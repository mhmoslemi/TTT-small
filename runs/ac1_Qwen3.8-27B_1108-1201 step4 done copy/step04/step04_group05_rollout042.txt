"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using hybrid multi-stage optimization:
- Nesterov-accelerated subgradient descent on the simplex (fast, many iterations)
- Multiplicative weights (mirror descent) for diverse exploration
- LP-based Frank-Wolfe for escape from local minima
- Coordinate descent for final polishing
- Multi-length exploration (300-450)
Key improvement: correct and efficient subgradient computation enables 10x more
iterations than LP-based approach, with periodic LP steps for escape.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgrad(a, conv, top_k=20, temp=0.0005):
    """Compute weighted subgradient of max(conv(a,a)) w.r.t. a."""
    n = len(a)
    num_peaks = min(top_k, len(conv))
    peaks = np.argpartition(conv, -(num_peaks - 1))[-num_peaks:]
    
    pv = conv[peaks]
    scaled = pv / max(temp, 1e-10)
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            sg[m_lo:m_hi + 1] += 2.0 * w[i] * a[k - m_hi:k - m_lo + 1][::-1]
    
    return sg


def sg_step(a, conv, lr, momentum, v, top_k, temp):
    """One step of Nesterov-accelerated subgradient descent on simplex."""
    sg = compute_subgrad(a, conv, top_k=top_k, temp=temp)
    v = momentum * v - lr * sg
    a_new = project_simplex(a + v)
    return a_new, v


def nesterov_sg(a0, time_limit, lr=0.001, momentum=0.9, top_k=20, temp=0.0005):
    """Nesterov-accelerated subgradient descent on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        a, v = sg_step(a, conv, cur_lr, momentum, v, top_k, cur_temp)
        conv = fast_convolve(a)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive learning rate
        cur_lr = max(lr * 0.9998, 0.0001)
        cur_temp = min(temp * 1.003, 0.01)
        
        # Restart if stuck
        if no_improve > 500:
            noise = np.random.default_rng(it).standard_normal(n) * 0.005
            a = project_simplex(best_a + noise)
            v = np.zeros(n)
            cur_lr = lr
            cur_temp = temp
            no_improve = 0
            conv = fast_convolve(a)
    
    return best_a, best_obj


def multiplicative_weights(a0, time_limit, lr=0.1, top_k=10, temp=0.001):
    """Mirror descent with entropy (multiplicative weights) on simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        sg = compute_subgrad(a, conv, top_k=top_k, temp=temp)
        
        # Multiplicative weights update
        log_update = -cur_lr * sg
        log_update = np.clip(log_update, -50, 50)
        a_new = a * np.exp(log_update)
        sn = a_new.sum()
        if sn < 1e-300:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / sn
        
        a = a_new
        conv = fast_convolve(a)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.999
        if cur_lr < 0.01:
            cur_lr = 0.01
        
        if no_improve > 300:
            break
    
    return best_a, best_obj


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def fw_lp_step(f, M, m_exp=1.0):
    """Single Frank-Wolfe step with LP oracle."""
    g = solve_lp(f, M * m_exp)
    if g is None:
        return None
    g_sum = g.sum()
    if g_sum < 1e-15:
        return None
    g = g / g_sum
    return g

    """

    best_a = a.copy()
    
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = 0.002
    cur_temp = 0.0005
    m_exp = 1.0
    sg_count = 0
    rng = np.random.default_rng(12345 + n)
    
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        # Decide: SG step or LP-FW step
        use_lp = (no_improve > 100 and no_improve % 50 == 0) or (sg_count > 0 and sg_count % 500 == 0 and no_improve > 50)
        
        if use_lp:
            # LP-based FW step
            g = fw_lp_step(a, best_obj, m_exp)
            if g is not None:
                # Line search
                best_alpha = 0.0
                best_val = best_obj
                for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:

                        no_improve = 0

                    v = np.zeros(n)

                    no_improve = 0

    while time.monotonic() - t0 < time_limit and improved_total:
        improved_total = False
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - t0 > time_limit - 1:
                break
            if a[j] < 1e-10:
                continue

            for source in [(j + 1) % n, (j - 1) % n, (j + n // 2) % n]:
                if source == j or a[source] < 1e-10:

                    a_new[j] += delta

                    if a_new[source] < -1e-15:

                    a_new[source] = max(0, a_new[source])
                    a_new[j] = a_new[source] + delta if a_new[source] == 0 and a_new[source] + delta != a_new[j] else a_new[j]

                        best_obj = val
                        best_a = a.copy()
                        improved_total = True
                        break
                if improved_total:
                    break
            if improved_total:
                break
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-stage hybrid approach:
1. FFT-based smooth gradient (cross-correlation) for fast true gradients
2. Nesterov-accelerated projected gradient descent on the simplex
3. Periodic LP-based Frank-Wolfe steps for escape from local minima
4. Multiplicative weights (mirror descent) for diverse exploration
5. Coordinate descent for final polishing
6. Multi-length exploration around 384
Key innovation: O(N log N) gradient via FFT cross-correlation enables ~50x more
iterations than the LP-based approach, with smooth objective for better convergence.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_gradient(a, conv, tau, N):
    """
    Compute gradient of tau*log(sum_k exp(conv[k]/tau)) w.r.t. a.
    Uses FFT cross-correlation for O(N log N) cost.
    """
    shifted = conv / tau
    shifted -= shifted.max()
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()

    W = np.fft.fft(w, N)
    A_fft = np.fft.fft(a, N)
    c = np.fft.ifft(W * np.conj(A_fft)).real[:len(a)]
    grad = (2.0 / tau) * c
    return grad


def subgradient_topk(a, conv, top_k=30, temp=0.001):
    """Compute (approximate) subgradient using top-k peaks."""
    n = len(a)
    num_peaks = min(top_k, len(conv))
    peaks = np.argpartition(conv, -(num_peaks - 1))[-num_peaks:]

    pv = conv[peaks]
    scaled = pv / max(temp, 1e-10)
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            sg[m_lo:m_hi + 1] += 2.0 * w[i] * a[k - m_hi:k - m_lo + 1][::-1]

    return sg


def nesterov_smooth(a0, time_limit, tau0=0.002, lr0=0.0, verbose=False):
    """
    Nesterov-accelerated projected gradient descent on softmax-smoothed objective.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    N = 1 << (3 * n).bit_length()

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    tau = tau0
    # Estimate Lipschitz constant: ||grad||_inf is bounded by 2/tau * max(conv)
    L = 4.0 * conv.max() / tau
    if lr0 > 0:
        lr = lr0
    else:
        lr = 1.0 / L

    y = a.copy()  # Nesterov point
    v_nest = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(42 + n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break

        # Compute gradient at y
        conv_y = fast_convolve(y)
        grad = smooth_gradient(y, conv_y, tau, N)

        # Gradient step
        z = project_simplex(y - lr * grad)

        # Nesterov update
        beta = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        beta = min(beta, 1.5)  # Cap for stability
        v_nest = (it / (it + beta)) * (z - y)
        y = z + v_nest
        y = project_simplex(y)

        # Evaluate
        conv = fast_convolve(z)
        obj = conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt tau (anneal slowly)
        if it % 200 == 0:
            tau = max(tau * 0.97, 0.0001)
            L = 4.0 * best_obj / tau
            lr = min(lr * 1.02, 1.0 / L)

        # Adaptive learning rate
        if no_improve % 100 == 50:
            lr *= 0.8

        # Restart
        if no_improve > 800:
            noise = rng.standard_normal(n) * (0.003 + 0.002 * (no_improve // 1000))
            y = project_simplex(best_a + noise)
            z = y.copy()
            v_nest = np.zeros(n)
            it = 0
            tau = max(tau0 * 0.6, 0.0002)
            lr = 1.0 / (4.0 * best_obj / tau)
            no_improve = 0
            conv = fast_convolve(z)
            if verbose:
                print(f"    NSG restart at iter {it}, obj={best_obj:.10f}")

    return best_a, best_obj

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9995
        if cur_lr < 0.005:
            cur_lr = 0.005

        if no_improve > 400:
            break

    return best_a, best_obj


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def fw_lp_step_single(f, M, m_exp=1.0):
    """Single FW step: solve LP and do line search."""
    g = solve_lp(f, M * m_exp)
    if g is None:
        return None, None, m_exp * 1.1
    g_sum = g.sum()
    if g_sum < 1e-15:
        return None, None, m_exp
    g = g / g_sum

    # Line search
    best_alpha = 0.0
    best_val = fast_convolve(f).max()
    for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
        ft = (1 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0)
        st = ft.sum()
        if st < 1e-15:
            continue
        ft = ft / st
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha

    if best_alpha < 0.01:
        return None, g, m_exp
    return g, g, m_exp


def hybrid_deep_optimize(a0, time_limit, verbose=False):
    """
    Deep hybrid optimizer: Nesterov smooth gradient as primary,
    LP-FW for escape, occasional MW for diversification.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    N = 1 << (3 * n).bit_length()

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    tau = 0.002
    lr = 1.0 / (4.0 * best_obj / tau)
    y = a.copy()
    v_nest = np.zeros(n)
    it_ns = 0

    t0 = time.monotonic()
    no_improve = 0
    m_exp = 1.0
    rng = np.random.default_rng(777 + n)
    phase = 0  # 0=NSG, 1=LP-FW, 2=MW

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        # Decide phase
        if no_improve > 300 and no_improve % 100 == 50:
            phase = 1  # LP-FW
        elif no_improve > 150 and no_improve % 100 == 25:
            phase = 2  # MW
        else:
            phase = 0  # NSG

        if phase == 0:
            # Nesterov smooth gradient step
            conv_y = fast_convolve(y)
            grad = smooth_gradient(y, conv_y, tau, N)
            z = project_simplex(y - lr * grad)
            beta = (1.0 + np.sqrt(1.0 + 4.0 * it_ns * it_ns)) / 2.0
            beta = min(beta, 1.5)
            v_nest = (it_ns / max(1, it_ns + beta)) * (z - y)
            y = project_simplex(z + v_nest)
            it_ns += 1

            a = z
            conv = fast_convolve(a)
            obj = conv.max()

            if obj < best_obj - 1e-15:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            # Adapt
            if it_ns % 300 == 0:
                tau = max(tau * 0.96, 0.0001)
                lr = min(lr * 1.01, 1.0 / (4.0 * best_obj / tau + 1e-10))

        elif phase == 1:
            # LP-FW step
            g = solve_lp(a, best_obj * m_exp)
            if g is not None and g.sum() > 1e-15:
                g = g / g.sum()
                best_alpha = 0.0
                best_val = best_obj
                for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                    ft = (1 - alpha) * a + alpha * g
                    ft = np.maximum(ft, 0)
                    st = ft.sum()
                    if st < 1e-15:
                        continue
                    ft = ft / st
                    val = fast_convolve(ft).max()
                    if val < best_val:
                        best_val = val
                        best_alpha = alpha
                if best_alpha > 0.01:
                    a = (1 - best_alpha) * a + best_alpha * g
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    conv = fast_convolve(a)
                    obj = conv.max()
                    if obj < best_obj - 1e-15:
                        best_obj = obj
                        best_a = a.copy()
                        no_improve = 0
                        m_exp = 1.0
                    else:
                        no_improve += 1
                    y = a.copy()
                    v_nest = np.zeros(n)
                    it_ns = 0
                else:
                    no_improve += 1
                    m_exp = min(m_exp * 1.15, 4.0)
            else:
                m_exp = min(m_exp * 1.1, 5.0)
                no_improve += 1

        else:
            # MW step (do a batch of 20)
            for _ in range(20):
                if time.monotonic() - t0 > time_limit - 3:
                    break
                sg = subgradient_topk(a, conv, top_k=15, temp=0.001)
                log_update = -0.05 * sg
                log_update = np.clip(log_update, -30, 30)
                a_new = a * np.exp(log_update)
                sn = a_new.sum()
                if sn > 1e-300:
                    a = a_new / sn
                conv = fast_convolve(a)
                obj = conv.max()
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            y = a.copy()
            v_nest = np.zeros(n)
            it_ns = 0

        # Big restart
        if no_improve > 500:
            scale = 0.003 + 0.003 * (no_improve // 500)
            noise = rng.standard_normal(n) * scale
            a = project_simplex(best_a + noise)
            y = a.copy()
            v_nest = np.zeros(n)
            it_ns = 0
            tau = 0.002
            lr = 1.0 / (4.0 * best_obj / tau)
            conv = fast_convolve(a)
            no_improve = 0
            m_exp = 1.0
            if verbose:
                print(f"    Hybrid restart, obj={best_obj:.10f}")

    return best_a, best_obj


def coordinate_descent_polish(a0, time_limit):
    """Coordinate descent for fine polishing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(31337)
    total_improved = True

    while time.monotonic() - t0 < time_limit and total_improved:
        total_improved = False
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - t0 > time_limit - 2:
                break
            if a[j] < 1e-12:
                continue
            # Try several sources
            sources = [(j + 1) % n, (j - 1) % n, (j + n // 3) % n,
                       (j + 2 * n // 3) % n]
            found = False
            for source in sources:
                if source == j or a[source] < 1e-12:
                    continue
                for frac in [0.2, 0.5, 0.8]:
                    delta = a[source] * frac
                    a_new = a.copy()
                    a_new[j] += delta
                    a_new[source] -= delta
                    a_new = np.maximum(a_new, 0.0)
                    sn = a_new.sum()
                    if sn < 1e-15:
                        continue
                    a_new = a_new / sn
                    val = fast_convolve(a_new).max()
                    if val < best_obj - 1e-16:
                        a = a_new
                        best_obj = val
                        best_a = a.copy()
                        total_improved = True
                        found = True
                        break
                if found:
                    break

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Concentrated Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    # Edge concentrated
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    # Two-peak
    peak = (np.exp(-((x - n / 3.0) ** 2) / (2.0 * (n / 12.0) ** 2)) +
            np.exp(-((x - 2.0 * n / 3.0) ** 2) / (2.0 * (n / 12.0) ** 2)))
    inits.append(peak / peak.sum())

    # Multi-peak
    num_peaks = max(4, n // 25)
    centers = rng.uniform(0, n, num_peaks)
    widths = rng.uniform(n / 40, n / 12, num_peaks)
    multi = np.zeros(n)
    for c, w in zip(centers, widths):
        multi += np.exp(-0.5 * ((x - c) / w) ** 2)
    inits.append(multi / multi.sum())

    # Sine-modulated
    for freq in [2, 3]:
        sin_seq = 0.5 + 0.4 * np.abs(np.sin(np.pi * freq * x / n))
        inits.append(sin_seq / sin_seq.sum())

    # Power law
    pl = (x + 1.0) ** (-0.4)
    inits.append(pl / pl.sum())

    # Parent interpolation
    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n),
                       np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
        # Blend with uniform
        for blend in [0.3, 0.5]:
            mixed = (1 - blend) * (px / ps) + blend * np.ones(n) / n
            inits.append(mixed / mixed.sum())

    return inits


def try_sequence(a, label, state):
    """Evaluate and update state."""
    a = np.maximum(a, 0.0)
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < state['best_score']:
        state['best_score'] = score
        state['best_seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return True
    return False


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-stage search:
    Phase 1: Quick multi-length exploration with NSG
    Phase 2: Deep hybrid optimization of best candidates
    Phase 3: Polishing (coordinate descent + fine NSG)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    state = {'best_seq': None, 'best_score': float('inf')}

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
            s = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(s):
                print(f"  Parent score: {s:.6f}")
                state['best_score'] = s
                state['best_seq'] = parent.copy()
    except Exception:
        pass

    # === Phase 1: Multi-length exploration ===
    print("=== Phase 1: Multi-length Exploration ===")
    phase1_end = time.monotonic() + float(budget_s) * 0.30

    lengths = [320, 360, 384, 400, 420]
    if parent is not None:
        pn = len(parent)
        if pn in lengths:
            lengths.remove(pn)
        lengths.insert(0, pn)

    for n in lengths:
        if time.monotonic() > phase1_end - 30:
            break

        inits = generate_inits(n, parent, rng)
        t_per = max(12, (phase1_end - time.monotonic() - 20) / max(1, len(inits) * max(1, len(lengths) - lengths.index(n))))
        t_per = min(t_per, 45)

        for i, f_init in enumerate(inits):
            if time.monotonic() > phase1_end - 15:
                break
            t_alloc = min(t_per, phase1_end - time.monotonic() - 10)
            if t_alloc < 8:
                break
            try:
                a_opt, obj = nesterov_smooth(f_init, time_limit=t_alloc, tau0=0.003)
                try_sequence(a_opt, f"P1_n{n}_i{i}", state)
            except Exception:
                pass

    print(f"  Phase 1 best: {state['best_score']:.6f}")

    # === Phase 2: Deep optimization ===
    print(f"\n=== Phase 2: Deep Optimization ===")
    phase2_end = time.monotonic() + float(budget_s) * 0.50

    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])
        print(f"  Optimizing n={n_best}")

        # Main deep run
        t_main = min(200, (phase2_end - time.monotonic()) * 0.4)
        if t_main > 30:
            try:
                a_opt, obj = hybrid_deep_optimize(state['best_seq'], time_limit=t_main, verbose=True)
                try_sequence(a_opt, "P2_deep_main", state)
            except Exception:
                pass

        # Perturbation runs
        perturb_scales = [0.003, 0.008, 0.015, 0.04]
        for ps in perturb_scales:
            rem = phase2_end - time.monotonic() - 20
            if rem < 20:
                break
            t_alloc = min(rem / 2.0, 100)
            if t_alloc < 12:
                break
            try:
                noise = rng.standard_normal(n_best) * ps
                f_init = project_simplex(state['best_seq'] + noise)
                a_opt, obj = nesterov_smooth(f_init, time_limit=t_alloc, tau0=0.002)
                try_sequence(a_opt, f"P2_pert_{ps}", state)
            except Exception:
                pass

        # Mirror blend
        rem = phase2_end - time.monotonic() - 15
        if rem > 25:
            try:
                f_mirror = project_simplex(0.7 * state['best_seq'] + 0.3 * state['best_seq'][::-1])
                a_opt, obj = nesterov_smooth(f_mirror, time_limit=min(rem, 80), tau0=0.002)
                try_sequence(a_opt, "P2_mirror", state)
            except Exception:
                pass

        # MW exploration
        rem = phase2_end - time.monotonic() - 10
        if rem > 20:
            try:
                a_opt, obj = multiplicative_weights_opt(state['best_seq'],
                                                        time_limit=min(rem, 60), lr=0.03)
                try_sequence(a_opt, "P2_mw", state)
            except Exception:
                pass

    # === Phase 3: Polishing ===
    print(f"\n=== Phase 3: Polishing ===")
    phase3_end = deadline

    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])
        print(f"  Current best: {state['best_score']:.6f}")

        # Fine NSG with small tau
        rem = phase3_end - time.monotonic() - 50
        if rem > 20:
            try:
                a_opt, obj = nesterov_smooth(state['best_seq'], time_limit=min(rem, 120), tau0=0.0005)
                try_sequence(a_opt, "P3_fine_nsg", state)
            except Exception:
                pass

        # Deep hybrid (second round)
        rem = phase3_end - time.monotonic() - 30
        if rem > 25:
            try:
                a_opt, obj = hybrid_deep_optimize(state['best_seq'], time_limit=min(rem, 100))
                try_sequence(a_opt, "P3_hybrid2", state)
            except Exception:
                pass

        # Coordinate descent
        rem = phase3_end - time.monotonic() - 10
        if rem > 15:
            try:
                a_opt, obj = coordinate_descent_polish(state['best_seq'], time_limit=min(rem, 80))
                try_sequence(a_opt, "P3_cd", state)
            except Exception:
                pass

        # Final NSG squeeze
        rem = phase3_end - time.monotonic()
        if rem > 10:
            try:
                a_opt, obj = nesterov_smooth(state['best_seq'], time_limit=min(rem, 50), tau0=0.0003)
                try_sequence(a_opt, "P3_final", state)
            except Exception:
                pass

    # Fallback
    if state['best_seq'] is None:
        state['best_seq'] = np.ones(384) / 384
        state['best_score'] = float(evaluate_sequence(state['best_seq'].tolist()))
        print(f"  Fallback uniform: {state['best_score']:.6f}")

    # Final verification
    fs = float(evaluate_sequence(state['best_seq'].tolist()))
    if np.isfinite(fs):
        state['best_score'] = fs

    print(f"\nFINAL SCORE: {state['best_score']:.6f} (n={len(state['best_seq'])})")
    return [float(x) for x in state['best_seq']]
