import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def compute_soft_gradient(a, conv, top_k=50, temp=0.01):
    """Compute softmax-weighted gradient of the peak autocorrelation."""
    n = len(a)
    # Use top-k peaks with softmax weighting
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum

    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += w[idx] * a[k - idxs]
    return grad


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def nesterov_optimize(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=50, temp_init=0.01, temp_decay=1.002, lr_decay=0.999):
    """Nesterov-accelerated gradient descent with adaptive parameters."""
    a = normalize(a0.astype(np.float64))
    n = len(a)
    v = np.zeros(n)
    y = a.copy()
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Nesterov lookahead
        y = a + mom * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        
        conv_y = fast_convolve(y)
        grad = compute_soft_gradient(y, conv_y, top_k=top_k, temp=temp)
        
        # Gradient step
        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive updates
        lr *= lr_decay
        temp *= temp_decay
        mom = min(0.97, mom + 0.0001)
        
        if no_improve >= 300:
            break
    
    return best_a, best_obj


def lp_solve(a, M, C=None):
    """Solve LP: max sum(g) s.t. C@g <= M, g >= 0."""
    n = len(a)
    if C is None:
        C = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True})
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g, g_sum, True
    except Exception:
        pass
    return a.copy(), 1.0, False


def lp_step(a, conv, n):
    """One LP-based step with line search."""
    M = conv.max()
    peak_idx = int(np.argmax(conv))
    C = build_conv_matrix(a)
    g, g_sum, success = lp_solve(a, M * 1.0001, C)
    
    if not success or g_sum <= 1.0 + 1e-8:
        return a.copy(), False
    
    d = g - a
    norm_d = np.linalg.norm(d)
    if norm_d < 1e-12:
        return a.copy(), False
    
    best_a = a.copy()
    best_obj = 2.0 * n * M
    
    for t in [0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]:
        a_try = a + t * d
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        conv_try = fast_convolve(a_try)
        obj_try = 2.0 * n * conv_try.max()
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            break
    
    if best_obj < 2.0 * n * M:
        return best_a, True
    return a.copy(), False


def lp_optimize(a0, time_limit, max_perturb=30):
    """LP-based iterative optimization."""
    a = normalize(a0.astype(np.float64))
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    no_improve = 0
    perturb_count = 0

            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj - 1e-12:

                best_a = a.copy()
            no_improve = 0

            no_improve += 1

            a = np.maximum(best_a + noise, 0.0)

            conv = fast_convolve(a)
            no_improve = 0
            perturb_count += 1

    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def normalize(a):
    """Project to non-negative simplex (sum=1)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_soft_gradient(a, conv, top_k=50, temp=0.01):
    """Compute softmax-weighted gradient of the peak autocorrelation."""
    n = len(a)
    if top_k >= len(conv):
        top_k = len(conv)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum

    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-10:
            continue
        k = int(peaks[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += w[idx] * a[k - idxs]
    return grad


def nesterov_optimize(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=50,
                      temp_init=0.01, temp_growth=1.003, lr_decay=0.999,
                      mom_growth=0.0001, no_improve_limit=400):
    """Nesterov-accelerated gradient descent with adaptive parameters."""
    a = normalize(a0.astype(np.float64))
    n = len(a)
    v = np.zeros(n)
    y = a.copy()

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 2:
            break

        # Nesterov lookahead
        y = a + mom * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        else:
            y = a.copy()

        conv_y = fast_convolve(y)
        grad = compute_soft_gradient(y, conv_y, top_k=top_k, temp=temp)

        # Gradient step
        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
            v = np.zeros(n)

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive updates
        lr *= lr_decay
        temp *= temp_growth
        mom = min(0.97, mom + mom_growth)

        if no_improve >= no_improve_limit:
            break

    return best_a, best_obj


def lp_solve(a, M_target, C=None):
    """Solve LP: max sum(g) s.t. C@g <= M_target, g >= 0."""
    n = len(a)
    if C is None:
        C = build_conv_matrix(a)
    c_vec = -np.ones(n)
    b_vec = np.full(2 * n - 1, M_target)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_vec, A_ub=C, b_ub=b_vec, bounds=bounds,
                         method='highs', options={'presolve': True,
                                                   'dual_feasibility_tolerance': 1e-9})
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g, g_sum, True
    except Exception:
        pass
    return a.copy(), 1.0, False


def lp_step(a, conv, n):
    """One LP-based step with geometric line search."""
    M = conv.max()
    C = build_conv_matrix(a)
    g, g_sum, success = lp_solve(a, M * 1.00005, C)

    if not success or g_sum <= 1.0 + 1e-7:
        return a.copy(), False

    d = g - a
    if np.linalg.norm(d) < 1e-12:
        return a.copy(), False

    best_a = a.copy()
    best_obj = 2.0 * n * M
    improved = False

    for t in [0.4, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005]:
        a_try = a + t * d
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        conv_try = fast_convolve(a_try)
        obj_try = 2.0 * n * conv_try.max()
        if obj_try < best_obj - 1e-13:
            best_obj = obj_try
            best_a = a_try.copy()
            improved = True
            break

    return best_a, improved


def lp_optimize(a0, time_limit, max_perturb=50, verbose=False):
    """LP-based iterative optimization with perturbative restarts."""
    a = normalize(a0.astype(np.float64))
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    perturb_count = 0
    lp_fail = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 10:
            break

        a_new, improved = lp_step(a, conv, n)

        if improved:
            a = a_new
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj - 1e-13:
                best_obj = cur_obj
                best_a = a.copy()
            no_improve = 0
            lp_fail = 0
        else:
            no_improve += 1
            lp_fail += 1

        if verbose and (no_improve + perturb_count * 50) % 100 == 0:
            print(f"    LP: obj={best_obj:.8f}, perturb={perturb_count}, t={time.monotonic()-t0:.0f}s")

        # Perturb on stagnation
        if no_improve >= 60:
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 999983)
            scale = 0.001 + 0.004 * min(perturb_count, 15)
            noise = rng.standard_normal(n) * scale
            a = np.maximum(best_a + noise, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            else:
                a = best_a.copy()
            conv = fast_convolve(a)
            no_improve = 0
            perturb_count += 1

            if lp_fail > 40:
                # Switch to gradient briefly
                a_g, obj_g = nesterov_optimize(best_a, time_limit=3.0,
                                               lr_init=0.006, top_k=60, temp_init=0.003)
                if obj_g < best_obj:
                    best_obj = obj_g
                    best_a = a_g.copy()
                    a = a_g.copy()
                    conv = fast_convolve(a)
                lp_fail = 0
                no_improve = 0

            if perturb_count > max_perturb:
                break

    return best_a, best_obj


def generate_hann(n):
    x = np.arange(n) / n
    return np.sin(np.pi * x) ** 2


def generate_flattop(n):
    x = np.arange(n) / n
    a1, a2, a3, a4, a5 = 0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368
    w = a1 - a2 * np.cos(2 * np.pi * x) + a3 * np.cos(4 * np.pi * x) \
        - a4 * np.cos(6 * np.pi * x) + a5 * np.cos(8 * np.pi * x)
    return np.abs(w)


def generate_fejer(n):
    half = n // 2
    c = np.arange(half + 1)
    w = 1.0 - c / (half + 1)
    if n % 2 == 0:
        w_full = np.zeros(n)
        w_full[:half] = w[:half][::-1]
        w_full[half:] = w[:half]
    else:
        w_full = np.zeros(n)
        w_full[:half] = w[:half][::-1]
        w_full[half] = w[half]
        w_full[half + 1:] = w[:half]
    return np.abs(w_full)


def generate_blackman(n):
    if n <= 1:
        return np.ones(n)
    x = np.arange(n) / (n - 1)
    return np.abs(0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))


def generate_blackman_harris(n):
    if n <= 1:
        return np.ones(n)
    x = np.arange(n) / (n - 1)
    a = [0.35875, 0.48829, 0.14128, 0.01168]
    w = np.zeros(n)
    for k, ak in enumerate(a):
        w += ak * np.cos(2 * np.pi * k * x)
    return np.abs(w)


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    inits['uniform'] = np.ones(n) / n
    inits['hann'] = normalize(generate_hann(n))
    inits['flattop'] = normalize(generate_flattop(n))
    inits['fejer'] = normalize(generate_fejer(n))
    inits['blackman'] = normalize(generate_blackman(n))
    inits['blackman_harris'] = normalize(generate_blackman_harris(n))
    inits['random1'] = normalize(rng.dirichlet(np.ones(n) * 0.3))
    inits['random2'] = normalize(rng.dirichlet(np.ones(n) * 1.5))

    # Bimodal
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Quasi-bimodal (Gaussian bumps at 1/4 and 3/4)
    x = np.arange(n) / n
    bump1 = np.exp(-50 * (x - 0.25) ** 2)
    bump2 = np.exp(-50 * (x - 0.75) ** 2)
    inits['quasi_bi'] = normalize(bump1 + bump2)

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps
                # Also try parent with noise
                for ns in [0.01, 0.03]:
                    pn = np.maximum(pp + rng.standard_normal(n) * ns * (pps / n), 0)
                    pns = pn.sum()
                    if pns > 1e-10:
                        inits[f'parent_noise{ns}'] = pn / pns

    return inits


def coordinate_refine(a0, time_limit, max_rounds=50000):
    """Local coordinate refinement: try small perturbations to reduce peak."""
    a = normalize(a0.astype(np.float64).copy())
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    peak_idx = int(np.argmax(conv))

    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 500:
        # Perturb the two largest contributors to the peak
        k = peak_idx
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        contributors = np.arange(j_start, j_end + 1)
        contrib_vals = a[k - contributors]
        top2 = np.argsort(contrib_vals)[-2:]

        i1 = int(contributors[top2[0]])
        i2 = int(contributors[top2[1]])

        # Try reducing these and adding to random positions
        for delta in [0.002, 0.005, 0.01, 0.02]:
            j_target = rng.integers(0, n)
            if j_target == i1 or j_target == i2:
                continue
            a_try = a.copy()
            a_try[i1] -= delta * a[i1]
            a_try[i2] -= delta * a[i2]
            a_try[j_target] += delta * (a[i1] + a[i2])
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-15:
                continue
            a_try = a_try / s
            conv_try = fast_convolve(a_try)
            obj_try = 2.0 * n * conv_try.max()
            if obj_try < best_obj - 1e-12:
                a = a_try
                conv = conv_try
                best_obj = obj_try
                best_a = a.copy()
                peak_idx = int(np.argmax(conv))
                no_improve = 0
                break
        else:
            no_improve += 1

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase hybrid optimization:
    1. Quick Nesterov scan across diverse initializations
    2. Extended Nesterov on top candidates
    3. LP-based deep refinement
    4. Coordinate fine-tuning and multi-length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass

    # =========================================================
    # Phase 1: Quick Nesterov scan (200s)
    # =========================================================
    print("=== Phase 1: Quick Scan ===")
    t_start = time.monotonic()

    target_n = 768
    inits = generate_initializations(target_n, parent, rng)

    # Priority ordering
    priority = ['parent', 'parent_interp', 'parent_noise0.01', 'parent_noise0.03',
                'uniform', 'hann', 'flattop', 'fejer', 'blackman_harris',
                'random1', 'quasi_bi', 'bimodal', 'blackman']
    init_names = [nm for nm in priority if nm in inits]
    init_names += [nm for nm in inits if nm not in priority]
    init_names = init_names[:10]

    t_per_scan = 15.0
    scan_results = []

    for name in init_names:
        if time.monotonic() > deadline - 600:
            break
        t_alloc = min(t_per_scan, deadline - time.monotonic() - 400)
        if t_alloc < 5:
            break

        a0 = inits[name]
        a_opt, obj = nesterov_optimize(a0, time_limit=t_alloc,
                                        lr_init=0.012, momentum=0.91,
                                        top_k=50, temp_init=0.008)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            scan_results.append((score, name, a_opt.copy()))
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
            print(f"  {name}: obj={obj:.6f}, score={score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 2: Extended Nesterov on top-3 (500s)
    # =========================================================
    print("=== Phase 2: Extended Nesterov ===")
    t_start2 = time.monotonic()

    top3 = scan_results[:3]
    t_per_ext = 150.0

    for i, (score, name, a0) in enumerate(top3):
        if time.monotonic() > deadline - 400:
            break
        t_alloc = min(t_per_ext, deadline - time.monotonic() - 300)
        if t_alloc < 20:
            break

        # Multiple Nesterov configs
        configs = [
            {'lr_init': 0.01, 'momentum': 0.90, 'top_k': 60, 'temp_init': 0.005,
             'temp_growth': 1.004, 'lr_decay': 0.999},
            {'lr_init': 0.006, 'momentum': 0.93, 'top_k': 40, 'temp_init': 0.003,
             'temp_growth': 1.003, 'lr_decay': 0.9992},
            {'lr_init': 0.015, 'momentum': 0.88, 'top_k': 80, 'temp_init': 0.01,
             'temp_growth': 1.005, 'lr_decay': 0.9988},
        ]

        cfg = configs[i % len(configs)]
        a_opt, obj = nesterov_optimize(a0, time_limit=t_alloc, **cfg)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Ext-{name}: {score:.6f} *NEW BEST*")
            else:
                print(f"  Ext-{name}: {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 3: LP Deep Optimization (600s)
    # =========================================================
    print("=== Phase 3: LP Deep Optimization ===")
    t_start3 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 200
        if remaining > 50:
            # Run LP on the best
            t_lp = min(remaining * 0.7, 400)
            print(f"  LP main: t={t_lp:.0f}s")
            a_opt, obj = lp_optimize(best_sequence, time_limit=t_lp, max_perturb=50,
                                     verbose=True)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  LP main: {score:.6f} *NEW BEST*")
                else:
                    print(f"  LP main: {score:.6f}")

            # Second LP run with perturbed start
            remaining = deadline - time.monotonic() - 100
            if remaining > 30:
                scale = 0.003
                noise = rng.standard_normal(len(best_sequence)) * scale
                a_pert = np.maximum(best_sequence + noise, 0)
                a_pert = a_pert / a_pert.sum()
                t_lp2 = min(remaining * 0.6, 100)
                print(f"  LP perturbed: t={t_lp2:.0f}s")
                a_opt, obj = lp_optimize(a_pert, time_limit=t_lp2, max_perturb=30)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score):
                    if score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  LP perturbed: {score:.6f} *NEW BEST*")
                    else:
                        print(f"  LP perturbed: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 4: Fine-tuning (200s)
    # =========================================================
    print("=== Phase 4: Fine-tuning ===")
    t_start4 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 50
        n_best = len(best_sequence)

        # Coordinate refinement
        t_cd = min(remaining * 0.3, 60)
        if t_cd > 10:
            a_opt, obj = coordinate_refine(best_sequence, time_limit=t_cd)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  CoordRefine: {score:.6f} *NEW BEST*")

        # Short Nesterov with different configs
        remaining = deadline - time.monotonic() - 30
        if remaining > 20:
            fine_configs = [
                {'lr_init': 0.003, 'momentum': 0.95, 'top_k': 100, 'temp_init': 0.002,
                 'temp_growth': 1.002, 'lr_decay': 0.9995},
                {'lr_init': 0.008, 'momentum': 0.90, 'top_k': 30, 'temp_init': 0.004,
                 'temp_growth': 1.003, 'lr_decay': 0.999},
                {'lr_init': 0.005, 'momentum': 0.92, 'top_k': 70, 'temp_init': 0.001,
                 'temp_growth': 1.002, 'lr_decay': 0.9993},
            ]
            for gi, cfg in enumerate(fine_configs):
                if time.monotonic() > deadline - 20:
                    break
                t_sg = min(remaining / (len(fine_configs) - gi) * 0.8, 40)
                if t_sg < 8:
                    break
                a_opt, obj = nesterov_optimize(best_sequence, time_limit=t_sg, **cfg)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Fine-{gi}: {score:.6f} *NEW BEST*")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 5: Try different lengths (100s)
    # =========================================================
    print("=== Phase 5: Multi-Length ===")
    t_start5 = time.monotonic()

    if best_sequence is not None and time.monotonic() < deadline - 40:
        # Try extending to a longer sequence
        for n_try in [1024, 1000]:
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - time.monotonic() - 25
            if remaining < 20:
                break

            # Interpolate to new length
            a_ext = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, len(best_sequence)),
                              best_sequence / best_sequence.sum())
            a_ext = np.maximum(a_ext, 0)
            a_ext = a_ext / a_ext.sum()

            t_trial = min(remaining * 0.6, 40)
            a_opt, obj = nesterov_optimize(a_ext, time_limit=t_trial,
                                            lr_init=0.008, top_k=50, temp_init=0.005)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  n={n_try}: {score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n_try}: {score:.6f} *NEW BEST*")

        # Try shorter (sometimes shorter is better due to the 2n factor)
        for n_try in [512, 640]:
            if time.monotonic() > deadline - 25:
                break
            if len(best_sequence) <= n_try:
                continue
            remaining = deadline - time.monotonic() - 20
            if remaining < 15:
                break

            a_shr = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, len(best_sequence)),
                              best_sequence / best_sequence.sum())
            a_shr = np.maximum(a_shr, 0)
            a_shr = a_shr / a_shr.sum()

            t_trial = min(remaining * 0.5, 25)
            a_opt, obj = nesterov_optimize(a_shr, time_limit=t_trial,
                                            lr_init=0.008, top_k=40, temp_init=0.005)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  n={n_try}: {score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n_try}: {score:.6f} *NEW BEST*")

    print(f"Phase 5 done ({time.monotonic()-t_start5:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Final verification
    # =========================================================
    if best_sequence is None:
        best_sequence = np.ones(768) / 768

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
