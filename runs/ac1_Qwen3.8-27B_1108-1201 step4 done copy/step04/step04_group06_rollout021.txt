1. FFT-based gradient descent with Adam optimizer

import numpy as np
import time
from scipy.optimize import linprog

def next_power_of_2(n):
    return 1 << (n - 1).bit_length() if n > 1 else 1

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = next_power_of_2(size)
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_full_gradient(a, temperature=0.01, var_weight=0.1):
    """
    Compute gradient of J(a) = softmax_smoothed_max(conv(a,a)) + var_weight * sum(conv(a,a)^2)
    assuming sum(a) = 1.
    Returns (gradient, conv, max_val).
    """
    n = len(a)
    conv = fast_convolve(a)
    max_val = conv.max()
    
    # FFT size for cross-correlation: need >= (2n-1) + n - 1 = 3n-2
    L = next_power_of_2(3 * n - 2)
    
    # Gradient of smoothed max
    scaled = conv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    w = w / w_sum
    
    # h[i] = sum_j w[i+j] * a[j] (cross-correlation of w and a)
    W = np.fft.fft(w, L)
    A = np.fft.fft(a, L)
    h = np.fft.ifft(W * np.conj(A)).real[:n]
    grad_max = 2.0 * h
    
    # Gradient of variance: 4 * sum_k conv[k] * a[k-i]
    CP = np.fft.fft(conv, L)
    h_var = np.fft.ifft(CP * np.conj(A)).real[:n]
    grad_var = 4.0 * h_var
    
    grad = grad_max + var_weight * grad_var
    return grad, conv, max_val

def adam_update(a, grad, state, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer step with non-negativity and sum=1 projection."""
    m, v, t = state
    t = t + 1
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad ** 2
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    # Project to non-negative
    a_new = np.maximum(a_new, 0.0)
    # Normalize to sum = 1
    s = a_new.sum()
    if s < 1e-15:
        a_new = np.ones(len(a)) / len(a)
    else:
        a_new = a_new / s
    return a_new, (m, v, t)

                      var_weight_init=0.1, var_weight_final=0.01,

    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based gradient descent with Adam optimizer and mixed objective (smoothed max + variance)
2. Multi-scale initialization with diverse structured starts
3. LP-based direction finding for polishing

import numpy as np
import time
from scipy.optimize import linprog


def next_power_of_2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = next_power_of_2(size)
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_gradient_mixed(a, temperature, var_weight):
    """Compute gradient of smoothed_max(conv(a,a)) + var_weight * sum(conv(a,a)^2)."""
    n = len(a)
    conv = fast_convolve(a)
    max_val = conv.max()

    # FFT size >= (2n-1) + n - 1 = 3n-2 for cross-correlation
    L = next_power_of_2(3 * n - 2)

    # Smoothed max gradient: grad[i] = 2 * sum_k w[k] * a[k-i] for valid k-i
    scaled = conv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w = w / w_sum

    W = np.fft.fft(w, L)
    A = np.fft.fft(a, L)
    # Cross-correlation: h[i] = sum_j w[i+j] * a[j]
    h = np.fft.ifft(W * np.conj(A)).real[:n]
    grad_max = 2.0 * h

    # Variance gradient: 4 * sum_k conv[k] * a[k-i]
    CP = np.fft.fft(conv, L)
    h_var = np.fft.ifft(CP * np.conj(A)).real[:n]
    grad_var = 4.0 * h_var

    grad = grad_max + var_weight * grad_var
    return grad, conv, max_val


def adam_step(a, grad, state, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    m, v, t = state
    t_new = t + 1
    m_new = beta1 * m + (1.0 - beta1) * grad
    v_new = beta2 * v + (1.0 - beta2) * (grad * grad)
    m_hat = m_new / (1.0 - beta1 ** t_new)
    v_hat = v_new / (1.0 - beta2 ** t_new)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        a_new = np.ones(len(a)) / len(a)
    else:
        a_new = a_new / s
    return a_new, (m_new, v_new, t_new)


def gradient_optimize(a0, time_limit, lr_init=0.01, temp_init=0.01, temp_final=0.002,

    t = 0
    state = (m, v, t)

def build_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C

    max_no_improve = 60
    perturb_count = 0

        c_obj = -np.ones(n)

        bounds = [(0.0, None)] * n

        g = None
        try:
            result = linprog(c_obj, A_ub=C_matrix, b_ub=b_ub, bounds=bounds,
                             method='highs',
                             options={'presolve': True, 'time_limit': time_limit - (time.monotonic() - t0)})
            if result.success and result.x is not None:
                g = result.x
        except Exception:
            pass

        improved = False
        if g is not None and g.sum() > a.sum() * (1.0 + 1e-8):
            d = g - a
            norm_d = np.linalg.norm(d)
            if norm_d > 1e-12:
                # Try step sizes
                for t_val in [0.5, 0.3, 0.2, 0.1, 0.05]:
                    a_try = a + t_val * d
                    a_try = np.maximum(a_try, 0.0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        continue
                    a_try = a_try / s_try
                    conv_try = fast_convolve(a_try)
                    obj_try = 2.0 * n * conv_try.max()
                    if obj_try < best_obj - 1e-12:
                        a = a_try
                        best_obj = obj_try
                        best_a = a_try.copy()
                        M = conv_try.max()
                        C_matrix = build_conv_matrix(a)
                        conv = conv_try
                        improved = True
                        break

        if improved:
            no_improve = 0
        else:
            no_improve += 1

        if verbose and no_improve % 20 == 0:
            print(f"    LP: obj={best_obj:.8f}, no_improve={no_improve}, t={time.monotonic()-t0:.1f}s")

        if no_improve >= max_no_improve:
            scale = 0.002 * (1 + perturb_count * 0.5)
            noise = rng.standard_normal(n) * scale
            a = np.maximum(best_a + noise, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            else:
                a = best_a.copy()
            conv = fast_convolve(a)
            M = conv.max()
            C_matrix = build_conv_matrix(a)
            no_improve = 0
            perturb_count += 1
            max_no_improve = min(200, 60 + 20 * perturb_count)
            if perturb_count > 30:
                break

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['rand_dirichlet'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['rand_dirichlet2'] = rng.dirichlet(np.ones(n) * 0.1)

    # Gaussian centered
    sigma = 0.25 * n
    g = np.exp(-0.5 * ((x - n / 2) / sigma) ** 2)
    inits['gaussian'] = g / g.sum()

    # V-shape (mass at edges)
    v = 1.0 + 2.0 * np.abs(x / n - 0.5)
    inits['vshape'] = v / v.sum()

    # Inverted V (mass at center)
    iv = 1.0 + 2.0 * (0.5 - np.abs(x / n - 0.5))
    inits['inv_v'] = iv / iv.sum()

    # Cosine modulated
    for freq in [1, 2, 3]:
        c = 1.0 + 0.5 * np.cos(2.0 * np.pi * freq * x / n)
        c = np.maximum(c, 0.01)
        inits[f'cos_{freq}'] = c / c.sum()

    # Alternating (two-tone)
    alt = np.ones(n)
    alt[::2] = 1.5
    alt[1::2] = 0.5
    inits['alternating'] = alt / alt.sum()

    # Exponential decay
    exp_d = np.exp(-0.5 * x / n)
    inits['exp_decay'] = exp_d / exp_d.sum()

    # Power law
    pw = (x + 1.0) ** 0.5
    inits['power'] = pw / pw.sum()

    # Bimodal (two bumps)
    b1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2)
    b2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    bi = b1 + b2
    inits['bimodal'] = bi / bi.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps

    return inits


def extend_sequence(a_short, n_target):
    """Extend via interpolation."""
    n_short = len(a_short)
    s = a_short.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    a_norm = a_short / s
    a_ext = np.interp(np.linspace(0, 1, n_target), np.linspace(0, 1, n_short), a_norm)
    a_ext = np.maximum(a_ext, 0)
    ses = a_ext.sum()
    if ses > 1e-15:
        a_ext = a_ext / ses
    return a_ext


def coordinate_descent(a0, time_limit, max_rounds=50000):
    """Random coordinate descent: swap mass between two positions."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * len(a) * fast_convolve(a).max()
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    no_improve = 0
    max_no_improve = 500

    while time.monotonic() - t0 < time_limit:
        for _ in range(min(max_rounds, max(1, 200000 // n))):
            if time.monotonic() - t0 > time_limit - 1.0:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            delta = 0.005 * a.sum() * rng.uniform(0.5, 2.0) * (1 if rng.random() > 0.5 else -1)
            if a[i] + delta < 0 or a[j] - delta < 0:
                continue
            a[i] += delta
            a[j] -= delta
            conv_try = fast_convolve(a)
            obj_try = 2.0 * n * conv_try.max()
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a.copy()
                conv = conv_try
                no_improve = 0
            else:
                a[i] -= delta
                a[j] += delta
                no_improve += 1
        if no_improve > max_no_improve:
            break

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Hybrid gradient + LP optimization for minimizing
    2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass

    def update_best(a, label=""):
        nonlocal best_sequence, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        if a.size == 0:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  {label}: score={score:.6f} (n={len(a)}) *NEW BEST*")
        elif np.isfinite(score):
            print(f"  {label}: score={score:.6f} (n={len(a)})")

    # =========================================================
    # Phase 1: Quick gradient scan (250s)
    # =========================================================
    print("=== Phase 1: Gradient Scan ===")
    t_start = time.monotonic()
    phase1_time = 250.0

    lengths = [384, 512, 768, 1024]
    scan_results = []

    for n in lengths:
        if time.monotonic() > deadline - 600:
            break
        inits = generate_inits(n, parent, rng)

        # Prioritize
        priority = ['parent', 'parent_interp', 'uniform', 'gaussian', 'vshape',
                    'cos_1', 'cos_2', 'rand_dirichlet', 'bimodal', 'alternating']
        init_names = [nm for nm in priority if nm in inits]
        init_names += [nm for nm in inits if nm not in priority]
        init_names = init_names[:5]

        t_per = max(15.0, (phase1_time - 20) / (len(lengths) * 5))

        for name in init_names:
            if time.monotonic() > deadline - 550:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 200)
            if t_alloc < 8:
                break

            a0 = inits[name]
            a_opt, obj = gradient_optimize(a0, time_limit=t_alloc,
                                           lr_init=0.015, temp_init=0.008,
                                           temp_final=0.001, var_init=0.15,
                                           var_final=0.02, patience=400)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Grad n={n}, {name}: {score:.6f} *NEW BEST*")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 2: Deep gradient optimization (500s)
    # =========================================================
    print("=== Phase 2: Deep Gradient Optimization ===")
    t_start2 = time.monotonic()

    # Select top candidates (diverse lengths)
    top_configs = []
    seen_lengths = set()
    for score, n, name, a_opt in scan_results:
        if n not in seen_lengths:
            top_configs.append((n, a_opt))
            seen_lengths.add(n)
        if len(top_configs) >= 3:
            break

    if best_sequence is not None:
        if len(best_sequence) not in seen_lengths:
            top_configs.append((len(best_sequence), best_sequence.copy()))

    # Also try extending best to larger lengths
    if best_sequence is not None:
        for ext_n in [1024, 1536]:
            if ext_n > len(best_sequence) and ext_n not in seen_lengths:
                ext_a = extend_sequence(best_sequence, ext_n)
                top_configs.append((ext_n, ext_a))
                seen_lengths.add(ext_n)

    remaining = deadline - time.monotonic() - 400.0
    n_cands = max(1, len(top_configs))
    t_per_deep = max(20.0, remaining * 0.8 / n_cands)

    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 350:
            break
        t_alloc = min(t_per_deep, deadline - time.monotonic() - 250)
        if t_alloc < 10:
            break

        print(f"  Deep grad candidate {i}: n={n}, t={t_alloc:.0f}s")
        a_opt, obj = gradient_optimize(a0, time_limit=t_alloc,
                                       lr_init=0.008, temp_init=0.005,
                                       temp_final=0.0005, var_init=0.1,
                                       var_final=0.01, temp_decay=0.9998,
                                       var_decay=0.9999, lr_decay=0.99997,
                                       patience=1000, restart_count=5,
                                       restart_scale=0.003)
        update_best(a_opt, f"Deep {i} n={n}")

    # Also do a fresh run at n=768 and n=1024 with best params
    if time.monotonic() < deadline - 300:
        for try_n in [768, 1024]:
            if time.monotonic() > deadline - 250:
                break
            a0 = best_sequence.copy() if best_sequence is not None else np.ones(try_n) / try_n
            if len(a0) != try_n:
                a0 = extend_sequence(a0, try_n)
            t_alloc = min(80.0, deadline - time.monotonic() - 200)
            if t_alloc > 10:
                a_opt, obj = gradient_optimize(a0, time_limit=t_alloc,
                                               lr_init=0.01, temp_init=0.004,
                                               temp_final=0.0003, var_init=0.08,
                                               var_final=0.005, temp_decay=0.99985,
                                               var_decay=0.99992, lr_decay=0.99996,
                                               patience=1500, restart_count=8,
                                               restart_scale=0.002)
                update_best(a_opt, f"Extra n={try_n}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 3: LP polishing (400s)
    # =========================================================
    print("=== Phase 3: LP Polishing ===")
    t_start3 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 200.0
        if remaining > 60:
            # Multiple LP runs with perturbations
            n_runs = min(4, max(1, int(remaining / 80)))
            t_per_lp = (remaining * 0.9) / n_runs

            for pi in range(n_runs):
                if time.monotonic() > deadline - 150:
                    break
                t_alloc = min(t_per_lp, deadline - time.monotonic() - 100)
                if t_alloc < 15:
                    break

                if pi == 0:
                    a0 = best_sequence.copy()
                else:
                    scale = 0.003 * pi
                    noise = rng.standard_normal(len(best_sequence)) * scale
                    a0 = np.maximum(best_sequence + noise, 0)
                    s = a0.sum()
                    if s > 1e-15:
                        a0 = a0 / s
                    else:
                        a0 = best_sequence.copy()

                print(f"  LP polish {pi}: n={len(a0)}, t={t_alloc:.0f}s")
                a_opt, obj = lp_polish(a0, time_limit=t_alloc, verbose=True)
                update_best(a_opt, f"LP {pi}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 4: Fine-tuning (150s)
    # =========================================================
    print("=== Phase 4: Fine-tuning ===")
    t_start4 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 30.0

        # Coordinate descent
        if remaining > 30:
            t_cd = min(remaining * 0.3, 40.0)
            if t_cd > 5:
                a_opt, obj = coordinate_descent(best_sequence, time_limit=t_cd)
                update_best(a_opt, "CoordDesc")

        # Short gradient runs with different configs
        remaining = deadline - time.monotonic() - 20.0
        if remaining > 15:
            configs = [
                {'lr_init': 0.003, 'temp_init': 0.002, 'temp_final': 0.0002,
                 'var_init': 0.05, 'var_final': 0.002, 'patience': 600},
                {'lr_init': 0.005, 'temp_init': 0.003, 'temp_final': 0.0003,
                 'var_init': 0.08, 'var_final': 0.005, 'patience': 500},
            ]
            n_cfg = min(2, max(1, int(remaining / 30)))
            t_per_cfg = remaining * 0.8 / max(1, n_cfg)

            for gi in range(n_cfg):
                if time.monotonic() > deadline - 10:
                    break
                t_sg = min(t_per_cfg, deadline - time.monotonic() - 10)
                if t_sg < 5:
                    break
                cfg = configs[gi % len(configs)]
                a_opt, obj = gradient_optimize(best_sequence, time_limit=t_sg, **cfg)
                update_best(a_opt, f"FineGrad {gi}")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 5: Final LP micro-polish
    # =========================================================
    if best_sequence is not None and time.monotonic() < deadline - 10:
        t_final = min(25.0, deadline - time.monotonic() - 5)
        if t_final > 5:
            print("=== Phase 5: Final Polish ===")
            a_opt, obj = lp_polish(best_sequence, time_limit=t_final)
            update_best(a_opt, "Final LP")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
