from scipy.optimize import minimize, linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax_fn(t):
    t = t - t.max()
    e = np.exp(t)
    s = e.sum()
    return e / s


def true_obj(a):
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def smoothed_obj_grad(t, tau):
    """Objective and gradient of smoothed objective w.r.t. t (softmax parameterization)."""
    a = softmax_fn(t)
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()

    x = np.clip(x, -100, 100)
    ex = np.exp(x)
    w = ex / ex.sum()
    
    lse = max_c + tau * np.log(ex.sum())
    obj = 2.0 * n * lse
    
    # Gradient computation
    N = 4 * n
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(W * np.conj(A), N)

    grad = 4.0 * n * a * (g_raw - dot_wc)

def solve_lp_oracle(f, M, time_limit_lp=10):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix: (f * g)[k] = sum_j f[max(0,k-n+1):min(n,k+1)] ... 
    # Actually (f*g)[k] = sum_j f[j] * g[k-j] for valid j, k-j
    # For fixed k: (f*g)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} f[j] * g[k-j]
    # This is a banded matrix.
    
    # For efficiency, build the matrix as a banded matrix
    # A_ub has shape (2n-1, n)
    # A_ub[k, j] = f[j] if k-j is in [0, n-1], i.e., max(0, k-n+1) <= j <= min(n-1, k)
    # Equivalently: A_ub[k, j] = f[j] if 0 <= k-j <= n-1, i.e., k-n+1 <= j <= k and 0 <= j <= n-1
    
    # Actually, let's use the same approach as before: for each j, f*g has f[j] * g shifted
    # (f * g)[k] includes term f[j] * g[k-j] when k-j is in [0, n-1], i.e., j is in [k-n+1, k]
    # So A_ub[k, j] = f[j] if max(0, k-n+1) <= j <= min(n-1, k)
    
    # For the LP, we can use the banded structure. But for simplicity with n=384,
    # let's just build the full matrix (767 x 384 = ~295K entries).
    
    A_ub = np.zeros((2*n-1, n))
    for j in range(n):
        # f[j] * g contributes to (f*g)[k] for k = j, j+1, ..., j+n-1
        # i.e., A_ub[k, j] = f[j] for k in [j, j+n-1]
        A_ub[j:j+n, j] = f
    
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit_lp})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def fw_lp_step(f, M, time_per_lp=5):
    """One Frank-Wolfe step: solve LP oracle and line search."""
    n = len(f)
    g = solve_lp_oracle(f, M, time_limit_lp=time_per_lp)
    if g is None:
        return f, M, False
    
    gs = g.sum()
    if gs < 1e-15:
        return f, M, False
    g = g / gs
    
    # Line search
    best_alpha = 0.0
    best_val = M
    for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        ft = (1 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0)
        s = ft.sum()
        if s < 1e-15:
            continue
        ft = ft / s
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
    
    if best_alpha > 0:
        f_new = (1 - best_alpha) * f + best_alpha * g
        f_new = np.maximum(f_new, 0)
        f_new = f_new / f_new.sum()
        return f_new, best_val, True
    return f, M, False


def subgradient_step(f, conv, top_k=64, temp=0.01, lr=0.01):
    """One subgradient step with softmax weighting."""
    n = len(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= wsum
    
    # Compute subgradient using FFT
    N = 4 * n
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(f, N)
    corr = np.fft.irfft(W * np.conj(A), N)
    g_raw = corr[:n]
    
    # Projected gradient step on simplex
    f_new = f - lr * 2.0 * n * g_raw  # scaled subgradient
    # Project onto simplex
    f_new = project_simplex(f_new)
    return f_new


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
