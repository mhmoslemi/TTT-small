"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed gradient with Adam optimizer
2. Frank-Wolfe with LP oracle for global escapes
3. Adaptive multi-restart with diverse initializations
4. Final coordinate-descent-style polishing
5. Multi-length exploration (256, 384, 512)
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp(f, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def logsumexp_gradient(f, conv, beta):
    """Compute gradient of (1/beta)*log(sum exp(beta*conv_k)) w.r.t. f."""
    n = len(f)
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-30:
        # Fallback to hard max gradient
        peak = np.argmax(conv)
        w = np.zeros(len(conv))
        w[peak] = 1.0
    else:
        w /= wsum

    # Gradient of conv_k w.r.t. f[i] is f[k-i] for valid i
    # Sum over k with weights w[k]
    grad = np.zeros(n)
    for k in range(len(conv)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            i = k - j
            if i != j:
                grad[i] += 2.0 * w[k] * f[j]
            else:
                grad[i] += w[k] * f[j]
    return grad


def logsumexp_gradient_fast(f, conv, beta, top_k=200):
    """Faster gradient: only consider top-k peaks."""
    n = len(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]

    pv = beta * conv[peaks]
    pv -= pv.max()
    w = np.exp(pv)
    wsum = w.sum()
    if wsum < 1e-30:
        peak = peaks[np.argmax(conv[peaks])]
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= wsum

    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        j_arr = np.arange(js, je + 1)
        i_arr = k - j_arr
        # grad[i] gets contribution w[k] * f[j] for each (i,j) pair
        # For i != j: contribution is 2*w[k]*f[j] to grad[i]
        # For i == j: contribution is w[k]*f[i] to grad[i]
        contrib = 2.0 * w[idx] * f[j_arr]
        same = (i_arr == j_arr)
        contrib[same] = w[idx] * f[j_arr[same]]
        # Use np.add.at for duplicate indices (shouldn't happen in valid range)
        np.add.at(grad, i_arr, contrib)
    return grad


def adam_simplex(f, conv, beta, lr, m, v, t, beta1, beta2, eps):
    """One Adam step with simplex projection."""
    n = len(f)
    grad = logsumexp_gradient_fast(f, conv, beta, top_k=min(300, 2 * n))
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad ** 2
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    f_new = project_simplex(f - lr * m_hat / (np.sqrt(v_hat) + eps))
    return f_new, m, v, t + 1


def adam_optimize(f0, time_limit, beta_init=5.0, beta_final=200.0,
                  lr_init=0.01, lr_decay=0.9995, seed=42):
    """Adam optimization with increasing log-sum-exp sharpness."""
    rng = np.random.default_rng(seed)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    beta = beta_init

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break

        f, m, v, t = adam_simplex(f, conv, beta, lr, m, v, t, beta1, beta2, eps)
        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Gradually increase beta (sharpen the approximation)
        progress = min(1.0, it / max(1, int(time_limit * 10)))
        beta = beta_init * (beta_final / beta_init) ** progress
        lr *= lr_decay

        # Periodic restart
        if no_improve > 300:
            rng_local = np.random.default_rng(it * 7 + seed)
            f = project_simplex(best_f + rng_local.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            beta = beta_init * 2
            lr = lr_init * 0.5
            no_improve = 0

    return best_f, best_obj


def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break

        g = solve_lp(f, M * m_exp)
        if g is None:
            m_exp *= 1.3
            if m_exp > 10.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 60:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = np.linspace(0.05, 1.0, 12)
        for alpha in alphas:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine search
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(10):
                if hi - lo < 1e-7:
                    break
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)

        if no_improve > 100:
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def coordinate_polish(f0, time_limit, seed=42):
    """
    Coordinate-style polishing: cycle through coordinates, doing small
    local searches around each one.
    """
    rng = np.random.default_rng(seed)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    order = np.arange(n)
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break

        # Shuffle order periodically
        if it % 5 == 0:
            rng.shuffle(order)

        for i in order:
            if time.monotonic() - t0 > time_limit - 1:
                break

            # Try small perturbations at coordinate i
            # Transfer mass from i to a random other coordinate
            j = rng.integers(0, n)
            if j == i:
                continue

            ai = f[i]
            aj = f[j]
            total = ai + aj
            if total < 1e-15:
                continue

            # Try several splits of the mass between i and j
            best_split = f[i]  # keep as is
            best_local_obj = conv.max()

            for frac in [0.0, 0.2, 0.4, 0.5, 0.6, 0.8, 1.0]:
                new_ai = total * frac
                new_aj = total * (1.0 - frac)
                f[i] = new_ai
                f[j] = new_aj
                conv_new = fast_convolve(f)
                obj_new = conv_new.max()
                if obj_new < best_local_obj:
                    best_local_obj = obj_new
                    best_split = new_ai

            # Restore and apply best
            f[i] = best_split
            f[j] = total - best_split
            conv = fast_convolve(f)
            obj = conv.max()

            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()

    return best_f, best_obj


def generate_inits(n, parent=None, rng=None):
    """Generate diverse structured initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Raised cosine (Hann-like)
    hc = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1))) + 0.001
    inits.append(hc / hc.sum())

    # Tukey window (alpha=0.5)
    t = x / (n - 1)
    tk = np.ones(n)
    mask = (t < 0.25) | (t > 0.75)
    tk[mask] = 0.5 * (1 + np.cos(np.pi * (2 * t[mask] - 0.5)))
    tk += 0.001
    inits.append(tk / tk.sum())

    # Sine (half period)
    sn = np.sin(np.pi * (x + 0.5) / n) + 0.01
    inits.append(sn / sn.sum())

    # Exponential (two-sided)
    center = n / 2.0
    e = np.exp(-0.5 * (x - center) ** 2 / (n / 4.0) ** 2) + 0.001
    inits.append(e / e.sum())

    # Edge-concentrated (U-shaped)
    u = 1.0 / (np.abs(x - center) + 1.0)
    inits.append(u / u.sum())

    # Random smooth
    r = rng.standard_normal(n)
    # Smooth with moving average
    kernel = np.ones(5) / 5
    rs = np.convolve(r, kernel, mode='same')
    rs = np.maximum(rs, 0) + 0.01
    inits.append(rs / rs.sum())

    # Another random smooth
    r2 = rng.standard_normal(n)
    kernel2 = np.ones(8) / 8
    rs2 = np.convolve(r2, kernel2, mode='same')
    rs2 = np.maximum(rs2, 0) + 0.01
    inits.append(rs2 / rs2.sum())

    # Parent-based
    if parent is not None:
        x_parent = np.linspace(0, 1, len(parent))
        x_new = np.linspace(0, 1, n)
        px = np.interp(x_new, x_parent, parent)
        px = np.maximum(px, 0) + 0.001
        inits.append(px / px.sum())
        # Blend with cosine
        blend = 0.7 * (px / px.sum()) + 0.3 * (hc / hc.sum())
        inits.append(blend / blend.sum())

    return inits


def hybrid_optimize(f0, time_limit, seed=42):
    """
    Hybrid: Adam for most of the time, with periodic LP FW escapes.
    """
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.008
    beta = 10.0
    no_improve = 0
    it = 0
    t0 = time.monotonic()
    lp_counter = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break

        # Adam step
        f_new, m, v, t = adam_simplex(f, conv, beta, lr, m, v, t, beta1, beta2, eps)
        f = f_new
        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive beta and lr
        progress = min(1.0, it / 2000.0)
        beta = 10.0 * (300.0 / 10.0) ** progress
        lr *= 0.9997

        # Periodic LP escape
        lp_counter += 1
        if no_improve > 150 and rem > 30 and lp_counter > 50:
            g = solve_lp(f, best_obj * 1.01, time_limit=min(15, rem // 4))
            if g is not None:
                gs = g.sum()
                if gs > 1e-10:
                    g = g / gs
                    for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                        ft = project_simplex((1 - alpha) * f + alpha * g)
                        c_try = fast_convolve(ft).max()
                        if c_try < best_obj:
                            best_obj = c_try
                            best_f = ft.copy()
                            f = ft
                            conv = fast_convolve(f)
                            break
            lp_counter = 0
            no_improve = 0
            lr = 0.006
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            beta = 10.0

        # Stuck restart
        if no_improve > 600:
            rng = np.random.default_rng(it * 13 + seed)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            conv = fast_convolve(f)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr = 0.008
            beta = 10.0
            no_improve = 0
            lp_counter = 0

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Multi-phase optimization combining Adam with
    log-sum-exp smoothing, LP Frank-Wolfe, and coordinate polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        if a.sum() < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            old = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}) (was {old:.8f})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
    except Exception:
        pass

    # === Phase 1: Optimize parent with hybrid method ===
    print("=== Phase 1: Parent Optimization ===")
    if parent is not None:
        n_p = len(parent)
        t1 = min(250, max(60, deadline - time.monotonic() - 500))
        if t1 > 30:
            print(f"  Hybrid on parent (n={n_p}), t={t1:.0f}s")
            a_opt, obj = hybrid_optimize(parent, time_limit=t1, seed=seed)
            try_seq(a_opt, "parent_hybrid")

            # FW follow-up
            rem = deadline - time.monotonic() - 400
            if rem > 30:
                a_opt2, obj2 = fw_lp_optimize(a_opt, time_limit=min(rem, 80))
                try_seq(a_opt2, "parent_fw")

    # === Phase 2: Multi-length exploration ===
    print("\n=== Phase 2: Multi-length Exploration ===")

    lengths = [256, 384, 512]
    if parent is not None:
        n_par = len(parent)
        if n_par not in lengths:
            lengths.insert(0, n_par)

    for n in lengths:
        if time.monotonic() > deadline - 150:
            break
        if parent is not None and n == len(parent):
            continue

        inits = generate_inits(n, parent, rng)
        rem = deadline - time.monotonic() - 120
        t_per = max(25, rem // (len(inits) * len(lengths)))
        t_per = min(t_per, 70)

        for i, f_init in enumerate(inits):
            if time.monotonic() > deadline - 100:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 60)
            if t_alloc < 15:
                break
            a_opt, obj = hybrid_optimize(f_init, time_limit=t_alloc, seed=seed + i * 100)
            try_seq(a_opt, f"n{n}_i{i}")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Deep optimization on best ===
    print(f"\n=== Phase 3: Deep Optimization ===")

    if best_seq is not None:
        n_best = len(best_seq)

        # Multiple FW runs with different perturbations
        for p in range(5):
            rem = deadline - time.monotonic() - 80
            if rem < 20:
                break
            t_alloc = min(rem // (5 - p), 100)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)
            elif p == 2:
                f_init = project_simplex(0.8 * best_seq + 0.2 * (best_seq[::-1]))
            elif p == 3:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.01)
            else:
                # Blend with cosine window
                x = np.arange(n_best)
                cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / (n_best - 1))) + 0.001
                cos_w /= cos_w.sum()
                f_init = project_simplex(0.6 * (best_seq / best_seq.sum()) + 0.4 * cos_w)

            a_opt, obj = fw_lp_optimize(f_init, time_limit=t_alloc)
            try_seq(a_opt, f"deep_fw_{p}")

        # Hybrid runs with different configs
        for p in range(3):
            rem = deadline - time.monotonic() - 60
            if rem < 15:
                break
            t_alloc = min(rem // (3 - p), 80)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.002)
            else:
                # Slightly different initialization
                noise = rng.standard_normal(n_best) * 0.005
                f_init = project_simplex(best_seq + noise)

            a_opt, obj = hybrid_optimize(f_init, time_limit=t_alloc, seed=seed + 1000 + p)
            try_seq(a_opt, f"deep_hyb_{p}")

    # === Phase 4: Coordinate polishing ===
    print(f"\n=== Phase 4: Coordinate Polishing ===")
    print(f"  Current best: {best_score:.8f}")

    if best_seq is not None:
        n_best = len(best_seq)

        # Coordinate polish
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            a_opt, obj = coordinate_polish(best_seq, time_limit=min(rem, 60), seed=seed)
            try_seq(a_opt, "coord_polish")

        # Final FW on polished
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_opt, obj = fw_lp_optimize(best_seq, time_limit=min(rem, 40))
            try_seq(a_opt, "final_fw")

        # Micro perturbations + polish
        rem = deadline - time.monotonic() - 5
        if rem > 8:
            for _ in range(3):
                rem = deadline - time.monotonic() - 3
                if rem < 5:
                    break
                f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.001)
                a_opt, obj = hybrid_optimize(f_pert, time_limit=min(rem, 10),
                                             seed=seed + 5000)
                try_seq(a_opt, "micro")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
