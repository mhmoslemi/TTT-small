"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines peak-directed subgradient descent, LP-based Frank-Wolfe,
smoothed gradient descent, and random restarts.
Key innovations: anti-symmetric initialization, adaptive peak targeting,
and efficient backtracking line search.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_peak_gradient(f, k):
    """Compute gradient of (f*f)[k] w.r.t. f.
    grad[i] = 2*f[k-i] if 0 <= k-i < n, else 0.
    """
    n = len(f)
    grad = np.zeros(n)
    # For each i in [0, n-1], grad[i] = 2*f[k-i] if 0 <= k-i < n
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    if js <= je:
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * f[k - idx]
    return grad


def peak_directed_optimize(f0, time_limit, lr_init=0.01, verbose=False):
    """
    Peak-directed subgradient optimization.
    At each step, find the worst peak in f*f and take a gradient step to reduce it.
    Uses backtracking line search.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = lr_init
    top_k = 3  # Number of peaks to target

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv = fast_convolve(f)
        cmax = conv.max()

        # Find top-k peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        # Compute weighted gradient (weighted by peak values)
        pv = conv[peaks]
        # Use softmax weighting with small temperature
        temp = cmax * 0.01
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            grad += w[i] * compute_peak_gradient(f, int(peaks[i]))

        # Project onto tangent space of simplex
        grad = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            # At a "flat" point, try random perturbation
            break

        # Backtracking line search
        alpha = lr
        accepted = False
        for _ in range(15):
            f_new = project_simplex(f - alpha * grad)
            conv_new = fast_convolve(f_new)
            obj_new = conv_new.max()
            if obj_new < cmax - 1e-14:
                f = f_new
                accepted = True
                break
            alpha *= 0.7

        if accepted:
            if cmax < best_obj - 1e-14:
                best_obj = cmax
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            lr *= 0.9

        # Adaptive: if stuck, increase top_k or restart
        if no_improve > 100:
            # Try larger top_k
            top_k = min(top_k + 1, 10)
            lr = lr_init * 0.5
            no_improve = 0

        if no_improve > 300:
            # Random restart
            rng = np.random.default_rng((it * 7919 + n) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            lr = lr_init
            top_k = 3
            no_improve = 0

        # Occasionally try a different top_k
        if it % 200 == 0:
            top_k = max(1, min(10, top_k + (1 if it % 400 == 0 else -1)))

    return best_f, best_obj


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive constraint and line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Solve LP
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M * m_exp)
        bounds = [(0, None)] * n
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success or res.x is None:
            m_exp *= 1.1
            if m_exp > 5.0:
                m_exp = 1.0
            continue

        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                # Try with tighter constraint
                m_exp = max(1.0, m_exp * 0.95)
                continue
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 4.0)

        if no_improve > 30:
            rng = np.random.default_rng((it * 7919 + n * 104729) % (2**31))
            scale = 0.003 + 0.002 * min(no_improve // 30, 3)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines peak-directed subgradient descent, LP-based Frank-Wolfe,
smoothed gradient descent, and anti-symmetric initialization.
Key innovations: targeted peak reduction with FFT-based gradients,
adaptive restarts, and diverse initialization including anti-symmetric constructions.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_loss_gradient(f, T):
    """Compute gradient of T*log(sum(exp((f*f)/T))) w.r.t. f using FFT."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = w / ws
    # grad[i] = 2 * sum_j f[j] * p[j+i] via FFT cross-correlation
    fp_size = 1 << (3 * n).bit_length()
    A = np.fft.rfft(f, fp_size)
    B = np.fft.rfft(p, fp_size)
    cross = np.fft.irfft(np.conj(A) * B, fp_size)
    return 2.0 * cross[:n], conv


def peak_directed_optimize(f0, time_limit, lr_init=0.008, top_k_init=3, verbose=False):
    """
    Peak-directed subgradient optimization with backtracking line search.
    Targets the worst peaks in f*f using exact subgradients.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = lr_init
    top_k = top_k_init

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv = fast_convolve(f)
        cmax = conv.max()

        # Find top-k peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        # Weighted gradient (softmax weighting)
        pv = conv[peaks]
        temp = max(cmax * 0.005, 1e-10)
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute gradient via FFT (treat weighted peaks as p)
        p = np.zeros(2 * n - 1)
        for i in range(len(peaks)):
            if w[i] > 1e-12:
                p[int(peaks[i])] += w[i]

        fp_size = 1 << (3 * n).bit_length()
        A = np.fft.rfft(f, fp_size)
        B = np.fft.rfft(p, fp_size)
        cross = np.fft.irfft(np.conj(A) * B, fp_size)
        grad = 2.0 * cross[:n]

        # Project onto tangent space of simplex
        grad = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Backtracking line search
        alpha = lr
        accepted = False
        for _ in range(12):
            f_new = project_simplex(f - alpha * grad)
            conv_new = fast_convolve(f_new)
            obj_new = conv_new.max()
            if obj_new < cmax - 1e-14:
                f = f_new
                accepted = True
                break
            alpha *= 0.7

        if accepted:
            new_cmax = conv_new.max()
            if new_cmax < best_obj - 1e-14:
                best_obj = new_cmax
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            lr *= 0.92

        # Adaptive restart
        if no_improve > 150:
            rng = np.random.default_rng((it * 7919 + n * 104729) % (2**31))

            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            if conv.max() < best_obj:
                best_obj = conv.max()
                best_f = f.copy()
            lr = lr_init * max(0.3, 1.0 - no_improve / 500.0)
            no_improve = 0

        if verbose and it % 500 == 0:
            print(f"    PD iter {it}: obj={best_obj:.10f}, lr={lr:.6f}, top_k={top_k}")

    return best_f, best_obj


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive constraint relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M * m_exp)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success or res.x is None:
            m_exp *= 1.1
            if m_exp > 5.0:
                m_exp = 1.0
            continue

        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 20:
                m_exp = max(1.0, m_exp * 0.95)
            if no_improve > 60:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 4.0)

        if no_improve > 40:
            rng = np.random.default_rng((it * 7919 + n * 104729) % (2**31))
            scale = 0.004 + 0.002 * min(no_improve // 40, 4)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

        if verbose and it % 100 == 0:
            print(f"    LP-FW iter {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def smoothed_grad_descent(f0, time_limit, T_init=0.003, T_min=0.00005, lr_init=0.004, verbose=False):
    """Gradient descent with log-sum-exp smoothed max objective."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()

        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    T = T_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = lr_init

    while time.monotonic() - t0 < time_limit:
        it += 1

        grad, conv = smoothed_loss_gradient(f, T)
        grad = grad - np.mean(grad)

        f_new = project_simplex(f - lr * grad)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            no_improve += 1
            if no_improve % 15 == 0:

        T = max(T * 0.996, T_min)

        if no_improve > 250:
            rng = np.random.default_rng((it * 331 + 17) % (2**31))

def generate_inits(n, parent, rng):
    """Generate diverse initializations including anti-symmetric constructions."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Linear ramp (anti-symmetric around center)
    ramp = x - (n - 1) / 2.0
    c_max = 2.0 / (n * (n - 1)) if n > 1 else 0.0
    for c_frac in [0.5, 0.7, 0.85, 1.0]:
        f = 1.0 / n + c_frac * c_max * ramp
        f = np.maximum(f, 0.0)
        fs = f.sum()
        if fs > 1e-10:
            inits.append(f / fs)

    # Cubic anti-symmetric
    t = x / (n - 1) - 0.5
    for c_val in [0.3, 0.5, 0.7]:
        f = 1.0 / n + c_val * (1.0 / n) * (t * abs(t))
        f = np.maximum(f, 0.0)
        fs = f.sum()
        if fs > 1e-10:
            inits.append(f / fs)

    # Hann window
    hann = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    inits.append(hann / hann.sum())

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits.append(g / g.sum())

    # Two humps
    h1 = np.exp(-0.5 * ((x - 0.2 * n) / (0.1 * n)) ** 2)
    h2 = np.exp(-0.5 * ((x - 0.8 * n) / (0.1 * n)) ** 2)
    two = h1 + h2
    inits.append(two / two.sum())

    # Random
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    # Anti-symmetric random
    half = n // 2
    left = rng.dirichlet(np.ones(half) * 0.3)
    f_anti = np.zeros(n)
    f_anti[:half] = left * 1.5
    f_anti[half:] = (1 - left) * 0.5
    f_anti = np.maximum(f_anti, 0)
    fs = f_anti.sum()
    if fs > 1e-10:
        inits.append(f_anti / fs)

    # Parent interpolation
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search combining peak-directed subgradient,
    LP-FW, smoothed gradient, and diverse initialization.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
    except Exception:
        pass

    # Determine base length
    if parent is not None:
        base_n = len(parent)
    else:
        base_n = 368

    print(f"=== Phase 1: Initialization and Coarse Search ===")
    t_p1 = time.monotonic()
    p1_budget = 200.0

    # Try base length and a few neighbors
    lengths = [base_n]
    for d in [-8, 8, -16, 16, -4, 4]:
        cand = base_n + d
        if 128 <= cand <= 1024 and cand not in lengths:
            lengths.append(cand)

    all_best = {}

    for n_try in lengths[:5]:
        rem = p1_budget - (time.monotonic() - t_p1)
        if rem < 20:
            break
        t_alloc = min(30.0, rem - 10.0)
        if t_alloc < 10:
            break

        inits = generate_inits(n_try, parent, rng)
        # Quick eval: pick best init by max(f*f)
        best_init = None
        best_init_val = float('inf')
        for f_init in inits:
            v = fast_convolve(f_init).max()
            if v < best_init_val:
                best_init_val = v
                best_init = f_init

        if best_init is None:
            best_init = np.ones(n_try) / n_try

        print(f"  n={n_try}: best_init_obj={best_init_val:.8f}, running quick opt (t={t_alloc:.0f}s)")

        # Quick peak-directed optimization
        a_opt, _ = peak_directed_optimize(best_init, time_limit=min(15.0, t_alloc / 2))
        try_seq(a_opt, f"p1_pd_n{n_try}")

        rem = p1_budget - (time.monotonic() - t_p1)
        if rem > 10:
            a_opt2, _ = lp_fw_optimize(best_init, time_limit=min(15.0, rem - 5.0))
            try_seq(a_opt2, f"p1_lp_n{n_try}")

        if best_seq is not None and n_try == base_n:
            all_best[n_try] = best_seq / best_seq.sum()

    print(f"  Phase 1 done: best={best_score:.6f}")

    # === Phase 2: Deep Optimization ===
    print(f"\n=== Phase 2: Deep Optimization ===")
    t_p2 = time.monotonic()
    p2_budget = 1300.0

    if best_seq is None:
        best_seq = np.ones(base_n) / base_n
        best_score = float(evaluate_sequence(best_seq.tolist()))

    n_best = len(best_seq)
    f_work = best_seq / best_seq.sum()

    # Alternating optimization cycles
    n_cycles = 8
    t_per_cycle = p2_budget / n_cycles

    for cycle in range(n_cycles):
        rem = p2_budget - (time.monotonic() - t_p2)
        if rem < 30:
            break

        # Sub-phase A: Peak-directed (fast, targeted)
        t_a = min(t_per_cycle * 0.4, rem * 0.4)
        if t_a > 10:
            print(f"  Cycle {cycle}: Peak-directed (t={t_a:.0f}s)")
            a_opt, _ = peak_directed_optimize(f_work, time_limit=t_a,
                                               lr_init=0.005, top_k_init=3)
            try_seq(a_opt, f"p2_pd_{cycle}")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

        rem = p2_budget - (time.monotonic() - t_p2)

        # Sub-phase B: LP-FW (global moves)
        t_b = min(t_per_cycle * 0.35, rem * 0.35)
        if t_b > 10:
            print(f"  Cycle {cycle}: LP-FW (t={t_b:.0f}s)")
            a_opt, _ = lp_fw_optimize(f_work, time_limit=t_b)
            try_seq(a_opt, f"p2_lp_{cycle}")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

        rem = p2_budget - (time.monotonic() - t_p2)

        # Sub-phase C: Smoothed gradient (intermediate smoothness)
        t_c = min(t_per_cycle * 0.25, rem * 0.3)
        if t_c > 10:
            conv_cur = fast_convolve(f_work)
            T_start = max(0.001, 0.003 * conv_cur.max())
            print(f"  Cycle {cycle}: Smoothed GD (t={t_c:.0f}s, T={T_start:.5f})")
            a_opt, _ = smoothed_grad_descent(f_work, time_limit=t_c,
                                              T_init=T_start, T_min=0.00003)
            try_seq(a_opt, f"p2_sgd_{cycle}")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

        rem = p2_budget - (time.monotonic() - t_p2)

        # Sub-phase D: Occasional random restart + re-optimize
        if rem > 20 and cycle % 3 == 2:
            rng2 = np.random.default_rng((cycle * 12345 + 678) % (2**31))
            f_pert = project_simplex(f_work + rng2.standard_normal(n_best) * 0.01)
            t_d = min(20.0, rem - 5.0)
            if t_d > 5:
                a_opt, _ = peak_directed_optimize(f_pert, time_limit=t_d, lr_init=0.003)
                try_seq(a_opt, f"p2_restart_{cycle}")
                if best_seq is not None:
                    f_work = best_seq / best_seq.sum()

    print(f"  Phase 2 done: best={best_score:.6f}")

    # === Phase 3: Fine Polish ===
    print(f"\n=== Phase 3: Fine Polish ===")
    t_p3 = time.monotonic()
    p3_budget = 250.0

    if best_seq is not None:
        n_best = len(best_seq)
        f_work = best_seq / best_seq.sum()

        # Multiple rounds of peak-directed with decreasing learning rate
        configs = [
            {"lr_init": 0.003, "top_k_init": 2},
            {"lr_init": 0.002, "top_k_init": 3},
            {"lr_init": 0.001, "top_k_init": 2},
            {"lr_init": 0.0005, "top_k_init": 1},
        ]

        for i, cfg in enumerate(configs):
            rem = p3_budget - (time.monotonic() - t_p3)
            if rem < 15:
                break
            t_alloc = min(50.0, rem - 10.0)
            print(f"  Polish round {i}: lr={cfg['lr_init']}, top_k={cfg['top_k_init']} (t={t_alloc:.0f}s)")
            a_opt, _ = peak_directed_optimize(f_work, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"p3_polish_{i}")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

        # Final LP-FW polish
        rem = p3_budget - (time.monotonic() - t_p3)
        if rem > 20:
            t_alloc = min(40.0, rem - 5.0)
            print(f"  Final LP-FW (t={t_alloc:.0f}s)")
            a_opt, _ = lp_fw_optimize(f_work, time_limit=t_alloc)
            try_seq(a_opt, "p3_final_lp")

        # Try a slightly different length
        rem = p3_budget - (time.monotonic() - t_p3)
        if rem > 30:
            for delta in [8, -8]:
                new_n = n_best + delta
                if new_n < 128 or new_n > 1024:
                    continue
                px = np.interp(np.linspace(0, 1, new_n),
                               np.linspace(0, 1, n_best), f_work)
                px = project_simplex(px)
                t_alloc = min(25.0, rem - 5.0)
                if t_alloc > 10:
                    a_opt, _ = peak_directed_optimize(px, time_limit=t_alloc, lr_init=0.003)
                    try_seq(a_opt, f"p3_len{new_n}")
                    rem = p3_budget - (time.monotonic() - t_p3)
                    if rem < 10:
                        break

    print(f"  Phase 3 done: best={best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(368) / 368
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(368) / 368
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
