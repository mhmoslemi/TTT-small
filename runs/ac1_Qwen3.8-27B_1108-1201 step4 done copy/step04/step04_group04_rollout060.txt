def project_simplex(v):

    n = len(v)
    u = np.sort(v)[::-1]

    rho = np.nonzero(cond)[0][-1]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

Optimized LP-Frank-Wolfe with adaptive M-tightening, FFT-based smoothed gradient

import numpy as np

from scipy.sparse import csr_matrix

def fast_autoconv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_sparse_A(a):
    """Build sparse constraint matrix for conv(a,g) <= M."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        rows.extend(range(j, j + n))
        cols.extend([j] * n)
        vals.extend(a.tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_direction(a, M, A_sparse):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)

    if res.success and res.x is not None:

    return None

        else:
            no_improve += 1
            lr = max(lr * 0.7, 1e-8)

        T = max(T * 0.9995, T_min)

        if no_improve > 150:
            rng = np.random.default_rng((it * 733 + 41) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            conv = fast_autoconv(f)
            M = conv.max()
            T = max(T_init * 0.4, T_min)
            lr = 0.003
            no_improve = 0

    return best_f, best_obj


def sg_peak_shave(a0, time_limit, verbose=False):
    """Subgradient method targeting active peaks via FFT cross-correlation."""
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_autoconv(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = 0.008
    fft_size = 1 << (3 * n).bit_length()

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv = fast_autoconv(f)
        peak = conv.max()
        threshold = peak * (1.0 - 0.005 * min(no_improve / 50.0, 3.0))
        active = conv >= threshold

        # Weighted subgradient via FFT
        w_vec = np.zeros(2 * n - 1)
        w_vec[active] = (conv[active] - (peak - threshold)) + 1.0
        F = np.fft.rfft(f, fft_size)
        W = np.fft.rfft(w_vec, fft_size)
        cross = np.fft.irfft(np.conj(F) * W, fft_size)
        sg = 2.0 * cross[:n]
        sg = sg - np.mean(sg)

        ft = project_simplex(f - lr * sg)
        conv_new = fast_autoconv(ft)
        val = conv_new.max()

        if val < best_obj - 1e-15:
            best_obj = val
            best_f = ft.copy()
            f = ft
            no_improve = 0
        else:
            no_improve += 1

        if it % 50 == 0:
            lr *= 0.98

        if no_improve > 200:
            rng = np.random.default_rng((it * 97 + 13) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.0008)
            lr = 0.004
            no_improve = 0

    return best_f, best_obj


def try_evaluate(a, best_tracker):
    """Evaluate a candidate sequence and update best if improved."""
    best_score, best_seq = best_tracker
    a = np.maximum(a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    if a.sum() < 0.01:
        return best_score, best_seq
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
    return best_score, best_seq


def generate_inits(n, parent, rng):
    """Generate diverse initializations for a given length."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.2))

    x = np.arange(n, dtype=np.float64)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits.append(g / g.sum())

    c1 = np.abs(np.cos(np.pi * x / n)) + 0.03
    inits.append(c1 / c1.sum())

    h1 = np.exp(-0.5 * ((x - 0.2 * n) / (0.08 * n)) ** 2)
    h2 = np.exp(-0.5 * ((x - 0.8 * n) / (0.08 * n)) ** 2)
    th = h1 + h2
    inits.append(th / th.sum())

    half_n = n // 2
    half = rng.dirichlet(np.ones(half_n) * 0.4)
    sym = np.zeros(n)
    sym[:half_n] = half
    sym[half_n:] = half[::-1]
    inits.append(sym / sym.sum())

    # U-shaped (stronger)
    u2 = 1.0 / (np.minimum(x, n - 1 - x) + 0.5) ** 1.5
    inits.append(u2 / u2.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: LP-FW (primary), smoothed gradient (secondary),
    subgradient peak shaving (tertiary), with multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_score = float('inf')
    best_seq = None

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent * parent.sum()  # restore original scale
        best_score, best_seq = try_evaluate(parent, (best_score, best_seq))
        print(f"Initial score: {best_score:.6f} (n={len(best_seq)})")
    else:
        parent = np.ones(368) / 368
        best_score, best_seq = try_evaluate(parent, (best_score, best_seq))
        print(f"Default init score: {best_score:.6f}")

    # === Phase 1: Quick multi-length exploration (120s) ===
    print("\n=== Phase 1: Multi-length exploration ===")
    t_p1 = time.monotonic()
    p1_budget = 120.0

    base_n = len(parent) if parent is not None else 368
    lengths = sorted(set([base_n] + [base_n + d for d in [-16, -8, 8, 16, 32, -32, 48, 64]
                               if 192 <= base_n + d <= 640]))

    for n_try in lengths:
        rem = p1_budget - (time.monotonic() - t_p1)
        if rem < 15:
            break
        if n_try == base_n:
            continue  # Already optimized in Phase 2
        t_alloc = min(25.0, rem / max(1, len(lengths)))
        if t_alloc < 8:
            continue

        inits = generate_inits(n_try, parent, rng)
        best_init = None
        best_init_obj = float('inf')
        for fi in inits:
            obj = fast_autoconv(fi).max()
            if obj < best_init_obj:
                best_init_obj = obj
                best_init = fi

        a_opt, _ = fw_optimize(best_init, time_limit=t_alloc)
        best_score, best_seq = try_evaluate(a_opt, (best_score, best_seq))
        print(f"  n={n_try}: {best_score:.6f}")

    print(f"Phase 1 done: best={best_score:.6f}")

    # === Phase 2: Deep LP-FW optimization (1200s) ===
    print(f"\n=== Phase 2: Deep LP-FW (n={len(best_seq)}) ===")
    t_p2 = time.monotonic()
    p2_budget = 1200.0

    n_best = len(best_seq)
    f_work = best_seq / best_seq.sum()

    # Multiple cycles of LP-FW + gradient
    n_cycles = 8
    t_per_cycle = p2_budget / n_cycles

    for cyc in range(n_cycles):
        rem = p2_budget - (time.monotonic() - t_p2)
        if rem < 30:
            break

        # LP-FW phase
        t_fw = min(t_per_cycle * 0.6, rem * 0.5)
        if t_fw > 10:
            a_opt, _ = fw_optimize(f_work, time_limit=t_fw, verbose=True)
            best_score, best_seq = try_evaluate(a_opt, (best_score, best_seq))
            f_work = best_seq / best_seq.sum()
            n_best = len(best_seq)

        if rem < 15:
            break
        t_gd = min(t_per_cycle * 0.35, rem * 0.4)
        if t_gd > 10:
            conv_cur = fast_autoconv(f_work)
            T_init = 0.2 * conv_cur.max() / n_best
            a_opt, _ = grad_descent_opt
