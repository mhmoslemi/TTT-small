def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M):
    """Solve LP: max 1^T g s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)

    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):

    b_ub = np.full(2 * n - 1, M)

    if res.success and res.x is not None:
        return res.x
    return None


def smoothed_gradient(conv, a, T):

        val = f_func(ft)
        vals.append(val)
        if val < best_val:
            best_val = val
            best_alpha = alpha
    
    if best_alpha <= 0.0:
        return 0.0, best_val, best_val
    
    # Fine search around best

    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.005

    for _ in range(num_rounds):
        if time.monotonic() - t0 >= time_limit:
            break

        conv = fast_convolve(f)
        # Find top 3 peaks
        top_peaks = np.argsort(conv)[-3:]
        top_vals = conv[top_peaks]
        
        # Compute subgradient from top peaks
        sg = np.zeros(n)
        for k in top_peaks:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * f[k - idx]
        
        sg /= max(1, len(top_peaks))
        f = project_simplex(f - lr * sg)
        
        conv = fast_convolve(f)
        if conv.max() < best_obj - 1e-14:
            best_obj = conv.max()
            best_f = f.copy()
        
        lr *= 0.95

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    x = np.arange(n, dtype=np.float64)
    # Edge-concentrated
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # One-sided exponential
    exp1 = np.exp(-x / (n * 0.3))
    inits.append(exp1 / exp1.sum())

    # Anti-symmetric: high on left, low on right
    anti = np.exp(-x / (n * 0.5)) + 0.1 * np.exp(-(n - 1 - x) / (n * 0.5))
    inits.append(anti / anti.sum())

    # U-shape
    u_shape = np.ones(n)
    u_shape[:n//4] *= 0.5
    u_shape[3*n//4:] *= 0.5
    inits.append(u_shape / u_shape.sum())

    # Cosine window
    c1 = 0.5 * (1 + np.cos(2 * np.pi * x / n)) + 0.1
    inits.append(c1 / c1.sum())

    # Two peaks (asymmetric)
    h1 = np.exp(-0.5 * ((x - 0.3 * n) / (0.08 * n)) ** 2)
    h2 = np.exp(-0.5 * ((x - 0.7 * n) / (0.12 * n)) ** 2)
    inits.append((h1 + h2) / (h1 + h2).sum())

    # Mirror-symmetric random
    half_n = n // 2
    half = rng.dirichlet(np.ones(half_n) * 0.5)
    sym = np.zeros(n)
    sym[:half_n] = half
    sym[half_n:] = half[::-1]
    inits.append(sym / sym.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    a = np.maximum(a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    if a.sum() < 0.01:
        return float('inf')
    try:
        score = float(evaluate_sequence(a.tolist()))
        return score
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search combining LP-FW, Nesterov smoothed
    gradient, subgradient polish, peak shaving, and multi-start.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        score = evaluate_safe(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
    except Exception:
        pass

    # === Phase 1: Broad Exploration (250s) ===
    print("=== Phase 1: Broad Exploration ===")
    t1 = time.monotonic()
    p1_budget = 250.0

    if parent is not None:
        n_p = len(parent)
        t_alloc = min(100.0, p1_budget - 50.0)
        if deadline - time.monotonic() > t_alloc + 10:
            print(f"  FW-LP on parent (n={n_p}), t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(parent, time_limit=t_alloc)
            try_seq(a_opt, "parent_fw")
            # Nesterov smoothed
            rem = p1_budget - (time.monotonic() - t1)
            if rem > 50:
                print(f"  NSGD on parent, t={rem-20:.0f}s")
                a_opt, _ = nesterov_smoothed_grad(parent, time_limit=rem - 30, T_init=0.005)
                try_seq(a_opt, "parent_nsgd")

    # Try multiple lengths
    base_n = len(parent) if parent is not None else 368
    lengths = [base_n]
    for d in [8, -8, 16, -16, 4, -4, 32, -32, 64]:
        cn = base_n + d
        if 128 <= cn <= 800 and cn not in lengths:
            lengths.append(cn)
    
    other_lengths = [l for l in lengths if l != base_n]
    rng.shuffle(other_lengths)

    for n in other_lengths[:3]:
        rem = p1_budget - (time.monotonic() - t1)
        if rem < 40:
            break
        t_alloc = min(50.0, rem - 20.0)
        if t_alloc < 15:
            break

        inits = generate_inits(n, parent, rng)
        best_init = None
        best_init_obj = float('inf')
        for f_init in inits[:4]:
            conv = fast_convolve(f_init)
            obj = conv.max()
            if obj < best_init_obj:
                best_init_obj = obj
                best_init = f_init

        if best_init is not None:
            print(f"  FW-LP n={n}, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(best_init, time_limit=t_alloc)
            try_seq(a_opt, f"explore_n{n}")

    print(f"  Phase 1 done: best={best_score:.6f}")

    # === Phase 2: Deep Optimization (700s) ===
    print(f"\n=== Phase 2: Deep Optimization ===")
    t2 = time.monotonic()
    p2_budget = 700.0

    if best_seq is not None:
        f_work = best_seq.copy()
        f_work = f_work / f_work.sum()
        n_best = len(f_work)

        n_cycles = 8
        t_per_cycle = p2_budget / n_cycles

        for cycle in range(n_cycles):
            rem = p2_budget - (time.monotonic() - t2)
            if rem < 30:
                break

            t_each = min(t_per_cycle * 0.4, rem / 3.0)
            if t_each > 10:
                print(f"  Cycle {cycle}: LP-FW (t={t_each:.0f}s)")
                a_opt, _ = fw_lp_optimize(f_work, time_limit=t_each)
                try_seq(a_opt, f"p2_fw_{cycle}")
                if best_seq is not None:
                    f_work = best_seq / best_seq.sum()

            rem = p2_budget - (time.monotonic() - t2)
            t_each2 = min(t_per_cycle * 0.3, rem / 3.0)
            if t_each2 > 10:
                T_start = max(0.002, 0.004 * fast_convolve(f_work).max())
                print(f"  Cycle {cycle}: NSGD (t={t_each2:.0f}s, T={T_start:.5f})")
                a_opt, _ = nesterov_smoothed_grad(f_work, time_limit=t_each2,
                                                   T_init=T_start, T_min=0.00003)
                try_seq(a_opt, f"p2_nsgd_{cycle}")
                if best_seq is not None:
                    f_work = best_seq / best_seq.sum()

            rem = p2_budget - (time.monotonic() - t2)
            t_each3 = min(t_per_cycle * 0.25, rem / 3.0)
            if t_each3 > 10:
                print(f"  Cycle {cycle}: PeakShave (t={t_each3:.0f}s)")
                a_opt, _ = peak_shave(f_work, time_limit=t_each3)
                try_seq(a_opt, f"p2_peak_{cycle}")
                if best_seq is not None:
                    f_work = best_seq / best_seq.sum()

    print(f"  Phase 2 done: best={best_score:.6f}")

    # === Phase 3: Fine Polish (450s) ===
    print(f"\n=== Phase 3: Fine Polish ===")
    t3 = time.monotonic()
    p3_budget = 450.0

    if best_seq is not None:
        f_work = best_seq / best_seq.sum()

        # Multiple SG rounds with decreasing parameters
        configs = [
            {"lr": 0.0008, "top_k_frac": 0.05, "temp": 0.00002},
            {"lr": 0.0004, "top_k_frac": 0.03, "temp": 0.00001},
            {"lr": 0.0002, "top_k_frac": 0.02, "temp": 0.000005},
        ]

        for i, cfg in enumerate(configs):
            rem = p3_budget - (time.monotonic() - t3)
            if rem < 30:
                break
            t_alloc = min(100.0, rem - 30.0)
            if t_alloc < 15:
                break
            print(f"  SG polish round {i} (t={t_alloc:.0f}s)")
            a_opt, _ = sg_polish(f_work, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"p3_sg_{i}")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

        # Very fine Nesterov
        rem = p3_budget - (time.monotonic() - t3)
        if rem > 40:
            T_init = max(0.0003, 0.0008 * fast_convolve(f_work).max())
            t_alloc = min(120.0, rem - 20.0)
            print(f"  Fine NSGD (t={t_alloc:.0f}s, T={T_init:.6f})")
            a_opt, _ = nesterov_smoothed_grad(f_work, time_limit=t_alloc,
                                               T_init=T_init, T_min=0.00001)
            try_seq(a_opt, "p3_fine_nsgd")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

        # Peak shave with many rounds
        rem = p3_budget - (time.monotonic() - t3)
        if rem > 30:
            t_alloc = min(80.0, rem - 15.0)
            print(f"  Fine peak shave (t={t_alloc:.0f}s)")
            a_opt, _ = peak_shave(f_work, time_limit=t_alloc, num_rounds=200)
            try_seq(a_opt, "p3_peak_fine")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

        # Final LP-FW
        rem = p3_budget - (time.monotonic() - t3)
        if rem > 20:
            t_alloc = min(50.0, rem - 10.0)
            print(f"  Final LP-FW (t={t_alloc:.0f}s)")
            a_opt, _ = fw_lp_optimize(f_work, time_limit=t_alloc)
            try_seq(a_opt, "p3_final_fw")

    print(f"  Phase 3 done: best={best_score:.6f}")

    # === Phase 4: Final Tricks (remaining time) ===
    print(f"\n=== Phase 4: Final Tricks ===")

    if best_seq is not None:
        n_best = len(best_seq)
        f_work = best_seq / best_seq.sum()

        # Mirror average
        mirrored = 0.5 * f_work + 0.5 * f_work[::-1]
        mirrored = project_simplex(mirrored)
        if np.max(np.abs(mirrored - f_work)) > 1e-8:
            rem = deadline - time.monotonic()
            if rem > 20:
                a_opt, _ = sg_polish(mirrored, time_limit=min(30, rem - 10),
                                      lr=0.0003, top_k_frac=0.03, temp=0.00001)
                try_seq(a_opt, "p4_mirror")

        # Tiny perturbations
        for trial in range(5):
            if time.monotonic() > deadline - 20:
                break
            noise_scale = 0.0003 * (trial + 1)
            noise = rng.standard_normal(n_best) * noise_scale
            f_pert = project_simplex(f_work + noise)
            a_opt, _ = sg_polish(f_pert, time_limit=8, lr=0.0002,
                                  top_k_frac=0.02, temp=0.000005)
            try_seq(a_opt, f"p4_perturb_{trial}")

        # Quick LP-FW
        if time.monotonic() < deadline - 40:
            a_opt, _ = fw_lp_optimize(f_work, time_limit=30)
            try_seq(a_opt, "p4_tiny_fw")

        # Try slightly different length
        rem = deadline - time.monotonic()
        if rem > 30:
            for delta in [4, -4]:
                new_n = n_best + delta
                if new_n < 128 or new_n > 800:
                    continue
                px = np.interp(np.linspace(0, 1, new_n),
                               np.linspace(0, 1, n_best), f_work)
                px = project_simplex(px)
                a_opt, _ = sg_polish(px, time_limit=10, lr=0.0003,
                                      top_k_frac=0.03, temp=0.00001)
                try_seq(a_opt, f"p4_len{new_n}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(368) / 368
        best_score = evaluate_safe(best_seq)

    # Final verification
    fs = evaluate_safe(best_seq)
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(368) / 368
        best_score = evaluate_safe(best_seq)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
