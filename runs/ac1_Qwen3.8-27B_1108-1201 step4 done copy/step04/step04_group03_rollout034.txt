import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f):
    """Build the (2n-1) x n matrix A such that A @ g = convolve(f, g)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def logsumexp_smoothed_max(conv, tau):
    """Smoothed max using log-sum-exp trick."""
    shifted = conv / tau
    shifted -= shifted.max()
    s = np.exp(shifted).sum()
    return conv.max() + tau * np.log(s)


def smoothed_max_grad(conv, tau, n):

def fw_lp_optimize(f0, time_limit, verbose=False):
    """
    Frank-Wolfe with LP oracle, adaptive M, and line search.
    Returns (best_f, best_obj) where best_obj is max(conv).
    """
    global f_global
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    f_global = f
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 5:
            break

        g = solve_lp_oracle(f, M * m_relax)
        if g is None:
            m_relax *= 1.2
            if m_relax > 10.0:
                m_relax = 1.0
                no_improve += 1
                if no_improve > 20:
                    break
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            m_relax = min(m_relax * 1.05, 5.0)
            if no_improve > 50:
                # Restart with noise
                rng = np.random.default_rng(it * 7919 + 13)
                scale = 0.003 + 0.005 * min(1.0, no_improve / 200.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_relax = 1.0
                no_improve = 0
            continue

        f_global = f

        # Line search: try multiple alphas
        best_alpha = 0.0
        best_val = M

        # Coarse search
        for alpha in [0.02, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine search around best
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.04)
            hi = min(1.0, best_alpha + 0.04)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f_global = f
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.1, 5.0)

    f_global = None
    return best_f, best_obj


def smoothed_gradient_descent(f0, time_limit, tau_init=0.001, tau_decay=1.003,
                               lr_init=0.01, lr_decay=0.9995, verbose=False):
    """
    Gradient descent on the log-sum-exp smoothed objective.
    Uses momentum and adaptive learning rate.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_tau = tau_init
    cur_lr = lr_init
    v = np.zeros(n)
    mom = 0.9
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 2:
            break

        conv = fast_convolve(f)
        M = conv.max()

        # Compute smoothed weights
        shifted = conv / cur_tau
        shifted -= shifted.max()
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws

        # Compute gradient
        grad = np.zeros(n)
        for k in range(len(conv)):
            if w[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += w[k] * 2.0 * f[k - idx]

        # Momentum update
        v = mom * v - cur_lr * grad
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            no_improve += 1
            # If no improvement, reduce momentum and try smaller step
            if no_improve % 50 == 0:
                v *= 0.5

        cur_lr *= lr_decay
        cur_tau *= tau_decay

        # Restart if stuck
        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
            noise_scale = 0.002 + 0.004 * min(1.0, no_improve / 1000.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            cur_lr = max(lr_init * 0.5, 0.001)
            cur_tau = tau_init * 2.0
            mom = 0.85
            no_improve = 0

    return best_f, best_obj


def peak_pair_refine(f0, time_limit, verbose=False):

        peaks = np.argpartition(conv, -top_k)[-top_k:]

        # Actually, we want to find a direction d with sum(d)=0 that reduces the peaks.

        c = A_con.sum(axis=0)  # We want to minimize c @ d

        b_eq = np.array([0.0])

def multi_length_search(parent, time_limit, verbose=True):
    """
    Try different sequence lengths by interpolating the parent.
    Returns the best (sequence, score) across all lengths.
    """
    lengths = [256, 384, 512, 768]
    best_seq = parent.copy()
    best_obj = fast_convolve(parent).max()
    best_n = len(parent)

    t0 = time.monotonic()
    t_per_len = time_limit / len(lengths)

    for n_new in lengths:
        if time.monotonic() - t0 > time_limit - 10:
            break
        t_rem = min(t_per_len, time_limit - (time.monotonic() - t0) - 5)
        if t_rem < 10:
            break

        # Interpolate parent to new length
        if n_new != len(parent):
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n_new)
            f_new = np.interp(x_new, x_old, parent)
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
        else:
            f_new = parent.copy()

        # Quick FW-LP refinement
        f_opt, obj_opt = fw_lp_optimize(f_new, time_limit=t_rem * 0.6)
        score = 2 * len(f_opt) * obj_opt
        if verbose:
            print(f"    n={n_new}: obj={obj_opt:.8f}, score={score:.8f}")

        if score < 2 * best_n * best_obj:
            best_obj = obj_opt
            best_seq = f_opt
            best_n = n_new

        # Also try smoothed GD on this length
        t_rem2 = time_limit - (time.monotonic() - t0) - 5
        if t_rem2 > 10:
            f_opt2, obj_opt2 = smoothed_gradient_descent(
                f_opt, time_limit=min(t_rem2, t_per_len * 0.4),
                tau_init=0.0005, lr_init=0.005
            )
            score2 = 2 * len(f_opt2) * obj_opt2
            if verbose:
                print(f"    n={n_new} (GD): score={score2:.8f}")
            if score2 < 2 * best_n * best_obj:
                best_obj = obj_opt2
                best_seq = f_opt2
                best_n = n_new

    return best_seq, best_obj

        # For each active position, try reducing it (and adding to a random other position)
        improved = False
        for _ in range(min(20, n)):
            j = rng.choice(active)
            # Find a position to add mass to (one not contributing to the peak)
            candidates = np.setdiff1d(np.arange(n), active)
            if len(candidates) == 0:
                break
            k = rng.choice(candidates)

            # Try moving a small amount from j to k
            for delta in [0.001, 0.005, 0.01, 0.02]:
                if f[j] < delta:
                    continue
                f_try = f.copy()
                f_try[j] -= delta
                f_try[k] += delta
                f_try = np.maximum(f_try, 0)
                f_try = f_try / f_try.sum()

                conv_try = fast_convolve(f_try)
                if conv_try.max() < M - 1e-14:
                    f = f_try
                    M = conv_try.max()
                    improved = True
                    break
            if improved:
                break

        if improved:
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                # Scramble a bit
                f = project_simplex(f + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search combining FW-LP, smoothed GD,
    peak-pairing, coordinate descent, and multi-length exploration.
    """
    global f_global
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            pscore = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={pscore:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        n_init = 384
        parent = np.ones(n_init) / n_init
        try_seq(parent, "uniform_init")

    n = len(parent)

    # === Phase 1: Aggressive FW-LP on parent (~400s) ===
    print("\n=== Phase 1: FW-LP on parent ===")
    rem = deadline - time.monotonic() - 1200
    if rem > 30:
        t_alloc = min(rem, 400)
        print(f"  FW-LP, t={t_alloc:.0f}s")
        f_opt, obj_opt = fw_lp_optimize(parent, time_limit=t_alloc)
        try_seq(f_opt, "fw1")

    # === Phase 2: Smoothed gradient descent (~300s) ===
    print("\n=== Phase 2: Smoothed GD ===")
    if best_seq is not None:
        f_start = best_seq.copy()
        rem = deadline - time.monotonic() - 900
        if rem > 30:
            t_alloc = min(rem, 300)
            print(f"  Smoothed GD, t={t_alloc:.0f}s")
            f_opt, obj_opt = smoothed_gradient_descent(
                f_start, time_limit=t_alloc,
                tau_init=0.001, tau_decay=1.002,
                lr_init=0.008, lr_decay=0.9996
            )
            try_seq(f_opt, "sgd1")

            # Second pass with different params
            rem = deadline - time.monotonic() - 800
            if rem > 20:
                t_alloc = min(rem, 200)
                print(f"  Smoothed GD (pass 2), t={t_alloc:.0f}s")
                f_opt2, obj_opt2 = smoothed_gradient_descent(
                    best_seq, time_limit=t_alloc,
                    tau_init=0.0005, tau_decay=1.004,
                    lr_init=0.004, lr_decay=0.9997
                )
                try_seq(f_opt2, "sgd2")

    # === Phase 3: Peak-pair refinement (~250s) ===
    print("\n=== Phase 3: Peak-Pair Refinement ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 600
        if rem > 30:
            t_alloc = min(rem, 250)
            print(f"  Peak-pair, t={t_alloc:.0f}s")
            f_opt, obj_opt = peak_pair_refine(best_seq, time_limit=t_alloc)
            try_seq(f_opt, "peakpair")

    # === Phase 4: More FW-LP on best (~300s) ===
    print("\n=== Phase 4: Deep FW-LP ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_alloc = min(rem, 300)
            print(f"  Deep FW-LP, t={t_alloc:.0f}s")
            f_opt, obj_opt = fw_lp_optimize(best_seq, time_limit=t_alloc)
            try_seq(f_opt, "fw_deep")

            # Perturbed restart
            rem = deadline - time.monotonic() - 300
            if rem > 20:
                noise = rng.standard_normal(len(best_seq)) * 0.004
                f_pert = project_simplex(best_seq + noise)
                t_alloc2 = min(rem, 150)
                f_opt2, obj_opt2 = fw_lp_optimize(f_pert, time_limit=t_alloc2)
                try_seq(f_opt2, "fw_perturbed")

    # === Phase 5: Coordinate descent (~200s) ===
    print("\n=== Phase 5: Coordinate Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t_alloc = min(rem, 200)
            print(f"  Coord descent, t={t_alloc:.0f}s")
            f_opt, obj_opt = coordinate_descent_refine(best_seq, time_limit=t_alloc)
            try_seq(f_opt, "coord")

    # === Phase 6: Multi-length exploration (~200s) ===
    print("\n=== Phase 6: Multi-length exploration ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 60:
            t_alloc = min(rem, 200)
            print(f"  Multi-length, t={t_alloc:.0f}s")
            f_opt, obj_opt = multi_length_search(best_seq, time_limit=t_alloc)
            try_seq(f_opt, "multi_len")

    # === Phase 7: Final fine polish (~100s) ===
    print("\n=== Phase 7: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            # Very fine smoothed GD
            t_alloc = min(rem, 50)
            print(f"  Fine GD, t={t_alloc:.0f}s")
            f_opt, obj_opt = smoothed_gradient_descent(
                best_seq, time_limit=t_alloc,
                tau_init=0.0002, tau_decay=1.005,
                lr_init=0.002, lr_decay=0.9998
            )
            try_seq(f_opt, "fine_gd")

            # Tiny perturbations + quick refine
            rem = deadline - time.monotonic() - 15
            if rem > 10:
                for trial in range(3):
                    rem = deadline - time.monotonic() - 10
                    if rem < 5:
                        break
                    noise = rng.standard_normal(len(best_seq)) * 0.001
                    f_pert = project_simplex(best_seq + noise)
                    f_opt, obj_opt = fw_lp_optimize(f_pert, time_limit=min(rem, 15))
                    try_seq(f_opt, f"perturb_{trial}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
