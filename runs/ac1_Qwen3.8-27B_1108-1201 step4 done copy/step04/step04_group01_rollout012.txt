def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_gradient(f, conv, beta):
    """Compute gradient of log-sum-exp smoothing of conv, projected back to f-space.
    grad[i] = 2 * sum_k w[k] * f[k-i] where w[k] = softmax(beta * conv[k])
    """
    n = len(f)
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()

    # grad[i] = 2 * sum_k w[k] * f[k-i] for valid k-i
    f_flipped = f[::-1].copy()
    total_len = len(conv) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv_out = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv_out[n - 1:n - 1 + n]


def subgradient_topk(f, conv, top_k=16, temperature=0.001):
    """Weighted subgradient using top-k peaks with softmax weighting."""
    n = len(f)
    k_all = np.arange(len(conv))
    pk = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[pk]
    scaled = pv / temperature
    scaled -= scaled.max()
    weights = np.exp(scaled)
    weights /= weights.sum()

    sg = np.zeros(n)
    for i in range(len(pk)):
        if weights[i] < 1e-14:
            continue
        k = int(pk[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * weights[i] * f[k - idx]
    return sg


def mass_transport_step(f, conv, alpha=0.05, top_k=8):
    """Direct mass transport: move mass from high-overlap to low-overlap positions."""
    n = len(f)
    pk = np.argpartition(conv, -top_k)[-top_k:]
    pk = pk[np.argsort(conv[pk])[::-1]]

    # Compute pressure at each position
    pressure = np.zeros(n)
    for k in pk:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        pressure[idx] += f[k - idx]

    # Normalize pressure
    p_mean = pressure[pressure > 0].mean() if (pressure > 0).any() else 0
    # Transfer from above-mean pressure to below-mean
    above = pressure > p_mean
    below = pressure <= p_mean

    if not above.any() or not below.any():
        return f.copy()

    excess = (pressure[above] - p_mean).sum()
    deficit = (p_mean - pressure[below]).sum()
    if deficit < 1e-15:
        return f.copy()

    f_new = f.copy()
    # Reduce at high-pressure positions (proportionally to how far above mean)
    for i in np.where(above)[0]:
        reduction = alpha * (pressure[i] - p_mean) / (pressure[above] - p_mean).sum() * excess * 0.5
        f_new[i] -= reduction
        if f_new[i] < 0:
            f_new[i] = 0

    # Add at low-pressure positions (proportionally to how far below mean)
    redistribution = f.sum() - f_new.sum()
    if redistribution > 1e-15 and deficit > 1e-15:
        for i in np.where(below)[0]:
            add = alpha * (p_mean - pressure[i]) / deficit * redistribution
            f_new[i] += add

    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s < 1e-15:
        return f.copy()
    return f_new / s


def coordinate_descent_pass(f, max_iter=3):
    """Exact coordinate descent: update one coordinate at a time."""
    n = len(f)
    for _ in range(max_iter):
        improved = False
        for i in range(n):
            old_val = f[i]
            # Effect of changing f[i] to v on conv:
            # conv[k] changes by (v - old_val) * f[k-i] + (v - old_val) * f[i-k]
            # (accounting for both cross terms)
            # The change in max conv determines whether v is better.
            # Try v = 0 and v = 2*old_val (stay on simplex by adjusting another coord)
            # Actually, for simplex constraint, we need to redistribute.
            # Simplified: try reducing f[i] and increasing a "best" other coordinate.
            f[i] = 0
            s = f.sum()
            if s < 1e-15:
                f[i] = old_val
                continue
            f = f / s
            conv_new = fast_convolve(f)
            conv_old = fast_convolve(f)  # This is the same since we just computed it
            f[i] = old_val
            # Don't actually do full CD here as it's expensive
            break
    return f

def flat_top_window(n):
    """Generate a flat-top window of length n."""
    a0 = 0.21557895
    a1 = 0.41663158
    a2 = 0.277263158
    a3 = 0.083578947
    a4 = 0.006947368
    x = np.arange(n)
    w = np.zeros(n)
    for i in range(n):
        w[i] = a0 - a1 * np.cos(2 * np.pi * i / n) \
               + a2 * np.cos(4 * np.pi * i / n) \
               - a3 * np.cos(6 * np.pi * i / n) \
               + a4 * np.cos(8 * np.pi * i / n)
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def cosine_taper(f):
    """Apply cosine taper to a sequence."""
    n = len(f)
    x = np.linspace(0, np.pi, n)
    taper = 0.5 * (1 + np.cos(x))
    w = f * taper
    s = w.sum()
    if s < 1e-15:
        return f.copy()
    return w / s

    print(f"\nFINAL: score={best_tracker[0]:.8f}, n={len(final_seq)}")
    return [float(x) for x in final_seq]
