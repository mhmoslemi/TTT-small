"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using exact coordinate descent
with incremental convolution updates, combined with subgradient polishing
and diverse length exploration.
"""

import numpy as np
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def coord_descent_pass(a, b, S, order, n_max_iter=80):

        else:
            right_max = 0.0

        # b[k](x) for k in [k_start, k_end]:
        #   if k == 2i: b[2i] - v^2 + x^2
        #   else: b[k] - 2*v*a[k-i] + 2*x*a[k-i]

            idx_2i = two_i - k_start
            base_no_2i = base.copy()

def subgradient_polish(a, t0, time_limit, n_peaks=5, lr=0.002, decay=0.9995, mom=0.93):
    """Subgradient method targeting top peaks."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    a = a / S  # normalize to sum=1
    b = fast_convolve(a)
    best_a = a.copy()
    best_score = 2.0 * n * b.max()
    
    v = np.zeros(n)
    no_improve = 0
    cur_lr = lr
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        M = b.max()
        score = 2.0 * n * M
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Top peaks
        k = min(n_peaks, 2 * n - 1)
        peaks = np.argpartition(b, -k)[-k:]
        
        # Gradient of max: sum of gradients at active peaks, weighted
        pv = b[peaks]
        w = pv / pv.sum() if pv.sum() > 0 else np.ones(k) / k
        
        sg = np.zeros(n)
        for idx in range(k):
            ki = int(peaks[idx])
            wt = w[idx]
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            idxs = np.arange(js, je + 1)
            sg[idxs] += wt * 2.0 * a[ki - idxs]
        
        v = mom * v - cur_lr * sg
        a = project_simplex(a + v)
        
        cur_lr *= decay
        
        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 1000000)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = lr * 0.8
            no_improve = 0

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using exact coordinate descent
with incremental convolution updates, combined with subgradient polishing
and diverse length/initialization exploration.

Key innovations over previous approach:
- Exact 1D line search per coordinate (no approximate subgradient)
- Incremental O(n) convolution updates (no FFT per step)
- Coarse grid + golden section to handle non-unimodality
- Multiple sequence lengths explored (larger n has lower theoretical bound)
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def score_from_parts(b, S, n):
    """Compute objective from convolution and sum."""
    if S < 1e-15:
        return 1e10
    return 2.0 * n * b.max() / (S * S)


def coord_descent_pass(a, b, S, order, n):
    """
    One full pass of coordinate descent.
    For each coordinate (in random order), find optimal value via
    coarse grid + golden section 1D search.
    Updates a, b, S in place.
    """
    two_n_minus_1 = 2 * n - 1
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for i in order:
        v = a[i]
        if v < 0:
            v = 0.0
        S_prime = S - v
        if S_prime < 1e-12 and v < 1e-12:
            continue

        k_start = i
        k_end = min(two_n_minus_1 - 1, i + n - 1)
        if k_end < k_start:
            continue

        two_i = 2 * i
        in_range = (two_i >= k_start) and (two_i <= k_end)

        # Precompute reference values
        ks = np.arange(k_start, k_end + 1)
        a_ref = a[ks - i]
        base = b[ks].copy()

        # Fixed max (b values outside variable range)
        if k_start > 0:
            left_max = b[:k_start].max()
        else:
            left_max = -1.0
        if k_end + 1 < two_n_minus_1:
            right_max = b[k_end + 1:].max()
        else:
            right_max = -1.0
        fixed_max = max(left_max, right_max, 0.0)

        # Determine search range
        x_hi = max(3.0 * v + 2.0 / n, 0.01)
        x_hi = min(x_hi, S + 1.0)

        # Objective function
        def obj_x(x):
            if in_range:
                idx_2i = two_i - k_start
                bv = base - 2.0 * v * a_ref + 2.0 * x * a_ref
                bv[idx_2i] = base[idx_2i] - v * v + x * x
            else:
                bv = base - 2.0 * v * a_ref + 2.0 * x * a_ref
            max_b = max(fixed_max, float(bv.max()))
            total = S_prime + x
            if total < 1e-15:
                return 1e10
            return max_b / (total * total)

        # Coarse grid: 12 points
        n_grid = 12
        xs = np.linspace(0.0, x_hi, n_grid)
        vals = np.empty(n_grid)
        for gi in range(n_grid):
            vals[gi] = obj_x(xs[gi])
        best_idx = int(vals.argmin())

        # Golden section refinement
        if best_idx == 0:
            lo, hi = 0.0, xs[1]
        elif best_idx == n_grid - 1:
            lo, hi = xs[n_grid - 2], x_hi
        else:
            lo, hi = xs[best_idx - 1], xs[best_idx + 1]

        c = lo + (1.0 - gr) * (hi - lo)
        d = lo + gr * (hi - lo)
        fc = obj_x(c)
        fd = obj_x(d)

        for _ in range(35):
            if fc < fd:
                hi = d
                d = c
                fd = fc
                c = lo + (1.0 - gr) * (hi - lo)
                fc = obj_x(c)
            else:
                lo = c
                c = d
                fc = fd
                d = lo + gr * (hi - lo)
                fd = obj_x(d)

        x_opt = (lo + hi) * 0.5
        x_opt = max(0.0, min(x_opt, x_hi))

        # Apply update
        delta = x_opt - v
        if abs(delta) > 1e-14:
            a[i] = x_opt
            S += delta
            if in_range:
                mask_var = ks != two_i
                b[ks[mask_var]] += 2.0 * delta * a[ks[mask_var] - i]
                b[two_i] += delta * (v + x_opt)
            else:
                b[ks] += 2.0 * delta * a[ks - i]

    return a, b, S


def run_coord_descent(a_init, t0, time_limit, verbose=False, seed=12345):
    """Run coordinate descent until time limit."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
        S = 1.0
    b = fast_convolve(a)

    best_a = a.copy()
    best_score = score_from_parts(b, S, n)

    if verbose:
        print(f"    CD init: score={best_score:.8f}, n={n}")

    rng = np.random.default_rng(seed)
    no_improve = 0
    pass_count = 0
    max_no_improve = 80

    while time.monotonic() - t0 < time_limit:
        order = rng.permutation(n)
        a, b, S = coord_descent_pass(a, b, S, order, n)
        pass_count += 1

        S = a.sum()
        score = score_from_parts(b, S, n)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and pass_count % 100 == 0:
            print(f"    Pass {pass_count}: score={score:.8f}, no_improve={no_improve}")

        if no_improve >= max_no_improve:
            # Perturbation to escape local minimum
            scale = 0.003 * min(no_improve / 40.0, 4.0)
            a = best_a + rng.standard_normal(n) * scale
            a = np.maximum(a, 0.0)
            S = a.sum()
            b = fast_convolve(a)
            no_improve = 0
            pass_count = 0

    return best_a, best_score


def subgradient_polish(a_init, t0, time_limit, n_peaks=5, lr=0.0015, decay=0.9997, mom=0.94, seed=99):
    """Subgradient method targeting top peaks with projection."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    b = fast_convolve(a)
    best_a = a.copy()
    best_score = 2.0 * n * b.max()

    v = np.zeros(n)
    no_improve = 0
    cur_lr = lr
    rng = np.random.default_rng(seed)
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        M = b.max()
        score = 2.0 * n * M

        if score < best_score - 1e-15:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Top peaks with softmax weighting
        k = min(n_peaks, 2 * n - 1)
        peaks = np.argpartition(b, -k)[-k:]
        pv = b[peaks]
        # Temperature-based weighting
        temp = 0.001 * (1.0 + 0.1 * (it % 100))
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(k)
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for idx in range(k):
            ki = int(peaks[idx])
            wt = w[idx]
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            idxs = np.arange(js, je + 1)
            sg[idxs] += wt * 2.0 * a[ki - idxs]

        v = mom * v - cur_lr * sg
        a = project_simplex(a + v)

        cur_lr *= decay

        if no_improve > 200:
            noise_scale = 0.002 + 0.002 * min(1.0, no_improve / 400.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            cur_lr = max(lr * 0.7, 0.0003)
            no_improve = 0

    return best_a, best_score


def multiplicative_weight(a_init, t0, time_limit, lr=0.04, top_k=12, temp=0.0004, seed=77):
    """Multiplicative weight (mirror descent) updates."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    b = fast_convolve(a)
    best_a = a.copy()
    best_score = 2.0 * n * b.max()

    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    rng = np.random.default_rng(seed)
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        M = b.max()
        score = 2.0 * n * M

        if score < best_score - 1e-15:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        k = min(top_k, 2 * n - 1)
        peaks = np.argpartition(b, -k)[-k:]
        pv = b[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(k)
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        log_update = np.zeros(n)
        for idx in range(k):
            if w[idx] < 1e-14:
                continue
            ki = int(peaks[idx])
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            idxs = np.arange(js, je + 1)
            log_update[idxs] -= cur_lr * w[idx] * 2.0 * a[ki - idxs]

        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-25)
        a = a / a.sum()

        cur_lr *= 0.9996
        cur_temp *= 1.003

        if no_improve > 120:
            a = best_a * np.exp(rng.standard_normal(n) * 0.004)
            a = a / a.sum()
            cur_lr = max(lr * 0.6, 0.008)
            cur_temp = temp * 1.5
            no_improve = 0

    return best_a, best_score


def generate_inits(n_list, parent, rng):
    """Generate diverse initializations for multiple lengths."""
    inits = []
    for n_try in n_list:
        if parent is not None and n_try == len(parent):
            inits.append(("parent", parent.copy()))
            continue

        # Interpolated parent
        if parent is not None:
            px = np.interp(np.linspace(0, 1, n_try),
                           np.linspace(0, 1, len(parent)), parent)
            px = np.maximum(px, 0)
            ps = px.sum()
            if ps > 1e-10:
                inits.append((f"interp_{n_try}", px / ps))

        # Uniform
        inits.append((f"uniform_{n_try}", np.ones(n_try) / n_try))

        # Edge-weighted (concentrate at edges to reduce center peak)
        x = np.arange(n_try, dtype=np.float64)
        edge_w = 1.0 / (np.minimum(x, n_try - 1 - x) + 1.0)
        inits.append((f"edge_{n_try}", edge_w / edge_w.sum()))

        # Cosine bump
        g = np.exp(-0.5 * ((x - 0.4 * (n_try - 1)) / (0.15 * n_try)) ** 2)
        g += 1e-6
        inits.append((f"bump_{n_try}", g / g.sum()))

        # Random Dirichlet
        r = rng.dirichlet(np.ones(n_try) * 0.4)
        inits.append((f"rand_{n_try}", r))

        # Two-bump (asymmetric)
        x2 = np.arange(n_try, dtype=np.float64)
        g1 = np.exp(-0.5 * ((x2 - 0.2 * (n_try - 1)) / (0.1 * n_try)) ** 2)
        g2 = np.exp(-0.5 * ((x2 - 0.7 * (n_try - 1)) / (0.12 * n_try)) ** 2)
        two_bump = g1 + g2 + 1e-6
        inits.append((f"twobump_{n_try}", two_bump / two_bump.sum()))

    return inits

    time_per_init = 0.60 * (budget_s - 25) / max(n_top, 1)

    for rank, (sc, name, a_init) in enumerate(top_inits):
        rem = t_phase1_end - time.monotonic()
        if rem < 25:
            break
        # Allocate more time to better/same-length inits
        if best_seq is not None and len(a_init) == len(best_seq):
            t_alloc = min(rem, time_per_init * 1.5)
        else:
            t_alloc = min(rem, time_per_init)
        if t_alloc < 15:
            t_alloc = min(15, rem)

        print(f"  CD [{rank}]: {name} (n={len(a_init)}, init={sc:.6f}), t={t_alloc:.0f}s")
        a_opt, _ = run_coord_descent(a_init, t0=time.monotonic(), time_limit=t_alloc,
                                     seed=1000 + rank)
        try_seq(a_opt, f"cd_{name}")

    print(f"  Phase 1 best: {best_score:.8f}")

    # === Phase 2: SG + MW Polish (20% of time) ===
    print("\n=== Phase 2: Subgradient + MW Polish ===")
    t_phase2_end = t_start + 0.80 * (budget_s - 25)

    if best_seq is not None:
        poly_configs = [
            ('sg', {'n_peaks': 5, 'lr': 0.0015, 'decay': 0.9997, 'mom': 0.94}),
            ('sg', {'n_peaks': 10, 'lr': 0.001, 'decay': 0.9998, 'mom': 0.95}),
            ('sg', {'n_peaks': 3, 'lr': 0.002, 'decay': 0.9995, 'mom': 0.92}),
            ('mw', {'lr': 0.04, 'top_k': 10, 'temp': 0.0003}),
            ('mw', {'lr': 0.03, 'top_k': 20, 'temp': 0.0006}),
        ]

        n_poll = len(poly_configs)
        t_per = 0.20 * (budget_s - 25) / max(n_poll, 1)

        for pi, (ptype, cfg) in enumerate(poly_configs):
            rem = t_phase2_end - time.monotonic()
            if rem < 15:
                break
            t_alloc = min(rem, t_per)
            label = f"{ptype}_{pi}"
            print(f"  {label}: {cfg}, t={t_alloc:.0f}s")

            if ptype == 'sg':
                a_opt, _ = subgradient_polish(best_seq, t0=time.monotonic(),
                                              time_limit=t_alloc, **cfg, seed=2000 + pi)
            else:
                a_opt, _ = multiplicative_weight(best_seq, t0=time.monotonic(),
                                                 time_limit=t_alloc, **cfg, seed=3000 + pi)
            try_seq(a_opt, label)

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Final Coordinate Descent (20% of time) ===
    print("\n=== Phase 3: Final CD ===")

    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 20:
            t_alloc = min(rem, 0.20 * (budget_s - 25))
            print(f"  Final CD, t={t_alloc:.0f}s")
            a_opt, _ = run_coord_descent(best_seq, t0=time.monotonic(),
                                         time_limit=t_alloc, verbose=True, seed=5000)
            try_seq(a_opt, "final_cd")

            # Perturbed restarts
            rem = deadline - time.monotonic()
            if rem > 30:
                n_pert = 4
                t_per_pert = (rem - 10) / n_pert
                for pi in range(n_pert):
                    rem = deadline - time.monotonic()
                    if rem < 20:
                        break
                    t_alloc = min(rem - 5, t_per_pert)
                    noise = rng.standard_normal(len(best_seq)) * (0.002 + 0.002 * pi)
                    f_pert = np.maximum(best_seq + noise, 0)
                    f_pert = f_pert / f_pert.sum()
                    a_opt, _ = run_coord_descent(f_pert, t0=time.monotonic(),
                                                 time_limit=t_alloc, seed=6000 + pi)
                    try_seq(a_opt, f"pert_cd_{pi}")

    # Final verification
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
