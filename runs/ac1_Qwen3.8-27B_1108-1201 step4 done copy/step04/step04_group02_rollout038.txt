```python
        idx = np.arange(L)
    
    cv = conv[idx]
    s = cv * beta
    s -= s.max()
    w = np.exp(s)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(idx))
        w[np.argmax(cv)] = 1.0
    else:
        w /= wsum
    
    grad = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-15:
            continue
        k = int(idx[ii])
        # For this peak k: grad[i] += 2 * w[ii] * a[k-i] for i in [max(0,k-n+1), min(n-1,k)]
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            # grad[i_lo:i_hi+1] += 2*w[ii]*a[k - i_hi : k - i_lo + 1][::-1]
            # a[k-i] for i = i_lo, ..., i_hi gives a[k-i_lo], ..., a[k-i_hi]
            # which is reversed: a[k-i_hi], ..., a[k-i_lo]
            segment = a[k-i_hi : k-i_lo+1][::-1]
            grad[i_lo:i_hi+1] += 2.0 * w[ii] * segment
    
    return grad
```

Now for the Adam optimizer:

```python
class AdamOpt:
    def __init__(self, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
        self.lr = lr
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps
        self.t = 0
        self.m = None
        self.v = None
    
    def step(self, params, grad):
        self.t += 1
        if self.m is None:
            self.m = np.zeros_like(params)
            self.v = np.zeros_like(params)
        
        self.m = self.beta1 * self.m + (1 - self.beta1) * grad
        self.v = self.beta2 * self.v + (1 - self.beta2) * grad**2
        
        m_hat = self.m / (1 - self.beta1**self.t)
        v_hat = self.v / (1 - self.beta2**self.t)
        
        return params - self.lr * m_hat / (np.sqrt(v_hat) + self.eps)
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth gradient-based optimization
with Adam optimizer, adaptive temperature annealing, LP-based Frank-Wolfe escapes,
and multi-length exploration. Key improvements over previous approach:
- Smooth log-sum-exp objective instead of hard max (better gradients)
- Adam optimizer for adaptive per-coordinate step sizes
- Backtracking line search for convergence guarantees
- Multi-length exploration (384, 512, 768)
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad(a, conv, beta, top_k=300):
    """
    Gradient of F(a) = (1/beta)*log(sum_k exp(beta*(a*a)_k)) w.r.t. a.
    Uses top-k peaks for efficiency.
    """
    n = len(a)
    L = len(conv)
    
    if top_k is not None and top_k < L:
        idx = np.argpartition(conv, -top_k)[-top_k:]
    else:
        idx = np.arange(L)
    
    cv = conv[idx].astype(np.float64)
    s = cv * beta
    s -= s.max()
    w = np.exp(s)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(idx))
        w[np.argmax(cv)] = 1.0
    else:
        w /= wsum
    
    grad = np.zeros(n)
    for ii in range(len(idx)):
        wi = w[ii]
        if wi < 1e-15:
            continue
        k = int(idx[ii])
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            # a[k-i] for i = i_lo..i_hi
            # = a[k-i_lo], a[k-i_lo-1], ..., a[k-i_hi]
            # = a[k-i_hi : k-i_lo+1][::-1]
            grad[i_lo:i_hi+1] += 2.0 * wi * a[k-i_hi:k-i_lo+1][::-1]
    
    return grad


def smooth_obj(a, conv, beta, top_k=300):
    """Smooth approximation of max(conv)."""
    if top_k is not None and top_k < len(conv):
        idx = np.argpartition(conv, -top_k)[-top_k:]
        cv = conv[idx]
    else:
        cv = conv
    s = cv * beta
    s -= s.max()
    return (s.max() + np.log(np.exp(s).sum())) / beta


def solve_lp_simplex(f, M, n_constraints=None):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0, sum(g) <= 1.
    Returns g or None.
    """
    n = len(f)
    # Variables: g_0, ..., g_{n-1}
    c = -np.ones(n)
    # Constraint: (f*g)[k] <= M for k = 0, ..., 2n-2
    # (f*g)[k] = sum_{j} f[|k-j|]... wait no.
    # (f*g)[k] = sum_j f[j] * g[k-j] (with g[k-j]=0 if out of range)
    # For fixed k: (f*g)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} f[j] * g[k-j]
    # Let i = k-j, so j = k-i:
    # = sum_{i=max(0,n-1-k)}^{min(n-1,k)} f[k-i] * g[i]
    
    # Actually for the Frank-Wolfe step, we need (f*g) where g is the new direction.
    # (f*g)[k] = sum_j f[j]*g[k-j]
    
    # In matrix form: A_ub[g] <= M where A_ub[k, i] = coefficient of g[i] in (f*g)[k]
    # (f*g)[k] = sum_i g[i] * f[k-i] (with f[k-i]=0 if k-i out of range)
    # So A_ub[k, i] = f[k-i] if 0 <= k-i < n, else 0
    
    if n_constraints is not None:
        # Only use a subset of constraints for speed
        k_vals = np.random.choice(2*n-1, size=n_constraints, replace=False)
    else:
        k_vals = np.arange(2*n-1)
    
    A_ub = np.zeros((len(k_vals), n))
    for row, k in enumerate(k_vals):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            # g[i] appears with coefficient f[k-i] for i in [i_lo, i_hi]
            A_ub[row, i_lo:i_hi+1] = f[k-i_hi:k-i_lo+1][::-1]
    
    b_ub = np.full(len(k_vals), M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 20})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def adam_smooth_optimize(f0, time_limit, initial_beta=50, final_beta=500,
                         lr=0.01, top_k=200, lp_every=200):
    """
    Smooth gradient optimization with Adam and adaptive temperature.
    """
    n = len(f0)
    a = np.maximum(f0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    cur_beta = initial_beta
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    lr_decay = 0.9997
    beta_growth = 1.005
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break
        
        # Compute smooth gradient
        grad = smooth_grad(a, conv, cur_beta, top_k=top_k)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)
        
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        conv = conv_new
        
        # Adaptive scheduling
        cur_beta *= beta_growth
        cur_beta = min(cur_beta, final_beta)
        lr *= lr_decay
        
        # LP escape
        if no_improve > lp_every and rem > 15:
            g = solve_lp_simplex(a, best_obj * 1.01)
            if g is not None:
                for alpha in [0.1, 0.3, 0.5, 0.7, 1.0]:
                    a_try = project_simplex((1-alpha)*a + alpha*g)
                    c_try = fast_convolve(a_try).max()
                    if c_try < best_obj:
                        a = a_try
                        conv = fast_convolve(a)
                        best_obj = c_try
                        best_a = a.copy()
                        break
            no_improve = 0
            lr = max(lr * 0.5, 0.001)
        
        # Random restart
        if no_improve > 500:
            rng = np.random.default_rng(t * 7 + 13)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            conv = fast_convolve(a)
            m = np.zeros(n)
            v = np.zeros(n)
            lr = initial_lr if 'initial_lr' in dir() else 0.005
            no_improve = 0
    
    return best_a, best_obj


def fw_lp_optimize(f0, time_limit):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        
        g = solve_lp_simplex(f, M)
        if g is None:
            continue
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        for alpha in alphas:
            ft = project_simplex((1.0 - alpha) * f + alpha * g)
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f = project_simplex((1.0 - best_alpha) * f + best_alpha * g)
            conv = fast_convolve(f)
            M = conv.max()
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 50:
            rng = np.random.default_rng(it * 31)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0
    
    return best_f, best_obj


def generate_inits(n, parent=None, seed=42):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    inits.append(np.ones(n) / n)
    
    # U-shaped (helps reduce center peak)
    u = (x - n/2)**2 + 1.0
    inits.append(u / u.sum())
    
    # Inverse U (bump in middle)
    inv_u = n**2 - (x - n/2)**2 + 1.0
    inv_u = np.maximum(inv_u, 0)
    if inv_u.sum() > 0:
        inits.append(inv_u / inv_u.sum())
    
    # Cosine window (raised to power)
    c = np.cos(np.pi * (x / (n-1) - 0.5))**6 + 0.01
    inits.append(c / c.sum())
    
    # Exponential from both edges
    e = np.exp(-x/(n/4)) + np.exp(-(n-1-x)/(n/4))
    inits.append(e / e.sum())
    
    # Random Dirichlet (concentrated)
    r = rng.dirichlet(np.ones(n) * 0.5)
    inits.append(r)
    
    # Random Dirichlet (spread)
    r2 = rng.dirichlet(np.ones(n) * 2.0)
    inits.append(r2)
    
    # Parent-based
    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            # Blend with U-shape
            u_norm = (x - n/2)**2 + 1.0
            u_norm = u_norm / u_norm.sum()
            blended = 0.7*(px/ps) + 0.3*u_norm
            inits.append(blended / blended.sum())
    
    return inits
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth log-sum-exp gradient optimization
with Adam optimizer, adaptive temperature annealing, LP-based Frank-Wolfe escapes,
and multi-length exploration. Key improvements: smooth gradients (not subgradients),
Adam for adaptive steps, better initialization with U-shaped priors.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad_full(a, conv, beta):
    """
    Exact gradient of F(a) = (1/beta)*log(sum_k exp(beta*(a*a)_k)).
    grad[i] = 2 * sum_k w_k * a[k-i] where w_k = softmax(beta*conv[k]), a[k-i]=0 if out of range.
    """
    n = len(a)
    L = len(conv)
    s = conv * beta
    s -= s.max()
    w = np.exp(s)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(L)
        w[np.argmax(conv)] = 1.0
    else:
        w /= wsum

    grad = np.zeros(n)
    for k in range(L):
        wk = w[k]
        if wk < 1e-15:
            continue
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            grad[i_lo:i_hi + 1] += 2.0 * wk * a[k - i_hi:k - i_lo + 1][::-1]
    return grad


def solve_lp_oracle(f, M, time_limit=15):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns normalized g or None.
    """
    n = len(f)
    c = -np.ones(n)
    num_constraints = 2 * n - 1

    A_ub = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            A_ub[k, i_lo:i_hi + 1] = f[k - i_hi:k - i_lo + 1][::-1]

    b_ub = np.full(num_constraints, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None

def generate_inits(n, parent=None, seed=42):
    """Generate diverse structured initializations for length n."""
    rng = np.random.default_rng(seed)
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # U-shaped (reduces center peak of convolution)
    u = (x - n / 2.0) ** 2 + 1.0
    inits.append(u / u.sum())

    # Strong U-shaped
    u2 = (x - n / 2.0) ** 4 + 1.0
    inits.append(u2 / u2.sum())

    # Cosine raised to high power (concentrated at edges)
    c = np.cos(np.pi * (x / (n - 1) - 0.5)) ** 8 + 0.01
    inits.append(c / c.sum())

    # Exponential from both edges
    e = np.exp(-x / (n / 5.0)) + np.exp(-(n - 1 - x) / (n / 5.0))
    inits.append(e / e.sum())

    # Inverse cosine (bump in middle - opposite)
    ic = 1.0 / (np.cos(np.pi * (x / (n - 1) - 0.5)) ** 2 + 0.1)
    inits.append(ic / ic.sum())

    # Random Dirichlet (concentrated)
    r = rng.dirichlet(np.ones(n) * 0.3)
    inits.append(r)

    # Random Dirichlet (spread)
    r2 = rng.dirichlet(np.ones(n) * 3.0)
    inits.append(r2)

    # Two-bump
    bump = np.exp(-((x - n * 0.2) / (n * 0.1)) ** 2) + np.exp(-((x - n * 0.8) / (n * 0.1)) ** 2) + 0.01
    inits.append(bump / bump.sum())

    # Parent-based
    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            u_norm = (x - n / 2.0) ** 2 + 1.0
            u_norm = u_norm / u_norm.sum()
            for blend in [0.3, 0.5, 0.7]:
                blended = blend * (px / ps) + (1 - blend) * u_norm
                inits.append(blended / blended.sum())

    return inits

        t_per = min(t_per, 70)

        print(f"  Length n={n}, {len(inits)} inits, ~{t_per:.0f}s each")

        for i, f_init in enumerate(inits):
            if time.monotonic() > deadline - 120:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 80)
            if t_alloc < 15:
                break

            a_opt, obj = adam_smooth_optimize(
                f_init, time_limit=t_alloc,
                initial_beta=20, final_beta=250,
                lr=0.015, top_k=150, lp_every=180)
            try_seq(a_opt, f"n{n}_i{i}")

    # === Phase 3: Deep optimization on best ===
    print(f"\n=== Phase 3: Deep Optimization (best={best_score:.8f}) ===")

    if best_seq is not None:
        n_best = len(best_seq)

        # Multiple Adam configurations
        configs = [
            {"initial_beta": 50, "final_beta": 500, "lr": 0.01, "top_k": 250, "lp_every": 100},
            {"initial_beta": 100, "final_beta": 1000, "lr": 0.005, "top_k": 400, "lp_every": 150},
            {"initial_beta": 30, "final_beta": 200, "lr": 0.02, "top_k": 150, "lp_every": 130},
            {"initial_beta": 200, "final_beta": 2000, "lr": 0.003, "top_k": 600, "lp_every": 300},
        ]

        for ci, cfg in enumerate(configs):
            rem = deadline - time.monotonic() - 50
            if rem < 15:
                break
            t_alloc = min(rem // (len(configs) - ci), 90)

            if ci == 0:
                f_init = best_seq.copy()
            elif ci == 1:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)
            elif ci == 2:
                f_init = project_simplex(0.85 * best_seq + 0.15 * (np.ones(n_best) / n_best))
            else:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.008)

            a_opt, obj = adam_smooth_optimize(f_init, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"deep_{ci}")

        # FW deep
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=min(rem, 80))
            try_seq(a_opt, "deep_fw")

    # === Phase 4: Final polishing ===
    print(f"\n=== Phase 4: Final Polish (best={best_score:.8f}) ===")

    if best_seq is not None:
        n_best = len(best_seq)

        # Very high beta polish (sharp max approximation)
        rem = deadline - time.monotonic() - 12
        if rem > 8:
            a_opt, _ = adam_smooth_optimize(
                best_seq, time_limit=min(rem, 40),
                initial_beta=500, final_beta=5000,
                lr=0.002, top_k=700, lp_every=500,
                lr_decay=0.9999, beta_growth=1.001,
                noise_scale=0.001, noise_every=800)
            try_seq(a_opt, "polish_high_beta")

        # Micro perturbations with short optimization
        rem = deadline - time.monotonic() - 8
        for mi in range(6):
            if rem < 4:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.0008)
            a_opt, _ = adam_smooth_optimize(
                f_pert, time_limit=min(rem, 8),
                initial_beta=1000, final_beta=10000,
                lr=0.001, top_k=800, lp_every=10000,
                lr_decay=1.0, beta_growth=1.0)
            try_seq(a_opt, f"micro_{mi}")
            rem = deadline - time.monotonic() - 8

        # Final FW on best
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=min(rem, 30))
            try_seq(a_opt, "final_fw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
