import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def cross_corr_at_lag(f, k):
    """Compute cross-correlation of f with itself at lag k: sum_j f[j]*f[k-j]
    This gives the subgradient contribution for convolution position k."""
    n = len(f)
    # (f*f)[k] = sum_{j} f[j] * f[k-j] where 0<=j<n and 0<=k-j<n
    # d/d f[i] (f*f)[k] = f[k-i] + f[k-i] if i != k-i, else 2*f[k-i]
    # Actually: d/d f[i] sum_j f[j]*f[k-j] = f[k-i] + f[k-i] = 2*f[k-i] (if 0<=k-i<n)
    # Wait: (f*f)[k] = sum_j f[j]*f[k-j]
    # d/d f[i]: the term f[i]*f[k-i] contributes f[k-i], and the term f[k-i]*f[i] contributes f[i]
    # So d/d f[i] = f[k-i] + f[i]... no that's wrong.
    # Let me be more careful. (f*f)[k] = sum_j f[j]*f[k-j]
    # The terms involving f[i] are: j=i gives f[i]*f[k-i], and k-j=i gives j=k-i, term is f[k-i]*f[i]
    # If i != k-i: two terms, each contributing f[k-i] to d/d f[i]. Total: 2*f[k-i]
    # If i == k-i (i.e., k=2i): one term f[i]*f[i], derivative is 2*f[i]
    # So in all cases: d/d f[i] (f*f)[k] = 2*f[k-i] (when 0<=k-i<n)
    result = np.zeros(n)
    # For position i in f, the contribution is 2*f[k-i] if 0 <= k-i < n
    idx = np.arange(n)
    k_minus_i = k - idx
    valid = (k_minus_i >= 0) & (k_minus_i < n)
    result[valid] = 2.0 * f[k_minus_i[valid]]
    return result

    best_obj = conv.max()
    best_f = f.copy()

    lr = lr_init
    no_improve = 0

            break

def lse_gradient(f, beta):
    """Compute LSE smoothed gradient."""
    n = len(f)
    conv = fast_convolve(f)
    # Numerically stable softmax
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()
    
    # Gradient: 2 * cross_correlation(w, f) of length n
    # grad[i] = 2 * sum_{m} w[m] * f[m - i] (where 0 <= m-i < n)
    # Equivalent: grad[i] = 2 * (w * f_reversed)[n-1+i] ... 
    # Let's use FFT
    f_rev = f[::-1]
    total_len = (2*n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f_rev, fft_size)
    conv_wf = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv_wf[n-1 : n-1+n]

def lse_descent(f0, time_limit, beta_start=50.0, beta_end=50000.0, lr_init=None):

    momentum = 0.92

    t0 = time.monotonic()
    no_improve = 0
    it = 0

                best_f = f.copy()

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) Exact active-peak subgradient descent, (2) LSE with Nesterov acceleration,
(3) Coordinate-wise peak equalization, (4) FW-LP with tightened active constraints,
(5) Alternating multi-phase refinement, (6) Length exploration.
Key innovations over previous: Nesterov momentum for LSE, direct peak-equalization
coordinate updates, and tighter LP constraint handling.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgrad_at_lag(f, k):
    """Subgradient of (f*f)[k] with respect to f. Gives 2*f[k-i] for valid i."""
    n = len(f)
    result = np.zeros(n)
    idx = np.arange(n)
    k_minus_i = k - idx
    valid = (k_minus_i >= 0) & (k_minus_i < n)
    result[valid] = 2.0 * f[k_minus_i[valid]]
    return result


def lse_gradient(f, beta):
    """Compute LSE-smoothed gradient of max(f*f)."""
    n = len(f)
    conv = fast_convolve(f)
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()
    # grad[i] = 2 * sum_m w[m] * f[m-i] for 0<=m-i<n
    # = 2 * (w * f_rev)[n-1+i] via circular conv
    f_rev = f[::-1]
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f_rev, fft_size)
    conv_wf = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv_wf[n - 1:n - 1 + n]


def active_peak_descent(f0, time_limit, top_k=5, lr_init=0.4, decay_rate=0.998):
    """Exact subgradient descent using top-k active convolution peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()

    lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        it += 1

        current_max = conv.max()

        else:
            active = np.argpartition(conv, -top_k)[-top_k:]

        w = conv[active]

        if w_sum < 1e-15:
            w = np.ones(len(active)) / len(active)
        else:
            w = w / w_sum

def lse_nesterov(f0, time_limit, beta_start=100.0, beta_end=200000.0, lr_init=None):
    """LSE-smoothed gradient descent with Nesterov acceleration."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    if lr_init is None:
        lr_init = 1.5 / np.sqrt(n)

    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = lr_init
    v = np.zeros(n)
    y = f.copy()  # Nesterov extrapolation point

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    t = 1.0  # Nesterov parameter

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        it += 1

        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        m = (t - 1.0) / (t_new + 1.0)
        y = f + m * (f - y)

        grad = lse_gradient(y, beta)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad = grad / gn

        v_new = m * v - lr / t_new * grad
        f_new = project_simplex(f + v_new)
        v = v_new

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            t = t_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 10 == 0:
                lr *= 0.97
                v *= 0.5
            if no_improve > 60:
                beta = min(beta * 1.5, beta_end)
                v = np.zeros(n)
                y = f.copy()
                t = 1.0
            if no_improve > 300:
                rng = np.random.default_rng(it * 23 + 11)

def coord_equalize(f0, time_limit, n_sweeps=50):

        if time.monotonic() - t0 > time_limit:
            break

        # Randomize coordinate order
        order = rng.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[i]
            if old_val < 1e-8:
                # Small: try setting to small positive
                candidates = [0.0, old_val + 0.001, 0.002, 0.005]
            else:
                # Try: 0, current, and values that would reduce top peaks
                conv = fast_convolve(f)
                top_peak = np.argmax(conv)
                # If we decrease f[i], peaks involving i go down
                # If we increase f[i], other peaks go up
                # Try: reduce by 10%, 30%, 50%, and set to 0
                candidates = [old_val, old_val * 0.7, old_val * 0.5, old_val * 0.3, 0.0]

    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    tighten_factor = 0.995

    for cycle in range(n_cycles):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        lp_time = min(remaining * 0.7, 60)

        conv = fast_convolve(f)
        M = conv.max()

        # Find active peaks (top peaks within 98% of max)
        threshold = M * 0.98
        active = np.where(conv >= threshold)[0]

        # Build LP matrix
        rows, cols, vals = [], [], []
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                js = np.arange(j_start, j_end + 1)
                rows.extend([k] * len(js))
                cols.extend(js.tolist())
                vals.extend([f[k - j] for j in js])

        A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

        b_ub = np.full(2 * n - 1, M)
        # Tighten active constraints
        b_ub[active] = M * tighten_factor

        c = -np.ones(n)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': lp_time})
            if res.success and res.x is not None:
                g = np.maximum(res.x, 0.0)
                gs = g.sum()
                if gs > 1e-15:
                    g = g / gs

                    # Line search
                    best_alpha = 0
                    best_val = M
                    for alpha in [0.03, 0.07, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
                        ft = (1 - alpha) * f + alpha * g
                        ft = np.maximum(ft, 0.0)
                        st = ft.sum()
                        if st < 1e-15:
                            continue
                        ft = ft / st
                        val = fast_convolve(ft).max()
                        if val < best_val:
                            best_val = val
                            best_alpha = alpha

                    if best_alpha > 0:
                        f = np.maximum((1 - best_alpha) * f + best_alpha * g, 0.0)
                        f = f / f.sum()
                        conv = fast_convolve(f)
                        M = conv.max()
                        if M < best_obj - 1e-15:
                            best_obj = M
                            best_f = f.copy()

                        # Decrease tighten factor slightly for next cycle
                        tighten_factor = max(0.98, tighten_factor - 0.003)
                    else:
                        tighten_factor = max(0.99, tighten_factor + 0.002)
                else:
                    tighten_factor = max(0.99, tighten_factor + 0.002)
            else:
                tighten_factor = max(0.99, tighten_factor + 0.002)
        except Exception as e:
            print(f"  FWLP error: {e}")
            tighten_factor = 0.995

        print(f"  FWLP cycle={cycle}, obj={best_obj:.12f}, tighten={tighten_factor:.4f}, t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def try_sequence(a, best_tracker):
    """Evaluate and track best."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  ★ NEW BEST: {score:.10f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Multi-phase strategy:
    Phase 1: Active-peak subgradient descent (exact, no smoothing)
    Phase 2: LSE with Nesterov acceleration (high beta range)
    Phase 3: Coordinate-wise peak equalization
    Phase 4: FW-LP with progressive tightening
    Phase 5: Alternating multi-round refinement
    Phase 6: Length exploration
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_tracker = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
            try_sequence(parent, best_tracker)
    except Exception as e:
        print(f"No parent: {e}")

    if parent is None:
        parent = np.ones(448) / 448
        try_sequence(parent, best_tracker)

    current = best_tracker[1].copy() if best_tracker[1] is not None else parent.copy()

    # === Phase 1: Active Peak Descent ===
    print(f"\n=== Phase 1: Active Peak Subgradient Descent ===")
    remaining = deadline - time.monotonic()
    t1 = min(250, remaining * 0.15)
    if t1 > 30:
        print(f"  Time: {t1:.0f}s")
        a_opt, _ = active_peak_descent(current, time_limit=t1, top_k=3, lr_init=0.5, decay_rate=0.999)
        try_sequence(a_opt, best_tracker)
        if best_tracker[1] is not None and len(best_tracker[1]) == len(current):
            current = best_tracker[1].copy()

    # === Phase 2: LSE Nesterov ===
    print(f"\n=== Phase 2: LSE + Nesterov ===")
    remaining = deadline - time.monotonic()
    t2 = min(350, remaining * 0.25)
    if t2 > 30:
        print(f"  Time: {t2:.0f}s")
        a_opt, _ = lse_nesterov(current, time_limit=t2, beta_start=100.0, beta_end=300000.0, lr_init=2.0/np.sqrt(len(current)))
        try_sequence(a_opt, best_tracker)
        if best_tracker[1] is not None and len(best_tracker[1]) == len(current):
            current = best_tracker[1].copy()

    # === Phase 3: Coordinate Equalization ===
    print(f"\n=== Phase 3: Coordinate Peak Equalization ===")
    remaining = deadline - time.monotonic()
    t3 = min(200, remaining * 0.12)
    if t3 > 20:
        print(f"  Time: {t3:.0f}s")
        a_opt, _ = coord_equalize(current, time_limit=t3, n_sweeps=30)
        try_sequence(a_opt, best_tracker)
        if best_tracker[1] is not None and len(best_tracker[1]) == len(current):
            current = best_tracker[1].copy()

    # === Phase 4: FW-LP Multi-Tighten ===
    print(f"\n=== Phase 4: FW-LP Progressive Tightening ===")
    remaining = deadline - time.monotonic()
    t4 = min(250, remaining * 0.18)
    if t4 > 30:
        print(f"  Time: {t4:.0f}s")
        a_opt, _ = fw_lp_multi_tighten(current, time_limit=t4, n_cycles=8)
        try_sequence(a_opt, best_tracker)
        if best_tracker[1] is not None and len(best_tracker[1]) == len(current):
            current = best_tracker[1].copy()

    # === Phase 5: Alternating Refinement ===
    print(f"\n=== Phase 5: Alternating Refinement ===")
    n_rounds = 3
    for r in range(n_rounds):
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break
        t_half = remaining / (2 * (n_rounds - r))
        if t_half < 25:
            break

        # Round: LSE then Active Peak
        t_lse = min(t_half * 0.6, 150)
        t_apd = min(t_half * 0.4, 100)

        if t_lse > 20:
            a_opt, _ = lse_nesterov(current, time_limit=t_lse,
                                     beta_start=500.0 + r * 500.0,
                                     beta_end=500000.0,
                                     lr_init=1.0 / np.sqrt(len(current)))
            try_sequence(a_opt, best_tracker)
            if best_tracker[1] is not None and len(best_tracker[1]) == len(current):
                current = best_tracker[1].copy()

        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_opt, _ = active_peak_descent(current, time_limit=min(t_apd, remaining - 10),
                                            top_k=2, lr_init=0.2, decay_rate=0.9995)
            try_sequence(a_opt, best_tracker)
            if best_tracker[1] is not None and len(best_tracker[1]) == len(current):
                current = best_tracker[1].copy()

        print(f"  Round {r+1} done, best={best_tracker[0]:.10f}")

    # === Phase 6: Length Exploration ===
    print(f"\n=== Phase 6: Length Exploration ===")
    n_parent = len(current)
    alt_lengths = [384, 480, 512, 576, 640]
    if n_parent in alt_lengths:
        alt_lengths.remove(n_parent)

    for n_alt in alt_lengths:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        t_alloc = min(80, (remaining - 20) / 2.0)
        if t_alloc < 30:
            break

        print(f"  Trying n={n_alt}, time={t_alloc:.0f}s")
        x_old = np.linspace(0, 1, len(current))
        x_new = np.linspace(0, 1, n_alt)
        f_interp = np.interp(x_new, x_old, current)
        f_interp = np.maximum(f_interp, 0.0)
        if f_interp.sum() > 1e-10:
            f_interp = f_interp / f_interp.sum()

        # Quick LSE on new length
        a_opt, _ = lse_nesterov(f_interp, time_limit=t_alloc,
                                 beta_start=50.0, beta_end=100000.0,
                                 lr_init=1.5 / np.sqrt(n_alt))
        try_sequence(a_opt, best_tracker)

    # === Final: One more polish on the best ===
    remaining = deadline - time.monotonic()
    if remaining > 30 and best_tracker[1] is not None:
        print(f"\n=== Final Polish ===")
        final_current = best_tracker[1].copy()
        t_final = min(remaining - 10, 100)
        a_opt, _ = lse_nesterov(final_current, time_limit=t_final,
                                 beta_start=1000.0, beta_end=500000.0,
                                 lr_init=0.5 / np.sqrt(len(final_current)))
        try_sequence(a_opt, best_tracker)

    # Final verification
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            best_score = final_score
    else:
        final_seq = np.ones(448) / 448
        best_score = float(evaluate_sequence(final_seq.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL: score={best_score:.10f}, n={len(final_seq)}")
    print(f"{'='*50}")
    return [float(x) for x in final_seq]
