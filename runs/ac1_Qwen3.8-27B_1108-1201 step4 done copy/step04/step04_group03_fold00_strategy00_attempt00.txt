def project_simplex(v):
    n = len(v)
    u = np.cumsum(v)[::-1]
    scl = u / (np.arange(0, len(u)) + 1)
    theta, rho = 1e-10, 0
    idx = np.arange(n) + 1
    uv = np.sort(v)[::-1]
    for i in range(1, n):
        if uv[i] < uv[i-1]:
            theta = (uv[i-1] - uv[i]) / (i + 1)
            rho = i
            break
        else:
            if i == n-1:
                rho = n-1
    x_star = np.maximum(v - theta, 0)
    return x_star

def find_peak_weights(seq, n, c=1.0, min_peak_val=0.0001):
    results = {}
    conv = np.zeros(2*n - 1)
    conv[:] = np.convolve(seq, seq, mode='full')

    top_peak = np.argmax(conv)
    top_val = conv[top_peak]
    top_peak_contrib = np.zeros(n)
    for offset in range(n):
        if top_val > 1e-15:
            # peak starting contribution only from valid indices

            weight = seq[i] * seq[j]

def reactive_convolution_reduction(seq, n, factor=0.95):

    if total_mass < 1e-15:
        return np.ones(n)/n

    scale_factor = factor*(1.0 / np.sqrt(np.log(0.0001 + total_mass)))
