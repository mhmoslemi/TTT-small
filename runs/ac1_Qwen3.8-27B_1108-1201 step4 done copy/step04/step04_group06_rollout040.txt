"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. SQP-like method: LP direction finding on active peak set + line search
2. Smooth gradient descent (log-sum-exp) with Nesterov acceleration
3. Peak shaving for fine-tuning
4. Adaptive active set management and restart strategy
Key improvement: smaller LPs (only top-K active constraints) for faster iterations,
plus FFT-based smooth gradient computation.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def simplex_project(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - s))[0]
    if len(rho) == 0:
        return np.full(n, s / n)
    rho = rho[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_gradient_fft(a, conv, temp):
    """Compute gradient of log-sum-exp smoothed max of conv, using FFT."""
    n = len(a)
    scaled = conv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        # Fallback to hard max gradient
        k_max = int(np.argmax(conv))
        grad = np.zeros(n)
        j_start = max(0, k_max - n + 1)
        j_end = min(n - 1, k_max)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] = 2.0 * a[k_max - idxs]
        return grad
    w /= w_sum
    # grad[j] = 2 * sum_k w[k] * a[k-j]
    # Use FFT cross-correlation: cross_corr[tau] = sum_n w[n]*a[n-tau]
    N = 1 << (len(conv) + n - 1).bit_length()
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    cross_corr = np.fft.irfft(W * A, N)[:n]
    return 2.0 * cross_corr


def build_lp_matrices(a, active_indices):
    """Build LP constraint matrices for active peak constraints."""
    n = len(a)
    K = len(active_indices)
    A = np.zeros((K, n))
    for ii, k in enumerate(active_indices):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[ii, j] = 2.0 * a[k - j]
    return A


def solve_lp_direction(a, conv, active_indices):
    """
    Solve LP to find direction d:
    min t s.t. A_active @ d <= t - R[active], sum(d)=0, d >= -a
    Returns direction d (or None if infeasible).
    """
    n = len(a)
    K = len(active_indices)
    if K == 0:
        return None

    A = build_lp_matrices(a, active_indices)
    # Objective: minimize t (last variable)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    # Inequality: A @ d - t <= -conv[active_indices]
    A_ub = np.hstack([A, -np.ones((K, 1))])
    b_ub = -conv[active_indices]
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    # Bounds: d_i >= -a_i, t free (but bounded for numerical safety)
    t_lower = -float(conv.max()) - 1.0
    bounds = [(-a_i, None) for a_i in a] + [(t_lower, None)]

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True})
        if result.success:
            d = result.x[:n]
            return d
    except Exception:
        pass
    return None


def line_search(a, d, current_peak, max_iter=15):
    """Backtracking line search along direction d from a."""
    best_alpha = 0.0
    best_a = a.copy()
    best_peak = current_peak
    alpha = 1.0
    s = a.sum()
    for _ in range(max_iter):
        a_try = a + alpha * d
        if np.any(a_try < -1e-12):
            alpha *= 0.5
            continue
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            alpha *= 0.5
            continue
        a_try = a_try / s_try
        conv_try = fast_convolve(a_try)
        peak_try = conv_try.max()
        if peak_try < best_peak - 1e-14:
            best_peak = peak_try
            best_a = a_try.copy()
            best_alpha = alpha
        alpha *= 0.5
        if alpha < 1e-6:
            break
    return best_a, best_alpha, best_peak


def peak_shave(a, conv, n_peaks=3, strength=0.1):
    """Direct peak shaving: reduce elements contributing to top peaks."""
    n = len(a)
    a_new = a.copy()
    s = a.sum()
    for _ in range(n_peaks):
        k = int(np.argmax(conv))
        # conv[k] = sum_i a[i]*a[k-i]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        contrib = a[idxs] * a[k - idxs]
        total_contrib = contrib.sum()
        if total_contrib < 1e-15:
            break
        # Reduce a at positions with high contribution, compensate elsewhere
        weights = contrib / total_contrib
        # Reduce peak contributors
        reduction = strength * total_contrib
        for j in idxs:
            reduce_amt = weights[j - j_start] * reduction * a[j]
            a_new[j] -= reduce_amt
        # Compensate at low-contribution positions
        mask = np.ones(n, dtype=bool)
        # Find positions NOT in the peak region
        for j in idxs:
            mask[j] = False
        candidates = np.where(mask)[0]
        if len(candidates) == 0:
            continue
        # Distribute compensation proportional to current value
        comp_weights = a_new[candidates]
        comp_total = comp_weights.sum()
        if comp_total < 1e-15:
            comp_weights = np.ones(len(candidates))
            comp_total = len(candidates)
        else:
            comp_weights = comp_weights / comp_total
        a_new[candidates] += reduction * comp_weights
    a_new = np.maximum(a_new, 0.0)
    sn = a_new.sum()
    if sn > 1e-15:
        a_new = a_new / sn
    return a_new


def sqp_optimize(a0, time_limit, verbose=False):
    """SQP-like optimization: LP direction + line search."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_peak = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    K = 20  # Active set size

    while time.monotonic() - t0 < time_limit:
        # Select active set: top K peaks
        top_idx = np.argpartition(conv, -K)[-K:]
        top_idx = top_idx[np.argsort(conv[top_idx])[::-1]]

        # Solve LP for direction
        d = solve_lp_direction(a, conv, top_idx)

        if d is not None:
            a_new, alpha, new_peak = line_search(a, d, best_peak)
            if alpha > 0 and new_peak < best_peak - 1e-14:
                a = a_new
                conv = fast_convolve(a)
                best_peak = conv.max()
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        iteration += 1

        # Adaptive: increase K when stuck
        if no_improve == 10:
            K = min(50, K + 5)
        elif no_improve == 30:
            K = 5
            # Try gradient step
            temp = 0.001 * max(best_peak, 1e-10)
            grad = smooth_gradient_fft(a, conv, temp)
            # Project gradient step onto simplex tangent
            grad_tan = grad - grad.mean()
            lr = 0.01
            a_try = simplex_project(a + lr * grad_tan)
            conv_try = fast_convolve(a_try)
            if conv_try.max() < best_peak - 1e-14:
                a = a_try
                conv = conv_try
                best_peak = conv.max()
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        # Perturb when very stuck
        if no_improve >= 60:
            rng = np.random.default_rng(iteration * 31 + 7)
            scale = 0.002 + 0.001 * min(no_improve // 60, 5)
            noise = rng.standard_normal(n) * scale
            a = simplex_project(best_a + noise)
            conv = fast_convolve(a)
            best_peak = conv.max()
            if best_peak > best_a is not None and conv.max() < best_peak:
                pass
            no_improve = 0
            K = 20
            if no_improve > 200:
                break

        if verbose and iteration % 200 == 0:
            print(f"    SQP iter {iteration}, peak={best_peak:.10f}, obj={2*n*best_peak:.8f}, t={time.monotonic()-t0:.1f}s")

    return best_a, 2.0 * n * best_peak


def gradient_optimize(a0, time_limit, lr_init=0.01, momentum=0.9, temp_init=0.002):
    """Nesterov-accelerated smooth gradient descent."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    y = a.copy()
    conv = fast_convolve(a)
    best_peak = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        y = a + mom * v
        y = simplex_project(np.maximum(y, 0))
        conv_y = fast_convolve(y)
        grad = smooth_gradient_fft(y, conv_y, temp)
        # Tangential gradient (project to sum-0 subspace)
        grad = grad - grad.mean()

        v = mom * v - lr * grad
        a_new = a + v
        a = simplex_project(a_new)

        conv = fast_convolve(a)
        cur_peak = conv.max()

        if cur_peak < best_peak - 1e-14:
            best_peak = cur_peak
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9997
        temp *= 1.005
        mom = min(0.97, mom + 0.0003)

        if no_improve >= 300:
            break

    return best_a, 2.0 * n * best_peak

def micro_step(a, conv, n):
    """Try a few micro-steps: reduce peak element's contributions."""
    best_a = a.copy()
    best_peak = conv.max()
    k_max = int(np.argmax(conv))
    j_start = max(0, k_max - n + 1)
    j_end = min(n - 1, k_max)

    # Try reducing each element in the peak region slightly
    for j in range(j_start, j_end + 1):
        partner = k_max - j
        if partner < 0 or partner >= n:
            continue
        # Reduce a[j] and increase a at a "safe" position
        for delta in [0.001, 0.005, 0.01]:
            if a[j] < delta:
                continue
            # Find a safe position: one that doesn't contribute to top peaks
            a_try = a.copy()
            a_try[j] -= delta
            # Add to a position far from the peak
            safe_idx = (j + n // 2) % n
            a_try[safe_idx] += delta
            if np.any(a_try < 0):
                continue
            s_try = a_try.sum()
            if abs(s_try - 1.0) > 1e-10:
                a_try = a_try / s_try
            conv_try = fast_convolve(a_try)
            if conv_try.max() < best_peak - 1e-12:
                best_a = a_try.copy()
                best_peak = conv_try.max()
                a = a_try
                conv = conv_try
                break

    return best_a, best_peak

    print(f"  Phase 4 done. Best: {best_score:.8f}")

    # =========================================================
    # Phase 5: Peak shaving fine-tuning
    # =========================================================
    print("=== Phase 5: Peak Shaving ===")
    if best_sequence is not None:
        t5 = min(60.0, deadline - time.monotonic() - 50)
        if t5 > 10:
            try_optimize(best_sequence, t5, method='shave')
    print(f"  Phase 5 done. Best: {best_score:.8f}")

    # =========================================================
    # Phase 6: Final micro-optimization
    # =========================================================
    print("=== Phase 6: Micro Steps ===")
    if best_sequence is not None and time.monotonic() < deadline - 20:
        a = best_sequence.copy()
        conv = fast_convolve(a)
        n = len(a)
        cur_peak = conv.max()
        t6_end = time.monotonic() + min(40.0, deadline - time.monotonic() - 10)
        while time.monotonic() < t6_end:
            a, cur_peak = micro_step(a, conv, n)
            conv = fast_convolve(a)
            obj = 2.0 * n * cur_peak
            if obj < best_score - 1e-10:
                best_score = obj
                best_sequence = a.copy()
    print(f"  Phase 6 done. Best: {best_score:.8f}")

    # =========================================================
    # Phase 7: Final SQP polish with remaining time
    # =========================================================
    if best_sequence is not None and time.monotonic() < deadline - 15:
        t7 = min(30.0, deadline - time.monotonic() - 5)
        if t7 > 5:
            a_opt, obj = sqp_optimize(best_sequence, time_limit=t7)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final SQP: {score:.8f} *NEW BEST*")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        # Should not happen, but ensure we return something valid
        best_sequence = np.ones(256)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
