def fast_convolve(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size-1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A*A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.ones(n)/n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def coord_descent_optimal(a, i):
    """Find optimal value of a[i] given all other coordinates fixed.
    Returns the optimal x >= 0."""
    n = len(a)
    x_old = a[i]
    a[i] = 0.0
    conv0 = fast_convolve(a)  # convolution with a[i]=0
    S0 = a.sum()
    a[i] = x_old  # restore
    
    if S0 < 1e-15:
        return x_old
    
    # new_conv[k] = conv0[k] + 2*x*a0[k-i] + x^2 * delta(k, 2i)
    # where a0 = a with a[i]=0
    a0 = a.copy()
    a0[i] = 0.0
    
    # For each k, the coefficient of x is 2*a0[k-i] (if valid)
    # and the constant is conv0[k]
    # Special: at k=2i, there's also an x^2 term
    
    c = np.zeros(2*n-1)  # coefficient of x
    for k in range(2*n-1):
        if 0 <= k - i < n:
            c[k] = 2.0 * a0[k-i]
    
    # Candidate x values
    candidates = [0.0]
    
    # Critical points from linear pieces: x = S0 - 2*b_j/m_j where m_j = c[j], b_j = conv0[j]
    for k in range(2*n-1):

            x_crit = S0 - 2.0 * conv0[k] / c[k]
            if x_crit > 0:
                candidates.append(x_crit)

    k2 = 2*i
    if k2 < 2*n-1:
        x_q = conv0[k2] / S0
        if x_q > 0:
            candidates.append(x_q)
    
    # Also add the current value and a few others
    candidates.append(x_old)
    candidates.append(S0 / n)  # uniform level
    candidates.append(S0)  # all mass here
    
    # Evaluate objective at each candidate
    best_x = 0.0
    best_obj = float('inf')
    
    for x in candidates:
        if x < 0:
            x = 0.0
        new_conv = conv0 + c * x
        new_conv[k2] += x * x  # add the x^2 term
        obj = new_conv.max() / (x + S0)**2
        if obj < best_obj:
            best_obj = obj
            best_x = x

def coordinate_descent(a, max_sweeps, time_limit, verbose=False):
    """Run coordinate descent for many sweeps."""
    n = len(a)
    a = a.copy().astype(np.float64)
    t0 = time.monotonic()
    
    conv = fast_convolve(a)
    S = a.sum()
    best_obj = conv.max() / S**2
    best_a = a.copy()
    no_improve = 0

            if obj < best_obj - 1e-14:

                best_a = a.copy()

        else:
            no_improve += 1
        
        if no_improve > 3:
            break
        
        if verbose and sweep % 10 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}")
    
    return best_a, 2*n*best_obj
