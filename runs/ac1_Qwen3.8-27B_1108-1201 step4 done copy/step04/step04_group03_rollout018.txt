def soft_max_gradient_descent(a0, time_limit, t_init=50, t_decay=1.001, lr_init=0.01):
    a = a0.copy()
    a = a / a.sum()
    n = len(a)
    t0 = time.monotonic()
    
    t = t_init
    lr = lr_init
    best_obj = float('inf')
    best_a = a.copy()
    
    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        M = c.max()
        
        if M < best_obj:
            best_obj = M
            best_a = a.copy()
        
        # Soft-max gradient
        shifted = t * c
        shifted -= shifted.max()
        w = np.exp(shifted)
        w /= w.sum()
        
        # Gradient
        grad = compute_grad(w, a)
        
        # Step
        a_new = a - lr * grad
        a_new = project_simplex(a_new)
        
        # Check improvement
        c_new = fast_convolve(a_new)
        if c_new.max() < M:
            a = a_new
        else:
            # Try smaller step
            a_try = a - lr * 0.5 * grad
            a_try = project_simplex(a_try)
            c_try = fast_convolve(a_try)
            if c_try.max() < M:
                a = a_try
                lr *= 0.5
            else:
                lr *= 0.5
        
        t *= t_decay
        
        if lr < 1e-8:
            lr = lr_init * 0.5
            t = t_init * 2
    
    return best_a, best_obj

        a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    t = t_init
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(a)
        M = c.max()

        if M < best_obj - 1e-15:
            best_obj = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Softmax weights on convolution values
        shifted = t * c
        shifted -= shifted.max()
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        else:
            w /= ws

        grad = compute_gradient(w, a)

        # Adaptive step: try the step, if it doesn't improve, halve
        a_try = a - lr * grad
        a_try = project_simplex(a_try)
        c_try = fast_convolve(a_try)
        obj_try = c_try.max()

        if obj_try < M - 1e-16:
            a = a_try
            # Occasionally increase lr slightly
            if no_improve == 0 and lr < lr_init:
                lr = min(lr * 1.02, lr_init)
        else:
            # Try smaller steps
            improved = False
            for factor in [0.5, 0.25, 0.1]:
                a_try2 = a - lr * factor * grad
                a_try2 = project_simplex(a_try2)
                c_try2 = fast_convolve(a_try2)
                if c_try2.max() < M - 1e-16:
                    a = a_try2
                    lr = lr * factor
                    improved = True
                    break
            if not improved:
                lr *= 0.5

        # Adaptive temperature
        t *= 1.003
        if t > 500.0:
            t = 500.0

        # Restart if stuck
        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 1000000)
            scale = 0.002 + 0.005 * min(1.0, no_improve / 400.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            lr = lr_init * 0.7
            t = t_init
            no_improve = 0

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SoftMaxGD iter {it}, obj={M:.8f}, best={best_obj:.8f}, t={t:.1f}, lr={lr:.6f}, elapsed={elapsed:.0f}s")

    return best_a, best_obj


def peak_reduction(a0, time_limit, verbose=False):
    """Iteratively reduce the highest peak by scaling down contributing elements."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(a)
        M = c.max()
        k_max = int(np.argmax(c))

        if M < best_obj - 1e-15:
            best_obj = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Find elements contributing most to the peak at k_max
        # c[k_max] = sum_j a[j] * a[k_max - j]
        # Contribution of a[j] to c[k_max] is a[j] * a[k_max - j]
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        contrib = a[js:je + 1] * a[k_max - je:k_max - js + 1][::-1]

        # Scale down the top contributors
        top_k = max(5, n // 10)
        top_idx_local = np.argpartition(contrib, -top_k)[:top_k]
        top_idx_global = js + top_idx_local

        # Reduce top contributors, increase others slightly to maintain sum
        reduction = 0.001 + 0.002 * min(1.0, no_improve / 100.0)
        a_new = a.copy()
        a_new[top_idx_global] *= (1.0 - reduction)

        # Redistribute to non-top elements
        deficit = a[top_idx_global].sum() - a_new[top_idx_global].sum()
        mask = np.ones(n, dtype=bool)
        mask[top_idx_global] = False
        if mask.sum() > 0:
            a_new[mask] += deficit / mask.sum()

        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()

        c_new = fast_convolve(a_new)
        if c_new.max() < M - 1e-16:
            a = a_new
        else:
            # Try even smaller reduction
            for r2 in [0.0005, 0.0002, 0.0001]:
                a_try = a.copy()
                a_try[top_idx_global] *= (1.0 - r2)
                deficit2 = a[top_idx_global].sum() - a_try[top_idx_global].sum()
                a_try[mask] += deficit2 / mask.sum()
                a_try = np.maximum(a_try, 0.0)
                a_try = a_try / a_try.sum()
                if fast_convolve(a_try).max() < M - 1e-16:
                    a = a_try
                    break
            else:
                # No improvement, try attacking second-highest peak
                c_sorted = np.sort(c)[::-1]
                if len(c_sorted) > 1 and c_sorted[1] > c_sorted[0] * 0.95:
                    # Multiple similar peaks - try averaging approach
                    pass

        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 1000000)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            no_improve = 0

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakRed iter {it}, obj={M:.8f}, best={best_obj:.8f}, elapsed={elapsed:.0f}s")

    return best_a, best_obj


def spectral_equalize(a0, time_limit, verbose=False):
    """Try to equalize the power spectrum to flatten the convolution."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = 0.01

    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(a)
        M = c.max()

        if M < best_obj - 1e-15:
            best_obj = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Compute power spectrum
        fft_size = 1 << (2 * n - 2).bit_length()
        A = np.fft.fft(a, fft_size)
        P = np.abs(A) ** 2

        # Target: flat spectrum (all bins equal)
        # Gradient of variance(P) with respect to a:
        # P[k] = |A[k]|^2 = A[k]*conj(A[k])
        # dP[k]/da[j] = 2*Re(A[k] * conj(dA[k]/da[j]))
        # A[k] = sum_j a[j] * exp(-2πijk/N)
        # dA[k]/da[j] = exp(-2πijk/N)
        # dP[k]/da[j] = 2*Re(A[k] * exp(2πijk/N)) = 2*Re(A[k] * conj(exp(-2πijk/N)))
        
        # We want to minimize max(c) ≈ related to variance of P
        # Use gradient of sum(P[k]^2) as proxy for flatness
        # d(sum P^2)/da[j] = 2*sum_k P[k] * dP[k]/da[j]
        
        # Simpler: use the soft-max gradient approach but in frequency domain
        # Actually, let's just use the time-domain gradient but weight by spectral info
        
        # Alternative: directly try to reduce variance of c
        c_mean = c.mean()
        c_var = ((c - c_mean) ** 2).sum()
        
        # Gradient of c_var w.r.t. a[j]:
        # c[k] = sum_i a[i]*a[k-i]
        # dc[k]/da[j] = 2*a[k-j]
        # d(c_var)/da[j] = 2*sum_k (c[k]-c_mean) * 2*a[k-j]
        
        shifted_c = c - c_mean
        grad = 2.0 * compute_gradient(shifted_c, a)
        
        a_try = a - lr * grad
        a_try = project_simplex(a_try)
        c_try = fast_convolve(a_try)
        if c_try.max() < M - 1e-16:
            a = a_try
        else:
            a_try2 = a - lr * 0.5 * grad
            a_try2 = project_simplex(a_try2)
            c_try2 = fast_convolve(a_try2)
            if c_try2.max() < M - 1e-16:
                a = a_try2
                lr *= 0.7
            else:
                lr *= 0.5

        if lr < 1e-6:
            lr = 0.01

        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 1000000)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            lr = 0.01
            no_improve = 0

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SpectralEq iter {it}, obj={M:.8f}, best={best_obj:.8f}, elapsed={elapsed:.0f}s")

    return best_a, best_obj


def coord_descent(a0, time_limit, verbose=False):
    """Coordinate descent: optimize one element at a time with line search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            c = fast_convolve(a)
            M = c.max()

            # Try changing a[j] while adjusting another element to keep sum=1
            # Simply: try a[j] * (1+delta) and renormalize
            for delta in [0.01, -0.01, 0.02, -0.02, 0.005, -0.005]:
                a_try = a.copy()
                a_try[j] *= (1.0 + delta)
                a_try = np.maximum(a_try, 0.0)
                ts = a_try.sum()
                if ts < 1e-15:
                    continue
                a_try = a_try / ts
                c_try = fast_convolve(a_try)
                if c_try.max() < M - 1e-16:
                    a = a_try
                    improved = True
                    break

            if a.sum() > 0 and fast_convolve(a).max() < best_obj - 1e-15:
                best_obj = fast_convolve(a).max()
                best_a = a.copy()

        if not improved:
            # Try larger perturbations
            rng = np.random.default_rng(it)
            for _ in range(10):
                j = rng.integers(n)
                k = rng.integers(n)
                if j == k:
                    continue
                a_try = a.copy()
                a_try[j] += 0.001 * a.max()
                a_try[k] -= 0.001 * a.max()
                a_try = np.maximum(a_try, 0.0)
                ts = a_try.sum()
                if ts < 1e-15:
                    continue
                a_try = a_try / ts
                c_try = fast_convolve(a_try)
                M = fast_convolve(a).max()
                if c_try.max() < M - 1e-16:
                    a = a_try
                    improved = True
                    break

    return best_a, best_obj


def multi_peak_sg(a0, time_limit, top_k=20, lr_init=0.005, mom=0.9, verbose=False):
    """Multi-peak subgradient with momentum, targeting multiple peaks simultaneously."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    v = np.zeros(n)

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(a)
        M = c.max()

        if M < best_obj - 1e-15:
            best_obj = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Weight multiple peaks with rank-based weights
        if top_k >= len(c):
            top_k = len(c)
        peaks = np.argpartition(c, -top_k)[-top_k:]
        pv = c[peaks]
        # Sharper weighting for top peaks
        order = np.argsort(pv)[::-1]
        weights = np.array([1.0 / (rank + 1) ** 2.0 for rank in range(len(peaks))])
        weights /= weights.sum()

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += weights[i] * 2.0 * a[k - idx]

        v = mom * v - lr * sg
        a_new = a + v
        a_new = project_simplex(a_new)

        c_new = fast_convolve(a_new)
        if c_new.max() < M - 1e-16:
            a = a_new
        else:
            # Try without momentum
            a_try = project_simplex(a - lr * sg)
            c_try = fast_convolve(a_try)
            if c_try.max() < M - 1e-16:
                a = a_try
                v = np.zeros(n)
            else:
                lr *= 0.8
                v = 0.5 * v

        lr = min(lr * 0.9998, lr_init)
        if lr < 1e-7:
            lr = lr_init * 0.5

        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 1000000)
            scale = 0.002 + 0.004 * min(1.0, no_improve / 600.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            lr = lr_init * max(0.3, 1.0 - no_improve / 1000.0)
            no_improve = 0

        if verbose and it % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MultiPeakSG iter {it}, obj={M:.8f}, best={best_obj:.8f}, elapsed={elapsed:.0f}s")

    return best_a, best_obj


def annealing_search(a0, time_limit, n_restarts=20, verbose=False):
    """Simulated annealing style search with periodic large jumps."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(a)
        M = c.max()

        if M < best_obj - 1e-15:
            best_obj = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Temperature based on iteration
        progress = (time.monotonic() - t0) / time_limit
        temp = 0.01 * (1.0 - progress) ** 2 + 0.001

        # Generate perturbation
        rng = np.random.default_rng(it * 7 + int(time.monotonic() * 100))
        
        if no_improve > 100:
            # Larger jump
            scale = temp * 2
            a_try = project_simplex(best_a + rng.standard_normal(n) * scale)
        else:
            # Small step toward reducing peaks
            peaks = np.argpartition(c, -5)[-5:]
            for k in peaks:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                a_try = a - 0.002 * a[k - idx]
                a_try[0] += 0.002  # maintain sum approximately
                break
            a_try = project_simplex(a_try)

        c_try = fast_convolve(a_try)
        obj_try = c_try.max()

        # Metropolis accept
        if obj_try < M:
            a = a_try
        else:
            # Accept with probability exp(-(obj_try - M) / temp_obj)
            delta = obj_try - M
            if delta < 0.01 and rng.random() < np.exp(-delta / (temp * 0.01)):
                a = a_try

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-strategy search combining soft-max GD, peak reduction,
    spectral equalization, coordinate descent, and multi-peak subgradient.
    Tries multiple sequence lengths and uses diverse initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    # === Phase 1: Exploration (~200s) ===
    print("\n=== Phase 1: Exploration ===")

    # Start with parent and try a few variations
    candidates = []
    if parent is not None:
        candidates.append((parent, len(parent)))
    
    # Try different lengths
    for n_try in [384, 512, 256, 640]:
        if parent is not None and n_try == len(parent):
            continue
        # Interpolate parent to new length or use structured init
        if parent is not None:
            px = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, len(parent)), parent)
            px = np.maximum(px, 0)
            ps = px.sum()
            if ps > 1e-10:
                candidates.append((px / ps, n_try))
        else:
            candidates.append((np.ones(n_try) / n_try, n_try))

    # Allocate time for quick optimization of each candidate

        # Quick soft-max GD

        if rem > 5:
            for attempt in range(10):
                rem = deadline - time.monotonic() - 5
                if rem < 2:
                    break
                scale = 0.0005 / (attempt + 1)
                a_pert = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * scale)
                try_seq(a_pert, f"micro_{attempt}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
