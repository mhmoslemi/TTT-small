"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using ramp-based initializations
combined with LP Frank-Wolfe, active-set refinement, and local search.
Key insight: monotonic (ramp) sequences avoid the Cauchy-Schwarz lower bound
on the central auto-correlation, achieving much better objectives than
symmetric sequences.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, conv_f, top_k=50):
    """Solve reduced LP to find descent direction.
    min t s.t. t >= conv_f[k] + 2*(B*g)[k] for top_k peaks,
    g >= -f, sum(g) = 0.
    Returns direction g or None."""
    n = len(f)
    
    # Select top_k peaks
    if top_k >= len(conv_f):
        peaks = np.arange(len(conv_f))
    else:
        peaks = np.argpartition(conv_f, -top_k)[-top_k:]
    
    # Build constraint matrix for only selected peaks
    # (B*g)[k] = sum_j f[k-j] * g[j] for valid j
    # For each peak k, the constraint is: -t + sum_j f[k-j]*g[j] <= -conv_f[k]/2
    # (dividing by 2 for numerical convenience)
    
    c = np.zeros(n + 1)
    c[0] = 1.0  # minimize t
    
    A_ub = np.zeros((len(peaks), n + 1))
    A_ub[:, 0] = -1.0
    b_ub = -conv_f[peaks] / 2.0
    
    for idx, k in enumerate(peaks):
        # (B*g)[k] = sum_j g[j] * f[k-j] where 0<=j<n and 0<=k-j<n
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A_ub[idx, j + 1] = 2.0 * f[k - j]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]
    return None


def lf_wolfe(f0, time_limit, top_k=30, verbose=True):
    """LP-based Frank-Wolfe with reduced constraints."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(f)
        current_obj = conv.max()
        
        g = solve_lp_direction(f, conv, top_k=top_k)
        if g is None:
            no_improve += 5
            if no_improve > 50:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                no_improve = 0
            continue
        
        target = f + g
        target = np.maximum(target, 0.0)
        ts = target.sum()
        if ts < 1e-15:
            no_improve += 1
            continue
        target = target / ts
        
        if np.max(np.abs(target - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                no_improve = 0
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = current_obj
        for alpha in np.linspace(0.05, 1.0, 10):
            f_new = (1.0 - alpha) * f + alpha * target
            f_new = np.maximum(f_new, 0)
            fs = f_new.sum()
            if fs < 1e-15:
                continue
            f_new = f_new / fs
            val = fast_convolve(f_new).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            # Refine
            lo = max(0.0, best_alpha * 0.7)
            hi = min(1.0, best_alpha * 1.3)
            for _ in range(6):
                mid = (lo + hi) / 2
                f_new = (1.0 - mid) * f + mid * target
                f_new = np.maximum(f_new, 0)
                fs = f_new.sum()
                if fs < 1e-15:
                    break
                f_new = f_new / fs
                val = fast_convolve(f_new).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * target
            f = np.maximum(f, 0)
            f = f / f.sum()
            
            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"    LF iter {it}: obj={best_obj:.10f} (t={elapsed:.1f}s)")
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 60:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            no_improve = 0
    
    return best_f, best_obj


def active_set_refine(f0, time_limit, tol=1e-10):
    """KKT-based active set refinement.
    At optimum: for active set A, sum_{k in A} lambda_k * grad_k = mu * 1 (constant).
    grad_k[j] = 2*f[(k-j) mod structure]"""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, fast_convolve(f).max()
    f = f / s
    n = len(f)
    
    t0 = time.monotonic()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    rng = np.random.default_rng(77)
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        current_max = conv.max()
        
        # Find active set (peaks within tolerance of max)
        threshold = current_max - max(1e-8, current_max * 1e-6)
        active = np.where(conv >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(conv)])
        if len(active) > 50:
            # Keep only top 50
            top_idx = np.argsort(conv[active])[::-1][:50]
            active = active[top_idx]
        
        # Solve KKT: find lambda >= 0, sum(lambda)=1 such that
        # sum_k lambda[k] * 2*f[k-j] = mu for all j
        # This is: M @ lambda = mu * 1, where M[j,k] = 2*f[k-j] (for valid)
        m = len(active)
        M = np.zeros((n, m))
        for idx, k in enumerate(active):
            j_start = max(0, k - n + 1)
            j_end = min(n, k + 1)
            for j in range(j_start, j_end):
                M[j, idx] = 2.0 * f[k - j]
        
        # We want: M @ lambda = mu * 1, lambda >= 0, sum(lambda) = 1
        # This is an underdetermined system (n equations, m+1 unknowns)
        # Use least squares with non-negativity
        
        # Actually, let's formulate as: minimize ||M@lambda - mu*1||^2 s.t. lambda >= 0, sum=1
        # For fixed mu, this is a QP. Let's use a simpler approach:
        # Find lambda that minimizes ||M@lambda||_inf - mean, s.t. lambda >= 0, sum = 1
        # Or: use the LP: min t s.t. -t <= M@lambda - mu <= t, lambda >= 0, sum(lambda)=1
        # But mu is also unknown...
        
        # Simpler approach: iteratively adjust f towards equalizing the active peaks
        # For each active peak k, compute "force" on f: f_new[j] should reduce conv[k]
        # conv[k] = sum_i f[i]*f[k-i], d(conv[k])/d(f[j]) = 2*f[k-j]
        # To reduce conv[k], decrease f[j] where f[k-j] is large
        
        # Water-filling: redistribute mass from "expensive" positions to "cheap" ones
        # Cost of f[j] = sum_k lambda[k] * 2*f[k-j]
        
        # Use the non-negative least squares approach
        from scipy.optimize import nnls
        # Minimize ||M@lambda - 1||^2 (target constant = 1 for simplicity)
        # s.t. lambda >= 0
        # Then normalize
        target = np.ones(n)
        lam, _ = nnls(M, target)
        lam_sum = lam.sum()
        if lam_sum < 1e-15:
            lam = np.ones(m) / m
        else:
            lam = lam / lam_sum
        
        # Compute the gradient at current f with these lambda weights
        grad = M @ lam  # This is sum_k lam[k] * 2*f[k-j]
        
        # Descent direction: decrease f where grad is high, increase where low
        # Projected gradient: f_new = project_simplex(f - lr * grad)
        lr = 0.01
        f_new = project_simplex(f - lr * grad)
        
        new_obj = fast_convolve(f_new).max()
        if new_obj < best_obj - 1e-12:
            best_obj = new_obj
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            # Try smaller step
            lr = 0.005
            f_new = project_simplex(f - lr * grad)
            new_obj = fast_convolve(f_new).max()
            if new_obj < best_obj - 1e-12:
                best_obj = new_obj
                best_f = f_new.copy()
                f = f_new
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 20:
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                    no_improve = 0
        
        if no_improve > 50:
            break
    
    return best_f, best_obj


def local_search(f0, time_limit, rng):
    """Structured local search: coordinate descent + random perturbations."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        # Try coordinate descent on a few random coordinates
        for _ in range(min(200, n // 4)):
            if time.monotonic() - t0 > time_limit:
                break
            j = rng.integers(0, n)
            # Try moving mass from j to j+1 and j-1
            for dj in [-2, -1, 1, 2]:
                j2 = j + dj
                if j2 < 0 or j2 >= n:
                    continue
                # Move small amount of mass
                for frac in [0.01, 0.005, 0.001]:
                    amount = f[j] * frac
                    if amount < 1e-15:
                        continue
                    f_temp = f.copy()
                    f_temp[j] -= amount
                    f_temp[j2] += amount
                    if f_temp[j] < 0:
                        continue
                    f_temp = f_temp / f_temp.sum()
                    obj = fast_convolve(f_temp).max()
                    if obj < best_obj - 1e-12:
                        best_obj = obj
                        best_f = f_temp.copy()
                        f = f_temp.copy()
                        improved = True
                        break
        
        # Random perturbation
        if not improved:
            scale = 0.003 * (1 + 0.1 * (time.monotonic() - t0) / max(time_limit, 1))
            f_pert = project_simplex(f + rng.standard_normal(n) * scale)
            obj = fast_convolve(f_pert).max()
            if obj < best_obj:
                best_obj = obj
                best_f = f_pert.copy()
                f = f_pert.copy()
    
    return best_f, best_obj


def generate_ramp_inits(n, rng):
    """Generate diverse ramp-like initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Linear ramp
    a = x + 1
    inits.append(("ramp_lin", a / a.sum()))
    
    # Power ramps
    for p in [0.5, 1.5, 2.0, 3.0]:
        a = (x + 1) ** p
        inits.append((f"ramp_p{p}", a / a.sum()))
    
    # Exponential ramps
    for alpha in [0.01, 0.02, 0.05]:
        a = np.exp(alpha * x)
        inits.append((f"ramp_exp{alpha}", a / a.sum()))
    
    # Reversed ramps (same by symmetry of the problem? No, not quite)
    # Actually the problem is not symmetric under reversal because
    # conv(a, a) is the same for a and reversed(a).
    # So reversed ramp gives same objective. Skip.
    
    # Sigmoid ramps
    for center in [0.3, 0.5]:
        a = 1.0 / (1.0 + np.exp(-10 * (x / n - center)))
        inits.append((f"ramp_sig{center}", a / a.sum()))
    
    # Quadratic with different curvatures
    for c in [0.3, 0.5, 0.7]:
        a = (x - c * n) ** 2 + 1  # Always positive
        inits.append((f"ramp_quad{c}", a / a.sum()))
    
    # Linear with noise
    a = (x + 1) * (1 + 0.1 * rng.standard_normal(n))
    a = np.maximum(a, 0)
    inits.append(("ramp_noisy", a / a.sum()))
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Minimizes 2*n*max(conv(a,a))/(sum(a))^2.
    Key insight: monotonic (ramp) sequences achieve much better objectives
    than symmetric sequences by avoiding the Cauchy-Schwarz bound on central
    auto-correlation. Uses LP Frank-Wolfe, active-set refinement, and
    structured local search starting from ramp initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
    except Exception:

    print("=== Phase 1: Initialization ===")

    if parent is not None:
        if len(parent) in lengths:
            lengths = [len(parent)] + [l for l in lengths if l != len(parent)]
        else:
            lengths.insert(0, len(parent))
    
    candidates = []
    for n in lengths:
        if time.monotonic() > deadline - 600:
            break
        for label, init in generate_ramp_inits(n, rng):
            candidates.append((n, label, init))
    
    # Add parent
    if parent is not None:
        candidates.append((len(parent), "parent", parent))
    
    # Evaluate all candidates

        if time.monotonic() > deadline - 500:
