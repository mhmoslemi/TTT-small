import numpy as np

def fast_convolve(a):
    """Full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_toeplitz(a):
    """Build matrix C such that C @ g = conv(a, g)."""
    n = len(a)
    size = 2 * n - 1
    C = np.zeros((size, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def lse_gradient(a, conv, tau):
    """Compute gradient of LSE-smoothed max of conv(a,a) w.r.t. a using FFT."""
    n = len(a)
    size = len(conv)

    scaled = conv / tau
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(size)
        p[np.argmax(conv)] = 1.0
    else:
        p /= p_sum

    # cross[i] = sum_m p[i+m] * a[m] via FFT cross-correlation
    fft_size = 1 << (size + n - 1 - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a, fft_size)
    cross_circ = np.fft.irfft(np.conj(A) * P, fft_size)
    cross = cross_circ[:n]

    return 2.0 * cross


def lse_gradient_direct(a, conv, tau, top_k=50):
    """Compute gradient using top-k peaks for efficiency."""
    n = len(a)
    size = len(conv)

    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    scaled = peak_vals / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum

    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        # grad[i] += 2 * w[idx] * a[k-i] for valid i
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min > i_max:
            continue
        idxs = np.arange(i_min, i_max + 1)
        grad[idxs] += 2.0 * w[idx] * a[k - idxs]

    return grad


def obj_value(a):
    """Compute the objective 2*n*max(conv)/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def lse_optimize(a0, time_limit, tau_init=0.01, lr_init=0.01, momentum=0.9,
                 top_k=50, decay_period=100, tau_decay=0.995):
    """LSE-smoothed gradient descent with Nesterov acceleration."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    a = a / a.sum()

    v = np.zeros(n)
    y = a.copy()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    tau = tau_init
    mom = momentum
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        # Nesterov lookahead
        y = a + mom * v
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        else:
            y = a.copy()

        conv_y = fast_convolve(y)
        grad = lse_gradient_direct(y, conv_y, tau, top_k=top_k)

        # Gradient step
        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
            break

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        iter_count += 1

        # Adaptive updates
        if iter_count % decay_period == 0:
            tau *= tau_decay
            lr *= 0.998
            mom = min(0.97, mom + 0.0003)

        if no_improve >= 300:
            break

    return best_a, best_obj

def lp_solve_direction(a, M, C=None):
    """Solve LP: max sum(g) s.t. C@g <= M, g >= 0."""
    n = len(a)
    if C is None:
        C = build_conv_toeplitz(a)
    c_vec = -np.ones(n)
    b_vec = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_vec, A_ub=C, b_ub=b_vec, bounds=bounds,
                         method='highs',
                         options={'presolve': True,
                                  'dual_feasibility_tolerance': 1e-9,
                                  'primal_feasibility_tolerance': 1e-9})
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g, g_sum, True
    except Exception:
        pass
    return a.copy(), a.sum(), False


def lp_step(a, conv, C=None):
    """One LP-based optimization step with bisection step sizing."""
    n = len(a)
    M = conv.max()

    if C is None:
        C = build_conv_toeplitz(a)

    # Allow slightly larger M to give some room
    g, g_sum, success = lp_solve_direction(a, M * 1.00001, C)

    if not success or g_sum <= 1.0 + 1e-8:
        return a.copy(), False, 0.0

    d = g - a
    norm_d = np.linalg.norm(d)
    if norm_d < 1e-12:
        return a.copy(), False, 0.0

    # Bisection to find best step size
    best_a = a.copy()
    best_obj = 2.0 * n * M
    best_t = 0.0

    t_lo, t_hi = 0.0, 1.0
    # First check if t=1.0 is good
    a_full = a + d
    a_full = np.maximum(a_full, 0.0)
    s_full = a_full.sum()
    if s_full > 1e-15:
        a_full = a_full / s_full
        conv_full = fast_convolve(a_full)
        obj_full = 2.0 * n * conv_full.max()
        if obj_full < best_obj:
            best_obj = obj_full
            best_a = a_full.copy()
            best_t = 1.0
            t_hi = 1.0
        else:
            t_hi = 0.5

    if best_t == 0.0:
        # Bisection search
        for _ in range(12):
            t_mid = 0.5 * (t_lo + t_hi)
            a_try = a + t_mid * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t_hi = t_mid
                continue
            a_try = a_try / s_try
            conv_try = fast_convolve(a_try)
            obj_try = 2.0 * n * conv_try.max()
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                best_t = t_mid
                t_lo = t_mid
            else:
                t_hi = t_mid

    if best_t > 0:
        return best_a, True, best_obj
    return a.copy(), False, 0.0


def lp_optimize(a0, time_limit, verbose=False):
    """LP-based iterative optimization with adaptive strategies."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:

    no_improve = 0
    perturb_count = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        # Rebuild C when a changes significantly
        a_new, improved, new_obj = lp_step(a, conv, C)

        if improved:
            a = a_new
            conv = fast_convolve(a)
            C = build_conv_toeplitz(a)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj - 1e-14:
                best_obj = cur_obj
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        iteration += 1

        if verbose and iteration % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iteration}, obj={best_obj:.9f}, t={elapsed:.1f}s")

        if no_improve >= 20:
            # Perturb and restart
            scale = 0.002 + 0.003 * min(perturb_count, 10)
            noise = rng.standard_normal(n) * scale
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            else:
                a = best_a.copy()
            conv = fast_convolve(a)

            # If too many perturbations, try gradient for a bit

                    best_obj = 2.0 * n * conv_temp.max()
                    best_a = a_temp.copy()
                    a = a_temp.copy()
                    conv = conv_temp

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Sine-based
    for freq in [1, 2, 3]:
        sig = (np.sin(np.pi * freq * x / n) ** 2 + 0.01)
        inits[f'sine{freq}'] = sig / sig.sum()

    # Bimodal
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.15 * n)) ** 2)
    inits['gaussian'] = g / g.sum()

    # From parent
    if parent is not None:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps

    # Perturbed parent
    if 'parent' in inits:
        for s in [0.005, 0.01, 0.02]:
            noise = rng.standard_normal(n) * s
            pert = np.maximum(inits['parent'] + noise, 0)
            pers = pert.sum()
            if pers > 1e-10:
                inits[f'parent_n{s}'] = pert / pers

    return inits


def spectral_reshape(a, n_freqs=5, power=2.0):
    """Reshape spectrum to be more peaked (concentrated at low freqs)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    A = np.fft.rfft(a, fft_size)
    S = np.abs(A) ** 2

    # Amplify low frequencies, suppress high
    n_half = fft_size // 2
    for i in range(n_half):
        frac = i / n_half
        S[i] *= (1.0 + power * (1.0 - frac) ** 2)

    # Reconstruct phase
    A_new = np.sqrt(S) * np.exp(1j * np.angle(A))
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = a.copy()

    return a_new

                print(f"  LP {i} (n={n}): {score:.7f} *NEW BEST*")
            else:
                print(f"  LP {i} (n={n}): {score:.7f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.7f}")

    # =========================================================
    # Phase 3: LSE-FFT gradient polish (300s)
    # =========================================================
    print("=== Phase 3: LSE-FFT Gradient Polish ===")
    t_start3 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 200.0
        if remaining > 50:
            # Multiple runs with different parameters
            configs = [
                {'tau_init': 0.003, 'lr_init': 0.005, 'momentum': 0.93},
                {'tau_init': 0.002, 'lr_init': 0.003, 'momentum': 0.95},
                {'tau_init': 0.005, 'lr_init': 0.008, 'momentum': 0.90},
            ]
            for ci, cfg in enumerate(configs):
                if time.monotonic() > deadline - 150:
                    break
                t_alloc = min(remaining / (len(configs) - ci) * 0.85,
                              deadline - time.monotonic() - 120)
                if t_alloc < 10:
                    break

                a0 = best_sequence.copy()
                if ci > 0:
                    noise_scale = 0.001 * ci
                    noise = rng.standard_normal(len(best_sequence)) * noise_scale
                    a0 = np.maximum(a0 + noise, 0)
                    s = a0.sum()
                    if s > 1e-15:
                        a0 = a0 / s

                a_opt, obj = lse_optimize_fft(a0, time_limit=t_alloc, **cfg)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score):
                    if score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  LSE-FFT {ci}: {score:.7f} *NEW BEST*")
                    else:
                        print(f"  LSE-FFT {ci}: {score:.7f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.7f}")

    # =========================================================
    # Phase 4: Final LP micro-polish + spectral reshape (200s)
    # =========================================================
    print("=== Phase 4: Final Micro-Optimization ===")
    t_start4 = time.monotonic()
    if best_sequence is not None:
        # Try spectral reshape
        remaining = deadline - time.monotonic() - 80.0
        if remaining > 30:
            a_spec = spectral_reshape(best_sequence, n_freqs=3, power=1.5)
            # Quick gradient refinement
            a_opt, obj = lse_optimize(a_spec, time_limit=min(20.0, remaining * 0.3),
                                       tau_init=0.003, lr_init=0.004, top_k=60)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Spectral: {score:.7f} *NEW BEST*")

        # Final LP
        remaining = deadline - time.monotonic() - 30.0
        if remaining > 20:
            a_opt, obj = lp_optimize(best_sequence, time_limit=remaining * 0.8)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final LP: {score:.7f} *NEW BEST*")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.7f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.7f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
