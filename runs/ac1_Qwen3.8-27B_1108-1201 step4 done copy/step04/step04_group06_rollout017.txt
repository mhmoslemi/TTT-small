import numpy as np

from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def compute_grad(a, conv, top_k, temp):
    """Compute softmax-weighted gradient. d conv[k]/da[j] = 2*a[k-j]."""
    n = len(a)
    m = len(conv)
    if top_k >= m:
        peaks = np.arange(m)
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]

    pv = conv[peaks]
    sc = pv / temp
    sc = sc - sc.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w = w / ws

    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-14:
            continue
        k = int(peaks[idx])
        wi = 2.0 * w[idx]
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        idxs = np.arange(j_lo, j_hi + 1)
        offsets = k - idxs
        grad[idxs] += wi * a[offsets]
    return grad

    A_ub[:, n] = -1.0

    t0 = time.monotonic()
    while time.monotonic() - t0 < t_limit:
        conv = fast_convolve(a)
        g = compute_grad(a, conv, topk, temp)
        v = mom * v - lr * g
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / sn
        cn = fast_convolve(a_new)
        on = 2.0 * n * cn.max()
        if on < best_obj - 1e-13:
            a = a_new
            best_obj = on
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 20:
                for al in [0.5, 0.25, 0.1]:
                    at = a - al * lr * g
                    at = np.maximum(at, 0.0)
                    st = at.sum()
                    if st < 1e-15:
                        continue
                    at = at / st
                    ct = fast_convolve(at)
                    ot = 2.0 * n * ct.max()
                    if ot < best_obj - 1e-13:
                        a = at
                        best_obj = ot
                        best_a = a.copy()
                        no_imp = 0
                        v *= al
                        break
                else:
                    if no_imp > patience:
                        rng = np.random.default_rng(int(time.monotonic() * 1e6))
                        a = best_a + rng.standard_normal(n) * 0.003
                        a = np.maximum(a, 0)
                        s2 = a.sum()
                        if s2 > 1e-15:
                            a = a / s2
                        v = np.zeros(n)
                        no_imp = 0
                        lr = lr0 * 0.6
                        temp = temp0
        lr *= 0.9992
        temp *= temp_mult
        mom = min(0.96, mom + 0.00008)
        if no_imp > patience * 4:
            break
    return best_a, best_obj


def spectral_init(n, n_iter=200, seed=0):
    """Gerchberg-Saxton style spectral initialization."""
    rng = np.random.default_rng(seed)
    a = rng.dirichlet(np.ones(n) * 0.5)
    L = 1 << (2 * n - 1).bit_length()
    target_mag = 1.0 / np.sqrt(L)
    for _ in range(n_iter):
        ap = np.zeros(L)
        ap[:n] = a
        A = np.fft.fft(ap)
        ang = np.angle(A)
        A_new = target_mag * np.exp(1j * ang)
        a_new = np.fft.ifft(A_new).real[:n]
        a = np.maximum(a_new, 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s
    return a


def full_optimize(a0, t_limit):
    """Combined LP + gradient optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_imp = 0
    use_lp = True
    while time.monotonic() - t0 < t_limit:
        if use_lp:
            a_new, imp = lp_step(a, conv)
            if imp:
                a = a_new
                conv = fast_convolve(a)
                obj = 2.0 * n * conv.max()
                if obj < best_obj - 1e-13:
                    best_obj = obj
                    best_a = a.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp > 40:
                    use_lp = False
        else:
            t_rem = t_limit - (time.monotonic() - t0)
            if t_rem < 1:
                break
            t_grad = min(5.0, t_rem * 0.3)
            a_opt, obj_opt = softmax_descend(best_a, t_grad,
                                              lr0=0.01, temp0=0.001,
                                              temp_mult=1.01, topk=60,
                                              mom0=0.9, patience=60)
            if obj_opt < best_obj - 1e-13:
                best_obj = obj_opt
                best_a = a_opt.copy()
                a = a_opt.copy()
                conv = fast_convolve(a)
            no_imp = 0
            use_lp = True
    return best_a, best_obj


def coordinate_refine(a0, t_limit):
    """Random coordinate perturbation refinement."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * len(a) * fast_convolve(a).max()
    a = a / s
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    no_imp = 0
    while time.monotonic() - t0 < t_limit:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        best_delta = 0.0
        best_val = best_obj
        for delta in [0.002, 0.005, 0.01, 0.02, -0.002, -0.005, -0.01, -0.02]:
            if a[i] + delta < 0 or a[j] - delta < 0:
                continue
            a_try = a.copy()
            a_try[i] += delta
            a_try[j] -= delta
            cv = fast_convolve(a_try)
            ob = 2.0 * n * cv.max()
            if ob < best_val:
                best_val = ob
                best_delta = delta
        if best_delta != 0.0:
            a[i] += best_delta
            a[j] -= best_delta
            conv = fast_convolve(a)
            best_obj = best_val
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 500:
                break
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase optimization:
    Phase 1: Diverse initialization + quick gradient scan across lengths
    Phase 2: Deep LP+gradient optimization on best candidates
    Phase 3: Multi-restart simulated annealing
    Phase 4: Final coordinate + gradient polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: n={len(p)}")
    except Exception:
        pass

    # Phase 1: Quick scan
    print("=== Phase 1: Quick Scan ===")
    t1 = time.monotonic()
    p1_limit = min(250.0, (deadline - time.monotonic()) * 0.2)
    lengths = [256, 512, 768, 1024]
    results = []

    for n in lengths:
        if time.monotonic() > deadline - 800:
            break
        inits = {}
        x = np.arange(n, dtype=np.float64)
        inits['uniform'] = np.ones(n) / n
        inits['sine'] = (np.sin(np.pi * x / n) ** 2 + 0.01)
        inits['sine'] /= inits['sine'].sum()
        inits['bimodal'] = np.zeros(n)
        edge = max(1, n // 4)
        inits['bimodal'][:edge] = 1.0
        inits['bimodal'][-edge:] = 1.0
        inits['bimodal'] /= inits['bimodal'].sum()
        inits['spec1'] = spectral_init(n, 150, seed=1)
        inits['spec2'] = spectral_init(n, 150, seed=42)
        inits['alternating'] = np.ones(n)
        inits['alternating'][::2] = 1.8
        inits['alternating'][1::2] = 0.4
        inits['alternating'] /= inits['alternating'].sum()
        if parent is not None:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent / parent.sum())
            pp = np.maximum(pp, 0)
            ss = pp.sum()
            if ss > 1e-10:
                inits['parent'] = pp / ss

        for name, a0 in inits.items():
            if time.monotonic() > deadline - 700:
                break
            t_al = min(15.0, p1_limit - (time.monotonic() - t1) + 5)
            if t_al < 3:
                break
            a_opt, obj = softmax_descend(a0, t_al, lr0=0.02, temp0=0.005,
                                         temp_mult=1.01, topk=50, mom0=0.85, patience=80)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  P1 n={n} {name}: {score:.6f}")

    results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t1:.0f}s). Best: {best_score:.6f}")

    # Phase 2: Deep optimization
    print("=== Phase 2: Deep Optimization ===")
    t2 = time.monotonic()
    top = []
    seen = set()
    for sc, nn, nm, ao in results:
        if nn not in seen:
            top.append((nn, ao))
            seen.add(nn)
        if len(top) >= 3:
            break
    if best_sequence is not None and len(best_sequence) not in seen:
        top.append((len(best_sequence), best_sequence.copy()))

    remaining = deadline - time.monotonic() - 400
    t_per = max(30.0, remaining * 0.7 / max(1, len(top)))
    for i, (nn, a0) in enumerate(top):
        if time.monotonic() > deadline - 350:
            break
        t_al = min(t_per, deadline - time.monotonic() - 300)
        if t_al < 10:
            break
        print(f"  P2 cand {i}: n={nn}, t={t_al:.0f}s")
        a_opt, obj = full_optimize(a0, t_al)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P2 {i}: {score:.6f} *NEW*")
            else:
                print(f"  P2 {i}: {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t2:.0f}s). Best: {best_score:.6f}")

    # Phase 3: Multi-restart refinement
    print("=== Phase 3: Multi-Restart ===")
    t3 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 150
        n_restarts = max(2, int(remaining / 40))
        for ri in range(min(n_restarts, 8)):
            if time.monotonic() > deadline - 100:
                break
            t_al = min(30.0, (deadline - time.monotonic() - 80) / max(1, n_restarts - ri))
            if t_al < 5:
                break
            if ri == 0:
                a0 = best_sequence.copy()
            else:
                scale = 0.002 * (ri + 1)
                noise = rng.standard_normal(len(best_sequence)) * scale
                a0 = np.maximum(best_sequence + noise, 0)
                s = a0.sum()
                if s > 1e-15:
                    a0 = a0 / s
                else:
                    a0 = best_sequence.copy()
            a_opt, obj = full_optimize(a0, t_al)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P3 restart {ri}: {score:.6f} *NEW*")

    print(f"Phase 3 done ({time.monotonic()-t3:.0f}s). Best: {best_score:.6f}")

    # Phase 4: Final polish
    print("=== Phase 4: Final Polish ===")
    t4 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10
        if remaining > 10:
            t_cd = min(remaining * 0.4, 30.0)
            if t_cd > 3:
                a_opt, obj = coordinate_refine(best_sequence, t_cd)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  CoordDesc: {score:.6f} *NEW*")
            remaining = deadline - time.monotonic() - 5
            if remaining > 5:
                for gi in range(3):
                    if time.monotonic() > deadline - 3:
                        break
                    t_sg = min(remaining / (3 - gi) * 0.7, 20.0)
                    if t_sg < 3:
                        break
                    cfgs = [
                        (0.004, 0.0005, 1.005, 80, 0.93),
                        (0.002, 0.0002, 1.003, 100, 0.95),
                        (0.008, 0.001, 1.008, 50, 0.90),
                    ]
                    lr_g, temp_g, tm, tk, mo = cfgs[gi % 3]
                    a_opt, obj = softmax_descend(best_sequence, t_sg,
                                                  lr0=lr_g, temp0=temp_g,
                                                  temp_mult=tm, topk=tk,
                                                  mom0=mo, patience=100)
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  SG {gi}: {score:.6f} *NEW*")

    print(f"Phase 4 done ({time.monotonic()-t4:.0f}s). Best: {best_score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
