The uniform part: (u*u)[k] = (k+1)/n^2 for k < n, (2n-1-k)/n^2 for k >= n. Peak at k=n-1: 1/n.

The cross term: (2*u*d)[k] = 2/n * d[k] (approximately, since u is uniform). More precisely, (u*d)[k] = (1/n) * sum_j d[j] (for the overlapping part). This is approximately (1/n) * sum_j d[j] = 0 (since the sum of a cosine over a full period is 0). So the cross term is small.

The d*d term: (d*d)[k] = (alpha/n)^2 * sum_j cos(2*pi*m*j/n + phi) * cos(2*pi*m*(k-j)/n + phi)
= (alpha/n)^2 * (1/2) * sum_j [cos(2*pi*m*k/n) + cos(2*pi*m*(2j-k)/n + 2*phi)]
≈ (alpha/(2n^2)) * n * cos(2*pi*m*k/n) + small
= (alpha/(2n)) * cos(2*pi*m*k/n) + small

So c[k] ≈ (k+1)/n^2 + (alpha/(2n)) * cos(2*pi*m*k/n) (for k < n)

The peak of this is at k = n-1: 1/n + (alpha/(2n)) * cos(2*pi*m*(n-1)/n) = 1/n + (alpha/(2n)) * cos(2*pi*m - 2*pi*m/n)

For m=1: cos(2*pi - 2*pi/n) ≈ cos(0) = 1 (for large n). So the peak increases!
For m=n/2: cos(2*pi*(n/2)*(n-1)/n) = cos(pi*(n-1)) = (-1)^(n-1). For n even, this is -1. So the peak decreases by alpha/(2n).

So using m = n/2 (Nyquist frequency) with positive alpha reduces the center peak!

But wait, we also need a[j] >= 0. With a[j] = (1 + alpha*cos(pi*j))/n = (1 + alpha*(-1)^j)/n, we need 1 + alpha*(-1)^j >= 0 for all j, which requires alpha <= 1.

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive subgradient descent
with Adam optimizer, temperature annealing, structured perturbations,
coordinate descent polishing, and multi-length exploration.

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgrad(a, conv, T, top_k):
    n = len(a)
    k_act = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_act)[-k_act:]
    pv = conv[peaks].copy()
    scaled = pv / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * a[k - idx]
    return sg, peaks, w

def adam_subgrad_optimize(a0, time_limit, verbose=True, seed=42):

    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)

    else:
        a = a / s
    n = len(a)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.01
    
    # Temperature
    T = 50.0
    T_min = 10.0
    T_max = 500.0
    
    top_k = max(5, n // 10)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    step_count = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        step_count += 1
        
        # Compute subgradient
        sg, peaks, w = compute_subgrad(a, conv, T, top_k)
        
        # Peak normalization: subtract mean to make it zero-sum (for simplex projection)
        sg_mean = sg.mean()
        sg_centered = sg - sg_mean
        
        # Adam update
        m = beta1 * m + (1 - beta1) * sg_centered
        v = beta2 * v + (1 - beta2) * sg_centered ** 2

        v_hat = v / (1 - beta2 ** t)

            best_a = a_new.copy()

            a = a_new
            conv = conv_new
        else:

            a = a_new  # still update to keep momentum going

def coordinate_descent_optimize(a0, time_limit, verbose=True):
    """Faster coordinate descent with adaptive multipliers."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    pass_count = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-10:
                continue
            
            best_j = old_val
            best_j_obj = best_obj
            
            # Try a range of multipliers
            for mult in [0.5, 0.7, 0.9, 1.1, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j = new_val
            
            if best_j != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j) / denom
                    a[j] = best_j
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
                    conv = fast_convolve(a)
        
        if not improved and pass_count > 2:
            break
    
    return best_a, best_obj

    no_improve = 0
    it = 0
    m_exp = 1.0
    
    # Precompute the constraint matrix structure
    # (f*g)[k] = sum_j f[j] * g[k-j] (with appropriate bounds)
    # This is a banded matrix
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Build LP
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):

        b_ub = np.full(2 * n - 1, M * m_exp)
        bounds = [(0, None)] * n
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            m_exp *= 1.1
            if m_exp > 5.0:
                m_exp = 1.0
            if no_improve > 20:
                break
            no_improve += 1
            continue
        
        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue
        
        # Fast line search: try a few alphas
        best_alpha = 0.0
        best_val = M
        for alpha in [0.5, 0.7, 0.85, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f = f + best_alpha * (g - f)
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 50 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}, time={elapsed:.1f}s")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 4.0)
    
    return best_f, best_obj

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgrad_weights(conv, T, top_k):
    """Compute softmax weights over top-k peaks."""
    n_conv = len(conv)
    k_act = min(top_k, n_conv)
    peaks = np.argpartition(conv, -k_act)[-k_act:]
    pv = conv[peaks].copy()
    scaled = pv / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    return peaks, w


def apply_subgrad(a, peaks, w):
    """Apply weighted subgradient: sg[j] = 2 * sum_i w[i] * a[k_i - j]."""
    n = len(a)
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * a[k - idx]
    return sg


def nesterov_subgrad_optimize(a0, time_limit, verbose=True, seed=42,
                               T_init=50.0, lr_init=0.02, top_k_frac=0.1):
    """Nesterov-accelerated subgradient descent with BB step size."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    top_k = max(5, int(n * top_k_frac))
    T = T_init

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    x_prev = a.copy()
    t_k = 1.0
    lr = lr_init

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    # BB step size history
    delta_prev = None
    grad_prev = None

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov extrapolation
        mu = (t_k - 1) / (t_k + 2)
        y = a + mu * (a - x_prev)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys < 1e-15:
            y = best_a.copy()
            ys = 1.0
        y = y / ys

        # Compute subgradient at y
        conv_y = fast_convolve(y)
        peaks, w = compute_subgrad_weights(conv_y, T, top_k)
        sg = apply_subgrad(y, peaks, w)
        # Center for simplex constraint
        sg_centered = sg - sg.mean()

        # BB adaptive step size
        if grad_prev is not None and delta_prev is not None:
            num = np.dot(delta_prev, grad_prev - sg_centered)
            den = np.dot(grad_prev - sg_centered, grad_prev - sg_centered)
            if den > 1e-30:
                bb_lr = num / den
                bb_lr = np.clip(bb_lr, lr * 0.5, lr * 2.0)

        # Gradient step
        x_prev = a.copy()
        a_new = y - lr * sg_centered
        a_new = project_simplex(a_new)

        # Evaluate
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        delta_prev = a_new - a
        grad_prev = sg_centered.copy()
        a = a_new
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive temperature and restart
        if no_improve > 200:
            T = min(T * 1.5, 800.0)
            lr = max(lr * 0.85, 0.001)
            if no_improve > 800:
                # Structured perturbation
                pert_type = rng.integers(0, 4)
                if pert_type == 0:
                    freq = rng.integers(1, n // 2 + 1)
                    phase = rng.uniform(0, 2 * np.pi)
                    pert = np.cos(2 * np.pi * freq * np.arange(n) / n + phase) * 0.015
                elif pert_type == 1:
                    pert = (np.arange(n) / n - 0.5) ** 2 * 0.03
                elif pert_type == 2:
                    # Two-frequency perturbation
                    f1, f2 = rng.integers(1, n // 4), rng.integers(n // 4 + 1, n // 2 + 1)
                    pert = (np.cos(2 * np.pi * f1 * np.arange(n) / n) * 0.008 +
                           np.cos(2 * np.pi * f2 * np.arange(n) / n) * 0.008)
                else:
                    pert = rng.standard_normal(n) * 0.006
                a = project_simplex(best_a + pert)
                x_prev = a.copy()
                conv = fast_convolve(a)
                t_k = 1.0
                lr = lr_init * 0.5
                T = T_init * 2
                no_improve = 0
                delta_prev = None
                grad_prev = None

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0

def lp_fw_optimize(f0, time_limit, verbose=False, top_peak_constraint=True):
    """LP-based Frank-Wolfe with optional constraint to top peaks only."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Select constraints: all or top peaks only
        if top_peak_constraint:
            n_constraints = min(2 * n - 1, max(n, (2 * n - 1) // 2))
            conv_full = fast_convolve(f)
            top_idx = np.argpartition(conv_full, -n_constraints)[-n_constraints:]
            A_ub = np.zeros((n_constraints, n))
            b_ub = np.zeros(n_constraints)
            for ii, k in enumerate(top_idx):
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                A_ub[ii, js:je + 1] = f[je:js - 1:-1] if js > 0 else f[je + 1][::-1]
                b_ub[ii] = conv_full[k] * m_exp
        else:
            A_ub = np.zeros((2 * n - 1, n))
            for j in range(n):
                A_ub[j:j + n, j] = f
            b_ub = np.full(2 * n - 1, M * m_exp)
            n_constraints = 2 * n - 1

        c = -np.ones(n)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

        if not res.success:
            m_exp *= 1.08
            if m_exp > 4.0:
                m_exp = 1.0
            no_improve += 1
            if no_improve > 30:
                break
            continue

        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 25:
                break
            m_exp = min(m_exp * 1.03, 3.0)
            continue

        # Line search: try a few alphas
        best_alpha = 0.0
        best_val = fast_convolve(f).max()
        for alpha in [0.3, 0.6, 0.8, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = f + best_alpha * (g - f)
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 30 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}, time={elapsed:.1f}s")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.06, 3.5)

    return best_f, best_obj


def coordinate_descent_fine(a0, time_limit, verbose=False):
    """Coordinate descent with finer multiplier grid and early exit."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    pass_count = 0

    multipliers = [0.6, 0.75, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue

            best_j = old_val
            best_j_obj = conv.max()

            denom = 1.0 - old_val
            if denom < 1e-10:
                continue

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()

                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j = new_val

            if best_j != old_val:
                scale = (1.0 - best_j) / denom
                a[j] = best_j
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                conv = fast_convolve(a)

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved and pass_count > 3:
            break
        if verbose and pass_count % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass {pass_count}: obj={best_obj:.10f}, time={elapsed:.1f}s")

    return best_a, best_obj


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

                f_interp, time_limit=t_alloc * 0.5, verbose=False, seed=seed + tn)
            eval_and_update(a_opt, f"nag_n{tn}")

            rem = deadline - time.monotonic() - 60.0
            if rem > 25:
                a_opt, _ = lp_fw_optimize(a_opt, time_limit=min(rem, t_alloc * 0.45),
                                           verbose=False, top_peak_constraint=True)
                eval_and_update(a_opt, f"lpfw_n{tn}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Short Nesterov
        rem = deadline - time.monotonic() - 30.0
        if rem > 15:
            a_opt, _ = nesterov_subgrad_optimize(
                best_seq, time_limit=min(rem, 50), verbose=False,
                seed=seed + 999, T_init=30.0, lr_init=0.008, top_k_frac=0.2)
            eval_and_update(a_opt, "final_nag")

        # Short LP-FW
        rem = deadline - time.monotonic() - 15.0
        if rem > 10:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem, 35),
                                       verbose=False, top_peak_constraint=True)
            eval_and_update(a_opt, "final_lpfw")

        # Short CD
        rem = deadline - time.monotonic() - 10.0
        if rem > 8:
            a_opt, _ = coordinate_descent_fine(best_seq, time_limit=min(rem, 20))
            eval_and_update(a_opt, "final_cd")

        # Perturbations
        for _ in range(4):
            rem = deadline - time.monotonic() - 8.0
            if rem < 6:
                break
            noise = rng.standard_normal(n_best) * 0.003
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = nesterov_subgrad_optimize(
                f_pert, time_limit=min(rem, 12), verbose=False,
                seed=seed + 700 + _, T_init=40.0, lr_init=0.01, top_k_frac=0.15)
            eval_and_update(a_opt, "perturb")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
