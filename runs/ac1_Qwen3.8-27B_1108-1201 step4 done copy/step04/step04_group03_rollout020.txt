"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an improved multi-phase approach:

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_lp_matrix(f):
    """Build the constraint matrix for the LP: (f*g)[k] <= M for all k."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A

def solve_lp(f, M, A_cache=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    if A_cache is None:
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            A[j:j + n, j] = f
    else:
        A = A_cache
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def peak_chase_with_line_search(f0, time_limit, lr_init=0.005, mom=0.93, verbose=True):
    """Peak-chasing subgradient with momentum and line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break

        conv = fast_convolve(f)
        M = conv.max()

        # Find top peaks
        top_k = max(5, min(n // 10, len(conv)))
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        
        # Sharper weighting on top peaks
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 2.0
        total_w = weights.sum()
        weights /= total_w

        # Subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += weights[i] * 2.0 * f[k - idx]

        v = mom * v - cur_lr * sg
        f_new = f + v
        f_new = project_simplex(f_new)
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * f_new
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            f = (1.0 - best_alpha) * f + best_alpha * f_new
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()

        if M < best_obj - 1e-15:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9995

        if no_improve > 100:
            rng = np.random.default_rng(int(time.monotonic() * 1000000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr_init * 0.7, 0.0005)
            no_improve = 0

        if verbose and it % 100 == 0:
            print(f"    PG iter {it}: M={M:.10f}, best={best_obj:.10f}")

    return best_f, best_obj


def multiplicative_weight_refine(f0, time_limit, lr_init=0.03, top_k=20, temp=0.0004, verbose=True):
    """Multiplicative weight method for positive sequence optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_temp = temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break

        conv = fast_convolve(f)
        M = conv.max()

        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Multiplicative update
        log_update = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_contrib = 2.0 * w[i] * f[k - idx]
            log_update[idx] -= cur_lr * grad_contrib

        f = f * np.exp(log_update)
        f = np.maximum(f, 1e-25)
        f = f / f.sum()

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9997
        cur_temp *= 1.003

        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 1000000) % 999983)
            f = best_f * np.exp(rng.standard_normal(n) * 0.003)
            f = f / f.sum()
            cur_lr = max(lr_init * 0.5, 0.005)
            cur_temp = temp * 1.5
            no_improve = 0

        if verbose and it % 100 == 0:
            print(f"    MW iter {it}: M={obj:.10f}, best={best_obj:.10f}")

    return best_f, best_obj

def try_symmetrize(f):
    """Try symmetrizing the sequence and see if it helps."""
    f_sym = (f + f[::-1]) / 2.0
    f_sym = np.maximum(f_sym, 0.0)
    s = f_sym.sum()
    if s < 1e-15:
        return f
    f_sym = f_sym / s
    obj_orig = fast_convolve(f).max()
    obj_sym = fast_convolve(f_sym).max()
    if obj_sym <= obj_orig:
        return f_sym
    return f

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase optimization:
1. FW-LP with adaptive M and warm-restart strategy
2. Nesterov-accelerated smoothed gradient (log-sum-exp smoothing)
3. Fine subgradient polish with EMA peak memory
4. Structured restarts (shifts, mirrors, ramps) to escape local minima
Key improvements: Better line search, peak memory for adaptive weighting,
Nesterov acceleration on smoothed objective, and structured restarts.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f):
    """Build the (2n-1) x n matrix A where (A * g)[k] = (f*g)[k]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_for_direction(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A = build_conv_matrix(f)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def line_search(f, g, M, n_iters_coarse=12, n_iters_fine=10):
    """Adaptive line search: coarse then fine bisection."""
    best_alpha = 0.0
    best_val = M

    # Coarse: logarithmically spaced alphas
    coarse_alphas = np.unique(np.concatenate([
        np.linspace(0.01, 0.5, n_iters_coarse),
        np.array([0.6, 0.7, 0.8, 0.9, 1.0])
    ]))

    for alpha in coarse_alphas:
        ft = (1.0 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0.0)
        st = ft.sum()
        if st < 1e-15:
            continue
        ft = ft / st
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha

    if best_alpha <= 0:
        return 0.0, M

    # Fine: bisection around best
    lo = max(0.0, best_alpha - 0.05)
    hi = min(1.0, best_alpha + 0.05)
    for _ in range(n_iters_fine):
        mid = (lo + hi) / 2.0
        ft = (1.0 - mid) * f + mid * g
        ft = np.maximum(ft, 0.0)
        st = ft.sum()
        if st < 1e-15:
            break
        ft = ft / st
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = mid
            lo = mid
        else:
            hi = mid

    return best_alpha, best_val


def fw_lp_optimize(f0, time_limit, verbose=True):
    """Frank-Wolfe with LP oracle, adaptive M, and structured restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_factor = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3.0:
            break

        g = solve_lp_for_direction(f, M * m_factor)
        if g is None:
            m_factor *= 1.12
            if m_factor > 8.0:
                m_factor = 1.0
                no_improve += 10
                continue
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 5
            m_factor = min(m_factor * 1.04, 6.0)
            if no_improve > 150:
                break
            continue

        best_alpha, best_val = line_search(f, g, M, n_iters_coarse=12, n_iters_fine=10)

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_factor = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_factor = min(m_factor * 1.06, 6.0)

        # Structured restart when stuck
        if no_improve > 100:
            rng = np.random.default_rng(it * 7919 + 13)
            restart_type = it % 4
            if restart_type == 0:
                # Random noise
                noise = rng.standard_normal(n) * 0.004
                f = project_simplex(best_f + noise)
            elif restart_type == 1:
                # Small circular shift
                shift = rng.integers(1, max(2, n // 50))
                f = np.roll(best_f, shift)
                f = project_simplex(f)
            elif restart_type == 2:
                # Add small ramp
                x = np.linspace(0, 1, n)
                ramp = x * rng.uniform(0.005, 0.02) * rng.choice([1, -1])
                f = project_simplex(best_f + ramp)
            else:
                # Add small sinusoid
                freq = rng.uniform(1.0, 4.0)
                phase = rng.uniform(0, 2 * np.pi)
                sine = np.sin(2 * np.pi * freq * np.arange(n) / n + phase) * 0.003
                f = project_simplex(best_f + sine)

            conv = fast_convolve(f)
            M = conv.max()
            m_factor = 1.0
            no_improve = 0

        if verbose and it % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW iter {it}: M={M:.10f}, best={best_obj:.10f}, no_imp={no_improve}, t={elapsed:.0f}s")

    return best_f, best_obj


def nesterov_smoothed(f0, time_limit, tau_init=1.0, tau_end=0.1, verbose=True):
    """
    Nesterov-accelerated gradient on log-sum-exp smoothed objective.
    Minimize (1/tau)*log(sum exp(tau*c[k])) which approximates max(c).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    # Nesterov state
    y = f.copy()
    t_val = 1.0
    f_prev = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        # Adaptive tau: start coarse, end fine
        progress = min(1.0, it / 5000.0)
        tau = tau_init * (tau_end / tau_init) ** progress

        # Compute smoothed gradient at y
        conv_y = fast_convolve(y)
        scaled = tau * conv_y
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros_like(conv_y)
            p[np.argmax(conv_y)] = 1.0
        else:
            p /= p_sum

        # grad[j] = (2/tau) * sum_k p[k] * a[k-j] for valid indices
        # = (2/tau) * np.correlate(p, y, mode='full')[:n]
        corr = np.correlate(p, y, mode='full')
        grad = (2.0 / tau) * corr[:n]

        # Lipschitz constant upper bound: L ~ 2*n/tau
        L = 2.0 * n / tau
        lr = 1.0 / L

        # Nesterov update
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        beta = (t_val - 1.0) / t_next

        # Gradient step from y
        f_new = y - lr * grad
        f_new = project_simplex(f_new)

        # Extrapolation
        y_new = f_new + beta * (f_new - f_prev)
        y_new = project_simplex(y_new)

        f_prev = f_new
        f = f_new
        y = y_new
        t_val = t_next

        # Track best
        conv_f = fast_convolve(f)
        obj = conv_f.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            y = f.copy()
            f_prev = f.copy()
            t_val = 1.0
            no_improve = 0

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Nesterov iter {it}: obj={obj:.10f}, best={best_obj:.10f}, tau={tau:.4f}, t={elapsed:.0f}s")

    return best_f, best_obj


def fine_polish_peak_memory(f0, time_limit, lr_init=0.002, mom=0.94, top_k=30,
                            temp_init=0.0002, verbose=True):
    """
    Fine subgradient polish with EMA peak memory.
    Tracks which peaks have been consistently high and weights them more.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv_len = 2 * n - 1

    # Peak memory: EMA of normalized peak heights
    peak_memory = np.zeros(conv_len)
    conv = fast_convolve(f)
    peak_memory = conv / conv.max()

    v = np.zeros(n)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_temp = temp_init
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        conv = fast_convolve(f)
        M = conv.max()

        # Update peak memory
        peak_memory = 0.95 * peak_memory + 0.05 * (conv / M)

        # Select top-k peaks using combined weight (current + memory)
        combined = conv / M + 0.5 * peak_memory
        if top_k >= conv_len:
            peaks = np.arange(conv_len)
        else:
            peaks = np.argpartition(combined, -top_k)[-top_k:]

        # Softmax weights with temperature
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += w[i] * 2.0 * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)

        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9998
        cur_temp *= 1.002

        if no_improve > 400:
            rng = np.random.default_rng(int(time.monotonic() * 1000000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            cur_lr = max(lr_init * 0.6, 0.0003)
            no_improve = 0

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Polish iter {it}: obj={co:.10f}, best={best_obj:.10f}, t={elapsed:.0f}s")

    return best_f, best_obj


def try_symmetrize(f):
    """Try symmetrizing and return the better version."""
    f_sym = (f + f[::-1]).astype(np.float64) / 2.0
    f_sym = np.maximum(f_sym, 0.0)
    s = f_sym.sum()
    if s < 1e-15:
        return f
    f_sym = f_sym / s
    obj_orig = fast_convolve(f).max()
    obj_sym = fast_convolve(f_sym).max()
    if obj_sym <= obj_orig + 1e-15:
        return f_sym
    return f


def evaluate_normalized(a):
    """Compute the objective for a normalized non-negative sequence."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a_norm = a / s
    conv = np.convolve(a_norm, a_norm)
    return float(2 * len(a_norm) * conv.max() / (a_norm.sum() ** 2))
