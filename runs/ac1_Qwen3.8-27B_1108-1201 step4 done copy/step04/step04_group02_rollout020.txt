import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp(f, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

    else:

    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= wsum
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    
    return sg

def golden_section_search(f, a, b, func, max_iter=40):
    """Golden section search for minimum of func on [a, b]."""
    gr = (np.sqrt(5) - 1) / 2
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = func(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = func(d)
    return (a + b) / 2

def nesterov_sg(f0, time_limit, initial_lr=0.01, top_k=128, temp=0.001):
    """Nesterov-accelerated subgradient descent."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    y = f.copy()  # Nesterov's lookahead point
    conv = fast_convolve(y)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    cur_lr = initial_lr
    cur_temp = temp
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break
        
        # Compute subgradient at y
        conv_y = fast_convolve(y)
        sg = compute_subgradient(y, conv_y, top_k=top_k, temp=cur_temp)
        
        # Take step from f (not y)
        y_new = project_simplex(f - cur_lr * sg)
        
        # Nesterov update
        t_k = (1 + np.sqrt(1 + 4 * (it + 1)**2)) / 2
        beta = (t_k - 1) / t_k
        f_new = beta * f + (1 - beta) * y_new
        
        f = f_new
        y = project_simplex(f - cur_lr * sg)
        
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
        
        # Adaptive parameters
        cur_lr *= 0.9995
        cur_temp *= 1.005
        
        # Periodic restart
        if it % 500 == 0:
            rng = np.random.default_rng(it * 31 + 7)
            noise = rng.standard_normal(n) * 0.003
            f_try = project_simplex(best_f + noise)
            conv_try = fast_convolve(f_try)
            if conv_try.max() < best_obj:
                f = f_try
                y = f_try.copy()
                best_obj = conv_try.max()
                best_f = f_try.copy()
            cur_lr = initial_lr * 0.5
            cur_temp = temp * 0.5
    
    return best_f, best_obj

def pairwise_cd(f0, time_limit, n_pairs_per_pass=100, n_passes=50):
    """Pairwise coordinate descent polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        # Pick random pairs
        for _ in range(n_pairs_per_pass):
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            
            # Try moving mass from i to j
            # a[i] -> a[i] - delta, a[j] -> a[j] + delta
            # Objective is a function of delta
            
            def obj_at_delta(delta):
                if delta < -f[i] or delta > -f[j] or f[i] - delta < 0 or f[j] + delta < 0:
                    return np.inf
                f_try = f.copy()
                f_try[i] -= delta
                f_try[j] += delta
                f_try = np.maximum(f_try, 0)
                f_try = f_try / f_try.sum()
                return fast_convolve(f_try).max()
            
            # Find best delta via golden section
            # Range: delta in [-f[i], -f[j]] but also need f[i]-delta >= 0 and f[j]+delta >= 0
            # So delta in [-f[j], f[i]]
            lo = -f[j] + 1e-10
            hi = f[i] - 1e-10
            if lo >= hi:
                continue
            
            best_delta = golden_section_search(f, lo, hi, obj_at_delta, max_iter=25)
            best_val = obj_at_delta(best_delta)
            
            if best_val < best_obj - 1e-14:
                f = best_f.copy()
                f[i] -= best_delta
                f[j] += best_delta
                f = np.maximum(f, 0)
                f = f / f.sum()
                conv = fast_convolve(f)
                obj = conv.max()
                if obj < best_obj:
                    best_obj = obj
                    best_f = f.copy()
        
        if time.monotonic() - t0 > time_limit - 2:
            break
    
    return best_f, best_obj

def generate_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:

    inits.append(np.ones(n) / n)

    inits.append(ramp_d / ramp_d.sum())
    
    ramp_i = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp_i / ramp_i.sum())

    eps = 1e-6
    arcs = 1.0 / np.sqrt(np.abs(x / (n - 1) - 0.5) ** 2 + 0.25) + eps

    c = np.cos(np.pi * x / n) ** 2 + 0.1
    inits.append(c / c.sum())
    
    # Random
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    # Random sparse
    sparse = np.zeros(n)
    idx = rng.choice(n, size=max(2, n // 4), replace=False)
    sparse[idx] = 1.0
    inits.append(sparse / sparse.sum())
    
    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)

        if ps > 1e-10:
            inits.append(px / ps)
            inits.append(project_simplex(px / ps + rng.standard_normal(n) * 0.02))
    
    return inits

            gr = (np.sqrt(5) - 1) / 2
            c_pt = hi - gr * (hi - lo)
            d_pt = lo + gr * (hi - lo)
            fc = line_search_obj(f, g, c_pt)
            fd = line_search_obj(f, g, d_pt)
            for _ in range(30):
                if hi - lo < 1e-8:
                    break
                if fc < fd:
                    hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = hi - gr * (hi - lo)
                    fc = line_search_obj(f, g, c_pt)
                else:
                    lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = lo + gr * (hi - lo)
                    fd = line_search_obj(f, g, d_pt)
            best_alpha = (lo + hi) / 2
            best_val = line_search_obj(f, g, best_alpha)
        
        if best_val < M - 1e-14:
            f = project_simplex((1.0 - best_alpha) * f + best_alpha * g)
            conv = fast_convolve(f)
            M = conv.max()
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if verbose and it % 20 == 0:
            print(f"    FW-LP iter {it}: M={M:.8f}, best={best_obj:.8f}")
    
    return best_f, best_obj

def nesterov_sg(f0, time_limit, initial_lr=0.01, top_k=128, temp=0.001):
    """Nesterov-accelerated subgradient descent."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f))
    else:
        f = f / s
    n = len(f)
    
    y = f.copy()
    conv = fast_convolve(y)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    cur_lr = initial_lr
    cur_temp = temp
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break
        
        conv_y = fast_convolve(y)
        sg = compute_subgradient(y, conv_y, top_k=top_k, temp=cur_temp)
        
        y_new = project_simplex(f - cur_lr * sg)
        
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1) ** 2)) / 2.0
        beta = (t_k - 1.0) / t_k
        f_new = beta * f + (1.0 - beta) * y_new
        
        f = f_new
        y = project_simplex(f - cur_lr * sg)
        
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
        
        cur_lr *= 0.9995
        cur_temp *= 1.005
        
        if it % 500 == 0:
            rng = np.random.default_rng(it * 31 + 7)
            noise = rng.standard_normal(n) * 0.003
            f_try = project_simplex(best_f + noise)
            conv_try = fast_convolve(f_try)
            if conv_try.max() < best_obj:
                f = f_try
                y = f_try.copy()
                best_obj = conv_try.max()
                best_f = f_try.copy()
            cur_lr = initial_lr * 0.5
            cur_temp = temp * 0.5
    
    return best_f, best_obj

def pairwise_cd(f0, time_limit, n_pairs_per_pass=200):
    """Pairwise coordinate descent polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f))
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        for _ in range(n_pairs_per_pass):
            i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
            if i == j:
                continue
            
            lo = -f[j] + 1e-10
            hi = f[i] - 1e-10
            if lo >= hi:
                continue
            
            # Golden section search for best delta
            gr = (np.sqrt(5) - 1) / 2
            c_pt = hi - gr * (hi - lo)
            d_pt = lo + gr * (hi - lo)
            fc = cd_step_obj(f, i, j, c_pt)
            fd = cd_step_obj(f, i, j, d_pt)
            for _ in range(20):
                if hi - lo < 1e-10:
                    break
                if fc < fd:
                    hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = hi - gr * (hi - lo)
                    fc = cd_step_obj(f, i, j, c_pt)
                else:
                    lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = lo + gr * (hi - lo)
                    fd = cd_step_obj(f, i, j, d_pt)
            
            best_delta = (lo + hi) / 2
            best_val = cd_step_obj(f, i, j, best_delta)
            
            if best_val < best_obj - 1e-14:
                f_try = f.copy()
                f_try[i] = max(0.0, f[i] - best_delta)
                f_try[j] = max(0.0, f[j] + best_delta)
                s_try = f_try.sum()
                if s_try > 1e-15:
                    f_try = f_try / s_try
                    f = f_try
                    conv = fast_convolve(f)
                    obj = conv.max()
                    if obj < best_obj:
                        best_obj = obj
                        best_f = f.copy()
        
        if time.monotonic() - t0 > time_limit - 2:
            break
    
    return best_f, best_obj

def generate_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    inits.append(np.ones(n) / n)
    
    ramp_d = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(ramp_d / ramp_d.sum())
    
    ramp_i = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp_i / ramp_i.sum())
    
    # Arcsine-like
    eps = 1e-6
    t = x / max(1, n - 1)
    arcs = 1.0 / np.sqrt((t - 0.5) ** 2 + 0.25) + eps
    inits.append(arcs / arcs.sum())
    
    # Two bumps
    bump1 = np.exp(-((x - n * 0.25) / max(1, n * 0.1)) ** 2)
    bump2 = np.exp(-((x - n * 0.75) / max(1, n * 0.1)) ** 2)
    inits.append((bump1 + bump2 + eps) / (bump1 + bump2 + eps).sum())
    
    # Cosine
    c = np.cos(np.pi * x / n) ** 2 + 0.1
    inits.append(c / c.sum())
    
    # Random
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    # Random sparse
    sparse = np.zeros(n)
    idx = rng.choice(n, size=max(2, n // 4), replace=False)
    sparse[idx] = 1.0
    inits.append(sparse / sparse.sum())
    
    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            inits.append(project_simplex(px / ps + rng.standard_normal(n) * 0.02))
    
    return inits

def try_sequence(a, best_state):
    """Try a sequence and update best state. Returns (best_seq, best_score)."""
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    if a.sum() < 0.01:
        return best_state
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_state[1]:
        old = best_state[1]
        best_state = (a.copy(), score)
        print(f"  NEW BEST: {score:.8f} (n={len(a)}) (was {old:.8f})")
    return best_state
