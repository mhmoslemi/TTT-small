Hybrid LP + Nesterov gradient optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements over previous approach:
1. Longer sequences (1024-2048) for more degrees of freedom
2. Efficient LP step with adaptive M and true-objective line search
3. Dual-mode LP (same-M for sum increase, lower-M for peak reduction)
4. Multi-start with diverse initializations including parent interpolation
5. Nesterov gradient polish with adaptive softmax temperature

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g) (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def compute_gradient(a, conv, top_k=30, temp=0.003):
    """Softmax-weighted gradient of peak autocorrelation."""
    n = len(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    max_pv = peak_vals.max()
    if max_pv < 1e-15:
        return np.zeros(n)
    scaled = peak_vals / (temp * max_pv)
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[peak_vals.argmax()] = 1.0
    else:
        w /= w_sum

    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += w[idx] * a[k - idxs]
    return grad


def obj_val(a):
    """Compute objective 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def lp_solve_direction(a, M, C=None):

    n = len(a)
    if C is None:

    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:

                         options={'presolve': True, 'time_limit': 30})
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:

    return None


def lp_optimize(a0, time_limit, verbose=False):

    conv = fast_convolve(a)

def nesterov_optimize(a0, time_limit, lr_init=0.008, momentum=0.92, top_k=40, temp_init=0.002):
    """Nesterov-accelerated gradient descent with adaptive parameters."""
    a = normalize(a0.copy())
    n = len(a)
    v = np.zeros(n)
    y = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        # Nesterov lookahead
        y = a + mom * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y

        conv_y = fast_convolve(y)
        grad = compute_gradient(y, conv_y, top_k=top_k, temp=temp)

        # Update
        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
            v = np.zeros(n)
            continue

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive scheduling
        lr *= 0.9997
        temp *= 1.004
        mom = min(0.97, mom + 0.0001)

        if no_improve >= 300:
            break
        if lr < 1e-6:
            break

    return best_a, best_obj


def gradient_polish(a0, time_limit, n_configs=5):
    """Multiple gradient runs with different hyperparameters."""
    best_a = normalize(a0.copy())
    best_obj = 2.0 * len(a0) * fast_convolve(best_a).max()

    configs = [
        {'lr_init': 0.005, 'momentum': 0.94, 'top_k': 60, 'temp_init': 0.001},
        {'lr_init': 0.003, 'momentum': 0.95, 'top_k': 100, 'temp_init': 0.0005},
        {'lr_init': 0.01, 'momentum': 0.89, 'top_k': 25, 'temp_init': 0.004},
        {'lr_init': 0.002, 'momentum': 0.96, 'top_k': 150, 'temp_init': 0.0003},
        {'lr_init': 0.015, 'momentum': 0.85, 'top_k': 15, 'temp_init': 0.006},
    ]

    t0 = time.monotonic()
    for i in range(n_configs):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3.0:
            break
        t_alloc = min(remaining / (n_configs - i) * 0.85, time_limit - (time.monotonic() - t0) - 1.0)
        if t_alloc < 2.0:
            break
        cfg = configs[i % len(configs)]
        a_opt, obj = nesterov_optimize(best_a, time_limit=t_alloc, **cfg)
        if obj < best_obj:
            best_obj = obj
            best_a = a_opt

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations for sequence of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Sine-based
    inits['sine1'] = (np.sin(np.pi * x / n) ** 2 + 0.01)
    inits['sine1'] /= inits['sine1'].sum()

    inits['sine2'] = (np.sin(2 * np.pi * x / n) ** 2 + 0.01)
    inits['sine2'] /= inits['sine2'].sum()

    # Hamming window
    inits['hamming'] = 0.54 - 0.46 * np.cos(2 * np.pi * x / (n - 1))
    inits['hamming'] = np.maximum(inits['hamming'], 0.01)
    inits['hamming'] /= inits['hamming'].sum()

    # Blackman window
    inits['blackman'] = (0.42 - 0.5 * np.cos(2 * np.pi * x / (n - 1))
                         + 0.08 * np.cos(4 * np.pi * x / (n - 1)))
    inits['blackman'] = np.maximum(inits['blackman'], 0.01)
    inits['blackman'] /= inits['blackman'].sum()

    # Bimodal (edges)
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Asymmetric ramp
    ramp = np.linspace(0.1, 1.0, n)
    inits['ramp'] = ramp / ramp.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps
                # Also try reversed parent
                pr = pp[::-1].copy()
                prs = pr.sum()
                if prs > 1e-10:
                    inits['parent_rev'] = pr / prs

    return inits


def extend_sequence(a_short, n_target):
    """Extend a shorter sequence to target length via interpolation."""
    n_short = len(a_short)
    s = a_short.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    a_norm = a_short / s
    a_ext = np.interp(np.linspace(0, 1, n_target), np.linspace(0, 1, n_short), a_norm)
    a_ext = np.maximum(a_ext, 0)
    ses = a_ext.sum()
    if ses > 1e-15:
        a_ext = a_ext / ses
    return a_ext


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Hybrid LP + gradient search with multi-scale exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass

    # =========================================================
    # Phase 1: Quick Nesterov scan across lengths and inits
    # =========================================================
    print("=== Phase 1: Nesterov Scan ===")
    t_start = time.monotonic()
    phase1_time = 250.0

    lengths = [1024, 1536, 2048]
    scan_results = []

    for n in lengths:
        elapsed = time.monotonic() - t_start
        if elapsed > phase1_time - 120:
            break
        inits = generate_initializations(n, parent, rng)

        # Priority order
        priority = ['parent', 'parent_interp', 'parent_rev', 'uniform', 'sine1',
                    'hamming', 'blackman', 'bimodal', 'ramp', 'random']
        init_names = [nm for nm in priority if nm in inits]
        init_names += [nm for nm in inits if nm not in priority]
        init_names = init_names[:5]

        t_per = max(15.0, (phase1_time - elapsed - 60) / (len(lengths) * len(init_names)))

        for name in init_names:
            remaining = deadline - time.monotonic() - 400
            if remaining < 30:
                break
            t_alloc = min(t_per, remaining / max(1, len(init_names)))
            if t_alloc < 8:
                break

            a0 = inits[name]
            a_opt, obj = nesterov_optimize(a0, time_limit=t_alloc,
                                            lr_init=0.01, momentum=0.91,
                                            top_k=40, temp_init=0.002)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, {name}: {score:.6f} *BEST*")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 2: LP deep optimization on top candidates
    # =========================================================
    print("=== Phase 2: LP Deep Optimization ===")
    t_start2 = time.monotonic()

    # Select top candidates (diverse lengths)
    top_configs = []
    seen_lengths = set()
    for score, n, name, a_opt in scan_results:
        if n not in seen_lengths:
            top_configs.append((n, a_opt, score))
            seen_lengths.add(n)
        if len(top_configs) >= 4:
            break

    # Include best overall if not already there
    if best_sequence is not None:
        if len(best_sequence) not in seen_lengths:
            top_configs.append((len(best_sequence), best_sequence.copy(), best_score))
            seen_lengths.add(len(best_sequence))

    remaining = deadline - time.monotonic() - 200.0
    n_candidates = max(1, len(top_configs))
    t_per_lp = max(40.0, remaining * 0.55 / n_candidates)

    for i, (n, a0, sc) in enumerate(top_configs):
        if time.monotonic() > deadline - 150:
            break
        t_alloc = min(t_per_lp, deadline - time.monotonic() - 120)
        if t_alloc < 15:
            break

        print(f"  LP cand {i}: n={n}, init_score={sc:.6f}, t_alloc={t_alloc:.0f}s")
        a_opt, obj = lp_optimize(a0, time_limit=t_alloc, verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  LP {i} (n={n}): {score:.6f} *NEW BEST*")
            else:
                print(f"  LP {i} (n={n}): {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 3: Extended LP with multi-restart on best
    # =========================================================
    print("=== Phase 3: LP Multi-Restart ===")
    t_start3 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 150.0
        n_restarts = 4
        t_per_restart = max(20.0, remaining * 0.5 / n_restarts)

        for ri in range(n_restarts):
            if time.monotonic() > deadline - 100:
                break
            t_alloc = min(t_per_restart, deadline - time.monotonic() - 80)
            if t_alloc < 10:
                break

            if ri == 0:
                a0 = best_sequence.copy()
            else:
                scale = 0.003 * (ri + 1)
                noise = rng.standard_normal(len(best_sequence)) * scale
                a0 = np.maximum(best_sequence + noise, 0)
                a0 = normalize(a0)

            a_opt, obj = lp_optimize(a0, time_limit=t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Restart {ri}: {score:.6f} *NEW BEST*")
                else:
                    print(f"  Restart {ri}: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 4: Gradient polish
    # =========================================================
    print("=== Phase 4: Gradient Polish ===")
    t_start4 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 50.0
        if remaining > 20:
            a_opt, obj = gradient_polish(best_sequence, time_limit=remaining * 0.6, n_configs=5)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  GradPolish: {score:.6f} *NEW BEST*")
                else:
                    print(f"  GradPolish: {score:.6f}")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 5: Short LP micro-polish
    # =========================================================
    print("=== Phase 5: Final LP Polish ===")
    if best_sequence is not None and time.monotonic() < deadline - 15:
        t_final = min(30.0, deadline - time.monotonic() - 8)
        if t_final > 5:
            a_opt, obj = lp_optimize(best_sequence, time_limit=t_final)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  FinalLP: {score:.6f} *NEW BEST*")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
