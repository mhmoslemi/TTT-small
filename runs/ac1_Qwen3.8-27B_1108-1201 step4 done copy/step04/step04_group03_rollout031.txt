"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using mirror descent with adaptive
peak weighting, multi-resolution initialization, and diverse restarts.
Key improvements over previous: better peak weighting based on excess over flat
level, adaptive step size with backtracking, multi-resolution warm start,

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(f, conv, top_k, temperature):
    """Compute a weighted subgradient concentrated on top peaks."""
    n = len(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    
    pv = conv[peaks]
    # Weight by excess over the mean (encourages flattening)
    mean_val = 1.0 / (2 * n - 1)  # ideal flat level for sum=1
    excess = np.maximum(pv - mean_val, 0)
    
    if temperature > 0:
        scaled = excess / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
    else:
        # Hard focus on the single highest peak
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += w[i] * f[k - idx]
    
    return sg

def mirror_descent_optimize(f0, time_limit, lr_init=0.1, top_k=20, 
                            temp_init=0.001, decay_lr=0.9999, 
                            decay_temp=1.001, verbose=False):
    """
    Mirror descent (multiplicative weight) with adaptive peak weighting.
    Uses excess-over-flat-level weighting and backtracking line search.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_temp = temp_init
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Compute subgradient
        sg = compute_subgradient(f, conv, top_k, cur_temp)
        
        # Multiplicative weight update
        f_new = f * np.exp(-cur_lr * sg)
        f_new = np.maximum(f_new, 1e-30)
        f_new = f_new / f_new.sum()
        
        # Check improvement
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            no_improve += 1
            # Try smaller step
            for shrink in [0.5, 0.25, 0.1]:
                f_try = f * np.exp(-cur_lr * shrink * sg)
                f_try = np.maximum(f_try, 1e-30)
                f_try = f_try / f_try.sum()
                conv_try = fast_convolve(f_try)
                if conv_try.max() < M:
                    f = f_try
                    break
            else:
                f = f_new  # Accept anyway to move
        
        cur_lr *= decay_lr
        cur_temp *= decay_temp
        
        # Restart if stuck
        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 1000000) % 1000000)
            perturbation = 0.01 * min(1.0, no_improve / 500.0)
            f = best_f * np.exp(rng.standard_normal(n) * perturbation)
            f = np.maximum(f, 0)
            f = f / f.sum()
            cur_lr = max(lr_init * 0.5, 0.01)
            cur_temp = temp_init * 2.0
            no_improve = 0
        
        if verbose and it % 1000 == 0:
            score = 2 * n * M
            print(f"    MD iter {it}: obj={M:.10f}, score={score:.8f}, lr={cur_lr:.6f}, temp={cur_temp:.6f}")
    
    return best_f, best_obj

def subgradient_optimize(f0, time_limit, lr_init=0.01, mom=0.9, top_k=30,
                         temp_init=0.001, verbose=False):
    """
    Subgradient method with momentum and adaptive peak weighting.
    Uses simplex projection.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_temp = temp_init
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Compute subgradient
        sg = compute_subgradient(f, conv, top_k, cur_temp)
        
        # Update with momentum
        v = mom * v - cur_lr * sg
        f_new = project_simplex(f + v)
        
        # Check improvement
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            no_improve += 1
            f = f_new  # Still move, but track best
        
        cur_lr *= 0.99995
        cur_temp *= 1.0005
        
        # Restart if stuck
        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 1000000) % 1000000)
            perturbation = 0.005 * min(1.0, no_improve / 1000.0)
            f = project_simplex(best_f + rng.standard_normal(n) * perturbation)
            v = np.zeros(n)
            cur_lr = max(lr_init * 0.5, 0.001)
            no_improve = 0
    
    return best_f, best_obj

def multi_peak_flatten(f0, time_limit, verbose=False):
    """
    Specialized algorithm that explicitly targets the top peaks.
    At each iteration, identifies the highest peak and computes the 
    optimal direction to reduce it (using the fact that the problem 
    is locally quadratic near the peak).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Find top peaks
        top_k = max(5, n // 10)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        # Weight peaks by rank (emphasize the top ones)
        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 0.8
        weights /= weights.sum()
        
        # Compute combined gradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += weights[i] * f[k - idx]
        
        # Adaptive step size: try multiple steps and pick the best
        best_alpha = 0.0
        best_val = M
        for alpha in [0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
            f_try = project_simplex(f - alpha * sg / sg.max())
            conv_try = fast_convolve(f_try)
            if conv_try.max() < best_val:
                best_val = conv_try.max()
                best_alpha = alpha
        
        if best_alpha > 0:
            f = project_simplex(f - best_alpha * sg / sg.max())
            conv = fast_convolve(f)
            if conv.max() < best_obj - 1e-15:
                best_obj = conv.max()
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Restart
        if no_improve > 50:
            rng = np.random.default_rng(int(time.monotonic() * 1000000) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            no_improve = 0
    
    return best_f, best_obj

def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for length n."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Raised cosine
    x = np.arange(n, dtype=np.float64)
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    inits.append(rc / rc.sum())
    
    # Tukey window (alpha=0.5)
    alpha = 0.5
    tw = np.ones(n)
    for i in range(n):
        if i < alpha * n / 2:
            tw[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * n) - 1)))
        elif i >= n - alpha * n / 2:
            tw[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * n) - 2 * n / (alpha * n) + 1)))
    inits.append(tw / tw.sum())
    
    # Center-dipped (try to reduce center peak)
    cd = np.ones(n)
    mid = n // 2
    dip_width = n // 4
    for i in range(n):
        d = abs(i - mid)
        if d < dip_width:
            cd[i] = 0.3 + 0.7 * (d / dip_width) ** 2
    inits.append(cd / cd.sum())
    
    # Two bumps
    g1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    inits.append((g1 + g2) / (g1 + g2).sum())
    
    # Cosine with offset
    c = np.abs(np.cos(np.pi * x / n)) ** 2 + 0.1
    inits.append(c / c.sum())
    
    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    
    # More structured random
    inits.append(rng.dirichlet(np.ones(n) * 1.0))
    
    # If parent available, interpolate
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
        
        # Perturbed parent
        pn = px * np.exp(rng.standard_normal(n) * 0.1)
        pn = np.maximum(pn, 0)
        if pn.sum() > 1e-10:
            inits.append(pn / pn.sum())
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach combining:
1. Multi-resolution initialization across different sequence lengths
2. Mirror descent (multiplicative weight) with adaptive peak weighting
3. Subgradient with momentum and simplex projection
4. LP-based Frank-Wolfe for precise direction finding
5. Fine-grained local search with diverse restarts
Key innovations: excess-over-flat-level peak weighting, adaptive temperature
scheduling, multi-length exploration, and aggressive final polish.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_weighted_subgradient(f, conv, top_k, temperature, use_excess=True):
    """Compute a weighted subgradient concentrated on top peaks."""
    n = len(f)
    num_conv = len(conv)
    if top_k >= num_conv:
        peaks = np.arange(num_conv)
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]

    pv = conv[peaks]
    if use_excess:
        mean_val = 1.0 / (2 * n - 1)
        excess = np.maximum(pv - mean_val, 0.0)
        if excess.max() < 1e-15:
            excess = pv - pv.min()
    else:
        excess = pv

    if temperature > 1e-10:
        scaled = excess / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
    else:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += w[i] * f[k - idx]

    return sg


def mirror_descent_optimize(f0, time_limit, lr_init=0.05, top_k=15,
                            temp_init=0.001, decay_lr=0.9999,
                            decay_temp=1.001, use_excess=True, verbose=False):
    """
    Mirror descent (multiplicative weight) with adaptive peak weighting.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_temp = temp_init
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        sg = compute_weighted_subgradient(f, conv, top_k, cur_temp, use_excess)

        # Multiplicative weight update
        log_step = -cur_lr * sg
        # Clamp to avoid overflow/underflow
        log_step = np.clip(log_step, -5.0, 5.0)
        f_new = f * np.exp(log_step)
        f_new = np.maximum(f_new, 1e-30)
        ns = f_new.sum()
        if ns < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / ns

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            no_improve += 1
            # Backtracking: try smaller steps
            accepted = False
            for shrink in [0.5, 0.25, 0.1, 0.05]:
                ls2 = -cur_lr * shrink * sg
                ls2 = np.clip(ls2, -5.0, 5.0)
                f_try = f * np.exp(ls2)
                f_try = np.maximum(f_try, 1e-30)
                st = f_try.sum()
                if st < 1e-15:
                    continue
                f_try = f_try / st
                ct = fast_convolve(f_try)
                if ct.max() < M - 1e-15:
                    f = f_try
                    accepted = True
                    break
            if not accepted:
                f = f_new

        cur_lr = max(cur_lr * decay_lr, 0.001)
        cur_temp *= decay_temp

        if no_improve > 150:
            rng = np.random.default_rng(int(time.monotonic() * 1000000) % 1000000)
            pert = 0.005 + 0.01 * min(1.0, no_improve / 400.0)
            f = best_f * np.exp(rng.standard_normal(n) * pert)
            f = np.maximum(f, 0)
            f = f / f.sum()
            cur_lr = max(lr_init * 0.5, 0.005)
            cur_temp = temp_init * 1.5
            no_improve = 0

        if verbose and it % 2000 == 0:
            print(f"      MD iter {it}: obj={M:.10f}, score={2*n*M:.8f}, lr={cur_lr:.5f}")

    return best_f, best_obj


def subgradient_optimize(f0, time_limit, lr_init=0.005, mom=0.92, top_k=25,
                         temp_init=0.001, use_excess=True, verbose=False):
    """
    Subgradient method with momentum and simplex projection.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_temp = temp_init
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        sg = compute_weighted_subgradient(f, conv, top_k, cur_temp, use_excess)

        # Normalize gradient for stability
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            no_improve += 100
        else:
            sg = sg / sg_norm

        v = mom * v - cur_lr * sg
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            no_improve += 1
            f = f_new

        cur_lr = max(cur_lr * 0.99995, 0.0001)
        cur_temp *= 1.0005

        if no_improve > 250:
            rng = np.random.default_rng(int(time.monotonic() * 1000000) % 1000000)
            pert = 0.003 + 0.008 * min(1.0, no_improve / 800.0)
            f = project_simplex(best_f + rng.standard_normal(n) * pert)
            v = np.zeros(n)
            cur_lr = max(lr_init * 0.6, 0.001)
            no_improve = 0

    return best_f, best_obj


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 30})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit, max_lp_calls=50, verbose=False):
    """Frank-Wolfe with LP oracle, with fallback to subgradient."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lp_count = 0
    use_subgrad = False

    while time.monotonic() - t0 < time_limit:
        it += 1

        if not use_subgrad and lp_count < max_lp_calls:
            # Try LP oracle
            g = solve_lp_oracle(f, M * 1.01)
            if g is not None:
                lp_count += 1
                gs = g.sum()
                if gs < 1e-15:
                    use_subgrad = True
                    continue
                g = g / gs

                if np.max(np.abs(g - f)) < 1e-9:
                    no_improve += 1
                    if no_improve > 20:
                        use_subgrad = True
                    continue

                # Line search
                best_alpha = 0.0
                best_val = M
                for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
                    ft = (1.0 - alpha) * f + alpha * g
                    ft = np.maximum(ft, 0.0)
                    st = ft.sum()
                    if st < 1e-15:
                        continue
                    ft = ft / st
                    val = fast_convolve(ft).max()
                    if val < best_val:
                        best_val = val
                        best_alpha = alpha

                if best_alpha > 1e-10:
                    f = (1.0 - best_alpha) * f + best_alpha * g
                    f = np.maximum(f, 0.0)
                    f = f / f.sum()
                    conv = fast_convolve(f)
                    M = conv.max()

                    if M < best_obj - 1e-14:
                        best_obj = M
                        best_f = f.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                use_subgrad = True
        else:
            # Subgradient step
            sg = compute_weighted_subgradient(f, conv, 20, 0.001, True)
            lr = 0.002 * (1.0 + 0.5 * (1 - (time.monotonic() - t0) / time_limit))
            f_new = project_simplex(f - lr * sg)
            conv_new = fast_convolve(f_new)
            obj_new = conv_new.max()
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_f = f_new.copy()
                f = f_new
                conv = conv_new
                M = obj_new
                no_improve = 0
            else:
                no_improve += 1
                f = f_new

        if no_improve > 40:
            rng = np.random.default_rng(it * 1000 + 7)
            scale = 0.003 + 0.005 * min(1.0, no_improve / 100.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0
            use_subgrad = False

    return best_f, best_obj


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / max(n - 1, 1)))
    inits.append(rc / rc.sum())

    # Center-dipped (reduces center peak)
    cd = np.ones(n)
    mid = n // 2
    dip_w = max(1, n // 4)
    for i in range(n):
        d = abs(i - mid)
        if d < dip_w:
            cd[i] = 0.3 + 0.7 * (d / dip_w) ** 2
    inits.append(cd / cd.sum())

    # Two bumps
    g1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.12 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.12 * n)) ** 2)
    inits.append((g1 + g2) / (g1 + g2).sum())

    # Edge-concentrated
    ew = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 0.5
    inits.append(ew / ew.sum())

    # Cosine squared
    c = (np.cos(np.pi * x / n) ** 2 + 0.05)
    inits.append(c / c.sum())

    # Random Dirichlet (concentrated)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    # Random Dirichlet (spread)
    inits.append(rng.dirichlet(np.ones(n) * 2.0))

    # Sinusoidal perturbation
    sp = 1.0 + 0.3 * np.sin(2 * np.pi * x / (n / 3.0))
    inits.append(sp / sp.sum())

    # If parent available
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
        # Perturbed parent
        pn = px * np.exp(rng.standard_normal(n) * 0.15)
        pn = np.maximum(pn, 0)
        if pn.sum() > 1e-10:
            inits.append(pn / pn.sum())

    return inits


def local_hill_climb(f0, num_trials, step_size, rng):
    """Try random directions and keep improvements."""
    f = f0.copy()
    n = len(f)
    best_obj = fast_convolve(f).max()

    for _ in range(num_trials):
        direction = rng.standard_normal(n)
        # Try both directions
        for sign in [1, -1]:
            f_try = project_simplex(f + sign * step_size * direction)
            obj = fast_convolve(f_try).max()
            if obj < best_obj - 1e-12:
                f = f_try
                best_obj = obj
                break

    return f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search:
    Phase 1: Quick exploration across lengths and initializations
    Phase 2: Deep optimization of best candidates (MD, SG, FW-LP)
    Phase 3: Fine polish with small steps and diverse restarts
    Phase 4: Local hill climbing for final extraction
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 45.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        # Compute score directly (faster than evaluate_sequence)
        n = len(a_norm)
        conv = fast_convolve(a_norm)
        score = 2.0 * n * conv.max()
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={n})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = 2.0 * len(parent) * fast_convolve(parent).max()
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(384) / 384

    # Sequence lengths to explore
    lengths = [len(parent)]
    for ln in [384, 512, 768, 1024]:
        if ln not in lengths:
            lengths.append(ln)
    lengths = sorted(set(lengths))

    # === Phase 1: Quick exploration (~350s) ===
    print("\n=== Phase 1: Quick Exploration ===")
    phase1_budget = min(350.0, deadline - time.monotonic() - 1200)
    t_start_p1 = time.monotonic()

    phase1_candidates = []

    for n in lengths:
        rem = deadline - time.monotonic() - 1200
        if rem < 20:
            break

        inits = generate_initializations(n, rng, parent)
        t_per = max(4.0, phase1_budget / (len(lengths) * len(inits)))

        for ii, init in enumerate(inits):
            rem = deadline - time.monotonic() - 1200
            if rem < 8:
                break

            # Quick mirror descent
            t_alloc = min(t_per, rem)
            a_opt, _ = mirror_descent_optimize(
                init, time_limit=t_alloc,
                lr_init=0.04, top_k=12, temp_init=0.001
            )
            sc = try_seq(a_opt, f"md_n{n}_i{ii}")
            if np.isfinite(sc):
                phase1_candidates.append((a_opt, sc, n))

            rem = deadline - time.monotonic() - 1200
            if rem < 8:
                break

            # Quick subgradient
            t_alloc = min(t_per, rem)
            a_opt, _ = subgradient_optimize(
                init, time_limit=t_alloc,
                lr_init=0.004, mom=0.9, top_k=20, temp_init=0.001
            )
            sc = try_seq(a_opt, f"sg_n{n}_i{ii}")
            if np.isfinite(sc):
                phase1_candidates.append((a_opt, sc, n))

    print(f"  Phase 1 best: {best_score:.8f}")

    # === Phase 2: Deep optimization (~700s) ===
    print("\n=== Phase 2: Deep Optimization ===")

    if best_seq is not None:
        n_best = len(best_seq)

        # Strategy: alternate between different optimizers on the best sequence
        # and on perturbed versions of it.

        deep_configs = [
            ('md', dict(lr_init=0.03, top_k=10, temp_init=0.0005, decay_lr=0.99995, decay_temp=1.001)),
            ('md', dict(lr_init=0.05, top_k=20, temp_init=0.001, decay_lr=0.9999, decay_temp=1.0015)),
            ('md', dict(lr_init=0.02, top_k=30, temp_init=0.002, decay_lr=0.99995, decay_temp=1.002)),
            ('sg', dict(lr_init=0.003, mom=0.93, top_k=20, temp_init=0.001)),
            ('sg', dict(lr_init=0.005, mom=0.90, top_k=35, temp_init=0.0005)),
            ('sg', dict(lr_init=0.002, mom=0.95, top_k=15, temp_init=0.002)),
            ('fw', dict(max_lp_calls=30)),
            ('md', dict(lr_init=0.04, top_k=25, temp_init=0.0008, decay_lr=0.9999, decay_temp=1.001)),
        ]

        rem_total = deadline - time.monotonic() - 400
        t_per_cfg = max(30.0, rem_total / len(deep_configs))

        for ci, (method, cfg) in enumerate(deep_configs):
            rem = deadline - time.monotonic() - 400
            if rem < 20:
                break
            t_alloc = min(t_per_cfg, rem)

            print(f"  Deep {method} cfg{ci}, n={n_best}, t={t_alloc:.0f}s")

            if method == 'md':
                a_opt, _ = mirror_descent_optimize(best_seq, time_limit=t_alloc, **cfg)
            elif method == 'sg':
                a_opt, _ = subgradient_optimize(best_seq, time_limit=t_alloc, **cfg)
            else:
                a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc, **cfg)

            try_seq(a_opt, f"deep_{method}_{ci}")

            # Also try from a perturbed starting point
            rem = deadline - time.monotonic() - 400
            if rem > 25:
                noise = rng.standard_normal(n_best) * 0.02
                f_pert = project_simplex(best_seq + noise)
                t_alloc2 = min(40.0, rem * 0.5)

                if method == 'md':
                    a_opt2, _ = mirror_descent_optimize(f_pert, time_limit=t_alloc2, **cfg)
                elif method == 'sg':
                    a_opt2, _ = subgradient_optimize(f_pert, time_limit=t_alloc2, **cfg)
                else:
                    a_opt2, _ = mirror_descent_optimize(
                        f_pert, time_limit=t_alloc2,
                        lr_init=0.03, top_k=15, temp_init=0.001
                    )
                try_seq(a_opt2, f"deep_{method}_{ci}_pert")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Fine polish (~350s) ===
    print("\n=== Phase 3: Fine Polish ===")

    if best_seq is not None:
        n_best = len(best_seq)

        # Fine MD with very small lr
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_alloc = min(rem, 120)
            print(f"  Fine MD, t={t_alloc:.0f}s")
            a_opt, _ = mirror_descent_optimize(
                best_seq, time_limit=t_alloc,
                lr_init=0.008, top_k=10, temp_init=0.0002,
                decay_lr=0.99997, decay_temp=1.0005
            )
            try_seq(a_opt, "fine_md")

        # Fine SG with high top_k
        rem = deadline - time.monotonic() - 80
        if rem > 15:
            t_alloc = min(rem, 100)
            print(f"  Fine SG, t={t_alloc:.0f}s")
            a_opt, _ = subgradient_optimize(
                best_seq, time_limit=t_alloc,
                lr_init=0.0008, mom=0.95, top_k=max(20, n_best // 4),
                temp_init=0.0002
            )
            try_seq(a_opt, "fine_sg")

        # Multiple small perturbations with quick optimization
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            num_pert = max(5, int(rem / 6))
            for pi in range(num_pert):
                rem = deadline - time.monotonic() - 20
                if rem < 4:
                    break
                noise_scale = 0.001 + 0.003 * (pi / max(num_pert, 1))
                noise = rng.standard_normal(n_best) * noise_scale
                f_pert = project_simplex(best_seq + noise)
                t_alloc = min(5.0, rem)
                a_opt, _ = mirror_descent_optimize(
                    f_pert, time_limit=t_alloc,
                    lr_init=0.01, top_k=10, temp_init=0.0003
                )
                try_seq(a_opt, f"perturb_{pi}")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: Local hill climbing and final squeeze (~100s) ===
    print("\n=== Phase 4: Local Hill Climb ===")

    if best_seq is not None:
        # Hill climbing with small steps
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            print(f"  Hill climbing, t={rem:.0f}s")
            t0_hc = time.monotonic()
            step = 0.002
            while time.monotonic() - t0_hc < min(rem, 60):
                f_hc, obj_hc = local_hill_climb(
                    best_seq, num_trials=500, step_size=step, rng=rng
                )
                sc = try_seq(f_hc, "hill_climb")
                if sc >= best_score:
                    break
                step *= 0.7

        # Very fine final MD
        rem = deadline - time.monotonic() - 10
        if rem > 8:
            print(f"  Final MD, t={min(rem, 30):.0f}s")
            a_opt, _ = mirror_descent_optimize(
                best_seq, time_limit=min(rem, 30),
                lr_init=0.003, top_k=8, temp_init=0.0001,
                decay_lr=0.99998, decay_temp=1.0003
            )
            try_seq(a_opt, "final_md")

        # Final SG
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_opt, _ = subgradient_optimize(
                best_seq, time_limit=min(rem, 15),
                lr_init=0.0003, mom=0.96, top_k=max(15, n_best // 5),
                temp_init=0.0001
            )
            try_seq(a_opt, "final_sg")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = 2.0 * 384 * fast_convolve(best_seq).max()

    # Final verification with evaluate_sequence
    final_check = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_check) and final_check < best_score:
        best_score = final_check

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
