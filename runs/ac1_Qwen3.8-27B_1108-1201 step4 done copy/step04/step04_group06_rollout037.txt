        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[i, j] = a[k - j]
    return A

def lp_direction(a, M, top_k=50):
    """Find LP direction: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    conv_a = fast_convolve(a)

def mwu_step(a, peak_idx, eta, n_top=5):
    """One MWU step: reduce mass at positions feeding the peak."""
    n = len(a)
    conv = fast_convolve(a)
    # Get top peaks
    top_peaks = np.argpartition(conv, -n_top)[-n_top:]
    
    log_w = np.zeros(n)
    for k in top_peaks:
        k = int(k)
        w = conv[k] - conv[top_peaks].min()  # weight by excess
        if w < 0:
            w = 0
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            log_w[j] -= eta * w * a[k - j]
    
    a_new = a * np.exp(log_w)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def nesterov_grad(a0, time_limit, lr_init=0.01, momentum=0.92, top_k=30, temp=0.003):

    no_improve = 0

        y = np.maximum(y, 0.0)

        peaks = np.argpartition(conv_y, -top_k)[-top_k:]

        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= ws

        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Sparse-constraint LP direction finding with bisection step sizing
2. Multiplicative Weight Update (MWU) for non-negative optimization
3. Multi-length exploration (n = 768, 1024, 1536)

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_rows(a, lags):
    """Build dense rows of convolution matrix for given lags.
    Returns A of shape (len(lags), n) where A[i] @ g = conv(a, g)[lags[i]].
    """
    n = len(a)
    A = np.zeros((len(lags), n))
    for i, k in enumerate(lags):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[i, j] = a[k - j]
    return A


def lp_direction(a, M, top_k, rng_seed=0):
    """Solve LP: max sum(g) s.t. conv(a,g)[top lags] <= M, g >= 0.
    Returns g or None if no useful direction found.
    """
    n = len(a)
    conv_a = fast_convolve(a)
    n_lags = min(top_k, 2 * n - 1)
    top_indices = np.argpartition(conv_a, -n_lags)[-n_lags:]

    A = build_conv_rows(a, top_indices)
    c_obj = -np.ones(n)
    b_ub = np.full(n_lags, M * 1.00005)
    bounds = [(0, None)] * n

    try:

                         options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})

def compute_softmax_grad(a, conv, top_k, temp):
    """Compute softmax-weighted subgradient of peak autocorrelation."""
    n = len(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum

        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)

    return grad

    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        y = a + momentum * v
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy

        conv_y = fast_convolve(y)
        grad = compute_softmax_grad(y, conv_y, top_k, temp)

        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn > 1e-15:
            a = a_new / sn
        else:
            a = best_a.copy()
            v = np.zeros(n)

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9992
        temp *= 1.004

        if no_improve >= 400:
            break

    return best_a, best_obj


def lp_optimize(a0, time_limit, top_k_init=40, verbose=False):
    """LP-based optimization with sparse constraints and bisection step sizing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    iter_count = 0
    top_k = top_k_init
    perturb_count = 0

    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        g = lp_direction(a, M, top_k)

        if g is not None:
            d = g - a
            norm_d = np.linalg.norm(d)

            if norm_d > 1e-10:
                # Bisection step sizing: find largest t in [0,1] that improves
                lo, hi = 0.0, 1.0
                best_t = 0.0
                best_a_try = None

                # First check t=1
                a_full = np.maximum(a + d, 0.0)
                sf = a_full.sum()
                if sf > 1e-15:
                    a_full = a_full / sf
                    obj_full = 2.0 * n * fast_convolve(a_full).max()
                    if obj_full < best_obj - 1e-13:
                        best_t = 1.0
                        best_a_try = a_full.copy()
                    else:
                        # Binary search for best step
                        for _ in range(8):
                            mid = (lo + hi) / 2.0
                            a_mid = np.maximum(a + mid * d, 0.0)
                            sm = a_mid.sum()
                            if sm > 1e-15:
                                a_mid = a_mid / sm
                                obj_mid = 2.0 * n * fast_convolve(a_mid).max()
                                if obj_mid < best_obj - 1e-13:
                                    lo = mid
                                    best_t = mid
                                    best_a_try = a_mid.copy()
                                else:
                                    hi = mid
                else:
                    best_t = 0.0

                if best_t > 0.01 and best_a_try is not None:
                    a = best_a_try
                    conv = fast_convolve(a)
                    cur_obj = 2.0 * n * conv.max()
                    if cur_obj < best_obj:
                        best_obj = cur_obj
                        best_a = a.copy()
                    no_improve = 0
                    iter_count += 1
                    if verbose and iter_count % 100 == 0:
                        print(f"    LP iter {iter_count}, obj={best_obj:.8f}, t={time.monotonic()-t0:.1f}s")
                    continue

        no_improve += 1
        iter_count += 1

        if no_improve >= 40:
            rng = np.random.default_rng(iter_count * 17 + perturb_count * 131)
            scale = 0.004 + 0.006 * min(perturb_count, 12)
            noise = rng.standard_normal(n) * scale
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s2 = a.sum()
            if s2 > 1e-15:
                a = a / s2
            else:
                a = best_a.copy()
            conv = fast_convolve(a)
            no_improve = 0
            perturb_count += 1
            if perturb_count > 60:
                break

        if iter_count % 80 == 0:
            top_k = min(2 * n - 1, top_k + 10)

    return best_a, best_obj


def mwu_optimize(a0, time_limit):
    """Multiplicative Weight Update optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    eta = 0.15
    n_peaks = 3

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()

        # Get top peaks
        top_peaks = np.argpartition(conv, -n_peaks)[-n_peaks:]
        peak_vals = conv[top_peaks]
        max_peak = peak_vals.max()

        # Compute log-update
        log_w = np.zeros(n)
        for k in top_peaks:
            k = int(k)
            excess = conv[k] - max_peak * 0.8
            if excess < 0:
                excess = 0.0
            if excess < 1e-15:
                continue
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idxs = np.arange(j_start, j_end + 1)
            log_w[idxs] -= eta * excess * a[k - idxs] / max_peak

        a_new = a * np.exp(log_w)
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn > 1e-15:
            a_new = a_new / sn
        else:
            a_new = best_a.copy()

        if (2.0 * n * fast_convolve(a_new).max()) > best_obj * 1.001:
            # Reverting if too bad
            eta *= 0.8
            continue

        a = a_new
        eta *= 0.998

    return best_a, best_obj


def coordinate_descent(a0, time_limit):
    """Random coordinate descent for fine-tuning."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * len(a) * fast_convolve(a).max()
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    no_improve = 0
    deltas = [0.001, 0.003, 0.005, 0.01, 0.02, -0.001, -0.003, -0.005, -0.01, -0.02]

    while time.monotonic() - t0 < time_limit:
        batch = max(1, 200000 // n)
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit - 0.2:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            found = False
            for delta in deltas:
                if a[i] + delta < 0 or a[j] - delta < 0:
                    continue
                a_try = a.copy()
                a_try[i] += delta
                a_try[j] -= delta
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-11:
                    a = a_try
                    best_obj = obj_try
                    best_a = a.copy()
                    no_improve = 0
                    found = True
                    break
            if not found:
                no_improve += 1
        if no_improve > 300:
            break

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.4))

    # Raised cosine
    rc = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
    rc = np.maximum(rc, 1e-8)
    inits.append(rc / rc.sum())

    # From parent (interpolated)
    if parent is not None and len(parent) > 1:
        pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        pp = np.maximum(pp, 0.0)
        if pp.sum() > 1e-10:
            inits.append(pp / pp.sum())

    # Sine window
    sw = np.sin(np.pi * (x + 0.5) / n) ** 2 + 0.005
    inits.append(sw / sw.sum())

    # Bimodal
    edge = max(2, n // 5)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits.append(bi / bi.sum())

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits.append(g / g.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Hybrid LP + MWU + gradient + coordinate descent search.
    Explores multiple sequence lengths and uses aggressive randomization.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass

    def try_update(a):
        nonlocal best_sequence, best_score
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  ** NEW BEST: score={score:.6f}, n={len(a)} **")

    # =========================================================
    # Phase 1: Quick gradient scan (fast exploration)
    # =========================================================
    print("=== Phase 1: Gradient Scan ===")
    t1 = time.monotonic()

    lengths_to_try = [768, 1024, 1536]
    if parent is not None:
        pn = len(parent)
        if pn not in lengths_to_try:
            lengths_to_try.insert(0, pn)

    phase1_budget = min(180.0, (deadline - time.monotonic()) * 0.15)
    per_init_time = max(8.0, phase1_budget / (len(lengths_to_try) * 3))

    for n in lengths_to_try:
        if time.monotonic() > deadline - 1000:
            break
        inits = generate_inits(n, parent, rng)

        # Prioritize: parent-based, raised cosine, random
        priority_indices = []
        if parent is not None and len(inits) > 3:
            priority_indices.append(3)  # parent interp
        priority_indices.extend([2, 1, 4])  # raised cosine, random, sine
        priority_indices = [i for i in priority_indices if 0 <= i < len(inits)][:3]

        for idx in priority_indices:
            if time.monotonic() > deadline - 900:
                break
            t_alloc = min(per_init_time, deadline - time.monotonic() - 500)
            if t_alloc < 5:
                break
            a_opt, obj = nesterov_grad(inits[idx], time_limit=t_alloc,
                                       lr_init=0.012, momentum=0.91,
                                       top_k=30, temp_init=0.0025)
            try_update(a_opt)

    print(f"Phase 1 done ({time.monotonic()-t1:.0f}s). Best={best_score:.6f}")

    # =========================================================
    # Phase 2: Main LP optimization on best candidate
    # =========================================================
    print("=== Phase 2: LP Optimization ===")
    t2 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 300
        if remaining > 80:
            # Run 1: direct LP from best
            t_lp1 = remaining * 0.5
            a_opt, obj = lp_optimize(best_sequence, time_limit=t_lp1,
                                     top_k_init=50, verbose=True)
            try_update(a_opt)

            # Run 2: perturbed LP
            if time.monotonic() < deadline - 150:
                remaining2 = deadline - time.monotonic() - 120
                noise = rng.standard_normal(len(best_sequence)) * 0.008
                a_noisy = np.maximum(best_sequence + noise, 0)
                if a_noisy.sum() > 1e-10:
                    a_noisy = a_noisy / a_noisy.sum()
                    t_lp2 = min(remaining2 * 0.6, 200.0)
                    a_opt2, obj2 = lp_optimize(a_noisy, time_limit=t_lp2,
                                               top_k_init=40, verbose=True)
                    try_update(a_opt2)

    print(f"Phase 2 done ({time.monotonic()-t2:.0f}s). Best={best_score:.6f}")

    # =========================================================
    # Phase 3: MWU refinement (different optimization landscape)
    # =========================================================
    print("=== Phase 3: MWU ===")
    t3 = time.monotonic()

    if best_sequence is not None and time.monotonic() < deadline - 100:
        t_mwu = min(80.0, (deadline - time.monotonic() - 80) * 0.4)
        if t_mwu > 10:
            a_opt, obj = mwu_optimize(best_sequence, time_limit=t_mwu)
            try_update(a_opt)

    print(f"Phase 3 done ({time.monotonic()-t3:.0f}s). Best={best_score:.6f}")

    # =========================================================
    # Phase 4: More LP with different top_k and from perturbed starts
    # =========================================================
    print("=== Phase 4: LP Diversification ===")
    t4 = time.monotonic()

    if best_sequence is not None and time.monotonic() < deadline - 80:
        remaining = deadline - time.monotonic() - 60
        n_best = len(best_sequence)

        # Try LP from several perturbed starts
        for pi in range(4):
            if time.monotonic() > deadline - 50:
                break
            t_alloc = min(remaining / (5 - pi) * 0.7, 80.0)
            if t_alloc < 10:
                break

            if pi == 0:
                a0 = best_sequence.copy()
            else:
                scale = 0.005 * pi
                noise = rng.standard_normal(n_best) * scale
                a0 = np.maximum(best_sequence + noise, 0)
                s = a0.sum()
                if s > 1e-10:
                    a0 = a0 / s
                else:
                    a0 = best_sequence.copy()

            tk = 30 + pi * 20
            a_opt, obj = lp_optimize(a0, time_limit=t_alloc, top_k_init=tk)
            try_update(a_opt)

    print(f"Phase 4 done ({time.monotonic()-t4:.0f}s). Best={best_score:.6f}")

    # =========================================================
    # Phase 5: Coordinate descent fine-tuning
    # =========================================================
    print("=== Phase 5: Coordinate Descent ===")
    t5 = time.monotonic()

    if best_sequence is not None and time.monotonic() < deadline - 20:
        t_cd = min(50.0, deadline - time.monotonic() - 15)
        if t_cd > 5:
            a_opt, obj = coordinate_descent(best_sequence, time_limit=t_cd)
            try_update(a_opt)

    print(f"Phase 5 done ({time.monotonic()-t5:.0f}s). Best={best_score:.6f}")

    # =========================================================
    # Phase 6: Final short LP polish
    # =========================================================
    print("=== Phase 6: Final Polish ===")
    if best_sequence is not None and time.monotonic() < deadline - 5:
        t_final = min(20.0, deadline - time.monotonic() - 3)
        if t_final > 5:
            a_opt, obj = lp_optimize(best_sequence, time_limit=t_final, top_k_init=60)
            try_update(a_opt)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        try_update(best_sequence)

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
