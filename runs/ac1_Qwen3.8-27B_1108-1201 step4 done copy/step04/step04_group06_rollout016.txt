import numpy as np

from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def lp_direction(a, C=None):
    """
    Solve LP: min t s.t. conv(a,g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns (g, t_val, success).
    """
    n = len(a)
    if C is None:
        C = build_conv_matrix(a)
    
    # Variables: g[0..n-1], t
    # Minimize: t
    # Constraints: C @ g - t <= 0 (2n-1 constraints)
    #              sum(g) = 1
    #              g >= 0, t >= 0 (t will be >= 0 automatically)
    
    # c vector: minimize t → c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # A_ub: [C, -I] @ [g; t] <= 0
    # Shape: (2n-1, n+1)
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = C
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    # A_eq: [ones, 0] @ [g; t] = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: g >= 0, t free (but t >= 0 implicitly from constraints)
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if result.success:
            g = result.x[:n]
            t_val = result.x[n]
            g = np.maximum(g, 0)  # Clean up numerical issues
            g_sum = g.sum()
            if g_sum < 1e-10:
                return None, None, False
            g = g / g_sum  # Renormalize
            return g, t_val, True
    except Exception:
        pass
    return None, None, False

def softmax_gradient(a, conv, top_k=50, temp=0.003):
    """Compute softmax-weighted gradient of the peak autocorrelation."""
    n = len(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum
    
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += w[idx] * 2.0 * a[k - idxs]
    return grad


def nesterov_optimize(a0, time_limit, lr_init=0.01, momentum=0.92, top_k=40, temp_init=0.003):
    """Nesterov-accelerated gradient descent with adaptive parameters."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    y = a.copy()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        y = a + mom * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        
        conv_y = fast_convolve(y)
        grad = softmax_gradient(y, conv_y, top_k=top_k, temp=temp)
        
        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= 0.9997
        temp *= 1.002
        mom = min(0.96, mom + 0.0001)
        
        if no_improve >= 300:
            break
    
    return best_a, best_obj


def coordinate_descent(a0, time_limit):
    """Random coordinate descent for final polishing."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * len(a) * fast_convolve(a).max()
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(99999)
    no_improve = 0
    
    # Precompute for efficiency
    while time.monotonic() - t0 < time_limit:
        for _ in range(max(1, 200000 // n)):
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            
            # Try moving mass from i to j
            for delta in [0.0005, 0.002, 0.005, -0.0005, -0.002, -0.005]:
                if a[i] - delta < 0:
                    continue
                a[i] -= delta
                a[j] += delta
                conv_try = fast_convolve(a)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a.copy()
                    conv = conv_try
                    no_improve = 0
                    break
                else:
                    a[i] += delta
                    a[j] -= delta
            else:
                no_improve += 1
        
        if no_improve > 500:
            break
    
    return best_a, best_obj

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def lp_direction(a, C=None):
    """
    Solve LP: min t s.t. conv(a,g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns (g, t_val, success).
    """
    n = len(a)
    if C is None:
        C = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = C
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'dual_feasibility_tolerance': 1e-10,
                                  'primal_feasibility_tolerance': 1e-10})
        if result.success:

            g = np.maximum(g, 0.0)

            if g_sum < 1e-10:
                return None, None, False
            g = g / g_sum
            return g, t_val, True
    except Exception:
        pass
    return None, None, False

def softmax_gradient(a, conv, top_k=50, temp=0.003):
    """Compute softmax-weighted gradient of the peak autocorrelation."""
    n = len(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum

    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += w[idx] * 2.0 * a[k - idxs]
    return grad

                conv_try = fast_convolve(a)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a.copy()
                    conv = conv_try
                    no_improve = 0
                    improved = True
                    break
                else:
                    a[i] += delta
                    a[j] -= delta
            if not improved:
                no_improve += 1

        if no_improve > 1000:
            break

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Uses LP-based direction finding (min cross-correlation peak)
    with backtracking line search as the primary optimizer, supplemented by
    Nesterov gradient descent and coordinate descent for fine-tuning.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass

    # Initialize
    if parent is not None:
        a_init = parent / parent.sum()
        conv = fast_convolve(a_init)
        init_score = 2.0 * len(a_init) * conv.max()
        print(f"Initial (normalized): score={init_score:.8f}, n={len(a_init)}")
        best_sequence = a_init.copy()
        best_score = float(evaluate_sequence(a_init.tolist()))
        if not np.isfinite(best_score):
            best_score = init_score
    else:
        n = 512
        x = np.arange(n, dtype=np.float64)
        a_init = 0.5 + 0.3 * np.sin(np.pi * x / n) + 0.15 * np.cos(2 * np.pi * x / n)
        a_init = np.maximum(a_init, 0.01)
        a_init = a_init / a_init.sum()
        best_sequence = a_init.copy()
        best_score = 2.0 * n * fast_convolve(a_init).max()
        print(f"Generated initial: n={n}, score={best_score:.6f}")

    # =========================================================
    # Phase 1: Primary LP optimization (majority of time)
    # =========================================================
    print("=== Phase 1: Primary LP Optimization ===")
    t_start = time.monotonic()
    t_alloc_p1 = min(500.0, (deadline - time.monotonic()) * 0.45)

    if t_alloc_p1 > 30:
        a_opt, obj = lp_optimize(best_sequence, time_limit=t_alloc_p1, verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"Phase 1 LP: {score:.8f} *NEW BEST*")
        else:
            print(f"Phase 1 LP: eval={score:.8f}, internal={obj:.8f}")

    # =========================================================
    # Phase 2: Nesterov gradient exploration
    # =========================================================
    print("=== Phase 2: Gradient Exploration ===")
    t_start2 = time.monotonic()

    remaining = deadline - time.monotonic() - 150
    if remaining > 30:
        configs = [
            {'lr_init': 0.006, 'momentum': 0.94, 'top_k': 60, 'temp_init': 0.0015},
            {'lr_init': 0.01, 'momentum': 0.91, 'top_k': 30, 'temp_init': 0.004},
            {'lr_init': 0.004, 'momentum': 0.95, 'top_k': 100, 'temp_init': 0.001},
        ]

        for ci, cfg in enumerate(configs):
            if time.monotonic() > deadline - 80:
                break
            t_alloc = min(60.0, (deadline - time.monotonic() - 60) / max(1, len(configs) - ci))
            if t_alloc < 8:
                break

            a_opt, obj = nesterov_optimize(best_sequence, time_limit=t_alloc, **cfg)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"Phase 2 Grad {ci}: {score:.8f} *NEW BEST*")
            else:
                print(f"Phase 2 Grad {ci}: {score:.8f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.8f}")

    # =========================================================
    # Phase 3: LP second pass (refined)
    # =========================================================
    print("=== Phase 3: LP Second Pass ===")
    t_start3 = time.monotonic()

    remaining = deadline - time.monotonic() - 80
    if remaining > 30 and best_sequence is not None:
        # Try with a small perturbation for diversity
        for attempt in range(2):
            if time.monotonic() > deadline - 40:
                break
            t_alloc = min(remaining / (2 - attempt) * 0.85, 200.0)
            if t_alloc < 15:
                break

            if attempt == 0:
                a_start = best_sequence.copy()
            else:
                noise = rng.standard_normal(len(best_sequence)) * 0.001
                a_start = np.maximum(best_sequence + noise, 0.0)
                a_start = a_start / a_start.sum()

            a_opt, obj = lp_optimize(a_start, time_limit=t_alloc, verbose=True)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"Phase 3 LP attempt {attempt}: {score:.8f} *NEW BEST*")
            else:
                print(f"Phase 3 LP attempt {attempt}: {score:.8f}")

    # =========================================================
    # Phase 4: Try slightly different lengths
    # =========================================================
    print("=== Phase 4: Length Variation ===")
    t_start4 = time.monotonic()

    n_best = len(best_sequence)
    nearby_lengths = []
    for factor in [0.875, 1.125]:
        nl = int(n_best * factor)
        if nl >= 128 and nl <= 2048 and nl != n_best:
            nearby_lengths.append(nl)

    for nl in nearby_lengths[:2]:
        if time.monotonic() > deadline - 40:
            break
        t_alloc = min(40.0, deadline - time.monotonic() - 25)
        if t_alloc < 10:
            break

        a_try = np.interp(np.linspace(0, 1, nl),
                          np.linspace(0, 1, len(best_sequence)),
                          best_sequence)
        a_try = np.maximum(a_try, 0.0)
        a_try = a_try / a_try.sum()

        a_opt, obj = lp_optimize(a_try, time_limit=t_alloc)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"Phase 4 n={nl}: {score:.8f} *NEW BEST*")
        else:
            print(f"Phase 4 n={nl}: {score:.8f}")

    # =========================================================
    # Phase 5: Coordinate descent final polish
    # =========================================================
    print("=== Phase 5: Coordinate Descent ===")
    t_start5 = time.monotonic()

    remaining = deadline - time.monotonic() - 5
    if remaining > 8 and best_sequence is not None:
        t_alloc = min(remaining * 0.9, 45.0)
        if t_alloc > 5:
            a_opt, obj = coordinate_descent(best_sequence, time_limit=t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"Phase 5 CD: {score:.8f} *NEW BEST*")
            else:
                print(f"Phase 5 CD: {score:.8f}")

    # =========================================================
    # Final verification
    # =========================================================
    if best_sequence is not None:
        final_score = float(evaluate_sequence(best_sequence.tolist()))
        if np.isfinite(final_score):
            best_score = final_score

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
