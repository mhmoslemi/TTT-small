"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using aggressive LSE gradient descent
with diverse restarts and peak-targeting heuristics.
Key improvements over previous approach:
- More aggressive beta schedule (exponential ramp)
- Adaptive learning rate with backtracking
- Diverse structured starting points (cosine, two-bump, three-bump)
- Peak-targeting restarts (perturb toward reducing the worst lag)
- Multiple n values explored
- No LP solver (pure gradient-based, allowing many more iterations)
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_gradient(f, beta):
    """Compute the LSE gradient: gradient of (1/beta)*log(sum(exp(beta*conv)))
    which approximates grad(max(conv)).
    Returns (grad, conv_values).
    """
    n = len(f)
    conv = fast_convolve(f)
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()
    
    # Gradient: 2 * cross_correlation(w, f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv_g = np.fft.irfft(W * F, fft_size)
    grad = 2.0 * conv_g[n - 1:n - 1 + n]
    
    return grad, conv, w


def aggressive_lse(f0, time_limit, beta_start=5.0, beta_multiplier=1.008,

    if s < 1e-15:

        f = f / s
    n = len(f)
    
    if lr_start is None:
        lr_start = 3.0 / n
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    beta = beta_start
    lr = lr_start
    momentum = momentum_start
    v = np.zeros(n)
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    consecutive_restarts = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute gradient
        grad, conv, w = lse_gradient(f, beta)
        
        # Adaptive learning rate
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 100
            break
        
        # Step
        v = momentum * v - lr * grad / grad_norm
        f_new = project_simplex(f + v)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            # Slightly increase lr when making progress

            consecutive_restarts = 0
        else:

                momentum = max(0.80, momentum - 0.01)
            if no_improve > 60:

                v = np.zeros(n)
                lr = max(lr * 0.5, 1.0 / n)
                beta = min(beta * 2.0, beta_max)
                momentum = 0.9

def peak_target_optimize(f0, time_limit, seed=42):
    """Target the peak lag specifically with coordinate-level operations."""
    rng = np.random.default_rng(seed)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    lr = 0.01
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        
        # Find top-K lags
        K = min(20, 2 * n - 1)
        top_k = np.argpartition(conv, -K)[-K:]
        
        # Weight: exponential over top-K
        tv = conv[top_k]
        scaled = tv / 0.001
        scaled -= scaled.max()
        w = np.exp(scaled)
        w = w / w.sum()
        
        # Subgradient from top-K lags
        grad = np.zeros(n)
        for idx in range(K):
            k = int(top_k[idx])
            wi = w[idx]
            if wi < 1e-14:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            grad[js:je+1] += 2.0 * wi * f[k - js:k - js + (je - js + 1)]
        
        # Normalize and step
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        f_new = project_simplex(f - lr * grad / gn)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.01, 0.1)
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                lr *= 0.9
            if no_improve > 500:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                lr = 0.01
                no_improve = 0
    
    return best_f, best_obj

def create_diverse_starts(n, parent, rng):
    """Create diverse starting points of length n."""
    starts = []
    x = np.arange(n)
    
    # 1. Uniform

    # 2. Cosine taper
    c = np.cos(np.pi * x / (n - 1)) ** 2
    starts.append(c / c.sum())
    
    # 3. Two bumps at edges
    b = np.zeros(n)
    m = n // 4
    b[:m] = 1.0
    b[n-m:] = 1.0
    starts.append(b / b.sum())
    
    # 4. Three bumps
    b3 = np.zeros(n)
    m3 = n // 6
    b3[:m3] = 1.0
    b3[n//2 - m3//2:n//2 + m3//2] = 1.0
    b3[n-m3:] = 1.0
    starts.append(b3 / b3.sum())
    
    # 5. Parent (if available and same length)
    if parent is not None and len(parent) == n:
        starts.append(parent.copy() / parent.sum())
    elif parent is not None:
        # Interpolate parent to new length
        x_old = np.linspace(0, 1, len(parent))
        x_new = np.linspace(0, 1, n)
        p_interp = np.interp(x_new, x_old, parent / parent.sum())
        p_interp = np.maximum(p_interp, 0)
        if p_interp.sum() > 1e-10:
            starts.append(p_interp / p_interp.sum())
    
    # 6. Linear ramp (decreasing)
    r = np.linspace(1.0, 0.1, n)
    starts.append(r / r.sum())
    
    # 7. Sine wave
    sw = np.abs(np.sin(2 * np.pi * x / n)) ** 2 + 0.1
    starts.append(sw / sw.sum())
    
    # 8. Random
    r_vec = rng.standard_normal(n)
    r_vec = np.maximum(r_vec, 0)
    if r_vec.sum() > 1e-10:
        starts.append(r_vec / r_vec.sum())
    
    return starts

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using aggressive LSE gradient descent
with diverse structured restarts, adaptive momentum, and small active-set LP steps.
Key differences from previous approach:
- Much faster beta ramp (exponential, doubling periodically)

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_grad_and_conv(f, beta):
    """Compute LSE gradient and convolution."""
    n = len(f)
    conv = fast_convolve(f)
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()
    # Gradient: 2 * cross_correlation(w, f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv_g = np.fft.irfft(W * F, fft_size)
    grad = 2.0 * conv_g[n - 1:n - 1 + n]
    return grad, conv, w


def small_active_lp(f, M, top_k_lags, max_time=2.0):
    """Solve small LP: max sum(g) s.t. (f*g)[k] <= M for k in top_k_lags, g >= 0."""
    n = len(f)
    k_list = list(top_k_lags)
    K = len(k_list)
    if K == 0:
        return None
    # Build constraint matrix: for each lag k, row k has f[k-j] at column j
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(k_list):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(f[k - j])
    if not rows:
        return None
    A = csr_matrix((vals, (rows, cols)), shape=(K, n))
    c = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': max_time})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def create_starts(n, parent, rng):
    """Create diverse starting points of length n."""
    starts = []
    x = np.arange(n)

    # Uniform
    starts.append(np.ones(n) / n)

    # Cosine taper
    c = np.cos(np.pi * x / (n - 1)) ** 2
    starts.append(c / c.sum())

    # Raised cosine (edges higher)
    rc = 1.0 - np.cos(np.pi * x / (n - 1))
    rc = rc ** 2 + 0.5
    starts.append(rc / rc.sum())

    # Sine wave
    sw = np.abs(np.sin(2 * np.pi * x / n)) ** 2 + 0.2
    starts.append(sw / sw.sum())

    # Double sine
    ds = np.abs(np.sin(4 * np.pi * x / n)) ** 2 + 0.3
    starts.append(ds / ds.sum())

    # Linear ramp
    r = np.linspace(1.0, 0.3, n)
    starts.append(r / r.sum())

    # Three bump
    b3 = np.zeros(n)
    m = n // 8
    b3[:m] = 1.0
    b3[n // 2 - m: n // 2 + m] = 1.0
    b3[n - m:] = 1.0
    starts.append(b3 / b3.sum())

    # Parent (interpolated if needed)
    if parent is not None:
        if len(parent) == n:
            starts.append(parent.copy() / parent.sum())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            p_int = np.interp(x_new, x_old, parent / parent.sum())
            p_int = np.maximum(p_int, 0)
            if p_int.sum() > 1e-10:
                starts.append(p_int / p_int.sum())

    # Random nonnegative
    rv = np.abs(rng.standard_normal(n))
    if rv.sum() > 1e-10:
        starts.append(rv / rv.sum())

    return starts

                                     use_lp=True, lp_every=20, lp_k=10,
                                     seed=seed + si * 100)
        try_sequence(f_opt, best_tracker)

    # Phase 2: Optimize at nearby lengths
    print("\n=== Phase 2: Nearby lengths ===")
    alt_lengths = []
    for dn in [1, -1, 2, -2, 3, 4, -3]:
        n_try = n_parent + dn
        if 32 <= n_try <= 600 and n_try != n_parent:
            alt_lengths.append(n_try)
    alt_lengths = alt_lengths[:4]

    for n_alt in alt_lengths:
        t_left = deadline - time.monotonic() - 200
        if t_left < 30:
            break
        t_alloc = min(t_left * 0.25, 120)
        if t_alloc < 15:
            continue
        print(f"  n={n_alt} ({t_alloc:.0f}s)...")
        # Interpolate parent
        x_old = np.linspace(0, 1, n_parent)
        x_new = np.linspace(0, 1, n_alt)
        f_start = np.interp(x_new, x_old, parent)
        f_start = np.maximum(f_start, 0)
        if f_start.sum() > 1e-10:
            f_start = f_start / f_start.sum()
            f_opt, obj = aggressive_lse(f_start, time_limit=t_alloc,
                                         beta_start=5.0, beta_rate=1.015,
                                         beta_max=100000.0,
                                         lr_start=3.0 / n_alt,
                                         use_lp=True, lp_every=15, lp_k=8,
                                         seed=seed + 500 + n_alt)
            try_sequence(f_opt, best_tracker)

    # Phase 3: Sustained search from best
    print("\n=== Phase 3: Sustained search ===")
    if best_tracker[1] is not None:
        t_left = deadline - time.monotonic() - 30
        if t_left > 60:
            print(f"  Time: {t_left:.0f}s")
            f_sust, obj_sust = sustained_search(best_tracker[1], time_limit=t_left,
                                                 seed=seed + 9000)
            try_sequence(f_sust, best_tracker)

    # Phase 4: Final aggressive LSE from best with high beta
    print("\n=== Phase 4: Final high-beta polish ===")
    if best_tracker[1] is not None:
        t_left = deadline - time.monotonic() - 15
        if t_left > 20:
            n_best = len(best_tracker[1])
            f_final, obj_final = aggressive_lse(best_tracker[1], time_limit=t_left,
                                                 beta_start=100.0, beta_rate=1.03,
                                                 beta_max=500000.0,
                                                 lr_start=1.5 / n_best,
                                                 mom_start=0.88,
                                                 use_lp=False,
                                                 seed=seed + 9999)
            try_sequence(f_final, best_tracker)

    # Verify
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    # Fallback
    fb = np.ones(448) / 448
    print(f"\nFALLBACK: score={evaluate_sequence(fb.tolist()):.10f}")
    return [float(x) for x in fb]
