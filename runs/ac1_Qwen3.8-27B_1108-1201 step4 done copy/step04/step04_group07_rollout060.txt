    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    base_delta = 0.008
    restart_count = 0

    while time.monotonic() - t0 < time_limit:
        no_improve = 0
        converged = False

        while no_improve < 80 and time.monotonic() - t0 < time_limit:
            # Find top peaks
            top_k = min(3, len(conv))
            peak_indices = np.argpartition(conv, -top_k)[-top_k:]

            # For the main peak, find most contributing coordinate
            k = int(peak_indices[0])
            js = max(0, k - n + 1)
            je = min(n - 1, k)

            if je < js:
                no_improve += 1
                continue

            idx = np.arange(js, je + 1)
            # Contribution of f[j] to lag k: f[j] * f[k-j]
            contrib = f[idx] * np.maximum(f[k - idx], 0.0)
            i_src = int(idx[np.argmax(contrib)])

            # Candidate destinations: spread across the sequence
            n_dest = min(6, n)
            dest_candidates = np.linspace(0, n - 1, n_dest + 2)[1:n_dest + 1].astype(int)
            dest_candidates = [d for d in dest_candidates if d != i_src and f[d] < f[i_src] + base_delta]
            if len(dest_candidates) > 4:
                dest_candidates = dest_candidates[:4]
                rng.shuffle(dest_candidates)

            if not dest_candidates:
                no_improve += 1
                continue

            # Try transfers
            best_improve = 1e-14
            best_move = None

            for dest in dest_candidates:
                for dm in [0.4, 0.7, 1.0, 1.5, 2.5]:
                    delta = base_delta * dm
                    if f[i_src] < delta:
                        continue
                    f_try = f.copy()
                    f_try[i_src] -= delta
                    f_try[dest] += delta
                    obj_try = fast_convolve(f_try).max()
                    improve = best_obj - obj_try
                    if improve > best_improve:
                        best_improve = improve
                        best_move = (i_src, dest, delta)

            if best_move is not None:
                i, j, d = best_move
                f[i] -= d
                f[j] += d
                fsum = f.sum()
                if fsum < 1e-10:
                    f = best_f.copy()
                    continue
                f = f / fsum
                conv = fast_convolve(f)
                obj = conv.max()
                if obj < best_obj:
                    best_obj = obj
                    best_f = f.copy()
                    no_improve = 0
                else:
                    f = best_f.copy()
                    conv = fast_convolve(f)
                    no_improve += 1
            else:
                no_improve += 1

        # Restart with perturbation
        restart_count += 1
        scale = min(0.01, 0.002 * restart_count)
        f = project_simplex(best_f + rng.standard_normal(n) * scale)
        conv = fast_convolve(f)
        base_delta *= 0.88
        if base_delta < 5e-5:
            base_delta = 0.003

    return best_f, best_obj


def subgrad_optimize(f0, time_limit, rng):
    """Adaptive softmax subgradient with momentum and cooling."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.004
    mom = 0.93
    temp = 0.0003
    top_k = max(10, n // 5)
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive learning rate
        lr *= 0.9998
        temp *= 1.003

        if no_improve > 200:
            scale = 0.003 + 0.001 * min(no_improve // 50, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            lr = max(0.002, lr * 2)
            temp = max(0.0001, temp * 0.5)
            no_improve = 0

    return best_f, best_obj


def fine_tune(f0, time_limit, rng):
    """Very fine coordinate-wise optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    step = 1e-4

    while time.monotonic() - t0 < time_limit:
        improved = False
        # Try small perturbations at random coordinate pairs
        for _ in range(200):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            if f[i] < step:
                continue
            f_try = f.copy()
            f_try[i] -= step
            f_try[j] += step
            obj = fast_convolve(f_try).max()
            if obj < best_obj - 1e-15:
                f = f_try / f_try.sum()
                conv = fast_convolve(f)
                best_obj = obj
                best_f = f.copy()
                improved = True

        if not improved:
            step *= 0.7
            if step < 1e-7:
                break

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)

    # Exponential decay (asymmetric - key insight for non-negative sequences)
    x = np.arange(n, dtype=np.float64)
    for rate in [0.02, 0.05, 0.1, 0.15]:
        e = np.exp(-rate * x)
        inits.append(e / e.sum())
    for rate in [0.02, 0.05, 0.1]:
        e = np.exp(-rate * (n - 1 - x))
        inits.append(e / e.sum())

    # Gaussian
    for sigma_frac in [0.15, 0.3, 0.5]:
        sigma = sigma_frac * n
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / sigma) ** 2)
        inits.append(g / g.sum())

    # Asymmetric Gaussian (shifted center)
    for center_frac in [0.3, 0.7]:
        c = center_frac * (n - 1)
        g = np.exp(-0.5 * ((x - c) / (0.3 * n)) ** 2)
        inits.append(g / g.sum())

    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c1 / c1.sum())

    # Power law
    p = (x + 1) ** 0.5
    inits.append(p / p.sum())
    p2 = (n - x) ** 0.5
    inits.append(p2 / p2.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase hybrid search:
    Phase 1: Initialization and evaluation
    Phase 2: LP-based Frank-Wolfe (primary)
    Phase 3: Transfer local search (novel)
    Phase 4: Subgradient polishing
    Phase 5: Fine-tuning
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
    except Exception:
        pass

    # === Phase 1: Initialization ===
    print("=== Phase 1: Initialization ===")

    if parent is not None:
        n_main = len(parent)
    else:
        n_main = 384
        parent = np.ones(n_main) / n_main

    # Evaluate parent
    try_seq(parent, "parent")

    # Try structured initializations at main length
    inits = generate_inits(n_main, parent, rng)
    for i, ini in enumerate(inits):
        if time.monotonic() > deadline - 600:
            break
        try_seq(ini, f"init_{i}")

    # Try a few other lengths
    for n_try in [384, 512, 640, 768, 256]:
        if n_try == n_main:
            continue
        if time.monotonic() > deadline - 500:
            break
        # Use the best known shape interpolated to new length
        ref = best_seq if best_seq is not None else parent
        px = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, len(ref)), ref)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            px = px / ps
            try_seq(px, f"len{n_try}")

    if best_seq is None:
        best_seq = np.ones(256) / 256
        try_seq(best_seq, "fallback")

    print(f"  Phase 1 best: {best_score:.8f} (n={len(best_seq)})")

    # === Phase 2: LP-based Frank-Wolfe (primary) ===
    print("\n=== Phase 2: LP Frank-Wolfe ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic() - 300
    t_alloc = min(remaining * 0.45, 600)

    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s, n={n_best}")
        a_opt, _ = fw_optimize(best_seq, time_limit=t_alloc * 0.7, verbose=True)
        try_seq(a_opt, "fw1")

        # Second run from perturbed start
        rem = deadline - time.monotonic() - 250
        if rem > 30:
            pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.008)
            a_opt2, _ = fw_optimize(pert, time_limit=min(rem * 0.5, 200))
            try_seq(a_opt2, "fw2_pert")

    print(f"  Phase 2 best: {best_score:.8f} (n={len(best_seq)})")

    # === Phase 3: Transfer Local Search (novel) ===
    print("\n=== Phase 3: Transfer Local Search ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic() - 200
    t_alloc = min(remaining * 0.35, 400)

    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_opt, _ = transfer_local_search(best_seq, time_limit=t_alloc * 0.6, rng=rng)
        try_seq(a_opt, "transfer1")

        rem = deadline - time.monotonic() - 150
        if rem > 20:
            a_opt2, _ = transfer_local_search(best_seq, time_limit=min(rem * 0.5, 150), rng=rng)
            try_seq(a_opt2, "transfer2")

    print(f"  Phase 3 best: {best_score:.8f} (n={len(best_seq)})")

    # === Phase 4: Subgradient Polish ===
    print("\n=== Phase 4: Subgradient Polish ===")
    remaining = deadline - time.monotonic() - 100
    t_alloc = min(remaining * 0.35, 250)

    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_opt, _ = subgrad_optimize(best_seq, time_limit=t_alloc * 0.6, rng=rng)
        try_seq(a_opt, "sg1")

        rem = deadline - time.monotonic() - 60
        if rem > 15:
            a_opt2, _ = subgrad_optimize(best_seq, time_limit=min(rem * 0.5, 100), rng=rng)
            try_seq(a_opt2, "sg2")

    print(f"  Phase 4 best: {best_score:.8f} (n={len(best_seq)})")

    # === Phase 5: Fine-tuning ===
    print("\n=== Phase 5: Fine-tuning ===")
    remaining = deadline - time.monotonic() - 30
    t_alloc = min(remaining * 0.8, 150)

    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_opt, _ = fine_tune(best_seq, time_limit=t_alloc * 0.5, rng=rng)
        try_seq(a_opt, "fine1")

        rem = deadline - time.monotonic() - 20
        if rem > 10:
            # One more FW pass on the fine-tuned result
            a_opt2, _ = fw_optimize(best_seq, time_limit=min(rem * 0.6, 60))
            try_seq(a_opt2, "fw_final")

    # === Try alternate lengths with best shape ===
    rem = deadline - time.monotonic() - 15
    if rem > 20:
        n_best = len(best_seq)
        for n_try in [n_best + 16, n_best - 16, n_best + 32]:
            if n_try < 64 or rem < 15:
                continue
            px = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, n_best), best_seq)
            px = np.maximum(px, 0)
            ps = px.sum()
            if ps > 1e-10:
                px = px / ps
                a_opt, _ = fw_optimize(px, time_limit=min(rem * 0.4, 30))
                try_seq(a_opt, f"alt_len{n_try}")
                rem = deadline - time.monotonic() - 15

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f}")

    return [float(x) for x in best_seq]
