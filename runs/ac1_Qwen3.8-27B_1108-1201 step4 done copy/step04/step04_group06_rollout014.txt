        # c[k] = sum_{j} a[j]*a[k-j] where 0<=j<n and 0<=k-j<n
        # dc[k]/da[i]: i appears in the sum as j=i (giving a[k-i]) and as k-j=i i.e. j=k-i (giving a[i])
        # So dc[k]/da[i] = a[k-i] (if 0<=k-i<n) + a[i] (if 0<=k-i<n)
        # Wait no: dc[k]/da[i] = a[k-i] + a[k-i] if both j=i and j=k-i are valid and distinct
        # If j=i: term is a[i]*a[k-i], derivative w.r.t. a[i] is a[k-i]
        # If j=k-i: term is a[k-i]*a[i], derivative w.r.t. a[i] is a[k-i]
        # If i == k-i (2i==k): same term counted once, derivative is 2*a[i]
        # So: dc[k]/da[i] = 2*a[k-i] if 0<=k-i<n and i!=k-i
        #                          2*a[i] if 2i==k (which is same as 2*a[k-i])
        # In all cases: dc[k]/da[i] = 2*a[k-i] if 0<=k-i<n (and k != 0 or i != 0 edge cases)
        # Wait, that's only true when i != k-i. When i = k-i, the term a[i]*a[i] has derivative 2*a[i] = 2*a[k-i].
        # So uniformly: dc[k]/da[i] = 2*a[k-i] if 0 < k-i < n-1... 
        # Hmm, let me just do it directly.
        
        # Valid j range: j from max(0, k-n+1) to min(k, n-1)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1)
        # c[k] = sum_{j in js} a[j]*a[k-j]
        # d(dk)/da[i] for i in js: contributes a[k-i] from the a[k-j] factor
        # d(dk)/da[i] for i = k-j for j in js (i.e., i = k-j_start to k-j): contributes a[j] = a[k-i]
        # So for each valid i: dc[k]/da[i] = a[k-i] + a[k-i] = 2*a[k-i] (if i != k-i)
        # or 2*a[i] (if i == k-i), which is the same value.
        # So: for i in [j_start, j_end]: dc[k]/da[i] = 2*a[k-i]
        # AND for i in [k-j_end, k-j_start]: dc[k]/da[i] = 2*a[k-i]
        # These two ranges might overlap or be the same.
        
        # Actually the simplest: dc[k]/da[i] is nonzero for i where either i in [j_start,j_end] or (k-i) in [j_start,j_end]
        # i.e., i in [j_start,j_end] OR i in [k-j_end, k-j_start]
        # And the value is a[k-i] + a[k-i] = 2*a[k-i] if both conditions hold and i != k-i
        # or just a[k-i] if only one condition holds (but for auto-convolution both always hold by symmetry of valid range)
        
        # For auto-convolution: if i is in [j_start, j_end], then k-i is in [k-j_end, k-j_start]
        # and since j_start = max(0,k-n+1), k-j_start = min(k, 2n-1-k) ... this is getting complex.
        
        # Simplest correct formula: dc[k]/da[i] = a[k-i] * (1 if 0<=k-i<n else 0) + a[i] * (1 if 0<=k-i<n else 0)
        # Wait no. Let me just use: dc[k]/da[i] = a[k-i] if 0 <= k-i < n, else 0
        # PLUS a[k-i] if 0 <= k-i < n, else 0 (from the symmetric term)
        # = 2*a[k-i] if 0 <= k-i < n, else 0
        # WAIT. That's only if k-i is a valid index for the OTHER factor.
        # c[k] = sum_j a[j] * a[k-j]
        # The term where j = i: a[i]*a[k-i], valid if 0<=i<n and 0<=k-i<n. Derivative wrt a[i]: a[k-i]
        # The term where k-j = i, i.e., j = k-i: a[k-i]*a[i], valid if 0<=k-i<n and 0<=i<n. Derivative wrt a[i]: a[k-i]
        # These are the SAME term when i = k-i. Otherwise they're different terms.
        # So dc[k]/da[i] = a[k-i] + a[k-i] = 2*a[k-i] if 0<=k-i<n (and i in [0,n-1])
        #                  = 0 otherwise
        
        # So: grad[i] += w[idx] * 2*a[k-i] for i where 0 <= k-i < n
        # i.e., i in [k-(n-1), k] intersected with [0, n-1]
        
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        ii = np.arange(i_start, i_end + 1)
        grad[ii] += 2.0 * w[idx] * a[k - ii]
    
    return grad


def lp_direction(a, M, C=None, top_constraints=None):
    """
    Solve LP: max 1^T g s.t. C[:top_gold] @ g <= M, g >= 0.
    Uses only the top 'top_constraints' rows of C for speed.
    Returns (g, success).
    """
    n = len(a)
    if C is None:
        C = build_sparse_conv_matrix(a)
    
    # Determine which constraints to use
    if top_constraints is not None:
        # Use only the most active constraints
        C_sub = C[top_constraints]
        b = np.full(len(top_constraints), M)
    else:
        C_sub = C
        b = np.full(2 * n - 1, M)
    
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=C_sub, b_ub=b, bounds=bounds,
                         method='highs', options={'presolve': True,
                                                   'dual_feasibility_tolerance': 1e-9,
                                                   'primal_feasibility_tolerance': 1e-9})
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0)
            if g.sum() > 1e-10:
                return g, True
    except Exception:
        pass
    return None, False


def golden_section_search(f, a, d, t_lo=0.0, t_hi=1.0, tol=0.01, max_iter=30):
    """Golden section search for optimal step t in a + t*d."""
    n = len(a)
    gr = (np.sqrt(5) + 1) / 2  # golden ratio
    c_ = 1.0 / gr
    d_ = 2.0 - gr
    
    best_t = t_lo
    best_f = f(a)
    
    for _ in range(max_iter):
        t1 = c_ * t_lo + d_ * t_hi
        t2 = d_ * t_lo + c_ * t_hi
        if abs(t1 - t2) < tol * max(1, abs(t_lo + t_hi)):
            break
        f1 = f(a + t1 * d)
        f2 = f(a + t2 * d)
        if f1 < f2:
            t_hi = t2
            if f1 < best_f:
                best_f = f1
                best_t = t1
        else:
            t_lo = t1
            if f2 < best_f:
                best_f = f2
                best_t = t2
    
    return best_t, best_f


def lp_optimize_sequence(a0, time_limit, verbose=False):
    """
    LP-based optimization with golden-section step sizing and adaptive constraints.
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a), dtype=np.float64) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M / (a.sum() ** 2)
    best_a = a.copy()
    
    C = build_sparse_conv_matrix(a)
    
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 50
    perturb_count = 0
    rng = np.random.default_rng(42 + n)
    
    while time.monotonic() - t0 < time_limit:
        # Identify active constraints (top peaks)
        top_k_constraints = min(2 * n - 1, max(50, n // 4))
        active_idx = np.argpartition(conv, -top_k_constraints)[-top_k_constraints:]
        
        # Solve LP with reduced constraints
        g, success = lp_direction(a, M * (1 + 1e-6), C, active_idx)
        
        if not success or g is None:
            no_improve += 1
        else:
            g = g / g.sum()  # Normalize
            d = g - a
            dn = np.linalg.norm(d)
            
            if dn < 1e-10:
                no_improve += 1
            else:
                d = d / dn  # Normalize direction
                obj_at_a = 2.0 * n * M / (a.sum() ** 2)
                
                # Line search
                def step_obj(t):
                    a_try = a + t * dn * d
                    a_try = np.maximum(a_try, 0.0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        return np.inf
                    conv_try = fast_convolve(a_try)
                    return 2.0 * n * conv_try.max() / (s_try ** 2)
                
                t_best, obj_best = golden_section_search(step_obj, a, d, 0.0, 2.0, tol=0.05, max_iter=20)
                
                if obj_best < best_obj - 1e-10:
                    a_new = a + t_best * dn * d
                    a_new = np.maximum(a_new, 0.0)
                    s_new = a_new.sum()
                    if s_new > 1e-15:
                        a_new = a_new / s_new
                        a = a_new
                        conv = fast_convolve(a)
                        M = conv.max()
                        best_obj = 2.0 * n * M / (a.sum() ** 2)
                        best_a = a.copy()
                        no_improve = 0
                        # Rebuild matrix (a changed)
                        C = build_sparse_conv_matrix(a)
                    else:
                        no_improve += 1
                else:
                    no_improve += 1
        
        if verbose and no_improve % 20 == 0:
            print(f"    LP: iter no_improve={no_improve}, obj={best_obj:.8f}, t={time.monotonic()-t0:.1f}s")
        
        if no_improve >= max_no_improve:
            # Perturb to escape
            scale = 0.001 + 0.002 * min(perturb_count, 10)
            # Target the peak: increase elements that don't contribute to peak, decrease those that do
            peak_lag = int(np.argmax(conv))
            # Elements contributing to peak: a[i] where 0 <= peak_lag-i < n
            contrib = np.zeros(n)
            i_start = max(0, peak_lag - n + 1)
            i_end = min(n - 1, peak_lag)
            contrib[i_start:i_end+1] = a[peak_lag - np.arange(i_start, i_end+1)]
            
            noise = rng.standard_normal(n) * scale
            # Bias: reduce contributing elements more
            noise -= scale * 0.5 * contrib / max(contrib.max(), 1e-10)
            
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            else:
                a = best_a.copy()
            conv = fast_convolve(a)
            M = conv.max()
            C = build_sparse_conv_matrix(a)
            no_improve = 0
            perturb_count += 1
            max_no_improve = min(200, 50 + 25 * perturb_count)
            
            if perturb_count > 30:
                break
    
    return best_a, best_obj


def gradient_optimize(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=30, tau_init=0.003):
    """Nesterov-accelerated gradient descent on smoothed max."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    y = a.copy()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    tau = tau_init
    mom = momentum
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        y = a + mom * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        else:
            y = a.copy()
        
        conv_y = fast_convolve(y)
        grad = smoothed_max_gradient(y, conv_y, top_k=top_k, tau=tau)
        
        # Subtract mean to preserve sum=1 (project gradient onto simplex tangent space)
        grad = grad - grad.mean()
        
        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive params
        lr *= 0.9997
        tau *= 1.002
        mom = min(0.97, mom + 0.0003)
        
        if no_improve >= 300:
            break
    
    return best_a, best_obj


def coordinate_polish(a0, time_limit):
    """Random coordinate descent: move mass between pairs."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        for _ in range(max(1, int(200000 / n))):
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            for delta in [0.002, 0.005, 0.01, 0.02, -0.002, -0.005, -0.01, -0.02]:
                if a[i] + delta < -1e-15 or a[j] - delta < -1e-15:
                    continue
                a[i] += delta
                a[j] -= delta
                conv_try = fast_convolve(a)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-10:
                    best_obj = obj_try
                    best_a = a.copy()
                    conv = conv_try
                    no_improve = 0
                    break
                else:
                    a[i] -= delta
                    a[j] += delta
        else:
            no_improve += 1
        
        if no_improve > 100:
            break
    
    return best_a


def generate_init_geometric(n, ratio=0.97):
    """Geometric sequence from one side."""
    x = np.arange(n, dtype=np.float64)
    a = ratio ** x
    return a / a.sum()


def generate_init_sine(n, freq=1.0):
    """Sine-based initialization."""
    x = np.arange(n, dtype=np.float64) / n
    a = (np.sin(freq * np.pi * x) ** 2 + 0.01)
    return a / a.sum()


def generate_init_bimodal(n, edge_frac=0.3):
    """Bimodal: high at edges."""
    a = np.zeros(n)
    edge = max(1, int(n * edge_frac))
    a[:edge] = 1.0
    a[-edge:] = 1.0
    return a / a.sum()


def generate_init_poly(n, coeffs):
    """Polynomial-shaped initialization."""
    x = np.arange(n, dtype=np.float64) / max(1, n - 1)
    a = np.ones(n)
    for c in coeffs:
        a = a + c * (x ** c[1]) if isinstance(c, tuple) else a + c * x
    a = np.maximum(a, 0.01)
    return a / a.sum()
