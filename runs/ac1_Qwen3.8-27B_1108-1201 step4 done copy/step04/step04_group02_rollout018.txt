"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using structured Newton methods
with Hankel Hessian-matrix products via FFT, combined with LP-based Frank-Wolfe
escapes and multi-scale exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def hankel_mv(w, v):
    """Compute H*v where H[i,l] = 2*w[i+l] (Hankel matrix from weight vector w)."""
    n = len(v)
    w_rev = w[::-1]
    size = 1 << (len(w_rev) + n - 1).bit_length()
    W = np.fft.rfft(w_rev, size)
    V = np.fft.rfft(v, size)
    conv = np.fft.irfft(W * V, size)[:len(w_rev) + n - 1]
    # result[i] = conv[2*n-2-i] for i = 0, ..., n-1
    idx = 2 * n - 2 - np.arange(n)
    return 2.0 * conv[idx]

def compute_weights(c, top_k=32, temp=0.01):
    """Compute softmax weights over top peaks of convolution."""
    if top_k >= len(c):
        peaks = np.arange(len(c))
    else:
        peaks = np.argpartition(c, -top_k)[-top_k:]
    pv = c[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w_full = np.zeros(len(c))
    w_vals = np.exp(scaled)
    wsum = w_vals.sum()
    if wsum < 1e-30:
        w_full[peaks[np.argmax(pv)]] = 1.0
    else:
        w_full[peaks] = w_vals / wsum
    return w_full

def compute_gradient(a, w):
    """Compute gradient of weighted sum of c[k] = (a*a)[k]."""
    n = len(a)
    # grad[i] = 2 * sum_k w[k] * a[k-i] (with bounds)
    # This is 2 * (w * a_reversed) ... let me think.
    # (a*a)[k] = sum_j a[j]*a[k-j]
    # d/d a[i] = 2 * a[k-i] (with zero extension)
    # So grad[i] = sum_k w[k] * 2*a[k-i] = 2 * sum_k w[k] * a[k-i]
    # Let's compute: for each i, sum over k where 0<=k-i<n: w[k] * a[k-i]
    # = sum_j w[i+j] * a[j] for j = 0, ..., n-1 (where i+j < 2n-1)
    # This is the cross-correlation of w and a!
    # result[i] = sum_j w[i+j] * a[j]
    # = (w_rev * a)[2n-2-i] (convolution)
    w_rev = w[::-1]
    size = 1 << (len(w_rev) + n - 1).bit_length()
    W = np.fft.rfft(w_rev, size)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(W * A, size)[:len(w_rev) + n - 1]
    idx = 2 * n - 2 - np.arange(n)
    return 2.0 * conv[idx]

def cg_solve(w, rhs, max_iter=50, tol=1e-10, reg=1e-8):
    """Solve H*x = rhs using CG, where H[i,l] = 2*w[i+l] + reg*I."""
    n = len(rhs)
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    r_norm_sq = np.dot(r, r)
    if r_norm_sq < tol * tol:
        return x
    for it in range(max_iter):
        Hp = hankel_mv(w, p) + reg * p
        p_Hp = np.dot(p, Hp)
        if p_Hp < 1e-15:
            break
        alpha = r_norm_sq / p_Hp
        x += alpha * p
        r -= alpha * Hp
        new_r_norm_sq = np.dot(r, r)
        if new_r_norm_sq < tol * r_norm_sq:
            break
        p = r + (new_r_norm_sq / r_norm_sq) * p
        r_norm_sq = new_r_norm_sq
    return x

def newton_optimize(a0, time_limit, top_k=32, temp_init=0.001, temp_growth=1.005,
                     max_cg_iter=50, reg=1e-6, verbose=False):
    """Structured Newton optimization with CG and line search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_temp = temp_init
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1:
            break
        
        # Compute weights and gradient
        w = compute_weights(c, top_k=top_k, temp=cur_temp)
        g = compute_gradient(a, w)
        
        # Compute Newton direction
        d = -cg_solve(w, g, max_iter=max_cg_iter, reg=reg)
        
        # If CG failed, use plain gradient
        if np.max(np.abs(d)) < 1e-15:
            d = -g / (np.linalg.norm(g) + 1e-15)
        
        # Line search
        best_alpha = 0.0
        best_val = best_obj
        alphas = [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125]
        for alpha in alphas:
            a_try = project_simplex(a + alpha * d)
            c_try = fast_convolve(a_try)
            val = c_try.max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            a = project_simplex(a + best_alpha * d)
            c = fast_convolve(a)
            obj = c.max()
            
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            cur_temp *= temp_growth
        else:
            no_improve += 1
            cur_temp *= temp_growth
        
        if no_improve > 50:
            # Escape: add noise
            rng = np.random.default_rng(it * 7 + 3)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            c = fast_convolve(a)
            no_improve = 0
            cur_temp = temp_init
    
    return best_a, best_obj

def solve_lp_oracle(f, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    n = len(f)
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def fw_escape(f, M, time_limit=30):
    """Frank-Wolfe step using LP oracle."""
    n = len(f)
    g = solve_lp_oracle(f, M, time_limit)
    if g is None:
        return None
    gs = g.sum()
    if gs < 1e-15:
        return None
    g = g / gs
    return g

def generate_inits(n, parent=None, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    inits.append(np.ones(n) / n)
    
    ramp = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(ramp / ramp.sum())
    
    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            blended = 0.5 * (px / ps) + 0.5 * (ramp / ramp.sum())
            inits.append(blended / blended.sum())
    
    # Random Dirichlet
    for alpha in [0.5, 1.0, 2.0]:
        inits.append(rng.dirichlet(np.ones(n) * alpha))
    
    # Bimodal
    center = n // 2
    bimodal = np.exp(-(x - center*0.3)**2 / (n*0.05)) + np.exp(-(x - center*0.7)**2 / (n*0.05))
    inits.append(bimodal / bimodal.sum())
    
    # White noise
    wn = np.abs(rng.standard_normal(n))

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using structured Newton methods with
Hankel Hessian-matrix products via FFT, combined with LP-based Frank-Wolfe
escapes, multi-scale exploration, and adaptive parameter tuning.

The key insight is that the Hessian of max_k (a*a)[k] has a Hankel structure:
H[i,l] = 2*w[i+l], enabling O(n log n) matrix-vector products via FFT.
This allows efficient Newton-type optimization via conjugate gradient.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT, matching np.convolve(a, a)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(w, v):
    """Compute H*v where H[i,l] = 2*w[i+l] (Hankel matrix from weight vector w of length 2n-1)."""
    n = len(v)
    w_rev = w[::-1]
    total_len = len(w_rev) + n - 1
    size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w_rev, size)
    V = np.fft.rfft(v, size)
    conv = np.fft.irfft(W * V, size)[:total_len]
    idx = 2 * n - 2 - np.arange(n)
    return 2.0 * conv[idx]


def compute_weights(c, top_k=32, temp=0.01):
    """Compute softmax weights concentrated on top peaks of convolution."""
    n_peaks = min(top_k, len(c))
    if n_peaks == len(c):
        peaks = np.arange(len(c))
    else:
        peaks = np.argpartition(c, -n_peaks)[-n_peaks:]
    pv = c[peaks]
    scaled = pv / max(temp, 1e-12)
    scaled -= scaled.max()
    w_full = np.zeros(len(c))
    w_vals = np.exp(scaled)
    wsum = w_vals.sum()
    if wsum < 1e-30:
        w_full[peaks[np.argmax(pv)]] = 1.0
    else:
        w_full[peaks] = w_vals / wsum
    return w_full


def compute_gradient(a, w):
    """Compute gradient of sum_k w[k]*c[k] where c = a*a."""
    n = len(a)
    # grad[i] = 2 * sum_k w[k] * a[k-i] (zero-extended)
    # = 2 * sum_j w[i+j] * a[j] for j=0..n-1
    # This is the cross-correlation, computable via FFT
    w_rev = w[::-1]
    total_len = len(w_rev) + n - 1
    size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w_rev, size)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(W * A, size)[:total_len]
    idx = 2 * n - 2 - np.arange(n)
    return 2.0 * conv[idx]


def cg_solve_hankel(w, rhs, max_iter=60, tol=1e-10, reg=1e-6):
    """Solve (H + reg*I)*x = rhs using CG, where H[i,l] = 2*w[i+l]."""
    n = len(rhs)
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    r_norm_sq = np.dot(r, r)
    if r_norm_sq < 1e-30:
        return x
    initial_norm = r_norm_sq
    for _ in range(max_iter):
        Hp = hankel_mv(w, p) + reg * p
        p_Hp = np.dot(p, Hp)
        if abs(p_Hp) < 1e-15:
            break
        alpha = r_norm_sq / p_Hp
        x += alpha * p
        r -= alpha * Hp
        new_r_norm_sq = np.dot(r, r)
        if new_r_norm_sq < tol * initial_norm:
            break
        if new_r_norm_sq > 1e10 * initial_norm:
            break
        p = r + (new_r_norm_sq / max(r_norm_sq, 1e-30)) * p
        r_norm_sq = new_r_norm_sq
    return x


def diagonal_preconditioned_step(a, w, g, w_vec):
    """Fallback: diagonal preconditioned gradient step."""
    # D[i,i] = H[i,i] = 2*w[2i]
    diag = 2.0 * w[0::2]  # w[0], w[2], w[4], ...
    diag = np.maximum(diag, 1e-10)
    d = -g / diag
    return d


def newton_optimize(a0, time_limit, top_k=32, temp_init=0.001, temp_growth=1.004,

    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    no_improve = 0
    it = 0
    cur_temp = temp_init

        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.5:

        g = compute_gradient(a, w)

            no_improve += 1
            cur_temp *= temp_growth
            continue

        d_norm = np.linalg.norm(d)

        # If CG failed or gave degenerate result, use diagonal preconditioning
        if d_norm < 1e-14 or not np.isfinite(d_norm):

            d_norm = np.linalg.norm(d)

                best_alpha = cur_alpha

            a = project_simplex(a + best_alpha * d)

                best_a = a.copy()

def solve_lp_oracle(f, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    if n > 2000:
        return None
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def fw_step(f, M, time_limit=15):
    """Frank-Wolfe direction via LP oracle."""
    n = len(f)
    g = solve_lp_oracle(f, M, time_limit)
    if g is None:
        return None
    gs = g.sum()
    if gs < 1e-15:
        return None
    return g / gs


def generate_inits(n, parent=None, rng=None):
    """Generate diverse structured initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Linear ramp decreasing
    ramp_d = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(ramp_d / ramp_d.sum())

    # Cosine window
    c = np.cos(np.pi * (x / max(n - 1, 1) - 0.5)) ** 4 + 0.01
    inits.append(c / c.sum())

    # Bimodal
    center = n / 2.0
    bimodal = np.exp(-(x - center * 0.3) ** 2 / (n * 0.05)) + np.exp(-(x - center * 0.7) ** 2 / (n * 0.05))
    inits.append(bimodal / bimodal.sum())

    # Parent-based
    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    return inits

        best_seq_ref[0] = a.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}) (was {old:.8f})")
        return True, score
    return False, score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: structured Newton optimization with Hankel Hessian,
    LP-based Frank-Wolfe escapes, and multi-scale exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = [None]
    best_score = [float('inf')]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
    except Exception:
        pass

    # === Phase 1: Newton on parent ===
    print("=== Phase 1: Newton on Parent ===")
    if parent is not None:
        n_p = len(parent)
        t1 = min(250, deadline - time.monotonic() - 400)
        if t1 > 30:
            print(f"  Running Newton on parent (n={n_p}), budget={t1:.0f}s")
            a_opt, obj = newton_optimize(parent, time_limit=t1, top_k=32,
                                         temp_init=0.001, reg=1e-6)
            try_score(a_opt, best_seq, best_score, "parent_N1")

            # Follow up with different params
            rem = deadline - time.monotonic() - 350
            if rem > 30:
                a_opt2, _ = newton_optimize(a_opt, time_limit=min(rem, 80),
                                            top_k=64, temp_init=0.0005, reg=1e-7)
                try_score(a_opt2, best_seq, best_score, "parent_N2")

            # LP escape
            rem = deadline - time.monotonic() - 300
            if rem > 20:
                a_n = best_seq[0] / best_seq[0].sum() if best_seq[0] is not None else parent
                M = fast_convolve(a_n).max()
                for relax in [1.01, 1.03, 1.05]:
                    rem = deadline - time.monotonic() - 250
                    if rem < 15:
                        break
                    g = fw_step(a_n, M * relax, time_limit=min(15, rem // 4))
                    if g is not None:
                        for alpha in [0.3, 0.5, 0.7, 1.0]:
                            a_try = project_simplex((1 - alpha) * a_n + alpha * g)
                            try_score(a_try, best_seq, best_score, f"parent_LP_{relax}_{alpha}")
                        # Newton from LP-mixed point
                        rem = deadline - time.monotonic() - 200
                        if rem > 15:
                            a_best_n = best_seq[0] / best_seq[0].sum()
                            a_opt3, _ = newton_optimize(a_best_n, time_limit=min(rem, 40),
                                                        top_k=32, temp_init=0.0003, reg=1e-7)
                            try_score(a_opt3, best_seq, best_score, "parent_postLP_N")

    # === Phase 2: Multi-length exploration ===
    print(f"\n=== Phase 2: Exploration (best={best_score[0]:.8f}) ===")

    lengths = [384, 256, 512]
    if parent is not None and len(parent) not in lengths:
        lengths.insert(0, len(parent))

    for n in lengths:
        if time.monotonic() > deadline - 200:
            break
        if parent is not None and n == len(parent):
            continue

        inits = generate_inits(n, parent, rng)
        rem = deadline - time.monotonic() - 150

        for i, f_init in enumerate(inits[:4]):
            if time.monotonic() > deadline - 100:
                break
            t_alloc = max(10, min(rem / max(1, len(inits[:4]) - i), 35))
            a_opt, obj = newton_optimize(f_init, time_limit=t_alloc,
                                         top_k=max(16, n // 10), temp_init=0.002, reg=1e-5)
            try_score(a_opt, best_seq, best_score, f"n{n}_i{i}")

    # === Phase 3: Deep optimization on best ===
    print(f"\n=== Phase 3: Deep Optimization (best={best_score[0]:.8f}) ===")

    if best_seq[0] is not None:
        n_best = len(best_seq[0])

        # Multiple Newton configurations
        configs = [
            (16, 0.0001, 1e-8),
            (32, 0.0005, 1e-7),
            (64, 0.002, 1e-5),
            (128, 0.01, 1e-4),
            (32, 0.0003, 1e-7),
        ]

        for ci, (tk, ti, rg) in enumerate(configs):
            rem = deadline - time.monotonic() - 80
            if rem < 15:
                break
            t_alloc = min(rem // (len(configs) - ci), 50)

            if ci == 0:
                f_init = best_seq[0].copy()
            elif ci == 1:
                f_init = project_simplex(best_seq[0] + rng.standard_normal(n_best) * 0.003)
            elif ci == 2:
                f_init = project_simplex(best_seq[0] + rng.standard_normal(n_best) * 0.008)
            elif ci == 3:
                f_init = project_simplex(0.9 * best_seq[0] / best_seq[0].sum() +
                                         0.1 * np.ones(n_best) / n_best)
            else:
                f_init = project_simplex(best_seq[0] + rng.standard_normal(n_best) * 0.001)

            a_opt, obj = newton_optimize(f_init, time_limit=t_alloc,
                                         top_k=tk, temp_init=ti, reg=rg,
                                         max_cg_iter=70)
            try_score(a_opt, best_seq, best_score, f"deep_{ci}")

        # LP escapes
        for le in range(4):
            rem = deadline - time.monotonic() - 50
            if rem < 15:
                break
            a_n = best_seq[0] / best_seq[0].sum()
            M = fast_convolve(a_n).max()
            relax = 1.01 + 0.03 * le
            g = fw_step(a_n, M * relax, time_limit=min(12, rem // 5))
            if g is not None:
                for alpha in [0.2, 0.4, 0.6, 0.8]:
                    a_try = project_simplex((1 - alpha) * a_n + alpha * g)
                    try_score(a_try, best_seq, best_score, f"LP3_{le}_{alpha}")
                # Polish with Newton
                rem = deadline - time.monotonic() - 40
                if rem > 10:
                    a_opt, _ = newton_optimize(best_seq[0], time_limit=min(rem, 20),
                                               top_k=16, temp_init=0.0002, reg=1e-8)
                    try_score(a_opt, best_seq, best_score, f"postLP3_{le}")

    # === Phase 4: Fine polishing ===
    print(f"\n=== Phase 4: Final Polish (best={best_score[0]:.8f}) ===")

    if best_seq[0] is not None:
        n_best = len(best_seq[0])

        # Very fine Newton with many CG iterations
        rem = deadline - time.monotonic() - 25
        if rem > 10:
            a_opt, _ = newton_optimize(best_seq[0], time_limit=min(rem, 35),
                                       top_k=8, temp_init=0.00005,
                                       reg=1e-10, max_cg_iter=100)
            try_score(a_opt, best_seq, best_score, "polish_fine")

        # Micro perturbations
        rem = deadline - time.monotonic() - 15
        if rem > 8:
            for mp in range(4):
                rem = deadline - time.monotonic() - 10
                if rem < 5:
                    break
                f_pert = project_simplex(best_seq[0] + rng.standard_normal(n_best) * 0.0005)
                a_opt, _ = newton_optimize(f_pert, time_limit=min(rem, 10),
                                           top_k=8, temp_init=0.00003,
                                           reg=1e-10, max_cg_iter=80)
                try_score(a_opt, best_seq, best_score, f"micro_{mp}")

        # Try with slightly larger n (interpolate)
        rem = deadline - time.monotonic() - 10
        if rem > 8 and n_best < 800:
            n_large = min(n_best + 64, 800)
            a_large = np.interp(np.arange(n_large),
                                np.linspace(0, 1, n_best),
                                best_seq[0])
            a_large = project_simplex(a_large)
            a_opt, _ = newton_optimize(a_large, time_limit=min(rem, 8),
                                       top_k=16, temp_init=0.001, reg=1e-5)
            try_score(a_opt, best_seq, best_score, "larger_n")

    # Fallback
    if best_seq[0] is None:
        print("  WARNING: No valid sequence found, using fallback")
        best_seq[0] = np.ones(384) / 384
        best_score[0] = float(evaluate_sequence(best_seq[0].tolist()))
        if not np.isfinite(best_score[0]):
            best_seq[0] = np.ones(256) / 256
            best_score[0] = float(evaluate_sequence(best_seq[0].tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq[0].tolist()))
    if np.isfinite(fs):
        best_score[0] = fs
    else:
        best_seq[0] = np.ones(256) / 256
        best_score[0] = float(evaluate_sequence(best_seq[0].tolist()))

    print(f"\nFINAL SCORE: {best_score[0]:.8f} (n={len(best_seq[0])})")
    return [float(x) for x in best_seq[0]]
