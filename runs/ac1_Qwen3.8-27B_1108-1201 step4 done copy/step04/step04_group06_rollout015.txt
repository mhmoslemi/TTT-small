"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Rapid Peak Descent:
1. Fast FFT-based gradient descent with soft-max peak weighting
2. Momentum with adaptive step sizing
3. Periodic LP refinement for escaping local minima
4. Multi-start with diverse initializations
5. Hierarchical multi-scale approach
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_gradient_peaks(a, conv, top_k=20, temperature=0.01):
    """
    Compute gradient of max peak using soft-max weighting.
    Efficient: O(n * top_k) using the structure of autocorrelation.
    """
    n = len(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    
    # Soft-max weights
    scaled = peak_vals / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum
    
    # Compute gradient efficiently
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-10:
            continue
        k = int(peaks[idx])
        wi = w[idx]
        if k == 0:
            # Special case: db_0/da_i = 2*a_i
            grad += wi * 2.0 * a
        else:
            # g_k[i] = a[i+k] (if i < n-k) + a[i-k] (if i >= k)
            grad[:n-k] += wi * a[k:]
            grad[k:] += wi * a[:n-k]
    
    return grad


def peak_descent(a0, time_limit, lr_init=0.1, momentum=0.9, top_k=30, 
                 temp_init=0.005, temp_growth=1.002, verbose=False):
    """
    Rapid Peak Descent: Fast gradient-based optimization.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    v = np.zeros(n)  # momentum
    lr = lr_init
    temp = temp_init
    
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    lr_decay = 0.999
    
    while time.monotonic() - t0 < time_limit:
        # Compute gradient
        grad = compute_gradient_peaks(a, conv, top_k=top_k, temperature=temp)
        
        # Project out mean (to maintain sum=1 approximately)
        grad = grad - np.mean(grad)
        
        # Momentum update
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        
        conv_new = fast_convolve(a_new)
        M_new = conv_new.max()
        obj_new = 2.0 * n * M_new
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            conv = conv_new
            no_improve = 0
        else:
            # Step too large or no improvement - try smaller step
            a_new2 = a + 0.5 * v
            a_new2 = np.maximum(a_new2, 0.0)
            s2 = a_new2.sum()
            if s2 > 1e-15:
                a_new2 = a_new2 / s2
            conv_new2 = fast_convolve(a_new2)
            if 2.0 * n * conv_new2.max() < best_obj - 1e-14:
                a = a_new2
                conv = conv_new2
                no_improve = 0
            else:
                a = a_new  # Keep the current (even if not improving)
                conv = conv_new
                no_improve += 1
        
        iteration += 1
        
        # Adaptive parameters
        lr *= lr_decay
        temp *= temp_growth
        
        if verbose and iteration % 500 == 0:
            print(f"    PD iter {iteration}, obj={best_obj:.8f}, lr={lr:.6f}, t={time.monotonic()-t0:.1f}s")
        
        if no_improve >= 200:
            break
    
    return best_a, best_obj


def lp_refine(a0, time_limit, max_iters=200):
    """
    LP-based refinement: solve LP to find direction that reduces max peak.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, 2.0 * len(a0) * fast_convolve(a0).max()
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Build convolution matrix
        C = np.zeros((2 * n - 1, n))
        for j in range(n):
            C[j:j+n, j] = a
        
        # LP: max sum(g) s.t. C@g <= M*(1+eps), g >= 0
        M_target = M * 1.0001
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M_target)
        bounds = [(0, None)] * n
        
        try:
            result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, 
                           method='highs', options={'presolve': True})
            if not result.success:
                break
            g = result.x
            g_sum = g.sum()
            if g_sum < 1.0 + 1e-8:
                break
            
            # Direction
            d = g - a
            norm_d = np.linalg.norm(d)
            if norm_d < 1e-10:
                break
            
            # Line search
            best_t = 0
            for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01]:
                a_try = a + t * d
                a_try = np.maximum(a_try, 0.0)
                st = a_try.sum()
                if st < 1e-15:
                    continue
                a_try = a_try / st
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    conv = conv_try
                    M = conv.max()
                    best_t = t
                    no_improve = 0
                    break
            else:
                no_improve += 1
        except Exception:
            break
        
        if no_improve >= 30:
            break
    
    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Sine
    inits['sine1'] = np.sin(np.pi * x / n) ** 2 + 0.01
    inits['sine1'] /= inits['sine1'].sum()
    
    # Bimodal (edges)
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # Alternating
    alt = np.ones(n)
    alt[::2] = 1.3
    alt[1::2] = 0.7
    inits['alternating'] = alt / alt.sum()
    
    # Triangular
    tri = 1.0 - 2.0 * np.abs(x - n/2) / n
    inits['triangular'] = tri / tri.sum()
    
    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps
            # Perturbed parent
            noise = rng.standard_normal(len(p)) * 0.01
            pp2 = np.maximum(p / ps + noise, 0)
            pp2s = pp2.sum()
            if pp2s > 1e-10:
                inits['parent_noisy'] = pp2 / pp2s
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Diagonally-preconditioned peak gradient descent with adaptive step sizing
2. Soft-max peak weighting with temperature annealing  
3. LP-based escape from local minima (used sparingly)
4. Multi-start with diverse initializations
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def peak_grad(a, conv, top_k, temp):
    """Compute soft-max weighted gradient of peak autocorrelation. O(n*top_k)."""
    n = len(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    sc = pv / temp
    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        wi = w[idx]
        if k == 0:
            grad += wi * 2.0 * a
        else:
            grad[:n-k] += wi * a[k:]
            grad[k:] += wi * a[:n-k]
    return grad


def preconditioner(a, conv, top_k):
    """Compute diagonal Hessian approximation for preconditioning."""
    n = len(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    sc = pv / (0.01 * (pv.max() + 1e-15))
    sc -= sc.max()
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-15:
        k = int(peaks[np.argmax(pv)])
        if k == 0:
            return 2.0 * np.maximum(a, 1e-10)
        hd = np.zeros(n)
        hd[:n-k] += a[k:]
        hd[k:] += a[:n-k]
        return np.maximum(2.0 * hd, 1e-10)
    w /= ws
    
    hd = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        wi = w[idx]
        if k == 0:
            hd += wi * 2.0 * a
        else:
            hd[:n-k] += wi * a[k:]
            hd[k:] += wi * a[:n-k]
    return np.maximum(2.0 * hd, 1e-10)

    Diagonally-preconditioned gradient descent with momentum.

    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        # Preconditioner (recompute every 100 iters or so)
        if it % 100 == 0:
            H = preconditioner(a, conv, top_k)
        
        # Gradient
        g = peak_grad(a, conv, top_k, temp)
        
        # Project out mean
        g = g - np.mean(g)
        
        # Preconditioned step
        v = momentum * v - lr * (g / H)
        
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn > 1e-15:
            a_new = a_new / sn
        else:
            a_new = best_a.copy()
            v = np.zeros(n)
            it += 1
            continue
        
        conv_new = fast_convolve(a_new)
        M_new = conv_new.max()
        obj = 2.0 * n * M_new
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            conv = conv_new
            no_improve = 0
        else:
            no_improve += 1
            # Try smaller step
            a_try = a + 0.3 * v
            a_try = np.maximum(a_try, 0.0)
            st = a_try.sum()
            if st > 1e-15:
                a_try = a_try / st
                ct = fast_convolve(a_try)
                if 2.0 * n * ct.max() < best_obj - 1e-14:
                    a = a_try
                    conv = ct
                    best_obj = min(best_obj, 2.0 * n * ct.max())
                    best_a = a.copy()
                    no_improve = 0
                else:
                    a = a_new  # Keep current
                    conv = conv_new
            else:
                a = a_new
                conv = conv_new
        
        # Adaptive lr
        if no_improve > 20:
            lr *= 0.95
        elif no_improve < 5:
            lr = min(lr * 1.02, 2.0)
        
        if no_improve > 500:
            break
        
        it += 1
    
    return best_a, best_obj


def lp_escape(a0, time_limit, max_iters=50):
    """LP-based escape from local minimum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), float('inf')
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    
    # Build matrix once
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j+n, j] = a
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        try:
            res = linprog(-np.ones(n), A_ub=C, b_ub=np.full(2*n-1, M*1.001),
                         bounds=[(0,None)]*n, method='highs')
            if not res.success:
                break
            g = res.x
            if g.sum() < 1.0 + 1e-8:
                break
            d = g - a
            if np.linalg.norm(d) < 1e-10:
                break
            
            improved = False
            for t in [0.3, 0.2, 0.1, 0.05, 0.02]:
                at = a + t * d
                at = np.maximum(at, 0)
                st = at.sum()
                if st < 1e-15:
                    continue
                at = at / st
                ct = fast_convolve(at)
                ot = 2.0 * n * ct.max()
                if ot < best_obj - 1e-12:
                    best_obj = ot
                    best_a = at.copy()
                    a = at
                    conv = ct
                    M = ct.max()
                    # Rebuild C
                    C = np.zeros((2*n-1, n))
                    for j in range(n):
                        C[j:j+n, j] = a
                    improved = True
                    break
            if not improved:
                break
        except Exception:
            break
    
    return best_a, best_obj


def make_inits(n, parent, rng):
    """Generate initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uni'] = np.ones(n) / n
    inits['rand'] = rng.dirichlet(np.ones(n) * 0.5)
    
    s1 = np.sin(np.pi * x / n)**2 + 0.01
    inits['sine'] = s1 / s1.sum()
    
    e = max(1, n//4)
    bi = np.zeros(n)
    bi[:e] = 1.0
    bi[-e:] = 1.0
    inits['bimod'] = bi / bi.sum()
    
    alt = np.ones(n)
    alt[::2] = 1.3
    alt[1::2] = 0.7
    inits['alt'] = alt / alt.sum()
    
    if parent is not None:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
                nz = rng.standard_normal(n) * 0.02
                pn = np.maximum(p/ps + nz, 0)
                inits['parent_nz'] = pn / pn.sum()
            else:
                pp = np.interp(np.linspace(0,1,n), np.linspace(0,1,len(p)), p/ps)
                pp = np.maximum(pp, 0)
                inits['parent_i'] = pp / pp.sum()
    
    return inits
