However, with optimization, longer sequences might allow better quality (more degrees of freedom). So trying a few different lengths is worthwhile.

Let me write the code now.

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Coordinate descent with exact line search (scipy minimize_scalar)
2. LP-based Frank-Wolfe with adaptive M
3. Spectral smoothing (Gerchberg-Saxton style)
4. Multi-length search with interpolation

Key improvement over parent: exact line search in coordinate descent
instead of fixed multiplier scanning, plus spectral smoothing step.
"""

import numpy as np
from scipy.optimize import linprog, minimize_scalar
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    if old_n >= 4:
        from scipy.interpolate import CubicSpline
        cs = CubicSpline(old_x, f, bc_type='clamped')
        f_interp = cs(new_x)
    else:
        f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def coord_descent_exact(f0, time_limit, verbose=False, max_passes=1000):
    """
    Coordinate descent with exact line search using scipy minimize_scalar.
    For each coordinate i, find the optimal value v that minimizes
    max_k (a'(v)*a')(v)[k] where a' has a_i = v and others scaled to keep sum=1.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    passes = 0

    # Precompute for speed
    # For each coordinate i, we need to evaluate g(v) = max_k (a'(v)*a')(v)[k]
    # This requires a full FFT each time, which is O(n log n)
    
    while time.monotonic() - t0 < time_limit and passes < max_passes:
        passes += 1
        improved = False
        pass_improvement = 0.0

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                    break
            
            old_val = f[j]
            if old_val < 1e-14:
                # Try setting to a small positive value
                candidates = [1e-8, 1e-7, 1e-6]
                best_cand_obj = best_obj
                best_cand_val = 0.0
                for cv in candidates:
                    f_trial = f.copy()
                    f_trial[j] = cv
                    f_trial[j] = 0.0  # set to 0 first
                    # Actually, let's skip zero coordinates for now
                continue
            
            # Define the objective function for minimize_scalar
            other_sum = 1.0 - old_val
            if other_sum < 1e-10:
                continue
            
            c = f / other_sum  # normalized "other" vector, sum = 1, c[j] is the relative weight
            
            def obj_func(v):
                # v is the new value for coordinate j
                # a'[j] = v, a'[k] = (1-v) * c[k] for k != j
                # We need max_k (a' * a')[k]
                # Build a'
                a_prime = (1.0 - v) * c.copy()
                a_prime[j] = v
                cv = fast_convolve(a_prime)
                return cv.max()
            
            # Use bounded scalar minimization
            # Restrict search to a reasonable range around old_val
            lo = max(0.0, old_val * 0.3 - 0.01)
            hi = min(0.9999, old_val * 3.0 + 0.01)
            
            try:
                res = minimize_scalar(obj_func, bounds=(lo, hi), method='bounded',
                                     options={'maxiter': 30, 'xatol': 1e-8})
                if res.success and res.fun < best_obj - 1e-12:
                    new_val = res.x
                    # Update f
                    f[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= (1.0 - new_val) / other_sum  # scale others to keep sum=1
                    # Renormalize to fix any numerical drift
                    f = f / f.sum()
                    
                    conv = fast_convolve(f)
                    new_obj = conv.max()
                    if new_obj < best_obj - 1e-14:
                        best_obj = new_obj
                        best_f = f.copy()
                        improved = True
                        pass_improvement += best_obj - new_obj  # negative, so this is wrong
                    else:
                        # Revert
                        f[j] = old_val
                        f[mask] *= other_sum / (1.0 - old_val)
                        f = f / f.sum()
                # If no improvement from minimize_scalar, also try a few fixed candidates
                # as a safety net
                for mult in [0.5, 1.5, 0.1, 2.0]:
                    nv = old_val * mult
                    if nv < 0 or nv > 0.999:
                        continue
                    f_trial = f.copy()
                    f_trial[j] = nv
                    mask2 = np.ones(n, dtype=bool)
                    mask2[j] = False
                    f_trial[mask2] *= (1.0 - nv) / other_sum
                    f_trial = f_trial / f_trial.sum()
                    cv_trial = fast_convolve(f_trial)
                    if cv_trial.max() < best_obj - 1e-14:
                        f = f_trial
                        best_obj = cv_trial.max()
                        best_f = f.copy()
                        improved = True
                        break
            except Exception:
                pass

        if not improved and passes > 5:
            # Try a random perturbation to escape local minimum
            if passes % 10 == 0:
                rng = np.random.default_rng(passes * 12345 + 67890)
                noise = rng.standard_normal(n) * 0.005
                f_pert = project_simplex(best_f + noise)
                conv_pert = fast_convolve(f_pert)
                if conv_pert.max() < best_obj:
                    f = f_pert
                    best_obj = conv_pert.max()
                    best_f = f.copy()
            else:
                break

        if verbose and passes % 50 == 0:
            print(f"    CD iter {passes}: obj={best_obj:.10f}")

    return best_f, best_obj


def coord_descent_fast(f0, time_limit, verbose=False, max_passes=5000):
    """
    Faster coordinate descent using precomputed structure and fewer FFT evaluations.
    Uses the fact that only a few coordinates near the peak are "active".
    Updates coordinates in a smart order (those contributing most to the peak first).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit and passes < max_passes:
        passes += 1
        improved = False

        # Sort coordinates by their contribution to the peak
        # Coordinates that appear in the peak convolution terms are most important
        peak_k = np.argmax(conv)
        active_js = set()
        for k in [peak_k + dk for dk in range(-3, 4)]:
            k = int(k)
            if 0 <= k < len(conv):
                js = range(max(0, k - n + 1), min(n, k + 1))
                active_js.update(js)
        
        # Process active coordinates first, then others
        order = list(active_js) + [j for j in range(n) if j not in active_js]
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-14:
                continue

            other_sum = 1.0 - old_val
            if other_sum < 1e-10:
                continue

            # Try a set of multipliers (faster than minimize_scalar)
            best_j_val = old_val
            best_j_obj = best_obj
            
            for mult in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 3.0]:
                nv = old_val * mult
                if nv < 0 or nv > 0.9999:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial = f.copy()
                f_trial[j] = nv
                f_trial[mask] *= (1.0 - nv) / other_sum
                f_trial = f_trial / f_trial.sum()
                cv_trial = fast_convolve(f_trial)
                if cv_trial.max() < best_j_obj - 1e-15:
                    best_j_obj = cv_trial.max()
                    best_j_val = nv

            if best_j_val != old_val:
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[j] = best_j_val
                f[mask] *= (1.0 - best_j_val) / other_sum
                f = f / f.sum()
                best_obj = best_j_obj
                best_f = f.copy()
                improved = True

        if not improved and passes > 3:
            # Random restart
            if passes % 20 == 0:
                rng = np.random.default_rng(passes * 777 + 42)
                noise = rng.standard_normal(n) * (0.003 + 0.002 * min(passes // 100, 5))
                f_pert = project_simplex(best_f + noise)
                f = f_pert
                conv = fast_convolve(f)
            else:
                break

        if verbose and passes % 100 == 0:
            print(f"    CD-fast iter {passes}: obj={best_obj:.10f}")

    return best_f, best_obj


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive M relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Solve LP: max sum(g) s.t. (f*g)[k] <= M*m_exp for all k, g >= 0
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M * m_exp)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        except Exception:
            res = None

        if res is None or not res.success:
            m_exp *= 1.15
            if m_exp > 5.0:
                m_exp = 1.0
                # Try with relaxed M
                continue
        
        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 50:
                break
            m_exp = min(m_exp * 1.03, 3.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M

        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            # Refine with binary search
            lo = max(0.0, best_alpha - 0.15)
            hi = min(1.0, best_alpha + 0.15)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 4.0)

        if no_improve > 60:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.002 + 0.003 * min(no_improve // 50, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def spectral_smooth(f0, time_limit, verbose=False):
    """
    Spectral smoothing: iteratively adjust the sequence to flatten its
    power spectrum, which reduces autocorrelation peaks.
    Uses a Gerchberg-Saxton like approach.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0

    # FFT size for full linear convolution
    conv_size = 2 * n - 1
    fft_size = 1 << (conv_size - 1).bit_length()

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it > 5000:
            break

        # Compute current spectrum
        A = np.fft.rfft(f, fft_size)
        power = np.abs(A) ** 2
        
        # Target: flat spectrum (all frequencies equal)
        # But we need to preserve the DC component (sum = 1)
        # DC = FFT[0] = sum(f) = 1 (real)
        
        # Try to reduce the variance of the power spectrum
        # while keeping the sequence nonnegative
        mean_power = power[1:].mean()  # exclude DC
        power_target = np.ones_like(power) * mean_power
        power_target[0] = 1.0  # keep DC
        
        # Compute new amplitude: blend between current and target
        alpha = 0.1  # smoothing parameter
        new_amp = np.sqrt((1 - alpha) * power + alpha * power_target)
        
        # Keep the phase from the current spectrum
        phase = np.angle(A)
        new_A = new_amp * np.exp(1j * phase)
        
        # Inverse FFT
        f_new = np.fft.irfft(new_A, fft_size)[:n]
        
        # Project to simplex
        f_new = project_simplex(f_new)
        
        # Evaluate
        conv = fast_convolve(f_new)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            f = f_new
        else:
            # Revert and try smaller alpha
            alpha /= 2
            if alpha < 0.001:
                # Try a different approach: reduce power at peak frequencies
                peak_k = np.argmax(conv)
                broken = True
                break

        if it % 100 == 0 and verbose:
            print(f"    Spectral iter {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.003, mom=0.94, top_k=50, temp=0.00005):
    """Subgradient polishing with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9997
        cur_temp *= 1.003

        if no_improve > 500:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0

    return best_f, best_obj
