def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def mirror_descent_step(f, subgrad, lr):
    """Mirror descent (softmax) step on the simplex."""
    n = len(f)
    # f * exp(-lr * subgrad), normalized
    log_f = np.where(f > 1e-30, np.log(f), -30.0)
    scores = log_f - lr * subgrad
    scores -= scores.max()
    new_f = np.exp(scores)
    s = new_f.sum()
    if s < 1e-30:
        return f.copy()
    return new_f / s

def compute_subgrad(f, conv, top_k, temperature):
    """Compute weighted subgradient from top-k peaks."""
    n = len(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    
    pv = conv[peaks]
    scaled = pv / max(temperature, 1e-10)
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    
    return sg, peaks, w

def lp_direction(f, M, method='highs'):
    """Find LP direction: max sum(g) s.t. (f*g) <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method)
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def spectral_init(n, rng):
    """Create initialization with approximately flat spectrum."""
    # Random phase in frequency domain
    freqs = np.fft.rfftfreq(n)
    # Create flat magnitude with random phase
    mag = np.ones(len(freqs))
    phase = rng.uniform(0, 2 * np.pi, len(freqs))
    spectrum = mag * np.exp(1j * phase)
    # Ensure Hermitian symmetry for real output
    a = np.fft.irfft(spectrum * n, n)
    a = np.abs(a)  # Take absolute value to ensure non-negative
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a / s

def lse_objective_and_grad(x, T, n):
    """Compute smoothed objective and gradient w.r.t. x."""
    # Softmax parametrization
    a = np.exp(x - x.max())
    a = a / a.sum()
    
    conv = fast_convolve(a)
    
    # Log-sum-exp
    Tconv = T * conv
    Tconv_max = Tconv.max()
    log_sum_exp = Tconv_max + np.log(np.sum(np.exp(Tconv - Tconv_max)))
    obj = log_sum_exp / T
    
    # Gradient w.r.t. a
    softmax_w = np.exp(Tconv - Tconv_max)
    softmax_w /= softmax_w.sum()
    
    # d(conv_k)/da[j] = 2*a[k-j] for valid j
    grad_a = np.zeros(n)
    for k in range(len(conv)):
        if softmax_w[k] < 1e-12:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * T * softmax_w[k] * a[k - idx]
    
    # Chain rule: d(x)/da for softmax
    # a = softmax(x), so da_i/dx_j = a_i*(delta_ij - a_j)

    # = a_j * (grad_a_j - sum_i grad_a_i * a_i)
    inner = np.sum(grad_a * a)
    grad_x = a * (grad_a - inner)
    
    return obj, grad_x

"""
Multi-stage optimization for minimal autocorrelation peak sequences.
Uses L-BFGS with power-mean smoothing (annealed), mirror descent,
aggressive coordinate descent, and multi-length exploration.
Key innovation: smooth optimization via power-mean objective with
softmax parametrization, enabling fast gradient-based convergence
to the basin of attraction, followed by direct coordinate descent
polishing on the true (non-smooth) objective.

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def power_mean_obj_and_grad(x, p, n):
    """
    Compute power-mean objective and gradient w.r.t. softmax parameters x.
    Objective: M_p(conv(a,a)) where a = softmax(x), M_p = (mean(b^p))^(1/p).
    """
    a = softmax(x)
    conv = np.maximum(fast_convolve(a), 1e-30)
    n_conv = len(conv)
    
    # Power mean
    log_pm = (np.sum(np.log(conv) * p) - np.log(n_conv)) / p
    pm = np.exp(log_pm)
    obj = float(pm)
    
    # Weights: w_k = pm^(1-p) * conv[k]^(p-1) / n_conv
    log_w = (1.0 - p) * np.log(pm) + (p - 1.0) * np.log(conv) - np.log(n_conv)
    # Subtract max for stability
    log_w_shifted = log_w - log_w.max()
    w = np.exp(log_w_shifted)
    
    # grad_a[j] = 2 * sum_k w[k] * a[k-j] for valid k,j
    # = 2 * (w cross-correlate a)[j]
    # Compute via direct method (fast for n~400)
    grad_a = np.zeros(n)
    for j in range(n):
        # Sum over k such that 0 <= k-j < n, i.e., j <= k <= j+n-1
        k_start = max(0, j)
        k_end = min(n_conv, j + n)
        if k_start >= k_end:
            continue
        idx = np.arange(k_start, k_end)
        grad_a[j] = 2.0 * np.dot(w[idx], a[idx - j])
    
    # Chain rule through softmax: grad_x = a * (grad_a - dot(grad_a, a))
    inner = np.dot(grad_a, a)
    grad_x = a * (grad_a - inner)
    
    return obj, grad_x


def power_mean_optimize(f0, time_limit, p_start=20, p_end=5000, 
                        max_iter=2000, tol=1e-10):
    """
    L-BFGS optimization with annealed power-mean smoothing.
    Works in softmax parameter space.
    """
    n = len(f0)
    a = np.maximum(f0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Initialize x from a (inverse softmax)
    a_safe = np.maximum(a, 1e-15)
    x = np.log(a_safe)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    current_p = p_start
    
    # Anneal p over the optimization
    n_restarts = 5
    iterations_per_restart = max_iter // n_restarts
    
    for restart in range(n_restarts):
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        p = p_start * (p_end / p_start) ** (restart / max(n_restarts - 1, 1))
        p = max(2.0, min(p, 100000.0))
        
        def fun_and_grad(x_):
            return power_mean_obj_and_grad(x_, p, n)
        
        try:
            res = minimize(fun_and_grad, x, method='L-BFGS-B',
                          jac=True, options={'maxiter': iterations_per_restart,
                                            'ftol': 1e-12, 'gtol': 1e-8,
                                            'maxls': 50})
        except Exception:
            x = np.log(np.maximum(a, 1e-15))
            continue
        
        a_new = softmax(res.x)
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()
        
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            x = res.x
            a = a_new.copy()
        
        # Also check the L-BFGS iterate at the smoothed level
        a_sm = softmax(res.x)
        a_sm = np.maximum(a_sm, 0.0) / max(a_sm.sum(), 1e-15)
        conv_sm = fast_convolve(a_sm)
        if conv_sm.max() < best_obj:
            best_obj = conv_sm.max()
            best_a = a_sm.copy()
            x = res.x
        
        # Add small perturbation for next restart
        if restart < n_restarts - 1:
            rng_r = np.random.default_rng(100 + restart * 37)
            x = x + rng_r.standard_normal(n) * 0.01
    
    return best_a, best_obj


def mirror_descent_opt(f0, time_limit, lr_init=0.005, lr_decay=0.9998,
                       top_k=50, temp_init=0.0001):
    """Mirror descent (softmax) on the simplex for max-convolution."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    lr = lr_init
    cur_temp = temp_init
    rng_local = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        pv = conv[peaks]
        scaled = pv / max(cur_temp, 1e-10)
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        # Mirror descent step
        log_f = np.where(f > 1e-30, np.log(f), -30.0)
        scores = log_f - lr * sg
        scores -= scores.max()
        new_f = np.exp(scores)
        ns = new_f.sum()
        if ns < 1e-30:
            f = best_f.copy()
        else:
            f = new_f / ns
        
        conv = fast_convolve(f)
        co = conv.max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= lr_decay
        cur_temp *= 1.003
        
        if no_improve > 250:
            f = project_simplex(best_f + rng_local.standard_normal(n) * 0.004)
            lr = max(lr_init * 0.4, 0.001)
            cur_temp = temp_init
            no_improve = 0
    
    return best_f, best_obj

def lp_direction(f, M):
    """LP: max sum(g) s.t. (f*g) <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit):

    f = np.maximum(f0.astype(np.float64), 0.0)

    best_f = f.copy()

    rng_local = np.random.default_rng(42 + n)
    
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        g = lp_direction(f, M * m_exp)

            if m_exp > 5.0:
                break

        if g_sum < 1e-15:
            break

        if diff < 1e-10:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 3.0)
            if no_improve > 50:
                break
            continue

            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs < 1e-15:
                f = best_f.copy()
            else:

            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1

            no_improve += 1
            m_exp = min(m_exp * 1.1, 4.0)

            if strat == 0:

            elif strat == 1:
                f = project_simplex(0.8 * best_f + 0.2 * best_f[::-1])
            else:
                f = project_simplex(rng_local.dirichlet(np.ones(n) * 0.5))
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
    
    return best_f, best_obj


def spectral_init(n, rng):
    """Create initialization with approximately flat magnitude spectrum."""
    freqs = np.fft.rfftfreq(n)
    mag = np.ones(len(freqs))
    phase = rng.uniform(0, 2 * np.pi, len(freqs))
    spectrum = mag * np.exp(1j * phase)
    a = np.fft.irfft(spectrum * n, n)
    a = np.abs(a) + 0.01
    s = a.sum()
    return a / s


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    inits.append(rng.dirichlet(np.ones(n) * 1.0))
    
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits.append(g / g.sum())
    
    sine = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine / sine.sum())
    
    for _ in range(3):
        inits.append(spectral_init(n, rng))
    
    cos_t = np.abs(np.cos(np.pi * x / n)) + 0.01
    inits.append(cos_t / cos_t.sum())
    
    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
        # Perturbed parent
        for scale in [0.01, 0.03]:
            pp = project_simplex(px / ps + rng.standard_normal(n) * scale)
            inits.append(pp)
    
    return inits


def try_sequence(a, label, state):
    """Evaluate and update state."""
    a = np.maximum(a, 0.0)
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < state['best_score']:
        state['best_score'] = score
        state['best_seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-stage search: L-BFGS with power-mean annealing, mirror descent,
    FW-LP, and aggressive coordinate descent for minimal autocorrelation peak.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    state = {'best_seq': None, 'best_score': float('inf')}
    
    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}")
            s = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(s):
                print(f"  Parent score: {s:.6f}")
                if s < state['best_score']:
                    state['best_score'] = s
                    state['best_seq'] = parent.copy()
    except Exception:
        pass
    
    # === Phase 1: Quick L-BFGS exploration (first 15%) ===
    print("\n=== Phase 1: Quick L-BFGS Exploration ===")
    phase1_end = time.monotonic() + float(budget_s) * 0.15
    
    if parent is not None:
        n_p = len(parent)
        # L-BFGS with moderate smoothing
        t_alloc = min(80, phase1_end - time.monotonic() - 10)
        if t_alloc > 15:
            print(f"  L-BFGS power-mean on parent (n={n_p}), t={t_alloc:.0f}s")
            a_opt, obj = power_mean_optimize(parent, time_limit=t_alloc,
                                             p_start=10, p_end=500, max_iter=1000)
            try_sequence(a_opt, "pm_parent", state)
        
        # Mirror descent quick
        rem = phase1_end - time.monotonic() - 10
        if rem > 15:
            print(f"  Mirror descent on parent, t={rem:.0f}s")
            a_opt, obj = mirror_descent_opt(parent, time_limit=rem,
                                             lr_init=0.004, top_k=30, temp_init=0.00005)
            try_sequence(a_opt, "md_parent", state)
    
    # Try a couple other lengths
    if parent is not None:
        n_parent = len(parent)
        test_lengths = [n_parent - 16, n_parent - 8, n_parent + 8, n_parent + 16]
        test_lengths = [t for t in test_lengths if 128 <= t <= 640]
        
        for n_t in test_lengths[:2]:
            rem = phase1_end - time.monotonic() - 10
            if rem < 15:
                break
            inits = generate_inits(n_t, parent, rng)
            t_alloc = min(rem, 30)
            if t_alloc < 10:
                break
            a_opt, obj = power_mean_optimize(inits[7 if len(inits) > 7 else 0], 
                                             time_limit=t_alloc,
                                             p_start=10, p_end=200, max_iter=500)
            try_sequence(a_opt, f"pm_n{n_t}", state)
    
    print(f"  Phase 1 best: {state['best_score']:.6f}")
    
    # === Phase 2: Deep optimization (next 40%) ===
    print(f"\n=== Phase 2: Deep Optimization ===")
    phase2_end = time.monotonic() + float(budget_s) * 0.40
    
    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])
        print(f"  Optimizing n={n_best}, current={state['best_score']:.6f}")
        
        # Progressive L-BFGS with increasing p (annealing)
        p_schedule = [20, 50, 100, 200, 500, 1000, 2000]
        for i, p_val in enumerate(p_schedule):
            rem = phase2_end - time.monotonic() - 10
            if rem < 10:
                break
            t_alloc = min(rem / (len(p_schedule) - i), 60)
            if t_alloc < 8:
                break
            print(f"  L-BFGS p={p_val}, t={t_alloc:.0f}s")
            a_opt, obj = power_mean_optimize(state['best_seq'], time_limit=t_alloc,
                                             p_start=p_val, p_end=p_val * 2,
                                             max_iter=800)
            try_sequence(a_opt, f"pm_p{p_val}", state)
        
        # FW-LP deep run
        rem = phase2_end - time.monotonic() - 10
        if rem > 20:
            t_alloc = min(rem * 0.4, 120)
            print(f"  FW-LP deep, t={t_alloc:.0f}s")
            a_opt, obj = fw_lp_optimize(state['best_seq'], time_limit=t_alloc)
            try_sequence(a_opt, "fwl_deep", state)
        
        # Perturbation + L-BFGS
        perturb_scales = [0.005, 0.01, 0.02, 0.04]
        for ps in perturb_scales:
            rem = phase2_end - time.monotonic() - 15
            if rem < 12:
                break
            t_alloc = min(rem * 0.3, 50)
            if t_alloc < 8:
                break
            noise = rng.standard_normal(n_best) * ps
            f_init = project_simplex(state['best_seq'] + noise)
            a_opt, obj = power_mean_optimize(f_init, time_limit=t_alloc,
                                             p_start=50, p_end=500, max_iter=500)
            try_sequence(a_opt, f"pert_pm_{ps}", state)
        
        # Mirror blend + optimize
        rem = phase2_end - time.monotonic() - 10
        if rem > 20:
            f_mirror = project_simplex(0.7 * state['best_seq'] + 0.3 * state['best_seq'][::-1])
            a_opt, obj = power_mean_optimize(f_mirror, time_limit=min(rem, 60),
                                             p_start=30, p_end=300, max_iter=600)
            try_sequence(a_opt, "mirror_pm", state)
    
    print(f"  Phase 2 best: {state['best_score']:.6f}")
    
    # === Phase 3: Polishing (remaining time) ===
    print(f"\n=== Phase 3: Polishing ===")
    phase3_end = deadline - 10
    
    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])
        print(f"  Current best: {state['best_score']:.6f} (n={n_best})")
        
        # Long aggressive coordinate descent
        rem = phase3_end - time.monotonic() - 30
        if rem > 20:
            t_alloc = min(rem * 0.5, 150)
            print(f"  Aggressive CD round 1, t={t_alloc:.0f}s")
            a_opt, obj = aggressive_coord_descent(state['best_seq'], time_limit=t_alloc, rng=rng)
            try_sequence(a_opt, "cd_r1", state)
        
        # High-p L-BFGS for fine structure
        rem = phase3_end - time.monotonic() - 20
        if rem > 15:
            t_alloc = min(rem * 0.4, 80)
            print(f"  L-BFGS high-p polish, t={t_alloc:.0f}s")
            a_opt, obj = power_mean_optimize(state['best_seq'], time_limit=t_alloc,
                                             p_start=2000, p_end=10000, max_iter=1000)
            try_sequence(a_opt, "pm_high_p", state)
        
        # Second CD round
        rem = phase3_end - time.monotonic() - 15
        if rem > 15:
            t_alloc = min(rem * 0.5, 100)
            print(f"  Aggressive CD round 2, t={t_alloc:.0f}s")
            a_opt, obj = aggressive_coord_descent(state['best_seq'], time_limit=t_alloc, rng=rng)
            try_sequence(a_opt, "cd_r2", state)
        
        # Fine mirror descent
        rem = phase3_end - time.monotonic() - 10
        if rem > 15:
            t_alloc = min(rem * 0.6, 60)
            print(f"  Fine MD polish, t={t_alloc:.0f}s")
            a_opt, obj = mirror_descent_opt(state['best_seq'], time_limit=t_alloc,
                                             lr_init=0.001, lr_decay=0.9999,
                                             top_k=20, temp_init=0.00001)
            try_sequence(a_opt, "md_fine", state)
        
        # Final CD squeeze
        rem = phase3_end - time.monotonic()
        if rem > 10:
            t_alloc = min(rem, 40)
            print(f"  Final CD squeeze, t={t_alloc:.0f}s")
            a_opt, obj = aggressive_coord_descent(state['best_seq'], time_limit=t_alloc, rng=rng)
            try_sequence(a_opt, "cd_final", state)
    
    # Fallback
    if state['best_seq'] is None:
        state['best_seq'] = np.ones(256) / 256
        state['best_score'] = float(evaluate_sequence(state['best_seq'].tolist()))
        print(f"  Fallback uniform: {state['best_score']:.6f}")
    
    # Final verification
    fs = float(evaluate_sequence(state['best_seq'].tolist()))
    if np.isfinite(fs):
        state['best_score'] = fs
    
    print(f"\nFINAL SCORE: {state['best_score']:.6f} (n={len(state['best_seq'])})")
    return [float(x) for x in state['best_seq']]
