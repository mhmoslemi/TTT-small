"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using exact 1D coordinate descent
with vectorized evaluations, multi-length exploration, and subgradient polishing.
"""

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def exact_coord_descent(f0, time_limit, verbose=False):
    """Exact 1D coordinate descent with vectorized peak evaluation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    
    # Precompute the D matrix structure (depends only on n and j)
    # For a given j, D[k] = b[k-j] + b[j-k]
    
    G_coarse = 30
    G_fine = 50
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        pass_improvement = 0.0
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            v = f[j]
            if v < 1e-14:
                continue
            
            alpha = 1.0 - v
            if alpha < 1e-10:
                continue
            
            b = f.copy()
            b[j] = 0.0
            
            C = fast_convolve(b)
            
            # Compute D
            D = np.zeros(2*n-1)
            end1 = min(j+n, 2*n-1)
            D[j:end1] += b[:end1-j]
            start2 = max(0, j-n+1)
            end2 = min(j, 2*n-2) + 1
            seg = b[j-start2 : j-end2+2 : -1]  # b[j-start2], b[j-start2-1], ..., b[j-end2+1]
            D[start2:end2] += seg
            
            # Coarse grid search
            tvals = np.linspace(0.0, min(1.0, v + alpha*0.5), G_coarse)
            u = (1.0 - tvals) / alpha
            conv_new = (u[:, None]**2) * C[None, :] + (tvals * u)[:, None] * D[None, :]
            if 2*j < 2*n-1:
                conv_new[:, 2*j] += tvals**2
            peaks = conv_new.max(axis=1)
            best_idx = np.argmin(peaks)
            
            # Fine grid search around best
            if 0 < best_idx < G_coarse - 1:
                lo = tvals[best_idx - 1]
                hi = tvals[best_idx + 1]
            elif best_idx == 0:
                lo = 0.0
                hi = tvals[1]
            else:
                lo = tvals[-2]
                hi = tvals[-1]
            
            tvals_fine = np.linspace(lo, hi, G_fine)
            u_fine = (1.0 - tvals_fine) / alpha
            conv_fine = (u_fine[:, None]**2) * C[None, :] + (tvals_fine * u_fine)[:, None] * D[None, :]
            if 2*j < 2*n-1:
                conv_fine[:, 2*j] += tvals_fine**2
            peaks_fine = conv_fine.max(axis=1)
            best_fine_idx = np.argmin(peaks_fine)
            t_opt = tvals_fine[best_fine_idx]
            
            # Apply update
            if t_opt > 1e-14 and abs(t_opt - v) > 1e-14:
                scale = (1.0 - t_opt) / alpha
                f_new = f.copy()
                f_new[j] = t_opt
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_new[mask] *= scale
                f_new = f_new / f_new.sum()  # numerical safety
                
                new_obj = fast_convolve(f_new).max()
                if new_obj < best_obj - 1e-15:
                    best_obj = new_obj
                    best_f = f_new.copy()
                    f = f_new
                    improved = True
                    pass_improvement += (best_obj - new_obj)
                elif new_obj < best_obj + 1e-12:
                    # Accept if roughly equal (to allow exploration)
                    f = f_new
            elif t_opt < 1e-14 and v > 1e-10:
                # Set to zero
                scale = 1.0 / alpha
                f_new = f.copy()
                f_new[j] = 0.0
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_new[mask] *= scale
                f_new = f_new / f_new.sum()
                new_obj = fast_convolve(f_new).max()
                if new_obj < best_obj - 1e-15:
                    best_obj = new_obj
                    best_f = f_new.copy()
                    f = f_new
                    improved = True
        
        if verbose and passes % 5 == 0:
            print(f"    CD pass {passes}: obj={best_obj:.10f}, improved={improved}")
        
        if not improved and passes > 3:
            # Try a different randomization
            if passes > 10:
                break
            rng = np.random.default_rng(passes * 12345)
            # Perturb a random subset of coordinates
            n_perturb = max(5, n // 10)
            indices = rng.choice(n, size=n_perturb, replace=False)
            perturbation = rng.standard_normal(n_perturb) * 0.01
            f_pert = best_f.copy()
            f_pert[indices] += perturbation
            f_pert = project_simplex(f_pert)
            f = f_pert
            obj = fast_convolve(f).max()
            if obj < best_obj:
                best_obj = obj
                best_f = f.copy()
    
    return best_f, best_obj

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using exact 1D coordinate descent
with vectorized evaluations, multi-length interpolation, and adaptive
subgradient polishing.

import numpy as np
import time

# --- Helper functions ---

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
