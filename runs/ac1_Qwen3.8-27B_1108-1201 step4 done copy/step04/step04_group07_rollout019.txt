def fast_convolve(a):

    n = len(a)
    size = 2 * n - 1

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_obj_grad(a, beta):
    """
    Compute (1/beta)*logsumexp(beta*conv(a,a)) and its gradient w.r.t. a.
    Assumes a >= 0 (but doesn't enforce during gradient computation).
    """
    n = len(a)
    conv = fast_convolve(a)
    bc = beta * conv
    mx = bc.max()
    shifted = bc - mx
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    obj = (mx + np.log(sum_exp)) / beta
    softmax = exp_shifted / sum_exp

    cross = np.fft.ifft(np.conj(Fh) * Fa).real

    return obj, grad

    Adam update with simplex projection.
    a is on the simplex (sum=1, a>=0).
    lam is the Lagrange multiplier for the sum constraint (handled by projection).
    Returns updated a, m, v.

    n = len(a)
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad ** 2

    v_hat = v / (1 - beta2 ** t)

    return a_new, m, v

        probs = contrib / contrib.sum()
        chosen = rng.choice(len(idx), size=n_reduce, replace=False, p=probs)

        new_a = best_a.copy()
        new_a[idx[chosen]] -= reduction / len(chosen)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth gradient-based optimization
with logsumexp continuation, Adam updates, peak-aware subgradient, multi-restart
with diverse initializations, and peak shaving. Tries multiple sequence lengths.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_obj_grad(a, beta):
    """
    Compute (1/beta)*logsumexp(beta*conv(a,a)) and its gradient w.r.t. a.
    Returns (obj, grad) where grad has same length as a.
    """
    n = len(a)
    conv = fast_convolve(a)
    bc = beta * conv
    mx = bc.max()
    shifted = bc - mx
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    obj = (mx + np.log(sum_exp)) / beta
    softmax = exp_shifted / sum_exp

    N = 1 << (3 * n).bit_length()
    Fh = np.fft.fft(softmax, N)
    Fa = np.fft.fft(a, N)
    cross = np.fft.ifft(np.conj(Fh) * Fa).real
    grad = 2.0 * cross[:n]

    return obj, grad


def peak_subgradient(a, top_k=30, temp=0.001):
    """
    Compute a weighted subgradient of max(conv(a,a)) using softmax over peaks.
    Returns the gradient vector (to be used for descent).
    """
    n = len(a)
    conv = fast_convolve(a)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]

    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w_peaks = np.exp(scaled)
    w_peaks /= w_peaks.sum()

    # Build the weight vector w over all lags
    w = np.zeros(len(conv))
    w[peaks] = w_peaks

    # grad[i] = 2 * sum_j a[j] * w[i+j] = 2 * cross_corr(a, w)[i]
    N = 1 << (3 * n).bit_length()
    Fw = np.fft.fft(w, N)
    Fa = np.fft.fft(a, N)
    cross = np.fft.ifft(np.conj(Fw) * Fa).real
    grad = 2.0 * cross[:n]

    return grad, conv.max()


def optimize_adam_smooth(a0, time_budget, beta_start=5.0, beta_end=500.0,
                         n_cont=12, lr_base=0.008):
    """
    Adam optimization with continuation on beta (logsumexp temperature).
    """
    n = len(a0)
    a = project_simplex(a0.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    betas = np.geomspace(beta_start, beta_end, n_cont)
    time_per_phase = time_budget * 0.85 / n_cont

    for phase in range(n_cont):
        if time.monotonic() - t0 > time_budget * 0.85:
            break
        beta = betas[phase]
        phase_end = t0 + (phase + 1) * time_per_phase
        lr = lr_base * (0.75 ** phase)

        m = np.zeros(n)
        v = np.zeros(n)
        t_adam = 0
        b1, b2, eps = 0.9, 0.999, 1e-8
        no_improve = 0

        while time.monotonic() < phase_end:
            t_adam += 1
            obj, grad = smooth_obj_grad(a, beta)

            m = b1 * m + (1 - b1) * grad
            v = b2 * v + (1 - b2) * grad ** 2
            m_hat = m / (1 - b1 ** t_adam)
            v_hat = v / (1 - b2 ** t_adam)

            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = project_simplex(a_new)

            if t_adam % 15 == 0:
                true_obj = fast_convolve(a).max()
                if true_obj < best_obj - 1e-15:
                    best_obj = true_obj
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1

                if no_improve > 100:
                    rng = np.random.default_rng(phase * 1000 + t_adam)
                    a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                    m = np.zeros(n)
                    v = np.zeros(n)
                    t_adam = 0
                    no_improve = 0
                    lr = max(lr * 0.5, 0.0001)

    # Final polish at high beta
    remaining = time_budget - (time.monotonic() - t0)
    if remaining > 1:
        beta = beta_end
        lr = lr_base * 0.15
        m = np.zeros(n)
        v = np.zeros(n)
        t_adam = 0
        b1, b2, eps = 0.9, 0.999, 1e-8

        while time.monotonic() - t0 < time_budget:
            t_adam += 1
            obj, grad = smooth_obj_grad(a, beta)
            m = b1 * m + (1 - b1) * grad
            v = b2 * v + (1 - b2) * grad ** 2
            m_hat = m / (1 - b1 ** t_adam)
            v_hat = v / (1 - b2 ** t_adam)
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = project_simplex(a_new)

            if t_adam % 30 == 0:
                true_obj = fast_convolve(a).max()
                if true_obj < best_obj - 1e-15:
                    best_obj = true_obj
                    best_a = a.copy()

    return best_a, best_obj


def optimize_subgrad(a0, time_budget, top_k=30, lr_init=0.0015, mom=0.92, temp_init=0.0008):
    """Peak-aware subgradient with momentum and adaptive temperature."""
    n = len(a0)
    a = project_simplex(a0.astype(np.float64).copy())
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_budget:
        grad, cur_obj = peak_subgradient(a, top_k=top_k, temp=temp)

        v = mom * v - lr * grad
        a = project_simplex(a + v)

        true_obj = fast_convolve(a).max()
        if true_obj < best_obj - 1e-15:
            best_obj = true_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9997
        temp *= 1.005

        if no_improve > 200:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            v = np.zeros(n)
            lr = max(lr_init * 0.5, 0.0002)
            temp = temp_init
            no_improve = 0

    return best_a, best_obj


def peak_shave(a0, n_trials=500, rng=None):
    """Direct peak shaving: reduce elements contributing to the peak."""
    if rng is None:
        rng = np.random.default_rng(42)
    n = len(a0)
    best_a = project_simplex(a0.astype(np.float64).copy())
    best_obj = fast_convolve(best_a).max()

    for _ in range(n_trials):
        conv = fast_convolve(best_a)
        # Get top few peaks
        n_peaks = min(5, len(conv))
        peak_idx = np.argsort(conv)[-n_peaks:]

        for pk in peak_idx:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            if je < js:
                continue
            idx = np.arange(js, je + 1)
            contrib = best_a[idx] * best_a[pk - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                continue

            # Pick elements to reduce (weighted by contribution)
            n_red = max(1, min(len(idx), rng.integers(1, len(idx) // 3 + 1)))
            probs = contrib / total_contrib
            chosen = rng.choice(len(idx), size=n_red, replace=False, p=probs)

            # Amount to reduce
            frac = rng.uniform(0.005, 0.02)
            reduction = frac * best_a[idx[chosen]].sum()

            new_a = best_a.copy()
            new_a[idx[chosen]] -= reduction / n_red
            new_a = np.maximum(new_a, 0)

            # Redistribute to random positions (prefer low-contributing ones)
            redist = rng.choice(n, size=n_red, replace=True)
            new_a[redist] += reduction / n_red
            new_a = project_simplex(new_a)

            obj = fast_convolve(new_a).max()
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_a = new_a
                break

    return best_a, best_obj


def coordinate_descent(a0, time_budget, rng=None):
    """Simple coordinate descent: try adjusting pairs of elements."""
    if rng is None:
        rng = np.random.default_rng(123)
    n = len(a0)
    best_a = project_simplex(a0.astype(np.float64).copy())
    best_obj = fast_convolve(best_a).max()

    t0 = time.monotonic()
    scale = 0.01

    while time.monotonic() - t0 < time_budget:
        improved = False
        for _ in range(200):
            if time.monotonic() - t0 > time_budget:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            delta = rng.normal(0, scale) * best_a[i]
            new_a = best_a.copy()
            new_a[i] -= delta
            new_a[j] += delta
            if new_a[i] < 0 or new_a[j] < 0:
                continue
            new_a = np.maximum(new_a, 0)
            s = new_a.sum()
            if s < 1e-10:
                continue
            new_a = new_a / s
            obj = fast_convolve(new_a).max()
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_a = new_a
                improved = True

        if not improved:
            scale *= 0.7
            if scale < 1e-6:
                scale = 0.005
                # Larger perturbation
                new_a = project_simplex(best_a + rng.standard_normal(n) * 0.01)
                obj = fast_convolve(new_a).max()
                if obj < best_obj:
                    best_obj = obj
                    best_a = new_a

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    inits.append(('uniform', np.ones(n) / n))

    # Sine-based
    for freq in [1, 2, 3, 5, 7]:
        for phase in [0, np.pi / 3, np.pi / 2, 2 * np.pi / 3]:
            s = np.sin(2 * np.pi * freq * x / n + phase)
            s = np.maximum(s, 0)
            if s.sum() > 1e-10:
                inits.append((f'sine_f{freq}_p{phase:.1f}', s / s.sum()))

    # Cosine
    for power in [1, 2, 3, 6, 12]:
        for freq in [1, 2, 3]:
            c = np.abs(np.cos(np.pi * freq * x / n)) ** power + 0.01
            inits.append((f'cos_f{freq}_p{power}', c / c.sum()))

    # Gaussian
    for cf in [0.3, 0.5, 0.7]:
        for sf in [0.08, 0.15, 0.25]:
            g = np.exp(-0.5 * ((x - cf * (n - 1)) / (sf * n)) ** 2)
            inits.append((f'g_c{cf}_s{sf}', g / g.sum()))

    # Two bumps
    for sf in [0.06, 0.1, 0.15]:
        c1 = np.exp(-0.5 * ((x - 0.3 * (n - 1)) / (sf * n)) ** 2)
        c2 = np.exp(-0.5 * ((x - 0.7 * (n - 1)) / (sf * n)) ** 2)
        two = c1 + c2
        inits.append((f'two_s{sf}', two / two.sum()))

    # Three bumps
    for sf in [0.06, 0.1]:
        c1 = np.exp(-0.5 * ((x - 0.2 * (n - 1)) / (sf * n)) ** 2)
        c2 = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (sf * n)) ** 2)
        c3 = np.exp(-0.5 * ((x - 0.8 * (n - 1)) / (sf * n)) ** 2)
        three = c1 + c2 + c3
        inits.append((f'three_s{sf}', three / three.sum()))

    # Dirichlet
    for alpha in [0.3, 0.5, 1.0]:
        d = rng.dirichlet(np.ones(n) * alpha)
        inits.append((f'dir_a{alpha}', d))

    # Comb
    for period in [2, 3]:
        comb = np.zeros(n)
        comb[::period] = 1.0
        inits.append((f'comb{period}', comb / comb.sum()))

    # Triangular
    tri = 1.0 - 2.0 * np.abs(x / (n - 1) - 0.5)
    inits.append(('tri', tri / tri.sum()))

    # Parent interpolation
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        if px.sum() > 1e-10:
            inits.append(('parent', px / px.sum()))

    return inits


def try_seq(seq, best_state):
    """Evaluate and update best state."""
    seq = np.maximum(seq, 0.0)
    if seq.sum() < 1e-10:
        return float('inf')
    score = evaluate_sequence(seq.tolist())
    if np.isfinite(score) and score < best_state['score']:
        best_state['score'] = score
        best_state['seq'] = seq.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(seq)})")
    return score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-strategy search combining smooth optimization,
    subgradient methods, peak shaving, and coordinate descent.
    """
    rng_master = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_state = {'score': float('inf'), 'seq': None}

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
    except Exception:
        pass

    # Determine lengths to try
    lengths_to_try = [256, 320, 384, 448, 512]
    if parent is not None:
        n_parent = len(parent)
        lengths_to_try = [n_parent] + [l for l in lengths_to_try if l != n_parent]

    print(f"Lengths to try: {lengths_to_try}")
    print(f"Time budget: {budget_s}s, deadline in {deadline - time.monotonic():.0f}s")

    # Phase 1: Quick evaluation of initializations
    print("\n=== Phase 1: Initialization ===")
    t_phase = time.monotonic()

    for n in lengths_to_try:
        if time.monotonic() > deadline - 300:
            break
        inits = generate_inits(n, parent, rng_master)
        # Quick eval of top initializations
        for label, init in inits[:20]:
            try_seq(init, best_state)

    if best_state['seq'] is None:
        best_state['seq'] = np.ones(384) / 384
        best_state['score'] = evaluate_sequence(best_state['seq'].tolist())

    n_best = len(best_state['seq'])
    print(f"  Best after init: {best_state['score']:.8f} (n={n_best})")

    # Phase 2: Smooth Adam optimization (primary)
    print("\n=== Phase 2: Adam Smooth Optimization ===")
    remaining = deadline - time.monotonic() - 100
    t_alloc = min(remaining * 0.35, 500)

    if t_alloc > 30:
        # Run on best
        a_opt, _ = optimize_adam_smooth(best_state['seq'], time_budget=t_alloc * 0.5,
                                        beta_start=5.0, beta_end=500.0,
                                        n_cont=12, lr_base=0.008)
        try_seq(a_opt, best_state)

        # Run with different params
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            a_opt, _ = optimize_adam_smooth(best_state['seq'], time_budget=min(rem * 0.4, 200),
                                            beta_start=2.0, beta_end=300.0,
                                            n_cont=10, lr_base=0.012)
            try_seq(a_opt, best_state)

    print(f"  Best after Adam: {best_state['score']:.8f}")
    n_best = len(best_state['seq'])

    # Phase 3: Subgradient optimization
    print("\n=== Phase 3: Peak Subgradient ===")
    remaining = deadline - time.monotonic() - 80
    t_alloc = min(remaining * 0.25, 350)

    if t_alloc > 20:
        a_opt, _ = optimize_subgrad(best_state['seq'], time_budget=t_alloc * 0.5,
                                    top_k=30, lr_init=0.0015, mom=0.92, temp_init=0.0008)
        try_seq(a_opt, best_state)

        rem = deadline - time.monotonic() - 60
        if rem > 15:
            a_opt, _ = optimize_subgrad(best_state['seq'], time_budget=min(rem * 0.4, 150),
                                        top_k=50, lr_init=0.001, mom=0.95, temp_init=0.0003)
            try_seq(a_opt, best_state)

    print(f"  Best after Subgrad: {best_state['score']:.8f}")
    n_best = len(best_state['seq'])

    # Phase 4: Peak shaving + coordinate descent
    print("\n=== Phase 4: Peak Shaving & Coordinate Descent ===")
    remaining = deadline - time.monotonic() - 60
    t_alloc = min(remaining * 0.2, 250)

    if t_alloc > 10:
        rng_ps = np.random.default_rng(999)
        a_opt, _ = peak_shave(best_state['seq'], n_trials=300, rng=rng_ps)
        try_seq(a_opt, best_state)

        rem = deadline - time.monotonic() - 40
        if rem > 10:
            a_opt, _ = coordinate_descent(best_state['seq'], time_budget=min(rem * 0.5, 100),
                                          rng=np.random.default_rng(888))
            try_seq(a_opt, best_state)

    print(f"  Best after shave: {best_state['score']:.8f}")
    n_best = len(best_state['seq'])

    # Phase 5: Alternating smooth + polish with restarts
    print("\n=== Phase 5: Restart & Polish ===")
    remaining = deadline - time.monotonic() - 40
    t_alloc = min(remaining * 0.6, 400)

    if t_alloc > 20:
        n_restarts = 4
        t_each = t_alloc / n_restarts
        for r in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break
            # Perturb
            noise_scale = 0.003 * (r + 1)
            f_pert = project_simplex(best_state['seq'] + rng_master.standard_normal(n_best) * noise_scale)

            # Quick Adam
            t_sub = min(t_each * 0.6, 80)
            a_opt, _ = optimize_adam_smooth(f_pert, time_budget=t_sub,
                                            beta_start=50.0, beta_end=400.0,
                                            n_cont=6, lr_base=0.004)
            try_seq(a_opt, best_state)

            # Quick subgrad
            rem = deadline - time.monotonic() - 25
            if rem > 10:
                a_opt, _ = optimize_subgrad(a_opt, time_budget=min(rem * 0.4, 40),
                                            top_k=30, lr_init=0.0008, mom=0.9, temp_init=0.0005)
                try_seq(a_opt, best_state)

    print(f"  Best after restarts: {best_state['score']:.8f}")
    n_best = len(best_state['seq'])

    # Phase 6: Final fine-tuning
    print("\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic() - 20
    if remaining > 10:
        # Multiple peak shaving rounds
        rng_final = np.random.default_rng(55555)
        for round_num in range(3):
            if time.monotonic() > deadline - 15:
                break
            a_opt, _ = peak_shave(best_state['seq'], n_trials=200, rng=rng_final)
            try_seq(a_opt, best_state)

        # Final subgradient with very fine parameters
        rem = deadline - time.monotonic() - 15
        if rem > 5:
            a_opt, _ = optimize_subgrad(best_state['seq'], time_budget=min(rem * 0.7, 60),
                                        top_k=80, lr_init=0.0005, mom=0.95, temp_init=0.0001)
            try_seq(a_opt, best_state)

    # Final verification
    if best_state['seq'] is not None:
        final_score = float(evaluate_sequence(best_state['seq'].tolist()))
        if np.isfinite(final_score) and final_score < best_state['score']:
            best_state['score'] = final_score
        print(f"\nFINAL SCORE: {best_state['score']:.10f} (n={len(best_state['seq'])})")
    else:
        best_state['seq'] = np.ones(256) / 256
        best_state['score'] = float(evaluate_sequence(best_state['seq'].tolist()))
        print(f"\nFALLBACK SCORE: {best_state['score']:.10f}")

    return [float(x) for x in best_state['seq']]
