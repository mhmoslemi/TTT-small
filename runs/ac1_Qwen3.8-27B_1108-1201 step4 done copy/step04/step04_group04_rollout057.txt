= a_j * dF/da_j - a_j * sum_i a_i * dF/da_i
= a_j * (dF/da_j - E_a[dF/da])

where E_a[dF/da] = sum_i a_i * dF/da_i.

This is computable! And it's the gradient in the "natural gradient" / Fisher metric sense.

    """Compute F_T(softmax(z)) and its gradient w.r.t. z."""
    # Numerical stability
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    # Compute conv and smoothed objective
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    F = T * (np.log(w_sum) + scaled.max())  # = T * log(sum(exp(conv/T)))
    
    # Gradient of F w.r.t. a
    p = w / w_sum
    # grad_a[i] = 2 * cross_correlation(a, p)[i]
    n = len(a)
    fft_size = 1 << (3*n).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    cross = np.fft.irfft(np.conj(A) * P, fft_size)
    grad_a = 2.0 * cross[:n]
    
    # Gradient w.r.t. z: grad_z_j = a_j * (grad_a_j - sum_i a_i * grad_a_i)
    ea = (a * grad_a).sum()
    grad_z = a * (grad_a - ea)
    
    return F, grad_z

from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    # Euclidean projection onto simplex
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax_and_grad(z, T):
    """Returns (F, grad_z) where F = T*log(sum(exp(conv(softmax(z))/T)))"""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    n = len(a)
    
    conv = fast_convolve(a)
    scaled = conv / T
    scaled_max = scaled.max()
    w = np.exp(scaled - scaled_max)
    w_sum = w.sum()
    F = T * (np.log(w_sum) + scaled_max)
    
    p = w / w_sum
    fft_size = 1 << (3*n).bit_length()
    A = rfft(a, fft_size)
    P = rfft(p, fft_size)
    cross = irfft(np.conj(A) * P, fft_size)
    grad_a = 2.0 * cross[:n]
    
    ea = (a * grad_a).sum()
    grad_z = a * (grad_a - ea)
    
    return F, grad_z

"""
Multi-method optimizer combining L-BFGS on smoothed objectives, LP-Frank-Wolfe,
subgradient polishing, and multi-length exploration to minimize
2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_obj_grad(a, T):
    """Compute smoothed objective and gradient w.r.t. a."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = conv / T
    scaled_max = scaled.max()
    w = np.exp(scaled - scaled_max)
    w_sum = w.sum()
    F = T * (np.log(w_sum) + scaled_max)
    p = w / w_sum
    fft_size = 1 << (3 * n).bit_length()
    A = rfft(a, fft_size)
    P = rfft(p, fft_size)
    cross = irfft(np.conj(A) * P, fft_size)
    grad_a = 2.0 * cross[:n]
    return F, grad_a


def softmax_obj_grad(z, T):
    """Compute smoothed objective and gradient w.r.t. softmax parameters z."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    n = len(a)
    F, grad_a = compute_smoothed_obj_grad(a, T)
    ea = (a * grad_a).sum()
    grad_z = a * (grad_a - ea)
    return F, grad_z


def lbgfs_smoothed(a0, T, max_iter=300):
    """L-BFGS optimization on smoothed objective with softmax parameterization."""
    n = len(a0)
    a0_clipped = np.clip(a0, 1e-12, None)
    a0_norm = a0_clipped / a0_clipped.sum()
    z0 = np.log(a0_norm)

    def obj_func(z):
        return softmax_obj_grad(z, T)[0]

    def grad_func(z):
        return softmax_obj_grad(z, T)[1]

    result = minimize(
        fun=obj_func,
        x0=z0,
        jac=grad_func,
        method='L-BFGS-B',
        options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-8, 'maxls': 50}
    )

    z_opt = result.x
    z_shifted = z_opt - z_opt.max()
    exp_z = np.exp(z_shifted)
    a_opt = exp_z / exp_z.sum()
    return a_opt


def projected_grad_steps(a0, T, n_steps=5000, lr_init=0.01, verbose=False):
    """Projected gradient descent with adaptive step size on smoothed objective."""
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    lr = lr_init

    for i in range(n_steps):
        conv = fast_convolve(f)
        F, grad = compute_smoothed_obj_grad(f, T)

        # Project gradient to tangent space
        grad = grad - np.mean(grad)

        f_new = project_simplex(f - lr * grad)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
        else:
            lr = max(lr * 0.9, 1e-6)

    return best_f, best_obj


def lf_wolfe(f0, time_limit):
    """LP-based Frank-Wolfe optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        # Build LP constraint matrix
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M * m_exp)
        bounds = [(0, None)] * n

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'time_limit': max(1.0, remaining - 1)})
        except Exception:
            break

        if not res.success or res.x is None:
            m_exp = min(m_exp * 1.15, 5.0)
            if m_exp > 5.0:
                break
            continue

        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 40:
                break
            m_exp = min(m_exp * 1.03, 3.0)
            continue

        # Line search (coarse + fine)
        best_alpha = 0.0
        best_val = M
        alphas = np.linspace(0.05, 1.0, 15)
        for alpha in alphas:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine bisection
        if best_alpha > 0.05:
            lo = max(0.0, best_alpha - 0.08)
            hi = min(1.0, best_alpha + 0.08)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:

            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def subgradient_polish(f0, time_limit, lr=0.001, temp=0.00003, top_k_frac=0.05):
    """Subgradient method with weighted top-k peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    top_k = max(5, int(n * top_k_frac))
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        f = project_simplex(f - lr * sg)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if it % 100 == 0:
            lr *= 0.998
            temp *= 1.005

        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 100000) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.0008)
            lr = 0.0006
            no_improve = 0

        if co > best_obj * 1.15:
            f = best_f.copy()

    return best_f, best_obj


def generate_diverse_inits(n, parent, rng):
    """Generate diverse initial sequences."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    # U-shape
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # W-shape (two edge peaks + center dip)

    w2 = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)

    inits.append(w_shape / w_shape.sum())

    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c1 / c1.sum())

    half_n = n // 2

    sym[:half_n] = half
    sym[half_n:] = half[::-1]

    if parent is not None:

        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:

            f_pert = project_simplex(f_work + noise)
            a_opt, _ = subgradient_polish(f_pert, time_limit=min(8, rem - 3),
                                          lr=0.0002, temp=0.000005, top_k_frac=0.02)
            best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "micro")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

        # One final L-BFGS at very low T
        rem = p4_budget - (time.monotonic() - t4)
        if rem > 20:
            try:
                a_opt = lbgfs_smoothed(f_work, 0.00005, max_iter=100)
                best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "final_lbgfs")
            except Exception:
                pass

    # Fallback
    if best_seq is None:
        print("  WARNING: No valid sequence found, using fallback")
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
