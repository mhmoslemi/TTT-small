    """Build matrix C such that C @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    C = np.zeros((m, n))
    for j in range(n):
        # C[k, j] = a[j] when 0 <= k-j < n, i.e., j <= k <= j+n-1
        # But also 0 <= k-j < n means k-j >= 0 (k >= j) and k-j < n (k < j+n)
        lo = j  # k >= j
        hi = min(j + n, m)  # k < j+n, but also k < m
        if lo < hi:
            C[lo:hi, j] = a[j]
    return C


def compute_subgradient(a, conv, top_k=50, temp=0.01):
    """Compute smoothed subgradient of max(conv) using top-k peaks."""
    n = len(a)
    # Get top-k peak positions
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        weights = np.ones(len(peaks)) / len(peaks)
    else:
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peak_indices]
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(top_k)
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum
        peaks = peak_indices
        weights = w

    # Subgradient: for peak at position k, d(b[k])/d(a[j]) = a[k-j] (if 0<=k-j<n)
    grad = np.zeros(n)
    for idx, k in enumerate(peaks):
        if weights[idx] < 1e-14:
            continue
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad[j] += w * a[k-j] for j in [lo, hi]
        # a[k-j] for j in [lo, hi] is a[k-hi : k-lo+1] reversed... 
        # Actually: for j from lo to hi, the term is a[k-j]
        # k-j goes from k-lo down to k-hi
        # So a[k-j] = a[k-hi], ..., a[k-lo] = a[hi-... ]
        # Let me just compute it directly
        js = np.arange(lo, hi + 1)
        grad[js] += weights[idx] * a[k - js]
    return grad


def lp_direction(a, M, C=None):
    """Solve LP to find direction that stays within peak constraint M."""
    n = len(a)
    if C is None:
        C = build_conv_matrix(a)
    
    c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True})
        if result.success:
            g = np.maximum(result.x, 0)
            if g.sum() > 1e-10:
                return g, True
    except Exception:
        pass
    return a.copy(), False


def lp_step(a, conv, n):
    """One LP-based step with quadratic step sizing."""
    M = conv.max()
    
    C = build_conv_matrix(a)
    g, success = lp_direction(a, M, C)
    
    if not success:
        return a.copy(), False
    
    d = g - a
    if np.linalg.norm(d) < 1e-12:
        return a.copy(), False
    
    # Quadratic step sizing
    # At peak k*, conv(a + t*d, a + t*d)[k*] = M + 2t*conv(a,d)[k*] + t^2*conv(d,d)[k*]
    # We want this <= M, so: 2*conv(a,d)[k*]*t + conv(d,d)[k*]*t^2 <= 0
    # If conv(a,d)[k*] < 0: t <= -2*conv(a,d)[k*] / conv(d,d)[k*]
    
    peak_idx = int(np.argmax(conv))
    # Compute conv(a, d) at peak
    # conv(a,d)[k] = sum_j a[j]*d[k-j]
    # Instead of full convolution, just compute at the peak
    ad_peak = 0.0
    lo = max(0, peak_idx - n + 1)
    hi = min(n - 1, peak_idx)
    for j in range(lo, hi + 1):
        ad_peak += a[j] * d[peak_idx - j]
    
    if ad_peak >= -1e-12:
        # The LP direction doesn't reduce the peak at k*
        # Try a smaller step
        t_candidates = [0.05, 0.02, 0.01]
    else:
        # Estimate optimal t
        dd_peak = 0.0
        for j in range(lo, hi + 1):
            dd_peak += d[j] * d[peak_idx - j]
        
        if dd_peak > 1e-12:
            t_opt = -2 * ad_peak / dd_peak
            t_candidates = [min(1.0, t_opt * 0.8), min(1.0, t_opt * 0.5), 0.3, 0.15, 0.05]
        else:
            t_candidates = [0.3, 0.15, 0.05, 0.02]
    
    t_candidates = [t for t in t_candidates if 0 < t < 1.0]
    if not t_candidates:
        t_candidates = [0.05]
    
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    for t in sorted(t_candidates, reverse=True):
        a_try = a + t * d
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        conv_try = fast_convolve(a_try)
        obj_try = 2.0 * n * conv_try.max()
        if obj_try < best_obj - 1e-10:
            best_obj = obj_try
            best_a = a_try.copy()
            break
    
    if np.linalg.norm(best_a - a) > 1e-12:
        return best_a, True
    return a.copy(), False


def subgradient_optimize(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=50, temp_init=0.01, l2_reg=1e-6):
    """Subgradient method with Nesterov acceleration."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    y = a.copy()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Nesterov lookahead
        y = a + mom * v
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        
        conv_y = fast_convolve(y)
        grad = compute_subgradient(y, conv_y, top_k=top_k, temp=temp)
        
        # Add small L2 regularization to encourage spreading
        grad += l2_reg * y
        
        # Update
        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn > 1e-15:
            a = a_new / sn
        else:
            a = best_a.copy()
            v = np.zeros(n)
            break
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-11:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive updates
        lr *= 0.999
        temp *= 1.005
        mom = min(0.98, mom + 0.0001)
        
        if no_improve >= 300:
            break
    
    return best_a, best_obj


def peak_shave(a, conv, n, strength=0.1):
    """Directly reduce the peak by shaving contributing elements."""
    peak_idx = int(np.argmax(conv))
    M = conv[peak_idx]
    
    # Find contributors to the peak
    lo = max(0, peak_idx - n + 1)
    hi = min(n - 1, peak_idx)
    contributors = np.arange(lo, hi + 1)
    weights = np.array([a[j] * a[peak_idx - j] for j in contributors])
    total = weights.sum()
    
    if total < 1e-15:
        return a.copy(), False
    
    # Reduce contributors proportionally
    a_new = a.copy()
    # Reduce by a fraction proportional to their contribution
    reductions = strength * weights / total
    # But we can only reduce so much (can't go below 0)
    reductions = np.minimum(reductions, a[contributors] * 0.5)
    a_new[contributors] -= reductions
    
    # Add the reduced mass to positions that DON'T contribute to the peak
    # Good positions: far from the peak
    mass_added = reductions.sum()
    if mass_added < 1e-15:
        return a.copy(), False
    
    # Spread mass to edge positions
    edge_size = max(1, n // 10)
    edge_indices = list(range(0, edge_size)) + list(range(n - edge_size, n))
    # Weight by current values (add more where there's already mass)
    edge_weights = a[edge_indices] + 1e-8
    edge_weights /= edge_weights.sum()
    a_new[edge_indices] += mass_added * edge_weights
    
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    
    return a_new, True


def combined_optimize(a0, time_limit):
    """Combined LP + subgradient + peak shaving optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    method = 'lp'  # alternate between methods
    lp_count = 0
    perturb_count = 0
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        if no_improve >= 50:
            # Try peak shaving
            a_try, success = peak_shave(best_a, fast_convolve(best_a), n, strength=0.05 + 0.02 * perturb_count)
            if success:
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-12:
                    a = a_try
                    best_obj = obj_try
                    best_a = a_try.copy()
                    no_improve = 0
                    perturb_count += 1
                    continue
            
            # Try perturbation
            scale = 0.003 + 0.003 * min(perturb_count, 10)
            noise = rng.standard_normal(n) * scale
            a = np.maximum(best_a + noise, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            else:
                a = best_a.copy()
            conv = fast_convolve(a)
            no_improve = 0
            perturb_count += 1
            
            if perturb_count > 50:
                break
            continue
        
        if method == 'lp':
            a_new, improved = lp_step(a, conv, n)
            if improved:
                a = a_new
                conv = fast_convolve(a)
                obj = 2.0 * n * conv.max()
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a.copy()
                no_improve = 0
                lp_count += 1
            else:
                no_improve += 1
                method = 'grad'
        else:
            # Gradient step
            t_grad = min(1.0, remaining * 0.02)
            a_grad, obj_grad = subgradient_optimize(a, time_limit=t_grad, lr_init=0.005, top_k=80, temp_init=0.005)
            if obj_grad < best_obj - 1e-12:
                a = a_grad
                best_obj = obj_grad
                best_a = a_grad.copy()
                conv = fast_convolve(a)
                no_improve = 0
            else:
                no_improve += 1
            method = 'lp'
    
    return best_a, best_obj


def generate_initializations(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Sine squared
    s1 = (np.sin(np.pi * x / n) ** 2 + 0.01)
    inits['sine1'] = s1 / s1.sum()
    
    s2 = (np.sin(2 * np.pi * x / n) ** 2 + 0.01)
    inits['sine2'] = s2 / s2.sum()
    
    # Bimodal
    edge = max(1, n // 6)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # Gaussian (centered)
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.15 * n)) ** 2)
    inits['gaussian'] = g / g.sum()
    
    # Cosine window
    cw = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    inits['cosine'] = cw / cw.sum()
    
    # Hamming window
    hw = 0.54 - 0.46 * np.cos(2 * np.pi * x / (n - 1))
    inits['hamming'] = hw / hw.sum()
    
    # Blackman window
    bw = 0.42 - 0.5 * np.cos(2 * np.pi * x / (n - 1)) + 0.08 * np.cos(4 * np.pi * x / (n - 1))
    inits['blackman'] = bw / bw.sum()
    
    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps
    
    return inits
