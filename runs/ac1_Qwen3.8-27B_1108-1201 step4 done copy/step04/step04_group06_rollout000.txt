"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using enhanced multi-peak LP, 
spectral initialization, and adaptive optimization.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def compute_gradient_softmax(a, conv, top_k=30, temp=0.01):
    """Compute gradient of softmax approximation to max convolution."""
    n = len(a)
    # Use top-k peaks with softmax weighting
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    
    peak_vals = conv[peaks]
    scaled = peak_vals / max(temp, 1e-10)
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum

    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-14:
            continue
        k = int(peaks[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += w[idx] * a[k - idxs]
    return grad


def lp_multipeak(a, M, top_k=5, C=None):
    """
    Solve LP: max sum(g) s.t. (a*g)[k] <= M for top-k peaks, g >= 0.
    Returns (g, success).
    """
    n = len(a)
    if C is None:
        C = build_conv_matrix(a)
    
    conv = C @ a  # This is a * a
    # Get top-k peak indices
    if top_k >= len(conv):
        peak_indices = np.arange(len(conv))
    else:
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
    peak_indices = sorted(peak_indices.tolist())
    
    # Build reduced constraint matrix
    C_red = C[peak_indices, :]
    b_red = np.full(len(peak_indices), M)
    
    c_vec = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_vec, A_ub=C_red, b_ub=b_red, bounds=bounds, 
                        method='highs', options={'presolve': True})
        if result.success:
            g = np.maximum(result.x, 0)
            if g.sum() > 1e-10:
                return g, True
    except Exception:
        pass
    return None, False

def spectral_flatten(a, iterations=50, strength=0.1):
    """
    Iteratively apply spectral equalization to flatten the auto-correlation.
    Works in log-frequency domain.
    """
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for _ in range(iterations):
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        # We want power to be flat (the auto-conv should be flat)
        # Apply gentle equalization
        avg_power = power.mean()
        # Reduce peaks in power spectrum
        ratio = np.minimum(1.0, (avg_power / power) ** strength)
        ratio = np.maximum(ratio, 0.1)
        A_eq = A * np.sqrt(ratio)
        a_new = np.fft.irfft(A_eq, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new
    
    return a


def generate_inits(n, parent, rng):
    """Generate diverse initializations for sequence of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Sine-based
    s1 = np.sin(np.pi * x / n) ** 2 + 0.001
    inits['sine1'] = s1 / s1.sum()
    
    s2 = np.sin(2 * np.pi * x / n) ** 2 + 0.001
    inits['sine2'] = s2 / s2.sum()
    
    # Gaussian
    g = np.exp(-0.5 * ((x - n/2) / (n/4)) ** 2) + 0.001
    inits['gaussian'] = g / g.sum()
    
    # Bimodal
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi += 0.01
    inits['bimodal'] = bi / bi.sum()
    
    # Symmetric triangular (inverted)
    tri = 1.0 - np.abs(x - n/2 + 0.5) / (n/2)
    tri = np.maximum(tri, 0.01)
    inits['inv_tri'] = tri / tri.sum()
    
    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            p = p / ps
            if len(p) == n:
                inits['parent'] = p
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps
    
    return inits

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def compute_gradient_softmax(a, conv, top_k, temp):
    """Compute gradient of softmax approximation to max convolution."""
    n = len(a)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    
    peak_vals = conv[peaks]
    scaled = peak_vals / max(temp, 1e-10)
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum

    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-14:
            continue
        k = int(peaks[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += w[idx] * a[k - idxs]
    return grad


def lp_multipeak(a, M, top_k, C):
    """
    Solve LP: max sum(g) s.t. (a*g)[k] <= M for top-k peaks, g >= 0.
    Returns (g, success).
    """
    n = len(a)
    conv_aa = C @ a
    if top_k >= len(conv_aa):
        peak_indices = np.arange(len(conv_aa))
    else:
        peak_indices = np.argpartition(conv_aa, -top_k)[-top_k:]
    peak_indices = np.sort(peak_indices)
    
    C_red = C[peak_indices, :]
    b_red = np.full(len(peak_indices), M)
    c_vec = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_vec, A_ub=C_red, b_ub=b_red, bounds=bounds,
                        method='highs', options={'presolve': True})
        if result.success:
            g = np.maximum(result.x, 0)
            if g.sum() > 1e-10:
                return g, True
    except Exception:
        pass
    return None, False

def spectral_flatten(a, iterations, strength):

    a = np.maximum(a.astype(np.float64), 0).copy()

        return a

        A = np.fft.rfft(a, fft_size)

        ratio = np.minimum(1.0, (avg_power / np.maximum(power, 1e-20)) ** strength)
        ratio = np.maximum(ratio, 0.05)

        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new
    
    return a


def generate_inits(n, parent, rng):
    """Generate diverse initializations for sequence of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    s1 = np.sin(np.pi * x / n) ** 2 + 0.001
    inits['sine1'] = s1 / s1.sum()
    
    s2 = np.sin(2 * np.pi * x / n) ** 2 + 0.001
    inits['sine2'] = s2 / s2.sum()
    
    g = np.exp(-0.5 * ((x - n / 2) / (n / 4)) ** 2) + 0.001
    inits['gaussian'] = g / g.sum()
    
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi += 0.01
    inits['bimodal'] = bi / bi.sum()
    
    tri = 1.0 - np.abs(x - n / 2 + 0.5) / (n / 2)
    tri = np.maximum(tri, 0.01)
    inits['inv_tri'] = tri / tri.sum()
    
    # Smoother: raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / n)) + 0.001
    inits['raised_cos'] = rc / rc.sum()
    
    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            p = p / ps
            if len(p) == n:
                inits['parent'] = p
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps
    
    return inits

        t_alloc = min(t_per_lp, deadline - time.monotonic() - 350)
        if t_alloc < 15:
            break
        
        print(f"  LP candidate {i}: n={n}, t={t_alloc:.0f}s")
        a_opt = np.maximum(a0, 0).copy()
        s = a_opt.sum()
        if s > 1e-15:
            a_opt = a_opt / s
        
        a_final, obj = lp_optimize(a_opt, time_limit=t_alloc,
                                     initial_top_k=3, max_top_k=12)
        best_sequence, best_score = try_update_best(a_final, best_sequence, best_score)
        print(f"  LP {i} (n={n}): obj={obj:.6f}, best_score={best_score:.6f}")
    
    print(f"Phase 2 done ({time.monotonic()-t0:.0f}s). Best: {best_score:.6f}")
    
    # =========================================================
    # Phase 3: Extended LP on best with perturbations
    # =========================================================
    print("=== Phase 3: Extended LP Polish ===")
    t0 = time.monotonic()
    
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 300
        if remaining > 30:
            n_rounds = 4
            for round_i in range(n_rounds):
                if time.monotonic() > deadline - 250:
                    break
                t_alloc = min(remaining / (n_rounds - round_i) * 0.7,
                             deadline - time.monotonic() - 200)
                if t_alloc < 10:
                    break
                
                if round_i == 0:
                    a0 = best_sequence.copy()
                elif round_i == 1:
                    noise = rng.standard_normal(len(best_sequence)) * 0.001
                    a0 = np.maximum(best_sequence + noise, 0)
                    s = a0.sum()
                    a0 = a0 / s if s > 1e-15 else best_sequence.copy()
                elif round_i == 2:
                    # Spectral perturbation
                    a0 = spectral_flatten(best_sequence, iterations=30, strength=0.08)
                else:
                    noise = rng.standard_normal(len(best_sequence)) * 0.003
                    a0 = np.maximum(best_sequence + noise, 0)
                    s = a0.sum()
                    a0 = a0 / s if s > 1e-15 else best_sequence.copy()
                
                a_opt, obj = lp_optimize(a0, time_limit=t_alloc,
                                           initial_top_k=5, max_top_k=20)
                best_sequence, best_score = try_update_best(a_opt, best_sequence, best_score)
                print(f"  Polish {round_i}: {best_score:.6f}")
    
    print(f"Phase 3 done ({time.monotonic()-t0:.0f}s). Best: {best_score:.6f}")
    
    # =========================================================
    # Phase 4: Spectral + gradient fine-tuning
    # =========================================================
    print("=== Phase 4: Fine-tuning ===")
    t0 = time.monotonic()
    
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20
        
        # Spectral approach
        if remaining > 40:
            a_spec = spectral_flatten(best_sequence, iterations=150, strength=0.04)
            t_sg = min(20.0, remaining * 0.25)
            if t_sg > 5:
                a_opt, obj = gradient_optimize(a_spec, time_limit=t_sg,
                                                lr_init=0.005, momentum=0.93,
                                                top_k=60, temp_init=0.001)
                best_sequence, best_score = try_update_best(a_opt, best_sequence, best_score)
                print(f"  Spectral: {best_score:.6f}")
        
        # Gradient refinement with varied params
        remaining = deadline - time.monotonic() - 15
        if remaining > 15:
            configs = [
                (0.003, 0.95, 80, 0.0005),
                (0.008, 0.90, 30, 0.002),
                (0.001, 0.97, 120, 0.0002),
            ]
            for ci, (lr, mom, tk, tp) in enumerate(configs):
                if time.monotonic() > deadline - 10:
                    break
                t_sg = min(remaining / (len(configs) - ci) * 0.8, 20.0)
                if t_sg < 5:
                    break
                a_opt, obj = gradient_optimize(best_sequence, time_limit=t_sg,
                                                lr_init=lr, momentum=mom,
                                                top_k=tk, temp_init=tp)
                best_sequence, best_score = try_update_best(a_opt, best_sequence, best_score)
                print(f"  GradFine {ci}: {best_score:.6f}")
    
    print(f"Phase 4 done ({time.monotonic()-t0:.0f}s). Best: {best_score:.6f}")
    
    # =========================================================
    # Phase 5: Final LP micro-polish
    # =========================================================
    if best_sequence is not None and time.monotonic() < deadline - 5:
        t_final = min(15.0, deadline - time.monotonic() - 2)
        if t_final > 5:
            a_opt, obj = lp_optimize(best_sequence, time_limit=t_final,
                                       initial_top_k=5, max_top_k=15)
            best_sequence, best_score = try_update_best(a_opt, best_sequence, best_score)
            print(f"  FinalLP: {best_score:.6f}")
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
