def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def eval_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

        else:

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x: x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(conv, a, top_k=10, temperature=0.01):
    """Compute weighted subgradient of max(conv) with respect to a.
    Returns: sum_i w_i * shift(a, m_i)
    where shift(a, m)[j] = a[m-j] if 0 <= m-j < n, else 0
    """
    n = len(a)
    conv_len = len(conv)
    
    if top_k >= conv_len:
        peaks = np.arange(conv_len)
        weights = np.ones(conv_len) / conv_len
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        # Softmax weights
        scaled = pv / temperature
        scaled -= scaled.max()
        exp_s = np.exp(scaled)
        weights = exp_s / exp_s.sum()
    
    # Compute weighted sum of shifted a's
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if weights[i] < 1e-15:
            continue
        m = int(peaks[i])
        # shift(a, m)[j] = a[m-j] for j where 0 <= m-j < n
        # i.e., j in [max(0, m-n+1), min(n-1, m)]
        j_lo = max(0, m - n + 1)
        j_hi = min(n - 1, m)
        if j_lo <= j_hi:
            # a[m-j] for j in [j_lo, j_hi] is a[m-j_hi : m-j_lo+1] reversed
            # a[m-j_lo], a[m-j_lo-1], ..., a[m-j_hi]
            # = a[m-j_lo : m-j_hi-1, -1] = a[m-j_hi : m-j_lo+1][::-1]
            grad[j_lo:j_hi+1] += weights[i] * a[m-j_hi:m-j_lo+1][::-1]
    
    return grad

def subgradient_optimize(a0, time_limit, top_k=10, temp_init=0.005, lr_init=None):
    """Subgradient descent with top-k peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    if lr_init is None:
        lr_init = 0.1 / top_k
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    lr = lr_init
    temperature = temp_init
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        grad = compute_subgradient(conv, a, top_k, temperature)
        
        # Normalize gradient for stable step
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm
        
        a_new = project_simplex(a - lr * grad)
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            # Slowly increase lr
            lr = min(lr * 1.005, lr_init * 3)
        else:
            a = a_new  # Still move (subgradient methods don't need to always decrease)
            no_improve += 1
            if no_improve >= 20:
                lr *= 0.9
            if no_improve > 200:
                # Restart from best
                a = best_a.copy()
                lr = lr_init
                temperature = temp_init * (1 + it / 1000)
                no_improve = 0
        
        # Increase temperature slowly (explore more peaks)
        temperature *= 1.002
    
    return best_a, best_obj

def coord_descent(a0, time_limit):
    """Coordinate descent with incremental conv updates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    conv = np.array(np.convolve(a, a), dtype=np.float64)
    best_max = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        # One full sweep
        for j in range(n):
            # Find delta that minimizes max_m(conv[m] + 2*delta*a[m-j] + delta^2*I(m=2j))
            # We search on [-a[j], 0] (decreasing a[j] only helps)
            
            if a[j] < 1e-15:
                continue
            
            # Precompute a_shifted: a_shifted[m] = a[m-j] for valid m
            a_shifted = np.zeros(len(conv))
            lo = max(0, j)
            hi = min(len(conv) - 1, j + n - 1)
            a_shifted[lo:hi+1] = a[0:hi-lo+1]
            # Note: a_shifted[m] = a[m-j], and for m=2j, a_shifted[2j] = a[j] (current value)
            
            # Ternary search for optimal delta in [-a[j], 0]
            lo_d = -a[j]
            hi_d = 0.0
            
            # Quick check: is delta=0 already optimal?
            # f(0) = conv.max()
            # f(-a[j]) = max(conv - 2*a[j]*a_shifted + a[j]^2 * I(m=2j))
            f_0 = conv.max()
            
            # Evaluate at a few points to bracket
            mid_d = lo_d / 2
            f_mid = (conv + 2.0 * mid_d * a_shifted).max()
            if 2*j < len(conv):
                f_mid = max(f_mid, conv[2*j] + 2.0*mid_d*a[j] + mid_d*mid_d)
            
            if f_mid >= f_0:
                # Optimum is at 0 or very close
                continue
            
            # Ternary search
            for _ in range(40):
                m1 = lo_d + (hi_d - lo_d) / 3.0
                m2 = hi_d - (hi_d - lo_d) / 3.0
                
                # f(m1)
                val1 = conv + 2.0 * m1 * a_shifted
                if 2*j < len(conv):
                    val1 = val1.copy()
                    val1[2*j] = conv[2*j] + 2.0*m1*a[j] + m1*m1
                f1 = val1.max()
                
                # f(m2)
                val2 = conv + 2.0 * m2 * a_shifted
                if 2*j < len(conv):
                    val2 = val2.copy()
                    val2[2*j] = conv[2*j] + 2.0*m2*a[j] + m2*m2
                f2 = val2.max()
                
                if f1 < f2:
                    hi_d = m2
                else:
                    lo_d = m1
            
            delta = (lo_d + hi_d) / 2.0
            
            if abs(delta) < 1e-14:
                continue
            
            # Update a and conv
            old_val = a[j]
            a[j] = max(0.0, old_val + delta)
            delta = a[j] - old_val
            
            # Incremental conv update: conv[m] += 2*delta*a_old[m-j] + delta^2*I(m=2j)
            # a_old[m-j] = a_shifted[m] (computed before the update)
            # But a_shifted[2j] = old_val (the old a[j])
            conv[lo:hi+1] += 2.0 * delta * a_shifted
            if 0 <= 2*j < len(conv):
                conv[2*j] += delta * delta
        
        # Normalize
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        conv = fast_convolve(a)  # Recompute for accuracy after normalization
        
        cur = conv.max()
        if cur < best_max:
            best_max = cur
            best_a = a.copy()
    
    return best_a, best_max

def make_symmetric(a):
    """Make a symmetric (mirror)."""
    n = len(a)
    result = np.zeros(n)
    for i in range(n):
        result[i] = max(a[i], a[n-1-i])
    return result

def generate_initializations(n, seed=42):
    """Generate diverse initializations of length n."""
    rng = np.random.default_rng(seed)
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Symmetric bell shape
    x = np.linspace(0, 1, n)
    bell = np.exp(-4 * (x - 0.5)**2)
    inits.append(bell / bell.sum())
    
    # Symmetric U-shape
    u_shape = (x - 0.5)**2 + 0.05
    inits.append(u_shape / u_shape.sum())
    
    # Cosine bell
    cos_bell = np.cos(np.pi * (x - 0.5))**2
    inits.append(cos_bell / cos_bell.sum())
    
    # Power law (asymmetric)
    power = x**2 + 0.01
    inits.append(power / power.sum())
    
    # Random (several)
    for _ in range(5):
        r = rng.random(n)
        r = r + 0.01  # Avoid zeros
        inits.append(r / r.sum())
    
    # Symmetric random
    half = n // 2
    for _ in range(3):
        r = rng.random(half) + 0.01
        if n % 2 == 0:
            sym = np.concatenate([r, r[::-1]])
        else:
            sym = np.concatenate([r, [r[half-1]], r[::-1]])[:n]
        inits.append(sym / sym.sum())
    
    # "Double peak"
    peak1 = np.exp(-50 * (x - 0.25)**2)
    peak2 = np.exp(-50 * (x - 0.75)**2)
    inits.append((peak1 + peak2) / (peak1 + peak2).sum())
    
    return inits

    """Evaluate and track best."""
    a = np.maximum(a, 0.0)

        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"    NEW BEST: {score:.8f} (n={len(a)})")
    return score
