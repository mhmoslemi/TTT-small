"""

log-sum-exp smoothing, and peak-reducing coordinate descent for final polish.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_dir(f, M):
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None

def fista_smoothed(f0, time_limit, T_init=0.003, T_min=0.00005, verbose=False):
    """FISTA (accelerated proximal gradient) with log-sum-exp smoothing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    y = f.copy()  # FISTA momentum variable
    t_k = 1.0  # FISTA step parameter
    T = T_init
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = 1.0  # Will be adapted
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute gradient at y
        conv_y = fast_convolve(y)
        scaled = conv_y / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            p = np.zeros(len(conv_y))
            p[np.argmax(conv_y)] = 1.0
        else:
            p = w / ws
        
        # Gradient: 2 * cross_correlation(y, p)
        fft_size = 1 << (3 * n).bit_length()
        Y = np.fft.rfft(y, fft_size)
        P = np.fft.rfft(p, fft_size)
        cross = np.fft.irfft(np.conj(Y) * P, fft_size)
        grad = 2.0 * cross[:n]
        
        # Project gradient to tangent space
        grad = grad - np.mean(grad)
        
        # Step size (use 1/L where L is Lipschitz constant estimate)
        # L ≈ 2 * max|cross| * n / T roughly, but use adaptive
        alpha = 1.0 / (2.0 * np.max(np.abs(grad)) * n / T + 1e-10)
        alpha = min(alpha, 0.1)  # Cap step size
        
        # FISTA update
        f_new = project_simplex(y - alpha * grad)
        
        # Momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        
        # Check objective
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        
        # FISTA momentum
        beta = (t_k - 1.0) / t_new
        y = f_new + beta * (f_new - f)
        y = np.maximum(y, 0.0)
        y = y / max(y.sum(), 1e-15)
        f = f_new
        t_k = t_new
        
        # Anneal T
        T = max(T * 0.998, T_min)
        
        if no_improve > 200:
            rng = np.random.default_rng((it * 331 + 17) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            y = f.copy()
            t_k = 1.0
            T = max(T_init * 0.5, T_min)
            no_improve = 0
        
        if verbose and it % 500 == 0:
            print(f"    FISTA iter {it}: obj={best_obj:.10f}, T={T:.6f}")
    
    return best_f, best_obj


def coord_descent_polish(f0, time_limit, verbose=False):
    """Peak-reducing coordinate descent: move mass from high-contribution to low-contribution elements."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Find top peaks
        top_k = max(3, n // 50)
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peak_indices]
        
        # Compute damage for each element: how much does increasing a[j] increase the peaks?
        damage = np.zeros(n)
        for k in peak_indices:
            w = peak_vals[k]  # weight proportional to peak value
            # (a*a)[k] = sum_j a[j] * a[k-j]
            # d(a*a)[k] / da[j] = a[k-j] (for valid indices)
            js = range(max(0, k - n + 1), min(n, k + 1))
            for j in js:
                damage[j] += w * f[k - j]
        
        # Normalize damage
        damage_max = damage.max()
        if damage_max < 1e-15:
            break
        damage = damage / damage_max
        
        # Move mass from high-damage to low-damage elements
        # Find the element with max damage and min damage (among positive elements)
        pos_mask = f > 1e-10
        if not pos_mask.any():
            break
        
        # Move a small amount from worst to best
        step = 0.001 * f.sum()
        src = np.argmax(damage * pos_mask)
        dst_candidates = np.where(pos_mask | (f < 0.01))[0]
        # Pick the destination with lowest damage
        dst = dst_candidates[np.argmin(damage[dst_candidates])]
        
        if src == dst or f[src] < step:
            # Try a more global move
            f_try = f.copy()
            # Move mass proportional to (damage - mean_damage)
            mean_d = np.mean(damage[pos_mask])
            delta = np.zeros(n)
            delta[pos_mask] = (damage[pos_mask] - mean_d) * step
            delta -= delta.sum()  # Ensure sum is zero
            f_try = f + delta
            f_try = np.maximum(f_try, 0.0)
            f_try = f_try / f_try.sum()
        else:
            f_try = f.copy()
            f_try[src] -= step
            f_try[dst] += step
            f_try = np.maximum(f_try, 0.0)
            f_try = f_try / f_try.sum()
        
        conv_try = fast_convolve(f_try)
        obj_try = conv_try.max()
        
        if obj_try < best_obj - 1e-15:
            best_obj = obj_try
            best_f = f_try.copy()
            f = f_try
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                step *= 0.5
                if step < 1e-8:
                    break
            if no_improve > 500:
                rng = np.random.default_rng((it * 733 + 41) % (2**31))
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                step = 0.001
                no_improve = 0
    
    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.001, top_k_frac=0.05, temp=0.00003):

    s = f.sum()

        f = np.ones(len(f)) / len(f)

        f = f / s
    n = len(f)

    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    top_k = max(5, int(n * top_k_frac))
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        f = project_simplex(f - lr * sg)
        conv = fast_convolve(f)
        co = conv.max()
        if co < best_obj - 1e-14:

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_dir(f, M):
    """Solve LP: max 1^T g s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None

                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 4.0)
        if no_improve > 35:
            rng = np.random.default_rng((it * 7919 + n * 104729) % (2**31))
            scale = 0.003 + 0.003 * min(no_improve // 35, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
    return best_f, best_obj


def compute_smoothed_gradient(f, T, n):
    """Compute gradient of T*log(sum(exp(conv/T))) w.r.t. f."""
    conv = fast_convolve(f)
    scaled = conv / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = w / ws
    fft_size = 1 << (3 * n).bit_length()
    A = np.fft.rfft(f, fft_size)
    P = np.fft.rfft(p, fft_size)
    cross = np.fft.irfft(np.conj(A) * P, fft_size)
    return 2.0 * cross[:n], conv


def momentum_grad_descent(f0, time_limit, T_init=0.004, T_min=0.00003, verbose=False):
    """Gradient descent with momentum and adaptive temperature."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    T = T_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = 0.005
    momentum = 0.7
    v = np.zeros(n)
    x_prev = f.copy()
    top_k_for_temp = max(5, n // 40)
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        grad, _ = compute_smoothed_gradient(f, T, n)
        grad = grad - np.mean(grad)
        v = momentum * v + (1 - momentum) * grad
        f_new = project_simplex(f - lr * v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            # Adapt temperature based on peak spread
            sorted_conv = np.sort(conv_new)[::-1]
            spread = sorted_conv[0] - sorted_conv[min(top_k_for_temp, len(sorted_conv)-1)]
            T_target = max(T_min, spread * 0.4)
            T = T * 0.95 + T_target * 0.05
            T = max(T, T_min)
            if verbose and it % 200 == 0:
                print(f"    MG iter {it}: obj={best_obj:.10f}, T={T:.6f}")
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr = max(lr * 0.9, 0.0001)
            # Still update T adaptively
            sorted_conv = np.sort(conv_new)[::-1]
            spread = sorted_conv[0] - sorted_conv[min(top_k_for_temp, len(sorted_conv)-1)]
            T_target = max(T_min, spread * 0.4)
            T = T * 0.97 + T_target * 0.03
            T = max(T, T_min)
        f_prev = f
        f = f_new
        if no_improve > 250:
            rng = np.random.default_rng((it * 331 + 17) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            lr = max(lr * 2.0, 0.003)
            T = max(T_init * 0.4, T_min)
            no_improve = 0
    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.001, top_k_frac=0.05, temp=0.00003):
    """Subgradient polish with weighted top-k peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    top_k = max(5, int(n * top_k_frac))
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        f = project_simplex(f - lr * sg)
        conv = fast_convolve(f)
        co = conv.max()
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        if it % 100 == 0:
            lr *= 0.995
            temp *= 1.01
        if no_improve > 300:
            rng = np.random.default_rng((it * 733 + 41) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            lr = 0.0006
            no_improve = 0
    return best_f, best_obj


def coord_polish(f0, time_limit, verbose=False):
    """Coordinate descent: move mass from high-peak-contribution to low-peak-contribution."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    step_size = 0.002
    top_k = max(5, n // 60)
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        damage = np.zeros(n)
        for k in peaks:
            weight = conv[k]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            damage[idx] += weight * f[k - idx]
        dm = damage.max()
        if dm < 1e-15:
            break
        damage = damage / dm
        pos_mask = f > 1e-8
        if pos_mask.sum() < 2:
            break
        mean_d = np.mean(damage[pos_mask])
        delta = np.zeros(n)
        delta[pos_mask] = (damage[pos_mask] - mean_d) * step_size
        delta -= delta.sum()
        f_try = f + delta
        f_try = np.maximum(f_try, 0.0)
        st = f_try.sum()
        if st < 1e-15:
            break
        f_try = f_try / st
        conv_try = fast_convolve(f_try)
        obj_try = conv_try.max()
        if obj_try < best_obj - 1e-15:
            best_obj = obj_try
            best_f = f_try.copy()
            f = f_try
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    Coord iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                step_size *= 0.85
            if no_improve > 800:
                break
        if step_size < 1e-7:
            break
    return best_f, best_obj


def generate_diverse_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    x = np.arange(n, dtype=np.float64)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c1 / c1.sum())
    h1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2)
    h2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    two_hump = h1 + h2
    inits.append(two_hump / two_hump.sum())
    half_n = n // 2
    half = rng.dirichlet(np.ones(half_n) * 0.5)
    sym = np.zeros(n)
    sym[:half_n] = half
    sym[half_n:] = half[::-1]
    inits.append(sym / sym.sum())
    # Anti-symmetric pressure: push mass away from center
    anti_center = np.abs(x - 0.5 * (n - 1)) + 0.5
    inits.append(anti_center / anti_center.sum())
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search:
    Phase 1: Quick exploration with LP-FW on parent and alternative lengths.
    Phase 2: Deep optimization with momentum GD and LP-FW alternation.
    Phase 3: Fine polishing with subgradient and coordinate descent.
    Phase 4: Final verification and micro-perturbations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
    except Exception:
        pass

    # === Phase 1: Quick Exploration (250s) ===
    print("=== Phase 1: Exploration ===")
    t_phase1 = time.monotonic()
    p1_budget = 250.0

    base_n = len(parent) if parent is not None else 384

    if parent is not None:
        t_alloc = min(100.0, p1_budget - 50.0)
        print(f"  FW-LP on parent (n={base_n}), t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(parent, time_limit=t_alloc)
        try_seq(a_opt, "parent_fw")

    # Try alternative lengths
    alt_lengths = []
    for d in [-8, 8, -16, 16, -32, 32, -48, 48]:
        cn = base_n + d
        if 160 <= cn <= 800 and cn != base_n:
            alt_lengths.append(cn)
    rng.shuffle(alt_lengths)

    for n_try in alt_lengths[:3]:
        rem = p1_budget - (time.monotonic() - t_phase1)
        if rem < 30:
            break
        inits = generate_diverse_inits(n_try, parent, rng)
        best_init = None
        best_init_obj = float('inf')
        for f_init in inits[:5]:
            c = fast_convolve(f_init)
            o = c.max()
            if o < best_init_obj:
                best_init_obj = o
                best_init = f_init
        if best_init is not None:
            t_alloc = min(35.0, rem - 15.0)
            if t_alloc > 15:
                print(f"  FW-LP n={n_try}, t={t_alloc:.0f}s")
                a_opt, _ = fw_lp_optimize(best_init, time_limit=t_alloc)
                try_seq(a_opt, f"n{n_try}")

    print(f"  Phase 1 done: best={best_score:.6f}")

    # === Phase 2: Deep Optimization (950s) ===
    print("=== Phase 2: Deep Optimization ===")
    t_phase2 = time.monotonic()
    p2_budget = 950.0

    if best_seq is not None:
        f_work = best_seq / best_seq.sum()
        n_best = len(best_seq)

        # Alternate: momentum GD -> LP-FW -> momentum GD -> ...
        n_cycles = 5
        for cycle in range(n_cycles):
            rem = p2_budget - (time.monotonic() - t_phase2)
            if rem < 50:
                break

            # Momentum gradient descent
            t_mg = min(120.0, rem * 0.4)
            if t_mg > 30:
                T_start = 0.004 / (1.0 + 0.3 * cycle)
                print(f"  Cycle {cycle}: Momentum GD (t={t_mg:.0f}s, T={T_start:.5f})")
                a_opt, _ = momentum_grad_descent(f_work, time_limit=t_mg, T_init=T_start)
                try_seq(a_opt, f"mg_{cycle}")
                if best_seq is not None:
                    f_work = best_seq / best_seq.sum()

            # LP-FW
            rem = p2_budget - (time.monotonic() - t_phase2)
            if rem > 40:
                t_fw = min(100.0, rem * 0.35)
                if t_fw > 25:
                    print(f"  Cycle {cycle}: LP-FW (t={t_fw:.0f}s)")
                    a_opt, _ = fw_lp_optimize(f_work, time_limit=t_fw)
                    try_seq(a_opt, f"fw_{cycle}")
                    if best_seq is not None:
                        f_work = best_seq / best_seq.sum()

            # Quick momentum GD with lower T
            rem = p2_budget - (time.monotonic() - t_phase2)
            if rem > 30:
                t_mg2 = min(60.0, rem * 0.3)
                if t_mg2 > 20:
                    T_start2 = 0.002 / (1.0 + 0.3 * cycle)
                    print(f"  Cycle {cycle}: Fine MG (t={t_mg2:.0f}s, T={T_start2:.5f})")
                    a_opt, _ = momentum_grad_descent(f_work, time_limit=t_mg2, T_init=T_start2)
                    try_seq(a_opt, f"mgf_{cycle}")
                    if best_seq is not None:
                        f_work = best_seq / best_seq.sum()

        # Try nearby lengths for best shape
        rem = p2_budget - (time.monotonic() - t_phase2)
        if rem > 50:
            for delta in [8, -8]:
                new_n = n_best + delta
                if new_n < 128 or new_n > 1024:
                    continue
                px = np.interp(np.linspace(0, 1, new_n),
                               np.linspace(0, 1, n_best), f_work)
                px = project_simplex(px)
                t_alloc = min(40.0, rem * 0.4)
                if t_alloc > 20:
                    a_opt, _ = fw_lp_optimize(px, time_limit=t_alloc)
                    try_seq(a_opt, f"len{new_n}")
                    rem = p2_budget - (time.monotonic() - t_phase2)
                    if rem < 30:
                        break

    print(f"  Phase 2 done: best={best_score:.6f}")

    # === Phase 3: Fine Polish (500s) ===
    print("=== Phase 3: Fine Polish ===")
    t_phase3 = time.monotonic()
    p3_budget = 500.0

    if best_seq is not None:
        f_work = best_seq / best_seq.sum()

        # Coordinate descent
        rem = p3_budget - (time.monotonic() - t_phase3)
        if rem > 50:
            t_cd = min(150.0, rem * 0.3)
            print(f"  Coord descent (t={t_cd:.0f}s)")
            a_opt, _ = coord_polish(f_work, time_limit=t_cd)
            try_seq(a_opt, "coord")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

        # Subgradient rounds with decreasing parameters
        sg_configs = [
            {"lr": 0.0008, "top_k_frac": 0.04, "temp": 0.00002},
            {"lr": 0.0005, "top_k_frac": 0.03, "temp": 0.00001},
            {"lr": 0.0003, "top_k_frac": 0.02, "temp": 0.000005},
            {"lr": 0.0002, "top_k_frac": 0.015, "temp": 0.000002},
        ]
        for i, cfg in enumerate(sg_configs):
            rem = p3_budget - (time.monotonic() - t_phase3)
            if rem < 35:
                break
            t_alloc = min(70.0, rem * 0.25)
            if t_alloc > 20:
                print(f"  SG polish {i} (t={t_alloc:.0f}s)")
                a_opt, _ = sg_polish(f_work, time_limit=t_alloc, **cfg)
                try_seq(a_opt, f"sg_{i}")
                if best_seq is not None:
                    f_work = best_seq / best_seq.sum()

        # Final very fine momentum GD
        rem = p3_budget - (time.monotonic() - t_phase3)
        if rem > 40:
            t_fine = min(100.0, rem * 0.4)
            if t_fine > 25:
                T_init_fine = 0.0005
                print(f"  Fine momentum GD (t={t_fine:.0f}s, T={T_init_fine:.6f})")
                a_opt, _ = momentum_grad_descent(f_work, time_limit=t_fine,
                                                  T_init=T_init_fine, T_min=0.00001)
                try_seq(a_opt, "fine_mg")
                if best_seq is not None:
                    f_work = best_seq / best_seq.sum()

        # One more LP-FW with tight constraint
        rem = p3_budget - (time.monotonic() - t_phase3)
        if rem > 35:
            t_final_fw = min(60.0, rem * 0.5)
            if t_final_fw > 20:
                print(f"  Final LP-FW (t={t_final_fw:.0f}s)")
                a_opt, _ = fw_lp_optimize(f_work, time_limit=t_final_fw)
                try_seq(a_opt, "final_fw")

    print(f"  Phase 3 done: best={best_score:.6f}")

    # === Phase 4: Final micro-optimizations ===
    print("=== Phase 4: Final Micro-Optimizations ===")

    if best_seq is not None:
        f_work = best_seq / best_seq.sum()

        # Micro-perturbations with quick polish
        for _ in range(5):
            if time.monotonic() > deadline - 15:
                break
            noise = rng.standard_normal(len(f_work)) * 0.0004
            f_pert = project_simplex(f_work + noise)
            a_opt, _ = sg_polish(f_pert, time_limit=8, lr=0.0003,
                                  top_k_frac=0.02, temp=0.00001)
            try_seq(a_opt, "micro_pert")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

        # Tiny coordinate polish
        if time.monotonic() < deadline - 20:
            a_opt, _ = coord_polish(f_work, time_limit=15)
            try_seq(a_opt, "micro_coord")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
