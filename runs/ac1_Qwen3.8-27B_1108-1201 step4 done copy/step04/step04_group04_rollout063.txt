"""
Optimized step function minimizer using adaptive smoothed gradient descent
with LP direction injection and water-filling peak reduction.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_smoothed_grad(conv, a, T):
    n = len(a)
    s = conv / T
    s -= s.max()
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = w / ws
    fft_size = 1 << (3 * n).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    cross = np.fft.irfft(np.conj(A) * P, fft_size)
    return 2.0 * cross[:n]

def solve_lp_direction(f, M, max_time=5.0):
    n = len(f)
    # LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0
    # For speed, only include constraints where f*g is near M
    c = -np.ones(n)
    # Build constraint matrix sparsely
    A_ub_rows = []
    b_ub_vals = []
    for k in range(2 * n - 1):
        # (f * g)[k] = sum_j f[j] * g[k-j] (valid j)
        # Coefficient of g[j] in row k is f[k-j] if valid
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        # Check if this constraint is likely active
        # (f * f)[k] as a proxy
        pass  # Include all constraints for correctness
        row = np.zeros(n)
        for j in range(js, je + 1):
            row[j] = f[k - j]
        A_ub_rows.append(row)
        b_ub_vals.append(M)
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    bounds = [(0, None)] * n
    
    t_start = time.monotonic()
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': max_time, 'presolve': True})
    
    if res.success and res.x is not None:
        g = res.x
        gs = g.sum()
        if gs > 1e-15:
            return g / gs
    return None

def subgradient_polish(f0, time_limit, verbose=True):

    f = np.maximum(f0, 0.0).copy()

    it = 0
    lr = 0.001
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)

        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        
        # Weight by excess above mean
        mean_pv = pv.mean()
        excess = np.maximum(pv - mean_pv, 0)
        total_excess = excess.sum()
        if total_excess < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = excess / total_excess
        
        # Subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        # Line search
        best_alpha = 0.0
        best_val = best_obj
        for alpha in [lr * 0.5, lr, lr * 2, lr * 4]:
            ft = project_simplex(f - alpha * sg)
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            f = project_simplex(f - best_alpha * sg)
            conv = fast_convolve(f)
            obj = conv.max()
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve % 50 == 0 and no_improve > 0:
            lr *= 0.8
            if lr < 1e-7:
                lr = 0.001
                rng = np.random.default_rng(it * 13 + 5)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                no_improve = 0
        
        if verbose and it % 200 == 0:
            print(f"    SG iter={it}, obj={best_obj:.10f}, lr={lr:.6f}")
    
    return best_f, best_obj

"""

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_grad(conv, a, T):
    """Gradient of T*log(sum(exp(conv/T))) w.r.t. a via FFT cross-correlation."""
    n = len(a)
    s = conv / T
    s -= s.max()
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = w / ws
    fft_size = 1 << (3 * n).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    cross = np.fft.irfft(np.conj(A) * P, fft_size)
    return 2.0 * cross[:n]


def compute_flatten_grad(conv, a, target):

    n = len(a)

    A = np.fft.rfft(a, fft_size)

def solve_lp_direction(f, M, max_time=5.0):
    """Solve LP: max 1^T g s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': max_time, 'presolve': True})
        if res.success and res.x is not None:
            g = res.x
            gs = g.sum()
            if gs > 1e-15:
                return g / gs
    except Exception:
        pass
    return None

def subgradient_polish(f0, time_limit, lr_init=0.0008, top_k_frac=0.04, verbose=False):
    """Subgradient polish with weighted top-k peaks and bisection line search."""
    f = np.maximum(f0, 0.0).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = lr_init
    top_k = max(3, int(n * top_k_frac))

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)

        # Weighted subgradient from top peaks
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        # Weight by excess above median
        med = np.median(pv)
        excess = np.maximum(pv - med, 1e-15)
        w = excess / excess.sum()

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        # Bisection line search
        best_alpha = 0.0
        best_val = best_obj
        lo, hi = 0.0, lr * 10
        for _ in range(8):
            mid = (lo + hi) / 2.0
            if mid < 1e-12:
                break
            ft = project_simplex(f - mid * sg)
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = mid
                lo = mid
            else:
                hi = mid

        if best_alpha > 1e-12:
            f = project_simplex(f - best_alpha * sg)
            conv = fast_convolve(f)
            obj = conv.max()
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Decay learning rate
        if no_improve > 0:
            lr = max(lr * 0.98, lr_init * 0.01)

        # Restart
        if no_improve > 200:
            rng = np.random.default_rng(it * 23 + 1)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            lr = lr_init * 0.5
            no_improve = 0

        if verbose and it % 300 == 0:
            print(f"    SG iter={it}, obj={best_obj:.10f}, lr={lr:.6f}")

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for exploration."""
    inits = []
    if parent is not None:
        inits.append(parent.copy())
    inits.append(np.ones(n) / n)

    x = np.arange(n, dtype=np.float64)

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / n)) + 0.05
    inits.append(rc / rc.sum())

    # Edge-concentrated (mild)
    edge = 1.0 / (np.minimum(x, n - 1 - x) + n * 0.1 + 1.0)
    inits.append(edge / edge.sum())

    # Two-lobe
    l1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.12 * n)) ** 2)
    l2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.12 * n)) ** 2)
    inits.append((l1 + l2 + 0.03) / (l1 + l2 + 0.03).sum())

    # Mirror-symmetric Dirichlet
    half_n = n // 2
    half = rng.dirichlet(np.ones(half_n) * 0.8)
    sym = np.zeros(n)
    sym[:half_n] = half
    sym[half_n:] = half[::-1]
    if n % 2 == 1:
        sym[half_n] = half[-1]
    sym = np.maximum(sym, 0)
    if sym.sum() > 1e-10:
        inits.append(sym / sym.sum())

    # Parent + noise
    if parent is not None:
        for sigma in [0.05, 0.1, 0.2]:
            noisy = project_simplex(parent + rng.standard_normal(n) * sigma)
            inits.append(noisy)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Phase 1: Quick multi-init exploration.
    Phase 2: Deep optimization with adaptive smoothed GD + LP injection.
    Phase 3: Fine subgradient polishing with decreasing parameters.
    Phase 4: Final tricks (mirror, perturbation, length adjustment).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(368) / 368
        print("Using default parent: uniform n=368")

    n_base = len(parent)
    print(f"Base length: {n_base}, parent score: {float(evaluate_sequence(parent.tolist())):.6f}")

    # === Phase 1: Quick Exploration (200s) ===
    print("\n=== Phase 1: Quick Exploration ===")
    t1_start = time.monotonic()
    phase1_budget = 200.0

    # Try parent
    try_seq(parent, "parent")

    # Generate and evaluate inits
    inits = generate_inits(n_base, parent, rng)
    init_objs = []
    for init in inits:
        conv = fast_convolve(init)
        init_objs.append(conv.max())
    init_order = np.argsort(init_objs)

    # Optimize top 3 diverse inits
    used = set()
    count = 0
    for idx in init_order:
        if count >= 3:
            break
        if idx in used:
            continue
        used.add(idx)
        rem = phase1_budget - (time.monotonic() - t1_start)
        if rem < 35:
            break
        t_alloc = min(50.0, rem - 10.0)
        if t_alloc < 15:
            break
        label = f"p1_{count}"
        print(f"  Phase 1: init idx={idx} obj={init_objs[idx]:.8f}, t={t_alloc:.0f}s")
        a_opt, _ = adaptive_smoothed_gd(inits[idx], time_limit=t_alloc,
                                         T_init=0.005 * init_objs[idx],
                                         T_min=0.0001, lr_init=0.008,
                                         use_flatten=True, flatten_weight=0.05,
                                         lp_every=300, verbose=False)
        try_seq(a_opt, label)
        count += 1

    print(f"  Phase 1 done: best={best_score:.6f}")

    # === Phase 2: Deep Optimization (1100s) ===
    print(f"\n=== Phase 2: Deep Optimization ===")
    t2_start = time.monotonic()
    phase2_budget = 1100.0

    if best_seq is not None:
        f_work = best_seq / best_seq.sum()
        n_best = len(f_work)

        # Multiple rounds of alternating methods
        n_rounds = 10
        t_per_round = phase2_budget / n_rounds

        for r in range(n_rounds):
            rem = phase2_budget - (time.monotonic() - t2_start)
            if rem < 20:
                break

            # A: Smoothed GD (55% of round)
            t_a = min(t_per_round * 0.55, rem * 0.5)
            if t_a > 15:
                T_init = 0.002 * fast_convolve(f_work).max()
                print(f"  R{r}: SmoothedGD t={t_a:.0f}s")
                a_opt, obj_opt = adaptive_smoothed_gd(
                    f_work, time_limit=t_a,
                    T_init=T_init, T_min=0.00003,
                    lr_init=0.005, use_flatten=True,
                    flatten_weight=0.08 + 0.04 * (r % 3),
                    lp_every=350 + 50 * r,
                    verbose=False)
                try_seq(a_opt, f"p2_gd_{r}")
                if best_seq is not None:
                    f_work = best_seq / best_seq.sum()

            # B: Subgradient polish (30% of round)
            rem = phase2_budget - (time.monotonic() - t2_start)
            t_b = min(t_per_round * 0.30, rem * 0.4)
            if t_b > 10:
                lr_sg = max(0.0003, 0.001 * (1.0 - 0.05 * r))
                tk = max(0.02, 0.05 - 0.003 * r)
                print(f"  R{r}: SubGrad t={t_b:.0f}s lr={lr_sg:.5f} tk={tk:.3f}")
                a_opt, _ = subgradient_polish(f_work, time_limit=t_b,
                                              lr_init=lr_sg, top_k_frac=tk,
                                              verbose=False)
                try_seq(a_opt, f"p2_sg_{r}")
                if best_seq is not None:
                    f_work = best_seq / best_seq.sum()

            # C: Length adjustment (15% of round, every other round)
            rem = phase2_budget - (time.monotonic() - t2_start)
            t_c = min(t_per_round * 0.15, rem * 0.3)
            if t_c > 12 and r % 2 == 1:
                deltas = [8, -8, 16, -16, 32]
                delta = deltas[r // 2 % len(deltas)]
                new_n = n_best + delta
                if 128 <= new_n <= 1024:
                    px = np.interp(np.linspace(0, 1, new_n),
                                   np.linspace(0, 1, n_best), f_work)
                    px = project_simplex(px)
                    print(f"  R{r}: Length n={new_n} t={t_c:.0f}s")
                    a_opt, _ = adaptive_smoothed_gd(
                        px, time_limit=t_c,
                        T_init=0.003 * fast_convolve(px).max(),
                        T_min=0.0001, lr_init=0.006,
                        use_flatten=True, flatten_weight=0.05,
                        lp_every=500, verbose=False)
                    try_seq(a_opt, f"p2_len{new_n}")
                    if best_seq is not None:
                        f_work = best_seq / best_seq.sum()
                        n_best = len(f_work)

    print(f"  Phase 2 done: best={best_score:.6f}")

    # === Phase 3: Fine Polish (400s) ===
    print(f"\n=== Phase 3: Fine Polish ===")
    t3_start = time.monotonic()
    phase3_budget = 400.0

    if best_seq is not None:
        f_work = best_seq / best_seq.sum()

        # Progressive subgradient polishing
        configs = [
            {"lr": 0.0004, "tk": 0.02},
            {"lr": 0.0002, "tk": 0.015},
            {"lr": 0.0001, "tk": 0.01},
            {"lr": 0.00005, "tk": 0.008},
        ]

        for i, cfg in enumerate(configs):
            rem = phase3_budget - (time.monotonic() - t3_start)
            if rem < 25:
                break
            t_alloc = min(70.0, rem - 10.0)
            if t_alloc < 10:
                break
            print(f"  SG polish {i}: lr={cfg['lr']}, tk={cfg['tk']}, t={t_alloc:.0f}s")
            a_opt, _ = subgradient_polish(f_work, time_limit=t_alloc,
                                          lr_init=cfg["lr"], top_k_frac=cfg["tk"],
                                          verbose=False)
            try_seq(a_opt, f"p3_sg{i}")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

        # Ultra-low temperature smoothed GD
        rem = phase3_budget - (time.monotonic() - t3_start)
        if rem > 40:
            t_alloc = min(100.0, rem - 10.0)
            T_init = 0.0005 * fast_convolve(f_work).max()
            print(f"  Ultra-fine GD: T0={T_init:.7f}, t={t_alloc:.0f}s")
            a_opt, _ = adaptive_smoothed_gd(
                f_work, time_limit=t_alloc,
                T_init=T_init, T_min=0.00001,
                lr_init=0.002, use_flatten=True,
                flatten_weight=0.15, lp_every=800,
                verbose=False)
            try_seq(a_opt, "p3_ultra")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

        # Final LP-driven polish
        rem = phase3_budget - (time.monotonic() - t3_start)
        if rem > 25:
            t_alloc = min(50.0, rem - 5.0)
            print(f"  Final LP-GD: t={t_alloc:.0f}s")
            a_opt, _ = adaptive_smoothed_gd(
                f_work, time_limit=t_alloc,
                T_init=0.001 * fast_convolve(f_work).max(),
                T_min=0.00001, lr_init=0.003,
                use_flatten=False, lp_every=100,
                verbose=False)
            try_seq(a_opt, "p3_lp")

    print(f"  Phase 3 done: best={best_score:.6f}")

    # === Phase 4: Final Tricks (60s) ===
    print(f"\n=== Phase 4: Final Tricks ===")

    if best_seq is not None:
        f_work = best_seq / best_seq.sum()
        n_work = len(f_work)

        # Mirror symmetry
        mirrored = (f_work + f_work[::-1]) / 2.0
        mirrored = project_simplex(mirrored)
        if np.max(np.abs(mirrored - f_work)) > 1e-7:
            t_rem = min(25.0, deadline - time.monotonic() - 10)
            if t_rem > 8:
                a_opt, _ = subgradient_polish(mirrored, time_limit=t_rem,
                                              lr_init=0.0003, top_k_frac=0.02,
                                              verbose=False)
                try_seq(a_opt, "p4_mirror")

        # Tiny perturbations with quick polish
        for p in range(6):
            if time.monotonic() > deadline - 10:
                break
            sigma = 0.0002 * (p + 1)
            noise = rng.standard_normal(n_work) * sigma
            f_pert = project_simplex(f_work + noise)
            t_rem = min(12.0, deadline - time.monotonic() - 5)
            if t_rem > 4:
                a_opt, _ = subgradient_polish(f_pert, time_limit=t_rem,
                                              lr_init=0.0002, top_k_frac=0.015,
                                              verbose=False)
                try_seq(a_opt, f"p4_pert{p}")

        # One last ultra-short GD burst
        if time.monotonic() < deadline - 20:
            t_rem = min(20.0, deadline - time.monotonic() - 5)
            T_init = 0.0003 * fast_convolve(f_work).max()
            a_opt, _ = adaptive_smoothed_gd(
                f_work, time_limit=t_rem,
                T_init=T_init, T_min=0.00001,
                lr_init=0.001, use_flatten=True,
                flatten_weight=0.2, lp_every=9999,
                verbose=False)
            try_seq(a_opt, "p4_burst")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(368) / 368
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        else:
            best_seq = np.ones(368) / 368
            best_score = float(evaluate_sequence(best_seq.tolist()))
    except Exception:
        best_seq = np.ones(368) / 368
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
