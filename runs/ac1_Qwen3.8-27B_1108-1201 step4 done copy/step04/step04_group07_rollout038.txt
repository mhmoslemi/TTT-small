def coordinate_descent(f0, time_limit):
    """Coordinate descent targeting the peak."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    f = f / s
    n = len(f)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        peak_k = int(np.argmax(conv))
        
        # The peak is at position peak_k. The gradient w.r.t. a[j] is 2*a[peak_k - j]
        # We want to redistribute mass to reduce the peak.
        # Find coordinates where a[j] * a[peak_k - j] is large (these contribute most to the peak)
        
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        js_count = je - js + 1
        
        if js_count < 2:
            break
        
        # Pick a pair (j, peak_k - j) and try to transfer mass
        # Try several random pairs
        for _ in range(200):
            if time.monotonic() - t0 > time_limit:
                break
            j1 = js + rng.integers(js_count)
            j2 = js + rng.integers(js_count)
            if j1 == j2 or j1 + j2 != peak_k:
                continue
            # j1 and j2 are a pair that contributes to the peak
            i1 = j1
            i2 = j2
            if i1 >= n or i2 >= n:
                continue
            a1 = f[i1]
            a2 = f[i2]
            if a1 < 1e-10 or a2 < 1e-10:
                continue
            
            # Try to reduce the product a1*a2 by transferring mass
            # Move delta from the larger to the smaller
            if a1 >= a2:
                for delta in [0.01 * a1, 0.05 * a1, 0.1 * a1, 0.2 * a1]:
                    f_try = f.copy()
                    f_try[i1] -= delta
                    f_try[i2] += delta
                    # Check non-negativity
                    if f_try[i1] < 0:
                        continue
                    s_try = f_try.sum()
                    f_try = f_try / s_try
                    obj = fast_convolve(f_try).max()
                    if obj < best_obj - 1e-15:
                        best_obj = obj
                        best_f = f_try.copy()
                        f = f_try.copy()
                        break
            else:
                for delta in [0.01 * a2, 0.05 * a2, 0.1 * a2, 0.2 * a2]:
                    f_try = f.copy()
                    f_try[i2] -= delta
                    f_try[i1] += delta
                    if f_try[i2] < 0:
                        continue
                    s_try = f_try.sum()
                    f_try = f_try / s_try
                    obj = fast_convolve(f_try).max()
                    if obj < best_obj - 1e-15:
                        best_obj = obj
                        best_f = f_try.copy()
                        f = f_try.copy()
                        break
    
    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an adaptive LP Frank-Wolfe method
with multiple exploration strategies: reduced/full LP oracles, subgradient polishing,
spectral balancing, coordinate descent, and multi-length search.
Key innovations over baseline:
- Adaptive LP constraint set (grows k as convergence slows)
- Spectral-domain peak flattening as an exploration operator
- Multi-track parallel optimization with perturbed restarts
- Systematic length exploration around the best found length

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f):
    """Build B matrix where (B*g)[k] = (f*g)[k]."""
    n = len(f)
    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        B[j:j + n, j] = f
    return B


def solve_reduced_lp(f, conv_f, top_k, B=None):
    """Solve reduced linearized LP using top_k peak constraints."""
    n = len(f)
    if B is None:
        B = build_conv_matrix(f)
    if top_k >= len(conv_f):
        peaks = np.arange(len(conv_f))
    else:
        peaks = np.argpartition(conv_f, -top_k)[-top_k:]
    peaks = np.sort(peaks)

    c = np.zeros(n + 1)
    c[0] = 1.0
    A_ub = np.zeros((len(peaks), n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = 2.0 * B[peaks, :]
    b_ub = -conv_f[peaks]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]
    return None


def solve_full_lp(f, conv_f, B=None):
    """Solve full linearized LP with all constraints."""
    n = len(f)
    if B is None:
        B = build_conv_matrix(f)
    c = np.zeros(n + 1)
    c[0] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = 2.0 * B
    b_ub = -conv_f
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]
    return None


def solve_maxsum_lp(f, M):
    """Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None and np.any(res.x > 0):
        return res.x
    return None

                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            M = fast_convolve(f).max()
            m_exp = 1.0
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 4.0)

        if no_improve > 50:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            M = fast_convolve(f).max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def subgrad_optimize(f0, time_limit, lr=0.003, mom=0.93, top_k=40, temp=0.0001):
    """Smoothed subgradient descent with softmax-weighted multi-peak targeting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9998
        cur_temp *= 1.003

        if no_improve > 250:
            rng = np.random.default_rng((it * 31 + 17) % 100000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0003)
            no_improve = 0

    return best_f, best_obj


def spectral_smooth(f0, time_limit, gamma=0.05):
    """
    Spectral-domain peak flattening: iterate between modifying |FFT(f)|^2
    to be flatter and projecting back to the valid simplex.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        F = np.fft.rfft(f, fft_size)
        F_mag = np.abs(F)
        F_ang = np.angle(F)

        power = F_mag ** 2
        avg_power = power.sum() / len(power)
        target_mag = np.sqrt(avg_power)

        # Reduce peaks in magnitude spectrum
        mask = F_mag > target_mag
        if mask.any():
            F_mag[mask] = F_mag[mask] * (target_mag / F_mag[mask]) ** gamma
            # Boost valleys slightly
            mask_low = F_mag < target_mag * 0.8
            if mask_low.any():
                F_mag[mask_low] = F_mag[mask_low] + 0.01 * (target_mag - F_mag[mask_low])

        F_new = F_mag * np.exp(1j * F_ang)
        f_new = np.fft.ifft(F_new, fft_size)[:n].real
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new = f_new / s

        f = f_new
        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()

    return best_f, best_obj


def coordinate_descent(f0, time_limit):
    """Coordinate descent targeting the auto-convolution peak."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        peak_k = int(np.argmax(conv))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        js_count = je - js + 1

        if js_count < 2:
            break

        improved = False
        for _ in range(300):
            if time.monotonic() - t0 > time_limit:
                break
            j1 = js + int(rng.integers(js_count))
            j2 = peak_k - j1
            if j2 < js or j2 > je or j1 == j2:
                continue
            a1 = f[j1]
            a2 = f[j2]
            if a1 < 1e-10 or a2 < 1e-10:
                continue

            if a1 >= a2:
                source, dest = j1, j2
                base = a1
            else:
                source, dest = j2, j1
                base = a2

            for frac in [0.05, 0.1, 0.2, 0.35]:
                delta = frac * base
                if f[source] - delta < 0:
                    continue
                f_try = f.copy()
                f_try[source] -= delta
                f_try[dest] += delta
                s_try = f_try.sum()
                f_try = f_try / s_try
                obj = fast_convolve(f_try).max()
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_f = f_try.copy()
                    f = f_try.copy()
                    improved = True
                    break

        if not improved:
            # Try random pair swaps
            for _ in range(100):
                if time.monotonic() - t0 > time_limit:
                    break
                i1 = int(rng.integers(n))
                i2 = int(rng.integers(n))
                if i1 == i2:
                    continue
                a1 = f[i1]
                a2 = f[i2]
                if a1 < 1e-10 or a2 < 1e-10:
                    continue
                direction = 1 if a1 > a2 else -1
                for frac in [0.02, 0.05, 0.1]:
                    delta = frac * max(a1, a2)
                    src = i1 if direction > 0 else i2
                    dst = i2 if direction > 0 else i1
                    if f[src] - delta < 0:
                        continue
                    f_try = f.copy()
                    f_try[src] -= delta
                    f_try[dst] += delta
                    f_try = f_try / f_try.sum()
                    obj = fast_convolve(f_try).max()
                    if obj < best_obj - 1e-15:
                        best_obj = obj
                        best_f = f_try.copy()
                        f = f_try.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved and rng.random() < 0.1:
            # Occasionally do a small random perturbation
            f = project_simplex(f + rng.standard_normal(n) * 0.002)

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search combining LP Frank-Wolfe (reduced/full),
    max-sum LP FW, subgradient, spectral smoothing, coordinate descent,
    and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={ps:.10f}")
            best_seq = parent.copy()
            best_score = ps
    except Exception:
        pass

    # Phase 1: Initialization exploration
    print("=== Phase 1: Initialization ===")
    if parent is None:
        n_base = 384
        best_seq = np.ones(n_base) / n_base
        best_score = float(evaluate_sequence(best_seq.tolist()))
    else:
        n_base = len(parent)

    # Try alternative initializations at parent length
    x = np.arange(n_base, dtype=np.float64)
    init_list = []
    init_list.append(("uniform", np.ones(n_base) / n_base))
    init_list.append(("dir03", rng.dirichlet(np.ones(n_base) * 0.3)))
    init_list.append(("dir10", rng.dirichlet(np.ones(n_base) * 1.0)))

    for sf in [0.2, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * (n_base - 1)) / (sf * n_base)) ** 2)
        init_list.append((f"gauss{int(sf*100)}", g / g.sum()))

    cos_v = np.abs(np.cos(np.pi * x / n_base)) + 0.05
    init_list.append(("cosine", cos_v / cos_v.sum()))

    beta_seq = (x + 1) ** 2 * (n_base - x) ** 2
    init_list.append(("beta", beta_seq / beta_seq.sum()))

    for label, ini in init_list:
        if time.monotonic() > deadline - 400:
            break
        ini = np.maximum(ini, 0.0)
        sc = float(evaluate_sequence(ini.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = ini.copy()
            print(f"  New best init [{label}]: {sc:.10f}")

    # Try different lengths
    for n_try in [256, 320, 448, 512, 640]:
        if n_try == n_base or time.monotonic() > deadline - 350:
            continue
        if parent is not None:
            px = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, n_base), parent)
        else:
            px = np.ones(n_try) / n_try
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            px = px / ps
            sc = float(evaluate_sequence(px.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = px.copy()
                print(f"  New best len{n_try}: {sc:.10f}")

    print(f"  Phase 1 done: best={best_score:.10f}, n={len(best_seq)}")

    # Phase 2: Main LP Frank-Wolfe (reduced, adaptive k)
    print("\n=== Phase 2: LP Frank-Wolfe (adaptive) ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    t_p2 = min(remaining * 0.35, 500)
    print(f"  FW reduced k=25, n={n_best}, budget={t_p2:.0f}s")
    a_opt, _ = fw_optimize(best_seq, time_limit=t_p2, reduced_k=25, use_full=False, adaptive_k=True)
    a_opt = np.maximum(a_opt, 0.0)
    sc = float(evaluate_sequence(a_opt.tolist()))
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_opt.copy()
        print(f"  Improved: {sc:.10f}")

    # Second FW round with smaller k (faster iterations)
    rem = deadline - time.monotonic()
    if rem > 80:
        t_p2b = min(rem * 0.15, 150)
        print(f"  FW reduced k=12, budget={t_p2b:.0f}s")
        a_opt, _ = fw_optimize(best_seq, time_limit=t_p2b, reduced_k=12, use_full=False, adaptive_k=True)
        a_opt = np.maximum(a_opt, 0.0)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  Improved: {sc:.10f}")

    # Full LP for larger jumps
    rem = deadline - time.monotonic()
    if rem > 100:
        t_p2c = min(rem * 0.20, 200)
        print(f"  FW full LP, budget={t_p2c:.0f}s")
        a_opt, _ = fw_optimize(best_seq, time_limit=t_p2c, reduced_k=30, use_full=True, adaptive_k=False)
        a_opt = np.maximum(a_opt, 0.0)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  Improved: {sc:.10f}")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # Phase 3: Max-sum LP Frank-Wolfe
    print("\n=== Phase 3: Max-sum LP FW ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t_p3 = min(rem * 0.18, 180)
        print(f"  Max-sum FW, budget={t_p3:.0f}s")
        a_opt, _ = maxsum_fw_optimize(best_seq, time_limit=t_p3)
        a_opt = np.maximum(a_opt, 0.0)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  Improved: {sc:.10f}")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # Phase 4: Subgradient polishing
    print("\n=== Phase 4: Subgradient ===")
    rem = deadline - time.monotonic()
    if rem > 50:
        n_best = len(best_seq)
        t_p4 = min(rem * 0.18, 180)
        print(f"  SG coarse, budget={t_p4 * 0.5:.0f}s")
        a_opt, _ = subgrad_optimize(best_seq, time_limit=t_p4 * 0.5,
                                     lr=0.002, mom=0.94,
                                     top_k=max(10, n_best // 8), temp=0.00005)
        a_opt = np.maximum(a_opt, 0.0)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  Improved: {sc:.10f}")

        rem = deadline - time.monotonic()
        if rem > 30:
            t_p4b = min(rem * 0.12, 100)
            print(f"  SG fine, budget={t_p4b:.0f}s")
            a_opt, _ = subgrad_optimize(best_seq, time_limit=t_p4b,
                                         lr=0.0008, mom=0.96,
                                         top_k=max(20, n_best // 4), temp=0.00002)
            a_opt = np.maximum(a_opt, 0.0)
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  Improved: {sc:.10f}")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # Phase 5: Spectral smoothing
    print("\n=== Phase 5: Spectral Smoothing ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t_p5 = min(rem * 0.12, 100)
        print(f"  Spectral, budget={t_p5:.0f}s")
        a_opt, _ = spectral_smooth(best_seq, time_limit=t_p5, gamma=0.03)
        a_opt = np.maximum(a_opt, 0.0)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  Improved: {sc:.10f}")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # Phase 6: Coordinate descent
    print("\n=== Phase 6: Coordinate Descent ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_p6 = min(rem * 0.12, 100)
        print(f"  Coord descent, budget={t_p6:.0f}s")
        a_opt, _ = coordinate_descent(best_seq, time_limit=t_p6)
        a_opt = np.maximum(a_opt, 0.0)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  Improved: {sc:.10f}")

    print(f"  Phase 6 done: best={best_score:.10f}")

    # Phase 7: Final fine-tuning with perturbed restarts
    print("\n=== Phase 7: Final Polish ===")
    rem = deadline - time.monotonic()
    n_best = len(best_seq)

    for attempt in range(6):
        if rem < 12:
            break
        t_each = min(rem / max(1, 6 - attempt), 35)
        noise_scale = 0.001 * (attempt + 1)
        f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise_scale)
        a_opt, _ = fw_optimize(f_pert, time_limit=t_each * 0.8,
                               reduced_k=20, use_full=False, adaptive_k=True)
        a_opt = np.maximum(a_opt, 0.0)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  Restart {attempt}: {sc:.10f}")
        rem = deadline - time.monotonic()

    # Try nearby lengths
    rem = deadline - time.monotonic()
    if rem > 15:
        for n_try in [n_best - 8, n_best + 8, n_best - 16, n_best + 16]:
            if n_try < 64 or rem < 8:
                continue
            px = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, n_best), best_seq)
            px = np.maximum(px, 0)
            ps = px.sum()
            if ps > 1e-10:
                px = px / ps
                a_opt, _ = fw_optimize(px, time_limit=min(rem * 0.3, 20),
                                       reduced_k=20, use_full=False, adaptive_k=False)
                a_opt = np.maximum(a_opt, 0.0)
                sc = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_opt.copy()
                    print(f"  len{n_try}: {sc:.10f}")
                rem = deadline - time.monotonic()

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK: {best_score:.10f}")

    return [float(x) for x in best_seq]
