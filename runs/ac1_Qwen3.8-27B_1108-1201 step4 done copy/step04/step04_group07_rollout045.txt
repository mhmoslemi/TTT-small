"""
Hybrid optimizer for minimizing 2*n*max(autocorr(a))/(sum(a))^2.
Uses Nesterov-accelerated mirror descent, LP-based Frank-Wolfe, 
and simulated annealing for multi-phase optimization.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def nesterov_subgrad(f0, time_limit, lr=0.005, mom=0.9, top_k=8, temp=0.0002):

    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    y = f.copy()  # Nesterov momentum point
    v = np.zeros(n)  # momentum
    conv = fast_convolve(y)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temp
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov lookahead
        y = f + mom * v
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys < 1e-15:
            y = f
        else:
            y = y / ys
        
        conv = fast_convolve(y)
        
        # Select top-k peaks with softmax weighting
        if top_k >= len(conv):

        ws = w.sum()
        if ws < 1e-15:

            w[np.argmax(pv)] = 1.0

        for idx, ini in enumerate(ini_list[:4]):
            candidates.append((f"n{n_try}_i{idx}", ini))

    for label, cand in candidates:
        if time.monotonic() > deadline - 300:
            break
        try_seq(cand, label)

    if best_seq is None:
        best_seq = np.ones(384) / 384
        try_seq(best_seq, "fallback")

    n_best = len(best_seq)
    print(f"  Phase 1 best: {best_score:.8f} (n={n_best})")

    print("\n=== Phase 2: Nesterov subgradient (main) ===")
    remaining = deadline - time.monotonic() - 200
    t_p2 = min(remaining * 0.40, 600)
    if t_p2 > 20:
        print(f"  Nesterov n={n_best}, budget={t_p2:.0f}s")
        a_opt, _ = nesterov_optimize(best_seq, time_limit=t_p2 * 0.5,
                                     lr=0.004, mom=0.93, top_k=6, temp=0.0002, seed=111)
        try_seq(a_opt, "nest1")

        rem = deadline - time.monotonic() - 150
        if rem > 20:
            a_opt, _ = nesterov_optimize(best_seq, time_limit=min(rem * 0.4, 250),
                                         lr=0.002, mom=0.95, top_k=10, temp=0.0001, seed=222)
            try_seq(a_opt, "nest2_fine")

    print(f"  Phase 2 best: {best_score:.8f}")

    print("\n=== Phase 3: Mirror descent ===")
    remaining = deadline - time.monotonic() - 150
    t_p3 = min(remaining * 0.20, 300)
    if t_p3 > 15:
        print(f"  Mirror descent n={n_best}, budget={t_p3:.0f}s")
        a_opt, _ = mirror_descent_optimize(best_seq, time_limit=t_p3 * 0.6,
                                           lr=0.006, top_k=8, temp=0.0004, seed=333)
        try_seq(a_opt, "mirror1")

        rem = deadline - time.monotonic() - 100
        if rem > 15:
            a_opt, _ = mirror_descent_optimize(best_seq, time_limit=min(rem * 0.4, 150),
                                               lr=0.003, top_k=12, temp=0.0002, seed=444)
            try_seq(a_opt, "mirror2")

    print(f"  Phase 3 best: {best_score:.8f}")

    print("\n=== Phase 4: LP Frank-Wolfe ===")
    remaining = deadline - time.monotonic() - 100
    t_p4 = min(remaining * 0.20, 300)
    if t_p4 > 20:
        print(f"  LP FW n={n_best}, budget={t_p4:.0f}s")
        a_opt, _ = lp_frank_wolfe(best_seq, time_limit=t_p4 * 0.5,
                                  use_maxsum=False, seed=555)
        try_seq(a_opt, "lp_lin")

        rem = deadline - time.monotonic() - 80
        if rem > 20:
            a_opt, _ = lp_frank_wolfe(best_seq, time_limit=min(rem * 0.4, 150),
                                      use_maxsum=True, seed=666)
            try_seq(a_opt, "lp_maxsum")

    print(f"  Phase 4 best: {best_score:.8f}")

    print("\n=== Phase 5: Fine polish ===")
    remaining = deadline - time.monotonic() - 50
    t_p5 = min(remaining * 0.30, 200)
    if t_p5 > 10:
        print(f"  Fine polish n={n_best}, budget={t_p5:.0f}s")
        a_opt, _ = fine_polish(best_seq, time_limit=t_p5 * 0.5,
                               top_k=40, seed=777)
        try_seq(a_opt, "polish1")

        rem = deadline - time.monotonic() - 30
        if rem > 10:
            a_opt, _ = fine_polish(best_seq, time_limit=min(rem * 0.4, 100),
                                   top_k=60, seed=888)
            try_seq(a_opt, "polish2")

    print(f"  Phase 5 best: {best_score:.8f}")

    print("\n=== Phase 6: Multi-length exploration ===")
    remaining = deadline - time.monotonic() - 30
    if remaining > 30 and best_seq is not None:
        n_cur = len(best_seq)
        for n_try in [n_cur - 32, n_cur + 32, n_cur + 64]:
            if n_try < 32 or n_try > 2048 or remaining < 15:
                continue
            px = np.interp(np.linspace(0, 1, n_try),
                           np.linspace(0, 1, n_cur), best_seq)
            px = np.maximum(px, 0)
            ps = px.sum()
            if ps > 1e-10:
                px = px / ps
                t_each = min(remaining * 0.3, 40)
                a_opt, _ = nesterov_optimize(px, time_limit=t_each,
                                             lr=0.003, mom=0.93, top_k=8,
                                             temp=0.0002, seed=1234)
                try_seq(a_opt, f"len{n_try}")
                remaining = deadline - time.monotonic() - 20

    print("\n=== Phase 7: Final Nesterov pass on best ===")
    remaining = deadline - time.monotonic() - 10
    if remaining > 15 and best_seq is not None:
        n_final = len(best_seq)
        t_final = min(remaining * 0.8, 100)
        print(f"  Final Nesterov n={n_final}, budget={t_final:.0f}s")
        a_opt, _ = nesterov_optimize(best_seq, time_limit=t_final,
                                     lr=0.001, mom=0.96, top_k=15,
                                     temp=0.00005, seed=4321)
        try_seq(a_opt, "final_nest")

    if best_seq is not None:
        final_score = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f}")

    return [float(x) for x in best_seq]
