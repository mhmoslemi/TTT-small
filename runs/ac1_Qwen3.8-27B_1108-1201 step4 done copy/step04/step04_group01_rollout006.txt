import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the simplex {x : sum(x)=1, x>=0}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def obj_value(f):
    """Compute the objective value 2n*max(conv)/(sum)^2."""
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = fast_convolve(f)
    return 2.0 * n * conv.max() / (s * s)

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:
1. FW-LP for structural optimization
2. LSE (log-sum-exp) gradient descent with adaptive beta
3. Pairwise coordinate descent with ternary search (new)

6. Final coordinate descent polish

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f):
    """Build sparse constraint matrix for the LP."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[k - js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A_cache):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 25})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def lse_optimize(f0, time_limit, beta_start=20.0, beta_end=5000.0, seed=0):
    """Log-sum-exp smoothed gradient descent."""
    rng = np.random.default_rng(seed)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    beta = beta_start
    lr = 2.0 / n
    momentum = 0.90
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)

        # Softmax weights (numerically stable)
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        # Gradient via FFT: grad[i] = 2 * sum_m w[m] * f[m-i] (cross correlation)
        f_flipped = f[::-1].copy()
        total_len = (2 * n - 1) + n - 1
        fft_size = 1 << (total_len - 1).bit_length()
        W = np.fft.rfft(w, fft_size)
        F = np.fft.rfft(f_flipped, fft_size)
        conv_full = np.fft.irfft(W * F, fft_size)
        grad = 2.0 * conv_full[n - 1:n - 1 + n]

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 100
            break
        grad = grad / grad_norm

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.003, 5.0 / n)
        else:
            no_improve += 1
            if no_improve >= 8:
                lr *= 0.93
                momentum = max(0.85, momentum - 0.005)
            if no_improve > 60:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(2.0 / n, 0.001)
                beta = min(beta * 1.8, beta_end * 2)
                momentum = 0.88
                no_improve = 0

        if beta < beta_end:
            beta *= 1.004

        if it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE it={it}, obj={best_obj:.10f}, beta={beta:.1f}, lr={lr:.6f}, t={elapsed:.0f}s")

    return best_f, best_obj


def pairwise_coord_descent(f0, time_limit, seed=123, n_pairs_per_sweep=2000):

            j = rng.integers(0, n)

                continue

            # Ternary search on t in [-(f[j])*0.95, f[i]*0.95]

            best_t = 0.0

            b_gs = hi

                fd[i] += d_gs

                fd = np.maximum(fd, 0.0)
                sd = fd.sum()
                if sd < 1e-15:
