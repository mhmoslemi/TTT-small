"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast coordinate descent with analytical 1D subproblems
2. LP-based Frank-Wolfe for global direction finding
3. Multi-phase time allocation with diverse initializations

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def objective_value(a):
    """Compute the objective 2n*max(conv)/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fast_convolve(a)
    return 2.0 * n * c.max() / (s * s)

def coord_descent(a, time_limit, n_grid=50, verbose=False, seed=42):

        a = a / s  # normalize to sum = 1

        order = rng.permutation(n)
        
        improved = False
        for i in order:

            old_v = a[i]
            S = 1.0 - old_v  # sum of all other coordinates (since sum = 1)
            
            if S < 1e-15:

            # Build B and C arrays
            # For k in [i, i+n-1]: if k != 2i, B[k] = 2*a[k-i]; if k == 2i, C[k] = 1

                C[:] = 0.0

                B[2*i] = 0.0  # override for the quadratic case

            for v in v_candidates:
                vals = A_arr + B * v + C * v * v

                    best_h = h_val

def solve_lp_oracle(f, M, max_time=10):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    num_constraints = 2 * n - 1
    c_obj = -np.ones(n)
    
    # Build constraint matrix A_ub
    A_ub = np.zeros((num_constraints, n))
    for j in range(n):
        # (f*g)[k] = sum_j f[k-j]*g[j] for valid k-j
        # For fixed j, the constraint is: for each k, f[k-j]*g[j] contributes to c[k]
        # So A_ub[k, j] = f[k-j] if 0 <= k-j < n, else 0
        k_start = j
        k_end = min(j + n, num_constraints)
        if k_end > k_start:
            A_ub[k_start:k_end, j] = f[:k_end - k_start]
    
    b_ub = np.full(num_constraints, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': max_time, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def fw_lp(f0, time_limit, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()

        f = np.ones(len(f)) / len(f)

        f = f / s

    M = c.max()
    best_obj = M
    best_f = f.copy()

    it = 0
    m_factor = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        
        g = solve_lp_oracle(f, M * m_factor, max_time=min(15, rem // 4))
        if g is None:
            m_factor = min(m_factor * 1.3, 5.0)
            if m_factor > 4.0:
                break
            continue
        
        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = np.concatenate([[0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0],

        for alpha in alphas:

            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-8:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            c = fast_convolve(f)
            M = c.max()
            m_factor = 1.0
            
            if M < best_obj - 1e-13:
                best_obj = M
                best_f = f.copy()
                no_improve = 0

                no_improve += 1

            no_improve += 1

            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            c = fast_convolve(f)
            M = c.max()
            m_factor = 1.0
            no_improve = 0
    
    return best_f, best_obj


def generate_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    inits.append(np.ones(n) / n)
    
    # Symmetric triangle
    tri = 1.0 - np.abs(2.0 * x / (n - 1) - 1.0)
    inits.append(tri / tri.sum())
    
    # Cosine
    cos = np.cos(np.pi * x / (n - 1 + 1e-10)) ** 2 + 0.01
    inits.append(cos / cos.sum())
    
    # Bipartite-like: alternating blocks
    alt = np.zeros(n)
    block = max(1, n // 8)
    for j in range(0, n, 2 * block):
        alt[j:j+block] = 1.0
    inits.append(alt / alt.sum())
    
    # Random binary
    rb = (rng.random(n) < 0.5).astype(float)
    if rb.sum() > n * 0.1:
        inits.append(rb / rb.sum())
    
    # Parent-based
    if parent is not None:
        p = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        p = np.maximum(p, 0)
        ps = p.sum()
        if ps > 1e-10:
            inits.append(p / ps)
    
    return inits
