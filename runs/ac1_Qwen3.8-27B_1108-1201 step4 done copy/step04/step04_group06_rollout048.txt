def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_objective(a):

    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def build_conv_matrix(a):

    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C

def logsumexp_smooth_max(c, tau):

    c_max = c.max()
    shifted = tau * (c - c_max)

        return c_max, weights
    weights = exp_vals / sum_exp
    soft_max = c_max + np.log(sum_exp) / tau
    return soft_max, weights

def subgradient_optimize(a0, time_limit, tau=50.0):

    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()

        a = np.ones(n) / n
    else:
        a = a / s

    obj_history = []

        c = fast_convolve(a)
        soft_max, weights = logsumexp_smooth_max(c, tau)
        obj = 2.0 * n * soft_max  # Normalized (sum=1)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        obj_history.append(obj)
        
        # Gradient
        grad = soft_gradient(a, c, weights)
        
        # Polyak step size
        if len(obj_history) > 10:
            gap = obj - best_obj
            avg_sq_grad = np.mean(np.array([np.sum(g**2) for g in []]) + [np.sum(grad**2)])  # approximate
            # Use a simpler step size
            step = max(1e-6, min(0.5, gap / (np.sum(grad**2) + 1e-10)))
        else:
            step = 0.1

        s = a.sum()
        if s > 1e-15:

        # Adaptive tau
        tau *= 1.01

def generate_diverse_inits(n, parent, rng):

    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # From parent
    if parent is not None:

        ps = p.sum()

            pps = pp.sum()
            if pps > 1e-10:

2. LP-based active-set direction with multi-peak constraints

import numpy as np
import time
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_objective(a):
    """Compute objective without external call."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def build_conv_matrix(a):
    """Build matrix C such that (C @ g)[k] = conv(a, g)[k]."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def logsumexp_smooth_max(c, tau):
    """Compute soft max and its probability weights."""
    c_max = c.max()
    shifted = tau * (c - c_max)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    sum_exp = exp_vals.sum()
    if sum_exp < 1e-300:
        weights = np.zeros(len(c))
        weights[int(np.argmax(c))] = 1.0
        return c_max, weights
    weights = exp_vals / sum_exp
    soft_max = c_max + np.log(sum_exp) / tau
    return soft_max, weights


def soft_gradient(a, c, weights):
    """Gradient of weighted sum of convolution lags w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    for k in range(len(c)):
        if weights[k] < 1e-14:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start > j_end:
            continue
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += weights[k] * 2.0 * a[k - idxs]
    return grad

    if obj < params.get('best_obj', np.inf):
        params['best_obj'] = obj
        params['best_x'] = xv_norm.copy()
    return obj


def lbfgs_gradient(x, params):
    """Gradient for scipy minimize."""
    tau = params['tau']
    n = params['n']
    xv = np.maximum(x, 0.0)
    sv = xv.sum()
    if sv < 1e-15:
        return np.zeros(n)
    xv_norm = xv / sv
    c = fast_convolve(xv_norm)
    _, weights = logsumexp_smooth_max(c, tau)
    grad = soft_gradient(xv_norm, c, weights)
    return 2.0 * n * grad


def lbfgs_optimize(a0, time_limit, tau=100.0, max_iter=5000):
    """L-BFGS-B optimization with log-sum-exp smoothing."""
    n = len(a0)
    x0 = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = x0.sum()
    if s < 1e-15:
        x0 = np.ones(n) / n
    else:
        x0 = x0 / s

    params = {
        'tau': tau, 'n': n, 't_limit': time_limit,
        't_start': time.monotonic(), 'best_obj': np.inf, 'best_x': x0.copy()
    }

    bounds = [(0.0, 1000.0)] * n
    try:
        minimize(
            lbfgs_objective, x0, args=(params,),
            jac=lbfgs_gradient,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': max_iter, 'maxfun': max_iter * 3,
                     'ftol': 1e-14, 'gtol': 1e-10, 'iprint': 0}
        )
    except (ValueError, OverflowError, Exception):
        pass

    best_x = params.get('best_x', x0)
    best_obj = params.get('best_obj', compute_objective(x0))
    return best_x, best_obj


def lp_step(a, top_k=3, slack=0.001):
    """LP step: max sum(g) s.t. conv(a,g)[k] <= M*(1+slack) for top-k lags, g>=0."""
    n = len(a)
    c = fast_convolve(a)
    peak_indices = np.argpartition(c, -top_k)[-top_k:]
    M = c.max()

    C = build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M * 50.0)
    for k in peak_indices:
        b_ub[k] = M * (1.0 + slack)

    c_obj = -np.ones(n)
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=[(0, None)] * n,
                         method='highs', options={'presolve': True})
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g, g_sum, True
    except Exception:
        pass
    return a.copy(), 1.0, False


def lp_optimize(a0, time_limit, top_k=3, slack=0.001):
    """LP-based iterative optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = 2.0 * n * c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    perturb_count = 0

    while time.monotonic() - t0 < time_limit:
        a_new, g_sum, success = lp_step(a, top_k=top_k, slack=slack)

        if success and g_sum > 1.0 + 1e-8:
            d = a_new - a
            improved = False
            for t in [0.4, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.01]:
                a_try = np.maximum(a + t * d, 0.0)
                st = a_try.sum()
                if st < 1e-15:
                    continue
                a_try = a_try / st
                c_try = fast_convolve(a_try)
                obj_try = 2.0 * n * c_try.max()
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    c = c_try
                    improved = True
                    no_improve = 0
                    break
            if not improved:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve >= 150:
            rng = np.random.default_rng(hash((int(time.monotonic() * 1e6), perturb_count)) % (2**31))
            scale = 0.003 + 0.004 * min(perturb_count, 10)
            noise = rng.standard_normal(n) * scale
            a = np.maximum(best_a + noise, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            else:
                a = best_a.copy()
            c = fast_convolve(a)
            no_improve = 0
            perturb_count += 1
            if perturb_count > 60:
                break

    return best_a, best_obj


def subgradient_optimize(a0, time_limit, tau_init=50.0):
    """Subgradient method with adaptive step."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_objective(a)
    tau = tau_init
    t0 = time.monotonic()
    step_base = 0.3
    step_decay = 0.999
    iters = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        soft_max, weights = logsumexp_smooth_max(c, tau)
        obj = 2.0 * n * soft_max

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

        grad = soft_gradient(a, c, weights)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        gap = obj - best_obj
        step = step_base * min(1.0, gap / (grad_norm * grad_norm + 1e-10))
        step = max(step, 1e-6)
        step *= step_decay ** (iters // 100)

        a = a - step * grad
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-15:
            a = a / s
        else:
            a = best_a.copy()

        tau *= 1.005
        iters += 1

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['sine1'] = (np.sin(np.pi * x / n) ** 2 + 0.01)
    inits['sine1'] /= inits['sine1'].sum()
    inits['sine2'] = (np.sin(2 * np.pi * x / n) ** 2 + 0.01)
    inits['sine2'] /= inits['sine2'].sum()

    # Hamming window
    ham = 0.54 - 0.46 * np.cos(2.0 * np.pi * x / max(1, n - 1))
    inits['hamming'] = ham / ham.sum()

    # Blackman window
    N = max(1, n - 1)
    black = 0.42 - 0.5 * np.cos(2.0 * np.pi * x / N) + 0.08 * np.cos(4.0 * np.pi * x / N)
    black = np.maximum(black, 0.0)
    if black.sum() > 1e-10:
        inits['blackman'] = black / black.sum()

    # Tukey (raised cosine) window with r=0.3
    r = 0.3
    tukey = np.ones(n)
    half = int(r * n / 2)
    if half > 0:
        for i in range(half):
            w = 0.5 * (1 + np.cos(np.pi * (2 * i / r - n + 1) / n))
            tukey[i] = w
            tukey[n - 1 - i] = w
        tukey = np.maximum(tukey, 0.01)
    inits['tukey'] = tukey / tukey.sum()

    # From parent
    if parent is not None:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
            pp = np.maximum(pp, 0.0)
            pps = pp.sum()
            if pps > 1e-10:
                inits['parent'] = pp / pps

    return inits


def coordinate_polish(a0, time_limit, n_trials=50000):
    """Try coordinate-level perturbations to fine-tune."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, compute_objective(a)
    a = a / s
    n = len(a)
    best_obj = compute_objective(a)
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(99999)
    no_improve = 0
    max_trials = n_trials

    for _ in range(max_trials):
        if time.monotonic() - t0 > time_limit - 1.0:
            break

        i = rng.integers(0, n)
        delta = rng.choice([0.002, 0.005, 0.01, -0.002, -0.005, -0.01])
        if a[i] + delta < 0:
            continue

        a_try = a.copy()
        a_try[i] += delta
        # Redistribute the delta to other random positions
        j = rng.integers(0, n)
        if j == i:
            continue
        a_try[j] -= delta
        if a_try[j] < 0:
            continue

        c_try = fast_convolve(a_try)
        obj_try = 2.0 * n * c_try.max()
        if obj_try < best_obj - 1e-10:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 2000:
                break

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main entry point. Hybrid L-BFGS-B + LP + subgradient optimization."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, obj={compute_objective(p):.8f}")
    except Exception:
        pass

    # Helper to update best (inline to avoid closure)
    def check_and_update(a, label):
        nonlocal best_sequence, best_score
        score = compute_objective(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f}, n={len(a)}")
        return score

    # =========================================================
    # Phase 1: L-BFGS-B scan (most important phase)
    # =========================================================
    print("=== Phase 1: L-BFGS-B Multi-Start ===")
    t_start = time.monotonic()

    # Focus on lengths near parent
    n_parent = len(parent) if parent is not None else 768
    candidate_lengths = sorted(set([n_parent, max(256, n_parent // 2), min(2048, n_parent * 2)]))

    phase1_budget = 400.0
    results = []

    for n in candidate_lengths:
        remaining_total = deadline - time.monotonic()
        if remaining_total < 300:
            break
        inits = generate_inits(n, parent, rng)
        priority = ['parent', 'uniform', 'hamming', 'sine1', 'random', 'blackman', 'tukey']
        names = [nm for nm in priority if nm in inits][:4]

        t_per = max(15.0, (phase1_budget / len(candidate_lengths)) / max(1, len(names)))

        for name in names:
            remaining = deadline - time.monotonic()
            if remaining < 250:
                break
            t_alloc = min(t_per, remaining - 200)
            if t_alloc < 8:
                break
            print(f"  L-BFGS n={n}, init={name}, tau=100, t={t_alloc:.0f}s")
            a_opt, obj = lbfgs_optimize(inits[name], time_limit=t_alloc, tau=100.0, max_iter=4000)
            score = check_and_update(a_opt, f"LBF n={n} {name}")
            results.append((score, n, a_opt.copy()))

    results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.8f}")

    # =========================================================
    # Phase 2: LP deep optimization on best candidates
    # =========================================================
    print("=== Phase 2: LP Optimization ===")
    t_start2 = time.monotonic()

    # Get top 2 unique-length candidates
    top_candidates = []
    seen = set()
    for score, n, a_opt in results:
        if n not in seen:
            top_candidates.append((n, a_opt))
            seen.add(n)
        if len(top_candidates) >= 2:
            break
    if best_sequence is not None:
        if len(best_sequence) not in seen:
            top_candidates.append((len(best_sequence), best_sequence.copy()))

    remaining = deadline - time.monotonic() - 150
    t_per_lp = max(30.0, remaining * 0.5 / max(1, len(top_candidates)))

    for i, (n, a0) in enumerate(top_candidates):
        remaining = deadline - time.monotonic()
        if remaining < 130:
            break
        t_alloc = min(t_per_lp, remaining - 100)
        if t_alloc < 15:
            break
        print(f"  LP top_k=3, n={n}, t={t_alloc:.0f}s")
        a_opt, obj = lp_optimize(a0, time_limit=t_alloc, top_k=3, slack=0.001)
        check_and_update(a_opt, f"LP3 n={n}")

    # Also try top_k=1 on the best
    remaining = deadline - time.monotonic() - 100
    if remaining > 30 and best_sequence is not None:
        t_alloc = min(80.0, remaining - 80)
        if t_alloc > 10:
            print(f"  LP top_k=1, t={t_alloc:.0f}s")
            a_opt, obj = lp_optimize(best_sequence, time_limit=t_alloc, top_k=1, slack=0.0005)
            check_and_update(a_opt, "LP1 best")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.8f}")

    # =========================================================
    # Phase 3: L-BFGS-B with high tau (tight smoothing) on best
    # =========================================================
    print("=== Phase 3: L-BFGS-B Tight Smoothing ===")
    t_start3 = time.monotonic()

    if best_sequence is not None:
        taus = [200.0, 500.0, 1000.0]
        remaining = deadline - time.monotonic() - 80
        t_per = remaining / len(taus) * 0.8

        for tau in taus:
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break
            t_alloc = min(t_per, remaining - 50)
            if t_alloc < 8:
                break
            print(f"  L-BFGS tau={tau}, t={t_alloc:.0f}s")
            a_opt, obj = lbfgs_optimize(best_sequence, time_limit=t_alloc, tau=tau, max_iter=5000)
            check_and_update(a_opt, f"LBF tau={tau}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.8f}")

    # =========================================================
    # Phase 4: Subgradient + LP alternation
    # =========================================================
    print("=== Phase 4: Subgradient + LP ===")
    t_start4 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 60
        if remaining > 40:
            t_sg = min(remaining * 0.3, 80.0)
            if t_sg > 10:
                print(f"  Subgradient t={t_sg:.0f}s")
                a_opt, obj = subgradient_optimize(best_sequence, time_limit=t_sg, tau_init=100.0)
                check_and_update(a_opt, "Subgrad")

            # LP polish
            remaining = deadline - time.monotonic() - 40
            if remaining > 20:
                t_lp = min(remaining * 0.6, remaining - 30)
                if t_lp > 10:
                    print(f"  LP top_k=5 t={t_lp:.0f}s")
                    a_opt, obj = lp_optimize(best_sequence, time_limit=t_lp, top_k=5, slack=0.0005)
                    check_and_update(a_opt, "LP5")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.8f}")

    # =========================================================
    # Phase 5: Coordinate polish
    # =========================================================
    print("=== Phase 5: Coordinate Polish ===")
    t_start5 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20
        t_cd = min(remaining * 0.5, 40.0)
        if t_cd > 5:
            print(f"  Coord polish t={t_cd:.0f}s")
            a_opt, obj = coordinate_polish(best_sequence, time_limit=t_cd, n_trials=30000)
            check_and_update(a_opt, "CoordPolish")

    print(f"Phase 5 done ({time.monotonic()-t_start5:.0f}s). Best: {best_score:.8f}")

    # =========================================================
    # Phase 6: Final L-BFGS-B micro-polish
    # =========================================================
    if best_sequence is not None and time.monotonic() < deadline - 10:
        t_final = min(20.0, deadline - time.monotonic() - 5)
        if t_final > 5:
            a_opt, obj = lbfgs_optimize(best_sequence, time_limit=t_final, tau=1000.0, max_iter=2000)
            check_and_update(a_opt, "Final")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(768) / 768
        best_score = compute_objective(best_sequence)

    # Verify with actual evaluator
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
