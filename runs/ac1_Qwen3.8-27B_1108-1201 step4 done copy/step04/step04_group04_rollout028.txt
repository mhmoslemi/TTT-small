Multi-phase optimizer minimizing 2*n*max(convolve(a,a))/(sum(a))^2.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def m_sequence_init(n, rng):
    """Generate m-sequence inspired initialization mapped to non-negative simplex."""
    # Generate a pseudo-random binary sequence with good autocorrelation properties
    # Use a maximal-length LFSR-inspired approach
    # For simplicity, use a random sequence with forced alternation pattern
    # and then optimize the "phase" to minimize peak autocorrelation
    
    # Simple approach: random signs, mapped to non-negative
    b = rng.choice([-1, 1], size=n)
    # Add some structure: blend with a chirp
    x = np.arange(n, dtype=np.float64)
    chirp = np.sign(np.cos(2 * np.pi * x * x / (4 * n)))
    b = (b + chirp) / 2  # now in {-1, 0, 1}
    a = (b + 1) / 2  # now in {0, 0.5, 1}
    a = np.maximum(a, 0.01)  # ensure positive
    return a / a.sum()


def lp_frank_wolfe(f0, time_limit, relax=1.0, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)

    best_f = f.copy()

        b_ub = np.full(2 * n - 1, M_eff)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

            # Try with more relaxed constraint
            relax = min(relax * 1.2, 5.0)
            continue

        if g_sum < 1e-15:
            break

            no_improve += 1
            if no_improve > 30:
                break
            continue
        
        # Line search

        best_val = M
        for alpha in [0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:

            if st < 1e-15:

                best_alpha = alpha

            hi = min(1.0, best_alpha + 0.05)

                ft = np.maximum(ft, 0.0)

def multi_peak_grad(f0, time_limit, T=0.001, verbose=False):
    """Gradient descent with log-sum-exp smoothed max, multi-peak aware."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = 0.005
    T_current = T
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        
        # Log-sum-exp gradient
        scaled = conv / T_current
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        p = w / ws
        
        # Gradient: d/d a[i] of log(sum_k exp(conv[k]/T))
        # = (1/T) * sum_k p[k] * d conv[k] / d a[i]
        # d conv[k]/d a[i] = a[k-i] (if valid) * 2 (for symmetry, when i != k-i)
        # More precisely: if 0 <= k-i < n, gradient contribution is a[k-i]
        # (and if k-i == i, it's 2*a[i]... wait no)
        # conv[k] = sum_j a[j]*a[k-j]
        # d/d a[i] = a[k-i] if max(0,k-n+1) <= i <= min(n-1,k) and 0 <= k-i <= n-1
        # But by symmetry of the sum, it's actually:
        # If i is in the valid range for k, then a[i] appears in the term j=i (contributing a[k-i])
        # and if k-i is also valid (which it is by symmetry), a[i] also appears as a[k-j] when j=k-i
        # So the total derivative is 2*a[k-i] if i != k-i, or a[i] if i == k-i (k=2i)
        # Wait, let me be more careful.
        # conv[k] = sum_{j} a[j] * a[k-j]
        # The terms where a[i] appears:
        #   j = i: term is a[i] * a[k-i], derivative w.r.t. a[i] is a[k-i]
        #   k-j = i, i.e., j = k-i: term is a[k-i] * a[i], derivative w.r.t. a[i] is a[k-i]
        # If i = k-i (k = 2i), these are the same term: a[i]*a[i], derivative is 2*a[i]
        # If i != k-i, we have two separate terms, each giving a[k-i], total 2*a[k-i]
        # So in all cases: d conv[k]/d a[i] = 2*a[k-i] (if valid), where valid means
        # max(0,k-n+1) <= i <= min(n-1,k)
        
        # Compute gradient efficiently
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if p[k] < 1e-10:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo > hi:
                continue
            idx = np.arange(lo, hi + 1)
            grad[idx] += p[k] * 2.0 * f[k - idx]
        
        grad /= T_current
        
        # Project to tangent space
        grad = grad - np.mean(grad)
        
        f_new = project_simplex(f - lr * grad)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"      MPG iter {it}: obj={best_obj:.10f}, T={T_current:.6f}")
        else:
            no_improve += 1
            if no_improve > 20:
                lr = max(lr * 0.9, 0.0001)
        
        # Anneal temperature
        T_current = max(T_current * 0.998, T * 0.1)
        
        if no_improve > 300:
            rng = np.random.default_rng(hash((it * 31 + n)) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            lr = 0.005
            T_current = T
            no_improve = 0
    
    return best_f, best_obj


def spectral_shaping(f0, n_iter=100, alpha=0.05):
    """Shape the spectrum to be flatter via iterative adjustment."""
    f = f0.copy()
    n = len(f)
    for _ in range(n_iter):
        F = np.fft.rfft(f, 2 * n)
        power = np.abs(F) ** 2
        # Target: flat power spectrum
        target_mean = power.mean()
        # The "error" spectrum
        error = power - target_mean
        # Gradient in time domain (approximate)
        # d/d f of sum(error^2) involves the IFFT of error * F
        # Simplification: use IFFT of error as a proxy gradient
        grad_t = np.fft.irfft(error, 2 * n)[:n]
        grad_t = grad_t / (np.max(np.abs(grad_t)) + 1e-10)
        f = project_simplex(f - alpha * grad_t)
    return f


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # M-sequence inspired
    inits.append(m_sequence_init(n, rng))
    
    # Edge-concentrated
    x = np.arange(n, dtype=np.float64)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 0.5
    inits.append(edge_w / edge_w.sum())

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)

    osc = 0.5 + 0.5 * np.cos(2 * np.pi * x / max(1, n // 7))

    # Random
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    half_n = n // 2

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits

"""
Multi-phase optimizer for min 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-FW, smoothed gradient descent, spectral shaping, and multi-length search.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgrad(f, conv, k):
    """Compute subgradient of conv[k] w.r.t. f."""
    n = len(f)
    sg = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        sg[idx] = 2.0 * f[k - idx]
    return sg


def lp_frank_wolfe(f0, time_limit, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    relax = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        M_eff = M * relax
        
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M_eff)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        
        if not res.success or res.x is None:
            relax = min(relax * 1.15, 5.0)
            if relax > 4.0:
                # Perturb and restart
                rng = np.random.default_rng(it * 7919 + n)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(f)
                M = conv.max()
                relax = 1.0
                no_improve = 0
            continue
        
        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 40:
                break
            relax = min(relax * 1.03, 3.0)
            continue
        
        # Line search with more points
        best_alpha = 0.0
        best_val = M
        alphas = [0.02, 0.03, 0.05, 0.07, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.65, 0.8, 1.0]
        for alpha in alphas:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Bisection refinement
        if 0.01 < best_alpha < 0.99:
            lo_a = best_alpha - 0.04
            hi_a = best_alpha + 0.04
            lo_a = max(0.0, lo_a)
            hi_a = min(1.0, hi_a)
            for _ in range(7):
                mid = (lo_a + hi_a) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo_a = mid
                else:
                    hi_a = mid
        
        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            relax = 1.0
            no_improve = 0
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
        else:
            no_improve += 1
            relax = min(relax * 1.05, 4.0)
        
        if no_improve > 60:
            rng = np.random.default_rng(it * 7919 + n * 31)
            scale = 0.001 + 0.001 * min(no_improve // 60, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            relax = 1.0
            no_improve = 0
    
    return best_f, best_obj


def smoothed_grad_descent(f0, time_limit, T_init, T_min, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = 0.006
    T = T_init
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        scaled = conv / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = w / ws
        
        # Compute gradient using FFT cross-correlation
        # grad[i] = 2 * sum_k p[k] * f[k-i] (where valid)
        # = 2 * cross_correlation(f, p reversed, shifted)
        # Actually: grad[i] = (2/T) * sum_k p[k] * f[k-i] for valid k-i
        # This is a cross-correlation
        
        # Efficient computation:
        # sum_k p[k] * f[k-i] for i = 0,...,n-1
        # Let's define: for each i, sum over k where 0<=k-i<n, i.e., i <= k < i+n
        # = sum over valid k of p[k] * f[k-i]
        # This is the cross-correlation of p and f at lag i
        # = (f * p_reversed)[i + n - 1] in convolution terms
        # Or more simply: use FFT
        
        fft_size = 1 << (3 * n).bit_length()
        F = np.fft.rfft(f, fft_size)
        P = np.fft.rfft(p, fft_size)
        # cross[i] = sum_k p[k] * f[k-i] = sum_j f[j] * p[j+i]
        # = (f convolved with reversed p)[i + offset]
        # Actually: sum_j f[j] * p[j+i] for j such that 0<=j<n and 0<=j+i<2n-1
        # This is the cross-correlation
        # In FFT: IFFT(conj(F) * P) gives cross-correlation
        cross = np.fft.irfft(np.conj(F) * P, fft_size)
        grad = 2.0 * cross[:n] / T
        
        # Project to tangent space
        grad = grad - np.mean(grad)
        
        f_new = project_simplex(f - lr * grad)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                lr = max(lr * 0.88, 0.0001)
        
        T = max(T * 0.997, T_min)
        
        if no_improve > 400:
            rng = np.random.default_rng(it * 331 + 17 + n)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.0008)
            lr = max(lr * 2.0, 0.003)
            T = max(T_init * 0.3, T_min)
            no_improve = 0
    
    return best_f, best_obj

def generate_inits(n, parent, rng):
    inits = []
    inits.append(np.ones(n) / n)
    
    # M-sequence style
    b = rng.choice([-1, 1], size=n).astype(np.float64)
    x = np.arange(n, dtype=np.float64)
    chirp = np.sign(np.cos(2 * np.pi * x * x / (4.0 * n)))
    b = b + 0.3 * chirp
    a = (b + np.max(np.abs(b))) / (2.0 * np.max(np.abs(b)))
    a = np.maximum(a, 0.01)
    inits.append(a / a.sum())
    
    # Edge
    x = np.arange(n, dtype=np.float64)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 0.7
    inits.append(edge_w / edge_w.sum())
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.12 * n)) ** 2)
    inits.append(g / g.sum())
    
    # Two peaks
    h1 = np.exp(-0.5 * ((x - 0.15 * n) / (0.06 * n)) ** 2)
    h2 = np.exp(-0.5 * ((x - 0.85 * n) / (0.06 * n)) ** 2)
    inits.append((h1 + h2) / (h1 + h2).sum())
    
    # Oscillatory (period ~n/5)
    osc = 0.5 + 0.5 * np.cos(2 * np.pi * x / max(1.0, n / 5.0))
    inits.append(osc / osc.sum())
    
    # Random
    inits.append(rng.dirichlet(np.ones(n) * 0.4))
    
    # Symmetric random
    half_n = n // 2
    half = rng.dirichlet(np.ones(half_n) * 0.5)
    sym = np.zeros(n)
    sym[:half_n] = half
    sym[half_n:] = half[::-1]
    inits.append(sym / sym.sum())
    
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits

"""
Multi-phase optimizer for minimizing 2*n*max(convolve(a,a))/(sum(a))^2.
Strategies: improved LP-FW, smoothed gradient descent, multi-length exploration,
spectral nudging, adaptive perturbations, and extended fine polishing.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_frank_wolfe(f0, time_limit):
    """LP-based Frank-Wolfe with adaptive relaxation and fine line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    relax = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        M_eff = M * relax

        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M_eff)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

        if not res.success or res.x is None:
            relax = min(relax * 1.12, 5.0)
            if relax > 4.5:
                rng = np.random.default_rng(it * 7919 + n)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(f)
                M = conv.max()
                relax = 1.0
                no_improve = 0
            continue

        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            relax = min(relax * 1.02, 2.5)
            continue

        # Multi-point line search
        best_alpha = 0.0
        best_val = M
        alphas = [0.02, 0.04, 0.06, 0.08, 0.1, 0.14, 0.18, 0.25, 0.35, 0.5, 0.65, 0.8, 1.0]
        for alpha in alphas:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Bisection refinement
        if 0.02 < best_alpha < 0.98:
            lo_a = max(0.0, best_alpha - 0.05)
            hi_a = min(1.0, best_alpha + 0.05)
            for _ in range(7):
                mid = (lo_a + hi_a) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo_a = mid
                else:
                    hi_a = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            relax = 1.0
            no_improve = 0
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
        else:
            no_improve += 1
            relax = min(relax * 1.04, 3.5)

        if no_improve > 70:
            rng = np.random.default_rng(it * 7919 + n * 31)

            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            relax = 1.0
            no_improve = 0

    return best_f, best_obj


def smoothed_grad_descent(f0, time_limit, T_init, T_min):
    """Gradient descent with log-sum-exp smoothed max objective."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = 0.006
    T = T_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        scaled = conv / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = w / ws

        # FFT-based cross-correlation for gradient
        fft_size = 1 << (3 * n).bit_length()
        F = np.fft.rfft(f, fft_size)
        P = np.fft.rfft(p, fft_size)
        cross = np.fft.irfft(np.conj(F) * P, fft_size)
        grad = 2.0 * cross[:n] / T

        grad = grad - np.mean(grad)
        f_new = project_simplex(f - lr * grad)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 25:
                lr = max(lr * 0.90, 0.0001)

        T = max(T * 0.997, T_min)

        if no_improve > 350:
            rng = np.random.default_rng(it * 331 + 17 + n)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.0008)
            lr = max(lr * 1.8, 0.003)
            T = max(T_init * 0.4, T_min)
            no_improve = 0

    return best_f, best_obj


def spectral_nudge(f0, n_iter=30, strength=0.03):
    """Nudge the spectrum toward flatness, then project back."""
    f = f0.copy()
    n = len(f)
    for _ in range(n_iter):
        F = np.fft.rfft(f, 2 * n)
        power = np.abs(F) ** 2
        target = power.mean()
        # Fractional adjustment: reduce high-power bins, increase low-power
        adjust = np.ones(len(power))
        high_mask = power > target * 1.5
        low_mask = power < target * 0.5
        adjust[high_mask] = 1.0 - strength
        adjust[low_mask] = 1.0 + strength
        F_new = F * np.sqrt(adjust)
        f_new = np.fft.irfft(F_new, 2 * n)[:n]
        f = project_simplex(f_new)
    return f


def generate_inits(n, parent, rng):
    """Generate diverse initializations on the simplex."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    inits.append(np.ones(n) / n)

    # M-sequence / pseudo-random with chirp blend
    b = rng.choice([-1, 1], size=n).astype(np.float64)
    chirp = np.sign(np.cos(2.0 * np.pi * x * x / (4.0 * n)))
    mixed = b + 0.4 * chirp
    a = (mixed - mixed.min()) / (mixed.max() - mixed.min() + 1e-10)
    a = np.maximum(a, 0.005)
    inits.append(a / a.sum())

    # Edge concentrated
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 0.6
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.1 * n)) ** 2)
    inits.append(g / g.sum())

    # Two humps
    h1 = np.exp(-0.5 * ((x - 0.18 * n) / (0.07 * n)) ** 2)
    h2 = np.exp(-0.5 * ((x - 0.82 * n) / (0.07 * n)) ** 2)
    inits.append((h1 + h2) / (h1 + h2).sum())

    # Oscillatory
    period = max(2.0, n / 6.0)
    osc = 0.5 + 0.5 * np.cos(2.0 * np.pi * x / period)
    inits.append(osc / osc.sum())

    # Pure random
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    # Symmetric random
    half_n = n // 2
    half = rng.dirichlet(np.ones(half_n) * 0.5)
    sym = np.zeros(n)
    sym[:half_n] = half
    sym[half_n:] = half[::-1]
    inits.append(sym / sym.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits
