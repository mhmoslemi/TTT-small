"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach combining:
1. Nesterov-accelerated proximal gradient with exponential peak weighting
2. LP-based direction finding with bisection step sizing
3. Multi-length exploration and hierarchical refinement
4. Peak-shaving coordinate descent for final polish
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def compute_subgradient(a, conv, weights):
    """Compute weighted subgradient of max autocorrelation."""
    n = len(a)
    grad = np.zeros(n)
    for k in range(len(weights)):
        if weights[k] < 1e-12:
            continue
        w = weights[k]
        if k == 0:
            grad += w * 2.0 * a
        else:
            if n > k:
                grad[:n - k] += w * a[k:]
            if k < n:
                grad[k:] += w * a[:n - k]
    return grad


def compute_peak_weights(conv, top_k=50, temp=0.005):
    """Compute weight distribution over peaks using softmax."""
    # Use top-k peaks with softmax weighting
    n_peaks = len(conv)
    if top_k >= n_peaks:
        scaled = conv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum > 1e-15:
            w /= w_sum
        else:
            w = np.zeros(n_peaks)
            w[np.argmax(conv)] = 1.0
        return w
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        w = np.zeros(n_peaks)
        peak_vals = conv[peaks]
        scaled = peak_vals / temp
        scaled -= scaled.max()
        exp_w = np.exp(scaled)
        exp_sum = exp_w.sum()
        if exp_sum > 1e-15:
            exp_w /= exp_sum
        else:
            exp_w = np.zeros(len(peaks))
            exp_w[np.argmax(peak_vals)] = 1.0
        w[peaks] = exp_w
        return w


def normalize(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def objective(a):
    """Compute objective value 2*n*max(conv)/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def nesterov_optimize(a0, time_limit, lr_init=0.01, momentum_init=0.9,
                      top_k=50, temp_init=0.003, temp_growth=1.002,
                      lr_decay=0.9997, mom_growth=0.0001, max_mom=0.97):
    """Nesterov-accelerated gradient descent with adaptive parameters."""
    a = normalize(a0.astype(np.float64).copy())
    n = len(a)
    v = np.zeros(n)
    y = a.copy()

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum_init
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        # Nesterov lookahead
        y = a + mom * v
        y = normalize(y)

        # Compute subgradient
        conv_y = fast_convolve(y)
        weights = compute_peak_weights(conv_y, top_k=top_k, temp=temp)
        grad = compute_subgradient(y, conv_y, weights)

        # Gradient step
        v = mom * v - lr * grad
        a_new = normalize(a + v)

        conv_new = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv_new.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            a_new = a  # Don't move if no improvement

        a = a_new
        conv = conv_new
        iter_count += 1

        # Adaptive parameter updates
        lr *= lr_decay
        temp *= temp_growth
        mom = min(max_mom, mom + mom_growth)

        if no_improve >= 300:
            break

    return best_a, best_obj


def lp_step(a, conv, M_factor=1.0001):
    """
    One LP-based step. Find direction g maximizing sum(g) s.t. conv(a,g) <= M, g >= 0.
    Returns (a_new, improved, g_sum_improvement).
    """
    n = len(a)
    M = conv.max() * M_factor
    C = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 10})
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0)
            g_sum = g.sum()
            a_sum = a.sum()
            if g_sum > a_sum * (1.0 + 1e-6):
                return g, True, g_sum / a_sum
    except Exception:
        pass
    return a.copy(), False, 1.0


def lp_optimize(a0, time_limit, verbose=False):
    """LP-based iterative optimization with adaptive restarts."""
    a = normalize(a0.astype(np.float64).copy())
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    perturb_count = 0
    max_no_improve = 60

    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        g, success, _ = lp_step(a, conv)

        if success:
            d = g - a
            norm_d = np.linalg.norm(d)
            if norm_d > 1e-12:
                # Try multiple step sizes
                best_t = 0
                best_local_obj = best_obj
                for t in [0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]:
                    a_try = normalize(a + t * d)
                    conv_try = fast_convolve(a_try)
                    obj_try = 2.0 * n * conv_try.max()
                    if obj_try < best_local_obj - 1e-12:
                        best_local_obj = obj_try
                        best_t = t
                        a = a_try
                        conv = conv_try
                        break

                if best_t > 0:
                    if best_local_obj < best_obj - 1e-12:
                        best_obj = best_local_obj
                        best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1

        if verbose and no_improve % 50 == 0:
            print(f"    LP: obj={best_obj:.8f}, no_improve={no_improve}, t={time.monotonic()-t0:.1f}s")

        if no_improve >= max_no_improve:
            # Perturb to escape
            rng = np.random.default_rng(int(time.monotonic() * 1000) + perturb_count)
            scale = 0.002 + 0.003 * min(perturb_count, 10)
            noise = rng.standard_normal(n) * scale
            a = normalize(best_a + noise)
            conv = fast_convolve(a)
            no_improve = 0
            perturb_count += 1
            max_no_improve = min(200, 60 + 20 * perturb_count)

            if perturb_count > 50:
                # Switch to gradient for a bit
                a, obj = nesterov_optimize(best_a, time_limit=1.0, lr_init=0.005,
                                           top_k=80, temp_init=0.001)
                conv = fast_convolve(a)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                perturb_count = 0
                max_no_improve = 60

    return best_a, best_obj


def peak_shaving(a0, time_limit):

    a = normalize(a0.astype(np.float64).copy())

    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        conv = fast_convolve(a)
        peak_idx = int(np.argmax(conv))
        M = conv[peak_idx]

        # Find elements contributing to this peak
        if peak_idx == 0:
            # conv[0] = sum a_j^2: reduce by making more uniform
            top_elements = np.argsort(a)[-10:]
            bottom_elements = np.argsort(a)[:10]
            for i in top_elements:
                for j in bottom_elements:
                    if a[i] - a[j] < 1e-6:
                        continue
                    delta = min(0.001, (a[i] - a[j]) * 0.1)
                    a_try = a.copy()
                    a_try[i] -= delta
                    a_try[j] += delta
                    a_try = normalize(a_try)
                    conv_try = fast_convolve(a_try)

                    if obj_try < best_obj - 1e-10:
                        a = a_try
                        best_obj = obj_try

                        break
                else:
                    continue
                break
        else:
            # conv[peak_idx] = sum a[j]*a[j+peak_idx]
            # Find the largest product and split it
            j_start = max(0, peak_idx - n + 1)
            j_end = min(n - 1, peak_idx)

                    a_try[i] += delta

                    a_try = normalize(a_try)
                    conv_try = fast_convolve(a_try)

            else:

                # This is getting complicated; just use random perturbation

def generate_inits(n, parent, rng):
    """Generate diverse initializations for given length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random'] = normalize(rng.dirichlet(np.ones(n) * 0.5))

    # Sine-based
    s1 = np.sin(np.pi * x / n) ** 2 + 0.01
    inits['sine'] = normalize(s1)

    s2 = np.sin(2 * np.pi * x / n) ** 2 + 0.01
    inits['sine2'] = normalize(s2)

    # Bimodal
    edge = max(1, n // 6)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = normalize(bi)

    # Triangular (peak at center)
    tri = 1.0 - np.abs(x - n / 2) / (n / 2)
    inits['triangular'] = normalize(np.maximum(tri, 0.01))

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent'] = pp / pps

    return inits

            a_opt, obj = nesterov_optimize(a0, time_limit=t_alloc,
                                           lr_init=0.015, momentum_init=0.91,
                                           top_k=60, temp_init=0.002,
                                           temp_growth=1.003, lr_decay=0.9996)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Grad n={n}, {name}: {score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 2: LP optimization on top candidates
    # =========================================================
    print("=== Phase 2: LP Optimization ===")
    t_start2 = time.monotonic()

    # Get top unique-length candidates
    top_configs = []
    seen_lengths = set()
    for score, n, name, a_opt in scan_results:
        if n not in seen_lengths:
            top_configs.append((n, a_opt))
            seen_lengths.add(n)
        if len(top_configs) >= 3:
            break

    if best_sequence is not None:
        if len(best_sequence) not in seen_lengths:
            top_configs.append((len(best_sequence), best_sequence.copy()))

    remaining = deadline - time.monotonic() - 400.0
    n_cands = max(1, len(top_configs))
    t_per_lp = max(30.0, remaining * 0.7 / n_cands)

    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 350:
            break
        t_alloc = min(t_per_lp, deadline - time.monotonic() - 300)
        if t_alloc < 10:
            break

        print(f"  LP cand {i}: n={n}, budget={t_alloc:.0f}s")
        a_opt, obj = lp_optimize(a0, time_limit=t_alloc)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  LP {i} (n={n}): {score:.6f} *NEW BEST*")
            else:
                print(f"  LP {i} (n={n}): {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 3: Alternating LP + Gradient refinement
    # =========================================================
    print("=== Phase 3: Alternating Refinement ===")
    t_start3 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 200.0
        n_best = len(best_sequence)

        # Multiple rounds of alternating approaches
        round_time = min(remaining / 4, 150.0)

        for r in range(4):
            if time.monotonic() > deadline - 150:
                break
            t_alloc = min(round_time, deadline - time.monotonic() - 120)
            if t_alloc < 10:
                break

            # Alternate between LP and gradient
            if r % 2 == 0:
                a_opt, obj = lp_optimize(best_sequence, time_limit=t_alloc)
            else:
                a_opt, obj = nesterov_optimize(best_sequence, time_limit=t_alloc,
                                               lr_init=0.005, momentum_init=0.93,
                                               top_k=100, temp_init=0.001,
                                               temp_growth=1.005, lr_decay=0.9995)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Round {r} ({'LP' if r%2==0 else 'Grad'}): {score:.6f} *NEW BEST*")
            else:
                print(f"  Round {r} ({'LP' if r%2==0 else 'Grad'}): {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 4: Peak shaving + fine gradient
    # =========================================================
    print("=== Phase 4: Fine Tuning ===")
    t_start4 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 30.0

        # Peak shaving
        t_ps = min(remaining * 0.3, 60.0)
        if t_ps > 5:
            a_opt, obj = peak_shaving(best_sequence, time_limit=t_ps)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  PeakShaving: {score:.6f} *NEW BEST*")

        # Very fine gradient
        remaining = deadline - time.monotonic() - 20.0
        if remaining > 10:
            t_fg = min(remaining * 0.5, 40.0)
            if t_fg > 5:
                a_opt, obj = nesterov_optimize(best_sequence, time_limit=t_fg,
                                               lr_init=0.002, momentum_init=0.95,
                                               top_k=200, temp_init=0.0005,
                                               temp_growth=1.01, lr_decay=0.9999)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  FineGrad: {score:.6f} *NEW BEST*")

        # Final LP micro-step
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 10:
            a_opt, obj = lp_optimize(best_sequence, time_limit=min(remaining * 0.8, 20.0))
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  FinalLP: {score:.6f} *NEW BEST*")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
