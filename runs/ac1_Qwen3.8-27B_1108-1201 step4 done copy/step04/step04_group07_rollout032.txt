                lr = max(lr_start * 0.4, 0.05)
                no_improve = 0
                if verbose:
                    print(f"    MD restart at iter {it}")

        tau = max(tau * tau_decay, tau_end)
        lr = max(lr * 0.9995, min_lr)

        if verbose and it % 500 == 0:
            print(f"    MD iter {it}: obj={best_obj:.10f}, tau={tau:.7f}, lr={lr:.4f}, no_imp={no_improve}")

    return best_f, best_obj


def pairwise_local_search(f0, time_limit, rng, pairs_per_iter=200, search_steps=25, verbose=False):
    """Local search via pairwise golden-section optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0

    def eval_pair(f_arr, i, j, t):
        """Evaluate max correlation after moving t from j to i."""
        fa = f_arr.copy()
        fa[i] += t
        fa[j] -= t
        # Only recompute affected lags for speed (but use full for safety)
        c = fast_convolve(fa)
        return c.max()

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Sample random pairs
        pairs = rng.integers(0, n, size=(pairs_per_iter, 2))
        for idx in range(pairs_per_iter):
            i = int(pairs[idx, 0])
            j = int(pairs[idx, 1])
            if i == j:
                continue
            if f[i] < 1e-12 or f[j] < 1e-12:
                continue

            # Golden section search for optimal t in [-f[i]*0.5, f[j]*0.5]
            # (conservative range to avoid too-large jumps)
            t_lo = -f[i] * 0.5
            t_hi = f[j] * 0.5
            if t_hi - t_lo < 1e-14:
                continue

            gr = (np.sqrt(5.0) - 1.0) / 2.0
            t1 = t_hi - gr * (t_hi - t_lo)
            t2 = t_lo + gr * (t_hi - t_lo)
            f1 = eval_pair(f, i, j, t1)
            f2 = eval_pair(f, i, j, t2)

            for _ in range(search_steps):
                if time.monotonic() - t0 > time_limit:
                    break
                if f1 < f2:
                    t_hi = t2
                    t2 = t1
                    f2 = f1
                    t1 = t_hi - gr * (t_hi - t_lo)
                    f1 = eval_pair(f, i, j, t1)
                else:
                    t_lo = t1
                    t1 = t2
                    f1 = f2
                    t2 = t_lo + gr * (t_hi - t_lo)
                    f2 = eval_pair(f, i, j, t2)

            # Best t is the midpoint
            t_best = (t_lo + t_hi) / 2.0
            f_candidate = f.copy()
            f_candidate[i] += t_best
            f_candidate[j] -= t_best
            f_candidate = np.maximum(f_candidate, 0.0)
            cs = f_candidate.sum()
            if cs < 1e-15:
                continue
            f_candidate = f_candidate / cs

            obj_cand = fast_convolve(f_candidate).max()
            if obj_cand < best_obj - 1e-14:
                best_obj = obj_cand
                best_f = f_candidate.copy()
                f = f_candidate

        if verbose and it % 10 == 0:
            print(f"    Pairwise iter {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def coordinate_descent(f0, time_limit, rng, verbose=False):
    """Coordinate descent: optimize single coordinates with sum constraint."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        improved = False
        # Try each coordinate
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-14:
                continue

            # Parameterize: a_j' = f[j] + t, rest scaled to maintain sum
            # a_i' = f[i] * (1 - t) / (1 - f[j]) for i != j (if 1-f[j] > 0)
            # Valid: t in [-f[j], (1 - f[j]) * large] ... actually just t in [-f[j], 1 - f[j]]
            # But we also need a_i' >= 0, which is automatic since f[i] >= 0 and (1-t)/(1-f[j]) >= 0 for t <= 1-f[j]

            denom = 1.0 - f[j]
            if denom < 1e-12:
                continue

            t_lo = -f[j]
            t_hi = 2.0 * denom  # allow increasing a_j

            g = 0.6180339887498949
            t1 = t_hi - g * (t_hi - t_lo)
            t2 = t_lo + g * (t_hi - t_lo)

            def eval_t(t):
                fj = f[j] + t
                if fj < 0:
                    return 1e10
                scale = (1.0 - fj) / denom
                if scale < 0:
                    return 1e10
                fa = f * scale
                fa[j] = fj
                return fast_convolve(fa).max()

            f1 = eval_t(t1)
            f2 = eval_t(t2)

            for _ in range(20):
                if time.monotonic() - t0 > time_limit:
                    break
                if f1 < f2:
                    t_hi = t2
                    t2 = t1
                    f2 = f1
                    t1 = t_hi - g * (t_hi - t_lo)
                    f1 = eval_t(t1)
                else:
                    t_lo = t1
                    t1 = t2
                    f1 = f2
                    t2 = t_lo + g * (t_hi - t_lo)
                    f2 = eval_t(t2)

            t_best = (t_lo + t_hi) / 2.0
            fj = f[j] + t_best
            if fj < 0 or fj > 1.0:
                continue
            scale = (1.0 - fj) / denom
            if scale < 0:
                continue
            fa = f * scale
            fa[j] = fj
            fa = np.maximum(fa, 0.0)
            fs = fa.sum()
            if fs < 1e-15:
                continue
            fa = fa / fs
            obj = fast_convolve(fa).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = fa.copy()
                f = fa
                improved = True

        if not improved and it > 50:
            # Perturb and try again
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            best_obj = fast_convolve(f).max()

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    x = np.arange(n, dtype=np.float64)
    for sigma_frac in [0.1, 0.2, 0.35, 0.5]:
        sigma = sigma_frac * n
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / sigma) ** 2)
        inits.append(g / g.sum())

    # U-shape (bimodal)
    edge_w = 1.0 / (np.abs(x - 0.5 * (n - 1)) / n + 0.05)
    inits.append(edge_w / edge_w.sum())

    # Cosine
    c = np.abs(np.cos(np.pi * x / n)) + 0.02
    inits.append(c / c.sum())

    # Power law
    p = (x + 1) ** 0.5 * (n - x + 1) ** 0.5
    inits.append(p / p.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
        # Perturbed parent
        for scale in [0.005, 0.01, 0.02]:
            pp = px + rng.standard_normal(n) * scale
            pp = np.maximum(pp, 0)
            pp = pp / pp.sum()
            inits.append(pp)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search:
    Phase 1: Multi-start initialization
    Phase 2: Mirror descent (multiplicative weights) with annealing
    Phase 3: Pairwise golden-section local search
    Phase 4: Coordinate descent polish
    Phase 5: Final refinement with alternating methods
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        try:
            score = float(evaluate_sequence([float(x) for x in a]))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence([float(x) for x in parent]))
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except NameError:
        pass
    except Exception:
        pass

    n_base = len(parent) if parent is not None else 384

    # === Phase 1: Multi-start initialization ===
    print("\n=== Phase 1: Multi-start Initialization ===")
    t0 = time.monotonic()

    inits = generate_inits(n_base, parent, rng)
    for i, ini in enumerate(inits):
        try_seq(ini, f"init_{i}")

    # Try a few different lengths
    for n_try in [n_base - 16, n_base + 16, n_base - 32, n_base + 32]:
        if n_try < 64 or n_try > 2000:
            continue
        inits_n = generate_inits(n_try, parent, rng)
        try_seq(inits_n[0], f"len{n_try}_uniform")
        try_seq(inits_n[2], f"len{n_try}_gauss")

    print(f"  Phase 1 best: {best_score:.8f} (n={len(best_seq)})")

    # === Phase 2: Mirror Descent ===
    print("\n=== Phase 2: Mirror Descent ===")
    remaining = deadline - time.monotonic() - 200
    t_md = min(remaining * 0.45, 700)

    if t_md > 30:
        print(f"  Running MD (n={len(best_seq)}), budget={t_md:.0f}s")
        # Split time for multiple MD runs
        t_each = t_md / 3.0

        # Run 1: Standard
        a_opt, _ = mirror_descent_optimize(best_seq, time_limit=t_each,
                                           tau_start=0.004, tau_end=0.00003,
                                           lr_start=0.8, momentum=0.92)
        try_seq(a_opt, "md1")

        # Run 2: Different hyperparams
        rem = deadline - time.monotonic() - 180
        t_each2 = min(rem * 0.5, 250)
        if t_each2 > 20:
            a_opt, _ = mirror_descent_optimize(best_seq, time_limit=t_each2,
                                               tau_start=0.002, tau_end=0.00001,
                                               lr_start=0.5, momentum=0.88)
            try_seq(a_opt, "md2")

        # Run 3: From perturbed start
        rem = deadline - time.monotonic() - 150
        t_each3 = min(rem * 0.5, 200)
        if t_each3 > 20:
            f_pert = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * 0.01)
            a_opt, _ = mirror_descent_optimize(f_pert, time_limit=t_each3,
                                               tau_start=0.003, tau_end=0.00002,
                                               lr_start=0.6, momentum=0.90)
            try_seq(a_opt, "md3_pert")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Pairwise Local Search ===
    print("\n=== Phase 3: Pairwise Local Search ===")
    remaining = deadline - time.monotonic() - 150
    t_pw = min(remaining * 0.35, 500)

    if t_pw > 20:
        print(f"  Running pairwise search, budget={t_pw:.0f}s")
        # Split into two runs
        t_half = t_pw / 2.0
        a_opt, _ = pairwise_local_search(best_seq, time_limit=t_half, rng=rng,
                                         pairs_per_iter=300, search_steps=20)
        try_seq(a_opt, "pairwise1")

        rem = deadline - time.monotonic() - 120
        t_half2 = min(rem * 0.5, 200)
        if t_half2 > 15:
            a_opt, _ = pairwise_local_search(best_seq, time_limit=t_half2, rng=rng,
                                             pairs_per_iter=500, search_steps=15)
            try_seq(a_opt, "pairwise2")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: Coordinate Descent ===
    print("\n=== Phase 4: Coordinate Descent ===")
    remaining = deadline - time.monotonic() - 100
    t_cd = min(remaining * 0.3, 300)

    if t_cd > 15:
        print(f"  Running coordinate descent, budget={t_cd:.0f}s")
        a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd, rng=rng)
        try_seq(a_opt, "coord_descent")

    print(f"  Phase 4 best: {best_score:.8f}")

    # === Phase 5: Final Alternating Refinement ===
    print("\n=== Phase 5: Final Alternating Refinement ===")
    remaining = deadline - time.monotonic() - 40
    t_final = min(remaining * 0.8, 200)

    if t_final > 15:
        # Alternate short MD + short pairwise
        n_rounds = 4
        t_per_round = t_final / n_rounds
        for r in range(n_rounds):
            if time.monotonic() > deadline - 30:
                break
            # Short MD
            t_md_small = t_per_round * 0.5
            if t_md_small > 5:
                a_opt, _ = mirror_descent_optimize(best_seq, time_limit=t_md_small,
                                                   tau_start=0.001, tau_end=0.00001,
                                                   lr_start=0.3, momentum=0.9)
                try_seq(a_opt, f"final_md_{r}")

            # Short pairwise
            rem = deadline - time.monotonic() - 25
            t_pw_small = min(rem * 0.4, 30)
            if t_pw_small > 5:
                a_opt, _ = pairwise_local_search(best_seq, time_limit=t_pw_small, rng=rng,
                                                 pairs_per_iter=100, search_steps=20)
                try_seq(a_opt, f"final_pw_{r}")

            # Small perturbation restart
            if time.monotonic() < deadline - 40:
                f_pert = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * 0.001)
                try_seq(f_pert, f"final_pert_{r}")

    print(f"  Phase 5 best: {best_score:.8f}")

    # === Final: Try nearby lengths ===
    rem = deadline - time.monotonic() - 15
    if rem > 10 and best_seq is not None:
        n_cur = len(best_seq)
        for n_try in [n_cur - 8, n_cur + 8]:
            if n_try < 32 or rem < 8:
                continue
            px = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, n_cur), best_seq)
            px = np.maximum(px, 0)
            ps = px.sum()
            if ps > 1e-10:
                px = px / ps
                try_seq(px, f"interp_n{n_try}")
                a_opt, _ = mirror_descent_optimize(px, time_limit=min(rem * 0.3, 15),
                                                   tau_start=0.002, tau_end=0.00002,
                                                   lr_start=0.5, momentum=0.9)
                try_seq(a_opt, f"interp_opt_n{n_try}")
                rem = deadline - time.monotonic() - 10

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence([float(x) for x in best_seq]))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence([float(x) for x in best_seq]))
        print(f"\nFALLBACK SCORE: {best_score:.10f} (n=256)")

    return [float(x) for x in best_seq]
