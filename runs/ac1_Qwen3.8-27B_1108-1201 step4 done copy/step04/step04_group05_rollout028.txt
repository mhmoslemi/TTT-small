def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_correlation(a, w):
    """(a ⊛ w)[j] = sum_{m=0}^{n-1} a[m] * w[m+j] for j=0,...,n-1."""
    n = len(a)
    w_rev = w[::-1]
    size = n + len(w) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)[:size]
    return conv_result[2*n-2 : n-1 : -1]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_mirror_descent(f0, time_limit, T_start=5.0, T_end=200.0, 
                          lr=0.01, momentum=0.9, n_anneal_steps=200):

    best_obj = conv.max()

    it = 0

    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        it += 1
        progress = min(1.0, it / float(n_anneal))
        T = T_start * (T_end / T_start) ** progress

        conv = fast_convolve(f)
        w = softmax_stable(conv, T)
        grad = 2.0 * cross_correlation(f, w)

        v = momentum * v - cur_lr * grad
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj = conv_new.max()

        if obj < best_obj - 1e-16:
            best_obj = obj
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 400:
                cur_lr = max(cur_lr * 0.5, 1e-7)
                v *= 0.3
                no_improve = 0
            elif no_improve > 800:
                scale = 0.002 + 0.003 * (it // 2000)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                v = np.zeros(n)
                cur_lr = lr_init
                no_improve = 0
            elif no_improve > 1500:
                scale = 0.01 + 0.01 * (it // 5000)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                v = np.zeros(n)
                cur_lr = lr_init * 1.5
                no_improve = 0

        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SmoothDesc it={it} T={T:.1f} lr={cur_lr:.2e} obj={best_obj:.10f} elapsed={elapsed:.0f}s")

    return best_f, best_obj


def solve_lp(f, M):
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_frank_wolfe(f0, time_limit, seed=0, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    rng = np.random.default_rng(seed + 777)

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 2:
            break
        it += 1
        g = solve_lp(f, M * m_exp)
        if g is None:
            m_exp *= 1.15
            if m_exp > 6.0:
                m_exp = 1.0
            continue
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        diff = np.max(np.abs(g - f))
        if diff < 1e-11:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 4.0)
            if no_improve > 40:
                break
            continue

        best_alpha = 0.0
        best_val = M
        alphas = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9, 1.0]
        for alpha in alphas:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0.15 and best_alpha < 0.85:
            lo = best_alpha - 0.08
            hi = best_alpha + 0.08
            lo = max(0.0, lo)
            hi = min(1.0, hi)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs < 1e-15:
                f = best_f.copy()
            else:
                f = f / fs
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    print(f"    LP-FW it={it} obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)

        if no_improve > 50:
            strat = no_improve % 3
            if strat == 0:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            elif strat == 1:
                blend = 0.2 + 0.1 * (no_improve // 60)
                f = project_simplex((1 - blend) * best_f + blend * best_f[::-1])
            else:
                f = project_simplex(best_f * (1 + 0.05 * rng.random(n)))
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def coord_descent_polish(f0, time_limit, seed=0):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed + 4242)
    stagnant = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 1:
            break
        improved_this_pass = False
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            candidates = []
            for offset in [1, -1, 2, -2, n // 2, n // 3]:
                c = (j + offset) % n
                if c != j:
                    candidates.append(c)
            candidates.append(int(rng.integers(n)))
            for source in candidates:
                source = int(source) % n
                if source == j:
                    continue
                if f[source] < 1e-12:
                    continue
                for frac in [0.1, 0.2, 0.05]:
                    delta = f[source] * frac
                    if delta < 1e-14:
                        continue
                    f_new = f.copy()
                    f_new[j] += delta
                    f_new[source] -= delta
                    if f_new[source] < 0:
                        f_new[j] += f_new[source]
                        f_new[source] = 0.0
                    f_new = np.maximum(f_new, 0.0)
                    sn = f_new.sum()
                    if sn < 1e-15:
                        continue
                    f_new = f_new / sn
                    val = fast_convolve(f_new).max()
                    if val < best_obj - 1e-15:
                        f = f_new
                        best_obj = val
                        best_f = f.copy()
                        improved_this_pass = True
                        break
                if improved_this_pass:
                    break
            if improved_this_pass:
                break

        if not improved_this_pass:
            stagnant += 1
            if stagnant > 5:
                break
        else:
            stagnant = 0

    return best_f, best_obj


def generate_inits(n, parent, rng):
    inits = []
    x = np.arange(n, dtype=np.float64)
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())
    cos_t = np.cos(np.pi * x / (2.0 * (n - 1)))
    cos_t = np.maximum(cos_t, 0.01)
    inits.append(cos_t / cos_t.sum())
    peak = (np.exp(-((x - n / 4.0) ** 2) / (2.0 * (n / 8.0) ** 2)) +
            np.exp(-((x - 3.0 * n / 4.0) ** 2) / (2.0 * (n / 8.0) ** 2)))
    inits.append(peak / peak.sum())
    sine = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine / sine.sum())
    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    return inits


def try_sequence(a, label, state):
    a = np.maximum(a, 0.0)
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < state['best_score']:
        state['best_score'] = score
        state['best_seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    state = {'best_seq': None, 'best_score': float('inf')}

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.6f}")
            try_sequence(parent, "parent", state)
    except Exception:
        pass

    print("=== Phase 1: Initial Exploration (30% time) ===")
    phase1_end = time.monotonic() + (float(budget_s) * 0.30)

    if parent is not None:
        n_p = len(parent)
        t1 = min(80, phase1_end - time.monotonic() - 20)
        if t1 > 20:
            print(f"  Smooth descent on parent n={n_p}, t={t1:.0f}s")
            a_opt, obj = smooth_descent(parent, time_limit=t1, T_start=3.0,
                                        T_end=200.0, lr_init=0.008,
                                        momentum=0.85, n_anneal=200, seed=1)
            try_sequence(a_opt, "parent_smooth", state)

    lengths = []
    if parent is not None:
        n_p = len(parent)
        lengths = [n_p]
        for dn in [-2, -1, 1, 2, 4, 8, -4, -8, 16]:
            cand = n_p + dn
            if 128 <= cand <= 800 and cand not in lengths:
                lengths.append(cand)
    else:
        lengths = [256, 320, 384, 448, 512]

    for n in lengths[:4]:
        if time.monotonic() > phase1_end - 30:
            break
        inits = generate_inits(n, parent, rng)
        t_per = min(30, (phase1_end - time.monotonic() - 15) / max(1, len(inits[:2])))
        if t_per < 10:
            break
        for i in range(min(2, len(inits))):
            if time.monotonic() > phase1_end - 15:
                break
            t_alloc = min(t_per, phase1_end - time.monotonic() - 10)
            if t_alloc < 8:
                break
            a_opt, obj = smooth_descent(inits[i], time_limit=t_alloc, T_start=3.0,
                                        T_end=150.0, lr_init=0.006,
                                        momentum=0.85, n_anneal=100, seed=seed + i + n)
            try_sequence(a_opt, f"n{n}_i{i}_smooth", state)

    print(f"  Phase 1 best: {state['best_score']:.6f}")

    print("\n=== Phase 2: Deep Optimization (45% time) ===")
    phase2_end = time.monotonic() + (float(budget_s) * 0.45)

    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])
        print(f"  Deep optimization n={n_best}, budget={(phase2_end - time.monotonic()):.0f}s")

        t_sm = min(200, (phase2_end - time.monotonic()) * 0.4)
        if t_sm > 30:
            print(f"  Long smooth descent, t={t_sm:.0f}s")
            a_opt, obj = smooth_descent(state['best_seq'], time_limit=t_sm,
                                        T_start=5.0, T_end=800.0,
                                        lr_init=0.004, momentum=0.9,
                                        n_anneal=500, seed=100)
            try_sequence(a_opt, "deep_smooth", state)

        rem = phase2_end - time.monotonic() - 20
        if rem > 30:
            t_lp = min(150, rem * 0.5)
            if t_lp > 20:
                print(f"  LP Frank-Wolfe, t={t_lp:.0f}s")
                a_opt, obj = lp_frank_wolfe(state['best_seq'], time_limit=t_lp, seed=200, verbose=True)
                try_sequence(a_opt, "deep_lp", state)

        rem = phase2_end - time.monotonic() - 20
        if rem > 30:
            t_sm2 = min(rem, 150)
            if t_sm2 > 20:
                print(f"  Smooth descent round 2, t={t_sm2:.0f}s")
                a_opt, obj = smooth_descent(state['best_seq'], time_limit=t_sm2,
                                            T_start=10.0, T_end=1000.0,
                                            lr_init=0.002, momentum=0.92,
                                            n_anneal=400, seed=300)
                try_sequence(a_opt, "smooth_r2", state)

        pert_scales = [0.003, 0.008, 0.02]
        for ps in pert_scales:
            rem = phase2_end - time.monotonic() - 20
            if rem < 20:
                break
            t_alloc = min(rem * 0.4, 100)
            if t_alloc < 15:
                break
            noise = rng.standard_normal(n_best) * ps
            f_init = project_simplex(state['best_seq'] + noise)
            a_opt, obj = smooth_descent(f_init, time_limit=t_alloc, T_start=5.0,
                                        T_end=500.0, lr_init=0.005,
                                        momentum=0.88, n_anneal=200, seed=400 + int(ps * 10000))
            try_sequence(a_opt, f"pert_{ps}", state)

    print(f"  Phase 2 best: {state['best_score']:.6f}")

    print("\n=== Phase 3: Polishing (20% time) ===")
    phase3_end = time.monotonic() + (float(budget_s) * 0.20)

    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])

        rem = phase3_end - time.monotonic() - 20
        if rem > 15:
            t_cd = min(rem * 0.5, 120)
            if t_cd > 10:
                print(f"  Coordinate descent, t={t_cd:.0f}s")
                a_opt, obj = coord_descent_polish(state['best_seq'], time_limit=t_cd, seed=500)
                try_sequence(a_opt, "cd1", state)

        rem = phase3_end - time.monotonic() - 15
        if rem > 15:
            t_sm3 = min(rem, 100)
            if t_sm3 > 10:
                print(f"  Fine smooth descent, t={t_sm3:.0f}s")
                a_opt, obj = smooth_descent(state['best_seq'], time_limit=t_sm3,
                                            T_start=50.0, T_end=2000.0,
                                            lr_init=0.001, momentum=0.95,
                                            n_anneal=200, seed=600)
                try_sequence(a_opt, "fine_smooth", state)

        rem = phase3_end - time.monotonic() - 10
        if rem > 10:
            t_cd2 = min(rem, 60)
            if t_cd2 > 8:
                a_opt, obj = coord_descent_polish(state['best_seq'], time_limit=t_cd2, seed=700)
                try_sequence(a_opt, "cd2", state)

    print(f"  Phase 3 best: {state['best_score']:.6f}")

    print("\n=== Phase 4: Final Squeeze (5% time) ===")
    phase4_end = deadline

    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])

        rem = phase4_end - time.monotonic() - 5
        if rem > 10:
            t_lp_fin = min(rem * 0.6, 80)
            if t_lp_fin > 10:
                a_opt, obj = lp_frank_wolfe(state['best_seq'], time_limit=t_lp_fin, seed=800)
                try_sequence(a_opt, "final_lp", state)

        rem = phase4_end - time.monotonic() - 3
        if rem > 8:
            t_cd_fin = min(rem, 40)
            if t_cd_fin > 5:
                a_opt, obj = coord_descent_polish(state['best_seq'], time_limit=t_cd_fin, seed=900)
                try_sequence(a_opt, "final_cd", state)

        rem = phase4_end - time.monotonic() - 2
        if rem > 5:
            t_sm_fin = min(rem, 30)
            if t_sm_fin > 4:
                a_opt, obj = smooth_descent(state['best_seq'], time_limit=t_sm_fin,
                                            T_start=100.0, T_end=3000.0,
                                            lr_init=0.0005, momentum=0.97,
                                            n_anneal=100, seed=1000)
                try_sequence(a_opt, "final_smooth", state)

    if state['best_seq'] is None:
        state['best_seq'] = np.ones(256) / 256
        state['best_score'] = float(evaluate_sequence(state['best_seq'].tolist()))
        print(f"  Fallback uniform: {state['best_score']:.6f}")

    fs = float(evaluate_sequence(state['best_seq'].tolist()))
    if np.isfinite(fs):
        state['best_score'] = fs

    print(f"\nFINAL SCORE: {state['best_score']:.6f} (n={len(state['best_seq'])})")
    return [float(x) for x in state['best_seq']]
