```python
def solve_lp_minimize_max(f, n):
    """Solve: minimize M subject to (f*g)[k] <= M for all k, g >= 0, sum(g) = 1."""
    from scipy.optimize import linprog
    
    # Variables: g[0], ..., g[n-1], M
    # Total: n + 1 variables
    # Objective: minimize M -> c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Constraints: (f*g)[k] - M <= 0 for k = 0, ..., 2n-2
    # (f*g)[k] = sum_j f[j] * g[k-j] (with appropriate bounds)
    # So: sum_j f[j] * g[k-j] - M <= 0
    # In matrix form: A_ub @ x <= b_ub where x = [g, M]
    
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    b_ub = np.zeros(n_constraints)
    
    for k in range(2 * n - 1):
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                A_ub[k, j] = f[j]
        A_ub[k, -1] = -1.0  # coefficient of M
    
    # Equality constraint: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: g[i] >= 0, M is free (but will be >= 0 automatically)
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                  method='highs', options={'presolve': True})
    
    if res.success:
        g = res.x[:n]
        g = np.maximum(g, 0)
        g_sum = g.sum()
        if g_sum > 1e-15:
            g = g / g_sum
        return g, res.x[-1]
    return None, np.inf
```

The objective we care about is max(a*a). After each full iteration (optimizing a then b), we have new a and b. The "effective" objective is max(a*b) (the mixed convolution). At convergence, a ≈ b, so max(a*b) ≈ max(a*a).

I think the best approach is:
1. Run alternating LP until convergence (a and b stabilize)
2. The result is a good local optimum
3. Polish with subgradient methods

Let me also consider: should I try to directly minimize max(a*a) using the LP formulation where both factors are the same? That would be:
- Minimize M subject to sum_j g[j]*g[k-j] <= M for all k, g >= 0, sum(g) = 1

This is a QP (quadratic constraints), not an LP. So we can't solve it directly as an LP.

The alternating approach linearizes one factor at a time, making each subproblem an LP. This is the key advantage.

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using alternating LP optimization
with subgradient polishing. The key innovation is using direct LP minimization
of the max-convolution (rather than the fix-M approach) and alternating between
two factors for coordinate descent on the non-convex problem.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def mixed_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_for_factor(fixed, n):
    """
    Solve: minimize M subject to (fixed * g)[k] <= M for all k,
           g >= 0, sum(g) = 1.
    Returns (g, M) or (None, inf) if infeasible.
    """
    # Variables: g[0], ..., g[n-1], M -> total n+1
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    # Inequality constraints: (fixed * g)[k] - M <= 0 for k = 0, ..., 2n-2
    n_cons = 2 * n - 1
    A_ub = np.zeros((n_cons, n + 1))
    
    for k in range(n_cons):
        # (fixed * g)[k] = sum_{j: max(0,k-n+1) <= j <= min(n-1,k)} fixed[j] * g[k-j]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = fixed[j]
        A_ub[k, n] = -1.0  # -M term
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds
    bounds = [(0, None)] * n + [(0, None)]  # M >= 0
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=np.zeros(n_cons),
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})
        if res.success:
            g = res.x[:n]
            g = np.maximum(g, 0)
            g_sum = g.sum()
            if g_sum > 1e-15:
                g = g / g_sum
            return g, float(res.x[n])
        elif res.status == 2:  # optimum found but not accurate
            g = res.x[:n]
            g = np.maximum(g, 0)
            g_sum = g.sum()
            if g_sum > 1e-15:
                g = g / g_sum
            return g, float(res.fun)
    except Exception:
        pass
    return None, np.inf


def alternating_lp(f_init, time_limit, verbose=True):
    """
    Alternating LP optimization.
    Maintains two factors a and b, alternately optimizing each via LP.
    """
    a = f_init.astype(np.float64).copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    b = a.copy()
    n = len(a)
    
    # Initial objective
    conv = fast_convolve(a)
    M = conv.max()
    best_M = M
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        
        # Step 1: Optimize a given b
        a_new, M1 = solve_lp_for_factor(b, n)
        if a_new is None:
            break
        
        # Step 2: Optimize b given a_new
        b_new, M2 = solve_lp_for_factor(a_new, n)
        if b_new is None:
            break
        
        # Evaluate the auto-convolution of a_new
        conv_new = fast_convolve(a_new)
        M_auto = conv_new.max()
        
        # Also evaluate cross-convolution
        conv_cross = mixed_convolve(a_new, b_new)
        M_cross = conv_cross.max()
        
        if verbose and it % 10 == 0:
            print(f"  Alt-LP it={it}: M_auto={M_auto:.8f}, M_cross={M_cross:.8f}, best={best_M:.8f}")
        
        if M_auto < best_M - 1e-12:
            best_M = M_auto
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Update
        a = a_new
        b = b_new
        
        # Convergence check
        if np.max(np.abs(a - b)) < 1e-8 and M_auto >= best_M - 1e-12:
            no_improve += 5
        
        if no_improve > 30:
            # Try a perturbation
            rng = np.random.default_rng(it * 7 + 3)
            a_pert = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            b = a_pert.copy()
            a = a_pert.copy()
            no_improve = 0
            M = fast_convolve(a).max()
    
    return best_a, best_M


def compute_subgradient(f, conv, top_k=64, temp=0.001):
    """Compute subgradient of max(conv) with softmax weighting over top peaks."""
    n = len(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= wsum
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            sg[j] += 2.0 * w[i] * f[k - j]
    
    return sg


def sg_polish(f0, time_limit, lr=0.001, mom=0.95, top_k=100, temp=0.0001):
    """Fine subgradient polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temp
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1:
            break
        
        sg = compute_subgradient(f, conv, top_k=top_k, temp=cur_temp)
        v = mom * v - cur_lr * sg
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj = conv_new.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            f = f_new
            conv = conv_new
            no_improve = 0
        else:
            no_improve += 1
            # Partial step
            f_half = project_simplex(f + 0.5 * v)
            conv_half = fast_convolve(f_half)
            if conv_half.max() < best_obj - 1e-15:
                best_obj = conv_half.max()
                best_f = f_half.copy()
                f = f_half
                conv = conv_half
                no_improve = 0
        
        cur_lr *= 0.9998
        cur_temp *= 1.001
        
        if no_improve > 300:
            rng = np.random.default_rng(it + 100)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            conv = fast_convolve(f)
            cur_lr = lr * 0.5
            no_improve = 0
    
    return best_f, best_obj
```
