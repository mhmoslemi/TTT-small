import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, conv, k):
    """Compute subgradient of max(conv) using top-k weighted peaks."""
    n = len(f)
    actual_k = min(k, len(conv))
    peaks = np.argpartition(conv, -actual_k)[-actual_k:]
    pv = conv[peaks]
    total = pv.sum()
    if total < 1e-30:
        w = np.zeros(actual_k)
        w[np.argmax(pv)] = 1.0
    else:
        w = pv / total
    sg = np.zeros(n)
    for i in range(actual_k):
        if w[i] < 1e-15:
            continue
        pos = int(peaks[i])
        js = max(0, pos - n + 1)
        je = min(n - 1, pos)
        idx = np.arange(js, je + 1)
        sg[idx] += w[i] * f[pos - idx]
    return 2.0 * sg

    v = np.zeros(n)

    lr = 2.0 / (3.0 * t)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

        it += 1

def solve_sparse_lp(f, M, n, tol_frac=0.05):
    """Solve LP with only near-active constraints for speed."""
    conv_current = fast_convolve(f)
    # Only include constraints within tol_frac of M
    active_mask = conv_current >= M * (1.0 - tol_frac)
    if not active_mask.any():
        active_mask = np.zeros(len(conv_current), dtype=bool)
        active_mask[np.argmax(conv_current)] = True

    n_constraints = int(active_mask.sum())
    # Limit to at most 500 constraints for speed
    if n_constraints > 500:
        # Pick the top 500
        thresholds = np.sort(conv_current)[::-1][:500]
        min_thresh = thresholds[-1]
        active_mask = conv_current >= min_thresh
        n_constraints = int(active_mask.sum())

    rows = np.where(active_mask)[0]
    c = -np.ones(n)
    A_ub = np.zeros((n_constraints, n))
    for i, k in enumerate(rows):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        j_idx = np.arange(js, je + 1)
        A_ub[i, j_idx] = f[k - j_idx]
    b_ub = np.full(n_constraints, M)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-8})
    if res.success:
        return res.x
    # Fallback: try with relaxed M

        res = linprog(c, A_ub=A_ub, b_ub=np.full(n_constraints, M * relax),
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x
    return None

            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.08)
            hi = min(1.0, best_alpha + 0.08)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_relax = 1.0
            tol_frac = max(0.02, tol_frac * 0.95)
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.12f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 60:
            rng = np.random.default_rng(it * 41 + 17)
            scale = 0.002 + 0.003 * min(no_improve // 80, 4)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.0
            no_improve = 0

    return best_f, best_obj


def coordinate_descent_fast(f0, time_limit, verbose=False):
    """Fast coordinate descent with multiplier-based updates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    conv = fast_convolve(f)
    best_obj = conv.max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        conv = fast_convolve(f)

        # Identify which coordinates matter most for the peak
        peak_pos = int(np.argmax(conv))
        js = max(0, peak_pos - n + 1)
        je = min(n - 1, peak_pos)
        peak_coords = list(range(js, je + 1))

        # Shuffle non-peak coords first, then do peak coords
        other_coords = [j for j in range(n) if j not in peak_coords]
        rng = np.random.default_rng(passes * 31)
        rng.shuffle(other_coords)
        ordered = other_coords[:n // 2] + peak_coords

        for j in ordered:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-10:
                if j not in peak_coords:
                    continue

            best_j_val = old_val
            best_j_obj = best_obj
            cur_conv = fast_convolve(f)
            cur_peak = cur_conv.max()

            multipliers = [0.5, 0.7, 0.85, 1.2, 1.4, 1.8]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 5:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def parametric_init(n, family='cosine', alpha=1.0):
    """Generate a parametric initial sequence."""
    if family == 'uniform':
        return np.ones(n) / n
    elif family == 'gaussian':
        x = np.linspace(0, 1, n)
        center = 0.5
        sigma = 0.25
        f = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        f = np.maximum(f, 0)
        return f / f.sum()
    elif family == 'cosine':
        x = np.linspace(0, 1, n)
        f = np.cos(np.pi * (x - 0.5)) ** (2 * alpha)
        f = np.maximum(f, 0)
        ss = f.sum()
        if ss < 1e-15:
            return np.ones(n) / n
        return f / ss
    elif family == 'power':
        x = np.linspace(0, 1, n)
        d = np.abs(x - 0.5) * 2
        f = (1 - d) ** alpha
        f = np.maximum(f, 0)
        ss = f.sum()
        if ss < 1e-15:
            return np.ones(n) / n
        return f / ss
    elif family == 'bimodal':
        x = np.linspace(0, 1, n)
        f = np.exp(-((x - 0.3) / 0.15) ** 2) + np.exp(-((x - 0.7) / 0.15) ** 2)
        f = np.maximum(f, 0)
        return f / f.sum()
    elif family == 'w_shape':
        x = np.linspace(0, 1, n)
        f = 1.0 + 0.3 * np.cos(2 * np.pi * x) - 0.5 * np.abs(x - 0.5)
        f = np.maximum(f, 0)
        ss = f.sum()
        if ss < 1e-15:
            return np.ones(n) / n
        return f / ss
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining Nesterov PGD, sparse LP-FW,
    and coordinate descent with multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        if a.sum() < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    parent_n = 0
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_n = len(parent)
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={parent_n}, score={p_score:.8f}")
            best_score = min(best_score, p_score)
            best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Multi-start with parametric inits (120s) ===
    print("=== Phase 1: Multi-start ===")
    t1_end = time.monotonic() + min(120, (deadline - time.monotonic()) * 0.1)

    candidate_ns = [512, 768, 1024]
    if parent is not None and parent_n in candidate_ns:
        candidate_ns = [parent_n] + [x for x in candidate_ns if x != parent_n]

    for target_n in candidate_ns:
        if time.monotonic() > t1_end:
            break
        rem_in_phase = t1_end - time.monotonic()
        if rem_in_phase < 15:
            break

        best_for_n = None
        best_obj_for_n = float('inf')

        if parent is not None:
            f_init = interpolate_sequence(parent, target_n)
            obj = fast_convolve(f_init).max()
            if obj < best_obj_for_n:
                best_obj_for_n = obj
                best_for_n = f_init

        families = ['cosine', 'gaussian', 'power', 'bimodal', 'w_shape', 'uniform']
        for fam in families:
            if time.monotonic() > t1_end:
                break
            for alpha in ([1.0, 2.0, 3.0] if fam in ('cosine', 'power') else [1.0]):
                f_init = parametric_init(target_n, fam, alpha)
                obj = fast_convolve(f_init).max()
                if obj < best_obj_for_n:
                    best_obj_for_n = obj
                    best_for_n = f_init

        if best_for_n is not None:
            t_quick = min(rem_in_phase * 0.8, 30)
            a_opt, _ = nesterov_pgd(best_for_n, time_limit=t_quick, top_k_init=10, top_k_max=80)
            score = try_seq(a_opt, f"init_n{target_n}")

    # === Phase 2: Nesterov PGD on best (400s) ===
    print(f"\n=== Phase 2: Nesterov PGD (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 600
        t2 = max(60, min(rem * 0.5, 400))
        print(f"  Time allocated: {t2:.0f}s")
        n_best = len(best_seq)

        # Use progressively larger top_k
        for run in range(3):
            rem = deadline - time.monotonic() - (600 - run * 100)
            if rem < 20:
                break
            t_alloc = min(rem * 0.5, 150)
            top_k_init = 20 + run * 30
            top_k_max = 100 + run * 80
            a_opt, obj = nesterov_pgd(best_seq, time_limit=t_alloc,
                                      top_k_init=top_k_init, top_k_max=top_k_max, verbose=False)
            try_seq(a_opt, f"nesterov_{run}")

    # === Phase 3: LP-FW optimization (500s) ===
    print(f"\n=== Phase 3: LP-FW (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        t3 = max(60, min(rem * 0.6, 500))
        print(f"  Time allocated: {t3:.0f}s")

        # Run LP-FW in chunks with different inits
        chunks = 3
        for chunk in range(chunks):
            rem = deadline - time.monotonic() - (300 - chunk * 50)
            if rem < 20:
                break
            t_alloc = min(rem / (chunks - chunk), 200)

            if chunk == 0:
                f_init = best_seq.copy()
            elif chunk == 1:
                noise = rng.standard_normal(len(best_seq)) * 0.003
                f_init = project_simplex(best_seq + noise)
            else:
                # Mix with uniform
                f_init = 0.7 * best_seq + 0.3 * (np.ones(len(best_seq)) / len(best_seq))
                f_init = project_simplex(f_init)

            a_opt, obj = lp_fw_optimize(f_init, time_limit=t_alloc, verbose=(chunk == 0))
            try_seq(a_opt, f"lpfw_{chunk}")

    # === Phase 4: Try larger n by interpolation (200s) ===
    print(f"\n=== Phase 4: Larger n exploration (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        for target_n in [1024, 1280, 1536]:
            if time.monotonic() > deadline - 200:
                break
            if len(best_seq) == target_n:
                continue
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min((deadline - time.monotonic() - 200) * 0.4, 80)
            if t_alloc < 15:
                break
            print(f"  Trying n={target_n}, t={t_alloc:.0f}s")
            a_opt, obj = nesterov_pgd(f_interp, time_limit=t_alloc,
                                       top_k_init=30, top_k_max=200)
            score = try_seq(a_opt, f"interp_n{target_n}")
            if score < best_score:
                # Follow up with LP-FW on the new best
                rem2 = deadline - time.monotonic() - 150
                if rem2 > 30:
                    a_opt2, _ = lp_fw_optimize(a_opt, time_limit=min(rem2, 60))
                    try_seq(a_opt2, f"lpfw_interp_n{target_n}")

    # === Phase 5: Coordinate descent polishing (250s) ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        t5 = max(30, min(rem, 250))
        print(f"  Time allocated: {t5:.0f}s")
        a_opt, obj = coordinate_descent_fast(best_seq, time_limit=t5)
        try_seq(a_opt, "cd")

    # === Phase 6: Final fine polish (80s) ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 30:
            # Very fine Nesterov
            t6 = min(rem * 0.6, 50)
            a_opt, _ = nesterov_pgd(best_seq, time_limit=t6,
                                     top_k_init=50, top_k_max=300)
            try_seq(a_opt, "final_nest")

            rem = deadline - time.monotonic() - 10
            if rem > 20:
                # One more LP-FW
                a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem, 40))
                try_seq(a_opt, "final_lpfw")

            rem = deadline - time.monotonic() - 5
            if rem > 10:
                # Tiny perturbations + quick polish
                n_best = len(best_seq)
                for trial in range(2):
                    rem = deadline - time.monotonic() - 5
                    if rem < 8:
                        break
                    noise = rng.standard_normal(n_best) * 0.0008
                    f_pert = project_simplex(best_seq + noise)
                    a_opt, _ = nesterov_pgd(f_pert, time_limit=min(rem, 15),
                                            top_k_init=50, top_k_max=200)
                    try_seq(a_opt, f"perturb_{trial}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
