def fast_convolve(a):
    """Compute full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = convolve(a, g) (full linear convolution)."""
    n = len(a)
    m = 2 * n - 1
    C = np.zeros((m, n))
    for j in range(n):
        start = j
        C[start:start + n, j] = a
    return C


def obj_value(a):
    """Compute the objective value (scale-invariant)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    conv = fast_convolve(a_norm)
    return 2.0 * n * conv.max()


def spectral_smooth(a, alpha=0.3, n_iter=50, time_limit=None):
    """
    Gerchberg-Saxton style spectral flattening.
    Compresses the dynamic range of |FFT(a)|^2 to reduce autocorrelation peaks.
    """
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    fft_size = 1 << (2 * n - 2).bit_length()

    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()

    t0 = time.monotonic()
    for it in range(n_iter):
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break

        A = np.fft.fft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        mag2 = mag ** 2

        # Compress dynamic range: mag_new = mag^(1-alpha) * mag_max^alpha
        mag_max = mag2.max()
        if mag_max < 1e-30:
            break
        mag2_new = (mag2 ** (1 - alpha)) * (mag_max ** alpha)
        mag_new = np.sqrt(mag2_new)

        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        conv = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a_new.copy()

        # Adaptive alpha: reduce as we converge
        alpha = max(0.05, alpha * 0.98)

    return best_a


def lp_direction(a, target_M):
    """
    Solve LP: max sum(g) s.t. C_a @ g <= target_M, g >= 0.
    Returns (g, g_sum, success).
    """
    n = len(a)
    C = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, target_M)
    bounds = [(0.0, None)] * n

    try:
        result = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 10})
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g, g_sum, True
    except Exception:
        pass
    return a.copy(), 1.0, False


def lp_step(a, conv, target_M=None):
    """
    One LP-based step. Returns (a_new, improved).
    Uses line search over candidate step sizes.
    """
    n = len(a)
    M = conv.max()
    if target_M is None:
        target_M = M

    g, g_sum, success = lp_direction(a, target_M)

    if not success or g_sum <= 1.0 + 1e-10:
        return a.copy(), False

    # Direction: move from a towards g (both normalized to sum 1)
    g_norm = g / g_sum
    # a is already normalized to sum 1

    # Line search: try multiple t values
    best_a = a.copy()
    best_obj = M

    candidates = [0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03]
    for t in candidates:
        a_try = (1 - t) * a + t * g_norm
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        conv_try = fast_convolve(a_try)
        obj_try = conv_try.max()
        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_a = a_try.copy()
            break  # First improvement in geometric sequence

    if best_obj < M - 1e-14:
        return best_a, True
    return a.copy(), False


def subgradient_optimize(a0, time_limit, top_k=50, temp_init=0.002, lr_init=0.005):
    """
    Nesterov-accelerated subgradient descent on the (non-smooth) max of conv.
    Uses softmax weighting over top-k peaks.
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    y = a.copy()
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = 0.9
    no_improve = 0
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        iteration += 1

        # Nesterov lookahead
        y = a + mom * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y

        conv_y = fast_convolve(y)

        # Compute softmax-weighted subgradient
        peaks = np.argpartition(conv_y, -top_k)[-top_k:]
        peak_vals = conv_y[peaks]
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum

        # Gradient of max_k (a*a)[k] w.r.t. a is a weighted sum of reflections
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idxs = np.arange(j_start, j_end + 1)
            grad[idxs] += w[idx] * y[k - idxs]

        # Update
        v = mom * v - lr * grad
        a_new = a - v  # Note: we're minimizing, so subtract gradient direction
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
            v = np.zeros(n)
            continue

        conv = fast_convolve(a)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive updates
        lr *= 0.999
        temp *= 1.002
        mom = min(0.96, mom + 0.0001)

        if no_improve >= 300:
            break

    return best_a, best_obj


def coordinate_polish(a0, time_limit):
    """Random coordinate swaps to reduce peak."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        # Try a batch of random pairwise transfers
        batch = max(100, int(200000 / n))
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit - 0.1:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            for delta in [0.002, 0.005, 0.01, 0.02, -0.002, -0.005, -0.01, -0.02]:
                if a[i] + delta < -1e-12 or a[j] - delta < -1e-12:
                    continue
                a_try = a.copy()
                a_try[i] = max(0.0, a_try[i] + delta)
                a_try[j] = max(0.0, a_try[j] - delta)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                # Quick check without full reconvolution: use old conv as upper bound
                # Only do full conv if likely to improve
                conv_try = fast_convolve(a_try)
                obj_try = conv_try.max()
                if obj_try < best_obj - 1e-12:
                    a = a_try
                    best_obj = obj_try
                    best_a = a.copy()
                    conv = conv_try
                    no_improve = 0
                    break
            else:
                no_improve += 1
                if no_improve > 500:
                    break
        if no_improve > 500:
            break

    return best_a


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    inits['sine1'] = np.sin(np.pi * x / n) ** 2 + 0.01
    inits['sine1'] /= inits['sine1'].sum()

    inits['sine2'] = np.sin(2 * np.pi * x / n) ** 2 + 0.01
    inits['sine2'] /= inits['sine2'].sum()

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    rc += 0.01
    inits['raised_cos'] = rc / rc.sum()

    # Bimodal (edges heavy)
    edge = max(1, n // 6)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Triangular (peak in center)
    tri = 1.0 - 2.0 * np.abs(x - n / 2) / n
    tri = np.maximum(tri, 0.01)
    inits['triangular'] = tri / tri.sum()

    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.1)

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            p = p / ps
            if len(p) == n:
                inits['parent'] = p
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent'] = pp / pps

    return inits


def lp_refine(a0, time_limit):
    """
    Iterative LP-based refinement with adaptive targets and perturbation.
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = M
    best_a = a.copy()

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    perturb_count = 0
    target_factor = 1.0

    while time.monotonic() - t0 < time_limit:
        iteration += 1

        # Try LP step with current target
        target = M * target_factor
        a_new, improved = lp_step(a, conv, target)

        if improved:
            a = a_new
            conv = fast_convolve(a)
            M = conv.max()
            if M < best_obj - 1e-14:
                best_obj = M
                best_a = a.copy()
            no_improve = 0
            # Gradually tighten target
            target_factor = max(0.95, target_factor * 0.999)
        else:
            no_improve += 1
            # Loosen target to find a new direction
            target_factor = min(1.05, target_factor * 1.02)

        # Periodic perturbation to escape local minima
        if no_improve >= 50 + 20 * perturb_count:
            rng = np.random.default_rng(iteration * 31 + 17 + perturb_count * 7)
            scale = 0.002 + 0.003 * min(perturb_count, 10)
            noise = rng.standard_normal(n) * scale
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            perturb_count += 1
            target_factor = 1.0

            if perturb_count > 30:
                break

        if iteration % 50 == 0:
            print(f"    LP iter {iteration}, obj={best_obj:.8f}, "
                  f"perturb={perturb_count}, t={time.monotonic()-t0:.1f}s")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Hybrid spectral + LP + subgradient search.
    Tries multiple lengths and initializations, then refines the best.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass

    # =========================================================
    # Phase 1: Spectral scan across lengths and inits (fast)
    # =========================================================
    print("=== Phase 1: Spectral Scan ===")
    t_start = time.monotonic()
    phase1_limit = 150.0

    lengths = [256, 512, 768, 1024]
    scan_results = []

    for n in lengths:
        if time.monotonic() > deadline - 800:
            break
        inits = generate_inits(n, parent, rng)
        # Prioritize informative inits
        priority = ['parent', 'sine1', 'raised_cos', 'uniform', 'sine2', 'bimodal', 'triangular', 'random']
        init_names = [nm for nm in priority if nm in inits][:5]

        t_per = max(5.0, (phase1_limit - 5) / (len(lengths) * 4))

        for name in init_names:
            if time.monotonic() > deadline - 700:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 400)
            if t_alloc < 3:
                break

            a0 = inits[name]
            a_opt = spectral_smooth(a0, alpha=0.35, n_iter=30, time_limit=t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Spec n={n}, {name}: {score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 2: LP deep optimization on best candidates
    # =========================================================
    print("=== Phase 2: LP Deep Optimization ===")
    t_start2 = time.monotonic()

    # Select top candidates (unique by length)
    top_configs = []
    seen_lengths = set()
    for score, n, name, a_opt in scan_results:
        if n not in seen_lengths:
            top_configs.append((n, a_opt))
            seen_lengths.add(n)
        if len(top_configs) >= 3:
            break

    # Always include parent if available
    if parent is not None:
        n_p = len(parent)
        if n_p not in seen_lengths:
            top_configs.append((n_p, parent / parent.sum()))
            seen_lengths.add(n_p)

    remaining = deadline - time.monotonic() - 350.0
    n_cands = max(1, len(top_configs))
    t_per_lp = max(40.0, remaining * 0.55 / n_cands)

    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 300:
            break
        t_alloc = min(t_per_lp, deadline - time.monotonic() - 250)
        if t_alloc < 15:
            break

        print(f"  LP candidate {i}: n={n}, t_alloc={t_alloc:.0f}s")
        a_opt, obj = lp_refine(a0, time_limit=t_alloc)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  LP {i} (n={n}): {score:.6f} *NEW BEST*")
            else:
                print(f"  LP {i} (n={n}): {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 3: Subgradient + LP alternating on best
    # =========================================================
    print("=== Phase 3: Alternating Subgrad + LP ===")
    t_start3 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 200.0
        n_best = len(best_sequence)

        # Alternate between subgradient and LP
        a_curr = best_sequence.copy()
        for round_i in range(6):
            if time.monotonic() > deadline - 150:
                break
            t_alloc = max(10.0, min(remaining / (6 - round_i) * 0.7, deadline - time.monotonic() - 100))
            if t_alloc < 8:
                break

            # Subgradient phase
            a_sg, obj_sg = subgradient_optimize(a_curr, time_limit=t_alloc * 0.5,
                                                 top_k=60, temp_init=0.001, lr_init=0.003)
            score_sg = float(evaluate_sequence(a_sg.tolist()))
            if np.isfinite(score_sg) and score_sg < best_score:
                best_score = score_sg
                best_sequence = a_sg.copy()
                a_curr = a_sg.copy()
                print(f"  Subgrad round {round_i}: {score_sg:.6f} *NEW BEST*")

            # LP phase
            if time.monotonic() < deadline - 100:
                t_lp = max(8.0, min(t_alloc * 0.5, deadline - time.monotonic() - 80))
                if t_lp > 5:
                    a_lp, obj_lp = lp_refine(a_curr, time_limit=t_lp)
                    score_lp = float(evaluate_sequence(a_lp.tolist()))
                    if np.isfinite(score_lp) and score_lp < best_score:
                        best_score = score_lp
                        best_sequence = a_lp.copy()
                        a_curr = a_lp.copy()
                        print(f"  LP round {round_i}: {score_lp:.6f} *NEW BEST*")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 4: Final polishing
    # =========================================================
    print("=== Phase 4: Final Polishing ===")
    t_start4 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20.0

        # Coordinate polish
        t_cd = min(remaining * 0.4, 60.0)
        if t_cd > 5:
            a_opt = coordinate_polish(best_sequence, time_limit=t_cd)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  CoordDesc: {score:.6f} *NEW BEST*")

        # Final short subgradient runs with varied params
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 10:
            configs = [
                (0.002, 0.0005, 80, 0.92),
                (0.005, 0.002, 40, 0.88),
                (0.001, 0.0002, 100, 0.95),
            ]
            for ci, (lr, temp, tk, mom_unused) in enumerate(configs):
                if time.monotonic() > deadline - 5:
                    break
                t_sg = min(remaining / (len(configs) - ci) * 0.7, 30.0)
                if t_sg < 5:
                    break
                a_opt, _ = subgradient_optimize(best_sequence, time_limit=t_sg,
                                                 top_k=tk, temp_init=temp, lr_init=lr)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  SG final {ci}: {score:.6f} *NEW BEST*")

        # Final LP micro-polish
        if time.monotonic() < deadline - 5:
            t_final = min(15.0, deadline - time.monotonic() - 3)
            if t_final > 5:
                a_opt, _ = lp_refine(best_sequence, time_limit=t_final)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Final LP: {score:.6f} *NEW BEST*")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
