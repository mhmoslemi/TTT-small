from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp(f, M):
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

We have:
- `a` of length `n`
- `w` of length `2n-1` (the softmax weights over the convolution)
- We want `h[m] = sum_{k=0}^{2n-2} w[k] * a[k-m]` for `m = 0, ..., n-1`
  where `a[i] = 0` for `i < 0` or `i >= n`.

So `h[m] = sum_{k=max(0,m)}^{min(2n-2, m+n-1)} w[k] * a[k-m]`
Let `j = k - m`, so `k = j + m`:
`h[m] = sum_{j=0}^{n-1} w[j+m] * a[j]` (with `w[i] = 0` for `i >= 2n-1`)

This is a "sliding dot product" - for each `m`, we take the slice `w[m:m+n]` and dot it with `a`.

Using FFT:
Define `w_rev[i] = w[2n-2-i]` for `i = 0, ..., 2n-2`.
Then `w_rev[j+m'] = w[2n-2-j-m']`.

The convolution `(a * w_rev)[k] = sum_j a[j] * w_rev[k-j]` (with zero padding).
`= sum_j a[j] * w[2n-2-k+j]`

Setting `k = 2n-2-m`:
`(a * w_rev)[2n-2-m] = sum_j a[j] * w[m+j]`

This is exactly `h[m]`! ✓

So: `h[m] = (a * w_rev)[2n-2-m]`

"""
Hybrid optimization combining mirror descent (multiplicative weights) with
temperature annealing as the primary engine, LP-based Frank-Wolfe for
structured refinement, and subgradient polishing. The mirror descent approach
achieves O(n log n) per iteration via FFT-based gradient computation, enabling

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_mw_gradient(a, w):
    """
    Compute gradient of smoothed max-convolution.
    grad[m] = a[m] + h[m] where h[m] = sum_k w[k]*a[k-m]
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    # h[m] = (a * w_rev)[2n-2-m]
    w_rev = w[::-1]
    conv_len = n + len(w_rev) - 1  # = 3n - 2
    fft_size = 1 << (conv_len - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    # h[m] = conv_result[2n-2-m] for m=0,...,n-1
    h = conv_result[2 * n - 2 : n - 1 : -1]
    return a + h


def md_optimize(a0, time_limit, t_start=5.0, t_end=200.0,
                eta_start=0.3, eta_end=0.01, verbose=False):
    """Mirror descent with temperature annealing on smoothed max-convolution."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    g = fast_convolve(a)
    best_obj = g.max()
    best_a = a.copy()

    t0 = time.monotonic()
    total_iters = 0
    no_improve = 0
    rng = np.random.default_rng(12345 + n)
    restart_count = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        # Anneal temperature and learning rate
        t = t_start + (t_end - t_start) * progress
        eta = eta_start * (eta_end / eta_start) ** progress

        # Compute convolution and check improvement
        g = fast_convolve(a)
        obj = g.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Restart if stuck
        if no_improve > 800:
            scale = 0.005 + 0.005 * (restart_count % 5)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            no_improve = 0
            restart_count += 1
            g = fast_convolve(a)
            continue

        # Compute softmax weights over convolution peaks
        scaled = t * g
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros_like(g)
            w[int(g.argmax())] = 1.0
        else:
            w /= ws

        # Compute gradient via FFT
        grad = compute_mw_gradient(a, w)

        # Multiplicative weights update (mirror descent with entropy)
        log_a = np.log(a + 1e-300)
        log_a_new = log_a - eta * grad
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-30:
            a = project_simplex(a - 0.1 * eta * grad)
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
        else:
            a = a_new / a_new_sum
            # Ensure no values are exactly zero (maintain full support)
            a = np.maximum(a, 1e-12)
            a = a / a.sum()

        total_iters += 1
        if verbose and total_iters % 5000 == 0:
            print(f"        MD iter={total_iters}, t={t:.1f}, eta={eta:.5f}, obj={best_obj:.10f}")

    return best_a, best_obj

def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def sg_polish(f0, time_limit, lr=0.002, mom=0.94, top_k=30, temp=0.00003):
    """Subgradient descent with multi-peak softmax weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    rng_local = np.random.default_rng(7777)
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[int(np.argmax(pv))] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9998
        cur_temp *= 1.003

        if no_improve > 400:
            f = project_simplex(best_f + rng_local.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0

    return best_f, best_obj


def local_search_polish(f0, time_limit, n_trials=50):
    """Random coordinate perturbation search for fine improvements."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(8888 + n)

    while time.monotonic() - t0 < time_limit:
        improved = False
        for _ in range(n_trials):
            if time.monotonic() - t0 > time_limit - 1:
                break
            # Pick two random coordinates and transfer mass
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            for frac in [0.01, 0.03, 0.08, 0.15, 0.3]:
                delta = f[j] * frac
                if delta < 1e-15:
                    continue
                f_new = f.copy()
                f_new[i] += delta
                f_new[j] -= delta
                f_new = np.maximum(f_new, 0.0)
                sn = f_new.sum()
                if sn < 1e-15:
                    continue
                f_new = f_new / sn
                val = fast_convolve(f_new).max()
                if val < best_obj - 1e-15:
                    f = f_new
                    best_obj = val
                    best_f = f.copy()
                    improved = True
                    break
        if not improved:
            # Try neighborhood perturbations
            noise = rng.standard_normal(n) * 0.003
            f_try = project_simplex(best_f + noise)
            val = fast_convolve(f_try).max()
            if val < best_obj - 1e-15:
                f = f_try
                best_obj = val
                best_f = f.copy()
                improved = True
            if not improved:
                break

    return best_f, best_obj

    for n in lengths:
        if time.monotonic() > phase1_end - 30:
            break

        inits = []
        x = np.arange(n, dtype=np.float64)

        # Uniform
        inits.append(np.ones(n) / n)
        # Raised cosine
        rc = 0.5 * (1 + np.cos(2 * np.pi * x / n))
        rc = np.maximum(rc, 0.01)
        inits.append(rc / rc.sum())
        # Two-peak
        pk = (np.exp(-((x - n / 4.0)**2) / (2.0 * (n / 6.0)**2)) +
              np.exp(-((x - 3.0 * n / 4.0)**2) / (2.0 * (n / 6.0)**2)))
        inits.append(pk / pk.sum())
        # Random Dirichlet
        inits.append(rng.dirichlet(np.ones(n) * 0.4))

        # Blend with parent if available
        if parent is not None:
            px = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            px = np.maximum(px, 0)
            px = px / px.sum()
            inits.append(0.5 * (inits[0] + px) / (0.5 * (inits[0] + px)).sum())

        n_try = min(3, len(inits))
        t_per = max(15, (phase1_end - time.monotonic() - 20) / max(1, (len(lengths) - lengths.index(n)) * n_try))
        t_per = min(t_per, 60)

        for i in range(n_try):
            if time.monotonic() > phase1_end - 10:
                break
            t_alloc = min(t_per, phase1_end - time.monotonic() - 5)
            if t_alloc < 8:
                break
            a_opt, obj = md_optimize(inits[i], time_limit=t_alloc,
                                     t_start=3.0, t_end=100.0,
                                     eta_start=0.4, eta_end=0.02)
            try_sequence(a_opt, f"md_n{n}_i{i}")

    print(f"  Phase 1 best: {state['best_score']:.6f}")

    # === Phase 2: Deep optimization ===
    print(f"\n=== Phase 2: Deep Optimization ===")
    phase2_end = time.monotonic() + float(budget_s) * 0.30

    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])

        # MD with different annealing schedules
        schedules = [
            (10.0, 300.0, 0.15, 0.003),
            (30.0, 500.0, 0.08, 0.002),
            (50.0, 800.0, 0.05, 0.001),
        ]
        for si, (ts, te, es, ee) in enumerate(schedules):
            rem = phase2_end - time.monotonic() - 20
            if rem < 20:
                break
            t_alloc = min(rem // 3, 120)
            if t_alloc < 15:
                break
            print(f"  MD schedule {si} (n={n_best}), t={t_alloc:.0f}s")
            a_opt, obj = md_optimize(state['best_seq'], time_limit=t_alloc,
                                     t_start=ts, t_end=te,
                                     eta_start=es, eta_end=ee, verbose=True)
            try_sequence(a_opt, f"md_sched{si}")

        # LP-based Frank-Wolfe
        rem = phase2_end - time.monotonic() - 15
        if rem > 25:
            t_alloc = min(rem, 150)
            print(f"  FW-LP (n={n_best}), t={t_alloc:.0f}s")
            a_opt, obj = fw_lp_optimize(state['best_seq'], time_limit=t_alloc,
                                        verbose=True)
            try_sequence(a_opt, "fwlp_deep")

        # Perturbation + MD
        for ps in [0.003, 0.01, 0.03, 0.08]:
            rem = phase2_end - time.monotonic() - 10
            if rem < 12:
                break
            noise = rng.standard_normal(n_best) * ps
            f_init = project_simplex(state['best_seq'] + noise)
            t_alloc = min(rem // 2, 60)
            if t_alloc < 10:
                break
            a_opt, obj = md_optimize(f_init, time_limit=t_alloc,
                                     t_start=10.0, t_end=200.0,
                                     eta_start=0.1, eta_end=0.01)
            try_sequence(a_opt, f"md_pert{ps}")

        # Mirror blend
        rem = phase2_end - time.monotonic() - 10
        if rem > 15:
            f_mirror = project_simplex(0.8 * state['best_seq'] + 0.2 * state['best_seq'][::-1])
            a_opt, obj = md_optimize(f_mirror, time_limit=min(rem, 60),
                                     t_start=10.0, t_end=200.0,
                                     eta_start=0.1, eta_end=0.01)
            try_sequence(a_opt, "mirror")

    print(f"  Phase 2 best: {state['best_score']:.6f}")

    # === Phase 3: Fine polishing ===
    print(f"\n=== Phase 3: Fine Polishing ===")
    phase3_end = deadline - 3

    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])

        # Fine MD at high temperature
        rem = phase3_end - time.monotonic() - 30
        if rem > 15:
            print(f"  MD fine polish, t={min(rem, 80):.0f}s")
            a_opt, obj = md_polish_fine(state['best_seq'], time_limit=min(rem, 80),
                                        t_start=200.0, t_end=2000.0, eta=0.02)
            try_sequence(a_opt, "md_fine1")

        # Subgradient
        rem = phase3_end - time.monotonic() - 20
        if rem > 12:
            a_opt, obj = sg_polish(state['best_seq'], time_limit=min(rem, 50),
                                   lr=0.001, mom=0.95,
                                   top_k=max(10, n_best // 4), temp=0.00002)
            try_sequence(a_opt, "sg")

        # Local search
        rem = phase3_end - time.monotonic() - 15
        if rem > 10:
            a_opt, obj = local_search_polish(state['best_seq'], time_limit=min(rem, 40))
            try_sequence(a_opt, "local")

        # Final LP squeeze
        rem = phase3_end - time.monotonic() - 8
        if rem > 10:
            a_opt, obj = fw_lp_optimize(state['best_seq'], time_limit=min(rem, 35))
            try_sequence(a_opt, "fwlp_final")

        # Final MD squeeze
        rem = phase3_end - time.monotonic() - 3
        if rem > 8:
            a_opt, obj = md_polish_fine(state['best_seq'], time_limit=min(rem, 25),
                                        t_start=500.0, t_end=3000.0, eta=0.01)
            try_sequence(a_opt, "md_squeeze")

    if state['best_seq'] is None:

        print(f"  Fallback uniform: {state['best_score']:.6f}")
