def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1} via simple clipping and normalization."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def nesterov_optimize(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=30, temp_init=0.003):
    """Nesterov-accelerated gradient descent with adaptive parameters."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    y = a.copy()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        y = a + mom * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()

        else:

        conv_y = fast_convolve(y)
        grad = softmax_gradient(y, conv_y, top_k=top_k, temp=temp)

        v = mom * v - lr * grad
        a_new = a + v
        a = project_nonneg_sum1(a_new)

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9995
        temp *= 1.003
        mom = min(0.97, mom + 0.0002)

        if no_improve >= 300:
            break

    return best_a, best_obj


def lp_step(a, C, conv):
    """One LP-based step with bisection line search.
    Returns (a_new, improved).
    """
    n = len(a)
    M = conv.max()

    # LP: max sum(g) s.t. C@g <= M, g >= 0
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True})
        if not result.success:
            return a.copy(), False
        g = result.x
        g_sum = g.sum()
        if g_sum <= 1.0 + 1e-6:
            return a.copy(), False
    except Exception:
        return a.copy(), False

    d = g - a
    norm_d = np.linalg.norm(d)
    if norm_d < 1e-12:
        return a.copy(), False

    # Bisection line search for optimal step
    best_a = a.copy()
    best_obj = 2.0 * n * M
    best_t = 0.0

    lo, hi = 0.0, 0.5
    # First check if hi gives improvement
    a_hi = project_nonneg_sum1(a + hi * d)
    conv_hi = fast_convolve(a_hi)
    obj_hi = 2.0 * n * conv_hi.max()

    if obj_hi >= best_obj - 1e-12:
        # Try smaller steps
        for t in [0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]:
            a_try = project_nonneg_sum1(a + t * d)
            conv_try = fast_convolve(a_try)
            obj_try = 2.0 * n * conv_try.max()
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try
                best_t = t
                break
        if best_t > 0:
            return best_a, True
        return a.copy(), False

    # Bisection: find largest t that gives improvement
    for _ in range(10):
        mid = (lo + hi) / 2
        a_mid = project_nonneg_sum1(a + mid * d)
        conv_mid = fast_convolve(a_mid)
        obj_mid = 2.0 * n * conv_mid.max()
        if obj_mid < best_obj - 1e-12:
            best_obj = obj_mid
            best_a = a_mid
            best_t = mid
            lo = mid
        else:
            hi = mid

    if best_t > 0:
        return best_a, True
    return a.copy(), False


def lp_optimize(a0, time_limit, verbose=False):
    """LP-based iterative optimization with perturbations on stagnation."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    max_no_improve = 60
    perturb_count = 0

    while time.monotonic() - t0 < time_limit:
        C = build_conv_matrix(a)
        a_new, improved = lp_step(a, C, conv)

        if improved:
            a = a_new
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        iteration += 1

        if verbose and iteration % 50 == 0:
            print(f"    LP iter {iteration}, obj={best_obj:.8f}, t={time.monotonic()-t0:.1f}s")

        if no_improve >= max_no_improve:
            rng = np.random.default_rng(iteration * 13 + perturb_count * 7)
            scale = 0.003 + 0.004 * min(perturb_count, 10)
            noise = rng.standard_normal(n) * scale
            a = project_nonneg_sum1(best_a + noise)
            conv = fast_convolve(a)
            no_improve = 0
            perturb_count += 1
            max_no_improve = min(200, 60 + 15 * perturb_count)

            if perturb_count > 50:
                break

    return best_a, best_obj


def coordinate_descent(a0, time_limit):
    """Coordinate descent: try adjusting pairs to reduce peak."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * len(a) * fast_convolve(a).max()
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        batch = max(1, int(200000 / n))
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit - 0.3:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            found = False
            for delta in [0.002, 0.005, 0.01, 0.02, -0.002, -0.005, -0.01]:
                if a[i] + delta < 0 or a[j] - delta < 0:
                    continue
                a[i] += delta
                a[j] -= delta
                conv_try = fast_convolve(a)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-10:
                    best_obj = obj_try
                    best_a = a.copy()
                    conv = conv_try
                    no_improve = 0
                    found = True
                    break
                else:
                    a[i] -= delta
                    a[j] += delta
            if not found:
                no_improve += 1

        if no_improve > 100:
            break

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Sine-based
    inits['sine1'] = (np.sin(np.pi * x / n) ** 2 + 0.01)
    inits['sine1'] /= inits['sine1'].sum()

    inits['sine2'] = (np.sin(2 * np.pi * x / n) ** 2 + 0.01)
    inits['sine2'] /= inits['sine2'].sum()

    # Hann window
    hann = 0.5 * (1 - np.cos(2 * np.pi * x / n))
    hann = np.maximum(hann, 0.01)
    inits['hann'] = hann / hann.sum()

    # Blackman window
    black = 0.42 - 0.5 * np.cos(2 * np.pi * x / n) + 0.08 * np.cos(4 * np.pi * x / n)
    black = np.maximum(black, 0.01)
    inits['blackman'] = black / black.sum()

    # Bimodal
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Quadratic (parabolic)
    quad = 1.0 - (2 * x / (n - 1) - 1) ** 2
    quad = np.maximum(quad, 0.01)
    inits['quadratic'] = quad / quad.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Hybrid gradient + LP + coordinate descent search
    with spectral-aware initialization and bisection line search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass

    # =========================================================
    # Phase 1: Quick gradient scan across lengths and inits
    # =========================================================
    print("=== Phase 1: Gradient Scan ===")
    t_start = time.monotonic()
    phase1_limit = 250.0

    lengths = [512, 768, 1024, 1536]
    scan_results = []

    for n in lengths:
        if time.monotonic() > deadline - 900:
            break
        inits = generate_initializations(n, parent, rng)

        priority = ['parent', 'parent_interp', 'hann', 'quadratic', 'sine1', 'blackman', 'uniform', 'random']
        init_names = [nm for nm in priority if nm in inits]
        init_names += [nm for nm in inits if nm not in priority]
        init_names = init_names[:5]

        t_per = max(8.0, (phase1_limit - 5) / (len(lengths) * 5))

        for name in init_names:
            if time.monotonic() > deadline - 800:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 500)
            if t_alloc < 5:
                break

            a0 = inits[name]
            a_opt, obj = nesterov_optimize(a0, time_limit=t_alloc,
                                           lr_init=0.012, momentum=0.92,
                                           top_k=40, temp_init=0.002)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Grad n={n}, {name}: {score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 2: LP-based deep optimization
    # =========================================================
    print("=== Phase 2: LP Deep Optimization ===")
    t_start2 = time.monotonic()

    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen and len(top_configs) < 5:
            top_configs.append((n, a_opt, score))
            seen.add(key)

    if best_sequence is not None:
        blen = len(best_sequence)
        if not any(n == blen for n, _, _ in top_configs):
            top_configs.append((blen, best_sequence.copy(), best_score))

    remaining = deadline - time.monotonic() - 300.0
    n_cand = max(1, len(top_configs))
    t_per_lp = max(40.0, remaining * 0.65 / n_cand)

    for i, (n, a0, sc) in enumerate(top_configs):
        if time.monotonic() > deadline - 250:
            break
        t_alloc = min(t_per_lp, deadline - time.monotonic() - 200)
        if t_alloc < 15:
            break

        print(f"  LP cand {i}: n={n}, init_score={sc:.6f}, t={t_alloc:.0f}s")
        a_opt, obj = lp_optimize(a0, time_limit=t_alloc, verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  LP {i} (n={n}): {score:.6f} *NEW BEST*")
            else:
                print(f"  LP {i} (n={n}): {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 3: Extended LP refinement on the best
    # =========================================================
    print("=== Phase 3: Extended LP Polish ===")
    t_start3 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 120.0
        if remaining > 60:
            for pi in range(4):
                if time.monotonic() > deadline - 80:
                    break
                t_alloc = min(remaining / (4 - pi) * 0.8, deadline - time.monotonic() - 60)
                if t_alloc < 15:
                    break

                if pi == 0:
                    a0 = best_sequence.copy()
                else:
                    scale = 0.002 * (pi + 1)
                    noise = rng.standard_normal(len(best_sequence)) * scale
                    a0 = project_nonneg_sum1(best_sequence + noise)

                a_opt, obj = lp_optimize(a0, time_limit=t_alloc)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Polish LP {pi}: {score:.6f} *NEW BEST*")
                else:
                    print(f"  Polish LP {pi}: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 4: Gradient + coordinate descent fine-tuning
    # =========================================================
    print("=== Phase 4: Fine-tuning ===")
    t_start4 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20.0
        n_best = len(best_sequence)

        # Coordinate descent
        t_cd = min(remaining * 0.3, 40.0)
        if t_cd > 5:
            a_opt, obj = coordinate_descent(best_sequence, time_limit=t_cd)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  CoordDesc: {score:.6f} *NEW BEST*")

        # Short gradient runs with different hyperparams
        remaining = deadline - time.monotonic() - 20.0
        if remaining > 10:
            configs = [
                {'lr_init': 0.004, 'momentum': 0.94, 'top_k': 60, 'temp_init': 0.001},
                {'lr_init': 0.002, 'momentum': 0.96, 'top_k': 100, 'temp_init': 0.0005},
                {'lr_init': 0.008, 'momentum': 0.89, 'top_k': 25, 'temp_init': 0.004},
            ]
            for gi, cfg in enumerate(configs):
                if time.monotonic() > deadline - 10:
                    break
                t_sg = min(remaining / (len(configs) - gi) * 0.7, 25.0)
                if t_sg < 5:
                    break
                a_opt, obj = nesterov_optimize(best_sequence, time_limit=t_sg, **cfg)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  SG {gi}: {score:.6f} *NEW BEST*")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 5: Final LP micro-polish
    # =========================================================
    if best_sequence is not None and time.monotonic() < deadline - 10:
        t_final = min(15.0, deadline - time.monotonic() - 5)
        if t_final > 5:
            a_opt, obj = lp_optimize(best_sequence, time_limit=t_final)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final: {score:.6f} *NEW BEST*")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
