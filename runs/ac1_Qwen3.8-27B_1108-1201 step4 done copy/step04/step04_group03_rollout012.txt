"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Fast coordinate descent with exact 1D line search (ternary search on convex function)
2. Periodic Frank-Wolfe with LP oracle on active peaks
3. Multi-start strategy with different sequence lengths

The coordinate descent is the key innovation: for each coordinate, the objective
is a convex 1D function that can be minimized exactly, making each step very cheap.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)[:size]
    return np.maximum(result, 0.0)


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def coordinate_descent(a_init, time_limit, n_restarts=5, restart_after=200, verbose=True):
    """
    Coordinate descent minimizing max(conv(a,a)) subject to a >= 0.
    Uses ternary search for exact 1D minimization at each coordinate.
    Normalizes after each coordinate update.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    sweep = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                    break
            
            x = a[i]
            # Compute base, d, q for coordinate i
            # conv[k] = base[k] + d[k]*x + q[k]*x^2
            # Where:
            #   d[k] = 2*a[k-i] if 0 <= k-i < n and k != 2i; 0 otherwise
            #   q[k] = 1 if k == 2i and 0 <= 2i <= 2n-2; 0 otherwise
            #   base[k] = conv[k] - d[k]*x - q[k]*x^2
            
            ki_bounds = (0, n - 1)  # valid range for k-i
            k_min = i  # k-i >= 0 → k >= i
            k_max = i + n - 1  # k-i <= n-1 → k <= i+n-1
            k_max = min(k_max, 2 * n - 2)
            
            d = np.zeros(2 * n - 1)
            q = np.zeros(2 * n - 1)
            
            # d[k] = 2*a[k-i] for k in [k_min, k_max], k != 2i
            for k in range(k_min, k_max + 1):
                ki = k - i
                if k == 2 * i:
                    q[k] = 1.0
                else:
                    d[k] = 2.0 * a[ki]
            
            # base = conv - d*x - q*x^2
            base = conv - d * x - q * x * x
            
            # Now minimize g(x) = max_k (base[k] + d[k]*x + q[k]*x^2) for x >= 0
            # Using ternary search on [0, x_hi]
            x_hi = max(2.0 * x, 2.0 / n)  # reasonable upper bound
            
            # Ternary search
            lo = 0.0
            hi = x_hi
            for _ in range(60):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                g1 = np.max(base + d * m1 + q * m1 * m1)
                g2 = np.max(base + d * m2 + q * m2 * m2)
                if g1 < g2:
                    hi = m2
                else:
                    lo = m1
            
            x_opt = (lo + hi) / 2.0
            
            # Update
            a[i] = x_opt
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
                break
            a = a / s
            
            # Update conv (incremental or full recompute)
            # Full recompute is O(n log n), which is fine
            conv = fast_convolve(a)
            cur_obj = conv.max()
            
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            # else: don't increment no_improve per coordinate, only per sweep
    
        # End of sweep
        cur_obj = fast_convolve(a).max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= restart_after:
            # Restart with perturbation
            a = best_a * np.exp(rng.standard_normal(n) * 0.01)
            a = project_simplex(a)
            conv = fast_convolve(a)
            best_obj = conv.max()
            best_a = a.copy()
            no_improve = 0
    
    return best_a, best_obj


def fw_active_peaks(a_init, time_limit, n_active=10, verbose=True):
    """
    Frank-Wolfe method using LP oracle, but only constraining the top n_active peaks.
    This is faster than the full LP since we have fewer constraints.
    """
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = M
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    # Precompute the sparse matrix structure
    # A_ub[k, j] = a[j] if 0 <= k-j < n, else 0
    # This is a Toeplitz-like matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    
    row_idx = np.array(rows)
    col_idx = np.array(cols)
    val_arr = np.array(vals)
    
    # Build sparse matrix (will need to update values when a changes)
    # Actually, for FW, the matrix changes at each iteration (a changes)
    # So we rebuild each time
    
    alpha = 1.0 / (2 * (it + 1))  # Will use standard FW step size
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Find active peaks (top n_active)
        n_act = min(n_active, 2 * n - 1)
        active_peaks = np.argpartition(conv, -n_act)[-n_act:]
        
        # Solve LP: max sum(g) s.t. (a*g)[k] <= M for k in active_peaks, g >= 0
        # In standard form: min -sum(g) s.t. (a*g)[k] <= M, g >= 0
        c = -np.ones(n)
        
        # Build constraint matrix for active peaks only
        A_rows = []
        A_cols = []
        A_vals = []
        b_ub = []
        for k in active_peaks:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                A_rows.append(len(b_ub))  # row index in A_ub
                A_cols.append(j)
                A_vals.append(a[j])
            b_ub.append(M)
        
        if len(A_rows) == 0:
            break
            
        A_ub_sp = csr_matrix((A_vals, (A_rows, A_cols)), shape=(len(b_ub), n))
        
        res = linprog(c, A_ub=A_ub_sp, b_ub=np.array(b_ub), 
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': 5})
        
        if not res.success or res.x is None:
            no_improve += 1
            if no_improve > 50:
                rng = np.random.default_rng(int(time.monotonic() * 10000))
                a = best_a * np.exp(rng.standard_normal(n) * 0.02)
                a = project_simplex(a)
                conv = fast_convolve(a)
                M = conv.max()
                no_improve = 0
            continue
        
        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        # Line search
        alpha_fw = 1.0 / (2 * (it + 1))
        # Try a few alphas
        best_alpha = 0.0
        best_val = M
        for alpha_try in [alpha_fw, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - alpha_try) * a + alpha_try * g
            a_try = np.maximum(a_try, 0)
            st = a_try.sum()
            if st < 1e-15:
                continue
            a_try = a_try / st
            val = fast_convolve(a_try).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha_try
        
        if best_alpha > 0:
            a = (1 - best_alpha) * a + best_alpha * g
            a = np.maximum(a, 0)
            a = a / a.sum()
            conv = fast_convolve(a)
            M = conv.max()
            
            if M < best_obj:
                best_obj = M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
    
    return best_a, best_obj
