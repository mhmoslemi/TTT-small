```python
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

class FW_LP_Solver:
    def __init__(self, n):
        self.n = n
        # Precompute sparsity pattern
        rows = []
        cols = []
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            js = np.arange(j_start, j_end + 1)
            rows.extend(k for _ in js)
            cols.extend(js.tolist())
        self.rows = np.array(rows)
        self.cols = np.array(cols)
        self.nnz = len(rows)
    
    def build_matrix(self, f):
        # Values: for constraint k, col j, value is f[k-j]
        vals = np.empty(self.nnz)
        for i in range(self.nnz):
            k = self.rows[i]
            j = self.cols[i]
            vals[i] = f[k - j]
        return csr_matrix((vals, (self.rows, self.cols)), shape=(2*self.n-1, self.n))
    
    def solve_direction(self, f, M):
        n = self.n
        A = self.build_matrix(f)
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        bounds = [(0.0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 10})
        if res.success:
            g = res.x
            g = np.maximum(g, 0)
            s = g.sum()
            if s > 1e-15:
                return g / s
        return None
```

Now, for the FW iteration with Friedman-LetsGo step:

- f_{t+1} = (1 - γ_t) * f_t + γ_t * g_t

This is known to give O(1/t) convergence for strongly convex problems and works well in practice for non-convex ones too.

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-start diverse initialization
2. Away-step Frank-Wolfe with LP oracle (fast, no line search)
3. Log-sum-exp gradient polishing with Adam
4. Active-set refinement
5. trying multiple sequence lengths
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_w(weights, f):
    """Compute gradient of sum_k weights[k] * conv_k(f) w.r.t. f.
    Returns array of length n.
    grad[i] = sum_k weights[k] * (f[k-i] + f[i-k]) (with 0 boundary)
    Using FFT: this is 2 * cross_correlation(weights, f)
    """
    n = len(f)
    # conv_k(f) = sum_j f[j] * f[k-j]
    # d(conv_k)/df[i] = f[k-i] + f[i-k] (with f[m]=0 for m<0 or m>=n)
    # grad[i] = sum_k weights[k] * (f[k-i] + f[i-k])
    #         = sum_k weights[k] * f[k-i] + sum_k weights[k] * f[i-k]
    #         = (weights * f)[i] reversed stuff...
    # Actually: sum_k weights[k] * f[k-i] is the cross-correlation of weights and f at lag i
    # And sum_k weights[k] * f[i-k] is also a cross-correlation (at lag -i = i for symmetric)
    # Since weights comes from conv which is symmetric, both terms are equal.
    # So grad[i] = 2 * sum_k weights[k] * f[k-i] (with boundary)
    # This is 2 * (weights cross-correlated with f)[i]
    
    # cross_correlation(w, f)[i] = sum_k w[k] * f[k-i]
    # = sum_k w[k+i] * f[k] (reindexing)
    # This is the convolution of w and reversed f, evaluated at position (len(w)-1-i)
    
    # Let's use FFT:
    # Let g = f reversed. Then conv(w, g)[position] = sum_k w[k] * g[position-k]
    # We want sum_k w[k] * f[k-i] = sum_k w[k] * g[n-1-(k-i)] = sum_k w[k] * g[n-1-k+i]
    # = conv(w, g)[n-1+i] where conv is linear convolution
    
    total_len = (2*n-1) + n - 1  # len(weights) + len(f) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    G = np.fft.rfft(f[::-1], fft_size)
    result = np.fft.irfft(W * G, fft_size)
    # result[m] = sum_k weights[k] * f_rev[m-k] = sum_k weights[k] * f[n-1-m+k]
    # We want: cc[i] = sum_k weights[k] * f[k-i]
    # Let's check: set j = k-i, so k = j+i: cc[i] = sum_j weights[j+i] * f[j]
    # result[m] = sum_k weights[k] * f[n-1-m+k]
    # Set n-1-m+k = j, so k = j - n + 1 + m: result[m] = sum_j weights[j-n+1+m] * f[j]
    # We want sum_j weights[j+i] * f[j], so j-n+1+m = j+i, i.e., m = i + n - 1
    # So cc[i] = result[i + n - 1]
    
    return 2.0 * result[n-1 : n-1+n]


class FastFW:
    def __init__(self, n, seed=42):
        self.n = n
        self.rng = np.random.default_rng(seed)
        # Precompute sparsity pattern
        rows = []
        cols = []
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
        self.rows = np.array(rows)
        self.cols = np.array(cols)
        self.nnz = len(rows)
        self.c_vec = -np.ones(n)
        self.bounds = [(0.0, None)] * n
    
    def _build_A(self, f):
        vals = np.empty(self.nnz)
        idx = 0
        for k in range(2 * self.n - 1):
            j_start = max(0, k - self.n + 1)
            j_end = min(self.n - 1, k)
            length = j_end - j_start + 1
            vals[idx:idx+length] = f[k - j_end : k - j_start + 1]
            idx += length
        return csr_matrix((vals, (self.rows, self.cols)), shape=(2*self.n-1, self.n))
    
    def solve_lp(self, f, M):
        A = self._build_A(f)
        b_ub = np.full(2*self.n-1, M)
        try:
            res = linprog(self.c_vec, A_ub=A, b_ub=b_ub, bounds=self.bounds,
                         method='highs', options={'presolve': True, 'time_limit': 5,
                                                   'dual_feasibility_tolerance': 1e-8,
                                                   'primal_feasibility_tolerance': 1e-8})
            if res.success:
                g = np.maximum(res.x, 0.0)
                s = g.sum()
                if s > 1e-15:
                    return g / s
        except Exception:
            pass
        return None
    
    def run(self, f0, time_limit, m_relax=1.05):
        f = np.maximum(f0.astype(np.float64), 0.0)
        s = f.sum()
        if s < 1e-15:
            f = np.ones(self.n) / self.n
        else:
            f = f / s
        
        f_away = f.copy()  # for away-step
        conv = fast_convolve(f)
        M = conv.max()
        best_obj = M
        best_f = f.copy()
        
        t0 = time.monotonic()
        no_improve = 0
        it = 0
        m_exp = 1.0
        lp_fails = 0
        
        while time.monotonic() - t0 < time_limit:
            it += 1
            
            g = self.solve_lp(f, M * m_exp)
            if g is None:
                lp_fails += 1
                m_exp *= 1.1
                if m_exp > 5.0:
                    # Fallback: use random direction
                    g = project_simplex(f + self.rng.standard_normal(self.n) * 0.01)
                    m_exp = 1.0
                if lp_fails > 20:
                    break
                continue
            lp_fails = 0
            m_exp = 1.0
            
            if np.max(np.abs(g - f)) < 1e-12:
                no_improve += 1
                if no_improve > 30:
                    break
                continue
            
            # Friedman-LetsGo step size
            gamma = 2.0 / (it + 2)
            
            # Away-step: check if we can remove f_away
            # f_new = (1 - gamma - gamma_away) * f + gamma * g + gamma_away * (away)
            # For away-step: gamma_away = gamma * (f - f_away) . (f - g) / ||f - f_away||^2
            # clipped to [0, 1]
            
            d_away = f - f_away
            d_away_norm_sq = np.dot(d_away, d_away)
            
            if d_away_norm_sq > 1e-15:
                gamma_away = gamma * np.dot(d_away, f - g) / d_away_norm_sq
                gamma_away = np.clip(gamma_away, 0, 1)
            else:
                gamma_away = 0.0
            
            if gamma_away > 0.01:
                f_new = (1 - gamma - gamma_away) * f + gamma * g + gamma_away * (f - d_away)
                # f - d_away = f - (f - f_away) = f_away
                f_new = (1 - gamma - gamma_away) * f + gamma * g + gamma_away * f_away
                f_new = np.maximum(f_new, 0.0)
                s_new = f_new.sum()
                if s_new > 1e-15:
                    f_new = f_new / s_new
                else:
                    gamma_away = 0.0
                    f_new = (1 - gamma) * f + gamma * g
                    f_new = np.maximum(f_new, 0.0)
                    f_new = f_new / f_new.sum()
            else:
                f_new = (1 - gamma) * f + gamma * g
                f_new = np.maximum(f_new, 0.0)
                s_new = f_new.sum()
                if s_new > 1e-15:
                    f_new = f_new / s_new
                else:
                    continue
            
            f_away = f_new
            
            conv_new = fast_convolve(f_new)
            M_new = conv_new.max()
            f = f_new
            M = M_new
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            if no_improve > 50:
                # Diversification
                scale = 0.005 + 0.01 * min(1.0, no_improve / 100.0)
                f = project_simplex(best_f + self.rng.standard_normal(self.n) * scale)
                f_away = f.copy()
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
                it = max(1, it // 2)  # reset step size schedule
                if it < 1:
                    it = 1
        
        return best_f, best_obj


def lse_optimize_adam(f0, time_limit, beta_start=20.0, beta_end=5000.0):
    """Log-sum-exp with Adam optimizer."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    beta = beta_start
    lr = 1.0 / np.sqrt(n)
    m_o = np.zeros(n)  # Adam first moment
    v_o = np.zeros(n)  # Adam second moment
    adam_t = 0
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(f)
        
        # Softmax weights
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()
        
        # Gradient
        grad = compute_gradient_w(w, f)
        
        # Adam update
        adam_t += 1
        m_o = beta1 * m_o + (1 - beta1) * grad
        v_o = beta2 * v_o + (1 - beta2) * grad ** 2
        m_hat = m_o / (1 - beta1 ** adam_t)
        v_hat = v_o / (1 - beta2 ** adam_t)
        
        f_new = project_simplex(f - lr * m_hat / (np.sqrt(v_hat) + eps))
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            f = f_new  # still accept for exploration
            no_improve += 1
            if no_improve >= 20:
                lr *= 0.95
            if no_improve > 100:
                f = best_f.copy()
                m_o = np.zeros(n)
                v_o = np.zeros(n)
                adam_t = 0
                lr = 1.0 / np.sqrt(n)
                beta = min(beta * 1.5, beta_end)
                no_improve = 0
        
        # Ramp up beta
        if beta < beta_end:
            beta *= 1.005
        
        if it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-Adam it={it}, obj={best_obj:.12f}, beta={beta:.1f}, lr={lr:.6f}, t={elapsed:.0f}s")
    
    return best_f, best_obj


def diverse_starts(n, parent=None, n_starts=8, seed=42):
    """Generate diverse starting points of length n."""
    rng = np.random.default_rng(seed)
    starts = []
    
    if parent is not None and len(parent) == n:
        starts.append(parent.astype(np.float64))
    
    # Cosine
    x = np.linspace(0, np.pi, n)
    starts.append(np.cos(x/2)**2)
    
    # Gaussian
    c = np.arange(n)
    mu = (n-1)/2
    for sigma in [n/4, n/3, n/5]:
        starts.append(np.exp(-0.5 * ((c-mu)/sigma)**2))
    
    # Triangular
    starts.append(1 - np.abs(c - mu)/(mu + 1))
    
    # Random
    starts.append(rng.dirichlet(np.ones(n)))
    starts.append(np.maximum(rng.standard_normal(n), 0))
    
    # Power law
    starts.append(1.0 / (1 + c/rng.uniform(1, n/3)))
    
    # Two peaks
    p1, p2 = n//3, 2*n//3
    starts.append(np.exp(-0.5*((c-p1)/(n/8))**2) + np.exp(-0.5*((c-p2)/(n/8))**2))
    
    result = []
    for s in starts[:n_starts]:
        s = np.maximum(s, 0)
        ss = s.sum()
        if ss > 1e-10:
            result.append(s / ss)
    
    return result
```

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_w(weights, f):
    n = len(f)
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    G = np.fft.rfft(f[::-1], fft_size)
    result = np.fft.irfft(W * G, fft_size)
    return 2.0 * result[n - 1:n - 1 + n]

        rows.extend([k] * len(js))
        cols.extend(js.tolist())
