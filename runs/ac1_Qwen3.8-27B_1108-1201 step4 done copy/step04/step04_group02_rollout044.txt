```python
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
```

```python
import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A*A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_grad_at_peak(y, k, n):
    """Gradient of conv_k at point y (length n)."""
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo > hi:
        return grad
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * y[k - idx]
    return grad

def compute_subgradient(y, conv, tol=1e-6):
    """Subgradient of max(conv) at y, averaged over near-active peaks."""
    n = len(y)
    max_val = conv.max()
    active = np.where(conv >= max_val - max_val*tol)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    grad = np.zeros(n)
    for k in active:
        grad += compute_grad_at_peak(y, k, n)
    grad /= len(active)
    return grad

def accelerated_subgrad(f_init, time_limit, tol=1e-6, verbose=False):
    n = len(f_init)
    f = project_simplex(f_init.copy())
    y = f.copy()
    t = 1.0
    alpha = 1.0 / np.sqrt(n)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1:
            break
        
        conv_y = fast_convolve(y)
        grad = compute_subgradient(y, conv_y, tol=tol)
        
        # Backtracking line search
        alpha = min(alpha * 1.5, 10.0)
        best_val = conv_y.max()
        best_x = None
        for _ in range(20):
            x_try = project_simplex(y - alpha * grad)
            val = fast_convolve(x_try).max()
            if val < best_val - 1e-14:
                best_val = val
                best_x = x_try
                alpha = min(alpha * 1.5, 10.0)
            else:
                alpha *= 0.5
                if alpha < 1e-10:
                    break
        
        if best_x is not None:
            f_new = best_x
        else:
            f_new = project_simplex(y - alpha * grad)
        
        # Nesterov momentum update
        t_next = (1.0 + np.sqrt(1.0 + 4.0*t*t)) / 2.0
        y = f_new + ((t - 1.0) / t_next) * (f_new - f)
        t = t_next
        f = f_new
        
        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Restart if stuck
        if no_improve > 500:
            y = f.copy()
            t = 1.0
            alpha = 1.0 / np.sqrt(n)
            no_improve = 0
        
        if verbose and it % 1000 == 0:
            print(f"  iter {it}: obj={best_obj:.8f}, alpha={alpha:.6f}, no_improve={no_improve}")
    
    return best_f, best_obj
```
