import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A*A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.ones(n)/n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0)/(rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lse_grad(f, beta, conv):
    n = len(f)
    bc = beta * conv
    bc_max = bc.max()
    w = np.exp(bc - bc_max)
    w = w / w.sum()
    
    # grad[i] = 2 * sum_{m} w[m] * f[m-i] for valid m, i
    # This is 2 * (w cross-correlate f)[i]
    # = 2 * (w * flip(f))[n-1+i] in full convolution terms
    ff = f[::-1]
    L = (2*n-1) + n - 1
    fs = 1 << (L-1).bit_length()
    W = np.fft.rfft(w, fs)
    F = np.fft.rfft(ff, fs)
    c = np.fft.irfft(W*F, fs)
    return 2.0 * c[n-1:n-1+n]
