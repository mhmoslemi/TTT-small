"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_lse_grad(weights, f):
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]

def build_lp_matrix(f):
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def solve_lp(f, M):
    n = len(f)
    A = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 15})
        if res.success:
            return np.maximum(res.x, 0.0)
    except:
        pass
    return None

def fw_optimize(f0, time_limit, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_mult = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        g = solve_lp(f, M * m_mult)
        if g is None:
            m_mult *= 1.15
            if m_mult > 5.0:
                m_mult = 1.0
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.abs(g - f).max() < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            m_mult = min(m_mult * 1.03, 3.0)
            continue
        
        # Armijo-style line search
        best_alpha = 0.0
        best_val = M
        # Coarse search
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = np.maximum((1 - alpha) * f + alpha * g, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft /= st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Fine search
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(10):
                mid = (lo + hi) / 2
                ft = np.maximum((1 - mid) * f + mid * g, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft /= st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-10:
            f = np.maximum((1 - best_alpha) * f + best_alpha * g, 0.0)
            f /= f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_mult = 1.0
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_mult = min(m_mult * 1.05, 5.0)
        
        if no_improve > 50:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.001 + 0.002 * min(1.0, (no_improve - 50) / 50)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_mult = 1.0
            no_improve = 0
            
        if verbose and it % 50 == 0:
            print(f"  FW it={it}, M={best_obj:.10f}, elapsed={time.monotonic()-t0:.0f}s")
    
    return best_f, best_obj

def lse_polish(f0, time_limit, beta_start=20, beta_end=5000):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    beta = beta_start
    lr = 2.0 / n
    momentum = 0.92
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(f)
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()
        
        grad = compute_lse_grad(w, f)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad /= grad_norm
        
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.003, 4.0 / n)
        else:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.9
            if no_improve > 100:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(2.0 / n, 0.001)
                beta = min(beta * 2.0, beta_end)
                no_improve = 0
        
        if beta < beta_end:
            beta *= 1.005
    
    return best_f, best_obj

def local_search(f0, time_limit):

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:

    n = len(f)

    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        # Pick a random subset of coordinates to perturb
        k = min(n, rng.integers(1, max(2, n // 10)))
        idx = rng.choice(n, size=k, replace=False)
        
        # Try different scales of perturbation
        improved = False
        for scale in [0.001, 0.005, 0.01, 0.02, 0.05]:
            f_trial = best_f.copy()
            delta = rng.standard_normal(k) * scale
            # Move mass from some coords to others to maintain sum
            total_delta = delta.sum()
            delta -= total_delta / n  # approximately maintain sum
            f_trial[idx] += delta
            f_trial = np.maximum(f_trial, 0)
            st = f_trial.sum()
            if st < 1e-15:
                continue
            f_trial /= st
            obj = fast_convolve(f_trial).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f_trial.copy()
                improved = True
                break
        
        if not improved:
            # Try golden section on a single coordinate
            i = rng.integers(0, n)
            f_base = best_f
            a_i = f_base[i]
            lo = 0.0
            hi = a_i * 3 + 0.001
            for _ in range(8):
                m1 = lo + (hi - lo) * 0.382
                m2 = lo + (hi - lo) * 0.618
                f1 = f_base.copy()
                f2 = f_base.copy()
                f1[i] = m1
                f2[i] = m2
                # Renormalize
                f1 = np.maximum(f1, 0)
                f2 = np.maximum(f2, 0)
                s1 = f1.sum()
                s2 = f2.sum()
                if s1 > 1e-15:
                    f1 /= s1
                if s2 > 1e-15:
                    f2 /= s2
                o1 = fast_convolve(f1).max()
                o2 = fast_convolve(f2).max()
                if o1 < o2:
                    hi = m2
                else:
                    lo = m1
            # Check the best point
            for m in [lo, hi, (lo+hi)/2]:
                f_trial = f_base.copy()
                f_trial[i] = m
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st > 1e-15:
                    f_trial /= st
                    obj = fast_convolve(f_trial).max()
                    if obj < best_obj - 1e-14:

                        best_f = f_trial.copy()
    
    return best_f, best_obj

def generate_initializations(n, parent=None):
    """Generate diverse initializations."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Linear ramp (increasing)
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp / ramp.sum())
    
    # Linear ramp (decreasing)

    # Quadratic (parabola, peak in middle)
    x = np.linspace(0, 1, n)
    para = 1 - (2*x - 1)**2
    para = np.maximum(para, 0)
    if para.sum() > 0:
        inits.append(para / para.sum())
    
    # Exponential decay
    exp = np.exp(-3 * x)
    inits.append(exp / exp.sum())
    
    # Two-level: first half high, second half low
    two_level = np.ones(n)
    two_level[:n//2] = 2.0
    inits.append(two_level / two_level.sum())
    
    # Sine-based
    sine = 1 + np.sin(np.pi * x)
    inits.append(sine / sine.sum())
    
    # Cosine-based (anti-palindromic)
    cos = 1 + np.cos(np.pi * x)
    cos = np.maximum(cos, 0.01)
    inits.append(cos / cos.sum())
    
    # Peak at 1/3
    peak13 = np.exp(-50 * (x - 1/3)**2)
    inits.append(peak13 / peak13.sum())
    
    # Peak at 2/3
    peak23 = np.exp(-50 * (x - 2/3)**2)
    inits.append(peak23 / peak23.sum())
    
    # Random
    rng = np.random.default_rng(42)
    for _ in range(5):
        r = rng.random(n)
        r = np.maximum(r, 0)
        if r.sum() > 0:
            inits.append(r / r.sum())
    
    # Parent (if available)
    if parent is not None:
        p = np.maximum(parent.astype(np.float64), 0)
        if p.sum() > 1e-15:
            inits.append(p / p.sum())
    
    return inits
