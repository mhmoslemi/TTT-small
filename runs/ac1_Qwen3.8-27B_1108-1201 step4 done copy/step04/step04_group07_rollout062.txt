"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Adam optimization with smooth max
approximation, temperature annealing, multiple restarts, and LP-based polishing.
Key innovations: (1) FFT-based gradient computation for the smooth max objective,
(2) adaptive temperature schedule, (3) diverse initializations including flat-spectrum
start, (4) multi-phase search with time allocation.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a, b=None):
    n = len(a)
    if b is None:
        b = a
        nb = n
    else:
        nb = len(b)
    size = n + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_gradient(a, w):
    """Compute grad of sum_k w_k * conv(a,a)_k wrt a.
    w should have length 2n-1 (same as conv).
    Returns gradient of shape (n,).
    """
    n = len(a)
    N = 1 << (2 * n - 1 - 1).bit_length()
    N = max(N, 2 * n - 1)
    if N != (1 << int(np.log2(N))):
        N = 1 << (N - 1).bit_length()
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(W * np.conj(A), N)
    return 2.0 * corr[:n]

def adam_optimize(a0, time_limit, lr=0.05, tau_start=0.005, tau_end=0.0005,
                  beta1=0.9, beta2=0.999, eps=1e-8, verbose=False):
    """Adam optimization with smooth max and temperature annealing."""
    n = len(a0)
    a = a0.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_obj = 2 * n * fast_convolve(a).max()
    
    t0 = time.monotonic()
    max_iters = int(time_limit * 500)  # estimate
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        # Progress-based temperature
        progress = min(1.0, t / max_iters)
        tau = tau_start * (tau_end / tau_start) ** progress
        
        # Compute conv and smooth max weights
        conv = fast_convolve(a)
        scaled = conv / tau
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w = w / w_sum
        
        # Compute gradient
        grad = compute_gradient(a, w)
        
        # Current objective
        obj = 2 * n * conv.max()
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project to simplex
        a = project_simplex(a)
        
        # Track best
        new_obj = 2 * n * fast_convolve(a).max()
        if new_obj < best_obj:
            best_obj = new_obj
            best_a = a.copy()
        
        if verbose and t % 1000 == 0:
            print(f"      iter {t}: obj={best_obj:.10f}, tau={tau:.6f}")
    
    return best_a, best_obj

def solve_linearized_lp(f, conv_f):
    """Frank-Wolfe LP oracle: min t s.t. t >= conv_f[k] + 2*(B*g)[k], g >= -f, sum(g)=0."""
    n = len(f)
    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        B[j:j + n, j] = f
    
    c = np.zeros(n + 1)
    c[0] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = 2.0 * B
    b_ub = -conv_f
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]
    return None

def fw_polish(a0, time_limit, max_iters=50):
    """Frank-Wolfe polishing step."""
    n = len(a0)
    a = a0.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = 2 * n * fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1
        conv = fast_convolve(a)
        g = solve_linearized_lp(a, conv)
        if g is None:
            break
        target = np.maximum(a + g, 0)
        ts = target.sum()
        if ts < 1e-15:
            break
        target = target / ts
        
        if np.max(np.abs(target - a)) < 1e-10:
            break
        
        # Line search
        best_alpha = 0.0
        best_val = 2 * n * conv.max()
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = (1 - alpha) * a + alpha * target
            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = 2 * n * fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            a = (1 - best_alpha) * a + best_alpha * target
            a = np.maximum(a, 0)
            a = a / a.sum()
            obj = 2 * n * fast_convolve(a).max()
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
        else:
            break
    
    return best_a, best_obj

def generate_flat_spectrum_init(n, rng):
    """Initialize with a flat-spectrum (CAZAC-like) sequence projected to non-negative."""
    S = 1 << (2 * n - 1 - 1).bit_length()
    S = max(S, 2 * n - 1)
    # Random phase
    phase = rng.uniform(0, 2 * np.pi, S // 2 + 1)
    # Ensure Hermitian symmetry for real output
    spectrum = np.ones(S // 2 + 1)  # flat magnitude
    # Apply random phase (with Hermitian constraint)
    spectrum[0] = 1.0  # DC
    if S % 2 == 0:
        spectrum[-1] = 1.0  # Nyquist
    spectrum[1:-1] = np.exp(1j * rng.uniform(0, 2 * np.pi, S // 2 - 1))
    
    a_complex = np.fft.irfft(spectrum, S)[:n]
    # Project to non-negative
    a = np.maximum(a_complex, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Adam optimization with smooth max
approximation, temperature annealing, diverse initializations including flat-spectrum
starts, and LP-based Frank-Wolfe polishing. Key innovations: FFT-based gradient for
smooth max objective, adaptive temperature schedule, multi-phase search with
time allocation across sequence lengths.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    n = len(a)
    if b is None:
        b = a
        nb = n
    else:
        nb = len(b)
    size = n + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(a, w):
    """Gradient of sum_k w_k*(a*a)_k w.r.t. a. w has length 2n-1."""
    n = len(a)
    N = 1 << (2 * n - 2).bit_length()
    N = max(N, 2 * n - 1)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(W * np.conj(A), N)
    return 2.0 * corr[:n]


def adam_optimize(a0, time_limit, lr=0.05, tau_start=0.004, tau_end=0.0003,
                  beta1=0.85, beta2=0.999, eps=1e-8, noise_scale=0.0, verbose=False):
    """Adam with smooth max, temperature annealing, optional noise for escaping plateaus."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    m = np.zeros(n)
    v_acc = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        progress = min(1.0, t / (time_limit * 800))
        # Exponential annealing
        tau = tau_start * (tau_end / tau_start) ** progress

        conv = fast_convolve(a)
        scaled = conv / tau
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum

        grad = compute_gradient(a, w)

        # Adam
        m = beta1 * m + (1.0 - beta1) * grad
        v_acc = beta2 * v_acc + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v_acc / (1.0 - beta2 ** t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Add noise if stuck
        if no_improve > 2000 and noise_scale > 0:
            a += np.random.default_rng(t).standard_normal(n) * noise_scale

        a = project_simplex(a)

        obj = 2.0 * n * fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"        t={t}, obj={best_obj:.10f}, tau={tau:.6f}, no_improve={no_improve}, elapsed={elapsed:.0f}s")

    return best_a, best_obj


def solve_linearized_lp_reduced(f, conv_f, top_k=40):
    """Reduced LP: only top_k peak constraints for speed."""
    n = len(f)
    peaks = np.argpartition(conv_f, -top_k)[-top_k:]

    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        B[j:j + n, j] = f

    c = np.zeros(n + 1)
    c[0] = 1.0
    A_ub = np.zeros((top_k, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = 2.0 * B[peaks, :]
    b_ub = -conv_f[peaks]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]
    return None

    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(99)

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        g = solve_linearized_lp_reduced(a, conv, top_k=top_k)
        if g is None:
            break

        target = np.maximum(a + g, 0.0)
        ts = target.sum()
        if ts < 1e-15:
            break
        target = target / ts

        if np.max(np.abs(target - a)) < 1e-11:
            break

        # Line search
        best_alpha = 0.0
        best_val = 2.0 * n * conv.max()
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * a + alpha * target
            fs = ft.sum()
            if fs < 1e-15:
                continue
            ft = ft / fs
            val = 2.0 * n * fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            a = (1.0 - best_alpha) * a + best_alpha * target
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            obj = 2.0 * n * fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                if verbose and it % 20 == 0:
                    print(f"        FW it={it}: obj={best_obj:.10f}")
        else:
            # Perturb and continue
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)

    return best_a, best_obj


def generate_flat_spectrum_init(n, rng):
    """Flat-spectrum initialization projected to non-negative."""
    S = 1 << (2 * n - 2).bit_length()
    S = max(S, 4 * n)
    spec = np.ones(S // 2 + 1)
    if S % 2 == 0 and S // 2 + 1 > 1:
        spec[1:-1] = np.exp(1j * rng.uniform(0, 2 * np.pi, S // 2 - 1))
    a_complex = np.fft.irfft(spec, S)[:n]
    a = np.maximum(a_complex, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def subgradient_optimize(a0, time_limit, lr_init=0.005, mom=0.93, top_k=50, temp=0.0001):
    """Subgradient descent with softmax-weighted multi-peak targeting."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    vel = np.zeros(n)
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    cur_lr = lr_init
    cur_temp = temp

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        vel = mom * vel - cur_lr * sg
        a = project_simplex(a + vel)
        obj = 2.0 * n * fast_convolve(a).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9998
        cur_temp *= 1.003

        if no_improve > 500:
            rng = np.random.default_rng(it * 31 + 17)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            vel = np.zeros(n)
            cur_lr = max(lr_init * 0.5, 0.001)
            no_improve = 0

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search:
    Phase 1: Broad exploration across n values and initializations
    Phase 2: Adam optimization with smooth max + annealing (fast, many iterations)
    Phase 3: Subgradient + FW LP polishing
    Phase 4: Final local search
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if len(a) == 0 or a.sum() < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={ps:.8f}")
    except Exception:
        pass

    # Determine n candidates
    if parent is not None:
        n_parent = len(parent)
        n_values = sorted(set([n_parent, max(64, n_parent - 32), n_parent + 32,
                                256, 320, 384, 448, 512, 640]))
    else:
        n_values = [256, 320, 384, 448, 512, 640]
    n_values = [n for n in n_values if 64 <= n <= 1024]

    # === Phase 1: Exploration ===
    print("=== Phase 1: Exploration ===")
    t_start = time.monotonic()

    init_pool = {}
    for n in n_values:
        inits = []
        inits.append(np.ones(n) / n)
        inits.append(rng.dirichlet(np.ones(n) * 0.4))

        x = np.arange(n, dtype=np.float64)
        for sf in [0.12, 0.2, 0.3, 0.45]:
            g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (sf * n)) ** 2)
            inits.append(g / g.sum())

        c = np.abs(np.cos(np.pi * x / n)) + 0.03
        inits.append(c / c.sum())

        # Beta-like
        for ab, bb in [(1.5, 1.5), (2.0, 3.0), (3.0, 2.0)]:
            b = (x + 1) ** ab * (n - x) ** bb
            inits.append(b / b.sum())

        # Flat spectrum
        for _ in range(3):
            inits.append(generate_flat_spectrum_init(n, rng))

        if parent is not None:
            px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            px = np.maximum(px, 0)
            inits.append(px / px.sum())

        init_pool[n] = inits
        # Quick eval of first few
        for i, ini in enumerate(inits[:4]):
            try_seq(ini, f"exp_n{n}_{i}")

    print(f"  Phase 1 done ({time.monotonic()-t_start:.1f}s), best={best_score:.8f}")

    # === Phase 2: Adam optimization on best n values ===
    print("\n=== Phase 2: Adam Optimization ===")

    # Find best init per n
    n_best_inits = {}
    for n in n_values:
        best_ini = None
        best_ini_obj = float('inf')
        for ini in init_pool[n]:
            obj = 2.0 * n * fast_convolve(np.maximum(ini, 0)).max()
            if obj < best_ini_obj:
                best_ini_obj = obj
                best_ini = ini
        n_best_inits[n] = best_ini

    # Rank n values
    n_ranks = sorted(n_values, key=lambda n: 2.0 * n * fast_convolve(n_best_inits[n]).max())
    top_ns = n_ranks[:4]
    print(f"  Top n: {top_ns}")

    remaining = deadline - time.monotonic() - 150
    time_per_n = remaining / max(1, len(top_ns))

    for n in top_ns:
        if time.monotonic() > deadline - 150:
            break

        print(f"  --- n={n}, budget={time_per_n:.0f}s ---")
        a0 = n_best_inits[n].copy()

        # Multiple Adam runs with varying hyperparameters
        configs = [
            {'lr': 0.04, 'tau_start': 0.005, 'tau_end': 0.0002},
            {'lr': 0.02, 'tau_start': 0.003, 'tau_end': 0.0001},
            {'lr': 0.06, 'tau_start': 0.008, 'tau_end': 0.0003, 'noise_scale': 0.001},
        ]
        t_each = time_per_n / (len(configs) + 1)

        for ci, cfg in enumerate(configs):
            if time.monotonic() > deadline - 150:
                break
            a_init = a0.copy()
            if ci > 0:
                a_init = a_init + rng.standard_normal(n) * (0.01 * ci)
                a_init = project_simplex(a_init)

            a_opt, obj = adam_optimize(a_init, time_limit=t_each,
                                       lr=cfg['lr'], tau_start=cfg['tau_start'],
                                       tau_end=cfg['tau_end'],
                                       noise_scale=cfg.get('noise_scale', 0.0))
            try_seq(a_opt, f"adam_n{n}_c{ci}")
            if a_opt is not None and obj < 2.0 * n * fast_convolve(a0).max():
                a0 = a_opt.copy()

        # Subgradient polish
        rem = min(deadline - time.monotonic() - 100, t_each)
        if rem > 15:
            a_opt, obj = subgradient_optimize(a0, time_limit=rem,
                                              lr_init=0.004, mom=0.94,
                                              top_k=n // 5, temp=0.00008)
            try_seq(a_opt, f"sg_n{n}")

    print(f"\n  Phase 2 done, best={best_score:.8f}")

    # === Phase 3: Deep optimization on overall best ===
    print("\n=== Phase 3: Deep Optimization ===")
    remaining = deadline - time.monotonic() - 80

    if remaining > 50 and best_seq is not None:
        n_best = len(best_seq)
        print(f"  n={n_best}, remaining={remaining:.0f}s")

        # Long Adam with very slow annealing
        t_adam = min(remaining * 0.35, 350)
        for attempt in range(2):
            if time.monotonic() > deadline - 80:
                break
            noise = 0.003 * (attempt + 1) if attempt > 0 else 0.0
            a_init = best_seq + rng.standard_normal(n_best) * noise
            a_init = project_simplex(a_init)
            t_each = t_adam / (2 - attempt)
            a_opt, obj = adam_optimize(a_init, time_limit=t_each,
                                       lr=0.015, tau_start=0.002, tau_end=0.00005,
                                       beta1=0.9, beta2=0.9995)
            try_seq(a_opt, f"deep_adam_{attempt}")

        # FW polish
        rem = deadline - time.monotonic() - 60
        if rem > 30:
            a_opt, obj = fw_polish(best_seq, time_limit=min(rem * 0.4, 80), top_k=50)
            try_seq(a_opt, "deep_fw")

        # Subgradient fine
        rem = deadline - time.monotonic() - 50
        if rem > 20:
            a_opt, obj = subgradient_optimize(best_seq, time_limit=min(rem * 0.5, 60),
                                              lr_init=0.002, mom=0.95,
                                              top_k=n_best // 3, temp=0.00003)
            try_seq(a_opt, "deep_sg")

    print(f"\n  Phase 3 done, best={best_score:.8f}")

    # === Phase 4: Final local search ===
    print("\n=== Phase 4: Final Local Search ===")
    remaining = deadline - time.monotonic() - 20

    if remaining > 10 and best_seq is not None:
        n_best = len(best_seq)
        scale = 0.003
        for _ in range(500):
            if time.monotonic() > deadline - 20:
                break
            a_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            obj = 2.0 * n_best * fast_convolve(a_pert).max()
            if obj < 2.0 * n_best * fast_convolve(best_seq).max() - 1e-12:
                cur_score = float(evaluate_sequence(a_pert.tolist()))
                if np.isfinite(cur_score) and cur_score < best_score:
                    best_score = cur_score
                    best_seq = a_pert.copy()
                    print(f"  Local improve: {best_score:.8f}")
                scale = max(0.0005, scale * 0.92)
            else:
                scale = min(0.01, scale * 1.02)

    # Final
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK: {best_score:.10f}")

    return [float(x) for x in best_seq]
