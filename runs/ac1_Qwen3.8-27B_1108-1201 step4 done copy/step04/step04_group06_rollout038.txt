```python
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g) (full linear convolution)."""
    n = len(a)
    m = 2 * n - 1
    C = np.zeros((m, n))
    for j in range(n):
        C[j:j+n, j] = a
    return C

def objective(a):
    """Compute the objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def lp_direction(a, M, C, top_k=50):
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. High-precision LP-based optimization with quadratic line search
2. Log-sum-exp smoothed gradient with Nesterov acceleration
3. Active-set-adaptive constraint selection
4. Multi-length exploration with parent interpolation
5. Coordinate descent micro-polishing
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g) (full linear convolution)."""
    n = len(a)
    m = 2 * n - 1
    C = np.zeros((m, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def compute_objective(a):
    """Compute 2*n*max(conv)/(sum)^2 for non-negative a."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def solve_lp(a, C, M):
    """Solve LP: max sum(g) s.t. C@g <= M, g >= 0. Returns (g, success, gsum)."""
    n = len(a)
    m = 2 * n - 1
    c_vec = -np.ones(n)
    b_vec = np.full(m, M)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c_vec, A_ub=C, b_ub=b_vec, bounds=bounds,
                         method='higer', 'presolve': True}
        if result.success:
            g = result.x
            gs = g.sum()
            if gs > 1e-10:
                return g, True, gs
    except Exception:
        pass
    # Fallback to highs
    try:
        result = linprog(c_vec, A_ub=C, b_ub=b_vec, bounds=bounds,
                         method='highs',
                         options={'presolve': True,
                                  'dual_feasibility_tolerance': 1e-12,
                                  'primal_feasibility_tolerance': 1e-12,
                                  'time_limit': 10})
        if result.success:
            g = result.x
            gs = g.sum()
            if gs > 1e-10:
                return g, True, gs
    except Exception:
        pass
    return None, False, 0.0


def lse_gradient_direct(a, conv, beta):
    """Compute gradient of (1/beta)*log(sum_k exp(beta*conv[k])) w.r.t. a using O(n^2) direct computation."""
    n = len(a)
    m = 2 * n - 1
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(m)
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= w_sum

    grad = np.zeros(n)
    # For each i, grad[i] = (1/beta) * sum_k w[k] * d(conv[k])/d(a[i])
    # d(conv[k])/d(a[i]) = a[k-i] if 0<=k-i<n and k-i != i
    #                     = 2*a[i] if k-i == i (i.e. k=2i)
    #                     = 0 otherwise (i.e. k-i not in [0,n))
    for i in range(n):
        k_lo = max(0, i)
        k_hi = min(m - 1, i + n - 1)
        total = 0.0
        for k in range(k_lo, k_hi + 1):
            j = k - i
            if j == i:
                total += w[k] * 2.0 * a[i]
            else:
                total += w[k] * a[j]
        grad[i] = total / beta
    return grad


def lse_gradient_fft(a, conv, beta):
    """Compute LSE gradient using FFT-based cross-correlation for speed."""
    n = len(a)
    m = 2 * n - 1
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(m)
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= w_sum

    # grad[i] = (1/beta) * sum_k w[k] * a[k-i] (for valid k-i in [0,n))
    #         + (1/beta) * sum_k w[k] * a[k-i] (for k-i != i, valid)
    # = (1/beta) * [2 * xcorr[i] - w[2i] * a[i] * I(2i < m)]
    # where xcorr[i] = sum_j a[j] * w[i+j] (j from 0 to n-1, i+j from 0 to m-1)

    # xcorr[i] = (a_flipped * w)[i + n - 1] where a_flipped = a[::-1]
    # Actually: xcorr[i] = sum_{j=0}^{n-1} a[j] * w[i+j]
    # Pad a to length m, then: xcorr = ifft(conj(fft(a_pad, 2*m)) * fft(w, 2*m)) doesn't work directly
    # Let's use: xcorr[i] = conv(w, a_rev)[i + n - 1] where a_rev = a[::-1]
    # conv(w, a_rev)[p] = sum_j w[j] * a_rev[p-j] = sum_j w[j] * a[n-1-p+j]
    # Setting p = i + n - 1: sum_j w[j] * a[n-1-i-n+1+j] = sum_j w[j] * a[j-i]
    # Hmm, that gives sum_j w[j]*a[j-i], not sum_j a[j]*w[i+j]

    # Let me use: xcorr[i] = sum_j a[j] * w[i+j]
    # = ifft(fft(a, 2*n) * conj(fft(w_flipped, 2*n)))... this is getting complicated.
    # For n=768, O(n^2) is fine (~600K ops). Let's just use the direct method.
    return lse_gradient_direct(a, conv, beta)


def lp_optimize_precise(a0, time_limit, verbose=False):
    """
    High-precision LP-based optimization with quadratic line search.
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    max_no_improve = 60
    perturb_count = 0
    rng = np.random.default_rng(42 + n)

    while time.monotonic() - t0 < time_limit - 5:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        M = conv.max()
        C = build_conv_matrix(a)

        # Solve LP with tight constraint
        g, success, g_sum = solve_lp(a, C, M * (1.0 + 1e-10))

        if not success or g_sum <= 1.0 + 1e-10:
            no_improve += 1
        else:
            d = g - a
            norm_d = np.linalg.norm(d)
            if norm_d < 1e-14:
                no_improve += 1
            else:
                # Quadratic line search
                peak_idx = int(np.argmax(conv))
                conv_ad = C @ d
                lin = 2.0 * conv_ad[peak_idx]
                dd = fast_convolve(d)
                quad = dd[peak_idx]

                # Optimal unconstrained step
                if quad > 1e-15:
                    t_quad = max(0.0, -lin / (2.0 * quad))
                elif lin < -1e-10:
                    t_quad = 1.0
                else:
                    t_quad = 0.1

                # Max step for non-negativity
                neg_mask = d < -1e-15
                if np.any(neg_mask):
                    t_max_nneg = (a[neg_mask] / (-d[neg_mask])).min()
                else:
                    t_max_nneg = 2.0

                t_max_nneg = min(t_max_nneg * 0.999, 2.0)

                # Generate candidates
                candidates = []
                if t_quad > 1e-6 and t_quad < t_max_nneg:
                    candidates.extend([t_quad, t_quad * 0.7, t_quad * 1.3,
                                       t_quad * 0.5, t_quad * 1.5])
                candidates.extend([0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01])
                if t_max_nneg > 0.01:
                    candidates.extend([t_max_nneg * 0.9, t_max_nneg * 0.5, t_max_nneg * 0.3])

                candidates = sorted(set([max(1e-6, min(c, t_max_nneg)) for c in candidates if c > 1e-7]))

                best_t = 0.0
                best_obj_try = best_obj
                best_a_try = a.copy()

                for t in candidates:
                    a_try = a + t * d
                    a_try = np.maximum(a_try, 0.0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        continue
                    a_try = a_try / s_try
                    conv_try = fast_convolve(a_try)
                    obj_try = 2.0 * n * conv_try.max()
                    if obj_try < best_obj_try - 1e-14:
                        best_obj_try = obj_try
                        best_a_try = a_try.copy()
                        best_t = t

                if best_t > 0:
                    a = best_a_try
                    conv = fast_convolve(a)
                    cur_obj = 2.0 * n * conv.max()
                    if cur_obj < best_obj - 1e-14:
                        best_obj = cur_obj
                        best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1

        iteration += 1

        if verbose and iteration % 200 == 0:
            print(f"    LP iter {iteration}, best_obj={best_obj:.8f}, t={time.monotonic()-t0:.1f}s")

        # Handle stagnation
        if no_improve >= max_no_improve:
            scale = 0.001 + 0.003 * min(perturb_count, 10)
            noise = rng.standard_normal(n) * scale
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            else:
                a = best_a.copy()
            conv = fast_convolve(a)
            no_improve = 0
            perturb_count += 1
            max_no_improve = min(200, 60 + 30 * perturb_count)

            if perturb_count > 60:
                break

            # Every 10th perturbation, try LSE gradient for a few steps
            if perturb_count % 10 == 0:
                beta = 30.0 + 5.0 * perturb_count
                lr = 0.005
                v = np.zeros(n)
                for _ in range(20):
                    conv_l = fast_convolve(a)
                    grad_l = lse_gradient_direct(a, conv_l, beta)
                    v = 0.9 * v - lr * grad_l
                    a_new = a + v
                    a_new = np.maximum(a_new, 0.0)
                    s_new = a_new.sum()
                    if s_new > 1e-15:
                        a = a_new / s_new
                    else:
                        a = best_a.copy()
                        break
                    lr *= 0.95
                conv = fast_convolve(a)
                cur_obj = 2.0 * n * conv.max()
                if cur_obj < best_obj - 1e-14:
                    best_obj = cur_obj
                    best_a = a.copy()

    return best_a, best_obj


def nesterov_lse_optimize(a0, time_limit, beta_init=20.0, lr_init=0.008, momentum=0.92):
    """Nesterov-accelerated gradient descent on LSE surrogate."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    y = a.copy()
    best_a = a.copy()
    best_obj = compute_objective(a)

    t0 = time.monotonic()
    beta = beta_init
    lr = lr_init
    mom = momentum
    no_improve = 0

    while time.monotonic() - t0 < time_limit - 2:
        y = a + mom * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        else:
            y = a.copy()

        conv_y = fast_convolve(y)
        grad = lse_gradient_direct(y, conv_y, beta)

        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
            v = np.zeros(n)
            break

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.999
        beta *= 1.005
        mom = min(0.97, mom + 0.0001)

        if no_improve >= 300:
            break

    return best_a, best_obj


def coordinate_descent_polish(a0, time_limit):
    """Fine-grained coordinate descent: adjust pairs of elements."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, compute_objective(a)
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(99999)
    no_improve = 0

    while time.monotonic() - t0 < time_limit - 1:
        # Focus on positions near the peak
        peak_idx = int(np.argmax(conv))
        # Positions that contribute to the peak
        i_start = max(0, peak_idx - n + 1)
        i_end = min(n - 1, peak_idx)
        peak_positions = np.arange(i_start, i_end + 1)

        # Pick random positions, biasing toward peak-contributing ones
        use_peak = rng.random() < 0.7
        if use_peak and len(peak_positions) > 1:
            i = peak_positions[rng.integers(0, len(peak_positions))]
        else:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # Try small transfers
        base = min(a[i], a[j])
        for frac in [0.002, 0.005, 0.01, 0.02, 0.001, -0.002, -0.005, -0.01]:
            delta = frac * base
            if abs(delta) < 1e-15:
                continue
            if a[i] + delta < 0 or a[j] - delta < 0:
                continue
            a_try = a.copy()
            a_try[i] += delta
            a_try[j] -= delta
            conv_try = fast_convolve(a_try)
            obj_try = 2.0 * n * conv_try.max()
            if obj_try < best_obj - 1e-12:
                a = a_try
                conv = conv_try
                best_obj = obj_try

def generate_initializations(n, parent, rng):

def extend_sequence(a_short, n_target):
    """Extend a shorter sequence to target length via interpolation."""
    n_short = len(a_short)
    s = a_short.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    a_norm = a_short / s
    if n_short == n_target:
        return a_norm.copy()
    a_ext = np.interp(np.linspace(0, 1, n_target), np.linspace(0, 1, n_short), a_norm)
    a_ext = np.maximum(a_ext, 0)
    ses = a_ext.sum()
    if ses > 1e-15:

    return a_ext
```
