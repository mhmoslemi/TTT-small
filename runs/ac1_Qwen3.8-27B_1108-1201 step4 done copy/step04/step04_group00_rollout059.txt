        je = min(n - 1, k)

        A_ub[idx, cols] = f[rows_f]

    bounds = [(0, None)] * n

    if res.success:

                    mask[j] = False
                    f_trial[mask] = f[mask] * (1.0 - val) / rest_sum
                f_trial = f_trial / f_trial.sum()
                return fast_convolve(f_trial).max()
            
            v1 = obj_at(c1)
            v2 = obj_at(c2)
            cur_obj = fast_convolve(f).max()

                if v1 < v2:
                    hi = c2
                    c2 = c1
                    v2 = v1
                    c1 = hi - gr * (hi - lo)
                    v1 = obj_at(c1)
                else:
                    lo = c1
                    c1 = c2
                    v1 = v2
                    c2 = lo + gr * (hi - lo)
                    v2 = obj_at(c2)
            
            best_c = (lo + hi) / 2
            best_v = obj_at(best_c)
            
            if best_v < cur_obj - 1e-14 and best_c > 1e-14:

                    best_obj = best_v
                    best_f = f.copy()
                    improved = True
        
        if not improved and passes > 2:
            break
    
    return best_f, best_obj

def fast_convolve(a):

def project_simplex(v):

        return np.ones(n) / n

            # Golden section search on f[j] value
            lo = 0.0
            hi = min(0.99, old_val * 4.0 + 0.05 / n)
            
            if hi <= lo:
                hi = lo + 0.01 / n
            
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            
            def eval_coord(val):
                f_trial = f.copy()
                f_trial[j] = val
                if val < 1.0 - 1e-12:
                    scale = (1.0 - val) / rest_sum
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                return fast_convolve(f_trial).max()
            
            v1 = eval_coord(c1)
            v2 = eval_coord(c2)
            
            for _ in range(10):
                if v1 < v2:
                    hi = c2
                    c2 = c1
                    v2 = v1
                    c1 = hi - gr * (hi - lo)
                    v1 = eval_coord(c1)
                else:
                    lo = c1
                    c1 = c2
                    v1 = v2
                    c2 = lo + gr * (hi - lo)
                    v2 = eval_coord(c2)
            
            best_c = (lo + hi) / 2.0
            best_v = eval_coord(best_c)
            
            if best_v < cur_obj - 1e-13 and best_c > 1e-14 and best_c < 1.0 - 1e-10:
                f[j] = best_c
                scale = (1.0 - best_c) / rest_sum
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f = f / f.sum()
                cur_obj = best_v
                
                if best_v < best_obj - 1e-12:
                    best_obj = best_v
                    best_f = f.copy()
                    improved = True
        
        if not improved and passes > 3:
            break
    
    return best_f, best_obj


def simulated_annealing(f0, time_limit, verbose=False):
    """SA on the simplex: random coordinate perturbations with annealing schedule."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    cur_obj = best_obj
    
    t0 = time.monotonic()
    total_time = time_limit
    rng = np.random.default_rng(999)
    
    # Temperature schedule: exponential cooling
    T_init = 0.0005
    T_final = 0.000005
    cool_rate = 0.9997
    
    T = T_init
    steps = 0
    
    while time.monotonic() - t0 < time_limit:
        T = max(T_final, T_init * (cool_rate ** steps))
        steps += 1
        
        # Pick random coordinate
        j = rng.integers(0, n)
        old_val = f[j]
        if old_val < 1e-14:
            continue
        
        rest_sum = 1.0 - old_val
        if rest_sum < 1e-12:
            continue
        
        # Perturb
        noise = rng.standard_normal() * (0.05 * old_val + 0.001 / n)
        new_val = old_val + noise
        new_val = max(0.0, min(new_val, 1.0 - 1e-10))
        
        if abs(new_val - old_val) < 1e-15:
            continue
        
        # Compute new objective
        f_trial = f.copy()
        f_trial[j] = new_val
        scale = (1.0 - new_val) / rest_sum
        mask = np.ones(n, dtype=bool)
        mask[j] = False
        f_trial[mask] *= scale
        f_trial = f_trial / f_trial.sum()
        
        new_obj = fast_convolve(f_trial).max()
        delta = new_obj - cur_obj
        
        # Metropolis acceptance
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            f = f_trial
            cur_obj = new_obj
            if new_obj < best_obj - 1e-13:
                best_obj = new_obj
                best_f = f.copy()
    
    return best_f, best_obj


def pairwise_cd(f0, time_limit, n_pairs=50):
    """Pairwise coordinate descent: optimize pairs of coordinates jointly."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    cur_obj = best_obj
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Pick random pair
            j1, j2 = rng.choice(n, size=2, replace=False)
            v1, v2 = f[j1], f[j2]
            if v1 < 1e-14 or v2 < 1e-14:
                continue
            
            rest_sum = 1.0 - v1 - v2
            if rest_sum < 1e-12:
                continue
            
            # 2D search: vary v1 and v2
            # Parametrize: new_v1 = v1 * (1 + dx), new_v2 = v2 * (1 + dy)
            # Then scale rest by (1 - new_v1 - new_v2) / rest_sum
            best_dx, best_dy = 0.0, 0.0
            best_pair_obj = cur_obj
            
            for dx in [-0.5, -0.2, -0.1, 0.0, 0.1, 0.2, 0.5]:
                for dy in [-0.5, -0.2, -0.1, 0.0, 0.1, 0.2, 0.5]:
                    nv1 = v1 * (1.0 + dx)
                    nv2 = v2 * (1.0 + dy)
                    if nv1 < 0 or nv2 < 0 or nv1 + nv2 > 1.0 - 1e-10:
                        continue
                    f_trial = f.copy()
                    f_trial[j1] = nv1
                    f_trial[j2] = nv2
                    scale = (1.0 - nv1 - nv2) / rest_sum
                    mask = np.ones(n, dtype=bool)
                    mask[j1] = False
                    mask[j2] = False
                    f_trial[mask] *= scale
                    f_trial = f_trial / f_trial.sum()
                    obj = fast_convolve(f_trial).max()
                    if obj < best_pair_obj - 1e-13:
                        best_pair_obj = obj
                        best_dx, best_dy = dx, dy
            
            if best_dx != 0.0 or best_dy != 0.0:
                nv1 = v1 * (1.0 + best_dx)
                nv2 = v2 * (1.0 + best_dy)
                f[j1] = nv1
                f[j2] = nv2
                scale = (1.0 - nv1 - nv2) / rest_sum
                mask = np.ones(n, dtype=bool)
                mask[j1] = False
                mask[j2] = False
                f[mask] *= scale
                f = f / f.sum()
                cur_obj = best_pair_obj
                
                if best_pair_obj < best_obj - 1e-12:
                    best_obj = best_pair_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Linear interpolation to change sequence length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: accelerated FW, SA escape, coordinate descent,
    pairwise updates, and diverse restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score
    
    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            try_seq(parent, "parent")
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(768) / 768.0
        try_seq(parent, "uniform_768")
    
    n_parent = len(parent)
    print(f"\nStarting optimization with n={n_parent}, initial score={best_score:.6f}")
    
    # === Phase 1: Accelerated FW on parent (400s) ===
    print("\n=== Phase 1: Accelerated FW (n={}) ===".format(n_parent))
    t1 = min(400, deadline - time.monotonic() - 600)
    if t1 > 60:
        a_opt, _ = accelerated_fw(parent, time_limit=t1, top_K=250, verbose=True)
        try_seq(a_opt, "acfw_p1")
    
    # === Phase 2: Coordinate descent polish (200s) ===
    print(f"\n=== Phase 2: CD Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 40:
            t_cd = min(rem, 200)
            a_opt, _ = coordinate_descent_v2(best_seq, time_limit=t_cd, verbose=True)
            try_seq(a_opt, "cd_p2")
        
        # Quick pairwise
        rem = deadline - time.monotonic() - 200
        if rem > 30:

    # === Phase 3: SA escape + FW recovery (400s total) ===
    print(f"\n=== Phase 3: SA + FW Recovery (best={best_score:.6f}) ===")
    if best_seq is not None:

        for restart in range(4):

            if rem < 60:
                break
            
            # Perturb
            noise_scale = 0.003 + 0.002 * restart
            noise = rng.standard_normal(n_best) * noise_scale
            f_pert = project_simplex(best_seq + noise)
            
            # SA phase
            t_sa = min(80, rem * 0.4)
            if t_sa > 20:

            # FW recovery
            rem = deadline - time.monotonic() - 150
            if rem > 30:

                t_fw = min(120, rem * 0.5)
                a_opt, _ = accelerated_fw(base_sa, time_limit=t_fw, top_K=200)

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
