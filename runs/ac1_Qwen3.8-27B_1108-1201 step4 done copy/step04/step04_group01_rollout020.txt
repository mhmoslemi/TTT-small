Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-start initialization with diverse patterns
2. Coarse-to-fine LSE optimization
3. Exact subgradient with Nesterov acceleration
4. Coordinate descent for final polish
5. Systematic length exploration

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_at_max(a):
    """Compute subgradient of max_k (a*a)_k at point a.
    Returns (sg, k_max) where sg is the subgradient vector."""
    n = len(a)
    conv = fast_convolve(a)
    k_max = int(np.argmax(conv))
    sg = np.zeros(n)
    lo = max(0, k_max - n + 1)
    hi = min(n - 1, k_max)
    js = np.arange(lo, hi + 1)
    sg[js] = 2.0 * a[k_max - js]
    return sg, k_max, conv

def lse_optimize(f0, time_limit, beta_start=20.0, beta_end=5000.0, lr_scale=2.0):
    """Log-sum-exp smoothed gradient descent with adaptive parameters and backtracking."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = lr_scale / n
    momentum = 0.90
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    best_lr = lr

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)

        # Numerically stable log-sum-exp gradient
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w_sum = exp_bc.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w = exp_bc / w_sum

        # Gradient: grad[i] = 2 * sum_m w[m] * a_{m-i} (for valid m-i)
        # This is 2 * (w * flip(f)) circular, but we need linear
        # Use FFT
        f_flipped = f[::-1].copy()
        total_len = (2 * n - 1) + n - 1
        fft_size = 1 << (total_len - 1).bit_length()
        W = np.fft.rfft(w, fft_size)
        F = np.fft.rfft(f_flipped, fft_size)
        conv_w = np.fft.irfft(W * F, fft_size)
        grad = 2.0 * conv_w[n - 1:n - 1 + n]

        # Backtracking line search
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 100
            break

        # Try several step sizes
        best_alpha = 0.0
        best_val = best_obj
        current_obj = conv.max()
        
        # Coarse search
        for alpha_mult in [3.0, 2.0, 1.0, 0.5, 0.25, 0.1]:
            alpha = alpha_mult * lr
            f_trial = project_simplex(f - alpha * grad)
            val = fast_convolve(f_trial).max()
            if val < current_obj:
                best_alpha = alpha
                best_val = val
                current_obj = val
        
        if best_alpha > 0:
            f = project_simplex(f - best_alpha * grad)
            # Update momentum
            v = momentum * v - best_alpha * grad
            f = project_simplex(f + v * 0.1)  # slight momentum blend
        else:
            # Reduce learning rate
            lr *= 0.8
            no_improve += 1

        conv = fast_convolve(f)
        obj = conv.max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:

def subgrad_nesterov(f0, time_limit, lr_scale=1.0):
    """Subgradient method with Nesterov acceleration for max(conv(a,a)) on simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    y = f.copy()
    t_param = 1.0
    lr = lr_scale / np.sqrt(n)
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        sg, k_max, conv = subgradient_at_max(y)
        
        # Adaptive step size
        lr_t = lr / np.sqrt(it + 1)
        
        f_new = project_simplex(y - lr_t * sg)
        
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param)) / 2.0
        y = f_new + (t_param / t_new - 1.0) * (f_new - f)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y / ys
        t_param = t_new
        f = f_new

        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                # Perturb and restart
                rng = np.random.default_rng(it + 12345)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                y = f.copy()
                t_param = 1.0
                no_improve = 0

        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG-N it={it}, obj={best_obj:.12f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj

def coordinate_descent(f0, time_limit, n_sweeps=5):
    """Coordinate descent: optimize one coordinate at a time."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            # Try several values for f[j]
            # The optimal x minimizes max_k (a*a)_k / (sum)^2
            # With sum = S - old_val + x
            # We search over x in [0, 1] (since f is on simplex)
            
            # Coarse grid
            best_x = old_val
            best_score = best_obj * (s ** 2)  # work with unnormalized for speed
            
            base_sum = s - old_val
            # Precompute the base convolution (without the j-th element)
            f_base = f.copy()
            f_base[j] = 0.0
            conv_base = fast_convolve(f_base)
            
            # For each k, (a*a)_k with a_j = x:
            # If 0 <= k-j < n and k-j != j: add 2*x*f_base[k-j]
            # If k == 2*j: add x^2
            # (Plus the cross terms are already in conv_base... wait no)
            
            # Actually conv_base already has a_j = 0, so:
            # (a*a)_k with a_j = x = conv_base[k] + x * (terms involving a_j)
            # Terms: x*f_{k-j} + f_{k-j}*x if k-j != j, or x^2 if k-j == j
            # But we need to be careful: conv_base[k] = sum_i f_i * f_{k-i} where f_j=0
            # So we're missing: f_j*f_{k-j} + f_{k-j}*f_j (if k-j != j) or f_j*f_j (if k-j==j)
            # = 2*x*f_{k-j} (if 0<=k-j<n and k-j != j) or x^2 (if k-j==j, i.e., k=2j)
            
            # Let delta[k] = additional contribution to (a*a)_k from setting a_j = x
            # delta[k] = 2*x*f[k-j] if 0<=k-j<n and k-j != j
            #           = x^2 if k == 2*j
            #           = 0 otherwise
            
            # For efficiency, let's compute the coefficient for x in each k:
            # lin_coeff[k] = 2*f[k-j] if 0<=k-j<n and k-j != j, else 0
            # quad_coeff = 1 if k == 2*j, else 0 (only one k has this)
            
            lin_coeff = np.zeros(2*n-1)
            lo = max(0, j)  # k-j >= 0 => k >= j
            hi = min(2*n-2, j+n-1)  # k-j <= n-1 => k <= j+n-1
            ks = np.arange(lo, hi+1)
            for k in ks:
                kj = k - j
                if kj != j:  # not the quadratic case
                    lin_coeff[k] = 2.0 * f[kj]
            # Quadratic case: k = 2j
            quad_k = 2 * j
            if 0 <= quad_k < 2*n-1:
                pass  # will handle separately
            
            # Objective for a given x:
            # max_k (conv_base[k] + lin_coeff[k]*x + (1 if k==2j else 0)*x^2) / (base_sum + x)^2
            
            # Search over x in [0, min(1, old_val*3)] (don't go too far)
            x_max = min(1.0, max(old_val * 3, 0.01))
            
            # Grid search
            xs = np.linspace(0, x_max, 20)
            best_x = old_val
            best_local = np.inf
            
            for x in xs:
                conv_x = conv_base + lin_coeff * x
                if 0 <= quad_k < 2*n-1:
                    conv_x[quad_k] += x * x
                total = base_sum + x
                if total < 1e-10:
                    continue
                score = conv_x.max() / (total * total)
                if score < best_local:
                    best_local = score
                    best_x = x
            
            # Refine around best_x
            if best_x > 0 and best_x < x_max:
                x_lo = max(0, best_x - (x_max/20))
                x_hi = min(x_max, best_x + (x_max/20))
                xs = np.linspace(x_lo, x_hi, 30)
                for x in xs:
                    conv_x = conv_base + lin_coeff * x
                    if 0 <= quad_k < 2*n-1:
                        conv_x[quad_k] += x * x
                    total = base_sum + x
                    if total < 1e-10:
                        continue
                    score = conv_x.max() / (total * total)
                    if score < best_local:
                        best_local = score
                        best_x = x
            
            f[j] = best_x
            s = base_sum + best_x
            f = f / s  # renormalize
            # This changes the objective, so recompute
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()

    return best_f, best_obj

def upsample(f, new_n):
    """Upsample by interpolation."""
    n = len(f)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new = f_new / s
    return f_new

def downsample(f, new_n):
    """Downsample by averaging blocks."""
    n = len(f)
    if new_n >= n:
        return f.copy()
    # Divide into new_n blocks and average
    block_size = n / new_n
    result = np.zeros(new_n)
    for i in range(new_n):
        start = int(i * block_size)
        end = int((i + 1) * block_size)
        if end > n:
            end = n
        result[i] = f[start:end].mean()
    s = result.sum()
    if s > 1e-15:
        result = result / s
    return result

def diverse_starts(n, parent=None):
    """Generate diverse starting points of length n."""
    starts = []
    
    if parent is not None:
        # Parent (resized to n if needed)
        if len(parent) == n:
            starts.append(parent.copy())
        else:
            p = upsample(parent, n)
            starts.append(p)
    
    # Uniform
    starts.append(np.ones(n) / n)
    
    # Cosine window
    x = np.linspace(0, 1, n)
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x))
    cos_w = np.maximum(cos_w, 1e-6)
    starts.append(cos_w / cos_w.sum())
    
    # Ramp
    ramp = np.arange(1, n+1, dtype=np.float64)
    starts.append(ramp / ramp.sum())
    
    # Bimodal
    x = np.linspace(0, 1, n)
    bimodal = np.exp(-50*(x-0.3)**2) + np.exp(-50*(x-0.7)**2)
    bimodal = np.maximum(bimodal, 1e-6)
    starts.append(bimodal / bimodal.sum())
    
    # Random
    rng = np.random.default_rng(12345)
    rand = rng.random(n)
    rand = np.maximum(rand, 1e-6)
    starts.append(rand / rand.sum())
    
    # "Golay-like": alternate high/low with some structure
    golay = np.ones(n)
    golay[::2] = 2.0
    golay[1::4] = 0.5
    golay = np.maximum(golay, 1e-6)
    starts.append(golay / golay.sum())
    
    return starts

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Coarse-to-fine LSE gradient descent with backtracking line search

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(weights, f):
    """Compute gradient of LSE-smoothed max: 2 * cross_correlation(weights, f)."""
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def lse_optimize(f0, time_limit, beta_start=20.0, beta_end=5000.0, lr_scale=2.0):
    """LSE-smoothed gradient descent with backtracking line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = lr_scale / n
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    v = np.zeros(n)
    momentum = 0.92

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        current_obj = conv.max()

        # Numerically stable softmax weights
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w_sum = exp_bc.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w = exp_bc / w_sum

        # Gradient via FFT
        grad = compute_lse_grad(w, f)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 100
            break
        grad = grad / grad_norm

        # Backtracking line search with momentum
        best_alpha = 0.0
        best_val = current_obj

        # Try a range of step sizes
        for mult in [4.0, 3.0, 2.0, 1.5, 1.0, 0.75, 0.5, 0.3, 0.15, 0.05]:
            alpha = mult * lr
            # With momentum: f_new = project(f + momentum*v - alpha*grad)
            v_candidate = momentum * v - alpha * grad
            f_trial = project_simplex(f + v_candidate)
            val = fast_convolve(f_trial).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
                v = v_candidate.copy()

        if best_alpha > 0:
            f = project_simplex(f + v)
        else:
            # Shrink learning rate
            lr *= 0.7
            v = np.zeros(n)
            no_improve += 1

        conv_new = fast_convolve(f)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
            lr = min(lr * 1.003, 6.0 / n)
        else:
            no_improve += 1
            if no_improve >= 20:
                lr = max(lr * 0.85, 0.5 / n)
                momentum = max(0.85, momentum - 0.01)
            if no_improve > 120:
                f = best_f.copy()
                v = np.zeros(n)
                beta = min(beta * 2.0, beta_end)
                lr = max(2.0 / n, 0.5 / n)
                momentum = 0.92
                no_improve = 0

        # Gradually increase beta (temperature)
        if beta < beta_end:
            beta *= 1.0015

        if it % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE it={it}, obj={best_obj:.12f}, beta={beta:.0f}, lr={lr:.6f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def subgrad_nesterov(f0, time_limit, lr_scale=1.0):
    """Subgradient method with Nesterov acceleration for exact max(conv(a,a))."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    y = f.copy()
    t_param = 1.0
    base_lr = lr_scale / (n ** 0.6)
    no_improve = 0
    it = 0
    last_obj = best_obj

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        # Compute subgradient at y
        conv_y = fast_convolve(y)
        k_max = int(np.argmax(conv_y))
        sg = np.zeros(n)
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        js = np.arange(lo, hi + 1)
        sg[js] = 2.0 * y[k_max - js]

        # Adaptive step size
        lr_t = base_lr / (it ** 0.4)

        # Projected subgradient step
        f_new = project_simplex(y - lr_t * sg)

        # Nesterov momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param)) / 2.0
        y_new = f_new + ((t_param - 1.0) / t_new) * (f_new - f)
        y_new = np.maximum(y_new, 0.0)
        yt = y_new.sum()
        if yt > 1e-15:
            y_new = y_new / yt
        y = y_new
        t_param = t_new
        f = f_new

        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                # Restart with perturbation
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.0005)
                y = f.copy()
                t_param = 1.0
                base_lr = min(base_lr * 1.2, 3.0 / (n ** 0.6))
                no_improve = 0

        if it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG-N it={it}, obj={best_obj:.12f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def coordinate_descent(f0, time_limit, n_sweeps=4):
    """Coordinate descent exploiting the 1D structure of each coordinate optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    n_conv = 2 * n - 1

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = 0
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            base_sum = s - old_val

            # Compute base convolution (with f[j] = 0)
            f[j] = 0.0
            conv_base = fast_convolve(f)
            f[j] = old_val

            # Precompute linear coefficients
            # (a*a)_k = conv_base[k] + 2*x*f[k-j] (if 0<=k-j<n, k-j!=j)
            #                            + x^2 (if k = 2j)
            lin_coeff = np.zeros(n_conv)
            lo_k = max(0, j)
            hi_k = min(n_conv - 1, j + n - 1)
            for k in range(lo_k, hi_k + 1):
                kj = k - j
                if kj != j:
                    lin_coeff[k] = 2.0 * f[kj]
            quad_k = 2 * j
            has_quad = (0 <= quad_k < n_conv)

            # 1D search: minimize max_k(conv_base[k] + lin_coeff[k]*x + quad*x^2) / (base_sum+x)^2
            # Grid search with refinement
            x_max = min(1.0, max(old_val * 4, 0.005))
            
            best_x = old_val
            best_score = np.inf

            # Coarse grid (25 points)
            xs = np.linspace(0.0, x_max, 25)
            for x in xs:
                conv_x = conv_base + lin_coeff * x
                if has_quad:
                    conv_x[quad_k] += x * x
                total = base_sum + x
                if total < 1e-10:
                    continue
                val = conv_x.max() / (total * total)
                if val < best_score:
                    best_score = val
                    best_x = x

            # Fine grid around best (30 points)
            if best_x > 0 and best_x < x_max:
                dx = x_max / 25
                x_lo = max(0.0, best_x - dx)
                x_hi = min(x_max, best_x + dx)
                xs2 = np.linspace(x_lo, x_hi, 30)
                for x in xs2:
                    conv_x = conv_base + lin_coeff * x
                    if has_quad:
                        conv_x[quad_k] += x * x
                    total = base_sum + x
                    if total < 1e-10:
                        continue
                    val = conv_x.max() / (total * total)
                    if val < best_score:
                        best_score = val
                        best_x = x

            if abs(best_x - old_val) > 1e-12:
                f[j] = best_x
                s = base_sum + best_x
                improved += 1

        # Renormalize and check improvement
        if s > 1e-15:
            f = f / s
            s = 1.0
        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            print(f"    CD sweep={sweep}, obj={best_obj:.12f}, improved_coords={improved}")

    return best_f, best_obj

def upsample(f, new_n):
    """Upsample by linear interpolation."""
    n = len(f)
    if n == new_n:
        return f.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0.0)
    ss = f_new.sum()
    if ss > 1e-15:
        f_new = f_new / ss
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def downsample(f, new_n):
    """Downsample by block averaging."""
    n = len(f)
    if new_n >= n:
        return f.copy()
    result = np.zeros(new_n)
    for i in range(new_n):
        start = int(i * n / new_n)
        end = int((i + 1) * n / new_n)
        if end > n:
            end = n
        if end > start:
            result[i] = f[start:end].mean()
    ss = result.sum()
    if ss > 1e-15:
        result = result / ss
    else:
        result = np.ones(new_n) / new_n
    return result


def diverse_starts(n, parent=None):
    """Generate diverse starting points of length n."""
    starts = []

    if parent is not None:
        if len(parent) == n:
            starts.append(parent.copy())
        else:
            starts.append(upsample(parent, n))

    # Uniform
    starts.append(np.ones(n) / n)

    # Cosine window
    x = np.linspace(0, 1, n)
    cos_w = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    cos_w = np.maximum(cos_w, 1e-6)
    starts.append(cos_w / cos_w.sum())

    # Triangular (Barker-like)
    tri = 1.0 - 2.0 * np.abs(x - 0.5)
    tri = np.maximum(tri, 0.01)
    starts.append(tri / tri.sum())

    # Bimodal
    bimodal = np.exp(-40.0 * (x - 0.3) ** 2) + np.exp(-40.0 * (x - 0.7) ** 2)
    bimodal = np.maximum(bimodal, 1e-6)
    starts.append(bimodal / bimodal.sum())

    # Random
    rng = np.random.default_rng(42 + n)
    rand = rng.random(n)
    rand = np.maximum(rand, 1e-6)
    starts.append(rand / rand.sum())

    # Alternating pattern
    alt = np.ones(n)
    alt[::2] = 1.8
    alt[1::2] = 0.6
    starts.append(alt / alt.sum())

    return starts

        f_opt, _ = lse_optimize(current_best, time_limit=t1,
                                beta_start=100, beta_end=8000, lr_scale=0.8)
        try_sequence(f_opt, best_tracker)

        current_best = best_tracker[1] if best_tracker[1] is not None else parent
        t2 = t_rem * 0.3
        if t2 > 15:
            f_opt, _ = coordinate_descent(current_best, time_limit=t2, n_sweeps=2)
            try_sequence(f_opt, best_tracker)

        current_best = best_tracker[1] if best_tracker[1] is not None else parent
        t3 = t_rem * 0.25
        if t3 > 10:
            f_opt, _ = lse_optimize(current_best, time_limit=t3,
                                    beta_start=200, beta_end=15000, lr_scale=0.5)
            try_sequence(f_opt, best_tracker)

    # === Final ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            best_score = final_score
        else:
            final_seq = parent
            best_score = evaluate_sequence(parent.tolist())
    else:
        final_seq = np.ones(448) / 448
        best_score = float(evaluate_sequence(final_seq.tolist()))

    print(f"\nFINAL: score={best_score:.8f}, n={len(final_seq)}")
    print(f"Target: 1.5027, Gap: {best_score - 1.5027:.8f}")
    return [float(x) for x in final_seq]
