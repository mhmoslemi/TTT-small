def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build matrix C such that C @ g gives conv(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C

def compute_gradient(a, conv, T, top_k=50):
    """Compute softmax-weighted gradient of the peak autocorrelation."""
    n = len(a)
    if top_k >= 2*n-1:
        k_sel = np.arange(2*n-1)
    else:
        k_sel = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[k_sel]
    scaled = peak_vals / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
