def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, conv, top_k=64, temp=0.001):
    """Subgradient of max(conv) with softmax weighting over top peaks."""
    n = len(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]

    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= wsum

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]

    return sg


def solve_lp_active(f, M, active_set, n_constraints_max=200, time_per_lp=2.0):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in active_set, g >= 0.
    Returns (g, active_set, success).
    """
    n = len(f)
    active_set = list(active_set)

    if len(active_set) > n_constraints_max:
        # Prioritize most violated
        conv_f = fast_convolve(f)
        scores = np.array([conv_f[k] for k in active_set])
        top_idx = np.argsort(scores)[::-1][:n_constraints_max]
        active_set = [active_set[i] for i in top_idx]

    c = -np.ones(n)
    A_ub = np.zeros((len(active_set), n))
    for i, k in enumerate(active_set):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A_ub[i, js:je + 1] = f[k - np.arange(js, je + 1)]
    b_ub = np.full(len(active_set), M)
    bounds = [(0, None)] * n

    t0 = time.monotonic()
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_per_lp})
        if res.success and res.x is not None:
            g = res.x
        else:
            return None, active_set, False
    except Exception:
        return None, active_set, False

    if time.monotonic() - t0 > time_per_lp * 1.5:
        return None, active_set, False

    # Check for new violations
    g_sum = g.sum()
    if g_sum < 1e-15:
        return None, active_set, False

    cross = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        cross[k] = np.dot(f[js:je + 1], g[k - np.arange(js, je + 1)])

    violations = np.where(cross > M * (1 + 1e-8))[0]
    new_violations = [k for k in violations if k not in active_set]

    return g, active_set, True

def local_search(f0, time_limit, verbose=False):
    """Local search: try small perturbations and keep improvements."""
    f = np.maximum(f0.astype(np.float64), 0.0)

    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1:
            break

        # Try random pair exchange
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # Move some mass from i to j (or vice versa)
        delta = f[i] * rng.uniform(0.05, 0.3)
        f_try = f.copy()
        f_try[i] -= delta
        f_try[j] += delta
        if f_try.min() < 0:
            f_try[i] = 0
            f_try[j] = f[i]
        f_try = f_try / f_try.sum()

        val = fast_convolve(f_try).max()
        if val < best_obj - 1e-15:
            best_obj = val
            best_f = f_try.copy()
            f = f_try
            conv = fast_convolve(f)

        # Try small noise
        if rng.random() < 0.3:
            f_try = f.copy()
            f_try[rng.integers(0, n, size=3)] += rng.standard_normal(3) * 0.01
            f_try = np.maximum(f_try, 0)
            f_try = f_try / f_try.sum()
            val = fast_convolve(f_try).max()
            if val < best_obj - 1e-15:
                best_obj = val
                best_f = f_try.copy()
                f = f_try
                conv = fast_convolve(f)

def generate_inits(n, parent=None, rng=None):
    """Generate diverse structured initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = []
    x = np.arange(n, dtype=np.float64)

    inits.append(np.ones(n) / n)

    ramp_d = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(ramp_d / ramp_d.sum())

    ramp_i = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp_i / ramp_i.sum())

    q = (x - n / 2) ** 2 + 1.0
    inits.append(q / q.sum())

    e1 = np.exp(-x / (n / 3.0))
    inits.append(e1 / e1.sum())

    e2 = np.exp(-x[::-1] / (n / 3.0))
    inits.append(e2 / e2.sum())

    c = np.cos(np.pi * (x / max(n - 1, 1) - 0.5)) ** 4 + 0.01
    inits.append(c / c.sum())

    sn = np.sin(np.pi * x / n) + 0.05
    inits.append(sn / sn.sum())

    r = rng.dirichlet(np.ones(n) * 0.3)
    inits.append(r)

    # Triangle
    tri = np.minimum(x + 1, n - x)
    tri = np.maximum(tri, 0.01)
    inits.append(tri / tri.sum())

    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            blended = 0.5 * (px / ps) + 0.5 * (ramp_d / ramp_d.sum())
            inits.append(blended / blended.sum())

    return inits

                f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.001)
                a_opt, _ = sg_polish(f_pert, time_limit=min(rem, 10),
                                     lr=0.0005, mom=0.97,
                                     top_k=max(32, n_best // 3), temp=0.00001)
                try_seq(a_opt, "micro")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
