"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Adam-based smoothed subgradient
descent with softmax peak weighting, LP Frank-Wolfe polishing, coordinate
descent refinement, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f):
    """Build B matrix where (B*g)[k] = (f*g)[k] for cross-convolution."""
    n = len(f)
    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        B[j:j + n, j] = f
    return B


def compute_subgradient(f, conv, top_k, temp):
    """Compute softmax-weighted subgradient of max(conv) w.r.t. f."""
    n = len(f)
    m = len(conv)
    if top_k >= m:
        peaks = np.arange(m)
        pv = conv
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
    
    # Softmax weights
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = int(peaks[i])
        # dc[k]/da[i] = 2*a[k-i] if 0<=k-i<n, else a[k-i] or 0
        # More precisely: dc[k]/da[i] = 2*w[i]*a[k-i] (with a[m]=0 outside [0,n-1])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
    
    return sg


def adam_optimize(f0, time_limit, lr=0.01, beta1=0.9, beta2=0.999,
                  top_k=50, temp=0.001, restart_interval=2000,
                  temp_growth=1.002, verbose=False):
    """Adam optimizer with softmax-weighted subgradient and simplex projection."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    cur_lr = lr
    cur_temp = temp
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    no_improve = 0
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Compute subgradient
        sg = compute_subgradient(f, conv, top_k, cur_temp)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        f_new = f - cur_lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        f_new = project_simplex(f_new)
        
        conv = fast_convolve(f_new)
        obj = conv.max()
        
        # Accept if improvement or within tolerance
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = f_new
        
        # Adaptive temperature
        cur_temp *= temp_growth
        
        # Restart
        if no_improve > restart_interval:
            rng = np.random.default_rng(t * 7919 + 13)
            scale = min(0.01, 0.003 * (no_improve // restart_interval))
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            cur_lr = lr
            no_improve = 0
            conv = fast_convolve(f)
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam iter {t}: obj={best_obj:.10f}, temp={cur_temp:.6f}, elapsed={elapsed:.1f}s")
    
    return best_f, best_obj


def solve_linearized_lp(f, conv_f, B=None):
    """Linearized LP: find direction g that minimizes the linearized peak."""
    n = len(f)
    if B is None:
        B = build_conv_matrix(f)
    
    c_vec = np.zeros(n + 1)
    c_vec[0] = 1.0
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = 2.0 * B
    b_ub = -conv_f
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]
    
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]
    return None


def solve_reduced_lp(f, conv_f, B, top_k=30):
    """Reduced LP using only top_k lags."""
    n = len(f)
    peaks = np.argpartition(conv_f, -top_k)[-top_k:]
    peaks = np.sort(peaks)
    
    c_vec = np.zeros(n + 1)
    c_vec[0] = 1.0
    
    A_ub = np.zeros((top_k, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = 2.0 * B[peaks, :]
    b_ub = -conv_f[peaks]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]
    
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]
    return None


def line_search(f, target, current_obj, max_alpha=1.0):
    """Line search for f + alpha*(target-f)."""
    best_alpha = 0.0
    best_obj = current_obj
    
    alphas = np.array([0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0])
    for alpha in alphas:
        f_new = (1.0 - alpha) * f + alpha * target
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-15:
            continue
        f_new = f_new / s
        obj = fast_convolve(f_new).max()
        if obj < best_obj:
            best_obj = obj
            best_alpha = alpha
    
    if best_alpha > 0:
        lo = max(0.0, best_alpha * 0.7)
        hi = min(max_alpha, best_alpha * 1.3)
        for _ in range(6):
            mid = (lo + hi) / 2.0
            f_new = (1.0 - mid) * f + mid * target
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-15:
                break
            f_new = f_new / s
            obj = fast_convolve(f_new).max()
            if obj < best_obj:
                best_obj = obj
                best_alpha = mid
                lo = mid
            else:
                hi = mid
    
    return best_alpha, best_obj


def lp_frank_wolfe(f0, time_limit, use_reduced=False, reduced_k=25, verbose=False):

    conv = fast_convolve(f)

    it = 0
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        current_obj = conv.max()
        
        if use_reduced:
            g = solve_reduced_lp(f, conv, B, top_k=reduced_k)
        else:
            g = solve_linearized_lp(f, conv, B)
        
        if g is None:
            no_improve += 5
            if no_improve > 50:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                B = build_conv_matrix(f)
                no_improve = 0
            continue
        
        target = f + g
        target = np.maximum(target, 0.0)
        ts = target.sum()
        if ts < 1e-15:
            no_improve += 1
            continue
        target = target / ts
        
        if np.max(np.abs(target - f)) < 1e-11:
            no_improve += 1
            if no_improve > 30:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                B = build_conv_matrix(f)
                no_improve = 0
            continue
        
        alpha, obj_val = line_search(f, target, current_obj)

            f = (1.0 - alpha) * f + alpha * target
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs < 1e-15:
                f = best_f.copy()
            else:
                f = f / fs
            B = build_conv_matrix(f)
            
            if obj_val < best_obj - 1e-14:
                best_obj = obj_val
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}, alpha={alpha:.4f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 60:
            f = project_simplex(best_f + rng.standard_normal(n) * (0.002 + 0.001 * (no_improve // 60)))
            B = build_conv_matrix(f)
            no_improve = 0
    
    return best_f, best_obj


def coordinate_descent(f0, time_limit, rng, verbose=False):
    """Coordinate descent: try adjusting pairs of coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Try adjusting coordinate i
            for delta_scale in [0.001, 0.005, 0.01, 0.02]:
                for sign in [1, -1]:
                    if f[i] < 0 and sign < 0:
                        continue
                    # Find a coordinate j to compensate
                    j_candidates = [j for j in range(n) if j != i and f[j] > 0]
                    if not j_candidates:
                        break
                    # Pick j with largest value (or random)
                    j = int(rng.integers(0, n))
                    if j == i:
                        j = (i + 1) % n
                    
                    f_new = f.copy()
                    delta = sign * delta_scale * f[i] if f[i] > 0 else sign * delta_scale * f.mean()
                    if f_new[i] + delta < -1e-15:
                        continue
                    f_new[i] += delta
                    f_new[j] -= delta
                    if f_new[j] < -1e-15:
                        continue
                    
                    f_new = np.maximum(f_new, 0)
                    s_new = f_new.sum()
                    if s_new < 1e-15:
                        continue
                    f_new = f_new / s_new
                    
                    obj = fast_convolve(f_new).max()
                    if obj < best_obj - 1e-14:
                        best_obj = obj
                        best_f = f_new.copy()
                        f = f_new
                        improved = True
                        no_improve = 0
                        break
        
        if not improved:
            no_improve += 1
            if no_improve > 20:
                # Larger perturbation
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                no_improve = 0
    
    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    x = np.arange(n, dtype=np.float64)
    for sigma_frac in [0.15, 0.25, 0.4]:
        sigma = sigma_frac * n
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / sigma) ** 2)
        inits.append(g / g.sum())
    
    # Asymmetric hump
    for center_frac in [0.3, 0.4, 0.6, 0.7]:
        c = center_frac * n
        g = np.exp(-0.5 * ((x - c) / (0.2 * n)) ** 2)
        inits.append(g / g.sum())
    
    # Random
    inits.append(rng.dirichlet(np.ones(n) * 0.1))
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    
    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits
