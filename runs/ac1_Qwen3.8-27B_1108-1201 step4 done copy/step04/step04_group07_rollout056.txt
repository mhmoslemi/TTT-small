def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    x0 = np.log(a0 / a0.sum())
    
    t0 = time.monotonic()
    
    def fun_and_grad(x):

            raise KeyboardInterrupt
        # Clip x to prevent overflow
        x = np.clip(x, -50, 50)
        return smoothed_obj_and_grad(x, beta)
    
    try:
        res = minimize(fun_and_grad, x0, jac=True, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'maxfun': 500, 'maxls': 50})
    except KeyboardInterrupt:
        return a0, float('inf')
    except Exception:
        return a0, float('inf')
    
    if res.success or res.fun < float('inf'):
        x_opt = np.clip(res.x, -50, 50)
        a_opt = np.exp(x_opt - x_opt.max())
        a_opt = a_opt / a_opt.sum()
        return a_opt, float(res.fun)
    return a0, float('inf')

        conv = fast_convolve(a)
        M = conv.max()

        if top_k < len(conv):
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        else:
            peaks = np.arange(len(conv))
        
        pv = conv[peaks]
        # Softmax weights (temperature 1)
        w = pv / pv.sum()
        
        # Gradient: grad_a[j] = sum_k w_k * (a[k-j] + a[j-k])
        # Using FFT
        w_full = np.zeros(2 * n - 1)
        w_full[peaks] = w
        
        fft_len = 1 << (3 * n - 1).bit_length()
        Wf = np.fft.rfft(w_full, fft_len)
        Af = np.fft.rfft(a, fft_len)
        Ar = a[::-1]
        Arf = np.fft.rfft(Ar, fft_len)
        
        conv_WA = np.fft.irfft(Wf * Af, fft_len)[:3*n-2]
        conv_WAr = np.fft.irfft(Wf * Arf, fft_len)[:3*n-2]
        
        term_B = conv_WA[:n]
        term_A = conv_WAr[n-1:n-1+n]
        grad_a = term_A + term_B
        
        # Mirror descent update: a_j <- a_j * exp(-lr * grad_a_j)
        a = a * np.exp(-lr * grad_a / max(grad_a.max(), 1e-15))
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        lr *= 0.999
        if lr < 0.01:
            lr = 0.01
    
    return best_a, best_obj


def solve_maxsum_lp(f, M):
    """Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None and np.any(res.x > 0):
        return res.x
    return None

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using annealed L-BFGS on a smoothed
log-sum-exp objective with FFT-based gradients, combined with Frank-Wolfe
(max-sum LP oracle), mirror descent, and local search polishing.
Key innovation: efficient gradient computation via FFT for the log-sum-exp
smoothing of the max-convolution, enabling rapid convergence through
scipy L-BFGS-B in log-parameterization space.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_a(a, w):
    """
    Compute grad_a[j] = sum_k w[k] * (a[k-j] + a[j-k]) efficiently via FFT.
    a has length n, w has length 2n-1.
    """
    n = len(a)
    fft_len = 1 << (3 * n - 1).bit_length()
    Wf = np.fft.rfft(w, fft_len)
    Af = np.fft.rfft(a, fft_len)
    Ar = a[::-1]
    Arf = np.fft.rfft(Ar, fft_len)

    conv_WA = np.fft.irfft(Wf * Af, fft_len)[:3 * n - 2]
    conv_WAr = np.fft.irfft(Wf * Arf, fft_len)[:3 * n - 2]

    term_B = conv_WA[:n]
    term_A = conv_WAr[n - 1:n - 1 + n]

    return term_A + term_B


def smoothed_obj_and_grad(x, beta):
    """
    Compute smoothed objective and gradient in log-parameterization.
    a = softmax(x), obj = 2n/beta * log(sum_k exp(beta*c_k))
    Returns (obj, grad_x).
    """
    n = len(x)
    x_shifted = x - x.max()
    ex = np.exp(x_shifted)
    a = ex / ex.sum()

    conv = fast_convolve(a)

    # Numerically stable log-sum-exp
    beta_conv = beta * conv
    beta_conv_max = beta_conv.max()
    shifted = beta_conv - beta_conv_max
    exp_shifted = np.exp(shifted)
    log_sum = beta_conv_max + np.log(exp_shifted.sum())
    llse = log_sum / beta

    obj = 2.0 * n * llse

    # Softmax weights
    total = exp_shifted.sum()
    if total < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w = exp_shifted / total

    # Gradient of llse wrt a
    grad_llse_a = compute_grad_a(a, w)

    # Convert to x-space
    obj_factor = 2.0 * n
    grad_obj_a = obj_factor * grad_llse_a
    dot_ga = np.dot(grad_obj_a, a)
    grad_x = a * (grad_obj_a - dot_ga)

    return obj, grad_x


def lbgfgs_smoothed(a0, beta, time_limit, maxiter=200):
    """Optimize using L-BFGS-B on the smoothed objective in log-parameterization."""
    n = len(a0)
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()
    x0 = np.log(a0)
    x0 = x0 - x0.max()

    t0 = time.monotonic()
    eval_count = [0]

    def fun_and_grad(x):
        eval_count[0] += 1
        if time.monotonic() - t0 > time_limit:
            raise KeyboardInterrupt
        x = np.clip(x, -60, 60)
        return smoothed_obj_and_grad(x, beta)

    try:
        res = minimize(fun_and_grad, x0, jac=True, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'maxfun': maxiter * 3,
                                'maxls': 30, 'gtol': 1e-8})
    except KeyboardInterrupt:
        if eval_count[0] > 1:
            # Use last evaluated point - approximate from x0
            return a0, 2.0 * n * fast_convolve(a0).max()
        return a0, float('inf')
    except Exception:
        return a0, float('inf')

    if hasattr(res, 'x') and res.x is not None:
        x_opt = np.clip(res.x, -60, 60)
        a_opt = np.exp(x_opt - x_opt.max())
        s = a_opt.sum()
        if s > 1e-15:
            a_opt = a_opt / s
            return a_opt, 2.0 * n * fast_convolve(a_opt).max()
    return a0, 2.0 * n * fast_convolve(a0).max()


def annealed_lbgfgs(a0, time_limit, betas=None, maxiter_per_beta=100):
    """Annealed L-BFGS: start with low beta (smooth) and increase."""
    if betas is None:
        betas = [5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0, 1000.0, 2000.0]

    n = len(a0)
    a = a0.copy()
    t0 = time.monotonic()

    for i, beta in enumerate(betas):
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 5:
            break
        # Allocate more time to later (higher beta) stages
        frac = 0.5 + 0.5 * (i / len(betas))
        t_alloc = min(remaining * 0.8 * frac, 80)
        if t_alloc < 3:
            t_alloc = 3

        a, obj = lbgfgs_smoothed(a, beta, time_limit=t_alloc,
                                 maxiter=maxiter_per_beta)
        exact_obj = 2.0 * n * fast_convolve(a).max()
        print(f"    Anneal beta={beta:.0f}: exact={exact_obj:.10f} "
              f"(time={time.monotonic()-t0:.1f}s)")

    return a


def mirrored_gradient_descent(a0, time_limit, lr_init=0.05, top_k=None):
    """Mirror descent (multiplicative weights) with peak targeting."""
    n = len(a0)
    if top_k is None:
        top_k = max(10, n // 5)
    a = np.maximum(a0.copy(), 1e-15)
    a = a / a.sum()

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        curr_obj = 2.0 * n * conv.max()

        if curr_obj < best_obj - 1e-13:
            best_obj = curr_obj
            best_a = a.copy()

        # Get top-k peaks
        if top_k < len(conv):
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        else:
            peaks = np.arange(len(conv))

        pv = conv[peaks]
        pmin = pv.min()
        # Use a sharp but not too sharp weighting
        w_local = (pv - pmin + 1e-10)
        w_local = w_local ** 2  # Square to emphasize peaks more
        w = w_local / w_local.sum()

        # Build full weight vector
        w_full = np.zeros(2 * n - 1)
        w_full[peaks] = w

        # Compute gradient
        grad_a = compute_grad_a(a, w_full)

        # Mirror descent update
        grad_max = grad_a.max()
        if grad_max > 1e-15:
            a = a * np.exp(-lr * grad_a / grad_max)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        lr *= 0.9995
        if lr < 0.005:
            lr = 0.005

    return best_a, best_obj


def solve_maxsum_lp(f, M):
    """Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None and np.any(res.x > 0):
        return res.x
    return None


def fw_maxsum(a0, time_limit, verbose=False):
    """Frank-Wolfe with max-sum LP oracle."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(99999)

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        g = solve_maxsum_lp(a, M * 1.01)
        if g is None:
            g = solve_maxsum_lp(a, M * 1.05)
            if g is None:
                g = solve_maxsum_lp(a, M * 1.1)
            if g is None:
                no_improve += 5
                if no_improve > 40:
                    a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
                    no_improve = 0
                continue

        g = g / g.sum()

        if np.max(np.abs(g - a)) < 1e-9:
            no_improve += 1
            if no_improve > 30:
                a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
                no_improve = 0
            continue

        # Line search
        best_alpha = 0.0
        best_val = best_obj
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * a + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = 2.0 * n * fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * a + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = 2.0 * n * fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-8:
            a = (1.0 - best_alpha) * a + best_alpha * g
            a = np.maximum(a, 0.0)
            a = a / a.sum()

            if best_val < best_obj - 1e-13:
                best_obj = best_val
                best_a = a.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"    FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 50:
            a = project_simplex(best_a + rng.standard_normal(n) *
                                0.003 * (1 + no_improve * 0.05))
            no_improve = 0

    return best_a, best_obj


def peak_shaving(a0, time_limit, rng):
    """Directly try to reduce the maximum convolution peak by adjusting
    elements that contribute most to it."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(best_a)
        peak_idx = int(np.argmax(conv))
        M = conv[peak_idx]

        # Find which pairs (i, j) with i+j = peak_idx (or |i-j| = peak_idx)
        # contribute to this peak
        # c_k = sum_i a_i * a_{k-i}
        contribs = []
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contribs.append((i, j, best_a[i] * best_a[j]))

        if not contribs:
            break

        # Try reducing the largest contributors
        contribs.sort(key=lambda x: -x[2])
        improved = False

        for i, j, c in contribs[:5]:
            # Try reducing a[i] (or a[j] if i==j)
            for frac in [0.5, 0.3, 0.2, 0.1]:
                a_try = best_a.copy()
                red = a_try[i] * frac
                a_try[i] -= red
                # Redistribute to minimize new peak
                # Simple: add to elements far from the peak
                # Find elements that when added don't increase the peak
                candidates = []
                for k in range(n):
                    # Check if adding to position k would increase any conv
                    # Simply try it
                    a_try2 = a_try.copy()
                    a_try2[k] += red
                    a_try2 = np.maximum(a_try2, 0)
                    s = a_try2.sum()
                    if s < 1e-15:
                        continue
                    a_try2 = a_try2 / s
                    obj_try = 2.0 * n * fast_convolve(a_try2).max()
                    if obj_try < best_obj - 1e-13:
                        candidates.append((obj_try, a_try2))

                if candidates:
                    candidates.sort(key=lambda x: x[0])
                    best_try = candidates[0]
                    if best_try[0] < best_obj - 1e-13:
                        best_obj = best_try[0]
                        best_a = best_try[1].copy()
                        improved = True
                        break

            if improved:
                break

        if not improved:
            # Try random perturbation
            a_try = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            obj_try = 2.0 * n * fast_convolve(a_try).max()
            if obj_try < best_obj - 1e-13:
                best_obj = obj_try
                best_a = a_try.copy()
            else:
                # Small random nudge
                a_try = project_simplex(best_a + rng.standard_normal(n) * 0.001)
                obj_try = 2.0 * n * fast_convolve(a_try).max()
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()

    return best_a, best_obj


def local_search(a0, time_limit, rng):
    """Local search with adaptive perturbation."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    scale = 0.002
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        f_pert = project_simplex(best_a + rng.standard_normal(n) * scale)
        obj = 2.0 * n * fast_convolve(f_pert).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = f_pert.copy()
            no_improve = 0
        else:
            no_improve += 1
            scale *= 0.995
            if no_improve > 300:
                scale = 0.004
                no_improve = 0

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)

    # Beta-like distributions
    for alpha_b in [1.5, 2.0, 3.0]:
        x = np.arange(1, n + 1, dtype=np.float64)
        b = x ** (alpha_b - 1) * (n - x + 1) ** (alpha_b - 1)
        inits.append(b / b.sum())

    # Cosine window
    x = np.arange(n, dtype=np.float64)
    c1 = 0.5 * (1 + np.cos(2 * np.pi * x / n)) + 0.01
    inits.append(c1 / c1.sum())

    # Triangular
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits.append(tri / tri.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-stage optimization:
    1. Annealed L-BFGS on smoothed log-sum-exp (primary, FFT gradients)
    2. Frank-Wolfe with max-sum LP oracle (larger jumps)
    3. Mirror descent (multiplicative weights)
    4. Peak shaving (targeted peak reduction)
    5. Local search polishing
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, "
                  f"score={evaluate_sequence(parent.tolist()):.8f}")
    except Exception:
        pass

    # === Phase 1: Initialization ===
    print("=== Phase 1: Initialization ===")

    candidates = []
    if parent is not None:
        candidates.append(("parent", parent.copy()))
        n_p = len(parent)
    else:
        n_p = 384
        candidates.append(("uniform", np.ones(n_p) / n_p))

    other_inits = generate_inits(n_p, parent, rng)
    for i, ini in enumerate(other_inits):
        candidates.append((f"init_{i}", ini))

    for label, cand in candidates:
        if time.monotonic() > deadline - 300:
            break
        try_seq(cand, f"init_{label}")

    if best_seq is None:
        best_seq = np.ones(384) / 384
        try_seq(best_seq, "fallback")

    print(f"  Phase 1 best: {best_score:.8f} (n={len(best_seq)})")
    n_best = len(best_seq)

    # === Phase 2: Annealed L-BFGS (primary) ===
    print("\n=== Phase 2: Annealed L-BFGS ===")
    remaining = deadline - time.monotonic() - 150
    t_alloc_p2 = min(remaining * 0.45, 600)

    if t_alloc_p2 > 30:
        print(f"  Running annealed L-BFGS, budget={t_alloc_p2:.0f}s")
        a_opt = annealed_lbgfgs(best_seq, time_limit=t_alloc_p2,
                                betas=[5, 10, 20, 50, 100, 200, 500, 1000, 2000],
                                maxiter_per_beta=80)
        try_seq(a_opt, "annealed_lbgfgs")

        # Second run from best
        rem = deadline - time.monotonic() - 120
        if rem > 20 and best_score < 1.51:
            print(f"  Second L-BFGS pass, budget={min(rem*0.4, 100):.0f}s")
            a_opt = annealed_lbgfgs(best_seq, time_limit=min(rem * 0.4, 100),
                                    betas=[100, 200, 500, 1000, 2000, 4000],
                                    maxiter_per_beta=60)
            try_seq(a_opt, "lbgfgs2")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Frank-Wolfe ===
    print("\n=== Phase 3: Frank-Wolfe (max-sum LP) ===")
    remaining = deadline - time.monotonic() - 100
    t_alloc_p3 = min(remaining * 0.3, 300)

    if t_alloc_p3 > 30:
        print(f"  Running FW, budget={t_alloc_p3:.0f}s")
        a_opt, _ = fw_maxsum(best_seq, time_limit=t_alloc_p3, verbose=True)
        try_seq(a_opt, "fw_maxsum")

        # Perturbed restart
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            noise = rng.standard_normal(n_best) * 0.015
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = fw_maxsum(f_pert, time_limit=min(rem * 0.4, 120))
            try_seq(a_opt, "fw_pert")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: Mirror Descent ===
    print("\n=== Phase 4: Mirror Descent ===")
    remaining = deadline - time.monotonic() - 60
    t_alloc_p4 = min(remaining * 0.2, 150)

    if t_alloc_p4 > 15:
        print(f"  Running mirror descent, budget={t_alloc_p4:.0f}s")
        a_opt, _ = mirrored_gradient_descent(best_seq, time_limit=t_alloc_p4,
                                             lr_init=0.04, top_k=max(10, n_best // 4))
        try_seq(a_opt, "mirror")

    print(f"  Phase 4 best: {best_score:.8f}")

    # === Phase 5: Peak Shaving ===
    print("\n=== Phase 5: Peak Shaving ===")
    remaining = deadline - time.monotonic() - 40
    t_alloc_p5 = min(remaining * 0.3, 120)

    if t_alloc_p5 > 10:
        print(f"  Running peak shaving, budget={t_alloc_p5:.0f}s")
        a_opt, _ = peak_shaving(best_seq, time_limit=t_alloc_p5, rng=rng)
        try_seq(a_opt, "peak_shave")

    print(f"  Phase 5 best: {best_score:.8f}")

    # === Phase 6: Local Search + Final Polish ===
    print("\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic() - 15
    t_alloc_p6 = min(remaining * 0.6, 100)

    if t_alloc_p6 > 10:
        a_opt, _ = local_search(best_seq, time_limit=min(t_alloc_p6 * 0.5, 50),
                                rng=rng)
        try_seq(a_opt, "local")

        # Final short L-BFGS
        rem = deadline - time.monotonic() - 10
        if rem > 8:
            a_opt = annealed_lbgfgs(best_seq, time_limit=min(rem, 20),
                                    betas=[500, 1000, 2000],
                                    maxiter_per_beta=40)
            try_seq(a_opt, "final_lbgfgs")

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f}")

    return [float(x) for x in best_seq]
