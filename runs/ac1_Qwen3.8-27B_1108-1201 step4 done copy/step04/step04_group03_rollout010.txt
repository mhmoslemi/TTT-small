def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_for_fw(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def solve_sdp(n, time_limit=300):
    """Solve SDP relaxation to get warm start."""
    try:
        X = cp.Variable((n, n), symmetric=True)
        M = cp.Variable()
        
        constraints = [X >> 0]
        
        # Convolution constraints
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            conv_term = sum(X[j, k - j] for j in idx)
            constraints.append(conv_term <= M)
        
        # Normalization: sum of all entries = 1
        constraints.append(cp.sum(X) == 1)
        
        prob = cp.Problem(cp.Minimize(M), constraints)
        
        t0 = time.monotonic()
        prob.solve(solver=cp.PDLP, verbose=False)
        
        if time.monotonic() - t0 > time_limit:
            return None, None
            
        if prob.status in ['optimal', 'optimal_inaccurate']:
            X_val = X.value
            if X_val is not None:
                # Extract top eigenvector
                X_val = (X_val + X_val.T) / 2  # Symmetrize
                eigenvalues, eigenvectors = np.linalg.eigh(X_val)
                idx = np.argmax(eigenvalues)
                a = np.abs(eigenvectors[:, idx])
                s = a.sum()
                if s > 1e-10:
                    a = a / s
                    return a, M.value
        return None, None
    except Exception as e:
        print(f"SDP failed: {e}")

def mw_step(a, eta, T):
    """One step of multiplicative weight with logsumexp smoothing."""
    n = len(a)
    conv = fast_convolve(a)
    
    # Softmax weights
    Tconv = T * conv
    Tconv -= Tconv.max()  # numerical stability
    p = np.exp(Tconv)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros_like(conv)
        p[np.argmax(conv)] = 1.0
    else:
        p /= p_sum
    
    # Cross-correlation: sg[i] = 2 * sum_k p[k] * a[k-i] (valid k-i)
    # This is the cross-correlation of p and a
    # Using FFT: IFFT(FFT(p) * conj(FFT(a))) gives circular cross-correlation
    # We need linear cross-correlation, so use zero-padding
    fft_size = 1 << (len(p) + n - 1 - 1).bit_length()
    P = np.fft.fft(p, fft_size)
    A = np.fft.fft(a, fft_size)
    corr = np.fft.ifft(P * np.conj(A)).real
    # corr[i] = sum_k p[k] * a[(k-i) mod fft_size]
    # For linear cross-correlation, we want the first n values
    # Actually, the cross-correlation (w ⊛ a)[i] = sum_k w[k] * a[k-i]

**Strategy for hyperparameters:**

The objective we're minimizing is:
F(a) = max_k conv[k] (with sum(a) = 1)

The logsumexp approximation is:
L_T(a) = (1/T) * log(sum_k exp(T * conv[k]))

As T → ∞, L_T → max_k conv[k]. For finite T, L_T is a smooth upper bound.

The gradient of L_T is:
∇L_T[i] = sum_k p_k * ∂conv[k]/∂a[i] = 2 * (p ⊛ a)[i]

where p_k = exp(T*conv[k]) / sum_j exp(T*conv[j]).

For the multiplicative weight update:
a_new[i] ∝ a[i] * exp(-eta * ∇L_T[i])

The choice of eta is crucial. If eta is too large, we overshoot. If too small, we converge slowly.

A good adaptive strategy:
- Start with a moderate eta (e.g., 0.01)
- Decrease eta geometrically over time
- Periodically increase T (to focus more on the max peak)
- Monitor the objective and restart if it increases
