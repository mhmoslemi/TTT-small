Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid optimization approach:
- Nesterov-accelerated subgradient descent with adaptive multi-peak weighting
- LP-based Frank-Wolfe for direction finding
- Backtracking line search

- Coordinate descent polishing

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Compute weighted subgradient of max(conv(f,f)) using top-k peaks."""

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid optimization approach:
- Accelerated subgradient descent with adaptive multi-peak softmax weighting
- LP-based Frank-Wolfe direction finding interleaved with SG steps
- Golden section line search for better step selection
- Multi-length exploration near the parent sequence length
- Coordinate descent and fine-grained subgradient polishing
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, top_k=15, temp=0.001):
    """Compute weighted subgradient of max(conv(f,f)) using top-k peaks with softmax."""
    n = len(f)
    conv = fast_convolve(f)

    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]

    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-13:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]

    return sg, conv


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0. Returns normalized g or None."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 30})
        if res.success and res.x is not None:
            g = res.x
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def golden_section_line_search(f, g, conv_current, max_iter=25):
    """
    Find best alpha in [0,1] minimizing max(conv((1-alpha)*f + alpha*g, ...)).
    Uses golden section search on [0,1].
    """
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c_pt = hi - phi * (hi - lo)
    d_pt = lo + phi * (hi - lo)

    def obj_at(alpha):
        ft = (1.0 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0.0)
        st = ft.sum()
        if st < 1e-15:
            return float('inf')
        ft = ft / st
        return fast_convolve(ft).max()

    fc = obj_at(c_pt)
    fd = obj_at(d_pt)

    best_alpha = 0.0
    best_val = conv_current

    for _ in range(max_iter):
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - phi * (hi - lo)
            fc = obj_at(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + phi * (hi - lo)
            fd = obj_at(d_pt)

        if abs(hi - lo) < 1e-5:
            break

    mid = (lo + hi) / 2.0
    fm = obj_at(mid)
    if fm < best_val:
        best_alpha = mid
        best_val = fm
    if fc < best_val:
        best_alpha = c_pt
        best_val = fc
    if fd < best_val:
        best_alpha = d_pt
        best_val = fd

    return best_alpha, best_val

def coordinate_descent_polish(f0, time_limit):

    f = np.maximum(f0.astype(np.float64), 0.0)

            if time.monotonic() - t0 > time_limit - 1:

            candidates.add((j + 1) % n)
            candidates.add((j - 1) % n)

                    if val < best_obj - 1e-14:

                        improved = True
                        break
                if improved:

            if improved:
                # Also try reverse direction
                improved = False

        if not improved:
            # Try one more full pass before giving up
            any_improve = False
            for j in range(n):
                if time.monotonic() - t0 > time_limit - 1:
                    break
                for source in [(j + 1) % n, (j + n // 2) % n]:
                    if source == j or f[source] < 1e-12:

                        f_new[source] -= delta
                        if f_new[source] < 0:
                            f_new[j] += f_new[source]

                        f_new = np.maximum(f_new, 0.0)
                        sn = f_new.sum()
                        if sn < 1e-15:
                            continue
                        f_new = f_new / sn
                        val = fast_convolve(f_new).max()
                        if val < best_obj - 1e-14:
                            f = f_new
                            best_obj = val
                            best_f = f.copy()
                            any_improve = True
                            break
                    if any_improve:
                        break
                if any_improve:
                    break
            if not any_improve:

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Random Dirichlet (concentrated)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    # Edge-concentrated
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine taper
    cos_t = np.cos(np.pi * x / (2.0 * (n - 1)))
    cos_t = np.maximum(cos_t, 0.01)
    inits.append(cos_t / cos_t.sum())

    # Two-peak
    peak = (np.exp(-((x - n / 4.0) ** 2) / (2.0 * (n / 8.0) ** 2)) +
            np.exp(-((x - 3.0 * n / 4.0) ** 2) / (2.0 * (n / 8.0) ** 2)))
    inits.append(peak / peak.sum())

    # Sine window
    sine = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine / sine.sum())

    # Power law
    pl = (x + 1.0) ** (-0.5)
    inits.append(pl / pl.sum())

    # Flat with small random perturbation
    flat_pert = np.ones(n) + rng.standard_normal(n) * 0.1
    flat_pert = np.maximum(flat_pert, 0.01)
    inits.append(flat_pert / flat_pert.sum())

    # Parent interpolation
    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            # Perturbed parent
            px_pert = px + rng.standard_normal(n) * 0.01 * px.max()
            px_pert = np.maximum(px_pert, 0)
            ps_pert = px_pert.sum()
            if ps_pert > 1e-10:
                inits.append(px_pert / ps_pert)

    return inits


def try_sequence(a, label, state):
    """Evaluate a sequence and update state if it's better."""
    a = np.maximum(a, 0.0)
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < state['best_score']:
        state['best_score'] = score
        state['best_seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase hybrid search using accelerated subgradient
    descent interleaved with LP-based Frank-Wolfe, with multi-length exploration
    and coordinate descent polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    state = {'best_seq': None, 'best_score': float('inf')}

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}")
            s = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(s):
                print(f"  Parent score: {s:.8f}")
                if s < state['best_score']:
                    state['best_score'] = s
                    state['best_seq'] = p.copy()
    except Exception:
        pass

    if parent is None:
        # Default parent
        parent = np.ones(384) / 384
        print("Using default uniform parent n=384")
        s = float(evaluate_sequence(parent.tolist()))
        state['best_score'] = s
        state['best_seq'] = parent.copy()
        print(f"  Default score: {s:.8f}")

    n_parent = len(parent)

    # === Phase 1: Multi-length Quick Exploration (20% budget) ===
    print("\n=== Phase 1: Multi-length Quick Exploration ===")
    phase1_end = time.monotonic() + (float(budget_s) * 0.18)

    # Candidate lengths: focus near parent
    candidate_lengths = set()
    for delta in range(-10, 11, 2):
        n_try = n_parent + delta
        if 200 <= n_try <= 500:
            candidate_lengths.add(n_try)
    # Add some fixed lengths
    for n_fixed in [320, 360, 384, 400, 420, 448]:
        if 200 <= n_fixed <= 500:
            candidate_lengths.add(n_fixed)
    # Sort by proximity to parent
    candidate_lengths = sorted(candidate_lengths, key=lambda x: abs(x - n_parent))
    # Limit to top 8 lengths
    candidate_lengths = candidate_lengths[:8]

    print(f"  Exploring lengths: {candidate_lengths}")

    # Quick runs on each length
    n_lengths = len(candidate_lengths)
    t_per_length = max(15, (phase1_end - time.monotonic() - 30) / n_lengths)

    for n in candidate_lengths:
        if time.monotonic() > phase1_end - 20:
            break
        t_alloc = min(t_per_length, phase1_end - time.monotonic() - 15)
        if t_alloc < 10:
            break

        inits = generate_inits(n, parent, rng)
        # Try 2 initializations
        n_try = min(2, len(inits))
        t_per_init = t_alloc / n_try

        for i in range(n_try):
            if time.monotonic() > phase1_end - 10:
                break
            t_i = min(t_per_init, phase1_end - time.monotonic() - 10)
            if t_i < 8:
                break
            f_init = inits[i]
            a_opt, obj = hybrid_optimize(f_init, time_limit=t_i)
            try_sequence(a_opt, f"ph1_n{n}_i{i}", state)

    print(f"  Phase 1 best: {state['best_score']:.8f} (n={len(state['best_seq']) if state['best_seq'] is not None else 0})")

    # === Phase 2: Deep Optimization (50% budget) ===
    print(f"\n=== Phase 2: Deep Optimization ===")
    phase2_end = time.monotonic() + (float(budget_s) * 0.50)

    n_best = len(state['best_seq'])
    print(f"  Primary optimization: n={n_best}, budget={(phase2_end - time.monotonic()):.0f}s")

    # Main deep run
    t_main = min(300, (phase2_end - time.monotonic()) * 0.45)
    if t_main > 30:
        print(f"  Main hybrid run: t={t_main:.0f}s")
        a_opt, obj = hybrid_optimize(state['best_seq'], time_limit=t_main, verbose=True)
        try_sequence(a_opt, "deep_main", state)

    # Perturbation runs from current best
    perturb_scales = [0.001, 0.003, 0.008, 0.02, 0.05]
    for ps in perturb_scales:
        rem = phase2_end - time.monotonic() - 20
        if rem < 15:
            break
        t_alloc = min(rem / (len(perturb_scales) - perturb_scales.index(ps) - 1 + 1), 100)
        if t_alloc < 10:
            break
        noise = rng.standard_normal(n_best) * ps
        f_init = project_simplex(state['best_seq'] + noise)
        a_opt, obj = hybrid_optimize(f_init, time_limit=t_alloc)
        try_sequence(a_opt, f"pert_{ps}", state)

    # Mirror blend
    rem = phase2_end - time.monotonic() - 15
    if rem > 20:
        f_mirror = project_simplex(0.8 * state['best_seq'] + 0.2 * state['best_seq'][::-1])
        a_opt, obj = hybrid_optimize(f_mirror, time_limit=min(rem, 80))
        try_sequence(a_opt, "mirror", state)

    # Anti-symmetric perturbation
    rem = phase2_end - time.monotonic() - 10
    if rem > 15:
        anti_sym = state['best_seq'] - 0.3 * (state['best_seq'] - state['best_seq'][::-1])
        f_anti = project_simplex(anti_sym)
        a_opt, obj = hybrid_optimize(f_anti, time_limit=min(rem, 60))
        try_sequence(a_opt, "antisym", state)

    print(f"\n  Phase 2 best: {state['best_score']:.8f} (n={len(state['best_seq'])})")

    # === Phase 3: Polishing (remaining time minus safety margin) ===
    print(f"\n=== Phase 3: Polishing ===")
    phase3_end = deadline - 15

    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])
        print(f"  Current best: {state['best_score']:.8f}")

        # Fine SG polish (long)
        rem = phase3_end - time.monotonic() - 50
        if rem > 20:
            t_sg = min(rem * 0.4, 120)
            print(f"  Fine SG polish: t={t_sg:.0f}s")
            a_opt, obj = fine_sg_polish(state['best_seq'], time_limit=t_sg,
                                        lr=0.0008, momentum=0.97,
                                        top_k=25, temp=0.00005)
            try_sequence(a_opt, "sg_fine_long", state)

        # Coordinate descent
        rem = phase3_end - time.monotonic() - 30
        if rem > 15:
            t_cd = min(rem * 0.35, 80)
            print(f"  Coordinate descent: t={t_cd:.0f}s")
            a_opt, obj = coordinate_descent_polish(state['best_seq'], time_limit=t_cd)
            try_sequence(a_opt, "cd", state)

        # Very fine SG
        rem = phase3_end - time.monotonic() - 20
        if rem > 10:
            t_sg2 = min(rem * 0.35, 60)
            print(f"  Ultra-fine SG: t={t_sg2:.0f}s")
            a_opt, obj = fine_sg_polish(state['best_seq'], time_limit=t_sg2,
                                        lr=0.0003, momentum=0.98,
                                        top_k=30, temp=0.00002)
            try_sequence(a_opt, "sg_ultra_fine", state)

        # Final hybrid squeeze
        rem = phase3_end - time.monotonic() - 5
        if rem > 10:
            t_final = min(rem, 40)
            print(f"  Final hybrid squeeze: t={t_final:.0f}s")
            a_opt, obj = hybrid_optimize(state['best_seq'], time_limit=t_final)
            try_sequence(a_opt, "final_squeeze", state)

        # One more CD pass
        rem = phase3_end - time.monotonic()
        if rem > 8:
            a_opt, obj = coordinate_descent_polish(state['best_seq'], time_limit=min(rem, 20))
            try_sequence(a_opt, "cd_final", state)

    # Fallback
    if state['best_seq'] is None:
        state['best_seq'] = np.ones(256) / 256
        state['best_score'] = float(evaluate_sequence(state['best_seq'].tolist()))
        print(f"  Fallback uniform: {state['best_score']:.6f}")

    # Final verification
    fs = float(evaluate_sequence(state['best_seq'].tolist()))
    if np.isfinite(fs):
        state['best_score'] = fs

    print(f"\nFINAL SCORE: {state['best_score']:.8f} (n={len(state['best_seq'])})")
    return [float(x) for x in state['best_seq']]
