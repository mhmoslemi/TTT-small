def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def project_simplex(v):

    n = len(v)

def coord_sweep(f0, time_limit, n_iter=200, verbose=False):
    """Coordinate descent sweep: move mass between pairs to reduce peak correlation."""
    f = np.maximum(f0.astype(np.float64), 0.0)

        f = np.ones(len(f)) / len(f)

        f = f / s
    n = len(f)

    best_obj = conv.max()
    best_f = f.copy()

    rng = np.random.default_rng(42)

    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        # Pick peak position
        k_peak = np.argmax(conv)

        # Find coordinates contributing most to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        contrib = f[js:je + 1]

        # Pick a high contributor to reduce
        local_idx = js + np.argmax(contrib)
        if f[local_idx] < 1e-10:
            continue

        # Try moving mass to a random low coordinate
        low_candidates = np.where(f < np.mean(f) * 0.5)[0]
        if len(low_candidates) == 0:
            continue
        target = rng.choice(low_candidates)
        if target == local_idx:
            continue

        # Binary search on transfer amount
        best_alpha = 0.0
        best_trial = conv.max()
        f_trial = f.copy()

        for _ in range(20):
            alpha = (best_alpha + 1.0) / 2.0
            amount = f[local_idx] * alpha
            if amount < 1e-15:
                break
            f_trial = f.copy()
            f_trial[local_idx] -= amount
            f_trial[target] += amount
            f_trial = np.maximum(f_trial, 0.0)
            f_trial = f_trial / f_trial.sum()
            trial_conv = fast_convolve(f_trial)
            trial_obj = trial_conv.max()
            if trial_obj < best_trial:
                best_trial = trial_obj
                best_alpha = alpha
            else:
                # alpha too large, reduce
                pass
            # Halve the search range
            if trial_obj < conv.max():
                # Can try more
                break

        if best_alpha > 0.01:
            amount = f[local_idx] * best_alpha
            f[local_idx] -= amount
            f[target] += amount
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            if conv.max() < best_obj - 1e-14:
                best_obj = conv.max()
                best_f = f.copy()

        # Occasionally do a full sweep reducing the top contributor
        if it % 10 == 0:
            k_peak = np.argmax(conv)
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            if je >= js:
                top_local = js + np.argmax(f[js:je + 1])
                if f[top_local] > 1e-8:
                    # Spread its mass to neighbors
                    spread = f[top_local] * 0.1
                    f[top_local] -= spread
                    left = max(0, top_local - 1)

                    f[right] += spread * 0.5
                    f = np.maximum(f, 0.0)
                    f = f / f.sum()
                    conv = fast_convolve(f)

def upsample_seq(a, factor=2):
    """Upsample a sequence by linear interpolation."""
    n = len(a)
    new_n = n * factor
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    new_a = np.interp(x_new, x_old, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s > 1e-15:
        new_a /= s
    return new_a

    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine with various frequencies
    for freq in [2, 3, 5, 8, 13]:
        c = 1.0 + 0.4 * np.cos(2 * np.pi * freq * x / n)
        c = np.maximum(c, 0.01)
        inits.append(c / c.sum())

    # Sine modulation
    for freq in [1, 2, 4]:
        s = 1.0 + 0.3 * np.sin(2 * np.pi * freq * x / n)
        s = np.maximum(s, 0.01)
        inits.append(s / s.sum())

    # Power law
    pw = (x + 1.0) ** 0.3
    inits.append(pw / pw.sum())

    # Inverse power law
    ipw = 1.0 / (x + 1.0) ** 0.3
    inits.append(ipw / ipw.sum())

    # Two-peaked
    peak1 = n // 3
    peak2 = 2 * n // 3
    p = np.exp(-((x - peak1) ** 2) / (n / 10.0)) + np.exp(-((x - peak2) ** 2) / (n / 10.0)) + 0.01
    inits.append(p / p.sum())

    # Random smooth (low-pass filtered noise)
    raw = rng.standard_normal(n)
    # Simple moving average for smoothing
    window = max(3, n // 50)
    padded = np.concatenate([raw[:window][::-1], raw, raw[-window:][::-1]])
    smooth = np.convolve(padded, np.ones(window) / window, mode='valid')[:n]
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())

    # Periodic pattern
    period = max(2, n // 16)
    pattern = np.tile([1.0, 0.3], (n // period + 1))[:n]
    inits.append(pattern / pattern.sum())

    # Parent-based
    if parent is not None and len(parent) > 1:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0.01)
        px = px / px.sum()
        inits.append(px)
        # Modulated parent
        mod = px * (1.0 + 0.2 * np.cos(2 * np.pi * 3 * x / n))
        mod = np.maximum(mod, 0.01)
        inits.append(mod / mod.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function: multi-phase optimization combining mirror descent,
    FISTA, coordinate descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        if a.sum() < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            old = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}) (was {old:.8f})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
    except Exception:
        pass

    # === Phase 1: Optimize parent with mirror descent ===
    print("=== Phase 1: Parent Optimization (Mirror Descent) ===")
    if parent is not None:
        n_p = len(parent)
        t1 = min(200, deadline - time.monotonic() - 400)
        if t1 > 30:
            print(f"  Mirror descent on parent (n={n_p}), t={t1:.0f}s")
            a_opt, obj = mirror_descent(parent, time_limit=t1, lr=0.5,
                                        temp_init=0.003, temp_final=0.0001, top_k=n_p // 4)
            try_seq(a_opt, "parent_md")

            # FISTA follow-up
            rem = deadline - time.monotonic() - 300
            if rem > 20:
                a_opt2, _ = fista_optimize(a_opt, time_limit=min(rem, 80),
                                           lr=0.03, temp_init=0.002, temp_final=0.0001,
                                           top_k=n_p // 4)
                try_seq(a_opt2, "parent_fista")

            # Coordinate sweep
            rem = deadline - time.monotonic() - 200
            if rem > 10:
                a_opt3, _ = coord_sweep(a_opt, time_limit=min(rem, 30))
                try_seq(a_opt3, "parent_coord")

    # === Phase 2: Multi-length exploration ===
    print("\n=== Phase 2: Multi-length Exploration ===")

    lengths = [384, 512, 768]
    if parent is not None:
        if len(parent) not in lengths:
            lengths.insert(0, len(parent))
        else:
            lengths.remove(len(parent))
            lengths.insert(0, len(parent))

    # Also try larger lengths
    lengths.extend([1024, 1280])
    lengths = list(set(lengths))
    lengths.sort()

    for n in lengths:
        if time.monotonic() > deadline - 150:
            break
        if parent is not None and n == len(parent) and best_score < 1.51:
            continue  # Already optimized parent

        inits = generate_inits(n, parent, rng)
        # Select subset of inits to keep time manageable
        n_inits = min(len(inits), 6)
        if len(inits) > n_inits:
            # Pick diverse ones
            chosen = [inits[0], inits[1], inits[len(inits)//3], inits[2*len(inits)//3],
                      inits[-2], inits[-1]][:n_inits]
        else:
            chosen = inits

        rem = deadline - time.monotonic() - 120
        t_per = max(15, rem / (n_inits * max(1, len(lengths) - lengths.index(n))))
        t_per = min(t_per, 80)

        for i, f_init in enumerate(chosen):
            if time.monotonic() > deadline - 100:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 60)
            if t_alloc < 10:
                break

            # Alternate between mirror descent and FISTA
            if i % 2 == 0:
                a_opt, obj = mirror_descent(f_init, time_limit=t_alloc, lr=0.4,
                                            temp_init=0.004, temp_final=0.0001,
                                            top_k=max(64, n // 6))
            else:
                a_opt, obj = fista_optimize(f_init, time_limit=t_alloc, lr=0.04,
                                            temp_init=0.003, temp_final=0.0001,
                                            top_k=max(64, n // 6))
            try_seq(a_opt, f"n{n}_i{i}")

    print(f"  Phase 2 best: {best_score:.8f} (n={len(best_seq) if best_seq is not None else 0})")

    # === Phase 3: Deep optimization on best ===
    print(f"\n=== Phase 3: Deep Optimization ===")

    if best_seq is not None:
        n_best = len(best_seq)

        # Mirror descent with different params
        runs = [
            {"method": "md", "lr": 0.3, "temp_i": 0.005, "temp_f": 0.00005, "tk": n_best // 3},
            {"method": "fista", "lr": 0.02, "temp_i": 0.003, "temp_f": 0.00005, "tk": n_best // 3},
            {"method": "md", "lr": 0.8, "temp_i": 0.001, "temp_f": 0.00003, "tk": n_best // 2},
            {"method": "fista", "lr": 0.08, "temp_i": 0.005, "temp_f": 0.0001, "tk": n_best // 4},
        ]

        for p, cfg in enumerate(runs):
            rem = deadline - time.monotonic() - 60
            if rem < 15:
                break
            t_alloc = min(rem // (len(runs) - p), 100)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)
            elif p == 2:
                f_init = project_simplex(0.8 * best_seq + 0.2 * (best_seq[::-1]))
            else:
                f_init = project_simplex(0.9 * best_seq + 0.1 * (np.ones(n_best) / n_best))

            if cfg["method"] == "md":
                a_opt, obj = mirror_descent(f_init, time_limit=t_alloc, lr=cfg["lr"],
                                            temp_init=cfg["temp_i"], temp_final=cfg["temp_f"],
                                            top_k=cfg["tk"])
            else:
                a_opt, obj = fista_optimize(f_init, time_limit=t_alloc, lr=cfg["lr"],
                                            temp_init=cfg["temp_i"], temp_final=cfg["temp_f"],
                                            top_k=cfg["tk"])
            try_seq(a_opt, f"deep_{cfg['method']}_{p}")

        # Coordinate descent on best
        rem = deadline - time.monotonic() - 40
        if rem > 10:
            a_opt, _ = coord_sweep(best_seq, time_limit=min(rem, 40))
            try_seq(a_opt, "deep_coord")

    # === Phase 4: Try upscaling to larger n ===
    print(f"\n=== Phase 4: Upscaling ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for factor in [2]:
            n_new = n_best * factor
            if n_new > 2048:
                continue
            rem = deadline - time.monotonic() - 60
            if rem < 20:
                break
            f_up = upsample_seq(best_seq, factor)
            t_alloc = min(rem, 60)
            a_opt, obj = mirror_descent(f_up, time_limit=t_alloc, lr=0.3,
                                        temp_init=0.003, temp_final=0.0001,
                                        top_k=n_new // 5)
            try_seq(a_opt, f"upscale_{factor}x")

    # === Phase 5: Final polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    print(f"  Current best: {best_score:.8f}")

    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine mirror descent
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            a_opt, _ = mirror_descent(best_seq, time_limit=min(rem, 40), lr=0.2,
                                      temp_init=0.0005, temp_final=0.00002,
                                      top_k=n_best // 2)
            try_seq(a_opt, "polish_md")

        # FISTA with very small LR
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_opt, _ = fista_optimize(best_seq, time_limit=min(rem, 30), lr=0.01,
                                      temp_init=0.0005, temp_final=0.00002,
                                      top_k=n_best // 2)
            try_seq(a_opt, "polish_fista")

        # Micro perturbations
        rem = deadline - time.monotonic() - 10
        if rem > 8:
            for _ in range(3):
                rem = deadline - time.monotonic() - 5
                if rem < 5:
                    break
                f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.001)
                a_opt, _ = mirror_descent(f_pert, time_limit=min(rem, 10), lr=0.1,
                                          temp_init=0.0003, temp_final=0.00001,
                                          top_k=n_best // 2)
                try_seq(a_opt, "micro")

    # Fallback
    if best_seq is None:
        n_fb = 384
        best_seq = np.ones(n_fb) / n_fb
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback used: score={best_score:.8f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
