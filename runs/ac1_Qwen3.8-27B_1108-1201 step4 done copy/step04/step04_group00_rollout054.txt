"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def objective_from_conv(conv):
    """Compute objective given the convolution and n."""
    n = (len(conv) - 1) // 2
    return 2.0 * n * conv.max()


def solve_lp_direction_active(f, M, n, active_set=None, tol=1e-8):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for active constraints, g >= 0.
    Uses active set management for efficiency.
    Returns (g, active_set_updated, success).
    """
    if active_set is None:
        active_set = list(range(2 * n - 1))
    
    m = len(active_set)
    if m == 0:
        return np.ones(n) / n, active_set, True
    
    c = -np.ones(n)
    A_ub = np.zeros((m, n))
    for row, k in enumerate(active_set):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for col in range(js, je + 1):
            A_ub[row, col] = f[col]
    b_ub = np.full(m, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # Fall back to all constraints
        active_set = list(range(2 * n - 1))
        m = len(active_set)
        c = -np.ones(n)
        A_ub = np.zeros((m, n))
        for row, k in enumerate(active_set):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for col in range(js, je + 1):
                A_ub[row, col] = f[col]
        b_ub = np.full(m, M)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            return None, active_set, False
    
    g = res.x
    g_sum = g.sum()
    if g_sum < 1e-15:
        return np.ones(n) / n, active_set, True
    
    # Check feasibility for all constraints
    conv_test = fast_convolve(g / g_sum)
    violated = np.where(conv_test > M + tol)[0]
    
    if len(violated) > 0 and len(active_set) < 2 * n - 1:
        # Add violated constraints to active set
        new_set = set(active_set)
        for v in violated[:20]:  # Add at most 20 new
            new_set.add(int(v))
        # Keep at most 200 constraints for efficiency
        if len(new_set) > 200:
            # Keep the most violated
            conv_full = fast_convolve(f)
            sorted_k = np.argsort(conv_full)[::-1]
            new_set = set(sorted_k[:200].tolist())
        active_set = sorted(new_set)
    
    return g, active_set, True

    M = conv.max()
    best_obj = M
    best_f = f.copy()

    active_set = np.argsort(conv)[::-1][:min(max_constraints, 2 * n - 1)].tolist()
    
    t0 = time.monotonic()

    it = 0
    m_relax = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        g_dir, active_set, success = solve_lp_direction_active(
            f, M * m_relax, n, active_set)
        
        if g_dir is None or not success:
            m_relax = min(m_relax * 1.1, 3.0)
            if m_relax > 3.0:
                m_relax = 1.0
            continue
        
        g_sum = g_dir.sum()
        if g_sum < 1e-15:
            break
        g_dir = g_dir / g_sum
        
        if np.max(np.abs(g_dir - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:

                    hi = mid

def nesterov_sg(f0, time_limit, lr=0.002, momentum=0.95, top_k=30, temp=1e-5):
    """Nesterov-accelerated subgradient polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    y = f.copy()  # Lookahead point
    m_vec = np.zeros(n)  # Momentum
    conv = fast_convolve(y)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    t_iter = 1
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    
    while time.monotonic() - t0 < time_limit:
        # Compute subgradient at y
        conv_y = fast_convolve(y)
        k_actual = min(top_k, len(conv_y))
        peaks = np.argpartition(conv_y, -k_actual)[-k_actual:]
        
        pv = conv_y[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += w[i] * (2.0 * y[k - idx] + 2.0 * y[idx - (k - n + 1) * 0])
            # More carefully: d/da_j [(a*a)[k]] = a[k-j] + a[j-k] (with 0 outside range)
        
        # Recompute subgradient more carefully
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            for j in range(n):
                # Contribution to (a*a)[k] from a[j]: a[k-j] (if valid) 
                ki = k - j
                if 0 <= ki < n:
                    sg[j] += w[i] * y[ki]
                # And a[j-k] if valid
                kj = j - k
                if 0 <= kj < n:
                    sg[j] += w[i] * y[kj]
        
        # Nesterov update
        m_vec = momentum * m_vec - cur_lr * sg
        y = f + m_vec
        y = project_simplex(y)
        
        # Evaluate at y
        conv = fast_convolve(y)
        co = conv.max()
        
        if co < best_obj - 1e-15:
            best_obj = co
            best_f = y.copy()
            f = y.copy()
            no_improve = 0
        else:
            f = project_simplex(f)
            no_improve += 1
        
        t_iter += 1
        cur_lr *= 0.9998
        cur_temp *= 1.002
        
        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            y = f.copy()
            m_vec = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0003)
            no_improve = 0
    
    return best_f, best_obj


def coordinate_descent(f0, time_limit, n_multipliers=10):
    """Coordinate descent with adaptive multipliers."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    multipliers = np.linspace(0.3, 2.0, n_multipliers)
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            cur_peak = fast_convolve(f).max()
            best_j_obj = cur_peak
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss
                
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f /= f.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved and passes > 2:
            break
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_init(n, pattern="cosine"):
    """Generate structured initial sequence."""
    x = np.arange(n) / n
    if pattern == "cosine":
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif pattern == "triangle":
        a = 1.0 - np.abs(2 * x - 1)
    elif pattern == "parabola":
        a = 1.0 - (2 * x - 1) ** 2
    elif pattern == "skew":
        a = x ** 0.7 + 0.1
    elif pattern == "bell":
        a = np.exp(-5 * (2 * x - 1) ** 2)
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining LP-FW with active sets,
    Nesterov subgradient, coordinate descent, and multi-length search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if len(a) == 0 or a.sum() < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            try_seq(parent, "parent")
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(384) / 384
        try_seq(parent, "uniform_384")
    
    n_parent = len(parent)
    
    # === Phase 1: LP-FW on parent length ===
    print(f"\n=== Phase 1: LP-FW on parent (n={n_parent}) ===")
    rem = deadline - time.monotonic() - 600
    t1 = min(250, rem)
    if t1 > 30:
        a_opt, _ = lp_fw_enhanced(parent, time_limit=t1, verbose=True)
        try_seq(a_opt, f"lpfw_n{n_parent}")
    
    # === Phase 2: Try structured inits at various lengths ===
    print(f"\n=== Phase 2: Structured inits ===")
    lengths_to_try = [512, 768, 1024]
    
    # Filter to reasonable lengths
    if parent is not None and n_parent in lengths_to_try:
        lengths_to_try.remove(n_parent)
    
    for target_n in lengths_to_try:
        rem = deadline - time.monotonic() - 400
        if rem < 30:
            break
        
        # Use best available as base for interpolation
        base = best_seq if best_seq is not None else parent
        base_n = len(base)
        
        if base_n != target_n:
            f_init = interpolate_sequence(base, target_n)
        else:
            f_init = base.copy()
        
        t_alloc = min(rem * 0.5, 200)
        print(f"  LP-FW n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_enhanced(f_init, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, f"lpfw_struct_n{target_n}")
    
    # === Phase 3: Multi-start LP-FW on best ===
    print(f"\n=== Phase 3: Multi-start deep optimization (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        
        starts = []
        # Start 1: current best
        starts.append(best_seq.copy())
        # Start 2: noisy perturbation
        noise = rng.standard_normal(n_best) * 0.005
        starts.append(project_simplex(best_seq + noise))
        # Start 3: reversed and mixed
        starts.append(project_simplex(0.7 * best_seq + 0.3 * best_seq[::-1]))
        # Start 4: cosine-shaped
        starts.append(structured_init(n_best, "cosine"))
        # Start 5: parabolic
        starts.append(structured_init(n_best, "parabola"))
        
        for si, f_init in enumerate(starts):
            rem = deadline - time.monotonic() - 200
            if rem < 20:
                break
            t_alloc = min(rem / (len(starts) - si), 150)
            print(f"  Multi-start {si}: t={t_alloc:.0f}s")
            a_opt, _ = lp_fw_enhanced(f_init, time_limit=t_alloc, verbose=(si == 0))
            try_seq(a_opt, f"multi_start_{si}")
    
    # === Phase 4: Coordinate descent polishing ===
    print(f"\n=== Phase 4: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_cd = min(rem * 0.6, 200)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd, n_multipliers=12)
            try_seq(a_opt, "cd")
            
            # Also try CD on a slightly perturbed version
            rem = deadline - time.monotonic() - 100
            if rem > 15:
                noise = rng.standard_normal(len(best_seq)) * 0.002
                f_pert = project_simplex(best_seq + noise)
                a_opt2, _ = coordinate_descent(f_pert, time_limit=min(rem, 60), n_multipliers=8)
                try_seq(a_opt2, "cd_pert")
    
    # === Phase 5: Nesterov subgradient polishing ===
    print(f"\n=== Phase 5: Nesterov SG (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        
        rem = deadline - time.monotonic() - 80
        if rem > 15:
            t_sg = min(rem * 0.5, 120)
            print(f"  Nesterov SG t={t_sg:.0f}s")
            a_opt, _ = nesterov_sg(best_seq, time_limit=t_sg,
                                  lr=0.0015, momentum=0.95,
                                  top_k=max(20, n_best // 6), temp=1e-5)
            try_seq(a_opt, "nesterov_sg")
        
        # Fine SG with smaller lr
        rem = deadline - time.monotonic() - 50
        if rem > 10:
            a_opt, _ = nesterov_sg(best_seq, time_limit=min(rem, 60),
                                  lr=0.0005, momentum=0.96,
                                  top_k=max(10, n_best // 10), temp=1e-6)
            try_seq(a_opt, "sg_fine")
    
    # === Phase 6: Final LP-FW on best ===
    print(f"\n=== Phase 6: Final LP-FW (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t_final = min(rem, 100)
            print(f"  Final LP-FW t={t_final:.0f}s")
            a_opt, _ = lp_fw_enhanced(best_seq, time_limit=t_final, verbose=True)
            try_seq(a_opt, "final_lpfw")
            
            # One more round of CD
            rem = deadline - time.monotonic() - 15
            if rem > 10:
                a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 40), n_multipliers=6)
                try_seq(a_opt, "cd_final")
    
    # === Phase 7: Try a few more lengths if time allows ===
    print(f"\n=== Phase 7: Additional lengths (best={best_score:.6f}) ===")
    if best_seq is not None:
        for extra_n in [1280, 1536]:
            rem = deadline - time.monotonic() - 20
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, extra_n)
            t_alloc = min(rem * 0.7, 120)
            print(f"  Extra LP-FW n={extra_n}, t={t_alloc:.0f}s")
            a_opt, _ = lp_fw_enhanced(f_interp, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, f"extra_n{extra_n}")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        # Should not happen, but just in case
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
