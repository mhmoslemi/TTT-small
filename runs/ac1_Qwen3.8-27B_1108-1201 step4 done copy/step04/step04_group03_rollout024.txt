import time


def fast_conv(a):
    """FFT-based full linear auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fs = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fs)
    return np.fft.irfft(A * A, fs)[:size]


def cross_conv(a, b):
    """FFT-based full linear cross-convolution."""
    n = max(len(a), len(b))
    size = len(a) + len(b) - 1
    fs = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fs)
    B = np.fft.rfft(b, fs)
    return np.fft.irfft(A * B, fs)[:size]


def proj_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cv[r] - 1.0) / (r + 1)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f, peaks):
    """Build sparse constraint matrix for the LP oracle."""
    n = len(f)
    nc = len(peaks)
    # Pre-allocate
    max_entries = nc * n
    rows = np.zeros(max_entries, dtype=np.int32)
    cols = np.zeros(max_entries, dtype=np.int32)
    vals = np.zeros(max_entries, dtype=np.float64)
    count = 0
    for i in range(nc):
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        j_range = np.arange(js, je + 1)
        length = len(j_range)
        if length == 0:
            continue
        rows[count:count + length] = i
        cols[count:count + length] = j_range
        vals[count:count + length] = f[k - j_range]
        count += length
    return csr_matrix((vals[:count], (rows[:count], cols[:count])), shape=(nc, n))


def solve_lp(f, M, peaks):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in peaks, g >= 0.
    Returns the optimal g (unnormalized) or None."""
    n = len(f)
    if len(peaks) == 0:
        return None
    A = build_lp_matrix(f, peaks)
    c = -np.ones(n)
    b = np.full(len(peaks), M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def find_best_alpha(conv_f, conv_fg, conv_g, conv):
    """Find optimal FW step size via 1D search over top peaks."""
    n = len(conv_f)
    top_k = min(100, n)
    peaks = np.argpartition(conv, -top_k)[-top_k:]

    best_alpha = 0.0
    best_val = conv_f[peaks].max()

    alphas = np.linspace(0.0, 1.0, 41)
    for alpha in alphas:
        c = (1 - alpha) ** 2 * conv_f[peaks] + 2 * alpha * (1 - alpha) * conv_fg[peaks] + alpha ** 2 * conv_g[peaks]
        val = c.max()
        if val < best_val:
            best_val = val
            best_alpha = alpha

    # Refine: find the active peak and solve the quadratic exactly
    alpha = best_alpha
    c_at = (1 - alpha) ** 2 * conv_f[peaks] + 2 * alpha * (1 - alpha) * conv_fg[peaks] + alpha ** 2 * conv_g[peaks]
    active_idx = np.argmax(c_at)
    k = peaks[active_idx]
    Af = conv_f[k]
    Bf = conv_fg[k]
    Cf = conv_g[k]
    A_q = Cf - 2 * Bf + Af
    B_q = 2 * Bf - 2 * Af
    if A_q > 1e-15:
        alpha_exact = -B_q / (2.0 * A_q)
        alpha_exact = float(np.clip(alpha_exact, 0.0, 1.0))
        c_check = (1 - alpha_exact) ** 2 * conv_f[peaks] + 2 * alpha_exact * (1 - alpha_exact) * conv_fg[peaks] + alpha_exact ** 2 * conv_g[peaks]
        if c_check.max() < best_val:
            best_val = c_check.max()
            best_alpha = alpha_exact

    return best_alpha, best_val


def fw_lp_optimize(f0, time_limit, top_k=100, verbose=True):
    """Frank-Wolfe with LP oracle using top-K adaptive constraints."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f), dtype=np.float64) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_conv(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    m_exp = 1.0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 200 == 0 and verbose:
            print(f"      FW iter {it}, obj={best_obj:.10f}, M={M:.8f}, m_exp={m_exp:.2f}, t={time.monotonic()-t0:.1f}s")

        # Top-K peaks as constraints
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peaks_sorted = np.sort(peaks)

        g = solve_lp(f, M * m_exp, peaks_sorted)
        if g is None:
            m_exp *= 1.2
            if m_exp > 10.0:
                m_exp = 1.0
                rng = np.random.default_rng(it * 13 + 7)
                f = proj_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_conv(f)
                M = conv.max()
                no_improve = 0
            continue

        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 80:
                rng = np.random.default_rng(it * 7 + 3)
                scale = 0.001 + 0.004 * min(1.0, no_improve / 150.0)
                f = proj_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_conv(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            else:
                m_exp = min(m_exp * 1.05, 5.0)
            continue

        # Compute cross-convolution and g*self-conv for step size
        conv_g = fast_conv(g)
        conv_fg = cross_conv(f, g)

        alpha, predicted = find_best_alpha(conv, conv_fg, conv_g, conv)

        if alpha > 1e-10:
            f_new = (1.0 - alpha) * f + alpha * g
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                no_improve += 1
                continue
            f_new = f_new / sn
            conv_new = fast_conv(f_new)
            obj_new = conv_new.max()

            if obj_new < best_obj - 1e-15:
                f = f_new
                conv = conv_new
                M = max(conv.max(), 1e-30)
                m_exp = 1.0
                best_obj = obj_new
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                m_exp = min(m_exp * 1.03, 3.0)
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 5.0)

    return best_f, best_obj


def softmax_grad_descent(f0, time_limit, tau_init=0.005, lr_init=0.05, verbose=True):
    """Smoothed softmax gradient descent with Nesterov momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f), dtype=np.float64) / len(f)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_conv(f).max()
    best_f = f.copy()
    no_improve = 0
    cur_lr = lr_init
    cur_tau = tau_init
    momentum = np.zeros(n)
    y = f.copy()  # Nesterov lookahead

    t0 = time.monotonic()
    while time.monotonic() - time.monotonic() < time_limit or True:
        if time.monotonic() - t0 >= time_limit:
            break

        # Nesterov step
        prev_y = y
        # Compute gradient at y
        conv_y = fast_conv(y)
        scaled = conv_y / cur_tau
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        w /= ws

        # Gradient of softmax-smoothed max
        grad = np.zeros(n)
        for k_idx in range(2 * n - 1):
            if w[k_idx] < 1e-14:
                continue
            js = max(0, k_idx - n + 1)
            je = min(n - 1, k_idx)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * w[k_idx] * y[k_idx - idx]

        # Update momentum (Nesterov)
        beta = 0.9
        momentum = beta * momentum - cur_lr * grad
        y = proj_simplex(f + momentum)
        f = proj_simplex(y)

        conv = fast_conv(f)
        obj = conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9995
        cur_tau *= 1.003

        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
            f = proj_simplex(best_f + rng.standard_normal(n) * 0.003)
            y = f.copy()
            momentum = np.zeros(n)
            cur_lr = max(lr_init * 0.5, 0.005)
            cur_tau = tau_init * 1.5
            no_improve = 0
        if verbose and no_improve % 200 == 0:
            print(f"      SGD obj={best_obj:.10f}, lr={cur_lr:.6f}, tau={cur_tau:.6f}, t={time.monotonic()-t0:.1f}s")

    return best_f, best_obj
