"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast subgradient optimization
with momentum, multi-start, peak-shaving, and coordinate descent polishing.
Key innovations: softmax-weighted multi-peak subgradients, adaptive learning rates,

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(a, conv, top_k, temperature):
    """Compute softmax-weighted subgradient of max(conv)."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    # Get top-k indices
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-13:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += w[i] * a[k - idx]
    return sg

def subgradient_optimize(f0, time_limit, top_k=50, temperature=0.0001, lr=0.005, momentum=0.9, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    v = np.zeros(n)  # momentum
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temperature
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute subgradient
        sg = compute_subgradient(f, conv, top_k, cur_temp)
        
        # Nesterov: evaluate at f + momentum * v
        f_nest = f + momentum * v
        f_nest = np.maximum(f_nest, 0.0)
        sn = f_nest.sum()
        if sn > 1e-15:
            f_nest = f_nest / sn
        conv_nest = fast_convolve(f_nest)
        sg_nest = compute_subgradient(f_nest, conv_nest, top_k, cur_temp)
        
        v = momentum * v - cur_lr * sg_nest
        f = f + v
        f = project_simplex(f)
        
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive learning rate
        cur_lr *= 0.9995
        if cur_lr < 0.0001:

            total = contributions.sum()

                    continue
                # Try reducing this value
                for frac in [0.7, 0.8, 0.9]:
                    new_val = old_val * frac

                    f_trial[int(i_from)] = new_val
                    # Redistribute to others
                    diff = old_val - new_val
                    mask = np.ones(n, dtype=bool)
                    mask[int(i_from)] = False

def coordinate_descent(f0, time_limit, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_init(n, rng):
    """Create structured initial sequences with good autocorrelation properties."""
    method = rng.integers(0, 5)
    if method == 0:
        # Uniform
        return np.ones(n) / n
    elif method == 1:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        f = 0.5 + 0.5 * np.cos(x)
        return f / f.sum()
    elif method == 2:
        # Raised cosine (Hann)
        x = np.linspace(0, 1, n)
        f = 0.5 * (1 - np.cos(2 * np.pi * x))
        f = np.maximum(f, 0)
        if f.sum() < 1e-15:
            return np.ones(n) / n
        return f / f.sum()
    elif method == 3:
        # Sinc-like (truncated)
        x = np.linspace(-2, 2, n)
        f = np.sinc(x)
        f = np.maximum(f, 0)
        if f.sum() < 1e-15:
            return np.ones(n) / n
        return f / f.sum()
    else:
        # Random with smoothing
        f = np.abs(rng.standard_normal(n))
        # Smooth
        kernel = np.ones(5) / 5
        f = np.convolve(f, kernel, mode='same')
        return f / f.sum()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multiplicative-weights subgradient
optimization with Nesterov momentum, adaptive multi-peak selection, structured
restarts, coordinate descent polishing, and multi-length exploration.
Key innovations: exponential weights update (O(n) per step vs O(n log n) for
simplex projection), adaptive temperature scheduling, and pair-exchange local search.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, conv, top_k, temperature):
    """Compute softmax-weighted subgradient of max(conv(a,a))."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-13:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * a[k - idx]
    return sg

        if not improved:
            break

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    # Center-out order
    mid = n // 2
    order = []
    for d in range(mid + 1):
        if mid + d < n:
            order.append(mid + d)
        if mid - d >= 0 and d > 0:
            order.append(mid - d)
    if mid - d >= 0:
        order.append(mid - d)

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue

            conv = fast_convolve(f)
            cur_peak = conv.max()
            best_j_val = old_val
            best_j_obj = cur_peak

            for mult in [0.4, 0.6, 0.8, 1.2, 1.5, 2.0]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f /= f.sum()

                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_inits(n, rng, count=4):
    """Generate structured initial sequences."""
    inits = []
    inits.append(np.ones(n) / n)

    x = np.linspace(0, 1, n)
    hann = 0.5 * (1 - np.cos(2 * np.pi * x))
    hann = np.maximum(hann, 1e-10)
    inits.append(hann / hann.sum())

    # Sharp central peak
    center = n / 2.0
    sigma = n / (4.0 + rng.random() * 4.0)
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    inits.append(gauss / gauss.sum())

    # Random smoothed
    r = np.abs(rng.standard_normal(n))
    kernel_len = max(3, n // 100)
    kernel = np.ones(kernel_len) / kernel_len
    r = np.convolve(r, kernel, mode='same')
    r = np.maximum(r, 1e-10)
    inits.append(r / r.sum())

    # Alternating pattern (for even n)
    if n % 2 == 0:
        alt = np.ones(n)
        alt[0::2] = 1.0 + rng.random() * 0.5
        alt[1::2] = 1.0 + rng.random() * 0.5
        inits.append(alt / alt.sum())

    return inits[:count]


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: multiplicative-weights subgradient with momentum,
    pair-exchange search, coordinate descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        print("No parent found, using uniform n=512")

    # === Phase 1: Fast subgradient on parent and structured inits ===
    print("\n=== Phase 1: Multiplicative-Weights Subgradient ===")

    candidates = [parent]
    for n_try in [512, 768]:
        candidates.extend(structured_inits(n_try, rng, count=3))

    for idx, cand in enumerate(candidates):
        rem = deadline - time.monotonic() - 250
        if rem < 20:
            break
        n_cand = len(cand)
        t_alloc = min(rem * 0.12, 100)
        if t_alloc < 8:
            t_alloc = min(rem * 0.25, 50)

        print(f"  Cand {idx}: n={n_cand}, t={t_alloc:.0f}s")
        a_opt, _ = mu_weights_optimize(cand, time_limit=t_alloc,
                                        top_k=max(20, n_cand // 8),
                                        temperature=0.00004, lr=0.003,
                                        momentum=0.85)
        try_seq(a_opt, f"mu_{idx}")

    # === Phase 2: Deep optimization on best ===
    print(f"\n=== Phase 2: Deep Optimization (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        configs = [
            {"top_k": max(30, n_best // 4), "temp": 0.00002, "lr": 0.002, "mom": 0.90},
            {"top_k": max(20, n_best // 6), "temp": 0.00008, "lr": 0.004, "mom": 0.82},
            {"top_k": max(10, n_best // 10), "temp": 0.00001, "lr": 0.001, "mom": 0.93},
            {"top_k": max(40, n_best // 3), "temp": 0.0001, "lr": 0.005, "mom": 0.78},
        ]

        for p, cfg in enumerate(configs):
            rem = deadline - time.monotonic() - 150
            if rem < 15:
                break
            t_alloc = min(rem * 0.3, 180)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.006
                f_init = project_simplex(best_seq + noise)
            elif p == 2:
                f_init = 0.8 * best_seq + 0.2 * (np.ones(n_best) / n_best)
                f_init = project_simplex(f_init)
            else:
                f_init = 0.6 * best_seq + 0.4 * best_seq[::-1]
                f_init = project_simplex(f_init)

            print(f"  Deep {p}: n={n_best}, t={t_alloc:.0f}s, cfg={cfg}")
            a_opt, _ = mu_weights_optimize(f_init, time_limit=t_alloc,
                                            top_k=cfg["top_k"], temperature=cfg["temp"],
                                            lr=cfg["lr"], momentum=cfg["mom"])
            try_seq(a_opt, f"deep_{p}")

    # === Phase 3: Length exploration ===
    print(f"\n=== Phase 3: Length Exploration (best={best_score:.6f}) ===")
    if best_seq is not None:
        target_lengths = [640, 1024, 1280, 1536]
        for target_n in target_lengths:
            rem = deadline - time.monotonic() - 120
            if rem < 25:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.25, 120)
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = mu_weights_optimize(f_interp, time_limit=t_alloc,
                                            top_k=max(30, target_n // 6),
                                            temperature=0.00004, lr=0.003,
                                            momentum=0.87)
            try_seq(a_opt, f"len_{target_n}")

    # === Phase 4: Pair exchange + coordinate descent ===
    print(f"\n=== Phase 4: Local Search (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_pex = min(rem * 0.4, 80)
            print(f"  Pair exchange t={t_pex:.0f}s")
            a_opt, _ = pair_exchange_search(best_seq, time_limit=t_pex)
            try_seq(a_opt, "pair_exch")

            rem = deadline - time.monotonic() - 50
            if rem > 15:
                t_cd = min(rem * 0.7, 100)
                print(f"  Coord descent t={t_cd:.0f}s")
                a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
                try_seq(a_opt, "cd")

    # === Phase 5: Final fine-tuning ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine subgradient
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            a_opt, _ = mu_weights_optimize(best_seq, time_limit=min(rem, 60),
                                            top_k=n_best, temperature=0.000001,
                                            lr=0.0003, momentum=0.96)
            try_seq(a_opt, "sg_fine")

        # Pair exchange on fine result
        rem = deadline - time.monotonic() - 25
        if rem > 10:
            a_opt, _ = pair_exchange_search(best_seq, time_limit=min(rem, 30))
            try_seq(a_opt, "pex_fine")

        # Perturbation + quick optimize
        for _ in range(4):
            rem = deadline - time.monotonic() - 10
            if rem < 8:
                break
            for scale in [0.001, 0.003]:
                rem = deadline - time.monotonic() - 8
                if rem < 5:
                    break
                noise = rng.standard_normal(n_best) * scale
                f_pert = project_simplex(best_seq + noise)
                a_opt, _ = mu_weights_optimize(f_pert, time_limit=min(rem, 12),
                                                top_k=max(20, n_best // 5),
                                                temperature=0.00002, lr=0.001,
                                                momentum=0.92)
                try_seq(a_opt, f"pert_{scale}")

        # Coordinate descent as absolute last step
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 20))
            try_seq(a_opt, "cd_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
