Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
import time
from scipy.optimize import minimize, linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_lse_gradient(a, conv, T, top_k=50):
    """Compute gradient of (1/T)*log(sum exp(T*conv_k)) w.r.t. a."""
    n = len(a)
    # Softmax weights (optionally restricted to top-k)
    if top_k is not None and top_k < len(conv):
        # Use top-k for efficiency
        idx = np.argpartition(conv, -top_k)[-top_k:]
        vals = conv[idx]
        scaled = T * vals
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        p = np.zeros(2*n-1)
        p[idx] = w / w_sum
    else:
        scaled = T * conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        p = w / w.sum()
    
    # Cross-correlation: grad[i] = sum_j p[j+i] * a[j]
    fft_size = 1 << (3*n - 2).bit_length()
    rev_p = p[::-1]
    R = np.fft.rfft(rev_p, fft_size)
    A = np.fft.rfft(a, fft_size)
    conv_result = np.fft.irfft(R * A, fft_size)[:3*n-2]
    grad = conv_result[n-1:2*n-1][::-1]
    
    return grad

def lse_objective_and_grad(a, T, top_k=50):
    """Compute LSE objective and gradient."""
    conv = fast_convolve(a)
    scaled = T * conv
    scaled_max = scaled.max()
    log_sum_exp = scaled_max + np.log(np.exp(scaled - scaled_max).sum()) / T

    no_improve = 0
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 50:
            break
        
        # Build constraint matrix: C @ g = a * g (linear conv)
        # C is (2n-1) x n
        # C[k, j] = a[k-j] if 0 <= k-j < n, else 0
        C = np.zeros((2*n-1, n))
        for j in range(n):
            C[j:j+n, j] = a
        
        # LP: max sum(g) s.t. C@g <= M, g >= 0
        c = -np.ones(n)
        b = np.full(2*n-1, M)
        bounds = [(0, None)] * n
        
        try:

                            options={'presolve': True})
            if not result.success:
                no_improve += 1
                continue
        except:
            no_improve += 1
            continue
        
        g = result.x
        g_sum = g.sum()
        if g_sum < 1e-10:
            no_improve += 1
            continue
        
        # Find step size
        d = g - a
        best_t = 0
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:

                a = a_try
                conv = conv_try
                M = conv_try.max()
                no_improve = 0
                break
        
        if best_t == 0:

    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based gradient computation for log-sum-exp smooth approximation
2. L-BFGS optimization with adaptive temperature annealing
3. LP-based direction finding for polishing
4. Multi-restart with diverse perturbations
5. Nesterov momentum gradient descent as fallback
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_lse_gradient(a, conv, T, top_k=50):
    """
    Compute gradient of (1/T)*log(sum_k exp(T*conv_k)) w.r.t. a.
    grad[i] = sum_j p[j+i] * a[j] where p = softmax(T*conv)
    """
    n = len(a)
    if top_k is not None and top_k < len(conv):
        idx = np.argpartition(conv, -top_k)[-top_k:]
        vals = conv[idx]
        scaled = T * vals
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            p = np.zeros(2 * n - 1)
            p[idx[np.argmax(vals)]] = 1.0
        else:
            p = np.zeros(2 * n - 1)
            p[idx] = w / w_sum
    else:
        scaled = T * conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        p = w / w_sum

    # Cross-correlation via FFT: grad[i] = sum_j p[j+i] * a[j]
    fft_size = 1 << (3 * n - 2).bit_length()
    rev_p = p[::-1].copy()
    R = np.fft.rfft(rev_p, fft_size)
    A = np.fft.rfft(a, fft_size)
    conv_result = np.fft.irfft(R * A, fft_size)[:3 * n - 2]
    grad = conv_result[n - 1:2 * n - 1][::-1].copy()
    return grad


def lse_fun(x, T, top_k=50):
    """LSE objective value."""
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s < 1e-15:
        return 1e10
    x = x / s
    conv = fast_convolve(x)
    scaled = T * conv
    scaled -= scaled.max()
    return np.log(np.exp(scaled).sum()) / T


def lse_jac(x, T, top_k=50):
    """LSE gradient."""
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s < 1e-15:
        return np.zeros(len(x))
    x = x / s
    conv = fast_convolve(x)
    return compute_lse_gradient(x, conv, T, top_k)


def lbfgs_optimize_single(a0, T, time_limit, top_k=50):
    """Single L-BFGS-B run on LSE approximation."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()

    # Run L-BFGS with time limit handling
    def fun(x):
        return lse_fun(x, T, top_k)

    def jac(x):
        if time.monotonic() - t0 > time_limit:
            raise KeyboardInterrupt
        return lse_jac(x, T, top_k)

    try:
        result = minimize(
            fun, a, method='L-BFGS-B', jac=jac,
            bounds=[(0, None)] * n,
            options={'maxiter': 300, 'maxfun': 2000, 'gtol': 1e-9, 'ftol': 1e-12,
                     'maxls': 50}
        )
        a_new = np.maximum(result.x, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        conv_new = fast_convolve(a_new)
        if conv_new.max() < best_obj:
            best_obj = conv_new.max()
            best_a = a_new.copy()
    except KeyboardInterrupt:
        pass
    except Exception:
        pass

    return best_a, 2.0 * n * best_obj


def nesterov_optimize(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=50, temp=100.0):
    """Nesterov-accelerated gradient descent on LSE."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    best_conv = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    lr = lr_init
    mom = momentum
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        # Nesterov look-ahead
        y = a + mom * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y

        conv_y = fast_convolve(y)
        grad = compute_lse_gradient(y, conv_y, temp, top_k)

        # Update
        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
            v = np.zeros(n)
            break

        conv = fast_convolve(a)
        cur_conv_max = conv.max()

        if cur_conv_max < best_conv - 1e-12:
            best_conv = cur_conv_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive
        lr *= 0.999
        temp *= 1.005
        mom = min(0.97, mom + 0.0001)

        if no_improve >= 300:
            break

    return best_a, 2.0 * n * best_conv


def lp_optimize(a0, time_limit, verbose=False):
    """LP-based direction finding optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > 60:
            break

        # Build constraint matrix
        C = np.zeros((2 * n - 1, n))
        for j in range(n):
            C[j:j + n, j] = a

        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M * 1.00001)
        bounds = [(0, None)] * n

        try:
            result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                            method='highs', options={'presolve': True,
                                                     'dual_feasibility_tolerance': 1e-9})
            if not result.success:
                no_improve += 1
                iter_count += 1
                continue
        except Exception:
            no_improve += 1
            iter_count += 1
            continue

        g = result.x
        g_sum = g.sum()
        if g_sum <= 1.0 + 1e-8:
            no_improve += 1
            iter_count += 1
            continue

        d = g - a
        improved = False
        for t in [0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]:
            a_try = a + t * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            conv_try = fast_convolve(a_try)
            M_try = conv_try.max()
            obj_try = 2.0 * n * M_try
            if obj_try < best_obj - 1e-12:
                a = a_try
                conv = conv_try
                M = M_try
                best_obj = obj_try
                best_a = a_try.copy()
                no_improve = 0
                improved = True
                break

        if not improved:
            no_improve += 1

        iter_count += 1
        if verbose and iter_count % 50 == 0:
            print(f"    LP iter {iter_count}, obj={best_obj:.8f}, t={time.monotonic()-t0:.1f}s")

    return best_a, best_obj


def generate_diverse_inits(n, parent, rng, num=6):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits.append(p / ps)
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits.append(pp / pps)

    # Uniform
    inits.append(np.ones(n) / n)

    # Sine (inverted U)
    s1 = np.sin(np.pi * x / n) ** 2 + 0.05
    inits.append(s1 / s1.sum())

    # Sine (U)
    s2 = np.cos(np.pi * x / n) ** 2 + 0.05
    inits.append(s2 / s2.sum())

    # Bimodal (edges)
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    # Smooth edges
    bi[:edge] = np.linspace(1, 0.3, edge)
    bi[-edge:] = np.linspace(0.3, 1, edge)
    inits.append(bi / bi.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.25 * n)) ** 2) + 0.05
    inits.append(g / g.sum())

    # Random
    r = rng.dirichlet(np.ones(n) * 0.5)
    inits.append(r)

    # Alternating smooth
    alt = 1.0 + 0.4 * np.sin(2 * np.pi * x / max(1, n // 4))
    inits.append(np.maximum(alt, 0.01) / np.maximum(alt, 0.01).sum())

    return inits[:num]
