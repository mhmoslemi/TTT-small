"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Multi-start initialization with diverse candidates
2. LSE (log-sum-exp) gradient descent with adaptive beta and momentum
3. Coordinate descent refinement within the simplex
4. Alternating LSE and coordinate descent for final convergence
Uses FFT-based convolution and gradient computation for efficiency.
"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex (sum=1, non-negative)."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_lse_grad(weights, f):
    """Compute the gradient of LSE(conv(f,f), beta) w.r.t. f via FFT.
    Returns 2 * cross_correlation(weights, f) of length n.
    """
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]

def obj_value(f):
    """Compute the objective value (with sum normalized to 1)."""
    s = f.sum()
    if s < 1e-15:
        return float('inf')
    fn = f / s
    conv = fast_convolve(fn)
    return 2 * len(fn) * conv.max()

def lse_optimize(f0, time_limit, beta_start=15.0, beta_end=10000.0, lr_start=None, momentum_start=0.90):
    """
    Log-sum-exp smoothed gradient descent with adaptive parameters.
    f0: initial point on the simplex (sum=1, non-negative)
    Returns: (best_f, best_obj)
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    if lr_start is None:
        lr_start = 2.0 / n
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    beta = beta_start
    lr = lr_start
    mom = momentum_start
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    total_obj_history = []
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(f)
        
        # Softmax weights (numerically stable)
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w_sum = exp_bc.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(conv)
            w[bc_max_idx if (bc_max_idx := int(np.argmax(conv))) else 0] = 1.0
        else:
            w = exp_bc / w_sum
        
        # Gradient via FFT
        grad = compute_lse_grad(w, f)
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 50
            break
        grad = grad / grad_norm
        
        # Momentum update
        v = mom * v - lr * grad
        f_new = project_simplex(f + v)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.003, 6.0 / n)
        else:
            no_improve += 1
            if no_improve >= 5:
                lr *= 0.95
                mom = max(0.85, mom - 0.003)
            if no_improve > 100:
                # Restart with noise
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(1.5 / n, 0.001)
                beta = min(beta * 2.0, beta_end)
                mom = 0.88
                no_improve = 0
                # Add small noise

        # Ramp up beta

            else:
                beta *= 1.002
        
        if it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE it={it}, obj={best_obj:.12f}, beta={beta:.1f}, lr={lr:.6f}, no_improve={no_improve}, elapsed={elapsed:.0f}s")
    
    return best_f, best_obj

def coord_descent(f0, time_limit, n_sweeps_max=50):
    """
    Coordinate descent within the simplex.
    For each coordinate, try several values and keep the best.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    
    for sweep in range(n_sweeps_max):
        if time.monotonic() - t0 > time_limit:
            break
        
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            x_old = f[i]
            if x_old < 1e-12:
                # Try to increase from zero
                candidates = [0.001/n, 0.01/n, 0.1/n]
            else:
                candidates = [0.0, 0.25*x_old, 0.5*x_old, 0.75*x_old, x_old, 
                              1.25*x_old, 1.5*x_old, 2.0*x_old, max(0, 1-2*x_old)]
            
            best_x = x_old
            best_val = best_obj
            
            for x_new in candidates:
                if x_new < 0 or x_new > 1 - 1e-12:
                    continue
                # Scale others to maintain sum = 1
                if abs(x_new - x_old) < 1e-15:
                    continue
                if x_old > 1 - 1e-12:
                    # Can't scale others (they're all ~0)
                    f_trial = f.copy()
                    f_trial[i] = x_new
                    f_trial = f_trial / f_trial.sum()
                else:
                    f_trial = f.copy()
                    scale = (1.0 - x_new) / (1.0 - x_old)
                    f_trial[i] = x_new
                    for j in range(n):
                        if j != i:
                            f_trial[j] = f[j] * scale
                    # Handle floating point: renormalize
                    f_trial_sum = f_trial.sum()
                    if f_trial_sum > 1e-15:
                        f_trial = f_trial / f_trial_sum
                    else:
                        continue
                
                conv_trial = fast_convolve(f_trial)
                val = conv_trial.max()
                
                if val < best_val - 1e-15:
                    best_val = val
                    best_x = x_new
                    improved = True
            
            if best_x != x_old:
                x_old = f[i]
                if x_old > 1 - 1e-12:
                    f[i] = best_x
                    f = f / f.sum()
                else:
                    scale = (1.0 - best_x) / (1.0 - x_old)
                    f[i] = best_x
                    for j in range(n):
                        if j != i:
                            f[j] = f[j] * scale
                    f_sum = f.sum()
                    if f_sum > 1e-15:
                        f = f / f_sum
        
        conv = fast_convolve(f)
        obj = conv.max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
        
        if not improved:
            break
    
    return best_f, best_obj
