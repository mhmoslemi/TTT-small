def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)

    return np.maximum(v - theta, 0.0)

2 * w_k * f * (reversed_slice - conv[k])

where reversed_slice is the appropriately shifted and reversed version of f.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multiplicative weights update (MWU)
with adaptive peak selection, spectral-domain reshaping, and multi-phase optimization.
Key innovations over previous approach: (1) entropic/MWU parameterization for
natural positivity and aggressive relative updates, (2) spectral flattening as
a large-step restart mechanism, (3) multi-frequency structured initializations,
(4) adaptive temperature scheduling for peak selection.
"""
import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_mwu_grad(f, conv, peaks, weights):
    """Compute MWU subgradient: grad[i] = sum_k w_k * 2*f[i]*(f[k-i] - conv[k])."""
    n = len(f)
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        k = int(peaks[idx])
        w = weights[idx]
        if w < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        i_idx = np.arange(js, je + 1)
        fkm = f[k - i_idx]
        grad[i_idx] += 2.0 * w * f[i_idx] * (fkm - conv[k])
    return grad


def mwu_optimize(f0, time_limit, lr_init=0.05, lr_decay=0.9998, top_k=30,
                 temp_init=0.001, temp_growth=1.003, restart_every=400,
                 dither_scale=0.003, verbose=False):
    """MWU optimizer with entropic parameterization and adaptive peak selection."""
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)
    log_theta = np.log(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    it = 0
    no_improve = 0
    rng = np.random.default_rng(999 + it)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(f)
        current_obj = conv.max()
        
        # Select active peaks
        k_sel = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_sel)[-k_sel:]
        pv = conv[peaks]
        
        # Softmax weights
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        # Compute gradient
        grad = compute_mwu_grad(f, conv, peaks, w)
        
        # Update log_theta
        log_theta = log_theta - lr * grad
        
        # Dithering to escape local minima
        if it % restart_every == 0:
            log_theta += rng.standard_normal(n) * dither_scale
        
        # Convert back to f
        log_theta -= log_theta.max()
        f = np.exp(log_theta)
        fs = f.sum()
        if fs < 1e-30:
            f = best_f.copy()
            log_theta = np.log(f)
        else:
            f = f / fs
        
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    MWU iter {it}: obj={best_obj:.10f}, lr={lr:.6f}")
        else:
            no_improve += 1
        
        lr *= lr_decay
        temp *= temp_growth
        
        # Restart if stuck
        if no_improve > restart_every:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.008)
            log_theta = np.log(np.maximum(f, 1e-15))
            lr = lr_init * 0.7
            temp = temp_init * 2
            no_improve = 0
    
    return best_f, best_obj


def spectral_reshape(f, alpha=0.3, n_harmonics=5):
    """
    Reshape the sequence in the frequency domain to flatten the autocorrelation.
    Returns a new candidate sequence.
    """
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    A = np.fft.rfft(f, fft_size)
    power = np.abs(A) ** 2
    total_power = power.sum()
    
    # Target: flatten the power spectrum (within the passband)
    # The DC component must be preserved (it equals sum(f)^2 = 1)
    # For non-DC bins, we want power to be more uniform
    
    # Compute target: geometric mean of non-DC power
    non_dc = power[1:]
    avg_non_dc = non_dc.mean() if len(non_dc) > 0 else 0
    
    # Blend current and target
    target = power.copy()
    target[1:] = avg_non_dc  # flat target for non-DC
    
    # Interpolate in log domain
    log_current = np.log(power + 1e-30)
    log_target = np.log(target + 1e-30)
    log_new = (1 - alpha) * log_current + alpha * log_target
    
    new_power = np.exp(log_new)
    
    # Preserve phase
    phase = np.angle(A)
    new_A = np.sqrt(new_power) * np.exp(1j * phase)
    
    # Convert back
    f_new = np.fft.irfft(new_A, fft_size)[:n]
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s < 1e-15:
        return f.copy()
    return f_new / s


def sinusoidal_init(n, harmonics=None, eps=0.3, rng=None):
    """Generate a sinusoidal-based initialization."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    x = np.arange(n, dtype=np.float64)
    if harmonics is None:
        harmonics = [1, 2, 3]
    
    best_f = None
    best_obj = np.inf
    
    for h in harmonics:
        for phi in [0, np.pi/2, np.pi/3]:
            for e in [eps * 0.5, eps, eps * 1.5]:
                f = 1.0 + e * np.cos(2 * np.pi * h * x / n + phi)
                f = np.maximum(f, 0.1)  # keep positive
                f = f / f.sum()
                obj = fast_convolve(f).max()
                if obj < best_obj:
                    best_obj = obj
                    best_f = f.copy()
    
    return best_f if best_f is not None else np.ones(n) / n

    best_f = f.copy()

    it = 0
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(f)
        
        k_sel = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_sel)[-k_sel:]

        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-30:

        for i in range(len(peaks)):

            je = min(n - 1, k)

        f = project_simplex(f + v)

        obj = conv.max()

            best_obj = obj

        cur_temp *= 1.003

            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0003)
            no_improve = 0
    
    return best_f, best_obj

    it = 0

def build_conv_matrix(f):
    """Build B matrix where (B*g)[k] = (f*g)[k]."""
    n = len(f)
    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        B[j:j+n, j] = f
    return B


def solve_linearized_lp(f, conv_f, top_k=30):
    """Solve reduced linearized LP for Frank-Wolfe direction."""
    n = len(f)
    k_sel = min(top_k, 2*n-1)
    peaks = np.argpartition(conv_f, -k_sel)[-k_sel:]
    peaks = np.sort(peaks)
    
    B = build_conv_matrix(f)
    
    c = np.zeros(n + 1)
    c[0] = 1.0
    
    A_ub = np.zeros((k_sel, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = 2.0 * B[peaks, :]
    b_ub = -conv_f[peaks]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]
    return None


def linfw_step(f, conv, top_k=25):
    """Single linearized FW step."""
    n = len(f)
    g = solve_linearized_lp(f, conv, top_k=top_k)
    if g is None:
        return f, conv.max()
    
    target = f + g
    target = np.maximum(target, 0.0)
    ts = target.sum()
    if ts < 1e-15:
        return f, conv.max()
    target = target / ts
    
    # Line search
    best_alpha = 0.0
    best_obj = conv.max()
    for alpha in np.linspace(0.05, 1.0, 10):
        f_new = (1-alpha)*f + alpha*target
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-15:
            continue
        f_new = f_new/s
        obj = fast_convolve(f_new).max()
        if obj < best_obj:
            best_obj = obj
            best_alpha = alpha
    
    if best_alpha > 1e-8:
        f_new = (1-best_alpha)*f + best_alpha*target
        f_new = np.maximum(f_new, 0)
        f_new = f_new / f_new.sum()
        return f_new, best_obj
    return f, conv.max()

            break

        conv = fast_convolve(f)
        current_obj = conv.max()

        k_sel = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_sel)[-k_sel:]
        pv = conv[peaks]

        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        grad = compute_mwu_grad(f, conv, peaks, w)

        log_theta = log_theta - lr * grad

        if it % restart_every == 0:
            log_theta += rng.standard_normal(n) * dither_scale

        log_theta -= log_theta.max()
        f = np.exp(log_theta)
        fs = f.sum()
        if fs < 1e-30:
            f = best_f.copy()
            log_theta = np.log(f)
        else:
            f = f / fs

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    MWU iter {it}: obj={best_obj:.10f}, lr={lr:.6f}")
        else:
            no_improve += 1

        lr *= lr_decay
        temp *= temp_growth

        if no_improve > restart_every:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.006)
            log_theta = np.log(np.maximum(f, 1e-15))
            lr = lr_init * 0.7
            temp = temp_init * 1.5
            no_improve = 0

    return best_f, best_obj


def spectral_reshape(f, alpha=0.4):
    """Reshape sequence in frequency domain to flatten autocorrelation."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    A = np.fft.rfft(f, fft_size)
    power = np.abs(A) ** 2

    non_dc = power[1:]
    avg_non_dc = non_dc.mean() if len(non_dc) > 0 else 0
    if avg_non_dc < 1e-30:
        return f.copy()

    log_current = np.log(power + 1e-30)
    log_target = np.full_like(log_current, np.log(avg_non_dc + 1e-30))
    log_target[0] = log_current[0]

    log_new = (1 - alpha) * log_current + alpha * log_target
    new_power = np.exp(log_new)

    phase = np.angle(A)
    new_A = np.sqrt(new_power) * np.exp(1j * phase)
    f_new = np.fft.irfft(new_A, fft_size)[:n]
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s < 1e-15:
        return f.copy()
    return f_new / s


def antisymmetric_perturb(f, rng, scale=0.01):
    """Apply antisymmetric perturbation to redistribute off-center energy."""
    n = len(f)
    b = rng.standard_normal(n)
    b_anti = b - b[::-1]
    b_anti = b_anti - b_anti.mean()
    b_anti = b_anti / (np.std(b_anti) + 1e-15)
    f_new = f + scale * b_anti
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s < 1e-15:
        return f.copy()
    return f_new / s
