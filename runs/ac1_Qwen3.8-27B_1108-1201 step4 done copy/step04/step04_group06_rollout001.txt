"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an enhanced LP-based approach
with structured perturbations, mirror transformations, and multi-length search.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C

def compute_gradient(a, conv, top_k=30, temp=0.002):
    n = len(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += w[idx] * a[k - idxs]
    return grad

def lp_solve(a, M, C=None):
    n = len(a)
    if C is None:
        C = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True})
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g, g_sum, True
    except:
        pass
    return a.copy(), 1.0, False

def lp_step(a, conv, n):
    M = conv.max()
    C = build_conv_matrix(a)
    g, g_sum, success = lp_solve(a, M * 1.0001, C)
    if not success or g_sum <= 1.0 + 1e-8:
        return a.copy(), False
    d = g - a
    norm_d = np.linalg.norm(d)
    if norm_d < 1e-12:
        return a.copy(), False
    
    # Line search: try geometrically decreasing step sizes
    best_a = a.copy()
    best_obj = 2.0 * n * M
    for t in [0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]:
        a_try = np.maximum(a + t * d, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        conv_try = fast_convolve(a_try)
        obj_try = 2.0 * n * conv_try.max()
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            break
    if best_obj < 2.0 * n * M:
        return best_a, True
    return a.copy(), False

def gradient_step(a, conv, n, lr=0.01, top_k=30, temp=0.002):
    grad = compute_gradient(a, conv, top_k=top_k, temp=temp)
    a_new = np.maximum(a - lr * grad, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy(), False
    a_new = a_new / s
    conv_new = fast_convolve(a_new)
    obj_new = 2.0 * n * conv_new.max()
    obj_old = 2.0 * n * conv.max()
    if obj_new < obj_old - 1e-12:

    return a.copy(), False

def structured_perturbation(a, rng):
    """Add a random sine wave perturbation."""
    n = len(a)
    x = np.arange(n, dtype=np.float64)
    freq = rng.uniform(0.5, 5.0)
    phase = rng.uniform(0, 2 * np.pi)
    amp = rng.uniform(0.01, 0.05)
    pert = amp * np.sin(2 * np.pi * freq * x / n + phase)

"""
Enhanced LP-based optimizer with mirror transforms, structured perturbations,
adaptive step sizing, and multi-length search for minimizing
2*n*max(conv(a,a))/sum(a)^2.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = convolve(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def compute_soft_grad(a, conv, top_k=30, temp=0.002):
    """Softmax-weighted gradient of peak autocorrelation."""
    n = len(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += w[idx] * a[k - idxs]
    return grad


def lp_direction(a, M, C=None):
    """Solve LP: max sum(g) s.t. C@g <= M, g >= 0."""
    n = len(a)
    if C is None:
        C = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True})
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g, g_sum, True
    except Exception:
        pass
    return a.copy(), 1.0, False


def lp_step_with_search(a, conv, n):
    """LP step with adaptive line search (bisection between 0 and best found)."""
    M = conv.max()
    C = build_conv_matrix(a)

    if not success or g_sum <= 1.0 + 1e-6:
        return a.copy(), False

    d = g - a
    if np.linalg.norm(d) < 1e-12:
        return a.copy(), False

    # Try a range of step sizes, preferring larger
    base_obj = 2.0 * n * M
    best_a = a.copy()
    best_obj = base_obj

    # First try large steps
    for t in [1.0, 0.6, 0.4, 0.25]:
        a_try = np.maximum(a + t * d, 0.0)
        s = a_try.sum()

            continue
        a_try = a_try / s
        cv = fast_convolve(a_try)
        obj = 2.0 * n * cv.max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_try.copy()

    if best_obj < base_obj - 1e-14:
        return best_a, True

            break

    return a.copy(), False


def gradient_step(a, conv, n, lr, top_k=40, temp=0.002):
    """Single gradient descent step with soft-max peak approximation."""
    grad = compute_soft_grad(a, conv, top_k=top_k, temp=temp)

    s = a_new.sum()

    obj_old = 2.0 * n * conv.max()

    # Alternating
    alt = np.ones(n)
    alt[::2] = 1.5
    alt[1::2] = 0.5
    inits['alternating'] = alt / alt.sum()

    # Power law
    pw = (x + 1.0) ** 0.5
    inits['power'] = pw / pw.sum()

    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n),
                               np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: hybrid LP + Nesterov gradient optimization with
    structured perturbations, mirror transforms, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: n={len(p)}, sum={p.sum():.4f}")
            ps = float(evaluate_sequence(p.tolist()))
            if np.isfinite(ps):
                best_score = ps
                best_sequence = p.copy()
                print(f"Parent score: {ps:.8f}")
    except Exception:
        pass

    # ============================================================
    # Phase 1: Quick scan across lengths and initializations
    # ============================================================
    print("=== Phase 1: Quick Scan ===")
    t1 = time.monotonic()

    lengths = [384, 512, 640, 768, 1024]
    if parent is not None:
        pn = len(parent)
        if pn not in lengths:
            lengths.insert(2, pn)

    results = []
    n_phase1 = min(200.0, (budget_s - 25) * 0.12)

    for n in lengths:
        if time.monotonic() > deadline - (budget_s - 25) * 0.5:
            break
        inits = generate_inits(n, parent, rng)
        # Prioritize parent-based
        priority = ['parent', 'parent_interp', 'uniform', 'sine1', 'random',
                    'sine2', 'bimodal', 'alternating', 'sine3', 'random2', 'power']
        init_names = [nm for nm in priority if nm in inits][:5]

        t_per = max(8.0, n_phase1 / (len(lengths) * 5))
        for name in init_names:
            if time.monotonic() > deadline - (budget_s - 25) * 0.4:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - (budget_s - 25) * 0.3)
            if t_alloc < 4:
                break
            a0 = inits[name]
            a_opt, obj = optimize_hybrid(a0, time_limit=t_alloc, rng=rng)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  P1 n={n} {name}: {score:.8f}")

    results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t1:.0f}s). Best: {best_score:.8f}")

    # ============================================================
    # Phase 2: Deep optimization on top candidates
    # ============================================================
    print("=== Phase 2: Deep Optimization ===")
    t2 = time.monotonic()

    top_configs = []
    seen_n = set()
    for score, n, name, a_opt in results:
        if n not in seen_n:
            top_configs.append((n, a_opt))
            seen_n.add(n)
        if len(top_configs) >= 4:
            break
    if best_sequence is not None and len(best_sequence) not in seen_n:
        top_configs.append((len(best_sequence), best_sequence.copy()))
        seen_n.add(len(best_sequence))

    remaining = deadline - time.monotonic() - (budget_s - 25) * 0.15
    n_cand = max(1, len(top_configs))
    t_per_deep = max(20.0, remaining * 0.55 / n_cand)

    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - (budget_s - 25) * 0.12:
            break
        t_alloc = min(t_per_deep, deadline - time.monotonic() - (budget_s - 25) * 0.1)
        if t_alloc < 10:
            break
        print(f"  Deep {i}: n={n}, t={t_alloc:.0f}s")
        a_opt, obj = optimize_hybrid(a0, time_limit=t_alloc, rng=rng)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Deep {i} n={n}: {score:.8f} *** NEW BEST ***")
            else:
                print(f"  Deep {i} n={n}: {score:.8f}")

    print(f"Phase 2 done ({time.monotonic()-t2:.0f}s). Best: {best_score:.8f}")

    # ============================================================
    # Phase 3: Intensive polish on the best solution
    # ============================================================
    print("=== Phase 3: Polish ===")
    t3 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20.0
        if remaining > 15:
            # Multiple polish attempts with different strategies
            n_polish = max(1, int(remaining / 80))
            n_polish = min(n_polish, 5)

            for pi in range(n_polish):
                if time.monotonic() > deadline - 10:
                    break
                t_alloc = min(remaining / (n_polish - pi) * 0.8,
                              deadline - time.monotonic() - 15)
                if t_alloc < 8:
                    break

                if pi == 0:
                    a0 = best_sequence.copy()
                elif pi == 1:
                    a0 = mirror_perturb(best_sequence, rng)
                elif pi == 2:
                    a0 = structured_perturb(best_sequence, rng)
                else:
                    a0 = gaussian_perturb(best_sequence, rng,
                                          scale=0.002 * pi)

                a_opt, obj = optimize_hybrid(a0, time_limit=t_alloc, rng=rng)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score):
                    if score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  Polish {pi}: {score:.8f} *** NEW BEST ***")
                    else:
                        print(f"  Polish {pi}: {score:.8f}")

    print(f"Phase 3 done ({time.monotonic()-t3:.0f}s). Best: {best_score:.8f}")

    # ============================================================
    # Phase 4: Final fine-tuning with Nesterov gradient
    # ============================================================
    print("=== Phase 4: Final Fine-tune ===")
    if best_sequence is not None and time.monotonic() < deadline - 5:
        t_remaining = deadline - time.monotonic() - 3
        if t_remaining > 5:
            # Try a few gradient configurations
            configs = [
                {'lr_init': 0.004, 'momentum': 0.94, 'top_k': 60, 'temp_init': 0.001},
                {'lr_init': 0.008, 'momentum': 0.91, 'top_k': 30, 'temp_init': 0.003},
                {'lr_init': 0.002, 'momentum': 0.96, 'top_k': 80, 'temp_init': 0.0005},
            ]
            t_each = t_remaining / len(configs)
            for gi, cfg in enumerate(configs):
                if time.monotonic() > deadline - 3:
                    break
                a_opt, obj = nesterov_optimize(best_sequence, time_limit=t_each,
                                                rng=rng, **cfg)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  FineTune {gi}: {score:.8f} *** NEW BEST ***")

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
