optimization with Adam, LP-based Frank-Wolfe, and multi-start exploration.
Key innovation: FFT-accelerated smooth gradient via log-sum-exp approximation
of the max convolution value, providing much better descent directions than
raw subgradient methods.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_grad_fft(a, w):
    """Compute 2*(w cross-correlate a)[0..n-1] using FFT.
    This gives d/d a[j] of sum_k w[k]*c[k] where c=conv(a,a)."""
    n = len(a)
    a_rev = a[::-1]
    N = 1 << (3 * n).bit_length()
    A = np.fft.rfft(a_rev, N)
    W = np.fft.rfft(w, N)
    conv = np.fft.irfft(A * W, N)

def project_simplex(v):
    """Euclidean projection onto simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 60})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def fine_polish(a0, time_limit, lr=0.0005, mom=0.93, top_k=64, tau=0.0002):
    """Very fine polishing with small steps."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    no_improve = 0

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.5:
            break

        obj = c.max()
        if obj < best_obj - 1e-16:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Compute gradient
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        w = np.zeros(len(c))
        c_top = c[top_idx]
        scaled = c_top / tau
        scaled -= scaled.max()
        exp_vals = np.exp(scaled)
        wsum = exp_vals.sum()
        if wsum > 0:
            w[top_idx] = exp_vals / wsum

        grad = compute_grad_fft(a, w)
        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / sn
        a = a_new
        c = fast_convolve(a)

        if no_improve > 3000:
            rng = np.random.default_rng(it + 200)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            c = fast_convolve(a)
            lr = max(0.0001, lr * 0.5)
            no_improve = 0

    return best_a, best_obj


def generate_varied_inits(n, parent, rng):
    """Generate diverse initializations for a given length."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Interpolated parent
    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    # Uniform
    inits.append(np.ones(n) / n)

    # Slight modifications of parent
    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        for noise in [0.005, 0.01, 0.02]:
            pn = px + rng.standard_normal(n) * noise
            pn = np.maximum(pn, 0)
            pn = pn / pn.sum()
            inits.append(pn)

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function: combines LSE-Adam, momentum LSE, LP Frank-Wolfe,
    and fine polishing with adaptive time allocation across multiple lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        if a.sum() < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            old = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}) (was {old:.8f})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score~1.5056")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384
        print("No parent, using uniform n=384")

    p_norm = parent / parent.sum()
    n_p = len(p_norm)

    # === Phase 1: LSE-Adam on parent (main workhorse) ===
    print("\n=== Phase 1: LSE-Adam on parent (n={}) ===".format(n_p))
    t1 = min(500, deadline - time.monotonic() - 400)
    if t1 > 50:
        print(f"  Running LSE-Adam for {t1:.0f}s...")
        a_opt, obj = lse_adam(p_norm, time_limit=t1, tau0=0.008, tau_min=0.0003,
                              tau_decay=0.9996, lr=0.004, beta1=0.85, beta2=0.999,
                              top_k=32, restart_every=10000, verbose=True)
        try_seq(a_opt, "lse_adam_p1")

    # === Phase 2: LSE-Momentum (different optimizer, same landscape) ===
    print(f"\n=== Phase 2: LSE-Momentum (n={n_p}) ===")
    t2 = min(250, deadline - time.monotonic() - 250)
    if t2 > 30:
        print(f"  Running LSE-Momentum for {t2:.0f}s...")
        a_opt2, obj2 = lse_momentum(best_seq, time_limit=t2, tau0=0.005, tau_min=0.0002,
                                    tau_decay=0.9997, lr=0.002, mom=0.92, top_k=48,
                                    verbose=True)
        try_seq(a_opt2, "lse_mom_p2")

    # === Phase 3: Try nearby lengths ===
    print(f"\n=== Phase 3: Nearby lengths ===")
    nearby_lengths = []
    for delta in [-8, -4, 4, 8, 16, -16]:
        nl = n_p + delta
        if 128 <= nl <= 1024:
            nearby_lengths.append(nl)
    # Also try some larger lengths
    nearby_lengths.extend([n_p * 2, 512, 640])
    nearby_lengths = list(set(l for l in nearby_lengths if l != n_p and 128 <= l <= 1024))
    nearby_lengths = sorted(nearby_lengths)[:4]  # Limit to 4

    for nl in nearby_lengths:
        rem = deadline - time.monotonic() - 150
        if rem < 30:
            break
        t_alloc = min(80, rem // max(1, len(nearby_lengths)))
        if t_alloc < 20:
            break
        print(f"  Trying n={nl}, t={t_alloc:.0f}s...")
        inits = generate_varied_inits(nl, p_norm, rng)
        for i, f_init in enumerate(inits[:2]):
            rem = deadline - time.monotonic() - 120
            if rem < 15:
                break
            t_a = min(40, rem // 2)
            if t_a < 10:
                break
            a_opt, obj = lse_adam(f_init, time_limit=t_a, tau0=0.01, tau_min=0.0004,
                                  tau_decay=0.9996, lr=0.004, top_k=32,
                                  restart_every=5000)
            try_seq(a_opt, f"n{nl}_i{i}")

    # === Phase 4: LP Frank-Wolfe on best ===
    print(f"\n=== Phase 4: LP Frank-Wolfe ===")
    print(f"  Current best: {best_score:.8f}")

    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(3):
            rem = deadline - time.monotonic() - 80
            if rem < 20:
                break
            t_alloc = min(rem // (3 - p), 100)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)
            else:
                f_init = project_simplex(0.9 * best_seq + 0.1 * (best_seq[::-1]))

            print(f"  FW-LP run {p}, t={t_alloc:.0f}s...")
            a_opt, obj = fw_lp_optimize(f_init, time_limit=t_alloc)
            try_seq(a_opt, f"fw_{p}")

    # === Phase 5: Deep LSE-Adam on best ===
    print(f"\n=== Phase 5: Deep LSE-Adam ===")
    print(f"  Current best: {best_score:.8f}")

    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(3):
            rem = deadline - time.monotonic() - 60
            if rem < 20:
                break
            t_alloc = min(rem // (3 - p), 80)

            if p == 0:
                f_init = best_seq.copy()
                cfg = dict(tau0=0.004, tau_min=0.0001, tau_decay=0.9998,
                           lr=0.002, top_k=24, restart_every=15000)
            elif p == 1:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.002)
                cfg = dict(tau0=0.006, tau_min=0.0002, tau_decay=0.9997,
                           lr=0.003, top_k=40, restart_every=8000)
            else:
                f_init = project_simplex(0.95 * best_seq + 0.05 * (np.ones(n_best) / n_best))
                cfg = dict(tau0=0.003, tau_min=0.0001, tau_decay=0.9998,
                           lr=0.001, top_k=16, restart_every=20000)

            print(f"  Deep LSE run {p}, t={t_alloc:.0f}s...")
            a_opt, obj = lse_adam(f_init, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"deep_lse_{p}")

    # === Phase 6: Final fine polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    print(f"  Current best: {best_score:.8f}")

    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(4):
            rem = deadline - time.monotonic() - 10
            if rem < 8:
                break
            t_alloc = min(rem // (4 - p), 30)
            if t_alloc < 5:
                break

            if p == 0:
                f_init = best_seq.copy()
            else:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.0005)

            a_opt, obj = fine_polish(f_init, time_limit=t_alloc,
                                     lr=0.0003, mom=0.95, top_k=48, tau=0.0001)
            try_seq(a_opt, f"polish_{p}")

        # Ultimate micro-polish
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_opt, obj = fine_polish(best_seq, time_limit=rem,
                                     lr=0.0001, mom=0.97, top_k=32, tau=0.00005)
            try_seq(a_opt, "micro_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
