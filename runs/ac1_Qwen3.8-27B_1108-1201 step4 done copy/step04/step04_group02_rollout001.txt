def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_objective(a):
    """Compute 2n*max(a*a)/(sum(a))^2."""
    n = len(a)
    conv = fast_convolve(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * conv.max() / (s * s)


def project_simplex(v, target_sum=1.0):
    """Project onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return np.array([target_sum])
    v = np.array(v, dtype=np.float64)
    # Project to sum=x, x>=0
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fw_lp_step(f, M, time_limit_lp=10.0):
    """Single Frank-Wolfe step: solve LP max sum(g) s.t. (f*g)[k] <= M, g>=0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit_lp})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

            rem = deadline - time.monotonic() - 100
            if rem < 20:
                break
            t_alloc = min(rem // (4 - p), 100)
            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.004,
                                         target_sum=best_seq.sum())
            elif p == 2:
                f_init = project_simplex(0.9 * best_seq + 0.1 * best_seq[::-1],
                                         target_sum=best_seq.sum())
            else:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.01,
                                         target_sum=best_seq.sum())
            a_opt, _ = coord_descent(f_init, time_limit=t_alloc)
            try_seq(a_opt, f"deep_cd_{p}")

        # FW-LP runs
        for p in range(3):
            rem = deadline - time.monotonic() - 80
            if rem < 20:
                break
            t_alloc = min(rem // (3 - p), 80)
            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.003,
                                         target_sum=best_seq.sum())
            else:
                f_init = project_simplex(0.85 * best_seq + 0.15 * (np.ones(n_best) / n_best),
                                         target_sum=best_seq.sum())
            a_opt, _ = fw_optimize(f_init, time_limit=t_alloc)
            try_seq(a_opt, f"deep_fw_{p}")
            # CD polish after FW
            rem = deadline - time.monotonic() - 60
            if rem > 15:
                a_opt2, _ = coord_descent(a_opt, time_limit=min(rem, 30))
                try_seq(a_opt2, f"deep_fw_cd_{p}")

        # Smoothed subgradient runs
        for p in range(2):
            rem = deadline - time.monotonic() - 60
            if rem < 20:
                break
            t_alloc = min(rem // (2 - p), 60)
            if p == 0:
                f_init = best_seq.copy()
                cfg = {"tau_init": 0.002, "lr_init": 0.003}
            else:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.005,
                                         target_sum=best_seq.sum())
                cfg = {"tau_init": 0.005, "lr_init": 0.004}
            a_opt, _ = smoothed_subgrad(f_init, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"deep_sg_{p}")
            # CD polish
            rem = deadline - time.monotonic() - 40
            if rem > 10:
                a_opt2, _ = coord_descent(a_opt, time_limit=min(rem, 20))
                try_seq(a_opt2, f"deep_sg_cd_{p}")

    # === Phase 4: Extended exploration with larger n ===
    print(f"\n=== Phase 4: Extended Exploration ===")
    larger_lengths = [640, 768]
    for n in larger_lengths:
        rem = deadline - time.monotonic() - 60
        if rem < 40:
            break
        # Interpolate from best if available
        if best_seq is not None:
            px = np.interp(np.arange(n, dtype=np.float64),
                           np.linspace(0, 1, len(best_seq)), best_seq)
            f_init = px / px.sum()
        else:
            f_init = np.ones(n) / n
        t_alloc = min(rem // 2, 120)
        a_opt, _ = coord_descent(f_init, time_limit=t_alloc)
        try_seq(a_opt, f"n{n}_explore")

    # === Phase 5: Final intensive polishing ===
    print(f"\n=== Phase 5: Final Polish ===")
    print(f"  Current best: {best_score:.8f}")

    if best_seq is not None:
        n_best = len(best_seq)

        # Repeated CD with tiny perturbations
        for p in range(8):
            rem = deadline - time.monotonic() - 10
            if rem < 8:
                break
            t_alloc = min(rem, 25)
            if p < 3:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.001,
                                         target_sum=best_seq.sum())
            elif p < 6:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.0005,
                                         target_sum=best_seq.sum())
            else:
                f_init = best_seq.copy()
            a_opt, _ = coord_descent(f_init, time_limit=t_alloc, tol=1e-14, max_sweeps=200)
            try_seq(a_opt, f"polish_{p}")

        # Final FW on very best
        rem = deadline - time.monotonic() - 5
        if rem > 15:
            a_opt, _ = fw_optimize(best_seq, time_limit=min(rem, 30))
            try_seq(a_opt, "final_fw")
            # CD after FW
            rem = deadline - time.monotonic() - 3
            if rem > 5:
                a_opt2, _ = coord_descent(a_opt, time_limit=min(rem, 10), tol=1e-14)
                try_seq(a_opt2, "final_fw_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
