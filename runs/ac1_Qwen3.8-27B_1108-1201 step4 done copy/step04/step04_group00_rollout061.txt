def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_grad(a, t):
    """Compute gradient of t*log(sum(exp(conv/t))) w.r.t. a."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = conv / t
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    # grad[i] = 2 * (w * a_flip)[i + n - 1]
    a_flip = a[::-1]
    conv_len = (2*n-1) + n - 1  # = 3n-2
    fft_size = 1 << (conv_len).bit_length()
    W = np.fft.rfft(w, fft_size)
    Af = np.fft.rfft(a_flip, fft_size)
    conv_result = np.fft.irfft(W * Af, fft_size)
    grad = 2.0 * conv_result[n-1 : 2*n-1]
    return grad

def smooth_obj_value(a, t):
    """Compute t*log(sum(exp(conv/t)))."""
    conv = fast_convolve(a)
    scaled = conv / t
    scaled -= scaled.max()
    return t * (np.log(np.sum(np.exp(scaled))) + scaled.max())

                             momentum_init=0.9, verbose=False):

    s = a.sum()
    if s < 1e-15:

        a = a / s

    current_max = conv.max()

    if t_min is None:
        t_min = current_max / 500.0

    momentum = np.zeros(n)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth gradient descent with
adaptive temperature, momentum, multi-length exploration, and coordinate descent
polishing. Uses log-sum-exp smooth approximation of the max function for efficient
gradient computation via FFT.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_gradient(a, t):
    """
    Compute gradient of F_t(a) = t * log(sum(exp(conv(a,a)/t))) w.r.t. a.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    conv = fast_convolve(a)
    scaled = conv / t
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()

    # grad[i] = 2 * (w * a_flip)[i + n - 1] where * is convolution
    a_flip = a[::-1]
    conv_len = (2 * n - 1) + n - 1
    fft_size = 1 << (conv_len).bit_length()
    W = np.fft.rfft(w, fft_size)
    Af = np.fft.rfft(a_flip, fft_size)
    conv_result = np.fft.irfft(W * Af, fft_size)
    grad = 2.0 * conv_result[n - 1:2 * n - 1]
    return grad


def peak_subgradient(a):
    """Compute subgradient of max(convolve(a,a)) w.r.t. a."""
    n = len(a)
    conv = fast_convolve(a)
    k_star = np.argmax(conv)
    sg = np.zeros(n)
    js = np.arange(n)
    partners = k_star - js
    valid = (partners >= 0) & (partners < n)
    sg[valid] = 2.0 * a[partners[valid]]
    return sg


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def smooth_gradient_descent(a0, time_limit, lr_init=0.005, mu_init=0.9,
                            t_ratio_init=0.3, t_ratio_min=0.001,
                            verbose=False, seed=42):
    """
    Smooth gradient descent with momentum and adaptive temperature.
    Optimizes F_t(a) = t*log(sum(exp(conv/t))) which approximates max(conv).
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    max_val = conv.max()
    best_obj = max_val
    best_a = a.copy()

    t = t_ratio_init * max_val
    t_min = t_ratio_min * max_val
    lr = lr_init
    mu = mu_init
    momentum = np.zeros(n)

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        grad = smooth_gradient(a, t)
        momentum = mu * momentum - lr * grad
        a_new = project_simplex(a + momentum)

        conv_new = fast_convolve(a_new)
        new_max = conv_new.max()

        if new_max < best_obj - 1e-13:
            best_obj = new_max
            best_a = a_new.copy()
            no_improve = 0
            t = max(t * 0.998, t_min)
            lr = min(lr * 1.0005, lr_init * 2.0)
        else:
            no_improve += 1
            t = max(t * 0.9997, t_min)

        a = a_new

        # Adaptive parameters
        if no_improve > 200:
            mu = max(mu * 0.995, 0.5)
            lr = max(lr * 0.99, 0.0005)

        # Restart with noise
        if no_improve > 800:
            noise_scale = 0.005 + 0.005 * min(no_improve // 1000, 5)
            noise = rng.standard_normal(n) * noise_scale
            a = project_simplex(best_a + noise)
            momentum = np.zeros(n)
            lr = lr_init * 0.7
            mu = mu_init
            no_improve = 0
            conv = fast_convolve(a)
            t = max(0.1 * conv.max(), t_min)

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD iter {it}: obj={best_obj:.10f}, t={t:.2e}, lr={lr:.2e}, elapsed={elapsed:.1f}s")

    return best_a, best_obj


def coordinate_descent_polish(a0, time_limit, n_trials=5, verbose=False):
    """
    Coordinate descent: for each coordinate, try several multipliers
    and accept the best improvement.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    multipliers = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.7]

    passes = 0
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = fast_convolve(a).max()

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 1.0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()
                    if best_j_obj < best_obj - 1e-14:
                        best_obj = best_j_obj
                        best_a = a.copy()
                        improved = True

        if not improved:
            break

    return best_a, best_obj


def block_sgd(a0, time_limit, block_size=32, lr=0.002, verbose=False):
    """
    Block coordinate descent using smooth gradient within each block.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for start in range(0, n, block_size):
            if time.monotonic() - t0 > time_limit:
                break
            end = min(start + block_size, n)
            # Optimize this block with a few gradient steps
            for _ in range(3):
                t_block = 0.1 * fast_convolve(a).max()
                t_block = max(t_block, 1e-6)
                grad_full = smooth_gradient(a, t_block)
                # Only update the block
                a_block = a[start:end]
                grad_block = grad_full[start:end]
                a_block_new = a_block - lr * grad_block
                a_block_new = np.maximum(a_block_new, 0)
                # Rescale block to maintain sum
                rest_sum = 1.0 - a_block.sum()
                block_sum = a_block_new.sum()
                if block_sum > 1e-15:
                    a_block_new = a_block_new * (a_block.sum() / block_sum)
                a[start:end] = a_block_new
                a = a / a.sum()

            new_obj = fast_convolve(a).max()
            if new_obj < best_obj - 1e-13:
                best_obj = new_obj
                best_a = a.copy()
                improved = True

        if not improved:
            break

    return best_a, best_obj


def peak_targeted_update(a, n_trials=10):
    """
    Target the peak of the convolution: find coordinates that contribute
    most to the peak and try to reduce them.
    """
    n = len(a)
    conv = fast_convolve(a)
    k_star = np.argmax(conv)
    best_a = a.copy()
    best_obj = conv.max()

    # Find contributions to the peak
    contributions = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            contributions[i] = a[i] * a[j]

    # Sort by contribution (largest first)
    order = np.argsort(contributions)[::-1]
    top_indices = order[:min(50, n)]

    for idx in top_indices:
        old_val = a[idx]
        if old_val < 1e-10:
            continue
        # Try reducing this coordinate
        for frac in [0.5, 0.7, 0.85]:
            new_val = old_val * frac
            denom = 1.0 - old_val
            if denom < 1e-10:
                break
            scale = (1.0 - new_val) / denom
            a_trial = a.copy()
            a_trial[idx] = new_val
            mask = np.ones(n, dtype=bool)
            mask[idx] = False
            a_trial[mask] *= scale
            val = fast_convolve(a_trial).max()
            if val < best_obj - 1e-14:
                best_obj = val
                best_a = a_trial.copy()
    return best_a, best_obj
