def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_gradient(a, p):
    """Compute gradient of LSE objective.
    a: length n (non-negative, sums to 1)
    p: length 2n-1 (probability weights from LSE)
    Returns: gradient of length n
    """
    n = len(a)
    # grad[m] = 2 * sum_{l=0}^{n-1} p[m+l] * a[l]
    # Use FFT for efficiency
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(np.conj(P) * A, fft_size)
    return 2.0 * corr[:n]


def lse_gradient_descent(f0, time_limit, t0=50, t_end=500, lr0=0.05, lr_end=0.001):
    """Gradient descent on LSE-smoothed max convolution."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    t = t0
    lr = lr0
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t_start = time.monotonic()
    no_improve = 0
    total_iters = 0
    
    while time.monotonic() - t_start < time_limit:
        conv = fast_convolve(f)
        # Compute LSE weights
        shifted = conv - conv.max()
        exp_vals = np.exp(t * shifted)
        total = exp_vals.sum()
        if total < 1e-30:
            p = np.zeros(len(conv))
            p[conv.argmax()] = 1.0
        else:
            p = exp_vals / total

            f = f_new

        # Anneal

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. LP-based Frank-Wolfe for coarse optimization (active constraints only)
2. LSE-smoothed gradient descent with FFT-based gradient for fine optimization
3. Spectral shaping to flatten power spectrum
4. Coordinate descent for final polish

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(a, p):
    """Compute gradient of LSE-smoothed max convolution objective.
    grad[m] = 2 * sum_l a[l] * p[m+l]
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    p_len = len(p)
    total_len = n + p_len - 1
    fft_size = 1 << (total_len - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(np.conj(P) * A, fft_size)
    return 2.0 * corr[:n]


def lse_gradient_descent(f0, time_limit, t_init=30.0, t_end=300.0, lr_init=0.03, lr_end=0.0005):
    """Gradient descent on LSE-smoothed max auto-convolution with annealing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s

    best_f = f.copy()

        remaining = time_limit - (time.monotonic() - t_start)
        if remaining < 0.1:
            break

        t_cur = t_init * (t_end / t_init) ** (progress ** 0.8)
        lr = lr_init * (lr_end / lr_init) ** (progress ** 0.6)

        conv = fast_convolve(f)
        shifted = conv - conv.max()
        exp_vals = np.exp(t_cur * shifted)
        total = exp_vals.sum()
        if total < 1e-30:
            p = np.zeros(len(conv))
            p[conv.argmax()] = 1.0
        else:
            p = exp_vals / total

        grad = compute_lse_grad(f, p)

        f_new = project_simplex(f - lr * grad)
        conv_new = fast_convolve(f_new)
        obj = conv_new.max()

            best_obj = obj
            best_f = f_new.copy()
            f = f_new
            no_improve = 0

def coordinate_descent_polish(f0, time_limit):

    f = np.maximum(f0.astype(np.float64), 0.0)

        order = rng_local.permutation(n)
        for j in order:
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            # Try moving mass from j to nearby positions
            for source in [(j + 1) % n, (j - 1) % n, (j + n // 2) % n,
                           (j + n // 3) % n, rng_local.integers(n)]:
                source = int(source) % n

                    if f_new[source] < 0:

def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    # Edge-concentrated
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine taper
    cos_t = np.cos(np.pi * x / (2.0 * (n - 1)))
    cos_t = np.maximum(cos_t, 0.01)
    inits.append(cos_t / cos_t.sum())

    # Two-peak
    peak = (np.exp(-((x - n / 4.0) ** 2) / (2.0 * (n / 10.0) ** 2)) +
            np.exp(-((x - 3.0 * n / 4.0) ** 2) / (2.0 * (n / 10.0) ** 2)))
    inits.append(peak / peak.sum())

    # Sine window
    sine = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine / sine.sum())

    # Chirp-inspired (quadratic phase magnitude)
    chirp = np.ones(n)
    for i in range(n):
        chirp[i] = 0.5 + 0.5 * np.cos(2 * np.pi * i * i / (4.0 * n))
    inits.append(np.maximum(chirp, 0.01) / np.maximum(chirp, 0.01).sum())

    # Parent interpolation
    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0.0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:

            print(f"Parent: n={len(parent)}, sum={parent.sum():.4f}")
            s = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(s):
                print(f"  Parent score: {s:.6f}")
                state['best_score'] = s
                state['best_seq'] = parent.copy()
    except Exception:
        pass

    if parent is not None:
        print(f"Initial best: {state['best_score']:.6f}")

    # === Phase 1: Multi-length exploration (30% time) ===
    print("\n=== Phase 1: Exploration ===")
    phase1_end = time.monotonic() + (float(budget_s) * 0.30)

    lengths_to_try = [350, 368, 384, 400, 416]
    if parent is not None:
        pn = len(parent)
        # Ensure parent length is included
        if pn not in lengths_to_try:
            lengths_to_try.append(pn)
            lengths_to_try.sort()
        # Put parent length first
        lengths_to_try.remove(pn)
        lengths_to_try.insert(0, pn)

    for n in lengths_to_try:
        if time.monotonic() > phase1_end - 40:
            break
        inits = generate_inits(n, parent, rng)

        # Try 2 initializations per length
        n_try = min(2, len(inits))
        t_per = max(15, (phase1_end - time.monotonic() - 30) / (n_try * max(1, len(lengths_to_try) - lengths_to_try.index(n))))
        t_per = min(t_per, 45)

        for i in range(n_try):
            if time.monotonic() > phase1_end - 20:
                break
            t_alloc = min(t_per, phase1_end - time.monotonic() - 10)
            if t_alloc < 10:
                break
            f_init = inits[i]
            # Quick FW
            a_opt, obj = fw_lp_optimize(f_init, time_limit=t_alloc)
            try_sequence(a_opt, f"p1_n{n}_i{i}", state)

    print(f"  Phase 1 best: {state['best_score']:.6f}")

    # === Phase 2: Deep optimization (45% time) ===
    print(f"\n=== Phase 2: Deep Optimization ===")
    phase2_end = time.monotonic() + (float(budget_s) * 0.45)

    if state['best_seq'] is not None:
        f_best = state['best_seq'].copy()
        n_best = len(f_best)

        # 2a: LSE gradient descent (main workhorse)
        rem = phase2_end - time.monotonic() - 120
        if rem > 30:
            print(f"  LSE-GD on n={n_best}, t={min(rem, 180):.0f}s")
            a_opt, obj = lse_gradient_descent(f_best, time_limit=min(rem, 180),
                                              t_init=20.0, t_end=200.0,
                                              lr_init=0.04, lr_end=0.001)
            try_sequence(a_opt, "p2_lse1", state)

        # 2b: Spectral shaping
        rem = phase2_end - time.monotonic() - 60
        if rem > 20:
            print(f"  Spectral shaping, t={min(rem, 80):.0f}s")
            a_opt, obj = spectral_shaping(state['best_seq'], time_limit=min(rem, 80))
            try_sequence(a_opt, "p2_spectral", state)

        # 2c: FW-LP deep run
        rem = phase2_end - time.monotonic() - 30
        if rem > 30:
            print(f"  FW-LP deep, t={min(rem, 120):.0f}s")
            a_opt, obj = fw_lp_optimize(state['best_seq'], time_limit=min(rem, 120),
                                        active_frac=0.2, verbose=True)
            try_sequence(a_opt, "p2_fw_deep", state)

        # 2d: Perturbation + LSE
        for ps in [0.005, 0.01, 0.02]:
            rem = phase2_end - time.monotonic() - 20
            if rem < 20:
                break
            noise = rng.standard_normal(n_best) * ps
            f_init = project_simplex(state['best_seq'] + noise)
            a_opt, obj = lse_gradient_descent(f_init, time_limit=min(rem, 60),
                                              t_init=40.0, t_end=400.0,
                                              lr_init=0.02, lr_end=0.0005)
            try_sequence(a_opt, f"p2_pert_{ps}", state)

        # 2e: Mirror blend + LSE
        rem = phase2_end - time.monotonic() - 20
        if rem > 25:
            f_mirror = project_simplex(0.7 * state['best_seq'] + 0.3 * state['best_seq'][::-1])
            a_opt, obj = lse_gradient_descent(f_mirror, time_limit=min(rem, 60),
                                              t_init=50.0, t_end=500.0,
                                              lr_init=0.015, lr_end=0.0003)
            try_sequence(a_opt, "p2_mirror", state)

    print(f"  Phase 2 best: {state['best_score']:.6f}")

    # === Phase 3: Polishing (remaining time) ===
    print(f"\n=== Phase 3: Polishing ===")
    phase3_end = deadline - 15

    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])

        # 3a: Fine LSE (higher temperature for more precise targeting)
        rem = phase3_end - time.monotonic() - 80
        if rem > 15:
            print(f"  Fine LSE, t={min(rem, 100):.0f}s")
            a_opt, obj = lse_gradient_descent(state['best_seq'], time_limit=min(rem, 100),
                                              t_init=100.0, t_end=1000.0,
                                              lr_init=0.008, lr_end=0.0001)
            try_sequence(a_opt, "p3_lse_fine", state)

        # 3b: Spectral shaping (fine)
        rem = phase3_end - time.monotonic() - 40
        if rem > 10:
            a_opt, obj = spectral_shaping(state['best_seq'], time_limit=min(rem, 50),
                                          alpha_init=0.2, alpha_end=0.01)
            try_sequence(a_opt, "p3_spectral_fine", state)

        # 3c: Coordinate descent
        rem = phase3_end - time.monotonic() - 20
        if rem > 10:
            print(f"  Coord descent, t={min(rem, 60):.0f}s")
            a_opt, obj = coordinate_descent_polish(state['best_seq'], time_limit=min(rem, 60))
            try_sequence(a_opt, "p3_cd", state)

        # 3d: Very fine LSE
        rem = phase3_end - time.monotonic() - 10
        if rem > 10:
            a_opt, obj = lse_gradient_descent(state['best_seq'], time_limit=min(rem, 40),
                                              t_init=200.0, t_end=2000.0,
                                              lr_init=0.003, lr_end=0.00005)
            try_sequence(a_opt, "p3_lse_vfine", state)

        # 3e: Final CD squeeze
        rem = phase3_end - time.monotonic()
        if rem > 8:
            a_opt, obj = coordinate_descent_polish(state['best_seq'], time_limit=min(rem, 25))
            try_sequence(a_opt, "p3_cd_final", state)

    # Fallback
    if state['best_seq'] is None:
        state['best_seq'] = np.ones(384) / 384
        state['best_score'] = float(evaluate_sequence(state['best_seq'].tolist()))
        print(f"  Fallback uniform: {state['best_score']:.6f}")

    # Final verification
    fs = float(evaluate_sequence(state['best_seq'].tolist()))
    if np.isfinite(fs):
        state['best_score'] = fs

    print(f"\nFINAL SCORE: {state['best_score']:.6f} (n={len(state['best_seq'])})")
    return [float(x) for x in state['best_seq']]
