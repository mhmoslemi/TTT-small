def nesterov_sg_optimize(f0, time_limit, lr0=0.01, mom0=0.9, 
                         top_k=30, temp0=0.005, temp_growth=1.002,
                         decay=0.9998, restart_after=500, seed=42):
    """Nesterov-accelerated subgradient with cooling."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    rng = np.random.default_rng(seed)
    y = f.copy()  # Nesterov lookahead point
    v = np.zeros(n)  # momentum velocity
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    cur_lr = lr0
    cur_mom = mom0
    cur_temp = temp0
    it = 0
    prev_f = f.copy()
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov lookahead
        y = project_simplex(f + cur_mom * v)
        
        # Compute gradient at y
        conv_y = fast_convolve(y)
        grad = get_subgradient(y, conv_y, top_k=top_k, temp=cur_temp)
        
        # Update
        v = cur_mom * v - cur_lr * grad
        f_new = project_simplex(f + v)
        
        # Evaluate
        conv_f = fast_convolve(f_new)
        obj = conv_f.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
        
        f = f_new
        cur_lr *= decay
        cur_temp *= temp_growth
        cur_mom = min(0.999, cur_mom * 1.0001)
        
        # Periodic restart
        if it % restart_after == 0:
            # Check if we're stuck
            if best_obj - conv_f.max() > 0.001:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                v = np.zeros(n)
                cur_lr = max(lr0 * 0.5, 0.002)
            else:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                v = np.zeros(n)
                cur_lr = max(lr0 * 0.3, 0.001)
    
    return best_f, best_obj

def coordinate_descent(f0, time_limit, n_sweeps=50, seed=42):
    """Coordinate descent: for each coordinate, try to improve."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    rng = np.random.default_rng(seed)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    order = np.arange(n)
    
    t0 = time.monotonic()
    sweep = 0
    
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        rng.shuffle(order)
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                    break
            # Try a few perturbations of coordinate i
            base = f.copy()
            old_val = f[i]
            
            # Try: increase, decrease, and a few random values
            candidates = [
                old_val * 0.5,
                old_val * 0.8,
                old_val * 1.2,
                old_val * 1.5,
                old_val + 0.01 / n,
                old_val - 0.01 / n,
                0.0,
            ]
            # Filter: must keep sum positive
            candidates = [max(0.0, c) for c in candidates if (s - old_val + max(0.0, c)) > 1e-10]
            
            best_cand = old_val
            best_cand_obj = best_obj
            
            for c in candidates:
                f[i] = c
                f_new = f / f.sum()
                obj = fast_convolve(f_new).max()
                if obj < best_cand_obj:
                    best_cand_obj = obj
                    best_cand = c
            
            f = base
            f[i] = best_cand
            f = f / f.sum()
            
            if best_cand_obj < best_obj - 1e-14:
                best_obj = best_cand_obj
                best_f = f.copy()
    
    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated subgradient with adaptive temperature and learning rate
2. Multiplicative weights (mirror descent) variant
3. Coordinate descent for local refinement
4. Pair-swap heuristic for fine-tuning
5. Multi-phase strategy with diverse restarts and length exploration
"""

import numpy as np
import time


def fast_convolve(a):
    """Full auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def get_subgradient(f, conv, top_k=30, temp=0.005):
    """Softmax-weighted subgradient of max(conv)."""
    n = len(f)
    m = len(conv)
    if top_k >= m:
        peaks = np.arange(m)
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]

    pv = conv[peaks]
    max_pv = pv.max()
    scaled = (pv - max_pv) / temp
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= wsum

    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * w[i] * f[k - idx]

    return grad


def nesterov_sg(f0, time_limit, lr0=0.015, mom0=0.92, top_k=25,
                temp0=0.004, temp_growth=1.002, decay=0.99985,
                restart_after=400, seed=42):
    """Nesterov-accelerated subgradient with cooling and restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    rng = np.random.default_rng(seed)
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    cur_lr = lr0
    cur_mom = mom0
    cur_temp = temp0
    it = 0
    stale = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov lookahead
        y = project_simplex(f + cur_mom * v)

        # Gradient at lookahead point
        conv_y = fast_convolve(y)
        grad = get_subgradient(y, conv_y, top_k=top_k, temp=cur_temp)

        # Update
        v = cur_mom * v - cur_lr * grad
        f_new = project_simplex(f + v)

        conv_f = fast_convolve(f_new)
        obj = conv_f.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            stale = 0
        else:
            stale += 1

        f = f_new
        cur_lr *= decay
        cur_temp *= temp_growth
        cur_mom = min(0.9995, cur_mom * 1.00005)

        # Periodic diversified restart
        if stale > restart_after:
            scale = 0.002 + 0.002 * min(stale / restart_after, 3.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            cur_lr = max(lr0 * 0.6, 0.003)
            cur_temp = temp0 * 0.8
            stale = 0

    return best_f, best_obj


def mirror_descent(f0, time_limit, lr0=0.02, top_k=30, temp=0.005,
                   decay=0.9999, seed=42):
    """Mirror descent (multiplicative weights) on the simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    rng = np.random.default_rng(seed)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    cur_lr = lr0
    it = 0
    stale = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        grad = get_subgradient(f, conv, top_k=top_k, temp=temp)

        # Multiplicative update
        log_f = np.log(f + 1e-30) - cur_lr * grad
        log_f -= log_f.max()
        f_new = np.exp(log_f)
        f_new = f_new / f_new.sum()

        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            stale = 0
        else:
            stale += 1

        f = f_new
        cur_lr *= decay

        if stale > 600:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            stale = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit, seed=42, n_candidates=8):
    """Randomized coordinate descent with multiple candidates per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    rng = np.random.default_rng(seed)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    order = np.arange(n)

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        rng.shuffle(order)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[i]
            s_other = s - old_val

            if old_val < 1e-15:
                candidates = [s / n, s / (2 * n), 2 * s / n]
            else:
                candidates = [
                    old_val * 0.5, old_val * 0.75, old_val * 1.25, old_val * 1.5,
                    old_val + 0.02 / n, old_val - 0.02 / n,
                    s / n, 0.0
                ]
                # Add some random candidates
                candidates.append(old_val * (1.0 + rng.standard_normal() * 0.5))

            best_cand = old_val
            best_cand_obj = np.inf

            for c in candidates:
                c = max(0.0, c)
                if s_other + c < 1e-10:
                    continue
                f[i] = c
                f_new = f / f.sum()
                obj = fast_convolve(f_new).max()
                if obj < best_cand_obj:
                    best_cand_obj = obj
                    best_cand = c
                f[i] = old_val

            if best_cand != old_val:
                f[i] = best_cand
                f = f / f.sum()
                if best_cand_obj < best_obj - 1e-14:
                    best_obj = best_cand_obj
                    best_f = f.copy()

    return best_f, best_obj


def pair_swap(f0, time_limit, seed=42):
    """Try moving small amounts of mass between pairs of positions."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    rng = np.random.default_rng(seed)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    delta_scales = [0.01, 0.005, 0.002, 0.001]
    di = 0

    while time.monotonic() - t0 < time_limit:
        for _ in range(500):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            delta = delta_scales[di % len(delta_scales)] * s * rng.uniform(0.3, 1.0)

            # Try moving delta from i to j
            if f[i] >= delta:
                f[i] -= delta
                f[j] += delta
                f_new = f / f.sum()
                obj = fast_convolve(f_new).max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_f = f_new.copy()
                else:
                    f[i] += delta
                    f[j] -= delta

            # Try moving delta from j to i
            if f[j] >= delta:
                f[j] -= delta
                f[i] += delta
                f_new = f / f.sum()
                obj = fast_convolve(f_new).max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_f = f_new.copy()
                else:
                    f[j] += delta
                    f[i] -= delta

        di += 1
        f = best_f.copy()

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    inits.append(rng.dirichlet(np.ones(n) * 1.0))

    for sf in [0.1, 0.2, 0.35]:
        sigma = sf * n
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / sigma) ** 2)
        inits.append(g / g.sum())

    for alpha in [0.5, 1.0, 2.0, 4.0]:
        u = (x + 1) ** (-alpha) * (n - x) ** (-alpha)
        inits.append(u / u.sum())

    for p in [2.0, 4.0, 8.0]:
        c = np.abs(np.cos(np.pi * x / n)) ** p + 0.05
        inits.append(c / c.sum())

    for frac in [0.15, 0.25, 0.35]:

        g2 = np.exp(-0.5 * ((x - (1 - frac) * n) / (0.08 * n)) ** 2)

    for frac in [0.1, 0.2]:

    # === Phase 5: Pair swap ===
    print("\n=== Phase 5: Pair Swap ===")
    remaining = deadline - time.monotonic() - 150
    t_p5 = min(remaining * 0.3, 200)

    if t_p5 > 15:
        a_opt, _ = pair_swap(best_seq, time_limit=t_p5, seed=333)
        try_seq(a_opt, "swap1")

        rem = deadline - time.monotonic() - 100
        if rem > 15:
            a_opt, _ = pair_swap(best_seq, time_limit=min(rem, 80), seed=444)
            try_seq(a_opt, "swap2")

    print(f"  Phase 5 best: {best_score:.8f}")

    # === Phase 6: Alternating refinement ===
    print("\n=== Phase 6: Alternating Refinement ===")
    remaining = deadline - time.monotonic() - 80
    t_p6 = min(remaining * 0.6, 300)

    if t_p6 > 30:
        n_rounds = 5
        t_each = t_p6 / n_rounds
        for r in range(n_rounds):
            if time.monotonic() > deadline - 60:
                break
            # Nesterov with tightening parameters
            t_sg = t_each * 0.5
            a_opt, _ = nesterov_sg(best_seq, time_limit=t_sg,
                                   lr0=0.005 * (1.0 - r * 0.1),
                                   mom0=0.96 + r * 0.005,
                                   top_k=20 + r * 10,
                                   temp0=0.002 * (1.0 - r * 0.1),
                                   temp_growth=1.001,
                                   restart_after=400,
                                   seed=500 + r)
            try_seq(a_opt, f"alt_sg{r}")

            # Coordinate descent
            t_cd = t_each * 0.3
            rem_cd = deadline - time.monotonic() - 50 - (n_rounds - r - 1) * t_each
            t_cd = min(t_cd, max(5, rem_cd * 0.3))
            if t_cd > 5:
                a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd, seed=600 + r)
                try_seq(a_opt, f"alt_cd{r}")

            # Pair swap
            rem_sw = deadline - time.monotonic() - 50 - (n_rounds - r - 1) * t_each
            t_sw = min(t_each * 0.2, max(5, rem_sw * 0.2))
            if t_sw > 5:
                a_opt, _ = pair_swap(best_seq, time_limit=t_sw, seed=700 + r)
                try_seq(a_opt, f"alt_sw{r}")

    print(f"  Phase 6 best: {best_score:.8f}")

    # === Phase 7: Length exploration ===
    print("\n=== Phase 7: Length Exploration ===")
    remaining = deadline - time.monotonic() - 50
    if remaining > 30 and best_seq is not None:
        for n_try in [n_best - 16, n_best + 16, n_best + 32, n_best - 32,
                      int(n_best * 1.25), int(n_best * 0.8)]:
            if n_try < 64 or n_try > 1500 or time.monotonic() > deadline - 40:
                continue
            px = np.interp(np.linspace(0, 1, n_try),
                          np.linspace(0, 1, n_best), best_seq)
            px = np.maximum(px, 0)
            ps = px.sum()
            if ps > 1e-10:
                px = px / ps
                t_opt = min(25, remaining * 0.3)
                rem = deadline - time.monotonic() - 35
                if rem < 10:
                    break
                a_opt, _ = nesterov_sg(px, time_limit=min(t_opt, rem * 0.6),
                                       lr0=0.01, mom0=0.93, top_k=20,
                                       temp0=0.003, temp_growth=1.002,
                                       restart_after=300, seed=n_try)
                try_seq(a_opt, f"len{n_try}")

    print(f"  Phase 7 best: {best_score:.8f}")

    # === Phase 8: Final fine polish ===
    print("\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic() - 30
    if remaining > 15 and best_seq is not None:
        a_opt, _ = nesterov_sg(best_seq, time_limit=min(remaining * 0.6, 80),
                               lr0=0.002, mom0=0.985, top_k=60,
                               temp0=0.001, temp_growth=1.0005,
                               restart_after=500, seed=999)
        try_seq(a_opt, "final_sg")

        rem = deadline - time.monotonic() - 25
        if rem > 10:
            a_opt, _ = pair_swap(best_seq, time_limit=min(rem, 30), seed=888)
            try_seq(a_opt, "final_swap")

    # Final check
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f}")

    return [float(x) for x in best_seq]
