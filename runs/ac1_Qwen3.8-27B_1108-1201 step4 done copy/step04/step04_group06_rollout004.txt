def compute_gradient_at_peak(a, conv, k_peak):
    """Compute gradient of (a*a)[k_peak] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):

            if i == j:
                grad[i] += a[i]
            else:
                grad[i] += 2 * a[j]

    # grad[i] = a[i] for i where 2*i == k_peak
    return grad

def mirror_descent_step(a, conv, eta):
    """One mirror descent step for minimizing max_k (a*a)[k]."""
    n = len(a)
    k_peak = int(np.argmax(conv))
    grad = compute_gradient_at_peak(a, conv, k_peak)

    log_a = np.where(a > 1e-300, np.log(a + 1e-300), -30)
    log_a_new = log_a - eta * grad
    # Normalize to max 0 for numerical stability
    log_a_new -= log_a_new.max()
    a_new = np.exp(log_a_new)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def flat_spectrum_init(n, rng, iters=200):
    """Generate a sequence with approximately flat power spectrum."""
    a = rng.dirichlet(np.ones(n) * 0.1)
    target = np.ones(n) / n  # flat target spectrum
    
    for _ in range(iters):
        # Forward transform
        A = np.fft.rfft(a)
        # Modify amplitude to be flat (keep phase)
        amplitude = np.abs(A)
        phase = np.angle(A)
        target_amp = np.sqrt(np.abs(target))  # We want |A|^2 ~ flat
        # Actually, we want |A(k)|^2 to be approximately constant
        # |A(k)|^2 = sum_k (a*a)[k] ... no, that's not right.
        # By Parseval: sum_k |A(k)|^2 = N * sum_i a[i]^2
        # The circular autocorrelation is ifft(|A|^2)
        # We want the circular autocorrelation to be flat (delta at 0)
        # which means |A(k)|^2 should be constant.
        
        # Set |A(k)| = constant (sqrt of target)
        const_amp = np.sqrt(np.sum(np.abs(A)**2) / len(A))
        A_new = const_amp * np.exp(1j * phase)
        # Make sure A_new[0] is real and positive (DC component)
        A_new[0] = np.abs(A_new[0])

        else:

        a = a_new

def lp_solve(a, M, C=None):
    """Solve LP: max sum(g) s.t. C@g <= M, g >= 0."""
    n = len(a)

                         options={'presolve': True})
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g, g_sum, True
    except Exception:
        pass
    return a.copy(), 1.0, False

def lp_step_with_linsearch(a, conv, n, max_evals=20):
    """LP step with backtracking line search."""
    M = conv.max()
    C = build_conv_matrix(a)
    g, g_sum, success = lp_solve(a, M * 1.0001, C)

    if not success or g_sum <= 1.0 + 1e-8:
        return a.copy(), False

    d = g - a
    norm_d = np.linalg.norm(d)
    if norm_d < 1e-12:
        return a.copy(), False

    # Backtracking line search
    best_a = a.copy()
    best_obj = 2.0 * n * M
    t = 1.0
    best_t = 0.0

            best_a = a_try.copy()
            best_t = t
            break
        else:
            t *= 0.5

    return a.copy(), False

def gradient_step(a, conv, n, lr):
    """Projected gradient step with subgradient of max conv."""

    top_k = max(10, n // 10)

    temp = max(peak_vals.max() - peak_vals.min(), 1e-10) * 0.1

    scaled -= scaled.max()
    w = np.exp(scaled)

            continue
        k = int(peaks[idx])
        for i in range(max(0, k - n + 1), min(n - 1, k) + 1):
            j = k - i
            if i == j:
                grad[i] += w[idx] * a[i]
            else:
                grad[i] += w[idx] * 2 * a[j]

    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy(), False
    a_new = a_new / s
    return a_new, True

        t = max(temp * peak_vals.max(), 1e-10)
        scaled = peak_vals / t
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum

    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            if i == j:
                grad[i] += w[idx] * a[i]
            else:
                grad[i] += w[idx] * 2.0 * a[j]
    return grad


def mirror_descent_step(a, conv, eta, top_k=10, temp=0.01):
    """One mirror descent step for minimizing max_k (a*a)[k]."""
    n = len(a)
    grad = compute_subgradient(a, conv, top_k=top_k, temp=temp)
    
    # Multiplicative update: a_new[i] ∝ a[i] * exp(-eta * grad[i])
    safe_a = np.maximum(a, 1e-30)
    log_ratio = -eta * grad
    # Clip for numerical stability
    log_ratio = np.clip(log_ratio, -50, 50)
    a_new = safe_a * np.exp(log_ratio)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def flat_spectrum_init(n, rng, iters=300):
    """Generate sequence with approximately flat power spectrum via error reduction."""
    a = rng.dirichlet(np.ones(n) * 0.05)
    
    for it in range(iters):
        A = np.fft.rfft(a, n)
        amp = np.abs(A)
        phase = np.angle(A)
        
        # Target: flat amplitude spectrum
        mean_amp = amp.mean()
        target_amp = np.full_like(amp, mean_amp)
        target_amp[0] = amp[0]  # Keep DC
        
        A_new = target_amp * np.exp(1j * phase)
        A_new[0] = np.abs(A_new[0])
        
        a_new = np.fft.irfft(A_new, n)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        a = a_new
    
    return a


def lp_solve(a, M, C=None):
    """Solve LP: max sum(g) s.t. C@g <= M, g >= 0."""
    n = len(a)
    if C is None:
        C = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 10})
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g, g_sum, True
    except Exception:
        pass
    return a.copy(), 1.0, False


def lp_step_with_linsearch(a, conv, n, max_backtrack=15):
    """LP step with backtracking line search."""
    M = conv.max()
    C = build_conv_matrix(a)
    g, g_sum, success = lp_solve(a, M * 1.00001, C)

    if not success or g_sum <= 1.0 + 1e-8:
        return None

    d = g - a
    if np.linalg.norm(d) < 1e-12:
        return None

    # Backtracking line search
    best_obj = 2.0 * n * M
    t = 0.5
    for _ in range(max_backtrack):
        a_try = a + t * d
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            t *= 0.5
            continue
        a_try = a_try / s
        conv_try = fast_convolve(a_try)
        obj_try = 2.0 * n * conv_try.max()
        if obj_try < best_obj - 1e-10:
            return a_try
        t *= 0.5
    return None


def gradient_step_with_linsearch(a, conv, n, lr_init=0.01, max_backtrack=8):
    """Projected subgradient step with line search."""
    top_k = max(10, n // 20)
    grad = compute_subgradient(a, conv, top_k=top_k, temp=0.005)
    
    best_obj = 2.0 * n * conv.max()
    lr = lr_init
    for _ in range(max_backtrack):
        a_try = a - lr * grad
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 1e-15:
            lr *= 0.5
            continue
        a_try = a_try / s
        conv_try = fast_convolve(a_try)
        obj_try = 2.0 * n * conv_try.max()
        if obj_try < best_obj - 1e-10:
            return a_try
        lr *= 0.5
    return None


def optimize_hybrid(a0, time_limit, verbose=False, seed=0):
    """
    Hybrid optimization loop: LP + mirror descent + gradient, with adaptive switching.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    mode = 'lp'  # 'lp', 'md', 'grad'
    mode_fail = 0
    perturb_count = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 0.5:
            break

        improved = False

        if mode == 'lp':
            result = lp_step_with_linsearch(a, conv, n)
            if result is not None:
                a = result
                conv = fast_convolve(a)
                cur_obj = 2.0 * n * conv.max()
                if cur_obj < best_obj - 1e-12:
                    best_obj = cur_obj
                    best_a = a.copy()
                improved = True
            else:
                mode_fail += 1
                if mode_fail >= 3:
                    mode = 'md' if mode_fail % 2 == 1 else 'grad'
                    mode_fail = 0

        elif mode == 'md':
            M = conv.max()
            # Adaptive eta based on current scale
            eta_base = 0.05 * M / max(a.max(), 1e-10)
            eta = eta_base / (1.0 + 0.1 * iteration)
            for top_k in [5, 10, 20]:
                a_try = mirror_descent_step(a, conv, eta, top_k=top_k, temp=0.01)
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-12:
                    a = a_try
                    conv = conv_try
                    best_obj = obj_try
                    best_a = a.copy()
                    improved = True
                    break
            if not improved:
                mode_fail += 1
                if mode_fail >= 5:
                    mode = 'lp'
                    mode_fail = 0

        else:  # gradient
            result = gradient_step_with_linsearch(a, conv, n, lr_init=0.02 / (1.0 + 0.01 * iteration))
            if result is not None:
                a = result
                conv = fast_convolve(a)
                cur_obj = 2.0 * n * conv.max()
                if cur_obj < best_obj - 1e-12:
                    best_obj = cur_obj
                    best_a = a.copy()
                improved = True
            else:
                mode_fail += 1
                if mode_fail >= 5:
                    mode = 'lp'
                    mode_fail = 0

        if improved:
            no_improve = 0
        else:
            no_improve += 1

        iteration += 1

        if verbose and iteration % 100 == 0:
            print(f"      iter={iteration}, obj={best_obj:.8f}, t={elapsed:.1f}s, mode={mode}")

        # Stagnation: perturb and restart
        stagnation_threshold = 40 + 10 * perturb_count
        if no_improve >= stagnation_threshold:
            scale = 0.003 + 0.003 * min(perturb_count, 10)
            noise = rng.standard_normal(n) * scale
            a = best_a + noise
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            else:
                a = best_a.copy()
            conv = fast_convolve(a)
            no_improve = 0
            mode = 'lp'
            mode_fail = 0
            perturb_count += 1

            if perturb_count > 30:
                # Try a more aggressive restart
                a2 = rng.dirichlet(np.ones(n) * 0.1)
                conv2 = fast_convolve(a2)
                obj2 = 2.0 * n * conv2.max()
                if obj2 < best_obj:
                    a = a2
                    conv = conv2
                    best_obj = obj2
                    best_a = a.copy()
                else:
                    break

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random1'] = rng.dirichlet(np.ones(n) * 0.1)
    inits['random2'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['random3'] = rng.dirichlet(np.ones(n) * 1.0)

    # Sine-based
    s1 = np.sin(np.pi * x / n) ** 2 + 0.01
    inits['sine1'] = s1 / s1.sum()
    s2 = np.sin(2 * np.pi * x / n) ** 2 + 0.01
    inits['sine2'] = s2 / s2.sum()
    s3 = np.cos(np.pi * x / n) ** 2 + 0.01
    inits['sine3'] = s3 / s3.sum()

    # Bimodal
    edge = max(1, n // 6)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi += 0.1 * rng.random(n)
    inits['bimodal'] = bi / bi.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.15 * n)) ** 2)
    inits['gaussian'] = g / g.sum()

    # Alternating
    alt = np.ones(n)
    alt[::2] = 1.5
    alt[1::2] = 0.5
    inits['alternating'] = alt / alt.sum()

    # Flat spectrum
    inits['flat_spec'] = flat_spectrum_init(n, rng, iters=200)

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps

    return inits


def solve_qcqp_small(n, time_limit=30):
    """Try to solve QCQP for small n using CVXPY as a warm-start template."""
    try:
        import cvxpy as cp
    except ImportError:
        return None

    t_start = time.monotonic()
    
    a_var = cp.Variable(n)
    t_var = cp.Variable()

    constraints = [a_var >= 0, cp.sum(a_var) == 1]

    # Build quadratic constraints using sparse matrices
    from scipy.sparse import lil_matrix

    quad_terms = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        rows = []
        cols = []
        for i in range(i_start, i_end + 1):
            j = k - i
            rows.append(i)
            cols.append(j)
        Q = lil_matrix((n, n))
        for i, j in zip(rows, cols):
            Q[i, j] = 1.0
        Q = Q.tocsr()
        constraints.append(cp.quad_form(a_var, Q) <= t_var)

    prob = cp.Problem(cp.Minimize(t_var), constraints)

    # Try solving with a non-convex solver
    try:
        # Try GUROBI first
        prob.solve(solver=cp.GUROBI, non_convex=2, time_limit=time_limit)
    except Exception:
        try:
            prob.solve(solver=cp.MOSEK, time_limit=time_limit)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iters=5000)
            except Exception:
                return None

    if time.monotonic() - t_start > time_limit + 5:
        return None

    if a_var.value is None:
        return None

    a_sol = np.asarray(a_var.value, dtype=np.float64)
    a_sol = np.maximum(a_sol, 0)
    s = a_sol.sum()
    if s < 1e-10:
        return None
    a_sol = a_sol / s
    return a_sol


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Hybrid LP + mirror descent + gradient search with
    QCQP warm start and multi-scale initialization.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass

    def update_best(a, label=""):
        nonlocal best_sequence, best_score
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  {label}: score={score:.6f} *NEW BEST* (n={len(a)})")
        elif np.isfinite(score):
            print(f"  {label}: score={score:.6f}")
        return score

    # === Phase 0: QCQP warm start for small n ===
    print("=== Phase 0: QCQP Warm Start ===")
    t0 = time.monotonic()
    qcqp_template = None
    for n_q in [64, 128]:
        if time.monotonic() > deadline - 1500:
            break
        qcqp_template = solve_qcqp_small(n_q, time_limit=20)
        if qcqp_template is not None:
            print(f"  QCQP n={n_q}: solved in {time.monotonic()-t0:.1f}s")
            break
    if qcqp_template is None:
        print(f"  QCQP not available or timed out ({time.monotonic()-t0:.1f}s)")
    print(f"Phase 0 done ({time.monotonic()-t0:.1f}s)")

    # === Phase 1: Multi-initialization gradient scan ===
    print("=== Phase 1: Initialization Scan ===")
    t0 = time.monotonic()
    phase1_budget = 250.0

    target_lengths = [768]
    if parent is not None:
        pn = len(parent)
        if pn not in target_lengths:
            target_lengths.insert(0, pn)

    scan_results = []

    for n in target_lengths:
        if time.monotonic() > deadline - 1000:
            break
        inits = generate_initializations(n, parent, rng)

        # Add QCQP template if available
        if qcqp_template is not None and len(qcqp_template) != n:
            q_ext = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(qcqp_template)), qcqp_template)
            q_ext = np.maximum(q_ext, 0)
            q_ext /= q_ext.sum()
            inits['qcqp'] = q_ext

        # Priority ordering
        priority = ['parent', 'parent_interp', 'flat_spec', 'qcqp', 'uniform',
                    'sine1', 'random1', 'bimodal', 'alternating', 'gaussian']
        init_names = [nm for nm in priority if nm in inits]
        init_names += [nm for nm in inits if nm not in priority]
        init_names = init_names[:5]

        t_per_init = max(15.0, (phase1_budget - 10) / (len(target_lengths) * len(init_names)))

        for name in init_names:
            if time.monotonic() > deadline - 900:
                break
            t_alloc = min(t_per_init, deadline - time.monotonic() - 800)
            if t_alloc < 8:
                break

            a0 = inits[name]
            a_opt, obj = optimize_hybrid(a0, time_limit=t_alloc, seed=hash(name) % 10000)
            score = update_best(a_opt, f"Scan n={n}/{name}")
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t0:.0f}s). Best so far: {best_score:.6f}")

    # === Phase 2: Deep LP optimization on top candidates ===
    print("=== Phase 2: Deep Optimization ===")
    t0 = time.monotonic()

    # Select top candidates (unique by length, prefer best score)
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n,)
        if key not in seen:
            top_configs.append((n, a_opt))
            seen.add(key)
        if len(top_configs) >= 3:
            break

    # Always include overall best
    if best_sequence is not None:
        bn = len(best_sequence)
        if bn not in [c[0] for c in top_configs]:
            top_configs.append((bn, best_sequence.copy()))

    remaining = max(60.0, deadline - time.monotonic() - 400.0)
    n_cand = max(1, len(top_configs))
    t_per_deep = remaining / n_cand

    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 350:
            break
        t_alloc = min(t_per_deep, deadline - time.monotonic() - 300)
        if t_alloc < 20:
            break

        print(f"  Deep opt {i}: n={n}, time={t_alloc:.0f}s")
        a_opt, obj = optimize_hybrid(a0, time_limit=t_alloc, verbose=True, seed=100 + i)
        score = update_best(a_opt, f"Deep {i} (n={n})")

    print(f"Phase 2 done ({time.monotonic()-t0:.0f}s). Best: {best_score:.6f}")

    # === Phase 3: Aggressive polish on best ===
    print("=== Phase 3: Polish ===")
    t0 = time.monotonic()

    if best_sequence is not None:
        n_best = len(best_sequence)
        remaining = max(30.0, deadline - time.monotonic() - 150.0)

        # Multiple polish runs with different perturbation levels
        for pi in range(4):
            if time.monotonic() > deadline - 120:
                break
            t_alloc = remaining / (4 - pi) * 0.8
            t_alloc = min(t_alloc, deadline - time.monotonic() - 100)
            if t_alloc < 10:
                break

            if pi == 0:
                a0 = best_sequence.copy()
            else:
                scale = 0.002 * pi + 0.001
                noise = rng.standard_normal(n_best) * scale
                a0 = np.maximum(best_sequence + noise, 0)
                s = a0.sum()
                if s > 1e-15:
                    a0 = a0 / s
                else:
                    a0 = best_sequence.copy()

            a_opt, obj = optimize_hybrid(a0, time_limit=t_alloc, seed=200 + pi)
            score = update_best(a_opt, f"Polish {pi}")

    print(f"Phase 3 done ({time.monotonic()-t0:.0f}s). Best: {best_score:.6f}")

    # === Phase 4: Final fine-tuning ===
    print("=== Phase 4: Fine-tuning ===")
    t0 = time.monotonic()

    if best_sequence is not None and time.monotonic() < deadline - 20:
        n_best = len(best_sequence)
        remaining = max(10.0, deadline - time.monotonic() - 15.0)

        # Short gradient runs with different hyperparameters
        configs = [
            {'lr_mult': 0.5},
            {'lr_mult': 2.0},
            {'lr_mult': 1.0},
        ]
        t_per = remaining / 3
        for gi in range(3):
            if time.monotonic() > deadline - 10:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 8)
            if t_alloc < 5:
                break
            a_opt, obj = optimize_hybrid(best_sequence, time_limit=t_alloc,
                                         seed=300 + gi * 7)
            score = update_best(a_opt, f"FineTune {gi}")

    print(f"Phase 4 done ({time.monotonic()-t0:.0f}s). Best: {best_score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_sequence.tolist()))
        print(f"  Fallback uniform: {best_score:.6f}")

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
