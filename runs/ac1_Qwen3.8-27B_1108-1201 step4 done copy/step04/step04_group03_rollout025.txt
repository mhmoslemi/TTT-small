def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax(x):
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()

"""
Multi-strategy optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses continuation-based smoothed L-BFGS, pairwise coordinate descent with
simulated annealing, multi-scale spectral initializations, and structured
restarts. Key innovations: smooth approximation with adaptive temperature
scheduling, second-order curvature exploitation via L-BFGS, and complementary
pairwise exchange moves that escape gradient-based local minima.
"""
import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_objective_normalized(a):
    """Compute 2n*max(conv) for normalized a (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()


def smooth_obj_and_grad_direct(a, eps):
    """
    Compute smoothed objective and gradient (w.r.t. a, not projected).
    f_eps(a) = conv.max() + eps * log(sum(exp((conv - conv.max)/eps)))
    grad = sum_k w[k] * d(conv[k])/d(a)
    """
    n = len(a)
    conv = fast_convolve(a)
    conv_max = conv.max()
    shifted = (conv - conv_max) / eps
    exp_shifted = np.exp(shifted - shifted.max())
    log_sum = shifted.max() + np.log(exp_shifted.sum())
    f_val = conv_max + eps * log_sum

    w = exp_shifted / exp_shifted.sum()

    # Gradient: d f / d a[j] = sum_k w[k] * (a[k-j] + a[j-k]) with bounds
    # T1[j] = sum_m a[m] * w[j+m] (cross-correlation)
    # T2[j] = sum_m a[m] * w[j-m] (convolution)
    L = 1 << (3 * n - 2).bit_length()
    A_fft = np.fft.rfft(a, L)
    W_fft = np.fft.rfft(w, L)
    T1 = np.fft.irfft(np.conj(A_fft) * W_fft, L)[:n]
    T2 = np.fft.irfft(A_fft * W_fft, L)[:n]
    grad = T1 + T2

    return f_val, grad


def continuation_lbfgs(a0, eps_schedule, time_limit, max_iter_per_phase=200):
    """
    Continuation method with L-BFGS on the simplex.
    Uses projected gradient with line search at each eps level.
    """
    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    best_obj = compute_objective_normalized(a)
    t0 = time.monotonic()

    for phase_idx, eps in enumerate(eps_schedule):
        if time.monotonic() - t0 >= time_limit:
            break
        phase_time = time_limit / len(eps_schedule) * 1.5
        phase_start = time.monotonic()

        # At each eps level, do projected gradient with backtracking
        k = 0
        momentum = np.zeros(n)
        lr = eps * 0.1
        no_improve = 0

        while time.monotonic() - phase_start < phase_time and k < max_iter_per_phase * 10:
            if time.monotonic() - t0 >= time_limit:
                break
            k += 1

            f_val, grad = smooth_obj_and_grad_direct(a, eps)

            # Project gradient to tangent space of simplex
            grad_mean = grad.mean()
            grad_tangent = grad - grad_mean

            # Backtracking line search
            alpha = lr
            y = a - alpha * grad_tangent
            a_new = project_simplex(y)
            obj_new = compute_objective_normalized(a_new)

            # FISTA-like acceleration
            if obj_new < f_val * 0.999 or obj_new < best_obj:
                momentum = 0.9 * momentum - alpha * grad_tangent
                a = project_simplex(a + momentum)
                momentum = momentum * 0.5  # damp after projection
                obj = compute_objective_normalized(a)
            else:
                a = a_new
                obj = obj_new
                momentum *= 0.8

            # Track best
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            # Decay learning rate
            lr *= 0.999
            if no_improve > 50:
                lr = max(lr * 0.5, eps * 0.01)
                no_improve = 0

    return best_a, best_obj


def pairwise_cd(a0, time_limit, n_iter_max=500000, anneal_start=0.8, anneal_rate=0.9999):

    """

    n = len(a)
    best_a = a.copy()
    best_obj = compute_objective_normalized(a)
    rng = np.random.default_rng(12345)

    it = 0
    temperature = 0.001
    accept_count = 0
    total_count = 0

    precompute_fft_size = 1 << (2 * n - 1).bit_length()

    while time.monotonic() - t0 < time_limit and it < n_iter_max:
        it += 1

        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue

        s = a[i] + a[j]
        if s < 1e-15:
            continue

        # Current fraction
        t_cur = a[i] / s

        # Evaluate objective at several t values
        # Try: current, and several perturbations
        candidates_t = [t_cur]
        # Golden section around current
        for delta in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1]:
            candidates_t.append(max(0.0, min(1.0, t_cur + delta)))
            candidates_t.append(max(0.0, min(1.0, t_cur - delta)))

        best_t = t_cur
        best_local_obj = best_obj

        # We need to compute the objective efficiently
        # The change in conv when transferring from j to i by delta:
        # delta_a[i] = s*(t - t_cur), delta_a[j] = s*(t_cur - t)
        # Let d = s*(t - t_cur)
        # Δconv[k] ≈ 2d*(a[k-i] - a[k-j]) (first order)
        # For second order: Δconv[k] = 2d*(a[k-i] - a[k-j]) + d^2*(1 if k==i+j or k==j+i adjusted)

        # For efficiency, just evaluate at a few points
        eval_count = 0
        for t_cand in candidates_t:
            if t_cand == t_cur or eval_count > 8:
                continue
            eval_count += 1
            a_test = a.copy()
            a_test[i] = s * t_cand
            a_test[j] = s * (1.0 - t_cand)
            obj_test = compute_objective_normalized(a_test)
            if obj_test < best_local_obj:
                best_local_obj = obj_test
                best_t = t_cand

        if best_t != t_cur:
            a[i] = s * best_t
            a[j] = s * (1.0 - best_t)
            obj = best_local_obj
            total_count += 1
            if obj < best_obj:
                accept_count += 1
            # Annealing acceptance
            if obj > best_obj:
                if temperature > 1e-10 and rng.random() < np.exp(-(obj - best_obj) / temperature):
                    pass  # keep the change
                else:
                    a[i] = s * t_cur
                    a[j] = s * (1.0 - t_cur)
                    obj = best_obj

        # Update best
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        # Anneal
        temperature *= anneal_rate
        if it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Pairwise CD: it={it}, obj={best_obj:.8f}, temp={temperature:.6f}, "
                  f"accept_rate={accept_count/max(1,total_count):.2f}, time={elapsed:.0f}s")

    return best_a, best_obj


def spectral_init(n, rng, mode=0):
    """Generate structured initializations using spectral methods."""
    x = np.arange(n, dtype=np.float64)

    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        # Chirp-like
        a = np.abs(np.cos(np.pi * x * x / (4.0 * n))) + 0.1
        return a / a.sum()
    elif mode == 2:
        # Multi-freq modulation
        a = np.ones(n)
        for l in range(1, 4):
            a += 0.3 * np.cos(2.0 * np.pi * l * x / n + rng.uniform(0, 2 * np.pi))
        a = np.abs(a) + 0.05
        return a / a.sum()
    elif mode == 3:
        # Edge-weighted
        w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
        return w / w.sum()
    elif mode == 4:
        # Gaussian
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
        return g / g.sum()
    elif mode == 5:
        # Random smooth
        freqs = rng.integers(1, 10, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        amps = rng.uniform(0.1, 0.5, size=5)
        a = np.ones(n)
        for l in range(5):
            a += amps[l] * np.cos(2.0 * np.pi * freqs[l] * x / n + phases[l])
        a = np.abs(a) + 0.05
        return a / a.sum()
    elif mode == 6:
        # Discrete chirp (real part)
        a = np.cos(np.pi * x * x / n) ** 2 + 0.1
        return a / a.sum()
    else:
        return rng.dirichlet(np.ones(n) * 0.5)

def structured_restart(a0, rng, n_structures=8):
    """Generate structured perturbations of a good solution."""
    n = len(a0)
    restarts = []

    # Small random perturbations
    for scale in [0.001, 0.003, 0.008]:

        restarts.append(project_simplex(perturbed))

    # Phase rotation in frequency domain
    for k_rot in [1, 2, 3]:
        A = np.fft.rfft(a0)
        phase = 2.0 * np.pi * k_rot / n
        A_rot = A * np.exp(1j * phase * np.arange(len(A)))
        a_rot = np.fft.irfft(A_rot, n)
        a_rot = np.abs(a_rot) + 1e-6
        restarts.append(a_rot / a_rot.sum())

    # Scale inversion (complement)
    a_comp = 1.0 / (a0 * n + 1e-6)
    restarts.append(a_comp / a_comp.sum())

    # Symmetric flip
    a_flip = a0[::-1].copy()
    restarts.append(a_flip / a_flip.sum())

    return restarts

        s = try_seq(a_init, f"init_{label}")
        scored_candidates.append((s, label, a_init))

    scored_candidates.sort(key=lambda x: x[0])
    print(f"  Top candidates: {[(l, f'{s:.6f}') for s, l, _ in scored_candidates[:5]]}")

    # Quick continuation on top 3 initializations
    t_remaining = deadline - time.monotonic() - 1400
    top_k = min(3, len(scored_candidates))
    eps_quick = [0.01, 0.003, 0.001]
    t_per = max(10, t_remaining / top_k)

    for ci in range(top_k):
        t_rem = deadline - time.monotonic() - 1300
        if t_rem < 10:
            break
        label, a_init = scored_candidates[ci][1], scored_candidates[ci][2]
        print(f"  Quick continuation on {label}, t={min(t_per, t_rem):.0f}s")
        a_opt, _ = continuation_lbfgs(a_init, eps_quick, time_limit=min(t_per, t_rem))
        try_seq(a_opt, f"quick_{label}")

    print(f"  Phase 1 best: {best_score:.8f}")

    # === Phase 2: Deep continuation L-BFGS (~600s) ===
    print("\n=== Phase 2: Deep Continuation ===")
    if best_seq is not None:
        t_remaining = deadline - time.monotonic() - 600
        p2_time = min(600, max(60, t_remaining))
        eps_deep = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00003]
        print(f"  Deep continuation, t={p2_time:.0f}s, eps schedule: {eps_deep}")
        a_opt, _ = continuation_lbfgs(best_seq, eps_deep, time_limit=p2_time)
        try_seq(a_opt, "deep_continuation")

        # Try from a perturbed start
        t_rem = deadline - time.monotonic() - 500
        if t_rem > 30 and best_seq is not None:
            noise = rng.standard_normal(len(best_seq)) * 0.01
            a_pert = project_simplex(best_seq + noise)
            eps_pert = [0.002, 0.001, 0.0003, 0.0001]
            a_opt2, _ = continuation_lbfgs(a_pert, eps_pert, time_limit=min(t_rem, 120))
            try_seq(a_opt2, "deep_perturbed")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Pairwise Coordinate Descent (~400s) ===
    print("\n=== Phase 3: Pairwise CD ===")
    if best_seq is not None:
        t_remaining = deadline - time.monotonic() - 300
        p3_time = min(400, max(30, t_remaining))
        print(f"  Pairwise CD, t={p3_time:.0f}s")
        a_opt, _ = pairwise_cd(best_seq, time_limit=p3_time)
        try_seq(a_opt, "pairwise_cd")

        # Another round with different seed
        t_rem = deadline - time.monotonic() - 250
        if t_rem > 30 and best_seq is not None:
            a_opt2, _ = pairwise_cd(best_seq, time_limit=min(t_rem, 150))
            try_seq(a_opt2, "pairwise_cd2")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: Fine Polish (~200s) ===
    print("\n=== Phase 4: Fine Polish ===")
    if best_seq is not None:
        t_remaining = deadline - time.monotonic() - 100
        p4_time = min(200, max(20, t_remaining))
        print(f"  Fine polish, t={p4_time:.0f}s")
        a_opt, _ = fine_polish(best_seq, time_limit=p4_time)
        try_seq(a_opt, "fine_polish")

        # Structured restarts
        t_rem = deadline - time.monotonic() - 60
        if t_rem > 20 and best_seq is not None:
            restarts = structured_restart(best_seq, rng)
            for ri, a_r in enumerate(restarts[:4]):
                t_r = deadline - time.monotonic() - 30
                if t_r < 10:
                    break
                eps_fine = [0.0003, 0.0001, 0.00005]
                a_opt_r, _ = continuation_lbfgs(a_r, eps_fine, time_limit=min(t_r, 40))
                try_seq(a_opt_r, f"restart_{ri}")
                t_rem = deadline - time.monotonic() - 30

    print(f"  Phase 4 best: {best_score:.8f}")

    # === Phase 5: Multi-length exploration (~100s) ===
    print("\n=== Phase 5: Multi-length ===")
    t_remaining = deadline - time.monotonic() - 30
    if t_remaining > 60:
        multi_length_search(t_remaining, best_seq, best_score, rng)

    # === Phase 6: Final ultra-fine polish (~30s) ===
    print("\n=== Phase 6: Final Polish ===")
    if best_seq is not None:
        t_rem = deadline - time.monotonic() - 5
        if t_rem > 10:
            # Very fine pairwise
            a_opt, _ = pairwise_cd(best_seq, time_limit=min(t_rem, 20),
                                   n_iter_max=200000, anneal_rate=0.99995)
            try_seq(a_opt, "final_pairwise")

            t_rem2 = deadline - time.monotonic() - 2
            if t_rem2 > 5:
                a_opt2, _ = fine_polish(best_seq, time_limit=t_rem2)
                try_seq(a_opt2, "final_fine")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    total_time = time.monotonic() - total_start
    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={total_time:.1f}s")
    return [float(x) for x in best_seq]
