"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- FFT-based gradient computation (O(n log n) per step)
- Entropy mirror descent (multiplicative updates)
- Adaptive LSE smoothing with backtracking line search
- Nesterov-style momentum
- Multi-length exploration with periodic restarts
"""

import numpy as np
import time

def fft_conv(a):
    """Compute linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fft_gradient(w, a):
    """Compute gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    grad[i] = 2 * sum_j w[i+j] * a[j]
    Uses FFT cross-correlation.
    """
    n = len(a)
    N = 1 << (2 * n - 1 + n - 1).bit_length()  # enough for full correlation
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(W * A.conjugate(), N)
    return 2.0 * corr[:n]

def lse_objective(a, beta):
    """Compute LSE of convolution."""
    c = fft_conv(a)
    bc = beta * c
    bc_max = bc.max()
    return bc_max / beta + np.log(np.sum(np.exp(bc - bc_max))) / beta

def project_simplex(v):
    """Euclidean projection onto simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def multiplicative_project(v, a):
    """Multiplicative update: a_new[i] = a[i] * exp(v[i]), then normalize."""
    a_new = a * np.exp(v - v.max())  # shift for numerical stability
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a_new / s

def lse_mirror_descent(a0, time_limit, beta_start=10.0, beta_end=20000.0,
                        lr_start=0.5, momentum_init=0.85, verbose=True):
    """LSE-smoothed entropy mirror descent with adaptive parameters."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    c = fft_conv(a)
    M0 = c.max()
    best_obj = M0
    best_a = a.copy()
    
    beta = beta_start
    lr = lr_start
    momentum = momentum_init
    v = np.zeros(n)  # momentum buffer (in log-space)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    best_beta = beta_start
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute current convolution
        c = fft_conv(a)
        M_cur = c.max()
        
        # LSE weights
        bc = beta * c
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()
        
        # Gradient via FFT
        grad = fft_gradient(w, a)
        
        # Nesterov extrapolation (in the multiplicative space)
        # v is the "velocity" in log-space
        v = momentum * v - lr * grad
        v_clipped = np.clip(v, -50, 50)  # prevent explosion
        
        # Try the step
        a_try = multiplicative_project(v_clipped, a)
        
        # Evaluate
        c_try = fft_conv(a_try)
        M_try = c_try.max()
        
        # Backtracking if needed
        alpha = 1.0
        accepted = False
        if M_try < M_cur or no_improve > 20:
            accepted = True
        else:
            for _ in range(10):
                alpha *= 0.5
                v_attempt = np.clip(alpha * v_clipped, -50, 50)
                a_try = multiplicative_project(v_attempt, a)
                c_try = fft_conv(a_try)
                M_try = c_try.max()
                if M_try < M_cur:
                    accepted = True
                    break
        
        if accepted:
            a = a_try
            if M_try < best_obj - 1e-15:
                best_obj = M_try
                best_a = a.copy()
                no_improve = 0
                # Small increase in learning rate
                lr = min(lr * 1.01, 2.0)
            else:
                no_improve += 1
        else:
            no_improve += 1
            v = np.zeros(n)
            lr *= 0.5
        
        # Adapt beta
        if no_improve > 30:
            beta = min(beta * 1.8, beta_end)
            momentum = min(momentum + 0.01, 0.97)
            lr = max(lr * 0.8, 0.01)
        
        if beta < beta_end:
            beta *= 1.005
        
        # Periodic restart
        if no_improve > 150:
            rng = np.random.default_rng((it * 7919 + 13) % 999983)
            a = best_a + rng.standard_normal(n) * 0.003
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = lr_start * 0.5
            momentum = 0.9
            no_improve = 0
            if verbose:
                print(f"    Restart at it={it}, beta={beta:.1f}, M={fft_conv(a).max():.10f}")
    
    # After main loop, do a final Euclidean projection pass
    # (sometimes helps to "snap" to the boundary)
    a_final = project_simplex(best_a)
    c_final = fft_conv(a_final)
    if c_final.max() < best_obj:
        best_obj = c_final.max()
        best_a = a_final.copy()
    
    return best_a, best_obj

def lse_euclidean_descent(a0, time_limit, beta_start=10.0, beta_end=20000.0,
                          lr_start=0.1, verbose=True):
    """LSE-smoothed gradient descent with Euclidean projection."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    c = fft_conv(a)
    best_obj = c.max()
    best_a = a.copy()
    
    beta = beta_start
    lr = lr_start
    momentum = 0.9
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fft_conv(a)
        bc = beta * c
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()
        
        grad = fft_gradient(w, a)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm
        
        v = momentum * v - lr * grad
        a_try = project_simplex(a + v)
        
        c_try = fft_conv(a_try)
        M_try = c_try.max()
        
        if M_try < best_obj - 1e-14:
            best_obj = M_try
            best_a = a_try.copy()
            a = a_try
            no_improve = 0
            lr = min(lr * 1.005, 2.0)
        else:
            no_improve += 1
            if no_improve > 10:
                lr *= 0.93
            if no_improve > 80:
                a = best_a.copy()
                v = np.zeros(n)
                lr = max(lr_start * 0.5, 0.001)
                beta = min(beta * 1.5, beta_end)
                no_improve = 0
        
        if beta < beta_end:
            beta *= 1.003
    
    return best_a, best_obj

def interpolate_to_length(f, new_n):
    """Resample a normalized sequence to new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- FFT-based gradient computation (O(n log n) per step) for both LSE and peak-targeting
- Entropy mirror descent (multiplicative updates) for fast non-negativity handling
- Euclidean projected gradient as complementary method
- Adaptive LSE beta schedule with backtracking line search
- Nesterov-style momentum
- Multi-length exploration with periodic restarts and perturbations
- U-shaped and random initializations to escape local minima
"""

import numpy as np
import time


def fft_conv(a):
    """Compute linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fft_gradient(w, a):
    """Compute gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a via FFT.
    grad[i] = 2 * sum_j w[i+j] * a[j]
    This is 2 times the cross-correlation of w and a, evaluated at lags 0..n-1.
    """
    n = len(a)
    # Need N >= len(w) + len(a) - 1 = (2n-1) + n - 1 = 3n-2
    N = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(W * A.conjugate(), N)
    return 2.0 * corr[:n]


def project_simplex(v):
    """Euclidean projection onto simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_mirror_descent(a0, time_limit, beta_start=10.0, beta_end=20000.0,
                       lr_start=0.3, momentum_init=0.85, verbose=False):
    """LSE-smoothed entropy mirror descent with adaptive parameters and FFT gradient."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c = fft_conv(a)
    best_obj = c.max()
    best_a = a.copy()

    beta = beta_start
    lr = lr_start
    momentum = momentum_init
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fft_conv(a)
        M_cur = c.max()

        # LSE weights (numerically stable)
        bc = beta * c
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        # FFT-based gradient
        grad = fft_gradient(w, a)

        # Momentum update in log-space
        v = momentum * v - lr * grad
        v_clipped = np.clip(v, -40.0, 40.0)

        # Multiplicative update with backtracking
        alpha = 1.0
        accepted = False
        for _ in range(8):
            v_attempt = alpha * v_clipped
            a_try = a * np.exp(np.clip(v_attempt, -40.0, 40.0))
            a_try = np.maximum(a_try, 1e-30)
            s_try = a_try.sum()
            if s_try < 1e-15:
                break
            a_try /= s_try
            c_try = fft_conv(a_try)
            M_try = c_try.max()
            if M_try <= M_cur * (1 + 1e-12):
                accepted = True
                break
            alpha *= 0.5

        if accepted:
            a = a_try
            if M_try < best_obj - 1e-15:
                best_obj = M_try
                best_a = a.copy()
                no_improve = 0
                lr = min(lr * 1.008, 3.0)
            else:
                no_improve += 1
        else:
            no_improve += 2
            v *= 0.5
            lr *= 0.85

        # Adapt beta and momentum on stagnation
        if no_improve > 40:
            beta = min(beta * 2.0, beta_end)
            momentum = min(momentum + 0.01, 0.97)
        if no_improve > 100:
            lr = max(lr * 0.7, 0.005)

        if beta < beta_end:
            beta *= 1.004

        # Periodic restart with perturbation
        if no_improve > 200:
            rng = np.random.default_rng((it * 7919 + 13) % 999983)
            scale = 0.002 + 0.003 * min(1.0, (no_improve - 200) / 100.0)
            a = np.maximum(best_a + rng.standard_normal(n) * scale, 0.0)
            a /= a.sum()
            v = np.zeros(n)
            lr = lr_start * 0.5
            momentum = 0.9
            no_improve = 0
            beta = min(beta * 0.5, beta_end)
            if verbose:
                print(f"    Mirror restart it={it}, beta={beta:.0f}, M={fft_conv(a).max():.10f}")

    return best_a, best_obj


def lse_euclidean_descent(a0, time_limit, beta_start=10.0, beta_end=20000.0,
                          lr_start=0.1, verbose=False):
    """LSE-smoothed gradient descent with Euclidean projection and FFT gradient."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c = fft_conv(a)
    best_obj = c.max()
    best_a = a.copy()

    beta = beta_start
    lr = lr_start
    momentum = 0.9
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fft_conv(a)
        bc = beta * c
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        grad = fft_gradient(w, a)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        v = momentum * v - lr * grad
        a_try = project_simplex(a + v)

        c_try = fft_conv(a_try)
        M_try = c_try.max()

        if M_try < best_obj - 1e-14:
            best_obj = M_try
            best_a = a_try.copy()
            a = a_try
            no_improve = 0
            lr = min(lr * 1.003, 1.5)
        else:
            no_improve += 1
            if no_improve > 15:
                lr *= 0.94
            if no_improve > 100:
                a = best_a.copy()
                v = np.zeros(n)
                lr = max(lr_start * 0.4, 0.001)
                beta = min(beta * 1.6, beta_end)
                no_improve = 0

        if beta < beta_end:
            beta *= 1.004

        # Restart
        if no_improve > 200:
            rng = np.random.default_rng((it * 104729 + 7) % 999983)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            lr = lr_start * 0.5
            no_improve = 0
            beta = min(beta * 0.8, beta_end)
            if verbose:
                print(f"    Euclid restart it={it}, beta={beta:.0f}, M={fft_conv(a).max():.10f}")

    return best_a, best_obj


def peak_target_descent(a0, time_limit, n_peaks=8, lr_start=0.05, momentum_init=0.93):
    """Directly target top peaks with subgradient + FFT-based computation."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    conv_len = 2 * n - 1

    c = fft_conv(a)
    best_obj = c.max()
    best_a = a.copy()

    lr = lr_start
    momentum = momentum_init
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fft_conv(a)

        # Select top peaks with soft weighting
        k_top = min(n_peaks, conv_len)
        peaks = np.argpartition(c, -k_top)[-k_top:]
        pv = c[peaks]
        # Exponential weighting emphasizing highest peaks
        scaled = pv - pv.min()
        w_peaks = np.exp(scaled - scaled.max())
        w_peaks /= w_peaks.sum()

        w_full = np.zeros(conv_len)
        w_full[peaks] = w_peaks

        grad = fft_gradient(w_full, a)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad /= grad_norm

        v = momentum * v - lr * grad
        a_try = project_simplex(a + v)
        c_try = fft_conv(a_try)
        M_try = c_try.max()

        if M_try < best_obj - 1e-14:
            best_obj = M_try
            best_a = a_try.copy()
            a = a_try
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                lr *= 0.95
            if no_improve > 150:
                rng = np.random.default_rng((it * 31 + 7) % 999983)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
                v = np.zeros(n)
                lr = max(lr * 0.5, 0.001)
                no_improve = 0
                # Increase peaks targeted
                n_peaks = min(n_peaks + 2, 64)

    return best_a, best_obj

def interpolate_to_length(f, new_n):
    """Resample a normalized sequence to new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def u_shape_init(n):
    """Create a U-shaped initialization (large at edges, small in middle)."""
    x = np.linspace(0, 1, n)
    f = 0.3 + 1.4 * (2 * x - 1) ** 2
    f = np.maximum(f, 0.01)
    f /= f.sum()
    return f


def random_init(n, seed=0):
    """Random non-negative initialization."""
    rng = np.random.default_rng(seed)
    f = rng.exponential(1.0, size=n)
    f = np.maximum(f, 0.01)
    f /= f.sum()
    return f


def gaussian_taper_init(n, center=0.3, width=0.2):
    """Asymmetric Gaussian-tapered initialization."""
    x = np.linspace(0, 1, n)
    f = np.exp(-((x - center) / width) ** 2) + 0.1 * np.exp(-((x - (1 - center)) / width) ** 2)
    f = np.maximum(f, 0.01)
    f /= f.sum()
    return f
