def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_gradient(a, beta):
    """
    Compute the gradient of log-sum-exp(beta * autocorrelation(a)) w.r.t. a.
    
    Using the formula:
    g_i = sum_k w_k * (a_{k-i} + a_{i-k})
    where w_k = softmax(beta * (a*a)_k)
    
    This can be computed as:
    g_i = sum_j a_j * w_{i+j} + sum_j a_j * w_{i-j}
        = (reverse_conv)(i) + (conv)(i)
    
    where we use w as a kernel of length 2n-1.
    """
    n = len(a)
    c = fast_convolve(a)  # length 2n-1
    
    # Compute softmax weights (numerically stable)
    c_max = c.max()
    exp_c = np.exp(beta * (c - c_max))
    w = exp_c / exp_c.sum()  # length 2n-1
    
    # g_i = sum_{j=0}^{n-1} a_j * w_{i+j} + sum_{j=0}^{i} a_j * w_{i-j}
    # First term: sum_j a_j * w_{i+j} for i = 0,...,n-1
    # This is a "cross-correlation" of a with w.
    # Let's compute it via FFT.
    
    # Second term: sum_{j=0}^{i} a_j * w_{i-j} = (a conv w)_i (truncated to first n elements)
    # This is just the linear convolution of a (length n) and w (length 2n-1), first n elements.
    
    # For the first term: sum_j a_j * w_{i+j}
    # Let's define w_reversed = w[::-1] (length 2n-1)
    # Then sum_j a_j * w_{i+j} = sum_j a_j * w_reversed[(2n-2)-(i+j)] = sum_j a_j * w_reversed[2n-2-i-j]
    # = (a conv w_reversed)_{2n-2-i}
    # So the first term for index i is (a conv w_reversed)_{2n-2-i}
    
    # Let's verify: (a conv w_reversed)_m = sum_j a_j * w_reversed[m-j] = sum_j a_j * w[(2n-2)-(m-j)]
    # For m = 2n-2-i: sum_j a_j * w[(2n-2)-(2n-2-i-j)] = sum_j a_j * w[i+j] ✓
    
    # So: g_i = (a conv w_reversed)_{2n-2-i} + (a conv w)_i
    
    # Compute both convolutions via FFT
    fft_size = 1 << (3 * n - 3).bit_length()  # need at least n + (2n-1) - 1 = 3n-2
    
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    W_rev = np.fft.rfft(w[::-1], fft_size)
    
    conv_aw = np.fft.irfft(A * W, fft_size)  # a conv w, length 3n-2
    conv_awrev = np.fft.irfft(A * W_rev, fft_size)  # a conv w_reversed, length 3n-2

    g = np.zeros(n)
    for i in range(n):
        g[i] = conv_awrev[2 * n - 2 - i] + conv_aw[i]
    
    return g * beta  # multiply by beta for the log-sum-exp gradient

2. LP-based Frank-Wolfe for polishing

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(a, beta):
    """Compute gradient of log-sum-exp(beta * autocorrelation)."""
    n = len(a)
    c = fast_convolve(a)
    
    c_max = c.max()
    shifted = beta * (c - c_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -700, 700)
    exp_c = np.exp(shifted)
    w = exp_c / exp_c.sum()
    
    fft_size = 1 << (3 * n - 3).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    W_rev = np.fft.rfft(w[::-1], fft_size)
    
    conv_aw = np.fft.irfft(A * W, fft_size)
    conv_awrev = np.fft.irfft(A * W_rev, fft_size)
    
    g = conv_awrev[n - 1:2 * n - 1][::-1] + conv_aw[0:n]
    return g * beta


def nesterov_optimize(f0, time_limit, beta_start=50, beta_end=5000, 
                       lr_start=0.01, momentum_start=0.9, verbose=False):
    """Nesterov-accelerated projected gradient with log-sum-exp smoothing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    y = f.copy()  # Nesterov lookahead
    v = np.zeros(n)
    
    beta = beta_start
    lr = lr_start
    momentum = momentum_start
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = elapsed / time_limit
        
        # Adaptive beta: increase over time
        beta = beta_start * (beta_end / beta_start) ** progress
        # Adaptive learning rate: decrease over time
        lr = lr_start / (1 + 0.5 * it)
        # Momentum: increase toward 1
        momentum = min(momentum_start + 0.001 * it, 0.999)
        
        # Compute gradient at y
        grad_y = compute_gradient(y, beta)
        
        # Nesterov update
        v_new = momentum * v - lr * grad_y
        y = f - v_new / (momentum + 1e-10)  # or y = f + (it-1)/(it+2) * v_new
        y = project_simplex(y)
        
        # Projected step
        f_new = project_simplex(f - lr * grad_y)
        
        # Actually, let me use the standard FTRL or Mirror Descent approach
        # which is better for simplex constraints.
        
        # Let me use the multiplicative weights / mirror descent approach:
        # f_new[i] ∝ f[i] * exp(-lr * grad[i])
        # This is the negative entropy mirror descent, which naturally stays in the simplex.
        
        logits = np.log(f + 1e-30) - lr * grad_y
        logits -= logits.max()  # for stability
        f_new = np.exp(logits)
        f_new = f_new / f_new.sum()
        
        conv = fast_convolve(f_new)
        obj = conv.max()
        
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = f_new
        v = v_new
        
        if verbose and it % 200 == 0:
            print(f"    Nesterov iter {it}: obj={best_obj:.10f}, beta={beta:.1f}")
        
        if no_improve > 200:
            # Restart with perturbation
            rng = np.random.default_rng(it * 104729 + 7)
            scale = 0.01 + 0.01 * min(no_improve // 500, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            no_improve = 0
    
    return best_f, best_obj

Minimizes 2*n*max(conv(a,a))/sum(a)^2 using a hybrid approach:

import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_smoothed_grad(a, beta):
    """Gradient of log-sum-exp(beta * autocorr) w.r.t. a."""
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    shifted = np.clip(beta * (c - c_max), -500, 500)
    exp_c = np.exp(shifted)
    w = exp_c / exp_c.sum()
    
    fft_size = 1 << (3 * n - 3).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    W_rev = np.fft.rfft(w[::-1], fft_size)
    conv_aw = np.fft.irfft(A * W, fft_size)
    conv_awrev = np.fft.irfft(A * W_rev, fft_size)
    
    g = conv_awrev[n - 1:2 * n - 1][::-1] + conv_aw[0:n]
    return beta * g

def lp_solve_min_peak(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    n_c = 2 * n - 1
    A_ub = np.zeros((n_c, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_c)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return None
