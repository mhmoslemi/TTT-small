"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive mirror descent
(multiplicative weights) with weighted subgradients from top peaks,
combined with coordinate descent refinement and multi-length exploration.
"""

import numpy as np
import time


def fast_max_conv(a):
    """Compute max of auto-convolution using FFT. Returns (max_val, argmax)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(A * A, fft_size)[:size]
    k = int(np.argmax(conv))
    return conv[k], k, conv


def compute_subgradient(a, conv, top_k=10, temperature=0.01):
    """
    Compute weighted subgradient of max_k conv[k].
    Uses softmax weighting over top-k peaks.
    Returns gradient vector g of length n.
    """
    n = len(a)
    conv_len = len(conv)
    
    # Find top-k peaks
    k_actual = min(top_k, conv_len)
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    
    # Softmax weights
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    # Compute subgradient
    g = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-10:
            continue
        k = int(peaks[i])
        # d conv[k]/d a[m] = 2*a[k-m] if 0 <= k-m < n, else 0
        m_lo = max(0, k - (n - 1))
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            indices = np.arange(m_lo, m_hi + 1)
            g[indices] += 2.0 * w[i] * a[k - indices]
    
    return g


def mirror_descent_step(a, g, eta):
    """Multiplicative weights update: a_new[m] = a[m] * exp(-eta*g[m]), normalized."""
    # Clip g to avoid overflow
    g_clipped = np.clip(g, -50/eta, 50/eta) if eta > 0 else np.zeros_like(g)
    log_update = -eta * g_clipped
    # For stability, subtract max
    log_update -= log_update.max()
    a_new = a * np.exp(log_update)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def cosine_init(n, freq, amp):
    """Cosine-modulated initialization."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def hann_init(n):
    """Hann window initialization."""
    j = np.arange(n)
    a = 0.5 * (1 + np.cos(2.0 * np.pi * j / n))
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n

def hamming_init(n):

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def cd_eval_at_x(a, j, x, old_val):
    """Evaluate max_conv when a[j]=x and others scaled to keep sum=1."""
    n = len(a)
    a_tmp = a.copy()
    a_tmp[j] = x
    denom = 1.0 - old_val
    if x < 1.0 and denom > 1e-12:
        scale = (1.0 - x) / denom
        mask = np.ones(n, dtype=bool)
        mask[j] = False
        a_tmp[mask] *= scale
    cm, _, _ = fast_max_conv(a_tmp)
    return cm


def cd_apply_optimal(a, j, x, old_val):
    """Apply optimal value x to coordinate j, scaling others."""
    n = len(a)
    a[j] = x
    denom = 1.0 - old_val
    if x < 1.0 and denom > 1e-12:
        scale = (1.0 - x) / denom
        mask = np.ones(n, dtype=bool)
        mask[j] = False
        a[mask] *= scale
    return a


def coordinate_descent(a0, time_limit):
    """
    Golden-section coordinate descent. Each pass cycles through all coordinates.
    Returns (best_a, best_obj).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj, _, _ = fast_max_conv(a)

    t0 = time.monotonic()
    no_pass_improve = 0
    inv_GR = 2.0 / (1.0 + np.sqrt(5.0))

    while time.monotonic() - t0 < time_limit:
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-13:
                continue
            denom = 1.0 - old_val
            if denom < 1e-11:
                continue

            lo = 0.0
            hi = min(0.9999, max(2.0 * old_val, old_val + 0.1))
            if hi <= lo:
                continue

            c1 = lo + inv_GR * (hi - lo)
            c2 = lo + (1.0 - inv_GR) * (hi - lo)
            f1 = cd_eval_at_x(a, j, c1, old_val)
            f2 = cd_eval_at_x(a, j, c2, old_val)

            for _ in range(12):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + inv_GR * (hi - lo)
                    f1 = cd_eval_at_x(a, j, c1, old_val)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + (1.0 - inv_GR) * (hi - lo)
                    f2 = cd_eval_at_x(a, j, c2, old_val)

            opt_x = (lo + hi) / 2.0
            opt_val = min(f1, f2)

            if opt_val < best_obj - 1e-13:
                a = cd_apply_optimal(a, j, opt_x, old_val)
                best_obj = opt_val
                best_a = a.copy()
                pass_improved = True

        if pass_improved:
            no_pass_improve = 0
        else:
            no_pass_improve += 1
            if no_pass_improve >= 2:
                break

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimizer: mirror descent + coordinate descent + length adaptation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    # === Phase 1: Quick initialization sweep ===
    print("=== Phase 1: Initialization sweep ===")
    t_phase1_start = time.monotonic()
    candidates = []

    if parent is not None:
        s = try_seq(parent, "parent")
        candidates.append(parent.copy())

    init_configs = [
        (768, "cos1", lambda n: make_cosine_init(n, 1, 0.3)),
        (768, "cos2", lambda n: make_cosine_init(n, 2, 0.2)),
        (768, "cos3", lambda n: make_cosine_init(n, 3, 0.15)),
        (768, "cos5", lambda n: make_cosine_init(n, 5, 0.1)),
        (768, "hann", lambda n: make_hann_init(n)),
        (768, "flat", lambda n: make_flat_top_init(n)),
        (700, "cos2", lambda n: make_cosine_init(n, 2, 0.25)),
        (800, "cos2", lambda n: make_cosine_init(n, 2, 0.25)),
        (850, "cos1", lambda n: make_cosine_init(n, 1, 0.3)),
        (850, "cos2", lambda n: make_cosine_init(n, 2, 0.2)),
        (900, "cos2", lambda n: make_cosine_init(n, 2, 0.2)),
        (1024, "cos2", lambda n: make_cosine_init(n, 2, 0.2)),
        (1024, "cos3", lambda n: make_cosine_init(n, 3, 0.15)),
    ]

    for n_init, name, init_fn in init_configs:
        a_init = init_fn(n_init)
        s = try_seq(a_init, f"init_{name}_n{n_init}")
        candidates.append(a_init)
        # Stop if time running short
        if time.monotonic() - t_phase1_start > 30:
            break

    # Sort candidates by their normalized objective (max_conv)
    scored_candidates = []
    for a_c in candidates:
        mc, _, _ = fast_max_conv(a_c)
        scored_candidates.append((mc, a_c))
    scored_candidates.sort(key=lambda x: x[0])
    print(f"  Best init max_conv: {scored_candidates[0][0]:.10f}")

    # === Phase 2: Mirror descent on top candidates ===
    print("\n=== Phase 2: Mirror descent optimization ===")
    for idx in range(min(4, len(scored_candidates))):
        rem = deadline - time.monotonic() - 300
        if rem < 30:
            break
        a_init = scored_candidates[idx][1]
        n_c = len(a_init)
        if idx == 0:
            t_alloc = min(rem * 0.35, 200)
        elif idx == 1:
            t_alloc = min(rem * 0.25, 120)
        else:
            t_alloc = min(rem * 0.15, 60)

        print(f"  MD cand {idx} (n={n_c}, t={t_alloc:.0f}s)")
        a_opt, obj = mirror_descent_optimize(a_init, time_limit=t_alloc,
                                             eta0=0.08, top_k=min(25, n_c // 12),
                                             temp0=0.004)
        try_seq(a_opt, f"md_cand{idx}_n{n_c}")

    # === Phase 3: Coordinate descent on best ===
    print(f"\n=== Phase 3: Coordinate descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t_cd = min(rem * 0.5, 150)
            n_b = len(best_seq)
            print(f"  CD t={t_cd:.0f}s, n={n_b}")
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd_main")

    # === Phase 4: Length adaptation ===
    print(f"\n=== Phase 4: Length adaptation (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        target_lengths = sorted([l for l in [600, 650, 700, 750, 768, 800, 850, 900, 950, 1024, 1100, 1200]
                                  if abs(l - n_best) > 20],
                                key=lambda l: abs(l - n_best))

        for target_n in target_lengths[:5]:
            rem = deadline - time.monotonic() - 100
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_md = min(rem * 0.3, 60)
            print(f"  Interp n={target_n}, MD t={t_md:.0f}s")
            a_opt, _ = mirror_descent_optimize(f_interp, time_limit=t_md,
                                               eta0=0.06, top_k=min(20, target_n // 15),
                                               temp0=0.004)
            try_seq(a_opt, f"interp_n{target_n}")

            rem = deadline - time.monotonic() - 80
            if rem > 15:
                a_opt2, _ = coordinate_descent(a_opt, time_limit=min(rem * 0.4, 30))
                try_seq(a_opt2, f"cd_n{target_n}")

    # === Phase 5: Fine-tuning via perturbations ===
    print(f"\n=== Phase 5: Fine-tuning (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for trial in range(8):
            rem = deadline - time.monotonic() - 40
            if rem < 15:
                break
            noise_scale = 0.0005 + 0.0005 * trial
            noise = rng.standard_normal(n_best) * noise_scale
            f_pert = project_simplex(best_seq + noise)
            t_md = min(rem * 0.25, 25)
            a_opt, _ = mirror_descent_optimize(f_pert, time_limit=t_md,
                                               eta0=0.04, top_k=12, temp0=0.003)
            try_seq(a_opt, f"perturb_{trial}")

    # === Phase 6: Final CD polish ===
    print(f"\n=== Phase 6: Final polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 8:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 40))
            try_seq(a_opt, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
