    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_at_peak(a, k, n):
    """Compute ∂b_k/∂a for linear convolution peak at index k."""
    # ∂b_k/∂a_i = 2*a_{k-i} if 0 <= k-i < n, else 0
    r = np.zeros(n)
    # k-i in [0, n-1] means i in [k-n+1, k]
    i_start = max(0, k - n + 1)
    i_end = min(n - 1, k)
    if i_start <= i_end:
        r[i_start:i_end+1] = 2.0 * a[k - np.arange(i_start, i_end+1)]
    return r


def top_k_subgradient(a, k_peaks, n, weights=None):
    """Compute weighted average subgradient from top-k peaks."""
    if weights is None:
        weights = np.ones(len(k_peaks)) / len(k_peaks)
    sg = np.zeros(n)
    for i, k in enumerate(k_peaks):
        sg += weights[i] * subgradient_at_peak(a, k, n)
    return sg


def fwl_oracle(a, M, n):
    """
    Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k=0,...,2n-2, g >= 0.
    Returns the optimal g (normalized to sum 1) or None.
    """
    c = -np.ones(n)
    # Constraints: for each k in 0,...,2n-2: sum_j a_j * g_{k-j} <= M
    # where g_{k-j} is valid only if 0 <= k-j < n
    # This gives: sum_{j=max(0,k-n+1)}^{min(n-1,k)} a_j * g_{k-j} <= M
    # Rearranging: sum_{i=max(0,k-n+1)}^{min(n-1,k)} a_{k-i} * g_i <= M
    
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            A_ub[k, i] = a[k - i]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            g = np.maximum(res.x, 0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None


def subgradient_descent(a0, time_limit, top_k=20, lr_init=0.01, momentum=0.9):
    """
    Projected subgradient descent with top-k peaks and momentum.
    """
    a = np.maximum(a0, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    patience = 500
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        
        # Get top-k peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            pvals = conv
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pvals = conv[peaks]
        
        # Weight by value (higher peaks get more weight)
        w = pvals ** 2  # Square weighting emphasizes the max
        w = w / w.sum()
        
        # Blend: also include some uniform weight to explore
        w_unif = np.ones(len(peaks)) / len(peaks)
        w_blend = 0.7 * w + 0.3 * w_unif
        w_blend = w_blend / w_blend.sum()
        
        sg = top_k_subgradient(a, peaks, n, w_blend)
        
        # Momentum update
        v = momentum * v - lr * sg
        candidate = project_simplex(a + v)
        
        obj = fast_convolve(candidate).max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = candidate.copy()
            a = candidate
            no_improve = 0
        else:
            no_improve += 1
            a = candidate  # Still move (for exploration)
        
        # Adaptive step size
        if no_improve > 50:
            lr *= 0.95
        if no_improve > 200:
            lr *= 0.9
        lr = max(lr, 1e-6)
        
        if no_improve > patience:
            # Random restart
            rng = np.random.default_rng((it * 7919) % (2**31))
            noise = 0.002 + 0.005 * min(1.0, (no_improve - patience) / 500)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            lr = lr_init * 0.5
            no_improve = 0
    
    return best_a, best_obj


def fw_lp_optimize(a0, time_limit, n_lp_constraints=None):
    """Frank-Wolfe with LP oracle."""
    a = np.maximum(a0, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = M
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 2:
            break
        
        g = fwl_oracle(a, M, n)
        if g is None:
            # Try relaxed M
            g = fwl_oracle(a, M * 1.1, n)
            if g is None:
                no_improve += 5
                if no_improve > 30:
                    break
                continue
        
        # Line search for best alpha
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            at = (1 - alpha) * a + alpha * g
            at = np.maximum(at, 0)
            st = at.sum()
            if st < 1e-15:
                continue
            at = at / st
            val = fast_convolve(at).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            # Refine with bisection
            lo = max(0, best_alpha - 0.05)
            hi = min(1, best_alpha + 0.05)
            for _ in range(6):
                mid = (lo + hi) / 2
                at = (1 - mid) * a + mid * g
                at = np.maximum(at, 0)
                st = at.sum()
                if st < 1e-15:
                    break
                at = at / st
                val = fast_convolve(at).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
            
            a = (1 - best_alpha) * a + best_alpha * g
            a = np.maximum(a, 0)
            a = a / a.sum()
            M = fast_convolve(a).max()
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 40:
            rng = np.random.default_rng((it * 104729) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            M = fast_convolve(a).max()
            no_improve = 0
    
    return best_a, best_obj


def generate_initializations(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Arcsine: a_j ∝ 1/sqrt((j+1)(n-j))
    j = np.arange(1, n + 1)
    arcs = 1.0 / np.sqrt(j * (n - j + 1))
    # Handle edge case
    arcs[0] = 1.0 / np.sqrt(n)
    arcs[-1] = 1.0 / np.sqrt(n)
    inits.append(arcs / arcs.sum())
    
    # Edge-concentrated (inverse distance to center)
    x = np.arange(n, dtype=np.float64)
    center = 0.5 * (n - 1)
    edge_w = 1.0 / (np.abs(x - center) + 1.0)
    inits.append(edge_w / edge_w.sum())
    
    # Two-peak
    g1 = np.exp(-0.5 * ((x - 0.2 * (n-1)) / (0.1 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.8 * (n-1)) / (0.1 * n)) ** 2)
    two_peak = g1 + g2
    inits.append(two_peak / two_peak.sum())
    
    # Discrete "flat convolution" approximation
    # a_j ∝ sqrt(1/(j+1)) for j < n/2, symmetric for j >= n/2
    half = n // 2
    af = np.zeros(n)
    for j in range(n):
        d = min(j, n - 1 - j) + 1
        af[j] = 1.0 / np.sqrt(d)
    inits.append(af / af.sum())
    
    # Cosine bell
    cos_bell = np.cos(np.pi * (x / (n - 1) - 0.5)) ** 2 + 0.01
    inits.append(cos_bell / cos_bell.sum())
    
    # Random
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    # Random with concentration
    inits.append(rng.dirichlet(np.ones(n) * 0.1))
    
    # From parent
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
        
        # Perturbed parent
        pp = project_simplex(parent + rng.standard_normal(len(parent)) * 0.01)
        ppx = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), pp)
        ppx = np.maximum(ppx, 0)
        pps = ppx.sum()
        if pps > 1e-10:
            inits.append(ppx / pps)
    
    return inits


def active_set_refine(a0, time_limit):
    """
    Active-set refinement: identify the peaks that are at or near the max,
    and solve a QP that equalizes them.
    """
    a = np.maximum(a0, 0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        M = conv.max()
        
        # Find peaks at or near the max (within 1%)
        threshold = M * 0.99
        active_peaks = np.where(conv >= threshold)[0]
        
        if len(active_peaks) == 0:
            active_peaks = np.array([np.argmax(conv)])
        
        # Build the QP: minimize t s.t. b_k(a) <= t for k in active_peaks, a >= 0, sum(a) = 1
        # This is equivalent to: find a on the simplex that minimizes max_{k in active} b_k(a)
        
        # Use a simple approach: project subgradient with only active peaks
        w = np.ones(len(active_peaks)) / len(active_peaks)
        sg = top_k_subgradient(a, active_peaks, n, w)
        
        # Backtracking line search
        lr = 0.01
        best_candidate = a.copy()
        best_val = M
        
        for trial in [lr, lr*2, lr*4, lr*0.5, lr*0.25]:
            candidate = project_simplex(a - trial * sg)
            val = fast_convolve(candidate).max()
            if val < best_val:
                best_val = val
                best_candidate = candidate
        
        if best_val < M - 1e-15:
            a = best_candidate
            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_a = a.copy()
        else:
            # Try with a wider threshold
            threshold = M * 0.95
            active_peaks = np.where(conv >= threshold)[0]
            if len(active_peaks) > 1:
                w = np.ones(len(active_peaks)) / len(active_peaks)
                sg = top_k_subgradient(a, active_peaks, n, w)
                candidate = project_simplex(a - 0.005 * sg)
                val = fast_convolve(candidate).max()
                if val < best_obj - 1e-14:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
            else:
                # Stuck, try random perturbation
                rng = np.random.default_rng(it * 31 + 7)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Hybrid search combining:
    1. Arcsine-inspired initializations
    2. Multi-n exploration
    3. Projected subgradient with adaptive peaks
    4. LP-based Frank-Wolfe
    5. Active-set refinement
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}, time={time.monotonic()-t_start:.0f}s)")
        return score
    
    t_start = time.monotonic()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if parent is None:
        parent = np.ones(384) / 384
    
    n_parent = len(parent)
    
    # === Phase 1: Quick multi-init, multi-n exploration ===
    print("\n=== Phase 1: Quick Exploration ===")
    
    # Try different n values
    n_candidates = sorted(set([
        n_parent,
        max(128, n_parent // 2),
        min(2048, n_parent * 2),
        256, 384, 512, 768, 1024,
    ]))
    n_candidates = [n for n in n_candidates if 64 <= n <= 2048]
    
    for n_try in n_candidates:
        if time.monotonic() > deadline - 600:
            break
        
        inits = generate_initializations(n_try, parent, rng)
        
        # Quick evaluation of all inits
        best_init = None
        best_init_score = float('inf')
        for idx, init in enumerate(inits):
            sc = try_seq(init, f"init{n_try}_{idx}")
            if sc < best_init_score:
                best_init_score = sc
                best_init = init
        
        if best_init is None:
            continue
        
        # Short subgradient refinement
        remaining = deadline - time.monotonic() - 500
        if remaining > 10:
            t_alloc = min(remaining, 30)
            a_opt, _ = subgradient_descent(best_init, time_limit=t_alloc, top_k=20, lr_init=0.02)
            try_seq(a_opt, f"sg_quick_n{n_try}")
    
    print(f"  Phase 1 best: {best_score:.8f}")
    
    # === Phase 2: Deep subgradient on best ===
    print("\n=== Phase 2: Deep Subgradient ===")
    
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 400
        if remaining > 20:
            t_alloc = min(remaining, 200)
            print(f"  Deep SG (k=30), n={len(best_seq)}, t={t_alloc:.0f}s")
            a_opt, _ = subgradient_descent(best_seq, time_limit=t_alloc, top_k=30, lr_init=0.01)
            try_seq(a_opt, "sg_deep_k30")
        
        remaining = deadline - time.monotonic() - 300
        if remaining > 20:
            t_alloc = min(remaining, 100)
            print(f"  Deep SG (k=50), t={t_alloc:.0f}s")
            a_opt, _ = subgradient_descent(best_seq, time_limit=t_alloc, top_k=50, lr_init=0.008)
            try_seq(a_opt, "sg_deep_k50")
    
    # === Phase 3: LP-based FW ===
    print("\n=== Phase 3: FW-LP ===")
    
    if best_seq is not None
