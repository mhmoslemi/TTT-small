def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_direction_reduced(f, M, n, active_indices, max_constraints=60):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for active k, g >= 0.
    Returns direction g (normalized to sum=1) or None."""
    # Build constraint matrix using only active constraints
    K = min(len(active_indices), max_constraints)
    if K == 0:
        return None
    
    rows = active_indices[:K]
    A_ub = np.zeros((K, n))
    for j in range(n):
        # (f*g)[k] = sum_i f[i]*g[k-i], valid when 0<=k-i<n
        # For fixed j (g[j]), the lags k where g[j] appears: k = i+j for i in [0,n-1]
        # So g[j] contributes f[i] to row (i+j) for i in [0, n-1]
        js = np.arange(j, j + n)
        valid = (js >= rows[0]) & (js <= rows[-1])
        if valid.any():
            idx_js = js[valid]
            # Find which rows these correspond to
            row_indices = np.searchsorted(rows, idx_js)
            row_indices = row_indices[(row_indices < K) & (rows[np.minimum(row_indices, K-1)] == idx_js)]
            if len(row_indices) > 0:
                f_offset = idx_js - j  # = i values
                A_ub[row_indices, j] = f[f_offset]
    
    c = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None


def lp_fw_optimize(f0, time_limit, K=40, verbose=False):
    """LP-based Frank-Wolfe with active constraint management."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    # Active set: top K lags
    active = np.argsort(conv)[::-1][:min(K, 2*n-1)]

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    prev_g = None

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        # Update active set periodically
        if it % 20 == 0:
            conv = fast_convolve(f)
            M = conv.max()
            active = np.argsort(conv)[::-1][:min(K, 2*n-1)]

        g = lp_direction_reduced(f, M, n, active, max_constraints=K)
        if g is None:
            # Try with relaxed M
            g = lp_direction_reduced(f, M * 1.01, n, active, max_constraints=K)

                g = lp_direction_reduced(f, M * 1.05, n, active, max_constraints=K)

                    no_improve += 1
                    if no_improve > 50:
                        break
                    continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 60:
                rng = np.random.default_rng(it * 137 + 42)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)

                active = np.argsort(conv)[::-1][:min(K, 2*n-1)]
                no_improve = 0
                continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:

            if st < 1e-15:

            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)

            conv = fast_convolve(f_new)

                best_f = f.copy()
                no_improve = 0

            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 80:

            conv = fast_convolve(f)
            M = conv.max()
            active = np.argsort(conv)[::-1][:min(K, 2*n-1)]
            no_improve = 0

    return best_f, best_obj

                best_obj = C.max()
                best_f = f.copy()
                pass_improved = True

        if pass_improved:
            no_improve_passes = 0
        else:
            no_improve_passes += 1
            if no_improve_passes >= 5:
                break

    return best_f, best_obj


def smooth_gradient_descent(f0, time_limit, verbose=False):
    """Adam optimizer on smoothed max using top-K sum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    base_lr = 0.003

    it = 0
    no_improve = 0
    max_iters = int(time_limit / 0.02)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / max(1, max_iters * 0.85))
        lr = base_lr * (1.0 - 0.6 * progress)
        top_k = max(10, int(100 * (1.0 - progress) + 20 * progress))
        T = 2.0 / n * (1.0 - progress) + 0.3 / n * progress

        conv = fast_convolve(f)
        total_len = len(conv)
        k_actual = min(top_k, total_len)
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = T * (pv - pv.max())
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws

        # Subgradient: for C[k] = sum_j f[j]*f[k-j], dC[k]/df[i] = f[k-i] + (f[i-k] if i-k>=0 and k-i>=0 else 0)
        # More precisely: dC[k]/df[i] = f[k-i] (if 0<=k-i<n) + f[i-k] (if 0<=i-k<n)
        # If i==k-i (i.e., 2i=k), it's 2*f[i]
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            # dC[k]/df[i] for all i:
            # term 1: f[k-i] valid when 0 <= k-i < n => i in [k-n+1, k]
            # term 2: f[i-k] valid when 0 <= i-k < n => i in [k, k+n-1]
            # So dC[k]/df[i] = f[k-i]*1[k-i in [0,n-1]] + f[i-k]*1[i-k in [0,n-1]]
            
            i_lo1 = max(0, k - n + 1)
            i_hi1 = min(n - 1, k)
            i_lo2 = max(0, k)
            i_hi2 = min(n - 1, k + n - 1)
            
            if i_lo1 <= i_hi1:
                idx1 = np.arange(i_lo1, i_hi1 + 1)
                vals1 = f[k - idx1]  # f[k-i] for i in range
                sg[idx1] += 2.0 * w[idx] * vals1  # factor of 2 because both terms contribute when i is in both ranges
            
            # Actually, let me be more careful:
            # dC[k]/df[i] = f[k-i] (when 0<=k-i<n) + f[i-k] (when 0<=i-k<n)
            # When both conditions hold (i.e., max(0,k) <= i <= min(n-1,k+n-1) AND max(0,k-n+1) <= i <= min(n-1,k)):
            # i.e., when both k-i and i-k are in [0,n-1], which means |i-k| <= n-1 and both are valid
            # In this case, dC[k]/df[i] = f[k-i] + f[i-k]
            # When only one holds, dC[k]/df[i] = f[k-i] or f[i-k]
            
            # Let me just compute it directly:
            sg_tmp = np.zeros(n)
            for i in range(n):
                val = 0.0
                if 0 <= k - i < n:
                    val += f[k - i]
                if 0 <= i - k < n:
                    val += f[i - k]
                sg_tmp[i] = val
            sg += 2.0 * w[idx] * sg_tmp

        # Adam update
        it1 = it + 1
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** it1)
        v_hat = v / (1 - beta2 ** it1)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        f_new = project_simplex(f - step)
        f = f_new
        new_M = fast_convolve(f).max()

        if new_M < best_obj - 1e-14:
            best_obj = new_M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                rng = np.random.default_rng(it * 41 + 13)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                m = np.zeros(n)
                v = np.zeros(n)
                no_improve = 0

    return best_f, best_obj


def fast_pairwise(f0, time_limit, verbose=False):
    """Optimized pairwise descent with precomputed structure."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    total_n = 2 * n - 1

    C = fast_convolve(f)
    best_obj = C.max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    no_improve_rounds = 0

    while time.monotonic() - t0 < time_limit:
        round_improved = False

        # Try several random pairs
        for _ in range(2 * n):
            if time.monotonic() - t0 > time_limit:
                break

            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue

            c = f[i] + f[j]
            if c < 1e-10:
                continue

            # Quick scan of t values
            best_t = f[i] / c
            best_val = best_obj  # conservative

            # Try several t values
            ts = np.array([0.0, 0.1, 0.3, 0.5, 0.7, 0.9, 1.0,
                           max(0, best_t - 0.15), min(1, best_t + 0.15),
                           max(0, best_t - 0.05), min(1, best_t + 0.05)])
            ts = np.unique(ts)

            for t in ts:
                if abs(t - best_t) < 1e-12:
                    continue
                f_trial = f.copy()
                f_trial[i] = t * c
                f_trial[j] = (1 - t) * c
                val = fast_convolve(f_trial).max()
                if val < best_val - 1e-13:
                    best_val = val
                    best_t = t

            if best_t != f[i] / c and best_val < best_obj - 1e-12:
                f[i] = best_t * c
                f[j] = (1 - best_t) * c
                C = C  # Will recompute below
                best_obj = best_val
                best_f = f.copy()
                round_improved = True

        if round_improved:
            no_improve_rounds = 0
        else:
            no_improve_rounds += 1
            if no_improve_rounds >= 3:
                break

    return best_f, best_obj


def asymmetric_init(n, seed=0):
    """Create asymmetric initial sequences."""
    rng = np.random.default_rng(seed)
    inits = []

    # Left-heavy
    x = np.linspace(0, 1, n)
    a = 0.5 * x + 0.5 * x**3
    a = np.maximum(a, 0) / a.sum()
    inits.append(a)

    # Right-heavy
    a = 0.5 * (1 - x) + 0.5 * (1 - x)**3
    a = np.maximum(a, 0) / a.sum()
    inits.append(a)

    # Peak left of center
    a = np.exp(-((x - 0.3) ** 2) / 0.05)
    a = a / a.sum()
    inits.append(a)

    # Peak right of center
    a = np.exp(-((x - 0.7) ** 2) / 0.05)
    a = a / a.sum()
    inits.append(a)

    # Bimodal asymmetric
    a = np.exp(-((x - 0.2) ** 2) / 0.02) + 0.5 * np.exp(-((x - 0.8) ** 2) / 0.04)
    a = a / a.sum()
    inits.append(a)

    # Linear ramp
    a = np.linspace(0.1, 1.0, n)
    a = a / a.sum()
    inits.append(a)

    # Noise
    a = rng.standard_normal(n)
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    inits.append(a)

    return inits


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
