5. Repeat until no constraints are violated

2. Solve: max 1^T g s.t. A_S g <= M * 1_S, g >= 0
3. Check all constraints: if (A g)[k] > M for some k not in S, add it
4. Repeat

This should be much faster when only a few constraints are active.

    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x : x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_rows(f, ks):
    """Build constraint matrix rows for given peak indices ks."""
    n = len(f)
    rows = []
    for k in ks:
        row = np.zeros(n)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            row[j] = f[k - j]
        rows.append(row)
    return np.array(rows)


def solve_lp_sub(f, M, ks):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in ks, g >= 0."""
    n = len(f)
    A_ub = build_rows(f, ks)
    b_ub = np.full(len(ks), M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_cg_optimize(f0, time_limit, top_k=30, verbose=False):
    """
    Frank-Wolfe with constraint generation LP oracle.
    Maintains a set of active constraints, solves LP on active set,
    checks for violated constraints, adds most violated if needed.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    # Initial active set: top peaks
    active_ks = np.argpartition(conv, -top_k)[-top_k:].tolist()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.1:
            break
        
        # Solve LP with active constraints
        g = solve_lp_sub(f, M, active_ks)
        if g is None:
            # Fallback: try with fewer constraints
            active_ks = active_ks[:max(5, len(active_ks)//2)]
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        # Check for violated constraints
        fg = fast_convolve(f) * 0  # dummy
        # Actually compute f*g
        # (f*g)[k] = sum_j f[j]*g[k-j]
        fg_conv = np.convolve(f, g)
        violations = fg_conv - M
        if violations.max() > 1e-10:
            # Add most violated constraint
            worst = int(np.argmax(violations))
            if worst not in active_ks:
                active_ks.append(worst)
                # Keep only top-K by violation
                if len(active_ks) > top_k + 10:
                    # Keep the ones with highest violation
                    viol_arr = np.array([fg_conv[k] - M for k in active_ks])
                    top_idx = np.argpartition(viol_arr, -top_k)[-top_k:]
                    active_ks = [active_ks[i] for i in top_idx]
                continue  # Re-solve with new constraint
        
        # Check convergence
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            continue
        
        # Line search: find best alpha in [0, 1]
        # Use ternary search since objective is convex in alpha
        best_alpha = 0.0
        best_val = M
        
        # First, try a few key values
        for alpha in [0.1, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Refine with ternary search around best_alpha
        lo = max(0.0, best_alpha - 0.15)
        hi = min(1.0, best_alpha + 0.15)
        for _ in range(30):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            ft1 = (1.0 - m1) * f + m1 * g
            ft1 = np.maximum(ft1, 0.0)
            st1 = ft1.sum()
            if st1 < 1e-15:
                v1 = best_val
            else:
                ft1 = ft1 / st1
                v1 = fast_convolve(ft1).max()
            
            ft2 = (1.0 - m2) * f + m2 * g
            ft2 = np.maximum(ft2, 0.0)
            st2 = ft2.sum()
            if st2 < 1e-15:
                v2 = best_val
            else:
                ft2 = ft2 / st2
                v2 = fast_convolve(ft2).max()
            
            if v1 < v2:
                hi = m2
            else:
                lo = m1
        
        alpha_final = (lo + hi) / 2.0
        ft = (1.0 - alpha_final) * f + alpha_final * g
        ft = np.maximum(ft, 0.0)
        st = ft.sum()
        if st < 1e-15:
            no_improve += 1
            continue
        ft = ft / st
        final_val = fast_convolve(ft).max()
        
        if final_val < best_obj - 1e-15:
            f = ft
            M = final_val
            best_obj = M
            best_f = f.copy()
            no_improve = 0
            
            # Update active set
            conv_new = fast_convolve(f)
            # Keep current active set, add new top peaks if needed
            new_top = np.argpartition(conv_new, -top_k)[-top_k:]
            # Merge: keep union, trim to top_k
            all_ks = list(set(active_ks + new_top.tolist()))
            # Sort by convolution value, keep top
            all_ks.sort(key=lambda k: -conv_new[k])
            active_ks = all_ks[:top_k]
        else:
            no_improve += 1
        
        # Perturbation when stuck
        if no_improve > 80:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.002 + 0.005 * min(1.0, no_improve / 200.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            active_ks = np.argpartition(conv, -top_k)[-top_k:].tolist()
            no_improve = 0
    
    return best_f, best_obj


def peak_balance_optimize(f0, time_limit, top_k=20, verbose=False):
    """
    Peak balancing: iteratively find the top K peaks and solve
    a QP to minimize the maximum of these peaks.
    Uses the fact that at the optimum, the top peaks should be equal.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(f)
        M = conv.max()
        
        # Get top K peaks
        top_ks = np.argpartition(conv, -top_k)[-top_k:]
        
        # Subgradient descent with momentum on the top peaks
        # Gradient of max of top peaks (smoothed)
        T = max(0.001, M * 0.05)
        scaled = conv[top_ks] / T
        scaled -= scaled.max()
        weights = np.exp(scaled)
        weights /= weights.sum()
        
        # Compute gradient: weighted sum of subgradients
        grad = np.zeros(n)
        for i, k in enumerate(top_ks):
            w = weights[i]
            if w < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                # d(a*a)[k]/da[j] = 2*a[k-j] (approx)
                grad[j] += w * 2.0 * a_val_at(f, k, j)
        
        # Projected gradient step with backtracking
        lr = 0.01
        best_candidate = f
        best_val = M
        
        for trial in [lr, lr*0.5, lr*0.25, lr*0.1, lr*0.05]:
            candidate = project_simplex(f - trial * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
        
        if best_val < best_obj - 1e-14:
            f = best_candidate
            best_obj = best_val
            best_f = f.copy()
            no_improve = 0
        else:
            f = best_candidate  # still move
            no_improve += 1
        
        if no_improve > 100:
            rng = np.random.default_rng((it * 31627 + 7) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            no_improve = 0
    
    return best_f, best_obj


def a_val_at(f, k, j):
    """Get a[k-j] if valid, else 0."""
    idx = k - j
    if 0 <= idx < len(f):
        return f[idx]
    return 0.0


def coordinate_polish(f0, time_limit):
    """
    Coordinate descent: for each coordinate in the support,
    try to reduce the highest peak by adjusting that coordinate.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    sweep = 0
    
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(f)
        k_peak = int(np.argmax(conv))
        
        # Try adjusting each coordinate in the "affected" set
        # The peak at k is affected by coordinates j where 0 <= k-j < n
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        
        improved = False
        for j in range(js, je + 1):
            if f[j] < 1e-12:
                continue
            # Try decreasing f[j] and distributing to others
            # Or try a small random perturbation centered at j
            rng = np.random.default_rng((sweep * 100000 + j) % (2**31))
            pert = np.zeros(n)
            pert[j] = -0.001 * f[j]
            # Distribute to neighbors
            for d in range(-3, 4):
                jj = j + d
                if 0 <= jj < n and jj != j:
                    pert[jj] = 0.0005 * f[j] / 6.0
            
            candidate = project_simplex(f + pert)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_obj - 1e-15:
                f = candidate
                best_obj = val
                best_f = f.copy()
                improved = True
                break
        
        if not improved and sweep > 500:
            # Try broad random perturbation
            rng = np.random.default_rng((sweep * 48271) % (2**31))
            candidate = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                f = candidate
                best_obj = c.max()
                best_f = f.copy()
    
    return best_f, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse starting points."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    # Edge concentration
    x = np.arange(n, dtype=np.float64)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    
    # Two Gaussian bumps
    g = np.exp(-0.5 * ((x - 0.25*n) / (0.1*n))**2) + np.exp(-0.5 * ((x - 0.75*n) / (0.1*n))**2)
    inits.append(g / g.sum())
    
    # Cosine
    c = np.abs(np.cos(np.pi * x / n)) + 0.1
    inits.append(c / c.sum())
    
    # Father wavelet-like
    f = np.cos(2 * np.pi * x / n) * np.exp(-0.5*(x/n - 0.5)**2/0.05)
    f = np.abs(f) + 0.01
    inits.append(f / f.sum())
    
    # Interpolated parent
    if parent is not None and len(parent) != n:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    elif parent is not None:
        inits.append(parent.copy())
    
    return inits
