    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):

    rho = np.nonzero(cond)[0][-1]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase hybrid approach:
- L-BFGS on softmax-parameterized log-sum-exp with temperature annealing
- Mirror descent (multiplicative weights) with Nesterov momentum
- FW-LP with exact LP oracle for global structure
- Structured initializations and adaptive restarts
- Key insight: gradient of weighted conv = 2 * sliding_window_view(p,n).T @ a

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_softmax_weights(conv, T):
    """Compute p = softmax(conv/T) for gradient computation."""
    shifted = conv / T
    shifted -= shifted.max()
    p = np.exp(shifted)
    s = p.sum()
    if s < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = p / s
    return p


def compute_grad(a, p):
    """
    Compute gradient of sum_k p[k]*conv[k] w.r.t. a.
    grad[j] = 2 * sum_m a[m] * p[j+m]
    Uses sliding window view for efficiency.
    """
    n = len(a)
    # sliding_window_view(p, n) gives shape (2n-1-n+1, n) = (n, n)
    # Row j is p[j:j+n]
    W = np.lib.stride_tricks.sliding_window_view(p, n)
    grad = W.T @ a
    return 2.0 * grad


def logsumexp_value(conv, T):
    """Smooth approximation to max(conv)."""
    shifted = conv / T
    shifted -= shifted.max()
    return T * (shifted.max() + np.log(np.sum(np.exp(shifted))))


def softmax_obj_grad(z, T):
    """
    Objective and gradient for softmax parameterization.
    a = softmax(z), objective = logsumexp(conv(a), T)
    Returns (obj, dz) where dz is gradient w.r.t. z.
    """
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    obj = logsumexp_value(conv, T)
    p = compute_softmax_weights(conv, T)
    grad_a = compute_grad(a, p)
    # Chain rule: dz[i] = a[i] * (grad_a[i] - dot(grad_a, a))
    ga = np.dot(grad_a, a)
    dz = a * (grad_a - ga)
    return obj, dz


def lbfgs_softmax_optimize(a0, time_limit, T_values=None, verbose=False):

    """
    n = len(a0)

    a = a / a.sum()
    z = np.log(a)
    z -= z.mean()

    if T_values is None:

    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()

            obj, _ = softmax_obj_grad(z_inner, T)
            return obj

            _, g = softmax_obj_grad(z_inner, T)
            return g

                a = a_new.copy()
        except Exception:
            pass

        if verbose:
            print(f"      L-BFGS T={T:.6f}, obj={best_obj:.10f}")

    return best_a, best_obj


def mirror_descent_optimize(a0, time_limit, lr_init=0.05, T_init=0.002, verbose=False):
    """
    Mirror descent (multiplicative weights) with Nesterov momentum.
    a_new = normalize(a * exp(-lr * momentum_grad))

            break

        conv = fast_convolve(a)
        p = compute_softmax_weights(conv, T)
        grad = compute_grad(a, p)

        # Nesterov momentum
        v = momentum * v + (1 - momentum) * grad
        # Use lookahead: compute gradient at a - lr*mom*v
        a_look = np.maximum(a - lr * momentum * v, 1e-15)
        a_look = a_look / a_look.sum()
        conv_look = fast_convolve(a_look)
        p_look = compute_softmax_weights(conv_look, T)
        grad_look = compute_grad(a_look, p_look)
        v = momentum * v + (1 - momentum) * grad_look

        # Multiplicative update
        log_update = -lr * v
        log_update -= log_update.max()
        update = np.exp(log_update)
        a_new = a * update
        s = a_new.sum()
        if s < 1e-30:
            break
        a_new = a_new / s
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()

        if obj < best_obj - 1e-14:

            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

def solve_lp_oracle(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    # Build A_ub: row k has f[0..n-1-k] in columns [k..k+n-1] (shifted)
    # Actually: (f*g)[k] = sum_j f[j] * g[k-j] for valid j, k-j
    # So (f*g)[k] = sum_j f[k-j] * g[j] where 0<=j<n and 0<=k-j<n
    # For a given k: j ranges from max(0, k-n+1) to min(n-1, k)
    # A_ub[k, j] = f[k-j] if 0 <= k-j <= n-1, else 0
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g[j] appears in (f*g)[k] for k = j, j+1, ..., j+n-1
        # with coefficient f[k-j] = f[0], f[1], ..., f[n-1]
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle and adaptive M relaxation."""
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()

    conv = fast_convolve(f)

            break
        it += 1

        g = solve_lp_oracle(f, M * m_exp, n)
        if g is None:
            m_exp = min(m_exp * 1.2, 4.0)
            if m_exp > 4.0 and no_improve > 30:
                break
            continue
        m_exp = 1.0

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 40:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Refine with bisection
        if best_alpha > 0.01:
            lo = max(0.0, best_alpha - 0.03)
            hi = min(1.0, best_alpha + 0.03)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_f, best_obj


def nesterov_smoothed(a0, time_limit, T_init=0.001, lr_init=0.02, verbose=False):

    return best_a, best_obj


def generate_structured_inits(n, rng, parent=None):
    """Generate diverse structured initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    # Gaussian (various widths)
    for sigma in [0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
        g = np.exp(-0.5 * ((x - center) / (sigma * n)) ** 2)
        g = np.maximum(g, 1e-10)
        inits.append(g / g.sum())

    # Cosine
    c = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c / c.sum())

    # Two Gaussians
    for sep in [0.2, 0.3, 0.4]:
        g2 = np.exp(-0.5 * ((x - center + sep * n) / (0.12 * n)) ** 2) + \
             np.exp(-0.5 * ((x - center - sep * n) / (0.12 * n)) ** 2)
        g2 = np.maximum(g2, 1e-10)
        inits.append(g2 / g2.sum())

    # Edge concentration
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 0.5
    inits.append(edge_w / edge_w.sum())

    # Quadratic (parabolic)
    quad = 1.0 - ((x - center) / center) ** 2
    quad = np.maximum(quad, 0.01)
    inits.append(quad / quad.sum())

    # Uniform
    inits.append(np.ones(n) / n)

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    inits.append(rng.dirichlet(np.ones(n) * 2.0))

    # If parent available, resample
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 1e-10)
        inits.append(px / px.sum())

    return inits

    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic() - 200
        if remaining > 30:
            t_alloc = min(remaining, 250)
            print(f"  FW-LP n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "fw_lp_main")

        # Perturbed FW-LP restarts
        remaining = deadline - time.monotonic() - 100
        if remaining > 20:
            for scale in [0.003, 0.01]:
                remaining = deadline - time.monotonic() - 80
                if remaining < 15:
                    break
                f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
                a_opt, _ = fw_lp_optimize(f_pert, time_limit=min(remaining, 60))
                try_seq(a_opt, f"fw_pert_{scale}")

    print(f"  Phase 4 best: {best_score:.10f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 5: Final fine-tuning (~200s) ===
    print("\n=== Phase 5: Final fine-tuning ===")
    t_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine L-BFGS
        remaining = deadline - time.monotonic() - 80
        if remaining > 10:
            t_alloc = min(remaining, 80)
            T_fine = [0.0001, 0.00005, 0.00002, 0.00001]
            print(f"  Fine L-BFGS t={t_alloc:.0f}s")
            a_opt, _ = lbfgs_softmax_optimize(best_seq, time_limit=t_alloc,
                                               T_values=T_fine)
            try_seq(a_opt, "fine_lbfgs")

        # Tiny perturbations + quick optimization
        remaining = deadline - time.monotonic() - 40
        if remaining > 5:
            for _ in range(5):
                remaining = deadline - time.monotonic() - 20
                if remaining < 5:
                    break
                noise = rng.standard_normal(n_best) * 0.0005
                pert = project_simplex(best_seq + noise)
                a_opt, _ = lbfgs_softmax_optimize(pert, time_limit=min(remaining, 8),
                                                   T_values=[0.0001, 0.00005])
                try_seq(a_opt, "perturb_fine")

        # One more FW-LP
        remaining = deadline - time.monotonic() - 20
        if remaining > 10:
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=min(remaining, 30))
            try_seq(a_opt, "final_fw")

    print(f"  Phase 5 best: {best_score:.10f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
