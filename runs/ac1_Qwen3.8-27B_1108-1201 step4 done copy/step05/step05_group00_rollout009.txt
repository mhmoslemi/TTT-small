import numpy as np

def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex (sum=1, non-negative)."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(conv, a, p):
    """
    Compute gradient of sum_k p[k]*conv[k] with respect to a.
    grad[m] = 2 * sum_k p[k] * a[k-m] (where indices are valid)
    """
    n = len(a)
    # For each k, the contribution to grad[m] is p[k] * a[k-m] (if k-m valid)
    # This is a correlation: grad[m] = 2 * sum_k p[k] * a[k-m]
    # = 2 * correlate(p, a) in the appropriate sense
    # (a*a)[k] = sum_j a[j]*a[k-j], d/da[m] = 2*a[k-m] (if valid)
    # So grad[m] = sum_k p[k] * 2 * a[k-m] (if 0<=k-m<n)
    # This is 2 * (p correlated with a), i.e., 2 * sum_k p[k]*a[k-m]
    # Which is 2 * np.correlate(p, a, 'valid') if len(p) >= len(a)
    # Actually conv has length 2n-1, a has length n.
    # grad[m] = 2 * sum_{k=m}^{m+n-1} p[k] * a[k-m] for m in 0..n-1
    # This is 2 * np.correlate(p, a, mode='valid') reversed? Let me think.
    # np.correlate(p, a, 'valid') computes sum_j p[j+shift] * a[j] for appropriate shifts
    # We want: for each m, sum_k p[k] * a[k-m] where k ranges over valid range
    # Let l = k-m, then k = l+m, range of l is [0, n-1] (since a is length n)
    # And k = l+m must be in [0, 2n-2]
    # So grad[m] = 2 * sum_{l=0}^{n-1} p[l+m] * a[l]
    # This is 2 * np.correlate(p, a, mode='valid')[m] if p is the longer array
    # np.correlate(p, a, 'valid') gives for offset m: sum_l p[m+l]*a[l]
    # Yes! That's exactly what we want.
    grad = 2.0 * np.correlate(p, a, mode='valid')
    return grad


def smoothed_gradient_step(a, T, lr, use_momentum=False, v=None, momentum=0.9):
    """
    One step of smoothed gradient descent with backtracking line search.
    Returns updated a, new objective, and updated momentum vector.
    """
    n = len(a)
    conv = fast_convolve(a)
    
    # Log-sum-exp smoothing (numerically stable)
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros_like(conv)
        p[np.argmax(conv)] = 1.0
    else:
        p = p / p_sum
    
    grad = compute_gradient(conv, a, p)
    
    if use_momentum and v is not None:
        v = momentum * v - lr * grad
    else:
        v = -lr * grad
    
    # Backtracking line search
    current_obj = conv.max()
    best_a = a.copy()
    best_obj = current_obj
    
    step = 1.0
    for _ in range(30):
        candidate = project_simplex(a + step * v)
        c = fast_convolve(candidate)
        val = c.max()
        if val < current_obj - 1e-15:
            best_a = candidate
            best_obj = val
            break
        step *= 0.5
    
    return best_a, best_obj, v


def coordinate_descent_step(a, i, conv):

    One coordinate descent step for variable i.
    Uses the 'rescale others' parameterization.
    Returns improved a or original if no improvement found.
    """
    n = len(a)
    a_i = a[i]
    s = 1.0 - a_i  # sum of others
    
    if s < 1e-15:
        return a
    
    # Peaks affected by changing a[i]: k in [i, i+n-1] intersect [0, 2n-2]
    k_start = max(0, i)
    k_end = min(2 * n - 2, i + n - 1)
    
    # Current max among affected peaks
    current_max = conv.max()
    
    # Try a grid of values for t = new a[i]
    # Center around current value
    best_t = a_i
    best_max = current_max
    
    # Generate candidate values
    candidates = [a_i]
    # Relative perturbations
    for delta in [0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
        candidates.append(a_i * (1 + delta))
        candidates.append(a_i * (1 - delta))
    # Absolute perturbations
    for delta in [0.001, 0.002, 0.005, 0.01, 0.02, 0.05]:
        candidates.append(a_i + delta)
        candidates.append(a_i - delta)
    # Some extreme values
    candidates.append(0.0)
    candidates.append(min(0.99, a_i * 3))
    
    t_lo = 0.0
    t_hi = min(1.0 - 1e-10, 0.99 * n / n)  # Can't exceed 1
    
    # Quick ternary-like search
    # First find the bracket
    for t in candidates:
        t = max(0.0, min(1.0 - 1e-10, t))
        if abs(t - a_i) < 1e-12:
            continue
        
        # Build new sequence
        scale = (1.0 - t) / s
        new_a = a * scale
        new_a[i] = t
        
        # Recompute only affected peaks
        # (new_a * new_a)[k] for k in [k_start, k_end]
        # = sum_j new_a[j] * new_a[k-j]
        # For efficiency, compute directly
        improved = True
        for k in range(k_start, k_end + 1):
            # (new_a*new_a)[k] = scale^2 * (a*a)[k] + contributions from a[i]
            # Actually let's just compute it directly for the affected range
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            val = 0.0
            for j in range(js, je + 1):
                val += new_a[j] * new_a[k - j]
            if val > best_max + 1e-12:
                improved = False
                break
        
        if improved and val <= best_max:
            # This is promising, but we need to check ALL peaks
            # For safety, compute full convolution
            full_conv = fast_convolve(new_a)
            actual_max = full_conv.max()
            if actual_max < best_max - 1e-14:
                best_max = actual_max
                best_t = t
    
    if best_t != a_i and best_t > 0:
        scale = (1.0 - best_t) / s
        new_a = a * scale
        new_a[i] = best_t
        return new_a
    return a


def coordinate_descent_full(a, n_rounds=3, order='random', rng=None):
    """Full cycle of coordinate descent."""
    n = len(a)
    for _ in range(n_rounds):
        if order == 'random' and rng is not None:
            perm = rng.permutation(n)
        else:
            perm = range(n)
        for i in perm:
            conv = fast_convolve(a)
            a = coordinate_descent_step(a, i, conv)
    return a

def fw_lp_optimize(a0, time_limit, verbose=False):

            break
        
        # Solve LP: max sum(g) s.t. (a*g)[k] <= M*m_exp for all k, g >= 0
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = a
        b_ub = np.full(2 * n - 1, M * m_exp)
        bounds = [(0, None)] * n
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': max(1.0, remaining * 0.3)})
            if res.success:
                g = np.maximum(res.x, 0.0)
            else:
                g = None
        except Exception:
            g = None
        
        if g is None:
            m_exp *= 1.3

            continue

            continue
        
        # Line search

                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

            conv = fast_convolve(a)

def nesterov_gradient_descent(a0, time_limit, T_init=1e-4, lr_init=0.1):
    """Nesterov accelerated gradient descent with adaptive parameters."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    T = T_init
    lr = lr_init
    v = np.zeros(n)
    y = a.copy()  # Nesterov lookahead
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    beta = 0.9  # Momentum
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.01:
            break
        
        # Nesterov lookahead
        y = a + beta * (a - (a - v))  # Simplified: y = a + beta*v
        
        conv_y = fast_convolve(y)
        scaled = conv_y / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum > 1e-300:
            p = p / p_sum
        else:
            p = np.zeros_like(conv_y)
            p[np.argmax(conv_y)] = 1.0
        
        grad = compute_gradient(conv_y, y, p)
        
        # Update
        v = beta * v - lr * grad
        a_new = project_simplex(a + v)
        
        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            a = a_new  # Still update to maintain momentum
            no_improve += 1
        
        # Adapt
        if no_improve > 30:
            T = max(T * 0.95, 1e-8)
            lr *= 0.95
        if no_improve > 100:
            beta = max(beta * 0.99, 0.7)
        if no_improve > 300:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise = 0.001 + 0.003 * min(1.0, (no_improve - 300) / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            T = T_init
            lr = lr_init
            beta = 0.9
            no_improve = 0
    
    return best_a, best_obj

def propose_candidate(seed=42, budget_s=1800):

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if parent is None:
        parent = np.ones(768) / 768
        try_seq(parent, "uniform")
    
    n_parent = len(parent)
    
    # === Phase 1: FW-LP (~350s) ===
    print("\n=== Phase 1: FW-LP ===")
    t_start = time.monotonic()
    
    remaining = deadline - time.monotonic() - 1100
    if remaining > 30:
        t_alloc = min(remaining, 300)
        print(f"  FW-LP n={n_parent}, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(parent, time_limit=t_alloc)
        try_seq(a_opt, "fw_lp")
    
    # Perturbed FW-LP attempts
    for scale in [0.005, 0.015]:
        remaining = deadline - time.monotonic() - 1050
        if remaining < 20:
            break
        f_pert = project_simplex(best_seq + rng.standard_normal(n_parent) * scale)
        t_alloc = min(remaining, 80)
        a_opt, _ = fw_lp_optimize(f_pert, time_limit=t_alloc)
        try_seq(a_opt, f"fw_pert_{scale}")
    
    print(f"  Phase 1 best: {best_score:.8f} ({time.monotonic()-t_start:.0f}s)")
    
    # === Phase 2: Active-set gradient (~500s) ===
    print("\n=== Phase 2: Active-Set Gradient ===")
    t_start = time.monotonic()
    
    # Multiple temperature settings
    for T_init in [1e-5, 1e-4, 1e-3]:
        remaining = deadline - time.monotonic() - 550
        if remaining < 15:
            break
        t_alloc = min(remaining, 150)
        print(f"  Active-set T={T_init}, t={t_alloc:.0f}s")
        a_opt, _ = active_set_optimize(best_seq, time_limit=t_alloc, T_init=T_init)
        try_seq(a_opt, f"active_T{T_init}")
    
    print(f"  Phase 2 best: {best_score:.8f} ({time.monotonic()-t_start:.0f}s)")
    
    # === Phase 3: Nesterov + Coordinate (~450s) ===
    print("\n=== Phase 3: Nesterov + Coordinate ===")
    t_start = time.monotonic()
    
    # Nesterov gradient
    for lr_init in [0.05, 0.1, 0.2]:
        remaining = deadline - time.monotonic() - 300
        if remaining < 15:
            break
        t_alloc = min(remaining, 120)
        print(f"  Nesterov lr={lr_init}, t={t_alloc:.0f}s")
        a_opt, _ = nesterov_gradient_descent(best_seq, time_limit=t_alloc, 
                                             T_init=1e-5, lr_init=lr_init)
        try_seq(a_opt, f"nesterov_{lr_init}")
    
    # Coordinate descent
    remaining = deadline - time.monotonic() - 200
    if remaining > 10:
        t_alloc = min(remaining, 80)
        print(f"  Coordinate descent, t={t_alloc:.0f}s")
        # Run coordinate descent for a fixed number of iterations
        n = len(best_seq)
        a_cd = best_seq.copy()
        t_cd_start = time.monotonic()
        rounds = 0
        while time.monotonic() - t_cd_start < t_alloc:
            perm = rng.permutation(n)
            for i in perm:
                if time.monotonic() - t_cd_start > t_alloc:
                    break
                conv = fast_convolve(a_cd)
                a_cd = coordinate_descent_step(a_cd, i, conv)
            rounds += 1
            if rounds % 2 == 0:
                try_seq(a_cd, f"cd_r{rounds}")
        try_seq(a_cd, "cd_final")
    
    print(f"  Phase 3 best: {best_score:.8f} ({time.monotonic()-t_start:.0f}s)")
    
    # === Phase 4: Pair coordinate + final polish (~250s) ===
    print("\n=== Phase 4: Pair Coord + Final Polish ===")
    t_start = time.monotonic()
    
    # Pair coordinate optimization
    remaining = deadline - time.monotonic() - 80
    if remaining > 20:
        t_alloc = min(remaining, 120)
        print(f"  Pair coord, t={t_alloc:.0f}s")
        a_opt = pair_coordinate_optimize(best_seq, time_limit=t_alloc, rng=rng)
        try_seq(a_opt, "pair_coord")
    
    # Final active-set with very low temperature
    remaining = deadline - time.monotonic() - 40
    if remaining > 10:
        t_alloc = min(remaining, 60)
        print(f"  Final active-set T=1e-6, t={t_alloc:.0f}s")
        a_opt, _ = active_set_optimize(best_seq, time_limit=t_alloc, T_init=1e-6)
        try_seq(a_opt, "final_active")
    
    # Tiny perturbations with quick Nesterov
    remaining = deadline - time.monotonic() - 15
    if remaining > 5:
        for idx in range(5):
            remaining = deadline - time.monotonic() - 5
            if remaining < 3:
                break
            noise = rng.standard_normal(len(best_seq)) * 0.0005
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = nesterov_gradient_descent(f_pert, time_limit=min(remaining, 8),
                                                 T_init=1e-6, lr_init=0.05)
            try_seq(a_opt, f"final_pert_{idx}")
    
    print(f"  Phase 4 best: {best_score:.8f} ({time.monotonic()-t_start:.0f}s)")
    
    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
