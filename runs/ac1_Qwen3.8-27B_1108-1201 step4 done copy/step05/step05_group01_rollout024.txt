import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fast_cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = max(len(a), len(b))
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def gradient_smoothed(a, conv, temp):
    """
    Compute gradient of temp * log(sum_k exp(conv[k]/temp)) w.r.t. a.
    gradient[m] = 2 * sum_{k} w[k] * a[k-m] where 0<=k-m<n
    = 2 * (conv(w, a_flip))[n-1 : 2n-1]
    """
    n = len(a)
    scaled = conv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    
    a_flip = a[::-1].copy()

    # = 2 * (w * a_flip)[m + n - 1]

    gradient = 2.0 * cross[n-1 : 2*n-1]
    
    return gradient

    it = 0
    no_improve = 0
    tau = tau0

            no_improve = 0

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)

        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def gradient_smoothed(a, conv, temp):
    """
    Compute gradient of temp * log(sum_k exp(conv[k]/temp)) w.r.t. a.
    Uses FFT for O(n log n) computation.
    gradient[m] = 2 * sum_{j=0}^{n-1} w[m+j] * a[j] = 2 * (w * a_flip)[m+n-1]
    """
    n = len(a)
    scaled = conv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    a_flip = a[::-1].copy()
    cross = fast_cross_convolve(w, a_flip)
    gradient = 2.0 * cross[n - 1: 2 * n - 1]
    return gradient

def build_lp_matrix(a, active_lags):
    """Build constraint matrix rows for given active lags.
    A[k, m] = a[k-m] if 0 <= k-m <= n-1, else 0.
    """
    n = len(a)
    K = len(active_lags)
    A = np.zeros((K, n))

            A[i, m_lo:m_hi + 1] = a[k - (m_hi):k - m_lo - 1:-1] if m_lo < m_hi else a[k - m_lo]
    return A

            any_improved = False
            conv = fast_convolve(a)
            for j in range(0, n, max(1, n // 50)):
                if time.monotonic() - t0 > time_limit:
                    break
                old = a[j]
                if old < 1e-12:
                    continue
                denom = 1.0 - old
                if denom < 1e-12:
                    continue
                candidates = [old * 0.5, old * 0.7, old * 1.3, old * 1.5, 0.0]
                best_x = old
                best_val = fast_convolve(a).max()
                for x in candidates:
                    if x < 0 or x >= 1.0:
                        continue
                    a_t = a.copy()
                    a_t[j] = x
                    scale = (1.0 - x) / denom
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a_t[mask] *= scale
                    a_t = np.maximum(a_t, 0.0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    val = fast_convolve(a_t / st).max()
                    if val < best_val - 1e-14:
                        best_val = val
                        best_x = x
                if best_x != old:
                    a[j] = best_x
                    scale = (1.0 - best_x) / denom
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    any_improved = True
                    new_obj = fast_convolve(a).max()
                    if new_obj < best_obj:
                        best_obj = new_obj
                        best_a = a.copy()
            if not any_improved:
                break

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def cosine_init(n, freq, amp, phase=0.0):
    """Cosine-modulated initialization."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:

    if parent is None:
        parent = np.ones(768) / 768
        best_seq = parent.copy()

            [float(max(0.0, min(1000.0, x))) for x in best_seq]))
