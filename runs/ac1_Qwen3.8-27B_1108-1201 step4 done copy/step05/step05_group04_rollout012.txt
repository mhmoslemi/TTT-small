    cross = np.fft.irfft(W * F_rev, fft_size)
    
    # cross[k] = sum_j w[j] * f[n-1-(k-j)] = sum_j w[j] * f[j-n+1+k]
    # Let me verify: (w * f_rev)[k] = sum_j w[j] * f_rev[k-j] = sum_j w[j] * f[n-1-k+j]
    # I want: sum_j w[m+j] * f[j] (part 1)
    # Let i = m+j, j = i-m: sum_i w[i]*f[i-m] for i in [m, m+n-1]
    # = (w * f_rev)[m + n - 1] ... let me just compute directly for clarity
    
    # Actually let me just do it with a loop over the non-zero weights for correctness
    # For large n this is O(n^2) in the worst case but the weight vector is often sparse
    # Let's use the FFT approach properly
    
    # Part 1: g1[m] = sum_{j=0}^{n-1} w[m+j] * f[j], valid when m+j < conv_len
    # This is a cross-correlation: g1 = (w cross f) 
    # g1[m] = sum_j w[j+m]*f[j]
    # = (flip(w) * f)[len(w)-1-m] ... this is getting confusing
    
    # Let me use a cleaner FFT approach
    # g1[m] = sum_{j=0}^{n-1} w[m+j]*f[j]
    # Extend f to length conv_len: f_ext[j] = f[j] for j < n, 0 otherwise
    # g1[m] = sum_j w[m+j]*f_ext[j]
    # This is the cross-correlation of w and f_ext, evaluated at lag m
    # = (w * flip(f_ext))[len(w)-1-m + len(f_ext)-1]... 
    
    # Simplest: just do it directly with FFT
    # g1[m] = sum_j w[j+m]*f[j] for j=0..n-1, j+m < 2n-1
    # = (circular cross-correlation) but with linear boundaries
    
    # Let me just use the direct formula with FFT:
    # g1 = ifft(fft(w) * conj(fft(f_padded)))
    f_pad = np.zeros(fft_size)
    f_pad[:n] = f
    F = np.fft.rfft(f_pad)
    W_full = np.fft.rfft(np.zeros(fft_size))  # placeholder
    # Actually W is already computed above with length fft_size
    # ifft(fft(w)*conj(fft(f))) gives cross-correlation
    cross1 = np.fft.irfft(W * np.conj(F), fft_size)
    # cross1[m] = sum_j w[j]*f_pad[j+m] (for appropriate wrapping, but since we zero-padded, it's linear)
    # Wait: ifft(fft(w)*conj(fft(f)))_m = sum_j f[j]*w[j+m] (circular)
    # Since f is zero-padded to fft_size >= conv_len + n, no wrapping issues for m < n
    # cross1[m] = sum_j f[j] * w[j+m] (lin

ear) = g1[m] ✓
    
    # Part 2: g2[m] = sum_{j=0}^{n-1} w[m-j]*f[j] for m-j >= 0, i.e., j <= m
    # = sum_{j=0}^{m} w[m-j]*f[j]
    # Let i = m-j: sum_i w[i]*f[m-i] for i=0..m
    # = (w * f)[m] (convolution!)
    # g2[m] = (w * f)[m] = ifft(fft(w)*fft(f))
    conv_wf = np.fft.irfft(W * F, fft_size)
    # conv_wf[m] = sum_j w[j]*f[m-j] = g2[m] ✓
    
    grad = cross1[:n] + conv_wf[:n]
    return grad


def lse_mirror_descent(f0, time_limit, beta_start=30.0, beta_end=20000.0, 
                       lr_init=None, max_peaks=200, verbose=False):
    """Mirror descent (exponentiated gradient) with LSE smoothing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    if lr_init is None:
        lr_init = 5.0 / n
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    best_params = seq_to_softmax_params(f)
    
    beta = beta_start
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    momentum = 0.85
    v = np.zeros(n)  # velocity in parameter space
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()
        
        # Compute gradient of LSE objective w.r.t. f
        grad_f = conv_gradient(f, w)
        
        # Convert to parameter space gradient
        # d/d theta_k = f_k * (grad_f_k - sum_j grad_f_j * f_j)
        mean_g = np.dot(grad_f, f)
        grad_theta = f * (grad_f - mean_g)
        
        grad_norm = np.linalg.norm(grad_theta)
        if grad_norm < 1e-20:
            no_improve += 100
            break
        grad_theta /= grad_norm
        
        # Momentum step in parameter space
        v = momentum * v - lr * grad_theta
        
        # Update parameters
        theta = best_params + v
        # Actually, let's track current params
        # Better: keep current params and update them
        # Let me restructure: keep theta as the state
        if it == 1:
            theta = seq_to_softmax_params(f)
        theta_new = theta + v
        f_new = softmax_params_to_seq(theta_new)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            best_params = theta_new.copy()
            f = f_new
            theta = theta_new
            no_improve = 0
            lr = min(lr * 1.005, 20.0 / n)
            if verbose and it % 500 == 0:
                print(f"    MD it={it}, obj={obj_new:.10f}, beta={beta:.1f}")
        else:
            no_improve += 1
            if no_improve >= 10:
                lr *= 0.94
                momentum = max(0.75, momentum - 0.003)
            if no_improve > 80:
                # Restart from best
                theta = best_params.copy()
                v = np.zeros(n)
                f = f.copy()
                lr = max(lr_init * 0.5, 0.001)
                beta = min(beta * 1.5, beta_end)
                momentum = 0.82
                no_improve = 0
        
        # Adaptive beta increase
        if beta < beta_end:
            beta *= 1.005
            if no_improve > 30:
                beta *= 1.02  # accelerate beta growth when stuck
    
    return best_f, best_obj


def fw_lp(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        A_cache = build_lp_matrix(f)
        g = None
        try:
            c = -np.ones(n)
            b_ub = np.full(2 * n - 1, M * m_exp)
            bounds = [(0.0, None)] * n
            res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': 20})
            if res.success and res.x is not None:
                g = np.maximum(res.x, 0.0)
        except Exception:
            pass
        
        if g is None:
            m_exp = min(m_exp * 1.15, 8.0)
            continue
        
        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs

        if np.max(np.abs(g - f)) < 1e-13:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.05, 5.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = np.maximum((1.0 - alpha) * f + alpha * g, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft /= st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
            f /= f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 8.0)

        if no_improve > 40:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.003 + 0.003 * min(1.0, (no_improve - 40) / 50.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def peak_targeting(f0, time_limit, n_peaks=10, lr_init=None, momentum=0.92):
    """Direct subgradient method targeting top peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    if lr_init is None:
        lr_init = 0.004
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    v = np.zeros(n)
    cur_mom = momentum
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        conv_len = 2 * n - 1
        
        k_top = min(n_peaks, conv_len)
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peaks]
        
        # Soft weighting emphasizing highest peaks
        scaled = pv - pv.min()
        w = np.exp(scaled - scaled.max())
        w /= w.sum()
        
        w_full = np.zeros(conv_len)
        w_full[peaks] = w
        
        grad_f = conv_gradient(f, w_full)
        grad_norm = np.linalg.norm(grad_f)
        if grad_norm < 1e-20:
            break
        grad_f /= grad_norm
        
        v = cur_mom * v - cur_lr * grad_f
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                cur_lr *= 0.95
            if no_improve > 300:
                rng = np.random.default_rng((it * 13 + 7) % 999983)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                v = np.zeros(n)
                cur_lr = max(lr_init * 0.3, 0.0001)
                no_improve = 0

    return best_f, best_obj


def try_sequence(a, best_tracker):
    """Evaluate and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.10f} (n={len(a)})")
    return score


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    return f_new


def gen_parametric_inits(n, parent=None):
    """Generate diverse parametric initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    t = x / (n - 1) if n > 1 else np.array([0.0])
    
    # U-shaped: (2t-1)^2
    for alpha in [0.3, 0.5, 0.7, 1.0]:
        f = 1.0 + alpha * (2 * t - 1) ** 2
        f = np.maximum(f, 1e-10)
        f /= f.sum()
        inits.append(f)
    
    # Cosine modulation
    for alpha in [0.2, 0.4, 0.6]:
        f = 1.0 - alpha * np.cos(2 * np.pi * t)
        f = np.maximum(f, 1e-10)
        f /= f.sum()
        inits.append(f)
    
    # Inverted Gaussian (U-shaped)
    for sigma in [0.15, 0.25, 0.35]:
        f = 1.0 - 0.5 * np.exp(-(t - 0.5) ** 2 / (2 * sigma ** 2))
        f = np.maximum(f, 1e-10)
        f /= f.sum()
        inits.append(f)
    
    # Raised cosine (bell) - for contrast
    f = 0.5 * (1 + np.cos(2 * np.pi * t))
    f = np.maximum(f, 1e-10)
    f /= f.sum()
    inits.append(f)
    
    # Sinusoidal U-shape
    f = np.sin(np.pi * t) ** 2  # bell
    f = np.maximum(f, 1e-10)
    f /= f.sum()
    inits.append(f)
    
    # Anti-bell
    f = 1.0 - 0.7 * np.sin(np.pi * t) ** 2
    f = np.maximum(f, 1e-10)
    f /= f.sum()
    inits.append(f)
    
    # Spectral design: flat-top spectrum
    # We want |F(omega)|^2 to be as flat as possible
    # Try: f = |DTF of ideal lowpass| but that gives sinc...
    # Better: try sequences with Approximately Constant Amplitude
    
    # Blackman-Harris-like window (inverted for U-shape)
    w = (1.0 - 0.42 * np.cos(2*np.pi*t) + 0.21 * np.cos(4*np.pi*t) 
         - 0.07 * np.cos(6*np.pi*t) + 0.02 * np.cos(8*np.pi*t))
    w = np.maximum(w, 1e-10)
    w /= w.sum()
    inits.append(w)
    
    # Inverted Blackman
    w_inv = 1.0 - w / w.max()
    w_inv = np.maximum(w_inv, 1e-10)
    w_inv /= w_inv.sum()
    inits.append(w_inv)
    
    # Parent
    if parent is not None:
        p = np.maximum(parent, 0.0)
        s = p.sum()
        if s > 1e-15:
            inits.append(p / s)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    return inits


def spectral_init(n, num_tries=50):
    """Generate initializations by random spectral shaping."""
    rng = np.random.default_rng(42)
    best_f = None
    best_obj = float('inf')
    
    for _ in range(num_tries):
        # Random amplitude spectrum with smooth shape
        freqs = np.arange(n)
        # Create smooth random spectrum
        n_harmonics = rng.integers(2, 8)
        amps = rng.uniform(0.3, 1.0, n_harmonics)
        phases = rng.uniform(0, 2 * np.pi, n_harmonics)
        
        F = np.zeros(n, dtype=np.complex128)
        F[0] = n * 0.5  # DC component for non-negativity
        
        for i in range(n_harmonics):
            k = (i + 1) * max(1, n // (2 * n_harmonics))
            if k < n:
                F[k] = amps[i] * np.exp(1j * phases[i])
                F[n - k] = amps[i] * np.exp(-1j * phases[i])
        
        f = np.real(np.fft.ifft(F))
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s < 1e-10:
            continue
        f /= s
        
        obj = fast_convolve(f).max()
        if obj < best_obj:
            best_obj = obj
            best_f = f.copy()
    
    return best_f if best_f is not None else np.ones(n) / n

                break
            f_init = interpolate_to_length(best_tracker[1], n_try)
            print(f"  Trying n={n_try}, t={t_alloc:.0f}s")
            f_opt, _ = lse_mirror_descent(f_init, time_limit=t_alloc,
                                           beta_start=50.0, beta_end=20000.0,
                                           lr_init=6.0/n_try)
            try_sequence(f_opt, best_tracker)
    
    # === Final: one more peak targeting squeeze ===
    current = best_tracker[1] if best_tracker[1] is not None else parent
    t_remaining = deadline - time.monotonic() - 5
    if t_remaining > 20 and current is not None:
        print(f"\n=== Final squeeze: t={t_remaining:.0f}s ===")
        f_opt, _ = peak_targeting(current, time_limit=t_remaining,
                                   n_peaks=3, lr_init=0.0005, momentum=0.97)
        try_sequence(f_opt, best_tracker)
    
    # Final verification
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]
    
    # Fallback
    print("WARNING: Using fallback")
    fallback = np.ones(448) / 448
    return [float(x) for x in fallback]
