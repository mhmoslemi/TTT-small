    fft_size = 1 << (size - 1).bit_length()

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def mm_lag_step(a, lag, n_iters=50, lr=0.5):
    """Minimize conv[lag] over the simplex via projected gradient."""
    n = len(a)
    a = a.copy()
    i_start = max(0, lag - n + 1)
    i_end = min(n - 1, lag)
    for _ in range(n_iters):
        grad = np.zeros(n)
        for i in range(i_start, i_end + 1):
            grad[i] = 2.0 * a[lag - i]
        a = project_simplex(a - lr * grad)
    return a


def mm_alternate_step(a, top_m=5, n_iters_per_lag=30):
    """Alternate MM minimization over top-m lags."""
    n = len(a)
    conv = fast_convolve(a)
    if len(conv) <= top_m:
        lags = np.argsort(conv)[::-1]
    else:
        lags = np.argpartition(conv, -top_m)[-top_m:]
        lags = lags[np.argsort(conv[lags])[::-1]]
    for k in lags:
        a = mm_lag_step(a, int(k), n_iters=n_iters_per_lag, lr=0.45)
    return a


def smooth_gradient_step(a, T, top_k, lr):
    """One step of smoothed gradient descent on the top-k peak objective."""
    n = len(a)
    conv = fast_convolve(a)
    k_actual = min(top_k, 2 * n - 1)
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = (pv - pv.max()) / max(T, 1e-12)
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w = w / ws
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            grad[i] += 2.0 * w[idx] * a[k - i]
    return a - lr * grad, conv.max()


def smooth_optimize_nesterov(f0, time_limit, verbose=False):
    """Smoothed gradient descent with Nesterov momentum and adaptive parameters."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()

        f = f / s
    n = len(f)

    best_f = f.copy()

    base_lr = 0.006

    T_start = 8.0 / n
    T_end = 0.3 / n
    top_k_start = max(100, n)
    top_k_end = max(10, n // 6)

    max_iters = max(1, int(time_limit / 0.005))

        # Cosine annealing for temperature
        T = T_end + 0.5 * (T_start - T_end) * (1 + np.cos(np.pi * progress))
        top_k = max(5, int(top_k_start + (top_k_end - top_k_start) * progress))
        lr = base_lr * (1.0 - 0.4 * progress)
        # Nesterov momentum schedule
        beta_t = beta * (1.0 - 0.3 * progress)

        f_nest = f + beta_t * m
        new_f, cur_M = smooth_gradient_step(f_nest, T, top_k, lr)
        new_f = project_simplex(np.maximum(new_f, 0.0))
        m = new_f - f
        f = new_f

        if new_M < best_obj - 1e-15:
            best_obj = new_M
            best_f = f.copy()

            m = np.zeros(n)

def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        rows = np.arange(j, j + n)
        A_ub[rows, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def cosine_init(n):
    """Cosine-shaped initialization."""
    x = np.linspace(0, np.pi, n)
    a = np.cos(x / 2) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def gaussian_init(n, sigma_frac=0.3):
    """Gaussian-shaped initialization."""
    x = np.arange(n) - (n - 1) / 2.0
    sigma = n * sigma_frac
    a = np.exp(-x ** 2 / (2 * sigma ** 2))
    s = a.sum()
    a /= s
    return a


def random_init(n, seed=0):
    """Random simplex point."""
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    a /= a.sum()
    return a


def mixed_init(n, seed=0):
    """Mix of cosine and random."""
    rng = np.random.default_rng(seed)
    a = 0.5 * cosine_init(n) + 0.5 * random_init(n, seed)
    a = project_simplex(a)
    return a

        t_alloc = min(rem * 0.15, 120)
        if t_alloc > 15:
            print(f"  MM n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = mm_optimize(f_interp, time_limit=t_alloc)
            try_seq(a_opt, f"mm_n{target_n}")

        # Smooth
        rem = deadline - time.monotonic()
        t_alloc = min(rem * 0.15, 100)
        if t_alloc > 15:
            print(f"  Smooth n={target_n}, t={t_alloc:.0f}s")
            a_opt2, _ = smooth_optimize_nesterov(a_opt, time_limit=t_alloc)
            try_seq(a_opt2, f"smooth_n{target_n}")

    # === Phase 5: Diverse starts ===
    print(f"\n=== Phase 5: Diverse starts (best={best_score:.8f}) ===")
    n_best = len(best_seq) if best_seq is not None else n_parent

    diverse_starts = []
    # Cosine
    diverse_starts.append(("cos", cosine_init(n_best)))
    # Gaussian
    diverse_starts.append(("gauss", gaussian_init(n_best, 0.25)))
    # Random
    diverse_starts.append(("rand", random_init(n_best, seed=123)))
    # Mixed
    diverse_starts.append(("mixed", mixed_init(n_best, seed=456)))
    # Noisy parent
    rng_d = np.random.default_rng(789)
    diverse_starts.append(("noisy", project_simplex(best_seq + rng_d.standard_normal(n_best) * 0.01)))

    for name, f_init in diverse_starts:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_alloc = min(rem * 0.1, 100)
        if t_alloc < 15:
            break

        # Alternate between MM and smooth
        if name in ("cos", "gauss", "rand"):
            a_opt, _ = mm_optimize(f_init, time_limit=t_alloc * 0.6)
            rem2 = deadline - time.monotonic()
            if rem2 > 10:
                a_opt, _ = smooth_optimize_nesterov(a_opt, time_limit=min(rem2, t_alloc * 0.4))
        else:
            a_opt, _ = smooth_optimize_nesterov(f_init, time_limit=t_alloc * 0.6)
            rem2 = deadline - time.monotonic()
            if rem2 > 10:
                a_opt, _ = mm_optimize(a_opt, time_limit=min(rem2, t_alloc * 0.4))
        try_seq(a_opt, f"diverse_{name}")

    # === Phase 6: Intensive MM + smooth alternation on best ===
    print(f"\n=== Phase 6: Intensive MM+Smooth (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t6 = min(rem * 0.25, 300)
        if t6 > 20:
            print(f"  Time budget: {t6:.0f}s")
            f_work = best_seq.copy()
            sub_best = f_work.copy()
            sub_best_obj = fast_convolve(f_work).max()
            t0 = time.monotonic()

            while time.monotonic() - t0 < t6:
                # MM phase
                mm_t = min((t6 - (time.monotonic() - t0)) * 0.5, 60)
                if mm_t > 5:
                    f_work, _ = mm_optimize(f_work, time_limit=mm_t)
                    obj = fast_convolve(f_work).max()
                    if obj < sub_best_obj:
                        sub_best_obj = obj
                        sub_best = f_work.copy()

                # Smooth phase
                rem_inner = t6 - (time.monotonic() - t0)
                sm_t = min(rem_inner * 0.5, 60)
                if sm_t > 5:
                    f_work, _ = smooth_optimize_nesterov(f_work, time_limit=sm_t)
                    obj = fast_convolve(f_work).max()
                    if obj < sub_best_obj:
                        sub_best_obj = obj
                        sub_best = f_work.copy()

            try_seq(sub_best, "mm_sm_alt")

    # === Phase 7: LP-FW on best ===
    print(f"\n=== Phase 7: LP-FW polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t7 = min(rem * 0.2, 200)
        if t7 > 15:
            print(f"  Time budget: {t7:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t7, verbose=True)
            try_seq(a_opt, "lpfw_p7")

    # === Phase 8: Pairwise optimization ===
    print(f"\n=== Phase 8: Pairwise (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t8 = min(rem * 0.2, 200)
        if t8 > 15:
            print(f"  Time budget: {t8:.0f}s, n={len(best_seq)}")
            a_opt, _ = pairwise_optimize(best_seq, time_limit=t8, verbose=True)
            try_seq(a_opt, "pairwise")

            # Quick smooth after pairwise
            rem = deadline - time.monotonic()
            if rem > 15:
                a_opt, _ = smooth_optimize_nesterov(a_opt, time_limit=min(rem * 0.5, 60))
                try_seq(a_opt, "smooth_after_pw")

    # === Phase 9: Final MM polish ===
    print(f"\n=== Phase 9: Final MM (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t9 = min(rem * 0.15, 100)
        if t9 > 10:
            a_opt, _ = mm_optimize(best_seq, time_limit=t9, verbose=True)
            try_seq(a_opt, "mm_final")

    # === Phase 10: Final perturbations ===
    print(f"\n=== Phase 10: Final perturbations (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_b = len(best_seq)
        for trial in range(6):
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            rng = np.random.default_rng(1000 + trial * 7)
            noise_scale = 0.002 + 0.001 * trial
            f_pert = project_simplex(best_seq + rng.standard_normal(n_b) * noise_scale)
            t_alloc = min(rem * 0.3, 40)
            if t_alloc < 8:
                break
            # Quick MM + smooth
            f_pert, _ = mm_optimize(f_pert, time_limit=t_alloc * 0.5)
            rem2 = deadline - time.monotonic()
            if rem2 > 5:
                f_pert, _ = smooth_optimize_nesterov(f_pert, time_limit=min(rem2, t_alloc * 0.4))
            try_seq(f_pert, f"perturb_{trial}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
