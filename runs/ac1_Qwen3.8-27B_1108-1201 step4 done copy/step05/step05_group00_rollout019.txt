                    improved = True
                    break

            if not improved and it % 100 == 0:
                # Larger jump
                a = project_simplex(best_a + rng2.standard_normal(n) * 0.01)

    return best_a, best_obj


def smoothed_grad(a0, time_limit, T_init=0.001, lr_init=0.03):
    """Smoothed gradient descent with adaptive temperature."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)
    mom = 0.9

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        grad = 2.0 * hankel_mv(p, a)
        v = mom * v - lr * grad
        a_new = project_simplex(a + v)

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            T = max(0.00005, T * 0.98)
            lr = min(lr * 1.01, 0.1)
        else:
            no_improve += 1
            if no_improve > 20:
                T *= 1.05
                lr *= 0.95
            if no_improve > 50:
                T *= 0.8
            if no_improve > 150:
                rng = np.random.default_rng((it * 48271 + 3) % (2**31))
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                v = np.zeros(n)
                T = T_init
                lr = lr_init
                no_improve = 0

    return best_a, best_obj


def fw_lp_optimize(f0, time_limit):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        # LP oracle: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M * m_exp)
        bounds = [(0, None)] * n

        g = None
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': max(1, (time_limit - (time.monotonic() - t0)) * 0.3)})
            if res.success:
                g = np.maximum(res.x, 0.0)
        except Exception:
            pass

        if g is None or g.sum() < 1e-15:
            m_exp = min(m_exp * 1.5, 5.0)
            if m_exp > 5.0:
                m_exp = 1.0
            no_improve += 1
            if no_improve > 30:
                rng = np.random.default_rng((it * 7919 + 13) % (2**31))
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue

        g = g / g.sum()

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 3.0)
            if no_improve > 40:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)

        if no_improve > 50:
            rng = np.random.default_rng((it * 7919 + 13) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def chirp_init(n, rng, variant=0):
    """Generate chirp-like initializations with good autocorrelation properties."""
    x = np.arange(n, dtype=np.float64)

    if variant == 0:
        # Quadratic chirp modulated
        phase = 2.0 * np.pi * x * x / (2.0 * n)
        a = 1.0 + 0.8 * np.cos(phase)
    elif variant == 1:
        # Cubic chirp
        phase = 2.0 * np.pi * x**3 / (n * n)
        a = 1.0 + 0.5 * np.cos(phase)
    elif variant == 2:
        # Multi-frequency
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * x / n) + 0.2 * np.cos(4.0 * np.pi * x / (3.0 * n))
    elif variant == 3:
        # Edge-weighted chirp
        envelope = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 0.3
        phase = 2.0 * np.pi * x * x / (2.0 * n)
        a = envelope * (1.0 + 0.5 * np.cos(phase))
    elif variant == 4:
        # Binary-like (approximate Barker)
        a = np.where((x * x % (n // 2)) < (n // 4), 1.0, 0.3)
    else:
        # Sine-squared
        a = np.sin(np.pi * x / n) ** 2 + 0.1

    a = np.maximum(a, 0.0)
    a = a + rng.standard_normal(n) * 0.01
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search: multi-phase optimization combining continuation game iterate,
    peak shaving, FW-LP, and smoothed gradient with diverse initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        n_default = 768
        parent = np.ones(n_default) / n_default
        try_seq(parent, "default_uniform")

    n_parent = len(parent)

    # === Phase 1: Multi-initialization exploration ===
    print("\n=== Phase 1: Multi-init Exploration ===")
    t_start = time.monotonic()

    initializations = []

    # Parent
    initializations.append(("parent", parent.copy()))

    # Chirp variants at parent length
    for v in range(5):
        initializations.append((f"chirp{v}", chirp_init(n_parent, rng, variant=v)))

    # Random Dirichlet
    initializations.append(("random", rng.dirichlet(np.ones(n_parent) * 0.5)))

    # Edge-concentrated
    x = np.arange(n_parent, dtype=np.float64)
    edge_w = 1.0 / (np.minimum(x, n_parent - 1 - x) + 1.0)
    initializations.append(("edge", edge_w / edge_w.sum()))

    # Try a couple different lengths
    for n_try in [n_parent - 96, n_parent + 96]:
        if n_try >= 128:
            interp = np.interp(np.linspace(0, 1, n_try),
                               np.linspace(0, 1, n_parent), parent)
            interp = np.maximum(interp, 0)
            interp = interp / interp.sum()
            initializations.append((f"interp_n{n_try}", interp))

    # Quick optimization of each initialization
    for label, init_a in initializations:
        remaining = deadline - time.monotonic() - 600
        if remaining < 10:
            break
        t_alloc = min(remaining, 40)
        if t_alloc < 5:
            continue
        print(f"  Quick optimize [{label}] n={len(init_a)}, t={t_alloc:.0f}s")
        a_opt, _ = game_iterate_continue(init_a, time_limit=t_alloc, T_start=0.005, T_end=0.0005)
        try_seq(a_opt, f"quick_{label}")

    print(f"  Phase 1 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 2: Deep continuation game iterate ===
    print("\n=== Phase 2: Deep Continuation Game Iterate ===")
    t_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Multiple runs with different temperature schedules
        schedules = [
            {'T_start': 0.01, 'T_end': 0.0001, 'time_frac': 0.4},
            {'T_start': 0.005, 'T_end': 0.00005, 'time_frac': 0.4},
            {'T_start': 0.002, 'T_end': 0.00002, 'time_frac': 0.2},
        ]

        total_alloc = min(deadline - time.monotonic() - 400, 600)
        for si, sched in enumerate(schedules):
            remaining = deadline - time.monotonic() - (400 - si * 100)
            if remaining < 15:
                break
            t_alloc = min(remaining, total_alloc * sched['time_frac'])
            if t_alloc < 10:
                continue
            print(f"  Game continue T=[{sched['T_start']},{sched['T_end']}], t={t_alloc:.0f}s")
            a_opt, _ = game_iterate_continue(best_seq, time_limit=t_alloc,
                                             T_start=sched['T_start'], T_end=sched['T_end'])
            try_seq(a_opt, f"game_cont_{si}")

        # FW-LP on best
        remaining = deadline - time.monotonic() - 300
        if remaining > 30:
            t_alloc = min(remaining, 180)
            print(f"  FW-LP n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "fw_lp_deep")

        # FW-LP from perturbed starts
        for pert_scale in [0.005, 0.015]:
            remaining = deadline - time.monotonic() - 200
            if remaining < 20:
                break
            t_alloc = min(remaining, 80)
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * pert_scale)
            a_opt, _ = fw_lp_optimize(f_pert, time_limit=t_alloc)
            try_seq(a_opt, f"fw_pert_{pert_scale}")

    print(f"  Phase 2 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 3: Peak shaving ===
    print("\n=== Phase 3: Peak Shaving ===")
    t_start = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 150
        if remaining > 20:
            t_alloc = min(remaining, 250)
            print(f"  Peak shaving, t={t_alloc:.0f}s")
            a_opt, _ = peak_shave(best_seq, time_limit=t_alloc, top_k=5)
            try_seq(a_opt, "peak_shave")

            # Second round with more peaks
            remaining = deadline - time.monotonic() - 100
            if remaining > 15:
                t_alloc = min(remaining, 100)
                a_opt, _ = peak_shave(a_opt, time_limit=t_alloc, top_k=10)
                try_seq(a_opt, "peak_shave_2")

    print(f"  Phase 3 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 4: Smoothed gradient refinement ===
    print("\n=== Phase 4: Smoothed Gradient ===")
    t_start = time.monotonic()

    if best_seq is not None:
        grad_configs = [
            {'T_init': 0.0005, 'lr_init': 0.02},
            {'T_init': 0.002, 'lr_init': 0.01},
            {'T_init': 0.0001, 'lr_init': 0.005},
        ]

        for ci, cfg in enumerate(grad_configs):
            remaining = deadline - time.monotonic() - (80 - ci * 20)
            if remaining < 10:
                break
            t_alloc = min(remaining, 80)
            print(f"  Smoothed grad config {ci}, T={cfg['T_init']}, t={t_alloc:.0f}s")
            a_opt, _ = smoothed_grad(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"smooth_{ci}")

    print(f"  Phase 4 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 5: Final fine-tuning ===
    print("\n=== Phase 5: Final Fine-tuning ===")
    t_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Very careful peak shaving
        remaining = deadline - time.monotonic() - 50
        if remaining > 10:
            t_alloc = min(remaining, 80)
            a_opt, _ = peak_shave(best_seq, time_limit=t_alloc, top_k=3)
            try_seq(a_opt, "final_peak")

        # Tiny game iterate
        remaining = deadline - time.monotonic() - 30
        if remaining > 10:
            t_alloc = min(remaining, 50)
            a_opt, _ = game_iterate_continue(best_seq, time_limit=t_alloc,
                                             T_start=0.0003, T_end=0.00005)
            try_seq(a_opt, "final_game")

        # Minute perturbations
        remaining = deadline - time.monotonic() - 15
        if remaining > 5:
            for _ in range(5):
                remaining = deadline - time.monotonic() - 5
                if remaining < 3:
                    break
                noise = rng.standard_normal(n_best) * 0.0005
                f_pert = project_simplex(best_seq + noise)
                a_opt, _ = smoothed_grad(f_pert, time_limit=min(remaining, 8),
                                         T_init=0.0002, lr_init=0.003)
                try_seq(a_opt, "fine_pert")

        # One more FW-LP
        remaining = deadline - time.monotonic() - 10
        if remaining > 8:
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=min(remaining, 30))
            try_seq(a_opt, "final_fw")

    print(f"  Phase 5 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
