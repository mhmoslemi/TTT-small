def project_simplex(v):

    n = len(v)

    return None

        f = f_new
        new_M = fast_convolve(f).max()

        if new_M < best_obj - 1e-15:
            best_obj = new_M
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    Adam iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            if no_improve > 300:
                break

    return best_f, best_obj


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with golden section line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        g = lp_direction(f, M, n)
        if g is None:
            g = lp_direction(f, M * 1.01, n)
            if g is None:
                g = lp_direction(f, M * 1.05, n)
                if g is None:
                    no_improve += 1
                    if no_improve > 60:
                        break
                    continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 60:
                rng = np.random.default_rng(it * 137 + 42)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
                continue
            continue

        best_alpha = 0.0
        best_val = M

        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0.01:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            gr = (np.sqrt(5) - 1) / 2
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            ft1 = np.maximum(f + c1 * (g - f), 0)
            ft1 = ft1 / max(ft1.sum(), 1e-15)
            fc1 = fast_convolve(ft1).max()
            ft2 = np.maximum(f + c2 * (g - f), 0)
            ft2 = ft2 / max(ft2.sum(), 1e-15)
            fc2 = fast_convolve(ft2).max()

            for _ in range(6):
                if hi - lo < 1e-7:
                    break
                if fc1 < fc2:
                    hi = c2
                    c2 = c1
                    fc2 = fc1
                    c1 = hi - gr * (hi - lo)
                    ft1 = np.maximum(f + c1 * (g - f), 0)
                    ft1 = ft1 / max(ft1.sum(), 1e-15)
                    fc1 = fast_convolve(ft1).max()
                else:
                    lo = c1
                    c1 = c2
                    fc1 = fc2
                    c2 = lo + gr * (hi - lo)
                    ft2 = np.maximum(f + c2 * (g - f), 0)
                    ft2 = ft2 / max(ft2.sum(), 1e-15)
                    fc2 = fast_convolve(ft2).max()

            cand = (lo + hi) / 2
            ft = np.maximum(f + cand * (g - f), 0)
            ft = ft / max(ft.sum(), 1e-15)
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = cand

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 30 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 80:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.002 + 0.002 * min(no_improve // 40, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_obj


def peak_chip_optimize(f0, time_limit, verbose=False):
    """Direct peak-chipping: iteratively reduce the top peak by redistributing mass."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(77777)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > 200:
            break

        conv = fast_convolve(f)
        peak_idx = int(np.argmax(conv))
        peak_val = conv[peak_idx]

        # Find contributing pairs for the peak
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        contributing = np.arange(js, je + 1)
        products = f[contributing] * f[peak_idx - contributing]

        # Find the largest contributor
        max_local = np.argmax(products)
        i = contributing[max_local]
        j = peak_idx - i

        if i == j:
            # Diagonal: reducing f[i] reduces c[peak] by 2*f[i]*delta
            # But also affects other convolution terms
            # Try reducing f[i] and adding to a "safe" position
            delta = f[i] * 0.1
            # Find a safe position: one that doesn't contribute to top peaks
            top_peaks = np.argpartition(conv, -max(10, n // 10))[-max(10, n // 10):]
            safe_positions = [p for p in range(n) if p not in top_peaks]
            if not safe_positions:
                safe_positions = [p for p in range(n) if p != i]
            if not safe_positions:
                no_improve += 1
                continue
            # Pick the position that contributes least to any peak
            target = safe_positions[np.random.randint(len(safe_positions))]
            f_trial = f.copy()
            f_trial[i] -= delta
            f_trial[target] += delta
            f_trial = np.maximum(f_trial, 0)
            f_trial = f_trial / f_trial.sum()
            val = fast_convolve(f_trial).max()
            if val < best_obj - 1e-14:
                f = f_trial
                best_obj = val
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Off-diagonal: reducing f[i] or f[j] helps
            # Try transferring from the larger of f[i], f[j]
            if f[i] >= f[j]:
                src, dst_pair = i, j
            else:
                src, dst_pair = j, i

            for frac in [0.05, 0.1, 0.15, 0.2]:
                delta = f[src] * frac
                # Find safe target
                top_peaks_set = set(np.argpartition(conv, -max(20, n // 8))[-max(20, n // 8):].tolist())
                safe = [p for p in range(n) if p not in top_peaks_set and p != src]
                if not safe:
                    safe = [p for p in range(n) if p != src]
                if not safe:
                    break
                target = safe[np.random.randint(len(safe))]
                f_trial = f.copy()
                f_trial[src] -= delta
                f_trial[target] += delta
                f_trial = np.maximum(f_trial, 0)
                f_trial = f_trial / f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_obj - 1e-14:
                    f = f_trial
                    best_obj = val
                    best_f = f.copy()
                    no_improve = 0
                    break
                else:
                    no_improve += 1
                    break

    return best_f, best_obj


def pairwise_optimize(f0, time_limit, verbose=False):
    """Optimize pairs of coordinates by mass transfer."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(98765)

    offsets = [1, 2, 3, 5, 7, 11, 17, 29, 47, 79, 127, 211]

    no_improve_passes = 0
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        if no_improve_passes >= 5:
            break
        pass_improved = False

        order = rng.permutation(n)

        for idx_i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = order[idx_i]
            for offset in offsets:
                j = order[(idx_i + offset) % n]
                if i == j:
                    continue
                c = f[i] + f[j]
                if c < 1e-12:
                    continue

                best_t = f[i] / c
                best_val = fast_convolve(f).max()
                candidates = [best_t, 0.0, 1.0, 0.25, 0.5, 0.75]
                if 0.1 < best_t < 0.9:
                    candidates.extend([best_t - 0.1, best_t + 0.1, best_t - 0.05, best_t + 0.05,
                                        best_t - 0.02, best_t + 0.02])

                for t in candidates:
                    t = max(0.0, min(1.0, t))
                    f_trial = f.copy()
                    f_trial[i] = t * c
                    f_trial[j] = (1 - t) * c
                    f_trial = f_trial / f_trial.sum()
                    val = fast_convolve(f_trial).max()
                    if val < best_val - 1e-14:
                        best_val = val
                        best_t = t

                if best_val < best_obj - 1e-13:
                    f[i] = best_t * c
                    f[j] = (1 - best_t) * c
                    f = f / f.sum()
                    best_obj = best_val
                    best_f = f.copy()
                    pass_improved = True

        if pass_improved:
            no_improve_passes = 0
        else:
            no_improve_passes += 1

    return best_f, best_obj


def simulate_anneal_polish(f0, time_limit, verbose=False):
    """Simulated annealing with small perturbations."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    temp_start = 0.01
    temp_end = 0.0001
    it = 0
    max_iters = int(time_limit / 0.05)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        progress = min(1.0, it / max(1, max_iters))
        temp = temp_start * (temp_end / temp_start) ** progress

        # Random perturbation
        f_trial = f + rng.standard_normal(n) * temp
        f_trial = project_simplex(f_trial)

        new_obj = fast_convolve(f_trial).max()
        old_obj = fast_convolve(f).max()

        delta = new_obj - old_obj
        if delta < 0 or rng.random() < np.exp(-delta / (temp * 0.01 + 1e-15)):
            f = f_trial
            if new_obj < best_obj - 1e-14:
                best_obj = new_obj
                best_f = f.copy()
                if verbose and it % 100 == 0:
                    print(f"    SA iter {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_init(n, kind=0, seed=42):
    """Create structured initializations."""
    rng = np.random.default_rng(seed)
    if kind == 0:
        # Uniform
        f = np.ones(n) / n
    elif kind == 1:
        # Gaussian centered
        x = np.arange(n)
        f = np.exp(-0.5 * ((x - n / 2.0) / (n / 6.0)) ** 2)
    elif kind == 2:
        # U-shaped (more at edges)
        x = np.arange(n) / (n - 1)
        f = 0.5 + 0.5 * np.cos(np.pi * x)
        f = f + 0.3 * (1 - f)
    elif kind == 3:
        # Random
        f = rng.random(n)
    elif kind == 4:
        # Cosine window
        x = np.arange(n)
        f = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    elif kind == 5:
        # Bimodal
        x = np.arange(n)
        f = np.exp(-0.5 * ((x - n * 0.2) / (n / 10.0)) ** 2) + np.exp(-0.5 * ((x - n * 0.8) / (n / 10.0)) ** 2)
    elif kind == 6:
        # Triangular
        x = np.arange(n) / (n - 1)
        f = 1 - np.abs(2 * x - 1)
    else:
        f = np.ones(n) / n
    f = np.maximum(f, 0)
    ss = f.sum()
    if ss > 1e-15:
        f /= ss
    return f


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining Adam-SG, LP-FW, peak-chipping,
    pairwise polish, and simulated annealing across multiple lengths.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if len(a) == 0:
            return float('inf')
        sc = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
        return sc

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.8f}")

    # Phase 1: Multi-length quick exploration
    print(f"\n=== Phase 1: Multi-length exploration (best={best_score:.8f}) ===")
    candidate_lengths = [384, 512, 640, 768, 896, 1024, 1280, 1536, 2048]
    if len(parent) in candidate_lengths:
        candidate_lengths.remove(len(parent))

    # Sort by proximity to parent length for interpolation quality
    candidate_lengths.sort(key=lambda x: abs(x - len(parent)))

    top_candidates = []  # list of (f, n, score)
    time_per_length = 25  # seconds per length for quick Adam

    for target_n in candidate_lengths[:6]:  # Try up to 6 lengths
        rem = deadline - time.monotonic() - 900
        if rem < 30:
            break

        t_alloc = min(rem / max(1, 6 - candidate_lengths.index(target_n)), time_per_length)
        if t_alloc < 10:
            t_alloc = 10

        # Try two initializations
        for init_kind in [0, 3]:
            rem = deadline - time.monotonic() - 850
            if rem < 8:
                break

            if init_kind == 0:
                f_init = interpolate_sequence(parent, target_n)
            else:
                f_init = structured_init(target_n, kind=init_kind, seed=seed + target_n)

            t_aw = min(rem * 0.45, t_alloc)
            if t_aw < 5:
                continue
            a_opt, obj = adam_peak_optimize(f_init, time_limit=t_aw)
            sc = try_seq(a_opt, f"adam_n{target_n}_k{init_kind}")
            if np.isfinite(sc):
                top_candidates.append((a_opt, target_n, sc))

    # Sort top candidates by score
    top_candidates.sort(key=lambda x: x[2])
    print(f"  Top candidates after Phase 1: {[(c[2], c[1]) for c in top_candidates[:5]]}")

    # Phase 2: Deep optimization on best candidates
    print(f"\n=== Phase 2: Deep optimization (best={best_score:.8f}) ===")
    num_deep = min(3, len(top_candidates) if top_candidates else 1)
    if num_deep == 0:
        num_deep = 1
        top_candidates = [(best_seq, len(best_seq), best_score)]

    for ci in range(num_deep):
        rem = deadline - time.monotonic() - 500
        if rem < 60:
            break

        f_cand, n_cand, sc_cand = top_candidates[ci]
        time_per_cand = min(rem / max(1, num_deep - ci), 180)

        # LP-FW
        rem = deadline - time.monotonic() - 450
        t_lpfw = min(rem * 0.4, 80)
        if t_lpfw > 15:
            print(f"  LP-FW on n={n_cand} (score={sc_cand:.8f}), t={t_lpfw:.0f}s")
            a_opt, _ = lp_fw_optimize(f_cand, time_limit=t_lpfw)
            try_seq(a_opt, f"lpfw_c{ci}")

        # Peak-chipping
        rem = deadline - time.monotonic() - 400
        t_chip = min(rem * 0.2, 40)
        if t_chip > 10:
            a_opt2, _ = peak_chip_optimize(a_opt if 'a_opt' in dir() else f_cand, time_limit=t_chip)
            try_seq(a_opt2, f"chip_c{ci}")

        # Adam again after LP-FW
        rem = deadline - time.monotonic() - 350
        t_adam = min(rem * 0.3, 60)
        if t_adam > 10:
            a_opt3, _ = adam_peak_optimize(a_opt2 if 'a_opt2' in dir() else (a_opt if 'a_opt' in dir() else f_cand), time_limit=t_adam)
            try_seq(a_opt3, f"adam_c{ci}_post")

    # Phase 3: Additional diverse starts
    print(f"\n=== Phase 3: Diverse starts (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        diverse_inits = []

        # Reverse
        diverse_inits.append((best_seq[::-1].copy() / best_seq[::-1].sum(), "rev"))
        # Noise
        for noise_level in [0.005, 0.01, 0.02]:
            rng = np.random.default_rng(seed + 100 + int(noise_level * 1000))
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise_level)
            diverse_inits.append((f_pert, f"noise_{int(noise_level*1000)}"))
        # Blend with uniform
        f_blend = 0.6 * best_seq + 0.4 * (np.ones(n_best) / n_best)
        diverse_inits.append((f_blend / f_blend.sum(), "blend"))

        for di, (f_init, label) in enumerate(diverse_inits):
            rem = deadline - time.monotonic() - 300
            if rem < 20:
                break
            t_alloc = min(rem / max(1, len(diverse_inits) - di), 60)
            if t_alloc < 8:
                continue

            # Alternate between methods
            if di % 2 == 0:
                a_opt, _ = adam_peak_optimize(f_init, time_limit=t_alloc)
            else:
                a_opt, _ = lp_fw_optimize(f_init, time_limit=min(t_alloc, 50))
            try_seq(a_opt, f"div_{label}")

    # Phase 4: Try different lengths with best structure
    print(f"\n=== Phase 4: Length transfer (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        nearby_lengths = [n_best - 64, n_best - 32, n_best + 32, n_best + 64, n_best + 128, n_best * 2]
        nearby_lengths = [x for x in nearby_lengths if 64 <= x <= 4096 and x != n_best]

        for nl in nearby_lengths[:4]:
            rem = deadline - time.monotonic() - 200
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, nl)
            t_alloc = min(rem * 0.3, 50)
            if t_alloc < 10:
                continue
            a_opt, _ = adam_peak_optimize(f_interp, time_limit=t_alloc)
            try_seq(a_opt, f"xl_n{nl}")

            rem = deadline - time.monotonic() - 180
            if rem > 15:
                a_opt2, _ = lp_fw_optimize(a_opt, time_limit=min(rem * 0.4, 30))
                try_seq(a_opt2, f"xp_n{nl}")

    # Phase 5: Intensive pairwise polish
    print(f"\n=== Phase 5: Pairwise polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_pw = min(rem * 0.7, 150)
            print(f"  Time: {t_pw:.0f}s, n={len(best_seq)}")
            a_opt, _ = pairwise_optimize(best_seq, time_limit=t_pw)
            try_seq(a_opt, "pairwise")

            # Adam after pairwise
            rem = deadline - time.monotonic() - 80
            if rem > 12:
                a_opt2, _ = adam_peak_optimize(a_opt, time_limit=min(rem, 50))
                try_seq(a_opt2, "adam_post_pw")

    # Phase 6: Peak-chipping on best
    print(f"\n=== Phase 6: Peak-chipping (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 15:
            t_chip = min(rem * 0.6, 60)
            a_opt, _ = peak_chip_optimize(best_seq, time_limit=t_chip)
            try_seq(a_opt, "chip_best")

    # Phase 7: Simulated annealing polish
    print(f"\n=== Phase 7: SA polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 10:
            t_sa = min(rem * 0.6, 40)
            a_opt, _ = simulate_anneal_polish(best_seq, time_limit=t_sa)
            try_seq(a_opt, "sa_polish")

    # Phase 8: Final intensive LP-FW + tiny perturbations
    print(f"\n=== Phase 8: Final polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t_final = min(rem * 0.5, 40)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_final)
            try_seq(a_opt, "lpfw_final")

            rem = deadline - time.monotonic() - 20
            if rem > 10:
                a_opt2, _ = adam_peak_optimize(a_opt, time_limit=min(rem, 25))
                try_seq(a_opt2, "adam_final")

        # Tiny perturbations
        for trial in range(5):
            rem = deadline - time.monotonic() - 10
            if rem < 8:
                break
            rng = np.random.default_rng(seed + 1000 + trial * 7)
            f_pert = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * 0.001)
            a_opt, _ = adam_peak_optimize(f_pert, time_limit=min(rem, 12))
            try_seq(a_opt, f"tiny_{trial}")

        # One final pairwise
        rem = deadline - time.monotonic() - 5
        if rem > 8:
            a_opt, _ = pairwise_optimize(best_seq, time_limit=min(rem, 25))
            try_seq(a_opt, "pw_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
