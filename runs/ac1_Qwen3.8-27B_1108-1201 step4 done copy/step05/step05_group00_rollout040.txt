"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using mirror-gradient descent with
Nesterov momentum, adaptive temperature softmax-smoothing, exact pair-swap
polish, and multi-n exploration. Key insight: the gradient of the peak
auto-convolution at position k is 2*a_reversed_at_k, which drives the
sequence toward mirror-asymmetry (anti-correlation between a[i] and a[n-1-i]).
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def conv_max(a):
    """Return max of auto-convolution."""
    return fast_convolve(a).max()


def mirror_gradient(a, peaks, weights):
    """
    Compute the weighted mirror gradient: grad[i] = sum_k w_k * 2*a[k-i].
    Uses FFT for efficiency.
    """
    n = len(a)
    a_rev = a[::-1]
    # w_full has length 2n-1, w_full[k] = weights for peak k
    w_full = np.zeros(2 * n - 1)
    for k, w in zip(peaks, weights):
        if 0 <= k < 2 * n - 1:
            w_full[k] = w
    # cross_conv(w_full, a_rev)[n-1 + i] = sum_k w_full[k] * a_rev[n-1+i-k]
    # = sum_k w[k] * a[k-i]
    conv_result = cross_convolve(w_full, a_rev)
    grad = 2.0 * conv_result[n - 1:2 * n - 1]
    return grad


def softmax_weights(conv, T, top_k=None):
    """Compute softmax weights over (top) peaks."""
    if top_k is not None and top_k < len(conv):
        idx = np.argpartition(conv, -top_k)[-top_k:]
        vals = conv[idx]
        scaled = vals / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        return idx, w
    else:
        scaled = conv / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        return np.arange(len(conv)), w


def mirror_grad_optimize(a0, time_limit, T_init=0.002, eta_init=0.03, beta_init=0.90,

    Mirror gradient descent with momentum and adaptive temperature.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()

    else:
        a = a / s
    n = len(a)

    T = T_init
    eta = eta_init
    beta = beta_init

    v = np.zeros(n)
    best_a = a.copy()
    best_obj = conv_max(a)

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    prev_obj = best_obj

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.05:
            break

        a_new = project_simplex(a + v)

            best_obj = new_obj

        else:

            for factor in [0.5, 0.25, 0.1, 0.05]:

def pair_swap_polish(a0, time_limit, verbose=False):
    """
    For the dominant peak, try exact 2-coordinate swaps to reduce it.
    Swap: a[i] -> a[i]+t, a[j] -> a[j]-t where i+j = peak_position.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = conv_max(a)

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        current_max = conv.max()
        k_peak = int(np.argmax(conv))

        # Find all valid pairs (i, j) with i + j == k_peak, i < j
        # i ranges from max(0, k_peak - n + 1) to min(n-1, k_peak)
        i_start = max(0, k_peak - n + 1)
        i_end = min(n - 1, k_peak)

        improved = False
        # Try pairs in order of most "promising" (largest asymmetry)
        candidates = []
        for i in range(i_start, i_end + 1):
            j = k_peak - i
            if i == j:
                continue
            # Change in conv[k_peak] = 2t*(a[j]-a[i]) - 2*t^2
            # To reduce: want a[j] > a[i] and t > 0 (move mass from i to j... wait)
            # Actually: if a[j] > a[i], then for small t > 0, conv decreases.
            # So: increase a[i], decrease a[j] when a[j] > a[i]? No wait.
            # delta = 2t*(a[j] - a[i]) - 2t^2
            # For delta < 0: need 2t*(a[j]-a[i]) < 2t^2, i.e., t > a[j]-a[i] (if a[j]>a[i])
            # Or: (a[i]-a[j]) > 0 and t > 0 gives delta < 0 for small t.
            # Wait: delta = 2t*(a[j]-a[i]) - 2t^2
            # If a[i] > a[j]: delta = 2t*(negative) - 2t^2 < 0 for all t > 0. Good!
            # If a[j] > a[i]: delta = 2t*(positive) - 2t^2, which is < 0 only for t > a[j]-a[i].
            # So the easy direction is: increase a[i] when a[i] > a[j] (move mass from j to i).
            asym = abs(a[i] - a[j])
            candidates.append((asym, i, j))

        candidates.sort(reverse=True)

        for asym, i, j in candidates[:min(50, len(candidates))]:
            if asym < 1e-12:
                break
            # Direction: if a[i] > a[j], increase a[i] (t > 0 means a[i]+=t, a[j]-=t)
            # if a[j] > a[i], decrease a[i] (t < 0 means a[i]+=t, a[j]-=t... wait)
            # Let me redefine: transfer t from j to i: a[i] += t, a[j] -= t
            # delta_conv[k_peak] = 2t*(a[j] - a[i]) - 2t^2 ... let me recheck
            # Actually from earlier: delta = 2t*(a[k-i] - a[k-j]) - 2t^2 where k-i=j, k-j=i
            # So delta = 2t*(a[j] - a[i]) - 2t^2
            # If a[i] > a[j]: delta = 2t*(negative) - 2t^2 < 0 for t > 0.
            #   So increase a[i] by t, decrease a[j] by t.
            # If a[j] > a[i]: delta = 2t*(positive) - 2t^2 < 0 for t > a[j]-a[i].
            #   So we need a large enough t, or we go the other way:
            #   Let t < 0: delta = 2t*(positive) - 2t^2. For t < 0, 2t*(positive) < 0 and -2t^2 < 0.
            #   So delta < 0 for all t < 0. That means decrease a[i], increase a[j].
            #   Which is the same as: transfer from i to j.

            # In both cases, the optimal direction is to move mass TOWARD the larger one
            # (increase the larger, decrease the smaller) to reduce the product a[i]*a[j].

            # Optimal t (unconstrained): t* = (a[j] - a[i]) / 2
            # But we need a[i] + t >= 0 and a[j] - t >= 0.
            # If t* > 0: t <= a[j] (to keep a[j] >= 0)
            # If t* < 0: t >= -a[i] (to keep a[i] >= 0)

            t_opt = (a[j] - a[i]) / 2.0

                t_max = a[j] * 0.99

            # Try several t values around t_opt

            for t in t_values:
                if abs(t) < 1e-10:
                    continue
                a[i] += t
                a[j] -= t
                if a[i] < 0 or a[j] < 0:
                    a[i] -= t
                    a[j] += t
                    continue
                a_norm = a / a.sum()
                new_obj = conv_max(a_norm)
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a_norm.copy()
                    a = a_norm
                    improved = True
                    break
                else:
                    # Undo
                    a[i] -= t
                    a[j] += t

            if improved:
                no_improve = 0
                break

        if not improved:
            no_improve += 1
            if no_improve > 20:
                # Try random perturbations

                    best_a = a_try.copy()
                    a = a_try
                    no_improve = 0
                elif no_improve > 100:

    return best_a, best_obj


def fw_lp_step(a, M, n):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = a
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit, max_lp_time=5.0):

        f = np.ones(len(f)) / len(f)
    else:
        f = f / s

    M = conv.max()
    best_obj = M
    best_f = f.copy()

    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)

            break

        t_lp_start = time.monotonic()
        g = fw_lp_step(f, M * m_exp, n)
        t_lp_end = time.monotonic()

        if g is None:
            m_exp *= 1.3
            if m_exp > 5.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            M = fast_convolve(f).max()
            m_exp = 1.0
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.15, 8.0)

        if no_improve > 50:
            rng = np.random.default_rng(it * 7919 + 13)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            M = fast_convolve(f).max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    x = np.arange(n, dtype=np.float64)

    # Linear ramp (mirror asymmetry)
    for alpha in [0.3, 0.5, 0.7, 1.0]:
        a = 1.0 + alpha * (2.0 * x / max(n - 1, 1) - 1.0)
        a = np.maximum(a, 0.05)
        inits.append(a / a.sum())

    # Gaussian at different centers
    for center in [0.3, 0.5, 0.7]:
        g = np.exp(-0.5 * ((x - center * (n - 1)) / (0.15 * n)) ** 2)
        inits.append(g / g.sum())

    # Two peaks (asymmetric)
    g2 = np.exp(-0.5 * ((x - 0.2 * (n - 1)) / (0.1 * n)) ** 2) + \
         0.5 * np.exp(-0.5 * ((x - 0.8 * (n - 1)) / (0.1 * n)) ** 2)
    inits.append(g2 / g2.sum())

    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.1
    inits.append(c1 / c1.sum())

    # Edge concentration
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 0.5
    inits.append(edge_w / edge_w.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search combining:
    1. Mirror gradient descent (main workhorse)
    2. FW-LP for global structure
    3. Pair-swap polish for local refinement
    4. Multi-n exploration and adaptive restarts
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent_initial")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768

    n_parent = len(parent)
    print(f"Starting optimization, parent n={n_parent}, current best={best_score:.8f}")

    # === Phase 1: Mirror gradient on parent (~600s) ===
    print("\n=== Phase 1: Mirror Gradient on Parent ===")
    t_start = time.monotonic()

    remaining = deadline - time.monotonic() - 800
    if remaining > 30:
        t_alloc = min(remaining, 400)
        print(f"  Mirror grad T=0.002, eta=0.03, t={t_alloc:.0f}s")
        a_opt, obj = mirror_grad_optimize(parent, time_limit=t_alloc, T_init=0.002, eta_init=0.03)
        try_seq(a_opt, "mgrad_T0.002")

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 600
        if remaining > 30:
            t_alloc = min(remaining, 200)
            print(f"  Mirror grad T=0.005, eta=0.01, t={t_alloc:.0f}s")
            a_opt, obj = mirror_grad_optimize(best_seq, time_limit=t_alloc, T_init=0.005, eta_init=0.01)
            try_seq(a_opt, "mgrad_T0.005")

    print(f"  Phase 1 done: best={best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 2: FW-LP for global structure (~300s) ===
    print("\n=== Phase 2: FW-LP ===")
    t_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic() - 500
        if remaining > 20:
            t_alloc = min(remaining, 180)
            print(f"  FW-LP n={n_best}, t={t_alloc:.0f}s")
            a_opt, obj = fw_lp_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "fw_lp_1")

        if best_seq is not None:
            remaining = deadline - time.monotonic() - 350
            if remaining > 20:
                # Perturbed restart
                for scale in [0.005, 0.015]:
                    remaining = deadline - time.monotonic() - 300
                    if remaining < 15:
                        break
                    n_best = len(best_seq)
                    f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
                    t_alloc = min(remaining, 60)
                    print(f"  FW-LP pert={scale}, t={t_alloc:.0f}s")
                    a_opt, obj = fw_lp_optimize(f_pert, time_limit=t_alloc)
                    try_seq(a_opt, f"fw_lp_pert_{scale}")

    print(f"  Phase 2 done: best={best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 3: Deep mirror gradient with multiple configs (~400s) ===
    print("\n=== Phase 3: Deep Mirror Gradient ===")
    t_start = time.monotonic()

    if best_seq is not None:
        configs = [
            {'T_init': 0.001, 'eta_init': 0.02, 'beta_init': 0.92},
            {'T_init': 0.003, 'eta_init': 0.04, 'beta_init': 0.88},
            {'T_init': 0.0005, 'eta_init': 0.01, 'beta_init': 0.94},
            {'T_init': 0.01, 'eta_init': 0.06, 'beta_init': 0.85},
        ]
        for ci, cfg in enumerate(configs):
            remaining = deadline - time.monotonic() - 150
            if remaining < 20:
                break
            t_alloc = min(remaining, 100)
            print(f"  Mgrad config {ci} (T={cfg['T_init']}, eta={cfg['eta_init']}), t={t_alloc:.0f}s")
            a_opt, obj = mirror_grad_optimize(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"mgrad_c{ci}")

    print(f"  Phase 3 done: best={best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 4: Pair-swap polish (~250s) ===
    print("\n=== Phase 4: Pair-Swap Polish ===")
    t_start = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 100
        if remaining > 15:
            t_alloc = min(remaining, 150)
            print(f"  Pair swap, t={t_alloc:.0f}s")
            a_opt, obj = pair_swap_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "pair_swap_1")

        if best_seq is not None:
            remaining = deadline - time.monotonic() - 60
            if remaining > 15:
                t_alloc = min(remaining, 80)
                print(f"  Pair swap round 2, t={t_alloc:.0f}s")
                a_opt, obj = pair_swap_polish(best_seq, time_limit=t_alloc)
                try_seq(a_opt, "pair_swap_2")

    print(f"  Phase 4 done: best={best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 5: Multi-n exploration (~150s) ===
    print("\n=== Phase 5: Multi-n Exploration ===")
    t_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        for n_try in [n_best, n_best - 1, n_best + 1, n_best + 2, n_best // 2, n_best * 2]:
            remaining = deadline - time.monotonic() - 60
            if remaining < 15:
                break
            if n_try < 100 or n_try > 4096:
                continue
            try:
                if n_try == n_best:
                    start = best_seq.copy()
                else:
                    start = np.interp(np.linspace(0, 1, n_try),
                                      np.linspace(0, 1, len(best_seq)), best_seq)
                    start = np.maximum(start, 0)
                    start = start / start.sum()

                t_alloc = min(remaining, 30)
                print(f"  Mgrad n={n_try}, t={t_alloc:.0f}s")
                a_opt, obj = mirror_grad_optimize(start, time_limit=t_alloc,
                                                   T_init=0.002, eta_init=0.03)
                try_seq(a_opt, f"mgrad_n{n_try}")
            except Exception:
                continue

    print(f"  Phase 5 done: best={best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 6: Final refinement (~60s) ===
    print("\n=== Phase 6: Final Refinement ===")
    t_start = time.monotonic()

    if best_seq is not None:
        # Very fine mirror gradient
        remaining = deadline - time.monotonic() - 20
        if remaining > 10:
            t_alloc = min(remaining, 25)
            print(f"  Fine mgrad, t={t_alloc:.0f}s")
            a_opt, obj = mirror_grad_optimize(best_seq, time_limit=t_alloc,
                                              T_init=0.0003, eta_init=0.005, beta_init=0.95)
            try_seq(a_opt, "fine_mgrad")

        # Quick pair swap
        remaining = deadline - time.monotonic() - 10
        if remaining > 5:
            t_alloc = min(remaining, 20)
            print(f"  Final pair swap, t={t_alloc:.0f}s")
            a_opt, obj = pair_swap_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_pswap")

        # Tiny perturbations
        remaining = deadline - time.monotonic() - 5
        if remaining > 5:
            n_best = len(best_seq)
            for _ in range(3):
                remaining = deadline - time.monotonic() - 3
                if remaining < 2:
                    break
                noise = rng.standard_normal(n_best) * 0.0005
                f_pert = project_simplex(best_seq + noise)
                a_opt, obj = mirror_grad_optimize(f_pert, time_limit=min(remaining, 3),
                                                   T_init=0.001, eta_init=0.01)
                try_seq(a_opt, "perturb_fine")

    print(f"  Phase 6 done: best={best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
