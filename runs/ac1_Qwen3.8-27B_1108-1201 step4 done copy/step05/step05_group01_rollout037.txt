"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-based projected gradient
descent with adaptive log-sum-exp smoothing, Nesterov momentum, and LP-FW polish.
Explores multiple sequence lengths with cosine-modulated and parent-interpolated
initializations.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_gradient(a, r, alpha):
    """
    Compute gradient of the log-sum-exp approximation to max(r).
    LSE = (1/t) * log(sum_k exp(t * r[k])) where t = alpha / max(r).
    Returns gradient w.r.t. a (length n).
    """
    n = len(a)
    max_r = r.max()
    if max_r < 1e-30:
        return np.zeros(n)
    t = alpha / max_r
    # Numerically stable softmax
    scaled = t * r - t * max_r
    exp_scaled = np.exp(scaled)
    w = exp_scaled / exp_scaled.sum()
    if w.sum() < 1e-30:
        w = np.zeros(len(r))
        w[np.argmax(r)] = 1.0
        return 2.0 * _correlate_slice(w, a)
    # gradient[m] = 2 * sum_k w[k] * a[k-m] (valid indices)
    # = 2 * cross_correlation(w, a)[m]
    grad = 2.0 * _correlate_slice(w, a)
    return grad


def _correlate_slice(w, a):
    """
    Compute R[m] = sum_j a[j] * w[j+m] for m = 0..n-1 via FFT.
    w has length 2n-1, a has length n.
    """
    n = len(a)
    L = 1 << (2 * n - 1 - 1).bit_length()
    A = np.fft.fft(a, L)
    W = np.fft.fft(w, L)
    R = np.fft.ifft(A * np.conj(W), L).real
    return R[:n]


def fft_gradient_optimize(a0, time_limit, alpha0=3.0, alpha_final=15.0,
                          lr0=0.05, mom0=0.8, verbose=False):
    """
    Nesterov-accelerated projected gradient descent on LSE approximation
    of max autocorrelation peak.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()

        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    r = fast_convolve(a)
    best_obj = r.max()
    best_a = a.copy()

    v = np.zeros(n)  # momentum
    y = a.copy()     # Nesterov lookahead
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        alpha = alpha0 + (alpha_final - alpha0) * progress * progress

        grad = compute_lse_gradient(y, r_y, alpha)

        a_new = y + v

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    # Build constraint matrix: (f*g)[k] <= M for k = 0..2n-2
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def cosine_modulated_init(n, freq, amp, seed=0):
    """Generate a cosine-modulated uniform initialization."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def raised_cosine_init(n):
    """Raised cosine window initialization."""
    j = np.arange(n)
    a = np.sin(np.pi * (j + 0.5) / n) ** 2
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

    # === Phase 4: LP-FW polish ===
    print(f"\n=== Phase 4: LP-FW polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 40:
            n_best = len(best_seq)
            # LP-FW is expensive for large n; limit to n <= 1536
            if n_best <= 1280:
                t4 = min(rem * 0.4, 200)
                print(f"  LP-FW n={n_best}, time={t4:.0f}s")
                a_opt, _ = lp_fw_optimize(best_seq, time_limit=t4, verbose=True)
                try_seq(a_opt, "lpfw_polish")

    # === Phase 5: Cosine-modulated inits at best length ===
    print(f"\n=== Phase 5: Cosine inits (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for freq, amp in [(1, 0.2), (2, 0.15), (3, 0.1), (2, 0.25), (1, 0.35)]:
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            a_cos = cosine_modulated_init(n_best, freq, amp)
            # Blend with best
            a_blend = 0.7 * best_seq + 0.3 * a_cos
            a_blend = project_simplex(a_blend)
            t5 = min(rem * 0.2, 80)
            if t5 > 15:
                a_opt, _ = fft_gradient_optimize(a_blend, time_limit=t5,
                                                 alpha0=5.0, alpha_final=15.0,
                                                 lr0=0.02, mom0=0.8)
                try_seq(a_opt, f"cos_f{freq}_a{amp}")

    # === Phase 6: Multi-start with perturbations ===
    print(f"\n=== Phase 6: Multi-start (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(6):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t6 = min(rem * 0.2, 80)

            if p == 0:
                noise = rng.standard_normal(n_best) * 0.003
                f_init = project_simplex(best_seq + noise)
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.005
                f_init = project_simplex(best_seq + noise)
            elif p == 2:
                # Frequency-domain perturbation
                L = 1 << (n_best - 1).bit_length()
                A = np.fft.rfft(best_seq, L)
                A_real = A.real + rng.standard_normal(len(A)) * 0.01
                A_imag = A.imag + rng.standard_normal(len(A)) * 0.01
                A_pert = A_real + 1j * A_imag
                a_pert = np.fft.irfft(A_pert, L)[:n_best]
                f_init = project_simplex(a_pert)
            elif p == 3:
                # Time-reversal blend
                f_init = project_simplex(0.8 * best_seq + 0.2 * best_seq[::-1])
            elif p == 4:
                # Local smoothing
                kernel = np.ones(5) / 5.0
                a_smooth = np.convolve(best_seq, kernel, mode='same')
                f_init = project_simplex(a_smooth)
            else:
                # Random frequency emphasis
                L = 1 << (n_best - 1).bit_length()
                A = np.fft.rfft(best_seq, L)
                mask = rng.random(len(A)) > 0.3
                A_pert = A * mask
                a_pert = np.fft.irfft(A_pert, L)[:n_best]
                f_init = project_simplex(a_pert)

            a_opt, _ = fft_gradient_optimize(f_init, time_limit=t6,
                                             alpha0=5.0, alpha_final=18.0,
                                             lr0=0.02, mom0=0.8)
            try_seq(a_opt, f"multi_{p}")

    # === Phase 7: Final fine-tuning ===
    print(f"\n=== Phase 7: Final tuning (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()

        # Very fine LSE
        if rem > 20:
            t7 = min(rem * 0.4, 120)
            a_opt, _ = fft_gradient_optimize(best_seq, time_limit=t7,
                                             alpha0=15.0, alpha_final=30.0,
                                             lr0=0.005, mom0=0.9, verbose=True)
            try_seq(a_opt, "final_fine")

        # Short LP-FW if n is small enough
        rem = deadline - time.monotonic()
        if rem > 15 and n_best <= 1024:
            t8 = min(rem * 0.5, 60)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t8)
            try_seq(a_opt, "final_lpfw")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
