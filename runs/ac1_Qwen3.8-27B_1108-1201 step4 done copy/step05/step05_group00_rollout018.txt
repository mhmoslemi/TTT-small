"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using peak-directed pairwise swap optimization
with analytical O(n) convolution update, simulated annealing, and multi-peak awareness.
Key innovation: exact 1D golden-section optimization for each candidate swap using the
closed-form expression for how the auto-convolution changes under a two-coordinate swap.
"""
import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_swap_vectors(a, p, q):
    """Precompute base and quad vectors for swap (p, q)."""
    n = len(a)
    conv_len = 2 * n - 1
    base = np.zeros(conv_len)
    base[p:p + n] -= 2.0 * a
    base[q:q + n] += 2.0 * a
    quad = np.zeros(conv_len)
    if 2 * p < conv_len:
        quad[2 * p] -= 1.0
    if 2 * q < conv_len:
        quad[2 * q] += 1.0
    return base, quad

def eval_swap(conv, base, quad, d):
    """Evaluate max of conv + d*base + d^2*quad."""
    return (conv + d * base + d * d * quad).max()

    best_d = 0.0
    best_val = conv.max()
    for d in d_vals:
        val = eval_swap(conv, base, quad, d)
        if val < best_val:
            best_val = val
            best_d = d
    if best_d == 0.0:
        return 0.0, best_val
    # Golden section refinement
    step = (hi_bound - lo_bound) / n_grid
    lo = max(lo_bound, best_d - 2 * step)
    hi = min(hi_bound, best_d + 2 * step)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    x1 = hi - gr * (hi - lo)
    x2 = lo + gr * (hi - lo)
    f1 = eval_swap(conv, base, quad, x1)
    f2 = eval_swap(conv, base, quad, x2)
    for _ in range(25):
        if f1 < f2:
            hi = x2
            x2 = x1
            f2 = f1
            x1 = hi - gr * (hi - lo)
            f1 = eval_swap(conv, base, quad, x1)
        else:
            lo = x1
            x1 = x2
            f1 = f2
            x2 = lo + gr * (hi - lo)
            f2 = eval_swap(conv, base, quad, x2)
    best_d = (lo + hi) / 2.0
    best_val = eval_swap(conv, base, quad, best_d)
    return best_d, best_val

def multi_swap_optimize(a, conv, peak_indices, n, rng, n_candidates=40, n_grid=40):
    """Try multiple candidate swaps and return the best one plus new state."""
    best_d = 0.0
    best_pair = None
    best_val = conv.max()
    active_set = set()
    for k in peak_indices[:6]:
        kk = int(k)
        js = max(0, kk - n + 1)
        je = min(n - 1, kk)
        for j in range(js, je + 1):
            active_set.add(j)
    active_list = list(active_set)
    non_active = [i for i in range(n) if i not in active_set]
    tried_pairs = set()
    for _ in range(n_candidates):
        if not active_list:
            break
        p = active_list[rng.integers(len(active_list))]
        if non_active and rng.random() < 0.75:
            q = non_active[rng.integers(len(non_active))]
        else:
            q = rng.integers(n)
        if p == q:
            continue
        pair_key = (min(p, q), max(p, q))
        if pair_key in tried_pairs:
            continue
        tried_pairs.add(pair_key)
        base, quad = build_swap_vectors(a, p, q)
        d, val = optimize_swap_fast(conv, a, p, q, base, quad, n_grid)
        if val < best_val:
            best_val = val
            best_d = d
            best_pair = (p, q)
    return best_pair, best_d, best_val

def peak_directed_descent(a0, time_limit, seed=42):
    """Main optimization loop using peak-directed pairwise swaps with annealing."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    conv = fast_convolve(a)
    best_a = a.copy()
    best_conv_max = conv.max()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    best_local_a = a.copy()
    best_local_obj = conv.max()
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        it += 1
        # Find top peaks
        n_peaks = min(15, 2 * n - 1)
        peak_idx = np.argpartition(conv, -n_peaks)[-n_peaks:]
        peak_idx = peak_idx[np.argsort(conv[peak_idx])[::-1]]
        # Generate and evaluate candidate swaps
        pair, d, val = multi_swap_optimize(a, conv, peak_idx, n, rng,
                                           n_candidates=35, n_grid=35)
        if pair is not None and abs(d) > 1e-13:
            p, q = pair
            a[p] -= d
            a[q] += d
            a = np.maximum(a, 0.0)
            ss = a.sum()
            if ss > 1e-15:
                a = a / ss
            conv = fast_convolve(a)
            cur_max = conv.max()
            if cur_max < best_local_obj - 1e-14:
                best_local_obj = cur_max
                best_local_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        # Track global best
        if best_local_obj < best_conv_max - 1e-14:
            best_conv_max = best_local_obj
            best_a = best_local_a.copy()
        # Annealing restart
        if no_improve > 300:
            scale = 0.002 + 0.004 * min(1.0, (no_improve - 300) / 400.0)
            a = project_simplex(best_local_a + rng.standard_normal(n) * scale)
            conv = fast_convolve(a)
            best_local_a = a.copy()
            best_local_obj = conv.max()
            no_improve = 0
        if no_improve > 1500:
            scale = 0.02
            a = project_simplex(best_local_a + rng.standard_normal(n) * scale)
            conv = fast_convolve(a)
            best_local_a = a.copy()
            best_local_obj = conv.max()
            no_improve = 0
    return best_a, best_conv_max

def batch_block_descent(a0, time_limit, seed=42, block_size=5):
    """Optimize blocks of coordinates jointly using inner coordinate descent."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    conv = fast_convolve(a)
    best_a = a.copy()
    best_obj = conv.max()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        it += 1
        peak_n = min(10, 2 * n - 1)
        peak_idx = np.argpartition(conv, -peak_n)[-peak_n:]
        peak_idx = peak_idx[np.argsort(conv[peak_idx])[::-1]]
        # Pick a block around the highest peak
        k_star = int(peak_idx[0])
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        block_range = list(range(js, je + 1))
        if len(block_range) > block_size:
            # Pick subset closest to peak center
            center = (js + je) // 2
            block_range.sort(key=lambda x: abs(x - center))
            block = block_range[:block_size]
        else:
            block = block_range
        # Inner coordinate descent on the block
        improved = False
        for _ in range(3):
            for j in block:
                if a[j] < 1e-12:
                    continue
                # Try moving mass from j to a random other coordinate
                others = [i for i in range(n) if i not in block]
                if not others:
                    continue
                q = others[rng.integers(len(others))]
                base, quad = build_swap_vectors(a, j, q)
                d, val = optimize_swap_fast(conv, a, j, q, base, quad, n_grid=30)
                if val < conv.max() - 1e-14 and abs(d) > 1e-13:
                    a[j] -= d
                    a[q] += d
                    a = np.maximum(a, 0.0)
                    ss = a.sum()
                    if ss > 1e-15:
                        a = a / ss
                    conv = fast_convolve(a)
                    improved = True
            if not improved:
                break
        if conv.max() < best_obj - 1e-14:
            best_obj = conv.max()
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 200:
            scale = 0.003 + 0.005 * min(1.0, no_improve / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            conv = fast_convolve(a)
            no_improve = 0
    return best_a, best_obj

def gradient_flow_descent(a0, time_limit, seed=42, lr=0.02, T=0.002):
    """Smoothed gradient descent using softmax-weighted peaks."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    conv = fast_convolve(a)
    best_a = a.copy()
    best_obj = conv.max()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.1:
            break
        it += 1
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        # Gradient: for each j, grad[j] = 2 * sum_k p[k] * a[k-j] (where valid)
        grad = np.zeros(n)
        for j in range(n):
            # Contributing k values: k in [j, j+n-1], term is a[k-j]
            ks = np.arange(j, j + n)
            ks = ks[(ks >= 0) & (ks < 2 * n - 1)]
            a_idx = ks - j
            grad[j] = 2.0 * np.dot(p[ks], a[a_idx])
        # Projected gradient step
        for trial_lr in [lr, lr * 0.5, lr * 0.25]:
            candidate = project_simplex(a - trial_lr * grad)
            c = fast_convolve(candidate)
            if c.max() < conv.max() - 1e-15:
                a = candidate
                conv = c
                break
        if conv.max() < best_obj - 1e-14:
            best_obj = conv.max()
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 50:
            T *= 1.1
            lr *= 0.95
        if no_improve > 200:
            noise = 0.003 + 0.004 * min(1.0, (no_improve - 200) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            conv = fast_convolve(a)
            T = 0.002
            lr = 0.02
            no_improve = 0
    return best_a, best_obj

def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search combining:
    1. Quick multi-n exploration
    2. Peak-directed pairwise swap descent (main engine)
    3. Gradient flow refinement
    4. Block coordinate descent polish
    5. Final fine-tuning with tiny swaps
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            try_seq(parent, "parent")
            print(f"Parent: n={len(parent)}, score={best_score:.8f}")
    except Exception as e:
        print(f"No parent: {e}")

    if best_seq is None:
        best_seq = np.ones(768) / 768
        try_seq(best_seq, "uniform768")

    # Phase 1: Quick exploration of nearby n values (~200s)
    print("\n=== Phase 1: Multi-n Exploration ===")
    t1 = time.monotonic()
    n_parent = len(best_seq)
    candidate_ns = sorted(set([n_parent, n_parent // 2, n_parent * 3 // 2, 384, 512, 1024]))
    candidate_ns = [n for n in candidate_ns if 128 <= n <= 2048]

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic()
        if remaining < 250:
            break
        t_alloc = min(remaining - 240, 40)
        if t_alloc < 5:
            break
        if n_try == n_parent:
            start = best_seq.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), best_seq)
            start = np.maximum(start, 0.0)
            start = start / start.sum()
        a_opt, _ = peak_directed_descent(start, time_limit=t_alloc, seed=seed + n_try)
        try_seq(a_opt, f"swap_n{n_try}")

    print(f"  Phase 1 done in {time.monotonic()-t1:.0f}s, best={best_score:.8f}")

    # Phase 2: Deep pairwise swap optimization (~900s)
    print("\n=== Phase 2: Deep Swap Descent ===")
    t2 = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    t_alloc = min(remaining, 900)
    if t_alloc > 30:
        a_opt, _ = peak_directed_descent(best_seq, time_limit=t_alloc, seed=seed + 100)
        try_seq(a_opt, "deep_swap")

    # Multiple restarts with perturbations
    remaining = deadline - time.monotonic()
    for scale in [0.005, 0.01, 0.02, 0.03]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        t_alloc = min(remaining, 120)
        perturbed = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
        a_opt, _ = peak_directed_descent(perturbed, time_limit=t_alloc, seed=seed + int(scale * 10000))
        try_seq(a_opt, f"swap_pert_{scale}")

    print(f"  Phase 2 done in {time.monotonic()-t2:.0f}s, best={best_score:.8f}")

    # Phase 3: Gradient flow refinement (~400s)
    print("\n=== Phase 3: Gradient Flow ===")
    t3 = time.monotonic()
    remaining = deadline - time.monotonic()
    if remaining > 100:
        t_alloc = min(remaining, 250)
        a_opt, _ = gradient_flow_descent(best_seq, time_limit=t_alloc, seed=seed + 200,
                                         lr=0.03, T=0.001)
        try_seq(a_opt, "grad_flow")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        t_alloc = min(remaining, 100)
        a_opt, _ = gradient_flow_descent(best_seq, time_limit=t_alloc, seed=seed + 300,
                                         lr=0.01, T=0.005)
        try_seq(a_opt, "grad_flow_warm")

    print(f"  Phase 3 done in {time.monotonic()-t3:.0f}s, best={best_score:.8f}")

    # Phase 4: Block coordinate descent (~200s)
    print("\n=== Phase 4: Block Descent ===")
    t4 = time.monotonic()
    remaining = deadline - time.monotonic()
    if remaining > 60:
        t_alloc = min(remaining, 150)
        a_opt, _ = batch_block_descent(best_seq, time_limit=t_alloc, seed=seed + 400,
                                       block_size=7)
        try_seq(a_opt, "block")

    print(f"  Phase 4 done in {time.monotonic()-t4:.0f}s, best={best_score:.8f}")

    # Phase 5: Final fine-tuning (~150s)
    print("\n=== Phase 5: Final Polish ===")
    t5 = time.monotonic()
    remaining = deadline - time.monotonic()
    if remaining > 30:
        t_alloc = min(remaining, 100)
        a_opt, _ = peak_directed_descent(best_seq, time_limit=t_alloc, seed=seed + 500)
        try_seq(a_opt, "final_swap")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        t_alloc = min(remaining, 30)
        a_opt, _ = gradient_flow_descent(best_seq, time_limit=t_alloc, seed=seed + 600,
                                         lr=0.005, T=0.0005)
        try_seq(a_opt, "final_grad")

    print(f"  Phase 5 done in {time.monotonic()-t5:.0f}s, best={best_score:.8f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
