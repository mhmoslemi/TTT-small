"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast pairwise coordinate descent
with analytical O(n) per-evaluation line search, combined with FFT-based global
refresh and smoothed gradient descent for final polish.

Key innovation: The change in auto-convolution from a pairwise swap (a[j]+=t, a[j']-=t)
can be computed in O(n) without FFT, enabling millions of exact 1D line searches.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_swap_delta_lin(a, j, jp, n, conv_len):

    Compute delta_linear[k] = 2*(a[k-j] - a[k-j']) for all k in [0, 2n-2].
    Out-of-bounds indices give 0.

    # a[k-j'] is nonzero for j' <= k <= j'+n-1
    
    delta = np.zeros(conv_len)
    
    # Contribution from a[k-j]: nonzero for k in [j, j+n-1]
    lo1 = max(0, j)
    hi1 = min(conv_len - 1, j + n - 1)
    if lo1 <= hi1:
        idx = np.arange(lo1, hi1 + 1)
        delta[lo1:hi1+1] += 2.0 * a[idx - j]
    
    # Contribution from -a[k-j']: nonzero for k in [j', j'+n-1]

    hi2 = min(conv_len - 1, jp + n - 1)
    if lo2 <= hi2:
        idx = np.arange(lo2, hi2 + 1)
        delta[lo2:hi2+1] -= 2.0 * a[idx - jp]
    
    return delta


def compute_swap_eps(j, jp, n, conv_len):
    """
    Compute eps[k] for the quadratic term.
    eps[2j] += 1, eps[j+j'] -= 2, eps[2j'] += 1 (if in range).
    """
    eps = np.zeros(conv_len)
    k1 = 2 * j
    k2 = j + jp
    k3 = 2 * jp
    if 0 <= k1 < conv_len:
        eps[k1] += 1.0
    if 0 <= k2 < conv_len:
        eps[k2] -= 2.0
    if 0 <= k3 < conv_len:
        eps[k3] += 1.0
    return eps


def line_search_swap(conv, a, j, jp, n, delta_lin, eps, t_min, t_max, n_evals=30):
    """
    Find t in [t_min, t_max] that minimizes max(conv + t*delta_lin + t^2*eps).
    Uses coarse-to-fine grid search.
    Returns (best_t, best_val).
    """
    # Coarse search
    ts = np.linspace(t_min, t_max, n_evals)
    best_val = np.inf
    best_t = ts[0]
    
    for t in ts:
        val = np.max(conv + t * delta_lin + t * t * eps)
        if val < best_val:
            best_val = val
            best_t = t
    
    # Fine search around best
    margin = (t_max - t_min) / n_evals
    lo = max(t_min, best_t - margin)
    hi = min(t_max, best_t + margin)
    ts_fine = np.linspace(lo, hi, 20)
    for t in ts_fine:
        val = np.max(conv + t * delta_lin + t * t * eps)
        if val < best_val:
            best_val = val
            best_t = t
    
    return best_t, best_val


def pairwise_cd(a0, time_limit, verbose=False):
    """
    Fast pairwise coordinate descent.
    a0: initial sequence (will be normalized to sum=1)
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    conv_len = 2 * n - 1
    
    conv = fast_convolve(a)
    best_a = a.copy()
    best_obj = conv.max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(42)
    
    # Precompute indices for efficiency
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.1:
            break
        
        # Find the peak
        k_peak = int(np.argmax(conv))
        
        # Find top contributors to the peak
        # a[i] * a[k_peak - i] for valid i
        lo_i = max(0, k_peak - n + 1)
        hi_i = min(n - 1, k_peak)
        if lo_i > hi_i:
            break
        contrib = a[lo_i:hi_i+1] * a[k_peak - hi_i:k_peak - lo_i:-1]
        
        # Source: index with largest contribution
        j_offset = int(np.argmax(contrib))
        j = lo_i + j_offset
        
        # Target: find j' that minimizes the "damage" (max contribution to any peak)
        # Approximate: minimize max over k of a[j'] * a[k-j']
        # This is approximately a[j'] * max_k a[k-j'] = a[j'] * max(a) (roughly)
        # Better: minimize the sum of contributions to top peaks
        
        # For efficiency, just try a few candidate targets
        # Candidate 1: the index with minimum a value
        # Candidate 2: random
        # Candidate 3: the index that minimizes a[k_peak - i] (opposite side of peak)
        
        candidates = []
        
        # Min a value (far from contributing to peaks)
        min_idx = int(np.argmin(a))
        candidates.append(min_idx)
        
        # Random
        candidates.append(int(rng.integers(n)))
        
        # Opposite side of peak
        j_opp = k_peak - j
        if 0 <= j_opp < n and j_opp != j:
            candidates.append(j_opp)
        
        # A few more random
        for _ in range(3):
            candidates.append(int(rng.integers(n)))
        
        best_move = None
        best_new_obj = conv.max()
        
        for jp in candidates:
            if jp == j:
                continue
            if a[jp] < 1e-15:
                continue  # Can't decrease further
            
            delta_lin = compute_swap_delta_lin(a, j, jp, n, conv_len)
            eps = compute_swap_eps(j, jp, n, conv_len)
            
            # t range: a[j] + t >= 0 => t >= -a[j], a[jp] - t >= 0 => t <= a[jp]
            t_min = -a[j]  # = -a[j] (we can increase j without limit in principle, but let's cap)
            t_max = a[jp]
            
            # Cap t to reasonable range
            t_max = min(t_max, 0.1)  # Don't move too much at once
            t_min = max(t_min, -0.1)
            
            if t_max <= t_min:
                continue
            
            best_t, best_val = line_search_swap(conv, a, j, jp, n, delta_lin, eps, t_min, t_max, n_evals=25)
            
            if best_val < best_new_obj - 1e-15:
                best_new_obj = best_val
                best_move = (j, jp, best_t, delta_lin, eps)
        
        if best_move is not None:
            j, jp, t, delta_lin, eps = best_move
            if abs(t) < 1e-15:
                no_improve += 1
                continue
            
            # Apply the swap
            a[j] += t
            a[jp] -= t
            a = np.maximum(a, 0)
            
            # Update conv
            conv = conv + t * delta_lin + t * t * eps
            # Slight normalization fix (should be negligible)
            s = a.sum()
            if abs(s - 1.0) > 1e-10:
                a = a / s
                # Recompute conv from scratch (rare)
                conv = fast_convolve(a)
            
            actual_obj = conv.max()
            
            if actual_obj < best_obj - 1e-14:
                best_obj = actual_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Periodic refresh every 10000 iterations to avoid drift
        if it % 10000 == 0:
            conv = fast_convolve(a)
        
        # Adaptivity
        if no_improve > 500:
            # Try a more aggressive move
            j = int(rng.integers(n))
            jp = int(rng.integers(n))
            if j != jp and a[jp] > 1e-6:
                delta_lin = compute_swap_delta_lin(a, j, jp, n, conv_len)
                eps = compute_swap_eps(j, jp, n, conv_len)
                t_min = max(-a[j], -0.05)
                t_max = min(a[jp], 0.05)
                if t_max > t_min:
                    best_t, best_val = line_search_swap(conv, a, j, jp, n, delta_lin, eps, t_min, t_max, n_evals=30)
                    if best_val < conv.max() - 1e-12:
                        a[j] += best_t
                        a[jp] -= best_t
                        a = np.maximum(a, 0)
                        conv = conv + best_t * delta_lin + best_t * best_t * eps
                        no_improve = 0
        
        if no_improve > 5000:
            # Perturb and restart
            noise_scale = 0.001 + 0.002 * min(1.0, (no_improve - 5000) / 5000.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            conv = fast_convolve(a)
            best_obj = conv.max()
            no_improve = 0
        
        if verbose and it % 50000 == 0:
            print(f"    PCD it={it}, obj={best_obj:.10f}, no_improve={no_improve}, elapsed={time.monotonic()-t0:.1f}s")
    
    return best_a, best_obj


def smoothed_grad_descent(a0, time_limit, T_init=0.0005, lr_init=0.05):
    """Smoothed gradient descent using softmax-weighted peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient: 2 * correlate(p, a)
        # (p convolved with a, then take the part of length n)
        grad = 2.0 * np.correlate(p, a, mode='valid')

        # Projected gradient with backtracking
        best_candidate = a
        best_val = conv.max()

        for trial_lr in [lr, lr * 0.5, lr * 0.25]:
            candidate = project_simplex(a - trial_lr * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        a = best_candidate
        obj = best_val

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 20:
            T *= 1.05
            lr *= 0.97
        if no_improve > 40:
            T *= 0.8
        if no_improve > 150:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise_scale = 0.002 + 0.004 * min(1.0, (no_improve - 150) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            T = T_init
            lr = lr_init
            no_improve = 0

    return best_a, best_obj


def fw_lp_optimize(f0, time_limit):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    bounds = [(0, None)] * n

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.1:
            break

        b_ub = np.full(2 * n - 1, M * m_exp)
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True})
            if res.success:
                g = np.maximum(res.x, 0.0)
            else:
                m_exp *= 1.2
                if m_exp > 4.0:
                    m_exp = 1.0
                continue
        except Exception:
            m_exp *= 1.2
            if m_exp > 4.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 40:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        best_alpha = 0.0
        best_val = M

        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)

        if no_improve > 60:
            rng = np.random.default_rng(it * 7919 + 13)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
            # Rebuild A_ub
            A_ub = np.zeros((2 * n - 1, n))
            for j in range(n):
                A_ub[j:j + n, j] = f

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Hybrid search combining:
    - Fast pairwise coordinate descent (O(n) per evaluation, no FFT)
    - Smoothed gradient descent for local refinement
    - FW-LP for global structure
    - Multi-n exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent_initial")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768
    n_parent = len(parent)

    # === Phase 1: Multi-n initial exploration (~100s) ===
    print("\n=== Phase 1: Quick Exploration ===")
    t_start = time.monotonic()

    candidate_ns = [n_parent]
    for n_try in [n_parent - 64, n_parent + 64, 640, 896]:
        if n_try not in candidate_ns and 128 <= n_try <= 2048:
            candidate_ns.append(n_try)

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1400
        if remaining < 15:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            start = start / start.sum()

        # Quick pairwise CD
        t_alloc = min(remaining, 25)
        a_opt, _ = pairwise_cd(start, time_limit=t_alloc)
        try_seq(a_opt, f"pcd_n{n_try}")

    print(f"  Phase 1 done (elapsed {time.monotonic()-t_start:.0f}s), best={best_score:.8f}")

    # === Phase 2: Intensive pairwise CD (~800s) ===
    print("\n=== Phase 2: Intensive Pairwise CD ===")
    t_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic() - 600
        if remaining > 30:
            t_alloc = min(remaining, 700)
            print(f"  PCD n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = pairwise_cd(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "pcd_deep")

    print(f"  Phase 2 done (elapsed {time.monotonic()-t_start:.0f}s), best={best_score:.8f}")

    # === Phase 3: Smoothed gradient + FW-LP (~400s) ===
    print("\n=== Phase 3: Gradient & FW ===")
    t_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Smoothed gradient with low temperature
        remaining = deadline - time.monotonic() - 250
        if remaining > 20:
            t_alloc = min(remaining, 150)
            print(f"  Smoothed grad T=0.0003, t={t_alloc:.0f}s")
            a_opt, _ = smoothed_grad_descent(best_seq, time_limit=t_alloc,
                                             T_init=0.0003, lr_init=0.02)
            try_seq(a_opt, "smooth_T0.0003")

        # Another smoothed gradient pass
        remaining = deadline - time.monotonic() - 150
        if remaining > 20:
            t_alloc = min(remaining, 100)
            print(f"  Smoothed grad T=0.001, t={t_alloc:.0f}s")
            a_opt, _ = smoothed_grad_descent(best_seq, time_limit=t_alloc,
                                             T_init=0001 - 0.001, lr_init=0.01)
            try_seq(a_opt, "smooth_T0.001")

        # FW-LP
        remaining = deadline - time.monotonic() - 80
        if remaining > 20:
            t_alloc = min(remaining, 80)
            print(f"  FW-LP, t={t_alloc:.0f}s")
