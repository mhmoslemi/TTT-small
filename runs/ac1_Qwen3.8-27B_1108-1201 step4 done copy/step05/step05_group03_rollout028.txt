Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Softmax parameterization + L-BFGS on power-sum smoothed objective

import numpy as np
from scipy.optimize import minimize, linprog
import time

# All helper functions at top level

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def next_pow2(n):
    """Next power of 2 >= n."""
    return 1 << (n - 1).bit_length()

def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax_obj(x, p, a_cache=None):
    """Compute log(f_p) where f_p = (sum C^p)^(1/p), C = conv(softmax(x), softmax(x))."""
    # Stable softmax
    xm = x - x.max()
    a = np.exp(xm)
    a_sum = a.sum()
    a = a / a_sum
    
    C = fast_convolve(a)
    C = np.maximum(C, 1e-300)
    log_C = np.log(C)
    p_log_C = p * log_C
    max_val = p_log_C.max()
    log_sum_exp = max_val + np.log(np.sum(np.exp(p_log_C - max_val)))
    log_fp = log_sum_exp / p
    return log_fp

def softmax_grad(x, p):
    """Compute gradient of log(f_p) w.r.t. x."""
    xm = x - x.max()
    a = np.exp(xm)
    a_sum = a.sum()
    a = a / a_sum
    n = len(a)
    
    C = fast_convolve(a)
    C = np.maximum(C, 1e-300)
    log_C = np.log(C)
    p_log_C = p * log_C
    max_val = p_log_C.max()
    exp_vals = np.exp(p_log_C - max_val)
    s = exp_vals.sum()  # = exp(log_sum_exp - max_val)
    
    # v[k] = C[k]^(p-1) / sum C^p
    # log_v[k] = (p-1)*log_C[k] - log_s where log_s = log_sum_exp
    log_s = max_val + np.log(s)
    # v[k] = exp((p-1)*log_C[k] - log_s)
    # For numerical stability:
    log_v = (p-1) * log_C - log_s
    # Clip to avoid underflow issues
    log_v_clipped = np.maximum(log_v, -700)
    v = np.exp(log_v_clipped)
    
    # Cross-correlation: grad_a[i] = 2 * sum_k v[k] * a[k-i]
    W_rev = v[::-1]
    fft_size = next_pow2(3 * n - 2)
    F = np.fft.ifft(np.fft.fft(W_rev, fft_size) * np.fft.fft(a, fft_size)).real
    G = F[n-1:2*n-1][::-1]
    grad_a = 2.0 * G
    
    # grad_x = a * (grad_a - dot(a, grad_a))
    dad = np.dot(a, grad_a)
    grad_x = a * (grad_a - dad)
    
    return grad_x

def lbfgs_optimize_a0(a0, p, time_limit, maxiter=200):
    """Optimize using L-BFGS on softmax-parameterized power-sum objective."""
    n = len(a0)
    a0 = np.maximum(a0, 1e-10)
    a0 = a0 / a0.sum()
    # Initialize x = log(a0)
    x0 = np.log(a0)
    
    best_x = x0.copy()
    best_f = softmax_obj(x0, p)
    
    t0 = time.monotonic()
    
    for attempt in range(3):
        if time.monotonic() - t0 > time_limit:
            break
        
        result = minimize(
            fun=lambda x: softmax_obj(x, p),
            x0=x0,
            jac=lambda x: softmax_grad(x, p),
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'gtol': 1e-10, 'maxfun': 500}
        )
        
        if result.fun < best_f:
            best_f = result.fun
            best_x = result.x.copy()
        
        # Check time and restart with perturbation
        if time.monotonic() - t0 > time_limit:
            break
        
        x0 = best_x + np.random.default_rng(attempt * 7 + 1).standard_normal(n) * 0.01
    
    # Convert back
    xm = best_x - best_x.max()
    a = np.exp(xm)
    a = a / a.sum()
    
    # True objective
    C = fast_convolve(a)
    obj = C.max()
    
    return a, obj

def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        rows = np.arange(j, min(j + n, 2 * n - 1))
        A_ub[rows, j] = f[:len(rows)]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0)
    return None

def lp_fw_optimize(f0, time_limit):
    """LP-based Frank-Wolfe with line search."""
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time_limit - (time.monotonic() - t0) < 3:
            break
        
        g = lp_direction(f, M, n)
        if g is None:
            g = lp_direction(f, M * 1.01, n)
            if g is None:
                no_improve += 1
                if no_improve > 30:
                    # Restart with perturbation
                    rng = np.random.default_rng(it * 31)
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                    conv = fast_convolve(f)
                    M = conv.max()
                    no_improve = 0
                continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                rng = np.random.default_rng(it * 17)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f_new = np.maximum(f + best_alpha * (g - f), 0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 50:
            rng = np.random.default_rng(it * 43)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0
    
    return best_f, best_obj

def pairwise_optimize(f0, time_limit):
    """Pairwise coordinate optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    offsets = [1, 2, 3, 5, 8, 13, 21]
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < 10:
        no_improve += 1
        improved = False
        order = rng.permutation(n)
        
        for idx in range(0, n, max(1, n // 200)):
            if time.monotonic() - t0 > time_limit:
                break
            i = order[idx]
            for offset in offsets:
                j = order[(idx + offset) % n]
                if i == j:
                    continue
                c = f[i] + f[j]
                if c < 1e-12:
                    continue
                
                best_t = f[i] / c
                best_val = best_obj
                
                for t in [0.0, 0.2, 0.4, 0.5, 0.6, 0.8, 1.0, best_t - 0.1, best_t + 0.1]:
                    t = np.clip(t, 0, 1)
                    ft = f.copy()
                    ft[i] = t * c
                    ft[j] = (1 - t) * c
                    val = fast_convolve(ft).max()
                    if val < best_val - 1e-14:
                        best_val = val
                        best_t = t
                
                if best_val < best_obj - 1e-13:
                    f[i] = best_t * c
                    f[j] = (1 - best_t) * c
                    f = f / f.sum()
                    best_obj = best_val
                    best_f = f.copy()
                    improved = True
        
        if improved:
            no_improve = 0
    
    return best_f, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp /= s
    return f_interp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def next_pow2(n):
    """Next power of 2 >= n."""
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
