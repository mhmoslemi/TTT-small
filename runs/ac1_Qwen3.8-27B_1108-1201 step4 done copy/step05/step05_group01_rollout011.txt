import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(a, conv, top_k=50):
    """Compute weighted subgradient from top-k peaks."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    # Get top-k peaks
    if k_actual == len(conv):
        peaks = np.arange(len(conv))
        weights = np.ones(len(peaks)) / len(peaks)
    else:
        # Use partition for efficiency
        top_indices = np.argpartition(conv, -k_actual)[-k_actual:]
        peaks = top_indices
        pv = conv[peaks]
        # Softmax weights
        max_pv = pv.max()
        weights = np.exp(pv / max(1e-10, max_pv) - 1)
        weights /= weights.sum()
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if weights[i] < 1e-14:
            continue
        k = int(peaks[i])
        # d conv[k]/d a[m] = 2*a[k-m] if 0<=k-m<n
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            sg[m_lo:m_hi+1] += 2.0 * weights[i] * a[k - np.arange(m_lo, m_hi + 1)]
    return sg

        a = a / s

    best_a = a.copy()

    eta = eta0
    it = 0
    no_improve = 0

        it += 1

        log_a = np.log(np.maximum(a, 1e-30))
        log_a -= eta * sg
        # Numerical stability
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            break
        a_new = a_new / a_new_sum
        
        # Evaluate
        conv = fast_convolve(a_new)
        val = conv.max()
        
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 100 == 0:

            a = project_simplex(best_a + noise)
            conv = fast_convolve(a)
            eta = eta0 * 0.5
            no_improve = 0
    
    return best_a, best_val

    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j+n, j] = f
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if res.success:
        return res.x
    return None

def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive strategies."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Solve LP with relaxed M
        g = solve_lp(f, M * m_relax, n)
        if g is None:
            m_relax *= 1.05
            if m_relax > 3.0:
                m_relax = 1.0
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        # Check if direction is useful
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 80:
                break
            m_relax = min(m_relax * 1.02, 2.0)
            continue
        
        # Line search: try multiple alphas
        best_alpha = 0.0
        best_val = M
        
        # FW step size and nearby values
        fw_alpha = 2.0 / (it + 2.0)
        candidates = [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, fw_alpha, 0.5, 0.7, 0.9, 1.0]
        candidates = sorted(set(max(0, min(1, c)) for c in candidates))
        
        for alpha in candidates:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Binary refinement
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_relax = 1.0
            f = f_new
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f} (t={elapsed:.0f}s)")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.03, 3.0)
        
        # Periodic restart
        if no_improve > 150:
            rng = np.random.default_rng(it * 31 + 17)
            scale = 0.002 + 0.002 * min(no_improve // 150, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.0
            no_improve = 0
    
    return best_f, best_obj
