**Peeling algorithm**:

2. The peak value is $p = (a*a)_{k^*} = \sum_i a_i a_{k^*-i}$.
3. To reduce $p$, we need to reduce the products $a_i a_{k^*-i}$.
4. For each pair $(i, k^*-i)$ that contributes, we can either:
   a. Reduce both $a_i$ and $a_{k^*-i}$ (but this reduces the sum)
   b. Transfer mass from the "heavy" side to the "light" side
5. Specifically, if $a_i > a_{k^*-i}$, transfer some mass from $a_i$ to $a_{k^*-i}$ (and vice versa). But this might create new peaks at other lags.

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using Nesterov-accelerated mirror descent
with adaptive peak weighting, block coordinate descent, active-set LP refinement,
and multi-length exploration. Combines multiple optimization strategies with

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def conv_peak(a):
    """Return the max of the auto-convolution."""
    return fast_convolve(a).max()


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_initialization(n, kind, rng):
    """Create various initializations."""
    x = np.arange(n)
    if kind == 0:
        a = np.ones(n)
    elif kind == 1:
        c = n / 2.0
        s = n / 4.0
        a = np.exp(-0.5 * ((x - c) / s) ** 2)
    elif kind == 2:
        a = np.exp(-x / (n / 3.0))
    elif kind == 3:
        a = np.sin(np.pi * x / n) ** 2 + 0.01
    elif kind == 4:
        a = 1.0 / (1.0 + (x / (n / 6.0)) ** 4)
    elif kind == 5:
        a = rng.random(n)
    elif kind == 6:
        # Triangular
        a = np.abs(1.0 - 2.0 * x / (n - 1.0)) + 0.01
    elif kind == 7:
        # Sine window
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1.0))) + 0.001
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def nesterov_subgradient(f0, time_limit, lr=None, momentum=0.93, top_k_frac=0.1,
                         temp=0.00005, lookahead_alpha=0.4):
    """Nesterov-accelerated subgradient with lookahead and temperature annealing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    if lr is None:
        lr = 0.003 * np.sqrt(n / 832.0)

    y = f.copy()  # lookahead point
    v = np.zeros(n)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temp
    cur_top_k = max(5, int(n * top_k_frac))

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Update lookahead
        y = f + lookahead_alpha * (y - f)

        # Compute gradient at lookahead point
        conv_y = fast_convolve(y)
        k_actual = min(cur_top_k, len(conv_y))
        peaks = np.argpartition(conv_y, -k_actual)[-k_actual:]

        pv = conv_y[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute weighted subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            kg = k - idx
            mask = (kg >= 0) & (kg < n)
            sg[idx[mask]] += 2.0 * w[i] * y[kg[mask]]

        # Nesterov momentum update
        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)

        # Evaluate
        conv = fast_convolve(f_new)
        co = conv.max()

        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new

        # Annealing
        if it % 200 == 0:
            cur_temp *= 1.015
            cur_lr *= 0.9997
            cur_top_k = max(3, int(cur_top_k * 0.98))

        # Periodic restart
        if no_improve > 600:
            rng = np.random.default_rng(int(time.monotonic() * 100000) % 999983)
            scale = 0.001 + 0.001 * min(no_improve // 200, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            y = f.copy()
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            cur_temp *= 0.5
            no_improve = 0

        if no_improve > 3000:
            break

    return best_f, best_obj


def lp_direction(f, M, n, top_peaks=15):
    """Solve LP to find direction g minimizing peak of (f+alpha(g-f))*g."""
    # Simplified: find g that minimizes max over top peaks of (f*g)_k
    # subject to g >= 0, sum(g) = 1
    conv = fast_convolve(f)
    peak_idx = np.argpartition(conv, -top_peaks)[-top_peaks:]

    c = np.ones(n)
    c[-1] = 0.0  # minimize t

    # Actually: min t s.t. sum_j f_{j} g_{k-j} <= t for k in peak_idx, g >= 0, sum g = 1
    n_con = len(peak_idx)
    A_ub = np.zeros((n_con, n + 1))
    for ii, k in enumerate(peak_idx):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[ii, j] = f[k - j]
        A_ub[ii, -1] = -1.0
    b_ub = np.zeros(n_con)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={"time_limit": 10, "presolve": True})
        if res.success:
            g = res.x[:n]
            gs = g.sum()
            if gs > 1e-15:
                g = g / gs
            else:
                g = np.ones(n) / n
            return g
    except Exception:
        pass
    return None


def lp_fw_step(f, n, max_lp_time=5.0):
    """One LP-FW step: find direction via LP, then line search."""
    t_start = time.monotonic()
    conv = fast_convolve(f)
    M = conv.max()

    g = lp_direction(f, M, n, top_peaks=20)
    if g is None:
        if time.monotonic() - t_start > max_lp_time:
            return f, M
        g = np.ones(n) / n

    g = np.maximum(g, 0)
    gs = g.sum()
    if gs < 1e-15:
        g = np.ones(n) / n
    else:
        g = g / gs

    # Line search
    best_alpha = 0.0
    best_val = M
    for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
        ft = f + alpha * (g - f)
        ft = np.maximum(ft, 0.0)
        st = ft.sum()
        if st < 1e-15:
            continue
        ft /= st
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha

    if best_alpha > 1e-12:
        # Refine
        lo = max(0.0, best_alpha - 0.05)
        hi = min(1.0, best_alpha + 0.05)
        for _ in range(8):
            if time.monotonic() - t_start > max_lp_time:
                break
            mid = (lo + hi) / 2.0
            ft = f + mid * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                break
            ft /= st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = mid
                lo = mid
            else:
                hi = mid

        f_new = f + best_alpha * (g - f)
        f_new = np.maximum(f_new, 0.0)
        sn = f_new.sum()
        if sn > 1e-15:
            f_new /= sn
        return f_new, best_val
    return f, M


def block_coordinate_descent(f0, time_limit, block_size=32):
    """Block coordinate descent: optimize blocks of coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        np.random.seed(passes * 7919)
        block_order = np.random.permutation(n // block_size + 1)

        for bi in block_order:
            if time.monotonic() - t0 > time_limit:
                break
            start = bi * block_size
            end = min(start + block_size, n)
            if start >= n:
                break
            bsize = end - start

            # Try perturbing this block
            best_block_obj = best_obj
            best_block_f = f.copy()

            # Gradient direction for this block
            conv = fast_convolve(f)
            top_k = min(10, len(conv))
            peaks = np.argpartition(conv, -top_k)[-top_k:]

            block_grad = np.zeros(bsize)
            for k in peaks:
                k = int(k)
                for j in range(start, end):
                    other = k - j
                    if 0 <= other < n:
                        block_grad[j - start] += 2.0 * f[other]

            # Try several step sizes
            for lr in [0.001, 0.003, 0.01, 0.03, 0.1]:
                new_block = f[start:end] - lr * block_grad
                new_block = np.maximum(new_block, 0)
                f_trial = f.copy()
                f_trial[start:end] = new_block
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st
                val = fast_convolve(f_trial).max()
                if val < best_block_obj - 1e-14:
                    best_block_obj = val
                    best_block_f = f_trial

            if best_block_obj < best_obj - 1e-14:
                f = best_block_f
                best_obj = best_block_obj
                best_f = f.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 5:
                break
        else:
            no_improve = 0

    return best_f, best_obj

                    if best_j_obj < best_obj - 1e-14:

                    if found:

        rem = t2_end - time.monotonic()
        if rem > 30:
            a_opt, _ = nesterov_subgradient(best_seq, time_limit=min(rem, 200),
                                            lr=0.004, momentum=0.93,
                                            top_k_frac=0.12, temp=0.00005,
                                            lookahead_alpha=0.4)
            try_seq(a_opt, "sg_main_1")

        # Second run: finer
        rem = t2_end - time.monotonic()
        if rem > 20:
            a_opt, _ = nesterov_subgradient(best_seq, time_limit=min(rem, 140),
                                            lr=0.0015, momentum=0.95,
                                            top_k_frac=0.08, temp=0.00002,
                                            lookahead_alpha=0.5)
            try_seq(a_opt, "sg_main_2")

    # === Phase 3: LP-FW refinement (250s) ===
    print("\n=== Phase 3: LP-FW (budget=250s) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        t3_end = time.monotonic() + 250.0
        f = best_seq.copy()
        conv = fast_convolve(f)
        M = conv.max()
        no_improve = 0
        lp_it = 0
        while time.monotonic() < t3_end and no_improve < 30:
            f, M = lp_fw_step(f, n_best, max_lp_time=8.0)
            lp_it += 1
            if M < best_score / (2.0 * n_best) - 1e-15:
                try_seq(f, "lpfw_{}".format(lp_it))
                no_improve = 0
            else:
                no_improve += 1
            # Occasionally add perturbation
            if lp_it % 10 == 0 and no_improve > 5:
                pert = rng.standard_normal(n_best) * 0.001
                f = project_simplex(f + pert)
                no_improve = 0

    # === Phase 4: Block coordinate descent (200s) ===
    print("\n=== Phase 4: Block CD (budget=200s) ===")
    if best_seq is not None:
        t4_end = time.monotonic() + 200.0
        for bsize in [16, 32, 64]:
            rem = t4_end - time.monotonic()
            if rem < 15:
                break
            a_opt, _ = block_coordinate_descent(best_seq, time_limit=min(rem, 70),
                                                block_size=bsize)
            try_seq(a_opt, "bcd_{}".format(bsize))

    # === Phase 5: Multi-length exploration (350s) ===
    print("\n=== Phase 5: Multi-length (budget=350s) ===")
    if best_seq is not None:
        t5_end = time.monotonic() + 350.0
        base = best_seq
        candidate_lengths = []
        for delta in [-128, -96, -64, -48, -32, -16, 16, 32, 48, 64, 96, 128, 192, 256]:
            ln = len(base) + delta
            if 256 <= ln <= 2048:
                candidate_lengths.append(ln)
        # Sort by proximity to current best length
        candidate_lengths.sort(key=lambda x: abs(x - len(base)))

        explored = set()
        for target_n in candidate_lengths[:8]:
            if target_n in explored:
                continue
            explored.add(target_n)
            rem = t5_end - time.monotonic()
            if rem < 40:
                break
            f_interp = interpolate_sequence(base, target_n)
            t_alloc = min(rem * 0.9, 80)
            a_opt, _ = nesterov_subgradient(f_interp, time_limit=t_alloc,
                                            lr=0.003 * np.sqrt(target_n / 832.0),
                                            momentum=0.93,
                                            top_k_frac=0.1, temp=0.00004)
            try_seq(a_opt, "len_{}".format(target_n))

    # === Phase 6: Coordinate descent polish (200s) ===
    print("\n=== Phase 6: CD Polish (budget=200s) ===")
    if best_seq is not None:
        t6_end = time.monotonic() + 200.0
        a_opt, _ = coordinate_descent_polish(best_seq, time_limit=min(120, t6_end - time.monotonic()))
        try_seq(a_opt, "cd_polish_1")

        rem = t6_end - time.monotonic()
        if rem > 30:
            a_opt, _ = coordinate_descent_polish(best_seq, time_limit=min(rem, 80))
            try_seq(a_opt, "cd_polish_2")

    # === Phase 7: Final fine-tuning with random restarts (remaining time) ===
    print("\n=== Phase 7: Final fine-tune ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            n_best = len(best_seq)
            # Multiple short restarts
            n_restarts = min(8, max(2, int(rem / 30)))
            restart_time = max(10, (rem - 10) / n_restarts)
            for ri in range(n_restarts):
                rem2 = deadline - time.monotonic()
                if rem2 < 10:
                    break
                scale = 0.0005 + 0.0005 * ri
                noise = rng.standard_normal(n_best) * scale
                f_pert = project_simplex(best_seq + noise)
                t_alloc = min(restart_time, rem2 - 5)
                if t_alloc < 5:
                    break
                a_opt, _ = nesterov_subgradient(f_pert, time_limit=t_alloc,
                                                lr=0.001, momentum=0.95,
                                                top_k_frac=0.05, temp=0.00001,
                                                lookahead_alpha=0.5)
                try_seq(a_opt, "final_restart_{}".format(ri))

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        else:
            best_seq = np.ones(512) / 512
            best_score = float(evaluate_sequence(best_seq.tolist()))
    except Exception:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print("\nFINAL SCORE: {:.8f} (n={})".format(best_score, len(best_seq)))
    return [float(x) for x in best_seq]
