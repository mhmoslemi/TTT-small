    a = np.exp(theta_centered)
    a /= a.sum()

    c = fast_convolve(a)  # length 2n-1

    Tc_max = Tc.max()

    phi = log_sum_exp / T  # smooth max value

    p = np.exp(Tc - Tc_max)
    p /= p.sum()
    
    # Gradient w.r.t. a

    # Full objective: 2n * phi / (sum a)^2 = 2n * phi (since sum a = 1)
    obj = 2.0 * n * phi

import numpy as np

import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def get_objective(a):
    """Compute the objective value for a non-negative sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    c = fast_convolve(a_norm)
    return 2.0 * n * c.max()

def smooth_max_obj_grad(theta, T, n):
    """
    Compute smooth max objective and gradient w.r.t. theta (softmax params).
    Returns (obj, grad).
    """
    # Softmax
    tc = theta - theta.max()
    exp_tc = np.exp(tc)
    a = exp_tc / exp_tc.sum()
    
    # Auto-convolution
    c = fast_convolve(a)
    
    # Smooth max
    Tc = T * c
    Tc_max = Tc.max()
    exp_shifted = np.exp(Tc - Tc_max)
    log_sum_exp = Tc_max + np.log(exp_shifted.sum())
    phi = log_sum_exp / T
    
    # Weights p_k
    p = exp_shifted / exp_shifted.sum()
    
    # Gradient w.r.t. a: (grad_a)_i = 2 * sum_k p[k] * a[k-i]
    # Computed as 2 * (p * a_rev)[n-1:2n-1]
    a_rev = a[::-1]
    conv_len = (2 * n - 1) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    AR = np.fft.rfft(a_rev, fft_size)
    conv_pa = np.fft.irfft(P * AR, fft_size)[:conv_len]
    grad_a = 2.0 * conv_pa[n-1 : 2*n-1]
    
    # Gradient w.r.t. theta
    dot_a = np.dot(a, grad_a)
    grad_theta = 2.0 * n * a * (grad_a - dot_a)
    
    obj = 2.0 * n * phi
    return obj, grad_theta

    best_a = a_init.copy()

            lambda th, T=T: smooth_max_obj_grad(th, T, n)[0],

        y = a_new + (v - v + 0)  # simplified: y = a_new for now
        # Actually, proper Nesterov:
        beta = (1 + np.sqrt(1 + 4 * momentum**2)) / 2
        y = a_new + (momentum / beta) * (a_new - a)
        
        a = a_new
        T *= T_growth
        
        if no_improve > 500:
            # Restart
            rng = np.random.default_rng(it * 12345 + 67)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            y = a.copy()
            T = T_start
            no_improve = 0
    
    return best_a, best_obj

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_min_peak(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        g = res.x[:n]
        g = np.maximum(g, 0)
        gs = g.sum()
        if gs > 1e-15:
            g /= gs
        else:
            g = np.ones(n) / n
        return g
    return None

def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        g = solve_lp_min_peak(f, n)
        if g is None:
            g = np.ones(n) / n
        
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft /= st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-12:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new /= f_new.sum()
            f = f_new
            M = best_val
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 40:
            rng = np.random.default_rng(it * 7919 + 13)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            M = fast_convolve(f).max()
            no_improve = 0
    
    return best_f, best_obj

def cosine_init(n, alpha=1.0):
    """Create a cosine-windowed initialization."""
    x = np.arange(n) / n
    a = np.cos(np.pi * (x - 0.5)) ** (2 * alpha)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def gaussian_init(n, sigma=0.3):
    """Create a Gaussian-windowed initialization."""
    x = np.arange(n) / n - 0.5
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
