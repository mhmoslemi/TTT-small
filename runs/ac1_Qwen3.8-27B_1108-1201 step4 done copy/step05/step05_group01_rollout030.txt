import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fast_grad_convolve(a, peak_positions):
    """Compute gradient of max_k conv(a,a)[k] (weighted) at given peaks."""
    n = len(a)
    grad = np.zeros(n)
    for k in peak_positions:
        k = int(k)
        # d conv[k]/d a[m] = 2*a[k-m] if 0 <= k-m < n
        m_lo = max(0, k - (n - 1))
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            indices = np.arange(m_lo, m_hi + 1)
            grad[indices] += 2.0 * a[k - indices]
    return grad

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

- Greedy "transfer" steps targeting worst autocorrelation peaks

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_weighted_grad(a, peaks, weights):
    """Compute weighted subgradient of sum_k w_k * r[k] where r = a*a."""
    n = len(a)
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        k = int(peaks[idx])
        w = weights[idx]
        if w < 1e-15:

        if m_lo <= m_hi:
            indices = np.arange(m_lo, m_hi + 1)
            grad[indices] += w * 2.0 * a[k - indices]
    return grad

def transfer_optimize(a0, time_limit, verbose=False):
    """Greedy transfer optimization: move mass from high-contribution to low-contribution positions."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    r = fast_convolve(a)
    best_obj = r.max()
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        r = fast_convolve(a)
        r_max = r.max()
        peak_k = int(np.argmax(r))

        # Compute contribution of each position to the peak
        m_lo = max(0, peak_k - (n - 1))
        m_hi = min(n - 1, peak_k)
        if m_lo > m_hi:
            break
        indices = np.arange(m_lo, m_hi + 1)
        contrib = a[peak_k - indices]  # contribution of a[m] to r[peak_k]

        # Find positions with high contribution (to decrease) and low contribution (to increase)
        # Only consider positions that already have mass (to decrease) and those with low values (to increase)
        above_mean = contrib > contrib.mean()
        below_mean = contrib < contrib.mean()

        if not above_mean.any() or not below_mean.any():
            break

        high_pos = indices[above_mean]
        low_pos = indices[below_mean]

        # Transfer a small amount from the highest contributor to the lowest
        transfer_amount = min(a[high_pos[0]] * 0.1, a[low_pos[0]] * 0.5, 0.01 / n)

        if transfer_amount < 1e-12:
            # Try a smaller transfer
            transfer_amount = 1e-6 / n
            if transfer_amount >= a[high_pos[0]]:
                break

        a[high_pos[0]] -= transfer_amount
        a[low_pos[0]] += transfer_amount

        r_new = fast_convolve(a)
        obj_new = r_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
        else:
            # Revert
            a[high_pos[0]] += transfer_amount
            a[low_pos[0]] -= transfer_amount
            no_improve += 1

        if no_improve > 200:
            # Try subgradient step instead
            r = fast_convolve(a)
            r_mean = r.mean()
            top_k = min(10, len(r))
            top_idx = np.argpartition(r, -top_k)[-top_k:]
            excess = np.maximum(r[top_idx] - r_mean, 0)
            if excess.sum() < 1e-15:
                weights = np.ones(len(top_idx)) / len(top_idx)
            else:
                weights = excess / excess.sum()
            grad = compute_weighted_grad(a, top_idx, weights)
            lr = 0.5 / np.sqrt(n)
            a = project_simplex(a - lr * grad)
            r_new = fast_convolve(a)
            if r_new.max() < best_obj:
                best_obj = r_new.max()
                best_a = a.copy()
            no_improve = 0

        if no_improve > 1000:
            break

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      transfer iter {it}, best={best_obj:.10f}, elapsed={elapsed:.1f}s")

    return best_a, best_obj


def coordinate_descent(a0, time_limit, verbose=False):
    """Coordinate descent with golden section search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()

    t0 = time.monotonic()
    GR = (1.0 + np.sqrt(5.0)) / 2.0
    inv_gr = 1.0 / GR
    no_pass = 0

    while time.monotonic() - t0 < time_limit:
        if no_pass >= 3:
            break
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-14:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            # Golden section: vary a[j], scale others to maintain sum=1
            lo = 0.0
            hi = min(0.9999, max(2.0 * old_val, 0.05 / n))

            mask = np.ones(n, dtype=bool)
            mask[j] = False

            def eval_coord(x):
                ft = a.copy()
                ft[j] = x
                ft[mask] = a[mask] * (1.0 - x) / denom
                return fast_convolve(ft).max()

            c1 = lo + inv_gr * (hi - lo)
            c2 = lo + (1.0 - inv_gr) * (hi - lo)
            f1 = eval_coord(c1)
            f2 = eval_coord(c2)

            for _ in range(10):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + inv_gr * (hi - lo)
                    f1 = eval_coord(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + (1.0 - inv_gr) * (hi - lo)
                    f2 = eval_coord(c2)

            opt_x = (lo + hi) / 2.0
            opt_val = min(f1, f2)
            cur_val = fast_convolve(a).max()

            if opt_val < cur_val - 1e-13:
                a[j] = opt_x
                a[mask] = a[mask] * (1.0 - opt_x) / denom
                a = a / a.sum()
                new_obj = fast_convolve(a).max()
                if new_obj < best_obj - 1e-12:
                    best_obj = new_obj
                    best_a = a.copy()
                pass_improved = True

        if pass_improved:
            no_pass = 0
        else:
            no_pass += 1

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def cosine_init(n, freq, amp, phase=0.0):
    """Cosine-modulated initialization."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def alt_multiplier_init(n, ratio=0.5):
    """Alternating high/low initialization."""
    a = np.ones(n)
    a[::2] = ratio
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def propose_candidate():
    """
    Multi-phase optimization combining subgradient, transfer, and coordinate descent
    with multi-length exploration.
    """
    deadline = time.monotonic() + 1740.0

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512

    best_seq = parent.copy()
    best_score = float(evaluate_sequence(parent.tolist()))
    print(f"Initial score: {best_score:.6f} (n={len(best_seq)})")

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a = a / s
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # === Phase 1: Multi-length quick exploration ===
    print("\n=== Phase 1: Multi-length exploration ===")
    candidate_lengths = [640, 768, 1024, 1280, 1536]

    for n_cand in candidate_lengths:
        rem = deadline - time.monotonic() - 600
        if rem < 20:
            break

        # Interpolate parent
        a_init = interpolate_sequence(parent, n_cand)
        s = try_seq(a_init, f"interp_n{n_cand}")

        # Short subgradient
        t_alloc = min(rem * 0.15, 60)
        if t_alloc > 10:
            print(f"  Subgrad n={n_cand}, t={t_alloc:.0f}s")
            a_opt, _ = subgrad_optimize(a_init, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, f"subgrad_n{n_cand}")

    # === Phase 2: Different initializations at best length ===
    print(f"\n=== Phase 2: Multi-init (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic() - 500
    if rem > 30:
        inits = [
            cosine_init(n_best, 1, 0.3),
            cosine_init(n_best, 2, 0.2),
            cosine_init(n_best, 3, 0.15),
            cosine_init(n_best, 1, 0.4, np.pi / 3),
            alt_multiplier_init(n_best, 0.4),
            alt_multiplier_init(n_best, 0.6),
        ]
        for i, a_init in enumerate(inits):
            rem = deadline - time.monotonic() - 400
            if rem < 15:
                break
            s = try_seq(a_init, f"init_{i}")
            if np.isfinite(s) and s < best_score + 0.3:
                t_alloc = min(rem * 0.2, 50)
                if t_alloc > 8:
                    a_opt, _ = subgrad_optimize(a_init, time_limit=t_alloc)
                    try_seq(a_opt, f"init_opt_{i}")

    # === Phase 3: Deep optimization on best ===
    print(f"\n=== Phase 3: Deep optimization (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 300
    if rem > 30:
        t_sub = min(rem * 0.5, 300)
        print(f"  Deep subgrad n={n_best}, t={t_sub:.0f}s")
        a_opt, _ = subgrad_optimize(best_seq, time_limit=t_sub, verbose=True)
        try_seq(a_opt, "deep_subgrad")

    # === Phase 4: Transfer optimization ===
    print(f"\n=== Phase 4: Transfer optimization (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 200
    if rem > 20:
        t_tr = min(rem * 0.6, 150)
        print(f"  Transfer t={t_tr:.0f}s")
        a_opt, _ = transfer_optimize(best_seq, time_limit=t_tr, verbose=True)
        try_seq(a_opt, "transfer")

    # === Phase 5: Subgradient round 2 ===
    print(f"\n=== Phase 5: Subgrad round 2 (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_sub2 = min(rem * 0.6, 200)
            print(f"  Subgrad2 t={t_sub2:.0f}s")
            a_opt, _ = subgrad_optimize(best_seq, time_limit=t_sub2, verbose=True)
            try_seq(a_opt, "subgrad2")

    # === Phase 6: Coordinate descent polish ===
    print(f"\n=== Phase 6: Coordinate descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 15:
            t_cd = min(rem * 0.7, 120)
            print(f"  CD t={t_cd:.0f}s, n={len(best_seq)}")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "coord_desc")

    # === Phase 7: Try longer length interpolation ===
    print(f"\n=== Phase 7: Longer length (best={best_score:.6f}) ===")
    if best_seq is not None:
        for target_n in [2048, 2560]:
            rem = deadline - time.monotonic() - 80
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            s = try_seq(f_interp, f"interp_long_n{target_n}")
            if np.isfinite(s) and s < best_score + 0.1:
                t_alloc = min(rem * 0.4, 80)
                if t_alloc > 10:
                    a_opt, _ = subgrad_optimize(f_interp, time_limit=t_alloc)
                    try_seq(a_opt, f"long_n{target_n}")

    # === Phase 8: Final fine-tuning ===
    print(f"\n=== Phase 8: Final polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 60
        if rem > 10:
            # Tiny subgradient
            t_final = min(rem * 0.4, 40)
            a_opt, _ = subgrad_optimize(best_seq, time_limit=t_final)
            try_seq(a_opt, "final_subgrad")

        rem = deadline - time.monotonic() - 30
        if rem > 8:
            # Tiny coordinate descent
            t_cd2 = min(rem * 0.5, 25)
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd2)
            try_seq(a_opt, "final_cd")

        # Perturbation restarts
        rng = np.random.default_rng(12345)
        rem = deadline - time.monotonic() - 15
        for _ in range(3):
            if rem < 8:
                break
            noise = rng.standard_normal(n_best) * 0.003
            f_pert = project_simplex(best_seq + noise)
            t_pert = min(rem * 0.3, 15)
            a_opt, _ = subgrad_optimize(f_pert, time_limit=t_pert)
            try_seq(a_opt, "perturb")

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
