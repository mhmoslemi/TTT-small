```python
"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using iterative reweighted QP,
LP-based Frank-Wolfe, subgradient descent, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.linalg import toeplitz
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_qp_simplex(Q, n):
    """Solve min a^T Q a s.t. a >= 0, sum(a) = 1 using CVXPY."""
    a = cp.Variable(n)
    objective = cp.Minimize(cp.quad_form(a, cp.psd_wrap(Q)))
    constraints = [a >= 0, cp.sum(a) == 1]
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.OSQP, eps_abs=1e-8, eps_rel=1e-8, max_iter=10000)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, eps=1e-8)
        except Exception:
            return project_simplex(np.ones(n) / n)
    if a.value is not None:
        result = np.maximum(a.value, 0)
        s = result.sum()
        if s > 1e-10:
            return result / s
    return project_simplex(np.ones(n) / n)


def reweighted_qp_optimize(a0, time_limit, n_peaks=5, p=2.0, verbose=False):
    """Iterative reweighted QP to minimize max autocorrelation."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        cmax = conv.max()
        
        if cmax < 1e-15:
            break
        
        # Select top peaks
        k_select = min(n_peaks, len(conv))
        peak_idx = np.argpartition(conv, -k_select)[-k_select:]
        peak_vals = conv[peak_idx]
        
        # Weights (emphasize peaks)
        w = (peak_vals / peak_vals.max()) ** p
        
        # Form Q (sparse banded approximation)
        Q = np.zeros((n, n))
        for i in range(len(peak_idx)):
            k = int(peak_idx[i])
            if k == 0:
                Q[np.diag_indices(n)] += w[i]
            elif k <= n - 1:
                idx = np.arange(n - k)
                Q[idx, idx + k] += w[i] / 2
                Q[idx + k, idx] += w[i] / 2
            else:
                # k > n-1: map to effective lag
                k_eff = 2 * n - 2 - k
                if k_eff <= n - 1:
                    idx = np.arange(n - k_eff)
                    Q[idx, idx + k_eff] += w[i] / 2
                    Q[idx + k_eff, idx] += w[i] / 2
        
        # Ensure PSD: add alpha * I
        alpha = float(np.sum(w)) + 1e-10
        Q[np.diag_indices(n)] += alpha
        
        # Solve QP
        a_new = solve_qp_simplex(Q, n)
        
        # Evaluate
        conv_new = fast_convolve(a_new)
        new_cmax = conv_new.max()
        
        if new_cmax < best_obj - 1e-12:
            best_obj = new_cmax
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 10 == 0:
                print(f"    RQP iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
        
        # Update a (with some mixing to avoid oscillation)
        a = 0.7 * a_new + 0.3 * a
        a = np.maximum(a, 0)
        ss = a.sum()
        if ss > 1e-15:
            a = a / ss
        
        if no_improve > 30:
            # Restart with perturbation
            rng = np.random.default_rng(it * 31 + 7)
            noise = rng.standard_normal(n) * 0.005
            a = project_simplex(best_a + noise)
            no_improve = 0
            # Increase n_peaks for diversity
            n_peaks = min(n_peaks + 2, 20)
            p = min(p * 1.1, 5.0)
    
    return best_a, best_obj
```

```python
def solve_qp_active_set(Q, n):
    """Solve min a^T Q a s.t. a >= 0, sum(a) = 1 using active set method."""
    # Start with all variables active (free)
    active = np.ones(n, dtype=bool)
    
    for iteration in range(n + 10):
        # Solve the equality-constrained QP on active set
        iact = np.where(active)[0]
        Q_sub = Q[np.ix_(iact, iact)]
        c_sub = np.ones(len(iact))
        
        try:
            Q_inv_c = np.linalg.solve(Q_sub, c_sub)
            denom = c_sub @ Q_inv_c
            if denom <= 1e-15:
                break
            a_sub = Q_inv_c / denom
        except np.linalg.LinAlgError:
            # Q_sub is singular, add regularization
            Q_sub_reg = Q_sub + 1e-10 * np.eye(len(iact))
            Q_inv_c = np.linalg.solve(Q_sub_reg, c_sub)
            denom = c_sub @ Q_inv_c
            a_sub = Q_inv_c / denom
        
        # Check nonnegativity
        if np.all(a_sub >= -1e-10):
            a = np.zeros(n)
            a[iact] = np.maximum(a_sub, 0)
            s = a.sum()
            if s > 1e-15:
                a /= s
            return a
        
        # Find the most violated constraint
        worst = np.argmin(a_sub)
        active[iact[worst]] = False
        
        if not active.any():
            return np.ones(n) / n
    
    # Fallback
    return np.ones(n) / n
```

```python
def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)

def make_Q(peak_idx, w, n):
    """Form the matrix Q = sum_i w_i * T_{peak_idx[i]} + alpha*I."""
    Q = np.zeros((n, n))
```

```python
def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_min_peak(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t, g>=0, sum(g)=1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None

            g = np.ones(n) / n
        else:
            g = g / gs

        if np.max(np.abs(g - f)) < 1e-13:
            no_improve += 1
            if no_improve > 50:
                break
            continue

        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.06)
            hi = min(1.0, best_alpha + 0.06)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = np.maximum(f + mid * (g - f), 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-12:
            f_new = np.maximum(f + best_alpha * (g - f), 0.0)
            sn = f_new.sum()
            f_new = f_new / sn if sn > 1e-15 else np.ones(n) / n
            f = f_new
            M = best_val
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"      LP-FW it {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 60:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.002 + 0.003 * min(no_improve // 50, 4)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_obj


def make_peak_Q(peak_idx, w, n):
    """Build sparse-ish Q from peak lag indices and weights, with PSD shift."""
    Q = np.zeros((n, n))
    for i in range(len(peak_idx)):
        k = int(peak_idx[i])
        wi = w[i]
        if k == 0:
            Q[np.diag_indices(n)] += wi
        elif k <= n - 1:
            idx = np.arange(n - k)
            Q[idx, idx + k] += wi * 0.5
            Q[idx + k, idx] += wi * 0.5
        else:
            k_eff = 2 * n - 2 - k
            if 0 <= k_eff <= n - 1:
                idx = np.arange(n - k_eff)
                Q[idx, idx + k_eff] += wi * 0.5
                Q[idx + k_eff, idx] += wi * 0.5
    alpha = float(np.sum(w)) + 1e-10
    Q[np.diag_indices(n)] += alpha
    return Q


def reweighted_pg(a0, time_limit, n_peaks=6, p=2.0, inner_iters=300, verbose=False):
    """Reweighted projected gradient with Nesterov acceleration."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    outer = 0
    no_improve = 0
    cur_n_peaks = n_peaks
    cur_p = p

    while time.monotonic() - t0 < time_limit:
        outer += 1
        if time.monotonic() - t0 > time_limit * 0.97:
            break

        conv = fast_convolve(a)
        cmax = conv.max()
        if cmax < 1e-16:
            break

        k_sel = min(cur_n_peaks, len(conv))
        peak_idx = np.argpartition(conv, -k_sel)[-k_sel:]
        peak_vals = conv[peak_idx]
        w = (peak_vals / peak_vals.max()) ** cur_p

        Q = make_peak_Q(peak_idx, w, n)

        # Estimate Lipschitz constant via power iteration
        rng_local = np.random.default_rng(outer * 31)
        v = rng_local.standard_normal(n)
        vn = np.linalg.norm(v)
        if vn < 1e-15:
            v = np.ones(n) / n
        else:
            v /= vn
        for _ in range(15):
            wv = Q @ v
            wn = np.linalg.norm(wv)
            if wn < 1e-15:
                break
            v = wv / wn
        L = 2.0 * (v @ (Q @ v))
        if L < 1e-10:
            L = 2.0 * (float(np.sum(w)) + 1.0)
        step_size = 1.0 / max(L, 1e-10)
        step_size = min(step_size, 0.5)

        # Nesterov accelerated projected gradient
        a_curr = a.copy()
        a_prev = a.copy()
        f_curr = a_curr @ Q @ a_curr

        for inner in range(inner_iters):
            if time.monotonic() - t0 > time_limit * 0.98:
                break
            mu = 1.0 + 1.0 / np.sqrt(float(inner + 1))
            a_y = a_curr + (mu - 1.0) / mu * (a_curr - a_prev)
            grad = 2.0 * (Q @ a_y)
            a_new = project_simplex(a_curr - step_size * grad)
            f_new = a_new @ Q @ a_new
            if f_new > f_curr + 1e-6:
                # Divergence guard: halve step and retry
                a_new = project_simplex(a_curr - step_size * 0.5 * grad)
                f_new = a_new @ Q @ a_new
            a_prev = a_curr
            a_curr = a_new
            f_curr = f_new

        a = a_curr
        conv = fast_convolve(a)
        new_obj = conv.max()

        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and outer % 10 == 0:
                print(f"      RPG outer {outer}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        if no_improve > 15:
            rng = np.random.default_rng(outer * 104729 + 7)
            scale = 0.003 + 0.002 * min(no_improve // 8, 5)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            no_improve = 0
            cur_n_peaks = min(cur_n_peaks + 2, 20)
            cur_p = min(cur_p * 1.15, 6.0)

    return best_a, best_obj


def sg_optimize(f0, time_limit, lr=0.003, momentum=0.94, top_k=40, temp=0.00005,
                verbose=False):
    """Subgradient with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit * 0.97:
            break

        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            kg = k - idx
            mask = (kg >= 0) & (kg < n)
            sg[idx[mask]] += 2.0 * w[i] * f[kg[mask]]

        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)
        conv = fast_convolve(f_new)
        co = conv.max()

        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 200 == 0:
                print(f"      SG it {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            f = f_new

        cur_lr *= 0.9999
        cur_temp *= 1.003

        if no_improve > 300:
            rng = np.random.default_rng(it * 6151 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def generate_initializations(n, rng):
    """Generate diverse initializations."""
    inits = []
    # Uniform
    inits.append(np.ones(n) / n)
    # Triangle
    tri = np.concatenate([np.linspace(0, 1, n // 2 + 1), np.linspace(1, 0, n - n // 2 - 1)])
    tri = np.maximum(tri, 1e-6)
    inits.append(tri / tri.sum())
    # Cosine
    cos = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    cos = np.maximum(cos, 1e-6)
    inits.append(cos / cos.sum())
    # Gaussian bump
    center = n / 2
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / (n / 4)) ** 2)
    gauss = np.maximum(gauss, 1e-6)
    inits.append(gauss / gauss.sum())
    # Random
    r = rng.standard_normal(n)
    r = np.maximum(r, 0)
    if r.sum() < 1e-10:
        r = np.ones(n)
    inits.append(r / r.sum())
    # Beta-distributed
    beta = rng.beta(2, 2, size=n)
    inits.append(beta / beta.sum())
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining LP-FW, reweighted projected gradient,
    subgradient descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 35.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # Get parent / starting point
    parent = None
    parent_n = 832
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
            parent_n = len(parent)
    except Exception:
        pass

    if parent is None:
        parent = np.ones(832) / 832
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.8f}")

    base_n = len(parent)

    # === Phase 1: LP-FW on parent ===
    print(f"\n=== Phase 1: LP-FW (n={base_n}) ===")
    rem = deadline - time.monotonic()
    t1 = min(350, rem * 0.25)
    if t1 > 30:
        print(f"  time={t1:.0f}s")
        a_opt, _ = lp_fw_optimize(parent, time_limit=t1, verbose=True)
        try_seq(a_opt, "lpfw_parent")

    # === Phase 2: Reweighted Projected Gradient ===
    print(f"\n=== Phase 2: Reweighted PG (n={len(best_seq)}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t2 = min(500, rem * 0.3)
        if t2 > 30:
            print(f"  n={len(best_seq)}, time={t2:.0f}s")
            a_opt, _ = reweighted_pg(best_seq, time_limit=t2,
                                     n_peaks=6, p=2.0, inner_iters=200, verbose=True)
            try_seq(a_opt, "reweighted_pg")

            # Second run with different params
            rem = deadline - time.monotonic()
            t2b = min(250, rem * 0.2)
            if t2b > 20:
                print(f"  n={len(best_seq)}, time={t2b:.0f}s (variant)")
                a_opt, _ = reweighted_pg(best_seq, time_limit=t2b,
                                         n_peaks=10, p=3.0, inner_iters=150, verbose=True)
                try_seq(a_opt, "reweighted_pg_v2")

    # === Phase 3: Try different initializations at same length ===
    print(f"\n=== Phase 3: Diverse initializations (n={base_n}) ===")
    inits = generate_initializations(base_n, rng)
    for i, init in enumerate(inits[:4]):
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(rem * 0.2, 150)
        if t_alloc < 20:
            break
        print(f"  Init {i}, time={t_alloc:.0f}s")
        a_opt, _ = reweighted_pg(init, time_limit=t_alloc,
                                 n_peaks=6, p=2.0, inner_iters=200)
        try_seq(a_opt, f"reweighted_pg_init{i}")

    # === Phase 4: Subgradient refinement ===
    print(f"\n=== Phase 4: Subgradient (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        t_sg = min(rem * 0.25, 250)
        if t_sg > 20:
            print(f"  n={n_best}, time={t_sg:.0f}s")
            a_opt, _ = sg_optimize(best_seq, time_limit=t_sg,
                                   lr=0.002, momentum=0.94,
                                   top_k=max(30, n_best // 8),
                                   temp=0.00004, verbose=True)
            try_seq(a_opt, "sg_main")

            rem = deadline - time.monotonic()
            t_sg2 = min(rem * 0.15, 120)
            if t_sg2 > 15:
                a_opt, _ = sg_optimize(best_seq, time_limit=t_sg2,
                                       lr=0.001, momentum=0.95,
                                       top_k=max(20, n_best // 12),
                                       temp=0.00002)
                try_seq(a_opt, "sg_fine")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.8f}) ===")
    if best_seq is not None:
        base = best_seq
        candidate_lengths = []
        for delta in [-128, -64, 64, 128, 256, 512]:
            ln = len(base) + delta
            if 256 <= ln <= 4096:
                candidate_lengths.append(ln)
        # Remove duplicates, sort by closeness to current
        candidate_lengths = sorted(set(candidate_lengths), key=lambda x: abs(x - len(base)))

        for target_n in candidate_lengths[:4]:
            rem = deadline - time.monotonic()
            if rem < 80:
                break
            t_alloc = min(rem * 0.3, 200)
            if t_alloc < 20:
                break
            f_interp = interpolate_sequence(base, target_n)
            print(f"  n={target_n}, time={t_alloc:.0f}s")
            a_opt, _ = reweighted_pg(f_interp, time_limit=t_alloc,
                                     n_peaks=6, p=2.0, inner_iters=150)
            try_seq(a_opt, f"multilength_n{target_n}")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Quick reweighted PG with tight params
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t_polish = min(rem * 0.4, 100)
            a_opt, _ = reweighted_pg(best_seq, time_limit=t_polish,
                                     n_peaks=4, p=4.0, inner_iters=400)
            try_seq(a_opt, "polish_pg")

        # Quick LP-FW
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t_lp = min(rem * 0.3, 80)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_lp)
            try_seq(a_opt, "polish_lpfw")

        # Quick SG
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            t_sg = min(rem * 0.3, 50)
            a_opt, _ = sg_optimize(best_seq, time_limit=t_sg,
                                   lr=0.0008, momentum=0.95,
                                   top_k=max(15, n_best // 10),
                                   temp=0.00002)
            try_seq(a_opt, "polish_sg")

        # Random perturbations + quick optimization
        for p_idx in range(5):
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break
            noise = rng.standard_normal(n_best) * (0.001 + 0.0005 * p_idx)
            f_pert = project_simplex(best_seq + noise)
            t_p = min(rem * 0.3, 30)
            if t_p < 5:
                break
            a_opt, _ = reweighted_pg(f_pert, time_limit=t_p,
                                     n_peaks=5, p=2.0, inner_iters=100)
            try_seq(a_opt, f"perturb_{p_idx}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Time remaining: {deadline - time.monotonic():.1f}s")
    return [float(x) for x in best_seq]
```
