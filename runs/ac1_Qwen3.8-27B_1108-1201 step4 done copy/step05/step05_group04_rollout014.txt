def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def softmax(x):
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()

def lse_obj_and_grad(x, beta):
    n = len(x)
    f = softmax(x)
    conv = fast_convolve(f)
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()
    
    # Objective
    obj = (bc_max + np.log(exp_bc.sum())) / beta
    
    # Gradient w.r.t. f
    # grad_f[j] = 2 * sum_m w[j+m] * f[m]
    # = 2 * corr(w, f)[j] where corr[k] = sum_i w[i] * f[i-k]
    size = 2 * n
    W = np.fft.rfft(w, size)
    F = np.fft.rfft(f, size)
    corr = np.fft.irfft(W * np.conj(F), size)[:n]
    grad_f = 2.0 * corr
    
    # Gradient w.r.t. x (chain rule through softmax)
    dot = np.dot(f, grad_f)
    grad_x = f * (grad_f - dot)
    
    return obj, grad_x
