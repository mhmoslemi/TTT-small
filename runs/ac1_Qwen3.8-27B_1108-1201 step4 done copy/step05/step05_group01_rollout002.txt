        a_pert = parent + rng.standard_normal(n_parent) * scale
        a_pert = project_simplex(a_pert)
        init_list.append((a_pert, f"pert_{scale}"))
    
    # Run optimization on each init (time-limited)
    for a_init, label in init_list:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        try_seq(a_init, label + "_init")
        
        # Quick gradient flatten
        t_flat = min(rem * 0.1, 15)
        if t_flat > 5:
            a_flat, _ = grad_flatten(a_init, time_limit=t_flat)
            try_seq(a_flat, label + "_flat")
        
        # LP-FW
        rem = deadline - time.monotonic()
        t_lpfw = min(rem * 0.15, 60)
        if t_lpfw > 5:
            a_opt, _ = lp_fw_optimize(a_init, time_limit=t_lpfw)
            try_seq(a_opt, label + "_lpfw")
    
    # Phase 3: Focus on best, try longer lengths
    print(f"\n=== Phase 3: Longer lengths (best={best_score:.6f}) ===")
    if best_seq is not None:
        for target_n in [int(len(best_seq) * 1.5), int(len(best_seq) * 2), int(len(best_seq) * 3)]:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            try_seq(f_interp, f"interp_best_n{target_n}")
            
            t_lpfw = min(rem * 0.2, 80)
            if t_lpfw > 10:
                a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_lpfw, verbose=True)
                try_seq(a_opt, f"lpfw_interp_n{target_n}")
    
    # Phase 4: Refine best with multiple methods
    print(f"\n=== Phase 4: Refine best (best={best_score:.6f}) ===")
    if best_seq is not None:
        # Gradient flatten
        rem = deadline - time.monotonic() - 60
        if rem > 10:
            t_gf = min(rem * 0.3, 100)
            a_opt, _ = grad_flatten(best_seq, time_limit=t_gf, lr=0.005, mom=0.92)
            try_seq(a_opt, "grad_flatten_refine")
        
        # LP-FW (longer)
        rem = deadline - time.monotonic() - 40
        if rem > 10:
            t_lpfw = min(rem * 0.5, 200)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_lpfw, verbose=True)
            try_seq(a_opt, "lpfw_refine")
        
        # Coordinate descent
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_cd = min(rem * 0.5, 60)
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd_refine")
