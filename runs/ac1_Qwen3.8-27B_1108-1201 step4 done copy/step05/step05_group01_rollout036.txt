    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_gradient(f, w):
    """
    Compute gradient of sum_k w[k]*conv(f,f)[k] with respect to f.
    grad[m] = 2 * sum_j f[j] * w[m+j]
    """
    n = len(f)
    # Use FFT-based cross-correlation
    # grad[m] = 2 * sum_j f[j] * w[m+j] for m in [0, n-1]
    # This equals 2 * conv(f, w[::-1])[2n-2-m]
    rev_w = w[::-1]
    fft_len = 1 << (3 * n - 2 - 1).bit_length()
    F = np.fft.rfft(f, fft_len)
    W = np.fft.rfft(rev_w, fft_len)
    conv_fw = np.fft.irfft(F * W, fft_len)
    # grad[m] = 2 * conv_fw[2n-2-m], m in [0, n-1]
    # indices: [2n-2, 2n-3, ..., n-1]
    grad = 2.0 * conv_fw[n-1:2*n-1][::-1]
    return grad


def smooth_max_descent(f0, time_limit, temp0=0.005, lr0=0.02, mom0=0.9, 
                       temp_growth=1.002, lr_decay=0.9995, mom_growth=1.00005):
    """Gradient descent with smooth max approximation and momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    temp = temp0
    lr = lr0
    mom = mom0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Softmax weights
        conv_shifted = conv / temp
        conv_shifted -= conv_shifted.max()
        w = np.exp(conv_shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum
        
        # Gradient
        grad = compute_smooth_gradient(f, w)
        
        # Momentum update
        v = mom * v - lr * grad
        f_new = project_simplex(f + v)
        
        # Evaluate
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = f_new
        conv = conv_new
        
        # Adapt hyperparameters
        temp = min(temp * temp_growth, 0.5)
        lr = max(lr * lr_decay, 0.0001)
        mom = min(mom * mom_growth, 0.99)
        
        if no_improve > 300:
            # Perturbation restart
            rng = np.random.default_rng((it * 7919 + n) % 2**31)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            v = np.zeros(n)
            conv = fast_convolve(f)
            no_improve = 0
            temp = max(0.001, temp * 0.3)
            lr = min(0.05, lr * 3)
            it = 0
        
        if it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD iter {it}: obj={best_obj:.8f}, t={elapsed:.1f}s, lr={lr:.6f}, temp={temp:.6f}")
    
    return best_f, best_obj


def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. f*g[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    # Build constraint matrix A_ub where row k corresponds to (f*g)[k] <= M
    # (f*g)[k] = sum_j f[max(0,k-n+1):min(n,k+1)][...] * g[...]
    # Actually: (f*g)[k] = sum_j f[j] * g[k-j] for valid j
    # For fixed f, this is linear in g: sum_j g[j] * f[k-j] (where k-j is valid)
    # So A_ub[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # But building this matrix explicitly for n=768 gives a 1535x768 matrix - that's fine
    
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = f[idx]
    
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive M and line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
            
        g = lp_direction(f, M * m_relax, n)
        if g is None:
            m_relax = min(m_relax * 1.2, 5.0)
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            m_relax = min(m_relax * 1.05, 3.0)
            continue
        
        # Simple line search: try a few alphas
        best_alpha = 0.0
        best_val = M
        alphas = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 2.0/(it+2)]
        
        for alpha in alphas:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_relax = 1.0
            f = f_new
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 10 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.05, 4.0)
    
    return best_f, best_obj


def coordinate_descent(f0, time_limit, max_passes=10):
    """Cyclic coordinate descent with golden section line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    GR = (1.0 + np.sqrt(5.0)) / 2.0
    inv_gr = 1.0 / GR
    
    for _ in range(max_passes):
        if time.monotonic() - t0 > time_limit:
            break
        pass_improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-14:
                continue
            
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue
            
            # Golden section search
            lo = 0.0
            hi = min(0.9999, 3.0 * old_val)
            
            c1 = lo + inv_gr * (hi - lo)
            c2 = lo + (1.0 - inv_gr) * (hi - lo)
            
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            others = f[mask].copy()
            
            def eval_at(x):
                ft = np.empty(n)
                ft[j] = x
                scale = (1.0 - x) / denom
                ft[mask] = others * scale
                return fast_convolve(ft).max()
            
            f1 = eval_at(c1)
            f2 = eval_at(c2)
            
            for _ in range(10):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + inv_gr * (hi - lo)
                    f1 = eval_at(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + (1.0 - inv_gr) * (hi - lo)
                    f2 = eval_at(c2)
            
            opt_x = (lo + hi) / 2.0
            opt_val = min(f1, f2)
            cur_val = fast_convolve(f).max()
            
            if opt_val < cur_val - 1e-14:
                ft = np.empty(n)
                ft[j] = opt_x
                scale = (1.0 - opt_x) / denom
                ft[mask] = others * scale
                f = ft
                pass_improved = True
                new_obj = fast_convolve(f).max()
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_f = f.copy()
        
        if not pass_improved:
            break
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def cosine_modulated_init(n, freq, amp, seed=0):
    """Generate a cosine-modulated uniform initialization."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:
1. Fast smooth-max gradient descent (softmax approximation) for rapid exploration
2. LP-based Frank-Wolfe for precise direction finding
3. Coordinate descent with golden-section search for local polish
4. Multi-length exploration and multi-start diversification
Key improvements: vectorized constraint matrix building, adaptive temperature
schedule, and aggressive exploration of sequence lengths 512-2048.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_gradient(f, w):
    """
    Compute gradient of sum_k w[k]*conv(f,f)[k] w.r.t. f.
    grad[m] = 2 * sum_j f[j] * w[m+j] for valid indices.
    Uses FFT-based cross-correlation.
    """
    n = len(f)
    rev_w = w[::-1].copy()
    fft_len = 1 << (3 * n - 2 - 1).bit_length()
    F = np.fft.rfft(f, fft_len)
    W = np.fft.rfft(rev_w, fft_len)
    conv_fw = np.fft.irfft(F * W, fft_len)
    grad = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]
    return grad


def smooth_max_descent(f0, time_limit, temp0=0.005, lr0=0.02, mom0=0.92,
                       temp_growth=1.003, lr_decay=0.9997, mom_growth=1.00003):
    """Gradient descent with smooth max (softmax) approximation and momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    temp = temp0
    lr = lr0
    mom = mom0

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv_shifted = conv / temp
        conv_shifted -= conv_shifted.max()
        w = np.exp(conv_shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum

        grad = compute_smooth_gradient(f, w)
        v = mom * v - lr * grad
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        conv = conv_new

        temp = min(temp * temp_growth, 0.3)
        lr = max(lr * lr_decay, 0.00005)
        mom = min(mom * mom_growth, 0.995)

        if no_improve > 400:
            rng = np.random.default_rng((it * 7919 + n * 31) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            v = np.zeros(n)
            conv = fast_convolve(f)
            no_improve = 0
            temp = max(0.001, temp * 0.4)
            lr = min(0.04, lr * 3.0)
            it = 0

        if it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SGD iter {it}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_f, best_obj


def build_A_ub(f):
    """Build the LP constraint matrix A_ub where A_ub[k,j] = f[k-j] if valid."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = build_A_ub(f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive relaxation and efficient line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        g = lp_direction(f, M * m_relax, n)
        if g is None:
            m_relax = min(m_relax * 1.15, 5.0)
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 80:
                break
            m_relax = min(m_relax * 1.04, 3.0)
            continue

        # Efficient line search: FW step + a few alternatives
        best_alpha = 0.0
        best_val = M
        fw_alpha = 2.0 / (it + 2.0)
        candidates = sorted(set([0.1, 0.3, 0.5, 0.7, 1.0, fw_alpha]))

        for alpha in candidates:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Binary refinement (4 steps)
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.2)
            hi = min(1.0, best_alpha + 0.2)
            for _ in range(5):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_relax = 1.0
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    print(f"      LP-FW iter {it}: obj={best_obj:.10f}, t={time.monotonic()-t0:.0f}s")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.05, 4.0)

    return best_f, best_obj


def golden_cd_step(f, j, n):
    """Golden section search for coordinate j, returning optimal value and new f."""
    old_val = f[j]
    if old_val < 1e-14:
        return f, fast_convolve(f).max()

    denom = 1.0 - old_val
    if denom < 1e-12:
        return f, fast_convolve(f).max()

    GR = (1.0 + np.sqrt(5.0)) / 2.0
    inv_gr = 1.0 / GR
    lo = 0.0
    hi = min(0.9999, 3.0 * old_val)

    c1 = lo + inv_gr * (hi - lo)
    c2 = lo + (1.0 - inv_gr) * (hi - lo)

    mask = np.ones(n, dtype=bool)
    mask[j] = False
    others = f[mask].copy()

    def eval_x(x):
        ft = np.empty(n)
        ft[j] = x
        scale = (1.0 - x) / denom
        ft[mask] = others * scale
        return fast_convolve(ft).max()

    f1 = eval_x(c1)
    f2 = eval_x(c2)

    for _ in range(10):
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = lo + inv_gr * (hi - lo)
            f1 = eval_x(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + (1.0 - inv_gr) * (hi - lo)
            f2 = eval_x(c2)

    opt_x = (lo + hi) / 2.0
    opt_val = min(f1, f2)
    cur_val = fast_convolve(f).max()

    if opt_val < cur_val - 1e-14:
        ft = np.empty(n)
        ft[j] = opt_x
        scale = (1.0 - opt_x) / denom
        ft[mask] = others * scale
        return ft, opt_val
    return f, cur_val


def coordinate_descent(f0, time_limit, max_passes=5):
    """Cyclic coordinate descent with golden section search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    for _ in range(max_passes):
        if time.monotonic() - t0 > time_limit:
            break
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            f, val = golden_cd_step(f, j, n)
            if val < best_obj - 1e-14:
                best_obj = val
                best_f = f.copy()
                pass_improved = True

        if not pass_improved:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def cosine_modulated_init(n, freq, amp):
    """Generate a cosine-modulated uniform initialization on the simplex."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def two_freq_init(n, f1, a1, f2, a2):
    """Two-frequency modulation for richer initialization."""
    j = np.arange(n)
    a = 1.0 + a1 * np.cos(2.0 * np.pi * f1 * j / n) + a2 * np.cos(2.0 * np.pi * f2 * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def simulated_annealing(f0, time_limit, alpha0=0.01, cooling=0.999):
    """Simulated annealing with mass-transfer moves."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    cur_obj = best_obj

    t0 = time.monotonic()
    alpha = alpha0
    rng = np.random.default_rng(12345)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        delta = rng.uniform(-0.005, 0.005)
        if f[i] + delta < 0 or f[j] - delta < 0:
            continue

        f[i] += delta
        f[j] -= delta
        conv_new = fast_convolve(f)
        new_obj = conv_new.max()
        delta_obj = new_obj - cur_obj

        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / max(alpha, 1e-10)):
            cur_obj = new_obj
            conv = conv_new
            if new_obj < best_obj - 1e-14:
                best_obj = new_obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            f[i] -= delta
            f[j] += delta

        alpha *= cooling
        if no_improve > 500:
            f = best_f.copy()
            conv = fast_convolve(f)
            cur_obj = conv.max()
            alpha = max(alpha * 2, 0.001)
            no_improve = 0

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining smooth-max gradient descent,
    LP-Frank-Wolfe, coordinate descent, and simulated annealing.
    Explores multiple sequence lengths with diverse initializations.
    """
    global _best_seq, _best_score
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    _best_seq = None
    _best_score = float('inf')

    def update_best(a, label):
        global _best_seq, _best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < _best_score:
            _best_score = score
            _best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768

    # === Phase 1: Smooth-max gradient descent (fast exploration) ===
    print("=== Phase 1: Smooth-max descent ===")
    t1 = min(150, deadline - time.monotonic() - 1200)
    if t1 > 40:
        a_opt, obj = smooth_max_descent(parent, time_limit=t1, temp0=0.004, lr0=0.025, mom0=0.93)
        update_best(a_opt, "smooth_parent")

    # Try different lengths with smooth-max
    base = _best_seq if _best_seq is not None else parent
    for n_try in [512, 1024, 1536]:
        rem = deadline - time.monotonic() - 800
        if rem < 30:
            break
        f_init = interpolate_sequence(base, n_try)
        t_alloc = min(rem * 0.2, 60)
        if t_alloc > 15:
            a_opt, obj = smooth_max_descent(f_init, time_limit=t_alloc, temp0=0.003, lr0=0.02, mom0=0.92)
            update_best(a_opt, f"smooth_n{n_try}")

    # Two-frequency initializations
    for n_try in [768, 1024]:
        rem = deadline - time.monotonic() - 600
        if rem < 20:
            break
        for f1, a1, f2, a2 in [(1, 0.3, 2, 0.15), (2, 0.2, 5, 0.1), (1, 0.2, 3, 0.2)]:
            f_init = two_freq_init(n_try, f1, a1, f2, a2)
            s = update_best(f_init, f"2freq_n{n_try}")
            rem = deadline - time.monotonic() - 400
            if rem > 20:
                t_alloc = min(rem * 0.15, 40)
                a_opt, _ = smooth_max_descent(f_init, time_limit=t_alloc, temp0=0.003, lr0=0.015, mom0=0.91)
                update_best(a_opt, f"2freq_opt_n{n_try}")

    # === Phase 2: LP-FW on best (precise optimization) ===
    print(f"\n=== Phase 2: LP-FW (best={_best_score:.6f}) ===")
    if _best_seq is not None:
        n_best = len(_best_seq)
        rem = deadline - time.monotonic() - 500
        if rem > 40:
            t_fw = min(rem * 0.5, 200)
            print(f"  LP-FW n={n_best}, t={t_fw:.0f}s")
            a_opt, _ = lp_fw_optimize(_best_seq, time_limit=t_fw, verbose=True)
            update_best(a_opt, "lpfw_main")

    # === Phase 3: Simulated annealing for escape ===
    print(f"\n=== Phase 3: Simulated annealing (best={_best_score:.6f}) ===")
    if _best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_sa = min(rem * 0.4, 100)
            a_opt, _ = simulated_annealing(_best_seq, time_limit=t_sa, alpha0=0.005, cooling=0.9995)
            update_best(a_opt, "sim_anneal")
            # Quick gradient polish on SA result
            rem = deadline - time.monotonic() - 200
            if rem > 20:
                a_opt2, _ = smooth_max_descent(a_opt, time_limit=min(rem, 40), temp0=0.002, lr0=0.01, mom0=0.94)
                update_best(a_opt2, "sa_polish")

    # === Phase 4: Second LP-FW round ===
    print(f"\n=== Phase 4: LP-FW round 2 (best={_best_score:.6f}) ===")
    if _best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 30:
            t_fw = min(rem * 0.5, 150)
            a_opt, _ = lp_fw_optimize(_best_seq, time_limit=t_fw, verbose=False)
            update_best(a_opt, "lpfw_r2")

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coordinate descent (best={_best_score:.6f}) ===")
    if _best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_cd = min(rem * 0.5, 100)
            a_opt, _ = coordinate_descent(_best_seq, time_limit=t_cd, max_passes=3)
            update_best(a_opt, "coord_desc")

    # === Phase 6: Multi-start diversification ===
    print(f"\n=== Phase 6: Multi-start (best={_best_score:.6f}) ===")
    if _best_seq is not None:
        n_best = len(_best_seq)
        for p in range(4):
            rem = deadline - time.monotonic() - 80
            if rem < 15:
                break
            t_alloc = min(rem * 0.25, 60)

            if p == 0:
                noise = rng.standard_normal(n_best) * 0.004
                f_init = project_simplex(_best_seq + noise)
            elif p == 1:
                f_init = 0.85 * _best_seq + 0.15 * cosine_modulated_init(n_best, 2, 0.25)
                f_init = project_simplex(f_init)
            elif p == 2:
                f_init = 0.9 * _best_seq + 0.1 * (np.ones(n_best) / n_best)
                f_init = project_simplex(f_init)
            else:
                f_init = 0.75 * _best_seq + 0.25 * _best_seq[::-1]
                f_init = project_simplex(f_init)

            a_opt, _ = lp_fw_optimize(f_init, time_limit=t_alloc)
            update_best(a_opt, f"multi_{p}")

    # === Phase 7: Long length exploration ===
    print(f"\n=== Phase 7: Long lengths (best={_best_score:.6f}) ===")
    if _best_seq is not None:
        for target_n in [2048, 2560]:
            rem = deadline - time.monotonic() - 60
            if rem < 30:
                break
            f_interp = interpolate_sequence(_best_seq, target_n)
            # Quick smooth-max
            t_alloc = min(rem * 0.3, 50)
            a_opt, _ = smooth_max_descent(f_interp, time_limit=t_alloc, temp0=0.003, lr0=0.015, mom0=0.91)
            update_best(a_opt, f"long_n{target_n}")
            # LP-FW if time
            rem = deadline - time.monotonic() - 40
            if rem > 20:
                a_opt2, _ = lp_fw_optimize(a_opt, time_limit=min(rem, 40))
                update_best(a_opt2, f"long_lpfw_n{target_n}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final polish (best={_best_score:.6f}) ===")
    if _best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            a_opt, _ = lp_fw_optimize(_best_seq, time_limit=min(rem, 40))
            update_best(a_opt, "final_lpfw")

        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_opt, _ = coordinate_descent(_best_seq, time_limit=min(rem, 30), max_passes=2)
            update_best(a_opt, "final_cd")

    # Fallback
    if _best_seq is None:
        _best_seq = parent.copy()
        _best_score = float(evaluate_sequence(_best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(_best_seq.tolist()))
    if np.isfinite(fs):
        _best_score = fs
    else:
        _best_seq = parent.copy()
        _best_score = float(evaluate_sequence(_best_seq.tolist()))

    print(f"\nFINAL SCORE: {_best_score:.6f} (n={len(_best_seq)})")
    return [float(x) for x in _best_seq]
