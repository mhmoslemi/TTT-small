fast coordinate descent with analytic subproblem solving, and multi-length

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_obj(f):
    """Compute max of self-convolution for a normalized sequence (sum=1)."""
    return float(fast_convolve(f).max())


def lpfw_lp_solve(f, M, active_lags):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in active_lags, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    n_act = len(active_lags)
    A_ub = np.zeros((n_act, n))
    for idx, k in enumerate(active_lags):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[idx, j] = f[k - j]
    b_ub = np.full(n_act, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    return None

        rem = float(deadline - time.monotonic() - 700)
        t2 = max(20, min(300, rem))
        a_opt, _ = fast_cd(best_seq, time_limit=t2, n_evals=6, verbose=True)
        update_best(a_opt, "cd_1")

    # === Phase 3: LP-FW round 2 ===
    print(f"\n=== Phase 3: LP-FW round 2 (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = float(deadline - time.monotonic() - 500)
        t3 = max(20, min(300, rem))
        a_opt, _ = active_set_lpfw(best_seq, time_limit=t3)
        update_best(a_opt, "lpfw_2")

        rem = float(deadline - time.monotonic() - 300)
        if rem > 20:
            a_opt, _ = fast_cd(best_seq, time_limit=min(rem, 200), n_evals=6)
            update_best(a_opt, "cd_2")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.8f}) ===")
    if best_seq is not None:
        target_lengths = [600, 650, 700, 800, 850, 960, 1024, 1152, 1280]
        for tl in target_lengths:
            rem = float(deadline - time.monotonic() - 200)
            if rem < 30:
                break
            if abs(tl - len(best_seq)) < 40:
                continue
            fi = interpolate_seq(best_seq, tl)
            t_alloc = min(rem * 0.35, 120)
            if t_alloc < 15:
                continue
            print(f"  n={tl}, t={t_alloc:.0f}s")
            a_opt, _ = active_set_lpfw(fi, time_limit=min(t_alloc, 90))
            s = update_best(a_opt, f"len_{tl}")
            if np.isfinite(s) and s < best_score + 0.01:
                rem = float(deadline - time.monotonic() - 150)
                if rem > 15:
                    a_opt2, _ = fast_cd(a_opt, time_limit=min(rem, 60))
                    update_best(a_opt2, f"cd_len_{tl}")

    # === Phase 5: Diverse restarts ===
    print(f"\n=== Phase 5: Diverse restarts (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_b = len(best_seq)
        restarts = [
            ("noise", project_simplex(best_seq + rng.standard_normal(n_b) * 0.005)),
            ("flip", project_simplex(0.5 * best_seq + 0.5 * best_seq[::-1])),
            ("blend", project_simplex(0.8 * best_seq + 0.2 * np.ones(n_b) / n_b)),
            ("cos_blend", project_simplex(0.7 * best_seq + 0.3 * cosine_init(n_b, 3, 0.3))),
        ]
        for label, f_init in restarts:
            rem = float(deadline - time.monotonic() - 100)
            if rem < 15:
                break
            t_r = min(rem * 0.4, 80)
            if t_r < 10:
                continue
            a_opt, _ = active_set_lpfw(f_init, time_limit=t_r)
            update_best(a_opt, f"restart_{label}")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = float(deadline - time.monotonic() - 20)
        if rem > 15:
            a_opt, _ = fast_cd(best_seq, time_limit=min(rem, 120), n_evals=8)
            update_best(a_opt, "final_cd")

        rem = float(deadline - time.monotonic() - 10)
        if rem > 10:
            a_opt, _ = active_set_lpfw(best_seq, time_limit=min(rem, 50))
            update_best(a_opt, "final_lpfw")

        # Tiny perturbations
        for pi in range(3):
            rem = float(deadline - time.monotonic() - 8)
            if rem < 8:
                break
            noise = rng.standard_normal(len(best_seq)) * 0.001
            fp = project_simplex(best_seq + noise)
            a_opt, _ = fast_cd(fp, time_limit=min(rem, 15), n_evals=7)
            update_best(a_opt, f"pert_{pi}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768, dtype=np.float64) / 768

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    print(f"\nFINAL SCORE: {fs:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
