"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA (Nesterov-accelerated projected subgradient) with exact convex subgradients
2. LP-Frank-Wolfe for global descent directions  
3. Multi-length exploration with interpolation
4. Adaptive phase switching based on convergence
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_max_conv(a, conv, top_k=1, temperature=0.0):
    """Compute subgradient of max(conv(a,a)) at point a.
    Returns gradient vector of length n."""
    n = len(a)
    if temperature > 0:
        # Smoothed: use softmax weighting over peaks
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        w = w / ws
    else:
        # Hard max: uniform weight over all argmax indices
        max_val = conv.max()
        peaks = np.where(conv >= max_val - 1e-12)[0]
        w = np.ones(len(peaks)) / len(peaks)
        k_actual = len(peaks)
        w = w[:top_k]  # Actually for hard max, all argmax get equal weight
    
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if i >= len(w):
            break
        wi = w[i]
        if wi < 1e-14:
            continue
        k = int(peaks[i])
        # d conv[k] / d a[m] = 2*a[k-m] if 0 <= k-m < n
        m_lo = max(0, k - (n - 1))
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            indices = np.arange(m_lo, m_hi + 1)
            grad[indices] += 2.0 * wi * a[k - indices]
    
    return grad

def fista_optimize(a0, time_limit, lr0=0.01, top_k=1, temperature=0.0, verbose=False):
    """FISTA (Nesterov-accelerated projected subgradient) for min max(conv(a,a)) s.t. a>=0, sum(a)=1."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    y = a.copy()
    a_prev = a.copy()
    t = 1.0
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    lr = lr0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute subgradient at y
        conv_y = fast_convolve(y)
        g = subgradient_max_conv(y, conv_y, top_k=top_k, temperature=temperature)
        
        # Gradient step
        z = y - lr / t * g
        
        # Project onto simplex
        a_new = project_simplex(z)
        
        # Update Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_new + ((t - 1.0) / t_new) * (a_new - a_prev)
        y = project_simplex(y)  # Keep y in simplex
        
        a_prev = a_new
        a = a_new
        t = t_new
        
        # Evaluate
        conv = fast_convolve(a)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    FISTA iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
        
        # Adaptive learning rate
        if no_improve > 200:
            lr *= 0.7
            no_improve = 0
            if lr < 1e-6:
                break
        
        if no_improve > 2000:
            break
    
    return best_a, best_obj

def lp_fw_optimize(a0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with proper step sizes and m_relaxation."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = M
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Solve LP: max sum(g) s.t. (a*g)[k] <= M*m_relax for all k, g >= 0
        c = -np.ones(n)
        # Build constraint matrix (sparse would be better but let's use dense for simplicity)
        # A[k, j] = a[j] if 0 <= k-j < n
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j+n, j] = a
        b_ub = np.full(2 * n - 1, M * m_relax)
        bounds = [(0, None)] * n
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        
        if not res.success:
            m_relax *= 1.1
            if m_relax > 5.0:
                m_relax = 1.0
            continue
        
        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - a)) < 1e-11:
            no_improve += 1
            if no_improve > 50:
                break
            m_relax = min(m_relax * 1.03, 3.0)
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        fw_alpha = 2.0 / (it + 2.0)
        alphas = sorted(set([0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, fw_alpha, 0.6, 0.7, 0.8, 0.9, 1.0]))
        
        for alpha in alphas:
            at = a + alpha * (g - a)
            at = np.maximum(at, 0.0)
            st = at.sum()
            if st < 1e-15:
                continue
            at = at / st
            val = fast_convolve(at).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            a_new = a + best_alpha * (g - a)
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            conv = fast_convolve(a_new)
            M = conv.max()
            m_relax = 1.0
            a = a_new
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_a = a.copy()
                no_improve = 0
                if verbose and it % 10 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.05, 4.0)
        
        if no_improve > 80:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.002 + 0.002 * min(no_improve // 80, 3)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            conv = fast_convolve(a)
            M = conv.max()
            m_relax = 1.0
            no_improve = 0
    
    return best_a, best_obj

def cosine_init(n, freq, amp, seed=0):
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. FISTA (Nesterov-accelerated projected subgradient) with exact convex subgradients
2. LP-Frank-Wolfe for globally valid descent directions with sparse constraint matrices
3. Multi-length exploration with interpolation
4. Adaptive phase switching and dual-guided subgradients
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgrad(a, conv, num_peaks=1, temperature=0.0):
    """
    Compute (sub)gradient of max_k conv_k(a) at point a.
    Uses top-k peaks with optional softmax weighting for smoothing.
    d conv[k]/d a[m] = 2*a[k-m] if 0 <= k-m < n, else 0.
    """
    n = len(a)
    grad = np.zeros(n)

    if temperature > 0:
        k_actual = min(num_peaks, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
    else:
        max_val = conv.max()
        peaks = np.where(conv >= max_val - 1e-12)[0]
        if len(peaks) == 0:
            peaks = np.array([np.argmax(conv)])
        w = np.ones(len(peaks)) / len(peaks)

    for i in range(len(peaks)):
        wi = w[i]
        if wi < 1e-14:
            continue
        k = int(peaks[i])
        m_lo = max(0, k - (n - 1))
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            indices = np.arange(m_lo, m_hi + 1)
            grad[indices] += 2.0 * wi * a[k - indices]

    return grad


def fista_optimize(a0, time_limit, lr0=0.01, num_peaks=5, temperature=0.001,
                   decay=0.9999, verbose=False):
    """
    FISTA (Nesterov-accelerated projected subgradient) for minimizing
    max(conv(a,a)) subject to a >= 0, sum(a) = 1.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    y = a.copy()
    a_prev = a.copy()
    t = 1.0

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    lr = lr0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute subgradient at y (lookahead point)
        conv_y = fast_convolve(y)
        g = compute_subgrad(y, conv_y, num_peaks=num_peaks, temperature=temperature)

        # Nesterov gradient step with diminishing step size
        step = lr / (1.0 + 0.1 * lr * it)
        z = y - step * g

        # Project onto simplex
        a_new = project_simplex(z)

        # Nesterov momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_new + ((t - 1.0) / t_new) * (a_new - a_prev)
        y = project_simplex(y)

        a_prev = a_new
        a = a_new
        t = t_new

        # Evaluate at actual point
        conv = fast_convolve(a)
        obj = conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    FISTA iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        # Adaptive learning rate decay
        lr *= decay

        if no_improve > 500:
            lr *= 0.5
            no_improve = 0
            if lr < 1e-8:
                break

        if no_improve > 5000:
            break

    return best_a, best_obj


def build_sparse_conv_matrix(a):
    """
    Build sparse matrix A where A[k,j] = a[j] if 0 <= k-j < n (else 0).
    This represents the constraint (a*g)[k] = sum_j A[k,j]*g[j].
    """
    n = len(a)
    rows = []
    cols = []
    data = []
    for j in range(n):
        # A[k, j] = a[k-j] for k in [j, j+n-1]
        ks = np.arange(j, j + n)
        vals = a
        rows.extend(ks)
        cols.extend([j] * n)
        data.extend(vals)
    row_arr = np.array(rows)
    col_arr = np.array(cols)
    data_arr = np.array(data)
    return csr_matrix((data_arr, (row_arr, col_arr)), shape=(2 * n - 1, n))


def solve_lp_direction(a, M, n, sparse_A=None):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    if sparse_A is None:
        sparse_A = build_sparse_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=sparse_A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, res
    return None, res


def lp_fw_optimize(a0, time_limit, verbose=False):
    """
    LP-based Frank-Wolfe with adaptive M relaxation, sparse constraint matrix,
    and golden-section line search.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = M
    best_a = a.copy()

    # Precompute sparse matrix (it changes when a changes, but structure is the same)
    sparse_A = build_sparse_conv_matrix(a)

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Solve LP with relaxed bound
        g, res = solve_lp_direction(a, M * m_relax, n, sparse_A)

        if g is None:
            m_relax *= 1.15
            if m_relax > 5.0:
                m_relax = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - a)) < 1e-12:
            no_improve += 1
            if no_improve > 40:
                break
            m_relax = min(m_relax * 1.04, 3.0)
            continue

        # Line search: FW step size + golden section refinement
        best_alpha = 0.0
        best_val = M
        fw_alpha = 2.0 / (it + 2.0)

        # Grid search first
        alphas = sorted(set([x * 0.1 for x in range(1, 11)] + [fw_alpha, 0.95]))
        for alpha in alphas:
            at = a + alpha * (g - a)
            at = np.maximum(at, 0.0)
            st = at.sum()
            if st < 1e-15:
                continue
            at = at / st
            val = fast_convolve(at).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Golden section refinement
        if 0 < best_alpha < 1:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            GR = (1.0 + np.sqrt(5.0)) / 2.0
            c1 = lo + (1.0 / GR) * (hi - lo)
            c2 = lo + (1.0 - 1.0 / GR) * (hi - lo)

            def eval_alpha(x):
                at = a + x * (g - a)
                at = np.maximum(at, 0.0)
                st = at.sum()
                if st < 1e-15:
                    return float('inf')
                at = at / st
                return fast_convolve(at).max()

            f1 = eval_alpha(c1)
            f2 = eval_alpha(c2)
            for _ in range(10):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + (1.0 / GR) * (hi - lo)
                    f1 = eval_alpha(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + (1.0 - 1.0 / GR) * (hi - lo)
                    f2 = eval_alpha(c2)
            opt_x = (lo + hi) / 2.0
            opt_val = min(f1, f2)
            if opt_val < best_val:
                best_val = opt_val
                best_alpha = opt_x

        if best_alpha > 1e-10:
            a_new = a + best_alpha * (g - a)
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            conv = fast_convolve(a_new)
            M = conv.max()
            m_relax = 1.0
            a = a_new
            # Update sparse matrix
            sparse_A = build_sparse_conv_matrix(a)

            if M < best_obj - 1e-14:
                best_obj = M
                best_a = a.copy()
                no_improve = 0
                if verbose and it % 5 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.06, 4.0)

        # Perturbation for escape from local minima
        if no_improve > 60:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.002 + 0.002 * min(no_improve // 60, 4)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            conv = fast_convolve(a)
            M = conv.max()
            m_relax = 1.0
            sparse_A = build_sparse_conv_matrix(a)
            no_improve = 0

    return best_a, best_obj


def cosine_init(n, freq, amp):
    """Cosine-modulated initialization."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def multi_cosine_init(n, freqs, amps):
    """Multi-frequency cosine initialization."""
    j = np.arange(n)
    a = np.ones(n)
    for f, amp in zip(freqs, amps):
        a += amp * np.cos(2.0 * np.pi * f * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def interpolate_seq(a, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_new = np.interp(new_x, old_x, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    return a_new


def smooth_max_subgrad_fista(a0, time_limit, lr0=0.005, n_peaks=10, temp=0.0005):
    """
    FISTA with soft-max smoothed objective for better convergence.
    Uses log-sum-exp as a smooth surrogate for the max.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    y = a.copy()
    a_prev = a.copy()
    t = 1.0
    lr = lr0

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv_y = fast_convolve(y)
        # Top-k peaks with softmax
        k_act = min(n_peaks, len(conv_y))
        peaks = np.argpartition(conv_y, -k_act)[-k_act:]
        pv = conv_y[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        w = w / ws

        # Compute gradient
        g = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            m_lo = max(0, k - (n - 1))
            m_hi = min(n - 1, k)
            if m_lo <= m_hi:
                indices = np.arange(m_lo, m_hi + 1)
                g[indices] += 2.0 * w[i] * y[k - indices]

        # Gradient step
        step = lr / (1.0 + 0.05 * lr * it)
        z = y - step * g
        a_new = project_simplex(z)

        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_new + ((t - 1.0) / t_new) * (a_new - a_prev)
        y = project_simplex(y)

        a_prev = a_new
        a = a_new
        t = t_new

        # Evaluate true objective
        conv = fast_convolve(a)
        obj = conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            lr *= 0.6
            no_improve = 0
            if lr < 1e-8:
                break

        if no_improve > 3000:
            break

    return best_a, best_obj


def propose_candidate(budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining LP-Frank-Wolfe, FISTA with exact
    subgradients, multi-length exploration, and adaptive switching.
    Target: minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(2024)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_a = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_a, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # --- Get parent sequence ---
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_a = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = cosine_init(768, 1, 0.3)
        best_a = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default init: n={len(parent)}, score={best_score:.6f}")

    n_parent = len(parent)

    # === Phase 1: LP-FW on parent (main workhorse) ===
    print(f"\n=== Phase 1: LP-FW on parent (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    t1 = min(250, rem * 0.18)
    if t1 > 30:
        print(f"  Time budget: {t1:.0f}s")
        a_opt, _ = lp_fw_optimize(parent, time_limit=t1, verbose=True)
        try_seq(a_opt, "lpfw_parent")

    # === Phase 2: FISTA refinement ===
    print(f"\n=== Phase 2: FISTA (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic()
        if rem > 80:
            t_fista = min(150, rem * 0.15)
            print(f"  FISTA t={t_fista:.0f}s, n={len(best_a)}")
            a_opt, _ = fista_optimize(best_a, time_limit=t_fista, lr0=0.008,
                                       num_peaks=5, temperature=0.002, verbose=True)
            try_seq(a_opt, "fista_1")

            rem = deadline - time.monotonic()
            if rem > 50:
                a_opt2, _ = smooth_max_subgrad_fista(a_opt, time_limit=min(rem * 0.3, 80),
                                                      lr0=0.004, n_peaks=15, temp=0.001)
                try_seq(a_opt2, "fista_smooth")

    # === Phase 3: Alternative initializations ===
    print(f"\n=== Phase 3: Alternative inits (best={best_score:.6f}) ===")
    init_configs = [
        (512, [1, 3], [0.25, 0.12]),
        (512, [2, 5], [0.2, 0.08]),
        (768, [1, 2, 7], [0.2, 0.1, 0.04]),
        (768, [3, 8], [0.15, 0.08]),
        (1024, [1, 3], [0.2, 0.1]),
        (1024, [2, 5], [0.15, 0.08]),
    ]

    for n_init, freqs, amps in init_configs:
        rem = deadline - time.monotonic()
        if rem < 120:
            break
        a_init = multi_cosine_init(n_init, freqs, amps)
        s = try_seq(a_init, f"init_n{n_init}_f{freqs}")
        if np.isfinite(s) and s < best_score + 0.08:
            t_alloc = min(rem * 0.12, 80)
            if t_alloc > 15:
                a_opt, _ = lp_fw_optimize(a_init, time_limit=t_alloc)
                try_seq(a_opt, f"lpfw_n{n_init}")

    # === Phase 4: Extended LP-FW on best ===
    print(f"\n=== Phase 4: Extended LP-FW (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic()
        if rem > 100:
            t_lp = min(350, rem * 0.3)
            print(f"  LP-FW t={t_lp:.0f}s, n={len(best_a)}")
            a_opt, _ = lp_fw_optimize(best_a, time_limit=t_lp, verbose=True)
            try_seq(a_opt, "lpfw_extended")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.6f}) ===")
    if best_a is not None:
        n_best = len(best_a)
        target_lengths = sorted(set([384, 512, 768, 1024, 1280, 1536]))
        for target_n in target_lengths:
            if target_n == n_best:
                continue
            rem = deadline - time.monotonic()
            if rem < 80:
                break
            a_interp = interpolate_seq(best_a, target_n)
            s_init = evaluate_sequence(a_interp.tolist())
            s = try_seq(a_interp, f"interp_n{target_n}")
            # Also try a fresh cosine init at this length
            a_cos = cosine_init(target_n, 1, 0.25)
            try_seq(a_cos, f"cos_n{target_n}")
            # Optimize whichever is better
            a_work = a_interp if s_init < evaluate_sequence(a_cos.tolist()) else a_cos
            t_alloc = min(rem * 0.15, 100)
            if t_alloc > 20:
                a_opt, _ = lp_fw_optimize(a_work, time_limit=t_alloc)
                try_seq(a_opt, f"lpfw_n{target_n}")

    # === Phase 6: Final refinement ===
    print(f"\n=== Phase 6: Final refinement (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic()
        if rem > 40:
            t_f = min(rem * 0.3, 100)
            a_opt, _ = fista_optimize(best_a, time_limit=t_f, lr0=0.003,
                                       num_peaks=3, temperature=0.0005)
            try_seq(a_opt, "final_fista")

        rem = deadline - time.monotonic()
        if rem > 30:
            a_opt, _ = lp_fw_optimize(best_a, time_limit=min(rem * 0.3, 60))
            try_seq(a_opt, "final_lpfw")

        # Perturbation-based escape
        rem = deadline - time.monotonic()
        n_best = len(best_a)
        for p in range(5):
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            if p == 0:
                noise = rng.standard_normal(n_best) * 0.003
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.005
            elif p == 2:
                # Frequency-based perturbation
                j = np.arange(n_best)
                noise = 0.003 * np.cos(2.0 * np.pi * rng.integers(1, 10) * j / n_best)
            elif p == 3:
                # Block perturbation
                block = n_best // 8
                noise = np.zeros(n_best)
                for _ in range(3):
                    start = rng.integers(0, n_best - block)
                    noise[start:start + block] += rng.standard_normal(block) * 0.004
            else:
                noise = rng.standard_normal(n_best) * 0.002

            f_pert = project_simplex(best_a + noise)
            t_alloc = min(rem * 0.3, 40)
            if t_alloc > 10:
                a_opt, _ = lp_fw_optimize(f_pert, time_limit=t_alloc)
                try_seq(a_opt, f"perturb_{p}")

    # === Phase 7: Last resort FISTA on best ===
    print(f"\n=== Phase 7: Last resort (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic()
        if rem > 15:
            a_opt, _ = smooth_max_subgrad_fista(best_a, time_limit=min(rem, 40),
                                                  lr0=0.002, n_peaks=20, temp=0.0005)
            try_seq(a_opt, "last_fista")

    # Fallback
    if best_a is None:
        best_a = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_a.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_a = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_a.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_a)})")
    return [float(x) for x in best_a]
