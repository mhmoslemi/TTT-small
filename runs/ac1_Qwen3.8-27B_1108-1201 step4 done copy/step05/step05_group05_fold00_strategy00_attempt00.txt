import numpy as np

def evaluate_sequence(sequence: list[float]) -> float:
    """Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid."""
    #---SecurityChecks---#
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf

    # Convert list elements to floats and perform defensive clipping
    sequence = [float(x) for x in sequence]
    # Defensive clipping against negative numbers
    sequence = [max(0, x) for x in sequence]
    # Defensive clipping against too large numbers
    sequence = [min(1000.0, x) for x in sequence]
    
    # Group processing for efficiency if needed
    # Check if the sum is too small
    sum_a = np.sum(sequence)
    if sum_a < 0.01:
        return np.inf

    n = len(sequence)
    # Auto-correlation using linear convolution (full mode)
    conv_result = np.convolve(sequence, sequence, mode='full')
    max_b = np.max(conv_result)
    # Objective is scale-invariant (division by sum^2)
    objective = float(2 * n * max_b / (sum_a ** 2))
    return objective

def fast_convolve(a):
    """Linear auto-correlation via FFT for efficiency."""
    n = len(a)
    conv_length = n + n - 1
    # If the convolution length is small, use direct convolution
    if conv_length < 10:
        return np.convolve(a, a)
    # Use FFT size (next power of two >= conv_length)
    fft_size = 1 << (conv_length - 1).bit_length() if conv_length else 0
    # Compute FFT
    A = np.fft.rfft(a, fft_size)
    conv_result = np.fft.irfft(A * A)

def project_simplex(v):
    """Projects v onto the non-negative simplex with sum(v)=1."""
    if isinstance(v, np.ndarray):
        s = np.sum(v)
    else:
        s = sum(v)
    if s < 1e-15:
        s = 1.0
    if isinstance(v, np.ndarray):
        v = v.astype(np.float64) / s
    else:

def create_uniform_population(n, size=50):

def tournament_selection(population, fitnesses, k=3):
    """Select one individual to be parent for crossover using tournament selection."""
    # Select k individuals randomly
    indices = sorted(random.sample(range(len(population)), k))
    # Get the best fitness and corresponding individual
    best_index = indices[0]
    best_fitness = fitnesses[best_index]
    for i in range(1, k):

            best_index = indices[i]
            best_fitness = fitnesses[indices[i]]
