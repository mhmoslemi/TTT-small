import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_peak(f, w_full):
    """Compute gradient of sum_k w_full[k] * (f*f)[k] w.r.t. f.
    w_full has length 2n-1. For efficiency, only process non-zero w entries.
    grad[m] = sum_k w[k] * (f[k-m] + f[m-k]) where out-of-range = 0
    """
    n = len(f)
    grad = np.zeros(n)
    nz = np.nonzero(w_full > 1e-15)[0]
    for k in nz:
        wk = w_full[int(k)]
        if wk < 1e-15:
            continue
        k = int(k)
        # Contribution from f[k-m]: valid when 0 <= k-m < n => k-n+1 <= m <= k
        lo1 = max(0, k - n + 1)
        hi1 = min(n - 1, k)
        if lo1 <= hi1:
            ms = np.arange(lo1, hi1 + 1)
            grad[ms] += wk * f[k - ms]
        # Contribution from f[m-k]: valid when 0 <= m-k < n => k <= m <= k+n-1
        lo2 = max(0, k)
        hi2 = min(n - 1, k + n - 1)
        if lo2 <= hi2:
            ms2 = np.arange(lo2, hi2 + 1)
            grad[ms2] += wk * f[ms2 - k]
    return grad

    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0

            w_sum = w_rel.sum()
            if w_sum < 1e-300:

            if w_sum > 1e10:

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    # Ensure no zeros for multiplicative update
    f = np.maximum(f, 1e-10)
    f = f / f.sum()
    n = len(f)
    conv_len = 2 * n - 1

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        cur_max = conv.max()

        # Compute peak weighting
        k_top = min(max_peaks, conv_len)
        idx = np.argpartition(conv, -k_top)[-k_top:]
        idx = idx[conv[idx] > 0.995 * cur_max]
        if len(idx) == 0:
            idx = np.array([np.argmax(conv)])
        vals = conv[idx]
        w_rel = np.exp(-beta * (cur_max - vals))
        w_sum = w_rel.sum()
        if w_sum < 1e-300:
            w_full = np.zeros(conv_len)
            w_full[int(idx[0])] = 1.0
        else:
            w_full = np.zeros(conv_len)
            w_full[idx] = w_rel / w_sum

        grad = compute_grad_peak(f, w_full)
        grad_max = np.abs(grad).max()
        if grad_max < 1e-15:
            break

        # Multiplicative weights update
        # f_new[i] ∝ f[i] * exp(-lr * grad[i] / grad_max)
        updates = np.exp(-lr * grad / grad_max)
        f_new = f * updates
        f_new_sum = f_new.sum()
        if f_new_sum < 1e-15 or not np.isfinite(f_new_sum):
            f_new = f.copy()
        else:
            f_new = f_new / f_new_sum
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                lr *= 0.95
            if no_improve > 100:
                # Try additive update instead
                f = best_f.copy()
                lr = lr_init * 0.5
                no_improve = 0
                # Switch to small additive step
                v = -lr * grad / grad_max
                f_new = project_simplex(f + v)
                conv_new = fast_convolve(f_new)
                if conv_new.max() < best_obj:
                    f = f_new
                    best_obj = conv_new.max()
                    best_f = f_new.copy()
                    no_improve = 0

        if beta < beta_end:
            beta *= 1.01

    return best_f, best_obj


def directed_peak_optimize(f0, time_limit, n_peaks=5, lr_init=0.003, momentum=0.93):
    """Focus gradient on the very top peaks only."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv_len = 2 * n - 1

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    v = np.zeros(n)
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        cur_max = conv.max()

        # Select top peaks
        k_top = min(n_peaks, conv_len)
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        # Weight: emphasize highest
        pv = conv[peaks].copy()
        pv -= pv.min()
        if pv.max() < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = pv / pv.sum()

        w_full = np.zeros(conv_len)
        w_full[peaks] = w

        grad = compute_grad_peak(f, w_full)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad /= grad_norm

        v = momentum * v - cur_lr * grad
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                cur_lr *= 0.95
            if no_improve > 200:
                rng = np.random.default_rng((it * 7919 + 13) % 100003)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                v = np.zeros(n)
                cur_lr = max(lr_init * 0.3, 0.0001)
                no_improve = 0

    return best_f, best_obj


def backtracking_lse(f0, time_limit, beta=100.0, max_peaks=150):
    """LSE with backtracking line search for guaranteed decrease."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv_len = 2 * n - 1

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    lr = 2.0 / n
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        cur_max = conv.max()

        # Weighted subgradient
        k_top = min(max_peaks, conv_len)
        idx = np.argpartition(conv, -k_top)[-k_top:]
        idx = idx[conv[idx] > 0.99 * cur_max]
        if len(idx) == 0:
            idx = np.array([np.argmax(conv)])
        # Use beta-weighted geometric
        vals = conv[idx]
        scaled = beta * (cur_max - vals)
        w_rel = np.exp(-scaled)
        w_sum = w_rel.sum()
        w_full = np.zeros(conv_len)
        w_full[idx] = w_rel / w_sum

        grad = compute_grad_peak(f, w_full)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad /= grad_norm

        # Backtracking line search
        best_alpha = 0.0
        best_val = cur_max
        alpha = lr
        for _ in range(10):
            f_try = project_simplex(f - alpha * grad)
            val = fast_convolve(f_try).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
            alpha *= 0.7
            if alpha < 1e-8:
                break

        if best_alpha > 1e-10:
            f = project_simplex(f - best_alpha * grad)
            conv = fast_convolve(f)
            obj = conv.max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
                lr = min(best_alpha * 1.2, 10.0 / n)
            else:
                no_improve += 1
                lr = best_alpha * 0.7
        else:
            no_improve += 1
            lr *= 0.7

        if no_improve > 100:
            f = best_f.copy()
            lr = max(1.0 / n, 0.001)
            no_improve = 0

        # Slowly increase beta
        if it % 50 == 0:
            beta = min(beta * 1.5, 20000.0)

    return best_f, best_obj


def coordinate_optimize(f0, time_limit, n_peaks=10):
    """Optimize individual coordinates that most affect the top peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv_len = 2 * n - 1

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        cur_max = conv.max()
        k_top = min(n_peaks, conv_len)
        peaks = np.argpartition(conv, -k_top)[-k_top:]

        # For each peak, find the coordinate m that most increases conv at that peak
        # and try to decrease it
        improved = False
        for k in peaks:
            k = int(k)
            if time.monotonic() - t0 > time_limit:
                break
            # grad of conv[k] w.r.t f[m] = f[k-m] + f[m-k] (valid terms)
            # We want to decrease conv[k], so decrease f at positions where grad is large
            # and increase f at positions where grad contribution is small
            lo1 = max(0, k - n + 1)
            hi1 = min(n - 1, k)
            lo2 = max(0, k)
            hi2 = min(n - 1, k + n - 1)

            grad_k = np.zeros(n)
            if lo1 <= hi1:
                ms = np.arange(lo1, hi1 + 1)
                grad_k[ms] += f[k - ms]
            if lo2 <= hi2:
                ms2 = np.arange(lo2, hi2 + 1)
                grad_k[ms2] += f[ms2 - k]

            # Try to shift mass from high-grad positions to low-grad positions
            # Pick the top and bottom of grad_k
            gmax_idx = np.argmax(grad_k)
            gmin_idx = np.argmin(grad_k)
            if grad_k[gmax_idx] > 0 and f[gmax_idx] > 2 * f.mean():
                # Try small transfer
                for frac in [0.01, 0.02, 0.05]:
                    delta = f[gmax_idx] * frac
                    f_try = f.copy()
                    f_try[gmax_idx] -= delta
                    f_try[gmin_idx] += delta
                    f_try = np.maximum(f_try, 0)
                    st = f_try.sum()
                    f_try /= st
                    val = fast_convolve(f_try).max()
                    if val < best_obj - 1e-13:
                        f = f_try
                        best_obj = val
                        best_f = f.copy()
                        improved = True
                        break

        if not improved and it % 50 == 0:
            # Random perturbation
            rng = np.random.default_rng(it * 31)
            f = project_simplex(f + rng.standard_normal(n) * 0.002)
            conv_check = fast_convolve(f).max()
            if conv_check > best_obj * 1.01:
                f = best_f.copy()

    return best_f, best_obj


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def propose_candidate(seed=42, budget_s=1800):
