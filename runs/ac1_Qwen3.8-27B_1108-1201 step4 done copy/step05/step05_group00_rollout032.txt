    - Frequency domain: flatten the power spectrum (A^2) to reduce peaks
    """

        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        A = np.fft.rfft(a, fft_size)
        B = A * A  # This is the spectrum of the circular convolution

        c = np.fft.irfft(B, fft_size)[:size]
        max_c = c.max()
        
        # Target: flatten the convolution
        # Modify B to reduce peaks
        # Strategy: where c has peaks, reduce the corresponding frequency components
        # But the relationship is through DFT, so we need a more global approach
        
        # Alternative: use "spectral factorization" approach
        # We want IDFT(B) to be flat, i.e., B should be close to constant (in DFT sense)
        # The DFT of c is B. If c were flat (all = mean), then B[0] = sum(c) and B[m] = 0 for m>0.
        # But that's impossible (B[0] = |sum(a)|^2 = 1, and B[m] for m>0 would need to be 0,
        # meaning A[m] = 0 for m > 0, which means a is a delta function).
        
        # Better approach: partially flatten
        # Reduce the "sidelobe" energy in the frequency domain
        # B[0] is the DC component = (sum a)^2 = 1
        # The other components determine the "ripple" in c
        
        # Modify B: reduce the magnitude of non-DC components
        # This makes c more "DC-like" (flat)
        B_mod = B.copy()
        # Keep DC, scale others
        B_mod[1:] *= (1.0 - alpha)
        
        # But this changes the total energy... let's be more careful
        # Actually, we want to minimize max(c), not just flatten
        # A better approach: iteratively reduce the peaks
        
        # Peak reduction: find peak locations, add "anti-peak" in frequency
        peak_idx = np.argmax(c)
        # The contribution of frequency m to c[k] is (1/fft_size) * B[m] * exp(2πi*m*k/fft_size)
        # To reduce c[peak_idx], we want to reduce the constructive interference at that point
        
        # Simpler approach: just reduce the overall ripple
        # Compute the "ripple" = c - mean(c)
        ripple = c - c.mean()
        # Scale the ripple down
        c_target = c - alpha * ripple  # = (1-alpha)*c + alpha*mean(c)*ones
        
        # Transform back to frequency
        B_target = np.fft.rfft(c_target, fft_size)
        
        # Blend: new B = (1-alpha)*B + alpha*B_target
        # But we need to recover a, not just c
        # a = IDFT(A) where A^2 = B
        # If we change B, we need to find new A such that A^2 = B_new and a = IDFT(A) >= 0
        
        # Use iterative amplitude change:
        # A_new[m] = A[m] * (B_target[m] / B[m])^0.5 (in magnitude), preserving phase
        A_new = A.copy()
        for m in range(len(A)):
            if abs(B[m]) > 1e-15:
                ratio = abs(B_target[m] / B[m])
                if ratio < 0:
                    ratio = 0
                A_new[m] = A[m] * np.sqrt(min(ratio, 2.0))  # clip for stability
            else:
                A_new[m] = A[m] * np.sqrt(max(0, abs(B_target[m])))
        
        # Recover a
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        
        # Enforce non-negativity and normalization
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:

        a_new = a_new / s_new

        if it % 100 == 0:

    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Minimum-norm subgradient (solves QP for optimal active peak weights)
- Nesterov momentum with adaptive backtracking line search
- Spectral error reduction for diversity
- Multi-n exploration with diverse initializations
- Bundle-based direction for escape from plateaus

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgrad(a, c, rel_eps):
    """Compute minimum-norm subgradient of max_k c_k(a)."""
    n = len(a)
    max_c = c.max()
    threshold = max_c - rel_eps * max_c
    active = np.where(c >= threshold)[0]
    K = len(active)

    if K == 0:
        k = int(np.argmax(c))
        g = np.zeros(n)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            i_vals = np.arange(js, je + 1)
            g[i_vals] = 2.0 * a[k - i_vals]
        return g

    # Gradient for each active peak: grad_k[i] = 2*a[k-i] for valid k-i
    grads = np.zeros((K, n))
    for idx in range(K):
        k = int(active[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            i_vals = np.arange(js, je + 1)
            grads[idx, i_vals] = 2.0 * a[k - i_vals]

    if K == 1:
        return grads[0].copy()

    # QP: min (1/2) lam^T H lam s.t. 1^T lam = 1, lam >= 0
    H = grads @ grads.T

    # Active set method
    lam_set = list(range(K))
    lam = np.ones(K) / K
    for _ in range(K + 1):
        m = len(lam_set)
        if m == 0:
            lam = np.ones(K) / K
            break
        if m == 1:
            lam = np.zeros(K)
            lam[lam_set[0]] = 1.0
            break
        idx_arr = np.array(lam_set)
        H_sub = H[np.ix_(idx_arr, idx_arr)]
        try:
            h_inv_1 = np.linalg.solve(H_sub, np.ones(m))
        except np.linalg.LinAlgError:
            h_inv_1 = np.linalg.lstsq(H_sub, np.ones(m), rcond=None)[0]
        s = h_inv_1.sum()
        if abs(s) < 1e-15:
            lam_sub = np.ones(m) / m
        else:
            lam_sub = h_inv_1 / s
        if (lam_sub >= -1e-12).all():
            lam = np.zeros(K)
            for i, orig_idx in enumerate(lam_set):
                lam[orig_idx] = max(0.0, lam_sub[i])
            break
        else:
            worst_local = np.argmin(lam_sub)
            lam_set.remove(lam_set[worst_local])

    return lam @ grads

def spectral_reduce(a0, time_limit):
    """
    Fienup-style spectral error reduction.
    Alternates time-domain (non-neg) and frequency-domain (flatten) constraints.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    alpha = 0.15
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        A = np.fft.rfft(a, fft_size)
        B = A * A
        c = np.fft.irfft(B, fft_size)[:size]

        # Flatten: reduce ripple in c
        c_target = c - alpha * (c - c.mean())
        B_target = np.fft.rfft(c_target, fft_size)

        # Update A: adjust magnitude toward target, preserve phase
        A_new = A.copy()
        for m in range(len(A)):
            B_m = B[m]
            Bt_m = B_target[m]
            if abs(B_m) > 1e-18:
                mag_ratio = abs(Bt_m) / abs(B_m)
                mag_ratio = np.clip(mag_ratio, 0.0, 2.0)
                A_new[m] = A[m] * np.sqrt(mag_ratio)
            else:
                A_new[m] = 0.0

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        c_new = fast_convolve(a_new)
        obj = c_new.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            alpha = min(alpha * 1.02, 0.4)
        else:
            a = project_simplex(0.5 * a + 0.5 * a_new)
            alpha *= 0.95
            if alpha < 0.02:
                alpha = 0.15
                rng = np.random.default_rng(it * 31 + 7)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.004)

    return best_a, best_obj

def generate_initializations(n, parent, rng):

    inits = []

    # Edge concentration
    x = np.arange(n, dtype=np.float64)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Ramp (front-loaded)
    ramp = np.linspace(3.0, 0.5, n)
    inits.append(ramp / ramp.sum())

    # Ramp (back-loaded)
    ramp2 = np.linspace(0.5, 3.0, n)
    inits.append(ramp2 / ramp2.sum())

    # Two peaks
    g1 = np.exp(-0.5 * ((x - 0.2 * (n - 1)) / (0.1 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.8 * (n - 1)) / (0.1 * n)) ** 2)
    inits.append((g1 + g2) / (g1 + g2).sum())

    # Cosine window
    c_win = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1))) + 0.1
    inits.append(c_win / c_win.sum())

    # Interpolated parent
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits

        256, 384, 512, 768, 1024, 1536, 2048
    ]))

    # Ensure parent length is included
    if n_parent not in candidate_ns:
        candidate_ns.append(n_parent)
    candidate_ns.sort()

    # Filter: skip very large if time is tight
    if time.monotonic() + (budget_s - 30) - time.monotonic() < 1200:
        candidate_ns = [n for n in candidate_ns if n <= 1536]

    n_results = {}
    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 900
        if remaining < 30:
            break

        # Generate a few initializations
        inits = generate_initializations(n_try, parent, rng)

        best_for_n = None
        best_for_n_obj = float('inf')

        t_alloc = min(remaining / max(1, len(candidate_ns) - candidate_ns.index(n_try)), 45)
        t_alloc = max(t_alloc, 15)

        for init_idx, init_a in enumerate(inits[:4]):  # Try up to 4 inits
            if time.monotonic() > deadline - 900:
                break
            t_sub = t_alloc / max(1, 4 - init_idx)
            a_opt, obj = subgrad_optimize(
                init_a, time_limit=min(t_sub, 30),
                rel_eps=0.01, lr_init=0.02
            )
            if obj < best_for_n_obj:
                best_for_n_obj = obj
                best_for_n = a_opt

        if best_for_n is not None:
            n_results[n_try] = (best_for_n, best_for_n_obj)
            try_seq(best_for_n, f"explore_n{n_try}")

    print(f"  Phase 1 best: {best_score:.8f} (elapsed {time.monotonic()-t1:.0f}s)")

    # Sort n_results by objective
    sorted_ns = sorted(n_results.items(), key=lambda x: x[1][1])
    print(f"  Top n candidates: {[(n, f'{obj:.6f}') for n, (a, obj) in sorted_ns[:4]]}")

    # Phase 2: Deep optimization on top candidates
    print("\n=== Phase 2: Deep Subgradient Optimization ===")
    t2 = time.monotonic()

    for n_cand, (a_cand, obj_cand) in sorted_ns[:3]:
        remaining = deadline - time.monotonic() - 400
        if remaining < 60:
            break

        t_alloc = min(remaining, 500)
        print(f"  Deep subgrad n={n_cand}, t={t_alloc:.0f}s, start_obj={obj_cand:.8f}")

        # Run main subgradient optimizer with tight epsilon
        a_opt, obj_opt = subgrad_optimize(
            a_cand, time_limit=t_alloc * 0.7,
            rel_eps=0.003, lr_init=0.012
        )
        try_seq(a_opt, f"deep_n{n_cand}")

        # Also try from the parent interpolated to this n
        if best_seq is not None:
            px = np.interp(np.linspace(0, 1, n_cand),
                           np.linspace(0, 1, len(best_seq)), best_seq)
            px = np.maximum(px, 0)
            px = px / px.sum()
            remaining2 = deadline - time.monotonic() - 300
            if remaining2 > 30:
                a_opt2, obj_opt2 = subgrad_optimize(
                    px, time_limit=min(remaining2, 120),
                    rel_eps=0.005, lr_init=0.015
                )
                try_seq(a_opt2, f"deep2_n{n_cand}")

    print(f"  Phase 2 best: {best_score:.8f} (elapsed {time.monotonic()-t2:.0f}s)")

    # Phase 3: Spectral reduction + coordinate polish
    print("\n=== Phase 3: Spectral + Coordinate ===")
    t3 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Spectral reduce
        remaining = deadline - time.monotonic() - 200
        if remaining > 30:
            t_alloc = min(remaining, 150)
            print(f"  Spectral reduce n={n_best}, t={t_alloc:.0f}s")
            a_spec, _ = spectral_reduce(best_seq, time_limit=t_alloc)
            try_seq(a_spec, "spectral")
            if a_spec is not None and best_seq is not None:
                # Blend spectral result with current best
                for blend in [0.3, 0.5, 0.7]:
                    a_blend = (1 - blend) * best_seq + blend * a_spec
                    a_blend = project_simplex(a_blend)
                    try_seq(a_blend, f"spec_blend_{blend}")

        # Coordinate polish
        remaining = deadline - time.monotonic() - 100
        if remaining > 20:
            t_alloc = min(remaining, 80)
            print(f"  Coord polish n={n_best}, t={t_alloc:.0f}s")
            a_coord, _ = coordinate_polish(best_seq, time_limit=t_alloc)
            try_seq(a_coord, "coord")

    print(f"  Phase 3 best: {best_score:.8f} (elapsed {time.monotonic()-t3:.0f}s)")

    # Phase 4: Final refinement - multiple short runs with different params
    print("\n=== Phase 4: Final Refinement ===")
    t4 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        configs = [
            {'rel_eps': 0.002, 'lr_init': 0.008},
            {'rel_eps': 0.005, 'lr_init': 0.015},
            {'rel_eps': 0.01, 'lr_init': 0.025},
            {'rel_eps': 0.003, 'lr_init': 0.01},
        ]

        for ci, cfg in enumerate(configs):
            remaining = deadline - time.monotonic() - 30
            if remaining < 15:
                break
            t_alloc = min(remaining, 60)
            print(f"  Refine config {ci} (eps={cfg['rel_eps']}, lr={cfg['lr_init']}), t={t_alloc:.0f}s")

            # Start from best or slightly perturbed
            if ci == 0:
                start = best_seq.copy()
            else:
                start = project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)

            a_opt, _ = subgrad_optimize(start, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"refine_{ci}")

        # Final coordinate polish
        remaining = deadline - time.monotonic() - 15
        if remaining > 10 and best_seq is not None:
            a_coord, _ = coordinate_polish(best_seq, time_limit=min(remaining, 30))
            try_seq(a_coord, "final_coord")

    print(f"  Phase 4 best: {best_score:.8f} (elapsed {time.monotonic()-t4:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
