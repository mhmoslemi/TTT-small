def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:

    # Check each element in the list for validity
    for x in sequence:

        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f):
    """Build sparse constraint matrix for the LP. Shape: (2n-1, n)."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A_cache=None):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    if A_cache is None:
        A_cache = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 25})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None

def compute_grad_weighted(f, w, max_peaks=300):
    """Compute gradient of sum_k w[k] * (f*f)[k] w.r.t. f.
    Uses only top max_peaks entries of w for efficiency.
    """
    n = len(f)
    conv_len = 2 * n - 1

    if len(w) <= max_peaks:
        sig_idx = np.argsort(w)[::-1]
    else:
        sig_idx = np.argpartition(w, -max_peaks)[-max_peaks:]
        sig_idx = sig_idx[w[sig_idx] > 1e-15 * w.max()]

    grad = np.zeros(n)
    for k in sig_idx:
        wk = w[k]
        if wk < 1e-15:
            continue
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)

        # I1: m in [lo, hi], contribution wk * f[k-m]
        if lo <= hi:
            ms = np.arange(lo, hi + 1)
            grad[ms] += wk * f[k - ms]

        # I2: k-m in [lo, hi] => m in [k-hi, k-lo], contribution wk * f[k-m]
        ms2_lo = max(0, k - hi)
        ms2_hi = min(n - 1, k - lo)
        if ms2_lo <= ms2_hi:
            ms2 = np.arange(ms2_lo, ms2_hi + 1)
            grad[ms2] += wk * f[k - ms2]

    return grad


def lse_optimize(f0, time_limit, beta_start=20.0, beta_end=5000.0, max_peaks=200):
    """Log-sum-exp smoothed gradient descent with corrected gradient."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = 2.0 / n
    momentum = 0.90
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        grad = compute_grad_weighted(f, w, max_peaks=max_peaks)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 200
            break
        grad /= grad_norm

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.003, 5.0 / n)
        else:
            no_improve += 1
            if no_improve >= 8:
                lr *= 0.93
                momentum = max(0.85, momentum - 0.004)
            if no_improve > 70:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(2.0 / n, 0.001)
                beta = min(beta * 1.6, beta_end)
                momentum = 0.88
                no_improve = 0

        if beta < beta_end:
            beta *= 1.004

    return best_f, best_obj


def directed_peak_grad(f0, time_limit, n_peaks=12, lr_init=0.003, momentum=0.93):
    """Directly target the top peaks with weighted subgradient."""
    f = np.maximum(f0.astype(np.float64), 0.0)

    conv_len = 2 * n - 1

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    v = np.zeros(n)
    cur_mom = momentum
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)

        # Select top peaks with exponential weighting
        k_top = min(n_peaks, conv_len)
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peaks]
        # Soft weighting: emphasize the highest peaks
        scaled = pv - pv.min()
        threshold = scaled.max() * 0.5
        w = np.where(scaled > threshold, np.exp(scaled - scaled.max()), 0.0)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= w_sum

        # Build full weight vector
        w_full = np.zeros(conv_len)
        w_full[peaks] = w

        grad = compute_grad_weighted(f, w_full, max_peaks=k_top)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 200
            break
        grad /= grad_norm

        v = cur_mom * v - cur_lr * grad
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 40 == 0:

                v = np.zeros(n)
                cur_lr = max(lr_init * 0.4, 0.0002)
                no_improve = 0

    return best_f, best_obj

def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def symmetric_init(n, freq=2.0, amplitude=0.1):
    """Create a symmetric sinusoidal perturbation of uniform."""
    base = np.ones(n) / n
    x = np.arange(n)
    sym = np.cos(2.0 * np.pi * freq * x / n)
    sym = sym[0:n//2]
    if n % 2 == 1:
        sym_full = np.concatenate([sym, [0.0], sym[::-1]])
    else:
        sym_full = np.concatenate([sym, sym[::-1]])
    f = base + amplitude * sym_full
    f = np.maximum(f, 0.0)
    f /= f.sum()
    return f
