Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using exact-subgradient Nesterov
acceleration, LP Frank-Wolfe with adaptive tightening, and multi-scale
coordinate descent with golden-section refinement.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(a, conv, top_k, temperature):
    """Compute weighted subgradient using top-k peaks."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    w = w / ws
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        m_lo = max(0, k - (n - 1))
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            indices = np.arange(m_lo, m_hi + 1)
            sg[indices] += 2.0 * w[i] * a[k - indices]
    return sg

def nesterov_subgrad(f0, time_limit, lr0=0.01, mom0=0.85, top_k=50, temp0=0.0001):
    """Nesterov-accelerated projected subgradient with exact gradient."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    y = f.copy()
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr0
    cur_mom = mom0
    cur_temp = temp0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute subgradient at current y (Nesterov lookahead)
        y_conv = fast_convolve(y)
        sg = compute_subgradient(y, y_conv, top_k, cur_temp)

        f = f_new
        
        conv = fast_convolve(f)
        co = conv.max()

            best_f = f.copy()
            no_improve = 0

def solve_lp_direction(f, M, n):
    c = -np.ones(n)
    # Build constraint matrix efficiently
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def coordinate_descent(f0, time_limit):
    """Coordinate descent with golden section on each coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    no_pass_improve = 0
    
    GR = (1.0 + np.sqrt(5.0)) / 2.0
    inv_gr = 1.0 / GR
    mask = np.ones(n, dtype=bool)
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        pass_improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-14:
                continue
            
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue
            
            # Golden section on x = f[j]
            lo = 0.0
            hi = min(0.9999, 3.0 * old_val)
            
            def eval_at(x):
                ft = f.copy()
                ft[j] = x
                rest = (1.0 - x) / denom
                if rest < 0:
                    return float('inf')
                mask[j] = False
                ft[mask] = f[mask] * rest
                mask[j] = True
                return fast_convolve(ft).max()
            
            c1 = lo + inv_gr * (hi - lo)
            c2 = lo + (1.0 - inv_gr) * (hi - lo)
            f1 = eval_at(c1)
            f2 = eval_at(c2)
            
            for _ in range(15):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + inv_gr * (hi - lo)
                    f1 = eval_at(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + (1.0 - inv_gr) * (hi - lo)
                    f2 = eval_at(c2)
            
            opt_x = (lo + hi) / 2.0
            opt_val = min(f1, f2)
            
            cur_obj = fast_convolve(f).max()
            if opt_val < cur_obj - 1e-13:
                f[j] = opt_x
                rest = (1.0 - opt_x) / denom
                mask[j] = False
                f[mask] = f[mask] * rest
                mask[j] = True
                f = f / f.sum()
                pass_improved = True
                new_obj = fast_convolve(f).max()
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_f = f.copy()
        
        if pass_improved:
            no_pass_improve = 0
        else:
            no_pass_improve += 1
            if no_pass_improve >= 3:
                break
    
    return best_f, best_obj

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def cosine_init(n, freq, amp):
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    else:
        a = np.ones(n) / n
    return a

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, conv, top_k, temperature):

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, conv, top_k, temperature):
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    w = w / ws
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        m_lo = max(0, k - (n - 1))
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            indices = np.arange(m_lo, m_hi + 1)
            sg[indices] += 2.0 * w[i] * a[k - indices]
    return sg

    # === Phase 7: Final polish ===
    rem = deadline - time.monotonic()
    if rem > 20 and best_seq is not None:
        print(f"\n=== Phase 7: Final polish (best={best_score:.8f}) ===")
        n_best = len(best_seq)

        # Very fine subgradient
        t7 = min(rem * 0.35, 80)
        if t7 > 10:
            a_opt, _ = nesterov_subgrad(best_seq, time_limit=t7, lr0=0.001, mom0=0.95,
                                        top_k=max(10, n_best // 15), temp0=0.000008)
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  NEW BEST [final_subgrad]: {best_score:.8f}")

        # Quick CD
        rem = deadline - time.monotonic()
        if rem > 15:
            t7b = min(rem * 0.3, 60)
            a_opt, _ = coordinate_descent(best_seq, time_limit=t7b)
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  NEW BEST [final_cd]: {best_score:.8f}")

        # LP-FW short burst
        rem = deadline - time.monotonic()
        if rem > 20:
            t7c = min(rem * 0.4, 80)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t7c, verbose=False)
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  NEW BEST [final_lpfw]: {best_score:.8f}")

        # Perturbation restarts
        rem = deadline - time.monotonic()
        for p in range(4):
            rem = deadline - time.monotonic()
            if rem < 12:
                break
            n_cur = len(best_seq)
            scale = 0.001 * (1.0 + 0.5 * p)
            noise = rng.standard_normal(n_cur) * scale
            f_pert = project_simplex(best_seq + noise)
            t_alloc = min(rem * 0.3, 30)
            a_opt, _ = nesterov_subgrad(f_pert, time_limit=t_alloc, lr0=0.001, mom0=0.93,
                                        top_k=max(10, n_cur // 15), temp0=0.000005)
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  NEW BEST [perturb_{p}]: {best_score:.8f} (n={len(best_seq)})")

    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
