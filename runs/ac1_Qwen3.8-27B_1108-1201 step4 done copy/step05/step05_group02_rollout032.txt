import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_min_peak_lp(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0

    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None

        if g is None:
            return None, None
        gs = g.sum()
        if gs < 1e-15:
            g = np.ones(n) / n
        else:
            g = g / gs
    return g, lp_t

def optimize_lp_fw(f0, time_limit, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()

    no_improve = 0

        g, lp_t = lp_fw_step(f, n, use_min_peak=True)

            else:

                alphas = [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]

                    ft = np.maximum(ft, 0.0)

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def solve_min_peak_lp(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0

    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None


def solve_topk_peak_lp(f, n, top_k=10):

    conv = fast_convolve(f)
    peak_indices = np.argpartition(conv, -top_k)[-top_k:]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((top_k, n + 1))
    for idx, k in enumerate(peak_indices):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[idx, j] = f[j]
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(top_k)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None

def subgradient_optimize(f0, time_limit, lr=0.003, momentum=0.93, top_k=30, temp=0.00005):
    """Subgradient method with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute weighted subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            kg = k - idx
            mask = (kg >= 0) & (kg < n)
            sg[idx[mask]] += 2.0 * w[i] * f[kg[mask]]

        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)
        conv = fast_convolve(f_new)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            no_improve += 1
            f = f_new

        cur_lr *= 0.9999
        cur_temp *= 1.005

        if no_improve > 300:
            rng = np.random.default_rng(it * 47 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0

    return best_f, best_obj

def make_w_shape(n, width_frac=0.15):
    """Create a W-shaped sequence (high at both ends, low in middle)."""
    f = np.zeros(n)
    center = n // 2
    width = int(n * width_frac)
    # Two Gaussian bumps at the ends
    for i in range(n):
        dist_left = abs(i - width // 2)
        dist_right = abs(i - (n - 1 - width // 2))
        f[i] = np.exp(-dist_left**2 / (2 * (width * 0.4)**2)) + \
               np.exp(-dist_right**2 / (2 * (width * 0.4)**2))
    f = np.maximum(f, 0)
    s = f.sum()
    if s > 1e-15:
        f /= s
    return f


def make_gaussian(n, sigma_frac=0.25):
    """Create a Gaussian-shaped sequence."""
    x = np.linspace(-1, 1, n)
    f = np.exp(-x**2 / (2 * sigma_frac**2))
    f = np.maximum(f, 0)
    s = f.sum()
    if s > 1e-15:
        f /= s
    return f


def make_cosine(n, freq=1.0):
    """Create a cosine-modulated sequence."""
    x = np.linspace(0, 1, n)
    f = (1 + np.cos(2 * np.pi * freq * x)) / 2
    f = np.maximum(f, 0)
    s = f.sum()
    if s > 1e-15:
        f /= s
    return f

    print(f"\n=== Phase 6: Peak splitting (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_ps = min(rem * 0.2, 250)
        if t_ps > 20:
            print(f"  n={len(best_seq)}, t={t_ps:.0f}s")
            a_opt, obj = peak_splitting_search(best_seq, time_limit=t_ps)
            try_seq(a_opt, "peak_split")

    # === Phase 7: LP-FW refined on best found so far ===
    print(f"\n=== Phase 7: LP-FW refined (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t7 = min(rem * 0.25, 300)
        if t7 > 30:
            print(f"  n={len(best_seq)}, t={t7:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t7, verbose=True)
            try_seq(a_opt, "lpfw_refined")

    # === Phase 8: Cosine init + LP-FW ===
    print(f"\n=== Phase 8: Cosine init ===")
    rem = deadline - time.monotonic()
    t8 = min(rem * 0.1, 120)
    if t8 > 15:
        cos_init = make_cosine(len(best_seq) if best_seq is not None else base_n, freq=2.0)
        print(f"  n={len(cos_init)}, t={t8:.0f}s")
        a_opt, obj = lp_fw_optimize(cos_init, time_limit=t8)
        try_seq(a_opt, "lpfw_cosine")

    # === Phase 9: Final polish with subgradient ===
    print(f"\n=== Phase 9: Final polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            a_opt, obj = subgradient_optimize(best_seq, time_limit=min(rem, 100),
                                              lr=0.0008, momentum=0.95,
                                              top_k=max(20, len(best_seq) // 12),
                                              temp=0.00002)
            try_seq(a_opt, "sg_polish")

        rem = deadline - time.monotonic() - 20
        if rem > 15:
            a_opt, obj = peak_splitting_search(best_seq, time_limit=min(rem, 80))
            try_seq(a_opt, "ps_polish")

    # === Phase 10: Perturbation + quick optimize ===
    print(f"\n=== Phase 10: Perturbations (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(8):
            rem = deadline - time.monotonic() - 10
            if rem < 15:
                break
            noise = rng.standard_normal(n_best) * (0.0008 + 0.0004 * p)
            f_pert = project_simplex(best_seq + noise)
            t_p = min(rem * 0.5, 40)
            if t_p < 10:
                break
            a_opt, obj = lp_fw_optimize(f_pert, time_limit=t_p)
            try_seq(a_opt, f"perturb_{p}")

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
