"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based Frank-Wolfe with
diverse initializations (flat-spectrum, cosine-modulated, chirp), multi-length
exploration, and subgradient refinement.

import numpy as np
from scipy.optimize import linprog
import time

# Global deadline
_deadline = None

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_constraint_matrix(f, n):
    """Build the (2n-1) x n constraint matrix for the LP."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A

def solve_lp(f, M, n):
    """Solve: max sum(g) s.t. A@f * g <= M, g >= 0.
    Actually: max sum(g) s.t. (convolve-like constraints) <= M, g >= 0.
    The constraint is: for each k, sum_j f[j] * g[k-j] <= M
    which is: A @ g <= M * ones, where A[k,j] = f[j] if 0<=k-j<n
    """
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def evaluate_local(a):
    """Fast local evaluation (same as evaluate_sequence but without overhead)."""
    a = np.maximum(a, 0.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    max_b = conv.max()
    return float(2 * n * max_b / (s * s))

def flat_spectrum_init(n, seed=0):
    """Initialize with approximately flat power spectrum."""
    rng = np.random.default_rng(seed)
    # Random phases
    m = 2 ** (int(np.log2(n)) + 1) if n > 1 else 1
    # Actually, let's use the next power of 2 >= n
    fft_n = 1
    while fft_n < n:
        fft_n *= 2
    
    phases = rng.uniform(0, 2 * np.pi, fft_n)
    F = np.exp(1j * phases) * (1.0 / np.sqrt(fft_n))
    
    # Make F conjugate-symmetric for real output
    F[0] = abs(F[0])
    if fft_n % 2 == 0:
        F[fft_n // 2] = abs(F[fft_n // 2])

        if i < fft_n - i:
            F[fft_n - i] = np.conj(F[i])
    
    a = np.real(np.fft.ifft(F))[:n]
    a = np.abs(a) + 1e-8
    a = a / a.sum()

def interpolate_to_length(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

                if ft2 < 0 or ft2 > 1.0:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = ft[mask] * (ft2 / (1.0 - old_val)) if (1.0 - old_val) > 1e-10 else 0
                if ft.sum() < 1e-10:
                    continue
                ft = ft / ft.sum()
                val = fast_convolve(ft).max()
                if val < best_val_at_x - 1e-14:
                    best_val_at_x = val
                    best_x = x

            if best_x != old_val:
                f[j] = best_x
                ft2 = 1.0 - best_x
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] = f[mask] * (ft2 / (1.0 - old_val)) if (1.0 - old_val) > 1e-10 else 0
                if f.sum() > 1e-10:
                    f = f / f.sum()
                pass_improved = True

        if pass_improved:
            no_pass_improve = 0
            cur_obj = fast_convolve(f).max()
            if cur_obj < best_obj - 1e-14:
                best_obj = cur_obj
                best_f = f.copy()
        else:
            no_pass_improve += 1
            if no_pass_improve >= 3:
                break

    return best_f, best_obj


def flat_spectrum_init(n, seed=0):
    """Initialize with approximately flat power spectrum (random phases)."""
    rng = np.random.default_rng(seed)
    fft_n = 1
    while fft_n < n:
        fft_n *= 2

    phases = rng.uniform(0, 2 * np.pi, fft_n)
    F = np.exp(1j * phases) / np.sqrt(fft_n)

    # Enforce conjugate symmetry for real output
    F[0] = abs(F[0])
    if fft_n % 2 == 0:
        F[fft_n // 2] = abs(F[fft_n // 2])
    for i in range(1, fft_n // 2 + 1):
        j = fft_n - i
        if j < fft_n and i != j:
            F[j] = np.conj(F[i])

    a = np.real(np.fft.ifft(F))[:n]
    a = np.abs(a) + 1e-10
    a = a / a.sum()
    return a


def cosine_init(n, freq, amp):
    """Cosine-modulated uniform initialization."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def evaluate_local(a):
    """Quick local evaluation matching evaluate_sequence formula."""
    a = np.maximum(a, 0.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: diverse initializations -> LP-Frank-Wolfe at
    multiple lengths -> subgradient refinement -> coordinate descent polish.
    """
    global _g_deadline
    _g_deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    def _try(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512

    # === Phase 1: Diverse initializations (4 min) ===
    print("\n=== Phase 1: Diverse initializations ===")
    inits = []

    _try(parent, "parent")
    inits.append(('parent', parent.copy(), evaluate_local(parent)))

    for n in [384, 512, 640, 768, 896, 1024]:
        for freq in [1, 2, 3]:
            for amp in [0.15, 0.3, 0.5]:
                a = cosine_init(n, freq, amp)
                s = evaluate_local(a)
                inits.append((f'cos_n{n}_f{freq}_a{amp}', a, s))
                if s < best_score:
                    _try(a, f'cos_n{n}_f{freq}_a{amp}')

    for n in [512, 768, 1024]:
        for s_idx in range(4):
            a = flat_spectrum_init(n, seed=s_idx * 1000 + n)
            s = evaluate_local(a)
            inits.append((f'flat_n{n}_s{s_idx}', a, s))
            if s < best_score:
                _try(a, f'flat_n{n}_s{s_idx}')

    # Chirp
    for n in [512, 768, 1024]:
        j = np.arange(n)
        a = 1.0 + 0.3 * np.cos(np.pi * j * j / n)
        a = np.maximum(a, 1e-10)
        a = a / a.sum()
        s = evaluate_local(a)
        inits.append((f'chirp_n{n}', a, s))
        if s < best_score:
            _try(a, f'chirp_n{n}')

    inits.sort(key=lambda x: x[2])
    print(f"  Top 5: " + ", ".join(f"{x[0]}={x[2]:.6f}" for x in inits[:5]))

    # === Phase 2: LP-FW on best initializations (50 min) ===
    print(f"\n=== Phase 2: LP-FW (best={best_score:.8f}) ===")

    # Select diverse top initializations (different lengths preferred)
    selected = []
    seen_n = set()
    for name, a, s in inits:
        n = len(a)
        if n not in seen_n and len(selected) < 4:
            selected.append((name, a, s))
            seen_n.add(n)
    # Always include the single best
    if inits[0][1] not in [x[1] for x in selected]:
        selected.insert(0, inits[0])

    for name, a, s in selected[:4]:
        rem = _g_deadline - time.monotonic() - 700
        if rem < 30:
            break
        t_alloc = min(rem * 0.35, 350)
        if t_alloc < 20:
            break
        n = len(a)
        print(f"  LP-FW '{name}' n={n} init={s:.6f} t={t_alloc:.0f}s")
        a_opt, obj = lp_fw_optimize(a, time_limit=t_alloc, deadline=_g_deadline, verbose=True)
        _try(a_opt, f"lpfw_{name}")

    # === Phase 3: Multi-length interpolation + LP-FW (40 min) ===
    print(f"\n=== Phase 3: Multi-length (best={best_score:.8f}) ===")

    if best_seq is not None:
        for tn in [896, 1024, 1152, 1280, 1536]:
            rem = _g_deadline - time.monotonic() - 450
            if rem < 30:
                break
            if tn <= len(best_seq):
                continue
            f_interp = interpolate_to_length(best_seq, tn)
            t_alloc = min(rem * 0.3, 200)
            print(f"  LP-FW interp n={tn}, t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(f_interp, time_limit=t_alloc, deadline=_g_deadline, verbose=False)
            _try(a_opt, f"lpfw_ml_n{tn}")

    # === Phase 4: Subgradient refinement (25 min) ===
    print(f"\n=== Phase 4: Subgradient (best={best_score:.8f}) ===")

    if best_seq is not None:
        n_b = len(best_seq)
        rem = _g_deadline - time.monotonic() - 250
        if rem > 30:
            t_ns = min(rem * 0.5, 200)
            print(f"  Subgrad n={n_b} t={t_ns:.0f}s")
            a_opt, obj = subgradient_refine(best_seq, time_limit=t_ns, deadline=_g_deadline,
                                             lr0=0.006, mom0=0.82,
                                             top_k=max(30, n_b // 5), temp0=0.0008)
            _try(a_opt, "subgrad_1")

            rem = _g_deadline - time.monotonic() - 150
            if rem > 20 and best_seq is not None:
                a_opt, obj = subgradient_refine(best_seq, time_limit=min(rem, 100), deadline=_g_deadline,
                                                 lr0=0.002, mom0=0.88,
                                                 top_k=max(20, n_b // 8), temp0=0.0003)
                _try(a_opt, "subgrad_2")

    # === Phase 5: Coordinate descent polish (20 min) ===
    print(f"\n=== Phase 5: Coord Descent (best={best_score:.8f}) ===")

    if best_seq is not None:
        rem = _g_deadline - time.monotonic() - 80
        if rem > 20:
            t_cd = min(rem * 0.7, 180)
            print(f"  CD n={len(best_seq)} t={t_cd:.0f}s")
            a_opt, obj = coordinate_descent_polish(best_seq, time_limit=t_cd, deadline=_g_deadline)
            _try(a_opt, "coord_descent")

    # === Phase 6: Final LP-FW + perturbations (remaining) ===
    print(f"\n=== Phase 6: Final polish (best={best_score:.8f}) ===")

    if best_seq is not None:
        n_b = len(best_seq)

        # Final LP-FW
        rem = _g_deadline - time.monotonic() - 50
        if rem > 15:
            t_alloc = min(rem * 0.5, 80)
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, deadline=_g_deadline, verbose=False)
            _try(a_opt, "final_lpfw")

        # Subgradient
        rem = _g_deadline - time.monotonic() - 30
        if rem > 10:
            a_opt, obj = subgradient_refine(best_seq, time_limit=min(rem, 40), deadline=_g_deadline,
                                             lr0=0.003, mom0=0.85,
                                             top_k=max(15, n_b // 8), temp0=0.0002)
            _try(a_opt, "final_subgrad")

        # Quick perturbations
        for trial in range(4):
            rem = _g_deadline - time.monotonic() - 15
            if rem < 8:
                break
            noise = rng.standard_normal(n_b) * 0.003
            f_pert = project_simplex(best_seq + noise)
            t_alloc = min(rem * 0.5, 25)
            if t_alloc < 5:
                break
            a_opt, obj = subgradient_refine(f_pert, time_limit=t_alloc, deadline=_g_deadline,
                                             lr0=0.002, mom0=0.88,
                                             top_k=max(15, n_b // 8), temp0=0.0001)
            _try(a_opt, f"perturb_{trial}")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final check
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
