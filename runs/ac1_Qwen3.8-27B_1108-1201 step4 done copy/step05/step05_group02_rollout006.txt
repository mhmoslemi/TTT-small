"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def logsumexp_gradient_step(a, tau, top_k=None):
    """Compute gradient of the log-sum-exp smooth approximation of max(conv(a,a)).
    Returns the gradient vector of length n."""
    n = len(a)
    b = fast_convolve(a)  # length 2n-1
    
    if top_k is not None and top_k < len(b):
        # Only use top-k peaks
        idx = np.argpartition(b, -top_k)[-top_k:]
        b_work = b[idx].copy()
        p_full = np.zeros(len(b))
    else:
        b_work = b
        p_full = np.zeros(len(b))
    
    # Softmax
    b_scaled = b_work / tau
    b_scaled -= b_scaled.max()
    w = np.exp(b_scaled)
    w /= w.sum()
    
    if top_k is not None and top_k < len(b):
        p_full[idx] = w
    else:
        p_full = w
    
    # Gradient: g_i = 2 * sum_j p_full[i+j] * a[j]
    # This is np.correlate(p_full, a, mode='valid')
    g = 2.0 * np.correlate(p_full, a, mode='valid')
    return g, b


def smooth_gradient_descent(a0, time_limit, tau_start=0.01, tau_end=0.0001,
                            lr_start=0.01, momentum=0.9, top_k=None,
                            verbose=False, restart_interval=500):

    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Adaptive temperature
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** progress
        
        # Adaptive learning rate (decay)
        lr = lr_start * (0.999) ** it
        
        # Compute gradient
        g, b = logsumexp_gradient_step(a, tau, top_k=top_k)
        
        # Momentum update
        v = momentum * v - lr * g
        a_new = project_simplex(a + v)
        
        # Evaluate
        obj = fast_convolve(a_new).max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
            if verbose and it % 100 == 0:
                print(f"    SGD iter {it}: obj={best_obj:.10f}, tau={tau:.6f}")
        else:
            no_improve += 1
            a = a_new
        
        # Restart if stuck
        if no_improve > restart_interval:
            rng = np.random.default_rng(hash((it, n)) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            no_improve = 0
    
    return best_a, best_obj


def solve_lp_min_peak(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    n_constraints = 2 * n - 1
    # Build constraint matrix more efficiently
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def cosine_window(n):
    """Create a cosine window of length n."""
    x = np.linspace(0, np.pi, n)
    w = np.cos(x / 2) ** 2
    w /= w.sum()
    return w


def gaussian_window(n, sigma_frac=0.3):
    """Create a Gaussian window of length n."""
    x = np.arange(n) - (n - 1) / 2.0
    sigma = sigma_frac * n
    w = np.exp(-x**2 / (2 * sigma**2))
    w /= w.sum()
    return w

Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
1. Smooth log-sum-exp gradient descent with adaptive temperature
2. LP-Frank-Wolfe with min-peak LP formulation

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    if top_k is not None and top_k < len(b):
        idx = np.argpartition(b, -top_k)[-top_k:]
        b_work = b[idx].copy()
    else:
        idx = np.arange(len(b))
        b_work = b.copy()

    b_scaled = b_work / tau
    b_scaled -= b_scaled.max()
    w = np.exp(b_scaled)
    w /= w.sum()

    p_full = np.zeros(len(b))
    p_full[idx] = w

    g = 2.0 * np.correlate(p_full, a, mode='valid')
    return g, b


def smooth_gradient_descent(a0, time_limit, tau_start=0.005, tau_end=0.0001,
                            lr_start=0.005, momentum=0.9, top_k=None,
                            restart_interval=300, verbose=False):
    """Smooth gradient descent with log-sum-exp approximation and momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    best_obj = fast_convolve(a).max()
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** progress
        lr = lr_start * (0.9992) ** it

        g, b = logsumexp_gradient(a, tau, top_k=top_k)

        v = momentum * v - lr * g
        a_new = project_simplex(a + v)

        obj = fast_convolve(a_new).max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
            if verbose and it % 200 == 0:
                print(f"    SGD iter {it}: obj={best_obj:.10f}, tau={tau:.7f}, lr={lr:.6f}")
        else:
            no_improve += 1
            a = a_new

        if no_improve > restart_interval:
            rng = np.random.default_rng((it * 2654435761 + n) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            no_improve = 0

    return best_a, best_obj


def solve_lp_min_peak(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0

    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with min-peak LP subproblem."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g, lp_t = solve_lp_min_peak(f, n)
        if g is None:
            g = np.ones(n) / n
        else:
            gs = g.sum()
            if gs < 1e-15:
                g = np.ones(n) / n
            else:
                g = g / gs

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        best_alpha = 0.0
        best_val = M

        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.08)
            hi = min(1.0, best_alpha + 0.08)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-12:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                f_new = np.ones(n) / n
            else:
                f_new = f_new / sn
            f = f_new
            M = best_val

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 10 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 50:
            rng = np.random.default_rng((it * 7919 + 13) % (2**31))
            scale = 0.002 + 0.002 * min(no_improve // 50, 3)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit, verbose=False):
    """Coordinate descent targeting peak-contributing coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 25:
        top_k = min(4, len(conv))
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]

        improved = False
        for pk in peak_positions:
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)

            for j in range(js, je + 1):
                if time.monotonic() - t0 > time_limit:
                    break
                old_val = f[j]
                if old_val < 1e-12:
                    continue

                best_j_val = old_val
                best_j_obj = best_obj
                multipliers = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]

                for mult in multipliers:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.5:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial = f_trial / f_trial.sum()

                    val = fast_convolve(f_trial).max()
                    if val < best_j_obj - 1e-14:
                        best_j_obj = val
                        best_j_val = new_val

                if best_j_val != old_val:
                    denom = 1.0 - old_val
                    if denom > 1e-10:
                        scale = (1.0 - best_j_val) / denom
                        f[j] = best_j_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f[mask] *= scale
                        f = f / f.sum()

                    if best_j_obj < best_obj - 1e-14:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True
                        conv = fast_convolve(f)

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            rng = np.random.default_rng(int(time.monotonic() * 1000000) % 999983)
            for _ in range(15):
                j = int(rng.integers(0, n))
                old_val = f[j]
                if old_val < 1e-10:
                    continue
                for mult in [0.6, 0.8, 1.2, 1.4]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.5:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial = f_trial / f_trial.sum()
                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-14:
                        f = f_trial
                        best_obj = val
                        best_f = f.copy()
                        conv = fast_convolve(f)
                        improved = True
                        break
                if improved:
                    break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def cosine_window(n):
    """Create a raised-cosine window of length n, normalized to sum=1."""
    x = np.linspace(0, np.pi, n)
    w = np.cos(x / 2) ** 2
    w /= w.sum()
    return w


def gaussian_window(n, sigma_frac=0.3):
    """Create a Gaussian window of length n, normalized to sum=1."""
    x = np.arange(n) - (n - 1) / 2.0
    sigma = sigma_frac * n
    w = np.exp(-x**2 / (2.0 * sigma**2))
    w /= w.sum()
    return w


def triangular_window(n):
    """Create a triangular (Bartlett) window."""
    if n % 2 == 0:
        w = np.concatenate([np.arange(1, n // 2 + 1), np.arange(n // 2, 0, -1)])
    else:
        w = np.concatenate([np.arange(1, n // 2 + 1), [n // 2 + 1], np.arange(n // 2, 0, -1)])
    w = w.astype(np.float64)
    w /= w.sum()
    return w


def evaluate_and_update(a, best_seq, best_score, label):
    """Evaluate a candidate and return (best_seq, best_score, score)."""
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        return best_seq, best_score, float('inf')
    a_norm = a / a.sum()
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score, score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: smooth gradient descent (log-sum-exp),
    LP-Frank-Wolfe, multi-length exploration, coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        best_init = None
        best_init_score = float('inf')
        for n_init in [256, 384, 512, 640, 768, 896, 1024]:
            for init_func in [cosine_window, gaussian_window, triangular_window]:
                a_init = init_func(n_init)
                s_val = float(evaluate_sequence(a_init.tolist()))
                if s_val < best_init_score:
                    best_init_score = s_val
                    best_init = a_init
            a_init = np.ones(n_init) / n_init
            s_val = float(evaluate_sequence(a_init.tolist()))
            if s_val < best_init_score:
                best_init_score = s_val
                best_init = a_init

        # Also try random initializations
        for n_init in [512, 768, 1024]:
            for _ in range(3):
                a_init = rng.dirichlet(np.ones(n_init))
                s_val = float(evaluate_sequence(a_init.tolist()))
                if s_val < best_init_score:
                    best_init_score = s_val
                    best_init = a_init

        parent = best_init
        best_seq = parent.copy()
        best_score = best_init_score
        print(f"Best init: n={len(parent)}, score={best_score:.8f}")

    base_n = len(parent)
    print(f"Starting optimization: n={base_n}, score={best_score:.8f}")

    # === Phase 1: Smooth gradient descent (main workhorse) ===
    print(f"\n=== Phase 1: Smooth GD (n={base_n}) ===")
    rem = deadline - time.monotonic()
    t1 = min(400, rem * 0.28)
    if t1 > 30:
        print(f"  Time: {t1:.0f}s, n={len(best_seq)}")
        a_opt, _ = smooth_gradient_descent(
            best_seq, time_limit=t1,
            tau_start=0.005, tau_end=0.0001,
            lr_start=0.005, momentum=0.9,
            top_k=max(50, len(best_seq) // 4),
            restart_interval=300, verbose=True
        )
        best_seq, best_score, _ = evaluate_and_update(a_opt, best_seq, best_score, "sgd_p1")

    # === Phase 2: LP-Frank-Wolfe ===
    print(f"\n=== Phase 2: LP-FW (n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    t2 = min(350, rem * 0.25)
    if t2 > 30:
        print(f"  Time: {t2:.0f}s, n={len(best_seq)}")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t2, verbose=True)
        best_seq, best_score, _ = evaluate_and_update(a_opt, best_seq, best_score, "lpfw_p2")

    print(f"\n=== Phase 3: Refined Smooth GD ===")
    rem = deadline - time.monotonic()
    t3 = min(250, rem * 0.2)

        print(f"  Time: {t3:.0f}s, n={len(best_seq)}")
        a_opt, _ = smooth_gradient_descent(

            lr_start=0.002, momentum=0.92,
            top_k=max(30, len(best_seq) // 6),
            restart_interval=200, verbose=True
        )
        best_seq, best_score, _ = evaluate_and_update(a_opt, best_seq, best_score, "sgd_p3")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length exploration ===")
    base = best_seq if best_seq is not None else parent
    candidate_lengths = set()
    for delta in [-128, -64, -32, 32, 64, 128, 256, 384]:
        ln = len(base) + delta
        if ln >= 128 and ln <= 2048:
            candidate_lengths.add(ln)
    candidate_lengths = sorted(candidate_lengths)

    for target_n in candidate_lengths[:6]:

    print(f"\n=== Phase 5: LP-FW polish ===")

    # === Phase 6: Coordinate descent ===

    rem = deadline - time.monotonic()
    t6 = min(120, rem * 0.25)
    if t6 > 15 and best_seq is not None:
        print(f"  Time: {t6:.0f}s, n={len(best_seq)}")

    # === Phase 7: Final perturbations with short optimization ===
    print(f"\n=== Phase 7: Perturbations ===")
    if best_seq is not None:
        for p in range(10):
            rem = deadline - time.monotonic()
            if rem < 12:

            noise_scale = 0.0003 + 0.0004 * p

            a_opt, _ = smooth_gradient_descent(

                tau_start=0.0005, tau_end=0.00005,
                lr_start=0.001, momentum=0.93,
                top_k=max(20, len(best_seq) // 8)
            )
            best_seq, best_score, _ = evaluate_and_update(a_opt, best_seq, best_score, f"pert_{p}")

    # === Phase 8: Very final LP-FW if time remains ===
    rem = deadline - time.monotonic()
    if rem > 30 and best_seq is not None:
        print(f"\n=== Phase 8: Final LP-FW ({rem:.0f}s remaining) ===")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem - 10, 100), verbose=True)
        best_seq, best_score, _ = evaluate_and_update(a_opt, best_seq, best_score, "final_lpfw")

    # Fallback

    else:

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
