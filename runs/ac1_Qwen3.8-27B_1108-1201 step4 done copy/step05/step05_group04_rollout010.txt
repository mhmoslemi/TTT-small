"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multiplicative subgradient (mirror descent on simplex) with adaptive active set
2. Euclidean subgradient with Nesterov momentum and adaptive step
3. LSE (log-sum-exp) high-beta gradient polishing

Key novelty: mirror descent (multiplicative weights) updates which handle the

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgrad(f, active_lags):
    """Compute subgradient of max over active_lags of b[k] w.r.t. f[j]."""
    n = len(f)
    grad = np.zeros(n)
    for k in active_lags:
        k = int(k)
        # Contribute a[k-j] for valid j: max(0, k-n+1) <= j <= min(n-1, k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            js = np.arange(j_lo, j_hi + 1)
            grad[js] += f[k - js]
        # Contribute a[j-k] for valid j: max(0, k) <= j <= min(n-1, k+n-1)
        j2_lo = max(0, k)
        j2_hi = min(n - 1, k + n - 1)
        if j2_lo <= j2_hi:
            js = np.arange(j2_lo, j2_hi + 1)
            grad[js] += f[js - k]
    grad /= max(1, len(active_lags))
    return grad


def multiplicative_subgrad(f0, time_limit, lr_init=0.05, tol_init=1e-8,
                           max_active=None, verbose=False):
    """Mirror descent (multiplicative weights) subgradient method.
    
    Uses adaptive active set based on tolerance around the max.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = lr_init
    tol = tol_init
    momentum_v = np.zeros(n)
    momentum = 0.7

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        max_b = conv.max()

        # Adaptive active set
        if max_active is None:
            n_active = max(5, n // 8)
        else:
            n_active = max_active
        active_lags = np.argpartition(conv, -n_active)[-n_active:]
        active_lags = active_lags[conv[active_lags] >= max_b - tol * max_b]
        if len(active_lags) == 0:
            active_lags = np.array([np.argmax(conv)])

        grad = compute_subgrad(f, active_lags)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            no_improve += 100
            break
        grad /= gnorm

        # Nesterov-like momentum in multiplicative domain
        momentum_v = momentum * momentum_v + grad
        effective_grad = momentum_v

        # Multiplicative update with clipping
        scaled = lr * effective_grad
        scaled = np.clip(scaled, -5.0, 5.0)
        f_new = f * np.exp(-scaled)
        s_new = f_new.sum()
        if s_new < 1e-15:
            f_new = f.copy()
        else:
            f_new /= s_new

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.01, 0.2)
            tol = max(tol * 0.99, 1e-10)
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.95
            if no_improve % 60 == 0:
                tol *= 1.05
            if no_improve > 200:
                rng = np.random.default_rng((it * 7919 + 13) % 999983)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                momentum_v = np.zeros(n)
                lr = max(lr_init * 0.5, 0.005)
                tol = tol_init
                no_improve = 0
                if verbose:
                    print(f"    MultSGD restart it={it}, obj={best_obj:.10f}")

    return best_f, best_obj

def lse_optimize(f0, time_limit, beta_start=100.0, beta_end=20000.0,
                 max_peaks=300, verbose=False):
    """Log-sum-exp smoothed gradient descent."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    beta = beta_start
    lr = 1.5 / n
    momentum = 0.90
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        # Weighted gradient using top peaks
        sig_idx = np.argpartition(w, -max_peaks)[-max_peaks:]
        sig_idx = sig_idx[w[sig_idx] > 1e-12 * w.max()]

        grad = np.zeros(n)
        for k in sig_idx:
            wk = w[k]
            if wk < 1e-15:
                continue
            k = int(k)
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                js = np.arange(j_lo, j_hi + 1)
                grad[js] += wk * f[k - js]
            j2_lo = max(0, k)
            j2_hi = min(n - 1, k + n - 1)
            if j2_lo <= j2_hi:
                js = np.arange(j2_lo, j2_hi + 1)
                grad[js] += wk * f[js - k]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            no_improve += 200
            break
        grad /= gnorm

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.003, 4.0 / n)
        else:
            no_improve += 1
            if no_improve >= 10:
                lr *= 0.94
                momentum = max(0.84, momentum - 0.003)
            if no_improve > 80:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(1.5 / n, 0.0005)
                beta = min(beta * 1.5, beta_end)
                momentum = 0.88
                no_improve = 0

        if beta < beta_end:
            beta *= 1.006

    return best_f, best_obj


def peak_shave(f0, time_limit, n_shaves=20, verbose=False):
    """Direct peak shaving: iteratively reduce the most contributing coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit and no_improve < 100:
        it += 1
        conv = fast_convolve(f)
        max_b = conv.max()
        peak_k = int(np.argmax(conv))

        # Find the coordinate j that most contributes to b[peak_k]
        j_lo = max(0, peak_k - n + 1)
        j_hi = min(n - 1, peak_k)
        contributions = np.zeros(n)
        if j_lo <= j_hi:
            js = np.arange(j_lo, j_hi + 1)
            contributions[js] = f[js] * f[peak_k - js]
        j2_lo = max(0, peak_k)
        j2_hi = min(n - 1, peak_k + n - 1)
        if j2_lo <= j2_hi:
            js = np.arange(j2_lo, j2_hi + 1)
            contributions[js] += f[js] * f[js - peak_k]

        top_j = int(np.argmax(contributions))
        if contributions[top_j] < 1e-15 * max_b:
            no_improve += 10
            continue

        # Reduce f[top_j] and redistribute
        reduce_amt = f[top_j] * 0.02
        f[top_j] -= reduce_amt
        f[top_j] = max(f[top_j], 0.0)
        # Redistribute to coordinates with low contribution
        low_mask = contributions < np.median(contributions)
        low_idx = np.where(low_mask)[0]
        if len(low_idx) == 0:
            low_idx = np.arange(n)
        redist = reduce_amt / len(low_idx)
        f[low_idx] += redist
        f = np.maximum(f, 0)
        f /= f.sum()

        conv_new = fast_convolve(f)
        obj_new = conv_new.max()
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                f = best_f.copy()

    return best_f, best_obj


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def propose_candidate(seed=42, budget_s=1800):
    """
    Main multi-phase optimization combining mirror descent, Euclidean subgradient,
    LSE polishing, peak shaving, and multi-start exploration.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_tracker = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_sequence(parent, best_tracker)
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        try_sequence(parent, best_tracker)

    n_parent = len(parent)
    current = best_tracker[1].copy() if best_tracker[1] is not None else parent.copy()
    n_cur = len(current)

    # === Phase 1: Multiplicative subgradient (mirror descent) ===
    print("\n=== Phase 1: Multiplicative Subgradient (Mirror Descent) ===")
    t1_end = deadline - 600
    t_alloc = min(200, t1_end - time.monotonic())
    if t_alloc > 30:
        print(f"  MultSGD n={n_cur}, t={t_alloc:.0f}s, lr=0.08")
        f_opt, obj = multiplicative_subgrad(current, time_limit=t_alloc,
                                             lr_init=0.08, tol_init=1e-6,
                                             max_active=n_cur // 6, verbose=True)
        try_sequence(f_opt, best_tracker)

    current = best_tracker[1].copy() if best_tracker[1] is not None else current
    t_alloc = min(180, t1_end - time.monotonic())
    if t_alloc > 30:
        print(f"  MultSGD round2 n={n_cur}, t={t_alloc:.0f}s, lr=0.04")
        f_opt, obj = multiplicative_subgrad(current, time_limit=t_alloc,
                                             lr_init=0.04, tol_init=1e-7,
                                             max_active=n_cur // 10, verbose=True)
        try_sequence(f_opt, best_tracker)

    # === Phase 2: Euclidean subgradient with Nesterov ===
    print(f"\n=== Phase 2: Euclidean Subgradient ===")
    t2_end = deadline - 400
    current = best_tracker[1].copy() if best_tracker[1] is not None else current

    t_alloc = min(150, t2_end - time.monotonic())
    if t_alloc > 30:
        print(f"  EuclSGD n={n_cur}, t={t_alloc:.0f}s, n_peaks=80")
        f_opt, _ = euclidean_subgrad(current, time_limit=t_alloc,
                                      lr_init=0.004, n_peaks=80, momentum=0.93,
                                      verbose=True)
        try_sequence(f_opt, best_tracker)

    current = best_tracker[1].copy() if best_tracker[1] is not None else current
    t_alloc = min(120, t2_end - time.monotonic())
    if t_alloc > 20:
        print(f"  EuclSGD fine n={n_cur}, t={t_alloc:.0f}s, n_peaks=30")
        f_opt, _ = euclidean_subgrad(current, time_limit=t_alloc,
                                      lr_init=0.002, n_peaks=30, momentum=0.94,
                                      verbose=True)
        try_sequence(f_opt, best_tracker)

    # === Phase 3: LSE high-beta polishing ===
    print(f"\n=== Phase 3: LSE High-Beta Polish ===")
    t3_end = deadline - 250
    current = best_tracker[1].copy() if best_tracker[1] is not None else current

    t_alloc = min(150, t3_end - time.monotonic())
    if t_alloc > 30:
        print(f"  LSE n={n_cur}, t={t_alloc:.0f}s, beta=[50, 15000]")
        f_opt, _ = lse_optimize(current, time_limit=t_alloc,
                                 beta_start=50.0, beta_end=15000.0, max_peaks=300)
        try_sequence(f_opt, best_tracker)

    current = best_tracker[1].copy() if best_tracker[1] is not None else current
    t_alloc = min(100, t3_end - time.monotonic())
    if t_alloc > 20:
        print(f"  LSE ultra-high beta n={n_cur}, t={t_alloc:.0f}s")
        f_opt, _ = lse_optimize(current, time_limit=t_alloc,
                                 beta_start=2000.0, beta_end=50000.0, max_peaks=200)
        try_sequence(f_opt, best_tracker)

    # === Phase 4: Peak shaving + alternating refinement ===
    print(f"\n=== Phase 4: Peak Shaving + Alternation ===")
    t4_end = deadline - 150
    cycles = 0
    while time.monotonic() < t4_end and cycles < 6:
        cycles += 1
        current = best_tracker[1].copy() if best_tracker[1] is not None else current
        remaining = t4_end - time.monotonic()
        if remaining < 30:
            break

        if cycles % 3 == 1:
            t_alloc = min(50, remaining * 0.4)
            if t_alloc > 15:
                print(f"  Cycle {cycles}: PeakShave t={t_alloc:.0f}s")
                f_opt, _ = peak_shave(current, time_limit=t_alloc)
                try_sequence(f_opt, best_tracker)
        elif cycles % 3 == 2:
            t_alloc = min(50, remaining * 0.4)
            if t_alloc > 15:
                print(f"  Cycle {cycles}: MultSGD t={t_alloc:.0f}s")
                f_opt, _ = multiplicative_subgrad(current, time_limit=t_alloc,
                                                   lr_init=0.02, tol_init=1e-7)
                try_sequence(f_opt, best_tracker)
        else:
            t_alloc = min(50, remaining * 0.4)
            if t_alloc > 15:
                print(f"  Cycle {cycles}: EuclSGD t={t_alloc:.0f}s")
                f_opt, _ = euclidean_subgrad(current, time_limit=t_alloc,
                                              lr_init=0.001, n_peaks=20, momentum=0.95)
                try_sequence(f_opt, best_tracker)

    # === Phase 5: Multi-start exploration ===
    print(f"\n=== Phase 5: Multi-Start ===")
    t5_end = deadline - 80
    best_f = best_tracker[1].copy() if best_tracker[1] is not None else current
    n_best = len(best_f)
    rng = np.random.default_rng(seed)

    n_starts = 4
    for i in range(n_starts):
        t_remaining = t5_end - time.monotonic()
        if t_remaining < 40:
            break
        t_alloc = min(60, t_remaining * 0.4)
        if t_alloc < 20:
            break
        # Different perturbation strategies
        if i == 0:
            pert = rng.standard_normal(n_best) * 0.003
        elif i == 1:
            # Asymmetric perturbation (shift emphasis)
            x = np.arange(n_best)
            shift = 0.3
            pert = np.exp(-((x - n_best * shift) ** 2) / (2 * (n_best * 0.1) ** 2)) * 0.004
            pert -= pert.mean()
        elif i == 2:
            # High-frequency noise
            freq = rng.integers(3, 10)
            x = np.arange(n_best)
            pert = np.cos(2 * np.pi * freq * x / n_best) * 0.002
            pert -= pert.mean()
        else:
            # Random sparse perturbation
            pert = rng.standard_normal(n_best) * 0.002
            mask = rng.random(n_best) < 0.3
            pert[~mask] = 0
            pert -= pert.mean()

        f_pert = project_simplex(best_f + pert)
        print(f"  Start {i+1}: perturbed, running LSE t={t_alloc:.0f}s")
        f_opt, _ = lse_optimize(f_pert, time_limit=t_alloc,
                                 beta_start=50.0, beta_end=10000.0, max_peaks=200)
        try_sequence(f_opt, best_tracker)

    # === Phase 6: Nearby length exploration ===
    print(f"\n=== Phase 6: Nearby Lengths ===")
    if best_tracker[1] is not None:
        n_best = len(best_tracker[1])
        offsets = [32, -32, 64, -64, 16, -16, 128, -128]
        for off in offsets:
            n_try = n_best + off
            if n_try < 256 or n_try > 2048:
                continue
            t_remaining = deadline - time.monotonic() - 20
            if t_remaining < 30:
                break
            t_alloc = min(50, t_remaining * 0.35)
            if t_alloc < 15:
                break
            f_init = interpolate_to_length(best_tracker[1], n_try)
            print(f"  n={n_try}, t={t_alloc:.0f}s")
            f_opt, _ = lse_optimize(f_init, time_limit=t_alloc,
                                     beta_start=30.0, beta_end=8000.0, max_peaks=200)
            try_sequence(f_opt, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    print("WARNING: Using uniform fallback")
    fallback = np.ones(768) / 768
    return [float(x) for x in fallback]
