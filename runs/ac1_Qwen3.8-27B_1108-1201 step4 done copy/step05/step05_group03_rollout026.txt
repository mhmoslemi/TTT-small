from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_obj_grad(f, T, top_k):
    """Compute smoothed objective and gradient."""
    n = len(f)
    conv = fast_convolve(f)
    # Select top_k peaks
    k_actual = min(top_k, 2 * n - 1)
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = T * (pv - pv.max())
    w = np.exp(scaled)
    ws = w.sum()
    w = w / ws
    
    obj = pv.max() + np.log(ws) / T
    # Actually, the log-sum-exp is: max(pv) + log(sum(exp(T*(pv - max(pv))))) / T
    obj = pv.max() + np.log(ws) / T
    
    # Gradient
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    
    return obj, sg, conv

    best_f = f.copy()

    t = 1.0

        lr = lr0 * (lr1 / lr0) ** progress
        
        obj, sg, conv = smooth_obj_grad(y, T, top_k)

import time

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f, peaks, n):
    K = len(peaks)
    A = np.zeros((K, n))
    for row, k in enumerate(peaks):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            A[row, js:je + 1] = f[k - np.arange(js, je + 1)]
    return A


def lp_direction_topk(f, M, n, top_k):

    c = -np.ones(n)
    A_ub = build_lp_matrix(f, peaks, n)
    b_ub = np.full(len(peaks), M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

                val = fast_convolve(f).max()
                if val < best_obj - 1e-13:
                    best_obj = val
                    best_f = f.copy()
                    improved = True
                else:
                    # Revert
                    f[i] = current_ratio * c
                    f[j] = (1.0 - current_ratio) * c

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_initializations(parent, seed=0):
    """Generate diverse initializations."""
    inits = []
    rng = np.random.default_rng(seed)

    # Parent
    inits.append(("parent", parent.copy()))

    # Uniform at parent length
    n = len(parent)
    inits.append(("uniform", np.ones(n) / n))

    # Reversed parent
    inits.append(("reversed", parent[::-1].copy() / parent.sum()))

    # Symmetric average
    sym = 0.5 * (parent + parent[::-1])
    sym = sym / sym.sum()
    inits.append(("symmetric", sym))

    # Noisy parent
    noisy = parent + rng.standard_normal(n) * 0.01
    inits.append(("noisy", project_simplex(noisy)))

    # Triangular
    tri = np.abs(np.arange(n) - n // 2)
    tri = 1.0 / (tri + 1.0)
    inits.append(("triangular", tri / tri.sum()))

    # Bimodal
    bi = np.exp(-0.5 * ((np.arange(n) - n * 0.25) / (n * 0.1))**2) + \
         np.exp(-0.5 * ((np.arange(n) - n * 0.75) / (n * 0.1))**2)
    inits.append(("bimodal", bi / bi.sum()))

    # Random
    rand = rng.dirichlet(np.ones(n))
    inits.append(("random", rand))

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return float('inf')
        try:
            sc = float(evaluate_sequence(a.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
        return sc

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = try_seq(parent, "parent_init")
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
    except (NameError, Exception):
        pass

    if parent is None:
        parent = np.ones(512) / 512
        try_seq(parent, "default_uniform")
        print("Using default uniform parent")

    # === Phase 1: Nesterov on parent ===
    print("\n=== Phase 1: Nesterov smoothed SG on parent ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.2, 200)
    if t1 > 30:
        print(f"  Time budget: {t1:.0f}s, n={len(parent)}")
        a_opt, _ = nesterov_smooth_opt(parent, time_limit=t1, verbose=True)
        try_seq(a_opt, "nesterov_p1")

    # === Phase 2: LP-FW on best ===
    print(f"\n=== Phase 2: Top-K LP-FW (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t2 = min(rem * 0.2, 250)
        if t2 > 30:
            n_best = len(best_seq)
            print(f"  Time budget: {t2:.0f}s, n={n_best}")
            a_opt, _ = topk_lpfw_opt(best_seq, time_limit=t2, top_k=min(200, 2 * n_best), verbose=True)
            try_seq(a_opt, "lpfw_p2")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent

    target_lengths = [384, 512, 768, 1024, 1536]
    # Remove current length
    target_lengths = [tl for tl in target_lengths if tl != len(base)]
    # Sort by proximity to current length
    target_lengths.sort(key=lambda x: abs(x - len(base)))

    for target_n in target_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break

        f_interp = interpolate_sequence(base, target_n)
        time_per_length = min(rem / max(1, 5 - target_lengths.index(target_n)) * 0.7, 180)

        # Nesterov
        t_alloc = time_per_length * 0.45
        if t_alloc > 20:
            print(f"  Nesterov n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = nesterov_smooth_opt(f_interp, time_limit=t_alloc,
                                           T_start=8.0, T_end=0.3,
                                           top_k_start=200, top_k_end=30)
            try_seq(a_opt, f"nesterov_n{target_n}")

        # LP-FW
        rem = deadline - time.monotonic()
        t_lpfw = min(rem * 0.4, 120)
        if t_lpfw > 20:
            start = a_opt if 'a_opt' in dir() else f_interp
            print(f"  LP-FW n={target_n}, t={t_lpfw:.0f}s")
            a_opt, _ = topk_lpfw_opt(start, time_limit=t_lpfw,
                                     top_k=min(150, 2 * target_n))
            try_seq(a_opt, f"lpfw_n{target_n}")

    # === Phase 4: Diverse restarts with ILS ===
    print(f"\n=== Phase 4: ILS with diverse starts (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        num_restarts = 6
        rem = deadline - time.monotonic()
        time_per_restart = rem * 0.35 / num_restarts

        for r in range(num_restarts):
            rem = deadline - time.monotonic()
            if rem < 15:
                break

            rng = np.random.default_rng(1000 + r * 37)

            if r == 0:
                f_init = best_seq.copy()
            elif r == 1:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.005)
            elif r == 2:
                f_init = best_seq[::-1].copy() / best_seq.sum()
            elif r == 3:
                f_init = 0.6 * best_seq + 0.4 * (np.ones(n_best) / n_best)
                f_init = project_simplex(f_init)
            elif r == 4:
                # Truncate small values and renormalize
                f_init = best_seq.copy()
                threshold = np.percentile(f_init[f_init > 0], 20) if (f_init > 0).any() else 0
                f_init[f_init < threshold] = 0
                f_init = project_simplex(f_init)
            else:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.015)

            t_alloc = min(time_per_restart, 120)

            # Alternate optimization methods
            if r % 3 == 0:
                a_opt, _ = nesterov_smooth_opt(f_init, time_limit=t_alloc * 0.6,
                                               T_start=6.0, T_end=0.2,
                                               top_k_start=150, top_k_end=20)
                rem = deadline - time.monotonic()
                if rem > 10:
                    a_opt, _ = topk_lpfw_opt(a_opt, time_limit=min(rem - 5, t_alloc * 0.4),
                                             top_k=min(100, 2 * n_best))
            elif r % 3 == 1:
                a_opt, _ = topk_lpfw_opt(f_init, time_limit=t_alloc * 0.7,
                                         top_k=min(150, 2 * n_best))
                rem = deadline - time.monotonic()
                if rem > 10:
                    a_opt, _ = nesterov_smooth_opt(a_opt, time_limit=min(rem - 5, t_alloc * 0.3),
                                                   T_start=3.0, T_end=0.15,
                                                   top_k_start=100, top_k_end=15)
            else:
                a_opt, _ = nesterov_smooth_opt(f_init, time_limit=t_alloc * 0.5,
                                               T_start=10.0, T_end=0.5,
                                               top_k_start=300, top_k_end=50)
                rem = deadline - time.monotonic()
                if rem > 10:
                    a_opt, _ = topk_lpfw_opt(a_opt, time_limit=min(rem - 5, t_alloc * 0.5),
                                             top_k=min(120, 2 * n_best))

            try_seq(a_opt, f"ils_{r}")

    # === Phase 5: Pairwise polish ===
    print(f"\n=== Phase 5: Pairwise polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            t_pw = min(rem * 0.3, 150)
            print(f"  Time: {t_pw:.0f}s, n={len(best_seq)}")
            a_opt, _ = pairwise_polish(best_seq, time_limit=t_pw)
            try_seq(a_opt, "pairwise")

    # === Phase 6: Coordinate descent ===
    print(f"\n=== Phase 6: Coord descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 20:
            t_cd = min(rem * 0.25, 100)
            print(f"  Time: {t_cd:.0f}s")
            a_opt, _ = coord_descent_polish(best_seq, time_limit=t_cd)
            try_seq(a_opt, "coord_desc")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        # Nesterov with very low temperature
        rem = deadline - time.monotonic()
        if rem > 15:
            t_final = min(rem * 0.3, 80)
            print(f"  Final Nesterov: {t_final:.0f}s")
            a_opt, _ = nesterov_smooth_opt(best_seq, time_limit=t_final,
                                           T_start=2.0, T_end=0.05,
                                           top_k_start=50, top_k_end=5,
                                           lr_start=0.003, lr_end=0.0005)
            try_seq(a_opt, "nesterov_final")

        # LP-FW final
        rem = deadline - time.monotonic()
        if rem > 15:
            t_lpfw = min(rem * 0.3, 60)
            print(f"  Final LP-FW: {t_lpfw:.0f}s")
            a_opt, _ = topk_lpfw_opt(best_seq, time_limit=t_lpfw, top_k=80)
            try_seq(a_opt, "lpfw_final")

        # Tiny perturbations + quick re-optimize
        for trial in range(5):
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            rng = np.random.default_rng(2000 + trial * 13)
            f_pert = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * 0.002)
            t_quick = min(rem * 0.4, 25)
            a_opt, _ = nesterov_smooth_opt(f_pert, time_limit=t_quick,
                                           T_start=1.0, T_end=0.05,
                                           top_k_start=40, top_k_end=5,
                                           lr_start=0.002, lr_end=0.0003)
            try_seq(a_opt, f"perturb_{trial}")

        # One final pairwise pass
        rem = deadline - time.monotonic()
        if rem > 10:
            a_opt, _ = pairwise_polish(best_seq, time_limit=min(rem, 30))
            try_seq(a_opt, "pw_final")

        # Final coordinate descent
        rem = deadline - time.monotonic()
        if rem > 8:
            a_opt, _ = coord_descent_polish(best_seq, time_limit=min(rem, 20))
            try_seq(a_opt, "cd_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
