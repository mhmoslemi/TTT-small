    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
1. Log-sum-exp smoothed gradient descent with adaptive temperature (main)

3. LP-based Frank-Wolfe (final polish)

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n <= 1:
        return np.array([max(v[0] if n == 1 else 0.0, 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_gradient(a, weights, fft_size, size, n):
    """Compute gradient of sum_k w[k]*conv_k using FFT.
    grad[j] = 2 * sum_k w[k] * a[k-j] = 2 * convolve(w, a_flipped)[j+n-1]
    """
    w_padded = np.zeros(fft_size)
    w_padded[:size] = weights
    a_flip = a[::-1]
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a_flip
    W = np.fft.rfft(w_padded)
    Af = np.fft.rfft(a_padded)
    grad_full = np.fft.irfft(W * Af, fft_size)
    return 2.0 * grad_full[n - 1:2 * n - 1]


def lse_gradient_descent(a0, time_limit, verbose=False, tau_init=0.003, lr_init=0.5,
                         tau_min=0.0003, tau_decay=0.9997, lr_decay=0.99995):
    """Log-sum-exp smoothed gradient descent on the simplex with adaptive params."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    tau = tau_init
    lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        c_max = conv.max()

        scaled = (conv - c_max) / tau
        weights = np.exp(scaled)
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights /= wsum

        grad = compute_lse_gradient(a, weights, fft_size, size, n)

        # Line search with backtracking
        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq < 1e-30:
            no_improve += 1
            tau = max(tau * 0.95, tau_min)
            continue

        best_alpha = 0.0
        best_val = c_max

        for _ in range(20):
            a_trial = a - alpha * grad
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-15:
                alpha *= 0.5
                continue
            a_trial = a_trial / st
            val = fast_convolve(a_trial).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
            # Armijo condition (simplified)

                break
            alpha *= 0.5
            if alpha < 1e-8:
                break

            a = a - best_alpha * grad
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            if best_val < best_obj - 1e-15:
                best_obj = best_val
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        lr = max(lr * lr_decay, 0.01)

    return best_a, best_obj


def spectral_flatten(a0, n_iter=2000, gamma=0.6, verbose=False):
    """Iterative spectral flattening with non-negativity projection."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    for it in range(n_iter):
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)

        mag_new = mag ** gamma
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s_new

        conv = fast_convolve(a_new)
        obj = conv.max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        a = a_new

    return best_a, best_obj


def solve_lp_min_peak(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def quadratic_residue_init(p):
    """Create a sequence based on quadratic residues mod p."""
    a = np.zeros(p)
    for x in range(1, p):
        if (x * x) % p == 1 or ((x * x) % p) % 2 == 0:
            a[x % p] += 1.0
    # Use the Legendre symbol pattern
    a2 = np.zeros(p)
    for x in range(1, p):
        ls = 1 if pow(x, (p - 1) // 2, p) == 1 else 0
        a2[x % p] = 1.0 + 0.5 * ls
    a2[0] = 1.0
    s = a2.sum()
    if s > 0:
        a2 /= s
    return a2
