import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def objective(a):
    """Compute the objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    conv = fast_convolve(a_norm)
    return 2.0 * n * conv.max()

def subgradient_step(a, step_size, top_k=10):
    """One subgradient step for minimizing max(conv(a,a)) on the simplex."""
    n = len(a)
    conv = fast_convolve(a)
    # Find top k peaks
    k = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k)[-k:]
    peak_vals = conv[peaks]
    # Weight by how much they exceed the mean
    mean_peak = peak_vals.mean()
    w = np.maximum(peak_vals - mean_peak, 0)
    ws = w.sum()
    if ws < 1e-15:
        w = np.ones(k) / k
    else:
        w = w / ws
    
    # Compute subgradient
    sg = np.zeros(n)
    for i in range(k):
        if w[i] < 1e-14:
            continue
        kpos = int(peaks[i])
        js = max(0, kpos - n + 1)
        je = min(n - 1, kpos)
        idx = np.arange(js, je + 1)
        sg[idx] += w[i] * a[kpos - idx]
        # Handle the diagonal term (when kpos - idx == idx, i.e., kpos = 2*idx)
        diag = kpos // 2
        if kpos % 2 == 0 and js <= diag <= je:
            sg[diag] += w[i] * a[diag]
    
    return sg

def nesterov_subgrad(a0, time_limit, verbose=False):

    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Palindromic-constrained optimization for dimensionality reduction
2. Adam optimizer with log-sum-exp smoothing and adaptive scheduling
3. Nesterov-accelerated subgradient on the non-smooth max objective
4. LP-based Frank-Wolfe with golden-section line search
5. Exact pairwise coordinate descent with golden-section 1D search
6. Multi-length exploration with interpolation
7. Structured diverse restarts and final polish
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_palin(first_half, n):
    """Create a palindromic sequence of length n from its first half."""
    if n % 2 == 0:
        m = n // 2
        fh = first_half[:m]
        return np.concatenate([fh, fh[::-1]])
    else:
        m = (n - 1) // 2
        fh = first_half[:m + 1]
        return np.concatenate([fh, fh[-2::-1]])


def extract_first_half(a, n):
    """Extract the first half of a palindromic-like sequence."""
    if n % 2 == 0:
        return a[:n // 2].copy()
    else:
        return a[:(n + 1) // 2].copy()


def smooth_objective_and_grad(a, T, top_k):
    """Compute smoothed objective and gradient using log-sum-exp on top peaks."""
    n = len(a)
    conv = fast_convolve(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    max_pv = pv.max()
    scaled = T * (pv - max_pv)
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w = w / ws

    obj = max_pv + T * np.log(np.sum(np.exp(scaled)))

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        kpos = int(peaks[i])
        js = max(0, kpos - n + 1)
        je = min(n - 1, kpos)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * a[kpos - idx]
        if kpos % 2 == 0:
            diag = kpos // 2
            if js <= diag <= je:
                sg[diag] += 2.0 * w[i] * a[diag]

    return obj, sg / (T + 1e-30)


def adam_sg_optimize(f0, time_limit, verbose=False, T_scale=1.0, lr_scale=1.0):
    """Adam optimizer on the log-sum-exp smoothed objective with simplex projection."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.005 * lr_scale

    T_start = 5.0 / n * T_scale
    T_end = 0.3 / n * T_scale
    top_k_start = max(50, n)
    top_k_end = max(8, n // 10)

    it = 0
    no_improve = 0
    max_iters = int(time_limit / 0.01)

    while no_improve < 800:
        if time.monotonic() - t0 > time_limit:
            break
        it += 1

        progress = min(1.0, it / (max_iters * 0.85))
        T = T_start * (T_end / T_start) ** progress
        top_k = max(5, int(top_k_start + (top_k_end - top_k_start) * progress))
        lr = base_lr * (1.0 - 0.4 * progress) * (1.0 + 0.5 * np.sin(it * 0.01))

        obj, sg = smooth_objective_and_grad(f, T, top_k)

        it1 = it + 1
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** it1)
        v_hat = v / (1 - beta2 ** it1)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        f_new = project_simplex(f - step)
        f = f_new

        new_M = fast_convolve(f).max()
        if new_M < best_obj - 1e-13:
            best_obj = new_M
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    Adam iter {it}: obj={best_obj:.10f}, T={T:.6f}, lr={lr:.6f}")
        else:
            no_improve += 1

        if no_improve in [300, 500, 700]:
            rng = np.random.default_rng(it * 41 + 13 + no_improve)
            scale = 0.003 + 0.002 * (no_improve // 200)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0

    return best_f, best_obj


def nesterov_subgrad_optimize(f0, time_limit, verbose=False):
    """Nesterov-accelerated subgradient method on the non-smooth max objective."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    y = f.copy()
    it = 0
    no_improve = 0
    top_k = min(30, 2 * n - 1)

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        it += 1

        if it % 1000 == 0 and verbose:
            print(f"    NSG iter {it}: obj={best_obj:.10f}")

        momentum = 2.0 / (2.0 + np.sqrt(it))
        step = 0.03 / (1.0 + 0.05 * np.sqrt(it))

        y = f + momentum * (f - y)

        conv_y = fast_convolve(y)
        k = min(top_k, len(conv_y))
        peaks = np.argpartition(conv_y, -k)[-k:]
        pv = conv_y[peaks]
        mean_pv = pv.mean()
        w = np.maximum(pv - mean_pv, 0)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(k) / k
        else:
            w = w / ws

        sg = np.zeros(n)
        for i in range(k):
            if w[i] < 1e-14:
                continue
            kpos = int(peaks[i])
            js = max(0, kpos - n + 1)
            je = min(n - 1, kpos)
            idx = np.arange(js, je + 1)
            sg[idx] += w[i] * y[kpos - idx]
            if kpos % 2 == 0:
                diag = kpos // 2
                if js <= diag <= je:
                    sg[diag] += w[i] * y[diag]

        f_new = y - step * sg
        f_new = project_simplex(f_new)
        f = f_new

        new_M = fast_convolve(f).max()
        if new_M < best_obj - 1e-12:
            best_obj = new_M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 400:
            rng = np.random.default_rng(it * 7 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            y = f.copy()
            no_improve = 0

    return best_f, best_obj


def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        rows = np.arange(j, j + n)
        A_ub[rows, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0)
    return None

            no_improve = 0

    return best_f, best_obj


def pairwise_exact_optimize(f0, time_limit, verbose=False):
    """Pairwise coordinate descent with exact golden-section 1D search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(424242)

    offsets = [1, 2, 3, 4, 5, 7, 8, 11, 13, 17, 21, 34, 55, 89, 144]
    no_improve_passes = 0
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        pass_improved = False

        if passes % 2 == 1:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for idx_i in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            i = order[idx_i]
            for offset in offsets:
                j = order[(idx_i + offset) % n]
                if i == j:
                    continue

                c = f[i] + f[j]
                if c < 1e-12:
                    continue

                # Golden section search on t in [0, 1]
                # f(t) = max(conv(a)) with a[i] = t*c, a[j] = (1-t)*c
                lo_t, hi_t = 0.0, 1.0
                gr = (np.sqrt(5) - 1) / 2

                cur_t = f[i] / c if c > 0 else 0.5
                # Evaluate at current point and boundaries
                vals = []
                for t_eval in [cur_t, 0.0, 1.0, 0.5, 0.25, 0.75]:
                    t_eval = np.clip(t_eval, 0.0, 1.0)
                    f_trial = f.copy()
                    f_trial[i] = t_eval * c
                    f_trial[j] = (1.0 - t_eval) * c
                    vals.append((fast_convolve(f_trial).max(), t_eval))

                best_val, best_t = min(vals)

                if best_val < best_obj - 1e-13:
                    # Refine with golden section around best_t
                    center = best_t
                    span = 0.15
                    lo_t = max(0.0, center - span)
                    hi_t = min(1.0, center + span)

                    c1 = hi_t - gr * (hi_t - lo_t)
                    c2 = lo_t + gr * (hi_t - lo_t)

                    f1 = f.copy()
                    f1[i] = c1 * c
                    f1[j] = (1.0 - c1) * c
                    v1 = fast_convolve(f1).max()

                    f2 = f.copy()
                    f2[i] = c2 * c
                    f2[j] = (1.0 - c2) * c
                    v2 = fast_convolve(f2).max()

                    for _ in range(10):
                        if hi_t - lo_t < 1e-8:
                            break
                        if v1 < v2:
                            hi_t = c2
                            c2 = c1
                            v2 = v1
                            c1 = hi_t - gr * (hi_t - lo_t)
                            f1 = f.copy()
                            f1[i] = c1 * c
                            f1[j] = (1.0 - c1) * c
                            v1 = fast_convolve(f1).max()
                        else:
                            lo_t = c1
                            c1 = c2
                            v1 = v2
                            c2 = lo_t + gr * (hi_t - lo_t)
                            f2 = f.copy()
                            f2[i] = c2 * c
                            f2[j] = (1.0 - c2) * c
                            v2 = fast_convolve(f2).max()

                    final_t = (lo_t + hi_t) / 2.0
                    f_trial = f.copy()
                    f_trial[i] = final_t * c
                    f_trial[j] = (1.0 - final_t) * c
                    final_val = fast_convolve(f_trial).max()

                    if final_val < best_val - 1e-14:
                        best_val = final_val
                        best_t = final_t

                    f[i] = best_t * c
                    f[j] = (1.0 - best_t) * c
                    best_obj = fast_convolve(f).max()
                    best_f = f.copy()
                    pass_improved = True

        if pass_improved:
            no_improve_passes = 0
        else:
            no_improve_passes += 1
            if no_improve_passes >= 3:
                break

        if verbose and passes % 5 == 0:
            print(f"    Pairwise pass {passes}: obj={best_obj:.10f}")

    return best_f, best_obj


def palindromic_optimize(f0, time_limit, verbose=False):
    """Optimize over the first half of a palindromic sequence."""
    n = len(f0)
    m = (n + 1) // 2  # number of free variables

    # Initialize first half from f0
    fh = extract_first_half(f0, n)
    fh = np.maximum(fh, 0)
    fh_sum = fh.sum()
    if fh_sum < 1e-15:
        fh = np.ones(m) / m
    else:
        fh = fh / fh_sum

    best_fh = fh.copy()
    best_obj = fast_convolve(make_palin(fh, n)).max()

    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    # Simple coordinate descent on the first half
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit and no_improve < 200:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        improved = False
        for i in range(m):
            if time.monotonic() - t0 > time_limit:
                break

            c = fh[i]
            if c < 1e-13:
                # Try to give it some mass
                for j in range(m):
                    if j == i:
                        continue
                    c_j = fh[j]
                    if c_j < 1e-13:
                        continue
                    trial = fh.copy()
                    trial[j] = c_j * 0.9
                    trial[i] = c_j * 0.1
                    trial = trial / trial.sum()
                    full = make_palin(trial, n)
                    val = fast_convolve(full).max()
                    if val < best_obj - 1e-13:
                        fh = trial
                        best_obj = val
                        best_fh = fh.copy()
                        improved = True
                        break
                continue

            # Try redistributing mass from fh[i] to other positions
            for j in range(m):
                if j == i:
                    continue
                total = fh[i] + fh[j]
                if total < 1e-12:
                    continue

                # Quick scan
                cur_t = fh[i] / total
                best_t = cur_t
                best_v = fast_convolve(make_palin(fh, n)).max()

                for t in [cur_t, 0.0, 1.0, 0.5, cur_t - 0.2, cur_t + 0.2,
                          cur_t - 0.1, cur_t + 0.1]:
                    t = np.clip(t, 0.0, 1.0)
                    trial = fh.copy()
                    trial[i] = t * total
                    trial[j] = (1.0 - t) * total
                    full = make_palin(trial, n)
                    val = fast_convolve(full).max()
                    if val < best_v - 1e-13:
                        best_v = val
                        best_t = t

                if best_v < best_obj - 1e-13:
                    fh[i] = best_t * total
                    fh[j] = (1.0 - best_t) * total
                    fh = fh / fh.sum()
                    best_obj = fast_convolve(make_palin(fh, n)).max()
                    best_fh = fh.copy()
                    improved = True

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                fh = project_simplex(best_fh + rng.standard_normal(m) * 0.01)
                no_improve = 0

    return make_palin(best_fh, n), best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining palindromic constraints, Adam-SG,
    Nesterov subgradient, LP-FW, exact pairwise, and multi-length exploration.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a = a / a.sum()
        sc = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
        return sc

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.8f}")

    # === Phase 1: Palindromic optimization ===
    print(f"\n=== Phase 1: Palindromic (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 1400
    t1 = min(rem, 180)
    if t1 > 20:
        print(f"  Time: {t1:.0f}s, n={len(parent)}")
        a_palin, _ = palindromic_optimize(parent, time_limit=t1, verbose=True)
        try_seq(a_palin, "palin_p1")

        # Also try palindromic with Adam on the full (palindromic-structured) sequence
        if best_seq is not None:
            a_palin2 = make_palin(extract_first_half(best_seq, len(best_seq)), len(best_seq))
            try_seq(a_palin2, "palin_from_best")

    # === Phase 2: Adam-SG on best ===
    print(f"\n=== Phase 2: Adam-SG (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 1150
        t2 = min(rem, 280)
        if t2 > 20:
            print(f"  Time: {t2:.0f}s, n={len(best_seq)}")
            a_opt, _ = adam_sg_optimize(best_seq, time_limit=t2, verbose=True, T_scale=1.0, lr_scale=1.0)
            try_seq(a_opt, "adam_p2")

            # Try with different hyperparameters
            rem = deadline - time.monotonic() - 1100
            if rem > 30:
                a_opt2, _ = adam_sg_optimize(a_opt, time_limit=min(rem, 80), verbose=False, T_scale=0.7, lr_scale=1.3)
                try_seq(a_opt2, "adam_p2b")

    # === Phase 3: Nesterov Subgradient ===
    print(f"\n=== Phase 3: Nesterov Subgrad (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 950
        t3 = min(rem, 200)
        if t3 > 20:
            print(f"  Time: {t3:.0f}s, n={len(best_seq)}")
            a_opt, _ = nesterov_subgrad_optimize(best_seq, time_limit=t3, verbose=True)
            try_seq(a_opt, "nsg_p3")

            # Chain: Nesterov then Adam
            rem = deadline - time.monotonic() - 900
            if rem > 20:
                a_opt2, _ = adam_sg_optimize(a_opt, time_limit=min(rem, 60), verbose=False)
                try_seq(a_opt2, "nsg_then_adam")

    # === Phase 4: LP-FW ===
    print(f"\n=== Phase 4: LP-FW (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 700
        t4 = min(rem, 250)
        if t4 > 20:
            print(f"  Time: {t4:.0f}s, n={len(best_seq)}")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t4, verbose=True)
            try_seq(a_opt, "lpfw_p4")

            # LP-FW from Nesterov result
            rem = deadline - time.monotonic() - 650
            if rem > 20:
                a_opt2, _ = lp_fw_optimize(a_opt, time_limit=min(rem, 80), verbose=False)
                try_seq(a_opt2, "lpfw_p4b")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent
    n_best = len(base)

    target_lengths = [384, 512, 768, 1024, 1280]
    for target_n in target_lengths:
        if target_n == n_best:
            continue
        if target_n > 1500:
            continue
        rem = deadline - time.monotonic() - 400
        if rem < 50:
            break

        f_interp = interpolate_sequence(base, target_n)

        # Quick Adam
        t_alloc = min(rem * 0.35, 100)
        if t_alloc > 15:
            print(f"  Adam n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = adam_sg_optimize(f_interp, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, f"adam_n{target_n}")

        # Quick LP-FW
        rem = deadline - time.monotonic() - 350
        if rem > 25:
            t_lpfw = min(rem * 0.4, 100)
            if t_lpfw > 15:
                print(f"  LP-FW n={target_n}, t={t_lpfw:.0f}s")
                a_opt2, _ = lp_fw_optimize(a_opt, time_limit=t_lpfw, verbose=False)
                try_seq(a_opt2, f"lpfw_n{target_n}")

    # === Phase 6: Pairwise exact optimization ===
    print(f"\n=== Phase 6: Pairwise (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 20:
            t_pw = min(rem * 0.6, 250)
            print(f"  Time: {t_pw:.0f}s, n={len(best_seq)}")
            a_opt, _ = pairwise_exact_optimize(best_seq, time_limit=t_pw, verbose=True)
            try_seq(a_opt, "pairwise_p6")

            # Adam after pairwise
            rem = deadline - time.monotonic() - 200
            if rem > 15:
                a_opt2, _ = adam_sg_optimize(a_opt, time_limit=min(rem, 60), verbose=False)
                try_seq(a_opt2, "adam_after_pw")

    # === Phase 7: Diverse restarts ===
    print(f"\n=== Phase 7: Diverse (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        diverse_inits = []
        # Reversed
        diverse_inits.append(best_seq[::-1].copy())
        # Noise
        rng = np.random.default_rng(999)
        diverse_inits.append(project_simplex(best_seq + rng.standard_normal(n_best) * 0.01))
        # Uniform mix
        diverse_inits.append(project_simplex(0.8 * best_seq + 0.2 * np.ones(n_best) / n_best))
        # Palindromic version
        diverse_inits.append(make_palin(extract_first_half(best_seq, n_best), n_best))

        for di, f_init in enumerate(diverse_inits):
            rem = deadline - time.monotonic() - 150
            if rem < 15:
                break
            t_alloc = min(rem / max(1, len(diverse_inits) - di), 80)
            t_alloc = max(t_alloc, 10)

            if di % 2 == 0:
                a_opt, _ = adam_sg_optimize(f_init, time_limit=min(t_alloc, 60), verbose=False)
            else:
                a_opt, _ = nesterov_subgrad_optimize(f_init, time_limit=min(t_alloc, 60), verbose=False)
            try_seq(a_opt, f"diverse_{di}")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Adam fine
        rem = deadline - time.monotonic() - 50
        if rem > 10:
            a_opt, _ = adam_sg_optimize(best_seq, time_limit=min(rem, 60), verbose=True, T_scale=0.5, lr_scale=0.7)
            try_seq(a_opt, "adam_final")

        # Pairwise final
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            a_opt, _ = pairwise_exact_optimize(a_opt, time_limit=min(rem, 40), verbose=False)
            try_seq(a_opt, "pw_final")

        # Tiny perturbations
        for trial in range(3):
            rem = deadline - time.monotonic() - 10
            if rem < 8:
                break
            rng = np.random.default_rng(1000 + trial)
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)
            a_opt, _ = adam_sg_optimize(f_pert, time_limit=min(rem, 20), verbose=False)
            try_seq(a_opt, f"perturb_{trial}")

        # One more Nesterov pass
        rem = deadline - time.monotonic() - 5
        if rem > 8:
            a_opt, _ = nesterov_subgrad_optimize(best_seq, time_limit=min(rem, 25), verbose=False)
            try_seq(a_opt, "nsg_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        # If something went wrong, use parent
        if parent is not None:
            best_seq = parent
            best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
