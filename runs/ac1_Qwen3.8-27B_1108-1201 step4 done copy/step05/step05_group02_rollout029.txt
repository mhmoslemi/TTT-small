import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_smoothed_subgradient(a, conv, temp):
    """
    Compute smoothed subgradient of max(conv) using softmax weighting and FFT cross-correlation.
    sg[i] = 2 * sum_k w[k] * a[k-i] (where 0 <= k-i < n)
    This equals 2 * cross_correlation(w, a)[i]
    """
    n = len(a)
    scaled = conv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= ws

    # Cross-correlation via FFT: cross[i] = sum_j w[i+j] * a[j]
    # Need FFT size >= len(w) + len(a) - 1 = (2n-1) + n - 1 = 3n - 2
    N = 1 << (3 * n - 2 - 1).bit_length()
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    cross = np.fft.irfft(np.conj(W) * A, N)
    sg = 2.0 * cross[:n]
    return sg


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_min_peak(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0

    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'time_limit': 30})
    if res.success and res.status in [0, 3]:
        return res.x[:n], res.x[-1]
    return None, None

            no_improve += 1
            if no_improve > 50:

        # Line search with golden ratio
        best_alpha = 0.0
        best_val = M

        for alpha in alphas:
            ft = f + alpha * (g - f)

            st = ft.sum()
            if st < 1e-15:

            ft /= st
            val = fast_convolve(ft).max()
            if val < best_val:

                best_alpha = alpha

        # Fine search around best

            if M < best_obj - 1e-16:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 80:
            break

    return best_f, best_obj


def mirror_descent_optimize(f0, time_limit, lr_init=0.002, temp_init=None,
                            temp_decay=0.9997, verbose=False):
    """Mirror descent (multiplicative weights) with adaptive temperature."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    if temp_init is None:
        conv_init = fast_convolve(f)
        temp_init = conv_init.max() * 0.05

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_temp = temp_init
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute smoothed subgradient
        sg = compute_smoothed_subgradient(f, conv, cur_temp)

        # Multiplicative weights update: f_new[i] ∝ f[i] * exp(-lr * sg[i])
        # To avoid underflow, subtract max
        exp_arg = -cur_lr * sg
        exp_arg -= exp_arg.max()
        mult = np.exp(exp_arg)
        f_new = f * mult
        f_new = np.maximum(f_new, 1e-30)
        sn = f_new.sum()
        if sn < 1e-15:
            f_new = np.ones(n) / n
        else:
            f_new /= sn

        conv = fast_convolve(f_new)
        co = conv.max()

        if co < best_obj - 1e-16:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 10000 == 0:
                print(f"    MD iter {it}: obj={best_obj:.10f}, temp={cur_temp:.2e}")
        else:
            no_improve += 1
            f = f_new

        # Decay temperature slowly
        cur_temp *= temp_decay
        # Slowly decay learning rate
        cur_lr *= 0.99999

        # Periodic restart with perturbation
        if no_improve > 20000:
            rng = np.random.default_rng(hash((it, n)) % 2**31)
            noise = rng.standard_normal(n) * 0.003
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            cur_lr = lr_init * 0.5
            no_improve = 0

        # Check time periodically (every 10000 iters to avoid overhead)
        if it % 10000 == 0 and time.monotonic() - t0 > time_limit:
            break

    return best_f, best_obj


def coord_descent_polish(f0, time_limit, verbose=False):
    """Coordinate descent: try small multiplicative changes to individual coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    pass_count = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False

        # Focus on coordinates that contribute to the top peaks
        top_k = min(5, len(conv))
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]

        # Gather candidate coordinates
        candidate_j = set()
        for pk in peak_positions:
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            # Add all contributing coordinates (or a subset if too many)
            contrib = list(range(js, je + 1))
            if len(contrib) > 50:
                # Take every Nth
                step = len(contrib) // 50 + 1
                contrib = contrib[::step]
            candidate_j.update(contrib)

        # Also add some random coordinates for exploration
        rng = np.random.default_rng(pass_count * 7919)
        for _ in range(min(20, n // 20)):
            candidate_j.add(int(rng.integers(0, n)))

        for j in sorted(candidate_j):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-14:
                # Try setting a small value
                for new_val in [0.001 / n, 0.01 / n, 0.05 / n]:
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    f_trial = f_trial / f_trial.sum()
                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-15:
                        f = f_trial
                        best_obj = val
                        best_f = f.copy()
                        conv = fast_convolve(f)
                        improved = True
                        old_val = new_val
                        break
                if improved:
                    continue
                continue

            # Try multiplicative changes
            multipliers = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]
            best_j_val = old_val
            best_j_obj = best_obj

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                f = f.copy()
                f[j] = best_j_val
                f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    conv = fast_convolve(f)
                    improved = True

        if not improved:
            no_improve += 1
            if no_improve > 15:
                break
        else:
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: LP-FW (large steps)
    Phase 2: Mirror descent (many small steps, fast per step)
    Phase 3: Coordinate descent polish
    Phase 4: Multi-length exploration
    Phase 5: Final combined polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 5.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.8f}")

    base_n = len(parent)
    print(f"Starting best: {best_score:.8f}")

    # === Phase 1: LP-FW on parent ===
    print("\n=== Phase 1: LP-FW ===")
    rem = deadline - time.monotonic()
    t1 = min(400, rem * 0.25)
    if t1 > 50:
        print(f"  n={base_n}, time={t1:.0f}s")
        a_opt, _ = lp_fw_optimize(parent, time_limit=t1, verbose=True)
        try_seq(a_opt, "lpfw_parent")

    # === Phase 2: Mirror descent ===
    print(f"\n=== Phase 2: Mirror Descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        t_md = min(rem * 0.35, 600)
        if t_md > 60:
            print(f"  n={n_best}, time={t_md:.0f}s")
            a_opt, _ = mirror_descent_optimize(best_seq, time_limit=t_md,
                                               lr_init=0.001, temp_init=None,
                                               temp_decay=0.9998, verbose=True)
            try_seq(a_opt, "md_main")

            # Second MD run with different params
            rem = deadline - time.monotonic()
            t_md2 = min(rem * 0.2, 300)
            if t_md2 > 40:
                print(f"  n={n_best}, time={t_md2:.0f}s (fine)")
                a_opt, _ = mirror_descent_optimize(best_seq, time_limit=t_md2,
                                                   lr_init=0.0003, temp_init=None,
                                                   temp_decay=0.9999, verbose=True)
                try_seq(a_opt, "md_fine")

    # === Phase 3: LP-FW again on improved sequence ===
    print(f"\n=== Phase 3: LP-FW round 2 (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t3 = min(rem * 0.2, 250)
        if t3 > 40:
            print(f"  n={len(best_seq)}, time={t3:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t3, verbose=True)
            try_seq(a_opt, "lpfw_r2")

    # === Phase 4: Coordinate descent ===
    print(f"\n=== Phase 4: Coord Descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_cd = min(rem * 0.2, 200)
        if t_cd > 30:
            print(f"  n={len(best_seq)}, time={t_cd:.0f}s")
            a_opt, _ = coord_descent_polish(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd_main")

            # Another CD round
            rem = deadline - time.monotonic()
            t_cd2 = min(rem * 0.1, 100)
            if t_cd2 > 20:
                a_opt, _ = coord_descent_polish(best_seq, time_limit=t_cd2)
                try_seq(a_opt, "cd_r2")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.8f}) ===")
    if best_seq is not None:
        base = best_seq
        # Try a few nearby lengths
        deltas = [0, -64, 64, -32, 32, -128, 128]
        tried_lengths = set()
        for d in deltas:
            rem = deadline - time.monotonic()
            if rem < 100:
                break
            target_n = len(base) + d
            if target_n in tried_lengths or target_n < 128 or target_n > 2048:
                continue
            tried_lengths.add(target_n)

            f_interp = interpolate_sequence(base, target_n)
            t_alloc = min(rem * 0.25, 150)
            if t_alloc < 30:
                continue
            print(f"  n={target_n}, time={t_alloc:.0f}s")

            # Quick MD on interpolated
            a_opt, _ = mirror_descent_optimize(f_interp, time_limit=t_alloc * 0.7,
                                               lr_init=0.001, temp_init=None,
                                               temp_decay=0.9998)
            try_seq(a_opt, f"md_n{target_n}")

            rem = deadline - time.monotonic()
            if rem > 40 and a_opt is not None:
                a_opt2, _ = coord_descent_polish(a_opt, time_limit=min(rem, 60))
                try_seq(a_opt2, f"cd_n{target_n}")

    # === Phase 6: Final combined polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Quick MD
        rem = deadline - time.monotonic() - 20
        if rem > 20:
            a_opt, _ = mirror_descent_optimize(best_seq, time_limit=min(rem, 80),
                                               lr_init=0.0002, temp_init=None,
                                               temp_decay=0.99995)
            try_seq(a_opt, "md_polish")

        # Quick CD
        rem = deadline - time.monotonic() - 15
        if rem > 15:
            a_opt, _ = coord_descent_polish(best_seq, time_limit=min(rem, 50))
            try_seq(a_opt, "cd_polish")

        # Perturbation + MD
        for p in range(8):
            rem = deadline - time.monotonic() - 10
            if rem < 15:
                break
            scale = 0.002 + 0.001 * p
            noise = rng.standard_normal(n_best) * scale
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = mirror_descent_optimize(f_pert, time_limit=min(rem, 30),
                                               lr_init=0.0005, temp_init=None,
                                               temp_decay=0.9999)
            try_seq(a_opt, f"pert_{p}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
