import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgrad(f, conv, top_k, temperature):
    n = len(f)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    # Weight by how much above mean
    mean_val = conv.sum() / len(conv)
    weights = np.maximum(pv - mean_val, 0)
    if weights.sum() < 1e-15:
        weights = np.ones(len(peaks))
    weights = weights / weights.sum()
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if weights[i] < 1e-14:
            continue
        k = int(peaks[i])
        m_lo = max(0, k - (n - 1))
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            indices = np.arange(m_lo, m_hi + 1)
            sg[indices] += weights[i] * f[k - indices]
    return sg * 2.0  # Factor of 2 from the derivative

def mirror_descent(f0, time_limit, eta0=0.1, top_k=20, temp=0.001):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    eta = eta0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        sg = compute_subgrad(f, conv, top_k, temp)
        
        # Mirror descent update
        log_f = np.log(f + 1e-30)
        log_f_new = log_f - eta * sg
        # Numerical stability
        log_f_new -= log_f_new.max()
        f_new = np.exp(log_f_new)
        f_new = f_new / f_new.sum()
        
        conv = fast_convolve(f_new)
        obj = conv.max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:

            if no_improve > 50:

                    best_obj = conv.max()
                    best_f = f.copy()
                    eta = eta0 * 0.5
                    no_improve = 0
        
        if it % 100 == 0:
            eta = max(eta * 0.95, 1e-6)
    
    return best_f, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)

        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
