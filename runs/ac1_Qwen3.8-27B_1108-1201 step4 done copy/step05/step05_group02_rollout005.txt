Now, for L-BFGS, I need to optimize over the constraint a >= 0, sum(a) = 1. I can parameterize using the softmax: a = exp(z) / sum(exp(z)). Then I optimize over z in R^n (with no constraints). The gradient w.r.t. z is the standard softmax gradient transform.

Alternatively, I can use L-BFGS-B with bounds [0, None] and an equality constraint. But equality constraints are handled poorly by L-BFGS-B.

The softmax parameterization is cleaner. Let me use that.

If a = softmax(z) = exp(z - logsumexp(z)), then:
d a_j / d z_i = a_j * (delta_{ji} - a_i)

So the gradient w.r.t. z is:
g_z_i = sum_j g_a_j * d a_j / d z_i = sum_j g_a_j * a_j * (delta_{ji} - a_i)
      = g_a_i * a_i - a_i * sum_j g_a_j * a_j
      = a_i * (g_a_i - sum_j g_a_j * a_j)

This is the standard softmax gradient.

So my optimization problem becomes: minimize f(softmax(z)) over z in R^n.

This is unconstrained, so L-BFGS works perfectly.

1. Helper functions (fast_convolve, smooth_obj_and_grad, softmax, etc.)
2. L-BFGS optimizer with adaptive temperature
3. Peak-targeted coordinate descent
4. Simulated annealing with structured moves
5. Main `propose_candidate` function that orchestrates everything

Let me also think about the temperature schedule for log-sum-exp:
- Start with high t (smooth, easy to optimize globally)
- Decrease t (sharper, closer to true max)
- This is like simulated annealing on the exponent

Actually, for minimization, I want:
- High t → log-sum-exp ≈ max (sharp, harder to optimize but closer to true objective)
- Low t → log-sum-exp ≈ mean (smooth, easier to optimize)

So I should start with low t (smooth) and increase t (sharper). This is the annealing schedule.

Let me think about the coordinate descent approach more carefully:

For coordinate j, if I change a_j to a_j' and rescale the rest to maintain sum=1:
- a_j' is the new value for coordinate j
- All other coordinates are scaled by (1-a_j')/(1-a_j)
- The new convolution can be computed efficiently:
  conv_new = conv_old - 2*a_j*(a vector at appropriate positions) + 2*a_j'*(scaled a vector) + (a_j'^2 - a_j^2) at position 2j
  
  Actually this is getting complicated. Let me just recompute the convolution from scratch for each candidate value (it's fast with FFT).

For the simulated annealing approach:
- State: the sequence a (on the simplex)
- Move: pick coordinate j, propose new value a_j' ~ some distribution, rescale others
- Energy: the objective value
- Accept if energy decreases, or with probability exp(-delta/T)

The key is to design good moves. Options:
1. Single coordinate: change one coordinate and rescale
2. Pair swap: move mass from coordinate j to coordinate k
3. Block move: change a block of consecutive coordinates

I think pair swaps are the most natural and effective.

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
1. L-BFGS on log-sum-exp smoothed objective with adaptive temperature
2. Peak-targeted coordinate descent with FFT-based evaluation
3. Structured simulated annealing with pair-swap moves
4. Multi-length exploration via interpolation
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax(z):
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    return e / e.sum()


def logsumexp(x):
    m = x.max()
    if m == -np.inf:
        return -np.inf
    return m + np.log(np.sum(np.exp(x - m)))


def smooth_obj(a, t):
    """Log-sum-exp smoothed max of convolution."""
    conv = fast_convolve(a)
    return logsumexp(t * conv) / t


def smooth_grad(a, t):
    """Gradient of log-sum-exp smoothed max of convolution w.r.t. a."""
    conv = fast_convolve(a)
    shifted = t * (conv - conv.max())
    w = np.exp(shifted)
    w /= w.sum()
    
    n = len(a)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        grad[js:je+1] += 2.0 * wk * a[k-je:k-js+1][::-1]
    return grad


def logsumexp_obj_z(z, t):
    """Objective in softmax parameterization."""
    a = softmax(z)
    return smooth_obj(a, t)


def logsumexp_grad_z(z, t):
    """Gradient in softmax parameterization."""
    a = softmax(z)
    g_a = smooth_grad(a, t)
    # Transform to z-space
    mean_ga = np.dot(g_a, a)
    g_z = a * (g_a - mean_ga)
    return g_z


def lbfgs_optimize(f0, t_start=10.0, t_end=5000.0, time_limit=60.0,
                   verbose=False):
    """Optimize using L-BFGS on log-sum-exp smoothed objective."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    # Initialize z in log space
    f_clipped = np.maximum(f, 1e-30)
    z = np.log(f_clipped)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    
    # Adaptive temperature schedule
    # Start smooth, get sharper
    n_restarts = 3
    time_per_restart = time_limit / n_restarts
    
    for restart in range(n_restarts):
        t_curr = t_start * (t_end / t_start) ** (restart / max(1, n_restarts - 1))
        
        def obj(z_in):
            a = softmax(z_in)
            conv = fast_convolve(a)
            return np.log(conv.max())  # Using log for numerical stability in L-BFGS
            # Actually let's use the true smoothed obj
            # return smooth_obj(a, t_curr)
        
        def grad(z_in):
            a = softmax(z_in)
            g_a = smooth_grad(a, t_curr)
            mean_ga = np.dot(g_a, a)
            return a * (g_a - mean_ga)
        
        def obj_true(z_in):
            a = softmax(z_in)
            return smooth_obj(a, t_curr)
        
        # Use scipy L-BFGS-B on the smooth objective
        res = minimize(
            obj_true, z, jac=grad,
            method='L-BFGS-B',
            options={'maxiter': 200, 'gtol': 1e-6, 'ftol': 1e-10}
        )
        
        if res.x is not None:
            f_new = softmax(res.x)
            f_new = np.maximum(f_new, 0)
            sn = f_new.sum()
            if sn > 1e-15:
                f_new /= sn
            obj_new = fast_convolve(f_new).max()
            if obj_new < best_obj:
                best_obj = obj_new
                best_f = f_new.copy()
                z = res.x.copy()
                if verbose:
                    print(f"    L-BFGS restart {restart}: obj={best_obj:.10f} (t={t_curr:.1f})")
        
        if time.monotonic() - t0 > time_limit:
            break
    
    return best_f, best_obj


def peak_cd_optimize(f0, time_limit=120.0, verbose=False):
    """Peak-targeting coordinate descent."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        # Find top peaks
        top_k = min(5, 2 * n - 1)
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        
        improved = False
        
        for pk in sorted(peak_indices, key=lambda x: -conv[x]):
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            
            # Get contributing coordinates sorted by their contribution to this peak
            contributions = np.zeros(je - js + 1)
            for j in range(js, je + 1):
                contributions[j - js] = 2.0 * f[peak_k - j] * f[j]  # Wait, this isn't quite right
                # Actually the contribution of (f[j], f[peak_k-j]) to conv[peak_k] is f[j]*f[peak_k-j]
                # But if peak_k-j == j, it's f[j]^2
                # The "contribution" of coordinate j to the peak is:
                # d conv[peak_k] / d f[j] * f[j] = 2*f[peak_k-j]*f[j] (if peak_k-j != j)
                # or 2*f[j]^2 (if peak_k-j == j)
                # Either way it's 2*f[j]*f[peak_k-j]
            
            # Try reducing the coordinate with highest contribution
            # Sort by contribution (descending)
            order = np.argsort(contributions)[::-1]
            
            for idx in order[:min(10, len(order))]:
                j = js + idx
                if f[j] < 1e-15:
                    continue
                
                # Try reducing f[j] by various amounts
                old_val = f[j]
                best_j_val = old_val
                best_j_obj = best_obj
                
                for fraction in [0.1, 0.3, 0.5, 0.7, 0.9]:
                    new_val = old_val * (1.0 - fraction)
                    if new_val < 0:
                        continue
                    # Rescale others
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial = f_trial / f_trial.sum()
                    
                    val = fast_convolve(f_trial).max()
                    if val < best_j_obj - 1e-15:
                        best_j_obj = val
                        best_j_val = new_val
                
                # Also try increasing
                for fraction in [0.1, 0.3, 0.5]:
                    new_val = old_val * (1.0 + fraction)
                    if new_val > 0.9:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial = f_trial / f_trial.sum()
                    
                    val = fast_convolve(f_trial).max()
                    if val < best_j_obj - 1e-15:
                        best_j_obj = val
                        best_j_val = new_val
                
                if best_j_val != old_val:
                    denom = 1.0 - old_val
                    if denom > 1e-10:
                        scale = (1.0 - best_j_val) / denom
                        f[j] = best_j_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f[mask] *= scale
                        f = f / f.sum()
                    conv = fast_convolve(f)
                    if best_j_obj < best_obj - 1e-15:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 20:
                # Perturb and try again
                rng = np.random.default_rng(iteration * 31 + 7)
                n_perturb = max(5, n // 50)
                indices = rng.choice(n, size=n_perturb, replace=False)
                for idx in indices:
                    f[idx] *= 1.0 + rng.standard_normal() * 0.01
                f = np.maximum(f, 0)
                f = f / f.sum()
                conv = fast_convolve(f)
                if conv.max() < best_obj:
                    best_obj = conv.max()
                    best_f = f.copy()
                no_improve = 0
            elif no_improve > 50:
                break
        else:
            no_improve = 0
    
    return best_f, best_obj


def simulated_annealing(f0, time_limit=120.0, verbose=False):
    """Simulated annealing with pair-swap moves on the simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    current_obj = conv.max()
    best_obj = current_obj
    best_f = f.copy()
    
    T = 1e-5  # Initial temperature
    T_end = 1e-9
    cooling = 0.9997
    
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Pair swap: move mass from j to k
        j = int(rng.integers(0, n))
        k = int(rng.integers(0, n))
        if j == k:
            continue
        
        # Move a fraction of f[j] to f[k]
        frac = rng.uniform(0.01, 0.2)
        delta = f[j] * frac
        f_trial = f.copy()
        f_trial[j] -= delta
        f_trial[k] += delta
        
        if f_trial[j] < 0:
            continue
        
        obj_trial = fast_convolve(f_trial).max()
        delta_obj = obj_trial - current_obj
        
        if delta_obj < 0 or (T > T_end and rng.random() < np.exp(-delta_obj / T)):
            f = f_trial
            current_obj = obj_trial
            if current_obj < best_obj:
                best_obj = current_obj
                best_f = f.copy()
                if verbose and it % 10000 == 0:
                    print(f"    SA iter {it}: best={best_obj:.10f}, T={T:.2e}")
        
        T = max(T_end, T * cooling)
        
        # Periodically inject randomness
        if it % 5000 == 0:
            n_perturb = max(3, n // 100)
            indices = rng.choice(n, size=n_perturb, replace=False)
            for idx in indices:
                f[idx] *= 1.0 + rng.standard_normal() * 0.005
            f = np.maximum(f, 0)
            f = f / f.sum()
            current_obj = fast_convolve(f).max()
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def lp_min_peak(f, n, max_iter=5000):
    """Solve LP: min t s.t. (f*g)[k] <= t, g >= 0, sum(g) = 1.
    Returns (g, t) or (None, None)."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None


def lp_fw_step(f, n):
    """One LP-based Frank-Wolfe step."""
    g, lp_t = lp_min_peak(f, n)
    if g is None:
        return None
    gs = g.sum()
    if gs < 1e-15:
        return None
    g = g / gs
    return g


def multi_start_lbfgs(f0, time_limit=60.0, n_starts=3):
    """Multi-start L-BFGS optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    time_per_start = time_limit / n_starts
    rng = np.random.default_rng(999)
    
    for start in range(n_starts):
        if time.monotonic() - t0 > time_limit:
            break
        
        if start == 0:
            f_curr = f.copy()
        else:
            # Perturbed start
            noise = rng.standard_normal(n) * (0.01 + 0.02 * start)
            f_curr = np.maximum(f + noise, 0)
            f_curr = f_curr / f_curr.sum()
        
        a_opt, obj_opt = lbfgs_optimize(f_curr, time_limit=time_per_start,
                                         t_start=5.0, t_end=2000.0, verbose=True)
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_f = a_opt.copy()
    
    return best_f, best_obj
