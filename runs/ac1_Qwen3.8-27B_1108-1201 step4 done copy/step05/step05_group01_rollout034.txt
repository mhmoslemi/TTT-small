```python
import numpy as np
import time
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def softmax(x):
    xm = x - x.max()
    ex = np.exp(xm)
    return ex / ex.sum()

def soft_max_objective(x, T):
    """Compute (2n/T) * log(sum exp(T * conv(softmax(x))))"""
    a = softmax(x)
    n = len(a)
    conv = fast_convolve(a)
    # Soft max with numerical stability
    Tconv = T * conv
    Tconv_max = Tconv.max()
    Z = np.sum(np.exp(Tconv - Tconv_max))
    obj = 2 * n * (Tconv_max / T + np.log(Z) / T)
    return obj

def soft_max_gradient(x, T):
    """Compute gradient of soft_max_objective w.r.t. x"""
    a = softmax(x)
    n = len(a)
    conv = fast_convolve(a)
    
    Tconv = T * conv
    Tconv_max = Tconv.max()
    p = np.exp(Tconv - Tconv_max)
    p = p / p.sum()
    
    # r = sum_k p[k] * conv[k]
    r = np.dot(p, conv)
    
    # q[j] = sum_k p[k] * a[k-j] * I(0<=k-j<n)
    # Direct computation
    q = np.zeros(n)
    for j in range(n):
        # k-j in [0, n-1] means k in [j, j+n-1]
        k_min = j
        k_max = min(j + n - 1, 2*n - 2)
        if k_min <= k_max:
            u = np.arange(k_min, k_max + 1)  # k values
            q[j] = np.dot(p[k_min:k_max+1], a[u - j])
    
    grad = (4.0 * n * a / T) * (q - r)
    return grad
```

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using soft-max smoothed
gradient optimization with T-annealing, multi-length exploration,
peak-directed moves, and simulated annealing polish.

import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(x):
    """Numerically stable softmax."""
    xm = x - x.max()
    ex = np.exp(xm)
    s = ex.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return ex / s


def compute_q(p, a):
    """Compute q[j] = sum_m a[m] * p[m+j] for j=0..n-1 using FFT."""
    n = len(a)
    N = 1 << (2 * n - 1 + n - 1).bit_length()  # >= 3n-2
    FA = np.fft.rfft(a, N)
    FP = np.fft.rfft(p, N)
    # Cross-correlation: ifft(FA * conj(FP)) gives sum_k a[k]*p[k+j]
    # Wait, I need to be careful about the convention.
    # numpy ifft(FA * conj(FP))[j] = sum_k a[k] * p[(k+j) % N]
    # For linear cross-correlation, we need N >= len(a) + len(p) - 1
    # With zero-padding, this gives the linear cross-correlation.
    corr = np.fft.irfft(FA * np.conj(FP), N)
    return corr[:n]


def soft_max_value_and_grad(x, T):
    """Compute soft-max objective and gradient w.r.t. x (softmax parameterization)."""
    a = softmax_fn(x)
    n = len(a)
    conv = fast_convolve(a)
    
    Tconv = T * conv
    Tconv_max = Tconv.max()
    exp_Tconv = np.exp(Tconv - Tconv_max)
    Z = exp_Tconv.sum()
    p = exp_Tconv / Z
    
    # Objective: 2n * (Tconv_max + log(Z)) / T
    obj = 2.0 * n * (Tconv_max + np.log(Z)) / T
    
    # Gradient computation
    r = np.dot(p, conv)
    q = compute_q(p, a)
    grad = (4.0 * n * a / T) * (q - r)
    
    return obj, grad

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length, maintaining sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using soft-max smoothed
L-BFGS optimization with T-annealing, multi-length exploration,
gradient descent on simplex, peak-directed moves, and simulated
annealing for final polish.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(x):
    """Numerically stable softmax."""
    xm = x - x.max()
    ex = np.exp(xm)
    s = ex.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return ex / s


def compute_q_fft(p, a):
    """Compute q[j] = sum_m a[m] * p[m+j] for j=0..n-1 using FFT."""
    n = len(a)
    N = 1 << (2 * n - 1 + n - 1).bit_length()
    FA = np.fft.rfft(a, N)
    FP = np.fft.rfft(p, N)
    corr = np.fft.irfft(FA * np.conj(FP), N)
    return corr[:n]


def soft_max_obj(x, T):
    """Compute soft-max smoothed objective value."""
    a = softmax_fn(x)
    n = len(a)
    conv = fast_convolve(a)
    Tconv = T * conv
    Tconv_max = Tconv.max()
    Z = np.sum(np.exp(Tconv - Tconv_max))
    return 2.0 * n * (Tconv_max + np.log(Z)) / T


def soft_max_grad(x, T):
    """Compute gradient of soft-max smoothed objective w.r.t. x."""
    a = softmax_fn(x)
    n = len(a)
    conv = fast_convolve(a)
    Tconv = T * conv
    Tconv_max = Tconv.max()
    exp_tc = np.exp(Tconv - Tconv_max)
    Z = exp_tc.sum()
    p = exp_tc / Z
    r = np.dot(p, conv)
    q = compute_q_fft(p, a)
    grad = (4.0 * n * a / T) * (q - r)
    return grad


def lp_bfgs_soft_max(a_init, time_limit, T_schedule):
    """Optimize using L-BFGS with soft-max parameterization and T-annealing."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a_init)) / len(a_init)
    else:
        a = a / s
    n = len(a)

    x = np.log(a + 1e-30)
    best_a = a.copy()
    best_max = fast_convolve(a).max()
    t0 = time.monotonic()

    for T in T_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        res = minimize(soft_max_obj, x, jac=soft_max_grad, args=(T,),
                       method='L-BFGS-B',
                       options={'maxiter': 300, 'ftol': 1e-14,
                                'gtol': 1e-10, 'maxfun': 800})
        a_new = softmax_fn(res.x)
        conv_new = fast_convolve(a_new)
        max_new = conv_new.max()
        if max_new < best_max:
            best_max = max_new
            best_a = a_new.copy()
        x = res.x.copy()
        print(f"    T={T}: max_conv={max_new:.12f}, obj={2*n*max_new:.6f}")

    return best_a, best_max


def grad_descent_simplex(a_init, time_limit, lr0=0.005, decay=0.9997):
    """Subgradient descent on the simplex with peak tracking."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a_init)) / len(a_init)
    else:
        a = a / s
    n = len(a)
    conv = fast_convolve(a)
    best_max = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    lr = lr0
    no_improve = 0
    top_k = max(5, n // 10)

    while time.monotonic() - t0 < time_limit:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        weights = np.zeros(len(peaks))
        if len(peaks) == 1:
            weights[0] = 1.0
        else:
            scaled = (pv - pv.max()) / max(pv.max() - pv.min(), 1e-15)
            weights = np.exp(scaled * 3.0)
            weights /= weights.sum()

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-12:
                continue
            k = int(peaks[i])
            m_lo = max(0, k - n + 1)
            m_hi = min(n - 1, k)
            sg[m_lo:m_hi + 1] += 2.0 * weights[i] * a[k - np.arange(m_lo, m_hi + 1)]

        a_new = a - lr * sg
        a_new = project_simplex(a_new)
        conv = fast_convolve(a_new)
        obj = conv.max()

        if obj < best_max - 1e-13:
            best_max = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 80:
                lr *= 0.85
                if lr < 1e-9:
                    break
        a = a_new
        lr = max(lr * decay, 1e-9)

    return best_a, best_max


def cosine_init(n, freq, amp):
    """Cosine-modulated initialization."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def two_cos_init(n, f1, a1, f2, a2):
    """Two-frequency cosine initialization."""
    j = np.arange(n)
    a = 1.0 + a1 * np.cos(2.0 * np.pi * f1 * j / n) + a2 * np.cos(2.0 * np.pi * f2 * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length, maintaining sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def simulated_annealing_opt(a_init, time_limit, T_start=0.04, T_end=0.002, seed=42):
    """Simulated annealing with mixed move types."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a_init)) / len(a_init)
    else:
        a = a / s
    n = len(a)
    conv = fast_convolve(a)
    cur_obj = conv.max()
    best_a = a.copy()
    best_obj = cur_obj
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(elapsed / time_limit, 1.0)
        T = T_start * (T_end / T_start) ** progress

        move_type = rng.integers(0, 5)
        a_new = a.copy()

        if move_type == 0:
            j = rng.integers(0, n)
            delta = rng.standard_normal() * 0.008
            a_new[j] = max(0.0, a[j] + delta)
        elif move_type == 1:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                amt = min(a[i], 0.1) * rng.uniform(0.1, 0.6)
                a_new[i] -= amt
                a_new[j] += amt
        elif move_type == 2:
            bs = min(8, n // 8)
            start = int(rng.integers(0, n - bs))
            noise = rng.standard_normal(bs) * 0.004
            a_new[start:start + bs] = np.maximum(a_new[start:start + bs] + noise, 0.0)
        elif move_type == 3:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j and a[i] > a[j]:
                amt = (a[i] - a[j]) * 0.4 * rng.uniform(0.1, 1.0)
                a_new[i] -= amt
                a_new[j] += amt
        else:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                swap_amt = min(a[i], a[j]) * 0.5 * rng.uniform(0.3, 1.0)
                a_new[i] -= swap_amt
                a_new[j] += swap_amt

        a_new = project_simplex(a_new)
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < cur_obj or rng.random() < np.exp(-(obj_new - cur_obj) / max(T, 1e-10)):
            a = a_new
            cur_obj = obj_new
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()

    return best_a, best_obj


def peak_reduction(a_init, time_limit, n_max_steps=10000):
    """Iteratively reduce the peak by redistributing mass from peak-contributing pairs."""
    rng = np.random.default_rng(123)
    a = np.maximum(a_init.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, 0.0
    a = a / s
    n = len(a)
    conv = fast_convolve(a)
    best_max = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    steps = 0

    for _ in range(n_max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        steps += 1
        k_peak = int(np.argmax(conv))
        m_lo = max(0, k_peak - n + 1)
        m_hi = min(n - 1, k_peak)
        if m_lo > m_hi:
            break
        indices = np.arange(m_lo, m_hi + 1)
        contribs = a[indices] * a[k_peak - indices]
        total = contribs.sum()
        if total < 1e-15:
            break
        max_idx = int(np.argmax(contribs))
        m_star = indices[max_idx]
        k_other = k_peak - m_star
        frac = 0.02 + 0.03 * (steps % 10) / 10.0
        eps = frac * a[m_star]
        if eps < 1e-12:
            continue
        a_new = a.copy()
        if m_star == k_other:
            a_new[m_star] = max(0.0, a[m_star] - 2.0 * eps)
            target = int(rng.integers(0, n))
            a_new[target] += 2.0 * eps
        else:
            a_new[m_star] = max(0.0, a[m_star] - eps)
            a_new[k_other] = max(0.0, a[k_other] - eps)
            min_idx = int(np.argmin(contribs))
            target = indices[min_idx]
            if target == m_star or target == k_other:
                target = int(rng.integers(0, n))
            a_new[target] += eps
            target2 = int(rng.integers(0, n))
            a_new[target2] += eps
        a_new = project_simplex(a_new)
        conv = fast_convolve(a_new)
        obj = conv.max()
        if obj < best_max - 1e-13:
            best_max = obj
            best_a = a_new.copy()
            a = a_new

    return best_a, best_max
