def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

Actually, for a continuous uniform distribution on [0,1], the sum of two independent draws has a triangular distribution with peak 1 (at sum=1). The maximum of the density is 1, which equals `1/1 = 1` (where 1 is the range of each variable). In the discrete case with n points, the uniform gives peak `n/n^2 = 1/n`.

For a continuous Gaussian with variance σ², the sum has variance 2σ², and the peak density is `1/sqrt(4*pi*σ²) = 1/(2*sigma*sqrt(pi))`.

In the discrete case, if p is approximately Gaussian with variance v, the peak of the discrete convolution is approximately `1/sqrt(4*pi*v)` (for large n).

To minimize the peak, we want to MAXIMIZE the variance v. The maximum variance of a distribution on {0,...,n-1} is achieved by a Bernoulli distribution (mass at 0 and n-1), giving v = (n-1)²/4.

But a Bernoulli distribution gives a very peaked convolution: `(p*p)[0] = p[0]^2`, `(p*p)[n-1] = 2*p[0]*p[n-1]`, `(p*p)[2(n-1)] = p[n-1]^2`. If p[0] = p[n-1] = 1/2, then `(p*p)[n-1] = 1/2`. That's much worse than uniform's 1/n for large n.

So there's a tradeoff: spreading out (high variance) helps the Gaussian peak but creates issues at the edges. The optimal is somewhere in between.

Actually, I think the key insight is this: for the discrete uniform distribution, the convolution is a triangle (trapezoid in the discrete case), with peak 1/n and support of size 2n-1. The "flatness" is determined by the ratio peak/mean = (1/n) / (1/(2n-1)) = (2n-1)/n ≈ 2.

For a distribution that makes the convolution more rectangular (flat), we'd get peak/mean closer to 1, which is the theoretical minimum.

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def gs_flatten(a_init, max_time, alpha=0.4, verbose=False):

def sg_descent(a_init, time_limit, lr=0.01, momentum=0.9, top_k=50, temp=0.001, verbose=False):

    a = np.maximum(a_init.astype(np.float64), 0.0)

        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temp
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        scaled -= scaled.max()
        w = np.exp(scaled)

            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0

                continue

            best_obj = obj
            best_a = a_new.copy()

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
