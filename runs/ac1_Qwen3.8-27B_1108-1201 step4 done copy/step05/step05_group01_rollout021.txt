    while time.monotonic() - t0 < time_limit:
        it += 1

        grad, cur_max, _ = smooth_max_subgrad(y, tau, K)

        # Adaptive step based on gradient norm
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-14:
            break
        # Step size: balance between smoothness and progress
        step = min(0.3, max(0.005, 0.5 / (g_norm * step_scale)))

        # Heavy ball momentum update
        y_new = y - step * grad + momentum * (y - prev_a)
        a_new = project_simplex(y_new)

        # Nesterov extrapolation for next y
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_new + ((t - 1.0) / t_new) * (a_new - prev_a)
        y = project_simplex(y)
        prev_a = a
        a = a_new
        t = t_new

        # Slowly decrease temperature (sharpen the approximation)
        tau = max(0.0002, tau * 0.9997)

        # Track best
        if cur_max < best_obj - 1e-13:
            best_obj = cur_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Periodic restart to escape slow convergence
        if no_improve > 800:
            rng = np.random.default_rng((it * 2654435761) % (2**31))
            noise_scale = 0.001 + 0.002 * min(no_improve // 800, 5)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            y = a.copy()
            prev_a = a.copy()
            t = 1.0
            tau = max(0.0005, tau * 2.0)
            step = 0.05 * step_scale
            no_improve = 0

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA it={it}, obj={best_obj:.10f}, tau={tau:.6f}, step={step:.4f}, time={elapsed:.0f}s")

    return best_a, best_obj


def coordinate_descent(a0, time_limit, max_passes=200, verbose=False):
    """
    Coordinate descent: for each coordinate j, find optimal value via golden section.
    Maintains sum(a)=1 by scaling others.
    """
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0.0).astype(np.float64))
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    GR = (1.0 + np.sqrt(5.0)) / 2.0
    inv_GR = 1.0 / GR

    passes = 0
    no_pass_improve = 0

    while time.monotonic() - t0 < time_limit and passes < max_passes:
        passes += 1
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-13:
                # Try setting to a small positive value
                candidates = [0.0, 1e-4, 1e-3, 1e-4 / (n - 1)]
            else:
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                lo = max(0.0, old_val * 0.3)
                hi = min(0.99, old_val * 3.0)
                candidates = None

            if candidates is not None:
                best_cand = candidates[0]
                best_cand_val = best_obj + 1
                for c in candidates:
                    a_try = a.copy()
                    a_try[j] = c
                    s = a_try.sum()
                    if s > 1e-15:
                        a_try = a_try / s
                    v = fast_convolve(a_try).max()
                    if v < best_cand_val:
                        best_cand_val = v
                        best_cand = c
                if best_cand_val < best_obj - 1e-13:
                    a[j] = best_cand
                    s = a.sum()
                    if s > 1e-15:
                        a = a / s
                    best_obj = best_cand_val
                    best_a = a.copy()
                    pass_improved = True
                continue

            # Golden section search for coordinate j
            c1 = lo + inv_GR * (hi - lo)
            c2 = lo + (1.0 - inv_GR) * (hi - lo)

            def eval_coord(x):
                a_try = a.copy()
                a_try[j] = x
                ft = 1.0 - x
                if ft < 0:
                    return float('inf')
                if ft < 1e-15:
                    a_try = a_try / max(a_try.sum(), 1e-15)
                    return fast_convolve(a_try).max()
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_try[mask] = a_try[mask] * (ft / denom)
                return fast_convolve(a_try).max()

            f1 = eval_coord(c1)
            f2 = eval_coord(c2)

            for _ in range(10):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + inv_GR * (hi - lo)
                    f1 = eval_coord(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + (1.0 - inv_GR) * (hi - lo)
                    f2 = eval_coord(c2)

            opt_x = (lo + hi) / 2.0
            opt_val = min(f1, f2)

            if opt_val < best_obj - 1e-13:
                a[j] = opt_x
                ft = 1.0 - opt_x
                if ft > 1e-15:
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] = a[mask] * (ft / denom)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                cur = fast_convolve(a).max()
                if cur < best_obj - 1e-13:
                    best_obj = cur
                    best_a = a.copy()
                pass_improved = True

        if pass_improved:
            no_pass_improve = 0
        else:
            no_pass_improve += 1
            if no_pass_improve >= 3:
                break

        if verbose and passes % 10 == 0:
            print(f"    CD pass {passes}: obj={best_obj:.10f}")

    return best_a, best_obj


def cosine_init(n, freq, amp, seed=0):
    """Generate cosine-modulated initialization on the simplex."""
    rng = np.random.default_rng(seed)
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, maintaining simplicity."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1.0, old_n)
    new_x = np.linspace(0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def multi_freq_init(n, seed=0):
    """Multi-frequency cosine initialization."""
    rng = np.random.default_rng(seed)
    j = np.arange(n)
    a = np.ones(n)
    for _ in range(rng.integers(1, 4)):
        freq = rng.integers(1, 6)
        amp = rng.uniform(0.05, 0.3)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * freq * j / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: FISTA subgradient + coordinate descent + multi-length
    exploration with adaptive parameters.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512

    n_parent = len(parent)

    # === Phase 1: Quick FISTA on parent ===
    print("=== Phase 1: FISTA on parent ===")
    rem = deadline - time.monotonic()
    t1 = min(150, rem * 0.12)
    if t1 > 20:
        a_opt, _ = fista_optimize(parent, time_limit=t1, K=80, momentum=0.88, step_scale=1.0)
        try_seq(a_opt, "fista_parent")

    # === Phase 2: Explore different lengths and initializations ===
    print("\n=== Phase 2: Multi-length exploration ===")
    base = best_seq if best_seq is not None else parent

    lengths_to_try = sorted(set([
        n_parent,
        max(256, n_parent - 128),
        n_parent + 128,
        n_parent + 256,
        n_parent * 2 if n_parent * 2 <= 3000 else n_parent + 512,
    ]))

    for n_try in lengths_to_try:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(rem * 0.15, 120)
        if t_alloc < 15:
            continue

        # Interpolate parent to new length
        if n_try == n_parent:
            a_init = base.copy()
        else:
            a_init = interpolate_sequence(base, n_try)

        # Quick FISTA
        a_opt, _ = fista_optimize(a_init, time_limit=t_alloc, K=60, momentum=0.9, step_scale=1.0)
        try_seq(a_opt, f"fista_n{n_try}")

    # === Phase 3: Diverse initializations ===
    print("\n=== Phase 3: Diverse initializations ===")
    if best_seq is not None:
        n_best = len(best_seq)
    else:
        n_best = n_parent

    init_configs = [
        (n_best, "cos1", 1, 0.25),
        (n_best, "cos2", 2, 0.2),
        (n_best, "multi", None, None),
        (max(512, n_best + 256), "cos3", 3, 0.15),
        (max(512, n_best + 256), "multi", None, None),
    ]

    for idx, (n_init, name, freq, amp) in enumerate(init_configs):
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t_alloc = min(rem * 0.1, 80)
        if t_alloc < 10:
            continue

        if name == "multi":
            a_init = multi_freq_init(n_init, seed=seed + idx * 137)
        else:
            a_init = cosine_init(n_init, freq, amp, seed=seed + idx * 73)

        # Blend with best
        if best_seq is not None and len(best_seq) == n_init:
            a_init = project_simplex(0.6 * a_init + 0.4 * best_seq)

        a_opt, _ = fista_optimize(a_init, time_limit=t_alloc, K=60, momentum=0.9, step_scale=1.0)
        try_seq(a_opt, f"init_{name}_n{n_init}")

    # === Phase 4: Main intensive FISTA ===
    print(f"\n=== Phase 4: Intensive FISTA (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        t4 = min(rem * 0.35, 600)
        if t4 > 30:
            print(f"  Running FISTA for {t4:.0f}s, n={n_best}")
            a_opt, _ = fista_optimize(best_seq, time_limit=t4, K=100, momentum=0.92, step_scale=0.8, verbose=True)
            try_seq(a_opt, "fista_main")

            # Second pass with different params
            rem = deadline - time.monotonic()
            if rem > 30:
                t4b = min(rem * 0.2, 200)
                a_opt, _ = fista_optimize(a_opt, time_limit=t4b, K=150, momentum=0.94, step_scale=0.5, verbose=False)
                try_seq(a_opt, "fista_main2")

    # === Phase 5: Coordinate descent refinement ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        t5 = min(rem * 0.25, 300)
        if t5 > 20:
            print(f"  CD for {t5:.0f}s, n={n_best}")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t5, max_passes=100, verbose=True)
            try_seq(a_opt, "cd")

    # === Phase 6: Interpolate to longer and optimize ===
    print(f"\n=== Phase 6: Longer lengths (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for target_n in [n_best + 256, n_best + 512, n_best * 2]:
            if target_n > 4000:
                continue
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            t_alloc = min(rem * 0.2, 150)
            if t_alloc < 15:
                continue
            f_interp = interpolate_sequence(best_seq, target_n)
            a_opt, _ = fista_optimize(f_interp, time_limit=t_alloc, K=80, momentum=0.9, step_scale=1.0)
            try_seq(a_opt, f"fista_long_n{target_n}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Fine FISTA
        rem = deadline - time.monotonic()
        if rem > 30:
            a_opt, _ = fista_optimize(best_seq, time_limit=min(rem, 120), K=200, momentum=0.95, step_scale=0.3)
            try_seq(a_opt, "final_fista")

        # Short CD
        rem = deadline - time.monotonic()
        if rem > 15:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 60), max_passes=50)
            try_seq(a_opt, "final_cd")

        # Tiny perturbations
        rem = deadline - time.monotonic()
        for p in range(3):
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            noise = rng.standard_normal(n_best) * (0.001 + 0.001 * p)
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = fista_optimize(f_pert, time_limit=min(rem, 40), K=100, momentum=0.93, step_scale=0.5)
            try_seq(a_opt, f"perturb_{p}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(n_parent) / n_parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
