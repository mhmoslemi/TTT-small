    conv = fast_convolve(a)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using momentum-accelerated gradient
descent on a log-sum-exp smoothed objective with geometric temperature annealing,
combined with simulated annealing restarts, Nesterov-like extrapolation, and
exact subgradient polishing. Key improvements over prior approach:
- Heavy-ball / Nesterov momentum for faster convergence
- Smooth geometric temperature schedule (not piecewise)
- More frequent shorter restarts with adaptive noise
- Final exact subgradient with adaptive line search
- Optional LP-based peak reduction for the hardest peaks
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """Compute M(p) @ a where M(p)_{j,m} = p[j+m]. p has length 2n-1."""
    return np.correlate(p, a, mode='valid')


def smooth_gradient(a, T):
    """Compute smoothed gradient: 2 * H(softmax(conv/T)) * a."""
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p /= p.sum()
    grad = 2.0 * hankel_mv(p, a)
    return grad, conv


def momentum_smoothed_descent(a0, time_limit, T_init=0.01, T_end=0.00005,
                               lr_init=0.3, mom_init=0.4, mom_end=0.85):
    """
    Momentum (heavy-ball) gradient descent on smoothed objective
    with geometric temperature annealing and Nesterov extrapolation.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    a_prev = a.copy()
    best_a = a.copy()
    best_exact = fast_convolve(a).max()
    no_improve = 0

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_init * (T_end / T_init) ** progress
        mom = mom_init + (mom_end - mom_init) * min(1.0, progress * 1.5)
        lr = lr_init / (1.0 + 0.005 * it)

        # Nesterov extrapolation
        y = a + mom * (a - a_prev)

        # Gradient at extrapolated point
        grad_y, _ = smooth_gradient(y, T)

        # Update
        v = mom * v - lr * grad_y
        a_new = project_simplex(y + v)

        if not np.all(np.isfinite(a_new)):
            a_new = a
            v = np.zeros(n)
            a_prev = a
            continue

        a_prev = a
        a = a_new

        # Track best exact objective
        exact = fast_convolve(a).max()
        if exact < best_exact - 1e-15:
            best_exact = exact
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive restart within this function
        if no_improve > 500:
            rng_local = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng_local.standard_normal(n) * 0.002)
            a_prev = a.copy()
            v = np.zeros(n)
            no_improve = 0

    return best_a, best_exact


def exact_subgradient_polish(a0, time_limit, lr_init=0.05):
    """
    Polish using exact subgradient from the argmax peak with adaptive line search.
    Cycles through top peaks for more effective descent.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_exact = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)

        # Use top few peaks weighted by value for a richer subgradient
        top_k = min(5, len(conv))
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        peak_values = conv[peak_indices]
        peak_values = peak_values - peak_values.min() + 1e-10
        weights = peak_values / peak_values.sum()

        # Combined subgradient
        grad = np.zeros(n)
        for w, k in zip(weights, peak_indices):
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += w * 2.0 * a[k - j]

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Multi-step line search
        best_candidate = a
        best_val = best_exact
        found = False

        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05, lr * 0.01]:
            candidate = project_simplex(a - trial_lr * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
                found = True
                break

        if found:
            a = best_candidate
            if best_val < best_exact - 1e-15:
                best_exact = best_val
                best_a = a.copy()
            no_improve = 0
            lr = min(lr * 1.02, 0.2)
        else:
            no_improve += 1
            lr *= 0.95
            if lr < 1e-12:
                break

        # Periodic perturbation to escape
        if no_improve > 100:
            rng_local = np.random.default_rng((it * 997 + 3) % (2**31))
            a = project_simplex(best_a + rng_local.standard_normal(n) * 0.001)
            lr = lr_init * 0.5
            no_improve = 0

    return best_a, best_exact


def lp_peak_reduction(a0, M, n_vars, max_iter=50):
    """
    LP-based peak reduction: find direction that reduces the max peak.
    Solve: max 1^T d s.t. (a*d)[k] <= 0 for all k, ||d||_inf <= 1
    This gives a direction that doesn't increase any convolution value.
    """
    from scipy.optimize import linprog

    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    # LP: minimize -sum(d) subject to:
    # (a*d)[k] <= 0 for all k=0,...,2n-2
    # -1 <= d_j <= 1 for all j
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = a
    b_ub = np.zeros(2 * n - 1)
    bounds = [(-1.0, 1.0)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.success:
            return np.maximum(res.x, 0.0) - np.minimum(res.x, 0.0)  # Just return d
    except Exception:
        pass
    return None
