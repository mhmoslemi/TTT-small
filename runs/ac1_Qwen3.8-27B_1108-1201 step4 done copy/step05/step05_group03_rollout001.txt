def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

            rng = np.random.default_rng(it * 31 + 7)
            scale = min(0.005, 0.001 + 0.001 * (no_improve // 20))

def adam_optimize(f0, time_limit, verbose=False):
    """Adam on log-sum-exp smoothed objective."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    base_lr = 0.003
    
    T_start = 8.0 / n
    T_end = 0.3 / n
    top_k_start = max(100, n // 2)
    top_k_end = max(10, n // 16)
    
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1:
            break
        
        progress = min(1.0, it / max(1, int(time_limit / 0.01)))
        T = T_start * (T_end / T_start) ** progress
        top_k = max(5, int(top_k_start + (top_k_end - top_k_start) * progress))
        lr = base_lr * (1.0 - 0.3 * progress)
        
        conv = fast_convolve(f)
        
        k_actual = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = T * (pv - pv.max())
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        it1 = it + 1
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** it1)
        v_hat = v / (1 - beta2 ** it1)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        
        f_new = project_simplex(f - step)
        
        if np.max(np.abs(f_new - f)) < 1e-14:
            no_improve += 1
            if no_improve > 200:
                rng = np.random.default_rng(it * 41 + 13)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                m = np.zeros(n)
                v = np.zeros(n)
                no_improve = 0
            continue
        
        f = f_new
        new_M = fast_convolve(f).max()
        
        if new_M < best_obj - 1e-14:
            best_obj = new_M
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    Adam it={it}: obj={best_obj:.10f}, T={T:.6e}, lr={lr:.6f}")
        else:
            no_improve += 1
    
    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Cyclic coordinate descent on pairs."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)

    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1:
            break
        
        improved = False
        # Random pair order
        pairs = rng.permutation(n * 4)
        
        for pi in range(min(n * 2, len(pairs))):
            if time.monotonic() - t0 > time_limit - 1:
                break
            
            i = int(pairs[pi]) % n
            j = int(pairs[(pi + 7919) % len(pairs)]) % n
            if i == j:
                j = (j + 1) % n
            
            c = f[i] + f[j]
            if c < 1e-12:
                continue
            
            # Try several splits
            t_vals = [f[i] / c]
            if 0 < f[i] / c < 1:
                ratio = f[i] / c
                t_vals.extend([ratio + d for d in [-0.2, -0.1, -0.05, 0.05, 0.1, 0.2]])
            t_vals.extend([0.0, 1.0, 0.25, 0.5, 0.75])
            
            best_t = f[i] / c
            best_val = best_obj
            
            for t in t_vals:
                if t < -0.05 or t > 1.05:
                    continue
                t = max(0.0, min(1.0, t))
                f_trial = f.copy()
                f_trial[i] = t * c
                f_trial[j] = (1 - t) * c
                f_trial = f_trial / f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_val - 1e-13:
                    best_val = val
                    best_t = t
            
            if best_val < best_obj - 1e-12:
                f[i] = best_t * c
                f[j] = (1 - best_t) * c
                f = f / f.sum()
                best_obj = fast_convolve(f).max()
                best_f = f.copy()
                improved = True
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 10:
                break
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based Frank-Wolfe with exact quadratic line search
2. Log-sum-exp gradient descent with adaptive temperature
3. Cooperative alternation between the two methods
4. Multi-length exploration with diverse restarts
5. Random-pair coordinate descent for final polish
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_solve(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        rows = np.arange(j, j + n)
        A[rows, j] = f[j]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={"presolve": True, "primal_feasibility_tolerance": 1e-9})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def quad_line_search(f_conv, g_conv, fg_conv):
    """Find alpha in [0,1] minimizing max_k q_k(alpha) where
    q_k(a) = (1-a)^2 f[k] + 2a(1-a) fg[k] + a^2 g[k]."""
    A_q = g_conv - 2.0 * fg_conv + f_conv
    B_q = 2.0 * (fg_conv - f_conv)
    C_q = f_conv

    best_alpha = 0.0
    best_val = C_q.max()

    # Coarse grid search
    for alpha in np.linspace(0.02, 1.0, 30):
        vals = A_q * alpha * alpha + B_q * alpha + C_q
        val = vals.max()
        if val < best_val:
            best_val = val
            best_alpha = alpha

    if best_alpha < 0.01:
        return 0.0, best_val

    # Golden section refinement
    lo = max(0.0, best_alpha - 0.06)
    hi = min(1.0, best_alpha + 0.06)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    v1 = (A_q * c1 * c1 + B_q * c1 + C_q).max()
    v2 = (A_q * c2 * c2 + B_q * c2 + C_q).max()

    for _ in range(35):
        if hi - lo < 1e-12:
            break
        if v1 <= v2:
            hi = c2
            c2 = c1
            v2 = v1
            c1 = hi - gr * (hi - lo)
            v1 = (A_q * c1 * c1 + B_q * c1 + C_q).max()
        else:
            lo = c1
            c1 = c2
            v1 = v2
            c2 = lo + gr * (hi - lo)
            v2 = (A_q * c2 * c2 + B_q * c2 + C_q).max()

    alpha = (lo + hi) / 2.0
    val = (A_q * alpha * alpha + B_q * alpha + C_q).max()

    # Check endpoints
    v0 = C_q.max()
    v1_end = g_conv.max()
    if v0 < val:
        return 0.0, v0
    if v1_end < val:
        return 1.0, v1_end

    return alpha, val


def fw_optimize(f0, time_limit, verbose=False):
    """LP Frank-Wolfe with exact quadratic line search and adaptive restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    f_conv = fast_convolve(f)
    M = f_conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 2.0:
            break

        g = lp_solve(f, M, n)
        if g is None:
            g = lp_solve(f, M * 1.003, n)
            if g is None:
                g = lp_solve(f, M * 1.008, n)
                if g is None:
                    no_improve += 1
                    if no_improve > 40:
                        break
                    continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                rng = np.random.default_rng(it * 997 + 13)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                f_conv = fast_convolve(f)
                M = f_conv.max()
                no_improve = 0
            continue

        # Exact quadratic line search
        g_conv = fast_convolve(g)
        fg_conv = cross_convolve(f, g)

        alpha, predicted = quad_line_search(f_conv, g_conv, fg_conv)

        if alpha > 1e-10 and predicted < M - 1e-12:
            f = np.maximum(f + alpha * (g - f), 0.0)
            f = f / f.sum()
            f_conv = fast_convolve(f)
            M = f_conv.max()

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 5 == 0:
                    print(f"    FW it={it}: obj={best_obj:.10f}, alpha={alpha:.4f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 50:
            rng = np.random.default_rng(it * 31 + 7)
            scale = min(0.006, 0.001 + 0.001 * (no_improve // 15))
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            f_conv = fast_convolve(f)
            M = f_conv.max()
            no_improve = 0

    return best_f, best_obj


def adam_optimize(f0, time_limit, verbose=False):
    """Adam optimizer on log-sum-exp smoothed objective with simplex projection."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    base_lr = 0.004

    T_start = 10.0 / n
    T_end = 0.2 / n
    top_k_start = max(100, n // 2)
    top_k_end = max(8, n // 20)

    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1.0:
            break

        progress = min(1.0, 1.0 - (time_limit - (time.monotonic() - t0)) / time_limit)
        T = T_start * (T_end / T_start) ** progress
        top_k = max(5, int(top_k_start + (top_k_end - top_k_start) * progress))
        lr = base_lr * (1.0 - 0.4 * progress)

        conv = fast_convolve(f)

        k_actual = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = T * (pv - pv.max())
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        it1 = it + 1
        m = beta1 * m + (1.0 - beta1) * sg
        v = beta2 * v + (1.0 - beta2) * sg * sg
        m_hat = m / (1.0 - beta1 ** it1)
        v_hat = v / (1.0 - beta2 ** it1)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        f_new = project_simplex(f - step)

        if np.max(np.abs(f_new - f)) < 1e-14:
            no_improve += 1
            if no_improve > 150:
                rng = np.random.default_rng(it * 41 + 13)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                m = np.zeros(n)
                v = np.zeros(n)
                no_improve = 0
            continue

        f = f_new
        new_M = fast_convolve(f).max()

        if new_M < best_obj - 1e-14:
            best_obj = new_M
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    Adam it={it}: obj={best_obj:.10f}, T={T:.6e}, lr={lr:.6f}")
        else:
            no_improve += 1

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Random-pair coordinate descent with local optimization of each pair."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    no_improve_rounds = 0
    rnd = 0

    while time.monotonic() - t0 < time_limit:
        rnd += 1
        if time.monotonic() - t0 > time_limit - 1.0:
            break

        round_improved = False

        # Generate random pairs
        num_pairs = min(n * 3, 5000)
        indices = rng.integers(0, n, size=2 * num_pairs)
        i_arr = indices[::2]
        j_arr = indices[1::2]

        for idx in range(num_pairs):
            if time.monotonic() - t0 > time_limit - 1.0:
                break

            i = int(i_arr[idx])
            j = int(j_arr[idx])
            if i == j:
                continue

            c = f[i] + f[j]
            if c < 1e-12:
                continue

            # Try several splits
            cur_ratio = f[i] / c
            t_candidates = [cur_ratio]
            for d in [-0.2, -0.1, -0.05, -0.02, 0.02, 0.05, 0.1, 0.2]:
                t = cur_ratio + d
                if 0.0 <= t <= 1.0:
                    t_candidates.append(t)
            for t_fix in [0.0, 0.1, 0.25, 0.5, 0.75, 0.9, 1.0]:
                if abs(t_fix - cur_ratio) > 0.05:
                    t_candidates.append(t_fix)

            best_t = cur_ratio
            best_val = best_obj

            for t in t_candidates:
                t = max(0.0, min(1.0, t))
                if abs(t - cur_ratio) < 1e-8:
                    continue
                f_trial = f.copy()
                f_trial[i] = t * c
                f_trial[j] = (1.0 - t) * c
                f_trial = f_trial / f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_val - 1e-13:
                    best_val = val
                    best_t = t

            if best_val < best_obj - 1e-12:
                f[i] = best_t * c
                f[j] = (1.0 - best_t) * c
                f = f / f.sum()
                best_obj = fast_convolve(f).max()
                best_f = f.copy()
                round_improved = True

        if round_improved:
            no_improve_rounds = 0
        else:
            no_improve_rounds += 1
            if no_improve_rounds > 5:
                if verbose:
                    print(f"    CoordDescent: no improvement for {no_improve_rounds} rounds, stopping")
                break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining LP-FW with exact quadratic line search,
    Adam-SG gradient descent, random-pair coordinate descent, and multi-length
    exploration with diverse restarts.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)
    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return float('inf')
        sc = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
        return sc

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception as e:
        print(f"Parent loading error: {e}")

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: LP-FW with exact quadratic line search on parent ===
    print(f"\n=== Phase 1: LP-FW on parent (n={n_parent}) ===")
    rem = deadline - time.monotonic() - 1000.0
    t1 = min(rem, 350.0)
    if t1 > 30:
        print(f"  Time budget: {t1:.0f}s")
        a_opt, _ = fw_optimize(parent, time_limit=t1, verbose=True)
        try_seq(a_opt, "fw_p1")

    # === Phase 2: Adam-SG refinement ===
    print(f"\n=== Phase 2: Adam-SG (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent
    rem = deadline - time.monotonic() - 800.0
    t2 = min(rem, 200.0)
    if t2 > 20:
        print(f"  Time budget: {t2:.0f}s, n={len(base)}")
        a_opt, _ = adam_optimize(base, time_limit=t2, verbose=True)
        try_seq(a_opt, "adam_p2")

    # === Phase 3: LP-FW refinement on best ===
    print(f"\n=== Phase 3: LP-FW refinement (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent
    rem = deadline - time.monotonic() - 550.0
    t3 = min(rem, 250.0)
    if t3 > 20:
        print(f"  Time budget: {t3:.0f}s, n={len(base)}")
        a_opt, _ = fw_optimize(base, time_limit=t3, verbose=True)
        try_seq(a_opt, "fw_p3")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent

    lengths = [640, 768, 896, 1024]
    if n_parent in lengths:
        lengths.remove(n_parent)

    for target_n in lengths:
        rem = deadline - time.monotonic() - 300.0
        if rem < 30:
            break

        f_interp = interpolate_sequence(base, target_n)
        a_opt = None

        # FW on interpolated
        t_fw = min(rem * 0.5, 130.0)
        if t_fw > 20:
            a_opt, _ = fw_optimize(f_interp, time_limit=t_fw, verbose=False)
            try_seq(a_opt, f"fw_n{target_n}")

        # Adam on result or interpolated
        rem = deadline - time.monotonic() - 250.0
        if rem > 15:
            t_ad = min(rem * 0.5, 80.0)
            if t_ad > 10:
                start = a_opt if a_opt is not None else f_interp
                a_opt2, _ = adam_optimize(start, time_limit=t_ad, verbose=False)
                try_seq(a_opt2, f"adam_n{target_n}")

    # === Phase 5: Diverse restarts on best ===
    print(f"\n=== Phase 5: Diverse restarts (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent
    n_base = len(base)

    for p in range(4):
        rem = deadline - time.monotonic() - 180.0
        if rem < 20:
            break
        t_alloc = min(rem / max(1, 4 - p), 100.0)

        if p == 0:
            f_init = best_seq.copy()
        elif p == 1:
            rng = np.random.default_rng(200 + p * 17)
            f_init = project_simplex(best_seq + rng.standard_normal(n_base) * 0.004)
        elif p == 2:
            f_init = best_seq[::-1].copy()
            f_init = f_init / f_init.sum()
        else:
            rng = np.random.default_rng(400 + p * 23)
            f_dirichlet = rng.dirichlet(np.ones(n_base) * 0.5)
            f_init = project_simplex(0.75 * best_seq + 0.25 * f_dirichlet)

        if p % 2 == 0:
            a_opt, _ = fw_optimize(f_init, time_limit=min(t_alloc, 80.0), verbose=False)
        else:
            a_opt, _ = adam_optimize(f_init, time_limit=min(t_alloc, 80.0), verbose=False)
        try_seq(a_opt, f"diverse_{p}")

    # === Phase 6: Coordinate descent polish ===
    print(f"\n=== Phase 6: Coord Descent (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent
    rem = deadline - time.monotonic() - 100.0
    if rem > 20:
        t_cd = min(rem * 0.8, 150.0)
        print(f"  Time budget: {t_cd:.0f}s, n={len(base)}")
        a_opt, _ = coordinate_descent(base, time_limit=t_cd, verbose=True)
        try_seq(a_opt, "coord")

    # === Phase 7: Final FW on best ===
    print(f"\n=== Phase 7: Final FW (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent
    rem = deadline - time.monotonic() - 40.0
    if rem > 15:
        a_opt, _ = fw_optimize(base, time_limit=min(rem, 60.0), verbose=True)
        try_seq(a_opt, "fw_final")

    # === Phase 8: Tiny perturbations + quick optimize ===
    print(f"\n=== Phase 8: Perturbations (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent
    n_base = len(base)

    for trial in range(8):
        rem = deadline - time.monotonic() - 5.0
        if rem < 10:
            break
        rng = np.random.default_rng(700 + trial * 11)
        scale = 0.001 + 0.0005 * trial
        f_pert = project_simplex(base + rng.standard_normal(n_base) * scale)
        # Alternate between FW and Adam
        if trial % 2 == 0:
            a_opt, _ = fw_optimize(f_pert, time_limit=min(rem, 15.0), verbose=False)
        else:
            a_opt, _ = adam_optimize(f_pert, time_limit=min(rem, 15.0), verbose=False)
        try_seq(a_opt, f"perturb_{trial}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
