def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """Compute sum_k p[k] * a[k-j] for all j (gradient of smooth objective w.r.t. a)."""
    return np.correlate(p, a, mode='valid')


def smooth_objective_and_grad(a, T):
    """Compute smooth objective T*logsumexp(conv/T) and gradient w.r.t. a."""
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    logsumexp_val = scaled.max() + np.log(exp_scaled.sum())
    F = T * logsumexp_val
    p = exp_scaled / exp_scaled.sum()
    grad = 2.0 * hankel_mv(p, a)
    return F, grad


def nesterov_smooth(a0, time_limit, T_init=0.001, T_final=0.00005, n_anneal=500):
    """
    Nesterov-accelerated gradient descent on smooth objective with temperature annealing.
    Works on the simplex via projection.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    y = a.copy()  # Nesterov momentum point
    beta = 1.0   # momentum parameter (starts at 1 for Nesterov)
    T = T_init

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    total_steps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.05:
            break

        total_steps += 1

        # Anneal temperature
        progress = min(1.0, total_steps / max(1, n_anneal))
        T = T_init * (T_final / T_init) ** progress

        # Nesterov momentum update
        if total_steps > 1:
            beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
            beta = beta_new

        # Compute gradient at y
        F_y, grad_y = smooth_objective_and_grad(y, T)

        # Step size: use 1/L approximation where L ~ 2*conv_max/T
        conv_y = fast_convolve(y)
        L = 2.0 * conv_y.max() / T + 1e-10
        lr = 1.0 / L
        lr = min(lr, 0.1)  # Cap learning rate

        # Nesterov step
        x_new = project_simplex(y - lr * grad_y)

        # Update y for next iteration
        gamma = (beta - 1.0) / beta_new
        y = x_new + gamma * (x_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        a = x_new

        # Evaluate true objective
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Reset momentum if stuck
        if no_improve > 100:
            y = best_a.copy()
            beta = 1.0
            no_improve = 0

        # Occasional restart with perturbation
        if no_improve > 300:
            rng = np.random.default_rng(total_steps * 7 + 13)
            scale = 0.002 + 0.005 * min(1.0, (no_improve - 300) / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            y = a.copy()
            beta = 1.0
            T = T_init * 0.5
            no_improve = 0

    return best_a, best_obj


def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        # Solve LP: max sum(g) s.t. H(f)g <= M*m_exp, g >= 0
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M * m_exp)
        bounds = [(0, None)] * n

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': max(1.0, remaining * 0.3)})
            if res.success:
                g = np.maximum(res.x, 0.0)
            else:
                g = None
        except Exception:
            g = None

        if g is None or g.sum() < 1e-15:
            m_exp = min(m_exp * 1.3, 4.0)
            continue

        g = g / g.sum()

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 50:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Refine line search
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)

        if no_improve > 80:
            rng = np.random.default_rng(it * 104729 + 3)
            scale = 0.002 + 0.004 * min(1.0, no_improve / 200.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def subgradient_polish(a0, time_limit, top_k=30, lr_init=0.001, mom=0.90):
    """Multi-peak subgradient with momentum for final polish."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)

        # Top-k peaks
        k = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k)[-k:]
        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 1.5
        ws = weights.sum()
        if ws > 0:
            weights /= ws

        # Subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            kk = int(peaks[i])
            js = max(0, kk - n + 1)
            je = min(n - 1, kk)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * 2.0 * a[kk - idx_arr]

        v = mom * v - lr * sg
        a_new = project_simplex(a + v)

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9998
        mom = min(0.97, mom + 0.00005)

        if no_improve > 200:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            noise = 0.001 + 0.003 * min(1.0, (no_improve - 200) / 400.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            lr = lr_init * 0.5
            no_improve = 0

    return best_a, best_obj


def coordinate_refine(a0, time_limit):
    """
    Coordinate descent: for each of the top peaks, try to reduce it by
    redistributing mass among the variables that contribute to it.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    sweep = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        peak_k = int(np.argmax(conv))
        peak_val = conv[peak_k]

        # Variables affecting this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        affected = np.arange(js, je + 1)

        if len(affected) == 0:
            break

        # Try to reduce the peak by moving mass from high-contributing to low-contributing vars
        # Contribution of a[j] to peak is a[peak_k - j]
        contributions = a[peak_k - affected]

        # Find the variable with highest contribution and try to reduce it
        max_idx = affected[np.argmax(contributions)]
        min_idx = affected[np.argmin(contributions)]

        if max_idx == min_idx:
            break

        # Try shifting mass from max_idx to min_idx
        improved = False
        for frac in [0.01, 0.03, 0.05, 0.1, 0.2, 0.3]:
            delta = a[max_idx] * frac
            if delta <= 0:
                continue
            a_trial = a.copy()
            a_trial[max_idx] -= delta
            a_trial[min_idx] += delta
            if a_trial[max_idx] < 0:
                continue
            ct = fast_convolve(a_trial)
            if ct.max() < best_obj - 1e-14:
                a = a_trial
                best_obj = ct.max()
                best_a = a.copy()
                improved = True
                break

        if not improved:
            # Try random perturbation in affected set
            rng = np.random.default_rng((sweep * 12345 + peak_k) % (2**31))
            for _ in range(3):
                pert = rng.standard_normal(n) * 0.001
                for j in range(n):
                    if j not in affected:
                        pert[j] = 0
                a_trial = project_simplex(a + pert)
                ct = fast_convolve(a_trial)
                if ct.max() < best_obj - 1e-14:
                    a = a_trial
                    best_obj = ct.max()
                    best_a = a.copy()
                    improved = True
                    break

        if not improved and sweep % 100 == 0:
            # Broader restart
            rng = np.random.default_rng((sweep * 67891) % (2**31))
            a_trial = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            ct = fast_convolve(a_trial)
            if ct.max() < best_obj - 1e-14:
                a = a_trial
                best_obj = ct.max()
                best_a = a.copy()
            elif sweep > 500:
                break

    return best_a, best_obj


def dual_informed_smooth(a0, time_limit, T_init=0.002):
    """
    Smooth gradient descent using the LP dual as the peak weighting.
    Solves the dual of the LP at each step to get optimal peak weights,
    then does gradient steps in that direction.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = 0.01
    T = T_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.1:
            break

        conv = fast_convolve(a)
        M = conv.max()

        # Solve dual: min M*sum(y) s.t. H(a)^T y >= 1, y >= 0
        # This is equivalent to: min sum(y) s.t. H(a)^T y >= 1/M, y >= 0
        # Dual of the primal LP
        # Primal: max 1^T g s.t. H(a)g <= M, g >= 0
        # Dual:   min M*1^T y s.t. H(a)^T y >= 1, y >= 0
        # H(a)^T y >= 1 means for each j: sum_k H(a)_{k,j} y_k >= 1
        # H(a)_{k,j} = a[k-j] (with bounds)
        # So: sum_k a[|k-j|] y_k >= 1 (approximately, with appropriate indexing)

        # This is large (1535 constraints for the dual of a 768-var LP)
        # Instead, use the softmax approximation as a cheaper substitute
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient
        grad = 2.0 * hankel_mv(p, a)

        # Backtracking line search
        best_candidate = a
        best_val = M
        for trial in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            candidate = project_simplex(a - trial * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        a = best_candidate
        obj = best_val

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            lr = min(lr * 1.02, 0.05)
        else:
            no_improve += 1
            lr *= 0.98

        # Adapt temperature
        if no_improve > 50:
            T *= 0.9
            if T < 0.0001:
                T = 0.0001
        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise_scale = 0.002 + 0.004 * min(1.0, (no_improve - 200) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            T = T_init
            lr = 0.01
            no_improve = 0

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample a sequence to a different length using interpolation."""
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search combining FW-LP,
    Nesterov smooth gradient, dual-informed gradient, and coordinate polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent_initial")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768

    n_parent = len(parent)
    print(f"Starting with n={n_parent}, best={best_score:.8f}")

    # === Phase 1: Quick FW-LP to establish baseline (120s) ===
    print("\n=== Phase 1: FW-LP Baseline ===")
    t_phase = time.monotonic()
    t_alloc = min(120, deadline - time.monotonic() - 1400)
    if t_alloc > 10:
        a_fw, obj_fw = fw_lp_optimize(parent, time_limit=t_alloc)
        try_seq(a_fw, "fw_baseline")

    print(f"  Phase 1 done: best={best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    if best_seq is None:
        best_seq = parent

    n_best = len(best_seq)

    # === Phase 2: Nesterov smooth gradient (400s) ===
    print("\n=== Phase 2: Nesterov Smooth Gradient ===")
    t_phase = time.monotonic()
    t_alloc = min(400, deadline - time.monotonic() - 1000)
    if t_alloc > 20:
        a_nest, _ = nesterov_smooth(best_seq, time_limit=t_alloc,
                                    T_init=0.002, T_final=0.0001, n_anneal=300)
        try_seq(a_nest, "nesterov_1")

    # Second pass with different params
    t_alloc = min(200, deadline - time.monotonic() - 800)
    if t_alloc > 20:
        a_nest2, _ = nesterov_smooth(best_seq, time_limit=t_alloc,
                                     T_init=0.0005, T_final=0.00002, n_anneal=200)
        try_seq(a_nest2, "nesterov_2")

    print(f"  Phase 2 done: best={best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Dual-informed smooth gradient (300s) ===
    print("\n=== Phase 3: Dual-Informed Gradient ===")
    t_phase = time.monotonic()
    t_alloc = min(300, deadline - time.monotonic() - 700)
    if t_alloc > 20:
        a_dual, _ = dual_informed_smooth(best_seq, time_limit=t_alloc, T_init=0.001)
        try_seq(a_dual, "dual_smooth")

    t_alloc = min(150, deadline - time.monotonic() - 500)
    if t_alloc > 15:
        a_dual2, _ = dual_informed_smooth(best_seq, time_limit=t_alloc, T_init=0.0003)
        try_seq(a_dual2, "dual_smooth_fine")

    print(f"  Phase 3 done: best={best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: FW-LP deep refinement (300s) ===
    print("\n=== Phase 4: Deep FW-LP ===")
    t_phase = time.monotonic()
    t_alloc = min(300, deadline - time.monotonic() - 400)
    if t_alloc > 20:
        a_fw2, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
        try_seq(a_fw2, "fw_deep")

    # Perturbed restart
    t_alloc = min(100, deadline - time.monotonic() - 250)
    if t_alloc > 15:
        for scale in [0.005, 0.015]:
            t_alloc = min(50, deadline - time.monotonic() - 150)
            if t_alloc < 10:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            a_p, _ = fw_lp_optimize(f_pert, time_limit=t_alloc)
            try_seq(a_p, f"fw_pert_{scale}")

    print(f"  Phase 4 done: best={best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Subgradient polish (200s) ===
    print("\n=== Phase 5: Subgradient Polish ===")
    t_phase = time.monotonic()
    t_alloc = min(200, deadline - time.monotonic() - 250)
    if t_alloc > 15:
        a_sg, _ = subgradient_polish(best_seq, time_limit=t_alloc, top_k=50, lr_init=0.001, mom=0.92)
        try_seq(a_sg, "sg_polish")

    t_alloc = min(100, deadline - time.monotonic() - 150)
    if t_alloc > 10:
        a_sg2, _ = subgradient_polish(best_seq, time_limit=t_alloc, top_k=15, lr_init=0.0005, mom=0.95)
        try_seq(a_sg2, "sg_fine")

    print(f"  Phase 5 done: best={best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 6: Coordinate refine (150s) ===
    print("\n=== Phase 6: Coordinate Refine ===")
    t_phase = time.monotonic()
    t_alloc = min(150, deadline - time.monotonic() - 150)
    if t_alloc > 10:
        a_coord, _ = coordinate_refine(best_seq, time_limit=t_alloc)
        try_seq(a_coord, "coord")

    print(f"  Phase 6 done: best={best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 7: Multi-start exploration (200s) ===
    print("\n=== Phase 7: Multi-Start ===")
    t_phase = time.monotonic()

    n_starts = 5
    t_per_start = max(10, min(40, (deadline - time.monotonic() - 100) / n_starts))

    for i in range(n_starts):
        remaining = deadline - time.monotonic() - 80
        if remaining < 10:
            break

        # Different perturbation strategies
        if i == 0:
            # Gaussian perturbation
            pert = rng.standard_normal(n_best) * 0.01
        elif i == 1:
            # Structured: alternating bump
            pert = np.zeros(n_best)
            block = n_best // 8
            shift = (i * block + rng.integers(0, block)) % n_best
            pert[shift:shift + block] = 0.02
            pert = pert - pert.sum() / n_best  # Keep roughly zero-sum
        elif i == 2:
            # Edge-focused perturbation
            x = np.arange(n_best)
            edge_w = np.exp(-x / (n_best * 0.1)) + np.exp(-(n_best - 1 - x) / (n_best * 0.1))
            pert = edge_w * rng.standard_normal(n_best) * 0.02
        elif i == 3:
            # Periodic perturbation
            freq = rng.integers(3, 10)
            pert = np.sin(2 * np.pi * freq * np.arange(n_best) / n_best) * 0.01
        else:
            # Larger random
            pert = rng.standard_normal(n_best) * 0.02

        a_pert = project_simplex(best_seq + pert)

        # Quick Nesterov on perturbed
        t_sub = min(t_per_start, deadline - time.monotonic() - 60)
        if t_sub > 5:
            a_opt, _ = nesterov_smooth(a_pert, time_limit=t_sub,
                                       T_init=0.001, T_final=0.00005, n_anneal=100)
            try_seq(a_opt, f"multi_{i}")

    print(f"  Phase 7 done: best={best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 8: Try different lengths (150s) ===
    print("\n=== Phase 8: Length Exploration ===")
    t_phase = time.monotonic()

    candidate_ns = []
    for n_try in [n_best // 2, n_best * 2, 384, 1024, 1536]:
        if 128 <= n_try <= 2048 and n_try != n_best and n_try not in candidate_ns:
            candidate_ns.append(n_try)

    for n_try in candidate_ns[:3]:
        remaining = deadline - time.monotonic() - 60
        if remaining < 15:
            break

        a_resampled = resample_sequence(best_seq, n_try)
        t_sub = min(40, remaining - 30)
        if t_sub > 5:
            a_opt, _ = nesterov_smooth(a_resampled, time_limit=t_sub,
                                       T_init=0.001, T_final=0.0001, n_anneal=100)
            try_seq(a_opt, f"n{n_try}_nesterov")

    print(f"  Phase 8 done: best={best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 9: Final fine polish (remaining time) ===
    print("\n=== Phase 9: Final Polish ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic() - 5

    if remaining > 10:
        # Very fine Nesterov
        t_sub = min(remaining * 0.4, 60)
        a_fine, _ = nesterov_smooth(best_seq, time_limit=t_sub,
                                    T_init=0.0002, T_final=0.00001, n_anneal=200)
        try_seq(a_fine, "final_nest")

    if deadline - time.monotonic() > 10:
        # Subgradient with very fine steps
        t_sub = min((deadline - time.monotonic()) * 0.4, 40)
        a_sg_final, _ = subgradient_polish(best_seq, time_limit=t_sub,
                                           top_k=10, lr_init=0.0003, mom=0.96)
        try_seq(a_sg_final, "final_sg")

    if deadline - time.monotonic() > 5:
        # Coordinate on the very best
        t_sub = min(deadline - time.monotonic() - 2, 30)
        a_c_final, _ = coordinate_refine(best_seq, time_limit=t_sub)
        try_seq(a_c_final, "final_coord")

    print(f"  Phase 9 done: best={best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
