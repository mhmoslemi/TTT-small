"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using multi-resolution p-norm
gradient optimization with diverse initialization and peak-focused refinement.
Key differences from previous approach:
1. p-norm smoothing (instead of LSE) for more stable optimization at high sharpness
2. Multi-resolution warm-starting (optimize at small n first, then upscale)
3. Nesterov-accelerated projected gradient
4. Diverse initialization pool with competition
5. Coordinate descent polish at the end
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex (sum=1, elements>=0)."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def p_norm_and_grad(a, p):
    """Compute p-norm of convolution and its gradient w.r.t. a.
    a should be normalized (sum=1, non-negative).
    Returns (p_norm_value, gradient).
    """
    n = len(a)
    conv = fast_convolve(a)
    if p == 2:
        # Special case: L2 norm
        # ||conv||_2, gradient = 2*conv cross-correlated with a (normalized)
        val = np.sqrt(np.sum(conv ** 2))
        # grad_i = (1/val) * 2 * sum_j a_j * conv_{i+j}
        # = (2/val) * crosscorr(a, conv)_i
        N = 1 << (2 * n - 1 - 1).bit_length()
        A = np.fft.rfft(a, N)
        B = np.fft.rfft(conv, N)
        cross = np.fft.irfft(np.conj(A) * B, N)[:n]
        grad = (2.0 / val) * cross if val > 1e-30 else np.zeros(n)
        return val, grad
    else:
        conv_clipped = np.maximum(conv, 1e-30)
        conv_p = conv_clipped ** p
        conv_p_nm1 = conv_clipped ** (p - 1)
        total = np.sum(conv_p)
        val = total ** (1.0 / p)
        scale = total ** (1.0 / p - 1.0)
        # grad_i = 2 * scale * sum_j a_j * conv_{i+j}^{p-1}
        N = 1 << (2 * n - 1 - 1).bit_length()
        A = np.fft.rfft(a, N)
        C = np.fft.rfft(conv_p_nm1, N)
        cross = np.fft.irfft(np.conj(A) * C, N)[:n]
        grad = 2.0 * scale * cross
        return val, grad


def nesterov_pnorm(a0, p, time_limit, lr=None, momentum=0.9):
    """Optimize p-norm of convolution using Nesterov-accelerated projected gradient."""
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    if lr is None:
        lr = 1.0 / (n * 0.5)  # initial learning rate
    
    best_obj, _, _ = p_norm_and_grad(f, p)
    best_f = f.copy()
    
    # Nesterov momentum
    y = f.copy()
    v = np.zeros(n)
    t = 1.0
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Nesterov look-ahead
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t_new - 1.0) / t_new
        t = t_new
        
        obj, grad = p_norm_and_grad(y, p)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 100
            break
        grad_unit = grad / grad_norm
        
        v = beta * v - lr * grad_unit
        y_new = project_simplex(y + v)
        
        # Check improvement
        obj_new = fast_convolve(y_new)
        obj_new = np.sum(np.maximum(obj_new, 0)**p) ** (1.0/p)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = y_new.copy()
            f = y_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                lr *= 0.95
            if no_improve > 100:
                # Restart
                rng = np.random.default_rng(no_improve * 7 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                y = f.copy()
                v = np.zeros(n)
                lr = 1.0 / (n * 0.5)
                t = 1.0
                no_improve = 0
    
    return best_f, best_obj


def l2_optimize(a0, time_limit, lr=None, momentum=0.85):
    """Optimize L2 norm of convolution (p=2 case) with momentum."""
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    if lr is None:
        lr = 2.0 / n
    
    conv = fast_convolve(f)
    best_obj = np.sqrt(np.sum(conv ** 2))
    best_f = f.copy()
    
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        val = np.sqrt(np.sum(conv ** 2))
        # grad = (2/val) * crosscorr(a, conv)
        N = 1 << (2 * n - 1 - 1).bit_length()
        A = np.fft.rfft(f, N)
        B = np.fft.rfft(conv, N)
        cross = np.fft.irfft(np.conj(A) * B, N)[:n]
        grad = (2.0 / val) * cross
        
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        
        conv_new = fast_convolve(f_new)
        obj_new = np.sqrt(np.sum(conv_new ** 2))
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.93
            if no_improve > 150:
                f = best_f.copy()
                v = np.zeros(n)
                lr = 2.0 / n
                no_improve = 0
    
    return best_f, best_obj


def peak_focused_grad(a0, time_limit, n_peaks=10, lr_init=0.001, decay=0.98):
    """Gradient descent focusing on the top n_peaks convolution values."""
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv_len = 2 * n - 1
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    v = np.zeros(n)
    momentum = 0.92
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        
        # Select top peaks
        k = min(n_peaks, conv_len)
        peak_idx = np.argpartition(conv, -k)[-k:]
        peak_vals = conv[peak_idx]
        
        # Weight: exponential of (peak_val - min_peak) / (max_peak - min_peak + eps)
        pv_range = peak_vals.max() - peak_vals.min()
        if pv_range < 1e-15:
            w = np.ones(k) / k
        else:
            w = np.exp((peak_vals - peak_vals.min()) / (pv_range))
            w /= w.sum()
        
        # Compute weighted gradient
        # grad_i = sum_{j in peaks} w_j * 2 * a_{j-i} (for valid j-i)
        # = 2 * crosscorr_a(weighted_conv)
        # where weighted_conv has w_j at peak positions
        
        w_conv = np.zeros(conv_len)
        w_conv[peak_idx] = 2.0 * w
        
        N = 1 << (conv_len - 1).bit_length()
        A = np.fft.rfft(f, N)
        W = np.fft.rfft(w_conv, N)
        cross = np.fft.irfft(np.conj(A) * W, N)[:n]
        grad = cross
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 200
            break
        grad /= grad_norm
        
        v = momentum * v - cur_lr * grad
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                cur_lr *= decay
            if no_improve > 300:
                rng = np.random.default_rng(no_improve * 13)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.0005)
                v = np.zeros(n)
                cur_lr = lr_init
                no_improve = 0
    
    return best_f, best_obj


def coordinate_descent_polish(a0, time_limit, window=5):
    """Local polish by trying small changes to groups of adjacent elements."""
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        # Pick a random position
        i = rng.integers(0, n - window)
        # Try a small perturbation
        delta = rng.standard_normal(window) * 0.001
        f_new = f.copy()
        f_new[i:i+window] += delta
        f_new = np.maximum(f_new, 0)
        f_new /= f_new.sum()
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
    
    return best_f, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex (sum=1, elements>=0)."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def cross_corr_ab(a, b):
    """Compute cross-correlation: result[m] = sum_j a[j]*b[m+j] for m=0..len(a)-1.
    a has length n, b has length >= 2n-1. Uses FFT."""
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    if N < len(b):
        N = 1 << (len(b)).bit_length()
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    result = np.fft.irfft(np.conj(A) * B, N)[:n]
    return result

        v = beta * v - lr * g_unit
        y_new = project_simplex(y + v)
        
        obj_new = fast_convolve(y_new)
        if p == 2:
            obj_new = np.sqrt(np.sum(np.maximum(obj_new, 0)**2))

                y = f.copy()
                v = np.zeros(n)
                t_param = 1.0
                lr = lr_init
                no_improve = 0
    
    return best_f, best_obj


def l2_optimize(a0, time_limit, lr_init=None, momentum=0.88):
    """Optimize L2 norm of convolution with heavy-ball momentum."""
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    if lr_init is None:
        lr_init = 3.0 / n
    
    best_obj, _ = pnorm_obj_grad(f, 2)
    best_f = f.copy()
    
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = lr_init
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        _, grad = pnorm_obj_grad(f, 2)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        g_unit = grad / gnorm
        
        v = momentum * v - lr * g_unit
        f_new = project_simplex(f + v)
        
        conv_new = fast_convolve(f_new)
        obj_new = np.sqrt(np.sum(np.maximum(conv_new, 0)**2))
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.93
            if no_improve > 150:
                rng = np.random.default_rng(it * 31)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                v = np.zeros(n)
                lr = lr_init
                no_improve = 0
    
    return best_f, best_obj
