Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using exact coordinate descent

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact coordinate descent with binary search (globally optimal per coordinate)
2. LP-based Frank-Wolfe for complementary directions
3. Multi-length exploration with diverse restarts
4. Spectral initialization using chirp/Alpager windows

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def exact_coord_descent(f, time_limit=60.0, verbose=False, initial_passes=50):
    """
    Exact coordinate descent: for each coordinate i, find the optimal x_i
    that minimizes max(conv(a,a)) using binary search on the threshold T.
    
    Key insight: if the current value old <= ub(T*), it's already optimal.
    Otherwise, set x = ub(T*).
    """
    f = np.maximum(f.astype(np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    
    n = len(f)
    size = 2 * n - 1
    conv = fast_convolve(f)
    best_M = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    k_range = np.arange(size)
    
    pass_count = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        pass_improved = False
        
        # Choose order
        if pass_count % 4 == 1:
            order = rng.permutation(n)
        elif pass_count % 4 == 2:
            order = np.arange(n)
        elif pass_count % 4 == 3:
            order = np.arange(n)[::-1]
        else:
            # Block order
            order = np.concatenate([np.arange(i, i + max(1, n//8)) for i in range(0, n, max(1, n//8))])
            order = order[:n]
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            old = f[i]
            
            # Compute base and coef1 (vectorized)
            j_range = k_range - i
            valid = (j_range >= 0) & (j_range < n)
            
            if not valid.any():
                continue
            
            base = conv.copy()
            coef1 = np.zeros(size)
            
            # Linear terms: where valid and k != 2*i
            mask_lin = valid & (k_range != 2 * i)
            if mask_lin.any():
                base[mask_lin] -= 2.0 * old * f[j_range[mask_lin]]
                coef1[mask_lin] = 2.0 * f[j_range[mask_lin]]
            
            # Quadratic term at k = 2*i
            quad_idx = 2 * i
            if 0 <= quad_idx < size:
                base[quad_idx] -= old * old
            
            # Lower bound on T
            lb = base[quad_idx]  # T >= base[quad_idx] always
            # Also T >= base[k] for k where coef1[k] == 0 and k != quad_idx
            mask_static = ~valid  # k where no linear term
            if mask_static.any():
                lb = max(lb, base[mask_static].max())
            
            # Upper bound
            ub_T = conv.max() + old * old + 1.0
            
            # If lb >= ub_T, current is already optimal (or close)
            if lb >= ub_T - 1e-12:
                continue
            
            # Binary search
            lo = lb
            hi = ub_T
            for _ in range(50):
                mid = (lo + hi) / 2.0
                if mid < base[quad_idx]:
                    lo = mid
                    continue
                # Upper bound on x from quadratic
                x_ub = np.sqrt(mid - base[quad_idx])
                # Upper bound on x from linear constraints
                mask_c1 = coef1 > 0
                if mask_c1.any():
                    x_ub = min(x_ub, ((mid - base[mask_c1]) / coef1[mask_c1]).min())
                
                if x_ub >= 0:
                    hi = mid
                else:
                    lo = mid
            
            T_star = hi
            
            # Compute x_ub at T_star
            x_ub = np.sqrt(T_star - base[quad_idx])
            mask_c1 = coef1 > 0
            if mask_c1.any():
                x_ub = min(x_ub, ((T_star - base[mask_c1]) / coef1[mask_c1]).min())
            x_ub = max(0.0, x_ub)
            
            # If old <= x_ub, current value is already optimal
            if old <= x_ub + 1e-13:
                # Check if we can improve by going to x_ub (edge case)
                continue
            
            # Set x = x_ub
            new_x = x_ub
            
            # Update conv incrementally
            if mask_lin.any():
                conv[mask_lin] += 2.0 * (new_x - old) * f[j_range[mask_lin]]
            if 0 <= quad_idx < size:
                conv[quad_idx] += new_x * new_x - old * old
            
            f[i] = new_x
            
            new_M = conv.max()
            if new_M < best_M - 1e-14:
                best_M = new_M
                best_f = f.copy()
                pass_improved = True
            elif new_M > best_M + 1e-10:
                # Shouldn't happen, but safety check
                f[i] = old
                # Revert conv
                if mask_lin.any():
                    conv[mask_lin] -= 2.0 * (new_x - old) * f[j_range[mask_lin]]
                if 0 <= quad_idx < size:
                    conv[quad_idx] -= new_x * new_x - old * old
        
        if verbose and pass_count % 20 == 0:
            print(f"  ExactCD pass {pass_count}: M={best_M:.12f}, n={n}")
        
        if not pass_improved:
            no_improve += 1
            if no_improve > 5:
                # Random perturb to escape local minimum
                if no_improve % 10 == 0:
                    perturb_size = max(1, n // 20)
                    idx = rng.choice(n, size=perturb_size, replace=False)
                    f[idx] += rng.standard_normal(perturb_size) * 0.02
                    f = np.maximum(f, 0)
                    ss = f.sum()
                    if ss > 1e-15:
                        f /= ss
                    conv = fast_convolve(f)
                    no_improve = 0
                else:
                    break
        else:
            no_improve = 0
    
    return best_f, best_M


def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        rows = np.arange(j, j + n)
        A_ub[rows, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0)
    return None


def lp_fw_optimize(f0, time_limit=60.0, verbose=False):
    """LP-based Frank-Wolfe with line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 2:
            break
        
        g = lp_direction(f, M * (1 + 1e-6), n)
        if g is None:
            g = lp_direction(f, M * 1.01, n)
            if g is None:
                no_improve += 1
                if no_improve > 50:
                    break
                continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 50:
                break
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft /= st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f = np.maximum(f + best_alpha * (g - f), 0)
            f /= f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 80:
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0
    
    return best_f, best_obj


def adam_smoothed(f0, time_limit=60.0, verbose=False):
    """Adam on log-sum-exp smoothed objective."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    base_lr = 0.003
    
    T_start = 3.0 / n
    T_end = 0.3 / n
    
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        progress = min(1.0, it / max(1, int(time_limit / 0.01)))
        T = T_start * (T_end / T_start) ** progress
        lr = base_lr * (1.0 - 0.3 * progress)
        
        conv = fast_convolve(f)
        top_k = max(10, n // 4)
        k_actual = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = T * (pv - pv.max())
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for ii in range(len(peaks)):
            if w[ii] < 1e-14:
                continue
            k = int(peaks[ii])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[ii] * f[k - idx]
        
        it1 = it + 1
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** it1)
        v_hat = v / (1 - beta2 ** it1)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        
        f_new = project_simplex(f - step)
        if np.max(np.abs(f_new - f)) < 1e-15:
            no_improve += 1
            if no_improve > 200:
                rng = np.random.default_rng(it * 41)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                m, v = np.zeros(n), np.zeros(n)
                no_improve = 0
            continue
        
        f = f_new
        new_M = fast_convolve(f).max()
        if new_M < best_obj - 1e-14:
            best_obj = new_M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def chirp_init(n, alpha=0.5, seed=0):
    """Initialize with a chirp-like window for diversity."""
    x = np.linspace(0, 1, n)
    rng = np.random.default_rng(seed)
    
    # Various window shapes
    style = seed % 5
    if style == 0:
        # Gaussian window
        f = np.exp(-alpha * (x - 0.5) ** 2 * 4)
    elif style == 1:
        # Raised cosine
        f = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif style == 2:
        # Uniform with slight taper
        f = np.ones(n)
        taper = max(1, n // 20)
        f[:taper] *= np.linspace(0.5, 1, taper)
        f[-taper:] *= np.linspace(1, 0.5, taper)
    elif style == 3:
        # Chirp
        f = np.abs(np.sin(np.pi * x * (1 + alpha * 20 * x)))
    else:
        # Random smooth
        f = np.ones(n)
        f += rng.standard_normal(n) * 0.1
        f = np.maximum(f, 0)
    
    f = np.maximum(f, 0)
    ss = f.sum()
    if ss > 1e-15:
        f /= ss
    return f

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast exact coordinate descent with binary search (globally optimal per coordinate)
2. LP-based Frank-Wolfe for complementary directions
3. Multi-length exploration with diverse spectral initializations
4. Adam smoothed optimization for final polish

Key innovation: The exact coordinate descent computes the globally optimal
update for each coordinate via binary search on the convolution threshold,
avoiding the local minima traps of gradient-based methods.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def exact_coord_descent(f0, time_limit=60.0, verbose=False, seed=42):
    """
    Fast exact coordinate descent: for each coordinate i, find the optimal x_i
    that minimizes max(conv(a,a)) using binary search on the threshold T.
    
    Key optimization: the support of coordinate i in the convolution is always
    exactly n positions (k=i to k=i+n-1), and f[k-i] ranges over all of f.
    
    If old <= ub(T*), the current value is already optimal (no cycling).
    """
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s

    n = len(f)
    size = 2 * n - 1
    conv = fast_convolve(f)
    best_M = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    arange_n = np.arange(n)

    pass_count = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        pass_improved = False

        if pass_count % 5 == 1:
            order = rng.permutation(n)
        elif pass_count % 5 == 2:
            order = np.arange(n)
        elif pass_count % 5 == 3:
            order = np.arange(n)[::-1]
        else:
            order = np.arange(n)
            # Stride-based order for better mixing
            stride = max(1, n // 16)
            order = np.sort(order[stride::1])

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            old = f[i]
            if old < 1e-15:
                continue

            # Support: k in [i, i+n-1], j = k-i in [0, n-1]
            conv_slice = conv[i:i + n]  # shape (n,)
            quad_local = i  # local index of k=2*i within the slice

            # Lower bound from quadratic constraint
            A = conv[2 * i] - old * old

            # Static constraints (positions where f[j]==0, j != quad_local)
            f_pos = f  # f_vals is always the full f array
            mask_static = (arange_n != quad_local) & (f_pos == 0)
            B = 0.0
            if mask_static.any():
                B = max(B, conv_slice[mask_static].max())

            # Outside support: k < i or k >= i+n
            if i > 0:
                B = max(B, conv[:i].max())
            if i + n < size:
                B = max(B, conv[i + n:].max())

            # Linear constraint lower bound
            mask_lin = (arange_n != quad_local) & (f_pos > 0)
            C = 0.0
            if mask_lin.any():
                C = (conv_slice[mask_lin] - 2.0 * old * f_pos[mask_lin]).max()

            T_lb = max(A, B, C, 0.0)
            T_ub = max(conv.max() + old * old, T_lb + 1.0)

            if T_lb >= T_ub - 1e-13:
                continue

            # Binary search on T
            lo = T_lb
            hi = T_ub
            for _ in range(45):
                mid = (lo + hi) * 0.5
                if mid < A:
                    lo = mid
                    continue
                x_ub = np.sqrt(mid - A)
                if mask_lin.any():
                    vals = old + (mid - conv_slice[mask_lin]) / (2.0 * f_pos[mask_lin])
                    x_ub = min(x_ub, vals.min())
                if x_ub >= 0:
                    hi = mid
                else:
                    lo = mid

            T_star = hi
            # Compute x upper bound at T_star
            x_ub = np.sqrt(max(0.0, T_star - A))
            if mask_lin.any():
                vals = old + (T_star - conv_slice[mask_lin]) / (2.0 * f_pos[mask_lin])
                x_ub = min(x_ub, vals.min())
            x_ub = max(0.0, x_ub)

            # Key: if old <= x_ub, current value is already optimal
            if old <= x_ub + 1e-12:
                continue

            new_x = x_ub

            # Update conv incrementally
            # For all j in [0,n-1]: conv[i+j] += 2*(new_x - old)*f[j]
            conv[i:i + n] += 2.0 * (new_x - old) * f_pos
            # Correct the quad position: need new_x^2 - old^2 instead of 2*(new_x-old)*old
            quad_correction = (new_x - old) ** 2
            conv[2 * i] += quad_correction

            f[i] = new_x
            new_M = conv.max()

            if new_M < best_M - 1e-14:
                best_M = new_M
                best_f = f.copy()
                pass_improved = True

        # Periodic recompute to avoid floating point drift
        if pass_count % 15 == 0:
            conv = fast_convolve(f)
            if conv.max() < best_M - 1e-14:
                best_M = conv.max()
                best_f = f.copy()

        if verbose and pass_count % 25 == 0:

            print(f"  ExactCD pass {pass_count}: M={best_M:.12f}, score~{score:.6f}")

        if not pass_improved:
            no_improve += 1
            if no_improve > 4:
                if no_improve % 8 == 0:
                    # Random perturbation to escape
                    n_perturb = max(3, n // 12)
                    idx = rng.choice(n, size=n_perturb, replace=False)
                    f[idx] += rng.standard_normal(n_perturb) * 0.02
                    f = np.maximum(f, 0)
                    ss = f.sum()
                    if ss > 1e-15:
                        f /= ss
                    conv = fast_convolve(f)
                    no_improve = 0
                elif no_improve > 12:
                    break
        else:
            no_improve = 0

    return best_f, best_M


def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        rows = np.arange(j, j + n)
        A_ub[rows, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0)
    return None


def lp_fw_optimize(f0, time_limit=60.0, verbose=False):
    """LP-based Frank-Wolfe with line search and restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 3:
            break

        g = lp_direction(f, M * (1 + 1e-6), n)
        if g is None:
            g = lp_direction(f, M * 1.01, n)
            if g is None:
                no_improve += 1
                if no_improve > 40:
                    break
                continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 40:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft /= st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = np.maximum(f + best_alpha * (g - f), 0)
            f /= f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 60:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_obj


def adam_smoothed(f0, time_limit=60.0, verbose=False):
    """Adam on log-sum-exp smoothed objective with simplex projection."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    base_lr = 0.003

    T_start = 3.0 / n
    T_end = 0.2 / n

    it = 0
    no_improve = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / max(1, int(time_limit / 0.012)))
        T = T_start * (T_end / T_start) ** progress
        lr = base_lr * (1.0 - 0.3 * progress)

        conv = fast_convolve(f)
        top_k = max(10, n // 4)
        k_actual = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = T * (pv - pv.max())
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= w.sum()

        sg = np.zeros(n)
        for ii in range(len(peaks)):
            if w[ii] < 1e-14:
                continue
            k = int(peaks[ii])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[ii] * f[k - idx]

        it1 = it + 1
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** it1)
        v_hat = v / (1 - beta2 ** it1)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        f_new = project_simplex(f - step)
        if np.max(np.abs(f_new - f)) < 1e-15:
            no_improve += 1
            if no_improve > 150:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                m, v = np.zeros(n), np.zeros(n)
                no_improve = 0
            continue

        f = f_new
        new_M = fast_convolve(f).max()
        if new_M < best_obj - 1e-14:
            best_obj = new_M
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    Adam iter {it}: obj={best_obj:.10f}, T={T:.6f}")
        else:
            no_improve += 1

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def spectral_init(n, seed=0):
    """Initialize using spectral methods for diversity."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    style = seed % 7

    if style == 0:
        # Gaussian
        f = np.exp(-3.0 * (x - 0.5) ** 2)
    elif style == 1:
        # Raised cosine
        f = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif style == 2:
        # Tapered uniform
        f = np.ones(n)
        tap = max(1, n // 16)
        f[:tap] *= np.linspace(0.3, 1, tap)
        f[-tap:] *= np.linspace(1, 0.3, tap)
    elif style == 3:
        # Chirp
        f = np.abs(np.sin(np.pi * x * (1 + 10 * x)))
    elif style == 4:
        # Random smooth (low-pass filtered noise)
        f = np.ones(n) + rng.standard_normal(n) * 0.15
        f = np.maximum(f, 0.1)
    elif style == 5:
        # Two-peak
        f = np.exp(-8*(x-0.3)**2) + np.exp(-8*(x-0.7)**2)
    else:
        # Inverse Gaussian (peaks at edges)
        f = np.exp(3.0 * (x - 0.5) ** 2)

    f = np.maximum(f, 0)
    ss = f.sum()
    if ss > 1e-15:
        f /= ss
    return f

        print(f"  Time: {t1:.0f}s")
        a_opt, M = exact_coord_descent(parent, time_limit=t1, verbose=True, seed=1)
        try_seq(a_opt, "excd_p1")

    # === Phase 2: LP-FW on best ===
    print(f"\n=== Phase 2: LP-FW (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 550
        t2 = min(rem, 180)
        if t2 > 20:
            print(f"  Time: {t2:.0f}s")
            a_opt2, _ = lp_fw_optimize(best_seq, time_limit=t2, verbose=True)
            try_seq(a_opt2, "lpfw_p2")

    # === Phase 3: Exact CD after LP-FW ===
    print(f"\n=== Phase 3: Exact CD after LP-FW (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 480
        t3 = min(rem, 220)
        if t3 > 20:
            print(f"  Time: {t3:.0f}s")
            a_opt3, _ = exact_coord_descent(best_seq, time_limit=t3, verbose=True, seed=2)
            try_seq(a_opt3, "excd_p3")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent

    target_lengths = [512, 768, 1024, 1536]
    # Remove the parent length to avoid redundancy
    target_lengths = [tn for tn in target_lengths if tn != len(base)]

    for target_n in target_lengths[:4]:
        rem = deadline - time.monotonic() - 300
        if rem < 50:
            break

        f_interp = interpolate_sequence(base, target_n)

        # Exact CD (fast)
        t_alloc = min(rem * 0.45, 130)
        if t_alloc > 15:
            print(f"  ExactCD n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = exact_coord_descent(f_interp, time_limit=t_alloc, verbose=False, seed=target_n)
            try_seq(a_opt, f"excd_n{target_n}")

        # LP-FW
        rem = deadline - time.monotonic() - 250
        if rem > 20:
            t_lpfw = min(rem * 0.35, 100)
            if t_lpfw > 15:
                print(f"  LP-FW n={target_n}, t={t_lpfw:.0f}s")
                a_opt2, _ = lp_fw_optimize(a_opt, time_limit=t_lpfw, verbose=False)
                try_seq(a_opt2, f"lpfw_n{target_n}")

                # Exact CD after LP-FW
                rem = deadline - time.monotonic() - 200
                if rem > 15:
                    a_opt3, _ = exact_coord_descent(a_opt2, time_limit=min(rem, 80), verbose=False, seed=target_n+1)
                    try_seq(a_opt3, f"excd_after_n{target_n}")

    # === Phase 5: Diverse initializations ===
    print(f"\n=== Phase 5: Diverse starts (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        divers_strategies = [
            ("noise_001", lambda: project_simplex(best_seq + np.random.default_rng(101).standard_normal(n_best) * 0.008)),
            ("reverse", lambda: best_seq[::-1] / best_seq[::-1].sum()),
            ("spectral_0", lambda: spectral_init(n_best, seed=10)),
            ("mix_uniform", lambda: (0.4 * best_seq + 0.6 * np.ones(n_best) / n_best)),
            ("spectral_1", lambda: spectral_init(n_best, seed=20)),
            ("mix_noise", lambda: project_simplex(0.8 * best_seq + 0.2 * np.random.default_rng(30).standard_normal(n_best))),
        ]

        for p, (label, gen_fn) in enumerate(divers_strategies):
            rem = deadline - time.monotonic() - 200
            if rem < 15:
                break
            t_alloc = min(rem / max(1, len(divers_strategies) - p), 130)

            f_init = gen_fn()
            f_init = np.maximum(f_init, 0)
            ss = f_init.sum()
            if ss > 1e-15:
                f_init /= ss

            # Exact CD (primary)
            a_opt, _ = exact_coord_descent(f_init, time_limit=min(t_alloc * 0.7, 90), verbose=False, seed=100 + p)
            try_seq(a_opt, f"div_{label}")

            # LP-FW if time
            rem = deadline - time.monotonic() - 150
            if rem > 15 and best_score < 1.51:
                a_opt2, _ = lp_fw_optimize(a_opt, time_limit=min(rem, 50), verbose=False)
                try_seq(a_opt2, f"div_lpfw_{label}")

    # === Phase 6: Final intensive optimization ===
    print(f"\n=== Phase 6: Final (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Long exact CD run
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_final = min(rem, 180)
            print(f"  Final ExactCD: t={t_final:.0f}s")
            a_opt, _ = exact_coord_descent(best_seq, time_limit=t_final, verbose=True, seed=777)
            try_seq(a_opt, "final_excd")

        # Adam polish
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            a_opt2, _ = adam_smoothed(best_seq, time_limit=min(rem, 60), verbose=False)
            try_seq(a_opt2, "adam_polish")

            # Exact CD after Adam
            rem = deadline - time.monotonic() - 15
            if rem > 8:
                a_opt3, _ = exact_coord_descent(a_opt2, time_limit=min(rem, 25), verbose=False, seed=888)
                try_seq(a_opt3, "excd_after_adam")

        # Tiny perturbations + quick optimization
        for trial in range(5):
            rem = deadline - time.monotonic() - 10
            if rem < 8:
                break
            rng = np.random.default_rng(500 + trial)
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)
            a_opt, _ = exact_coord_descent(f_pert, time_limit=min(rem, 18), verbose=False, seed=900 + trial)
            try_seq(a_opt, f"perturb_{trial}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
