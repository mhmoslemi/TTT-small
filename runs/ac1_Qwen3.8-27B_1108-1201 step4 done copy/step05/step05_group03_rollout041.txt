import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

def objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2 for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def evaluate_safe(a):

def smoothed_gradient(f, T, n):

    conv = fast_convolve(f)
    size = 2 * n - 1
    # Use top-k for efficiency
    k = min(size, max(20, n // 2))
    top_k_idx = np.argpartition(conv, -k)[-k:]

    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:

        w[np.argmax(top_k_vals)] = 1.0
    else:
        w = w / w_sum

    for i in range(k):

        valid_hi = min(n - 1, m)
        if valid_lo <= i <= valid_hi:  # This is always true if 0 <= i < n and m is in [0, 2n-2]
            # Actually we need to check if i is in [valid_lo, valid_hi]
            pass
        # Let's just compute directly:

        # and when 0 <= i-m < n (as f[i-m]*f[i]), i.e., m-n+1 <= i <= m
        # Combined: max(0, m-n+1) <= i <= min(n-1, m)
        lo = max(0, m - n + 1)

                # Both give f[m-i], so total = 2*f[m-i]
                mi = m - i
                if 0 <= mi < n:
                    grad[i] += 2.0 * w[i] * f[mi]
    return grad, conv.max()

def adam_smooth_optimize(f0, time_limit, verbose=False, seed=42):
    """Adam optimizer on log-sum-exp smoothed objective with simplex projection."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_obj = objective(f)
    best_f = f.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Adam state
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    # Temperature schedule
    T0 = 8.0 / n
    T_min = 0.3 / n
    
    # Hyperparameters
    base_lr = 0.003
    max_no_improve = 200
    
    it = 0
    no_improve = 0
    patience = 50
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, it / (time_limit / 0.02))
        
        # Temperature: cosine annealing with restarts
        if it % (patience * 4) == 0 and no_improve > patience:
            # Reheat
            T = T0 * (1.0 - progress) * 0.5 + T_min
            no_improve = 0
        else:
            T = T0 * (T_min / T0) ** progress
        
        # Adaptive top-k
        top_k = max(10, int(n * (1.0 - 0.7 * progress)))
        top_k = min(top_k, 2 * n - 1)
        
        # Learning rate decay
        lr = base_lr * (1.0 + 0.5 * np.cos(np.pi * progress))
        
        grad, current_M = smoothed_gradient(f, T, n)
        
        # Project gradient to tangent space of simplex
        grad_t = grad - np.mean(grad)
        
        # Adam update
        it1 = it + 1
        m_state = beta1 * m_state + (1 - beta1) * grad_t
        v_state = beta2 * v_state + (1 - beta2) * grad_t ** 2
        m_hat = m_state / (1 - beta1 ** it1)
        v_hat = v_state / (1 - beta2 ** it1)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        
        f_new = project_simplex(f - step)
        
        if np.max(np.abs(f_new - f)) < 1e-15:
            no_improve += 1
            if no_improve > max_no_improve:
                # Diversify
                scale = 0.003 * min(1 + no_improve / max_no_improve, 3.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                m_state = np.zeros(n)
                v_state = np.zeros(n)
                no_improve = 0
                patience = max(20, patience // 2)
            continue
        
        f = f_new
        new_obj = objective(f)
        
        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    Adam iter {it}: obj={best_obj:.10f}, T={T:.6f}, lr={lr:.6f}")
        else:
            no_improve += 1
    
    return best_f, best_obj


def pairwise_optimize(f0, time_limit, verbose=False, seed=123):
    """Optimize pairs of coordinates by mass transfer with exact 1D search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = objective(f)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Precompute offsets for pairing
    # Use "strategic" offsets: small (local smoothing) and large (global balancing)
    small_offsets = list(range(1, 32))
    large_offsets = [n // 2, n // 3, n // 4, 2 * n // 3, n // 5, 3 * n // 5]
    large_offsets = [o for o in large_offsets if 0 < o < n]
    
    pass_num = 0
    no_improve_passes = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_num += 1
        pass_improved = False
        
        # Alternate between random and structured order
        if pass_num % 4 == 1:
            order = rng.permutation(n)
        elif pass_num % 4 == 2:
            order = np.arange(n)
        else:
            order = np.arange(n - 1, -1, -1)
        
        for idx_i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            i = order[idx_i]
            
            # Choose a few partners
            partners = set()
            for off in small_offsets[:8]:
                j1 = order[(idx_i + off) % n]
                j2 = order[(idx_i - off) % n]
                partners.add(j1)
                partners.add(j2)
            for off in large_offsets[:4]:
                j1 = (i + off) % n
                j2 = (i - off) % n
                partners.add(j1)
                partners.add(j2)
            
            # Limit partners for speed
            if len(partners) > 12:
                partners = list(partners)[:12]
            
            for j in partners:
                if i == j:
                    continue
                
                ci = f[i]
                cj = f[j]
                c = ci + cj
                if c < 1e-12:
                    continue
                
                # Current fraction
                t0_val = ci / c
                
                # Find best transfer
                best_t = t0_val
                best_val = best_obj  # Use current best as reference
                
                # Coarse grid
                candidates = np.linspace(0, 1, 21)
                # Add refinement around current
                candidates = np.unique(np.concatenate([candidates, 
                    np.linspace(max(0, t0_val - 0.1), min(1, t0_val + 0.1), 11)]))
                
                for t in candidates:
                    f_trial = f.copy()
                    f_trial[i] = t * c
                    f_trial[j] = (1 - t) * c
                    val = objective(f_trial)
                    if val < best_val:
                        best_val = val
                        best_t = t
                
                if best_val < best_obj - 1e-13:
                    f[i] = best_t * c
                    f[j] = (1 - best_t) * c
                    best_obj = best_val
                    best_f = f.copy()
                    pass_improved = True
        
        if pass_improved:
            no_improve_passes = 0
        else:
            no_improve_passes += 1
            if no_improve_passes >= 3:
                break
    
    return best_f, best_obj


def spectral_shaping(f0, time_limit, verbose=False, seed=777):
    """Try to flatten the power spectrum via iterative amplitude adjustment."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = objective(f)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute DFT
        A = np.fft.rfft(f)
        power = np.abs(A) ** 2
        
        # Target: flat power spectrum (except DC)
        # power_target[k] = (sum(power) - power[0]) / (len(power) - 1) for k > 0
        target = (power.sum() - power[0]) / max(1, len(power) - 1)
        
        # New amplitude: adjust to move towards flat
        # new_power[k] = (1-alpha) * power[k] + alpha * target for k > 0
        alpha = 0.1
        new_power = power.copy()
        new_power[1:] = (1 - alpha) * power[1:] + alpha * target
        new_power[0] = power[0]  # Keep DC component
        
        # Take square root to get new amplitudes
        new_A = np.sqrt(new_power)
        # Keep phases from original
        phases = np.angle(A)
        new_A = new_A * np.exp(1j * phases)
        
        # Inverse DFT
        f_candidate = np.fft.irfft(new_A, n)
        
        # Ensure non-negativity and normalize
        f_candidate = np.maximum(f_candidate, 0)
        fc = f_candidate.sum()
        if fc < 1e-15:
            f_candidate = project_simplex(f + rng.standard_normal(n) * 0.01)
        else:
            f_candidate = f_candidate / fc
        
        obj = objective(f_candidate)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f_candidate.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    Spectral iter {it}: obj={best_obj:.10f}")
        else:

            else:

        f = best_f  # Always work from best

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def cosine_initialization(n, seed=0):
    """Create a cosine-window-like initialization."""
    x = np.arange(n) / n
    f = np.cos(np.pi * (x - 0.5)) ** 2
    f = np.maximum(f, 0)
    f = f / f.sum()
    return f


def geometric_initialization(n, seed=0):
    """Create a geometric/bell-shaped initialization."""
    x = np.arange(n) / n
    sigma = 0.3
    f = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    f = f / f.sum()
    return f


def random_initialization(n, seed=0):
    """Random initialization on simplex."""
    rng = np.random.default_rng(seed)
    f = np.abs(rng.standard_normal(n))
    f = f / f.sum()
    return f

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed projected gradient (Adam) with adaptive temperature, momentum, and cosine annealing
2. Fast pairwise coordinate descent with convolution-delta optimization
3. Spectral shaping via DFT amplitude adjustment
4. Multi-length exploration with interpolation
5. Diverse restarts with simulated annealing schedule
Key improvement: exact convolution delta for pairwise (O(n) per eval instead of O(n log n))
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return np.array([])
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_objective(f):
    """Compute 2*n*max(conv(f,f)) for f with sum=1."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()


def smoothed_subgradient(f, T, top_k=None):
    """Compute smoothed gradient using softmax-weighted peaks."""
    n = len(f)
    conv = fast_convolve(f)
    size = 2 * n - 1
    if top_k is None:
        top_k = min(size, max(20, n // 3))
    top_k = min(top_k, size)
    
    top_idx = np.argpartition(conv, -top_k)[-top_k:]
    top_vals = conv[top_idx]
    
    scaled = top_vals / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(top_k)
        w[np.argmax(top_vals)] = 1.0
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for i in range(top_k):
        if w[i] < 1e-14:
            continue
        m = int(top_idx[i])
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        # For all i_idx in [lo, hi], grad[i_idx] gets contribution
        # d(b[m])/d(f[i_idx]) = 2*f[m-i_idx] if m != 2*i_idx, else 2*f[i_idx]
        # We need to add w[i] * d(b[m])/d(f) to grad
        
        # Vectorized: for i_idx in [lo, hi], the coefficient is 2*f[m - i_idx]
        # except when m - i_idx == i_idx (i.e., i_idx = m//2 and m is even), where it's 2*f[i_idx]
        # But 2*f[m-i_idx] when i_idx = m//2 gives 2*f[m//2] = 2*f[i_idx], same thing!
        # So actually grad[lo:hi+1] += w[i] * 2 * f[m - (lo:hi+1)]
        
        idx_range = np.arange(lo, hi + 1)
        coeff = 2.0 * w[i] * f[m - idx_range]
        grad[lo:hi + 1] += coeff
    
    return grad, conv.max()


def adam_optimize(f0, time_limit, verbose=False, seed=42, lr_scale=1.0):
    """Adam optimizer on smoothed objective with simplex projection."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_obj = compute_objective(f)
    best_f = f.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    T0 = 10.0 / n
    T_min = 0.2 / n
    base_lr = 0.005 * lr_scale
    
    it = 0
    no_improve = 0
    total_improvements = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        
        # Temperature: cosine annealing
        T = T_min + (T0 - T_min) * 0.5 * (1.0 + np.cos(np.pi * progress))
        
        # Top-k decreases over time
        top_k = max(10, int(n * (1.0 - 0.6 * progress)))
        
        # Learning rate with cosine schedule
        lr = base_lr * 0.5 * (1.0 + np.cos(np.pi * progress))
        
        grad, current_M = smoothed_subgradient(f, T, top_k)
        
        # Project to tangent space
        grad_t = grad - np.mean(grad)
        
        # Adam update
        it1 = it + 1
        m_state = beta1 * m_state + (1 - beta1) * grad_t
        v_state = beta2 * v_state + (1 - beta2) * grad_t ** 2
        m_hat = m_state / (1 - beta1 ** it1)
        v_hat = v_state / (1 - beta2 ** it1)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        
        f_new = project_simplex(f - step)
        
        if np.max(np.abs(f_new - f)) < 1e-16:
            no_improve += 1
            if no_improve > 300:
                scale = 0.003 * min(1.0 + no_improve / 200.0, 4.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                m_state = np.zeros(n)
                v_state = np.zeros(n)
                no_improve = 0
            continue
        
        f = f_new
        new_obj = compute_objective(f)
        
        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_f = f.copy()
            no_improve = 0
            total_improvements += 1
            if verbose and it % 200 == 0:
                print(f"    Adam iter {it}: obj={best_obj:.10f}, T={T:.6f}, lr={lr:.6f}, imp={total_improvements}")
        else:
            no_improve += 1
    
    return best_f, best_obj


def conv_delta_update(b, f, i, j, new_fi, new_fj, n):
    """Compute updated convolution values when f[i] and f[j] change.
    Returns the new full convolution b_new."""
    di = new_fi - f[i]
    dj = new_fj - f[j]
    
    b_new = b.copy()
    
    # Update for coordinate i: b[m] += di * f[m-i] for valid m-i
    # b[m] = sum_k f[k]*f[m-k], changing f[i] by di adds di*f[m-i] for each valid m-i
    lo_i = max(0, i - (n - 1))  # m - i >= 0 => m >= i, and m - i <= n-1 => m <= i+n-1
    # Actually: f[i] appears in b[m] when m-i is in [0, n-1], i.e., m in [i, i+n-1]
    # Also when i is used as the second factor: m-k=i => k=m-i, valid if m-i in [0,n-1]
    # Same condition. But when m-i == i (m=2i), it's counted once not twice.
    # Actually in b[m] = sum_k f[k]*f[m-k], f[i] appears as:
    #   k=i: term is f[i]*f[m-i], contributes di*f[m-i]
    #   k=m-i: term is f[m-i]*f[i], contributes f[m-i]*di
    # Total: 2*di*f[m-i] if m-i != i, or di*f[i] if m=i... 
    # Wait: when m-i=i (m=2i), k=i and k=m-i are the same k, so only one term: di*f[i]+f[i]*di... no.
    # b[m] = sum_k f[k]*f[m-k]. If we change f[i] to f[i]+di:
    # New b[m] = sum_k f'[k]*f'[m-k] where f'[i]=f[i]+di
    # = b[m] + di*f[m-i] (from k=i) + f[m-i]*di (from k=m-i, if m-i != i)
    # = b[m] + di*f[m-i] + (di*f[m-i] if m-i != i else 0)
    # Wait no. Let me be more careful.
    # b[m] = sum_k f[k] * f[m-k]
    # After change: b'[m] = sum_k f'[k] * f'[m-k]
    # = sum_{k != i} f[k]*f[m-k] + f'[i]*f'[m-i] + sum_{k != i, m-k != i} ... 
    # This is getting complicated. Let me just do it directly.
    # b'[m] - b[m] = sum_k (f'[k]*f'[m-k] - f[k]*f[m-k])
    # = sum_k f'[k]*(f'[m-k]-f[m-k]) + sum_k (f'[k]-f[k])*f[m-k]
    # = di * f[m-i] + di * f'[m-i] ... no this is wrong too.
    
    # Simplest: b'[m] - b[m] = di*f[m-i] + dj*f[m-j] + di*f[i-m] ... 
    # Actually the cleanest way:
    # b'[m] = sum_k f'[k] f'[m-k]
    # b[m] = sum_k f[k] f[m-k]
    # b'[m] - b[m] = sum_k (f'[k]f'[m-k] - f[k]f[m-k])
    # = sum_k f'[k](f'[m-k] - f[m-k]) + sum_k (f'[k] - f[k])f[m-k]
    # = di * f'[m-i] + dj * f'[m-j] + di * f[m-i] + dj * f[m-j]  (if all indices valid)
    # But f'[m-i] = f[m-i] + di*(1 if m-i==i else 0) + dj*(1 if m-i==j else 0)
    # This is getting messy. Let me just use the direct formula:
    # b'[m] - b[m] = sum_k f'[k]f'[m-k] - f[k]f[m-k]
    # The only terms that change are where k=i or m-k=i (i.e., k=m-i), or k=j or m-k=j.
    # Terms involving i:
    #   k=i: f'[i]*f'[m-i] - f[i]*f[m-i] = (f[i]+di)*f'[m-i] - f[i]*f[m-i]
    #   k=m-i (if m-i != i and valid): f'[m-i]*f'[i] - f[m-i]*f[i] = f'[m-i]*(f[i]+di) - f[m-i]*f[i]
    # Terms involving j: similar
    # Cross terms (if i and j both appear in the same b[m]): need to be careful.
    
    # For simplicity and correctness, let me just compute the affected b[m] directly.
    # The affected m values are those where b[m] involves index i or j.
    # b[m] involves index i if 0 <= m-i < n, i.e., i <= m < i+n
    # b[m] involves index j if 0 <= m-j < n, i.e., j <= m < j+n
    
    affected = set()
    for m in range(max(0, i), min(2*n-1, i+n)):
        affected.add(m)
    for m in range(max(0, j), min(2*n-1, j+n)):
        affected.add(m)
    
    if not affected:
        return b_new
    
    affected = sorted(affected)
    f_new = f.copy()
    f_new[i] = new_fi
    f_new[j] = new_fj
    
    for m in affected:
        # Recompute b[m] from scratch using f_new
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        b_new[m] = np.dot(f_new[lo:hi+1], f_new[m-lo:m-hi])
    
    return b_new


def pairwise_optimize_fast(f0, time_limit, verbose=False, seed=456):
    """Fast pairwise optimization using convolution delta updates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    b = fast_convolve(f)
    best_obj = 2.0 * n * b.max()
    best_f = f.copy()
    best_b = b.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    pass_num = 0
    no_improve_passes = 0
    total_evals = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_num += 1
        pass_improved = False
        pass_evals = 0
        
        if pass_num % 5 == 1:
            order = rng.permutation(n)
        elif pass_num % 5 == 3:
            order = np.arange(n)
        else:
            order = np.arange(n - 1, -1, -1)
        
        for idx_i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if pass_evals > 50000:  # Limit per pass
                break
            
            i = order[idx_i]
            
            # Choose partners: mix of local and far
            partners = []
            # A few local
            for off in [1, 2, 3, 5, 8]:
                j1 = order[(idx_i + off) % n]
                j2 = order[(idx_i - off) % n]
                if j1 != i:
                    partners.append(j1)
                if j2 != i:
                    partners.append(j2)
            # A few far
            for frac in [0.25, 0.5, 0.75]:
                j1 = (i + int(n * frac)) % n
                j2 = (i - int(n * frac)) % n
                if j1 != i:
                    partners.append(j1)
                if j2 != i:
                    partners.append(j2)
            
            for j in partners:
                if i == j:
                    continue
                pass_evals += 1
                
                ci = f[i]
                cj = f[j]
                c = ci + cj
                if c < 1e-12:
                    continue
                
                t0_frac = ci / c
                
                # Try a few transfer values
                best_t = t0_frac
                new_ci = ci
                new_cj = cj
                best_local_val = np.inf
                
                # Adaptive candidates based on current
                candidates = [t0_frac, 0.5, 0.0, 1.0, 0.25, 0.75]
                if 0.1 < t0_frac < 0.9:
                    candidates.extend([t0_frac - 0.1, t0_frac + 0.1, t0_frac - 0.05, t0_frac + 0.05])
                
                for t in candidates:
                    t = max(0.0, min(1.0, t))
                    new_ci = t * c
                    new_cj = (1.0 - t) * c
                    
                    b_trial = conv_delta_update(b, f, i, j, new_ci, new_cj, n)
                    val = b_trial.max()
                    
                    if val < best_local_val:
                        best_local_val = val
                        best_t = t
                        best_new_ci = new_ci
                        best_new_cj = new_cj
                        best_b_trial = b_trial.copy()
                
                new_obj = 2.0 * n * best_local_val
                if new_obj < best_obj - 1e-13:
                    f[i] = best_new_ci
                    f[j] = best_new_cj
                    b = best_b_trial
                    best_obj = new_obj
                    best_f = f.copy()
                    best_b = b.copy()
                    pass_improved = True
        
        if pass_improved:
            no_improve_passes = 0
            if verbose and pass_num % 50 == 0:
                print(f"    Pairwise pass {pass_num}: obj={best_obj:.10f}, evals={pass_evals}")
        else:
            no_improve_passes += 1
            if no_improve_passes >= 4:
                break
    
    return best_f, best_obj

                t0_frac = ci / c
                best_t = t0_frac
                best_val = best_obj
                
                for t in [t0_frac, 0.5, 0.0, 1.0, max(0, t0_frac-0.15), min(1, t0_frac+0.15)]:
                    t = max(0.0, min(1.0, t))
                    f_trial = f.copy()
                    f_trial[i] = t * c
                    f_trial[j] = (1 - t) * c
                    val = compute_objective(f_trial)
                    if val < best_val:
                        best_val = val
                        best_t = t
                
                if best_val < best_obj - 1e-13:
                    f[i] = best_t * c
                    f[j] = (1 - best_t) * c
                    best_obj = best_val
                    best_f = f.copy()
                    pass_improved = True
        
        if pass_improved:
            no_improve_passes = 0
            if verbose and pass_num % 20 == 0:
                print(f"    PairwiseSimple pass {pass_num}: obj={best_obj:.10f}")
        else:
            no_improve_passes += 1
            if no_improve_passes >= 5:
                break
    
    return best_f, best_obj


def spectral_shaping(f0, time_limit, verbose=False, seed=101):
    """Iteratively flatten the power spectrum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = compute_objective(f)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if no_improve > 150:
            break
        
        A = np.fft.rfft(f)
        power = np.abs(A) ** 2
        
        # Target flat spectrum (excluding DC)
        n_freqs = len(power)
        target = (power.sum() - power[0]) / max(1, n_freqs - 1)
        
        # Progressively stronger flattening
        progress = min(1.0, it / 200)
        alpha = 0.05 + 0.2 * progress
        
        new_power = power.copy()
        new_power[1:] = (1 - alpha) * power[1:] + alpha * target
        # Keep DC fixed
        new_power[0] = power[0]
        
        # Reconstruct with original phases
        phases = np.angle(A)
        new_A = np.sqrt(new_power) * np.exp(1j * phases)
        f_candidate = np.fft.irfft(new_A, n)
        
        # Enforce non-negativity
        f_candidate = np.maximum(f_candidate, 0)
        fc = f_candidate.sum()
        if fc < 1e-15:
            no_improve += 1
            continue
        f_candidate = f_candidate / fc
        
        obj = compute_objective(f_candidate)
        
        improved = False
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f_candidate.copy()
            no_improve = 0
            improved = True
        else:
            # Try mixing
            for mix in [0.2, 0.4, 0.6]:
                f_mix = (1 - mix) * best_f + mix * f_candidate
                f_mix = np.maximum(f_mix, 0)
                fs = f_mix.sum()
                if fs > 1e-15:
                    f_mix = f_mix / fs
                    obj_mix = compute_objective(f_mix)
                    if obj_mix < best_obj - 1e-12:
                        best_obj = obj_mix
                        best_f = f_mix.copy()
                        no_improve = 0
                        improved = True
                        break
        
        if not improved:
            no_improve += 1
        
        f = best_f
        if verbose and it % 100 == 0:
            print(f"    Spectral iter {it}: obj={best_obj:.10f}, alpha={alpha:.4f}")
    
    return best_f, best_obj


def block_optimize(f0, time_limit, block_size=16, verbose=False, seed=202):
    """Optimize blocks of coordinates via local random search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = compute_objective(f)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    num_blocks = (n + block_size - 1) // block_size
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if no_improve > 200:
            break
        
        improved = False
        block_order = rng.permutation(num_blocks)
        
        for bi in block_order:
            if time.monotonic() - t0 > time_limit:
                break
            
            start = bi * block_size
            end = min(start + block_size, n)
            bs = end - start
            block_sum = f[start:end].sum()
            if block_sum < 1e-12:
                continue
            
            for _ in range(3):
                # Random perturbation scaled to block sum
                scale = 0.02 * block_sum
                pert = rng.standard_normal(bs) * scale
                trial_block = np.maximum(f[start:end] + pert, 0)
                ts = trial_block.sum()
                if ts < 1e-12:
                    continue
                trial_block = trial_block * (block_sum / ts)
                
                f_trial = f.copy()
                f_trial[start:end] = trial_block
                f_trial = f_trial / f_trial.sum()
                obj = compute_objective(f_trial)
                
                if obj < best_obj - 1e-13:
                    f = f_trial
                    best_obj = obj
                    best_f = f.copy()
                    improved = True
                    break
        
        if improved:
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    Block iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
