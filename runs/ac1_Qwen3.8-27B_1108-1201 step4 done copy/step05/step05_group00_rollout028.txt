**Key idea: Active set refinement**

Near the optimum, a small number of convolution positions will be "active" (at or very close to the maximum). We can:
1. Identify the active set S (positions where `c[k] >= max(c) - ε`).
2. Solve the constrained problem: minimize `max_{k in S} c[k]` subject to `a >= 0`, `sum(a) = 1`.
3. This can be formulated as a QCQP: minimize `t` subject to `c[k] <= t` for `k in S`, `a >= 0`, `sum(a) = 1`.
4. Use a QP solver or iterative method for this.

For the QCQP, we can use the approach:
- Linearize around the current point: `c[k] ≈ c_0[k] + ∇c[k]^T * (a - a_0)`.
- Solve the resulting LP: minimize `t` subject to `c_0[k] + ∇c[k]^T * (a - a_0) <= t`, `a >= 0`, `sum(a) = 1`.
- Take a step in this direction.
- Repeat.

This is essentially a sequential linear programming (SLP) approach, which should converge fast near the optimum.

**Even better: Use the structure of the gradient**

Since `∇c[k]` is a shifted version of `a`, the LP becomes:
Minimize `t` subject to:
`c_0[k] + 2 * sum_m a[m] * a_0[k-m] <= t` for `k in S` (wait, this is still quadratic because both `a[m]` and `a_0[k-m]` appear... no, `a_0` is fixed, so it's linear in `a`).

Actually wait: `c[k] = sum_j a[j] * a[k-j]`. The linearization at `a_0` is:
`c[k] ≈ c_0[k] + sum_j (a[j] - a_0[j]) * a_0[k-j] + sum_j a_0[j] * (a[k-j] - a_0[k-j])`
`= c_0[k] + sum_j (a[j] - a_0[j]) * a_0[k-j] + sum_j a_0[j] * (a[k-j] - a_0[k-j])`
`= c_0[k] + 2 * sum_m (a[m] - a_0[m]) * a_0[k-m]` (by the symmetry and reindexing I worked out earlier)

Wait, that's the first-order approximation. But `c[k]` is actually quadratic, so the linearization is:
`c[k] ≈ c_0[k] + ∇c[k](a_0)^T * (a - a_0)`

where `∇c[k](a_0)[m] = 2*a_0[k-m]` (for valid indices).

So the linearized constraint is:
`c_0[k] + 2 * sum_m (a[m] - a_0[m]) * a_0[k-m] <= t`

This is linear in `a`! So we can solve an LP:
Minimize `t`
Subject to: `2 * sum_m a[m] * a_0[k-m] <= t - c_0[k] + 2 * sum_m a_0[m] * a_0[k-m]` for `k in S`

So the LP has `|S|` functional constraints plus the simplex constraint. For `|S|` around 10-50 and `n = 768`, this is very fast to solve.

def hankel_qp_solve(p, n, max_iter=200):
    """Solve min a^T M a s.t. a in simplex, M[i,j] = p[i+j]. Active set method."""
    active = list(range(n))
    
    for _ in range(max_iter):
        if len(active) <= 1:
            a = np.zeros(n)
            if len(active) == 1:
                a[active[0]] = 1.0
            else:
                a[0] = 1.0
            return a
        
        S = active
        k = len(S)
        
        # Build M_SS (sparse Hankel)
        M_SS = np.array([[p[S[i] + S[j]] for j in range(k)] for i in range(k)])
        
        # KKT: [2M_SS, 1; 1^T, 0][x; lambda] = [0; 1]
        KKT = np.zeros((k + 1, k + 1))
        KKT[:k, :k] = 2.0 * M_SS
        KKT[:k, k] = 1.0
        KKT[k, :k] = 1.0
        rhs = np.zeros(k + 1)
        rhs[k] = 1.0
        
        try:
            sol = np.linalg.solve(KKT, rhs)
            x = sol[:k]
            lam = sol[k]
        except np.linalg.LinAlgError:
            # Add regularization
            KKT[:k, :k] += 1e-8 * np.eye(k)
            try:
                sol = np.linalg.solve(KKT, rhs)
                x = sol[:k]
                lam = sol[k]
            except:
                x = np.ones(k) / k
                lam = 0
        
        # Check for negative components
        if np.min(x) < -1e-10:
            min_idx = np.argmin(x)
            active.remove(S[min_idx])
            continue
        
        # Build full a
        a = np.zeros(n)
        for i, idx in enumerate(S):
            a[idx] = max(0, x[i])
        
        # Compute Ma for all coordinates
        # Ma[j] = sum_i M[j,i] * a[i] = sum_i p[j+i] * a[i]
        Ma = np.zeros(n)
        for j in range(n):
            Ma[j] = 0
            for i in range(n):
                if a[i] > 0:
                    Ma[j] += p[j + i] * a[i]
        
        # KKT check for inactive
        violated = False
        for j in range(n):
            if j not in active:
                if 2.0 * Ma[j] + lam < -1e-8:
                    active.append(j)
                    violated = True
        
        if not violated:
            return a
    
    a = np.zeros(n)
    a[active] = 1.0 / len(active)
    return a

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_active_set_lp(a, S):
    """
    Solve LP: min t s.t. linearized c[k] <= t for k in S, a' >= 0, sum(a') = 1.
    Linearization: c[k] ~ c[k]_current + sum_m 2*a[k-m]*(a'[m] - a[m])
    Simplified: 2*sum_m a[k-m]*a'[m] - t <= c[k]_current
    """
    n = len(a)
    c = fast_convolve(a)

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    A_ub = np.zeros((len(S), n + 1))
    b_ub = np.zeros(len(S))

    for i, k in enumerate(S):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for m in range(js, je + 1):
            A_ub[i, m] = 2.0 * a[k - m]
        A_ub[i, -1] = -1.0
        b_ub[i] = c[k]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(-np.inf, np.inf)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            a_new = np.maximum(res.x[:n], 0.0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
            return a_new, res.x[-1]
    except Exception:
        pass
    return None, None

def hankel_matvec(p, a, n):
    """Compute M @ a where M[i,j] = p[i+j]. Efficient for Hankel structure."""
    result = np.zeros(n)
    for i in range(n):
        # result[i] = sum_j p[i+j] * a[j]
        result[i] = np.dot(p[i:i+n], a)
    return result

            if 2.0 * Ma[j] + lam < -1e-8:
                active.add(j)
                added = True

        if not added:
            return a

    # Fallback
    a = np.zeros(n)
    a[list(active)] = 1.0 / len(active)
    return a
