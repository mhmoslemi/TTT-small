def local_polish(a0, time_limit, rng, verbose=True):
    """Local polish using targeted moves on peak-contributing coordinates."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = a0.astype(np.float64).copy()
    a = np.clip(a, 0.0, None)
    a = normalize_seq(a, S)
    
    current_score = fast_score(a)
    best_a = a.copy()
    best_score = current_score
    
    t0 = time.monotonic()
    temp = 0.001
    cooling = 0.9998
    
    while time.monotonic() - t0 < time_limit:
        candidate = a.copy()
        r = rng.random()
        
        if r < 0.4:
            # Targeted: reduce top contributors to peak, increase others
            c = fft_convolve(a, a)
            peak = int(np.argmax(c))
            # Find top contributors to peak
            lo = max(0, peak - n + 1)
            hi = min(n - 1, peak)
            contribs = np.zeros(n)
            for m in range(lo, hi + 1):
                contribs[m] = a[m] * a[peak - m]
            top_k = max(3, n // 8)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.02, 0.008, len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 3:
                n_inc = min(10, len(other) // 3)
                inc_idx = rng.choice(other, n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.02, 0.008, n_inc))
        elif r < 0.7:
            # Pair swap
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = rng.uniform(-0.1, 0.1) * max(a[i], a[j])
                candidate[i] = max(0, a[i] - delta)
                candidate[j] = max(0, a[j] + delta)
        elif r < 0.85:
            # Small block
            w = max(1, int(rng.integers(1, max(2, n // 10))))
            s = int(rng.integers(0, n - w))
            candidate[s:s+w] *= np.exp(rng.normal(0, 0.015, w))
        else:
            # Tiny global
            candidate *= np.exp(rng.normal(0, 0.004, n))
        
        candidate = np.clip(candidate, 0.0, None)
        cs = np.sum(candidate)
        if cs < 0.01:
            candidate = a.copy()
        else:
            candidate = normalize_seq(candidate, S)

            a = candidate

            a = candidate
            current_score = new_score
        
        temp *= cooling
    
    return best_a, best_score

def make_init(n, rng, init_type='random'):

"""
Subgradient-based optimization with momentum for minimizing peak self-convolution
of non-negative step functions. Uses fast O(n) subgradient computation via FFT,
adaptive step sizing, multiple near-peak positions, periodic perturbations,
and targeted local polish. Explores multiple sequence lengths and initializations.

import numpy as np
import time


def fft_convolve(a, b):
    """Linear convolution of two 1-D arrays via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def fast_score(a):
    """Compute the evaluation score efficiently."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (s * s))


def compute_subgradient(a, peak_fraction=0.97, max_peaks=10):
    """
    Compute the subgradient of max_k (a*a)[k] averaged over near-peak positions.
    Uses the formula: d(a*a)[k]/d(a)[m] = 2*a[k-m] * I(0<=k-m<n)
    """
    n = len(a)
    c = fft_convolve(a, a)
    max_c = np.max(c)
    thresh = max_c * peak_fraction
    peak_positions = np.where(c >= thresh)[0]
    if len(peak_positions) > max_peaks:
        # Keep the highest ones
        peak_values = c[peak_positions]
        top_indices = np.argsort(peak_values)[::-1][:max_peaks]
        peak_positions = peak_positions[top_indices]
    if len(peak_positions) == 0:
        peak_positions = np.array([np.argmax(c)])

    sg = np.zeros(n)
    for k in peak_positions:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            length = hi - lo + 1
            sg[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]

    sg /= len(peak_positions)
    return sg, max_c, len(peak_positions)

    mu = 0.88

    stall = 0

                n_inc = min(12, len(other) // 3)
                inc_idx = rng.choice(other, n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.018, 0.007, n_inc))
        elif r < 0.55:
            # Pair redistribution
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = rng.uniform(-0.12, 0.12) * max(a[i], a[j], 0.1)
                candidate[i] = max(0.0, a[i] - delta)
                candidate[j] = max(0.0, a[j] + delta)
        elif r < 0.70:
            # Block perturbation
            w = max(1, int(rng.integers(1, max(2, n // 8))))
            s = int(rng.integers(0, n - w))
            candidate[s:s + w] *= np.exp(rng.normal(0.0, 0.012, w))
        elif r < 0.85:
            # Multi-point small adjustment
            for _ in range(4):
                idx = int(rng.integers(0, n))
                candidate[idx] *= np.exp(rng.normal(0.0, 0.02))
        else:
            # Tiny global
            candidate *= np.exp(rng.normal(0.0, 0.003, n))

        candidate = np.clip(candidate, 0.0, None)
        cs = np.sum(candidate)
        if cs < 0.01:
            candidate = a.copy()
        else:
            candidate = normalize_seq(candidate, S)

        new_score = fast_score(candidate)
        count += 1

        if new_score < current_score:
            a = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
                if verbose and count % verbose_interval == 0:
                    elapsed = time.monotonic() - t0
                    print(f"        polish: score={best_score:.6f}, t={elapsed:.1f}s")
        elif rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-15)):
            a = candidate
            current_score = new_score

        temp *= cooling

    return best_a, best_score
