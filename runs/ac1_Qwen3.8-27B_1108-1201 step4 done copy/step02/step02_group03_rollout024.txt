"""
Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based soft-max gradient descent, LP refinement, and coordinate descent.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Next power of 2 for FFT efficiency
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    fa = np.fft.rfft(a, fft_size)
    fa2 = fa * fa
    result = np.fft.irfft(fa2, fft_size)[:size]
    #Clip small negative values from numerical error
    result = np.maximum(result, 0)
    return result

def fast_grad_soft(a, w):
    """
    Compute soft-max gradient.
    w: weights of length 2n-1 (probability distribution over lags)
    grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    """
    n = len(a)
    # This is a cross-correlation: grad[i] = 2 * <w[i:i+n], a>
    # Using FFT: pad w and a, compute cross-correlation
    fft_size = 1
    while fft_size < 2 * n:
        fft_size *= 2
    # w has length 2n-1, a has length n
    # We want: for each i in [0, n-1], dot(w[i:i+n], a)
    # This is the "valid" cross-correlation of w (reversed) with a
    # corr[i] = sum_j w[i+j] * a[j]
    # = conv(w_rev, a)[i + n - 1] where w_rev = w[::-1]
    # Actually: conv(w_rev, a)[k] = sum_j w_rev[j] * a[k-j] = sum_j w[n-1-j] * a[k-j]
    # Let m = n-1-j, j = n-1-m: = sum_m w[m] * a[k-n+1+m]
    # For k = i + n - 1: = sum_m w[m] * a[i+m]
    # So conv(w_rev, a)[i+n-1] = sum_m w[m] * a[i+m] which is what we want (for m in valid range)
    
    w_rev = w[::-1]
    fw = np.fft.rfft(w_rev, fft_size)
    fa = np.fft.rfft(a, fft_size)
    conv_result = np.fft.irfft(fw * fa, fft_size)
    grad = 2.0 * conv_result[n-1:n-1+n]
    return grad

def compute_obj(a):
    """Compute the objective."""
    n = len(a)
    c = fast_conv(a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def normalize(a, target=None):
    """Normalize sequence to have a specific sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)

def soft_max_weights(c, tau):
    """Compute soft-max weights from convolution values."""
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    return w

def gradient_step(a, tau, lr, momentum_buf, momentum_decay):
    """One step of soft-max gradient descent with momentum."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a, S_t)
    
    c = fast_conv(a)
    w = soft_max_weights(c, tau)
    grad = fast_grad_soft(a, w)
    
    # Project out the mean to maintain sum constraint
    grad -= grad.mean()
    
    # Momentum
    momentum_buf[:] = momentum_decay * momentum_buf - lr * grad
    
    a_new = a + momentum_buf
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize(a_new, S_t)

def lp_refine(f, M, time_limit=30):
    """
    Solve LP: max sum(g) s.t. conv(f,g) <= M (elementwise), g >= 0.
    """
    n = len(f)
    if n > 600:
        return None
    
    c_obj = -np.ones(n)
    
    # Build constraint matrix efficiently
    # A[m, j] = f[m-j] if 0 <= m-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo_m = j
        hi_m = min(j + n - 1, 2 * n - 2)
        for m in range(lo_m, hi_m + 1):
            A[m, j] = f[m - j]
    
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

"""
Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses soft-max gradient descent with adaptive tau/lr, FFT-based convolution,
LP refinement, and diverse multi-start strategy with restart mechanisms.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size + n:
        fft_size *= 2
    fa = np.fft.rfft(a, fft_size)
    fa2 = fa * fa
    result = np.fft.irfft(fa2, fft_size)[:size]
    return np.maximum(result, 0.0)


def fast_grad_soft(a, w):
    """
    Compute soft-max gradient: grad[i] = 2 * sum_j w[i+j] * a[j]
    w has length 2n-1, a has length n.
    Using matrix-free approach with FFT cross-correlation.
    """
    n = len(a)
    # We want for each i: dot(w[i:i+n], a[0:n])
    # This equals the valid cross-correlation of w with a
    # conv(w[::-1], a) gives us what we need at specific indices
    w_rev = w[::-1]
    fft_size = 1
    while fft_size < len(w_rev) + n - 1:
        fft_size *= 2
    fw = np.fft.rfft(w_rev, fft_size)
    fa = np.fft.rfft(a, fft_size)
    conv_res = np.fft.irfft(fw * fa, fft_size)
    # conv(w_rev, a)[k] = sum_j w_rev[j] * a[k-j]
    # = sum_j w[2n-2-j] * a[k-j]
    # Let m = 2n-2-j => j = 2n-2-m
    # = sum_m w[m] * a[k-(2n-2-m)] = sum_m w[m] * a[k-2n+2+m]
    # We want sum_m w[m] * a[i+m] for m in [0, n-1]
    # So k-2n+2+m = i+m => k = i+2n-2
    # Hmm, that's beyond the array. Let me use a different approach.
    
    # Direct O(n^2) computation - fine for n <= 600
    grad = np.zeros(n)
    for i in range(n):
        k_start = i
        k_end = i + n
        if k_end > len(w):
            k_end = len(w)
        if k_start < len(w):
            grad[i] = 2.0 * np.dot(w[k_start:k_end], a[:k_end - k_start])
    return grad


def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=None):
    """Normalize sequence to have a specific sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def soft_max_weights(c, tau):
    """Compute soft-max (Boltzmann) weights from convolution values."""
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    return w


def grad_descent_step(a, tau, lr, mom_buf, mom_decay):
    """One soft-max gradient descent step with momentum."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a, S_t)
    c = fast_conv(a)
    w = soft_max_weights(c, tau)
    grad = fast_grad_soft(a, w)
    # Project to maintain sum constraint
    grad -= grad.mean()
    # Momentum update
    mom_buf[:] = mom_decay * mom_buf - lr * grad
    a_new = a + mom_buf
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize(a_new, S_t)
    return a_new, mom_buf, c


def lp_refine(f, M, time_limit=20):
    """LP: max sum(g) s.t. conv(f,g)[k]<=M, g>=0."""
    n = len(f)
    if n > 512:
        return None
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for m in range(j, min(j + n, 2 * n - 1)):
            A[m, j] = f[m - j]
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

    a = normalize(a.copy(), S_t)

            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            orig = a[i]
            if orig < 1e-8:
                continue
            best_val = orig
            best_s = best_score
            for scale in [0.4, 0.7, 1.4, 1.8]:
                a[i] = orig * scale
                a = normalize(a, S_t)
                s = compute_obj(a)
                if s < best_s - 1e-12:
                    best_s = s
                    best_val = orig * scale
                a[i] = orig
            if best_val != orig:
                a[i] = best_val
                a = normalize(a, S_t)
                best_score = best_s
                improved = True
        if not improved:
            break
    return a, best_score

    # Phase 1: Soft-max gradient descent
    mom = np.zeros(n)
    mom_decay = 0.85
    no_improve = 0
    it = 0
    p1_frac = 0.5
    p1_deadline = time.monotonic() + p1_frac * (deadline - time.monotonic())
    
    while time.monotonic() < p1_deadline and it < 80000:
        it += 1
        frac = min(1.0, it / 3000.0)
        tau = 2.0 + 100.0 * (frac ** 0.5)
        tau = min(tau, 120.0)
        lr = 0.4 * (0.9995 ** it)
        lr = max(lr, 0.001)
        
        a, mom, c = grad_descent_step(a, tau, lr, mom, mom_decay)
        
        if it % 40 == 0:
            s = compute_obj(a)
            if s < best_score - 1e-10:
                best_score = s
                best = a.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 12:
                    # Restart mechanism
                    mom[:] = 0
                    no_improve = 0
                    # Random perturbation
                    n_pert = max(1, n // 10)
                    idx = rng.choice(n, size=n_pert, replace=False)
                    a[idx] += rng.normal(0, 0.1 * best[idx + 0].sum()/n, n_pert)
                    a = np.maximum(a, 0)
                    a = normalize(a, S_t)
    
    print(f"    P1: score={best_score:.6f}, it={it}")
    
    if time.monotonic() >= deadline - 20:
        return best, best_score
    
    # Phase 2: LP refinement
    if n <= 448:
        a = best.copy()
        c = fast_conv(a)
        M = float(c.max())
        lp_stuck = 0
        p2_deadline = min(deadline - 25, time.monotonic() + 0.25 * (deadline - time.monotonic()))
        it_lp = 0
        
        while time.monotonic() < p2_deadline and it_lp < 40:
            it_lp += 1
            tl = max(5, int(p2_deadline - time.monotonic() - 5))
            tl = min(tl, 25)
            g = lp_refine(a, M, time_limit=tl)
            if g is None:
                break
            g_sum = g.sum()
            if g_sum < 1e-6:
                break
            g_norm = g * (S_t / g_sum)
            
            improved = False
            for t_b in [0.4, 0.25, 0.15, 0.08, 0.04]:
                a_new = (1.0 - t_b) * a + t_b * g_norm
                a_new = np.maximum(a_new, 0.0)
                a_new = normalize(a_new, S_t)
                s_new = compute_obj(a_new)
                if s_new < best_score - 1e-10:
                    a = a_new
                    best = a.copy()
                    best_score = s_new
                    improved = True
                    break
            
            if not improved:
                lp_stuck += 1
                if lp_stuck > 4:
                    break
                # Small gradient nudge
                a, mom, _ = grad_descent_step(a, 15.0, 0.03, np.zeros(n), 0.0)
                s = compute_obj(a)
                if s < best_score:
                    best = a.copy()
                    best_score = s
                    lp_stuck = 0
            else:
                lp_stuck = 0
            c = fast_conv(a)
            M = float(c.max())
        
        print(f"    P2(LP): score={best_score:.6f}, it={it_lp}")
    
    if time.monotonic() >= deadline - 15:
        return best, best_score
    
    # Phase 3: Coordinate descent
    if n <= 384:
        a_p, s_p = coord_polish(best, deadline - 8, max_rounds=3)
        if s_p < best_score:
            best = a_p
            best_score = s_p
        print(f"    P3(coord): score={best_score:.6f}")
    
    return best, best_score


def make_starts(rng, lengths):
    """Generate diverse starting sequences."""
    starts = []
    
    # From height_sequence_1 if available
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 8 and np.all(np.isfinite(gs)):
            for L in lengths:
                if L <= len(gs):
                    starts.append(gs[:L].copy())
                else:
                    reps = int(np.ceil(L / len(gs)))
                    starts.append(np.tile(gs, reps)[:L])
            if len(gs) >= 256:
                starts.append(gs[:384][::-1].copy() if len(gs) >= 384 else gs[::-1].copy())
                # Smoothed
                k = np.ones(7) / 7
                sm = np.convolve(gs[:min(len(gs), 512)], k, mode='same')
                starts.append(np.maximum(sm, 0))
    except (NameError, Exception):
        pass
    
    for n in lengths:
        i = np.arange(n, dtype=np.float64)
        # Uniform
        starts.append(np.ones(n) * (1.0 / n))
        # Random
        starts.append(rng.random(n) * 0.6 + 0.2)
        # Raised cosine
        starts.append(0.5 + 0.5 * np.cos(2 * np.pi * i / max(n - 1, 1)))
        # Hann
        starts.append(0.5 * (1 - np.cos(2 * np.pi * i / max(n - 1, 1))))
        # Multi-frequency
        starts.append(0.4 + 0.3 * np.cos(2 * np.pi * 3 * i / n) + 0.2 * np.cos(2 * np.pi * 7 * i / n))
        # Gaussian center
        starts.append(np.exp(-0.5 * ((i - n / 2) / (n * 0.15)) ** 2))
        # Two Gaussians
        starts.append(np.exp(-0.5 * ((i - n * 0.25) / (n * 0.08)) ** 2) +
                      np.exp(-0.5 * ((i - n * 0.75) / (n * 0.08)) ** 2))
        # Frequency-domain flat spectrum
        fft_sz = 1
        while fft_sz < 2 * n:
            fft_sz *= 2
        mag = rng.random(fft_sz // 2 + 1) ** 0.5 + 0.1
        phase = rng.random(fft_sz // 2 + 1) * 2 * np.pi
        X = mag * np.exp(1j * phase)
        X_full = np.zeros(fft_sz, dtype=complex)
        X_full[:fft_sz // 2 + 1] = X
        for k in range(1, fft_sz // 2):
            X_full[fft_sz - k] = np.conj(X[k])
        a_fd = np.fft.irfft(X_full, fft_sz)[:n]
        a_fd = np.maximum(a_fd, 0)
        if a_fd.sum() > 0.01:
            starts.append(a_fd)
        # Sparse clusters
        n_cl = rng.integers(3, 8)
        a_sp = np.zeros(n)
        for _ in range(n_cl):
            c = rng.integers(0, n)
            w = rng.integers(3, n // 5 + 1)
            for off in range(-w, w + 1):
                idx = (c + off) % n
                a_sp[idx] += np.exp(-0.5 * (off / max(w // 2, 1)) ** 2)
        a_sp += 0.03
        starts.append(a_sp)
    
    return starts


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: multi-start soft-max gradient descent + LP + coordinate descent.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 12.0)
    
    best_seq = None
    best_score = float('inf')
    
    lengths = [256, 320, 384, 448, 512]
    print(f"Budget={budget_s}s, starting at n in {lengths}")
    
    starts = make_starts(rng, lengths)
    starts.sort(key=lambda x: len(x))
    
    # Deduplicate
    unique = []
    seen = set()
    for s in starts:
        key = (len(s),)
        unique.append(s)
        seen.add(key)
    starts = unique
    print(f"Total starts: {len(starts)}")
    
    # Pass 1: Screen all starts
    for idx, a0 in enumerate(starts):
        if time.monotonic() > deadline - 90:
            break
        remaining = deadline - time.monotonic() - 90
        n_left = len(starts) - idx
        t_alloc = min(80.0, remaining / max(1, n_left) * 0.9)
        t_alloc = max(12.0, t_alloc)
        if time.monotonic() + t_alloc > deadline - 60:
            t_alloc = max(10.0, deadline - time.monotonic() - 60)
            if t_alloc < 10:
                break
        
        seq, score = optimize_one(a0, time.monotonic() + t_alloc, rng)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  [s{idx},n={len(seq)}] BEST: {best_score:.6f}")
        else:
            print(f"  [s{idx},n={len(seq)}] {score:.6f}")
    
    # Pass 2: Focus on best with perturbations
    if best_seq is not None and time.monotonic() < deadline - 60:
        bn = len(best_seq)
        print(f"\nPass 2: n={bn}, best={best_score:.6f}")
        
        for trial in range(6):
            if time.monotonic() > deadline - 40:
                break
            remaining = deadline - time.monotonic() - 40
            t_alloc = min(100.0, remaining / max(1, 6 - trial))
            if t_alloc < 15:
                break
            
            if trial == 0:
                a0 = np.maximum(best_seq + rng.normal(0, 0.08, bn), 0)
            elif trial == 1:
                a0 = best_seq * (0.8 + 0.4 * rng.random(bn))
            elif trial == 2:
                a0 = rng.random(bn) * 0.5 + 0.25
            elif trial == 3:
                a0 = best_seq[rng.permutation(bn)]
            elif trial == 4:
                kern = np.ones(7) / 7
                a0 = np.convolve(best_seq, kern, mode='same')
                a0 = np.maximum(a0, 0)
            else:
                # Invert (peak becomes valley and vice versa)
                a0 = best_seq.max() - best_seq + 0.01
                a0 = np.maximum(a0, 0)
            
            seq, score = optimize_one(a0, deadline - 20, rng)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  [p2-{trial}] BEST: {best_score:.6f}")
            else:
                print(f"  [p2-{trial}] {score:.6f}")
    
    # Final polish
    if best_seq is not None and time.monotonic() < deadline - 20:
        rem = deadline - time.monotonic() - 8
        if rem > 30:
            print(f"\nFinal polish: {best_score:.6f}")
            seq, score = optimize_one(best_seq, deadline - 8, rng)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  Final: {best_score:.6f}")
    
    if best_seq is not None:
        fs = compute_obj(best_seq)
        print(f"\nResult: n={len(best_seq)}, score={fs:.6f}")
        try:
            ev = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence: {ev:.6f}")
        except Exception as e:
            print(f"Eval error: {e}")
        return [float(x) for x in best_seq]
    
    print("Fallback")
    return [1.0] * 256
