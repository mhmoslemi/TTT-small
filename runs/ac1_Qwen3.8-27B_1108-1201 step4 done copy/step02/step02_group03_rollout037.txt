        return a.copy()
    return a * (target / S)

def gradient_hardmax(a):
    """Gradient of max_k conv(a,a)[k] w.r.t. a (assuming unique max)."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad, k_star

def gradient_softmax(a, tau):
    """Gradient of soft-max of conv(a,a)."""
    n = len(a)
    c = np.convolve(a, a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    # grad[i] = 2 * sum_j w[i+j] * a[j]
    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_result[n-1:n-1+n]
    return grad

def solve_lp(f, M, n_max=1024):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M (elementwise), g >= 0."""
    n = len(f)
    if n > n_max:
        return None
    
    # Build constraint matrix A: A[k, j] = f[k-j] if 0 <= k-j < n
    A = np.zeros((2*n-1, n))
    for j in range(n):
        for k in range(j, j+n):
            A[k, j] = f[k-j]
    
    c_obj = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

def lp_iteration(a, M, t_blend=0.5):
    """One LP iteration: solve LP, blend, return new a and new M."""
    n = len(a)
    S = a.sum()
    g = solve_lp(a, M)
    if g is None:
        return a, M, False
    
    g_sum = g.sum()
    if g_sum < 1e-6:
        return a, M, False
    
    # Normalize g to have same sum as a
    g = g * (S / g_sum)
    
    # Blend
    a_new = (1.0 - t_blend) * a + t_blend * g
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize_to_sum(a_new, S)
    
    c_new = np.convolve(a_new, a_new)
    M_new = float(c_new.max())
    
    return a_new, M_new, True

def soft_max_descent(a, deadline, tau_start=1.0, tau_end=50.0, max_iters=20000):
    """Soft-max gradient descent with annealing."""
    n = len(a)
    S = a.sum()
    target = np.sqrt(2.0 * n)
    a = normalize_to_sum(a.copy(), target)
    best = a.copy()
    best_score = compute_obj(a)
    
    momentum = np.zeros(n)
    decay = 0.9
    lr = 0.3
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Anneal tau
        progress = min(1.0, it / max_iters * 3)  # reach tau_end in 1/3 of iterations
        tau = tau_start * (tau_end / tau_start) ** progress
        tau = min(tau, tau_end)
        
        # Anneal learning rate
        lr = 0.3 * (0.999 ** it)
        lr = max(lr, 0.002)
        
        grad = gradient_softmax(a, tau)
        grad -= grad.mean()  # project out mean
        
        momentum = decay * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_to_sum(a_new, target)
        
        if it % 100 == 0:
            s = compute_obj(a_new)
            if s < best_score - 1e-7:
                best_score = s
                best = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 10:
                    break
            a = a_new
        else:
            a = a_new
    
    return best, best_score
