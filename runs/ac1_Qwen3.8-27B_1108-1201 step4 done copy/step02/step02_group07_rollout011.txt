```python
"""
Smoothed gradient descent with log-sum-exp for minimizing peak autocorrelation
of non-negative sequences. Uses FFT-based convolution, matrix-based gradient
computation, adaptive temperature, and momentum. Multiple starts across different
sequence lengths and initializations.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def make_init(n, init_type, rng):
    """Generate initial sequence."""
    if init_type == 'hann':
        t = np.arange(n, dtype=np.float64)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
        a += 0.03 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif init_type == 'cosine':
        t = np.arange(n, dtype=np.float64)
        a = np.cos(np.pi * t / n) ** 2
        a += 0.02 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif init_type == 'gaussian':
        t = np.arange(n, dtype=np.float64)
        center = n / 2
        sigma = n / 4
        a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
        a += 0.02 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif init_type == 'random_smooth':
        a = rng.random(n)
        # Low-pass filter
        from scipy.ndimage import uniform_filter1d
        a = uniform_filter1d(a, size=max(2, n // 16))
        a = a / max(a.max(), 1e-10)
        a += 0.01 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif init_type == 'provided':
        global height_sequence_1
        if height_sequence_1 is not None:
            x_old = np.linspace(0, 1, len(height_sequence_1))
            x_new = np.linspace(0, 1, n)
            return np.interp(x_new, x_old, np.asarray(height_sequence_1, dtype=np.float64))
        return make_init(n, 'hann', rng)
    else:
        t = np.arange(n, dtype=np.float64)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
        return np.clip(a, 0.01, None)


def smoothed_grad_descent(a_init, time_limit, T_init=0.5, T_min=0.005,
                          lr_init=0.05, momentum=0.9, floor=0.001,
                          verbose=True, seed=0):
    """Smoothed gradient descent using log-sum-exp approximation of max."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = np.sum(a)
    a = a / S  # normalize to sum=1
    
    best_a = a.copy()
    best_score = score_fast(best_a)
    t0 = time.monotonic()
    m = np.zeros(n)
    it = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        
        # Temperature schedule: exponential decay
        T = max(T_min, T_init * np.exp(-5.0 * progress))
        
        # Learning rate: cosine decay
        lr = lr_init * 0.5 * (1.0 + np.cos(np.pi * progress))
        lr = max(lr, lr_init * 0.01)
        
        # Compute convolution
        c = fft_convolve(a, a)
        
        # Compute softmax weights (numerically stable)
        c_max = np.max(c)
        exp_c = np.exp((c - c_max) / T)
        w = exp_c / np.sum(exp_c)
        
        # Compute gradient: g[i] = 2 * sum_j a[j] * w[i+j]
        i_idx = np.arange(n)
        # For memory efficiency with large n, chunk the computation
        if n <= 2048:
            M = w[i_idx[:, None] + i_idx[None, :]]
            g = 2.0 * M.dot(a)
        else:
            g = np.zeros(n)
            chunk = 256
            for start in range(0, n, chunk):
                end = min(start + chunk, n)
                M_chunk = w[i_idx[start:end, None] + i_idx[None, :]]
                g[start:end] = 2.0 * M_chunk.dot(a)
        
        # Remove mean to maintain sum constraint
        g -= np.mean(g)
        
        # Momentum update
        m = momentum * m + (1.0 - momentum) * g
        a = a - lr * m
        
        # Project and normalize
        a = np.clip(a, floor, None)
        S = np.sum(a)
        a = a / S
        
        it += 1
        
        # Periodic scoring
        if it % 50 == 0:
            s = score_fast(a)
            if s < best_score:
```

```python
"""
Smoothed gradient descent with log-sum-exp for minimizing peak autocorrelation
of non-negative sequences. Uses FFT convolution, matrix-based gradient computation,
adaptive temperature scheduling, momentum, and multi-start across lengths/initializations.
Includes peak-shaving local search as a final refinement phase.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def make_hann(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_cosine(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * t / n) ** 2
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_gaussian(n, rng):
    t = np.arange(n, dtype=np.float64)
    center = n / 2.0
    sigma = n / 4.0
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_bell(n, rng):
    t = np.arange(n, dtype=np.float64)
    center = n / 2.0
    sigma = n / 3.0
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2) + 0.1 * np.exp(-0.5 * ((t - center) / (sigma * 0.4)) ** 2)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_smooth_random(n, rng):
    """Low-pass filtered random signal."""
    a = rng.random(n)
    # Simple moving average smoothing
    k = max(3, n // 16)
    kernel = np.ones(k) / k
    from scipy.signal import convolve as sig_convolve
    a = sig_convolve(a, kernel, mode='same')
    a = a / max(a.max(), 1e-10)
    a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_chirp(n, rng):
    t = np.arange(n, dtype=np.float64)
    window = 0.5 * (1.0 + np.cos(np.pi * t / max(1, n - 1)))
    a = 0.3 + 0.5 * window + 0.2 * np.cos(2.0 * np.pi * t * (t + 1) / (2.0 * n))
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init(n, init_type, rng):
    """Generate initial sequence of length n."""
    if init_type == 'hann':
        return make_hann(n, rng)
    elif init_type == 'cosine':
        return make_cosine(n, rng)
    elif init_type == 'gaussian':

    elif init_type == 'bell':

    elif init_type == 'random_smooth':

    elif init_type == 'chirp':

def smoothed_grad_descent(a_init, time_limit, T_init=0.5, T_min=0.003,
                          lr_init=0.08, momentum=0.9, floor=0.001,
                          verbose=True, seed=0):
    """
    Smoothed gradient descent using log-sum-exp approximation of max.
    Normalizes sequence to sum=1 at each step (scale invariance).
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_score = score_fast(best_a)
    t0 = time.monotonic()
    m = np.zeros(n)
    it = 0
    rng = np.random.default_rng(seed)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 0.5:
            break

        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        # Temperature: exponential decay from T_init to T_min
        T = max(T_min, T_init * np.exp(-4.0 * progress))

        # Learning rate: cosine schedule with warmup
        warmup = min(1.0, it / 50.0)
        cosine = 0.5 * (1.0 + np.cos(np.pi * progress))
        lr = lr_init * warmup * max(cosine, 0.01)

        # Compute convolution
        c = fft_convolve(a, a)

        # Softmax weights (numerically stable)
        c_max = np.max(c)
        exp_c = np.exp((c - c_max) / T)
        w_exp = exp_c / np.sum(exp_c)

        # Gradient: g[i] = 2 * sum_j a[j] * w[i+j]
        # Using matrix formulation: M[i,j] = w[i+j], g = 2 * M @ a
        i_idx = np.arange(n)
        g = 2.0 * w_exp[i_idx[:, None] + i_idx[None, :]].dot(a)

        # Remove mean to maintain sum constraint
        g -= np.mean(g)

        # Momentum update
        m = momentum * m + (1.0 - momentum) * g
        a = a - lr * m

        # Project to non-negative and renormalize
        a = np.clip(a, floor, None)
        S = np.sum(a)
        a = a / S

        it += 1

        # Periodic scoring
        if it % 100 == 0:
            s = score_fast(a)
            if s < best_score - 1e-10:
                best_score = s
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            if verbose and it % 500 == 0:
                print(f"    [{elapsed:6.1f}s] it={it:6d} T={T:.4f} lr={lr:.5f} score={s:.6f} best={best_score:.6f}")

            a += 0.02 * rng.standard_normal(n)
            a = np.clip(a, floor, None)
            a = a / np.sum(a)
            m *= 0.5
            no_improve = 0

    s = score_fast(a)
    if s < best_score:
        best_score = s
        best_a = a.copy()

    return best_a, best_score

            t_alloc = deadline - time.monotonic() - 120
        if t_alloc < 10:
            break

        a0 = make_init(n, init_type, rng)
        elapsed_so_far = time.monotonic() - t_start
        print(f"  [{elapsed_so_far:6.1f}s] n={n:4d} init={init_type:10s} t={t_alloc:.0f}s")

        a_best, s_best = smoothed_grad_descent(
            a0, t_alloc, T_init=0.5, T_min=0.003,
            lr_init=0.08, momentum=0.9, verbose=False
        )

        if s_best < best_score:
            best_score = s_best
            best_seq = a_best.copy()
            print(f"  *** NEW BEST: n={n} score={best_score:.6f} ***")
        else:
            print(f"  score={s_best:.6f}")

    print(f"\nAfter multi-start: best={best_score:.6f} n={len(best_seq) if best_seq is not None else '?'}")

    # Phase 2: Two-phase refine on best
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 60
        if remaining > 30:
            t2 = min(150.0, remaining)
            print(f"\n=== Phase 2: Two-phase refine ({t2:.0f}s) ===")
            a_r, s_r = two_phase_optimize(best_seq, t2, verbose=True)
            if s_r < best_score:
                best_score = s_r
                best_seq = a_r.copy()
                print(f"  Refined: {best_score:.6f}")

    # Phase 3: Peak shaving
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 30
        if remaining > 20:
            t3 = min(80.0, remaining)
            print(f"\n=== Phase 3: Peak shaving ({t3:.0f}s) ===")
            a_p, s_p = peak_shave(best_seq, t3, verbose=True)
            if s_p < best_score:
                best_score = s_p
                best_seq = a_p.copy()
                print(f"  After peak shaving: {best_score:.6f}")

    # Phase 4: Final ultra-low-T gradient pass
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 10
        if remaining > 10:
            t4 = min(50.0, remaining)
            print(f"\n=== Phase 4: Final polish ({t4:.0f}s) ===")
            a_f, s_f = smoothed_grad_descent(
                best_seq, t4, T_init=0.02, T_min=0.001,
                lr_init=0.01, momentum=0.95, verbose=True
            )
            if s_f < best_score:
                best_score = s_f
                best_seq = a_f.copy()
                print(f"  Final: {best_score:.6f}")

    # Finalize
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*50}")
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'='*50}")
        return final_list

    return [1.0] * 256
```
