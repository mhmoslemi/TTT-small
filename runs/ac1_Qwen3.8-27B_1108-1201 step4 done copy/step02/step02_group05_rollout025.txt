Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using mirror descent with adaptive
step sizes, pairwise coordinate descent, and soft-max smoothing.
Explores multiple sequence lengths and diverse initializations.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_value(p, n=None):
    """Compute 2*n*max(conv(p,p)) with sum(p)=1 assumed."""
    if n is None:
        n = len(p)
    conv = fast_convolve(p)
    return 2.0 * n * conv.max()


def mirror_descent_step(p, eta, top_k=3):
    """One mirror descent step: multiplicative update on simplex."""
    n = len(p)
    conv = fast_convolve(p)
    
    # Get top k lags
    top_indices = np.argsort(conv)[-top_k:][::-1]
    weights = np.ones(top_k)  # equal weight for now
    # Could weight by peak value
    peak_vals = conv[top_indices]
    weights = peak_vals / peak_vals.sum()
    
    # Compute weighted subgradient
    g = np.zeros(n)
    for w, k in zip(weights, top_indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi+1] += w * 2.0 * p[k - np.arange(lo, hi+1)]
    
    # Mirror descent update (KL projection on simplex)
    log_p = np.log(p + 1e-300) - eta * g
    log_p -= log_p.max()
    p_new = np.exp(log_p)
    p_new /= p_new.sum()
    
    return p_new, conv, top_indices

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using mirror descent with softmax
smoothing and annealing, followed by pairwise coordinate descent.
"""

import numpy as np
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_gradient_soft(p, T):
    """Compute soft gradient using softmax weighting."""
    n = len(p)
    conv = fast_convolve(p)
    # Softmax weights
    Tmax = conv.max()
    exp_conv = np.exp(T * (conv - Tmax))  # shifted for stability
    w = exp_conv / exp_conv.sum()
    
    # Gradient: g[i] = 2 * sum_k w[k] * p[k-i]
    g = np.zeros(n)
    for k in range(len(conv)):
        if w[k] < 1e-10:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi+1] += 2.0 * w[k] * p[k - np.arange(lo, hi+1)]
    
    return g, conv


def mirror_descent_optimize(p0, time_limit=60.0, T_start=1.0, T_end=100.0, eta_init=0.2):
    n = len(p0)
    p = np.maximum(p0, 0.0).astype(np.float64).copy()
    s = p.sum()
    if s < 1e-12:
        p = np.ones(n) / n
    else:
        p /= s
    
    best_p = p.copy()
    conv = fast_convolve(p)
    best_obj = 2.0 * n * conv.max()
    
    t0 = time.monotonic()
    eta = eta_init
    total_iters = 5000
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and it < total_iters and no_improve < 500:
        # Anneal T
        progress = min(1.0, it / total_iters)
        T = T_start + (T_end - T_start) * progress
        
        # Compute gradient
        g, conv = compute_gradient_soft(p, T)
        
        # Mirror descent step
        log_p = np.log(p + 1e-300) - eta * g
        log_p -= log_p.max()
        p_new = np.exp(log_p)
        p_new /= p_new.sum()
        
        # Evaluate
        conv_new = fast_convolve(p_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_p = p_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        p = p_new
        it += 1
        
        # Adaptive step size
        if no_improve > 100:
            eta *= 0.5
            no_improve = 0
    
    return best_p, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using mirror descent with softmax
annealing, efficient pairwise local search with O(n) convolution updates,
and diverse multi-length exploration. Avoids LP for speed.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_value(p, n=None):
    """Objective value assuming sum(p)=1."""
    if n is None:
        n = len(p)
    return 2.0 * n * fast_convolve(p).max()


def compute_soft_gradient(p, T):
    """Compute soft gradient: g[i] = 2 * sum_k softmax(T*conv)[k] * p[k-i]."""
    n = len(p)
    conv = fast_convolve(p)
    conv_max = conv.max()
    exp_conv = np.exp(T * (conv - conv_max))
    w = exp_conv / exp_conv.sum()
    
    # g[i] = 2 * sum_{r=0}^{n-1} w[i+r] * p[r]  (cross-correlation)
    # Using numpy correlate: np.correlate(w, p, 'full')[i + n - 1]
    corr = np.correlate(w, p, mode='full')
    g = 2.0 * corr[n - 1: n - 1 + n]
    
    return g, conv


def mirror_descent_optimize(p0, time_limit=60.0, T_start=0.5, T_end=80.0, eta_init=0.3, total_iters=20000):
    """Mirror descent with softmax annealing on the probability simplex."""
    n = len(p0)
    p = np.maximum(p0.astype(np.float64), 0.0).copy()
    s = p.sum()
    if s < 1e-12:
        p = np.ones(n) / n
    else:
        p /= s
    
    conv = fast_convolve(p)
    best_p = p.copy()
    best_obj = 2.0 * n * conv.max()
    
    t0 = time.monotonic()
    eta = eta_init
    it = 0
    no_improve = 0
    patience = 300
    
    while time.monotonic() - t0 < time_limit and it < total_iters and no_improve < patience:
        progress = min(1.0, float(it) / float(total_iters))
        T = T_start + (T_end - T_start) * progress
        
        g, conv = compute_soft_gradient(p, T)
        
        # Mirror descent update (KL projection on simplex)
        log_p = np.log(p + 1e-300) - eta * g
        log_p -= log_p.max()
        p_new = np.exp(log_p)
        p_new /= p_new.sum()
        
        conv_new = fast_convolve(p_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_p = p_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        p = p_new
        it += 1
        
        if no_improve > 100:
            eta *= 0.6
            no_improve = 0
    
    return best_p, best_obj


def conv_after_transfer(conv, p, i, j, delta):
    """Compute new convolution after transferring delta from index i to index j. O(n) time."""
    n = len(p)
    size = 2 * n - 1
    # new_conv[k] = conv[k] + 2*delta*(p[k-j] - p[k-i]) + delta^2*c[k]
    pi = np.zeros(size)
    pj = np.zeros(size)
    pi[i:i + n] = p
    pj[j:j + n] = p
    
    new_conv = conv + 2.0 * delta * (pj - pi)
    
    c = np.zeros(size)
    if 2 * i < size:
        c[2 * i] += 1.0
    if 2 * j < size:
        c[2 * j] += 1.0
    if i + j < size:
        c[i + j] -= 2.0
    new_conv += delta * delta * c
    
    return new_conv


def local_search_pq(p, time_limit=60.0, max_passes=100):
    """Pairwise local search: transfer mass from high-peak-contributing to low ones."""
    n = len(p)
    p = np.maximum(p.astype(np.float64), 0.0).copy()
    s = p.sum()
    if s < 1e-12:
        return p, np.inf
    p /= s
    
    conv = fast_convolve(p)
    best_obj = 2.0 * n * conv.max()
    best_p = p.copy()
    t0 = time.monotonic()
    stuck = 0
    
    while time.monotonic() - t0 < time_limit and stuck < 50:
        k_star = int(np.argmax(conv))
        
        # Compute contribution of each element to the peak
        contrib = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        contrib[lo:hi + 1] = p[k_star - np.arange(lo, hi + 1)]
        
        # Sources: high contribution; Targets: low contribution
        order = np.argsort(contrib)
        n_try = max(5, n // 8)
        targets = order[:n_try]
        sources = order[-n_try:]
        
        improved = False
        for si in sources:
            if p[si] < 1e-12:
                continue
            for ti in targets:
                if si == ti:
                    continue
                
                best_delta = 0.0
                best_local = best_obj
                
                # Try several transfer fractions
                for frac in [0.03, 0.08, 0.15, 0.25, 0.4, 0.65]:
                    delta = p[si] * frac
                    if delta < 1e-14:
                        continue
                    new_conv = conv_after_transfer(conv, p, int(si), int(ti), delta)
                    new_peak = new_conv.max()
                    new_obj = 2.0 * n * new_peak
                    if new_obj < best_local - 1e-13:
                        best_local = new_obj
                        best_delta = delta
                
                if best_delta > 1e-12:
                    p[int(si)] -= best_delta
                    p[int(ti)] += best_delta
                    conv = conv_after_transfer(conv, p, int(si), int(ti), -best_delta)
                    # Recompute conv from scratch to avoid accumulated errors
                    conv = fast_convolve(p)
                    best_obj = best_local
                    best_p = p.copy()
                    improved = True
        
        if improved:
            stuck = 0
        else:
            stuck += 1
    
    return best_p, best_obj

def make_initializations(n, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    for sf in [0.15, 0.3, 0.5, 0.7]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        s = g.sum()
        if s > 1e-10:
            inits[f'gauss_{sf}'] = g / s
    
    # Chirp-based (non-negative after abs)
    phase = 2.0 * np.pi * (x ** 2) / n
    chirp = np.abs(np.cos(phase)) ** 0.5
    s = chirp.sum()
    if s > 1e-10:
        inits['chirp'] = chirp / s
    
    # Sparse random
    alpha = max(0.05, 1.0 / n * 3)
    r = rng.dirichlet(np.ones(n) * alpha)
    inits['sparse'] = r
    
    # Edge-concentrated
    edges = np.zeros(n)
    m = max(1, n // 6)
    edges[:m] = 1.0
    edges[-m:] = 1.0
    inits['edges'] = edges / edges.sum()
    
    # Comb pattern
    comb = np.zeros(n)
    step = max(2, n // 20)
    comb[::step] = 1.0
    inits['comb'] = comb / comb.sum()
    
    # Parent
    parent = get_parent_resized(n)
    if parent is not None:
        inits['parent'] = parent
    
    return inits

    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_sequence = None
    best_score = float('inf')
    
    lengths = [256, 512, 1024, 2048]
    
    print(f"Budget: {budget_s}s, deadline in {deadline - time.monotonic():.1f}s")
    
    # Phase 1: Mirror descent exploration (70% of time)
    print("Phase 1: Mirror descent exploration...")
    phase1_end = time.monotonic() + (float(budget_s) - 30.0) * 0.70
    
    all_configs = []
    for n in lengths:
        inits = make_initializations(n, rng)
        for name, a0 in inits.items():
            all_configs.append((n, name, a0))
    
    # Priority: parent first, then shorter lengths, then diverse inits
    def config_priority(item):
        n, name, _ = item
        if name == 'parent':
            return (0, n)
        if name in ['gauss_0.3', 'chirp', 'sparse']:
            return (1, n)
        return (2, n)
    
    all_configs.sort(key=config_priority)
    
    # Allocate time per config
    n_configs = len(all_configs)
    time_per_config = max(5.0, (phase1_end - time.monotonic()) / n_configs)
    
    phase1_results = []
    for idx, (n, name, a0) in enumerate(all_configs):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        
        t_alloc = min(time_per_config, 120.0, remaining - 60.0)
        if t_alloc < 3.0:
            break
        
        # Run mirror descent
        a_opt, obj_opt = soft_mirror_descent(a0, time_limit=t_alloc, T_schedule='linear', eta_init=0.2)
        
        # Quick local search
        ls_time = min(10.0, t_alloc * 0.3, remaining - 60.0)
        if ls_time > 2.0:
            a_ls, obj_ls = local_search_pq(a_opt, time_limit=ls_time)
            if obj_ls < obj_opt:
                a_opt = a_ls
                obj_opt = obj_ls
        
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            phase1_results.append((score, n, name, a_opt))
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P1: n={n}, {name}, score={score:.6f} (best so far)")
    
    print(f"  Phase 1 done. Best: {best_score:.6f}, found {len(phase1_results)} valid results")
    
    # Phase 2: Deep optimization on top candidates (20% of time)
    print("Phase 2: Deep optimization on top candidates...")
    phase2_end = time.monotonic() + (float(budget_s) - 30.0) * 0.20
    
    phase1_results.sort(key=lambda x: x[0])
    top_candidates = []
    seen_keys = set()
    for score, n, name, a in phase1_results:
        key = (n, name)
        if key not in seen_keys and len(top_candidates) < 6:
            top_candidates.append((n, name, a))
            seen_keys.add(key)
    
    if best_sequence is not None:
        top_candidates.append((len(best_sequence), 'best', best_sequence.copy()))
    
    remaining_p2 = max(5.0, phase2_end - time.monotonic())
    time_per_deep = remaining_p2 / max(1, len(top_candidates))
    
    for n, name, a0 in top_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        
        t_md = min(time_per_deep * 0.6, remaining - 30.0)
        if t_md < 5.0:
            break
        
        # Deeper mirror descent with exponential T schedule
        a_opt, obj_opt = soft_mirror_descent(a0, time_limit=t_md, T_schedule='exponential', eta_init=0.15)
        
        # Local search
        t_ls = min(time_per_deep * 0.3, remaining - 30.0)
        if t_ls > 3.0:
            a_ls, obj_ls = local_search_pq(a_opt, time_limit=t_ls)
            if obj_ls < obj_opt:
                a_opt = a_ls
                obj_opt = obj_ls
        
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  P2: n={n}, {name}, score={score:.6f}")
    
    print(f"  Phase 2 done. Best: {best_score:.6f}")
    
    # Phase 3: Intensive polish on best (remaining time)
    print("Phase 3: Intensive polish...")
    if best_sequence is not None:
        polish_deadline = deadline - 15.0
        polish_iter = 0
        while time.monotonic() < polish_deadline:
            remaining = polish_deadline - time.monotonic()
            if remaining < 5.0:
                break
            
            # Alternate between mirror descent and local search
            if polish_iter % 2 == 0:
                t_alloc = min(remaining * 0.4, 60.0)
                a_opt, _ = soft_mirror_descent(best_sequence, time_limit=t_alloc,
                                               T_schedule='exponential', eta_init=0.1)
            else:
                t_alloc = min(remaining * 0.4, 60.0)
                a_opt, _ = local_search_pq(best_sequence, time_limit=t_alloc)
            
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P3 iter {polish_iter}: score={score:.6f}")
            
            polish_iter += 1
    
    # Final check
    if best_sequence is None:
        best_sequence = np.ones(512)
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score
    
    print(f"FINAL: score={best_score:.6f}, length={len(best_sequence)}")
    return [float(x) for x in best_sequence]
