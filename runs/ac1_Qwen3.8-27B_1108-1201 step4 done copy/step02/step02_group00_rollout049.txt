import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)

    A = np.fft.rfft(a, fft_size)

def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]

    if not rho.size:

    return np.maximum(v - theta, 0.0)

    # Constraints: sum(a) = 1, a >= 0

def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Asymmetric ramps
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp_inc'] = ramp
    inits['ramp_dec'] = ramp[::-1].copy()
    
    # Quadratic
    quad = x * x + 1.0
    quad /= quad.sum()
    inits['quad'] = quad
    
    # Exponential decay (asymmetric)
    for rate in [0.5, 1.0, 2.0]:
        exp_d = np.exp(-rate * x / n)
        exp_d /= exp_d.sum()
        inits[f'exp_{rate}'] = exp_d
        inits[f'exp_{rate}_r'] = exp_d[::-1].copy()
    
    # Bimodal asymmetric
    w = max(1, n // 5)
    bi = np.zeros(n)
    bi[:w] = 1.0
    bi[-w//2:] = 1.0
    bi /= bi.sum()
    inits['bimodal_asym'] = bi
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_crop'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                         np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                         np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps
    
    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search:
    Phase 1: Quick scan of lengths and initializations with FISTA
    Phase 2: Deep FISTA optimization on top candidates
    Phase 3: Scipy polish
    Phase 4: Final verification
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: len={len(p)}, sum={p.sum():.2f}")
            # Evaluate parent
            ps = float(evaluate_sequence(p.tolist()))
            if np.isfinite(ps):
                best_score = ps
                best_sequence = p.copy()
                print(f"Parent score: {best_score:.6f}")
    except Exception as e:
        print(f"No parent: {e}")
    
    # Phase 1: Quick scan
    print("=== Phase 1: Quick Scan ===")
    lengths = [128, 192, 256, 320, 384, 512]
    
    scan_results = []
    phase1_deadline = time.monotonic() + min(400.0, budget_s * 0.25)
    
    for n in lengths:
        if time.monotonic() > phase1_deadline - 20:
            break
        inits = generate_inits(n, parent, rng)
        init_names = list(inits.keys())
        
        # Priority ordering
        priority = ['parent_crop', 'parent_ds', 'parent_up', 'ramp_inc',
                    'ramp_dec', 'uniform', 'exp_1.0', 'exp_1.0_r',
                    'quad', 'random', 'bimodal_asym']
        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in init_names if nm not in priority]
        init_names = init_names[:8]
        
        time_per = max(10.0, 250.0 / (len(lengths) * 5))
        
        for name in init_names:
            if time.monotonic() > phase1_deadline - 10:
                break
            t_alloc = min(time_per, phase1_deadline - time.monotonic() - 5)
            if t_alloc < 5:
                break
            
            a0 = inits[name]
            a_opt, obj_opt = fista_smoothed_v2(
                a0, time_limit=t_alloc,
                beta_schedule=[10, 50, 200, 1000, 5000])
            
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Scan: n={n}, {name}, score={score:.6f}")
    
    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}")
    
    # Phase 2: Deep optimization
    print("=== Phase 2: Deep Optimization ===")
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen and len(top_configs) < 10:
            top_configs.append((n, name, a_opt))
            seen.add(key)
    
    # Always include current best
    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'best', best_sequence.copy()))
    
    remaining = deadline - time.monotonic() - 120.0
    time_per_deep = max(20.0, remaining / max(1, len(top_configs)))
    
    for i, (n, name, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 100:
            break
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 80)
        if t_alloc < 10:
            break
        
        # Use deeper beta schedule
        a_opt, obj_opt = fista_smoothed_v2(
            a0, time_limit=t_alloc,
            beta_schedule=[30, 100, 500, 2000, 8000, 30000])
        
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Deep {i}: n={n}, {name}, score={score:.6f}")
    
    print(f"Phase 2 best: {best_score:.6f}")
    
    # Phase 3: Scipy polish
    print("=== Phase 3: Scipy Polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 30.0
        n_best = len(best_sequence)
        
        # Multiple scipy attempts from slightly different starting points
        for attempt in range(min(5, int(remaining / 30) + 1)):
            if time.monotonic() > deadline - 20:
                break
            t_alloc = min(60.0, remaining - 10)
            if t_alloc < 10:
                break
            
            if attempt == 0:
                a_start = best_sequence.copy()
            elif attempt == 1:
                noise = rng.standard_normal(n_best) * 0.002
                a_start = project_simplex(best_sequence + noise)
            elif attempt == 2:
                # Slightly asymmetric perturbation
                drift = np.linspace(-0.001, 0.001, n_best)
                a_start = project_simplex(best_sequence + drift)
            else:
                noise = rng.standard_normal(n_best) * 0.001
                a_start = project_simplex(best_sequence + noise)
            
            a_opt, obj_opt = scipy_polish(a_start, time_limit=t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Scipy {attempt}: score={score:.6f}")
            remaining = deadline - time.monotonic() - 30.0
    
    # Phase 4: Final FISTA polish with very high beta
    print("=== Phase 4: Final Polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 15.0
        if remaining > 20:
            a_opt, obj_opt = fista_smoothed_v2(
                best_sequence, time_limit=remaining,
                beta_schedule=[100, 500, 2000, 10000, 50000])
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final FISTA: score={score:.6f}")
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    
    # Ensure all values are valid
    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    
    print(f"\nFINAL: score={best_score:.8f}, n={len(best_sequence)}, "
          f"sum={best_sequence.sum():.4f}")
    return [float(x) for x in best_sequence]
