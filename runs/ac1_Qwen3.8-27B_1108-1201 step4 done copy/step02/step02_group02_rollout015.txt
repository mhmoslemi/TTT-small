    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

    if s < 0.01:

    a: sequence of length n
    p: probability vector of length 2n-1 (softmax of b/t)

    j_idx = np.arange(n)[None, :]
    P = p[i_idx + j_idx]  # P[i, j] = p[i+j]
    g = 2.0 * (a @ P)

    rho = rho_list[-1]

Combines: smoothed gradient descent, block coordinate descent, peak-driven adjustments,
and simulated annealing.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n+m-1]

def eval_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = np.sum(a)
    if s < 0.01 or n == 0:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s*s)

def normalize_to_sum(a, target=1.0):
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    return a * (target / s)

def simplex_project(v):

    u = np.sort(v)[::-1]
    cs = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cs - 1)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]

    return np.maximum(v - theta, 0)

    j_idx = np.arange(n)[None, :]
    P = p[i_idx + j_idx]
    return 2.0 * (a @ P)

def smooth_ls(a0, time_limit, t0=5.0, t_min=0.01, verbose=True):
    """Smooth log-sum-exp gradient descent with FISTA on simplex."""
    n = len(a0)
    a = simplex_project(a0 / max(np.sum(a0), 1e-10))
    a_prev = a.copy()
    alpha = 1.0
    t = t0
    best_a = a.copy()
    best_score = eval_score(a)
    t_start = time.monotonic()
    
    # Adaptive t schedule
    n_expected = int(time_limit / 0.1)  # ~10 iterations per 0.1s
    t_ratio = (t_min / t0) ** (1.0 / max(n_expected, 1))
    
    it = 0
    while time.monotonic() - t_start < time_limit:
        b = fft_convolve(a, a)
        b_max = np.max(b)
        exp_b = np.exp((b - b_max) / t)
        p = exp_b / np.sum(exp_b)
        
        # FISTA: evaluate gradient at extrapolated point
        y = simplex_project(a + ((alpha - 1) / (0.5 * (1 + np.sqrt(1 + 4*alpha*alpha)))) * (a - a_prev))
        # Simpler: just use a directly
        y = a
        b_y = fft_convolve(y, y)
        b_y_max = np.max(b_y)
        exp_by = np.exp((b_y - b_y_max) / t)
        p_y = exp_by / np.sum(exp_by)
        grad = compute_smoothing_grad(y, p_y)
        
        # Step size: 1/L where L is Lipschitz constant
        # L ~ 2 * sum(p) * max(a) * n / t = 2 * max(a) * n / t (rough)
        L = 4.0 * np.max(a) * n / t + 1.0
        step = 1.0 / L
        
        # Backtracking
        f_y = b_y_max + t * np.log(np.sum(exp_by))
        trial = simplex_project(y - step * grad)
        b_t = fft_convolve(trial, trial)
        b_t_max = np.max(b_t)
        f_t = b_t_max + t * np.log(np.sum(np.exp((b_t - b_t_max) / t)))
        
        c1 = 0.5
        while f_t > f_y - c1 * step * np.dot(grad, grad) and step > 1e-10:
            step *= 0.5
            trial = simplex_project(y - step * grad)
            b_t = fft_convolve(trial, trial)
            b_t_max = np.max(b_t)
            f_t = b_t_max + t * np.log(np.sum(np.exp((b_t - b_t_max) / t)))
        
        a_prev = a.copy()
        a = trial
        alpha_new = 0.5 * (1 + np.sqrt(1 + 4 * alpha * alpha))
        a_prev = simplex_project(a + ((alpha - 1) / alpha_new) * (a - a_prev))
        alpha = alpha_new
        
        t *= t_ratio
        t = max(t, t_min)
        it += 1
        
        if it % 200 == 0:
            score = eval_score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
            if verbose:
                print(f"    sgd iter={it}, t={t:.5f}, score={score:.6f}, best={best_score:.6f}")
    
    score = eval_score(a)
    if score < best_score:
        best_score = score
        best_a = a.copy()
    return best_a, best_score

def block_cd(a0, time_limit, block_size=32, verbose=True):
    """Block coordinate descent: optimize blocks of coordinates at a time."""
    n = len(a0)
    a = a0.copy()
    a = normalize_to_sum(a, 1.0)
    best_a = a.copy()
    best_score = eval_score(a)
    t_start = time.monotonic()
    n_blocks = (n + block_size - 1) // block_size
    
    it = 0
    while time.monotonic() - t_start < time_limit:
        for bi in range(n_blocks):
            if time.monotonic() - t_start > time_limit:
                break
            start = bi * block_size
            end = min(start + block_size, n)
            
            # Optimize a[start:end] while keeping rest fixed
            # UseSciPy minimize with a few parameters
            a_fixed = a.copy()
            a_free = a[start:end].copy()
            
            def obj(x):
                a_temp = a_fixed.copy()
                a_temp[start:end] = np.maximum(x, 0)
                s = np.sum(a_temp)
                if s < 0.01:
                    return 1e10
                c = fft_convolve(a_temp, a_temp)
                return np.max(c) / (s*s)
            
            res = minimize(obj, a_free, method='L-BFGS-B', 
                          bounds=[(0, None)] * len(a_free),
                          options={'maxiter': 50, 'gtol': 1e-8})
            
            if res.fun < obj(a_free):
                a[start:end] = np.maximum(res.x, 0)
                a = normalize_to_sum(a, 1.0)
                it += 1
        
        score = eval_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        if verbose and it % 10 == 0:
            print(f"    bcd iter={it}, score={score:.6f}, best={best_score:.6f}")
    
    return best_a, best_score

def peak_reduction(a0, time_limit, verbose=True):
    """Direct peak reduction: identify peak contributors and redistribute."""
    n = len(a0)
    a = normalize_to_sum(a0, 1.0)
    best_a = a.copy()
    best_score = eval_score(a)
    t_start = time.monotonic()
    
    it = 0
    while time.monotonic() - t_start < time_limit:
        c = fft_convolve(a, a)
        peak_val = np.max(c)
        peak_indices = np.where(c >= peak_val * 0.999)[0]
        
        improved = False
        for pk in peak_indices:
            # Find contributing pairs
            pairs = []
            for j in range(max(0, pk - n + 1), min(n, pk + 1)):
                k = pk - j
                if 0 <= k < n and j != k:
                    pairs.append((j, k, a[j] * a[k]))
            pairs.sort(key=lambda x: -x[2])
            
            # Try reducing the top contributors
            for (j1, j2, val) in pairs[:5]:
                # Reduce a[j1] slightly, increase a[other]
                delta = 0.01 * a[j1]
                a[j1] -= delta
                # Add to a random position far from peak
                add_pos = (j1 + n // 2) % n
                a[add_pos] += delta
                a = np.clip(a, 0, None)
                a = normalize_to_sum(a, 1.0)
                
                new_score = eval_score(a)
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
                    improved = True
                else:
                    a[j1] += delta
                    a[add_pos] -= delta
                    a = normalize_to_sum(a, 1.0)
            it += 1
        
        if not improved and it > 100:
            break
    
    return best_a, best_score

def simulated_annealing(a0, time_limit, verbose=True, seed=42):
    """SA polish."""
    rng = np.random.default_rng(seed)
    n = len(a0)

        if r < 0.4:

            pert[add] *= np.exp(rng.normal(0.03, 0.02, len(add)))
        elif r < 0.7:

            s = rng.integers(0, n - w)
            pert[s:s+w] *= np.exp(rng.normal(0, 0.03, w))
        else:
            # Small global
            pert *= np.exp(rng.normal(0, 0.01, n))
        
        pert = np.clip(pert, 0, None)
        pert = normalize_to_sum(pert, 1.0)
        new_score = eval_score(pert)
        
        if new_score < curr_score or rng.random() < np.exp(-(new_score - curr_score) / temp):
            a = pert
            curr_score = new_score
            if curr_score < best_score:
                best_score = curr_score
                best_a = a.copy()

        it += 1
        if verbose and it % 5000 == 0:
            print(f"    sa iter={it}, temp={temp:.6f}, curr={curr_score:.6f}, best={best_score:.6f}")
    
    return best_a, best_score

"""
Multi-phase optimization for minimizing peak autocorrelation of non-negative
step functions. Uses log-sum-exp smoothed gradient descent with FISTA
acceleration, adaptive temperature annealing, multi-length search, and
simulated annealing polish.
"""

import numpy as np
import time

def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n+m-1]

def eval_score(a):
    """Compute the objective score (fast local version)."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (s * s))

def normalize(a, target=1.0):
    """Normalize sequence to have given sum."""
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a * (target / s)

def simplex_project(v):
    """Euclidean projection onto the probability simplex {a>=0, sum(a)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cs = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cs - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cs[r] - 1.0) / (r + 1)
    return np.maximum(v - theta, 0.0)

def gradient_smooth(a, p):
    """Compute gradient of t*log(sum exp(b_k/t)) w.r.t. a.
    
    a: sequence, length n
    p: softmax probabilities, length 2n-1
    Returns: gradient, length n
    """
    n = len(a)
    # g_i = 2 * sum_j a[j] * p[i+j]
    # = 2 * (reversed_a * p)[i + n - 1]
    L = 1
    while L < n + (2*n - 1):
        L <<= 1
    Ar = np.fft.rfft(a[::-1], L)
    P = np.fft.rfft(p, L)
    conv = np.fft.irfft(Ar * P, L)
    return 2.0 * conv[n-1:2*n-1]

def smooth_fista(a0, time_limit, t0=8.0, t_min=0.005, verbose=True, seed=0):
    """FISTA on smoothed log-sum-exp objective over the simplex."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = simplex_project(normalize(a0, 1.0))
    a_prev = a.copy()
    alpha = 1.0
    t = t0
    best_a = a.copy()
    best_score = eval_score(a)
    t_start = time.monotonic()
    
    # Pre-compute: expected iterations ~ time_limit / 0.005s per iter
    it = 0
    patience = 0
    
    while time.monotonic() - t_start < time_limit:
        # Compute softmax weights
        b = fft_convolve(a, a)
        b_max = np.max(b)
        # Avoid overflow
        shifted = (b - b_max) / t
        shifted = np.clip(shifted, -50, 50)
        exp_b = np.exp(shifted)
        p = exp_b / np.sum(exp_b)
        
        # Gradient at 'a' (simplified FISTA without extrapolation for stability)
        grad = gradient_smooth(a, p)
        
        # Lipschitz constant estimate
        L = 4.0 * np.max(a) * (2.0 * n) / t + 1.0
        step = 1.0 / L
        
        # Backtracking line search
        f_val = b_max + t * np.log(np.sum(exp_b))
        c1 = 1e-4  # Armijo parameter
        
        for _ in range(20):
            a_new = simplex_project(a - step * grad)
            b_new = fft_convolve(a_new, a_new)
            b_new_max = np.max(b_new)
            shifted_new = np.clip((b_new - b_new_max) / t, -50, 50)
            f_new = b_new_max + t * np.log(np.sum(np.exp(shifted_new)))
            
            if f_new <= f_val - c1 * step * np.dot(grad, grad):
                break
            step *= 0.5
        
        # FISTA momentum update
        a_prev_new = a + ((alpha - 1.0) / (0.5 * (1.0 + np.sqrt(1.0 + 4.0 * alpha * alpha)))) * (a - a_prev)
        a_prev_new = simplex_project(a_prev_new)
        a = a_new
        a_prev = a_prev_new
        alpha = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * alpha * alpha))
        
        # Anneal temperature
        t *= 0.9995
        t = max(t, t_min)
        it += 1
        
        # Track best
        if it % 50 == 0:
            score = eval_score(a)
            if score < best_score:
                best_score = score
