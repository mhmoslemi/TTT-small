"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Projected subgradient descent on the simplex
2. LP-based direction finding (periodic)
3. Multi-coordinate local search
4. Spectral and structured initializations

import numpy as np
import scipy.optimize
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_value(a):
    """Objective with sum(a)=1 assumed."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    # Algorithm from Wang & Carreira-Perpinan (2013)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def subgradient_direction(a, conv):
    """
    Compute subgradient of max_k conv[k] at point a.
    Returns the gradient for the steepest peak, and also a weighted
    subgradient for all near-peaks.
    """
    n = len(a)
    max_val = conv.max()
    # Find near-peaks (within 5% of max)
    near_peak_mask = conv >= 0.95 * max_val
    near_peaks = np.where(near_peak_mask)[0]
    
    if len(near_peaks) == 0:
        near_peaks = [int(np.argmax(conv))]
    
    # Compute weighted subgradient
    # For lag k, grad_k(a)[i] = a[k-i] if 0 <= k-i < n
    # Weight by how close each peak is to the max
    weights = (conv[near_peaks] - 0.9 * max_val) / (max_val - 0.9 * max_val + 1e-15)
    weights = np.maximum(weights, 0)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for idx, k in enumerate(near_peaks):
        for i in range(n):
            j = int(k) - i
            if 0 <= j < n:
                grad[i] += weights[idx] * a[j]
    
    return grad


def projected_gradient_step(a, lr):
    """Take a projected gradient step."""
    conv = fast_convolve(a)
    grad = subgradient_direction(a, conv)
    a_new = project_simplex(a - lr * grad)
    return a_new, conv, grad

    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    
    def f_of_t(t):
        a_t = (1.0 - t) * a + t * direction
        a_t = np.maximum(a_t, 0.0)
        s = a_t.sum()
        if s < 1e-15:
            return np.inf
        a_t /= s
        return obj_value(a_t)
    
    fc = f_of_t(c)
    fd = f_of_t(d)
    for _ in range(max_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f_of_t(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f_of_t(d)
    
    t = (lo + hi) / 2.0
    obj_t = f_of_t(t)
    if obj_t < best_obj:
        best_t = t
        best_obj = obj_t
    
    return best_t, best_obj


def solve_lp_direction(f, M):
    """LP: find g >= 0, sum(g)=1, maximizing... actually minimizing the peak of conv(f,g)."""
    n = len(f)
    # We want to find g on the simplex that minimizes max_k conv(f,g)[k]
    # This is a min-max problem, solve via LP:
    # minimize t
    # subject to: conv(f,g)[k] <= t for all k
    #              sum(g) = 1, g >= 0
    
    # conv(f,g)[k] = sum_{i: valid} f[i] * g[k-i] (or f[k-i]*g[i])
    # Actually: conv(f,g)[k] = sum_i f[i] * g[k-i] where both indices are valid
    # = sum_i f[i] * g[k-i] for 0<=i<n, 0<=k-i<n
    # = sum_{j: max(0,k-n+1)<=j<=min(k,n-1)} f[k-j] * g[j]
    
    # So constraint k: sum_j f[k-j] * g[j] <= t for valid j
    
    # Variables: g[0],...,g[n-1], t => n+1 variables
    # Objective: minimize t => c = [0,...,0,1]
    # Constraints: A_ub * [g, t] <= b_ub
    
    num_constraints = 2 * n - 1
    # A_ub is (num_constraints) x (n+1)
    A_ub = np.zeros((num_constraints, n + 1))
    b_ub = np.zeros(num_constraints)
    
    for k in range(num_constraints):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A_ub[k, j] = f[i]
        A_ub[k, n] = -1.0  # -t
        b_ub[k] = 0.0
    
    # sum(g) = 1 => A_eq * [g, t] = b_eq
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = scipy.optimize.linprog(
            c=np.array([0.0] * n + [1.0]),
            A_ub=A_ub, b_ub=b_ub,
            A_eq=A_eq, b_eq=b_eq,
            bounds=bounds,
            method='highs',
            options={'maxiter': 5000}
        )
        if res.success:
            g = np.maximum(res.x[:n], 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return None


def gradient_optimize(a0, time_limit, verbose=False):
    """
    Iterative projected subgradient optimization with adaptive step size.
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones_like(a) / len(a)
    else:
        a /= s
    
    best_f = a.copy()
    best_obj = obj_value(a)
    t0 = time.monotonic()
    
    # Adaptive learning rate
    lr = 0.1
    min_lr = 1e-5
    max_lr = 1.0
    no_improve = 0
    max_no_improve = 100
    step_count = 0
    
    # Use Polyak step size: lr_k = (f_k - f*) / ||grad||^2
    # We don't know f*, so use a heuristic
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        max_val = conv.max()
        grad = subgradient_direction(a, conv)
        grad_norm = np.linalg.norm(grad)
        
        if grad_norm < 1e-15:
            break
        
        # Polyak-like step
        # lr = (max_val - target) / (grad_norm^2 + 1e-10)
        # where target is a lower bound (use current best * 0.9 as optimistic target)
        target = max(best_obj / (2.0 * len(a)) * 0.8, 1e-6)  # rough lower bound for max(conv)
        lr = np.clip((max_val - target) / (grad_norm ** 2 + 1e-10), min_lr, max_lr)
        
        a_new = project_simplex(a - lr * grad)
        obj_new = obj_value(a_new)
        
        if obj_new < best_obj - 1e-12:
            a = a_new
            best_obj = obj_new
            best_f = a.copy()
            no_improve = 0
        else:
            # Try smaller step
            a_new2 = project_simplex(a - lr * 0.1 * grad)
            obj_new2 = obj_value(a_new2)
            if obj_new2 < best_obj - 1e-12:
                a = a_new2
                best_obj = obj_new2
                best_f = a.copy()
                no_improve = 0
                lr *= 0.5
            else:
                no_improve += 1
                lr *= 0.7
        
        step_count += 1
        if verbose and step_count % 100 == 0:
            print(f"    Grad step {step_count}: obj={best_obj:.8f}, lr={lr:.6f}")
        
        if no_improve >= max_no_improve:
            break
    
    return best_f, best_obj


def lp_optimize(a0, time_limit):
    """LP-based iteration: find direction via LP, line search, repeat."""
    a = np maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones_like(a) / len(a)
    else:
        a /= s
    
    best_f = a.copy()
    best_obj = obj_value(a)
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        
        g = solve_lp_direction(a, M)
        if g is None:
            break
        
        t, obj_t = line_search(a, g, max_iter=40)
        if obj_t < best_obj - 1e-12:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            ns = a_new.sum()
            if ns < 1e-15:
                break
            a_new /= ns
            a = a_new
            best_obj = obj_t
            best_f = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 15:
                break
    
    return best_f, best_obj


def local_search_multi(a, time_limit):
    """
    Multi-coordinate local search: try moving mass from multiple
    high-contribution positions to multiple low-contribution positions.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    best_obj = obj_value(a)
    t0 = time.monotonic()
    stuck = 0
    max_stuck = 200
    
    while time.monotonic() - t0 < time_limit and stuck < max_stuck:
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))
        
        # Compute contribution of each pair to the peak
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]
        
        # Also consider the second peak
        conv_copy = conv.copy()
        conv_copy[k_star] = 0
        k_star2 = int(np.argmax(conv_copy))
        contrib2 = np.zeros(n)
        for i in range(n):
            j = k_star2 - i
            if 0 <= j < n:
                contrib2[i] = a[i] * a[j]
        
        # Combined pressure
        pressure = contrib + 0.5 * contrib2
        
        # Sort by pressure
        order = np.argsort(pressure)
        n_movers = max(2, n // 32)
        sources = order[-n_movers:]  # highest pressure
        targets = order[:n_movers]    # lowest pressure
        
        found = False
        for si in sources[:8]:
            if found:
                break
            if a[si] < 1e-10:
                continue
            for ti in targets[:8]:
                if found:
                    break
                if si == ti:
                    continue
                for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                    amount = a[si] * frac
                    if amount < 1e-14:
                        continue
                    a_trial = a.copy()
                    a_trial[si] -= amount
                    a_trial[ti] += amount
                    a_trial = np.maximum(a_trial, 0.0)
                    ts = a_trial.sum()
                    if ts < 1e-12:
                        continue
                    a_trial /= ts
                    o = obj_value(a_trial)
                    if o < best_obj - 1e-12:
                        best_obj = o
                        a = a_trial.copy()
                        found = True
                        break
        
        # Also try 3-point moves: move from 2 sources to 1 target
        if not found:
            for si1 in sources[:5]:
                if found:
                    break
                if a[si1] < 1e-10:
                    continue
                for si2 in sources[:5]:
                    if found:
                        break
                    if si2 == si1 or a[si2] < 1e-10:
                        continue
                    for ti in targets[:5]:
                        if found:
                            break
                        if ti == si1 or ti == si2:
                            continue
                        for frac in [0.1, 0.2, 0.3]:
                            amt1 = a[si1] * frac
                            amt2 = a[si2] * frac
                            if amt1 < 1e-14 or amt2 < 1e-14:
                                continue
                            a_trial = a.copy()
                            a_trial[si1] -= amt1
                            a_trial[si2] -= amt2
                            a_trial[ti] += amt1 + amt2
                            a_trial = np.maximum(a_trial, 0.0)
                            ts = a_trial.sum()
                            if ts < 1e-12:
                                continue
                            a_trial /= ts
                            o = obj_value(a_trial)
                            if o < best_obj - 1e-12:
                                best_obj = o
                                a = a_trial.copy()
                                found = True
                                break
        
        if not found:
            stuck += 1
    
    return a, best_obj


def spectral_init(n, rng):
    """
    Generate a sequence designed to have a relatively flat power spectrum.
    Uses iterative spectral shaping.
    """
    a = np.ones(n) / n
    for _ in range(50):
        A = np.fft.rfft(a)
        P = np.abs(A) ** 2
        # We want P to be flat; compute the "excess"
        P_mean = P.mean()
        # Where P is above average, reduce corresponding time-domain energy
        # This is heuristic
        P_clipped = np.minimum(P, P_mean * 1.5)
        # Inverse: modify a to reduce high-P frequencies
        # a_new ∝ a * exp(-beta * |P - P_mean|) projected
        # Simpler: just do a few ICM iterations
        A_mod = A * (1.0 - 0.1 * (P - P_mean) / (P.max() + 1e-15))
        A_mod = np.maximum(A_mod.real, 0) + 1j * A_mod.imag
        a_new = np.fft.irfft(A_mod, n)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-12:
            break
        a_new /= s
        if np.allclose(a_new, a):
            break
        a = a_new
    return a


def get_parent(n):
    """Get parent sequence resized to length n."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim != 1 or p.size == 0 or not np.isfinite(p).all():
            return None
        if p.size >= n:
            pc = p[:n].copy()
        else:
            pc = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, p.size), p)
        pc = np.maximum(pc, 0.0)
        s = pc.sum()
        if s < 1e-10:
            return None
        return pc / s
    except Exception:
        return None


def make_inits(n, rng):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64)
    inits = {}
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Parent
    parent = get_parent(n)
    if parent is not None:
        inits['parent'] = parent
    
    # Gaussian (various widths)
    for sf in [0.15, 0.25, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g
    
    # Off-center Gaussian
    for center in [0.3, 0.7]:
        for sf in [0.2, 0.3]:
            g = np.exp(-0.5 * ((x - center * n) / (sf * n)) ** 2)
            g /= g.sum()
            inits[f'gauss_c{center}_s{sf}'] = g
    
    # Bimodal (mass at edges)
    for width in [0.1, 0.15, 0.2]:
        bim = np.exp(-0.5 * (x / (width * n)) ** 2) + np.exp(-0.5 * ((x - n) / (width * n)) ** 2)
        bim = np.maximum(bim, 0)
        bim /= bim.sum()
        inits[f'bim_{width}'] = bim
    
    # Cosine window (Hann)
    if n > 1:
        hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
        hann = np.maximum(hann, 0)
        inits['hann'] = hann / hann.sum()
    
    # Flat-top window
    flat = np.ones(n)
        flat[:n//10] = 0.5
        flat[-n//10:] = 0.5
        inits['flat_top'] = flat / flat.sum()
    
    # Random
    r = rng.dirichlet(np.ones(n) * 0.5)
    inits['random'] = r
    
    # Spectral init
    try:
        inits['spectral'] = spectral_init(n, rng)
    except:
        pass
    
    return inits


def combined_optimize(a0, time_limit):
    """
    Combined optimizer: alternates between gradient steps, LP directions,
    and local search.
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones_like(a) / len(a)
    else:
        a /= s
    
    best_f = a.copy()
    best_obj = obj_value(a)
    t0 = time.monotonic()
    
    # Strategy: alternate between methods
    # Each "round" consists of: gradient phase + LP phase + local search phase
    round_num = 0
    
    while time.monotonic() - t0 < time_limit:
        round_num += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        # Phase A: Gradient steps (40% of round)
        grad_time = remaining * 0.4
        a_grad, obj_grad = gradient_optimize(a, time_limit=grad_time)
        if obj_grad < best_obj - 1e-12:
            best_obj = obj_grad
            best_f = a_grad.copy()
            a = a_grad.copy()
        
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        # Phase B: LP direction (30% of round)
        lp_time = remaining * 0.3
        if len(a) <= 256:  # LP is expensive for large n
            a_lp, obj_lp = lp_optimize(a, time_limit=lp_time)
            if obj_lp < best_obj - 1e-12:
                best_obj = obj_lp
                best_f = a_lp.copy()
                a = a_lp.copy()
        
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        # Phase C: Local search (30% of round)
        ls_time = remaining * 0.3
        a_ls, obj_ls = local_search_multi(a, time_limit=ls_time)
        if obj_ls < best_obj - 1e-12:
            best_obj = obj_ls
            best_f = a_ls.copy()
            a = a_ls.copy()
        
        if round_num % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Round {round_num} ({elapsed:.1f}s): obj={best_obj:.8f}")
    
    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Uses combined optimization (gradient + LP + local search)
    with multiple initializations and sequence lengths to minimize
    2n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    total_budget = max(10.0, float(budget_s))
    deadline = time.monotonic() + total_budget - 30.0  # leave 30s buffer
    
    best_sequence = None
    best_score = float('inf')
    
    # Try multiple lengths
    lengths = [128, 256, 512, 1024]
    
    # Phase 1: Scan initializations with gradient optimization
    print("Phase 1: Gradient-based scan...")
    p1_end = time.monotonic() + total_budget * 0.35
    
    all_configs = []
    for n in lengths:
        inits = make_inits(n, rng)
        for name, a0 in inits.items():
            all_configs.append((n, name, a0))
    
    # Prioritize parent and shorter sequences
    def config_priority(item):
        n, name, _ = item
        if name == 'parent':
            return (0, n)
        if name in ('gauss_0.35', 'bim_0.15', 'hann'):
            return (1, n)
        return (2, n)
    
    all_configs.sort(key=config_priority)
    
    n_configs = len(all_configs)
    for i, (n, name, a0) in enumerate(all_configs):
        remaining = p1_end - time.monotonic()
        if remaining < 10:
            break
        t_alloc = min(30.0, max(5.0, remaining / max(1, n_configs - i)))
        t_alloc = min(t_alloc, deadline - time.monotonic() - 60.0)
        if t_alloc < 3.0:
            break
        
        a_opt, obj_opt = gradient_optimize(a0, time_limit=t_alloc)
        score = float(evaluate_sequence(a_opt.tolist()))
        
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P1: n={n}, {name}, score={score:.6f}")
    
    print(f"  Phase 1 best: {best_score:.6f}")
    
    # Phase 2: Combined optimization on best candidates
    print("Phase 2: Combined optimization...")
    p2_end = time.monotonic() + total_budget * 0.45
    
    if best_sequence is not None:
        n_best = len(best_sequence)
        # Also try nearby lengths
        candidates = [(n_best, 'best', best_sequence.copy())]
        
        # Add parent at same length
        parent = get_parent(n_best)
        if parent is not None:
            candidates.append((n_best, 'parent', parent))
        
        # Add some diverse inits at best length
        inits_best = make_inits(n_best, rng)
        for name in ['gauss_0.35', 'bim_0.15', 'hann', 'spectral']:
            if name in inits_best:
                candidates.append((n_best, name, inits_best[name]))
        
        n_cand = len(candidates)
        for i, (n, name, a0) in enumerate(candidates):
            remaining = p2_end - time.monotonic()
            if remaining < 20:
                break
            t_alloc = min(120.0, max(20.0, remaining / max(1, n_cand - i)))
            t_alloc = min(t_alloc, deadline - time.monotonic() - 60.0)
            if t_alloc < 10.0:
                break
            
            a_opt, obj_opt = combined_optimize(a0, time_limit=t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))
            
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P2: n={n}, {name}, score={score:.6f}")
    
    print(f"  Phase 2 best: {best_score:.6f}")
    
    # Phase 3: Intensive polish
    print("Phase 3: Intensive polish...")
    if best_sequence is not None:
        polish_deadline = deadline - 10.0
        no_improve_round = 0
        
        while time.monotonic() < polish_deadline and no_improve_round < 10:
            remaining = polish_deadline - time.monotonic()
            if remaining < 15:
                break
            
            # Alternate between different optimization strategies
            t_alloc = min(remaining * 0.4, 120.0)
            
            # Try gradient
            a_grad, obj_grad = gradient_optimize(best_sequence, time_limit=t_alloc * 0.5)
            score = float(evaluate_sequence(a_grad.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_grad.copy()
                no_improve_round = 0
                print(f"  P3 grad: {score:.6f}")
                continue
            no_improve_round += 1
            
            remaining = polish_deadline - time.monotonic()
            if remaining < 15:
                break
            
            # Try local search (more aggressive)
            t_ls = min(remaining * 0.4, 90.0)
            a_ls, obj_ls = local_search_multi(best_sequence, time_limit=t_ls)
            score = float(evaluate_sequence(a_ls.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_ls.copy()
                no_improve_round = 0
                print(f"  P3 ls: {score:.6f}")
                continue
            no_improve_round += 1
            
            remaining = polish_deadline - time.monotonic()
            if remaining < 15 or len(best_sequence) > 256:
                break
            
            # Try LP
            t_lp = min(remaining * 0.4, 60.0)
            a_lp, obj_lp = lp_optimize(best_sequence, time_limit=t_lp)
            score = float(evaluate_sequence(a_lp.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_lp.copy()
                no_improve_round = 0
                print(f"
