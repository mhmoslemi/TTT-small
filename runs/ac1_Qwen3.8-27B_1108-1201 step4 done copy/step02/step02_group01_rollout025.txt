Multi-strategy optimization for minimizing the normalized peak autocorrelation
of non-negative step functions. Combines Nesterov-accelerated multi-peak gradient
descent, LP-based refinement, simulated annealing, and coordinate descent.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sa = np.sum(a)
    if sa < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sa ** 2))

def normalize_seq(a, S=None):
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-12:
        return a * S / len(a)
    return a * S / s

def build_lp_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A

def lp_solve(a, M):
    n = len(a)
    A = build_lp_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.status == 0:
        return res.x
    return np.zeros(n)

def nesterov_gradient_optimize(a, time_limit, K=5, verbose=True):
    """Nesterov-accelerated multi-peak gradient descent."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv_max = np.max(fft_convolve(a, a))
    
    # Nesterov momentum
    v = np.zeros(n)
    beta = 0.9
    lr = 1.0 / n
    
    step = 0
    while time.monotonic() - t0 < time_limit:
        # Nesterov look-ahead
        a_star = a - beta * v
        
        # Compute gradient at a_star
        conv_star = fft_convolve(a_star, a_star)
        sorted_idx = np.argsort(conv_star)[::-1]
        top_K = sorted_idx[:K]
        max_val = conv_star[sorted_idx[0]]
        
        grad = np.zeros(n)
        for k in top_K:
            w = conv_star[k] / max_val
            for j in range(n):
                kj = k - j
                if 0 <= kj < n:
                    grad[j] += w * a_star[kj]
        
        # Project out mean
        grad -= np.mean(grad)
        
        # Update
        v = beta * v + grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_seq(a_new, S)
        
        new_conv = fft_convolve(a_new, a_new)
        new_max = np.max(new_conv)
        
        if new_max < best_conv_max:
            best_conv_max = new_max
            best_a = a_new.copy()
            a = a_new
            # Increase lr slightly
            lr = min(lr * 1.01, 5.0 / n)
        else:
            a = a_new  # Still move forward (Nesterov)
            lr *= 0.98
        
        step += 1
        if verbose and step % 200 == 0:
            score = 2.0 * n * best_conv_max / (S * S)
            print(f"    Nesterov step {step}: score={score:.6f}, lr={lr:.4f}")
    
    return best_a

def simulated_annealing(a, time_limit, verbose=True):
    """Simulated annealing with peak-focused perturbations."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = np.max(fft_convolve(a, a))
    
    rng = np.random.default_rng(42)
    T_initial = 0.1
    T_final = 0.001
    total_steps = 10000
    
    step = 0
    while time.monotonic() - t0 < time_limit:
        T = T_initial * (T_final / T_initial) ** (step / total_steps)
        
        candidate = a.copy()
        conv = fft_convolve(candidate, candidate)
        peak_idx = int(np.argmax(conv))
        
        # Perturb elements that contribute to the peak
        for _ in range(rng.integers(3, 10)):
            j = rng.integers(0, n)
            kj = peak_idx - j
            if 0 <= kj < n:
                candidate[j] *= np.exp(rng.normal(0, T))
                if 0 <= kj < n and kj != j:
                    candidate[kj] *= np.exp(rng.normal(0, T))
        
        candidate = np.clip(candidate, 0.01, None)
        candidate = normalize_seq(candidate, S)
        
        new_conv = fft_convolve(candidate, candidate)
        new_max = np.max(new_conv)
        
        delta = new_max - np.max(fft_convolve(a, a))
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = candidate
            if new_max < best_score:
                best_score = new_max
                best_a = candidate.copy()
        
        step += 1
        if verbose and step % 500 == 0:
            score = 2.0 * n * best_score / (S * S)
            print(f"    SA step {step}: T={T:.4f}, score={score:.6f}")
    
    return best_a

The function `f(a) = max_k (a*a)[k]` is the maximum of `2n-1` quadratic forms. Each quadratic form `a^T B_k a` is indefinite (since `B_k` has both positive and negative eigenvalues). So `f(a)` is non-convex.

The key challenge is that reducing one peak might increase another. The multi-peak gradient approach addresses this by considering multiple peaks simultaneously, but it's still a heuristic.

A potentially better approach: **use the KKT conditions more directly**.

At the optimum, the subdifferential of `f(a)` at `a` should contain a vector of the form `mu * 1 - nu` where `nu >= 0` (for `a >= 0`) and `nu_j > 0` only if `a_j = 0`.

The subdifferential of `f(a) = max_k a^T B_k a` is the convex hull of `{2 * B_k * a : k in active set}`, where the active set is `{k : a^T B_k a = f(a)}`.

So at the optimum, there exist weights `lambda_k >= 0` (sum to 1 over the active set) such that:
`2 * sum_k lambda_k * B_k * a = mu * 1 - nu`

where `nu >= 0` and `nu_j * a_j = 0`.

This means: `sum_k lambda_k * B_k * a` should be approximately constant (for coordinates where `a_j > 0`).

We can use this as a stopping criterion and also as a guide for the optimization.

**Algorithm: KKT-based iterative method**

1. Start with `a`
2. Compute `conv = a*a`, find active set `K` (peaks)
3. Solve for weights `lambda_k` such that `sum_k lambda_k * B_k * a` is as constant as possible
4. Use this to determine the direction of improvement
5. Update `a`

def fft_convolve(a, b):

    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):

    n = len(a)

        S = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-12:
        return a.copy() * S / len(a)
    return a * S / s

    n = len(a)
    conv = self_convolve(a)
    sorted_idx = np.argsort(conv)[::-1]
    top_K = sorted_idx[:K]
    max_val = conv[sorted_idx[0]]
