import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)

    # The subgradient at position j is sum_k in peak_lags [a[k-j] + a[j-k]]
    # = convolve(a, kernel)[j] + convolve(a, kernel_flipped)[j] ... 
    # Actually: sum_k a[k-j] for k in peak_lags = (a * delta_peak)[j] where delta is sparse
    # Let's just use direct computation for the peak lags (usually few)
    grad = np.zeros(n)
    for k in peak_lags:
        # a[k-j] valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            grad[j] += a[k - j]
        # a[j-k] valid when 0 <= j-k < n, i.e., k <= j <= k+n-1
        j_min2 = max(k, 0)
        j_max2 = min(k + n - 1, n - 1)
        for j in range(j_min2, j_max2 + 1):
            grad[j] += a[j - k]
    return grad


def adaptive_gradient_step(a, lr_init, max_iter=5000, deadline=None, verbose=False):
    """
    Adaptive subgradient descent with backtracking line search.
    Targets the top K peak lags simultaneously.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a.copy(), S)
    best_a = a.copy()
    best_conv = np.max(fft_convolve(a, a))
    
    t0 = time.monotonic()
    step = 0
    stall = 0
    lr = lr_init
    K = min(5, n // 10)  # Number of peak lags to target
    
    while step < max_iter:
        if deadline is not None and time.monotonic() > deadline:
            break
        if step % 1000 == 0 and time.monotonic() - t0 > 300:
            break
        
        conv = fft_convolve(a, a)
        current_max = np.max(conv)
        
        # Find top K peak lags (within 10% of max)
        threshold = current_max * 0.95
        peak_indices = np.where(conv >= threshold)[0]
        if len(peak_indices) > K:
            # Take the top K
            sorted_peaks = peak_indices[np.argsort(conv[peak_indices])[::-1]][:K]
        else:
            sorted_peaks = peak_indices
        
        # Compute subgradient
        grad = compute_subgradient_fast(a, sorted_peaks)
        
        # Backtracking line search
        alpha = 1.0
        improved = False
        for _ in range(15):
            step_size = lr * alpha
            a_trial = a - step_size * grad / np.max(np.abs(grad) + 1e-15)
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = normalize_seq(a_trial, S)
            new_conv_max = np.max(fft_convolve(a_trial, a_trial))
            if new_conv_max < current_max * (1.0 - 0.001 * alpha):
                a = a_trial
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            # Try a smaller fixed step
            a_trial = a - lr * 0.01 * grad / (np.max(np.abs(grad)) + 1e-15)
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = normalize_seq(a_trial, S)
            new_conv_max = np.max(fft_convolve(a_trial, a_trial))
            if new_conv_max < current_max:
                a = a_trial
                improved = True
        
        if improved:
            if np.max(fft_convolve(a, a)) < best_conv:
                best_conv = np.max(fft_convolve(a, a))
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
            lr = min(lr * 1.05, 2.0 / n)
        else:
            stall += 1
            lr *= 0.9
            if stall > 100:
                # Restart with perturbation
                rng = np.random.default_rng(step)
                a = a * (1 + 0.05 * rng.standard_normal(n))
                a = np.maximum(a, 0.0)
                a = normalize_seq(a, S)
                stall = 0
                lr = 0.5 / n
        
        step += 1
        if verbose and step % 500 == 0:
            sc = 2.0 * n * best_conv / (S * S)
            print(f"      Grad step {step}: score={sc:.6f}, lr={lr:.5f}, conv_max={best_conv:.4f}")
    
    return best_a, best_conv


def build_convolution_matrix(a):
    """Build sparse convolution matrix for LP."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return rows, cols, vals


def lp_refine(a, time_limit=60.0, deadline=None, verbose=False):
    """
    LP-based iterative refinement with aggressive blending.
    """
    from scipy.sparse import csr_matrix
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a.copy(), S)
    best_a = a.copy()
    best_conv = np.max(fft_convolve(a, a))
    
    # Pre-build the matrix
    rows, cols, vals = build_convolution_matrix(a)
    
    t0 = time.monotonic()
    iteration = 0
    
    while iteration < 200:
        if time.monotonic() - t0 > time_limit:
            break
        if deadline is not None and time.monotonic() > deadline:
            break
        
        conv_a = fft_convolve(a, a)
        M = np.max(conv_a)
        
        # Build sparse matrix
        A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        
        # Solve LP: maximize sum(g) s.t. A@g <= M, g >= 0
        c = -np.ones(n)
        res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True})
        
        if res.status != 0:
            break
        
        g = res.x
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        
        g = normalize_seq(g, S)
        
        # Find best blend
        best_t = 0.0
        best_blend_conv = M
        for t in [0.4, 0.3, 0.2, 0.15, 0.1, 0.08, 0.05, 0.03, 0.02, 0.01]:
            blend = (1 - t) * a + t * g
            blend = np.maximum(blend, 0.0)
            blend = normalize_seq(blend, S)
            bc = np.max(fft_convolve(blend, blend))
            if bc <= M + 1e-10:
                best_t = t
                best_blend_conv = bc
                break
        
        if best_t == 0.0:
            # Try even smaller
            for t in [0.005, 0.002]:
                blend = (1 - t) * a + t * g
                blend = np.maximum(blend, 0.0)
                blend = normalize_seq(blend, S)
                bc = np.max(fft_convolve(blend, blend))
                if bc <= M + 1e-10:
                    best_t = t
                    best_blend_conv = bc
                    break
        
        if best_t == 0.0:
            break
        
        a = (1 - best_t) * a + best_t * g
        a = np.maximum(a, 0.0)
        a = normalize_seq(a, S)
        
        new_conv = np.max(fft_convolve(a, a))
        if new_conv < best_conv - 1e-10:
            best_conv = new_conv
            best_a = a.copy()
        
        if verbose and iteration % 20 == 0:
            sc = 2.0 * n * best_conv / (S * S)
            print(f"      LP iter {iteration}: score={sc:.6f}, M={M:.4f}, t={best_t:.3f}")
        
        if best_t < 0.005:
            break
        iteration += 1
    
    return best_a


def coordinate_descent(a, time_limit=60.0, deadline=None):
    """
    Coordinate descent: try adjusting individual elements to reduce the peak.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a.copy(), S)
    best_a = a.copy()
    best_conv = np.max(fft_convolve(a, a))
    rng = np.random.default_rng(999)
    
    t0 = time.monotonic()
    it = 0
    
    while it < 50000:
        if time.monotonic() - t0 > time_limit:
            break
        if deadline is not None and time.monotonic() > deadline:
            break
        
        conv = fft_convolve(a, a)
        peak_idx = int(np.argmax(conv))
        current_peak = conv[peak_idx]
        
        # Try modifying elements that contribute to the peak
        # (a*a)[peak_idx] = sum_j a[j]*a[peak_idx-j]
        contrib = np.zeros(n)
        j_min = max(0, peak_idx - n + 1)
        j_max = min(peak_idx, n - 1)
        for j in range(j_min, j_max + 1):
            contrib[j] = a[j] * a[peak_idx - j]
        
        # Pick the top contributor and try reducing it
        if np.max(contrib) > 0:
            top_j = int(np.argmax(contrib))
            # Try reducing a[top_j]
            for frac in [0.5, 0.3, 0.1, 0.05]:
                candidate = a.copy()
                old_val = candidate[top_j]
                candidate[top_j] = old_val * (1 - frac)
                # Redistribute the lost mass
                lost = old_val - candidate[top_j]
                # Add to a random other position
                other = int(rng.integers(0, n))
                if other == top_j:
                    other = (other + 1) % n
                candidate[other] += lost
                candidate = np.maximum(candidate, 0.0)
                candidate = normalize_seq(candidate, S)
                nc = np.max(fft_convolve(candidate, candidate))
                if nc < best_conv:
                    best_conv = nc
                    best_a = candidate.copy()
                    a = candidate
                    break
                elif nc < current_peak:
                    a = candidate
                    break
        
        # Also try small random perturbations
        if it % 10 == 0:
            idx = int(rng.integers(0, n))
            candidate = a.copy()
            candidate[idx] *= np.exp(rng.normal(0, 0.02))
            candidate = np.maximum(candidate, 0.0)
            candidate = normalize_seq(candidate, S)
            nc = np.max(fft_convolve(candidate, candidate))
            if nc < best_conv:
                best_conv = nc
                best_a = candidate.copy()
                a = candidate
            elif nc < current_peak and rng.random() < 0.5:
                a = candidate
        
        it += 1
    
    return best_a


def two_level_init(n, ratio=2.0, split=0.5):
    """Create a two-level step function initialization."""
    k = int(n * split)
    a = np.zeros(n)
    a[:k] = 1.0 / ratio
    a[k:] = 1.0
    return a


def chirp_init(n, alpha=0.5):
    """Create a chirp (accelerating frequency) initialization."""
    t = np.arange(n)
    a = 1.0 + 0.3 * np.cos(2 * np.pi * alpha * t**2 / n)
    return np.maximum(a, 0.1)


def piecewise_init(n, n_pieces=4):
    """Create a piecewise constant initialization."""
    sizes = np.random.multinomial(n, np.ones(n_pieces) / n_pieces)
    a = np.zeros(n)
    pos = 0
    for i, s in enumerate(sizes):
        a[pos:pos+s] = 0.5 + 0.5 * (i / max(1, n_pieces - 1))
        pos += s
    return np.maximum(a, 0.1)


def interpolate_seq(a, n_new):
    """Interpolate a sequence to a new length."""
    x = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x, a)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining adaptive gradient descent, LP refinement,
    multi-length exploration, and coordinate descent polish.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 30.0
    rng = np.random.default_rng(seed)
    
    best_a = None
    best_score = np.inf
    
    def update_best(a):
        nonlocal best_a, best_score
        a_clip = np.clip(a, 0.0, 1000.0)
        s = fast_score(a_clip)
        if s < best_score:
            best_score = s
            best_a = a_clip.copy()
        return best_score
    
    # Get provided sequence
    provided = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        print(f"Provided: n={len(provided)}, score={fast_score(provided):.6f}")
    
    # Phase 0: Quick gradient from provided sequence
    print("\n=== Phase 0: Quick gradient (provided seq) ===")
    if provided is not None and time.monotonic() < deadline - 300:
        a_grad, _ = adaptive_gradient_step(provided, lr_init=0.3/len(provided),
                                           max_iter=2000, deadline=deadline - 300)
        s = update_best(a_grad)
        print(f"  After quick gradient: {s:.6f}")
    
    # Phase 1: LP refinement on best so far
    print("\n=== Phase 1: LP refinement ===")
    start_a = best_a if best_a is not None else (provided if provided is not None else np.ones(256))
    if time.monotonic() < deadline - 200:
        a_lp = lp_refine(start_a, time_limit=120.0, deadline=deadline - 200, verbose=True)
        s = update_best(a_lp)
        print(f"  After LP: {s:.6f}")
    
    # Phase 2: Gradient polish on LP result
    print("\n=== Phase 2: Gradient polish ===")
    if best_a is not None and time.monotonic() < deadline - 100:
        a_grad2, _ = adaptive_gradient_step(best_a, lr_init=0.2/len(best_a),
                                            max_iter=3000, deadline=deadline - 100)
        s = update_best(a_grad2)
        print(f"  After gradient polish: {s:.6f}")
    
    # Phase 3: Multi-length exploration
    print("\n=== Phase 3: Multi-length exploration ===")
    lengths_to_try = [384, 448, 512, 576, 640, 768, 896]
    for n_try in lengths_to_try:
        if time.monotonic() > deadline - 60:
            break
        # Interpolate current best to new length
        a0 = interpolate_seq(best_a, n_try)
        a0 = np.maximum(a0, 0.01)
        
        # Quick gradient
        if time.monotonic() > deadline - 40:
            break
        a_g, _ = adaptive_gradient_step(a0, lr_init=0.3/n_try,
                                        max_iter=1000, deadline=deadline - 40)
        s = fast_score(a_g)
        
        # LP if time allows
        if time.monotonic() < deadline - 30 and s < best_score + 0.01:
            a_l = lp_refine(a_g, time_limit=30.0, deadline=deadline - 30)
            s = update_best(a_l)
            if s < best_score:
                print(f"  n={n_try}: {s:.6f} *** NEW BEST ***")
    
    # Phase 4: Coordinate descent polish
    print("\n=== Phase 4: Coordinate descent polish ===")
    if best_a is not None and time.monotonic() < deadline - 10:
        a_cd = coordinate_descent(best_a, time_limit=45.0, deadline=deadline - 10)
        s = update_best(a_cd)
        print(f"  After coord descent: {s:.6f}")
    
    # Phase 5: Final LP if time remains
    print("\n=== Phase 5: Final LP ===")
    if best_a is not None and time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            a_final = lp_refine(best_a, time_limit=remaining, deadline=deadline)
            s = update_best(a_final)
            print(f"  After final LP: {s:.6f}")
    
    # Phase 6: Last moment coordinate descent
    if best_a is not None and time.monotonic() < deadline - 2:
        remaining = deadline - time.monotonic() - 2
        if remaining > 5:
            a_last = coordinate_descent(best_a, time_limit=remaining, deadline=deadline)
            s = update_best(a_last)
    
    print(f"\n=== FINAL: score={best_score:.6f}, n={len(best_a) if best_a is not None else 0} ===")
    
    # Verification
    if best_a is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_a]
        eval_score = evaluate_sequence(final_list)
        print(f"evaluate_sequence confirms: {eval_score:.6f}")
        return final_list
    
    # Fallback
    if provided is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided]
    return [1.0] * 256
