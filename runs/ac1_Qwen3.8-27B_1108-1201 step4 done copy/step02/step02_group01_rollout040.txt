import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))

def subgradient_at_peak(a, peak_idx):
    """Compute subgradient of conv(a,a)[peak_idx] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    for j in range(n):
        k_minus_j = peak_idx - j
        if 0 <= k_minus_j < n:
            grad[j] = 2.0 * a[k_minus_j]
    return grad

def subgradient_descent(a, time_limit, lr_init=0.1, verbose=True):
    """Subgradient descent with momentum."""
    n = len(a)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    
    lr = lr_init
    momentum = 0.9
    velocity = np.zeros(n)
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute convolution and find peak
        conv = fft_convolve(a, a)
        peak_val = np.max(conv)
        # Find all near-peak locations
        threshold = peak_val * 0.95
        peak_indices = np.where(conv >= threshold)[0]
        
        # Compute combined subgradient (average over near-peak locations)
        grad = np.zeros(n)
        for pi in peak_indices:
            grad += subgradient_at_peak(a, pi)
        grad /= len(peak_indices)
        
        # Momentum update
        velocity = momentum * velocity + lr * grad
        a_new = a - velocity
        a_new = np.maximum(a_new, 0.0)
        
        # Check improvement
        new_score = compute_score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            stall = 0
        elif new_score < compute_score(a):
            a = a_new
            stall += 1
        else:
            stall += 1
            velocity *= 0.5  # Reduce momentum

Hybrid optimization combining subgradient descent with LP-based refinement
for minimizing peak self-convolution of non-negative sequences.
Uses FFT-based convolution, momentum-accelerated subgradient, sparse LP solving,

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))


def subgradient_at_peak(a, peak_idx):
    """Compute subgradient of conv(a,a)[peak_idx] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    j = np.arange(n)
    k_minus_j = peak_idx - j
    valid = (k_minus_j >= 0) & (k_minus_j < n)
    grad[valid] = 2.0 * a[k_minus_j[valid]]
    return grad

def build_lp_matrix_sparse(a):
    """Build sparse convolution constraint matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []

def lp_refine(a, time_limit, verbose=True, max_iters=500):
    """LP-based iterative refinement with sparse matrix."""
    n = len(a)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    sum_target = np.sum(a)

    for iteration in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        conv_a = fft_convolve(a, a)
        M = np.max(conv_a)

        # Build sparse constraint matrix
        A = build_lp_matrix_sparse(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)

        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')

        if res.status != 0:
            break

        g = res.x
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break

        # Normalize g to match sum_target
        g = g * sum_target / g_sum

        # Find best blend parameter
        best_t = 0.0
        best_score_t = np.max(fft_convolve(a, a))
        for t_try in [0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.08, 0.05, 0.03]:
            b = (1 - t_try) * a + t_try * g
            b = np.maximum(b, 0.0)
            b = b * sum_target / np.sum(b)
            new_peak = np.max(fft_convolve(b, b))
            new_score = 2.0 * n * new_peak / (sum_target ** 2)
            if new_score < best_score_t:
                best_score_t = new_score
                best_t = t_try

        if best_t == 0.0:
            break

        a = (1 - best_t) * a + best_t * g
        a = np.maximum(a, 0.0)
        a = a * sum_target / np.sum(a)

        current_score = compute_score(a)
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()

        if verbose and iteration % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iteration} ({elapsed:.0f}s): score={best_score:.6f}, t={best_t:.3f}")

    return best_a


def local_search(a, time_limit, verbose=True):
    """Coordinate and block-level local search."""
    n = len(a)

        # Strategy 1: Coordinate descent on peak-contributing elements
        conv = fft_convolve(best_a, best_a)
        peak_idx = int(np.argmax(conv))

        # Find top contributors to the peak
        contribs = np.zeros(n)
        j = np.arange(n)
        k_minus_j = peak_idx - j
        valid = (k_minus_j >= 0) & (k_minus_j < n)
        contribs[valid] = best_a[valid] * best_a[k_minus_j[valid]]

        top_k = max(5, n // 10)
        top_idx = np.argsort(contribs)[-top_k:]

        # Try reducing top contributors and increasing others

            if len(other) > 0:
                n_inc = max(3, n // 15)
                inc_idx = rng.choice(other, size=min(n_inc, len(other)), replace=False)
                candidate[inc_idx] *= np.exp(rng.uniform(0.02, 0.15, size=len(inc_idx)))

            candidate = np.maximum(candidate, 0.0)
            candidate = candidate * sum_target / np.sum(candidate)
            new_score = compute_score(candidate)

            if new_score < best_score:
                best_score = new_score
                best_a = candidate.copy()

        # Strategy 2: Block perturbation
        if rng.random() < 0.4:
            width = max(2, n // 25)
            start = int(rng.integers(0, n - width))
            candidate = best_a.copy()
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.05, size=width))
            candidate = np.maximum(candidate, 0.0)
            candidate = candidate * sum_target / np.sum(candidate)
            new_score = compute_score(candidate)
            if new_score < best_score:
                best_score = new_score
                best_a = candidate.copy()

                    sj, ej = j_ * block_size, (j_ + 1) * block_size
                    candidate[si:ei], candidate[sj:ej] = candidate[sj:ej].copy(), candidate[si:ei].copy()
            new_score = compute_score(candidate)

def make_initialization(n, init_type, rng):

def interpolate_sequence(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)

    print("\n=== Phase 0: Initialization ===")
    candidates = []
    if provided_seq is not None:
        candidates.append(provided_seq)

    # Try interpolating to other lengths
    if provided_seq is not None:
        for n_try in [512, 768, 896]:
            if time.monotonic() > deadline - 1500:
                break
            a0 = interpolate_sequence(provided_seq, n_try)
            a0 = np.clip(a0, 0.01, None)
            sc = compute_score(a0)
            print(f"  Interpolated n={n_try}: score={sc:.6f}")
            candidates.append(a0)

    # Try a few structured inits
    for n in [512, 640, 768]:
        if time.monotonic() > deadline - 1400:
            break
        for init_type in ['chirp', 'multi_tone']:
            a0 = make_initialization(n, init_type, rng)
            sc = compute_score(a0)
            candidates.append(a0)

    # Quick SGD on each candidate to find the best starting point
    print("  Running quick SGD on candidates...")
    quick_results = []
    for i, a0 in enumerate(candidates):
        if time.monotonic() > deadline - 1300:
            break
        t_left = min(15.0, (deadline - time.monotonic()) * 0.05)
        if t_left < 3:
            break
        result = sgd_optimize(a0, t_left, verbose=False)
        sc = compute_score(result)
        quick_results.append((sc, result))
        print(f"    Candidate {i} (n={len(a0)}): score={sc:.6f}")

    # Pick the best
    quick_results.sort(key=lambda x: x[0])
    if quick_results[0][0] < best_score:
        best_score = quick_results[0][0]
        best_seq = quick_results[0][1].copy()
        print(f"  Best initial: n={len(best_seq)}, score={best_score:.6f}")
    else:
        print(f"  Keeping provided: score={best_score:.6f}")

    # Phase 1: Main subgradient descent
    print(f"\n=== Phase 1: Subgradient descent (n={len(best_seq)}) ===")
    t_limit = min(400.0, (deadline - time.monotonic()) * 0.25)
    result = sgd_optimize(best_seq, t_limit, lr_init=2.0 / len(best_seq), verbose=True)
    update_best(result)
    print(f"  After SGD: score={best_score:.6f}")

    # Phase 2: LP refinement
    print(f"\n=== Phase 2: LP refinement ===")
    t_limit = min(200.0, (deadline - time.monotonic()) * 0.15)
    if t_limit > 20:
        result = lp_refine(best_seq, t_limit, verbose=True)
        update_best(result)
        print(f"  After LP: score={best_score:.6f}")

    # Phase 3: More subgradient
    print(f"\n=== Phase 3: Subgradient refinement ===")
    t_limit = min(300.0, (deadline - time.monotonic()) * 0.2)
    if t_limit > 20:
        result = sgd_optimize(best_seq, t_limit, lr_init=1.0 / len(best_seq),
                              momentum_init=0.95, verbose=True)
        update_best(result)
        print(f"  After SGD2: score={best_score:.6f}")

    # Phase 4: LP refinement again
    print(f"\n=== Phase 4: LP refinement 2 ===")
    t_limit = min(150.0, (deadline - time.monotonic()) * 0.15)
    if t_limit > 20:
        result = lp_refine(best_seq, t_limit, verbose=True)
        update_best(result)
        print(f"  After LP2: score={best_score:.6f}")

    # Phase 5: Local search
    print(f"\n=== Phase 5: Local search ===")
    t_limit = min(200.0, (deadline - time.monotonic()) * 0.2)
    if t_limit > 10:
        result = local_search(best_seq, t_limit, verbose=True)
        update_best(result)
        print(f"  After local search: score={best_score:.6f}")

    # Phase 6: Random restarts
    print(f"\n=== Phase 6: Random restarts ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n_restarts = min(5, int(remaining / 40))
        for r in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break
            # Perturb the best sequence
            noise_scale = 0.02 + 0.03 * r
            candidate = best_seq * np.exp(rng.normal(0, noise_scale, size=len(best_seq)))
            candidate = np.maximum(candidate, 0.01)
            candidate = candidate * np.sum(best_seq) / np.sum(candidate)

            t_per = max(10.0, (deadline - time.monotonic() - 30) / (n_restarts - r))
            result = sgd_optimize(candidate, t_per, lr_init=2.0 / len(candidate), verbose=False)
            if update_best(result):
                print(f"  Restart {r}: NEW BEST score={best_score:.6f}")
            else:
                print(f"  Restart {r}: score={compute_score(result):.6f} (no improvement)")

    # Phase 7: Final polish
    print(f"\n=== Phase 7: Final polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Short LP
        t_lp = min(60.0, remaining * 0.4)
        if t_lp > 10:
            result = lp_refine(best_seq, t_lp, verbose=False)
            update_best(result)

        # Short local search
        t_ls = min(60.0, remaining * 0.4)
        if t_ls > 5:
            result = local_search(best_seq, t_ls, verbose=False)
            update_best(result)

        # Very short SGD
        t_sgd = remaining * 0.2
        if t_sgd > 5:
            result = sgd_optimize(best_seq, t_sgd, lr_init=0.5 / len(best_seq), verbose=False)
            update_best(result)

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = compute_score(np.array(final_list))
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        # Verify with evaluate_sequence
        eval_score = evaluate_sequence(final_list)
        print(f"=== evaluate_sequence confirms: {eval_score:.6f} ===")
        return final_list

    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    return [1.0] * 640
