Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Soft-max gradient descent with multiplicative updates (main driver)
2. LP-based direction finding (periodic refinement)
3. Targeted local search via mass transfer (final polish)
Starts from the parent sequence and uses adaptive temperature/step-size schedules.

import numpy as np
import scipy.optimize
import time


def fft_convolve(a, b=None):
    """Linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def compute_objective(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fft_convolve(a)
    return 2.0 * float(n) * float(conv.max())


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    if v.sum() <= 0:
        return np.ones(n) / n
    # Binary search for the threshold
    mu_max = float(v.max())
    mu_min = float(v.min())
    lo, hi = mu_min, mu_max
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    x = np.maximum(v - tau, 0.0)
    sx = float(x.sum())
    if sx < 1e-30:
        return np.ones(n) / n
    x /= sx
    return x


def gradient_softmax(a, t):
    """Compute gradient of soft-max objective 2n*t*log(sum(exp(conv/t)))."""
    n = len(a)
    conv = fft_convolve(a)
    conv_scaled = conv / t
    conv_scaled -= conv_scaled.max()
    w = np.exp(conv_scaled)
    w /= w.sum()
    # grad[i] = 4n * sum_j a[j] * w[i+j]
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = 4.0 * n * float(np.dot(a, w[i:i+n]))
    return grad


def gradient_hardmax(a):
    """Compute subgradient of 2n * max(conv(a,a)) at the argmax."""
    n = len(a)
    conv = fft_convolve(a)
    k_star = int(np.argmax(conv))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 4.0 * n * a[j]
    return grad


def multiplicative_update(a, t, lr):
    """One step of multiplicative gradient descent."""
    grad = gradient_softmax(a, t)
    # Normalize gradient for stable updates
    g_max = float(grad.max())
    if g_max < 1e-30:
        return a.copy()
    log_factor = -lr * grad / g_max
    # Clip to avoid underflow
    log_factor = np.maximum(log_factor, -50.0)
    a_new = a * np.exp(log_factor)
    s = float(a_new.sum())
    if s < 1e-30:
        return a.copy()
    a_new /= s
    return a_new


def solve_lp_direction(f, M):
    """LP: maximize sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    # Build constraint matrix A where A[k, j] = f[k-j] if 0<=k-j<n else 0
    # conv(f,g)[k] = sum_j f[k-j]*g[j]
    # But we need A[k,j] such that (A @ g)[k] = conv(f,g)[k]
    # conv(f,g)[k] = sum_j f[k-j] * g[j] (for valid indices)
    # So A[k, j] = f[k-j] if 0 <= k-j < n else 0
    # Wait, that's not right. Let me reconsider.
    # conv(f,g)[k] = sum_j f[j] * g[k-j]
    # So A[k, j] = g[k-j]... no, A should be in terms of g.
    # conv(f,g)[k] = sum_j f[j] * g[k-j]
    # Let me reindex: = sum_m f[k-m] * g[m] (substituting m = k-j)
    # So A[k, m] = f[k-m] if 0 <= k-m < n else 0
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for m in range(n):

            if 0 <= j < n:
                A[k, m] = f[j]
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b = np.full(2 * n - 1, M)

    try:

        if res.success:
            g = np.maximum(res.x, 0.0)
            s = float(g.sum())
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return f.copy()

def local_search(a, time_limit):
    """Targeted local search: transfer mass from peak-contributing to non-contributing elements."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = float(a.sum())
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    best_obj = compute_objective(a)
    t0 = time.monotonic()
    stuck = 0
    max_stuck = 200
    
    while time.monotonic() - t0 < time_limit and stuck < max_stuck:
        conv = fft_convolve(a)
        k_star = int(np.argmax(conv))
        
        # Compute contribution of each pair to conv[k_star]
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]
        
        max_c = float(contrib.max())
        if max_c < 1e-15:
            break
        
        # Elements with high contribution (danger) and low contribution (safe)
        thr_high = 0.5 * max_c
        thr_low = 0.1 * max_c
        danger = np.where(contrib > thr_high)[0]
        safe = np.where(contrib < thr_low)[0]
        
        if len(danger) == 0 or len(safe) == 0:
            stuck += 1
            continue
        
        found = False
        # Try transferring from danger to safe
        for di in danger[:20]:
            if found:
                break
            for si in safe[:20]:
                if found:
                    break
                if di == si:
                    continue
                for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                    amount = a[di] * frac
                    if amount < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[di] -= amount
                    a_trial[si] += amount
                    a_trial = np.maximum(a_trial, 0.0)
                    ts = float(a_trial.sum())
                    if ts < 1e-12:
                        continue
                    a_trial /= ts
                    o = compute_objective(a_trial)
                    if o < best_obj - 1e-10:
                        best_obj = o
                        a = a_trial.copy()
                        found = True
                        break
        
        if not found:
            stuck += 1
        else:
            stuck = 0
    
    return a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Soft-max multiplicative gradient descent with temperature annealing (main driver)
2. LP-based direction finding (periodic refinement)
3. Targeted local search via mass transfer and fine-tuning (final polish)
Starts from parent sequence, explores perturbations, and uses adaptive schedules.
"""

import numpy as np
import scipy.optimize
import time


def fft_convolve(a, b=None):
    """Linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def compute_objective(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1, a>=0."""
    n = len(a)
    conv = fft_convolve(a)
    return 2.0 * float(n) * float(conv.max())


def build_conv_matrix(f):
    """Build matrix A such that A @ g = conv(f, g) for any g of same length."""
    n = len(f)
    size = 2 * n - 1
    A = np.zeros((size, n))
    for m in range(n):
        k_start = m
        k_end = m + n
        A[k_start:k_end, m] = f
    return A


def solve_lp_direction(f, M):
    """LP: maximize sum(g) s.t. conv(f,g) <= M, g >= 0. Returns normalized g."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                                     method='highs', options={'presolve': True})
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = float(g.sum())
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return f.copy()


def gradient_softmax(a, t):
    """Compute gradient of soft-max objective. Returns grad where grad[i] = 4n*sum_j a[j]*w[i+j]."""
    n = len(a)
    conv = fft_convolve(a)
    conv_scaled = conv / t
    conv_scaled -= conv_scaled.max()
    w = np.exp(conv_scaled)
    w /= w.sum()
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = 4.0 * n * float(np.dot(a, w[i:i + n]))
    return grad


def multiplicative_step(a, t, lr):
    """One step of multiplicative gradient descent with soft-max objective."""
    grad = gradient_softmax(a, t)
    g_max = float(grad.max())
    if g_max < 1e-30:
        return a.copy()
    log_factor = -lr * grad / g_max
    log_factor = np.maximum(log_factor, -50.0)
    a_new = a * np.exp(log_factor)
    s = float(a_new.sum())
    if s < 1e-30:
        return a.copy()
    a_new /= s
    return a_new


def lp_refine(a, time_limit):
    """LP-based refinement: iterate LP direction + line search."""
    f = np.maximum(a.astype(np.float64), 0.0).copy()
    s = float(f.sum())
    if s < 1e-12:
        f = np.ones_like(f) / len(f)
    else:
        f /= s
    best_f = f.copy()
    best_obj = compute_objective(f)
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 25

    while time.monotonic() - t0 < time_limit:
        if no_improve >= max_no_improve:
            break
        conv = fft_convolve(f)
        M = float(conv.max())
        if M < 1e-15:
            break

        g = solve_lp_direction(f, M)
        if np.allclose(g, f, atol=1e-10):
            no_improve += 5
            continue

        best_t = 0.0
        best_obj_t = best_obj
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_new = (1.0 - t) * f + t * g
            a_new = np.maximum(a_new, 0.0)
            ns = float(a_new.sum())
            if ns < 1e-12:
                continue
            a_new /= ns
            obj_t = compute_objective(a_new)
            if obj_t < best_obj_t - 1e-12:
                best_obj_t = obj_t
                best_t = t

        if best_t > 0 and best_obj_t < best_obj - 1e-10:
            a_new = (1.0 - best_t) * f + best_t * g
            a_new = np.maximum(a_new, 0.0)
            a_new /= a_new.sum()
            f = a_new
            best_obj = best_obj_t
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

    return best_f, best_obj


def local_search(a, time_limit):
    """Targeted local search: transfer mass from peak-contributing to safe elements."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = float(a.sum())
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    best_obj = compute_objective(a)
    t0 = time.monotonic()
    stuck = 0
    max_stuck = 300

    while time.monotonic() - t0 < time_limit and stuck < max_stuck:
        conv = fft_convolve(a)
        k_star = int(np.argmax(conv))

        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]

        max_c = float(contrib.max())
        if max_c < 1e-15:
            break

        thr_high = 0.5 * max_c
        thr_low = 0.1 * max_c
        danger = np.where(contrib > thr_high)[0]
        safe = np.where(contrib < thr_low)[0]

        if len(danger) == 0 or len(safe) == 0:
            stuck += 1
            continue

        found = False
        for di in danger[:25]:
            if found:
                break
            for si in safe[:25]:
                if found:
                    break
                if di == si:
                    continue
                for frac in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7]:
                    amount = a[di] * frac
                    if amount < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[di] -= amount
                    a_trial[si] += amount
                    a_trial = np.maximum(a_trial, 0.0)
                    ts = float(a_trial.sum())
                    if ts < 1e-12:
                        continue
                    a_trial /= ts
                    o = compute_objective(a_trial)
                    if o < best_obj - 1e-10:
                        best_obj = o
                        a = a_trial.copy()
                        found = True
                        break

        if not found:
            stuck += 1
        else:
            stuck = 0

    return a, best_obj


def fine_tune(a, time_limit, rng):
    """Fine-tuning: small random multiplicative perturbations, keep improvements."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = float(a.sum())
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    best_obj = compute_objective(a)
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 10000

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        num_pert = int(rng.integers(1, 4))
        indices = rng.integers(0, n, size=num_pert)

        a_trial = a.copy()
        for idx in indices:
            factor = 1.0 + rng.uniform(-0.03, 0.03)
            a_trial[idx] *= factor
            if a_trial[idx] < 0.0:
                a_trial[idx] = 0.0

        a_trial = np.maximum(a_trial, 0.0)
        ts = float(a_trial.sum())
        if ts < 1e-10:
            no_improve += 1
            continue
        a_trial /= ts
        o = compute_objective(a_trial)
        if o < best_obj - 1e-12:
            best_obj = o
            a = a_trial.copy()
            no_improve = 0
        else:
            no_improve += 1

    return a, best_obj


def get_parent(n):
    """Get parent sequence truncated/interpolated to length n, normalized."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim != 1 or p.size == 0 or not np.isfinite(p).all():
            return None
        if p.size >= n:
            pc = p[:n].copy()
        else:
            pc = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, p.size), p)
        pc = np.maximum(pc, 0.0)
        s = float(pc.sum())
        if s < 1e-10:
            return None
        return pc / s
    except Exception:
        return None


def main_optimize(a0, time_limit, use_lp=True, lp_interval=300):
    """Main optimization: multiplicative gradient descent + periodic LP."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = float(a.sum())
    if s < 1e-12:
        a = np.ones_like(a) / len(a)
    else:
        a /= s

    best_a = a.copy()
    best_obj = compute_objective(a)
    t0 = time.monotonic()
    it = 0

    t_max = 0.08
    t_min = 0.003
    lr_start = 0.6
    lr_end = 0.03
    warmup_frac = 0.1

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        t_cur = t_max * (t_min / t_max) ** progress
        lr_cur = lr_start * (lr_end / lr_start) ** progress

        a_new = multiplicative_step(a, t_cur, lr_cur)
        obj_new = compute_objective(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        it += 1

        if use_lp and it % lp_interval == 0:
            remaining = time_limit - (time.monotonic() - t0)
            if remaining > 10:
                lp_time = min(20.0, remaining * 0.08)
                a_lp, obj_lp = lp_refine(a, time_limit=lp_time)
                if obj_lp < best_obj - 1e-10:
                    best_obj = obj_lp
                    best_a = a_lp.copy()
                    a = a_lp.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Minimizes 2n*max(conv(a,a))/(sum(a))^2 using
    multiplicative gradient descent with soft-max annealing, LP refinement,
    local search, and fine-tuning. Starts from parent sequence with perturbations.
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + total_budget - 10.0

    best_sequence = None
    best_score = float('inf')

    n = 512
    parent = get_parent(n)
    if parent is None:
        parent = np.ones(n) / n

    parent_score = float(evaluate_sequence(parent.tolist()))
    print(f"Parent: score={parent_score:.6f}, length={n}")

    # Phase 1: Quick exploration (15% of time)
    phase1_time = (total_budget - 10.0) * 0.15
    print(f"Phase 1: Quick exploration ({phase1_time:.0f}s)")
    t0 = time.monotonic()

    # Optimize parent directly
    a1, obj1 = main_optimize(parent, time_limit=phase1_time * 0.6, use_lp=False)
    score1 = float(evaluate_sequence(a1.tolist()))
    if np.isfinite(score1) and score1 < best_score:
        best_score = score1
        best_sequence = a1.copy()
    print(f"  Parent optimized: obj={obj1:.6f}, score={score1:.6f}")

    # Try perturbations of the parent
    elapsed = time.monotonic() - t0
    remaining_p1 = phase1_time - elapsed
    if remaining_p1 > 5:
        for pert_scale in [0.02, 0.05, 0.1]:
            if time.monotonic() > deadline:
                break
            a_pert = parent * (1.0 + pert_scale * rng.standard_normal(n))
            a_pert = np.maximum(a_pert, 0.0)
            s = float(a_pert.sum())
            if s < 1e-10:
                continue
            a_pert /= s
            t_alloc = min(8.0, max(2.0, remaining_p1 * 0.25))
            a_opt, obj_opt = main_optimize(a_pert, time_limit=t_alloc, use_lp=False)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Pert {pert_scale}: score={score:.6f}")
            remaining_p1 = phase1_time - (time.monotonic() - t0)

    # Phase 2: Deep optimization with LP (50% of time)
    phase2_time = (total_budget - 10.0) * 0.50
    remaining = min(phase2_time, deadline - time.monotonic())
    print(f"Phase 2: Deep optimization with LP ({remaining:.0f}s)")

    if best_sequence is not None and remaining > 10:
        a2, obj2 = main_optimize(best_sequence, time_limit=remaining, use_lp=True,
                                 lp_interval=200)
        score2 = float(evaluate_sequence(a2.tolist()))
        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_sequence = a2.copy()
        print(f"  Deep opt: obj={obj2:.6f}, score={score2:.6f}")

    # Phase 3: Polish - local search + fine tune + final LP (35% of time)
    phase3_time = (total_budget - 10.0) * 0.35
    remaining = min(phase3_time, deadline - time.monotonic())
    print(f"Phase 3: Polish ({remaining:.0f}s)")

    if best_sequence is not None and remaining > 5:
        # Local search
        ls_time = remaining * 0.35
        remaining = deadline - time.monotonic()
        ls_time = min(ls_time, max(3.0, remaining - 5.0))
        a3, obj3 = local_search(best_sequence, time_limit=ls_time)
        score3 = float(evaluate_sequence(a3.tolist()))
        if np.isfinite(score3) and score3 < best_score:
            best_score = score3
            best_sequence = a3.copy()
        print(f"  Local search: obj={obj3:.6f}, score={score3:.6f}")

        # Fine tune
        remaining = deadline - time.monotonic()
        ft_time = min(remaining * 0.3, max(3.0, (total_budget - 10.0) * 0.10))
        ft_time = min(ft_time, max(3.0, remaining - 5.0))
        a4, obj4 = fine_tune(best_sequence, time_limit=ft_time, rng=rng)
        score4 = float(evaluate_sequence(a4.tolist()))
        if np.isfinite(score4) and score4 < best_score:
            best_score = score4
            best_sequence = a4.copy()
        print(f"  Fine tune: obj={obj4:.6f}, score={score4:.6f}")

        # Final LP
        remaining = deadline - time.monotonic()
        if remaining > 5:
            lp_time = min(remaining - 5.0, (total_budget - 10.0) * 0.10)
            a5, obj5 = lp_refine(best_sequence, time_limit=lp_time)
            score5 = float(evaluate_sequence(a5.tolist()))
            if np.isfinite(score5) and score5 < best_score:
                best_score = score5
                best_sequence = a5.copy()
            print(f"  Final LP: obj={obj5:.6f}, score={score5:.6f}")

        # One more round of fine tune + local search
        remaining = deadline - time.monotonic()
        if remaining > 5:
            ft2_time = min(remaining * 0.4, max(3.0, 10.0))
            a6, obj6 = fine_tune(best_sequence, time_limit=ft2_time, rng=rng)
            score6 = float(evaluate_sequence(a6.tolist()))
            if np.isfinite(score6) and score6 < best_score:
                best_score = score6
                best_sequence = a6.copy()

            remaining = deadline - time.monotonic()
            if remaining > 5:
                ls2_time = min(remaining - 5.0, 10.0)
                a7, obj7 = local_search(best_sequence, time_limit=ls2_time)
                score7 = float(evaluate_sequence(a7.tolist()))
                if np.isfinite(score7) and score7 < best_score:
                    best_score = score7
                    best_sequence = a7.copy()

    # Final verification
    if best_sequence is None:
        best_sequence = parent
        best_score = parent_score

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"FINAL: score={best_score:.6f}, length={len(best_sequence)}")
    return [float(x) for x in best_sequence]
