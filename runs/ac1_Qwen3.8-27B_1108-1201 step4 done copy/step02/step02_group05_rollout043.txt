```
def fast_convolve(a):
    a=np.asarray(a, dtype=float)
    n=len(a)
    if n==0: return np.array([])
    size=2*n-1
    fft_size=1 << (size-1).bit_length()
    A=np.fft.rfft(a, fft_size)
    c=np.fft.irfft(A*A, fft_size)[:size]
    c[c<0.0]=0.0
    return c
```

```python
"""
Search for low-peak nonnegative convolution coefficients.

The objective is scale-invariant:
    2 * n * max_k (a * a)[k] / (sum a)^2.

This program combines several complementary local searches:

1. Fast FFT-based evaluations.
2. Exact one-coordinate minimization:
   Change one coefficient by an unnormalized shift and normalize.
   The induced objective is a 1D max-of-quadratics problem, solved
   by a small grid plus golden-section refinement.
3. Exact two-coordinate mass-transfer moves:
   Move mass between a "dangerous" coefficient and a "safe" one.
   Again the induced objective is 1D and is minimized directly.
4. A low-dimensional exponential/Legendre-like family optimization for
   strong global starting points.
5. A linear-programming direction from the previous state-of-art idea,
   used only as a final polish.

The returned sequence is the best sequence found within the time budget.
"""

import math
import time

import numpy as np
import scipy.optimize


# ----------------------------------------------------------------------
# Basic convex-construction operations
# ----------------------------------------------------------------------

def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    a = np.asarray(a, dtype=float)
    n = len(a)
    if n == 0:
        return np.array([], dtype=float)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]
    c[c < 0.0] = 0.0
    return c


def normalize(a):
    """Clip to nonnegative finite values and normalize sum to 1."""
    a = np.asarray(a, dtype=float).copy()
    a[~np.isfinite(a)] = 0.0
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-12:
        if len(a) > 0:
            a.fill(1.0 / float(len(a)))
        else:
            a = np.ones(1, dtype=float)
        return a
    a /= s
    return a


def internal_score(a):
    """Internal objective assumes a is already normalized to sum 1."""
    a = np.asarray(a, dtype=float)
    if len(a) == 0:
        return np.inf
    s = float(a.sum())
    if s < 1e-12:
        return np.inf
    b = fast_convolve(a / s)
    return 2.0 * float(len(a)) * float(b.max()) / (s * s)


# ----------------------------------------------------------------------
# Weights for active diagonals
# ----------------------------------------------------------------------

def active_weights(A, M, p=20.0):
    """Smooth weights emphasizing diagonals near the current peak."""
    K = len(A)
    if K == 0:
        return np.array([], dtype=float)
    if M <= 1e-18:
        return np.full(K, 1.0 / float(K))
    z = A / M
    z = np.clip(z, 0.0, 1.0)
    w = np.power(z, p)
    w[A < 1e-14 * M] = 0.0
    s = float(w.sum())
    if s <= 1e-18:
        w = np.ones(K, dtype=float)
        s = float(K)
    return w / s


def cross_weights(a, w):
    """
    r_i = sum_j a_j w_{i+j}, with w outside [0, 2n-2] treated as 0.
    This is the row-gradient vector for bilinear forms with diagonal weights w.
    """
    a = np.asarray(a, dtype=float)
    w = np.asarray(w, dtype=float)
    n = len(a)
    if n == 0:
        return np.array([], dtype=float)
    K = 2 * n - 2
    wr = w[::-1]
    conv_len = n + len(w) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    c = np.fft.irfft(np.fft.rfft(a, fft_size) * np.fft.rfft(wr, fft_size), fft_size)
    seg = c[n - 1:K + 1][:n][::-1]
    seg[seg < 0.0] = 0.0
    return seg


# ----------------------------------------------------------------------
# Exact one-coordinate optimization
# ----------------------------------------------------------------------

def coord_eval(d, A, E2, self_idx):
    """Objective value when coordinate is shifted by d, then normalized."""
    den = 1.0 + d
    if den <= 1e-12:
        return np.inf
    num = A + d * E2
    num[self_idx] += d * d
    return float(num.max()) / (den * den)


def golden_min_coord(lo, hi, A, E2, self_idx, iters=25):
    """Golden-section minimization for one-coordinate objective."""
    if lo > hi:
        lo, hi = hi, lo
    if hi - lo < 1e-12:
        val = coord_eval(lo, A, E2, self_idx)
        return lo, val

    gr = (math.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = coord_eval(c, A, E2, self_idx)
    fd = coord_eval(d, A, E2, self_idx)

    for _ in range(iters):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = coord_eval(c, A, E2, self_idx)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = coord_eval(d, A, E2, self_idx)

    t = 0.5 * (lo + hi)
    ft = coord_eval(t, A, E2, self_idx)

    best_t = t
    best_v = ft
    for ep in (lo, hi, c, d):
        vv = coord_eval(ep, A, E2, self_idx)
        if vv < best_v:
            best_v = vv
            best_t = ep
    return best_t, best_v


def optimize_coordinate(a, i, A, M, dlo, dhi):
    """
    Find the best unnormalized shift d for coordinate i:
        y = a, y_i = a_i + d, then normalize y / sum(y).
    Returns (d, estimated_new_peak) or (None, None).
    """
    n = len(a)
    K = len(A)
    if dhi <= dlo:
        return None, None
    if dlo < -1.0 + 1e-10:
        dlo = -1.0 + 1e-10
    if 1.0 + dlo <= 1e-9:
        dlo = -1.0 + 1e-8
    if 1.0 + dhi <= 1e-9:
        return None, None

    E = np.zeros(K, dtype=float)
    E[i:i + n] = a
    E2 = 2.0 * E
    self_idx = 2 * i

    ng = 24
    ds = np.linspace(dlo, dhi, ng)
    ds = ds[(1.0 + ds) > 1e-9]
    if ds.size == 0:
        return None, None

    den = (1.0 + ds) ** 2
    num = A[None, :] + ds[:, None] * E2[None, :]
    num[:, self_idx] += ds[:, None] * ds[:, None]
    R = num.max(axis=1) / den

    idx = int(np.argmin(R))
    best_R = float(R[idx])
    best_d = float(ds[idx])

    if idx > 0 and idx < ds.size - 1:
        lo = ds[idx - 1]
        hi = ds[idx + 1]
    elif idx == 0:
        lo = ds[0]
        hi = ds[min(1, ds.size - 1)]
    else:
        lo = ds[max(0, ds.size - 2)]
        hi = ds[-1]

    if hi > lo:
        d2, R2 = golden_min_coord(lo, hi, A, E2, self_idx, 25)
        if R2 < best_R:
            best_R = R2
            best_d = d2

    if best_R < M - 1e-10:
        return best_d, best_R
    return None, None


def apply_coord_update(a, i, d):
    """Apply an exact one-coordinate shift update."""
    ns = 1.0 + d
    if ns <= 1e-12:
        return None
    na = a / ns
    na[i] = (a[i] + d) / ns
    na = np.maximum(na, 0.0)
    s = float(na.sum())
    if s < 1e-12:
        return None
    return na / s


def coordinate_descent(a, end_time, top_frac=0.4, max_accepts=25):
    """
    Cyclic coordinate descent with exact 1D coordinate minimization.
    Coordinates are prioritized by distance of their active-diagonal gradient
    from the current peak.
    """
    a = normalize(a)
    n = len(a)
    A = fast_convolve(a)
    M = float(A.max())

    sweep = 0
    no_improve = 0
    tf = float(top_frac)

    while time.monotonic() < end_time - 2.0:
        A = fast_convolve(a)
        M = float(A.max())
        if M <= 1e-18:
            break

        w = active_weights(A, M, 20.0)
        r = cross_weights(a, w)
        dhi_global = max(0.5, 5.0 * float(np.max(a)))

        pot = np.abs(r - M)
        order = np.argsort(pot)[::-1]

        num_try = max(8, int(n * tf))
        if num_try > n:
            num_try = n

        accepts = 0
        for idx in range(num_try):
            if time.monotonic() > end_time - 5.0:
                break
            i = int(order[idx])
            dlo = -float(a[i])
            d, newM = optimize_coordinate(a, i, A, M, dlo, dhi_global)
            if d is None:
                continue

            na = apply_coord_update(a, i, d)
            if na is None:
                continue

            Anew = fast_convolve(na)
            Mnew = float(Anew.max())
            if Mnew < M - 1e-10:
                a = na
                A = Anew
                M = Mnew
                accepts += 1
                if accepts >= max_accepts:
                    break

        sweep += 1
        if accepts == 0:
            no_improve += 1
            if no_improve >= 3:
                break
            tf = min(1.0, tf * 1.6)
        else:
            no_improve = 0

    return a


# ----------------------------------------------------------------------
# Exact two-coordinate mass transfer optimization
# ----------------------------------------------------------------------

def pair_eval(d, A, D2, Q):
    """Objective for moving d from i to j (positive d: i -> j)."""
    num = A + d * D2 + (d * d) * Q
    return float(num.max())


def golden_min_pair(lo, hi, A, D2, Q, iters=25):
    """Golden-section minimization for two-coordinate move."""
    if lo > hi:
        lo, hi = hi, lo
    if hi - lo < 1e-12:
        return lo, pair_eval(lo, A, D2, Q)

    gr = (math.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = pair_eval(c, A, D2, Q)
    fd = pair_eval(d, A, D2, Q)

    for _ in range(iters):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = pair_eval(c, A, D2, Q)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = pair_eval(d, A, D2, Q)

    t = 0.5 * (lo + hi)
    ft = pair_eval(t, A, D2, Q)

    best_t = t
    best_v = ft
    for ep in (lo, hi, c, d):
        vv = pair_eval(ep, A, D2, Q)
        if vv < best_v:
            best_v = vv
            best_t = ep
    return best_t, best_v


def optimize_pair(a, i, j, A, M):
    """
    Find best mass transfer d from i to j.
    Positive d moves mass i -> j; negative d moves j -> i.
    """
    n = len(a)
    K = len(A)
    if i == j:
        return None, None

    dlo = -float(a[j])
    dhi = float(a[i])
    if dhi <= dlo:
        return None, None

    D = np.zeros(K, dtype=float)
    D[j:j + n] += a
    D[i:i + n] -= a
    D2 = 2.0 * D

    Q = np.zeros(K, dtype=float)
    Q[2 * i] += 1.0
    Q[2 * j] += 1.0
    Q[i + j] -= 2.0

    ng = 24
    ds = np.linspace(dlo, dhi, ng)
    ds2 = ds * ds
    num = A[None, :] + ds[:, None] * D2[None, :] + ds2[:, None] * Q[None, :]
    R = num.max(axis=1)

    idx = int(np.argmin(R))
    best_R = float(R[idx])
    best_d = float(ds[idx])

    if idx > 0 and idx < ds.size - 1:
        lo = ds[idx - 1]
        hi = ds[idx + 1]
    elif idx == 0:
        lo = ds[0]
        hi = ds[min(1, ds.size - 1)]
    else:
        lo = ds[max(0, ds.size - 2)]
        hi = ds[-1]

    if hi > lo:
        d2, R2 = golden_min_pair(lo, hi, A, D2, Q, 25)
        if R2 < best_R:
            best_R = R2
            best_d = d2

    if best_R < M - 1e-10:
        return best_d, best_R
    return None, None


def apply_pair(a, i, j, d):
    """Apply a two-coordinate mass transfer update."""
    na = a.copy()
    na[i] -= d
    na[j] += d
    na = np.maximum(na, 0.0)
    return normalize(na)


def pair_polish(a, end_time, count=16):
    """
    Polish with exact two-coordinate moves.
    Pairs high-gradient "danger" coordinates with low-gradient "safe" ones.
    """
    a = normalize(a)
    no_improve = 0

    while time.monotonic() < end_time - 2.0 and no_improve < 4:
        A = fast_convolve(a)
        M = float(A.max())
        if M <= 1e-18:
            break

        w = active_weights(A, M, 20.0)
        r = cross_weights(a, w)
        order = np.argsort(r)

        c = min(count, len(order))
        safes = order[:c]
        dangers = order[::-1][:c]

        accepted = False
        for ii in dangers:
            if accepted:
                break
            for jj in safes:
                i = int(ii)
                j = int(jj)
                if i == j:
                    continue

                d, newM = optimize_pair(a, i, j, A, M)
                if d is None:
                    continue

                na = apply_pair(a, i, j, d)
                Anew = fast_convolve(na)
                Mnew = float(Anew.max())
                if Mnew < M - 1e-10:
                    a = na
                    A = Anew
                    M = Mnew
                    accepted = True
                    break

        if not accepted:
            no_improve += 1
        else:
            no_improve = 0

    return a

# Optional LP polish from the previous method of record
# ----------------------------------------------------------------------

def build_convolution_matrix(f):
    """A[k, j] = f[k-j] when valid, else 0."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=float)
    for j in range(n):
        L = n - j
        A[j:j + L, j] = f[:L]
    return A


def solve_lp_direction(f, M):
    """
    Maximize sum(g) subject to conv(f, g) <= M, g >= 0.
    Return normalized g, or None.
    """
    n = len(f)
    if n > 1200:
        return None
    try:
        A = build_convolution_matrix(f)
        c = -np.ones(n)
        b = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        res = scipy.optimize.linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method="highs"
        )
    except Exception:
        return None

    if not res.success or res.x is None:
        return None
    g = np.maximum(np.asarray(res.x, dtype=float), 0.0)
    s = float(g.sum())
    if s < 1e-12:
        return None
    return g / s


def combo_max_conv(a, g, t):
    """max conv coefficient for (1-t)a + t g."""
    if t <= 0.0:
        return float(fast_convolve(a).max())
    if t >= 1.0:
        return float(fast_convolve(g).max())
    arr = (1.0 - t) * a + t * g
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if s < 1e-12:
        return np.inf
    arr = arr / s
    return float(fast_convolve(arr).max())


def golden_line_search(a, g, iters=35):
    """Best t in [0,1] for path t -> (1-t)a + t g."""
    lo = 0.0
    hi = 1.0
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = combo_max_conv(a, g, c)
    fd = combo_max_conv(a, g, d)

    for _ in range(iters):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = combo_max_conv(a, g, c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = combo_max_conv(a, g, d)

    t = 0.5 * (lo + hi)
    ft = combo_max_conv(a, g, t)

    best_t = t
    best_v = ft
    for ep in (0.0, 1.0, lo, hi, c, d):
        vv = combo_max_conv(a, g, ep)
        if vv < best_v:
            best_v = vv
            best_t = ep
    return best_t, best_v


def lp_local_optimize(a, end_time):
    """
    A few LP directions with line search.
    Used only as a final polish; not the main engine.
    """
    a = normalize(a)
    no_improve = 0

    while time.monotonic() < end_time - 5.0 and no_improve < 12:
        A = fast_convolve(a)
        M = float(A.max())
        if M <= 1e-18:
            break

        g = solve_lp_direction(a, M)
        if g is None:
            no_improve += 1
            continue

        if np.allclose(g, a, atol=1e-10):
            no_improve += 5
            continue

        t, _ = golden_line_search(a, g, 35)
        cand = (1.0 - t) * a + t * g
        cand = normalize(cand)
        Mcand = float(fast_convolve(cand).max())
        if Mcand < M - 1e-10:
            a = cand
            no_improve = 0
        else:
            no_improve += 1

    return a

# Initial constructions

def resample_parent(parent, n):
    """Interpolate global parent construction to length n."""
    try:
        p = np.asarray(parent, dtype=float)
        p = p[np.isfinite(p)]
        if p.size == 0:
            return None
        p = np.clip(p, 0.0, 1000.0)
        if p.sum() <= 1e-12:
            return None
        if p.size == 1:
            arr = np.full(n, float(p[0]), dtype=float)
        else:
            xi = np.linspace(0.0, 1.0, p.size)
            x = np.linspace(0.0, 1.0, n)
            arr = np.interp(x, xi, p)
        return normalize(arr)
    except Exception:
        return None


def make_initializations(n, rng, parent):
    """Generate a diverse set of normalized starting vectors."""
    inits = []

    if parent is not None:
        p = resample_parent(parent, n)
        if p is not None:
            inits.append(p)

    inits.append(np.ones(n, dtype=float) / float(n))

    inits.append(normalize(rng.dirichlet(np.full(n, 0.3))))
    inits.append(normalize(rng.dirichlet(np.full(n, 1.0))))

    x = np.arange(n, dtype=float)

    for theta in (0.6, 0.9):
        a = 1.0 + theta * np.sin(2.0 * np.pi * (x + 0.5) / float(n))
        a = np.maximum(a, 0.0)
        if a.sum() > 1e-12:
            inits.append(normalize(a))

    xf = (x + 0.5) / float(n)

    a = np.power(xf, 1.0) * np.power(1.0 - xf, 2.0)
    if a.sum() > 1e-12:
        inits.append(normalize(a))

    a = np.power(xf, 2.0) * np.power(1.0 - xf, 1.0)
    if a.sum() > 1e-12:
        inits.append(normalize(a))

    m = max(1, int(0.35 * n))
    a = np.zeros(n, dtype=float)
    a[:m] = 1.0
    a[n - m:] = 1.0
    a = np.clip(a, 0.0, 1.0)
    if a.sum() > 1e-12:
        inits.append(normalize(a))

    return inits


# ----------------------------------------------------------------------
# Low-dimensional parametric family
# ----------------------------------------------------------------------

def family_make_array(params, n, u):
    """Build nonnegative array from exponential polynomial coefficients."""
    u2 = u * u
    u3 = u2 * u
    u4 = u2 * u2
    u5 = u4 * u

    z = (
        params[0] * u
        + params[1] * u2
        + params[2] * u3
        + params[3] * u4
        + params[4] * u5
    )
    z = z - float(np.max(z))
    a = np.exp(z)
    return normalize(a)


def family_objective(params, n, u):
    """Objective for the exponential polynomial family."""
    a = family_make_array(params, n, u)
    if a is None or len(a) != n:
        return 1e100
    M = float(fast_convolve(a).max())
    return 2.0 * float(n) * M


def family_optimize(n, end_time, rng):
    """Optimize a low-dimensional exponential polynomial family."""
    if n < 8:
        return None

    u = 2.0 * np.arange(n, dtype=float) / float(n - 1) - 1.0

    starts = [
        np.zeros(5, dtype=float),
        rng.normal(0.0, 0.6, 5).astype(float),
        np.array([0.6, 0.0, -0.4, 0.0, 0.3], dtype=float),
        np.array([-0.6, 0.0, 0.4, 0.0, -0.3], dtype=float),
    ]

    best_params = None
    best_val = np.inf

    for sp in starts:
        if time.monotonic() >= end_time - 2.0:
            break
        try:
            res = scipy.optimize.minimize(
                family_objective,
                sp,
                args=(n, u),
                method="Nelder-Mead",
                options={
                    "maxiter": 350,
                    "xatol": 1e-5,
                    "fatol": 1e-7,
                    "adaptive": True
                }
            )
            if res.fun < best_val:
                best_val = float(res.fun)
                best_params = np.asarray(res.x, dtype=float)
        except Exception:
            pass

    if best_params is None:
        return None
    return family_make_array(best_params, n, u)

# Main entry point
# ----------------------------------------------------------------------

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Return the best sequence found within budget_s seconds.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    budget = float(budget_s)
    if budget < 300.0:
        budget = 300.0
    deadline = start + budget - 30.0

    # Try to use the provided parent construction, if present.
    parent = None
    try:
        pv = globals().get("height_sequence_1")
        if pv is not None:
            parent = np.asarray(pv, dtype=float)
            if parent.size == 0 or not np.all(np.isfinite(parent)):
                parent = None
    except Exception:
        parent = None

    candidates = []

    lengths = [256, 512, 1024]
    if parent is not None:
        pl = int(parent.size)
        if 64 <= pl <= 2048:
            lengths.append(pl)
    if budget > 1500.0:
        lengths.append(2048)

    unique_lengths = []
    for n in lengths:
        if n not in unique_lengths:
            unique_lengths.append(n)
    unique_lengths.sort()

    # ------------------------------------------------------------------
    # Phase 1: broad quick scan
    # ------------------------------------------------------------------
    phase1_end = start + (budget - 30.0) * 0.40

    for n in unique_lengths:
        if time.monotonic() > phase1_end - 30.0:
            break

        inits = make_initializations(n, rng, parent)
        max_inits = 6 if n <= 512 else 4

        for a0 in inits[:max_inits]:
            if time.monotonic() > phase1_end - 15.0:
                break

            quick_time = 6.0 if n <= 512 else 4.0
            quick_end = min(phase1_end - 5.0, time.monotonic() + quick_time)
            a_opt = coordinate_descent(a0, quick_end, top_frac=0.35, max_accepts=12)
            score = 2.0 * float(n) * float(fast_convolve(a_opt).max())
            if np.isfinite(score):
                candidates.append([score, a_opt])

        # Low-dimensional family for medium lengths.
        if n in (512, 1024) and time.monotonic() < phase1_end - 30.0:
            fam_end = min(phase1_end - 15.0, time.monotonic() + 20.0)
            af = family_optimize(n, fam_end, rng)
            if af is not None:
                coord_end = min(phase1_end - 3.0, time.monotonic() + 10.0)
                a_opt = coordinate_descent(af, coord_end, top_frac=0.45, max_accepts=20)
                score = 2.0 * float(n) * float(fast_convolve(a_opt).max())
                if np.isfinite(score):
                    candidates.append([score, a_opt])

    if not candidates:
        a = np.ones(512, dtype=float) / 512.0
        candidates.append([2.0 * 512.0 * float(fast_convolve(a).max()), a])

    candidates.sort(key=candidate_key)
    top_candidates = candidates[:8]

    best_seq = None
    best_score = np.inf
    for score, a in top_candidates:
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()

    print(f"Phase 1 best internal score: {best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 2: deep optimization on top candidates
    # ------------------------------------------------------------------
    phase2_end = start + (budget - 30.0) * 0.85
    deep_count = min(5, len(top_candidates))

    for idx in range(deep_count):
        if time.monotonic() > phase2_end - 60.0:
            break

        remaining = phase2_end - time.monotonic()
        remain_count = deep_count - idx
        alloc = remaining / float(max(1, remain_count))
        alloc = min(300.0, max(40.0, alloc))
        deep_end = time.monotonic() + alloc
        if deep_end > phase2_end - 15.0:
            deep_end = phase2_end - 15.0

        a0 = top_candidates[idx][1]
        a_cur = coordinate_descent(a0, deep_end, top_frac=0.65, max_accepts=50)
        score = 2.0 * float(len(a_cur)) * float(fast_convolve(a_cur).max())
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_seq = a_cur.copy()
            top_candidates[idx] = [score, a_cur]

        # Exact pair-move polish.
        if time.monotonic() < phase2_end - 30.0:
            pe = min(phase2_end - 10.0, time.monotonic() + min(60.0, alloc * 0.25))
            a_p = pair_polish(a_cur, pe, count=16)
            sp = 2.0 * float(len(a_p)) * float(fast_convolve(a_p).max())
            if np.isfinite(sp) and sp < best_score:
                best_score = sp
                best_seq = a_p.copy()
            top_candidates[idx] = [sp, a_p]
            a_cur = a_p

        # LP polish for moderate lengths.
        if len(a_cur) <= 1024 and time.monotonic() < phase2_end - 40.0:
            le = min(phase2_end - 5.0, time.monotonic() + min(60.0, alloc * 0.25))
            a_l = lp_local_optimize(a_cur, le)
            sl = 2.0 * float(len(a_l)) * float(fast_convolve(a_l).max())
            if np.isfinite(sl) and sl < best_score:
                best_score = sl
                best_seq = a_l.copy()

    print(f"Phase 2 best internal score: {best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 3: intensive polish on the best candidate
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.ones(512, dtype=float) / 512.0
        best_score = 2.0 * 512.0 * float(fast_convolve(best_seq).max())

    p1_end = min(deadline - 120.0, time.monotonic() + 240.0)
    a_opt = coordinate_descent(best_seq, p1_end, top_frac=1.0, max_accepts=80)
    s = 2.0 * float(len(a_opt)) * float(fast_convolve(a_opt).max())
    if np.isfinite(s) and s < best_score:
        best_score = s
        best_seq = a_opt

    p2_end = min(deadline - 60.0, time.monotonic() + 90.0)
    a_opt = pair_polish(best_seq, p2_end, count=20)
    s = 2.0 * float(len(a_opt)) * float(fast_convolve(a_opt).max())
    if np.isfinite(s) and s < best_score:
        best_score = s
        best_seq = a_opt

    if len(best_seq) <= 1024 and time.monotonic() < deadline - 50.0:
        le = min(deadline - 20.0, time.monotonic() + 90.0)
        a_opt = lp_local_optimize(best_seq, le)
        s = 2.0 * float(len(a_opt)) * float(fast_convolve(a_opt).max())
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_seq = a_opt

    # Light annealed restarts near the incumbent.
    if time.monotonic() < deadline - 80.0:
        n = len(best_seq)
        if n <= 1024:
            for sigma in (0.15, 0.4):
                if time.monotonic() >= deadline - 40.0:
                    break
                noise = np.exp(rng.normal(0.0, sigma, n))
                a_noise = normalize(best_seq * noise)
                ne = min(deadline - 30.0, time.monotonic() + 60.0)
                a_opt = coordinate_descent(a_noise, ne, top_frac=0.7, max_accepts=40)
                s = 2.0 * float(n) * float(fast_convolve(a_opt).max())
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_seq = a_opt

    final = normalize(best_seq)
    print(f"Final internal score: {best_score:.9f}, length: {len(final)}")

    # Final official evaluation, if available.
    try:
        official = evaluate_sequence([float(x) for x in final])
        if np.isfinite(official):
            print(f"Official score: {official:.9f}")
    except Exception:
        pass

    return [float(x) for x in final]
```
