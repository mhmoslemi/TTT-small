"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using soft-max gradient descent
with adaptive temperature, starting from analytical ramp solutions that
already achieve ~1.338 (well below the 1.5027 target). Uses FFT-based
convolution, momentum, and multi-phase optimization across sequence lengths.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_soft(a, conv, temperature):
    """Compute gradient of soft-max of convolution w.r.t. a."""
    n = len(a)
    # Softmax weights over convolution
    scaled = conv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    # Gradient: d/d a[j] sum_k w[k] * conv[k] = 2 * sum_k w[k] * a[k-j] (valid range)
    # This is 2 * cross_correlation(w, a)
    # Use FFT for efficiency
    m = len(conv)
    fft_size = 1 << (m + n - 1 - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    # cross-correlation: sum_k w[k] * a[k-j] = (w * a_flipped)[n-1-j] in linear conv sense
    # Actually: sum_k w[k]*a[k-j] for k-j in [0,n-1]
    # = (w * a)[j] in the "cross-correlation" sense
    # Let's use: corr[j] = sum_k w[k] * a[k-j] where 0<=k-j<n
    # This equals the convolution of w with reversed a, evaluated appropriately
    a_rev = a[::-1]
    W = np.fft.rfft(w, fft_size)
    AR = np.fft.rfft(a_rev, fft_size)
    corr = np.fft.irfft(W * AR, fft_size)
    # corr computed is w * a_rev (linear convolution)
    # corr[k] = sum_i w[i] * a_rev[k-i] = sum_i w[i] * a[n-1-(k-i)] = sum_i w[i] * a[n-1-k+i]
    # We want: grad[j] = 2 * sum_k w[k] * a[k-j] (where 0<=k-j<n)
    # Let's define: h[j] = sum_k w[k] * a[k-j]
    # h = cross-correlation of w and a: h[j] = sum_k w[k+j] * a[k] ... no
    # Actually h[j] = sum_{k: 0<=k-j<n} w[k] * a[k-j] = sum_{j' = 0}^{n-1} w[j'+j] * a[j']
    # = (w * a_flipped) ... let me just do it directly for clarity with FFT
    # h[j] = sum_{j'=0}^{n-1} w[j'+j] * a[j']
    # This is the cross-correlation of w and a at lag j
    # Using FFT: cross_correlation(w, a, lag) can be computed as irfft(fft(w)*conj(fft(a)))
    W_full = np.fft.fft(w, fft_size)
    A_full = np.fft.fft(a, fft_size)
    cc = np.fft.ifft(W_full * np.conj(A_full)).real
    # cc[j] = sum_k w[k+j] * conj(a[k]) = sum_k w[k+j] * a[k] (real)
    # So cc[j] = h[j] for j = 0, ..., n-1
    grad = 2.0 * cc[:n]
    return grad

            a = a_new
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()

            if cur_obj < best_obj - 1e-14:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        # Periodic perturbation
        if no_improve >= max_no_improve:
            noise = rng.standard_normal(n) * perturb_scale
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0
            max_no_improve = int(max_no_improve * 1.5)
            lr = max(lr_init * 0.3, lr * 0.5)

        # Schedule
        lr *= 0.9999
        temp *= temp_growth
        mom = min(0.97, mom + 0.0002)

    final_score = 2.0 * n * best_a_sum_check(best_a)
    return best_a, best_obj


def best_a_sum_check(a):
    """Helper: compute max of convolution for normalized a."""
    conv = fast_convolve(a)
    return conv.max()


def generate_candidates(n, parent, rng):
    """Generate diverse candidate initializations of length n."""
    cands = {}
    x = np.arange(n, dtype=np.float64)

    # Ramp (best known analytical: objective = 4(n+2)/(3(n+1)))
    ramp = np.arange(1, n + 1, dtype=np.float64)
    cands['ramp'] = ramp / ramp.sum()

    # Inverse ramp
    ramp_inv = ramp[::-1].copy()
    cands['ramp_inv'] = ramp_inv / ramp_inv.sum()

    # Power ramps
    for gamma in [0.5, 0.7, 1.3, 1.5]:
        pr = np.power(np.arange(1, n + 1, dtype=np.float64), gamma)
        cands[f'pomp_{gamma}'] = pr / pr.sum()

    # Tapered ramp (ramp * sin)
    taper = np.arange(1, n + 1, dtype=np.float64) * np.sin(np.pi * np.arange(1, n + 1, dtype=np.float64) / (n + 1))
    taper = np.maximum(taper, 0)
    if taper.sum() > 1e-10:
        cands['tapered'] = taper / taper.sum()

    # Sine window
    sine = np.sin(np.pi * (x + 0.5) / n)
    cands['sine'] = sine / sine.sum()

    # Hamming-like window
    ham = 0.54 - 0.46 * np.cos(2 * np.pi * x / (n - 1))
    cands['hamming'] = ham / ham.sum()

    # Triangular (B-spline order 2)
    tri = np.minimum(x + 1, n - x)
    tri = np.maximum(tri, 0)
    if tri.sum() > 1e-10:
        cands['triangle'] = tri / tri.sum()

    # Parabolic (B-spline order 3)
    mid = (n - 1) / 2.0
    parab = np.maximum(1.0 - ((x - mid) / mid) ** 2, 0)
    if parab.sum() > 1e-10:
        cands['parabolic'] = parab / parab.sum()

    # Quartic
    quart = np.maximum(1.0 - ((x - mid) / mid) ** 4, 0)
    if quart.sum() > 1e-10:
        cands['quartic'] = quart / quart.sum()

    # Flat top (uniform)
    cands['uniform'] = np.ones(n) / n

    # Random
    cands['random'] = rng.dirichlet(np.ones(n) * 0.1)

    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                cands['parent'] = pp / ps
        if len(parent) < n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
            ps = pp.sum()
            if ps > 1e-10:
                cands['parent_up'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
            ps = pp.sum()
            if ps > 1e-10:
                cands['parent_ds'] = pp / ps

    return cands


def mirror_descent_step(a, conv, eta, t_param):
    """One step of multiplicative weights / mirror descent."""
    n = len(a)
    max_conv = conv.max()
    if max_conv < 1e-15:
        return a

    # Softmax over convolution with temperature
    scaled = t_param * conv / max_conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        return a
    w /= w_sum

    # Gradient (same as before)
    # Use simple computation for moderate n
    thresh = w.max() * 0.01
    active = np.where(w > thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(w)])

    sg = np.zeros(n)
    for k_int in active:
        k = int(k_int)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        sg[idx] += w[k_int] * 2.0 * a[k - idx]

    # Center the gradient (invariant to adding constant)
    sg_centered = sg - sg.mean()

    # Multiplicative update
    log_update = -eta * sg_centered
    log_update -= log_update.max()
    a = a * np.exp(log_update)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def optimize_mirrordesc(a0, time_limit, eta_init=0.5, t_init=20.0, seed=42):
    """Mirror descent optimization."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    eta = eta_init
    t_param = t_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        a = mirror_descent_step(a, conv, eta, t_param)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 100:
            no_improve = 0
            eta *= 0.7
            t_param *= 1.5
            if eta < 0.05:
                # Reset with perturbation
                noise = rng.standard_normal(n) * 0.01
                a = project_simplex(best_a + noise)
                conv = fast_convolve(a)
                eta = eta_init
                t_param = t_init
                best_obj = 2.0 * n * conv.max()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search to minimize 2n*max(conv)/(sum)^2.
    Starts from analytical ramp solutions (achieving ~1.338) and refines
    with soft-max gradient descent and mirror descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent available: length={len(p)}, sum={p.sum():.2f}")
    except Exception:
        pass

    # Phase 1: Evaluate analytical candidates quickly
    print("=== Phase 1: Analytical Candidates ===")
    phase1_deadline = time.monotonic() + 120.0
    lengths = [256, 384, 512, 768, 1024, 1536, 2048]
    analytical_results = []

    for n in lengths:
        if time.monotonic() > phase1_deadline - 5:
            break
        cands = generate_candidates(n, parent, rng)
        for name, a in cands.items():
            conv = fast_convolve(a)
            objective = 2.0 * n * conv.max() / (a.sum() ** 2)
            analytical_results.append((objective, n, name, a.copy()))
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a.copy()

    analytical_results.sort(key=lambda x: x[0])
    for obj, n, name, a in analytical_results[:5]:
        print(f"  n={n}, {name}: obj={obj:.6f}")
    print(f"Phase 1 best: {best_score:.6f}")

    # Phase 2: Gradient optimization from best candidates
    print("\n=== Phase 2: Soft-Max Gradient Optimization ===")
    phase2_deadline = time.monotonic() + 900.0

    # Select top starting points
    top_starts = []
    seen_configs = set()
    for obj, n, name, a in analytical_results:
        key = (n, name)
        if key not in seen_configs:
            top_starts.append((n, name, a))
            seen_configs.add(key)
        if len(top_starts) >= 12:
            break

    # Always include the current best
    if best_sequence is not None and (len(best_sequence), 'best') not in seen_configs:
        top_starts.append((len(best_sequence), 'best', best_sequence.copy()))

    remaining = phase2_deadline - time.monotonic() - 30
    time_per_opt = max(20.0, remaining / max(1, len(top_starts) * 2))

    for idx, (n, name, a0) in enumerate(top_starts):
        if time.monotonic() > phase2_deadline - 30:
            break
        t_alloc = min(time_per_opt, phase2_deadline - time.monotonic() - 25)
        if t_alloc < 10:
            break

        # Try soft-max gradient
        t_sg = t_alloc * 0.6
        a_opt, obj_opt = optimize_softgrad(
            a0, time_limit=t_sg, lr_init=0.03, momentum=0.88,
            temp_init=0.005, temp_growth=1.008,
            perturb_every=1500, perturb_scale=0.004, batch=80, seed=seed+idx)

        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  SG n={n}, {name}: score={score:.6f}")

        # Try mirror descent
        t_md = min(t_alloc * 0.4, phase2_deadline - time.monotonic() - 20)
        if t_md > 8:
            a_opt2, obj_opt2 = optimize_mirrordesc(
                a_opt, time_limit=t_md, eta_init=0.4, t_init=30.0, seed=seed+100+idx)
            score2 = float(evaluate_sequence(a_opt2.tolist()))
            if np.isfinite(score2) and score2 < best_score:
                best_score = score2
                best_sequence = a_opt2.copy()
                print(f"  MD n={n}, {name}: score={score2:.6f}")

    print(f"Phase 2 best: {best_score:.6f}")

    # Phase 3: Deep optimization on the best
    print("\n=== Phase 3: Deep Optimization ===")
    if best_sequence is not None:
        phase3_deadline = time.monotonic() + 500.0
        n_best = len(best_sequence)

        configs_sg = [
            {'lr_init': 0.015, 'momentum': 0.92, 'temp_init': 0.002, 'temp_growth': 1.005, 'perturb_scale': 0.002},
            {'lr_init': 0.008, 'momentum': 0.95, 'temp_init': 0.001, 'temp_growth': 1.003, 'perturb_scale': 0.001},
            {'lr_init': 0.025, 'momentum': 0.88, 'temp_init': 0.008, 'temp_growth': 1.01, 'perturb_scale': 0.004},
            {'lr_init': 0.01, 'momentum': 0.93, 'temp_init': 0.004, 'temp_growth': 1.006, 'perturb_scale': 0.003},
        ]

        t_per = 500.0 / (len(configs_sg) + 3)

        for cfg_idx, cfg in enumerate(configs_sg):
            if time.monotonic() > phase3_deadline - 30:
                break
            t_alloc = min(t_per, phase3_deadline - time.monotonic() - 25)
            if t_alloc < 8:
                break

            a_opt, obj_opt = optimize_softgrad(
                best_sequence, time_limit=t_alloc, **cfg, batch=100, seed=seed+1000+cfg_idx)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Deep SG {cfg_idx}: score={score:.6f}")

        # Mirror descent polish
        if time.monotonic() < phase3_deadline - 20:
            t_md = min(100.0, phase3_deadline - time.monotonic() - 15)
            a_opt, obj_opt = optimize_mirrordesc(
                best_sequence, time_limit=t_md, eta_init=0.2, t_init=80.0, seed=seed+2000)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Deep MD: score={score:.6f}")

    # Phase 4: Try different lengths from the best shape
    print("\n=== Phase 4: Length Transfer ===")
    if best_sequence is not None:
        phase4_deadline = time.monotonic() + 200.0
        n_best = len(best_sequence)
        # Resample to other lengths and optimize
        for n_try in [2 * n_best, n_best + 128, max(256, n_best - 128)]:
            if time.monotonic() > phase4_deadline - 30:
                break
            if n_try <= 64 or n_try > 4096:
                continue
            if n_try == n_best:
                continue
            # Interpolate
            a_interp = np.interp(np.linspace(0, 1, n_try),
                                  np.linspace(0, 1, n_best), best_sequence)
            a_interp = np.maximum(a_interp, 0)
            a_interp /= a_interp.sum()

            t_alloc = min(50.0, phase4_deadline - time.monotonic() - 20)
            if t_alloc < 5:
                break
            a_opt, obj_opt = optimize_softgrad(
                a_interp, time_limit=t_alloc, lr_init=0.02, momentum=0.9,
                temp_init=0.003, temp_growth=1.006, perturb_scale=0.003, seed=seed+3000)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Transfer n={n_try}: score={score:.6f}")

    # Phase 5: Final random restarts
    print("\n=== Phase 5: Random Restarts ===")
    if best_sequence is not None:
        phase5_deadline = time.monotonic() + 100.0
        n_best = len(best_sequence)
        for attempt in range(5):
            if time.monotonic() > phase5_deadline - 20:
                break
            # Perturb the best and re-optimize
            scales = [0.005, 0.01, 0.02, 0.015, 0.008]
            noise = rng.standard_normal(n_best) * scales[attempt % len(scales)]
            a_pert = project_simplex(best_sequence + noise)
            t_alloc = min(20.0, phase5_deadline - time.monotonic() - 15)
            if t_alloc < 3:
                break
            a_opt, obj_opt = optimize_softgrad(
                a_pert, time_limit=t_alloc, lr_init=0.01, momentum=0.92,
                temp_init=0.002, temp_growth=1.004, perturb_scale=0.001,
                batch=100, seed=seed+4000+attempt)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Restart {attempt}: score={score:.6f}")

    # Final verification
    if best_sequence is None:
        # Fallback: use the ramp
        n = 256
        best_sequence = np.arange(1, n + 1, dtype=np.float64)
        best_sequence /= best_sequence.sum()
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Ensure all values are valid
    best_sequence = np.maximum(best_sequence, 0.0)
    best_sequence = np.minimum(best_sequence, 1000.0)
    best_sequence = best_sequence[np.isfinite(best_sequence)]

    if len(best_sequence) == 0:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        # Emergency fallback
        best_sequence = np.arange(1, 257, dtype=np.float64)
        best_sequence /= best_sequence.sum()
        final_score = float(evaluate_sequence(best_sequence.tolist()))

    # Check if our computed score matches
    print(f"\nFINAL: n={len(best_sequence)}, score={final_score:.6f}")
    return [float(x) for x in best_sequence]
