                a_try = normalize_to(a_try, S_target)
                s_try = compute_obj(a_try)
                if s_try < best_score - 1e-12:
                    a = a_try
                    best_a = a.copy()

def softmax_gradient_step(a, tau, lr, momentum, decay):
    """One step of soft-max gradient descent."""
    n = len(a)
    S_target = float(np.sqrt(2.0 * n))
    c = np.convolve(a, a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    
    # grad[i] = 2 * sum_j w[i+j] * a[j]
    a_rev = a[::-1]
    conv_res = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_res[n-1:n-1+n]
    
    # Project out mean
    grad = grad - grad.mean()
    
    momentum[:] = decay * momentum - lr * grad
    a_new = a + momentum
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize_to(a_new, S_target)
    return a_new, momentum

"""
Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Primary method: Iterated LP with golden-section blend optimization and adaptive M.
Secondary: Soft-max gradient descent, adaptive perturbation, coordinate descent polishing.
Starts from known best (height_sequence_1) and refines aggressively with restarts.

import numpy as np
import time
from scipy.optimize import linprog

_global_deadline = None

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * c.max() / (S * S))

def normalize_to(a, S_target):
    """Normalize a to have sum = S_target."""
    S = a.sum()
    if S < 1e-15:
        return a.copy()
    return a * (S_target / S)

def build_conv_matrix(n, f):
    """Build the convolution constraint matrix A where (A @ g)[m] = (f*g)[m]."""
    A = np.zeros((2 * n - 1, n))
    idx = np.arange(2 * n - 1)[:, None] - np.arange(n)[None, :]
    mask = (idx >= 0) & (idx < n)
    A[mask] = f[idx[mask]]
    return A

def solve_lp(f, M):
    """Solve: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    n = len(f)
    c_obj = -np.ones(n)
    A = build_conv_matrix(n, f)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'mip_rel_gap': 0})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def golden_blend(a, g_norm):
    """Find optimal t in [0,1] minimizing obj((1-t)*a + t*g_norm) via golden section."""
    def f_of_t(t):
        blend = (1.0 - t) * a + t * g_norm
        return compute_obj(blend)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = f_of_t(c_pt)
    fd = f_of_t(d_pt)

    for _ in range(45):
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = f_of_t(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = f_of_t(d_pt)

    t_best = (lo + hi) / 2.0
    return t_best, f_of_t(t_best)

        if g is None or g.sum() < 1e-10:
            # Retry with full M
            if m_factor < 1.0:
                g = solve_lp(a, M)
            if g is None or g.sum() < 1e-10:
                break

        g_norm = normalize_to(g, S_target)

        # Check if g is meaningfully different from a
        diff = np.linalg.norm(g_norm - a)
        if diff < 1e-8 * np.linalg.norm(a):
            stall_count += 5
        else:
            # Golden section for optimal blend
            t_best, score_at_t = golden_blend(a, g_norm)

            if score_at_t < best_score - 1e-13:
                a_new = (1.0 - t_best) * a + t_best * g_norm
                a_new = np.maximum(a_new, 0.0)
                a_new = normalize_to(a_new, S_target)
                actual_score = compute_obj(a_new)
                if actual_score < best_score:
                    best_a = a_new.copy()
                    best_score = actual_score
                    a = a_new
                    stall_count = 0
                else:
                    stall_count += 1
            else:
                # Try small blends
                improved = False
                for t_try in [0.08, 0.04, 0.02, 0.01]:
                    a_try = (1.0 - t_try) * a + t_try * g_norm
                    a_try = np.maximum(a_try, 0.0)
                    a_try = normalize_to(a_try, S_target)
                    s_try = compute_obj(a_try)
                    if s_try < best_score - 1e-13:
                        a = a_try
                        best_a = a.copy()
                        best_score = s_try
                        stall_count = 0
                        improved = True
                        break
                if not improved:
                    stall_count += 1

        # Handle stalls with perturbation
        if it % check_every == 0 and stall_count > 20:
            noise_scale = 0.02 * max(a.mean(), 1e-6)
            noise = rng_local.normal(0, noise_scale, n)
            a_pert = np.maximum(a + noise, 0.0)
            a_pert = normalize_to(a_pert, S_target)
            s_pert = compute_obj(a_pert)
            if s_pert < best_score:
                best_a = a_pert.copy()
                best_score = s_pert
            a = a_pert
            stall_count = 0

        # Log progress
        if it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {it}: score={best_score:.8f}, elapsed={elapsed:.1f}s")

    return best_a, best_score

def softmax_gradient_step(a, tau, lr, momentum, decay):
    """One step of soft-max gradient descent with momentum."""
    n = len(a)
    S_target = float(np.sqrt(2.0 * n))
    c = np.convolve(a, a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum

    # grad[i] = 2 * sum_j w[i+j] * a[j] via convolution
    a_rev = a[::-1]
    conv_res = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_res[n - 1:n - 1 + n]

    # Project out mean to preserve sum
    grad = grad - grad.mean()

    momentum[:] = decay * momentum - lr * grad
    a_new = a + momentum
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize_to(a_new, S_target)
    return a_new, momentum

def softmax_refine(a, time_limit):
    """Soft-max gradient refinement with adaptive tau and learning rate."""
    global _global_deadline
    n = len(a)
    S_target = float(np.sqrt(2.0 * n))
    a = normalize_to(np.asarray(a, dtype=np.float64).copy(), S_target)
    best_a = a.copy()
    best_score = compute_obj(a)
    momentum = np.zeros(n)
    t0 = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if _global_deadline is not None and time.monotonic() > _global_deadline - 30:
            break

        tau = min(3.0 + 0.02 * it, 60.0)
        lr = max(0.005, 0.15 * (0.9997 ** it))

        a, momentum = softmax_gradient_step(a, tau, lr, momentum, 0.9)

        if it % 50 == 0:
            s = compute_obj(a)
            if s < best_score - 1e-12:
                best_score = s
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
                if stall > 30:
                    momentum[:] = 0
                    stall = 0

    return best_a, best_score

def hardmax_gradient_step(a, lr, momentum, decay):
    """One step of hard-max gradient descent (at the peak lag)."""
    n = len(a)
    S_target = float(np.sqrt(2.0 * n))
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    grad -= grad.mean()
    momentum[:] = decay * momentum - lr * grad
    a_new = a + momentum
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize_to(a_new, S_target)
    return a_new, momentum

def hardmax_refine(a, time_limit):
    """Hard-max gradient refinement."""
    global _global_deadline
    n = len(a)

    a = normalize_to(np.asarray(a, dtype=np.float64).copy(), S_target)

def coordinate_polish(a, time_limit):
    """Coordinate descent polishing with adaptive candidates."""
    global _global_deadline
    n = len(a)
    S_target = float(np.sqrt(2.0 * n))
    a = normalize_to(np.asarray(a, dtype=np.float64).copy(), S_target)
    best_a = a.copy()
    best_score = compute_obj(a)
    t0 = time.monotonic()
    patience = 0
    rng_local = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        if _global_deadline is not None and time.monotonic() > _global_deadline - 10:
            break
        improved = False
        c = np.convolve(a, a)
        k_star = int(np.argmax(c))

        # Randomly shuffle order for diversity
        order = np.arange(n)
        if patience > 1:
            rng_local.shuffle(order)

        for i in order:
            if _global_deadline is not None and time.monotonic() > _global_deadline - 5:
                break
            orig = a[i]
            j = k_star - i
            candidates = [0.0, orig * 0.5, orig * 1.5]
            if 0 <= j < n and j != i:
                candidates.append(orig * 0.7)
                candidates.append(orig * 1.2)
                candidates.append((orig + a[j]) / 2.0)

            best_val = orig
            best_s = best_score
            for val in candidates:
                if val < 0:
                    continue
                a[i] = val
                a = normalize_to(a, S_target)
                s = compute_obj(a)
                if s < best_s - 1e-13:
                    best_s = s
                    best_val = val
            a[i] = best_val
            a = normalize_to(a, S_target)
            if best_val != orig:
                best_score = best_s
                best_a = a.copy()
                improved = True

        if not improved:
            patience += 1
            if patience >= 4:
                break
        else:
            patience = 0

    return best_a, best_score

def mixed_refine(a, time_limit):
    """Alternate between LP, hard-max gradient, and soft-max gradient."""
    global _global_deadline
    n = len(a)
    S_target = float(np.sqrt(2.0 * n))
    a = normalize_to(np.asarray(a, dtype=np.float64).copy(), S_target)
    best_a = a.copy()
    best_score = compute_obj(a)
    t0 = time.monotonic()
    phase_idx = 0
    rng_local = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        if _global_deadline is not None and time.monotonic() > _global_deadline - 30:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        if phase_idx == 0:
            # LP phase
            chunk = min(120, remaining)
            a_chunk, s_chunk = lp_refine(a, chunk, check_every=20)
            if s_chunk < best_score:
                best_score = s_chunk
                best_a = a_chunk.copy()
            a = a_chunk
            phase_idx = 1
        elif phase_idx == 1:
            # Hard-max gradient
            chunk = min(60, remaining)
            a_chunk, s_chunk = hardmax_refine(a, chunk)
            if s_chunk < best_score:
                best_score = s_chunk
                best_a = a_chunk.copy()
            a = a_chunk
            phase_idx = 2
        else:
            # Soft-max gradient
            chunk = min(60, remaining)
            a_chunk, s_chunk = softmax_refine(a, chunk)
            if s_chunk < best_score:
                best_score = s_chunk
                best_a = a_chunk.copy()
            a = a_chunk
            phase_idx = 0

        # Periodic perturbation
        if int(time.monotonic() - t0) % 90 < 180 and phase_idx == 0:
            noise = rng_local.normal(0, 0.005 * max(a.mean(), 1e-6), n)
            a_pert = np.maximum(a + noise, 0.0)
            a_pert = normalize_to(a_pert, S_target)
            if compute_obj(a_pert) < best_score:
                best_a = a_pert.copy()
                best_score = compute_obj(a_pert)
                a = a_pert

    return best_a, best_score

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Uses iterated LP with golden-section blending as the primary method,
    supplemented by gradient methods and coordinate descent polishing.
    Returns the best sequence found within the time budget.
    """
    global _global_deadline
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    _global_deadline = t_start + max(0.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    start_seq = None
    start_n = 384
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 16 and np.all(np.isfinite(gs)):
            start_seq = gs.copy()
            start_n = len(gs)
            s_init = compute_obj(start_seq)
            print(f"Loaded height_sequence_1: n={start_n}, score={s_init:.8f}")
            best_seq = start_seq.copy()
            best_score = s_init
    except NameError:
        print("height_sequence_1 not available, using structured start")

    if start_seq is None:
        # Create a structured starting point
        n = 384
        i = np.arange(n, dtype=np.float64)
        start_seq = 1.0 + 0.3 * np.cos(2 * np.pi * i / n) + 0.1 * np.cos(4 * np.pi * i / n)
        start_seq = np.maximum(start_seq, 0.01)
        s_init = compute_obj(start_seq)
        print(f"Structured start: n={n}, score={s_init:.8f}")
        best_seq = start_seq.copy()
        best_score = s_init

    n_best = len(best_seq)
    print(f"\nTarget: score < 1.5027. Current best: {best_score:.8f}")
    print(f"Time budget: {budget_s}s, effective deadline in { _global_deadline - t_start:.0f}s")

    # === Phase 1: Aggressive LP on the known best ===
    if time.monotonic() < _global_deadline - 400:
        print(f"\n{'='*60}")

        print(f"{'='*60}")
        t_alloc = min(400, _global_deadline - time.monotonic() - 300)
        a1, s1 = lp_refine(best_seq, t_alloc)
        if s1 < best_score:
            best_score = s1
            best_seq = a1.copy()
            print(f"  Phase 1: NEW BEST score={best_score:.8f}")
        else:
            print(f"  Phase 1: no improvement (best={best_score:.8f})")

    # === Phase 2: Mixed refinement (LP + gradients alternating) ===
    if best_seq is not None and time.monotonic() < _global_deadline - 200:
        print(f"\n{'='*60}")
        print(f"Phase 2: Mixed refinement (n={len(best_seq)})")
        print(f"{'='*60}")
        t_alloc = min(300, _global_deadline - time.monotonic() - 200)
        a2, s2 = mixed_refine(best_seq, t_alloc)
        if s2 < best_score:
            best_score = s2
            best_seq = a2.copy()
            print(f"  Phase 2: NEW BEST score={best_score:.8f}")
        else:
            print(f"  Phase 2: no improvement (best={best_score:.8f})")

    # === Phase 3: More LP from perturbed starts ===
    if best_seq is not None and time.monotonic() < _global_deadline - 150:
        print(f"\n{'='*60}")
        print(f"Phase 3: LP from perturbed starts")
        print(f"{'='*60}")
        n = len(best_seq)
        for trial in range(3):
            if time.monotonic() > _global_deadline - 100:
                break
            noise_scale = [0.01, 0.03, 0.05][trial]
            noise = rng.normal(0, noise_scale * best_seq.mean(), n)
            a_pert = np.maximum(best_seq + noise, 0.0)
            t_alloc = min(80, _global_deadline - time.monotonic() - 60)
            if t_alloc < 20:
                break
            a3, s3 = lp_refine(a_pert, t_alloc)
            if s3 < best_score:
                best_score = s3
                best_seq = a3.copy()
                print(f"  Perturb {trial} (scale={noise_scale}): NEW BEST score={best_score:.8f}")

    # === Phase 4: Try nearby lengths ===
    if best_seq is not None and time.monotonic() < _global_deadline - 100:
        print(f"\n{'='*60}")
        print(f"Phase 4: Try nearby lengths")
        print(f"{'='*60}")
        base_n = len(best_seq)
        nearby_lengths = [base_n - 8, base_n - 4, base_n + 4, base_n + 8, base_n + 12]
        for trial_n in nearby_lengths:
            if trial_n < 32:
                continue
            if time.monotonic() > _global_deadline - 50:
                break
            if trial_n > base_n:
                a0 = np.zeros(trial_n)
                a0[:base_n] = best_seq
                # Smooth the junction
                pad = trial_n - base_n
                a0[base_n:] = best_seq[-1] * np.exp(-np.arange(pad) / (pad * 0.5))
            else:
                a0 = best_seq[:trial_n].copy()
            t_alloc = min(50, _global_deadline - time.monotonic() - 30)
            if t_alloc < 15:
                break
            a4, s4 = lp_refine(a0, t_alloc)
            if s4 < best_score:
                best_score = s4
                best_seq = a4.copy()
                print(f"  n={trial_n}: NEW BEST score={best_score:.8f}")
            else:
                print(f"  n={trial_n}: score={s4:.8f}")

    # === Phase 5: Final LP + gradient combo ===
    if best_seq is not None and time.monotonic() < _global_deadline - 40:
        print(f"\n{'='*60}")
        print(f"Phase 5: Final refinement")
        print(f"{'='*60}")
        remaining = _global_deadline - time.monotonic() - 25
        t_lp = remaining * 0.5
        a5, s5 = lp_refine(best_seq, max(10, t_lp))
        if s5 < best_score:
            best_score = s5
            best_seq = a5.copy()
            print(f"  Final LP: NEW BEST score={best_score:.8f}")

        if time.monotonic() < _global_deadline - 20:
            t_sm = max(5, _global_deadline - time.monotonic() - 15)
            a5b, s5b = softmax_refine(best_seq, t_sm)
            if s5b < best_score:
                best_score = s5b
                best_seq = a5b.copy()
                print(f"  Final softmax: NEW BEST score={best_score:.8f}")

    # === Phase 6: Coordinate descent polish ===
    if best_seq is not None and time.monotonic() < _global_deadline - 10:
        print(f"\n{'='*60}")
        print(f"Phase 6: Coordinate descent polish")
        print(f"{'='*60}")
        n = len(best_seq)
        if n <= 512:
            t_cd = min(45, _global_deadline - time.monotonic() - 5)
            if t_cd > 5:
                a6, s6 = coordinate_polish(best_seq, t_cd)
                if s6 < best_score:
                    best_score = s6
                    best_seq = a6.copy()
                    print(f"  Coordinate polish: NEW BEST score={best_score:.8f}")
                else:
                    print(f"  Coordinate polish: no improvement (best={best_score:.8f})")

    # === Final validation ===
    if best_seq is not None:
        # Ensure all values are valid
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = np.minimum(best_seq, 1000.0)
        final_score = compute_obj(best_seq)
        print(f"\n{'='*60}")
        print(f"FINAL RESULT: n={len(best_seq)}, score={final_score:.8f}")
        print(f"{'='*60}")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence confirms: {eval_score:.8f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in best_seq]

    # Fallback
    print("WARNING: Falling back to uniform sequence.")
    return [1.0] * 256
