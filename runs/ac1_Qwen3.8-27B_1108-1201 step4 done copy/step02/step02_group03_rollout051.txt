Uses multi-phase approach: soft-max gradient descent, hard-max gradient with

import numpy as np

def compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def normalize(a, target=None):
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)

def compute_grad_hard(a):
    """Gradient of max_k conv(a,a)[k] w.r.t. a, using the argmax lag."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    # dc[k_star]/da[i] = 2*a[k_star-i] if 0 <= k_star-i < n
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad, k_star

def compute_grad_soft(a, tau):
    """Gradient of soft-max (log-sum-exp) approximation of max_k c[k]."""
    n = len(a)
    c = np.convolve(a, a)
    c_max = c.max()

    w = np.exp(tau * (c - c_max))
    w_sum = w.sum()

    conv_result = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_result[n-1:n-1+n]
    return grad, w

    n = len(a)

    a = normalize(a.copy(), S_t)
    best = a.copy()
    best_score = compute_obj(a)

    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline:

        progress = min(1.0, it / 10000.0)
        tau = tau_start + (tau_end - tau_start) * progress

            grad, _ = compute_grad_hard(a)
        else:
            grad, _ = compute_grad_soft(a, tau)
        
        # Center the gradient (to maintain sum)
        grad -= grad.mean()

                best = a.copy()
                no_improve = 0
            else:

Multi-phase approach: soft-max gradient descent with adaptive annealing,

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=None):
    """Normalize to target sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def grad_hard(a):
    """Gradient of max_k c[k] at the argmax lag."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad, k_star


def grad_soft(a, tau):
    """Gradient of soft-max approximation of max c[k]."""
    n = len(a)
    c = np.convolve(a, a)
    c_max = c.max()
    w = np.exp(tau * (c - c_max))
    w_sum = w.sum()
    if w_sum < 1e-300:

    a_rev = a[::-1]

    grad = 2.0 * conv_result[n - 1:n - 1 + n]
    return grad, w

    step = max(1, n // 384)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = np.convolve(a, a)
        k_star = int(np.argmax(c))
        for i in range(0, n, step):
            if time.monotonic() > deadline:
                break
            orig = a[i]
            cands = [0.0, orig * 0.4, orig * 0.6, orig * 1.4, orig * 1.6]
            j = k_star - i
            if 0 <= j < n and j != i:
                cands.append(orig * 0.2)
            best_val = orig
            best_s = best_score
            for val in cands:
                a[i] = val
                a = normalize(a, S_t)
                s = compute_obj(a)
                if s < best_s - 1e-10:
                    best_s = s
                    best_val = val
                a[i] = orig
            if best_val != orig:
                a[i] = best_val
                a = normalize(a, S_t)
                best_score = best_s

    return a, best_score


def optimize_from_start(a0, deadline):
    """Full optimization pipeline from one starting point."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n) * 0.5
    a = normalize(a)
    best = a.copy()
    best_score = compute_obj(a)

    rem = deadline - time.monotonic()
    if rem < 5:
        return best, best_score

    # Phase 1: Soft gradient
    t1 = time.monotonic() + rem * 0.30
    a1, s1 = phase_gradient(a, t1, tau_init=0.5, tau_final=50.0,
                            lr_init=0.6, lr_final=0.03, patience=12)
    if s1 < best_score:
        best_score = s1
        best = a1

    # Phase 2: Hard gradient
    rem2 = deadline - time.monotonic()
    if rem2 > 10:
        t2 = time.monotonic() + rem2 * 0.30
        a2, s2 = phase_gradient(best, t2, tau_init=0.0, tau_final=0.0,
                                lr_init=0.4, lr_final=0.02, patience=10,
                                use_soft=False)
        if s2 < best_score:
            best_score = s2
            best = a2

    # Phase 3: Peak cancellation
    rem3 = deadline - time.monotonic()
    if rem3 > 10:
        t3 = time.monotonic() + rem3 * 0.20
        a3, s3 = peak_cancellation(best, t3, max_iters=150)
        if s3 < best_score:
            best_score = s3
            best = a3

    # Phase 4: LP refinement
    rem4 = deadline - time.monotonic()
    if rem4 > 10 and n <= 1024:
        t4 = time.monotonic() + rem4 * 0.35
        a4, s4 = lp_phase(best, t4, max_iters=60)
        if s4 < best_score:
            best_score = s4
            best = a4

    # Phase 5: Coordinate polish
    rem5 = deadline - time.monotonic()
    if rem5 > 5 and n <= 768:
        a5, s5 = coord_polish(best, time.monotonic() + rem5 * 0.8, max_rounds=3)
        if s5 < best_score:
            best_score = s5
            best = a5

    return best, best_score


def make_init(n, rng, method='random'):
    """Generate diverse initializations."""
    i = np.arange(n, dtype=np.float64)
    if method == 'random':
        return rng.random(n) * 0.5 + 0.25
    elif method == 'chirp':
        return np.abs(np.cos(2 * np.pi * i * i / (4.0 * n))) + 0.1
    elif method == 'chirp2':
        return np.abs(np.cos(2 * np.pi * i * i / (3.0 * n) + 0.5)) + 0.1
    elif method == 'cos':
        return np.abs(np.cos(2 * np.pi * 3 * i / n)) + 0.1
    elif method == 'cos7':
        return np.abs(np.cos(2 * np.pi * 7 * i / n)) + 0.1
    elif method == 'gaussian':
        return np.exp(-0.5 * ((i - n * 0.5) / (n * 0.25)) ** 2) + 0.05
    elif method == 'sinc':
        x = (i - n * 0.5) / (n * 0.08)
        return np.abs(np.sinc(x)) + 0.05
    elif method == 'sparse':
        a = np.zeros(n)
        mask = rng.random(n) < 0.25
        a[mask] = rng.random(int(mask.sum())) * 0.5 + 0.2
        return a
    elif method == 'ramp':
        return 0.2 + 0.4 * i / n
    elif method == 'two_peak':
        return (np.exp(-0.5 * ((i - n * 0.2) / (n * 0.1)) ** 2) +
                np.exp(-0.5 * ((i - n * 0.8) / (n * 0.1)) ** 2)) + 0.05
    elif method == 'flat_noise':
        return np.maximum(1.0 + rng.normal(0, 0.4, n), 0.05)
    elif method == 'alternating':
        base = 0.5 + 0.2 * np.cos(2 * np.pi * i / 8)
        return np.maximum(base + rng.normal(0, 0.1, n), 0.05)
    return rng.random(n) * 0.5 + 0.25


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: multi-phase gradient/LP/coord optimization from diverse
    starts at multiple lengths with adaptive time allocation.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 12.0)

    best_seq = None
    best_score = float('inf')

    # Load previous best if available
    prev_seq = None
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 16 and np.all(np.isfinite(gs)):
            prev_seq = gs.copy()
            prev_score = compute_obj(prev_seq)
            print(f"Loaded previous: n={len(gs)}, score={prev_score:.6f}")
            best_seq = prev_seq.copy()
            best_score = prev_score
    except NameError:
        pass

    # Define trials: (method, n)
    lengths = [256, 384, 512, 640]
    methods_all = ['random', 'chirp', 'chirp2', 'cos', 'cos7',
                   'gaussian', 'sinc', 'sparse', 'ramp', 'two_peak',
                   'flat_noise', 'alternating']

    # Phase A: Screen combinations
    # Give short time to each to identify promising (length, method) pairs
    print(f"\n=== Phase A: Screening ===")
    screen_trials = []
    if prev_seq is not None:
        n_prev = len(prev_seq)
        screen_trials.append(('prev', n_prev))
    for n in lengths:
        for m in ['random', 'chirp', 'cos', 'gaussian', 'sinc']:
            screen_trials.append((m, n))

    total_screen_time = (deadline - t_start) * 0.45
    time_per = max(8.0, total_screen_time / len(screen_trials))

    screen_results = []
    for idx, (method, n) in enumerate(screen_trials):
        if time.monotonic() > deadline - total_screen_time * 0.3:
            break
        if method == 'prev':
            a0 = prev_seq.copy() if len(prev_seq) == n else prev_seq[:n].copy()
            mlabel = 'prev'
        else:
            a0 = make_init(n, rng, method)
            mlabel = method

        t_alloc = time_per
        if time.monotonic() + t_alloc > deadline - total_screen_time * 0.3:
            t_alloc = deadline - time.monotonic() - total_screen_time * 0.3
            if t_alloc < 8:
                break

        seq, score = optimize_from_start(a0, time.monotonic() + t_alloc)
        screen_results.append((score, n, mlabel, seq))
        tag = " *" if score < best_score else ""
        print(f"  [{idx:2d}] n={n:4d} {mlabel:12s} -> {score:.6f}{tag}")
        if score < best_score:
            best_score = score
            best_seq = seq.copy()

    # Phase B: Focused optimization at best length
    print(f"\n=== Phase B: Focused (n={len(best_seq) if best_seq is not None else 'N/A'}) ===")
    if best_seq is not None:
        best_n = len(best_seq)
        rem = deadline - time.monotonic() - 20
        n_focused = min(len(methods_all), max(3, int(rem / 55)))

        for idx in range(n_focused):
            if time.monotonic() > deadline - 20:
                break
            method = methods_all[idx % len(methods_all)]
            a0 = make_init(best_n, rng, method)
            t_alloc = min(100.0, rem / max(1, n_focused - idx))
            t_alloc = max(20.0, t_alloc)
            if time.monotonic() + t_alloc > deadline - 20:
                t_alloc = deadline - time.monotonic() - 20
                if t_alloc < 10:
                    break
            seq, score = optimize_from_start(a0, time.monotonic() + t_alloc)
            tag = " *" if score < best_score else ""
            print(f"  [B-{idx}] n={best_n} {method:12s} -> {score:.6f}{tag}")
            if score < best_score:
                best_score = score
                best_seq = seq.copy()

        # Perturbations of best
        for trial in range(4):
            if time.monotonic() > deadline - 20:
                break
            sigma = 0.08 + 0.03 * trial
            pert = rng.normal(0, sigma, best_n)
            a0 = np.maximum(best_seq + pert, 0.0)
            t_alloc = min(80.0, deadline - time.monotonic() - 20)
            if t_alloc < 10:
                break
            seq, score = optimize_from_start(a0, time.monotonic() + t_alloc)
            tag = " *" if score < best_score else ""
            print(f"  [pert-{trial}] sigma={sigma:.2f} -> {score:.6f}{tag}")
            if score < best_score:
                best_score = score
                best_seq = seq.copy()

    # Phase C: Final intensive polish
    print(f"\n=== Phase C: Final Polish ===")
    if best_seq is not None and time.monotonic() < deadline - 15:
        t_alloc = deadline - time.monotonic() - 8
        if t_alloc > 30:
            seq, score = optimize_from_start(best_seq, time.monotonic() + t_alloc)
            tag = " *" if score < best_score else ""
            print(f"  Final: {score:.6f}{tag}")
            if score < best_score:
                best_score = score
                best_seq = seq.copy()

    # Validation and return
    if best_seq is not None:
        best_seq = np.clip(best_seq, 0.0, 1000.0)
        final = compute_obj(best_seq)
        print(f"\n=== Result: n={len(best_seq)}, score={final:.6f} ===")
        try:
            ev = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence: {ev:.6f}")
        except Exception as e:
            print(f"Eval error: {e}")
        # Make sure sum is above threshold
        if best_seq.sum() < 0.01:
            best_seq = best_seq * 100.0
        return [float(x) for x in best_seq]

    # Fallback
    print("Fallback: uniform sequence")
    return [1.0] * 384
