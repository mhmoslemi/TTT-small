"""
Optimizes non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses FISTA gradient descent with soft-max smoothing, first-order LP-based local
search with golden-section step size, and multi-start with diverse initializations.
Key innovation: the LP directly approximates the peak autocorrelation minimization
via first-order Taylor expansion, giving a principled descent direction.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=None):
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = float(a.sum())
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def build_conv_matrix(f):
    """Build matrix A where A[k,j] = f[k-j] if 0<=k-j<n, else 0.
    Then conv(f,g)[k] = A[k] @ g."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def soft_max_gradient(a, tau):
    """Compute gradient of soft-max of autocorrelation."""
    n = len(a)
    c = np.convolve(a, a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum
    # grad[i] = 2 * sum_j w[i+j] * a[j]
    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_result[n - 1:n - 1 + n]
    return grad


def project_sum_nonneg(v, S):
    """Project v onto {a: a >= 0, sum(a) = S} via water-filling."""
    n = len(v)
    # Bisection on lambda: find lambda s.t. sum(max(v - lambda, 0)) = S
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.maximum(v - mid, 0.0).sum())
        if s > S:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.maximum(v - lam, 0.0)


def fista_setter(a, grad, y, t, step, S):
    """One FISTA iteration. Returns new x, new y, new t."""
    n = len(a)
    # Gradient step
    grad_proj = grad - grad.mean()  # project out mean direction
    x_new = project_sum_nonneg(y - step * grad_proj, S)
    # FISTA update
    t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
    momentum = (t - 1.0) / (t_new + 1.0)
    y_new = x_new + momentum * (x_new - y)
    return x_new, y_new, t_new


def solve_lp_first_order(f, S):
    """
    Solve: min M s.t. 2*conv(f,g)[k] - C_f[k] <= M, sum(g) = S, g >= 0, M >= 0.
    This is the first-order Taylor approximation of min max(conv(g,g)).
    Returns (g_opt, M_opt) or (None, None) on failure.
    """
    n = len(f)
    C_f = np.convolve(f, f)
    
    # Variables: [g_0, ..., g_{n-1}, M] -> n+1 variables
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize M
    
    # Inequality: 2*conv(f,g)[k] - C_f[k] - M <= 0 for all k
    A_mat = build_conv_matrix(f)
    A_ub = np.hstack([2.0 * A_mat, -np.ones((2 * n - 1, 1))])
    b_ub = C_f.copy()
    
    # Equality: sum(g) = S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    # Bounds
    bounds = [(0.0, None)] * n + [(0.0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.status == 0:
            g_opt = np.maximum(res.x[:n], 0.0)
            M_opt = res.x[n]
            return g_opt, M_opt
    except Exception:
        pass
    return None, None


def golden_section_peak(f, g, S, n_iter=40):
    """Find t in [0,1] minimizing max(conv((1-t)f+tg, (1-t)f+tg)).
    Precomputes convolutions for speed."""
    C_f = np.convolve(f, f)
    C_fg = np.convolve(f, g)
    C_g = np.convolve(g, g)
    
    def peak_val(t):
        val = (1.0 - t) ** 2 * C_f + 2.0 * (1.0 - t) * t * C_fg + t ** 2 * C_g
        return float(val.max())
    
    a_lo, a_hi = 0.0, 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a1 = a_hi - gr * (a_hi - a_lo)
    a2 = a_lo + gr * (a_hi - a_lo)
    f1 = peak_val(a1)
    f2 = peak_val(a2)
    for _ in range(n_iter):
        if f1 < f2:
            a_hi = a2
            a2 = a1
            f2 = f1
            a1 = a_hi - gr * (a_hi - a_lo)
            f1 = peak_val(a1)
        else:
            a_lo = a1
            a1 = a2
            f1 = f2
            a2 = a_lo + gr * (a_hi - a_lo)
            f2 = peak_val(a2)
    t_opt = 0.5 * (a_lo + a_hi)
    return t_opt, peak_val(t_opt)


def lp_local_search(f, S, deadline, max_iters=500):
    """Iterative LP-based local search with golden section step size."""
    n = len(f)
    best = f.copy()
    best_peak = float(np.convolve(best, best).max())
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        if no_improve > 15:
            break
        
        g_opt, M_opt = solve_lp_first_order(f, S)
        if g_opt is None:
            break
        
        # Ensure g_opt has the right sum
        g_sum = g_opt.sum()
        if g_sum < 1e-10:
            break
        g_opt = g_opt * (S / g_sum)
        
        # Golden section line search
        t_opt, new_peak = golden_section_peak(f, g_opt, S)
        
        if new_peak < best_peak - 1e-10:
            f_new = (1.0 - t_opt) * f + t_opt * g_opt
            f_new = np.maximum(f_new, 0.0)
            f_new = normalize(f_new, S)
            f = f_new
            actual_peak = float(np.convolve(f, f).max())
            if actual_peak < best_peak - 1e-12:
                best = f.copy()
                best_peak = actual_peak
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if it % 50 == 0:
            print(f"      LP iter {it}: peak={best_peak:.8f}, no_improve={no_improve}")
    
    return best, best_peak

    """FISTA gradient descent with soft-max, increasing tau."""

    t = 1.0

        if gs.size >= 8 and np.all(np.isfinite(gs)) and np.all(gs >= 0):
            existing_best = gs
            print(f"Using existing best: n={len(gs)}, score={compute_obj(gs):.6f}")
    except NameError:
        pass
    
    # Primary length: use the length of existing best or 384
    if existing_best is not None:
        primary_n = len(existing_best)
    else:
        primary_n = 384
    
    print(f"Starting optimization, budget={budget_s}s, primary_n={primary_n}")
    
    # Generate starts for primary length
    S = np.sqrt(2.0 * primary_n)
    starts = generate_starts(rng, primary_n, existing_best)
    
    # Sort: put existing best first, then perturbed, then others
    starts.sort(key=lambda x: -float(np.std(x)))
    
    # Time allocation
    n_starts = len(starts)
    time_per_start = (deadline - t_start) / max(1, n_starts) * 0.6
    time_per_start = max(30.0, min(300.0, time_per_start))
    
    for idx, a0 in enumerate(starts):
        if time.monotonic() > deadline - 60.0:
            break
        
        remaining = deadline - time.monotonic() - 30.0
        t_alloc = min(time_per_start, remaining * 0.4)
        t_alloc = max(20.0, t_alloc)
        
        if time.monotonic() + t_alloc > deadline - 30:
            t_alloc = deadline - time.monotonic() - 30
            if t_alloc < 15:
                break
        
        print(f"\n  Start {idx}/{n_starts} (n={len(a0)}), alloc={t_alloc:.0f}s")
        start_time = time.monotonic()
        
        seq, score = optimize_single(a0, S, start_time + t_alloc)
        
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  [start {idx}] New best: score={best_score:.6f}")
        else:
            print(f"  [start {idx}] score={score:.6f} (not better)")
    
    # Try other lengths if time permits
    other_lengths = [256, 512, 320, 448]
    other_lengths = [l for l in other_lengths if l != primary_n]
    
    if best_seq is not None and time.monotonic() < deadline - 300:
        for n_try in other_lengths[:2]:
            if time.monotonic() > deadline - 120:
                break
            S_try = np.sqrt(2.0 * n_try)
            a0 = rng.random(n_try) * 0.5 + 0.25
            # Also try truncating/padding the best
            if best_seq is not None:
                if n_try < len(best_seq):
                    a0 = best_seq[:n_try].copy()
                else:
                    a0 = np.zeros(n_try)
                    a0[:len(best_seq)] = best_seq
                    a0[len(best_seq):] = best_seq[-1]
            
            print(f"\n  Trying n={n_try} from existing best")
            t_alloc = min(200.0, deadline - time.monotonic() - 30)
            seq, score = optimize_single(a0, S_try, time.monotonic() + t_alloc)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  [n={n_try}] New best: score={best_score:.6f}")
    
    # Final intensive polish on the best
    if best_seq is not None and time.monotonic() < deadline - 100:
        print(f"\nFinal intensive polish (current best={best_score:.6f})")
        S_final = np.sqrt(2.0 * len(best_seq))
        
        # Aggressive LP search
        a_final = best_seq.copy()
        lp_deadline = min(deadline - 10, time.monotonic() + (deadline - time.monotonic()) * 0.6)
        a_final, _ = lp_local_search(a_final, S_final, lp_deadline, max_iters=500)
        s_final = compute_obj(a_final)
        if s_final < best_score:
            best_score = s_final
            best_seq = a_final.copy()
            print(f"  After final LP: score={best_score:.6f}")
        
        # Quick gradient
        if time.monotonic() < deadline - 20:
            a_g, s_g = gradient_phase(best_seq, S_final, deadline - 10, max_iters=5000)
            if s_g < best_score:
                best_score = s_g
                best_seq = a_g.copy()
                print(f"  After final grad: score={best_score:.6f}")
        
        # Coordinate polish
        if len(best_seq) <= 512 and time.monotonic() < deadline - 10:
            a_c, s_c = coordinate_polish(best_seq, S_final, deadline - 3, max_rounds=5)
            if s_c < best_score:
                best_score = s_c
                best_seq = a_c.copy()
                print(f"  After final coord: score={best_score:.6f}")
    
    # Final validation
    if best_seq is not None:
        final_score = compute_obj(best_seq)
        print(f"\nFinal result: n={len(best_seq)}, score={final_score:.6f}")
        # Ensure all values are valid
        best_seq = np.clip(best_seq, 0.0, 1000.0)
        final_score = compute_obj(best_seq)
        print(f"After clip: score={final_score:.6f}")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence: {eval_score:.6f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in best_seq]
    
    # Fallback
    print("Falling back to uniform sequence.")
    return [1.0] * 384
