"""
Log-sum-exp smoothed optimization for minimizing peak autocorrelation
of non-negative step functions. Uses adaptive temperature annealing with
L-BFGS-B, multiple starting points, and peak-targeting local search.
"""

import numpy as np
from scipy.optimize import minimize
import time


def local_score(a):
    """Fast local computation of the score."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def smooth_objective_and_grad(a, T):
    """
    Compute the log-sum-exp smoothed score and its gradient.
    Score = 2n * T * logsumexp(c/T) / s^2
    where c = convolve(a, a) and s = sum(a).
    """
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf, np.zeros(n)
    
    c = np.convolve(a, a)  # length 2n-1
    
    # Numerically stable log-sum-exp
    max_c = np.max(c)
    exp_c = np.exp((c - max_c) / T)
    sum_exp = np.sum(exp_c)
    lse = max_c + T * np.log(sum_exp)
    
    score = 2.0 * n * lse / (s * s)
    
    # Gradient
    # d lse / da[i] = T * sum_k exp_c[k] / sum_exp * dc[k]/da[i]
    # = T * sum_k w[k] * 2*a[k-i] * 1[0 <= k-i < n]
    # where w[k] = exp_c[k] / sum_exp
    
    w = exp_c / sum_exp
    
    # Compute h[i] = sum_k w[k] * a[k-i] * 1[0 <= k-i < n]
    # = sum_{j=0}^{n-1} w[i+j] * a[j]
    # = (w * a_reversed)[n-1+i]
    
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    size = 1
    while size < conv_len:
        size <<= 1
    W = np.fft.rfft(w, size)
    AR = np.fft.rfft(a_rev, size)
    conv_result = np.fft.irfft(W * AR, size)[:conv_len]
    h = conv_result[n-1:n-1+n]
    
    # d(score)/da[i] = 2n/s^2 * (2T*h[i] - 2*lse/s)
    # = 4n*T*h[i]/s^2 - 4n*lse/s^3
    grad = (4.0 * n * T * h) / (s * s) - (4.0 * n * lse) / (s ** 3)
    
    return score, grad


def optimize_smoothed(a0, T_schedule, max_iter=500, time_limit=300, seed=42):
    """
    Optimize using log-sum-exp smoothing with temperature annealing.
    T_schedule: list of temperatures to try (decreasing).
    """
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    
    for T in T_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        
        def obj_and_grad(x):
            x = np.maximum(x, 0.0)
            val, g = smooth_objective_and_grad(x, T)
            # The gradient is with respect to the unconstrained variable.
            # For L-BFGS-B, we just need the gradient of the objective.
            # But we need to be careful: if x[i] = 0, the true gradient
            # might point into negative territory, which L-BFGS-B handles
            # via the box constraint.
            return val, g
        
        result = minimize(
            obj_and_grad, a, jac=True,
            method='L-BFGS-B',
            bounds=[(0.0, 1000.0)] * n,
            options={'maxiter': max_iter, 'gtol': 1e-8, 'ftol': 1e-10}
        )
        
        a_new = np.clip(result.x, 0.0, 1000.0)
        score_new = local_score(a_new)
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
        
        # Update current iterate (even if not best, for continuation)
        if result.success or score_new < local_score(a):
            a = a_new
    
    return best_a, best_score


def peak_targeting_search(a0, time_limit=60, seed=42):
    """
    Local search that targets specific peak positions for reduction.
    """
    rng = np.random.default_rng(seed)
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        c = np.convolve(a, a)
        s = np.sum(a)
        current_score = 2.0 * n * np.max(c) / (s * s)
        
        # Find top peaks
        peak_indices = np.argsort(c)[::-1][:5]
        
        # For each peak, try to reduce it
        for k in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Elements contributing to c[k]: a[i] * a[k-i] for valid i
            # Try to reduce the largest contributors
            contribs = np.zeros(n)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            
            # Find top contributors
            top_k = max(3, n // 20)
            top_idx = np.argsort(contribs)[-top_k:]
            
            # Try decreasing top contributors and increasing non-contributors
            for attempt in range(5):
                candidate = a.copy()
                # Decrease top contributors
                decrease_factor = rng.uniform(0.7, 0.95)
                candidate[top_idx] *= decrease_factor
                
                # Increase some non-contributors (those with small contrib)
                non_top = np.setdiff1d(np.arange(n), top_idx)
                if len(non_top) > 0:
                    # Pick elements that don't contribute much to this peak
                    low_contrib = non_top[np.argsort(contribs[non_top])[:max(3, len(non_top)//10)]]
                    increase_factor = rng.uniform(1.05, 1.3)
                    candidate[low_contrib] *= increase_factor
                
                candidate = np.clip(candidate, 0.0, 1000.0)
                new_score = local_score(candidate)
                
                if new_score < current_score:
                    a = candidate
                    current_score = new_score
                    if current_score < best_score:
                        best_score = current_score
                        best_a = a.copy()
                    break
    
    return best_a, best_score


def make_symmetric_tent(n, gamma, rng):
    """Create a tent-like initialization with parameter gamma controlling flatness."""
    t = np.arange(n)
    center = (n - 1) / 2.0
    # Tent shape: 1 at center, decreasing linearly
    a = 1.0 - gamma * np.abs(t - center) / center
    a = np.clip(a, 0.01, None)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_cosine_init(n, rng, harmonics=5):
    """Create a cosine-based initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(1, harmonics + 1):
        amp = 0.3 / (k + 1)
        freq = rng.uniform(0.5, 3.0) * k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def resample_to_n(seq, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)

"""
Log-sum-exp smoothed optimization for minimizing peak autocorrelation of
non-negative step functions. Uses adaptive temperature annealing with
L-BFGS-B, mirror descent, multiple starting points, and peak-targeting
local search to beat the LP-based state of the art.
"""

import numpy as np
from scipy.optimize import minimize
import time


def local_score(a):
    """Fast local computation of the score."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def smooth_obj_grad(a, T):
    """
    Compute log-sum-exp smoothed score and gradient.
    Score_T = 2n * T * logsumexp(c/T) / s^2
    """
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return 1e10, np.zeros(n)

    c = np.convolve(a, a)  # length 2n-1

    # Numerically stable log-sum-exp
    max_c = np.max(c)
    exp_c = np.exp((c - max_c) / T)
    sum_exp = np.sum(exp_c)
    L = max_c + T * np.log(sum_exp)  # This is T*logsumexp(c/T)

    score = 2.0 * n * L / (s * s)

    # w[k] = softmax(c/T)
    w = exp_c / sum_exp

    # h[i] = sum_{j=0}^{n-1} w[i+j] * a[j]
    # = (w * a_rev)[n-1+i] where a_rev = a[::-1]
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    size = 1
    while size < conv_len:
        size <<= 1
    W = np.fft.rfft(w, size)
    AR = np.fft.rfft(a_rev, size)
    conv_result = np.fft.irfft(W * AR, size)[:conv_len]
    h = conv_result[n - 1:n - 1 + n]

    # d(Score_T)/da[i] = 2n/s^2 * (2*h[i] - 2*L/s)
    grad = 4.0 * n * h / (s * s) - 4.0 * n * L / (s ** 3)

    return score, grad


def verify_gradient(a, T, idx=None):
    """Verify gradient against finite differences (for debugging)."""
    score, grad = smooth_obj_grad(a, T)
    eps = 1e-6
    n = len(a)
    if idx is None:
        idx = np.random.randint(0, n, 5)
    else:
        idx = [idx]
    max_err = 0.0
    for i in idx:
        a_plus = a.copy()
        a_plus[i] += eps
        a_minus = a.copy()
        a_minus[i] = max(0, a_minus[i] - eps)
        s_plus, _ = smooth_obj_grad(a_plus, T)
        s_minus, _ = smooth_obj_grad(a_minus, T)
        fd = (s_plus - s_minus) / (a_plus[i] - a_minus[i])
        err = abs(fd - grad[i])
        max_err = max(max_err, err)
    return max_err


def optimize_smoothed_lbfgs(a0, T_schedule, max_iter=500, time_limit=200):
    """Optimize using L-BFGS-B with temperature annealing."""
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()

    for T in T_schedule:
        if time.monotonic() - t0 > time_limit:
            break

        def obj_and_grad(x):
            x_safe = np.maximum(x, 1e-15)
            val, g = smooth_obj_grad(x_safe, T)
            # For coordinates at zero, nudge gradient to non-negative
            g = np.where(x <= 0, np.maximum(g, 0), g)
            return val, g

        result = minimize(
            obj_and_grad, a, jac=True,
            method='L-BFGS-B',
            bounds=[(0.0, 1000.0)] * n,
            options={'maxiter': max_iter, 'gtol': 1e-9, 'ftol': 1e-12}
        )

        a_new = np.clip(result.x, 0.0, 1000.0)
        score_new = local_score(a_new)

        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()

        # Continue from new point if it improved
        if score_new < local_score(a):
            a = a_new

    return best_a, best_score


def mirror_descent(a0, T, eta0=0.1, n_iter=500, time_limit=60):
    """Mirror descent (exponential weights) on the simplex."""
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.clip(a, 1e-10, None)
    a = a / a.sum()  # normalize to simplex

    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    eta = eta0
    stall = 0

    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute gradient of L (the T*logsumexp part) w.r.t. a
        c = np.convolve(a, a)
        max_c = np.max(c)
        exp_c = np.exp((c - max_c) / T)
        w = exp_c / np.sum(exp_c)

        # h[i] = sum_j w[i+j] * a[j]
        a_rev = a[::-1]
        conv_len = len(w) + len(a) - 1
        size = 1
        while size < conv_len:
            size <<= 1
        W = np.fft.rfft(w, size)
        AR = np.fft.rfft(a_rev, size)
        conv_result = np.fft.irfft(W * AR, size)[:conv_len]
        h = conv_result[len(a) - 1:len(a) - 1 + len(a)]

        # Gradient of L w.r.t. a (before normalization)
        L = max_c + T * np.log(np.sum(exp_c))
        grad_L = 2.0 * h - 2.0 * L  # dL/da[i] = 2*h[i] - 2*L (since s=1)

        # Project gradient onto tangent space of simplex
        grad_L = grad_L - np.mean(grad_L)

        # Mirror descent update: a_new = a * exp(-eta * grad_L)
        a_new = a * np.exp(-eta * grad_L)
        a_new = np.clip(a_new, 1e-15, None)
        a_new = a_new / a_new.sum()

        score_new = local_score(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()

        if score_new < local_score(a):
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 30:
                eta *= 0.7
                stall = 0
            if eta < 1e-6:
                break

    return best_a, best_score


def peak_targeting_search(a0, time_limit=80, seed=42):
    """Local search targeting specific peak positions."""
    rng = np.random.default_rng(seed)
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        c = np.convolve(a, a)
        s = np.sum(a)
        current_score = 2.0 * n * np.max(c) / (s * s)

        # Normalize for computation
        a_norm = a / s
        c_norm = c / (s * s)

        # Find top peaks
        top_peak_count = min(5, 2 * n - 1)
        peak_indices = np.argsort(c_norm)[::-1][:top_peak_count]

        improved = False
        for k in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break

            # Contributions to c[k]: a[i] * a[k-i]
            contribs = np.zeros(n)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    contribs[i] = a_norm[i] * a_norm[j]

            # Strategy: decrease high contributors to this peak,
            # increase elements that don't contribute
            top_k = max(3, n // 15)
            top_idx = np.argsort(contribs)[-top_k:]
            low_idx = np.argsort(contribs)[:max(3, n // 10)]

            for attempt in range(8):
                candidate = a.copy()
                dec = rng.uniform(0.75, 0.95)
                candidate[top_idx] *= dec
                inc = rng.uniform(1.05, 1.25)
                candidate[low_idx] *= inc
                candidate = np.clip(candidate, 0.0, 1000.0)

                new_score = local_score(candidate)
                if new_score < current_score:
                    a = candidate
                    current_score = new_score
                    improved = True
                    if current_score < best_score:
                        best_score = current_score
                        best_a = a.copy()
                    break

        if not improved:
            no_improve += 1
            if no_improve > 50:
                # Small random perturbation to escape
                a = a * np.exp(rng.normal(0, 0.01, n))
                a = np.clip(a, 0.0, 1000.0)
                no_improve = 0

    return best_a, best_score


def block_local_search(a0, time_limit=60, seed=42):
    """Block-based local search with simulated annealing."""
    rng = np.random.default_rng(seed)
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    temp = 0.001
    cooling = 0.9995

    while time.monotonic() - t0 < time_limit:
        candidate = a.copy()
        r = rng.random()

        if r < 0.4:
            # Targeted: reduce top contributors to main peak
            c = np.convolve(a, a)
            s = np.sum(a)
            peak_idx = int(np.argmax(c / (s * s)))
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            top_k = max(4, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.015, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 5:
                n_inc = min(8, len(other) // 3)
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.015, size=n_inc))
        elif r < 0.7:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width + 1))
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.025, size=width))
        elif r < 0.85:
            # Pair swap
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = rng.uniform(-0.2, 0.2) * a[i]
                candidate[i] = max(0, a[i] - delta)
                candidate[j] = max(0, a[j] + delta)
        else:
            # Small global
            candidate = a * np.exp(rng.normal(0, 0.008, size=n))

        candidate = np.clip(candidate, 0.0, 1000.0)
        new_score = local_score(candidate)

        if new_score < best_score:
            best_score = new_score
            best_a = candidate.copy()

        if new_score < local_score(a):
            a = candidate
        elif rng.random() < np.exp(-(new_score - local_score(a)) / max(temp, 1e-15)):
            a = candidate

        temp *= cooling

    return best_a, best_score


def make_tent_init(n, gamma, rng):
    """Tent-shaped initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = 1.0 - gamma * np.abs(t - center) / (center + 1)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_cosine_init(n, rng, n_harm=6):
    """Multi-harmonic cosine initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(1, n_harm + 1):
        amp = 0.25 / (k + 0.5)
        freq = rng.uniform(0.4, 2.5) * k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_bivariate_init(n, rng):
    """Initialize based on a product of two smooth functions."""
    t = np.arange(n, dtype=np.float64) / n
    f1 = 1.0 + rng.uniform(0.3, 0.8) * np.cos(2 * np.pi * rng.uniform(1, 3) * t + rng.uniform(0, 2 * np.pi))
    f2 = 1.0 + rng.uniform(0.2, 0.6) * np.sin(2 * np.pi * rng.uniform(1, 4) * t + rng.uniform(0, 2 * np.pi))
    a = f1 * f2 + 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def resample_to_n(seq, new_n):
    """Resample sequence to new length."""
    if len(seq) == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)
