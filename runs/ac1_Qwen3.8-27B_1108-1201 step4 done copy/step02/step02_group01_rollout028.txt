import numpy as np
import time

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_objective(a, beta=100.0):
    """Compute smoothed max of autocorrelation."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf, 0.0
    conv = fft_convolve(a, a)
    # Smoothed max
    max_conv = np.max(conv)
    shifted = beta * (conv - max_conv)
    log_sum = max_conv + np.log(np.sum(np.exp(shifted))) / beta
    # Score
    score = 2.0 * n * log_sum / (S * S)
    return score, log_sum

def fista_optimize(a, time_limit, beta=100.0, verbose=True):
    """
    FISTA (accelerated proximal gradient) for minimizing the smoothed
    max autocorrelation with L2 regularization for stability.
    """
    n = len(a)
    S_target = np.sqrt(2.0 * n)
    a = a * S_target / np.sum(a)
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = np.inf
    
    # Lipschitz constant estimate (we'll adapt it)
    L = 1.0
    y = a.copy()
    tk = 1.0
    
    for iteration in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute gradient at y
        conv_y = fft_convolve(y, y)
        max_conv = np.max(conv_y)
        shifted = beta * (conv_y - max_conv)
        probs = np.exp(shifted)
        probs /= np.sum(probs)
        
        # Gradient of smoothed max
        grad_M = np.zeros(n)
        for k in range(len(conv_y)):
            if probs[k] < 1e-15:
                continue
            k_int = int(k)
            j_min = max(0, k_int - n + 1)
            j_max = min(k_int, n - 1)
            for j in range(j_min, j_max + 1):
                # ∂(a*a)[k]/∂a[j]
                if 2 * j == k_int:
                    grad_M[j] += 2.0 * y[j] * probs[k]
                else:
                    grad_M[j] += 2.0 * y[k_int - j] * probs[k]
        
        # Full gradient of score = 2n * M / S^2 (with S fixed by normalization)
        # Since we normalize after each step, we just need gradient of M
        # and then normalize.
        
        # Backtracking line search
        step_size = 1.0 / L
        x_new = y - step_size * grad_M
        x_new = np.maximum(x_new, 0.0)
        x_new = x_new * S_target / np.sum(x_new)
        
        # Check if step is good (backtracking)
        # For simplicity, use fixed step with periodic adjustment
        a_new = x_new
        
        # FISTA update
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        y = a_new + ((tk - 1.0) / t_next) * (a_new - a)
        y = np.maximum(y, 0.0)
        tk = t_next
        
        a = a_new
        
        # Track best
        S_a = np.sum(a)
        conv_a = fft_convolve(a, a)
        score = 2.0 * n * np.max(conv_a) / (S_a * S_a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if verbose and iteration % 1000 == 0:
            print(f"    FISTA iter {iteration}: score={best_score:.6f}, L={L:.2f}")
        
        # Adapt L
        if iteration % 500 == 0:
            L *= 1.01
    
    return best_a

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))

def smoothed_max_conv(a, beta):
    """Compute smoothed max of autocorrelation and its gradient."""
    n = len(a)
    conv = fft_convolve(a, a)
    max_conv = np.max(conv)
    shifted = beta * (conv - max_conv)
    # Numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    log_sum_exp = max_conv + np.log(np.sum(exp_shifted)) / beta
    probs = exp_shifted / np.sum(exp_shifted)
    
    # Gradient: grad_M[i] = 2 * sum_{j=0}^{n-1} probs[i+j] * a[j]
    # This is 2 * cross-correlation of probs and a
    # Use FFT-based computation
    N = 1
    while N < (2 * n - 1) + n:
        N <<= 1
    F_p = np.fft.rfft(probs, N)
    F_a = np.fft.rfft(a, N)
    corr = np.fft.irfft(F_p * F_a, N)[:n]
    grad = 2.0 * corr
    
    return log_sum_exp, grad, conv

def coordinate_descent(a_in, time_limit, verbose=True):
    """
    Coordinate descent with exact line minimization.
    For each coordinate, minimize the peak autocorrelation along that axis.
    """

    S_target = np.sqrt(2.0 * n)
    a = a_in * S_target / max(np.sum(a_in), 1e-10)

    sweep = 0

        np.random.shuffle(coord_order)
        
        for i in coord_order:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Minimize over a[i] keeping sum fixed
            # Adjust a[0] to maintain sum (or distribute to other coords)
            # Simple approach: vary a[i] and a[j] for j != i proportionally
            
            # Try different values of a[i]
            current_val = a[i]
            if current_val < 1e-10:
                # Try setting it to a small positive value
                candidates = [current_val + 0.01 * S_target / n]
            else:
                candidates = [current_val * 0.5, current_val * 0.8, current_val * 1.2, current_val * 1.5, current_val * 0.3]

            best_cand_score = best_score
            
            for new_val in candidates:
                if new_val < 0:
                    continue
                test_a = a.copy()
                # Distribute the difference to maintain sum
                diff = new_val - test_a[i]
                test_a[i] = new_val
                # Add diff to a random other coordinate (or spread evenly)
                if abs(diff) > 1e-15:
                    # Spread to all other coordinates proportionally
                    others = np.ones(n)
                    others[i] = 0
                    others = np.maximum(others, 0)
                    total_others = np.sum(others * a)
                    if total_others > 1e-10:
                        test_a += diff * others * a / total_others
                    else:
                        test_a[i] = new_val  # Can't maintain sum, just set it
                
                test_a = np.maximum(test_a, 0)
                t_sum = np.sum(test_a)
                if t_sum < 1e-10:
                    continue
                test_a = test_a * S_target / t_sum

            print(f"    CD sweep {sweep} (t={elapsed:.1f}s): score={best_score:.6f}")
    
    return best_a

"""
FISTA with smoothed max autocorrelation + multi-scale search + SA polish.
Uses FFT-based gradient computation for O(n log n) per iteration.
Anneals the smoothing parameter beta for progressively sharper optimization.
"""

import numpy as np
import time


def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))


def compute_grad_smoothed_max(a, beta):
    n = len(a)
    conv = fft_convolve(a, a)
    max_conv = np.max(conv)
    shifted = beta * (conv - max_conv)
    shifted = np.clip(shifted, -500, 0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    M_smooth = max_conv + np.log(sum_exp) / beta
    probs = exp_shifted / sum_exp
    N = 1
    while N < (2 * n - 1) + n:
        N <<= 1
    F_p = np.fft.rfft(probs, N)
    F_a = np.fft.rfft(a, N)
    corr_full = np.fft.irfft(F_p * F_a, N)
    grad = 2.0 * corr_full[:n]
    return M_smooth, grad, conv

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 150
        if remaining > 60:
            t_limit = min(remaining, 250.0)
            print(f"  limit={t_limit:.0f}s")
            result = sa_optimize(best_seq, t_limit, verbose=True)
            s = local_score(result)
            if s < best_score:
                best_score = s
                best_seq = result.copy()
                print(f"  *** SA: {best_score:.6f} ***")

    # Phase 5: Final FISTA with high beta
    print("\n=== Phase 5: Final FISTA ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 15
        if remaining > 20:
            t_limit = min(remaining, 120.0)
            print(f"  limit={t_limit:.0f}s")
            result = fista_optimize(best_seq, t_limit, beta_start=200, beta_end=2000, verbose=True)
            s = local_score(result)
            if s < best_score:
                best_score = s
                best_seq = result.copy()
                print(f"  *** Final: {best_score:.6f} ***")

    # Final output
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]

    return [1.0] * 256
