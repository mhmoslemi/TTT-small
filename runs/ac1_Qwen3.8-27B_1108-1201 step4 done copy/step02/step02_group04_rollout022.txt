def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def softmax(x):
    x_shifted = x - x.max()
    ex = np.exp(x_shifted)
    return ex / ex.sum()

def compute_objective(a):
    """Compute the normalized objective value (sum(a) assumed to be 1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-start L-BFGS optimization
with softmax parameterization, adaptive log-sum-exp smoothing, coordinate descent
polishing, and diverse structured initializations (sigmoid, spectral, anti-palindromic).
Exploits the non-palindromic structure of optimal solutions.
"""

import numpy as np
import scipy.optimize as optimize
import scipy.signal as sig
import time


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    ex = np.exp(x_shifted)
    return ex / ex.sum()


def compute_objective(a):
    """Compute the normalized objective 2n*max(conv) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def smoothed_obj_and_grad(x, beta):
    """Compute smoothed objective and gradient for L-BFGS.
    x: raw parameters, a = softmax(x), sum(a) = 1.
    Objective: 2n * (1/beta) * log(sum_k exp(beta * conv[k]))
    """
    a = softmax(x)
    n = len(a)
    conv = fast_convolve(a)

    # Log-sum-exp for numerical stability
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    shifted_max = shifted.max()
    log_sum = shifted_max + np.log(np.sum(np.exp(shifted - shifted_max)))
    smoothed_max = conv_max + log_sum / beta

    value = 2.0 * n * smoothed_max

    # Weights (softmax in conv-space)
    weights = np.exp(shifted - shifted_max)
    weights /= weights.sum()

    # dcda[i] = d(smoothed_max)/d(a[i]) = 2 * sum_{j=0}^{n-1} weights[i+j] * a[j]
    # This is 2 * cross_correlation(weights, a, mode='valid')
    dcda = 2.0 * sig.correlate(weights, a, mode='valid')

    # grad_x[j] = 2n * a[j] * (dcda[j] - dot(dcda, a))
    dot_da = np.dot(dcda, a)
    grad_x = 2.0 * n * a * (dcda - dot_da)

    return value, grad_x


def lbfgs_optimize(x0, betas, max_iter_each=300, time_limit=60.0):
    """Run L-BFGS with a schedule of increasing beta values."""
    t0 = time.monotonic()
    best_x = x0.copy()
    best_val = compute_objective(softmax(x0))

    for beta in betas:
        if time.monotonic() - t0 > time_limit:
            break

        def func(x):
            return smoothed_obj_and_grad(x, beta)[0]

        def grad(x):
            return smoothed_obj_and_grad(x, beta)[1]

        try:
            res = optimize.minimize(
                func, x0, jac=grad, method='L-BFGS-B',
                options={'maxiter': max_iter_each, 'ftol': 1e-14, 'gtol': 1e-10,
                         'maxfun': 20000}
            )
        except Exception:
            break

        a = softmax(res.x)
        val = compute_objective(a)
        if val < best_val:
            best_val = val
            best_x = res.x.copy()

        x0 = res.x

    return best_x, best_val


def coord_descent_polish(a0, time_limit=20.0):
    """Polish using coordinate descent with multiple step sizes."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a0, compute_objective(softmax(np.zeros(n)))
    a /= s

    best_val = compute_objective(a)
    t0 = time.monotonic()
    scales = [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 1.7, 2.0]

    stagnant = 0
    while time.monotonic() - t0 < time_limit:
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_ai = a[i]
            if old_ai < 1e-15:
                # Try small positive value
                for scale in [0.01, 0.05]:
                    a_test = a.copy()
                    a_test[i] = a.mean() * scale
                    a_test /= a_test.sum()
                    val = compute_objective(a_test)
                    if val < best_val - 1e-13:
                        best_val = val
                        a = a_test.copy()
                        improved = True
                        break
                continue

            for scale in scales:
                a_test = a.copy()
                a_test[i] = old_ai * scale
                a_test = np.maximum(a_test, 0.0)
                s_test = a_test.sum()
                if s_test < 1e-10:
                    continue
                a_test /= s_test
                val = compute_objective(a_test)
                if val < best_val - 1e-13:
                    best_val = val
                    a = a_test.copy()
                    improved = True
                    break

        if not improved:
            stagnant += 1
            if stagnant > 3:
                break
        else:
            stagnant = 0

    return a, best_val


def gen_initializations(n, rng):
    """Generate diverse initializations of length n, each summing to 1."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0

    # Uniform
    inits['uni'] = np.ones(n) / n

    # Sigmoid profiles (anti-palindromic)
    for s_name, s_val in [('sig5', 0.05), ('sig10', 0.10), ('sig20', 0.20), ('sig30', 0.30)]:
        sig_v = 1.0 / (1.0 + np.exp(-(x - mid) / (s_val * n)))
        sig_v = np.maximum(sig_v, 1e-8)
        inits[s_name] = sig_v / sig_v.sum()

    # Ramp (linear decreasing)
    ramp = np.linspace(1.0, 0.05, n)
    inits['ramp'] = ramp / ramp.sum()

    # Anti-ramp (linear increasing)
    inits['aramp'] = np.flip(ramp) / ramp.sum()

    # Cosine (Hann window)
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
    hann = np.maximum(hann, 1e-8)
    inits['hann'] = hann / hann.sum()

    # Blackman window
    blackman = (0.42 - 0.5 * np.cos(2.0 * np.pi * x / n)
                + 0.08 * np.cos(4.0 * np.pi * x / n))
    blackman = np.maximum(blackman, 1e-8)
    inits['black'] = blackman / blackman.sum()

    # Two-hump
    hump = (np.exp(-((x - n / 4) / (n / 8)) ** 2)
            + np.exp(-((x - 3 * n / 4) / (n / 8)) ** 2))
    hump = np.maximum(hump, 1e-8)
    inits['hump'] = hump / hump.sum()

    # Alternating
    alt = np.where(x.astype(int) % 2 == 0, 1.0, 0.3)
    inits['alt'] = alt / alt.sum()

    # Block (first half high, second half low)
    block = np.concatenate([np.ones(n // 2), np.full(n - n // 2, 0.15)])
    inits['block'] = block / block.sum()

    # Spectral: random power spectrum -> time domain
    for trial in range(3):
        L = 1 << (2 * n - 1).bit_length()
        P = rng.chisquare(2.0, L)  # random power spectrum
        P = np.maximum(P, 0.0)
        t = np.fft.irfft(P)[:n]
        t = np.maximum(t, 1e-8)
        inits[f'spec{trial}'] = t / t.sum()

    # Random Dirichlet (concentrated)
    inits['dir_conc'] = rng.dirichlet(np.ones(n) * 0.3)
    # Random Dirichlet (uniform)
    inits['dir_uni'] = rng.dirichlet(np.ones(n))
    # Random exponential
    r = rng.exponential(1.0, n)
    inits['exp_rnd'] = r / r.sum()

    # Asymmetric Gaussian
    for mu_frac in [0.3, 0.4, 0.6, 0.7]:
        g = np.exp(-0.5 * ((x - mu_frac * n) / (0.15 * n)) ** 2)
        g = np.maximum(g, 1e-8)
        inits[f'g{int(mu_frac * 100)}'] = g / g.sum()

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main entry point: multi-start L-BFGS + coordinate descent optimization."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Try parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = parent.copy()
            print(f"Parent: n={len(parent)}, score={score:.6f}")
    except Exception as e:
        print(f"No parent available: {e}")

    # Choose lengths to try
    lengths = [256, 384, 512, 768, 1024]

    # Filter lengths based on time
    active_lengths = []
    for n in lengths:
        if time.monotonic() < deadline - 80:
            active_lengths.append(n)
            print(f"Will try n={n}")

    all_results = []

    # Phase 1: Scan across lengths and initializations
    print("\n=== Phase 1: Initialization scan ===")
    for n in active_lengths:
        if time.monotonic() > deadline - 60:
            break

        inits = gen_initializations(n, rng)

        # Add parent interpolation
        if parent is not None:
            if parent.size == n:
                inits['parent'] = parent.copy()
            elif parent.size >= n:
                pp = np.maximum(parent[:n], 0.0)
                if pp.sum() > 1e-10:
                    inits['parent'] = pp / pp.sum()
            else:
                pp = np.interp(np.linspace(0, 1, n),
                               np.linspace(0, 1, parent.size), parent)
                pp = np.maximum(pp, 0.0)
                if pp.sum() > 1e-10:
                    inits['parent'] = pp / pp.sum()

        n_inits = len(inits)
        remaining_for_phase1 = max(0, deadline - time.monotonic() - 120)
        total_inits_remaining = sum(
            len(gen_initializations(l, rng)) + 1
            for l in active_lengths if l >= n
        )
        time_per_init = max(8.0, remaining_for_phase1 / max(1, total_inits_remaining))

        for name, a0 in inits.items():
            if time.monotonic() > deadline - 60:
                break

            t0 = time.monotonic()
            time_budget = min(time_per_init, deadline - time.monotonic() - 60)
            if time_budget < 6:
                continue

            # Convert to log-space for softmax parameterization
            x0 = np.log(np.maximum(a0, 1e-12)) + rng.normal(0, 0.01, n)

            # L-BFGS with adaptive smoothing
            if time_budget > 50:
                betas = [10, 30, 80, 200, 500]
                max_iter = 200
            elif time_budget > 25:
                betas = [20, 60, 150, 400]
                max_iter = 150
            else:
                betas = [50, 150, 400]
                max_iter = 100

            x_opt, val_opt = lbfgs_optimize(x0, betas, max_iter_each=max_iter,
                                            time_limit=time_budget * 0.6)
            a_opt = softmax(x_opt)
            score = compute_objective(a_opt)

            # Coordinate descent polish
            if time.monotonic() < deadline - 40:
                cd_time = min(15.0, time_budget * 0.35)
                a_cd, cd_val = coord_descent_polish(a_opt, time_limit=cd_time)
                if cd_val < score:
                    score = cd_val
                    a_opt = a_cd

            eval_score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(eval_score):
                all_results.append((eval_score, n, name, a_opt))
                if eval_score < best_score:
                    best_score = eval_score
                    best_seq = a_opt.copy()
                    print(f"  n={n:4d} {name:10s} -> {eval_score:.6f} ({time.monotonic()-t0:.0f}s)")

    print(f"\nPhase 1 done. Best: {best_score:.6f}, candidates: {len(all_results)}")

    # Phase 2: Deep optimization on top results
    all_results.sort(key=lambda x: x[0])
    top_k = min(6, len(all_results))

    remaining = deadline - time.monotonic() - 30
    if remaining > 40 and top_k > 0:
        print(f"\n=== Phase 2: Deep optimization (top {top_k}) ===")
        t_each = max(20.0, remaining / top_k)

        for idx in range(top_k):
            if time.monotonic() > deadline - 25:
                break
            sc, n, name, a = all_results[idx]
            ta = min(t_each, deadline - time.monotonic() - 25)
            if ta < 15:
                break

            print(f"  [{idx}] n={n} {name} score={sc:.6f}, budget={ta:.0f}s")

            # Aggressive L-BFGS with higher betas
            x0 = np.log(np.maximum(a, 1e-12))
            x_opt, val_opt = lbfgs_optimize(x0, [50, 150, 400, 1000, 2500],
                                            max_iter_each=300, time_limit=ta * 0.5)
            a_opt = softmax(x_opt)
            score = compute_objective(a_opt)

            # Coordinate descent
            if time.monotonic() < deadline - 15:
                a_cd, cd_val = coord_descent_polish(a_opt, time_limit=min(25.0, ta * 0.35))
                if cd_val < score:
                    score = cd_val
                    a_opt = a_cd

            eval_score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(eval_score) and eval_score < best_score:
                best_score = eval_score
                best_seq = a_opt.copy()
                print(f"    -> {eval_score:.6f}")

    # Phase 3: Random perturbations + local opt on the best
    if best_seq is not None and time.monotonic() < deadline - 15:
        print(f"\n=== Phase 3: Perturbation + refinement ===")
        n = len(best_seq)
        a = best_seq.copy()
        perturbation_scales = [0.005, 0.01, 0.02, 0.05]

        for ps in perturbation_scales:
            if time.monotonic() > deadline - 15:
                break
            for trial in range(3):
                if time.monotonic() > deadline - 15:
                    break
                noise = rng.normal(0, ps * max(a.mean(), 1e-6), n)
                a_pert = np.maximum(a + noise, 0.0)
                s = a_pert.sum()
                if s < 1e-10:
                    continue
                a_pert /= s

                x0 = np.log(np.maximum(a_pert, 1e-12))
                x_opt, val_opt = lbfgs_optimize(x0, [100, 300, 800],
                                                max_iter_each=200, time_limit=25)
                a_opt = softmax(x_opt)
                score = compute_objective(a_opt)

                # Quick coord descent
                if time.monotonic() < deadline - 10:
                    a_cd, cd_val = coord_descent_polish(a_opt, time_limit=10)
                    if cd_val < score:
                        score = cd_val
                        a_opt = a_cd

                eval_score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(eval_score) and eval_score < best_score:
                    best_score = eval_score
                    best_seq = a_opt.copy()
                    a = best_seq.copy()
                    print(f"  ps={ps:.4f} trial={trial} -> {eval_score:.6f}")

    # Phase 4: Try LP-guided step on the best (different from Phase 1/2)
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"\n=== Phase 4: Final LP-guided refinement ===")
        a = best_seq.copy()
        n = len(a)
        conv = fast_convolve(a)
        M = conv.max()

        # Solve LP: max sum(g) s.t. g>=0, (a*g)[k] <= M
        from scipy.sparse import csc_matrix
        rows_list = []
        cols_list = []
        for i in range(n):
            for j in range(n):
                rows_list.append(i + j)
                cols_list.append(i)
        rows_arr = np.array(rows_list, dtype=np.int32)
        cols_arr = np.array(cols_list, dtype=np.int32)
        vals = np.tile(a, n)
        A_mat = csc_matrix((vals, (rows_arr, cols_arr)), shape=(2 * n - 1, n))

        c_vec = -np.ones(n)
        b_vec = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        remaining_lp = deadline - time.monotonic() - 5
        if remaining_lp > 10:
            try:
                res = optimize.linprog(c_vec, A_ub=A_mat, b_ub=b_vec, bounds=bounds,
                                       method='highs', options={'disp': False,
                                                                'time_limit': remaining_lp * 0.7})
                if res.success:
                    g = np.maximum(res.x, 0.0)
                    g_sum = g.sum()
                    if g_sum > 1e-10:
                        g /= g_sum
                        # Try several blend ratios
                        for alpha in [0.3, 0.5, 0.7, 0.9, 1.0]:
                            a_mix = (1 - alpha) * a + alpha * g
                            a_mix = np.maximum(a_mix, 0.0)
                            s_mix = a_mix.sum()
                            if s_mix < 1e-10:
                                continue
                            a_mix /= s_mix
                            eval_sc = float(evaluate_sequence(a_mix.tolist()))
                            if np.isfinite(eval_sc) and eval_sc < best_score:
                                best_score = eval_sc
                                best_seq = a_mix.copy()
                                a = a_mix.copy()
                                print(f"  LP blend alpha={alpha:.1f} -> {eval_sc:.6f}")
            except Exception as e:
                print(f"  LP failed: {e}")

    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL: n={len(best_seq)}, score={best_score:.6f}")
    print(f"{'='*50}")
    return [float(x) for x in best_seq]
