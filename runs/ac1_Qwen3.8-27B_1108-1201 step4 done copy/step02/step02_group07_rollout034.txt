import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def fft_convolve(a, b=None):
    n = len(a)
    if b is None:
        b = a
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n+m-1]

def score_fast(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))

def subgradient_x(x, active_indices):
    """Compute subgradient of max_k sum_j exp(x_j + x_{k-j}) w.r.t. x."""
    n = len(x)
    a = np.exp(x)
    g = np.zeros(n)

        k = int(k)
        lo = max(0, k - (n-1))
        hi = min(n-1, k)
        for j in range(lo, hi+1):
            g[j] += 2.0 * a[j] * a[k-j]
    g /= len(active_indices)
    return g

                    break

Hybrid optimization for minimum peak aperiodic autocorrelation of non-negative

def fft_convolve(a, b=None):
    n = len(a)
    if b is None:

    while size < n + m - 1:

def compute_subgrad_x(x, active_lags):

def project_onto_sum_exp(y, S, max_bisect=200):

                print(f"    [xPG {time.monotonic()-t0:6.1f}s] it={it} s={s:.6f} best={best_score:.6f} lr={lr:.4f}")

        lr = max(lr_min, lr * lr_decay)

        if stall >= 80:
            restarts += 1
            momentum = np.zeros(n)
            active_frac = min(0.5, active_frac * 1.4)
            lr = max(0.3, lr * 2.0)
            rng_r = np.random.default_rng(999 + restarts * 31)
            x += 0.1 * rng_r.standard_normal(n)
            x = np.clip(x, -40, 40)
            x = project_sum_exp(x, S)
            stall = 0

    a = np.exp(np.clip(x, -40, 40))
    s = score_fast(a)
    if s < best_score:
        best_score = s
        best_a = a.copy()

    return best_a, best_score


def lp_optimize(f_init, time_limit, verbose=True):
    """LP-based optimization with quadratic step size and multi-M search."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 5.0:
            break

        c_f = fft_convolve(f, f)
        M = float(np.max(c_f))
        S = float(np.sum(f))

        b = solve_lp_direction(f, M, timeout=4.0)
        if b is None:
            break

        Sb = float(np.sum(b))
        if Sb < 1e-12:
            break
        b = b * (S / Sb)

        c_fb = fft_convolve(f, b)
        c_bb = fft_convolve(b, b)

        best_t = 0.0
        best_s = np.inf
        for alpha in [1.0, 0.98, 0.95, 0.9, 0.85, 0.8, 0.7, 0.6]:
            M_target = M * alpha
            t = quadratic_step_size(c_f, c_fb, c_bb, M_target, t_max=0.8)
            if t > 0.005:
                f_cand = (1.0 - t) * f + t * b
                f_cand = np.clip(f_cand, 0.0, 1000.0)
                s_cand = score_fast(f_cand)
                if s_cand < best_s:
                    best_s = s_cand
                    best_t = t

        if best_t < 0.005:
            for t_try in [0.02, 0.05, 0.1, 0.15, 0.2]:
                f_cand = (1.0 - t_try) * f + t_try * b
                f_cand = np.clip(f_cand, 0.0, 1000.0)
                s_cand = score_fast(f_cand)
                if s_cand < best_s:
                    best_s = s_cand
                    best_t = t_try

        if best_t < 0.001:
            break

        f = (1.0 - best_t) * f + best_t * b
        f = np.clip(f, 0.0, 1000.0)
        it += 1

        if best_s < best_score:
            best_score = best_s
            best_f = f.copy()

        if verbose and it % 15 == 0:
            print(f"    [LP {time.monotonic()-t0:6.1f}s] it={it} s={best_s:.6f} best={best_score:.6f} t={best_t:.4f}")

    return best_f, best_score


def spsa_optimize(a_init, time_limit, verbose=True):
    """SPSA with adaptive step sizes."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = float(np.sum(a))
    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(4242)

    c = 0.05
    A = 0.3
    alpha_p = 0.6
    gamma = 0.6
    A1 = 10.0
    A2 = 100000.0

    it = 0
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 2.0:
            break

        delta = rng.choice([-1.0, 1.0], size=n)

        a_plus = np.clip(a + c * delta, 1e-10, 1000.0)
        a_minus = np.clip(a - c * delta, 1e-10, 1000.0)

        Sp = float(np.sum(a_plus))
        Sm = float(np.sum(a_minus))
        if Sp < 0.01 or Sm < 0.01:
            c *= 0.9
            it += 1
            continue

        a_plus = a_plus * (S / Sp)
        a_minus = a_minus * (S / Sm)

        s_plus = score_fast(a_plus)
        s_minus = score_fast(a_minus)

        if np.isinf(s_plus) or np.isinf(s_minus) or s_plus == s_minus:
            c *= 0.95
            it += 1
            continue

        g_est = (s_plus - s_minus) / (2.0 * c) * delta
        a_step = A / (A1 + it) ** alpha_p
        a = a - a_step * g_est
        a = np.clip(a, 1e-10, 1000.0)
        a = a * (S / float(np.sum(a)))

        it += 1
        if it % 20 == 0:
            s = score_fast(a)
            if s < best_score:
                best_score = s
                best_a = a.copy()
            if verbose and it % 500 == 0:
                print(f"    [SPSA {time.monotonic()-t0:6.1f}s] it={it} s={s:.6f} best={best_score:.6f} c={c:.4f}")

        c = c / (1.0 + gamma * a_step / c ** alpha_p) ** 1.0
        c = max(c, 1e-6)

    s = score_fast(a)
    if s < best_score:
        best_score = s
        best_a = a.copy()

    return best_a, best_score


def block_refine(a_init, time_limit, verbose=True):
    """Block coordinate descent: optimize random subsets of coordinates."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = float(np.sum(a))
    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(555)
    block_sizes = [max(4, n // 16), max(8, n // 8), max(12, n // 4)]

    it = 0
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 2.0:
            break

        bs = block_sizes[it % len(block_sizes)]
        idx = rng.choice(n, size=bs, replace=False)
        idx.sort()

        a_sub = a[idx].copy()
        fixed = a.copy()

        best_sub = a_sub.copy()
        best_sub_score = best_score
        sub_comps = 0

        for _ in range(30):
            sub_comps += 1
            trial = a.copy()
            trial[idx] = a_sub
            S_trial = float(np.sum(trial))
            if S_trial < 0.01:
                break
            trial = trial * (S / S_trial)
            s = score_fast(trial)
            if s < best_sub_score:
                best_sub_score = s
                best_sub = a_sub.copy()

            # Perturb the block
            r = rng.random()
            if r < 0.5:
                a_sub *= np.exp(rng.normal(-0.02, 0.03, size=bs))
            elif r < 0.8:
                a_sub += rng.normal(0, 0.01 * np.max(a_sub), size=bs)
            else:
                peak = int(np.argmax(fft_convolve(a, a)))
                contribs = np.zeros(bs)
                for ii in range(bs):
                    j = idx[ii]
                    ki = peak - j
                    if 0 <= ki < n:
                        contribs[ii] = a[j] * a[ki]
                max_c = np.max(contribs)
                if max_c > 0:
                    mask = contribs > 0.7 * max_c
                    a_sub[mask] *= np.exp(rng.normal(-0.05, 0.02, size=int(np.sum(mask))))
                    a_sub[~mask] *= np.exp(rng.normal(0.02, 0.02, size=int(np.sum(~mask))))

            a_sub = np.clip(a_sub, 1e-10, 1000.0)

        a[idx] = best_sub
        it += 1

        s = score_fast(a)
        if s < best_score:
            best_score = s
            best_a = a.copy()

        if verbose and it % 50 == 0:
            print(f"    [BLOCK {time.monotonic()-t0:6.1f}s] it={it} s={s:.6f} best={best_score:.6f}")

    return best_a, best_score


def make_init_hann(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_bell(n, rng):
    t = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = np.exp(-0.5 * ((t - mid) / (n * 0.3)) ** 2)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_chirp(n, rng):
    t = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    window = np.exp(-0.3 * ((t - mid) / (n * 0.4)) ** 2)
    chirp = 0.5 + 0.5 * np.cos(2.0 * np.pi * t * t / (3.0 * n))
    a = window * (0.3 + 0.7 * chirp)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800):
    """Main search: multi-phase hybrid optimization across sequence lengths."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Try provided sequence
    provided = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = score_fast(provided)
        print(f"Provided: n={len(provided)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = provided.copy()

    # Phase 1: Projected gradient in x-space on provided sequence
    if provided is not None and time.monotonic() < deadline - 600:
        t_alloc = min(300.0, deadline - time.monotonic() - 600.0)
        print(f"\n=== Phase 1: x-space projected gradient on provided (n={len(provided)}, {t_alloc:.0f}s) ===")
        f1, s1 = projected_gradient_xspace(provided, t_alloc, verbose=True)
        if s1 < best_score:
            best_score = s1
            best_seq = f1.copy()
            print(f"  NEW BEST: {best_score:.6f}")

    # Phase 2: LP optimization on best sequence
    if best_seq is not None and time.monotonic() < deadline - 400:
        t_alloc = min(250.0, deadline - time.monotonic() - 400.0)
        print(f"\n=== Phase 2: LP optimization (n={len(best_seq)}, {t_alloc:.0f}s) ===")
        f2, s2 = lp_optimize(best_seq, t_alloc, verbose=True)
        if s2 < best_score:
            best_score = s2
            best_seq = f2.copy()
            print(f"  NEW BEST: {best_score:.6f}")

    # Phase 3: Another round of x-space gradient
    if best_seq is not None and time.monotonic() < deadline - 300:
        t_alloc = min(200.0, deadline - time.monotonic() - 300.0)
        print(f"\n=== Phase 3: x-space gradient round 2 ({t_alloc:.0f}s) ===")
        f3, s3 = projected_gradient_xspace(best_seq, t_alloc, verbose=True)
        if s3 < best_score:
            best_score = s3
            best_seq = f3.copy()
            print(f"  NEW BEST: {best_score:.6f}")

    # Phase 4: Block coordinate refinement
    if best_seq is not None and time.monotonic() < deadline - 200:
        t_alloc = min(150.0, deadline - time.monotonic() - 200.0)
        print(f"\n=== Phase 4: Block coordinate ({t_alloc:.0f}s) ===")
        f4, s4 = block_refine(best_seq, t_alloc, verbose=True)
        if s4 < best_score:
            best_score = s4
            best_seq = f4.copy()
            print(f"  NEW BEST: {best_score:.6f}")

    # Phase 5: Try other lengths with smart inits
    other_ns = [128, 192, 320, 384, 512]
    if time.monotonic() < deadline - 100:
        remaining = deadline - time.monotonic() - 80.0
        per_task = max(20.0, remaining / (len(other_ns) + 1))
        print(f"\n=== Phase 5: Other lengths (budget {remaining:.0f}s total) ===")

        for n in other_ns:
            if time.monotonic() > deadline - 80:
                break
            t_task = min(per_task, deadline - time.monotonic() - 60.0)
            if t_task < 15:
                break

            # Use best_seq as template if length is compatible
            if best_seq is not None:
                x_old = np.linspace(0, 1, len(best_seq))
                x_new = np.linspace(0, 1, n)
                f0 = np.interp(x_new, x_old, best_seq)
            else:
                f0 = make_init_hann(n, rng)

            f0 = np.clip(f0, 0.01, None)
            print(f"\n  n={n}: starting score={score_fast(f0):.6f}, budget={t_alloc:.0f}s")

            # Quick x-space gradient
            t_sub = max(10.0, t_task * 0.6)
            fc, sc = projected_gradient_xspace(f0, t_sub, verbose=False)
            if sc < best_score:
                best_score = sc
                best_seq = fc.copy()
                print(f"  NEW BEST from n={n}: {best_score:.6f}")

            # LP polish if time remains
            if time.monotonic() < deadline - 80:
                t_lp = max(10.0, min(60.0, deadline - time.monotonic() - 80.0))
                fl, sl = lp_optimize(fc, t_lp, verbose=False)
                if sl < best_score:
                    best_score = sl
                    best_seq = fl.copy()
                    print(f"  NEW BEST from n={n} LP: {best_score:.6f}")

    # Phase 6: SPSA final polish
    if best_seq is not None and time.monotonic() < deadline - 40:
        t_alloc = min(120.0, deadline - time.monotonic() - 20.0)
        if t_alloc > 20:
            print(f"\n=== Phase 6: SPSA polish ({t_alloc:.0f}s) ===")
            f6, s6 = spsa_optimize(best_seq, t_alloc, verbose=True)
            if s6 < best_score:
                best_score = s6
                best_seq = f6.copy()
                print(f"  NEW BEST: {best_score:.6f}")

    # Phase 7: Final LP refinement
    if best_seq is not None and time.monotonic() < deadline - 15:
        t_alloc = min(60.0, deadline - time.monotonic() - 10.0)
        if t_alloc > 10:
            print(f"\n=== Phase 7: Final LP ({t_alloc:.0f}s) ===")
            f7, s7 = lp_optimize(best_seq, t_alloc, verbose=True)
            if s7 < best_score:
                best_score = s7
                best_seq = f7.copy()
                print(f"  NEW BEST: {best_score:.6f}")

    # Finalize
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'=' * 50}")
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'=' * 50}")
        return final_list

    # Fallback
    return [1.0] * 256
