import numpy as np
import time

def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def softmax_with_scale(theta, S):
    """Compute a = S * softmax(theta)."""
    theta_shifted = theta - np.max(theta)
    e = np.exp(theta_shifted)
    return S * e / np.sum(e)

def compute_gradient(a, w):
    """
    Compute grad_a[i] = 2 * sum_j w[i+j] * a[j]
    This is 2 * cross_correlate(w, a)
    """

    # Can be computed as: if we reverse a, it becomes a convolution

    # 
    # cross_corr = ifft(conj(fft(a, size)) * fft(w, size)) ...

    # For real signals: (f ⋆ g)[k] = sum_j f[j] * g[j+k]

    #

def gradient_descent_optimize(a_init, time_limit, tau_start=None, tau_end=None, lr_base=0.5, verbose=False):
    """
    Adam gradient descent on log-softmax parameterization.
    Minimizes soft_max(conv(a,a)) with annealed temperature.
    """
    n = len(a_init)
    S = np.sum(a_init)
    a_safe = np.clip(a_init, 1e-10, None)
    theta = np.log(a_safe / S)  # so that softmax(theta) * S ≈ a_init
    
    if tau_start is None:
        tau_start = 10.0 / n
    if tau_end is None:
        tau_end = 0.5 / n
    
    momentum1 = np.zeros(n)
    momentum2 = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    step_count = 0
    best_theta = theta.copy()
    best_score = local_score(a_init)
    best_a = a_init.copy()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        # Exponential annealing
        tau = tau_start * (tau_end / tau_start) ** elapsed_frac
        
        a = softmax_with_scale(theta, S)
        
        # Compute convolution
        conv = fft_convolve(a, a)
        
        # Compute softmax weights
        conv_shifted = conv - np.max(conv)
        exp_conv = np.exp(tau * conv_shifted)
        sum_exp = np.sum(exp_conv)
        w = exp_conv / sum_exp
        
        # Compute gradient w.r.t. a
        grad_a = compute_gradient(a, w)
        
        # Convert to gradient w.r.t. theta
        # grad_theta[j] = a[j] * (grad_a[j] - sum_i grad_a[i]*a[i]/S)
        inner = np.dot(grad_a, a) / S
        grad_theta = a * (grad_a - inner)
        
        # Adam update
        step_count += 1
        momentum1 = beta1 * momentum1 + (1.0 - beta1) * grad_theta
        momentum2 = beta2 * momentum2 + (1.0 - beta2) * (grad_theta ** 2)
        m1_hat = momentum1 / (1.0 - beta1 ** step_count)
        m2_hat = momentum2 / (1.0 - beta2 ** step_count)
        
        # Adaptive learning rate
        lr = lr_base / (1.0 + 0.05 * step_count ** 0.5)
        # Scale lr by S to account for parameterization
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)
        
        # Check score periodically
        if step_count % 50 == 0:
            a_check = softmax_with_scale(theta, S)
            score = local_score(a_check)
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_a = a_check.copy()
                best_theta = theta.copy()
            
            if verbose and step_count % 500 == 0:
                print(f"    GD step={step_count}, tau={tau:.5f}, score={score:.6f}, best={best_score:.6f}")
    
    # Final check
    a_final = softmax_with_scale(best_theta, S)
    final_score = local_score(a_final)
    if final_score < best_score:
        best_score = final_score
        best_a = a_final.copy()
    
    return best_a, best_score

def simulated_annealing(a_init, time_limit, rng, verbose=False):
    """
    Simulated annealing with peak-aware moves.
    """
    a = a_init.astype(np.float64).copy()
    n = len(a)
    S = np.sum(a)
    best = a.copy()
    best_score = local_score(a)
    current_score = best_score
    t0 = time.monotonic()
    
    # Temperature schedule
    T_start = 0.01
    T_end = 0.0001
    n_total_expected = int(time_limit / 0.002)  # rough estimate of iterations
    cooling_rate = (T_end / T_start) ** (1.0 / max(1, n_total_expected))
    T = T_start
    
    while time.monotonic() - t0 < time_limit:
        # Choose a move
        r = rng.random()
        candidate = a.copy()
        
        if r < 0.3:
            # Peak-targeted: reduce top contributors, boost low contributors
            conv = fft_convolve(a, a)
            peak_val = np.max(conv)
            # Find all near-peak positions
            peak_mask = conv >= peak_val * 0.95
            peak_indices = np.where(peak_mask)[0]
            
            contribs = np.zeros(n)
            for k in peak_indices[:10]:  # Limit to 10 peak positions
                for i in range(n):
                    j = k - i
                    if 0 <= j < n:
                        contribs[i] += a[i] * a[j]
            
            n_mod = max(3, n // 8)
            top = np.argsort(contribs)[-n_mod:]
            bottom = np.argsort(contribs)[:n_mod]
            
            noise_scale = 0.02 * T / T_start + 0.005
            candidate[top] *= np.exp(rng.normal(-noise_scale, noise_scale/3, size=n_mod))
            candidate[bottom] *= np.exp(rng.normal(noise_scale, noise_scale/3, size=n_mod))
            
        elif r < 0.5:
            # Block perturbation
            width = max(1, int(rng.integers(1, n // 4 + 1)))
            start = int(rng.integers(0, n - width))
            scale = rng.normal(0, 0.03 * T / T_start + 0.01)
            candidate[start:start+width] *= np.exp(rng.normal(scale, 0.02, size=width))
            
        elif r < 0.7:
            # Single coordinate perturbation
            i = int(rng.integers(0, n))
            candidate[i] *= np.exp(rng.normal(0, 0.1))
            
        elif r < 0.85:
            # Global scaling of a subset
            subset_size = max(4, n // 4)
            subset = rng.choice(n, size=subset_size, replace=False)
            candidate[subset] *= np.exp(rng.normal(0, 0.02, size=subset_size))
        else:
            # Smooth perturbation (low-pass)
            perturbation = rng.normal(0, 0.05, size=n)
            # Low-pass filter
            fa = np.fft.rfft(perturbation)
            fa[int(n//8):] = 0
            perturbation = np.fft.irfft(fa, n)
            candidate *= np.exp(perturbation)
        
        candidate = np.clip(candidate, 0.0, None)
        cand_sum = np.sum(candidate)
        if cand_sum < 1e-10:
            T *= cooling_rate
            continue
        candidate = candidate * (S / cand_sum)
        
        new_score = local_score(candidate)
        
        # Metropolis acceptance
        if new_score < current_score:
            a = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = candidate.copy()
        else:
            delta = new_score - current_score
            if delta < 1.0 and rng.random() < np.exp(-delta / max(T, 1e-12)):
                a = candidate
                current_score = new_score
        
        T *= cooling_rate
        
        if verbose and int((time.monotonic() - t0) * 100) % 1000 == 0:
            pass  # too verbose
    
    return best, best_score

"""
Multi-phase optimization for minimizing normalized peak autocorrelation of
non-negative step functions. Uses FFT-based gradient descent with softmax
parameterization, multi-restart strategy, peak-aware simulated annealing,
and coordinate refinement. Key insight: asymmetric sequences can achieve
lower peaks than symmetric ones since the convolution need not peak at
the center.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (sum_a ** 2))


def softmax_with_scale(theta, S):
    """Compute a = S * softmax(theta)."""
    theta_shifted = theta - np.max(theta)
    e = np.exp(theta_shifted)
    return S * e / np.sum(e)


def compute_grad_a(a, w):
    """
    Compute grad_a[i] = 2 * sum_j w[i+j] * a[j]
    Using FFT cross-correlation.
    """
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w, size)
    cross_corr = np.fft.irfft(fa * np.conj(fw), size)
    return 2.0 * cross_corr[:n]


def gradient_optimize(a_init, time_limit, tau_start=None, tau_end=None,
                      lr_base=0.5, beta1=0.8, beta2=0.999, verbose=False):
    """
    Adam gradient descent on log-softmax parameterization.
    Minimizes annealed soft-max of conv(a,a).
    """
    n = len(a_init)
    S = np.sum(a_init)
    a_safe = np.clip(a_init, 1e-10, None)
    theta = np.log(a_safe / S)

    if tau_start is None:
        tau_start = 12.0 / n
    if tau_end is None:
        tau_end = 0.3 / n

    momentum1 = np.zeros(n)
    momentum2 = np.zeros(n)
    step_count = 0
    best_theta = theta.copy()
    best_score = local_score(a_init)
    best_a = a_init.copy()
    t0 = time.monotonic()

    # Precompute FFT size
    conv_size = 1
    while conv_size < 2 * n - 1:
        conv_size <<= 1
    grad_size = 1
    while grad_size < 2 * n:
        grad_size <<= 1

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** elapsed_frac

        a = softmax_with_scale(theta, S)

        # Compute convolution
        fa = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fa * fa, conv_size)[:2 * n - 1]

        # Compute softmax weights (shifted for numerical stability)
        conv_max = np.max(conv)
        exp_conv = np.exp(tau * (conv - conv_max))
        sum_exp = np.sum(exp_conv)
        w = exp_conv / sum_exp

        # Compute gradient w.r.t. a
        fw = np.fft.rfft(w, grad_size)
        fa_full = np.fft.rfft(a, grad_size)
        cross_corr = np.fft.irfft(fa_full * np.conj(fw), grad_size)
        grad_a = 2.0 * cross_corr[:n]

        # Convert to gradient w.r.t. theta
        inner = np.dot(grad_a, a) / S
        grad_theta = a * (grad_a - inner)

        # Adam update
        step_count += 1
        momentum1 = beta1 * momentum1 + (1.0 - beta1) * grad_theta
        momentum2 = beta2 * momentum2 + (1.0 - beta2) * (grad_theta ** 2)
        m1_hat = momentum1 / (1.0 - beta1 ** step_count)
        m2_hat = momentum2 / (1.0 - beta2 ** step_count)

        lr = lr_base / (1.0 + 0.1 * np.sqrt(step_count))
        eps = 1e-8
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        # Periodic score check
        if step_count % 100 == 0:
            a_check = softmax_with_scale(theta, S)
            score = local_score(a_check)
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_a = a_check.copy()
                best_theta = theta.copy()

            if verbose and step_count % 1000 == 0:
                print(f"      GD step={step_count}, tau={tau:.5f}, "
                      f"score={score:.6f}, best={best_score:.6f}")

    a_final = softmax_with_scale(best_theta, S)
    final_score = local_score(a_final)
    if final_score < best_score:
        best_score = final_score
        best_a = a_final.copy()

    return best_a, best_score


def multi_tau_gradient(a_init, time_limit, verbose=False):
    """
    Try multiple tau schedules and keep the best result.
    """
    n = len(a_init)
    best_a = a_init.copy()
    best_score = local_score(a_init)
    t0 = time.monotonic()

    tau_configs = [
        (15.0 / n, 0.2 / n, 0.6),
        (8.0 / n, 0.5 / n, 0.5),
        (20.0 / n, 1.0 / n, 0.4),
        (5.0 / n, 0.3 / n, 0.7),
        (12.0 / n, 0.1 / n, 0.5),
    ]

    total_time = time_limit
    for i, (ts, te, lr) in enumerate(tau_configs):
        remaining = total_time - (time.monotonic() - t0)
        if remaining < 15:
            break
        alloc = min(remaining * 0.35, remaining)
        if i < len(tau_configs) - 1:
            alloc = min(alloc, total_time / len(tau_configs) * 1.5)

        print(f"    tau_config {i}: start={ts:.4f}, end={te:.4f}, lr={lr}, time={alloc:.0f}s")
        a_result, score = gradient_optimize(a_init, alloc, tau_start=ts, tau_end=te,
                                            lr_base=lr, verbose=verbose)
        if score < best_score:
            best_score = score
            best_a = a_result.copy()
            print(f"      *** Improved: {score:.6f} ***")

    return best_a, best_score


def sa_optimize(a_init, time_limit, rng, verbose=False):
    """
    Simulated annealing with peak-aware moves.
    """
    a = a_init.astype(np.float64).copy()
    n = len(a)
    S = np.sum(a)
    best = a.copy()
    best_score = local_score(a)
    current_score = best_score
    t0 = time.monotonic()

    T_start = 0.008
    T_end = 0.00005
    cooling_rate = (T_end / T_start) ** (1.0 / max(1, int(time_limit / 0.001)))
    T = T_start
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        r = rng.random()
        candidate = a.copy()

        if r < 0.35:
            # Peak-targeted move
            conv = fft_convolve(a, a)
            peak_val = np.max(conv)
            peak_threshold = peak_val * 0.92
            peak_mask = conv >= peak_threshold
            peak_indices = np.where(peak_mask)[0]

            contribs = np.zeros(n)
            for k in peak_indices[:8]:
                for i in range(n):
                    j = k - i
                    if 0 <= j < n:
                        contribs[i] += a[i] * a[j]

            n_mod = max(3, n // 10)
            top = np.argsort(contribs)[-n_mod:]
            bottom = np.argsort(contribs)[:n_mod]

            scale = 0.015 + 0.02 * (T / T_start)
            candidate[top] *= np.exp(rng.normal(-scale, scale * 0.4, size=n_mod))
            candidate[bottom] *= np.exp(rng.normal(scale, scale * 0.4, size=n_mod))

        elif r < 0.55:
            # Block perturbation
            width = max(1, int(rng.integers(1, n // 5 + 1)))
            start = int(rng.integers(0, n - width))
            base_scale = 0.01 + 0.03 * (T / T_start)
            candidate[start:start + width] *= np.exp(rng.normal(0, base_scale, size=width))

        elif r < 0.70:
            # Pair swap-like move (transfer mass)
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                transfer = a[i] * rng.uniform(0.02, 0.1)
                candidate[i] -= transfer
                candidate[j] += transfer

        elif r < 0.85:
            # Smooth low-frequency perturbation
            pert = rng.normal(0, 0.03, size=n)
            fp = np.fft.rfft(pert)
            cutoff = max(2, n // 10)
            fp[cutoff:] = 0
            pert = np.fft.irfft(fp, n)
            candidate *= np.exp(pert)

        else:
            # Global smooth scaling
            candidate *= np.exp(rng.normal(0, 0.01, size=n))

        candidate = np.clip(candidate, 0.0, None)
        cand_sum = np.sum(candidate)
        if cand_sum < 1e-10:
            T *= cooling_rate
            continue
        candidate = candidate * (S / cand_sum)

        new_score = local_score(candidate)
        iter_count += 1

        if new_score < current_score:
            a = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = candidate.copy()
        else:
            delta = new_score - current_score
            if delta < 0.5 and rng.random() < np.exp(-delta / max(T, 1e-12)):
                a = candidate
                current_score = new_score

        T *= cooling_rate

        if verbose and iter_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SA iter={iter_count}, T={T:.6f}, "
                  f"score={current_score:.6f}, best={best_score:.6f}, t={elapsed:.0f}s")

    return best, best_score


def make_chirp_init(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / (2 * n))
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multitone_init(n, rng, n_frequencies=5):
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for _ in range(n_frequencies):
        k = int(rng.integers(1, max(2, n // 4)))
        amp = rng.uniform(0.05, 0.25)
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_asymmetric_init(n, rng):
    """Create intentionally asymmetric initialization."""
    t = np.arange(n, dtype=np.float64)
    # Skewed envelope
    a = 1.0 + 0.3 * np.sin(np.pi * t / n) ** 2
    # Add chirp
    a += 0.2 * np.cos(2 * np.pi * 3 * t / n + 0.5 * t / n)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)
