def gaussian_sequence(n):
    x = np.arange(n)
    # Gaussian peak with control over width
    center = n/2
    width = 0.4 * (n/100)
    # This needs adjustment, might be better to use a relative measure.
    y = np.exp(-0.5*( (x-center)/width )**2 )
    y_sum = y.sum()
    if y_sum < 1e-10:
        y = np.ones(n)
    y = y * np.sqrt(2*n) / y_sum
    return y

def hanning_sequence(n):
    if n == 1:
        return np.array([1.0])
    else:
        x = np.arange(n)
        h = 0.5 * (1 + np.cos(2 * np.pi * x / float(n-1)))
        h = np.clip(h, 0, None)
        h_sum = h.sum()
        obj_val = np.sqrt(2*n)
        y = h * obj_val / h_sum
        return y
