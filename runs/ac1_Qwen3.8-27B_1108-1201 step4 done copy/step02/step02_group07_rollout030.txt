"""
Multi-strategy optimizer for minimizing peak aperiodic convolution of
non-negative sequences. Uses cutting-plane LP refinement, simulated
annealing polish, and multi-start across lengths with adaptive time.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * float(np.max(c)) / (S * S))


def normalize(a):
    """Normalize to sum=1, ensuring non-negativity."""
    a = np.clip(a, 0.0, None)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.ones(len(a)) / len(a)
    return a / S


def conv_row(f, k):
    """Return vector v such that v @ a = conv(f, a)[k]."""
    n = len(f)
    v = np.zeros(n)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    for j in range(lo, hi + 1):
        v[j] = f[k - j]
    return v


def solve_lp_cutting_plane(f, active_k, M):
    """
    Solve LP: minimize t
    s.t. for each k in active_k: 2*conv(f,a)[k] - t <= conv(f,f)[k]
         sum(a) = 1, a >= 0
    Returns (a, t) or None.
    """
    n = len(f)
    c_ff = fft_convolve(f, f)
    # Variables: a[0..n-1], t (index n)
    # Objective: minimize t -> c_obj has 1 at position n
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    A_ub = []
    b_ub = []
    for k in active_k:
        v = conv_row(f, k)
        # 2*conv(f,a)[k] - t <= c_ff[k]
        # 2*v[0]*a[0] + ... + 2*v[n-1]*a[n-1] - t <= c_ff[k]
        row = np.zeros(n + 1)
        row[:n] = 2.0 * v
        row[n] = -1.0
        A_ub.append(row)
        b_ub.append(c_ff[k])

    # Equality: sum(a) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c_obj, A_ub=np.array(A_ub), b_ub=np.array(b_ub),
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                      method='highs',
                      options={'time_limit': 10, 'presolve': True})
        if res.success:
            return res.x[:n], float(res.x[n])
    except Exception:
        pass
    return None, None


def cutting_plane_optimize(f0, time_limit, verbose=True):
    """Cutting-plane LP optimization for peak minimization."""
    n = len(f0)
    f = normalize(f0.copy())
    best_f = f.copy()
    best_peak = float(np.max(fft_convolve(f, f)))
    best_score = score_fast(f)
    t0 = time.monotonic()

    # Start with center constraint
    active_k = [n - 1]
    iteration = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 3.0:
            break

        # Find current peak locations
        c = fft_convolve(f, f)
        current_peak = float(np.max(c))
        # Add top-k peak locations to active set
        top_count = min(len(active_k) + 3, 20)
        top_idx = np.argsort(c)[-top_count:]
        for k in top_idx:
            ki = int(k)
            if ki not in active_k:
                active_k.append(ki)
        # Keep only top 30 to limit LP size
        if len(active_k) > 30:
            active_k = list(np.argsort(c)[-30:])

        M = current_peak
        a_new, t_new = solve_lp_cutting_plane(f, active_k, M)

        if a_new is None:
            break

        a_new = np.clip(a_new, 0.0, None)
        S = float(np.sum(a_new))
        if S < 1e-10:
            break
        a_new = a_new / S

        # Line search: blend with current
        best_blend = f.copy()
        best_blend_score = score_fast(f)
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.1]:
            candidate = normalize((1.0 - alpha) * f + alpha * a_new)
            s = score_fast(candidate)
            if s < best_blend_score:
                best_blend_score = s
                best_blend = candidate

        f = best_blend
        iteration += 1

        current_score = score_fast(f)
        current_peak = float(np.max(fft_convolve(f, f)))

        if current_score < best_score - 1e-10:
            best_score = current_score
            best_f = f.copy()
            stall = 0
        else:
            stall += 1

        if verbose and iteration % 20 == 0:
            print(f"    [CP {time.monotonic()-t0:6.1f}s] it={iteration:3d} peak={current_peak:.8f} score={current_score:.6f} best={best_score:.6f} stall={stall} |A|={len(active_k)}")

        if stall >= 30:
            # Perturb and restart active set
            rng = np.random.default_rng(iteration * 31 + 7)
            f = normalize(f + 0.01 * rng.standard_normal(n))
            active_k = [n - 1]
            stall = 0

        if iteration >= 500:
            break

    return best_f, best_score


def lp_direction_optimize(f0, time_limit, verbose=True):
    """Original LP direction method: find b maximizing sum(b) s.t. conv(f,b)<=M, b>=0."""
    n = len(f0)
    f = f0.astype(np.float64).copy()
    S = float(np.sum(f))
    if S < 1e-12:
        f = np.ones(n) / n
    else:
        f = f / S

    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 3.0:
            break

        c = fft_convolve(f, f)
        M = float(np.max(c))

        # Build LP: max sum(b) s.t. conv(f,b) <= M, b >= 0
        # conv(f,b)[k] = sum_j f[k-j]*b[j] -> row k of A is f reversed appropriately
        A_rows = []
        for k in range(2 * n - 1):
            row = np.zeros(n)
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                row[j] = f[k - j]
            A_rows.append(row)
        A = np.array(A_rows)

        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n,
                          method='highs',
                          options={'time_limit': 5, 'presolve': True})
            if not res.success:
                break
            b = res.x
        except Exception:
            break

        Sb = float(np.sum(b))
        if Sb < 1e-12:
            break
        b = b / Sb  # normalize b to sum=1

        # Line search
        best_s = np.inf
        best_t = 0.05
        for t_try in [0.1, 0.05, 0.02, 0.01]:
            f_cand = normalize((1.0 - t_try) * f + t_try * b)
            s_cand = score_fast(f_cand)
            if s_cand < best_s:
                best_s = s_cand
                best_t = t_try

        f = normalize((1.0 - best_t) * f + best_t * b)
        it += 1

        if best_s < best_score - 1e-10:
            best_score = best_s
            best_f = f.copy()
            stall = 0
        else:
            stall += 1

        if verbose and it % 30 == 0:
            print(f"    [LP {time.monotonic()-t0:6.1f}s] it={it:3d} score={best_s:.6f} best={best_score:.6f} stall={stall}")

        if stall >= 50:
            rng = np.random.default_rng(999 + it * 104729)
            f = normalize(f + 0.02 * rng.standard_normal(n))
            stall = 0

        if it >= 1000:
            break

    return best_f, best_score


def simulated_annealing(f0, time_limit, verbose=True):
    """Simulated annealing for fine-tuning the sequence."""
    n = len(f0)
    f = f0.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    temp = 0.001
    cooling = 0.99995
    current_score = best_score
    it = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break

        candidate = f.copy()
        r = rng.random()

        if r < 0.3:
            # Targeted reduction at peak
            c = fft_convolve(f, f)
            peak_idx = int(np.argmax(c))
            # Reduce elements that contribute most to the peak
            contribs = np.zeros(n)
            lo = max(0, peak_idx - (n - 1))
            hi = min(n - 1, peak_idx)
            for j in range(lo, hi + 1):
                contribs[j] = f[j] * f[peak_idx - j]
            top_k = max(2, n // 10)
            top_mask = np.zeros(n, dtype=bool)
            top_idx = np.argsort(contribs)[-top_k:]
            top_mask[top_idx] = True
            candidate[top_mask] *= np.exp(rng.normal(-0.02, 0.01, size=top_mask.sum()))
            # Slightly increase others
            other = ~top_mask
            if other.sum() > 0:
                inc_count = min(5, other.sum())
                inc_idx = rng.choice(np.where(other)[0], size=inc_count, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.02, 0.01, size=inc_count))
        elif r < 0.6:
            # Smooth perturbation
            width = max(1, int(rng.integers(1, max(2, n // 16))))
            start = int(rng.integers(0, max(1, n - width)))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.03, size=width))
        elif r < 0.8:
            # Global smooth perturbation
            candidate *= np.exp(rng.normal(0.0, 0.008, size=n))
        else:
            # Swap-based: move mass from high to low
            hi_idx = int(np.argmax(f))
            lo_idx = int(np.argmin(f))
            amount = f[hi_idx] * 0.05
            candidate[hi_idx] -= amount
            candidate[lo_idx] += amount

        candidate = np.clip(candidate, 0.0, 1000.0)
        new_score = score_fast(candidate)

        if new_score < current_score:
            f = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = candidate.copy()
        else:
            if rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-15)):
                f = candidate
                current_score = new_score

        temp *= cooling
        it += 1

        if verbose and it % 500 == 0:
            print(f"    [SA {time.monotonic()-t0:6.1f}s] it={it:5d} temp={temp:.6f} score={current_score:.6f} best={best_score:.6f}")

    return best_f, best_score


def make_initialization(n, init_type, rng):
    """Generate various initializations."""
    t = np.arange(n, dtype=np.float64)
    if init_type == 'uniform':
        return np.ones(n) / n
    elif init_type == 'hann':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
        a += 0.02 * rng.standard_normal(n)
        return normalize(np.clip(a, 0.01, None))
    elif init_type == 'raised_cosine':
        a = 0.5 * (1.0 + np.cos(np.pi * t / max(1, n - 1)))
        a += 0.01 * rng.standard_normal(n)
        return normalize(np.clip(a, 0.001, None))
    elif init_type == 'random_smooth':
        # Generate smooth random via low-freq Fourier
        freqs = np.arange(n)
       amps = rng.exponential(1.0, size=n)
        phases = rng.uniform(0, 2 * np.pi, size=n)
        a = np.zeros(n)
        for k in range(min(10, n)):
            a += amps[k] * np.cos(2 * np.pi * k * t / n + phases[k])
        a = np.abs(a) + 0.1
        return normalize(a)
    elif init_type == 'broad_peak':
        # High in middle, low at edges (opposite of what we want, but as init)
        center = n / 2.0
        a = 1.0 + 0.3 * np.exp(-((t - center) ** 2) / (2 * (n / 4) ** 2))
        a += 0.01 * rng.standard_normal(n)
        return normalize(np.clip(a, 0.01, None))
    elif init_type == 'edge_concentrated':
        # High at edges
        dist = np.minimum(t, n - 1 - t)
        a = 0.3 + 0.7 * np.exp(-dist / (n * 0.1))
        a += 0.01 * rng.standard_normal(n)
        return normalize(np.clip(a, 0.01, None))
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    1. Try multiple initializations and lengths with cutting-plane LP
    2. Refine with LP direction method
    3. Polish with simulated annealing
    4. Final refinement pass
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

                best_score = s
                best_seq = provided.copy()

    # Define search space: (length, init_type)
    tasks = []
    lengths = [128, 192, 256, 320, 384, 512, 768]
    inits = ['uniform', 'hann', 'random_smooth', 'edge_concentrated']

    if best_seq is not None:
        tasks.append((len(best_seq), 'provided'))

    for n in lengths:
        for init in inits:
            tasks.append((n, init))

    # Weight tasks: smaller n = faster, so give more attempts
    # But also try larger n since "larger sequences have better attack surface"
    for idx, (n, init) in enumerate(tasks):
        if time.monotonic() > deadline - 120:
            break

        elapsed = time.monotonic() - t_start
        remaining = deadline - time.monotonic() - 120
        if remaining < 10:
            break

        # Allocate time per task
        if n <= 256:
            t_alloc = min(40.0, remaining * 0.15)
        elif n <= 512:
            t_alloc = min(30.0, remaining * 0.12)
        else:
            t_alloc = min(25.0, remaining * 0.10)
        t_alloc = max(8.0, t_alloc)

        # Generate initialization
        if init == 'provided' and best_seq is not None:
            if len(best_seq) == n:
                f0 = normalize(best_seq.copy())
            else:
                x_old = np.linspace(0, 1, len(best_seq))

        print(f"\n[{elapsed:6.1f}s] Task {idx+1}/{len(tasks)}: n={n:4d} init={init:18s} alloc={t_alloc:.0f}s")

        # Phase A: Cutting plane
        f_cp, s_cp = cutting_plane_optimize(f0, t_alloc * 0.5, verbose=False)

        # Phase B: LP direction (if time remains)
        t_rem = t_alloc * 0.5
        if time.monotonic() < deadline - 120 and t_rem > 5:
            f_lp, s_lp = lp_direction_optimize(f_cp, t_rem, verbose=False)
            if s_lp < s_cp:
                f_cp, s_cp = f_lp, s_lp

        if s_cp < best_score:
            best_score = s_cp
            best_seq = f_cp.copy()
            print(f"  *** NEW BEST: n={n}, score={best_score:.6f} ***")
        else:
            print(f"  score={s_cp:.6f} (best={best_score:.6f})")

    print(f"After main search: best_score={best_score:.6f}")
    print(f"{'='*50}")

    # Phase 2: Simulated annealing polish

            f_fin, s_fin = cutting_plane_optimize(best_seq, cp_time, verbose=True)

    return [1.0] * 256
