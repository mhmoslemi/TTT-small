import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def compute_objective(a):
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    c = fast_convolve(a)
    return 2.0 * n * c.max() / (s * s)

def active_set_gradient(a, p=10):
    """Compute gradient of max_k C[k] using top-p active set."""
    c = fast_convolve(a)
    n = len(a)
    # Get top p indices
    top = np.argsort(c)[-p:]
    # For each active k, gradient is 2*a[k-i] for valid i
    # Equivalently, gradient[i] = 2 * sum_{k in top, k-i valid} a[k-i]
    grad = np.zeros(n)
    for k in top:
        k = int(k)
        # grad[i] += 2*a[k-i] for i where 0<=k-i<n, i.e., k-n+1 <= i <= k
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        # a[k-i] for i in [i_start, i_end]: a[k-i_end] ... a[k-i_start]
        grad[i_start:i_end+1] += 2.0 * a[k-i_end:k-i_start+1][::-1]
    grad /= p
    return grad, c.max()

def simplex_project(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    mu = np.sort(v)[::-1]
    rho = np.arange(1, n+1)
    cond = mu - (1.0)/rho > 0
    rho_hat = rho[cond][-1]
    theta = (mu[:int(rho_hat)].sum() - 1.0) / rho_hat
    return np.maximum(v - theta, 0)

def lp_direction(f, M, n, timeout=20):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)[k] <= M for all k."""
    # Build constraint matrix
    # (f*g)[k] = sum_i f[k-i] * g[i] for valid i
    # Row k: coefficients are f[k-i] for i where 0<=k-i<n
    rows = []
    cols = []
    vals = []
    for i in range(n):
        for k in range(i, i + n):  # k = i + j for j in 0..n-1
            j = k - i
            if 0 <= j < n:
                rows.append(k)
                cols.append(i)
                vals.append(f[j])
    
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    c_obj = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = optimize.linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def optimize_lp(f0, time_limit, verbose=False):
    """LP-guided alternating optimization."""
    f = np.maximum(f0, 0.0).copy()
    n = len(f)
    f_sum = f.sum()
    if f_sum < 1e-12:
        f = np.ones(n) / n
    else:
        f /= f_sum
    
    c = fast_convolve(f)
    M = c.max()
    best_obj = 2.0 * n * M  # with sum=1
    best_f = f.copy()
    
    t = 1.0
    patience = 0
    t0 = time.monotonic()
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        g = lp_direction(f, M, n)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g /= g_sum
        
        if np.max(np.abs(g - f)) < 1e-14:
            break
        
        improved = False
        for t_try in [t, 0.5*t, 0.25*t, 0.1*t, 0.05*t]:
            f_new = (1 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0)
            s = f_new.sum()
            if s < 1e-12:

            f_new /= s
            c_new = fast_convolve(f_new)
            obj = 2.0 * n * c_new.max()
            if obj < best_obj - 1e-14:
                f = f_new
                M = c_new.max()
                best_obj = obj
                best_f = f.copy()
                patience = 0
                t = min(1.0, t * 1.15)

        if verbose and iteration % 20 == 0:

    return best_f, best_obj

def active_set_descent(f0, time_limit, p=10, eta0=0.5, verbose=False):

    n = len(f)
    f /= f.sum()

    best_obj = 2.0 * n * c.max()

    patience = 0
    t0 = time.monotonic()

        grad, cmax = active_set_gradient(f, p)
        # Project gradient onto simplex tangent
        grad = grad - grad.mean()
        
        # Multiplicative update
        f_new = f * np.exp(-eta * grad)
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-12:
            f_new = np.ones(n) / n

            f = f_new

        else:

            eta *= 0.8

def gen_inits(n, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uni'] = np.ones(n) / n

        if s < 1e-12:

        f_pert /= s

            best_f = f.copy()
        elif rng.random() < np.exp(-(obj_pert - best_obj) / T):
            f = f_pert
        
        # Cool

    return best_f, best_obj

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_obj_normalized(a):
    """Compute 2*n*max(conv) assuming sum(a)=1."""
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * c.max()


def make_constraint_matrix(n):
    """Build sparse constraint matrix for cross-convolution (f*g)[k] = sum_i f[k-i]*g[i]."""
    rows = []
    cols = []
    for i in range(n):
        # For variable g[i], it appears in rows k = i..i+n-1 with coefficient f[k-i]
        rows.extend(range(i, i + n))
        cols.extend([i] * n)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp(f, M, n, rows, cols, timeout=10):
    """Solve max sum(g) s.t. g>=0, (f*g)[k] <= M for all k."""
    vals = np.repeat(f, n)  # For each column i, the values are f[0],f[1],...,f[n-1] at rows i..i+n-1
    # Actually need to be more careful: row k, col i gets f[k-i]
    # Let me rebuild properly
    vals2 = []
    for i in range(n):
        # column i: rows i to i+n-1, values f[0] to f[n-1]
        vals2.extend(f.tolist())
    vals2 = np.array(vals2)
    A = sparse.csc_matrix((vals2, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def active_set_grad(f, p):
    """Compute gradient of max_k C[k] using top-p active set. Returns (grad, cmax)."""
    n = len(f)
    c = fast_convolve(f)
    cmax = c.max()
    top = np.argsort(c)[-p:]
    grad = np.zeros(n)
    for k in top:
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_end >= i_start:
            # grad[i] += 2*f[k-i] for i in [i_start, i_end]
            # f[k-i] for i=i_start..i_end is f[k-i_end]..f[k-i_start]
            grad[i_start:i_end + 1] += 2.0 * f[k - i_end:k - i_start + 1][::-1]
    grad /= float(p)
    return grad, cmax

def active_set_descent(f0, time_limit, verbose=False):

        sn = f_new.sum()
        if sn < 1e-12:

        else:
            f_new /= sn

        obj_new = compute_obj_normalized(f_new)

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new

def gen_inits(n, rng):
    """Generate diverse initializations summing to 1."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uni'] = np.ones(n) / n

    for sf in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'g{int(sf * 100)}'] = g

    inits['rnd'] = rng.dirichlet(np.ones(n) * 0.4)

    alt = np.zeros(n)
    alt[::2] = 1.0
    inits['alt'] = alt / alt.sum()

    blk = np.zeros(n)
    w = max(1, n // 3)
    blk[:w] = 1.0
    blk[-w:] = 1.0
    inits['blk'] = blk / blk.sum()

    cb = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits['cos'] = cb / cb.sum()

    # Multi-bump
    mb = np.zeros(n)
    for frac in [0.2, 0.5, 0.8]:
        center = int(frac * n)
        width = max(1, n // 20)
        mb[center - width:center + width + 1] += 1.0
    mb = np.maximum(mb, 0)
    inits['multi'] = mb / mb.sum()

    return inits


def anneal_polish(f0, time_limit, verbose=False):
    """Simulated annealing: random perturbations with cooling schedule."""
    n = len(f0)
    f = np.maximum(f0, 0.0).copy()
    f /= f.sum()

    best_obj = compute_obj_normalized(f)
    best_f = f.copy()

    rng = np.random.default_rng(12345)
    T = 0.05
    T_min = 0.0005
    t0 = time.monotonic()
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        iteration += 1
        if time.monotonic() - t0 > time_limit - 2:
            break

        # Perturbation magnitude scales with T
        scale = T * 3.0
        n_perturb = max(1, int(n * scale))
        f_pert = f.copy()
        idx = rng.integers(0, n, size=min(n_perturb, n))
        f_pert[idx] += rng.normal(0, max(T, 0.01), size=len(idx))
        f_pert = np.maximum(f_pert, 0)
        s = f_pert.sum()
        if s < 1e-12:
            continue
        f_pert /= s

        obj_pert = compute_obj_normalized(f_pert)

        # Metropolis acceptance
        if obj_pert < best_obj - 1e-14:
            f = f_pert
            best_obj = obj_pert
            best_f = f.copy()
        elif rng.random() < np.exp(-(obj_pert - best_obj) / max(T, 1e-8)):
            f = f_pert

        # Cool
        T = max(T_min, T * 0.9995)

        if verbose and iteration % 500 == 0:
            print(f"    Anneal iter {iteration}, obj={best_obj:.8f}, T={T:.6f}")

    return best_f, best_obj

                if np.isfinite(sc_new) and sc_new < best_score:
                    best_score = sc_new
                    best_seq = f_new.copy()
                    print(f"  Cross-length n={n_try}: {sc_new:.6f}")

    print(f"\nFINAL: n={len(best_seq) if best_seq is not None else 0}, score={best_score:.6f}")
    return [float(x) for x in best_seq]
