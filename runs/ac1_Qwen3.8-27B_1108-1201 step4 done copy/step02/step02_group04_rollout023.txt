Minimizes 2*n*max(auto-conv)/(sum)^2 using:
- Self-convolution linearized LP (novel)

- Multi-scale log-sum-exp subgradient polishing
- Diverse initializations, multiple lengths

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time

def fast_convolve(a):
    """Auto-convolution via FFT."""
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def make_conv_indices(n):
    """Row/col indices for convolution matrix (2n-1) x n."""
    rows = np.repeat(np.arange(n), n)
    cols = np.tile(np.arange(n), n)
    # For column i, rows are i, i+1, ..., i+n-1
    # So rows[i*n + j] = i + j
    rows = (np.arange(n).reshape(-1, 1) + np.arange(n).reshape(1, -1)).ravel()
    cols = np.tile(np.arange(n), n)
    return rows.astype(np.int32), cols.astype(np.int32)

def solve_conv_lp(weight, rhs, rows, cols, n, lb, ub, timeout=8):
    """max sum(x) s.t. W@x <= rhs, lb<=x<=ub where W[k,i]=weight[k-i] if valid."""
    vals = np.tile(weight, n)
    W = sparse.csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    c = -np.ones(n)
    bounds = list(zip(lb, ub))  # Hmm, this might be slow for large n

                           bounds=[(lb[i], ub[i]) for i in range(n)],
                           method='highs', options={'disp': False, 'time_limit': timeout})

                break
    
    return best_f, best_obj

def logsumexp_polish(f0, time_limit):
    """Polish using log-sum-exp objective with increasing concentration."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f /= f.sum()
    
    c = fast_convolve(f)
    best_obj = 2.0 * n * c.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    
    schedules = [
        (20, 0.5, 100),   # (beta, eta0, iters)
        (50, 0.3, 100),
        (100, 0.2, 100),
        (200, 0.1, 80),
        (500, 0.05, 60),
    ]
    
    for beta, eta0, n_iter in schedules:
        eta = eta0
        for _ in range(n_iter):
            if time.monotonic() - t0 > time_limit:
                break
            
            c = fast_convolve(f)
            c_max = c.max()
            # Softmax weights
            scaled = beta * (c - c_max)
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum < 1e-300:
                w = np.zeros(2*n-1)
                w[int(np.argmax(c))] = 1.0
            else:
                w /= w_sum
            
            # Subgradient via FFT
            sg = compute_sg_fft(f, w)
            
            # Mirror descent
            f_log = np.log(f + 1e-300)
            f_log -= eta * sg
            f_log -= f_log.max()
            f_new = np.exp(f_log)
            s = f_new.sum()
            if s < 1e-12:
                break
            f_new /= s
            
            obj = 2.0 * n * fast_convolve(f_new).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_f = f_new.copy()
                f = f_new
                eta = min(eta * 1.02, eta0)
            else:
                eta *= 0.8
                if eta < eta0 * 0.05:
                    break
        
        if time.monotonic() - t0 > time_limit:
            break
    
    return best_f, best_obj

def gen_inits(n, rng):
    """Generate diverse initial probability vectors of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uni'] = np.ones(n) / n
    
    for sf in [0.1, 0.15, 0.25, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5*n) / (sf*n))**2)
        g /= g.sum()
        inits[f'g{int(sf*100)}'] = g
    
    r = np.arange(1, n+1, dtype=np.float64)
    inits['ramp'] = r / r.sum()
    
    w_b = max(1, n // 4)
    bi = np.zeros(n)
    bi[:w_b] = 1.0
    bi[-w_b:] = 1.0
    inits['bi'] = bi / bi.sum()
    
    cb = 0.5 * (1 + np.cos(2*np.pi*x/n))
    inits['cos'] = cb / cb.sum()
    
    rr = rng.dirichlet(np.ones(n) * 0.5)
    inits['rnd'] = rr
    
    # Alternating pattern
    alt = np.ones(n)
    alt[1::2] = 0.5
    inits['alt'] = alt / alt.sum()
    
    # Triangle
    tri = 1.0 - 2.0 * np.abs(x/(n-1) - 0.5)
    inits['tri'] = tri / tri.sum()
    
    return inits
