import numpy as np

def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

import numpy as np

from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def build_A(f):
    """Build matrix A where A@b = conv(f,b) (full length 2n-1)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp(f, M, timeout=5.0):
    """Solve: max sum(b) s.t. A@b <= M*1, b >= 0. Returns b or None."""
    n = len(f)
    A = build_A(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def gradient_polish(f_init, time_limit, verbose=True):
    """Subgradient descent with momentum using analytic gradient of max(conv)."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    S = float(np.sum(f))
    f = f * (float(n) / S)
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    momentum = np.zeros(n)
    beta_m = 0.9
    lr = 0.003
    it = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break

        c = fft_convolve(f, f)
        top_k = min(5, len(c))
        top_idx = np.argsort(c)[-top_k:]

        g = np.zeros(n)
        for k in top_idx:
            ki = int(k)
            # dc[k]/da[j] = 2*f[ki-j] if 0 <= ki-j < n
            lo = max(0, ki - (n - 1))
            hi = min(n - 1, ki)
            for j in range(lo, hi + 1):
                g[j] += 2.0 * f[ki - j]
        g /= float(top_k)
        g -= np.mean(g)

        momentum = beta_m * momentum + (1.0 - beta_m) * g
        f = f - lr * momentum
        f = np.clip(f, 0.01, None)
        f = f * (float(n) / float(np.sum(f)))

        it += 1
        if it % 300 == 0:
            s = score_fast(f)
            if s < best_score:
                best_score = s
                best_f = f.copy()
            if verbose:
                print(f"    [grad {time.monotonic()-t0:6.1f}s] it={it} score={s:.6f} best={best_score:.6f}")

    s = score_fast(f)
    if s < best_score:
        best_score = s
        best_f = f.copy()
    return best_f, best_score


def local_search(f_init, time_limit, verbose=True):
    """Perturbation-based local search for fine-tuning."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()

    temp = 0.002
    cooling = 0.9998

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break

        candidate = f.copy()
        r = rng.random()

        if r < 0.4:
            peak_idx = int(np.argmax(fft_convolve(f, f)))
            contribs = np.zeros(n)
            for j in range(n):
                idx = peak_idx - j
                if 0 <= idx < n:
                    contribs[j] = f[j] * f[idx]
            top_k = max(3, n // 12)
            top_mask = contribs >= np.sort(contribs)[-top_k]
            candidate[top_mask] *= np.exp(rng.normal(-0.03, 0.02, size=top_mask.sum()))
            other = ~top_mask
            if other.sum() > 0:
                inc_mask = np.zeros(n, dtype=bool)
                idxs = np.where(other)[0]
                chosen = rng.choice(idxs, size=min(6, len(idxs)), replace=False)
                inc_mask[chosen] = True
                candidate[inc_mask] *= np.exp(rng.normal(0.03, 0.02, size=inc_mask.sum()))
        elif r < 0.7:
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, max(1, n - width)))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.04, size=width))
        else:
            candidate *= np.exp(rng.normal(0.0, 0.015, size=n))

        candidate = np.clip(candidate, 0.01, 1000.0)
        new_score = score_fast(candidate)

        if new_score < best_score:
            best_score = new_score
            best_f = candidate.copy()
            f = candidate
        elif rng.random() < np.exp(-(new_score - score_fast(f)) / max(temp, 1e-15)):
            f = candidate

        temp *= cooling

    return best_f, best_score
