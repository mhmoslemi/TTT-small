Multi-phase optimization for minimizing normalized peak autocorrelation
of non-negative step functions. Uses improved LP-based refinement, 
multi-lag subgradient with momentum, and targeted coordinate descent.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))

def normalize_seq(a, S=None):
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-15:
        return np.full_like(a, S / len(a))
    return a * (S / s)

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    # Use the standard algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - S))[0][-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_lp_matrix(a):
    n = len(a)
    # A[k, j] = contribution of g[j] to conv(a,g)[k]
    # conv(a,g)[k] = sum_j a[k-j] * g[j] (with bounds)
    # So A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # For each j, A[k, j] = a[k-j] for k = j, j+1, ..., j+n-1
        # (where k-j ranges from 0 to n-1)
        A[j:j+n, j] = a
    return A

def lp_solve(a, M):
    n = len(a)
    A = build_lp_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 10})
    if res.status == 0:
        return res.x
    return np.zeros(n)

def lp_refine(a, time_limit, verbose=False):
    n = len(a)

    a = normalize_seq(a, S)

def subgradient_step(a, lr, momentum, beta, K_frac=0.25):
    """Multi-lag subgradient step with momentum."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    conv = fft_convolve(a, a)
    max_conv = np.max(conv)
    
    # Find active lags (top K)
    K = max(1, int(K_frac * len(conv)))
    threshold = np.sort(conv)[::-1][min(K, len(conv)-1)]
    active = np.where(conv > threshold * 0.99)[0]
    weights = conv[active] - threshold * 0.99 + 1e-10
    weights = weights / np.sum(weights)
    
    # Compute weighted subgradient
    sg = np.zeros(n)
    for w, k in zip(weights, active):
        for i in range(n):
            j1 = k - i
            j2 = i - k
            if 0 <= j1 < n:
                sg[i] += w * a[j1]
            if 0 <= j2 < n:
                sg[i] += w * a[j2]
    
    # Momentum update
    momentum = beta * momentum - lr * sg
    a_new = project_simplex(a + momentum, S)
    
    return a_new, momentum, max_conv

def coordinate_descent(a, time_limit, verbose=False):
    """Targeted coordinate descent on the peak lag."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv_max = np.max(fft_convolve(a, a))
    rng = np.random.default_rng(42)
    
    stall = 0
    iteration = 0
    
    while time.monotonic() - t0 < time_limit and stall < 200:
        conv = fft_convolve(a, a)
        peak_idx = int(np.argmax(conv))
        current_peak = conv[peak_idx]
        
        # Find top contributors to the peak
        contribs = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]
            else:
                # Also check the "other" contribution
                j2 = i - peak_idx
                if 0 <= j2 < n:
                    contribs[i] += a[i] * a[j2]
        
        # Try moves: transfer mass from high-contributing elements to low ones
        n_tries = 20
        improved = False
        
        for _ in range(n_tries):
            candidate = a.copy()
            
            # Pick a random element to reduce (weighted by contribution)
            probs = contribs / (np.sum(contribs) + 1e-10)
            src = rng.choice(n, p=probs)
            
            # Pick a random element to increase (uniform or weighted by low contribution)
            low_mask = contribs < np.median(contribs)
            low_idx = np.where(low_mask)[0]
            if len(low_idx) == 0:
                low_idx = np.arange(n)
            dst = rng.choice(low_idx)
            
            if src == dst:
                continue
            
            # Transfer some mass
            frac = rng.uniform(0.01, 0.1)
            amount = candidate[src] * frac
            candidate[src] -= amount
            candidate[dst] += amount
            candidate = np.maximum(candidate, 0.0)
            candidate = normalize_seq(candidate, S)
            
            new_conv = fft_convolve(candidate, candidate)
            new_peak = np.max(new_conv)
            
            if new_peak < best_conv_max:
                best_conv_max = new_peak
                best_a = candidate.copy()
                a = candidate
                improved = True
                stall = 0
                break
            elif new_peak < current_peak * (1 - 0.001):
                a = candidate  # Accept small improvements to escape local minima
        
        if not improved:
            stall += 1
        
        iteration += 1
        if verbose and iteration % 100 == 0:
            score = 2.0 * n * best_conv_max / (S * S)
            print(f"    CD iter {iteration}: score={score:.6f}, peak_idx={peak_idx}, stall={stall}")
    
    return best_a

def make_initialization(n, seed=0):

"""
Multi-phase optimization for minimizing normalized peak autocorrelation
of non-negative step functions. Uses improved LP with adaptive peak-targeted
constraints, backtracking line search, multi-lag subgradient with momentum,
and targeted coordinate descent. Key innovation: the LP constraint tightens
disproportionally at the current peak lag to force faster reduction.

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the score 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))


def normalize_seq(a, S=None):
    """Normalize sequence to have a given sum."""
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-15:
        return np.full_like(a, float(S) / len(a))
    return a * (S / s)


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - S)
    rho_indices = np.nonzero(cond)[0]
    if len(rho_indices) == 0:
        return np.full(n, S / n)
    rho = rho_indices[-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(a):
    """Build matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_solve(a, b_ub):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= b_ub[k], g >= 0."""
    n = len(a)
    A = build_lp_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 5})
    if res.status == 0:
        return res.x
    return np.zeros(n)

def compute_multi_lag_subgradient(a, K_frac=0.3):
    """Compute weighted subgradient using top K lags."""
    n = len(a)
    conv = fft_convolve(a, a)
    max_conv = np.max(conv)
    K = max(3, int(K_frac * (2 * n - 1)))
    top_indices = np.argsort(conv)[::-1][:K]
    weights = conv[top_indices]
    weights = weights / (np.sum(weights) + 1e-15)

    sg = np.zeros(n)
    for w, k in zip(weights, top_indices):
        # Subgradient for lag k: sg[m] += 2*w*a[k-m] (if 0<=k-m<n)
        k_int = int(k)
        m_min = max(0, k_int - n + 1)
        m_max = min(n - 1, k_int)
        if m_min <= m_max:
            indices = np.arange(m_min, m_max + 1)
            sg[indices] += 2.0 * w * a[k_int - indices]

    return sg, max_conv

    a = normalize_seq(a, S).astype(np.float64)
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv_max = np.max(fft_convolve(a, a))

    momentum = np.zeros(n)
    beta = 0.85
    lr = 2.0 / n
    K_frac = 0.3
    t = 1
    stall = 0
    patience = 500

    while time.monotonic() - t0 < time_limit:
        if t > 100000:
            break

        sg, _ = compute_multi_lag_subgradient(a, K_frac)

        # Momentum update
        momentum = beta * momentum - lr * sg
        a_new = project_simplex(a + momentum, S)

        new_conv_max = np.max(fft_convolve(a_new, a_new))
        curr_conv_max = np.max(fft_convolve(a, a))

        if new_conv_max < best_conv_max:
            best_conv_max = new_conv_max
            best_a = a_new.copy()
            a = a_new
            stall = 0
        elif new_conv_max < curr_conv_max:
            a = a_new
            stall += 1
        else:
            stall += 1
            lr *= 0.92
            momentum *= 0.5

        if stall > patience:
            lr *= 0.5
            K_frac = min(1.0, K_frac * 2.0)
            beta = max(0.5, beta * 0.95)
            stall = 0
            # Small random perturbation to escape
            rng = np.random.default_rng(t)
            a = normalize_seq(a * np.exp(rng.normal(0, 0.01, n)), S)
            a = np.maximum(a, 0.0)
            a = normalize_seq(a, S)

        # Adaptive K: focus more on peak over time
        if t % 500 == 0:
            K_frac = max(0.05, K_frac * 0.95)

        t += 1
        if verbose and t % 500 == 0:
            score = 2.0 * n * best_conv_max / (S * S)
            print(f"    SG step {t}: score={score:.6f}, lr={lr:.5f}, K_frac={K_frac:.3f}")

    return best_a


def coordinate_descent(a, time_limit, verbose=False):
    """Targeted coordinate descent focusing on the peak lag."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S).astype(np.float64)
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv_max = np.max(fft_convolve(a, a))
    rng = np.random.default_rng(12345)

    stall = 0
    iteration = 0
    temp = 0.01  # "temperature" for accepting small worsening moves

    while time.monotonic() - t0 < time_limit:
        if iteration > 50000:
            break

        conv = fft_convolve(a, a)
        peak_idx = int(np.argmax(conv))
        current_peak = conv[peak_idx]

        n_tries = 30
        improved = False

        for _ in range(n_tries):
            candidate = a.copy()

            # Strategy 1: Reduce element contributing most to peak
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]

            if np.sum(contribs) > 0:
                probs = contribs / np.sum(contribs)
                src = rng.choice(n, p=probs)

                # Find low-contribution targets
                low_threshold = np.percentile(contribs, 30)
                low_idx = np.where(contribs < low_threshold)[0]
                if len(low_idx) == 0:
                    low_idx = np.arange(n)
                dst = rng.choice(low_idx)

                if src != dst:
                    frac = rng.uniform(0.02, 0.15)
                    amount = candidate[src] * frac
                    candidate[src] -= amount
                    candidate[dst] += amount

            # Strategy 2: Pair swap (reduce product a[i]*a[j] for i+j=peak)
            elif rng.random() < 0.5:
                valid_i = []
                for i in range(n):
                    j = peak_idx - i
                    if 0 <= j < n and i < j:
                        valid_i.append(i)
                if len(valid_i) > 0:
                    i = valid_i[rng.integers(len(valid_i))]
                    j = peak_idx - i
                    # Transfer mass from larger to smaller in the pair
                    if candidate[i] > candidate[j]:
                        move = (candidate[i] - candidate[j]) * rng.uniform(0.1, 0.4)
                        candidate[i] -= move
                        candidate[j] += move
                    else:
                        move = (candidate[j] - candidate[i]) * rng.uniform(0.1, 0.4)
                        candidate[j] -= move
                        candidate[i] += move

            candidate = np.maximum(candidate, 0.0)
            candidate = normalize_seq(candidate, S)

            new_conv = fft_convolve(candidate, candidate)
            new_peak = np.max(new_conv)

            if new_peak < best_conv_max:
                best_conv_max = new_peak
                best_a = candidate.copy()
                a = candidate
                improved = True
                stall = 0
                temp = max(0.001, temp * 0.98)
                break
            elif new_peak < current_peak and rng.random() < temp:
                a = candidate  # Simulated annealing acceptance

        if not improved:
            stall += 1
            if stall > 50:
                temp = max(0.001, temp * 0.9)

        iteration += 1
        if verbose and iteration % 200 == 0:
            score = 2.0 * n * best_conv_max / (S * S)
            print(f"    CD iter {iteration}: score={score:.6f}, peak_idx={peak_idx}, stall={stall}")

        if stall > 500:
            break

    return best_a


def make_structured_init(n, seed=0):
    """Create structured initial sequences that tend to have flat autocorrelation."""
    rng = np.random.default_rng(seed)
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    # Raised cosine (Hann window)
    a_hann = 0.5 + 0.5 * np.cos(2 * np.pi * (t - center) / n)
    a_hann = np.maximum(a_hann, 0.001)

    # Blackman window
    a_blackman = (0.42 - 0.5 * np.cos(2 * np.pi * t / (n - 1))
                  + 0.08 * np.cos(4 * np.pi * t / (n - 1)))
    a_blackman = np.maximum(a_blackman, 0.001)

    # Trapezoidal
    width = n * (0.3 + 0.4 * rng.random())
    a_trapezoid = np.zeros(n)
    start = int((n - width) / 2)
    end = int((n + width) / 2)
    a_trapezoid[start:end] = 1.0
    ramp = max(1, int(width * 0.3))
    for i in range(ramp):
        frac = i / ramp
        if start - 1 - i >= 0:
            a_trapezoid[start - 1 - i] = frac
        if end + i < n:
            a_trapezoid[end + i] = frac
    a_trapezoid = np.maximum(a_trapezoid, 0.001)

    # Uniform with noise
    a_uniform = np.ones(n) * (1.0 + 0.1 * rng.standard_normal(n))
    a_uniform = np.maximum(a_uniform, 0.01)

    # Return the best-looking one
    candidates = [a_hann, a_blackman, a_trapezoid, a_uniform]
    best = min(candidates, key=lambda x: local_score(x))
    return best


def interpolate_sequence(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    1. LP-based refinement with adaptive peak-targeted constraints
    2. Multi-lag subgradient with momentum
    3. Targeted coordinate descent
    4. Final LP polish
    Returns the best sequence found.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 25.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    def update_best(a):
        nonlocal best_score, best_seq
        a_clip = np.clip(a, 0.0, 1000.0)
        a_clip = np.maximum(a_clip, 1e-10)
        s = local_score(a_clip)
        if s < best_score:
            best_score = s
            best_seq = a_clip.copy()
            print(f"    *** NEW BEST: score={s:.6f}, n={len(a_clip)} ***")
            return True
        return False

    # Get provided sequence
    provided = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        provided = np.maximum(provided, 0.01)
        print(f"Provided: n={len(provided)}, score={local_score(provided):.6f}")

    # Build candidate pool
    candidates = []

    if provided is not None:
        candidates.append(('provided', provided.copy()))
        # Interpolated versions
        for n_try in [len(provided) - 64, len(provided) + 64, len(provided) + 128]:
            if n_try >= 128 and n_try <= 2048:
                a_interp = interpolate_sequence(provided, n_try)
                a_interp = np.maximum(a_interp, 0.01)
                candidates.append((f'interp_n{n_try}', a_interp))

    # Structured initializations
    for n in [256, 384, 512, 640, 768]:
        a_struct = make_structured_init(n, seed=seed + n)
        candidates.append((f'struct_n{n}', a_struct))

    # Evaluate all candidates
    print("\n=== Initial Candidates ===")
    for name, cand in candidates:
        s = local_score(cand)
        print(f"  {name}: n={len(cand)}, score={s:.6f}")
        update_best(cand)

    # Sort candidates by score
    scored = [(local_score(c), name, c) for name, c in candidates]
    scored.sort(key=lambda x: x[0])

    # Phase 1: LP refinement on top candidates
    print("\n=== Phase 1: LP Refinement (v2) ===")
    phase1_budget = min(500.0, (deadline - time.monotonic()) * 0.35)
    t_phase1_end = time.monotonic() + phase1_budget

    top_k = min(3, len(scored))
    for idx in range(top_k):
        if time.monotonic() > t_phase1_end - 60:
            break
        s, name, cand = scored[idx]
        n = len(cand)
        if n > 1200:
            continue
        t_per = min(150.0, t_phase1_end - time.monotonic())
        if t_per < 15:
            break
        print(f"  LP on '{name}' (n={n}, init_score={s:.6f}), budget={t_per:.0f}s")
        result = lp_refine_v2(cand, t_per, verbose=True)
        update_best(result)
        # Update scored list
        scored[idx] = (local_score(result), name, result)

    print(f"  Phase 1 best: {best_score:.6f}")

    # Phase 2: Subgradient with momentum
    print("\n=== Phase 2: Subgradient with Momentum ===")
    if best_seq is not None:
        phase2_budget = min(250.0, (deadline - time.monotonic()) * 0.25)
        if phase2_budget > 20:
            n = len(best_seq)
            print(f"  Subgradient (n={n}), budget={phase2_budget:.0f}s")
            result = subgradient_optimize(best_seq, phase2_budget, verbose=True)
            update_best(result)

    print(f"  Phase 2 best: {best_score:.6f}")

    # Phase 3: Coordinate descent
    print("\n=== Phase 3: Coordinate Descent ===")
    if best_seq is not None:
        phase3_budget = min(200.0, (deadline - time.monotonic()) * 0.25)
        if phase3_budget > 15:
            n = len(best_seq)
            print(f"  Coord descent (n={n}), budget={phase3_budget:.0f}s")
            result = coordinate_descent(best_seq, phase3_budget, verbose=True)
            update_best(result)

    print(f"  Phase 3 best: {best_score:.6f}")

    # Phase 4: Final LP polish
    print("\n=== Phase 4: Final LP Polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 20
        if remaining > 30:
            t_per = min(remaining, 120.0)
            n = len(best_seq)
            if n <= 1200:
                print(f"  Final LP (n={n}), budget={t_per:.0f}s")
                result = lp_refine_v2(best_seq, t_per, verbose=True)
                update_best(result)

    # Phase 5: Final coordinate descent polish
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 10
        if remaining > 10:
            result = coordinate_descent(best_seq, remaining, verbose=False)
            update_best(result)

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate_score={final_score:.6f}, n={len(final_list)}, local={best_score:.6f} ===")
        if np.isfinite(final_score):
            return final_list

    # Fallback
    if provided is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided]

    return [1.0] * 256
