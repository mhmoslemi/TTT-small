Smooth-max gradient optimization for minimizing peak autocorrelation
of non-negative step functions. Uses log-sum-exp smooth approximation
with adaptive temperature, L-BFGS-B optimizer, multiple restarts,
and coordinate descent polish.
"""

import numpy as np
import time
from scipy.optimize import minimize

def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(f):
    """Compute the true score."""
    n = len(f)
    S = np.sum(f)
    if S < 0.01:
        return np.inf
    c = fft_convolve(f, f)
    return 2.0 * n * np.max(c) / (S * S)

def smooth_max_obj_and_grad(f, beta, fixed_sum=None):
    """
    Compute smooth max of autocorrelation (log-sum-exp) and its gradient.
    If fixed_sum is provided, we fix sum(f) = fixed_sum and optimize only M.
    """
    n = len(f)
    c = fft_convolve(f, f)  # length 2n-1
    c_max = np.max(c)
    c_shifted = c - c_max
    exp_c = np.exp(beta * c_shifted)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp  # softmax weights
    M_smooth = c_max + np.log(sum_exp) / beta
    
    # Gradient of M_smooth with respect to f
    # dM_smooth/df_i = 2 * sum_k w[k] * f[k-i] (valid k)
    # = 2 * (cross-correlation of w and f at lag i)
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    W = np.fft.rfft(w, N)
    F = np.fft.rfft(f, N)
    h = np.fft.irfft(W * np.conj(F), N)[:n]
    grad_M = 2.0 * h
    
    if fixed_sum is not None:
        # If we fix sum, the gradient of M is just grad_M
        # (we project out the component along the all-ones vector)
        # Actually, we just return grad_M and let the optimizer handle it
        # Or we can subtract the mean to project
        obj = M_smooth
    else:
        # score = 2*n*M/S^2
        S = np.sum(f)
        obj = 2.0 * n * M_smooth / (S * S)
        grad = (2.0 * n / (S * S)) * grad_M - (4.0 * n * M_smooth / (S * S * S)) * np.ones(n)
        return obj, grad
    
    return obj, grad_M

    """Smooth approximation of the full score = 2nM/S^2."""
    n = len(f)
    c = fft_convolve(f, f)
    c_max = np.max(c)
    c_shifted = c - c_max
    exp_c = np.exp(beta * c_shifted)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp
    M_smooth = c_max + np.log(sum_exp) / beta
    
    S = np.sum(f)
    obj = 2.0 * n * M_smooth / (S * S)
    
    # Gradient
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    W = np.fft.rfft(w, N)
    F = np.fft.rfft(f, N)
    h = np.fft.irfft(W * np.conj(F), N)[:n]
    grad_M = 2.0 * h
    
    grad = (2.0 * n / (S * S)) * grad_M - (4.0 * n * M_smooth / (S ** 3)) * np.ones(n)
    return obj, grad

def make_chirp(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / (2.0 * n))
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def make_multitone(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(1, 8):
        amp = 0.3 / k
        freq = rng.uniform(0.2, 3.0) * k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def make_block(n, rng, n_blocks=4):
    a = np.zeros(n)
    for i in range(n_blocks):
        start = i * n // n_blocks
        width = n // n_blocks
        a[start:start+width] = rng.uniform(0.5, 1.5)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def resample(seq, new_n):
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)

For a "flat" w (all equal, which happens when β is small), (w ⊛ f)[i] ≈ S/(2n-1) * (number of terms) ≈ S/(2n-1) * n ≈ S/2.
So ||grad||^2 ≈ 4 * n * (S/2)^2 = n*S^2 = n*2n = 2n^2.
And M ≈ S^2/(2n-1) ≈ 1.
So lr ≈ 1/(2n^2) ≈ 1/500000 for n=512. That's tiny!

But Adam should handle this automatically since it normalizes by the gradient magnitude. Let me just use a moderate learning rate and let Adam's normalization handle the scaling.

Actually, let me reconsider. With Adam, the effective step size is approximately lr (the sign of the gradient times lr). So lr = 0.01 means I'm moving about 0.01 in each coordinate per step. Since the f_i values are about S/n ≈ 0.06, a step of 0.01 is about 15% of the value, which is reasonable.

Let me also add a "rejection" mechanism: if the true score increases significantly after a step, revert.

Now let me also think about a completely different approach: **simulated annealing on the smooth objective**. Instead of gradient descent, I can do:
1. Compute the smooth objective and gradient
2. Take a step in the negative gradient direction plus random noise
3. Accept/reject based on the smooth objective (Metropolis criterion)
4. Anneal both the temperature and the β parameter

This might escape local minima better than pure gradient descent.

Let me also think about **projected gradient descent** with a proper projection onto the constraint set {f ≥ 0, sum(f) = S_target}.

The projection onto {f ≥ 0} is just clipping at 0.
The projection onto {sum(f) = S_target} is: f -> f + (S_target - sum(f))/n (broadcast).
The projection onto {f ≥ 0, sum(f) = S_target} is: clip, then rescale.

OK let me now write the final, clean solution. I'll combine several approaches:

1. Smooth max with adaptive β (main optimization)
2. Multiple restarts with different initializations
3. A "peak-targeting" heuristic (reduce elements that contribute most to the current peak)
4. Final coordinate descent polish

Let me also consider: should I vary n? The previous best was n=512. Let me try n=512 primarily but also explore nearby values.

One more thought: the "soft" peak reduction. Instead of just targeting the single maximum lag, I can target the top-K lags simultaneously. This is naturally handled by the smooth max with a moderate β (it gives significant weight to the top few peaks).

"""
Smooth-max gradient optimization with adaptive beta annealing for minimizing
peak autocorrelation of non-negative step functions. Combines Adam optimizer
on log-sum-exp smoothed objective, peak-targeting heuristics, multi-start
exploration, and coordinate descent polish.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fft_convolve(a, b):
    """Compute linear convolution using FFT (fast path)."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(f):
    """Compute the true evaluation score."""
    n = len(f)
    S = np.sum(f)
    if S < 0.01:
        return np.inf
    c = fft_convolve(f, f)
    return float(2.0 * n * np.max(c) / (S * S))


def smooth_max_and_grad(f, beta):
    """
    Compute smooth max of autocorrelation and its gradient w.r.t. f.
    Returns (M_smooth, grad) where grad = dM_smooth/df.
    """
    n = len(f)
    c = fft_convolve(f, f)  # length 2n-1
    c_max = np.max(c)
    c_shifted = c - c_max
    exp_c = np.exp(beta * c_shifted)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp
    M_smooth = c_max + np.log(sum_exp) / beta

    # Gradient: grad_i = 2 * sum_k w[k] * f[k-i] (valid k)
    # = 2 * cross_correlation(w, f)[i]
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    W = np.fft.rfft(w, N)
    F = np.fft.rfft(f, N)
    h = np.fft.irfft(W * np.conj(F), N)[:n]
    grad = 2.0 * h
    return M_smooth, grad


def score_obj_and_grad(f, beta):
    """Smooth approximation of score = 2nM/S^2 and its gradient."""
    n = len(f)
    M, grad_M = smooth_max_and_grad(f, beta)
    S = np.sum(f)
    obj = 2.0 * n * M / (S * S)
    grad = (2.0 * n / (S * S)) * grad_M - (4.0 * n * M / (S ** 3)) * np.ones(n)
    return obj, grad


def adam_smooth_optimize(f0, time_limit, rng, beta_start=2.0, beta_end=60.0, verbose=True):
    """
    Adam optimization on the smooth max objective with adaptive beta annealing.
    Normalizes to fixed sum at each step for stability.
    """
    n = len(f0)
    S_target = np.sqrt(2.0 * n)
    f = f0.astype(np.float64).copy()
    f = np.clip(f, 0, None)
    s = np.sum(f)
    if s < 0.01:
        f = np.ones(n) * S_target / n
    else:
        f = f * (S_target / s)

    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    beta = beta_start
    lr_base = 0.02
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # Schedule for beta
    total_steps_est = int(time_limit / 0.05)  # rough estimate of steps per second * time
    total_steps_est = max(100, min(total_steps_est, 20000))

    consecutive_rejects = 0
    max_rejects = 30

    while time.monotonic() - t0 < time_limit:
        # Anneal beta (exponential schedule)
        progress = min(1.0, step / total_steps_est)
        beta = beta_start * (beta_end / beta_start) ** (progress ** 0.6)

        # Compute gradient
        M, grad = smooth_max_and_grad(f, beta)

        # Adam update
        step += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)

        lr = lr_base / (1.0 + 0.005 * step)
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = np.clip(f_new, 0.0, None)

        # Renormalize to fixed sum
        s_new = np.sum(f_new)
        if s_new < 0.01:
            f_new = f.copy()
        else:
            f_new = f_new * (S_target / s_new)

        # Evaluate true score
        true_score = compute_score(f_new)

        if true_score < best_score:
            best_score = true_score
            best_f = f_new.copy()
            consecutive_rejects = 0
            f = f_new
        else:
            consecutive_rejects += 1
            if consecutive_rejects < max_rejects:
                # Accept with some probability (simulated annealing flavor)
                temp = 0.005 / (1.0 + step * 0.001)
                delta = true_score - best_score
                if rng.random() < np.exp(-delta / max(temp, 1e-10)):
                    f = f_new
                    consecutive_rejects = 0
            else:
                # Rejected too many times, perturb to escape
                f = f * np.exp(rng.normal(0, 0.02, n))
                f = np.clip(f, 0, None)
                s_p = np.sum(f)
                if s_p > 0.01:
                    f = f * (S_target / s_p)
                consecutive_rejects = 0

        # Periodic reset of Adam state to avoid stale momentum
        if step % 500 == 0:
            m = m * 0.5
            v = v * 0.5

        if verbose and step % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      step={step}, beta={beta:.1f}, M={M:.4f}, "
                  f"score={true_score:.6f}, best={best_score:.6f}, "
                  f"t={elapsed:.0f}s")

    return best_f, best_score


def peak_targeting_step(f, rng):
    """
    Targeted step that reduces contributions to the current peak lag.
    Returns a new candidate.
    """
    n = len(f)
    c = fft_convolve(f, f)
    peak_idx = int(np.argmax(c))

    # Compute contribution of each element to the peak
    contribs = np.zeros(n)
    for i in range(n):
        j = peak_idx - i
        if 0 <= j < n:
            contribs[i] = f[i] * f[j]

    # Reduce top contributors, increase bottom contributors
    f_new = f.copy()
    top_k = max(4, n // 12)
    sorted_idx = np.argsort(contribs)
    top = sorted_idx[-top_k:]
    bottom = sorted_idx[:top_k]

    for idx in top:
        f_new[idx] *= np.exp(-rng.uniform(0.01, 0.05))
    for idx in bottom:
        if f_new[idx] > 0:
            f_new[idx] *= np.exp(rng.uniform(0.01, 0.04))

    # Renormalize
    s = np.sum(f_new)
    S_target = np.sqrt(2.0 * n)
    if s > 0.01:
        f_new = f_new * (S_target / s)

    return f_new


def lbfgs_refine(f0, beta, max_iter=100):
    """Single L-BFGS-B refinement on the smooth objective."""
    n = len(f0)
    S_target = np.sqrt(2.0 * n)

    def obj_grad(x):
        x = np.clip(x, 0.0, None)
        val, g = smooth_max_and_grad(x, beta)
        return val, g

    result = minimize(obj_grad, f0, jac=True, method='L-BFGS-B',
                      bounds=[(0.0, 100.0)] * n,
                      options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10})

    f_new = np.clip(result.x, 0, None)
    s = np.sum(f_new)
    if s > 0.01:
        f_new = f_new * (S_target / s)
    return f_new


def coordinate_descent_polish(f0, time_limit, rng):
    """Coordinate descent polish for final improvement."""
    n = len(f0)
    S_target = np.sqrt(2.0 * n)
    f = f0.copy()
    current_score = compute_score(f)
    best_f = f.copy()
    best_score = current_score
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(f, f)
        max_c = np.max(c)
        peak_idx = int(np.argmax(c))

        # Compute contributions to peak
        contribs = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] = f[i] * f[j]

        improved = False
        top_k = max(3, n // 16)
        top_idx = np.argsort(contribs)[-top_k:]
        other_idx = np.setdiff1d(np.arange(n), top_idx)

        # Try reducing top contributors
        for idx in top_idx:
            for scale in [0.95, 0.9, 0.85]:
                f_trial = f.copy()
                f_trial[idx] *= scale
                score = compute_score(f_trial)
                if score < current_score:
                    f = f_trial
                    current_score = score
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try increasing non-peak elements
            shuffled = other_idx[rng.permutation(len(other_idx))[:8]]
            for idx in shuffled:
                for scale in [1.05, 1.1, 1.15]:
                    f_trial = f.copy()
                    f_trial[idx] *= scale
                    score = compute_score(f_trial)
                    if score < current_score:
                        f = f_trial
                        current_score = score
                        improved = True
                        break
                if improved:
                    break

        if current_score < best_score:
            best_score = current_score
            best_f = f.copy()

        if not improved:
            # Random perturbation
            f = f * np.exp(rng.normal(0, 0.008, n))
            f = np.clip(f, 0, None)
            s = np.sum(f)
            if s > 0.01:
                f = f * (S_target / s)
            current_score = compute_score(f)

    return best_f, best_score


def make_chirp(n, alpha, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / (2.0 * n))
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multitone(n, rng):
    """Multi-tone initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(1, 8):
        amp = 0.25 / k
        freq = rng.uniform(0.2, 2.5) * k
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_asymmetric(n, rng, bias=0.3):
    """Asymmetric initialization (helps reduce center lag)."""
    t = np.arange(n, dtype=np.float64)
    x = t / (n - 1)
    a = 1.0 + bias * (2.0 * x - 1.0)  # linear ramp
    a += 0.1 * np.sin(3.0 * np.pi * x)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def resample(seq, new_n):
    """Resample a sequence to a new length."""
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)
