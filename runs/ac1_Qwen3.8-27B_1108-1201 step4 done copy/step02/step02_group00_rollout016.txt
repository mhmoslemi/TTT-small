Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using block coordinate descent with LP subproblems.
The key idea: fix current sequence f, solve LP to find best g that maintains autocorrelation bound,
then blend f towards g. This is the "turning tower" algorithm adapted for non-negative sequences.
"""
import numpy as np
import time
import cvxpy as cp

def fast_convolve(a, b=None):
    """Linear auto- or cross-convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def build_conv_matrix(f):
    """Build matrix C such that C @ g = conv(f, g) for fixed f."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        C[k, j_start:j_end+1] = f[k - j_start:k - j_end + 1]
    return C

def solve_lp_natural(f, M):
    """
    Solve: max sum(g) s.t. g >= 0, conv(f, g) <= M.
    Uses CVXPY with ECOS or CLARABEL.
    Returns (g, optimal_value) or (None, 0) if infeasible.
    """
    n = len(f)
    g = cp.Variable(n, non_neg=True)
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.CLARABEL)
    except:
        try:
            prob.solve(solver=cp.ECOS, max_iter=10000)
        except:
            return None, 0.0
    
    if g.value is None:
        return None, 0.0
    g_val = np.maximum(g.value, 0)
    return g_val, float(prob.value) if prob.value is not None else 0.0

def turning_tower(f0, time_limit, t_init=0.3, t_min=0.01, m_decay=0.999, 
                  m_floor_ratio=0.5, batch=20, verbose=False):
    """
    Turning tower algorithm: iterate LP-based updates to minimize max autocorrelation.
    f0: initial sequence (will be normalized to sum=1)
    Returns (best_f, best_objective)
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_f = f.copy()
    best_obj = 2.0 * n * M  # since sum(f)=1, obj = 2n*max(conv)/1^2
    
    t0 = time.monotonic()
    t = t_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            
            g, g_sum = solve_lp_natural(f, M)
            if g is None or g_sum < 1e-10:
                break
            
            # Normalize g to sum=1
            g = g / g.sum()
            
            # Blend
            f_new = (1 - t) * f + t * g
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
            
            conv = fast_convolve(f_new)
            cur_M = conv.max()
            cur_obj = 2.0 * n * cur_M
            
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_f = f_new.copy()
                no_improve = 0
                # Accept and potentially tighten M
                f = f_new
                M = cur_M * m_decay  # tighten the bound slightly
            else:
                f = f_new  # still accept to keep moving
                no_improve += 1
        
        # Schedule
        t = max(t_min, t * 0.995)
        m_decay = min(1.0, m_decay + 0.00001)
        
        if no_improve >= batch * 5:
            # Reset with perturbation
            rng = np.random.default_rng()
            noise = rng.standard_normal(n) * 0.01
            f = np.maximum(f + noise, 0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            t = t_init * 0.5
            no_improve = 0
    
    return best_f, best_obj

    return np.fft.irfft(A * B, fft_size)[:size]


def build_conv_matrix(f):
    """Build matrix C such that (C @ g)[k] = conv(f, g)[k]."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        C[k, js:je + 1] = f[k - js:k - je + 1]
    return C

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using the turning tower algorithm with LP subproblems.
The turning tower iteratively solves LPs to find optimal blend directions that reduce
peak autocorrelation, combined with multi-start exploration, adaptive step scheduling,
and a smooth-max gradient descent hybrid for final polish.
"""
import numpy as np
import time
import cvxpy as cp


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def build_conv_matrix(f):
    """Build matrix C such that (C @ g)[k] = conv(f, g)[k] for fixed f."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        C[k, js:je + 1] = f[k - js:k - je + 1]
    return C


def turning_tower_lp(f0, time_limit, t_init=0.5, t_decay=0.999, t_min=0.02,
                     restart_every=500, perturb_scale=0.02):
    """
    Turning tower with LP subproblems to minimize max autocorrelation.
    f0: initial sequence (will be normalized to sum=1)
    Returns (best_f_normalized, best_2n_maxconv).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_f = f.copy()
    best_obj = 2.0 * n * M

    t0 = time.monotonic()
    t = t_init
    iteration = 0
    no_improve = 0
    rng = np.random.default_rng(12345 + n)

    while time.monotonic() - t0 < time_limit:
        # Build LP
        C = build_conv_matrix(f)
        g_var = cp.Variable(n, non_neg=True)
        constraints = [C @ g_var <= M]
        objective = cp.Maximize(cp.sum(g_var))
        prob = cp.Problem(objective, constraints)

        # Solve LP
        g_val = None
        try:
            prob.solve(solver=cp.CLARABEL)
            if g_var.value is not None:
                g_val = np.maximum(g_var.value, 0.0)
        except Exception:
            pass

        if g_val is None:
            try:
                prob.solve(solver=cp.ECOS, max_iter=10000)
                if g_var.value is not None:
                    g_val = np.maximum(g_var.value, 0.0)
            except Exception:
                break

        if g_val is None:
            break

        g_sum = g_val.sum()
        if g_sum < 1e-12:
            break

        g_norm = g_val / g_sum

        # Blend
        f_new = (1.0 - t) * f + t * g_norm
        f_new = np.maximum(f_new, 0.0)
        f_s = f_new.sum()
        if f_s < 1e-15:
            f_new = np.ones(n) / n
        else:
            f_new = f_new / f_s

        conv = fast_convolve(f_new)
        cur_M = conv.max()
        cur_obj = 2.0 * n * cur_M

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        M = cur_M
        iteration += 1
        t = max(t_min, t * t_decay)

        # Restart if stuck
        if no_improve >= restart_every:
            noise = rng.standard_normal(n) * perturb_scale
            f = np.maximum(f + noise, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            t = max(t_min, t_init * 0.4)
            no_improve = 0
            perturb_scale *= 1.3

    return best_f, best_obj


def smooth_grad_descent(f0, time_limit, lr=0.01, T=50.0, momentum=0.9):
    """
    Gradient descent using smooth max (log-sum-exp) approximation.
    Minimizes 2n * max(conv(f,f)) with sum(f)=1, f>=0.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    f = f / s if s > 1e-15 else np.ones(len(f)) / len(f)
    n = len(f)

    best_f = f.copy()
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()

    t0 = time.monotonic()
    v = np.zeros(n)
    T_cur = T
    lr_cur = lr

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        max_conv = conv.max()
        if max_conv < 1e-15:
            break

        # Softmax weights over convolution peaks
        scaled = T_cur * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        w = w / w_sum if w_sum > 1e-15 else np.zeros_like(conv)

        # Compute gradient: d/d f_i of sum_k w_k * conv(f,f)[k]
        # conv(f,f)[k] = sum_j f[j] * f[k-j]
        # d/d f_i = sum_k w_k * (f[k-i] + f[i-k]) for valid indices
        grad = np.zeros(n)
        # Use FFT for speed: grad_i = sum_k w_k * (conv contributions)
        # Actually: grad_i = 2 * (w * f)[i] (cyclic) approx, but for linear conv it's:
        # grad_i = sum_k w_k * (f[k-i] + f[i-k]) where terms are 0 if index out of range
        # This is 2 * linear_cross_correlation(w, f)[i] approximately
        # Let me just do it directly for correctness
        active = np.where(w > 1e-10 * w.max())[0]
        for k_idx in active:
            k = int(k_idx)
            wi = w[k]
            # d conv[k]/d f[i] = f[k-i] + f[i-k] (symmetric)
            # for valid i, k-i in [0,n-1], i-k in [0,n-1]
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i_idx in range(i_start, i_end + 1):
                i = i_idx
                term = 0.0
                if 0 <= k - i < n:
                    term += f[k - i]
                if 0 <= i - k < n:
                    term += f[i - k]
                grad[i] += wi * 2.0 * term

        # Projected gradient step with momentum
        v = momentum * v - lr_cur * grad
        f_new = f + v
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()

        cur_conv = fast_convolve(f_new)
        cur_obj = 2.0 * n * cur_conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f_new.copy()

        f = f_new
        T_cur *= 1.01
        lr_cur *= 0.9995

    return best_f, best_obj


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def generate_initializations(n, parent, rng):
    """Generate diverse initializations for search."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Triangular
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = tri / tri.sum()

    # Gaussians
    for sf in [0.1, 0.2, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (sf * n)) ** 2)
        inits[f'g_{sf}'] = g / g.sum()

    # Bimodal
    bi = np.ones(n) * 0.05
    edge = max(1, n // 5)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = ramp / ramp.sum()

    # Cosine window
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    inits['cosine'] = cos_w / cos_w.sum()

    # Random
    inits['rand1'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['rand2'] = rng.dirichlet(np.ones(n) * 2.0)

    # Sparse
    sparse = np.zeros(n)
    k = max(2, n // 3)
    idx = rng.choice(n, k, replace=False)
    sparse[idx] = 1.0
    inits['sparse'] = sparse / sparse.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps

    return inits
