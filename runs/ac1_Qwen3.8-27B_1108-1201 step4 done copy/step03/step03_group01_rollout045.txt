def fast_convolve(a):
    n = len(a)

    A = np.fft.rfft(a, fft_size)

def build_conv_matrix(a, n):
    C = np.zeros((2 * n - 1, n))

Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses LP-based direction finding, log-sum-exp smoothed gradient with
Nesterov acceleration, and multi-length diversification.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (linear cross-convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def objective_value(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Binary search on the threshold
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def lse_objective(a, tau):
    """Log-sum-exp smoothed objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    scaled = conv / tau
    m = scaled.max()
    lse = tau * (m + np.log(np.sum(np.exp(scaled - m))))
    return 2.0 * n * lse / (s * s)


def lse_gradient(a, tau):
    """Gradient of the log-sum-exp smoothed objective."""
    n = len(a)
    s = a.sum()
    conv = fast_convolve(a)
    
    # Softmax weights

    m = scaled.max()
    exp_scaled = np.exp(scaled - m)
    w = exp_scaled / exp_scaled.sum()

    # Use FFT cross-correlation
    N = 1 << (3 * n).bit_length()
    Af = np.fft.rfft(a, N)
    Wf = np.fft.rfft(w, N)
    corr = np.fft.irfft(Af * np.conj(Wf), N)
    grad_lse = 2.0 * corr[:n]
    
    # Full gradient of 2n * lse / s^2
    grad = 2.0 * n * (grad_lse / (s * s) - 2.0 * lse_val / (s**3))
    return grad

                            a = a_new / s_new
                        else:
                            a = best_a.copy()
                        conv = fast_convolve(a)
                        M = conv.max()
                        if 2.0 * n * M < best_obj:
                            best_obj = 2.0 * n * M
                            best_a = a.copy()
                            no_improve = 0
                        else:
                            no_improve += 1
                    else:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        iteration += 1
        
        if no_improve >= max_no_improve:
            rng = np.random.default_rng(iteration * 7 + 13)
            scale = 0.003 + 0.005 * min(perturb_count, 10)
            noise = rng.standard_normal(n) * scale
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            perturb_count += 1
            max_no_improve = min(300, 100 + 20 * perturb_count)
            
            if perturb_count > 50:
                break
    
    return best_a, best_obj


def lse_optimize(a0, time_limit, tau_init=0.01, tau_decay=0.9995,
                 lr_init=0.01, momentum=0.9, verbose=False):
    """Log-sum-exp smoothed optimization with Nesterov acceleration."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))
    
    v = np.zeros(n)  # Momentum
    y = a.copy()  # Nesterov lookahead
    tau = tau_init
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Nesterov step
        y = a + momentum * v
        
        # Clamp y to non-negative
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = a.copy()
        else:
            y = y / s_y
        
        # Compute gradient at y
        grad = lse_gradient(y, tau)
        
        # Update velocity
        v = momentum * v - lr * grad
        
        # Take step and project
        a_new = a - lr * grad  # Actually, for Nesterov: a_new = y - lr * grad
        a_new = project_simplex(y - lr * grad)
        
        v = a_new - a
        a = a_new
        
        tau *= tau_decay
        lr *= 0.9999
        
        # Check improvement
        score = objective_value(a)
        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 2000:
            # Restart with perturbation
            rng = np.random.default_rng(int(time.monotonic() * 1000))
            a = best_a + rng.standard_normal(n) * 0.01
            a = project_simplex(a)
            v = np.zeros(n)
            no_improve = 0
    
    # Verify with actual objective
    final_score = float(evaluate_sequence(best_a.tolist()))
    if final_score < best_score:
        best_score = final_score
    
    return best_a, best_score
