"""
Optimized LP-based descent for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses a minimization LP (min t s.t. (a*b)[k] <= t, sum(b)=1, b>=0) for direction finding,
combined with line search, subgradient escape, and multi-start exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build the matrix A where A[k,j] = a[k-j] if 0<=k-j<n, else 0."""
    n = len(a)
    num_k = 2 * n - 1
    A = np.zeros((num_k, n))
    for k in range(num_k):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        j_idx = np.arange(j_min, j_max + 1)
        A[k, j_idx] = a[k - j_idx]
    return A


def solve_direction_lp(a, verbose=False):
    """
    Solve: min t s.t. (a*b)[k] <= t for all k, sum(b) = 1, b >= 0.
    Returns (b_opt, t_opt).
    """
    n = len(a)
    num_k = 2 * n - 1
    A_conv = build_conv_matrix(a)
    
    # Variables: b_0, ..., b_{n-1}, t
    # Minimize: t
    # Constraints:
    #   A_conv @ b - t <= 0  (num_k constraints)
    #   sum(b) = 1  (1 equality)
    #   b >= 0, t free (but t >= 0 implied by b >= 0 and A_conv >= 0)
    
    # c: minimize t → c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A_conv @ b - t <= 0
    # → [A_conv, -1] @ [b, t] <= 0
    A_ub = np.hstack([A_conv, -np.ones((num_k, 1))])
    b_ub = np.zeros(num_k)
    
    # Equality: sum(b) = 1
    A_eq = np.hstack([np.ones(n), np.zeros(1)])
    b_eq = np.array([1.0])
    
    # Bounds: b_i >= 0, t >= 0
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True})
    
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None


def subgradient_of_max_conv(a):
    """Compute subgradient of max_k (a*a)[k] w.r.t. a."""
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    # Find all indices where conv is close to max
    tolerance = 1e-10
    peak_indices = np.where(conv >= M - tolerance)[0]
    
    # Subgradient: (1/|K*|) * sum_{k in K*} grad_k
    # grad_k[i] = a[k-i] if 0 <= k-i < n, else 0
    # (plus the symmetric term, but for auto-convolution, the gradient of
    #  (a*a)[k] w.r.t. a[i] is:
    #  a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
    #  But for auto-convolution, (a*a)[k] = sum_j a[j]*a[k-j]
    #  d/d a[i] = a[k-i] (if 0<=k-i<n) [from the j=i term]
    #  + a[i-k] (if 0<=i-k<n) [from the k-j=i term, i.e., j=k-i... wait]
    
    # Let me be more careful. (a*a)[k] = sum_{j} a[j] * a[k-j]
    # d/d a[i] of this = sum_j [d a[j]/d a[i] * a[k-j] + a[j] * d a[k-j]/d a[i]]
    # = sum_j [delta(j,i) * a[k-j] + a[j] * delta(k-j, i)]
    # = a[k-i] (if 0 <= k-i < n) + sum_j a[j] * delta(k-j, i)
    # = a[k-i] (if 0 <= k-i < n) + a[k-i] (if 0 <= k-i < n)
    # Wait, the second term: delta(k-j, i) means k-j = i, so j = k-i.
    # So the second term is a[k-i] (if 0 <= k-i < n).
    
    # So d(a*a)[k] / d a[i] = 2 * a[k-i] (if 0 <= k-i < n), else 0.
    # UNLESS k-i = i, i.e., k = 2i, in which case it's a[i] * 2 but we should
    # count it only once? No, let me redo this.
    
    # (a*a)[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j] * a[k-j]
    # d/d a[i] = sum_j [delta_{j,i} * a[k-j] + a[j] * delta_{k-j,i}]
    # First sum: only j=i contributes (if 0<=k-i<n): gives a[k-i]
    # Second sum: only k-j=i contributes, i.e., j=k-i (if 0<=k-i<n): gives a[k-i]
    # Total: 2 * a[k-i] (if 0 <= k-i < n), else 0
    # Special case: if k-i = i (i.e., k=2i), then both terms give a[i], total = 2*a[i]
    # But that's already captured: 2 * a[k-i] = 2 * a[i] when k=2i. Correct.
    
    grad = np.zeros(n)
    for k in peak_indices:
        # grad += a[k-i] for valid i
        i_min = max(0, k - n + 1)  # k-i >= 0 → i <= k
        i_max = min(n - 1, k)       # k-i <= n-1 → i >= k-(n-1)
        i_range = np.arange(i_min, i_max + 1)
        grad[i_range] += a[k - i_range]
    
    grad /= len(peak_indices)
    return grad, M


def lp_descent(a0, time_limit, alpha_init=0.5, seed=0, verbose=False):
    """LP-based descent on the simplex."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    alpha = alpha_init
    no_improve = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        
        b, t_opt = solve_direction_lp(a)
        if b is None:
            break
        
        gap = M - t_opt
        if gap < 1e-12:
            # At local minimum of this subproblem
            no_improve += 1
            if no_improve > 20:
                # Escape: perturb
                noise = rng.standard_normal(n) * 0.01
                a = project_simplex(a + noise)
                no_improve = 0
                alpha = alpha_init
            continue
        
        # Line search: try different alphas
        best_alpha = alpha
        best_obj_try = best_obj
        for a_try in [alpha, min(1.0, alpha * 1.5), max(0.1, alpha * 0.5), 0.3, 0.7]:
            a_try = np.clip(a_try, 0.05, 1.0)
            a_candidate = (1.0 - a_try) * a + a_try * b
            a_candidate = np.maximum(a_candidate, 0.0)
            s_c = a_candidate.sum()
            if s_c < 1e-15:
                continue
            a_candidate = a_candidate / s_c
            conv_c = fast_convolve(a_candidate)
            obj_c = 2.0 * n * conv_c.max()
            if obj_c < best_obj_try:
                best_obj_try = obj_c
                best_alpha = a_try
        
        a_new = (1.0 - best_alpha) * a + best_alpha * b
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            # Increase alpha slightly if making progress
            alpha = min(1.0, alpha * 1.05)
        else:
            no_improve += 1
            alpha = max(0.1, alpha * 0.8)
            if no_improve > 30:
                noise = rng.standard_normal(n) * 0.005
                a = project_simplex(a + noise)
                alpha = alpha_init
                no_improve = 0
        
        iter_count += 1
        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iter_count}: obj={best_obj:.8f}, M={M:.6f}, t_opt={t_opt:.6f}, gap={gap:.2e}, alpha={alpha:.3f}, t={elapsed:.1f}s")
    
    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


def subgradient_descent(a0, time_limit, lr_init=0.01, seed=0, verbose=False):
    """Subgradient descent on the simplex."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        grad, M = subgradient_of_max_conv(a)
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        
        a_new = project_simplex(a - lr * grad)
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                lr = max(1e-5, lr * 0.9)
            if no_improve > 500:
                noise = rng.standard_normal(n) * 0.01
                a = project_simplex(a + noise)
                lr = lr_init * 0.5
                no_improve = 0
        
    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Ramps
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp_inc'] = ramp
    inits['ramp_dec'] = ramp[::-1].copy()
    
    # Exponential
    exp_r = np.exp(-x / (n / 5.0))
    exp_r /= exp_r.sum()
    inits['exp_left'] = exp_r
    inits['exp_right'] = exp_r[::-1].copy()
    
    # Bimodal
    bi = np.zeros(n)
    bi[:n//3] = 1.0
    bi[-n//3:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Triangle (peak at center)
    tri = 1.0 - np.abs(x - (n-1)/2.0) / ((n-1)/2.0)
    tri = np.maximum(tri, 0.01)
    tri /= tri.sum()
    inits['triangle_c'] = tri
    
    # Gaussian
    g = np.exp(-0.5 * ((x - n/2) / (n/4))**2)
    g /= g.sum()
    inits['gauss'] = g
    
    # Cosine
    cos_w = np.cos(np.pi * x / n)**2
    cos_w = np.maximum(cos_w, 0.01)
    cos_w /= cos_w.sum()
    inits['cosine'] = cos_w
    
    # Random
    inits['rand1'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['rand2'] = rng.dirichlet(np.ones(n) * 2.0)
    inits['rand3'] = rng.dirichlet(np.ones(n) * 0.1)
    
    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_res'] = pp / ps
    
    return inits

"""
LP-minimization-based descent for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Key innovation: uses min t s.t. (a*b)[k]<=t, sum(b)=1, b>=0 LP for direction finding,
combined with line search, parametric initialization, subgradient escape,

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A where A[k,j] = a[k-j] if 0<=k-j<n, else 0."""
    n = len(a)
    num_k = 2 * n - 1
    A = np.zeros((num_k, n))
    for k in range(num_k):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[k, j_min:j_max+1] = a[k - np.arange(j_min, j_max+1)]
    return A


def solve_direction_lp(a):
    """Solve: min t s.t. (a*b)[k] <= t for all k, sum(b)=1, b>=0. Returns (b, t)."""
    n = len(a)
    num_k = 2 * n - 1
    A_conv = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.hstack([A_conv, -np.ones((num_k, 1))])
    b_ub = np.zeros(num_k)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True})

    if res.success:
        return res.x[:n], res.x[-1]
    return None, None


def solve_sota_lp(a, M):
    """SOTA LP: max sum(b) s.t. b>=0, (a*b)[k] <= M. Returns b (rescaled to simplex)."""
    n = len(a)
    num_k = 2 * n - 1
    A_conv = build_conv_matrix(a)

    c = -np.ones(n)
    A_ub = A_conv
    b_ub = np.full(num_k, M)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})

    if res.success:
        s = res.x.sum()
        if s > 1e-15:
            return res.x / s, s
    return None, None


def subgradient_of_max_conv(a):
    """Subgradient of max_k (a*a)[k] w.r.t. a."""
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    tolerance = max(1e-12, M * 1e-8)
    peak_indices = np.where(conv >= M - tolerance)[0]

    grad = np.zeros(n)
    for k in peak_indices:
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        i_range = np.arange(i_min, i_max + 1)
        grad[i_range] += a[k - i_range]

    grad /= len(peak_indices)
    return grad, M

    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()

def subgradient_descent(a0, time_limit, lr_init=0.01, seed=0, verbose=False):
    """Subgradient descent with adaptive step on the simplex."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)

    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        grad, M = subgradient_of_max_conv(a)
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        grad = grad / g_norm

        a_new = project_simplex(a - lr * grad)
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr = max(1e-6, lr * 0.92)
            if no_improve > 300:
                noise = rng.standard_normal(n) * 0.01
                a = project_simplex(a + noise)
                lr = lr_init * 0.5
                no_improve = 0

    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj

    scan_results = []

    for n in lengths:
        if time.monotonic() - t_phase1 > phase1_time - 15:
            break

        # Parametric search (quick)
        t_param = min(20.0, phase1_time - 15 - (time.monotonic() - t_phase1))
        a_param, obj_param = parametric_search(n, t_param, rng)
        score_param = float(evaluate_sequence(a_param.tolist()))
        if np.isfinite(score_param):
            scan_results.append((score_param, n, 'param', a_param.copy()))
            if score_param < best_score:
                best_score = score_param
                best_sequence = a_param.copy()
                print(f"  Param n={n}: {score_param:.6f}")

        # A few structured inits
        inits = generate_initializations(n, parent, rng)
        names = ['parent_res', 'parent', 'ramp_inc', 'exp_left', 'bimodal',
                 'skew_right', 'power_law', 'uniform']
        names = [nm for nm in names if nm in inits][:3]

        time_remaining = phase1_time - 10 - (time.monotonic() - t_phase1)
        time_per = max(3.0, time_remaining / max(1, len(lengths) - lengths.index(n)) / max(1, len(names)))

        for name in names:
            if time.monotonic() - t_phase1 > phase1_time - 5:
                break
            t_alloc = min(time_per, phase1_time - 5 - (time.monotonic() - t_phase1))
            if t_alloc < 2:
                break
            a0 = inits[name]
            a_opt, obj_opt = lp_descent(a0, time_limit=t_alloc,
                                         alpha_init=0.6, seed=n * 7 + hash(name) % 100)
            score = float(evaluate_sequence(a_opt.tolist()))
            scan_results.append((score, n, name, a_opt.copy()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Scan n={n} {name}: {score:.6f}")

    scan_results = [r for r in scan_results if np.isfinite(r[0])]
    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done: best={best_score:.6f}, candidates={len(scan_results)}")

    # ===== Phase 2: Deep LP optimization (600s) =====
    print("=== Phase 2: Deep LP (600s) ===")
    t_phase2 = time.monotonic()
    phase2_time = 600.0

    # Build candidate list
    candidates = []
    seen_n = set()
    for score, n, name, a_opt in scan_results[:8]:
        if n not in seen_n or score < best_score + 0.001:
            candidates.append((n, a_opt.copy(), f"scan_{name}"))
            seen_n.add(n)
        if len(candidates) >= 8:
            break

    if best_sequence is not None:
        candidates.append((len(best_sequence), best_sequence.copy(), "best"))

    # Add fresh starts at promising lengths
    best_n = len(best_sequence) if best_sequence is not None else 320
    for n_try in [best_n, best_n - 40, best_n + 40]:
        if n_try < 64 or n_try > 800:
            continue
        inits = generate_initializations(n_try, parent, rng)
        for nm in ['ramp_inc', 'exp_left', 'skew_right']:
            if nm in inits:
                candidates.append((n_try, inits[nm].copy(), f"fresh_{nm}"))
        # Also add parametric result
        a_p, _ = parametric_search(n_try, 5.0, rng)
        candidates.append((n_try, a_p.copy(), "fresh_param"))

    # Deduplicate by length+name
    seen_keys = set()
    unique_candidates = []
    for n_c, a_c, name_c in candidates:
        key = (n_c, name_c)
        if key not in seen_keys:
            unique_candidates.append((n_c, a_c, name_c))
            seen_keys.add(key)
    candidates = unique_candidates[:12]

    time_per_cand = max(15.0, phase2_time / max(1, len(candidates)))
    print(f"  {len(candidates)} candidates, ~{time_per_cand:.0f}s each")

    for i, (n_c, a_c, name_c) in enumerate(candidates):
        if time.monotonic() - t_phase2 > phase2_time - 15:
            break
        t_alloc = min(time_per_cand, phase2_time - 15 - (time.monotonic() - t_phase2))
        if t_alloc < 8:
            break

        # Alternate between my LP and SOTA LP
        use_sota = (i % 3 == 2)
        alpha_init = 0.5 if not use_sota else 0.7

        a_opt, obj_opt = lp_descent(a_c, time_limit=t_alloc,
                                     alpha_init=alpha_init,
                                     seed=i * 13 + 5,
                                     use_sota_lp=use_sota)
        score = float(evaluate_sequence(a_opt.tolist()))
        tag = "SOTA-LP" if use_sota else "Min-LP"
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  [{i}] {tag} n={n_c} {name_c}: {score:.6f}")
        elif i % 3 == 0:
            print(f"  [{i}] {tag} n={n_c} {name_c}: {score:.6f} (no improvement)")

    print(f"Phase 2 done: best={best_score:.6f}")

    # ===== Phase 3: Subgradient refinement (250s) =====
    print("=== Phase 3: Subgradient (250s) ===")
    t_phase3 = time.monotonic()
    phase3_time = 250.0

    if best_sequence is not None:
        n_best = len(best_sequence)
        sg_configs = [
            (0.003, 0), (0.008, 1), (0.015, 2), (0.001, 3),
        ]
        time_per_sg = phase3_time / len(sg_configs)

        for lr, sd in sg_configs:
            if time.monotonic() - t_phase3 > phase3_time - 10:
                break
            t_alloc = min(time_per_sg, phase3_time - 10 - (time.monotonic() - t_phase3))
            if t_alloc < 8:
                break

            # Start from slightly perturbed best
            noise = rng.standard_normal(n_best) * 0.002
            a_start = project_simplex(best_sequence + noise)

            a_opt, obj_opt = subgradient_descent(a_start, time_limit=t_alloc,
                                                  lr_init=lr, seed=sd * 23 + 11)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  SG lr={lr}: {score:.6f}")

    print(f"Phase 3 done: best={best_score:.6f}")

    # ===== Phase 4: Final LP polish (remaining time) =====
    print("=== Phase 4: Polish ===")
    t_phase4 = time.monotonic()
    phase4_time = deadline - time.monotonic()

    if best_sequence is not None and phase4_time > 30:
        # Run multiple short LP passes with different parameters
        polish_configs = [
            {'alpha': 0.3, 'seed': 101, 'sota': False},
            {'alpha': 0.5, 'seed': 202, 'sota': True},
            {'alpha': 0.7, 'seed': 303, 'sota': False},
            {'alpha': 0.4, 'seed': 404, 'sota': True},
        ]
        time_per_polish = max(10.0, (phase4_time - 30) / len(polish_configs))

        for cfg in polish_configs:
            if time.monotonic() - t_phase4 > phase4_time - 25:
                break
            t_alloc = min(time_per_polish, phase4_time - 25 - (time.monotonic() - t_phase4))
            if t_alloc < 8:
                break

            a_opt, obj_opt = lp_descent(best_sequence, time_limit=t_alloc,
                                         alpha_init=cfg['alpha'],
                                         seed=cfg['seed'],
                                         use_sota_lp=cfg['sota'])
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Polish (a={cfg['alpha']}, sota={cfg['sota']}): {score:.6f}")

        # Tiny perturbations
        remaining = deadline - time.monotonic() - 20
        if remaining > 15:
            n_best = len(best_sequence)
            n_tries = min(int(remaining / 2), 40)
            for i in range(n_tries):
                if time.monotonic() > deadline - 25:
                    break
                improved = False
                for scale in [0.0005, 0.001, 0.002, 0.005, 0.01]:
                    noise = rng.standard_normal(n_best) * scale
                    a_pert = project_simplex(best_sequence + noise)
                    score_pert = float(evaluate_sequence(a_pert.tolist()))
                    if np.isfinite(score_pert) and score_pert < best_score:
                        best_score = score_pert
                        best_sequence = a_pert.copy()
                        print(f"  Perturb[{i}] s={scale}: {score_pert:.6f}")
                        improved = True
                        break
                if not improved and i < 5:
                    # Try directional perturbations
                    conv = fast_convolve(best_sequence)
                    peak_idx = conv.argmax()
                    # Reduce elements that contribute to the peak
                    a_dir = best_sequence.copy()
                    i_min = max(0, peak_idx - n_best + 1)
                    i_max = min(n_best - 1, peak_idx)
                    for j in range(i_min, i_max + 1):
                        a_dir[j] *= 0.95
                        a_dir[peak_idx - j] *= 0.95
                    a_dir = project_simplex(a_dir)
                    score_dir = float(evaluate_sequence(a_dir.tolist()))
                    if np.isfinite(score_dir) and score_dir < best_score:
                        best_score = score_dir
                        best_sequence = a_dir.copy()
                        print(f"  DirPerturb[{i}]: {score_dir:.6f}")

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(200) / 200

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(200) / 200
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
