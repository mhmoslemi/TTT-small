"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using sparse support optimization
with LP-based weight optimization on the reduced support space.
Combines support selection (greedy add/remove) with weight optimization
(LP direction finding + line search) for efficient global search.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def conv_sparse(a, support):
    """Compute convolution using only the support positions.
    Returns dict mapping position -> value."""
    m = len(support)
    conv = {}
    for i in range(m):
        for j in range(i, m):
            pos = support[i] + support[j]
            if i == j:
                conv[pos] = conv.get(pos, 0) + a[support[i]] * a[support[j]]
            else:
                conv[pos] = conv.get(pos, 0) + 2 * a[support[i]] * a[support[j]]
    return conv

def conv_sparse_array(a, support, n):
    """Compute convolution as array, using support for efficiency."""
    full_conv = np.zeros(2 * n - 1)
    m = len(support)
    for i in range(m):
        for j in range(i, m):
            pos = support[i] + support[j]
            if i == j:
                full_conv[pos] += a[support[i]] * a[support[j]]
            else:
                full_conv[pos] += 2 * a[support[i]] * a[support[j]]
    return full_conv

def cross_conv_sparse(a, support, g, support_g, n):
    """Compute cross-convolution of a (full) and g (sparse on support_g).
    Returns array of length 2*n-1."""
    result = np.zeros(2 * n - 1)
    m = len(support_g)
    for j_idx in range(m):
        pos_j = support_g[j_idx]
        gj = g[j_idx]
        # conv[k] includes a[k-pos_j] * gj for valid k-pos_j
        # k ranges from pos_j to pos_j + n - 1
        k_start = pos_j
        k_end = min(2 * n - 2, pos_j + n - 1)
        # a[k - pos_j] for k in [k_start, k_end]
        idx_start = max(0, k_start - (n - 1))  # not needed, k starts at pos_j >= 0
        # a[i] * gj where i = k - pos_j, so k = i + pos_j
        # i ranges from 0 to n-1, k from pos_j to pos_j+n-1
        valid_i_start = 0
        valid_i_end = n - 1
        result[pos_j:pos_j + n] += a[:n] * gj
    return result

def obj_from_conv(conv, n):
    """Compute objective from convolution array."""
    return 2.0 * n * conv.max()

def lp_direction_on_support(a, support, M, n):
    """
    Solve LP on the support: maximize sum(g) s.t. cross_conv(a, g) <= M, g >= 0.
    Here g is defined only on the support positions.
    """
    m = len(support)
    # Build constraint matrix: for each convolution position k,
    # cross_conv[k] = sum_j g[j] * a[k - support[j]]
    # This is linear in g.
    # Number of constraints: 2*n - 1
    # Number of variables: m
    
    # For efficiency, only include constraints that could be binding.
    # Actually, let's just build the full matrix.
    # A_ub has shape (2*n-1, m)
    # A_ub[k, j] = a[k - support[j]] if 0 <= k - support[j] < n, else 0
    
    # Build sparsely
    A_rows = []
    A_cols = []
    A_vals = []
    for j in range(m):
        sj = support[j]
        # a[k - sj] is non-zero for k in [sj, sj+n-1]
        k_start = sj
        k_end = min(2 * n - 2, sj + n - 1)
        # a[k-sj] for k in [k_start, k_end] is a[0], a[1], ..., a[k_end-sj]
        for i in range(n):
            k = sj + i
            if k < 2 * n - 1:
                A_rows.append(k)
                A_cols.append(j)
                A_vals.append(a[k - sj])  # = a[i]
    
    # Use sparse matrix
    from scipy.sparse import csc_matrix
    A_sparse = csc_matrix((A_vals, (A_rows, A_cols)), shape=(2*n-1, m))
    
    c = -np.ones(m)  # minimize -sum(g)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * m
    
    result = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    return None

def optimize_weights_lp(a, support, n, time_limit, verbose=False):

    """
    m = len(support)

    s = w.sum()
    if s < 1e-15:
        w = np.ones(m) / m
    else:
        w = w / s

    def make_full(w):
        full = np.zeros(n)
        full[support] = w
        return full
    
    a_full = make_full(w)
    conv = fast_convolve(a_full)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_w = w.copy()
    
    t0 = time.monotonic()
    iter_count = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        a_full = make_full(w)
        
        # Solve LP for direction
        g = lp_direction_on_support(a_full, support, M * 1.001, n)
        
        if g is None or g.sum() <= 1.0 + 1e-8:
            no_improve += 1
            if no_improve > 100:
                # Perturb
                rng = np.random.default_rng(iter_count)
                w = best_w + rng.standard_normal(m) * 0.01
                w = np.maximum(w, 0)
                s = w.sum()
                if s > 1e-15:
                    w = w / s
                no_improve = 0
            iter_count += 1
            continue
        
        d = g / g.sum() - w  # direction in normalized space
        
        # Line search
        best_t = 0.0
        best_obj_ls = best_obj
        for t in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
            w_try = w + t * d
            w_try = np.maximum(w_try, 0)
            s_try = w_try.sum()
            if s_try < 1e-15:
                break
            w_try = w_try / s_try
            a_try = make_full(w_try)
            conv_try = fast_convolve(a_try)
            obj_try = 2.0 * n * conv_try.max()
            if obj_try < best_obj_ls - 1e-14:
                best_obj_ls = obj_try
                best_t = t
        
        if best_t > 0 and best_obj_ls < best_obj - 1e-13:
            w = w + best_t * d
            w = np.maximum(w, 0)
            s = w.sum()
            if s > 1e-15:
                w = w / s
            a_full = make_full(w)
            conv = fast_convolve(a_full)
            M = conv.max()
            best_obj = 2.0 * n * M
            best_w = w.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        iter_count += 1
    
    return best_w, best_obj

def optimize_weights_gradient(a, support, n, time_limit, lr=0.01, verbose=False):
    """
    Optimize weights using gradient descent with multiplicative updates.
    """
    m = len(support)
    w = a[support].copy()
    s = w.sum()
    if s < 1e-15:
        w = np.ones(m) / m
    else:
        w = w / s
    
    def make_full(w):
        full = np.zeros(n)
        full[support] = w
        return full
    
    best_w = w.copy()
    best_obj = float('inf')
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        a_full = make_full(w)
        conv = fast_convolve(a_full)
        obj = 2.0 * n * conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_w = w.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            break
        
        # Find active peaks
        threshold = conv.max() * 0.9
        active = np.where(conv > threshold)[0]
        
        # Compute gradient of max(conv) w.r.t. w
        # For each active position k, grad_w_j = 2 * a[k - support[j]] (if valid)
        grad = np.zeros(m)
        for k in active:
            for j in range(m):
                idx = k - support[j]
                if 0 <= idx < n:
                    grad[j] += 2.0 * a_full[idx]
        
        # Normalize gradient
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        
        # Multiplicative update
        eta = lr / (1.0 + no_improve * 0.01)
        w = w * np.exp(-eta * grad / g_norm)
        
        # Renormalize
        s = w.sum()
        if s < 1e-15:
            w = best_w.copy()
        else:
            w = w / s
    
    return best_w, best_obj

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using sparse support optimization
with LP-based direction finding and adaptive peak-shaving. Exploits the
non-convexity through diverse random initializations on sparse supports,
with iterative LP refinement and support modification.

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_sparse_cross_matrix(a, support, n):
    """Build sparse matrix A where A @ g = cross_conv(a, g_sparse).
    g_sparse is defined on support positions."""
    m = len(support)
    rows = []
    cols = []
    vals = []
    for j in range(m):
        sj = support[j]
        # cross_conv[k] = sum_j g[j] * a[k - support[j]]
        # For column j: a[k - sj] for k in [sj, sj+n-1]
        k_start = sj
        k_end = min(2 * n - 2, sj + n - 1)
        length = k_end - k_start + 1
        rows.extend(range(k_start, k_end + 1))
        cols.extend([j] * length)
        vals.extend(a[:length])  # a[0], a[1], ..., a[length-1]
    A = csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, m))
    return A

def solve_lp(a, support, M, n, A_cache=None):
    """Solve LP: max sum(g) s.t. cross_conv(a,g) <= M, g >= 0."""
    m = len(support)
    if A_cache is None:
        A_cache = build_sparse_cross_matrix(a, support, n)
    c = -np.ones(m)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * m
    try:
        result = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None

def peak_shave(a, support, n, top_k=20, strength=0.1):
    """Reduce peaks in convolution by adjusting weights."""
    m = len(support)
    conv = fast_convolve(a)
    threshold = conv.max() * 0.85
    active = np.where(conv > threshold)[0]
    
    w = a[support].copy()
    # Compute blame: for each support element, how much does it contribute to peaks?
    blame = np.zeros(m)
    for k in active:
        for j in range(m):
            idx = k - support[j]
            if 0 <= idx < n:
                blame[j] += a[idx]
    
    # Reduce blamed elements, increase others
    max_blame = blame.max()
    if max_blame < 1e-15:
        return w
    w = w * np.exp(-strength * blame / max_blame)
    s = w.sum()
    if s < 1e-15:
        return a[support]
    return w / s
