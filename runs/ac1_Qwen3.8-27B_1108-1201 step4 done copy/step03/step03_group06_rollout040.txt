    n = len(a)

            j = k - i
            G[k, i] += a[j]
            if j != i:
                G[k, j] += a[i]
    return G

def solve_lpr(a, conv):
    """
    Solve the linearized peak reduction LP.
    min t s.t. G[k,:]@d - t <= -conv[k], sum(d)=0, d>=-a
    Returns direction d or None if infeasible.
    """
    n = len(a)
    G = build_gradient_matrix(a)
    num_k = 2 * n - 1
    
    # Objective: minimize t
    c = np.concatenate([np.zeros(n), [1.0]])

    A_ub = np.zeros((num_k, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -conv

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])

    bounds = [(-a[i], None) for i in range(n)] + [(0.0, None)]

                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            d = res.x[:n]
            return d
    except Exception:
        pass
    return None

            if s_new < 1e-15:

            print(f"  LPR iter: obj={best_obj:.8f}, step_scale={step_scale:.3f}")

Linearized Peak Reduction (LPR) optimization for minimizing 2*n*max(conv(a,a))
on the probability simplex. Uses a sequence of LPs that solve the first-order
linearization of the peak reduction subproblem, combined with global LP kicks
and multi-length exploration.

import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def build_gradient_matrix(a):
    """Build G[k,j] = d(conv[k])/d(a[j]) for conv = np.convolve(a, a)."""
    n = len(a)
    num_k = 2 * n - 1
    G = np.zeros((num_k, n))
    for k in range(num_k):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            G[k, i] += a[j]
            if j != i:
                G[k, j] += a[i]
    return G


def solve_lpr(a, conv):

    Solve: min t s.t. G[k,:]>=d - t <= -conv[k] for all k,
             sum(d) = 0, d >= -a.
    Returns direction d or None.

    G = build_gradient_matrix(a)
    num_k = 2 * n - 1

    c = np.zeros(n + 1)
    c[n] = 1.0

    A_ub = np.zeros((num_k, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -conv.copy()

    A_eq[0, :n] = 1.0

                      options={'presolve': True, 'time_limit': 5.0})
        if res.success:
            d = res.x[:n]
            if np.abs(d).max() > 1e-15:
                return d
    except Exception:
        pass
    return None

Linearized Peak Reduction (LPR) optimization for minimizing 2*n*max(conv(a,a))
on the probability simplex. Uses LP-based first-order direction finding with
adaptive line search, combined with global LP kicks, multi-length exploration,
and perturbation-based escape from local minima.

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])

    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    step_scale = 1.0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()

        d = solve_lpr(a, conv)
        improved = False

        if d is not None:
            d = d * step_scale
            for alpha in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]:
                a_new = a + alpha * d
                if a_new.min() < -1e-10:
                    continue
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    continue
                a_new = a_new / s_new
                conv_new = fast_convolve(a_new)
                obj_new = conv_new.max()
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    improved = True
                    no_improve = 0
                    break

        if not improved:
            # Try global LP kick
            b = solve_global_lp(a, M, top_k=max(20, n // 8), conv=conv)
            if b is not None:
                b_sum = b.sum()
                if b_sum > 1e-15:
                    b = b / b_sum
                    for mix in [0.5, 0.3, 0.7]:
                        a_new = (1.0 - mix) * a + mix * b
                        a_new = np.maximum(a_new, 0.0)
                        s_new = a_new.sum()
                        if s_new < 1e-15:
                            continue
                        a_new = a_new / s_new
                        conv_new = fast_convolve(a_new)
                        obj_new = conv_new.max()
                        if obj_new < best_obj - 1e-14:
                            best_obj = obj_new
                            best_a = a_new.copy()
                            a = a_new
                            improved = True
                            no_improve = 0
                            break

        if not improved:
            no_improve += 1
            if no_improve == 3:
                step_scale *= 0.6
            elif no_improve == 8:
                noise = rng.standard_normal(n) * 0.002
                a = project_simplex(a + noise)
                step_scale = 1.0
                no_improve = 0
            elif no_improve == 15:
                noise = rng.standard_normal(n) * 0.008
                a = project_simplex(a + noise)
                step_scale = 1.0
                no_improve = 0
            elif no_improve == 30:
                noise = rng.standard_normal(n) * 0.02
                a = project_simplex(a + noise)
                step_scale = 1.0
                no_improve = 0
            elif no_improve > 60:
                break

        iter_count += 1
        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPR iter {iter_count}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n (normalized to sum=1)."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp_inc'] = ramp
    inits['ramp_dec'] = ramp[::-1].copy()

    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    for sf in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits['gauss_%s' % str(sf)] = g

    if n > 1:
        cos_w = np.cos(np.pi * x / (n - 1)) ** 2
        cos_w = np.maximum(cos_w, 0.01)
        cos_w /= cos_w.sum()
    else:
        cos_w = np.ones(1)
    inits['cosine'] = cos_w

    inits['random1'] = rng.dirichlet(np.ones(n) * 0.1)
    inits['random2'] = rng.dirichlet(np.ones(n) * 1.0)

    alt = np.ones(n)
    alt[::2] = 2.0
    alt /= alt.sum()
    inits['alternating'] = alt

    tri = np.minimum(x, n - 1 - x).astype(np.float64) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri

    # Two delta-like bumps
    c1 = n // 4
    c2 = 3 * n // 4
    w = max(1, n // 20)
    two_bump = np.sum(np.exp(-0.5 * ((x - nc) / w) ** 2) for nc in [c1, c2])
    two_bump /= two_bump.sum()
    inits['two_bump'] = two_bump

    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_crop'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_interp'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps

    return inits


def try_update(a, best_score, best_sequence, label):
    """Try to update best. Returns (new_best_score, new_best_sequence, improved)."""
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        return score, a.copy(), True
    return best_score, best_sequence, False


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase LPR optimization:
    Phase 1: Quick exploration over lengths and initializations
    Phase 2: Deep LPR optimization on best candidates
    Phase 3: Polish with perturbations and nearby lengths
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print("Parent: length=%d, sum=%.2f" % (len(p), p.sum()))
    except NameError:
        pass
    except Exception:
        pass

    # ========== Phase 1: Quick Exploration ==========
    print("=== Phase 1: Quick Exploration ===")
    t_phase1 = time.monotonic()
    phase1_time = 140.0

    lengths = [256, 320, 384, 512]
    scan_results = []

    for n in lengths:
        if time.monotonic() + 15 > t_phase1 + phase1_time:
            break
        inits = generate_initializations(n, parent, rng)
        init_names = list(inits.keys())

        priority_order = ['parent_interp', 'parent_crop', 'parent_ds', 'bimodal',
                          'uniform', 'alternating', 'gauss_0.25', 'random1', 'cosine']
        ordered = [nm for nm in priority_order if nm in inits] + \
                  [nm for nm in init_names if nm not in priority_order]
        init_names = ordered[:4]

        for name in init_names:
            if time.monotonic() + 6 > t_phase1 + phase1_time:
                break
            t_alloc = min(12.0, t_phase1 + phase1_time - time.monotonic() - 6)
            if t_alloc < 4:
                break

            a0 = inits[name]
            a_opt, obj_opt = lpr_optimize(a0, time_limit=t_alloc,
                                          seed=n * 7 + hash(name) % 10000)
            new_score, new_seq, improved = try_update(
                a_opt, best_score, best_sequence, "P1 n=%d %s" % (n, name))
            if improved:
                best_score = new_score
                best_sequence = new_seq
                print("  NEW BEST (P1 %s n=%d): score=%.6f" % (name, n, best_score))
            scan_results.append((obj_opt, n, name, a_opt.copy()))

    scan_results.sort(key=lambda x: x[0])

    # ========== Phase 2: Deep LPR Optimization ==========
    print("=== Phase 2: Deep Optimization ===")
    phase2_start = time.monotonic()

    deep_starts = []

    if best_sequence is not None:

        if len(deep_starts) >= 5:
            break
        skip = False
        for existing in deep_starts:

                skip = True
                break
        if not skip:
            deep_starts.append(a_opt.copy())

    if parent is not None and len(parent) <= 600:
        pp = np.maximum(parent, 0)
        ps = pp.sum()
        if ps > 1e-10:
