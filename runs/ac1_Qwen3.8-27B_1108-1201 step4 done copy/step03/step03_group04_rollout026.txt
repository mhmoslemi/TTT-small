But without the sum constraint, the objective is scale-invariant, so we'd have a degree of freedom. We need to fix the scale somehow. Let's fix sum(a) = 1 and handle the constraint in coordinate descent.

Actually, for coordinate descent on the simplex:
- Fix all a_i for i ≠ j, let s = sum_{i≠j} a_i
- We need a_j = 1 - s (forced by the simplex constraint)
- So there's no freedom! We can't do coordinate descent on the simplex in the usual way.

Instead, we can do:
- Fix all a_i for i ≠ j, l (two coordinates)
- Let a_j + a_l = c (fixed by the sum constraint)
- Optimize the split: a_j = t, a_l = c - t, with 0 ≤ t ≤ c
- This is a 1D optimization

Or better: do "block coordinate descent" where we update a group of coordinates.

Actually, the simplest approach for the simplex: parameterize as a_j for j = 0, ..., n-2 and a_{n-1} = 1 - sum_{j=0}^{n-2} a_j. Then do coordinate descent on the first n-1 coordinates with the constraint that each is in [0, 1 - sum of others].

    n = len(a0)
    a = a0.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    v = np.zeros(n)  # momentum
    momentum = 0.9
    lr = 0.01
    mu = mu_init
    
    best_obj = compute_local_obj(a)
    best_a = a.copy()
    best_mu = mu
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute smoothed gradient
        conv = fast_convolve(a)
        max_conv = conv.max()
        exp_vals = np.exp((conv - max_conv) / mu)
        weights = exp_vals / exp_vals.sum()
        
        # Gradient via FFT cross-correlation
        fft_size = 1 << (3*n - 2 - 1).bit_length()
        W = np.fft.rfft(weights, fft_size)
        A = np.fft.rfft(a, fft_size)
        # cross_corr[j] = sum_k weights[k] * a[k-j]
        # = ifft(FFT(weights) * conj(FFT(a)))[j]
        # For rfft, we need to be careful
        # Let's just use full fft
        W_full = np.fft.fft(weights, fft_size)
        A_full = np.fft.fft(a, fft_size)
        cross_corr = np.fft.ifft(W_full * np.conj(A_full)).real[:n]
        grad = 2.0 * cross_corr
        
        # Nesterov momentum
        v = momentum * v - lr * grad
        a_new = np.maximum(a + v, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        
        # Evaluate
        obj_new = compute_local_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Backtrack: try smaller step
            for factor in [0.5, 0.25, 0.1]:
                a_try = np.maximum(a + factor * v, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                obj_try = compute_local_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    a_new = a_try
                    best_obj = obj_try
                    best_a = a_new.copy()
                    no_improve = 0
                    lr *= 0.5
                    break
            else:
                lr *= 0.8
        
        a = a_new
        mu *= mu_decay
        mu = max(mu, mu_min)
        it += 1
        
        if no_improve > 100:
            break
    
    return best_a, best_obj

"""
Hybrid optimizer combining LP-based Frank-Wolfe with mirror descent,
adaptive peak targeting, and diverse restarts to minimize
2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Key improvements over previous: faster LP with warm starts, log-sum-exp
smoothed mirror descent between LP calls, adaptive peak weighting,
and structured perturbation for escape from local minima.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, coo_matrix


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    if b is a:
        B = A
    else:
        B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def compute_local_obj(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)~1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(a):
    """Build sparse matrix A such that (A @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_oracle(a, M, time_limit=2.0):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': time_limit})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def lp_oracle_relaxed(a, M, relax=1.05, time_limit=2.0):
    """LP with relaxed constraint to allow more exploration."""
    return lp_oracle(a, M * relax, time_limit)


def mirror_descent_step(a, grad, lr):
    """Exponential weights / mirror descent step (stays in simplex)."""
    n = len(a)
    log_a = np.where(a > 1e-30, np.log(a), -30.0)
    new_log_a = log_a - lr * grad
    new_log_a -= new_log_a.max()  # numerical stability
    a_new = np.exp(new_log_a)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_new / s


def compute_smoothed_gradient(a, mu, top_k=None):
    """Compute gradient of log-sum-exp smoothed max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    # Use top-k peaks for sharpness
    if top_k is not None and top_k < len(conv):
        threshold_idx = np.argsort(conv)[-top_k]
        mask = np.zeros(len(conv), dtype=bool)
        mask[threshold_idx] = True
        conv_masked = np.where(mask, conv, -np.inf)
    else:
        conv_masked = conv
    
    max_cv = conv_masked.max()
    if not np.isfinite(max_cv):
        max_cv = conv.max()
        conv_masked = conv
    
    exp_vals = np.exp((conv_masked - max_cv) / mu)
    exp_vals[exp_vals < 1e-30] = 0.0
    total = exp_vals.sum()
    if total < 1e-30:
        # Fallback: uniform weights on top peaks
        peaks = np.where(conv_masked > max_cv - 1e-10)[0]
        weights = np.zeros(len(conv))
        if len(peaks) > 0:
            weights[peaks] = 1.0 / len(peaks)
        else:
            weights[conv.argmax()] = 1.0
    else:
        weights = exp_vals / total
    
    # Gradient via FFT cross-correlation
    # grad[j] = 2 * sum_k weights[k] * a[k-j]
    fft_size = 1 << (3 * n - 3).bit_length()
    W = np.fft.fft(weights, fft_size)
    A = np.fft.fft(a, fft_size)
    cross_corr = np.fft.ifft(W * np.conj(A)).real[:n]
    grad = 2.0 * cross_corr
    return grad, conv


def backtracking_line_search(a, direction, obj_current, max_steps=20):
    """Backtracking line search on the simplex."""
    n = len(a)
    best_t = 0.0
    best_obj = obj_current
    t = 1.0
    for _ in range(max_steps):
        candidate = (1 - t) * a + t * direction
        candidate = np.maximum(candidate, 0)
        s = candidate.sum()
        if s < 1e-15:
            t *= 0.5
            continue
        candidate = candidate / s
        obj = compute_local_obj(candidate)
        if obj < obj_current:
            best_t = t
            best_obj = obj
        t *= 0.5
        if t < 0.01:
            break
    return best_t, best_obj


def fpw_optimize(a0, time_limit, verbose=False, verbose_every=500,

    """LP-based Frank-Wolfe with adaptive features."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    max_stall = 150
    lp_count = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 5:
            break

        # Solve LP oracle
        b = lp_oracle(a, M, time_limit=lp_time)
        if b is None:
            # Try relaxed
            b = lp_oracle_relaxed(a, M, relax=relax_factor, time_limit=lp_time)
        if b is None:
            no_improve += 10
        else:
            lp_count += 1
            sb = b.sum()
            if sb < 1e-15:
                no_improve += 10
            else:
                b = b / sb
                # Line search
                t, obj_new = backtracking_line_search(a, b, best_obj, max_steps=8)
                if t < 0.005:
                    no_improve += 1
                else:
                    a_new = (1 - t) * a + t * b
                    a_new = np.maximum(a_new, 0)
                    a_new = a_new / a_new.sum()
                    conv_new = fast_convolve(a_new)
                    M = conv_new.max()
                    cur_obj = 2.0 * n * M
                    if cur_obj < best_obj - 1e-14:
                        a = a_new
                        best_obj = cur_obj
                        best_a = a.copy()
                        no_improve = 0
                    else:
                        a = a_new
                        no_improve += 1

        # Stall handling
        if no_improve >= max_stall:
            rng = np.random.default_rng(hash((it, n)) & 0xFFFFFFFF)
            # Structured perturbation: perturb in directions that reduce peaks
            conv_cur = fast_convolve(a)
            peak_idx = conv_cur.argmax()
            # Reduce mass near the peak-contributing positions
            peak_j_start = max(0, peak_idx - n + 1)
            peak_j_end = min(n - 1, peak_idx)
            noise = rng.standard_normal(n) * 0.005
            # Extra reduction at peak-contributing positions

            a = np.maximum(a + noise, 0)
            a = a / a.sum()
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            max_stall = int(max(max_stall * 1.5, 100))

        it += 1
        if verbose and it % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"    FPW it={it}, lp={lp_count}, obj={best_obj:.8f}, M={M:.6e}, elapsed={elapsed:.1f}s")

    return best_a, best_obj


def mirror_descent_optimize(a0, time_limit, lr_init=0.05, mu_init=0.005,
                            mu_decay=0.998, mu_min=0.0005, top_k=10,
                            verbose=False, verbose_every=2000):

    mu = mu_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        grad, conv = compute_smoothed_gradient(a, mu, top_k=top_k)

        # Mirror descent step
        a_new = mirror_descent_step(a, grad, lr)
        obj_new = compute_local_obj(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Try smaller step
            a_try = mirror_descent_step(a, grad, lr * 0.3)
            obj_try = compute_local_obj(a_try)
            if obj_try < best_obj - 1e-13:
                best_obj = obj_try
                best_a = a_try.copy()
                a_new = a_try
                no_improve = 0
                lr *= 0.5
            else:
                lr *= 0.7

        a = a_new
        mu = max(mu * mu_decay, mu_min)
        lr = min(lr * 1.001, 0.5)  # very slow lr growth

        if no_improve > 500:
            # Perturb

            noise = rng.standard_normal(n) * 0.002

            a = a / a.sum()
            no_improve = 0

        it += 1
        if verbose and it % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD it={it}, obj={best_obj:.8f}, mu={mu:.5f}, lr={lr:.4f}, elapsed={elapsed:.1f}s")

    return best_a, best_obj


def coordinate_descent_polish(a0, time_limit, verbose=False):
    """Polish using coordinate descent on pairs."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    a = a / a.sum()
    best_obj = compute_local_obj(a)
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        # Pick a random pair of coordinates to optimize
        j = rng.integers(0, n)
        l = rng.integers(0, n)
        if l == j:
            l = (j + 1) % n

        c = a[j] + a[l]
        if c < 1e-15:
            continue

        # Try different splits of the mass c between j and l
        best_split = a[j] / c if c > 0 else 0.5
        best_split_obj = best_obj

        for frac in np.linspace(0, 1, 21):
            a[j] = frac * c
            a[l] = (1 - frac) * c
            obj = compute_local_obj(a)
            if obj < best_split_obj - 1e-14:
                best_split_obj = obj
                best_split = frac

        a[j] = best_split * c
        a[l] = (1 - best_split) * c

        if best_split_obj < best_obj - 1e-14:
            best_obj = best_split_obj
            best_a = a.copy()

    return best_a, best_obj


def hybrid_optimize(a0, time_limit, verbose=True):
    """Combined LP-FW and mirror descent optimizer."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    a = a / a.sum()

    best_a = a.copy()

    phase_flip = 0
    stall_count = 0
    max_stall = 80

        else:

                                                     lr_init=0.02, mu_init=0.003,
                                                     top_k=8)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
            # Try the new point anyway (might help later)
            a = a_new.copy() * 0.5 + best_a * 0.5
            a = np.maximum(a, 0)
            a = a / a.sum()

        if stall_count >= max_stall:
            rng = np.random.default_rng(int(time.monotonic() * 1000))
            noise = rng.standard_normal(n) * 0.003
            a = np.maximum(best_a + noise, 0)
            a = a / a.sum()
            stall_count = 0
            max_stall = int(max_stall * 1.4)

        phase_flip += 1

        if verbose and phase_flip % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Hybrid phase={phase_flip}, obj={best_obj:.8f}, elapsed={elapsed:.1f}s")

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}

    # Parent-based
    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0.0)
        ps = pp.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = pp / ps
            else:
                pp_interp = np.interp(np.linspace(0, 1, n),
                                      np.linspace(0, 1, len(parent)), pp)
                ps2 = pp_interp.sum()
                if ps2 > 1e-10:
                    inits['parent_interp'] = pp_interp / ps2

    # Symmetric patterns
    if 'parent' in inits:
        # Symmetrized parent
        p = inits['parent'].copy()
        sym = (p + p[::-1]) / 2
        sym = np.maximum(sym, 0)
        s = sym.sum()
        if s > 1e-10:
            inits['parent_sym'] = sym / s

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Ramped
    r = np.arange(1, n + 1, dtype=np.float64)
    r /= r.sum()
    inits['ramp'] = r.copy()

    # Edge concentrated
    edge = max(2, n // 4)
    ec = np.zeros(n)
    ec[:edge] = 1.0
    ec[-edge:] = 1.0
    ec /= ec.sum()
    inits['edge'] = ec.copy()

    # Gaussian
    x = np.arange(n, dtype=np.float64)
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g.copy()

    # Random sparse
    sp = rng.random(n)
    sp[sp < 0.4] = 0.0
    if sp.sum() > 1e-10:
        inits['sparse'] = sp / sp.sum()

    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Riemann-like
    r2 = 1.0 / np.arange(1, n + 1)
    inits['riemann'] = r2 / r2.sum()

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase hybrid search: LP-FW + mirror descent + coordinate polish.
    Explores multiple lengths and initializations, then deeply optimizes the best.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')
    best_n = 0

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.2f}")
    except Exception:
        pass

    def try_sequence(seq, label=""):
        nonlocal best_sequence, best_score, best_n
        if seq is None:
            return
        seq = np.maximum(np.asarray(seq, dtype=np.float64), 0.0)
        if seq.sum() < 1e-10:
            return
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = seq.copy()
                best_n = len(seq)
                if label:
                    print(f"  NEW BEST: {score:.8f} ({label}, n={len(seq)})")
            return score
        return None

    # === Phase 1: Quick exploration across lengths and inits ===
    print("=== Phase 1: Exploration ===")
    t0 = time.monotonic()

    # Focus on parent length and a couple others
    if parent is not None:
        parent_n = len(parent)
        lengths = sorted(set([parent_n, max(128, parent_n // 2), parent_n + 128, parent_n + 256]))
    else:
        lengths = [256, 384, 512]

    # Filter to reasonable sizes
    lengths = [n for n in lengths if 64 <= n <= 1024]

    for n in lengths:
        if time.monotonic() > deadline - 200:
            break
        print(f"  Exploring n={n}...")
        inits = generate_initializations(n, parent, rng)
        init_names = list(inits.keys())

        # Priority order
        priority = ['parent', 'parent_interp', 'parent_sym', 'uniform',
                    'edge', 'gauss', 'ramp', 'riemann', 'sparse', 'random']
        init_names.sort(key=lambda x: priority.index(x) if x in priority else 99)

        # Quick optimization for each init (mirror descent, fast)
        for name in init_names[:5]:
            if time.monotonic() > deadline - 150:
                break
            t_alloc = min(25.0, (deadline - time.monotonic() - 100) / max(1, len(init_names[:5])))
            if t_alloc < 3:
                break

            a0 = inits[name]
            # Quick mirror descent
            a_opt, _ = mirror_descent_optimize(a0, time_limit=t_alloc,
                                               lr_init=0.03, mu_init=0.01,
                                               top_k=15)
            score = try_sequence(a_opt, f"n={n},init={name},MD")

        # Also try quick LP-FW on best init
        if time.monotonic() < deadline - 100 and 'parent' in inits:
            t_alloc = min(30.0, deadline - time.monotonic() - 80)
            if t_alloc > 5:
                a_opt, _ = fpw_optimize(inits['parent'], time_limit=t_alloc,
                                        verbose=False, lp_time=1.0)
                try_sequence(a_opt, f"n={n},init=parent,LPFW")

    print(f"  After Phase 1: best={best_score:.8f}, n={best_n}, elapsed={time.monotonic()-t0:.1f}s")

    # === Phase 2: Deep hybrid optimization ===
    if best_sequence is not None and time.monotonic() < deadline - 100:
        print("=== Phase 2: Deep Hybrid Optimization ===")
        t0 = time.monotonic()
        remaining = deadline - time.monotonic() - 80
        t_deep = remaining * 0.6

        a_opt, _ = hybrid_optimize(best_sequence, time_limit=t_deep, verbose=True)
        try_sequence(a_opt, "Phase2-hybrid")

        # Also dedicated LP-FW
        remaining2 = deadline - time.monotonic() - 60
        if remaining2 > 20:
            t_lp = remaining2 * 0.6
            a_opt, _ = fpw_optimize(best_sequence, time_limit=t_lp,
                                    verbose=True, verbose_every=200,
                                    lp_time=2.0, relax_factor=1.01)
            try_sequence(a_opt, "Phase2-LPFW")

    # === Phase 3: Coordinate descent polish ===
    if best_sequence is not None and time.monotonic() < deadline - 30:
        print("=== Phase 3: Coordinate Polish ===")
        remaining = deadline - time.monotonic() - 20
        t_cd = remaining * 0.5
        if t_cd > 5:
            a_opt, _ = coordinate_descent_polish(best_sequence, time_limit=t_cd)
            try_sequence(a_opt, "Phase3-CD")

    # === Phase 4: Final LP refinement ===
    if best_sequence is not None and time.monotonic() < deadline - 10:
        print("=== Phase 4: Final LP ===")
        remaining = deadline - time.monotonic() - 5
        if remaining > 5:
            a_opt, _ = fpw_optimize(best_sequence, time_limit=remaining,
                                    verbose=False, lp_time=1.5, relax_factor=1.0)
            try_sequence(a_opt, "Phase4-finalLP")

    # === Phase 5: Perturbation restarts ===
    if best_sequence is not None and time.monotonic() < deadline - 15:
        print("=== Phase 5: Perturbation Restarts ===")
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 10
        n_restarts = min(max(1, int(remaining // 20)), 6)

        for i in range(n_restarts):
            if time.monotonic() > deadline - 8:
                break
            scale = 0.0005 * (1 + i * 0.3)
            noise = rng.standard_normal(n_best) * scale
            a_pert = np.maximum(best_sequence + noise, 0)
            a_pert = a_pert / a_pert.sum()
            t_alloc = min(15.0, deadline - time.monotonic() - 5)
            if t_alloc < 3:
                break
            a_opt, _ = fpw_optimize(a_pert, time_limit=t_alloc, verbose=False, lp_time=1.0)
            try_sequence(a_opt, f"Restart{i}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence([float(x) for x in best_sequence]))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        # Should not happen, but safety
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
