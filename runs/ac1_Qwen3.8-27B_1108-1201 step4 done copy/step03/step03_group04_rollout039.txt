    temp = max(peak_vals.max() * 0.1, 1e-15)

def fast_convolve(a, b=None):
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def spectral_init(n, n_iters=200, rng=None):
    """Gerchberg-Saxton style initialization targeting flat spectrum."""
    if rng is None:

    target_mag = 1.0 / np.sqrt(n)  # flat magnitude

        phase = np.angle(A)
        # Set magnitude to target
        A_new = target_mag * np.exp(1j * phase)

        if s < 1e-15:

            a = a / a.sum()

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2:
1. Spectral (Gerchberg-Saxton) initialization with diverse targets
2. Nesterov-accelerated multi-peak subgradient descent
3. LP-based Frank-Wolfe polish
4. Perturbation restarts

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def gs_init(n, n_iters=500, rng=None, target_type='flat', alpha=1.0):
    """
    Gerchberg-Saxton initialization.
    target_type: 'flat', 'raised_dc', 'raised_edges', 'random_phase'
    """
    if rng is None:
        rng = np.random.default_rng()
    
    a = rng.random(n) + 0.1
    a = a / a.sum()
    
    # Target magnitude spectrum
    freqs = np.fft.rfftfreq(n)
    if target_type == 'flat':
        target_mag = np.ones(n // 2 + 1) / np.sqrt(n)
    elif target_type == 'raised_dc':
        target_mag = np.ones(n // 2 + 1) / np.sqrt(n)
        target_mag[0] = (1 + alpha) / np.sqrt(n)
        # Renormalize
        target_mag /= np.sqrt(np.sum(target_mag**2))
    elif target_type == 'raised_edges':
        target_mag = np.ones(n // 2 + 1) / np.sqrt(n)
        # Raise edges
        target_mag[-max(1, n//10):] *= (1 + alpha)
        target_mag /= np.sqrt(np.sum(target_mag**2))
    else:  # random_phase
        target_mag = np.ones(n // 2 + 1) / np.sqrt(n)
    
    for it in range(n_iters):
        A = np.fft.rfft(a)
        phase = np.angle(A)
        # Preserve DC phase as positive
        A[0] = abs(A[0])
        A_new = target_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, n)
        a = np.maximum(a_new, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = rng.random(n) + 0.1
            a = a / a.sum()
        else:
            a = a / s
    
    return a


def compute_subgradient(a, conv, n_peaks=30, temperature=0.05):
    """Compute weighted subgradient using top peaks."""
    n = len(a)
    k = min(n_peaks, len(conv))
    peaks = np.argpartition(conv, -k)[-k:]
    peak_vals = conv[peaks]
    
    temp = max(peak_vals.max() * temperature, 1e-15)
    w = np.exp(peak_vals / temp)
    w /= w.sum()
    
    g = np.zeros(n)
    for i in range(k):
        if w[i] < 1e-12:
            continue
        ki = int(peaks[i])
        wi = 2.0 * w[i]
        j_start = max(0, ki - n + 1)
        j_end = min(n - 1, ki)
        for j in range(j_start, j_end + 1):
            g[j] += wi * a[ki - j]
    
    return g

    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    a = a / s
    
    a_prev = a.copy()
    t_nest = 1.0
    v = np.zeros(n)
    lr = lr_init
    temp = temp_init
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    best_nest_t = t_nest
    best_a_prev = a_prev.copy()

        t_new = (1 + np.sqrt(1 + 4 * t_nest * t_nest)) / 2
        beta = (t_nest - 1) / t_new
        y = a + beta * (a - a_prev)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        else:
            y = a.copy()
        
        # Gradient at y
        conv_y = fast_convolve(y)
        g = compute_subgradient(y, conv_y, n_peaks, temp)
        
        # Update
        v = momentum * v - lr * g
        a_new = np.maximum(y + v, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        a_prev = a
        a = a_new
        t_nest = t_new
        
        # Evaluate
        conv = fast_convolve(a)
        cur_obj = conv.max()
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
            best_nest_t = t_nest
            best_a_prev = a_prev.copy()

            v = np.zeros(n)
            no_improve = 0

        temp *= temp_decay
        it += 1
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG it={it}, obj={best_obj:.8f}, elapsed={elapsed:.1f}s")

def build_conv_matrix(a):
    """Build sparse matrix for conv(a, b) as linear function of b."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_oracle(a, M):
    """LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': 3.0})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def fpw_lp_optimize(a0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    a = a / s
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = M
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        b = lp_oracle(a, M)
        if b is None:
            break
        sb = b.sum()
        if sb < 1e-15:
            break
        b = b / sb
        
        if np.allclose(a, b, atol=1e-10):
            no_improve += 1
            if no_improve > 50:
                break
        else:
            # Line search
            best_t = 0.0
            best_local = best_obj
            for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                cand = (1 - t) * a + t * b
                local_obj = fast_convolve(cand).max()
                if local_obj < best_local:
                    best_local = local_obj
                    best_t = t
            
            if best_t < 0.05:
                no_improve += 1
            else:
                a = (1 - best_t) * a + best_t * b
                conv = fast_convolve(a)
                M = conv.max()
                if M < best_obj - 1e-13:
                    best_obj = M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
        
        if no_improve >= 100:
            # Perturb
            rng = np.random.default_rng(it)
            noise = rng.standard_normal(n) * 0.001
            a = np.maximum(a + noise, 0)
            a = a / a.sum()
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
        
        it += 1
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP it={it}, obj={best_obj:.8f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj

    except Exception:
        pass

    # === Phase 1: Generate diverse initializations ===
    print("=== Phase 1: Spectral Initialization ===")
    t_phase1 = time.monotonic()

    lengths = []
    if parent is not None:
        lengths.append(len(parent))
    for n_try in [384, 512, 768, 1024]:
        if n_try not in lengths:
            lengths.append(n_try)

    candidates = []  # list of (obj, n, array, name)

    for n in lengths:
        if time.monotonic() > deadline - 300:
            break

        for target_type in ['flat', 'dc_boost', 'edge_boost']:
            if time.monotonic() > deadline - 250:
                break
            a0 = gs_spectral_init(n, n_iters=400, rng=rng, target_type=target_type)
            conv = fast_convolve(a0)
            obj = conv.max()
            candidates.append((obj, n, a0, f"gs_{target_type}"))

        # Parent interpolation
        if parent is not None:
            p_interp = np.interp(np.linspace(0, 1, n),
                                 np.linspace(0, 1, len(parent)),
                                 np.maximum(parent, 0.0))
            ps = p_interp.sum()
            if ps > 1e-10:
                p_interp = p_interp / ps
                conv = fast_convolve(p_interp)
                obj = conv.max()
                candidates.append((obj, n, p_interp, "parent_interp"))

    # Sort by initial objective (lower is better)
    candidates.sort(key=lambda x: x[0])
    print(f"  Generated {len(candidates)} candidates in {time.monotonic()-t_phase1:.1f}s")
    print(f"  Top 5: {[(c[3], f'{c[0]:.6f}') for c in candidates[:5]]}")

    # === Phase 2: Smoothed gradient optimization ===
    print("=== Phase 2: Smoothed Gradient Optimization ===")
    t_phase2 = time.monotonic()

    time_reserve = 350.0  # reserve for later phases
    available = deadline - time.monotonic() - time_reserve
    n_top = min(5, len(candidates))
    if n_top > 0:
        time_per = max(20.0, available / n_top)

    for idx in range(n_top):
        if time.monotonic() > deadline - time_reserve:
            break
        obj_init, n, a0, name = candidates[idx]
        t_alloc = min(time_per, deadline - time.monotonic() - time_reserve + 50)
        if t_alloc < 10:
            break

        print(f"  [{idx}] {name}, n={n}, init_obj={obj_init:.6f}, "
              f"alloc={t_alloc:.0f}s")
        a_opt, obj_opt = momentum_optimize(
            a0, time_limit=t_alloc,
            lr_init=0.003, beta_init=50.0, momentum=0.90,
            lr_decay=0.9997, beta_growth=1.0002,
            verbose_every=20000)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"    ** New best: score={score:.6f}, obj={obj_opt:.6f}")
        else:
            print(f"    score={score:.6f}, obj={obj_opt:.6f}")

    print(f"  Phase 2 done in {time.monotonic()-t_phase2:.1f}s, best={best_score:.6f}")

    # === Phase 3: LP Polish ===
    print("=== Phase 3: LP Polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 50.0
        t_lp = remaining * 0.45
        if t_lp > 20:
            print(f"  LP polish, alloc={t_lp:.0f}s, n={len(best_sequence)}")
            a_opt, obj_opt = fpw_lp_optimize(best_sequence, time_limit=t_lp,
                                             verbose_every=200)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  ** LP improved: score={score:.6f}")
            else:
                print(f"  LP result: score={score:.6f}")

    # === Phase 4: Continued smoothed gradient on best ===
    if best_sequence is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic() - 25.0
        t_sg = remaining * 0.5
        if t_sg > 20:
            print(f"=== Phase 4: Continued Gradient ===")
            print(f"  alloc={t_sg:.0f}s, n={len(best_sequence)}")
            a_opt, obj_opt = momentum_optimize(
                best_sequence, time_limit=t_sg,
                lr_init=0.001, beta_init=150.0, momentum=0.92,
                lr_decay=0.9999, beta_growth=1.0001,
                verbose_every=30000)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  ** Improved: score={score:.6f}")
            else:
                print(f"  Result: score={score:.6f}")

    # === Phase 5: Perturbation restarts ===
    if best_sequence is not None and time.monotonic() < deadline - 20:
        print("=== Phase 5: Perturbation Restarts ===")
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 10.0
        n_restarts = max(1, int(remaining // 25))
        n_restarts = min(n_restarts, 6)

        for i in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break
            scale = 0.003 * (1.0 + i * 0.7)
            noise = rng.standard_normal(n_best) * scale
            a_pert = np.maximum(best_sequence + noise, 0.0)
            a_pert = a_pert / a_pert.sum()
            t_alloc = min(20.0, deadline - time.monotonic() - 8)
            if t_alloc < 5:
                break
            a_opt, obj_opt = momentum_optimize(
                a_pert, time_limit=t_alloc,
                lr_init=0.002, beta_init=100.0, momentum=0.90,
                lr_decay=0.9997, beta_growth=1.0003,
                verbose_every=0)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Restart {i}: ** improved to {score:.6f}")
            else:
                print(f"  Restart {i}: score={score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
