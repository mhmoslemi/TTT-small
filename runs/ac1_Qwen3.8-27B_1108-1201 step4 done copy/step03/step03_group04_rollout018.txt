def propose_candidate(seed=42, budget_s=1800, **kwargs):
    import numpy as np
    import time
    
    deadline = time.monotonic() + budget_s - 30
    
    best_seq = None
    best_score = float('inf')
    
    def eval_obj(a):
        """Fast objective computation (assumes sum(a) > 0)."""
        n = len(a)
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        A = np.fft.rfft(a, fft_size)
        conv = np.fft.irfft(A * A, fft_size)[:size]
        max_b = conv.max()
        sum_a = a.sum()
        if sum_a < 0.01:
            return float('inf')
        return 2.0 * n * max_b / (sum_a ** 2)
    
    def subgradient_step(a, lr, n_peaks=5):
        """One subgradient step to reduce max convolution."""
        n = len(a)
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        A = np.fft.rfft(a, fft_size)
        conv = np.fft.irfft(A * A, fft_size)[:size]
        
        # Find top peaks
        top_k = min(n_peaks, size)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        
        # Weight peaks by their value (higher peaks get more weight)
        temp = max(peak_vals.max() * 0.1, 1e-15)
        weights = peak_vals / temp
        weights = np.exp(weights - weights.max())
        weights /= weights.sum()
        
        # Compute subgradient
        g = np.zeros(n)
        for w, k in zip(weights, peaks):
            if w < 1e-10:
                continue
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            g[idx] += 2.0 * w * a[k - idx]
        
        # Step
        a_new = np.maximum(a - lr * g, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        return a_new
    
    # Phase 1: Try parametric families
    lengths = [64, 128, 256, 384, 512, 768, 1024]
    
    for n in lengths:
        if time.monotonic() > deadline - 60:
            break
        
        # Linear ramp
        a = np.arange(1, n + 1, dtype=np.float64)
        a /= a.sum()
        score = eval_obj(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
        
        # Try a few power ramps
        for p in [0.7, 0.8, 0.9, 1.0, 1.1, 1.2, 1.3]:
            a = np.arange(1, n + 1, dtype=np.float64) ** p
            a /= a.sum()
            score = eval_obj(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
        
        # Reverse ramp (should be same as ramp by symmetry of the problem)
        a = np.arange(n, 0, -1, dtype=np.float64)
        a /= a.sum()
        score = eval_obj(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
    
    # Phase 2: Subgradient polish
    if best_seq is not None:
        a = best_seq.copy()
        n = len(a)
        lr = 0.01
        t0 = time.monotonic()
        for it in range(200000):
            if time.monotonic() > deadline - 60:
                break
            a_new = subgradient_step(a, lr, n_peaks=10)
