"""

Key innovation: uses the "ideal flat-convolution" sequence (central binomial coefficients)

import numpy as np
import time
from scipy.optimize import linprog
from scipy.special import binom


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(a):
    """Build constraint matrix for LP: (a*b)[k] <= M for all k."""
    n = len(a)
    num_k = 2 * n - 1
    A_ub = np.zeros((num_k, n))
    for k in range(num_k):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A_ub[k, idx] = a[k - idx]
    return A_ub


def solve_lp_direction(a, M, top_k=None, conv=None, n_constraints=None):
    """Solve LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M for all (or top-k) k."""
    n = len(a)
    num_k = 2 * n - 1

    if top_k is not None and top_k < num_k and conv is not None:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        A_ub_full = build_lp_matrix(a)
        A_ub = A_ub_full[peaks]
        b_ub = np.full(len(peaks), M)
    elif n_constraints is not None and n_constraints < num_k and conv is not None:
        # Use constraints where conv is above a threshold
        threshold = conv.max() * 0.7
        active = np.where(conv > threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv)])
        A_ub_full = build_lp_matrix(a)
        A_ub = A_ub_full[active]
        b_ub = np.full(len(active), M)
    else:
        A_ub = build_lp_matrix(a)
        b_ub = np.full(num_k, M)

    c = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.success:
            return res.x
    except Exception:
        pass

    # Fallback: try with fewer constraints
    try:
        peaks = np.argsort(conv)[-max(10, n // 10):]
        A_ub_full = build_lp_matrix(a)
        A_ub = A_ub_full[peaks]
        b_ub = np.full(len(peaks), M)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass

    return None


def ideal_sequence(n):
    """Generate the 'ideal' flat-convolution sequence using central binomial coefficients."""
    k = np.arange(n, dtype=np.float64)
    # binom(2k, k) / 4^k
    # Use log for numerical stability
    log_vals = np.array([np.log(binom(2 * int(ki), int(ki))) - int(ki) * np.log(4) for ki in k])
    # For large k, use Stirling: binom(2k,k)/4^k ~ 1/sqrt(pi*k)
    # But for k up to ~500, direct computation is fine
    vals = np.exp(log_vals - log_vals.max())
    vals = np.maximum(vals, 0)
    s = vals.sum()
    if s < 1e-15:
        vals = np.ones(n) / n
    else:
        vals /= s
    return vals


def tapered_ideal_sequence(n, alpha=1.0):
    """Ideal sequence with edge tapering to reduce boundary effects."""
    base = ideal_sequence(n)
    k = np.arange(n, dtype=np.float64)
    taper = np.minimum(1.0, np.minimum(k + 1, n - k) / max(1, n * 0.1)) ** alpha
    vals = base * taper
    vals = np.maximum(vals, 0)
    s = vals.sum()
    if s < 1e-15:
        vals = np.ones(n) / n
    else:
        vals /= s
    return vals


def smooth_gradient(a, beta=500.0):
    """Compute smooth gradient of max(conv(a,a)) using softmax approximation."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    size = 2 * n
    fft_size = 1 << (size - 1).bit_length()

    w_padded = np.zeros(fft_size)
    w_padded[:len(w)] = w
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a

    W = np.fft.fft(w_padded)
    A = np.fft.fft(a_padded)
    cross = np.fft.ifft(A * np.conj(W)).real[:n]
    conv_wa = np.fft.ifft(A * W).real[:n]

    grad = cross + conv_wa
    return grad, conv


def lp_optimize(a0, time_limit, step_init=1.0, top_k=None, relax=1.0,
                verbose=False, seed=0, use_threshold=False):
    """LP-based optimization on the simplex."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    step = step_init
    no_improve = 0
    max_no_improve = 60
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()

        if use_threshold:
            b = solve_lp_direction(a, M * relax, n_constraints=n)
        elif top_k is not None:
            b = solve_lp_direction(a, M * relax, top_k=top_k, conv=conv)
        else:
            b = solve_lp_direction(a, M * relax, top_k=None)

        if b is None:
            b = solve_lp_direction(a, M * relax, top_k=max(20, n // 10), conv=conv)
            if b is None:
                break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break

        b = b / b_sum

        a_new = (1.0 - step) * a + step * b
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            if step < step_init:
                step = min(step_init, step * 1.05)
        else:
            no_improve += 1
            if step > 0.05:
                step *= 0.8
                if no_improve > 5:
                    step = max(0.01, step * 0.5)
            else:
                if no_improve > 8:
                    noise = rng.standard_normal(n) * 0.003
                    a = project_simplex(a + noise)
                    step = step_init * 0.4
                    no_improve = 0

        if no_improve >= max_no_improve:
            noise = rng.standard_normal(n) * 0.008
            a = project_simplex(a + noise)
            step = step_init * 0.6
            no_improve = 0
            max_no_improve = min(150, int(max_no_improve * 1.3))

        iter_count += 1
        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iter_count}: obj={best_obj:.6f}, step={step:.3f}, t={elapsed:.1f}s")

    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


def gradient_optimize(a0, time_limit, beta_init=500.0, lr_init=0.03,
                      momentum=0.9, verbose=False, seed=0):

    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    v = np.zeros(n)

    t0 = time.monotonic()
    beta = beta_init
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        grad, conv = smooth_gradient(a, beta)
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-15:
            grad = grad / g_norm

        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.85
                v *= 0.5
            if no_improve > 300:
                noise = rng.standard_normal(n) * 0.005
                a = project_simplex(a + noise)
                v = np.zeros(n)
                lr = lr_init * 0.5
                no_improve = 0

        beta = min(5000.0, beta * 1.002)

        if verbose and no_improve % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Grad iter: obj={best_obj:.6f}, beta={beta:.0f}, lr={lr:.5f}, t={elapsed:.1f}s")

    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


def peak_shave_optimize(a0, time_limit, seed=0, verbose=False):
    """Active-set peak shaving: iteratively reduce the top peaks."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    peak_count = 5

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        
        # Find top peaks
        top_peaks = np.argpartition(conv, -peak_count)[-peak_count:]
        peak_vals = conv[top_peaks]
        
        # For each peak, identify which a[i] contribute most
        # and try to reduce them while increasing others
        improvement = False
        for pk in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            # Find contributing indices
            i_min = max(0, pk - n + 1)
            i_max = min(n - 1, pk)
            contributors = np.arange(i_min, i_max + 1)
            if len(contributors) == 0:
                continue
                
            # Contribution of each a[i] to conv[pk]
            contribs = np.array([a[i] * a[pk - i] for i in contributors])
            total_contrib = contribs.sum()
            
            if total_contrib < 1e-15:
                continue
            
            # Reduce the top contributors, increase the least contributing
            top_contrib_idx = contributors[np.argmax(contribs)]
            low_contrib_idx = contributors[np.argmin(contribs)]
            
            if top_contrib_idx == low_contrib_idx:
                continue
            
            # Move mass from top contributor to low contributor
            transfer = a[top_contrib_idx] * 0.05
            a_new = a.copy()
            a_new[top_contrib_idx] -= transfer
            a_new[low_contrib_idx] += transfer
            
            if a_new[top_contrib_idx] < 0:
                a_new[low_contrib_idx] += a_new[top_contrib_idx]
                a_new[top_contrib_idx] = 0
            
            conv_new = fast_convolve(a_new)
            obj_new = 2.0 * n * conv_new.max()
            
            if obj_new < best_obj - 1e-13:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                improvement = True
        
        if not improvement:
            # Try LP step
            b = solve_lp_direction(a, M, top_k=max(20, n // 8), conv=conv)
            if b is not None:
                b_sum = b.sum()
                if b_sum > 1e-15:
                    b = b / b_sum
                    a_new = 0.7 * a + 0.3 * b
                    a_new = np.maximum(a_new, 0)
                    s = a_new.sum()
                    if s > 1e-15:
                        a_new = a_new / s
                    conv_new = fast_convolve(a_new)
                    obj_new = 2.0 * n * conv_new.max()
                    if obj_new < best_obj - 1e-13:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        a = a_new
                        improvement = True
        
        if not improvement:
            # Perturb
            noise = rng.standard_normal(n) * 0.002
            a = project_simplex(a + noise)
            peak_count = max(3, peak_count - 1)
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave: obj={best_obj:.6f}, t={elapsed:.1f}s")

    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj

    # The KEY initialization: ideal flat-convolution sequence
    inits['ideal'] = ideal_sequence(n)
    inits['ideal_taper1'] = tapered_ideal_sequence(n, alpha=1.0)
    inits['ideal_taper2'] = tapered_ideal_sequence(n, alpha=2.0)
    inits['ideal_taper05'] = tapered_ideal_sequence(n, alpha=0.5)

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Cosine window
    cos_w = np.cos(np.pi * x / (n - 1)) ** 2 if n > 1 else np.ones(1)
    cos_w = np.maximum(cos_w, 0.01)
    cos_w /= cos_w.sum()
    inits['cosine'] = cos_w

    # Bimodal (edges)
    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Gaussians
    for sf in [0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf:.1f}'] = g

    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: ideal-sequence initialization + LP + gradient + peak shaving.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent available: length={len(p)}, sum={p.sum():.2f}")
    except NameError:
        pass
    except Exception:
        pass

    # ========== Phase 1: Quick Exploration with Ideal Sequence ==========
    print("=== Phase 1: Quick Exploration ===")
    t_start = time.monotonic()
    phase1_time = 150.0

    lengths = [256, 320, 384, 448, 512]
    scan_results = []

    for n in lengths:
        if time.monotonic() + 20 > t_start + phase1_time:
            break
        inits = generate_initializations(n, parent, rng)
        
        # Priority: ideal first, then diverse
        priority = ['ideal', 'parent', 'parent_ds', 'ideal_taper1', 'cosine',
                    'bimodal', 'gauss_0.3', 'random', 'ideal_taper2']
        init_names = [nm for nm in priority if nm in inits]
        init_names = init_names[:4]

        for name in init_names:
            if time.monotonic() + 10 > t_start + phase1_time:
                break
            t_alloc = min(25.0, t_start + phase1_time - time.monotonic() - 5)
            if t_alloc < 5:
                break

            a0 = inits[name]
            a_opt, obj_opt = lp_optimize(a0, time_limit=t_alloc,
                                         step_init=0.9, top_k=max(20, n // 10),
                                         relax=1.0, seed=n + hash(name) % 1000)

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Scan: n={n}, {name}, obj={obj_opt:.6f}, score={score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.6f}, n={len(best_sequence) if best_sequence is not None else 0}")

    # ========== Phase 2: Deep LP on Best Candidates ==========
    print("=== Phase 2: Deep LP Optimization ===")
    phase2_start = time.monotonic()
    phase2_time = 500.0

    if best_sequence is None:
        best_sequence = ideal_sequence(320)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    n_best = len(best_sequence)

    # Collect starting points
    deep_starts = []
    seen = set()
    for score, n, name, a_opt in scan_results[:6]:
        key = (n, name)
        if key not in seen:
            deep_starts.append((n, a_opt.copy()))
            seen.add(key)

    # Always add current best with ideal init
    deep_starts.append((n_best, best_sequence.copy()))
    deep_starts.append((n_best, ideal_sequence(n_best)))

    # Configurations
    lp_configs = [
        {'step_init': 1.0, 'top_k': max(30, n_best // 8), 'relax': 1.0, 'use_threshold': False},
        {'step_init': 0.8, 'top_k': max(50, n_best // 5), 'relax': 1.0, 'use_threshold': False},
        {'step_init': 0.9, 'top_k': max(20, n_best // 12), 'relax': 1.0, 'use_threshold': True},
        {'step_init': 1.0, 'top_k': max(40, n_best // 6), 'relax': 1.01, 'use_threshold': False},
        {'step_init': 0.7, 'top_k': None, 'relax': 1.0, 'use_threshold': False},
    ]

    num_runs = len(deep_starts) * len(lp_configs)
    time_per_run = max(15.0, phase2_time / max(1, num_runs))

    run_count = 0
    for i, (n_s, a_s) in enumerate(deep_starts):
        for j, cfg in enumerate(lp_configs):
            if time.monotonic() + 15 > phase2_start + phase2_time:
                break
            t_alloc = min(time_per_run, phase2_start + phase2_time - time.monotonic() - 10)
            if t_alloc < 10:
                break

            cfg_copy = dict(cfg)
            if cfg_copy['top_k'] is not None:
                cfg_copy['top_k'] = max(10, int(cfg_copy['top_k'] * n_s / n_best))

            a_opt, obj_opt = lp_optimize(a_s, time_limit=t_alloc,
                                         seed=run_count * 13 + 7, **cfg_copy)

            score = float(evaluate_sequence(a_opt.tolist()))
            run_count += 1
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                n_best = len(best_sequence)
                print(f"  Deep LP [{run_count}]: n={n_s}, cfg{j}, score={score:.6f}")

    print(f"Phase 2 done. Best: {best_score:.6f}")

    # ========== Phase 3: Peak Shaving + Gradient ==========
    print("=== Phase 3: Peak Shaving & Gradient ===")
    phase3_start = time.monotonic()
    phase3_time = 350.0

    if best_sequence is not None:
        n_b = len(best_sequence)

        # Peak shaving
        t_alloc = min(100.0, phase3_time * 0.3)
        if t_alloc > 10:
            a_opt, obj_opt = peak_shave_optimize(best_sequence, time_limit=t_alloc, seed=42)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                n_b = len(best_sequence)
                print(f"  PeakShave: score={score:.6f}")

        # Gradient configs
        grad_configs = [
            {'beta_init': 300, 'lr_init': 0.02, 'momentum': 0.92},
            {'beta_init': 800, 'lr_init': 0.015, 'momentum': 0.94},
            {'beta_init': 1500, 'lr_init': 0.008, 'momentum': 0.95},
            {'beta_init': 100, 'lr_init': 0.04, 'momentum': 0.88},
            {'beta_init': 500, 'lr_init': 0.025, 'momentum': 0.91},
        ]

        time_per_grad = max(10.0, (phase3_time - (time.monotonic() - phase3_start) - 20) / len(grad_configs))

        for i, cfg in enumerate(grad_configs):
            if time.monotonic() + 10 > phase3_start + phase3_time:
                break
            t_alloc = min(time_per_grad, phase3_start + phase3_time - time.monotonic() - 15)
            if t_alloc < 8:
                break

            a_opt, obj_opt = gradient_optimize(best_sequence, time_limit=t_alloc,
                                               seed=i * 17 + 3, **cfg)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Grad [{i}]: score={score:.6f}")

    print(f"Phase 3 done. Best: {best_score:.6f}")

    # ========== Phase 4: Hybrid Alternating ==========
    print("=== Phase 4: Hybrid Alternating ===")
    phase4_start = time.monotonic()
    phase4_time = 400.0

    if best_sequence is not None:
        n_b = len(best_sequence)
        a = best_sequence.copy()

        for hcfg_i in range(3):
            if time.monotonic() + 60 > phase4_start + phase4_time:
                break
            t_alloc = min(phase4_time / 3, phase4_start + phase4_time - time.monotonic() - 30)
            if t_alloc < 30:
                break

            a_curr = a.copy()
            best_in_phase = a.copy()
            best_obj_phase = 2.0 * n_b * fast_convolve(a).max()
            v = np.zeros(n_b)
            step = 0.8 - 0.15 * hcfg_i
            top_k_h = max(30, n_b // (6 + hcfg_i * 2))

            phase4_t0 = time.monotonic()

            while time.monotonic() - phase4_t0 < t_alloc:
                conv = fast_convolve(a_curr)
                M = conv.max()

                # LP step
                b = solve_lp_direction(a_curr, M, top_k=top_k_h, conv=conv)
                if b is not None:
                    b_sum = b.sum()
                    if b_sum > 1e-15:
                        b = b / b_sum
                        a_curr = (1.0 - step) * a_curr + step * b
                        a_curr = np.maximum(a_curr, 0.0)
                        s = a_curr.sum()
                        if s > 1e-15:
                            a_curr = a_curr / s

                # Gradient step
                beta_g = 500 + 500 * hcfg_i
                grad, conv = smooth_gradient(a_curr, beta_g)
                g_norm = np.linalg.norm(grad)
                if g_norm > 1e-15:
                    grad = grad / g_norm
                lr_g = 0.02 - 0.005 * hcfg_i
                mom_g = 0.9 + 0.02 * hcfg_i
                v = mom_g * v - lr_g * grad
                a_curr = project_simplex(a_curr + v)

                conv = fast_convolve(a_curr)
                obj = 2.0 * n_b * conv.max()

                if obj < best_obj_phase - 1e-13:
                    best_obj_phase = obj
                    best_in_phase = a_curr.copy()

            score = float(evaluate_sequence(best_in_phase.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = best_in_phase.copy()
                a = best_in_phase.copy()
                print(f"  Hybrid [{hcfg_i}]: score={score:.6f}")

    print(f"Phase 4 done. Best: {best_score:.6f}")

    # ========== Phase 5: Final Polish ==========
    print("=== Phase 5: Final Polish ===")
    phase5_start = time.monotonic()
    phase5_time = max(10.0, deadline - time.monotonic())

    if best_sequence is not None:
        n_b = len(best_sequence)

        # Fine LP
        t_alloc = min(phase5_time * 0.3, 60.0)
        if t_alloc > 5:
            a_opt, obj_opt = lp_optimize(best_sequence, time_limit=t_alloc,
                                         step_init=0.3, top_k=max(50, n_b // 5),
                                         relax=1.0, seed=999)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                n_b = len(best_sequence)
                print(f"  Polish LP: score={score:.6f}")

        # Fine gradient
        t_alloc = min(phase5_time * 0.3, 50.0)
        if t_alloc > 5:
            a_opt, obj_opt = gradient_optimize(
                best_sequence, time_limit=t_alloc,
                beta_init=2000, lr_init=0.005, momentum=0.96, seed=777)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                n_b = len(best_sequence)
                print(f"  Polish Grad: score={score:.6f}")

        # Fine peak shaving
        t_alloc = min(phase5_time * 0.2, 40.0)
        if t_alloc > 5:
            a_opt, obj_opt = peak_shave_optimize(best_sequence, time_limit=t_alloc, seed=555)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                n_b = len(best_sequence)
                print(f"  Polish PeakShave: score={score:.6f}")

        # Tiny perturbations
        remaining = deadline - time.monotonic() - 20.0
        if remaining > 10:
            n_tries = min(30, int(remaining / 3))
            for i in range(n_tries):
                if time.monotonic() > deadline - 25:
                    break
                for scale in [0.0005, 0.001, 0.002, 0.005]:
                    noise = rng.standard_normal(n_b) * scale
                    a_pert = project_simplex(best_sequence + noise)
                    score_pert = float(evaluate_sequence(a_pert.tolist()))
                    if np.isfinite(score_pert) and score_pert < best_score:
                        best_score = score_pert
                        best_sequence = a_pert.copy()
                        print(f"  Perturb [{i}] s={scale}: score={score:.6f}")
                        break

    # Final verification
    if best_sequence is None:
        best_sequence = ideal_sequence(320)

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})")
    return [float(x) for x in best_sequence]
