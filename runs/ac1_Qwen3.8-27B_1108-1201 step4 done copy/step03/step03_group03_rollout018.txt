def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    n = len(v)
    v = v - v.max() * 0  # just to ensure it's a copy
    # Standard simplex projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u * rho > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def active_set_lp(a, M, S, n):

    A_ub = np.zeros((len(S), n))

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using active-set LP optimization,
subgradient polishing, spectral refinement, and multi-length search.
Key improvement: active-set LP solves only binding constraints (~30-60)
instead of all 2n-1, enabling 10-20x more LP iterations and deeper
optimization. Combined with adaptive blending, momentum subgradient,
and coordinate descent for final polish.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - target_sum) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_active_lp(a, M, peak_indices, n):
    """Solve LP: max sum(g) s.t. g>=0, (a*g)[k] <= M for k in peak_indices."""
    S = list(peak_indices)
    m = len(S)
    A_ub = np.zeros((m, n))
    for i, k in enumerate(S):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[i, j] = a[k - j]
    b_ub = np.full(m, M)
    c = -np.ones(n)
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub,
                         bounds=[(0.0, None)] * n,
                         method='highs', options={'presolve': True,
                                                   'time_limit': 30})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None


def optimize_active_lp(a_init, time_limit, blend_init=0.3, blend_min=0.003,
                       n_peaks=40, blend_halve_every=8, sparsity_blend=0.2):
    """Active-set LP optimization with adaptive blending."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    uniform = np.ones(n) / n

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    blend = blend_init
    stall = 0

    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        if M < 1e-18:
            break

        # Find active peaks
        k_peaks = min(n_peaks, 2 * n - 1)
        peak_idx = np.argpartition(conv, -k_peaks)[-k_peaks:]
        S = tuple(peak_idx.tolist())

        g = solve_active_lp(a, M, S, n)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs

        # Check constraint satisfaction; if violated, add more peaks
        cross = cross_convolve(a, g)
        if cross.max() > M * 1.005:
            k_peaks2 = min(k_peaks + 20, 2 * n - 1)
            peak_idx2 = np.argpartition(cross, -k_peaks2)[-k_peaks2:]
            S = tuple(peak_idx2.tolist())
            g2 = solve_active_lp(a, M, S, n)
            if g2 is not None and g2.sum() > 1e-15:
                g = g2 / g2.sum()

        # Sparsity control
        if g.max() > 0.4:
            g = (1.0 - sparsity_blend) * g + sparsity_blend * uniform

        a_new = (1.0 - blend) * a + blend * g
        a = project_simplex(a_new)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall >= blend_halve_every:
            blend = max(blend_min, blend * 0.5)
            stall = 0
            if blend < blend_min * 1.5:
                break

        if stall > blend_halve_every * 4:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 2**31)
            a = project_simplex(a + rng.standard_normal(n) * 0.008)
            conv = fast_convolve(a)
            stall = 0
            blend = max(blend, 0.1)

    return best_a, best_obj


def subgrad_optimize(a_init, time_limit, lr_init=0.012, momentum=0.92,
                     top_k=35, temp_scale=0.04):
    """Subgradient descent with momentum and softmax peak weighting."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        k_peaks = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_peaks)[-k_peaks:]
        peak_vals = conv[peaks]
        pmax = peak_vals.max()
        if pmax < 1e-18:
            break

        # Softmax weighting: emphasize the highest peaks
        scaled = peak_vals / (pmax * temp_scale + 1e-30)
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= ws

        # Compute subgradient
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            wi = 2.0 * w[idx]
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                sg[j] += wi * a[k - j]

        v = momentum * v - lr * sg
        a_new = project_simplex(a + v)
        a = a_new
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 150:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 2**31)
            a = project_simplex(a + rng.standard_normal(n) * 0.004)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0
            lr = max(0.001, lr * 0.6)

        lr *= 0.9995

    return best_a, best_obj


def coordinate_polish(a_init, max_rounds=50, n_trials=5, seed=0):
    """Coordinate descent: move small mass from high-contribution to low positions."""
    a = np.maximum(a_init.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    n = len(a)
    rng = np.random.default_rng(seed)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()

    for rnd in range(max_rounds):
        improved = False
        # Focus on removing mass from positions that contribute to peaks
        peak_k = int(np.argmax(conv))
        # Positions contributing to the main peak
        j_start = max(0, peak_k - n + 1)
        j_end = min(n - 1, peak_k)
        contributors = list(range(j_start, j_end + 1))
        # Sort by contribution
        contribs = [(j, a[j] * a[peak_k - j]) for j in contributors if a[j] > 1e-8]
        contribs.sort(key=lambda x: -x[1])

        for j_don in [c[0] for c in contribs[:10]]:
            if a[j_don] < 1e-7:
                continue
            for _ in range(n_trials):
                j_rec = int(rng.integers(n))
                if j_rec == j_don:
                    continue
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    delta = a[j_don] * frac
                    if delta < 1e-9:
                        continue
                    a_test = a.copy()
                    a_test[j_don] -= delta
                    a_test[j_rec] += delta
                    a_test = np.maximum(a_test, 0.0)
                    st = a_test.sum()
                    if st < 1e-15:
                        continue
                    a_test /= st
                    c_test = fast_convolve(a_test)
                    obj_test = 2.0 * n * c_test.max()
                    if obj_test < best_obj - 1e-14:
                        a = a_test
                        conv = c_test
                        best_obj = obj_test
                        improved = True

        if not improved:
            # Try random pairs
            for _ in range(20):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-7:
                    continue
                delta = a[i] * 0.05
                a_test = a.copy()
                a_test[i] -= delta
                a_test[j] += delta
                a_test = np.maximum(a_test, 0.0)
                st = a_test.sum()
                if st < 1e-15:
                    continue
                a_test /= st
                c_test = fast_convolve(a_test)
                obj_test = 2.0 * n * c_test.max()
                if obj_test < best_obj - 1e-14:
                    a = a_test
                    conv = c_test
                    best_obj = obj_test
                    improved = True
            if not improved:
                break

    return a, best_obj


def resample_sequence(parent, target_n):
    """Resample a sequence to a different length using interpolation."""
    m = len(parent)
    if m == target_n:
        return parent.copy()
    x_orig = np.linspace(0, 1, m)
    x_new = np.linspace(0, 1, target_n)
    result = np.interp(x_new, x_orig, parent)
    result = np.maximum(result, 0)
    s = result.sum()
    if s > 1e-15:
        result /= s
    else:
        result = np.ones(target_n) / target_n
    return result


def gen_initializations(n, parent=None, rng=None):
    """Generate diverse initializations for optimization."""
    if rng is None:
        rng = np.random.default_rng(123)
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Edge-concentrated
    e = max(1, n // 5)
    edge = np.zeros(n)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()

    # Triangle
    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()

    # Bimodal
    g2 = np.exp(-0.5 * ((x - 0.2 * n) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.8 * n) / (0.1 * n)) ** 2)
    inits['bimodal'] = g2 / g2.sum()

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Parent resampled
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            inits['parent'] = resample_sequence(p, n)

    return inits


def spectral_polish(a_init, time_limit, alpha=0.5):
    """
    Spectral polishing: iteratively modify the power spectrum to flatten
    the autocorrelation peak, then project back to non-negative simplex.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    n = len(a)

    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = 0.1

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()

        # Compute A = FFT(a)
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2

        # Find time-domain peaks and reduce their contribution in frequency
        k_peaks = 20
        peaks = np.argpartition(conv, -k_peaks)[-k_peaks:]

        # For each peak, compute the "unwanted" frequency pattern
        # and subtract a small amount from the power spectrum
        delta_power = np.zeros_like(power)
        for k in peaks:
            # The peak at lag k corresponds to a specific frequency pattern
            # We approximate by reducing high-frequency components
            weight = (conv[int(k)] - M * 0.7) / (M + 1e-30)
            if weight > 0:
                # Reduce power at frequencies that contribute to this peak
                freq_idx = int(fft_size / 2 * (k / (2 * n)))
                for di in range(-2, 3):
                    fi = (freq_idx + di) % (fft_size // 2 + 1)
                    delta_power[fi] += weight * power[fi] * 0.01

        new_power = np.maximum(power - alpha * lr * delta_power, 0)
        # Take square root to get new amplitude (preserve phase)
        new_A = np.sqrt(new_power) * (A / (np.abs(A) + 1e-30))
        new_a = np.fft.irfft(new_A, fft_size)[:n]
        new_a = np.maximum(new_a, 0)
        ns = new_a.sum()
        if ns < 1e-15:
            break
        new_a /= ns

        a = new_a
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
        else:
            lr *= 0.8

        lr = max(lr, 0.001)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Active-set LP scan over lengths and initializations.
    Phase 2: Deep LP optimization with smaller blend factors.
    Phase 3: Subgradient + spectral + coordinate descent polishing.
    Phase 4: Random restarts and final refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            par_score = float(evaluate_sequence(p.tolist()))
            print(f"Parent: n={len(p)}, score={par_score:.6f}")
            best_score = par_score
            best_sequence = p.copy()
    except Exception:
        pass

    all_results = []

    # Phase 1: Scan over lengths and initializations
    print("=== Phase 1: Active-Set LP Scan ===")
    lengths = [384, 512, 640, 768]
    phase1_end = time.monotonic() + min(450.0, (deadline - time.monotonic()) * 0.30)

    for n in lengths:
        if time.monotonic() > phase1_end - 30:
            break
        inits = gen_initializations(n, parent, rng)
        # Priority: parent first, then diverse structures
        prio = ['parent', 'edge', 'bimodal', 'uniform', 'triangle', 'random']
        names = [nm for nm in prio if nm in inits][:4]

        t_per = max(15.0, 200.0 / (len(lengths) * len(names)))

        for name in names:
            if time.monotonic() > phase1_end - 10:
                break
            t_alloc = min(t_per, phase1_end - time.monotonic() - 8)
            if t_alloc < 8:
                break

            a0 = inits[name]
            a_opt, obj_opt = optimize_active_lp(a0, t_alloc, blend_init=0.35,
                                                blend_min=0.01, n_peaks=40)
            # Quick subgrad
            t_sg = min(4.0, t_alloc * 0.15)
            if t_sg > 2:
                a_opt2, obj2 = subgrad_optimize(a_opt, t_sg, lr_init=0.008,
                                                momentum=0.9, top_k=25)
                if obj2 < obj_opt:
                    a_opt, obj_opt = a_opt2, obj2

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                all_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n} {name}: {score:.6f}")

    all_results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}")

    # Phase 2: Deep LP optimization on top candidates
    print("=== Phase 2: Deep LP Optimization ===")
    top_candidates = []
    seen = set()
    for score, n, name, a_opt in all_results[:10]:
        key = (n, name)
        if key not in seen:
            top_candidates.append((n, a_opt))
            seen.add(key)
        if len(top_candidates) >= 4:
            break
    if best_sequence is not None:
        top_candidates.append((len(best_sequence), best_sequence.copy()))

    remaining = deadline - time.monotonic() - 150.0
    t_per = max(20.0, remaining / max(1, len(top_candidates)))

    for n, a0 in top_candidates:
        if time.monotonic() > deadline - 130:
            break
        t_alloc = min(t_per, deadline - time.monotonic() - 120)
        if t_alloc < 10:
            break

        for bi in [0.2, 0.1, 0.05]:
            if time.monotonic() > deadline - 110:
                break
            t_s = t_alloc / 3.0
            if t_s < 6:
                break
            a_opt, obj_opt = optimize_active_lp(a0, t_s, blend_init=bi,
                                                blend_min=0.002, n_peaks=50)
            # Subgrad polish
            t_sg = min(8.0, t_s * 0.3)
            if t_sg > 3:
                a_opt2, obj2 = subgrad_optimize(a_opt, t_sg, lr_init=0.003,
                                                momentum=0.93, top_k=40)
                if obj2 < obj_opt:
                    a_opt, obj_opt = a_opt2, obj2
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Deep n={n} bi={bi}: {score:.6f}")

    # Phase 3: Polishing
    print("=== Phase 3: Polishing ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 60.0

        # Subgradient deep polish
        t_sg = min(80.0, remaining * 0.3)
        if t_sg > 10:
            a_opt, obj_opt = subgrad_optimize(best_sequence, t_sg,
                                              lr_init=0.002, momentum=0.94,
                                              top_k=50, temp_scale=0.03)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Subgrad: {score:.6f}")

        # Spectral polish
        remaining = deadline - time.monotonic() - 50.0
        t_spec = min(30.0, remaining * 0.2)
        if t_spec > 5:
            a_opt, obj_opt = spectral_polish(best_sequence, t_spec, alpha=0.3)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Spectral: {score:.6f}")

        # Coordinate descent
        remaining = deadline - time.monotonic() - 40.0
        t_cd = min(60.0, remaining * 0.3)
        n_rounds = max(10, int(t_cd / 2))
        a_opt, obj_opt = coordinate_polish(best_sequence, max_rounds=n_rounds,
                                           n_trials=4, seed=seed)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  CD: {score:.6f}")

        # Final LP + subgrad cycles
        remaining = deadline - time.monotonic() - 30.0
        for cyc in range(3):
            if time.monotonic() > deadline - 25:
                break
            t_cyc = min(remaining / max(1, 3 - cyc), 40.0)
            if t_cyc < 8:
                break

            a_opt, _ = optimize_active_lp(best_sequence, t_cyc * 0.5,
                                          blend_init=0.08, blend_min=0.002,
                                          n_peaks=55)
            a_opt2, obj2 = subgrad_optimize(a_opt, t_cyc * 0.35,
                                            lr_init=0.002, momentum=0.94,
                                            top_k=50, temp_scale=0.025)
            if obj2 < 2.0 * len(a_opt) * fast_convolve(a_opt).max():
                a_opt = a_opt2
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Cycle {cyc}: {score:.6f}")

            # Quick CD
            a_cd, _ = coordinate_polish(best_sequence, max_rounds=20,
                                        n_trials=3, seed=seed + cyc)
            score_cd = float(evaluate_sequence(a_cd.tolist()))
            if np.isfinite(score_cd) and score_cd < best_score:
                best_score = score_cd
                best_sequence = a_cd.copy()
                print(f"  CD cyc {cyc}: {score_cd:.6f}")

        # Random restart perturbations
        if time.monotonic() < deadline - 15:
            n_b = len(best_sequence)
            for ps in [0.002, 0.005, 0.01]:
                if time.monotonic() > deadline - 12:
                    break
                for _ in range(3):
                    if time.monotonic() > deadline - 10:
                        break
                    noise = rng.standard_normal(n_b) * ps
                    a_p = project_simplex(best_sequence + noise)
                    a_opt, _ = optimize_active_lp(a_p, 5.0, blend_init=0.12,
                                                  blend_min=0.005, n_peaks=45)
                    a_opt2, obj2 = subgrad_optimize(a_opt, 3.0, lr_init=0.003,
                                                    momentum=0.92, top_k=35)
                    if obj2 < 2.0 * n_b * fast_convolve(a_opt).max():
                        a_opt = a_opt2
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  Perturb({ps}): {score:.6f}")

    # Fallback
    if best_sequence is None or not np.isfinite(best_score):
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final validation
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(256) / 256
        final_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
