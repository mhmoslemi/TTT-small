"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterative refinement,
mirror descent (multiplicative weights), and adaptive step-size control.
Combines fast LP subproblem solving with efficient subgradient updates.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A[k, j_start:j_end+1] = a[k - j_start : k - j_end + 1]
    return A

    A_eq[0, n] = 0.0
    b_eq = np.array([1.0])
    A_ub = np.zeros((num_rows, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(num_rows)
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        b = np.maximum(res.x[:n], 0)
        return b, res.x[n]
    return None, None


def lp_max_sum(a, M):
    """Solve: max sum(b) s.t. conv(a,b)[k] <= M for all k, b >= 0.
    Returns b or None."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2. Assumes a >= 0, sum(a) > 0."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def subgradient_at_peak(a, conv, top_k=5, temperature=0.001):
    """Compute weighted subgradient at top-k peaks of convolution."""
    n = len(a)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        peak_vals = conv.copy()
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
    
    if temperature > 1e-15:
        scaled = peak_vals / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum > 1e-15:
            w /= w_sum
        else:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
    else:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        wi = w[i] * 2.0
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        sg[idx] += wi * a[k - idx]
    return sg


def mirror_descent(a, lr, top_k, temperature):
    """One step of multiplicative weights / mirror descent."""
    n = len(a)
    conv = fast_convolve(a)
    sg = subgradient_at_peak(a, conv, top_k, temperature)
    # Multiplicative update: a[j] *= exp(-lr * sg[j])
    a = a * np.exp(-lr * sg)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n, conv
    a = a / s
    return a, conv


def opt_lp_iterative(a0, time_limit, method='min_peak', step_init=0.5,
                     step_max=0.9, step_min=0.05, max_iters=5000):
    """LP-based iterative refinement. Returns (best_a, best_obj)."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    step = step_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1
        M = conv.max()
        if M < 1e-15:
            break

        if method == 'min_peak':
            b, t_opt = lp_min_peak(a)
            if b is None:
                break
            b_sum = b.sum()
            if b_sum < 1e-15:
                break
            b = b / b_sum
        else:
            b = lp_max_sum(a, M)
            if b is None:
                break
            b_sum = b.sum()
            if b_sum < 1e-15:
                break
            b = b / b_sum

        if np.linalg.norm(b - a) < 1e-10:
            break

        a_new = (1.0 - step) * a + step * b
        a_new = np.maximum(a_new, 0.0)
        a_sum = a_new.sum()
        if a_sum < 1e-15:
            step *= 0.5
            continue
        a_new = a_new / a_sum

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            step = min(step_max, step * 1.02)
        else:
            no_improve += 1
            step = max(step_min, step * 0.95)

        a = a_new
        conv = conv_new

        if no_improve > 200:
            # Try switching method
            step = step_init
            no_improve = 0
            method = 'min_peak' if method == 'max_sum' else 'max_sum'

        if no_improve > 500:
            break

    return best_a, best_obj

def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps

    # Cosine inits
    for freq in [1, 2, 3]:
        a = 1.0 + 0.8 * np.cos(2.0 * np.pi * freq * x / n)
        a = np.maximum(a, 0.0)
        ss = a.sum()
        if ss > 1e-10:
            inits[f'cos_{freq}'] = a / ss

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g

    # Random (Dirichlet)
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp

    # Two-lobe
    a = 1.0 + 0.5 * np.cos(2.0 * np.pi * x / n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    inits['two_lobe'] = a

    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterative refinement
with adaptive step control, multiplicative-weights (mirror descent) polishing,
and multi-scale exploration. Combines fast LP subproblem solving (HiGHS) with
efficient subgradient and mirror-descent updates for comprehensive search.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def lp_min_peak(a):
    """Solve: min t s.t. sum(b)=1, b>=0, conv(a,b)[k] <= t for all k.
    Returns (b, t) or (None, None)."""
    n = len(a)
    A = build_conv_matrix(a)
    num_rows = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    A_ub = np.zeros((num_rows, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(num_rows)
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        b = np.maximum(res.x[:n], 0)
        bs = b.sum()
        if bs < 1e-15:
            return None, None
        return b / bs, res.x[n]
    return None, None


def lp_max_sum(a, M):
    """Solve: max sum(b) s.t. conv(a,b)[k] <= M for all k, b >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        b = np.maximum(res.x, 0)
        bs = b.sum()
        if bs < 1e-15:
            return None
        return b / bs
    return None


def compute_obj_norm(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def subgradient_at_peak(a, conv, top_k, temperature):
    """Weighted subgradient at top-k convolution peaks."""
    n = len(a)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        peak_vals = conv.copy()
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
    if temperature > 1e-15:
        scaled = peak_vals / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum > 1e-15:
            w /= w_sum
        else:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
    else:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        wi = w[i] * 2.0
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        sg[idx] += wi * a[k - idx]
    return sg


def opt_lp_refine(a0, time_limit, method='min_peak', step_init=0.5,
                  step_max=0.85, step_min=0.05, max_iters=100000):
    """LP-based iterative refinement with adaptive step and method switching."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    step = step_init
    no_improve = 0
    it = 0
    cur_method = method

    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1
        M = conv.max()
        if M < 1e-14:
            break
        b = None
        if cur_method == 'min_peak':
            b, _ = lp_min_peak(a)
        else:
            b = lp_max_sum(a, M)
        if b is None:
            break
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum
        if np.linalg.norm(b - a) < 1e-10:
            break
        a_new = (1.0 - step) * a + step * b
        a_new = np.maximum(a_new, 0.0)
        a_sum = a_new.sum()
        if a_sum < 1e-15:
            step *= 0.5
            continue
        a_new = a_new / a_sum
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            step = min(step_max, step * 1.03)
        else:
            no_improve += 1
            step = max(step_min, step * 0.96)
        a = a_new
        conv = conv_new
        if no_improve > 100:
            step = step_init
            no_improve = 0
            cur_method = 'min_peak' if cur_method == 'max_sum' else 'max_sum'
        if no_improve > 400:
            break
    return best_a, best_obj
