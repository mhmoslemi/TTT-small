import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def lp_direction(a):
    """
    Find direction b that minimizes projected peak:
    min t s.t. sum(b)=1, b>=0, conv(a,b)[k] <= t for all k.
    Returns (b_normalized, t) or (None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)
    num_rows = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    A_ub = np.zeros((num_rows, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(num_rows)
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        b = np.maximum(res.x[:n], 0)
        s = b.sum()
        if s > 1e-15:
            return b / s, res.x[n]
    return None, None

def cd_sweep(a, conv, S, order=None):
    """One full coordinate descent sweep. Returns updated a, conv, S."""
    n = len(a)
    if order is None:
        order = np.arange(n)
    for i in order:
        new_x = cd_single_update(a, i, conv, S)
        if new_x != a[i]:
            delta = new_x - a[i]
            a[i] = new_x
            S += delta
            # Update conv incrementally
            for k in range(max(0, i), min(2 * n - 1, 2 * i + n)):
                ki = k - i
                if 0 <= ki < n:
                    if ki == i:
                        conv[k] += new_x * new_x - (new_x - delta) * (new_x - delta)
                    else:
                        conv[k] += 2.0 * delta * a[ki]
    return a, conv, S

def simulated_annealing(a0, time_limit, T_init=0.05, T_min=0.001, cooldown=0.9997,
                        move_scale_init=0.02, verbose=False):
    """Simulated annealing with adaptive moves."""
    a = np.maximum(a0.astype(np.float64).copy(), 0.0)
    S = a.sum()

    conv = fast_convolve(a)
    obj = 2.0 * n * conv.max() / (S * S)
    best_a = a.copy()
    best_obj = obj
    t0 = time.monotonic()
    T = T_init
    move_scale = move_scale_init
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        # Generate move
        move_type = rng.random()
        if move_type < 0.3:
            # Single coordinate change
            i = rng.integers(0, n)
            delta = rng.standard_normal() * move_scale * S / n
            a_new = a.copy()
            a_new[i] = max(0.0, a_new[i] + delta)
        elif move_type < 0.5:
            # Block change
            block_size = rng.integers(3, max(4, n // 20))
            start = rng.integers(0, n - block_size)
            delta = rng.standard_normal(block_size) * move_scale * S / n
            a_new = a.copy()
            a_new[start:start + block_size] = np.maximum(
                a_new[start:start + block_size] + delta, 0.0)
        elif move_type < 0.7:
            # Uniform perturbation
            a_new = a + rng.standard_normal(n) * move_scale * S / n
            a_new = np.maximum(a_new, 0.0)
        else:
            # Coordinate descent step (exact)
            i = rng.integers(0, n)
            a_new = a.copy()
            a_new[i] = cd_single_update(a, i, conv, S)

        S_new = a_new.sum()
        if S_new < 0.01:
            T *= cooldown
            continue

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max() / (S_new * S_new)

        delta_obj = obj_new - obj
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / T):
            a = a_new
            S = S_new
            conv = conv_new
            obj = obj_new
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                if verbose:
                    print(f"    SA new best: {best_obj:.8f}")

                     [nm for nm in init_names if nm not in priority]
        init_names = init_names[:3]

        time_per_init = max(15.0, remain / max(1, len(init_names)))

        for name in init_names:
            remain2 = deadline - time.monotonic() - 600
            if remain2 < 10:
                break
            t_alloc = min(time_per_init, remain2)
            if t_alloc < 5:
                break
            a0 = inits[name]
            a_opt, _ = lp_refine(a0, time_limit=t_alloc, max_iters=300, step_init=0.35)
            try_update(a_opt)

    print(f"Phase 1 done: best={best_score:.6f}, elapsed={time.monotonic()-t1_start:.0f}s")

    # Phase 2: Coordinate descent polishing (the main workhorse)
    print("=== Phase 2: Coordinate Descent ===")
    t2_start = time.monotonic()

    if best_sequence is not None:
        phase2_deadline = deadline - 500
        n_best = len(best_sequence)
        t_cd = min(300.0, phase2_deadline - time.monotonic() - 5)
        if t_cd > 30:
            a_opt, obj_opt = cd_optimize(best_sequence, time_limit=t_cd, verbose=True)
            try_update(a_opt)

    # Also try CD from a few other initializations
    if parent is not None:
        for n_try in [512, 768]:
            remain3 = deadline - time.monotonic() - 400
            if remain3 < 30:
                break
            if n_try == len(best_sequence) if best_sequence is not None else False:
                continue
            a0 = parent[:n_try] if len(parent) >= n_try else np.interp(
                np.linspace(0, 1, n_try), np.linspace(0, 1, len(parent)), parent)
            a0 = np.maximum(a0, 0)
            s = a0.sum()
            if s > 1e-10:
                a0 = a0 / s
            t_cd2 = min(60.0, remain3 / 2 - 5)
            if t_cd2 > 10:
                a_opt, _ = cd_optimize(a0, time_limit=t_cd2)
                try_update(a_opt)

    print(f"Phase 2 done: best={best_score:.6f}, elapsed={time.monotonic()-t2_start:.0f}s")

    # Phase 3: Simulated annealing for exploration
    print("=== Phase 3: Simulated Annealing ===")
    t3_start = time.monotonic()

    if best_sequence is not None:
        phase3_deadline = deadline - 300
        n_best = len(best_sequence)
        for T_init in [0.03, 0.08, 0.015]:
            if time.monotonic() > phase3_deadline:
                break
            t_alloc = min(100.0, phase3_deadline - time.monotonic() - 5)
            if t_alloc < 15:
                break
            a_opt, _ = simulated_annealing(
                best_sequence, time_limit=t_alloc,
                T_init=T_init, move_scale_init=0.015)
            try_update(a_opt)
            # Follow up with CD
            t_cd_follow = min(60.0, deadline - time.monotonic() - 200)
            if t_cd_follow > 10:
                a_opt2, _ = cd_optimize(a_opt, time_limit=t_cd_follow)
                try_update(a_opt2)

    print(f"Phase 3 done: best={best_score:.6f}, elapsed={time.monotonic()-t3_start:.0f}s")

    # Phase 4: LP refinement + CD alternating
    print("=== Phase 4: LP + CD Alternating ===")
    t4_start = time.monotonic()

    if best_sequence is not None:
        phase4_deadline = deadline - 150
        for step_init in [0.25, 0.4, 0.15]:
            if time.monotonic() > phase4_deadline:
                break
            t_lp = min(50.0, phase4_deadline - time.monotonic() - 30)
            if t_lp < 5:
                break
            a_opt, _ = lp_refine(best_sequence, time_limit=t_lp,
                                 max_iters=300, step_init=step_init)
            try_update(a_opt)
            t_cd = min(40.0, deadline - time.monotonic() - 100)
            if t_cd > 5:
                a_opt2, _ = cd_optimize(a_opt, time_limit=t_cd)
                try_update(a_opt2)

    print(f"Phase 4 done: best={best_score:.6f}, elapsed={time.monotonic()-t4_start:.0f}s")

    # Phase 5: Deep CD with block updates and perturbations
    print("=== Phase 5: Deep CD with Blocks ===")
    t5_start = time.monotonic()

    if best_sequence is not None:
        a = best_sequence.copy()
        S = a.sum()
        n = len(a)
        conv = fast_convolve(a)
        best_obj5 = 2.0 * n * conv.max() / (S * S)
        best_a5 = a.copy()
        t0_5 = time.monotonic()
        t_limit_5 = min(120.0, deadline - time.monotonic() - 30)
        rng5 = np.random.default_rng(777)

        while time.monotonic() - t0_5 < t_limit_5 and time.monotonic() < deadline - 30:
            # Do a few full CD sweeps
            for _ in range(3):
                order = rng5.permutation(n)
                a, conv, S = cd_sweep(a, conv, S, order)
            obj = 2.0 * n * conv.max() / (S * S)
            if obj < best_obj5 - 1e-12:
                best_obj5 = obj
                best_a5 = a.copy()

            # Block update
            a, conv, S = block_cd_update(a, conv, S,
                                         block_size=rng5.integers(5, max(6, n // 10)),
                                         rng=rng5)
            obj = 2.0 * n * conv.max() / (S * S)
            if obj < best_obj5 - 1e-12:
                best_obj5 = obj
                best_a5 = a.copy()

            # Occasionally perturb
            if rng5.random() < 0.1:
                noise = rng5.standard_normal(n) * (0.005 / np.sqrt(n))
                a = np.maximum(a + noise, 0)
                S = a.sum()
                conv = fast_convolve(a)

        try_update(best_a5)

    print(f"Phase 5 done: best={best_score:.6f}, elapsed={time.monotonic()-t5_start:.0f}s")

    # Phase 6: Final polish with mixed approaches
    print("=== Phase 6: Final Polish ===")
    t6_start = time.monotonic()

    if best_sequence is not None and time.monotonic() < deadline - 20:
        t_final = min(60.0, deadline - time.monotonic() - 15)
        if t_final > 10:
            # CD with very fine tolerances
            a_opt, _ = cd_optimize(best_sequence, time_limit=t_final * 0.6,
                                   max_sweeps=50000)
            try_update(a_opt)
            # Small SA
            if time.monotonic() < deadline - 10:
                t_sa = min(30.0, deadline - time.monotonic() - 10)
                if t_sa > 5:
                    a_opt2, _ = simulated_annealing(
                        a_opt, time_limit=t_sa, T_init=0.01, move_scale_init=0.005)
                    try_update(a_opt2)

    print(f"Phase 6 done: best={best_score:.6f}, elapsed={time.monotonic()-t6_start:.0f}s")

    # Final check
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
