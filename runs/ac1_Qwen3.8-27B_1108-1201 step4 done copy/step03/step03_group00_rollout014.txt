import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def solve_lp_dual(f, M):
    """Solve LP and return both primal and dual info."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        # Dual variables (reduced costs / shadow prices)
        dual = res.ineqlin.marginals if hasattr(res, 'ineqlin') else None
        return res.x, dual
    return None, None

def line_search(f, g, M, n_max_bisect=10):
    """Bisection line search for best alpha in [0,1] minimizing max((f_t)*(f_t))."""
    best_alpha = 0.0
    best_val = M
    
    # Coarse scan
    alphas = np.linspace(0.05, 1.0, 15)
    for alpha in alphas:
        ft = (1.0 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0.0)
        st = ft.sum()
        if st < 1e-15:
            continue
        ft = ft / st
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
    
    # Fine bisection
    if best_alpha > 0:
        lo = max(0.0, best_alpha - 0.07)
        hi = min(1.0, best_alpha + 0.07)
        for _ in range(n_max_bisect):
            mid = (lo + hi) / 2.0
            ft = (1.0 - mid) * f + mid * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                break
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = mid
                lo = mid
            else:
                hi = mid
    
    return best_alpha, best_val

def structural_perturb(f, n_peaks=5, scale=0.01, rng=None):
    """Perturb f by reducing mass at positions correlated with peak lags."""
    n = len(f)
    if rng is None:
        rng = np.random.default_rng()
    
    conv = fast_convolve(f)
    # Find the top peaks (excluding position 0 and n-1 which are the ends)
    peaks = np.argsort(conv)[::-1][:n_peaks]
    
    pert = f.copy()
    for k in peaks:
        # The convolution at lag k gets contributions from f[j]*f[k-j]
        # Reduce the largest contributors
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        idx = np.arange(js, je + 1)
        contrib = f[idx] * f[k - idx]
        # Find top contributors
        top_idx = idx[np.argsort(contrib)[::-1][:min(10, len(idx))]]
        for j in top_idx:
            pert[j] -= scale * f[j]
    
    pert = np.maximum(pert, 0)
    ps = pert.sum()
    if ps > 1e-15:
        pert /= ps
    return pert

def sg_polish(f0, time_limit, lr=0.003, mom=0.94, top_k=30, temp=0.00005):

    conv = fast_convolve(f)

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    
    while time.monotonic() - t0 < time_limit:
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()

            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0

            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()
        
        if co < best_obj - 1e-13:

            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)

    return best_f, best_obj

def coordinate_descent(f0, time_limit, passes=5):
    """Simple coordinate descent: optimize one element at a time."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    
    for _ in range(passes):
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj
            # Try a few values for f[i]
            old = f[i]
            rest_sum = 1.0 - old
            candidates = [old, old * 0.9, old * 1.1, old * 0.5, old * 1.5, 0.0, rest_sum / n]
            best_candidate = old
            best_val = best_obj
            
            for c in candidates:
                c = max(0.0, min(c, 1.0))
                # After changing f[i] to c, we need to renormalize
                # Actually, let's just try changing f[i] and see if obj improves
                f[i] = c
                fs = f.sum()
                if fs < 1e-15:
                    f[i] = old
                    continue
                f_norm = f / fs
                val = fast_convolve(f_norm).max()
                if val < best_val:
                    best_val = val
                    best_candidate = c
            
            f[i] = best_candidate
            f = f / f.sum()
            conv = fast_convolve(f)
            if conv.max() < best_obj - 1e-12:
                best_obj = conv.max()
                best_f = f.copy()
    
    return best_f, best_obj

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f, k_indices):
    """Build A_ub matrix for given lag indices."""
    n = len(f)
    A_ub = np.zeros((len(k_indices), n))
    for row, k in enumerate(k_indices):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je >= js:
            A_ub[row, js:je + 1] = f[k - np.arange(js, je + 1)]
    return A_ub


def solve_lp_cg(f, M, init_k=None, max_iters=30):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0 using constraint generation."""
    n = len(f)
    if init_k is None:
        conv = fast_convolve(f)
        init_k = list(np.argsort(conv)[::-1][:min(80, 2 * n - 1)])
    
    k_indices = list(init_k)
    g = None
    
    for it in range(max_iters):
        A_ub = build_lp_matrix(f, k_indices)
        c = -np.ones(n)
        b_ub = np.full(len(k_indices), M)
        bounds = [(0, None)] * n
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            return None
        g = res.x
        
        if g.sum() < 1e-15:
            break
        
        # Check full constraint satisfaction
        conv_g = fast_convolve(g)
        violations = conv_g - M
        if violations.max() < 1e-10:
            break
        
        # Add most violated constraints
        worst_idx = np.argsort(violations)[::-1][:10]
        added = False
        for k in worst_idx:
            k = int(k)
            if violations[k] > 1e-12 and k not in k_indices:
                k_indices.append(k)
                added = True
        if not added:
            break
    
    return g


def solve_lp_full(f, M):
    """Solve the full LP with all constraints (slower but exact)."""
    n = len(f)

    return None


def line_search(f, g, M_current, n_coarse=12, n_fine=8):

def structural_perturb(f, n_peaks=5, scale=0.008, rng=None):
    """Perturb f by reducing mass at positions contributing to peak lags."""
    n = len(f)
    if rng is None:
        rng = np.random.default_rng()
    
    conv = fast_convolve(f)
    peaks = np.argsort(conv)[::-1][:n_peaks]
    
    pert = f.copy()
    for k in peaks:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        idx = np.arange(js, je + 1)
        contrib = f[idx] * f[k - idx]
        top_count = min(5, len(idx))
        top_idx = idx[np.argsort(contrib)[::-1][:top_count]]
        for j in top_idx:
            pert[j] -= scale * f[j]
    
    pert = np.maximum(pert, 0)
    ps = pert.sum()
    if ps > 1e-15:
        pert /= ps
    else:
        pert = f.copy()
    return pert

    n = len(f)

        rev = f[::-1]
        alpha = 0.1 + 0.2 * rng.random()

        new_f[i:i + block], new_f[j:j + block] = new_f[j:j + block].copy(), new_f[i:i + block].copy()

            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            k_cache = list(np.argsort(conv)[::-1][:min(80, 2 * n - 1)])
            
            if M < best_obj - 1e-13:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"      iter {it}: obj={best_obj:.10f}, alpha={best_alpha:.4f}, t={elapsed:.0f}s, m_exp={m_exp:.2f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.15, 4.0)
        
        if no_improve > 35:
            f = multi_perturb(best_f, rng)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            k_cache = list(np.argsort(conv)[::-1][:min(80, 2 * n - 1)])
            no_improve = 0
    
    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.003, mom=0.94, top_k=30, temp=0.00005):
    """Subgradient descent with multi-peak softmax weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    
    while time.monotonic() - t0 < time_limit:
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()
        
        if co < best_obj - 1e-13:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.9997
        cur_temp *= 1.003
        
        if no_improve > 400:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 100000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0
    
    return best_f, best_obj


def coordinate_descent(f0, time_limit):
    """Coordinate descent: try small changes to individual elements."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit:
        # Randomly select elements to try
        for _ in range(200):
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj
            i = rng.integers(0, n)
            old = f[i]
            if old < 1e-10:
                continue
            
            # Try several multipliers
            factors = [0.7, 0.85, 1.15, 1.3, 0.5, 1.5, 2.0]
            rng.shuffle(factors)
            
            best_factor = 1.0
            best_val = best_obj
            for fac in factors[:3]:
                f[i] = old * fac
                fs = f.sum()
                if fs < 1e-15:
                    f[i] = old
                    continue
                fn = f / fs
                val = fast_convolve(fn).max()
                if val < best_val:
                    best_val = val
                    best_factor = fac
            
            if best_factor != 1.0:
                f[i] = old * best_factor
                f = f / f.sum()
                conv = fast_convolve(f)
                if conv.max() < best_obj - 1e-12:
                    best_obj = conv.max()
                    best_f = f.copy()
            else:
                f[i] = old
    
    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.4))
    
    x = np.arange(n, dtype=np.float64)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits.append(g / g.sum())
    
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.1
    inits.append(c1 / c1.sum())
    
    ramp = np.linspace(0.1, 1.0, n)
    inits.append(ramp / ramp.sum())
    
    humps = (np.exp(-0.5 * ((x - 0.3 * n) / (0.15 * n)) ** 2) +
             np.exp(-0.5 * ((x - 0.7 * n) / (0.15 * n)) ** 2))
    inits.append(humps / humps.sum())
    
    # Frequency-structured
    freq = 1.0 + 0.3 * np.cos(2 * np.pi * 3 * x / n)
    inits.append(np.maximum(freq, 0.01) / np.maximum(freq, 0.01).sum())
    
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            inits.append(px[::-1] / px[::-1].sum())
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search using constraint-generation LP Frank-Wolfe,
    structural and random perturbations, subgradient polishing, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score
    
    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}")
    except Exception:
        pass
    
    # === Phase 1: Quick evaluation ===
    print("=== Phase 1: Quick Evaluation ===")
    t_phase1 = time.monotonic()
    
    if parent is not None:
        try_seq(parent, "parent_raw")
        n_p = len(parent)
        t1 = min(250, deadline - time.monotonic() - 400)
        if t1 > 30:
            print(f"  FW-LP on parent (n={n_p}), t={t1:.0f}s")
            a_opt, _ = fw_lp_optimize(parent, time_limit=t1, verbose=True)
            try_seq(a_opt, "parent_fwlp")
    
    # Try 1-2 other lengths
    if parent is not None:
        n_parent = len(parent)
        other_lengths = [l for l in [256, 512, 320] if l != n_parent]
    else:
        other_lengths = [256, 384]
    
    for n_try in other_lengths[:1]:
        remaining = deadline - time.monotonic() - 300
        if remaining < 60:
            break
        inits = generate_inits(n_try, parent, rng)
        for i, f_init in enumerate(inits[:2]):
            t_alloc = min(40, remaining / 2)
            if t_alloc < 15:
                break
            a_opt, _ = fw_lp_optimize(f_init, time_limit=t_alloc)
            try_seq(a_opt, f"n{n_try}_i{i}")
    
    print(f"  Phase 1 done in {time.monotonic() - t_phase1:.0f}s, best: {best_score:.6f}")
    
    # === Phase 2: Deep optimization ===
    print(f"\n=== Phase 2: Deep Optimization ===")
    t_phase2 = time.monotonic()
    
    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic() - 200
        
        # Long run on best
        t_long = min(remaining * 0.5, 600)
        if t_long > 30:
            print(f"  Long FW-LP run, t={t_long:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_long, verbose=True)
            try_seq(a_opt, "deep_long")
        
        remaining = deadline - time.monotonic() - 150
        
        # Diversified runs
        for p in range(4):
            rem = deadline - time.monotonic() - 100
            if rem < 25:
                break
            t_alloc = min(rem / max(1, 4 - p), 150)
            
            if p == 0:
                f_init = multi_perturb(best_seq, rng)
            elif p == 1:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.01)
            elif p == 2:
                f_init = project_simplex(0.9 * best_seq + 0.1 * best_seq[::-1])
            else:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.05)
            
            label = ["div0", "div1", "div2", "div3"][p]
            print(f"  Diversified run [{label}], t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(f_init, time_limit=t_alloc)
            try_seq(a_opt, label)
    
    # SG polish
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            n_best = len(best_seq)
            a_opt, _ = sg_polish(best_seq, time_limit=min(rem, 80),
                                  lr=0.002, mom=0.95, top_k=max(10, n_best // 8), temp=0.00003)
            try_seq(a_opt, "sg_p2")
    
    print(f"  Phase 2 done in {time.monotonic() - t_phase2:.0f}s, best: {best_score:.6f}")
    
    # === Phase 3: Final polish ===
    print(f"\n=== Phase 3: Final Polish ===")
    t_phase3 = time.monotonic()
    print(f"  Current best: {best_score:.6f}")
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Very fine SG
        rem = deadline - time.monotonic() - 50
        if rem > 15:
            a_opt, _ = sg_polish(best_seq, time_limit=min(rem, 60),
                                  lr=0.0008, mom=0.97, top_k=max(5, n_best // 4), temp=0.00001)
            try_seq(a_opt, "sg_fine")
        
        # Coordinate descent
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 40))
            try_seq(a_opt, "coord_desc")
        
        # Final FW-LP
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=min(rem, 25), verbose=True)
            try_seq(a_opt, "fwl_final")
        
        # Tiny perturbations + quick SG
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            for _ in range(3):
                rem = deadline - time.monotonic() - 3
                if rem < 3:
                    break
                f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.001)
                a_opt, _ = sg_polish(f_pert, time_limit=min(rem, 8),
                                      lr=0.0005, mom=0.97, top_k=max(5, n_best // 4), temp=0.000005)
                try_seq(a_opt, "final_pert")
    
    print(f"  Phase 3 done in {time.monotonic() - t_phase3:.0f}s")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
