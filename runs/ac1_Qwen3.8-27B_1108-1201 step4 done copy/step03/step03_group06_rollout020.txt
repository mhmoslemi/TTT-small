from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():

    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)

    A = np.zeros((len(k_indices), n))
    for row, k in enumerate(k_indices):

        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)

def solve_lp(a, M, k_indices):
    """Solve LP: max sum(b) s.t. b>=0, (a*b)[k] <= M for k in k_indices."""
    n = len(a)
    A_ub = build_A_ub(a, k_indices)
    b_ub = np.full(len(k_indices), M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def optimize_lp(a0, time_limit, seed=0, verbose=False):
    """LP-based optimization with adaptive constraints and line search."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    top_k = max(10, n // 10)  # Start with fewer constraints
    max_top_k = 2 * n - 1
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        
        # Select active constraints: top-k peaks
        if top_k < 2 * n - 1:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        else:
            peaks = np.arange(2 * n - 1)
        
        b = solve_lp(a, M, peaks)
        if b is None:
            # Try with all constraints
            if top_k < 2 * n - 1:
                b = solve_lp(a, M, np.arange(2 * n - 1))
            if b is None:
                break
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum
        
        # Line search: try several step sizes
        best_step = 0.0
        best_obj_candidate = best_obj
        for step in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1]:
            a_new = (1.0 - step) * a + step * b
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            conv_new = fast_convolve(a_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj_candidate:
                best_obj_candidate = obj_new
                best_step = step
        
        if best_step > 0:
            a_new = (1.0 - best_step) * a + best_step * b
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            a = a_new
            best_obj = best_obj_candidate
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Perturb when stuck
            if no_improve > 10:
                noise = rng.standard_normal(n) * (0.01 + 0.01 * no_improve / 50.0)
                a = project_simplex(a + noise)
                no_improve = 0
                # Increase top_k to use more constraints
                top_k = min(max_top_k, int(top_k * 1.5))
            elif no_improve > 5:
                top_k = min(max_top_k, int(top_k * 1.3))
        
        # Gradually increase top_k
        if no_improve == 0 and top_k < max_top_k:
            top_k = min(max_top_k, top_k + max(1, n // 50))
        
        if verbose and no_improve % 50 == 0:
            print(f"    LP: obj={best_obj:.6f}, top_k={top_k}, t={time.monotonic()-t0:.1f}s")

def optimize_gradient(a0, time_limit, seed=0, verbose=False):

    rng = np.random.default_rng(seed)

        if obj_new < best_obj - 1e-12:

            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.85
                v *= 0.5

                noise = rng.standard_normal(n) * 0.01
                a = project_simplex(a + noise)

def coordinate_descent_polish(a0, time_limit, seed=0):
    """Coordinate descent: for each coordinate, try to decrease the peak."""
    rng = np.random.default_rng(seed)

    best_a = a.copy()
    
    t0 = time.monotonic()
    order = np.arange(n)
    
    while time.monotonic() - t0 < time_limit:
        np.random.default_rng(seed + int(time.monotonic()*1000) % 10000).shuffle(order)
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Try a few perturbations of coordinate i
            current_val = a[i]
            best_local = current_val
            best_local_obj = best_obj
            
            # Gradient-based suggestion
            conv_curr = fast_convolve(a)
            peak_k = np.argmax(conv_curr)
            
            # The peak (a*a)[peak_k] involves a[i] through:
            # a[i] * a[peak_k - i] + a[peak_k - i] * a[i] (if valid)
            # = 2 * a[i] * a[peak_k - i] (if peak_k != 2i)
            # = a[i]^2 (if peak_k == 2i)
            
            # To reduce the peak, we should decrease a[i] if it contributes positively
            # But we need to maintain the simplex constraint
            
            # Simple: try a[i] * 0.9, a[i] * 1.1, a[i] - 0.01/max(a), etc.
            for factor in [0.7, 0.85, 1.15, 1.3, 0.5]:
                trial = a.copy()
                trial[i] = current_val * factor
                if trial[i] < 0:
                    trial[i] = 0
                s_trial = trial.sum()
                if s_trial < 1e-15:
                    continue
                trial = trial / s_trial
                conv_trial = fast_convolve(trial)
                obj_trial = 2.0 * n * conv_trial.max()
                if obj_trial < best_local_obj:
                    best_local_obj = obj_trial
                    best_local = factor
            
            if best_local != current_val:
                a[i] = current_val * best_local
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a / s_a
                conv_check = fast_convolve(a)
                obj_check = 2.0 * n * conv_check.max()
                if obj_check < best_obj:
                    best_obj = obj_check
                    best_a = a.copy()
    
    return best_a, 2.0 * n * fast_convolve(best_a).max()

"""
Multi-phase optimization of non-negative sequences to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based direction finding with adaptive constraints, smooth gradient descent,
structured perturbations, and exploration of multiple sequence lengths.
Key improvements: curriculum-based constraint addition, structured restarts,
wider n exploration, and hybrid LP+gradient alternating scheme.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def build_A_rows(a, k_indices):
    """Build sparse-ish constraint matrix rows for given lag indices."""
    n = len(a)
    A = np.zeros((len(k_indices), n))
    for row, k in enumerate(k_indices):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min > i_max:
            continue
        idx = np.arange(i_min, i_max + 1)
        A[row, idx] = a[k - idx]
    return A


def solve_lp_direction(a, M, k_indices, timeout=5):
    """Solve LP: max sum(b) s.t. b>=0, (a*b)[k] <= M for k in k_indices."""
    n = len(a)
    if len(k_indices) == 0:
        return None
    A_ub = build_A_rows(a, k_indices)
    b_ub = np.full(len(k_indices), M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout,
                               'dual_feasibility_tolerance': 1e-7})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def compute_smooth_gradient(a, beta=500.0):
    """Compute gradient of log-sum-exp approximation of max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    fft_size = 1 << (2 * n).bit_length()
    w_pad = np.zeros(fft_size)
    w_pad[:len(w)] = w
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a

    W_full = np.fft.fft(w_pad)
    A_full = np.fft.fft(a_pad)

    cross = np.fft.ifft(np.conj(A_full) * W_full).real[:n]
    conv_term = np.fft.ifft(A_full * W_full).real[:n]

    grad = cross + conv_term
    return grad, conv


def lp_optimize(a0, time_limit, seed=0, top_k_init=None, verbose=False):
    """LP-based optimization with adaptive constraint curriculum."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    if top_k_init is None:
        top_k_init = max(10, n // 8)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    top_k = top_k_init
    max_top_k = 2 * n - 1
    no_improve = 0
    step = 1.0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()

        if top_k < max_top_k:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        else:
            peaks = np.arange(max_top_k)

        b = solve_lp_direction(a, M, peaks, timeout=3)
        if b is None:
            b = solve_lp_direction(a, M, np.arange(max_top_k), timeout=5)
        if b is None:
            break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum

        # Adaptive step: try full step first, then reduce
        accepted = False
        for trial_step in [step, 0.7, 0.5, 0.3, 0.15]:
            if trial_step < 0.01:
                break
            a_new = (1.0 - trial_step) * a + trial_step * b
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            conv_new = fast_convolve(a_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-14:
                a = a_new
                best_obj = obj_new
                best_a = a.copy()
                step = min(1.0, trial_step * 1.1)
                no_improve = 0
                accepted = True
                break

        if not accepted:
            no_improve += 1
            step = max(0.05, step * 0.85)

        # Curriculum: gradually increase constraints
        if no_improve == 0:
            top_k = min(max_top_k, top_k + max(1, n // 40))
        elif no_improve == 3:
            top_k = min(max_top_k, int(top_k * 1.3))
        elif no_improve >= 15:
            # Structured perturbation: add a smooth bump
            center = rng.integers(n // 4, 3 * n // 4)
            width = n * rng.uniform(0.05, 0.25)
            x = np.arange(n)
            bump = np.exp(-0.5 * ((x - center) / width) ** 2)
            bump /= bump.sum()
            mix = rng.uniform(0.05, 0.2)
            a = project_simplex((1 - mix) * a + mix * bump)
            no_improve = 0
            step = 0.8
            top_k = max(top_k_init, int(top_k * 0.7))

        # Occasional random restart
        elif no_improve == 25:
            a = project_simplex(a + rng.standard_normal(n) * 0.02)
            no_improve = 0
            step = 0.7

        if verbose and no_improve % 30 == 0:
            print(f"      LP: obj={best_obj:.6f}, top_k={top_k}/{max_top_k}, "
                  f"step={step:.2f}, t={time.monotonic()-t0:.1f}s")

    conv_final = fast_convolve(best_a)
    return best_a, 2.0 * n * conv_final.max()


def gradient_optimize(a0, time_limit, seed=0, verbose=False):

    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)

    if s < 1e-15:

    momentum = 0.9
    no_improve = 0

def hybrid_optimize(a0, time_limit, seed=0, verbose=False):

    rng = np.random.default_rng(seed)

    max_top_k = 2 * n - 1
    no_improve = 0

    lr = 0.04
    lp_step = 0.8
    mode = 0  # 0=LP, 1=gradient

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()

        if mode == 0:
            # LP step
            if top_k < max_top_k:
                peaks = np.argpartition(conv, -top_k)[-top_k:]
            else:
                peaks = np.arange(max_top_k)

            b = solve_lp_direction(a, M, peaks, timeout=3)
            if b is not None and b.sum() > 1e-15:
                b = b / b.sum()
                a_new = (1.0 - lp_step) * a + lp_step * b
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a_new = a_new / s_new
                    conv_new = fast_convolve(a_new)
                    obj_new = 2.0 * n * conv_new.max()
                    if obj_new < best_obj:
                        a = a_new
                        best_obj = obj_new
                        best_a = a.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                        lp_step = max(0.2, lp_step * 0.8)
                top_k = min(max_top_k, top_k + max(1, n // 60))
            else:
                no_improve += 1

        else:
            # Gradient step
            grad, _ = compute_smooth_gradient(a, beta)
            g_norm = np.linalg.norm(grad)
            if g_norm > 1e-15:
                grad = grad / g_norm
            v = 0.9 * v - lr * grad
            a_new = project_simplex(a + v)
            conv_new = fast_convolve(a_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj:
                a = a_new
                best_obj = obj_new
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
                lr *= 0.95
            beta = min(6000.0, beta * 1.002)

        # Switch mode every few iterations or when stuck
        if no_improve >= 8:
            mode = 1 - mode
            if mode == 0:
                lp_step = 0.7
            else:
                lr = 0.03
                v *= 0.5

        if no_improve >= 20:

                x = np.arange(n)

                bump /= bump.sum()
                a_test = project_simplex(0.92 * a + 0.08 * bump)

                obj_test = 2.0 * n * conv_test.max()
                if obj_test < best_obj:

                    best_obj = obj_test
                    best_a = a.copy()
                    no_improve = 0
                    break
            else:
                a = project_simplex(a + rng.standard_normal(n) * 0.015)
                no_improve = 0
                mode = 0
                lp_step = 0.6

        if verbose and no_improve % 40 == 0:
            print(f"      Hybrid: obj={best_obj:.6f}, mode={mode}, "
                  f"top_k={top_k}, t={time.monotonic()-t0:.1f}s")

    conv_final = fast_convolve(best_a)
    return best_a, 2.0 * n * conv_final.max()


def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp

    tri = np.minimum(x, n - 1 - x).astype(np.float64) + 1.0
    tri /= tri.sum()
    inits['tri_edge'] = tri

    tri_c = (n - 1 - np.minimum(x, n - 1 - x)).astype(np.float64)
    tri_c /= tri_c.sum()
    inits['tri_center'] = tri_c

    for sf in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf:.2f}'] = g

    # Bimodal (edges)
    edge = max(1, n // 5)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # One-sided
    bump = np.exp(-x / (0.25 * n))
    bump /= bump.sum()
    inits['bump_left'] = bump
    inits['bump_right'] = bump[::-1].copy()

    # Cosine
    if n > 1:
        cos_w = np.cos(np.pi * x / (n - 1)) ** 2
        cos_w = np.maximum(cos_w, 0.05)
        cos_w /= cos_w.sum()
    else:
        cos_w = np.ones(1)
    inits['cosine'] = cos_w

    # Random Dirichlet with different concentrations
    inits['rand_sparse'] = rng.dirichlet(np.ones(n) * 0.05)
    inits['rand_flat'] = rng.dirichlet(np.ones(n) * 1.0)

    # Mixture of edge bumps
    w1 = rng.uniform(0.3, 0.7)
    mix = w1 * np.exp(-x / (0.15 * n)) + (1 - w1) * np.exp(-(n - 1 - x) / (0.15 * n))
    mix /= mix.sum()
    inits['asym_edges'] = mix

    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_interp'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps

    return inits

    t3_start = time.monotonic()
    t3_budget = min(500.0, (deadline - time.monotonic()) * 0.4)

    if best_sequence is not None:
        n_best = len(best_sequence)

        # Run hybrid from best and a few perturbed versions
        hybrid_starts = [best_sequence.copy()]
        for i in range(3):
            center = rng.integers(n_best // 4, 3 * n_best // 4)
            width = n_best * rng.uniform(0.1, 0.25)
            x = np.arange(n_best)
            bump = np.exp(-0.5 * ((x - center) / width) ** 2)
            bump /= bump.sum()
            hybrid_starts.append(project_simplex(0.9 * best_sequence + 0.1 * bump))

        t_per_hybrid = t3_budget / len(hybrid_starts)

        for idx, a_start in enumerate(hybrid_starts):
            if time.monotonic() > t3_start + t3_budget - 10:
                break
            t_alloc = min(t_per_hybrid, t3_start + t3_budget - time.monotonic() - 5)
            if t_alloc < 30:
                break

            a_opt, obj_opt = hybrid_optimize(a_start, time_limit=t_alloc,
                                             seed=3000 + idx * 11)
            update_best(a_opt, f"hybrid n={n_best} start={idx}")

    print(f"Phase 3 done. Best: {best_score:.6f}")

    # ========== Phase 4: Fine-tuning with Multiple Methods ==========
    print("=== Phase 4: Fine-tuning ===")
    t4_start = time.monotonic()
    t4_budget = min(400.0, (deadline - time.monotonic()) * 0.5)

    if best_sequence is not None:
        n_best = len(best_sequence)

        # Extended LP with very small steps (fine polish)
        t_lp = t4_budget * 0.3
        if t_lp > 30:
            a_opt, _ = lp_optimize(best_sequence, time_limit=t_lp,
                                   seed=5000, top_k_init=n_best // 4)
            update_best(a_opt, "polish_LP")

        # Extended gradient with high beta (sharp max approximation)
        t_gr = t4_budget * 0.3
        if t_gr > 30:
            a_opt, _ = gradient_optimize(best_sequence, time_limit=t_gr,
                                         seed=5001)
            update_best(a_opt, "polish_Grad")

        # More hybrid
        t_hy = t4_budget * 0.2
        if t_hy > 30:
            a_opt, _ = hybrid_optimize(best_sequence, time_limit=t_hy, seed=5002)
            update_best(a_opt, "polish_Hybrid")

        # Try different n: slightly larger
        remaining_after_4phases = deadline - time.monotonic() - 20
        if remaining_after_4phases > 100:
            # Try one more n value that we haven't tried deeply
            for n_try in [n_best + 64, n_best + 128, n_best * 2]:
                if n_try > 2048 or n_try < 128:
                    continue
                if time.monotonic() > deadline - 80:
                    break
                inits = generate_inits(n_try, best_sequence, rng)
                if 'parent_interp' in inits:
                    a0 = inits['parent_interp']
                elif 'uniform' in inits:
                    a0 = inits['uniform']
                else:
                    a0 = next(iter(inits.values()))
                t_try = min(80.0, deadline - time.monotonic() - 30)
                if t_try < 30:
                    break
                a_opt, _ = lp_optimize(a0, time_limit=t_try, seed=6000 + n_try)
                update_best(a_opt, f"explore n={n_try}")

    print(f"Phase 4 done. Best: {best_score:.6f}")

    # ========== Phase 5: Final Polish ==========
    print("=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    if best_sequence is not None and remaining > 10:
        n_best = len(best_sequence)

        # Tiny perturbations
        t_poll = remaining * 0.4
        if t_poll > 20:
            t0 = time.monotonic()
            best_local = best_sequence.copy()
            best_local_obj = best_score
            attempts = 0
            while time.monotonic() - t0 < t_poll:
                attempts += 1
                for scale in [0.002, 0.005, 0.01, 0.02]:
                    noise = rng.standard_normal(n_best) * scale
                    a_pert = project_simplex(best_local + noise)
                    obj_pert = evaluate_obj_normalized(a_pert)
                    if obj_pert < best_local_obj - 1e-12:
                        best_local = a_pert
                        best_local_obj = obj_pert
                        break
                if attempts % 50 == 0:
                    pass  # could print progress

            if best_local_obj < best_score:
                update_best(best_local, f"perturb ({attempts} iters)")
            else:
                best_sequence = best_local  # keep the best we found

        # Final LP micro-polish
        t_final = deadline - time.monotonic() - 5
        if t_final > 15:
            a_opt, _ = lp_optimize(best_sequence, time_limit=t_final,
                                   seed=9999, top_k_init=max(20, n_best // 3))
            update_best(a_opt, "final_LP")

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Ensure all values are in valid range
    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    final_check = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_check):
        best_score = final_check

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
