```python
from scipy.optimize import linprog

def solve_lp(f, M):
    """Solve the LP: maximize sum(g) subject to g >= 0 and (f*g)(k) <= M for all k."""
    n = len(f)
    
    # Variables: g_0, ..., g_{n-1}
    # Objective: maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: (f*g)(k) <= M for k = 0, ..., 2n-2
    # (f*g)(k) = sum_i f_i * g_{k-i} = sum_j f_{k-j} * g_j
    # So the constraint matrix A_eq has A[k, j] = f_{k-j} if 0 <= k-j < n, else 0
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = f[idx]
    b_ub = np.full(2*n-1, M)
    
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x
```

```python
import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_for_g(f, M):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)(k) <= M for all k."""
    n = len(f)
    
    # Build constraint matrix
    # (f*g)(k) = sum_j f_{k-j} * g_j for j where 0 <= k-j < n
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        # j ranges from max(0, k-n+1) to min(k, n-1)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            A[k, j] = f[k - j]
    
    c = -np.ones(n)  # minimize -sum(g)
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                     options={'time_limit': 30})
    
    if result.success:
        return result.x
    return None
```

```python
    # A[k, j] = f[k-j] if 0 <= k-j < n

    for lag in range(n):
```

```python
def fast_convolve(a):

    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
```

```python
def compute_objective(a):
    """Compute 2*n*max(conv)/sum^2."""
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max() / (s * s)


def solve_lp_for_g(f, M, time_limit=5.0):
    """Solve: max sum(g) s.t. g >= 0, (f*g)(k) <= M for all k."""
    n = len(f)
    
    # Build constraint matrix
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                     options={'time_limit': time_limit, 'presolve': True})
    
    if result.success and result.x is not None:
        return np.maximum(result.x, 0.0)
    return None


def lp_optimize(f0, max_iters=5000, time_budget=60.0, verbose=False):
    """Iterative LP-based optimization."""
    n = len(f0)
    f = np.maximum(f0, 0.0).copy()
    s = f.sum()
    if s < 1e-10:
        return f0
    f = f / s
    
    t0 = time.monotonic()
    best_M = fast_convolve(f).max()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_budget:
            break
        
        M = best_M
        g = solve_lp_for_g(f, M, time_limit=5.0)
        
        if g is None or g.sum() < 1e-12:
            break
        
        g = g / g.sum()

        best_t = 0.0
        best_new_M = M
        # Try a geometric schedule
        for t_candidate in [0.5, 1.0, 0.3, 0.7, 0.1, 0.9, 0.2, 0.8]:

            new_M = fast_convolve(f_try).max()
            if new_M < best_new_M:
                best_new_M = new_M
                best_t = t_candidate

    lr = 0.1 / np.sqrt(n)

        k_star = int(np.argmax(conv))

            best_f = f_new.copy()
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using iterative LP-based peak reduction,
FFT-based smooth gradient optimization, and subgradient polishing with multi-start.
The core LP method at each step finds the maximal non-negative direction g that
keeps (f*g) below the current peak, then performs a line search for the best mixing.
"""

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_objective(a):
    """Compute 2*n*max(conv)/sum^2 for a sequence."""
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max() / (s * s)


def solve_lp_for_g(f, M, time_limit=5.0):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)(k) <= M for all k=0..2n-2."""
    n = len(f)
    # Build constraint matrix: A[k,j] = f[k-j] if 0 <= k-j < n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                     options={'time_limit': time_limit, 'presolve': True})
    if result.success and result.x is not None:
        return np.maximum(result.x, 0.0)
    return None


def solve_lp_dual(f, M, time_limit=5.0):
    """Solve dual LP: min M*sum(y) s.t. (f*y shifted)_j >= 1, y >= 0.
    Returns primal solution g via the solver's shadow prices or recomputation."""
    n = len(f)
    # Dual: min M * sum_k y_k
    # s.t. sum_{m=0}^{n-1} f[m] * y[m+j] >= 1 for j=0..n-1
    #      y_k >= 0 for k=0..2n-2
    # Decision: just solve primal, it's fine for n=256
    return solve_lp_for_g(f, M, time_limit)

    """Iterative LP-based optimization to reduce max(auto-convolution)."""
    n = len(f0)
    f = np.maximum(f0, 0.0).copy()
    s = f.sum()
    if s < 1e-10:
        return f0.copy()
    f = f / s

    t0 = time.monotonic()
    M = fast_convolve(f).max()
    stagnation = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_budget:
            break
        if M < 1e-12:
            break

        g = solve_lp_for_g(f, M, time_limit=5.0)

        if g is None or g.sum() < 1e-14:
            break

        g_norm = g / g.sum()

        # Line search: try multiple step sizes
        best_t = 0.0
        best_new_M = M

        t_candidates = [0.5, 1.0, 0.3, 0.7, 0.1, 0.9, 0.2, 0.8, 0.4, 0.6]
        for tc in t_candidates:
            f_try = (1.0 - tc) * f + tc * g_norm
            new_M = fast_convolve(f_try).max()
            if new_M < best_new_M:
                best_new_M = new_M
                best_t = tc

        if best_t < 1e-8 or best_new_M >= M * (1.0 - 1e-12):
            # Try smaller steps
            for tc in [0.05, 0.02, 0.01]:
                f_try = (1.0 - tc) * f + tc * g_norm
                new_M = fast_convolve(f_try).max()
                if new_M < best_new_M:
                    best_new_M = new_M
                    best_t = tc

        if best_t < 1e-8:
            stagnation += 1
            if stagnation > 10:
                break
            continue

        f = (1.0 - best_t) * f + best_t * g_norm
        f = np.maximum(f, 0.0)
        fs = f.sum()
        if fs < 1e-10:
            break
        f = f / fs
        improvement = (M - best_new_M) / (M + 1e-15)
        M = best_new_M
        stagnation = 0 if improvement > 1e-10 else stagnation + 1
        if stagnation > 50:
            break

        if verbose and it % 200 == 0:
            print("    LP iter %d: M=%.10f, t=%.3f, obj=%.8f" % (it, M, best_t, 2.0 * n * M))

    return f


def fft_smooth_grad(f, w):
    """Compute gradient of sum_k w[k] * (f*f)(k) w.r.t. f using FFT cross-correlation."""
    n = len(f)
    N = 1 << (n + len(w) - 1).bit_length()
    W = fft(w, N)
    F = fft(f, N)
    grad = np.real(ifft(np.conj(W) * F, N))[:n] * 2.0
    return grad


def smooth_beta_optimize(f0, beta_start=40.0, max_iters=3000, time_budget=60.0, verbose=False):
    """Optimize using smooth max (log-sum-exp) with adaptive beta and Nesterov acceleration."""
    n = len(f0)
    f = np.maximum(f0, 0.0).copy()
    s = f.sum()
    if s < 1e-10:
        return f0.copy()
    f = f / s

    t0 = time.monotonic()
    best_f = f.copy()
    best_M = fast_convolve(f).max()
    beta = beta_start
    f_ynest = f.copy()
    t_mom = 1.0 / (1.0 + 1.0)  # Nesterov momentum parameter

    for it in range(max_iters):
        if time.monotonic() - t0 > time_budget:
            break
        if best_M < 1e-12:
            break

        # FGM step with Nesterov
        f_eval = f_ynest
        conv = fast_convolve(f_eval)
        conv_max = conv.max()
        shifted = beta * (conv - conv_max)
        shifted = np.clip(shifted, -60, 60)
        exp_v = np.exp(shifted)
        tot = exp_v.sum()
        if tot < 1e-30:
            w = np.zeros_like(conv)
            w[int(np.argmax(conv))] = 1.0
        else:
            w = exp_v / tot

        grad = fft_smooth_grad(f_eval, w)

        # Step
        lr = 2.0 / (beta * (it + 1 + 1.0))
        f_step = project_simplex(f_eval - lr * grad)

        # Nesterov update
        t_next = 1.0 / (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom) / 2.0)
        f_ynest = f_step + (t_next / t_mom - 1.0) * (f_step - f)
        f_ynest = np.maximum(f_ynest, 0.0)
        fs = f_ynest.sum()
        if fs > 1e-10:
            f_ynest = f_ynest / fs
        t_mom = t_next
        f = f_step

        # Track best
        cur_M = fast_convolve(f).max()
        if cur_M < best_M:
            best_M = cur_M
            best_f = f.copy()

        # Adapt beta (increasing for more precision)
        beta *= 1.005

    # Also check f_ynest
    M_ynest = fast_convolve(f_ynest).max()
    if M_ynest < best_M:
        best_f = f_ynest.copy()

    return best_f


def subgradient_optimize(f0, max_iters=10000, time_budget=60.0, verbose=False):
    """Subgradient descent on max(conv) with simplex projection and restarting."""
    n = len(f0)
    f = np.maximum(f0, 0.0).copy()
    s = f.sum()
    if s < 1e-10:
        return f0.copy()
    f = f / s

    t0 = time.monotonic()
    best_f = f.copy()
    best_M = fast_convolve(f).max()

    # Polyak-like step size
    M_init = best_M
    M_target = M_init * 0.99  # We want to beat this

    lr = 0.05 / np.sqrt(n)
    best_iter = 0
    patience = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_budget:
            break
        if patience > 3000:
            break

        conv = fast_convolve(f)
        k_star = int(np.argmax(conv))
        M = conv.max()

        # Subgradient computation
        idx = k_star - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        s_grad = np.zeros(n)
        s_grad[mask] = 2.0 * f[idx[mask]]
        # Fix diagonal
        if k_star % 2 == 0:
            jd = k_star // 2
            if 0 <= jd < n:
                s_grad[jd] = f[jd]

        # Step
        f_new = project_simplex(f - lr * s_grad)
        f = f_new

        new_M = fast_convolve(f).max()
        if new_M < best_M:
            best_M = new_M
            best_f = f.copy()
            best_iter = it
            patience = 0
        else:
            patience += 1

        # Decrease learning rate
        lr = max(0.001 / np.sqrt(n), 0.05 / np.sqrt(n) / (1.0 + 0.01 * it))

        # Periodic restart from best
        if patience == 1000:
            f = best_f + rng_reproducible_noise(n, it) * 0.005
            f = project_simplex(f)
            patience = 0

    return best_f


def rng_reproducible_noise(n, seed):
    """Generate deterministic noise for restarts."""
    r = np.random.default_rng(seed)
    return r.standard_normal(n)


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    inits['random1'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['random2'] = rng.dirichlet(np.ones(n) * 0.1)
    inits['random3'] = rng.dirichlet(np.ones(n) * 2.0)

    for sf in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (sf * n)) ** 2)
        g /= g.sum()
        inits['gauss_%s' % str(sf)] = g

    # Bimodal
    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Frequency-domain flat spectrum
    fft_size = 1 << (n - 1).bit_length()
    for trial in range(8):
        A = np.zeros(fft_size, dtype=complex)
        A[0] = np.sqrt(n)
        phases = rng.uniform(0, 2 * np.pi, fft_size // 2)
        for k in range(1, fft_size // 2):
            A[k] = np.exp(1j * phases[k - 1])
            A[fft_size - k] = np.exp(-1j * phases[k - 1])
        if fft_size % 2 == 0:
            A[fft_size // 2] = np.exp(1j * phases[-1])
        a = np.real(ifft(A, fft_size))[:n]
        a = np.maximum(a, 0.0)
        sa = a.sum()
        if sa > 1e-10:
            a /= sa
            inits['flatspec_%d' % trial] = a

    # Parent-based
    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0)
        ps = pp.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = pp / ps
            elif len(parent) > n:
                interp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), pp)
                interp = np.maximum(interp, 0)
                is_ = interp.sum()
                if is_ > 1e-10:
                    inits['parent_ds'] = interp / is_
            else:
                interp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), pp)
                interp = np.maximum(interp, 0)
                is_ = interp.sum()
                if is_ > 1e-10:
                    inits['parent_up'] = interp / is_

    return inits


def peak_shaving_iterative(a, n_steps=50, reduction=0.97):
    """Iteratively reduce the peak of the convolution by shaving contributing elements."""
    n = len(a)
    a = np.maximum(a, 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    for _ in range(n_steps):
        conv = fast_convolve(a)
        k_max = int(np.argmax(conv))
        target = conv.max()
        j_start = max(0, k_max - n + 1)
        j_end = min(n - 1, k_max)
        # Reduce contributors to the peak
        for j in range(j_start, j_end + 1):
            other = k_max - j
            if 0 <= other < n and a[other] > 1e-15:
                a[j] *= reduction
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return np.ones(n) / n
        a = a / s
    return a


def coordinate_descent_refine(a, n_iters=50, verbose=False):
    """Simple coordinate descent: for each coordinate, try scaling up/down."""
    n = len(a)
    a = np.maximum(a, 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    best_M = fast_convolve(a).max()

    for it in range(n_iters):
        improved = False
        # Use random order
        order = np.random.default_rng(42 + it).permutation(n)
        for i in order:
            old = a[i]
            if old < 1e-15:
                continue
            for scale in [0.9, 0.95, 0.8, 1.1, 1.05, 1.2, 0.7]:
                a[i] = old * scale
                new_M = fast_convolve(a).max()
                if new_M < best_M - 1e-15:
                    best_M = new_M
                    improved = True
                    break
            a[i] = old  # restore if no improvement found in this round
            # Actually let's keep the best
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a /= s
        if not improved:
            break
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Uses iterative LP optimization (primary), FFT smooth gradient
    with Nesterov acceleration (secondary), subgradient polishing, and multi-start
    to minimize 2n*max(conv)/(sum)^2.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and p.sum() > 1e-10:
            parent = p
            print("Parent available: length=%d, sum=%.2f" % (len(p), p.sum()))
    except Exception:
        pass

    def score_and_update(a):
        nonlocal best_sequence, best_score
        if a is None:
            return float('inf')
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print("  ** New best: %.8f (n=%d, t=%.0fs)" % (score, len(a), time.monotonic() - t_start))
        return score if np.isfinite(score) else float('inf')

    # Determine working length
    n = 256
    if parent is not None:
        n = len(parent)

    print("Working length: n=%d" % n)

    # Generate initializations
    inits = generate_initializations(n, parent, rng)
    init_names = list(inits.keys())
    priority = ['parent', 'parent_ds', 'parent_up', 'flatspec_0', 'flatspec_1',
                'flatspec_2', 'flatspec_3', 'uniform', 'random1', 'gauss_0.25',
                'random2', 'flatspec_4', 'gauss_0.15', 'random3']
    init_names = [nm for nm in priority if nm in inits] + \
                 [nm for nm in init_names if nm not in priority]

    # ================================================================
    # Phase 1: LP optimization on top initializations
    # ================================================================
    print("=== Phase 1: LP Optimization (multi-start) ===")
    phase1_deadline = t_start + 400.0
    n_tries = min(8, len(init_names))
    time_per_try = max(30.0, (phase1_deadline - t_start) / n_tries)

    candidates = []

    for i in range(n_tries):
        if time.monotonic() > phase1_deadline - 15:
            break
        name = init_names[i]
        t_alloc = min(time_per_try, phase1_deadline - time.monotonic() - 10)
        if t_alloc < 15:
            break

        f_init = inits[name]
        print("  [%d/%d] LP on '%s' (budget=%.0fs)" % (i + 1, n_tries, name, t_alloc))

        # LP optimization (main method)
        t_lp = t_alloc * 0.55
        f_opt = lp_optimize(f_init, max_iters=5000, time_budget=t_lp)
        score = score_and_update(f_opt)
        candidates.append((score, f_opt.copy(), name))

        # Smooth beta optimization (secondary)
        t_smooth = min(t_alloc * 0.25, phase1_deadline - time.monotonic() - 10)
        if t_smooth > 10:
            f_smooth = smooth_beta_optimize(f_opt, beta_start=30.0,
                                            max_iters=2000, time_budget=t_smooth)
            score_s = score_and_update(f_smooth)
            candidates.append((score_s, f_smooth.copy(), name + '_smooth'))

        # Subgradient polish
        t_sub = min(t_alloc * 0.2, phase1_deadline - time.monotonic() - 5)
        if t_sub > 5:
            f_sub = subgradient_optimize(f_opt, max_iters=5000, time_budget=t_sub)
            score_sub = score_and_update(f_sub)
            candidates.append((score_sub, f_sub.copy(), name + '_sub'))

    candidates.sort(key=lambda x: x[0] if np.isfinite(x[0]) else 1e10)
    print("Phase 1 complete. Best: %.8f" % best_score)

    # ================================================================
    # Phase 2: Deep optimization on top candidates
    # ================================================================
    print("=== Phase 2: Deep Optimization ===")
    phase2_deadline = deadline - 250.0

    for score_c, f_c, name_c in candidates[:4]:
        if time.monotonic() > phase2_deadline - 20:
            break
        t_alloc = min(150.0, phase2_deadline - time.monotonic() - 15)
        if t_alloc < 20:
            break

        print("  Deep on '%s' (score=%.6f, budget=%.0fs)" % (name_c, score_c, t_alloc))

        # Long LP run
        t_lp = t_alloc * 0.45
        f_opt = lp_optimize(f_c, max_iters=15000, time_budget=t_lp)
        score = score_and_update(f_opt)

        # Smooth with higher beta
        t_sm = min(t_alloc * 0.25, phase2_deadline - time.monotonic() - 10)
        if t_sm > 10:
            f_s = smooth_beta_optimize(f_opt, beta_start=50.0,
                                       max_iters=3000, time_budget=t_sm)
            score_and_update(f_s)

        # Subgradient
        t_sb = min(t_alloc * 0.2, phase2_deadline - time.monotonic() - 5)
        if t_sb > 5:
            f_sub = subgradient_optimize(f_opt, max_iters=8000, time_budget=t_sb)
            score_and_update(f_sub)

        # Peak shaving + re-optimize
        if time.monotonic() < phase2_deadline - 15:
            f_shaved = peak_shaving_iterative(f_opt, n_steps=30, reduction=0.96)
            score_sh = score_and_update(f_shaved)
            if np.isfinite(score_sh) and score_sh <= best_score:
                f_re = lp_optimize(f_shaved, max_iters=2000,
                                   time_budget=min(20.0, phase2_deadline - time.monotonic() - 10))
                score_and_update(f_re)

    print("Phase 2 complete. Best: %.8f" % best_score)

    # ================================================================
    # Phase 3: Iterative refinement loop on best
    # ================================================================
    print("=== Phase 3: Iterative Refinement ===")
    if best_sequence is not None:
        f_best = best_sequence / best_sequence.sum()
        remaining = deadline - time.monotonic() - 30

        # Alternating LP / smooth / subgradient with decreasing budgets
        methods = ['lp', 'smooth', 'sub', 'lp', 'smooth', 'lp', 'peak', 'lp']
        for method in methods:
            if remaining < 15:
                break
            t_single = min(60.0, remaining * 0.3)
            if t_single < 5:
                break

            if method == 'lp':
                f_new = lp_optimize(f_best, max_iters=10000, time_budget=t_single)
            elif method == 'smooth':
                f_new = smooth_beta_optimize(f_best, beta_start=60.0,
                                             max_iters=3000, time_budget=t_single)
            elif method == 'sub':
                f_new = subgradient_optimize(f_best, max_iters=5000, time_budget=t_single)
            elif method == 'peak':
                f_new = peak_shaving_iterative(f_best, n_steps=40, reduction=0.95)
            else:
                f_new = f_best

            score = score_and_update(f_new)
            if np.isfinite(score) and score <= best_score:
                f_best = f_new
            remaining = deadline - time.monotonic() - 30

        # Random perturbation restarts
        if remaining > 30:
            print("  Random perturbation restarts (budget=%.0fs)" % remaining)
            n_perturb = min(20, int(remaining / 8))
            for pi in range(n_perturb):
                if time.monotonic() > deadline - 40:
                    break
                scale = 0.002 + 0.003 * rng.uniform()
                noise = rng.standard_normal(n) * scale
                f_pert = project_simplex(f_best + noise)
                t_single = min(15.0, (deadline - 40 - time.monotonic()) / max(1, n_perturb - pi))
                if t_single < 3:
                    break
                f_opt = lp_optimize(f_pert, max_iters=1000, time_budget=t_single)
                score_and_update(f_opt)

        # Coordinate descent refinement
        if time.monotonic() < deadline - 20:
            f_coord = coordinate_descent_refine(f_best, n_iters=30)
            score = score_and_update(f_coord)
            if np.isfinite(score) and score <= best_score:
                # Re-optimize from here
                f_re = lp_optimize(f_coord, max_iters=2000,
                                   time_budget=min(30.0, deadline - time.monotonic() - 15))
                score_and_update(f_re)

    # ================================================================
    # Phase 4: Try alternate lengths
    # ================================================================
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10
        if remaining > 50:
            print("=== Phase 4: Alternate Lengths ===")
            for alt_n in [128, 192, 320, 384]:
                if time.monotonic() > deadline - 15:
                    break
                t_single = min(30.0, remaining / 3)
                if t_single < 10:
                    break
                # Downsample/upsample best to new length
                if len(best_sequence) == alt_n:
                    f_alt = best_sequence.copy()
                elif len(best_sequence) > alt_n:
                    f_alt = np.interp(np.linspace(0, 1, alt_n),
                                      np.linspace(0, 1, len(best_sequence)), best_sequence)
                else:
                    f_alt = np.interp(np.linspace(0, 1, alt_n),
                                      np.linspace(0, 1, len(best_sequence)), best_sequence)
                f_alt = np.maximum(f_alt, 0.0)
                sa = f_alt.sum()
                if sa < 1e-10:
                    continue
                f_alt = f_alt / sa
                f_opt = lp_optimize(f
