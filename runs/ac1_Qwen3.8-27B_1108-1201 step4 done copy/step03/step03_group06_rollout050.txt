"""
Hybrid LP + subgradient optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses full-constraint LP directions, exact subgradients with FFT, backtracking line search,
and adaptive restarts. Focuses on precise polishing of the parent solution while exploring

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def subgradient_fft(a, conv, rel_tol=0.01):
    """
    Compute (scaled) subgradient of max_k (a*a)[k] using FFT.
    Uses entries within rel_tol of the max as active set.
    Returns grad vector of length n.
    """
    n = len(a)
    M = conv.max()
    threshold = M * (1.0 - rel_tol)
    w = (conv >= threshold).astype(np.float64)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
        w_sum = 1.0
    else:
        w = w / w_sum

    size = 2 * n
    fft_size = 1 << (size - 1).bit_length()

    w_pad = np.zeros(fft_size)
    w_pad[:len(w)] = w
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a

    W = np.fft.fft(w_pad)
    A = np.fft.fft(a_pad)

    # cross[i] = sum_j a[j] * w[j+i] (linear cross-correlation)
    cross = np.fft.ifft(A * np.conj(W)).real[:n]

    # grad = 2 * cross (as derived)
    grad = 2.0 * cross * w_sum  # scale back
    return grad

                found = False
                for t in [step * 0.3, step * 0.15, step * 0.075, 0.02, 0.01]:
                    a_candidate = a - t * grad
                    a_candidate = project_simplex(a_candidate)
                    conv_c = fast_convolve(a_candidate)
                    if conv_c.max() < best_obj - 1e-14:
                        best_obj = conv_c.max()
                        best_a = a_candidate.copy()
                        a = a_candidate
                        no_improve = 0
                        found = True
                        break
                if not found:
                    no_improve += 1
                    if no_improve % 15 == 0:
                        step = max(0.05, step * 0.9)

        iteration += 1

    return best_a, 2.0 * n * best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Bimodal (edges)
    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g

    # Cosine
    if n > 1:
        cos_w = np.cos(np.pi * x / (n - 1)) ** 2
    else:
        cos_w = np.ones(1)
    cos_w = np.maximum(cos_w, 0.01)
    cos_w /= cos_w.sum()
    inits['cosine'] = cos_w

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['random2'] = rng.dirichlet(np.ones(n) * 2.0)

    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    Phase 1: Quick exploration of lengths (120s)
    Phase 2: Deep LP+subgradient optimization of best candidates (1400s)
    Phase 3: Final polish with micro-perturbations (200s)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 60.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            p_norm = p / p.sum()
            p_conv = fast_convolve(p_norm)
            p_obj = 2.0 * len(p) * p_conv.max()
            print(f"Parent: length={len(p)}, obj={p_obj:.6f}")
    except NameError:
        pass
    except Exception:
        pass

    # ========== Phase 1: Quick Length Exploration ==========
    print("=== Phase 1: Quick Length Exploration ===")
    t_phase1 = time.monotonic()
    phase1_time = 120.0

    lengths_to_try = [256, 280, 300, 312, 320, 328, 336, 352, 384]
    scan_results = []

    for n in lengths_to_try:
        if time.monotonic() - t_phase1 > phase1_time - 20:
            break

        inits = generate_initializations(n, parent, rng)
        # Prioritize parent-based inits
        priority = ['parent', 'parent_ds', 'parent_up', 'bimodal', 'gauss', 'random']
        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in inits if nm not in priority]
        init_names = init_names[:3]

        for name in init_names:
            if time.monotonic() - t_phase1 > phase1_time - 15:
                break
            t_alloc = min(12.0, phase1_time - (time.monotonic() - t_phase1) - 10)
            if t_alloc < 5:
                break

            a0 = inits[name]
            a_opt, obj_val = optimize_core(a0, time_limit=t_alloc, seed=n + hash(name) % 997,
                                           use_full_lp=True, rel_tol=0.02)
            score = 2.0 * n * fast_convolve(a_opt).max()
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
            scan_results.append((score, n, name, a_opt.copy()))
            print(f"  Quick: n={n}, {name}, obj={score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}")

    # ========== Phase 2: Deep Optimization ==========
    print("=== Phase 2: Deep Optimization ===")
    t_phase2 = time.monotonic()
    phase2_time = 1350.0

    if best_sequence is not None:
        best_n = len(best_sequence)
    else:
        best_n = 320
        best_sequence = np.ones(best_n) / best_n

    # Start from best and a few alternatives
    starts = [(best_sequence.copy(), f"main")]
    for score, n, name, a_opt in scan_results[:4]:
        if (n, name) != (best_n, "main"):
            starts.append((a_opt.copy(), f"alt_n{n}_{name}"))
        if len(starts) >= 5:
            break

    total_time = phase2_time
    num_starts = len(starts)
    time_per_start = total_time / max(1, num_starts)

    for idx, (a_start, label) in enumerate(starts):
        if time.monotonic() > deadline - 200:
            break
        t_alloc = min(time_per_start, deadline - time.monotonic() - 200)
        if t_alloc < 20:
            break

        print(f"  Deep [{label}]: n={len(a_start)}, alloc={t_alloc:.0f}s")

        # Split time into segments with restarts
        n_seg = max(1, int(t_alloc / 120))
        seg_time = t_alloc / n_seg

        a_curr = a_start.copy()
        for seg in range(n_seg):
            if time.monotonic() > deadline - 180:
                break
            seg_alloc = min(seg_time, deadline - time.monotonic() - 180)
            if seg_alloc < 5:
                break

            # Adjust parameters for later segments (finer)
            rel_tol = 0.02 if seg < n_seg // 2 else 0.005
            use_full = True

            a_opt, obj_val = optimize_core(a_curr, time_limit=seg_alloc,
                                           seed=idx * 100 + seg,
                                           use_full_lp=use_full, rel_tol=rel_tol)

            score = 2.0 * len(a_opt) * fast_convolve(a_opt).max()
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"    Seg {seg}: score={score:.6f}")

            a_curr = a_opt

    print(f"Phase 2 best: {best_score:.6f}")

    # ========== Phase 3: Final Polish ==========
    print("=== Phase 3: Final Polish ===")
    t_phase3 = time.monotonic()
    phase3_time = max(30.0, deadline - time.monotonic())

    if best_sequence is not None:
        n_best = len(best_sequence)

        # Multiple short optimization runs with different seeds and params
        polish_configs = [
            {'rel_tol': 0.001, 'full': True},
            {'rel_tol': 0.005, 'full': True},
            {'rel_tol': 0.01, 'full': False},
            {'rel_tol': 0.002, 'full': True},
            {'rel_tol': 0.0, 'full': True},
        ]

        time_per = phase3_time / max(1, len(polish_configs) + 4)

        for i, cfg in enumerate(polish_configs):
            if time.monotonic() > deadline - 30:
                break
            t_alloc = min(time_per, deadline - time.monotonic() - 30)
            if t_alloc < 5:
                break

            a_opt, obj_val = optimize_core(best_sequence, time_limit=t_alloc,
                                           seed=500 + i, use_full_lp=cfg['full'],
                                           rel_tol=cfg['rel_tol'])
            score = 2.0 * n_best * fast_convolve(a_opt).max()
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Polish [{i}]: score={score:.6f}")

        # Micro-perturbations
        remaining = deadline - time.monotonic() - 20
        if remaining > 10:
            n_tries = int(remaining / 4)
            n_tries = min(n_tries, 15)
            for i in range(n_tries):
                if time.monotonic() > deadline - 25:
                    break
                for scale in [0.0005, 0.001, 0.002, 0.004]:
                    noise = rng.standard_normal(n_best) * scale
                    a_pert = project_simplex(best_sequence + noise)
                    score_pert = 2.0 * n_best * fast_convolve(a_pert).max()
                    if score_pert < best_score - 1e-10:
                        best_score = score_pert
                        best_sequence = a_pert.copy()
                        print(f"  Micro [{i}] s={scale}: score={score:.6f}")
                        break

    # Final verification against evaluate_sequence
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    # Scale check: ensure sum >= 0.01
    if best_sequence.sum() < 0.01:
        best_sequence = best_sequence * 1.0

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score
    elif not np.isfinite(final_score):
        # Fallback
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
