    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_exact_subgradient(a, conv, tolerance=1e-10):
    """Compute the exact subgradient (average of active peak gradients)."""
    n = len(a)
    cmax = conv.max()
    # Active peaks are those within tolerance of the max
    active = np.where(conv >= cmax - tolerance)[0]
    
    sg = np.zeros(n)
    for k in active:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            sg[j] += a[k - j]
    sg /= len(active)
    return sg


def compute_weighted_subgradient(a, conv, top_k=20, temperature=0.01):
    """Compute a softmax-weighted subgradient focusing on top peaks."""
    n = len(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    cmax = peak_vals.max()
    w = np.exp(-(peak_vals - cmax) / temperature)
    w /= w.sum()

        j_end = min(n - 1, k)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Parameterized family search for good initializations
2. Accelerated subgradient descent with adaptive step sizes
3. Simulated annealing for escape from local optima
4. Population-based multi-length search
5. Coordinate descent and final polishing

import numpy as np
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def objective(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def compute_subgrad(a, conv, top_k=50, beta=5.0):
    """Compute softmax-weighted subgradient of max(conv)."""
    n = len(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    cmax = pv.max()
    w = np.exp(beta * (pv - cmax))
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    sg = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-14:
            continue
        k = int(peaks[idx])
        wi = w[idx]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            sg[j] += wi * a[k - j]
    return sg


def subgrad_optimize(a0, time_limit, lr=0.02, mom=0.92, top_k=40, beta=5.0):
    """Momentum subgradient descent on simplex."""
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_imp = 0
    
    while time.monotonic() - t0 < time_limit:
        sg = compute_subgrad(a, conv, top_k, beta)
        v = mom * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        obj = 2.0 * n * conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        if no_imp > 200:
            lr *= 0.85
            v *= 0.5
            no_imp = 0
            if lr < 0.0005:
                break
        
        if no_imp > 800:
            rng = np.random.default_rng(int(time.monotonic() * 1e6))
            a = project_simplex(a + rng.standard_normal(n) * 0.02)
            v = np.zeros(n)
            conv = fast_convolve(a)
            no_imp = 0
            lr = max(0.01, lr * 2)
    
    return best_a, best_obj


def simulated_annealing(a0, time_limit, T_init=0.05, T_min=0.001, lr=0.01):
    """Simulated annealing: accept worse solutions with probability exp(-dE/T)."""
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    T = T_init
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        # Propose a move: random perturbation
        noise = rng.standard_normal(n) * (0.01 + 0.03 * T / T_init)
        a_prop = project_simplex(a + noise)
        conv_p = fast_convolve(a_prop)
        obj_p = 2.0 * n * conv_p.max()
        
        delta = obj_p - cur_obj
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_prop
            cur_obj = obj_p
            conv = conv_p
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        
        # Also try subgradient step
        if rng.random() < 0.3:
            sg = compute_subgrad(a, conv, 30, 5.0)
            a_prop = project_simplex(a - lr * sg)
            conv_p = fast_convolve(a_prop)
            obj_p = 2.0 * n * conv_p.max()
            delta = obj_p - cur_obj
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                a = a_prop
                cur_obj = obj_p
                conv = conv_p
                if cur_obj < best_obj:
                    best_obj = cur_obj
                    best_a = a.copy()
        
        # Anneal
        T = max(T_min, T * 0.99995)
    
    return best_a, best_obj


def coordinate_descent(a0, max_rounds=50, delta_frac=0.03):
    """Coordinate descent: redistribute mass between pairs."""
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    
    for rnd in range(max_rounds):
        improved = False
        # Process in random order for diversity
        order = np.random.default_rng(rnd).permutation(n)
        for i in order:
            if a[i] < 1e-8:
                continue
            # Find best target j to move mass to
            best_j = -1
            best_obj_j = best_obj
            for j in order:
                if j == i or a[j] > 0.5:
                    continue
                delta = a[i] * delta_frac
                if delta < 1e-9:
                    continue
                a_test = a.copy()
                a_test[i] = max(0, a_test[i] - delta)
                a_test[j] += delta
                st = a_test.sum()
                a_test /= st
                conv_t = fast_convolve(a_test)
                obj_t = 2.0 * n * conv_t.max()
                if obj_t < best_obj_j - 1e-13:
                    best_obj_j = obj_t
                    best_j = j
                    best_delta = delta
            
            if best_j >= 0:
                a[i] = max(0, a[i] - best_delta)
                a[best_j] += best_delta
                a /= a.sum()
                conv = fast_convolve(a)
                best_obj = 2.0 * n * conv.max()
                improved = True
        
        if not improved:
            break
    
    return a
