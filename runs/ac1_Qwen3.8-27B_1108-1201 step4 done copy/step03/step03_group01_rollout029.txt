Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS-B with adaptive softmax
temperature, multi-scale optimization, LP-based refinement, and diverse restarts.
Combines smooth optimization (L-BFGS-B) with global exploration (multiple
initializations at different scales and lengths) for robust convergence.

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def true_objective(a):
    """Compute the true objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (S * S)


def soft_obj(a, beta):
    """Softmax-approximated objective."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return 1e10
    a_norm = a / S
    conv = fast_convolve(a_norm)
    max_c = conv.max()
    shifted = beta * (conv - max_c)
    # Numerical stability: clip for very large shifted values
    shifted = np.minimum(shifted, 500)
    exp_vals = np.exp(shifted)
    Z = exp_vals.sum()
    return 2.0 * n * (max_c + np.log(Z) / beta)


def soft_grad(a, beta):
    """Gradient of softmax-approximated objective."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.zeros(n)
    a_norm = a / S
    conv = fast_convolve(a_norm)
    max_c = conv.max()
    shifted = beta * (conv - max_c)
    shifted = np.minimum(shifted, 500)
    exp_vals = np.exp(shifted)
    Z = exp_vals.sum()
    weights = exp_vals / Z
    
    # Cross-correlation of weights and a_norm
    fft_size = 1 << (2 * n - 1).bit_length()
    W = rfft(weights, fft_size)
    A = rfft(a_norm, fft_size)
    g_an = irfft(W * np.conj(A), fft_size)[:n]
    
    # Chain rule: a_norm = a/S
    g_an_dot_a = np.dot(g_an, a)
    grad = 2.0 * n * (g_an / S - g_an_dot_a * a / (S * S))
    return grad

    S = a.sum()
    if S < 1e-15:
        return a0, float('inf')
    a = a / S

    best_a = a.copy()
    best_obj = true_objective(a)
    t0 = time.monotonic()

        if best_obj < 1.001:  # Can't do much better

            soft_obj, a, jac=soft_grad,

        best_t = 0.0
        best_local_obj = best_obj
        for t in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
            a_try = a + t * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            obj_try = true_objective(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_t = t
        
        if best_local_obj < best_obj - 1e-12:
            a_new = a + best_t * d
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            a = a_new / s_new
            conv = fast_convolve(a)
            M = conv.max()
            best_obj = 2.0 * n * M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                # Perturb and restart
                rng = np.random.default_rng(int(time.monotonic() * 1000))
                noise = rng.standard_normal(n) * 0.01
                a = best_a + noise
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                    conv = fast_convolve(a)
                    M = conv.max()
                no_improve = 0
    
    return best_a, best_obj


def generate_initialization(n, parent, rng, method):
    """Generate an initialization of length n."""
    if method == 'uniform':
        return np.ones(n) / n
    elif method == 'random':
        return rng.dirichlet(np.ones(n) * 0.5)
    elif method == 'gaussian':
        x = np.arange(n)
        g = np.exp(-0.5 * ((x - n / 2) / (n / 6)) ** 2)
        return g / g.sum()
    elif method == 'triangle':
        x = np.arange(n)
        t = np.minimum(x, n - 1 - x) + 1.0
        return t / t.sum()
    elif method == 'bimodal':
        a = np.zeros(n)
        edge = max(1, n // 4)
        a[:edge] = 1.0
        a[-edge:] = 1.0
        return a / a.sum()
    elif method == 'sine':
        x = np.arange(n)
        s = np.sin(np.pi * (x + 0.5) / n) + 0.1
        return s / s.sum()
    elif method == 'parent_interp':
        if parent is None:
            return np.ones(n) / n
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps < 1e-10:
            return np.ones(n) / n
        p = p / ps
        if len(p) == n:
            return p.copy()
        elif len(p) > n:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
            pp = np.maximum(pp, 0)
            pps = pp.sum()
            if pps < 1e-10:
                return np.ones(n) / n
            return pp / pps
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
            pp = np.maximum(pp, 0)
            pps = pp.sum()
            if pps < 1e-10:
                return np.ones(n) / n
            return pp / pps
    elif method == 'parent_noisy':
        if parent is None:
            return np.ones(n) / n
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps < 1e-10:
            return np.ones(n) / n
        p = p / ps
        if len(p) != n:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
            p = pp
            ps = p.sum()
            if ps < 1e-10:
                return np.ones(n) / n
            p = p / ps
        noise = rng.standard_normal(n) * 0.05
        a = p + noise
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            return np.ones(n) / n
        return a / s
    elif method == 'palindromic':
        half = np.ones((n + 1) // 2)
        if n % 2 == 0:
            a = np.concatenate([half, half[::-1]])
        else:
            a = np.concatenate([half[:-1], half[-1:], half[-2::-1]])
        return a / a.sum()
    elif method == 'raised_cosine':
        x = np.arange(n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x / n))
        a += 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n
