import numpy as np
import time

def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses Nesterov-accelerated subgradient descent with adaptive multi-peak weighting,
coordinate descent polishing, parametric initialization, and multi-length scanning.
Key innovations: exponential temperature annealing for peak focus, backtracking
step sizes, and coordinate descent exploiting the scale-invariance of the objective.
"""

import numpy as np
import time


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def compute_obj_raw(a):
    """Compute 2*n*max_conv/sum^2 for raw (unnormalized) a."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def project_simplex(v):
    """Project vector onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def multi_peak_subgrad(a, conv, top_k, temperature):
    """Compute weighted subgradient using top-k peaks with softmax weighting."""
    n = len(a)
    k_use = min(top_k, len(conv))
    if k_use <= 0:
        return np.zeros(n)
    peaks = np.argpartition(conv, -k_use)[-k_use:]
    vals = conv[peaks]
    max_val = vals.max()
    if max_val < 1e-15:
        w = np.ones(k_use) / k_use
    else:
        w = np.exp((vals - max_val) / (temperature * max_val + 1e-15))
        w /= w.sum()

    g = np.zeros(n)
    for i in range(k_use):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je >= js:
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * w[i] * a[k - idx]
    return g


def nesterov_optimize(a0, time_limit, top_k=80, temp_init=0.5, verbose=False, tag=""):
    """
    Nesterov-style accelerated subgradient descent on the simplex.
    Uses adaptive temperature, backtracking step size, and stall recovery.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    momentum = 0.9
    lr = 1.0 / np.sqrt(n)
    temperature = temp_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    stall_limit = 400
    best_lr = lr

    while time.monotonic() - t0 < time_limit:
        batch_end = min(t0 + 0.05, t0 + time_limit)  # ~20ms per iteration target
        conv = fast_convolve(a)
        g = multi_peak_subgrad(a, conv, top_k, temperature)
        g -= g.mean()

        v = momentum * v - lr * g
        a_new = project_simplex(a + v)

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            best_lr = lr
        else:
            no_improve += 1

        a = a_new

        # Backtracking: if stuck, reduce lr
        if no_improve > 200:
            lr = max(lr * 0.95, 1e-6)
        elif no_improve == 0:
            lr = min(lr * 1.005, best_lr * 2)

        # Temperature annealing
        if it % 500 == 499:
            temperature = max(0.05, temperature * 0.97)

        # Momentum decay when stuck
        if no_improve > 200:
            momentum = max(0.5, momentum * 0.9995)

        # Stall recovery with perturbation
        if no_improve >= stall_limit:
            rng = np.random.default_rng((it * 7919 + n) % (2**31))
            noise = rng.standard_normal(n) * (0.005 + 0.003 * np.log1p(it / 100))
            a = project_simplex(a + noise)
            v = np.zeros(n)
            no_improve = 0
            stall_limit = int(stall_limit * 1.5)
            temperature = min(1.5, temperature * 2.0)
            momentum = 0.9
            lr = max(lr, 0.5 / np.sqrt(n))

        it += 1
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {tag} Nesterov it={it}, obj={best_obj:.6f}, lr={lr:.4e}, "
                  f"temp={temperature:.3f}, mom={momentum:.3f}, elapsed={elapsed:.0f}s")

    return best_a, best_obj


def gradient_descent_simple(a0, time_limit, lr_init=0.1, verbose=False, tag=""):
    """
    Simple projected subgradient with adaptive lr (no momentum for stability).
    Used as a fallback or for short bursts.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    lr = lr_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        # Use top peaks with softmax
        top_k = min(50, 2 * n - 1)
        g = multi_peak_subgrad(a, conv, top_k, temperature=0.3)
        g -= g.mean()

        a_new = project_simplex(a - lr * g)
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            lr = min(lr * 1.01, 1.0)
        else:
            no_improve += 1
            lr *= 0.98

        a = a_new
        if no_improve > 500:
            rng = np.random.default_rng(it + 999)
            a = project_simplex(a + rng.standard_normal(n) * 0.008)
            no_improve = 0
            lr = lr_init

        it += 1
        if verbose and it % 2000 == 0:
            print(f"    {tag} GradDesc it={it}, obj={best_obj:.6f}, lr={lr:.4e}, elapsed={time.monotonic()-t0:.0f}s")

    return best_a, best_obj


def coordinate_descent(a0, time_limit, n_trials=5, verbose=False, tag=""):
    """
    Cyclic coordinate descent for the scale-invariant objective.
    For each coordinate j, tries several candidate values and picks the best.
    Works without simplex constraint (exploits scale invariance).
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s  # Start normalized

    best_obj = compute_obj_raw(a)
    best_a = a.copy()

    t0 = time.monotonic()
    pass_count = 0
    rng = np.random.default_rng(12345)
    no_full_improve = 0

    while time.monotonic() - t0 < time_limit:
        # Check time budget per pass
        remaining = time_limit - (time.monotonic() - t0)
        estimated_pass_time = n * n_trials * 0.002  # rough estimate
        if estimated_pass_time > remaining * 0.8:
            break

        order = rng.permutation(n)
        pass_improved = False

        for j in order:
            if time.monotonic() - t0 > time_limit - 0.5:
                break

            x_cur = a[j]
            if x_cur < 1e-15:
                # Try small values
                candidates = [x_cur, 1.0 / n * 0.1, 1.0 / n * 0.5, 1.0 / n]
            else:
                # Try relative perturbations
                candidates = [0.0, x_cur * 0.3, x_cur * 0.7, x_cur, x_cur * 1.3, x_cur * 1.7]
                # Add a value that might balance the peak
                candidates.append(x_cur * (0.5 + 0.5 * np.random.random()))

            best_x = x_cur
            best_val = best_obj  # Use current best as reference (avoids recomputing)

            # Compute current conv once
            conv_cur = fast_convolve(a)
            s_cur = a.sum()
            cur_val = 2.0 * n * conv_cur.max() / (s_cur * s_cur) if s_cur > 1e-15 else float('inf')

            for x in candidates:
                if x < -1e-15:
                    continue
                x = max(0.0, x)
                old = a[j]
                a[j] = x
                new_s = a.sum()
                if new_s < 1e-15:
                    a[j] = old
                    continue
                # Quick objective: recompute conv (expensive but necessary)
                # For efficiency, only do full conv for the best few candidates
                # Use a cheap upper bound first
                # Actually just compute it - for n<=1024 it's fast enough
                conv_try = fast_convolve(a)
                val_try = 2.0 * n * conv_try.max() / (new_s * new_s)
                if val_try < best_val:
                    best_val = val_try
                    best_x = x
                a[j] = old

            if best_x != x_cur:
                a[j] = best_x
                pass_improved = True

        # End of pass
        final_conv = fast_convolve(a)
        final_s = a.sum()
        cur_obj = 2.0 * n * final_conv.max() / (final_s * final_s) if final_s > 1e-15 else float('inf')
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_full_improve = 0
        else:
            no_full_improve += 1

        pass_count += 1
        if verbose and (pass_count % 5 == 0 or no_full_improve > 5):
            print(f"    {tag} CoordDesc pass={pass_count}, obj={best_obj:.6f}, "
                  f"improved={pass_improved}, elapsed={time.monotonic()-t0:.0f}s")

        # If no improvement for several passes, perturb
        if no_full_improve > 8:
            scale = 0.002 * (no_full_improve - 5)
            a = np.maximum(a + rng.standard_normal(n) * scale, 0.0)
            no_full_improve = 0

    return best_a, best_obj


def generate_parametric_inits(n, parent=None, rng=None):
    """Generate diverse parametric initializations."""
    if rng is None:
        rng = np.random.default_rng(42)

    inits = {}
    x = np.arange(n, dtype=np.float64)
    t = (x + 0.5) / n  # Normalized position in [0, 1]

    # Raised cosine (Hann window)
    hann = 0.5 * (1 + np.cos(2 * np.pi * t))
    inits['hann'] = hann / hann.sum()

    # Sine squared
    sin2 = np.sin(np.pi * t) ** 2
    inits['sin2'] = sin2 / sin2.sum()

    # Generalized: sin^p
    for p in [0.5, 1.0, 2.0, 4.0]:
        sp = np.sin(np.pi * t) ** p
        inits[f'sin_{p}'] = sp / sp.sum()

    # Edge concentrated (flat at edges, zero in middle)
    edge_width = max(1, n // 4)
    edge = np.zeros(n)
    edge[:edge_width] = 1.0
    edge[-edge_width:] = 1.0
    inits['edge_flat'] = edge / edge.sum()

    # Smooth edge (Gaussian at each edge)
    sigma = n * 0.1
    g1 = np.exp(-0.5 * (x / sigma) ** 2)
    g2 = np.exp(-0.5 * ((x - n + 1) / sigma) ** 2)
    edge_smooth = g1 + g2
    inits['edge_gauss'] = edge_smooth / edge_smooth.sum()

    # Power law edge
    for alpha in [0.5, 1.0, 2.0]:
        d = np.minimum(t, 1 - t)  # Distance to nearest edge
        pw = d ** (-alpha)
        pw[d == 0] = 1e10  # Handle edge
        inits[f'pow_{alpha}'] = pw / pw.sum()

    # W-shape (high at edges and center)
    center_g = np.exp(-0.5 * ((t - 0.5) / 0.1) ** 2)
    w_shape = edge_smooth + 0.5 * center_g
    inits['w_shape'] = w_shape / w_shape.sum()

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Sparse random
    sparse = rng.random(n)
    sparse[sparse < 0.4] = 0.0
    if sparse.sum() < 1e-10:
        sparse = rng.random(n)
    inits['sparse'] = sparse / sparse.sum()

    # Parent interpolation
    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0.0)
        ps = pp.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = pp / ps
            else:
                pp_interp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), pp)
                ps2 = pp_interp.sum()
                if ps2 > 1e-10:
                    inits['parent_interp'] = pp_interp / ps2

    return inits

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
