def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    A_ub = np.zeros((2 * n - 1, n))

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options=opts)
    if res.success:
        return res.x
    return None

Wait, I realize the Nesterov acceleration for FW is a bit different. The standard Nesterov FW (NFA) works as follows:

1. Initialize x_0, y_0 = x_0, t_0 = 1
2. At iteration t:

   b. γ_t = 2/(t+2)
   c. x_{t+1} = (1-γ_t) y_t + γ_t s_t
   d. t_{t+1} = (1 + sqrt(1 + 4t_t^2))/2
   e. y_{t+1} = x_{t+1} + (t_t/t_{t+1} - 1) * (x_{t+1} - x_t)

But this assumes smoothness. For non-smooth objectives (like max of convex functions), we need a variant.

Let me also try a "direct" approach: since the objective is `2n * max_k (a*a)[k] / (sum a)^2`, and we're on the simplex (sum=1), we want to minimize `max_k (a*a)[k]`. The optimal solution satisfies:
- The gradient of the active constraints (peak lags) is in the normal cone of the simplex.
- This means the subgradient is a constant vector (on the support) and ≤ 0 (outside).

So at the optimum, there exist weights w_k >= 0 for the active lags such that:
`sum_{k active} w_k * grad_k = λ * 1_vector` (on the support)
`sum_{k active} w_k * grad_k[i] <= λ` (outside the support)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov-style accelerated Frank-Wolfe
with LP oracle, multi-peak subgradient polishing, and a novel greedy peak-shaving
coordinate descent. Exploits asymmetry of optimal sequences and uses adaptive
restarts with structured perturbations.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, peaks, weights=None):
    """Compute subgradient of max_k (f*f)[k] at f, using given peak lags."""
    n = len(f)
    if weights is None:
        weights = np.ones(len(peaks)) / len(peaks)
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if weights[i] < 1e-14:
            continue
        k = int(peaks[i])
        # d(f*f)[k]/df[j] = 2*f[k-j] if 0<=k-j<n else 0
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * weights[i] * f[k - idx]
    return sg


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle, bisection line search, and adaptive restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = solve_lp_direction(f, M * m_exp)
        if g is None:
            m_exp *= 1.15
            if m_exp > 5.0:
                m_exp = 1.0
                # Subgradient fallback
                peaks = np.where(conv >= 0.9 * M)[0]
                pv = conv[peaks]
                scaled = pv / max(M * 0.001, 1e-15)
                scaled -= scaled.max()
                w = np.exp(scaled)
                w /= w.sum()
                sg = compute_subgradient(f, peaks, w)
                f = project_simplex(f - 0.01 * sg / max(sg.max(), 1e-15))
                conv = fast_convolve(f)
                M = conv.max()
                continue
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Fixed point check
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 50:
                break
            m_exp = min(m_exp * 1.03, 2.5)
            continue

        # Bisection line search with broader initial scan
        best_alpha = 0.0
        best_val = M

        alphas_coarse = np.linspace(0.02, 1.0, 25)
        for alpha in alphas_coarse:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine bisection
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-13:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"    FW iter {it}: obj={best_obj:.10f}, alpha={best_alpha:.4f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 4.0)

        if no_improve > 80:
            rng = np.random.default_rng(it * 31 + 17)
            scale = 0.002 + 0.004 * min((no_improve - 80) // 50, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.003, mom=0.93, temp=0.00005):
    """Subgradient descent with multi-peak softmax weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    top_k = max(10, n // 6)

    while time.monotonic() - t0 < time_limit:
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = compute_subgradient(f, peaks, w)
        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-13:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9995
        cur_temp *= 1.002

        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 100000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0

    return best_f, best_obj


def coord_descent_polish(f0, time_limit, verbose=False):
    """
    Greedy peak-shaving coordinate descent.
    At each step, identify the peak lag and shift mass from high-gradient
    coordinates to low-gradient coordinates to reduce the peak.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    step = 0.005  # initial step size
    patience = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()

        # Find all near-peak lags
        peak_threshold = 0.98 * M
        peaks = np.where(conv >= peak_threshold)[0]

        if len(peaks) == 0:
            peaks = np.array([np.argmax(conv)])

        # Compute weighted subgradient
        pv = conv[peaks]
        # Use inverse-value weighting: peaks that are higher get more weight
        w = (pv / M) ** 4
        w /= w.sum()

        sg = compute_subgradient(f, peaks, w)
        sg_norm = sg.max()
        if sg_norm < 1e-15:
            break
        sg = sg / sg_norm

        # Find coordinates to decrease (high sg) and increase (low sg)
        # Only consider coordinates with sufficient mass to decrease
        decrease_mask = (f > 0.01 * f.sum() / n) & (sg > np.median(sg))
        increase_mask = (sg < np.median(sg))

        if not decrease_mask.any() or not increase_mask.any():
            step *= 0.8
            patience += 1
            if patience > 50:
                break
            continue

        # Water-filling: move mass from high-sg to low-sg coords
        total_decrease = f[decrease_mask].sum()
        # Limit step
        max_step = 0.3 * total_decrease
        actual_step = min(step, max_step)

        # Proportional decrease from high-sg coords
        dec_amount = sg[decrease_mask]
        dec_amount /= dec_amount.sum()
        f[decrease_mask] -= actual_step * dec_amount

        # Proportional increase to low-sg coords (inverse sg weighting)
        inc_weight = np.maximum(1.0 - sg[increase_mask], 0.01)
        inc_weight /= inc_weight.sum()
        f[increase_mask] += actual_step * inc_weight

        f = np.maximum(f, 0.0)
        f = f / f.sum()

        conv = fast_convolve(f)
        new_obj = conv.max()

        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_f = f.copy()
            patience = 0
            step = min(step * 1.01, 0.02)
        else:
            patience += 1
            step *= 0.92

        if patience > 100:
            break

    return best_f, best_obj
