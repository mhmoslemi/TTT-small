import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_convolve(a, b=None):
    n = len(a)
    if b is None:
        b = a
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    rows, cols, vals = [], [], []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_oracle(a, M):
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': 10.0})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def compute_obj(a):

    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

"""
Hybrid Nesterov-accelerated subgradient + LP-based Frank-Wolfe optimizer
for minimizing 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Key improvements over previous: better diverse initializations (V-shape,
exponential edge, chirp), Nesterov momentum for faster convergence,
peak-equalization local search, and more aggressive multi-length exploration.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build sparse matrix A such that (A @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_oracle(a, M, time_limit=8.0):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': time_limit})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def compute_obj(a):
    """Compute 2*n*max(conv) assuming sum(a)~1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def nesterov_optimize(a0, time_limit, lr_init=0.008, top_k=20, temp_frac=0.03):
    """
    Nesterov-accelerated subgradient method on the simplex.
    Uses soft-max weighting of top peaks for subgradient computation.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = compute_obj(a)
    best_a = a.copy()

    y = a.copy()
    v = np.zeros(n)
    t_k = 1.0
    lr = lr_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    best_no_improve_lr = lr_init

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(y)
        top_k_eff = min(top_k, len(conv))
        peaks = np.argpartition(conv, -top_k_eff)[-top_k_eff:]
        peak_vals = conv[peaks]
        temp = max(peak_vals.max() * temp_frac, 1e-15)
        scaled = (peak_vals - peak_vals.min()) / temp
        w = np.exp(scaled)
        w /= w.sum()

        g = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            g[idx] += 2.0 * w[i] * y[k - idx]

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = (t_k - 1.0) / t_new

        y = np.maximum(a + v, 0.0)
        s = y.sum()
        if s < 1e-15:
            y = np.ones(n) / n
            s = 1.0
        y = y / s
        a = y
        t_k = t_new
        lr *= 0.9992

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

            t_k = 1.0

        it += 1

def fpw_lp_optimize(a0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive step and stagnation handling."""
    n = len(a0)
    a = np.maximum(a0, 0.0).copy()
    a = a / a.sum()

    conv = fast_convolve(a)
    M = conv.max()

    no_improve = 0
    max_stall = 80

        # Use tighter M to force improvement
        M_try = M * (1.0 - 5e-5)
        tl = min(8.0, max(2.0, remaining * 0.3))
        b = lp_oracle(a, M_try, time_limit=tl)
        if b is None:
            M_try = M * 1.0005

        if b is None:
            break
        sb = b.sum()
        if sb < 1e-15:
            break
        b = b / sb

        if np.allclose(a, b, atol=1e-10):
            no_improve += 1
        else:
            # Line search with more points early on
            n_ls = 8 if it < 50 else 5
            best_t = 0.0
            best_local = best_obj
            ts = np.concatenate([np.linspace(0.05, 0.5, n_ls // 2),
                                 np.linspace(0.5, 0.95, n_ls - n_ls // 2)])

                candidate = (1.0 - t) * a + t * b

def peak_equalize(a0, time_limit, lr=0.005):
    """
    Local optimization that explicitly tries to equalize the active peak
    convolution positions. Projects gradient of (max - mean_peaks)^2.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0).copy()
    a = a / a.sum()

    conv = fast_convolve(a)
    best_obj = compute_obj(a)
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        # Identify peaks: positions where conv is within 2% of max
        threshold = conv.max() * 0.98
        peak_indices = np.where(conv >= threshold)[0]
        if len(peak_indices) == 0:
            peak_indices = np.array([np.argmax(conv)])
        if len(peak_indices) > 30:
            # Keep only the top 30
            top30 = np.argpartition(conv, -30)[-30:]
            peak_indices = top30

        # Compute gradient to reduce variance of peak values
        peak_vals = conv[peak_indices]
        mean_peak = peak_vals.mean()
        g = np.zeros(n)
        for k in peak_indices:
            residual = conv[k] - mean_peak
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            g[idx] += 2.0 * residual * a[k - idx]

        a_new = np.maximum(a - lr * g, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a
        else:
            a_new = a_new / s

        conv_new = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv_new.max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.98

        if no_improve > 300:
            lr = max(lr, 0.0001)
            no_improve = 0

        it += 1

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # V-shape (high at edges, low center)
    v = np.maximum(x, n - 1 - x).astype(np.float64) / n
    v += 0.05
    v /= v.sum()
    inits['v_shape'] = v

    # Exponential edge decay (strong edges)
    decay = 0.4
    e = np.maximum(np.exp(-decay * x), np.exp(-decay * (n - 1 - x)))
    e /= e.sum()
    inits['exp_edge'] = e

    # Moderate exponential edge
    decay2 = 0.15
    e2 = np.maximum(np.exp(-decay2 * x), np.exp(-decay2 * (n - 1 - x)))
    e2 /= e2.sum()
    inits['exp_edge_mod'] = e2

    # Asymmetric ramps
    ramp = np.linspace(0.05, 1.0, n)
    ramp /= ramp.sum()
    inits['ramp'] = ramp
    inits['ramp_rev'] = ramp[::-1].copy()

    # Inverted V (center heavy)
    iv = np.minimum(x, n - 1 - x).astype(np.float64) / n
    iv += 0.05
    iv /= iv.sum()
    inits['inv_v'] = iv

    # Gaussian (center)
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g

    # Cosine window
    cw = 0.5 * (1 + np.cos(np.pi * x / (n - 1))) if n > 1 else np.ones(1)
    cw += 0.01
    cw /= cw.sum()
    inits['cosine'] = cw

    # Chirp-modulated
    t = np.linspace(0, 1, n)
    chirp = 0.5 + 0.5 * np.cos(2 * np.pi * 5 * t)
    chirp += 0.02
    chirp /= chirp.sum()
    inits['chirp'] = chirp

    # Random (Dirichlet)
    inits['random'] = rng.dirichlet(np.ones(n) * 1.0)

    # Random heavy-tailed
    heavy = rng.pareto(2.0, n) + 0.01
    heavy /= heavy.sum()
    inits['heavy'] = heavy

    # Two-peak (bimodal)
    peak1 = n // 4
    peak2 = 3 * n // 4
    sigma = n // 10
    b1 = np.exp(-0.5 * ((x - peak1) / sigma) ** 2)
    b2 = np.exp(-0.5 * ((x - peak2) / sigma) ** 2)
    bim = b1 + b2 + 0.01
    bim /= bim.sum()
    inits['bimodal'] = bim

    # Parent (interpolated if needed)
    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0.0)
        if len(parent) == n:
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp_i = np.interp(np.linspace(0, 1, n),
                             np.linspace(0, 1, len(parent)), pp)
            ps = pp_i.sum()
            if ps > 1e-10:
                inits['parent'] = pp_i / ps

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search:
    Phase 1: Fast Nesterov subgradient scan over lengths and initializations
    Phase 2: LP-based Frank-Wolfe refinement on top candidates
    Phase 3: Peak equalization + final LP pass
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 45.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.2f}, score={evaluate_sequence(p.tolist()):.6f}")
    except Exception:
        pass

    # Determine candidate lengths
    if parent is not None:
        pn = len(parent)
        lengths = sorted(set([pn, pn - 32, pn + 32, pn - 64, pn + 64,
                              256, 320, 384, 448, 512]))
    else:
        lengths = [256, 320, 384, 448, 512]
    lengths = [l for l in lengths if 128 <= l <= 768]

    # === Phase 1: Nesterov scan ===
    print("=== Phase 1: Nesterov Scan ===")
    phase1_results = []
    total_time = budget_s - 45.0
    phase1_time = total_time * 0.50

    for n in lengths:
        if time.monotonic() > deadline - total_time * 0.40:
            break
        inits = generate_inits(n, parent, rng)
        init_names = list(inits.keys())
        priority = ['parent', 'v_shape', 'exp_edge', 'exp_edge_mod', 'uniform',
                    'ramp', 'cosine', 'bimodal', 'heavy', 'gauss',
                    'random', 'chirp', 'inv_v', 'ramp_rev']
        init_names.sort(key=lambda x: priority.index(x) if x in priority else 99)

        # Allocate time: more for parent, less for others
        n_to_try = min(len(init_names), 10)
        time_per = max(4.0, phase1_time / (len(lengths) * n_to_try))

        for name in init_names[:n_to_try]:
            if time.monotonic() > deadline - total_time * 0.45:
                break
            t_alloc = min(time_per * 2, deadline - time.monotonic() - total_time * 0.45)
            if t_alloc < 3:
                break

            a0 = inits[name]
            a_opt, obj_opt = nesterov_optimize(a0, time_limit=t_alloc,
                                               lr_init=0.008, top_k=20)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                phase1_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    print(f"After Phase 1: best={best_score:.6f} ({len(phase1_results)} candidates)")

    # Sort results
    phase1_results.sort(key=lambda x: x[0])
    for s, n, nm, _ in phase1_results[:5]:
        print(f"  Candidate: score={s:.6f}, n={n}, init={nm}")

    # === Phase 2: LP refinement on top candidates ===
    print("\n=== Phase 2: LP Refinement ===")
    phase2_time = (budget_s - 45.0) * 0.25

    for i, (score_i, n_i, name_i, a_i) in enumerate(phase1_results[:4]):
        if time.monotonic() > deadline - phase2_time - 60:
            break
        elapsed_p1 = time.monotonic() - ((budget_s - 45.0) * 1.0)  # rough
        remaining_p2 = deadline - time.monotonic() - 90
        if remaining_p2 < 10:
            break
        t_alloc = min(150.0, remaining_p2 * 0.4)
        if t_alloc < 10:
            break

        print(f"  LP on {name_i} (n={n_i}, score={score_i:.6f}), t={t_alloc:.0f}s")
        a_opt, obj_opt = fpw_lp_optimize(a_i, time_limit=t_alloc, verbose=False)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            print(f"    -> score={score:.6f}")
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()

    # === Phase 3: Deep optimization on best ===
    print(f"\n=== Phase 3: Deep Optimization (best={best_score:.6f}) ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 60
        if remaining > 30:
            # LP-based FW
            t_lp = remaining * 0.5
            print(f"  Deep LP: {t_lp:.0f}s")
            a_opt, obj_opt = fpw_lp_optimize(best_sequence, time_limit=t_lp, verbose=True)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"    Deep LP -> {score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # === Phase 4: Peak equalization ===
    if best_sequence is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic() - 25
        if remaining > 15:
            print(f"\n=== Phase 4: Peak Equalization ({remaining:.0f}s) ===")
            a_opt, obj_opt = peak_equalize(best_sequence, time_limit=remaining * 0.6, lr=0.003)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"    Peak eq -> {score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # === Phase 5: Final LP pass ===
    if best_sequence is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic() - 10
        if remaining > 10:
            print(f"\n=== Phase 5: Final LP Pass ({remaining:.0f}s) ===")
            a_opt, obj_opt = fpw_lp_optimize(best_sequence, time_limit=remaining, verbose=False)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"    Final LP -> {score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
