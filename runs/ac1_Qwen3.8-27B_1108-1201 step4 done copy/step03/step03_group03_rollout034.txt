Key innovation: uses log-sum-exp (soft max) as a smooth surrogate for the max,

    A = np.fft.rfft(a, fft_size)

    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])

    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():

    rho = np.nonzero(cond)[0][-1]

    b_shifted = b / T - (b.max() / T)
    return T * (b.max() + np.log(np.exp(b_shifted).sum()))

            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                if a[j] > 1e-6 and a[k - j] > 1e-6:
                    candidates.add(j)
                    candidates.add(k - j)

        # Try random moves from candidates
        cand_list = list(candidates)
        if not cand_list:
            break
        rng.shuffle(cand_list)

        for i in cand_list[:min(50, len(cand_list))]:
            if time.monotonic() - t0 > time_limit - 2:
                break
            if a[i] < 1e-7:
                continue
            # Try redistributing a fraction of a[i] to random other positions
            for _ in range(3):
                j = rng.integers(0, n)
                if j == i:
                    continue
                for frac in [0.05, 0.1, 0.2]:
                    delta = a[i] * frac
                    if delta < 1e-9:
                        continue
                    a_try = a.copy()
                    a_try[i] -= delta
                    a_try[j] += delta
                    if a_try[i] < 0:
                        continue
                    a_try = np.maximum(a_try, 0)
                    st = a_try.sum()
                    if st < 1e-15:
                        continue
                    a_try /= st
                    conv_try = fast_convolve(a_try)
                    obj_try = 2.0 * n * conv_try.max()
                    if obj_try < best_obj - 1e-12:
                        a = a_try
                        best_obj = obj_try
                        conv = conv_try
                        improved = True
                        break
        if not improved:
            break

    return a


def gen_inits(n, parent=None, rng=None):
    """Generate diverse edge-concentrated initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)
    xm = x - 0.5 * (n - 1)
    xnorm = xm / (0.5 * (n - 1) + 1e-10)  # in [-1, 1]

    # U-shaped (quadratic)
    for gamma in [1.5, 2.0, 3.0, 4.0]:
        a = np.abs(xnorm) ** gamma + 0.01
        inits[f'u{gamma}'] = a / a.sum()

    # Cosine-based edge concentration
    for amp in [0.5, 1.0, 2.0, 3.0]:
        a = 1.0 + amp * (xnorm ** 2)
        inits[f'cos{amp}'] = a / a.sum()

    # Sharp edge (step-like)
    e = max(1, n // 8)
        a_step = np.zeros(n)
    a_step[:e] = 1.0
    a_step[-e:] = 1.0
    a_step += 0.01
    inits['step8'] = a_step / a_step.sum()

    e2 = max(1, n // 4)
    a_step2 = np.zeros(n)
    a_step2[:e2] = 1.0
    a_step2[-e2:] = 1.0
    a_step2 += 0.005
    inits['step4'] = a_step2 / a_step2.sum()

    # Bimodal (two edge bumps)
    for width in [0.1, 0.15, 0.2]:
        g = np.exp(-(xnorm - 0.8) ** 2 / (2 * width ** 2)) + \
            np.exp(-(xnorm + 0.8) ** 2 / (2 * width ** 2)) + 0.005
        inits[f'bim{width}'] = g / g.sum()

    # Uniform (baseline)
    inits['uniform'] = np.ones(n) / n

    # Random
    inits['rand'] = rng.dirichlet(np.ones(n) * 0.3)

    # Parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = p / ps
            else:
                p_interp = np.interp(np.linspace(0, 1, n),
                                     np.linspace(0, 1, len(parent)), p)
                inits['parent_interp'] = p_interp / p_interp.sum()

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search combining soft-annealing gradient descent, LP corrections,
    subgradient polish, and coordinate descent. Explores multiple sequence lengths
    and edge-concentrated initializations.
    """
    rng = np.random.default_rng(seed)
    time_start = time.monotonic()
    deadline = time_start + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            par_score = float(evaluate_sequence(p.tolist()))
            print(f"Parent: n={len(p)}, score={par_score:.6f}")
            if par_score < best_score:
                best_score = par_score
                best_sequence = p.copy()
    except Exception:
        pass

    # Phase 1: Soft-annealing scan over lengths and initializations
    print("=== Phase 1: Soft-Annealing Scan ===")
    lengths = [384, 512, 640, 768, 1024]
    results = []

    phase1_deadline = time_start + min(700.0, (deadline - time_start) * 0.45)

    for n in lengths:
        if time.monotonic() > phase1_deadline - 30:
            break
        inits = gen_inits(n, parent, rng)

        # Priority: parent first, then edge-concentrated
        prio = ['parent', 'parent_interp', 'u2.0', 'u3.0', 'cos1.0',
                'step8', 'bim0.1', 'u4.0', 'cos2.0', 'rand', 'uniform']
        names = [nm for nm in prio if nm in inits]
        names = names[:6]

        t_per_init = max(15.0, 150.0 / max(1, len(lengths) * 6))

        for name in names:
            if time.monotonic() > phase1_deadline - 10:
                break
            t_alloc = min(t_per_init, phase1_deadline - time.monotonic() - 8)
            if t_alloc < 8:
                break

            a0 = inits[name]
            t_where = time.monotonic()
            a_opt, obj_opt = soft_anneal_optimize(
                a0, t_alloc,
                T_start=None, T_end=None,
                eta_init=0.08, eta_min=1e-7,
                halve_every=150, lp_every=250, n_lp_max=5)

            # Quick subgrad polish
            t_rem_polish = min(8.0, t_alloc * 0.1)
            if t_rem_polish > 2:
                a_opt2, obj2 = subgrad_polish(a_opt, t_rem_polish,
                                              lr_init=0.008, momentum=0.92,
                                              top_k=20, temp_scale=0.04)
                if obj2 < obj_opt:
                    a_opt, obj_opt = a_opt2, obj2

            score = float(evaluate_sequence(a_opt.tolist()))
            elapsed = time.monotonic() - t_where
            if np.isfinite(score):
                results.append((score, n, name, a_opt.copy()))
                tag = f"  n={n:4d} {name:12s}: {score:.6f} ({elapsed:.1f}s)"
                print(tag)
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    results.sort(key=lambda x: x[0])
    print(f"After Phase 1: best={best_score:.6f}")

    # Phase 2: Deep optimization on top candidates
    print("=== Phase 2: Deep Optimization ===")
    top_list = []
    seen = set()
    for score, n, name, a_opt in results[:10]:
        key = (n, name)
        if key not in seen:
            top_list.append((n, a_opt, score))
            seen.add(key)
        if len(top_list) >= 5:
            break
    if best_sequence is not None:
        top_list.append((len(best_sequence), best_sequence.copy(), best_score))

    phase2_deadline = deadline - 250.0
    n_candidates = len(top_list)
    t_per_cand = max(30.0, (phase2_deadline - time.monotonic()) / max(1, n_candidates * 2))

    for cand_idx, (n, a0, sc) in enumerate(top_list):
        if time.monotonic() > phase2_deadline - 20:
            break
        t_remaining = phase2_deadline - time.monotonic() - 20
        t_alloc = min(t_per_cand, t_remaining)
        if t_alloc < 15:
            break

        print(f"  Deep {cand_idx}: n={n}, current={sc:.6f}")

        # Deep soft-anneal with more LP corrections
        t_sa = t_alloc * 0.55
        a_opt, obj_opt = soft_anneal_optimize(
            a0, t_sa,
            T_start=None, T_end=None,
            eta_init=0.04, eta_min=1e-8,
            halve_every=200, lp_every=150, n_lp_max=10)

        # Subgrad polish
        t_sg = min(30.0, t_alloc * 0.2)
        if time.monotonic() < phase2_deadline - 30:
            a_opt2, obj2 = subgrad_polish(a_opt, t_sg,
                                          lr_init=0.003, momentum=0.94,
                                          top_k=50, temp_scale=0.02)
            if obj2 < obj_opt:
                a_opt, obj_opt = a_opt2, obj2

        # Another round of soft-anneal with very low T
        if time.monotonic() < phase2_deadline - 25:
            t_final = min(20.0, phase2_deadline - time.monotonic() - 25)
            if t_final > 5:
                conv_tmp = fast_convolve(a_opt)
                T_very_low = conv_tmp.max() * 0.005
                a_opt3, obj3 = soft_anneal_optimize(
                    a_opt, t_final,
                    T_start=T_very_low, T_end=T_very_low * 0.1,
                    eta_init=0.02, eta_min=1e-9,
                    halve_every=300, lp_every=0, n_lp_max=0)
                if obj3 < obj_opt:
                    a_opt, obj_opt = a_opt3, obj2 if obj3 < obj2 else obj_opt

        score = float(evaluate_sequence(a_opt.tolist()))
        print(f"    Result: {score:.6f}")
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()

    print(f"After Phase 2: best={best_score:.6f}")

    # Phase 3: Final polish
    print("=== Phase 3: Final Polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 30.0

        # Coordinate descent
        t_cd = min(100.0, remaining * 0.4)
        if t_cd > 5:
            a_cd = coordinate_polish(best_sequence, time_limit=t_cd)
            score_cd = float(evaluate_sequence(a_cd.tolist()))
            print(f"  CD: {score_cd:.6f}")
            if np.isfinite(score_cd) and score_cd < best_score:
                best_score = score_cd
                best_sequence = a_cd.copy()

        # Final soft-anneal cycles with very low temperature
        remaining = deadline - time.monotonic() - 25.0
        for cyc in range(3):
            if time.monotonic() > deadline - 20:
                break
            t_cyc = min(remaining / (3 - cyc), 50.0)
            if t_cyc < 5:
                break
            conv_cur = fast_convolve(best_sequence)
            T_low = conv_cur.max() * 0.003
            a_opt, obj_opt = soft_anneal_optimize(
                best_sequence, t_cyc,
                T_start=T_low, T_end=T_low * 0.05,
                eta_init=0.01, eta_min=1e-9,
                halve_every=400, lp_every=0, n_lp_max=0)

            # Quick subgrad
            if time.monotonic() < deadline - 15:
                a_opt2, obj2 = subgrad_polish(a_opt, min(8.0, t_cyc * 0.2),
                                              lr_init=0.001, momentum=0.95,
                                              top_k=60, temp_scale=0.01)
                if obj2 < obj_opt:
                    a_opt = a_opt2

            score = float(evaluate_sequence(a_opt.tolist()))
            print(f"  Cycle {cyc}: {score:.6f}")
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()

            # Quick CD
            if time.monotonic() < deadline - 10:
                a_cd2 = coordinate_polish(best_sequence, time_limit=10.0)
                score_cd2 = float(evaluate_sequence(a_cd2.tolist()))
                if np.isfinite(score_cd2) and score_cd2 < best_score:
                    best_score = score_cd2
                    best_sequence = a_cd2.copy()
                    print(f"  CD cycle {cyc}: {score_cd2:.6f}")

        # Random perturbation restarts
        if time.monotonic() < deadline - 15:
            n_b = len(best_sequence)
            for pscale in [0.002, 0.005, 0.01]:
                if time.monotonic() > deadline - 10:
                    break
                for _ in range(3):
                    if time.monotonic() > deadline - 8:
                        break
                    noise = rng.standard_normal(n_b) * pscale
                    a_p = project_simplex(best_sequence + noise)
                    t_rem = min(10.0, deadline - time.monotonic() - 5)
                    if t_rem < 3:
                        break
                    conv_p = fast_convolve(a_p)
                    T_p = conv_p.max() * 0.01
                    a_fp, _ = soft_anneal_optimize(
                        a_p, t_rem,
                        T_start=T_p, T_end=T_p * 0.1,
                        eta_init=0.02, eta_min=1e-9,
                        halve_every=200, lp_every=0, n_lp_max=0)
                    score_fp = float(evaluate_sequence(a_fp.tolist()))
                    if np.isfinite(score_fp) and score_fp < best_score:
                        best_score = score_fp
                        best_sequence = a_fp.copy()
                        print(f"  Perturb({pscale}): {score_fp:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    # Final validation
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(256) / 256
        final_score = float(evaluate_sequence(best_sequence.tolist()))

    total_time = time.monotonic() - time_start
    print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}, time={total_time:.1f}s")
    return [float(x) for x in best_sequence]
