"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact subgradient descent (top-k weighted) on the simplex
2. Smooth max (log-sum-exp) with adaptive annealing
3. Alternating projection (flat spectrum + non-negativity) for initialization
4. Peak shaving and local perturbation for fine-tuning
5. Multi-length, multi-restart search with adaptive time allocation

import numpy as np
import time
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    if size <= 1:
        return a * a.copy()
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def grad_weighted_conv(a, w):
    """
    Compute gradient of sum_k w[k] * conv(a,a)[k] w.r.t. a.
    Formula: g[i] = 2 * sum_m a[m] * w[i+m] (with w zero outside [0, 2n-2]).
    Returns array of length n.
    """
    n = len(a)
    if n == 1:
        return np.array([2.0 * a[0] * w[0]])
    N = 1 << (3 * n - 2).bit_length()
    A = fft(a, N)
    W = fft(w, N)
    c1 = np.real(ifft(np.conj(A) * W, N))[:n]
    return 2.0 * c1


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max(x, beta):
    """Log-sum-exp approximation of max. Returns (smooth_max, softmax_weights)."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -200, 200)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-300:
        w = np.zeros_like(x)
        w[int(np.argmax(x))] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def topk_weights(conv, k):
    """Compute weights that are 1/k on the top-k values and 0 elsewhere."""
    k = min(k, len(conv))
    top_k = np.argpartition(conv, -k)[-k:]
    w = np.zeros_like(conv)
    w[top_k] = 1.0 / k
    return w


def alternating_projection(a0, n_iters=200, c_ratio=1.0):
    """
    Alternating projection: flat magnitude spectrum + non-negativity.
    Returns a non-negative sequence with sum=1.
    """
    n = len(a0)
    N = 1 << (n - 1).bit_length()
    a = a0.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n

    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()

    for it in range(n_iters):
        # Step 1: Force flat magnitude spectrum
        A = fft(a, N)
        mags = np.abs(A)
        # Set all non-DC magnitudes to the mean magnitude
        non_dc_mags = mags[1:]
        if N % 2 == 0:
            non_dc_mags = mags[1:N // 2 + 1]
        c = non_dc_mags.mean() * c_ratio
        if c < 1e-15:
            c = mags[1:].mean()
        phases = A / (mags + 1e-30)
        new_A = np.zeros(N, dtype=complex)
        new_A[0] = 1.0  # DC = 1 means sum = 1
        for k in range(1, N):
            if k < N - k:  # Avoid double-counting conjugate pairs
                new_A[k] = c * phases[k]
        # Handle conjugate symmetry for real signal
        for k in range(1, N // 2):
            new_A[N - k] = np.conj(new_A[k])
        if N % 2 == 0:
            new_A[N // 2] = np.abs(new_A[N // 2])  # Nyquist must be real

        a_new = np.real(ifft(new_A, N))[:n]

        # Step 2: Project onto non-negativity + sum=1
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        else:
            a_new = a.copy()

        # Check improvement
        conv = fast_convolve(a_new)
        obj = 2.0 * n * conv.max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

        a = a_new

    return best_a


def subgradient_descent(a0, max_iters=500, topk_schedule=None, smooth_transition=200,
                        lr_init=0.05, lr_decay=0.999, backtracks=8):
    """
    Hybrid subgradient/smooth-max descent on the simplex.
    - For it < smooth_transition: use exact top-k subgradient
    - For it >= smooth_transition: blend in smooth max with annealed beta
    """
    n = len(a0)
    a = project_simplex(a0.copy())
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    lr = lr_init

    for it in range(max_iters):
        conv = fast_convolve(a)
        obj = 2.0 * n * conv.max()

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        # Compute gradient
        if it < smooth_transition:
            # Exact top-k subgradient
            k = 1 if topk_schedule is None else min(topk_schedule, 2 * n - 1)
            w = topk_weights(conv, k)
            grad = grad_weighted_conv(a, w)
        else:
            # Annealed smooth max
            beta = max(1.0, 50.0 * (1.0 - (it - smooth_transition) / max(1, max_iters - smooth_transition)))
            sm, w = smooth_max(conv, beta)
            grad = grad_weighted_conv(a, w)
            obj = 2.0 * n * sm  # Use smooth objective for convergence check

        # Backtracking line search
        norm_g = np.linalg.norm(grad)
        if norm_g < 1e-15:
            break

        step_accepted = False
        t = lr
        for bt in range(backtracks):
            a_new = project_simplex(a - t * grad)
            conv_new = fast_convolve(a_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj + 1e-12:
                a = a_new
                step_accepted = True
                break
            t *= 0.5

        if not step_accepted:
            # Try smaller fixed step or break
            if t < 1e-10:
                break
            a = project_simplex(a - t * grad)

        lr *= lr_decay

    return best_a, best_obj


def peak_shave(a, n_steps=50, reduction=0.985):
    """Iteratively reduce the peak of the convolution by shrinking contributing elements."""
    n = len(a)
    a = a.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s

    for _ in range(n_steps):
        conv = fast_convolve(a)
        max_k = int(np.argmax(conv))
        # Elements contributing to conv[max_k]: a[i] * a[max_k - i] for valid i
        i_start = max(0, max_k - n + 1)
        i_end = min(n - 1, max_k)
        seg = a[i_start:i_end + 1]
        partners = a[max_k - i_end:max_k - i_start + 1][::-1]
        contrib = seg * partners  # Product terms contributing to the peak
        total = contrib.sum()
        if total < 1e-15:
            break
        # Reduce proportionally to contribution
        excess = conv.max() - conv.mean()
        if excess <= 0:
            break
        reduction_amount = reduction
        a[i_start:i_end + 1] -= seg * (contrib / total) * (1 - reduction)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-15:
            a = a / s
        else:
            break

    return a


def random_perturb(a, scale, rng):
    """Add random noise and project back to simplex."""
    noise = rng.standard_normal(len(a)) * scale
    return project_simplex(a + noise)


def generate_inits(n, parent, rng):

    inits = {}

    edge = max(1, n // 4)

    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Random

    inits['random_sparse'] = rng.dirichlet(np.ones(n) * 0.1)

    # Alternating projection init
    a_ap = alternating_projection(rng.dirichlet(np.ones(n) * 0.5), n_iters=100)
    inits['alt_proj'] = a_ap

    # Flat spectrum init (random phases)
    N = 1 << (n - 1).bit_length()
    best_fs = np.ones(n) / n
    best_fs_obj = 2.0 * n * fast_convolve(best_fs).max()
    for trial in range(5):
        A = np.zeros(N, dtype=complex)
        A[0] = 1.0
        phases = rng.uniform(0, 2 * np.pi, N)
        for k in range(1, N // 2 + 1):
            A[k] = np.exp(1j * phases[k])
        for k in range(1, N // 2):
            A[N - k] = np.conj(A[k])
        if N % 2 == 0:
            A[N // 2] = 1.0 + 0j
        a_fs = np.real(ifft(A, N))[:n]
        a_fs = np.maximum(a_fs, 0.0)
        s = a_fs.sum()
        if s > 1e-10:
            a_fs = a_fs / s
            obj = 2.0 * n * fast_convolve(a_fs).max()
            if obj < best_fs_obj:
                best_fs_obj = obj
                best_fs = a_fs.copy()
    inits['flat_spec'] = best_fs

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0.0)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0.0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps

    return inits

def smooth_max_descent(a0, beta, max_iters, lr=0.04):
    """Simple projected gradient descent using smooth max with fixed beta."""
    n = len(a0)
    a = project_simplex(a0.copy())
    best_a = a.copy()
    best_obj = float('inf')

    for it in range(max_iters):
        conv = fast_convolve(a)
        sm, w = smooth_max(conv, beta)
        obj = 2.0 * n * sm

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        grad = grad_weighted_conv(a, w)
        norm_g = np.linalg.norm(grad)
        if norm_g < 1e-15:
            break

        # Backtracking
        t = lr
        for _ in range(8):
            a_new = project_simplex(a - t * grad)
            conv_new = fast_convolve(a_new)
            sm_new, _ = smooth_max(conv_new, beta)
            obj_new = 2.0 * n * sm_new
            if obj_new <= sm * 2.0 * n + 1e-12:
                a = a_new
                break
            t *= 0.5

        lr *= 0.999

    return best_a

    focus_lengths = focus_lengths[:3]

    for n in focus_lengths:
        elapsed = time.monotonic() - t_start
        remaining = phase2_end - time.monotonic()
        if remaining < 10:
            break
        t_alloc = min(remaining - 5, (deadline - t_start) * 0.2)
        print("  Deep optimizing n=%d (budget=%.0fs)" % (n, t_alloc))

        inits = generate_inits(n, parent, rng)
        # Add current best as init
        if best_sequence is not None:
            if len(best_sequence) == n:
                inits['current_best'] = best_sequence / best_sequence.sum()
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(best_sequence)),
                               np.maximum(best_sequence, 0))
                pp = np.maximum(pp, 0.0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['current_best'] = pp / pps

        a_opt, sc = optimize_length(n, inits, parent, rng, t_alloc)
        if a_opt is not None:
            update_best(a_opt)

    print("Phase 2 best: %.6f" % best_score)

    # === Phase 3: Fine-tuning (remaining time) ===
    print("=== Phase 3: Fine-tuning ===")
    if best_sequence is not None:
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 10.0

        a_norm = best_sequence / best_sequence.sum()

        # 3a: Extended subgradient with very low temperature
        if remaining > 15:
            t_alloc = min(60.0, remaining * 0.3)
            n_iters = int(min(500, t_alloc * 50))
            a_opt, _ = subgradient_descent(a_norm, max_iters=n_iters,
                                           smooth_transition=n_iters // 3,
                                           lr_init=0.02, lr_decay=0.9995)
            update_best(a_opt)
            remaining = deadline - time.monotonic() - 10.0

        # 3b: Smooth max with very low beta (near-exact)
        if remaining > 10:
            for beta in [10, 5, 3, 2]:
                if remaining < 8:
                    break
                t_single = remaining / 4
                n_iters = int(min(300, t_single * 40))
                a_opt = smooth_max_descent(a_norm, beta, n_iters, lr=0.02)
                update_best(a_opt)
                a_norm = best_sequence / best_sequence.sum()
                remaining = deadline - time.monotonic() - 10.0

        # 3c: Peak shaving
        if remaining > 10:
            a_shaved = peak_shave(a_norm, n_steps=100, reduction=0.992)
            update_best(a_shaved)
            # Re-optimize from shaved
            if remaining > 10:
                n_iters = int(min(200, remaining * 20))
                a_opt, _ = subgradient_descent(a_shaved, max_iters=n_iters,
                                               smooth_transition=n_iters // 2,
                                               lr_init=0.02)
                update_best(a_opt)
            remaining = deadline - time.monotonic() - 10.0

        # 3d: Multiple small perturbations
        if remaining > 10:
            n_pert = int(min(20, remaining / 3))
            for i in range(n_pert):
                if deadline - time.monotonic() < 5:
                    break
                scale = 0.002 + 0.003 * rng.uniform()
                a_pert = random_perturb(a_norm, scale, rng)
                # Quick optimization
                n_iters = int(min(80, max(20, remaining / 4)))
                a_opt, _ = subgradient_descent(a_pert, max_iters=n_iters,
                                               smooth_transition=n_iters // 2,
                                               lr_init=0.03)
                update_best(a_opt)
                a_norm = best_sequence / best_sequence.sum()

        # 3e: Alternating projection polish
        if remaining > 10:
            a_ap = alternating_projection(a_norm, n_iters=200, c_ratio=1.0)
            update_best(a_ap)
            if remaining > 10:
                n_iters = int(min(200, remaining * 15))
                a_opt, _ = subgradient_descent(a_ap, max_iters=n_iters,
                                               smooth_transition=n_iters // 2,
                                               lr_init=0.02)
                update_best(a_opt)

    # Final: try one more different length if time permits
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 30:
            alt_n = len(best_sequence) + 64
            if alt_n <= 768:
                print("  Trying alternate length n=%d" % alt_n)
                inits = generate_inits(alt_n, parent, rng)
                if best_sequence is not None:
                    pp = np.interp(np.linspace(0, 1, alt_n),
                                   np.linspace(0, 1, len(best_sequence)),
                                   np.maximum(best_sequence, 0))
                    pp = np.maximum(pp, 0.0)
                    pps = pp.sum()
                    if pps > 1e-10:
                        inits['from_best'] = pp / pps
                a0 = inits.get('from_best', inits.get('alt_proj', inits['uniform']))
                n_iters = int(min(300, remaining * 30))
                a_opt, _ = subgradient_descent(a0, max_iters=n_iters,
                                               smooth_transition=n_iters // 2,
                                               lr_init=0.05)
                update_best(a_opt)

    # Ensure we have a valid sequence
    if best_sequence is None:
        best_sequence = np.ones(128) / 128
        best_score = score_sequence(best_sequence)

    # Final verification
    final_score = score_sequence(best_sequence)
    if np.isfinite(final_score):
        best_score = final_score

    print("\nFINAL: score=%.6f (n=%d, items=%d)" % (best_score, len(best_sequence), len(best_sequence)))
    return [float(x) for x in best_sequence]
