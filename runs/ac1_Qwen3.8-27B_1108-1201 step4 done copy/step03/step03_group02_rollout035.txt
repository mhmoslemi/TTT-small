import time
from scipy.optimize import minimize

def fast_conv(a):
    """Fast auto-convolution via FFT."""
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fast_conv(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_obj_and_grad(a, t):
    """Compute 2n*logsumexp(t*conv(a,a)) and its gradient, assuming sum(a)=1."""
    n = len(a)
    conv = fast_conv(a)
    scaled = t * conv
    m = scaled.max()
    exp_vals = np.exp(scaled - m)
    sum_exp = exp_vals.sum()
    p = exp_vals / sum_exp
    obj = 2.0 * n * (m/t + np.log(sum_exp)/t)
    
    # Gradient: 2t * cross_corr(p, a)
    L = 1 << (3*n - 1).bit_length()
    P = np.fft.rfft(p, L)
    A = np.fft.rfft(a, L)
    cross = np.fft.irfft(np.conj(P) * A, L)[:n]
    grad = 2.0 * t * cross
    return obj, grad

            o, g = smooth_obj_and_grad(x, t)

def optimize_projected_subgrad(a0, time_limit, lr_init=0.05, momentum=0.9, 
                                top_k=50, temp=0.001, t_smooth=None):

    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s

    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if t_smooth is not None:
            # Use smooth gradient
            _, sg = smooth_obj_and_grad(a, t_smooth)
        else:
            # Use top-k peak subgradient

Actually, the subgradient at a single peak k is:

So sg is just a reversed and shifted version of a. For multiple peaks with weights w:

This is a weighted sum of shifted/reversed copies of a. We can compute this via convolution:

Actually: sg[i] = 2 * sum_j w_j * a[k_j - i]

Let q[m] = sum_j w_j * delta(m - k_j) (a discrete measure on the conv indices).
Then sg[i] = 2 * sum_m q[m] * a[m-i] = 2 * cross_corr(q, a)[i].

And cross_corr via FFT is O(n log n).

def fast_conv(a):
    """Full auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fast_cross_corr(p, a):
    """Cross-correlation: result[i] = sum_m p[m]*a[m-i] (valid range).
    p has length 2n-1, a has length n. Returns array of length n."""
    n = len(a)
    L = 1 << (3 * n - 1).bit_length()
    P = np.fft.rfft(p, L)
    A = np.fft.rfft(a, L)
    return np.fft.irfft(np.conj(P) * A, L)[:n]

def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def normalize(a):
    """Normalize to probability simplex."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

    """Smoothed objective 2n*logsumexp(t*conv(a,a)) and gradient.
    Assumes sum(a)=1."""
    n = len(a)
    conv = fast_conv(a)
    scaled = t * conv
    m = scaled.max()
    exp_vals = np.exp(scaled - m)
    sum_exp = exp_vals.sum()
    p = exp_vals / sum_exp
    obj = 2.0 * n * (m / t + np.log(sum_exp) / t)
    grad = 2.0 * t * fast_cross_corr(p, a)
    return obj, grad

    """Compute subgradient as weighted combination of top-k peak subgradients.
    Uses FFT for efficiency."""
    n = len(a)
    if top_k >= len(conv):
        p = np.ones(len(conv)) / len(conv)
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        p = np.zeros(len(conv))
        pv = conv[peaks]
        pv_scaled = pv / (temperature + 1e-15)
        pv_scaled -= pv_scaled.max()
        w = np.exp(pv_scaled)
        w /= w.sum()
        p[peaks] = w
    grad = 2.0 * fast_cross_corr(p, a)
    return grad

    best_a = a.copy()

        else:
            progress = (elapsed - warmup_time) / (time_limit - warmup_time + 1e-9)
            t_cur = t_final * (1.0 + 2.0 * progress)
        
        _, grad = smooth_obj_grad(a, t_cur)
        v = momentum * v - lr * grad
        a = project_simplex(a + v)
        
        conv = fast_conv(a)
        cur_obj = 2.0 * n * conv.max()
        total_iters += 1

            best_a = a.copy()
            no_improve = 0
        else:

import numpy as np
import time
from scipy.optimize import linprog


def fast_conv(a):
    """Full auto-convolution via FFT. O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_corr(p, a):
    """Cross-correlation result[i] = sum_m p[m]*a[m-i]. p len 2n-1, a len n. Returns len n."""
    n = len(a)
    L = 1 << (3 * n - 1).bit_length()
    P = np.fft.rfft(p, L)
    A = np.fft.rfft(a, L)
    return np.fft.irfft(np.conj(P) * A, L)[:n]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def normalize(a):
    """Project to probability simplex (non-negative, sum=1)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def smooth_obj_grad(a, t):
    """Compute 2n*logsumexp(t*conv(a,a)) and gradient. Assumes sum(a)=1."""
    n = len(a)
    conv = fast_conv(a)
    scaled = t * conv
    m = scaled.max()
    exp_vals = np.exp(scaled - m)
    sum_exp = exp_vals.sum()
    p = exp_vals / sum_exp
    obj = 2.0 * n * (m / t + np.log(sum_exp) / t)
    grad = 2.0 * t * fast_cross_corr(p, a)
    return obj, grad


def subgrad_topk(a, conv, top_k, temperature):
    """FFT-based subgradient from top-k weighted peaks."""
    n = len(a)
    num_conv = len(conv)
    if top_k >= num_conv:
        p = np.ones(num_conv) / num_conv
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        p = np.zeros(num_conv)
        pv = conv[peaks]
        pv_s = pv / (temperature + 1e-15)
        pv_s -= pv_s.max()
        w = np.exp(pv_s)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= w_sum
        p[peaks] = w
    return 2.0 * fast_cross_corr(p, a)

def optimize_lp_iter(a0, time_limit, step=0.35, max_iters=200):
    """LP best-response iteration: find b maximizing sum(b) s.t. conv(a,b)<=M, b>=0."""
    a = normalize(a0.copy())
    n = len(a)
    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()

    # Pre-build the structure (which (i,j) pairs are valid)
    valid_pairs = []
    for i in range(2 * n - 1):
        j_lo = max(0, i - n + 1)
        j_hi = min(n - 1, i)
        for j in range(j_lo, j_hi + 1):
            valid_pairs.append((i, j, i - j))

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        M = conv.max()
        A_mat = np.zeros((2 * n - 1, n))
        for (i, j, val_idx) in valid_pairs:
            A_mat[i, j] = a[val_idx]

        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs')

        if res.status != 0:
            break

        b = np.maximum(res.x, 0.0)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum

        a_new = (1.0 - step) * a + step * b
        a_new = project_simplex(a_new)
        conv = fast_conv(a_new)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a_new.copy()

        a = a_new

    return best_a, best_obj


def gen_init_uniform(n):

def gen_init_cosine(n, freq, alpha):
    x = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * freq * x / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def gen_init_gaussian(n, sigma_frac):
    x = np.arange(n, dtype=np.float64)
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((x - 0.5 * n) / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def gen_init_sparse(n, k, rng):
    positions = np.sort(rng.choice(n, size=min(k, n), replace=False))
    a = np.zeros(n)
    a[positions] = 1.0
    return a / a.sum()


def gen_init_bimodal(n, gap, rng):
    """Two Gaussian bumps separated by gap fraction of n."""
    x = np.arange(n, dtype=np.float64)
    center1 = n * (0.5 - gap / 2)
    center2 = n * (0.5 + gap / 2)
    sigma = n * 0.1
    a = np.exp(-0.5 * ((x - center1) / sigma) ** 2) + np.exp(-0.5 * ((x - center2) / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def get_initializations(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = {}
    inits['uniform'] = gen_init_uniform(n)
    inits['cos1_05'] = gen_init_cosine(n, 1, 0.5)
    inits['cos2_04'] = gen_init_cosine(n, 2, 0.4)
    inits['cos1_08'] = gen_init_cosine(n, 1, 0.8)
    inits['cos3_03'] = gen_init_cosine(n, 3, 0.3)
    inits['gauss_w'] = gen_init_gaussian(n, 0.4)
    inits['gauss_n'] = gen_init_gaussian(n, 0.2)
    inits['bimodal'] = gen_init_bimodal(n, 0.4, rng)
    inits['sparse16'] = gen_init_sparse(n, 16, rng)
    inits['sparse32'] = gen_init_sparse(n, 32, rng)
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0.0)
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), np.maximum(parent, 0.0))
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search combining smooth annealed gradient,
    direct subgradient, LP refinement, and structured exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: len={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass

    def try_update(a):
        nonlocal best_sequence, best_score
        if a is None or len(a) == 0:
            return best_score
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return best_score
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  New best: {score:.6f} (n={len(a)})")
        return score

    # === Phase 1: Broad exploration (first ~400s) ===
    print("=== Phase 1: Broad Exploration ===")
    t_p1 = time.monotonic()

    lengths = [512, 256]
    if parent is not None and len(parent) not in lengths and 32 <= len(parent) <= 2048:
        lengths.insert(0, len(parent))
    lengths.extend([1024, 2048])

    candidates = []

    for n in lengths:
        remain = deadline - time.monotonic() - 600
        if remain < 60:
            break

        inits = get_initializations(n, parent, rng)
        # Prioritize initializations
        priority = ['parent', 'uniform', 'cos1_05', 'gauss_w', 'sparse32',
                    'cos2_04', 'bimodal', 'random', 'gauss_n', 'cos1_08']
        names = [nm for nm in priority if nm in inits]
        names += [nm for nm in inits if nm not in priority]
        names = names[:6]

        time_per_init = max(15.0, remain / (len(lengths) * len(names) * 1.5))

        for name in names:
            remain = deadline - time.monotonic() - 400
            if remain < 10:
                break
            alloc = min(time_per_init, remain)
            if alloc < 8:
                break

            a0 = inits[name]
            print(f"  n={n}, init={name}", end="")

            a_opt, obj = optimize_smooth_anneal(
                a0, time_limit=alloc, t_init=5.0, t_end=2000.0,
                lr_init=0.015, momentum=0.94)
            print(f" obj={obj:.6f}", end="")
            candidates.append((obj, a_opt))
            try_update(a_opt)

    print(f"Phase 1 done: best={best_score:.6f}, elapsed={time.monotonic()-t_p1:.0f}s")

    # Sort candidates by objective
    candidates.sort(key=lambda x: x[0])

    # === Phase 2: Deep optimization of top candidates (next ~600s) ===
    print("=== Phase 2: Deep Optimization ===")
    t_p2 = time.monotonic()

    top_candidates = candidates[:5] if len(candidates) >= 5 else candidates
    if best_sequence is not None:
        top_candidates.append((best_score, best_sequence.copy()))

    phase2_deadline = deadline - 500
    for idx, (obj_init, a_cand) in enumerate(top_candidates[:4]):
        remain = phase2_deadline - time.monotonic()
        if remain < 30:
            break
        alloc = min(120.0, remain / (len(top_candidates[:4]) - idx))
        if alloc < 15:
            break

        n = len(a_cand)
        print(f"  Deep opt candidate {idx} (n={n}, obj={obj_init:.6f}):")

        # Run smooth anneal with different params
        a_opt, obj = optimize_smooth_anneal(
            a_cand, time_limit=alloc * 0.5, t_init=3.0, t_end=5000.0,
            lr_init=0.01, momentum=0.95)
        print(f"    smooth: {obj:.6f}")
        try_update(a_opt)

        remain = phase2_deadline - time.monotonic()
        if remain > 15:
            a_opt2, obj2 = optimize_subgrad_direct(
                a_opt, time_limit=min(alloc * 0.5, remain),
                lr_init=0.03, momentum=0.92, top_k=max(20, n // 20),
                temp_init=0.003)
            print(f"    subgrad: {obj2:.6f}")
            try_update(a_opt2)

    print(f"Phase 2 done: best={best_score:.6f}, elapsed={time.monotonic()-t_p2:.0f}s")

    # === Phase 3: LP refinement (next ~300s) ===
    print("=== Phase 3: LP Refinement ===")
    t_p3 = time.monotonic()

    if best_sequence is not None:
        n_best = len(best_sequence)
        phase3_deadline = deadline - 200

        if n_best <= 1024:
            for step in [0.4, 0.25, 0.6]:
                remain = phase3_deadline - time.monotonic()
                if remain < 20:
                    break
                alloc = min(80.0, remain / 3)
                if alloc < 10:
                    break
                a_opt, obj = optimize_lp_iter(
                    best_sequence, time_limit=alloc, step=step)
                print(f"  LP step={step}: {obj:.6f}")
                try_update(a_opt)

    print(f"Phase 3 done: best={best_score:.6f}, elapsed={time.monotonic()-t_p3:.0f}s")

    # === Phase 4: Fine-tuning with subgradient (next ~200s) ===
    print("=== Phase 4: Fine-tuning ===")
    t_p4 = time.monotonic()

    if best_sequence is not None:
        n_best = len(best_sequence)
        phase4_deadline = deadline - 60

        for (lr, top_k, temp) in [(0.02, 30, 0.002), (0.01, 50, 0.005),
                                   (0.04, 20, 0.001), (0.015, 40, 0.003)]:
            remain = phase4_deadline - time.monotonic()
            if remain < 10:
                break
            alloc = min(50.0, remain / 3)
            if alloc < 8:
                break
            a_opt, obj = optimize_subgrad_direct(
                best_sequence, time_limit=alloc, lr_init=lr,
                momentum=0.93, top_k=top_k, temp_init=temp)
            try_update(a_opt)

        # Also try smooth with very high temperature
        remain = phase4_deadline - time.monotonic()
        if remain > 30:
            a_opt, obj = optimize_smooth_anneal(
                best_sequence, time_limit=min(60.0, remain),
                t_init=500.0, t_end=5000.0, lr_init=0.005, momentum=0.96)
            try_update(a_opt)

    print(f"Phase 4 done: best={best_score:.6f}, elapsed={time.monotonic()-t_p4:.0f}s")

    # === Phase 5: Perturbation + quick re-optimization ===
    print("=== Phase 5: Perturbations ===")
    t_p5 = time.monotonic()

    if best_sequence is not None:
        n_best = len(best_sequence)
        phase5_deadline = deadline - 20

        for scale in [0.003, 0.01, 0.005, 0.015, 0.002]:
            remain = phase5_deadline - time.monotonic()
            if remain < 15:
                break
            rng_sub = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
            noise = rng_sub.standard_normal(n_best) * scale
            a_pert = project_simplex(best_sequence + noise)
            alloc = min(25.0, remain - 5)
            if alloc < 5:
                break
            a_opt, obj = optimize_smooth_anneal(
                a_pert, time_limit=alloc, t_init=50.0, t_end=3000.0,
                lr_init=0.008, momentum=0.94)
            try_update(a_opt)

    print(f"Phase 5 done: best={best_score:.6f}, elapsed={time.monotonic()-t_p5:.0f}s")

    # === Final: try larger n from scratch if time permits ===
    print("=== Final: Large-n attempt ===")
    t_fin = time.monotonic()
    remain_final = deadline - time.monotonic()
    if remain_final > 60 and best_sequence is not None:
        for n_large in [2048, 4096]:
            if remain_final - (time.monotonic() - t_fin) < 40:
                break
            a0 = normalize(best_sequence) if len(best_sequence) <= n_large else \
                np.interp(np.linspace(0, 1, n_large),
                          np.linspace(0, 1, len(best_sequence)), best_sequence)
            a0 = normalize(a0)
            alloc = min(50.0, remain_final - (time.monotonic() - t_fin) - 10)
            if alloc < 20:
                break
            a_opt, obj = optimize_smooth_anneal(
                a0, time_limit=alloc, t_init=5.0, t_end=2000.0,
                lr_init=0.01, momentum=0.94)
            print(f"  n={n_large}: {obj:.6f}")
            try_update(a_opt)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    # Final evaluation
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
