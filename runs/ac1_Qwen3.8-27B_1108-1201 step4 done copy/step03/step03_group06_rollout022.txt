def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, conv=None):
    """Compute subgradient at argmax of conv(a,a)."""
    if conv is None:
        conv = fast_convolve(a)
    n = len(a)
    k_star = np.argmax(conv)
    # Subgradient: gradient of (a*a)_{k*} w.r.t. a
    # (a*a)_{k*} = sum_{i+j=k*} a_i * a_j
    # d/da_i = sum_{j: i+j=k*} a_j + sum_{j: j+i=k*} a_j (with i=j term counted once)
    # = a[k*-i] if valid + a[i-k*] if valid, with correction for double counting
    grad = np.zeros(n)
    for i in range(n):
        j1 = k_star - i
        j2 = i - k_star
        val = 0.0
        if 0 <= j1 < n:
            val += a[j1]
        if 0 <= j2 < n and j2 != j1:
            val += a[j2]
        # If j1 == j2 == i (i.e., 2i = k*), we already counted it once
        # Actually: d/da_i of sum_{p+q=k*} a_p a_q = sum_{q: i+q=k*} a_q + sum_{p: p+i=k*} a_p - a_i (if 2i=k*)
        # = a[k*-i] (if valid) + a[i-k*] (if valid) - a_i (if 2i=k*)
        if 2 * i == k_star:
            val -= a[i]
        grad[i] = val
    return grad, k_star, conv


def lp_solve_active(a, M, active_indices):
    """Solve LP: max sum(b) s.t. b>=0, (a*b)[k] <= M for k in active_indices."""
    n = len(a)
    num_constraints = len(active_indices)
    
    A_ub = np.zeros((num_constraints, n))
    for idx, k in enumerate(active_indices):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for j in range(i_min, i_max + 1):
            A_ub[idx, j] = a[k - j]
    
    c = -np.ones(n)
    b_ub = np.full(num_constraints, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def mirror_descent_optimize(a0, time_limit, lr_init=0.1, seed=0, verbose=False):
    """
    Mirror descent (multiplicative weights) with subgradient.
    Update: a_i = a_i * exp(-lr * g_i) / Z where g_i is subgradient.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    avg_a = a.copy()
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        if it > 0 and it % 50 == 0:
            conv_avg = fast_convolve(avg_a)
            obj_avg = 2.0 * n * conv_avg.max()
            if obj_avg < best_obj:
                best_obj = obj_avg
                best_a = avg_a.copy()
        
        conv = fast_convolve(a)
        M = conv.max()
        grad, k_star, _ = compute_subgradient(a, conv)
        
        # Normalize gradient
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        grad = grad / g_norm
        
        # Mirror descent update (multiplicative weights)
        # a_i_new = a_i * exp(-lr * grad_i) / Z
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - lr * grad
        # Numerical stability
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / a_new_sum
        
        # Check improvement
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                lr *= 0.9
                if no_improve > 500:
                    # Restart with perturbation
                    noise = rng.standard_normal(n) * 0.005
                    a = project_simplex(a + noise)
                    avg_a = a.copy()
                    lr = lr_init * 0.5
                    no_improve = 0
                    if lr < 0.001:
                        break
        
        a = a_new
        # Averaging
        decay = 1.0 / (it + 1)
        avg_a = (1.0 - decay) * avg_a + decay * a_new
    
    # Final check of average
    conv_avg = fast_convolve(avg_a)
    obj_avg = 2.0 * n * conv_avg.max()
    if obj_avg < best_obj:
        best_obj = obj_avg
        best_a = avg_a.copy()
    
    return best_a, best_obj


def coordinate_descent_polish(a0, time_limit, seed=0, verbose=False):
    """
    Coordinate descent: try perturbing pairs of coordinates to improve objective.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        # Try random pairs
        for _ in range(500):
            if time.monotonic() - t0 > time_limit:
                break
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            if a[i] < 1e-10 and a[j] < 1e-10:
                continue
            
            # Try transferring mass from j to i and vice versa
            for delta_scale in [0.1, 0.25, 0.5, 0.75, 0.9]:
                # Move from j to i
                delta = delta_scale * a[j]
                if delta > a[j]:
                    continue
                a_try = best_a.copy()
                a_try[j] -= delta
                a_try[i] += delta
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    improved = True
                    break
            
            if time.monotonic() - t0 > time_limit:
                break
        
        if not improved:
            # Try broader perturbations
            for scale in [0.005, 0.01, 0.02, 0.05]:
                noise = rng.standard_normal(n) * scale
                a_try = project_simplex(best_a + noise)
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    improved = True
                    break
        
        if not improved and scale > 0.05:
            break
    
    return best_a, best_obj


def generate_initializations(n, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(('uniform', np.ones(n) / n))
    
    # Bimodal (edges)
    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits.append(('bimodal', bi / bi.sum()))
    
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n-1)) / (0.25 * n)) ** 2)
    inits.append(('gauss_c', g / g.sum()))
    
    # Gaussian wide
    g2 = np.exp(-0.5 * ((x - 0.5 * (n-1)) / (0.5 * n)) ** 2)
    inits.append(('gauss_w', g2 / g2.sum()))
    
    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits.append(('ramp', ramp / ramp.sum()))
    
    # Cosine
    cos_w = np.cos(np.pi * (x + 0.5) / n) ** 2
    cos_w = np.maximum(cos_w, 0.001)
    inits.append(('cosine', cos_w / cos_w.sum()))
    
    # Random
    inits.append(('rand1', rng.dirichlet(np.ones(n) * 0.5)))
    inits.append(('rand2', rng.dirichlet(np.ones(n) * 2.0)))
    
    # Sparse random
    sp = np.zeros(n)
    n_nonzero = max(5, n // 5)
    idx = rng.choice(n, size=n_nonzero, replace=False)
    sp[idx] = 1.0
    inits.append(('sparse', sp / sp.sum()))
    
    return inits


def interp_to_length(a, new_n):
    """Interpolate sequence to new length, preserving non-negativity and normalization."""
    if len(a) == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp = np.maximum(a_interp, 0)
    s = a_interp.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / s

        starts = []
        if parent is not None:
            p_interp = interp_to_length(parent, n)
            starts.append(('parent', p_interp))
        
        inits = generate_initializations(n, rng)
        for name, init in inits[:4]:  # Limit initializations
            starts.append((name, init))
        
        time_per_start = max(8.0, time_left / (len(starts) + len(lengths)))
        
        for sname, s_arr in starts:
            time_left = phase1_start + phase1_time - time.monotonic()
            if time_left < 10:
                break
            t_alloc = min(time_per_start, time_left - 10)
            if t_alloc < 5:
                break
            
            # Quick LP optimization
            a_opt, obj_opt = lp_optimize_v2(s_arr, time_limit=t_alloc,
                                             top_k=max(20, n // 5), seed=n + hash(sname) % 1000)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                candidates.append((score, n, sname, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Explored: n={n}, {sname}, score={score:.6f}")
    
    print(f"Phase 1 done. Best: {best_score:.6f}")
    
    # ========== Phase 2: Deep LP Optimization ==========
    print("=== Phase 2: Deep LP Optimization ===")
    phase2_start = time.monotonic()
    phase2_time = 600.0
    
    # Sort candidates by score
    candidates.sort(key=lambda x: x[0])
    
    # Get top starting points
    deep_starts = []
    seen = set()
    for score, n, name, a_opt in candidates[:10]:
        key = (n, name)
        if key not in seen:
            deep_starts.append((n, a_opt.copy()))
            seen.add(key)
        if len(deep_starts) >= 5:
            break
    
    # Ensure parent's length is represented
    if best_sequence is not None:
        best_n = len(best_sequence)
        found = False
        for n_s, _ in deep_starts:
            if n_s == best_n:
                found = True
                break
        if not found:
            deep_starts.append((best_n, best_sequence.copy()))
    
    # Run deep LP with different configurations
    configs = [
        {'top_k': None, 'max_iters': 10000},
        {'top_k': 40, 'max_iters': 10000},
        {'top_k': 60, 'max_iters': 10000},
        {'top_k': 80, 'max_iters': 10000},
        {'top_k': 120, 'max_iters': 10000},
    ]
    
    total_runs = len(deep_starts) * len(configs)
    time_per_run = max(15.0, phase2_time / total_runs)
    
    run_i = 0
    for n_s, a_s in deep_starts:
        for cfg in configs:
            time_left = phase2_start + phase2_time - time.monotonic()
            if time_left < 15:
                break
            t_alloc = min(time_per_run, time_left - 15)
            if t_alloc < 10:
                break
            
            # Scale top_k for this n
            cfg_local = dict(cfg)
            if cfg_local['top_k'] is not None:
                cfg_local['top_k'] = max(20, int(cfg_local['top_k'] * n_s / 320))
            
            a_opt, obj_opt = lp_optimize_v2(a_s, time_limit=t_alloc, seed=run_i * 7 + 1, **cfg_local)
            score = float(evaluate_sequence(a_opt.tolist()))
            run_i += 1
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Deep LP [{run_i}]: n={n_s}, top_k={cfg_local['top_k']}, score={score:.6f}")
    
    print(f"Phase 2 done. Best: {best_score:.6f}")
    
    # ========== Phase 3: Mirror Descent ==========
    print("=== Phase 3: Mirror Descent ===")
    phase3_start = time.monotonic()
    phase3_time = 350.0
    
    if best_sequence is not None:
        n_best = len(best_sequence)
        md_configs = [
            {'lr_init': 0.05},
            {'lr_init': 0.1},
            {'lr_init': 0.15},
            {'lr_init': 0.03},
        ]
        
        time_per_md = max(20.0, phase3_time / len(md_configs))
        
        for i, cfg in enumerate(md_configs):
            time_left = phase3_start + phase3_time - time.monotonic()
            if time_left < 20:
                break
            t_alloc = min(time_per_md, time_left - 20)
            if t_alloc < 15:
                break
            
            a_opt, obj_opt = mirror_descent_optimize(best_sequence, time_limit=t_alloc,
                                                      seed=i * 31 + 1, **cfg)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Mirror [{i}] lr={cfg['lr_init']}: score={score:.6f}")
        
        # Mirror descent from perturbed starts
        for i in range(3):
            time_left = phase3_start + phase3_time - time.monotonic()
            if time_left < 15:
                break
            noise = rng.standard_normal(n_best) * 0.01
            a_pert = project_simplex(best_sequence + noise)
            t_alloc = min(30.0, time_left - 15)
            if t_alloc < 10:
                break
            
            a_opt, obj_opt = mirror_descent_optimize(a_pert, time_limit=t_alloc,
                                                      lr_init=0.08, seed=100 + i)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Mirror pert [{i}]: score={score:.6f}")
    
    print(f"Phase 3 done. Best: {best_score:.6f}")
    
    # ========== Phase 4: Coordinate Descent Polish ==========
    print("=== Phase 4: Coordinate Polish ===")
    phase4_start = time.monotonic()
    phase4_time = 200.0
    
    if best_sequence is not None:
        time_left = phase4_start + phase4_time - time.monotonic()
        if time_left > 30:
            a_opt, obj_opt = coordinate_descent_polish(best_sequence, time_limit=min(time_left - 20, 150.0),
                                                        seed=555)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Coord polish: score={score:.6f}")
        
        # More LP after coordinate polish
        time_left = phase4_start + phase4_time - time.monotonic()
        if time_left > 30:
            a_opt, obj_opt = lp_optimize_v2(best_sequence, time_limit=min(time_left - 20, 100.0),
                                             top_k=max(30, len(best_sequence) // 5), seed=888)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Post-coord LP: score={score:.6f}")
    
    print(f"Phase 4 done. Best: {best_score:.6f}")
    
    # ========== Phase 5: Final Polish ==========
    print("=== Phase 5: Final Polish ===")
    phase5_start = time.monotonic()
    remaining = total_deadline - time.monotonic()
    
    if best_sequence is not None and remaining > 30:
        n_final = len(best_sequence)
        
        # Tiny perturbations
        n_tries = min(30, int(remaining / 5))
        for i in range(n_tries):
            if time.monotonic() > total_deadline - 25:
                break
            for scale in [0.0005, 0.001, 0.002, 0.005]:
                noise = rng.standard_normal(n_final) * scale
                a_pert = project_simplex(best_sequence + noise)
                score_pert = float(evaluate_sequence(a_pert.tolist()))
                if np.isfinite(score_pert) and score_pert < best_score:
                    best_score = score_pert
                    best_sequence = a_pert.copy()
                    print(f"  Final perturb [{i}] s={scale}: score={score_pert:.6f}")
                    break
        
        # Final short LP
        time_left = total_deadline - time.monotonic() - 20
        if time_left > 15:
            a_opt, obj_opt = lp_optimize_v2(best_sequence, time_limit=time_left,
                                             top_k=max(50, n_final // 4), seed=9999)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final LP: score={score:.6f}")
    
    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(128) / 128
    
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})")
    return [float(x) for x in best_sequence]
