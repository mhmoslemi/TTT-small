def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        val = max(v[0], 0.0)
        return np.array([val]) if val > 0 else np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def lp_maximize_sum(a, M):
    """Solve: max sum(b) s.t. b >= 0, conv(a,b)[k] <= M for all k."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None

def optimize_smoothed_gradient(a0, time_limit, t_init=100.0, t_decay=0.999,

    """

    if S < 1e-15:

    else:

    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    lr = lr_init
    no_improve = 0

        # Compute gradient
        _, grad, obj_true = smoothed_objective_and_grad(a, t_cur)
        
        # Gradient step with momentum
        v = momentum * v - lr * grad
        a_new = a + v
        
        # Project onto simplex
        a_new = project_simplex(a_new)

        conv = fast_convolve(a_new)

            length = rng.integers(5, min(30, n // 4))
            noise = np.zeros(n)
            noise[start:start+length] = rng.standard_normal(length) * noise_scale
            noise = project_simplex(a + noise) - a
        
        a_new = project_simplex(a + noise)
        conv_new = fast_convolve(a_new)
        new_obj = 2.0 * n * conv_new.max()
        
        # Metropolis acceptance
        delta = new_obj - cur_obj
        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = a_new
            cur_obj = new_obj
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            T = max(1e-6, T * T_decay)
        else:
            no_improve += 1
            T = max(1e-6, T * T_decay)
        
        if no_improve > 5000:
            # Restart with larger temperature
            T = T_init * 0.5
            no_improve = 0
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()
    
    return best_a, best_obj


def optimize_lbfgs_smoothed(a0, time_limit, t_values=None):
    """
    Use scipy L-BFGS-B to minimize the smoothed objective.
    Tries multiple temperature values.
    """
    a = np.maximum(a0, 0.0).copy()
    S = a.sum()
    if S < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / S
    n = len(a)
    
    if t_values is None:
        t_values = [50.0, 30.0, 20.0, 15.0]
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for t_val in t_values:
        if time.monotonic() - t0 > time_limit:
            break
        
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        t_alloc = remaining / len([x for x in t_values if x >= t_val])
        
        def fun(x):
            x_pos = np.maximum(x, 0)
            s = x_pos.sum()
            if s < 1e-15:
                return 1e10
            x_norm = x_pos / s
            conv = fast_convolve(x_norm)
            t_c = t_val * conv
            mv = t_c.max()
            lse = mv + np.log(np.exp(t_c - mv).sum())
            return 2.0 * n * lse / t_val
        
        def grad(x):
            x_pos = np.maximum(x, 0)
            s = x_pos.sum()
            if s < 1e-15:
                return np.zeros(len(x))
            x_norm = x_pos / s
            conv = fast_convolve(x_norm)
            t_c = t_val * conv
            mv = t_c.max()
            p = np.exp(t_c - mv)
            p /= p.sum()
            
            fft_size = 1 << (3 * len(x) - 2).bit_length()
            P = np.fft.rfft(p, fft_size)
            A = np.fft.rfft(x_norm, fft_size)
            corr = np.fft.irfft(np.conj(A) * P, fft_size)
            grad_L = corr[:len(x)]
            
            L_t = (mv + np.log(p.sum())) / t_val
            # dFun/dx where Fun = 2n * L_t(x/sum(x)) ... 
            # Actually, let's just use the subgradient approach:
            # Since we're on the simplex, the effective gradient is:
            grad_result = 2.0 * n * (grad_L - L_t * np.ones(len(x))) 
            # This is the gradient projected to the tangent space of the simplex
            # (we subtract the mean to make it orthogonal to the all-ones vector)
            return grad_result
        
        bounds = [(0, None)] * n
        res = minimize(fun, a, jac=grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': 2000, 'maxfun': 5000,
                                               'ftol': 1e-12, 'gtol': 1e-8})
        
        if res.success or res.fun < fun(a):
            a_new = np.maximum(res.x, 0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
                conv_new = fast_convolve(a_new)
                obj_new = 2.0 * n * conv_new.max()
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                a = a_new
    
    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps
    
    for freq in [1, 2, 3, 5]:
        a = 1.0 + 0.9 * np.cos(2.0 * np.pi * freq * x / n)
        a = np.maximum(a, 0)
        a /= a.sum()
        inits[f'cos_{freq}'] = a
    
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g
    
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Asymmetric patterns
    a = np.ones(n)
    a[:n//2] = 1.0 + rng.uniform(0, 0.5, n//2)
    a[n//2:] = 1.0 + rng.uniform(-0.5, 0, n//2)
    a = np.maximum(a, 0)
    a /= a.sum()
    inits['asym'] = a
    
    # Linear ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp
    
    # Bimodal
    g1 = np.exp(-0.5 * ((x - 0.3 * n) / (0.1 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.7 * n) / (0.1 * n)) ** 2)
    b = g1 + g2
    b /= b.sum()
    inits['bimodal'] = b
    
    return inits
