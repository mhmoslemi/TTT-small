"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based peak reduction,
smooth max L-BFGS polishing, and multi-start over diverse initializations and lengths.
"""

import numpy as np
import time
import scipy
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import scipy.sparse as sp
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max_and_weights(x, beta):
    """Compute smooth max and softmax weights."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def compute_grad_smooth_conv(a, w):
    """Compute gradient of sum_k w[k]*conv[k] w.r.t. a."""
    n = len(a)
    N = 1 << (len(w) + n - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0
    return grad


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_constraint_matrix(f, n):
    """Build sparse matrix C where C[k,j] = f[k-j] for valid (k,j)."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    C = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return C


def solve_lp_direction(f, M, max_time=30):
    """Solve the LP to find the best direction b."""
    n = len(f)
    b = cp.Variable(n, nonneg=True)
    C = build_conv_constraint_matrix(f, n)
    
    constraints = [C @ b <= M]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    
    # Try solving
    try:
        prob.solve(solver=cp.SCS, max_iters=5000)
    except Exception:
        try:
            prob.solve(solver=cp.CLARABEL)
        except Exception:
            return None
    
    if b.value is None:
        return None
    
    b_val = np.maximum(b.value, 0)
    if b_val.sum() < 1e-10:
        return None
    return b_val

    for it in range(n_iter):
        if time.monotonic() - t_start > max_time:
            break
        
        conv = fast_convolve(f)
        M = conv.max()
        
        b = solve_lp_direction(f, M, max_time=20)
        if b is None:
            break
        
        b_sum = b.sum()
        if b_sum < 1e-10:
            break
        
        # Rescale b to have same sum as f
        b = b * (s / b_sum)
        
        # Update
        t = step_size
        f_new = (1 - t) * f + t * b
        f_new = np.maximum(f_new, 0)
        s_new = f_new.sum()
        if s_new < 1e-10:
            break
        f_new = f_new / s_new
        
        # Check if peak decreased
        conv_new = fast_convolve(f_new)
        peak_new = conv_new.max()
        
        if peak_new < best_peak:
            best_peak = peak_new
            best_f = f_new.copy()
        
        f = f_new
        
        # Decay step size
        step_size = max(0.01, step_size * 0.95)
    
    return best_f


def optimize_lbfgsb(a0, beta, maxiter=300):
    """Optimize using L-BFGS-B with log parametrization."""
    n = len(a0)
    a0_clipped = np.maximum(a0, 1e-30)
    a0_clipped = a0_clipped / a0_clipped.sum()
    x0 = np.log(a0_clipped)
    
    def obj(x):
        a = np.exp(x - x.max())
        a = a / a.sum()
        conv = fast_convolve(a)
        sm, _ = smooth_max_and_weights(conv, beta)
        return 2.0 * n * sm
    
    def jac(x):
        a = np.exp(x - x.max())
        a = a / a.sum()
        conv = fast_convolve(a)
        sm, w = smooth_max_and_weights(conv, beta)
        grad_a = compute_grad_smooth_conv(a, w)
        lagrange = np.dot(a, grad_a)
        return a * (grad_a - lagrange)
    
    result = minimize(obj, x0, jac=jac, method='L-BFGS-B',
                      options={'maxiter': maxiter, 'maxls': 50})
    
    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt, result.fun


def cosine_mod_init(n, c=0.5, p=1.0, rng=None):
    """Initialize with cosine-modulated sequence."""
    k = np.arange(n)
    if rng is not None:
        phase = rng.uniform(0, 2 * np.pi)
    else:
        phase = 0
    a = (1 + c * np.cos(2 * np.pi * k / n + phase)) ** p
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def flat_spectrum_init(n, rng):
    """Initialize with approximately flat power spectrum."""
    fft_size = 1 << (n - 1).bit_length()
    A = np.zeros(fft_size, dtype=complex)
    A[0] = np.sqrt(n)
    phases = rng.uniform(0, 2 * np.pi, fft_size)
    for k in range(1, fft_size // 2 + 1):
        A[k] = np.exp(1j * phases[k])
        if k < fft_size - k:
            A[fft_size - k] = np.exp(-1j * phases[k])
    a = np.real(ifft(A, fft_size))[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def multi_harmonic_init(n, n_harmonics=3, rng=None):
    """Initialize with sum of harmonics (kept non-negative)."""
    k = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    if rng is not None:
        for h in range(1, n_harmonics + 1):
            amp = 0.3 / h
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * h * k / n + phase)
    else:
        a += 0.3 * np.cos(2 * np.pi * k / n)
        a += 0.15 * np.cos(4 * np.pi * k / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Cosine modulated
    for c in [0.3, 0.5, 0.7]:
        inits['cos_%s' % str(c)] = cosine_mod_init(n, c=c, p=1.0, rng=rng)
    inits['cos_sq'] = cosine_mod_init(n, c=0.5, p=2.0, rng=rng)
    inits['cos_cub'] = cosine_mod_init(n, c=0.5, p=3.0, rng=rng)

    for sf in [0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (sf * n)) ** 2)
        g /= g.sum()
        inits['gauss_%s' % str(sf)] = g

    inits['multi_harm'] = multi_harmonic_init(n, n_harmonics=3, rng=rng)
    inits['multi_harm5'] = multi_harmonic_init(n, n_harmonics=5, rng=rng)
    
    # Flat spectrum
    inits['flat_spec'] = flat_spectrum_init(n, rng)
    inits['flat_spec2'] = flat_spectrum_init(n, rng)

    inits['random_sparse'] = rng.dirichlet(np.ones(n) * 0.1)
    
    # Bimodal (edges)
    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp
    
    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0)
        ps = pp.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = pp / ps
            elif len(parent) > n:
                interp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), pp)
                interp = np.maximum(interp, 0)
                s2 = interp.sum()
                if s2 > 1e-10:
                    inits['parent_ds'] = interp / s2
            else:
                interp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), pp)
                interp = np.maximum(interp, 0)
                s2 = interp.sum()
                if s2 > 1e-10:
                    inits['parent_up'] = interp / s2
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based peak reduction,
smooth max L-BFGS polishing, coordinate descent, and multi-start over
diverse initializations (cosine-modulated, multi-harmonic, flat-spectrum)
and multiple sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import scipy.sparse as sp

try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max_and_weights(x, beta):
    """Compute smooth max and softmax weights."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def compute_grad_smooth_conv(a, w):
    """Compute gradient of sum_k w[k]*conv[k] w.r.t. a."""
    n = len(a)
    N = 1 << (len(w) + n - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0
    return grad


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, n):
    """Build sparse matrix C where C[k,j] = f[k-j] for valid (k,j)."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    C = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return C


def solve_lp_direction(f, M):
    """Solve LP: max sum(b) s.t. b>=0, C@b <= M."""
    if not HAS_CVXPY:
        return None
    n = len(f)
    b = cp.Variable(n, nonneg=True)
    C = build_conv_matrix(f, n)
    constraints = [C @ b <= M]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=10000, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.CLARABEL, verbose=False)
        except Exception:
            return None
    if b.value is None:
        return None
    b_val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if not np.all(np.isfinite(b_val)):
        return None
    if b_val.sum() < 1e-10:
        return None
    return b_val


def lp_based_optimization(f, n_iter=40, step_size=0.15, max_time=90):
    """Iterative LP-based peak reduction."""
    n = len(f)
    f = np.maximum(f.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-10:
        return f
    f = f / s

    best_f = f.copy()
    best_peak = fast_convolve(f).max()
    t_start = time.monotonic()

    current_step = step_size
    for it in range(n_iter):
        if time.monotonic() - t_start > max_time:
            break

        conv = fast_convolve(f)
        M = conv.max()

        b = solve_lp_direction(f, M)
        if b is None:
            break

        b_sum = b.sum()
        if b_sum < 1e-10:
            break

        b = b * (1.0 / b_sum)

        f_new = (1.0 - current_step) * f + current_step * b
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-10:
            break
        f_new = f_new / s_new

        conv_new = fast_convolve(f_new)
        peak_new = conv_new.max()

        if peak_new < best_peak - 1e-12:
            best_peak = peak_new
            best_f = f_new.copy()

        f = f_new
        current_step = max(0.03, current_step * 0.92)

    return best_f


def optimize_lbfgsb(a0, beta, maxiter=300):
    """Optimize using L-BFGS-B with log parametrization on simplex."""
    n = len(a0)
    a0_clipped = np.maximum(a0, 1e-30)
    a0_clipped = a0_clipped / a0_clipped.sum()
    x0 = np.log(a0_clipped)

    def obj_fn(x):
        a = np.exp(x - x.max())
        a = a / a.sum()
        conv = fast_convolve(a)
        sm, _ = smooth_max_and_weights(conv, beta)
        return 2.0 * n * sm

    def jac_fn(x):
        a = np.exp(x - x.max())
        a = a / a.sum()
        conv = fast_convolve(a)
        sm, w = smooth_max_and_weights(conv, beta)
        grad_a = compute_grad_smooth_conv(a, w)
        lagrange = np.dot(a, grad_a)
        return a * (grad_a - lagrange)

    result = minimize(obj_fn, x0, jac=jac_fn, method='L-BFGS-B',
                      options={'maxiter': maxiter, 'maxls': 50})
    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt, result.fun


def cosine_mod_init(n, c=0.5, p=1.0, rng=None):
    """Initialize with cosine-modulated sequence."""
    k = np.arange(n, dtype=np.float64)
    phase = 0.0
    if rng is not None:
        phase = rng.uniform(0, 2 * np.pi)
    a = (1.0 + c * np.cos(2.0 * np.pi * k / n + phase)) ** p
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def flat_spectrum_init(n, rng):
    """Initialize with approximately flat power spectrum (random phases)."""
    fft_size = 1 << (n - 1).bit_length()
    A = np.zeros(fft_size, dtype=complex)
    A[0] = np.sqrt(n)
    phases = rng.uniform(0, 2.0 * np.pi, fft_size)
    for k in range(1, fft_size // 2 + 1):
        A[k] = np.exp(1j * phases[k])
        if k < fft_size - k:
            A[fft_size - k] = np.exp(-1j * phases[k])
    a = np.real(ifft(A, fft_size))[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def multi_harmonic_init(n, n_harmonics=3, rng=None):
    """Initialize with sum of harmonics, clipped to non-negative."""
    k = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    if rng is not None:
        for h in range(1, n_harmonics + 1):
            amp = 0.35 / (h + 0.5)
            phase = rng.uniform(0, 2.0 * np.pi)
            a += amp * np.cos(2.0 * np.pi * h * k / n + phase)
    else:
        a += 0.3 * np.cos(2.0 * np.pi * k / n)
        a += 0.15 * np.cos(4.0 * np.pi * k / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    for c in [0.3, 0.5, 0.7, 0.85]:
        inits['cos_%s' % str(c)] = cosine_mod_init(n, c=c, p=1.0, rng=rng)
    inits['cos_sq'] = cosine_mod_init(n, c=0.6, p=2.0, rng=rng)
    inits['cos_cub'] = cosine_mod_init(n, c=0.5, p=3.0, rng=rng)

    for sf in [0.2, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (sf * n)) ** 2)
        g /= g.sum()
        inits['gauss_%s' % str(sf)] = g

    inits['multi_harm3'] = multi_harmonic_init(n, n_harmonics=3, rng=rng)
    inits['multi_harm5'] = multi_harmonic_init(n, n_harmonics=5, rng=rng)

    inits['flat_spec'] = flat_spectrum_init(n, rng)
    inits['flat_spec2'] = flat_spectrum_init(n, rng)

    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['random_sparse'] = rng.dirichlet(np.ones(n) * 0.1)

    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp

    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0.0)
        ps = pp.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = pp / ps
            elif len(parent) > n:
                interp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), pp)
                interp = np.maximum(interp, 0.0)
                s2 = interp.sum()
                if s2 > 1e-10:
                    inits['parent_ds'] = interp / s2
            else:
                interp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), pp)
                interp = np.maximum(interp, 0.0)
                s2 = interp.sum()
                if s2 > 1e-10:
                    inits['parent_up'] = interp / s2

    return inits


def coordinate_descent_refine(a, n_passes=5, max_steps_per_coord=3):
    """Simple coordinate descent to reduce max convolution."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()

    for _ in range(n_passes):
        conv = fast_convolve(a)
        max_k = int(np.argmax(conv))
        for i in range(n):
            old_val = a[i]
            for scale in [0.85, 0.7, 0.9, 1.1, 1.2]:
                a[i] = old_val * scale
                if a[i] < 0:
                    a[i] = old_val
                    continue
                s_new = a.sum()
                if s_new < 1e-10:
                    a[i] = old_val
                    continue
                a_norm = a / s_new
                obj_new = 2.0 * n * fast_convolve(a_norm).max()
                if obj_new < best_obj - 1e-10:
                    best_obj = obj_new
                    best_a = a_norm.copy()
                    a = a_norm
                    break
            a[i] = a[i]  # keep current
        a = best_a.copy()
        rebased = fast_convolve(a)
        peak = rebased.max()
        if peak * 2.0 * n < best_obj * 0.999:
            best_obj = 2.0 * n * peak

    return best_a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search combining LP-based peak reduction,
    smooth L-BFGS-B optimization, coordinate descent, and diverse initializations
    to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and p.sum() > 1e-10:
            parent = p
            print("Parent available: length=%d, sum=%.2f" % (len(p), p.sum()))
    except Exception:
        pass

    # Phase 1: Quick scan over lengths and initializations with L-BFGS
    print("=== Phase 1: Quick L-BFGS Scan ===")
    lengths = [128, 192, 256, 320, 384, 512]
    phase1_deadline = t_start + 350.0

    scan_results = []

    for n in lengths:
        if time.monotonic() > phase1_deadline - 25:
            break
        inits = generate_initializations(n, parent, rng)

        priority = ['parent', 'parent_ds', 'parent_up', 'cos_0.7', 'cos_0.85',
                    'cos_sq', 'multi_harm3', 'flat_spec', 'gauss_0.35', 'uniform']
        init_names = [nm for nm in priority if nm in inits]
        for nm in inits:
            if nm not in init_names:
                init_names.append(nm)
        init_names = init_names[:6]

        budget_per_init = max(5.0, (phase1_deadline - t_start - 20) / max(1, len(lengths) * 4))

        for name in init_names:
            if time.monotonic() > phase1_deadline - 15:
                break
            t_alloc = min(budget_per_init, phase1_deadline - time.monotonic() - 10)
            if t_alloc < 3:
                break

            a0 = inits[name]
            maxiter = int(min(200, max(30, t_alloc * 25)))
            a_opt, _ = optimize_lbfgsb(a0, beta=30.0, maxiter=maxiter)

            a_check = np.maximum(a_opt, 0.0)
            s_check = a_check.sum()
            if s_check > 1e-10:
                score = float(evaluate_sequence(a_check.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_check.copy()
                    print("  New best: score=%.6f, n=%d, init=%s (t=%.0fs)" %
                          (score, len(a_check), name, time.monotonic() - t_start))
                if np.isfinite(score):
                    scan_results.append((score, n, name, a_opt.copy()))

    scan_results.sort(key=lambda x: x[0])
    print("Phase 1 done. Best: %.6f" % best_score)

    # Phase 2: LP-based optimization on top candidates
    print("=== Phase 2: LP-Based Peak Reduction ===")
    phase2_deadline = deadline - 250.0

    top_candidates = scan_results[:12] if len(scan_results) >= 6 else scan_results

    for score_c, n_c, name_c, a_c in top_candidates:
        if time.monotonic() > phase2_deadline - 25:
            break
        t_alloc = min(100.0, phase2_deadline - time.monotonic() - 15)
        if t_alloc < 20:
            break

        a_lp = lp_based_optimization(a_c, n_iter=30, step_size=0.12, max_time=t_alloc - 15)

        a_check = np.maximum(a_lp, 0.0)
        s_check = a_check.sum()
        if s_check > 1e-10:
            score = float(evaluate_sequence(a_check.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_check.copy()
                print("  LP new best: score=%.6f, n=%d (t=%.0fs)" %
                      (score, len(a_check), time.monotonic() - t_start))

        # Polish with L-BFGS after LP
        if time.monotonic() < phase2_deadline - 25:
            a_pol, _ = optimize_lbfgsb(a_lp, beta=15.0, maxiter=150)
            a_check2 = np.maximum(a_pol, 0.0)
            if a_check2.sum() > 1e-10:
                score2 = float(evaluate_sequence(a_check2.tolist()))
                if np.isfinite(score2) and score2 < best_score:
                    best_score = score2
                    best_sequence = a_check2.copy()
                    print("  LP+LBFGS new best: score=%.6f, n=%d (t=%.0fs)" %
                          (score2, len(a_check2), time.monotonic() - t_start))

    # Try LP on different lengths using best sequence as init
    if best_sequence is not None:
        best_n = len(best_sequence)
    else:
        best_n = 256

    alt_lengths = sorted(set([l for l in
                              [best_n, best_n // 2, best_n * 2,
                               128, 192, 256, 320, 384, 512]
                              if 64 <= l <= 1024]))

    for n in alt_lengths:
        if time.monotonic() > phase2_deadline - 25:
            break

        if best_sequence is not None:
            if len(best_sequence) == n:
                a0 = best_sequence / best_sequence.sum()
            elif len(best_sequence) > n:
                interp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(best_sequence)), best_sequence)
                interp = np.maximum(interp, 0.0)
                s2 = interp.sum()
                a0 = interp / s2 if s2 > 1e-10 else np.ones(n) / n
            else:
                interp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(best_sequence)), best_sequence)
                interp = np.maximum(interp, 0.0)
                s2 = interp.sum()
                a0 = interp / s2 if s2 > 1e-10 else np.ones(n) / n
        else:
            a0 = cosine_mod_init(n, c=0.7, p=1.0, rng=rng)

        t_alloc = min(80.0, phase2_deadline - time.monotonic() - 15)
        if t_alloc < 20:
            break

        a_lp = lp_based_optimization(a0, n_iter=25, step_size=0.10, max_time=t_alloc - 15)
        a_check = np.maximum(a_lp, 0.0)
        if a_check.sum() > 1e-10:
            score = float(evaluate_sequence(a_check.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_check.copy()
                print("  Alt-length LP best: score=%.6f, n=%d (t=%.0fs)" %
                      (score, n, time.monotonic() - t_start))

        if time.monotonic() < phase2_deadline - 20:
            a_pol, _ = optimize_lbfgsb(a_lp, beta=12.0, maxiter=120)
            a_check2 = np.maximum(a_pol, 0.0)
            if a_check2.sum() > 1e-10:
                score2 = float(evaluate_sequence(a_check2.tolist()))
                if np.isfinite(score2) and score2 < best_score:
                    best_score = score2
                    best_sequence = a_check2.copy()
                    print("  Alt-length polish: score=%.6f, n=%d (t=%.0fs)" %
                          (score2, len(a_check2), time.monotonic() - t_start))

    print("Phase 2 done. Best: %.6f" % best_score)

    # Phase 3: Intensive fine-tuning on the best
    print("=== Phase 3: Fine-tuning ===")

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20.0

        # Multi-beta L-BFGS annealing
        for beta in [10.0, 6.0, 4.0, 2.5, 1.5]:
            if remaining < 8:
                break
            t_single = remaining / 6.0
            if t_single < 4:
                break
            maxiter = int(min(400, max(50, t_single * 30)))
            a_opt, _ = optimize_lbfgsb(best_sequence / best_sequence.sum(), beta=beta, maxiter=maxiter)
            a_check = np.maximum(a_opt, 0.0)
            if a_check.sum() > 1e-10:
                score = float(evaluate_sequence(a_check.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_check.copy()
                    print("  Beta-polish: score=%.6f (beta=%.1f, t=%.0fs)" %
                          (score, beta, time.monotonic() - t_start))
            remaining = deadline - time.monotonic() - 20.0

        # LP polish on the best
        if remaining > 40:
            lp_time = min(70.0, remaining - 30.0)
            a_lp = lp_based_optimization(best_sequence / best_sequence.sum(),
                                         n_iter=25, step_size=0.08, max_time=lp_time)
            a_check = np.maximum(a_lp, 0.0)
            if a_check.sum() > 1e-10:
                score = float(evaluate_sequence(a_check.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_check.copy()
                    print("  LP polish: score=%.6f (t=%.0fs)" %
                          (score, time.monotonic() - t_start))
            remaining = deadline - time.monotonic() - 20.0

            # L-BFGS after LP
            if remaining > 10:
                a_opt, _ = optimize_lbfgsb(best_sequence / best_sequence.sum(),
                                           beta=5.0, maxiter=int(min(300, remaining * 25)))
                a_check = np.maximum(a_opt, 0.0)
                if a_check.sum() > 1e-10:
                    score = float(evaluate_sequence(a_check.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_check.copy()
                        print("  Post-LP L-BFGS: score=%.6f (t=%.0fs)" %
                              (score, time.monotonic() - t_start))
                remaining = deadline - time.monotonic() - 20.0

        # Random perturbations + re-optimize
        if remaining > 50:
            n_best = len(best_sequence)
            for trial in range(5):
                if remaining < 15:
                    break
                scale = 0.002 + 0.004 * rng.uniform()
                noise = rng.standard_normal(n_best) * scale
                a_pert = project_simplex(best_sequence / best_sequence.sum() + noise)

                t_single = min(50.0, remaining - 15.0)
                if t_single < 10:
                    break

                a_lp = lp_based_optimization(a_pert, n_iter=15, step_size=0.10,
                                             max_time=max(10, t_single - 15))
                a_check = np.maximum(a_lp, 0.0)
                if a_check.sum() > 1e-10:
                    score = float(evaluate_sequence(a_check.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_check.copy()
                        print("  Perturb+LP: score=%.6f (trial=%d, t=%.0fs)" %
                              (score, trial, time.monotonic() - t_start))

                if remaining > 10:

                                               maxiter=int(min(200, (remaining - 10) * 20)))
                    a_check = np.maximum(a_opt, 0.0)

                            best_sequence = a_check.copy()
                            print("  Perturb+LBFGS: score=%.6f (trial=%d, t=%.0fs)" %
                                  (score, trial, time.monotonic() - t_start))
                remaining = deadline - time.monotonic() - 20.0

        # Coordinate descent final polish
        if remaining > 20:
            a_cd = coordinate_descent_refine(best_sequence / best_sequence.sum(), n_passes=3)
            a_check = np.maximum(a_cd, 0.0)
            if a_check.sum() > 1e-10:

                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_check.copy()
                    print("  Coord descent: score=%.6f (t=%.0fs)" %
                          (score, time.monotonic() - t_start))
                remaining = deadline - time.monotonic() - 20.0

                # One more L-BFGS from coordinate descent result
                if remaining > 8:
                    a_opt, _ = optimize_lbfgsb(a_cd, beta=4.0,
                                               maxiter=int(min(200, remaining * 20)))
                    a_check = np.maximum(a_opt, 0.0)
                    if a_check.sum() > 1e-10:
                        score = float(evaluate_sequence(a_check.tolist()))
                        if np.isfinite(score) and score < best_score:
                            best_score = score
                            best_sequence = a_check.copy()
                            print("  CD+LBFGS: score=%.6f (t=%.0fs)" %
                                  (score, time.monotonic() - t_start))

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128) / 128.0

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print("\nFINAL SCORE: %.6f (n=%d)" % (best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
