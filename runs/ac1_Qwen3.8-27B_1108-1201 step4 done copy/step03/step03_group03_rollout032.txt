import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_gradient(a, w):
    """Compute gradient of sum_k w_k * c_k with respect to a."""
    # g_i = 2 * sum_j w[i+j] * a[j]
    # = 2 * np.convolve(a, w[::-1], mode='full')[n-1 : 2n-1][::-1]
    n = len(a)
    result = np.convolve(a, w[::-1], mode='full')
    return 2.0 * result[n-1 : 2*n-1][::-1]

def optimize_softmax(a_init, time_limit, T_init=None, T_min=0.0001, 
                     lr_init=0.01, momentum=0.9, restart_interval=200):
    n = len(a_init)
    a = project_simplex(np.maximum(a_init, 0.0))
    
    if T_init is None:
        c = fast_convolve(a)
        T_init = c.max() / 10.0
    
    T = T_init
    lr = lr_init
    mu = momentum
    v = np.zeros(n)
    
    c = fast_convolve(a)
    best_obj = 2.0 * n * c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    stall = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute softmax weights
        scaled = c / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(2*n-1)
            w[np.argmax(c)] = 1.0
        else:
            w /= w_sum
        
        # Compute gradient
        g = compute_gradient(a, w)
        
        # Nesterov momentum step
        v = mu * v - lr * g
        a_new = project_simplex(a + v)
        
        # Evaluate
        c_new = fast_convolve(a_new)
        obj_new = 2.0 * n * c_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        c = c_new
        iter_count += 1
        
        # Adapt parameters
        T = max(T_min, T * 0.9995)
        lr = max(0.00001, lr * 0.9999)
        mu = min(0.999, mu * 1.0001)  # slowly increase momentum
        
        # Restart if stuck
        if stall > restart_interval:
            rng = np.random.default_rng(iter_count * 7919)
            a = project_simplex(a + rng.standard_normal(n) * 0.01)
            c = fast_convolve(a)
            v *= 0.5
            stall = 0
            T = max(T_min * 10, T * 2)  # increase temperature to escape
    
    return best_a, best_obj

    a = project_simplex(a_init)
    x = np.log(a + 1e-30)
    x -= x.mean()  # center for numerical stability
    
    T = T_init
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        # Define objective and gradient for L-BFGS
        def obj_func(x):
            a = np.exp(x - x.max())
            a /= a.sum()
            c = fast_convolve(a)
            c_max = c.max()
            return c_max + T * np.log(np.sum(np.exp((c - c_max) / T)))

            a = np.exp(x - x.max())
            a /= a.sum()

"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using convex optimization insights.
Key ideas: (1) The problem min max_k c_k(a) s.t. sum(a)=1, a>=0 is convex.
(2) Use log-sum-exp smoothing + L-BFGS for efficient convex optimization.
(3) Multi-scale: optimize at smaller n, then upsample.
(4) Coordinate descent polishing on the active set.
(5) Mirror descent (exponential weights) as alternative to Euclidean projection.
"""

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_param(x):
    """Convert x to probability vector via softmax."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return a / s


def compute_grad_wrt_a(a, w):
    """Gradient of sum_k w_k * c_k(a) with respect to a.
    g_i = 2 * sum_j w[i+j] * a[j]
    Computed via FFT-based convolution.
    """
    n = len(a)
    # g_i = 2 * convolve(a, w[::-1], 'full')[2n-2-i] for i=0..n-1
    result = np.convolve(a, w[::-1], mode='full')
    return 2.0 * result[n - 1:2 * n - 1][::-1]


def compute_grad_wrt_x(x, T):
    """Gradient of smoothed objective w.r.t. x (where a = softmax(x))."""
    a = softmax_param(x)
    c = fast_convolve(a)
    c_max = c.max()
    w = np.exp((c - c_max) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    g = compute_grad_wrt_a(a, w)
    # Transform: dx_i = a_i * (g_i - sum_j a_j * g_j)
    ga = np.dot(a, g)
    return a * (g - ga)


def smooth_objective(x, T):
    """Log-sum-exp smoothed objective."""
    a = softmax_param(x)
    c = fast_convolve(a)
    c_max = c.max()
    return c_max + T * np.log(np.sum(np.exp((c - c_max) / T)))


def optimize_lbfgs(a_init, time_limit, T_start=None, T_min_frac=0.001,
                   n_levels=8, maxiter_per_level=300):
    """Multi-level L-BFGS optimization with decreasing temperature."""
    n = len(a_init)
    a = project_simplex(np.maximum(a_init, 0.0))
    if a.sum() < 1e-15:
        a = np.ones(n) / n

    c = fast_convolve(a)
    c_max = c.max()
    if T_start is None:
        T_start = c_max / 20.0

    best_obj = 2.0 * n * c_max
    best_a = a.copy()
    x = np.log(a + 1e-30)
    x -= x.mean()

    t0 = time.monotonic()
    T = T_start

    for level in range(n_levels):
        if time.monotonic() - t0 > time_limit - 5.0:
            break

        Tw = T

        def obj_fn(xv):
            return smooth_objective(xv, Tw)

        def grad_fn(xv):
            return compute_grad_wrt_x(xv, Tw)

        rem = time_limit - (time.monotonic() - t0) - 5.0
        t_level = max(2.0, min(rem / max(1, n_levels - level), 30.0))

        try:
            result = scipy_minimize(obj_fn, x, jac=grad_fn, method='L-BFGS-B',
                                    options={'maxiter': maxiter_per_level,
                                             'ftol': 1e-14, 'gtol': 1e-12,
                                             'maxls': 50})
            if result.x is not None:
                a_new = softmax_param(result.x)
                c_new = fast_convolve(a_new)
                obj_new = 2.0 * n * c_new.max()
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    x = result.x.copy()
        except Exception:
            pass

        # Decrease temperature
        T = T * (T_min_frac / max(T_start / c_max if c_max > 0 else 1.0, 0.01))
        T = max(T, c_max * 0.0005)

    return best_a, best_obj


def optimize_momentum(a_init, time_limit, lr_init=0.005, momentum_init=0.85,
                      T_init=None, T_decay=0.999, T_min=None):
    """Nesterov momentum with softmax weights on the simplex."""
    n = len(a_init)
    a = project_simplex(np.maximum(a_init, 0.0))
    if a.sum() < 1e-15:
        a = np.ones(n) / n

    c = fast_convolve(a)
    c_max = c.max()
    if T_init is None:
        T_init = c_max / 15.0
    if T_min is None:
        T_min = c_max * 0.001

    T = T_init
    lr = lr_init
    mu = momentum_init
    v = np.zeros(n)

    best_obj = 2.0 * n * c_max
    best_a = a.copy()
    t0 = time.monotonic()
    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        # Softer weights for stability early on
        scaled = c / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        else:
            w /= ws

        g = compute_grad_wrt_a(a, w)

        # Nesterov step
        v = mu * v - lr * g
        a_new = project_simplex(a + v)

        c_new = fast_convolve(a_new)
        obj_new = 2.0 * n * c_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new
        c = c_new
        it += 1

        T = max(T_min, T * T_decay)
        lr = max(0.0005, lr * 0.9998)
        mu = min(0.999, mu + 0.00002)

        if stall > 300:
            rng = np.random.default_rng(it * 31 + 7)
            a = project_simplex(a + rng.standard_normal(n) * 0.005)
            c = fast_convolve(a)
            v *= 0.3
            stall = 0
            T = max(T_min * 5.0, T * 3.0)

    return best_a, best_obj


def optimize_mirror(a_init, time_limit, eta_init=0.1, T_init=None,
                    T_decay=0.9995, T_min=None):
    """Mirror descent (exponential weights) with softmax smoothing."""
    n = len(a_init)
    a = project_simplex(np.maximum(a_init, 0.0))
    if a.sum() < 1e-15:
        a = np.ones(n) / n

    c = fast_convolve(a)
    c_max = c.max()
    if T_init is None:
        T_init = c_max / 10.0
    if T_min is None:
        T_min = c_max * 0.002

    T = T_init
    eta = eta_init
    best_obj = 2.0 * n * c_max
    best_a = a.copy()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        scaled = c / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        else:
            w /= ws

        g = compute_grad_wrt_a(a, w)

        # Mirror (exponential) update
        log_a = np.log(a + 1e-30)
        log_a -= eta * g
        a_new = np.exp(log_a - log_a.max())
        a_new /= a_new.sum()

        c_new = fast_convolve(a_new)
        obj_new = 2.0 * n * c_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new
        c = c_new

        T = max(T_min, T * T_decay)
        eta = max(0.01, eta * 0.9999)

        if stall > 200:
            rng = np.random.default_rng(stall * 13)
            a = project_simplex(np.log(a + 1e-30) + rng.standard_normal(n) * 0.01)
            a = np.exp(np.log(a + 1e-30) - np.log(a + 1e-30).max())
            a /= a.sum()
            c = fast_convolve(a)
            stall = 0

    return best_a, best_obj


def coordinate_descent_polish(a_in, max_rounds=50, verbose=False):
    """Coordinate descent: transfer mass between pairs to reduce peak."""
    a = project_simplex(np.maximum(a_in, 0.0))
    n = len(a)

    c = fast_convolve(a)
    best_obj = 2.0 * n * c.max()
    best_a = a.copy()

    # Find the top peaks
    for rnd in range(max_rounds):
        improved = False
        peak_idx = int(np.argmax(c))
        # Also consider near-peaks
        top_k = min(20, 2 * n - 1)
        peaks = np.argpartition(c, -top_k)[-top_k:]

        for i in range(n):
            if a[i] < 1e-10:
                continue
            best_j = -1
            best_delta = 0.0
            best_obj_try = best_obj

            # Try several target positions
            candidates = []
            for j in range(n):
                if j == i:
                    continue
                candidates.append(j)
            # Sort by potential benefit: prefer j where a_{peak-j} is large
            candidates.sort(key=lambda j: -a[peak_idx - j] if 0 <= peak_idx - j < n else 0)
            candidates = candidates[:min(50, len(candidates))]

            for j in candidates:
                # Try transferring different fractions
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    delta = a[i] * frac
                    if delta < 1e-10:
                        continue
                    a_try = a.copy()
                    a_try[i] -= delta
                    a_try[j] += delta
                    if a_try[i] < -1e-12:
                        continue
                    a_try = np.maximum(a_try, 0.0)
                    st = a_try.sum()
                    if st < 1e-15:
                        continue
                    a_try /= st
                    c_try = fast_convolve(a_try)
                    obj_try = 2.0 * n * c_try.max()
                    if obj_try < best_obj_try - 1e-14:
                        best_obj_try = obj_try
                        best_j = j
                        best_delta = delta

            if best_j >= 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = np.maximum(a, 0.0)
                a /= a.sum()
                c = fast_convolve(a)
                best_obj = 2.0 * n * c.max()
                best_a = a.copy()
                improved = True

        if not improved:
            break

    return best_a


def upsample(a, new_n):
    """Upsample sequence a to length new_n via interpolation."""
    n = len(a)
    if new_n == n:
        return a.copy()
    if new_n < n:
        x_old = np.linspace(0, 1, n)
        x_new = np.linspace(0, 1, new_n)
        return np.interp(x_new, x_old, a)
    # Upsample
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def gen_inits(n, parent=None, rng=None):
    """Generate diverse initializations on the simplex."""
    if rng is None:
        rng = np.random.default_rng(123)
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Triangular
    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()

    # Edge-concentrated
    edge = np.zeros(n)
    e = max(1, n // 4)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Bimodal
    g2 = (np.exp(-0.5 * ((x - 0.3 * n) / (0.1 * n)) ** 2) +
          np.exp(-0.5 * ((x - 0.7 * n) / (0.1 * n)) ** 2))
    inits['bimodal'] = g2 / g2.sum()

    # Sine-modulated
    sine = 1.0 + 0.5 * np.sin(2 * np.pi * x / n * 3)
    inits['sine'] = sine / sine.sum()

    # Random smooth
    freq = rng.integers(2, 8)
    smooth = 1.0 + 0.3 * np.sin(2 * np.pi * freq * x / n + rng.uniform(0, 2 * np.pi))
    inits['smooth_rand'] = smooth / smooth.sum()

    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10 and len(p) > 0:
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                inits['parent_interp'] = upsample(p / ps, n)

    return inits

        # L-BFGS + CD cycle
        for cycle in range(3):
            if time.monotonic() > deadline - 40:
                break
            rem = deadline - time.monotonic() - 35.0
            t_cyc_lb = min(rem * 0.4, 40.0)
            t_cyc_cd = min(rem * 0.3, 40.0)
            if t_cyc_lb < 5:
                break

            a_new, _ = optimize_lbfgs(best_sequence, t_cyc_lb, n_levels=8,
                                       maxiter_per_level=300)
            score_new = float(evaluate_sequence(a_new.tolist()))
            if np.isfinite(score_new) and score_new < best_score:
                best_score = score_new
                best_sequence = a_new.copy()
                print(f"  Cycle {cycle} L-BFGS: {score_new:.6f}")

            if t_cyc_cd > 5:
                a_cd2 = coordinate_descent_polish(best_sequence, max_rounds=30)
                score_cd2 = float(evaluate_sequence(a_cd2.tolist()))
                if np.isfinite(score_cd2) and score_cd2 < best_score:
                    best_score = score_cd2
                    best_sequence = a_cd2.copy()
                    print(f"  Cycle {cycle} CD: {score_cd2:.6f}")

        # Final small perturbations + quick optimize
        if time.monotonic() < deadline - 20:
            n_b = len(best_sequence)
            for ps in [0.002, 0.005, 0.01]:
                if time.monotonic() > deadline - 15:
                    break
                for _ in range(3):
                    if time.monotonic() > deadline - 12:
                        break
                    noise = rng.standard_normal(n_b) * ps
                    a_p = project_simplex(best_sequence + noise)
                    a_opt, _ = optimize_lbfgs(a_p, 5.0, n_levels=4,
                                               maxiter_per_level=100)
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  Perturb({ps}): {score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    # Final validation
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(256) / 256
        final_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
