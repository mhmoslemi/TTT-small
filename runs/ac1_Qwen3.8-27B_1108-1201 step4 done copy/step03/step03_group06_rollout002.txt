**Frank-Wolfe (Conditional Gradient) Method:**
- At each iteration, find the vertex `v` of the simplex (i.e., a standard basis vector `e_i`) that minimizes `∇f(a)^T (e_i - a)`.
- Take a step: `a_new = (1-t)*a + t*e_i`

The subgradient `∇f(a)` at `a` (choosing the argmax `k*`) is:
`g_i = a[k*-i] + a[i-k*]` (with 0 for out-of-range)

Then `∇f(a)^T (e_i - a) = g_i - sum_j g_j * a[j]`

We want the `i` that minimizes this, i.e., the `i` that minimizes `g_i - sum_j g_j * a[j]`, i.e., the `i` with the smallest `g_i`.

So the Frank-Wolfe step is: move mass from `a` toward the coordinate `i` with the smallest `g_i = a[k*-i] + a[i-k*]`.

This is actually quite intuitive! We're moving mass to a coordinate that has low "interaction" with the peak.

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)

def subgradient_at_conv(a, conv):
    """Compute exact subgradient of max(conv) w.r.t. a at the argmax."""
    n = len(a)
    k_star = np.argmax(conv)
    # g_i = a[k_star - i] + a[i - k_star] (with 0 for out of range)
    g = np.zeros(n)
    for i in range(n):
        idx1 = k_star - i
        idx2 = i - k_star
        if 0 <= idx1 < n:
            g[i] += a[idx1]
        if 0 <= idx2 < n:
            g[i] += a[idx2]
    return g, k_star

def subgradient_fast(a, conv):
    """Compute exact subgradient using FFT-ish approach."""
    n = len(a)
    k_star = np.argmax(conv)
    # g[i] = a[k_star - i] (if 0 <= k_star-i < n) + a[i - k_star] (if 0 <= i-k_star < n)
    g = np.zeros(n)
    # First term: a[k_star - i] for i where 0 <= k_star - i < n, i.e., k_star - n + 1 <= i <= k_star
    i_min = max(0, k_star - n + 1)
    i_max = min(n - 1, k_star)
    if i_min <= i_max:
        indices = np.arange(i_min, i_max + 1)
        g[indices] += a[k_star - indices]
    # Second term: a[i - k_star] for i where 0 <= i - k_star < n, i.e., k_star <= i <= k_star + n - 1
    i_min2 = max(0, k_star)
    i_max2 = min(n - 1, k_star + n - 1)
    if i_min2 <= i_max2:
        indices = np.arange(i_min2, i_max2 + 1)
        g[indices] += a[indices - k_star]
    return g, k_star

def fw_step(a, g, step_size=None):
    """Frank-Wolfe step: move toward e_i where g_i is minimized."""
    i = np.argmin(g)
    if step_size is None:
        # Use standard FW step size: 2/(t+2) approximation, or just use a fixed fraction
        step_size = 0.5  # We'll use line search instead
    a_new = (1 - step_size) * a + step_size * np.zeros(len(a))
    a_new[i] = step_size
    return a_new, i

def fw_optimize(a0, time_limit, verbose=False, seed=0):
    """Frank-Wolfe optimization on the simplex with line search."""
    rng = np.random.default_rng(seed)
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:

    else:

    n = len(a)

    best_a = a.copy()

        conv = fast_convolve(a)
        M = conv.max()
        g, k_star = subgradient_fast(a, conv)
        
        # Find the FW direction: e_i - a where i = argmin(g)
        i = np.argmin(g)
        s = a[i]

            continue

                if obj_trial < best_obj_ls:

                    best_gamma = gamma

def lp_direction(a, M, top_k=None, conv=None):

"""
Hybrid Frank-Wolfe + LP optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses exact subgradients for fast FW iterations, LP for larger moves,
multi-start exploration across sequence lengths, and adaptive time management.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def subgradient_fast(a, conv):
    """Compute exact subgradient of max(conv) at the argmax. O(n) time."""
    n = len(a)
    k_star = int(np.argmax(conv))
    g = np.zeros(n)
    # g[i] = a[k_star - i] (if valid) + a[i - k_star] (if valid)
    # First term: a[k_star - i] valid when 0 <= k_star - i < n => k_star - n + 1 <= i <= k_star
    i_min = max(0, k_star - n + 1)
    i_max = min(n - 1, k_star)
    if i_min <= i_max:
        indices = np.arange(i_min, i_max + 1)
        g[indices] += a[k_star - indices]
    # Second term: a[i - k_star] valid when 0 <= i - k_star < n => k_star <= i <= k_star + n - 1
    i_min2 = max(0, k_star)
    i_max2 = min(n - 1, k_star + n - 1)
    if i_min2 <= i_max2:
        indices = np.arange(i_min2, i_max2 + 1)
        g[indices] += a[indices - k_star]
    return g, k_star


def fw_line_search(a, gamma_candidates):
    """Evaluate FW step with line search. Returns (best_a, best_obj)."""
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    g, k_star = subgradient_fast(a, conv)
    i = int(np.argmin(g))

    if a[i] > 1 - 1e-12:
        return None, 2.0 * n * M

    best_trial = None
    best_obj = 2.0 * n * M

    for gamma in gamma_candidates:
        a_trial = a.copy()
        a_trial[i] += gamma * (1 - a[i])  # move toward e_i
        # Actually, FW step: a_new = (1-gamma)*a + gamma*e_i
        a_trial = (1 - gamma) * a
        a_trial[i] += gamma
        conv_trial = fast_convolve(a_trial)
        obj_trial = 2.0 * n * conv_trial.max()
        if obj_trial < best_obj:
            best_obj = obj_trial
            best_trial = a_trial.copy()

    return best_trial, best_obj

            print(f"    Hybrid: obj={best_obj:.8f}, no_improve={no_improve}, t={elapsed:.1f}s")

    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


def pairwise_swap_optimize(a0, time_limit, verbose=False, seed=0):
    """Optimize by trying pairwise mass transfers (i->j)."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        g, k_star = subgradient_fast(a, conv)

        # Find best pair: move mass from high-g to low-g coordinate
        # Sort by g
        order = np.argsort(g)
        improved = False

        # Try moving mass to the lowest-g coordinate from the highest-g one
        target = int(order[0])
        # Source: weighted random from high-g coordinates
        for attempt in range(20):
            source = int(order[n - 1 - rng.integers(0, min(n // 2, n - 1))])
            if source == target:
                continue
            if a[source] < 1e-10:
                continue

            for frac in [0.5, 0.25, 0.1]:
                delta = a[source] * frac
                a_trial = a.copy()
                a_trial[source] -= delta
                a_trial[target] += delta
                conv_trial = fast_convolve(a_trial)
                obj_trial = 2.0 * n * conv_trial.max()
                if obj_trial < best_obj - 1e-14:
                    best_obj = obj_trial
                    best_a = a_trial.copy()
                    a = a_trial
                    improved = True
                    break
            if improved:
                break

        if not improved:
            no_improve += 1
            if no_improve > 100:
                a = project_simplex(a + rng.standard_normal(n) * 0.01)
                no_improve = 0

    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


def generate_init(name, n, rng, parent=None):
    """Generate a specific initialization."""
    x = np.arange(n, dtype=np.float64)

    if name == 'uniform':
        return np.ones(n) / n
    elif name == 'ramp_inc':
        v = np.arange(1, n + 1, dtype=np.float64)
        return v / v.sum()
    elif name == 'ramp_dec':
        v = np.arange(n, 0, -1, dtype=np.float64)
        return v / v.sum()
    elif name == 'triangle_edge':
        v = np.minimum(x, n - 1 - x) + 1.0
        return v / v.sum()
    elif name == 'triangle_center':
        v = n - 1 - np.minimum(x, n - 1 - x) + 1.0
        return v / v.sum()
    elif name.startswith('gauss'):
        sf = float(name.split('_')[1])
        v = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        return v / v.sum()
    elif name == 'bimodal':
        v = np.zeros(n)
        edge = max(1, n // 4)
        v[:edge] = 1.0
        v[-edge:] = 1.0
        return v / v.sum()
    elif name == 'cosine':
        v = np.cos(np.pi * x / max(1, n - 1)) ** 2
        v = np.maximum(v, 0.01)
        return v / v.sum()
    elif name == 'random_sparse':
        v = rng.dirichlet(np.ones(n) * 0.1)
        return v
    elif name == 'random':
        v = rng.dirichlet(np.ones(n) * 1.0)
        return v
    elif name == 'bump':
        v = np.exp(-x / (0.3 * n))
        return v / v.sum()
    elif name == 'palindrome':
        # Generate half and mirror
        half = rng.dirichlet(np.ones(n // 2 + 1) * 0.5)
        v = np.zeros(n)
        v[:n // 2 + 1] = half
        v[n // 2 + 1:] = half[:max(0, n - n // 2 - 1)][::-1]
        v = np.maximum(v, 0)
        return v / v.sum()
    elif name == 'parent':
        if parent is not None:
            if len(parent) >= n:
                pp = np.maximum(parent[:n], 0)
            else:
                pp = np.interp(np.linspace(0, 1, n),
                               np.linspace(0, 1, len(parent)), parent)
                pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                return pp / ps
        return np.ones(n) / n
    else:
        return np.ones(n) / n


def get_parent():
    """Try to retrieve the parent sequence."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p / p.sum()
    except NameError:
        pass
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase hybrid optimization: FW + LP + pairwise swaps.
    Explores multiple sequence lengths, uses exact subgradients for fast
    FW iterations, and LP for larger moves. Final polish with fine steps.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_a = None
    best_score = float('inf')

    parent = get_parent()
    if parent is not None:
        print(f"Parent available: length={len(parent)}, sum={parent.sum():.4f}")

    def check_and_update(a):
        nonlocal best_a, best_score
        if a is None:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"  New best: score={score:.8f}, n={len(a)}")
            return True
        return False

    # ========== Phase 1: Quick scan with FW ==========
    print("=== Phase 1: Quick FW Scan ===")
    t_phase1_end = time.monotonic() + 200.0

    scan_n = [128, 192, 256, 320, 384, 512]
    scan_inits = ['uniform', 'bimodal', 'gauss_0.3', 'random', 'triangle_edge']

    scan_results = []
    for n in scan_n:
        if time.monotonic() > t_phase1_end - 20:
            break
        for name in scan_inits:
            if time.monotonic() > t_phase1_end - 10:
                break
            a0 = generate_init(name, n, rng, parent=parent)
            # Also try parent initialized at this length
            if parent is not None:
                a0 = generate_init('parent', n, rng, parent=parent)

            t_alloc = min(30.0, t_phase1_end - time.monotonic() - 5)
            if t_alloc < 5:
                break

            a_opt, obj = fw_optimize(a0, time_limit=t_alloc, seed=n + hash(name) % 1000)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                check_and_update(a_opt)

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.8f}")

    # ========== Phase 2: Deep optimization on best candidates ==========
    print("=== Phase 2: Deep Optimization ===")
    t_phase2_end = time.monotonic() + 600.0

    # Collect starting points
    deep_starts = []
    for score, n, name, a_opt in scan_results[:5]:
        deep_starts.append((n, a_opt.copy()))

    if parent is not None:
        deep_starts.append((len(parent), parent.copy()))

    # Always include best
    if best_a is not None:
        deep_starts.append((len(best_a), best_a.copy()))

    # Remove duplicates (keep first occurrence)
    seen = set()
    unique_starts = []
    for n, a in deep_starts:
        key = n
        if key not in seen:
            unique_starts.append((n, a))
            seen.add(key)
    deep_starts = unique_starts[:6]

    # Run hybrid FW+LP on each
    time_per_start = max(30.0, (t_phase2_end - time.monotonic()) / max(1, len(deep_starts)))

    for i, (n_start, a_start) in enumerate(deep_starts):
        if time.monotonic() > t_phase2_end - 30:
            break
        t_alloc = min(time_per_start, t_phase2_end - time.monotonic() - 20)
        if t_alloc < 15:
            break

        print(f"  Deep [{i}]: n={n_start}, t_alloc={t_alloc:.0f}s")
        a_opt, obj = fw_lp_hybrid(a_start, time_limit=t_alloc, seed=i * 7 + 1)
        check_and_update(a_opt)

    print(f"Phase 2 done. Best: {best_score:.8f}")

    # ========== Phase 3: LP refinement ==========
    print("=== Phase 3: LP Refinement ===")
    t_phase3_end = time.monotonic() + 400.0

    if best_a is not None:
        n_best = len(best_a)
        # Run LP with full constraints (if n is small enough)
        if n_best <= 512:
            t_alloc = min(120.0, t_phase3_end - time.monotonic() - 20)
            if t_alloc > 15:
                a_opt, obj = lp_optimize(best_a, time_limit=t_alloc, seed=123)
                check_and_update(a_opt)

        # Also run FW with more patience
        t_alloc = min(150.0, t_phase3_end - time.monotonic() - 20)
        if t_alloc > 15:
            a_opt, obj = fw_optimize(best_a, time_limit=t_alloc, seed=456)
            check_and_update(a_opt)

        # Try pairwise swap
        t_alloc = min(80.0, t_phase3_end - time.monotonic() - 20)
        if t_alloc > 10:
            a_opt, obj = pairwise_swap_optimize(best_a, time_limit=t_alloc, seed=789)
            check_and_update(a_opt)

    print(f"Phase 3 done. Best: {best_score:.8f}")

    # ========== Phase 4: Explore slightly different n ==========
    print("=== Phase 4: Adjacent n exploration ===")
    t_phase4_end = time.monotonic() + 250.0

    if best_a is not None:
        n_best = len(best_a)
        nearby_ns = [n_best - 32, n_best - 16, n_best + 16, n_best + 32]
        nearby_ns = [n for n in nearby_ns if 32 <= n <= 1024]

        for n_try in nearby_ns:
            if time.monotonic() > t_phase4_end - 30:
                break
            # Interpolate best_a to new length
            a_interp = np.interp(np.linspace(0, 1, n_try),
                                  np.linspace(0, 1, n_best), best_a)
            a_interp = np.maximum(a_interp, 0)
            s = a_interp.sum()
            if s > 1e-10:
                a_interp /= s

            t_alloc = min(40.0, t_phase4_end - time.monotonic() - 15)
            if t_alloc < 10:
                break
            a_opt, obj = fw_optimize(a_interp, time_limit=t_alloc, seed=n_try)
            check_and_update(a_opt)

    print(f"Phase 4 done. Best: {best_score:.8f}")

    # ========== Phase 5: Final polish ==========
    print("=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    if best_a is not None and remaining > 30:
        n_best = len(best_a)

        # Very fine FW
        t_alloc = min(remaining * 0.4, 120.0)
        if t_alloc > 10:
            a_opt, obj = fw_optimize(best_a, time_limit=t_alloc, seed=999)
            check_and_update(a_opt)

        # Fine LP
        remaining = deadline - time.monotonic()
        if n_best <= 512 and remaining > 30:
            t_alloc = min(remaining * 0.3, 80.0)
            if t_alloc > 10:
                a_opt, obj = lp_optimize(best_a, time_limit=t_alloc, seed=555)
                check_and_update(a_opt)

        # Tiny perturbations
        remaining = deadline - time.monotonic()
        if remaining > 10:
            for i in range(min(20, int(remaining / 3))):
                if time.monotonic() > deadline - 10:
                    break
                for scale in [0.0005, 0.001, 0.002, 0.005]:
                    noise = rng.standard_normal(n_best) * scale
                    a_pert = project_simplex(best_a + noise)
                    score_pert = float(evaluate_sequence(a_pert.tolist()))
                    if np.isfinite(score_pert) and score_pert < best_score:
                        best_score = score_pert
                        best_a = a_pert.copy()
                        print(f"  Perturb [{i}] s={scale}: {best_score:.8f}")
                        break

    # Final verification
    if best_a is None:
        best_a = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_a = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_a.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_a)})")
    return [float(x) for x in best_a]
