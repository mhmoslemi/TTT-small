"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding with
iterative M reduction, peak-directed coordinate descent, and smoothed gradient
polish. Multi-phase search across sequence lengths 384-768.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(a, M, n):
    """
    Solve LP: max sum(b_j) s.t. b_j >= 0, (a*b)[k] <= M for all k.
    Returns b (unnormalized) or None.
    """
    c_obj = -np.ones(n)
    
    # Build constraint matrix
    A_ub = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = i
        k_end = min(2 * n - 1, i + n)
        A_ub[k_start:k_end, i] = a[:k_end - k_start]
    
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
    except Exception:
        return None
    
    if not result.success or result.x is None:
        return None
    
    b = np.maximum(result.x, 0)
    if b.sum() < 1e-15:
        return None
    
    return b


def smoothed_grad(a, conv, beta):
    """Gradient of logsumexp(beta * conv) / beta w.r.t. a."""
    n = len(a)
    scaled = beta * conv
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        return 2.0 * a * conv.max()
    p /= p_sum
    
    w = conv * p
    
    fft_size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    cross = np.fft.irfft(A * np.conj(W), fft_size)
    
    return 2.0 * cross[:n]

                best_a = a_try.copy()
                cur_a = a_try.copy()
                improved = True
                break
        
        if not improved:
            # Try very small steps
            for t in [0.01, 0.02, 0.05]:
                a_try = (1 - t) * cur_a + t * b_norm
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                conv_try = fast_convolve(a_try)
                obj_try = conv_try.max()
                
                if obj_try < M_cur * (1 - 1e-6):
                    cur_a = a_try.copy()
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                    break

                    noise = rng.standard_normal(n) * 0.002
                    cur_a = project_simplex(cur_a + noise)

def peak_descent(a0, time_limit, seed=0):
    """
    Peak-directed coordinate descent: identify convolution peaks and
    reduce them by moving mass to "quiet" positions.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        conv_cur = fast_convolve(a)
        max_val = conv_cur.max()
        
        # Find top peaks
        top_k = min(5, len(conv_cur))
        peak_idx = np.argpartition(conv_cur, -top_k)[-top_k:]
        peak_idx = peak_idx[conv_cur[peak_idx] > max_val * 0.98]
        
        if len(peak_idx) == 0:
            break
        
        improved = False
        
        for k in peak_idx:
            k = int(k)
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            
            if i_lo > i_hi:
                continue
            
            # Find the pair with largest product
            contributions = np.array([a[i] * a[k - i] for i in range(i_lo, i_hi + 1)])
            if len(contributions) == 0:
                continue
            
            max_ci = np.argmax(contributions)
            i_star = i_lo + max_ci
            
            if a[i_star] < 1e-8:
                continue
            
            # Find "quiet" positions (low total contribution to peaks)
            # Just try a few random positions far from i_star
            for attempt in range(5):
                j = rng.integers(0, n)
                if abs(j - i_star) < 3:
                    continue
                
                # Binary search for best transfer amount
                lo, hi = 0.0, a[i_star] * 0.8
                best_amt = 0.0
                best_obj_try = best_obj
                
                for _ in range(8):
                    mid = 0.5 * (lo + hi)
                    a_try = a.copy()
                    a_try[i_star] -= mid
                    a_try[j] += mid
                    a_try = np.maximum(a_try, 0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        break
                    a_try = a_try / s_try
                    conv_try = fast_convolve(a_try)
                    obj_try = conv_try.max()
                    
                    if obj_try < best_obj_try:
                        best_obj_try = obj_try
                        best_amt = mid
                        lo = mid
                    else:
                        hi = mid
                
                if best_amt > 0 and best_obj_try < best_obj - 1e-12:
                    a[i_star] -= best_amt
                    a[j] += best_amt
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    best_obj = best_obj_try
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            iter_count += 1
            if iter_count % 2000 == 0:
                # Random perturbation
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i != j and a[i] > 1e-6:
                    amt = a[i] * 0.1
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0)
                    a = a / a.sum()
    
    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Bell shape
    tri = 1.0 / (1.0 + 0.5 * ((x - n / 2) / (n / 4)) ** 2)
    tri /= tri.sum()
    inits['bell'] = tri
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.3 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g
    
    # Bimodal
    bi = np.zeros(n)
    edge = max(2, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Raised cosine
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / max(n - 1, 1)))
    cos_w = np.maximum(cos_w, 1e-6)
    cos_w /= cos_w.sum()
    inits['raised_cos'] = cos_w
    
    # Random
    inits['rand1'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['rand2'] = rng.dirichlet(np.ones(n) * 1.5)
    
    # Parent-based
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps
    
    return inits

"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using an enhanced LP-based iterative
method with binary-search M tightening, peak-directed coordinate descent,
and smoothed gradient polish. Key improvements: adaptive LP constraint
tightening, backtracking line search, and dual-variable-guided peak targeting.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp(a, M, n, time_limit=10.0):
    """
    Solve LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0.
    Returns (b, dual_vars) or (None, None).
    """
    c_obj = -np.ones(n)

    A_ub = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = i
        k_end = min(2 * n - 1, i + n)
        A_ub[k_start:k_end, i] = a[:k_end - k_start]

    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True,
                                                   'time_limit': time_limit})
    except Exception:
        return None, None

    if not result.success or result.x is None:
        return None, None

    b = np.maximum(result.x, 0)
    if b.sum() < 1e-15:
        return None, None

    dual = None
    if result.ineqlin_marginals is not None:
        dual = np.maximum(result.ineqlin_marginals[:2*n-1], 0)

    return b, dual


def find_min_M(a, n, M_current, M_floor, iters=15):
    """Binary search for minimum feasible M (where LP has positive optimum)."""
    M_lo = M_floor
    M_hi = M_current
    for _ in range(iters):
        if M_hi - M_lo < 1e-6 * M_lo:
            break
        M_mid = 0.5 * (M_lo + M_hi)
        b, _ = solve_lp(a, M_mid, n, time_limit=5.0)
        if b is not None and b.sum() > 1e-8:
            M_hi = M_mid
        else:
            M_lo = M_mid
    return M_hi


def smoothed_grad(a, conv, beta):
    """Gradient of logsumexp(beta * conv) / beta w.r.t. a."""
    n = len(a)
    scaled = beta * conv
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        peak_idx = np.argmax(conv)
        p = np.zeros(len(conv))
        p[peak_idx] = 1.0
    else:
        p /= p_sum

    w = conv * p
    fft_size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    cross = np.fft.irfft(A * np.conj(W), fft_size)
    return 2.0 * cross[:n]


def gradient_phase(a0, time_limit, beta_start=50.0, beta_end=500.0,
                   lr=0.01, momentum=0.93, seed=0):
    """Nesterov projected gradient descent with log-sum-exp smoothing."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    perturb_every = 250

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        beta = beta_start + (beta_end - beta_start) * frac

        y = a + momentum * v
        conv_y = fast_convolve(y)
        grad = smoothed_grad(y, conv_y, beta)

        v = momentum * v - lr * grad
        a = project_simplex(a + v)

        conv = fast_convolve(a)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= perturb_every:
            noise_scale = 0.002 * (1 + 0.5 * np.sin(no_improve * 0.01))
            noise = rng.standard_normal(n) * noise_scale
            a = project_simplex(a + noise)
            v *= 0.3
            no_improve = 0
            perturb_every = int(perturb_every * 1.3)
            lr = max(0.0005, lr * 0.7)

    return best_a, best_obj


def lp_phase(a0, time_limit, seed=0, use_binary_M=True):
    """
    Enhanced LP-based optimization with binary M search and
    backtracking line search.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    cur_a = a.copy()

    t0 = time.monotonic()
    stuck = 0
    M_target = best_obj

    while time.monotonic() - t0 < time_limit:
        conv_cur = fast_convolve(cur_a)
        M_cur = conv_cur.max()

        if M_cur < M_target:
            M_target = M_cur

        # Option: binary search for tighter M
        M_lp = M_cur
        if use_binary_M and (time.monotonic() - t0) < time_limit * 0.8:
            M_floor = M_target * 0.95
            M_lp = find_min_M(cur_a, n, M_cur * 1.001, M_floor, iters=10)
            M_lp = max(M_lp, M_floor)

        b, dual = solve_lp(cur_a, M_lp, n, time_limit=8.0)

        if b is None:
            stuck += 1
            if stuck > 15:
                noise = rng.standard_normal(n) * 0.004
                cur_a = project_simplex(cur_a + noise)
                stuck = 0
            continue

        stuck = 0
        b_norm = b / b.sum()

        # Backtracking line search
        improved = False
        t = 1.0
        for _ in range(12):
            t *= 0.7
            a_try = (1 - t) * cur_a + t * b_norm
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            conv_try = fast_convolve(a_try)
            obj_try = conv_try.max()

            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                cur_a = a_try.copy()
                improved = True
                break
            elif obj_try < M_cur * (1 - 1e-5):
                cur_a = a_try.copy()
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                improved = True
                break

        if not improved:
            # Try fixed small steps
            for t_fixed in [0.05, 0.1, 0.15, 0.2]:
                a_try = (1 - t_fixed) * cur_a + t_fixed * b_norm
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                conv_try = fast_convolve(a_try)
                obj_try = conv_try.max()
                if obj_try < M_cur * (1 - 1e-6):
                    cur_a = a_try.copy()
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                    break
            else:
                # Use dual info to guide perturbation
                if dual is not None and np.any(dual > 0):
                    peak_lags = np.where(dual > np.max(dual) * 0.5)[0]
                    # Perturb elements contributing most to those peaks
                    for _ in range(3):
                        k = int(rng.choice(peak_lags))
                        i_lo = max(0, k - n + 1)
                        i_hi = min(n - 1, k)
                        if i_lo <= i_hi:
                            i = int(rng.integers(i_lo, i_hi + 1))
                            if a[i] > 1e-6:
                                j = int(rng.integers(0, n))
                                if j != i:
                                    amt = a[i] * 0.05
                                    a_try = cur_a.copy()
                                    a_try[i] -= amt
                                    a_try[j] += amt
                                    a_try = np.maximum(a_try, 0)
                                    a_try = a_try / a_try.sum()
                                    conv_try = fast_convolve(a_try)
                                    if conv_try.max() < M_cur:
                                        cur_a = a_try
                                        if conv_try.max() < best_obj:
                                            best_obj = conv_try.max()
                                            best_a = a_try.copy()
                                    break
                else:
                    if rng.random() < 0.2:
                        noise = rng.standard_normal(n) * 0.002
                        cur_a = project_simplex(cur_a + noise)

    return best_a, best_obj


def peak_coordinate_descent(a0, time_limit, seed=0):
    """
    Peak-directed coordinate descent: for each convolution peak,
    identify the worst-contributing pair and transfer mass to reduce it.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv_cur = fast_convolve(a)
        max_val = conv_cur.max()

        # Find peaks above 95% of max
        threshold = max_val * 0.97
        peak_indices = np.where(conv_cur >= threshold)[0]

        if len(peak_indices) == 0:
            break

        improved = False

        for k in peak_indices[:8]:
            k = int(k)
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo > i_hi:
                continue

            # Compute contributions
            idx = np.arange(i_lo, i_hi + 1)
            contribs = a[idx] * a[k - idx]
            if len(contribs) == 0:
                continue

            # Try reducing the top-3 contributing elements
            top_contrib = np.argsort(contribs)[::-1][:3]
            for ci in top_contrib:
                i_star = idx[ci]
                if a[i_star] < 1e-8:
                    continue

                # Try moving to several candidate positions
                for _ in range(4):
                    j = int(rng.integers(0, n))
                    if j == i_star or abs(j - i_star) < 2:
                        continue

                    # Binary search for best transfer
                    lo_amt = 0.0
                    hi_amt = a[i_star] * 0.7
                    best_amt = 0.0
                    best_obj_try = best_obj

                    for _ in range(6):
                        mid_amt = 0.5 * (lo_amt + hi_amt)
                        if mid_amt < 1e-10:
                            break
                        a_try = a.copy()
                        a_try[i_star] -= mid_amt
                        a_try[j] += mid_amt
                        a_try = np.maximum(a_try, 0)
                        s_try = a_try.sum()
                        if s_try < 1e-15:
                            break
                        a_try = a_try / s_try
                        obj_try = fast_convolve(a_try).max()

                        if obj_try < best_obj_try - 1e-12:
                            best_obj_try = obj_try
                            best_amt = mid_amt
                            lo_amt = mid_amt
                        else:
                            hi_amt = mid_amt

                    if best_amt > 0 and best_obj_try < best_obj - 1e-12:
                        a[i_star] -= best_amt
                        a[j] += best_amt
                        a = np.maximum(a, 0)
                        a = a / a.sum()
                        best_obj = best_obj_try
                        best_a = a.copy()
                        improved = True
                        break

            if improved:
                no_improve = 0
                break

        if not improved:
            no_improve += 1
            if no_improve > 500:
                # Escape: random perturbation
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i != j and a[i] > 1e-6:
                    amt = a[i] * 0.08
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                no_improve = 0

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    tri = 1.0 / (1.0 + 0.3 * ((x - n / 2) / (n / 5)) ** 2)
    tri /= tri.sum()
    inits['bell'] = tri

    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.25 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g

    bi = np.zeros(n)
    edge = max(2, n // 5)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / max(n - 1, 1)))
    cos_w = np.maximum(cos_w, 1e-6)
    cos_w /= cos_w.sum()
    inits['raised_cos'] = cos_w

    inits['rand_a'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['rand_b'] = rng.dirichlet(np.ones(n) * 2.0)

    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        if len(parent) < n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search combining enhanced LP optimization with binary M-search,
    peak-directed coordinate descent, and smoothed gradient polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            parent_score = evaluate_sequence(p.tolist())
            print(f"Parent: n={len(p)}, score={parent_score:.6f}", flush=True)
            if np.isfinite(parent_score):
                best_score = parent_score
                best_sequence = p.copy()
    except Exception as e:
        print(f"No parent: {e}", flush=True)

    # Phase 1: Quick gradient scan over lengths
    print("=== Phase 1: Quick Scan ===", flush=True)
    t_p1 = time.monotonic()
    p1_budget = 180.0
    lengths = [384, 512, 768]

    scan_results = []

    for n in lengths:
        if time.monotonic() > deadline - 500:
            break

        inits = generate_inits(n, parent, rng)
        names = list(inits.keys())
        priority = ['parent', 'parent_ds', 'parent_up', 'bell', 'gauss',
                    'uniform', 'raised_cos', 'rand_a']
        names = [nm for nm in priority if nm in names] + \
                [nm for nm in names if nm not in priority]
        names = names[:5]

        time_rem = p1_budget - (time.monotonic() - t_p1)
        per_init = max(8.0, time_rem / max(1, len(names)))

        for name in names:
            if time.monotonic() > deadline - 480:
                break
            t_alloc = min(per_init, 35.0)
            if t_alloc < 5:
                break

            a0 = inits[name]
            a_opt, _ = gradient_phase(a0, t_alloc, beta_start=40,
                                      beta_end=200, lr=0.012,
                                      momentum=0.92,
                                      seed=rng.integers(0, 100000))
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, {name}: {score:.6f}", flush=True)

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}", flush=True)

    # Phase 2: Deep LP optimization
    print("=== Phase 2: Deep LP ===", flush=True)

    # Build candidate list: best from scan + parent
    candidates = []
    if best_sequence is not None:
        candidates.append((len(best_sequence), best_sequence.copy()))

    for score, n, name, a_opt in scan_results:
        if len(candidates) >= 3:
            break
        # Avoid duplicate lengths
        if any(c[0] == n for c in candidates):
            continue
        candidates.append((n, a_opt))

    # Ensure we have at least one candidate
    if not candidates:
        candidates.append((384, np.ones(384) / 384))

    remaining = deadline - time.monotonic() - 400.0
    per_cand = max(40.0, remaining / max(1, len(candidates)))

    for ci, (n_c, a_c) in enumerate(candidates):
        if time.monotonic() > deadline - 380:
            break
        t_alloc = min(per_cand, 300.0)
        if t_alloc < 15:
            break

        print(f"  LP candidate {ci}: n={n_c}", flush=True)

        # LP phase (60% of allocation)
        t_lp = t_alloc * 0.55
        a_lp, obj_lp = lp_phase(a_c, t_lp, seed=rng.integers(0, 999999),
                                use_binary_M=True)
        score_lp = float(evaluate_sequence(a_lp.tolist()))
        if np.isfinite(score_lp) and score_lp < best_score:
            best_score = score_lp
            best_sequence = a_lp.copy()
            print(f"    LP: {score_lp:.6f}", flush=True)

        # Gradient polish (25%)
        t_g = min(t_alloc * 0.25, 50.0)
        if t_g > 5 and time.monotonic() < deadline - 200:
            a_g, _ = gradient_phase(a_lp, t_g, beta_start=200,
                                    beta_end=1000, lr=0.004,
                                    momentum=0.95,
                                    seed=rng.integers(0, 999999))
            score_g = float(evaluate_sequence(a_g.tolist()))
            if np.isfinite(score_g) and score_g < best_score:
                best_score = score_g
                best_sequence = a_g.copy()
                print(f"    Grad: {score_g:.6f}", flush=True)

        # Second LP from improved point (20%)
        t_lp2 = min(t_alloc * 0.20, 40.0)
        if t_lp2 > 5 and time.monotonic() < deadline - 180:
            start_a = best_sequence if best_sequence is not None else a_lp
            a_lp2, _ = lp_phase(start_a, t_lp2,
                                seed=rng.integers(0, 999999),
                                use_binary_M=True)
            score_lp2 = float(evaluate_sequence(a_lp2.tolist()))
            if np.isfinite(score_lp2) and score_lp2 < best_score:
                best_score = score_lp2
                best_sequence = a_lp2.copy()
                print(f"    LP2: {score_lp2:.6f}", flush=True)

    # Phase 3: Peak coordinate descent
    print("=== Phase 3: Peak Descent ===", flush=True)
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 80.0
        t_pd = min(remaining * 0.5, 150.0)
        if t_pd > 10:
            a_pd, _ = peak_coordinate_descent(best_sequence, t_pd, seed=424242)
            score_pd = float(evaluate_sequence(a_pd.tolist()))
            if np.isfinite(score_pd) and score_pd < best_score:
                best_score = score_pd
                best_sequence = a_pd.copy()
                print(f"  PeakD: {score_pd:.6f}", flush=True)

    # Phase 4: Final polish
    print("=== Phase 4: Final Polish ===", flush=True)
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10.0

        # High-beta gradient
        t_f1 = min(remaining * 0.3, 50.0)
        if t_f1 > 5:
            a_f1, _ = gradient_phase(best_sequence, t_f1,
                                     beta_start=500, beta_end=3000,
                                     lr=0.001, momentum=0.97, seed=111111)
            score_f1 = float(evaluate_sequence(a_f1.tolist()))
            if np.isfinite(score_f1) and score_f1 < best_score:
                best_score = score_f1
                best_sequence = a_f1.copy()
                print(f"  FineGrad: {score_f1:.6f}", flush=True)

        # Final LP
        remaining = deadline - time.monotonic() - 5.0
        t_f2 = min(remaining * 0.35, 40.0)
        if t_f2 > 5:
            a_f2, _ = lp_phase(best_sequence, t_f2, seed=222222,
                               use_binary_M=True)
            score_f2 = float(evaluate_sequence(a_f2.tolist()))
            if np.isfinite(score_f2) and score_f2 < best_score:
                best_score = score_f2
                best_sequence = a_f2.copy()
                print(f"  FinalLP: {score_f2:.6f}", flush=True)

        # Final peak descent
        remaining = deadline - time.monotonic() - 2.0
        if remaining > 5:
            a_f3, _ = peak_coordinate_descent(best_sequence, remaining,
                                              seed=333333)
            score_f3 = float(evaluate_sequence(a_f3.tolist()))
            if np.isfinite(score_f3) and score_f3 < best_score:
                best_score = score_f3
                best_sequence = a_f3.copy()
                print(f"  FinalPeak: {score_f3:.6f}", flush=True)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(384) / 384

    # Ensure non-negative and valid
    best_sequence = np.maximum(best_sequence, 0.0)
    if best_sequence.sum() < 0.01:
        best_sequence = np.ones(384) / 384

    # Final check
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: {best_score:.6f} (n={len(best_sequence)})", flush=True)
    return [float(x) for x in best_sequence]
