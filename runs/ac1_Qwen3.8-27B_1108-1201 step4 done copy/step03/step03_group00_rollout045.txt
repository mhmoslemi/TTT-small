"""
Multi-strategy optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines LP-based Frank-Wolfe, l2-norm QP directions, peak shaving,
and adaptive multi-length search with fine polishing.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def l2_direction(f):
    """Compute direction to reduce l2 norm of convolution.
    Returns g = projection of (f - grad) onto simplex, where grad
    points in the direction of increasing l2 norm."""
    n = len(f)
    conv = fast_convolve(f)
    # Gradient of sum(conv^2) w.r.t. f:
    # d/df sum_k conv[k]^2 = 2 * sum_k conv[k] * shift_k(f)
    grad = np.zeros(n)
    for k in range(len(conv)):

        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * conv[k] * f[k - idx]
    # Move opposite to gradient
    g = project_simplex(f - 0.01 * grad)
    return g

    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    m_exp = 1.0
    use_l2 = False

    while time.monotonic() - t0 < time_limit:
        # Occasionally use l2 direction
        use_l2 = (no_improve > 20 and no_improve % 5 == 0)

        if use_l2:
            g = l2_direction(f)
            g_sum = g.sum()
            if g_sum < 1e-15:
                use_l2 = False
                g = None
        else:
            g = solve_lp_direction(f, M * m_exp)
            if g is None:
                m_exp *= 1.08
                if m_exp > 4.0:
                    rng = np.random.default_rng(int(time.monotonic() * 1000))
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                    conv = fast_convolve(f)
                    M = conv.max()
                    m_exp = 1.0
                continue
            g_sum = g.sum()
            if g_sum < 1e-15:
                m_exp *= 1.05
                continue
            g = g / g_sum

        if use_l2 and g is not None:
            g = g / g.sum()

        # Fixed point check
        if np.max(np.abs(g - f)) < 5e-10:
            no_improve += 1
            if no_improve > 50:
                break
            m_exp = min(m_exp * 1.04, 3.0)
            if not use_l2:
                continue

        # Line search: coarse then fine
        best_val = M
        best_alpha = 0.0

        alphas = np.array([0.03, 0.05, 0.08, 0.12, 0.17, 0.25, 0.35, 0.5, 0.65, 0.8, 0.95, 1.0])
        for alpha in alphas:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine bisection
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(7):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-9:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            use_l2 = False

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and no_improve == 0:
                    print(f"    FW iter: obj={best_obj:.10f}, alpha={best_alpha:.4f}", end="\r")
            else:
                no_improve += 1
            if not use_l2:
                m_exp = 1.0
        else:
            no_improve += 1
            if not use_l2:
                m_exp = min(m_exp * 1.06, 3.5)

        if no_improve > 50:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 1000000)
            scale = 0.002 + 0.004 * min(1.0, no_improve / 150.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def peak_shave_polish(f0, time_limit):
    """Subgradient polishing targeting the top peaks with adaptive step size."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.003
    momentum = 0.92
    v = np.zeros(n)
    no_improve = 0
    top_k = max(5, n // 16)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)

        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        # Exponential weighting favoring the top
        temp = 0.0001 + 0.0005 * (1.0 - pv.min() / max(pv.max(), 1e-15))
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        # Small L2 regularization
        sg += 0.005 * f

        v = momentum * v - lr * sg
        f_new = project_simplex(f + v)
        new_obj = fast_convolve(f_new).max()

        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.98
            f = best_f.copy()

        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.0015)
            v = np.zeros(n)
            lr = 0.002
            no_improve = 0
            if not np.any(np.abs(f - best_f) > 1e-10):
                break

        if lr < 1e-6:
            break

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit):
    """Pairwise coordinate swaps to reduce peak convolution."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    # Find the peak location
    peak_k = int(np.argmax(conv))

    while time.monotonic() - t0 < time_limit:
        improved = False
        # Try pairwise swaps
        for _ in range(200):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or f[i] < 1e-10 or f[j] < 1e-10:
                continue
            # Try moving a small amount from i to j
            for frac in [0.1, 0.3, 0.5]:
                amt = frac * min(f[i], f[j])
                f_try = f.copy()
                f_try[i] -= amt
                f_try[j] += amt
                if f_try[i] < -1e-15 or f_try[j] < -1e-15:
                    continue
                f_try = np.maximum(f_try, 0)
                new_conv = fast_convolve(f_try)
                new_obj = new_conv.max()
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_f = f_try.copy()
                    f = f_try
                    improved = True
                    break
        if not improved:
            # Try larger perturbations
            for _ in range(10):
                i = rng.integers(0, n)
                if f[i] < 1e-8:
                    continue
                # Distribute f[i] to random neighbors
                targets = rng.choice(n, size=5, replace=False)
                f_try = f.copy()
                f_try[i] = 0
                for t in targets:
                    f_try[t] += f[i] / 5.0
                f_try = np.maximum(f_try, 0)
                f_try /= f_try.sum()
                new_obj = fast_convolve(f_try).max()
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_f = f_try.copy()
                    f = f_try
                    improved = True

        if not improved:
            # Check if we're stuck
            conv_now = fast_convolve(f)
            if conv_now.max() < best_obj + 1e-12:
                if time.monotonic() - t0 > 30:
                    break

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    x = np.arange(n, dtype=np.float64)
    # Edge-concentrated
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 0.5)
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c1 / c1.sum())

    # Two-peak
    p1 = np.exp(-0.5 * ((x - 0.2 * n) / (0.1 * n)) ** 2)
    p2 = np.exp(-0.5 * ((x - 0.8 * n) / (0.1 * n)) ** 2)
    two_peak = p1 + p2
    inits.append(two_peak / two_peak.sum())

    # Periodic
    per = 1.0 + 0.3 * np.cos(2 * np.pi * x * 3 / n)
    inits.append(per / per.sum())

    if parent is not None:
        if len(parent) == n:
            inits.append(parent.copy())
        else:
            px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            px = np.maximum(px, 0)
            ps = px.sum()
            if ps > 1e-10:
                inits.append(px / ps)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: explore lengths, use FW-LP + peak shaving + coordinate
    descent for polishing. Returns best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.6f}")
    except Exception:
        pass

    # === Phase 1: Explore a few lengths with diverse inits ===
    print("=== Phase 1: Multi-length Exploration ===")
    n_parent = len(parent) if parent is not None else 384

    # Focus lengths: parent length and a few nearby
    candidate_lengths = [n_parent]
    for delta in [2, -2, 4, -4, 6, -6, 8, 16, -16]:
        nl = n_parent + delta
        if 64 <= nl <= 1024 and nl not in candidate_lengths:
            candidate_lengths.append(nl)
    candidate_lengths = candidate_lengths[:8]

    # Allocate time: most to parent length
    phase1_time = min(600, (deadline - time.monotonic()) * 0.4)
    per_length_time = phase1_time / len(candidate_lengths)
    per_length_time = max(per_length_time, 30)

    for n in candidate_lengths:
        if time.monotonic() > deadline - 300:
            break
        t_alloc = min(per_length_time, deadline - time.monotonic() - 250)
        if t_alloc < 15:
            break

        inits = generate_inits(n, parent, rng)
        # Limit inits to save time
        max_inits = 3 if n == n_parent else 2
        inits = inits[:max_inits]

        t_per_init = t_alloc / len(inits)
        for i, f_init in enumerate(inits):
            rem = deadline - time.monotonic() - 200
            if rem < 10:
                break
            t_init = min(t_per_init, rem)
            if t_init < 10:
                break

            # FW-LP optimization
            a_opt, _ = fw_lp_optimize(f_init, time_limit=t_init * 0.7)
            try_seq(a_opt, f"n{n}_fw{i}")

            # Quick peak shave
            rem2 = deadline - time.monotonic() - 150
            if rem2 > 10:
                a_opt2, _ = peak_shave_polish(a_opt, time_limit=min(rem2 * 0.5, 20))
                try_seq(a_opt2, f"n{n}_ps{i}")

    print(f"  Phase 1 best: {best_score:.6f}")

    # === Phase 2: Deep optimization on best ===
    print(f"\n=== Phase 2: Deep Optimization ===")

    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic() - 100

        # Multiple FW-LP runs with perturbations
        perturbations = [
            ("exact", best_seq.copy()),
            ("noise_small", project_simplex(best_seq + rng.standard_normal(n_best) * 0.002)),
            ("noise_med", project_simplex(best_seq + rng.standard_normal(n_best) * 0.01)),
            ("mirror_blend", project_simplex(0.6 * best_seq + 0.4 * best_seq[::-1])),
            ("edge_push", project_simplex(best_seq + 0.1 * np.concatenate([np.ones(n_best//4), np.zeros(n_best//2), np.ones(n_best//4)]))),
        ]

        t_per_pert = remaining / (len(perturbations) + 3)

        for p_idx, (p_name, f_init) in enumerate(perturbations):
            rem = deadline - time.monotonic() - 80
            if rem < 15:
                break
            t_alloc = min(t_per_pert * 1.5, rem)
            a_opt, _ = fw_lp_optimize(f_init, time_limit=t_alloc)
            try_seq(a_opt, f"deep_{p_name}")

            # Peak shave
            rem2 = deadline - time.monotonic() - 60
            if rem2 > 10:
                a_opt2, _ = peak_shave_polish(a_opt, time_limit=min(rem2 * 0.4, 30))
                try_seq(a_opt2, f"deep_{p_name}_ps")

    # === Phase 3: Fine polishing ===
    print(f"\n=== Phase 3: Fine Polishing ===")
    print(f"  Current best: {best_score:.6f}")

    if best_seq is not None:
        n_best = len(best_seq)

        # Peak shave with very small steps
        rem = deadline - time.monotonic() - 50
        if rem > 15:
            a_opt, _ = peak_shave_polish(best_seq, time_limit=min(rem, 60))
            try_seq(a_opt, "fine_ps")

        # Coordinate descent
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            a_opt, _ = coordinate_descent_polish(best_seq, time_limit=min(rem, 40))
            try_seq(a_opt, "coord")

        # Final FW-LP
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=min(rem, 30))
            try_seq(a_opt, "fw_final")

        # Tiny perturbations + quick polish
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            for _ in range(4):
                rem = deadline - time.monotonic() - 5
                if rem < 5:
                    break
                noise = rng.standard_normal(n_best) * 0.0008
                f_pert = project_simplex(best_seq + noise)
                a_opt, _ = peak_shave_polish(f_pert, time_limit=min(rem, 15))
                try_seq(a_opt, "micro")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
