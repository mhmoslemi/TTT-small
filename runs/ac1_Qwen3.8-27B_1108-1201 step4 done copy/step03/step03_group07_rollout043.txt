def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    return np.real(ifft(A * A, N))[:size]


def smooth_max_val(x, beta):
    """Compute smooth max value."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -500, 500)
    log_sum = np.log(np.exp(shifted).sum())
    return x_max + log_sum / beta


def smooth_max_weights(x, beta):
    """Compute softmax weights for smooth max."""

def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        conv = fast_convolve(a)
        sm = smooth_max_val(conv, beta)

            best_a = a.copy()

        N = 1 << (n + len(conv) - 1).bit_length()
        A = fft(a, N)
        W = fft(w, N)
        grad_a = 2.0 * np.real(ifft(np.conj(A) * W, N))[:n]

                # Upsample
                pp_us = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(pp)), pp)
                pp_us = np.maximum(pp_us, 0)
                ps2 = pp_us.sum()
                if ps2 > 1e-10:
                    inits['parent'] = pp_us / ps2
    
    return inits


def peak_sharing_optimize(a0, n_active=5, n_iters=200):
    """
    Optimize by encouraging the top peaks to share the same value.
    This is a heuristic that works well at the optimum where multiple
    peaks are at the same level.
    """
    n = len(a0)
    a = a0.copy()
    
    for it in range(n_iters):
        conv = fast_convolve(a)
        cmax = conv.max()
        
        # Find top n_active peaks
        top_idx = np.argsort(conv)[::-1][:n_active]
        top_vals = conv[top_idx]
        target = top_vals.mean()
        
        # Penalty for spread among top peaks
        # Objective: cmax + lambda * variance of top peaks
        # Use smooth approximation
        beta = 50.0
        sm = smooth_max_val(conv, beta)
        
        # Gradient
        w = smooth_max_weights(conv, beta)
        N = 1 << (n + len(conv) - 1).bit_length()
        A = fft(a, N)
        W = fft(w, N)
        grad_a = 2.0 * np.real(ifft(np.conj(A) * W, N))[:n]
        grad_proj = grad_a - np.mean(grad_a)
        
        # Step
        lr = 0.005 * (1.0 + 0.5 * np.cos(it / n_iters * np.pi))
        a_new = a - lr * grad_proj
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        # Accept if true objective improved
        conv_new = fast_convolve(a_new)
        if conv_new.max() <= cmax + 1e-10:
            a = a_new
    
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search with corrected gradients:
    1. Multi-start L-BFGS-B with adaptive beta
    2. Subgradient polishing with true max
    3. Peak-sharing optimization
    4. Random perturbation + re-optimization
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print("Parent available: length=%d, sum=%.4f" % (len(p), p.sum()))
    except Exception:
        pass
    
    def score_and_update(a, verbose=True):
        nonlocal best_sequence, best_score
        if a is None:
            return float('inf')
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        seq = a.tolist()
        score = float(evaluate_sequence(seq))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            if verbose:
                print("  New best: score=%.6f, n=%d (t=%.0fs)" % (score, len(a), time.monotonic() - t_start))
        return score if np.isfinite(score) else float('inf')
    
    # Phase 1: Quick multi-start L-BFGS-B
    print("=== Phase 1: Multi-start L-BFGS-B ===")
    
    n = 256
    inits = get_initializations(n, parent, rng)
    
    # Priority order
    priority = ['parent', 'parent_ds', 'flat_spec', 'ramp', 'sine', 'uniform', 'random']
    init_names = [nm for nm in priority if nm in inits]
    init_names += [nm for nm in inits.keys() if nm not in priority]
    
    phase1_time = min(400.0, (deadline - t_start) * 0.3)
    time_per_init = max(10.0, phase1_time / len(init_names))
    
    for name in init_names:
        if time.monotonic() > t_start + phase1_time:
            break
        a0 = inits[name]
        # Adaptive beta: start high, decrease
        a_cur = a0.copy()
        for beta in [200.0, 100.0, 50.0, 25.0]:
            if time.monotonic() > t_start + phase1_time:
                break
            maxiter = int(min(300, time_per_init / 4 * 30))
            a_opt, _ = opt_lbfgsb(a_cur, beta=beta, maxiter=maxiter)
            score = score_and_update(a_opt, verbose=(beta == 25.0))
            a_cur = a_opt
    
    print("Phase 1 done. Best: %.6f" % best_score)
    
    # Phase 2: Deep optimization on best
    print("=== Phase 2: Deep Optimization ===")
    if best_sequence is not None:
        best_n = len(best_sequence)
        phase2_deadline = deadline - 200.0
        remaining = phase2_deadline - time.monotonic()
        
        # Try multiple betas with many iterations
        for beta in [500.0, 300.0, 200.0, 100.0, 50.0, 25.0]:
            if time.monotonic() > phase2_deadline:
                break
            t_alloc = remaining / 6
            maxiter = int(min(1000, max(100, t_alloc * 20)))
            a_opt, _ = opt_lbfgsb(best_sequence, beta=beta, maxiter=maxiter)
            score = score_and_update(a_opt)
        
        # Subgradient polishing
        if time.monotonic() < phase2_deadline - 30:
            n_iters = int(min(500, (phase2_deadline - time.monotonic()) / 0.05))
            a_pol = subgrad_polish(best_sequence, n_iters=n_iters, lr_init=0.0005)
            score = score_and_update(a_pol)
        
        # Peak sharing
        if time.monotonic() < phase2_deadline - 30:
            a_ps = peak_sharing_optimize(best_sequence, n_active=5, n_iters=200)
            score = score_and_update(a_ps)
            if score < best_score:
                # Re-optimize from here
                a_opt, _ = opt_lbfgsb(best_sequence, beta=100.0, maxiter=200)
                score_and_update(a_opt)
    
    print("Phase 2 done. Best: %.6f" % best_score)
    
    # Phase 3: Try different lengths
    print("=== Phase 3: Different Lengths ===")
    if best_sequence is not None:
        for alt_n in [128, 384, 512]:
            if time.monotonic() > deadline - 60:
                break
            inits_alt = get_initializations(alt_n, best_sequence, rng)
            a0 = inits_alt.get('parent', inits_alt.get('flat_spec', inits_alt.get('uniform', inits_alt['ramp'])))
            for beta in [100.0, 50.0]:
                if time.monotonic() > deadline - 40:
                    break
                maxiter = int(min(300, (deadline - time.monotonic() - 20) * 5))
                a_opt, _ = opt_lbfgsb(a0, beta=beta, maxiter=maxiter)
                score = score_and_update(a_opt)
    
    print("Phase 3 done. Best: %.6f" % best_score)
    
    # Phase 4: Final polishing
    print("=== Phase 4: Final Polishing ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10
        
        # Multiple random perturbations + re-optimize
        n_best = len(best_sequence)
        n_pert = min(10, int(remaining / 30))
        for i in range(n_pert):
            if time.monotonic() > deadline - 15:
                break
            scale = 0.002 + 0.003 * rng.uniform()
            noise = rng.standard_normal(n_best) * scale
            a_pert = project_simplex(best_sequence + noise)
            maxiter = int(min(300, max(50, (deadline - time.monotonic() - 10) * 5)))
            a_opt, _ = opt_lbfgsb(a_pert, beta=200.0, maxiter=maxiter)
            score = score_and_update(a_opt)
        
        # Final subgradient polish
        if time.monotonic() < deadline - 10:
            n_iters = int((deadline - time.monotonic() - 10) / 0.02)
            n_iters = min(n_iters, 1000)
            a_pol = subgrad_polish(best_sequence, n_iters=n_iters, lr_init=0.0003)
            score = score_and_update(a_pol)
        
        # Final ultra-high beta L-BFGS-B
        if time.monotonic() < deadline - 10:
            maxiter = int((deadline - time.monotonic() - 10) * 5)
            maxiter = min(maxiter, 1000)
            a_opt, _ = opt_lbfgsb(best_sequence, beta=1000.0, maxiter=maxiter)
