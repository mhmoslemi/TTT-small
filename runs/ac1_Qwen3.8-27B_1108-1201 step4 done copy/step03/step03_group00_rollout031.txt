    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def fw_lp_optimize(f0, time_limit, verbose=False, m_exp_start=1.0):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = solve_lp(f, M * m_exp_start)
        if g is None:
            m_exp_start *= 1.2
            if m_exp_start > 10.0:
                m_exp_start = 1.0
                # Perturb and restart
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Fixed point check
        if np.max(np.abs(g - f)) < 1e-9:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp_start = min(m_exp_start * 1.1, 5.0)
            continue

        # Line search (bisection)
        best_alpha = 0.0
        best_val = M

        # Coarse scan
        for alpha in np.concatenate([[0.0], np.linspace(0.05, 1.0, 10), [1.0]]):
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine bisection
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp_start = 1.0

            if M < best_obj - 1e-13:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp_start = min(m_exp_start * 1.1, 5.0)

        if no_improve > 50:
            rng = np.random.default_rng(it * 17 + 3)
            scale = 0.002 + 0.003 * min(no_improve // 50, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp_start = 1.0
            no_improve = 0

    return best_f, best_obj


def coord_descent(f0, time_limit, max_sweeps=500):
    """Coordinate descent on the simplex to polish the solution.
    For each coordinate j, find the optimal value by minimizing max(conv).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    sweep = 0

    while time.monotonic() - t0 < time_limit and sweep < max_sweeps:
        sweep += 1
        improved = False

        for j in range(n):
            # Determine how conv depends on f[j]
            # conv[k] = sum_i f[i]*f[k-i]
            # f[j] appears in conv[k] for k in [j, j+n-1] (clamped to [0, 2n-2])
            # Specifically: conv[j+i] includes term f[j]*f[i] for i in [0, n-1]
            # If j+i == 2j (i.e., i == j), the term is f[j]^2
            
            # To optimize f[j] while maintaining sum=1, we'll:
            # 1. Temporarily set f[j] = 0
            # 2. Compute the "base" convolution
            # 3. Find the optimal f[j] given the base
            # 4. Redistribute the sum difference uniformly (or to a specific coordinate)
            
            # Actually, a cleaner approach: since the problem is homogeneous of degree 0 in the
            # normalized space, let's work slightly differently.
            # Fix all coordinates except j. The objective max(conv) as a function of f[j]
            # (with sum constraint maintained by adjusting a "reference" coordinate) is:
            # Let f_ref be the coordinate we adjust to maintain sum.
            # f[ref] = 1 - sum_{i != j, i != ref} f[i] - f[j]
            
            # This is getting complex. Let me use a simpler approach:
            # Just do a small coordinate step and check if it improves.
            
            aj = f[j]
            if aj < 1e-15:
                continue
                
            # Try reducing f[j] by a factor and redistributing
            for frac in [0.5, 0.7, 0.9]:
                delta = aj * (1 - frac)
                if delta < 1e-15:
                    continue
                f_new = f.copy()
                f_new[j] = aj * frac
                # Redistribute delta to all other coordinates proportionally
                others = f_new.copy()
                others[j] = 0
                os = others.sum()
                if os < 1e-15:
                    f_new = project_simplex(f_new)
                else:
                    f_new += delta * others / os
                f_new = np.maximum(f_new, 0)
                f_new = f_new / f_new.sum()
                
                conv_new = fast_convolve(f_new)
                if conv_new.max() < conv.max() - 1e-14:
                    f = f_new
                    conv = conv_new
                    improved = True
                    if conv.max() < best_obj - 1e-13:
                        best_obj = conv.max()
                        best_f = f.copy()
                    break

        if not improved:
            # Try increasing coordinates that are currently zero or near-zero
            zero_mask = f < 1e-8
            if zero_mask.any():
                for j in np.where(zero_mask)[0]:
                    # Try adding a small amount to f[j] and reducing others
                    for frac in [0.001, 0.005, 0.01]:
                        f_new = f.copy()
                        # Take from the largest coordinate
                        max_idx = np.argmax(f_new)
                        if f_new[max_idx] < frac + 1e-10:
                            continue
                        f_new[j] = frac
                        f_new[max_idx] -= frac
                        f_new = np.maximum(f_new, 0)
                        f_new = f_new / f_new.sum()
                        conv_new = fast_convolve(f_new)
                        if conv_new.max() < conv.max() - 1e-14:
                            f = f_new
                            conv = conv_new
                            improved = True
                            break
            
            if not improved:
                break

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n, emphasizing asymmetry."""
    inits = []
    
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Edge-concentrated (symmetric)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    
    # Asymmetric: more mass on the left
    asym_left = np.exp(-x / (n * 0.3))
    inits.append(asym_left / asym_left.sum())
    
    # Asymmetric: more mass on the right
    asym_right = np.exp(-(n - 1 - x) / (n * 0.3))
    inits.append(asym_right / asym_right.sum())
    
    # Two-peak asymmetric
    peak1 = np.exp(-((x - 0.15 * n) / (0.1 * n)) ** 2)
    peak2 = np.exp(-((x - 0.75 * n) / (0.15 * n)) ** 2)
    two_peak = 0.6 * peak1 + 0.4 * peak2
    inits.append(two_peak / two_peak.sum())
    
    # Cosine ramp (asymmetric)
    cos_ramp = np.abs(np.cos(np.pi * x / (2 * n))) + 0.05
    inits.append(cos_ramp / cos_ramp.sum())
    
    # Power law (asymmetric)
    power_law = 1.0 / (x + 1.0) ** 0.5
    inits.append(power_law / power_law.sum())
    
    # Sawtooth
    saw = np.abs(x - n / 2) / (n / 2) + 0.1
    inits.append(saw / saw.sum())
    
    # Random
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    # Random with edge bias
    w = 0.5 + 0.5 * np.minimum(x, n - 1 - x) / (n / 4 + 1)
    inits.append(rng.dirichlet(w))
    
    # Parent (resized)
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
        # Also try flipped parent
        pf = parent[::-1].astype(np.float64)
        pfx = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(pf)), pf)
        pfx = np.maximum(pfx, 0)
        pfs = pfx.sum()
        if pfs > 1e-10:
            inits.append(pfx / pfs)
    
    return inits

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:
1. LP-based Frank-Wolfe with adaptive constraint relaxation for global moves
2. Smooth gradient descent (log-sum-exp approximation) for refined optimization

Key insight: optimal sequences are asymmetric, with more mass at edges to

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def fw_lp_optimize(f0, time_limit, m_start=1.0):
    """Frank-Wolfe with LP oracle and adaptive constraint relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = solve_lp(f, M * m_start)
        if g is None:
            m_start *= 1.25
            if m_start > 8.0:
                m_start = 1.0
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.008)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 25:
                break
            m_start = min(m_start * 1.1, 5.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = np.concatenate([[0.0], np.linspace(0.05, 1.0, 12), [1.0]])
        for alpha in alphas:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Bisection refinement
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.08)
            hi = min(1.0, best_alpha + 0.08)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_start = 1.0

            if M < best_obj - 1e-13:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_start = min(m_start * 1.1, 5.0)

        if no_improve > 40:
            rng = np.random.default_rng(it * 17 + 3)
            scale = 0.002 + 0.002 * min(no_improve // 40, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_start = 1.0
            no_improve = 0

    return best_f, best_obj


def smooth_gradient_optimize(f0, time_limit, t_init=0.005, t_decay=0.9995,
                             lr_init=0.15, mom_init=0.85, max_iters=100000):
    """Smooth optimization using log-sum-exp approximation of max(conv)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    t = t_init
    lr = lr_init
    mom = mom_init

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute softmax weights over convolution values
        scaled = conv / t
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws

        # Compute gradient: grad[j] = sum_k w[k] * d(conv[k])/d(f[j])
        # d(conv[k])/d(f[j]) = f[k-j] if 0<=k-j<n (plus extra f[j] if k=2j)
        grad = np.zeros(n)
        for k in range(len(conv)):
            if w[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            grad[idx] += w[k] * f[k - idx]

        # Heavy-ball momentum update
        v = mom * v - lr * grad
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj = conv_new.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        conv = conv_new
        t *= t_decay
        lr *= 0.99995
        mom = min(mom + 0.00005, 0.98)

        if no_improve > 300:
            rng = np.random.default_rng(it * 7 + 1)
            scale = 0.003 + 0.002 * min(no_improve // 300, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            t = t_init * 0.3
            lr = lr_init * 0.4
            mom = mom_init
            no_improve = 0
            conv = fast_convolve(f)

    return best_f, best_obj


def coord_descent_polish(f0, time_limit, max_sweeps=200):
    """Coordinate descent: try adjusting each coordinate to reduce max(conv)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    sweep = 0
    stagnant = 0

    while time.monotonic() - t0 < time_limit and sweep < max_sweeps:
        sweep += 1
        improved = False

        for j in range(n):
            aj = f[j]
            if aj < 1e-12:
                continue

            # Try reducing f[j] and redistributing
            best_local = conv.max()
            best_local_f = None

            for frac in [0.3, 0.5, 0.7, 0.85]:
                trial = f.copy()
                trial[j] = aj * frac
                delta = aj - trial[j]
                # Redistribute to others proportionally
                others_mask = np.ones(n, dtype=bool)
                others_mask[j] = False
                others_sum = trial[others_mask].sum()
                if others_sum < 1e-15:
                    continue
                trial[others_mask] += delta * trial[others_mask] / others_sum
                trial = np.maximum(trial, 0.0)
                trial = trial / trial.sum()

                c_new = fast_convolve(trial)
                if c_new.max() < best_local - 1e-14:
                    best_local = c_new.max()
                    best_local_f = trial.copy()

            # Try increasing f[j] (take from others)
            if best_local_f is None and aj < 0.5:
                for frac in [1.3, 1.5, 1.8]:
                    trial = f.copy()
                    trial[j] = aj * frac
                    delta = trial[j] - aj
                    others_mask = np.ones(n, dtype=bool)
                    others_mask[j] = False
                    others_sum = trial[others_mask].sum()
                    if others_sum < delta + 1e-15:
                        continue
                    trial[others_mask] -= delta * trial[others_mask] / others_sum
                    trial = np.maximum(trial, 0.0)
                    if trial.sum() < 1e-15:
                        continue
                    trial = trial / trial.sum()

                    c_new = fast_convolve(trial)
                    if c_new.max() < best_local - 1e-14:
                        best_local = c_new.max()
                        best_local_f = trial.copy()

            if best_local_f is not None:
                f = best_local_f
                conv = fast_convolve(f)
                improved = True
                if conv.max() < best_obj - 1e-13:
                    best_obj = conv.max()
                    best_f = f.copy()

        if not improved:
            stagnant += 1
            if stagnant > 3:
                break
        else:
            stagnant = 0

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations emphasizing asymmetry."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Edge-concentrated symmetric
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Asymmetric: exponential decay from left
    inits.append(np.exp(-x / (n * 0.25)) / np.exp(-x / (n * 0.25)).sum())

    # Asymmetric: exponential decay from right
    inits.append(np.exp(-(n - 1 - x) / (n * 0.25)) /
                 np.exp(-(n - 1 - x) / (n * 0.25)).sum())

    # Two-peak asymmetric (stronger left peak)
    p1 = np.exp(-((x - 0.12 * n) / (0.08 * n)) ** 2)
    p2 = np.exp(-((x - 0.72 * n) / (0.12 * n)) ** 2)
    two_peak = 0.55 * p1 + 0.45 * p2
    inits.append(two_peak / two_peak.sum())

    # Cosine asymmetric ramp
    cr = np.abs(np.cos(np.pi * x / (2.0 * n))) + 0.03
    inits.append(cr / cr.sum())

    # Power law
    pw = 1.0 / (x + 1.0) ** 0.4
    inits.append(pw / pw.sum())

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    # Edge-biased random
    w = 0.3 + 0.7 * (np.minimum(x, n - 1 - x) / (n / 4.0 + 1.0)) ** 0.5
    inits.append(rng.dirichlet(w))

    # Concentrated center (to contrast with edge strategies)
    gc = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(gc / gc.sum())

    # Parent resized
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
        # Flipped parent
        pf = parent[::-1].astype(np.float64)
        pfx = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(pf)), pf)
        pfx = np.maximum(pfx, 0)
        pfs = pfx.sum()
        if pfs > 1e-10:
            inits.append(pfx / pfs)

    return inits

                    a_opt if 'a_opt' in dir() else best_seq,
                    time_limit=min(rem, 25), t_init=0.001, lr_init=0.03)
                record(a_opt, f"final_smooth_{round_i}")

            # Tiny CD
            rem = deadline - time.monotonic() - 6
            if rem > 5:
                a_opt, _ = coord_descent_polish(
                    a_opt if 'a_opt' in dir() else best_seq, time_limit=min(rem, 15))
                record(a_opt, f"final_cd_{round_i}")

    # Fallback
    if best_seq is None:
        print("  FALLBACK: using uniform sequence")
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
