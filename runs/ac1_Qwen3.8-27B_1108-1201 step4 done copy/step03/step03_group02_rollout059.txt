"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using accelerated LP-based iteration
with CVXPY/PDLP solver, adaptive step sizes, and multi-phase polishing.
Key improvements: faster LP solving, line-search step sizes, frequency-domain

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a, b)[k]."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A

def solve_lp_min_peak_cvx(a, M, cache_matrix=None):
    """Solve: min M' s.t. sum(b)=1, b>=0, conv(a,b)[k] <= M' for all k.
    Returns (b, M') or (None, None)."""
    n = len(a)
    if cache_matrix is None:
        cache_matrix = build_conv_matrix(a)
    A = cache_matrix
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    objective = cp.Minimize(M_var)
    constraints = [
        cp.sum(b_var) == 1,
        A @ b_var <= M_var
    ]
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.PDLP)
        if b_var.value is not None:
            b = np.maximum(b_var.value, 0)
            b_sum = b.sum()
            if b_sum > 1e-15:
                b = b / b_sum
            return b, M_var.value
    except Exception:
        pass
    
    # Fallback to scipy
    from scipy.optimize import linprog
    num_rows = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    A_ub = np.zeros((num_rows, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(num_rows)
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        b = np.maximum(res.x[:n], 0)
        b_sum = b.sum()
        if b_sum > 1e-15:
            b = b / b_sum
        return b, res.x[n]
    return None, None

def solve_lp_max_sum_cvx(a, M, cache_matrix=None):
    """Solve: max sum(b) s.t. b>=0, conv(a,b)[k] <= M for all k."""
    n = len(a)
    if cache_matrix is None:
        cache_matrix = build_conv_matrix(a)
    A = cache_matrix
    
    b_var = cp.Variable(n, nonneg=True)
    objective = cp.Maximize(cp.sum(b_var))
    constraints = [A @ b_var <= M]
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.PDLP)
        if b_var.value is not None:
            b = np.maximum(b_var.value, 0)
            b_sum = b.sum()
            if b_sum > 1e-15:
                b = b / b_sum
            return b
    except Exception:
        pass
    
    # Fallback
    from scipy.optimize import linprog
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M),

    if res.status == 0:
        b = np.maximum(res.x, 0)
        b_sum = b.sum()
        if b_sum > 1e-15:

def optimize_soft_grad(a0, time_limit, lr_init=0.01, momentum=0.9, 
                       t_init=20.0, t_growth=1.05, t_max=200.0, max_iters=5000):
    """Projected gradient with momentum on soft-max smoothed objective."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)
    t0 = time.monotonic()
    lr = lr_init
    t_cur = t_init
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        _, grad = softmax_obj_and_grad(a, t_cur)
        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)
        
        conv = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            a = a_new  # Still move, just don't update best
        
        if no_improve > 300:
            lr = max(0.001, lr * 0.5)
            v *= 0.5
            no_improve = 0
        
        if it % 500 == 0 and t_cur < t_max:
            t_cur *= t_growth
    
    return best_a, best_obj

def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
        else:
            pp = np.interp(np.linspace(0, 1, n),
                          np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps
    
    for freq in [1, 2, 3, 5]:
        a = 1.0 + 0.8 * np.cos(2.0 * np.pi * freq * x / n)
        a = np.maximum(a, 0.001)
        a /= a.sum()
        inits[f'cos_{freq}'] = a
    
    # Gaussian centered
    g = np.exp(-0.5 * ((x - 0.5*n) / (0.3*n))**2)
    g /= g.sum()
    inits['gauss'] = g
    
    # Two Gaussians
    g2 = np.exp(-0.5*((x-0.25*n)/(0.15*n))**2) + np.exp(-0.5*((x-0.75*n)/(0.15*n))**2)
    g2 /= g2.sum()
    inits['two_gauss'] = g2
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Ramp
    ramp = np.arange(1, n+1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp
    
    # Triangular (peak in middle)
    tri = 1.0 - np.abs(x - n/2) / (n/2)
    tri = np.maximum(tri, 0.001)
    tri /= tri.sum()
    inits['tri'] = tri
    
    # Barker-like (alternating with smoothing)
    barker = np.ones(n)
    for i in range(1, n):
        barker[i] = 0.3 + 0.7 * (0.5 + 0.5*np.cos(2*np.pi*i*3/n))
    barker = np.maximum(barker, 0.001)
    barker /= barker.sum()
    inits['barker'] = barker
    
    return inits

def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search: LP iteration (CVXPY/PDLP) + soft-max gradient + perturbations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: len={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass
    
    def try_update(a):
        nonlocal best_sequence, best_score
        if a is None:
            return best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return best_score
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  New best: {score:.6f} (n={len(a)})")
        return score
    
    N = 512
    
    # === Phase 1: LP exploration from multiple starts ===
    print("=== Phase 1: LP Exploration ===")
    t1_start = time.monotonic()
    
    inits = generate_inits(N, parent, rng)
    init_names = list(inits.keys())
    priority = ['parent', 'uniform', 'random', 'cos_1', 'gauss', 'two_gauss', 'tri', 'barker']
    init_names = [nm for nm in priority if nm in inits] + \
                [nm for nm in init_names if nm not in priority]
    
    # Time allocation: ~700s for phase 1
    phase1_time = 700.0
    time_per_init = phase1_time / len(init_names)
    
    for name in init_names:
        remain = deadline - time.monotonic() - 600.0
        if remain < 30:
            break
        t_alloc = min(time_per_init, remain)
        if t_alloc < 15:
            break
        
        a0 = inits[name]
        print(f"  Init: {name}, alloc={t_alloc:.0f}s")
        
        # Try min_peak mode
        t_single = t_alloc * 0.6
        a_opt, obj = lp_iteration(a0, time_limit=t_single, mode='min_peak', 
                                   step_init=0.3, use_line_search=True)
        try_update(a_opt)
        
        # Try max_sum mode
        remain = deadline - time.monotonic() - 500.0
        if remain > 15:
            t_single = min(t_alloc * 0.4, remain)
            a_opt, obj = lp_iteration(a0, time_limit=t_single, mode='max_sum',
                                       step_init=0.25, use_line_search=True)
            try_update(a_opt)
    
    print(f"Phase 1 done: best={best_score:.6f}, elapsed={time.monotonic()-t1_start:.0f}s")
    
    # === Phase 2: Deep LP refinement on best ===
    print("=== Phase 2: Deep LP Refinement ===")
    t2_start = time.monotonic()
    
    if best_sequence is not None:
        phase2_time = 400.0
        phase2_deadline = deadline - 300.0
        for step_init in [0.3, 0.15, 0.5, 0.1]:
            if time.monotonic() > phase2_deadline:
                break
            t_alloc = min(phase2_time / 4.0, phase2_deadline - time.monotonic() - 5)
            if t_alloc < 10:
                break
            for method in ['min_peak', 'max_sum']:
                if time.monotonic() > phase2_deadline:
                    break
                t_single = t_alloc / 2.0
                if t_single < 5:
                    break
                print(f"  LP {method}, step={step_init}")
                a_opt, obj = lp_iteration(best_sequence, time_limit=t_single,
                                           mode=method, step_init=step_init,
                                           use_line_search=True)
                try_update(a_opt)
    
    print(f"Phase 2 done: best={best_score:.6f}, elapsed={time.monotonic()-t2_start:.0f}s")
    
    # === Phase 3: Soft-max gradient polishing ===
    print("=== Phase 3: Soft-max Gradient ===")
    t3_start = time.monotonic()
    
    if best_sequence is not None:
        phase3_deadline = deadline - 200.0
        for lr in [0.008, 0.004, 0.012]:
            if time.monotonic() > phase3_deadline:
                break
            t_alloc = min(80.0, phase3_deadline - time.monotonic() - 5)
            if t_alloc < 10:
                break
            print(f"  Soft-grad lr={lr}")
            a_opt, obj = optimize_soft_grad(best_sequence, time_limit=t_alloc,
                                             lr_init=lr, momentum=0.92,
                                             t_init=30.0)
            try_update(a_opt)
    
    print(f"Phase 3 done: best={best_score:.6f}, elapsed={time.monotonic()-t3_start:.0f}s")
    
    # === Phase 4: Perturbation + re-optimization ===
    print("=== Phase 4: Perturbations ===")
    t4_start = time.monotonic()
    
    if best_sequence is not None:
        phase4_deadline = deadline - 30.0
        n_best = len(best_sequence)
        for i, scale in enumerate([0.003, 0.01, 0.001, 0.02, 0.005]):
            if time.monotonic() > phase4_deadline:
                break
            noise = rng.standard_normal(n_best) * scale
            a_pert = project_simplex(best_sequence + noise)
            t_alloc = min(40.0, phase4_deadline - time.monotonic() - 5)
            if t_alloc < 10:
                break
            print(f"  Perturb {i}, scale={scale}")
            a_opt, obj = lp_iteration(a_pert, time_limit=t_alloc, mode='min_peak',

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = A[k, j] + a[k - j] if False else a[k - j]
    return A


def build_conv_matrix_fast(a):
    """Build convolution matrix using FFT-based approach for speed."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for j in range(n):
        # a[j] contributes to conv[k] for k = j, j+1, ..., j+n-1
        # Specifically, conv[k] gets a[j] * a[k-j] from this element
        # But wait, the matrix should encode: (A @ b)[k] = sum_j A[k,j] * b[j]
        # And conv(a,b)[k] = sum_j a[k-j] * b[j] (where valid)
        # So A[k,j] = a[k-j] if 0<=k-j<n, else 0
        k_start = j
        k_end = min(num_rows - 1, j + n - 1)
        A[k_start:k_end + 1, j] = a[0:k_end - k_start + 1]
    return A


def solve_lp_min_peak(a, M, conv_matrix, use_cvxpy=True):
    """Solve: min M' s.t. sum(b)=1, b>=0, conv(a,b)[k] <= M' for all k."""
    n = len(a)
    A = conv_matrix
    num_rows = 2 * n - 1

    if use_cvxpy:
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        objective = cp.Minimize(M_var)
        constraints = [cp.sum(b_var) == 1, A @ b_var <= M_var]
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.PDLP)
            if b_var.value is not None and not np.any(np.isnan(b_var.value)):
                b = np.maximum(b_var.value, 0)
                b_sum = b.sum()
                if b_sum > 1e-15:
                    b = b / b_sum
                    return b
        except Exception:
            pass

    # Fallback to scipy
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    A_ub = np.zeros((num_rows, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(num_rows)
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        b = np.maximum(res.x[:n], 0)
        b_sum = b.sum()
        if b_sum > 1e-15:
            b = b / b_sum
            return b
    return None


def solve_lp_max_sum(a, M, conv_matrix, use_cvxpy=True):
    """Solve: max sum(b) s.t. b>=0, conv(a,b)[k] <= M for all k."""
    n = len(a)
    A = conv_matrix
    num_rows = 2 * n - 1

    if use_cvxpy:
        b_var = cp.Variable(n, nonneg=True)
        objective = cp.Maximize(cp.sum(b_var))
        constraints = [A @ b_var <= M]
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.PDLP)
            if b_var.value is not None and not np.any(np.isnan(b_var.value)):
                b = np.maximum(b_var.value, 0)
                b_sum = b.sum()
                if b_sum > 1e-15:
                    b = b / b_sum
                    return b
        except Exception:
            pass

    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(num_rows, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.status == 0:
        b = np.maximum(res.x, 0)
        b_sum = b.sum()
        if b_sum > 1e-15:
            b = b / b_sum
            return b
    return None


def solve_qp_flatness(a, weights=None, conv_matrix=None, use_cvxpy=True):
    """Solve QP: min sum_k w_k*(conv(a,b)_k - c)^2 s.t. sum(b)=1, b>=0.
    where c = 1/(2n-1) (target flat value)."""
    n = len(a)
    A = conv_matrix
    num_rows = 2 * n - 1
    c_target = 1.0 / num_rows

    if weights is None:
        weights = np.ones(num_rows)

    if use_cvxpy:
        b_var = cp.Variable(n, nonneg=True)
        conv_vals = A @ b_var
        objective = cp.Minimize(cp.sum_squares(cp.multiply(weights, conv_vals - c_target)))
        constraints = [cp.sum(b_var) == 1]
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.CLARBAK)
            if b_var.value is not None and not np.any(np.isnan(b_var.value)):
                b = np.maximum(b_var.value, 0)
                b_sum = b.sum()
                if b_sum > 1e-15:
                    b = b / b_sum
                    return b
        except Exception:
            pass
        try:
            prob.solve(solver=cp.SCS)
            if b_var.value is not None and not np.any(np.isnan(b_var.value)):
                b = np.maximum(b_var.value, 0)
                b_sum = b.sum()
                if b_sum > 1e-15:
                    b = b / b_sum
                    return b
        except Exception:
            pass

    return None

def softmax_obj_and_grad(a, t):
    """Compute soft-max smoothed objective and its gradient."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = t * conv
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    sum_exp = exp_vals.sum()
    p = exp_vals / sum_exp

    obj = 2.0 * n * (conv.max() + (1.0 / t) * np.log(sum_exp))

    grad = np.zeros(n)
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        if k_start <= k_end:
            grad[j] = 4.0 * n * np.dot(p[k_start:k_end + 1], a[0:k_end - k_start + 1])

    return obj, grad


def optimize_soft_grad(a0, time_limit, lr_init=0.008, momentum=0.92,

    a = np.maximum(a0, 0.0)

    best_a = a.copy()

    v = np.zeros(n)
    t_start = time.monotonic()
    lr = lr_init
    t_cur = t_init
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() - t_start > time_limit:
            break

        if it % 20 == 0:
            _, grad = softmax_obj_and_grad(a, t_cur)
            v = momentum * v - lr * grad
            a_new = project_simplex(a + v)
            a = a_new

            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()

            if cur_obj < best_obj - 1e-10:

                no_improve = 0
            else:
                no_improve += 1

        if no_improve > 400:
            lr = max(0.001, lr * 0.5)
            v *= 0.3

                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps

    for freq in [1, 2, 3]:
        a = 1.0 + 0.85 * np.cos(2.0 * np.pi * freq * x / n)
        a = np.maximum(a, 0.001)
        a /= a.sum()
        inits[f'cos_{freq}'] = a

    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.25 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g

    g2 = (np.exp(-0.5 * ((x - 0.25 * n) / (0.15 * n)) ** 2) +
          np.exp(-0.5 * ((x - 0.75 * n) / (0.15 * n)) ** 2))
    g2 /= g2.sum()
    inits['two_gauss'] = g2

    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp

    tri = 1.0 - np.abs(x - n / 2) / (n / 2)
    tri = np.maximum(tri, 0.001)
    tri /= tri.sum()
    inits['tri'] = tri

    # Superposition of cosines (designed for flat convolution)
    a_flat = np.ones(n)
    for k in range(1, 6):
        a_flat += 0.3 * np.cos(2 * np.pi * k * x / n + rng.uniform(0, 2 * np.pi))
    a_flat = np.maximum(a_flat, 0.001)
    a_flat /= a_flat.sum()
    inits['multi_cos'] = a_flat

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search combining LP/QP-based iteration with soft-max gradient
    polishing and multi-restart perturbation to minimize the objective.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 35.0)

    best_sequence = None
    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: len={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass

    def try_update(a):
        nonlocal best_sequence, best_score
        if a is None or len(a) == 0:
            return best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return best_score
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return best_score
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  New best: {score:.6f} (n={len(a)})")
        return score

    N = 512

    # === Phase 1: Multi-start LP exploration ===
    print("=== Phase 1: Multi-Start LP Exploration ===")
    t1_start = time.monotonic()

    inits = generate_inits(N, parent, rng)
    init_names = list(inits.keys())
    priority = ['parent', 'uniform', 'multi_cos', 'random', 'cos_1', 'gauss', 'two_gauss', 'tri']
    init_names = [nm for nm in priority if nm in inits] + \
                 [nm for nm in init_names if nm not in priority]

    phase1_budget = 650.0
    time_per_init = phase1_budget / len(init_names)

    for name in init_names:
        remain = deadline - time.monotonic() - 550.0
        if remain < 20:
            break
        t_alloc = min(time_per_init, remain)
        if t_alloc < 10:
            break

        a0 = inits[name]
        print(f"  Init: {name}, alloc={t_alloc:.0f}s")

        # LP min_peak
        t_sub = t_alloc * 0.45
        a_opt, obj = lp_iteration(a0, time_limit=t_sub, mode='min_peak',
                                  step_init=0.3, use_line_search=True)
        try_update(a_opt)

        # QP flatness from best so far
        remain = deadline - time.monotonic() - 500.0
        if remain > 15:
            t_sub = min(t_alloc * 0.35, remain)
            start = a_opt if a_opt is not None else a0
            a_opt2, obj2 = lp_iteration(start, time_limit=t_sub, mode='qp_flat',
                                        step_init=0.25, use_line_search=True)
            try_update(a_opt2)

        # LP max_sum
        remain = deadline - time.monotonic() - 450.0
        if remain > 10:
            t_sub = min(t_alloc * 0.2, remain)
            a_opt3, obj3 = lp_iteration(a0, time_limit=t_sub, mode='max_sum',
                                        step_init=0.25, use_line_search=True)
            try_update(a_opt3)

    print(f"Phase 1 done: best={best_score:.6f}, elapsed={time.monotonic()-t1_start:.0f}s")

    # === Phase 2: Deep refinement on best ===
    print("=== Phase 2: Deep Refinement ===")
    t2_start = time.monotonic()

    if best_sequence is not None:
        phase2_deadline = deadline - 350.0
        for step_init in [0.25, 0.15, 0.4, 0.1]:
            if time.monotonic() > phase2_deadline:
                break
            t_alloc = min(70.0, phase2_deadline - time.monotonic() - 5)
            if t_alloc < 8:
                break
            for mode in ['min_peak', 'qp_flat', 'max_sum']:
                if time.monotonic() > phase2_deadline:
                    break
                t_single = t_alloc / 3.0
                if t_single < 5:
                    break
                print(f"  LP {mode}, step={step_init}")
                a_opt, obj = lp_iteration(best_sequence, time_limit=t_single,
                                          mode=mode, step_init=step_init,
                                          use_line_search=True)
                try_update(a_opt)

    print(f"Phase 2 done: best={best_score:.6f}, elapsed={time.monotonic()-t2_start:.0f}s")

    # === Phase 3: Soft-max gradient polishing ===
    print("=== Phase 3: Soft-max Gradient ===")
    t3_start = time.monotonic()

    if best_sequence is not None:
        phase3_deadline = deadline - 250.0
        for lr in [0.006, 0.003, 0.01]:
            if time.monotonic() > phase3_deadline:
                break
            t_alloc = min(70.0, phase3_deadline - time.monotonic() - 5)
            if t_alloc < 10:
                break
            print(f"  Soft-grad lr={lr}")
            a_opt, obj = optimize_soft_grad(best_sequence, time_limit=t_alloc,
                                            lr_init=lr, momentum=0.93,
                                            t_init=40.0)
            try_update(a_opt)

    print(f"Phase 3 done: best={best_score:.6f}, elapsed={time.monotonic()-t3_start:.0f}s")

    # === Phase 4: Multi-perturbation restart ===
    print("=== Phase 4: Perturbation Restart ===")
    t4_start = time.monotonic()

    if best_sequence is not None:
        phase4_deadline = deadline - 80.0
        n_best = len(best_sequence)
        scales = [0.002, 0.008, 0.004, 0.012, 0.001, 0.015]
        for scale in scales:
            if time.monotonic() > phase4_deadline:
                break
            noise = rng.standard_normal(n_best) * scale
            a_pert = project_simplex(best_sequence + noise)
            t_alloc = min(35.0, phase4_deadline - time.monotonic() - 5)
            if t_alloc < 8:
                break
            print(f"  Perturb scale={scale}")
            a_opt, obj = lp_iteration(a_pert, time_limit=t_alloc, mode='min_peak',
                                      step_init=0.2, use_line_search=True)
            try_update(a_opt)

    print(f"Phase 4 done: best={best_score:.6f}, elapsed={time.monotonic()-t4_start:.0f}s")

    # === Phase 5: Final polish ===
    print("=== Phase 5: Final Polish ===")
    t5_start = time.monotonic()

    if best_sequence is not None:
        remain = deadline - time.monotonic() - 5.0
        if remain > 15:
            for mode in ['min_peak', 'qp_flat']:
                if time.monotonic() > deadline - 5:
                    break
                t_single = remain / 2.0
                if t_single < 5:
                    break
                a_opt, obj = lp_iteration(best_sequence, time_limit=t_single,
                                          mode=mode, step_init=0.15,
                                          use_line_search=True)
                try_update(a_opt)

    print(f"Phase 5 done: best={best_score:.6f}, elapsed={time.monotonic()-t5_start:.0f}s")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
