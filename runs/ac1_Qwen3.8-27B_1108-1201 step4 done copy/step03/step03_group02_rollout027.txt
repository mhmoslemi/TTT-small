(multiplicative weights) with log-sum-exp smoothing, combined with LP-based
polishing and multi-restart exploration. The key innovation is using the
multiplicative update rule a[j] *= exp(-eta * grad[j]) which naturally
respects non-negativity and the simplex constraint, enabling many more
iterations than LP-based methods within the time budget.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def compute_grad_lse(a, w):
    """Compute gradient of log-sum-exp smoothed objective.
    
    grad[m] = 2 * sum_k w[k] * a[k-m] (with a out-of-range = 0)
    """

    # Using np.correlate: f[m] = cross[n-1-m] where cross = correlate(w, a, 'full')
    cross = np.correlate(w, a, mode='full')
    # f[m] = cross[n-1-m]
    grad = 2.0 * cross[n-1::-1][:n]
    return grad


def compute_grad_fft(a, w):

        # In FFT terms: R = ifft(fft(w, N) * conj(fft(a, N))) gives circular cross-corr
        # For linear: R_linear[m] = R_circ[m] for m >= 0 (with proper zero-padding)
        # Actually this is getting complicated. Let me just use correlate.
        pass

def optimize_mirror_descent(a0, time_limit, eta_init=0.5, T_init=20.0,
                            use_momentum=False, T_schedule='linear'):
    """Mirror descent (multiplicative weights) with log-sum-exp smoothing."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = 2.0 * n * c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    eta = eta_init
    T = T_init
    iter_count = 0
    no_improve = 0
    a_prev = a.copy()
    t_nest = 1.0

    while time.monotonic() - t0 < time_limit:
        # Compute softmax weights over convolution peaks
        scaled = T * c
        scaled -= scaled.max()
        exp_vals = np.exp(scaled)
        w = exp_vals / exp_vals.sum()

        # Compute gradient
        grad = compute_grad_lse(a, w)

        # Mirror descent step
        eta_eff = eta * n
        log_update = -eta_eff * grad
        log_update -= log_update.max()

            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

            coeff = (t_nest - 1.0) / (t_nest + 1.0)

            a = a_new
        a_prev = a_new.copy()

        elif no_improve > 50:

            c = fast_convolve(a)
            no_improve = 0
            eta = eta_init

def optimize_lp_polish(a0, time_limit, max_iters=200, step_init=0.3):

    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = 2.0 * n * c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    step = step_init

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        M = c.max()
        A = build_conv_matrix(a)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status != 0:
            break
        b = np.maximum(res.x, 0)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum

        if np.linalg.norm(b - a) < 1e-10:
            break

        a_new = (1.0 - step) * a + step * b
        a_new = np.maximum(a_new, 0.0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            break
        a_new = a_new / a_new_sum

        c_new = fast_convolve(a_new)
        obj_new = 2.0 * n * c_new.max()

        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            step = min(0.7, step * 1.03)
        else:
            step = max(0.05, step * 0.9)

        a = a_new
        c = c_new

    return best_a, best_obj


def generate_initializations(n, parent, rng):

        inits[f'cos_{freq}'] = a

    inits['gauss'] = g

    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp

    # Tukey window
    r = 0.3
    tukey = np.ones(n)
    half = int(r * n / 2)
    for i in range(half):
        tukey[i] = 0.5 * (1 + np.cos(np.pi * (i / half - 1)))
        tukey[n-1-i] = tukey[i]
