import numpy as np
import time
from scipy.optimize import minimize, linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_objective(a):
    """Compute the raw objective (before evaluate_sequence bounds)."""
    n = len(a)
    conv = fast_convolve(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a), conv

def build_conv_matrix(a):
    """Build matrix C such that (C @ g)[k] = sum_i a[k-i] * g[i] (cross-correlation a*g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C

def solve_lp(a, M):
    """Solve LP: maximize sum(g) s.t. C@g <= M, g >= 0. Returns g."""
    n = len(a)
    C = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, result.fun
    return None, None

def line_search(a, d, n, f_current):
    """Find best step size t in [0, 1] for a + t*d."""
    best_t = 0.0
    best_f = f_current
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2
    a_lb, a_ub = 0.0, 1.0
    c = a_ub - gr * (a_ub - a_lb)
    d_ = a_lb + gr * (a_ub - a_lb)
    f_c = eval_line(a, d, c, n)
    f_d = eval_line(a, d, d_, n)
    
    for _ in range(30):
        if f_c < f_d:
            a_ub = d_
            d_ = c
            f_d = f_c
            c = a_ub - gr * (a_ub - a_lb)
            f_c = eval_line(a, d, c, n)
        else:
            a_lb = c
            c = d_
            f_c = f_d
            d_ = a_lb + gr * (a_ub - a_lb)
            f_d = eval_line(a, d, d_, n)
    
    best_t = (a_lb + a_ub) / 2
    best_f = eval_line(a, d, best_t, n)
    return best_t, best_f

def eval_line(a, d, t, n):
    """Evaluate objective at a + t*d (with projection to non-negative and normalization)."""
    a_try = a + t * d
    a_try = np.maximum(a_try, 0.0)
    s = a_try.sum()
    if s < 1e-15:
        return float('inf')
    a_try = a_try / s
    f, _ = compute_objective(a_try)
    return f

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. L-BFGS-B on a log-sum-exp smoothed objective with softmax parameterization
2. LP-based direction finding for escaping local minima
3. Multi-start with diverse initializations and multiple sequence lengths
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_obj_normalized(b):
    """Compute 2n * max(conv(b,b)) where sum(b)=1."""
    n = len(b)
    conv = fast_convolve(b)
    return 2.0 * n * conv.max(), conv

def softmax(x):
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    return e / e.sum()

def smoothed_obj_and_grad(x, T):
    """
    Compute the smoothed objective and gradient for the softmax parameterization.
    F(x) = 2n * logsumexp(T * c(softmax(x)))
    """
    n = len(x)
    b = softmax(x)
    
    conv = fast_convolve(b)
    Tconv = T * conv
    max_Tconv = Tconv.max()
    log_sum_exp = max_Tconv + np.log(np.sum(np.exp(Tconv - max_Tconv)))
    F = 2.0 * n * log_sum_exp
    
    # Weights for the gradient
    w = np.exp(Tconv - max_Tconv)  # unnormalized
    w_sum = w.sum()
    w = w / w_sum  # normalized: w_k = exp(T*c_k) / sum exp(T*c_j)
    
    # Gradient of G(b) = 2n * LSE(T*c(b)) w.r.t. b:
    # dG/db_i = 4nT * sum_k w_k * b_{k-i}
    # = 4nT * sum_{j=0}^{n-1} w_{i+j} * b_j
    H = np.zeros(n)
    for i in range(n):
        H[i] = np.dot(w[i:i+n], b)
    H *= 4.0 * n * T
    
    # Gradient w.r.t. x: dF/dx_i = b_i * (H_i - H·b)
    Hb = np.dot(H, b)
    grad = b * (H - Hb)
    
    return F, grad

direction finding for escaping local minima, multi-start strategies,
and adaptive temperature scheduling.

import numpy as np
import time
from scipy.optimize import minimize, linprog

# Global variable for current temperature (to avoid closures/lambdas)
_CURRENT_T = 1.0
_CURRENT_N = 0


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax_func(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def compute_objective_val(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    conv = fast_convolve(a)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


def _obj_func(x):
    """Objective function using global _CURRENT_T."""
    n = len(x)
    b = softmax_func(x)
    conv = fast_convolve(b)
    Tc = _CURRENT_T * conv
    max_Tc = Tc.max()
    lse = max_Tc + np.log(np.sum(np.exp(Tc - max_Tc)))
    return 2.0 * n * lse


def _grad_func(x):
    """Gradient function using global _CURRENT_T."""
    n = len(x)
    b = softmax_func(x)
    conv = fast_convolve(b)
    Tc = _CURRENT_T * conv
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    w = exp_Tc / exp_Tc.sum()
    
    # H[i] = 4*n*T * sum_j w[i+j] * b[j]

    H = w[idx_i + idx_j] @ b
    H *= 4.0 * n * _CURRENT_T

    grad = b * (H - Hb)

def build_conv_matrix(a):
    """Build matrix C such that (C @ g)[k] = sum_i a[k-i] * g[i]."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def solve_lp_escape(a, M, margin=1.05):
    """
    Solve LP: maximize sum(g) s.t. C@g <= M*margin, g >= 0.
    Returns normalized g or None.
    """
    n = len(a)
    C = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M * margin)
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        g = result.x
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g / g_sum
    return None

    return best_t, best_obj


def combined_optimize(a_init, time_limit):
    """
    Combined optimization: L-BFGS-B with LP-based escape.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_objective_val(a)
    
    t0 = time.monotonic()
    no_improve_count = 0
    max_no_improve = 20
    lp_count = 0
    
    while time.monotonic() - t0 < time_limit:
        # Phase: L-BFGS-B optimization
        remaining = time_limit - (time.monotonic() - t0)
        t_lbfgs = min(30.0, remaining * 0.4)
        if t_lbfgs < 2:
            break
        
        b_opt, obj_opt = lbfgs_optimize(a, T_start=50.0, T_end=1.0, time_limit=t_lbfgs)
        
        if obj_opt < best_obj - 1e-10:
            best_obj = obj_opt
            best_a = b_opt.copy()
            a = b_opt.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        # Phase: LP-based escape
        if no_improve_count >= max_no_improve and time.monotonic() - t0 < time_limit - 5:
            conv = fast_convolve(a)
            M = conv.max()
            
            g = solve_lp_escape(a, M, margin=1.02)
            if g is not None:
                d = g - a
                if np.linalg.norm(d) > 1e-8:
                    t_best, obj_best = line_search_1d(a, d, n, best_obj)
                    if obj_best < best_obj - 1e-10:
                        a_new = a + t_best * d
                        a_new = np.maximum(a_new, 0.0)
                        s_new = a_new.sum()
                        if s_new > 1e-15:
                            a = a_new / s_new
                            best_obj = obj_best
                            best_a = a.copy()
                            no_improve_count = 0
                            lp_count += 1
            
            # If LP didn't help, perturb
            if no_improve_count >= max_no_improve + 10:
                rng = np.random.default_rng(int(time.monotonic() * 1000))
                noise = rng.standard_normal(n) * 0.01
                a = best_a + noise
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                else:
                    a = best_a.copy()
                no_improve_count = 0
                max_no_improve = min(50, max_no_improve + 5)
        
        if no_improve_count > 100:
            break
    
    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = tri / tri.sum()
    
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
    inits['gaussian'] = g / g.sum()
    
    sine = (np.sin(np.pi * x / n) + 0.5)
    inits['sine'] = sine / sine.sum()
    
    # Bimodal
    bi = np.zeros(n)
    edge = max(1, n // 3)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # Power law
    pw = (x + 1.0) ** 0.3
    inits['power'] = pw / pw.sum()
    
    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            elif len(p) > n:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_ds'] = pp / pps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_up'] = pp / pps
    
    return inits
