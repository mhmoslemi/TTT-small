```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using simulated annealing with
peak-targeted moves, combined with smooth L-BFGS-B initialization and polishing.
The SA step uses efficient O(n) convolution updates for individual swaps.
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max_and_weights(x, beta):
    """Compute smooth max and softmax weights."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def compute_grad_smooth_conv(a, w):
    """Gradient of weighted sum of conv coefficients w.r.t. a."""
    n = len(a)
    N = 1 << (len(w) + n - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0
    return grad


def project_simplex(v):
    """Project onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_obj(x, n, beta):
    """Objective for L-BFGS-B (softmax parametrization)."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, _ = smooth_max_and_weights(conv, beta)
    return 2.0 * n * sm


def _softmax_jac(x, n, beta):
    """Jacobian for L-BFGS-B."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, w = smooth_max_and_weights(conv, beta)
    grad_a = compute_grad_smooth_conv(a, w)
    lagrange = np.dot(a, grad_a)
    return a * (grad_a - lagrange)


def optimize_lbfgsb(a0, beta, maxiter=500):
    """Optimize using L-BFGS-B with softmax reparametrization."""
    n = len(a0)
    a0_clipped = np.maximum(a0, 1e-30)
    a0_clipped = a0_clipped / a0_clipped.sum()
    x0 = np.log(a0_clipped)
    result = minimize(_softmax_obj, x0, jac=_softmax_jac, args=(n, beta),
                      method='L-BFGS-B', options={'maxiter': maxiter, 'maxls': 50})
    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt, result.fun


def flat_spectrum_init(n, rng):
    """Initialize with approximately flat power spectrum."""
    fft_size = 1 << (n - 1).bit_length()
    A = np.zeros(fft_size, dtype=complex)
    A[0] = 1.0
    phases = rng.uniform(0, 2 * np.pi, fft_size)
    for k in range(1, fft_size // 2 + 1):
        A[k] = np.exp(1j * phases[k])
        if k < fft_size - k:
            A[fft_size - k] = np.exp(-1j * phases[k])
    a = np.real(ifft(A, fft_size))[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def compute_conv_change(conv_old, a, i, j, delta, n):
    """
    Compute new convolution after moving delta from position i to position j.
    Returns the new conv array.
    """
    # delta_conv[k] = 2*delta*(a[k-j] - a[k-i])
    # a[m] = 0 if m < 0 or m >= n
    
    change = np.zeros(2 * n - 1)
    
    # Contribution from a[k-j]: valid when 0 <= k-j < n, i.e., j <= k < j+n
    lo = max(0, j)
    hi = min(2 * n - 2, j + n - 1)

    hi = min(2 * n - 2, i + n - 1)

    return conv_old + change


def sa_peak_targeted(a, conv, k_star, rng, peak_fraction=0.3, bottom_fraction=0.3):
    """
    Propose a peak-targeted swap move.
    Returns (i, j, delta) - move delta from position i to position j.
    """
    n = len(a)
    
    # Compute influence: how much each position contributes to peak at k_star
    # influence[i] = a[k_star - i] if 0 <= k_star - i < n, else 0
    influence = np.zeros(n)
    # a[k_star - i] for i in [0, n-1]: valid when 0 <= k_star - i < n
    # i.e., k_star - n + 1 <= i <= k_star
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    if lo <= hi:
        influence[lo:hi + 1] = a[k_star - hi:k_star - lo + 1][::-1]
    
    # Pick i from top contributors (high influence)
    top_count = max(1, int(n * peak_fraction))
    bottom_count = max(1, int(n * bottom_fraction))
    
    # For i: pick from positions with highest influence
    top_indices = np.argsort(influence)[::-1][:top_count]
    # Only pick from positions with non-zero a values
    top_indices = top_indices[a[top_indices] > 1e-10]
    if len(top_indices) == 0:
        return None
    
    i = top_indices[rng.integers(len(top_indices))]
    
    # For j: pick from positions with lowest influence
    bottom_indices = np.argsort(influence)[:bottom_count]
    # Exclude i from consideration
    bottom_indices = bottom_indices[bottom_indices != i]
    if len(bottom_indices) == 0:
        return None
    j = bottom_indices[rng.integers(len(bottom_indices))]
    
    # Delta: a fraction of a[i]
    max_delta = a[i] * 0.3  # Can't move more than 30% of a[i]
    delta = max_delta * rng.uniform(0.3, 1.0)
    
    if delta < 1e-12:
        return None
    
    return (i, j, delta)


def sa_random(a, rng):
    """Propose a random swap move."""
    n = len(a)
    # Pick two random distinct positions
    indices = rng.integers(0, n, size=2)
    i, j = int(indices[0]), int(indices[1])
    if i == j:
        j = (i + 1) % n
    
    # Small random delta
    delta = a[i] * rng.uniform(0.05, 0.3)
    if a[i] < 1e-12:
        # Can't take from i, try the reverse
        i, j = j, i
        delta = a[i] * rng.uniform(0.05, 0.3)
        if delta < 1e-12:
            return None
    
    # Determine direction: can go either way
    if rng.random() < 0.5:
        return (i, j, delta)
    else:
        return (j, i, delta)


def simulated_annealing(a, n, rng, t_start, t_end, n_iters, deadline, peak_controls=0.7):

    Returns (best_a, best_obj, final_a, final_obj).

    conv = fast_convolve(a)
    current_obj = 2.0 * n * conv.max() / (sum_a ** 2)
    
    best_a = a.copy()
    best_obj = current_obj
    best_iter = 0
    
    t = t_start
    cooling = (t_end / t_start) ** (1.0 / n_iters)
    
    no_improve = 0
    reheat_threshold = 50000
    reheat_factor = 0.5
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        # Check deadline every 10000 iterations
        if it % 10000 == 0 and time.monotonic() > deadline:
            break
        
        # Decay temperature
        t *= cooling
        
        # Reheat if stuck
        if no_improve > reheat_threshold:
            t *= reheat_factor
            no_improve = 0
            print(f"    Reheat at iter {it}, T={t:.2e}")
        
        # Propose move
        use_peak = rng.random() < peak_controls
        if use_peak:
            k_star = int(np.argmax(conv))
            move = sa_peak_targeted(a, conv, k_star, rng)
        else:
            move = sa_random(a, rng)
        
        if move is None:
            continue
        
        i, j, delta = move
        
        # Evaluate the move
        new_conv = compute_conv_change(conv, a, i, j, delta, n)
        
        # Check non-negativity (should be guaranteed by delta <= a[i], but just in case)
        if delta > a[i]:
            continue
        
        new_max = new_conv.max()
        new_obj = 2.0 * n * new_max / (sum_a ** 2)
        
        # Accept/reject
        delta_obj = new_obj - current_obj
        if delta_obj <= 0 or rng.random() < np.exp(-delta_obj / t):
            # Accept
            a[i] -= delta
            a[j] += delta
            conv = new_conv
            current_obj = new_obj
            
            if current_obj < best_obj:
                best_obj = current_obj
                best_a = a.copy()
                best_iter = it
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
    
    return best_a, best_obj, a, current_obj


def subgradient_step(a, n, step_size=0.01):
    """
    One subgradient step: move in the direction of -grad at the argmax.
    Returns updated a (projected to simplex).
    """
    conv = fast_convolve(a)
    k_star = int(np.argmax(conv))
    
    # Subgradient of conv[k_star] w.r.t. a[i] is 2*a[k_star - i] (if valid)
    grad = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    if lo <= hi:
        grad[lo:hi + 1] = 2.0 * a[k_star - hi:k_star - lo + 1][::-1]
    
    # Projected gradient step
    a_new = a - step_size * grad
    a_new = project_simplex(a_new)
    return a_new


def peak_reduction_deterministic(a, n, n_steps=100, deadline=None):
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Multi-start smooth L-BFGS-B initialization
2. Simulated annealing with efficient O(n) incremental convolution updates
   and peak-targeted move proposals
3. Exact subgradient polishing with the true max (not smoothed)
4. Final L-BFGS-B refinement
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max_and_weights(x, beta):
    """Compute smooth max and softmax weights."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def compute_grad_smooth_conv(a, w):
    """Gradient of sum_k w[k]*conv[k] w.r.t. a."""
    n = len(a)
    N = 1 << (len(w) + n - 1 + 7).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0
    return grad


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_obj(x, n, beta):
    """Objective for L-BFGS-B with softmax parametrization."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, _ = smooth_max_and_weights(conv, beta)
    return 2.0 * n * sm


def _softmax_jac(x, n, beta):
    """Jacobian for L-BFGS-B with softmax parametrization."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, w = smooth_max_and_weights(conv, beta)
    grad_a = compute_grad_smooth_conv(a, w)
    lagrange = np.dot(a, grad_a)
    return a * (grad_a - lagrange)


def optimize_lbfgsb(a0, beta, maxiter=300):
    """Optimize using L-BFGS-B with softmax reparametrization."""
    n = len(a0)
    a0_clipped = np.maximum(a0, 1e-30)
    a0_clipped = a0_clipped / a0_clipped.sum()
    x0 = np.log(a0_clipped)
    result = minimize(_softmax_obj, x0, jac=_softmax_jac, args=(n, beta),
                      method='L-BFGS-B', options={'maxiter': maxiter, 'maxls': 40})
    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt, result.fun


def flat_spectrum_init(n, rng):
    """Initialize with approximately flat power spectrum."""
    fft_size = 1 << (n - 1).bit_length()
    A = np.zeros(fft_size, dtype=complex)
    A[0] = 1.0
    phases = rng.uniform(0, 2 * np.pi, fft_size)
    for k in range(1, fft_size // 2 + 1):
        A[k] = np.exp(1j * phases[k])
        if k < fft_size - k:
            A[fft_size - k] = np.exp(-1j * phases[k])
    a = np.real(ifft(A, fft_size))[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def compute_conv_incremental(conv, a, i, j, delta, n):
    """
    Compute new convolution after moving delta from position i to position j.
    delta_conv[k] = 2*delta*(a[k-j] - a[k-i]) where a[m]=0 for m outside [0,n-1].
    """
    change = np.zeros(2 * n - 1)
    # a[k-j] contribution: valid for k in [j, j+n-1]
    lo = j
    hi = min(2 * n - 2, j + n - 1)
    if lo <= hi:
        change[lo:hi + 1] += 2.0 * delta * a[:hi - lo + 1]
    # a[k-i] contribution (negative): valid for k in [i, i+n-1]
    lo = i
    hi = min(2 * n - 2, i + n - 1)
    if lo <= hi:
        change[lo:hi + 1] -= 2.0 * delta * a[:hi - lo + 1]
    return conv + change

    cur_obj = 2.0 * n * conv.max() / sum_sq

    best_a = a.copy()
    best_obj = cur_obj

    t = float(t_init)
    cooling = (t_final / t_init) ** (1.0 / max(1, n_iters))

    no_improve = 0
    reheat_every = 80000
    periodic_recompute = 50000

        if it % periodic_recompute == 0 and it > 0:
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max() / sum_sq

        t *= cooling
        if t < 1e-8:
            t = 1e-8

        # Reheat if stuck
        if no_improve > reheat_every and reheat_every < 500000:
            t = max(t, t_init * 0.1)
            no_improve = 0
            reheat_every = min(500000, reheat_every * 1.5)
            if it % 100000 == 0:
                print("    SA reheat at iter %d, best=%.6f" % (it, best_obj))

        # Propose move
        if rng.random() < peak_prob:
            # Peak-targeted move
            k_star = int(np.argmax(conv))
            # Influence: a[k_star - i] for valid i
            lo = max(0, k_star - n + 1)
            hi = min(n - 1, k_star)
            if hi < lo:
                continue
            # influence[i] = a[k_star - i]
            # For i in [lo, hi]: k_star - i ranges from k_star-hi to k_star-lo
            influence_seg = a[k_star - hi : k_star - lo + 1][::-1]
            # Find threshold: median of influence
            med = np.median(influence_seg)
            # Source i: from upper half of influence
            upper_mask = influence_seg > med
            if not upper_mask.any():
                upper_mask = influence_seg == influence_seg.max()
            upper_positions = np.where(upper_mask)[0] + lo
            # Filter to those with significant a value
            upper_positions = upper_positions[a[upper_positions] > 1e-10]
            if len(upper_positions) == 0:
                continue
            i = int(upper_positions[rng.integers(len(upper_positions))])

            # Target j: from lower half of influence
            lower_mask = influence_seg <= med
            if not lower_mask.any():
                lower_mask = influence_seg == influence_seg.min()
            lower_positions = np.where(lower_mask)[0] + lo
            lower_positions = lower_positions[lower_positions != i]
            if len(lower_positions) == 0:
                continue
            j = int(lower_positions[rng.integers(len(lower_positions))])
        else:
            # Random move
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                j = (i + 1) % n

        # Delta
        ai = a[i]
        if ai < 1e-12:
            # Swap direction
            i, j = j, i
            ai = a[i]
            if ai < 1e-12:
                continue
        delta = ai * rng.uniform(0.03, 0.25)

        # Evaluate
        new_conv = compute_conv_incremental(conv, a, i, j, delta, n)
        new_max = new_conv.max()
        new_obj = 2.0 * n * new_max / sum_sq

        # Accept/reject
        d_obj = new_obj - cur_obj
        if d_obj <= 0 or (t > 1e-15 and rng.random() < np.exp(-d_obj / t)):
            a[i] -= delta
            a[j] += delta
            conv = new_conv
            cur_obj = new_obj
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_a, best_obj, a.copy(), cur_obj


def subgradient_polish(a_init, n, lr_init=0.02, n_iters=2000, deadline=None):
    """
    Projected subgradient method using exact max.
    At each step, moves opposite to the subgradient at the argmax.
    Uses decreasing learning rate with occasional reheat.
    """
    a = np.maximum(a_init, 0.0).copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    sum_sq = s * s

    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max() / sum_sq

    lr = lr_init
    eps = 1e-10

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        conv = fast_convolve(a)
        cur_max = conv.max()
        cur_obj = 2.0 * n * cur_max / sum_sq

        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

        # Find top-3 peaks and average their subgradients
        top_k = min(3, 2 * n - 1)
        sorted_idx = np.argsort(conv)[::-1][:top_k]
        weights = np.ones(top_k) / top_k

        sg = np.zeros(n)
        for idx in sorted_idx:
            k_star = int(idx)
            lo = max(0, k_star - n + 1)
            hi = min(n - 1, k_star)
            if lo <= hi:
                sg[lo:hi + 1] += 2.0 * a[k_star - hi : k_star - lo + 1][::-1]
        sg *= weights.sum()  # already divided by top_k

        # Projected gradient step
        a_new = project_simplex(a - lr * sg)
        a = a_new
        lr *= 0.999

    return best_a, best_obj
