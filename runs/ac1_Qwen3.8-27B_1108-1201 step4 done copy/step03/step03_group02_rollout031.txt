"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-phase optimization:
(1) LP-based direction finding with backtracking line search,
(2) L-BFGS-B on log-sum-exp smoothed surrogate with annealing,
(3) Structured peak-transfer perturbations,
(4) Multi-length exploration and final polish.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (cssv - 1.0) / np.arange(1, n + 1) < u
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def compute_obj_normalized(a):
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * c.max()


def smooth_obj(a, t):
    n = len(a)
    c = fast_convolve(a)
    scaled = t * c
    scaled -= scaled.max()
    return 2.0 * n * (scaled.max() + np.log(np.exp(scaled).sum())) / t


def smooth_grad(a, t):
    n = len(a)
    c = fast_convolve(a)
    scaled = t * c
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    sum_exp = exp_vals.sum()
    p = exp_vals / sum_exp
    grad = np.zeros(n)
    for j in range(n):
        k_lo = j
        k_hi = min(2 * n - 2, j + n - 1)
        if k_lo <= k_hi:
            grad[j] = (4.0 * n / t) * np.dot(p[k_lo:k_hi + 1], a[k_lo - j:k_hi - j + 1])
    return grad


def lp_direction(a, method, top_k_frac=0.1):
    """LP-based direction finding. Returns normalized direction or None."""
    n = len(a)
    num_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c_conv = fast_convolve(a)
    
    if method == 'max_sum':
        M = c_conv.max()
        c_obj = -np.ones(n)
        b_ub = np.full(num_rows, M)
        bounds = [(0, None)] * n
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 30})
        if res.status == 0:
            b = np.maximum(res.x, 0)
            s = b.sum()
            if s > 1e-10:
                return b / s
        return None
    elif method == 'weighted':
        # Weight constraints by peak proximity
        M = c_conv.max()
        M_min = c_conv.min()
        if M - M_min < 1e-15:
            w = np.ones(num_rows)
        else:
            w = 1.0 + 5.0 * (c_conv - M_min) / (M - M_min)
        # Scale constraints: w[k] * conv[k] <= w[k] * M
        A_w = A * w[:, None]
        b_ub = w * M
        c_obj = -np.ones(n)
        bounds = [(0, None)] * n
        res = linprog(c_obj, A_ub=A_w, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 30})
        if res.status == 0:
            b = np.maximum(res.x, 0)
            s = b.sum()
            if s > 1e-10:
                return b / s
        return None
    elif method == 'min_peak':
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_eq = np.ones((1, n))
        b_eq = np.array([1.0])
        A_ub_full = np.zeros((num_rows, n + 1))
        A_ub_full[:, :n] = A
        A_ub_full[:, n] = -1.0
        b_ub = np.zeros(num_rows)
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c_obj, A_ub=A_ub_full, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': 30})
        if res.status == 0:
            b = np.maximum(res.x[:n], 0)
            s = b.sum()
            if s > 1e-10:
                return b / s
        return None
    return None


def lp_iterate(a0, time_limit, method='max_sum', max_iters=500):
    """LP-based iterative refinement with backtracking line search."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = 2.0 * n * c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    step = 0.3
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = lp_direction(a, method)
        if b is None:
            break
        
        if np.linalg.norm(b - a) < 1e-10:
            break
        
        # Backtracking line search
        found = False
        for t_trial in [step, step * 0.7, step * 0.5, step * 0.3, step * 0.1]:
            a_new = (1.0 - t_trial) * a + t_trial * b
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            c_new = fast_convolve(a_new)
            obj_new = 2.0 * n * c_new.max()
            
            if obj_new <= best_obj + 1e-8:  # Allow very slight increase
                a = a_new
                c = c_new
                if obj_new < best_obj - 1e-10:
                    best_obj = obj_new
                    best_a = a.copy()
                    no_improve = 0
                    step = min(0.6, step * 1.1)
                else:
                    no_improve += 1
                found = True
                break
        
        if not found:
            no_improve += 1
            step = max(0.05, step * 0.7)
        
        if no_improve > 100:
            break
    
    return best_a, best_obj


def lbfgs_smooth(a0, time_limit, t_init=20.0, t_min=1.0, t_decay=0.995):
    """L-BFGS-B optimization on log-sum-exp smoothed objective."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = 2.0 * n * c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    t_cur = t_init
    
    while time.monotonic() - t0 < time_limit:
        # Define objective and gradient for scipy
        def obj_fun(x):
            x_pos = np.maximum(x, 0)
            s_x = x_pos.sum()
            if s_x < 1e-15:
                return 1e10
            x_norm = x_pos / s_x
            return smooth_obj(x_norm, t_cur)
        
        def grad_fun(x):
            x_pos = np.maximum(x, 0)
            s_x = x_pos.sum()
            if s_x < 1e-15:
                return np.ones(len(x))
            x_norm = x_pos / s_x
            g = smooth_grad(x_norm, t_cur)
            # Chain rule for normalization: d(x/s)/dx_i = (s - x_i)/s^2 = (1-x_i/s)/s
            # But since we're on the simplex, the gradient should be projected
            # Actually, let's just work in the unnormalized space and project
            return g / s_x
        
        # Project current point to simplex
        a_proj = project_simplex(a)
        
        res = minimize(obj_fun, a_proj, jac=grad_fun, method='L-BFGS-B',
                       options={'maxiter': 200, 'gtol': 1e-6, 'maxfun': 500})
        
        if res.x is not None:
            a_res = project_simplex(res.x)
            s_res = a_res.sum()
            if s_res > 1e-15:
                a_res = a_res / s_res
                c_res = fast_convolve(a_res)
                obj_res = 2.0 * n * c_res.max()
                if obj_res < best_obj - 1e-10:
                    best_obj = obj_res
                    best_a = a_res.copy()
                    a = a_res
            else:
                break
        
        # Cool down
        t_cur = max(t_min, t_cur * t_decay)
        
        # Check if we should stop
        if t_cur <= t_min + 1e-6:
            # Do one more pass at minimum temperature
            t_cur = t_min
            # ... (will exit after this iteration due to time or no improvement)
    
    # Final evaluation
    c_final = fast_convolve(best_a)
    final_obj = 2.0 * n * c_final.max()
    return best_a, final_obj


def peak_transfer_perturb(a, n_transfers=10, transfer_scale=0.01):
    """Transfer mass from peak-contributing positions to non-contributing ones."""
    n = len(a)
    c = fast_convolve(a)
    peak_k = np.argmax(c)
    
    # Compute blame: how much each position contributes to the peak
    blame = np.zeros(n)
    j_start = max(0, peak_k - n + 1)
    j_end = min(n - 1, peak_k)
    for j in range(j_start, j_end + 1):
        blame[j] = a[peak_k - j]
    
    # High blame positions (sources) and low blame positions (targets)
    blame_sorted = np.argsort(blame)[::-1]
    sources = blame_sorted[:n // 4]  # top quarter have highest blame
    targets = blame_sorted[n // 4:]  # rest are targets
    
    a_new = a.copy()
    rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
    
    for _ in range(n_transfers):
        si = rng.choice(sources)
        ti = rng.choice(targets)
        if si == ti:
            continue
        delta = transfer_scale * a[si]
        if delta > 0 and a_new[si] > delta:
            a_new[si] -= delta
            a_new[ti] += delta
    
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def generate_inits(n, parent, rng):
    inits = {}
    inits['uniform'] = np.ones(n) / n
    
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
        else:
            pp = np.interp(np.linspace(0, 1, n),
                          np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps
    
    x = np.arange(n, dtype=np.float64)
    for freq in [1, 2, 3, 5]:
        a = 1.0 + 0.9 * np.cos(2.0 * np.pi * freq * x / n)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum > 1e-10:
            inits[f'cos{freq}'] = a / a_sum
    
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g
    
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Two-lobe
    a = np.zeros(n)
    a[:n//4] = 1.0
    a[3*n//4:] = 1.0
    a /= a.sum()
    inits['two_lobe'] = a
    
    # Anti-symmetric-ish (high in middle, low at edges)
    a = np.sin(np.pi * x / n) ** 2
    a /= a.sum()
    inits['sin2'] = a
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-phase optimization:
(1) LP-based direction finding with backtracking line search,
(2) L-BFGS-B on log-sum-exp smoothed surrogate with temperature annealing,

import numpy as np
import time
from scipy.optimize import linprog, minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (cssv - 1.0) / np.arange(1, n + 1) < u
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def compute_obj_normalized(a):
    """Objective for a normalized (sum=1) non-negative sequence."""
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * c.max()

    """Log-sum-exp smoothed objective value."""
    n = len(a)
    c = fast_convolve(a)
    scaled = t * c
    scaled -= scaled.max()
    return 2.0 * n * (scaled.max() + np.log(np.exp(scaled).sum())) / t


def smooth_grad_fft(a, t):
    """Compute gradient of log-sum-exp smoothed objective using FFT."""
    n = len(a)
    c = fast_convolve(a)
    scaled = t * c
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    sum_exp = exp_vals.sum()
    if sum_exp < 1e-300:
        return np.zeros(n)
    p = exp_vals / sum_exp
    # grad[j] = 4n * sum_k p[k] * a[k-j]
    # Compute via cross-correlation using FFT
    size = 4 * n
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a, size)
    cross = np.fft.irfft(np.conj(P) * A, size)
    return 4.0 * n * cross[:n]

    """Combined smooth objective and gradient."""
    n = len(a)
    obj = smooth_obj_val(a, t)

        b_ub = np.full(num_rows, M)
        bounds = [(0, None)] * n
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_budget, 'presolve': True})
        if res.status == 0:
            b = np.maximum(res.x, 0)
            s = b.sum()
            if s > 1e-10:
                return b / s
        return None
    elif method == 'weighted_peak':
        M = c_conv.max()
        c_range = c_conv.max() - c_conv.min()
        if c_range < 1e-15:
            w = np.ones(num_rows)
        else:
            w = 1.0 + 8.0 * ((c_conv - c_conv.min()) / c_range) ** 2
        A_w = A * w[:, None]
        b_ub = w * M
        c_obj = -np.ones(n)
        bounds = [(0, None)] * n
        res = linprog(c_obj, A_ub=A_w, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_budget, 'presolve': True})
        if res.status == 0:
            b = np.maximum(res.x, 0)
            s = b.sum()
            if s > 1e-10:
                return b / s
        return None
    elif method == 'min_peak':
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_eq = np.ones((1, n))
        b_eq = np.array([1.0])
        A_ub_full = np.zeros((num_rows, n + 1))
        A_ub_full[:, :n] = A
        A_ub_full[:, n] = -1.0
        b_ub = np.zeros(num_rows)
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c_obj, A_ub=A_ub_full, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': time_budget})
        if res.status == 0:
            b = np.maximum(res.x[:n], 0)
            s = b.sum()
            if s > 1e-10:
                return b / s
        return None
    return None


def lp_iterate(a0, time_limit, method='max_sum', max_iters=400, step_init=0.3):
    """LP-based iterative refinement with backtracking line search."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = 2.0 * n * c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    step = step_init
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = lp_direction(a, method)
        if b is None:
            break

        if np.linalg.norm(b - a) < 1e-10:
            break

        found = False
        for t_trial in [step, step * 0.7, step * 0.5, step * 0.3, step * 0.15, step * 0.05]:
            a_new = (1.0 - t_trial) * a + t_trial * b
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            c_new = fast_convolve(a_new)
            obj_new = 2.0 * n * c_new.max()

            if obj_new <= best_obj + 1e-9:
                a = a_new
                c = c_new
                if obj_new < best_obj - 1e-10:
                    best_obj = obj_new
                    best_a = a.copy()
                    no_improve = 0
                    step = min(0.5, step * 1.15)
                else:
                    no_improve += 1
                found = True
                break

        if not found:
            no_improve += 1
            step = max(0.03, step * 0.6)

        if no_improve > 80:
            break

    return best_a, best_obj


def lbfgs_smooth_optimize(a0, time_limit, t_init=15.0, t_min=1.5, t_decay=0.993):
    """L-BFGS-B optimization on log-sum-exp smoothed objective with annealing."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = 2.0 * n * c.max()
    best_a = a.copy()

    t_start = time.monotonic()
    t_cur = t_init
    stall_count = 0

    while time.monotonic() - t_start < time_limit:
        prev_best = best_obj

        def obj_wrapper(x):
            x_pos = np.maximum(x, 0.0)
            s_x = x_pos.sum()
            if s_x < 1e-15:
                return 1e10
            x_norm = x_pos / s_x
            return smooth_obj_val(x_norm, t_cur)

        def grad_wrapper(x):
            x_pos = np.maximum(x, 0.0)
            s_x = x_pos.sum()
            if s_x < 1e-15:
                return np.ones(len(x))
            x_norm = x_pos / s_x
            g = smooth_grad_fft(x_norm, t_cur)
            return g / s_x

        a_proj = project_simplex(a)
        res = minimize(obj_wrapper, a_proj, jac=grad_wrapper, method='L-BFGS-B',
                       bounds=[(0, None)] * n,
                       options={'maxiter': 300, 'gtol': 1e-7, 'maxfun': 800})

        if res.x is not None:
            a_res = project_simplex(res.x)
            s_res = a_res.sum()
            if s_res > 1e-15:
                a_res = a_res / s_res
                c_res = fast_convolve(a_res)
                obj_res = 2.0 * n * c_res.max()
                if obj_res < best_obj - 1e-10:
                    best_obj = obj_res
                    best_a = a_res.copy()
                    a = a_res
                    stall_count = 0
                else:
                    stall_count += 1
                    a = a_res
            else:
                break

        t_cur = max(t_min, t_cur * t_decay)

        if stall_count > 5:
            break
        if time.monotonic() - t_start > time_limit * 0.8:
            break

    c_final = fast_convolve(best_a)
    return best_a, 2.0 * n * c_final.max()


def peak_transfer_perturb(a, n_transfers=12, transfer_scale=0.008, seed=None):
    """Transfer mass from peak-contributing positions to non-contributing ones."""
    n = len(a)
    c = fast_convolve(a)
    # Use top 3 peaks
    top_peaks = np.argsort(c)[::-1][:3]

    blame = np.zeros(n)
    for pk in top_peaks:
        pk = int(pk)
        j_start = max(0, pk - n + 1)
        j_end = min(n - 1, pk)
        for j in range(j_start, j_end + 1):
            blame[j] += a[pk - j]

    blame_sorted = np.argsort(blame)[::-1]
    sources = blame_sorted[:max(1, n // 6)]
    targets = blame_sorted[n // 6:]

    rng = np.random.default_rng(seed)
    a_new = a.copy()

    for _ in range(n_transfers):
        si = rng.choice(sources)
        ti = rng.choice(targets)
        if si == ti:
            continue
        delta = transfer_scale * a[si]
        if delta > 0 and a_new[si] > delta * 1.1:
            a_new[si] -= delta
            a_new[ti] += delta

    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def random_perturb(a, scale, seed=None):
    """Simple random perturbation projected to simplex."""
    rng = np.random.default_rng(seed)
    noise = rng.standard_normal(len(a)) * scale
    return project_simplex(a + noise)


def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    inits['uniform'] = np.ones(n) / n

    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps

    x = np.arange(n, dtype=np.float64)
    for freq in [1, 2, 3, 5, 7]:
        a = 1.0 + 0.9 * np.cos(2.0 * np.pi * freq * x / n)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum > 1e-10:
            inits[f'cos{freq}'] = a / a_sum

    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g

    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    a = np.zeros(n)
    a[:n // 4] = 1.0
    a[3 * n // 4:] = 1.0
    a /= a.sum()
    inits['two_lobe'] = a

    a = np.sin(np.pi * x / n) ** 2
    a /= a.sum()
    inits['sin2'] = a

    a = np.ones(n)
    a[n // 3:2 * n // 3] = 0.3
    a /= a.sum()
    inits['dip'] = a

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point: multi-phase search to minimize the evaluation objective.
    Combines LP direction finding, L-BFGS smooth optimization, structured
    perturbations, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: len={len(p)}")
    except Exception:
        pass

    def try_update(a):
        nonlocal best_sequence, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  New best: {score:.6f} (n={len(a)})")

    # Determine lengths to try
    lengths = [512]
    if parent is not None:
        L = len(parent)
        if L != 512 and 64 <= L <= 2048:
            lengths.insert(0, L)
    for L in [384, 448, 640]:
        if L not in lengths:
            lengths.append(L)

    seen = set()
    unique_lengths = []
    for L in lengths:
        if L not in seen and 32 <= L <= 4096:
            unique_lengths.append(L)
            seen.add(L)
    lengths = unique_lengths

    print(f"Lengths to try: {lengths}")
    print(f"Time budget: {budget_s}s, deadline in {deadline - time.monotonic():.0f}s")

    # === Phase 1: LP-based exploration across lengths and inits ===
    print("\n=== Phase 1: LP Exploration ===")
    t1_start = time.monotonic()

    for n in lengths:
        if time.monotonic() > deadline - 600:
            break
        print(f"  n={n}")

        inits = generate_inits(n, parent, rng)
        priority = ['parent', 'uniform', 'random', 'cos1', 'sin2', 'dip', 'two_lobe', 'gauss']
        init_names = [nm for nm in priority if nm in inits] + \
                    [nm for nm in inits if nm not in priority]
        init_names = init_names[:5]

        remain = deadline - time.monotonic() - 500
        n_trials = len(lengths) * len(init_names) * 2
        time_per = max(8.0, remain / max(1, n_trials))

        for name in init_names:
            if time.monotonic() > deadline - 500:
                break
            t_alloc = min(time_per * 1.5, deadline - time.monotonic() - 400)
            if t_alloc < 6:
                break

            a0 = inits[name]
            for method in ['max_sum', 'weighted_peak']:
                t_single = t_alloc / 2.0
                if t_single < 4:
                    continue
                a_opt, _ = lp_iterate(a0, time_limit=t_single, method=method,
                                      max_iters=250, step_init=0.3)
                try_update(a_opt)

    print(f"Phase 1 done: best={best_score:.6f}, elapsed={time.monotonic()-t1_start:.0f}s")

    # === Phase 2: L-BFGS smooth optimization ===
    print("\n=== Phase 2: L-BFGS Smooth Optimization ===")
    t2_start = time.monotonic()

    if best_sequence is not None:
        phase2_deadline = deadline - 350
        for t_init in [20.0, 8.0, 4.0]:
            if time.monotonic() > phase2_deadline:
                break
            t_alloc = min(100.0, phase2_deadline - time.monotonic() - 10)
            if t_alloc < 15:
                break
            print(f"  L-BFGS with t_init={t_init}, time={t_alloc:.0f}s")
            a_opt, _ = lbfgs_smooth_optimize(best_sequence, time_limit=t_alloc,
                                             t_init=t_init, t_min=1.5, t_decay=0.99)
            try_update(a_opt)

    print(f"Phase 2 done: best={best_score:.6f}, elapsed={time.monotonic()-t2_start:.0f}s")

    # === Phase 3: Peak-transfer perturbations + re-optimization ===
    print("\n=== Phase 3: Peak Transfer Perturbations ===")
    t3_start = time.monotonic()

    if best_sequence is not None:
        phase3_deadline = deadline - 200
        for scale in [0.005, 0.01, 0.015, 0.003, 0.02]:
            if time.monotonic() > phase3_deadline:
                break
            a_pert = peak_transfer_perturb(best_sequence, n_transfers=15,
                                           transfer_scale=scale,
                                           seed=int(rng.integers(0, 100000)))
            t_alloc = min(50.0, phase3_deadline - time.monotonic() - 10)
            if t_alloc < 10:
                break
            a_opt, _ = lp_iterate(a_pert, time_limit=t_alloc, method='max_sum',
                                  max_iters=200, step_init=0.25)
            try_update(a_opt)

    # Also try random perturbations
    if best_sequence is not None and time.monotonic() < deadline - 100:
        for scale in [0.003, 0.008, 0.015]:
            if time.monotonic() > deadline - 100:
                break
            a_pert = random_perturb(best_sequence, scale, seed=int(rng.integers(0, 99999)))
            t_alloc = min(40.0, deadline - time.monotonic() - 80)
            if t_alloc < 10:
                break
            a_opt, _ = lp_iterate(a_pert, time_limit=t_alloc, method='weighted_peak',
                                  max_iters=150, step_init=0.2)
            try_update(a_opt)

    print(f"Phase 3 done: best={best_score:.6f}, elapsed={time.monotonic()-t3_start:.0f}s")

    # === Phase 4: Final LP polish with multiple methods ===
    print("\n=== Phase 4: Final LP Polish ===")
    t4_start = time.monotonic()

    if best_sequence is not None:
        phase4_deadline = deadline - 60
        for method in ['max_sum', 'weighted_peak', 'min_peak']:
            if time.monotonic() > phase4_deadline:
                break
            t_alloc = min(45.0, phase4_deadline - time.monotonic() - 5)
            if t_alloc < 8:
                break
            a_opt, _ = lp_iterate(best_sequence, time_limit=t_alloc, method=method,
                                  max_iters=400, step_init=0.2)
            try_update(a_opt)

    print(f"Phase 4 done: best={best_score:.6f}, elapsed={time.monotonic()-t4_start:.0f}s")

    # === Phase 5: Final L-BFGS with low temperature ===
    print("\n=== Phase 5: Final L-BFGS Polish ===")

    if best_sequence is not None:
        t_alloc = max(10.0, deadline - time.monotonic() - 5)
        if t_alloc > 15:
            a_opt, _ = lbfgs_smooth_optimize(best_sequence, time_limit=t_alloc,
                                             t_init=5.0, t_min=1.0, t_decay=0.985)
            try_update(a_opt)

    # === Phase 6: One more round of perturbations if time allows ===
    if best_sequence is not None and time.monotonic() < deadline - 30:
        for _ in range(3):
            if time.monotonic() > deadline - 25:
                break
            a_pert = peak_transfer_perturb(best_sequence, n_transfers=8,
                                           transfer_scale=0.004,
                                           seed=int(rng.integers(0, 99999)))
            t_alloc = max(5.0, deadline - time.monotonic() - 10)
            if t_alloc < 8:
                break
            a_opt, _ = lp_iterate(a_pert, time_limit=t_alloc, method='max_sum',
                                  max_iters=100, step_init=0.15)
            try_update(a_opt)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128) / 128
        print("WARNING: Using fallback uniform sequence!")

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\n{'='*50}")
    print(f"FINAL: score={best_score:.6f}, n={len(best_sequence)}")
    print(f"Total time: {time.monotonic() - (deadline - budget_s + 20):.0f}s")
    print(f"{'='*50}")

    return [float(x) for x in best_sequence]
