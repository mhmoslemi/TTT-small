import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_lse_gradient(a, p):
    """Compute grad[j] = sum_l a[l]*p[j+l] efficiently using FFT."""
    n = len(a)
    a_r = a[::-1]
    # Full convolution of a_r (length n) and p (length 2n-1)
    # Result length: n + (2n-1) - 1 = 3n - 2
    csize = 3 * n - 2
    fsize = 1 << (csize - 1).bit_length()
    Ar = np.fft.rfft(a_r, fsize)
    P = np.fft.rfft(p, fsize)
    conv = np.fft.irfft(Ar * P, fsize)[:csize]
    return conv[n-1:n-1+n]

def lse_objective_and_grad(a, t):
    """Compute LSE-smoothed objective and gradient (for sum(a)=1)."""
    n = len(a)
    c = fast_convolve(a)
    tc = t * c
    tc_shift = tc - tc.max()
    exp_c = np.exp(tc_shift)
    sum_exp = exp_c.sum()
    p = exp_c / sum_exp
    g = (1.0/t) * (np.log(sum_exp) + tc.max())  # LSE value
    
    # Gradient of g w.r.t. a
    grad_g = compute_lse_gradient(a, p)
    
    # Full objective: f = 2n * g (since sum(a) = 1)
    obj = 2.0 * n * g
    grad = 2.0 * n * grad_g
    
    return obj, grad, c

def build_conv_matrix(a):
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A

def lp_refine(a, M, time_limit=30):
    """LP: max sum(b) s.t. b >= 0, conv(a,b) <= M."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    t0 = time.monotonic()
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using projected gradient with
log-sum-exp smoothing, Adam-like adaptive learning rates, LP polishing,
and multi-scale exploration. Combines smooth optimization with exact
peak-tracking for rapid convergence to near-optimal sequences.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_cross_grad(a, p):
    """
    Compute grad[j] = sum_l a[l] * p[j+l] for j=0..n-1
    where a has length n and p has length 2n-1.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    a_r = a[::-1]
    csize = 3 * n - 2
    fsize = 1 << (csize - 1).bit_length()
    Ar = np.fft.rfft(a_r, fsize)
    P = np.fft.rfft(p, fsize)
    conv = np.fft.irfft(Ar * P, fsize)[:csize]
    return conv[n - 1:n - 1 + n]


def lse_obj_grad(a, t):
    """
    Compute LSE-smoothed objective and gradient for sum(a)=1.
    Returns (objective, gradient, convolution).
    """
    n = len(a)
    c = fast_convolve(a)
    tc = t * c
    tc_max = tc.max()
    exp_c = np.exp(tc - tc_max)
    sum_exp = exp_c.sum()
    p = exp_c / sum_exp
    g = (1.0 / t) * (np.log(sum_exp) + tc_max)
    grad_g = compute_cross_grad(a, p)
    obj = 2.0 * n * g
    grad = 2.0 * n * grad_g
    return obj, grad, c


def variance_obj_grad(a):
    """
    Compute sum(c_k^2) objective and gradient as an alternative smooth loss.
    For sum(a)=1, this encourages flat autocorrelation.

    n = len(a)
    c = fast_convolve(a)
    obj = np.dot(c, c)
    grad = 2.0 * compute_cross_grad(a, c)
    return obj, grad, c

def lp_polish(a, time_limit=60):
    """LP-based iterative polishing."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = 2.0 * n * c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    step = 0.3
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > 150:
            break

        M = c.max()
        A = build_conv_matrix(a)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')

        if res.status != 0:
            break

        b = np.maximum(res.x, 0)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum

        if np.linalg.norm(b - a) < 1e-10:
            break

        a_new = (1.0 - step) * a + step * b
        a_new = np.maximum(a_new, 0)
        a_sum = a_new.sum()
        if a_sum < 1e-15:
            break
        a_new = a_new / a_sum

        c_new = fast_convolve(a_new)
        obj_new = 2.0 * n * c_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            step = min(0.7, step * 1.03)
        else:
            no_improve += 1
            step = max(0.05, step * 0.93)

        a = a_new
        c = c_new

    return best_a, best_obj


def generate_structured_inits(n, rng):
    """Generate diverse structured initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Cosine profiles
    for freq in [1, 2, 3, 5]:
        for amp in [0.3, 0.6, 0.85]:
            a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)
            a = np.maximum(a, 0.0)
            if a.sum() > 1e-10:
                a /= a.sum()
                inits[f'cos_f{freq}_a{amp}'] = a

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g

    # Asymmetric ramp
    ramp = np.linspace(0.1, 1.0, n)
    ramp /= ramp.sum()
    inits['ramp'] = ramp

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Sparse
    a = np.zeros(n)
    mask = rng.random(n) > 0.7
    a[mask] = 1.0
    if a.sum() > 1e-10:
        a /= a.sum()
        inits['sparse'] = a

    return inits

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using projected gradient with
log-sum-exp smoothing, variance-based regularization, Adam optimizer,
LP polishing, and multi-scale exploration. Key innovations over prior
approach: correct gradient computation (factor of 2 from bilinear
convection), Adam adaptive learning rates, objective switching to escape
plateaus, and targeted temperature adaptation.
"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_cross_grad(a, p):
    """
    Compute cross_grad[j] = sum_l a[l] * p[j+l] for j=0..n-1.
    a has length n, p has length 2n-1.
    This is the core building block for convolution gradients.
    Uses FFT: result = conv(a[::-1], p)[n-1 : 2n-1]
    """
    n = len(a)
    a_r = a[::-1]
    csize = 3 * n - 2
    fsize = 1 << (csize - 1).bit_length()
    Ar = np.fft.rfft(a_r, fsize)
    P = np.fft.rfft(p, fsize)
    conv = np.fft.irfft(Ar * P, fsize)[:csize]
    return conv[n - 1:n - 1 + n]


def lse_obj_and_grad(a, t):
    """
    Compute LSE-smoothed objective and true gradient.
    Objective: 2n * (1/t) * log(sum_k exp(t*c_k))
    Gradient: 4n * compute_cross_grad(a, p) where p = softmax(t*c)
    Returns (lse_value, true_objective, gradient, convolution).
    """
    n = len(a)
    c = fast_convolve(a)
    tc = t * c
    tc_max = tc.max()
    exp_c = np.exp(tc - tc_max)
    sum_exp = exp_c.sum()
    p = exp_c / sum_exp
    lse_val = (1.0 / t) * (np.log(sum_exp) + tc_max)
    grad = 4.0 * n * compute_cross_grad(a, p)
    true_obj = 2.0 * n * c.max()
    return lse_val, true_obj, grad, c


def variance_obj_and_grad(a):
    """
    Compute sum(c_k^2) objective and gradient.
    Gradient: 4 * compute_cross_grad(a, c)
    Returns (variance_value, true_objective, gradient, convolution).
    """
    n = len(a)
    c = fast_convolve(a)
    var_val = np.dot(c, c)
    grad = 4.0 * compute_cross_grad(a, c)
    true_obj = 2.0 * n * c.max()
    return var_val, true_obj, grad, c
