"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach combining:
1. Exact subgradient with Nesterov momentum for fast local optimization
2. LP-based direction finding (from the state-of-the-art) for escaping local minima
3. Coordinate descent and random perturbations for final refinement
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft

def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]

def objective_value(a):
    """Compute the objective 2n*max(conv)/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    conv = fast_autocorr(a)
    return 2.0 * n * conv.max() / (s * s)

def smooth_max(x, beta):
    """Smooth approximation of max using log-sum-exp."""
    x_max = x.max()
    shifted = np.clip(beta * (x - x_max), -100, 100)
    exp_s = np.exp(shifted)
    total = exp_s.sum()
    if total < 1e-300:
        return x_max, np.zeros_like(x)
    w = exp_s / total
    return x_max + np.log(total) / beta, w

def grad_smooth_conv(a, w):
    """Gradient of weighted sum of convolution w.r.t. a."""
    n = len(a)
    N = 1 << (len(w) + n - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(np.conj(W) * A, N))[:n]
    return 2.0 * grad

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho >= n:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def grad_exact_conv(a, peak_indices):
    """Exact gradient at peak positions of autocorrelation."""
    n = len(a)
    grad = np.zeros(n)
    for k in peak_indices:
        # d(conv[k])/d(a[i]) = a[k-i] if 0 <= k-i < n
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            grad[j] += a[k - j]
    return grad / len(peak_indices)  # Average over peaks

def nesterov_pg(a0, beta, n_iter=500, lr_init=0.1, momentum=0.9):
    """Nesterov accelerated projected gradient with smooth max."""
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    y = a.copy()
    v = np.zeros(n)
    
    best_obj = float('inf')
    best_a = a.copy()
    
    for it in range(n_iter):
        lr = lr_init / (1.0 + 0.01 * it)
        
        # Nesterov step
        conv_y = fast_autocorr(y)
        sm, w = smooth_max(conv_y, beta)
        obj = 2.0 * n * sm
        
        if obj < best_obj:
            best_obj = obj
            best_a = y.copy()
        
        if obj < 1.002:  # Close to theoretical minimum
            break
        
        grad_y = grad_smooth_conv(y, w)
        
        # Gradient step
        y_new = y - lr * grad_y
        # Projection
        y_new = project_simplex(y_new)
        
        # Momentum update
        v = momentum * v + (y_new - y)
        y = y_new + momentum * v
        y = project_simplex(y)
    
    return best_a, best_obj

def solve_lp_direction(f, M, solver=cp.SCS):
    """Solve the LP to find direction g."""
    n = len(f)
    
    # Build constraint matrix A where A[k,j] = f[j] if 0 <= k-j < n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = j + n - 1
        A[k_start:k_end+1, j] = f[j]
    
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=solver, max_iters=5000)
        if g.value is not None and np.all(np.isfinite(g.value)):
            return g.value
    except Exception:
        pass
    
    return None

def lp_update_step(f, t_values=(0.1, 0.2, 0.3, 0.5)):
    """One step of the LP-based update."""
    n = len(f)
    # Normalize f so sum = sqrt(2n)
    s = f.sum()
    if s < 1e-10:
        return f
    f_norm = f * np.sqrt(2.0 * n) / s
    
    M = fast_autocorr(f_norm).max()
    
    g = solve_lp_direction(f_norm, M)
    if g is None or g.sum() < 1e-10:
        return f
    
    # Rescale g to have sum = sqrt(2n)
    g = g * np.sqrt(2.0 * n) / g.sum()
    
    # Try different t values
    best_f = f
    best_obj = objective_value(f)
    
    for t in t_values:
        f_new = (1 - t) * f_norm + t * g
        f_new = np.maximum(f_new, 0)
        s_new = f_new.sum()
        if s_new < 1e-10:
            continue
        f_new = f_new / s_new
        obj = objective_value(f_new)
        if obj < best_obj:

"""
Hybrid optimization combining Nesterov accelerated gradient, LP-based direction
finding (from the state-of-the-art LP formulation), and coordinate descent to
minimize 2n*max(conv(a,a))/(sum(a))^2. Key innovations: exact subgradient with
momentum, LP-based exploration to escape local minima, and adaptive phase switching.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft


def fast_autocorr(a):
    """Compute linear autocorrelation via FFT (matches np.convolve(a,a))."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def objective_value(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    conv = fast_autocorr(a)
    return float(2.0 * n * conv.max() / (s * s))


def smooth_max(x, beta):
    """Smooth max via log-sum-exp with numerical stability."""
    x_max = x.max()
    shifted = np.clip(beta * (x - x_max), -100, 100)
    exp_s = np.exp(shifted)
    total = exp_s.sum()
    if total < 1e-300:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_s / total
    return x_max + np.log(total) / beta, w


def grad_smooth_conv(a, w):
    """Gradient of sum_k w[k]*conv[k] w.r.t. a using FFT."""
    n = len(a)
    N = 1 << (len(w) + n - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(np.conj(W) * A, N))[:n]
    return 2.0 * grad


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.argmax(~cond))
    if rho >= n:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def nesterov_pg(a0, beta, n_iter=300, lr_init=0.08, momentum=0.85):
    """Nesterov accelerated projected gradient with smooth max objective."""
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    prev_a = a.copy()

    best_obj = float('inf')
    best_a = a.copy()

    for it in range(n_iter):
        lr = lr_init / (1.0 + 0.003 * it)

        # Nesterov lookahead
        y = a + momentum * (a - prev_a)
        y = project_simplex(y)

        conv_y = fast_autocorr(y)
        sm, w = smooth_max(conv_y, beta)
        obj = 2.0 * n * sm

        if obj < best_obj:
            best_obj = obj
            best_a = y.copy()

        if best_obj < 1.002:
            break

        grad_y = grad_smooth_conv(y, w)

        a_new = y - lr * grad_y
        a_new = project_simplex(a_new)

        prev_a = a
        a = a_new

    # Final exact evaluation
    exact_obj = objective_value(best_a)
    if exact_obj < best_obj:
        best_obj = exact_obj
    return best_a, exact_obj

    n = len(a0)

        if best_obj < 1.002:

        grad_norm = np.linalg.norm(grad)

        for _ in range(8):
            a_new = project_simplex(a - t * grad)
            obj_new = 2.0 * n * smooth_max(fast_autocorr(a_new), beta)[0]
            if obj_new <= obj + 1e-12:
                break
            t *= 0.5

        a = a_new

    exact_obj = objective_value(best_a)
    return best_a, exact_obj


def build_lp_matrix(f):
    """Build the constraint matrix A where A[k,j] = f[j] if 0 <= k-j < n."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f[j]
    return A


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. g >= 0, A@f <= M where A is cross-corr matrix."""
    n = len(f)
    A = build_lp_matrix(f)

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver=cp.SCS, max_iters=10000, eps=1e-6)
        if g.value is not None and np.all(np.isfinite(g.value)):
            gv = np.maximum(g.value, 0)
            if gv.sum() > 1e-10:
                return gv
    except Exception:
        pass

    try:
        prob.clear()
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, max_iters=10000)
        if g.value is not None and np.all(np.isfinite(g.value)):
            gv = np.maximum(g.value, 0)
            if gv.sum() > 1e-10:
                return gv
    except Exception:
        pass

    return None


def lp_step(f, t_candidates=(0.05, 0.1, 0.15, 0.2, 0.3, 0.5)):
    """One LP-based update step. Returns improved sequence or f unchanged."""
    n = len(f)
    s = f.sum()
    if s < 1e-10:
        return f

    # Work in normalization sum = sqrt(2n)
    f_norm = f * np.sqrt(2.0 * n) / s
    M = fast_autocorr(f_norm).max()

    if M < 1e-15:
        return f

    g = solve_lp_direction(f_norm, M)
    if g is None:
        return f

    # Rescale g to same normalization
    g_sum = g.sum()
    if g_sum < 1e-10:
        return f
    g = g * np.sqrt(2.0 * n) / g_sum

    # Try different mixing parameters
    best_f = f
    best_obj = objective_value(f)

    for t in t_candidates:
        f_new = (1.0 - t) * f_norm + t * g
        f_new = np.maximum(f_new, 0.0)
        sn = f_new.sum()
        if sn < 1e-10:
            continue
        f_new = f_new / sn
        obj = objective_value(f_new)
        if obj < best_obj:
            best_obj = obj
            best_f = f_new

    return best_f


def coordinate_descent(a, n_sweeps=5, scores=(0.8, 0.9, 0.95, 1.05, 1.1, 1.2)):
    """Coordinate descent minimizing the exact objective."""
    n = len(a)
    a = a.copy()
    best_obj = objective_value(a)

    for sweep in range(n_sweeps):
        for i in range(n):
            old_val = a[i]
            if old_val < 1e-15:
                continue
            improved = False
            for sc in scores:
                a[i] = old_val * sc
                s = a.sum()
                if s < 1e-10:
                    a[i] = old_val
                    break
                obj = objective_value(a / s)
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    a = a / s
                    improved = True
                    break
            if not improved:
                a[i] = old_val

    return a, best_obj


def momentum_pg_exact(a0, n_iter=500, lr_init=0.05, momentum=0.9):
    """Projected gradient using exact subgradient at the peak."""
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    prev_a = a.copy()

    best_obj = objective_value(a)
    best_a = a.copy()

    for it in range(n_iter):
        lr = lr_init / (1.0 + 0.005 * it)

        conv = fast_autocorr(a)
        max_val = conv.max()
        # Find peak indices (within 1% of max)
        threshold = max_val * 0.99
        peak_idx = np.where(conv >= threshold)[0]

        # Exact subgradient: grad[i] = (1/|P|) * sum_{k in P} a[k-i] for valid k-i
        grad = np.zeros(n)
        for k in peak_idx:
            j_start = max(0, int(k) - n + 1)
            j_end = min(n - 1, int(k))
            for j in range(j_start, j_end + 1):
                grad[j] += a[int(k) - j]
        grad /= len(peak_idx)

        # Nesterov step
        y = a + momentum * (a - prev_a)
        y = project_simplex(y)

        a_new = y - lr * grad
        a_new = project_simplex(a_new)

        obj = objective_value(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

        prev_a = a
        a = a_new

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initial sequences of length n."""
    inits = {}
    inits['uniform'] = np.ones(n) / n

    x = np.arange(n, dtype=np.float64)
    for sf in [0.2, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (sf * n)) ** 2)
        g /= g.sum()
        inits['gauss_%s' % str(sf)] = g

    tri = np.minimum(x, n - 1 - x).astype(float) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri

    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Flat spectrum initializations
    fft_size = 1 << (n - 1).bit_length()
    for trial in range(8):
        A = np.zeros(fft_size, dtype=complex)
        A[0] = 1.0
        phases = rng.uniform(0, 2 * np.pi, fft_size)
        for k in range(1, fft_size // 2 + 1):
            A[k] = np.exp(1j * phases[k])
            if k < fft_size - k:
                A[fft_size - k] = np.exp(-1j * phases[k])
        a = np.real(ifft(A, fft_size))[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a /= s
            inits['flat_%d' % trial] = a

    if parent is not None:
        if len(parent) == n:
            pp = np.maximum(parent, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)),
                           np.maximum(parent, 0))
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_r'] = pp / ps

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: multi-phase hybrid optimization combining Nesterov PG,
    exact subgradient momentum, LP-based direction finding, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print("Parent: n=%d, sum=%.4f" % (len(p), p.sum()))
    except Exception:
        pass

    def update_best(a):
        nonlocal best_sequence, best_score
        if a is None:
            return float('inf')
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print("  NEW BEST: %.6f (n=%d, t=%ds)" % (score, len(a), int(time.monotonic() - t_start)))
        return score if np.isfinite(score) else float('inf')

    # Determine starting length
    n = len(parent) if parent is not None else 256

    # ========== Phase 1: Quick optimization (0-250s) ==========
    print("=== Phase 1: Quick Nesterov + Exact PG ===")
    phase1_end = t_start + 250.0

    if parent is not None:
        a = np.maximum(parent, 0)
        a = a / a.sum()
    else:
        a = np.ones(n) / n

    # Smooth max Nesterov with decreasing beta
    for beta in [80.0, 40.0, 20.0, 10.0, 5.0]:
        if time.monotonic() > phase1_end - 15:
            break
        t_alloc = min(40.0, phase1_end - time.monotonic() - 10)
        n_iter = int(min(300, max(50, t_alloc * 8)))
        a, _ = nesterov_pg(a, beta=beta, n_iter=n_iter, lr_init=0.06, momentum=0.85)
        update_best(a)

    # Exact subgradient momentum
    if time.monotonic() < phase1_end - 20:
        t_alloc = min(50.0, phase1_end - time.monotonic() - 10)
        n_iter = int(min(400, max(100, t_alloc * 10)))
        a, _ = momentum_pg_exact(a, n_iter=n_iter, lr_init=0.03, momentum=0.8)
        update_best(a)

    # Try other initializations briefly
    if time.monotonic() < phase1_end - 30:
        inits = generate_inits(n, parent, rng)
        for name in ['flat_0', 'flat_1', 'flat_2', 'gauss_0.35', 'random']:
            if time.monotonic() > phase1_end - 10:
                break
            if name in inits:
                a_t, _ = nesterov_pg(inits[name], beta=20.0, n_iter=80, lr_init=0.05, momentum=0.8)
                update_best(a_t)

    print("Phase 1 done. Best: %.6f" % best_score)

    # ========== Phase 2: LP-assisted optimization (250-900s) ==========
    print("=== Phase 2: LP-assisted Optimization ===")
    phase2_end = t_start + 900.0

    if best_sequence is not None:
        a = best_sequence / best_sequence.sum()
    else:
        a = a / a.sum()

    no_improve_count = 0
    last_best = best_score

    while time.monotonic() < phase2_end - 40:
        # Gradient refinement
        a, _ = nesterov_pg(a, beta=15.0, n_iter=80, lr_init=0.03, momentum=0.8)
        update_best(a)

        if time.monotonic() > phase2_end - 80:
            break

        # LP step
        t_lp_start = time.monotonic()
        a_lp = lp_step(a, t_candidates=(0.05, 0.1, 0.2, 0.3, 0.5))
        update_best(a_lp)
        t_lp = time.monotonic() - t_lp_start

        if best_score < last_best - 1e-8:
            no_improve_count = 0
            last_best = best_score
        else:
            no_improve_count += 1

        a = a_lp

        # Safety: if LP is too slow, skip more of them
        if t_lp > 30:
            print("  LP took %.1fs, reducing frequency" % t_lp)

    print("Phase 2 done. Best: %.6f" % best_score)

    # ========== Phase 3: Deep local refinement (900-1550s) ==========
    print("=== Phase 3: Deep Refinement ===")
    phase3_end = t_start + 1550.0

    if best_sequence is not None:
        a = best_sequence / best_sequence.sum()
    n_best = len(a)

    while time.monotonic() < phase3_end - 30:
        # Low-beta gradient
        a, _ = nesterov_pg(a, beta=8.0, n_iter=150, lr_init=0.02, momentum=0.75)
        update_best(a)

        if time.monotonic() > phase3_end - 60:
            break

        # Exact subgradient with low momentum
        a, _ = momentum_pg_exact(a, n_iter=200, lr_init=0.015, momentum=0.7)
        update_best(a)

        if time.monotonic() > phase3_end - 30:
            break

        # Coordinate descent
        a_cd, _ = coordinate_descent(a, n_sweeps=5)
        update_best(a_cd)
        a = a_cd

    print("Phase 3 done. Best: %.6f" % best_score)

    # ========== Phase 4: Explore nearby lengths + perturbations (1550-1750s) ==========
    print("=== Phase 4: Exploration ===")
    phase4_end = t_start + 1750.0

    # Try random perturbations from best
    if best_sequence is not None:
        a = best_sequence / best_sequence.sum()
        n_best = len(a)

        for trial in range(15):
            if time.monotonic() > phase4_end - 20:
                break
            # Different perturbation scales
            scale = 0.005 + 0.03 * rng.uniform()
            noise = rng.standard_normal(n_best) * scale
            a_p = project_simplex(a + noise)

            # Quick refine
            a_opt, _ = nesterov_pg(a_p, beta=12.0, n_iter=80, lr_init=0.02, momentum=0.8)
            update_best(a_opt)

    # Try a different length
    if time.monotonic() < phase4_end - 60:
        alt_n = n + 32 if n < 400 else n - 32
        if alt_n >= 64:
            inits_alt = generate_inits(alt_n, parent, rng)
            if best_sequence is not None:
                pp = np.interp(np.linspace(0, 1, alt_n), np.linspace(0, 1, n),
                               best_sequence)
                pp = np.maximum(pp, 0)
                ps = pp.sum()
                if ps > 1e-10:
                    inits_alt['from_best'] = pp / ps
            a0 = inits_alt.get('from_best', inits_alt['uniform'])
            t_alloc = min(80.0, phase4_end - time.monotonic() - 20)
            n_iter = int(min(200, max(50, t_alloc * 5)))
            a_opt, _ = nesterov_pg(a0, beta=20.0, n_iter=n_iter, lr_init=0.05, momentum=0.85)
            update_best(a_opt)

    print("Phase 4 done. Best: %.6f" % best_score)

    # ========== Phase 5: Final polish (1750-1770s) ==========
    print("=== Phase 5: Final Polish ===")
    if best_sequence is not None:
        a = best_sequence / best_sequence.sum()

        # Very careful gradient
        if time.monotonic() < deadline - 15:
            a, _ = nesterov_pg(a, beta=5.0, n_iter=200, lr_init=0.01, momentum=0.6)
            update_best(a)

        # Final coordinate descent
        if time.monotonic() < deadline - 10:
            a_cd, _ = coordinate_descent(a, n_sweeps=8)
            update_best(a_cd)

        # Tiny perturbations
        if time.monotonic() < deadline - 5:
            for _ in range(5):
                if time.monotonic() > deadline - 3:
                    break
                noise = rng.standard_normal(len(a)) * 0.001
                a_p = project_simplex(a + noise)
                a_opt, _ = nesterov_pg(a_p, beta=6.0, n_iter=50, lr_init=0.008, momentum=0.7)
                update_best(a_opt)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(n) / n
        best_score = objective_value(best_sequence)
        final = [float(x) for x in best_sequence]
        s = float(evaluate_sequence(final))
        if np.isfinite(s):
            best_score = s

    final_score = float(evaluate_sequence([float(x) for x in best_sequence]))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print("\nFINAL SCORE: %.6f (n=%d)" % (best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
