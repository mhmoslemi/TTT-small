import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_cross_conv_matrix_sparse(a):
    """
    Build sparse matrix C such that C @ g = a * g (full linear convolution).
    C is (2n-1) x n.
    """
    n = len(a)
    rows = []
    cols = []
    vals = []
    for i in range(n):
        for j in range(n):
            rows.append(i + j)
            cols.append(j)
            vals.append(a[i])
    C = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return C


def build_cross_conv_matrix_dense(a):
    """Build dense matrix C such that C @ g = a * g."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C

import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    """Build dense matrix C such that C @ g = a * g (full linear convolution)."""

    b_eq = np.array([1.0])

        gs = g.sum()
        if gs > 1e-15:

def subgradient_optimize(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=30, temp=0.0002):
    """Projected subgradient descent with momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a))
    a = a / a.sum()
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_M = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        for _ in range(50):
            if time.monotonic() - t0 > time_limit:
                break
            
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            peak_vals = conv[peaks]
            scaled = peak_vals / temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum < 1e-15:
                w = np.zeros(len(peaks))
                w[np.argmax(peak_vals)] = 1.0
            else:
                w /= w_sum
            
            sg = np.zeros(n)
            for i in range(len(peaks)):
                if w[i] < 1e-12:
                    continue
                k = int(peaks[i])
                j_start = max(0, k - n + 1)
                j_end = min(n - 1, k)
                idx = np.arange(j_start, j_end + 1)
                sg[idx] += w[i] * 2.0 * a[k - idx]
            
            v = momentum * v - lr * sg
            a_new = a + v
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a = a_new / s_new
            else:
                a = best_a.copy()
            
            conv = fast_convolve(a)
            M = conv.max()
            
            if M < best_M - 1e-14:
                best_M = M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        lr *= 0.998
        temp *= 1.003
        momentum = min(0.97, momentum + 0.0002)
        
        if no_improve >= 250:
            break
    
    return best_a, best_M


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Skewed uniform (linear ramp)
    ramp = np.linspace(0.3, 1.0, n)
    inits['ramp'] = ramp / ramp.sum()
    inits['ramp_soft'] = (0.5 + 0.5 * x / n) / (0.5 + 0.5 * x / n).sum()
    
    # Cosine window
    cos_w = np.cos(np.pi * x / (2 * n)) ** 2
    inits['cos_window'] = cos_w / cos_w.sum()
    
    # Two bumps
    bump1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2)
    bump2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    inits['two_bump'] = (bump1 + bump2) / (bump1 + bump2).sum()
    
    # Edge concentration
    edge = np.exp(-x / (0.2 * n)) + np.exp(-(n-1-x) / (0.2 * n))
    inits['edge'] = edge / edge.sum()
    
    # Sine
    inits['sine'] = (np.sin(np.pi * x / n) + 0.2) / (np.sin(np.pi * x / n) + 0.2).sum()
    
    # Quadratic
    quad = (x - n/2)**2 + n
    inits['quad'] = quad / quad.sum()
    
    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_resample'] = pp / pps
    
    return inits

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_cross_conv_matrix(a):
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def lp_minimize_peak(a, conv_a, C):
    """
    Solve: min t s.t. C@g - 0.5*t <= conv_a/2, sum(g)=1, g>=0, t>=0
    Linearizes (g*g)[k] at point a.
    """
    n = len(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.empty((2 * n - 1, n + 1))
    A_ub[:, :n] = C
    A_ub[:, n] = -0.5
    b_ub = conv_a / 2.0
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            g = np.maximum(res.x[:n], 0.0)
            gs = g.sum()
            if gs > 1e-15:
                g /= gs
            return g, res.x[-1]
    except Exception:
        pass
    return None, None


def lp_maximize_direction(a, M, C):
    """
    Fallback: max sum(g) s.t. C@g <= M, g>=0.
    Returns normalized g.
    """
    n = len(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-15:
                return g / gs
    except Exception:
        pass
    return None


def lp_optimize(a0, time_limit, verbose=False, verbose_period=100):
    """LP-based optimization loop."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a))
    else:
        a = a / s
    n = len(a)
    
    conv_a = fast_convolve(a)
    best_M = conv_a.max()
    best_a = a.copy()
    C = build_cross_conv_matrix(a)
    
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 40
    iteration = 0
    perturb_count = 0
    
    while time.monotonic() - t0 < time_limit:
        # Try direct LP
        g, t_lp = lp_minimize_peak(a, conv_a, C)
        
        if g is not None and t_lp < best_M - 1e-12:
            # Check actual improvement
            conv_g = fast_convolve(g)
            M_g = conv_g.max()
            if M_g < best_M - 1e-14:
                best_a = g.copy()
                best_M = M_g
                a = g.copy()
                conv_a = conv_g
                C = build_cross_conv_matrix(a)
                no_improve = 0
            else:
                # Blend
                improved = False
                for alpha in [0.7, 0.5, 0.3, 0.1]:
                    a_try = (1 - alpha) * best_a + alpha * g
                    a_try = np.maximum(a_try, 0.0)
                    st = a_try.sum()
                    if st > 1e-15:
                        a_try /= st
                        Mt = fast_convolve(a_try).max()
                        if Mt < best_M - 1e-14:
                            best_a = a_try
                            best_M = Mt
                            a = a_try.copy()
                            conv_a = fast_convolve(a)
                            C = build_cross_conv_matrix(a)
                            no_improve = 0
                            improved = True
                            break
                if not improved:
                    no_improve += 1
        else:
            # Fallback: simple LP
            g2 = lp_maximize_direction(a, best_M * 1.001, C)
            if g2 is not None:
                d = g2 - a
                if np.linalg.norm(d) > 1e-12:
                    improved = False
                    for alpha in [1.0, 0.5, 0.3, 0.1]:
                        a_try = a + alpha * d
                        a_try = np.maximum(a_try, 0.0)
                        st = a_try.sum()
                        if st > 1e-15:
                            a_try /= st
                            Mt = fast_convolve(a_try).max()
                            if Mt < best_M - 1e-14:
                                best_a = a_try
                                best_M = Mt
                                a = a_try.copy()
                                conv_a = fast_convolve(a)
                                C = build_cross_conv_matrix(a)
                                no_improve = 0
                                improved = True
                                break
                    if not improved:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        
        iteration += 1
        
        if verbose and iteration % verbose_period == 0:
            print(f"      LP iter {iteration}: M={best_M:.10f}, elapsed={time.monotonic()-t0:.1f}s")
        
        if no_improve >= max_no_improve:
            rng = np.random.default_rng(hash((iteration, perturb_count)))
            scale = 0.003 + 0.002 * perturb_count
            noise = rng.standard_normal(n) * scale
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a /= s
            else:
                a = best_a.copy()
            conv_a = fast_convolve(a)
            C = build_cross_conv_matrix(a)
            no_improve = 0
            perturb_count += 1
            max_no_improve = min(80, 40 + 10 * perturb_count)
            if perturb_count > 20:
                break
    
    return best_a, best_M


def subgradient_optimize(a0, time_limit, lr_init=0.01, momentum=0.92, top_k=40, temp=0.00015):
    """Projected subgradient with softmax-weighted peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a))
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_M = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        for _ in range(40):
            if time.monotonic() - t0 > time_limit:
                break
            
            if top_k >= n:
                peaks = np.arange(n)
            else:
                peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
            sc = pv / temp
            sc -= sc.max()
            w = np.exp(sc)
            ws = w.sum()
            if ws < 1e-15:
                w = np.zeros(len(peaks))
                w[np.argmax(pv)] = 1.0
            else:
                w /= ws
            
            sg = np.zeros(n)
            for i in range(len(peaks)):
                if w[i] < 1e-12:
                    continue
                k = int(peaks[i])
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                sg[idx] += w[i] * 2.0 * a[k - idx]
            
            v = momentum * v - lr * sg
            a_new = a + v
            a_new = np.maximum(a_new, 0.0)
            sn = a_new.sum()
            if sn > 1e-15:
                a = a_new / sn
            else:
                a = best_a.copy()
            
            conv = fast_convolve(a)
            M = conv.max()
            if M < best_M - 1e-14:
                best_M = M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        lr *= 0.998
        temp *= 1.004
        momentum = min(0.97, momentum + 0.0002)
        if no_improve >= 200:
            break
    
    return best_a, best_M


def make_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    inits['rand'] = rng.dirichlet(np.ones(n) * 0.5)
    
    r = np.linspace(0.4, 1.0, n)
    inits['ramp'] = r / r.sum()
    
    cw = np.cos(np.pi * x / (2 * n)) ** 2
    inits['cosw'] = cw / cw.sum()
    
    b1 = np.exp(-0.5 * ((x - 0.3 * n) / (0.12 * n)) ** 2)
    b2 = np.exp(-0.5 * ((x - 0.7 * n) / (0.12 * n)) ** 2)
    inits['twob'] = (b1 + b2) / (b1 + b2).sum()
    
    edge = np.exp(-x / (0.15 * n)) + np.exp(-(n - 1 - x) / (0.15 * n))
    inits['edge'] = edge / edge.sum()
    
    sq = (1.0 + 0.3 * np.cos(2 * np.pi * x / n))
    inits['cosmod'] = sq / sq.sum()
    
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['tri'] = tri / tri.sum()
    
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_r'] = pp / pps
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: direct LP optimization, subgradient polishing,
    adaptive length selection, and perturbation-based restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.4f}")
    except Exception:
        print("No parent available, starting fresh.")
    
    # Phase 1: Quick scan across lengths and inits
    print("=== Phase 1: Quick LP Scan ===")
    t0 = time.monotonic()
    p1_budget = min(350.0, (deadline - time.monotonic()) * 0.3)
    p1_end = t0 + p1_budget
    
    # Choose lengths to try
    lens = []
    if parent is not None:
        lens.append(len(parent))
    for n in [128, 256, 384, 512, 768, 1024]:
        if n not in lens:
            lens.append(n)
    lens = [n for n in lens if 64 <= n <= 2048]
    # Limit to 4 lengths
    lens = lens[:4]
    
    n_inits_per = 3
    t_per = max(12.0, p1_budget / (len(lens) * n_inits_per))
    
    all_results = []
    
    for n in lens:
        if time.monotonic() > p1_end - 25:
            break
        inits = make_inits(n, parent, rng)
        priority = ['parent', 'parent_r', 'uniform', 'ramp', 'cosw', 'twob', 'cosmod', 'rand', 'edge', 'tri']
        names = [nm for nm in priority if nm in inits][:n_inits_per]
        
        for name in names:
            if time.monotonic() > p1_end - 15:
                break
            ta = min(t_per, p1_end - time.monotonic() - 10)
            if ta < 8:

            a0 = inits[name]
            a_opt, M_opt = lp_optimize(a0, time_limit=ta)

                all_results.append((score, n, name, a_opt.copy()))

                    best_score = score
                    best_seq = a_opt.copy()
                    print(f"  [{time.monotonic()-t0:.0f}s] n={n}, {name}: {score:.6f}")
    
    all_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t0:.0f}s). Best: {best_score:.6f}")
    
    # Phase 2: Deep LP on top candidates
    print("=== Phase 2: Deep LP ===")
    t0 = time.monotonic()
    
    # Pick diverse top candidates
    top = []

            top.append((n, ao.copy()))
            seen_n.add(n)
