"""
Accelerated multi-phase optimization for min 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) diverse initializations with low-autocorrelation structure,
(2) LP-based direction finding with adaptive active set,
(3) Nesterov-accelerated smoothed gradient descent,
(4) coordinate descent polish.

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)

def build_LP_matrix(a):
    """Build constraint matrix for LP: (a*b)[k] <= M."""
    n = len(a)
    num_k = 2 * n - 1
    A_ub = np.zeros((num_k, n))
    for k in range(num_k):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A_ub[k, idx] = a[k - idx]
    return A_ub

def solve_lp(a, M, active_indices=None):
    """Solve LP: max sum(b) s.t. b>=0, (a*b)[k] <= M for k in active_indices (or all)."""
    n = len(a)
    num_k = 2 * n - 1
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    if active_indices is not None and len(active_indices) < num_k:
        idx = active_indices
        A_ub = np.zeros((len(idx), n))
        for row, k in enumerate(idx):
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            col_idx = np.arange(i_min, i_max + 1)
            A_ub[row, col_idx] = a[k - col_idx]
        b_ub = np.full(len(idx), M)
    else:
        A_ub = build_LP_matrix(a)
        b_ub = np.full(num_k, M)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def lp_optimize(a0, time_limit, step_init=0.8, active_frac=0.3, relax=1.0, seed=0):

    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)

    else:
        a = a / s
    n = len(a)
    num_k = 2 * n - 1
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    step = step_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        
        # Adaptive active set: top fraction of peaks
        n_active = max(5, int(active_frac * num_k))
        active_idx = np.argpartition(conv, -n_active)[-n_active:]
        
        b = solve_lp(a, M * relax, active_indices=active_idx)
        if b is None:

        if b_sum < 1e-15:

        b = b / b_sum

        if s > 1e-15:

def smooth_gradient_descent(a0, time_limit, beta_init=500.0, lr_init=0.03,
                            momentum=0.9, nesterov=True, seed=0):
    """Nesterov-accelerated smoothed gradient descent on simplex."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    # Nesterov variables
    y = a.copy()
    v = np.zeros(n)
    beta = beta_init
    lr = lr_init
    no_improve = 0
    t_count = 1.0
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        # Compute gradient at y
        conv_y = fast_convolve(y)
        scaled = beta * conv_y
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(conv_y))
            w[np.argmax(conv_y)] = 1.0
        else:
            w /= w_sum
        
        # Gradient: sum_k w[k] * d(conv[k])/da[i]
        # d(conv[k])/da[i] = a[k-i] + a[i-k] (for valid indices in linear conv)
        # Use FFT-based cross-correlation
        size = 2 * n + 1
        fft_size = 1 << (size - 1).bit_length()
        
        w_pad = np.zeros(fft_size)
        w_pad[:len(w)] = w
        y_pad = np.zeros(fft_size)
        y_pad[:n] = y
        
        W = np.fft.fft(w_pad)
        Y = np.fft.fft(y_pad)
        
        # term1[i] = sum_k w[k] * y[k-i] (cross-correlation)
        cross = np.fft.ifft(Y * np.conj(W)).real[:n]
        # term2[i] = sum_k w[k] * y[i-k] (convolution)
        conv_wy = np.fft.ifft(Y * W).real[:n]
        
        grad = cross + conv_wy
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-15:
            grad = grad / g_norm
        
        if nesterov:
            # Nesterov update
            t_new = 1.0 + (t_count + 1) / 2.0
            momentum_coeff = (t_count - 1) / t_new
            t_count = t_new
        else:
            momentum_coeff = momentum
        
        v = momentum_coeff * v - lr * grad
        a_new = project_simplex(a + v)
        
        # Update y for Nesterov
        if nesterov:
            y = a_new + ((t_count - 1) / t_count) * (a_new - a)
            y = project_simplex(y)
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                lr *= 0.85

def coordinate_descent_polish(a0, time_limit, seed=0):
    """Coordinate descent: optimize one coordinate at a time."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a0, 2.0 * len(a0) * fast_convolve(a0).max()
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    order = rng.permutation(n)
    pos = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        i = order[pos % n]
        pos += 1
        
        # Save current value
        ai_old = a[i]
        
        # For coordinate i, the convolution entries that change are those k
        # where the indices (i, k-i) or (k-i, i) are valid
        # (a*a)[k] = sum_j a[j]*a[k-j]
        # When we change a[i], the terms that change are:
        #   (a*a)[i+j] += (a_new[i]-a_old[i]) * a[j] for valid j
        #   (a*a)[j+i] += (a_new[i]-a_old[i]) * a[j] for valid j (same as above by symmetry)
        # Actually (a*a)[k] = sum_j a[j]*a[k-j], so changing a[i] affects:
        #   (a*a)[k] for all k where i or k-i is i, i.e., k=i+j for any j (first term)
        #   and k=k where k-i=i, i.e., k=2i (second term, already counted)
        # More precisely: (a*a)[k] = sum_j a[j]*a[k-j]
        # If we change a[i] by delta, then:
        #   (a*a)[k] changes by delta * a[k-i] (from j=i term) + delta * a[i] if k-i=i... 
        #   Wait, let's be more careful.
        #   (a*a)[k] = sum_j a[j] * a[k-j]
        #   Terms involving a[i]: j=i gives a[i]*a[k-i], and k-j=i gives j=k-i gives a[k-i]*a[i]
        #   So total contribution of a[i] to (a*a)[k] is:
        #     a[i]*a[k-i] (if 0<=k-i<n) + a[k-i]*a[i] (if 0<=k-i<n) = 2*a[i]*a[k-i] (if k!=2i)
        #     a[i]^2 (if k=2i, since j=i and k-j=i are the same)
        #   Actually for k=2i: j=i gives a[i]*a[i]=a[i]^2, and there's no separate term.
        #   Wait no: (a*a)[k] = sum_j a[j]*a[k-j]. For k=2i:
        #     j=i: a[i]*a[2i-i] = a[i]*a[i] = a[i]^2
        #     That's the only term where either index is i (since k-j=i means j=k-i=i).
        #   So for k=2i: contribution is a[i]^2
        #   For k != 2i: contribution is a[i]*a[k-i] + a[k-i]*a[i] = 2*a[i]*a[k-i]
        #     (one from j=i, one from j=k-i, which are different when k!=2i)
        #   But wait, we need k-i to be valid (0 <= k-i < n).
        
        # New (a*a)[k] when a[i] changes to a_new_i:
        #   For k = 2i: (a*a)[k] = old_conv[k] - a[i]^2 + a_new_i^2
        #   For k != 2i, 0 <= k-i < n: (a*a)[k] = old_conv[k] - 2*a[i]*a[k-i] + 2*a_new_i*a[k-i]
        #   For other k: unchanged
        
        # We want to minimize max of new conv subject to a[i] >= 0 and sum stays = 1
        # (rest of a stays fixed, so we need a_new_i >= 0, and sum = 1 is automatic
        #  if we just change a[i]... wait no, we need sum to stay 1)
        
        # Actually, for coordinate descent on the simplex, we should fix the sum.
        # One approach: vary a[i] and renormalize the rest. But that changes other coords.
        # Alternative: just vary a[i] freely (a[i] >= 0) and accept if objective improves,
        # then renormalize at the end. Since the objective is scale-invariant, 
        # we can work with sum=1 and just project back.
        
        # Simpler approach: just try a few values of a[i] and pick the best.
        # The optimal a[i] is where the max is minimized.
        
        # For efficiency, let's just try a geometric grid around the current value.
        # Or better: the objective as a function of a_new_i is:
        #   max( max_k' old_conv[k'],  max_k'' (old_conv[k''] + alpha_k'' * a_new_i + beta_k'' * a_new_i^2) )
        # where the second max is over affected k values.
        # This is the max of linear and quadratic functions, so it's convex in a_new_i.
        # We can minimize it with a few bisection steps.
        
        # For simplicity and speed, let's try: bisection on a_new_i in [0, 2*ai_old+0.1]
        # Actually, let's just do a golden section search or simple grid.
        
        # Even simpler: try the current value scaled by some factors
        # and also try 0.
        candidates = [0.0, ai_old * 0.1, ai_old * 0.3, ai_old * 0.5, ai_old * 0.7,
                      ai_old, ai_old * 1.3, ai_old * 1.5, ai_old * 2.0, ai_old * 3.0]
        # Also try values based on what would equalize the peak
        best_val = ai_old
        best_local_obj = np.inf
        
        for new_val in candidates:
            a[i] = new_val
            # Renormalize
            s_curr = a.sum()
            if s_curr < 1e-15:
                a[i] = ai_old
                break
            a_norm = a / s_curr
            conv_cand = fast_convolve(a_norm)
            obj_cand = 2.0 * n * conv_cand.max()
            if obj_cand < best_local_obj:
                best_local_obj = obj_cand
                best_val = new_val
        
        a[i] = best_val
        s_curr = a.sum()
        a = a / s_curr
        
        conv = fast_convolve(a)
        obj = 2.0 * n * conv.max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > n * 3:
                break
    
    return best_a, 2.0 * n * fast_convolve(best_a).max()

def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Low autocorrelation inspired: use a pseudo-random binary sequence mapped to [0,1]
    # m-sequence like
    state = rng.integers(1, 2**8)
    bits = []
    for _ in range(n):
        # Simple LFSR
        msb = (state >> 7) ^ (state >> 3)
        state = ((state << 1) | msb) & 0xFF
        bits.append(state & 1)
    mseq = np.array(bits, dtype=np.float64) * 0.8 + 0.1
    mseq /= mseq.sum()
    inits['mseq'] = mseq
    
    # Cosine-modulated
    inits['cos_mod'] = (1.0 + np.cos(2 * np.pi * x / 4)) / 2 + 0.1
    inits['cos_mod'] /= inits['cos_mod'].sum()
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5*n) / (0.15*n))**2)
    g /= g.sum()
    inits['gauss'] = g
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Parent interpolation
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        elif len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
    
    return inits

"""
Multi-phase optimization for min 2*n*max(conv(a,a))/(sum(a))^2.
Uses: diverse initializations (including flat-spectrum), LP-based direction finding
with adaptive active set, Nesterov-accelerated smoothed gradient descent, and
coordinate descent polish with analytical structure exploitation.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def build_lp_matrix_rows(a, indices):
    """Build constraint matrix rows for given convolution positions."""
    n = len(a)
    A = np.zeros((len(indices), n))
    for row, k in enumerate(indices):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        col_idx = np.arange(i_min, i_max + 1)
        A[row, col_idx] = a[k - col_idx]
    return A


def solve_lp_direction(a, M, active_indices=None):
    """Solve LP: max sum(b) s.t. b>=0, (a*b)[k] <= M for active k."""
    n = len(a)
    num_k = 2 * n - 1
    c = -np.ones(n)
    bounds = [(0, None)] * n

    if active_indices is not None and len(active_indices) < num_k:
        A_ub = build_lp_matrix_rows(a, active_indices)
        b_ub = np.full(len(active_indices), M)
    else:
        A_ub = build_lp_matrix_rows(a, range(num_k))
        b_ub = np.full(num_k, M)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_optimize(a0, time_limit, step_init=0.8, active_frac=0.2, relax=1.0, seed=0):
    """LP-based optimization with adaptive active set."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    num_k = 2 * n - 1

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    step = step_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()

        n_active = max(5, int(active_frac * num_k))
        if n_active < num_k:
            active_idx = np.argpartition(conv, -n_active)[-n_active:]
        else:
            active_idx = None

        b = solve_lp_direction(a, M * relax, active_indices=active_idx)
        if b is None:
            b = solve_lp_direction(a, M * relax)
            if b is None:
                break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum

        a_new = (1.0 - step) * a + step * b
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            step = min(1.0, step * 1.03)

            step = max(0.05, step * 0.92)

    return best_a, 2.0 * n * conv_final.max()


def nesterov_smoothed_gradient(a0, time_limit, beta_init=500.0, lr_init=0.03,

    best_a = a.copy()

    lr = lr_init
    no_improve = 0
    t_count = 1.0

            w /= w_sum

        y_pad = np.zeros(fft_size)
        y_pad[:n] = y

        W = np.fft.fft(w_pad)
        Y = np.fft.fft(y_pad)
        cross = np.fft.ifft(Y * np.conj(W)).real[:n]
        conv_wy = np.fft.ifft(Y * W).real[:n]

            no_improve += 1
            if no_improve % 30 == 0:

                noise = rng.standard_normal(n) * 0.01
                a = project_simplex(a + noise)
                y = a.copy()
                v = np.zeros(n)
                lr = lr_init * 0.5
                t_count = 1.0
                no_improve = 0

        beta = min(10000.0, beta * 1.003)

    return best_a, 2.0 * n * conv_final.max()

def flatten_spectrum_init(n, rng):
    """Generate initialization with approximately flat power spectrum."""
    N = 2 * n  # oversample
    phases = rng.uniform(0, 2 * np.pi, N)
    # Enforce conjugate symmetry for real result
    half = N // 2
    sym_phases = np.zeros(N)
    sym_phases[0] = 0.0
    if N % 2 == 0:
        sym_phases[N // 2] = 0.0
    sym_phases[1:half] = phases[1:half]
    sym_phases[half + 1:] = -phases[1:half][::-1]
    
    spectrum = np.exp(1j * sym_phases)
    signal = np.real(np.fft.ifft(spectrum))
    signal = np.maximum(signal, 0.0)
    sig_sum = signal.sum()
    if sig_sum < 1e-10:
        return np.ones(n) / n
    # Resample to length n
    if len(signal) != n:
        x_old = np.linspace(0, 1, len(signal))
        x_new = np.linspace(0, 1, n)
        signal = np.interp(x_new, x_old, signal)
        signal = np.maximum(signal, 0.0)
        sig_sum = signal.sum()
        if sig_sum < 1e-10:
            return np.ones(n) / n
    return signal / sig_sum


def generate_initializations(n, parent, rng):

    inits = {}

    inits['flat_spec'] = flatten_spectrum_init(n, rng)

    # Low-contrast pseudo-random
    pr = rng.uniform(0.3, 1.0, n)
    pr /= pr.sum()
    inits['low_rand'] = pr

    # Cosine window (raised cosine)
    if n > 1:
        cos_w = 0.5 * (1.0 + np.cos(2 * np.pi * x / (n - 1)))
        cos_w = np.maximum(cos_w, 0.01)
        cos_w /= cos_w.sum()
        inits['cosine'] = cos_w

    # Bimodal (concentrated at edges)
    edge = max(2, n // 5)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Gaussian bump
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)

    inits['gauss'] = g

    # Parent-based
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        elif len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)

            if ps > 1e-10:

            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)

            if ps > 1e-10:

        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in init_names if nm not in priority]
        init_names = init_names[:5]

        time_per = max(6.0, remaining / max(1, len(init_names)))

        for name in init_names:
            if time.monotonic() > t1 + phase1_time - 4:
                break
            t_alloc = min(time_per, t1 + phase1_time - time.monotonic() - 2)
            if t_alloc < 4:
                break

            a0 = inits[name]
            a_opt, obj_opt = lp_optimize(a0, time_limit=t_alloc,
                                         step_init=0.9, active_frac=0.25,
                                         relax=1.0, seed=n * 13 + hash(name) % 2000)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, {name}: score={score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}")

    # ========== Phase 2: Deep LP Optimization ==========
    print("=== Phase 2: Deep LP ===")
    t2 = time.monotonic()
    phase2_time = min(550.0, deadline - time.monotonic() - 500.0)
    if phase2_time < 150:
        phase2_time = 200.0

    # Collect diverse starting points
    starts = []
    seen = set()
    for score, n, name, a_opt in scan_results[:12]:
        key = (n, name)
        if key not in seen:
            starts.append((n, a_opt.copy()))
            seen.add(key)
        if len(starts) >= 7:
            break

    # Ensure we have the current best
    if best_sequence is not None:
        starts.append((len(best_sequence), best_sequence.copy()))

    # LP configurations (diversify step size and active set fraction)
    lp_cfgs = [
        {'step_init': 1.0, 'active_frac': 0.12, 'relax': 1.0},
        {'step_init': 0.8, 'active_frac': 0.2, 'relax': 1.0},
        {'step_init': 0.6, 'active_frac': 0.35, 'relax': 1.0},
        {'step_init': 1.0, 'active_frac': 0.08, 'relax': 1.01},
        {'step_init': 0.5, 'active_frac': 0.5, 'relax': 0.99},
        {'step_init': 0.7, 'active_frac': 0.15, 'relax': 1.02},
    ]

    total = len(starts) * len(lp_cfgs)
    time_per = phase2_time / max(1, total)
    time_per = max(12.0, time_per)

    run_count = 0
    for i, (n_s, a_s) in enumerate(starts):
        for j, cfg in enumerate(lp_cfgs):
            if time.monotonic() > t2 + phase2_time - 8:
                break
            t_alloc = min(time_per, t2 + phase2_time - time.monotonic() - 4)
            if t_alloc < 8:
                break

            a_opt, obj_opt = lp_optimize(a_s, time_limit=t_alloc,
                                         seed=run_count * 17 + 5, **cfg)
            score = float(evaluate_sequence(a_opt.tolist()))
            run_count += 1
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  LP[{run_count}] n={n_s} cfg{j}: {score:.6f}")

    print(f"Phase 2 best: {best_score:.6f}")

    # ========== Phase 3: Nesterov Smoothed Gradient ==========
    print("=== Phase 3: Nesterov Gradient ===")
    t3 = time.monotonic()
    phase3_time = min(350.0, deadline - time.monotonic() - 300.0)
    if phase3_time < 100:
        phase3_time = 120.0

    if best_sequence is not None:
        n_best = len(best_sequence)
        grad_cfgs = [
            {'beta_init': 400, 'lr_init': 0.025, 'momentum': 0.9, 'use_nesterov': True},
            {'beta_init': 800, 'lr_init': 0.015, 'momentum': 0.92, 'use_nesterov': True},
            {'beta_init': 1500, 'lr_init': 0.008, 'momentum': 0.94, 'use_nesterov': True},
            {'beta_init': 300, 'lr_init': 0.04, 'momentum': 0.85, 'use_nesterov': False},
            {'beta_init': 1000, 'lr_init': 0.012, 'momentum': 0.93, 'use_nesterov': True},
            {'beta_init': 600, 'lr_init': 0.02, 'momentum': 0.91, 'use_nesterov': True},
        ]

        time_per = phase3_time / max(1, len(grad_cfgs) + 3)

        for i, cfg in enumerate(grad_cfgs):
            if time.monotonic() > t3 + phase3_time - 8:
                break
            t_alloc = min(time_per, t3 + phase3_time - time.monotonic() - 4)
            if t_alloc < 8:
                break

            a_opt, obj_opt = nesterov_smoothed_gradient(
                best_sequence, time_limit=t_alloc, seed=100 + i * 7, **cfg)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Grad[{i}]: {score:.6f}")

        # Perturbed gradient restarts
        for i in range(3):
            if time.monotonic() > t3 + phase3_time - 8:
                break
            noise = rng.standard_normal(n_best) * 0.006
            a_pert = project_simplex(best_sequence + noise)
            t_alloc = min(time_per * 0.6, t3 + phase3_time - time.monotonic() - 4)
            if t_alloc < 6:
                break

            a_opt, obj_opt = nesterov_smoothed_gradient(
                a_pert, time_limit=t_alloc, beta_init=800, lr_init=0.015,
                momentum=0.92, use_nesterov=True, seed=500 + i)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  GradPert[{i}]: {score:.6f}")

    print(f"Phase 3 best: {best_score:.6f}")

    # ========== Phase 4: Coordinate Descent Polish ==========
    print("=== Phase 4: Coordinate Descent ===")
    t4 = time.monotonic()
    phase4_time = deadline - time.monotonic() - 30.0
    if phase4_time < 30:
        phase4_time = 30.0

    if best_sequence is not None:
        # Split: 70% coordinate descent, 20% more gradient, 10% random
        cd_time = phase4_time * 0.65
        grad_time = phase4_time * 0.20
        rand_time = phase4_time * 0.15

        # Coordinate descent
        if cd_time > 20:
            a_opt, obj_opt = coordinate_descent_polish(
                best_sequence, time_limit=cd_time, seed=42)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  CoordDesc: {score:.6f}")

        # Another gradient pass
        if grad_time > 15 and best_sequence is not None:
            a_opt, obj_opt = nesterov_smoothed_gradient(
                best_sequence, time_limit=grad_time,
                beta_init=2000, lr_init=0.005, momentum=0.95,
                use_nesterov=True, seed=999)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  FinalGrad: {score:.6f}")

        # Random perturbations (tiny)
        if rand_time > 5 and best_sequence is not None:
            n_best = len(best_sequence)
            t_rand_start = time.monotonic()
            attempt = 0
            while time.monotonic() < t_rand_start + rand_time:
                scales = [0.0005, 0.001, 0.002, 0.004, 0.008]
                for scale in scales:
                    noise = rng.standard_normal(n_best) * scale
                    a_pert = project_simplex(best_sequence + noise)
                    score_p = float(evaluate_sequence(a_pert.tolist()))
                    if np.isfinite(score_p) and score_p < best_score:
                        best_score = score_p
                        best_sequence = a_pert.copy()
                        print(f"  Perturb s={scale} [{attempt}]: {score:.6f}")
                        break
                attempt += 1
                if time.monotonic() > deadline - 10:
                    break

    print(f"Phase 4 best: {best_score:.6f}")

    # ========== Final Verification ==========
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    # Safety: verify the sequence is valid
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Clip to ensure non-negative and within bounds
    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})")
    return [float(x) for x in best_sequence]
