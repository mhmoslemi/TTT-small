    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))

def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

    """Compute the log-sum-exp smoothed objective and its gradient."""
    c = fft_convolve_self(a)  # length 2n-1
    c_max = c.max()

    shifted = (c - c_max) / eps
    exp_shifted = np.exp(shifted)

    grad = compute_gradient(a, n, p)
    
    # Project gradient onto simplex tangent space (subtract mean)
    S = a.sum()
    grad = grad - np.mean(grad)
    
    return logsum, grad

**Coordinate descent with the sweep approach:**

For each coordinate j, find the optimal value of a[j] given all other entries, subject to the sum constraint.

If we fix sum(a) = S and change a[j] from c to x, we need to adjust another coordinate. The simplest is to distribute the change (c-x) evenly among all other coordinates. But that's expensive.

Alternative: use a "Lagrangian" approach. The problem is:
minimize max_k conv[k] subject to a ≥ 0, sum(a) = S.

Lagrangian: L(a, λ, μ) = max_k conv[k] + λ*(sum(a) - S) - sum_j μ_j * a_j

For a fixed λ, the subproblem is: minimize max_k conv[k] + λ*sum(a) subject to a ≥ 0.

This is still non-convex, but in the coordinate descent, for a fixed j:
conv_new[k] as a function of x = a[j] is:
- Linear: conv[k] + 2*(x-c)*a[k-j] (for most k)

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences using smoothed
log-sum-exp gradient with FFT-based computation, Adam optimizer,
adaptive smoothing, and multi-phase optimization strategy.

Key improvements over previous approach:
- Full softmax-weighted gradient (not just top-K) via FFT cross-correlation
- Adaptive log-sum-exp smoothing parameter (geometric annealing)
- Backtracking line search for robust step size
- Multi-phase: coarse (large eps) -> fine (small eps) -> polish
- Uses height_sequence_1 as primary initialization
- Periodic stochastic perturbation to escape local minima

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    c[np.abs(c) < 1e-14] = 0.0
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def smooth_gradient_fft(a, n, p):
    """Compute grad of smoothed max-conv using FFT cross-correlation.
    p: probability weights over 2n-1 lags (sum to 1).
    Returns gradient of length n.
    """
    q_flipped = p[::-1]
    size = (2 * n - 1) + n - 1
    fsize = 1 << (size - 1).bit_length()
    Q = fft(q_flipped, fsize)
    A = fft(a, fsize)
    conv_full = np.real(ifft(Q * A))
    # r[j] = conv_full[2n-2-j] for j=0..n-1
    r = conv_full[2 * n - 2 : n - 2 : -1]
    return 2.0 * r


def compute_smoothed(a, n, eps):
    """Compute log-sum-exp smoothed objective and its gradient."""
    c = fft_convolve_self(a)
    c_max = c.max()
    shifted = (c - c_max) / eps
    shifted = np.clip(shifted, -80.0, 80.0)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    logsum = c_max + eps * np.log(exp_shifted.sum())
    grad = smooth_gradient_fft(a, n, p)
    grad = grad - np.mean(grad)
    return logsum, grad, c

    c = fft_convolve_self(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)

                # Distribute -diff to other entries proportionally
                others = np.ones(n)
                others[j] = 0.0
                total_others = a.sum() - a[j]
                if total_others > 1e-15:
                    a = a - diff * others * (a / total_others)
                    a = np.maximum(a, 0.0)
                    a = a * (S / a.sum())
                else:
                    a = project_simplex(a, S)


def get_initializations(rng):
    """Generate a list of initializations to try."""
    inits = []

    # Primary: use the given best sequence
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given = np.clip(given, 0.0, 1000.0)
        if given.size >= 64 and given.sum() > 0.01:
            inits.append(("given", given.copy()))
    except Exception:
        pass

    n_base = 8192

    # Smooth periodic initializations
    for freq in [2, 3, 5, 8, 13]:
        tt = np.linspace(0, 1, n_base)
        arr = 1.0 + 0.3 * np.sin(2 * np.pi * freq * tt) + 0.1 * np.cos(2 * np.pi * (freq * 3) * tt)
        arr = np.clip(arr, 0.01, None)
        inits.append((f"smooth_{freq}_{n_base}", arr))

    # Random smooth (Gaussian filtered)
    for _ in range(3):
        raw = rng.random(n_base)
        # Simple moving average smoothing
        window = 64
        kernel = np.ones(window) / window
        arr = np.convolve(raw, kernel, mode='same')
        arr = np.clip(arr, 0.01, None)
        inits.append((f"rand_smooth_{n_base}", arr.copy()))

    # Different lengths
    for n_try in [4096, 16384]:
        tt = np.linspace(0, 1, n_try)
        arr = 1.0 + 0.2 * np.sin(2 * np.pi * 3 * tt)
        arr = np.clip(arr, 0.01, None)
        inits.append((f"len_{n_try}", arr))

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(10.0, budget_s - 15.0)
    print(f"Starting optimization with budget {budget_s}s, deadline in {budget_s-15:.0f}s")

    global_best = np.inf
    global_a = None

    inits = get_initializations(rng)
    n_inits = len(inits)
    print(f"Number of initializations: {n_inits}")

    # Time allocation: 60% explore, 30% focused on best, 10% polish
    elapsed = time.monotonic() - t0
    remaining = deadline - t0 - elapsed
    explore_budget = 0.55 * remaining
    focus_budget = 0.25 * remaining
    polish_budget = remaining - explore_budget - focus_budget

    # Phase 1: Explore multiple initializations
    per_init = explore_budget / max(1, n_inits)
    for i, (name, arr) in enumerate(inits):
        if time.monotonic() > deadline - focus_budget - polish_budget - 2.0:
            print(f"  Stopping exploration at init {i}/{n_inits}")
            break

        n = len(arr)
        if n < 64:
            continue

        t_init_start = time.monotonic()
        left = max(2.0, per_init - (time.monotonic() - t_init_start))

        # Adam with smoothed gradient
        seg = left * 0.6
        a_best, obj_best = run_adam_smoothed(arr, n, seg, seed=seed + i)
        print(f"  [{name}] adam: obj={obj_best:.6f} ({seg:.0f}s)")
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()

        # Multiplicative update on the result
        if time.monotonic() < deadline - focus_budget - polish_budget - 2.0 and seg > 1.0:
            seg2 = min(left * 0.3, 30.0)
            a_m, obj_m = run_multiplicative(a_best, n, seg2, seed=seed + 100 + i)
            print(f"  [{name}] mirror: obj={obj_m:.6f}")
            if obj_m < global_best:
                global_best = obj_m
                global_a = a_m.copy()
            elif obj_best < global_best:
                global_best = obj_best
                global_a = a_best.copy()

    print(f"After exploration: best={global_best:.6f}")

    # Phase 2: Focused optimization on the best found
    if global_a is not None and time.monotonic() < deadline - polish_budget - 2.0:
        n = len(global_a)
        left_focus = min(focus_budget, deadline - time.monotonic() - polish_budget - 2.0)
        if left_focus > 5.0:
            print(f"Phase 2: Focused optimization for {left_focus:.0f}s on n={n}")
            # Run Adam with smaller initial epsilon for fine tuning
            a_f, obj_f = run_adam_smoothed(
                global_a, n, left_focus, seed=seed + 1000,
                eps_init=0.1 * np.sqrt(2.0 * n) / n,
                lr_init=0.2 * np.sqrt(2.0 * n) / n
            )
            print(f"  Focused adam: obj={obj_f:.6f}")
            if obj_f < global_best:
                global_best = obj_f
                global_a = a_f.copy()

            # Also try multiplicative
            if time.monotonic() < deadline - polish_budget - 2.0:
                left_m = min(30.0, deadline - time.monotonic() - polish_budget - 2.0)
                if left_m > 3.0:
                    a_m2, obj_m2 = run_multiplicative(global_a, n, left_m, seed=seed + 2000)
                    print(f"  Focused mirror: obj={obj_m2:.6f}")
                    if obj_m2 < global_best:
                        global_best = obj_m2
                        global_a = a_m2.copy()

    # Phase 3: Coordinate sweep polish
    if global_a is not None and time.monotonic() < deadline - 5.0:
        n = len(global_a)
        left_polish = max(0.0, deadline - time.monotonic() - 5.0)
        print(f"Phase 3: Coordinate polish for {left_polish:.0f}s on n={n}")
        a_p, obj_p = coordinate_sweep_polish(global_a, n, left_polish, seed=seed + 3000)
        print(f"  Polish: obj={obj_p:.6f}")
        if obj_p < global_best:
            global_best = obj_p
            global_a = a_p.copy()

    # Final verification
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    ver = evaluate_sequence(global_a.tolist())
    print(f"Final verification: {ver:.6f}")
    if not np.isfinite(ver):
        # Fallback
        global_a = np.ones(256, dtype=np.float64)
        ver = evaluate_sequence(global_a.tolist())
        print(f"Fallback verification: {ver:.6f}")

    elapsed = time.monotonic() - t0
    print(f"Done in {elapsed:.0f}s. Final objective: {ver:.6f}")

    return [float(x) for x in global_a]
