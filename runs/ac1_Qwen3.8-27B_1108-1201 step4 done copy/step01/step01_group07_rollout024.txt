"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP outer iteration (alternating
minimization of cross-convolution) combined with subgradient refinement.
The LP step finds the optimal direction g minimizing max(f*g) on the simplex,
then a line search determines the best step size. Multiple sequence lengths
and diverse initializations are explored in a time-budgeted search.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """FFT-based linear auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_max_conv(a):
    """Max of auto-convolution."""
    return fast_convolve(a).max()


def compute_objective(a):
    """Compute 2*n*max_conv/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * compute_max_conv(a) / (s * s)


def build_lp_matrices(f):
    """Build LP constraint matrices for: min M s.t. (f*g)[k] <= M, sum(g)=1, g>=0."""
    n = len(f)
    m = 2 * n - 1
    A_ub = np.zeros((m, n + 1))
    for j in range(n):
        end = min(j + n, m)
        A_ub[j:end, j] = f[j]
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    c = np.zeros(n + 1)
    c[n] = 1.0
    bounds = [(0, None)] * (n + 1)
    return c, A_ub, b_ub, A_eq, b_eq, bounds


def lp_step(f):
    """
    Solve LP: minimize M s.t. (f*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, m_opt) or (None, msg).
    """
    n = len(f)
    c, A_ub, b_ub, A_eq, b_eq, bounds = build_lp_matrices(f)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0:
        return None, res.message
    g = np.maximum(res.x[:n], 0.0)
    g_sum = g.sum()
    if g_sum < 1e-15:
        return None, "g is zero"
    g /= g_sum
    return g, res.x[n]


def line_search(f, g, M_current, max_evals=8):
    """
    Find best step size t in [0,1] for f_new = (1-t)*f + t*g
    that minimizes max(f_new * f_new).
    """
    best_t = 0.0
    best_m = M_current
    ts = [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1]
    for t in ts[:max_evals]:
        f_new = (1.0 - t) * f + t * g
        m_new = compute_max_conv(f_new)
        if m_new < best_m:
            best_m = m_new
            best_t = t
    return best_t, best_m


def subgradient_refine(a, time_limit, eta0=0.1, decay=0.999):
    """
    Subgradient method on the simplex to reduce max convolution.
    a should be normalized to sum=1.
    """
    n = len(a)
    a = np.maximum(a.copy(), 0.0)
    a /= a.sum()
    conv = fast_convolve(a)
    best_m = conv.max()
    best_a = a.copy()
    eta = eta0
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        m = conv.max()
        k_star = int(np.argmax(conv))

        # Compute subgradient: dc[k_star]/da[i]
        sg = np.zeros(n)
        for i in range(n):
            d1 = k_star - i
            if 0 <= d1 < n:
                sg[i] += a[d1]
            d2 = i - k_star
            if 0 <= d2 < n and d2 != 0:
                sg[i] += a[d2]
            elif d2 == 0:
                sg[i] += a[0]  # double count for the a[i]^2 term

        # Projected gradient step on simplex
        a_new = a - eta * sg
        # Project onto {x >= 0, sum(x) = 1}
        a_new -= np.minimum(a_new, 0.0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new /= a_new_sum

        conv_new = fast_convolve(a_new)
        m_new = conv_new.max()

        if m_new < best_m - 1e-12:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new
        eta *= decay
        if stall > 200:
            eta = max(eta0 * 0.1, eta * 0.5)
            stall = 0
            if eta < eta0 * 0.01:
                break

    return best_a, best_m


def lp_outer_iteration(f0, time_limit, max_iters=5000):
    """
    LP outer iteration: alternately solve LP for g, update f via line search.
    f0 should be normalized to sum=1.
    """
    n = len(f0)
    f = np.maximum(f0.copy(), 0.0)
    f /= f.sum()
    conv = fast_convolve(f)
    best_m = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    M = best_m

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        g, msg = lp_step(f)
        if g is None:
            break

        # Line search
        t_best, m_after = line_search(f, g, M, max_evals=6)

        if t_best > 0 and m_after < best_m - 1e-12:
            f = (1.0 - t_best) * f + t_best * g
            f = np.maximum(f, 0.0)
            f /= f.sum()
            best_m = m_after
            best_f = f.copy()
            no_improve = 0
            M = m_after
        else:
            # Try full step with the g from LP
            f_try = g.copy()
            m_try = compute_max_conv(f_try)
            if m_try < best_m - 1e-12:
                f = f_try.copy()
                best_m = m_try
                best_f = f.copy()
                M = m_try
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 50:
                    break
                # Perturb to escape
                if no_improve % 10 == 0:
                    eps = 0.01
                    f = f + eps * np.random.default_rng(it).uniform(-1, 1, n)
                    f = np.maximum(f, 0)
                    f /= f.sum()

    return best_f, best_m


def generate_initializations(n, seed, parent=None):
    """Generate diverse initializations for sequence of length n."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = {}

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Various cosine windows
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        w = np.cos(np.pi * (x / n - 0.5)) ** (2 * p)
        w = np.maximum(w, 0)
        if w.sum() > 1e-10:
            inits[f'cos_{p}'] = w / w.sum()

    # Gaussian bumps
    for sf in [0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
        g = np.exp(-0.5 * ((x - n / 2) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g

    # U-shaped (edges)
    t = 2 * x / (n - 1) - 1  # -1 to 1
    for p in [0.5, 1.0, 2.0, 4.0]:
        u = np.abs(t) ** p
        if u.sum() > 1e-10:
            inits[f'u_{p}'] = u / u.sum()

    # Bimodal
    bi = np.zeros(n)
    bi[:n // 4] = 1.0
    bi[-n // 4:] = 1.0
    if bi.sum() > 0:
        inits['bimodal'] = bi / bi.sum()

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['random2'] = rng.dirichlet(np.ones(n) * 2.0)

    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = parent[:n].copy()
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                inits['parent'] = pp / pp.sum()
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                inits['parent_interp'] = pp / pp.sum()

    return inits


def run_local_refinement(a, time_limit):
    """Combine subgradient and small random perturbations for final polish."""
    n = len(a)
    a = np.maximum(a.copy(), 0.0)
    a /= a.sum()
    conv = fast_convolve(a)
    best_m = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    # Subgradient phase
    sub_time = time_limit * 0.6
    if sub_time > 1:
        a_sg, m_sg = subgradient_refine(a, time_limit=sub_time, eta0=0.05, decay=0.998)
        if m_sg < best_m:
            best_m = m_sg
            best_a = a_sg.copy()
            a = a_sg.copy()

    # Random perturbation phase
    remaining = time_limit - (time.monotonic() - t0)
    if remaining > 1:
        t_noise = 0.05
        for _ in range(200):
            if time.monotonic() - t0 > time_limit:
                break
            delta = rng.uniform(-1, 1, n) * t_noise
            a_try = a + delta
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try /= s
            m_try = compute_max_conv(a_try)
            if m_try < best_m:
                best_m = m_try
                best_a = a_try.copy()
                a = a_try.copy()
                t_noise = max(0.001, t_noise * 0.95)
            else:
                t_noise = max(0.001, t_noise * 0.98)

    return best_a, best_m
