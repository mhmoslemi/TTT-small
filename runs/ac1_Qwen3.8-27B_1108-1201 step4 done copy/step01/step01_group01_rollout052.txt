def _compute_score(a):
    """Compute 2*n*max(convolve(a,a)) / sum(a)^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-15:
        return np.inf
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b) / (s * s)

def _adam_step(x, tau, m, v, t, lr):
    """Single Adam step in log-domain. Returns (x_new, obj, m, v)."""
    a = _softmax(x)
    n = len(a)

    b_max = np.max(b)

    exp_b = np.exp(tau * b_shifted)

    wb = np.dot(w, b)
    grad = 2.0 * a * (wa - wb)

    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)

    score_final = _compute_score(a_final)
    if score_final < best_score:
        best_score = score_final
        best_x = x.copy()

    return best_x, best_score


def _init_two_level(n):
    a = np.ones(n)
    a[:n // 2] = 1.4
    a[n // 2:] = 0.6
    return a / np.sum(a)


def _init_cosine(n, freq=1, amp=0.4):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * i / n)
    a = np.maximum(a, 0.05)
    return a / np.sum(a)


def _init_tapered(n):
    i = np.arange(n, dtype=np.float64)
    t = 2.0 * i / (n - 1) - 1.0
    a = 0.5 * (1.0 + np.cos(np.pi * t))
    a = np.maximum(a, 0.01)
    return a / np.sum(a)


def _init_four_level(n):
    a = np.ones(n)
    q = n // 4
    a[:q] = 1.3
    a[q:2*q] = 1.1
    a[2*q:3*q] = 0.9
    a[3*q:] = 0.7
    return a / np.sum(a)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Log-domain optimization (L-BFGS + Adam) with tau annealing and multi-restart
    to minimize 2*n*max(autocorr)/(sum)^2 for a non-negative sequence.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 20.0)

    # Get parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = np.ones(6121)
    parent = np.clip(parent, 0.0, 1000.0)
    if parent.size == 0 or np.sum(parent) < 1e-10:
        parent = np.ones(6121)
    n = parent.size
    a_parent = parent / np.sum(parent)

    global_best_a = a_parent.copy()
    global_best_score = _compute_score(a_parent)
    print(f"n={n}, Parent score: {global_best_score:.6f}")

    # === Phase 1: L-BFGS tau annealing from parent (40% of time) ===
    phase1_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.40
    x_current = np.log(a_parent + 1e-15)

    tau_schedule_p1 = [2.0, 10.0, 50.0, 200.0, 800.0]
    for tau in tau_schedule_p1:
        if time.monotonic() >= phase1_deadline:
            break
        x_current, fun = _lbfgs_run(x_current, tau, phase1_deadline,
                                     maxiter=150, gtol=1e-4)
        a_check = _softmax(x_current)
        score_check = _compute_score(a_check)
        if score_check < global_best_score:
            global_best_score = score_check
            global_best_a = a_check.copy()
        elapsed = time.monotonic() - (phase1_deadline - (deadline - time.monotonic()) * 0.40)
        print(f"  P1 tau={tau:6.0f}: obj={fun:.6f} score={score_check:.6f} best={global_best_score:.6f}")

    # === Phase 2: Adam from parent with annealing (25% of time) ===
    remaining = deadline - time.monotonic()
    phase2_deadline = time.monotonic() + remaining * 0.50
    if time.monotonic() < phase2_deadline:
        x_start = np.log(global_best_a + 1e-15)
        x_adam, score_adam = _adam_optimize(
            x_start, phase2_deadline,
            tau_start=1.0, tau_end=1000.0,
            lr_start=0.03, lr_end=0.0005,
            total_steps=4000
        )
        a_adam = _softmax(x_adam)
        if score_adam < global_best_score:
            global_best_score = score_adam
            global_best_a = a_adam.copy()
        print(f"  P2 Adam: score={score_adam:.6f} best={global_best_score:.6f}")

    # === Phase 3: Multi-restart L-BFGS (25% of time) ===
    remaining = deadline - time.monotonic()
    phase3_deadline = time.monotonic() + remaining * 0.55
    if time.monotonic() < phase3_deadline:
        inits = []
        # Perturbed best
        inits.append(np.log(global_best_a + 1e-15) + rng.normal(0, 0.4, n))
        # Two-level
        inits.append(np.log(_init_two_level(n) + 1e-15))
        # Cosine
        inits.append(np.log(_init_cosine(n, 1, 0.5) + 1e-15))
        # Tapered
        inits.append(np.log(_init_tapered(n) + 1e-15))
        # Four-level
        inits.append(np.log(_init_four_level(n) + 1e-15))
        # Random with smoothing
        a_rand = rng.random(n)
        # Smooth via cumulative average
        a_rand = np.convolve(a_rand, np.ones(20)/20.0, mode='same')
        a_rand = np.maximum(a_rand, 0.01)
        inits.append(np.log(a_rand / np.sum(a_rand) + 1e-15))

        n_inits = len(inits)
        per_init = (phase3_deadline - time.monotonic()) / max(1, n_inits)

        for i, x_init in enumerate(inits):
            if time.monotonic() >= phase3_deadline:
                break
            init_dl = min(time.monotonic() + per_init, phase3_deadline)

            # Quick tau anneal: 3 levels
            for tau in [5.0, 80.0, 400.0]:
                if time.monotonic() >= init_dl:
                    break
                x_init, _ = _lbfgs_run(x_init, tau, init_dl, maxiter=120, gtol=1e-4)

            a_i = _softmax(x_init)
            score_i = _compute_score(a_i)
            if score_i < global_best_score:
                global_best_score = score_i
                global_best_a = a_i.copy()
                print(f"  P3 Init {i+1}/{n_inits}: NEW BEST score={score_i:.6f}")
            else:
                print(f"  P3 Init {i+1}/{n_inits}: score={score_i:.6f}")

    # === Phase 4: Final polish with very sharp tau ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"  P4: Polishing with {remaining:.1f}s")
        x_best = np.log(global_best_a + 1e-15)
        for tau in [2000.0, 10000.0]:
            if time.monotonic() >= deadline:
                break
            x_best, _ = _lbfgs_run(x_best, tau, deadline, maxiter=100, gtol=1e-6)
            a_best = _softmax(x_best)
            score_best = _compute_score(a_best)
            if score_best < global_best_score:
                global_best_score = score_best
                global_best_a = a_best.copy()
            print(f"    Polish tau={tau:.0f}: score={score_best:.6f}")

    # Final output
    result = [float(max(0.0, x)) for x in global_best_a]
    final_score = float(evaluate_sequence(result))
    print(f"FINAL SCORE: {final_score:.6f} (true best: {global_best_score:.6f})")
    return result
