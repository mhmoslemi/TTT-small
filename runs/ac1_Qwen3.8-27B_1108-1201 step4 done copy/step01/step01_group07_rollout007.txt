import numpy as np
from scipy.optimize import linprog, minimize
import time

def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def cross_convolve(f, g):
    """Linear cross-convolution f*g using FFT."""
    n = len(f)
    m = len(g)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    G = np.fft.rfft(g, fft_size)
    result = np.fft.irfft(F * G, fft_size)
    return result[:size]

def lp_step(f, M):
    """
    Given f (normalized, sum=1) and M (convolution bound),
    find g >= 0 maximizing sum(g) subject to (f*g)[k] <= M for all k.
    Returns g (not normalized).
    """
    n = len(f)
    num_constraints = 2 * n - 1
    
    # c = -1 for all variables (maximize sum = minimize -sum)
    c = -np.ones(n)
    
    # Constraint matrix A_ub: row k has f[j] in column (k-j) if 0 <= k-j < n
    # A_ub @ g <= b_ub where b_ub = M for each row
    A_ub = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, idx] = f[j]
    
    b_ub = np.full(num_constraints, M)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    else:
        # Fallback: uniform
        return np.ones(n)

def lp_optimize(f0, time_limit, max_iter=500):
    """
    LP-alternation method:
    1. Compute M = max(f*f)
    2. Solve LP to get g
    3. Rescale g to sum=1
    4. Update f = (1-t)*f + t*g
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    
    best_f = f.copy()
    best_obj = 2.0 * n * fast_convolve(f).max()
    
    t0 = time.monotonic()
    t = 0.5  # mixing parameter
    patience = 0
    max_patience = 50
    
    for iteration in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(f)
        M = conv.max()
        
        if M < 1e-15:
            break
        
        g_raw = lp_step(f, M)
        g_sum = g_raw.sum()
        if g_sum < 1e-10:
            break
        g = g_raw / g_sum
        
        # Try different mixing parameters
        best_local_obj = best_obj
        best_local_f = best_f
        
        for t_try in [t, t * 0.5, t * 1.5, 0.1, 0.3, 0.7]:
            if t_try <= 0 or t_try >= 1:
                continue
            f_new = (1 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0)
            f_sum = f_new.sum()
            if f_sum < 1e-10:
                continue
            f_new /= f_sum
            obj = 2.0 * n * fast_convolve(f_new).max()
            if obj < best_local_obj:
                best_local_obj = obj
                best_local_f = f_new.copy()
                t = t_try
        
        f = best_local_f
        
        if best_local_obj < best_obj - 1e-10:
            best_obj = best_local_obj
            best_f = f.copy()
            patience = 0
        else:
            patience += 1
            if patience >= max_patience:
                # Reduce step size
                t *= 0.5
                patience = 0
                max_patience = max(20, max_patience // 2)
                if t < 0.01:
                    break
    
    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-alternation (block coordinate
descent with linear subproblem), log-space Adam polishing, and multi-scale
search. Combines the SOTA LP approach with improved step-size adaptation
and gradient-based refinement.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def cross_corr(w, a):
    """Compute sg[j] = sum_k w[k] * a[k-j] for valid j."""
    n = len(a)
    mw = len(w)
    sg = np.zeros(n)
    # Direct computation (efficient for n <= 512)
    for k in range(mw):
        wk = w[k]
        if wk < 1e-12:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            sg[j_start:j_end + 1] += wk * a[k - j_end:k - j_start + 1][::-1]
    return sg


def build_lp_matrix(f):
    """Build the constraint matrix for the LP subproblem."""
    n = len(f)
    num_constraints = 2 * n - 1
    # A_ub[k, j] = f[j] if 0 <= k-j < n, else 0
    # i.e., A_ub[k, j] = f[j] if k-n+1 <= j <= k (and 0 <= j < n)
    rows = []
    cols = []
    vals = []
    for k in range(num_constraints):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[j])
    A = csr_matrix((vals, (rows, cols)), shape=(num_constraints, n))
    return A


def lp_step(f, M, A_prebuilt=None):
    """Solve the LP subproblem: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    n = len(f)
    num_constraints = 2 * n - 1
    
    if A_prebuilt is None:
        A_prebuilt = build_lp_matrix(f)
    
    c = -np.ones(n)
    b_ub = np.full(num_constraints, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_prebuilt, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 5})
    
    if res.success:
        return np.maximum(res.x, 0)

def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)

def cross_corr_direct(w, a):

def build_lp_matrix(f):

            no_improve = 0
            if T > 10000 or lr < 0.001:
                break
    return best_a, best_obj


def make_initializations(n, rng, parent=None):
    x = np.arange(n, dtype=np.float64)
    inits = {}
    inits['uniform'] = np.ones(n) / n
    for sf in [0.1, 0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        inits['gauss_' + str(sf)] = g / g.sum()
    sine = np.maximum(0.0, np.sin(np.pi * x / n))
    inits['sine'] = sine / sine.sum()
    cw = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
    inits['cosine'] = cw / cw.sum()
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    r = rng.exponential(1.0, n)
    inits['exp_random'] = r / r.sum()
    if parent is not None:
        if len(parent) >= n:
            pp = parent[:n].copy()
            pp = np.maximum(pp, 0.0)
            if pp.sum() > 1e-10:
                inits['parent'] = pp / pp.sum()
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0.0)
            if pp.sum() > 1e-10:
                inits['parent_interp'] = pp / pp.sum()
    return inits


def run_phase1(rng, deadline, parent):
    print("Phase 1: Quick scan with LP-alternation...")
    lengths = [64, 128, 256, 512]
    results = []
    best_score = float('inf')
    best_seq = None
    phase1_budget = min(500.0, (deadline - time.monotonic()) * 0.35)

    for n in lengths:
        if time.monotonic() > deadline - 150:
            break
        inits = make_initializations(n, rng, parent)
        priority = ['uniform', 'gauss_0.2', 'random', 'sine', 'parent', 'parent_interp', 'cosine', 'gauss_0.3']
        selected = [p for p in priority if p in inits]
        for name in inits:
            if name not in selected:
                selected.append(name)
        selected = selected[:5]
        time_per_init = phase1_budget / (len(lengths) * max(1, len(selected)))

        for name in selected:
            if time.monotonic() > deadline - 150:
                break
            t_alloc = min(time_per_init, deadline - time.monotonic() - 150)
            if t_alloc < 5:
                break
            a0 = inits[name]
            a_opt, obj = lp_optimize(a0, time_limit=t_alloc, max_iter=500, initial_t=0.5)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                results.append((score, n, name, a_opt))
                if score < best_score:
                    best_score = score
                    best_seq = a_opt.copy()
                    print("  P1: n=%d, %s, score=%.6f" % (n, name, score))
    return results, best_score, best_seq


def run_phase2(scan_results, deadline):
    best_score = float('inf')
    best_seq = None
    for score, n, name, a_opt in scan_results:
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
    print("Phase 2: Deep optimization. Best: %.6f" % best_score)

    top_configs = []
    seen_n = set()
    sorted_results = sorted(scan_results, key=lambda x: x[0])
    for score, n, name, a_opt in sorted_results:
        if n not in seen_n and len(top_configs) < 6:
            top_configs.append((n, a_opt.copy()))
            seen_n.add(n)
    if best_seq is not None:
        top_configs.append((len(best_seq), best_seq.copy()))

    remaining = deadline - time.monotonic() - 100
    time_per = max(15.0, remaining / max(1, len(top_configs)))

    for n, a0 in top_configs:
        if time.monotonic() > deadline - 100:
            break
        t_alloc = min(time_per, deadline - time.monotonic() - 100)
        if t_alloc < 10:
            break
        for t_init in [0.3, 0.6, 0.9]:
            if time.monotonic() > deadline - 100:
                break
            t_single = t_alloc / 3.0
            a_opt, obj = lp_optimize(a0, time_limit=t_single, max_iter=3000, initial_t=t_init)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print("  P2: n=%d, t=%.1f, score=%.6f" % (n, t_init, score))
    return best_score, best_seq


def run_phase3(best_seq, best_score, deadline):
    print("Phase 3: Adam polish. Best: %.6f" % best_score)
    remaining = deadline - time.monotonic() - 40
    for T_val in [50.0, 100.0, 200.0, 400.0]:
        if remaining < 10:
            break
        t_single = remaining / 4.0
        a_opt, obj = adam_polish(best_seq, time_limit=t_single, T=T_val, lr=0.03)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print("  P3: T=%.0f, score=%.6f" % (T_val, score))
        remaining = deadline - time.monotonic() - 40
    return best_score, best_seq


def run_phase4(best_seq, best_score, deadline, rng):
    print("Phase 4: Perturbation + LP. Best: %.6f" % best_score)
    base = best_seq.copy()
    n = len(base)
    remaining = deadline - time.monotonic() - 5
    max_attempts = int(remaining / 0.03)

    for i in range(min(max_attempts, 20000)):
        if time.monotonic() > deadline - 5:
            break
        sigma = 0.03 + 0.07 * (i / max(1, max_attempts))
        delta = rng.normal(0, sigma, n)
        a_pert = base + delta
        a_pert = np.maximum(a_pert, 0.0)
        if a_pert.sum() < 1e-10:
            continue
        a_pert /= a_pert.sum()
        conv = fast_convolve(a_pert)
        M = conv.max()
        A = build_lp_matrix(a_pert)
        g_raw = lp_step(a_pert, M, A)
        g_sum = g_raw.sum()
        if g_sum > 1e-10:
            g = g_raw / g_sum
            for t_mix in [0.4, 0.7]:
                a_new = (1.0 - t_mix) * a_pert + t_mix * g
                a_new = np.maximum(a_new, 0.0)
                if a_new.sum() < 1e-10:
                    continue
                a_new /= a_new.sum()
                score = float(evaluate_sequence(a_new.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    base = a_new.copy()
                    best_seq = a_new.copy()
                    print("  P4: iter=%d, score=%.6f" % (i, score))
    return best_score, best_seq


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 20.0)

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print("Parent loaded: n=%d" % len(parent))
    except Exception:
        pass

    results, best_score, best_seq = run_phase1(rng, deadline, parent)

    if best_seq is None:
        best_seq = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_seq.tolist()))

    best_score, best_seq = run_phase2(results, deadline)
    best_score, best_seq = run_phase3(best_seq, best_score, deadline)
    best_score, best_seq = run_phase4(best_seq, best_score, deadline, rng)

    print("FINAL SCORE: %.6f" % best_score)
    return [float(x) for x in best_seq]
