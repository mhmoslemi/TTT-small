        tasks.append((4, rng.random(n512) * 0.5 + 0.25, True, f"rand_{trial}"))
    
    # Priority 5: n=1024 variants
    i1024 = np.arange(n1024, dtype=np.float64)
    tasks.append((5, 0.5 + 0.5 * np.cos(2 * np.pi * i1024 * 5.0 / n1024), False, "chirp5_1024"))
    tasks.append((5, rng.random(n1024) * 0.3 + 0.35, False, "rand_1024"))
    
    # Priority 6: n=256 (sometimes shorter is better normalized)
    i256 = np.arange(256, dtype=np.float64)
    ideal_256 = ideal_flat_seq(256)
    tasks.append((6, ideal_256.copy(), True, "ideal256"))
    tasks.append((6, 0.5 + 0.5 * np.cos(2 * np.pi * i256 * 2.0 / 256), True, "chirp2_256"))
    
    # Sort by priority
    tasks.sort(key=lambda x: x[0])
    
    # Run tasks
    for priority, a0, use_lp, label in tasks:
        if time.monotonic() > deadline - 60:
            break
        try_seq(a0, use_lp, label)
    
    # Final refinement of best
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"\nFinal refinement of best (score={best_score:.6f}, n={len(best_seq)})...")
        for ref_pass in range(10):
            if time.monotonic() > deadline - 15:
                break
            seq, score = optimize_one(best_seq, deadline - 15, use_lp=(len(best_seq) <= 1024),
                                      max_grad_iters=2000)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  Refine pass {ref_pass}: score={best_score:.6f}")
            else:
                break
        
        # Extra coordinate descent
        if time.monotonic() < deadline - 10:
            a, s = coordinate_descent(best_seq, n_passes=500, deadline=deadline - 10)
            if s < best_score:
                best_score = s
                best_seq = a.copy()
                print(f"  Coord descent: score={best_score:.6f}")
    
    # Final output
    if best_seq is not None:
        final_score = compute_obj(best_seq)
        print(f"\nFinal: n={len(best_seq)}, score={final_score:.6f}")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence: {eval_score:.6f}")
        except Exception as e:
            print(f"eval error: {e}")
        return [float(x) for x in best_seq]
    
    print("Falling back to uniform.")
    return [1.0] * 512
