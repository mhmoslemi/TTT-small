Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using improved multiplicative mirror
descent with adaptive top-k peak focusing, diverse initializations including

def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def top_k_subgradient(a, conv, top_k=5, temperature=1.0):
    """
    Compute a weighted subgradient of max(conv) focusing on top-k peaks.
    Returns subgradient vector and the effective max value.
    """
    n = len(a)
    # Find top-k peaks
    k = min(top_k, len(conv))
    top_indices = np.argsort(conv)[-k:]
    top_values = conv[top_indices]
    
    # Soft-max weights over top-k
    if temperature < 1e-10:
        # Hard: equal weight on all top-k (or just the max)
        w = np.ones(k) / k
    else:
        scaled = temperature * (top_values - top_values.min())
        w = np.exp(scaled)
        w /= w.sum()
    
    # Compute subgradient: for each peak at index pk,
    # d(conv[pk])/d(a[j]) = 2*a[pk-j] if 0 <= pk-j < n
    sg = np.zeros(n)
    for i, pk in enumerate(top_indices):
        if w[i] < 1e-12:
            continue
        # Gradient of conv[pk] wrt a[j] = 2*a[pk-j] for j where 0<=pk-j<n
        # j ranges from max(0, pk-(n-1)) to min(n-1, pk)
        j_start = max(0, pk - n + 1)
        j_end = min(n - 1, pk)
        if j_start <= j_end:
            # a[pk - j] for j in [j_start, j_end]
            # = a[pk - j_end : pk - j_start + 1][::-1]
            idx = pk - np.arange(j_end + 1, j_start - 1, -1)

    return sg


def mirror_descent_step(a, sg, eta):
    """Multiplicative update on the simplex."""
    # Center the subgradient
    sg_centered = sg - sg.mean()
    # Multiplicative update
    log_factor = -eta * sg_centered
    log_factor -= log_factor.max()
    a_new = a * np.exp(log_factor)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-12:
        return np.ones_like(a) / len(a)
    return a_new / s

def generate_initializations(n, rng, parent=None):

    # Power-law
    for p in [2.0, 3.0, 5.0]:

        inits[f'power_{p}_rev'] = inits[f'power_{p}'][::-1].copy()

        p = np.maximum(p, 0)

def local_search(a, n_iterations=200, step_size=0.05, rng=None):
    """Simple local search: random perturbations and keep improvements."""
    if rng is None:
        rng = np.random.default_rng()
    
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-12:
        return a
    a = a / s
    
    conv = fast_convolve(a)
    best_obj = 2.0 * len(a) * conv.max()
    best_a = a.copy()
    
    for _ in range(n_iterations):
        # Random perturbation
        a_pert = best_a.copy()
        idx = rng.integers(0, len(a_pert))
        delta = rng.uniform(-step_size, step_size) * best_a.max()
        a_pert[idx] = max(0, a_pert[idx] + delta)

        a_pert /= s
        
        conv_p = fast_convolve(a_pert)
        obj_p = 2.0 * len(a_pert) * conv_p.max()
        
        if obj_p < best_obj - 1e-9:
            best_obj = obj_p
            best_a = a_pert.copy()
    
    return best_a

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
- Diverse asymmetric initializations (ramps, skewed Gaussians, power laws)
- Multiplicative mirror descent with adaptive top-k peak focusing
- L-BFGS-B refinement with smooth max approximation
- Local search polish
- Multi-length search (64 to 1024)
"""

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def obj_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def compute_sg_topk(a, conv, top_k, temperature):
    """Compute subgradient focusing on top-k peaks."""
    n = len(a)
    k = min(top_k, len(conv))
    # Get top-k indices
    if k == len(conv):
        top_idx = np.arange(len(conv))
        top_vals = conv.copy()
    else:
        # Partial sort for efficiency
        if len(conv) > 2 * k:
            part = np.argpartition(conv, -k)[-k:]
        else:
            part = np.argsort(conv)[-k:]
        top_idx = part
        top_vals = conv[part]
    
    # Weights
    if temperature < 1e-8:
        w = np.ones(k) / k
    else:
        t_vals = top_vals - top_vals.min()
        w = np.exp(temperature * t_vals)
        s = w.sum()
        if s < 1e-30:
            w = np.ones(k) / k
        else:
            w /= s
    
    # Build subgradient
    sg = np.zeros(n)
    for i in range(k):
        if w[i] < 1e-12:
            continue
        pk = int(top_idx[i])
        wi = w[i]
        j_start = max(0, pk - n + 1)
        j_end = min(n - 1, pk)
        if j_start <= j_end:
            idx = pk - np.arange(j_start, j_end + 1)
            sg[j_start:j_end + 1] += wi * 2.0 * a[idx]
    
    return sg


def mirror_descent_optimize(a0, time_limit, eta0=0.4, top_k=3, temp0=2.0,
                            patience_max=40, eta_min=0.005):
    """
    Multiplicative mirror descent on simplex with top-k peak focusing.
    a0: initial non-negative vector (will be normalized to sum=1)
    Returns (best_a, best_normalized_obj)
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    eta = eta0
    temp = temp0
    tk = top_k
    patience = 0
    t0 = time.monotonic()
    stall_count = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute subgradient
        sg = compute_sg_topk(a, conv, tk, temp)
        
        # Center and apply multiplicative update
        sg_c = sg - sg.mean()
        log_f = -eta * sg_c
        log_f -= log_f.max()
        a = a * np.exp(log_f)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = best_a.copy()
            break
        a /= s
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-9:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
            stall_count = 0
        else:
            patience += 1
            if patience > 20:
                stall_count += 1
        
        # Adaptive parameter updates
        if patience >= patience_max:
            patience = 0
            eta *= 0.7
            temp *= 1.8
            tk = max(1, tk - 1)
            
            if eta < eta_min:
                # Significant restart
                eta = max(0.15, eta0 * 0.5)
                temp = 0.5
                tk = min(5, top_k)
                patience_max = max(20, patience_max - 5)
                # Small random perturbation to escape local min
                a = a + 0.01 * np.random.default_rng(int(patience_max * 1000)).random(n)
                a = np.maximum(a, 0)
                a /= a.sum()
                conv = fast_convolve(a)
    
    return best_a, best_obj


def smooth_obj_and_grad(x, n, inv_temp=200.0):
    """Smooth objective and gradient for L-BFGS-B."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-12:
        return 1e10, np.zeros(n)
    
    conv = fast_convolve(a)
    # Soft-max smooth max
    scaled = inv_temp * (conv - conv.max())
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        smooth_max = conv.max()
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
        smooth_max = w @ conv
    
    obj = 2.0 * n * smooth_max / (s * s)
    
    # Gradient of smooth_max wrt a
    sg = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for ki in active:
        k = int(ki)
        wi = w[k]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = k - np.arange(j_start, j_end + 1)
            sg[j_start:j_end + 1] += wi * 2.0 * a[idx]
    
    # d(obj)/d(a_j) = 2n/(s^2) * sg_j - 2n*smooth_max/s^3 * 1
    grad = (2.0 * n / (s * s)) * sg - (2.0 * n * smooth_max / (s ** 3))
    
    return obj, grad


def lbfgs_refine(a0, time_limit, inv_temp=200.0):
    """Refine using L-BFGS-B with smooth objective."""
    n = len(a0)
    x0 = a0.astype(np.float64).copy()
    s0 = x0.sum()
    if s0 < 1e-12:
        return a0
    x0 /= s0
    
    t0 = time.monotonic()
    
    def wrapper(x):
        o, g = smooth_obj_and_grad(x, n, inv_temp)
        return o, g
    
    try:
        res = scipy_minimize(
            wrapper, x0, method='L-BFGS-B',
            options={'maxiter': 500, 'maxfun': 2000,
                     'ftol': 1e-12, 'gtol': 1e-8,
                     'maxls': 50, 'iprint': -1}
        )
        if res.success or res.fun < obj_normalized(x0):
            a_new = np.maximum(res.x, 0.0)
            a_new /= a_new.sum()
            return a_new
    except Exception:
        pass
    
    return a0


def local_search(a, n_iters=2000, step_scale=0.02, rng=None):
    """Coordinate-descent style local search."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    a = np.maximum(a.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return a
    a /= s
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    step = step_scale * best_a.max()
    
    for i in range(n_iters):
        j = rng.integers(0, n)
        for sign in [1, -1]:
            a_try = best_a.copy()
            a_try[j] = max(0.0, a_try[j] + sign * step * (0.5 + rng.random()))
            st = a_try.sum()
            if st < 1e-12:
                continue
            a_try /= st
            conv_try = fast_convolve(a_try)
            obj_try = 2.0 * n * conv_try.max()
            if obj_try < best_obj - 1e-10:
                best_obj = obj_try
                best_a = a_try
                step = max(step * 0.9, 1e-6)
                break
        else:
            step = min(step * 1.1, 0.1 * best_a.max())
    
    return best_a


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Ramps
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = ramp / ramp.sum()
    inits['ramp_rev'] = inits['ramp'][::-1].copy()
    
    # Power laws
    for p in [1.5, 2.0, 3.0, 5.0]:
        pl = np.arange(1, n + 1, dtype=np.float64) ** p
        inits[f'pw{p:.0f}'] = pl / pl.sum()
        inits[f'pw{p:.0f}r'] = inits[f'pw{p:.0f}'][::-1].copy()
    
    # Exponential
    for rate in [0.5, 1.0, 2.0]:
        e = np.exp(-rate * x / n)
        inits[f'exp{rate:.1f}'] = e / e.sum()
        inits[f'exp{rate:.1f}r'] = inits[f'exp{rate:.1f}'][::-1].copy()
    
    # Asymmetric Gaussians
    for center in [0.25, 0.35, 0.45, 0.55, 0.65, 0.75]:
        g = np.exp(-0.5 * ((x - center * n) / (0.25 * n)) ** 2)
        inits[f'g{center:.2f}'] = g / g.sum()
    
    # Cosine (asymmetric)
    cw = 0.5 * (1 + np.cos(np.pi * (2 * x / n - 0.5)))
    cw = np.maximum(cw, 0)
    if cw.sum() > 0:
        inits['cos_asym'] = cw / cw.sum()
    
    # Bimodal
    bi = np.zeros(n)
    bi[:n//4] = 1.0
    bi[-n//6:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # Random
    inits['rand'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Parent
    if parent is not None:
        p = np.maximum(parent, 0)
        if p.sum() > 1e-10:
            if len(p) >= n:
                pc = p[:n].copy()
                if pc.sum() > 1e-10:
                    inits['parent'] = pc / pc.sum()
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                if pp.sum() > 1e-10:
                    inits['parent_interp'] = pp / pp.sum()
    
    return inits
