"""
Multi-phase optimization for low-max-autocorrelation non-negative sequences.
Improves on previous by using aggressive LP best-response iteration combined
with gradient refinement and parametric initialization.
"""

import time
import numpy as np
from scipy.fft import fft, ifft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _conv_full(a):
    """Full linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0)
    L = next_fast_len(2 * n - 1)
    F = fft(a, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1]


def _score(a):
    """Internal score matching evaluate_sequence."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    c = _conv_full(a)
    mx = float(np.max(c))
    return 2.0 * n * mx / (s * s)


def _normalize(a):
    """Project onto {a>=0, sum=1}."""
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-12:
        return np.full(a.size, 1.0 / a.size)
    return a / s


def _lp_best_response(f, M):
    """
    Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns g or None on failure.
    """
    n = int(f.size)
    if n <= 0 or n > 3000:
        return None
    # Build banded constraint matrix
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        js = np.arange(jmin, jmax + 1)
        rows.extend((np.full(len(js), k)).tolist())
        cols.extend(js.tolist())
        data.extend(f[k - js].tolist())
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None),
                      method='highs', options={'disp': False, 'presolve': True})
        if res.status == 0:
            g = np.asarray(res.x, dtype=np.float64)
            g = np.maximum(g, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def _gradient_smooth_max(a, T_factor=0.05):
    """
    Gradient of smoothed max of convolution.
    Returns (gradient, temperature, smoothed_value).
    """
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    L = next_fast_len(3 * n - 2)
    c = _conv_full(a)
    cmax = float(np.max(c))
    if cmax <= 0:
        return np.zeros(n), 0.0, 0.0
    T = max(T_factor * cmax, 1e-30)
    diff = (c - cmax) / T
    e = np.exp(diff)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # Gradient: grad_j = 2 * sum_k w_k * a[k-j] (circular, sliced)
    Fw = fft(w[::-1], L)
    Fa = fft(a, L)
    h = np.real(ifft(Fa * Fw))
    if n == 1:
        G = h[:1]
    else:
        G = h[2 * n - 2: n - 2: -1]
    G = np.asarray(G, dtype=np.float64)
    if G.size != n:
        tmp = np.zeros(n)
        m = min(n, G.size)
        tmp[:m] = G[:m]
        G = tmp
    g = 2.0 * np.maximum(G, 0.0)
    surr = cmax + T * np.log(s) if s > 0 else cmax
    return g, T, surr


def _project_simplex(v):
    """Euclidean projection onto probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.0, posinf=1.0, neginf=0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / (k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if s < 1e-300:
        return np.full(n, 1.0 / n)
    return out / s


def _make_linear_taper(n, alpha, reverse=False):
    """Linear taper: a[j] = 1 - alpha*j/(n-1)."""
    x = np.linspace(0, 1, n)
    a = 1.0 - float(alpha) * x
    if reverse:
        a = a[::-1]
    return _normalize(np.maximum(a, 1e-10))


def _make_power_taper(n, p, reverse=False):
    """Power taper: a[j] = (1 - j/(n-1))^p."""
    x = np.linspace(0, 1, n)
    a = np.power(np.maximum(1.0 - x, 1e-10), float(p))
    if reverse:
        a = a[::-1]
    return _normalize(a)


def _make_geometric(n, r, reverse=False):
    """Geometric: a[j] = r^j."""
    a = np.power(float(r), np.arange(n, dtype=np.float64))
    if reverse:
        a = a[::-1]
    return _normalize(a)


def _make_quadratic(n, a, b, c):
    """Quadratic: a[j] = a + b*x + c*x^2 where x in [0,1]."""
    x = np.linspace(0, 1, n)
    a_val = float(a) + float(b) * x + float(c) * x * x
    return _normalize(np.maximum(a_val, 1e-10))


def _make_cosine_taper(n, depth):
    """Cosine taper: a[j] = 1 - depth*cos(pi*j/(n-1))."""
    x = np.linspace(0, 1, n)
    a = 1.0 - float(depth) * np.cos(np.pi * x)
    return _normalize(np.maximum(a, 1e-10))

        freq = int(rng.integers(1, max(2, n // 50)))
        phase = float(rng.uniform(0, 2 * np.pi))
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * freq * x + phase)
        inits.append(_normalize(np.maximum(a, 0.05)))
    return inits


def _lp_iteration(f, max_time_per_lp=30.0):
    """
    One LP best-response iteration.
    Returns (new_f, improved, new_score).
    """
    f = _normalize(f)
    c = _conv_full(f)
    M = float(np.max(c))
    current_score = float(_score(f))
    if M <= 0 or not np.isfinite(M):
        return f, False, current_score

    t_start = time.monotonic()
    g = _lp_best_response(f, M)
    lp_time = time.monotonic() - t_start

    if g is None:
        return f, False, current_score

    g = _normalize(g)
    g_score = _score(g)

    # Try different mixing ratios
    best_f = f.copy()
    best_sc = current_score
    for t in [1.0, 0.8, 0.6, 0.4, 0.25, 0.1, 0.05]:
        cand = _normalize((1.0 - t) * f + t * g)
        sc = _score(cand)
        if sc < best_sc:
            best_sc = sc
            best_f = cand.copy()

    improved = best_sc < current_score - 1e-12
    return best_f, improved, best_sc


def _gradient_refine(f, rng, n_iters=200, time_limit=30.0):
    """
    Gradient-based refinement on the simplex.
    """
    f = _normalize(f)
    n = int(f.size)
    best_f = f.copy()
    best_sc = _score(f)
    current_sc = best_sc
    T_factor = 0.05
    end = time.monotonic() + time_limit
    prev = f.copy()

    for it in range(n_iters):
        if time.monotonic() > end:
            break
        g, T, surr = _gradient_smooth_max(f, T_factor)
        if not np.all(np.isfinite(g)):
            g = np.zeros(n)
        gabs = float(np.max(np.abs(g)))
        if gabs < 1e-30:
            break
        # Line search
        p_scale = max(1.0 / n, 0.1 * np.max(f))
        base_lr = p_scale / gabs
        accepted = False
        for lr in [0.5 * base_lr, 0.1 * base_lr, 0.02 * base_lr, 0.004 * base_lr]:
            v = f - lr * g
            v = v + 0.3 * (f - prev) if it > 20 else v
            cand = _project_simplex(v)
            sc = _score(cand)
            surr_c = None
            if sc < current_sc:
                if sc < best_sc:
                    best_sc = sc
                    best_f = cand.copy()
                current_sc = sc
                prev = f
                f = cand
                accepted = True
                break
        if not accepted:
            T_factor = max(0.001, T_factor * 0.5)
            # Random perturbation
            sig = 0.02 / n
            cand = _project_simplex(f + rng.normal(0, sig, n))
            sc = _score(cand)
            if sc < best_sc:
                best_sc = sc
                best_f = cand.copy()
                current_sc = sc
                f = cand
            else:
                T_factor = 0.05

    return best_f, best_sc


def _lp_optimize(f0, rng, time_limit, label=""):
    """
    Main LP-based optimization loop.
    Alternates between LP best-response and gradient refinement.
    """
    f = _normalize(f0.copy())
    best_f = f.copy()
    best_sc = _score(f)
    print(f"  {label} start: score={best_sc:.10f}", flush=True)
    end = time.monotonic() + time_limit
    no_improve = 0
    lp_count = 0
    grad_count = 0

    while time.monotonic() < end:
        remaining = end - time.monotonic()
        if remaining < 5.0:
            break

        # LP step (primary driver)
        if remaining > 3.0:
            f_new, improved, sc_new = _lp_iteration(f)
            lp_count += 1
            if improved:
                no_improve = 0
                if sc_new < best_sc:
                    best_sc = sc_new
                    best_f = f_new.copy()
                    if lp_count % 10 == 0:
                        print(f"  {label} LP#{lp_count}: {sc_new:.10f}", flush=True)
                f = f_new
            else:
                no_improve += 1

        # Gradient refinement between LP steps
        if no_improve >= 1 and remaining > 15.0:
            f_grad, sc_grad = _gradient_refine(f, rng, n_iters=50,
                                                time_limit=min(20.0, remaining * 0.3))
            grad_count += 1
            if sc_grad < best_sc:
                best_sc = sc_grad
                best_f = f_grad.copy()
                no_improve = 0
                print(f"  {label} grad#{grad_count}: {sc_grad:.10f}", flush=True)
            f = _normalize(0.5 * f_grad + 0.5 * f)

        # Occasional random restart from best
        if no_improve > 5 and remaining > 30.0:
            sig = 0.05 / f.size
            f = _project_simplex(best_f + rng.normal(0, sig, f.size))
            no_improve = 0

        # Adaptive: if LP stops improving, try different M scaling
        if no_improve > 3 and remaining > 10.0:
            c = _conv_full(f)
            M = float(np.max(c))
            # Try with slightly relaxed constraint
            g = _lp_best_response(f, M * 1.001)
            if g is not None:
                g = _normalize(g)
                for t in [0.3, 0.2, 0.1, 0.05]:
                    cand = _normalize((1 - t) * f + t * g)
                    sc = _score(cand)
                    if sc < best_sc:
                        best_sc = sc
                        best_f = cand.copy()
                        f = cand
                        no_improve = 0
                        break

    return best_f, best_sc


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/sum(a)^2.
    Uses aggressive LP best-response with gradient refinement.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 15.0

    # Get parent if available
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.where(np.isfinite(parent_arr), parent_arr, 0.0)
        parent_arr = np.maximum(parent_arr, 0.0)
        if parent_arr.ndim == 1 and parent_arr.size >= 16 and parent_arr.sum() > 0.01:
            parent = parent_arr
            print(f"Parent loaded: n={parent_arr.size}", flush=True)
    except Exception:
        pass

    # Decide sequence lengths to try
    n_primary = 700
    n_alt = 1000
    if parent is not None:
        n_primary = int(parent.size)
        if n_primary < 100:
            n_primary = 500

    best_p = None
    best_score = np.inf

    def try_record(p, sc):
        nonlocal best_p, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64)
        if p.size == 0:
            return
        p = _normalize(p)
        if not np.isfinite(sc):
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_p = p.copy()
            print(f"NEW BEST: {sc:.10f} (n={p.size}) at t={time.monotonic()-t_start:.1f}s", flush=True)

    # Phase 1: Generate and evaluate initial points
    print("Phase 1: Generating initial points...", flush=True)
    all_inits = []

    # Parent as initial
    if parent is not None:
        p_norm = _normalize(parent)
        sc = _score(p_norm)
        all_inits.append((sc, p_norm, n_primary))
        try_record(p_norm, sc)

    # Parametric initials at primary length
    for a in _generate_initials(n_primary, rng):
        sc = _score(a)
        if np.isfinite(sc):
            all_inits.append((sc, a, n_primary))
            try_record(a, sc)

    # Also try at alternate length
    if n_alt != n_primary:
        for a in _generate_initials(n_alt, rng)[:15]:
            sc = _score(a)
            if np.isfinite(sc):
                all_inits.append((sc, a, n_alt))
                try_record(a, sc)

    all_inits.sort(key=lambda x: x[0])
    print(f"Phase 1 done: {len(all_inits)} inits, best={best_score:.10f}", flush=True)

    # Phase 2: LP optimization on top candidates
    print("Phase 2: LP optimization...", flush=True)
    remaining = deadline - time.monotonic()
    if remaining > 100.0:
        # Pick top distinct initial points
        candidates = []
        seen_scores = []
        for sc, p, n in all_inits:
            if len(candidates) >= 4:
                break
            # Avoid very similar starting points (simple check)
            dominated = False
            for csc, cp, cn in candidates:
                if n == cn and np.max(np.abs(p - cp)) < 0.01:
                    dominated = True
                    break
            if not dominated:
                candidates.append((sc, p, n))

        n_cands = len(candidates)
        time_per = max(60.0, min(400.0, (remaining - 200.0) / n_cands))

        for i, (sc_init, p_init, n) in enumerate(candidates):
            rem = deadline - time.monotonic() - 150.0
            if rem < 30.0:
                break
            t_alloc = min(time_per, rem)
            print(f"  Candidate {i}: n={n}, init_score={sc_init:.8f}, time={t_alloc:.0f}s", flush=True)
            p_best, sc_best = _lp_optimize(p_init, rng, t_alloc, label=f"C{i}")
            try_record(p_best, sc_best)

    # Phase 3: Final intensive polish on the best
    print("Phase 3: Final polish...", flush=True)
    remaining = deadline - time.monotonic()
    if best_p is not None and remaining > 30.0:
        t_polish = min(remaining - 10.0, 120.0)
        if t_polish > 20.0:
            p_final, sc_final = _lp_optimize(best_p, rng, t_polish, label="FINAL")
            try_record(p_final, sc_final)

        # Extra gradient polish
        remaining2 = deadline - time.monotonic()
        if best_p is not None and remaining2 > 15.0:
            p_g, sc_g = _gradient_refine(best_p, rng, n_iters=2000,
                                         time_limit=min(remaining2 - 10.0, 60.0))
            try_record(p_g, sc_g)

    # Ensure valid output
    if best_p is None:
        best_p = np.full(n_primary, 1.0 / n_primary)
        best_score = _score(best_p)

    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    s = float(best_p.sum())
    if s < 0.01:
        best_p = best_p * (0.01 / s) if s > 0 else np.full(best_p.size, 0.01 / best_p.size)
    best_p = np.minimum(best_p, 1000.0)

    elapsed = time.monotonic() - t_start
    final_sc = _score(best_p)
    print(f"DONE: best_score={best_score:.10f}, verified={final_sc:.10f}, n={best_p.size}, time={elapsed:.1f}s", flush=True)

    return [float(x) for x in best_p]
