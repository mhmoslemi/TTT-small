"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy:
  * Normalize to sum = sqrt(2n) so objective = max of full self-convolution.
  * Use soft-max subgradient (smooth approximation of max) with adaptive temperature.
  * Adam optimizer with simplex projection for non-negativity and sum constraint.
  * Periodic spectral shaping moves to escape local minima (flatten DFT magnitude).
  * Multiple initializations: given sequence, smooth functions, random.
  * Multiple lengths: 4096, 8192, 16384.
  * Final polishing phase with very fine steps.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_softgradient(c, a, alpha):
    """Soft-max weighted subgradient of max(c) w.r.t. a.
    
    grad[m] = 2 * sum_{t=0}^{n-1} p[m+t] * a[t]
    where p = softmax(alpha * c)
    Computed via FFT for efficiency.
    """
    n = len(a)
    size_c = 2 * n - 1

    c_shifted = c - c.max()
    exp_c = np.exp(alpha * c_shifted)
    p = exp_c / exp_c.sum()

    # grad[m] = 2 * sum_t a[t] * p[m+t]
    # = 2 * (a * rev_p)[2n-2-m] where rev_p = p[::-1]
    N = 1 << (3 * n - 1).bit_length()
    A = fft(a, N)
    rev_p = p[::-1]
    Rp = fft(rev_p, N)
    conv = np.real(ifft(A * Rp))
    grad = 2.0 * conv[n - 1:2 * n - 1][::-1]
    return grad


def compute_topk_gradient(c, a, top_k):
    """Hard top-k subgradient (fallback for very large alpha)."""
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    grad = np.zeros(n, dtype=np.float64)
    total = c[top_indices].sum()
    if total <= 1e-30:
        return grad
    weights = c[top_indices] / total
    for w_val, k in zip(weights, top_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += w_val * 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad


def spectral_shaping(a, strength=0.3, target_profile='flat'):
    """Modify DFT magnitude to flatten the autocorrelation.
    
    The idea: the autocorrelation peak at the "bulk" is related to the
    variance of |A|^2. We reduce dynamic range of |A| to flatten it.
    """
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)

    # Compute current autocorrelation to identify the problem
    autocorr = np.real(ifft(mag * mag))[:size]

    # Target: reduce the contrast in |A|^2
    # New magnitude: geometric mean between current and flat
    # Flat would be sqrt(mean(|A|^2)) for all frequencies
    flat_mag = np.sqrt(np.mean(mag ** 2))
    
    # Blend: reduce peaks, boost valleys
    log_mag = np.log(mag + 1e-15)
    log_flat = np.log(flat_mag + 1e-15)
    new_log_mag = (1 - strength) * log_mag + strength * log_flat
    new_mag = np.exp(new_log_mag)

    A_new = new_mag * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new * (a.sum() / s)
    else:
        a_new = a.copy()
    return a_new


def run_adam_soft(a_init, n, S, max_time, rng, lr_init, alpha_init, alpha_max):
    """Adam with soft-max gradient and adaptive temperature."""
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    a = a * (S / s) if s > 1e-15 else np.full(n, S / n)

    m_buf = np.zeros(n, dtype=np.float64)
    v_buf = np.zeros(n, dtype=np.float64)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    alpha = alpha_init

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        # Adaptive temperature: increase over time to sharpen
        alpha = min(alpha_max, alpha_init * (1.0 + 0.002 * t))

        # Choose between soft and hard gradient based on alpha
        if alpha > 50.0:
            grad = compute_topk_gradient(c, a, top_k=16)
        else:
            grad = compute_softgradient(c, a, alpha)

        gnorm = np.linalg.norm(grad)
        if gnorm <= 1e-14:
            break
        grad = grad / gnorm

        m_buf = beta1 * m_buf + (1.0 - beta1) * grad
        v_buf = beta2 * v_buf + (1.0 - beta2) * (grad ** 2)
        m_hat = m_buf / (1.0 - beta1 ** t)
        v_hat = v_buf / (1.0 - beta2 ** t)

        lr = lr_init * (0.999 ** t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new, S)

        # Periodic spectral shaping as global move
        if t % 2000 == 0 and rng.random() < 0.3:
            a_spec = spectral_shaping(a, strength=0.1 + 0.2 * rng.random())
            a_spec = project_simplex(a_spec, S)
            c_spec = fft_convolve_self(a_spec)
            if c_spec.max() < best_score:
                a = a_spec
                best_score = c_spec.max()
                best_local = a.copy()
            elif rng.random() < 0.1:
                # Sometimes accept even slightly worse to escape
                a = 0.7 * a + 0.3 * a_spec
                a = project_simplex(a, S)

        # Random perturbation to escape local minima
        if t % 5000 == 0 and rng.random() < 0.15:
            perturbation = rng.normal(0.0, 0.05 * S / n, n)
            a = project_simplex(a + perturbation, S)

    return best_local, best_score

    """Fine Adam for polishing - smaller steps, higher alpha."""
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    a = a * (S / s) if s > 1e-15 else np.full(n, S / n)

    best_score = c.max()
    best_local = a.copy()

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = c.max()

        if alpha > 50.0:
            grad = compute_topk_gradient(c, a, top_k=32)
        else:
            grad = compute_softgradient(c, a, alpha)

        gnorm = np.linalg.norm(grad)

        m_buf = beta1 * m_buf + (1.0 - beta1) * grad
        v_buf = beta2 * v_buf + (1.0 - beta2) * (grad ** 2)
        m_hat = m_buf / (1.0 - beta1 ** t)
