def conv_sq(p):
    """Full self-convolution of a 1D vector, sliced to length 2n-1."""
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = next_fast_len(2 * n - 1)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1].astype(np.float64, copy=False)


def internal_score(p):
    """Score = 2*n*max(conv(p,p))/sum(p)^2"""
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.inf
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 1e-12:
        return np.inf
    c = conv_sq(p)
    cmax = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(cmax) or cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (s * s))


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.0, posinf=1.0, neginf=0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / float(k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def safe_normalize(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if not np.isfinite(s) or p.size == 0 or s <= 1e-12:
        n = max(1, int(p.size))
        return np.full(n, 1.0 / n, dtype=np.float64)
    return p / s


def make_uniform(n):

def make_oscillatory(n, freq, amp, phase=0.0):
    """Non-negative oscillatory sequence: 1 + amp*sin(2*pi*freq*x + phase), clipped."""
    x = np.linspace(0.0, 1.0, n)
    p = 1.0 + amp * np.sin(2.0 * np.pi * freq * x + phase)
    p = np.maximum(p, 0.01)
    return safe_normalize(p)


def make_multi_osc(n, freqs, amps, phases=None):
    """Sum of multiple sinusoids, non-negative."""
    x = np.linspace(0.0, 1.0, n)
    p = np.ones(n, dtype=np.float64)
    for f, a in zip(freqs, amps):
        ph = 0.0
        if phases is not None:
            ph = phases[list(freqs).index(f)]
        p += a * np.sin(2.0 * np.pi * f * x + ph)
    p = np.maximum(p, 0.01)
    return safe_normalize(p)


def make_envelope_osc(n, freq, amp, env_pow):
    """Oscillation with envelope: (1 + amp*sin(...)) * env(x)^pow"""
    x = np.linspace(0.0, 1.0, n)
    env = 1.0 - 0.5 * (2.0 * x - 1.0) ** env_pow
    p = env * (1.0 + amp * np.sin(2.0 * np.pi * freq * x))
    p = np.maximum(p, 0.001)
    return safe_normalize(p)


def make_tapered(n, taper):
    """Uniform with tapered edges."""
    x = np.linspace(0.0, 1.0, n)
    p = 1.0 - taper * (2.0 * x - 1.0) ** 2

    p = 1.0 + slope * x
    p = np.maximum(p, 0.01)

def gradient_active_set(p, active_lags, T):
    """
    Compute gradient using only active lags (where autocorrelation is near max).
    This is more focused than the full softmax approach.
    """
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    L = next_fast_len(3 * n - 2)
    Fp = fft(p, L)
    
    # Compute autocorrelation
    c_full = np.real(ifft(Fp * Fp))[:2 * n - 1]
    
    # Weight the active lags
    w = np.zeros(2 * n - 1, dtype=np.float64)
    for k in active_lags:
        if 0 <= k < 2 * n - 1:
            w[k] = 1.0 / T
    
    # Softmax over active lags
    c_active = c_full[active_lags]
    cmax_a = float(np.max(c_active))
    diff = (c_active - cmax_a) / T
    e = np.exp(diff)
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        e = np.zeros_like(c_active)
        e[int(np.argmax(c_active))] = 1.0
        s = 1.0
    w_active = e / s
    
    # Build full weight vector
    w_full = np.zeros(2 * n - 1, dtype=np.float64)
    for i, k in enumerate(active_lags):
        if 0 <= k < 2 * n - 1:
            w_full[k] = w_active[i]
    
    # Gradient: g_j = 2 * sum_k w_k * p_{k-j} (circular conv of w with p)
    Fw = fft(w_full[::-1], L)
    h = np.real(ifft(Fp * Fw))
    
    if n == 1:
        G = h[:1]
    else:
        G = h[2 * n - 2:n - 2:-1]
    
    G = np.asarray(G, dtype=np.float64)
    if G.size != n:
        tmp = np.zeros(n, dtype=np.float64)
        m = min(n, G.size)
        tmp[:m] = G[:m]
        G = tmp
    
    g = 2.0 * np.maximum(G, 0.0)
    return g


def find_active_lags(c, n_active=10, threshold_frac=0.95):
    """Find lags where autocorrelation is within threshold_frac of max."""
    cmax = float(np.max(c))
    if cmax <= 0:
        return list(range(len(c)))
    thresh = threshold_frac * cmax
    lags = np.where(c >= thresh)[0]
    # Also include the top n_active lags
    top = np.argsort(c)[::-1][:n_active]
    all_lags = set(list(lags) + list(top))
    return sorted(all_lags)

def lp_refine(p, best_score, budget_s):
    """LP best-response for short sequences."""
    if not _HAVE_LP:
        return p, best_score
    
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n <= 0 or n > 500:
        return p, best_score
    
    if budget_s <= 1.0:
        return p, best_score
    
    c = conv_sq(p)
    M = float(np.max(c))
    if not np.isfinite(M) or M <= 0:
        return p, best_score
    
    f = p + 1e-14
    fs = float(f.sum())
    f = f / fs
    
    try:
        mid = np.arange(n, dtype=np.int32)
        rows = (np.arange(n, dtype=np.int32)[:, None] + mid[None, :]).ravel().astype(np.int32)
        cols = np.repeat(np.arange(n, dtype=np.int32), n)
        data = np.tile(f, n).astype(np.float64)
        A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0.0, None), method="highs", options={"disp": False})
    except Exception:
        return p, best_score
    
    if res is None:
        return p, best_score
    
    g = getattr(res, "x", None)
    if g is None:
        return p, best_score
    
    g = np.asarray(g, dtype=np.float64)
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return p, best_score
    g = g / s
    
    for t in (1.0, 0.7, 0.5, 0.3, 0.15):
        q = project_simplex((1.0 - t) * p + t * g)
        sc = internal_score(q)
        if np.isfinite(sc) and sc < best_score:
            return q, sc
    
    return p, best_score

def conv_sq(p):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = next_fast_len(2 * n - 1)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1].astype(np.float64, copy=False)


def internal_score(p):
    p = np.asarray(p, dtype=np.float64)

def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.0, posinf=1.0, neginf=0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / float(k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s

    x = np.linspace(0.0, 1.0, n)
    env = 1.0 + env_amp * np.sin(2.0 * np.pi * env_freq * x)
    p = env * (1.0 + amp * np.sin(2.0 * np.pi * freq * x))
    p = np.maximum(p, 0.01)
    return safe_normalize(p)


def make_step_shape(n, n_steps, heights):
    x = np.linspace(0.0, 1.0, n)
    p = np.interp(x, np.linspace(0.0, 1.0, len(heights)), heights)
    p = np.maximum(p, 0.01)
    return safe_normalize(p)


def make_random_smooth(n, rng):
    x = np.linspace(0.0, 1.0, n)
    p = np.ones(n, dtype=np.float64)
    n_harms = int(rng.integers(2, 6))
    for _ in range(n_harms):
        f = int(rng.integers(1, max(2, n // 30)))
        a = float(rng.uniform(0.1, 0.5))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        p += a * np.sin(2.0 * np.pi * f * x + ph)
    p = np.maximum(p, 0.02)
    return safe_normalize(p)


def smoothed_max_gradient(c, T, active_mask=None):
    """Compute softmax weights over autocorrelation values, optionally restricted to active set."""
    c = np.asarray(c, dtype=np.float64)
    m = c.size
    if m == 0:
        return np.array([])
    if active_mask is not None:
        idx = np.where(active_mask)[0]
        if idx.size == 0:
            w = np.zeros(m, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
            return w
        c_sub = c[idx]
    else:
        idx = np.arange(m)
        c_sub = c
    
    cmax = float(np.max(c_sub))
    if cmax <= 0 or not np.isfinite(cmax):
        w = np.zeros(m, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return w
    
    T = max(float(T), 1e-300)
    diff = (c_sub - cmax) / T
    e = np.exp(diff)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros(m, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return w
    w_sub = e / s
    
    w = np.zeros(m, dtype=np.float64)
    w[idx] = w_sub
    return w


def compute_gradient(p, c, T, active_mask=None):
    """Compute gradient of smoothed max autocorrelation."""
    n = int(p.size)
    if n == 0:
        return np.empty(0)
    L = next_fast_len(3 * n - 2)
    Fp = fft(p, L)
    w = smoothed_max_gradient(c, T, active_mask)
    Fw = fft(w[::-1], L)
    h = np.real(ifft(Fp * Fw))
    if n == 1:
        G = h[:1]
    else:
        G = h[2 * n - 2:n - 2:-1]
    G = np.asarray(G, dtype=np.float64)
    if G.size != n:
        tmp = np.zeros(n, dtype=np.float64)
        m = min(n, G.size)
        tmp[:m] = G[:m]
        G = tmp
    return 2.0 * np.maximum(G, 0.0)


def find_active_mask(c, frac=0.92, top_k=20):
    """Find mask of lags where autocorrelation is near max."""
    cmax = float(np.max(c))
    if cmax <= 0:
        return np.ones(c.size, dtype=bool)
    thresh = frac * cmax
    mask = c >= thresh
    top = np.argsort(c)[::-1][:top_k]
    for t in top:
        mask[t] = True
    return mask


def optimize_local(p0, rng, time_limit):
    """
    Main local optimizer: projected gradient with adaptive smoothed-max,
    plus stochastic pair moves and coordinate descent.
    """
    p = project_simplex(np.asarray(p0, dtype=np.float64))
    n = int(p.size)
    if n == 0:
        return p, np.inf

    best_p = p.copy()
    best_score = internal_score(best_p)
    current_score = best_score
    prev_p = p.copy()

    c = conv_sq(p)
    active_mask = find_active_mask(c)

    end = time.monotonic() + max(1.0, float(time_limit))
    iter_no = 0
    since_improve = 0

    while time.monotonic() < end:
        iter_no += 1

        # Adaptive temperature
        T_frac = 0.15 * (0.997 ** iter_no)
        if since_improve > 150:
            T_frac = max(0.005, T_frac * 0.5)
        T_frac = max(0.003, T_frac)

        cmax_cur = float(np.max(c))
        T = T_frac * cmax_cur if cmax_cur > 0 else 1e-10

        # Recompute active mask periodically
        if iter_no % 80 == 0:
            c = conv_sq(p)
            active_mask = find_active_mask(c)

        # Gradient
        g = compute_gradient(p, c, T, active_mask)
        if not np.all(np.isfinite(g)):
            g = np.zeros(n, dtype=np.float64)

        gabs = float(np.max(np.abs(g))) if g.size else 0.0
        p_mean = 1.0 / n
        p_scale = max(p_mean, 0.02 * float(np.max(p)), 1e-14)

        if gabs > 1e-300:
            base_lr = p_scale / gabs
            lrs = (0.4 * base_lr, 0.15 * base_lr, 0.04 * base_lr, 0.008 * base_lr, 0.001 * base_lr)
        else:
            lrs = (0.02, 0.005, 0.001)

        accepted = False
        momentum = 0.25 if iter_no > 80 else 0.0

        for lr in lrs:
            v = p - lr * g
            if momentum > 0.0:
                v = v + momentum * (p - prev_p)
            pt = project_simplex(v)
            sc = internal_score(pt)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_p = pt.copy()
                current_score = sc
                prev_p = p
                p = pt
                since_improve = 0
                c = conv_sq(p)
                accepted = True
                break
            elif np.isfinite(sc) and sc < current_score:
                current_score = sc
                prev_p = p
                p = pt
                since_improve += 1
                c = conv_sq(p)
                accepted = True
                break

        if not accepted:
            since_improve += 1

            # Stochastic pair moves
            if since_improve > 80:
                n_pairs = 8 if since_improve > 300 else 4
                for _ in range(n_pairs):
                    if time.monotonic() >= end:
                        break
                    i = int(rng.integers(0, n))
                    j = int(rng.integers(0, n))
                    if i == j:
                        continue
                    # Try moving mass from i to j
                    for direction in (1, -1):
                        if direction == 1:
                            src, dst = i, j
                        else:
                            src, dst = j, i
                        if p[src] <= 0:
                            continue
                        for frac in (0.1, 0.03, 0.008):
                            delta = frac * p[src]
                            if delta <= 0:
                                break
                            pt = p.copy()
                            pt[src] -= delta
                            pt[dst] += delta
                            pt = project_simplex(pt)
                            sc = internal_score(pt)
                            if np.isfinite(sc) and sc < current_score:
                                if sc < best_score:
                                    best_score = sc
                                    best_p = pt.copy()
                                    since_improve = 0
                                current_score = sc
                                prev_p = p
                                p = pt
                                c = conv_sq(p)
                                accepted = True
                                break
                        if accepted:
                            break
                    if accepted:
                        break

            # Larger random perturbation when very stuck
            if not accepted and since_improve > 400:
                sigma = 0.3 / n
                pt = project_simplex(p + rng.normal(0.0, sigma, n))
                sc = internal_score(pt)
                if np.isfinite(sc) and sc < current_score:
                    if sc < best_score:
                        best_score = sc
                        best_p = pt.copy()
                        since_improve = 0
                    current_score = sc
                    prev_p = p
                    p = pt
                    c = conv_sq(p)
                    accepted = True

            # Hard restart
            if not accepted and since_improve > 800:
                p = project_simplex(0.6 * best_p + 0.4 * p + rng.normal(0.0, 0.2 / n, n))
                current_score = internal_score(p)
                prev_p = p
                c = conv_sq(p)
                since_improve = 0

            # Softer pullback
            elif not accepted and since_improve > 250:
                p = project_simplex(0.8 * best_p + 0.2 * p + rng.normal(0.0, 0.03 / n, n))
                current_score = internal_score(p)
                prev_p = p
                c = conv_sq(p)

        if iter_no % 3000 == 0:
            print("    iter=%d best=%.10f cur=%.10f" % (iter_no, best_score, current_score), flush=True)

        if best_score < 1.498 and since_improve > 8000:
            break

    if not np.all(np.isfinite(best_p)):
        best_p = project_simplex(p)
    best_p = project_simplex(best_p)
    return best_p, float(best_score) if np.isfinite(best_score) else np.inf


def coordinate_refine(p, rng, time_limit, n_trials=500):
    """Coordinate descent: try reducing individual coordinates and redistributing."""
    p = project_simplex(np.asarray(p, dtype=np.float64))
    n = int(p.size)
    if n == 0:
        return p, np.inf

    best_p = p.copy()
    best_score = internal_score(best_p)

    end = time.monotonic() + max(1.0, float(time_limit))
    trial = 0

    while time.monotonic() < end and trial < n_trials:
        trial += 1
        # Weighted random coordinate selection (prefer coordinates with larger values)
        weights = p + 1e-10
        w_sum = float(weights.sum())
        probs = weights / w_sum
        i = int(rng.choice(n, p=probs))

        if p[i] <= 0:
            continue

        for frac in (0.03, 0.01, 0.003):
            delta = frac * p[i]
            if delta <= 0:
                break

            # Option 1: redistribute to random coordinate
            j = int(rng.integers(0, n))
            if j != i:
                pt = p.copy()
                pt[i] -= delta
                pt[j] += delta
                pt = project_simplex(pt)
                sc = internal_score(pt)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_p = pt.copy()
                    p = pt.copy()
                    break

            # Option 2: redistribute evenly to all other coordinates
            pt = p.copy()
            pt[i] -= delta
            # Redistribute delta to all other positions
            others = np.ones(n, dtype=np.float64)
            others[i] = 0.0
            others = others / float(others.sum())
            pt += delta * others
            pt = project_simplex(pt)
            sc = internal_score(pt)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_p = pt.copy()
                p = pt.copy()
                break

        if trial % 100 == 0:
            print("      CD trial=%d best=%.10f" % (trial, best_score), flush=True)

    return best_p, float(best_score) if np.isfinite(best_score) else np.inf


def lp_best_response(p, budget_s):
    """LP: maximize sum(g) s.t. p*g <= max(p*p), g >= 0."""
    if not _HAVE_LP:
        return p, np.inf
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n <= 0 or n > 400:
        return p, np.inf
    if budget_s <= 1.0:
        return p, np.inf

    c = conv_sq(p)
    M = float(np.max(c))
    if not np.isfinite(M) or M <= 0:
        return p, np.inf

    f = p + 1e-14
    f = f / float(f.sum())

    try:
        mid = np.arange(n, dtype=np.int32)
        rows = (np.arange(n, dtype=np.int32)[:, None] + mid[None, :]).ravel().astype(np.int32)
        cols = np.repeat(np.arange(n, dtype=np.int32), n)
        data = np.tile(f, n).astype(np.float64)
        A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0.0, None),
                      method="highs", options={"disp": False})
    except Exception:
        return p, np.inf

    if res is None:
        return p, np.inf
    g = getattr(res, "x", None)
    if g is None:
        return p, np.inf
    g = np.asarray(g, dtype=np.float64)
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return p, np.inf
    g = g / s
    return g, internal_score(g)


def generate_initials_for_n(n, rng, parent=None):
    """Generate a diverse set of initial sequences of length n."""
    x = np.linspace(0.0, 1.0, n)
    inits = []

    if parent is not None and int(parent.size) == n:
        inits.append(safe_normalize(parent))

    inits.append(make_uniform(n))

    # Tapered / cap shapes
    for t in (0.2, 0.4, 0.6, 0.8, 0.95):
        inits.append(make_tapered(n, t, 2.0))
        inits.append(make_tapered(n, t, 4.0))

    # Ramps (asymmetric)
    for s in (-0.5, -1.0, -2.0, -4.0, 0.5, 1.0, 2.0, 4.0):
        inits.append(make_ramp(n, s))

    # V-shapes
    for d in (0.3, 0.5, 0.7, 0.9):
        inits.append(make_v_shape(n, d, 2.0))
        inits.append(make_v_shape(n, d, 3.0))

    # Oscillatory
    for f in (3, 5, 7, 10, 15, 20):
        for a in (0.3, 0.5, 0.7):
            inits.append(make_oscillatory(n, f, a))

    # Modulated oscillations
    for ff, fa, ef, ea in [(7, 0.4, 2, 0.3), (10, 0.3, 3, 0.4), (5, 0.5, 2, 0.5)]:
        inits.append(make_modulated_osc(n, ff, fa, ef, ea))

    # Step shapes
    inits.append(make_step_shape(n, 4, [1.0, 0.5, 0.5, 1.0]))
    inits.append(make_step_shape(n, 4, [1.0, 0.3, 0.3, 1.0]))
    inits.append(make_step_shape(n, 5, [1.0, 0.7, 0.4, 0.7, 1.0]))
    inits.append(make_step_shape(n, 3, [1.0, 0.5, 1.0]))
    inits.append(make_step_shape(n, 6, [1.0, 0.8, 0.4, 0.4, 0.8, 1.0]))

    # Power / beta
    for a in (0.5, 1.0, 2.0, 4.0):
        inits.append(safe_normalize(x ** a))
        inits.append(safe_normalize((1.0 - x) ** a))
    for a, b in [(0.5, 1.0), (1.0, 2.0), (2.0, 1.0), (0.5, 3.0), (3.0, 0.5)]:
        p = (np.maximum(x, 1e-8) ** a) * (np.maximum(1.0 - x, 1e-8) ** b)
        inits.append(safe_normalize(p))

    # Gaussian
    for mu, sig in [(0.5, 0.1), (0.5, 0.2), (0.3, 0.1), (0.7, 0.1), (0.5, 0.3)]:
        inits.append(safe_normalize(np.exp(-((x - mu) ** 2) / (2.0 * sig ** 2))))

    # Two peaks
    for m1, m2 in [(0.25, 0.75), (0.2, 0.8), (0.35, 0.65)]:
        p = np.exp(-((x - m1) ** 2) / 0.008) + np.exp(-((x - m2) ** 2) / 0.008)
        inits.append(safe_normalize(p))

    # Random smooth
    for _ in range(6):
        inits.append(make_random_smooth(n, rng))

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for non-negative step function minimizing 2*n*max(conv(p,p))/sum(p)^2.
    
    Uses multi-start local optimization with diverse initializations,
    active-set gradient focusing, coordinate descent, and LP refinement.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 20.0)

    # Load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.where(np.isfinite(parent_arr), parent_arr, 0.0)
        parent_arr = np.maximum(parent_arr, 0.0)
        if parent_arr.ndim == 1 and parent_arr.size >= 32 and float(parent_arr.sum()) > 0.0:
            parent = parent_arr
    except Exception:
        parent = None

    best_p = None
    best_score = np.inf

    def try_record(p, sc):
        nonlocal best_p, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64)
        if p.size == 0 or not np.all(np.isfinite(p)):
            return
        p = safe_normalize(p)
        if not np.isfinite(sc):
            sc = internal_score(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_p = p.copy()
            print("  *** NEW BEST: %.10f (n=%d, t=%.0fs) ***" % (best_score, p.size, time.monotonic() - start_time), flush=True)

    # === Generate and score initializations for multiple lengths ===
    print("Generating initializations...", flush=True)

    lengths = [1024, 2048, 4096]
    if parent is not None:
        pn = int(parent.size)
        if pn not in lengths:
            lengths.insert(0, pn)
    if not lengths:
        lengths = [2048]

    all_jobs = []  # (score, p, n, label)

    for n in lengths:
        inits = generate_initials_for_n(n, rng, parent)
        scored = []
        for p in inits:
            if p is None:
                continue
            p = safe_normalize(p)
            sc = internal_score(p)
            if np.isfinite(sc):
                scored.append((sc, p))
                try_record(p, sc)
        scored.sort(key=lambda x: x[0])
        for sc, p in scored[:5]:
            all_jobs.append((sc, p, n, "n=%d" % n))

    all_jobs.sort(key=lambda x: x[0])
    print("Initializations done. Best=%.10f, %d jobs." % (best_score, len(all_jobs)), flush=True)

    # === Local optimization phase ===
    print("Local optimization phase...", flush=True)

    # Time allocation: ~60% for local opt, ~25% for CD polishing, ~10% for final, ~5% reserve
    total_avail = max(60.0, deadline - time.monotonic())
    opt_time = total_avail * 0.60
    cd_time_total = total_avail * 0.25
    final_time = total_avail * 0.10
    reserve = total_avail * 0.05

    # Sort jobs and allocate time
    n_opt_jobs = min(len(all_jobs), 8)
    if n_opt_jobs > 0:
        # Weighted allocation: top jobs get more time
        weights = []
        for i in range(n_opt_jobs):
            w = 1.0 / (i + 1.5)
            weights.append(w)
        total_w = sum(weights)
        weights = [w / total_w for w in weights]

        for i in range(n_opt_jobs):
            sc, p, n, label = all_jobs[i]
            t_alloc = opt_time * weights[i]
            remaining_now = deadline - time.monotonic() - cd_time_total - final_time - reserve
            if remaining_now < 15.0:
                break
            t_alloc = min(t_alloc, max(10.0, remaining_now))

            print("  Job %d/%d: n=%d, init=%.8f, time=%.1fs" % (i + 1, n_opt_jobs, n, sc, t_alloc), flush=True)
            t0 = time.monotonic()
            p_best, sc_best = optimize_local(p, rng, t_alloc)
            try_record(p_best, sc_best)
            print("    Job %d done: %.10f (took %.1fs)" % (i + 1, sc_best, time.monotonic() - t0), flush=True)

    # === Coordinate descent refinement on top results ===
    print("Coordinate descent refinement...", flush=True)
    if best_p is not None and time.monotonic() < deadline - final_time - reserve:
        cd_time = min(cd_time_total, deadline - time.monotonic() - final_time - reserve)
        if cd_time > 5.0:
            p_cd, sc_cd = coordinate_refine(best_p, rng, cd_time, n_trials=400)
            try_record(p_cd, sc_cd)

    # === LP refinement for short sequences ===
    if best_p is not None and int(best_p.size) <= 400:
        lp_time = min(30.0, deadline - time.monotonic() - final_time - reserve)
        if lp_time > 3.0:
            print("  LP refinement...", flush=True)
            p_lp, sc_lp = lp_best_response(best_p, lp_time)
            if sc_lp < best_score:
                try_record(p_lp, sc_lp)
                p_best2, sc_best2 = optimize_local(p_lp, rng, min(30.0, deadline - time.monotonic() - 10.0))
                try_record(p_best2, sc_best2)

    # === Final polish ===
    print("Final polish...", flush=True)
    if best_p is not None and time.monotonic() < deadline - 5.0:
        remaining_final = max(5.0, deadline - time.monotonic() - 5.0)
        p_best, sc_best = optimize_local(best_p, rng, remaining_final * 0.7)
        try_record(p_best, sc_best)

        if time.monotonic() < deadline - 5.0:
            cd_rem = max(3.0, deadline - time.monotonic() - 3.0)
            p_cd, sc_cd = coordinate_refine(best_p, rng, cd_rem, n_trials=200)
            try_record(p_cd, sc_cd)

    # === Last-resort: try a completely different seed if score is still high ===
    if best_score > 1.505 and time.monotonic() < deadline - 20.0:
        print("Last-resort random restart...", flush=True)
        rng2 = np.random.default_rng(seed + 12345)
        n_try = 2048
        for attempt in range(3):
            if time.monotonic() >= deadline - 10.0:
                break
            p_try = make_random_smooth(n_try, rng2)
            # Add oscillatory component
            x = np.linspace(0.0, 1.0, n_try)
            freq = int(rng2.integers(4, 12))
            amp = float(rng2.uniform(0.3, 0.6))
            p_try = safe_normalize(p_try + amp * np.sin(2.0 * np.pi * freq * x))
            sc_try = internal_score(p_try)
            try_record(p_try, sc_try)
            if sc_try < best_score + 0.05:
                t_go = min(60.0, deadline - time.monotonic() - 8.0)
                if t_go > 10:
                    p_best, sc_best = optimize_local(p_try, rng2, t_go)
                    try_record(p_best, sc_best)

    # Final validation and return
    if best_p is None:
        best_p = make_uniform(2048)
        best_score = internal_score(best_p)

    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    s = float(best_p.sum())
    if s <= 1e-12:
        best_p = make_uniform(2048)
    best_p = np.minimum(best_p, 1000.0)

    print("FINAL: score=%.10f, n=%d, time=%.1fs" % (best_score, best_p.size, time.monotonic() - start_time), flush=True)
    return [float(x) for x in best_p]
