"""
LP-based iterative optimization for minimum peak autocorrelation.
Uses the "best response" LP to guide updates toward flatter autocorrelation.
"""

import time
import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
from scipy.fft import fft, ifft, next_fast_len

def _conv_full(p):
    n = len(p)
    L = next_fast_len(2*n - 1)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2*n-1]

def _score(p):
    n = len(p)
    s = float(np.sum(p))
    if s < 0.01:
        return float('inf')
    c = _conv_full(p)
    return float(2.0 * n * np.max(c) / (s * s))

def _project_simplex(v):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.arange(1, n+1, dtype=np.float64)
    cond = (u * idx) > (css - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0/n)
    k = int(np.count_nonzero(cond) - 1)
    theta = (css[k] - 1.0) / (k + 1)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s <= 1e-10:
        return np.full(n, 1.0/n)
    return out / s

def _build_conv_matrix(p):
    """Build sparse convolution matrix A where (A @ g)[k] = (p*g)[k]."""
    n = len(p)
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            data.append(p[k - j])
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return A

def _lp_best_response(p, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (p*g)[k] <= M, g >= 0."""
    n = len(p)
    A = _build_conv_matrix(p)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None),
                      method='highs', options={'time_limit': time_limit, 'disp': False})
    except Exception:
        return None, None
    if res.status != 0:
        return None, None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.maximum(g, 0)
    s = g.sum()
    if s <= 1e-10:
        return None, None
    g = g / s
    # Get dual variables
    try:
        duals = np.asarray(res.ineqlin.marginals, dtype=np.float64)
        duals = np.maximum(duals, 0)
    except Exception:
        duals = None
    return g, duals

def _make_initials(n, rng):
    """Generate diverse initial sequences."""
    inits = []
    # Uniform
    inits.append(np.full(n, 1.0/n))
    # Cap (peaked in middle)
    x = np.linspace(0, 1, n)
    for a in [0.5, 1.0, 2.0, 4.0]:
        p = 1.0 + a * (1 - (2*x-1)**2)
        inits.append(_project_simplex(p))
    # Cup (peaked at ends)
    for a in [0.5, 1.0, 2.0]:
        p = 1.0 + a * (2*x-1)**2
        inits.append(_project_simplex(p))
    # Geometric
    for s in [0.5, 1.0, 2.0, 5.0]:
        p = np.exp(-s * x)
        inits.append(_project_simplex(p))
        p = np.exp(-s * (1-x))
        inits.append(_project_simplex(p))
    # Power
    for a in [0.5, 1.0, 2.0, 4.0]:
        p = np.power(x + 1e-6, a)
        inits.append(_project_simplex(p))
        p = np.power(1-x + 1e-6, a)
        inits.append(_project_simplex(p))
    # Sinusoidal
    for f in [1, 2, 3, 5]:
        for amp in [0.1, 0.3, 0.5]:
            p = 1.0 + amp * np.sin(2*np.pi*f*x)
            inits.append(_project_simplex(np.maximum(p, 0)))
    # Beta-like
    for alpha, beta in [(1,1), (2,1), (1,2), (3,1), (1,3), (4,1), (1,4)]:
        p = np.power(x+1e-6, alpha) * np.power(1-x+1e-6, beta)
        inits.append(_project_simplex(p))
    # Random smooth
    for _ in range(5):
        p = 1.0 + 0.3*rng.normal(size=n)
        # Smooth
        for _ in range(3):
            p = np.convolve(p, np.ones(5)/5, mode='same')
        inits.append(_project_simplex(np.maximum(p, 0)))
    return inits

    p = _project_simplex(p0.copy())
    n = len(p)
    best_p = p.copy()
    best_score = _score(p)
    cur_score = best_score
    
    end = time.monotonic() + time_limit
    iteration = 0
    stuck = 0

        c = _conv_full(p)
        M = float(np.max(c))
        
        if M <= 0:

        remaining = max(1, end - time.monotonic() - 5)
        tl = min(lp_time, remaining * 0.1)
        g, duals = _lp_best_response(p, M, time_limit=tl)
        
        if g is None:
            stuck += 1
            if stuck > 10:
                # Try random perturbation
                p = _project_simplex(p + rng.normal(0, 0.1/n, n))
                continue

        for t in [1.0, 0.8, 0.6, 0.4, 0.25, 0.1, 0.05]:

            sc = _score(p_new)

                p = p_new
                cur_score = sc
                if sc < best_score:

                    best_p = p.copy()

                grad = np.zeros(n)
                for k in range(2*n-1):
                    if w[k] > 0:
                        j_start = max(0, k-n+1)
                        j_end = min(k, n-1)
                        for j in range(j_start, j_end+1):
                            grad[j] += w[k] * p[k-j]
                grad *= 2.0

                    sc = _score(p_new)
                    if sc < cur_score:
                        p = p_new
                        cur_score = sc
                        if sc < best_score:
                            best_score = sc
                            best_p = p.copy()
                        improved = True
                        break
            
            if not improved:
                # Random perturbation
                sigma = max(0.01, 0.5 / n)
                for _ in range(20):
                    p_new = _project_simplex(p + rng.normal(0, sigma, n))
                    sc = _score(p_new)

                        p = p_new
                        cur_score = sc
                        if sc < best_score:
                            best_score = sc
                            best_p = p.copy()

```python
"""
LP-iterative search for nonnegative step function minimizing 2*n*max(conv(p,p))/(sum(p))^2.
Uses best-response LP + subgradient + blending for iterative improvement.
"""

import time
import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
from scipy.fft import fft, ifft, next_fast_len

def _fft_len(n):
    return int(next_fast_len(max(1, n)))

def _conv_full(p, L=None):
    n = len(p)
    if L is None:
        L = _fft_len(2*n - 1)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2*n-1]

def _score(p):
    n = len(p)
    s = float(np.sum(p))
    if s < 0.01:
        return float('inf')
    c = _conv_full(p)
    return float(2.0 * n * np.max(c) / (s * s))

def _project_simplex(v):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    v = np.maximum(v, 0)  # Actually, we need proper projection
    # Proper projection
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.arange(1, n+1, dtype=np.float64)
    cond = (u * idx) > (css - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0/n, dtype=np.float64)
    k = int(np.count_nonzero(cond) - 1)
    theta = (css[k] - 1.0) / (k + 1)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s <= 1e-10:
        return np.full(n, 1.0/n, dtype=np.float64)
    return out / s

def _build_indices(n):
    rows = []
    cols = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int64), np.array(cols, dtype=np.int64)

_lp_rows = None
_lp_cols = None
_lp_n = 0

def _get_indices(n):
    global _lp_rows, _lp_cols, _lp_n
    if _lp_n != n:
        _lp_rows, _lp_cols = _build_indices(n)
        _lp_n = n
    return _lp_rows, _lp_cols

def _lp_solve(p, M, time_limit=10.0):
    n = len(p)
    rows, cols = _get_indices(n)
    data = p[rows - cols]
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M, dtype=np.float64)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None),
                      method='highs', options={'time_limit': time_limit, 'disp': False})
    except Exception as e:
        return None, None
    if res.status != 0:
        return None, None
    g = np.zeros(n)
    nz = np.where(res.x > 1e-12)[0]
    g[nz] = res.x[nz]
    s = g.sum()
    if s <= 1e-10:
        return None, None
    g /= s
    # Get duals
    duals = None
    try:
        duals = np.asarray(res.ineqlin.marginals, dtype=np.float64)
        duals = np.maximum(duals, 0)
        ds = duals.sum()
        if ds > 0:
            duals /= ds
        else:
            duals = None
    except Exception:
        duals = None
    return g, duals

def _subgrad(p, duals):
    """Compute subgradient of sum_k duals[k] * c[k] where c = p*p."""
    n = len(p)
    L = _fft_len(3*n - 2)
    # grad[i] = 2 * sum_v duals[i+v] * p[v]
    # This is cross-correlation of duals and p
    F_duals = fft(duals, L)
    F_p = fft(p, L)
    cross = np.real(ifft(F_duals * np.conj(F_p)))
    grad = 2.0 * cross[:n]
    return grad

def _make_inits(n, rng):
    inits = []
    x = np.linspace(0, 1, n)
    # Uniform
    inits.append(np.full(n, 1.0/n))
    # Caps
    for a in [0.3, 0.8, 1.5, 3.0, 6.0]:
        p = 1.0 + a * (1.0 - (2*x - 1)**2)
        inits.append(_project_simplex(np.maximum(p, 0)))
    # Cups
    for a in [0.3, 0.8, 1.5, 3.0]:
        p = 1.0 + a * (2*x - 1)**2
        inits.append(_project_simplex(p))
    # Geometric
    for s in [0.5, 1.0, 2.0, 4.0, 8.0]:
        inits.append(_project_simplex(np.exp(-s*x)))
        inits.append(_project_simplex(np.exp(-s*(1-x))))
    # Power
    for a in [0.5, 1.0, 2.0, 3.0, 5.0]:
        inits.append(_project_simplex(np.power(x + 1e-8, a)))
        inits.append(_project_simplex(np.power(1-x + 1e-8, a)))
    # Sinusoidal
    for f in [1, 2, 3, 4, 6]:
        for amp in [0.1, 0.3, 0.5, 0.8]:
            p = 1.0 + amp * np.sin(2*np.pi*f*x)
            inits.append(_project_simplex(np.maximum(p, 1e-6)))
    # Two-hump
    for mu in [0.2, 0.3, 0.4]:
        for sig in [0.03, 0.05, 0.08]:
            p = np.exp(-((x-mu)**2)/(2*sig**2)) + np.exp(-((x-(1-mu))**2)/(2*sig**2)) + 1e-6
            inits.append(_project_simplex(p))
    # Random smooth
    for _ in range(8):
        coeffs = rng.normal(0, 1, 6)
        p = np.ones(n)
        for i, c in enumerate(coeffs, 1):
            p += c * 0.3 * np.sin(2*np.pi*i*x + rng.uniform(0, 2*np.pi))
        inits.append(_project_simplex(np.maximum(p, 1e-4)))
    return inits

    """Run LP-iterative optimization starting from p0."""
    p = _project_simplex(p0.copy().astype(np.float64))
    n = len(p)
    best_p = p.copy()
    best_score = _score(p)
    cur_score = best_score
    L_score = _
