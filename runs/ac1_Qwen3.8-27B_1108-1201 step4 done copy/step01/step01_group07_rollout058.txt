"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using LP-based direction finding
combined with L-BFGS polishing. The LP-based method finds directions that
reduce the peak of the auto-convolution by solving a linear program at each
step. Multi-start with diverse lengths and initializations.

import numpy as np
import time
from scipy.optimize import linprog, minimize

def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def cross_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    result = np.fft.irfft(A * B, fft_size)
    return result[:size]

def build_lp_matrix(f):
    """Build the LP constraint matrix A where A[k,j] = f[j] if 0<=k-j<n."""
    n = len(f)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n-1, k)
        A[k, j_start:j_end+1] = f[j_start:j_end+1]
    return A

def solve_lp_direction(f, M):
    """
    Solve LP: max sum(g) s.t. g>=0, (f*g)[k] <= M for all k.
    Returns g (unnormalized, non-negative).
    """
    n = len(f)
    A = build_lp_matrix(f)
    c = -np.ones(n)  # minimize -sum(g)
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return np.ones(n) / n

def smoothed_objective(b, tau, n):
    """
    Objective and gradient for L-BFGS.
    b: log-parameters, a = softmax(b)
    Returns (obj, grad_b)
    """
    a = np.exp(b - b.max())
    a /= a.sum()
    
    conv = fast_convolve(a)
    max_conv = conv.max()
    
    # Smoothed max: (1/tau) * log(sum exp(tau * conv))
    scaled = tau * (conv - max_conv)
    log_sum = np.log(np.sum(np.exp(scaled)))
    smooth_max = max_conv + log_sum / tau
    obj = 2.0 * n * smooth_max
    
    # Weights
    w = np.exp(scaled) / np.sum(np.exp(scaled))
    
    # Gradient of smoothed_conv_max w.r.t. a[j]:
    # = sum_k w[k] * d conv[a,a][k] / d a[j]
    # d conv[a,a][k] / d a[j] = a[k-j] + a[j+k] (with valid ranges)
    
    # Compute using cross-correlation via FFT
    N = 1 << (3*n).bit_length()
    A_fft = np.fft.fft(a, N)
    W_fft = np.fft.fft(w, N)
    
    # (a ⋆ w)[m] = sum_i a[i] * w[i+m]
    # = ifft(conj(FFT(a)) * FFT(w))[m] (circular, period N)
    R = np.fft.ifft(np.conj(A_fft) * W_fft).real
    
    # T1[j] = sum_i a[i]*w[i+j] = (a⋆w)[j] = R[j] for j in [0, n-1]
    T1 = R[:n]
    
    # T2[j] = sum_k w[k]*a[j+k] = sum_i a[i]*w[i-j] = (a⋆w)[-j]
    # For j=0: (a⋆w)[0] = R[0]
    # For j>0: (a⋆w)[-j] = R[N-j]
    T2 = np.zeros(n)
    T2[0] = R[0]
    if n > 1:
        T2[1:] = R[N-1:N-n:-1]
    
    grad_a = 2.0 * n * (T1 + T2)
    
    # Convert to gradient w.r.t. b (softmax parameterization)
    dot = np.dot(grad_a, a)
    grad_b = a * (grad_a - dot)
    
    return obj, grad_b

def lbfgs_polish(f0, time_limit, tau=100.0):
    """
    Polish using L-BFGS on the smoothed objective.
    f0 should sum to 1.
    Returns (best_f, best_obj).
    """
    n = len(f0)
    b0 = np.log(np.maximum(f0, 1e-15))
    
    t0 = time.monotonic()
    best_obj = 2.0 * n * fast_convolve(f0).max()
    best_f = f0.copy()
    
    result = minimize(
        fun=lambda b: smoothed_objective(b, tau, n)[0],
        x0=b0,
        jac=lambda b: smoothed_objective(b, tau, n)[1],
        method='L-BFGS-B',
        options={'maxiter': 10000, 'maxfun': 50000,
                 'iprint': 0}
    )
    
    b_opt = result.x
    a_opt = np.exp(b_opt - b_opt.max())
    a_opt /= a_opt.sum()
    a_opt = np.maximum(a_opt, 0)
    a_opt /= a_opt.sum()
    
    obj_opt = 2.0 * n * fast_convolve(a_opt).max()
    if obj_opt < best_obj:
        best_obj = obj_opt
        best_f = a_opt.copy()
    
    return best_f, best_obj

def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Gaussian bumps
    for sf in [0.15, 0.25, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - n*0.5) / (sf*n))**2)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g
    
    # Bimodal (edges)
    bi = np.zeros(n)
    w = max(1, n // 4)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Cosine
    cos_b = 0.5 * (1 + np.cos(2*np.pi*x/n))
    cos_b /= cos_b.sum()
    inits['cosine'] = cos_b
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Binary-like (random 0/1 pattern with optimization potential)
    binary = rng.random(n) > 0.5
    inits['binary'] = binary.astype(float)
    inits['binary'] /= inits['binary'].sum()
    
    # Parent
    if parent is not None:
        if parent.size >= n:
            p = parent[:n].copy()
        else:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
        p = np.maximum(p, 0)
        if p.sum() > 1e-10:
            p /= p.sum()
            inits['parent'] = p
    
    return inits

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using LP-based directional optimization
combined with L-BFGS polishing. The LP method finds directions that reduce the
auto-convolution peak by solving a linear program at each step. Multi-start with
diverse sequence lengths (including primes) and structured initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def build_lp_matrix(f):
    """Build LP constraint matrix A where A[k,m] = f[k-m] if valid."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        A[k, m_start:m_end + 1] = f[k - m_end:k - m_start + 1]
    return A


def solve_lp_direction(f, M):
    """
    Solve LP: max sum(g) s.t. g>=0, (f*g)[k] <= M for all k.
    Returns non-negative g or None on failure.
    """
    n = len(f)
    A = build_lp_matrix(f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-8})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_optimize(f0, time_limit, t_init=0.12, verbose=False):
    """
    LP-based iterative optimization minimizing 2n*max(conv(a,a)) with sum(a)=1.
    Returns (best_f, best_obj).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f_sum = f.sum()
    if f_sum < 1e-12:
        f = np.ones(n) / n
    else:
        f /= f_sum

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    M = conv.max()
    t = t_init
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        g = solve_lp_direction(f, M)

        if g is None or g.sum() < 1e-10:
            # Try with slightly relaxed M
            M *= 1.005
            g = solve_lp_direction(f, M)
            if g is None or g.sum() < 1e-10:
                break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g /= g_sum

        # Check similarity to f; if too similar, add perturbation
        diff = np.linalg.norm(g - f)
        if diff < 1e-5:
            g = f + 0.02 * rng.normal(0, 1, n)
            g = np.maximum(g, 0.0)
            gs = g.sum()
            if gs > 1e-12:
                g /= gs
            else:
                g = f.copy()

        # Update
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        f_new_sum = f_new.sum()
        if f_new_sum < 1e-12:
            break
        f_new /= f_new_sum

        conv_new = fast_convolve(f_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-10:

            best_f = f_new.copy()
            M = conv_new.max()
            no_improve = 0
        else:
            no_improve += 1

def smoothed_obj_grad(b, tau, n):
    """
    Compute smoothed objective and gradient w.r.t. log-parameters b.
    a = softmax(b). Objective = 2n * smooth_max(conv(a,a)).
    """
    a = np.exp(b - b.max())
    a /= a.sum()

    conv = fast_convolve(a)
    max_conv = conv.max()

    scaled = tau * (conv - max_conv)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(np.sum(exp_scaled))
    smooth_max = max_conv + log_sum / tau
    obj = 2.0 * n * smooth_max

    w = exp_scaled / np.sum(exp_scaled)

    # Gradient: grad_a[j] = 2n * sum_k w[k] * (a[k-j] + a[j+k]) * I(0<=k-j<n)
    # T1[j] = sum_i a[i]*w[i+j] (cross-correlation)
    # T2[j] = a[j] * sum_{k=j}^{j+n-1} w[k]
    N = 1 << (3 * n).bit_length()
    A_fft = np.fft.fft(a, N)
    W_fft = np.fft.fft(w, N)
    R = np.fft.ifft(np.conj(A_fft) * W_fft).real
    T1 = R[:n].copy()

    # S[j] = sum of w[k] for k in [j, j+n-1] (capped at 2n-2)
    w_ext = np.zeros(2 * n)
    w_ext[:2 * n - 1] = w
    cumsum = np.concatenate([[0.0], np.cumsum(w_ext)])
    S = cumsum[n:2 * n] - cumsum[:n]
    T2 = a * S

    grad_a = 2.0 * n * (T1 + T2)

    # Convert to gradient w.r.t. b via softmax chain rule
    dot = np.dot(grad_a, a)
    grad_b = a * (grad_a - dot)

    return obj, grad_b


def lbfgs_polish(f0, time_limit, tau=200.0):
    """Polish using L-BFGS on smoothed objective. f0 sums to 1."""
    n = len(f0)
    b0 = np.log(np.maximum(f0, 1e-15))

    t0 = time.monotonic()
    best_obj = 2.0 * n * fast_convolve(f0).max()
    best_f = f0.copy()

    def fun(b):
        return smoothed_obj_grad(b, tau, n)[0]

    def jac(b):
        return smoothed_obj_grad(b, tau, n)[1]

    max_iter = int((time_limit - (time.monotonic() - t0)) * 20)
    max_iter = max(50, min(max_iter, 5000))

    try:
        result = minimize(fun, b0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'maxfun': max_iter * 2, 'iprint': 0})
        b_opt = result.x
        a_opt = np.exp(b_opt - b_opt.max())
        a_opt = np.maximum(a_opt, 0.0)
        a_opt_sum = a_opt.sum()
        if a_opt_sum > 1e-12:
            a_opt /= a_opt_sum
            obj_opt = 2.0 * n * fast_convolve(a_opt).max()
            if obj_opt < best_obj:
                best_obj = obj_opt
                best_f = a_opt.copy()
    except Exception:
        pass

    return best_f, best_obj


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations for sequence of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    for sf in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g

    bi = np.zeros(n)
    w = max(1, n // 4)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    cos_b = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
    cos_b /= cos_b.sum()
    inits['cosine'] = cos_b

    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    binary = (rng.random(n) > 0.5).astype(float)
    if binary.sum() > 0:
        inits['binary'] = binary / binary.sum()

    # Zipper pattern (alternating high/low)
    zip_pat = np.ones(n)
    zip_pat[::2] = 2.0
    zip_pat /= zip_pat.sum()
    inits['zipper'] = zip_pat

    if parent is not None:
        if parent.size >= n:
            p = parent[:n].copy()
        else:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
        p = np.maximum(p, 0.0)
        if p.sum() > 1e-10:
            p /= p.sum()
            inits['parent'] = p

    return inits
