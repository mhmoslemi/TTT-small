"""
LP-iterative optimization with top-K constraints, line search,
subgradient polish, and block coordinate descent.
Minimizes 2*n*max(conv(f,f))/sum(f)^2 for non-negative f.
"""
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time

def fast_score(f):
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))

def build_conv_matrix(n, f):
    """Sparse (2n-1) x n matrix A where (A @ g)[k] = (convolve(f,g))[k]."""
    col_idx = np.repeat(np.arange(n), n)
    offsets = np.tile(np.arange(n), n)
    base = np.repeat(np.arange(n), n)
    row_idx = base + offsets
    data = np.tile(f, n)
    A = sp.coo_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n)).tocsr()
    return A

def solve_lp(f, M_vals, time_limit=5.0):
    """Solve max sum(g) s.t. (f*g)[k] <= M_k for k in M_vals, g >= 0."""
    n = len(f)
    A = build_conv_matrix(n, f)
    lags = sorted(M_vals.keys())
    A_sub = A[lags, :].tocsr()
    b = np.array([M_vals[k] for k in lags])
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_sub, b_ub=b, bounds=bounds,
                      method='highs', options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def line_search(f, g, current_score, max_shrinks=10):
    """Find best t such that score((1-t)f + t*g) < current_score."""
    best_t = 0.0
    best_sc = current_score
    t = 1.0
    for _ in range(max_shrinks):
        cand = (1 - t) * f + t * g
        cand = np.maximum(cand, 0)
        s = cand.sum()
        if s < 1e-10:
            t *= 0.5
            continue
        cand = cand / s  # normalize
        sc = fast_score(cand)
        if sc < best_sc:
            best_sc = sc
            best_t = t
        t *= 0.5
    return best_t, best_sc

def make_initial(n, rng, method='uniform'):
    if method == 'uniform':
        a = np.ones(n)
    elif method == 'perturbed':
        a = 1.0 + rng.normal(0, 0.3, n)
    elif method == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
    elif method == 'edge_peak':
        x = np.arange(n)
        mid = (n-1)/2
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x / n)
    elif method == 'bimodal':
        x = np.arange(n, dtype=float)
        a = np.exp(-((x - n*0.25)/(n*0.1))**2) + np.exp(-((x - n*0.75)/(n*0.1))**2) + 0.1
    elif method == 'arch':
        x = np.arange(n, dtype=float) / (n-1)
        a = 1.0 - 0.3 * (2*x - 1)**2  # arch: higher in middle
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a

def block_perturb(f, rng, block_frac=0.1, sigma=0.3):
    n = len(f)
    cand = f.copy()
    w = max(1, int(n * block_frac))
    start = int(rng.integers(0, n - w + 1))
    cand[start:start+w] *= np.exp(rng.normal(0, sigma, w))
    cand = np.maximum(cand, 0)
    s = cand.sum()
    if s < 1e-10:
        return f
    return cand / s

def lp_iterate(f, budget_s, K=None, lp_time=5.0, max_stall=30):
    """Run LP iterations on f until budget exhausted or stalled."""
    n = len(f)
    if K is None:
        K = min(n, max(5, n // 10))
    f = f.copy()
    f = f / f.sum()
    best_f = f.copy()
    best_sc = fast_score(f)
    stall = 0
    it = 0
    
    while time.monotonic() < budget_s:
        conv = np.convolve(f, f)
        M = conv.max()
        
        # Get top-K lags
        if K < len(conv):
            top_k = np.argsort(conv)[-K:]
            M_vals = {int(k): float(conv[k]) for k in top_k}
        else:
            M_vals = {k: float(v) for k, v in enumerate(conv)}
        
        g = solve_lp(f, M_vals, time_limit=lp_time)
        if g is None or g.sum() < 1e-10:
            # Perturb and try again
            f = block_perturb(f, np.random.default_rng(it + 42))
            stall += 1
            if stall > max_stall:
                break
            it += 1
            continue
        
        g = g / g.sum()
        
        # Line search
        t, sc = line_search(f, g, best_sc)
        if t > 0:
            f = (1 - t) * f + t * g
            f = np.maximum(f, 0)
            s = f.sum()
            if s > 1e-10:
                f = f / s
            if sc < best_sc:
                best_sc = sc
                best_f = f.copy()
                stall = 0
            else:
                stall += 1
        else:
            # No improvement, perturb
            f = block_perturb(f, np.random.default_rng(it + 42),
                             block_frac=0.05, sigma=0.2)
            stall += 1
        
        it += 1
        if stall > max_stall:
            break
    
    return best_f, best_sc

def subgradient_polish(f, budget_s, steps=10000):
    """Subgradient method with step size decay."""
    n = len(f)
    f = f.copy()
    f = f / f.sum()
    best_f = f.copy()
    best_sc = fast_score(f)
    eta = 0.1
    decay = 0.999
    
    for it in range(steps):
        if time.monotonic() > budget_s:
            break
        conv = np.convolve(f, f)
        M = conv.max()
        # Find all lags achieving the max (or within epsilon)
        active = np.where(conv >= M * (1 - 1e-6))[0]
        
        # Subgradient: average of reflected f at active lags
        grad = np.zeros(n)
        for k in active:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += f[j]
        grad /= len(active)
        
        # Projected subgradient step (project onto simplex)
        f_new = f - eta * grad
        # Project onto simplex {x >= 0, sum(x) = 1}
        f_new = project_simplex(f_new)
        
        sc = fast_score(f_new)
        if sc < best_sc:
            best_sc = sc
            best_f = f_new.copy()
        f = f_new
        eta *= decay
    
    return best_f, best_sc

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

"""
LP-iterative optimization with top-K peak constraints, golden-section line search,
subgradient polish, and diverse multi-start to minimize 2*n*max(conv(f,f))/sum(f)^2.
"""
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog, minimize_scalar
import time


def fast_score(f):
    """Compute 2*n*max(conv)/sum^2 efficiently."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))


def build_conv_matrix(n, f):
    """Sparse (2n-1) x n matrix A: (A @ g)[k] = (convolve(f, g))[k]."""
    col_idx = np.repeat(np.arange(n), n)
    base = np.repeat(np.arange(n), n)
    offsets = np.tile(np.arange(n), n)
    row_idx = base + offsets
    data = np.tile(f, n)
    A = sp.coo_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n)).tocsr()
    return A


def solve_lp(f, M_vals, time_limit=5.0):
    """Solve max sum(g) s.t. (f*g)[k] <= M_k for k in M_vals keys, g >= 0."""
    n = len(f)
    A = build_conv_matrix(n, f)
    lags = sorted(M_vals.keys())
    A_sub = A[lags, :].tocsr()
    b = np.array([M_vals[k] for k in lags])
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_sub, b_ub=b, bounds=bounds,
                      method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def golden_section_min(C, D, E):
    """Find t in [0,1] minimizing max_k[(1-t)^2*C_k + 2t(1-t)*D_k + t^2*E_k]."""
    def obj(t):
        vals = (1.0 - t) ** 2 * C + 2.0 * t * (1.0 - t) * D + t ** 2 * E
        return vals.max()
    res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-8})
    return res.x


def make_initial(n, rng, method='uniform'):

    elif method == 'arch':

        a = 1.0 - 0.4 * (2.0 * x - 1.0) ** 2
    elif method == 'random':
        a = rng.random(n)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def block_perturb(f, rng, block_frac=0.1, sigma=0.3):
    """Multiplicative block perturbation, renormalized."""
    n = len(f)
    cand = f.copy()
    w = max(1, int(n * block_frac))
    start = int(rng.integers(0, max(1, n - w + 1)))
    stop = min(n, start + w)
    cand[start:stop] *= np.exp(rng.normal(0.0, sigma, stop - start))
    cand = np.maximum(cand, 0.0)
    s = cand.sum()
    if s < 1e-10:
        return f
    return cand / s

    best_f = f.copy()

    stall = 0
    it = 0

        else:

        g = solve_lp(f, M_vals, time_limit=lp_time)

            continue

        g = g / g.sum()

def subgradient_polish(f, deadline, initial_eta=0.05, decay=0.9995):
    """Subgradient method projected onto simplex."""
    n = len(f)
    f = f.copy()
    f = f / f.sum()
    best_f = f.copy()
    best_sc = fast_score(f)
    eta = initial_eta

    while time.monotonic() < deadline:
        if time.monotonic() > deadline - 2:
            break
        conv = np.convolve(f, f)
        M = conv.max()
        tol = max(1e-10, M * 1e-7)
        active = np.where(conv >= M - tol)[0]
        if len(active) == 0:
        active = np.array([np.argmax(conv)])

        # Compute subgradient (average reflected f at active lags)
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for i in range(lo, hi):
                j = k - i
                if 0 <= j < n:
                    grad[i] += f[j]
        grad /= max(1, len(active))

        # Projected subgradient step
        f_new = f - eta * grad
        # Project onto simplex {x >= 0, sum(x) = 1}
        f_new = _project_simplex(f_new)

        sc = fast_score(f_new)
        if sc < best_sc:
            best_sc = sc
            best_f = f_new.copy()
        f = f_new
        eta *= decay

    return best_f, best_sc


def _project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def local_search_polish(f, deadline, rng_seed=12345):
    """Try targeted local moves: swap mass between pairs of positions."""
    n = len(f)
    best_f = f.copy()
    best_sc = fast_score(best_f)
    rng = np.random.default_rng(rng_seed)
    it = 0

    while time.monotonic() < deadline:
        if time.monotonic() > deadline - 2:
            break
        it += 1
        # Pick two positions and try moving mass between them
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        # Try several fractions of mass transfer
        for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
            amt = best_f[i] * frac
            if amt < 1e-12:
                continue
            cand = best_f.copy()
            cand[i] -= amt
            cand[j] += amt
            if cand[i] < 0:
                continue
            sc = fast_score(cand)
            if sc < best_sc - 1e-10:
                best_sc = sc
                best_f = cand
                break

        # Occasionally try smoothing a small window
        if it % 5 == 0:
            w = max(2, n // 50)
            start = int(rng.integers(0, n - w))
            cand = best_f.copy()
            window = cand[start:start + w].copy()
            for d in range(1, w - 1):
                window[d] = 0.5 * (window[d] + 0.25 * (window[d - 1] + window[d + 1]))
            cand[start:start + w] = window
            s = cand.sum()
            if s > 1e-10:
                cand = cand / s
                sc = fast_score(cand)
                if sc < best_sc - 1e-10:
                    best_sc = sc
                    best_f = cand

    return best_f, best_sc

            bs = sc
            best = cand.copy()
            print(f"  NEW BEST: {bs:.6f} (n={best.size}) [t={time.monotonic() - (deadline - budget_s + 30):.0f}s]")
            return True
        return False

    # ---- Phase 1: LP iterations with multiple starts ----
    # Define combinations of (length, method)
    lengths = [512]
    if best.size not in lengths and best.size <= 2048:
        lengths.append(int(best.size))
    for ln in [384, 768, 1024, 1536, 2048]:
        if ln not in lengths:
            lengths.append(ln)

    methods = ['uniform', 'perturbed', 'cosine', 'edge_peak', 'bimodal', 'arch', 'random']

    # Allocate time: give more to the parent length, less to others
    # Phase 1 gets 70% of total time
    phase1_budget = (deadline - time.monotonic()) * 0.70
    n_combos = len=lengths) * len(methods)
    # Weight: parent length gets 3x, others get 1x
    weights = []
    for n in lengths:
        w = 3.0 if n == best.size else 1.0
        for _ in range(len(methods)):
            weights.append(w)
    total_w = sum(weights)
    per_combo_time = [w / total_w * phase1_budget for w in weights]
    # Cap each combo at 120s, floor at 10s
    per_combo_time = [max(10.0, min(120.0, t)) for t in per_combo_time]

    # Also try the parent directly first
    print(f"[Phase 1] Parent LP: n={best.size}, budget={min(120, per_combo_time[0])}s")
    p_deadline = min(deadline, time.monotonic() + min(120.0, phase1_budget * 0.15))
    pb, psc = lp_phase(best, p_deadline, K=max(5, best.size // 8), lp_time=3.0)
    try_update(pb, psc)

    # Now try other combos
    combo_idx = 0
    for n in lengths:
        if time.monotonic() > deadline - 60:
            break
        for m in methods:
            if time.monotonic() > deadline - 60:
                break
            start = make_initial(n, rng, m)
            # For the parent length, also try starting from the current best (reshaped/padded)
            budget = per_combo_time[combo_idx] if combo_idx < len(per_combo_time) else 30.0
            combo_deadline = min(deadline - 60, time.monotonic() + budget)
            if combo_deadline - time.monotonic() < 5:
                combo_idx += 1
                continue
            print(f"[Phase 1] n={n} method={m} budget={budget:.0f}s", flush=True)
            cb, csc = lp_phase(start, combo_deadline,
                               K=max(5, n // 8), lp_time=3.0)
            try_update(cb, csc)
            combo_idx += 1

    # ---- Phase 2: Subgradient polish on best ----
    remaining = deadline - time.monotonic()
    if remaining > 30:
        phase2_budget = min(remaining * 0.4, 300.0)
        print(f"[Phase 2] Subgradient polish: budget={phase2_budget:.0f}s")
        phase2_deadline = time.monotonic() + phase2_budget
        pb2, psc2 = subgradient_polish(best, phase2_deadline,
                                       initial_eta=0.03, decay=0.9997)
        try_update(pb2, psc2)

    # ---- Phase 3: Local search ----
    remaining = deadline - time.monotonic()
    if remaining > 20:
        phase3_budget = min(remaining * 0.5, 200.0)
        print(f"[Phase 3] Local search: budget={phase3_budget:.0f}s")
        phase3_deadline = time.monotonic() + phase3_budget
        pb3, psc3 = local_search_polish(best, phase3_deadline, rng_seed=seed + 999)
        try_update(pb3, psc3)

    # ---- Phase 4: Final LP polish on the best ----
    remaining = deadline - time.monotonic()
    if remaining > 20:
        phase4_budget = min(remaining * 0.8, 200.0)
        print(f"[Phase 4] Final LP polish: budget={phase4_budget:.0f}s")
        phase4_deadline = time.monotonic() + phase4_budget
        pb4, psc4 = lp_phase(best, phase4_deadline,
                             K=max(5, best.size // 6), lp_time=2.0, max_stall=20)
        try_update(pb4, psc4)

    # Final verification
    final_score = fast_score(best)
    best = np.clip(best, 0.0, 1000.0)
    if best.size == 0 or not np.isfinite(best).all():
        best = np.ones(512)
    s = best.sum()
    if s < 0.01:
        best = np.ones(512)
    else:
        # Keep original scale (score is scale-invariant, but let's keep values reasonable)
        best = best * (s / s)  # no-op, just ensure clarity

    final_check = float(evaluate_sequence([float(x) for x in best]))
    print(f"FINAL: score={final_check:.6f} (n={best.size})")
    return [float(max(0.0, min(1000.0, x))) for x in best]
