**Alternative: Use the structure more directly**

The objective is 2n * max(c) / sum(a)^2. With normalization sum(a) = sqrt(2n), we minimize max(c).

The convolution c = a * a has 2n-1 elements. The peak is what we want to minimize.

Minimize 2*n*max(self-convolution)/sum^2 for non-negative sequences.
Uses FFT-based convolution, gradient methods for initial optimization,
and coordinate descent with exact 1D subproblem for polishing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_to_sum(a, S):
    """Clamp to [0, 1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        a = np.full(len(a), S / len(a), dtype=np.float64)
    else:
        a = a * (S / s)
    return a

def compute_grad_topk(a, c, k=10):
    """Compute subgradient using top-k active lags with softmax weights."""
    n = len(a)
    # Use softmax weighting over top-k
    top_idx = np.argpartition(c, -k)[-k:]
    top_vals = c[top_idx]
    # Softmax weights
    t_max = top_vals.max()
    exp_v = np.exp(top_vals - t_max)
    weights = exp_v / exp_v.sum()
    
    grad = np.zeros(n, dtype=np.float64)
    for w, ki in zip(weights, top_idx):
        ki = int(ki)
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if lo <= hi:
            grad[lo:hi+1] += 2.0 * w * a[ki-hi:ki-lo+1][::-1]
    return grad

def adam_optimize(a_init, n, S, max_time, rng, lr_init=None, top_k=10):
    """Adam optimizer with simplex projection."""
    if lr_init is None:
        lr_init = 0.3 * S / n
    a = normalize_to_sum(a_init, S)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        t += 1
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        grad = compute_grad_topk(a, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad = grad / gnorm
        
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        lr = lr_init * (1.0 / (1.0 + 0.002 * t))
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new, S)
        
        # Recompute convolution periodically
        if t % 5 == 0:
            c = fft_convolve_self(a)
    
    c = fft_convolve_self(a)
    if c.max() < best_score:
        best_score = c.max()
        best_local = a.copy()
    
    return best_local, best_score

def mirror_descent(a_init, n, S, max_time, rng, eta_init=0.2, top_k=10):
    """Mirror descent (multiplicative weights)."""
    a = normalize_to_sum(a_init, S)
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    t = 0
    
    while time.monotonic() - start < max_time:
        t += 1
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        grad = compute_grad_topk(a, c, top_k)
        gmax = np.abs(grad).max()
        if gmax < 1e-12:
            break
        grad = grad / gmax
        
        eta = eta_init * (1.0 / (1.0 + 0.005 * t))
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.full(n, S / n, dtype=np.float64)
        else:
            a = a * (S / s)
        
        if t % 5 == 0:
            c = fft_convolve_self(a)
    
    c = fft_convolve_self(a)
    if c.max() < best_score:
        best_score = c.max()
        best_local = a.copy()
    
    return best_local, best_score

"""
Minimize 2*n*max(self-convolution)/sum^2 for non-negative sequences.
Strategy:
  1. Use FFT-based convolution for speed.
  2. Gradient-based methods (Adam, Mirror Descent) for initial optimization.
  3. Coordinate descent with exact 1D subproblem (bisection) for intensive polishing.
  4. Multiple initializations and random restarts to escape local minima.
  5. Time-budgeted phases: explore -> coordinate descent -> final polish.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT (length 2n-1)."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clamp to [0, 1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.full(len(a), S / len(a), dtype=np.float64)
    return a * (S / s)


def compute_grad_topk(a, c, k=16):
    """Subgradient using top-k lags with softmax weighting."""
    n = len(a)
    k = min(k, len(c))
    top_idx = np.argpartition(c, -k)[-k:]
    top_vals = c[top_idx]
    tmax = top_vals.max()
    exp_v = np.exp(top_vals - tmax)
    weights = exp_v / exp_v.sum()
    grad = np.zeros(n, dtype=np.float64)
    for w_val, ki in zip(weights, top_idx):
        ki = int(ki)
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if lo <= hi:
            grad[lo:hi + 1] += 2.0 * w_val * a[ki - hi:ki - lo + 1][::-1]
    return grad


def adam_optimize(a_init, n, S, max_time, rng, lr_init=None, top_k=16):
    """Adam with simplex projection."""
    if lr_init is None:
        lr_init = 0.3 * S / n
    a = normalize_to_sum(a_init, S)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()

    while time.monotonic() - start < max_time:
        t += 1
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        grad = compute_grad_topk(a, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad = grad / gnorm

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_init / (1.0 + 0.003 * t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new, S)

        if t % 3 == 0:
            c = fft_convolve_self(a)

    c = fft_convolve_self(a)
    if c.max() < best_score:
        best_score = c.max()
        best_local = a.copy()
    return best_local, best_score


def mirror_descent(a_init, n, S, max_time, eta_init=0.15, top_k=16):
    """Mirror descent (multiplicative weights), respects non-negativity."""
    a = normalize_to_sum(a_init, S)
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    t = 0

    while time.monotonic() - start < max_time:
        t += 1
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        grad = compute_grad_topk(a, c, top_k)
        gmax = np.abs(grad).max()
        if gmax < 1e-12:
            break
        grad = grad / gmax

        eta = eta_init / (1.0 + 0.004 * t)
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.full(n, S / n, dtype=np.float64)
        else:
            a = a * (S / s)

        if t % 3 == 0:
            c = fft_convolve_self(a)

    c = fft_convolve_self(a)
    if c.max() < best_score:
        best_score = c.max()
        best_local = a.copy()
    return best_local, best_score


def coordinate_descent(a_init, n, S, max_time, verbose=True, skip_threshold=1e-10):
    """
    Coordinate descent with exact 1D subproblem solved by bisection.
    For each coordinate i, find t* = argmin_{t>=0} max_k c_new[k]
    where c_new is the convolution after updating a[i] to t.
    Uses linear approximation for the 1D problem.
    """
    a = normalize_to_sum(a_init, S).copy()
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    n_conv = 2 * n - 1
    num_passes = 0

    while time.monotonic() - start < max_time:
        num_passes += 1
        order = np.random.permutation(n)
        improved = 0

        for i in order:
            if time.monotonic() - start > max_time:
                break

            ai = a[i]

            # Affected range in convolution: k in [i, i+n-1] clipped to [0, 2n-2]
            lo_aff = i
            hi_aff = min(i + n - 1, n_conv - 1)
            m_hi = hi_aff - lo_aff + 1

            # Direction vector: g[k] = 2*a[k-i] for k in [lo_aff, hi_aff]
            g_vec = 2.0 * a[:m_hi]

            # base_max: max of c outside affected range
            if lo_aff > 0 and hi_aff < n_conv - 1:
                base_max = max(c[:lo_aff].max(), c[hi_aff + 1:].max())
            elif lo_aff > 0:
                base_max = c[:lo_aff].max()
            elif hi_aff < n_conv - 1:
                base_max = c[hi_aff + 1:].max()
            else:
                base_max = 0.0

            # b[m] = c[lo_aff+m] - ai * g_vec[m]  (convolution without a[i] contribution)
            b = c[lo_aff:hi_aff + 1] - ai * g_vec

            # Minimize h(t) = max_m (b[m] + t * g_vec[m]) over t >= 0
            # Check if t=0 is optimal: subgradient at 0 is g_vec[argmax(b)]
            argmax_b = np.argmax(b)
            if g_vec[argmax_b] >= 0:
                t_opt = 0.0
            else:
                # Bisection to find where subgradient changes sign
                b_range = b.max() - b.min()
                g_range = g_vec.max() - g_vec.min()
                if g_range < 1e-15:
                    t_opt = 0.0
                else:
                    t_lo = 0.0
                    t_hi = max(1.0, 3.0 * b_range / g_range)
                    # Ensure t_hi is large enough
                    for _ in range(5):
                        vals_chk = b + t_hi * g_vec
                        if g_vec[np.argmax(vals_chk)] > 0:
                            break
                        t_lo = t_hi
                        t_hi *= 2.0
                    # Bisection
                    for _ in range(30):
                        t_mid = (t_lo + t_hi) / 2.0
                        vals = b + t_mid * g_vec
                        if g_vec[np.argmax(vals)] > 0:
                            t_hi = t_mid
                        else:
                            t_lo = t_mid
                    t_opt = (t_lo + t_hi) / 2.0

            # Also check that t_opt doesn't make things worse due to base_max
            # f(t) = max(base_max, max_m(b[m] + t*g[m]))
            # Since we minimized the inner max, and f = max(base_max, inner),
            # the optimum of f is at t_opt (or any t where inner <= base_max)

            if abs(t_opt - ai) > 1e-14:
                delta = t_opt - ai
                c[lo_aff:hi_aff + 1] += delta * g_vec
                a[i] = t_opt
                new_max = c.max()
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_local = a.copy()
                    improved += 1

        if verbose and num_passes % 3 == 0:
            elapsed = time.monotonic() - start
            print(f"  CD pass {num_passes}: max_c={best_score:.8f}, improved={improved}, time={elapsed:.1f}s")

        if improved == 0:
            if verbose:
                print(f"  CD converged at pass {num_passes}")
            break

    return best_local, best_score


def try_initialization(init_name, init_arr, n, S, budget, rng):
    """Try several optimizers on an initialization, return best."""
    best_a = None
    best_s = np.inf
    methods = [

    ]
    for name, func in methods:
        if time.monotonic() > 0:  # just to keep structure
            pass
        try:

            if s < best_s:

                best_a = a.copy()
                print(f"    {init_name}/{name}: max_c={s:.8f}")
        except Exception as e:
            print(f"    {init_name}/{name} failed: {e}")

        print(f"  FINAL: evaluate_sequence = {final_ver:.6f}" if np.isfinite(final_ver) else "  FINAL: inf")
        # Also compute our estimate
        obj_est = 2.0 * len(global_a) * global_best / (global_a.sum() ** 2)
        print(f"  FINAL estimated objective: {obj_est:.6f}")
        print(f"  Total time: {time.monotonic() - total_start:.1f}s")
    else:
        print("  WARNING: No valid sequence found, using fallback.")
        global_a = np.ones(256, dtype=np.float64)

    return [float(x) for x in global_a]
