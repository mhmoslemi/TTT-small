The FZ algorithm:

2. Compute the autocorrelation
3. Set a threshold T (initially high)

6. Update: `a = a + delta_a` (constrained to be real and have the right energy)
7. Lower the threshold and repeat

   d. Update `a` to reduce the peaks, while maintaining non-negativity and normalization
3. **Repeat** with decreasing threshold

Hybrid subgradient + LP optimization for minimizing peak auto-convolution
of non-negative sequences. Combines fast subgradient descent with precise
LP-based refinement, using multiple initializations and sequence lengths.
"""

from scipy.optimize import linprog

import time


def project_to_simplex(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]

    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w / w.sum()

    """Compute the score: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def compute_subgradient_fft(a, active_mask):
    """
    Compute the subgradient of max(c_k) where c = a*a.
    g[j] = (2/|S|) * sum_{k in S} a[k-j] for valid k-j.
    """
    n = len(a)
    size = 1 << max(1, (2 * n - 1).bit_length())
    A = np.fft.rfft(a, size=size)
    B = np.fft.rfft(active_mask, size=size)
    g = np.fft.irfft(B * A.conjugate(), size=size)[:n]
    g *= 2.0 / active_mask.sum()
    return g


def subgradient_optimize(a, time_limit=60.0, verbose=True):
    """Subgradient descent with momentum to minimize max(auto-conv)."""
    n = len(a)
    a = np.maximum(0.0, a.copy().astype(np.float64))
    s = a.sum()
    if s < 1e-15:

    best_score = score_sequence(a)
    
    v = np.zeros(n)
    mu = 0.9
    lr0 = 0.5
    start = time.monotonic()
    it = 0
    stagnant = 0
    
    while time.monotonic() - start < time_limit:
        c = np.convolve(a, a)
        M = c.max()
        
        # Adaptive active set: start narrow, widen over time
        frac = min(0.4, 0.02 + 0.002 * it)
        n_active = max(1, int(frac * len(c)))
        sorted_c = np.sort(c)[::-1]
        threshold = sorted_c[n_active - 1]
        active_mask = (c >= threshold).astype(np.float64)
        
        g = compute_subgradient_fft(a, active_mask)
        
        lr = lr0 / (1.0 + 0.01 * it)
        v = mu * v - lr * g
        a_new = a + v
        a_new = project_to_simplex(a_new)

        if score < best_score - 1e-10:

            stagnant = 0

def build_conv_matrix(a):
    """Build sparse matrix for convolution with fixed a."""
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_step(a, M, timeout=10.0):
    """Solve LP: maximize sum(g) s.t. A@g <= M, g >= 0."""
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def lp_refine(a, time_limit=300.0, verbose=True):
    """LP-based iterative refinement."""
    n = len(a)
    a = np.maximum(0.0, a.copy().astype(np.float64))
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    
    best_a = a.copy()
    best_score = score_sequence(a)
    start = time.monotonic()
    stagnant = 0
    it = 0
    
    while time.monotonic() - start < time_limit:
        elapsed = time.monotonic() - start
        c = np.convolve(a, a)
        score = 2.0 * n * c.max() / (a.sum() ** 2)
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 10 == 0 or it < 3):
                print(f"    LP iter {it:3d} ({elapsed:5.1f}s): score={score:.6f}")
        else:
            stagnant += 1
        
        if stagnant > 30:
            if verbose:
                print(f"    LP stagnant at iter {it}, best={best_score:.6f}")
            break
        
        M = c.max()
        remaining = time_limit - elapsed
        if remaining < 3.0:
            break
        timeout = max(2.0, min(30.0, remaining * 0.2))
        g = solve_lp_step(a, M, timeout=timeout)
        if g is None:
            break
        
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            if verbose:
                print(f"    LP no improvement (sum_g={sum_g:.8f}) at iter {it}")
            break
        
        g = g / sum_g
        alpha = min(0.5, max(0.05, 1.0 / (1.0 + 0.5 * it)))
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = np.maximum(0.0, a_new)
        a_new = a_new / a_new.sum()
        
        new_score = score_sequence(a_new)
        if new_score <= score + 1e-10:
            a = a_new
        else:
            a2 = (1.0 - alpha * 0.3) * a + alpha * 0.3 * g
            a2 = np.maximum(0.0, a2)
            a2 = a2 / a2.sum()
            a = a2
        
        it += 1
    
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    if verbose:
        print(f"    LP done: best={best_score:.6f} after {it+1} iters, {time.monotonic()-start:.1f}s")
    return best_a, best_score

def init_raised_cosine_p(n, p=1.0):
    """Generalized raised cosine: (0.5*(1+cos(2*pi*t/n)))^p"""
    t = np.arange(n, dtype=np.float64)
    base = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))
    return np.maximum(0.0, base ** p)


def init_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def init_beta_kernel(n, alpha=2.0, beta=2.0):
    """Beta distribution kernel on [0, n-1]."""
    t = np.arange(n, dtype=np.float64)
    x = (t + 0.5) / n  # in (0, 1)
    # Beta density (unnormalized): x^(alpha-1) * (1-x)^(beta-1)
    if alpha <= 0 or beta <= 0:
        return np.ones(n) / n
    a = x ** (alpha - 1.0) * (1.0 - x) ** (beta - 1.0)
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_fourier_smooth(n, rng, n_freqs=5):
    """Random smooth function built from a few Fourier modes."""
    a = np.ones(n) / n
    t = np.arange(n, dtype=np.float64)
    for _ in range(n_freqs):
        freq = float(rng.integers(1, 4))
        phase = float(rng.uniform(0, 2 * np.pi))
        amp = float(rng.uniform(0.1, 0.4))
        a += amp * np.cos(2 * np.pi * freq * t / n + phase)
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

"""
Hybrid subgradient + LP optimization for minimizing peak auto-convolution
of non-negative sequences. Uses FFT-based subgradient descent for fast initial
convergence, then LP-based iterative refinement for precise polishing.
Multiple initializations and sequence lengths are explored within a time budget.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def project_to_simplex(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w / w.sum()


def score_sequence(a):
    """Compute the score: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)

    B = np.fft.rfft(active_mask, size=size)
    g = np.fft.irfft(B * A.conjugate(), size=size)[:n]
    g *= 2.0 / active_mask.sum()
    return g

            stagnant = 0
        else:
            stagnant += 1

        a = a_new
        it += 1

        if verbose and it % 50 == 0:
            print(f"    SG iter {it:4d}: score={score:.6f}, M={M:.8f}")

        if stagnant > 200:
            v = np.zeros(n)
            stagnant = 0

    return best_a, best_score


def build_conv_matrix(a):
    """Build sparse convolution matrix: (M @ g)[k] = sum_j a[j]*g[k-j]."""
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_step(a, M, timeout=10.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def lp_refine(a, time_limit=300.0, verbose=True):
    """LP-based iterative refinement to reduce peak auto-convolution."""
    n = len(a)
    a = np.maximum(0.0, a.copy().astype(np.float64))
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s

    best_a = a.copy()
    best_score = score_sequence(a)
    start = time.monotonic()
    stagnant = 0
    it = 0

    while time.monotonic() - start < time_limit:
        elapsed = time.monotonic() - start
        c = np.convolve(a, a)
        score = 2.0 * n * c.max() / (a.sum() ** 2)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 10 == 0 or it < 5):
                print(f"    LP iter {it:3d} ({elapsed:5.1f}s): score={score:.6f}")
        else:
            stagnant += 1

        if stagnant > 25:
            if verbose:
                print(f"    LP stagnant at iter {it}, best={best_score:.6f}")
            break

        M = c.max()
        remaining = time_limit - elapsed
        if remaining < 3.0:
            break
        timeout = max(2.0, min(30.0, remaining * 0.15))
        g = solve_lp_step(a, M, timeout=timeout)
        if g is None:
            break

        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            if verbose:
                print(f"    LP no improvement (sum_g={sum_g:.8f}) at iter {it}")
            break

        g = g / sum_g
        # Adaptive step size
        alpha = min(0.4, max(0.03, 0.8 / (1.0 + 0.3 * it)))
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = np.maximum(0.0, a_new)
        a_new = a_new / a_new.sum()

        new_score = score_sequence(a_new)
        if new_score <= score + 1e-10:
            a = a_new
        else:
            a2 = (1.0 - alpha * 0.3) * a + alpha * 0.3 * g
            a2 = np.maximum(0.0, a2)
            a2 = a2 / a2.sum()
            a = a2

        it += 1

    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    if verbose:
        print(f"    LP done: best={best_score:.6f}, {it+1} iters, {time.monotonic()-start:.1f}s")
    return best_a, best_score


def local_search_refine(a, time_limit=60.0, rng=None, verbose=True):
    """Local search with random segment perturbations and symmetry enforcement."""
    if rng is None:
        rng = np.random.default_rng(42)
    n = len(a)
    a = np.maximum(0.0, a.copy().astype(np.float64))
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s

    best_a = a.copy()
    best_score = score_sequence(a)
    start = time.monotonic()
    accept_count = 0

    while time.monotonic() - start < time_limit:
        # Random segment perturbation
        width = max(3, int(rng.integers(n // 150, max(4, n // 8))))
        pos = int(rng.integers(0, max(1, n - width)))
        candidate = a.copy()
        candidate[pos:pos + width] *= np.exp(rng.normal(0.0, 0.04, size=width))

        # Occasionally enforce symmetry
        if rng.random() < 0.25:
            candidate = 0.5 * (candidate + candidate[::-1])

        candidate = np.maximum(0.0, candidate)
        cs = candidate.sum()
        if cs > 1e-15:
            candidate = candidate / cs
            score = score_sequence(candidate)
            if score < best_score - 1e-12:
                best_score = score
                best_a = candidate.copy()
                a = candidate.copy()
                accept_count += 1
                if verbose and accept_count % 30 == 0:
                    print(f"    LS accept {accept_count} ({time.monotonic()-start:.1f}s): score={best_score:.6f}")

    if verbose:
        print(f"    LS done: best={best_score:.6f}, {time.monotonic()-start:.1f}s")
    return best_a, best_score


def init_raised_cosine_p(n, p=1.0):
    """Generalized raised cosine: (0.5*(1+cos(2*pi*t/n)))^p."""
    t = np.arange(n, dtype=np.float64)
    base = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))
    return np.maximum(0.0, base ** p)


def init_gaussian(n, sigma_frac=0.25):
    """Gaussian bump centered in the sequence."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def init_beta_kernel(n, alpha=3.0, beta=3.0):
    """Beta distribution kernel (symmetric when alpha=beta)."""
    t = np.arange(n, dtype=np.float64)
    x = (t + 0.5) / n
    x = np.clip(x, 1e-10, 1 - 1e-10)
    a = x ** (alpha - 1.0) * (1.0 - x) ** (beta - 1.0)
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_fourier_smooth(n, rng, n_freqs=4):
    """Random smooth function from few Fourier modes (ensures non-negative)."""
    a = np.ones(n) / n
    t = np.arange(n, dtype=np.float64)
    for _ in range(n_freqs):
        freq = float(rng.integers(1, 3))
        phase = float(rng.uniform(0, 2 * np.pi))
        amp = float(rng.uniform(0.15, 0.35))
        a += amp * np.cos(2 * np.pi * freq * t / n + phase)
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Hybrid subgradient + LP search minimizing 2*n*max(conv(a,a))/sum(a)^2.
    
    Strategy:
    1. Try to use height_sequence_1 as a starting point
    2. Generate diverse initializations across lengths and shapes
    3. Quick subgradient pre-optimization for each
    4. Extended LP refinement on the best candidates
    5. Final local search polishing
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + max(100.0, budget_s - 40.0)

    best_score = float('inf')
    best_seq = None

    # Try to load existing best sequence
    try:
        hs1 = height_sequence_1
        if hs1 is not None and len(hs1) > 0:
            hs1_arr = np.array(hs1, dtype=np.float64)
            hs1_arr = np.maximum(0, hs1_arr)
            s = hs1_arr.sum()
            if s > 0.01:
                hs1_score = score_sequence(hs1_arr)
                print(f"Loaded height_sequence_1: n={len(hs1_arr)}, score={hs1_score:.6f}")
                best_seq = hs1_arr.copy()
                best_score = hs1_score
    except (NameError, TypeError):
        pass

    # Phase 1: Quick evaluation of multiple initializations
    print("\n--- Phase 1: Initialize and quick subgradient ---")
    candidates = []  # (score, sequence, n, label)

    init_configs = [
        (500, 'cos_p1', 1.0),
        (500, 'cos_p2', 2.0),
        (500, 'cos_p05', 0.5),
        (500, 'gauss_02', 0.2),
        (500, 'gauss_03', 0.3),
        (500, 'beta_33', 3.0),
        (1000, 'cos_p1', 1.0),
        (1000, 'cos_p2', 2.0),
        (1000, 'gauss_02', 0.2),
        (2000, 'cos_p1', 1.0),
        (2000, 'gauss_025', 0.25),
    ]

    for n, init_type, param in init_configs:
        remaining = deadline - time.monotonic()
        if remaining < 30.0:
            break

        if init_type == 'cos_p1':
            a = init_raised_cosine_p(n, p=1.0)
        elif init_type == 'cos_p2':
            a = init_raised_cosine_p(n, p=2.0)
        elif init_type == 'cos_p05':
            a = init_raised_cosine_p(n, p=0.5)
        elif init_type == 'gauss_02':
            a = init_gaussian(n, sigma_frac=0.2)
        elif init_type == 'gauss_03':
            a = init_gaussian(n, sigma_frac=0.3)
        elif init_type == 'gauss_025':
            a = init_gaussian(n, sigma_frac=0.25)
        elif init_type == 'beta_33':
            a = init_beta_kernel(n, alpha=3.0, beta=3.0)
        else:
            a = init_raised_cosine_p(n, p=1.0)

        a = np.maximum(0.0, a.astype(np.float64))
        s = a.sum()
        if s < 1e-15:
            continue
        a = a / s
        sc = score_sequence(a)
        candidates.append((sc, a.copy(), n, f"n={n}_{init_type}"))
        print(f"  {init_type} n={n}: score={sc:.6f}")

    # Sort by score and pick top candidates for refinement
    candidates.sort(key=lambda x: x[0])
    print(f"\nTop initializations: {[c[3] for c in candidates[:3]]}")

    # Add existing best as a candidate for refinement
    if best_seq is not None:
        candidates.insert(0, (best_score, best_seq.copy(), len(best_seq), "existing"))

    # Phase 2: Refine top candidates with SG + LP
    print("\n--- Phase 2: Subgradient + LP refinement ---")
    top_k = min(4, len(candidates))
    # Allocate time proportionally
    time_per_candidate = (deadline - time.monotonic()) / (top_k + 1)  # +1 for final polish
    time_per_candidate = max(30.0, min(300.0, time_per_candidate))

    for idx in range(top_k):
        remaining = deadline - time.monotonic()
        if remaining < 30.0:
            break

        sc_init, a_init, n, label = candidates[idx]
        alloc = min(time_per_candidate, remaining * 0.7)
        sg_time = min(15.0, alloc * 0.15)
        lp_time = alloc - sg_time - 5.0

        if lp_time < 10.0:
            continue

        print(f"\n  Refining {label} (init={sc_init:.6f}, n={n}, alloc={alloc:.0f}s)")

        # Quick subgradient pre-step
        a_sg, sg_score = subgradient_optimize(a_init, time_limit=sg_time, verbose=False)
        if sg_score < sc_init:
            a_work = a_sg
            print(f"    SG improved: {sc_init:.6f} -> {sg_score:.6f}")
        else:
            a_work = a_init

        # LP refinement
        a_lp, lp_score = lp_refine(a_work, time_limit=lp_time, verbose=True)
        final_sc = min(lp_score, sg_score, sc_init)
        if lp_score < sg_score and lp_score < sc_init:
            a_final = a_lp
        elif sg_score < sc_init:
            a_final = a_sg
        else:
            a_final = a_init

        if final_sc < best_score:
            best_score = final_sc
            best_seq = a_final.copy()
            print(f"  *** NEW BEST: {label} -> score={best_score:.6f} (t={time.monotonic()-total_start:.0f}s)")

    # Phase 3: Final polishing on the best sequence
    print("\n--- Phase 3: Final polishing ---")
    if best_seq is not None:
        # Extended LP on best
        remaining = deadline - time.monotonic()
        if remaining > 30.0:
            lp_final_time = min(120.0, remaining * 0.4)
            print(f"  Final LP ({lp_final_time:.0f}s):")
            a_lp_f, lp_f_score = lp_refine(best_seq, time_limit=lp_final_time, verbose=True)
            if lp_f_score < best_score:
                best_score = lp_f_score
                best_seq = a_lp_f.copy()
                print(f"  *** After final LP: {best_score:.6f}")

        # Local search
        remaining = deadline - time.monotonic()
        if remaining > 20.0:
            ls_time = min(60.0, remaining * 0.4)
            print(f"  Local search ({ls_time:.0f}s):")
            a_ls, ls_score = local_search_refine(best_seq, time_limit=ls_time, rng=rng, verbose=True)
            if ls_score < best_score:
                best_score = ls_score
                best_seq = a_ls.copy()
                print(f"  *** After LS: {best_score:.6f}")

        # One more short LP
        remaining = deadline - time.monotonic()
        if remaining > 15.0:
            lp2_time = min(60.0, remaining * 0.5)
            print(f"  Additional LP ({lp2_time:.0f}s):")
            a_lp2, lp2_score = lp_refine(best_seq, time_limit=lp2_time, verbose=True)
            if lp2_score < best_score:
                best_score = lp2_score
                best_seq = a_lp2.copy()
                print(f"  *** After extra LP: {best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = init_raised_cosine_p(500, p=1.0)
        best_score = score_sequence(best_seq)
        print(f"  Fallback: score={best_score:.6f}")

    print(f"\n{'='*60}")
    print(f"Final result: score={best_score:.6f}, n={len(best_seq)}")
    verify_score = evaluate_sequence([float(x) for x in best_seq])
    print(f"Verified score: {verify_score:.6f}")
    print(f"Total time: {time.monotonic()-total_start:.1f}s")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
