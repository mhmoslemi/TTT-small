import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond].astype(int))
    theta = cssv[i - 1] / i
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

def compute_objective_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def softmax_weights(conv, T):
    """Compute softmax weights over convolution values."""
    if conv.max() < 1e-15:
        return np.ones(len(conv)) / len(conv)
    scaled = T * conv / conv.max()
    scaled -= scaled.max()
    w = np.exp(scaled)
    s = w.sum()
    if s < 1e-15:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= s
    return w

def compute_subgradient(a, conv, T):
    """Compute the subgradient of the smooth max approximation."""
    n = len(a)
    w = softmax_weights(conv, T)
    sg = np.zeros(n)
    thresh = w.max() * 0.001
    active = np.where(w > thresh)[0]
    for k_int in active:
        k = int(k_int)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            sg[j_start:j_end + 1] += w[k] * a[k - j_end:k - j_start + 1][::-1]
    return sg

            g /= g.sum()
            inits[f'trunc_gauss_{sf}'] = g
    
    # Parent-derived
    if parent is not None:
        if parent.size >= n:
            pp = parent[:n].copy()
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent'] = pp
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0)
