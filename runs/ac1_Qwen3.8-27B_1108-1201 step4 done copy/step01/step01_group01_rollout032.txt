"""
FFT-accelerated smooth optimization for minimizing peak autocorrelation
of non-negative sequences. Uses log-sum-exp smoothing, momentum, and
simulated annealing to escape local minima. Multiple diverse starting
points with adaptive time allocation.
"""

import numpy as np
import time
from scipy.fft import fft, ifft


def _fft_size(n):
    """Get FFT size for convolution of two length-n sequences."""
    size = 2 * n - 1
    # Round up to next "good" FFT size (power of 2 or 2^a * 3^b * 5^c)
    # For simplicity, use next power of 2
    return 1 << (size - 1).bit_length()


def _autocorr_fft(a, L=None):
    """Compute full autocorrelation a*a using FFT."""
    n = len(a)
    if L is None:
        L = _fft_size(n)
    A = fft(a, L)
    b = ifft(A * A).real
    return b[:2*n-1]


def _softmax(x, tau=1.0):
    """Stable softmax with temperature."""
    x = np.asarray(x)
    x_max = np.max(x)
    e = np.exp((x - x_max) * tau)
    return e / np.sum(e)


def _logsumexp_approx(x, tau=1.0):
    """Smooth approximation to max(x) using log-sum-exp."""
    x = np.asarray(x)
    x_max = np.max(x)
    return x_max + np.log(np.sum(np.exp((x - x_max) * tau))) / tau


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_score(a):
    """Compute 2*n*max(convolve(a,a)) for normalized sequence."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-15:
        return np.inf
    b = _autocorr_fft(a)
    return 2.0 * n * np.max(b) / (S * S)


def _make_two_level(n, eps):
    """Anti-symmetric two-level sequence."""
    a = np.ones(n)
    a[:n // 2] = 1.0 + eps
    a[n // 2:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    return a / S


def _make_smooth_anti(n, eps, freq=1):
    """Smooth anti-symmetric: 1 + eps*cos(2*pi*freq*i/n)."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(2*np.pi*freq*i/n)
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    return a / S


def _make_four_level(n, eps):
    """Four-level anti-symmetric."""
    a = np.ones(n)
    q = n // 4
    a[:q] = 1.0 + eps
    a[q:2*q] = 1.0 + eps/3.0
    a[2*q:3*q] = 1.0 - eps/3.0
    a[3*q:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    return a / S


def _smooth_optimize(a, deadline, rng, tau_init=100.0, tau_end=500.0,
                     eta_init=None, momentum=0.9, L=None):
    """
    Smooth gradient descent with log-sum-exp objective, momentum,
    and adaptive learning rate.
    """
    n = len(a)
    if L is None:
        L = _fft_size(n)
    
    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    else:
        a = np.ones(n) / n
    
    if eta_init is None:
        eta_init = 1.0 / np.sqrt(n)
    
    best = a.copy()
    best_score = _compute_score(best)
    
    v = np.zeros(n)  # momentum
    t = 0
    tau = tau_init
    
    # Adaptive parameters
    no_improve = 0
    reheat_every = 500
    
    while time.monotonic() < deadline:
        t += 1
        
        # Compute autocorrelation
        b = _autocorr_fft(a, L)
        
        # Compute scores
        cur_score = 2.0 * n * np.max(b)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Reheat: increase tau (smoother) and add noise
        if no_improve >= reheat_every:
            tau = max(tau_init, tau * 0.9)  # actually make it smoother (larger tau = sharper, wait no)
            # tau larger = more like max (sharper), tau smaller = smoother
            # For reheat, we want smoother to escape: decrease tau
            tau = max(10.0, tau * 0.7)
            # Add noise
            noise = rng.normal(0, 0.01/np.sqrt(n), n)
            a = a + noise
            a = _project_simplex(a, 1.0)
            no_improve = 0
            v = np.zeros(n)
        
        # Adaptive tau: start sharp (large tau), get smoother over time? Or vice versa?
        # Actually, we want to always track the max, so keep tau large enough.
        # Let's use a fixed large tau for the gradient, but track the exact max for scoring.
        
        # Compute smooth gradient using softmax weights
        # w_k = exp(tau * b_k) / sum(exp(tau * b_j))
        b_max = np.max(b)
        log_w = tau * (b - b_max)
        log_w -= np.max(log_w)  # numerical stability
        w = np.exp(log_w)
        w /= np.sum(w)
        
        # Gradient: grad_i = 2 * sum_k w_k * a_{k-i}
        # = 2 * ifft(fft(a, L) * conj(fft(w, L))).real[:n]
        A = fft(a, L)
        W = fft(w, L)
        grad = 2.0 * ifft(A * np.conj(W)).real[:n]
        
        # Center gradient (for sum constraint)
        grad -= np.mean(grad)
        
        # Momentum update
        v = momentum * v + (1.0 - momentum) * grad
        eta = eta_init / (1.0 + 0.01 * t)
        a = a - eta * v
        
        # Project
        a = _project_simplex(a, 1.0)
        if np.sum(a) < 1e-10:
            a = best.copy()
            break
        
        # Early termination if converged
        if t > 1000 and no_improve > reheat_every * 3:
            break
    
    return best, best_score


def _spectral_flatten(a, n_iters=100, blend=0.3, L=None):
    """
    GS-style spectral flattening. Iteratively:
    1. Compute DFT of a
    2. Reduce magnitude peaks (partial flattening)
    3. Inverse DFT
    4. Project onto simplex
    """
    n = len(a)
    if L is None:
        L = _fft_size(n)
    
    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    
    for _ in range(n_iters):
        A = fft(a, L)
        mag = np.abs(A)
        mag_max = np.max(mag)
        if mag_max < 1e-15:
            break
        # Partial flattening: mag_new = mag^(1-alpha) * mag_max^alpha
        # This reduces peaks while maintaining some structure
        alpha = blend
        # A_new = A / mag^(1-alpha) * mag_max^(-alpha) ... 
        # Actually: want |A_new| = mag^alpha * (1/mag)^(1-alpha) = mag^alpha / mag^(1-alpha)
        # Simpler: A_new = A * (mag_max / mag)^alpha
        # This brings all magnitudes closer to mag_max
        # Wait, that increases small values. Let me think...
        # We want to reduce the dynamic range of |A|
        # A_new = A * (mean_mag / mag)^beta for beta in (0,1)
        mean_mag = np.mean(mag)
        ratio = np.ones_like(mag)
        nonzero = mag > 1e-15
        ratio[nonzero] = (mean_mag / mag[nonzero]) ** alpha
        A_new = A * ratio
        a_new = ifft(A_new, L).real[:n]
        a_new = _project_simplex(a_new, 1.0)
        a = a_new
    
    return a

"""
FFT-accelerated smooth optimization for minimizing peak autocorrelation of
non-negative sequences. Uses log-sum-exp smoothed objective with momentum,
spectral flattening (GS algorithm), and simulated annealing. Multiple diverse
starting points with adaptive time allocation and periodic reheating.
"""

import numpy as np
import time
from scipy.fft import fft, ifft


def _next_power_of_2(n):
    """Return next power of 2 >= n."""
    return 1 << (n - 1).bit_length()


def _fft_size_conv(n):
    """FFT size for autocorrelation of length-n sequence."""
    return _next_power_of_2(2 * n - 1)


def _fft_size_xcorr(n):
    """FFT size for cross-correlation of length-(2n-1) and length-n sequences."""
    return _next_power_of_2(3 * n - 2)


def _autocorr(a, L):
    """Full autocorrelation a*a via FFT, length 2n-1."""
    A = fft(a, L)
    return ifft(A * A).real[:2 * len(a) - 1]


def _cross_corr_wa(w, a, L):
    """
    Compute (w cross-correlate a)[i] = sum_k w[k+i] * a[k] for i=0..n-1.
    Uses ifft(fft(w) * conj(fft(a))).
    """
    W = fft(w, L)
    A = fft(a, L)
    return ifft(W * np.conj(A)).real[:len(a)]


def _project_simplex(v, target=1.0):
    """Project vector onto {x >= 0, sum x = target} using sort method."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _score(a, L):
    """Compute 2*n*max(a*a) for sum-normalized sequence."""
    n = len(a)
    b = _autocorr(a, L)
    return 2.0 * n * np.max(b)


def _make_two_level(n, eps):
    """Anti-symmetric two-level, normalized to sum=1."""
    a = np.ones(n)
    a[:n // 2] = 1.0 + eps
    a[n // 2:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    return a / np.sum(a)


def _make_smooth_anti(n, eps, freq=1):
    """Smooth anti-symmetric: 1 + eps*cos(2*pi*freq*i/n), normalized."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * i / n)
    a = np.maximum(a, 0.0)
    return a / np.sum(a)


def _make_four_level(n, eps):
    """Four-level anti-symmetric, normalized."""
    a = np.ones(n)
    q = n // 4
    a[:q] = 1.0 + eps
    a[q:2 * q] = 1.0 + eps / 3.0
    a[2 * q:3 * q] = 1.0 - eps / 3.0
    a[3 * q:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    return a / np.sum(a)


def _make_triangular_anti(n, eps):
    """Triangular anti-symmetric: peaks at edges, dip at center."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = 1.0 + eps * np.abs(i - center) / center
    return a / np.sum(a)


def _spectral_flatten(a, n_iters, blend, L):
    """
    GS-style spectral flattening: reduce dynamic range of |DFT(a)|,
    project back to simplex.
    """
    n = len(a)
    a = a.copy()
    for _ in range(n_iters):
        A = fft(a, L)
        mag = np.abs(A)
        mean_mag = np.mean(mag)
        if mean_mag < 1e-15:
            break
        ratio = np.ones_like(mag)
        nz = mag > 1e-15
        ratio[nz] = np.power(mean_mag / mag[nz], blend)
        A_new = A * ratio
        a_new = ifft(A_new, L).real[:n]
        a = _project_simplex(a_new, 1.0)
    return a


def _smooth_descend(a, deadline, rng, L_aut, L_xc, tau=500.0,
                    eta_init=None, momentum=0.9, top_m=30):
    """
    Smooth gradient descent using log-sum-exp / top-M peak objective.
    Uses FFT-accelerated gradient computation.
    """
    n = len(a)
    a = a.copy()
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    else:
        a = np.ones(n) / n

    if eta_init is None:
        eta_init = 3.0 / np.sqrt(n)

    best = a.copy()
    best_s = _score(best, L_aut)
    v = np.zeros(n)
    t = 0
    no_improve = 0
    reheat_count = 0

    while time.monotonic() < deadline:
        t += 1
        b = _autocorr(a, L_aut)
        cur_s = 2.0 * n * np.max(b)

        if cur_s < best_s - 1e-13:
            best_s = cur_s
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Reheat: perturb and reset momentum
        if no_improve >= 800:
            scale = 0.05 / np.sqrt(n) * (1 + reheat_count * 0.5)
            scale = min(scale, 0.5 / np.sqrt(n))
            a = a + rng.normal(0, scale, n)
            a = _project_simplex(a, 1.0)
            v = np.zeros(n)
            no_improve = 0
            reheat_count += 1
            if reheat_count > 20:
                break

        # Compute weights: top-M softmax
        if top_m >= len(b):
            idx = np.arange(len(b))
            b_sel = b
        else:
            idx = np.argpartition(b, -top_m)[-top_m:]
            b_sel = b[idx]

        b_max = np.max(b_sel)
        log_w = tau * (b_sel - b_max)
        log_w -= np.max(log_w)
        w_sel = np.exp(log_w)
        w_sel /= np.sum(w_sel)

        # Build full weight vector
        w = np.zeros(len(b))
        w[idx] = w_sel

        # Gradient via FFT cross-correlation
        grad = 2.0 * _cross_corr_wa(w, a, L_xc)

        # Center for sum constraint
        grad -= np.mean(grad)

        # Adaptive learning rate
        eta = eta_init / (1.0 + 0.005 * t)

        # Momentum update
        v = momentum * v + (1.0 - momentum) * grad
        a = a - eta * v

        # Project
        a = _project_simplex(a, 1.0)
        if np.sum(a) < 1e-10:
            a = best.copy()
            break

        # Convergence check
        if t > 5000 and no_improve > 3000:
            break

    return best, best_s


def _smooth_descend_multi_tau(a, deadline, rng, L_aut, L_xc,
                              taus=(200, 500, 1000, 2000),
                              eta_init=None, momentum=0.9, top_m=30):
    """
    Run smooth descent with increasing tau (sharper peak tracking) in stages.
    """
    n = len(a)
    a = a.copy()
    S = np.sum(a)
    if S > 1e-10:
        a /= S

    if eta_init is None:
        eta_init = 3.0 / np.sqrt(n)

    best = a.copy()
    best_s = _score(best, L_aut)

    n_taus = len(taus)
    per_tau_deadline = (deadline - time.monotonic()) / n_taus

    for tau in taus:
        if time.monotonic() >= deadline:
            break
        tau_deadline = min(time.monotonic() + per_tau_deadline, deadline)
        a_cur, s_cur = _smooth_descend(
            a, tau_deadline, rng, L_aut, L_xc,
            tau=tau, eta_init=eta_init, momentum=momentum, top_m=top_m
        )
        if s_cur < best_s:
            best = a_cur.copy()
            best_s = s_cur
        a = a_cur

    return best, best_s
