def project_simplex(v, S):
    """Project onto the simplex (that is, non-negative and sum to S)."""
    n = len(v)
    if n == 0:
        return np.array([])
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if valid.any():
        rho = np.max(np.where(valid)[0]) + 1
        theta = (cumsum[rho-1] - S) / rho
    else:
        idx_min = np.min(idx)
        theta = (cumsum[idx_min-1] - S) / idx_min
        if np.isinf(theta):
            theta = 0.0
