def compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)

def softmax_grad_fft(a, tau):
    """Compute softmax-weighted gradient using FFT for efficiency."""
    n = len(a)
    c = np.convolve(a, a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(c))
        w[int(c.argmax())] = 1.0
    else:
        w /= w_sum
    
    # Compute grad[j] = 2 * sum_m a[m] * w[m+j] for j=0..n-1
    # This is the cross-correlation (w ⊛ a)[j] for j=0..n-1
    # Using FFT: correlate(w, a) = ifft(conj(FFT(w)) * FFT(a))
    N = 1
    while N < 3 * n:
        N *= 2
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    corr = np.fft.irfft(np.conj(W) * A, N)
    grad = 2.0 * corr[:n]
    
    # Project out the mean (for fixed-sum constraint)
    grad -= grad.mean()
    
    return grad

    """One Adam step with projection onto non-negative, fixed-sum simplex."""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad ** 2
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize(a_new)
    return a_new, m, v

"""
Optimizes non-negative sequences to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses Adam optimizer with FFT-accelerated softmax gradients, top-k gradient variants,
LP-based refinement, random perturbations, and multiple initialization strategies.
Leverages warm start from provided sequence when available.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=None):
    """Normalize sequence to have a specific sum (default sqrt(2n))."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def convolve_self(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    return c[:2 * n - 1]


def softmax_weighted_gradient(a, tau):
    """
    Compute gradient of softmax approximation to max(conv(a,a)).
    Returns gradient projected onto the fixed-sum hyperplane.
    """
    n = len(a)
    c = convolve_self(a)
    c_max = c.max()
    c_shifted = c - c_max
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum

    # grad[j] = 2 * sum_m a[m] * w[m+j] (cross-correlation of w and a)
    N = 1
    while N < 3 * n:
        N *= 2
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    corr = np.fft.irfft(np.conj(W) * A, N)
    grad = 2.0 * corr[:n]

    grad -= grad.mean()
    return grad


def topk_weighted_gradient(a, k=3):
    """Compute gradient using uniform weights on top-k peaks."""
    n = len(a)
    c = convolve_self(a)
    top_k = np.argsort(c)[-k:]
    w = np.zeros(len(c))
    for idx in top_k:
        w[idx] = 1.0 / k

    N = 1
    while N < 3 * n:
        N *= 2
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    corr = np.fft.irfft(np.conj(W) * A, N)
    grad = 2.0 * corr[:n]

    grad -= grad.mean()
    return grad


def hard_max_gradient(a):
    """Compute gradient at the single argmax of conv(a,a)."""
    n = len(a)
    c = convolve_self(a)
    p = int(c.argmax())
    grad = np.zeros(n)
    lo = max(0, p - n + 1)
    hi = min(n - 1, p)
    for j in range(lo, hi + 1):
        grad[j] = 2.0 * a[p - j]
    grad -= grad.mean()
    return grad


def adam_step(a, grad, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """One Adam update with non-negativity clip and renormalization."""
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize(a_new)
    return a_new, m, v


def lp_solve(f, M):
    """Solve LP: max sum(g) s.t. convolve(f,g) <= M elementwise, g >= 0."""
    n = len(f)
    if n > 1500:
        return None

    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[m, j] = f[m - j]

    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def run_adam_phase(a0, deadline, max_iters=20000, tau_start=5.0, tau_end=100.0,
                   lr_start=0.1, lr_end=0.003, use_topk=False, topk_k=3):
    """Adam optimization phase with softmax or topk gradients."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    a = normalize(a)

    best = a.copy()
    best_score = compute_obj(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stall_counter = 0
    stall_best = best_score

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        t += 1
        progress = min(1.0, it / max(max_iters - 1, 1))

        # Adaptive temperature: start moderate, increase to focus on peak
        tau = tau_start + (tau_end - tau_start) * progress
        # Learning rate: cosine decay
        lr = lr_start * 0.5 * (1.0 + np.cos(np.pi * progress))
        lr = max(lr, lr_end)

        # Choose gradient method
        if use_topk and t % 3 == 0:
            grad = topk_weighted_gradient(a, topk_k)
        elif t % 5 == 0:
            grad = hard_max_gradient(a)
        else:
            grad = softmax_weighted_gradient(a, tau)

        a, m, v = adam_step(a, grad, m, v, t, lr)

        # Check for improvement every 500 steps
        if (it + 1) % 500 == 0:
            s = compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()
                stall_counter = 0
                stall_best = best_score
            else:
                stall_counter += 1
                # If stalled for a while, add small perturbation
                if stall_counter >= 4:
                    rng_local = np.random.default_rng(t)
                    a = a + 0.02 * rng_local.standard_normal(n)
                    a = np.maximum(a, 0.0)
                    a = normalize(a)
                    m *= 0.5  # reduce momentum to escape
                    stall_counter = 0

    s = compute_obj(a)
    if s < best_score:
        best_score = s
        best = a.copy()

    return best, best_score


def run_lp_phase(a0, deadline, max_iters=60):
    """LP-based refinement phase."""
    n = len(a0)
    if n > 1024:
        return a0.copy(), compute_obj(a0)

    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    a = normalize(a)

    best = a.copy()
    best_score = compute_obj(a)
    rng_local = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = convolve_self(a)
        M = float(c.max())

        g = lp_solve(a, M)
        if g is None:
            break

        improved = False
        for t_frac in [0.5, 0.3, 0.7, 0.2, 0.9, 0.4, 0.15, 0.6]:
            a_new = (1.0 - t_frac) * a + t_frac * g
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new)
            s_new = compute_obj(a_new)
            if s_new < best_score:
                a = a_new
                best = a.copy()
                best_score = s_new
                improved = True
                break

        if not improved:
            # Perturb to escape
            a = a + 0.01 * rng_local.standard_normal(n)
            a = np.maximum(a, 0.0)
            a = normalize(a)

    return best, best_score


def run_adam_lp_combined(a0, deadline, max_adam_iters=15000, max_lp_iters=50):
    """Run Adam phase followed by LP refinement."""
    remaining = deadline - time.monotonic()
    if remaining < 5.0:
        return a0.copy(), compute_obj(a0)

    # Phase 1: Adam (65% of time)
    adam_deadline = time.monotonic() + remaining * 0.65
    a1, s1 = run_adam_phase(a0, adam_deadline, max_iters=max_adam_iters)

    # Phase 2: LP (remaining time)
    a2, s2 = run_lp_phase(a1, deadline, max_iters=max_lp_iters)

    if s2 < s1:
        return a2, s2
    return a1, s1

    alpha = (n - 1) / 2.0
    if alpha > 0:
        x = np.clip((i - alpha) / alpha, -1.0, 1.0)
        starts.append(np.maximum(np.exp(-8.6 * np.sqrt(np.maximum(1.0 - x * x, 0.0))), 0.01))

    # Chirps
    for freq in [2, 3, 5, 7]:
        starts.append(np.maximum(0.5 + 0.5 * np.cos(2.0 * np.pi * i * freq / n), 0.01))

    # Random
    starts.append(rng.random(n) * 0.5 + 0.25)
    starts.append(rng.random(n) * 0.8 + 0.1)

    # Bimodal
    center = n / 2.0
    starts.append(np.maximum(0.2 + 0.8 * np.abs((i - center) / max(center, 1.0)), 0.01))

    # Gaussian
    starts.append(np.maximum(np.exp(-0.5 * ((i - center) / max(n * 0.25, 1.0)) ** 2), 0.01))

    # Two peaks
    starts.append(np.maximum(
        np.exp(-0.5 * ((i - n * 0.25) / (n * 0.1)) ** 2) +
        np.exp(-0.5 * ((i - n * 0.75) / (n * 0.1)) ** 2), 0.01))

    return starts


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: Adam+softmax gradient, top-k gradient, LP refinement,
    random restarts, and multiple initializations including warm start.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 12.0)

    best_seq = None
    best_score = float('inf')

    # Collect warm start
    warm_seq = None
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 8 and np.all(np.isfinite(gs)):
            warm_seq = gs.copy()
            print(f"Warm start available: length {len(gs)}, "
                  f"obj={compute_obj(gs):.6f}")
    except NameError:
        print("No warm start available.")

    # Build list of (sequence, label) to try, ordered by priority
    tasks = []

    if warm_seq is not None:
        tasks.append((warm_seq.copy(), "warm_full", "combined"))
        for L in [128, 192, 256, 384, 512, 640, 768, 1024]:
            if L <= len(warm_seq):
                tasks.append((warm_seq[:L].copy(), f"warm_{L}", "combined"))
            else:
                reps = int(np.ceil(L / len(warm_seq)))
                padded = np.tile(warm_seq, reps)[:L]
                tasks.append((padded, f"warmpad_{L}", "combined"))

    # Generated starts for various lengths
    for n in [256, 512, 768, 1024]:
        for s in generate_diverse_starts(rng, n):
            tasks.append((s, f"gen_{n}", "combined"))

    # Also try topk variant on some
    for n in [256, 512]:
        tasks.append((rng.random(n) * 0.5 + 0.25, f"topk_{n}", "topk"))

    # Limits per-task time budget
    total_time = budget_s - 15.0
    n_tasks = len(tasks)
    # Give warm starts more time
    base_time = total_time / max(n_tasks, 1)
    warm_time = base_time * 3.0
    gen_time = base_time

    print(f"\nTotal tasks: {n_tasks}, budget: {budget_s}s")
    print(f"Warm time/task: {warm_time:.1f}s, Gen time/task: {gen_time:.1f}s\n")

    task_idx = 0
    while task_idx < len(tasks) and time.monotonic() < deadline - 20.0:
        a0, label, method = tasks[task_idx]
        n = len(a0)

        # Determine deadline for this task
        if "warm" in label:
            task_budget = max(30.0, min(warm_time, deadline - time.monotonic() - 15.0))
        else:
            task_budget = max(15.0, min(gen_time, deadline - time.monotonic() - 15.0))

        task_deadline = time.monotonic() + task_budget

        t0 = time.monotonic()
        if method == "combined":
            a_opt, score = run_adam_lp_combined(a0, task_deadline,
                                                 max_adam_iters=20000,
                                                 max_lp_iters=40)
        else:
            a_opt, score = run_adam_topk(a0, task_deadline, max_iters=15000)

        elapsed = time.monotonic() - t0

        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  *** NEW BEST: {score:.6f} (n={n}, {label}, {elapsed:.1f}s) ***")
        else:
            print(f"  {label}: n={n}, score={score:.6f} ({elapsed:.1f}s), "
                  f"best={best_score:.6f}")

        task_idx += 1

    # ===== Final refinement phases =====
    if best_seq is not None and time.monotonic() < deadline - 10.0:
        print(f"\n=== Final refinement (score={best_score:.6f}) ===")

        # Multiple Adam+LP passes on the best
        for pass_i in range(8):
            if time.monotonic() > deadline - 5.0:
                break
            a_opt, score = run_adam_lp_combined(best_seq, deadline - 5.0,
                                                 max_adam_iters=8000,
                                                 max_lp_iters=30)
            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"  Refine {pass_i}: {best_score:.6f}")
            else:
                break

        # Try Adam-only refinement with different hyperparams
        if time.monotonic() < deadline - 5.0:
            a_opt, score = run_adam_phase(best_seq, deadline - 5.0,
                                           max_iters=10000,
                                           tau_start=2.0, tau_end=40.0,
                                           lr_start=0.05, lr_end=0.001)
            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"  Adam-refine: {best_score:.6f}")

        # Try topk refinement
        if time.monotonic() < deadline - 5.0:
            a_opt, score = run_adam_phase(best_seq, deadline - 5.0,
                                           max_iters=8000,
                                           use_topk=True, topk_k=5,
                                           tau_start=3.0, tau_end=50.0,
                                           lr_start=0.06, lr_end=0.002)
            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"  Topk-refine: {best_score:.6f}")

    # Try truncation
    if best_seq is not None and time.monotonic() < deadline - 2.0:
        for L in [64, 96, 128, 160, 192, 256, 320, 384, 448, 512]:
            if L < len(best_seq) and L >= 16:
                truncated = best_seq[:L]
                s = compute_obj(truncated)
                if s < best_score:
                    best_score = s
                    best_seq = truncated.copy()
                    print(f"  Truncated to {L}: {best_score:.6f}")

    # Final validation
    if best_seq is not None:
        # Ensure all values are non-negative and finite
        best_seq = np.clip(best_seq, 0.0, 1000.0)
        final_score = compute_obj(best_seq)
        print(f"\nFinal result: n={len(best_seq)}, score={final_score:.6f}")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence confirms: {eval_score:.6f}")
        except Exception as e:
            print(f"Evaluation error: {e}")
        return [float(x) for x in best_seq]

    print("Falling back to uniform sequence of length 256.")
    return [1.0] * 256
