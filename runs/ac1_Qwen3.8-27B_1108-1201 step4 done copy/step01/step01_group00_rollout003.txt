    k = np.arange(2*n-1)[:, None]

    corr = fft_convolve(rev_a, w)
    g = 2.0 * corr[n - 1: 2 * n - 1]
    return value, g, c


def build_conv_matrix(f):
    """Build the convolution matrix A where A[k,j] = f[k-j] if valid."""
    n = len(f)
    k = np.arange(2 * n - 1)[:, None].astype(np.int64)
    j = np.arange(n)[None, :].astype(np.int64)
    i = k - j
    A = np.where((i >= 0) & (i < n), f[np.clip(i, 0, n - 1)], 0.0)
    # Zero out invalid entries (where i was out of range)
    A[~((i >= 0) & (i < n))] = 0.0
    return A


def lp_solve(f, M):
    """
    Solve LP: maximize sum(b) subject to b>=0 and (f*b)[k] <= M for all k.
    Returns the optimal b vector, or None if infeasible.
    """
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.success:
        return res.x
    return None


def lp_refine(f, time_limit, rng, blend_factor=0.25):
    """
    Iterative LP-based refinement:
    1. Solve LP with current f and M = ||f*f||_inf
    2. Normalize solution g to match sum(f)
    3. Blend: f <- (1-t)*f + t*g
    4. Repeat until time runs out or convergence
    """
    n = len(f)
    current = f.astype(np.float64).copy()
    best = current.copy()
    S = np.sum(current)
    conv = fft_convolve(current, current)
    M = np.max(conv)
    best_M = M
    t0 = time.monotonic()
    stagnation = 0

    print(f"    LP refine start: n={n}, M={M:.6f}, score={2.0*n*M/(S*S):.6f}")

    for it in range(2000):
        if time.monotonic() - t0 > time_limit:
            break

        g = lp_solve(current, M)
        if g is None:
            break

        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break

        # Normalize g to have same sum as current
        g_norm = g * (S / sum_g)

        # Check the peak of g_norm * g_norm
        conv_g = fft_convolve(g_norm, g_norm)
        M_g = np.max(conv_g)

        if M_g < best_M:
            best_M = M_g
            best = g_norm.copy()
            stagnation = 0
        else:
            stagnation += 1

        # Blend
        t = blend_factor * (0.5 + 0.5 * np.exp(-it / 50.0))
        current = (1 - t) * current + t * g_norm

        # Update M for next iteration
        conv_cur = fft_convolve(current, current)
        M_new = np.max(conv_cur)

        if M_new > M * 1.01:
            # Bad step, revert
            current = best.copy()
            M = M_new
        else:
            M = min(M, M_new)

        if it % 20 == 0:
            score_cur = 2.0 * n * np.max(fft_convolve(current, current)) / (S * S)
            print(f"    LP iter={it}, M={M:.6f}, best_M={best_M:.6f}, score_cur={score_cur:.6f}, t={time.monotonic()-t0:.1f}s")

        if stagnation > 50:
            # Try a different blend factor or restart
            blend_factor = 0.1 + 0.3 * rng.random()
            stagnation = 0

    final_score = 2.0 * n * np.max(fft_convolve(best, best)) / (S * S)
    print(f"    LP refine done: best_M={best_M:.6f}, score={final_score:.6f}, time={time.monotonic()-t0:.1f}s")
    return best, local_score(best)


def optimize_annealed_softmax(n, theta_init, time_limit, rng):
    """
    Annealed softmax gradient descent with Adam optimizer.
    Returns (best_theta, best_score).
    """
    S = np.sqrt(2.0 * n)
    theta = theta_init.astype(np.float64).copy()
    momentum1 = np.zeros(n)
    momentum2 = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr_base = 4.0
    t0 = time.monotonic()
    step = 0
    best_theta = theta.copy()
    best_score = np.inf
    tau_start = 80.0 / n
    tau_end = 0.05 / n
    plateau = 0
    patience = 4000
    eval_interval = max(10, 500 // n)

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1e-6))
        # Use a power curve for slower early cooling
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.5)

        a = softmax_normalize(theta) * S
        value, g_a, c = grad_soft_max_conv(a, tau)
        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner / S)

        step += 1
        momentum1 = beta1 * momentum1 + (1 - beta1) * g_theta
        momentum2 = beta2 * momentum2 + (1 - beta2) * (g_theta * g_theta)
        m1_hat = momentum1 / (1.0 - beta1 ** step)
        m2_hat = momentum2 / (1.0 - beta2 ** step)
        lr = lr_base / np.sqrt(step + 1)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        if step % eval_interval == 0:
            a_check = softmax_normalize(theta) * S
            score = 2.0 * n * np.max(fft_convolve(a_check, a_check)) / (S * S)
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_theta = theta.copy()
                plateau = 0
            else:
                plateau += eval_interval

            if step % (eval_interval * 20) == 0:
                print(f"      step={step}, score={score:.6f}, best={best_score:.6f}, tau={tau:.2e}, t={time.monotonic()-t0:.1f}s")

        if plateau > patience:
            # Perturb and restart temperature partially
            theta = theta + rng.standard_normal(n) * (1.0 / np.sqrt(n))
            momentum1 = np.zeros(n)
            momentum2 = np.zeros(n)
            plateau = 0
            step = 0

    return best_theta, best_score


def make_chirp_init(n, alpha, rng):
    """Create a chirp-like initial sequence."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng, spread=0.3):
    """Create a random initial sequence centered around 1."""
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multi_tone_init(n, rng):
    """Create a multi-tone initial sequence."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    num_tones = rng.integers(3, 8)
    for _ in range(num_tones):
        k = rng.integers(1, n // 4 + 1)
        amp = rng.uniform(0.05, 0.3)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_windowed_init(n, rng):
    """Create a windowed initial sequence (raised cosine, Hann, etc)."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    window_type = rng.integers(0, 3)
    if window_type == 0:
        a = 0.5 * (1 + np.cos(2 * np.pi * t))  # Hann
    elif window_type == 1:
        a = 0.5 - 0.5 * np.cos(2 * np.pi * t)  # Inverted Hann
    else:
        a = np.sin(np.pi * t) ** 0.5  # Sinc-like
    a = a / np.mean(a)  # Normalize to mean 1
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_symmetric_init(n, rng, base_func):
    """Create a symmetric (palindromic) initial sequence."""
    half = n // 2
    a_half = base_func(half, rng)
    if n % 2 == 0:
        a = np.concatenate([a_half, a_half[::-1]])
    else:
        a = np.concatenate([a_half, [a_half[half - 1]], a_half[:half - 1][::-1]])
    return a[:n]


def local_search(a_init, time_limit, rng):
    """
    Enhanced simulated annealing with peak-targeted perturbations.
    """
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temperature = 0.002
    cooling = 0.9998

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()

        if r < 0.30:
            # Peak-targeted: reduce elements contributing to the peak
            conv = fft_convolve(current, current)
            # Find top 3 peaks
            peak_indices = np.argsort(conv)[-3:]
            for peak_idx in peak_indices:
                peak_idx = int(peak_idx)
                contribs = np.zeros(n)
                for i in range(n):
                    j = peak_idx - i
                    if 0 <= j < n:
                        contribs[i] = current[i] * current[j]
                # Reduce top contributors
                threshold = np.percentile(contribs, 70)
                reduce_mask = contribs > threshold
                candidate[reduce_mask] *= np.exp(rng.normal(-0.03, 0.02, size=np.sum(reduce_mask)))
                # Increase non-contributors
                inc_mask = ~reduce_mask
                if np.sum(inc_mask) > 0:
                    n_inc = min(5, np.sum(inc_mask))
                    inc_idx = np.where(inc_mask)[0]
                    chosen = rng.choice(inc_idx, size=n_inc, replace=False)
                    candidate[chosen] *= np.exp(rng.normal(0.03, 0.02, size=n_inc))
        elif r < 0.50:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.04, size=width))
        elif r < 0.65:
            # Multi-block
            for _ in range(4):
                width = max(1, int(rng.integers(1, max(2, n // 15))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.025, size=width))
        elif r < 0.80:
            # Partial symmetrization
            current_sym = 0.5 * (current + current[::-1])
            candidate = current_sym * np.exp(rng.normal(0.0, 0.015, size=n))
        elif r < 0.90:
            # Swap-based: move mass from high to low
            sorted_idx = np.argsort(current)
            n_move = max(1, n // 10)
            low_idx = sorted_idx[:n_move]
            high_idx = sorted_idx[-n_move:]
            amount = current[high_idx] * rng.uniform(0.05, 0.15, size=n_move)
            candidate[high_idx] -= amount
            candidate[low_idx] += amount
            candidate = np.clip(candidate, 0.01, 1000.0)
        else:
            # Small global perturbation
            candidate = current * np.exp(rng.normal(0.0, 0.015, size=n))

        candidate = np.clip(candidate, 0.01, 1000.0)
        new_score = local_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            current = candidate
            current_score = new_score

        temperature *= cooling

    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search:
    Phase 1: Annealed softmax gradient descent on diverse initializations
    Phase 2: LP-based refinement on top candidates
    Phase 3: Combined gradient + LP alternation
    Phase 4: Local search polish
    """
    t_global = time.monotonic()
    time_budget = max(10.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check for provided starting sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        try:
            start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
            s = local_score(start_seq)
            print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_seq = start_seq.copy()
        except Exception as e:
            print(f"Failed to load provided sequence: {e}")

    # ==================== Phase 1: Gradient-based optimization ====================
    print("\n" + "=" * 60)
    print("PHASE 1: Annealed softmax gradient descent")
    print("=" * 60)

    configs = [
        (256, 4, ['chirp', 'multi_tone', 'random', 'windowed']),
        (384, 3, ['chirp', 'multi_tone', 'windowed']),
        (512, 3, ['chirp', 'random', 'multi_tone']),
        (768, 2, ['chirp', 'multi_tone']),
        (1024, 2, ['chirp', 'windowed']),
    ]

    # Also try resampling the provided sequence if available
    if start_seq is not None and len(start_seq) > 10:
        resample_lengths = [384, 512, 768]
        for rl in resample_lengths:
            if time.monotonic() > deadline - 300:
                break
            x_old = np.linspace(0, 1, len(start_seq))
            x_new = np.linspace(0, 1, rl)
            a0 = np.interp(x_new, x_old, start_seq)
            a0 = np.clip(a0, 0.01, 1000.0)
            # Try optimizing this resampled version
            theta_init = np.log(a0 / np.sum(a0) + 1e-10)
            t_per = min(60.0, (deadline - time.monotonic()) / 4.0)
            if t_per > 5:
                theta_best, grad_score = optimize_annealed_softmax(rl, theta_init, t_per, rng)
                a_best = softmax_normalize(theta_best) * np.sqrt(2.0 * rl)
                actual_score = local_score(a_best)
                print(f"  Resampled n={rl}: score={actual_score:.6f}")
                if actual_score < best_score:
                    best_score = actual_score
                    best_seq = a_best.copy()
                    print(f"    *** New best (resampled): {best_score:.6f} ***")

    for n, n_inits, init_types in configs:
        if time.monotonic() > deadline - 180:
            break
        time_left = deadline - time.monotonic()
        time_per_init = min(80.0, time_left / (n_inits + 0.5))

        for init_i in range(n_inits):
            if time.monotonic() > deadline - 60:
                break
            init_type = init_types[min(init_i, len(init_types) - 1)]

            if init_type == 'chirp':
                alpha = 0.3 + 0.4 * rng.random()
                a0 = make_chirp_init(n, alpha, rng)
            elif init_type == 'multi_tone':
                a0 = make_multi_tone_init(n, rng)
            elif init_type == 'windowed':
                a0 = make_windowed_init(n, rng)
            else:
                a0 = make_random_init(n, rng, spread=0.2 + 0.4 * rng.random())

            a0 = np.clip(a0, 0.01, 1000.0)
            theta_init = np.log(a0 / np.sum(a0) + 1e-10)

            t_actual = min(time_per_init, deadline - time.monotonic() - 10)
            if t_actual < 5:
                continue

            theta_best, grad_score = optimize_annealed_softmax(n, theta_init, t_actual, rng)
            a_best = softmax_normalize(theta_best) * np.sqrt(2.0 * n)
            actual_score = local_score(a_best)
            print(f"  n={n}, init='{init_type}': score={actual_score:.6f} (time={time.monotonic()-t_global:.0f}s)")

            if actual_score < best_score:
                best_score = actual_score
                best_seq = a_best.copy()
                print(f"    *** New best: {best_score:.6f} ***")

    # Also try to optimize the provided sequence directly if it matches a length
    if start_seq is not None:
        n_prov = len(start_seq)
        if time.monotonic() < deadline - 60:
            theta_init = np.log(start_seq / np.sum(start_seq) + 1e-10)
            t_per = min(90.0, deadline - time.monotonic() - 30)
            if t_per > 10:
                print(f"\n  Optimizing provided sequence (n={n_prov})...")
                theta_best, grad_score = optimize_annealed_softmax(n_prov, theta_init, t_per, rng)
                a_best = softmax_normalize(theta_best) * np.sqrt(2.0 * n_prov)
                actual_score = local_score(a_best)
                print(f"  Provided optimized: score={actual_score:.6f}")
                if actual_score < best_score:
                    best_score = actual_score
                    best_seq = a_best.copy()
                    print(f"    *** New best: {best_score:.6f} ***")

    # ==================== Phase 2: LP-based refinement ====================
    print("\n" + "=" * 60)
    print("PHASE 2: LP-based refinement")
    print("=" * 60)

    if best_seq is not None:
        # Try LP refinement on the best sequence
        lp_time = min(600.0, max(30.0, (deadline - time.monotonic()) * 0.4))
        print(f"  Starting LP refinement, current best={best_score:.6f}, time={lp_time:.0f}s")

        lp_start = best_seq.astype(np.float64)
        lp_result, lp_score = lp_refine(lp_start, lp_time, rng, blend_factor=0.2)

        if lp_score < best_score:
            best_score = lp_score
            best_seq = lp_result.copy()
            print(f"  *** LP improved to: {best_score:.6f} ***")
        else:
            print(f"  LP best: {lp_score:.6f} (no improvement over {best_score:.6f})")

    # Try LP on other good candidates too
    if start_seq is not None and time.monotonic() < deadline - 120:
        lp_time2 = min(200.0, deadline - time.monotonic() - 60)
        if lp_time2 > 30:
            print(f"\n  LP refinement on provided sequence (n={len(start_seq)})...")
            lp2_result, lp2_score = lp_refine(start_seq, lp_time2, rng, blend_factor=0.15)
            if lp2_score < best_score:
                best_score = lp2_score
                best_seq = lp2_result.copy()
                print(f"  *** LP (provided) improved to: {best_score:.6f} ***")

    # ==================== Phase 3: Gradient + LP alternation ====================
    print("\n" + "=" * 60)
    print("PHASE 3: Alternating gradient + LP")
    print("=" * 60)

    if best_seq is not None and time.monotonic() < deadline - 60:
        a_cur = best_seq.astype(np.float64).copy()
        n_cur = len(a_cur)
        S = np.sum(a_cur)
        theta_cur = np.log(a_cur / S + 1e-10)
        alt_time = min(400.0, deadline - time.monotonic() - 30)

        # Alternate between gradient steps and LP steps
        while time.monotonic() < deadline - 30 and alt_time > 0:
            # Gradient phase: 30 seconds
            grad_time = min(30.0, alt_time)
            if grad_time > 5:
                theta_cur, _ = optimize_annealed_softmax(n_cur, theta_cur, grad_time, rng)
                a_cur = softmax_normalize(theta_cur) * S
                alt_time -= grad_time

            # LP phase: 30 seconds
            lp_time = min(30.0, alt_time)
            if lp_time > 5:
                a_cur, sc = lp_refine(a_cur, lp_time, rng, blend_factor=0.15)
                S = np.sum(a_cur)
                theta_cur = np.log(a_cur / S + 1e-10)
                alt_time -= lp_time

            cur_score = local_score(a_cur)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a_cur.copy()
                print(f"  Alt improved: {best_score:.6f}")

    # ==================== Phase 4: Local search polish ====================
    print("\n" + "=" * 60)
    print("PHASE 4: Local search polish")
    print("=" * 60)

    if best_seq is not None and time.monotonic() < deadline - 10:
        polish_time = max(5.0, deadline - time.monotonic() - 5)
        print(f"  Polishing, current best={best_score:.6f}, time={polish_time:.0f}s")
        polished, polish_score = local_search(best_seq, polish_time, rng)
        if polish_score < best_score:
            best_score = polish_score
            best_seq = polished.copy()
            print(f"  *** Polish improved to: {best_score:.6f} ***")
        else:
            print(f"  Polish best: {polish_score:.6f} (no improvement)")

    # Final verification
    print("\n" + "=" * 60)
    print("FINAL RESULT")
    print("=" * 60)

    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"Score: {final_score:.6f}, n={len(final_list)}")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]

    # Absolute fallback: uniform sequence
    a = np.ones(256)
    return [float(x) for x in a]
