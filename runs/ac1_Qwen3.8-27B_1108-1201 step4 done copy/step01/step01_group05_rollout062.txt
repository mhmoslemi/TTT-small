def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, float)
    n = len(v)
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n+1, dtype=float)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / (k + 1)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return out / s

def project_simplex(v):
    v = np.asarray(v, float)
    n = len(v)
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=float)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / (k + 1)

        return None
    return g / g.sum()


def optimize_lp(p0, rng, time_limit):
    """LP best-response iteration with blending."""
    p = project_simplex(p0.copy())
    best_p = p.copy()
    best_score = score(p)
    n = len(p)

    end = time.monotonic() + time_limit
    it = 0

    while time.monotonic() < end:
        it += 1
        g = lp_best_response(p, budget_s=min(20.0, end - time.monotonic() - 5))
        if g is None:
            break

        improved = False
        for t in [1.0, 0.75, 0.5, 0.3, 0.15, 0.05]:
            q = project_simplex((1 - t) * p + t * g)
            sc = score(q)
            if sc < best_score:
                best_score = sc
                best_p = q.copy()
                p = q.copy()
                improved = True
                break

        if not improved:
            p = project_simplex(p + rng.normal(0, 0.1 / n, n))

        if it % 3 == 0:
            print(f"    LP: it={it} best={best_score:.8f}", flush=True)

        if best_score < 1.505 and it > 50:
            break

    return best_p, best_score


def sparsify(p, rng, n_zeros):
    """Try zeroing out n_zeros entries and re-optimizing."""
    n = len(p)
    best_p = p.copy()
    best_score = score(p)

    # Try removing entries at various positions
    for _ in range(5):
        p_try = p.copy()
        if n_zeros >= n:
            continue
        # Choose positions to zero out (prefer low-value entries)
        vals = p_try.copy()
        # Bias toward edges and low values
        x = np.linspace(0, 1, n)
        edge_weight = x ** 2 + (1 - x) ** 2
        priority = vals * (1 + edge_weight)
        remove_idx = np.argsort(priority)[:n_zeros]
        p_try[remove_idx] = 0
        p_try = normalize(p_try)
        sc = score(p_try)
        if sc < best_score:
            best_score = sc
            best_p = p_try.copy()

    return best_p, best_score


def make_candidates(n, rng):
    """Generate diverse initial candidates of length n."""
    cands = []
    x = np.linspace(0, 1, n)
    xm = 2 * x - 1  # centered in [-1, 1]

    # Uniform
    cands.append(np.ones(n) / n)

    # Geometric / exponential tapers
    for s in [0.3, 0.8, 2.0, 5.0, 15.0]:
        cands.append(normalize(np.exp(-s * x)))
        cands.append(normalize(np.exp(-s * (1 - x))))
        cands.append(normalize(np.exp(-s * np.abs(xm))))

    # Power laws
    for a in [-3, -2, -1, -0.5, 0.5, 1, 2, 4]:
        cands.append(normalize(np.power(x + 1e-10, a)))
        cands.append(normalize(np.power(1 - x + 1e-10, a)))

    # Polynomial shapes
    for a in [0.5, 1, 2, 4, 8]:
        cands.append(normalize(np.maximum(1 + a * xm ** 2, 0.01)))
        cands.append(normalize(np.maximum(1 - a * xm ** 2, 0.01)))
        cands.append(normalize(np.maximum(1 + a * xm ** 4, 0.01)))
        cands.append(normalize(np.maximum(1 - a * xm ** 4, 0.01)))

    # Chebyshev-like oscillations
    for d in [3, 5, 7, 10]:
        for amp in [0.3, 0.6, 1.0]:
            T = np.zeros(n)
            T1 = xm.copy()
            for dd in range(d):
                if dd == 0:
                    T2 = np.ones(n)
                elif dd == 1:
                    T2 = xm.copy()
                else:
                    T2 = 2 * xm * T1 - T
                T = T2
                T1 = T
            cands.append(normalize(np.maximum(1 + amp * T, 0.01)))

    # Sine/cosine
    for f in [1, 2, 3, 5, 8]:
        for amp in [0.2, 0.5, 0.8, 1.2]:
            cands.append(normalize(np.maximum(1 + amp * np.sin(np.pi * f * x), 0.01)))
            cands.append(normalize(np.maximum(1 + amp * np.cos(np.pi * f * x), 0.01)))
            cands.append(normalize(np.maximum(1 + amp * np.sin(2 * np.pi * f * x), 0.01)))

    # Beta-like
    for a, b in [(0.5, 0.5), (1, 3), (3, 1), (0.3, 5), (5, 0.3), (2, 2), (1.5, 1.5)]:
        cands.append(normalize(np.power(x + 1e-8, a) * np.power(1 - x + 1e-8, b)))

    # Two-peak
    for mu, w in [(0.25, 0.05), (0.3, 0.08), (0.2, 0.03), (0.4, 0.1)]:
        p1 = np.exp(-((x - mu) ** 2) / (2 * w ** 2))
        p2 = np.exp(-((x - (1 - mu)) ** 2) / (2 * w ** 2))
        cands.append(normalize(p1 + p2))

    # Random smooth
    for _ in range(15):
        nf = int(rng.integers(2, 8))
        freqs = np.sort(rng.uniform(1, 15, nf))
        amps = rng.uniform(0.1, 1.0, nf) * np.exp(-freqs / (n * 0.1))
        phases = rng.uniform(0, 2 * np.pi, nf)
        f = np.ones(n)
        for i in range(nf):
            f += amps[i] * np.sin(2 * np.pi * freqs[i] * x + phases[i])
        cands.append(normalize(np.maximum(f, 0.02)))

    # "Sandwich" shapes (high at both ends)
    for a in [1, 2, 4, 8]:
        cands.append(normalize(x ** a + (1 - x) ** a + 0.01))

    return cands


def resample_to_length(p, n_new):
    """Resample a 1D array to a new length using linear interpolation."""
    n_old = len(p)
    if n_old == n_new:
        return p.copy()
    xp = np.linspace(0, 1, n_old)
    fp = p / p.sum()
    xp_new = np.linspace(0, 1, n_new)
    return np.interp(xp_new, xp, fp)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(p,p))/(sum(p))^2.
    Uses multi-length exploration, gradient optimization with top-k max,
    LP best-response, and sparsification.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + budget_s - 20.0

    # Get parent sequence
    parent = None
    try:
        pa = np.asarray(height_sequence_1, float)
        pa = np.where(np.isfinite(pa), pa, 0.0)
        pa = np.maximum(pa, 0.0)
        if pa.ndim == 1 and pa.size >= 32 and pa.sum() > 0:
            parent = pa
    except Exception:
        parent = None

    best_p = None
    best_score = np.inf

    def try_record(p, sc):
        nonlocal best_p, best_score
        if p is None:
            return
        p = np.asarray(p, float)
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.maximum(p, 0.0)
        if p.size == 0 or p.sum() < 1e-12:
            return
        p = p / p.sum()
        if not np.isfinite(sc):
            sc = score(p)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = float(sc)
            best_p = p.copy()
            print(f"  *** NEW BEST: {sc:.8f} (n={len(p)})", flush=True)

    # Determine lengths to explore
    lengths = [400, 700, 1000, 1500, 2000, 3000, 5000]
    if parent is not None:
        pn = len(parent)
        if pn not in lengths:
            lengths.insert(0, pn)

    print(f"Lengths to explore: {lengths}", flush=True)
    print(f"Parent length: {len(parent) if parent is not None else 'None'}", flush=True)

    # Phase 1: Quick evaluation
    print("Phase 1: Candidate evaluation", flush=True)
    t_phase1_end = start + (deadline - start) * 0.12
    all_cands = []

    for n in lengths:
        if time.monotonic() > t_phase1_end - 5:
            break
        cands = make_candidates(n, rng)

        if parent is not None:
            if len(parent) == n:
                cands.append(normalize(parent))
            else:
                cands.append(normalize(resample_to_length(parent, n)))

        for c in cands:
            c = normalize(c)
            sc = score(c)
            if np.isfinite(sc):
                all_cands.append((sc, c, n))

    all_cands.sort(key=lambda x: x[0])
    n_cands = len(all_cands)
    print(f"  Generated {n_cands} candidates. Top 5:", flush=True)
    for i in range(min(5, len(all_cands))):
        print(f"    [{i}] score={all_cands[i][0]:.6f} n={all_cands[i][2]}", flush=True)

    for sc, c, n in all_cands[:4]:
        try_record(c, sc)

    # Phase 2: Local optimization
    print("Phase 2: Local optimization", flush=True)
    t_phase2_end = start + (deadline - start) * 0.75

    top_count = min(6, len(all_cands))
    # Allocate time: give more to shorter sequences (faster per-iteration)
    jobs = []
    for i in range(top_count):
        sc, c, n = all_cands[i]
        # Time budget proportional to 1/sqrt(n) (since each iteration costs O(n log n))
        w = 1.0 / np.sqrt(max(n, 1))
        jobs.append((w, sc, c, n, i))

    total_w = sum(j[0] for j in jobs)
    available = t_phase2_end - time.monotonic()

    for w, sc, c, n, idx in jobs:
        remaining = t_phase2_end - time.monotonic()
        if remaining < 15:
            break
        t_alloc = max(10.0, min(available * (w / total_w), remaining * 0.9))
        print(f"  Job {idx}: n={n} init={sc:.6f} time={t_alloc:.0f}s", flush=True)

        p_cur = c.copy()
        sc_cur = sc

        # Strategy: gradient first, then LP if short enough
        t_grad = t_alloc * (0.4 if n <= 800 else 0.7)
        p1, sc1 = optimize_gradient(p_cur, rng, t_grad, topk=15)
        try_record(p1, sc1)
        p_cur = p1
        sc_cur = sc1

        if n <= 800 and _HAVE_LP:
            t_lp = min(60.0, t_alloc - t_grad - 5)
            if t_lp > 10 and time.monotonic() < t_phase2_end - 10:
                p2, sc2 = optimize_lp(p_cur, rng, t_lp)
                try_record(p2, sc2)
                p_cur = p2
                sc_cur = sc2

        # Final gradient polish
        t_rem = min(t_alloc, t_phase2_end - time.monotonic() - 5)
        if t_rem > 10:
            p3, sc3 = optimize_gradient(p_cur, rng, t_rem, topk=20)
            try_record(p3, sc3)

        print(f"  Job {idx} done: best={score(p_cur):.8f}", flush=True)

    # Phase 3: Sparsification attempts on best
    if best_p is not None:
        print("Phase 3: Sparsification + final polish", flush=True)
        remaining = deadline - time.monotonic()
        if remaining > 30:
            n_best = len(best_p)
            # Try various sparsification levels
            for nz in [5, 10, 20, 50]:
                if nz >= n_best // 2:
                    continue
                if time.monotonic() > deadline - 15:
                    break
                p_sp, sc_sp = sparsify(best_p, rng, nz)
                try_record(p_sp, sc_sp)
                if sc_sp < best_score:
                    best_p = p_sp.copy()

            # Final gradient polish
            t_final = max(5.0, deadline - time.monotonic() - 5)
            if t_final > 10:
                p_f, sc_f = optimize_gradient(best_p, rng, t_final, topk=25)
                try_record(p_f, sc_f)

            # If still short enough, try LP one more time
            if len(best_p) <= 800 and _HAVE_LP and time.monotonic() < deadline - 25:
                p_l, sc_l = optimize_lp(best_p, rng, min(30.0, deadline - time.monotonic() - 10))
                try_record(p_l, sc_l)

    if best_p is None:
        print("WARNING: No valid candidate found, using uniform fallback", flush=True)
        best_p = np.ones(1000) / 1000
        best_score = 2.0

    # Ensure valid output
    best_p = np.asarray(best_p, float)
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    best_p = np.minimum(best_p, 1000.0)
    s = best_p.sum()
    if s < 0.01:
        best_p = best_p * (1.0 / max(s, 1e-15))

    final_sc = score(best_p)
    print(f"\nFINAL: score={final_sc:.8f}, n={len(best_p)}, sum={best_p.sum():.4f}", flush=True)
    print(f"   max(conv)={conv_full(best_p).max():.10f}, 2n/(sum^2)={2*len(best_p)/(best_p.sum()**2):.4f}", flush=True)

    return [float(x) for x in best_p]
