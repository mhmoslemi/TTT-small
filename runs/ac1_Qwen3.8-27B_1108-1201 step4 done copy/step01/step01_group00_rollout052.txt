LP-based alternating minimization with gradient refinement and simulated annealing
for minimizing the normalized peak autocorrelation of non-negative step functions.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n+m-1]

def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)

def lp_direction(f, S, time_limit=5.0):
    """Solve LP: min t s.t. (f*g)[k] <= t, sum(g)=S, g>=0"""
    n = len(f)
    c = np.append(np.zeros(n), 1.0)
    
    A_ub_list = []
    for k in range(2*n-1):
        row = np.zeros(n+1)
        lo = max(0, k-n+1)
        hi = min(n-1, k)
        for i in range(lo, hi+1):
            row[k-i] = f[i]
        row[-1] = -1.0
        A_ub_list.append(row)
    A_ub = np.array(A_ub_list)
    b_ub = np.zeros(2*n-1)
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])

            g = g / g.sum() * S
            return g, res.x[n] if len(res.x) > n else np.max(fft_convolve(f, g))
    return None, None

    c = fft_convolve(f, f)
    k_star = int(np.argmax(c))
    # Subgradient: sg[j] = 2*f[k_star - j] if 0 <= k_star-j < n
    sg = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            sg[j] = 2.0 * f[idx]
    # Move in negative subgradient direction
    g = f - 0.01 * sg
    g = np.maximum(g, 0)
    if g.sum() < 1e-10:
        g = np.ones(n) / n * S
    else:
        g = g / g.sum() * S
    return g

def lp_iteration(f, S, eta=0.5, time_limit=5.0):
    """One LP-based iteration. Returns updated f."""
    g, t = lp_direction(f, S, time_limit)
    if g is None:
        g = subgradient_direction(f, S)
    # Update: f_new = (1-eta)*f + eta*g
    f_new = (1-eta) * f + eta * g
    f_new = np.maximum(f_new, 0)
    f_new = f_new / f_new.sum() * S
    return f_new

def anneal_polish(a, time_limit, rng):
    """Simulated annealing polish."""
    current = a.copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temp = 0.005
    
    while time.monotonic() - t0 < time_limit:
        r = rng.random()
        candidate = current.copy()
        
        if r < 0.4:
            # Targeted peak reduction
            c = fft_convolve(current, current)
            k = int(np.argmax(c))
            # Find top contributors to the peak
            contribs = np.zeros(n)
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    contribs[j] = current[j] * current[idx]
            top_k = max(3, n // 8)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=len(top_idx)))
            others = np.setdiff1d(np.arange(n), top_idx)
            if len(others) > 0:
                n_inc = min(10, len(others))
                inc_idx = rng.choice(others, n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.05, 0.02, size=n_inc))
        elif r < 0.6:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start+width] *= np.exp(rng.normal(0, 0.04, size=width))
        elif r < 0.75:
            # Symmetrize with noise
            sym = 0.5 * (current + current[::-1])
            candidate = sym * np.exp(rng.normal(0, 0.02, size=n))
        else:
            # Global small perturbation
            candidate = current * np.exp(rng.normal(0, 0.015, size=n))
        
        candidate = np.clip(candidate, 0.0, 1000.0)
        new_score = local_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best = current.copy()
                best_score = current_score
        elif rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-12)):
            current = candidate
            current_score = new_score
        
        temp *= 0.9998
    
    return best, best_score

def gradient_refine(a, time_limit, rng):
    """Adam gradient descent with softmax parameterization."""
    n = len(a)
    S = np.sum(a)
    theta = np.log(a / S + 1e-10)
    
    momentum1 = np.zeros(n)
    momentum2 = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0
    t0 = time.monotonic()
    best_theta = theta.copy()
    best_score = local_score(a)
    
    while time.monotonic() - t0 < time_limit:
        elapsed = (time.monotonic() - t0) / time_limit
        tau = (30.0 / n) * (0.3 / 30.0) ** min(1.0, elapsed)
        
        # Softmax normalization
        t_shift = theta - np.max(theta)
        e = np.exp(t_shift)
        p = e / e.sum()
        a_curr = p * S
        
        # Compute gradient of soft-max(conv(a,a), tau)
        c = fft_convolve(a_curr, a_curr)
        c_max = c.max()
        shifted = (c - c_max) / tau
        exp_shifted = np.exp(shifted)
        w = exp_shifted / exp_shifted.sum()
        
        # Gradient w.r.t. a
        rev_a = a_curr[::-1]
        corr = fft_convolve(rev_a, w)
        g_a = 2.0 * corr[n-1:2*n-1]
        
        # Gradient w.r.t. theta (chain rule through softmax)
        inner = np.dot(a_curr, g_a)
        g_theta = p * (g_a - inner / S)
        
        step += 1
        momentum1 = beta1 * momentum1 + (1-beta1) * g_theta
        momentum2 = beta2 * momentum2 + (1-beta2) * (g_theta**2)
        m1_hat = momentum1 / (1 - beta1**step)
        m2_hat = momentum2 / (1 - beta2**step)
        lr = 1.0 / np.sqrt(step + 1)
        theta -= lr * m1_hat / (np.sqrt(m2_hat) + eps)
        
        # Check score periodically
        if step % 100 == 0:
            a_check = softmax_from_theta(theta) * S
            score = local_score(a_check)
            if score < best_score:
                best_score = score
                best_theta = theta.copy()
    
    a_best = softmax_from_theta(best_theta) * S
    return a_best, best_score

def softmax_from_theta(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / e.sum()

"""
LP-based alternating minimization with gradient refinement and simulated annealing
for minimizing the normalized peak autocorrelation of non-negative step functions.
Uses linear programming to find optimal descent directions that reduce the autocorrelation
peak, combined with Adam gradient descent and stochastic polishing for final refinement.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def local_score(a):
    """Compute the score (same formula as evaluate_sequence)."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def build_lp_constraint_matrix(f):
    """Build A_ub and b_ub for the LP: (f*g)[k] - t <= 0."""
    n = len(f)
    num_constraints = 2 * n - 1
    A_ub = np.zeros((num_constraints, n + 1))
    for k in range(num_constraints):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            A_ub[k, k - i] = f[i]
        A_ub[k, -1] = -1.0
    b_ub = np.zeros(num_constraints)
    return A_ub, b_ub


def lp_solve_direction(f, S, time_limit=8.0):
    """
    Solve LP: minimize t subject to (f*g)[k] <= t, sum(g) = S, g >= 0.
    Returns (g, t) or (None, None) on failure.
    """
    n = len(f)
    c = np.append(np.zeros(n), 1.0)
    A_ub, b_ub = build_lp_constraint_matrix(f)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.x is not None and len(res.x) == n + 1:
            g = np.maximum(res.x[:n], 0.0)
            g_sum = g.sum()
            if g_sum < 1e-10:
                return None, None
            g = g / g_sum * S
            t_val = res.x[n]
            return g, t_val
    except Exception:
        pass
    return None, None


def subgradient_direction(f, S, step=0.02):
    """Subgradient-based direction: reduce peak lag contributions."""
    n = len(f)
    c = fft_convolve(f, f)
    k_star = int(np.argmax(c))
    sg = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            sg[j] = 2.0 * f[idx]
    g = f - step * sg
    g = np.maximum(g, 0.0)
    g_sum = g.sum()
    if g_sum < 1e-10:
        g = np.ones(n) / n * S
    else:
        g = g / g_sum * S
    return g


def multi_peak_subgradient(f, S, num_peaks=3, step=0.015):
    """Subgradient targeting multiple peak lags simultaneously."""
    n = len(f)
    c = fft_convolve(f, f)
    c_sorted = np.sort(c)[::-1]
    threshold = c_sorted[min(num_peaks - 1, len(c_sorted) - 1)] * 0.95
    peak_lags = np.where(c >= threshold)[0]
    sg = np.zeros(n)
    for k in peak_lags:
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                sg[j] += f[idx]
    g = f - step * sg
    g = np.maximum(g, 0.0)
    g_sum = g.sum()
    if g_sum < 1e-10:
        g = np.ones(n) / n * S
    else:
        g = g / g_sum * S
    return g

    while time.monotonic() - t0 < time_budget:
        iteration += 1
        elapsed_frac = (time.monotonic() - t0) / time_budget

        # Solve LP for direction
        g, t_lp = lp_solve_direction(f, S, time_limit=min(5.0, max(1.0, (time_budget - (time.monotonic() - t0)) / 2)))

        if g is not None:
            # Damped update
            f_new = (1.0 - eta) * f + eta * g
            f_new = np.maximum(f_new, 0.0)
            f_new_sum = f_new.sum()
            if f_new_sum > 1e-10:
                f_new = f_new / f_new_sum * S
            else:
                f_new = f.copy()
        else:
            # Fallback: subgradient
            if rng.random() < 0.5:
                f_new = subgradient_direction(f, S, step=0.03 * (1 + 0.5 * rng.random()))
            else:
                f_new = multi_peak_subgradient(f, S, num_peaks=2 + int(rng.integers(0, 3)), step=0.02)

        score = local_score(f_new)

        if score < best_score:
            best_score = score
            best_f = f_new.copy()
            patience = 0
            if verbose and iteration % 5 == 0:
                print(f"    iter {iteration}: score={score:.6f} *** best ***")
        else:
            patience += 1
            if verbosity := verbose:
                if iteration % 10 == 0:
                    print(f"    iter {iteration}: score={score:.6f}, best={best_score:.6f}, eta={eta:.3f}, patience={patience}")

        # Adaptive step size
        if patience > 0:
            eta = max(0.05, eta * (1.0 - 0.005 * patience))
        else:
            eta = min(0.8, eta * 1.02)

        # Accept/reject
        f = f_new

        # Restart with perturbation if stuck too long
        if patience > 80:
            if verbose:
                print(f"    Restarting with perturbation at iter {iteration}")
            f = best_f * np.exp(rng.normal(0, 0.05, size=n))
            f = np.maximum(f, 0.01)
            f = f / f.sum() * S
            patience = 0
            eta = 0.5

        # Periodic best-so-far checkpoint
        if patience > 150:
            break

    return best_f, best_score


def gradient_refine(a_init, time_limit, rng):
    """Adam gradient descent with softmax parameterization and annealed softmax."""
    n = len(a_init)
    S = np.sum(a_init)
    theta = np.log(np.maximum(a_init, 1e-10) / S)

    momentum1 = np.zeros(n)
    momentum2 = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0
    t0 = time.monotonic()
    best_theta = theta.copy()
    best_score = local_score(a_init)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = (time.monotonic() - t0) / time_limit
        tau = (20.0 / n) * (0.2 / 20.0) ** min(1.0, elapsed * 1.2)

        # Softmax
        t_shift = theta - np.max(theta)
        e = np.exp(t_shift)
        p = e / e.sum()
        a_curr = p * S

        # Compute soft-max of conv and its gradient
        c = fft_convolve(a_curr, a_curr)
        c_max = c.max()
        shifted = (c - c_max) / tau
        exp_shifted = np.exp(shifted)
        exp_sum = exp_shifted.sum()
        w = exp_shifted / exp_sum

        # Gradient of soft_max(conv(a,a)) w.r.t. a
        rev_a = a_curr[::-1]
        corr = fft_convolve(rev_a, w)
        g_a = 2.0 * corr[n - 1:2 * n - 1]

        # Chain rule through softmax: d(soft_max)/d(theta)
        inner = np.dot(a_curr, g_a) / S
        g_theta = p * (g_a / S - inner)

        step += 1
        momentum1 = beta1 * momentum1 + (1 - beta1) * g_theta
        momentum2 = beta2 * momentum2 + (1 - beta2) * (g_theta * g_theta)
        m1_hat = momentum1 / (1.0 - beta1 ** step)
        m2_hat = momentum2 / (1.0 - beta2 ** step)
        lr = 1.5 / np.sqrt(step + 10)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        # Evaluate periodically
        if step % 50 == 0:
            t_shift2 = theta - np.max(theta)
            e2 = np.exp(t_shift2)
            p2 = e2 / e2.sum()
            a_check = p2 * S
            score = local_score(a_check)
            if score < best_score:
                best_score = score
                best_theta = theta.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 40:
                break

    # Extract best
    t_shift = best_theta - np.max(best_theta)
    e = np.exp(t_shift)
    p = e / e.sum()
    a_best = p * S
    return a_best, best_score


def anneal_polish(a_init, time_limit, rng):
    """Simulated annealing with targeted perturbations."""
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temp = 0.004
    cooling = 0.9997
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        r = rng.random()
        candidate = current.copy()

        if r < 0.35:
            # Targeted: reduce peak contributors
            c = fft_convolve(current, current)
            peak_idx = int(np.argmax(c))
            contribs = np.zeros(n)
            for j in range(n):
                idx = peak_idx - j
                if 0 <= idx < n:
                    contribs[j] = current[j] * current[idx]
            top_k = max(3, n // 10)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.06, 0.02, size=len(top_idx)))
            others = np.setdiff1d(np.arange(n), top_idx)
            if len(others) > 0:
                n_inc = min(12, len(others))
                inc_idx = rng.choice(others, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.06, 0.02, size=n_inc))
        elif r < 0.5:
            # Second peak reduction
            c = fft_convolve(current, current)
            sorted_idx = np.argsort(c)[::-1]
            peak2 = int(sorted_idx[1])
            for j in range(n):
                idx = peak2 - j
                if 0 <= idx < n:
                    candidate[j] *= np.exp(0.03 * current[idx] / max(current.sum() / n, 1e-10))
        elif r < 0.65:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 6))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.04, size=width))
        elif r < 0.8:
            # Symmetrize with noise
            sym = 0.5 * (current + current[::-1])
            candidate = sym * np.exp(rng.normal(0, 0.015, size=n))
        elif r < 0.9:
            # Local swap-like
            i1 = int(rng.integers(0, n))
            i2 = int(rng.integers(0, n))
            if i1 != i2:
                delta = 0.05 * rng.standard_normal()
                candidate[i1] *= np.exp(delta)
                candidate[i2] *= np.exp(-delta)
        else:
            # Global small
            candidate = current * np.exp(rng.normal(0, 0.01, size=n))

        candidate = np.clip(candidate, 0.0, 1000.0)
        new_score = local_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
                if iter_count % 500 == 0:
                    print(f"    anneal iter {iter_count}: score={best_score:.6f}")
        elif rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-12)):
            current = candidate
            current_score = new_score

        temp *= cooling

    return best, best_score


def make_chirp(n, alpha, rng):
    """Chirp-like initial sequence."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multitone(n, rng):
    """Multi-tone initial sequence."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k, amp in [(1, 0.3), (2, 0.15), (3, 0.1), (5, 0.06), (7, 0.04)]:
        a += amp * np.cos(2.0 * np.pi * k * t / n + rng.uniform(0, 2 * np.pi))
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_spread(n, rng, spread=0.25):
    """Random initial with controlled spread."""
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: LP-based alternating minimization, gradient refinement,
    and simulated annealing polish. Returns best sequence found.
    """
    t_global = time.monotonic()
    deadline = t_global + max(0.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Get starting sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = local_score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()

    # === PHASE 1: LP-based optimization on provided sequence ===
    if best_seq is not None:
        n = len(best_seq)
        lp_time = min(900.0, max(60.0, (deadline - time.monotonic()) * 0.55))
        print(f"\n=== Phase 1: LP optimization (n={n}, budget={lp_time:.0f}s) ===")
        t0 = time.monotonic()
        lp_result, lp_score = lp_optimize(best_seq, lp_time, rng, verbose=True)
        elapsed = time.monotonic() - t0
        print(f"  LP done in {elapsed:.0f}s: score={lp_score:.6f}")
        if lp_score < best_score:
            best_score = lp_score
            best_seq = lp_result.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # === PHASE 2: Multi-start LP on different initializations ===
    print(f"\n=== Phase 2: Multi-start LP ===")
    configs = [
        (256, lambda: make_chirp(256, 0.4 + 0.2 * rng.random(), rng)),
        (256, lambda: make_multitone(256, rng)),
        (384, lambda: make_chirp(384, 0.5, rng)),
        (512, lambda: make_chirp(512, 0.45, rng)),
        (512, lambda: make_multitone(512, rng)),
        (384, lambda: make_random_spread(384, rng, 0.3)),
    ]

    for n, init_fn in configs:
        if time.monotonic() > deadline - 90:
            break
        a0 = init_fn()
        S0 = np.sum(a0)
        a0 = a0 / S0 * np.sqrt(2.0 * n)
        initial_s = local_score(a0)
        print(f"  n={n}: initial score={initial_s:.6f}")

        if initial_s >= best_score + 0.05:
            continue  # Skip if clearly worse

        remaining = min(180.0, (deadline - time.monotonic()) * 0.3)
        if remaining < 20:
            continue
        lp_res, lp_s = lp_optimize(a0, remaining, rng, verbose=False)
        print(f"  n={n}: after LP score={lp_s:.6f}")
        if lp_s < best_score:
            best_score = lp_s
            best_seq = lp_res.copy()
            print(f"  *** New best: {best_score:.6f} (n={n}) ***")

    # === PHASE 3: Gradient refinement ===
    if best_seq is not None and time.monotonic() < deadline - 60:
        print(f"\n=== Phase 3: Gradient refinement ===")
        grad_time = min(300.0, max(30.0, (deadline - time.monotonic()) * 0.25))
        print(f"  Budget: {grad_time:.0f}s, current best={best_score:.6f}")
        a_grad, grad_score = gradient_refine(best_seq, grad_time, rng)
        print(f"  Gradient result: {grad_score:.6f}")
        if grad_score < best_score:
            best_score = grad_score
            best_seq = a_grad.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # === PHASE 4: Annealing polish ===
    if best_seq is not None and time.monotonic() < deadline - 15:
        print(f"\n=== Phase 4: Annealing polish ===")
        polish_time = max(15.0, deadline - time.monotonic() - 10.0)
        print(f"  Budget: {polish_time:.0f}s, current best={best_score:.6f}")
        polished, polish_score = anneal_polish(best_seq, polish_time, rng)
        if polish_score < best_score:
            best_score = polish_score
            best_seq = polished.copy()
            print(f"  *** New best: {polish_score:.6f} ***")
        else:
            print(f"  No improvement from polish (best={best_score:.6f})")

    # === Phase 5: Quick final LP restart if time permits ===
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic() - 5
        if remaining > 30:
            n = len(best_seq)
            # Small perturbation + LP
            perturbed = best_seq * np.exp(rng.normal(0, 0.02, size=n))
            perturbed = np.maximum(perturbed, 0.01)
            perturbed = perturbed / perturbed.sum() * np.sum(best_seq)
            lp_res5, lp_s5 = lp_optimize(perturbed, min(60.0, remaining * 0.7), rng, verbose=False)
            if lp_s5 < best_score:
                best_score = lp_s5
                best_seq = lp_res5.copy()
                print(f"  Phase 5 restart: *** New best: {best_score:.6f} ***")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
