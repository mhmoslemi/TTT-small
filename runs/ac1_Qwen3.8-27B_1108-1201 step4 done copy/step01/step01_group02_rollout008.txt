import time


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)

        return float('inf')

def initial_raised_cosine(n):

    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))
    return normalize_nonneg(a)


def initial_gaussian(n, sigma_frac=0.3):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    return normalize_nonneg(a)


def initial_blackman(n):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    x = t / (n - 1)
    a = np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))

    return normalize_nonneg(a)


def random_bumps(n, rng, n_bumps=4, max_width_frac=0.4):

def build_convolve_matrix_cols(a):
    """Build sparse matrix A where A @ g = conv(a, g) for the full convolution."""
    n = len(a)
    n_rows = 2 * n - 1
    row_idx = []
    col_idx = []
    data = []
    for i in range(n):
        # g[i] contributes to conv positions i, i+1, ..., i+n-1
        # conv[a,g][k] = sum_j a[j] * g[k-j]
        # So g[i] appears in conv[k] with coefficient a[k-i] when 0 <= k-i < n
        # i.e., k in [i, i+n-1], coefficient is a[k-i]
        for k in range(i, min(i + n, n_rows)):
            row_idx.append(k)
            col_idx.append(i)
            data.append(a[k - i])
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def lp_minimax_step(a, time_limit=30.0):
    """Solve LP: minimize t s.t. conv(a,g)_k <= t for all k, sum(g)=1, g>=0.
    Returns (g, t_opt) or (None, None) if failed."""
    n = len(a)
    n_rows = 2 * n - 1
    A = build_convolve_matrix_cols(a)
    
    # Variables: [t, g_0, ..., g_{n-1}]
    c = np.zeros(n + 1)
    c[0] = 1.0
    
    # Inequality: -t + (A @ g)_k <= 0 for all k
    A_ub = np.hstack([
        -np.ones((n_rows, 1)),
        A.toarray()
    ])
    b_ub = np.zeros(n_rows)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([1.0])

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x[1:], res.x[0]
    except Exception:
        pass
    return None, None

            if res.success or res.fun < obj_and_grad(x0)[0]:

"""
Hybrid optimization for minimizing peak auto-convolution of non-negative sequences.
Combines LP minimax refinement, L-BFGS-B with soft-max smoothing, and adaptive local search.
Key improvements: better LP formulation (minimize max cross-conv), line search for step size,
adaptive local search, and multi-scale initialization.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog, minimize
import time


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))
    return normalize_nonneg(a)


def initial_gaussian(n, sigma_frac=0.3):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    return normalize_nonneg(a)


def initial_blackman(n):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    x = t / (n - 1)
    a = np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))
    return normalize_nonneg(a)


def initial_flat_taper(n, taper_frac=0.35):
    """Flat in middle, smooth taper at edges."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    edge = max(1, int(n * taper_frac / 2))
    if edge < n // 2:
        ramp = 0.5 * (1.0 + np.cos(np.pi * np.arange(edge) / edge))
        a[:edge] = ramp
        a[-edge:] = ramp[::-1]
    return normalize_nonneg(a)


def random_bumps(n, rng, n_bumps=4, max_width_frac=0.4):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(2, n * max_width_frac // 10), max(3, n * max_width_frac // 2)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    return normalize_nonneg(a)


def build_conv_matrix(a):
    """Build sparse matrix A where (A @ g)[k] = conv(a, g)[k] for full convolution."""
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = col_idx + np.tile(np.arange(n, dtype=np.int32), n)
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def lp_minimax_step(a, time_limit=30.0):
    """LP: minimize t s.t. conv(a,g)_k <= t for all k, sum(g)=1, g>=0.
    Returns (g, t_opt) or (None, None)."""
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[0] = 1.0

    A_ub = np.hstack([
        -np.ones((n_rows, 1)),
        A.toarray()
    ])
    b_ub = np.zeros(n_rows)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] + [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x[1:], res.x[0]
    except Exception:
        pass
    return None, None


def lp_refine(a, time_limit=300.0, verbose=True):
    """Iteratively refine using LP minimax with line search."""
    start_time = time.monotonic()
    a = normalize_nonneg(a.astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)

    if verbose:
        print(f"    LP start: score={best_score:.6f}, n={n}")

    stagnant = 0
    for it in range(1000):
        elapsed = time.monotonic() - start_time
        remaining = time_limit - elapsed
        if remaining < 5.0:
            break

        g, t_opt = lp_minimax_step(a, time_limit=min(30.0, remaining * 0.4))
        if g is None:
            if verbose:
                print(f"    LP solver failed at iter {it}")
            break
        g = normalize_nonneg(g)

        current_score = score_sequence(a)
        best_alpha = 0.0
        best_sc = current_score

        # Fine-grained line search
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 0.9, 1.0]:
            a_new = (1.0 - alpha) * a + alpha * g
            a_new = normalize_nonneg(a_new)
            sc = score_sequence(a_new)
            if sc < best_sc:
                best_sc = sc
                best_alpha = alpha

        if best_alpha == 0.0:
            stagnant += 1
            if stagnant > 15:
                if verbose:
                    print(f"    LP stagnant at iter {it}, best={best_score:.6f}")
                break
            continue

        a = (1.0 - best_alpha) * a + best_alpha * g
        a = normalize_nonneg(a)

        if best_sc < best_score - 1e-13:
            best_score = best_sc
            best_a = a.copy()
            stagnant = 0
            if verbose and (it < 5 or it % 20 == 0):
                print(f"    LP iter {it:3d}: score={best_score:.6f}, alpha={best_alpha:.2f}, t_opt={t_opt:.6f}")
        else:
            stagnant += 1
            if stagnant > 15:
                break

    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    if verbose:
        print(f"    LP done: best={best_score:.6f}, time={time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def lbfgsb_refine(a, time_limit=120.0, verbose=True):
    """Refine using L-BFGS-B with soft-max approximation and log-parameterization."""
    start_time = time.monotonic()
    a = normalize_nonneg(a.astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)

    if verbose:
        print(f"    L-BFGS start: score={best_score:.6f}, n={n}")

    x0 = np.log(a + 1e-10)
    tau_values = [30.0, 100.0, 500.0, 2000.0]

    for tau in tau_values:
        remaining = time_limit - (time.monotonic() - start_time)
        if remaining < 15.0:
            break

        def obj_and_grad(x):
            a_sm = np.exp(x - x.max())
            a_sm = a_sm / a_sm.sum()
            conv = np.convolve(a_sm, a_sm)
            max_conv = conv.max()
            exp_conv = np.exp(tau * (conv - max_conv))
            denom = exp_conv.sum()
            val = max_conv + np.log(denom) / tau
            weights = exp_conv / denom
            grad_a = np.correlate(weights, a_sm, 'full')[:n]
            grad_x = a_sm * (grad_a - np.dot(grad_a, a_sm))
            return val, grad_x

        try:
            res = minimize(obj_and_grad, x0, method='L-BFGS-B',
                          options={'maxiter': 300, 'maxfun': 1000,
                                   'maxls': 50, 'ftol': 1e-14})
            a_new = np.exp(res.x - res.x.max())
            a_new = a_new / a_new.sum()
            sc = score_sequence(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                x0 = res.x.copy()
                if verbose:
                    print(f"    L-BFGS tau={tau:.0f}: score={sc:.6f}")
            else:
                if verbose:
                    print(f"    L-BFGS tau={tau:.0f}: no improvement (got {sc:.6f})")
        except Exception as e:
            if verbose:
                print(f"    L-BFGS tau={tau:.0f} error: {e}")

    if verbose:
        print(f"    L-BFGS done: best={best_score:.6f}, time={time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def local_search(a, time_limit=60.0, rng=None, verbose=True):
    """Adaptive local search with smooth random perturbations."""
    if rng is None:
        rng = np.random.default_rng(42)
    start_time = time.monotonic()
    a = normalize_nonneg(a.astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    step_size = 0.08

    if verbose:
        print(f"    LS start: score={best_score:.6f}")

    accepts = 0
    for it in range(100000):
        if time.monotonic() - start_time > time_limit:
            break

        # Multi-scale perturbation
        r = rng.random()
        if r < 0.5:
            # Local smooth perturbation
            width = max(2, int(rng.integers(n // 300, n // 4)))
            start = int(rng.integers(0, max(1, n - width)))
            stop = min(n, start + width)
            candidate = a.copy()
            candidate[start:stop] *= np.exp(rng.normal(0, step_size, size=stop - start))
        elif r < 0.7:
            # Two-local perturbation
            candidate = a.copy()
            for _ in range(2):
                width = max(2, int(rng.integers(n // 200, n // 5)))
                start = int(rng.integers(0, max(1, n - width)))
                stop = min(n, start + width)
                candidate[start:stop] *= np.exp(rng.normal(0, step_size * 0.7, size=stop - start))
        elif r < 0.85:
            # Global shape change
            t = np.arange(n, dtype=np.float64)
            center = (n - 1) / 2.0
            candidate = a * np.exp(rng.normal(0, step_size * 0.5) * (t - center) / (n / 2.0))
        else:
            # Scale locally
            width = max(3, int(n * rng.uniform(0.1, 0.5)))
            start = int(rng.integers(0, max(1, n - width)))
            stop = min(n, start + width)
            candidate = a.copy()
            candidate[start:stop] *= rng.uniform(0.5, 1.5)

        candidate = normalize_nonneg(candidate)
        sc = score_sequence(candidate)

        if sc < best_score:
            best_score = sc
            best_a = candidate.copy()
            a = candidate.copy()
            step_size = min(0.3, step_size * 1.03)
            accepts += 1
        else:
            step_size = max(0.01, step_size * 0.97)

        if verbose and it % 5000 == 0 and it > 0:
            print(f"    LS iter {it:5d}: score={best_score:.6f}, step={step_size:.4f}, accepts={accepts}")

    if verbose:
        print(f"    LS done: best={best_score:.6f}, time={time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def coordinate_descent(a, time_limit=60.0, verbose=True):
    """Coordinate descent: try adjusting pairs of elements to reduce the peak."""
    start_time = time.monotonic()
    a = normalize_nonneg(a.astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    conv = np.convolve(a, a)
    peak_idx = conv.argmax()

    if verbose:
        print(f"    CD start: score={best_score:.6f}, peak at {peak_idx}")

    for it in range(20000):
        if time.monotonic() - start_time > time_limit:
            break

        # Find elements contributing most to the peak
        # conv[peak_idx] = sum_j a[j] * a[peak_idx - j]
        # Try reducing the largest contributors
        contributions = np.zeros(n)
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        for j in range(lo, hi + 1):
            contributions[j] = a[j] * a[peak_idx - j]

        top_idx = np.argsort(contributions)[::-1][:5]
        j1 = top_idx[rng_int := int(np.random.randint(len(top_idx)))]
        j2 = int(np.random.randint(n))

        # Try reducing j1 and increasing j2
        candidate = a.copy()
        delta = candidate[j1] * np.random.uniform(0.05, 0.3)
        if delta > 0:
            candidate[j1] -= delta
            candidate[j2] += delta
            candidate = normalize_nonneg(candidate)
            sc = score_sequence(candidate)
            if sc < best_score:
                best_score = sc
                best_a = candidate.copy()
                a = candidate.copy()
                conv = np.convolve(a, a)
                peak_idx = conv.argmax()

    if verbose:
        print(f"    CD done: best={best_score:.6f}, time={time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses LP minimax refinement as primary method, with L-BFGS-B and local search for polishing.
    Multi-start with diverse initializations at multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 25.0

    best_score = float('inf')
    best_seq = None

    # Try to load parent sequence
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        parent = normalize_nonneg(parent)
        parent_score = score_sequence(parent)
        print(f"Parent: n={len(parent)}, score={parent_score:.6f}")
        if parent_score < best_score:
            best_score = parent_score
            best_seq = parent.copy()
    except Exception:
        pass

    # Phase 1: Quick scan of diverse initializations
    print("\n--- Phase 1: Quick scan ---")
    init_configs = []

    if best_seq is not None:
        init_configs.append((len(best_seq), 'parent', best_seq.copy()))

    for n in [300, 500, 700, 1000, 1200, 1500]:
        init_configs.append((n, 'cosine', initial_raised_cosine(n)))
        init_configs.append((n, 'gauss', initial_gaussian(n, 0.2 + 0.15 * rng.random())))
        init_configs.append((n, 'blackman', initial_blackman(n)))
        init_configs.append((n, 'flat', initial_flat_taper(n, 0.25 + 0.2 * rng.random())))
        if n <= 1000:
            init_configs.append((n, 'random', random_bumps(n, rng, n_bumps=int(rng.integers(3, 8)))))

    # Score all quick
    scored = []
    for n, name, a in init_configs:
        sc = score_sequence(a)
        scored.append((sc, n, name, a))
    scored.sort(key=lambda x: x[0])

    print("Top 10 initializations:")
    for sc, n, name, _ in scored[:10]:
        print(f"  n={n:4d} {name:10s}: {sc:.6f}")

    # Phase 2: LP refinement on top candidates
    print("\n--- Phase 2: LP Refinement ---")
    # Select diverse top candidates (different n values preferred)
    selected = []
    seen_ns = set()
    for sc, n, name, a in scored:
        if len(selected) >= 5:
            break
        if n in seen_ns:
            continue
        seen_ns.add(n)
        selected.append((sc, n, name, a))

    # Ensure we include n from parent if it's good
    if best_seq is not None:
        pn = len(best_seq)
        if pn not in seen_ns:
            selected.insert(0, (best_score, pn, 'parent', best_seq.copy()))

    for sc, n, name, a in selected:
        remaining = deadline - time.monotonic()
        if remaining < 60.0:
            break

        print(f"\n=== LP refine: n={n} ({name}), init_score={sc:.6f} ===")
        lp_time = min(180.0, remaining * 0.45)
        a_out, sc_out = lp_refine(a, time_limit=lp_time, verbose=True)

        if sc_out < best_score:
            best_score = sc_out
            best_seq = a_out.copy()
            print(f"  *** NEW BEST: {sc_out:.6f} (n={n})")

    # Phase 3: L-BFGS-B polish on best
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        print(f"\n--- Phase 3: L-BFGS-B polish (remaining={remaining:.0f}s) ---")
        if remaining > 40:
            lb_time = min(80.0, remaining * 0.35)
            a_out, sc_out = lbfgsb_refine(best_seq, time_limit=lb_time, verbose=True)
            if sc_out < best_score:
                best_score = sc_out
                best_seq = a_out.copy()
                print(f"  *** NEW BEST after L-BFGS: {sc_out:.6f}")

    # Phase 4: Local search polishing
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        print(f"\n--- Phase 4: Local search (remaining={remaining:.0f}s) ---")
        if remaining > 30:
            ls_time = min(150.0, remaining * 0.5)
            a_out, sc_out = local_search(best_seq, time_limit=ls_time, rng=rng, verbose=True)
            if sc_out < best_score:
                best_score = sc_out
                best_seq = a_out.copy()
                print(f"  *** NEW BEST after LS: {sc_out:.6f}")

    # Phase 5: Final LP refinement on the best
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        print(f"\n--- Phase 5: Final LP (remaining={remaining:.0f}s) ---")
        if remaining > 30:
            lp_time2 = min(100.0, remaining * 0.6)
            a_out, sc_out = lp_refine(best_seq, time_limit=lp_time2, verbose=True)
            if sc_out < best_score:
                best_score = sc_out
                best_seq = a_out.copy()
                print(f"  *** NEW BEST after final LP: {sc_out:.6f}")

    # Phase 6: Final local search
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            print(f"\n--- Phase 6: Final LS (remaining={remaining:.0f}s) ---")
            ls_time2 = min(60.0, remaining * 0.7)
            a_out, sc_out = local_search(best_seq, time_limit=ls_time2, rng=rng, verbose=True)
            if sc_out < best_score:
                best_score = sc_out
                best_seq = a_out.copy()

    if best_seq is None:
        best_seq = initial_raised_cosine(500)
        best_score = score_sequence(best_seq)
        print(f"  Fallback: {best_score:.6f}")

    # Verify
    verify_score = evaluate_sequence([float(x) for x in best_seq])
    print(f"\n{'='*50}")
    print(f"FINAL: score={best_score:.6f}, verified={verify_score:.6f}, n={len(best_seq)}")
    print(f"Total time: {time.monotonic()-total_start:.1f}s")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
