    # For row k, valid j range: max(0, k-(n-1)) to min(n-1, k)

def lp_step(a, M, timeout=10.0):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0"""
    n = len(a)
    n_rows = 2 * n - 1

    # C[k, j] = a[k-j] if 0 <= k-j < n
    row_idx = []
    col_idx = []
    data = []
    for k in range(n_rows):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)

    return normalize_nonneg(a)


def build_conv_matrix(a):
    """Build sparse matrix C where C @ g = conv(a, g)."""
    n = len(a)
    n_rows = 2 * n - 1
    # C[k, j] = a[k-j] if 0 <= k-j < n
    row_idx = []
    col_idx = []
    data = []
    for k in range(n_rows):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            row_idx.append(k)
            col_idx.append(j)
            data.append(a[k - j])
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def build_conv_matrix_fast(a):
    """Faster construction using vectorized operations."""
    n = len(a)
    n_rows = 2 * n - 1
    # For each k, the valid j range is [max(0,k-n+1), min(n-1,k)]
    # Let's build row by row but use numpy for the data
    row_parts = []
    col_parts = []
    data_parts = []
    for k in range(n_rows):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        length = j_hi - j_lo + 1
        row_parts.append(np.full(length, k, dtype=np.int32))
        col_parts.append(np.arange(j_lo, j_hi + 1, dtype=np.int32))
        data_parts.append(a[k - np.arange(j_lo, j_hi + 1)])
    row_idx = np.concatenate(row_parts)
    col_idx = np.concatenate(col_parts)
    data = np.concatenate(data_parts)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def lp_step(a, M, C, timeout=10.0):
    """Solve LP: max sum(g) s.t. C@g <= M, g >= 0. Returns g or None."""
    n = len(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9,
                               'dual_objective_bound': -1.0})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
        elif res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def lp_refine(a, time_limit_s=120.0, verbose=True):
    """
    Iteratively refine using LP steps.
    At each step: find g maximizing sum(g) s.t. conv(a,g) <= M, g >= 0.
    Mix a with g to get new a.
    """
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    stagnant = 0
    it = 0
    
    # Pre-build the conv matrix (a changes, but we rebuild each iter)
    while time.monotonic() - start_time < time_limit_s:
        it += 1
        conv = np.convolve(a, a)
        M = conv.max()
        score = 2.0 * n * M  # sum(a) = 1
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 20 == 0 or it <= 3):
                print(f"    LP iter {it:4d}: score={score:.8f}, M={M:.8f}")
        else:
            stagnant += 1
        
        if stagnant > 40:
            if verbose:
                print(f"    LP stagnant at iter {it}, best={best_score:.8f}")
            break
        
        remaining = time_limit_s - (time.monotonic() - start_time)
        if remaining < 5:
            break
        
        timeout = max(2.0, min(30.0, remaining * 0.15))
        
        # Build conv matrix for current a
        C = build_conv_matrix_fast(a)
        
        g = lp_step(a, M, C, timeout=timeout)
        if g is None:
            if verbose:
                print(f"    LP failed at iter {it}")
            break
        
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-9:
            if verbose:
                print(f"    LP no improvement (sum_g={sum_g:.10f}) at iter {it}")
            break
        
        g = g / sum_g
        # Adaptive step size
        alpha = min(0.4, max(0.02, 3.0 / (it + 20.0)))
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = normalize_nonneg(a_new)
        a = a_new
    
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    if verbose:
        print(f"    LP done: best={best_score:.8f} after {it} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def subgradient_refine(a, time_limit_s=60.0, verbose=True):
    """
    Subgradient method with momentum and mirror descent.
    Minimizes max(conv(a,a)) subject to a>=0, sum(a)=1.
    """
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    v = np.zeros(n)
    momentum = 0.8
    lr = 0.1 / np.sqrt(n)
    best_lr = lr
    no_improve = 0
    
    it = 0
    while time.monotonic() - start_time < time_limit_s:
        it += 1
        conv = np.convolve(a, a)
        k_star = conv.argmax()
        score = 2.0 * n * conv.max()
        
        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                lr *= 0.5
                if lr < 1e-10:
                    break
                no_improve = 0
        
        if verbose and it % 500 == 0:
            print(f"    SG iter {it:5d}: score={score:.8f}, lr={lr:.6f}")
        
        # Subgradient: g[j] = 2*a[k_star - j] if 0 <= k_star - j < n
        g = np.zeros(n)
        j_lo = max(0, k_star - n + 1)
        j_hi = min(n - 1, k_star)
        for j in range(j_lo, j_hi + 1):
            g[j] = 2.0 * a[k_star - j]
        
        # Mirror descent with exponential update
        v = momentum * v + (1.0 - momentum) * g
        a_new = a * np.exp(-lr * v)
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
    
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    if verbose:
        print(f"    SG done: best={best_score:.8f} after {it} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def local_search_refine(a, time_limit_s=30.0, rng=None, verbose=True):
    """Local search with random multiplicative perturbations."""
    if rng is None:
        rng = np.random.default_rng(42)
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    it = 0
    while time.monotonic() - start_time < time_limit_s:
        it += 1
        width = max(2, int(n / 200))
        start = int(rng.integers(0, max(1, n - width)))
        candidate = a.copy()
        sigma = 0.02 + 0.03 * np.random.exponential(1.0)
        candidate[start:start + width] *= np.exp(rng.normal(0.0, sigma, size=width))
        candidate = normalize_nonneg(candidate)
        score = score_sequence(candidate)
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
    
    if verbose:
        print(f"    LS done: best={best_score:.8f} after {it} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score
