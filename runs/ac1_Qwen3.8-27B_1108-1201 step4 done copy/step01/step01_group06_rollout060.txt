    p = p / S  # normalize to sum = 1
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_obj = np.inf
    best_p = p.copy()
    
    T = T_init
    
    for t in range(n_iter):
        conv = fft_convolve(p, p)
        max_conv = conv.max()
        
        # Smoothed objective
        obj = np.log(np.exp(T * (conv - max_conv)).sum()) / T + max_conv  # LogSumExp
        
        if obj < best_obj:
            best_obj = obj
            best_p = p.copy()
        
        # Gradient
        g = compute_logsumexp_gradient(p, conv, T)
        
        # Adam update
        t_adam = t + 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** t_adam)
        v_hat = v / (1 - beta2 ** t_adam)
        
        p_new = p - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project onto {x >= 0, sum x = 1}
        p_new = np.maximum(p_new, 0)
        s = p_new.sum()
        if s > 0:
            p_new = p_new / s
        else:
            p_new = np.ones(n) / n
        p = p_new
        
        # Anneal T
        T = min(T_max, T_init * (1 + t / 1000))
    
    return best_p, best_obj
