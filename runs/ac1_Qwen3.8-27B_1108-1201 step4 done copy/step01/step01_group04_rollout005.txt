def _compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""

`= (c * a_rev)[j+n-1]`

For the objective `2*n*max_c/(sum_a)^2`:
- If `max_c` scales as `1/n` (flat convolution), objective ≈ 2.
- If we can get `max_c` to scale faster than `1/n`, we get better than 2.
- The theoretical minimum is `2n/(2n-1) ≈ 1`.

For a "random" non-negative sequence, what's the expected peak?
If `a[j]` are i.i.d. with mean `μ` and variance `σ^2`, then `c[k] ≈ (number of terms) * μ^2 + noise`. The number of terms at position k is `min(k+1, n, 2n-1-k)`. At the peak (k ≈ n-1), there are n terms, each with mean `μ^2`. So `E[c[peak]] ≈ n*μ^2`. The standard deviation of each term is `σ^2`, and there are n independent terms (approximately), so `std ≈ σ`.

If `a[j]` are Bernoulli(p) scaled, then `μ = p`, `σ^2 = p(1-p)`. `sum(a) ≈ np`. `max_c ≈ np^2 + O(sqrt(n*p(1-p)))`.
Objective ≈ `2*n*n*p^2/(n*p)^2 = 2`. Same as uniform!

So random sequences don't help. We need a structured approach.

The key insight is that we need to arrange the elements so that the "heavy" elements are positioned where they contribute to positions with fewer pairs, and the "light" elements are where there are many pairs.

For position k, the number of pairs is `min(k+1, n, 2n-1-k)`. This is smallest at the edges (k=0 and k=2n-2, where it's 1) and largest at the center (k=n-1, where it's n).

So we want the products `a[j]*a[i]` (where `j+i=k`) to be small at the center and can be larger at the edges.

One strategy: make `a[j]` large for j near 0 and n-1 (edges) and small for j near n/2 (center). This way, the "heavy" elements pair with few partners at the edge positions, and the "light" elements are at the center where there are many pairs.

Let me try this: `a[j] = 1 + alpha * (1 - |2j - (n-1)|/(n-1))` or something that's peaked at the edges.

Actually, `a[j]` large at edges means `a[0]` and `a[n-1]` are large. But `c[0] = a[0]^2` and `c[2n-2] = a[n-1]^2`, so if `a[0]` is large, `c[0]` is large, which might become the new peak!

So we need to balance: make `a[0]^2` not too large, but also not make the center too large.

This is a trade-off. The optimal solution likely has a specific structure that balances all positions.

Optimizes a non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated gradient descent with two-phase optimization (L2 smoothing then peak reduction),
LP-based refinement, and multi-start strategy.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft

def _next_pow2(x):
    """Next power of 2 >= x."""
    p = 1
    while p < x:
        p *= 2
    return p

def _fft_convolve(a, b=None):
    """Compute full linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = _next_pow2(size)
    A = fft(a, fft_size)
    B = fft(b, fft_size)
    return ifft(A * B).real[:size]

def _compute_conv(a):
    """Compute convolve(a, a) using FFT."""
    return _fft_convolve(a)

def _compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = _compute_conv(a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def _normalize(a, target=None):
    """Normalize to target sum (default sqrt(2n))."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return np.ones_like(a)
    return a * (target / S)

def _grad_l2(a):
    """Gradient of sum(c[k]^2) w.r.t. a, where c = convolve(a,a)."""
    n = len(a)
    c = _compute_conv(a)  # length 2n-1
    a_rev = a[::-1]
    conv_ca = _fft_convolve(c, a_rev)  # length (2n-1) + n - 1 = 3n-2
    grad = 2.0 * conv_ca[n-1 : 2*n-1]
    return grad

def _grad_peak(a, tau=50.0):
    """Soft-max weighted gradient for peak reduction."""
    n = len(a)
    c = _compute_conv(a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w /= w.sum()
    # grad[j] = sum_k w[k] * a[k-j] for valid k-j
    # This is similar to _grad_l2 but with w instead of c
    # grad[j] = sum_k w[k] * a[k-j] = (w * a_rev)[j + n - 1] ... let me verify
    # Actually: grad[j] = sum_{k: 0<=k-j<n} w[k] * a[k-j]
    # Let i = k-j, k = j+i: grad[j] = sum_{i=0}^{n-1} w[j+i] * a[i]
    # This is the same structure as _grad_l2!
    # So grad[j] = 2 * (convolve(w, a_rev))[j+n-1] / 2 ... wait, no factor of 2 here
    # grad[j] = sum_{i=0}^{n-1} w[j+i] * a[i]
    # = (convolve(w, a_rev))[j+n-1] (by the same derivation as before, without the factor of 2)
    
    w_padded = np.zeros(2*n-1)  # w is already length 2n-1
    conv_wa = _fft_convolve(w, a_rev)
    grad = conv_wa[n-1 : 2*n-1]
    return grad

def _grad_hard(a):
    """Hard peak gradient (subgradient at the argmax)."""
    n = len(a)
    c = _compute_conv(a)
    k_star = int(c.argmax())
    grad = np.zeros(n)
    for j in range(n):
        i = k_star - j
        if 0 <= i < n:
            grad[j] = a[i]
    return grad

def _lp_solve(f, M, n_max=2048):
    """Solve LP: max sum(g) s.t. convolve(f,g) <= M, g >= 0."""
    n = len(f)
    if n > n_max:
        return None
    
    c_obj = -np.ones(n)
    # A[m, j] = f[m-j] if 0 <= m-j < n
    A = np.zeros((2*n-1, n))
    for m in range(2*n-1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[m, j] = f[m - j]
    
    b = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

def _optimize_two_phase(a0, deadline, max_phase1=2000, max_phase2=3000):
    """Two-phase optimization: L2 smoothing then peak reduction."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    a = _normalize(a)
    S_t = np.sqrt(2.0 * n)
    
    best = a.copy()
    best_score = _compute_obj(a)
    
    # Phase 1: Minimize ||c||_2^2
    lr = 0.1
    for it in range(max_phase1):
        if time.monotonic() > deadline - 60.0:

"""
Optimizes a non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated two-phase gradient descent (L2 smoothing then soft-peak reduction),
LP-based refinement, structured initializations from load-balancing insights,
and multi-start strategy with reheat mechanism.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft


def _next_pow2(x):
    p = 1
    while p < x:
        p *= 2
    return p


def _fft_convolve(a, b=None):
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fs = _next_pow2(size)
    A = fft(a, fs)
    B = fft(b, fs)
    return ifft(A * B).real[:size]


def _compute_conv(a):
    return _fft_convolve(a)


def _compute_obj(a):
    n = len(a)
    c = _compute_conv(a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def _normalize(a, target=None):
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return np.ones_like(a)
    return a * (target / S)


def _grad_l2(a):
    n = len(a)
    c = _compute_conv(a)
    a_rev = a[::-1]
    conv_ca = _fft_convolve(c, a_rev)
    grad = 2.0 * conv_ca[n - 1:2 * n - 1]
    return grad


def _grad_soft_peak(a, tau):
    n = len(a)
    c = _compute_conv(a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(c)
        w[int(c.argmax())] = 1.0
    else:
        w /= w_sum
    a_rev = a[::-1]
    conv_wa = _fft_convolve(w, a_rev)
    grad = conv_wa[n - 1:2 * n - 1]
    return grad


def _grad_hard_peak(a):
    n = len(a)
    c = _compute_conv(a)
    k_star = int(c.argmax())
    grad = np.zeros(n)
    for j in range(n):
        i = k_star - j
        if 0 <= i < n:
            grad[j] = a[i]
    return grad


def _lp_solve(f, M):
    n = len(f)
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[m, j] = f[m - j]
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

def _make_edge_weighted(n, alpha):
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    dist = np.abs(i - center) / center
    return 1.0 + alpha * dist


def _make_cosine_mod(n, freq, phase):
    i = np.arange(n, dtype=np.float64)
    return 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * i / n + phase)


def _make_bimodal(n, sigma_ratio):
    i = np.arange(n, dtype=np.float64)
    c1 = n * 0.25

    s = n * sigma_ratio
    return np.exp(-0.5 * ((i - c1) / s) ** 2) + np.exp(-0.5 * ((i - c2) / s) ** 2) + 0.05
