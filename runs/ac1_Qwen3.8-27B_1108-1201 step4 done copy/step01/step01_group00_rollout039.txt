"""
Peak-reducing optimization for non-negative step functions minimizing
2n * max(autocorrelation) / (sum)^2.
Uses water-filling for aggressive peak reduction, LP-based direction finding,
multi-restart with diverse initializations, and coordinate-wise local search polish.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score directly."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)


def gradient_at_lag(a, k):
    """Compute gradient of conv[k] w.r.t. a. Result: 2*a[k-m] for valid m."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        g[lo:hi + 1] = 2.0 * a[k - hi:k - lo][::-1]
    return g


def waterfill_step(a, k_star, alpha=1.0):
    """
    Apply water-filling step to reduce conv at lag k_star.
    Moves mass from high-gradient positions to low-gradient positions.
    """
    n = len(a)
    S = np.sum(a)
    w = gradient_at_lag(a, k_star)

    # Water-filling: delta[m] = max(-a[m], nu - w[m])
    # Find nu such that sum(delta) = 0
    # f(nu) = sum max(-a[m], nu - w[m]) is monotonically increasing in nu

    w_min = w.min()
    w_max = w.max()
    a_min = a.min()

    lo_nu = w_min - max(a.max(), 1.0)
    hi_nu = w_max + max(a.max(), 1.0)

    for _ in range(50):
        mid = (lo_nu + hi_nu) / 2.0
        delta = np.maximum(-a, mid - w)
        s = delta.sum()
        if s > 0:
            hi_nu = mid
        else:
            lo_nu = mid

    nu = (lo_nu + hi_nu) / 2.0
    delta = np.maximum(-a, nu - w)
    # Rescale so sum is exactly 0 (bisection should have done this)
    delta_sum = delta.sum()
    if abs(delta_sum) > 1e-10:
        delta = delta - delta_sum / n

    a_new = a + alpha * delta
    a_new = np.maximum(a_new, 0.0)
    new_sum = a_new.sum()
    if new_sum < 1e-10:
        return a.copy()
    a_new = a_new * (S / new_sum)
    return a_new


def lp_direction(a, M, max_iter=2000):
    """
    Solve LP: max sum(b) s.t. b >= 0, conv(a,b)[k] <= M for all k.
    Returns optimal b or None if infeasible.
    """
    n = len(a)
    # Build constraint matrix: A[k, j] = a[k-j] if 0 <= k-j < n
    num_constraints = 2 * n - 1
    A = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[k, lo:hi + 1] = a[hi - lo + 1:hi - lo + 1 + (hi - lo) + 1][::-1]
            # More carefully: A[k, j] = a[k-j] for j in [lo, hi]
            for j in range(lo, hi + 1):
                A[k, j] = a[k - j]

    c = -np.ones(n)  # minimize -sum(b)
    bounds = [(0.0, None)] * n
    b_ub = np.full(num_constraints, M)

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'maxiter': max_iter})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def soft_peak_gradient(a, conv, tau):
    """Compute gradient of soft-max of conv w.r.t. a."""
    n = len(a)
    conv_max = conv.max()
    p = np.exp((conv - conv_max) / tau)
    p = p / p.sum()

    # g[m] = 2 * sum_k p[k] * a[k-m] * 1_{0<=k-m<n}
    # This is a cross-correlation, computable via FFT
    size = 1
    while size < 4 * n:
        size <<= 1
    A_fft = np.fft.rfft(a, size)
    P_fft = np.fft.rfft(p, size)
    g = 2.0 * np.fft.irfft(A_fft * np.conj(P_fft), size)[:n]
    return g

        elif r < 0.6:
            # Block reduction at peak contributors
            width = max(1, int(rng.integers(1, max(2, n // 16))))
            # Find where gradient is highest
            g_pos = np.maximum(g, 0)
            start = max(0, min(n - width, int(np.argmax(g_pos[:n-width+1]))))
            candidate = current.copy()
            scale = np.exp(rng.uniform(-0.08, -0.02, width))
            candidate[start:start + width] *= scale
            # Compensate elsewhere
            other = np.setdiff1d(np.arange(n), np.arange(start, start + width))
            if len(other) > 0:
                n_comp = min(8, len(other))
                comp_idx = rng.choice(other, size=n_comp, replace=False)
                comp_scale = np.exp(rng.uniform(0.02, 0.08, n_comp))
                candidate[comp_idx] *= comp_scale
            candidate = np.maximum(candidate, 0.0)
            candidate = candidate * S / candidate.sum()

        elif r < 0.8:
            # Symmetrize toward peak lag
            # If k_star ≈ n-1, symmetrize around center
            if abs(k_star - (n - 1)) <= n // 8:
                candidate = 0.5 * (current + current[::-1])
                candidate *= np.exp(rng.normal(0, 0.01, n))
            else:
                candidate = current * np.exp(rng.normal(0, 0.02, n))
            candidate = np.maximum(candidate, 0.0)
            candidate = candidate * S / candidate.sum()

        else:
            # Small random perturbation
            candidate = current * np.exp(rng.normal(0, 0.015, n))
            candidate = np.maximum(candidate, 0.0)
            candidate = candidate * S / candidate.sum()

        new_score = compute_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            no_improve = 0
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            current = candidate
            current_score = new_score
            no_improve += 1
        else:
            no_improve += 1

        temperature *= 0.9999
        if no_improve > 200:
            no_improve = 0
            temperature = max(temperature, 0.0005)

    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search combining water-filling peak reduction, LP direction
    finding, and local search polishing across multiple initializations
    and sequence lengths.
    """
    t_global = time.monotonic()
    time_budget = max(10.0, float(budget_s) - 25.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check for provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = compute_score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()

    # Define search configurations
    # (length, list of initialization functions)
    def init_chirp(n, r):
        return make_chirp(n, 0.3 + 0.3 * r.random(), r, freq_scale=r.uniform(0.5, 2.0))

    def init_sine(n, r):
        f = int(r.integers(1, max(2, n // 3)))
        return make_sine(n, f, 0.2 + 0.3 * r.random(), r)

    def init_multi(n, r):
        return make_multi_sine(n, r)

    def init_random(n, r):
        return make_random(n, r, spread=0.2 + 0.4 * r.random())

    def init_provided(n, r):
        if start_seq is not None:
            if len(start_seq) == n:
                return start_seq.copy()
            x_old = np.linspace(0, 1, len(start_seq))
            x_new = np.linspace(0, 1, n)
            return np.interp(x_new, x_old, start_seq)
        return make_chirp(n, 0.4, r)

    # Phase 1: Quick screening of different lengths and inits
    print("\n=== Phase 1: Screening ===")
    configs = [
        (256, [init_chirp, init_sine, init_multi, init_random, init_provided]),
        (320, [init_chirp, init_sine, init_random]),
        (384, [init_chirp, init_sine, init_multi]),
        (448, [init_chirp, init_sine]),
        (512, [init_chirp, init_sine, init_random]),
        (640, [init_chirp]),
        (768, [init_chirp]),
        (1024, [init_chirp]),
    ]

    # Allocate time: give more time to smaller sequences (faster iterations)
    # Quick pass: 5s per config for screening
    quick_time = 4.0

    candidates = []  # (a, n, init_name)
    for n, inits in configs:
        if time.monotonic() > deadline - 60:
            break
        for init_fn in inits:
            if time.monotonic() > deadline - 30:
                break
            a0 = init_fn(n, rng)
            a0 = np.maximum(a0, 0.01)
            S0 = np.sum(a0)
            a0 = a0 * np.sqrt(2.0 * n) / S0
            s0 = compute_score(a0)
            candidates.append((a0, n, s0))
            if s0 < best_score:
                best_score = s0
                best_seq = a0.copy()

    # Sort candidates by initial score
    candidates.sort(key=lambda x: x[2])
    print(f"  Best initial: score={candidates[0][2]:.6f}, n={candidates[0][1]}")
    print(f"  Overall best after screening: {best_score:.6f}")

    # Phase 2: Focused optimization on top candidates
    print("\n=== Phase 2: Focused optimization ===")
    # Allocate remaining time to top candidates
    remaining = deadline - time.monotonic()
    top_k = min(len(candidates), 8)

    # Weighted time allocation: better initial scores get more time
    scores = np.array([c[2] for c in candidates[:top_k]])
    weights = 1.0 / (scores - scores.min() + 0.01)
    weights = weights / weights.sum()

    # But also cap: no single candidate gets more than 40% of time
    max_weight = 0.4
    weights = np.minimum(weights, max_weight)
    weights = weights / weights.sum()

    for i in range(top_k):
        if time.monotonic() > deadline - 30:
            break
        a_init, n, _ = candidates[i]
        time_alloc = remaining * weights[i]
        time_alloc = max(5.0, time_alloc)  # minimum 5s per candidate
        time_alloc = min(time_alloc, deadline - time.monotonic() - 20)
        if time_alloc < 5:
            break

        use_lp = n <= 512  # LP is feasible for smaller n

        print(f"  Optimizing candidate {i+1}/{top_k}: n={n}, init_score={compute_score(a_init):.6f}, time={time_alloc:.1f}s")

        a_opt, s_opt = optimize_waterfill(a_init, time_alloc, rng, use_lp=use_lp)
        if s_opt < best_score:
            best_score = s_opt
            best_seq = a_opt.copy()
            print(f"    *** New best: {best_score:.6f} (n={n}) ***")
        else:
            print(f"    Best for this candidate: {s_opt:.6f}")

        # Also try to update candidates with the optimized solution
        if s_opt < candidates[i][2]:
            candidates[i] = (a_opt, n, s_opt)

    # Phase 3: Polish the overall best
    print(f"\n=== Phase 3: Polish ===")
    if best_seq is not None:
        polish_time = min(120.0, max(10.0, deadline - time.monotonic()))
        print(f"  Polishing best (score={best_score:.6f}, n={len(best_seq)}), time={polish_time:.1f}s")
        polished, polish_score = polish_local_search(best_seq, polish_time, rng)
        if polish_score < best_score:
            best_score = polish_score
            best_seq = polished.copy()
            print(f"  After polish: {best_score:.6f}")

    # Phase 4: If time remains, try one more round on slightly modified best
    if time.monotonic() < deadline - 20 and best_seq is not None:
        remaining2 = deadline - time.monotonic() - 10
        if remaining2 > 30:
            print(f"\n=== Phase 4: Additional refinement ===")
            n = len(best_seq)
            # Try a few variants
            for variant in range(3):
                if time.monotonic() > deadline - 15:
                    break
                a_var = best_seq * np.exp(rng.normal(0, 0.03, n))
                a_var = np.maximum(a_var, 0.01)
                a_var = a_var * np.sqrt(2.0 * n) / a_var.sum()
                t_var = min(40.0, remaining2 / 2.0)
                a_opt, s_opt = optimize_waterfill(a_var, t_var, rng, use_lp=(n <= 512))
                if s_opt < best_score:
                    best_score = s_opt
                    best_seq = a_opt.copy()
                    print(f"  Variant {variant+1}: new best {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")

        if final_score < best_score:
            best_score = final_score
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
