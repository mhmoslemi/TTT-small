def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    mu = np.sort(v)[::-1]
    cumulative = np.cumsum(mu)
    rho = np.where(mu * np.arange(1, n+1) > (cumulative - S))[0][-1]
    theta = (cumulative[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(np.max(c))
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def project_onto_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    sorted_v = np.sort(v)[::-1]
    cumsum = np.cumsum(sorted_v)
    # Find rho
    arange = np.arange(1, n + 1, dtype=np.float64)
    cond = sorted_v * arange > (cumsum - S)
    if not np.any(cond):
        return np.full(n, S / n)
    rho = np.where(cond)[0][-1]
    theta = (cumsum[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax_grad(a, tau):
    """Gradient of sum_k exp(tau*c[k]) / (tau * (2n-1)) w.r.t. a."""
    n = len(a)
    c = np.convolve(a, a)
    w = np.exp(tau * (c - c.max()))
    w /= w.sum()

        m_hi = min(n, 2*n-1-i)
        if m_hi > 0:

    return grad
