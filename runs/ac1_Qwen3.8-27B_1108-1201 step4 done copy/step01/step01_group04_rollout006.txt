    if S < 1e-12:

    """Gradient of soft-max approximation to peak of convolve(a,a)."""
    n = len(a)
    c = np.convolve(a, a)
    c_max = c.max()

def grad_topk(a, top_k=5):
    """Gradient using average of top-k peak positions."""
    n = len(a)
    c = np.convolve(a, a)
    top_idx = np.argsort(c)[-top_k:]
    grad = np.zeros(n)
    for k in top_idx:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi >= lo:
            grad[lo:hi + 1] += 2.0 * a[k - np.arange(lo, hi + 1)]
    grad /= top_k
    return grad


def adam_optimize(a0, deadline, max_iters=3000, lr=0.1, tau_init=3.0, tau_end=40.0,
                  use_l2=False, top_k=1):
    """Adam optimization with adaptive soft-max or L2-conv gradient."""
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0, 1000)
    if a.sum() < 1e-6:
        a = np.ones(n)
    a = normalize_seq(a)

    best = a.copy()
    best_score = compute_obj(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    S_t = np.sqrt(2.0 * n)

    for it in range(max_iters):
        if time.monotonic() > deadline - 3.0:
            break

        progress = min(1.0, (it + 1) / max_iters)
        tau = tau_init + (tau_end - tau_init) * progress

        if use_l2:
            grad = grad_l2conv(a)
        elif top_k > 1:
            grad = grad_topk(a, top_k)
        else:
            grad = grad_softmax(a, tau)

        # Project out mean (sum constraint)
        grad -= grad.mean()

        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        a = normalize_seq(a_new, S_t)

        if (it + 1) % 30 == 0 or it == 0:
            score = compute_obj(a)
            if score < best_score:
                best_score = score
                best = a.copy()

    return best, best_score


def lp_refine_step(a, M):
    """One LP refinement step: find g maximizing sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    if n > 1500:
        return None
    c_obj = -np.ones(n)
    n_constraints = 2 * n - 1
    A = np.zeros((n_constraints, n))
    for m in range(n_constraints):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        if hi >= lo:
            A[m, lo:hi + 1] = a[m - np.arange(lo, hi + 1)]
    b = np.full(n_constraints, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def lp_optimize(a0, deadline, max_iters=50):
    """LP-based alternating optimization."""
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0, 1000)
    if a.sum() < 1e-6:
        a = np.ones(n)
    a = normalize_seq(a)
    best = a.copy()
    best_score = compute_obj(a)
    S_t = np.sqrt(2.0 * n)

    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break

        c = np.convolve(a, a)
        M = float(c.max())

        g = lp_refine_step(a, M)
        if g is None:
            # Try with slightly relaxed M
            g = lp_refine_step(a, M * 1.01)
            if g is None:
                break

        S_g = g.sum()
        if S_g < 1e-10:
            break

        # Try blending
        improved = False
        for t_blend in [0.9, 0.7, 0.5, 0.3, 0.15]:
            a_new = (1.0 - t_blend) * a + t_blend * g
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_seq(a_new, S_t)
            s_new = compute_obj(a_new)
            if s_new < best_score:
                a = a_new
                best = a.copy()
                best_score = s_new
                improved = True
                break

        if not improved:
            # Small perturbation to escape
            rng_local = np.random.default_rng(it + 12345)
            a_pert = a + 0.05 * rng_local.standard_normal(n)
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = normalize_seq(a_pert, S_t)
            s_pert = compute_obj(a_pert)
            if s_pert < best_score:
                a = a_pert
                best = a.copy()
                best_score = s_pert
            else:
                # Try gradient step
                grad = grad_softmax(a, 10.0)
                grad -= grad.mean()
                a_new = a - 0.05 * grad
                a_new = np.maximum(a_new, 0.0)
                a_new = normalize_seq(a_new, S_t)
                s_new = compute_obj(a_new)
                if s_new < best_score:
                    a = a_new
                    best = a.copy()
                    best_score = s_new
                else:
                    break

    return best, best_score


def simulated_anneal(a0, deadline, n_restarts=200, rng=None):
    """Simulated annealing to escape local minima."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0, 1000)
    a = normalize_seq(a)
    best = a.copy()
    best_score = compute_obj(a)
    S_t = np.sqrt(2.0 * n)

    T = 0.05
    for it in range(n_restarts):
        if time.monotonic() > deadline - 5.0:
            break
        T *= 0.97

        # Random perturbation
        scale = 0.1 + 0.3 * T
        a_pert = a + scale * rng.standard_normal(n)
        a_pert = np.maximum(a_pert, 0.0)
        a_pert = normalize_seq(a_pert, S_t)
        s_pert = compute_obj(a_pert)

        if s_pert < best_score or rng.random() < np.exp(-(s_pert - best_score) / (T + 1e-10)):
            a = a_pert
            if s_pert < best_score:
                best_score = s_pert
                best = a.copy()

    return best, best_score


def generate_starts(rng, maxlen=2048):
    """Generate diverse starting sequences."""
    starts = []

    # Random starts
    for n in [128, 256, 384, 512, 768, 1024]:
        starts.append(rng.random(n) * 0.5 + 0.25)
        starts.append(rng.random(n) ** 2)  # More peaked

    # Structured starts
    for n in [256, 512, 1024]:
        i = np.arange(n, dtype=np.float64)
        # Chirp
        starts.append(0.5 + 0.5 * np.cos(2 * np.pi * i * 3.0 / n))
        # Raised cosine
        starts.append(0.5 + 0.5 * np.cos(2 * np.pi * i / n))
        # Double chirp
        starts.append(0.3 + 0.7 * np.abs(np.sin(2 * np.pi * i * 2.0 / n)))
        # Gaussian
        center = n / 2.0
        starts.append(np.exp(-0.5 * ((i - center) / (n * 0.2)) ** 2))
        # Binary-like (smoothed)
        starts.append((rng.random(n) > 0.5).astype(float) * 0.8 + 0.2)
        # Periodic modulation
        starts.append(0.5 + 0.3 * np.sin(2 * np.pi * i * 7.0 / n) + 0.2 * np.sin(2 * np.pi * i * 13.0 / n))

    # Near-uniform with noise
    for n in [512, 1024, 1536]:
        starts.append(1.0 + 0.2 * rng.standard_normal(n))

    # Length sweep with random
    for n in [192, 320, 448, 640, 896, 1280, 1792]:
        starts.append(rng.random(n) * 0.5 + 0.25)

    return starts


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Uses Adam with soft-max/L2 gradients, LP refinement,
    and simulated annealing to find near-optimal non-negative sequences.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = float('inf')

    def update_best(seq, score):
        nonlocal best_seq, best_score
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  [UPDATE] score={score:.6f}, n={len(seq)}, t={time.monotonic()-t_start:.1f}s")

    # Try to use height_sequence_1 if available
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 8 and np.all(np.isfinite(gs)):
            gs = np.maximum(gs, 0)
            for L in [256, 384, 512, 768, min(1024, len(gs)), min(2048, len(gs))]:
                if L <= len(gs):
                    candidate = gs[:L].copy()
                    # Quick Adam optimization
                    seq, score = adam_optimize(candidate, t_start + 120,
                                               max_iters=800, lr=0.08,
                                               tau_init=5, tau_end=35)
                    update_best(seq, score)
    except NameError:
        pass

    # Generate diverse starts
    starts = generate_starts(rng)
    print(f"Generated {len(starts)} starting sequences")

    # Phase 1: Quick Adam on all starts
    phase1_deadline = t_start + 0.55 * (float(budget_s) - 10.0)
    print(f"Phase 1 deadline: {phase1_deadline - t_start:.0f}s")

    # Sort by length (shorter first for quick wins)
    start_scores = []
    for idx, a0 in enumerate(starts):
        if time.monotonic() > phase1_deadline - 5.0:
            break
        n = len(a0)
        # Quick Adam
        seq, score = adam_optimize(a0, phase1_deadline - 5.0,
                                   max_iters=500, lr=0.08,
                                   tau_init=3, tau_end=30)
        start_scores.append((score, n, idx, seq))
        update_best(seq, score)
        if (idx + 1) % 5 == 0:
            print(f"  Phase 1: {idx+1}/{len(starts)} done, best={best_score:.6f}")

    # Phase 2: Deeper optimization on top candidates
    phase2_deadline = t_start + 0.80 * (float(budget_s) - 10.0)
    print(f"Phase 2 deadline: {phase2_deadline - t_start:.0f}s")

    start_scores.sort(key=lambda x: x[0])
    top_k = min(10, len(start_scores))

    for i in range(top_k):
        if time.monotonic() > phase2_deadline - 10.0:
            break
        score, n, idx, seq = start_scores[i]
        print(f"  Phase 2: refining candidate {i} (score={score:.6f}, n={n})")

        # Deeper Adam with L2 gradient
        seq2, score2 = adam_optimize(seq, phase2_deadline - 10.0,
                                     max_iters=2000, lr=0.05,
                                     use_l2=True)
        update_best(seq2, score2)

        # Adam with top-k gradient
        seq3, score3 = adam_optimize(seq, phase2_deadline - 10.0,
                                     max_iters=1500, lr=0.05,
                                     top_k=5)
        update_best(seq3, score3)

        # Adam with soft-max (higher tau)
        seq4, score4 = adam_optimize(seq, phase2_deadline - 10.0,
                                     max_iters=1500, lr=0.04,
                                     tau_init=10, tau_end=60)
        update_best(seq4, score4)

    # Phase 3: LP refinement on best
    phase3_deadline = t_start + 0.92 * (float(budget_s) - 10.0)
    print(f"Phase 3 deadline: {phase3_deadline - t_start:.0f}s")

    if best_seq is not None and len(best_seq) <= 1024:
        print(f"  LP refinement (n={len(best_seq)}, score={best_score:.6f})")
        seq_lp, score_lp = lp_optimize(best_seq, phase3_deadline, max_iters=30)
        update_best(seq_lp, score_lp)

    # Also try LP on a few other candidates
    for i in range(1, min(4, top_k)):
        if time.monotonic() > phase3_deadline - 10.0:
            break
        _, n, idx, seq = start_scores[i]
        if n <= 768:
            seq_lp, score_lp = lp_optimize(seq, phase3_deadline - 5.0, max_iters=20)
            update_best(seq_lp, score_lp)

    # Phase 4: Simulated annealing + final polish
    final_deadline = deadline
    print(f"Phase 4: final refinement, deadline in {final_deadline - time.monotonic():.0f}s")

    if best_seq is not None:
        # Simulated annealing
        seq_sa, score_sa = simulated_anneal(best_seq, final_deadline - 30.0,
                                            n_restarts=500, rng=rng)
        update_best(seq_sa, score_sa)

        # Short Adam polish
        if time.monotonic() < final_deadline - 10.0:
            seq_pol, score_pol = adam_optimize(best_seq, final_deadline - 5.0,
                                               max_iters=1000, lr=0.02,
                                               tau_init=20, tau_end=80)
            update_best(seq_pol, score_pol)
            if score_pol < best_score:
                best_score = score_pol
                best_seq = seq_pol.copy()

        # Final LP if time allows and n is small enough
        if len(best_seq) <= 512 and time.monotonic() < final_deadline - 10.0:
            seq_final, score_final = lp_optimize(best_seq, final_deadline - 3.0, max_iters=10)
            update_best(seq_final, score_final)

    # Final validation
    if best_seq is not None:
        final_score = compute_obj(best_seq)
        print(f"Final result: n={len(best_seq)}, score={final_score:.6f}")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence: {eval_score:.6f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in best_seq]

    # Fallback
    print("Falling back to uniform sequence.")
    return [1.0] * 256
