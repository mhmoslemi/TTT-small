```python
def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def score_fast(a):

    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)

def project_simplex(v):

        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
```

```python
from scipy.optimize import linprog

def solve_lp(f, M):
    """Solve: max sum(b) s.t. conv(f,b) <= M, b >= 0."""
    n = len(f)
    # c[k] = sum_i f[k-i] * b[i] for k = 0, ..., 2n-2
    # C[k, i] = f[k-i] if 0 <= k-i < n, else 0
    C = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for i in range(max(0, k-n+1), min(n, k+1)):
            C[k, i] = f[k-i]
    
    # Maximize sum(b) = minimize -sum(b)
    c_obj = -np.ones(n)
    # Constraints: C b <= M
    A_ub = C
    b_ub = np.full(2*n-1, M)
    # b >= 0 (bounds)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    else:
        return None
```

```python
    """LP-based optimization for minimizing the peak of conv(f,f)."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    S = np.sqrt(2.0 * n)
    f = f / np.sum(f) * S  # Normalize
    
    t0 = time.monotonic()
    best_f = f.copy()
    best_score = 2.0 * n * np.max(fft_convolve(f, f)) / (S * S)
    
    for iteration in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_f = fft_convolve(f, f)
        M = np.max(conv_f)
        
        # Solve LP
        b = solve_lp(f, M)
        if b is None:
            break
        
        # Rescale b to have sum = S
        sum_b = np.sum(b)
        if sum_b < 1e-10:
            break
        b_rescaled = b / sum_b * S
        
        # Line search for step size
        best_t = 0
        best_t_score = np.inf
        for t in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:

            f_new = np.clip(f_new, 0, None)
            s_new = np.sum(f_new)
            if s_new < 0.01:
                continue
            score_new = 2.0 * n * np.max(fft_convolve(f_new, f_new)) / (s_new * s_new)
            if score_new < best_t_score:
                best_t_score = score_new
                best_t = t
                best_f_new = f_new.copy()
        
        if best_t > 0 and best_t_score < best_score:
            f = best_f_new
            best_score = best_t_score
            best_f = f.copy()
        else:

            f = f * np.exp(rng.normal(0, 0.01, n))
            f = np.clip(f, 0, None)
            s = np.sum(f)
            if s > 0.01:
                f = f / s * S
    
    return best_f, best_score
```
