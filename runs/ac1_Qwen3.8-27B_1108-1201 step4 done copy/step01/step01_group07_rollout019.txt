    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (a_sum ** 2)


def compute_normalized_obj(a):

def compute_adam_gradient(a, conv, w, n):
    """
    Gradient of 2n * soft_max(conv) w.r.t. log-space variables x where a=softmax(x).
    grad[i] = 4n * a[i] * (h[i] - h2)
    where h[i] = sum_k w[k]*a[k-i], h2 = sum_k w[k]*conv[k]
    """

    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:

        j_end = min(n - 1, k)
        h[j_start:j_end + 1] += wk * a[k - j_end:k - j_start + 1][::-1]
    h2 = np.dot(w, conv)
    return 4.0 * n * a * (h - h2)


def adam_update(state, grad, lr, t, beta1=0.9, beta2=0.999, eps=1e-8):

    m = beta1 * state['m'] + (1 - beta1) * grad
    v = beta2 * state['v'] + (1 - beta2) * (grad * grad)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    update = lr * m_hat / (np.sqrt(v_hat) + eps)
    state['m'] = m
    state['v'] = v
    return update

        score = 2.0 * n * obj_norm  # Since sum=1, score = 2n*max_conv
        if np.isfinite(score):
            scan_results.append((score, n, name, a_opt))
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Scan: n={n}, {name}, score={score:.6f}", flush=True)

    # Phase 2: Deep optimization
    print(f"Phase 2: Deep optimization. Best: {best_score:.6f}", flush=True)
    scan_results.sort(key=lambda x: x[0])

    # Select top configs with diversity
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        if len(top_configs) < 10:
            if n not in seen or len(top_configs) < 5:
                top_configs.append((n, name, a_opt))
                seen.add(n)
        if len(top_configs) >= 7:
            break

    # Always include current best
    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'best', best_sequence.copy()))

    remaining = deadline - time.monotonic() - 100
    time_per_deep = max(20.0, remaining / max(1, len(top_configs)))

    for i, (n, name, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 80:
            break
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 80)
        if t_alloc < 10:
            break

        # Try two different parameter settings
        for bs, be, lr_val in [(12, 350, 0.006), (20, 500, 0.004)]:
            if time.monotonic() > deadline - 80:
                break
            t_single = t_alloc / 2.0
            a_opt, obj_norm = optimize_adam(a0, time_limit=t_single,
                                             beta_start=bs, beta_end=be,
                                             lr=lr_val, pert_every=150,
                                             pert_scale=0.1)
            score = 2.0 * n * obj_norm
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Deep: n={n}, {name}, bs={bs}, score={score:.6f}", flush=True)

    # Phase 3: Final polish with LP
    print(f"Phase 3: Final polish. Best: {best_score:.6f}", flush=True)
    if best_sequence is not None:
        n_best = len(best_sequence)

        # First: extended Adam optimization
        remaining = deadline - time.monotonic() - 20
        t_adam = min(remaining * 0.5, 300.0)
        if t_adam > 10:
            a_opt, obj_norm = optimize_adam(best_sequence, time_limit=t_adam,
                                             beta_start=50, beta_end=600,
                                             lr=0.003, pert_every=200,
                                             pert_scale=0.06)
            score = 2.0 * n_best * obj_norm
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Adam polish: score={score:.6f}", flush=True)

        # LP polish (only for moderate sizes)
        remaining = deadline - time.monotonic() - 10
        if remaining > 30 and n_best <= 512:
            t_lp = min(remaining * 0.5, 180.0)
            a_lp, score_lp = lp_polish(best_sequence, time_limit=t_lp,
                                        max_iters=60)
            if np.isfinite(score_lp) and score_lp < best_score:
                best_score = score_lp
                best_sequence = a_lp.copy()
                print(f"  LP polish: score={score_lp:.6f}", flush=True)

        # One more Adam pass after LP
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            a_opt, obj_norm = optimize_adam(best_sequence, time_limit=remaining,
                                             beta_start=80, beta_end=800,
                                             lr=0.002, pert_every=300,
                                             pert_scale=0.04)
            score = 2.0 * len(a_opt) * obj_norm
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final Adam: score={score:.6f}", flush=True)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Verify with evaluate_sequence
    final_score = float(evaluate_sequence([float(x) for x in best_sequence]))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"FINAL SCORE: {best_score:.6f}, length={len(best_sequence)}", flush=True)
    return [float(x) for x in best_sequence]
