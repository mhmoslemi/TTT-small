```python
import numpy as np
import time

def _fft_convolve(a, b):
    """Fast convolution using FFT."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - target))[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1)
    return np.maximum(v - theta, 0.0)
```

```python
import numpy as np

def _fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    out_len = len(a) + len(b) - 1
    size = 1
    while size < out_len:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:out_len]


def _project_simplex(v, target=1.0):
    """Project v onto the probability simplex {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v, axis=0)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.full(n, target / n)
    rho = rho_idx[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_b(a):
    """Compute the autocorrelation b = convolve(a, a) using FFT."""
    return _fft_convolve(a, a)


def _smoothed_grad(a, b, t):
    """
    Compute gradient of (1/t)*log(sum_k exp(t*b[k])) w.r.t. a.
    Uses FFT for efficiency.
    """
    n = len(a)
    # Compute weights w_k = exp(t*b[k]) / sum(exp(t*b[k]))
    # Use log-sum-exp trick for numerical stability
    t_b = t * b
    t_b_max = np.max(t_b)
    exp_tb = np.exp(t_b - t_b_max)
    w = exp_tb / np.sum(exp_tb)
    
    # grad[m] = 2 * sum_j w[m+j] * a[j] for j=0..n-1
    # This is 2 * correlate(w, a, 'valid')
    # Using FFT: convolve w with reversed a, take appropriate slice
    a_rev = a[::-1]
    # (w * a_rev)[k] = sum_i w[i] * a_rev[k-i]

    conv_result = _fft_convolve(w, a_rev)
    # conv_result has length (2n-1) + n - 1 = 3n-2
    # We need indices n-1 to 2n-2
    grad = 2.0 * conv_result[n-1:2*n-1]
    return grad


def _l2_grad(a, b):
    """Compute gradient of ||b||_2^2 = sum_k b[k]^2 w.r.t. a."""
    n = len(a)

    # = 4 * correlate(b, a, 'valid')
    a_rev = a[::-1]
    conv_result = _fft_convolve(b, a_rev)
    grad = 4.0 * conv_result[n-1:2*n-1]
    return grad

    return 2.0 * n * np.max(b)
```

```python
Multi-phase optimization for minimizing the normalized autocorrelation peak
of a non-negative step function sequence.

Strategy:
- Phase 1: Smooth L2 surrogate with Nesterov-accelerated gradient descent
- Phase 2: Log-sum-exp surrogate with adaptive temperature (FISTA-style)
- Phase 3: Subgradient descent on the actual max objective
- FFT-based convolution and gradient computation for O(n log n) efficiency
- Multi-start with diverse initializations including theoretically-motivated shapes

import numpy as np

def _fft_convolve(a, b):
    """Linear convolution via FFT."""
    out_len = len(a) + len(b) - 1
    size = 1
    while size < out_len:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:out_len]

def _project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.full(n, target / n)
    rho = idx[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _autocorr(a):
    """Compute b = convolve(a, a) via FFT."""
    return _fft_convolve(a, a)

def _grad_broadcast(weights, a):
    """
    Compute grad[m] = 2 * sum_j weights[m+j] * a[j]
    where weights has length 2n-1 and a has length n.
    Uses FFT: convolve(weights, a_reversed), slice [n-1 : 2n-1].
    """
    n = len(a)
    a_rev = a[::-1]
    conv = _fft_convolve(weights, a_rev)
    return 2.0 * conv[n-1 : 2*n-1]

def _score(a):
    """Compute 2*n*max(b) for a with sum=1."""
    n = len(a)
    b = _autocorr(a)
    return 2.0 * n * np.max(b)

def _optimize_l2(a, deadline, rng, eta0=None, momentum=0.85):

    n = len(a)
    a = a.copy()
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    else:

    if eta0 is None:

    best_score = _score(best)
    eta = eta0
    v = np.zeros(n)
    stuck = 0
    
    while time.monotonic() < deadline:
        b = _autocorr(a)
        obj = np.sum(b * b)
        cur_score = 2.0 * n * np.max(b)
        
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 200:

                a = a + 0.02 * rng.standard_normal(n)
                a = _project_simplex(a, 1.0)
                v = np.zeros(n)
                stuck = 0
        
        # grad of ||b||^2 = 4 * correlate(b, a)
        grad = _grad_broadcast(b, a)
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:

        a = a - eta * v
        a = _project_simplex(a, 1.0)

            eta *= 0.95
    
    return best, best_score

def _optimize_lse(a, deadline, rng, t0=10.0, t1=300.0, eta0=None, momentum=0.7):
    """Minimize (1/t)*log(sum exp(t*b)) with adaptive t."""
    n = len(a)
    a = a.copy()

        a /= S
    else:
        a = np.ones(n) / n

        eta0 = 0.3 / n
    
    best = a.copy()
    best_score = _score(best)
    eta = eta0
    t = t0
    v = np.zeros(n)
    stuck = 0
    n_iters = 0
    
    while time.monotonic() < deadline:
        b = _autocorr(a)
        b_max = np.max(b)
        cur_score = 2.0 * n * b_max
        
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        n_iters += 1

        t = t0 + (t1 - t0) * progress

        exp_tb = np.exp(tb_shift)

def _optimize_subgrad(a, deadline, rng, M=30, eta0=None):
    """Subgradient on actual max with FFT-accelerated gradient."""
    n = len(a)
    a = a.copy()
    S = np.sum(a)

        a /= S
    else:
        a = np.ones(n) / n

    best_score = _score(best)
    eta = eta0

        b = _autocorr(a)
        b_max = np.max(b)
        cur_score = 2.0 * n * b_max
        
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best = a.copy()
        
        # Top-M peaks with softmax weights
        if M >= len(b):
            w = np.ones(len(b)) / len(b)
        else:
```

"""
Multi-phase optimizer for minimizing the normalized autocorrelation peak of a
non-negative sequence. Uses FFT-accelerated gradient computation, smooth
surrogate objectives (L2 norm and log-sum-exp with adaptive temperature),
and final subgradient polishing. Multi-start with theoretically-motivated
diverse initializations.
"""

import numpy as np
import time


def _fft_convolve(a, b):
    """Linear convolution via FFT."""
    out_len = len(a) + len(b) - 1
    size = 1
    while size < out_len:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:out_len]


def _project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.full(n, target / n)
    rho = idx[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _autocorr(a):
    """Compute b = convolve(a, a) via FFT."""
    return _fft_convolve(a, a)


def _grad_broadcast(weights, a):
    """
    Compute grad[m] = 2 * sum_j weights[m+j] * a[j]
    where weights has length 2n-1 and a has length n.
    Uses FFT: convolve(weights, a_reversed)[n-1 : 2n-1].
    """
    n = len(a)
    a_rev = a[::-1]
    conv = _fft_convolve(weights, a_rev)
    return 2.0 * conv[n - 1:2 * n - 1]


def _score(a):
    """Compute 2*n*max(b) for a with sum=1."""
    n = len(a)
    b = _autocorr(a)
    return 2.0 * n * np.max(b)


def _optimize_l2(a, deadline, rng, eta0=None):
    """Minimize ||b||^2 with momentum gradient descent + simplex projection."""
    n = len(a)
    a = a.copy()
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    else:
        a = np.ones(n) / n
    if eta0 is None:
        eta0 = 1.0 / n

    best = a.copy()
    best_score = _score(best)
    eta = eta0
    v = np.zeros(n)
    stuck = 0
    momentum = 0.85

    while time.monotonic() < deadline:
        b = _autocorr(a)
        cur_score = 2.0 * n * np.max(b)

        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best = a.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 300:
                a = a + 0.03 * rng.standard_normal(n)
                a = _project_simplex(a, 1.0)
                v = np.zeros(n)
                stuck = 0

        # grad of ||b||^2 w.r.t. a: grad[m] = 4*sum_k b[k]*a[k-m]
        grad = _grad_broadcast(b, a) * 2.0
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break

        v = momentum * v + (1.0 - momentum) * (grad / gn)
        a = a - eta * v
        a = _project_simplex(a, 1.0)

        if stuck % 100 == 0 and stuck > 0:
            eta *= 0.97

    return best, best_score


def _optimize_lse(a, deadline, rng, t0=10.0, t1=500.0, eta0=None):
    """Minimize (1/t)*log(sum exp(t*b)) with adaptive temperature and momentum."""
    n = len(a)
    a = a.copy()
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    else:
        a = np.ones(n) / n
    if eta0 is None:
        eta0 = 0.5 / n

    best = a.copy()
    best_score = _score(best)
    eta = eta0
    t = t0
    v = np.zeros(n)
    stuck = 0
    n_iters = 0
    momentum = 0.75

    while time.monotonic() < deadline:
        b = _autocorr(a)
        b_max = np.max(b)
        cur_score = 2.0 * n * b_max

        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best = a.copy()
            stuck = 0
        else:
            stuck += 1

        n_iters += 1

        # Ramp temperature
        frac = min(1.0, n_iters / 8000.0)
        t = t0 + (t1 - t0) * frac

        # Weights via log-sum-exp
        tb = t * b
        tb_shift = tb - np.max(tb)
        exp_tb = np.exp(tb_shift)
        w = exp_tb / np.sum(exp_tb)

        grad = _grad_broadcast(w, a)
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break

        # Effective step decreases as t increases (steeper landscape)
        eff_eta = eta * t0 / max(t, t0)

        v = momentum * v + (1.0 - momentum) * (grad / gn)
        a = a - eff_eta * v
        a = _project_simplex(a, 1.0)

        if stuck > 150:
            eff_eta_check = eta * t0 / max(t, t0)
            if eff_eta_check > 1e-6:
                eta *= 0.95
            if stuck > 600:
                a = a + 0.015 * rng.standard_normal(n)
                a = _project_simplex(a, 1.0)
                v = np.zeros(n)
                stuck = 200

    return best, best_score


def _optimize_subgrad(a, deadline, rng, M=30, eta0=None):
    """Subgradient on actual max with FFT-accelerated gradient."""
    n = len(a)
    a = a.copy()
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    else:
        a = np.ones(n) / n
    if eta0 is None:
        eta0 = 0.15 / np.sqrt(n)

    best = a.copy()
    best_score = _score(best)
    eta = eta0

    while time.monotonic() < deadline:
        b = _autocorr(a)
        b_max = np.max(b)
        cur_score = 2.0 * n * b_max

        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best = a.copy()

        # Top-M peaks with softmax weights
        if M >= len(b):
            w = np.ones(len(b)) / len(b)
        else:
            top_idx = np.argpartition(b, -M)[-M:]
            top_b = b[top_idx]
            w = np.zeros(len(b))
            sw = np.exp(-6.0 * (b_max - top_b) / max(b_max, 1e-15))
            w[top_idx] = sw / np.sum(sw)

        grad = _grad_broadcast(w, a)
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break

        a = a - eta * grad / gn
        a = _project_simplex(a, 1.0)
        eta *= 0.9997

    return best, best_score
