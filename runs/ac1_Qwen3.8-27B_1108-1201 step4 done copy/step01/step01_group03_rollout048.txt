                cols.append(j)
                data.append(f[idx])

        if g is None:
            # Try smaller LP (reduced constraints)
            g, t_new = lp_min_peak_cross(f[:n//2] if n > 64 else f, time_limit=3.0)
            if g is None:
                break
            if len(g) < n:
                # Pad or truncate - skip this
                break

        # Blend
        f_new = (1.0 - t_blend) * f + t_blend * g
        f_new = normalize(f_new)

        sc = fast_score(f_new)
        if sc < best_sc:
            best_sc = sc
            best_f = f_new.copy()
        f = f_new

        # Occasional perturbation
        if i % 2 == 1:
            w = max(1, n // 20)
            start = int(rng.integers(0, max(1, n - w)))
            f[start:start + w] *= np.exp(rng.normal(0, 0.2, w))
            f = normalize(f)

    return best_f, best_sc


def make_initial(n, rng, method='cosine'):
    """Create a non-uniform initial sequence."""
    if method == 'cosine':
        x = np.linspace(0.0, np.pi, n)
        a = 0.5 * (1.0 + np.cos(x))
    elif method == 'sin2':
        x = np.linspace(0.0, np.pi, n)
        a = np.sin(x) ** 2
    elif method == 'random':
        a = rng.random(n) + 0.01
    elif method == 'bimodal':
        x = np.arange(n, dtype=np.float64)
        s = max(1, n // 12)
        a = (np.exp(-0.5 * ((x - n // 4) / s) ** 2) +
             np.exp(-0.5 * ((x - 3 * n // 4) / s) ** 2))
    elif method == 'triangular':
        x = np.arange(n, dtype=np.float64)
        a = np.abs(1.0 - 2.0 * x / (n - 1))
    elif method == 'quarter_wave':
        x = np.linspace(0.0, np.pi / 2, n)
        a = np.sin(x)
    elif method == 'perturbed':
        a = np.ones(n) + rng.normal(0.0, 0.8, n)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def block_perturb(f, rng, strength=0.3):
    """Random multiplicative block perturbation."""
    n = len(f)
    cand = f.copy()
    n_blocks = int(rng.integers(1, 5))
    for _ in range(n_blocks):
        w = int(rng.integers(1, max(2, n // 8)))
        start = int(rng.integers(0, max(1, n - w)))
        cand[start:start + w] *= float(np.exp(rng.normal(0.0, strength)))
    return normalize(cand)


def propose_candidate(seed=42, budget_s=1800):
    """Search for the best non-negative sequence minimizing the score."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 10.0)

    # Get baseline
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except Exception:
        parent = np.ones(256, dtype=np.float64)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)

    best = parent.copy()
    bs = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(bs):
        bs = fast_score(best)
    print("Baseline score = %.6f (n=%d)" % (bs, best.size), flush=True)

    # Define search configurations: (length, method)
    configs = []
    lengths = [256, 512, 1024, 2048, 4096]
    methods = ['cosine', 'sin2', 'random', 'bimodal', 'triar_perturbed',
               'triar_perturbed', 'quarter_wave', 'perturbed']

    # Filter methods to valid ones
    valid_methods = ['cosine', 'sin2', 'random', 'bimodal', 'triangular',
                     'quarter_wave', 'perturbed']

    for n in lengths:
        for m in valid_methods:
            configs.append((n, m))

    # Shuffle configs for diversity
    rng.shuffle(configs)

    # Time budget per configuration
    n_configs = len(configs)
    time_per_config = max(20.0, (float(budget_s) - 60.0) / n_configs)
    time_per_config = min(time_per_config, 120.0)

    for cfg_idx, (n, method) in enumerate(configs):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        local_deadline = min(deadline, time.monotonic() + min(time_per_config, remaining - 5))

        print("[%d/%d] n=%d method=%s time=%.0fs" % (
            cfg_idx + 1, n_configs, n, method,
            local_deadline - time.monotonic()), flush=True)

        init = make_initial(n, rng, method)

        # Phase 1: Fast gradient descent
        t_split = time.monotonic() + (local_deadline - time.monotonic()) * 0.6
        g_f, g_sc = gradient_descent(init, t_split, lr_init=0.3,
                                     momentum=0.85, perturb_every=100, top_k=1)
        if np.isfinite(g_sc) and g_sc < bs:
            bs = g_sc
            best = g_f.copy()
            print("   grad: %.6f" % bs, flush=True)

        # Phase 2: LP refinement
        if time.monotonic() < local_deadline - 10:
            lp_deadline = min(local_deadline, time.monotonic() + 40.0)
            lp_f, lp_sc = lp_refinement(g_f, lp_deadline,
                                        n_lp_steps=3, t_blend=0.4, lp_time=6.0)
            if np.isfinite(lp_sc) and lp_sc < bs:
                bs = lp_sc
                best = lp_f.copy()
                print("   lp: %.6f" % bs, flush=True)

        # Phase 3: Perturbation + re-optimization
        if time.monotonic() < local_deadline - 5:
            cand = block_perturb(best, rng, strength=0.4)
            if cand.shape[0] == g_f.shape[0]:
                t2 = min(local_deadline, time.monotonic() + 20.0)
                p_f, p_sc = gradient_descent(cand, t2, lr_init=0.2,
                                             momentum=0.8, perturb_every=80, top_k=1)
                if np.isfinite(p_sc) and p_sc < bs:
                    bs = p_sc
                    best = p_f.copy()
                    print("   perturb: %.6f" % bs, flush=True)

    # Final polish on the best solution
    print("\n[Polish] Final refinement on best (n=%d, score=%.6f)" % (
        best.size, bs), flush=True)
    polish_deadline = deadline
    n_best = best.size

    # Try multiple perturbations + gradient on the best
    polish_rng = np.random.default_rng(seed + 42)
    for i in range(50):
        if time.monotonic() > polish_deadline:
            break
        cand = block_perturb(best, polish_rng, strength=0.15)
        t_rem = polish_deadline - time.monotonic()
        if t_rem < 3:
            break
        p_f, p_sc = gradient_descent(cand,
                                     time.monotonic() + min(5.0, t_rem - 1),
                                     lr_init=0.2, momentum=0.8,
                                     perturb_every=60, top_k=1)
        if np.isfinite(p_sc) and p_sc < bs:
            bs = p_sc
            best = p_f.copy()
            if i % 5 == 0:
                print("   polish[%d]: %.6f" % (i, bs), flush=True)

    # One more LP attempt on the best
    if time.monotonic() < polish_deadline - 10:
        lp_f, lp_sc = lp_refinement(best, polish_deadline,
                                    n_lp_steps=2, t_blend=0.3, lp_time=10.0)
        if np.isfinite(lp_sc) and lp_sc < bs:
            bs = lp_sc
            best = lp_f.copy()
            print("   final LP: %.6f" % bs, flush=True)

    # Validate and return
    best = np.clip(np.asarray(best, dtype=np.float64), 0.0, 1000.0)
    if best.size == 0 or not np.isfinite(best).all() or best.sum() < 0.01:
        best = np.ones(256, dtype=np.float64)

    # Final evaluation
    final_sc = float(evaluate_sequence([float(x) for x in best]))
    print("\nFINAL score = %.6f (n=%d)" % (final_sc, best.size), flush=True)
    if not np.isfinite(final_sc):
        best = np.ones(256, dtype=np.float64)
        final_sc = 2.0
        print("FALLBACK to uniform, score = %.6f" % final_sc, flush=True)

    return [float(x) for x in best]
