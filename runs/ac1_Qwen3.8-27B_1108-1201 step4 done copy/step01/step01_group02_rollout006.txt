A simpler and potentially more effective approach: **use the exact subgradient when the maximum is at a unique point**.

At a point where `conv[k*]` is the unique maximum:
- The gradient is `2n * (a[k*-j] + a[j-k*])` (just two shifted copies of a)
- This is very cheap to compute!

We can use a "non-smooth" proximal gradient method (like proximal subgradient) with this exact subgradient.

But the issue is that at the optimum, the max might be at multiple points (which would actually be desirable - a flat peak). In that case, we need to use a subgradient that's a convex combination.

2. Compute subgradient: `g = 2n * mean_{k in K*} (a[k-j] + a[j-k])`
3. Update: `a_new = proj_simplex(a - step * g + momentum * (a - a_prev))`
4. Adaptive step size (e.g., `step = c / ||g||` or backtracking)

```python
def subgradient_method(a_init, time_limit_s, verbose=True):
    """
    Subgradient method with momentum for minimizing 2n*max(conv(a,a)) on the simplex.
    """
    start_time = time.monotonic()
    n = len(a_init)
    a = normalize_nonneg(a_init.copy().astype(np.float64))
    a_prev = a.copy()
    
    best_a = a.copy()
    best_score = score_sequence(a)
    
    t_mom = 1.0
    momentum = 0.9
    step = 0.1  # Initial step size
    
    for it in range(1000000):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        
        # Compute convolution and find active set
        conv = np.convolve(a, a)
        max_val = conv.max()
        
        # Use softmax with adaptive temperature
        T = max_val / max(1.0, math.log(2 * n)) * 0.5
        logits = conv / T
        logits -= logits.max()
        w = np.exp(logits)
        w = w / w.sum()
        
        # Compute weighted subgradient
        # grad_j = 2n * sum_k w[k] * (a[k-j] + a[j-k])
        # Use FFT for efficiency
        L = 1
        while L < 3 * n - 2:
            L *= 2
        A = np.fft.rfft(a, L)
        W = np.fft.rfft(w, L)
        S2 = np.fft.irfft(A * W, L)[:n]
        S1 = np.fft.irfft(np.conj(W) * A, L)[:n]
        grad = 2.0 * n * (S1 + S2)
        
        # Momentum update (Nesterov acceleration)
        t_new = (1.0 + math.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        beta = (t_mom - 1.0) / t_new
        
        # Step
        step_size = step / (1.0 + 0.01 * it)  # Decaying step
        a_candidate = a - step_size * grad + beta * (a - a_prev)
        a_candidate = proj_simplex(a_candidate)
        
        a_prev = a
        a = a_candidate
        t_mom = t_new
        
        # Track best
        if it % 5 == 0:
            score = score_sequence(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                if verbose and (best_score < 1.52 or it < 100):
                    print(f"        iter {it}: score={score:.6f}, T={T:.4f}")
        
        if time.monotonic() - start_time > time_limit_s:
            break
    
    # Final evaluation
    score = score_sequence(a)
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    if verbose:
        print(f"      Subgrad done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    
    return best_a, best_score
```
