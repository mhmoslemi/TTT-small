    # Cross-correlation via FFT:
    # xcorr[j] = sum_i w[i+j] * a[i] for i = 0..n-1 (with w[i+j] = 0 if out of range)

def grad_step(a, lr, tau, momentum=None, v=None):
    """One gradient step with soft-max weighting and optional momentum."""
    n = len(a)
    grad, c = soft_gradient(a, tau)
    # Project out mean component (to handle sum constraint)
    grad -= grad.mean()
    
    if momentum is not None and v is not None:
        v_new = momentum * v - lr * grad
        a = a + v_new
        v = v_new
    else:
        a = a - lr * grad
        v = None
    
    a = np.maximum(a, 0.0)
    a = normalize(a)
    return a, c, v


def lp_solve(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M (elementwise), g >= 0."""
    n = len(f)
    if n > 2048:
        return None
    
    c_obj = -np.ones(n)
    
    # Build constraint matrix
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[m, j] = f[m - j]
    
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def lp_solve_sparse(f, M):
    """Solve LP using sparse matrix for efficiency."""
    from scipy.sparse import csr_matrix
    import scipy.sparse as sp
    
    n = len(f)
    c_obj = -np.ones(n)
    
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            rows.append(m)
            cols.append(j)
            vals.append(f[m - j])
    
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:

    except Exception:
        pass
    return None

"""
Optimizes a non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses adaptive soft-max gradient descent with momentum, LP-based refinement
with reduced peak targets, top-k peak focusing, and multiple restart strategies.
Incorporates sparse LP for efficiency and periodic diversification to escape local minima.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize_seq(a, target=None):
    """Normalize sequence to have a specific sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def topk_softmax_weights(c, tau, k=None):
    """Compute top-k soft-max weights over convolution values."""
    if k is not None and k < len(c):
        # Keep only top k values
        c_work = c.copy()
        threshold = np.partition(c, -k)[-k]
        c_work[c_work < threshold] = -np.inf
    c_max = c.max()
    w = np.exp(tau * (c_work - c_max))
    w = np.nan_to_num(w)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(c)
        w[c.argmax()] = 1.0
    else:
        w /= w_sum
    return w


def compute_gradient(a, w):
    """Compute gradient given weights w over convolution values.
    grad[j] = 2 * sum_k w[k] * a[k-j] (valid indices).
    """
    n = len(a)
    N = 1
    while N < 2 * n:
        N *= 2
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    xcorr = np.fft.irfft(W * A.conj(), N)
    grad = 2.0 * xcorr[:n]
    return grad


def grad_step_momentum(a, lr, tau, momentum, v, topk=None):
    """One gradient step with soft-max weighting, top-k focusing, and momentum."""
    n = len(a)
    c = np.convolve(a, a)
    w = topk_softmax_weights(c, tau, topk=topk)
    grad = compute_gradient(a, w)
    # Project out mean (sum constraint)
    grad -= grad.mean()
    
    v_new = momentum * v - lr * grad
    a = a + v_new
    a = np.maximum(a, 0.0)
    a = normalize_seq(a)
    return a, c, v_new


def build_sparse_constraint(f):
    """Build sparse constraint matrix for LP: A[m,j] = f[m-j] if valid."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            rows.append(m)
            cols.append(j)
            vals.append(f[m - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns g or None."""
    n = len(f)
    if n > 2048:
        return None
    if M <= 0:
        return None
    
    c_obj = -np.ones(n)
    A = build_sparse_constraint(f)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def optimize_gradient(a0, deadline, max_iters=3000, verbose=False):
    """Optimize using adaptive soft-max gradient with momentum and top-k focusing."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    a = normalize_seq(a)
    score = compute_obj(a)
    best = a.copy()
    best_score = score
    
    tau = 20.0
    tau_end = 80.0
    lr_base = 0.12
    momentum = 0.85
    topk = min(n, 50)
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break
        
        progress = it / max_iters
        # Cosine annealing for lr
        lr = lr_base * 0.5 * (1.0 + np.cos(np.pi * progress))
        lr = max(lr, 0.003)
        # Anneal temperature (increase focus on peaks)
        tau = tau + (tau_end - tau) * 0.005
        # Anneal topk (start broad, end narrow)
        current_topk = max(5, int(topk * (1.0 - 0.5 * progress)))
        
        a, c, v = grad_step_momentum(a, lr, tau, momentum, v, topk=current_topk)
        
        if (it + 1) % 200 == 0:
            s = compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()
                if verbose:
                    print(f"    grad iter {it}: score={s:.6f}")
    
    # Final check
    s = compute_obj(a)
    if s < best_score:
        best_score = s
        best = a.copy()
    
    return best, best_score


def optimize_lp_refine(a0, deadline, max_iters=200, verbose=False):
    """Optimize using LP-based refinement with reduced peak targets."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    a = normalize_seq(a)
    score = compute_obj(a)
    best = a.copy()
    best_score = score
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break
        
        c = np.convolve(a, a)
        M = float(c.max())
        
        # Try LP with current peak
        g = solve_lp(a, M)
        
        if g is None:
            break
        
        # Try blending with various t values
        improved = False
        blend_options = [0.9, 0.7, 0.5, 0.3, 0.15, 0.05]
        for t in blend_options:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_seq(a_new)
            s_new = compute_obj(a_new)
            if s_new < best_score - 1e-8:
                a = a_new
                best = a.copy()
                best_score = s_new
                improved = True
                if verbose:
                    print(f"    LP iter {it}, t={t}: score={s_new:.6f}")
                break
        
        if not improved:
            # Try with reduced M (more aggressive)
            for m_factor in [0.98, 0.95, 0.90]:
                M_reduced = m_factor * M
                g2 = solve_lp(a, M_reduced)
                if g2 is not None and g2.sum() > 1e-6:
                    for t in [0.5, 0.3, 0.1]:
                        a_new = (1.0 - t) * a + t * g2
                        a_new = np.maximum(a_new, 0.0)
                        a_new = normalize_seq(a_new)
                        s_new = compute_obj(a_new)
                        if s_new < best_score - 1e-8:
                            a = a_new
                            best = a.copy()
                            best_score = s_new
                            improved = True
                            break
                    if improved:
                        break
        
        if not improved:
            # Small gradient step to escape local minimum
            tau_esc = 40.0
            a, c, _ = grad_step_momentum(a, 0.03, tau_esc, 0.0, np.zeros(n))
            s = compute_obj(a)
            if s < best_score:
                best = a.copy()
                best_score = s
            else:
                # No improvement in several iterations, diversify
                rng_local = np.random.default_rng(it)
                a_perturbed = a + 0.01 * rng_local.standard_normal(n)
                a_perturbed = np.maximum(a_perturbed, 0.0)
                a_perturbed = normalize_seq(a_perturbed)
                s_pert = compute_obj(a_perturbed)
                if s_pert < best_score:
                    a = a_perturbed
                    best = a.copy()
                    best_score = s_pert
                else:
                    a = a_perturbed  # Accept perturbation anyway for exploration
    
    return best, best_score


def optimize_combined(a0, deadline, verbose=False):
    """Combined optimization: gradient + LP alternation with refinement."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    a = normalize_seq(a)
    score = compute_obj(a)
    best = a.copy()
    best_score = score
    
    if verbose:
        print(f"  Starting optimization n={n}, initial score={score:.6f}")
    
    # Phase 1: Gradient descent (coarse)
    time_for_grad = min(60.0, (deadline - time.monotonic()) * 0.4)
    if time.monotonic() + time_for_grad > time.monotonic() + 5.0:
        a, best_score = optimize_gradient(a, time.monotonic() + time_for_grad,
                                           max_iters=2000, verbose=verbose)
        best = a.copy()
        if verbose:
            print(f"  After gradient: score={best_score:.6f}")
    
    # Phase 2: LP refinement
    if n <= 1024 and time.monotonic() < deadline - 15.0:
        a, score = optimize_lp_refine(a, deadline - 10.0, max_iters=150, verbose=verbose)
        if score < best_score:
            best_score = score
            best = a.copy()
            if verbose:
                print(f"  After LP: score={best_score:.6f}")
    
    # Phase 3: Fine gradient polish
    if time.monotonic() < deadline - 10.0:
        grad_deadline = min(deadline, time.monotonic() + 30.0)
        a, score = optimize_gradient(best, grad_deadline, max_iters=1500, verbose=verbose)
        if score < best_score:
            best_score = score
            best = a.copy()
            if verbose:
                print(f"  After polish: score={best_score:.6f}")
    
    # Phase 4: More LP if time remains
    if n <= 1024 and time.monotonic() < deadline - 10.0:
        a, score = optimize_lp_refine(best, deadline, max_iters=100, verbose=verbose)
        if score < best_score:
            best_score = score
            best = a.copy()
            if verbose:
                print(f"  After final LP: score={best_score:.6f}")
    
    return best, best_score


def diversify(a, rng, scale=0.05):
    """Add diversity to a sequence while maintaining non-negativity."""
    n = len(a)
    # Random mask perturbation
    mask = (rng.random(n) < 0.3)
    perturbation = scale * rng.standard_normal(n) * mask
    a_new = a + perturbation
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize_seq(a_new)
    return a_new
