    # dc[0]/da[1] = 0 (from formula: 2*a[0-1] = 2*a[-1] = 0) ✓
    # dc[1]/da[0] = 2*a[1] (from formula: 2*a[1-0] = 2*a1)

    # 
    # Therefore: grad[j'] = sum_k grad_conv[k] * 2*a[k-j']
    # = 2 * sum_k grad_conv[k] * a[k-j']

    # (grad_conv ⊛ a)[j'] = sum_k grad_conv[k+j'] * a[k]... 
    # Hmm, let me define it correctly.
    # sum_k grad_conv[k] * a[k-j']

    # = sum_m grad_conv[m+j'] * a[m]
    # This is the cross-correlation: (a ⊛ grad_conv)[j'] = sum_m a[m] * grad_conv[m+j']

    # For real signals: (x ⊛ y)[n] = sum_m x[m]*y[m-n]
    # FFT(x ⊛ y) = conj(X) * Y

    # with appropriate zero-padding.

    # np.correlate(a, v, mode='full') computes: out[n] = sum_k a[k] * conj(v[k-n])

    # np.correlate(a, v, 'full')[n] = sum_k a[k] * v[k-n]
    # So np.correlate(g, a, 'full')[j] = sum_k g[k] * a[k-j]

    # Actually for g of length 2n-1 and a of length n, the full correlation has length (2n-1)+n-1 = 3n-2

    N = 1

    # For n up to ~5000, the direct computation of the gradient is O(n^2) which is fine for a few iterations.
    # But for a gradient descent loop with many iterations, we need it faster.
    
    # Alternative: Just use the fact that grad[j] = 2 * sum_k g[k] * a[k-j]
    # This is a convolution-like operation. Let me use scipy.signal or just np.convolve.

    # len(g) = 2n-1, len(a) = n, so valid gives length n.

    # = g[j]*a[0] + g[j+1]*a[1] + ... + g[j+n-1]*a[n-1]
    # That's exactly h[j]!
    
    # So: grad = 2 * np.correlate(grad_conv, a, mode='valid')
    
    # But wait, this requires len(grad_conv) >= len(a), which is (2n-1) >= n, yes.
    # And 'valid' mode gives length 2n-1 - n + 1 = n.

    # g = [g0, g1, g2], a = [a0, a1]
    # np.correlate(g, a, 'valid') should give length 3-2+1 = 2
    # [0]: g[0]*a[0] + g[1]*a[1] = g0*a0 + g1*a1
    # [1]: g[1]*a[0] + g[2]*a[1] = g1*a0 + g2*a1
    # grad = 2 * [g0*a0 + g1*a1, g1*a0 + g2*a1]
    # Which matches! Great.
    
    grad = 2.0 * np.correlate(grad_conv, a, mode='valid')
    
    # Now the full objective gradient:
    # f(a) = 2*n*softmax(conv(a,a)) / sum(a)^2
    # df/da_j = 2*n * (d(softmax)/da_j * sum(a)^2 - softmax * 2*sum(a)*1) / sum(a)^4
    # = 2*n * (grad[j]*sum(a)^2 - 2*softmax*sum(a)*1) / sum(a)^4
    # = 2*n * (grad[j]/sum(a)^2 - 2*softmax/(sum(a)^2 * sum(a)))... 
    # Let me be more careful.
    # f = 2*n*S / S_a^2 where S = softmax(conv), S_a = sum(a)
    # df/da_j = 2*n * (dS/da_j * S_a^2 - S * 2*S_a * d(S_a)/da_j) / S_a^4
    # = 2*n * (grad_softmax[j] * S_a^2 - S * 2*S_a) / S_a^4
    # = 2*n * (grad_softmax[j] / S_a^2 - 2*S / S_a^3)
    # = 2*n/S_a^2 * (grad_softmax[j] - 2*S/S_a)
    
    # But since we're normalizing (we can fix sum(a)=1), let's just minimize S = softmax(conv(a,a))
    # with the constraint sum(a)=1, a>=0.
    # With the constraint sum(a)=1, the gradient is just grad_softmax (projected to the constraint).
    
    # Actually, for simplicity, let's just minimize softmax(conv(a,a)) with a>=0,
    # and normalize at the end. The score depends on the shape, not the scale.
    
    obj = soft_max_val
    return obj, grad


def adam_smooth_optimize(a, time_limit=120.0, lr=0.01, tau=1.0, verbose=True):
    """Optimize using Adam with soft-max objective."""
    start_time = time.monotonic()
    a = a.copy().astype(np.float64)
    a = np.maximum(0, a)
    s = a.sum()
    if s > 0:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_score = score_sequence(a)
    
    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    
    # Learning rate schedule
    lr_init = lr
    patience = 100
    stale = 0
    
    if verbose:
        print(f"    Adam start: score={best_score:.6f}, n={n}, lr={lr}, tau={tau}")
    
    while time.monotonic() - start_time < time_limit:
        # Compute gradient
        conv = np.convolve(a, a)
        scaled = conv / tau
        max_val = scaled.max()
        shifted = scaled - max_val
        exp_shifted = np.exp(shifted)
        softmax_weights = exp_shifted / exp_shifted.sum()
        grad = 2.0 * np.correlate(softmax_weights, a, mode='valid')
        
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Adaptive learning rate
        effective_lr = lr_init * (0.95 ** (t // 200))
        
        a_new = a - effective_lr * m_hat / (np.sqrt(v_hat) + epsilon)
        a_new = np.maximum(0, a_new)
        
        # Normalize (reparameterization trick)
        s = a_new.sum()
        if s > 1e-10:
            a_new = a_new / s * s  # keep same scale for now
            # Actually, let's not normalize here, just clip
            a_new = np.maximum(0, a_new)
            s = a_new.sum()
            if s > 1e-10:
                a = a_new / s  # normalize to sum=1
            else:
                a = best_a.copy()
        else:
            a = best_a.copy()
        
        score = score_sequence(a)
        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            stale = 0
        else:
            stale += 1
        
        if stale > patience:
            # Reduce learning rate
            lr_init *= 0.5
            stale = 0
            if lr_init < 1e-7:
                break
        
        if verbose and t % 100 == 0:
            print(f"    Adam iter {t}: score={best_score:.6f}, lr={lr_init:.6f}, t={time.monotonic()-start_time:.1f}s")
    
    if verbose:
        print(f"    Adam done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def spectral_peak_reduction(a, time_limit=120.0, verbose=True):
    """
    Spectral method for peak reduction.
    Repeatedly: compute power spectrum, identify peaks in IFFT, reduce them,
    then reconstruct a non-negative signal.
    """
    start_time = time.monotonic()
    a = a.copy().astype(np.float64)
    a = np.maximum(0, a)
    s = a.sum()
    if s > 0:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_score = score_sequence(a)
    if verbose:
        print(f"    Spectral start: score={best_score:.6f}, n={n}")
    
    # FFT size (need at least 2n-1 for linear convolution)
    N = 1
    while N < 2 * n:
        N *= 2
    
    it = 0
    for it in range(10000):
        if time.monotonic() - start_time > time_limit:
            break
        
        # Compute power spectrum
        A = np.fft.rfft(a, N)
        P = np.abs(A) ** 2  # power spectrum
        
        # IFFT of P gives the autocorrelation (up to scaling)
        c = np.fft.irfft(P, N)[:2*n-1]
        
        # Identify what fraction is above the mean
        c_mean = c.mean()
        c_max = c.max()
        
        # Apply peak reduction: compress the peak
        # Use a soft clip
        threshold = c_mean + 0.1 * (c_max - c_mean)
        P_reduced = P.copy()
        c_reduced = c.copy()
        
        # Reduce peaks by reducing P at frequency bins that contribute most
        # Simple approach: modify P to make IFFT(P) flatter
        # This is like an iterative water-filling
        
        # Alternative: use the "ripple" method
        # Step 1: Compute current peak
        # Step 2: In frequency domain, reduce the contribution to the peak
        # The peak of IFFT(P) at position p is related to P through:
        # c[p] = sum_f P[f] * exp(2*pi*i*f*p/N) / N
        # To reduce c[p], we want to reduce P[f] where exp(2*pi*i*f*p/N) is large
        
        # Simpler: just scale down P proportionally at the peak location
        # This is essentially: a = a (unchanged), c has peaks, so we modify a directly
        
        # Let me use a different approach: iterate on a directly
        # Given c = IFFT(|A|^2), find the peak location p*
        # Then a should be modified to reduce c[p*]
        # c[p*] = sum_k a[k] * a[k-p*] (with appropriate bounds)
        # To reduce this, we should reduce a where it contributes most to c[p*]
        # Contribution of a[j] to c[p*] is a[p*-j] (roughly)
        
        # This is getting circular. Let me use a simpler approach:
        # Modify a by: a_new = a - eta * grad_peak
        # where grad_peak[j] = a[j - p*] (contribution to peak)
        # This is essentially a step of projected gradient descent on the peak objective.
        
        p_star = c.argmax()
        # Gradient of c[p_star] w.r.t. a: it's 2*a shifted by p_star
        # From our earlier derivation: dc[k]/da[j] = 2*a[k-j]
        # So dc[p_star]/da[j] = 2*a[p_star - j] (with a[m]=0 for m outside [0,n-1])
        
        grad_peak = np.zeros(n)
        # a[p_star - j] is non-zero when 0 <= p_star - j < n, i.e., p_star - n < j <= p_star
        # j in [max(0, p_star-n+1), min(n-1, p_star)]
        j_start = max(0, p_star - n + 1)
        j_end = min(n, p_star + 1)
        for j in range(j_start, j_end):
            m = p_star - j
            if 0 <= m < n:
                grad_peak[j] = 2.0 * a[m]
        
        # But this only handles one peak simultaneously. For multiple peaks, sum gradients.
        # Let's handle top few peaks:
        top_k = min(5, 2 * n - 1)
        c_sorted = np.sort(c)[::-1]
        # Find all lags where c is within 90% of max
        threshold = 0.9 * c_max
        peak_lags = np.where(c >= threshold)[0]
        
        if len(peak_lags) > 1:
            grad_total = np.zeros(n)
            for p in peak_lags:
                j_start = max(0, p - n + 1)
                j_end = min(n, p + 1)
                for j in range(j_start, j_end):
                    m = p - j
                    if 0 <= m < n:
                        grad_total[j] += 2.0 * a[m]
            # Normalize gradient by number of peaks
            grad_total /= len(peak_lags)
        else:
            grad_total = grad_peak
        
        # Learning rate
        eta = 0.01
        a_new = a - eta * grad_total
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s > 1e-10:
            a_new = a_new / s
        else:
            a = best_a.copy()
            break
        
        score = score_sequence(a_new)
        if score < best_score - 1e-10:
            best_score = score
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # Try smaller step
            a_new = a - 0.1 * eta * grad_total
            a_new = np.maximum(0, a_new)
            s = a_new.sum()
            if s > 1e-10:
                a_new = a_new / s
                score2 = score_sequence(a_new)
                if score2 < best_score:
                    a = a_new.copy()
                    best_a = a_new.copy()
                    best_score = score2
            # If still no improvement, break or reduce step size further
            if it > 200 and score_sequence(a) >= best_score * (1 + 1e-5):
                break
        
        if verbose and it % 200 == 0:
            print(f"    Spectral iter {it}: score={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    
    if verbose:
        print(f"    Spectral done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def lp_refine(a, time_limit_s=180.0, verbose=True):
    """LP-based refinement (improved version)."""
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    if verbose:
        print(f"    LP start: score={best_score:.6f}, n={n}")
    
    stagnant = 0
    it = 0
    max_iters = 200
    
    for it in range(max_iters):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        
        conv = np.convolve(a, a)
        score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        if stagnant > 20:
            break
        
        M = conv.max()
        remaining = time_limit_s - elapsed
        if remaining < 10.0:
            break
        
        # Build constraint matrix: conv(a, g) <= M for all lags
        # This is a (2n-1) x n matrix
        timeout = max(5.0, min(60.0, remaining * 0.3))
        
        # Build the convolution matrix
        # (A @ g)[k] = sum_j A[k,j] * g[j] = conv(a, g)[k]
        # A[k,j] = a[k-j] if 0 <= k-j < n, else 0
        # For efficiency, use sparse
        n_rows = 2 * n - 1
        row_idx = []
        col_idx = []
        data = []
        for k in range(n_rows):
            j_start = max(0, k - n + 1)
            j_end = min(n, k + 1)
            for j in range(j_start, j_end):
                m = k - j
                if 0 <= m < n:
                    row_idx.append(k)
                    col_idx.append(j)
                    data.append(a[m])
        
        A_mat = csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))
        
        c = -np.ones(n)
        b = np.full(n_rows, M)
        bounds = [(0.0, None)] * n
        
        try:
            res = linprog(c, A_ub=A_mat, b_ub=b, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': timeout})
            if res.success and res.x is not None:
                g = np.maximum(0.0, res.x)
            else:
                break
        except Exception:
            break
        
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            break
        
        g = g / sum_g
        alpha = min(0.5, max(0.1, 2.0 / (1.0 + 0.5 * it)))
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = normalize_nonneg(a_new)
        
        new_score = score_sequence(a_new)
        if new_score <= score + 1e-10:
            a = a_new
        else:
            alpha2 = alpha * 0.3
            a_new = (1.0 - alpha2) * a + alpha2 * g
            a_new = normalize_nonneg(a_new)
            a = a_new
    
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    if verbose:
        print(f"    LP done: best={best_score:.6f} after {it+1} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def coordinate_descent_refine(a, time_limit=60.0, verbose=False):
    """
    Coordinate descent: for each coordinate, find the optimal value
    that minimizes the peak convolution.
    """
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    while time.monotonic() - start_time < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - start_time > time_limit:
                break
            
            # Try a few values for a[j]
            old_val = a[j]
            # Try: old * 0.5, old * 0.7, old, old * 1.3, old * 1.5, 0
            candidates = [old_val * f for f in [0.3, 0.5, 0.7, 1.3, 1.5, 2.0]]
            candidates.append(0.0)
            if old_val > 0:
                candidates.append(old_val)
            
            for new_val in candidates:
                a[j] = max(0, new_val)
                s = a.sum()
                if s < 0.01:
                    a[j] = old_val
                    continue
                a_norm = a / s
                score = score_sequence(a_norm)
                if score < best_score:
                    best_score = score
                    best_a = a_norm.copy()
                    improved = True

def random_perturb_refine(a, time_limit=60.0, rng=None, verbose=True):

            best_a = candidate.copy()

        print(f"    RP done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")

def sparse_lp_step(a, M, n_active=20, timeout=10):
    """Solve LP with only the most active constraints."""
    n = len(a)
    conv = np.convolve(a, a)
    # Get the lags with highest convolution values
    top_lags = np.argsort(conv)[::-1][:n_active]
    
    # Build constraint matrix with only these rows
    n_rows = len(top_lags)
    rows_data = []
    row_idx = []
    col_idx = []
    for i, k in enumerate(top_lags):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            m = k - j
            if 0 <= m < n:
                row_idx.append(i)
                col_idx.append(j)
                rows_data.append(a[m])
    
    A = csr_matrix((rows_data, (row_idx, col_idx)), shape=(n_rows, n))
    c = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success:
            return np.maximum(0, res.x)
    except:
        pass
    return None

"""
Multi-strategy optimization for minimizing peak auto-convolution of non-negative sequences.
Combines: (1) smooth softmax-based Adam optimization with adaptive temperature,
(2) sparse LP refinement with active constraint selection, (3) multi-start with diverse

import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_blackman(n):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))


def initial_hann(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 - np.cos(2.0 * np.pi * t / n))


def initial_flat_top(n, frac=0.4):
    a = np.zeros(n)
    center = n // 2
    width = int(frac * n)
    start = max(0, center - width // 2)
    end = min(n, start + width)
    a[start:end] = 1.0
    # Smooth edges
    edge = max(1, width // 10)
    if start > 0:
        for i in range(edge):
            a[start + i] *= 0.5 * (1 - np.cos(np.pi * i / edge))
            a[end - 1 - i] *= 0.5 * (1 - np.cos(np.pi * i / edge))
    return np.maximum(0, a)


def get_softmax_grad(a, conv, tau):
    """Compute gradient of soft-max(conv) w.r.t. a using the correlation formula."""
    n = len(a)
    scaled = conv / tau
    max_val = scaled.max()
    shifted = scaled - max_val
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / exp_shifted.sum()
    # grad[j] = 2 * sum_k weights[k] * a[k-j]
    # = 2 * np.correlate(weights, a, mode='valid')
    grad = 2.0 * np.correlate(weights, a, mode='valid')
    soft_max_val = (max_val + np.log(np.exp(shifted).sum())) * tau
    return soft_max_val, grad, weights


def adam_softmax_optimize(a, time_limit=120.0, lr=0.02, tau_start=1.0, tau_end=0.01,
                          verbose=False):
    """Adam optimizer with adaptive temperature for soft-max objective."""
    start_time = time.monotonic()
    a = a.copy().astype(np.float64)
    a = np.maximum(0, a)
    s = a.sum()
    if s > 0:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_score = score_sequence(a)

    beta1 = 0.85
    beta2 = 0.999
    epsilon = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    patience = 150
    stale = 0
    current_lr = lr

    if verbose:
        print(f"    AdamSoft start: score={best_score:.6f}, n={n}, lr={lr}, tau={tau_start}→{tau_end}")

    while time.monotonic() - start_time < time_limit:
        # Adaptive temperature
        progress = min(1.0, (time.monotonic() - start_time) / max(1.0, time_limit))
        tau = tau_start * (tau_end / tau_start) ** progress

        conv = np.convolve(a, a)
        _, grad, _ = get_softmax_grad(a, conv, tau)

        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # Cosine annealing for learning rate
        lr_t = 0.5 * current_lr * (1 + np.cos(np.pi * progress)) + 0.1 * current_lr

        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + epsilon)
        a_new = np.maximum(0, a_new)
        s_new = a_new.sum()
        if s_new > 1e-10:
            a = a_new / s_new
        else:
            a = best_a.copy()
            break

        score = score_sequence(a)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stale = 0
        else:
            stale += 1

        if stale > patience:
            current_lr *= 0.5
            stale = 0
            if current_lr < 1e-8:
                break

        if verbose and t % 200 == 0:
            print(f"    AdamSoft iter {t}: score={best_score:.6f}, lr={current_lr:.6f}, "
                  f"tau={tau:.4f}, {time.monotonic()-start_time:.1f}s")

    if verbose:
        print(f"    AdamSoft done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def build_conv_matrix_rows(a, lag_indices):
    """Build sparse matrix rows for conv(a, g) at specified lag indices."""
    n = len(a)
    data = []
    row_idx = []
    col_idx = []
    for i, k in enumerate(lag_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            m = k - j
            if 0 <= m < n:
                row_idx.append(i)
                col_idx.append(j)
                data.append(a[m])
    n_rows = len(lag_indices)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def sparse_lp_step(a, M, n_active=100, timeout=15.0):
    """Solve LP with only the most active (highest conv) constraints."""
    n = len(a)
    conv = np.convolve(a, a)
    # Select top n_active lags
    n_active = min(n_active, 2 * n - 1)
    top_lags = np.argsort(conv)[::-1][:n_active]

    A = build_conv_matrix_rows(a, top_lags)
    c = -np.ones(n)
    b = np.full(len(top_lags), M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def full_lp_step(a, M, timeout=30.0):
    """Solve full LP with all constraints."""
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix_rows(a, np.arange(n_rows))
    c = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def lp_refine(a, time_limit_s=180.0, verbose=True, use_full_lp=False):
    """LP-based iterative refinement with adaptive step size."""
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    if verbose:
        print(f"    LP start: score={best_score:.6f}, n={n}")

    stagnant = 0
    it = 0
    alpha_base = 0.3

    while it < 500:
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break

        conv = np.convolve(a, a)
        M = conv.max()
        score = 2.0 * n * M / (a.sum() ** 2)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        if stagnant > 15:
            break

        remaining = time_limit_s - elapsed
        if remaining < 10.0:
            break

        # Choose LP type based on time
        timeout = max(5.0, min(40.0, remaining * 0.25))
        if use_full_lp and n <= 2000:
            g = full_lp_step(a, M, timeout=timeout)
        else:
            g = sparse_lp_step(a, M, n_active=min(200, 2 * n - 1), timeout=timeout)

        if g is None:
            break

        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            if verbose:
                print(f"    LP no improvement (sum_g={sum_g:.6f}) at iter {it}")
            break

        g = g / sum_g

        # Line search for alpha
        best_alpha = 0.1
        best_new_score = score_sequence((1 - 0.1) * a + 0.1 * g)
        for alpha_try in [0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            candidate = (1 - alpha_try) * a + alpha_try * g
            candidate = normalize_nonneg(candidate)
            s_try = score_sequence(candidate)
            if s_try < best_new_score:
                best_new_score = s_try
                best_alpha = alpha_try

        a = (1 - best_alpha) * a + best_alpha * g
        a = normalize_nonneg(a)
        it += 1

        if verbose and (it % 10 == 0 or it < 5):
            print(f"    LP iter {it}: score={best_score:.6f}, alpha={best_alpha:.3f}, "
                  f"{time.monotonic()-start_time:.1f}s")

    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    if verbose:
        print(f"    LP done: best={best_score:.6f} after {it} iters, "
              f"{time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def subgradient_optimize(a, time_limit=60.0, verbose=False):
    """Subgradient method with line search on the active peaks."""
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)

    it = 0
    while time.monotonic() - start_time < time_limit:
        conv = np.convolve(a, a)
        M = conv.max()
        # Find active lags (within 1% of max)
        active = np.where(conv >= 0.99 * M)[0]
        if len(active) == 0:
            active = np.array([conv.argmax()])

        # Compute subgradient: average gradient over active lags
        # dc[k]/da[j] = 2*a[k-j] (with a[m]=0 outside [0,n-1])
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n, k + 1)
            for j in range(j_start, j_end):
                m = k - j
                if 0 <= m < n:
                    grad[j] += 2.0 * a[m]
        grad /= len(active)

        # Line search
        best_eta = 0.001
        best_new_score = score_sequence(a)
        for eta in [0.001, 0.003, 0.01, 0.03, 0.01, 0.005]:
            candidate = a - eta * grad
            candidate = np.maximum(0, candidate)
            s = candidate.sum()
            if s > 1e-10:
                candidate = candidate / s
                s_try = score_sequence(candidate)
                if s_try < best_new_score:
                    best_new_score = s_try
                    best_eta = eta

        if best_new_score < best_score - 1e-12:
            a = a - best_eta * grad
            a = np.maximum(0, a)
            s = a.sum()
            if s > 1e-10:
                a = a / s
            best_score = best_new_score
            best_a = a.copy()
        else:
            # No improvement, reduce step
            if best_eta < 1e-6:
                break

        it += 1
        if verbose and it % 50 == 0:
            print(f"    SubGrad iter {it}: score={best_score:.6f}, {time.monotonic()-start_time:.1f}s")

    if verbose:
        print(f"    SubGrad done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def random_perturb_refine(a, time_limit=60.0, rng=None, verbose=True):
    """Random perturbation with acceptance criterion."""
    if rng is None:
        rng = np.random.default_rng(42)
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    accept_count = 0
    sigma = 0.1

    while time.monotonic() - start_time < time_limit:
        width = max(2, int(rng.integers(n // 200, n // 8)))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        candidate = a.copy()
        candidate[start:stop] *= np.exp(rng.normal(0.0, sigma, size=stop - start))
        if rng.random() < 0.2:
            candidate = 0.5 * (candidate + candidate[::-1])
        candidate = normalize_nonneg(candidate)
        score = score_sequence(candidate)
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if verbose and accept_count % 100 == 0:
                print(f"    RP accept {accept_count}: score={best_score:.6f}")
        else:
            sigma *= 0.999
            if sigma < 0.01:
                sigma = 0.1

    if verbose:
        print(f"    RP done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score
