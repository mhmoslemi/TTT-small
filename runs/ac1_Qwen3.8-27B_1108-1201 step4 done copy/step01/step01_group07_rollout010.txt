```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
- Smooth gradient (log-sum-exp) with mirror descent on the simplex
- Periodic LP-based direction updates (state of the art approach)
- Multiple sequence lengths with warm starts

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def compute_grad_direct(a, w):
    """Compute gradient: grad[i] = 2 * sum_k w[k] * a[k-i] (valid range)."""
    n = len(a)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            grad[j_start:j_end+1] += wk * a[k-j_end:k-j_start+1][::-1]
    return 2.0 * grad

def smooth_obj_and_grad(a, t):
    """
    Compute smooth approximation of max(conv) and its gradient.
    Returns (smooth_obj, gradient).
    """
    n = len(a)
    conv = fast_convolve(a)
    max_conv = conv.max()
    
    if t < 1e6:
        scaled = t * (conv - max_conv)
        # numerically stable log-sum-exp
        scaled_max = scaled.max()
        if scaled_max > 500:
            scaled -= 500  # prevent overflow, doesn't change the result
        exp_sum = np.exp(scaled).sum()
        if exp_sum < 1e-300:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp(scaled) / exp_sum
        obj = max_conv + np.log(exp_sum) / t
    else:
        # Hard max
        w = np.zeros(len(conv))
        max_idx = np.where(conv >= max_conv - 1e-10)[0]
        w[max_idx] = 1.0 / len(max_idx)
        obj = max_conv
    
    grad = compute_grad_direct(a, w)
    return obj, grad

def lp_direction(f, M):
    """
    Solve LP: max sum(g) s.t. g >= 0, ||f*g||_inf <= M.
    Returns g (non-negative).
    """
    n = len(f)
    
    # Constraint matrix: for each k, (f*g)[k] = sum_j f[k-j]*g[j] <= M
    # For k=0..2n-2, the coefficient of g[j] is f[k-j] if 0<=k-j<n, else 0.
    
    # Build constraint matrix A_ub and b_ub
    # A_ub has shape (2n-1, n), b_ub = M * ones(2n-1)
    
    # For efficiency, use sparse if available, but for n<=2048, dense is fine
    # Actually for n=2048, the matrix is 4095 x 2048 = ~8M entries, which is fine.
    
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = f[idx]
    
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    
    if res.success:
        return np.maximum(res.x, 0)
    else:
        return None

def lp_direction_sparse(f, M):
    """
    More efficient LP direction using scipy sparse.
    """
    from scipy.sparse import csr_matrix
    
    n = len(f)
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []

                cols.append(j)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return np.maximum(res.x, 0)
    return None
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth mirror descent (log-sum-exp approximation) with adaptive parameters
2. Periodic LP-based direction updates (state of the art)
3. Multiple sequence lengths with diverse initializations
4. Nesterov momentum for acceleration
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def compute_grad_direct(a, w):
    n = len(a)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            grad[j_start:j_end+1] += wk * a[k-j_end:k-j_start+1][::-1]
    return 2.0 * grad

def get_smooth_obj_and_grad(a, t):
    n = len(a)
    conv = fast_convolve(a)
    max_conv = conv.max()
    
    if t < 1e5:
        scaled = t * (conv - max_conv)
        exp_sum = np.exp(scaled).sum()
        if exp_sum < 1e-300:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp(scaled) / exp_sum
        obj = max_conv + np.log(exp_sum) / t
    else:
        w = np.zeros(len(conv))
        max_idx = np.where(conv >= max_conv * 0.999)[0]
        if len(max_idx) == 0:
            max_idx = np.array([np.argmax(conv)])
        w[max_idx] = 1.0 / len(max_idx)
        obj = max_conv
    
    grad = compute_grad_direct(a, w)
    return obj, grad, conv

def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k."""
    n = len(f)
    
    # Build sparse constraint matrix
    # A[k, j] = f[k-j] if 0 <= k-j < n
    rows = np.repeat(np.arange(n), n) + np.tile(np.arange(n), n)
    cols = np.repeat(np.arange(n), n)
    vals = np.tile(f, n)
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    c = -np.ones(n)
    bounds = [(0, None)] * n
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    
    if res.success:
        return np.maximum(res.x, 0)
    return None

def optimize_sequence(a0, time_limit, use_lp=True):
    """
    Optimize the sequence using smooth mirror descent with optional LP direction.
    a0 is a non-negative array (will be normalized to sum=1).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    eta = 0.5  # learning rate
    t = 20.0  # temperature for log-sum-exp
    momentum = np.zeros(n)  # for Nesterov
    gamma = 0.9  # momentum coefficient
    
    t0 = time.monotonic()
    patience = 0
    max_patience = 60
    lp_interval = 100  # do LP every this many iterations
    count = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute smooth objective and gradient
        obj, grad, conv = get_smooth_obj_and_grad(a, t)
        raw_obj = 2.0 * n * conv.max()
        
        if raw_obj < best_obj - 1e-10:
            best_obj = raw_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        # Nesterov momentum
        momentum = gamma * momentum + (1-gamma) * grad
        grad_eff = grad + gamma * (grad - momentum / (1-gamma)) if gamma > 0 else grad
        # Simpler: just use momentum as the gradient
        grad_eff = momentum
        
        # Mirror descent update (multiplicative)
        # a <- a * exp(-eta * (grad_eff - mean(grad_eff)))
        centered = grad_eff - grad_eff.mean()
        
        # Adaptive step size: scale eta so that the update doesn't change a too much
        max_change = eta * np.abs(centered).max()
        if max_change > 1.0:
            eta *= 0.8
            max_change = eta * np.abs(centered).max()
        
        log_factor = -eta * centered
        log_factor -= log_factor.max()
        a = a * np.exp(log_factor)
        a = np.maximum(a, 1e-30)
        a /= a.sum()
        
        # Periodic LP direction step
        if use_lp and count % lp_interval == lp_interval - 1 and n <= 1024:
            M = conv.max()
            g = solve_lp_direction(a, M)
            if g is not None and g.sum() > 0:
                g /= g.sum()  # normalize to sum=1
                # Mix
                mix = 0.3
                a_new = (1-mix) * a + mix * g
                a_new = np.maximum(a_new, 0)
                a_new /= a_new.sum()
                # Check if it improved
                conv_new = fast_convolve(a_new)
                if 2*n*conv_new.max() < raw_obj:
                    a = a_new
                else:
                    pass  # keep a
        
        count += 1
        
        # Adjust parameters
        if patience == max_patience:
            patience = 0
            eta *= 0.7
            t = min(t * 1.5, 500.0)
            if eta < 0.01:
                gamma = max(gamma * 0.9, 0.5)
                eta = 0.3
        
        if patience < max_patience // 3:
            # Making progress, keep going
            t = min(t * 1.02, 200.0)
    
    return best_a, best_obj
```

Let me now think about the LP more carefully. The LP is:
- Maximize sum(g)
- Subject to: (f*g)[k] <= M for k=0,...,2n-2
- And: g >= 0

Where M = max(f*f).

The dual of this LP is:
- Minimize M * sum(λ_k)
- Subject to: sum_k λ_k * f[k-j] >= 1 for j=0,...,n-1
- And: λ_k >= 0

The dual might be easier to solve since it has 2n-1 variables and n constraints (vs. n variables and 2n-1 constraints in the primal). For large n, the primal has fewer variables, so it might be better to solve the primal.

Actually, for HiGHS, either formulation should work fine. Let me stick with the primal.

One important observation: the LP solution g will typically have many zero components (due to the non-negativity constraints being active). This means the direction g is "sparse" or "structured", which could be useful.

Let me also think about **what happens after the LP step**. We have:
- Current f with ||f*f||_inf = M
- LP solution g with ||f*g||_inf <= M and sum(g) maximal

If we set h = αf + (1-α)g for some α in [0,1], then:
- sum(h) = α*sum(f) + (1-α)*sum(g)
- We want to choose α such that the objective 2n*max(h*h)/sum(h)^2 is minimized.

Note that h*h = α²(f*f) + 2α(1-α)(f*g) + (1-α)²(g*g).

We know max(f*f) = M and max(f*g) <= M, but we don't know max(g*g). If max(g*g) is also bounded by M, then max(h*h) <= M for all α, and we should take α=0 (pure g). But g*g might have a larger max.

So the optimal α depends on ||g*g||_inf. We can do a 1D line search over α to minimize the objective.

Let me implement this:

```python
def lp_update_step(f, time_budget=1.0):
    """
    Do one LP-based update step.
    Returns updated sequence and new objective.
    """
    n = len(f)
    f_sum = f.sum()
    conv_f = fast_convolve(f)
    M = conv_f.max()
    
    # Solve LP
    g = solve_lp_direction(f, M)
    if g is None or g.sum() < 1e-10:
        return f, 2*n*M/f_sum**2
    
    g_sum = g.sum()
    
    # Line search over alpha: h = (1-alpha)*f + alpha*g (normalized)
    # Actually, let's parameterize as: h = (1-t)*f/f_sum + t*g/g_sum (so sum(h)=1)
    # Then h*h = (1-t)^2 (f*f)/f_sum^2 + 2t(1-t)(f*g)/(f_sum*g_sum) + t^2 (g*g)/g_sum^2
    
    f_norm = f / f_sum
    g_norm = g / g_sum
    
    best_t = 0
    best_obj = 2*n*M  # current objective (normalized)
    
    for t in np.linspace(0, 1, 20):
        h = (1-t) * f_norm + t * g_norm
        h = np.maximum(h, 0)
        h_sum = h.sum()
        if h_sum < 1e-10:
            continue
        h /= h_sum
        conv_h = fast_convolve(h)

        if obj < best_obj:

        h = np.maximum(h, 0)
        h /= h.sum()
        return h, best_obj
    else:
        return f, best_obj
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using hybrid optimization:
- Smooth mirror descent with Nesterov momentum
- Periodic LP-based direction (state of the art) with line search
- Multiple sequence lengths and initializations
- Adaptive parameter scheduling
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# === Helper Functions ===

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def compute_grad_direct(a, w):
    n = len(a)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            grad[j_start:j_end+1] += wk * a[k-j_end:k-j_start+1][::-1]
    return 2.0 * grad

def smooth_max_and_grad(a, t):
    """Returns (smoothed_max, gradient, raw_max, conv)."""
    n = len(a)
    conv = fast_convolve(a)
    max_conv = conv.max()
    
    if t < 1e5:
        scaled = t * (conv - max_conv)
        e = np.exp(scaled)
        e_sum = e.sum()
        if e_sum < 1e-300:
            w = np.zeros(2*n-1)
            w[np.argmax(conv)] = 1.0
        else:
            w = e / e_sum
        smooth_val = max_conv + np.log(e_sum) / t
    else:
        w = np.zeros(2*n-1)
        threshold = max_conv * 0.999
        max_idx = np.where(conv >= threshold)[0]
        w[max_idx] = 1.0 / len(max_idx)
        smooth_val = max_conv
    
    grad = compute_grad_direct(a, w)
    return smooth_val, grad, max_conv, conv

def solve_lp(f, M):
    """LP: max sum(g) s.t. g>=0, (f*g)[k] <= M for all k."""
    n = len(f)
    rows = np.repeat(np.arange(n), n) + np.tile(np.arange(n), n)
    cols = np.repeat(np.arange(n), n)
    vals = np.tile(f, n)
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    c = -np.ones(n)
    bounds = [(0, None)] * n
    b_ub = np.full(2*n-1, M)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if res.success:
        return np.maximum(res.x, 0)
    return None

def lp_step(f):
    """
    One LP-based update step with line search.
    Returns (new_f, improved).
    """
    n = len(f)
    f = f / f.sum()  # ensure normalized
    conv_f = fast_convolve(f)
    M = conv_f.max()
    
    g = solve_lp(f, M)
    if g is None or g.sum() < 1e-10:
        return f, False
    
    g = g / g.sum()
    
    # Line search
    best_t = 0.0
    best_obj = 2.0 * n * M
    for t in np.linspace(0.05, 1.0, 15):
        h = (1-t)*f + t*g
        h = np.maximum(h, 0)
        hs = h.sum()
        if hs < 1e-10:
            continue
        h = h / hs
        obj = 2.0 * n * fast_convolve(h).max()
        if obj < best_obj:
            best_obj = obj
            best_t = t
    
    if best_t > 0:
        h = (1-best_t)*f + best_t*g
        h = np.maximum(h, 0)
        h /= h.sum()
        return h, True
    return f, False

def gradient_steps(f, time_limit, t_param=30.0, eta0=0.5):
    """
    Run gradient-based optimization for the given time.
    Returns (best_f, best_obj).
    """
    f = np.maximum(f.astype(np.float64), 0).copy()
    n = len(f)
    f /= f.sum()
    
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    
    eta = eta0
    t = t_param
    momentum = np.zeros(n)
    gamma = 0.85
    patience = 0
    max_pat = 50
    
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        smooth_val, grad, raw_max, conv = smooth_max_and_grad(f, t)
        obj = 2.0 * n * raw_max
        
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_f = f.copy()
            patience = 0
        else:
            patience += 1
        
        # Momentum
        momentum = gamma * momentum + (1-gamma) * grad
        
        # Mirror descent update
        centered = momentum - momentum.mean()
        scale = eta * np.abs(centered).max()
        if scale > 2.0:
            eta *= 0.8
        elif scale < 0.01 and patience < max_pat // 2:
            eta = min(eta * 1.1, 2.0)
        
        log_mult = -eta * centered
        log_mult -= log_mult.max()
        f = f * np.exp(log_mult)
        f = np.maximum(f, 1e-30)
        f /= f.sum()
        
        if patience >= max_pat:
            patience = 0
            eta *= 0.6
            t = min(t * 1.5, 500.0)
            if eta < 0.005:
                gamma = min(gamma + 0.05, 0.95)
                eta = 0.3
            t = min(t * 1.3, 300.0)
    
    return best_f, best_obj
