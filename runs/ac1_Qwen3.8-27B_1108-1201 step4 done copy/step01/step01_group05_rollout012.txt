import numpy as np

def ideal_flat_conv(n, L):
    """Compute the ideal sequence that would give flat convolution."""
    # Target: c[k] = 1/(2n-1) for k=0,...,2n-2
    # DFT of target: C[m] = sum_{k=0}^{2n-2} (1/(2n-1)) * exp(-2pi i m k / L)
    #             = (1/(2n-1)) * sum_{k=0}^{2n-2} exp(-2pi i m k / L)
    #             = (1/(2n-1)) * (1 - exp(-2pi i m (2n-1)/L)) / (1 - exp(-2pi i m / L))
    
    m = np.arange(L)
    # Handle m=0 separately
    C = np.ones(L, dtype=complex)
    C[0] = 1.0  # (1/(2n-1)) * (2n-1) = 1
    
    for idx in range(1, L):
        numerator = 1.0 - np.exp(-2j * np.pi * idx * (2*n-1) / L)
        denominator = 1.0 - np.exp(-2j * np.pi * idx / L)
        C[idx] = (1.0 / (2*n-1)) * numerator / denominator
    
    # |A[m]|^2 = C[m] (real part, since C should be real for this target)
    # Actually C[m] should be real because the target is real and symmetric... 
    # Wait, the target c[k] for k=0,...,2n-2 is NOT symmetric in general.
    # c[k] = sum_j a_j a_{k-j}. For the target to be flat (constant), it IS symmetric.
    # So DFT(c) should be real (up to numerical error).
    
    C_real = np.real(C)
    # C_real should be >= 0 for all m
    C_real = np.maximum(C_real, 0)
    
    # |A[m]| = sqrt(C_real[m])
    # Phase: for real a, A[L-m] = conj(A[m]), so phase should be symmetric
    
    # For the "ideal" solution, we want a to be real, so:
    # A[m] = sqrt(C_real[m]) * exp(-i * phi[m]) where phi is the appropriate phase
    # For a real signal, A[m] and A[L-m] are conjugates.
    # The phase that makes a real is: phi[m] = -pi*m*(n-1)/L (linear phase, centered at (n-1)/2)
    # Actually, for a non-symmetric real signal, the phase is more complex.
    
    # Let's just take the real part of the IDFT of sqrt(C_real):
    A = np.sqrt(C_real).astype(complex)
    a = np.real(np.ifft(A))
    
    return a  # Take first n elements
