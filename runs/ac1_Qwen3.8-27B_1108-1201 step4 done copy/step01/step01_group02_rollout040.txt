```python
    n = len(a)

    # Use FFT for speed
    N = 2 * (2 * n - 1)  # zero-pad to avoid circular convolution
```

```python
"""
Multi-phase optimization for minimizing peak auto-convolution of non-negative sequences.
Uses soft-max gradient descent (FFT-accelerated), LP refinement, and peak-chasing local search.
Key insight: optimal sequences are NON-symmetric (symmetric sequences have score >= 2).
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.2):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) * 0.4  # slightly off-center to break symmetry
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_blackman(n):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))


def initial_asymmetric_bump(n, rng, center_frac=0.4, width_frac=0.15):
    t = np.arange(n, dtype=np.float64)
    center = center_frac * (n - 1)
    width = width_frac * n
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    # Add a smaller secondary bump to break symmetry
    center2 = (1.0 - center_frac) * (n - 1) * 0.7
    width2 = width * 0.6
    a += 0.3 * np.exp(-0.5 * ((t - center2) / width2) ** 2)
    return np.maximum(0, a)


def initial_random_nonneg(n, rng, n_bumps=4):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(2, n // 15), max(3, n // 4)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    return np.maximum(0, a)


def get_fft_size(n):
    N = 1
    while N < 3 * n:
        N *= 2
    return N


def compute_conv_fft(a, N):
    A = np.fft.rfft(a, N)
    conv = np.fft.irfft(A * A, N)
    return conv[:2 * len(a) - 1]


def compute_grad_fft(p, a, N):
    n = len(a)
    P = np.fft.rfft(p, N)
    A_rev = np.fft.rfft(a[::-1], N)
    conv_pa = np.fft.irfft(P * A_rev, N)
    return conv_pa[n - 1:2 * n - 2]


def softmax_grad_descent(a, time_budget=60.0, verbose=True):
    start_time = time.monotonic()
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    N = get_fft_size(n)

    best_a = a.copy()
    best_score = score_sequence(a)

    lr = 0.3 * n
    stagnant = 0
    it = 0

    while time.monotonic() - start_time < time_budget:
        conv = compute_conv_fft(a, N)
        score = 2.0 * n * conv.max()

        if score < best_score - 1e-15:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it < 10 or it % 500 == 0):
                print(f"    SGD iter {it:5d}: score={score:.6f}")
        else:
            stagnant += 1

        if stagnant > 500:
            # Add small perturbation to escape
            a = a * np.exp(np.random.default_rng(it).normal(0, 0.05, n))
            a = normalize(a)
            stagnant = 0
            continue

        # Adaptive beta: start moderate, increase over time
        progress = min(1.0, (time.monotonic() - start_time) / time_budget)
        beta = n * (0.5 + 4.0 * progress)

        # Softmax weights (numerically stable)
        conv_max = conv.max()
        p = np.exp(beta * (conv - conv_max))
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.zeros(2 * n - 1)
            p[np.argmax(conv)] = 1.0
        else:
            p /= p_sum

        # Gradient via FFT
        grad = compute_grad_fft(p, a, N)

        # Multiplicative update
        step = lr * grad
        step = np.clip(step, -5.0, 5.0)
        a = a * np.exp(-step)
        a = np.maximum(a, 1e-30)
        a = normalize(a)

        # Slowly decay learning rate
        if it % 1000 == 999:
            lr *= 0.9

        it += 1

    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    if verbose:
        print(f"    SGD done: best={best_score:.6f} after {it} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def build_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    row_idx = []
    col_idx = []
    data = []
    for k in range(n_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            row_idx.append(k)
            col_idx.append(j)
            data.append(a[j])
    A = csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))
    return A


def solve_lp_step(a, M, timeout=20.0):
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def lp_refine(a, time_limit_s=60.0, verbose=True):
    start_time = time.monotonic()
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    if verbose:
        print(f"    LP start: score={best_score:.6f}, n={n}")

    stagnant = 0
    it = 0
    while time.monotonic() - start_time < time_limit_s:
        conv = np.convolve(a, a)
        score = 2.0 * n * conv.max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        if stagnant > 20:
            break

        M = conv.max()
        remaining = time_limit_s - (time.monotonic() - start_time)
        if remaining < 3.0:
            break
        timeout = max(2.0, min(30.0, remaining * 0.4))
        g = solve_lp_step(a, M, timeout=timeout)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            break
        g = g / sum_g
        alpha = min(0.5, max(0.05, 1.0 / (1.0 + 0.2 * it)))
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = normalize(a_new)
        new_score = score_sequence(a_new)
        if new_score <= score + 1e-14:
            a = a_new
        else:
            alpha2 = alpha * 0.3
            a_new = (1.0 - alpha2) * a + alpha2 * g
            a_new = normalize(a_new)
            a = a_new
        it += 1

    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    if verbose:
        print(f"    LP done: best={best_score:.6f} after {it+1} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def peak_chasing(a, time_budget=30.0, verbose=True):
    """Directly chase and reduce the peak of the convolution."""
    start_time = time.monotonic()
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)

    it = 0
    stagnant = 0
    rng = np.random.default_rng(123)

    while time.monotonic() - start_time < time_budget:
        conv = np.convolve(a, a)
        k_star = int(np.argmax(conv))
        C_max = conv[k_star]

        if 2.0 * n * C_max < best_score - 1e-14:
            best_score = 2.0 * n * C_max
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        if stagnant > 200:
            break

        # Find indices contributing to the peak
        j_start = max(0, k_star - n + 1)
        j_end = min(n, k_star + 1)

        # Reduce elements contributing to the peak, proportional to their partner's value
        alpha = 0.1
        for j in range(j_start, j_end):
            partner = k_star - j
            if 0 <= partner < n:
                a[j] *= (1.0 - alpha * a[partner] / (C_max + 1e-30))

        # Also try increasing elements far from the peak to redistribute mass
        if it % 5 == 0:
            # Add small mass at random positions far from peak contributors
            for _ in range(3):
                pos = int(rng.integers(0, n))
                # Check if this position contributes to the peak
                if k_star - pos < 0 or k_star - pos >= n:
                    a[pos] += 0.001 * a.sum() / n

        a = np.maximum(a, 0.0)
        a = normalize(a)
        it += 1

    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    if verbose:
        print(f"    PeakChase done: best={best_score:.6f} after {it} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def local_search_refine(a, time_limit_s=60.0, rng=None, verbose=True):
    if rng is None:
        rng = np.random.default_rng(0)
    start_time = time.monotonic()
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)

    accept_count = 0
    while time.monotonic() - start_time < time_limit_s:
        # Random segment perturbation
        width = max(2, int(rng.integers(n // 300, n // 6)))
        start_pos = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start_pos + width)
        candidate = a.copy()
        candidate[start_pos:stop] *= np.exp(rng.normal(0.0, 0.08, size=stop - start_pos))

        # Occasionally try a global shift (breaks symmetry)
        if rng.random() < 0.2:
            shift = int(rng.integers(-n // 10, n // 10 + 1))
            candidate = np.roll(candidate, shift)
            # Fade edges to avoid wrap-around artifacts
            fade = min(n // 10, abs(shift)) + 1
            if shift > 0:
                candidate[:fade] *= np.linspace(0, 1, fade)
                candidate[-fade:] *= np.linspace(1, 0, fade)
            elif shift < 0:
                candidate[-fade:] *= np.linspace(0, 1, fade)
                candidate[:fade] *= np.linspace(1, 0, fade)

        candidate = normalize(candidate)
        score = score_sequence(candidate)
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if verbose and accept_count % 30 == 0:
                print(f"    LS accept {accept_count}: score={best_score:.6f}")

    if verbose:
        print(f"    LS done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def propose_candidate():
    """
    Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses soft-max gradient descent, LP refinement, peak-chasing, and local search.
    Explores multiple lengths and initializations. Optimal sequences are non-symmetric.
    """
    rng = np.random.default_rng(666)
    total_start = time.monotonic()
    budget_s = 1800.0
    deadline = total_start + budget_s - 20.0

    best_score = float('inf')
    best_seq = None

    # Try to load previous best
    prev_seq = None
    try:
        if 'height_sequence_1' in dir() or 'height_sequence_1' in globals():
            prev_seq = np.array(globals()['height_sequence_1'], dtype=np.float64)
            if len(prev_seq) > 0 and np.all(np.isfinite(prev_seq)):
                prev_score = score_sequence(prev_seq)
                print(f"Previous best: score={prev_score:.6f}, n={len(prev_seq)}")
                best_score = prev_score
                best_seq = prev_seq.copy()
    except Exception:
        pass

    # Define configurations: (length, init_type)
    configs = [
        (500, 'raised_cosine'),
        (500, 'asymmetric'),
        (700, 'raised_cosine'),
        (700, 'gaussian'),
        (1000, 'raised_cosine'),
        (500, 'gaussian'),
        (600, 'asymmetric'),
        (800, 'gaussian'),
    ]

    # If we have a previous best, include it as a config
    if best_seq is not None:
        configs.insert(0, (len(best_seq), 'previous'))

    for cfg_idx, (n, init_type) in enumerate(configs):
        remaining = deadline - time.monotonic()
        if remaining < 60.0:
            break

        # Allocate time: favor earlier configs more
        n_configs_left = len(configs) - cfg_idx
        time_limit = min(remaining * 0.85 / max(1, n_configs_left) * 1.5, 300.0)
        time_limit = max(30.0, time_limit)

        print(f"\n=== Config {cfg_idx+1}/{len(configs)}: n={n}, init={init_type}, time={time_limit:.0f}s ===")

        # Initialize
        if init_type == 'previous' and best_seq is not None and len(best_seq) == n:
            a = best_seq.copy()
        elif init_type == 'raised_cosine':
            a = initial_raised_cosine(n)
        elif init_type == 'gaussian':
            a = initial_gaussian(n, sigma_frac=0.2)
        elif init_type == 'asymmetric':
            a = initial_asymmetric_bump(n, rng,
                                        center_frac=0.3 + 0.2 * rng.random(),
                                        width_frac=0.1 + 0.1 * rng.random())
        else:
            a = initial_raised_cosine(n)

        a = normalize(a)
        init_score = score_sequence(a)
        print(f"    Init score: {init_score:.6f}")

        # Phase 1: Soft-max gradient descent (fast, many iterations)
        t1 = min(time_limit * 0.35, 80.0)
        a, s1 = softmax_grad_descent(a, time_budget=t1, verbose=True)

        # Phase 2: LP refinement
        t2 = min(time_limit * 0.30, 60.0)
        a, s2 = lp_refine(a, time_limit_s=t2, verbose=True)

        # Phase 3: Soft-max gradient descent again
        t3 = min(time_limit * 0.20, 50.0)
        a, s3 = softmax_grad_descent(a, time_budget=t3, verbose=True)

        # Phase 4: Peak chasing
        t4 = min(time_limit * 0.10, 20.0)
        a, s4 = peak_chasing(a, time_budget=t4, verbose=True)

        # Phase 5: Local search
        t5 = max(5.0, time_limit - t1 - t2 - t3 - t4)
        t5 = min(t5, 30.0)
        a, s5 = local_search_refine(a, time_limit_s=t5, rng=rng, verbose=True)

        # Check if improved
        if s5 < best_score:
            best_score = s5
            best_seq = a.copy()
            elapsed = time.monotonic() - total_start
            print(f"  *** NEW BEST: score={best_score:.6f}, n={n}, elapsed={elapsed:.1f}s")

    # Final polishing on the best sequence
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30.0:
            print(f"\n=== Final polishing (remaining={remaining:.0f}s) ===")
            n = len(best_seq)

            # LP polish
            t_lp = min(remaining * 0.3, 60.0)
            a, s = lp_refine(best_seq, time_limit_s=t_lp, verbose=True)
            if s < best_score:
                best_score = s
                best_seq = a.copy()

            # SGD polish
            remaining = deadline - time.monotonic()
            if remaining > 20.0:
                t_sgd = min(remaining * 0.3, 40.0)
                a, s = softmax_grad_descent(best_seq, time_budget=t_sgd, verbose=True)
                if s < best_score:
                    best_score = s
                    best_seq = a.copy()

            # Peak chase polish
            remaining = deadline - time.monotonic()
            if remaining > 15.0:
                t_pc = min(remaining * 0.2, 20.0)
                a, s = peak_chasing(best_seq, time_budget=t_pc, verbose=True)
                if s < best_score:
                    best_score = s
                    best_seq = a.copy()

            # Final local search
            remaining = deadline - time.monotonic()
            if remaining > 10.0:
                t_ls = min(remaining * 0.8, 30.0)
                a, s = local_search_refine(best_seq, time_limit_s=t_ls, rng=rng, verbose=True)
                if s < best_score:
                    best_score = s
                    best_seq = a.copy()

    # Fallback
    if best_seq is None:
        best_seq = initial_raised_cosine(500)
        best_score = score_sequence(best_seq)
        print(f"  Fallback: score={best_score:.6f}")

    # Verify
    total_time = time.monotonic() - total_start
    print(f"\nFinal result: score={best_score:.6f}, n={len(best_seq)}, time={total_time:.1f}s")
    verify_score = evaluate_sequence([float(x) for x in best_seq])
    print(f"Verified score via evaluate_sequence: {verify_score:.6f}")

    return [float(x) for x in best_seq]
```
