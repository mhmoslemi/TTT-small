        return np.inf
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * c.max() / (s ** 2)

def softmax(theta):
    t = theta - theta.max()
    e = np.exp(t)
    s = e.sum()
    if s == 0:
        return np.ones(len(theta)) / len(theta)
    return e / s

def make_obj_grad(n, S, tau):
    """Create objective and gradient functions for L-BFGS-B on theta."""
    
    def obj(th):
        a = softmax(th) * S
        c = fft_convolve(a, a)
        tc = tau * c
        mc = tc.max()
        return float((mc + np.log(np.exp(tc - mc).sum())) / tau)
    
    def grad(th):
        a = softmax(th) * S
        c = fft_convolve(a, a)
        tc = tau * c
        mc = tc.max()
        ev = np.exp(tc - mc)
        w = ev / ev.sum()
        # Gradient w.r.t. a: g_a[j] = 2 * (w * a_rev)[j + n - 1]
        a_rev = a[::-1]
        corr = fft_convolve(w, a_rev)
        g_a = 2.0 * corr[n-1:2*n-1]
        # Gradient w.r.t. theta
        inner = np.dot(g_a, a) / S
        return a * (g_a - inner)
    
    return obj, grad
