def _convolve_fft(a, b):
    """Compute linear convolution using FFT."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

def _cross_correlate_fft(a, b):
    """Compute cross-correlation: (a ⋆ b)[k] = sum_i a[i] * b[k+i]."""
    # cross-correlation = convolution of a with reversed b
    b_rev = b[::-1]
    return _convolve_fft(a, b_rev)

def _simplex_project(v, target=1.0):

            lo = mid
        else:

```python
"""
FFT-accelerated log-sum-exp gradient descent with mirror descent updates
to minimize the normalized autocorrelation peak of a non-negative sequence.
Uses adaptive temperature, exponential updates, and periodic LP polishing.
"""

import numpy as np
import time


def _fft_convolve(a, b):
    """Compute linear convolution of a and b using FFT."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]


def _compute_gradient(a, p):
    """
    Compute grad[i] = 2 * sum_k p[k] * a[k-i] for i=0,...,n-1.
    This is the gradient of (1/tau)*log(sum_k exp(tau*b[k])) w.r.t. a[i].
    p has length 2n-1, a has length n.
    """
    n = len(a)
    p_rev = p[::-1].copy()
    total_len = n + len(p) - 1  # = n + (2n-1) - 1 = 3n-2
    size = 1 << (total_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv = np.fft.irfft(fa * fp, size)[:total_len]
    # grad[i] = 2 * conv[2n-2-i] for i=0,...,n-1
    # conv[2n-2-i] for i=0,...,n-1 gives indices 2n-2, 2n-3, ..., n-1
    grad = conv[n-1:2*n-1][::-1]
    return 2.0 * grad


def _score(a):
    """Compute 2*n*max(convolve(a,a)) for normalized a (sum=1)."""
    n = len(a)
    b = _fft_convolve(a, a)
    return 2.0 * n * np.max(b)


def _simplex_project(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target} in O(n log n)."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.full(n, target / n)
    rho = idx[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _mirror_descent_step(a, eta, grad):
    """Exponential update: a_new ∝ a * exp(-eta * grad), normalized."""
    log_update = -eta * grad
    # Clip for numerical stability
    log_update = np.clip(log_update, -50.0, 50.0)
    a_new = a * np.exp(log_update)
    s = np.sum(a_new)
    if s < 1e-300:
        return a.copy()
    a_new /= s
    return a_new


def _run_mgd(a, deadline, tau_start=1.0, tau_end=None, eta_start=1.0, eta_end=0.01,
             reheat_every=500, reheat_scale=0.5, max_no_improve=1000):
    """
    Mirror descent + gradient (log-sum-exp) optimization.
    Returns (best_a, best_score).
    """
    n = len(a)
    if tau_end is None:
        tau_end = n
    
    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(n) / n
    else:
        a /= S
    
    best = a.copy()
    best_score = _score(best)
    
    eta = eta_start
    tau = tau_start
    no_improve = 0
    step = 0
    improves_total = 0
    
    while time.monotonic() < deadline:
        # Compute b and softmax weights
        b = _fft_convolve(a, a)
        # Softmax with temperature tau
        tb = tau * b
        tb_max = np.max(tb)
        exp_tb = np.exp(tb - tb_max)
        p = exp_tb / np.sum(exp_tb)
        
        # Compute gradient
        grad = _compute_gradient(a, p)
        
        # Mirror descent update
        a_new = _mirror_descent_step(a, eta, grad)
        
        # Ensure non-negativity (should be automatic with exp, but clip for safety)
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-300:
            break
        a_new /= s
        
        # Evaluate
        b_new = _fft_convolve(a_new, a_new)
        score_new = 2.0 * n * np.max(b_new)
        
        if score_new < best_score - 1e-12:
            best_score = score_new
            best = a_new.copy()
            no_improve = 0
            improves_total += 1
        else:
            no_improve += 1
        
        a = a_new
        
        # Adaptive schedule
        step += 1
        progress = min(1.0, step / 50000.0)  # normalize over expected steps
        tau = tau_start + (tau_end - tau_start) * progress
        eta = eta_start + (eta_end - eta_start) * progress
        
        # Reheating
        if no_improve >= reheat_every:
            eta *= reheat_scale
            no_improve = 0
            if eta < 1e-6:
                break
        
        # Hard limit on no improvement
        if no_improve >= max_no_improve:
            break
    
    return best, best_score


def _lp_polish(a, deadline, M=100):
    """
    LP-based polishing: linearize top-M peaks and solve LP.
    Returns (best_a, best_score).
    """
    try:
        from scipy.optimize import linprog
    except ImportError:
        return a, _score(a)
    
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    a /= S
    
    best = a.copy()
    best_score = _score(best)
    
    for iteration in range(20):
        if time.monotonic() >= deadline:
            break
        
        # Compute b and find top M peaks
        b = _fft_convolve(a, a)
        peak_idx = np.argpartition(b, -M)[-M:]
        peak_idx = peak_idx[np.argsort(b[peak_idx])[::-1]]
        
        # Build LP: minimize t
        # s.t. b[k] + sum_i grad_bk[i] * (a_new[i] - a[i]) <= t for k in peak_idx
        #      sum(a_new) = 1
        #      a_new >= 0
        #
        # Variables: x = [a_new[0], ..., a_new[n-1], t] (total n+1 variables)
        # 
        # Constraints (inequality, <= form):
        # For each k in peak_idx: sum_i grad_bk[i] * a_new[i] - t <= b[k] - sum_i grad_bk[i] * a[i]
        #
        # grad_bk[i] = 2 * a[k-i] (for valid k-i)
        
        # Build constraint matrix
        # c_k^T * a_new - t <= rhs_k
        # In scipy linprog form: A_ub @ x <= b_ub
        # x = [a_new, t], so A_ub has shape (M, n+1)
        
        A_ub = np.zeros((M, n + 1))
        b_ub = np.zeros(M)
        
        for li, k in enumerate(peak_idx):
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    A_ub[li, i] = 2.0 * a[j]
            A_ub[li, n] = -1.0  # coefficient of t
            b_ub[li] = b[k] - np.sum(2.0 * a[j] * a[j] for j in range(n) if 0 <= k - j < n)
            # Actually: b[k] = sum_i a[i]*a[k-i], and grad_bk = 2*a[k-i]
            # Linearization: b_new[k] ≈ b[k] + grad_bk^T (a_new - a)
            # = b[k] + sum_i 2*a[k-i]*(a_new[i] - a[i])
            # Constraint: b[k] + sum_i 2*a[k-i]*a_new[i] - 2*sum_i a[k-i]*a[i] <= t
            # = sum_i 2*a[k-i]*a_new[i] - t <= 2*sum_i a[k-i]*a[i] - b[k]
            # Note: 2*sum_i a[k-i]*a[i] = 2*b[k] (since b[k] = sum_i a[i]*a[k-i])
            # So RHS = 2*b[k] - b[k] = b[k]
        
        # Wait, let me redo this.
        # b[k] = sum_i a[i] * a[k-i]
        # db[k]/da[i] = a[k-i] + a[k-i] = 2*a[k-i] (for i ≠ k-i; if i = k-i, it's a[i] not 2a[i])
        # Actually: b[k] = sum_i a[i]*a[k-i]. If we differentiate w.r.t. a[j]:
        # db[k]/da[j] = a[k-j] + a[j-k+...]... 
        # More carefully: b[k] = sum_i a[i]*a[k-i]
        # db[k]/da[j] = a[k-j] (from the first factor, when i=j) + a[j] (from the second factor, when k-i=j, i.e., i=k-j)
        # = a[k-j] + a[k-j] = 2*a[k-j] (when j ≠ k-j, i.e., 2j ≠ k)
        # = a[j] (when j = k-j, i.e., 2j = k, only if k is even)
        # 
        # For simplicity, I'll use the approximation grad_bk[i] = 2*a[k-i] (which is exact except at the diagonal).
        #
        # Linearization: b_new[k] ≈ b[k] + sum_i 2*a[k-i] * (a_new[i] - a[i])
        # Constraint: b_new[k] <= t
        # => b[k] + sum_i 2*a[k-i]*a_new[i] - 2*sum_i a[k-i]*a[i] <= t
        # => sum_i 2*a[k-i]*a_new[i] - t <= 2*b[k] - b[k] = b[k]
        
        # So: A_ub[li, :n] = 2*a[k-i] for i=0,...,n-1 (with a[k-i]=0 if k-i out of range)
        #     A_ub[li, n] = -1
        #     b_ub[li] = b[k]
        
        # Let me rebuild properly:
        A_ub = np.zeros((M, n + 1))
        b_ub = np.zeros(M)
        
        for li, k in enumerate(peak_idx):
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    A_ub[li, i] = 2.0 * a[j]
            A_ub[li, n] = -1.0
            b_ub[li] = b[k]
        
        # Equality constraint: sum(a_new) = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        # Objective: minimize t
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        
        # Bounds: a_new[i] >= 0, t unconstrained (but t >= 0 in practice)
        bounds = [(0, None)] * n + [(0, None)]
        
        # Solve
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        
        if result.success:
            a_new = result.x[:n]
            a_new = np.maximum(a_new, 0.0)
            s = np.sum(a_new)
            if s > 1e-10:
                a_new /= s
                score_new = _score(a_new)
                if score_new < best_score:
                    best_score = score_new
                    best = a_new.copy()
                a = a_new
        else:
            break
    
    return best, best_score
