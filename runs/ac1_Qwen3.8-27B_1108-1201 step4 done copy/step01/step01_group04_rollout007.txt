"""
Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses analytical gradients of the peak autocorrelation (both hard-max and soft-max),
momentum-based acceleration, LP-based refinement, coordinate descent polishing,
and multi-start strategy with diverse initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=None):
    """Normalize sequence to have a specific sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def autocorr(a):
    """Compute full autocorrelation."""
    return np.convolve(a, a)


def gradient_hard(a, tau=None):
    """
    Compute gradient of the (soft-)max autocorrelation w.r.t. a.
    Returns gradient vector of same length as a.
    If tau is None, uses hard max (single peak).
    If tau is given, uses soft-max weighting.
    """
    n = len(a)
    c = np.convolve(a, a)

    if tau is None:
        # Hard max: gradient of c[k*] w.r.t. a[i] = 2*a[k*-i] (if valid)
        k_star = int(np.argmax(c))
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
    else:
        # Soft max: weight each lag by exp(tau * c[k])
        c_shifted = c - c.max()
        w = np.exp(tau * c_shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w /= w_sum
        # grad[i] = 2 * sum_k w[k] * a[k-i] (for valid k-i)
        # This is 2 * (w * a) at appropriate lags
        # w has length 2n-1, a has length n
        # For each i in [0,n-1], grad[i] = 2 * sum_{k: 0<=k-i<n} w[k] * a[k-i]
        # = 2 * sum_{j: 0<=j<n} w[i+j] * a[j]  (where j = k-i)
        grad = np.zeros(n)
        for i in range(n):
            # j ranges over [0, n-1], k = i+j ranges over [i, i+n-1]
            # We need k < 2n-1, so i+j < 2n-1, which is always true for i,n <= 2n-2
            k_start = i
            k_end = i + n
            if k_end > len(w):
                k_end = len(w)
            if k_start < len(w):
                grad[i] = 2.0 * np.dot(w[k_start:k_end], a[:k_end - k_start])

    return grad


def gradient_fast(a, tau=None):
    """
    Faster gradient using the structure of the problem.
    For soft-max: grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    This is a cross-correlation that can be computed with FFT.
    """
    n = len(a)
    c = np.convolve(a, a)

    if tau is None:
        k_star = int(np.argmax(c))
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
    else:
        c_shifted = c - c.max()
        w = np.exp(tau * c_shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w /= w_sum
        # Use FFT-based cross-correlation
        # grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
        # Let's compute this as: for each i, dot(w[i:i+n], a[:n])
        # This is the cross-correlation of w (length 2n-1) with a (length n)
        # Specifically, it's the "valid" part of the correlation
        # corr[i] = sum_j w[i+j] * a[j] for j=0..n-1
        # This is like flipping a and doing convolution, or...
        # Actually: convolve(w, a_reversed) gives sum_k w[k] * a_rev[k-i]
        # Let's just use a direct loop for n <= 1024 (fast enough)
        a_rev = a[::-1]
        # conv(w, a_rev)[i+n-1] = sum_j w[i+j] * a_rev[n-1-j] = sum_j w[i+j] * a[j]
        conv_result = np.convolve(w, a_rev, mode='full')
        # The valid part starts at index n-1
        grad = 2.0 * conv_result[n-1:n-1+n]

    return grad


def gradient_descent_step(a, lr, momentum_buf, momentum_decay, tau=None):
    """
    One step of gradient descent with momentum, projecting onto sum and non-negativity.
    """
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a_curr = normalize(a, S_t)

    grad = gradient_fast(a_curr, tau)

    # Project out the mean (to maintain sum constraint)
    grad -= grad.mean()

    # Momentum
    momentum_buf[:] = momentum_decay * momentum_buf - lr * grad

    a_new = a_curr + momentum_buf
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize(a_new, S_t)

    return a_new, momentum_buf


def lp_refine(f, M, n_vars=None):
    """
    Solve LP: max sum(g) s.t. convolve(f,g) <= M (elementwise), g >= 0.
    Returns the optimal g or None if infeasible/failed.
    """
    n = len(f)
    if n_vars is None:
        n_vars = n
    if n > 2048:
        return None

    c_obj = -np.ones(n)

    # Build constraint matrix: A[m,j] = f[m-j] if 0 <= m-j < n, else 0
    # Use sparse-like construction for efficiency
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[m, j] = f[m - j]

    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def lp_refine_fast(f, M):
    """
    Faster LP refinement using the structure of the constraint matrix.
    The matrix A is a banded Toeplitz-like matrix.
    """
    n = len(f)
    if n > 1024:
        return None

    # For smaller n, direct LP is fine
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # f[j] appears in rows j through j+n-1 (where row m has A[m,j] = f[m-j])
        # Actually A[m,j] = f[m-j] if 0 <= m-j < n
        # So for fixed j, m-j ranges over [0, n-1], i.e., m ranges over [j, j+n-1]
        for m in range(j, j + n):
            A[m, j] = f[m - j]

    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-8})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def coordinate_descent_polish(a, deadline, max_rounds=10):
    """
    Coordinate descent: for each coordinate, try a small set of values
    and keep the best one.
    """
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S_t)
    best_score = compute_obj(a)
    patience = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            # Try perturbing a[i]
            orig = a[i]
            # Try: 0, orig*0.5, orig*1.5, and a few small changes
            candidates = [0.0, orig * 0.5, orig * 1.5]
            # Also try based on what's at the peak lag
            c = np.convolve(a, a)
            k_star = int(np.argmax(c))
            j = k_star - i
            if 0 <= j < n and j != i:
                # Reducing a[i] when a[j] is large helps
                candidates.append(orig * 0.3)
                candidates.append(orig * 0.7)

            best_val = orig
            best_s = best_score
            for val in candidates:
                a[i] = val
                a = normalize(a, S_t)
                s = compute_obj(a)
                if s < best_s - 1e-10:
                    best_s = s
                    best_val = val
                a[i] = orig  # restore

            if best_val != orig:
                a[i] = best_val
                a = normalize(a, S_t)
                best_score = best_s
                improved = True

        if not improved:
            patience += 1
            if patience >= 2:
                break
        else:
            patience = 0

    return a, best_score


def optimize_from_start(a0, deadline, max_time_single=None):
    """
    Full optimization pipeline from a single starting point.
    Returns (best_sequence, best_score).
    """
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n) * 0.5
    a = normalize(a)
    score = compute_obj(a)
    best = a.copy()
    best_score = score
    S_t = np.sqrt(2.0 * n)

    if max_time_single is None:
        max_time_single = deadline - time.monotonic() - 30.0
    single_deadline = time.monotonic() + max(10.0, max_time_single)

    print(f"    n={n}, initial_score={score:.6f}")

    # Phase 1: Gradient descent with momentum (hard max)
    momentum_buf = np.zeros(n)
    momentum_decay = 0.9
    lr = 0.3
    no_improve = 0
    phase1_deadline = time.monotonic() + max(10.0, max_time_single * 0.35)

    it = 0
    while time.monotonic() < phase1_deadline and it < 50000:
        it += 1
        # Anneal learning rate
        lr = 0.3 * (0.9995 ** it)
        lr = max(lr, 0.005)

        a, momentum_buf = gradient_descent_step(a, lr, momentum_buf, momentum_decay, tau=None)

        if it % 200 == 0:
            s = compute_obj(a)
            if s < best_score - 1e-8:
                best_score = s
                best = a.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 5:
                    # Reset momentum, try soft-max
                    momentum_buf[:] = 0
                    no_improve = 0
                    phase1_deadline = time.monotonic() + max(5.0, (single_deadline - time.monotonic()) * 0.5)
                    break

    print(f"    Phase 1 (hard grad): score={best_score:.6f}, iters={it}")

    # Phase 2: Soft-max gradient descent
    if time.monotonic() < single_deadline - 10:
        momentum_buf = np.zeros(n)
        tau = 5.0
        lr = 0.2
        no_improve = 0
        phase2_deadline = time.monotonic() + max(10.0, (single_deadline - time.monotonic()) * 0.35)

        it = 0
        while time.monotonic() < phase2_deadline and it < 30000:
            it += 1
            tau = 5.0 + 20.0 * (it / 1000.0)
            tau = min(tau, 50.0)
            lr = 0.2 * (0.9997 ** it)
            lr = max(lr, 0.005)

            a, momentum_buf = gradient_descent_step(a, lr, momentum_buf, 0.9, tau=tau)

            if it % 200 == 0:
                s = compute_obj(a)
                if s < best_score - 1e-8:
                    best_score = s
                    best = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                    if no_improve > 8:
                        break

        a = best.copy()
        print(f"    Phase 2 (soft grad): score={best_score:.6f}, iters={it}")

    # Phase 3: LP-based refinement
    if n <= 768 and time.monotonic() < single_deadline - 15:
        a = best.copy()
        c = np.convolve(a, a)
        M = float(c.max())
        no_improve_lp = 0
        phase3_deadline = time.monotonic() + max(5.0, (single_deadline - time.monotonic()) * 0.25)

        it_lp = 0
        while time.monotonic() < phase3_deadline and it_lp < 100:
            it_lp += 1
            g = lp_refine_fast(a, M)
            if g is None:
                break

            # Normalize g
            g_sum = g.sum()
            if g_sum < 1e-6:
                break
            g_norm = g * (S_t / g_sum)

            # Try blending
            improved = False
            for t_blend in [0.7, 0.5, 0.3, 0.15, 0.05]:
                a_new = (1.0 - t_blend) * a + t_blend * g_norm
                a_new = np.maximum(a_new, 0.0)
                a_new = normalize(a_new, S_t)
                s_new = compute_obj(a_new)
                if s_new < best_score - 1e-10:
                    a = a_new
                    best = a.copy()
                    best_score = s_new
                    improved = True
                    break

            if not improved:
                no_improve_lp += 1
                # Small gradient nudge to escape
                a, _ = gradient_descent_step(a, 0.05, np.zeros(n), 0.0, tau=None)
                s = compute_obj(a)
                if s < best_score:
                    best = a.copy()
                    best_score = s
                    no_improve_lp = 0
                if no_improve_lp > 5:
                    break
            else:
                no_improve_lp = 0

            # Update M
            c = np.convolve(a, a)
            M = float(c.max())

        print(f"    Phase 3 (LP): score={best_score:.6f}, iters={it_lp}")

    # Phase 4: Coordinate descent polish (only for n <= 512)
    if n <= 512 and time.monotonic() < single_deadline - 5:
        a_polished, s_polished = coordinate_descent_polish(
            best, single_deadline - 2.0, max_rounds=3)
        if s_polished < best_score:
            best = a_polished
            best_score = s_polished
        print(f"    Phase 4 (coord desc): score={best_score:.6f}")

    return best, best_score


def generate_starts(rng, lengths):
    """Generate diverse starting sequences."""
    starts = []

    # From height_sequence_1 if available
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 8 and np.all(np.isfinite(gs)):
            for L in lengths:
                if L <= len(gs):
                    starts.append(gs[:L].copy())
            # Also try shuffled versions
            for L in [min(len(gs), 256), min(len(gs), 512)]:
                idx = rng.permutation(L)
                starts.append(gs[idx].copy())
    except NameError:
        pass

    # Random starts
    for n in lengths:
        starts.append(rng.random(n) * 0.5 + 0.25)

    # Structured starts
    for n in lengths:
        i = np.arange(n, dtype=np.float64)
        # Various chirps and modulations
        starts.append(0.5 + 0.5 * np.cos(2 * np.pi * i * 3.0 / n))
        starts.append(0.5 + 0.5 * np.cos(2 * np.pi * i * 7.0 / n))
        starts.append(np.abs(np.sin(2 * np.pi * i * 2.0 / n)) + 0.1)
        starts.append(np.abs(np.sin(2 * np.pi * i * 5.0 / n) + 0.3 * np.cos(2*np.pi*i*3.0/n)) + 0.05)
        # Gaussian bumps
        center = n / 2.0
        starts.append(np.exp(-0.5 * ((i - center) / (n * 0.2)) ** 2))
        # Two bumps
        starts.append(np.exp(-0.5 * ((i - n*0.25) / (n*0.1))**2) + np.exp(-0.5 * ((i - n*0.75) / (n*0.1))**2))

    # Sparse-like starts (many zeros)
    for n in [256, 512]:
        a = np.zeros(n)
        a[::3] = rng.random(n // 3 + 1) * 0.5 + 0.1
        starts.append(a[:n])

    return starts


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Uses multi-phase optimization (gradient, LP, coordinate descent)
    from multiple diverse starting points. Returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = float('inf')

    # Promising lengths to try
    lengths = [256, 384, 512, 640]

    print(f"Starting optimization, budget={budget_s}s, deadline in {deadline - t_start:.1f}s")

    # --- First pass: try each length with a couple of starts ---
    starts = generate_starts(rng, lengths)

    # Sort by length (shorter first for quick wins)
    starts.sort(key=lambda x: len(x))

    # Deduplicate roughly
    seen_sums = set()
    unique_starts = []
    for s in starts:
        key = (len(s), round(float(s.sum()), 1))
        if key not in seen_sums:
            seen_sums.add(key)
            unique_starts.append(s)
    starts = unique_starts

    print(f"Total starting sequences: {len(starts)}")

    # Allocate time: give more time to shorter sequences first
    # Then focus remaining time on the best length
    n_starts = len(starts)
    # Give each start at most 120 seconds, but proportional to remaining budget
    time_per_start = min(120.0, (deadline - t_start) / max(1, n_starts) * 0.7)

    for idx, a0 in enumerate(starts):
        if time.monotonic() > deadline - 60.0:
            break

        remaining = deadline - time.monotonic()
        # Allocate: first few get less time, later (promising) ones get more
        if idx < 4:
            t_alloc = min(time_per_start, remaining * 0.15)
        else:
            t_alloc = min(time_per_start * 1.5, remaining * 0.2)

        if t_alloc < 20.0:
            t_alloc = min(180.0, remaining * 0.3)

        t_alloc = max(15.0, t_alloc)
        if time.monotonic() + t_alloc > deadline - 30:
            t_alloc = deadline - time.monotonic() - 30
            if t_alloc < 10:
                break

        seq, score = optimize_from_start(a0, deadline, max_time_single=t_alloc)

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  [start {idx}] New best: score={best_score:.6f}, n={len(seq)}")
        else:
            print(f"  [start {idx}] score={score:.6f}, n={len(seq)} (not better)")

    # --- Second pass: focus on the best length with more diverse starts ---
    if best_seq is not None:
        best_n = len(best_seq)
        # Try a few more starts at the best length
        extra_starts = []
        for trial in range(3):
            a0 = rng.random(best_n) * 0.5 + 0.25
            extra_starts.append(a0)

        # Also try perturbed versions of the best
        for trial in range(2):
            perturbation = rng.normal(0, 0.1, best_n)
            a0 = np.maximum(best_seq + perturbation, 0)
            extra_starts.append(a0)

        print(f"\nSecond pass: {len(extra_starts)} extra starts at n={best_n}")

        for idx, a0 in enumerate(extra_starts):
            if time.monotonic() > deadline - 30.0:
                break
            remaining = deadline - time.monotonic() - 30
            t_alloc = min(180.0, remaining / max(1, len(extra_starts) - idx))
            if t_alloc < 15:
                break

            seq, score = optimize_from_start(a0, deadline, max_time_single=t_alloc)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  [extra {idx}] New best: score={best_score:.6f}")
            else:
                print(f"  [extra {idx}] score={score:.6f}")

    # --- Final polish on the best sequence ---
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"\nFinal polish on best (score={best_score:.6f}, n={len(best_seq)})...")
        remaining = deadline - time.monotonic() - 10
        # Quick gradient + LP polish
        seq, score = optimize_from_start(best_seq, deadline, max_time_single=remaining)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  Final polish: score={best_score:.6f}")

    # --- Final validation ---
    if best_seq is not None:
        final_score = compute_obj(best_seq)
        print(f"\nFinal: n={len(best_seq)}, computed_score={final_score:.6f}, tracked_best={best_score:.6f}")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence score: {eval_score:.6f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in best_seq]

    # Fallback
    print("Falling back to uniform sequence.")
    return [1.0] * 256
