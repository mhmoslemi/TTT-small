import time
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def compute_objective(z, tau):
    """Compute the smoothed objective given log-parameters z."""
    a = softmax(z)
    n = len(a)
    conv = fast_convolve(a)
    # Log-sum-exp of tau * conv
    max_conv = conv.max()
    if max_conv < 1e-15:
        return 2.0 * n * max_conv
    scaled = tau * conv
    scaled -= scaled.max()
    log_sum_exp = scaled.max() + np.log(np.exp(scaled).sum())
    return 2.0 * n * (log_sum_exp / tau)
