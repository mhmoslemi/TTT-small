    """Full convolution of a with b (or with itself if b is None) via FFT."""

    return exp_c / np.sum(exp_c)

def run_lse_adam(a_init, n, S, max_time, rng, T_init=0.05, T_min=0.001, lr_init=0.1):
    a = np.clip(a_init, 0, 1000).astype(np.float64)
    s = a.sum()
    a = (a * (S/s) if s > 1e-15 else np.full(n, S/n))
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    b1, b2, eps = 0.8, 0.999, 1e-8
    t = 0
    start = time.monotonic()
    
    _, best_score = lse_grad_and_value(a, T_init)
    best_local = a.copy()
    last_best_t = 0
    lr = lr_init
    
    while time.monotonic() - start < max_time:
        t += 1
        T = max(T_min, T_init * 0.9995**t)
        grad, cur = lse_grad_and_value(a, T)
        
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
            last_best_t = t
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        
        m_buf = b1*m_buf + (1-b1)*grad
        v_buf = b2*v_buf + (1-b2)*grad**2
        m_hat = m_buf / (1 - b1**t)
        v_hat = v_buf / (1 - b2**t)
        
        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps), S)
        lr = lr_init / (1 + 0.0005*t)
        
        # Restart if stuck
        if t - last_best_t > 10000 and rng.random() < 0.01:
            a = project_simplex(a + rng.normal(0, 0.002*S/n, n), S)
    
    return best_local, best_score
