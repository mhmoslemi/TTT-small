import numpy as np

def fft_convolve(a, b):
    """Fast linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    c = np.fft.irfft(fa * fb, size)
    return c[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score directly."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    C = fft_convolve(a, a)
    return 2.0 * n * np.max(C) / (S * S)


def grad_max_convolution(a, C, S, tau, top_k=20):

    Compute gradient of soft-max approximation to score w.r.t. a.
    Uses top-k weighted peaks for stability.

def convex_project(v, bounds_low, bounds_high, target_sum):
    """Project v onto {x: bounds_low <= x <= bounds_high, sum(x) = target_sum}."""
    n = len(v)
    lo = bounds_low
    hi = bounds_high
    
    # First, check if projecting to box already satisfies sum
    v_clip = np.clip(v, lo, hi)
    s = np.sum(v_clip)
    
    if abs(s - target_sum) < 1e-10:
        return v_clip
    
    # Binary search for lambda
    lo_lambda = -1e6
    hi_lambda = 1e6
    
    for _ in range(100):
        mid = (lo_lambda + hi_lambda) / 2.0
        x = np.clip(v - mid, lo, hi)
        s = np.sum(x)
        if s > target_sum:
            lo_lambda = mid
        else:
            hi_lambda = mid
    
    x = np.clip(v - (lo_lambda + hi_lambda) / 2.0, lo, hi)
    # Fix numerical issues
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-12:
        # Distribute remaining to elements not at boundary
        mask = (x > lo + 1e-10) & (x < hi - 1e-10)
        if np.any(mask):
            x[mask] += diff / np.sum(mask)
    return x


def adam_step(a, g, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """One step of Adam optimizer."""
    m = beta1 * m + (1 - beta1) * g
    v = beta2 * v + (1 - beta2) * (g * g)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    return a_new, m, v

def convolution_flattening(a, target_peak, iter_count, rng):
    """
    Directly flatten the convolution by reducing peak contributors
    and increasing valley contributors.
    """
    n = len(a)
    a = a.copy()
    S = np.sum(a)
    
    for it in range(iter_count):
        C = fft_convolve(a, a)
        peak = np.max(C)
        
        if peak <= target_peak:
            break
        
        # Find peaks above target
        peak_indices = np.where(C > target_peak * 1.01)[0]
        
        if len(peak_indices) == 0:
            break
        
        # For each peak, compute contributions
        total_excess = 0.0
        adjust = np.zeros(n)
        
        for k in peak_indices[:10]:  # Limit to top 10 peaks
            excess = C[k] - target_peak
            total_excess += excess
            # Contributors: a[i] * a[k-i] for valid i
            i_range = range(max(0, k - n + 1), min(n, k + 1))
            contribs = np.array([a[i] * a[k - i] for i in i_range])
            i_vals = np.array(list(i_range))
            if np.sum(contribs) > 1e-15:
                # Reduce each contributor proportionally
                frac = excess / (2 * np.sum(contribs))  # Factor 2 because each pair counted twice
                for i in i_vals:
                    if 0 <= k - i < n:
                        reduce_amt = frac * 2 * a[i]
                        adjust[i] -= min(reduce_amt, 0.3 * a[i])
    
        # Add back to low-contribution positions
        if np.sum(np.abs(adjust)) > 1e-10:
            # Find flat regions in C to add amplitude
            valley_indices = np.where(C < target_peak * 0.9)[0]
            if len(valley_indices) > 0:
                # Distribute the removed mass to positions that contribute least to peaks
                total_removed = -np.sum(np.minimum(adjust, 0))
                if total_removed > 0:
                    # Add to positions not strongly contributing to peaks
                    peak_contrib = np.zeros(n)
                    for k in peak_indices[:5]:
                        i_range = range(max(0, k - n + 1), min(n, k + 1))
                        for i in i_range:
                            j = k - i
                            if 0 <= j < n:
                                peak_contrib[i] += a[j]
                    
                    # Prefer adding where peak_contrib is low
                    weights = 1.0 / (peak_contrib + 0.1 * np.max(peak_contrib))
                    weights /= np.sum(weights)
                    add_mask = (a < np.mean(a))  # Add to below-average positions
                    if np.sum(add_mask) > 0:
                        w_masked = weights.copy()
                        w_masked[~add_mask] *= 0.1
                        w_masked /= np.sum(w_masked)
                        adjust += total_removed * w_masked
                        adjust[~add_mask] *= 0.1
                        adjust /= np.sum(adjust + total_removed) * 0 + 1  # keep as is
                        # Renormalize adjust to sum to 0
                        adjust_sum = np.sum(adjust)
                        adjust -= adjust_sum / n
        
        a = a + 0.1 * adjust
        a = np.clip(a, 0.0, None)
        a = a / np.sum(a) * S  # Renormalize
    
    return a


def local_search(a0, time_limit, rng):
    """Local search with diverse perturbation strategies."""
    n = len(a0)
    a = a0.copy()
    S = np.sum(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    temp = 0.002
    cooling = 0.9995
    stuck_count = 0
    
    while time.monotonic() - t0 < time_limit:
        candidate = a.copy()
        r = rng.random()
        
        if r < 0.25:
            # Targeted peak reduction
            C = fft_convolve(a, a)
            peak_idx = int(np.argmax(C))
            # Identify top contributors to the peak
            i_start = max(0, peak_idx - n + 1)
            i_end = min(n, peak_idx + 1)
            i_vals = np.arange(i_start, i_end)
            contribs = a[i_vals] * a[peak_idx - i_vals]
            top_k = max(3, len(i_vals) // 8)
            top_local = np.argsort(contribs)[-top_k:]
            top_global = i_vals[top_local]
            # Reduce top contributors
            candidate[top_global] *= np.exp(rng.normal(-0.03, 0.01, len(top_global)))
            # Increase non-contributors
            others = np.setdiff1d(np.arange(n), top_global)
            n_inc = min(10, len(others))
            if n_inc > 0:
                inc_idx = rng.choice(others, n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.01, n_inc))
        
        elif r < 0.45:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 12))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.025, width))
        
        elif r < 0.60:
            # Symmetry breaking
            asym = np.sin(np.pi * np.arange(n) / n) * 0.05
            candidate = a * np.exp(rng.normal(0, 0.01, n) + asym * rng.uniform(-1, 1))
        
        elif r < 0.75:
            # Multi-scale perturbation
            for _ in range(3):
                width = max(1, int(rng.integers(1, max(2, n // 15))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0, 0.02, width))
        
        elif r < 0.85:
            # Convolution-guided move
            C = fft_convolve(a, a)
            mean_C = np.mean(C)
            # Where C is high, reduce; where low, increase
            guidance = np.zeros(n)
            for i in range(n):
                # How much does a[i] contribute to high-C regions?
                contrib = np.sum(C[max(0, i):min(2*n-1, i+n)] * a[np.clip(np.arange(max(0, i), min(2*n-1, i+n)) - i, 0, n-1)])
                guidance[i] = contrib - mean_C * 2 * a[i] * n
            # Normalize guidance
            g_norm = guidance / (np.std(guidance) + 1e-10)
            candidate = a * np.exp(-0.01 * g_norm + rng.normal(0, 0.005, n))
        
        else:
            # Small random
            candidate = a * np.exp(rng.normal(0, 0.008, n))
        
        candidate = np.clip(candidate, 0.0, 1000.0)
        if np.sum(candidate) < 0.01:
            continue
        
        new_score = compute_score(candidate)
        
        if new_score < best_score:
            best_score = new_score
            best_a = candidate.copy()
            stuck_count = 0
        if new_score < best_score + temp:
            a = candidate
        else:
            stuck_count += 1
        
        temp *= cooling
        
        if stuck_count > 2000:
            a = best_a * np.exp(rng.normal(0, 0.03, n))
            a = np.clip(a, 0.01, 1000.0)
            stuck_count = 0
    
    return best_a, best_score


def make_structured_init(n, rng, style):
    """Create structured initial sequences."""
    t = np.arange(n, dtype=np.float64)
    if style == 0:  # Chirp
        alpha = 0.3 + 0.4 * rng.random()
        a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    elif style == 1:  # Asymmetric chirp
        alpha = 0.3 + 0.3 * rng.random()
        a = 1.0 + alpha * np.cos(np.pi * t * (t + n/3) / n)
    elif style == 2:  # Multi-tone
        a = np.ones(n)
        for k in range(1, 6):
            amp = 0.15 / k
            freq = rng.integers(1, n // 4)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * freq * t / n + phase)
    elif style == 3:  # Exponential taper (asymmetric)
        rate = rng.uniform(0.5, 2.0)
        center = rng.uniform(0.2, 0.8) * n
        a = 1.0 + 0.3 * np.exp(-rate * (t - center) ** 2 / n)
    elif style == 4:  # Step + noise
        a = np.ones(n)
        split = int(rng.uniform(0.3, 0.7) * n)
        a[:split] *= 1.0 + rng.uniform(-0.2, 0.3)
        a[split:] *= 1.0 + rng.uniform(-0.2, 0.3)
    else:  # Pure random
        a = 1.0 + 0.3 * rng.standard_normal(n)
    
    a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.01, 10.0)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: gradient descent + convolution flattening + local search.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    # Load provided sequence if available
    start_seq = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
            s = compute_score(start_seq)
            print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
            if s < best_score:
                best_score = s
                best_seq = start_seq.copy()
    except:
        pass
    
    # Phase 1: Gradient optimization with multiple starts
    print("\n=== Phase 1: Gradient optimization ===")
    
    # Determine which lengths to try
    lengths_to_try = []
    
    # Always try the provided length
    if start_seq is not None:
        lengths_to_try.append(len(start_seq))
    
    # Add other lengths
    for n_try in [192, 256, 320, 384, 448, 512, 640]:
        if n_try not in lengths_to_try:
            lengths_to_try.append(n_try)
    
    lengths_to_try = sorted(set(lengths_to_try))
    
    # Allocate time: more time for promising lengths
    n_configs = len(lengths_to_try)
    time_per_config = time_budget * 0.55 / n_configs  # 55% of budget for phase 1
    
    for n in lengths_to_try:
        if time.monotonic() > deadline - 60:
            break
        
        time_left = min(time_per_config * 1.5, deadline - time.monotonic() - 30)
        if time_left < 5:
            break
        
        print(f"\n  Optimizing n={n}, time_available={time_left:.1f}s")
        
        # Number of initializations based on time
        n_inits = max(2, int(time_left / 15))
        n_inits = min(n_inits, 8)
        
        time_per_init = time_left / (n_inits + 0.5)
        
        for init_i in range(n_inits):
            if time.monotonic() > deadline - 30:
                break
            
            # Choose initialization
            if init_i == 0 and start_seq is not None and len(start_seq) == n:
                a0 = start_seq.copy()
            elif init_i == 0 and start_seq is not None:
                # Resample provided sequence
                x_old = np.linspace(0, 1, len(start_seq))
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, start_seq)
            else:
                style = (init_i - 1) % 6 if init_i > 0 else rng.integers(0, 6)
                a0 = make_structured_init(n, rng, style)
            
            a0 = np.clip(a0, 0.01, 100.0)
            S = np.sum(a0)
            a0 = a0 / S * np.sqrt(2.0 * n)  # Normalize
            
            a_best, score = optimize_sequence(a0, time_per_init, rng)
            actual_score = compute_score(a_best)
            
            if init_i % 2 == 0 or actual_score < best_score:
                print(f"    init={init_i}, score={actual_score:.6f}, best_so_far={best_score:.6f}")
            
            if actual_score < best_score:
                best_score = actual_score
                best_seq = a_best.copy()
                print(f"    *** New best: n={n}, score={best_score:.6f} ***")
    
    # Phase 2: Convolution flattening on best candidate
    print(f"\n=== Phase 2: Convolution flattening ===")
    print(f"  Current best: score={best_score:.6f}, n={len(best_seq)}")
    
    if best_seq is not None:
        a = best_seq.copy()
        S = np.sum(a)
        n = len(a)
        C = fft_convolve(a, a)
        target_peak = np.mean(C) * 1.3  # Target 1.3x mean
        
        # Try flattening
        time_left = min(60, deadline - time.monotonic() - 30)
        if time_left > 5:
            a_flat = convolution_flattening(a, target_peak, 200, rng)
            score_flat = compute_score(a_flat)
            print(f"  After flattening: score={score_flat:.6f}")
            if score_flat < best_score:
                best_score = score_flat
                best_seq = a_flat.copy()
                print(f"  *** New best: {best_score:.6f} ***")
            
            # Try with different targets
            for tf in [1.2, 1.4, 1.5]:
                a_flat = convolution_flattening(best_seq, tf * np.mean(fft_convolve(best_seq, best_seq)), 100, rng)
                s = compute_score(a_flat)
                if s < best_score:
                    best_score = s
                    best_seq = a_flat.copy()
                    print(f"  After flattening (target={tf}): {best_score:.6f} ***")
    
    # Phase 3: Additional gradient optimization on best
    print(f"\n=== Phase 3: Refinement ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        time_left = min(90, deadline - time.monotonic() - 20)
        if time_left > 10:
            # Fine-tune with tighter parameters
            a_ref, s_ref = optimize_sequence(best_seq, time_left, rng)
            s_ref = compute_score(a_ref)
            print(f"  After refinement: score={s_ref:.6f}")
            if s_ref < best_score:
                best_score = s_ref
                best_seq = a_ref.copy()
                print(f"  *** New best: {best_score:.6f} ***")
    
    # Phase 4: Local search polish
    print(f"\n=== Phase 4: Local search polish ===")
    if best_seq is not None and time.monotonic() < deadline - 15:
        time_left = min(120, deadline - time.monotonic() - 10)
        if time_left > 10:
            print(f"  Starting local search, time={time_left:.1f}s")
            a_ls, s_ls = local_search(best_seq, time_left, rng)
            if s_ls < best_score:
                best_score = s_ls
                best_seq = a_ls.copy()
                print(f"  After local search: {best_score:.6f} ***")
            else:
                print(f"  Local search best: {s_ls:.6f} (no improvement)")
    
    # Final: try a few more quick optimizations if time permits
    if time.monotonic() < deadline - 10 and best_seq is not None:
        # Quick coordinate-based refinement
        a = best_seq.copy()
        n = len(a)
        S = np.sum(a)
        for coord in range(n):
            if time.monotonic() > deadline - 10:
                break
            # Try small changes to this coordinate
            best_local = a.copy()
            best_local_score = compute_score(a)
            for scale in [0.9, 0.95, 1.05, 1.1]:
                trial = a.copy()
                trial[coord] *= scale
                trial = np.clip(trial, 0, 1000)
                if np.sum(trial) < 0.01:
                    continue
                s = compute_score(trial)
                if s < best_local_score:
                    best_local_score = s
                    best_local = trial.copy()
            a = best_local
        s_final_coord = compute_score(a)
        if s_final_coord < best_score:
            best_score = s_final_coord
            best_seq = a.copy()
            print(f"  After coordinate refinement: {best_score:.6f} ***")
    
    # Final verification
    print(f"\n=== FINAL RESULTS ===")
    print(f"Best score: {best_score:.6f}")
    print(f"Sequence length: {len(best_seq)}")
    
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"Verified score: {final_score:.6f}")
        return final_list
    
    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
