def compute_objective(a):
    """Compute 2n * max(c) / S^2 where c = a*a (convolution)."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = np.max(c)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    return 2.0 * n * max_c / (S * S)

def soft_max_and_weights(c, beta):
    """Compute soft-max of c and the weights (gradient w.r.t. c)."""
    max_c = np.max(c)
    shifted = beta * (c - max_c)
    # Clip for numerical stability
    shifted = np.maximum(shifted, -50)  # exp(-50) ~ 2e-22, safe
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    soft_max_val = max_c + np.log(sum_exp) / beta
    return soft_max_val, weights

    Gradient of 2n * soft_max(c, beta) / S^2 w.r.t. a.
    where c = a * a (convolution), S = sum(a).
