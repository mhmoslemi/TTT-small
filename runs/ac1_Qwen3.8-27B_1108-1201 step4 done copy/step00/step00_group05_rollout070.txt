import time


def _next_pow2(m):
    s = 1
    while s < m:
        s <<= 1
    return s

def _alt_inits(a, rng):
    """Return a list of alternative nonnegative init vectors of same length."""
    n = a.size
    k = np.arange(n)
    outs = []
    # sum-of-cosines "spectral shaping" starts (nonnegative by 1 + bounded)
    for (p, r) in [(1, 0.4), (2, 0.3), (3, 0.3), (5, 0.25), (1, 0.7)]:
        v = 1.0 + r * np.cos(2 * np.pi * p * k / max(1, n - 1))
        v = np.maximum(v, 0.05)
        v = v * (n / v.sum())
        outs.append(v)
    # raised cosine bump (edges emphasized to flatten convolution)
    v = 1.0 - 0.6 * np.cos(2 * np.pi * k / max(1, n - 1))
    v = np.maximum(0.05 * v, 0.05)
    outs.append(v * (n / v.sum()))
    # smoothed random
    v = np.maximum(rng.standard_normal(n), 0.0) + 0.3
    outs.append(v * (n / v.sum()))
    return outs


def _flatten_grad(a, N):
    """Gradient of sum(convolve(a,a)**2) == (4/N) sum|A|^4  w.r.t. a.  O(N log N)."""
    fa = np.fft.rfft(a, N)
    Hh = (fa * np.conj(fa)) * np.conj(fa)
    H = np.empty(N, dtype=complex)
    H[:len(Hh)] = Hh
    H[len(Hh):] = np.conj(Hh[N // 2 - 1:0:-1])
    G = np.fft.ifft(H).real * N
    return 4.0 * G[np.mod(-np.arange(a.size), N)]


def _peak_grad(a, w, M):
    """grad[j] = sum_i w[j+i]*a[i]  (softmax-peak direction)."""
    Fw = np.fft.rfft(w, M)
    Fa = np.fft.rfft(a, M)
    c = np.fft.irfft(np.conj(Fw) * Fa, M)
    return c[:a.size]


def _self_conv(a, N):
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[: 2 * a.size - 1]
