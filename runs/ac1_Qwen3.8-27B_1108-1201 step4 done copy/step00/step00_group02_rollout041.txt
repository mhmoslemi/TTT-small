import numpy as np
import time

def _convolve_fft(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2*n-1]
    np.maximum(c, 0, out=c)
    return c

def _gradient_at_hot(c, a, top_frac=0.05):
    """
    Compute (1/|K|) * sum_{k in K} a[k-i] for valid indices.
    This is the "hot cross-correlation" which appears in the gradient.
    """
    n = len(a)
    threshold = np.percentile(c, 100 * (1 - top_frac))
    hot = (c >= threshold).astype(np.float64)
    n_hot = np.sum(hot)
    if n_hot == 0:
        hot = np.zeros(len(c))
        hot[np.argmax(c)] = 1.0
        n_hot = 1.0
    
    # Compute cross-correlation: g[i] = sum_k hot[k] * a[k-i] / n_hot
    # = ifft(fft(hot) * conj(fft(a))) / n_hot (circular, so we need zero-padding)
    size = 1
    while size < 3 * n - 2:
        size *= 2
    F_hot = np.fft.rfft(hot, size)
    F_a = np.fft.rfft(a, size)
    g = np.fft.irfft(F_hot * np.conj(F_a), size)[:n]
    g /= n_hot
    return g, n_hot

The sum is sum_i binom(2i,i)/4^i = 1/sqrt(1-1) which diverges! So this infinite series doesn't converge. For finite n, we'd truncate, but the last elements would be very small.

Actually, the sum of binom(2i,i)/4^i from i=0 to n-1 is approximately sqrt(2n/pi) for large n (by the Central Limit Theorem, binom(2i,i)/4^i ≈ 1/sqrt(pi*i), and sum 1/sqrt(i) from 1 to n ≈ 2*sqrt(n)).

So sum(a) ≈ sqrt(C) * 2*sqrt(n) (roughly). And C = a_0^2 = C, so... this is circular. Let me just set C = 1 for now:

The point is: for finite n, the "ideal" sequence (making all c_k equal) doesn't exist because the infinite series diverges. But a truncated version might give a good approximation.
