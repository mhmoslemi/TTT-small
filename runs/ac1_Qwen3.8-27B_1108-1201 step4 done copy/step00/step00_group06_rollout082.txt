def compute_smoothed_grad(a, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*conv(a,a))) w.r.t. a."""
    n = a.size
    conv_a = np.convolve(a, a)
    mx = conv_a.max()
    ev = np.exp(tau * (conv_a - mx))
    Z = ev.sum()
    w = ev / Z
    # grad[j] = 2 * sum_l a[l] * w[l+j], computed via FFT
    wf = w[::-1]
    sz = 1
    total_len = n + wf.size - 1
    while sz < total_len:
        sz *= 2
    conv_full = np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(wf, sz), sz)[:total_len]
    grad = 2.0 * conv_full[n - 1:2 * n - 1][::-1]
    return grad, conv_a


def try_lp_refinement(a, max_time=60.0):
    """Try one LP-based direction step. Returns improved a or None."""
    n = a.size
    conv_a = np.convolve(a, a)
    M = float(conv_a.max())
    S = float(a.sum())
    if M < 1e-15 or S < 0.01:
        return None

    rows = []
    cols = []
    vals = []

        hi = min(n, k + 1)
        for m in range(lo, hi):
            rows.append(k)
            cols.append(m)
            vals.append(a[k - m])
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    b = res.x

import numpy as np

def autocorrelation_fft(a):
    """Compute linear autocorrelation conv(a,a) via FFT."""
    n = a.size
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    ac = np.fft.irfft(fa * fa.conj(), size)
    return ac[:2 * n - 1]


def compute_grad(a, tau, size, fa):
    """Compute gradient of smoothed max of autocorrelation using precomputed FFT of a."""
    n = a.size
    conv_a = np.fft.irfft(fa * fa.conj(), size)
    conv_a = conv_a[:2 * n - 1]

    mx = conv_a.max()
    ev = np.exp(tau * (conv_a - mx))
    Z = ev.sum()
    w = ev / Z

    # grad[j] = 2 * sum_{l=0}^{n-1} a[l] * w[l+j]
    # Computed as 2 * convolve(a, w[::-1])[n-1:2n-1][::-1]
    wf = w[::-1]
    fw = np.fft.rfft(wf, size)
    conv_full = np.fft.irfft(fa * fw, size)
    total_len = n + wf.size - 1
    grad = 2.0 * conv_full[n - 1:2 * n - 1][::-1]

    return grad, conv_a


def try_lp_refinement(a, max_time=40.0):
    """Solve LP for direction b: max sum(b) s.t. conv(b,a)<=M, b>=0."""
    n = a.size
    conv_a = np.convolve(a, a)
    M = float(conv_a.max())
    S = float(a.sum())
    if M < 1e-15 or S < 0.01:
        return None

    # Build sparse matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for m in range(lo, hi):
            rows.append(k)
            cols.append(m)
            vals.append(a[k - m])
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)

                  method='highs', options={'time_limit': max_time})
    if res.status != 0 or res.x is None:
        return None
    b = res.x
    S_b = float(b.sum())
    if S_b < 1e-10:
        return None
    return b * (S / S_b)
