"""
LP-based iterative optimization for minimizing the autocorrelation peak of non-negative sequences.
Uses the approach: at each step, solve an LP to find g maximizing sum(g) subject to (f*g)[k] <= max(f*f),
then update f = (1-t)*f + t*g. Also employs block perturbations and multiple starting points.
"""
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time


def build_conv_matrix(n, f):
    """Build sparse matrix A (2n-1 x n) where A[k,j] = f[k-j] if 0<=k-j<n else 0."""
    i_idx = np.tile(np.arange(n), n)
    j_idx = np.repeat(np.arange(n), n)
    row_idx = j_idx + i_idx
    data = f[i_idx]
    A = sp.coo_matrix((data, (row_idx, j_idx)), shape=(2 * n - 1, n)).tocsr()
    return A


def solve_lp_for_g(f, M, time_limit_s=15):
    """Solve max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0. Returns g or None."""
    n = len(f)
    A = build_conv_matrix(n, f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit_s, 'presolve': True})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_one_step(f, t, time_limit_s=15):
    """Perform one LP-based update step on f. Returns new f."""
    n = len(f)
    conv = np.convolve(f, f)
    M = conv.max()
    if M < 1e-15:
        return f
    g = solve_lp_for_g(f, M, time_limit_s)
    if g is None or g.sum() < 1e-10:
        return f
    g = g / g.sum()
    f_new = (1.0 - t) * f + t * g
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-10:
        return f
    return f_new / s
