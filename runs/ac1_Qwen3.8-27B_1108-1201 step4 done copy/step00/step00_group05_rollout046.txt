"""
Iterative peak-shaving and block-perturbation search for minimizing
2n * max(a*a) / (sum(a))^2 over non-negative sequences.
"""

import numpy as np
import time


def compute_objective(a):
    """Compute the objective function value."""
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    b = np.convolve(a, a)
    max_b = np.max(b)
    n = len(a)
    return 2.0 * n * max_b / (S * S)


def make_initial_shape(n, shape_type='hann'):
    """Create an initial sequence shape."""
    if shape_type == 'uniform':
        return np.ones(n)
    elif shape_type == 'hann':
        return np.hanning(n) + 0.1  # add small floor to keep positive
    elif shape_type == 'gaussian':
        x = np.linspace(-3, 3, n)
        return np.exp(-x**2 / 2) + 0.01
    elif shape_type == 'blackman':
        return np.blackman(n) + 0.05
    elif shape_type == 'welch':
        x = np.arange(n)
        return 1.0 - (x - (n-1)/2) / ((n-1)/2) ** 2
    else:
        return np.ones(n)


def normalize_sum(a, target_sum=None):
    """Normalize sequence to have a specific sum."""
    S = np.sum(a)
    if S <= 0:
        return a
    if target_sum is None:
        target_sum = np.sqrt(2 * len(a))
    return a * target_sum / S


def peak_shaving_step(a, peak_positions, num_redistribute=10):

def block_perturbation(a, rng, scale=0.1):
    """Apply a block perturbation to the sequence."""
    n = len(a)
    a_new = a.copy()
    width = max(1, int(n * rng.uniform(0.005, 0.05)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    end = min(n, start + width)
    
    # Multiplicative perturbation
    factor = np.exp(rng.normal(0, scale, size=end - start))
    a_new[start:end] *= factor
    
    # Ensure non-negative
    a_new = np.clip(a_new, 0, 1000.0)
    
    # Optional: symmetrize
    if rng.random() < 0.3:
        a_new = 0.5 * (a_new + a_new[::-1])
    
    return a_new

def compute_b(a):
    """Compute the full autocorrelation a*a."""
    return np.convolve(a, a)


def get_objective(a):
    """Compute 2n*max(a*a)/S^2, the objective to minimize."""
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    b = np.convolve(a, a)
    max_b = np.max(b)
    n = len(a)
    return 2.0 * n * max_b / (S * S)


def normalize(a, target=None):
    """Normalize sequence to have a target sum."""
    S = np.sum(a)
    if S <= 0:
        return a
    if target is None:
        target = float(len(a))
    return a * (target / S)


def init_uniform(n):
    return np.ones(n, dtype=np.float64)


def init_gaussian(n, sigma_frac=0.3):
    """Gaussian centered in the sequence."""
    c = (n - 1) / 2.0
    x = np.arange(n) - c
    sigma = sigma_frac * n
    return np.exp(-x**2 / (2.0 * sigma**2))


def init_cosine(n, power=1.0):
    """Cosine window raised to a power."""
    x = np.arange(n)
    w = 0.5 * (1.0 + np.cos(np.pi * x / (n - 1)))
    return w ** power

    """Parabolic dip: 1 - alpha*(t^2) where t in [-1,1]."""
    t = 2.0 * np.arange(n) / (n - 1) - 1.0
    return np.maximum(1.0 - alpha * t**2, 0.0)
