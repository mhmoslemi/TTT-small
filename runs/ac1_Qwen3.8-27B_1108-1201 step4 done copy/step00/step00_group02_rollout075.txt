    while time.monotonic() < deadline:
        a = best
        g, M = compute_subgradient(a, n)
        g_norm = np.linalg.norm(g)

        if g_norm < 1e-14:
            break

        # Backtracking line search
        improved = False
        local_eta = eta
        for _ in range(40):
            a_tilde = a - local_eta * g
            a_new = simplex_project(a_tilde)
            score_new = objective_on_simplex(a_new, n)

            if np.isfinite(score_new) and score_new < best_score - 1e-15:
                best = a_new
                best_score = score_new
                no_improve = 0
                improved = True
                # Increase eta slightly on success
                eta = min(eta * 1.02, eta_max)
                break
            else:
                local_eta *= 0.5
                if local_eta < eta_min:
                    break

        if not improved:
            no_improve += 1
            eta = max(eta * 0.95, eta_min)

        total_iters += 1

        # Print progress
        if time.monotonic() - last_print_time > 20:
            print(f"  iter={total_iters}, J={best_score:.6f}, eta={eta:.2e}, no_improve={no_improve}")
            last_print_time = time.monotonic()

        # Restart if stuck
        if no_improve > max_no_improve:
            restart_count += 1
            print(f"  Restart {restart_count} at iter {total_iters}, J={best_score:.6f}")
            # Try several new initial points plus perturbation
            candidates = []
            for i in range(8):
                candidates.append(make_initial_point(i, n, rng, parent))
            # Perturbed copy of current best
            for sigma in [0.1, 0.3, 0.5]:
                perturbed = best * np.exp(rng.normal(0, sigma, size=n))
                perturbed = np.maximum(perturbed, 0)
                ps = np.sum(perturbed)
                if ps > 1e-10:
                    candidates.append(perturbed / ps)
            # Mirror-symmetric version
            mirrored = 0.5 * (best + best[::-1])
            mirrored = np.maximum(mirrored, 0)
            ms = np.sum(mirrored)
            if ms > 1e-10:
                candidates.append(mirrored / ms)

            for cand in candidates:
                sc = objective_on_simplex(cand, n)
                if np.isfinite(sc) and sc < best_score:
                    best = cand
                    best_score = sc

            no_improve = 0
            eta = 0.05
            print(f"  After restart: J={best_score:.6f}")

    # Final: scale result to have sum = n (values O(1)) for safe evaluation
    result = best * n
    result = np.clip(result, 0.0, 1000.0)

    final_score = float(evaluate_sequence([float(x) for x in result]))
    print(f"Final J = {final_score:.6f} (from simplex score {best_score:.6f})")

    if not np.isfinite(final_score):
        result = np.ones(n)
        final_score = float(evaluate_sequence([float(x) for x in result]))
        print(f"Fallback J = {final_score:.6f}")

    return [float(x) for x in result]
