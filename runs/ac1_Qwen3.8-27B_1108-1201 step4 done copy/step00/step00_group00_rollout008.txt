```python
def subgradient_step(a, lr):
    n = len(a)
    c = np.convolve(a, a)
    k = np.argmax(c)
    g = np.zeros(n)
    for i in range(n):
        if 0 <= k - i < n:
            g[i] += a[k - i]
        if 0 <= i - k < n:
            g[i] += a[i - k]
    a_new = a - lr * g
    a_new = np.maximum(0, a_new)
    s = a_new.sum()
    if s > 0:
        a_new = a_new / s
    else:
        a_new = np.ones(n) / n
    return a_new
```
