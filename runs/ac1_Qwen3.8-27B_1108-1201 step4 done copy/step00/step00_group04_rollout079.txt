Coordinate descent with fast O(n) convolution tracking to minimize
2n*max(autocorrelation)/sum^2 for non-negative sequences.
Tries multiple lengths and initializations.
"""

import numpy as np
import time


def max_conv_after_change(c, a, i, x):
    """Compute max of convolution after hypothetical change of a[i] to x.
    Does NOT modify a or c. Returns the new maximum value of the full convolution."""
    n = len(a)
    old = a[i]
    diff = x - old
    if abs(diff) < 1e-15:
        return c.max()
    k_lo = i
    k_hi = min(i + n - 1, 2 * n - 2)
    length = k_hi - k_lo + 1
    new_affected = c[k_lo:k_hi + 1].copy() + 2.0 * diff * a[0:length]
    if k_lo <= 2 * i <= k_hi:
        idx = 2 * i - k_lo
        new_affected[idx] = c[2 * i] + x * x - old * old
    max_aff = new_affected.max()
    max_unaff = 0.0
    if k_lo > 0:
        m = c[:k_lo].max()
        if m > max_unaff:
            max_unaff = m
    if k_hi + 1 < 2 * n - 1:
        m = c[k_hi + 1:].max()
        if m > max_unaff:
            max_unaff = m
    return max(max_aff, max_unaff)


def apply_change(c, a, i, x):
    """Apply change a[i] = x, updating c in-place. Returns change in sum."""
    n = len(a)
    old = a[i]
    diff = x - old
    if abs(diff) < 1e-15:
        return 0.0
    k_lo = i
    k_hi = min(i + n - 1, 2 * n - 2)
    length = k_hi - k_lo + 1
    c[k_lo:k_hi + 1] += 2.0 * diff * a[0:length]
    if k_lo <= 2 * i <= k_hi:
        idx = 2 * i - k_lo
        c[2 * i] -= 2.0 * diff * a[idx]
        c[2 * i] += x * x - old * old
    a[i] = x
    return diff

    x0 = a[i]

        return max_conv_after_change(c, a, i, x)

    f_mid = f(x_hi * 0.5)
    f_hi = f(x_hi)

        if f_hi < f_mid:
            x_lo = x_hi
            f_lo = f_hi
            x_hi = 2.0 * x_hi
            f_mid = f(x_hi * 0.5)
            f_hi = f(x_hi)
        else:
            break

    if f_hi <= f_mid:
        return x_hi

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c_pt = x_hi - gr * (x_hi - x_lo)
    d_pt = x_lo + gr * (x_hi - x_lo)
    fc = f(c_pt)
    fd = f(d_pt)

    tol_abs = max(1e-8, x0 * 1e-7)
    for _ in range(50):
        if x_hi - x_lo < tol_abs:
            break
        if fc <= fd:
            x_hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = x_hi - gr * (x_hi - x_lo)
            fc = f(c_pt)
        else:
            x_lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = x_lo + gr * (x_hi - x_lo)
            fd = f(d_pt)

    x_opt = 0.5 * (x_lo + x_hi)
    return max(0.0, x_opt)


def run_coordinate_descent(a0, max_sweeps=300, deadline=None, rng=None):
    """Run coordinate descent on a copy of a0. Returns (best_a, best_inner_obj)."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    c = np.convolve(a, a)
    c_max = float(c.max())
    S = float(a.sum())
    obj = 2.0 * n * c_max / (S * S) if S > 0.01 else 1e10

    best_a = a.copy()
    best_obj = obj

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break

        order = rng.permutation(n)
        prev_c_max = c_max

        for i in order:
            if deadline is not None and time.monotonic() > deadline:
                break
            x_opt = optimize_coordinate(c, a, i)
            old_val = a[i]
            if abs(x_opt - old_val) > 1e-12:
                diff = apply_change(c, a, i, x_opt)
                S += diff
                c_max = float(c.max())

        if c_max < prev_c_max - 1e-10:
            obj = 2.0 * n * c_max / (S * S) if S > 0.01 else 1e10
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

        if sweep > 0 and abs(c_max - prev_c_max) < 1e-10 * max(c_max, 1.0):
            if sweep > 50 and c_max == prev_c_max:
                if best_obj < 1.6:
                    break
                if sweep > 150:
                    break

    return best_a, best_obj
