def _lp_dual(f, M, n, remaining):
    if not np.isfinite(M) or M <= 0.0: return None
    rows = 2*n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = f
    b = np.full(rows, float(M), dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)
    fmin = float(np.min(f))
    if fmin <= 0.0 or not np.isfinite(fmin):
        fmin = 1e-12
    ub = 10.0 * float(M) / fmin
    ub = min(max(ub, 1.0), 1e6)
    bounds = [(0.0, float(ub))] * n
    tl = float(min(30.0, max(8.0, remaining / 4.0)))
    if n > 900:
        tl = float(min(60.0, max(15.0, remaining / 3.0)))
    opts = {'presolve': 'on', 'time_limit': tl}
    try:
        res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options=opts)
    except Exception:
        try:
            res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve':'on'})
        except Exception:
            return None
    x = getattr(res, 'x', None)
    if x is None:
        return None
    x = np.asarray(x, dtype=np.float64).ravel()
    if x.size != n or not np.all(np.isfinite(x)):
        return None
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-9:
        return None
    if not getattr(res, 'success', False):
        # check feasibility optionally if n <= 1200 and maybe violation not huge
        if n <= 1200:
            try:
                viol = float(np.max(A.dot(x) - float(M)))
                if viol > 1e-4 * float(M) + 1e-8:
                    # If slight violation, clip? reject.
                    return None
            except Exception:
                return None
    return x

import numpy as np
import time
from scipy import optimize


"""
Heuristic search for low-peak nonnegative autocorrelation step functions.

The objective is scale invariant.  Internally all trial sequences are scaled to
sum sqrt(2N), so the objective equals max(convolve(a, a)).  The optimizer uses:
  - a small pool of analytically/random-shaped starts,
  - an LP "dual direction" iteration inspired by the state-of-the-art method,
  - line-searched convex blends,
  - local subgradient/random refinement as a fallback.
"""


def _ceil_pow2(m):
    p = 1
    m = int(m)
    while p < m:
        p <<= 1
    return p


def _normalize_to_S(arr, S):
    a = np.asarray(arr, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.maximum(a, 0.0)
    if not np.all(np.isfinite(a)):
        return None
    s = float(a.sum())
    if s <= 1e-12:
        return None
    return a * (float(S) / s)


def _conv_raw(a, conv_len, fft_len):
    F = np.fft.rfft(a, fft_len)
    c = np.fft.irfft(F * F, fft_len)
    return c[:conv_len]


def _max_conv_raw(a, conv_len, fft_len):
    c = _conv_raw(a, conv_len, fft_len)
    return float(np.max(c))


def _interp_to_n_nS(seq, n, S):
    if seq is None:
        return None
    arr = np.asarray(seq, dtype=np.float64).ravel()
    if arr.size == 0:
        return None
    arr = np.maximum(arr, 0.0)
    if not np.all(np.isfinite(arr)):
        return None
    if arr.size == 1:
        return np.full(n, float(S) / float(n), dtype=np.float64)
    if arr.size != n:
        old = np.linspace(0.0, 1.0, arr.size)
        new = np.linspace(0.0, 1.0, n)
        arr = np.interp(new, old, arr)
    return _normalize_to_S(arr, S)


def _update_pool(pool, arr, n, conv_len, fft_len, S, max_pool):
    a = _normalize_to_S(arr, S)
    if a is None:
        return pool
    sc = _max_conv_raw(a, conv_len, fft_len)
    if not np.isfinite(sc):
        return pool

    if len(pool) < max_pool:
        pool.append([a, sc])
    else:
        worst_i = 0
        worst_s = pool[0][1]
        for i in range(1, len(pool)):
            if pool[i][1] > worst_s:
                worst_s = pool[i][1]
                worst_i = i
        if sc < worst_s - 1e-12:
            pool[worst_i] = [a, sc]
    return pool


def _shape_pool(n, rng, external, max_s=30.0, max_pool=6):
    if rng is None:
        rng = np.random.default_rng(12345)

    pool = []
    S = float(np.sqrt(2.0 * int(n)))
    conv_len = 2 * int(n) - 1
    fft_len = _ceil_pow2(conv_len)
    t = np.linspace(0.0, 1.0, int(n))

    # Deterministic useful shapes.
    _update_pool(pool, np.full(n, S / float(n)), n, conv_len, fft_len, S, max_pool)
    _update_pool(pool, np.arange(1, n + 1, dtype=np.float64), n, conv_len, fft_len, S, max_pool)
    _update_pool(pool, np.arange(n, 0, -1, dtype=np.float64), n, conv_len, fft_len, S, max_pool)

    for alpha in (0.25, 0.5, 1.0, 2.0, 3.0, 4.0, -0.5, -1.0, -2.0, -3.0):
        a = np.exp(float(alpha) * t)
        _update_pool(pool, a, n, conv_len, fft_len, S, max_pool)

    x = t - 0.5
    for A in (-2.0, -1.5, -1.0, -0.5, 0.5, 1.0, 1.5, 2.0):
        a = 1.0 + float(A) * x
        _update_pool(pool, a, n, conv_len, fft_len, S, max_pool)

    for A in (-2.0, -1.0, 0.0, 1.0, 2.0):
        for B in (-2.0, -.1, 0.0, .1, 1.0, 2.0):
            a = 1.0 + float(A) * x + float(B) * x * x
            _update_pool(pool, a, n, conv_len, fft_len, S, max_pool)

    if external is not None:
        e = _interp_to_n_nS(external, n, S)
        if e is not None:
            _update_pool(pool, e, n, conv_len, fft_len, S, max_pool)

    # Random parameter search.  The shapes are positive by construction and
    # can produce useful asymmetric start points.
    start = time.monotonic()
    limit = max(0.0, float(max_s))
    deadline = start + limit
    it = 0
    while time.monotonic() < deadline:
        it += 1

        u = float(rng.uniform(0.0, 2.5))
        v = float(rng.uniform(0.0, 2.5))
        lin = float(rng.uniform(-3.0, 3.0))
        quad = float(rng.uniform(-3.0, 3.0))
        cub = float(rng.uniform(-4.0, 4.0))

        base = (np.maximum(t, 1e-8) ** u) * (np.maximum(1.0 - t, 1e-8) ** v)
        a = base * np.exp(lin * x + quad * x * x + cub * x * x * x)
        _update_pool(pool, a, n, conv_len, fft_len, S, max_pool)

        if it % 3 == 0:
            a2 = 1.0 + lin * x + quad * x * x + cub * x * x * x
            _update_pool(pool, a2, n, conv_len, fft_len, S, max_pool)

    if not pool:
        a = np.full(n, S / float(n), dtype=np.float64)
        pool.append([a, _max_conv_raw(a, conv_len, fft_len)])
    return pool


def _lp_dual(f, M, n, remaining):
    """
    Solve the LP:
        maximize sum(g)
        subject to g >= 0, convolve(f, g) <= M.
    """
    n = int(n)
    if not np.isfinite(M) or M <= 0.0:
        return None

    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f

    b = np.full(rows, float(M), dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)

    fmin = float(np.min(f))
    if not np.isfinite(fmin) or fmin <= 0.0:
        fmin = 1e-12

    ub = 10.0 * float(M) / fmin
    ub = min(max(ub, 1.0), 1.0e6)
    bounds = [(0.0, float(ub))] * n

    if n > 900:
        tl = float(min(60.0, max(15.0, float(remaining) / 3.0)))
    else:
        tl = float(min(30.0, max(8.0, float(remaining) / 4.0)))

    res = None
    try:
        res = optimize.linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method='highs',
            options={'presolve': 'on', 'time_limit': tl}
        )
    except Exception:
        try:
            res = optimize.linprog(
                c,
                A_ub=A,
                b_ub=b,
                bounds=bounds,
                method='highs',
                options={'presolve': 'on'}
            )
        except Exception:
            return None

    x = getattr(res, 'x', None)
    if x is None:
        return None
    x = np.asarray(x, dtype=np.float64).ravel()
    if x.size != n or not np.all(np.isfinite(x)):
        return None

    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-9:
        return None

    if not getattr(res, 'success', False):
        # If the LP only hit its local time limit, accept only if the returned
        # iterate is approximately feasible.
        try:
            resid = A.dot(x)
            viol = float(np.max(resid - float(M)))
            scale = abs(float(M)) + float(np.max(resid)) + 1.0
            if viol > 1e-4 * scale + 1e-8:
                return None
        except Exception:
            return None

    return x


def _local_gradient_refine(f, n, conv_len, fft_len, S, rng, deadline):
    """
    Cheap nonconvex refinement using a top-lag subgradient direction.
    """
    if rng is None:
        rng = np.random.default_rng(98765)

    f = _normalize_to_S(f, S)
    if f is None:
        return None, np.inf

    best = f.copy()
    best_score = _max_conv_raw(best, conv_len, fft_len)
    no = 0

    for _ in range(200000):
        if time.monotonic() >= deadline:
            break

        b = _conv_raw(f, conv_len, fft_len)
        M = float(np.max(b))
        if not np.isfinite(M):
            break

        K = min(16, int(conv_len))
        idx = np.argpartition(b, -K)[-K:]
        w = b[idx] * b[idx]

        grad = np.zeros(n, dtype=np.float64)
        for i in range(len(idx)):
            k = int(idx[i])
            wk = float(w[i])
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            if lo <= hi:
                if lo == 0:
                    seg = f[hi::-1]
                else:
                    seg = f[hi:lo - 1:-1]
                grad[lo:hi + 1] += 2.0 * wk * seg

        gs = float(np.max(grad))
        if gs <= 1e-300:
            break

        meanp = float(np.mean(f))
        base = meanp / gs
        etas = (0.01, 0.03, 0.1, 0.3, 1.0)

        best_cand = None
        best_cand_score = best_score
        for eta in etas:
            c = f - base * eta * grad
            c = _normalize_to_S(np.maximum(c, 0.0), S)
            if c is None:
                continue
            sc = _max_conv_raw(c, conv_len, fft_len)
            if sc < best_cand_score:
                best_cand_score = sc
                best_cand = c.copy()

        if best_cand is not None and best_cand_score < best_score - 1e-12:
            f = best_cand
            best = best_cand.copy()
            best_score = best_cand_score
            no = 0
        else:
            no += 1
            if no > 8:
                c = f.copy()
                wdt = max(1, int(n // 64))
                if wdt >= n:
                    wdt = 1
                start = int(rng.integers(0, n - wdt))
                c[start:start + wdt] *= float(np.exp(rng.normal(0.0, 0.15)))
                c = _normalize_to_S(np.maximum(c, 0.0), S)
                if c is not None:
                    sc = _max_conv_raw(c, conv_len, fft_len)
                    if sc < best_score:
                        f = c
                        best = c.copy()
                        best_score = sc
                        no = 0
                    elif sc < best_score + 0.02:
                        f = c
                        no = 0

    return best, best_score


def _run_lp_phase(n, pool, external, external_score, deadline, rng):
    """
    Run LP-based local search for one sequence length n.
    Returns the best sequence and its internal scaled objective.
    """
    if rng is None:
        rng = np.random.default_rng(246810)

    n = int(n)
    S = float(np.sqrt(2.0 * n))
    conv_len = 2 * n - 1
    fft_len = _ceil_pow2(conv_len)

    if pool is None or len(pool) == 0:
        pool = _shape_pool(n, rng, external, 30.0, 6)
    else:
        pool = [list(entry) for entry in pool]
        if external is not None:
            e = _interp_to_n_nS(external, n, S)
            if e is not None:
                pool = _update_pool(pool, e, n, conv_len, fft_len, S, 8)

    if not pool:
        a = np.full(n, S / float(n), dtype=np.float64)
        pool = [[a, _max_conv_raw(a, conv_len, fft_len)]]

    best_i = 0
    best_score = float(pool[0][1])
    for i in range(1, len(pool)):
        if float(pool[i][1]) < best_score:
            best_score = float(pool[i][1])
            best_i = i

    best_seq = np.asarray(pool[best_i][0], dtype=np.float64).copy()
    best_seq = _normalize_to_S(best_seq, S)
    if best_seq is None:
        best_seq = np.full(n, S / float(n), dtype=np.float64)
        best_score = _max_conv_raw(best_seq, conv_len, fft_len)

    # If the externally supplied sequence already has a better raw score, use
    # it as the starting point.
    if external is not None and np.isfinite(external_score):
        e = _interp_to_n_nS(external, n, S)
        if e is not None:
            e_score = _max_conv_raw(e, conv_len, fft_len)
            if e_score < best_score - 1e-12:
                best_seq = e
                best_score = e_score

    last_improve = time.monotonic()

    while time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        if rem < 3.0:
            break

        f = best_seq
        M = float(best_score)
        if not np.isfinite(M):
            f = np.full(n, S / float(n), dtype=np.float64)
            M = _max_conv_raw(f, conv_len, fft_len)
            best_seq = f.copy()
            best_score = float(M)

        improved = False

        # LP direction.
        if rem > 8.0:
            g = _lp_dual(f, M, n, rem)
            if g is not None:
                g = _normalize_to_S(g, S)
                if g is not None:
                    old = f.copy()
                    cur_best_cand = None
                    cur_best_sc = M

                    # Line search for blends of old and LP direction.
                    for tv in (1.0, 0.75, 0.5, 0.33, 0.2, 0.1, 0.05, 0.02):
                        c = (1.0 - tv) * old + tv * g
                        c = _normalize_to_S(c, S)
                        if c is None:
                            continue
                        sc = _max_conv_raw(c, conv_len, fft_len)
                        if sc < cur_best_sc:
                            cur_best_sc = sc
                            cur_best_cand = c.copy()

                    # Optionally add a lightly smoothed version of g.
                    gs = g
                    if n >= 5:
                        kern = np.ones(3, dtype=np.float64) / 3.0
                        gs_raw = np.convolve(g, kern, mode='same')
                        gs = _normalize_to_S(gs_raw, S)
                    if gs is not None:
                        c = 0.5 * old + 0.5 * gs
                        c = _normalize_to_S(c, S)
                        if c is not None:
                            sc = _max_conv_raw(c, conv_len, fft_len)
                            if sc < cur_best_sc:
                                cur_best_sc = sc
                                cur_best_cand = c.copy()

                    if cur_best_cand is not None and cur_best_sc < M - 1e-12:
                        best_seq = cur_best_cand
                        best_score = float(cur_best_sc)
                        improved = True
                        last_improve = time.monotonic()

        # Small random neighborhood around the current best.
        if not improved:
            for _ in range(12):
                c = f.copy()
                wdt = max(1, int(n // 64))
                if wdt >= n:
                    wdt = 1
                start = int(rng.integers(0, n - wdt))
                c[start:start + wdt] *= float(np.exp(rng.normal(0.0, 0.2)))
                c = _normalize_to_S(np.maximum(c, 0.0), S)
                if c is None:
                    continue
                sc = _max_conv_raw(c, conv_len, fft_len)
                if sc < best_score - 1e-12:
                    best_seq = c
                    best_score = float(sc)
                    improved = True
                    last_improve = time.monotonic()
                    break

        # Subgradient refinement if LP/random did not help.
        if not improved:
            local_deadline = min(deadline, time.monotonic() + 25.0)
            if local_deadline > time.monotonic() + 2.0:
                gseq, gsc = _local_gradient_refine(
                    f, n, conv_len, fft_len, S, rng, local_deadline
                )
                if gseq is not None and np.isfinite(gsc) and gsc < best_score - 1e-12:
                    best_seq = gseq
                    best_score = float(gsc)
                    improved = True
                    last_improve = time.monotonic()

        # Last-resort restarting from a fresh small shape pool.
        if not improved:
            small_deadline = time.monotonic() + 8.0
            if small_deadline <= deadline:
                sub_pool = _shape_pool(n, rng, best_seq, 8.0, 3)
                for entry in sub_pool:
                    if float(entry[1]) < best_score - 1e-12:
                        best_seq = np.asarray(entry[0], dtype=np.float64).copy()
                        best_score = float(entry[1])
                        improved = True
                        last_improve = time.monotonic()
                        break

        if time.monotonic() - last_improve > 200.0:
            break

    best_seq = _normalize_to_S(best_seq, S)
    if best_seq is None:
        best_seq = np.full(n, S / float(n), dtype=np.float64)
        best_score = _max_conv_raw(best_seq, conv_len, fft_len)
    return best_seq, float(best_score)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Generate a search sequence and return it as a list of floats.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    budget = max(60.0, float(budget_s) - 15.0)
    deadline = t0 + budget

    height_seq = None
    try:
        height_seq = height_sequence_1
    except Exception:
        height_seq = None

    best_seq = None
    best_score = np.inf

    def _try_update(seq, sc):
        nonlocal_best = None
        # Python helper for readability only; no nested closures are used for
        # helper functions honoring the stated rules.
        return seq, sc

    # Phase 1: robust medium size.
    if time.monotonic() < deadline - 30.0:
        n = 512
        S = float(np.sqrt(2.0 * n))
        conv_len = 2 * n - 1
        fft_len = _ceil_pow2(conv_len)
        pool = _shape_pool(n, rng, height_seq, min(60.0, deadline - time.monotonic() - 40.0), 8)
        if best_seq is not None:
            e = _interp_to_n_nS(best_seq, n, S)
            if e is not None:
                pool = _update_pool(pool, e, n, conv_len, fft_len, S, 8)
        seq, sc = _run_lp_phase(
            n, pool, None, np.inf,
            min(deadline, time.monotonic() + 700.0),
            rng
        )
        if seq is not None and np.isfinite(sc) and sc < best_score:
            best_seq = seq
            best_score = float(sc)
        print(f"N={n} phase score={best_score:.10f}")

    # Phase 2: larger size if time remains.
    if time.monotonic() < deadline - 250.0:
        n = 768
        S = float(np.sqrt(2.0 * n))
        conv_len = 2 * n - 1
        fft_len = _ceil_pow2(conv_len)
        pool = _shape_pool(n, rng, height_seq, min(25.0, deadline - time.monotonic() - 40.0), 6)
        if best_seq is not None:
            e = _interp_to_n_nS(best_seq, n, S)
            if e is not None:
                pool = _update_pool(pool, e, n, conv_len, fft_len, S, 6)
        seq, sc = _run_lp_phase(
            n, pool, best_seq, best_score,
            min(deadline, time.monotonic() + 450.0),
            rng
        )
        if seq is not None and np.isfinite(sc) and sc < best_score:
            best_seq = seq
            best_score = float(sc)
        print(f"N={n} phase score={best_score:.10f}")

    # Phase 3: even larger if possible.
    if time.monotonic() < deadline - 160.0:
        n = 1024
        S = float(np.sqrt(2.0 * n))
        conv_len = 2 * n - 1
        fft_len = _ceil_pow2(conv_len)
        pool = _shape_pool(n, rng, height_seq, min(15.0, deadline - time.monotonic() - 40.0), 5)
        if best_seq is not None:
            e = _interp_to_n_nS(best_seq, n, S)
            if e is not None:
                pool = _update_pool(pool, e, n, conv_len, fft_len, S, 5)
        seq, sc = _run_lp_phase(
            n, pool, best_seq, best_score,
            min(deadline, time.monotonic() + 220.0),
            rng
        )
        if seq is not None and np.isfinite(sc) and sc < best_score:
            best_seq = seq
            best_score = float(sc)
        print(f"N={n} phase score={best_score:.10f}")

    # Final polish on the best sequence using any remaining time.
    if best_seq is not None and time.monotonic() < deadline - 5.0:
        n = int(best_seq.size)
        S = float(np.sqrt(2.0 * n))
        conv_len = 2 * n - 1
        fft_len = _ceil_pow2(conv_len)
        gseq, gsc = _local_gradient_refine(
            best_seq, n, conv_len, fft_len, S, rng, deadline - 5.0
        )
        if gseq is not None and np.isfinite(gsc) and gsc < best_score:
            best_seq = gseq
            best_score = float(gsc)
        print(f"final polish score={best_score:.10f}")

    if best_seq is None:
        n = 512
        S = float(np.sqrt(2.0 * n))
        best_seq = np.full(n, S / float(n), dtype=np.float64)
        best_score = 2.0

    best_seq = np.maximum(np.asarray(best_seq, dtype=np.float64), 0.0)
    S = float(np.sqrt(2.0 * best_seq.size))
    norm = _normalize_to_S(best_seq, S)
    if norm is not None:
        best_seq = norm

    final = [float(x) for x in best_seq]

    # Verify with the authoritative evaluator when available.
    try:
        authoritative = float(evaluate_sequence(final))
        print(f"authoritative score={authoritative:.10f}")
        if not np.isfinite(authoritative):
            n = 512
            S = float(np.sqrt(2.0 * n))
            final = [float(x) for x in np.full(n, S / float(n), dtype=np.float64)]
    except Exception:
        pass

    return final
