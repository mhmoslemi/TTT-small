import numpy as np
import time
from scipy.optimize import linprog


"""
LP-based optimization for minimizing 2*n*max(convolve(a,a))/(sum(a))^2.
Uses iterative linear programming to find sequences compatible with the
current peak convolution bound, then blends with the current iterate.
Multiple restarts and adaptive stepping ensure robust convergence.
"""


def normalize_seq(x, target_sum, n):
    """Normalize a non-negative sequence to have the target sum."""
    s = float(np.sum(x))
    if s < 1e-10:
        return np.full(n, target_sum / n)
    return np.clip(x, 0, 1000) * target_sum / s


def build_A(f):
    """Build (2n-1) x n matrix where A[k,j] = f[k-j] if valid, else 0."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp(f, M):
    """Solve max sum(b) s.t. b>=0, (f*b)[k] <= M for all k."""
    n = len(f)
    A = build_A(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True})
    if res.status == 0 and res.x is not None:
        return res.x
    return None


def conv_max(f):
    """Compute max of full self-convolution."""
    return float(np.max(np.convolve(f, f)))

            best = new_f
            stagnant = 0

            best = normalize_seq(best, target_sum, n)
            best_M = conv_max(best)
            stagnant = 0
            print(f"    [{label}] Restart: M={best_M:.6f}", flush=True)
            if best_M < start_M * 0.9:
                break
