def _pow2_at_least(x):
    p=1

def _norm_pos(x, floor=0.0):

def _dense_grad_top(a,c,K,beta):
   n=a.size
   cmax=float(c.max())

       pos=np.argmax(c)

          if s>0: w/=s
          else: w=np.ones(idx.size)/idx.size

      je=mpar
      if je>=n: je=n-1
      if js>je: continue
      l_start=mpar-je

def _dense_candidate(a, grad, step, rng, noise_sigma=0.0):
   n=a.size
   t=grad - grad.mean()
   norm=np.sqrt(np.dot(t,t))
   x=a.copy()
   if norm>1e-15 and step!=0:
      x = x - (step/norm)*t

def _dense_local(a, deadline, rng, K=60, beta_param=50.0, base_step=0.002, min_step=1e-5, max_step=0.1, verbose=False):
   n=a.size
   a=_norm_pos(a)
   m=_pow2_at_least(2*n-1)
   sc,c=_dense_score_c(a,m)
   best=a.copy(); best_sc=sc; current=sc
   step=base_step; T=0.02; no_improve=0
   it=0
   while time.monotonic()<deadline:
      it+=1
      beta=beta_param*n
      g=_dense_grad_top(a,c,K,beta)
      best_trial=a; best_trial_sc=current
      # line search
      for mult in (0.25,0.5,1.0,2.0):

def _pow2_at_least(x):
    p=1
    while p < x:
        p <<= 1
    return p

def _norm_pos(x):
    x=np.asarray(x,dtype=np.float64)

    s=np.sum(x)

"""
Dense smooth-max local search with strong structured starts.

The method normalizes the sequence to sum 1 and minimizes
    2*n*max(conv(a,a))
using a projected gradient step for a smoothed maximum
(exp(beta * conv(a,a))) plus adaptive line-sampling and simulated
annealing.  It starts from many nonnegative structured sequences,
especially shifted inverse-sqrt laws, which are already near the target.
"""

import numpy as np
import time


def _pow2_at_least(x):
    p = 1
    while p < x:
        p <<= 1
    return p


def _norm_pos(x):
    x = np.asarray(x, dtype=np.float64)
    x = np.maximum(x, 0.0)
    s = float(np.sum(x))
    if s <= 0.0:
        return np.ones_like(x, dtype=np.float64) / float(x.size)
    return x / s


def _dense_c(a, m):
    A = np.fft.rfft(a, m)
    c = np.fft.irfft(A * A, m)
    return c[:2 * a.size - 1]


def _dense_score_c(a, m):
    c = _dense_c(a, m)
    return 2.0 * float(a.size) * float(np.max(c)), c


def _dense_grad_top(a, c, K, beta):
    n = a.size
    cmax = float(np.max(c))
    K = int(K)
    if K <= 0:
        K = 1
    K = min(K, c.size)

    if K == c.size:
        idx = np.arange(c.size, dtype=np.int64)
        vals = c
        w = np.ones(c.size, dtype=np.float64)
        w /= float(c.size)
    else:
        if K == 1:
            pos = int(np.argmax(c))
            idx = np.array([pos], dtype=np.int64)
            vals = c[idx]
        else:
            idx = np.argpartition(c, -K)[-K:]
            vals = c[idx]

        scaled = vals * float(n) - cmax * float(n)
        w = np.exp(beta * scaled)
        s = float(np.sum(w))
        if s <= 0.0 or not np.isfinite(s):
            w = np.ones(idx.size, dtype=np.float64)
            w /= float(idx.size)
        else:
            w = w / s

    g = np.zeros(n, dtype=np.float64)

    for ii in range(idx.size):
        mpar = int(idx[ii])
        wt = float(w[ii])
        if wt == 0.0:
            continue

        js = mpar - (n - 1)
        if js < 0:
            js = 0
        je = mpar
        if je >= n:
            je = n - 1
        if js > je:
            continue

        l_start = mpar - je
        l_len = je - js + 1
        g[js:je + 1] += wt * a[l_start:l_start + l_len][::-1]

    return 2.0 * g


def _dense_grad_full(a, c, beta, m_grad):
    n = a.size
    sn = c * float(n)
    smax = float(np.max(sn))
    u = np.exp(beta * (sn - smax))
    s = float(np.sum(u))
    if s <= 0.0 or not np.isfinite(s):
        u = np.ones(c.size, dtype=np.float64)
        s = float(c.size)
    u = u / s

    Ar = np.fft.rfft(a[::-1], m_grad)
    Ur = np.fft.rfft(u, m_grad)
    conv = np.fft.irfft(Ar * Ur, m_grad)
    return 2.0 * conv[n - 1:n - 1 + n]


def _dense_candidate(a, grad, step, rng, noise=0.0):
    n = a.size
    t = grad - float(np.mean(grad))
    norm = float(np.linalg.norm(t))
    x = a.copy()

    if norm > 1e-14 and step != 0.0:
        x = x - (step / norm) * t

    if noise > 0.0:
        x = x + rng.normal(0.0, noise, n)

    x = np.maximum(x, 0.0)
    x = x + 1e-16
    s = float(np.sum(x))
    if s <= 0.0:
        return a.copy()
    return x / s


def _dense_local(a0, limit, rng, m_self, m_grad, K=80, base_step=0.003, verbose_tag=""):
    n = a0.size
    a = _norm_pos(a0)
    sc, c = _dense_score_c(a, m_self)

    best = a.copy()
    best_sc = sc
    current = sc

    step = base_step
    T = 0.02
    no_improve = 0
    it = 0
    beta_scale = 20.0

    while time.monotonic() < limit:
        it += 1
        beta_scale = min(500.0, 20.0 + 0.2 * float(it))

        if it % 5 == 0:
            g = _dense_grad_top(a, c, K, beta_scale)
        else:
            g = _dense_grad_full(a, c, beta_scale, m_grad)

        if not np.all(np.isfinite(g)):
            g = _dense_grad_top(a, c, K, beta_scale)

        best_trial = a
        best_trial_sc = current

        for mult in (0.25, 0.5, 1.0, 2.0):
            cand = _dense_candidate(a, g, step * mult, rng, 0.0)
            s2, _ = _dense_score_c(cand, m_self)
            if np.isfinite(s2) and s2 < best_trial_sc:
                best_trial_sc = s2
                best_trial = cand

        if no_improve > 50 or rng.random() < 0.05:
            sig = 0.02 + 0.8 * T
            cand = a * np.exp(rng.normal(0.0, sig, n))
            cand = _norm_pos(cand)
            s2, _ = _dense_score_c(cand, m_self)
            if np.isfinite(s2) and s2 < best_trial_sc:
                best_trial_sc = s2
                best_trial = cand

        accept = False
        if best_trial_sc < current:
            accept = True
        else:
            diff = best_trial_sc - current
            if diff < 1e-4:
                accept = rng.random() < 0.25
            else:
                z = diff / max(T, 1e-12)
                if z < 700.0:
                    accept = rng.random() < np.exp(-z)

        if accept:
            if best_trial_sc < best_sc:
                best = best_trial.copy()
                best_sc = best_trial_sc
                no_improve = 0
                T = max(1e-5, T * 0.98)
            else:
                no_improve += 1
                T = max(1e-5, T * 0.995)

            current = best_trial_sc
            a = best_trial
            c = _dense_c(a, m_self)

            if best_trial_sc < current - 1e-12:
                step *= 1.1
            else:
                step *= 0.92
        else:
            step *= 0.85
            no_improve += 1
            T = max(1e-5, T * 0.995)

        step = float(np.clip(step, 1e-5, 0.05))

        if no_improve > 350:
            a = best * np.exp(rng.normal(0.0, 0.06, n))
            a = _norm_pos(a)
            current, c = _dense_score_c(a, m_self)
            step = base_step
            no_improve = 0
            beta_scale = 20.0

        if verbose_tag != "" and (it % 500 == 0):
            print(verbose_tag, "iter", it, "best", best_sc, "cur", current, "step", step)

    return best, best_sc


def _make_dense_inits(n, rng, parent):
    seqs = []
    i = np.arange(n, dtype=np.float64)
    x = i / float(max(1, n - 1))

    seqs.append(np.ones(n, dtype=np.float64))

    for gamma in (0.3, 0.5, 0.8, 1.2, 1.8, 2.6):
        seqs.append(np.exp(-gamma * x))

    for p in (0.45, 0.5, 0.55):
        for d in (0.25, 0.5, 1.0, 2.0, 5.0):
            seqs.append((i + d) ** (-p))

    for d in (0.5, 1.0):
        for gamma in (0.2, 0.5, 1.0, 2.0):
            seqs.append((i + d) ** (-0.5) * np.exp(-gamma * x))

    base = (i + 0.5) ** (-0.5)
    for A in (0.25, 0.5, 0.8, 1.2):
        seqs.append(base * (1.0 + A * x))
        seqs.append(base * (1.0 + A * (2.0 * x - 1.0)))

    base2 = (float(n - 1) - i + 0.5) ** (-0.5)
    for alpha in (0.05, 0.1, 0.2, 0.4):
        seqs.append(base + alpha * base2)

    for sigma in (0.05, 0.1, 0.2):
        for _ in range(3):
            mult = np.exp(rng.normal(0.0, sigma, n))
            seqs.append(base * mult)

    if parent is not None and parent.size == n and float(np.sum(parent)) > 0.0:
        seqs.append(parent)

    return seqs
