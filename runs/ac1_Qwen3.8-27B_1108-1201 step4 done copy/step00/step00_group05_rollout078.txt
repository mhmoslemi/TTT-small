"""

simplex projection. Uses FFT for O(n log n) per-iteration cost. The score
is scale-invariant, so we normalize sum(a)=1 and minimize 2*n*max(c) where
c = convolve(a,a). The theoretical lower bound is 2n/(2n-1) ≈ 1.

def fft_convolve(a):
    """Full linear convolution of a with itself via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    C = np.fft.irfft(A * A, fft_size)[:size]

    """Project onto {x >= 0, sum(x) = 1} via bisection (water-filling)."""

    rs = result.sum()
    if rs > 1e-15:

    else:

    return result

    spec = np.fft.rfft(a_rev, fs) * np.fft.rfft(w, fs)
    conv_res = np.fft.irfft(spec, fs)[:conv_len]
    return 2.0 * conv_res[n - 1:2 * n - 1]


def grad_l2(a, c):
    """Gradient of sum_k c[k]^2. O(n log n) via FFT.
    d(sum c^2)/da[j] = 4 * sum_k c[k]*a[k-j] = 4*conv(c, a_rev)[j+n-1]"""
    n = len(a)
    a_rev = a[::-1].copy()
    conv_len = (2 * n - 1) + n - 1
    fs = 1 << (conv_len - 1).bit_length()
    spec = np.fft.rfft(c, fs) * np.fft.rfft(a_rev, fs)
    conv_res = np.fft.irfft(spec, fs)[:conv_len]
    return 4.0 * conv_res[n - 1:2 * n - 1]


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for sequence minimizing 2*n*max(conv(a,a))/sum(a)^2."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 5.0)

    # Initialization
    try:
        a0 = np.array(height_sequence_1, dtype=np.float64)
        a0 = np.clip(a0, 0.0, 1000.0)
        if a0.ndim != 1 or len(a0) == 0 or not np.all(np.isfinite(a0)) or a0.sum() < 1e-10:
            raise ValueError("bad")
    except (NameError, ValueError, Exception):
        a0 = np.ones(4498, dtype=np.float64)

    n = len(a0)
    best_a = a0 / a0.sum()
    best_score = 2.0 * n * fft_convolve(best_a).max()
    print(f"n={n}, initial score={best_score:.6f}", flush=True)

    no_improve = 0
    restart_count = 0
    max_restarts = 20

    while time.monotonic() < deadline and restart_count < max_restarts:
        # Perturb to break symmetry
        if restart_count == 0:
            a = best_a.copy()
            # Small random perturbation
            a += rng.normal(0, 0.01 / np.sqrt(n), size=n)
            a = project_simplex(a)
        else:
            a = best_a.copy()
            # Try different structural perturbations
            mode = restart_count % 5
            if mode == 0:
                freq = rng.integers(1, n // 2)
                a += 0.05 * np.cos(2 * np.pi * freq * np.arange(n) / n)
            elif mode == 1:
                freq = rng.integers(1, n // 2)
                a += 0.05 * np.sin(2 * np.pi * freq * np.arange(n) / n)
            elif mode == 2:
                a += 0.03 * rng.normal(0, 1, size=n) / np.sqrt(n)
            elif mode == 3:
                # Bathtub: reduce center
                center = n // 2
                width = n // 4
                a[center - width:center + width] *= 0.7
            else:
                # Edge boost
                edge = n // 5
                a[:edge] *= 1.3
                a[-edge:] *= 1.3
            a = project_simplex(a)

        # Gradient descent loop
        eps = 2.0 / n
        step = 0.3
        local_best = a.copy()
        local_best_score = 2.0 * n * fft_convolve(local_best).max()
        stall = 0
        max_stall = 300

        t_start = time.monotonic()
        while time.monotonic() < deadline:
            rem = deadline - time.monotonic()
            if rem < 2.0:
                break

            c = fft_convolve(a)
            cur_score = 2.0 * n * c.max()

            # Blend L2 and logsumexp gradients
            if eps > 5.0 / n:
                g = grad_l2(a, c) * 0.5 + grad_logsumexp(a, c, eps) * 0.5
            else:
                g = grad_logsumexp(a, c, eps)

            # Adaptive line search
            improved = False
            for trial in [step, step * 0.5, step * 0.25]:
                a_try = project_simplex(a - trial * g)
                c_try = fft_convolve(a_try)
                s_try = 2.0 * n * c_try.max()
                if s_try < cur_score:
                    a = a_try
                    cur_score = s_try
                    improved = True
                    break

            if not improved:
                step = max(step * 0.8, 0.01)
                stall += 1
            else:
                if step < 1.0:
                    step = min(step * 1.1, 1.0)
                stall = 0

            if cur_score < local_best_score:
                local_best_score = cur_score
                local_best = a.copy()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            # Cool eps
            eps = max(eps * 0.995, 1.0 / n)

            if stall > max_stall:
                break

            if (int(time.monotonic() - t_start) // 10) > (int(time.monotonic() - t_start - 0.1) // 10):
                print(f"  restart={restart_count} score={cur_score:.6f} best={best_score:.6f} eps={eps:.1e} step={step:.3f}", flush=True)
                t_start = time.monotonic()

        restart_count += 1
        if local_best_score < best_score:
            best_score = local_best_score
            best_a = local_best.copy()

        print(f"Restart {restart_count} done: best_score={best_score:.6f}", flush=True)
        if best_score < 1.51:
            break

    print(f"FINAL: score={best_score:.8f}, n={n}", flush=True)
    return [float(x) for x in best_a]
