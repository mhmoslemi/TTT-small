import numpy as np
from numpy.fft import rfft, irfft

def autoconv(a):
    """Compute autoconvolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Use next power of 2 for FFT efficiency
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    fa = rfft(a, fft_size)
    return irfft(fa * fa.conj(), fft_size)[:size]

import time
from numpy.lib.stride_tricks import as_strided

def fast_autoconv(a):
    """Compute full autoconvolution (length 2n-1) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(fa * fa, fft_size)
    return result[:size]

def compute_norm_score(a):
    """Compute score assuming sum(a) = 1. Returns 2*n*max(autoconv)."""
    n = len(a)
    c = fast_autoconv(a)
    return 2.0 * n * np.max(c)

def grad_step(a, temp, lr):
    """One gradient descent step on max(autoconv) with log-sum-exp smoothing."""
    n = len(a)
    c = fast_autoconv(a)
    max_c = np.max(c)
    t_c = temp * (c - max_c)
    # Clip to avoid overflow
    t_c = np.clip(t_c, -50, 50)
    p = np.exp(t_c)
    p_sum = p.sum()
    if p_sum < 1e-30:
        idx = np.argmax(c)
        p = np.zeros_like(c)
        p[idx] = 1.0

    p_mat = as_strided(p, shape=(n, n), strides=(p.strides[0], p.strides[0]))
    grad = p_mat @ a
    # Update

    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s

Minimizes 2*n*max(autocorrelation)/(sum)^2 using a two-phase gradient method:
Phase 1: L2-smoothed gradient (minimize sum(c^2)) for fast convergence to flat autoconv.
Phase 2: Log-sum-exp (soft-max) gradient for fine-tuning the peak.
Includes adaptive temperature, random restarts, and structured initializations.
Score is scale-invariant, so we work with sum(a)=1 sequences.

def fast_autoconv(a):
    """Compute full autoconvolution (length 2n-1) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(fa * fa, fft_size)
    return np.maximum(result[:size], 0.0)


def cross_corr_sliding(weights, a):
    """Compute g[i] = sum_j weights[i+j] * a[j] for i=0..n-1 via stride tricks + BLAS."""
    n = len(a)
    w_mat = as_strided(weights, shape=(n, n), strides=(weights.strides[0], weights.strides[0]))
    return w_mat @ a


def _normalize(a, n):
    """Ensure a is non-negative and sums to 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(n, 1.0 / n)
    return a / s


def _score_norm(a):
    """Compute 2*n*max(autoconv) for a sequence with sum=1."""
    n = len(a)
    c = fast_autoconv(a)
    return 2.0 * n * float(np.max(c))

            break
        c = fast_autoconv(a)
        g = cross_corr_sliding(c, a)
        a = a - lr * 2.0 * g

        lr *= 0.9995
        if lr < 0.005:

        a = a - lr * g
