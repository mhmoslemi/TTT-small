- Decrease the larger of a[j] and a[k*-j] by a small amount
- Increase a "quiet" position (one that doesn't contribute much to any peak) by the same amount
- Repeat

This is essentially a coordinate descent / coordinate swap approach. It's simple but might be slow.

import time
from scipy.optimize import linprog

def compute_score(a):
    """Fast score computation without security checks."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    R = np.convolve(a, a)
    return float(2.0 * n * R.max() / (S * S))

def build_conv_matrix(a):
    """Build dense convolution matrix A where A @ b = convolve(a, b)."""
    n = len(a)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = a
    return A

def lp_step(a, M):
    """Solve the LP: max sum(b) s.t. convolve(a,b) <= M, b >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    res = linprog(c, A_ub=A, b_ub=b_ub, 
                 bounds=[(0, None)]*n, method='highs')
    if res.success:
        return np.maximum(res.x, 0)
    return None

        i_min = max(0, k - n + 1)
