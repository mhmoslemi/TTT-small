LP-guided optimization for minimizing 2*n*max(convolve(f,f))/(sum(f))^2.
Uses a convex linear program to find descent directions, with adaptive
mixing, line search, and random perturbations to escape local optima.
Key insight: solving max sum(g) s.t. conv(f,g) <= max(f*f) elementwise
finds a "spread out" direction that reduces the autocorrelation peak.
"""

import numpy as np
import time
from scipy.optimize import linprog


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where (A @ g) = np.convolve(f, g)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(n):
        A[k:k + n, k] = f
    return A


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M (elementwise), g >= 0.
    Returns optimal g array or None if LP fails."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def compute_objective(f):
    """Directly compute 2*n*max(f*f)/(sum(f))^2."""
    s = f.sum()
    if s < 1e-10:
        return np.inf
    conv = np.convolve(f, f)
    M = conv.max()
    return 2.0 * len(f) * M / (s * s)

            else:

            g_sum = g.sum()
            if g_sum < 1e-10:

            else:

        best_score = final_score
    print(f"Final: n={n}, best_score={best_score:.6f}, iterations={iterations}",
          flush=True)
    return [float(x) for x in best_f]
