"""
Minimize 2*n*max(f*f)/(sum(f))^2 using gradient descent in log-space
with multiple parametric initializations, adaptive learning rate, and restarts.
Uses the subgradient of the piecewise-linear (in max) objective at the
current argmax of the auto-convolution.
"""

import numpy as np
import time


def _convolve_max(a):
    """Return (max_val, max_pos) of np.convolve(a, a)."""
    conv = np.convolve(a, a)
    max_pos = int(np.argmax(conv))
    return float(conv[max_pos]), max_pos


def _score(a):
    """Compute the evaluation score for a non-negative sequence."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    max_b, _ = _convolve_max(a)
    return 2.0 * n * max_b / (s * s)


def _grad_step(a, lr, kstar):
    """
    One subgradient step in log-space for minimizing F = 2n*M/S^2.
    a: non-negative array (will be clipped internally)
    lr: learning rate
    kstar: current argmax of the auto-convolution
    Returns updated a.
    """
    a = np.clip(a, 1e-10, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return a

    conv = np.convolve(a, a)
    M = float(conv[kstar])

    # Valid range for i in dC[kstar]/da[i]
    lo = max(0, kstar - n + 1)
    hi = min(n - 1, kstar)

    # In log-space: dF/dx[i] = a[i] * dF/da[i]
    # dF/da[i] = 2n*(2*a[kstar-i]/S^2 - 2*M*a[i]/S^3) for i in [lo,hi]
    # dF/da[i] = 2n*(0 - 2*M*a[i]/S^3) for i outside [lo,hi]

    x = np.log(a)

    # Compute gradient in log-space
    # Outside [lo,hi]:
    grad = np.full(n, -4.0 * n * M * a * a / (S ** 3))
    # Inside [lo,hi]:
    idx = np.arange(lo, hi + 1)
    grad[lo:hi + 1] = (4.0 * n * a[idx] * a[kstar - idx] / (S ** 2)
                       - 4.0 * n * M * a[idx] * a[idx] / (S ** 3))

    # Update
    x -= lr * grad
    x = np.clip(x, np.log(1e-10), np.log(1000.0))
    return np.exp(x)

        print(f"  NEW BEST: {label} score={s:.6f} n={len(a)} "

    return s


def _parametric_seqs(n):
    """Generate a list of parametric initial sequences of length n."""
    i = np.arange(n, dtype=np.float64)
    seqs = []
    labels = []

    # Uniform
    seqs.append(np.ones(n))
    labels.append("uniform")

    # Gaussian
    for sigma_ratio in [0.2, 0.4, 0.6, 1.0]:
        sigma = sigma_ratio * n / 2.0
        seqs.append(np.exp(-((i - n / 2.0) ** 2) / (2.0 * sigma * sigma)))
        labels.append(f"gauss_{sigma_ratio}")

    # Cosine (half period, peaking at edges)
    for beta in [0.3, 0.6, 0.9, 1.2]:
        seqs.append(np.clip(1.0 + beta * np.cos(np.pi * i / (n - 1)), 0.01, None))
        labels.append(f"cos_{beta}")

    # Cos^2 (peaking at edges, zero at center)
    seqs.append(np.cos(np.pi * i / (2.0 * (n - 1))) ** 2 + 0.01)
    labels.append("cos2")

    # Sin^2 (zero at edges, peaking at center)
    seqs.append(np.sin(np.pi * i / (n - 1)) ** 2 + 0.01)
    labels.append("sin2")

    # Raised cosine (one full period, peaking at center)
    seqs.append(0.5 * (1.0 + np.cos(2.0 * np.pi * i / (n - 1))) + 0.01)
    labels.append("raised_cos")

    # Power law decay
    for p in [1.0, 2.0, 4.0]:
        seqs.append((1.0 - i / (n - 1)) ** p + 0.01)
        labels.append(f"power_{p}")

    # Triangle
    seqs.append(1.0 - np.abs(2.0 * i / (n - 1) - 1.0) + 0.01)
    labels.append("tri")

    # Flat top
    for width in [0.5, 0.7, 0.9]:
        t = i / (n - 1)
        flat = np.where(np.abs(t - 0.5) < width / 2.0, 1.0,
                        np.maximum(0.0, 1.0 - (np.abs(t - 0.5) - width / 2.0) * 4.0))
        seqs.append(flat + 0.01)
        labels.append(f"flat_{width}")

    return seqs, labels
