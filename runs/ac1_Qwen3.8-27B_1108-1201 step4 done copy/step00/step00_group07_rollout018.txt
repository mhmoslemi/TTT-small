import numpy as np
import time

def full_conv(f):
    """Compute full self-convolution using FFT."""
    n = len(f)
    size = 2 * n - 1
    P = 1
    while P < size:
        P *= 2
    F = np.fft.rfft(f, P)
    result = np.fft.irfft(F * F, P)[:size]
    return result

def compute_grad(f, w):
    """Compute gradient of sum_k w[k]*g[k] where g = f*f."""
    n = len(f)
    size = 2 * n - 1
    P = 1
    while P < 3 * n:
        P *= 2
    
    F = np.fft.rfft(f, P)
    W = np.fft.rfft(w, P)
    W_rev = np.fft.rfft(w[::-1], P)
    
    # term1[m + 2n - 2] where term1 = ifft(FFT(w[::-1]) * FFT(f))
    term1 = np.fft.irfft(W_rev * F, P)
    # term2[m] where term2 = ifft(FFT(f) * FFT(w))
    term2 = np.fft.irfft(F * W, P)
    
    # grad[m] = term1[m + 2n - 2] + term2[m] for m = 0, ..., n-1
    grad = term1[2*n-2 : 2*n-2+n] + term2[:n]
    return grad
