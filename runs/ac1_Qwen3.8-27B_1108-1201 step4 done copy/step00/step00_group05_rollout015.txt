```python
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix

def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    num_cons = 2*n - 1
    
    # Build constraint matrix A where A[k, i] = f[i] if 0 <= k-i < n else 0
    # A has shape (2n-1, n)
    A = lil_matrix((num_cons, n), dtype=np.float64)
    for k in range(num_cons):
        # For row k: A[k, i] = f[i] when max(0, k-n+1) <= i <= min(n-1, k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            A[k, i_start:i_end+1] = f[i_start:i_end+1]
    
    A = csr_matrix(A)
    
    # Objective: minimize -sum(g)
    c = -np.ones(n)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    # Solve
    res = linprog(c, A_ub=A, b_ub=np.full(num_cons, M), bounds=bounds, method='highs')
    
    if res.status == 0:
        return res.x
    return None
```

```python
"""
Optimizes a non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based direction finding combined with gradient refinement.
Starts from a two-block initialization (ratio 2, achieving ~1.778) and iteratively
improves via linear programming for strong descent directions.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _two_block_init(n):
    """Create two-block initialization [x^(n/2), y^(n/2)] with x/y = 2."""
    m = n // 2
    a = np.zeros(n)
    a[:m] = 2.0
    a[m:] = 1.0
    return a


def _normalize(a):
    """Normalize to sum=1."""
    s = a.sum()
    if s < 1e-12:
        return np.ones(len(a)) / len(a)
    return a / s


def _compute_objective(a):
    """Compute 2*n*max(conv)/(sum)^2 directly."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    return float(2.0 * n * conv.max() / (s * s))


def _build_constraint_matrix(f):
    """Build sparse (2n-1) x n matrix where A[k,i] = f[k-i] if valid."""
    n = len(f)
    num_rows = 2 * n - 1
    nnz = n * n

    data = np.empty(nnz, dtype=np.float64)
    col_ind = np.empty(nnz, dtype=np.int32)
    row_ind = np.empty(nnz, dtype=np.int32)

    pos = 0
    for k in range(num_rows):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        count = i_end - i_start + 1
        row_ind[pos:pos + count] = k
        col_ind[pos:pos + count] = np.arange(i_start, i_end + 1, dtype=np.int32)
        idx = k - np.arange(i_end, i_start - 1, -1, dtype=np.int32)
        data[pos:pos + count] = f[idx]
        pos += count

    return csr_matrix((data, (row_ind, col_ind)), shape=(num_rows, n))


def _solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g)[k] <= M for all k, g >= 0.
    Returns g (normalized to sum=1) or None."""
    n = len(f)
    num_rows = 2 * n - 1

    A = _build_constraint_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(num_rows, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.status == 0 and res.x is not None and res.x.sum() > 1e-10:
            return _normalize(res.x)
    except Exception:
        pass
    return None


def _gradient_step(a, n_steps=50, initial_tau=0.05):
    """Smoothed gradient descent on max(conv) with sum(a)=1 constraint."""
    n = len(f) if 'f' in dir() else len(a)
    n = len(a)
    tau = initial_tau
    eta = 0.01
    a = a.copy().astype(np.float64)

    for step in range(n_steps):
        conv = np.convolve(a, a)
        # Softmax weights (smoothed max)
        shifted = conv / tau - conv.max()
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w /= w_sum

        # Gradient: grad[j] ≈ 2 * cross-correlation of w and a at lag j
        # (w ⋆ a)[j] = sum_k w[k] * a[k-j] (linear cross-correlation)
        size = 2 * n
        W = np.fft.rfft(w, size)
        A = np.fft.rfft(a, size)
        corr = np.fft.irfft(W * np.conj(A), size)[:n]
        grad = 2.0 * corr

        # Update with clipping and renormalization
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a
        else:
            a_new /= s

        # Check if objective improved (simple acceptance)
        old_obj = conv.max()
        new_conv_max = np.convolve(a_new, a_new).max()
        if new_conv_max < old_obj:
            a = a_new
        else:
            # Try smaller step
            a_trial = a - 0.5 * eta * grad
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st > 1e-10:
                a_trial /= st
                if np.convolve(a_trial, a_trial).max() < old_obj:
                    a = a_trial
        eta *= 0.98
        tau = max(tau * 0.95, 0.001)

    return a


def _random_perturbation(a, rng, scale=0.1):
    """Apply multiplicative random perturbation."""
    n = len(a)
    width = max(1, n // 50)
    start = rng.integers(0, n - width)
    pert = np.exp(rng.normal(0, scale, size=width))
    a_new = a.copy()
    a_new[start:start + width] *= pert
    return _normalize(a_new)
```
