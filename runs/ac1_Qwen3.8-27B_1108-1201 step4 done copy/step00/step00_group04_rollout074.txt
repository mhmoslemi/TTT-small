"""
Hybrid optimizer for minimizing the peak-to-average convolution ratio.
Uses soft-max gradient descent for fast exploration, then LP-based
refinement for precise optimization.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Full linear convolution of two 1D arrays using FFT."""
    n = len(a) + len(b) - 1
    fft_size = 1
    while fft_size < n:
        fft_size <<= 1
    fa = np.fft.rfft(a, fft_size)
    fb = np.fft.rfft(b, fft_size)
    return np.fft.irfft(fa * fb, fft_size)[:n]


def compute_convolution(a):
    """Compute full autocorrelation a*a using FFT."""
    n = len(a)
    total = 2 * n - 1
    fft_size = 1
    while fft_size < total:
        fft_size <<= 1
    fa = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(fa * fa, fft_size)[:total]
    return result


def compute_gradient(a, p, n):
    """
    Compute gradient of sum_k p[k] * (a*a)[k] w.r.t. a.
    grad[j] = 2 * sum_m a[m] * p[m+j]
    Uses the identity: grad = 2 * (a_flipped * p)[n-1 : 2n-1]

    # conv[k] = sum_i a_flip[i] * p[k-i] for valid pairs
    # We want conv[n-1+j] for j = 0, ..., n-1
    grad = 2.0 * conv[n - 1 : 2 * n - 1]
    return grad
