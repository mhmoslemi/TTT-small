def build_conv_constraint_matrix(a):
    """Build sparse matrix A where A[k,j] = a[k-j] if 0<=k-j<=n-1 else 0."""
    n = len(a)
    rows_list = []
    cols_list = []
    vals_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        js = np.arange(j_start, j_end + 1)
        rows_list.extend([k] * len(js))
        cols_list.extend(js)
        vals_list.extend(a[k - js])
    from scipy.sparse import csr_matrix
    return csr_matrix((vals_list, (rows_list, cols_list)), shape=(2*n-1, n))


def fft_convolve(a, b):
    """Compute full convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    result = np.fft.irfft(fa * fb, size)
    return result[:n + m - 1]


def subgradient_update(a, eta, top_k=50):
    """Subgradient step to reduce peak autocorrelation."""
    n = len(a)
    c = fft_convolve(a, a)
    M = c.max()
    if M <= 0:
        return a.copy()
    
    # Find top K positions
    if top_k >= len(c):
        active = np.arange(len(c))
        weights = np.ones(len(c))
    else:
        # Use top K with smooth weights
        top_indices = np.argsort(c)[::-1][:top_k]
        top_values = c[top_indices]
        # Weight by how close to max (smooth version)
        threshold = M * 0.9  # include positions with value >= 90% of max
        active_mask = c >= threshold
        active = np.where(active_mask)[0]
        if len(active) == 0:
            active = top_indices[:1]
        weights = c[active] / M  # normalized weights
    
    # Compute subgradient: sg[i] = sum_{k in active} weights[k] * 2*a[k-i]
    sg = np.zeros(n)
    for k, w in zip(active, weights):
        # a[k-i] for i in [0, n-1], valid when 0 <= k-i <= n-1
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_start <= i_end:
            j_start = k - i_end
            j_end = k - i_start
            sg[i_start:i_end+1] += w * 2 * a[j_start:j_end+1]
    
    # Multiplicative update
    a_new = a * np.exp(-eta * sg / (n * a.max() + 1e-10))
    a_new = np.clip(a_new, 0, 1000)
    S = a.sum()
    if a_new.sum() > 0:
        a_new = a_new * (S / a_new.sum())
    return a_new

This should be very fast. Let me estimate: each iteration is about 2 FFTs (one for convolution, one for cross-correlation) plus some O(n) work. For n=3809, each FFT takes about 0.1ms. So each iteration is about 1ms. In 1800 seconds: 1.8M iterations.

But we need to be careful: the subgradient method might not converge to the global optimum. It might get stuck in local minima. To counter this:
- Use a decreasing step size (like 1/sqrt(t))
- Occasionally do a "restart" (perturb the current solution)
- Try multiple initializations

Let me also think about the weight function w. Options:
1. w[k] = 1 for k in active set (hard threshold): only reduces the actual peak
2. w[k] = (c[k]/M)^p for p > 1: smooth version, p larger = more peak-focused
3. w[k] = exp(β*c[k]) / sum(exp(β*c[k])): softmax, β controls sharpness
4. w[k] = c[k]/M: linear weighting

I think option 3 (softmax) is the best compromise between focusing on the peak and smoothing. With large β, it approximates the hard max.

For the step size, I'll use:
- eta_t = eta_0 / sqrt(t + 1) (diminishing step size)
- Or eta_t = 2 / (L * (t + 1)) (standard subgradient)

Where L is an upper bound on the gradient norm.

The gradient norm: ||g||_2 ≤ 2 * ||w||_2 * ||a||_2 (by Cauchy-Schwarz in the cross-correlation)
Actually, ||g||_2 = 2 * ||cross_corr(w, a)||_2 ≤ 2 * ||w||_1 * ||a||_∞ (by properties of cross-correlation)
Hmm, this is getting complicated. Let me just use an adaptive step size:
- Start with eta = 1.0
- If the step makes M worse, halve eta
- If M improves, keep eta (or increase slightly)

Actually, a simpler approach: use the "mirror descent" (multiplicative) update:
a_new[i] = a[i] * exp(-eta * g[i] / (n * mean(a)))

This naturally keeps a positive and is more robust to step size choices.

Let me also think about whether the subgradient direction is correct. We want to minimize M = max_k c[k]. The subgradient points in the direction of increasing M, so we should move in the negative gradient direction:
a_new = a - eta * g (for additive) or a_new = a * exp(-eta * g) (for multiplicative)

This should reduce M. But we also need to maintain the constraint sum(a) = S. The normalization step handles this.

One concern: the multiplicative update a * exp(-eta * g) might make some a[i] very small (close to 0), which could lead to a sequence with many zero entries. This might not be optimal but should still be valid.

"""
Iterative subgradient optimization to minimize peak autocorrelation of a non-negative sequence.
Uses FFT-based convolution, softmax-weighted subgradients, and multiplicative updates.
Combines fast per-iteration subgradient steps with occasional LP-based improvements.
"""

import numpy as np
import time


def _fft_convolve_full(a, b):
    """Compute full linear convolution using FFT."""
    na, nb = len(a), len(b)
    n = na + nb - 1
    size = 1
    while size < n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]


def _fft_xcorr(a, b):
    """
    Compute cross-correlation: result[i] = sum_k a[k] * b[k-i]
    where b[j] = 0 for j outside [0, len(b)-1].
    Returns array of length len(b).
    """
    na, nb = len(a), len(b)
    n = na + nb - 1
    size = 1
    while size < n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    r = np.fft.irfft(fa * np.conj(fb), size)
    return r[:nb]


def _compute_score(a):
    """Compute the objective: 2*n*M/S^2 where M=max(conv(a,a)), S=sum(a)."""
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = _fft_convolve_full(a, a)
    M = c.max()
    return 2.0 * len(a) * M / (S * S)


def _normalize(a, target_sum):
    """Normalize non-negative array to have target_sum."""
    s = a.sum()
    if s > 1e-15:
        a = a * (target_sum / s)
    return np.clip(a, 0.0, 1000.0)


def _softmax_weights(c, beta):
    """Compute softmax weights on convolution values."""
    c_shifted = c - c.max()
    w = np.exp(beta * c_shifted)
    w_sum = w.sum()
    if w_sum > 0:
        w = w / w_sum
    else:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    return w


def _subgradient_step(a, eta, beta, mode='mul'):
    """
    One subgradient step to reduce peak autocorrelation.
    Returns updated a (unnormalized) and the current max convolution value.
    """
    n = len(a)
    c = _fft_convolve_full(a, a)
    M = c.max()
    if M <= 0:
        return a.copy(), M

    # Compute weights
    w = _softmax_weights(c, beta)

    # Compute gradient: g[i] = 2 * sum_k w[k] * a[k-i]
    # This is 2 * cross_correlation(w, a)
    g = 2.0 * _fft_xcorr(w, a)

    # Update
    if mode == 'mul':
        # Multiplicative update (mirror descent)
        scale = n * a.mean() + 1e-10
        a_new = a * np.exp(-eta * g / scale)
    else:
        # Additive update
        scale = n * a.mean() + 1e-10
        a_new = a - eta * g / scale
        a_new = np.clip(a_new, 0.0, None)

    return a_new, M


def _lp_step(a, time_limit=30):
    """
    LP-based best-response step.
    Solves: max sum(b) s.t. conv(a,b)[k] <= M for all k, 0 <= b <= 1000.
    Returns the LP solution if successful, None otherwise.
    """
    from scipy.optimize import linprog
    from scipy.sparse import toeplitz as sp_toeplitz

    n = len(a)
    c = _fft_convolve_full(a, a)
    M = c.max()
    if M <= 0:
        return a.copy()

    # Build Toeplitz constraint matrix
    col = np.zeros(2 * n - 1)
    col[:n] = a
    row = np.zeros(n)
    row[0] = a[0]
    A = sp_toeplitz(col, row)

    # LP: maximize sum(b) => minimize -sum(b)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, 1000.0)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def _random_perturb(a, rng, sigma=0.1):
    """Apply random multiplicative perturbation."""
    pert = np.exp(rng.normal(0, sigma, len(a)))
    a_new = a * pert
    return _normalize(np.clip(a_new, 0, 1000), a.sum())

    3. Periodically attempt LP-based improvements

    n = 3809
    try:
        a_init = np.clip(np.array(height_sequence_1, dtype=np.float64), 0, 1000)

            raise ValueError

    # Two-scale: high in middle, low at edges
