        if max_conv < 1e-15:
            break
        
        scaled = t_param * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum
        
        # Compute subgradient
        sg = np.zeros(n)
        threshold = w.max() * 0.005
        active_k = np.where(w > threshold)[0]
        
        for k in active_k:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                sg[j_start:j_end+1] += w[k] * a[k-j_end : k-j_start+1][::-1]
        
        # Multiplicative update
        sg_centered = sg - sg.mean()
        # a[j] *= exp(-eta * sg_centered[j])
        # For numerical stability:
        log_step = -eta * sg_centered
        log_step -= log_step.max()  # prevent overflow
        a = a * np.exp(log_step)
        a = np.maximum(a, 0)
        a_sum = a.sum()
        if a_sum < 1e-15:
            a = np.ones(n) / n
        else:
            a /= a_sum
        
        # Recompute convolution and objective
        conv = fast_convolve(a)
        current_obj = 2.0 * n * conv.max()
        
        if current_obj < best_obj - 1e-10:
            best_obj = current_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > max_patience:
            eta *= 0.7
            patience = 0
            if eta < 0.001:
                # Try increasing t_param for sharper focus
                t_param *= 1.5
                eta = 0.1
                max_patience = 50
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multiplicative mirror descent
(exponential weights) with soft-max smoothing over convolution peaks.
Tries multiple sequence lengths (64-2048) and diverse initializations
(uniform, Gaussian, ramp, bimodal, random, parent-derived), with a
two-phase strategy: quick scan then deep optimization on best configs.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_obj_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def do_optimize(a0, time_limit, t_param=40.0, eta0=0.5, top_k_frac=0.02):
    """
    Multiplicative mirror descent on the probability simplex.
    a0 should be non-negative (will be normalized to sum=1).
    Returns (best_a, best_objective_normalized).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.ones(n) / n
    else:
        a /= a_sum

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    eta = eta0
    patience = 0
    max_patience = 80
    current_t = t_param
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-15:
            break

        # Soft-max weights over convolution
        scaled = current_t * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum

        # Compute weighted subgradient: sg[j] = sum_k w[k] * a[k-j] (valid range)
        sg = np.zeros(n)
        # Only process significant weights
        thresh = w.max() * 0.003
        active_idx = np.where(w > thresh)[0]

        for k_int in active_idx:
            k = int(k_int)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                # sg[j] gets w[k] * a[k-j] for j in [j_start, j_end]
                # a[k-j] for j=j_start..j_end gives a[k-j_start], a[k-j_start-1], ...
                # = a[k-j_end : k-j_start+1][::-1]
                sg[j_start:j_end + 1] += w[k] * a[k - j_end:k - j_start + 1][::-1]

        # Multiplicative update with centering
        sg_centered = sg - sg.mean()
        log_factor = -eta * sg_centered
        log_factor -= log_factor.max()
        a = a * np.exp(log_factor)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum < 1e-12:
            a = np.ones(n) / n
        else:
            a /= a_sum

        # Evaluate
        conv = fast_convolve(a)
        current_obj = 2.0 * n * conv.max()

        if current_obj < best_obj - 1e-9:
            best_obj = current_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience >= max_patience:
            patience = 0
            eta *= 0.6
            if eta < 0.005:
                # Change strategy: sharpen the focus
                current_t *= 2.0
                eta = 0.3
                max_patience = 40
            else:
                current_t = np.clip(current_t * 1.2, 5.0, 200.0)

    return best_a, best_obj
