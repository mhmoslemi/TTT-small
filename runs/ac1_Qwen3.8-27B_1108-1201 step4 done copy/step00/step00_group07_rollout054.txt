            f += amp * np.exp(-0.5 * ((j - center) / width) ** 2)
        f = np.maximum(f, 0.1)
    else:
        f = np.ones(n)
    f = np.maximum(f, 1e-6)
    return f


def _run_optimization(f_init, n, config_deadline, rng, T_init=5.0, lr_init=0.05):
    """Run log-space gradient optimization for one configuration."""
    # Work in log-space: g = log(f)
    g = np.log(f_init)
    
    N_fft = 1 << (2 * n - 1).bit_length()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    lr = lr_init
    T = T_init
    t = 0
    
    best_local = f_init.copy()
    best_local_score = _compute_objective_fast(f_init)
    
    f = f_init.copy()
    
    while time.monotonic() < config_deadline:
        # Compute gradient in log-space
        # dL/dg[j] = (dL/df[j]) * f[j]
        # dL/df[j] ≈ 2 * sum_k w[k] * f[k-j] = 2 * (w cross-correlate f)[j]
        
        b = fftconvolve(f, f, mode='full')
        
        # Soft-max weights with numerical stability
        Tb = T * (b - b.max())
        exp_Tb = np.exp(Tb)
        w = exp_Tb / exp_Tb.sum()
        
        # Cross-correlation via FFT
        W = np.fft.rfft(w, N_fft)
        F = np.fft.rfft(f, N_fft)
        cross = np.fft.irfft(W * np.conj(F), N_fft)[:n]
        
        # Gradient in f-space, then convert to g-space
        grad_f = 2.0 * cross
        grad_g = grad_f * f
        
        # Adam update
        t += 1
        m = b1 * m + (1.0 - b1) * grad_g
        v = b2 * v + (1.0 - b2) * grad_g * grad_g
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        g = g - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Convert back to f-space, clip
        f = np.exp(g)
        f = np.clip(f, 0.0, 1000.0)
        
        # Light normalization to prevent drift (keep max(f) in reasonable range)
        fmax = np.max(f)
        if fmax > 100.0 or fmax < 0.01:
            f = f / fmax * 1.0
            g = np.log(np.maximum(f, 1e-30))
        
        # Schedule: anneal temperature, decay learning rate
        if t % 50 == 0:
            T = max(1.0, T * 0.96)
            lr *= 0.995
            # Occasionally reset Adam state to escape local minima
            if t % 500 == 0 and rng.random() < 0.3:
                m = np.zeros(n)
                v = np.zeros(n)
                lr = max(0.01, lr * 1.5)
        
        # Check for improvement
        if t % 25 == 0:
            obj = _compute_objective_fast(f)
            if obj < best_local_score:
                best_local_score = obj
                best_local = f.copy()
    
    return best_local, best_local_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Minimizes 2*n*max(convolve(f,f))/(sum(f))^2 using log-space gradient descent
    with soft-max smoothing, Adam optimizer, and FFT-based convolutions.
    Uses multiple restarts with different initial shapes and temperature schedules.
    """
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    rng = np.random.default_rng(seed)
    
    best_sequence = None
    best_score = np.inf
    
    # Configuration list: (n, shape_type, T_init, lr_init)
    configs = [
        (6733, 'inv_sqrt', 5.0, 0.05),
        (6733, 'inv_sqrt_shift', 8.0, 0.04),
        (6733, 'asym_inv_sqrt', 5.0, 0.05),
        (6733, 'sine_mod', 5.0, 0.04),
        (4096, 'inv_sqrt', 5.0, 0.05),
        (8192, 'inv_sqrt', 5.0, 0.04),
        (6733, 'power_opt', 6.0, 0.05),
        (6733, 'random_smooth', 5.0, 0.05),
    ]
    
    # Time allocation: give more time to earlier configs
    total_configs = len(configs)
    weights = [3, 3, 2, 2, 1.5, 1.5, 2, 2]
    total_weight = sum(weights)
    available_time = deadline - time.monotonic()
    
    for idx, (n, shape, T_init, lr_init) in enumerate(configs):
        if time.monotonic() >= deadline:
            break
        
        # Allocate time proportionally
        time_alloc = available_time * weights[idx] / sum(weights[idx:])
        config_deadline = min(time.monotonic() + time_alloc, deadline)
        
        if config_deadline - time.monotonic() < 30:
            config_deadline = min(time.monotonic() + 30, deadline)
        
        # Try up to 2 random restarts for this config
        n_restarts = 2 if weights[idx] >= 3 else 1
        for r in range(n_restarts):
            if time.monotonic() >= deadline or time.monotonic() >= config_deadline - 10:
                break
            
            f_init = _make_initial(n, shape, rng)
            # Normalize
            f_init = f_init / np.sum(f_init) * np.sqrt(2.0 * n)
            
            # Quick initial check
            init_score = _compute_objective_fast(f_init)
            if np.isfinite(init_score):
                if init_score < best_score:
                    best_score = init_score
                    best_sequence = f_init.copy()
                    # Verify with exact evaluator
                    exact = float(evaluate_sequence(f_init.tolist()))
                    if np.isfinite(exact):
                        best_score = exact
                        best_sequence = f_init.copy()
                        print(f"Config {idx} restart {r} ({n},{shape}): init score = {best_score:.6f}")
            
            # Run optimization
            local_best, local_score = _run_optimization(
                f_init, n, config_deadline, rng,
                T_init=T_init, lr_init=lr_init
            )
            
            if local_score < best_score:
                best_score = local_score
                best_sequence = local_best.copy()
                # Verify
                exact = float(evaluate_sequence(best_sequence.tolist()))
                if np.isfinite(exact):
                    best_score = exact
                    print(f"Config {idx} restart {r} ({n},{shape}): improved to {best_score:.6f}")
        
        print(f"After config {idx}: best = {best_score:.6f}, time left = {deadline - time.monotonic():.1f}s")
    
    # Final refinement: take the best sequence and do a short local search
    if best_sequence is not None and time.monotonic() < deadline - 30:
        n = len(best_sequence)
        f = best_sequence.copy()
        N_fft = 1 << (2 * n - 1).bit_length()
        m = np.zeros(n)
        v = np.zeros(n)
        g = np.log(np.maximum(f, 1e-30))
        lr = 0.01
        T = 20.0  # High temperature for final sharpening
        b1, b2, eps = 0.9, 0.999, 1e-8
        t = 0
        refine_deadline = min(deadline, time.monotonic() + 60)
        
        while time.monotonic() < refine_deadline:
            f = np.exp(g)
            b = fftconvolve(f, f, mode='full')
            Tb = T * (b - b.max())
            exp_Tb = np.exp(Tb)
            w = exp_Tb / exp_Tb.sum()
            W = np.fft.rfft(w, N_fft)
            F = np.fft.rfft(f, N_fft)
            cross = np.fft.irfft(W * np.conj(F), N_fft)[:n]
            grad_g = 2.0 * cross * f
            
            t += 1
            m = b1 * m + (1 - b1) * grad_g
            v = b2 * v + (1 - b2) * grad_g * grad_g
            m_hat = m / (1 - b1 ** t)
            v_hat = v / (1 - b2 ** t)
            g = g - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            if t % 20 == 0:
                f_check = np.clip(np.exp(g), 0.0, 1000.0)
                score = _compute_objective_fast(f_check)
                if score < best_score:
                    exact = float(evaluate_sequence(f_check.tolist()))
                    if np.isfinite(exact) and exact < best_score:
                        best_score = exact
                        best_sequence = f_check.copy()
                        print(f"Refinement t={t}: {best_score:.6f}")
    
    # Final verification
    if best_sequence is not None:
        final = float(evaluate_sequence(best_sequence.tolist()))
        print(f"FINAL verified score: {final:.6f}")
        if np.isfinite(final):
            best_score = final
        return [float(x) for x in best_sequence]
    
    # Fallback
    return [1.0]
