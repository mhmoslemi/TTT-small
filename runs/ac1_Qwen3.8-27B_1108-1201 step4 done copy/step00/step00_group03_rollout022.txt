    # b is the autocorrelation array (length 2n-1)

def _autocorr(f):
    return np.convolve(f, f)

    n = len(f)

    w = np.exp(beta * m)
    s = w.sum()
    if s < 1e-300:

        w[b.argmax()] = 1.0
