```
def _smooth_log_noise(n, rng, sigma, lowpass):
    N2 = n//2+1
    k = np.arange(N2, dtype=np.float64)
    decay = np.exp(-0.5*(k/max(1.0, lowpass))**2)
    re = rng.normal(0.0, decay)
    im = rng.normal(0.0, decay)
    # enforce Hermitian by irfft
    spec = re + 1j*im
    spec[0]=0.0
    if n%2==0:
        spec[-1]=re[-1]
    noise = np.fft.irfft(spec, n)
    noise -= noise.mean()
    # optionally std normalize
    std=noise.std()
    if std>1e-12: noise = sigma*noise/std
    else: noise = np.zeros(n)
    return np.exp(noise)
```

```python
"""
Search for nonnegative step-function heights minimizing
    2n * max(np.convolve(a, a)) / (sum(a))**2.

Strategy:
- Start from diverse asymmetric analytic profiles (especially one-sided
  power-law profiles, which avoid the symmetric lower bound of 2.
- Generate many randomized smooth/multiplicative variants.
- Refine using accelerated gradient-like multiplicative updates derived
  from a softmax-weighted peak subgradient of the convolution.
- Use FFT for fast exact convolution and gradient computation.
- Keep a small elite top list and periodically restart from it.
- Finalize by exact-evaluated top candidates with the provided
  evaluate_sequence function when available.
"""

import numpy as np
import time
from operator import itemgetter

_TOP_LIMIT = 16


def _get_base_sequence():
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        return None
    if arr.ndim != 1:
        arr = arr.ravel()
    if arr.size == 0:
        return None
    return np.clip(arr, 0.0, 1000.0)


def _next_pow2(m):
    if m <= 1:
        return 1
    s = 1
    while s < m:
        s <<= 1
    return s


def _normalize_to_target(a, target, cap=999.0):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return a.copy()
    if not np.all(np.isfinite(a)):
        a = np.nan_to_num(a, nan=0.0, posinf=cap, neginf=0.0)
    a = np.maximum(a, 0.0)

    for _ in range(8):
        s = float(a.sum())
        if s <= 1e-300 or not np.isfinite(s):
            return np.full(a.size, max(target / a.size, 1e-300), dtype=np.float64)
        a = a * (target / s)
        if cap is None or cap <= 0:
            break
        m = float(a.max())
        if m <= cap:
            break
        over = np.maximum(a - cap, 0.0)
        total_over = float(over.sum())
        a = np.minimum(a, cap)
        mask = a < cap
        if not np.any(mask):
            break
        mask_sum = float(a[mask].sum())
        if mask_sum <= 1e-300:
            break
        a[mask] += total_over * (a[mask] / mask_sum)

    return a


def _score_and_C(a, fsize):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    s = float(a.sum())
    if s <= 1e-8 or not np.isfinite(s):
        return np.inf, np.zeros(2 * n - 1, dtype=np.float64)
    fa = np.fft.rfft(a, fsize)
    C = np.fft.irfft(fa * fa, fsize)[:2 * n - 1]
    if C.size == 0:
        return np.inf, C
    mx = float(np.max(C))
    if not np.isfinite(mx):
        return np.inf, C
    if mx < 0.0:
        mx = 0.0
    return float(2.0 * n * mx / (s * s)), C


def _soft_weights(C, tau_frac, min_weight=1e-8):
    C = np.asarray(C, dtype=np.float64)
    m = C.size
    if m == 0:
        return np.array([], dtype=np.float64)
    C = np.maximum(C, 0.0)
    cmax = float(np.max(C))
    if not np.isfinite(cmax) or cmax <= 1e-300:
        return np.full(m, 1.0 / m, dtype=np.float64)
    tau = max(float(tau_frac) * cmax, 1e-12)
    e = np.exp((C - cmax) / tau)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(m, 1.0 / m, dtype=np.float64)
    w = e / s
    if min_weight is not None and min_weight > 0.0:
        mw = float(min_weight)
        if 0.0 < mw < 1.0:
            w = (1.0 - mw) * w + (mw / m)
            s2 = float(w.sum())
            if s2 > 0.0:
                w = w / s2
    return w


def _raw_gradient(a, weights, gsize):
    a = np.asarray(a, dtype=np.float64)
    weights = np.asarray(weights, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    ar = a[::-1]
    W = np.fft.rfft(weights, gsize)
    Ar = np.fft.rfft(ar, gsize)
    conv = np.fft.irfft(W * Ar, gsize)
    return np.ascontiguousarray(conv[n - 1:2 * n - 1])


def _step_linear(a, T, target, eta):
    a = np.asarray(a, dtype=np.float64)
    T = np.asarray(T, dtype=np.float64)
    meanT = float(np.mean(T))
    if meanT <= 0.0 or not np.isfinite(meanT):
        return a.copy()
    arg = -float(eta) * (T - meanT) / meanT
    arg = np.clip(arg, -5.0, 5.0)
    return _normalize_to_target(a * np.exp(arg), target)


def _step_log(a, T, target, eta):
    a = np.asarray(a, dtype=np.float64)
    T = np.asarray(T, dtype=np.float64)
    meanT = float(np.mean(T))
    if meanT <= 0.0 or not np.isfinite(meanT):
        return a.copy()
    floorT = max(1e-6 * meanT, 1e-300)
    arg = float(eta) * np.log(meanT / np.maximum(T, floorT))
    arg = np.clip(arg, -5.0, 5.0)
    return _normalize_to_target(a * np.exp(arg), target)


def _step_additive(a, T, target, eta):
    a = np.asarray(a, dtype=np.float64)
    T = np.asarray(T, dtype=np.float64)
    meanT = float(np.mean(T))
    if meanT <= 0.0 or not np.isfinite(meanT):
        return a.copy()
    delta = T - meanT
    lim = 10.0 * meanT
    delta = np.clip(delta, -lim, lim)
    anew = a - float(eta) * delta
    anew = np.maximum(anew, 0.0)
    return _normalize_to_target(anew, target)


def _random_block(a, rng, sigma):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return a.copy()
    minw = max(1, n // 500)
    maxw = max(minw, n // 8)
    w = int(rng.integers(minw, maxw + 1))
    if w > n:
        w = n
    st = int(rng.integers(0, n - w + 1))
    cand = a.copy()
    cand[st:st + w] *= float(np.exp(rng.normal(0.0, float(sigma))))
    return _normalize_to_target(cand, float(n))


def _smooth_log_noise(n, rng, sigma, lowpass):
    if n <= 0:
        return np.ones(0, dtype=np.float64)
    N2 = n // 2 + 1
    k = np.arange(N2, dtype=np.float64)
    lp = max(1.0, float(lowpass))
    decay = np.exp(-0.5 * (k / lp) ** 2)
    re = rng.normal(0.0, decay)
    im = rng.normal(0.0, decay)
    spec = re + 1j * im
    spec[0] = 0.0
    if n % 2 == 0:
        spec[-1] = re[-1]
    noise = np.fft.irfft(spec, n)
    noise -= float(noise.mean())
    std = float(noise.std())
    if std > 1e-12:
        noise = float(sigma) * noise / std
    else:
        noise = np.zeros(n, dtype=np.float64)
    return np.exp(noise)


def _random_style_candidate(n, target, rng, style):
    i = np.arange(n, dtype=np.float64)
    t = i / max(1.0, float(n - 1))
    shifts = (0.5, 1.0, 2.0, 4.0)
    style = int(style) % 6

    if style == 0:
        alpha = float(rng.uniform(0.35, 0.60))
        shift = float(shifts[int(rng.integers(0, 4))])
        a = (i + shift) ** (-alpha)
    elif style == 1:
        alpha = float(rng.uniform(0.35, 0.55))
        a = (i + 1.0) ** (-alpha)
        rw = float(rng.uniform(0.0, 0.25))
        a = a + rw * ((n - i + 1.0) ** (-alpha))
    elif style == 2:
        gamma = float(rng.uniform(0.05, 3.0))
        a = np.exp(-gamma * t)
        if rng.random() < 0.35:
            a = a + float(rng.uniform(0.0, 0.25)) * np.exp(-gamma * (1.0 - t))
    elif style == 3:
        alpha = float(rng.uniform(0.45, 0.55))
        base = (i + 1.0) ** (-alpha)
        r = int(rng.integers(1, 10))
        amp = float(rng.uniform(0.0, 0.85))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        fac = 1.0 + amp * np.cos(2.0 * np.pi * r * i / float(n) + phase)
        a = base * np.maximum(fac, 0.0)
    elif style == 4:
        base = (i + 1.0) ** (-0.5)
        c = float(rng.uniform(0.1, 4.0))
        p = float(rng.uniform(0.5, 4.0))
        a = base * (1.0 + c * np.power(t, p))
    else:
        base = (i + 1.0) ** (-0.5)
        sigma = float(rng.uniform(0.04, 0.28))
        a = base * np.exp(rng.normal(0.0, sigma, n))

    return _normalize_to_target(a, target)


def _consider_candidate(score, arr, top, limit=_TOP_LIMIT):
    if not np.isfinite(score):
        return
    arr = np.asarray(arr, dtype=np.float64)
    if arr.size == 0 or not np.all(np.isfinite(arr)):
        return
    if len(top) < limit:
        top.append([float(score), arr.copy()])
    else:
        worst = 0
        for idx in range(len(top)):
            if top[idx][0] > top[worst][0]:
                worst = idx
        if score < top[worst][0]:
            top[worst] = [float(score), arr.copy()]


def _sorted_top(top):
    if not top:
        return []
    return sorted(top, key=itemgetter(0))


def _gradient_restart(initial, best_score, best_arr, top, fsize, gsize, target,

    sc, C = _score_and_C(cur, fsize)

        return best_score, best_arr

        best_score = float(sc)

    eta = 0.35

    it = 0
    start = time.monotonic()

        elapsed = max(0.0, time.monotonic() - start)
        progress = min(1.0, elapsed / total_time)
        tau_frac = 0.25 * (1.0 - progress) + 0.008 * progress

            best_trial = cand

        cand = _step_log(cur, T, target, max(0.05, 0.6 * eta))

            sc = best_trial_score
            C = best_trial_C
            eta = min(1.0, eta * 1.08)
            no_improve = 0
        else:
            if (best_trial is not None and
                    best_trial_score < sc + 1e-5 * max(1.0, sc) and

                cur = best_trial
                sc = best_trial_score
                C = best_trial_C
                eta *= 0.85
                no_improve += 1
            else:
                eta = max(0.02, eta * 0.75)
                no_improve += 1
                if no_improve > 25:
                    try:
                        kick = _smooth_log_noise(n, rng, 0.03, max(5, n // 20))
                    except Exception:
                        kick = np.exp(rng.normal(0.0, 0.03, n))
                    cur = _normalize_to_target(cur * kick, target)
                    sc, C = _score_and_C(cur, fsize)
                    no_improve = 5
                if no_improve > 110:
                    break

        if it % 80 == 0 and best_arr is not None and np.isfinite(best_score):
            if sc > best_score + 0.03 * max(1.0, best_score):
                try:
                    kick = _smooth_log_noise(n, rng, 0.05, max(5, n // 20))
                except Exception:
                    kick = np.exp(rng.normal(0.0, 0.05, n))

    sc, C = _score_and_C(cur, fsize)

            last_printed = best_score

        restart += 1
        if time.monotonic() >= search_deadline - 1.0:
            break

    print(f"after refinement best_fft={best_score:.6f}, elapsed="
          f"{time.monotonic() - start_time:.2f}")

    final_candidates = _sorted_top(top)
    if best_arr is not None and np.isfinite(best_score):
        final_candidates = final_candidates + [[float(best_score), np.asarray(best_arr, dtype=np.float64).copy()]]

    best_exact = np.inf
    best_seq = None

    seen = 0
    for item in final_candidates:
        if seen >= 10:
            break
        if time.monotonic() > final_hard_deadline - 1.0:
            break
        arr = np.asarray(item[1], dtype=np.float64)
        arr = _normalize_to_target(arr, target)
        try:
            ex = evaluate_sequence([float(x) for x in arr])
        except Exception:
            ex = np.inf
        if np.isfinite(ex) and ex < best_exact:
            best_exact = float(ex)
            best_seq = arr
        seen += 1

    if best_arr is not None and (best_seq is None or not np.isfinite(best_exact)):
        try:
            ex = evaluate_sequence([float(x) for x in best_arr])
            if np.isfinite(ex) and ex < best_exact:
                best_exact = float(ex)
                best_seq = best_arr
        except Exception:
            pass

    if best_seq is None:
        best_seq = _normalize_to_target(np.ones(n, dtype=np.float64), target)

    best_seq = _normalize_to_target(best_seq, target)
    print(f"final_fft_best={best_score:.6f}, final_exact_best={best_exact:.6f}")
    return [float(x) for x in best_seq]
```
