```python
def compute_gradient(a, p):
    """Compute grad[i] = sum_j a[j] * p[i+j] using FFT."""
    n = len(a)
    aRev = a[::-1]
    # Convolve aRev with p (length 2n-1)
    # Result has length n + (2n-1) - 1 = 3n-1
    # We want elements at positions n-1 to 2n-2
    # Use FFT
    fft_size = 1
    while fft_size < 3*n - 1:
        fft_size *= 2
    A = np.fft.rfft(aRev, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv_result = np.fft.irfft(A * P, fft_size)
    return conv_result[n-1:2*n-1]
```

```python
    """Generate the non-negative sequence with flat linear autocorrelation."""
    a = np.zeros(n)
    a[0] = 1.0

            s += a[i] * a[k-i]

    return a
```
