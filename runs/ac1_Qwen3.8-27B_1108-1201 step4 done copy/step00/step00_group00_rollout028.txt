def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:

    theta = (cssv[rho] - 1) / (rho + 1.0)

import time

def compute_full_convolution(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Use next fast FFT length
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    fa = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(fa * fa, fft_size)[:size]
    return conv

def project_simplex(v):

    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)

    rho = np.argmax(cond)  # First True

    theta = (cssv[rho] - 1) / (rho + 1.0)

def full_convolution_fft(a):

    n = len(a)
    size = 2 * n - 1

Actually, let me think about this more carefully. The continuous analog of our problem is:
minimize max_t (f*f)(t) subject to f >= 0, integral f = 1, support in [0,1].

The "flat spectrum" condition means |F(ξ)|^2 = constant, which means |F(ξ)| = c. But F is the Fourier transform of a non-negative function, so |F(ξ)| >= F(0) * [some bound]... actually, for a non-negative f, F(0) = integral f = 1, and |F(ξ)| <= F(0) = 1 (by the triangle inequality). So |F(ξ)| <= 1 for all ξ, with equality at ξ = 0.

For |F(ξ)|^2 to be constant (= c), we'd need c = 1 at ξ = 0, so c = 1, meaning |F(ξ)| = 1 for all ξ. But this means F(ξ) = e^{iφ(ξ)} for some phase function, and f = IfFT(e^{iφ}) which is generally not non-negative.

So the continuous optimum is not achievable. The best we can do is make |F(ξ)| as close to 1 as possible for ξ ≠ 0.

In the discrete case, this translates to: make |FFT(a)[k]| as close to |FFT(a)[0]| = sum(a) = 1 as possible for k ≠ 0.

This is the "flat-top" or "CAZAC" condition, but restricted to non-negative sequences.

For non-negative sequences, the DFT at 0 is always the largest (by the triangle inequality). The question is how "flat" the rest can be.

This gives me another optimization approach: minimize the variance of |FFT(a)|^2 (excluding DC). This promotes flatness of the spectrum and thus flatness of the convolution.

import numpy as np
import time

def full_convolution_fft(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    fa = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(fa * fa, fft_size)[:size]
    return np.maximum(conv, 0.0)

def project_simplex(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][0]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def gradient_conv_at_k(a, k):
    """Gradient of the k-th element of the full self-convolution with respect to a."""
    n = len(a)
    grad = np.zeros(n)
    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    if j_start <= j_end:
        j = np.arange(j_start, j_end + 1)
        i = k - j
        grad[i] = 2.0 * a[j]
    return grad

def optimize_from_init(a_init, n, deadline, rng, top_k=3, lr_init=1.0):
    """Run gradient descent from a given initialization."""
    a = np.maximum(a_init, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a_init, float('inf')
    a /= s
    
    best_a = a.copy()
    best_score = 2.0 * n * full_convolution_fft(a).max()
    
    lr = lr_init
    no_improve = 0
    patience = 20000
    
    while time.monotonic() < deadline:
        conv = full_convolution_fft(a)
        max_val = conv.max()
        score = 2.0 * n * max_val
        
        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > patience:
            # Restart with perturbation
            a = best_a + rng.normal(0, 0.01, n)
            a = project_simplex(a)
            no_improve = 0
            lr = lr_init * 0.5
            continue
        
        # Top-k gradient
        k = min(top_k, len(conv))
        top_idx = np.argsort(conv)[::-1][:k]
        
        grad = np.zeros(n)
        for idx in top_idx:
            grad += gradient_conv_at_k(a, int(idx))
        grad /= float(k)

            continue
        
        a = a_new
        
        # Learning rate decay
        if abs(lr - lr * 0.9995) > 1e-15:
            lr = max(1e-6, lr * 0.9995)
    
    return best_a, best_score
