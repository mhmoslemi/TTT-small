def _next_pow2(x):
    """Return next power of 2 >= x."""
    p = 1
    while p < x:
        p *= 2

                print(f"    n={n} it={it} obj={obj:.6f} best={best_obj:.6f} t={time.monotonic()-g_start:.0f}s")

    # Final exact evaluation
    score = evaluate_sequence(best_a.tolist())
    if np.isfinite(score) and score < best_obj:
        best_obj = score

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Projected subgradient descent on min max(autocorr(a)) s.t. sum(a)=1, a>=0.
    Scale-invariant reformulation of min 2n*max(conv(a,a))/(sum(a))^2.
    Multiple restarts with varying n and initializations.
    """
    t0 = time.monotonic()
    deadline = t0 + budget_s - 20.0
    rng = np.random.default_rng(seed)

    best_a = None
    best_score = np.inf

    def record(a, score):
        nonlocal best_a, best_score
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"  ** NEW BEST: {score:.8f} (n={len(a)}, t={time.monotonic()-t0:.1f}s)")

    # Phase 1: height_sequence_1
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        if h.ndim == 1 and h.size > 10 and np.all(np.isfinite(h)):
            h = np.clip(h, 0.0, 1000.0)
            if h.sum() > 1e-10:
                n = len(h)
                print(f"[Phase 1] height_sequence_1, n={n}")
                d1 = min(deadline, t0 + 400.0)
                a1, s1 = _optimize(h, d1, t0)
                s1 = evaluate_sequence(a1.tolist())
                record(a1, s1)
                print(f"  [Phase 1] done: {s1:.8f}")
    except Exception as ex:
        print(f"  [Phase 1] error: {ex}")

    # Phase 2: uniform + perturbation, various n
    n_list = [4096, 6144, 8192, 10240]
    t_each = max(150.0, (deadline - time.monotonic()) / len(n_list) * 0.9)
    for n in n_list:
        if time.monotonic() > deadline - 60:
            break
        a_init = np.ones(n) + rng.normal(0, 0.08, n)
        a_init = np.clip(a_init, 0, None)
        print(f"[Phase 2] uniform+perturb, n={n}, budget={t_each:.0f}s")
        a2, s2 = _optimize(a_init, min(deadline - 30, time.monotonic() + t_each), t0)
        s2 = evaluate_sequence(a2.tolist())
        record(a2, s2)
        print(f"  [Phase 2] n={n} done: {s2:.8f}")

    # Phase 3: random init
    for n in [4096, 8192]:
        if time.monotonic() > deadline - 60:
            break
        a_init = rng.random(n) + 0.01
        a_init /= a_init.sum()
        print(f"[Phase 3] random, n={n}")
        a3, s3 = _optimize(a_init, min(deadline - 30, time.monotonic() + 200.0), t0)
        s3 = evaluate_sequence(a3.tolist())
        record(a3, s3)
        print(f"  [Phase 3] n={n} done: {s3:.8f}")

    # Phase 4: use best found as seed for further refinement
    if best_a is not None and time.monotonic() < deadline - 60:
        n = len(best_a)
        print(f"[Phase 4] refine best, n={n}, current best={best_score:.8f}")
        a4, s4 = _optimize(best_a, deadline - 10, t0)
        s4 = evaluate_sequence(a4.tolist())
        record(a4, s4)
        print(f"  [Phase 4] done: {s4:.8f}")

    # Fallback
    if best_a is None or not np.isfinite(best_score):
        print("  Fallback to uniform sequence")
        best_a = np.ones(4096)
        best_score = 2.0

    result = [float(x) for x in best_a]
    print(f"\nFinal: score={best_score:.8f}, n={len(result)}")
    return result
