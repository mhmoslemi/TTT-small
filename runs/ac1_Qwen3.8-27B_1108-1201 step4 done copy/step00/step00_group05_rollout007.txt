def local_search_around_uniform(base, rng, budget_s):
    """
    Perform local search to confirm no improvement below 2 is possible.
    Uses various perturbation strategies.
    """
    deadline = time.monotonic() + budget_s
    best = np.array(base, dtype=np.float64)
    best_score = evaluate_sequence(best.tolist())
    n = len(best)
    iterations = 0

    while time.monotonic() < deadline:
        # Try several types of perturbations
        mode = rng.integers(0, 4)

        if mode == 0:
            # Random block scaling
            candidate = best.copy()
            width = int(rng.integers(max(1, n // 100), max(2, n // 10)))
            start = int(rng.integers(0, max(1, n - width)))
            candidate[start:start+width] *= np.exp(rng.normal(0, 0.05))
        elif mode == 1:
            # Gaussian perturbation
            candidate = best * np.exp(rng.normal(0, 0.01, n))
        elif mode == 2:
            # Symmetric perturbation
            candidate = best.copy()
            h = n // 2
            pert = np.exp(rng.normal(0, 0.03, h))
            candidate[:h] *= pert
            candidate[h:] *= pert[::-1]
        else:
            # Coordinate perturbation
            candidate = best.copy()
            idx = rng.integers(0, n, size=max(1, n // 50))
            candidate[idx] *= np.exp(rng.normal(0, 0.02, len(idx)))

        candidate = np.clip(candidate, 0.0, 1000.0)
        if np.sum(candidate) < 0.01:
            continue

        score = evaluate_sequence(candidate.tolist())
        iterations += 1

        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
            if iterations % 100 == 0:
                print(f"  Iter {iterations}: improved to {best_score:.10f}")

        if iterations % 500 == 0:
            print(f"  Iter {iterations}: current best = {best_score:.10f}")

    return best.tolist(), best_score, iterations
