projected gradient descent with power-norm (l_p) smoothing, FFT-based

import numpy as np

def _project_simplex(v):
    """Project vector v onto {x >= 0, sum(x) = 1} using sorting-based method."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.where(cond)[0][-1]
    rho_val = idx + 1
    theta = (cv[idx] - 1.0) / rho_val
    proj = np.maximum(v - theta, 0.0)
    s = proj.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return proj / s


def _fft_convolve_same(a):
    """Fast self-convolution a*a using FFT. Returns full linear convolution."""
    n = len(a)
    out_len = 2 * n - 1
    N = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:out_len]

    Returns gradient of length n (unnormalized).

        return np.zeros(len(f))
    # Power weights: w_k = (b_k / b_max)^(p-1)
    safe_b = np.maximum(b, 0.0)
    w = (safe_b / b_max) ** (p - 1.0)
    w_sum = w.sum()
