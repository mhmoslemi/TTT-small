import numpy as np
import time
from scipy.optimize import linprog

def compute_convolution(f):
    """Compute full convolution of f with itself using FFT."""
    n = len(f)
    size = 2 * n
    F = np.fft.rfft(f, size)
    C = np.fft.irfft(F * F, size)
    return C[:2*n-1]

def subgradient_step(f, eta, top_m=5):
    """One subgradient descent step on the peak of f*f."""
    n = len(f)
    c = compute_convolution(f)
    M = c.max()
    if M <= 0:
        return f.copy()
    
    # Get top-m peaks
    top_indices = np.argsort(c)[::-1][:top_m]
    
    # Compute responsibility
    pressure = np.zeros(n)
    for k in top_indices:
        weight = c[k] / M
        # f_j contributes f_j * f_{k-j} to c_k
        # So pressure on f_j is weight * f_{k-j}
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                pressure[j] += weight * f[idx]
    
    # Update
    f_new = f - eta * pressure
    f_new = np.clip(f_new, 0, None)
    
    # Normalize (optional, since score is scale-invariant)
    s = f_new.sum()
    if s > 1e-10:
        f_new = f_new / s
    
    return f_new

def lp_direction(f, M):

    n = len(f)
    # Build constraint matrix A: (2n-1) x n
    # A[k, j] = f[j] if 0 <= k-j <= n-1, else 0
    # Actually: A[k, i] = f[k-i] if 0 <= k-i <= n-1 and 0 <= i <= n-1
    
    # This is a big matrix. For n=6121, it's 12241 x 6121 = 75M entries.
    # Too large for dense. Use sparse.
    from scipy.sparse import lil_matrix
    
    A = lil_matrix((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        i_min = max(0, k - n + 1)
        i_max = min(n-1, k)
        for i in range(i_min, i_max + 1):
            A[k, i] = f[k - i]
    
    A = A.tocsr()
    
    # Objective: maximize sum(b) => minimize -sum(b)
    c_obj = -np.ones(n)
    
    # Constraints: A @ b <= M * ones
    b_rhs = np.full(2*n-1, M)
    
    # Bounds: b >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A, b_ub=b_rhs, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    return None
