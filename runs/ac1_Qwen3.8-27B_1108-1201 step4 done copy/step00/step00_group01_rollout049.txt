```python
import numpy as np
import time

def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    # Use the water-filling algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - s))[0]
    if len(rho) == 0:
        return np.zeros(n)
    theta = (cssv[rho[-1]] - s) / (rho[-1] + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_step(x, lr):
    """One subgradient step on the normalized sequence."""
    n = len(x)
    conv = np.convolve(x, x)
    max_val = conv.max()
    argmax = np.where(conv >= max_val * (1 - 1e-12))[0]
    
    # Subgradient: average of the "reflected" x over argmax positions
    grad = np.zeros(n)
    for k in argmax:
        # r_k[i] = x[k-i] if 0 <= k-i <= n-1 else 0
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += x[j]
    grad /= len(argmax)
    
    # Step
    x_new = x - lr * grad
    x_new = project_simplex(x_new, 1.0)
    return x_new
```

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    theta = (cssv[rho[-1]] - 1.0) / (rho[-1] + 1.0)
    return np.maximum(v - theta, 0.0)
