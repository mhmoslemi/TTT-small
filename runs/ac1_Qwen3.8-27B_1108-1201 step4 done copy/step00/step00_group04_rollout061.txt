import numpy as np
import time
from scipy.optimize import linprog

def compute_objective(a):
    n = len(a)
    c = np.convolve(a, a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    return 2 * n * np.max(c) / S**2

def solve_lp(a, M):
    """Solve: max sum(g) s.t. convolve(a, g) <= M, g >= 0"""
    n = len(a)
    # Variables: g_0, ..., g_{n-1} (n variables)
    # Constraints: for each k = 0, ..., 2n-2:
    #   sum_i a[i] * g[k-i] <= M
    #   (where sum is over valid i)
    
    # Objective: maximize sum(g) → minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Inequality constraints: A_ub @ g <= b_ub
    # For each k, the constraint is:
    # sum_{i: 0<=i<n, 0<=k-i<n} a[i] * g[k-i] <= M
    
    # Build A_ub matrix (2n-1 rows, n columns)
    # A_ub[k, j] = coefficient of g[j] in constraint k
    # g[j] appears in constraint k when j = k-i, i.e., i = k-j
    # So A_ub[k, j] = a[k-j] if 0 <= k-j < n, else 0
    
    # Actually, for constraint k: sum_i a[i]*g[k-i] where 0<=i<n and 0<=k-i<n
    # Let j = k-i, so i = k-j. Constraint: sum_j a[k-j]*g[j] where 0<=k-j<n and 0<=j<n
    # So A_ub[k, j] = a[k-j] if 0 <= k-j < n, else 0
    
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A_ub[k, j] = a[i]
    
    b_ub = np.full(2*n-1, M)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    else:

def optimize_sequence(a, n_iterations=10000, beta=100, lr=0.01):
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    a /= a.sum()
    
    best = a.copy()
    best_obj = np.max(np.convolve(a, a))
    
    for it in range(n_iterations):
        c = np.convolve(a, a)
        # Log-sum-exp
        c_max = np.max(c)
        exp_c = np.exp(beta * (c - c_max))
        softmax = exp_c / np.sum(exp_c)
        
        # Gradient

        # Update

        a_new = np.maximum(a_new, 0)
        a_new /= a_new.sum()
        
        # Check improvement
        obj = np.max(np.convolve(a_new, a_new))
        if obj < best_obj:
            best = a_new.copy()
            best_obj = obj
        
        a = a_new
        
        # Adaptive learning rate
        if it % 100 == 0:
            lr *= 0.99
    
    return best, best_obj

def _logsumexp_obj_grad(x, beta, n):
    """Compute log-sum-exp smoothed objective and its gradient w.r.t. x (log-params)."""
    # a = softmax(x)
    x_shift = x - x.max()
    a = np.exp(x_shift)
    a_sum = a.sum()
    if a_sum == 0 or not np.isfinite(a_sum):
        return 1e30, np.zeros(n)
    a = a / a_sum

    # Convolution
    c = np.convolve(a, a)

    # Log-sum-exp of beta * c
    c_max = c.max()
    shifted = beta * (c - c_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -700, 700)
    exp_shifted = np.exp(shifted)
    exp_sum = exp_shifted.sum()
    if exp_sum == 0 or not np.isfinite(exp_sum):
        return 1e30, np.zeros(n)

    obj = c.max() + np.log(exp_sum) / beta
    softmax_c = exp_shifted / exp_sum

    # Gradient w.r.t. a: grad_a[j] = 2 * sum_k softmax_c[k] * a[k-j]
    # Using convolution: sum_i a[i] * softmax_c[i+j] = convolve(a[::-1], softmax_c)[n-1+j]
    conv_grad = np.convolve(a[::-1], softmax_c)
    grad_a = 2.0 * conv_grad[n - 1:2 * n - 1]

    # Gradient w.r.t. x: grad_x[j] = a[j] * (grad_a[j] - sum_k grad_a[k]*a[k])
    dot = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot)

    return obj, grad_x


def _run_lbfgs(n, x0, betas, max_iter=500):
    """Run L-BFGS-B with a schedule of beta values."""
    current_x = x0.copy()
    best_a = None
    best_true_obj = np.inf

    for beta in betas:
        def fun(x):
            return _logsumexp_obj_grad(x, beta, n)[0]

        def jac(x):
            return _logsumexp_obj_grad(x, beta, n)[1]

        try:
            res = minimize(
                fun, current_x, jac=jac, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-8}
            )
            current_x = res.x
        except Exception:
            pass

        # Recover a
        x_shift = current_x - current_x.max()
        a = np.exp(x_shift)
        a_sum = a.sum()
        a = a / a_sum

        # True objective
        c = np.convolve(a, a)
        true_obj = c.max()
        if true_obj < best_true_obj:
            best_true_obj = true_obj
            best_a = a.copy()

    return best_a, best_true_obj


def _geometric_init(n, r, rng):
    """Create a geometric initial sequence."""
    a = np.array([r ** i for i in range(n)])
    a *= np.exp(rng.normal(0, 0.05, n))
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    return a
