import numpy as np
import time

def compute_score(a):
    """Compute J = 2n*max(conv(a))/(sum(a))^2 without calling evaluate_sequence."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    conv = np.convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)

def _gradient_peaks(a, num_peaks=80, smooth_width=5):
    """Compute gradient direction to reduce convolution peaks."""
    n = len(a)
    conv = fftconvolve(a, a)
    
    # Smooth the excess to get a smoother gradient
    if smooth_width > 1:
        kernel = np.ones(2 * smooth_width + 1) / (2 * smooth_width + 1)
        conv_smooth = np.convolve(conv, kernel, mode='same')
    else:
        conv_smooth = conv
    
    mean_c = conv_smooth.mean()
    excess = np.maximum(conv_smooth - mean_c, 0)
    
    # Get top K peak positions
    K = min(num_peaks, len(conv))
    top_idx = np.argsort(conv_smooth)[-K:]
    weights = np.zeros(len(conv))
    for k in top_idx:
        weights[k] = max(0, conv_smooth[k] - mean_c)
    
    # Gradient: g[i] = sum_k weights[k] * a[k-i] (where valid)
    g = np.zeros(n)
