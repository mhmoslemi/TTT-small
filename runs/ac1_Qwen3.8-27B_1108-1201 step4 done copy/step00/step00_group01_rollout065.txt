def _project_simplex(a, target_sum):
    """Project vector a onto {x: sum(x) = target_sum, x >= 0} via bisection."""
    n = a.size
    if n == 0:
        return a.astype(np.float64)
    lo = float(a.min()) - target_sum - 1.0
    hi = float(a.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.maximum(0.0, a - mid)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    nu = (lo + hi) * 0.5
    return np.maximum(0.0, a - nu)


def _add_grad(g, a, k, w):
    """In-place add w * (grad of b_k wrt a) to g, where b = convolve(a,a,full)."""
    n = a.size
    # Term 1: d/d a_i [sum_j a_j * a_{k-j}] = a_{k-i} for valid 0<=k-i<n
    lo1 = max(0, k - n + 1)
    hi1 = min(n - 1, k)
    if lo1 <= hi1:
        g[lo1:hi1 + 1] += w * a[k - hi1:k - lo1 + 1][::-1]
    # Term 2: a_{i-k} for valid 0<=i-k<n
    lo2 = max(0, k)
    hi2 = min(n - 1, k + n - 1)
    if lo2 <= hi2:
        g[lo2:hi2 + 1] += w * a[lo2 - k:hi2 - k + 1]


def _lse_subgradient(a, b, tau, max_terms=80):

    return tau * g
