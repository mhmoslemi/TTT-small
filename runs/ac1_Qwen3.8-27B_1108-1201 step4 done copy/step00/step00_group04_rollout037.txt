    """Project y onto the simplex {x >= 0, sum(x) = s}."""

"""
Multi-phase optimization for minimizing 2*n*max(convolve(a,a)) / (sum(a))^2.
Phases: (1) diverse initialization, (2) (3) mirror descent with multi-peak
subgradients, (4) LP-based directional refinement, (5) final polish.
"""

import numpy as np
import time


def fast_convolve(a):
    """Full auto-convolution a*a using FFT (faster than np.convolve for large n)."""
    n = len(a)
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:out_len]


def compute_score(a):
    """Compute the objective: 2*n*max(conv) / (sum)^2."""
    n = len(a)
    c = fast_convolve(a)
    max_c = np.max(c)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


def build_conv_matrix(f, top_K):
    """
    Build the top-K constraint rows of the convolution matrix A.
    A[k,j] = f[k-j] if 0<=k-j<n else 0.
    Returns (A_K, indices) where A_K has shape (K, n).
    """
    n = len(f)
    num_constraints = 2 * n - 1
    K = min(top_K, num_constraints)
    A = np.zeros((K, n))
    # For row k: A[k, j] = f[k-j] for j in [max(0,k-n+1), min(k,n-1)]
    # Vectorized: for each k, the valid j range
    for k in range(K):
        j_end = min(k, n - 1)
        j_start = max(0, k - n + 1)
        # A[k, j] = f[k-j] for j=j_start..j_end
        # = f[k-j_start], f[k-j_start-1], ..., f[k-j_end]
        A[k, j_start:j_end + 1] = f[j_end:j_start - 1, -1] if j_start > 0 else f[0:j_end + 1][::-1]
    return A


def build_conv_matrix_all(f):
    """Build full convolution matrix A of shape (2n-1, n)."""
    n = len(f)
    num_constraints = 2 * n - 1
    A = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        j_end = min(k, n - 1)
        j_start = max(0, k - n + 1)
        A[k, j_start:j_end + 1] = f[j_end:j_start - 1, -1] if j_start > 0 else f[0:j_end + 1][::-1]
    return A


def solve_lp_direction(f, M, top_K=100):
    """
    Solve LP: max 1'g s.t. (f*g)_k <= M for top-K peaks, g >= 0.
    Returns g (normalized) or None.
    """
    from scipy.optimize import linprog

    n = len(f)
    A_K = build_conv_matrix(f, top_K)
    c_obj = -np.ones(n)
    b_ub = np.full(top_K, M)

    res = linprog(c_obj, A_ub=A_K, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs', options={'time_limit': 60, 'presolve': True})

    if res.status != 0 or res.x is None:
        return None
    g = res.x
    total = g.sum()
    if total < 1e-10:
        return None
    return g / total


def lp_refinement(a, best_score, deadline, rng):
    """LP-based directional refinement phase."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_local = a.copy()
    best_local_score = best_score
    it = 0
    stuck_count = 0

    while time.monotonic() < deadline - 10:
        t0 = time.monotonic()
        c = fast_convolve(a)
        M = np.max(c)
        # Get top K peak indices
        K = min(200, len(c))
        peak_indices = np.argsort(c)[::-1][:K]

        if len(peak_indices) < 2:
            break

        g = solve_lp_direction(a, M, top_K=len(peak_indices))
        if g is None:
            break
        g = g / g.sum() * a.sum()

        cgg = fast_convolve(g)
        max_gg = np.max(cgg)

        # Find optimal t via bisection
        def loss_at_t(t):
            ct = (1 - t) ** 2 * c + 2 * t * (1 - t) * fast_convolve(a) * 0  # placeholder
            return 0

        # Actually compute properly: f_t*f_t = (1-t)^2 c + 2t(1-t) fg + t^2 gg
        # We need f*g for the cross term
        # f*g is expensive to compute each time. Let's use the LP constraint: (f*g)_k <= M
        # So the worst case upper bound for the cross term is M.
        # For the actual value, let's compute it once.
        # Actually, for the line search we need the actual convolution.
        # Let's compute f*g once (it's the same as g*f for the cross term).
        # f*g: convolution of f and g
        ng = len(g)
        out_len = n + ng - 1
        size = 1 << (out_len - 1).bit_length()
        F = np.fft.rfft(a, size)
        G = np.fft.rfft(g, size)
        fg = np.fft.irfft(F * G, size)[:out_len]

        def max_conv_t(t):
            val = (1 - t) ** 2 * c + 2.0 * t * (1.0 - t) * fg + t ** 2 * cgg
            return np.max(val)

        # Binary search for max t such that max_conv_t(t) < M
        lo, hi = 0.0, 1.0
        if max_conv_t(1.0) >= M:
            for _ in range(30):
                mid = (lo + hi) / 2.0
                if max_conv_t(mid) < M:
                    lo = mid
                else:
                    hi = mid
            t_opt = lo * 0.95
        else:
            t_opt = 1.0

        if t_opt < 1e-4:
            stuck_count += 1
            if stuck_count > 20:
                break
            # Try smaller peaks
            continue

        a_t = (1 - t_opt) * a + t_opt * g
        a_t = np.maximum(a_t, 0)
        s_at = a_t.sum()
        if s_at < 0.01:
            break
        a_t = a_t / s_at

        score_t = compute_score(a_t)
        if score_t < best_local_score - 1e-10:
            best_local_score = score_t
            best_local = a_t.copy()
            stuck_count = 0
        else:
            stuck_count += 1
            if stuck_count > 30:
                break

        a = a_t
        it += 1
        elapsed = time.monotonic() - t0
        if it % 10 == 0:
            print(f"  LP iter {it}: score={best_local_score:.6f}, t_opt={t_opt:.4f}, "
                  f"time/s={elapsed:.2f} ({time.monotonic()-t0:.2f}s/iter)")

    return best_local, best_local_score


def polish(a, best_score, deadline, rng):
    """Final polish with small random perturbations."""
    n = len(a)
    best = a.copy()
    best_sc = best_score
    it = 0
    while time.monotonic() < deadline - 5:
        # Random block perturbation
        width = max(1, n // 50)
        start = rng.integers(0, n - width)
        noise = np.exp(rng.normal(0, 0.05, size=width))
        c = best.copy()
        c[start:start + width] *= noise
        s = c.sum()
        if s < 0.01 or s > 100:
            continue
        c = c / s  # normalize back
        sc = compute_score(c)
        if sc < best_sc:
            best = c.copy()
            best_sc = sc
        it += 1
        if it % 5000 == 0:
            print(f"    Polish iter {it}: best={best_sc:.6f}")
    return best, best_sc


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search for optimal non-negative sequence minimizing
    2*n*max(full_convolve(a,a)) / (sum(a))^2.
    
    Phases:
      1. Diverse initialization (parametric families + parent)
      2. Extended mirror descent (multiplicative weights, multi-peak)
      3. LP-based directional refinement
      4. Final polish
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 10.0  # leave 10s buffer

    # --- Get parent sequence ---
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64)
        parent_raw = np.clip(parent_raw, 0, 1000)
    except (NameError, Exception):
        parent_raw = None

    if parent_raw is not None and parent_raw.size > 0 and np.sum(parent_raw) > 0.01:
        n = parent_raw.size
        parent = parent_raw / parent_raw.sum()
    else:
        n = 2048
        parent = np.ones(n, dtype=np.float64) / n

    print(f"n = {n}, parent score = {compute_score(parent):.6f}")

    # --- Phase 1: Initialization ---
    print("Phase 1: Initialization...")
    t_phase1 = time.monotonic()
    candidates = []

    # Parent and symmetrized
    candidates.append(parent.copy())
    candidates.append((parent + parent[::-1]) / 2.0)

    # Parametric families
    x = np.arange(n, dtype=np.float64)
    xn = x / n
    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * xn))
    rc = np.clip(rc, 0, None)
    if rc.sum() > 0:
        candidates.append(rc / rc.sum())
    # Pure cosine (non-neg part)
    for m in [1, 2, 3, 4, 5, 7, 8, 11, 13, 16, 20, 30, 50]:
        c = 0.5 * (1 + np.cos(2 * np.pi * m * xn))
        c = np.clip(c, 1e-10, None)
        candidates.append(c / c.sum())
    # Two-frequency
    for m1, m2 in [(1, 3), (2, 5), (3, 8), (5, 13), (7, 20)]:
        c = 0.5 * (1 + 0.5 * np.cos(2 * np.pi * m1 * xn) + 0.3 * np.cos(2 * np.pi * m2 * xn))
        c = np.clip(c, 1e-10, None)
        candidates.append(c / c.sum())
    # Chirp
    for alpha in [0.2, 0.4, 0.6, 0.8]:
        c = 1 + alpha * np.cos(2 * np.pi * (x ** 2) / n)
        c = np.clip(c, 1e-10, None)
        candidates.append(c / c.sum())
    # Gaussian windows
    for sigma in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((xn - 0.5) / sigma) ** 2)
        candidates.append(g / g.sum())
    # Hanning
    h = 0.5 * (1 - np.cos(2 * np.pi * xn))
    h = np.clip(h, 1e-10, None)
    candidates.append(h / h.sum())
    # Random
    for _ in range(10):
        c = rng.random(n)
        candidates.append(c / c.sum())
    # Structured random (smooth)
    for _ in range(5):
        freqs = rng.integers(1, 20, size=10)
        amps = rng.random(10)
        phases = rng.uniform(0, 2 * np.pi, size=10)
        c = np.ones(n)
        for f_, a_, p_ in zip(freqs, amps, phases):
            c += a_ * np.cos(2 * np.pi * f_ * xn + p_)
        c = np.clip(c, 1e-10, None)
        candidates.append(c / c.sum())

    # Evaluate all candidates
    best = None
    best_score = np.inf
    for idx, c in enumerate(candidates):
        c = np.clip(c, 0, None)
        s = c.sum()
        if s < 0.01:
            continue
        c = c / s  # normalize
        sc = compute_score(c)
        if sc < best_score:
            best_score = sc
            best = c.copy()

                print(f"  Init {idx}: score = {sc:.6f}")

    print(f"Phase 1 done: best_score = {best_score:.6f} (time: {time.monotonic()-t_phase1:.1f}s)")

        best_score = 2.0
        best = parent.copy()

    print("Phase 2: Mirror Descent...")

    top_m = 3
    bs = 8
    diag_cache = None
    ds_cache = None
    it = 0
    stall = 0

    while time.monotonic() < deadline - 300:  # save 300s for LP

        max_c = np.max(c)

        # Batch mirror descent
        for b in range(bs):
            # Update diagonal cache every 32 iters
            if b % 32 == 0:
                top_idx = np.argsort(c)[::-1][:top_m]
                top_vals = c[top_idx]
                w = top_vals ** 2
                w /= w.sum()
                diag_cache = np.zeros(n)
                ds_cache = np.zeros(n)
                for idx, wt in zip(top_idx, w):
                    idx = int(idx)
                    js = max(0, idx - n + 1)
                    je = min(idx, n - 1)
                    contrib = wt * 2.0 * a[js:je + 1][::-1]
                    diag_cache[js:je + 1] += contrib
                    ds_cache[js:je + 1] += a[js:je + 1] * contrib

            eta = eta0 / np.sqrt(float(it + b + 1))
            log_f = -eta * diag_cache
            log_f -= np.max(log_f)
            w_new = a * np.exp(log_f)
            ds = float(np.dot(a, ds_cache))
            denom = ds * (np.exp(eta) - 1.0) + 1.0
            w_new /= denom
            w_new = np.maximum(w_new, 1e-30)
            w_new /= w_new.sum()
            a = w_new

        it += bs

                best = a.copy()
                stall = 0
            else:

                eta0 = eta0 * (0.8 if stall > 80 else 1.2)
                eta0 = min(eta0, 0.5 * n)

                # Small random kick

                    a *= rng.uniform(0.9, 1.1, size=n)
                    a = np.maximum(a, 1e-15)

                print(f"  MD iter {it}: score={best_score:.6f}, iters/s={it/elapsed:.0f}")

    # --- Phase 3: LP-based refinement ---
    print("Phase 3: LP Refinement...")
    t_phase3 = time.monotonic()
    try:

    t_phase4 = time.monotonic()
    try:

    print(f"Final score (internal): {final_score:.6f}")
