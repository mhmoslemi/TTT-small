    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.ndim != 1 or parent.size < 2 or not np.isfinite(parent).all():
        parent = np.ones(2535)
    n = parent.size
    j = np.arange(n, dtype=np.float64)
    x = j / max(n - 1, 1)
    inits = [
        _normalize(parent.copy()),
        _normalize(j + 1.0),
        _normalize(n - j),
        _normalize(1.0 + 2.0 * np.sin(0.5 * np.pi * x)),
        _normalize(np.exp(1.5 * x) * (1.0 - 0.5 * x)),
        _normalize((1.0 + x * x) * np.exp(-2.0 * x * (1.0 - x))),
    ]
    scored = []
    for init in inits:
        sc = _score(init)
        if np.isfinite(sc):
            scored.append((sc, init.copy()))
    scored.sort(key=lambda t: t[0])
    if not scored:
        lin = j + 1.0
        lin /= lin.sum()
        return [float(v) for v in lin]
    best_all = scored[0][1].copy()
    best_sc = scored[0][0]
    top_k = min(4, len(scored))
    rem = max(deadline - time.monotonic(), 4.0)
    time_per = rem / top_k
    for k in range(top_k):
        if time.monotonic() >= deadline - 5.0:
            break
        dl = min(time.monotonic() + time_per, deadline - 10.0)
        b, sc = _fft_optimize(scored[k][1].copy(), dl, rng)
        if sc < best_sc:
            best_sc = sc
            best_all = b.copy()
    rem2 = deadline - time.monotonic()
    if rem2 > 80.0:
        b2, sc2 = _fft_optimize(best_all.copy(), deadline - 8.0, rng)
        if sc2 < best_sc:
            best_sc = sc2
            best_all = b2.copy()
    if not np.isfinite(best_all).all() or best_all.sum() < 0.01:
        lin = j + 1.0
        best_all = lin / lin.sum()
    return [float(v) for v in best_all]
