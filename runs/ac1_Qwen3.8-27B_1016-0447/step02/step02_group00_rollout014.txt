    m = 2*n - 1
    # Build using diagonals
    # For lag d = 0, 1, ..., n-1:
    #   The entry a[d] appears at position (k, k-d) for k = d, d+1, ..., n-1
    #   And at position (k, k+d) for k = n-d, ..., 2n-2 (wait, need to think more carefully)

    # For a fixed d = k-j (0 <= d <= n-1):
    #   j can range from 0 to n-1, and k = j+d
    #   Need 0 <= k <= 2n-2, so 0 <= j+d <= 2n-2, i.e., 0 <= j <= 2n-2-d
    #   Combined: 0 <= j <= min(n-1, 2n-2-d)
    
    all_rows = []
    all_cols = []
    all_vals = []
    for d in range(n):
        j_max = min(n-1, 2*n-2-d)
        if j_max < 0:

        ks = js + d
        all_rows.append(ks)
        all_cols.append(js)
        all_vals.append(np.full(j_max+1, a[d]))

    cols = np.concatenate(all_cols)
    vals = np.concatenate(all_vals)
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

    return a / s


def _score(a):
    return float(evaluate_sequence(a.tolist()))


def _fconv(a):
    n = a.size
    sz = 1
    while sz < 2 * n - 1:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa * fa, sz)[:2 * n - 1]


def _subgrad(a, c, topk):
    n = a.size
    kk = min(topk, 2 * n - 1)
    idx = np.argpartition(-c, kk - 1)[:kk]
    v = c[idx]
    w = v / v.sum()
    g = np.zeros(n)
    for ii in range(kk):
        ki = int(idx[ii])
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if lo <= hi:
            g[lo:hi + 1] += w[ii] * 2.0 * a[ki - hi:ki - lo + 1][::-1]
    return g


def _grad_iter(a, eta, topk):
    c = _fconv(a)
    g = _subgrad(a, c, topk)
    gm = g.max()
    if gm < 1e-15:
        return a, c
    a2 = a * np.exp(-eta * g / gm)
    a2 = np.maximum(a2, 1e-300)
    return _norm(a2), c


def _build_conv_mat(a):
    n = a.size
    m = 2 * n - 1
    rs, cs, vs = [], [], []
    for d in range(n):
        jmax = min(n - 1, m - 1 - d)
        if jmax < 0:
            continue
        js = np.arange(jmax + 1)
        rs.append(js + d)
        cs.append(js)
        vs.append(np.full(jmax + 1, a[d]))
    A = sparse.csr_matrix((np.concatenate(vs), (np.concatenate(rs), np.concatenate(cs))), shape=(m, n))
    return A


def _lp_step(a, tlimit=4.0):
    n = a.size
    c = _fconv(a)
    M = c.max()
    A = _build_conv_mat(a)
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M),
                      bounds=[(0, None)]*n, method='highs',
                      options={'time_limit': tlimit, 'presolve': True})
        if res.status == 0 and res.x is not None:
            b = res.x
            sb = b.sum()
            if sb > a.sum() * 1.001:
                return b / sb * a.sum()
    except Exception:
        pass
    return a


def _run(a0, deadline, rng, topk=5, lp_every=300):
    a = _norm(a0.astype(np.float64, copy=True))
    n = a.size
    best, best_sc = a.copy(), _score(a)
    eta = 0.5
    stall = 0
    it = 0
    while time.monotonic() < deadline:
        a, c = _grad_iter(a, eta, topk)
        it += 1
        if it % 60 == 0:
            eta *= 0.985
            if eta < 0.02:
                eta = 0.02
        if it % 30 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-10:
                best_sc, best, stall = sc, a.copy(), 0
            else:
                stall += 1
            if stall >= 15:
                m = rng.random(n) < 0.4
                a[m] *= np.exp(0.08 * rng.standard_normal(int(m.sum())))
                a = _norm(a)
                stall = 0
                eta = 0.5
        if it % lp_every == 0 and time.monotonic() < deadline - 20:
            a = _norm(_lp_step(a))
            sc = _score(a)
            if sc < best_sc - 1e-10:
                best_sc, best = sc, a.copy()
    sc = _score(a)
    if sc < best_sc:
        best, best_sc = a.copy(), sc
    return best, best_sc
