import numpy as np
import time
import math
import scipy.optimize as sopt
from scipy.sparse import csr_matrix

def _fft_conv(a, b, N):
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    out = np.fft.irfft(fa * fb, N)
    return out[:len(a)+len(b)-1]

def _score(a):
    n = len(a)
    N = 1 << (2*n-1).bit_length()
    c = _fft_conv(a, a, N)
    return float(2*n*c.max() / a.sum()**2)

def _active_lp(a, S, M, N):
    n = len(a)
    # LP: min t s.t. 2*conv(a,b)[k] - conv(a,a)[k] <= t for k in S
    #      b >= 0, sum(b) = 1
    # Variables: b_0,...,b_{n-1}, t
    # A_eq: [ones(1,n) | 0] * [b; t] = 1
    # A_ub: [C | -I] * [b; t] <= conv(a,a)[S]  where C[k,j] = 2*a[k-j]
    # b >= 0: A_ub_row: [-I | 0] * [b;t] <= 0
    # t free: (handled by linprog)
    
    caa = _fft_conv(a, a, N)  # conv(a,a)
    S_idx = np.array(S)
    caa_S = caa[S_idx]
    
    # Build C matrix (|S| x n)
    # C[i, j] = 2*a[k_i - j] if 0 <= k_i - j < n else 0
    C = np.zeros((len(S_idx), n))
    for i, k in enumerate(S_idx):
        for j in range(n):
            d = k - j
            if 0 <= d < n:
                C[i, j] = 2 * a[d]
    # This is O(|S| * n) = O(50 * 2535) = 126K ops, fine
    
    nvar = n + 1  # b_0..b_{n-1}, t
    # A_ub: C - I (for t) rows, plus -I for b>=0
    # 2*conv(a,b)[k] - t <= conv(a,a)[k] => C@b - t <= caa_S
    A_ub = np.zeros((len(S_idx) + n, nvar))
    A_ub[:len(S_idx), :n] = C
    A_ub[:len(S_idx), n] = -1.0
    b_ub = caa_S.copy()
    # b_j >= 0 => -b_j <= 0
    A_ub[len(S_idx):, :n] = -np.eye(n)
    b_ub[len(S_idx):] = 0.0
    
    # A_eq: sum(b) = 1
    A_eq = np.zeros((1, nvar))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    c_obj = np.zeros(nvar)
    c_obj[n] = 1.0  # minimize t
    
    bounds = [(0, None)]*n + [(None, None)]  # b >= 0, t free
    
    sol = sopt.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
