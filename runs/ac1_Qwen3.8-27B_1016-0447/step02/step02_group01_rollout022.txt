        p <<= 1
    return p

def sc(x):
    return float(evaluate_sequence(x.tolist()))

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    dl = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    
    try:
        par = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except Exception:
        par = np.ones(512)
    if par.ndim != 1 or par.size < 2 or not np.all(np.isfinite(par)):
        par = np.ones(512)
    
    n = par.size
    ts = np.sqrt(2.0 * n)
    fs = npow2(2 * n - 1)
    
    x = par / max(par.sum(), 0.01) * ts
    np.clip(x, 0.0, 1000.0, out=x)
    
    best = x.copy()
    bsc = sc(best)
    
    if not np.isfinite(bsc):
        return [float(v) for v in np.clip(x, 0.0, 1000.0)]
    
    T = max(0.05, bsc * 0.08)
    T_min = 0.003
    lr = 0.3
    v = np.zeros(n)
    mu = 0.3  # momentum
    stuck = 0
    it = 0
    
    while time.monotonic() < dl - 80.0:
        if it % 500 == 0 and time.monotonic() > dl - 82.0:
            break
        
        A = sfft.rfft(x, fs)
        c = np.real(sfft.irfft(A * A, fs))[:2*n-1]
        np.maximum(c, 0.0, out=c)
        
        cm = c.max()
        K = min(32, c.size)
        idx = np.argpartition(c, -K)[-K:]
        pv = np.exp((c[idx] - cm) / T)
        s = pv.sum()
        p = np.zeros(2*n-1)
        if s > 1e-300:
            p[idx] = pv / s
        else:
            p[int(np.argmax(c))] = 1.0
        
        P = sfft.rfft(p, fs)
        g = 2.0 * np.real(sfft.irfft(P * np.conj(A), fs))[:n]
        g -= g.mean()
        
        v = mu * v + (1.0 - mu) * g
        vn = np.linalg.norm(v)
        if vn > 1e-15:
            x -= lr * (ts / n) * v / vn

        if s < 0.01:
            x = best.copy()
        else:
            x *= ts / s
        np.clip(x, 0.0, 1000.0, out=x)
        
        T = max(T_min, T * 0.9998)
        
        if it > 0 and it % 2000 == 0:
            s_ = sc(x)
            if np.isfinite(s_) and s_ < bsc - 1e-10:
                bsc = s_
