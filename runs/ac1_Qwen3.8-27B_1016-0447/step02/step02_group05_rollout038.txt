        p[n - 1] = 1.0
    else:

def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _normalize(x, target_sum):
    s = x.sum()
    if s < 1e-10:
        x = np.maximum(x, 1e-6)
        s = x.sum()
    x = x / s * target_sum

def _conv_peak(x, fft_size):
    X = sfft.rfft(x, fft_size)

    return np.maximum(c, 0.0)


def _subgrad(x, T, lr, mom, beta, fft_size, corr_size, n):
