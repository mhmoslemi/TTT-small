            gn = np.linalg.norm(g)
            if gn > 1e-15:
                x -= lr * (ts / n) * g / gn
                np.clip(x, 0.0, 1000.0, out=x)
                s = x.sum()
                if s < 0.01:
                    x = best + 0.01 * rng.standard_normal(n)
                    np.clip(x, 0.0, 1000.0, out=x)
                    s = x.sum()
                x *= ts / s
                np.clip(x, 0.0, 1000.0, out=x)
            T = max(T_min, T * 0.9995)
            it += 1
        
        # Check score
        s_ = sc(x)
        if np.isfinite(s_) and s_ < bsc - 1e-12:
            bsc = s_
            best = x.copy()
            stuck = 0
        else:
            stuck += 50
            if stuck > 2000:
                x = best + 0.03 * (ts / n) * rng.standard_normal(n)
                np.clip(x, 0.0, 1000.0, out=x)
                s = x.sum()
                if s > 0.01:
                    x *= ts / s
                    np.clip(x, 0.0, 1000.0, out=x)
                T = max(0.1, T * 3)
                lr = min(1.0, lr * 1.3)
                stuck += -2000  # won't work, let me use stuck = 0
        
    # Phase 2: Exact subgradient polish
