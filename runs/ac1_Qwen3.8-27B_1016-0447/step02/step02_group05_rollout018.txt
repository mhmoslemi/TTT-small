    v = np.zeros(n)       # Nesterov velocity
    T = 0.2 * best_sc
    T_floor = 0.003
    T_decay = 0.9996
    lr = 0.25
    beta = 0.9
    eval_iv = 400
    stagnant = 0
    restart_ct = 0
    MAX_RESTARTS = 30
    STAG_LIMIT = 6000
    it = 0
    
    while time.monotonic() < deadline - 40.0:
        # conv
        Xf = sfft.rfft(x, F)
        conv = np.real(sfft.irfft(Xf * Xf, F))[:2*n-1]
        conv = np.maximum(conv, 0.0)
        
        # softmax
        cm = conv.max()
        lw = np.clip((conv - cm) / T, -55.0, 0.0)
        p = np.exp(lw)
        ps = p.sum()
        if ps < 1e-300:
            p[:] = 0.0
            p[n-1] = 1.0
        else:
            p /= ps
        
        # gradient via cross-correlation
        Pf = sfft.rfft(p, C)
        Xc = sfft.rfft(x, C)
        corr = np.real(sfft.irfft(Pf * np.conj(Xc), C))
        g = 2.0 * corr[:n]
        g -= g.mean()
        
        # Nesterov
        v = beta * v - lr * g
        xn = x + v
        x = _norm(np.maximum(xn, 0.0), S)
        
        T = max(T_floor, T * T_decay)
        if it > 20000:
            lr = max(0.03, lr * 0.99995)
        
        it += 1
        if it % eval_iv == 0:
            sc = float(evaluate_sequence(x.tolist()))
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best = x.copy()
                stagnant = 0
            else:
                stagnant += eval_iv
            if stagnant > STAG_LIMIT and restart_ct < MAX_RESTARTS:
                restart_ct += 1
                x = best * (1.0 + 0.08 * rng.standard_normal(n))
                x = _norm(np.maximum(x, 0.0), S)
                v[:] = 0.0
                T = max(0.08, best_sc * 0.25)
                lr = 0.25
                stagnant = 0
        # occasional noise
        if it % 5000 == 0:
            x += 1e-4 * (S/n) * rng.standard_normal(n)
            x = _norm(np.maximum(x, 0.0), S)
    
    # Phase 2: polish
    if time.monotonic() < deadline - 10.0:
        T2 = 0.002
        lr2 = 0.03
        v2 = np.zeros(n)
        while time.monotonic() < deadline - 10.0:
            Xf = sfft.rfft(x, F)
            conv = np.real(sfft.irfft(Xf * Xf, F))[:2*n-1]
            conv = np.maximum(conv, 0.0)
            cm = conv.max()
            lw = np.clip((conv - cm) / T2, -55.0, 0.0)
