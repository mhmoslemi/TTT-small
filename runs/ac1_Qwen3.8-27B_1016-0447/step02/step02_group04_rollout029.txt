    best = None
    bsc = np.inf
    for c in candidates:
        sc = _score(c)
        if np.isfinite(sc) and sc < bsc:
            bsc, best = sc, c.copy()

    if best is None:
        return [float(ts / n)] * n

    x = best.copy()

    # Main gradient optimization
    T = 0.03
    lr = 0.25
    it = 0
    no_improve = 0

    while time.monotonic() < dl - 3.0:
        X = sfft.rfft(x, fs)
        conv = np.maximum(np.real(sfft.irfft(X * X, fs))[:2 * n - 1], 0.0)
        cm = conv.max()

        if cm < bsc - 1e-10:
            bsc = cm
            best = x.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Softmax peak weighting
        ex = np.exp((conv - cm) / T)
        es = ex.sum()
        if es < 1e-300:
            p = np.zeros(2 * n - 1)
            p[np.argmax(conv)] = 1.0
        else:
            p = ex / es

        # Gradient via FFT
        pr = p[::-1]
        P = sfft.rfft(pr, fs)
        gr = np.real(sfft.irfft(P * X, fs))
        g = gr[n - 1:2 * n - 1][::-1]
        g -= g.mean()

        x = x - lr * g
        x = np.maximum(x, 0.0)
        x = _norm(x)
        x = np.clip(x, 0, 1000)

        # Schedule
        T = max(0.002, T * 0.997)
        if it > 4000:
            lr = max(0.02, lr * 0.998)
        it += 1

        # Noise injection
        if it % 2500 == 0:
            x += 8e-5 * rng.standard_normal(n)
            x = np.maximum(x, 0)
            x = _norm(x)
            x = np.clip(x, 0, 1000)

        # Anti-palindromic projection every 10000 iters
        if it % 10000 == 0:
            d = (x - x[::-1]) / 2
            c0 = x.mean()
            x = c0 + 1.2 * d
            x = np.maximum(x, 0)
            x = _norm(x)
            x = np.clip(x, 0, 1000)

        # Restart if stuck
        if no_improve > 5000:
            mode = rng.integers(0, 4)
            if mode == 0:
                d = (best - best[::-1]) / 2
                r = _anti(rng.standard_normal(n) * best.std() * 0.08)
                x = best + r + 0.5 * d
            elif mode == 1:
                x = best + rng.normal(0, max(best.std() * 0.05, 0.01), n)
            elif mode == 2:
                f = rng.uniform(1, 15)
                x = best + 0.1 * best.std() * np.sin(2 * np.pi * f * np.arange(n) / n)
            else:
                r = _norm(np.clip(rng.standard_normal(n) * 0.3 + 0.5, 0, 1000))
                d = (r - r[::-1]) / 2
                x = 0.5 * (best + r) + 1.5 * d
            x = np.maximum(x, 0)
            x = _norm(x)
            x = np.clip(x, 0, 1000)
            T = max(0.04, 0.02)
            lr = 0.25
            no_improve = 0

    # Post-process: try anti-palindromic sharpening
    for alpha in [1.0, 1.3, 1.6, 2.0]:
        d = (best - best[::-1]) / 2
        c0 = best.mean()
        t = c0 + alpha * d
        t = np.clip(t, 0, 1000)
        if t.sum() > 0.01:
            t = _norm(t)
            sc = _score(t)
            if np.isfinite(sc) and sc < bsc:
                bsc, best = sc, t.copy()

    # Verify and return
    final_sc = _score(best)
    if not np.isfinite(final_sc):
        if np.isfinite(_score(par)):
            return par.tolist()
        return [float(ts / n)] * min(n, 200)
    return best.tolist()
