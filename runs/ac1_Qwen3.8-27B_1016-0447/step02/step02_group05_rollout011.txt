def _np2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _norm(x, ts):
    s = x.sum()
    if s < 1e-10:
        x = np.ones_like(x) * 1e-4
        s = x.sum()
    return np.clip(x * (ts / s), 0.0, 1000.0)


def _acorr(x, fs):
    X = sfft.rfft(x, fs)
    return np.real(sfft.irfft(X * X, fs))[:2 * len(x) - 1]


def _gsog(x, p, cs):
    n = len(x)
    P = sfft.rfft(p, cs)
    X = sfft.rfft(x, cs)
    c = np.real(sfft.irfft(P * np.conj(X), cs))
    g = 2.0 * c[:n]
    return g - g.mean()

    return g - g.mean()
