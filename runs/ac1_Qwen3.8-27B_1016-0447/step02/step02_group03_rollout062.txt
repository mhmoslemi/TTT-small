def norm1(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))

    return a / s

def b_and_score(a):
    L = 2 * GN - 1
    fa = np.fft.rfft(a, GNF)
    b = np.fft.irfft(fa * fa, GNF)[:L]
    mb = float(np.max(b))
    if not np.isfinite(mb):
        return b, 1e100
    s = float(np.sum(a))
    if not np.isfinite(s) or s < 0.01:
        return b, 1e100
    if mb < 0.0:
        mb = 0.0
    return b, float(2.0 * GN * mb / (s * s))


def score_fast(a):
    return b_and_score(a)[1]


def softmax_vec(z):
    z = np.asarray(z, dtype=np.float64)
    if z.size == 0:
        return None
    m = float(np.max(z))
    if not np.isfinite(m):
        return np.ones(z.size, dtype=np.float64) / float(z.size)
    e = np.exp(z - m)
    s = float(e.sum())
    if not np.isfinite(s) or s < 1e-300:
        return np.ones(z.size, dtype=np.float64) / float(z.size)
    e = e / s
    e = np.clip(e, 0.0, 1000.0)
    s2 = float(e.sum())
    if s2 < 0.01:
        return np.ones(z.size, dtype=np.float64) / float(z.size)
    return e / s2


def log_vec(a):
    a = np.asarray(a, dtype=np.float64)
    z = np.log(np.maximum(a, 1e-300))
    if not np.isfinite(z).all():
        z = np.zeros_like(a, dtype=np.float64)
    z = z - float(np.mean(z))
    return np.clip(z, -30.0, 30.0)


def cross_t1(a, w):
    ar = np.ascontiguousarray(a[::-1])
    fa = np.fft.rfft(ar, GNF)
    fw = np.fft.rfft(w, GNF)
    c = np.fft.irfft(fa * fw, GNF)
    return c[GN - 1:GN - 1 + GN].copy()


def lse_weights(b, beta):
    Lb = b.size
    if beta >= 1e6:
        K = 24
        if K > Lb:
            K = Lb
        if K <= 0:
            K = 1
        idx = np.argpartition(b, Lb - K)[Lb - K:]
        w = np.zeros(Lb, dtype=np.float64)
        w[idx] = 1.0 / float(K)
        return w
    sh = b - float(np.max(b))
    bm = beta * sh
    bm = np.clip(bm, -50.0, 0.0)
    e = np.exp(bm)
    S = float(e.sum())
    if not np.isfinite(S) or S < 1e-300:
        w = np.zeros(Lb, dtype=np.float64)
        w[int(np.argmax(b))] = 1.0
        return w
    return e / S
