def npow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def feval(x, fs, n):
    X = sfft.rfft(x, fs)
    c = np.real(sfft.irfft(X * X, fs))[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    nb = c.max()
    sa = x.sum()
    if sa < 0.01:
        return np.inf
    return 2.0 * n * nb / (sa * sa)

def sgrad(p, x, cs, n):
    P = sfft.rfft(p, cs)
    X = sfft.rfft(x, cs)
    g = 2.0 * np.real(sfft.irfft(P * np.conj(X), cs))[:n]
    g -= g.mean()
    return g

def nrm(x, ts):
    np.clip(x, 0.0, 1000.0, out=x)
    s = x.sum()
    if s < 0.01:
        x.fill(ts / x.size)
    else:
        x *= ts / s
    np.clip(x, 0.0, 1000.0, out=x)

def autocorr(x, fs, n):
    X = sfft.rfft(x, fs)
    c = np.real(sfft.irfft(X * X, fs))[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

    T = max(0.08, bsc * 0.12)
    Tmin = 0.003
    lr = 0.35
    K = 48
    it = 0
    EI = 350
    stuck = 0
    SL = 12000

    while time.monotonic() < dl - 60.0:
        c = autocorr(x, fs, n)
        cmax = c.max()

        Kc = min(K, 2 * n - 1)
        idx = np.argpartition(c, -Kc)[-Kc:]
        pw = np.exp((c[idx] - cmax) / T)
        ps = pw.sum()
        p = np.zeros(2 * n - 1)
        if ps > 1e-300:
            p[idx] = pw / ps
        else:
            p[int(c.argmax())] = 1.0

        g = sgrad(p, x, cs, n)
        gn = np.linalg.norm(g)

        if gn > 1e-18:
            x -= lr * (ts / n) * g / gn
            nrm(x, ts)

        if it > 20000:

                bsc = s2
                best = x.copy()
                stuck = 0

                nrm(x, ts)
                T = max(0.08, bsc * 0.12)
                lr = 0.35
                K = 48
                stuck = 0
        it += 1

    rem = dl - time.monotonic()

        thr = c.max() * 0.96
