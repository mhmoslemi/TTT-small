from scipy import fft as sfft

def npow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def fft_conv_sq(x, fs, m):
    X = sfft.rfft(x, fs)
    c = np.real(sfft.irfft(X * X, fs))[:m]
    return c

def fft_grad(p, x, cs, n):
    P = sfft.rfft(p, cs)
    Xc = sfft.rfft(x, cs)
    g = 2.0 * np.real(sfft.irfft(P * np.conj(Xc), cs))[:n]
    g -= g.mean()
    return g

def peak_weights(c, T, K):
    sz = c.size

    else:
        idx = np.argpartition(c, -K)[-K:]
    cm = c.max()
    w = np.exp((c[idx] - cm) / T)
    s = w.sum()
    p = np.zeros(sz)
    if s > 1e-300:
        p[idx] = w / s
    else:
        p[int(np.argmax(c))] = 1.0
    return p
