    return best_x, best_sc


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 20.0)

    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256)
    n = parent.size

    psum = parent.sum()
    if psum < 0.01:
        parent = np.ones(n)
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(best_score):
        best_score = float('inf')

    fft_size = _next_pow2(2 * n)
    corr_size = _next_pow2(3 * n - 1)
    target_sum = float(n)

    schedules = [(0.12, 0.35), (0.04, 0.2), (0.008, 0.08)]

    inits = [
        (parent.copy(), 0),
        (_make_init(n, rng, None, 0), 1),
        (_make_init(n, rng, None, 1), 2),
        (_make_init(n, rng, None, 3), 3),
    ]

    for idx in range(len(inits)):
        if time.monotonic() > deadline - 80.0:
            break
        x_init, _ = inits[idx]
        x_init = _normalize(x_init, target_sum)

        for Ti, lr_i in schedules:
            if time.monotonic() > deadline - 80.0:
                break
            bx, bsc = _opt_loop(x_init, n, fft_size, corr_size, target_sum,
                                Ti, lr_i, deadline, 30000, rng)
            if np.isfinite(bsc) and bsc < best_score:
                best_score = bsc
                best = bx.copy()
            x_init = bx.copy()

        if idx < 3 and time.monotonic() < deadline - 120.0:
            xp = best + 0.1 * rng.standard_normal(n)
            xp = np.maximum(xp, 0.01)
            xp = _normalize(xp, target_sum)
            inits.append((xp, 4 + idx))

    if time.monotonic() < deadline - 40.0:
        x_pol = best.copy()
        bx, bsc = _opt_loop(x_pol, n, fft_size, corr_size, target_sum,
                            0.003, 0.025, deadline, 8000, rng)
        if np.isfinite(bsc) and bsc < best_score:
            best_score = bsc
            best = bx.copy()

    fc = float(evaluate_sequence(best.tolist()))
    if np.isfinite(fc) and fc <= best_score + 1e-12:
        return [float(v) for v in best]

    pc = float(evaluate_sequence(parent.tolist()))
    if np.isfinite(pc):
        return [float(v) for v in parent]
    return [1.0] * n
