        p <<= 1
    return p

    s = x.sum()
    if s < 1e-10:
        x[:] = target_sum / x.size
        return x
    x *= target_sum / s
    np.clip(x, 0.0, 1000.0, out=x)
    return x

    S = x.sum()

        return 1e10
    fs = _next_pow2(2 * n - 1)

def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _normalize(x, target_sum):
    s = x.sum()
    if s < 1e-10:
        x[:] = target_sum / x.size
        return x
    x *= target_sum / s
    np.clip(x, 0.0, 1000.0, out=x)
    return x

    return float(2.0 * n * b.max() / (S * S))
