Let me think about the **LP approach** more carefully, but in a constraint-generation fashion:

1. Start with g = x (current best, normalized).
2. M = max(conv(x, x)).
3. Repeat (a few times):
   a. Compute c = conv(x, g).
   b. Find k* = argmax c.
   c. If c[k*] <= M, done (g is feasible).
   d. The constraint is: sum_i x_i * g_{k*-i} <= M.
   e. Define the "violating weight vector": w_j = x_{k*-j} (for valid j).
   f. We want to reduce g in the direction of w (the positions that contribute most to the violated constraint), while maintaining sum(g) as large as possible.
   g. Update: g = g - epsilon * w / ||w||, then clip to >= 0, renormalize.

This is a projected subgradient method for the LP. It's O(n) per step and uses O(n) storage.

    """LP-inspired correction: find g that maximizes sum with conv(x,g) <= M."""
    M = fft_conv(x, fs).max()
    if M < 1e-15:
        return x.copy()
    g = x.copy()
    for _ in range(niters):
        # Compute cross-conv x * g
        # This is the key: we need conv(x, g) not conv(x, x)
        A = sfft.rfft(x, fs)
        B = sfft.rfft(g, fs)
        c = sfft.irfft(A * B, fs).real[:2*n-1]
        c = np.maximum(c, 0)
        k = int(np.argmax(c))
        if c[k] <= M:
            break
        # Violating direction: w_j = x_{k-j} for valid j
        lo = max(0, k - (n-1))
        hi = min(n-1, k)
        w = np.zeros(n)
        w[lo:hi+1] = x[k-hi:k-lo+1][::-1]
        # Reduce g in direction w
        eps = 0.01 * g.mean()
        g = np.maximum(g - eps * w / (np.linalg.norm(w) + 1e-15), 0)
        g = g / g.sum()
    return g

def npow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def fconv(a, fs):
    A = sfft.rfft(a, fs)
    return np.maximum(sfft.irfft(A * A, fs).real[:2 * len(a) - 1], 0.0)

def sgrad(conv, a, thr, work):
    n = len(a)
    mx = conv.max()
    if mx < 1e-30:
        work[:n] = 0.0
        return
    idx = np.nonzero(conv > thr * mx)[0]

        work[:n] = 0.0
        return
    v = conv[idx]
    s = v.sum()
    work[:n] = 0.0
    for j in range(len(idx)):
        k = int(idx[j])
        w = 2.0 * v[j] / s
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        work[lo:hi + 1] += w * a[k - hi:k - lo + 1][::-1]
