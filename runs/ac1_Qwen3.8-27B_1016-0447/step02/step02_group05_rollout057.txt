import numpy as np
import time
from scipy import fft as sfft


def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _normalize(x, target_sum):
    x = np.clip(x, 0.0, 1000.0)
    s = x.sum()
    if s < 1e-12:
        x = np.ones_like(x) * (target_sum / x.size)
        return x
    return (target_sum / s) * x


def _subgrad(x, T, N, n, target_sum):
    X = sfft.rfft(x, N)
    conv = np.real(sfft.irfft(X * X, N))[:2 * n - 1]
    conv = np.maximum(conv, 0.0)
    m = conv.max()
    p = np.exp((conv - m) / T)
    ps = p.sum()
    if ps < 1e-300:
        p = np.zeros(2 * n - 1)
        p[n - 1] = 1.0
    else:
        p /= ps
    P = sfft.rfft(p, N)
    A = sfft.rfft(x, N)
    c = np.real(sfft.irfft(P * np.conj(A), N))[:n]
    g = c + x
    g -= g.mean()
    gmax = np.abs(g).max()
    if gmax > 1e6:
        g *= 1e6 / gmax
    return g, conv
