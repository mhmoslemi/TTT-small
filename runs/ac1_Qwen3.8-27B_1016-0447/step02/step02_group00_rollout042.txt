    return best, best_score


def _lp_direction(a, M, rng, iters=30):
    n = a.size
    b = np.ones(n) / n
    fft_sz = 1
    while fft_sz < 2 * n - 1:
        fft_sz <<= 1
    fa = np.fft.rfft(a, fft_sz)
    eta = 0.3 / n
    for _ in range(iters):
        fb = np.fft.rfft(b, fft_sz)
        ab = np.fft.irfft(fa * fb, fft_sz)[:2 * n - 1]
        viol = np.maximum(ab - M, 0.0)
        log_v = np.log(np.maximum(viol, 1e-30))
        z = eta * log_v
        z -= z.max()
        w = np.exp(z)
        b = b * (1.0 + w)
        b = np.maximum(b, 1e-15)
        b /= b.sum()
    return b


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.ndim != 1 or parent.size < 2 or not np.isfinite(parent).all():
        parent = np.ones(2535, dtype=np.float64)
    n = parent.size
    parent = np.maximum(parent, 0.0)
    if parent.sum() < 1e-10:
        parent = np.ones(n)
    parent = parent / parent.sum()

    best_all = parent.copy()
    best_sc = _score(parent)

    j = np.arange(n, dtype=np.float64)
    x = j / (n - 1.0)
    inits = [
        parent.copy(),
        (n - j).astype(np.float64),
        (j + 1.0),
        np.exp(2.5 * (1.0 - x)),
        1.0 + 0.5 * np.sin(np.pi * (x + 0.1)),
        np.exp(-8.0 * (x - 0.5) ** 2),
    ]

    scored = []
    for ia in inits:
        ia = _normalize(ia.copy())
        sc = _score(ia)
        if np.isfinite(sc) and sc < 50.0:
            scored.append((sc, ia))
    scored.sort(key=lambda tt: tt[0])

    if not scored:
        lin = j + 1.0
        return [float(vv) for vv in lin / lin.sum()]

    best_all = scored[0][1].copy()
    best_sc = scored[0][0]

    rem = max(total_deadline - time.monotonic(), 5.0)
    top_k = min(3, len(scored))
    polish_frac = 0.25
    search_time = rem * (1.0 - polish_frac)
    time_per = search_time / top_k

    for k in range(top_k):
        if time.monotonic() > total_deadline - 15.0:
            break
        dl = min(time.monotonic() + time_per, total_deadline - 15.0)
        b, sc = _run_opt(scored[k][1].copy(), dl, rng, P=30, lr_init=0.6)
        if sc < best_sc:
            best_sc = sc
            best_all = b.copy()

    # Sharpening phase
    rem2 = total_deadline - time.monotonic()
    if rem2 > 20.0:
        b2, sc2 = _run_opt(best_all.copy(), total_deadline - 3.0, rng, P=60, lr_init=0.3, b1=0.9, b2p=0.995 if 'b2p' in dir() else 0.99)
        if sc2 < best_sc:
            best_sc = sc2
            best_all = b2.copy()

    # LP-inspired refinement
    rem3 = total_deadline - time.monotonic()
    if rem3 > 5.0:
        fa_tmp = np.fft.rfft(best_all, 1 << (2 * n - 1).bit_length())
        c_tmp = np.fft.irfft(fa_tmp * fa_tmp, fa_tmp.size)[:2 * n - 1]
        M_tmp = c_tmp.max()
        b_lp = _lp_direction(best_all, M_tmp, rng, iters=20)
        for mix in [0.05, 0.1, 0.15, 0.2]:
            blended = (1.0 - mix) * best_all + mix * b_lp
            blended = _normalize(blended)
            sc_b = _score(blended)
            if sc_b < best_sc:
                best_sc = sc_b
                best_all = blended.copy()

    # Final micro-polish
    rem4 = total_deadline - time.monotonic()
    if rem4 > 10.0:
        b3, sc3 = _run_opt(best_all.copy(), total_deadline - 2.0, rng, P=80, lr_init=0.15, b1=0.92, b2=0.995)
        if sc3 < best_sc:
            best_sc = sc3
            best_all = b3.copy()

    if not np.isfinite(best_all).all() or best_all.sum() < 0.01:
        best_all = np.ones(n) / n

    return [float(vv) for vv in best_all]
