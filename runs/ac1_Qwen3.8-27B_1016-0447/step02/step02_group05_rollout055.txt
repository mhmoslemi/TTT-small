import numpy as np
import time
from scipy import fft as sfft

def _npo2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def _grad_step(x, T, fft_sz, corr_sz):
    n = x.size
    X = sfft.rfft(x, fft_sz)
    conv = np.real(sfft.irfft(X * X, fft_sz))[:2*n-1]
    conv = np.maximum(conv, 0.0)
    cmax = conv.max()
    # Softmax weights
    w = np.exp(np.clip((conv - cmax) / T, -60.0, 0.0))
    w /= w.sum()
    # Gradient: 2 * cross-correlation of w and x
    W = sfft.rfft(w, corr_sz)
    Xc = sfft.rfft(x, corr_sz)
    g = 2.0 * np.real(sfft.irfft(W * np.conj(Xc), corr_sz))[:n]
    g -= g.mean()  # preserve sum
    return g, cmax
