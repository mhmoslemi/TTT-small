        g /= gm
    return g


def _fast_score(x, fsz):
    X = sfft.rfft(x, fsz)
    b = np.real(sfft.irfft(X * X, fsz))
    n = len(x)
    bm = b[:2 * n - 1].max()
    sa = x.sum()
    if sa < 1e-300:
        return np.inf
    return 2.0 * n * bm / (sa * sa)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t_end = time.monotonic() + float(budget_s) - 15.0
    par = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if par.ndim != 1 or par.size == 0 or not np.all(np.isfinite(par)):
        par = np.ones(256)
    n = par.size
    S = np.sqrt(2.0 * n)
    best = par.copy()
    best_sc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(best_sc):
        return [float(v) for v in par]
    fsz = _np2(2 * n)
    csz = _np2(3 * n)

    j = np.arange(n, dtype=np.float64)
    inits = [
        _norm(par.copy(), S),
        _norm(np.maximum(par * 0.4 + 0.6 + 0.05 * rng.standard_normal(n), 0.001), S),
        _norm(0.5 + 0.5 * np.cos(2.0 * np.pi * j / n), S),
        _norm(1.0 + 0.3 * np.sin(2.0 * np.pi * 3.0 * j / n), S),
        _norm(np.maximum(1.0 + 0.2 * np.cos(4.0 * np.pi * j / n + 0.5), 0.01), S),
    ]
    scs = [_fast_score(c, fsz) for c in inits]
    x = inits[int(np.argmin(scs))].copy()

    T = 0.06
    lr = 0.4
    beta = 0.82
    mom = np.zeros(n)
    gap = 0
    ri = 0
    MAXR = 40
    it = 0
    sc_local = np.inf

    while time.monotonic() < t_end - 20.0:
        g = _grad(x, T, fsz, csz, n)
        mom = beta * mom + (1.0 - beta) * g
        x = np.maximum(x - lr * mom, 0.0)
        x = _norm(x, S)

        T = max(0.001, T * 0.9996)
        if it > 50000:
            lr = max(0.01, lr * 0.99995)

        if it % 400 == 0:
            sc = float(evaluate_sequence(x.tolist()))
            if np.isfinite(sc):
                if sc < best_sc:
                    best_sc = sc
                    best = x.copy()
                if sc < sc_local:
                    sc_local = sc
                    gap = 0
                else:
                    gap += 400
            else:
                gap = 100000

            if gap > 4000 and ri < MAXR:
                ri += 1
                x = _norm(np.maximum(best + 0.03 * rng.standard_normal(n), 0.001), S)
                mom.fill(0.0)
                T = 0.05
                lr = 0.35
                gap = 0
                sc_local = np.inf

        if it % 5000 == 0 and it > 0:
            x += 1e-4 * rng.standard_normal(n)
            x = np.maximum(x, 0.0)
            x = _norm(x, S)

        it += 1

    if time.monotonic() < t_end - 5.0:
        x = best.copy()
        T = 0.002
        lr = 0.02
        m2 = np.zeros(n)
        pi = 0
        while time.monotonic() < t_end - 5.0 and pi < 5000:
            g = _grad(x, T, fsz, csz, n)
            m2 = 0.9 * m2 + 0.1 * g
            x = np.maximum(x - lr * m2, 0.0)
            x = _norm(x, S)
            if pi % 300 == 0:
                sc = float(evaluate_sequence(x.tolist()))
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best = x.copy()
            pi += 1
        sc = float(evaluate_sequence(x.tolist()))
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = x.copy()

    fsc = float(evaluate_sequence(best.tolist()))
    if np.isfinite(fsc) and fsc <= best_sc:
        return [float(v) for v in best]
    return [float(v) for v in par]
