import numpy as np

from scipy import fft as sfft


def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _normalize(x, s):
    t = x.sum()
    if t < 1e-6:
        x = np.maximum(x, 0.01)
        t = x.sum()
    x = x * (s / t)
    return np.clip(x, 0.0, 1000.0)


def _perturb(n, rng, base, amp):
    j = np.arange(n, dtype=np.float64)
    x = base.astype(np.float64).copy()
    for fr in [3, 7, 13, 29, 47, 83]:
        x += amp * np.sin(2.0 * np.pi * fr * j / n + rng.uniform(0.0, 6.2832))
    x += 0.4 * amp * rng.standard_normal(n)
    return np.maximum(x, 1e-3)


def _do_step(x, thresh, lr, beta, mom, f1, f2, n):
    X = sfft.rfft(x, f1)
    conv = np.real(sfft.irfft(X * X, f1))[:2 * n - 1]
    conv = np.maximum(conv, 0.0)
    cmax = conv.max()
    if cmax < 1e-15:
        mom[:] = 0.0
        return x, cmax
    active = conv >= (cmax - thresh)
    na = active.sum()
    if na < 3:
        idx = np.argsort(conv)[-3:]
        active = np.zeros(2 * n - 1, dtype=bool)
        active[idx] = True
    p = active.astype(np.float64)
    p /= p.sum()
    P = sfft.rfft(p, f2)
    Xc = sfft.rfft(x, f2)
    corr = np.real(sfft.irfft(P * np.conj(Xc), f2))
    g = 2.0 * corr[:n]
    g -= g.mean()
    gm = np.abs(g).max()
    if gm > 1e4:
        g *= 1e4 / gm
    if gm < 1e-12:
        return x, cmax
    mom[:] = beta * mom + (1.0 - beta) * g
    x_new = x - lr * mom
    np.clip(x_new, 0.0, 1000.0, out=x_new)
    return x_new, cmax
