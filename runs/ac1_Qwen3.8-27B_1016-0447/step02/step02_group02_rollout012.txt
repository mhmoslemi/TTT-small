import numpy as np
import time
from scipy import fft as sfft


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def fft_conv(x, fs):
    A = sfft.rfft(x, fs)
    return np.maximum(sfft.irfft(A * A, fs).real[: 2 * len(x) - 1], 0.0)


def grad_topm(conv, x, m, work, n):
    k = min(m, len(conv))
    idx = np.argpartition(conv, -k)[-k:]
    mx = conv[idx].max()
    w = np.exp(8.0 * (conv[idx] - mx) / max(mx, 1e-15))
    w /= w.sum()
    work[:n] = 0.0
    for j in range(k):
        kk = int(idx[j])
        wi = float(w[j])
        lo = max(0, kk - (n - 1))
        hi = min(kk, n - 1)
        if lo <= hi:
            work[lo:hi + 1] += 2.0 * wi * x[kk - hi: kk - lo + 1][::-1]
    return work[:n]


def grad_single(kk, x, work, n):
    work[:] = 0.0
    lo = max(0, kk - (n - 1))
    hi = min(kk, n - 1)
    if lo <= hi:
        work[lo:hi + 1] = 2.0 * x[kk - hi: kk - lo + 1][::-1]
    return work[:n]
