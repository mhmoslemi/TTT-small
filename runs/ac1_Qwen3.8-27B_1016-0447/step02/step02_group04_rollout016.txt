    par = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if par.ndim != 1 or par.size == 0 or not np.all(np.isfinite(par)):
        par = np.ones(1024)
    n = par.size
    ts = np.sqrt(2.0 * n)
    fs = _n2(2 * n)
    jj = np.arange(n, dtype=np.float64)

    x = _norm(par.copy(), ts, n)
    best = x.copy()
    bsc = _ev(best)
    if not np.isfinite(bsc):
        best = np.full(n, ts / n)
        bsc = _ev(best)

    for _ in range(25):
        freq = rng.uniform(1, 30)
        ph = rng.uniform(0, 6.283)
        amp = rng.uniform(0.02, 0.25)
        t = par.copy() + amp * np.sin(2 * np.pi * freq * jj / n + ph)
        t = _norm(t, ts, n)
        sc = _ev(t)
        if np.isfinite(sc) and sc < bsc:
            bsc, best, x = sc, t.copy(), t.copy()

    mom = np.zeros(n)
    it = 0
    P_lp = 150
    stool = 0

    while time.monotonic() < dl - 5.0:
        conv = _fc(x, fs)
        cm = conv.max()

        if it % 40 == 0:
            sc = _ev(x)
            if np.isfinite(sc) and sc < bsc:
                bsc = sc
                best = x.copy()
                stool = 0
            else:
                stool += 1

        do_lp = (it % 4 == 0) and P_lp <= 2 * n - 1

        if do_lp:
            Pk = min(P_lp, 2 * n - 1)
            pidx = np.argpartition(conv, -Pk)[-Pk:]
            pidx = pidx[np.argsort(conv[pidx])[::-1]]

            A = np.zeros((Pk, n))
            for i in range(Pk):
                k = pidx[i]
                lo = max(0, k - n + 1)
                hi = min(k, n - 1) + 1
                if hi > lo:
                    A[i, lo:hi] = x[k - lo: k - lo + (hi - lo)][::-1]

            lo_b = -x * 0.4
            hi_b = x * 0.4
            try:
                tlim = max(0.5, min(45.0, dl - time.monotonic() - 15.0))
                res = linprog(-np.ones(n), A_ub=A, b_ub=np.zeros(Pk),
                             bounds=list(zip(lo_b, hi_b)),
                             method='highs',
                             options={'time_limit': tlim, 'presolve': True})
            except Exception:
                res = None

            moved = False
            if res is not None and res.success and np.abs(res.x).max() > 1e-10:
                d = res.x
                bt, bv = 0.0, cm
                for tt in [0.5, 1.0, 2.0, 3.0, 5.0, 8.0]:
                    cand = _norm(x + tt * d, ts, n)
                    cvm = _fc(cand, fs).max()
                    if cvm < bv:
                        bv = cvm
                        bt = tt
                if bt > 0:
                    x = _norm(x + bt * d, ts, n)
                    moved = True

            if not moved:
                T = max(0.004, 0.2 * cm)
                ex = np.exp((conv - cm) / T)
                es = ex.sum()
                if es < 1e-300:
                    p = np.zeros(2 * n - 1)
                    p[np.argmax(conv)] = 1.0
                else:
                    p = ex / es
                pr = p[::-1]
                Pfr = sfft.rfft(pr, fs)
                Xfr = sfft.rfft(x, fs)
                gr = np.real(sfft.irfft(Pfr * Xfr, fs))
                g = gr[n - 1:2 * n - 1][::-1]
                g -= g.mean()
                mom = 0.85 * mom + 0.15 * g
                x = _norm(x - 0.35 * mom, ts, n)
        else:
            T = max(0.004, 0.2 * cm)
            ex = np.exp((conv - cm) / T)
            es = ex.sum()
            if es < 1e-300:
                p = np.zeros(2 * n - 1)
                p[np.argmax(conv)] = 1.0
            else:
                p = ex / es
            pr = p[::-1]
            Pfr = sfft.rfft(pr, fs)
            Xfr = sfft.rfft(x, fs)
            gr = np.real(sfft.irfft(Pfr * Xfr, fs))
            g = gr[n - 1:2 * n - 1][::-1]
            g -= g.mean()
            mom = 0.9 * mom + 0.1 * g
            x = _norm(x - 0.3 * mom, ts, n)

        it += 1

        if it % 80 == 0:
            x += 1e-4 * rng.standard_normal(n)
            x = _norm(x, ts, n)

        if stool > 30 and it % 200 == 0:
            x = best + rng.normal(0, max(best.std(), 0.01) * 0.05, n)
            x = _norm(x, ts, n)
            mom *= 0.3
            stool = 0

        if it % 2000 == 0 and it > 0:
            P_lp = min(P_lp + 25, 400)

    sc = _ev(best)
    if not np.isfinite(sc):
        sc2 = _ev(par)
        if np.isfinite(sc2):
            return par.tolist()
        return [float(ts / n)] * n
    return best.tolist()
