        # Using scipy.signal.correlate: g = signal.correlate(w, x, mode='valid')
        # But we want to avoid scipy.signal for speed. Let's use FFT.
        #
        # g[j] = sum_{l=0}^{n-1} x[l] * w[l+j]
        # Let's define a new sequence: x_padded of length n (already is)
        # w of length 2n-1
        # The cross-correlation (w ⋆ x)[j] = sum_l x[l] * w[l+j] (with x = 0 outside [0,n))
        # = sum_l x_padded[l] * w[l+j]
        # 
        # FFT approach: pad both to length >= n + (2n-1) - 1 = 3n-2
        # cross_corr = ifft(fft(x, N) * conj(fft(w, N)))[:n]
        # Wait: ifft(fft(a)*conj(fft(b)))[k] = sum_j a[j] * b[j-k] (circular)
        # For linear (with enough padding): = sum_j a[j] * b[j-k] where b[j-k]=0 for j-k outside b's range
        # I want sum_l x[l] * w[l+j] = sum_l x[l] * w[l-(-j)]
        # So with a=x, b=w, k=-j: sum_l x[l] * w[l-(-j)] = sum_l x[l] * w[l+j] ✓
        # But k ranges 0 to N-1, and -j mod N = N-j for j>0
        # So: cross_corr[j] = ifft(fft(x,N)*conj(fft(w,N)))[(N-j) % N] for j=0..n-1
        # Or equivalently: take the last n elements in reverse?
        
        # Actually, let me just verify with a small example and use the correct indexing.
        # For now, let's use the "flip w" trick:
        # g[j] = sum_l x[l] * w[l+j]
        # Let w_flip[m] = w[2n-2-m]
        # (x * w_flip)[k] = sum_i x[i] * w_flip[k-i] = sum_i x[i] * w[2n-2-k+i]
        # Set 2n-2-k+i = l+j, so i = k-2n+2+j+l... this is getting complicated.
        
        # Simpler: just use np.correlate for the gradient (it's O(n*(2n)) = O(n^2) but n=4498 means ~20M ops, which is < 1ms in numpy)
        # Actually np.correlate is C-optimized and should be fast enough.
        
        # Or even simpler: since we already have the FFT of x, we can compute the gradient differently.
        # 
        # Actually, the cleanest FFT approach:
        # g[j] = sum_{l=0}^{n-1} x[l] * w[l+j] for j = 0, ..., n-1
        # Pad x to length N (N >= 3n-2), zero-pad w to length N
        # Compute Y = fft(w, N), X = fft(x, N)
        # The circular cross-correlation: c[k] = ifft(X * conj(Y))[k] = sum_l x[l] * w[(l-k) mod N]
        # For k = 0, ..., n-1 (which corresponds to our j):
        # c[k] = sum_l x[l] * w[l-k] (assuming no wrap-around since N is large enough)
        # But I want w[l+k], not w[l-k].
        # So c[N-k] = sum_l x[l] * w[l-(N-k) mod N] = sum_l x[l] * w[l+k] (for k=0..n-1, since N-k is large and l-(N-k) underflows to l+k-N+N = l+k)... 
        # Hmm, this is getting confusing. Let me just use:
        # 
        # g[j] = sum_l x[l] * w[l+j]
        # Define reversed w: wr[m] = w[2n-2-m] for m = 0, ..., 2n-2
        # (x * wr)[k] = sum_i x[i] * wr[k-i] = sum_i x[i] * w[2n-2-(k-i)] = sum_i x[i] * w[2n-2-k+i]
        # I want this to equal sum_i x[i] * w[i+j] for some j.
        # 2n-2-k+i = i+j => j = 2n-2-k. So k = 2n-2-j.
        # (x * wr)[2n-2-j] = sum_i x[i] * w[i+j] = g[j] ✓
        # 
        # So: conv_xw = ifft(fft(x, N) * fft(wr, N)).real
        # g[j] = conv_xw[2n-2-j] for j = 0, ..., n-1
        # i.e., g = conv_xw[2n-2 : n-2 : -1] (reversed slice)
        
        # Let me just use the direct approach with np.correlate since it's fast enough for n~4500
        # g = 2 * np.correlate(w, x, mode='valid')  # This gives sum_l x[l] * w[l+j]
        # Wait: np.correlate(a, v, 'valid') computes sum over v: a[i+k] * v[i] for k in range
        # np.correlate(w, x, 'valid')[j] = sum_i w[i+j] * x[i] (for valid i)
        # Since len(w) = 2n-1 > len(x) = n, 'valid' gives n elements
        # = sum_{i=0}^{n-1} w[i+j] * x[i] = g[j] / 2 ✓
        
        grad = 2.0 * np.correlate(w, x, mode='valid')
        
        # Center the gradient (remove mean to preserve sum)
        grad = grad - grad.mean()
        
        # Adam update
        it += 1
        m_v = beta1 * m_v + (1 - beta1) * grad
        v_v = beta2 * v_v + (1 - beta2) * grad * grad
        m_hat = m_v / (1 - beta1 ** it)
        v_hat = v_v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        
        x_new = x - step
        x_new = np.maximum(x_new, 0.0)
        s = x_new.sum()
        if s < 1e-14:
            x_new = np.ones(n) / n
        else:
            x_new = x_new / s
        
        x = x_new
        
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = x.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Increase temperature
        lam = min(lam_max, lam * 1.002)
        
        # Restart if stuck
        if no_imp >= IMP_LIM:
            x = best + 0.03 * rng.standard_normal(n)
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            m_v[:] = 0
            v_v[:] = 0
            it_reset = 0
            no_imp = 0
            lam = 2.0
            lr = max(0.01, lr * 0.8)
        
        # Check deadline
        if it % 5000 == 0:
            sc = float(evaluate_sequence(x.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best = x.copy()
    
    # Final polish
    if time.monotonic() < deadline - 5.0:
        x = best.copy()
        lam = lam_max
        lr = 0.005
        for _ in range(5000):
            if time.monotonic() > deadline - 5.0:
                break
            conv = fft_conv(x, fs)
            c2 = 2.0 * n * conv.max()
            if c2 < best_score - 1e-12:
                best_score = c2
                best = x.copy()
            conv_shifted = conv - conv.max()
