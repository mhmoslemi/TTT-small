            if np.isfinite(s_) and s_ < bsc - 1e-10:
                bsc = s_
                best = x.copy()
                stuck = 0
            else:
                stuck += 1000
                if stuck > max_stuck:
                    x = best + 0.02 * rng.standard_normal(n)
                    np.clip(x, 0.0, 1000.0, out=x)
                    sc_c = x.sum()
                    if sc_c < 1e-10:
                        x = best.copy()
                    else:
                        x /= sc_c
                    T = 0.03
                    lr0 = 0.2
                    stuck = 0
        it += 1

    for _ in range(300):
        if time.monotonic() > dl - 2.0:
            break
        X = sfft.rfft(best, fs)
        c = np.real(sfft.irfft(X * X, fs))[:2 * n - 1]
        np.maximum(c, 0.0, out=c)
        ks = int(np.argmax(c))
        p = np.zeros(2 * n - 1)
        p[ks] = 1.0
        P = sfft.rfft(p, fs)
        g = 2.0 * np.real(sfft.irfft(P * np.conj(X), fs))[:n]
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        improved = False
        for st in (0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005):
            cand = best - st * g
            np.clip(cand, 0.0, 1000.0, out=cand)
            sc_c = cand.sum()
            if sc_c < 1e-10:
                continue
            cand /= sc_c
            s_ = sc(cand)
            if np.isfinite(s_) and s_ < bsc - 1e-12:
                bsc = s_
                best = cand.copy()
                improved = True
                break
        if not improved:
            break

    if not np.isfinite(bsc):
        x = np.clip(x, 0.0, 1000.0)
        if x.sum() < 0.01:
            x = np.ones(n)
        return [float(v) for v in x]
    return [float(v) for v in best]
