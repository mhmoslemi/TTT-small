def _grad_soft(a, tau):
    """Compute soft gradient g[i] = sum_j w[i+j] * 2*a[j] where w = softmax(c/tau)."""
    n = a.size
    ft = 2 * n - 1
    # FFT size for convolution
    sz = 1
    while sz < ft:
        sz <<= 1
    # Compute c = a * a
    fa = np.fft.rfft(a, sz)
    c = np.fft.irfft(fa * fa, sz)[:ft]

    c_norm = c - c.max()
    e = np.exp(c_norm / tau)
    w = e / e.sum()

def _lp_polish(a, M, deadline):
    """Solve LP: maximize sum(b) s.t. a*b <= M, b >= 0."""
    n = a.size
    ft = 2 * n - 1
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for k in range(ft):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = scipy.sparse.coo_matrix((vals, (rows, cols)), shape=(ft, n)).tocsc()
    
    # LP: max sum(b) s.t. A @ b <= M, b >= 0

        c=-np.ones(n),
        A_ub=A,
        b_ub=np.full(ft, M),
        bounds=[(0, None)] * n,
        method='highs',

    )
    if res.status == 0:
        b = res.x
        b = np.maximum(b, 0)
        return b
    return None
