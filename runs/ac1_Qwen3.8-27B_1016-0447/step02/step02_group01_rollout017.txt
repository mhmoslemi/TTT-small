    p = 1

    X = sfft.rfft(x, fs)
    c = np.real(sfft.irfft(X * X, fs))[:2*n-1]
    np.maximum(c, 0.0, out=c)
    return c

def _dw(c, T, K):
    K = min(K, c.size)
    cm = c.max()

    p = np.zeros(c.size)

This is O(n) and doesn't need FFT. But for multiple peaks (softmax), I need the FFT cross-correlation.

Let me use the single-peak gradient for the polish phase (faster) and the multi-peak for the main phase.

Actually, the single-peak gradient via FFT is the same cost as the multi-peak (one FFT of p + one FFT of a + one inverse FFT). But if p is a delta, I can skip the FFT of p and just do the shift. Let me implement both.

v = beta * v + (1-beta) * g

def _ac(x, fs, n):
    X = sfft.rfft(x, fs)
    c = np.real(sfft.irfft(X * X, fs))[:2*n-1]
    np.maximum(c, 0.0, out=c)
    return c

def _dw(c, T, K):
    K = min(K, c.size)
    cm = c.max()
    idx = np.argpartition(c, -K)[-K:]
    v = np.exp((c[idx] - cm) / T)

    p = np.zeros(c.size)

        p[idx] = v / s
    else:
        p[int(np.argmax(c))] = 1.0
