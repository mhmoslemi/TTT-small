    p = np.exp(logits)
    ps = p.sum()
    if ps < 1e-300:
        p[:] = 0.0
        p[n - 1] = 1.0
    else:
        p /= ps
    P_f = sfft.rfft(p, fs)
    corr = np.real(sfft.irfft(P_f * np.conj(X), fs))[:n]
    S = x.sum()
    M = float(np.dot(p, conv))
    grad = 4.0 * n * corr / (S * S) - 2.0 * n * M / (S * S * S)
    grad -= grad.mean()
    gn = np.linalg.norm(grad)
    if gn > 1e-15:
        grad /= gn
    return grad


def _optimize(x0, tgt, fs, n, rng, deadline, best, best_score, T0, lr0, max_iters, beta):
    x = _norm(x0.copy(), tgt)
    mom = np.zeros(n)
    T = T0
    lr = lr0
    no_imp = 0
    for it in range(max_iters):
        if time.monotonic() > deadline - 10.0:
            break
        grad = _grad_step(x, T, fs, n)
        mom = beta * mom + (1.0 - beta) * grad
        x = x - lr * mom
        _norm(x, tgt)
        T = max(0.005, T * 0.9993)
        if it > 10000:
            lr *= 0.9997
        if it % 150 == 0:
            sc = float(evaluate_sequence(x.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best[:] = x
                no_imp = 0
            else:
                no_imp += 150
            if no_imp > 2500:
                x += 0.015 * rng.standard_normal(n)
                _norm(x, tgt)
                mom *= 0.3
                no_imp = 0
    return x, best_score


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 15.0)
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(4498)
    n = parent.size
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(best_score):
        return [float(v) for v in parent]
    tgt = np.sqrt(2.0 * n)
    fs = _np2(3 * n)

    # Phase 1: Main gradient optimization from parent
    x, best_score = _optimize(parent, tgt, fs, n, rng, deadline, best,
                              best_score, 0.5, 0.003, 30000, 0.85)

    # Phase 2: Fine-tune with low temperature
    if time.monotonic() < deadline - 20.0:
        x, best_score = _optimize(best, tgt, fs, n, rng, deadline, best,
                                  best_score, 0.02, 0.0008, 20000, 0.9)

    # Phase 3: Medium-temperature restart from perturbed best
    if time.monotonic() < deadline - 15.0:
        x0 = best + 0.04 * rng.standard_normal(n)
        x, best_score = _optimize(x0, tgt, fs, n, rng, deadline, best,
                                  best_score, 0.3, 0.002, 15000, 0.85)

    # Phase 4: Additional diverse starts
    if time.monotonic() < deadline - 10.0:
        j = np.arange(n, dtype=np.float64)
        for trial in range(3):
            if time.monotonic() > deadline - 5.0:
                break
            freqs = rng.integers(3, 60, size=4).astype(np.float64)
            phases = rng.uniform(0.0, 2.0 * np.pi, size=4)
            amps = rng.uniform(0.1, 0.4, size=4)
            x0 = 1.0
            for k in range(4):
                x0 += amps[k] * np.sin(2.0 * np.pi * freqs[k] * j / n + phases[k])
            x0 = np.maximum(x0, 0.01)
            x, best_score = _optimize(x0, tgt, fs, n, rng, deadline, best,
                                      best_score, 0.4, 0.003, 12000, 0.85)

    # Final verification
    fc = float(evaluate_sequence(best.tolist()))
    if np.isfinite(fc) and fc <= best_score:
        return [float(v) for v in best]
    return [float(v) for v in parent]
