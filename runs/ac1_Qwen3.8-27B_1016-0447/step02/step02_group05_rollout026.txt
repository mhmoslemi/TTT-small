```python
def _simplex_project(x, S):
    n = len(x)
    # Find tau such that sum(max(x - tau, 0)) = S
    # Sort x in descending order
    u = np.sort(x)[::-1]
    # Find the right threshold
    # cumsum(1:(n)) / (1:(n))
    cs = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u - (cs - S) / rho > 0
    if not cond.any():
        tau = x - S / n
        return np.maximum(x - tau, 0)
    rho_max = np.max(rho[cond])
    tau = (cs[rho_max-1] - S) / rho_max
    result = np.maximum(x - tau, 0.0)
    # Clip to 1000
    result = np.minimum(result, 1000.0)
    return result
```

```python
import numpy as np
import time
from scipy import fft as sfft


def _np2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _spj(x, S):
    """Project onto {x>=0, sum(x)=S} via water-filling."""
    x = np.clip(x, 0.0, 1000.0)
    n = x.size
    s = x.sum()
    if s < 1e-15:
        return np.full(n, S / n)
    if abs(s - S) < 1e-10:
        return x
    u = np.sort(x)[::-1]
    cs = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    d = u - (cs - S) / rho
    idx = np.where(d > 0)[0]
    if len(idx) == 0:
        return np.clip(x - (s - S) / n, 0.0, 1000.0)
    tau = (cs[idx[-1]] - S) / rho[idx[-1]]
    return np.clip(x - tau, 0.0, 1000.0)


def _gsm(x, T, fs, csz, n):
    """Smoothed gradient via softmax over autocorrelation lags."""
    X = sfft.rfft(x, fs)
    cv = np.real(sfft.irfft(X * X, fs))[:2 * n - 1]
    cv = np.maximum(cv, 0.0)
    cm = cv.max()
    lg = np.clip((cv - cm) / T, -50.0, 0.0)
    p = np.exp(lg)
    s = p.sum()
    if s < 1e-300:
        p = np.zeros(2 * n - 1)
        p[(2 * n - 1) // 2] = 1.0
    else:
        p /= s
    P = sfft.rfft(p, csz)
    Xc = sfft.rfft(x, csz)
    cr = np.real(sfft.irfft(P * np.conj(Xc), csz))
    g = 2.0 * cr[:n]
    g -= g.mean()
    m = np.abs(g).max()
    if m > 1e-12:
        g /= m
    return g


def _gex(x, fs, n):
    """Exact subgradient at the single active (max) lag."""
    X = sfft.rfft(x, fs)
    cv = np.real(sfft.irfft(X * X, fs))[:2 * n - 1]
    cv = np.maximum(cv, 0.0)
    k = int(cv.argmax())
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        g[lo:hi + 1] = 2.0 * x[k - np.arange(lo, hi + 1)]
    g -= g.mean()
    m = np.abs(g).max()
    if m > 1e-12:
        g /= m
    return g


def _inits(n, rng, base, S):

    a = _spj(np.maximum(base, 0.01) + 0.2 * rng.standard_normal(n), S)
    b = _spj(np.maximum(base, 0.01) + 0.5 * np.sin(2 * np.pi * rng.integers(2, 25) * j / n + rng.uniform(0, 6.28)), S)
    c = int(rng.integers(0, n))
    w = max(10.0, n / 10.0)
    d = _spj(np.exp(-0.5 * ((j - c) / w) ** 2) + 0.05 * np.maximum(base, 0), S)
    e = _spj(1.0 + 0.4 * np.cos(2 * np.pi * rng.uniform(0.5, 6.0) * j / n) + 0.1 * rng.standard_normal(n), S)
    return [a, b, d, e]

    Tm = 0.002

    return bs, bv
```
