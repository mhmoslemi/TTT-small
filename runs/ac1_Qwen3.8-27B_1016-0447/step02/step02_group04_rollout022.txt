Also, I want to double-check: is `conv ** 2` the right weighting? For a flat autocorrelation (all values equal), `conv ** 2` is also flat, so the gradient would be uniform (after subtracting mean, it's zero). That's fine - we don't want to move if the autocorrelation is already flat.

For a peaked autocorrelation, `conv ** 2` amplifies the peaks, so the gradient focuses on reducing the peaks. Good.

But there's a subtlety: the gradient `g[j] = sum_k p[k] * f[k-j]` (the cross-correlation of p and f). If p is concentrated at lag 0, then `g[j] = f[0] * p[0] + ... ≈ f[j] * p[0]` (approximately, for the central part). So the update would be `x[j] *= exp(-lr * p[0] * f[j])`, which reduces larger values more. This is like a "leveling" operation, which makes sense for reducing the peak at lag 0.

If p is concentrated at some other lag k*, the gradient pushes to reduce the overlap at that lag. Good.

from scipy import fft as sfft

    while p < m:

    par = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if par.ndim != 1 or par.size == 0 or not np.all(np.isfinite(par)):

    n = par.size
    ts = np.sqrt(2.0 * n)
    fs = _np2(2 * n)

    x = _nrm(par.copy(), n, ts)
    best = x.copy()
    bsc = _sc(best)

    # Quick structured initializations
    for frac in (0.5, 0.35, 0.65, 0.25):
        t = np.zeros(n)
        m = int(n * frac)
        t[:m] = 1.0
        if 1.0 - frac > 0.15:
            t[-int(n * (1 - frac)):] = 1.0
        t = _nrm(t, n, ts)
        sc = _sc(t)
        if np.isfinite(sc) and sc < bsc:
            bsc, best, x = sc, t.copy(), t.copy()

    # Random perturbations of parent
    jj = np.arange(n, dtype=np.float64)
    for _ in range(8):
        t = par.copy()
        t += rng.uniform(0.05, 0.30) * np.sin(2 * np.pi * rng.uniform(1, 15) * jj / n + rng.uniform(0, 6.28))
        t = _nrm(t, n, ts)
        sc = _sc(t)
        if np.isfinite(sc) and sc < bsc:
            bsc, best, x = sc, t.copy(), t.copy()

    # Main multiplicative gradient descent
    lr = 0.35
    g_ema = np.zeros(n)
    beta = 0.4
    it = 0
    noi = 0

    while time.monotonic() < dl - 5.0:

        conv = np.maximum(np.real(sfft.irfft(X * X, fs))[:2 * n - 1], 0.0)
        cm = conv.max()

        if cm < bsc - 1e-9:

            best = x.copy()
            noi = 0
        else:

        # Peak-focused power weighting
        p = conv ** 2
        ps = p.sum()
        if ps < 1e-300:
            p = np.zeros(2 * n - 1)
            p[np.argmax(conv)] = 1.0
        else:
            p /= ps

        # Gradient: g[j] = sum_k p[k] * f[k-j]  (cross-correlation)
        pr = p[::-1]
        P = sfft.rfft(pr, fs)
        gr = np.real(sfft.irfft(P * X, fs))
        g = gr[n - 1:2 * n - 1][::-1]
        g -= g.mean()

        # Exponential moving average for smoothing
        g_ema = beta * g_ema + (1.0 - beta) * g

        # Multiplicative (mirror) update with clamped exponent
        e = np.clip(-lr * g_ema, -3.5, 3.5)
        x = x * np.exp(e)
        x = _nrm(x, n, ts)
        x = np.clip(x, 0, 1000)

        # Adaptive schedule
        if it < 20000:
            lr = 0.35 - 0.15 * (it / 20000.0)
        else:
            lr = max(0.02, lr * 0.99997)
        it += 1

            x = np.clip(x, 0, 1000)

            if len(idx) > 0:

            else:

            P2 = sfft.rfft(pr2, fs)

        return par.tolist()
    # Also try current x
    csc = _sc(x)
    if np.isfinite(csc) and csc < fsc:
        best = x.copy()
    return best.tolist()
