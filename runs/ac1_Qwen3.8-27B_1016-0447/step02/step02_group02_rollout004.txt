import numpy as np
import time
from scipy import fft as sfft


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def fft_conv(a, fs):
    A = sfft.rfft(a, fs)
    out = sfft.irfft(A * A, fs).real[:2 * len(a) - 1]
    return np.maximum(out, 0.0)


def topm_grad(conv, a, m, work):
    n = len(a)
    L = len(conv)
    k = min(m, L)
    idx = np.argpartition(conv, -k)[-k:]
    vals = conv[idx]
    total = vals.sum()
    if total < 1e-30:
        work[:n] = 0.0
        return work
    ws = vals / total
    work[:n] = 0.0
    for j in range(k):
        kk = int(idx[j])
        w = float(ws[j])
        lo = max(0, kk - (n - 1))
        hi = min(kk, n - 1)
        if lo <= hi:
            work[lo:hi + 1] += 2.0 * w * a[kk - hi:kk - lo + 1][::-1]
    return work
