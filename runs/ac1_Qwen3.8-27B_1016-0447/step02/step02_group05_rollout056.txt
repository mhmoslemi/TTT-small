import numpy as np
import time
from scipy import fft as sfft


def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _normalize(x, target_sum):
    s = x.sum()
    if s < 1e-10:
        x = np.ones_like(x)
        s = x.sum()
    x = x * (target_sum / s)
    return np.clip(x, 0.0, 1000.0)


def _fast_score(x):
    n = len(x)
    sa = x.sum()
    if sa < 0.01:
        return float('inf')
    fs = _next_pow2(2 * n - 1)
    X = sfft.rfft(x, fs)
    conv = np.real(sfft.irfft(X * X, fs))[:2 * n - 1]
    return 2.0 * n * conv.max() / (sa * sa)


def _subgrad_step(x, T, lr, momentum, beta, fft_size, corr_size, n):
    X = sfft.rfft(x, fft_size)
    conv = np.real(sfft.irfft(X * X, fft_size))[:2 * n - 1]
    conv = np.maximum(conv, 0.0)
    cmax = conv.max()
    if cmax < 1e-300:
        return x, conv
    logits = np.clip((conv - cmax) / T, -50.0, 0.0)
    p = np.exp(logits)
    ps = p.sum()
    if ps < 1e-300:
        p = np.zeros(2 * n - 1)
        p[n - 1] = 1.0
    else:
        p /= ps
    P = sfft.rfft(p, corr_size)
    Xc = sfft.rfft(x, corr_size)
    corr = np.real(sfft.irfft(P * np.conj(Xc), corr_size))
    g = 2.0 * corr[:n]
    g -= g.mean()
    gmax = np.abs(g).max()
    if gmax > 1e5:
        g *= 1e5 / gmax
    momentum[:] = beta * momentum + (1.0 - beta) * g
    x_new = x - lr * momentum
    return np.clip(x_new, 0.0, 1000.0), conv
