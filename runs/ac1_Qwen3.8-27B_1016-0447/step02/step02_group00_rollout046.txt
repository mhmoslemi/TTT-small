def _score(a):

def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _optimize(a, deadline, rng, tau0, lr0):

    a = _normalize(a.astype(np.float64, copy=True))
    best = a.copy()
    best_score = _score(a)

    tau = tau0
    lr = lr0
