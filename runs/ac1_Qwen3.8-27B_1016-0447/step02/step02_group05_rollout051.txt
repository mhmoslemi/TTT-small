import numpy as np
import time
from scipy import fft as sfft


def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _normalize(x, target_sum):
    s = x.sum()
    if s < 1e-10:
        return np.full_like(x, target_sum / x.size)
    x = x * (target_sum / s)
    return np.clip(x, 0.0, 1000.0)


def _subgrad_step(x, T, lr, mom, beta, fft_size, corr_size, n):
    X = sfft.rfft(x, fft_size)
    conv = np.real(sfft.irfft(X * X, fft_size))[:2 * n - 1]
    conv = np.maximum(conv, 0.0)
    cmax = conv.max()
    logits = np.clip((conv - cmax) / T, -50.0, 0.0)
    p = np.exp(logits)
    ps = p.sum()
    if ps < 1e-300:
        p = np.zeros(2 * n - 1)
        p[n - 1] = 1.0
    else:
        p /= ps
    P = sfft.rfft(p, corr_size)
    Xc = sfft.rfft(x, corr_size)
    corr = np.real(sfft.irfft(P * np.conj(Xc), corr_size))
    g = 2.0 * corr[:n]
    g -= g.mean()
    gnorm = np.sqrt(np.dot(g, g))

        g *= min(1.0, 1.0 / gnorm) * 0.01
    mom[:] = beta * mom + (1.0 - beta) * g
    x_new = x - lr * mom
    x_new = np.clip(x_new, 0.0, 1000.0)
    return x_new, conv
