def _subgrad_direct(x, k_top, lr, fft_size, n):
    # Compute full convolution via FFT
    X = sfft.rfft(x, fft_size)
    conv = np.real(sfft.irfft(X * X, fft_size))[:2*n-1]
    conv = np.maximum(conv, 0)
    
    # Find top-k lags
    k_top = min(k_top, 2*n-1)
    top_indices = np.argsort(conv)[-k_top:]
    
    # Compute gradient as weighted sum of gradients at top lags
    g = np.zeros(n)
    for k in top_indices:
        # d(conv[k])/da[i] for all i
        valid = (k - np.arange(n) >= 0) & (k - np.arange(n) < n)
        idx = k - np.arange(n)
        contrib = np.where(valid, x[idx], 0)
        # Double count except when i == k-i
        diag = (np.arange(n) * 2 == k)
        contrib[diag] *= 0.5
        g += contrib
    
    # Step
    x_new = x - lr * g
    x_new = np.clip(x_new, 0, 1000)
    return x_new, conv
