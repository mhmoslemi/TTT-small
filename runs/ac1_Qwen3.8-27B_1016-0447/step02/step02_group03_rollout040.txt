    s = p_new.sum()

    p_new /= s

def _ac(p, N, n):
    return np.real(np.fft.irfft(np.fft.rfft(p, N) ** 2, N))[:2 * n - 1]


def _dual(y, p, N, n):
    Yf = np.fft.rfft(y, N)
    Pf = np.fft.rfft(p, N)
    return np.real(np.fft.irfft(Yf * np.conj(Pf), N))[:n]


def _prop_step(p, N, n, beta, alpha, mix):
    c = _ac(p, N, n)
    mx = c.max()
    y = np.exp(beta * (c - mx))
    ys = y.sum()
    if ys < 1e-300:
        return p, mx, False
    y /= ys
    d = _dual(y, p, N, n)
    d = np.maximum(d, 1e-30)
    p_new = d ** (-alpha)
    p_new = np.maximum(p_new, 0.0)
    s = p_new.sum()
    if s < 1e-30:
        return p, mx, False
    p_new /= s
    for t in (mix, mix * 0.5, mix * 0.2, mix * 0.1):
        p2 = (1.0 - t) * p + t * p_new
        p2 = np.maximum(p2, 0.0)
        s2 = p2.sum()

            continue
        p2 /= s2
        c2 = _ac(p2, N, n)
        if c2.max() < mx - 1e-16:
            return p2, c2.max(), True
    return p, mx, False


def _shave_step(p, N, n, beta):
