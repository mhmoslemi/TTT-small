def _normalize(a):

def _grad_from_p(a, p, fft_sz):
    n = a.size
    af = np.fft.rfft(a, fft_sz)
    prf = np.fft.rfft(p[::-1], fft_sz)
    ca = np.fft.irfft(af * prf, fft_sz)
    return 2.0 * ca[n - 1:2 * n - 1][::-1]


def _do_grad_step(a, fft_sz, tau, lr):
    n = a.size
    af = np.fft.rfft(a, fft_sz)
    b = np.fft.irfft(af * af, fft_sz)[:2 * n - 1]
