    # LP: max sum(b) s.t. (a*b)_k <= M for k in active_set, b >= 0
    # Decision variables: b (size n)
    # Constraints: (a*b)_k <= M for k in active_set
    #               -b_j <= 0 for j in 0..n-1
    
    # Build constraint matrix (sparse) for active_set
    # Each row k: nonzeros at positions max(0,k-n+1) to min(k, n-1)
    # with values a[j]
    
    from scipy.sparse import coo_matrix
    from scipy.optimize import linprog
    
    rows = []
    cols = []
    vals = []
    rhs = []
    
    for k in active_set:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            rows.append(len(rhs))
            cols.append(j)
            vals.append(a[j])
        rhs.append(M)
    
    # b >= 0: handled by bounds
    # -b <= 0: A_ub * b <= b_ub
    A_ub = coo_matrix((vals, (rows, cols)), shape=(len(rhs), n))
    b_ub = np.array(rhs)
    
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')

    fa = np.fft.rfft(a, sz)

def _solve_lp(a, M, active, n, tlimit):
    """Solve max sum(b) s.t. (a*b)_k <= M for k in active, 0 <= b_j <= B."""
    m = len(active)
    if m == 0:
        return np.ones(n, dtype=np.float64) / n
    
    rows = np.empty(m * n, dtype=np.int32)  # worst case
    cols = np.empty(m * n, dtype=np.int32)
    vals = np.empty(m * n, dtype=np.float64)
    
    # Build sparsely
    r_list = []
    c_list = []
    v_list = []
    for i, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            r_list.append(i)
            c_list.append(j)
            v_list.append(a[j])
    
    A = coo_matrix((v_list, (r_list, c_list)), shape=(m, n))
    
    B = 10.0 * M + 1.0  # upper bound for b_j
    
    res = linprog(
        -np.ones(n, dtype=np.float64),
        A_ub=A,
        b_ub=np.full(m, M, dtype=np.float64),
        bounds=[(0.0, B)] * n,
        method='highs',
        options={'time_limit': tlimit}
    )
    
    if res.status in (0, 3):  # optimal or time-limited with solution
        return np.maximum(res.x, 0.0)

    a is normalized (sum=1).
    """
    conv = _fft_conv(a, a)
    M = float(conv.max())
    
    # Initial active set: top constraints
    n_conv = len(conv)
    top_k_init = min(20, n_conv)
    active = list(np.argsort(conv)[-top_k_init:])
    
    b = None
    for it in range(15):
        if time.monotonic() > deadline - 2:
            break
        
        tlimit = min(10.0, max(1.0, (deadline - time.monotonic()) * 0.02))
        b = _solve_lp(a, M, active, n, tlimit)
        
        if b.sum() < 1e-30:
            break
        
        # Check all convolution constraints
        ab = _fft_conv(a, b)
        viol = np.where(ab > M + 1e-12)[0]
        
        if len(viol) == 0:
            break
        
        # Add violated constraints
        if len(viol) <= max_active - len(active):
            active = sorted(set(active) | set(viol.tolist()))
        else:
            # Keep most important violations
            all_idx = sorted(set(active) | set(viol.tolist()))
            if len(all_idx) > max_active:
                # Score by violation amount
                viol_set = set(viol.tolist())
                remaining = [k for k in all_idx if k in viol_set]
                all_idx = sorted(remaining, key=lambda k: -float(ab[k]))[:max_active]
            active = all_idx
    
    return b, M, active


def _do_search(a, n, deadline, rng):

    a = _normalize(a.astype(np.float64))
    best_a = a.copy()
    best_sc = _score(a)
    stall = 0

        else:

                    stall = 0
