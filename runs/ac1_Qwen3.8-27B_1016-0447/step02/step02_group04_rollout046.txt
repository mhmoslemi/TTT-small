import time
from scipy import fft as sfft


def _np2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def _fft_maxconv(x, fs):
    X = sfft.rfft(x, fs)
    return np.maximum(np.real(sfft.irfft(X * X, fs))[:2 * len(x) - 1], 0.0)


def _grad_from_weights(p, x, fs_g):
    Pf = sfft.rfft(p[::-1], fs_g)
    Xg = sfft.rfft(x, fs_g)
    r = np.real(sfft.irfft(Pf * Xg, fs_g))[:3 * len(x) - 2]
    return r[len(x) - 1:2 * len(x) - 1][::-1]


def _renorm(x, ts):
    sx = x.sum()
    if sx < 1e-10:
        return np.full(len(x), ts / len(x))
    return x * (ts / sx)
