from scipy import fft as sfft

def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def fft_conv(a, fs):
    A = sfft.rfft(a, fs)
    return np.maximum(sfft.irfft(A * A, fs).real[:2 * len(a) - 1], 0.0)

def subgrad(conv, a, m, work):
    n = len(a)
    kc = min(m, len(conv))
    idx = np.argpartition(conv, -kc)[-kc:]
    vals = conv[idx]
    vs = vals.sum()
    if vs < 1e-30:
        work[:] = 0.0
        return
    work[:] = 0.0
    for j in range(kc):
        kk = int(idx[j])
        w = 2.0 * float(vals[j]) / vs
        lo = kk - n + 1 if kk - n + 1 > 0 else 0
        hi = kk if kk < n else n - 1
        if lo <= hi:
            work[lo:hi + 1] += w * a[kk - hi:kk - lo + 1][::-1]
