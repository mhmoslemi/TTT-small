import numpy as np
import time
from scipy import fft as sfft

def npow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def sc(x):
    return float(evaluate_sequence(x.tolist()))

def autocorr(x, fs, n):
    X = sfft.rfft(x, fs)
    c = np.real(sfft.irfft(X * X, fs))[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def softmax_wts(c, T, K=64):
    cm = c.max()
    Kk = min(K, c.size)
    idx = np.argpartition(c, -Kk)[-Kk:]
    pv = np.exp((c[idx] - cm) / T)
    s = pv.sum()
    p = np.zeros(c.size)
    if s > 1e-300:
        p[idx] = pv / s
    else:
        p[int(np.argmax(c))] = 1.0
    return p

def xgrad(p, x, cs, n):
    P = sfft.rfft(p, cs)
    Xc = sfft.rfft(x, cs)
    g = 2.0 * np.real(sfft.irfft(P * np.conj(Xc), cs))[:n]
    g -= g.mean()
    return g

def normalize(x, ts):
    np.clip(x, 0.0, 1000.0, out=x)
    s = x.sum()
    if s < 0.01:
        x[:] = ts / x.size
    else:
        x *= ts / s
    np.clip(x, 0.0, 1000.0, out=x)
    return x

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    dl = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    try:
        par = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except Exception:
        par = np.ones(512.0)
    if par.ndim != 1 or par.size < 2 or not np.all(np.isfinite(par)):
        par = np.ones(512.0)

    n = par.size
    ts = np.sqrt(2.0 * n)
    fs = npow2(2 * n - 1)
    cs = npow2(3 * n - 2)

    x = par.copy()
    normalize(x, ts)
    best = x.copy()
    bsc = sc(best)
    if not np.isfinite(bsc):
        x = np.ones(n) * (ts / n)
        best = x.copy()
        bsc = sc(best)

    T = max(0.05, bsc * 0.10)
    Tm = 0.002
    lr0 = 0.4
    stuck = 0
    SL = 4000
    it = 0
    eval_interval = max(200, n // 5)

    while time.monotonic() < dl:
        if it > 0 and it % 200 == 0 and time.monotonic() > dl - 10.0:
            break

        c = autocorr(x, fs, n)
        p = softmax_wts(c, T)
        g = xgrad(p, x, cs, n)

        gn = np.linalg.norm(g)
        if gn > 1e-15:
            lr = lr0 * min(1.0, T / 0.05)
            x -= lr * (ts / n) * (g / gn)
        normalize(x, ts)
        T = max(Tm, T * 0.9994)

        if it > 0 and it % eval_interval == 0:
            s_ = sc(x)
            if np.isfinite(s_) and s_ < bsc - 1e-10:
                bsc = s_
                best = x.copy()
                stuck = 0
            else:
                stuck += eval_interval
            if stuck > SL:
                x = best.copy()
                noise = 0.03 * (ts / n) * rng.standard_normal(n)
                x += noise
                normalize(x, ts)
                T = max(0.05, bsc * 0.10)
                stuck = 0

        it += 1

    # Polish phase: exact subgradient with multiple peaks
    polish_iter = 0
    while polish_iter < 500 and time.monotonic() < dl - 3.0:
        c = autocorr(best, fs, n)
        mx = c.max()
        if mx < 1e-15:

    final = np.clip(best, 0.0, 1000.0)
    if final.sum() < 0.01:
        final = np.ones(n) * 1.0
    return [float(v) for v in final]
