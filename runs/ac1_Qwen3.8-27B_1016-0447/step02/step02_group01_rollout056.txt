import time
from scipy import fft as sfft

def npow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def rz(x, S):
    np.clip(x, 0.0, 1000.0, out=x)
    s = x.sum()
    if s > 1e-10:
        x *= S / s
    else:
        x[:] = 1.0
    np.clip(x, 0.0, 1000.0, out=x)
    return x

def ac(x, fs, n):
    X = sfft.rfft(x, fs)
    c = sfft.irfft(X * X, fs).real[:2*n-1]
    np.maximum(c, 0.0, out=c)
    return c

def grad_wn(w, x, cs, n):

    Xf = sfft.rfft(x, cs)

    return g
