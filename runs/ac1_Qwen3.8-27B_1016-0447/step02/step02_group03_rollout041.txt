    W[:2*GN-1] = w
    Pf = np.fft.rfft(p, GNF)

import numpy as np
import time

DEADLINE = 0.0
BEST_P = None
BEST_SC = float('inf')
GN = 0
GNF = 0

def conve(a, N):
    fa = np.fft.rfft(a, N)
    return np.real(np.fft.irfft(fa * fa, N)[:2 * a.size - 1])

def sc_of(p):
    return float(2.0 * GN * conve(p, GNF).max())

def upb(p):
    global BEST_P, BEST_SC

    if p.sum() < 0.01 or not np.isfinite(p).all():

    if np.isfinite(sc) and sc < BEST_SC:
        BEST_P = p.copy()
        BEST_SC = sc

def fft_grad(p, w):
    W = np.zeros(GNF)
    W[:2 * GN - 1] = w
    Pf = np.fft.rfft(p, GNF)
    Wf = np.fft.rfft(W)
    R = np.fft.irfft(Wf * np.conj(Pf), GNF)
    g = R[:GN].copy()
    g -= g.mean()
    return g

def descent(p, iters, mode):
    p = np.clip(np.asarray(p, dtype=np.float64).copy(), 0, None)
    if p.sum() < 1e-15:
        return p
    p /= p.sum()
    best_sc = sc_of(p)
    t = 0.03 if mode == 'peak' else 0.04
    stall = 0
    for _ in range(iters):
        if time.monotonic() > DEADLINE - 8:
            break
        c = conve(p, GNF)
        cm = c.max()

            break

            S = idx[c[idx] >= thr]
            if S.size == 0:

            w = np.zeros(2 * GN - 1)
            w[S] = 1.0 / S.size

        g = fft_grad(p, w)

                found = True

        if not found:

            stall += 1

def init_cands(n):
    L = []
    j = np.arange(n, dtype=np.float64)

    L.append(np.ones(n))

    return L
