    for _ in range(5):
        r = np.abs(rng.standard_normal(n))
        R = sfft.rfft(r, fs)
        fr = np.abs(np.fft.rfftfreq(fs))
        R *= np.exp(-fr * n * 0.02)
        r = np.abs(sfft.irfft(R, fs)[:n])
        inits.append(norm(r))
    
    # Parent + noise
    for sc in [0.05, 0.1, 0.2]:
        r = par + sc * max(par.std(), 0.01) * rng.standard_normal(n)
        inits.append(norm(r))
    
    # Score all
    scored = []
    for ini in inits:
        try:
            sc = _score(ini)
            if np.isfinite(sc) and sc < 100:
                scored.append((sc, ini))
        except:
            pass
    scored.sort(key=lambda t: t[0])
    
    if not scored:
        return [target_sum / n] * n
    
    best_sc = scored[0][0]
    best = scored[0][1].copy()
    
    # Main loop
    x = best.copy()
    T = 0.1
    lr = 0.4
    it = 0
    no_imp = 0
    
    while time.monotonic() < dl - 5.0:
        X = sfft.rfft(x, fs)
        conv = np.real(sfft.irfft(X * X, fs))[:2*n-1]
        cm = conv.max()
        
        # Score in normalized form (sum = target_sum, so (sum)^2 = 2n)
        sc = 2.0 * n * cm / (2.0 * n)  # = cm... wait
        # Actually score = 2n * max_b / sum_a^2 = 2n * cm / (2n) = cm
        # No wait: 2*n*cm / (sum_a)^2 where sum_a = target_sum = sqrt(2n)
        # So score = 2n * cm / (2n) = cm
        # Hmm, that means the score equals the max of the convolution when sum = sqrt(2n)?
        # Let me recheck: score = 2 * n * max_b / sum_a^2
        # sum_a = sqrt(2n), sum_a^2 = 2n
        # score = 2*n*cm / (2n) = cm
        # So for sum = sqrt(2n), the score IS the max convolution value!
        # That's convenient.
        
        sc = cm  # score when sum = sqrt(2n)
        
        if sc < best_sc - 1e-10:
            best_sc = sc
            best = x.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Softmax weights
        diff = conv - cm
        exps = np.exp(diff / T)
        ew = exps.sum()
        if ew < 1e-300:
            p = np.zeros(2*n-1)
            p[np.argmax(conv)] = 1.0
        else:
            p = exps / ew
