def entropy_step(x, g, eta):
    """Mirror descent step with entropy proximal operator."""
    log_x = np.log(np.maximum(x, 1e-300))
    log_x_next = log_x - eta * g
    log_x_next -= log_x_next.max()  # numerical stability
    x_next = np.exp(log_x_next)
    x_next /= x_next.sum()
    np.minimum(x_next, 1000.0, out=x_next)
    return x_next
