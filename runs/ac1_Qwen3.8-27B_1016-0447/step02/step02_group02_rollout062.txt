def lfgs_obj_grad(z, fs, rev_idx, T, n):
    z = np.asarray(z, dtype=np.float64).reshape(-1)
    if z.size != n:
       z = np.zeros(n)
    z = np.clip(z, -60.0, 60.0)
    zm = z - np.max(z)
    e = np.exp(zm)
    s = e.sum()
    if s <= 1e-300:
       y = np.ones(n, dtype=np.float64)
    else:
       y = float(n) * e / s
    A = sfft.rfft(y, fs)
    C = sfft.irfft(A*A, fs).real[:2*n-1]
    C = np.maximum(C, 0.0)
    mx = float(C.max())
    if mx <= 1e-300:
       return 0.0, np.zeros(n, dtype=np.float64)
    T = float(max(T, 1e-3))
    sh = (C - mx) / T
    # clip sh to avoid issues? lower no problem; upper maybe 0. If nan?
    expsh = np.exp(sh)
    ess = float(expsh.sum())
    if not np.isfinite(ess) or ess <= 0.0:
       # fallback one-hot at max
       w = np.zeros_like(C)
       w[int(np.argmax(C))] = 1.0
       ess = 1.0
    else:
       w = expsh / ess
    val = float(mx + T * np.log(ess))
    B = sfft.rfft(w[::-1], fs)
    G = sfft.irfft(A * B, fs).real
    Gy = 2.0 * G[rev_idx]
    dot = float(np.dot(Gy, y))
    Gz = Gy * y - (dot / float(n)) * y
    if not np.isfinite(val) or not np.all(np.isfinite(Gz)):
       return 1e100, np.zeros(n, dtype=np.float64)
    return val, Gz

import numpy as np
import time
from scipy import fft as sfft

try:
    from scipy.optimize import minimize
except Exception:
    minimize = None


def next_pow2(m):
    m = max(1, int(m))
    p = 1
    while p < m:
        p <<= 1
    return p


def fft_conv(y, fs):
    y = np.asarray(y, dtype=np.float64)
    A = sfft.rfft(y, fs)
    C = sfft.irfft(A * A, fs).real[:2 * len(y) - 1]
    return np.maximum(C, 0.0)


def fft_score(y, fs, n):
    return 2.0 * float(fft_conv(y, fs).max()) / float(n)


def z_to_y(z, n):
    z = np.asarray(z, dtype=np.float64).reshape(-1)
    if z.size != n:
        return np.ones(n, dtype=np.float64)
    z = np.clip(z, -40.0, 40.0)
    zm = z - np.max(z)
    e = np.exp(zm)
    s = e.sum()
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64)
    return float(n) * e / s


def y_to_z(y, n):
    y = np.asarray(y, dtype=np.float64).reshape(-1)
    if y.size != n:
        return np.zeros(n, dtype=np.float64)
    return np.log(np.maximum(y, 1e-12))


def make_pattern_y(kind, n, rng):
    t = np.arange(n, dtype=np.float64) / float(max(n, 1))
    y = np.ones(n, dtype=np.float64)
    if kind == 1:
        y += 0.65 * np.sin(2.0 * np.pi * t)
        y += 0.28 * np.sin(4.0 * np.pi * t - 0.6)
        y += 0.12 * np.sin(6.0 * np.pi * t - 1.2)
        y += 0.05 * np.sin(10.0 * np.pi * t + 0.3)
    elif kind == 2:
        for k in range(1, 13):
            amp = 0.55 / np.sqrt(float(k))
            ph = rng.uniform(0.0, 6.28318)
            y += amp * np.sin(2.0 * np.pi * k * t + ph)
    elif kind == 3:
        y += 0.55 * np.sin(2.0 * np.pi * t)
        y += 0.30 * np.sin(6.0 * np.pi * t - 1.1)
        y += 0.18 * np.sin(10.0 * np.pi * t - 0.4)
        y += 0.10 * np.sin(14.0 * np.pi * t + 0.9)
    y = np.maximum(y, 1e-6)
    s = y.sum()
    if s <= 1e-12:

    return y * (float(n) / s)

    try:
        return float(evaluate_sequence((y / float(n)).tolist()))
    except Exception:

def lfgs_obj_grad(z, fs, rev_idx, T, n):
    z = np.asarray(z, dtype=np.float64).reshape(-1)
    if z.size != n:
        z = np.zeros(n, dtype=np.float64)
    z = np.clip(z, -40.0, 40.0)
    zm = z - np.max(z)

        y = np.ones(n, dtype=np.float64)
    else:
        y = float(n) * e / s

    T = max(1e-3, float(T))
    sh = (C - mx) / T
    expsh = np.exp(sh)
    ess = float(expsh.sum())
    if not np.isfinite(ess) or ess <= 0.0:
        w = np.zeros_like(C)
        w[int(np.argmax(C))] = 1.0
        ess = 1.0
    else:
        w = expsh / ess

    val = float(mx + T * np.log(ess))

    B = sfft.rfft(w[::-1], fs)
    G = sfft.irfft(A * B, fs).real
    Gy = 2.0 * G[rev_idx]

    dot = float(np.dot(Gy, y))
    Gz = Gy * y - (dot / float(n)) * y

    if not np.isfinite(val) or not np.all(np.isfinite(Gz)):
        return 1e100, np.zeros(n, dtype=np.float64)
    return val, Gz


def run_lfgs_step(z, fs, rev_idx, T, n, maxfun, bounds):
    if minimize is None:
        return z
    opts = {
        "maxiter": 100,
        "maxfun": int(max(10, maxfun)),
        "ftol": 1e-12,
        "gtol": 1e-10,
        "disp": False,
        "maxls": 30,
    }
    try:
        res = minimize(
            lfgs_obj_grad,
            z,
            method="L-BFGS-B",
            jac=True,
            args=(fs, rev_idx, T, n),
            bounds=bounds,
            options=opts,
        )
    except Exception:
        return z
    x = np.asarray(res.x, dtype=np.float64).reshape(-1)
    if x.size != n or not np.all(np.isfinite(x)):
        return z
    x = x - np.mean(x)
    return x


def manual_polish(y, fs, n, end_time, seed=123):
    rng = np.random.default_rng(seed)
    y = np.asarray(y, dtype=np.float64).copy()
    mo = np.zeros(n, dtype=np.float64)
    cv = np.zeros(n, dtype=np.float64)
    work = np.zeros(n, dtype=np.float64)
    C0 = fft_conv(y, fs)
    fbest = 2.0 * float(C0.max()) / float(n)
    fbest_y = y.copy()
    lr = 0.04
    no_imp = 0

    while time.monotonic() < end_time:
        C = fft_conv(y, fs)
        cmax = float(C.max())
        sc = 2.0 * cmax / float(n)

        if sc < fbest - 1e-10:
            fbest = sc
            fbest_y = y.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 200:
                lr = max(1e-4, lr * 0.85)
                if no_imp > 1200:
                    y = fbest_y + 0.05 * rng.normal(size=n)
                    y = np.maximum(y, 1e-9)
                    y *= float(n) / y.sum()
                    mo[:] = 0.0
                    cv[:] = 0.0
                    no_imp = 0
                    lr = 0.04

        k = 1 if sc < 1.55 else 3
        if k > 1 and k < C.size:
            chosen = np.argpartition(C, -k)[-k:]
        else:
            chosen = np.array([int(np.argmax(C))], dtype=int)

        vals = C[chosen]
        vs = float(vals.sum())
        if vs <= 1e-30:
            break
        ws = vals / vs
        work[:] = 0.0

        for jj in range(chosen.size):
            kk = int(chosen[jj])
            w = 2.0 * float(ws[jj])
            lo = kk - (n - 1)
            if lo < 0:
                lo = 0
            hi = kk
            if hi > n - 1:
                hi = n - 1
            if lo <= hi:
                work[lo:hi + 1] += w * y[kk - hi:kk - lo + 1][::-1]

        G = work
        G -= G.mean()
        gn = float(np.linalg.norm(G))
        if gn < 1e-14:
            break

        mo = 0.9 * mo + 0.1 * G
        cv = 0.999 * cv + 0.001 * G * G
        y = y - lr * mo / (np.sqrt(cv) + 1e-7)
        y = np.maximum(y, 1e-12)
        s = y.sum()
        if s <= 1e-12:
            y = fbest_y.copy()
        else:
            y *= float(n) / s

    return y, fbest_y


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(int(seed))
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 30.0)

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(1024, dtype=np.float64)

    if parent.size == 0:
        return [1.0]

    parent = np.clip(parent, 0.0, 1000.0)
    if not np.all(np.isfinite(parent)):
        parent = np.ones(parent.size, dtype=np.float64)

    n = int(parent.size)
    if n < 1:
        return [1.0]

    s = float(parent.sum())
    if s < 1e-12:
        y0 = np.ones(n, dtype=np.float64)
    else:
        y0 = parent * (float(n) / s)
    y0 = np.maximum(y0, 1e-12)
    y0 *= float(n) / y0.sum()

    fs = next_pow2(3 * n - 2)
    if n == 1:
        rev_idx = np.array([0], dtype=int)
    else:
        rev_idx = np.arange(2 * n - 2, n - 2, -1, dtype=int)

    exact_y = y0.copy()
    best_exact = exact_score_y(exact_y, n)
    if not np.isfinite(best_exact):
        best_exact = 10.0

    search_y = y0.copy()
    best_fft = fft_score(search_y, fs, n)
    exact_calls = 0

    starts = [y_to_z(search_y, n)]
    starts.append(y_to_z(make_pattern_y(1, n, rng), n))
    starts.append(y_to_z(make_pattern_y(2, n, rng), n))
    starts.append(y_to_z(make_pattern_y(3, n, rng), n))
    starts.append(np.zeros(n, dtype=np.float64))

    bounds = [(-40.0, 40.0)] * n
    T_list = [
        max(2.0, float(n) / 8.0),
        max(2.0, float(n) / 32.0),
        max(2.0, float(n) / 128.0),
        max(2.0, float(n) / 512.0),
    ]
    last_phase = len(T_list) - 1

    for ti, T in enumerate(T_list):
        if time.monotonic() > deadline - 60.0:
            break
        for si in range(len(starts)):
            if time.monotonic() > deadline - 50.0:
                break
            if ti >= 2 and si >= 2 and time.monotonic() > deadline - 150.0:
                break

            if ti == 0:
                base_mf = 80
                max_runs = 1
            elif ti == 1:
                base_mf = 120
                max_runs = 2 if si == 0 else 1
            else:
                if si == 0:
                    base_mf = 180
                    max_runs = 4
                elif si == 1:
                    base_mf = 100
                    max_runs = 1
                else:
                    if time.monotonic() > deadline - 120.0:
                        break
                    base_mf = 60
                    max_runs = 1

            z = starts[si]
            for _ in range(max_runs):
                if time.monotonic() > deadline - 50.0:
                    break
                rem = deadline - time.monotonic()
                mf = base_mf
                if rem < 90.0:
                    mf = max(20, mf // 4)
                elif rem < 240.0:
                    mf = max(30, mf // 2)

                z = run_lfgs_step(z, fs, rev_idx, T, n, mf, bounds)
                z = z - np.mean(z)
                starts[si] = z
                y = z_to_y(z, n)

                scf = fft_score(y, fs, n)
                if scf < best_fft - 1e-10:
                    best_fft = scf
                    search_y = y.copy()
                    starts[0] = y_to_z(search_y, n)

                    if exact_calls < 35 and scf < best_exact - 1e-5:
                        exact_calls += 1
                        ex = exact_score_y(y, n)
                        if np.isfinite(ex) and ex < best_exact:
                            best_exact = ex
                            exact_y = y.copy()
                            search_y = exact_y.copy()
                            starts[0] = y_to_z(search_y, n)

                if time.monotonic() > deadline - 80.0:
                    break

    if time.monotonic() < deadline - 120.0:
        zb = y_to_z(search_y, n)
        T = max(2.0, float(n) / 1024.0)
        for _ in range(3):
            if time.monotonic() > deadline - 100.0:
                break
            rem = deadline - time.monotonic()
            mf = 150 if rem > 180.0 else 60
            zb = run_lfgs_step(zb, fs, rev_idx, T, n, mf, bounds)
            zb = zb - np.mean(zb)
            y = z_to_y(zb, n)
            scf = fft_score(y, fs, n)
            if scf < best_fft - 1e-10:
                best_fft = scf
                search_y = y.copy()
            if exact_calls < 45 or True:
                ex = exact_score_y(y, n)
                if np.isfinite(ex) and ex < best_exact:
                    best_exact = ex
                    exact_y = y.copy()
                    search_y = exact_y.copy()

    ex = exact_score_y(exact_y, n)
    if np.isfinite(ex) and ex < best_exact:
        best_exact = ex
    ex = exact_score_y(search_y, n)
    if np.isfinite(ex) and ex < best_exact:
        best_exact = ex
        exact_y = search_y.copy()

    if time.monotonic() < deadline - 60.0:
        base = search_y.copy()
        end_polish = deadline - 20.0
        y_now, y_fbest = manual_polish(base, fs, n, end_polish, int(seed) + 17)
        for cand in (y_now, y_fbest, exact_y, search_y):
            c = np.asarray(cand, dtype=np.float64).copy()
            if c.sum() <= 0.0:
                continue
            c = np.maximum(c, 1e-12)
            c *= float(n) / c.sum()
            ex = exact_score_y(c, n)
            if np.isfinite(ex) and ex < best_exact:
                best_exact = ex
                exact_y = c.copy()

    best = exact_y
    if best is None:
        best = y0.copy()
    best = np.asarray(best, dtype=np.float64).reshape(-1)
    if best.size != n or not np.all(np.isfinite(best)):
        best = y0.copy()

    x = best / float(n)
    x = np.clip(x, 0.0, 1000.0)
    if x.sum() < 0.01:
        return [1.0]
    return [float(v) for v in x]
