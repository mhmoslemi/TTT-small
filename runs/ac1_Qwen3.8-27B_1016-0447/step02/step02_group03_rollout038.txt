def _peak_shave(p, niter):
    """Direct subgradient peak shaving."""
    global _DL, _BEST, _BEST_SC
    p = p.copy()
    cur_sc = _score(p)
    for _ in range(niter):
        if time.monotonic() > _DL - 10:
            break
        c = _conv_fft(p, _FN)
        pk = c.max()
        if pk < 1e-18:
            break
        # Find all peaks within 5% of max
        thr = 0.97 * pk
        peaks = np.where(c >= thr)[0]
        peaks = peaks[:20]  # limit to 20 peaks
        npeaks = len(peaks)
        if npeaks == 0:
            break
        # weights proportional to peak values
        wts = c[peaks].copy()
        wts /= wts.sum()
        # Compute gradient
        g = np.zeros(_N)
        for i in range(npeaks):
            k = int(peaks[i])
            wi = wts[i]
            # g[m] += wi * (a[k-m] + a[m-k]) for valid indices
            # First term: a[k-m] valid when 0 <= k-m < n => k-n+1 <= m <= k
            lo1 = max(0, k - _N + 1)
            hi1 = min(_N - 1, k)
            if hi1 >= lo1:
                g[lo1:hi1+1] += wi * p[k-hi1:k-lo1+1][::-1]
            # Second term: a[m-k] valid when 0 <= m-k < n => k <= m <= k+n-1
            lo2 = max(0, k)
            hi2 = min(_N - 1, k + _N - 1)
            if hi2 >= lo2:
                g[lo2:hi2+1] += wi * p[lo2-k:hi2-k+1]
        g *= 2.0 * _N
        # Project onto tangent
        d = p @ g
        gv = p * (g - d)
        # Line search
        found = False
        for t in (0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001):
            p2 = p - t * gv
            mn = p2.min()
            if mn < 0:
                p2 = np.maximum(p2, 0)
                s2 = p2.sum()
                if s2 < 1e-15:
                    break
                p2 /= s2
            else:
                p2 /= p2.sum()
            sc2 = _score(p2)
            if sc2 < cur_sc - 1e-14:
                p = p2
                cur_sc = sc2
                found = True
                break
        if not found:
            break
    _update_best(p)
    return p

def _inits(n):
    """Generate initial candidates."""
    L = []
    j = np.arange(n, dtype=np.float64)
    m = 0.5*(n-1)
    # Uniform
    L.append(np.ones(n))
    # Cosine taper
    L.append(0.5*(1+np.cos(np.pi*j/(n-1))))
    # Inverse cosine taper
    L.append(0.5*(1-np.cos(np.pi*j/(n-1))) + 0.01)
    # Power laws
    for p in (-0.3, -0.5, -0.8, -1.2, -1.8, -2.5, -3.0):
        L.append((j+1)**p)
    for p in (0.3, 0.5, 0.8, 1.2, 1.8):
        L.append((n-1-j+1)**p)
    # Bump functions
    for w in (0.5, 1.0, 1.5, 2.5, 4.0, 8.0):
        L.append(np.exp(-w*(j-m)/(m+1))**2)
    # Two bumps
    t1, t2 = 0.3*n, 0.7*n
    sd = 0.15*n
    L.append(np.exp(-((j-t1)/sd)**2) + np.exp(-((j-t2)/sd)**2) + 0.01)
    # Distance from edges
    d = np.minimum(j, n-1-j) + 1.0
    for pv in (0.5, 1.0, 1.5, 2.0, 3.0):
        L.append(d**pv)
    for pv in (0.5, 1.0, 2.0):
        L.append(1.0/d**pv)
    # Sinusoidal modulation
    for freq in (1, 2, 3, 5):
        L.append(0.5*(1+np.cos(np.pi*j/(n-1))) * (1+0.3*np.cos(2*np.pi*freq*j/n)) + 0.01)
    return L

def _lse_run(z0, n_beta=8):
    """Run LSE optimization with increasing beta."""
    global _GF, _DL
    z = z0.copy()
    z = z - z.mean()
    z = np.clip(z, -25, 25)
    vb = _lse_val(z)
    betas = (4, 8, 16, 32, 64, 128, 256, 512, 1024, 2048, 4096, 8192)
    maxits = (30, 40, 50, 60, 80, 100, 120, 150, 200, 250, 300, 350)
    for i in range(min(n_beta, len(betas))):
        if time.monotonic() > _DL - 30:
            break
        _GF = float(betas[i])
        mi = maxits[i]
        try:
            r = sopt.minimize(_lse_val, z, jac=_lse_grad, method='L-BFGS-B',
                            options={'maxiter': mi, 'ftol': 1e-15, 'gtol': 1e-10, 'maxls': 30})
            if r is not None and r.x is not None and np.isfinite(r.fun):
                if r.fun <= vb + 1e-8:
                    z = np.asarray(r.x)
                    vb = float(r.fun)
        except Exception:
            pass
        p = np.exp(z - z.max())
        p /= p.sum()
        p = np.clip(p, 0, 1000)
        if p.sum() > 0.01:
            _update_best(p)
    return z

def _diversify(p, rng, k=3):
    """Generate diversified restart candidates."""
    L = [p.copy()]
    # Symmetrize
    sym = (p + p[::-1]) / 2.0
    L.append(sym)
    # Random perturbation in log space
    for _ in range(k):
        z = np.log(p + 1e-20)
        z += rng.normal(0, 0.1, size=z.size)
        q = np.exp(z - z.max())
        q /= q.sum()
        q = np.clip(q, 0, 1000)
        if q.sum() > 0.01:
            L.append(q)
    return L
