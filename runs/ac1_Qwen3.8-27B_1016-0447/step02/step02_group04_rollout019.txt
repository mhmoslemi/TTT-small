from scipy import fft as sfft


def _np2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _eval(v):

def _norm1(x):
    s = x.sum()
    if s < 1e-15:
        return np.ones_like(x) / x.size
    return x / s


def _fft_conv(x, fs):
    X = sfft.rfft(x, fs)
    c = np.real(sfft.irfft(X * X, fs))[:2 * len(x) - 1]
    return np.maximum(c, 0.0)


def _grad_peaks(x, conv, topk, fs):
    m = 2 * len(x) - 1
    if topk >= m:
        w = np.ones(m) / m
    else:
        idx = np.argpartition(conv, -topk)[-topk:]
        w = np.zeros(m)
        w[idx] = 1.0 / topk
    wr = w[::-1]
    W = sfft.rfft(wr, fs)
    X = sfft.rfft(x, fs)
    gf = np.real(sfft.irfft(W * X, fs))
    g = gf[len(x) - 1:2 * len(x) - 1][::-1]
    g -= g.mean()
    return g


def _soft_grad(x, conv, T, fs):
    m = 2 * len(x) - 1
    cm = conv.max()
    ex = np.exp((conv - cm) / T)
    es = ex.sum()
    if es < 1e-300:
        w = np.zeros(m)
        w[np.argmax(conv)] = 1.0
    else:
        w = ex / es
    wr = w[::-1]
    W = sfft.rfft(wr, fs)
    X = sfft.rfft(x, fs)
    gf = np.real(sfft.irfft(W * X, fs))
    g = gf[len(x) - 1:2 * len(x) - 1][::-1]
    g -= g.mean()
    return g


def _step_proj(x, k, shift):
