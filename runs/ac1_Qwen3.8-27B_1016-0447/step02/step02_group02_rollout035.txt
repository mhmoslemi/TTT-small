    A = sfft.rfft(a, fs)
    out = sfft.irfft(D * np.conj(A), fs).real[:n]
    return out


def top1_grad(conv, a):
    n = len(a)
    idx = int(np.argmax(conv))
    lo = max(0, idx - n + 1)
    hi = min(idx, n - 1)
    g = np.zeros(n)
    if lo <= hi:
        g[lo:hi + 1] = 2.0 * a[idx - hi:idx - lo + 1][::-1]
    return g


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 30.0)

    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = parent.size
    if n < 2 or parent.sum() < 0.01 or not np.isfinite(parent).all():
        return [1.0]

    fs = next_pow2(3 * n - 2)
    x = parent / parent.sum()

    conv = fft_conv(x, fs)
    best = x.copy()
    best_score = 2.0 * n * conv.max()
    sc0 = float(evaluate_sequence(best.tolist()))
    if np.isfinite(sc0):
        best_score = min(best_score, sc0)

    # Phase 1: smooth Lp gradient descent (increasing p for sharpening)
    p = 2.0
    step_frac = 0.03
    no_imp = 0
    IMP_LIM = 2500
    it = 0

    while time.monotonic() < deadline - 130.0:
        conv = fft_conv(x, fs)
        cur = 2.0 * n * conv.max()

        grad = fft_smooth_grad(conv, x, p, fs)
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-18:
            break

        x = np.maximum(x - step_frac * x.mean() * grad / gn, 0.0)
        s = x.sum()
        if s < 1e-14:
            x = best.copy()
        else:
            x = x / s

        it += 1
        if cur < best_score - 1e-13:
            best_score = cur
            best = x.copy()
            no_imp = 0
        else:
            no_imp += 1

        p = min(24.0, p + 0.02)

        if no_imp >= IMP_LIM:
            scale = max(float(np.std(best)), float(np.mean(best)))
            x = np.maximum(best + 0.06 * scale * rng.standard_normal(n), 1e-12)
            x = x / x.sum()
            no_imp = 0
            step_frac = max(0.005, step_frac * 0.82)
            p = max(2.0, p * 0.6)

        if it % 5000 == 0:
            sc = float(evaluate_sequence(best.tolist()))
            if sc < best_score:
                best_score = sc
            if best_score <= 1.5028:
                break

    # Phase 2: top-1 subgradient polish
    if time.monotonic() < deadline - 50.0:
        x = best.copy()
        step = 0.008
        for _ in range(200000):
            if time.monotonic() > deadline - 50.0:
                break
            conv = fft_conv(x, fs)
            c2 = 2.0 * n * conv.max()
            if c2 < best_score - 1e-14:
                best_score = c2
                best = x.copy()
            g = top1_grad(conv, x)
            g -= g.mean()
            gn = np.linalg.norm(g)
            if gn < 1e-18:
                break
            x = np.maximum(x - step * x.mean() * g / gn, 0.0)
            s = x.sum()
            if s > 1e-14:
                x = x / s

    # Phase 3: ultra-fine top-1 polish
    if time.monotonic() < deadline - 5.0:
        x = best.copy()
        step = 0.002
        for _ in range(200000):
            if time.monotonic() > deadline - 5.0:
                break
            conv = fft_conv(x, fs)
            c2 = 2.0 * n * conv.max()
            if c2 < best_score - 1e-14:
                best_score = c2
                best = x.copy()
            g = top1_grad(conv, x)
            g -= g.mean()
            gn = np.linalg.norm(g)
            if gn < 1e-18:
                break
            x = np.maximum(x - step * x.mean() * g / gn, 0.0)
            s = x.sum()
            if s > 1e-14:
                x = x / s

    sc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(sc):
        pn = parent / parent.sum()
        sc2 = float(evaluate_sequence(pn.tolist()))
        if np.isfinite(sc2):
            return [float(v) for v in pn]
        return [1.0]
    return [float(v) for v in best]
