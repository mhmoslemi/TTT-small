import numpy as np
import time
from scipy import fft as sfft


def _np2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _ev(v):
    return float(evaluate_sequence(v.tolist()))


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t_dead = time.monotonic() + max(0.0, float(budget_s) - 20.0)

    par = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if par.ndim != 1 or par.size == 0 or not np.all(np.isfinite(par)):
        par = np.ones(1024)
    n = par.size
    fs = _np2(3 * n)
    jj = np.arange(n, dtype=np.float64)
    clen = 2 * n - 1

    x = np.maximum(par, 0.0).copy()
    if x.sum() < 1e-10:
        x = np.ones(n) / n
    else:
        x = x / x.sum()
    best = x.copy()
    bs = _ev(best)

    # Perturbation search for better init
    for _ in range(30):
        t = par.astype(np.float64).copy()
        f1 = rng.uniform(0.5, 30.0)
        f2 = rng.uniform(0.5, 20.0)
        t += rng.uniform(0.03, 0.30) * np.sin(2.0 * np.pi * f1 * jj / n + rng.uniform(0.0, 6.283))
        t += rng.uniform(0.02, 0.15) * np.cos(2.0 * np.pi * f2 * jj / n + rng.uniform(0.0, 6.283))
        t = np.maximum(t, 0.0)
        st = t.sum()

        else:

        sc = _ev(t)
        if np.isfinite(sc) and sc < bs:

    x = best.copy()

    # Main gradient optimization with momentum
    lr = 0.35
    mu = 0.82
    v = np.zeros(n)
    it = 0
    gap = 0
    last_eval = 0

    while time.monotonic() < t_dead - 30.0:
        X = sfft.rfft(x, fs)
        conv = np.maximum(sfft.irfft(X * X, fs).real[:clen], 0.0)

        if cm < bs - 1e-12:
            bs = cm
            best = x.copy()
            gap = 0
        else:

        # Periodic exact eval
        if it - last_eval > 5000:
            es = _ev(x)
            if np.isfinite(es) and es < bs:

                best = x.copy()
                gap = 0
            last_eval = it

        # Top-k power-weighted importance
        frac = 0.12 + 0.10 * (1.0 - min(1.0, float(it) / 60000))
        k = max(4, int(frac * clen))
        thr = np.partition(conv, clen - 1 - k)[clen - 1 - k]

        final = _ev(best)
    if not np.isfinite(final):

    return best.tolist()
