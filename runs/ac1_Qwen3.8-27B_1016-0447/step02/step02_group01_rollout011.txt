```python
def simplex_proj(x, c):
    n = x.size
    u = np.sort(x)[::-1]
    css = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (css - c))[0][-1]
    theta = (css[rho] - c) / (rho + 1)
    return np.maximum(x - theta, 0)
```

This is O(n log n) per projection due to sorting. For n=4498, that's fast.

from scipy import fft as sfft

def npow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def autocorr(x, fs, n):
    X = sfft.rfft(x, fs)
    c = np.real(sfft.irfft(X * X, fs))[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def peak_wts(c, T, K):
    K = min(K, c.size)
    idx = np.argpartition(c, -K)[-K:]
