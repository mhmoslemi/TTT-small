                ii = rng.integers(0, n, size=npx)
                best[ii] += 0.006 * (S / n) * rng.standard_normal(npx)
                np.clip(best, 0.0, 1000.0, out=best)
                s = best.sum()
                if s > 0.01:
                    best *= S / s
                    np.clip(best, 0.0, 1000.0, out=best)
            else:
                break

    fsc = evaluate_sequence(best.tolist())
    if not np.isfinite(fsc):
        fx = np.clip(a, 0.0, 1000.0)
        if fx.sum() < 0.01:
            fx = np.full(n, S / n)
        best = fx
    return [float(v) for v in best]
