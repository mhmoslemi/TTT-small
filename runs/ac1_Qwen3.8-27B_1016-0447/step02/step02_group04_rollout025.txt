def _make_step(k, n, heights):

    X = sfft.rfft(x, fs)
    c = np.real(sfft.irfft(X * X, fs))[:2*n-1]

def _grad_fft(x, p, n, fs):
    """Gradient of max(p * conv) wrt x, using FFT."""
    X = sfft.rfft(x, fs)
    pr = p[::-1]
    P = sfft.rfft(pr, fs)
    gr = np.real(sfft.irfft(P * X, fs))
    g = gr[n-1:2*n-1][::-1].copy()
    g -= g.mean()
    return g
