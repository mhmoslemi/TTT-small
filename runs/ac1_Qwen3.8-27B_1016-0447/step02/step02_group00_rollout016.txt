        # Multiplicative update

        if gm > 1e-30:
            a *= np.exp(-lr * grad / gm)
        a = np.maximum(a, 1e-15)
        a /= a.sum()
        
        it += 1
        if it % 50 == 0:
            tau *= 0.99  # cool down (harder max)
            if tau < 0.5/n:
                tau = 0.5/n
            lr *= 0.995
        
        if it % eval_every == 0:
            sc = _score(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best = a.copy()
                stalled = 0
            else:
                stalled += 1
            if stalled >= 10:
                # Perturb
                mask = rng.random(n) < 0.2
                a[mask] *= np.exp(0.05 * rng.standard_normal(mask.sum()))
                a = _normalize(a)
                stalled = 0
    
    sc = _score(a)
    if sc < best_score:
        best = a.copy()
        best_score = sc
    return best, best_score


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 10.0)
    
    # Get parent
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0).flatten()
    except Exception:
        parent = None
    
    if parent is not None and parent.size >= 2 and np.isfinite(parent).all():
        n = parent.size
    else:
        n = 2535
        parent = np.ones(n)
    
    parent = np.maximum(parent, 0.0)
    if parent.sum() < 1e-10:
        parent = np.ones(n)
    parent = parent / parent.sum()
    
    # Generate initializations
    j = np.arange(n, dtype=np.float64)
    x = j / (n - 1.0)
    
    inits = []
    inits.append(parent.copy())
    inits.append(np.ones(n))  # uniform (theoretically optimal)
    inits.append((j + 1.0) / (j + 1.0).sum())
    inits.append((n - j) / (n - j).sum())
    inits.append(np.exp(2.0 * x) / np.exp(2.0 * x).sum())
    inits.append((1 + 0.5*x) / (1 + 0.5*x).sum())
    inits.append(np.exp(-2.0 * (x - 0.5)**2) / np.exp(-2.0 * (x - 0.5)**2).sum())
    
    # Score all inits
    scored = []
    for ia in inits:
        ia = _normalize(ia)
        sc = _score(ia)
        if np.isfinite(sc):
            scored.append((sc, ia))
    
    scored.sort(key=lambda t: t[0])
    
    if not scored:
        return [float(1.0/n)] * n
    
    best_all = scored[0][1].copy()
    best_sc = scored[0][0]
    
    # Optimize top candidates
    rem = max(deadline - time.monotonic(), 1.0)
    top_k = min(4, len(scored))
    time_per = rem / (top_k + 1)  # +1 for final polish
    
    for k in range(top_k):
        if time.monotonic() >= deadline - 30:
