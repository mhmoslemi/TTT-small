    for i in range(K):
        idx = top[i]
        lo = max(0, idx - n + 1)
        hi = min(n - 1, idx)
        s1 = idx - hi
        s2 = idx - lo
        g[lo:hi + 1] += 2.0 * w[i] * x[s1:s2 + 1][::-1]
    g -= np.mean(g)
    return g


def _normalize(x, n):
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s < 0.01:
        x = np.full(n, 1.0 / n)
    else:
        x = x * n / s
    return np.clip(x, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 18.0)

    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.ndim != 1 or parent.size == 0:
        parent = np.ones(4498)
    n = parent.size
    if n < 2 or parent.sum() < 0.01:
        parent = np.ones(n if n >= 2 else 4498)
        n = parent.size

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(best_score):
        best = np.ones(n)
        best_score = float(evaluate_sequence(best.tolist()))

    x = _normalize(parent, n)
    fft_size = _next_pow2(2 * n)
    conv = _fft_conv(x, fft_size)[:2 * n - 1]

    lr = 0.12
    K = 6
    T = 0.8
    no_improve = 0
    max_no_improve = 25000
    restart_count = 0
    last_eval_time = time.monotonic()
    step = 0
    best_T = T
    best_K = K

    while time.monotonic() < deadline - 20.0:
        g = _topk_gradient(x, conv, K, T)
        x = np.maximum(x - lr * g, 0.0)
        x = _normalize(x, n)
        step += 1

        if step % 30 == 0:
            conv = _fft_conv(x, fft_size)[:2 * n - 1]

        if time.monotonic() - last_eval_time > 0.6:
            last_eval_time = time.monotonic()
            sc = float(evaluate_sequence(x.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best = x.copy()
                no_improve = 0
                best_T = T
                best_K = K
            else:
                no_improve += 80

            T = max(0.05, T * 0.98)
            if step > 100000:
                T = max(0.02, T * 0.995)

            if no_improve > max_no_improve and restart_count < 20:
                restart_count += 1
                noise_scale = 0.03 + 0.04 * restart_count
                x = best + noise_scale * (best.std() + 0.1) * rng.standard_normal(n)
                x = _normalize(x, n)
                conv = _fft_conv(x, fft_size)[:2 * n - 1]
                T = max(0.2, 0.4 + 0.05 * restart_count)
                K = min(12, best_K + restart_count % 3)
                no_improve = 0

        if step % 20000 == 0 and step > 0:
            x += 0.003 * rng.standard_normal(n)
            x = _normalize(x, n)

    # Final polish phase
    if time.monotonic() < deadline - 25.0:
        lr2 = 0.04
        T2 = 0.03
        K2 = 3
        for _ in range(20000):
            if time.monotonic() > deadline - 25.0:
                break
            g = _topk_gradient(x, conv, K2, T2)
            x = np.maximum(x - lr2 * g, 0.0)
            x = _normalize(x, n)
            step += 1
            if step % 15 == 0:
                conv = _fft_conv(x, fft_size)[:2 * n - 1]

    sc_final = float(evaluate_sequence(x.tolist()))
    if np.isfinite(sc_final) and sc_final < best_score:
        best = x.copy()

    final = np.clip(best, 0.0, 1000.0)
    fs = float(evaluate_sequence(final.tolist()))
    if not np.isfinite(fs) or fs > best_score + 0.1:
        final = np.clip(parent, 0.0, 1000.0)

    return [float(v) for v in final]
