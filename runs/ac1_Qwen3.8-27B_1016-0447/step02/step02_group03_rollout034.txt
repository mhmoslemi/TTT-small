The objective with sum_a = 1 is: minimize `2*n*max(conv(a,a))`.

The gradient of max(conv(a,a)) with respect to a is `2 * (conv(a, e_k))` where e_k is the indicator of the argmax of conv(a,a). But this is only at the active set.

For a smooth approximation, we can use the log-sum-exp: `LSE_beta(max_i c_i) ≈ (1/beta) * log(sum(exp(beta * c_i)))`.

The gradient of this with respect to a_j is: `(2/beta) * sum_k w_k * c_k_indexed_back`, where `w_k = exp(beta*c_k)/sum(exp(beta*c))` and `c_k_indexed_back` gives the back-convolution.

Actually, let me think more carefully. If `c = conv(a, a)`, then `dc/d a_j` at the max index k is `2 * a[max(0,k-n+1):min(n,k+1)]` reversed appropriately.

The log-sum-exp gradient:
- `LSE = (1/β) * log(Σ exp(β*c_k))`
- `dLSE/d a_j = (1/β) * Σ_k w_k * dc_k/da_j` where `w_k = exp(β*c_k)/Σ exp(β*c)`
- `dc_k/da_j = 2 * a[j']` where `j'` is the index in `a` that contributes to `c_k`

Actually for `c = conv(a,a)`, `c_k = Σ_i a_i * a_{k-i}` (with bounds). So `dc_k/da_j = 2*a_{k-j}` (when valid).

So `dLSE/da_j = (2/β) * Σ_k w_k * a_{k-j}` (when k-j is valid)

This can be computed as `(2/β) * conv(w, a)_j` using FFT!

That's the key insight - the gradient can be computed entirely with FFT in O(n log n).

- Find indices where conv is near maximum
- Construct a "direction" that reduces those peaks
- The direction at peak index k is: for each j, the contribution to c_k from a_j is a_{k-j}, so to reduce c_k we want to decrease a at positions that contribute to c_k.

Actually, the gradient of the max at index k is:

For multiple active indices with weights w_k: `g_j = 2 * Σ_k w_k * a_{k-j}` = `2 * (w * a)_j` (reverse conv).

This is computable with FFT!

import numpy as np
import time
import math
try:
    import scipy.optimize as sopt
except Exception:
    sopt = None

DL = 0.0
BP = None
BS = float('inf')
GN = 0
NF = 0
NG = 0

def clip_norm(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return None
    p = a / s
    if not np.isfinite(p).all():
        return None
    return p

def fft_conv2(a, b, N):
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.real(np.fft.irfft(fa * fb, N))

def self_conv(p, N):
    fa = np.fft.rfft(p, N)
    return np.real(np.fft.irfft(fa * fa, N))

def score_of(p):
    c = self_conv(p, NF)
    return float(2.0 * GN * c.max())

def upd(p):
    global BP, BS
    p = clip_norm(p)
    if p is None:
        return
    sc = float(evaluate_sequence(p.tolist()))
    if np.isfinite(sc) and sc < BS:
        BP = p.copy()
        BS = sc

def lse_val(z, beta):
    zm = z - z.max()
    e = np.exp(np.clip(zm, -60, 60))
    s = e.sum()
    if s < 1e-300:
        return 1e100
    p = e / s
    c = self_conv(p, NF)
    u = float(GN) * c
    um = u.max()
    if um < 0:
        um = 0
    ev = np.exp(np.clip(beta * (u - um), -60, 60))
    s2 = ev.sum()
    if s2 < 1e-300:
        return 1e100
    return float((beta * um + np.log(s2)) / beta)

def lse_grad(z, beta):
    n = GN
    zm = z - z.max()
    p = np.exp(np.clip(zm, -60, 60))
    s = p.sum()
    if s < 1e-300:
        return np.zeros(n)
    p = p / s
    c = self_conv(p, NF)
    u = float(n) * c
    um = u.max()
    ev = np.exp(np.clip(beta * (u - um), -60, 60))
    s2 = ev.sum()
    if s2 < 1e-300:
        return np.zeros(n)
    w = ev / s2
    # g_dp[j] = (2n/beta) * (w cross-corr p)[j] = (2n/beta) * conv(w, p_flipped)[j+n-1]
    p_flipped = p[::-1].copy()
    Nc = NG
    conv_wc = fft_conv2(w, p_flipped, Nc)
    # (w ⋆ p)[d] = conv(w, p_flipped)[d + n - 1] for d in [0, n-1]
    g_dp = (2.0 * float(n) / beta) * conv_wc[n - 1:n - 1 + n]
    # Chain rule for softmax: dz = p * (g_dp - p·g_dp)
    d = float(p @ g_dp)
    g_z = p * (g_dp - d)
    return g_z

def lse_run(z0):
    if sopt is None:
        return
    z = z0.copy()
    z -= z.mean()
    z = np.clip(z, -30, 30)
    stages = [(4, 30), (16, 40), (64, 50), (256, 60), (1024, 80), (4096, 100), (16384, 120)]
    cur = lse_val(z, stages[0][0])
    for beta, maxit in stages:
        if time.monotonic() > DL - 20:
            break
        try:
            res = sopt.minimize(lse_val, z, args=(beta,), jac=lambda zz: lse_grad(zz, beta),
                               method='L-BFGS-B',
                               options={'maxiter': maxit, 'ftol': 1e-14, 'gtol': 1e-12, 'maxls': 30})
            if res is not None and res.x is not None and np.isfinite(res.fun):
                if res.fun <= cur + 1e-6:
                    z = np.asarray(res.x, dtype=np.float64)
                    cur = float(res.fun)
        except Exception:
            pass
        p = np.exp(z - z.max())
        s = p.sum()
        if s > 1e-300:
            p /= s
            upd(p)
    return z

def peak_shave(p, iters):
    p = p.copy()
    cur_sc = score_of(p)
    for _ in range(iters):
        if time.monotonic() > DL - 15:
            break
        c = self_conv(p, NF)
        pk = float(c.max())
        if pk < 1e-15:
            break
        top_k = 30
        idx = np.argsort(c)[::-1][:top_k]
        thr = pk * 0.90
        active = [int(k) for k in idx if c[k] >= thr]
        if not active:
            break
        nw = len(active)
        wts = np.array([c[k] for k in active])
        wts /= wts.sum()
        p_flipped = p[::-1].copy()
        wv = np.zeros(2 * GN - 1, dtype=np.float64)
        for ii in range(nw):
            wv[active[ii]] = wts[ii]
        conv_wc = fft_conv2(wv, p_flipped, NG)
        g = (2.0 * GN) * conv_wc[GN - 1:GN - 1 + GN]
        d = float(p @ g)
        g = g - d
        found = False
        for t in (0.05, 0.03, 0.01, 0.005, 0.002, 0.001):
            p2 = p - t * g
            if p2.min() < 0:
                p2 = np.maximum(p2, 0)
                s2 = p2.sum()
                if s2 < 1e-15:
                    break
                p2 /= s2
            else:
                p2 /= p2.sum()
            sc2 = score_of(p2)
            if np.isfinite(sc2) and sc2 < cur_sc - 1e-14:
                p = p2
                cur_sc = sc2
                found = True
                break
        if not found:
            break
    upd(p)
    return p

def gen_inits(n, rng):
    L = []
    j = np.arange(n, dtype=np.float64)
    m = 0.5 * (n - 1)
    L.append(np.ones(n, dtype=np.float64))
    L.append(0.5 * (1.0 + np.cos(np.pi * j / (n - 1))))
    L.append(0.5 * (1.0 - np.cos(np.pi * j / (n - 1))) + 0.01)
    L.append(np.exp(-((j - m) / (0.3 * m)) ** 2))
    L.append(np.exp(-((j - 0.3 * n) / (0.15 * n)) ** 2) + np.exp(-((j - 0.7 * n) / (0.15 * n)) ** 2))
    L.append((j + 1.0) ** (-0.5) + (n - j) ** (-0.5) + 0.01)
    for _ in range(8):
        c = rng.normal(0, 1, n)
        L.append(0.5 + 0.5 * np.tanh(c))
    return L

        p = clip_norm(a)
        if p is None:
