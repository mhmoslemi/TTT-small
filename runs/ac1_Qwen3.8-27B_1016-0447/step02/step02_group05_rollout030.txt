But the previous code takes `corr[:n]`, which is `corr[0], corr[1], ..., corr[n-1]` = `sum_i p[i]*a[i], sum_i p[i]*a[i+1], ..., sum_i p[i]*a[i+n-1]`.

And we want `grad[j] = sum_k p[k] * a[k-j]` for j = 0, ..., n-1.

`grad[0] = sum_k p[k] * a[k] = corr[0]` ✓
`grad[1] = sum_k p[k] * a[k-1] = sum_i p[i+1] * a[i]`... hmm, this is NOT `corr[1]`.

`corr[1] = sum_i p[i] * a[i+1] = sum_j p[j-1] * a[j]` (shifting index).

`grad[1] = sum_k p[k] * a[k-1] = sum_j p[j+1] * a[j]`.

These are different! `corr[1]` uses `p[j-1]*a[j]` while `grad[1]` uses `p[j+1]*a[j]`.

So the previous code has the gradient "shifted" by 1. But since p is approximately symmetric (it's a softmax over the convolution, which is symmetric), the shift doesn't matter much. The convolution c[k] = c[2n-2-k], so p[k] ≈ p[2n-2-k], and the shift is a minor perturbation.

import numpy as np
import time
from scipy import fft as sfft


def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _normalize(x, target_sum):
    s = x.sum()
    if s < 1e-12:
        return np.full(len(x), target_sum / len(x))
    return np.clip(x * (target_sum / s), 0.0, 1000.0)


def _step(x, T, lr, beta, mom, fN, cN, n):
    X = sfft.rfft(x, fN)
    conv = np.real(sfft.irfft(X * X, fN))[:2 * n - 1]
    conv = np.maximum(conv, 0.0)
    cmax = conv.max()
    logit = np.clip((conv - cmax) / max(T, 1e-10), -50.0, 0.0)
    p = np.exp(logit)
    ps = p.sum()
    if ps < 1e-300:
        p = np.zeros(2 * n - 1)
        p[n - 1] = 1.0
    else:
        p /= ps
    P = sfft.rfft(p, cN)
    Xc = sfft.rfft(x, cN)
    g = 2.0 * np.real(sfft.irfft(P * np.conj(Xc), cN))[:n]
    g -= g.mean()
    gm = np.abs(g).max()
    if gm > 1e5:
        g *= 1e5 / gm
    mom[:] = beta * mom + (1.0 - beta) * g
    return np.clip(x - lr * mom, 0.0, 1000.0)
