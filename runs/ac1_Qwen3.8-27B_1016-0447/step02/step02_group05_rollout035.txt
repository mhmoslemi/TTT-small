When we do x <- clip(x - lr*mom) then x <- x * (ts / x.sum()), the effective update is:

This is NOT the same as x - lr*(mom - mean(mom)) because of the clipping and the nonlinear scaling.

For small updates (lr*mom << x), the scaling is approximately:
x_new ≈ x - lr*mom + x * lr * sum(mom) / sum(x)

So the effective gradient is mom - x * sum(mom) / sum(x) = mom - x * mean(mom) / mean(x)

This is a weighted mean subtraction, not a simple mean subtraction. For a nearly-constant x, it's approximately the same. For a non-constant x, it's different.

The previous code does g -= g.mean() before the step, which subtracts the unweighted mean. The subsequent normalization subtracts the weighted mean. There's a small inconsistency, but for small steps, it shouldn't matter much.

from scipy import fft as sfft
import time


def _npow2(m):
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _norm(x, ts):
    s = x.sum()
    if s < 1e-8:
        return np.full(x.shape, ts / x.size, dtype=np.float64)
    return np.clip(x * (ts / s), 0.0, 1000.0)


def _step(x, T, lr, mom, bt, fs, cfs, n):
    X = sfft.rfft(x, n=fs)
    c = np.real(sfft.irfft(X * X, n=fs))[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    cm = c.max()
    lg = (c - cm) / max(T, 1e-10)
    np.clip(lg, -90.0, 0.0, out=lg)
    p = np.exp(lg)
    ps = p.sum()
    if ps < 1e-300:
        p[:] = 0.0
        p[n - 1] = 1.0
    else:
        p /= ps
    P = sfft.rfft(p, n=cfs)
    Xc = sfft.rfft(x, n=cfs)
    cr = np.real(sfft.irfft(P * np.conj(Xc), n=cfs))
    g = cr[:n].copy()
    g *= 2.0
    g -= g.mean()
    gn = np.sqrt(np.dot(g, g))
    xm = x.mean()
    if gn > 1e-15 and xm > 1e-15:
        g *= xm / gn
    mom[:] = bt * mom + (1.0 - bt) * g
    x2 = x - lr * mom
    np.clip(x2, 0.0, 1000.0, out=x2)
    return x2

import numpy as np
from scipy import fft as sfft
import time


def _npow2(m):
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _norm(x, ts):
    s = x.sum()
    if s < 1e-8:
        return np.full(x.shape, ts / x.size, dtype=np.float64)
    return np.clip(x * (ts / s), 0.0, 1000.0)


def _step(x, T, lr, mom, bt, fs, cfs, n):
    X = sfft.rfft(x, n=fs)
    c = np.real(sfft.irfft(X * X, n=fs))[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    cm = c.max()
    lg = (c - cm) / max(T, 1e-10)
    np.clip(lg, -90.0, 0.0, out=lg)
    p = np.exp(lg)
    ps = p.sum()
    if ps < 1e-300:
        p[:] = 0.0
        p[n - 1] = 1.0
    else:
        p /= ps
    P = sfft.rfft(p, n=cfs)
    Xc = sfft.rfft(x, n=cfs)
    cr = np.real(sfft.irfft(P * np.conj(Xc), n=cfs))
    g = cr[:n].copy()
    g *= 2.0
    g -= g.mean()
    gn = np.sqrt(np.dot(g, g))
    xm = x.mean()
    if gn > 1e-15 and xm > 1e-15:
        g *= xm / gn
    mom[:] = bt * mom + (1.0 - bt) * g
    x2 = x - lr * mom
    np.clip(x2, 0.0, 1000.0, out=x2)
    return x2
