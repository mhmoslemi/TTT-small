    n = GN
    # Build sparse constraint matrix A (2n-1) x n
    # A[k,m] = f[k-m] if 0 <= k-m < n
    from scipy import sparse
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):

        hi = min(n-1, k)
        for m in range(lo, hi+1):
            rows.append(k)
            cols.append(m)

        s = g.sum()
