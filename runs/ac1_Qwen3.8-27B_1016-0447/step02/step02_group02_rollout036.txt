        kk = int(idx[i])
        lo = kk - n + 1 if kk > n - 1 else 0
        hi = kk if kk < n else n - 1
        g[lo:hi + 1] += 2.0 * float(w[i]) * a[kk - hi:kk - lo + 1][::-1]
    return g


def _norm1(x):
    s = x.sum()
    return x / s if s > 1e-18 else None


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    dl = time.monotonic() + max(0.0, float(budget_s) - 20.0)
    par = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = par.size
    if n < 3 or par.sum() < 0.01:
        return [1.0]
    fs = _np2(2 * n - 1)
    x0 = par / par.sum()
    cb = _fconv(x0, fs)
    sb = 2.0 * n * cb.max()
    xb = x0.copy()

    def upd(x):
        nonlocal xb, sb
        cv = _fconv(x, fs)
        sc = 2.0 * n * cv.max()
        if sc < sb:
            sb, xb = sc, x.copy()
        return sc

    def oloop(x, fr, mf, alpha, ilim, d, mlo, mhi):
        nonlocal xb, sb
        x = x.copy()
        cv = _fconv(x, fs)
        sc0 = 2.0 * n * cv.max()
        sc0 -= x.mean() * (2.0 * alpha * x.mean())
        ni = 0
        for it in range(ilim):
            if time.monotonic() > d:
                break
            cv = _fconv(x, fs)
            sc = 2.0 * n * cv.max() - x.mean() * (2.0 * alpha * x.mean())
            if sc < sb:
                sb, xb = sc, x.copy()
            g = _tsa(cv, x, mf)
            g -= g.mean()
            gn = np.linalg.norm(g)
            if gn < 1e-18:
                break
            for _ in range(14):
                x2 = np.maximum(x - fr * g / gn, 0.0)
                x2 = _norm1(x2)
                if x2 is None:
                    break
                c2 = _fconv(x2, fs)
                s2 = 2.0 * n * c2.max() - x2.mean() * (2.0 * alpha * x2.mean())
                if s2 < sc - 1e-16:
                    x = x2
                    sc = s2
                    break
                fr *= 0.45
                if fr < 1e-8:
                    break
            if fr > 1e-8:
                fr = min(fr * 1.25, 0.15)
            ni = 0 if sc < sc0 - 1e-13 else ni + 1
            if ni >= 300:
                x = xb.copy()
                x[x > 1e-10] += 0.02 * rng.standard_normal(np.sum(x > 1e-10))
                x = np.maximum(x, 0.0)
                x = _norm1(x)
                if x is None:
                    break
                sc0 = 2.0 * n * _fconv(x, fs).max() - x.mean() * (2.0 * alpha * x.mean())
                ni, fr = 0, 0.06
            if it % 400 == 0:
                mfn = int(np.clip(mlo + (mhi - mlo) * float(it) / ilim, mlo, mhi))
                mf = mfn
        return x

    # Phase 1: moderate exploration
    x1 = oloop(x0, 0.06, 8, 0.0, 25000, dl - 180, 4, 10)
    upd(x1)

    # Phase 2: fine exploitation with tiny m
    x2 = oloop(xb, 0.006, 1, 0.0, 35000, dl - 60, 1, 1)
    upd(x2)

    # Phase 3: very fine with occasional m=2
    if time.monotonic() < dl - 25:
        x3 = xb.copy()
        cv3 = _fconv(x3, fs)
        sc3 = 2.0 * n * cv3.max()
        nf3 = 0
        for it in range(20000):
            if time.monotonic() > dl - 25:
                break
            cv3 = _fconv(x3, fs)
            s3m = 2.0 * n * cv3.max()
            if s3m < sb:
                sb, xb = s3m, x3.copy()
            g3 = _tsa(cv3, x3, 1 if it % 6 < 4 else 2)
            g3 -= g3.mean()
            gn3 = np.linalg.norm(g3)
            if gn3 < 1e-18:
                break
            x3 = np.maximum(x3 - 0.003 * g3 / gn3, 0.0)
            x3 = _norm1(x3)
            if x3 is None:
                break
            nf3 = 0 if s3m < sb + 1e-15 else nf3 + 1
            if nf3 >= 600:
                x3 = xb.copy()
                x3[x3 > 1e-12] += 0.005 * rng.standard_normal(np.sum(x3 > 1e-12))
                x3 = np.maximum(x3, 0.0)
                x3 = _norm1(x3)
                if x3 is None:
                    break
                nf3 = 0
        upd(x3)

    # Final exact validation
    out = [float(v) for v in xb]
    sc = float(evaluate_sequence(out))
    if not np.isfinite(sc):
        alt = [float(v) for v in x0]
        sc2 = float(evaluate_sequence(alt))
        if np.isfinite(sc2):
            return alt
        return [1.0]
    return out
