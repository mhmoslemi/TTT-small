            cand = np.maximum(best - st * g, 0.0)
            sc_sum = cand.sum()
            if sc_sum < 1e-300:
                continue
            cand /= sc_sum
            cy = acorr(cand, fs, n)
            fy = 2.0 * n * cy.max()
            if fy < exact - 1e-12:
                exact = fy
                best = cand
                improved = True
                break
        if not improved:
            break

    # Verify with evaluator
    final_sc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_sc):
        best = np.ones(n)
        final_sc = 2.0 * float(n)

    # Scale output to use range well (score is scale-invariant)
    mx = best.max()
    if mx > 0:
        scale = min(500.0 / mx, 1000.0 / mx)
        out = best * scale
    else:
        out = best
    out = np.clip(out, 0.0, 1000.0)
    if out.sum() < 0.01:
        out = np.ones(n)

    return [float(v) for v in out]
