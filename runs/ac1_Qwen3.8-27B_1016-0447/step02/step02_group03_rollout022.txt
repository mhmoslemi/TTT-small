import numpy as np
import time

DDL = 0.0
BEST = None
BEST_SC = float('inf')
N = 0
NF = 0

def normclip(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return None
    p = a / s
    return p if np.isfinite(p).all() else None

def conv(p):
    fa = np.fft.rfft(p, NF)
    return np.real(np.fft.irfft(fa * fa, NF)[:2 * N - 1])

def score(p):
    return float(2.0 * N * conv(p).max())

def update_best(p):
    global BEST, BEST_SC
    p = normclip(p)
    if p is None:
        return

def peak_shave(p, max_iter=800, top_k=1):

            break

        if top_k <= 1:
            lo = max(0, pk - (N - 1))
            hi = min(N - 1, pk)

        else:

            for idx in range(len(active)):
                ks = int(active[idx])

                hi = min(N - 1, ks)
                g[lo:hi + 1] += wts[idx] * p[ks - hi:ks - lo + 1][::-1]
            g *= 2.0
        gm = g - p @ g

                return

            u = beta * N * c

            if sw < 1e-300:
                return

            fp = np.fft.rfft(p, NF)
            fw = np.fft.rfft(w, NF)

                p2 = np.exp(zm2)

def gen_inits(n):
    j = np.arange(n, dtype=np.float64)
    m = 0.5 * (n - 1)
    L = []
    L.append(np.ones(n))
    for a in (0.5, 1.0, 1.5, 2.0, 3.0, 5.0):

    for pp in (-0.3, -0.7, -1.2, -2.0, -3.0):
        L.append((j + 1.0) ** pp)
    for s in (0.3, 0.35, 0.4, 0.5, 0.6, 0.65, 0.7):
        sd = 0.25 * (n - 1)
        L.append(np.exp(-((j - s * (n - 1)) / sd) ** 2) + 0.005)
    for a in (1.0, 2.0, 4.0):
        L.append((np.minimum(j, n - 1 - j) + 1.0) ** a)
    for f in (1, 2, 3):
        mod = 1.0 + 0.3 * np.cos(2.0 * np.pi * f * j / n)
        L.append(0.5 * (1.0 + np.cos(np.pi * j / (n - 1))) * mod + 0.01)
    return L
