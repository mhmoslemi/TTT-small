    remaining = deadline - 3.0 - time.monotonic()
    if remaining > 10.0 and np.isfinite(best_score):
        x_refine = best * (1.0 + 0.02 * rng.standard_normal(n))
        x_refine = normalize_x(x_refine, target_sum)
        best, best_score = run_subgradient(
            x_refine, n, fft_size, corr_size, target_sum,
            T0=0.15, lr0=0.5, min_T=0.002,
            max_iter=8000000, deadline=deadline, rng=rng,
            best_score=best_score, best=best,
            eval_interval=800, no_improve_limit=2500
        )

    # Final polish: very low temperature, small steps
    remaining = deadline - 3.0 - time.monotonic()
    if remaining > 5.0 and np.isfinite(best_score):
        x_polish = best.copy()
        T = 0.003
        lr = 0.15
        for _ in range(int(remaining * 800)):
            if time.monotonic() > deadline - 3.0:
                break
            X = sfft.rfft(x_polish, fft_size)
            conv = np.real(sfft.irfft(X * X, fft_size))[:2 * n - 1]
            conv = np.maximum(conv, 0.0)
            cmax = conv.max()
            if cmax < 1e-30:
                break
            p = np.exp((conv - cmax) / T)
            ps = p.sum()
            if ps < 1e-300:
                continue
            p /= ps
            P = sfft.rfft(p, corr_size)
            Xc = sfft.rfft(x_polish, corr_size)
            corr = np.real(sfft.irfft(P * np.conj(Xc), corr_size))[:n]
            g = 2.0 * corr
            g -= g.mean()
            x_polish -= lr * g
            x_polish = normalize_x(x_polish, target_sum)
            sc = safe_score(x_polish)
            if sc < best_score - 1e-10:
                best_score = sc
                best = x_polish.copy()

    if not np.isfinite(best_score) or best is None:
        return [float(v) for v in parent]

    fc = safe_score(best)
    if not np.isfinite(fc) or fc > 999.0:
        return [float(v) for v in parent]

    return [float(v) for v in best]
