import numpy as np

def _fft_conv(x, y):
    m = len(x) + len(y) - 1
    sz = 1
    while sz < m:
        sz <<= 1
    fx = np.fft.rfft(x, sz)
    fy = np.fft.rfft(y, sz)
    return np.fft.irfft(fx * fy, sz)[:m]


def _compute_grad(a, p):
    """Compute gradient of sum_k p[k]*b[k] w.r.t. a, where b = conv(a,a)."""
    n = a.size
    sz = 1
    while sz < 3 * n:
        sz <<= 1
    P = np.fft.rfft(p, sz)
    A = np.fft.rfft(a, sz)
    crosscorr = np.fft.irfft(P * np.conj(A), sz)
    return 2.0 * crosscorr[:n]


def _optimize(a, deadline, rng, label=""):

    n = a.size
    a = np.maximum(a, 0.0).astype(np.float64)

        s = n
    a = a / s  # normalize to sum=1
    
    best = a.copy()
    b = _fft_conv(best, best)

    tau = 0.5 / n  # initial temperature for softmax
    lr = 1.0 / n  # initial learning rate
    it = 0
    stalled = 0

    while time.monotonic() < deadline:

        # Compute softmax weights

        e = np.exp(z)

        # Compute gradient

        a_new = a * np.exp(-lr * g)
        a_new = np.maximum(a_new, 1e-30)
        sn = a_new.sum()
        if sn < 1e-10:

        else:

        # Accept only if score doesn't increase too much (or always for small lr)

import numpy as np
import time


def _fft_conv(x, y):
    m = len(x) + len(y) - 1
    sz = 1
    while sz < m:
        sz <<= 1
    fx = np.fft.rfft(x, sz)
    fy = np.fft.rfft(y, sz)
    return np.fft.irfft(fx * fy, sz)[:m]


def _grad_smooth(a, p):
    n = a.size
    sz = 1
    while sz < 3 * n:
        sz <<= 1
    P = np.fft.rfft(p, sz)
    A = np.fft.rfft(a, sz)
    cc = np.fft.irfft(P * np.conj(A), sz)
    return 2.0 * cc[:n]


def _optimize(a, deadline, rng):
    n = a.size
    a = np.maximum(a, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        a[:] = 1.0
        s = float(n)
    a = a / s

    best = a.copy()
    best_score = 2.0 * n * _fft_conv(best, best).max()

    tau = 2.0 / n
    lr = 2.0 / n
    it = 0
    stalled = 0
    patience = 8
    eval_every = 400

    while time.monotonic() < deadline - 3.0:
        it += 1
        b = _fft_conv(a, a)
        z = b / tau
        z -= z.max()
        e = np.exp(z)
        p = e / e.sum()

        g = _grad_smooth(a, p)
        a *= np.exp(-lr * g)
        a = np.maximum(a, 1e-25)
        sn = a.sum()
        if sn > 1e-15:
            a /= sn

        if it % 40 == 0:
            tau *= 0.995
            if tau < 0.01 / n:
                tau = 0.01 / n
            lr *= 0.999

        if it % eval_every == 0:
            b_now = _fft_conv(a, a)
            sc = 2.0 * n * b_now.max()
            if sc < best_score - 1e-12:
                best_score = sc
                best = a.copy()
                stalled = 0
            else:
                stalled += 1
            if stalled >= patience:
                eps = 0.005 * (1.0 + 0.5 * stalled)
                a += eps * rng.standard_normal(n)
                a = np.maximum(a, 0.0)
                s2 = a.sum()
                if s2 > 1e-10:
                    a /= s2
                tau = min(tau * 5.0, 1.0 / n)
                lr = 3.0 / n
                stalled = 0
            if time.monotonic() >= deadline - 3.0:
                break

    b = _fft_conv(a, a)
    sc = 2.0 * n * b.max()
    if sc < best_score:
        best_score = sc
        best = a.copy()
    return best, best_score


def _polish(a, deadline, rng):
    """Fine-tune with smaller steps."""
    n = a.size
    a = np.maximum(a, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a, float('inf')
    a = a / s
    best = a.copy()
    best_score = 2.0 * n * _fft_conv(best, best).max()

    tau = 0.1 / n
    lr = 0.1 / n
    it = 0
    while time.monotonic() < deadline - 3.0:
        it += 1
        b = _fft_conv(a, a)
        z = b / tau
        z -= z.max()
        e = np.exp(z)
        p = e / e.sum()
        g = _grad_smooth(a, p)
        a *= np.exp(-lr * g)
        a = np.maximum(a, 1e-25)
        sn = a.sum()
        if sn > 1e-15:
            a /= sn
        if it % 2 == 0:
            lr *= 0.9995
            tau *= 0.999
            if tau < 1e-5 / n:
                tau = 1e-5 / n
        if it % 300 == 0:
            b_now = _fft_conv(a, a)
            sc = 2.0 * n * b_now.max()
            if sc < best_score - 1e-14:
                best_score = sc
                best = a.copy()
    return best, best_score
