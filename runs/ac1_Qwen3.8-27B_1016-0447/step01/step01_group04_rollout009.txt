                trial = norm1((1 - blend) * a_best + blend * norm1(p))
                tsc, _, _ = eval_fft(trial, n)
                if np.isfinite(tsc) and tsc < sc_best * 1.02:
                    a = trial
                    sc = tsc
                    restarted = True
                    break
            if not restarted:
                b = a_best * np.exp(rng.normal(0, 0.04, n))
                b = norm1(b)
                tsc, _, _ = eval_fft(b, n)
                if np.isfinite(tsc) and tsc < sc_best * 1.02:
                    a = b
                    sc = tsc
            stall = 0

        # Temperature: start moderate, decrease
        frac = min(1.0, it / 600.0)
        tau_frac = 0.12 * (1 - frac) + 0.015 * frac
        tau = max(tau_frac * M, 1e-14)

        w = sof(c, tau)
        h = croscorr(w, a)
        g = 2.0 * h / tau
        g = np.maximum(g, 0.0)
        g = np.where(np.isfinite(g), g, 0.0)

        gmax = g.max()
        if gmax < 1e-18:
            break

        found = False
        for f in (0.02, 0.05, 0.10, 0.20, 0.40):
            eta = f / gmax
            logit = np.clip(-eta * g, -10.0, 10.0)
            b = a * np.exp(logit)
            b = norm1(b)
            cb = conv_full(b)
            Mb = cb.max()
            scb = 2.0 * n * Mb / (b.sum() ** 2)
            if np.isfinite(scb) and scb < sc - 1e-15:
                a = b
                sc = scb
                found = True
                break

        if not found and stall > 40 and it % 3 == 0:
            b = a * np.exp(rng.normal(0, 0.015, n))
            b = norm1(b)
            cb = conv_full(b)
            scb = 2.0 * n * cb.max() / (b.sum() ** 2)
            if np.isfinite(scb) and scb < sc:
                a = b
                sc = scb

        it += 1

    # Phase 2: Fine polishing with near-exact gradient
    a = a_best.copy()
    c = conv_full(a)
    M = c.max()
    s = a.sum()
    sc = 2.0 * n * M / (s * s)

    for trial in range(3000):
        if time.monotonic() > deadline:
            break
        active = np.where(c >= M - 0.005 * M)[0]
        w = np.zeros(c.size)
        w[active] = 1.0 / len(active)
        h = croscorr(w, a)
        g = 2.0 * h
        g = np.maximum(g, 0.0)
        gmax = g.max()
        if gmax < 1e-18:
            break
        improved = False
        for sf in (0.0003, 0.001, 0.003, 0.008):
            step = sf * s / gmax
            b = a - step * g
            b = np.maximum(b, 0.0)
            bs = b.sum()
            if bs < 1e-300:
                continue
            b = b / bs
            cb = conv_full(b)
            Mb = cb.max()
            scb = 2.0 * n * Mb / (b.sum() ** 2)
            if np.isfinite(scb) and scb < sc_best - 1e-14:
                a = b
                sc = scb
                sc_best = scb
                a_best = b.copy()
                c = cb
                M = Mb
                s = b.sum()
                improved = True
                break
        if not improved:
            b = a * np.exp(rng.normal(0, 0.005, n))
            b = norm1(b)
            cb = conv_full(b)
            scb = 2.0 * n * cb.max() / (b.sum() ** 2)
            if np.isfinite(scb) and scb < sc_best:
                a = b
                sc_best = scb
                a_best = b.copy()
                sc = scb
                c = cb
                M = cb.max()
                s = a.sum()

    # Final validation
    out = np.clip(a_best, 0.0, 1000.0)
    if out.sum() < 0.01:
        out = np.ones(n) / n

    try:
        ev = evaluate_sequence(out.tolist())
    except Exception:
        ev = sc_best

    # Also check parent blend candidates
    for blend in (0.05, 0.1, 0.2):
        trial = norm1((1.0 - blend) * a_best + blend * norm1(p))
        trial = np.clip(trial, 0.0, 1000.0)
        try:
            tev = evaluate_sequence(trial.tolist())
        except Exception:
            tev = np.inf
        if np.isfinite(tev) and tev < ev:
            out = trial
            ev = tev

    out = np.clip(out, 0.0, 1000.0)
    if out.sum() < 0.01:
        out = np.ones(n) / n
    if not np.all(np.isfinite(out)):
        out = np.ones(n) / n
    return [float(x) for x in out]
