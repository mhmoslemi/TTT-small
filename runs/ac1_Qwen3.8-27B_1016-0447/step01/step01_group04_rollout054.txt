def _fft_len(m):
    L = 1
    while L < m:
        L <<= 1
    return L

def _normalize(a):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-300:
        return np.ones_like(a) / len(a)
    return a / s

def _score(a, flen):
    af = np.fft.rfft(a, flen)
    c = np.fft.irfft(af * af, flen)[:2*len(a)-1]
    c = np.maximum(c, 0.0)
    M = c.max()
    return 2.0 * len(a) * M / (a.sum()**2)

def _npo2(m):
    L = 1
    while L < m:
        L <<= 1
    return L

def _norm(a):

    s = a.sum()
    if s < 1e-300:
        return np.ones_like(a) / a.size

def _score(a, flen):

    c = np.fft.irfft(af * af, flen)[:2*a.size-1]

    S = a.sum()
