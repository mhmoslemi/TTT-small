    n = len(a)
    c = _fft_conv(a, a)
    sb = c.max()

def _fft_conv(x, y):
    m = len(x) + len(y) - 1
    sz = 1
    while sz < m:
        sz <<= 1
    fx = np.fft.rfft(x, sz)
    fy = np.fft.rfft(y, sz)
    return np.fft.irfft(fx * fy, sz)[:m]

    c = _fft_conv(a, a)
    return 2.0 * n * c.max() / (a.sum() ** 2)
