Wait: sum_n a[n]*w[n+k]. Let m = n+k, n = m-k: sum_m a[m-k]*w[m] = sum_m w[m]*a[m-k]. Yes!

So `grad_a[k] = 2 * irfft(conj(FA) * FW)[k]` where FA = rfft(a, N) and FW = rfft(w, N).

Let me re-verify with the example:
a = [1,2,3,0,0], w = [0,0,0,1,0]
conj(FA)*FW IFFT at j:
= sum_n a[n]*w[n+j]
j=0: a[0]*w[0] + a[1]*w[1] + a[2]*w[2] + a[3]*w[3] + a[4]*w[4] = 1*0 + 2*0 + 3*0 + 0*1 + 0*0 = 0
j=1: a[0]*w[1] + a[1]*w[2] + a[2]*w[3] + ... = 0 + 0 + 3*1 = 3
j=2: a[0]*w[2] + a[1]*w[3] + a[2]*w[4] = 0 + 2*1 + 3*0 = 2

So grad_a = 2*[0, 3, 2] = [0, 6, 4].

And from the direct computation: dc[3]/da[k] = 2*a[3-k]:
k=0: 2*a[3] = 0
k=1: 2*a[2] = 6
k=2: 2*a[1] = 4

grad_a = [0, 6, 4]. ✓

So the formula is: `grad_a = 2 * irfft(conj(rfft(a,N)) * rfft(w,N))[:n]`

import numpy as np
import time
import scipy.optimize as so

_G_N = 0
_G_L = 0
_G_fftN = 0
_G_beta = 1.0
_G_deadline = 0.0


class _TMO(Exception):
    pass


def _check_time():
    if time.monotonic() > _G_deadline:
        raise _TMO


def _set_globals(n):
    global _G_N, _G_L, _G_fftN
    _G_N = int(n)
    _G_L = 2 * int(n) - 1
    _G_fftN = 1 << max(0, (3 * int(n) - 3).bit_length())


def _autocorr(x):
    fx = np.fft.rfft(x, _G_fftN)
    c = np.fft.irfft(fx * fx, _G_fftN)[:_G_L]
    return np.clip(c, 0.0, None)


def _lr_score(x):
    c = _autocorr(x)
    return float(2.0 * _G_N * np.max(c))


def _obj(z):
    _check_time()
    beta = _G_beta
    n = _G_N
    z = z - z.max()
    aa = np.exp(z)
    s = aa.sum()
    if s < 1e-300:
        return 1e100
    aa /= s
    c = _autocorr(aa)
    u = float(n) * c
    umax = float(u.max())
    ex = np.exp(beta * (u - umax))
    ls = np.sum(ex)
    if ls < 1e-300 or not np.isfinite(ls):
        return 1e100
    return float((beta * umax + np.log(ls)) / beta)


def _obj_grad(z):
    _check_time()
    beta = _G_beta
    n = _G_N
    z = z - z.max()
    aa = np.exp(z)
    s = aa.sum()
    if s < 1e-300:
        return np.zeros(n)
    aa /= s
    c = _autocorr(aa)
    u = float(n) * c
    umax = float(u.max())
    ex = np.exp(beta * (u - umax))
    ls = ex.sum()
    if ls < 1e-300 or not np.isfinite(ls):
        return np.zeros(n)
    w = ex / ls
    fa = np.fft.rfft(aa, _G_fftN)
    fw = np.fft.rfft(w, _G_fftN)
    corr = np.fft.irfft(np.conj(fa) * fw, _G_fftN)[:n]
    grad_a = 2.0 * float(n) * corr
    dot = float(np.dot(aa, grad_a))
    gz = aa * (grad_a - dot)
    if not np.isfinite(gz).all():
        return np.zeros(n)
    return gz


def _cb(_):
    _check_time()


def _norm(a):
    a = np.clip(a, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 0.01:
        return None
    return a / s


def _try_update(a, best, best_score):
    a = _norm(a)
    if a is None:
        return best, best_score
    sc = float(evaluate_sequence(a.tolist()))
    if np.isfinite(sc) and sc < best_score:
        return a.copy(), sc
    return best, best_score


def _make_starts(n, rng, parent):
    starts = [parent.copy()]
    j = np.arange(n, dtype=np.float64)
    # Power-law family
    for p in (-0.7, -0.5, -0.3, 0.0, 0.3, 0.5):
        a = (j + 1.0) ** p
        a = _norm(a)
        if a is not None:
            starts.append(a)
    # Symmetric power-law
    for p in (0.3, 0.5, 1.0):
        a = 1.0 + p * (1.0 - np.abs(2.0 * j / (n - 1) - 1.0))
        a = _norm(a)
        if a is not None:
            starts.append(a)
    # Gaussian-ish
    for sig in (0.3, 0.5, 0.7):
        a = np.exp(-0.5 * ((j - n / 2) / (sig * n)) ** 2)
        a = _norm(a)
        if a is not None:
            starts.append(a)
    # Two-plateau
    m = n // 3
    a = np.ones(n)
    a[:m] = 1.5
    a[n - m:] = 1.5
    starts.append(_norm(a))
    return starts


def _lbfgs_optimize(z, betas, deadline):
    """Run multi-beta L-BFGS-B and return best z."""
    _G_deadline = deadline
    for beta in betas:
        _check_time()
        global _G_beta
        _G_beta = float(beta)
        z = z - z.mean()
        amax = float(np.max(np.abs(z)))
        if amax > 35.0:
            z *= 35.0 / amax
        n = _G_N
        maxiter = max(30, min(400, int(80000 // n)))
        try:
            res = so.minimize(
                _obj, z, jac=_obj_grad, method="L-BFGS-B",
                callback=_cb,
                options={"maxiter": maxiter, "ftol": 1e-14, "gtol": 1e-9, "maxls": 40}
            )
            if res is not None and res.x is not None and np.isfinite(res.x).all():
                z_new = res.x - res.x.mean()
                p_new = np.exp(z_new - z_new.max())
                p_new /= p_new.sum()
                if _lr_score(p_new) <= _lr_score(np.exp(z - z.max()) / np.exp(z - z.max()).sum()) + 1e-8:
                    z = z_new
        except Exception:
            pass
    return z


def _subgradient_polish(a, deadline, rng):
    """Subgradient steps on the true max of autocorrelation."""
    best = a.copy()
    best_sc = _lr_score(a)
    eta = 0.02
    stall = 0
    for _ in range(5000):
        if time.monotonic() > deadline:
            break
        c = _autocorr(a)
        umax = c.max()
        peaks = np.where(c > umax * 0.999)[0]
        w = np.zeros(_G_L)
        w[peaks] = 1.0 / len(peaks)
        fa = np.fft.rfft(a, _G_fftN)
        fw = np.fft.rfft(w, _G_fftN)
        corr = np.fft.irfft(np.conj(fa) * fw, _G_fftN)[:_G_N]
        g = 2.0 * float(_G_N) * corr
        a_try = a - eta * g / (np.abs(g).max() + 1e-30)
        a_try = np.clip(a_try, 0.0, None)
        s = a_try.sum()
        if s < 0.01:
            a_try = a + eta * 0.1 * rng.standard_normal(_G_N)
            a_try = np.clip(a_try, 0.0, None)
            s = a_try.sum()
            if s < 0.01:
                break
        a_try /= s
        sc = _lr_score(a_try)
        if sc < best_sc - 1e-12:
            best = a_try.copy()
            best_sc = sc
            a = a_try.copy()
            eta = min(0.15, eta * 1.08)
            stall = 0
        else:
            stall += 1
            if stall > 50:
                eta *= 0.7
                stall = 0
            if eta < 1e-6:
                break
    return best, best_sc


def _noise_polish(a, deadline, rng):
    best = a.copy()
    best_sc = _lr_score(a)
    eta = 0.03
    for _ in range(3000):
        if time.monotonic() > deadline:
            break
        k = rng.integers(0, _G_N, size=min(20, _G_N))
        a_try = a.copy()
        a_try[k] *= np.exp(rng.normal(0.0, eta, len(k)))
        a_try = np.clip(a_try, 0.0, None)
        s = a_try.sum()
        if s < 0.01:
            continue
        a_try /= s
        sc = _lr_score(a_try)
        if sc < best_sc - 1e-12:
            best = a_try.copy()
            best_sc = sc
            a = a_try.copy()
            eta = min(0.2, eta * 1.05)
        else:
            eta *= 0.95
            if eta < 1e-7:
                break
    return best, best_sc
