import time

def npo2(m):
    L = 1
    while L < m:
        L <<= 1
    return L

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.ones_like(a)
    return a / s

def fft_conv(a):
    n = len(a)
    L = npo2(2*n-1)
    af = np.fft.rfft(a, L)
    c = np.fft.irfft(af*af, L)
    return c[:2*n-1]

    n = len(a)

    s = a.sum()

        w /= w.sum() + 1e-300
