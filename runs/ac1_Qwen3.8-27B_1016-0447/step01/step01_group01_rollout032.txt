def _fft_conv(a, b, N):
    """Full convolution of a and b via FFT, result length len(a)+len(b)-1."""
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)

def _self_conv(a, N):
    """Full autocorrelation (a*a) via FFT."""
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)

def _compute_grad(a, n, N1, N2, K):
    """Compute gradient of score at top-K peaks."""
    S = a.sum()
    C = _self_conv(a, N1)[:2*n-1]
