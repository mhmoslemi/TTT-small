import time
from scipy import fft as sfft


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def make_init(n, rng, variant):

    x = np.ones(n) * (0.8 + 0.1 * (variant % 4))
    freqs = [3, 7, 11, 17, 23, 41, 53, 71]
    for i, f in enumerate(freqs):

    x += 0.02 * rng.standard_normal(n)
    x += 0.03 * (variant % 7) * 0.05 * np.cos(4 * np.pi * j / n)
    return np.maximum(x, 0.01)


def normalize_seq(x, S):
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s < 0.01:
        return np.ones_like(x) * (S / x.size)
    return np.clip(x * (S / s), 0.0, 1000.0)
