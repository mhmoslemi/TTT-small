import numpy as np
import time


def _fft_conv(x, y):
    m = len(x) + len(y) - 1
    sz = 1
    while sz < m:
        sz <<= 1
    fx = np.fft.rfft(x, sz)
    fy = np.fft.rfft(y, sz)
    return np.fft.irfft(fx * fy, sz)[:m]


def _score(a):
    n = a.size
    sa = a.sum()
    if sa < 0.01:
        return np.inf
    conv = _fft_conv(a, a)
    return 2.0 * n * conv.max() / (sa * sa)


def _grad(a, tau):
    n = a.size
    s = _fft_conv(a, a)
    z = s / tau
    z -= z.max()
    e = np.exp(z)
    p = e / e.sum()
    c = p[::-1]
    ca = _fft_conv(c, a)
    return 2.0 * ca[n - 1:2 * n - 1][::-1]
