import time
from scipy import fft as sfft

def npow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def sc(x):

def autocorr(x, fs, n):
    X = sfft.rfft(x, fs)
    c = np.real(sfft.irfft(X * X, fs))[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def softmax_wts(c, T):
    cm = c.max()
    K = min(48, c.size)
    idx = np.argpartition(c, -K)[-K:]
    pv = np.exp((c[idx] - cm) / T)
    s = pv.sum()
    p = np.zeros(c.size)
    if s > 1e-300:
        p[idx] = pv / s
    else:
        p[int(np.argmax(c))] = 1.0
    return p

def xgrad(p, x, cs, n):
    P = sfft.rfft(p, cs)
    Xc = sfft.rfft(x, cs)
    g = 2.0 * np.real(sfft.irfft(P * np.conj(Xc), cs))[:n]
    g -= g.mean()
    return g
