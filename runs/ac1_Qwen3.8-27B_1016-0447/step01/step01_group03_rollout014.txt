def exact_subgradient_step(a, n_peaks=10, step_size=1e-3):
    n = len(a)
    N = next_pow2(2*n-1)
    C = ifft(fft(a, N)**2)[:2n-1].real
    
    # Find top peaks
    max_C = C.max()
    # Consider peaks within some fraction of max
    threshold = max_C * (1 - 1e-6)
    peaks = np.where(C >= threshold)[0]
    if len(peaks) > n_peaks:
        # Take top n_peaks
        idx = np.argsort(C[peaks])[::-1][:n_peaks]
        peaks = peaks[idx]
    
    # Compute subgradient
    grad = np.zeros(n)
    for k in peaks:
        # grad[j] += 2*a[k-j] for valid j
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_max >= j_min:
            grad[j_min:j_max+1] += 2 * a[k - j_max:k - j_min + 1][::-1]
    
    # Normalize gradient for step
    # Project: grad - mean(grad) to handle sum constraint
    grad -= grad.mean()
    
    # Adaptive step size
    step = step_size * (a.mean() / (grad.std() + 1e-10))
    a_new = a - step * grad
    a_new = np.clip(a_new, 0, None)
    a_new /= a_new.sum()
    return a_new, C, peaks

    fa = np.fft.rfft(a, N)
    return np.real(np.fft.irfft(fa * fa, N)[:L])

    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * len(a) * _conv(a).max() / (s * s)

    s = a.sum()
    if s < 1e-30 or not np.isfinite(s):

    p /= p.sum()
    C = _conv(p)
    if tk and tk < len(C):
        vi = np.argpartition(C, -tk)[-tk:]
        v = C[vi]
    else:
        v = C
    vm = v.max()

    for _ in range(niter):

            break
        C = _conv(p)
        mx = C.max()
        peaks = np.where(C >= mx * (1 - 1e-4))[0]
