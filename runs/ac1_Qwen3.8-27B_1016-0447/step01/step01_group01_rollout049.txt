import time

def _fft_conv(a, N):
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)

def _compute_grad(a, b, n, peaks, S, M):
    """Subgradient of score 2n*M/S^2 w.r.t. a, using active peaks."""
    n_peaks = len(peaks)
    gv = b[peaks]
    w = gv / gv.sum()
    grad = np.zeros(n)
    for w_k, k in zip(w, peaks):
        ki = int(k)
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if lo <= hi:
            # a[ki - j] for j in [lo, hi]
            # ki-j ranges from ki-hi (>=0) to ki-lo (<=n-1)
            seg = a[ki - hi : ki - lo + 1]
            # This gives a[ki-hi], a[ki-hi+1], ..., a[ki-lo]
            # For j=lo: a[ki-lo] (last element of seg)
            # For j=hi: a[ki-hi] (first element of seg)
            # So we need seg reversed
            seg = seg[::-1]
            grad[lo:hi+1] += w_k * (seg - M / S)
    return grad

import numpy as np

def _fft_conv(a, N):
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)

def _score_from_b(b, n, S):
    if S < 0.01:
        return float('inf')
    return 2.0 * n * b.max() / (S * S)

def _grad_peaks(a, b, n, peak_idx, S, M):
    n_p = len(peak_idx)
    if n_p == 0:
        return np.zeros(n)
    gv = b[peak_idx]
    w = gv / gv.sum()
    grad = np.zeros(n)
    mrs = M / S
    for w_k, k in zip(w, peak_idx):
        ki = int(k)
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if lo <= hi:
            seg = a[ki - lo : ki - hi + 1][::-1]
            grad[lo : hi + 1] += w_k * (seg - mrs)
    return grad

def _normalize(a, S_t):
    s = a.sum()
    if s > 1e-30:
        a = a * (S_t / s)
    return np.clip(a, 0.0, 1000.0)

def _perturb(a, rng, n, S_t, amp=0.03):
    a = a * (1.0 + amp * rng.standard_normal(n))
    return _normalize(np.clip(a, 0.0, 1000.0), S_t)

    s0 = rng.integers(0, n - w)
    a[s0 : s0 + w] *= np.exp(rng.normal(0, 0.12))
    return _normalize(np.clip(a, 0.0, 1000.0), S_t)
