import numpy as np
import time

def _conv2(a, n):
    """Compute a*a via FFT. n = len(a)."""
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    fa = np.fft.rfft(a, L)
    return np.fft.irfft(fa * fa, L)[:2 * n - 1]

def _grad(a, w, n):
    """Gradient of sum_k w[k]*(a*a)[k] w.r.t. a."""
    L = 1
    while L < 3 * n - 2:
        L <<= 1
    fw = np.fft.rfft(w, L)
    fa = np.fft.rfft(a, L)
    x = np.fft.irfft(fw * fa.conj(), L)
    return 2.0 * x[:n]

def _score_direct(a, n):
    b = _conv2(a, n)
    s = a.sum()
    return 2.0 * n * b.max() / (s * s)
