            if time.monotonic() > deadline - 50:
                break

    # Fine-tuning phase
    a = best.copy()
    eta_ft = 0.008
    for _ in range(400):
        if time.monotonic() > deadline - 5:
            break
        b = _fft_conv(a, N)
        peak_idx = np.argpartition(b, -5)[-5:]
        grad = _subgrad(a, peak_idx, n)
        grad -= grad.mean()
        a_new = a * np.exp(-eta_ft * grad)
        a_new = _normalize(np.clip(a_new, 0.0, 1000.0), target_sum)
        sc = _score(a_new, n, N)
        if sc < best_score:
            best = a_new.copy()
            best_score = sc
            a = a_new.copy()
        elif time.monotonic() > deadline - 10:
            break

    fs = _score(best, n, N)
    if not np.isfinite(fs):
        best = np.ones(n)

    return [float(x) for x in best]
