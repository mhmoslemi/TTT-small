```python
import numpy as np
import time
import scipy.sparse
import scipy.optimize

def _fft_conv(x, y):
    m = len(x) + len(y) - 1
    sz = 1
    while sz < m:
        sz <<= 1
    fx = np.fft.rfft(x, sz)
    fy = np.fft.rfft(y, sz)
    return np.fft.irfft(fx * fy, sz)[:m]

def _score(a):
    """Compute the score directly (for tracking during optimization)."""
    n = len(a)
    conv = _fft_conv(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2 * n * conv.max() / (s * s))

def _build_conv_matrix(f):
    """Build sparse (2n-1) x n matrix A where A[k,i] = f[k-i] if valid."""
    n = len(f)
    m = 2 * n - 1
    # Column i: f[0..n-1] in rows i..i+n-1
    rows = np.concatenate([np.arange(i, i + n) for i in range(n)])
    cols = np.concatenate([np.full(n, i, dtype=np.int32) for i in range(n)])
    vals = np.tile(f, n)
    A = scipy.sparse.coo_matrix((vals, (rows, cols)), shape=(m, n)).tocsr()
    return A

def _lp_step(f, deadline):
    """LP-based improvement step. Returns improved sequence or None."""
    n = len(f)
    conv = _fft_conv(f, f)
    M = conv.max()
    total_time = deadline - time.monotonic()
    if total_time < 2.0:
        return None
    
    A = _build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    try:
        res = scipy.optimize.linprog(
            c_obj, A_ub=A, b_ub=b_ub,
            bounds=[(0, None)] * n,
            method='highs',
            options={'time_limit': max(0.1, total_time * 0.3), 'presolve': True}
        )
        if res.status == 0 and res.x is not None:
            b = np.maximum(res.x, 0)
            sb = b.sum()
            if sb < 1e-10:
                return None
            # Rescale to match sum of f
            sf = f.sum()
            b = b * (sf / sb)
            return b
    except Exception:
        pass
    return None

def _local_search(a, deadline):
    """Coordinate-wise local search."""
    n = a.size
    a = a.copy()
    conv = _fft_conv(a, a)
    s = a.sum()
    best_score = 2.0 * n * conv.max() / (s * s)
    
    improved = True
    while improved and time.monotonic() < deadline - 5.0:
        improved = False
        for i in range(n):
            if time.monotonic() > deadline - 5.0:
                break
            # Impact vector for coordinate i
            # imp[k] = a[k-i] + a[i-k] if indices valid
            # a[k-i] valid when 0 <= k-i < n, i.e., i <= k <= i+n-1
            # a[i-k] valid when 0 <= i-k < n, i.e., i-n+1 <= k <= i
            imp = np.zeros(2 * n - 1)
            lo1 = max(0, i)
            hi1 = min(2 * n - 2, i + n - 1)
            imp[lo1:hi1+1] += a[:hi1-lo1+1]  # a[k-i] for k from lo1 to hi1
            lo2 = max(0, i - n + 1)
            hi2 = min(2 * n - 2, i)
            # a[i-k] for k from lo2 to hi2: a[i-lo2], a[i-lo2-1], ...
            vals2 = a[i - hi2:i - lo2 + 1][::-1]
            imp[lo2:hi2+1] += vals2
            
            # New conv ≈ conv + delta * imp
            # Minimize max(conv + delta * imp) subject to a[i] + delta >= 0
            # 1D problem: minimize max_k (conv[k] + delta * imp[k])
            # The optimal delta: we want conv[k] + delta*imp[k] <= M_new for all k
            # If imp[k] > 0: delta <= (M_new - conv[k]) / imp[k]
            # If imp[k] < 0: delta >= (M_new - conv[k]) / imp[k]
            # If imp[k] == 0: need conv[k] <= M_new
            # Binary search on M_new, or just try a few delta values.
            
            # Simple approach: try a few multipliers
            current = a[i]
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 2.0, 0.0]:
                delta = current * (mult - 1.0)
                if delta == 0:

                a[i] += delta
                new_conv_max = (conv + delta * imp).max()
                new_s = s + delta
                if new_s > 0.01:

                        s = new_s

    return a, best_score
```

import numpy as np
import time
import scipy.sparse
import scipy.optimize


def _fft_conv(x, y):
    m = len(x) + len(y) - 1
    sz = 1
    while sz < m:
        sz <<= 1
    fx = np.fft.rfft(x, sz)
    fy = np.fft.rfft(y, sz)
    return np.fft.irfft(fx * fy, sz)[:m]


def _fast_score(a):
    n = len(a)
    conv = _fft_conv(a, a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    return float(2.0 * n * conv.max() / (s * s))


def _build_conv_mat(f):
    n = f.size
    m = 2 * n - 1
    rows_parts = [np.arange(i, i + n) for i in range(n)]
    rows = np.concatenate(rows_parts)
    cols = np.repeat(np.arange(n), n)
    vals = np.tile(f, n)
    return scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(m, n))


def _lp_improve(f, deadline):
    n = f.size
    conv = _fft_conv(f, f)
    M = float(conv.max())
    remaining = deadline - time.monotonic()
    if remaining < 3.0:
        return None
    tl = min(remaining * 0.25, 60.0)
    A = _build_conv_mat(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = scipy.optimize.linprog(
            c_obj, A_ub=A, b_ub=b_ub,
            bounds=[(0.0, None)] * n,
            method='highs',
            options={'time_limit': tl, 'presolve': True}
        )
        if res.status == 0 and res.x is not None:
            b = np.maximum(res.x, 0.0)
            sb = b.sum()
            if sb < 1e-10:
                return None
            b *= f.sum() / sb
            return b
    except Exception:
        pass
    return None


def _local_search(a, deadline):
    n = a.size
    a = a.copy()

    s = float(a.sum())

            if time.monotonic() > deadline - 8.0:
                break
            imp = np.zeros(2 * n - 1)
            lo1 = i
            hi1 = min(2 * n - 2, i + n - 1)
            imp[lo1:hi1 + 1] += a[:hi1 - lo1 + 1]
            lo2 = max(0, i - n + 1)
            hi2 = min(2 * n - 2, i)
            imp[lo2:hi2 + 1] += a[i - hi2:i - lo2 + 1][::-1]
            ci = a[i]
            best_delta = 0.0
            best_trial = best_sc
            for frac in (0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 0.0):
                delta = ci * (frac - 1.0)
                if ci + delta < -1e-15 or abs(delta) < 1e-15:
                    continue
                new_max = (conv + delta * imp).max()
                new_s = s + delta
                if new_s < 1e-10:

                trial_sc = 2.0 * n * new_max / (new_s * new_s)
                if trial_sc < best_trial:
                    best_trial = trial_sc
                    best_delta = delta

                conv += best_delta * imp
