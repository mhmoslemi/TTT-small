    return 2.0 * ca[n - 1:2 * n - 1][::-1]

def _sg(a):
    n = a.size
    s = _fc(a, a)
    k = int(s.argmax())
    idx = k - np.arange(n)
    m = (idx >= 0) & (idx < n)
    g = np.zeros(n)
    g[m] = 2.0 * a[idx[m]]
    return g

def _opt(a, dl, rng):
    n = a.size
    a = np.maximum(a, 0.0).astype(np.float64)
    s = a.sum()
    a = a / s if s > 1e-10 else np.full(n, 1.0 / n)
    best = a.copy()
    bs = float(evaluate_sequence(a.tolist()))
    tau = 5.0 / n
    lr = 0.5
    it = 0
    stall = 0
    
    while time.monotonic() < dl:
        if tau < 2.0 / n:
            g = _sg(a)
        else:
            g = _sdg(a, tau)
        st = np.clip(lr * g, -50.0, 50.0)
        an = a * np.exp(-st)
        an = np.maximum(an, 1e-30)
        an /= an.sum()
        if _qs(an) < _qs(a):
            a = an
            lr = min(lr * 1.05, 5.0)
        else:
            lr *= 0.85
            st = np.clip(lr * g, -50.0, 50.0)
            an = a * np.exp(-st)
            an = np.maximum(an, 1e-30)
            an /= an.sum()
            a = an
        
        it += 1
        if it % 20 == 0:
            tau = max(tau * 0.95, 0.5 / n)
            # Symmetrize
            a_sym = 0.5 * (a + a[::-1])
            a_sym = np.maximum(a_sym, 1e-30)
            a_sym /= a_sym.sum()
            if _qs(a_sym) < _qs(a):
                a = a_sym
        
        if it % 100 == 0:
            sc = float(evaluate_sequence(a.tolist()))
            if sc < bs - 1e-13:
                bs = sc
                best = a.copy()
                stall = 0
            else:
                stall += 1
            if stall >= 4:
                a += 0.02 * stall * rng.standard_normal(n)
                a = np.maximum(a, 0.0)
                a /= a.sum()
                tau = max(6.0 / n, tau * 1.5)
                lr = 0.5
                stall = 0
            if time.monotonic() >= dl:
                break
    
    sc = float(evaluate_sequence(a.tolist()))
    if sc < bs:
        best = a.copy()
        bs = sc
    return best, bs

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    dl = time.monotonic() + max(0.0, float(budget_s) - 20.0)
    
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0).ravel()
    if p.size < 2:
        p = np.ones(2535)
    n = p.size
    x = np.linspace(0.0, 1.0, n)
    
    inits = [
        p.copy(),
        0.5 * (1.0 + np.cos(np.pi * x)),
        np.exp(-5.0 * ((x - 0.5) / 0.2) ** 2),
        np.exp(-6.0 * ((x - 0.3) / 0.15) ** 2) + np.exp(-6.0 * ((x - 0.7) / 0.15) ** 2) + 0.01,
    ]
    
    for i in range(len(inits)):
        inits[i] = np.maximum(inits[i], 0.0)
        ss = inits[i].sum()
        inits[i] = inits[i] / ss if ss > 1e-10 else np.full(n, 1.0 / n)
    
    ba = inits[0].copy()
    bsc = float(evaluate_sequence(ba.tolist()))
    
    rem = max(dl - time.monotonic(), 10.0)
    fracs = [0.45, 0.25, 0.15, 0.15]
    
    for i, ini in enumerate(inits):
        if time.monotonic() >= dl:
            break
        t_i = fracs[i] * rem
        d = min(time.monotonic() + t_i, dl)
        b, sc = _opt(ini.copy(), d, rng)
        if sc < bsc - 1e-13:
            bsc = sc
            ba = b.copy()
    
    if not np.isfinite(ba).all() or ba.sum() < 0.01:
        ba = np.maximum(p, 0.0)
        ss = ba.sum()
        ba = ba / ss if ss > 0 else np.full(n, 1.0 / n)
    
    return [float(v) for v in ba]
