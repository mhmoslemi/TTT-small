```python
def _simplex_project(v, S):
    n = len(v)
    u = np.sort(v)[::-1]
    cs = np.cumsum(u)
    # Find largest k in {1,...,n} such that u[k-1] - (cs[k-1]-S)/k > 0
    # Equivalently: k*u[k-1] > cs[k-1] - S
    # i.e., k*u[k-1] + S > cs[k-1]
    ks = np.arange(1, n+1)
    cond = ks * u + S > cs
    rho = np.sum(cond)
    if rho == 0:
        return np.full(n, S/n)
    tau = (cs[rho-1] - S) / rho
    return np.maximum(v - tau, 0)
```
