        if time.monotonic() > deadline - 30:
            break
        b = _fft_sq(a, N)[:2*n - 1]
        M = b.max()
        x = t_sm * (b - M)
        x -= x.max()
        pw = np.exp(x)
        pw /= pw.sum()
        p_rev = pw[::-1]
        conv_res = np.fft.irfft(
            np.fft.rfft(a, N2) * np.fft.rfft(p_rev, N2), N2)
        g = 2.0 * conv_res[n-1:2*n-1][::-1] - 2.0 * (a.sum() / n)
        a = a * np.exp(-0.005 * g)
        a = _normalize(np.clip(a, 0.0, 1000.0), tgt)
        sc = evaluate_sequence(a.tolist())
        if sc < best_sc:
            best = a.copy()
            best_sc = sc

    fs = evaluate_sequence(best.tolist())
    if not np.isfinite(fs) or fs > 10.0:
        best = np.ones(n)
    return [float(x) for x in best]
