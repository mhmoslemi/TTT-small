def _fft_conv(a, N):
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)


def _score_fft(a, n, N):
    S = a.sum()

        return float('inf')
    b = _fft_conv(a, N)
    return 2.0 * n * b.max() / (S * S)


def _normalize(a, target):
    s = a.sum()
    if s > 1e-30:
        a = a * (target / s)
    return np.clip(a, 0.0, 1000.0)


def _grad_from_peaks(a, b, n, n_peaks=48, thresh_frac=0.5):

    if len(peaks) > n_peaks:

        peaks = peaks[idx]
