```python
import numpy as np
import time

def _fft_conv(a, N):
    A = np.fft.rfft(a, N)
    return A, np.fft.irfft(A * A, N)

def _grad_smooth(a, b, n, S, M, power, n_peaks):
    if M < 1e-30 or S < 1e-30:
        return np.zeros(n)
    c = np.clip(b[:2*n-1], 0.0, None)
    w_raw = c ** (power - 1)
    ws = w_raw.sum()
    if ws < 1e-30:
        return np.zeros(n)
    if n_peaks < 2*n - 1:
        peak_idx = np.argpartition(w_raw, -n_peaks)[-n_peaks:]
        pv = w_raw[peak_idx]
        ps = pv.sum()
        if ps < 1e-30:
            return np.zeros(n)
        w = pv / ps
    else:
        peak_idx = np.arange(2*n-1)
        w = w_raw / ws
    g = np.zeros(n)
    for j in range(len(w)):
        ki = int(peak_idx[j])
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if lo <= hi:
            g[lo:hi+1] += w[j] * a[ki-hi:ki-lo+1][::-1]
    g /= M
