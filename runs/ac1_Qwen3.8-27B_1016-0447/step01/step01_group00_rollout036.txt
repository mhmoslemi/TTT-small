        if time.monotonic() > deadline - 30:
            break
        for _ in range(eval_every):
            if time.monotonic() > deadline - 30:
                break
            x, mom = do_step(x, fft_size, n, T, lr, mom)
            iters += 1
        sc = float(evaluate_sequence(x.tolist()))
        T = max(0.005, T * 0.997)
        lr = max(0.03, lr * 0.998)
        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best = x.copy()
            no_imp = 0
        else:
            no_imp += eval_every
        if no_imp > eval_every * 15:
            x = best + 0.05 * rng.standard_normal(n)
            x = np.clip(x, 0.0, None)
            x /= x.sum()
            mom = np.zeros(n)
            T = max(0.05, T)
            no_imp = 0

    # Phase 2: low temperature polishing
    x = best.copy()
    mom = np.zeros(n)
    T = 0.05
    lr = 0.25
    iters = 0
    max_iters_p2 = 400000
    eval_every = 200
    no_imp = 0

    while iters < max_iters_p2:
        if time.monotonic() > deadline - 25:
            break
        for _ in range(eval_every):
            if time.monotonic() > deadline - 25:
                break
            x, mom = do_step(x, fft_size, n, T, lr, mom)
            iters += 1
        sc = float(evaluate_sequence(x.tolist()))
        T = max(0.003, T * 0.998)
        lr = max(0.02, lr * 0.998)
        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best = x.copy()
            no_imp = 0
        else:
            no_imp += eval_every
        if no_imp > eval_every * 12:
            x = best + 0.02 * rng.standard_normal(n)
            x = np.clip(x, 0.0, None)
            x /= x.sum()
            mom = np.zeros(n)
            no_imp = 0

    # Phase 3: very low temperature, fine-tune
    x = best.copy()
    mom = np.zeros(n)
    T = 0.01
    lr = 0.15
    iters = 0
    max_iters_p3 = 300000
    eval_every = 300
    no_imp = 0

    while iters < max_iters_p3:
        if time.monotonic() > deadline - 15:
            break
        for _ in range(eval_every):
            if time.monotonic() > deadline - 15:
                break
            x, mom = do_step(x, fft_size, n, T, lr, mom)
            iters += 1
        sc = float(evaluate_sequence(x.tolist()))
        T = max(0.002, T * 0.999)
        lr = max(0.01, lr * 0.999)
        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best = x.copy()
            no_imp = 0
        else:
            no_imp += eval_every
        if no_imp > eval_every * 10:
            break

    # Final validation
    final = np.clip(best, 0.0, 1000.0)
    if final.sum() < 0.01:
        final = np.ones(n) / n
    fc = float(evaluate_sequence(final.tolist()))
    if not np.isfinite(fc):
        final = np.ones(n) / n
        fc = float(evaluate_sequence(final.tolist()))
    return [float(v) for v in final]
