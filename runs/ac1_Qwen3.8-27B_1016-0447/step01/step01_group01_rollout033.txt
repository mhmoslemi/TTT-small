import numpy as np
import time


def _fft_conv(a, N):
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:2 * len(a) - 1]


def _score(a, n, N):
    S = a.sum()
    if S < 0.01:
        return 1e18
    b = _fft_conv(a, N)
    return float(2.0 * n * b.max() / (S * S))


def _pg(a, k, n):
    """Gradient of conv[k] w.r.t. a, O(n) time, O(n) space."""
    g = np.zeros(n)
    if k < n:
        g[:k + 1] = 2.0 * a[:k + 1][::-1]
    else:
        lo = k - n + 1
        g[lo:n] = 2.0 * a[lo:n][::-1]
    return g


def _norm(a, tgt):
    s = a.sum()
    if s > 1e-30:
        a = a * (tgt / s)
    return np.clip(a, 0.0, 1000.0)


def _main_loop(a, n, N, tgt, K, eta, max_iter, deadline, rng):

    b = _fft_conv(best, N)
