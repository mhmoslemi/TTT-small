import numpy as np
import time


def _fc(a, N):
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)


def _sc(a, N):
    S = a.sum()
    if S < 0.01:
        return float('inf')
    return 2.0 * a.size * _fc(a, N).max() / (S * S)


def _grad(a, b, K):
    n = a.size
    bcc = b[:2 * n - 1]
    med = np.median(bcc)
    idx = np.argpartition(bcc, -K)[-K:]
    w = np.maximum(bcc[idx] - med, 0.0)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w /= ws
    g = np.zeros(n)
    for wi, ki in zip(w, idx):
        if wi < 1e-14:
            continue
        ki = int(ki)
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        g[lo:hi + 1] += wi * a[ki - hi:ki - lo + 1][::-1]
    g -= g.mean()
    return g
