    S = a.sum()
    mean_a = S / n
    peak_idx = np.argpartition(b, -n_peaks)[-n_peaks:]
    pv = b[peak_idx]
    w_sum = pv.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    grad = np.zeros(n)
    for w, k in zip(pv / w_sum, peak_idx):
        ki = int(k)
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if lo <= hi:

    return grad
