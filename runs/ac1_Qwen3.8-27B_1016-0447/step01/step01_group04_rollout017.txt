                    b /= sb
                    scb = 2.0 * n * float(fconv(b).max())
                    if scb < sc:
                        a = b

            if not accepted and stg > 500 and it % 50 == 0:
                mix = np.exp(np.linspace(0.0, 1.0, n))
                mix /= mix.sum()
                b = 0.7 * a + 0.3 * mix
                b = np.maximum(b, 1e-18)
                sb = b.sum()
                if sb > 1e-10:
                    b /= sb
                    scb = 2.0 * n * float(fconv(b).max())
                    if scb < sc:
                        a = b
                        stg = 0

    cands = [best.copy(), par.copy()]
    idx = np.arange(1, n + 1, dtype=np.float64)
    for p in (-0.7, -0.5, -0.2, 0.2):
        cands.append(np.power(idx, p))
    cands.append(np.ones(n, dtype=np.float64))
    if n > 1:
        x = np.linspace(0.0, 1.0, n)
        cands.append(1.0 + 0.5 * np.cos(np.pi * x))
        cands.append(1.0 + 0.5 * x)
        cands.append(1.0 - 0.3 * x)

    best_f = None
    best_fs = np.inf
    for f in cands:
        if f is None or f.size != n:
            continue
        f = np.clip(np.nan_to_num(f, nan=0.0, posinf=0.0, neginf=0.0), 0.0, 1000.0)
        if f.sum() < 0.01:
            continue
        try:
            sv = float(evaluate_sequence(f.tolist()))
        except Exception:
            sv = np.inf
        if np.isfinite(sv) and sv < best_fs:
            best_fs = sv
            best_f = f.copy()

    if best_f is None:
        f = np.clip(best, 0.0, 1000.0)
        if f.sum() < 0.01:
            f = np.ones(max(1, n), dtype=np.float64)
        best_f = f

    return [float(x) for x in best_f]
