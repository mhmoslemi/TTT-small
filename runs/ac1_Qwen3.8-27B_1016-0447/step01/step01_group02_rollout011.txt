def _fft_conv(x, y):
    m = len(x) + len(y) - 1
    sz = 1
    while sz < m:
        sz <<= 1
    fx = np.fft.rfft(x, sz)
    fy = np.fft.rfft(y, sz)
    return np.fft.irfft(fx * fy, sz)[:m]

    n = a.size
    s = a.sum()
    if s < 1e-15:
        return 1e10
    return 2.0 * n * _fft_conv(a, a).max() / (s * s)
