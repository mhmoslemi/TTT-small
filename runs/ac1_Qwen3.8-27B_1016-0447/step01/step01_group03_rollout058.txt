import numpy as np

def npwp2(L):
    L = int(L)
    if L <= 1:
        return 1
    return 1 << (L - 1).bit_length()


def conv_self(p):
    n = len(p)
    if n == 0:
        return np.zeros(0)
    L = 2 * n - 1
    N = npwp2(L)
    fa = np.fft.rfft(p, N)
    out = np.fft.irfft(fa * fa, N)[:L]
    return out


def conv_ab(a, b):
    if len(a) == 0 or len(b) == 0:
        return np.zeros(0)
    L = len(a) + len(b) - 1
    N = npwp2(L)
    out = np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N), N)[:L]
    return out


def smooth_val(b, beta):
    if b.size == 0:
        return 0.0
    m = float(np.max(b))
    s = float(np.sum(np.exp(beta * (b - m))))
    if s <= 0.0 or not np.isfinite(s):
        return 1e100
    return m + np.log(s) / beta


def softmax_b(b, beta):
    if b.size == 0:
        return np.zeros(0)
    m = float(np.max(b))
    ex = np.exp(beta * (b - m))
    s = float(np.sum(ex))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(b.shape, 1.0 / b.size)
    return ex / s


def grad_lse(p, w):
    n = len(p)
    c = conv_ab(p[::-1], w)
    cross = c[n - 1:2 * n - 1]
    idx = 2 * np.arange(n)
    corr = np.zeros(n, dtype=np.float64)
    valid = idx < w.size
    corr[valid] = p[valid] * w[idx[valid]]
    g = 2.0 * cross - corr
    if not np.isfinite(g).all():
        return np.zeros(n, dtype=np.float64)
    return g


def simplex_project(v):

            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    p = np.maximum(v - theta, 0.0)
    s = float(np.sum(p))
    if s < 1e-30:
        return np.full(n, 1.0 / n, dtype=np.float64)
    p /= s
    return p

    p = a / s

    return p
