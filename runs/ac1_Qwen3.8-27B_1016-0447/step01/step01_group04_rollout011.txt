        out = np.ones(n, dtype=np.float64) / n

    try:
        sc_check = evaluate_sequence([float(x) for x in out])
        if not np.isfinite(sc_check):
            raise ValueError
    except Exception:
        out = norm_seq(parent)
        out = np.clip(out, 0.0, 1000.0)
        if out.sum() < 0.01:
            out = np.ones(n, dtype=np.float64) / n

    return [float(x) for x in out]
