import numpy as np
import time
from scipy import fft as sfft


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s)) - 12.0

    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)
    n = parent.size
    s0 = parent.sum()
    if s0 < 0.01:
        parent = np.ones(n)
        s0 = float(n)

    y = parent / s0
    best = y.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(best_score):
        best_score = 10.0
        best = np.ones(n) / n

    fft_size = next_pow2(2 * n)
    corr_size = next_pow2(3 * n)

    def run_phase(y_init, T_start, T_end, lr_start, lr_end, duration, mom):
        nonlocal best, best_score, deadline
        t0 = time.monotonic()
        T = T_start
        lr = lr_start
        v = np.zeros(n)
        y = y_init.copy()
        n_inner = max(1, duration // 0.5)
        T_decay = 1.0
        lr_decay = 1.0
        if T_start > T_end and T_start > 1e-9:
            T_decay = (T_end / T_start) ** (1.0 / max(1, n_inner))
        if lr_start > lr_end and lr_start > 1e-9:
            lr_decay = (lr_end / lr_start) ** (1.0 / max(1, n_inner))

        it_count = 0
        stall = 0
        local_best_score = best_score
        while time.monotonic() < min(deadline, t0 + duration):
            X = sfft.rfft(y, fft_size)
            conv = np.real(sfft.irfft(X * X, fft_size))[:2 * n - 1]
            conv = np.maximum(conv, 0.0)
            cmax = conv.max()

            if T > 1e-7:
                zv = np.clip((conv - cmax) / T, -80.0, 0.0)
                p = np.exp(zv)
                ps = p.sum()
                if ps < 1e-300:
                    p = np.zeros(2 * n - 1)
                    p[n - 1] = 1.0
                else:
                    p /= ps
            else:
                p = np.zeros(2 * n - 1)
                p[np.argmax(conv)] = 1.0

            P = sfft.rfft(p, corr_size)
            Y = sfft.rfft(y, corr_size)
            corr = np.real(sfft.irfft(P * np.conj(Y), corr_size))[:n]
            g = 2.0 * corr
            g -= g.mean()
            gm = cmax
            if gm > 1e-15:
                g /= gm

            v = mom * v + (1.0 - mom) * g
            v_max = np.abs(v).max()
            if v_max < 1e-15:
                v = rng.standard_normal(n)
                v /= np.linalg.norm(v)
                v_max = 1.0
            step = lr / (n * v_max)
            y_new = y - step * v
            y_new = np.maximum(y_new, 0.0)
            ss = y_new.sum()
            if ss < 1e-12:
                y_new = np.ones(n) / n
            else:
                y_new /= ss
            y = y_new

            T = max(T_end, T * T_decay)
            lr = max(lr_end, lr * lr_decay)

            it_count += 1
            if it_count % 800 == 0:
                sc = float(evaluate_sequence(y.tolist()))
                if np.isfinite(sc) and sc < local_best_score - 1e-9:
                    local_best_score = sc
                    stall = 0
                else:
                    stall += 800
                if sc < best_score - 1e-9:
                    best_score = sc
                    best = y.copy()
                if stall > 6000:
                    break

        if local_best_score < best_score - 1e-9:
            best_score = local_best_score
            # best already updated above
        return y

    # Phase 1: aggressive from parent
    y = run_phase(y, 0.8, 0.15, 1.2, 0.3, 200.0, 0.7)

    # Phase 2: refinement
    y = run_phase(y, 0.15, 0.03, 0.5, 0.1, 250.0, 0.8)

    # Phase 3: restarts with structured perturbations
    j = np.arange(n, dtype=np.float64)
    for r in range(20):
        if time.monotonic() > deadline - 20.0:
            break
        t_rem = deadline - time.monotonic()
        if t_rem < 15.0:
            break
        dur = min(40.0, t_rem * 0.4)
        # Construct perturbation
        style = r % 5
        amp = 0.05 + 0.05 * (r % 3)
        if style == 0:
            f1 = rng.uniform(1.0, 6.0)
            pert = amp * np.sin(2.0 * np.pi * f1 * j / n + rng.uniform(0, 6.28))
        elif style == 1:
            f1 = rng.uniform(2.0, 12.0)
            pert = amp * np.cos(2.0 * np.pi * f1 * j / n + rng.uniform(0, 6.28))
        elif style == 2:
            pert = amp * rng.standard_normal(n)
            # Smooth it
            from scipy.ndimage import gaussian_filter1d
            pert = gaussian_filter1d(pert, sigma=10.0)
        elif style == 3:
            # Bimodal flip
            center = n // 2
            pert = np.zeros(n)
            pert[:center // 2] = amp
            pert[3 * center // 2:] = amp
            pert[center // 2:3 * center // 2] = -amp * 0.5
        else:
            f1 = rng.uniform(0.5, 3.0)
            pert = amp * np.sin(2.0 * np.pi * f1 * j / n) * np.cos(2.0 * np.pi * rng.uniform(1, 5) * j / n)

        y_init = best + pert
        y_init = np.maximum(y_init, 0.0)
        y_init /= y_init.sum()
        y = run_phase(y_init, 0.3, 0.02, 0.8, 0.1, dur, 0.85)

    # Final polish from best
    time_left = deadline - time.monotonic()
    if time_left > 30.0:
        y = run_phase(best.copy(), 0.05, 0.005, 0.3, 0.05, time_left - 15.0, 0.9)

    # Validate and return
    final = best * s0
    final = np.clip(final, 0.0, 1000.0)
    fs = float(evaluate_sequence(final.tolist()))
    if not np.isfinite(fs):
        final = parent
    if final.sum() < 0.01:
        final = parent
    return [float(x) for x in final]
