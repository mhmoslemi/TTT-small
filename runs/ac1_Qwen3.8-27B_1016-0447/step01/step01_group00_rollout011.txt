def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 10.0)

    parent = np.asarray(height_sequence_1, dtype=np.float64)

    p = np.maximum(parent, 0.0)
    if p.sum() < 0.01:
        p = np.ones(n, dtype=np.float64)
    p = p / p.sum()

    best = p.copy()
    best_score = evaluate_sequence(best.tolist())

    fft_size = next_pow2(2 * n)
    corr_size = next_pow2(3 * n)

    def compute_conv(x):
        X = sfft.rfft(x, fft_size)
        return np.real(sfft.irfft(X * X, fft_size))[:2 * n - 1]

    def grad_step(x, T, lr):
        conv = compute_conv(x)
        conv = np.maximum(conv, 0.0)
        emin = conv.min()
        emax = conv.max()
        if emax - emin < 1e-15:
            pp = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            sh = (conv - emin) / (emax - emin)
            pp = np.exp((sh - 1.0) / T)
            ss = pp.sum()
            if ss < 1e-300:
                pp = np.zeros(2 * n - 1)
                pp.argmax()
                pp[pp.argmax()] = 1.0
            else:
                pp /= ss
        P = sfft.rfft(pp, corr_size)
        X2 = sfft.rfft(x, corr_size)
        corr = np.real(sfft.irfft(P * np.conj(X2), corr_size))[:n]
        g = 2.0 * corr
        g -= g.mean()
        x_new = x - lr * g
        x_new = np.maximum(x_new, 0.0)
        s = x_new.sum()
        if s < 1e-15:
            x_new = np.random.default_rng(0).uniform(0.01, 1.0, n)
            s = x_new.sum()
        x_new /= s
        return x_new
