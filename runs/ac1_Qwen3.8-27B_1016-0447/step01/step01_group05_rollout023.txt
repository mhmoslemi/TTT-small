import time

def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.where(u > (cssv - 1) / np.arange(1, n + 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[0]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def _score(a, n):
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (s * s)

def _smooth_grad(a, p, tau, n):
    L = 1
    while L < 3 * n:
        L <<= 1
    fa = np.fft.rfft(a, L)
    fp = np.fft.rfft(p, L)
    # s2[m] = (p * a)[m] for m=0..n-1
    s2 = np.fft.irfft(fp * fa, L)[:n]
    # s1[m] = (p * a_flipped)[m + n - 1] for m=0..n-1
    a_flipped = a[::-1]
    faf = np.fft.rfft(a_flipped, L)
    s1_full = np.fft.irfft(fp * faf, L)
    s1 = s1_full[n - 1:2 * n - 1]
    return (s1 + s2) / tau

import numpy as np
import time

def _project_simplex(v):
    n = len(v)
    if n <= 1:
        return v.copy() if n == 1 else v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u > (cssv - 1.0) / idx
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[0]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _conv_max(a):
    return np.convolve(a, a).max()

def _score_norm(a, n):
    return 2.0 * n * _conv_max(a)

def _smooth_grad(a, p, tau, n):
    L = 1
    while L < 3 * n:
        L <<= 1
    fa = np.fft.rfft(a, L)
    fp = np.fft.rfft(p, L)
    s2 = np.fft.irfft(fp * fa, L)[:n]
    af = a[::-1]
    faf = np.fft.rfft(af, L)
    s1_full = np.fft.irfft(fp * faf, L)
    s1 = s1_full[n - 1 : 2 * n - 1]
    return (s1 + s2) / tau

def _init_variants(n):
