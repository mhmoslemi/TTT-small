        inits.append(norm_a(np.power(2.0 * n + 1.0 - base, e) + 0.01))

    best = None
    best_sc = np.inf
    for a0 in inits:
        if a0.size != n:
            continue
        _, sc0 = eval_fft(a0, flen)
        if np.isfinite(sc0) and sc0 < best_sc:
            best_sc = sc0
            best = a0.copy()
    if best is None:
        best = np.ones(n, dtype=np.float64) / n
        best_sc = 2.0

    a = best.copy()
    sc = best_sc
    stagnant = 0

    while time.monotonic() < deadline:
        c, sc = eval_fft(a, flen)
        M = float(c.max())
        if np.isfinite(sc) and sc < best_sc - 1e-13:
            best_sc = sc
            best = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        if M < 1e-15:
            break

        if stagnant < 40:
            tau_frac = 0.06
        elif stagnant < 100:
            tau_frac = 0.15
        elif stagnant < 200:
            tau_frac = 0.30
        else:
            tau_frac = 0.50
        tau = max(tau_frac * M, 1e-12)

        grad = compute_grad(a, c, tau, gflen)
        S = float(a.sum())
        wa = float(np.dot(a, grad))
        rel = grad - wa / S
        rel = np.where(np.isfinite(rel), rel, 0.0)
        gmax = float(np.max(np.abs(rel)))

        improved = False
        if gmax > 1e-14:
            for maxlog in (0.02, 0.05, 0.12, 0.3, 0.7):
                eta = maxlog / gmax
                logit = np.clip(-eta * rel, -5.0, 5.0)
                b = norm_a(a * np.exp(logit))
                _, scb = eval_fft(b, flen)
                if np.isfinite(scb) and scb < sc - 1e-14:
                    a, sc = b, scb
                    improved = True
                    break

        if not improved and stagnant > 4:
            D = min(10, n)
            contrib = a * grad
            donors = np.argpartition(contrib, n - D)[-D:]
            receivers = np.argpartition(grad, D - 1)[:D]
            dsum = float(a[donors].sum())
            for mfrac in (0.004, 0.015, 0.06, 0.15):
                mmove = mfrac * dsum
                if mmove < 1e-15:
                    continue
                b = a.copy()
                wd = b[donors]
                ws = float(wd.sum())
                if ws < 1e-15:
                    break
                b[donors] = np.maximum(b[donors] - mmove * wd / ws, 0.0)
                b[receivers] += mmove / D
                b = norm_a(b)
                _, scb = eval_fft(b, flen)
                if np.isfinite(scb) and scb < sc - 1e-14:
                    a, sc = b, scb
                    improved = True
                    break

        if not improved and stagnant > 25:
            b = a.copy()
            lo = max(1, n // 250)
            hi = max(lo + 1, n // 15)
            wlen = int(rng.integers(lo, hi))
            st = int(rng.integers(0, n - wlen))
            b[st:st + wlen] *= np.exp(rng.normal(0.0, 0.08))
            b = norm_a(b)
            _, scb = eval_fft(b, flen)
            if np.isfinite(scb) and scb < sc + 0.004:
                a, sc = b, scb

        if not improved and stagnant > 130:
            restarted = False
            for t in (0.2, 0.4, 0.6):
                for init in inits[:8]:
                    b = norm_a((1.0 - t) * a + t * init)
                    _, scb = eval_fft(b, flen)
                    if np.isfinite(scb) and scb < best_sc + 0.015:
                        a, sc = b, scb
                        restarted = True
                        break
                if restarted:
                    break
            if not restarted:
                mask = 0.5 + 0.5 * rng.random(n)
                a = norm_a(a * mask)
                sc = np.inf
            stagnant = 0

    cands = [best]
    if parent is not None and parent.size == n:
        cands.append(parent)

    final_best = None
    final_sc = np.inf
    for arr in cands:
        out = np.clip(norm_a(arr), 0.0, 1000.0)
        if out.sum() < 0.01:
            out = np.ones(n) / n
        try:
            s = float(evaluate_sequence(out.tolist()))
        except Exception:
            s = np.inf
        if np.isfinite(s) and s < final_sc:
            final_sc = s
            final_best = out

    if final_best is None:
        final_best = np.clip(norm_a(best), 0.0, 1000.0)
    return [float(x) for x in final_best]
