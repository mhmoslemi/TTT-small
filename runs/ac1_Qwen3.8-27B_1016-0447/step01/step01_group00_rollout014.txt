                x = np.minimum(x, 1000.0)
                T = 2.0 * conv_typ
                lr = 0.4
                beta = 0.90
                v = np.zeros(n)
                no_improve = 0
                restarts += 1

    # Final polish: a few iterations with very low T
    if time.monotonic() < deadline - 5.0:
        T = T_min
        lr = 0.05
        v = np.zeros(n)
        for _ in range(5000):
            if time.monotonic() > deadline - 3.0:
                break
            conv = _convolve_fft(x, fft_sz)[:2 * n - 1]
            conv = np.maximum(conv, 0.0)
            cmax = conv.max()
            p = np.exp((conv - cmax) / T)
            ps = p.sum()
            if ps < 1e-300:
                p = np.zeros(2 * n - 1)
                p[n - 1] = 1.0
            else:
                p /= ps
            Xr = sfft.rfft(x[::-1], corr_sz)
            P = sfft.rfft(p, corr_sz)
            full_corr = np.real(sfft.irfft(Xr * P, corr_sz))
            g = 2.0 * full_corr[n - 1: 2 * n - 1]
            g -= g.mean()
            v = 0.8 * v - lr * g
            x = np.maximum(x + v, 0.0)
            x = x * (S / x.sum())
            x = np.minimum(x, 1000.0)
            sc = _score(x)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best = x.copy()

    result = np.clip(best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.clip(parent, 0.0, 1000.0)
        if result.sum() < 0.01:
            result = np.ones(n)

    fs = _score(result)
    if not np.isfinite(fs):
        result = np.clip(parent, 0.0, 1000.0)
        if result.sum() < 0.01:
            result = np.ones(n)

    return [float(a) for a in result]
