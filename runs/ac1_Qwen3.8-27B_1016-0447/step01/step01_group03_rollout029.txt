```python
def simplex_projection(v, n):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.sort(v)[::-1]
    cumsum = np.cumsum(v)
    rho = np.arange(1, n+1)
    cond = v * rho > (cumsum - 1)
    rho_star = np.max(rho[cond])
    theta = (cumsum[rho_star-1] - 1) / rho_star
    return np.maximum(v, theta)
```
