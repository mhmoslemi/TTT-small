import numpy as np
import time

def _fft_conv_efficient(a, b):
    """Linear convolution using next-power-of-2 FFT."""
    m = len(a) + len(b) - 1
    sz = 1
    while sz < m:
        sz <<= 1
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz), sz)[:m]

def _compute_grad(a, p):
    """Compute gradient of soft-max objective.
    a: length n
    p: length 2n-1 (softmax weights)
    Returns: length n gradient vector
    """
    n = len(a)
    c = p[::-1]  # reversed p, length 2n-1
    ca = _fft_conv_efficient(c, a)  # length (2n-1)+n-1 = 3n-2
    return 2.0 * ca[n-1:2*n-1][::-1]

def _evaluate_normalized(a):
    """Score with a already normalized to sum 1."""
    n = len(a)
    conv = _fft_conv_efficient(a, a)
    return 2.0 * n * conv.max()  # since sum(a)=1, sum²=1

def _optimize(a, deadline, rng):
    n = len(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
    else:
        a = a / s
    
    best = a.copy()
    best_score = _evaluate_normalized(best)
    
    tau = 1.0 / (4 * n)
    lr = 1.0
    stalled = 0
    it = 0
    prev_obj = None
    
    while time.monotonic() < deadline:
        # Compute soft-max weights
        conv = _fft_conv_efficient(a, a)
        z = conv / tau
        z -= z.max()
        p = np.exp(z)
        p /= p.sum()
        
        # Compute gradient

            lr *= 0.9995

3. Adaptive temperature and learning rate
4. Periodic evaluation and perturbation for escape from local minima

def _fft_conv(a, b):
    m = len(a) + len(b) - 1
    sz = 1
    while sz < m:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    fb = np.fft.rfft(b, sz)
    return np.fft.irfft(fa * fb, sz)[:m]
