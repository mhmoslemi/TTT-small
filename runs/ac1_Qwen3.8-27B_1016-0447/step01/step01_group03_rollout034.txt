        return None
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:

def _acorr(p):
    N = FZ
    fp = np.fft.rfft(p, N)
    c = np.fft.irfft(fp * fp, N)[: 2 * p.size - 1]
    return np.clip(np.real(c), 0.0, None)


def _fastsc(p):
    c = _acorr(p)
    return float(2.0 * p.size * np.max(c))


def _lobj(z):
    z = z - z.max()
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0:
        return 1e100
    p = e / s
    c = _acorr(p)
    u = float(p.size) * c
    um = float(u.max())
    ex = np.exp(CB * (u - um))
    ss = float(ex.sum())
    if ss <= 0.0 or not np.isfinite(ss):
        return 1e100
    return float((CB * um + math.log(ss)) / CB)


def _lgrad(z):
    n = int(z.size)
    z = z - z.max()
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0:
        return np.zeros(n)
    p = e / s
    c = _acorr(p)
    u = float(n) * c
    um = float(u.max())
    ex = np.exp(CB * (u - um))
    ss = float(ex.sum())
    if ss <= 0.0:
        return np.zeros(n)
    w = ex / ss
    N = FZ
    fp = np.fft.rfft(p, N)
    fw = np.fft.rfft(w, N)
    corr = np.fft.irfft(fp * np.conj(fw), N)
    conv = np.fft.irfft(fp * fw, N)
    dm = conv[:n]
    dp = np.empty(n)
    dp[0] = corr[0]
    dp[1:] = corr[N - 1 : N - n : -1]
    g = (float(n) / CB) * (dp + dm)
    if not np.isfinite(g).all():
        return np.zeros(n)
    dot = float(np.dot(p, g))
    grad = p * (g - dot)
    return grad if np.isfinite(grad).all() else np.zeros(n)


def _gen_cands(n, rng):
    out = []
    j = np.arange(n, dtype=float)
    for pp in (0.3, 0.5, 0.7, 1.0, 1.5, 2.0):
        out.append((n - j) ** pp)
        out.append((j + 1.0) ** pp)
    for frac in (0.25, 0.35, 0.5, 0.65, 0.75):
        m = max(1, int(frac * n))
        a = np.full(n, 1.0)
        a[:m] = 4.0
        out.append(a.copy())
        b = np.full(n, 1.0)
        b[:m] = 0.25
        out.append(b.copy())
    for k in (1, 2, 3):
        for amp in (0.3, 0.6, 0.9):
            a = 1.0 + amp * np.cos(2.0 * np.pi * k * j / max(1, n - 1))
            out.append(np.clip(a, 0.01, None))
    # Random smooth
    for _ in range(6):
        a = 1.0 + rng.normal(0, 0.3, n).cumsum()
        a = a - a.min() + 0.1
        out.append(a)
    return out


def _peak_shave(p, maxit=500, eta0=0.008):
    n = p.size
    p = p.copy()
    for it in range(maxit):
        c = _acorr(p)
        kstar = int(np.argmax(c))
        lo = max(0, kstar - n + 1)
        hi = min(kstar, n - 1)
        idx = np.arange(lo, hi + 1)
        partner = kstar - idx
        g = np.zeros(n)
        g[idx] = 2.0 * p[partner]
        if kstar % 2 == 0:
            i0 = kstar // 2
            if 0 <= i0 < n:
                g[i0] = p[i0]
        gm = float(g.max())
        if gm < 1e-15:
            break
        eta = eta0 * (1.0 + 0.02 * it)
        p = np.clip(p * (1.0 - eta * g / gm), 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            break
        p = p / s
    return p

                    options={"maxiter": mit, "ftol": 1e-13,
                             "gtol": 1e-9, "maxls": 30})
            except Exception:
                res = None
            if (res is not None and res.x is not None
                    and np.asarray(res.x).size == n):
                z = np.asarray(res.x, dtype=np.float64)
                z -= float(z.mean())
            p = _norm(np.exp(z - float(z.max()) + 1e-300))
            if p is not None:
                sc = float(evaluate_sequence(p.tolist()))
                if np.isfinite(sc) and sc < best_sc:
                    best, best_sc = p, sc

    p = best.copy()
    eta = 0.005
    kern = np.ones(5, dtype=np.float64) / 5.0
    pat = 0
    while time.monotonic() < dl - 15.0 and eta > 1e-5 and pat < 400:
        noise = rng.normal(0.0, eta, n)
        noise = np.convolve(noise, kern, "same")
        cand = p * np.exp(noise)
        cand = _norm(cand)
        if cand is not None:
            sc = _fastsc(cand)
            if sc < best_sc - 1e-11:
                best_sc = sc
                best, p = cand, cand.copy()
                eta = min(0.10, eta * 1.05)
                pat = 0
            else:
                eta *= 0.88
                pat += 1

    if time.monotonic() < dl - 10.0:
        shaved = _peak_shave(best, 400, 0.004)
        p = _norm(shaved)
        if p is not None:
            sc = float(evaluate_sequence(p.tolist()))
            if np.isfinite(sc) and sc < best_sc:
                best, best_sc = p, sc

    p = _norm(best)
    if p is None or p.size == 0:
        p = np.ones(n) / n
    if not np.isfinite(evaluate_sequence(p.tolist())):
        p = np.ones(n) / n
    return [float(x) for x in p]
