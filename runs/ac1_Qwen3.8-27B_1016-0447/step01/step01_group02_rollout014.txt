def _fft_conv(x, y):
    m = len(x) + len(y) - 1
    sz = 1
    while sz < m:
        sz <<= 1
    return np.fft.irfft(np.fft.rfft(x, sz) * np.fft.rfft(y, sz), sz)[:m]

    """grad[i] = 2 * sum_k w[k] * a[k-i], w has length 2n-1."""
