import numpy as np

def _flen(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def _conv(a, L):
    af = np.fft.rfft(a, n=L)
    c = np.fft.irfft(af * af, n=L)
    c = c[:2 * a.size - 1]
    return np.maximum(c, 0.0)


def _score(a, L):
    n = a.size
    c = _conv(a, L)
    M = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf, c, M
    return 2.0 * float(n) * M / (s * s), c, M


def _grad(a, w, L2):
    n = a.size
    au = np.fft.rfft(a, n=L2)
    aw = np.fft.rfft(w, n=L2)
    cc = np.fft.irfft(au * np.conjugate(aw), n=L2)
    idx = (-np.arange(n)) % L2
    g1 = cc[idx]
    prefix = np.concatenate(([0.0], np.cumsum(w)))
    w2 = prefix[n:] - prefix[:n]
    g = g1 + a * w2
    return np.where(np.isfinite(g), g, 0.0)


def _softmax(c, tau):

    if parent is not None and parent.size == n and float(np.sum(parent)) > 0.0:
        final_pool.append(parent)
    for ini in inits[:5]:
        if len(final_pool) >= 8:
            break
        final_pool.append(ini)

    best_out = None
    best_final = np.inf
    use_exact = 0 < n <= 15000
    for arr in final_pool:
        if arr is None or arr.size != n:
            continue
        o = _norm(np.asarray(arr, dtype=np.float64).copy())
        o = np.clip(o, 0.0, 1000.0)
        if float(np.sum(o)) < 0.01:
            o = np.ones(n, dtype=np.float64)
        sct = np.inf
        if use_exact:
            try:
                sct = float(evaluate_sequence([float(x) for x in o]))
            except Exception:
                sct = np.inf
        if not np.isfinite(sct):
            sct, _, _ = _score(o, L)
        if np.isfinite(sct) and sct < best_final:
            best_final = sct
            best_out = o.copy()

    if best_out is None:
        best_out = _norm(np.asarray(best, dtype=np.float64))
        best_out = np.clip(best_out, 0.0, 1000.0)
    if float(np.sum(best_out)) < 0.01:
        best_out = np.ones(n, dtype=np.float64)
    return [float(x) for x in best_out]
