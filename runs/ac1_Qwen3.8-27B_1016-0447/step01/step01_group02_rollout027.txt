    fy = np.fft.rfft(y, sz)
    return np.fft.irfft(fx * fy, sz)[:m]

    z = s / tau
    z -= z.max()
    e = np.exp(z)
    p = e / e.sum()
    c = p[::-1]

    idx = np.arange(max(0, k-n+1), min(n-1, k)+1)

    return np.fft.irfft(np.fft.rfft(x, sz) * np.fft.rfft(y, sz), sz)[:m]

def _fc(x, y):
    m = len(x) + len(y) - 1
    s = 1
    while s < m:
        s <<= 1
    return np.fft.irfft(np.fft.rfft(x, s) * np.fft.rfft(y, s), s)[:m]

def _ac(a):
    return _fc(a, a)

    n = a.size
