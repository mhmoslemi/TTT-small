        x = idx / float(n - 1)
        for amp in (0.3, 0.6, 1.0):
            a = 1.0 + amp * np.cos(np.pi * x)
            np.maximum(a, 1e-10, out=a)
            res.append(a / float(np.sum(a)))
    if parent is not None:
        for eps in (0.01, 0.05, 0.15):
            a = parent * (1.0 + eps * rng.standard_normal(n))
            a = np.maximum(a, 0.0)
            ss = float(np.sum(a))
            if ss > 1e-10:
                res.append(a / ss)
    return res


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    parent = get_parent()
    n = parent.size if parent is not None else 1024
    if n < 1:
        n = 1
        parent = np.ones(1)
    flen = npx2(3 * n - 2)
    rng = np.random.default_rng(seed)

    inits = make_inits(n, parent, rng)
    best = None
    best_sc = np.inf
    for a0 in inits:
        sc, _, _ = conv_and_score(a0, flen)
        if sc < best_sc:
            best_sc = sc
            best = a0.copy()
    if best is None:
        best = np.ones(n) / float(n)
        best_sc = 2.0

    a = best.copy()
    start = time.monotonic()
    deadline = start + budget_s - 60.0

    tau = 0.003
    base_lr = 0.4
    mom = 0.25
    v = np.zeros(n)
    stagnant = 0
    phase = 0

    while time.monotonic() < deadline:
        sc, c, M = conv_and_score(a, flen)
        if sc < best_sc - 1e-13:
            best_sc = sc
            best = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        if M < 1e-18:
            break

        # Softmax weights on auto-conv
        z = (c - M) / tau
        np.clip(z, -60.0, 0.0, out=z)
        w = np.exp(z)
        ws = float(np.sum(w))
        if ws < 1e-30:
            w = np.zeros(2 * n - 1)
            w[n - 1] = 1.0
        else:
            w /= ws

        # Gradient of soft-max objective
        g = gradient(a, w, flen)
        g = np.where(np.isfinite(g), g, 0.0)
        g_c = g - float(np.mean(g))

        # Momentum
        v = mom * v + (1.0 - mom) * g_c
        v_max = float(np.max(np.abs(v)))

        if v_max < 1e-16:
            a = a * (1.0 + 0.008 * rng.standard_normal(n))
            a = np.maximum(a, 0.0)
            ss = float(np.sum(a))
            if ss > 1e-10:
                a /= ss
            else:
                a = best.copy()
            v = np.zeros(n)
            continue

        accepted = False
        for lr in (base_lr, base_lr * 0.4, base_lr * 0.1):
            step = lr * v / v_max
            np.clip(step, -6.0, 6.0, out=step)
            a_new = a * np.exp(-step)
            a_new = np.maximum(a_new, 0.0)
            sn = float(np.sum(a_new))
            if sn < 1e-15:
                continue
            a_new /= sn
            sc_new, _, _ = conv_and_score(a_new, flen)
            if sc_new < sc - 1e-14:
                a = a_new
                sc = sc_new
                accepted = True
                break

        if not accepted:
            scale = 0.003 * min(stagnant, 100) + 0.002
            a = a * (1.0 + scale * rng.standard_normal(n))
            a = np.maximum(a, 0.0)
            ss = float(np.sum(a))
            if ss > 1e-10:
                a /= ss
            v *= 0.5

        # Adaptive temperature
        if stagnant > 20:
            tau *= 0.98
        tau = max(tau, 1e-6)
        if stagnant > 100:
            tau = min(tau * 1.5, 0.01)

        # Restart from best if very stuck
        if stagnant > 300:
            blend_t = 0.5 + 0.3 * rng.random()
            a = (1.0 - blend_t) * best + blend_t * (best * (1.0 + 0.05 * rng.standard_normal(n)))
            a = np.maximum(a, 0.0)
            ss = float(np.sum(a))
            a = a / ss if ss > 1e-10 else best.copy()
            v = np.zeros(n)
            tau = 0.01
            stagnant = 0

        # Periodic diverse restart
        if stagnant > 500:
            a0 = make_inits(n, parent, rng)[int(rng.integers(0, len(make_inits(n, parent, rng))))]
            a = (0.3 * a + 0.7 * a0)
            a = np.maximum(a, 0.0)
            ss = float(np.sum(a))
            a = a / ss if ss > 1e-10 else best.copy()
            v = np.zeros(n)
            tau = 0.005
            stagnant = 0

    # Final polish: small random perturbations
    for _ in range(50):
        a_try = best * (1.0 + 0.003 * rng.standard_normal(n))
        a_try = np.maximum(a_try, 0.0)
        ss = float(np.sum(a_try))
        if ss < 1e-10:
            continue
        a_try /= ss
        sc_try, _, _ = conv_and_score(a_try, flen)
        if sc_try < best_sc:
            best_sc = sc_try
            best = a_try.copy()

    # Verify and return
    out = [float(x) for x in best]
    for i in range(len(out)):
        if not np.isfinite(out[i]) or out[i] < 0:
            out[i] = 1.0 / float(n)
    if len(out) == 0 or sum(out) < 0.01:
        out = [1.0 / float(n)] * n
    return out
