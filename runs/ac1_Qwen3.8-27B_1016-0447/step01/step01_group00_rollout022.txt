def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1

def _normalize(a, S):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        a = np.maximum(a, 1e-4)
        s = a.sum()
    a *= S / s
    return np.clip(a, 0.0, 1000.0)
