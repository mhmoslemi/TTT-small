import numpy as np
import time


def _fft_conv(a, N):
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)


def _compute_grad(a, b, n, m, temp):
    idx = np.argpartition(b, -m)[-m:]
    w = b[idx]
    w = np.exp((w - w.max()) * temp)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    for wi, k in zip(w, idx):
        ki = int(k)
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if lo <= hi:
            grad[lo:hi + 1] += wi * a[ki - hi:ki - lo + 1][::-1]
    grad -= np.mean(grad)
    return grad


def _normalize(a, target_sum):
    s = a.sum()
    if s > 1e-30:
        a = a * (target_sum / s)
    return np.clip(a, 0.0, 1000.0)


def _score_from_b(b, n, S):
