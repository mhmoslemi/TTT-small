import numpy as np
import time
from scipy import fft as sfft


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_score(arr):
    return float(evaluate_sequence(arr.tolist()))


def run_gd(x, n, fsz, clen, tgt, T, lr, max_it, dl, rng, lim_mult):
    best = x.copy()
    best_sc = compute_score(best)
    noimp = 0
    lim = int(lim_mult * 5000)
    for it in range(max_it):
        if time.monotonic() > dl:
            break
        X = sfft.rfft(x, fsz)
        conv = np.maximum(np.real(sfft.irfft(X * X, fsz)), 0.0)[:clen]
        cmax = conv.max()
        if T < 1e-4:
            mask = conv >= cmax - 1e-8
            p = np.zeros(clen)
            p[mask] = 1.0 / mask.sum()
        else:
            p = np.exp((conv - cmax) / T)
            ps = p.sum()
            if ps > 1e-300:
                p /= ps
            else:
                p = np.zeros(clen)
                p[n - 1] = 1.0
        P = sfft.rfft(p, fsz)
        g = 2.0 * np.real(sfft.irfft(P * np.conj(X), fsz))[:n]
        g -= g.mean()
        x = np.clip(x - lr * g, 0.0, 1e4)
        xs = x.sum()
        if xs < 0.01:
            x = np.maximum(x, 0.05)
            xs = x.sum()
        x *= tgt / xs
        T = max(0.01, T * 0.9985)
        if it > 12000:
            lr = max(0.02, lr * 0.998)
        if it % 400 == 0:
            sc = compute_score(x)
            if np.isfinite(sc) and sc < best_sc - 1e-12:
                best_sc = sc
                best = x.copy()
                noimp = 0
            else:
                noimp += 400
            if noimp > lim:
                x = best + 0.35 * rng.standard_normal(n)
                x = np.maximum(x, 0.01)
                x *= tgt / max(x.sum(), 0.01)
                T = max(0.05, best_sc * 0.2)
                lr = 0.4
                noimp = 0
        if it % 10000 == 0 and it > 0:
            x += 0.04 * rng.standard_normal(n)
            x = np.maximum(x, 0.0)
            x *= tgt / x.sum()
    return best, best_sc


def make_alt_init(n, rng, tgt):
    j = np.arange(n, dtype=np.float64)
    x = np.ones(n)
    x += 0.4 * np.sin(2 * np.pi * 11 * j / n + rng.uniform(0, 2 * np.pi))
    x += 0.25 * np.sin(2 * np.pi * 29 * j / n + rng.uniform(0, 2 * np.pi))
    x += 0.1 * np.cos(4 * np.pi * 3 * j / n + rng.uniform(0, 2 * np.pi))
    x = np.maximum(x, 0.02)
    x *= tgt / x.sum()
    return x
