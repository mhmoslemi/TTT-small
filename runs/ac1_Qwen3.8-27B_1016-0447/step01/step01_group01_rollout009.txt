def _fft_conv(a, N):

    return np.fft.irfft(A * A, N)

        A = rng.uniform(0.1, 0.8)
        return 1.0 + A * np.cos(2 * np.pi * f * i / n + rng.uniform(0, 2*np.pi))
    elif kind == 2:
        f1 = rng.integers(1, 10)
        f2 = rng.integers(1, 10)
        A1 = rng.uniform(0.1, 0.6)
        A2 = rng.uniform(0.1, 0.6)
        return 1.0 + A1*np.cos(2*np.pi*f1*i/n) + A2*np.cos(2*np.pi*f2*i/n)
    elif kind == 3:
        return 1.0 + 0.3 * rng.standard_normal(n)
    elif kind == 4:
        # Two-bump
        c1 = rng.integers(n//4, 3*n//4)
        c2 = (c1 + n//2) % n
        s1 = n * rng.uniform(0.03, 0.15)
        s2 = n * rng.uniform(0.03, 0.15)
        return np.exp(-0.5*((i-c1)/s1)**2) + np.exp(-0.5*((i-c2)/s2)**2) + 0.1
    else:
        return 1.0 + 0.5 * rng.standard_normal(n)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 10.0)
    
    # Get parent
    try:
        pa = np.asarray(height_sequence_1, dtype=np.float64)
        if pa.ndim != 1 or pa.size == 0 or not np.all(np.isfinite(pa)):
            raise ValueError
        a_parent = np.clip(pa, 0.0, 1000.0)
    except (NameError, ValueError):
        a_parent = np.ones(6962)
    
    n = a_parent.size
    N = 1 << (2 * n - 1).bit_length()
    S_t = np.sqrt(2.0 * n)
    
    # Build initializations
    inits = []
    inits.append(_norm_seq(a_parent.copy(), S_t))
    for k in range(1, 10):
        inits.append(_norm_seq(np.maximum(0, _init_type(n, rng, k)), S_t))
    
    best = inits[0].copy()
    b_best = _fft_conv(best, N)
    best_score = 2.0 * n * b_best[:2*n-1].max() / (S_t * S_t)
    
    for idx_init in range(len(inits)):
        if time.monotonic() > deadline - 60.0:
            break
        
        a = inits[idx_init].copy()
        # Small perturbation
        a = _norm_seq(np.maximum(0.0, a * (1.0 + 0.01 * rng.standard_normal(n))), S_t)
        
        eta = 20.0
        T = 2.0
        patience = 0
        max_pat = 200
        local_best = a.copy()
        local_score = best_score
        
        for it in range(200000):
            if time.monotonic() > deadline - 30.0:
                break
            
            b = _fft_conv(a, N)
            bv = b[:2*n-1]
            S = a.sum()
            sc = 2.0 * n * bv.max() / (S * S)
            
            if sc < local_score - 1e-10:
                local_score = sc
                local_best = a.copy()
                patience = 0
            else:
                patience += 1
            
            if sc < best_score:
                best_score = sc
                best = a.copy()
            
            if patience >= max_pat:
                a = local_best * (1.0 + 0.05 * rng.standard_normal(n))
                a = _norm_seq(np.maximum(0.0, a), S_t)
                eta = min(eta * 1.5, 80.0)
                patience = 0
                T = max(T * 1.3, 5.0)
                if max_pat < 40:
                    break
                max_pat = max(40, max_pat * 3 // 4)
            
            g = _grad_lse(bv, a, n, T, n_act=128)
            
            # Mirror descent
            a = a * np.exp(-eta * g)
            a = _norm_seq(a, S_t)
            
            # Anneal
            eta *= 0.9998
            T = max(0.05, T * 0.998)
            
            if it % 200 == 0 and rng.random() < 0.3:
                # Block perturbation
                w = rng.integers(max(10, n//100), max(20, n//20))
