import numpy as np
import time
from scipy import fft as sfft


def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _normalize(x, target_sum):
    s = x.sum()
    if s < 0.01:
        x = np.maximum(x, 0.01)
        s = x.sum()
    x = x * (target_sum / s)
    return np.clip(x, 0.0, 1000.0)


def _fft_conv(x, fft_size):
    X = sfft.rfft(x, fft_size)
    return np.real(sfft.irfft(X * X, fft_size))


def _make_init(n, rng, base):
    j = np.arange(n, dtype=np.float64)
    x = base.copy() if base.size == n else np.ones(n)
    x += 0.15 * np.sin(2.0 * np.pi * 7.0 * j / n + rng.uniform(0, 6.28))
    x += 0.08 * np.sin(2.0 * np.pi * 19.0 * j / n + rng.uniform(0, 6.28))
    x += 0.05 * rng.standard_normal(n)
    x = np.maximum(x, 0.01)
    return x


def _subgrad_step(x, T, lr, momentum, beta, fft_size, corr_size, n):
    X = sfft.rfft(x, fft_size)
    conv = np.real(sfft.irfft(X * X, fft_size))[:2 * n - 1]
    conv = np.maximum(conv, 0.0)

    cmax = conv.max()
    logits = np.clip((conv - cmax) / T, -50.0, 0.0)
    p = np.exp(logits)
    ps = p.sum()
    if ps < 1e-300:
        p = np.zeros(2 * n - 1)
        p[(2 * n - 1) // 2] = 1.0
    else:
        p /= ps

    P = sfft.rfft(p, corr_size)
    Xc = sfft.rfft(x, corr_size)
    corr = np.real(sfft.irfft(P * np.conj(Xc), corr_size))
    g = 2.0 * corr[:n]
    g -= g.mean()
    gmax = np.abs(g).max()
    if gmax > 1e5:
        g *= 1e5 / gmax

    momentum[:] = beta * momentum + (1.0 - beta) * g
    x_new = x - lr * momentum
    x_new = np.clip(x_new, 0.0, 1000.0)
    return x_new, conv
