Actually, for the scipy optimizer approach, we can use the log-sum-exp smoothing of the max:
`LSE(p) = (1/beta) * log(sum_k exp(beta * (p*p)[k]))`

This is a smooth approximation to `max_k (p*p)[k]` that becomes exact as `beta -> infinity`.

The gradient of `(p*p)[k]` with respect to `p` can be computed efficiently using FFT.

Let me think about the gradient. Let `c = np.convolve(p, p)` (length 2n-1). For each `p_i`:
`dc[k]/dp[i] = p[i + k - (n-1)] + p[i - k + (n-1)]` (with boundary handling)

Actually, the gradient of `c[k] = sum_j p[j]*p[j+k-(n-1)]` with respect to `p[i]`:
- Term where `j = i`: `p[i + k - (n-1)]`
- Term where `j + k - (n-1) = i`, i.e., `j = i - k + (n-1)`: `p[i]` → no wait...

Let me be more careful. `c[k] = sum_j p[j] * p[j + k - (n-1)]` where both indices must be in [0, n-1].

`dc[k]/dp[i] = p[i + k - (n-1)] * I(0 <= i+k-(n-1) <= n-1) + p[i - k + (n-1)] * I(0 <= i-k+(n-1) <= n-1)`

This is essentially `w[k] * (shifted p)`. We can compute the full gradient as:
`grad[i] = sum_k w[k] * (dc[k]/dp[i])`

which is a convolution-like operation. Specifically:
`grad = w_convolved_with_p + w_reflected_convolved_with_p`

More precisely, if we define `w = exp(beta * c) / sum(exp(beta * c))` (the softmax weights), then:
`grad[i] = sum_k w[k] * p[i+k-(n-1)] + sum_k w[k] * p[i-k+(n-1)]`

This is `conv(w, p) + conv(flip(w), p)` evaluated at appropriate indices. We can do this with FFT in O(n log n).

2. Use L-BFGS-B or projected gradient to minimize the smoothed max autocorrelation.
3. Use FFT for all convolution/gradient computations.

5. Anneal beta from low to high.

Let me also think about what sequences are good. For minimizing max autocorrelation:
- We want the autocorrelation to be as flat as possible.
- For probability distributions, this is related to "minimum redundancy" sequences.
- Step functions (piecewise constant) are good because they create a structured autocorrelation.
- The "optimal" shape tends to be somewhat like a half-sine or a raised cosine, spreading mass to flatten the peak.

Actually, for a step function with two levels, the autocorrelation will have a distinctive structure. Let me think about what minimizes max autocorrelation...

For a two-level step function: first m values are h, remaining n-m values are l, with m*h + (n-m)*l = 1.
The autocorrelation at lag 0: m*h^2 + (n-m)*l^2
The autocorrelation at lag d: depends on the overlap.

For a "bang-bang" sequence (binary), the minimum peak sidelobe is related to Barker codes etc. But we're in continuous nonnegative.

```python
import numpy as np
import time
import scipy.optimize as sopt
import sys

TIME_DEADLINE = 0.0
G_BETA = 50.0
G_N = 0
G_NFFT = 0

class _ST(Exception):
    pass

def _cb(_):
    if time.monotonic() > TIME_DEADLINE:
        raise _ST

def _init_n(n):
    global G_N, G_NFFT
    G_N = n
    L = 3 * n - 2
    G_NFFT = 1 << (L - 1).bit_length()

def _autocorr(p, N=None):
    n = len(p)
    L = 2 * n - 1
    if N is None:
        N = 1 << (L - 1).bit_length()
    fp = np.fft.rfft(p, N)
    return np.fft.irfft(fp * fp, N)[:L]

def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.full(z.shape, 1.0/z.size)
    return e / s

def _obj(z):
    n = G_N
    p = _softmax(z)
    c = _autocorr(p, 1 << ((2*n-2)).bit_length())
    b = G_BETA
    u = b * c
    um = u.max()
    s = np.exp(u - um).sum()
    if s < 1e-300 or not np.isfinite(s):
        return 1e100
    return float((b * um + np.log(s)) / b + 1e-6 * np.mean(z)**2)

def _grad(z):
    n = G_N
    p = _softmax(z)
    c = _autocorr(p, 1 << ((2*n-2)).bit_length())
    b = G_BETA
    u = b * c
    um = u.max()
    e = np.exp(u - um)
    s = e.sum()
    if s < 1e-300:
        return np.zeros(n)
    w = e / s  # length 2n-1
    
    N = G_NFFT
    fp = np.fft.rfft(p, N)
    fw = np.fft.rfft(w, N)
    fwf = np.fft.rfft(w[::-1], N)
    
    conv1 = np.fft.irfft(fp * fwf, N)[:3*n-2]
    conv2 = np.fft.irfft(fp * fw, N)[:3*n-2]
    
    grad_p = conv1[n-1:2*n-1] + conv2[n-1:2*n-1]
    
    dot = float(np.dot(p, grad_p))
    gz = p * (grad_p - dot)
    gz += 2e-6 * np.mean(z)
    
    if not np.isfinite(gz).all():
        return np.zeros(n)
    return gz

def _norm_seq(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1 or a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01 or not np.isfinite(s):
        return None
    return a / s

def _score(a):
    p = _norm_seq(a)
    if p is None:
        return float('inf')
    return float(evaluate_sequence(p.tolist()))

def _heuristic_cands(n):
    cands = []
    j = np.arange(n, dtype=np.float64)
    cands.append(np.ones(n))
    # Power laws
    for pw in (0.0, -0.3, -0.5, 0.3, 1.0):
        a = (j + 1.0) ** pw
        if np.isfinite(a).all() and a.min() > 0:
            cands.append(a)
    # Symmetric power
    for pw in (-0.3, -0.5, 0.3):
        a = (np.abs(j - (n-1)/2.0) + 1.0) ** pw
        if np.isfinite(a).all() and a.min() > 0:
            cands.append(a)
    # Step functions
    for frac in (0.3, 0.4, 0.5, 0.6, 0.7):
        m = int(round(frac * n))
        a = np.ones(n)
        a[:m] = 2.0
        cands.append(a)
        a = np.ones(n)
        a[m:] = 2.0
        cands.append(a)
    # Cosine taper
    a = 0.5 + 0.5 * np.cos(2 * np.pi * j / n)
    a = np.clip(a, 0.01, None)
    cands.append(a)
    # Raised cosine
    a = 0.5 * (1 + np.cos(2 * np.pi * j / n))
    a = np.clip(a, 0.01, None)
    cands.append(a)
    # Two-step symmetric
    for m1 in (0.3, 0.4, 0.5):
        m = int(round(m1 * n / 2))
        a = np.ones(n)
        a[:m] = 3.0
        a[m:2*m] = 1.5
        a[n-2*m:n-m] = 1.5
        a[n-m:n] = 3.0
        cands.append(a)
    return cands

def _optimize(p0, wall_dl):
    global G_BETA
    _init_n(len(p0))
    z = np.log(p0 + 1e-300)
    z -= z.mean()
    amax = np.abs(z).max()
    if amax > 40:
        z *= 40 / amax
    
    best_z = z.copy()
    best_val = _obj(z)
    
    betas = [8, 16, 32, 64, 128, 256, 512, 1024]
    n = G_N
    if n > 5000:
        fracs = [0.3]
    elif n > 2000:
        fracs = [0.5]
    else:
        fracs = [1.0]
    
    for beta in betas:
        if time.monotonic() > wall_dl - 30:
            break
        G_BETA = float(beta)
        
        mit = int(80 * fracs[0])
        if n > 5000:
            mit = int(mit * 0.5)
        
        try:
            res = sopt.minimize(
                _obj, z, jac=_grad, method='L-BFGS-B',
                callback=_cb,
                options={'maxiter': mit, 'ftol': 1e-12, 'gtol': 1e-8, 'maxls': 30}
            )
            if res.x is not None and np.isfinite(res.fun):
                z = res.x - res.x.mean()
                amax = np.abs(z).max()
                if amax > 60:
                    z *= 60 / amax
        except _ST:
            break
        except Exception:
            pass
        
        v = _obj(z)
        if np.isfinite(v) and v < best_val:
            best_val = v
            best_z = z.copy()
    
    return _softmax(best_z)

def _polish(p, wall_dl, rng):
    p = p.copy()
```
