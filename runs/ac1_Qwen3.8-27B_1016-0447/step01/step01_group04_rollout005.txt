import numpy as np

def next_power_two(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def normalize(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, None)
    s = float(np.sum(a))
    if not np.isfinite(s) or s < 1e-300:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.size)
    return a / s


def full_conv(a, flen):
    n = a.size
    af = np.fft.rfft(a, n=flen)
    c = np.fft.irfft(af * af, n=flen)
    return np.maximum(c[:2 * n - 1], 0.0)


def wconv(w, a, flen):
    r = a[::-1]
    wf = np.fft.rfft(w, n=flen)
    rf = np.fft.rfft(r, n=flen)
    out = np.fft.irfft(wf * rf, n=flen)
    n = a.size
    return out[n - 1:n - 1 + n]
