import numpy as np
import time

def _fft_conv(a):
    n = len(a)
    m = 2 * n - 1
    sz = 1
    while sz < m:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa * fa, sz)[:m]

def _score_direct(a):
    """Compute score without calling evaluate_sequence."""
    n = a.size
    conv = _fft_conv(a)
    max_b = conv.max()
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

def _grad_smoothmax(a, m=8):
    """Gradient of smoothed max of convolution(a,a)."""
    n = a.size
    conv = _fft_conv(a)
    # Top m lags
    if m >= len(conv):
        m = len(conv)
    idx = np.argpartition(conv, -m)[-m:]
    top = conv[idx]
    # Softmax weights over top-m
    tmax = top.max()
    tmin = top.min()
    range_v = tmax - tmin
    if range_v < 1e-15:
        w = np.ones(m) / m
    else:
        beta = 5.0 / range_v  # sensitivity parameter
        w = np.exp(beta * (top - tmax))
        w /= w.sum()
    
    # Gradient: for each lag k, contribution is 2*a[k-j] for valid j
    grad = np.zeros(n)
    for wi, ki in zip(w, idx):
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if lo <= hi:
            grad[lo:hi+1] += 2.0 * wi * a[ki - hi:ki - lo + 1][::-1]
    
    # Normalize by sum of a so the gradient is scale-invariant
    s = a.sum()
    if s > 1e-30:
        grad /= s
    return grad, conv.max(), idx[0]
