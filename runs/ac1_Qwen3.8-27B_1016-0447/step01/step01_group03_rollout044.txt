```python
from scipy.sparse import csr_matrix
from scipy.optimize import linprog

def solve_dual_lp(a, M):
    """Solve: max sum(b) s.t. A@b <= M, b >= 0 where A is convolution matrix"""
    n = len(a)
    L = 2*n - 1
    
    # Build sparse convolution matrix A (L x n)
    # A[k, j] = a[k-j] if 0 <= k-j < n
    rows = []
    cols = []
    vals = []
    for k in range(L):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(L, n))
    
    # max 1^T b s.t. A b <= M, b >= 0
    c = -np.ones(n)  # minimize -sum(b)
    res = linprog(c, A_ub=A, b_ub=np.full(L, M), bounds=[(0, None)]*n, method='highs')
    return res.x
```

```python
def solve_lp(a, M):
    """Solve max sum(b) s.t. A(b) <= M, b >= 0 where (a*b)[k] <= M"""
    n = len(a)
    L = 2*n - 1
    
    # Build sparse conv matrix A (L x n): A[k,j] = a[k-j] if 0<=k-j<n
    # Column j: rows j to j+n-1, values a[0] to a[n-1]
    j = np.arange(n)
    i_offset = np.arange(n)
    rows = (j[:, None] + i_offset[None, :]).ravel()
    cols = np.repeat(j, n)
    vals = np.tile(a, n)
    A = csr_matrix((vals, (rows, cols)), shape=(L, n))
    
    c_obj = -np.ones(n)
    b_ub = np.full(L, M)
    
    res = sopt.linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.success and res.x is not None:
        return np.clip(res.x, 0, None)
    return None
```

    fp = np.fft.rfft(p, N)
    return np.real(np.fft.irfft(fp * fp, N)[:L])
