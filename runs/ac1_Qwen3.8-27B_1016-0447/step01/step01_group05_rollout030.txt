        np.exp(-4.0 * (t - 0.5) ** 2),
        np.sin(np.pi * (i + 1.0) / (n + 1.0)),
        1.0 / (i + 1.0),
        c ** 4,
        np.exp(-2.0 * i / n),
    ]


def _optimize(a, deadline, rng):
    n = len(a)
    a = _norm1(a.astype(np.float64))
    if a is None:
        return None, np.inf
    B = _fft_conv(a, a)
    cmax = B.max()
    best_a = a.copy()
    best_max = cmax
    step = 0
    stale = 0
    beta = 15.0
    L2 = 1
    while L2 < 2 * n - 1:
        L2 <<= 1
    L3 = 1
    while L3 < 3 * n - 2:
        L3 <<= 1
    fa_cache = None
    while time.monotonic() < deadline and step < 400_000:
        if cmax > 1e-15:
            W = np.exp(beta * (B - cmax) / cmax)
        else:
            W = np.ones(len(B)) / len(B)
        W += 1e-30
        W /= W.sum()
        V = W[::-1]
        fa = np.fft.rfft(a, L3)
        cWa = np.fft.irfft(fa * np.fft.rfft(W, L3), L3)[:3 * n - 2]
        cVa = np.fft.irfft(fa * np.fft.rfft(V, L3), L3)[:3 * n - 2]
        grad = np.empty(n)
        grad[:n] = cWa[:n]
        grad += cVa[2 * n - 2 : n - 1 : -1]
        gn = np.linalg.norm(grad)
        if gn < 1e-18:
            break
        lr = 0.5 / (gn + 1e-15)
        ok = False
        for _ in range(25):
            at = np.maximum(a - lr * grad, 0.0)
            st = at.sum()
            if st < 1e-15:
                lr *= 0.5
                continue
            at /= st
            fa_t = np.fft.rfft(at, L2)
            Bt = np.fft.irfft(fa_t * fa_t, L2)[:2 * n - 1]
            nm = Bt.max()
            if nm <= cmax + 1e-12:
                a, B, cmax = at, Bt, nm
                ok = True
                break
            lr *= 0.5
        if not ok:
            beta = max(3.0, beta * 0.97)
            if step > 500:
                break
            continue
        if cmax < best_max * (1.0 - 1e-9):
            best_a, best_max, stale = a.copy(), cmax, 0
        else:
            stale += 1
        step += 1
        if stale > 1500:
            a = best_a * np.exp(rng.normal(0.0, 0.35, n))
            a = _norm1(a)
            if a is None:
                break
            fa_p = np.fft.rfft(a, L2)
            B = np.fft.irfft(fa_p * fa_p, L2)[:2 * n - 1]
            cmax = B.max()
            beta = 15.0
            stale = 0
    return best_a, best_max


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    deadline = time.monotonic() + max(0.0, float(budget_s)) - 45.0
    rng = np.random.default_rng(seed)
    a0 = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if a0.size == 0 or not np.all(np.isfinite(a0)):
        a0 = np.ones(256)
    n = a0.size
    a0 = a0 / a0.sum()
    fa0 = np.fft.rfft(a0, 1)
    while fa0.size < 2 * n - 1:
        Lz = 1
        while Lz < 2 * n - 1:
            Lz <<= 1
        break
    fz = np.fft.rfft(a0, Lz)
    Bz = np.fft.irfft(fz * fz, Lz)[:2 * n - 1]
    best_sc = 2.0 * n * Bz.max()
    best_a = a0.copy()
    inits = [a0.copy()] + _inits(n)
    rem = deadline - time.monotonic()
    tper = rem / len(inits)
    for init in inits:
        if time.monotonic() >= deadline:
            break
        init = np.maximum(init, 0.0)
        s = init.sum()
        if s < 1e-12:
            continue
        init = init / s
        dl = min(time.monotonic() + tper * 2.0, deadline)
        ba, bm = _optimize(init, dl, rng)
        if ba is not None and 2.0 * n * bm < best_sc:
            best_sc = 2.0 * n * bm
            best_a = ba.copy()
    if time.monotonic() < deadline:
        ba, bm = _optimize(best_a, deadline, rng)
        if ba is not None and 2.0 * n * bm < best_sc:
            best_sc = 2.0 * n * bm
            best_a = ba.copy()
    res = [float(x) for x in best_a]
    fs = float(evaluate_sequence(res))
    if not np.isfinite(fs):
        res = [float(x) for x in a0]
    return res
