                    bsc, bst, old = s, a.copy(), 0
        if not ok:
            D = min(48, n)
            ct = a * mom
            dn = np.argpartition(ct, n - D)[-D:]
            rc = np.argsort(mom)[:D]
            for mf in (0.004, 0.015, 0.04):
                b = a.copy()
                ds = float(np.sum(b[dn]))
                mv = min(mf * S, 0.45 * ds)
                if mv < 1e-15:
                    continue
                wd = b[dn]
                swd = float(np.sum(wd))
                if swd > 0:
                    b[dn] -= mv * wd / swd
                b[rc] += mv / float(D)
                b = norm1(b)
                sb, _, _ = esc(b, fl)
                if sb < s - 1e-12:
                    a, s, ok = b, sb, True
                    if s < bsc - 1e-13:
                        bsc, bst, old = s, a.copy(), 0
                    break
        if not ok:
            b = a.copy()
            bw = max(4, n // 50)
            bs = int(rng.integers(0, max(1, n - bw)))
            b[bs:bs + bw] *= float(np.exp(rng.normal(0.0, 0.2)))
            b = norm1(b)
            sb, _, _ = esc(b, fl)
            if sb < bsc + 0.02:
                a, s = b, sb
    return bst, bsc

def propose_candidate(seed=42, budget_s=1100, **kwargs):

    t0 = time.monotonic()
    tot = max(1.0, float(budget_s) - 40.0)
