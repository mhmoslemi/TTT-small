import numpy as np
import time

def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def fft_conv_sq(a):
    """Compute full a*a via FFT."""
    n = len(a)
    L = next_pow2(2*n - 1)
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)
    return c[:2*n-1]

def fft_cross_corr(w, a):
    """Compute sum_m w[j+m]*a[m] for j=0..n-1."""
    n = len(a)
    L = next_pow2(2*len(w) - 1)
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)
    cc = np.fft.irfft(W * A.conj(), L)
    return cc[:n]
