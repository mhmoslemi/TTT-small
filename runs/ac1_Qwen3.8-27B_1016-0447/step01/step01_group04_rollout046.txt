def soft_weights(c, tau):
    cmax = c.max()
    if cmax <= 0:
        w = np.zeros(len(c))
        w[len(c) // 2] = 1.0
        return w
    z = np.clip((c - cmax) / tau, -60.0, 0.0)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros(len(c))
        w[c.argmax()] = 1.0
        return w
    return e / s


def make_inits(n, parent):
    idx = np.arange(n, dtype=np.float64)
    out = []
    out.append(np.ones(n))
    out.append(1.0 / (1.0 + idx / (n / 4.0)))
    out.append(np.power(1.0 + idx, -0.3))
    out.append(np.power(1.0 + idx, -0.5))
    out.append(np.power(1.0 + idx, -0.7))
    out.append(np.power(1.0 + idx, -1.0))
    rev = 1.0 / (1.0 + (n - 1 - idx) / (n / 4.0))
    out.append(rev)
    out.append(np.power(1.0 + (n - 1 - idx), -0.5))
    x = idx / max(1, n - 1)
    out.append(np.cos(np.pi * x / 2.0))
    out.append(np.cos(np.pi * x / 2.0) ** 2)
    out.append(1.0 + 0.3 * np.cos(2 * np.pi * x))
    out.append(1.0 + 0.5 * x)
    out.append(1.0 - 0.3 * x)
    if parent is not None and len(parent) == n:
        p = np.clip(parent, 0, None)
        if p.sum() > 0.01:
            out.append(p.copy())
    return out

    while time.monotonic() < deadline:

        if sc < best_sc - 1e-14:
            best_sc = sc
            best = a.copy()
            stagnant = 0

        if M <= 0:
            break

        if stagnant > 300:
            restarted = False
            for t in (0.4, 0.7):
                for ini in inits_norm:
                    b = norm_pos((1 - t) * best + t * ini)
                    cb = full_conv(b, flen)

                    if scb < best_sc + 0.02:

                        sc = scb

    final = np.clip(final, 0.0, 1000.0)

    return [float(x) for x in final]
