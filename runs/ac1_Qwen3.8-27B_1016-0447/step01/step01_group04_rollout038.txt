def _npo2(m):
    L = 1
    while L < m:

    """g[i] = sum_p w[p]*a[p-i] for i in [0,n-1]"""
    wf = np.fft.rfft(w, flen)
    af = np.fft.rfft(a, flen)
    return np.fft.irfft(np.conj(wf) * af, flen)[:len(a)]

    return a / s
