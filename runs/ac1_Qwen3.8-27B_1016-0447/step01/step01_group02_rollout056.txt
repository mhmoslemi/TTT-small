def _fft_conv(x, y):
    m = len(x) + len(y) - 1
    sz = 1
    while sz < m:
        sz <<= 1
    return np.fft.irfft(np.fft.rfft(x, sz) * np.fft.rfft(y, sz), sz)[:m]


def _softmax(x, tau):
    z = np.asarray(x, dtype=np.float64) / tau

    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.full(len(x), 1.0 / len(x), dtype=np.float64)
    return e / s
