import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse

def _fft_conv(x, y):
    m = len(x) + len(y) - 1
    sz = 1
    while sz < m:
        sz <<= 1
    fx = np.fft.rfft(x, sz)
    fy = np.fft.rfft(y, sz)
    return np.fft.irfft(fx * fy, sz)[:m]

def _score(a):
    n = len(a)
    ac = _fft_conv(a, a)
    s = a.sum()
    if s < 1e-30:
        return float('inf')
    return float(2.0 * n * np.max(ac) / (s * s))

def _build_conv_matrix(n, f):
    rows = np.add.outer(np.arange(n), np.arange(n)).ravel()
    cols = np.repeat(np.arange(n), n)
    vals = np.repeat(f, n)
    return sparse.csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def _lp_solve(A, f, M, n, time_limit=30):
    A.data[:] = np.repeat(f, n).astype(np.float64)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n,
                      method='highs', options={'time_limit': time_limit})
        if res.success and res.x is not None and res.x.sum() > 1e-15:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse


def _fft_conv(x, y):
    m = len(x) + len(y) - 1
    sz = 1
    while sz < m:
        sz <<= 1
    fx = np.fft.rfft(x, sz)
    fy = np.fft.rfft(y, sz)
    return np.fft.irfft(fx * fy, sz)[:m]


def _score(a):
    n = len(a)
    ac = _fft_conv(a, a)
    s = a.sum()
    if s < 1e-30:
        return float('inf')
    return float(2.0 * n * np.max(ac) / (s * s))


def _build_sparse_conv(n, f):
    rows = np.add.outer(np.arange(n), np.arange(n)).ravel()
    cols = np.repeat(np.arange(n), n)
    A = sparse.csc_matrix((np.repeat(f, n), (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_direction(A, f, M, n, tlim):
    A.data[:] = np.repeat(f, n).astype(np.float64)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n,
                  method='highs', options={'time_limit': tlim})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-14:
            return g
    return None


def _grad_step(a, tau, lr):
    n = len(a)
    ac = _fft_conv(a, a)
    z = ac / (tau * n)
    z -= z.max()
    e = np.exp(z)
    p = e / e.sum()
    p_rev = p[::-1]
    ca = _fft_conv(p_rev, a)
    grad = 2.0 * ca[n - 1:2 * n - 1][::-1]
    a = a * np.exp(-lr * grad / n)
    a = np.maximum(a, 1e-30)
    a /= a.sum()
    return a
