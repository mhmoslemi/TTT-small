    fy = np.fft.rfft(y, sz)
    return np.fft.irfft(fx * fy, sz)[:m]

The bottleneck is the score evaluation (which we do every 30-50 steps). At 1ms per eval, we can do 1000 evals per second, or about 1.5M evals total. That's more than enough.

import numpy as np
import time


def _fft_conv(x, y):
    m = len(x) + len(y) - 1
    sz = 1
    while sz < m:
        sz <<= 1
    fx = np.fft.rfft(x, sz)
    fy = np.fft.rfft(y, sz)
    return np.fft.irfft(fx * fy, sz)[:m]


def _score(a):
    n = a.size
    conv = _fft_conv(a, a)
    s = a.sum()
    return 2.0 * n * conv.max() / (s * s)


def _grad_smooth(a, tau):
    n = a.size
    s = _fft_conv(a, a)
    z = (s - s.max()) / tau
    p = np.exp(z)
    p /= p.sum()
    c = p[::-1]
    ca = _fft_conv(c, a)
    return 2.0 * ca[n - 1:2 * n - 1][::-1]


def _grad_exact(a):
    n = a.size
    s = _fft_conv(a, a)
    k = int(s.argmax())
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    if lo <= hi:
        idx = k - np.arange(lo, hi + 1)
        g[lo:hi + 1] = 2.0 * a[idx]
    return g


def _phase(a, deadline, rng, tau, lr, eval_int, stall_limit, reheat_amp):
    n = a.size
    best = a.copy()
    best_sc = _score(a)
    it = 0
    stalled = 0
    while time.monotonic() < deadline:
        g = _grad_smooth(a, tau)
        if not np.all(np.isfinite(g)):
            break
        a = a * np.exp(-lr * g)
        a = np.maximum(a, 0.0)
        ss = a.sum()
        if ss < 1e-12:
            a = np.ones(n) / n
        else:
            a /= ss
        it += 1
        if it % 25 == 0:
            tau *= 0.993
            if tau < 5e-7 / n:
                tau = 5e-7 / n
            lr *= 0.9985
        if it % eval_int == 0:
            sc = _score(a)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best = a.copy()
                stalled = 0
            else:
                stalled += 1
            if stalled >= stall_limit:
                a += reheat_amp * rng.standard_normal(n) * (best_sc / (2.0 * n))
                a = np.maximum(a, 0.0)
                a /= a.sum()
                tau = max(tau * 2.5, 1.0 / n)
                lr = max(lr * 1.3, 0.8)
                stalled = 0
            if time.monotonic() >= deadline:
                break
    sc = _score(a)
    if sc < best_sc:
        best = a.copy()
        best_sc = sc
    return best, best_sc


def _polish(a, deadline):
    n = a.size
    best = a.copy()
    best_sc = _score(a)
    lr = 0.05
    for _ in range(3000):
        if time.monotonic() >= deadline:
            break
        g = _grad_exact(a)
        if not np.any(g > 0):
            break
        for trial in range(6):
            a_new = np.maximum(a - lr * g, 0.0)
            ss = a_new.sum()
            if ss < 1e-12:
                lr *= 0.4
                break
            a_new /= ss
            sc = _score(a_new)
            if sc < best_sc - 1e-14:
                best = a_new.copy()
                best_sc = sc
                a = a_new
                lr = min(lr * 1.15, 0.8)
                break
            lr *= 0.55
        if lr < 1e-7:
            break
    return best, best_sc
