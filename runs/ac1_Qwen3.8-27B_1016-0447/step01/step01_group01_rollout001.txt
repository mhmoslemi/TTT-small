import numpy as np

def _fft_conv(a, N):
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)


def _score_ff(a, n, N):
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = _fft_conv(a, N)
    return float(2.0 * n * b.max() / (S * S))


def _grad_smooth(a, b, n, n_peaks, temp_scale):
    S = a.sum()
    mean_a = S / n
    idx = np.argpartition(b, -n_peaks)[-n_peaks:]
    pv = b[idx]
    pmax, pmin = pv.max(), pv.min()
    temp = (pmax - pmin) * temp_scale / max(1, n_peaks) + 1e-10
    w = np.exp(-(pv - pmax) / temp)
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    w = w / wsum
    grad = np.zeros(n)
    for wi, ki in zip(w, idx):
        k = int(ki)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            seg = a[k - hi: k - lo + 1][::-1]
            grad[lo:hi + 1] += wi * seg
    grad -= mean_a
    return grad
