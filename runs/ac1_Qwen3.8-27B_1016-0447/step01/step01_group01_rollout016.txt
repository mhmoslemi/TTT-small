def _fft_conv(a, N):
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)


def _score(a, n, N):
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = _fft_conv(a, N)
    return float(2.0 * n * b.max() / (S * S))


def _subgradient(a, b, n, K):
    peak_idx = np.argpartition(b, -K)[-K:]
    pv = b[peak_idx]
    pv_max = pv.max()
    if pv_max < 1e-30:
        return np.zeros(n)
    w = np.exp(4.0 * (pv - pv_max) / (pv_max + 1e-30))
    w /= w.sum()
    grad = np.zeros(n)
    for wi, k in zip(w, peak_idx):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        j = np.arange(lo, hi + 1)
        k_j = k - j
        valid = (k_j >= 0) & (k_j < n)
        g = np.zeros(n)
        g[j[valid]] = a[k_j[valid]] + a[j[valid]]
        diag = valid & (k_j == j)
        g[j[diag]] = 2.0 * a[j[diag]]
        grad += wi * g
    return grad
