def _fft_conv(x, y):
    m = len(x) + len(y) - 1
    sz = 1
    while sz < m:
        sz <<= 1
    return np.fft.irfft(np.fft.rfft(x, sz) * np.fft.rfft(y, sz), sz)[:m]


def _score(a):
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = _fft_conv(a, a)
    return 2.0 * a.size * float(c.max()) / (s * s)

    n = a.size
    s = _fft_conv(a, a)
    z = s / max(tau, 1e-15)
    z -= z.max()
    e = np.exp(z)
    p = e / e.sum()
    c = p[::-1]
    ca = _fft_conv(c, a)
    return 2.0 * ca[n - 1:2 * n - 1][::-1]

    cols = np.repeat(np.arange(n), n)

                  options={'time_limit': time_limit})

        return np.maximum(res.x, 0.0)
    return None
