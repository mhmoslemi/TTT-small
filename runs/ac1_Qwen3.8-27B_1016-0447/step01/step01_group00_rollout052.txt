from scipy import fft as sfft


def _np2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _fft_score(x, fs, n):
    Xf = sfft.rfft(x, fs)
    c = np.real(sfft.irfft(Xf * Xf, fs))[:2 * n - 1]
    sm = x.sum()
    if sm < 1e-10:
        return 1e10
    return 2.0 * n * c.max() / (sm * sm)

    if s < 0.1:
