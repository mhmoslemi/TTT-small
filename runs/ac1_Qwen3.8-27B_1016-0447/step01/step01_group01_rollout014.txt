import numpy as np
import time


def _fft_conv(a):
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:2 * n - 1]


def _normalize(a, n):
    s = a.sum()
    if s < 1e-30:
        return np.full(n, 1.0 / n)
    return np.maximum(a, 0.0) / s


def _score(a, n):
    S = a.sum()
    if S < 0.01:
        return float('inf')
    c = _fft_conv(a)
    return 2.0 * n * c.max() / (S * S)


def _subgrad(a, c, n, n_peaks):
    g = np.zeros(n)
    n_p = min(n_peaks, 2 * n - 1)
    peaks = np.argpartition(c, -n_p)[-n_p:]
    pv = c[peaks]
    pw = pv / pv.sum()
    for w, k in zip(pw, peaks):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * w * a[k - hi:k - lo + 1][::-1]
    return g


def _init_ramp(n, i):
    return (i + 1.0) / (n * (n + 1) / 2.0)


def _init_hill(n, i):
    v = np.exp(-1.5 * ((i - (n - 1) / 2.0) / (n * 0.25)) ** 2)
    return v / v.sum()


def _init_cos(n, i, f1, f2, alpha):
    v = 1.0 + alpha * np.cos(2 * np.pi * f1 * i / n) + 0.5 * alpha * np.cos(2 * np.pi * f2 * i / n)
    return np.maximum(v, 0.1) / np.maximum(v, 0.1).sum()

    f = np.fft.irfft(F * mask, n)
    v = np.abs(f) + 0.2
    return v / v.sum()
