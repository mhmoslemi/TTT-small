    m = len(x) + len(y) - 1

    return np.fft.irfft(fx * fy, sz)[:m]

def _peak_shave(a, rng):

    n = a.size
    conv = _fft_conv(a, a)

Actually, there's a simpler way to do the LP update: use a projected subgradient method on the dual, or use a multiplicative update (similar to the gradient descent but for the LP).

The LP: max 1^T g s.t. T_f * g <= M * 1, g >= 0.
where T_f is the Toeplitz matrix.

A simple approach: iterative projection.
1. Start with g = 1
2. Compute h = T_f * g (via FFT: h = f * g)
3. For each k where h_k > M, reduce g proportionally: g_i *= M / h_k for all i that contribute to h_k (i.e., 0 <= k-i <= n-1)
4. Repeat

This is like a "power iteration" for the LP. Each step is O(n log n) via FFT.

def _lp_update(f, M, iters=100):
    """LP-style update: find g >= 0 maximizing sum(g) s.t. f*g <= M."""
    n = f.size
    g = np.ones(n)
    conv_len = 2 * n - 1
    fft_sz = 1
    while fft_sz < conv_len:
        fft_sz <<= 1
    
    ff = np.fft.rfft(f, fft_sz)
    
    for it in range(iters):
        fg = np.fft.irfft(ff * np.fft.rfft(g, fft_sz), fft_sz)[:conv_len]
        violations = np.maximum(fg - M, 0)
        if violations.max() < 1e-10:
            break
        # For each violating k, scale down contributing g_i
        # Do it in a single pass
        max_viol = violations.max()
        if max_viol < 1e-15:

        # Scale: g *= (M / max(fg, M)) element-wise won't work directly

def _fft_size(m):
    s = 1
    while s < m:
