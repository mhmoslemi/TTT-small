    m = len(x) + len(y) - 1

    while sz < m:

    fx = np.fft.rfft(x, sz)
    fy = np.fft.rfft(y, sz)
    return np.fft.irfft(fx * fy, sz)[:m]

    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    conv = _fft_conv(a, a)
    max_b = conv.max()
    return float(2 * n * max_b / s**2)

def _subgrad(a, K=50):
    """Compute subgradient of max_k conv_k using top-K active constraints"""
    n = len(a)
    conv = _fft_conv(a, a)
    m = len(conv)  # 2n-1
    # Find top-K values
    if K >= m:
        idx = np.argsort(conv)[::-1]
    else:
        idx = np.argpartition(conv, m-K)[m-K:]
        idx = idx[np.argsort(conv[idx])[::-1]]
    vals = conv[idx]
    # Softmax over top-K for smoothness
    vals -= vals.max()
    w = np.exp(vals)
    w /= w.sum()
    # Build weight vector
    cw = np.zeros(m)
    cw[idx] = w
    # Gradient: for each j, g[j] = 2 * sum_k cw[k] * a[k-j] (valid indices)
    # This is 2 * (reverse(cw) * a)[shifted]
    c = cw[::-1]
    ca = _fft_conv(c, a)
    g = 2.0 * ca[n-1:2*n-1][::-1]
    # Also return the conv for reuse
    return g, conv

    m = state['m']
    v = state['v']

    v = b2 * v + (1 - b2) * grad**2
    m_hat = m / (1 - b1**t)
    v_hat = v / (1 - b2**t)
    state['m'] = m
    state['v'] = v
    return -lr * m_hat / (np.sqrt(v_hat) + 1e-8)
