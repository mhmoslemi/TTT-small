import numpy as np
import time


def _fft_conv(a, N):
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)


def _score(a, n, N):
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = _fft_conv(a, N)
    return 2.0 * n * b.max() / (S * S)


def _compute_grad(a, n, peak_idx, peak_w):
    g = np.zeros(n)
    for w, k in zip(peak_w, peak_idx):
        kk = int(k)
        lo = max(0, kk - n + 1)
        hi = min(n - 1, kk)
        if lo <= hi:
            seg = a[kk - hi: kk - lo + 1][::-1]
            g[lo: hi + 1] += 2.0 * w * seg
    return g


def _normalize(a, target_sum):
    s = a.sum()
    if s > 1e-30:
        a = a * (target_sum / s)
    return np.clip(a, 0.0, 1000.0)


def _param_init(n, rng, kind):
    i = np.arange(n)
    if kind == 0:
        return np.ones(n)
    elif kind == 1:
        return 1.0 + 0.25 * np.cos(2.0 * np.pi * i / n)
    elif kind == 2:
        c = n // 2
        return np.exp(-0.5 * ((i - c) / (n * 0.12)) ** 2)
    elif kind == 3:
        f1 = rng.integers(1, 10)
        f2 = rng.integers(1, 10)
        ph = rng.uniform(0, 2 * np.pi)
        return 1.0 + 0.2 * np.cos(2.0 * np.pi * f1 * i / n + ph) + 0.2 * np.sin(2.0 * np.pi * f2 * i / n)
    elif kind == 4:
        return 1.0 + 0.15 * rng.standard_normal(n)
    else:
        c1 = n // 3
        c2 = 2 * n // 3
        return np.exp(-0.5 * ((i - c1) / (n * 0.08)) ** 2) + np.exp(-0.5 * ((i - c2) / (n * 0.08)) ** 2) + 0.1

    target_sum = np.sqrt(2.0 * n)

    S = a.sum()
    sc = 2.0 * n * M / (S * S)
    best_local = a.copy()
    best_sc = sc
    momentum = np.zeros(n)
    mu = 0.6
    eta = eta0
    patience = 0
    max_pat = 250

    for it in range(200000):
        if time.monotonic() > deadline:

        # Find top peaks with softmax

        peak_vals = b[peak_idx]
        spread = peak_vals.max() - peak_vals.min()
        temp = max(1e-10, spread * 0.4) if spread > 1e-10 else 1.0
        peak_w = np.exp((peak_vals - peak_vals.max()) / temp)
        peak_w /= peak_w.sum()

        # Gradient

        g = g - g.mean()

        # Momentum
        momentum = mu * momentum + (1.0 - mu) * g
        direction = momentum - momentum.mean()
