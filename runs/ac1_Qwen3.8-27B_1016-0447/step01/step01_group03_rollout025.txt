        fc = _qs(c2)
        if np.isfinite(fc) and fc < fb - 1e-9:
            fb = fc
            a = c2.copy()
            eta = min(eta * 1.08, 0.12)
            if fc < bs:
                bs = fc
                best = a.copy()
        else:
            eta *= 0.85
    return best, bs

def propose_candidate(seed=42, budget_s=1100, **kw):
    t0 = time.monotonic()
    dl = t0 + max(5.0, float(budget_s) - 40.0)
    rng = np.random.default_rng(seed)
    n = 2535
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64)
        if raw.ndim == 1 and raw.size > 0:
            n = int(raw.size)
            p0 = np.clip(raw, 0, 1000)
            p0 = p0 / p0.sum() if p0.sum() > 0.01 else np.ones(n) / n
        else:
            p0 = np.ones(n) / n
    except Exception:
        p0 = np.ones(n) / n
    _init(n)
    best = p0.copy()
    bs = _es(best)
    if not np.isfinite(bs):
        best = np.ones(n) / n
        bs = 2.0
    top = [(bs, best.copy())]
    for c in _mkcands(n):
        if time.monotonic() > dl - 80:
            break
        c = np.clip(c, 0.01, 1000)
        cs = float(c.sum())
        if cs < 0.01:
            continue
        c = c / cs
        qs = _qs(c)
        if np.isfinite(qs):
            top.append((qs, c.copy()))
            top = sorted(top)[:5]
    starts = []
    for _, p in top:
        if len(starts) >= 3:
            break
        skip = False
        for q in starts:
            if np.linalg.norm(p - q) / max(1.0, np.linalg.norm(p)) < 0.02:
                skip = True
                break
        if not skip:
            starts.append(np.clip(p, 1e-6, None))
    for s in starts:
        if time.monotonic() > dl - 40:
            break
        z0 = np.log(s + 1e-30)
        z0 = z0 - z0.mean()
        best, bs = _lbt(z0, best, bs, dl)
    if time.monotonic() < dl - 20:
        best, bs = _polish(best, best, bs, dl, rng)
    fsc = _es(best)
    if np.isfinite(fsc) and fsc < bs:
        bs = fsc
    if not np.isfinite(bs) or not all(np.isfinite(best)):
        best = np.ones(n) / n
    return [float(x) for x in best]
