_G_beta = 0.0

def _fft_conv(a, b):
    L = len(a) + len(b) - 1
    N = 1 << (L - 1).bit_length()
    return np.real(np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N), N)[:L])

def _norm(a):
    a = np.clip(np.asarray(a, float), 0.0, 1000.0)
    s = a.sum()
    return a / s if s >= 0.01 else None

def _score(p):
    c = _fft_conv(p, p)
    return 2.0 * p.size * c.max()

def _eval(p):

    p /= p.sum()
    u = _G_n * _fft_conv(p, p)
    m = u.max()
    v = (np.log(np.sum(np.exp(_G_beta * (u - m)))) + _G_beta * m) / _G_beta
    return v + 1e-4 * z.mean() ** 2

def _lse_grad(z):
    p = np.exp(z - z.max())
    p /= p.sum()
    n = _G_n
    c = _fft_conv(p, p)
    u = n * c
    m = u.max()
    w = np.exp(_G_beta * (u - m))
    w /= w.sum()
    # G[j] = sum_i w[i+j] p[i]
    w_rev = w[::-1]
    L2 = len(w_rev) + n - 1
    N2 = 1 << (L2 - 1).bit_length()
    Fw = np.fft.rfft(w_rev, N2)
    Fp = np.fft.rfft(p, N2)
    conv = np.real(np.fft.irfft(Fw * Fp, N2))
    G = conv[n - 1:2 * n - 1][::-1]
    gp = 2.0 * n * p * (G - np.dot(p, G))
    return p * (gp - np.dot(p, gp)) + 2e-4 * z.mean()

def _lp_step(f):
    n = f.size
    M = float(_fft_conv(f, f).max())
    cols = np.tile(np.arange(n), n)
    rows = np.arange(n)[:, None] + np.arange(n)[None, :]
    rows = rows.ravel()

    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'time_limit': 20, 'presolve': True})
    if res.status == 0 and res.x is not None:
        g = np.clip(res.x, 0, None)
        s = g.sum()
        if s > 1e-10:
            return g / s
    return None
