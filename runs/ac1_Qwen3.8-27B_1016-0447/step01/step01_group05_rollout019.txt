import numpy as np
import time


def _fft_grad(a, w, n, L, idx):
    fa = np.fft.rfft(a, L)
    fw = np.fft.rfft(w, L)
    xc = np.fft.irfft(fa * fw.conj(), L)
    return 2.0 * xc[idx]
