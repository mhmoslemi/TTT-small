    c = _autocorr(a)
    return float(2.0 * a.size * np.max(c) / (s * s))


def _soft_beta_obj(z, beta, n, lam):
    m = z.size
    zc = z - np.max(z)
    e = np.exp(zc)
    p = e / np.sum(e)
    c = _autocorr(p)
    u = n * c
    um = np.max(u)
    ex = np.exp(beta * (u - um))
    s = np.sum(ex)
    if s <= 0 or not np.isfinite(s):
        return 1e10, None, None
    val = float(beta * um + np.log(s)) / beta + lam * float(np.mean(z))**2
    w = ex / s
    return val, p, w


def _soft_beta_grad(z, beta, n, lam):
    m = z.size
    zc = z - np.max(z)
    e = np.exp(zc)
    p = e / np.sum(e)
    c = _autocorr(p)
    u = n * c
    um = np.max(u)
    ex = np.exp(beta * (u - um))
    s = np.sum(ex)
    if s <= 0 or not np.isfinite(s):
        return np.zeros(m)
    w = ex / s
    N = 1 << max(0, (2 * m - 1 - 1).bit_length())
    fp = np.fft.rfft(p, N)
    fw = np.fft.rfft(w, N)
    corr = np.fft.irfft(fp * np.conj(fw), N)
    conv = np.fft.irfft(fp * fw, N)
    dminus = conv[:m]
    dplus = np.empty(m)
    dplus[0] = corr[0]
    if m > 1:
        dplus[1:] = corr[N - 1:N - m:-1]
    g = (n / beta) * (dplus + dminus)
    if not np.isfinite(g).all():
        return np.zeros(m)
    grad = p * (g - float(np.dot(p, g)))
    grad += 2.0 * lam * float(np.mean(z))
    return grad


def _make_candidates(n):
    cands = []
    j = np.arange(n, dtype=np.float64)
    cands.append(np.ones(n))
    for p in (-0.3, -0.2, -0.1, 0.0, 0.1, 0.2):
        cands.append((j + 1.0) ** p)
    for p in (-0.3, -0.1, 0.0, 0.1, 0.3):
        cands.append((np.abs(j - n / 2.0) + 1.0) ** p)
    for k in (1, 2, 3):
        cands.append(1.0 + 0.3 * np.cos(2.0 * np.pi * k * j / n))
    half = n // 2
    for r in (0.5, 0.7, 0.8, 0.9):
        a = np.ones(n)
        a[:half] = r
        cands.append(a)
    for k in (2, 3, 4, 6):
        seg = n // k
        a = np.ones(n)
        for i in range(k):
            a[i * seg:(i + 1) * seg] = 1.0 + 0.2 * (i % 2)
        cands.append(a)
    return cands


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(5.0, float(budget_s) - 30.0)

    parent = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64)
        if raw.ndim == 1 and raw.size > 0 and np.isfinite(raw).all():
            parent = np.clip(raw, 0.0, 1000.0)
    except Exception:
        pass
    if parent is None:
        parent = np.ones(2535)
    n = int(parent.size)

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(best_score):
        best = np.ones(n)
        best_score = float(evaluate_sequence(best.tolist()))
        if not np.isfinite(best_score):
            best_score = 2.0

    for cand in _make_candidates(n):
        if time.monotonic() > deadline - 60:
            break
        cand = np.clip(np.asarray(cand, dtype=np.float64), 0.0, 1000.0)
        if np.sum(cand) < 0.01:
            continue
        s = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(s) and s < best_score:
            best = cand.copy()
            best_score = s

    if _sopt is not None and time.monotonic() < deadline - 40:
        p0 = best / np.sum(best)
        z0 = np.log(p0 + 1e-300)
        z0 -= np.mean(z0)
        for beta in (16.0, 32.0, 64.0, 128.0):
            if time.monotonic() > deadline - 20:
                break
            lam = 1e-5
            try:
                res = _sopt.minimize(
                    lambda zz: _soft_beta_obj(zz, beta, n, lam)[0],
                    z0.copy(),
                    jac=lambda zz: _soft_beta_grad(zz, beta, n, lam),
                    method='L-BFGS-B',
                    options={'maxiter': 60, 'ftol': 1e-12, 'gtol': 1e-8},
                )
                if res is not None and res.x is not None and np.isfinite(res.x).all():
                    p_new = np.exp(res.x - np.max(res.x))
                    p_new /= np.sum(p_new)
                    p_new = np.clip(p_new, 0.0, 1000.0)
                    s = float(evaluate_sequence(p_new.tolist()))
                    if np.isfinite(s) and s < best_score:
                        best = p_new.copy()
                        best_score = s
            except Exception:
                pass
            z0 = res.x.copy() if res is not None and res.x is not None else z0
            z0 -= np.mean(z0)

    if time.monotonic() < deadline - 10:
        eta = 0.05
        p = best / np.sum(best)
        pb = _score_fast(p)
        while time.monotonic() < deadline - 5 and eta > 1e-5:
            noise = rng.normal(0, eta, n)
            k5 = np.ones(7) / 7.0
            noise = np.convolve(noise, k5, mode='same')
            cand = p * np.exp(noise)
            cand = np.clip(cand, 0.0, 1000.0)
            s = _score_fast(cand)
            if np.isfinite(s) and s < pb - 1e-12:
                es = float(evaluate_sequence(cand.tolist()))
                if np.isfinite(es) and es < best_score:
                    best = cand.copy()
                    best_score = es
                pb = s
                p = cand.copy()
                eta = min(0.15, eta * 1.03)
            else:
                eta *= 0.85

    best, bs = best, best_score
    if bs == float('inf') or not np.isfinite(bs):
        best = np.ones(n)
    bs = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(bs):
        best = np.ones(n)
    out = np.clip(best, 0.0, 1000.0)
    if np.sum(out) < 0.01:
        out = np.ones(n)
    return [float(x) for x in out]
