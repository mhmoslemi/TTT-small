    """Linear convolution via FFT. Both a and b are 1D arrays."""
    n = len(a) + len(b) - 1
    N = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n]

    """Self-convolution a*a via FFT."""
    n = a.size
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:L]

    """Clip to [0, 1000], ensure sum > 0, return normalized (sum=1) or None."""

    if s < 0.01:

    a = np.clip(a, 0.0, 1000.0)

    if s < 0.01:

    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:L]

def _conv_ab(a, b, trim):
    L = a.size + b.size - 1
    if L < trim:
        trim = L
    N = 1 << (L - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N), N)[:trim]

def _sm(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    return e / s if s > 0 else np.ones_like(z) / z.size

def _norm(a):
    a = np.clip(np.asarray(a, float), 0.0, 1000.0)
    s = a.sum()
    return a / s if s >= 0.01 else None

def _sc(a):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = _conv_self(a)
    return float(2.0 * a.size * c.max() / (s * s))

def _lse_val(z, beta):
    p = _sm(z)
    c = _conv_self(p)
    u = p.size * c
    um = u.max()
    v = um + np.log(np.exp(u - um).sum()) / beta
    return float(v) if np.isfinite(v) else 1e100

def _lse_grad(z, beta):
    n = z.size
    p = _sm(z)
    c = _conv_self(p)
    u = n * c
    um = u.max()
    w = np.exp(beta * (u - um))
    w /= w.sum()
    cp = _conv_ab(w, p, n)
    g = 2.0 * n * cp
    return p * (g - g.dot(p))

def _optimize(p0, tdl):
    global _DL
    if _min is None:
        return p0
    n = p0.size
    z0 = np.log(p0 + 1e-300)
    z0 -= z0.mean()
    gdl = tdl - 20.0
    _DL = gdl - 5.0
    z = z0.copy()
    for beta, mit in [(8, 60), (20, 60), (50, 80), (120, 80), (300, 100)]:
        if time.monotonic() > gdl:
            break
        try:
            r = _min(_lse_val, z.copy(), jac=_lse_grad, args=(beta,),
                     method='L-BFGS-B', callback=_cb,
                     options={'maxiter': mit, 'ftol': 1e-12, 'gtol': 1e-8, 'maxls': 30})
        except Exception:
            continue
        if r is not None and r.x is not None and np.isfinite(r.x).all():
            z2 = r.x - r.x.mean()
            _DL = gdl
            p2 = _sm(z2)
            if np.isfinite(p2).all():
                z = z2
    z = np.clip(z, -25, 25)
    p = _sm(z - z.mean())
    return p if p is not None else p0

def _peak_shave(p, tdl):

def _cands(n):
    L = []
    j = np.arange(n, float)
    L.append(np.ones(n, float))
    for e in (-2.0, -1.5, -1.0, -0.7, -0.5, -0.3, -0.1, 0.0, 0.1, 0.3, 0.5):
        a = (j + 1) ** e
        if np.isfinite(a).all() and (a > 0).any():
            L.append(a)
    for alpha in (0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0):
        a = np.exp(-alpha * (j / n - 0.5) ** 2)
        L.append(a)
    for d in (0.2, 0.4, 0.6, 0.8):
        a = np.exp(-((j / n - d) ** 2) / 0.01)
        L.append(a)
    for k in (3, 5, 8, 13):
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * k * j / n)
        L.append(a)
    return L
