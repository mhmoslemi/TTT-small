def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1

def normalize(x, ts):
    s = x.sum()
    if s > 0.01:
        x = x * (ts / s)
    else:
        x = np.ones_like(x) * ts / len(x)
    return np.clip(x, 0.0, 1000.0)


def conv_fft(x, sz):
    X = sfft.rfft(x, sz)
    return np.real(sfft.irfft(X * X, sz))[:2 * len(x) - 1]


def cross_corr_fft(p, x, sz):
    P = sfft.rfft(p, sz)
    X = sfft.rfft(x, sz)
    return np.real(sfft.irfft(P * np.conj(X), sz))[:len(x)]


def softmax_weights(conv, T):
    cmax = conv.max()
    p = np.exp((conv - cmax) / T)
    ps = p.sum()
    if ps < 1e-300:
        p = np.zeros_like(conv)
        p[(len(conv) - 1) // 2] = 1.0
    else:
        p /= ps
    return p, cmax


def topk_weights(conv, frac=0.995):
