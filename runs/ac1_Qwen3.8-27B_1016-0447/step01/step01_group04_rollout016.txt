    a = np.where(np.isfinite(a), a, 0.0)

    s = float(a.sum())
    if s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s

def _conv_sq(a, fl):
    n = a.size
    af = np.fft.rfft(a, n=fl)
    c = np.fft.irfft(af * af, n=fl)
    return np.maximum(c[:2 * n - 1], 0.0)

def _hankel(w, a, fl):
    n = a.size
    r = a[::-1]
    out = np.fft.irfft(np.fft.rfft(w, n=fl) * np.fft.rfft(r, n=fl), n=fl)
    return out[n - 1:2 * n - 1]

def _sw(c, tau):
    cm = float(c.max())
    if tau <= 1e-15 or cm <= 0:
        w = (c >= cm - 1e-10 * max(1.0, cm)).astype(np.float64)
        s = w.sum()
        if s > 0:
            w /= s
        else:
            w[0] = 1.0
        return w
    e = np.exp(np.clip((c - cm) / tau, -60.0, 0.0))
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        w = (c >= cm - 1e-6 * max(1.0, cm)).astype(np.float64)
        s = w.sum()
        w /= s if s > 0 else 1.0
        return w
    return e / s

def _sc(a, fl):
    c = _conv_sq(a, fl)
    return 2.0 * a.size * float(c.max())

def _inits(n, par):
    out = []
    if par is not None and par.size == n and par.sum() > 0:
        out.append(par.copy())
    idx = np.arange(n, dtype=np.float64)
    b = 1.0 + idx
    for p in (0.0, 0.3, 0.5, -0.2, -0.5, -0.7, -1.0, 1.0, 1.5):
        out.append(np.power(b, p))
    for bt in (0.002, 0.005, 0.01, 0.03):
        out.append(np.exp(-bt * idx))
    ct = (n - 1) / 2.0
    for sg in (0.08, 0.15, 0.25, 0.4):
        out.append(np.exp(-(idx - ct) ** 2 / (2.0 * (sg * n) ** 2)))
    for p in (0.5, -0.5):
        rb = float(n) + 1.0 - idx
        out.append(np.power(b, 0.3) + 0.1 * np.power(rb, p))
    return out

def _block(a, rng):
    b = a.copy()
    if a.size > 2:
        lo = max(1, a.size // 300)
        hi = max(lo + 1, a.size // 15)
        wi = int(rng.integers(lo, hi))
        st = int(rng.integers(0, a.size - wi))
        b[st:st + wi] *= float(np.exp(rng.normal(0.0, 0.15)))
    return _norm(b)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    try:
        par = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        par = np.where(np.isfinite(par), par, 0.0)
        par = np.clip(par, 0.0, None)
    except Exception:
        par = None
    n = int(par.size) if par is not None and par.size > 0 else 256
    fl1 = _npt(2 * n - 1)
    fl2 = _npt(3 * n - 2)
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    dl = t0 + max(60.0, float(budget_s) - 40.0)

    best = None
    best_sc = np.inf
    for arr in _inits(n, par):
        a = _norm(arr)
        sc = _sc(a, fl1)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = a.copy()
    if best is None:
        best = np.ones(n, dtype=np.float64) / n
        best_sc = 2.0
    inits = [best.copy()]

    a = best.copy()
    sc = best_sc
    tau_f = 0.06
    stag = 0
    it = 0

    while time.monotonic() < dl:
        it += 1
        c = _conv_sq(a, fl1)
        M = float(c.max())
        if not np.isfinite(M) or M <= 0:
            break
        sc = 2.0 * n * M
        if sc < best_sc - 1e-13:
            best_sc = sc
            best = a.copy()
            stag = 0
        else:
            stag += 1

        if stag > 120:
            tau_f = 0.06
            r = rng.random()
            if r < 0.4:
                alt = inits[int(rng.integers(0, min(len(inits), 8)))]
                a = _norm(0.5 * a + 0.5 * alt)
            elif r < 0.7:
                a = _block(best, rng)
            else:
                idx2 = np.arange(n, dtype=np.float64)
                a = _norm(best * (1.0 + 0.2 * np.cos(2.0 * np.pi * rng.integers(1, 5) * idx2 / n)))
            stag = 0

        tau = max(tau_f * M, 1e-12)
        tau_f = max(0.01, tau_f * (0.995 if stag < 40 else 1.01))

        w = _sw(c, tau)
        h = _hankel(w, a, fl2)
        grad = (2.0 / tau) * h
        grad = np.where(np.isfinite(grad), grad, 0.0)
        dot = float(np.dot(a, grad))
        rel = grad - dot
        rel = np.where(np.isfinite(rel), rel, 0.0)
        gm = float(np.abs(rel).max())

        cand = None
        cand_sc = sc
        if gm > 1e-14:
            for ml in (0.04, 0.1, 0.25, 0.5):
                et = ml / gm
                lg = np.clip(-et * rel, -12.0, 12.0)
                b = _norm(a * np.exp(lg))
                sb = _sc(b, fl1)
                if np.isfinite(sb) and sb < cand_sc:
                    cand = b
                    cand_sc = sb

        D = min(16, n)
        if D > 1:
            ct2 = a * np.abs(grad)
            dn = np.argpartition(ct2, n - D)[-D:]
            rv = np.argpartition(grad, D - 1)[:D]
            for mf in (0.004, 0.015, 0.05):
                b = a.copy()
                ds = float(b[dn].sum())
                mm = min(mf, 0.45 * ds)
                if mm > 1e-14:
                    b[dn] -= mm * b[dn] / ds
                    b[rv] += mm / D
                b = _norm(b)
                sb = _sc(b, fl1)
                if np.isfinite(sb) and sb < cand_sc:
                    cand = b
                    cand_sc = sb

        if cand is not None:
            a, sc = cand, cand_sc
            if sc < best_sc - 1e-13:
                best_sc, best = sc, a.copy()
                stag = 0
                if it < 80 and best_sc < best_sc + 1:
                    inits.append(a.copy())
                    if len(inits) > 12:

        elif stag % 30 == 0:
            b = _block(a, rng)
            sb = _sc(b, fl1)
            if np.isfinite(sb) and sb < sc:
                a, sc = b, sb

    out = np.clip(best, 0.0, 1000.0)
    s = float(out.sum())

    try:

            raise ValueError
    except Exception:
        out = _norm(best)
        out = np.clip(out, 0.0, 1000.0)
    return [float(x) for x in out]
