def _fft_conv(x, y):
    m = len(x) + len(y) - 1
    sz = 1
    while sz < m:
        sz <<= 1
    fx = np.fft.rfft(x, sz)
    fy = np.fft.rfft(y, sz)
    return np.fft.irfft(fx * fy, sz)[:m]


def _softmax_w(s, tau):
    z = s / tau
    z -= z.max()
    e = np.exp(z)
    t = e.sum()
    if t < 1e-300:
        return np.ones_like(s) / len(s)
    return e / t

    p = _softmax_w(s, tau)
    c = p[::-1]

    g = 2.0 * ca[n - 1:2 * n - 1][::-1]
    gm = g.max()
    if gm > 1e-30:

    return g, s.max()
