        sc, _, _ = score_of(normalize(arr), L)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = normalize(arr).copy()
    if best is None:
        best = np.ones(n, dtype=np.float64) / n
        best_sc = 2.0

    a = best.copy()
    sc = best_sc
    stagnant = 0
    phase = 0

    while time.monotonic() < deadline:
        phase += 1
        c = conv_full(a, L)
        M = float(c.max())
        s = float(a.sum())
        sc = 2.0 * n * M / (s * s)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        if M < 1e-30:
            break

        frac = stagnant / max(1, phase)
        tau = M * (0.03 + 0.07 * (1.0 - min(1.0, phase / 200.0)))
        tau = max(tau, 1e-8)

        z = (c - M) / tau
        z = np.clip(z, -40.0, 0.0)
        e = np.exp(z)
        ws = float(e.sum())
        if ws < 1e-300:
            w = np.zeros(c.size, dtype=np.float64)
            w[np.argmax(c)] = 1.0
        else:
            w = e / ws

        g = gradient_stats(w, a, Lg)
        gm = g - g.mean()
        gm = np.where(np.isfinite(gm), gm, 0.0)
        gmax = float(np.max(np.abs(gm)))

        improved = False
        if gmax > 1e-15:
            for eta in (0.02 / gmax, 0.06 / gmax, 0.12 / gmax, 0.25 / gmax, 0.5 / gmax):
                logit = -eta * gm
                logit = np.clip(logit, -12.0, 12.0)
                b = a * np.exp(logit)
                b = normalize(b)
                scb = score_of(b, L)[0]
                if np.isfinite(scb) and scb < sc - 1e-12:
                    a = b
                    sc = scb
                    if sc < best_sc - 1e-12:
                        best_sc = sc
                        best = a.copy()
                        stagnant = 0
                    improved = True
                    break

        if not improved:
            D = min(128, n)
            contrib = a * g
            if D < n:
                top_d = np.argpartition(contrib, n - D)[n - D:]
                low_g = np.argpartition(g, D - 1)[:D]
            else:
                top_d = np.arange(n)
                low_g = np.arange(n)
            for frac_m in (0.002, 0.005, 0.01, 0.02, 0.04):
                b = a.copy()
                ds = float(b[top_d].sum())
                mv = min(frac_m, 0.4 * ds)
                if mv < 1e-15:
                    continue
                wd = b[top_d]
                swd = float(wd.sum())
                if swd > 1e-30:
                    b[top_d] -= mv * wd / swd
                b[low_g] += mv / float(len(low_g))
                b = normalize(b)
                scb = score_of(b, L)[0]
                if np.isfinite(scb) and scb < sc - 1e-12:
                    a = b
                    sc = scb
                    if sc < best_sc - 1e-12:
                        best_sc = sc
                        best = a.copy()
                        stagnant = 0
                    improved = True
                    break

        if not improved and stagnant % 15 == 0:
            bs = max(1, n // 80)
            s0 = int(rng.integers(0, max(1, n - bs)))
            s1 = min(n, s0 + bs)
            b = a.copy()
            b[s0:s1] *= np.exp(rng.normal(0.0, 0.08))
            b = normalize(b)
            scb = score_of(b, L)[0]
            if np.isfinite(scb) and scb < sc:
                a = b
                sc = scb

        if stagnant > 200:
            ri = int(rng.integers(len(inits)))
            t = 0.3 + 0.3 * (stagnant % 3) / 3.0
            a = normalize((1.0 - t) * best + t * inits[ri])
            sc = score_of(a, L)[0]
            stagnant = 0

        if stagnant > 600:
            a = normalize(best * (1.0 + 0.05 * rng.standard_normal(n)))
            sc = score_of(a, L)[0]
            stagnant = 0

    out = np.clip(best, 0.0, 1000.0)
    if out.sum() < 0.01:
        out = np.ones(n, dtype=np.float64)
    sl = [float(x) for x in out]
    try:
        es = evaluate_sequence(sl)
        if np.isfinite(es) and es <= 1.5 * best_sc + 0.1:
            return sl
    except Exception:
        pass
    return sl
