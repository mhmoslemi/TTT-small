            p = np.zeros(2 * n - 1)
            p[n - 1] = 1.0
        else:
            p /= ps

        # Gradient via FFT: g[j] = 2*sum_l x[l]*p[l+j]
        p_rev = p[::-1]
        Pf = do_rfft(p_rev, F)
        cfull = do_irfft(Xf * Pf, F)
        g = 2.0 * cfull[n - 1:2 * n - 1][::-1]
        g -= g.mean()

        # Momentum step
        v = mu * v - lr * g
        x = x + v
        np.maximum(x, 0.0, out=x)
        np.minimum(x, 1000.0, out=x)
        x = norm_seq(x, target_sum)

        # Anneal
        T = max(min_T, T * 0.99985)

        # Adaptive lr
        if it > 100000:
            lr = max(0.08, lr * 0.99995)

        # Periodic evaluation
        if it % eval_iv == 0:
            sc = float(evaluate_sequence(x.tolist()))
            if np.isfinite(sc) and sc < best_sc - 1e-13:
                best_sc = sc
                best = x.copy()
                no_imp = 0
            else:
                no_imp += eval_iv

            # Restart
            if no_imp > 4 * eval_iv and restarts < max_rest and time.monotonic() < deadline - 60.0:
                x = best.copy()
                x += 0.04 * rng.standard_normal(n)
                x = np.maximum(x, 0.01)
                x = norm_seq(x, target_sum)
                v = np.zeros(n)
                T = max(0.12, 0.06 * best_sc)
                lr = 0.25
                no_imp = 0
                restarts += 1

        # Small noise
        if it % 1200 == 0 and it > 0:
            x += 2e-4 * rng.standard_normal(n)
            x = np.maximum(x, 0.01)
            x = norm_seq(x, target_sum)

        it += 1

    # Final polish: try small perturbations of best
    if time.monotonic() < deadline - 12.0:
        for _ in range(40):
            if time.monotonic() > deadline - 12.0:
                break
            xp = best + 0.005 * rng.standard_normal(n)
            xp = np.maximum(xp, 0.01)
            xp = norm_seq(xp, target_sum)
            xp = np.clip(xp, 0.0, 1000.0)
            sc = float(evaluate_sequence(xp.tolist()))
            if np.isfinite(sc) and sc < best_sc - 1e-13:
                best_sc = sc
                best = xp.copy()

    fc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(fc):
        # Fallback to parent
        pc = float(evaluate_sequence(parent.tolist()))
        if np.isfinite(pc):
            return [float(v) for v in parent]
        return [1.0] * n
    return [float(v) for v in best]
