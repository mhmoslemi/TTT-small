def _fft_conv(x, y):
    m = len(x) + len(y) - 1
    sz = 1
    while sz < m:
        sz <<= 1
    fx = np.fft.rfft(x, sz)
    fy = np.fft.rfft(y, sz)
    return np.fft.irfft(fx * fy, sz)[:m]

def _mpdescent(a, deadline, rng):
    n = a.size
    a = np.maximum(a, 0.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a[:] = 1.0
        s = n
    a /= s
    
    best = a.copy()
    best_score = _score(best)
    
    tau = 2.0 / n
    tau_min = 0.01 / n
    lr = 0.5
    stalled = 0
    
    while time.monotonic() < deadline:
        conv_aa = _fft_conv(a, a)
        z = conv_aa / tau
        z -= z.max()
        e = np.exp(z)
        es = e.sum()
        if es < 1e-300:
            p = np.full(2*n-1, 1.0/(2*n-1))
        else:
            p = e / es
        
        c = p[::-1]
        ca = _fft_conv(c, a)
        grad = 2.0 * ca[n-1:2*n-1][::-1]
        grad = np.maximum(grad, 0.0)
        
        # Adaptive learning rate
        a *= np.exp(-lr * grad)
        a = np.maximum(a, 1e-30)
        a /= a.sum()
        
        # Schedule
        tau *= 0.998
        if tau < tau_min:
            tau = tau_min
        lr = max(0.05, lr * 0.9997)
        
        if (stalled + 1) % 300 == 0:
            sc = _score(a)
            if sc < best_score - 1e-14:
                best_score = sc
                best = a.copy()
                stalled = 0
            else:
                stalled += 1
            if stalled >= 8:
                # Restart with perturbation
                a += 0.005 * rng.standard_normal(n)
                a = np.maximum(a, 0.0)
                a /= a.sum()
                tau = 2.0 / n
                lr = 0.5
                stalled = 0
    
    sc = _score(a)
    if sc < best_score:
        best = a.copy()
        best_score = sc
    return best, best_score

def _coord_descent(a, deadline, rng, max_rounds=50):
    """Coordinate descent refinement."""
    n = a.size
    a = a.copy()
    s = a.sum()
    a /= s
    best = a.copy()
    best_score = _score(best)
    
    idx = np.arange(n)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        rng.shuffle(idx)
        improved = False
        for i in idx:
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            # Try a few candidate values
            # The optimal a_i minimizes max_k (a*a)_k
            # For fixed others, (a*a)_k is linear in a_i (except k=2i)
            # Try: current, 2x, 0.5x, 0, and a bisection-based optimum
            candidates = [old_val, old_val * 0.5, old_val * 2.0, 0.0]
            # Also try the value that equalizes the top constraint
            best_cand = old_val
            best_local = best_score
            for cv in candidates:
                a[i] = max(0.0, cv)
                sc = _score(a)
                if sc < best_local:
                    best_local = sc
                    best_cand = cv
            a[i] = best_cand
            if best_local < best_score - 1e-14:
                improved = True
        
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best = a.copy()
        if not improved:
            break
    
    return best, best_score

Actually, for coordinate descent, I can be smarter. Instead of calling evaluate_sequence (which does a full conv), I can compute just the max of the convolution efficiently. When I change a[i] from old to new, the change in the convolution is:
- Δconv[k] = (new - old) * (a_{k-i} + a_{k'-i}) where k' is the "mirror" index... 

Actually, the change in conv(a,a) when a_i changes by δ:
conv(a', a')[k] - conv(a, a')[k] = δ * a_{k-i} + a_{k-i} * δ = 2δ * a_{k-i} (for the cross terms)
Plus if k = 2i: additional δ^2 term.

Wait, conv(a',a')[k] = sum_j a'_j a'_{k-j}
conv(a,a)[k] = sum_j a_j a_{k-j}

Difference = sum_j (a'_j a'_{k-j} - a_j a_{k-j})
= sum_j [(a_j + δ_j)(a_{k-j} + δ_{k-j}) - a_j a_{k-j}]
where δ_j = δ if j=i else 0

= sum_j [a_j δ_{k-j} + δ_j a_{k-j} + δ_j δ_{k-j}]
= a_i δ_{k-i} + δ_i a_{k-i} + δ_i δ_{k-i}
= 2δ * a_{k-i} + δ^2 (if k-i = i, i.e., k = 2i)
= 2δ * a_{k-i} (if k != 2i and 0 <= k-i < n)
= 0 (if k-i not in [0, n-1])

So the change is very sparse! Only 2n-1 - (some) terms change, and only if k-i is in [0, n-1].

This means I can compute the new max much more efficiently than a full FFT. But it's still O(n) per coordinate change (to check all k). So coordinate descent is O(n^2) per full sweep, which for n=2535 is about 6.4 million operations per sweep. That's fast in numpy.

    n = a.size

            break

        for i in idx:
            if time.monotonic() > deadline - 5:
                break
            old = a[i]
            if old < 1e-15:
                continue
            # Try a range of multipliers
            # The optimal scales old by some factor in [0, inf)
            # Try: 0.25, 0.5, 0.75, 1.25, 1.5, 2.0, and the exact optimum
            # For efficiency, just try a few
            best_val = old
            best_cv = _score(a)  # current score (cache it)
            
            # Actually, compute the new convolution incrementally
            # Δconv[k] = 2*(new-old)*a[k-i] + (new-old)^2 * [k==2i]
            # Only for k where 0 <= k-i < n
            
            for factor in (0.3, 0.6, 1.0, 1.5, 2.0):
                new = old * factor
                delta = new - old
                # Quick max estimate
                # conv_new[k] = conv_curr[k] + 2*delta*a[k-i] + delta^2*[k==2i]
                # We need max of conv_new
                # This is O(n) to compute exactly
                # But we can approximate: the max will be near the current max
                # For now, just compute fully
                if abs(delta) < 1e-15:
                    continue
                # Update convolution
                lo = max(0, i)  # k-i >= 0 => k >= i
                hi = min(2*n-2, i+n-1)  # k-i <= n-1 => k <= i+n-1
                k_range = np.arange(lo, hi+1)
                delta_conv = np.zeros(2*n-1)
                delta_conv[lo:hi+1] = 2*delta*a[k_range-i]
                if 2*i < 2*n-1:
                    delta_conv[2*i] += delta*delta
                conv_new = conv_curr + delta_conv
                new_max = conv_new.max()
                # Score = 2*n*new_max/(sum_a)^2, sum_a unchanged (we normalize at end)
                # Actually sum changes! new_sum = s - old + new
                new_sum = s - old + new
                sc = 2*n*new_max/(new_sum**2)
                if sc < best_cv:
                    best_cv = sc
                    best_val = new
            
            if best_val != old:
                delta = best_val - old
                lo = max(0, i)
                hi = min(2*n-2, i+n-1)
                conv_curr[lo:hi+1] += 2*delta*a[lo-i:hi+1-i]
                # Wait, a[lo-i:hi+1-i]... a is indexed 0..n-1
                # k ranges lo..hi, k-i ranges 0..n-1
                conv_curr[lo:hi+1] += 2*delta*a[k_range-i]
                if 2*i < 2*n-1:
                    conv_curr[2*i] += delta*delta
                s += delta
                a[i] = best_val
                any_improved = True
        
        # Renormalize
        a /= a.sum()
        conv_curr = _fft_conv(a, a)  # recomputing to avoid drift
        s = a.sum()
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best = a.copy()
        if not any_improved:
            break
    
    return best, best_score

    m = len(x) + len(y) - 1
    sz = 1
    while sz < m:
        sz <<= 1
    return np.fft.irfft(np.fft.rfft(x, sz) * np.fft.rfft(y, sz), sz)[:m]

import numpy as np
import time

def _fc(x, y):
    m = len(x) + len(y) - 1
    sz = 1
    while sz < m:
        sz <<= 1
    return np.fft.irfft(np.fft.rfft(x, sz) * np.fft.rfft(y, sz), sz)[:m]

def _sc(a):
    n = a.size
    c = _fc(a, a)
    s = a.sum()
    return float(2.0 * n * c.max() / (s * s))

def _mpd(a, dl, rng):
    n = a.size
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    if a.sum() < 1e-10:
        a[:] = 1.0
    a /= a.sum()
    best = a.copy()
    bs = _sc(best)
    tau = 3.0 / n
    tmin = 0.001 / n
    lr = 1.0
    st = 0
    stp = 0
    while time.monotonic() < dl:
        cv = _fc(a, a)
        z = cv / tau
        z -= z.max()
        e = np.exp(z)
        p = e / e.sum()

        gm = g.mean()

        a = np.maximum(a, 1e-20)
        a /= a.sum()
