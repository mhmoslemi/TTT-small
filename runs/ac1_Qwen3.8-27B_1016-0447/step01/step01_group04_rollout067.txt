import numpy as np
import time

def _npt(x):
    L = 1
    while L < x:
        L <<= 1
    return L

def _norm(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s

def _conv(a, N):
    n = len(a)
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]

def _grad(a, w, N2):
    n = len(a)
    X = np.fft.rfft(w, N2)
    Y = np.fft.rfft(a, N2)
    corr = np.fft.irfft(X * np.conj(Y), N2)
    return 2.0 * corr[:n]
