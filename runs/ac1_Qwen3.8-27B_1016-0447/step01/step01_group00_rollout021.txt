    iters_since = 0
    stall_limit = 30000

        X = sfft.rfft(x, S)
        conv = np.real(sfft.irfft(X * X, S))[:2*n-1]
        conv = np.maximum(conv, 0.0)
        
        # Softmax weights
        cmax = conv[0]  # Actually should be max over all
        cmax = conv.max()
        exp_vals = np.exp((conv - cmax) / T)
        psum = exp_vals.sum()
        if psum < 1e-300:
            p = np.zeros(2*n-1)
            idx = int(np.argmax(conv))
            p[idx] = 1.0
        else:
            p = exp_vals / psum
        
        # Gradient: 2 * cross_corr(p, x)
        P = sfft.rfft(p, S)
        grad = 2.0 * np.real(sfft.irfft(np.conj(P) * X, S))[:n]
        
        # Momentum
        momentum = beta * momentum + (1 - beta) * grad
        gnorm = np.linalg.norm(momentum)
        if gnorm > 1e-15:
            momentum = momentum / gnorm
        else:
            momentum = np.zeros(n)
        
        # Step size: scale gradient to a fraction of x
        step = lr * np.linalg.norm(x) / np.sqrt(n)
        x_new = x - step * momentum
        x_new = np.maximum(x_new, 0.0)
        
        # Rescale
        s = x_new.sum()
        if s < 1e-6:
            x_new = np.ones(n)
            s = n
        x_new = x_new / s * target_sum
        x = x_new
        
        # Anneal
        T = max(T_min, T * T_decay)
        
        # Evaluate periodically
        if it % eval_every == 0:
            sc = evaluate_sequence(x.tolist())
            if np.isfinite(sc) and sc < best_score - 1e-10:
                best_score = sc
                best = x.copy()
                iters_since = 0
            else:

            if iters_since > stall_limit:
                # Perturb and restart
                alpha = 0.01 + 0.05 * (iters_since / stall_limit)
                x = best + alpha * target_sum * rng.standard_normal(n)

                s = x.sum()

                    s = best.sum()
                x = x / s * target_sum
                momentum = np.zeros(n)
                T = max(0.05, best_score * 0.1)
                iters_since = 0
        
        it += 1
    
    # Final validation
    fc = evaluate_sequence(best.tolist())
    if not np.isfinite(fc) or fc > best_score:
        fc = best_score
    return [float(v) for v in best]
