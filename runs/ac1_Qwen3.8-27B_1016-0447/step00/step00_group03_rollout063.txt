def _fft_conv(x, y):
    m = len(x) + len(y) - 1
    sz = 1
    while sz < m:
        sz <<= 1
    fx = np.fft.rfft(x, sz)
    fy = np.fft.rfft(y, sz)
    return np.fft.irfft(fx * fy, sz)[:m]


def _softmax(x, tau):
    z = np.asarray(x, dtype=np.float64) / tau
    z -= z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.full(len(x), 1.0 / len(x), dtype=np.float64)
    return e / s
