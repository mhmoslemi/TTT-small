def compute_grad(f, k_min, k_max):
    n = len(f)
    P = np.empty(n + 1, dtype=np.float64)
    P[0] = 0.0
    np.cumsum(f, out=P[1:])
    js = np.arange(n, dtype=np.int64)
    lo = np.maximum(0, k_min - js)
    hi = np.minimum(n - 1, k_max - js)
    valid = hi >= lo
    g = np.zeros(n, dtype=np.float64)
    if np.any(valid):
        g[valid] = 2.0 * (P[hi[valid] + 1] - P[lo[valid]])
    return g
