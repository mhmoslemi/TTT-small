def block_perturb(base, rng, n, rng_strength=0.05):
    """One O(n) nonnegative perturbation of base (block multiplicative jitter)."""
    cand = base.copy()
    width = int(rng.integers(max(1, n // 300), max(2, n // 8)))

    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, rng_strength)))
    return np.clip(cand, 0.0, 1000.0)
