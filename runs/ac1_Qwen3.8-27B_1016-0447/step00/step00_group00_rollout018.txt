def _subgradient(a, n):
    """Compute score and subgradient of 2n*max(conv(a,a))/S^2 w.r.t. a."""
    c = np.convolve(a, a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf, None, c
    k_star = int(np.argmax(c))
    max_b = c[k_star]
    score = 2.0 * n * max_b / (S * S)
    # d c[k_star]/d a[i] = a[k_star - i] if 0 <= k_star - i < n
    i_lo = max(0, k_star - n + 1)
    i_hi = min(k_star, n - 1)
    grad_c = np.zeros(n)
    if i_hi >= i_lo:
        grad_c[i_lo:i_hi + 1] = a[k_star - np.arange(i_lo, i_hi + 1)]
    grad = (2.0 * n / (S ** 3)) * (S * grad_c - 2.0 * a * max_b)
    # Add small flatness-encouraging component: push toward uniform
    grad -= 0.01 * (a - a.mean())
    return score, grad, c


def _soft_subgradient(a, n, topk=3):
    """Subgradient using average of top-k conv values for smoother objective."""
    c = np.convolve(a, a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf, None, c
    idx = np.argsort(c)[::-1][:topk]
    weights = np.zeros(2 * n - 1)
    for j in range(topk):
        weights[idx[j]] = 1.0 / topk
    c_soft = np.dot(weights, c)
    score = 2.0 * n * c.max() / (S * S)  # actual score still uses true max
    # Gradient of c_soft w.r.t. a
    grad_c = np.zeros(n)
    for j in range(topk):
        k_j = idx[j]
        i_lo = max(0, k_j - n + 1)
        i_hi = min(k_j, n - 1)
        if i_hi >= i_lo:
            grad_c[i_lo:i_hi + 1] += (1.0 / topk) * a[k_j - np.arange(i_lo, i_hi + 1)]
    grad = (2.0 * n / (S ** 3)) * (S * grad_c - 2.0 * a * c_soft)
    grad -= 0.005 * (a - a.mean())
    return score, grad, c

        i_hi = min(k, n - 1)
        if i_hi >= i_lo:
            contrib[i_lo:i_hi + 1] += c[k] / (i_hi - i_lo + 1) * (a[k - np.arange(i_lo, i_hi + 1)] / (S / n + 1e-12))
    # Scale update
    ratio = contrib / (S / n + 1e-12)
    factor = 1.0 + scale * (1.0 - ratio)
    return np.clip(a * np.clip(factor, 0.0, 2.0), 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 10.0)
    n = len(height_sequence_1)
    a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0).copy()

    # Ensure valid
    if np.sum(a) < 0.01:
        a = np.ones(n)

    best = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))
    if not np.isfinite(best_score):
        a = np.ones(n)
        best = a.copy()
        best_score = float(evaluate_sequence(a.tolist()))

    # Phase 1: Gradient descent with adaptive LR
    lr = 1.0 / (2.0 * n)
    no_improve = 0
    phase1_end = deadline - (budget_s * 0.55)

    while time.monotonic() < phase1_end:
        score, grad, c = _soft_subgradient(a, n, topk=4)
        if not np.isfinite(score):
            break
        if score < best_score - 1e-12:
            best = a.copy()
            best_score = score
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive learning rate
        if score < best_score:
            lr = min(lr * 1.03, 5.0 / n)
        elif no_improve > 20:
            lr *= 0.92
            if no_improve > 100:
                # Add perturbation to escape
                a = np.clip(a + rng.normal(0, 0.02 * a.mean(), n), 0, 1000)
                no_improve = 0
                continue

        a_new = np.clip(a - lr * grad, 0.0, 1000.0)
        S_new = np.sum(a_new)
        if S_new < 0.01:
            continue
        a = a_new

    # Phase 2: Multiplicative / ratio-based updates
    phase2_end = deadline - (budget_s * 0.20)
    while time.monotonic() < phase2_end:
        score_cur, grad, c = _subgradient(a, n)
        if not np.isfinite(score_cur):
            break
        if score_cur < best_score - 1e-12:
            best = a.copy()
            best_score = score_cur

        # Multiplicative update
        a_alt = _multiplicative_update(a, c, n, scale=0.05)
        s_alt = float(evaluate_sequence(a_alt.tolist()))
        if np.isfinite(s_alt) and s_alt < score_cur:
            a = a_alt
        else:
            # Small gradient step
            a = np.clip(a - lr * grad / 2.0, 0.0, 1000.0)

        # Occasional random kick
        if rng.random() < 0.02:
            a = np.clip(a * np.exp(rng.normal(0, 0.05, n)), 0, 1000)

    # Phase 3: Fine-tuning with small perturbations and exact eval
    step_size = 0.02
    while time.monotonic() < deadline:
        # Try several perturbation sizes
        for s_scale in (step_size, step_size * 0.5, step_size * 0.1):
            a_pert = np.clip(a + rng.normal(0, s_scale * max(a.mean(), 0.1), n), 0, 1000)
            sc = float(evaluate_sequence(a_pert.tolist()))
            if np.isfinite(sc) and sc < best_score - 1e-9:
                best = a_pert.copy()
                best_score = sc
                a = a_pert.copy()
            elif np.isfinite(sc) and sc < 2.0:
                # Still good, nudge toward it
                a = 0.9 * a + 0.1 * a_pert
                a = np.clip(a, 0, 1000)
        # Also try scaling the whole sequence
        for mult in (0.95, 1.05, 0.9, 1.1):
            a_sc = np.clip(a * mult, 0, 1000)
            sc2 = float(evaluate_sequence(a_sc.tolist()))
            if np.isfinite(sc2) and sc2 < best_score - 1e-9:
                best = a_sc.copy()
                best_score = sc2
                a = a_sc.copy()

    # Final validation
    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score) or final_score > best_score + 1e-6:
        # Fallback: uniform
        best = np.ones(n, dtype=np.float64)
        final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):
        best = np.ones(n, dtype=np.float64) * 0.1

    return [float(x) for x in best]
