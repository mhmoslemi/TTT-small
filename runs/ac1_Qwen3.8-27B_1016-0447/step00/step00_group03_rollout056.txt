import numpy as np

def _autocorr(x):
    return np.convolve(x, x, mode='full')


def _peak_grad(x, topk=1):
    """Subgradient of max of linear autocorrelation. O(n)."""
    b = _autocorr(x)
    n = x.size
    # indices of top peaks
    kk = np.argpartition(b, -topk)[-topk:]
    g = np.zeros(n)
    for k in kk:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi < lo:
            continue
        idx = np.arange(lo, hi + 1)
        g[idx] += 2.0 * x[k - idx]
    return b, g


def _normalize(a):
    s = a.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones(a.size) / a.size
    return a / s
