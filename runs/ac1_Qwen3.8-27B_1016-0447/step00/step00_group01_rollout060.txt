import time

def _auto_conv(a, n):
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

def _flip_shift(a, k, n):
    out = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        out[lo:hi+1] = a[k - np.arange(lo, hi + 1)]
    return out

def _project(a):
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s > 1e-30:
        a /= s
    else:
        a = np.ones_like(a)
        a /= a.size
    return a

def _score_fft(a, n):
    c = _auto_conv(a, n)
    return 2.0 * n * max(0.0, float(c.max()))

def _grad_step(a, n, lr):

    M = float(conv.max())

    g *= 2.0 / len(active)
    g -= g.mean()
    return a - lr * g

        s_est = _score_fft(a_new, n)

            s_exact = float(evaluate_sequence(a_new.tolist()))
            if np.isfinite(s_exact) and s_exact < best_score:
                best = a_new.copy()
                best_score = s_exact
                a = a_new.copy()
                stall = 0
                lr = min(lr * 1.03, 50.0 / n)
            else:
                a = a_new.copy()
        else:
            if it % 200 == 0:
                s_exact = float(evaluate_sequence(a.tolist()))
                if np.isfinite(s_exact) and s_exact < best_score:
                    best = a.copy()
                    best_score = s_exact
            stall += 1
            if stall > 60:
                lr *= 0.5
                if lr < 0.01 / n:
                    # Escape: add fresh perturbation
                    freq = rng.uniform(1.0, 12.0)
                    phase = rng.uniform(0.0, 2.0 * np.pi)
                    perturb = np.sin(2.0 * np.pi * freq * xn + phase)
                    perturb -= perturb.mean()
                    a = _project(a + 0.05 * perturb)
                    lr = 1.0 / n
                    stall = 0

    # Final verification
    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):
        return [1.0] * n
    return [float(max(0.0, v)) for v in best]
