import time
import scipy.sparse as sp
from scipy.optimize import linprog

def build_conv_pattern(n):
    total = n * n
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    idx = 0
    for m in range(2 * n - 1):
        j_start = max(0, m - n + 1)
        j_end = min(n - 1, m)
        cnt = j_end - j_start + 1
        rows[idx:idx+cnt] = m
        cols[idx:idx+cnt] = np.arange(j_start, j_end + 1)
        idx += cnt

    M = np.max(np.convolve(a, a))

        rows[idx:idx + cnt] = m
