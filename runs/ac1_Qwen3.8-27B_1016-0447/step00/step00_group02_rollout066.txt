import numpy as np
import time


def _fft_conv(a, b, size):
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)


def _score_fft(a, size, n):
    b = _fft_conv(a, a, size)[:2 * n - 1]
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    return float(2.0 * n * np.max(b) / (s * s))


def _ramp_init(n):
    a = (np.arange(n, dtype=np.float64) + 1.0)
    a /= np.sum(a)
    return a


def _subgradient_step(a, b, size, n, lr, top_k=8):
    k_star = int(np.argmax(b))
    peak = b[k_star]
    # Find top_k lags closest to peak
    threshold = peak * 0.85
    active_mask = b >= threshold
    active_idx = np.where(active_mask)[0]
    if len(active_idx) > top_k:
        order = np.argsort(-b[active_idx])[:top_k]
        active_idx = active_idx[order]
    s = np.zeros(n, dtype=np.float64)
    for k in active_idx:
        k = int(k)
        j_min = max(0, k - n + 1)
        j_max = min(n, k + 1)
        if j_max > j_min:
            idx = np.arange(j_min, j_max)
            s[idx] += 2.0 * a[k - idx]
    a = a - lr * s
    np.clip(a, 0.0, None, out=a)
    sa = np.sum(a)
    if sa < 1e-15:
        return a, False
    a /= sa
    return a, True


def _block_perturb(a, rng, n):
    c = a.copy()
    width = max(1, n // 50)
    start = int(rng.integers(0, n - width))
    c[start:start + width] *= float(np.exp(rng.normal(0, 0.05)))
    np.clip(c, 0.0, None, out=c)
    s = np.sum(c)
    if s > 1e-15:
        c /= s
    return c
