import numpy as np
import time


def _fft_conv(a, fft_size):
    """Compute linear convolution a*a via FFT, returned as length 2n-1."""
    n = a.size
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:2 * n - 1]


def _grad_softmax(conv, a, n, beta, M):
    """Compute weighted gradient of soft-max of conv with respect to a."""
    cm = conv.max()
    if cm <= 0:
        return np.zeros(n)
    scaled = beta * conv / cm
    shifted = scaled - scaled.max()
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w /= ws
    # Focus on top M lags
    if M < 2 * n - 1:
        top_idx = np.argpartition(conv, -M)[-M:]
        mask = np.zeros(2 * n - 1, dtype=bool)
        mask[top_idx] = True
        w_m = w[mask]
        s = w_m.sum()
        if s < 1e-30:
            return np.zeros(n)
        w_m /= s
        ks = top_idx
    else:
        w_m = w
        ks = np.arange(2 * n - 1)
    grad = np.zeros(n)
    for idx in range(len(ks)):
        k = int(ks[idx])
        wk = float(w_m[idx])
        if wk < 1e-12:
            continue
        i_lo = max(0, k - (n - 1))
        i_hi = min(n - 1, k)
        # a[k-i] for i=i_lo..i_hi gives a indices k-i_lo down to k-i_hi
        lo = k - i_hi
        hi = k - i_lo
        grad[i_lo:i_hi + 1] += 2.0 * wk * a[lo:hi + 1][::-1]
    return grad
