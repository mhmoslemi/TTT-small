from numpy.fft import rfft, irfft
import time

def full_autocorr(a):
    """Full linear autocorrelation via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    A = rfft(a, size)
    return irfft(A * A, size)[:2 * n - 1]

def subgrad_step(a, lr):
    """One subgradient step. Returns new a and the peak lag."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return a, 0
    b = full_autocorr(a)
    b = np.maximum(b, 0)  # numerical safety
    M = b.max()
    k_star = int(np.argmax(b))
    
    # Subgradient of 2n*M/S^2 w.r.t. a
    j_min = max(0, k_star - n + 1)
    j_max = min(n - 1, k_star)
    jj = np.arange(j_min, j_max + 1)
    fi = k_star - jj  # k_star - j for each valid j
    grad = np.zeros(n)
    grad[jj] = (2.0 * n / S**2) * (a[fi] + a[jj]) - (4.0 * n * M / S**3) * a[jj]
    
    # Projected gradient step
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0)
    s_new = a_new.sum()
    if s_new > 1e-10:
        a_new = a_new * S / s_new  # restore original sum
    return a_new, k_star

def smooth_subgrad_step(a, lr, frac=0.1):
    """Subgradient step using near-peak lags."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return a, 0
    b = full_autocorr(a)
    b = np.maximum(b, 0)
    M = b.max()
    if M < 1e-30:
        return a, n // 2
    
    # Use lags where b[k] > (1-frac)*M
    threshold = (1 - frac) * M
    active = b > threshold
    
    # For each active lag k, the contribution to grad[j] is (a[k-j]+a[j]) for valid j
    # This is a sum over active k, which could be many
    # Let me accumulate
    grad = np.zeros(n)
    active_indices = np.where(active)[0]
    for k in active_indices:

        j_max = min(n - 1, k)
        jj = np.arange(j_min, j_max + 1)
        fi = k - jj
        grad[jj] += a[fi] + a[jj]
    
    # Average over active lags

    if n_active > 0:
        grad /= n_active
        grad = (2.0 * n / S**2) * grad - (4.0 * n * M / S**3) * a
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()

            a_new = a_new * S / s_new
        return a_new, active_indices[0]

import numpy as np
from numpy.fft import rfft, irfft
import time
from scipy.optimize import linprog
from scipy.sparse import coo_array

def full_ac(a):
    n = len(a)
    sz = 1 << (2*n-2).bit_length()
    F = rfft(a, sz)
    return irfft(F*F, sz)[:2*n-1]

def qsc(a):
    n = len(a)
    s = a.sum()
    if s < 1e-12: return np.inf
    return 2*n*max(0.0, full_ac(a).max()) / s**2

def sgs(a, lr):
    n = len(a); s = a.sum()
    if s < 1e-12: return a
    b = np.maximum(full_ac(a), 0); M = b.max()
    if M < 1e-12: return a
    k = int(np.argmax(b))
    lo = max(0, k-n+1); hi = min(n-1, k)
    jj = np.arange(lo, hi+1); fi = k-jj
    g = np.zeros(n); g[jj] = a[fi]+a[jj]
    a2 = np.maximum(a - lr*g, 0)
    s2 = a2.sum()
    return a2*s/s2 if s2 > 1e-12 else a2

def bld_A(a):
    n = len(a)
    jj = np.arange(n)
    rr = (jj[None,:]+np.arange(n)[:,None]).ravel()
    cc = np.repeat(jj, n)
    vv = np.tile(a, n)
    return coo_array((vv,(rr,cc)), shape=(2*n-1,n)).tocsc()

def lp_go(a, tl):
    n = len(a)
    s = a.sum()
    if s < 1e-12: return a
    M = max(0.0, full_ac(a).max())
    if M < 1e-12: return a
    A = bld_A(a)
    res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M),
                  bounds=[(0,None)]*n, method='highs',
                  options={'time_limit': tl})
    del A
    if res.success:
        g = np.maximum(res.x, 0)
        gs = g.sum()
        if gs > 1e-12:
            g *= s/gs
            return g
    return a
