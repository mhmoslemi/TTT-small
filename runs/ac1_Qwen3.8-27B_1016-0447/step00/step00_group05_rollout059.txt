import numpy as np
import time


def _autocorr_max(arr):
    x = np.clip(np.asarray(arr, dtype=np.float64), 0.0, 1000.0)
    s = float(x.sum())
    if s < 0.01:
        return float("inf")
    n = x.size
    r = 2 * n - 1
    nfft = 1
    while nfft < r:
        nfft *= 2
    X = np.fft.rfft(x, nfft)
    conv = np.fft.irfft(X * X, nfft)[:r]
    return float(2.0 * n * float(conv.max()) / (s * s))


def _norm_shape(x):
    m = float(x.max())
    if m <= 0.0 or not np.isfinite(m):
        return np.ones_like(x)
    y = x / m
    return np.clip(y, 0.0, 1.0)


def _block_mul(x, rng, n, wmin, wmax, sigma):
    y = x.copy()
    w = int(rng.integers(wmin, wmax + 1))
    if w > n - 1:
        w = n - 1
    s = int(rng.integers(0, n - w))
    y[s:s + w] *= np.exp(rng.normal(0.0, sigma, size=w))
    return y


def _point_mul(x, rng, n, sigma):
    y = x.copy()
    i = int(rng.integers(0, n))
    y[i] *= np.exp(rng.normal(0.0, sigma))
    return y


def _seg_scale(x, rng, n, sigma):
    y = x.copy()
    w = int(rng.integers(max(2, n // 100), max(3, n // 6)))
    if w > n - 1:
        w = n - 1
    s = int(rng.integers(0, n - w))
    y[s:s + w] *= float(np.exp(rng.normal(0.0, sigma * 0.6)))
    return y


def _log_smooth(x, rng, n, amp, arange_n):
    base = np.log(np.maximum(x, 1e-8))
    add = np.zeros(n, dtype=np.float64)
    for _ in range(3):
        f = int(rng.integers(1, max(2, n // 150) + 1))
        ph = float(rng.uniform(0.0, 6.2831853))
        a = amp * float(rng.uniform(-1.0, 1.0))
        add += a * np.sin(6.2831853 * f * arange_n / float(n) + ph)
    y = np.exp(base + add)
    return np.clip(y, 0.0, 1000.0)


def _blur(x, rng, n):
    k = float(rng.uniform(0.15, 0.55))
    c = np.array([1.0, k, 1.0])
    sm = np.convolve(x, c / c.sum(), mode="same")
    y = (1.0 - k) * x + k * sm
    return np.clip(y, 0.0, 1000.0)


def _make_initial(n, idx, parent):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1)
    tn = t / float(n - 1)
    if idx == 0:
        return 0.4 + 1.2 * tn
    if idx == 1:
        return 0.2 + 1.4 * tn
    if idx == 2:
        return 0.3 + 1.1 * tn * tn
    if idx == 3:
        return np.exp(0.9 * tn)
    if idx == 4:
        return 0.5 + 0.7 * np.sqrt(tn)
    if idx == 5:
        return 0.4 + 0.8 * np.sin(3.14159265 * tn * 0.85)
    if idx == 6:
        return 0.6 + 0.6 * np.cos(3.14159265 * (tn - 0.35)) ** 2
    if idx == 7:
        return np.clip(parent, 0.0, 1.0) * 0.5 + 0.5 * tn
    return np.ones(n)


def _local_search(x, rng, n, arange_n, deadline):
    best = _norm_shape(x.copy())
    best_sc = _autocorr_max(best)
    if not np.isfinite(best_sc):
        best_sc = 1e9

    sigma = 0.06
    rejects = 0
    max_sigma = 0.6

    while time.monotonic() < deadline:
        r = rng.random()
        if r < 0.45:
            y = _block_mul(best, rng, n, max(2, n // 300), n // 8, sigma)
        elif r < 0.70:
            y = _log_smooth(best, rng, n, sigma * 1.2, arange_n)
        elif r < 0.85:
            y = _seg_scale(best, rng, n, sigma * 1.5)
        elif r < 0.95:
            y = _point_mul(best, rng, n, sigma)
        else:
            y = _blur(best, rng, n)

        y = _norm_shape(y)
        sc = _autocorr_max(y)

        if np.isfinite(sc) and sc < best_sc:
            best = y
            best_sc = sc
            rejects = 0
            sigma = max(0.004, sigma * 0.9985)
        else:
            rejects += 1
            if rejects > 60:
                sigma = min(max_sigma, sigma * 1.15)
                if rng.random() < np.exp(-(sc - best_sc) / (sigma + 1e-6)):
                    best = y
                    best_sc = sc
                    rejects = 0
                elif rejects > 400:
                    sigma = 0.12
                    rejects = 0
    return best, best_sc
