            if ss > 1e-15:
                a /= ss
    return a


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 12.0)

    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = parent.size

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    inits = [
        parent.copy(),
        np.ones(n, dtype=np.float64),
        1.0 - np.abs(np.linspace(-1.0, 1.0, n)),
        np.abs(np.linspace(-1.0, 1.0, n)),
        0.5 + 0.5 * np.cos(2.0 * np.pi * np.arange(n, dtype=np.float64) / n),
        np.where(np.arange(n) % 2 == 0, 1.0, 0.4),
        1.0 + 0.3 * np.sin(2.0 * np.pi * np.arange(n, dtype=np.float64) / n),
    ]

    for a0 in inits:
        if time.monotonic() > deadline:
            break
        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 0.01:
            continue
        a = _optimize(a0, deadline, 200000, rng)
        sc = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = a.copy()

    # Perturbation + re-optimize loop
    while time.monotonic() < deadline:
        a = best.copy()
        m = max(1, n // 300)
        idx = rng.integers(0, n, size=m)
        a[idx] *= np.exp(rng.normal(0.0, 0.1, size=m))
        np.maximum(a, 0.0, out=a)
        a = _optimize(a, deadline, 5000, rng)
        sc = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = a.copy()

    # Scale output for safety (score is scale-invariant)
    best = np.clip(best, 0.0, 1000.0)
    sm = best.sum()
    if sm < 0.01:
        best = np.ones(n) * 1.0
    else:
        best = best * (float(n) / sm)
        best = np.clip(best, 0.0, 1000.0)

    return [float(v) for v in best]
