def compute_conv(a):
    """Compute full self-convolution."""
    return np.convolve(a, a, mode='full')

def compute_score(a):
    """Compute the score (same as evaluate_sequence but for numpy arrays)."""
    n = len(a)
    conv = np.convolve(a, a, mode='full')
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))

def gradient_at_k(a, k):
    """Gradient of conv(a,a)[k] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * a[k - idx]
    return grad
