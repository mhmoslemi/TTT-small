def targeted_reduce(a, eta=0.01):
    """Reduce the peak of the autocorrelation via targeted mass transfer."""
    b = np.convolve(a, a)
    k_star = int(np.argmax(b))
    n = len(a)
    # Contributing j: max(0, k_star-n+1) <= j <= min(n-1, k_star)
    j_lo = max(0, k_star - n + 1)
    j_hi = min(n - 1, k_star)
    # For each contributing j, the partner is k_star - j
    js = np.arange(j_lo, j_hi + 1)
    partners = k_star - js
    # Reduce a at positions where both a[j] and a[partner] are large
    weight = a[js] * a[partners]
    max_w = weight.max() if max_w > 0 else 1
    # Apply reduction
    a_new = a.copy()
    a_new[js] *= (1.0 - eta * weight / max_w)
    # Renormalize
    s = a_new.sum()
    if s > 0.01:
        a_new = a_new / s * a.sum()  # preserve sum
    return np.clip(a_new, 0, 1000)

import numpy as np
import time


def _norm_sum(a, target):
    s = np.sum(a)
    if s < 1e-15:
        return a
    return a * (target / s)


def _try_update(a, best, best_score):
    a = np.clip(a, 0.0, 1000.0)
    if np.sum(a) < 0.01:
        return best, best_score
    sc = float(evaluate_sequence(a.tolist()))
    if np.isfinite(sc) and sc < best_score:
        return a.copy(), sc
    return best, best_score


def _gradient_reduce(a, lr):
    b = np.convolve(a, a)
    k_star = int(np.argmax(b))
    n = len(a)
    j_lo = max(0, k_star - n + 1)
    j_hi = min(n - 1, k_star)
    js = np.arange(j_lo, j_hi + 1, dtype=np.int64)
    partners = k_star - js
    partners = partners[(partners >= 0) & (partners < n)]
    js = js[(js >= 0) & (js < n)]
    mask = (partners >= 0) & (partners < n)
    js = js[mask]
    partners = partners[mask]
    step = lr * a[partners]
    a_new = a.copy()
    a_new[js] -= step
    a_new = np.clip(a_new, 0.0, 1000.0)
    return a_new
