def eval_score(a):
    """Fast score computation (same as evaluator but without list conversion)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    if n == 0:
        return np.inf
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (sum_a * sum_a))

def smooth_perturb(base, rng, n_modes=8, sigma=0.03):
    """Add smooth multiplicative noise."""
    n = base.size
    kmax = min(n_modes, n // 2)
    noise = np.zeros(n)
    for k in range(1, kmax + 1):
        coef = rng.normal(0, 1.0 / np.sqrt(k))
        noise += coef * np.cos(2.0 * np.pi * k * np.arange(n) / n)
    noise = noise - noise.mean()
    std = noise.std()
    if std > 1e-10:
        noise = noise / std
    return np.clip(base * np.exp(sigma * noise), 0.0, 1000.0)

def peak_reduction(a, c, k_star, n, depth=20, factor=0.98):
    """Reduce the values at positions contributing most to c[k_star]."""
    result = a.copy()
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    # Contribution of position i to c[k_star] is a[i]*a[k_star-i]
    contrib = np.zeros(hi - lo + 1)
