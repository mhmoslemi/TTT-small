def fft_conv(a):
    """Compute np.convolve(a,a) using FFT. O(n log n)."""
    n = a.size
    m = 2 * n - 1
    fsize = 1
    while fsize < m:
        fsize <<= 1
    A = np.fft.rfft(a, fsize)
    out = np.fft.irfft(A * A, fsize)
    return out[:m]

    out = np.fft.irfft(A * A, fsize)
    return out[:m]


def active_grad(a, k):
    """Gradient of B(k)=sum_j a[j]*a[k-j] w.r.t. a. O(n)."""
    n = a.size
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        g[idx] = 2.0 * a[k - idx]
    return g

    if s < 1e-12:

    return a / s
