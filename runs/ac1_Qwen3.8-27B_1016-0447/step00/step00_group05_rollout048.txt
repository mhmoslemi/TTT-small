def compute_grad(a, peak_indices):
    """Compute subgradient of max(b) w.r.t. a, where b=convolve(a,a)."""

def _subgrad(a, peaks, n):
    """Subgradient of max(conv(a,a)) w.r.t. a."""
    g = np.zeros(n)
    for k in peaks:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        g[idx] += 2.0 * a[k - idx]
    return g

def _simplex_proj(v):

    n = v.size

                break
