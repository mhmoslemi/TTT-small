```python
def autocorr_gradient(a, m):
    """Gradient of b[m] = (a*a)[m] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    # a[j] appears in b[m] as a[j]*a[m-j]
    # derivative = a[m-j] for first factor
    # plus a[m-j] for second factor (when m-j is valid and != j)
    # Combined: 2*a[m-j] with a[k]=0 outside [0,n)
    # So g[j] = 2*a[m-j] if 0<=m-j<n, else 0
    # But when m-j == j, it's 2*a[j] (same thing)
    lo = max(0, m - n + 1)
    hi = min(n - 1, m)
    # For j in [lo, hi]: m-j is in [0, n-1]
    # g[j] = 2 * a[m-j]
    idx = np.arange(lo, hi + 1)
    g[idx] = 2.0 * a[m - idx]
    return g
```

```python
import time

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    rho = int(np.max(np.nonzero(cond)[0]))
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def auto_conv(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    b = np.fft.irfft(fa * fa, fsize)
    return b[:size]

def conv_gradient(a, k):
    """Gradient of (a*a)[k] w.r.t. a, where * is linear convolution."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        j = np.arange(lo, hi + 1)
        g[j] = 2.0 * a[k - j]
    return g

    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    
    a = np.asarray(height_sequence_1, dtype=np.float64)

    else:

    best_score = float(evaluate_sequence(best_a.tolist()))
```

So the smoothed approach is impractical for high precision. The subgradient method, while slow, is more practical because it directly targets the max.

```python
import numpy as np
import time


def project_simplex(v):
    """Project v onto the probability simplex in O(n log n)."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    mask = (u * np.arange(1, n + 1)) > (cssv - 1.0)
    if not mask.any():
        return np.full(n, 1.0 / n)
    r = int(np.flatnonzero(mask)[-1])
    return np.maximum(v - (cssv[r] - 1.0) / (r + 1), 0.0)


def fast_conv(a):
    """Linear convolution a*a via FFT, O(n log n)."""
    n = len(a)
    sz = 2 * n - 1
    fs = 1 << (sz - 1).bit_length()
    fa = np.fft.rfft(a, fs)
    return np.fft.irfft(fa * fa, fs)[:sz]


def conv_score(a):
    """Quick score using FFT convolution (avoids O(n^2) np.convolve)."""
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return 1e18
    return 2.0 * n * fast_conv(a).max() / (s * s)


def subgrad_of_max(a, b, k=3):
    """Weighted subgradient of max(b) w.r.t. a, combining top-k active indices."""
    n = len(a)
    kk = min(k, len(b))
    tk = np.argpartition(b, -kk)[-kk:]
    tv = b[tk]
    spread = max(1e-30, tv.max() - tv.min())
    w = np.exp(5.0 * (tv - tv.max()) / spread)
    w /= w.sum()
    g = np.zeros(n)
    for wi, ki in zip(w, tk):
        ki = int(ki)
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            g[idx] += wi * 2.0 * a[ki - idx]
    return g
```
