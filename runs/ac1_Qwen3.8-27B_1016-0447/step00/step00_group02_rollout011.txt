But implementing all this is complex. Let me go with a simpler but effective approach:

1. Start with uniform
2. Apply a "deflate center, inflate edges" transformation
3. Iterate: at each step, find the lag k where C[k] is largest, and adjust a to reduce C[k]

For step 3, to reduce C[k] = sum_i a[i]*a[k-i], we can:
- Decrease a[i] for some i where a[i]*a[k-i] is large
- But this might increase other C[k'] values

A simpler iterative scheme:
- Compute C = convolve(a, a)
- Find k* = argmax C
- For the pairs (i, k*-i) that contribute to C[k*], slightly decrease a[i] and a[k*-i]
- Renormalize to maintain sum(a) = sqrt(2n)
- Repeat

This is like a "coordinate descent" that focuses on the bottleneck.

import numpy as np
import time

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0, float(budget_s) - 5.0)

    if a.ndim != 1 or a.size == 0:
        a = np.ones(256)
    
    n = a.size
    # Normalize
    s = a.sum()
    if s < 0.01:
        a = np.ones(n)
    a = a * np.sqrt(2.0 * n) / s
    
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    
    while time.monotonic() < deadline:
        # Compute convolution
        C = np.convolve(a, a)
        max_b = C.max()
        # Find lags where C is near max
        threshold = max_b * 0.9
        hot_lags = np.where(C >= threshold)[0]
        
        # For each hot lag, find contributing indices
        # and slightly reduce them
        step = 0.001
        for k in hot_lags[:5]:  # limit to 5 hot lags per iteration
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                j = k - i
                # Reduce the pair (i, j) proportionally
                a[i] *= (1 - step)
                if j != i:
                    a[j] *= (1 - step)
        
        # Renormalize
        s = a.sum()
        if s < 0.01:
            a = best.copy()
            continue
        a = a * np.sqrt(2.0 * n) / s
        a = np.clip(a, 0.0, 1000.0)
        
        # Evaluate
        score = evaluate_sequence(a.tolist())
        if np.isfinite(score) and score < best_score:
            best = a.copy()
            best_score = score
    
    return [float(x) for x in best]
