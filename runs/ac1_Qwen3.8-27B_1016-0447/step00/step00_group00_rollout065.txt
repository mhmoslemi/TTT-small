def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    # Binary search for tau such that sum(max(0, v - tau)) = s
    lo = min(v) - s  # ensure sum >= s
    hi = max(v) + s  # ensure sum <= s
    for _ in range(100):
        mid = (lo + hi) / 2
        s_val = np.sum(np.maximum(0, v - mid))
        if s_val > s:
            lo = mid
        else:
            hi = mid
    tau = mid
    return np.maximum(0, v - tau)
