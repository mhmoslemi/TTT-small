def compute_score(f):
    """Compute the AC1 score. f is a numpy array, nonnegative, sum > 0."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max() / (f.sum() ** 2)

def subgradient_step(f, m, eta):
    """One subgradient step to reduce (f*f)[m]."""
    n = len(f)
    grad = np.zeros(n)
    for i in range(n):
        mi = m - i
        if 0 <= mi < n:
            if mi == i:
                grad[i] = 2.0 * f[i]
            else:
                grad[i] = f[mi] + f[i]
    # Normalize
    gn = np.linalg.norm(grad)
    if gn > 0:
        grad /= gn
    return f - eta * grad
