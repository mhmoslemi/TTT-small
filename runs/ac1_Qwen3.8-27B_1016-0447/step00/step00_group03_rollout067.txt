import scipy.sparse as sp

    idx = 0

        hi = min(n - 1, m)
        cnt = hi - lo + 1
        rows[idx:idx + cnt] = m
        cols[idx:idx + cnt] = np.arange(lo, hi + 1)
        idx += cnt
    return rows, cols


def score_of(a):
    return float(evaluate_sequence([float(x) for x in a]))
