```python
def simplex_project(z):
    """Project z onto the positive simplex {x: sum(x)=1, x>=0}."""
    n = len(z)
    sv, idx = np.sort(z, kind='mergesort')[::-1], np.argsort(z)[::-1]
    cssv = np.cumsum(sv)
    r = np.nonzero(sv * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(z - theta, 0)
```

```python
import numpy as np
import time

def simplex_project(z):
    """Project z onto {x: sum(x)=1, x>=0}."""
    n = z.size
    if n == 0:
        return z
    order = np.argsort(z)[::-1]
    z_sorted = z[order]
    cssv = np.cumsum(z_sorted)
    rho = np.nonzero(z_sorted * np.arange(1, n+1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    out = np.maximum(z - theta, 0.0)
    return out / out.sum()  # ensure exact normalization

def conv_fft(x, n):
    """Compute linear convolution x*x via FFT."""
    size = 2 * n - 1
    f = np.fft.rfft(x, n=2 * n)
    return np.fft.irfft(f * f, n=size)
```

```python
def _simplex_proj(z):
    n = z.size
    if n <= 1:
        return np.ones(n)
    order = np.argsort(z)[::-1]
    z_s = z[order]
    cum = np.cumsum(z_s)
    rho = np.nonzero(z_s * np.arange(1, n + 1) > (cum - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cum[r] - 1.0) / (r + 1.0)

    if s < 1e-30:
```

```python
import numpy as np
import time


def _simplex_proj(z):
    n = z.size
    if n <= 1:
        return np.ones(n)
    order = np.argsort(z)[::-1]
    z_s = z[order]
    cum = np.cumsum(z_s)
    rho = np.nonzero(z_s * np.arange(1, n + 1) > (cum - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cum[r] - 1.0) / (r + 1.0)
    out = np.maximum(z - theta, 0.0)
    s = out.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return out / s


def _conv2(x, n):
    L = 2 * n
    F = np.fft.rfft(x, L)
    return np.fft.irfft(F * F, L)[:2 * n - 1]


def _subgrad(y, n, top_k):
    b = _conv2(y, n)
    m = min(10, 2 * n - 1)
    top = np.argsort(b)[::-1][:m]
    s = np.zeros(n)
    for k in top:
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        js = np.arange(lo, hi)
        ks = k - js
        s[js] += y[ks]
    s /= m
    return s, b.max()
```
