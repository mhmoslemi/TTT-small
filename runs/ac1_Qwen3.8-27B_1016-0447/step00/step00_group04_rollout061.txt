def _next_pow2(m):
    s = 1
    while s < m:
        s <<= 1
    return s


def _fft_conv(a, fs):
    F = np.fft.rfft(a, fs)
    return np.fft.irfft(F * F, fs)


def _fft_xcorr(a, w, L):
    n = a.size
    b = a[::-1]
    W = np.fft.rfft(w, L)
    B = np.fft.rfft(b, L)
    conv = np.fft.irfft(W * B, L)
    return conv[n - 1:2 * n - 1]

    if s < 0.01:
