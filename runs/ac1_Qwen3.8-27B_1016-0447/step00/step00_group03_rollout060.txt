2. Compute the "influence" of each a[i] on b[k*]: this is a[k*-i] (the shifted version of a)

5. Iterate

def peak_shave_step(a, eta):
    n = len(a)
    b = np.convolve(a, a)
    k_star = np.argmax(b)
    # Gradient of b[k_star] w.r.t. a:
    # db[k_star]/da[m] = a[k_star - m] + (a[m - k_star] if m > k_star else 0)
    # More precisely:
    # b[k] = sum_i a[i]*a[k-i] (with a[j]=0 for j<0 or j>=n)
    # db[k]/da[m] = a[k-m] (if 0<=k-m<n) + a[m-k] (if 0<=m-k<n)
    
    grad = np.zeros(n)
    # First part: a[k-m] for m where 0 <= k-m < n, i.e., k-n+1 <= m <= k
    for m in range(max(0, k_star-n+1), min(n, k_star+1)):
        grad[m] += a[k_star - m]
    # Second part: a[m-k] for m where 0 <= m-k < n, i.e., k <= m <= k+n-1
    for m in range(max(0, k_star), min(n, k_star+n)):
        grad[m] += a[m - k_star]
    
    # Projected gradient step
    a_new = a - eta * grad
    # Renormalize to maintain sum
    a_new = a_new / a_new.sum() * a.sum()
    a_new = np.clip(a_new, 0, 1000)
    return a_new

    n = len(a)
