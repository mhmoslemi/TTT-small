import numpy as np
import scipy.fft
import scipy.optimize
from scipy import sparse

def convolve_full(a, b):
    """Full convolution using FFT, O(n log n)."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    fa = scipy.fft.fft(a, size)
    fb = scipy.fft.fft(b, size)
    return scipy.fft.ifft(fa * fb).real[:n]

def build_sparse_A(f, active_lags, n):
    """Build sparse constraint matrix for the LP.
    A[k, j] = f[k-j] if 0 <= k-j < n and j in [0, n-1].
    active_lags: list of lag indices k."""
    rows = []
    cols = []
    data = []
    for idx, k in enumerate(active_lags):
        # j ranges from max(0, k-n+1) to min(k, n-1)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start > j_end:
            continue
        js = range(j_start, j_end + 1)
        for j in js:
            val = f[k - j]
            if val != 0:
                rows.append(idx)
                cols.append(j)
                data.append(val)
    return sparse.csr_matrix((data, (rows, cols)), shape=(len(active_lags), n))

import numpy as np
from scipy import sparse, optimize, fft
import time


def fast_convolve(a, b):
    n, m = len(a), len(b)
    out_len = n + m - 1
    size = 1
    while size < out_len:
        size <<= 1
    fa = fft.rfft(a, size)
    fb = fft.rfft(b, size)
    return fft.irfft(fa * fb, size)[:out_len]


def build_A_sparse(f, lags, n):
    m = len(lags)
    ri, ci, da = [], [], []
    for i, k in enumerate(lags):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo > hi:

        js = np.arange(lo, hi + 1)
        vals = f[k - js]
        nz = vals != 0
        if nz.any():
            ri.extend([i] * int(nz.sum()))
            ci.extend(js[nz].tolist())
            da.extend(vals[nz].tolist())
    if not da:

    return sparse.csr_matrix((da, (ri, ci)), shape=(m, n))

    x = np.arange(n, dtype=np.float64)

    return np.exp(-((x - center) ** 2) / (2.0 * sigma * sigma)) + 1e-4


def make_init_cosine(n, rng):

    phase = float(rng.uniform(0.0, 6.28))
