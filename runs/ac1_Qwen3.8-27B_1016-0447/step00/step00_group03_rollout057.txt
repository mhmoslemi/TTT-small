def _fast_score(a):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * len(a) * c.max() / (s * s)


def _block_perturb(a, rng, sigma, n_blocks):
    n = len(a)
    b = a.copy()
    min_w = max(2, n // 300)
    max_w = max(min_w + 1, n // 6)
    for _ in range(n_blocks):
        w = int(rng.integers(min_w, max_w + 1))
        start = int(rng.integers(0, n - w + 1))
        end = min(n, start + w)
        mid = (start + end) / 2.0
        idx = np.arange(start, end, dtype=np.float64)
        half = w / 3.0
        taper = np.exp(-0.5 * ((idx - mid) / half) ** 2)
        f = float(np.exp(rng.normal(0.0, sigma)))
        b[start:end] *= (1.0 - taper + taper * f)
    return np.clip(b, 0.0, 1000.0)


def _sym_perturb(a, rng, sigma):
