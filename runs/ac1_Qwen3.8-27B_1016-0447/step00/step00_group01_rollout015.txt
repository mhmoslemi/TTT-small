import numpy as np
import time

def fft_convolve(x):
    """Compute linear convolution x * x using FFT. O(n log n)."""
    n = len(x)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    X = np.zeros(size, dtype=np.complex128)
    X[:n] = x
    Xf = np.fft.fft(X)
    cf = np.fft.ifft(Xf * Xf)
    return np.real(cf[:2*n-1])

def compute_gradient_at_lag(x, k):
    """Compute gradient of c[k] w.r.t. x. O(n)."""
    n = len(x)
    g = np.zeros(n, dtype=np.float64)
    # c[k] = sum_i x[i]*x[k-i], valid i: max(0, k-(n-1)) <= i <= min(n-1, k)

def fft_convolve(x):
    """Linear convolution of x with itself via FFT."""
    n = x.size
    size = 1
    while size < 2 * n:
        size <<= 1
    X = np.fft.rfft(x, size)
    result = np.fft.irfft(X * X, size)[:2 * n - 1]
    return result
