def _lp_update(f, M):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for all k."""
    n = len(f)
    # Build constraint matrix A (2n-1) x n, A[k,j] = f[k-j] if 0<=k-j<n else 0
    # This is a convolution-type matrix
    rows, cols, vals = [], [], []
    for k in range(2*n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            if f[k - j] > 0:
                rows.append(k)
                cols.append(j)
                vals.append(f[k - j])
    A = sparse.coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()
