import time


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # Initialize from parent
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)

        parent = np.ones(6121, dtype=np.float64)

    n = parent.size
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Uniform sequence is a known optimal (score = 2 exactly)
    uniform = np.ones(n, dtype=np.float64)
    u_score = float(evaluate_sequence(uniform.tolist()))
    if np.isfinite(u_score) and u_score < best_score:
        best = uniform.copy()
        best_score = u_score

    # Try mild structured perturbations to confirm optimality / search for edge cases
    step = 0
    while time.monotonic() < deadline - 3.0:
        step += 1
        candidate = best.copy()

        if step % 3 == 1:
            # Small multiplicative block perturbation
            width = int(rng.integers(1, max(2, n // 200)))
            start = int(rng.integers(0, n - width))

        elif step % 3 == 2:

            freq = rng.uniform(0.1, 3.0)
