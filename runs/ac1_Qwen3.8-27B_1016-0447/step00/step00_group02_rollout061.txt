        a_new = a + step * direction
        a_new = np.clip(a_new, 0.0, 1000.0)

        accepted, new_score = _try_candidate(a_new, n, best, best_score, deadline)
        if accepted:
            best = a_new.copy()
            best_score = new_score
            a = a_new.copy()
        else:
            # Try a pure random perturbation
            if rng.random() < 0.4:
                width = int(rng.integers(1, max(2, n // 30)))
                start = int(rng.integers(0, n - width + 1))
                a_trial = a.copy()
                a_trial[start:start + width] *= np.exp(rng.normal(0, 0.2, width))
                a_trial = np.clip(a_trial, 0.0, 1000.0)
                accepted, new_score = _try_candidate(a_trial, n, best, best_score, deadline)
                if accepted:
                    best = a_trial.copy()
                    best_score = new_score
                    a = a_trial.copy()
                else:
                    # Smaller random step
                    a_trial = a + rng.normal(0, base_step * 0.5, n)
                    a_trial = np.clip(a_trial, 0.0, 1000.0)
                    accepted, new_score = _try_candidate(a_trial, n, best, best_score, deadline)
                    if accepted:
                        best = a_trial.copy()
                        best_score = new_score
                        a = a_trial.copy()
                    else:
                        a = np.clip(a + rng.normal(0, base_step * 0.2, n), 0.0, 1000.0)

        # Occasional restart from best with perturbation
        if trial_count % 2000 == 0:
            a = best + rng.normal(0, 0.1, n)
            a = np.clip(a, 0.0, 1000.0)

    return [float(x) for x in best]
