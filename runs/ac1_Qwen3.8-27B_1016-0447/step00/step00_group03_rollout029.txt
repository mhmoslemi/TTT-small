def _fft_conv(a):
    n = a.size
    m = 1
    while m < 2 * n:
        m <<= 1
    fa = np.fft.rfft(a, m)
    return np.fft.irfft(fa * fa, m)[:2 * n - 1]

def _score(a):
    S = a.sum()
    if S < 1e-8:
        return np.inf
    c = _fft_conv(a)
    return 2.0 * a.size * c.max() / (S * S)
