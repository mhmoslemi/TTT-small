import numpy as np
import time

def next_pow2(x):
    y=1
    while y < x:
        y *= 2
    return y

def normalize_positive(arr, eps=1e-12):
    a=np.asarray(arr, dtype=np.float64)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a=np.where(np.isfinite(a), a, 0.0)
    a=np.maximum(a, eps)
    s=a.sum()
    if not np.isfinite(s) or s <= eps:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s

def make_ramp(n):
    x=np.arange(n, dtype=np.float64) + 1.0
    return normalize_positive(x, 1e-12)

def make_power(n, p):
    x=(np.arange(n, dtype=np.float64) + 0.5) / float(n)
    x=np.clip(x, 1e-7, 1.0 - 1e-7)
    f=np.power(x, float(p)) + 1e-9
    return normalize_positive(f, 1e-12)

def make_beta(n, a, b):
    x=(np.arange(n, dtype=np.float64) + 0.5) / float(n)
    x=np.clip(x, 1e-7, 1.0 - 1e-7)
    f=np.power(x, float(a) - 1.0) * np.power(1.0 - x, float(b) - 1.0) + 1e-9
    return normalize_positive(f, 1e-12)

def convolution_fft(f, n, m):
    F=np.fft.rfft(f, m)
    c=np.fft.irfft(F * F, m)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    return c

def weighted_shift(f, w, n, m):
    fr=f[::-1]
    Fr=np.fft.rfft(fr, m)
    W=np.fft.rfft(w, m)
    conv=np.fft.irfft(W * Fr, m)
    out=conv[n - 1:n - 1 + n]
    out=np.where(np.isfinite(out), out, 0.0)
    return np.maximum(out, 0.0)

def score_normalized(f, n, m):
    c=convolution_fft(f, n, m)
    return float(2.0 * float(n) * float(np.max(c)))

def refine_local(f0, t_end, rng, n, m):
    f=normalize_positive(f0, 1e-9)
    score=score_normalized(f, n, m)
    best=f.copy()
    best_score=score
    lr=0.08
    smooth=12.0
    no_improve=0
    while time.monotonic() < t_end:
        c=convolution_fft(f, n, m)
        q=float(n) * c
        qmax=float(np.max(q))
        smooth = min(120.0, smooth + 0.05)
        w=np.exp(smooth * (q - qmax))
        wsum=float(np.sum(w))
        if not np.isfinite(wsum) or wsum <= 0.0:
            w=np.ones(q.size, dtype=np.float64) / float(q.size)
        else:
            w = w / wsum
        h=weighted_shift(f, w, n, m)
        g=float(n) * h
        g = g - np.dot(f, g)
        scale=np.dot(f, np.abs(g))
        if scale <= 1e-14 or not np.isfinite(scale):
            f = f * np.exp(rng.normal(0.0, 0.1, n))
            f = normalize_positive(f, 1e-9)
            score = score_normalized(f, n, m)
            if score < best_score:
                best_score = score
                best = f.copy()
            no_improve = 0
            continue
        g = g / scale
        accepted = False
        trial = lr
        for _ in range(4):
            if time.monotonic() > t_end:
                break
            f2 = f * np.exp(-trial * g)
            s2 = float(np.sum(f2))
            if not np.isfinite(s2) or s2 <= 0.0:
                trial *= 0.5
                continue
            f2 = f2 / s2
            sc2 = score_normalized(f2, n, m)
            if np.isfinite(sc2) and sc2 + 1e-12 < score:
                f = f2
                score = sc2
                accepted = True
                if sc2 < best_score:
                    best_score = sc2
                    best = f.copy()
                lr = min(2.0, lr * 1.05)
                no_improve = 0
                break
            trial *= 0.5

import numpy as np

def next_pow2(x):
    y = 1
    while y < x:
        y *= 2
    return y


def normalize_positive(arr, eps=1e-12):
    a = np.asarray(arr, dtype=np.float64)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, eps)
    s = a.sum()
    if not np.isfinite(s) or s <= eps:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def make_ramp(n):
    x = np.arange(n, dtype=np.float64) + 1.0
    return normalize_positive(x, 1e-12)


def make_power(n, p):
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    x = np.clip(x, 1e-7, 1.0 - 1e-7)
    f = np.power(x, float(p)) + 1e-9
    return normalize_positive(f, 1e-12)


def make_beta(n, a, b):
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    x = np.clip(x, 1e-7, 1.0 - 1e-7)
    f = np.power(x, float(a) - 1.0) * np.power(1.0 - x, float(b) - 1.0) + 1e-9
    return normalize_positive(f, 1e-12)


def convolution_fft(f, n, m):
    F = np.fft.rfft(f, m)
    c = np.fft.irfft(F * F, m)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    return c


def weighted_shift(f, w, n, m):
    fr = f[::-1]
    Fr = np.fft.rfft(fr, m)
    W = np.fft.rfft(w, m)
    conv = np.fft.irfft(W * Fr, m)
    out = conv[n - 1:n - 1 + n]
    out = np.where(np.isfinite(out), out, 0.0)
    return np.maximum(out, 0.0)


def score_normalized(f, n, m):
    c = convolution_fft(f, n, m)
    return float(2.0 * float(n) * float(np.max(c)))
