import time

def fft_conv(a, size):
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)

def cross_corr(w, a, size):
    W = np.fft.rfft(w, size)
    A = np.fft.rfft(a, size)
    return np.fft.irfft(np.conj(W) * A, size)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 5.0

    n = len(a0)
    
    # Normalize
    S = a0.sum()
    if S < 0.01:
        a = np.ones(n)
    else:
        a = a0 / S
    
    # FFT size for convolution (2n-1 elements, pad to power of 2)
    conv_size = 1
    while conv_size < 2 * n - 1:
        conv_size *= 2
    
    # FFT size for cross-correlation (3n-2 elements)
    cc_size = 1
    while cc_size < 3 * n - 2:
        cc_size *= 2
    
    best = a0.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    
    # Subgradient descent
    lr = 0.005
    momentum = np.zeros(n)
    K = min(100, 2 * n - 1)  # number of top peaks to track
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        
        # Compute convolution
        conv = fft_conv(a, conv_size)[:2*n-1]
        peak_val = conv.max()
        
        # If improving, update best (exact eval periodically)
        if it % 200 == 0:
            candidate = a * S  # denormalize
            candidate = np.clip(candidate, 0, 1000)
            if candidate.sum() >= 0.01:
                score = float(evaluate_sequence(candidate.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best = candidate.copy()
        
        # Find top-K peaks
        if K < 2*n-1:
            top_idx = np.argpartition(conv, -K)[-K:]

            top_idx = np.arange(2*n-1)

        if w_sum < 1e-15:
            w_vals = np.ones(len(top_idx)) / len(top_idx)
        else:
            w_vals = w_vals / w_sum

        w = np.zeros(2*n-1)
        w[top_idx] = w_vals
        
        # Compute subgradient via cross-correlation
        sg = 2.0 * cross_corr(w, a, cc_size)[:n]

        sg -= sg.mean()

def _conv_fft(a, size):
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)

        hi = min(n - 1, p)

        idx = np.arange(lo, hi + 1)
        sg[idx] += 2.0 * a[p - idx]
