def _fftconv(a, b):
    no = len(a) + len(b) - 1
    L = 1 << (no - 1).bit_length()

        np.maximum(1.0 / (np.sqrt(j + 0.5) + np.sqrt(n - j + 0.5)), 1e-15),
        np.maximum(1.0 + np.cos(2.0 * np.pi * j / n), 1e-15),
        np.maximum(1.0 / (np.sqrt(j + 0.5) + 1.0), 1e-15),
        np.maximum(1.0 + 0.5 * np.sin(np.pi * j / n), 1e-15),
    ]
    for e in inits:
        es = e.sum()
        if es < 1e-10:
            continue
        e /= es
        sc = _sc(e)
        if sc < bs:
            bs = sc
            best = e.copy()

    a = best.copy()
    eta = 1.0
    stall = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() < deadline:
        c = _fftc(a, a)
        g = _sg(a, c)
        if not g.any():
            break
        ok = False
        for _ in range(4):
            at = a * np.exp(-eta * g)
            at = np.clip(at, 1e-300, 1000.0)
            at /= at.sum()
            sc = _sc(at)
            if sc < bs:
                a = at
                best = at.copy()
                bs = sc
                ok = True
                break
            eta *= 0.5
        if ok:
            stall = 0
            if bs < 1.52:
                eta = max(eta * 0.99, 0.005)
        else:
            eta = max(eta * 0.5, 0.001)
            stall += 1
            if stall % 100 == 0:
                a *= np.exp(rng.normal(0.0, 0.05, n))
                a = np.clip(a, 1e-300, 1000.0)
                a /= a.sum()
                eta = 1.0

    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.clip(a_sym, 1e-300, 1000.0)
    a_sym /= a_sym.sum()
    if _sc(a_sym) < bs:
        bs = _sc(a_sym)
        best = a_sym.copy()

    fs = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(fs):
        return [float(x) for x in a0]
    return [float(x) for x in best]
