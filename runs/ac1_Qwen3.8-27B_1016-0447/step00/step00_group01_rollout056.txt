def fft_conv(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1
    while fsize < size:
        fsize <<= 1
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def make_asym_gauss(n, center, sigma, asym):
