import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def simplex_project(v, s):
    """Project v onto {x >= 0, sum(x) = s} via water-filling."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u > (cssv - s) / np.arange(1, n + 1))[0]
    if len(rho) == 0:
        return np.full(n, s / n)
    theta = (cssv[rho[-1]] - s) / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def make_conv_mat(f):
    """Sparse (2n-1) x n matrix A with (A @ b)[k] = (f * b)[k]."""
    n = len(f)
    idx = np.arange(n)
    rows = np.repeat(idx, n)
    cols = np.tile(idx, n)
    data = np.tile(f, n)
    offset = np.repeat(np.arange(n), n)
    return csr_matrix((data, (rows + offset, cols)), shape=(2 * n - 1, n))


def subgrad_dir(f, k_star):
    """Subgradient of max(f*f) at position k_star: g[m]=2*f[k_star-m]."""
    n = len(f)
    g = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    g[lo:hi + 1] = 2.0 * f[k_star - hi:k_star - lo + 1:-1]
    return g


def lp_step(f, M, deadline):
    """Solve max sum(b) s.t. (f*b)_k <= M, b >= 0. Returns b or None."""
    n = len(f)
    A = make_conv_mat(f)
    try:
        res = linprog(
            -np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
            bounds=[(0.0, None)] * n, method='highs',
            options={'time_limit': max(1.0, deadline - time.monotonic() - 20.0)}
        )
        if res.status == 0 and res.x is not None:
            b = np.clip(res.x, 0.0, None)
            if b.sum() > 1e-12:
                return b / b.sum()
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = len(parent)

    # Build initial candidates from structured patterns
    j = np.arange(n)
    candidates = []
    for period in [3, 4, 5, 6, 7, 8, 9, 10]:
        fp = 1.0 + 0.9 * np.cos(2.0 * np.pi * j / period)
        fp = np.clip(fp, 0.0, None)
        if fp.sum() > 0.01:
            candidates.append(fp / fp.sum())
    if np.sum(parent) > 0.01:
        candidates.append(parent / np.sum(parent))

    best_f = None
    best_score = np.inf
    for c in candidates:
        s = evaluate_sequence(c.tolist())
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_f = c.copy()

    f = best_f.copy().tolist()
    f = np.array(f, dtype=np.float64)

    # --- Main optimization loop ---
    eta = 0.05
    stales = 0
    lp_count = 0
    last_lp_time = 0.0

    while time.monotonic() < deadline:
        h = np.convolve(f, f)
        S = np.sum(f)
        if S < 0.01:
            f = simplex_project(f + rng.normal(0, 0.01, n), 1.0)
            continue
        M = float(np.max(h))
        score = 2.0 * n * M / (S * S)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_f = f.copy()
            stales = 0
        else:
            stales += 1

        # Periodic LP refinement
        if stales % 80 == 0 and lp_count < 120 and time.monotonic() - last_lp_time > 8.0:
            last_lp_time = time.monotonic()
            lp_count += 1
            b = lp_step(f, M, deadline)
            if b is not None:
                for t in (0.15, 0.3, 0.5, 0.7, 1.0):
                    candidate = (1.0 - t) * f + t * b
                    candidate = np.clip(candidate, 0.0, None)
                    cs = evaluate_sequence(candidate.tolist())
                    if np.isfinite(cs) and cs < best_score:
                        best_score = cs
                        best_f = candidate.copy()
                        f = candidate
                        break
                else:
                    # use best blend if any improved
                    f = simplex_project(f, 1.0)
                continue

        # Subgradient step
        k_star = int(np.argmax(h))
        g = subgrad_dir(f, k_star)
        f_new = simplex_project(f - eta * g, 1.0)

        # Occasional random perturbation to escape local structure
        if stales % 150 == 7:
            mask = rng.random(n) < 0.1
            f_new[mask] *= np.exp(rng.normal(0, 0.3, mask.sum() if mask.sum() > 0 else 1))
            f_new = simplex_project(f_new, 1.0)

        f = f_new

        # Adapt step size
        if stales % 20 == 19 and stales > 0:
            eta *= 0.8
        elif stales == 0:
            eta = min(eta * 1.1, 0.5)

        if stales > 1500:
            break

    # Final exact check
    final_score = evaluate_sequence(best_f.tolist())
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    return [float(x) for x in best_f]
