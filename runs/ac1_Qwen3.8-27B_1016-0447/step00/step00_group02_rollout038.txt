import time

def fast_autocorr(a):
    n = len(a)
    size = 2*n - 1
    # Use next fast FFT size
    fsize = 1
    while fsize < size:
        fsize <<= 1
    fa = np.fft.rfft(a, fsize)
    ac = np.fft.irfft(fa * fa, fsize)[:size]
    return ac

    # Two-tone: high for first half, low for second
    inits.append(np.concatenate([np.full(n//2, 2.0), np.full(n-n//2, 0.5)]))
    
    for init in inits[1:]:

        s = float(evaluate_sequence(init.tolist()))
        if np.isfinite(s) and s < best_score:

            best_score = s
    
    # Local search loop
    while time.monotonic() < deadline:

        k_peak = int(np.argmax(b))

        i_lo = max(0, k_peak - n + 1)

        # Contributors: best[i] * best[k_peak - i] for i in [i_lo, i_hi]

        cand[idx] = max(0.0, best[idx] - step)
        
        # Strategy 2: Scale a block near the peak center
        if rng.random() < 0.4:

            if np.isfinite(sc2) and sc2 < best_score:
                best = cand2
                best_score = sc2
                continue
        
        # Strategy 3: Random perturbation
        if rng.random() < 0.3:
            w = int(rng.integers(1, max(2, n // 200)))
            s = int(rng.integers(0, max(1, n - w)))
            cand3 = best.copy()
