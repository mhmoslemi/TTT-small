def damp_update(x, grad, eta, floor=1e-12):

def normalize_positive(x):
    x = np.asarray(x, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.abs(x)
    x = np.where(x >= FLOOR, x, FLOOR)
    s = float(x.sum())
    if s <= 1e-300 or not np.isfinite(s):
        x = np.ones_like(x, dtype=np.float64)
        if x.size > 0:
            x /= x.size
        else:
            x = np.ones(1, dtype=np.float64)
    else:
        x /= s
    return x

def self_conv(a, m, n):
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[: 2 * n - 1]

def self_maxc(a, m, n):
    c = self_conv(a, m, n)
    return float(np.max(c))

def lag_weights(c, bf):
    m = float(np.max(c))
    std = float(np.std(c))
    if std > 1e-300:
        beta = bf / std
    else:
        beta = bf
    u = np.clip(beta * (c - m), -70.0, 0.0)
    w = np.exp(u)
    w += 1e-12
    return w / float(w.sum())

    q = p[::-1]
    fq = np.fft.rfft(q, m2)
    t = np.fft.irfft(Ff * fq, m2)
    return t[n - 1:2 * n - 1][::-1]

    gm = float(np.max(grad))
    if gm <= 1e-300 or not np.isfinite(gm):
        return x
    y = x / (1.0 + eta * grad / gm)

        sub_deadline = min(deadline, time.monotonic() + max(0.75, 0.18 * max(0.0, deadline - time.monotonic())))
        g = solve_subproblem(a, g, m1, m2, n, inner_iters, sub_deadline)
        if time.monotonic() >= deadline:
            break

        best_score = best_mc
        best_cand = None

        c_self = self_conv(a, m1, n)
        w_self = lag_weights(c_self, 14.0)
        Fs = np.fft.rfft(a, m2)
        grad_self = cross_corr_precomputed(Fs, w_self, m2, n) * 2.0
        cand_sg = damp_update(a, grad_self, 0.34)
        mc = self_maxc(cand_sg, m1, n)
        if mc < best_score:
            best_score = mc
            best_cand = cand_sg

        for t in (0.35, 0.15, 0.60, 0.08, 0.22):
            cand = (1.0 - t) * a + t * g
            cand = normalize_positive(cand)
            mc = self_maxc(cand, m1, n)
            if mc < best_score:
                best_score = mc
                best_cand = cand

        mc = self_maxc(g, m1, n)
        if mc < best_score:
            best_score = mc
            best_cand = g.copy()

        if best_cand is not None:
            a = best_cand
            best = a.copy()
            best_mc = best_score
            no_imp = 0
            if best_mc <= target_mc:
                break
        else:
            no_imp += 1
            if no_imp == 5:
                a = block_perturb(a, rng, 0.18)
                g = block_perturb(g, rng, 0.18)
                mc = self_maxc(a, m1, n)
                if mc < best_mc:
                    best = a.copy()
                    best_mc = mc
                no_imp = 0
            elif no_imp == 3:
                a = block_perturb(a, rng, 0.08)

    return best

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 6.0)

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except NameError:
        parent = np.ones(128, dtype=np.float64)

    if parent.size == 0 or not np.all(np.isfinite(parent)):
        parent = np.ones(128, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    if float(parent.sum()) <= 1e-300:
        parent = np.ones(parent.size, dtype=np.float64)

    n = int(parent.size)
    parent = normalize_positive(parent)

    m1 = _next_pow2(2 * n - 1)
    m2 = _next_pow2(3 * n - 2)

    global_best = parent.copy()
    global_mc = self_maxc(global_best, m1, n)

    seeds = derive_seed_strong(n, parent, rng)

    scored = []
    for s in seeds:
        arr = normalize_positive(s)
        mc = self_maxc(arr, m1, n)
        if np.isfinite(mc):
            scored.append((mc, arr))

    if scored:
        scores = np.empty(len(scored), dtype=np.float64)
        arrs = []
        for i in range(len(scored)):
            scores[i] = scored[i][0]
            arrs.append(scored[i][1])
        order = np.argsort(scores)

        selected = [global_best]
        for idx in order:
            if len(selected) >= 4:
                break
            selected.append(arrs[idx])
    else:
        selected = [global_best]

    for i in range(len(selected)):
        if time.monotonic() >= deadline:
            break
        remaining = deadline - time.monotonic()
        share = remaining / float(len(selected) - i)
        sub_deadline = min(deadline, time.monotonic() + share)
        inner_iters = 26 if i < 2 else 22
        cand = refine_sequence(selected[i], rng, m1, m2, n, inner_iters, sub_deadline)
        mc = self_maxc(cand, m1, n)
        if mc < global_mc:
            global_best = cand
            global_mc = mc
        if (2.0 * n * global_mc) < TARGET:
            break

    if time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        sub_deadline = time.monotonic() + 0.62 * remaining
        cand = refine_sequence(global_best, rng, m1, m2, n, 36, sub_deadline)
        mc = self_maxc(cand, m1, n)
        if mc < global_mc:
            global_best = cand
            global_mc = mc

    best_list = [float(v) for v in global_best]
    exact_score = evaluate_sequence(best_list)

    parent_list = [float(v) for v in parent]
    parent_score = evaluate_sequence(parent_list)

    if not np.isfinite(exact_score) or (np.isfinite(parent_score) and parent_score < exact_score):
        return parent_list
    return best_list
