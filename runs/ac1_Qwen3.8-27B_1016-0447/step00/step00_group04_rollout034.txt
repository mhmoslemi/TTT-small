def fast_conv_max(a):

import numpy as np
import time

def fft_conv(a, size=None):
    """Linear convolution of a with itself via FFT."""
    n = len(a)
    if size is None:
        size = 1
        while size < 2*n-1:
            size *= 2
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def fast_score(a):
    """Compute 2n * max_conv / sum^2 quickly."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    c = fft_conv(a)
    return float(2.0 * n * np.max(c) / (s*s))

def subgrad_descent(a, n_iter, deadline, rng):
    """Subgradient descent on the peak of the convolution."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_s = fast_score(a)
    lr = 0.01
    for i in range(n_iter):
        if time.monotonic() > deadline:
            break
        c = fft_conv(a)
        peak_idx = int(np.argmax(c))
        peak_val = c[peak_idx]
        s = np.sum(a)
        # Subgradient of c[peak] w.r.t. a
        # ∂c[k]/∂a[j] = a[k-j] + a[j-k] (0 if out of range)
        j = np.arange(n)
        idx1 = peak_idx - j
        idx2 = j - peak_idx
        sg = np.zeros(n)
        m1 = (idx1 >= 0) & (idx1 < n)
        m2 = (idx2 >= 0) & (idx2 < n)
        sg[m1] += a[idx1[m1]]
        sg[m2] += a[idx2[m2]]
        # Normalized gradient step
        step = lr * sg / peak_val
        a = np.maximum(0.0, a - step)
        if np.sum(a) < 1e-10:
            a = best_a.copy()
            break
        sc = fast_score(a)
        if sc < best_s:
            best_a = a.copy()
            best_s = sc
        # Adaptive learning rate
        if i % 50 == 0:
            lr *= 0.95
    return best_a, best_s

import numpy as np
import time

def _fft_conv(a):
    n = len(a)
    sz = 1 << (2*n-1).bit_length()
    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa*fa, sz)[:2*n-1]

def _fast_score(a):
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    return float(2.0*n*np.max(_fft_conv(a))/(s*s))

def _subgrad(a, n_iter, deadline, rng):
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_s = _fast_score(a)
    lr = 0.05
    js = np.arange(n)
    for _ in range(n_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_conv(a)
        k = int(np.argmax(c))
        pk = c[k]
        if pk < 1e-15:
            break
        i1 = k - js
        i2 = js - k
        sg = np.zeros(n)
        m1 = (i1 >= 0) & (i1 < n)
        m2 = (i2 >= 0) & (i2 < n)
        sg[m1] += a[i1[m1]]
        sg[m2] += a[i2[m2]]
        a = np.maximum(0.0, a - lr*sg/pk)
        s = np.sum(a)
        if s < 1e-10:
            a = best_a.copy()
            break
        sc = _fast_score(a)
        if sc < best_s:
            best_a = a.copy()
            best_s = sc
        lr *= 0.998
    return best_a, best_s
