    fa = np.fft.rfft(a, fs)
    return np.fft.irfft(fa * fa, fs)[:size]


def quick_score(a):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * a.size * np.max(fft_convolve(a)) / (s * s)


def make_two_level(n, ratio, split_frac):
    x = np.empty(n, dtype=np.float64)
    m = int(n * split_frac)
    x[:m] = ratio
    x[m:] = 1.0
    return x


def peak_pressure_move(best, step, rng):
    conv = fft_convolve(best)
    kl = int(np.argmax(conv))
    n = len(best)
    lo = max(0, kl - n + 1)
    hi = min(n - 1, kl)
    if hi <= lo:
        return best.copy()
    rev = best[hi:lo - 1:-1]
    mean_rev = rev.mean()
    best[lo:hi + 1] -= step * (rev - mean_rev)
    return best


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)
    n = parent.size

    cands = [parent.copy()]
    cands.append(make_two_level(n, 2.0, 0.5))
    cands.append(make_two_level(n, 1.5, 0.5))
    cands.append(make_two_level(n, 2.5, 0.5))
    cands.append(make_two_level(n, 2.0, 0.45))
    cands.append(make_two_level(n, 2.0, 0.55))
    cands.append(np.linspace(2.0, 0.5, n, dtype=np.float64))
    cands.append(1.0 + 0.8 * np.exp(-np.linspace(0.0, 3.0, n, dtype=np.float64)))
    cands.append(1.0 - 0.5 * np.linspace(0.0, 1.0, n, dtype=np.float64))

    best = cands[0].copy()
    best_score = float(evaluate_sequence(best.tolist()))
    for c in cands[1:]:
        c = np.clip(c, 0.0, 1000.0)
        s = float(evaluate_sequence(c.tolist()))
        if np.isfinite(s) and s < best_score:
            best = c
            best_score = s

    step = 0.015
    patience = 0
    max_patience = 300

    while time.monotonic() < deadline:
        r = rng.random()
        if r < 0.30:
            best = peak_pressure_move(best.copy(), step, rng)
        elif r < 0.50:
            best = best + rng.normal(0.0, step * max(1e-6, best.mean()), n)
        elif r < 0.65:
            w = int(rng.integers(max(2, n // 300), max(3, n // 6)))
            st = int(rng.integers(0, max(1, n - w)))
            best[st:st + w] *= float(np.exp(rng.normal(0.0, 0.12)))
        elif r < 0.75:
            mu = best.mean()
            best = (1.0 - step) * best + step * mu
        elif r < 0.85:
            half = step * 0.5
            best = (1.0 - half) * best + half * best[::-1]
        else:
            m = int(rng.integers(n // 4, 3 * n // 4))
            R = float(rng.uniform(1.3, 3.0))
            x = np.empty(n, dtype=np.float64)
            x[:m] = R
            x[m:] = 1.0
            best = (1.0 - step) * best + step * x

        best = np.clip(best, 0.0, 1000.0)

        qs = quick_score(best)
        if np.isfinite(qs) and qs < best_score * 1.002:
            s = float(evaluate_sequence(best.tolist()))
            if np.isfinite(s) and s < best_score:
                best_score = s
                patience = 0
                step = max(0.001, step * 0.995)
            else:
                patience += 1
        else:
            patience += 1

        if patience > max_patience:
            step = min(0.05, step * 2.0)
            patience = 0

    return [float(x) for x in best]
