    return irfft(A * A, fft_n)[:2 * len(a) - 1]

def compute_pressure(b, n):

    target = np.mean(b)
    return np.maximum(0.0, b - target)

def compute_gradient(a, pressure, fft_n):

    n = len(a)

    P = rfft(pressure, L2)
    cross = irfft(A * P.conjugate(), L2)

    return w / s
