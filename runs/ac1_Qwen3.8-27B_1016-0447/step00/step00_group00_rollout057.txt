def peak_gradient(a, C, n, topk):
    if topk >= C.size:
        top_idx = np.arange(C.size)
    else:
        top_idx = np.argpartition(C, -topk)[-topk:]

        kt = int(k_t)
        lo = max(0, kt - n + 1)
        hi = min(n - 1, kt)
        if hi >= lo:
            idx = np.arange(lo, hi + 1)
            g[lo : hi + 1] += 2.0 * a[kt - idx]
    g /= len(top_idx)
    g -= g.mean()
    return g

import numpy as np
import time


def _fft_conv(a, N):
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[: 2 * a.size - 1]


def _peak_grad(a, C, n, topk):
    if topk >= C.size:
        top_idx = np.arange(C.size)
    else:
        top_idx = np.argpartition(C, -topk)[-topk:]
    g = np.zeros(n)
    for k_t in top_idx:
        kt = int(k_t)
        lo = max(0, kt - n + 1)
        hi = min(n - 1, kt)
        if hi >= lo:
            idx = np.arange(lo, hi + 1)
            g[lo : hi + 1] += 2.0 * a[kt - idx]
    g /= len(top_idx)
    g -= g.mean()
    return g


def _score_raw(a, C):
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * len(a) * float(np.max(C)) / (s * s)


def _make_inits(parent, n):

    inits = []
    sp = parent.sum()
    if sp > 0.01:
        inits.append(np.maximum(parent, 0.0) / sp)
    inits.append(np.maximum(1.0 + 0.5 * np.cos(2.0 * np.pi * j / n), 0.0))
    inits.append(np.maximum(1.0 + 0.3 * np.cos(4.0 * np.pi * j / n), 0.0))
    inits.append(np.clip(1.0 - ((2.0 * j / n) - 1.0) ** 2, 0.0, None))
    inits.append(np.sin(np.pi * (j + 0.5) / n) ** 2)
    inits.append(np.maximum(1.0 - 0.4 * np.abs(2.0 * j / n - 1.0), 0.0))
    out = []
    for x in inits:
        x = np.maximum(x, 0.0)
        sx = x.sum()
        if sx > 0.01:
            out.append(x / sx)
    return out
