import numpy as np
import time

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    v_sorted = np.sort(v)[::-1]
    # Find k such that v_sorted[k] > (sum(v_sorted[:k+1]) - 1) / (k+1)
    cumsum = np.cumsum(v_sorted)
    k = np.arange(1, n + 1)
    tau_candidates = (cumsum - 1.0) / k
    mask = v_sorted > tau_candidates
    k_star = np.sum(mask)  # number of elements above their tau
    if k_star == 0:
        return np.ones(n) / n
    tau = tau_candidates[k_star - 1]
    return np.maximum(v - tau, 0.0)

    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    n = a.size
    if n == 0:
        return [1.0]
    S = a.sum()
    if S < 1e-10:
        a = np.ones(n)
    else:
        a = a / S
    
    best_a = a.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))
    
    # Parameters
    gamma = 500.0  # exponential temperature
    momentum = 0.9
    
    v = np.zeros(n)  # momentum vector
    alpha = 1.0  # base step size
    
    for t in range(200000):
        if time.monotonic() > deadline:
            break
        
        # Compute convolution
        conv = np.convolve(a, a)  # length 2n-1
        m = conv.max()
        
        # Exponential weights
        w = np.exp(gamma * (conv - m))  # in [0, 1]
        
        # Gradient via convolution
        conv_aw = np.convolve(a[::-1], w)
        grad = conv_aw[n-1 : 2*n-1]  # length n
        grad *= 2.0 * gamma
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:

        # Momentum update

        v = momentum * v - step * grad / gnorm
        a_new = a + v

        # Ensure nonnegativity and sum
        a = a_new

import numpy as np
import time


def project_simplex(v):
    n = v.size
    if n <= 1:
        return np.ones(max(n, 1), dtype=np.float64) / max(n, 1)
    s = np.sort(v)[::-1]
    cs = np.cumsum(s)
    k = np.arange(1, n + 1, dtype=np.float64)
    tau_c = (cs - 1.0) / k
    ks = int(np.sum(s > tau_c))
    if ks == 0:
        return np.ones(n, dtype=np.float64) / n
    return np.maximum(v - tau_c[ks - 1], 0.0)
