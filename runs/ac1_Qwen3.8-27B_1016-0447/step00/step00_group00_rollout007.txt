import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse


def build_conv_matrix(f, n):
    """Sparse (2n-1 x n) matrix A where (A@g)[k] = (f*g)[k]."""
    row_parts = []
    col_parts = []
    val_parts = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_lo > j_hi:
            continue
        js = np.arange(j_lo, j_hi + 1)
        row_parts.append(np.full(len(js), k, dtype=np.int32))
        col_parts.append(js.astype(np.int32))
        val_parts.append(f[k - js])
    if not row_parts:
        return sparse.csr_matrix((2 * n - 1, n))
    rows = np.concatenate(row_parts)
    cols = np.concatenate(col_parts)
    vals = np.concatenate(val_parts)
    return sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_dual_lp(f, n, timeout_s=90):
    """Min t s.t. (f*g)[k]<=t (all k), sum(g)=1, g>=0. Returns g or None."""
    A = build_conv_matrix(f, n)
    m = 2 * n - 1
    c = np.zeros(n + 1)
    c[n] = 1.0
    t_col = sparse.csc_matrix(np.full((m, 1), -1.0))
    A_ub = sparse.hstack([A.tocsr(), t_col], format='csr')
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'time_limit': timeout_s, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x[:n], 0.0)
        s = g.sum()
        if s > 1e-12:
            return g / s
    return None
