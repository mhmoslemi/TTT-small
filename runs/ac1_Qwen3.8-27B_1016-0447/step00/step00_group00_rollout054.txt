import time


def ac_conv(a):
    """Linear auto-convolution via FFT: O(n log n)."""
    n = len(a)
    sz = 1 << (2 * n - 2).bit_length()
    f = np.fft.rfft(a, sz)
    return np.fft.irfft(f * f, sz)[:2 * n - 1]
