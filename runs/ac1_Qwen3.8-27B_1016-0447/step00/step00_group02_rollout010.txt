    for harm, amp in [(1, 0.4), (1, 0.7), (2, 0.5), (3, 0.4), (5, 0.3), (13, 0.25)]:
        inits.append(1.0 + amp * np.sin(2.0 * np.pi * harm * r / n))
    # linear ramps
    inits.append(1.0 + 0.5 * (2.0 * t - 1.0))
    inits.append(1.0 - 0.5 * (2.0 * t - 1.0))
    # bumps and decays
    inits.append(np.exp(-3.0 * (r - 0.3 * n) ** 2 / n))
    inits.append(np.exp(-3.0 * (r - 0.7 * n) ** 2 / n))
    inits.append(np.exp(-4.0 * t))
    inits.append(np.exp(-4.0 * (1.0 - t)))
    # two-tone
    inits.append(1.0 + 0.3 * np.sin(2.0 * np.pi * r / n * 2.0) * np.cos(2.0 * np.pi * r / n * 5.0))
    # random starts
    rs = np.random.RandomState(seed)
    inits.append(rs.uniform(0.3, 1.7, n))
    inits.append(1.0 + 0.3 * rs.normal(size=n))
    inits.append(1.0 + 0.5 * np.sign(rs.normal(size=n)) * np.abs(rs.normal(scale=0.2, size=n)))
    return inits


def _subgradient_step(a, k_star, n, lr):
    sg = a.copy()
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    if lo <= hi:
        sg[lo:hi + 1] += a[k_star - hi:k_star - lo + 1][::-1]
    mx = np.max(sg)
    if mx > 1e-30:
        sg /= mx
    return np.maximum(0.0, a - lr * sg)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(6962, dtype=np.float64)
    n = parent.size

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Phase 1: evaluate structured + random initializations
    inits = _try_init_list(n, rng, seed)
    for init in inits:
        if time.monotonic() > deadline:
            break
        init = np.clip(init, 0.0, 1000.0).astype(np.float64)
        if np.sum(init) < 0.01:
            continue
        s = _score_fast(init, n)
        if np.isfinite(s) and s < best_score:
            best_score = s
            best = init.copy()

    # Phase 2: local search
    a = best.copy()
    stall = 0
    stall_limit = 8000

    while time.monotonic() < deadline and stall < stall_limit:
        c = np.convolve(a, a, mode='full')
        s_a = np.sum(a)
        if s_a < 0.01:
            break
        k_star = int(np.argmax(c))
        cur_score = 2.0 * n * c[k_star] / (s_a * s_a)

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            stall = 0
        else:
            stall += 1

        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        improved = False

        # Strategy A: shrink top contributors to the peak
        if lo <= hi and (hi - lo + 1) > 2:
            contribs = a[lo:hi + 1] * a[k_star - hi:k_star - lo + 1][::-1]
            topk = min(15, hi - lo + 1)
            top_local = np.argpartition(contribs, -topk)[-topk:]
            top_global = lo + top_local
            for frac in (0.5, 0.3, 0.7):
                new_a = a.copy()
                new_a[top_global] *= frac
                ns = _score_fast(new_a, n)
                if np.isfinite(ns) and ns < cur_score:
                    a = new_a
                    improved = True
                    break
        if improved:
            continue

        # Strategy B: subgradient step (a few step sizes)
        for lr in (0.05, 0.02, 0.08, 0.01):
            new_a = _subgradient_step(a, k_star, n, lr)
            if np.sum(new_a) < 0.01:
                continue
            ns = _score_fast(new_a, n)
            if np.isfinite(ns) and ns < cur_score:
                a = new_a
                improved = True
                break
        if improved:
            continue

        # Strategy C: random block multiplicative perturbation
        for _try in range(3):
            w = max(1, int(rng.integers(1, max(2, n // 80))))
            st = int(rng.integers(0, max(1, n - w)))
            new_a = a.copy()
            new_a[st:st + w] *= float(np.exp(rng.normal(0.0, 0.12)))
            new_a = np.clip(new_a, 0.0, 1000.0)
            ns = _score_fast(new_a, n)
            if np.isfinite(ns) and ns < cur_score:
                a = new_a
                improved = True
                break
        if improved:
            continue

        # Strategy D: move mass from heavy to light element
        if n > 20:
            for _try in range(2):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                delta = a[i] * float(rng.uniform(0.05, 0.35))
                if delta <= 0 or delta >= a[i]:
                    continue
                new_a = a.copy()
                new_a[i] -= delta
                new_a[j] += delta
                ns = _score_fast(new_a, n)
                if np.isfinite(ns) and ns < cur_score:
                    a = new_a
                    improved = True
                    break
        if improved:
            continue

        # Fallback: pure random small perturbation to escape plateau
        idx = int(rng.integers(0, n))
        wd = max(1, int(rng.integers(1, max(2, n // 200))))
        lo2 = max(0, idx - wd)
        hi2 = min(n, idx + wd)
        a[lo2:hi2] *= float(np.exp(rng.normal(0.0, 0.08)))
        a = np.clip(a, 0.0, 1000.0)

    # Final: use the authoritative evaluator on the best candidate
    final = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final):
        final = float(evaluate_sequence(parent.tolist()))
        best = parent.copy()

    return [float(x) for x in best]
