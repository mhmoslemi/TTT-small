            h0_scaled = np.tile(initial_h_values, n_point_val // n_points_parent_val)
            print(f"Upscale error: Using fallback parent initialization.")

        print(f"Optimizing with n_points={n_point_val}")
        try:
            result = optimize_with_n(n_point_val, h0_scaled)
            h_optimal = result.x
            c5_optimal = evaluate_erdos_solution(h_optimal)
            print(f"Optimized n_points={n_point_val} -> c5={c5_optimal}")
        except Exception as e:
            print(f"Optimization error for n_points={n_point_val}: {e}")
            h_optimal = np.copy(h0_scaled)  # Fallback
            c5_optimal = 1.0  # Default high value

        # Update if better
        if best_score > c5_optimal:
            best_score = c5_optimal
            best_h = h_optimal
            best_n_points = n_point_val

    # End execution
    total_time = time() - t_start
    if total_time > budget_s:
        # Ensure we exit promptly
        pass  # Already scheduled to end

    return best_h, best_score, best_n_points
