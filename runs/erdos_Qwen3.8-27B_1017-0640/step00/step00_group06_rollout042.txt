def exact_score(h, n):
    dx = 2.0 / n
    q = np.correlate(h, 1.0 - h, mode="full")

def gauss_seidel_sweep(h, q, score, pairs, n, rng):
    h = h.copy()
    n_pairs = len(pairs)
    order = rng.permutation(n_pairs)
    for oi in order:
        i, j = pairs[oi]
        try:
            delta = line_minimize(h, i, j, q, n)
            if abs(delta) < 1e-14:
                continue
            h_trial = h.copy()
            h_trial[i] += delta
            h_trial[j] -= delta
            if not np.all(np.isfinite(h_trial)):
                continue
            if h_trial[i] < -1e-12 or h_trial[i] > 1.0 + 1e-12:
                continue
            if h_trial[j] < -1e-12 or h_trial[j] > 1.0 + 1e-12:
                continue
            h_trial[i] = min(max(h_trial[i], 0.0), 1.0)
            h_trial[j] = min(max(h_trial[j], 0.0), 1.0)

            m = n / 2.0
            r = m - math.fsum(h_trial.tolist())
            for _ in range(3):

                    break
                if r > 0:

                        if r <= 0:
                            break

                        h_trial[t] += add
                        r -= add

                        h_trial[t] -= sub
                        r += sub

                q = q_trial

        except Exception:
            continue

def exact_score(h, n):

def is_feasible(h, n, tol=1e-10):

        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.min(h) < -tol or np.max(h) > 1.0 + tol:
        return False
    return abs(math.fsum(h.tolist()) - n / 2.0) <= tol
