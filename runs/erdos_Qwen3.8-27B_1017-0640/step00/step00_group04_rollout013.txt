                    best_h, best_s = hb, sb
            except: pass
    if use_doubled and n0 <= 64:
        n1 = 2 * n0
        anchors1 = [np.repeat(fallback, 2), np.repeat(best_h, 2)]
        starts1 = build_starts(n1, anchors1, rng, n_starts)
        for st in starts1:
            try:
                hb, sb = pgd_search(st, tau_sched, max_iter)
                if np.isfinite(sb) and sb < best_s - 1e-12:
                    best_h, best_s = hb, sb
            except: pass
    if not check_feasible(best_h):
        pr = capped_project(best_h, len(best_h))
        if pr is not None and check_feasible(pr):
            best_h = pr
        else:
            best_h = fallback
    s_final = exact_score(best_h)
    if not np.isfinite(s_final):
        best_h, s_final = fallback, fb_score
    return (best_h.astype(np.float64), float(s_final), int(len(best_h)))
