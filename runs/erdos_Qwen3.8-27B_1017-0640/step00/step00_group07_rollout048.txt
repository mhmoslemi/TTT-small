def is_feasible(h, n):
    try:
        return (h.ndim == 1 and len(h) == n and np.all(np.isfinite(h))

def project_capped(y, n):

        y = np.asarray(y, dtype=np.float64).ravel()
        if len(y) != n or not np.all(np.isfinite(y)) or np.max(np.abs(y)) > 1e12:
            return np.full(n, 0.5)
    except Exception:
        return np.full(n, 0.5)
    m = n / 2.0
    lo, hi = np.min(y) - 1.0, np.max(y) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(y - mid, 0.0, 1.0)) > m:
            lo = mid
        else:
            hi = mid
    p = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    resid = m - np.sum(p)
    for _ in range(3):
        if abs(resid) <= 2e-14 * max(1, n):
            break
        for i in range(n):
            if resid > 1e-15:
                a = min(resid, 1.0 - p[i]); p[i] += a; resid -= a
            elif resid < -1e-15:
                a = min(-resid, p[i]); p[i] -= a; resid += a
            else:
                break
    return p if is_feasible(p, n) else np.full(n, 0.5)

def maj_val(h, r, n):
    dx, d = 2.0 / n, h - r
    Q = np.empty(2 * n - 1)
    Q[0] = dx * np.sum(r * (1.0 - r) + (1.0 - 2.0 * r) * d)
    for s in range(1, n):

        Q[2*s] = dx * np.sum(r[s:]*(1.0-r[:p]) + (1.0-r[:p])*d[s:] - r[s:]*d[:p] + 0.5*d[:p]**2 + 0.5*d[s:]**2)
    return Q

def maj_jac(h, r, n):

    G = np.zeros((2 * n - 1, n))
    G[0] = dx * (1.0 - 2.0 * r)
    for s in range(1, n):
        p = n - s
        G[2*s-1, :p] += dx * (1.0 - r[s:] + d[:p])
        G[2*s-1, s:] += dx * (-r[:p] + d[s:])
        G[2*s, :p] += dx * (-r[s:] + d[:p])
        G[2*s, s:] += dx * (1.0 - r[:p] + d[s:])
    return G

    g = np.zeros(n + 1); g[n] = 1.0; return g
