                    if float(np.max(np.abs(delta))) <= 1e-9:
                        return best
                    break
                eta *= 0.5
            if not accepted:
                break
            Fc, g = smooth_val_grad(current, tau, True)
            if Fc is None:
                break
    return best

def run(seed=42, **kwargs):
    try:
        s = int(seed)
        if s < 0 or s > 4294967295:
            s = 42
    except (ValueError, TypeError):
        s = 42
    rng = np.random.default_rng(s)
    parent = None
    n = 47
    try:
        ph = globals().get('initial_h_values')
        if ph is not None:
            arr = np.array(ph, dtype=np.float64).flatten()
            if 2 <= len(arr) < 1000 and np.all(np.isfinite(arr)) and arr.min() >= -1e-7 and arr.max() <= 1.0 + 1e-7:
                n = len(arr)
                proj = project_capped_simplex(arr)
                if proj is not None:
                    parent = proj
    except Exception:
        parent = None
    if parent is None:
        n = 47
        parent = np.full(n, 0.5, dtype=np.float64)
    n = len(parent)
    if n <= 64:
        temps = [0.04, 0.01, 0.0025, 0.0005, 0.0001]
        iters, max_st, rstart = 80, 8, [0.04, 0.12]
    elif n <= 128:
        temps = [0.04, 0.01, 0.0025, 0.0005]
        iters, max_st, rstart = 45, 5, [0.06]
    elif n <= 512:
        temps = [0.03, 0.006, 0.001]
        iters, max_st, rstart = 25, 3, []
    else:
        temps = [0.02, 0.003]
        iters, max_st, rstart = 12, 2, []
    starts = [parent, np.full(n, 0.5, dtype=np.float64)]
    base = parent
    i_arr = np.arange(n, dtype=np.float64)
    for freq in [1, 2]:
        phi = rng.uniform(0, 2 * math.pi)
        v = 0.5 + 0.30 * np.sin(2 * math.pi * freq * (i_arr + 0.5) / n + phi)
        p = project_capped_simplex(v)
        if p is not None:
            starts.append(p)
    for sc in [0.06, 0.18]:
        v = base + sc * rng.standard_normal(n)
        p = project_capped_simplex(v)
        if p is not None:
            starts.append(p)
    for _ in range(2):
        p = project_capped_simplex(rng.uniform(0, 1, n).astype(np.float64))
        if p is not None:
            starts.append(p)
    if parent is None or parent is np.full(n, 0.5, dtype=np.float64):
        phi = rng.uniform(0, 2 * math.pi)
        v = 0.5 + 0.30 * np.cos(2 * math.pi * 3 * (i_arr + 0.5) / n + phi)
        p = project_capped_simplex(v)
        if p is not None:
            starts.append(p)
    unique = [starts[0]]
    for st in starts[1:]:
        if not any(float(np.max(np.abs(st - u))) <= 1e-9 for u in unique):
            unique.append(st)
    starts = unique[:max_st]
    gb = parent.copy()
    gbs = exact_score(gb)
    if gbs is None:
        gbs = 999.0
    for st in starts:
        try:
            sc = exact_score(st)
            if sc is not None and sc < gbs - 1e-12:
                gb, gbs = st.copy(), sc
            res = projected_smooth_search(st, temps, iters)
            rs = exact_score(res)
            if rs is not None and rs < gbs - 1e-12:
                gb, gbs = res.copy(), rs
        except Exception:
            continue
    for sc in rstart:
        try:
            v = gb + sc * rng.standard_normal(n)
            p = project_capped_simplex(v)
            if p is None:
                continue
            ps = exact_score(p)
            if ps is not None and ps < gbs - 1e-12:
                gb, gbs = p.copy(), ps
            res = projected_smooth_search(p, temps, iters)
            rs = exact_score(res)
            if rs is not None and rs < gbs - 1e-12:
                gb, gbs = res.copy(), rs
        except Exception:
            continue
    fin = project_capped_simplex(gb)
    if fin is None:
        fin = gb
    fs = exact_score(fin)
    if fs is None:
        fp = project_capped_simplex(parent)
        if fp is not None:
            fin, fs = fp, exact_score(fp)
    if fs is None:
        fin = np.full(n, 0.5, dtype=np.float64)
        fs = 0.5
    return (fin, fs, n)
