                hull.append([b, c, -1e18])
            else:
                if b == hull[-1][0]:
                    if c <= hull[-1][1]: continue
                    hull.pop()
                    if not hull:
                        hull.append([b, c, -1e18]); continue
                xi = (hull[-1][1] - c) / (b - hull[-1][0])
                while hull and xi <= hull[-1][2] + 1e-15:
                    hull.pop()
                    if not hull: xi = -1e18; break
                    xi = (hull[-1][1] - c) / (b - hull[-1][0])
                hull.append([b, c, xi])
        segments = []
        for idx in range(len(hull)):
            xs = hull[idx][2]
            xe = hull[idx + 1][2] if idx + 1 < len(hull) else 1e18
            s0, s1 = max(xs, lo), min(xe, hi)
            if s1 - s0 > 1e-15:
                segments.append((hull[idx][0], hull[idx][1], s0, s1))
    sp = [(float(A[k]), float(B[k]), float(D[k])) for k in spec]
    best_d, best_v = 0.0, float('inf')
    for (lb, lc, s0, s1) in segments:
        polys = [(0.0, float(lb), float(lc))]
        polys.extend(sp)
        cands = [s0, s1]
        if lo <= 0.0 <= hi: cands.append(0.0)
        for p in range(len(polys)):
            pa, pb, pc = polys[p]
            if abs(pa) > 1e-15:
                st = -pb / (2.0 * pa)
                if s0 - 1e-12 <= st <= s1 + 1e-12:
                    cands.append(max(s0, min(s1, st)))
            for p2 in range(p + 1, len(polys)):
                qa, qb, qc = polys[p2]
                da, db, dc = pa - qa, pb - qb, pc - qc
                if abs(da) < 1e-15:
                    if abs(db) > 1e-15:
                        rt = -dc / db
                        if s0 - 1e-12 <= rt <= s1 + 1e-12:
                            cands.append(max(s0, min(s1, rt)))
                else:
                    disc = db * db - 4.0 * da * dc
                    tol = 1e-12 * (db * db + abs(4.0 * da * dc) + 1.0)
                    if disc > -tol:
                        disc = max(0.0, disc)
                        sq = math.sqrt(disc)
                        z = -0.5 * (db + (sq if db >= 0 else -sq))
                        if abs(z) > 1e-15:
                            rts = [z / da, dc / z]
                        else:
                            rts = [-db / (2.0 * da)]
                        for rt in rts:
                            if np.isfinite(rt) and s0 - 1e-12 <= rt <= s1 + 1e-12:
                                cands.append(max(s0, min(s1, rt)))
        for dc2 in cands:
            val = max(ppa * dc2 * dc2 + ppb * dc2 + ppc for ppa, ppb, ppc in polys)
            if not np.isfinite(val): continue
            if val < best_v - 1e-14 * max(1.0, abs(best_v)):
                best_v, best_d = val, dc2
            elif abs(val - best_v) <= 1e-14 * max(1.0, abs(best_v)) and abs(dc2) < abs(best_d):
                best_d = dc2
    if (2.0 / n) * (float(np.max(q)) - best_v) <= 1e-12: return 0.0
    return best_d
