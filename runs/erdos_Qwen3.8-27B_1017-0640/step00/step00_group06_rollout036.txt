                elite_idx = int(order[0])
                worst_idx = list(range(P - 1, P - 1 - R, -1))
                worst_idx = [int(order[j]) for j in worst_idx]
                for k, wi in enumerate(worst_idx):
                    if wi == elite_idx:
                        continue
                    if k % 2 == 0:
                        v = global_best + rng.standard_normal(n) * restart_scales[restarts_done % 4]
                    else:
                        v = rng.beta(0.7, 0.7, n).astype(np.float64)
                    proj = project_capped(v, n, m)
                    if proj is not None and is_feasible(proj, n, m):
                        pop[wi] = proj
                        eval_count += 1
                        s = exact_score(proj, n)
                        sc[wi] = s if s is not None else 1e18
                        if s is not None and s < global_best_score - 1e-12:
                            global_best = proj.copy()
                            global_best_score = s
                    else:
                        pop[wi] = global_best.copy()
                        sc[wi] = global_best_score
                restarts_done += 1
                stagnation = 0
    h_final = global_best
    if not is_feasible(h_final, n, m):
        proj = project_capped(h_final, n, m)
        if proj is not None and is_feasible(proj, n, m):
            h_final = proj
        else:
            h_final = fallback
    if not is_feasible(h_final, n, m):
        h_final = half_block
    c5 = exact_score(h_final, n)
    if c5 is None or not math.isfinite(c5):
        c5 = exact_score(fallback, n)
    if c5 is None or not math.isfinite(c5):
        c5 = 0.5
    return h_final, float(c5), n
