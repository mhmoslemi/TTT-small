def _project(v, l, u, m):
    v = np.asarray(v, dtype=np.float64)
    l = np.asarray(l, dtype=np.float64)
    u = np.asarray(u, dtype=np.float64)
    if not np.all(np.isfinite(v)) or not np.all(np.isfinite(l)) or not np.all(np.isfinite(u)):
        return None
    if math.fsum(l) > m + 1e-8 or math.fsum(u) < m - 1e-8:
        return None
    lo = float(np.min(v) - np.max(u)) - 1.0
    hi = float(np.max(v) - np.min(l)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        x = np.clip(v - mid, l, u)
        mass = math.fsum(x)
        if abs(mass - m) < 1e-13:
            lo = hi = mid
            break
        if mass > m:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-15:
            break
    x = np.clip(v - 0.5 * (lo + hi), l, u)
    for _ in range(3):
        r = m - math.fsum(x)
        if abs(r) < 1e-13:
            break
        if r > 0:
            for i in range(len(x)):
                a = min(r, u[i] - x[i])
                x[i] += a
                r -= a
                if r <= 0:
                    break
        else:
            for i in range(len(x)):
                s = min(-r, x[i] - l[i])
                x[i] -= s
                r += s
                if r >= 0:
                    break
    if not np.all(np.isfinite(x)) or np.any(x < l - 1e-10) or np.any(x > u + 1e-10):
        return None
    if abs(math.fsum(x) - m) > 1e-9:
        return None
    return x


def _lag_data(h, n):
    q = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(q)):
        return None, None
    G = np.zeros((2 * n - 1, n))
    for a in range(2 * n - 1):
        k = a - (n - 1)
        if k >= 0:
            rlo = 0
            rhi = n - 1
            for r in range(rlo, rhi + 1):
                if 0 <= r - k:
                    G[a, r] += (1.0 - h[r - k])
                if r + k < n:
                    G[a, r] -= h[r + k]
        else:
            for r in range(n):
                if r - k < n and r - k >= 0:
                    G[a, r] += (1.0 - h[r - k])
                if r + k < n and r + k >= 0:
                    G[a, r] -= h[r + k]
    if not np.all(np.isfinite(G)):
        return None, None
    return q, G


def _lp_step(h, delta, q, G, m, n):
    l = np.maximum(0.0, h - delta)
    u = np.minimum(1.0, h + delta)
    nv = n + 1
    c = np.zeros(nv)
    c[-1] = 1.0
    A_ub = np.hstack([G, -np.ones((len(q), 1))])
    b_ub = G @ h - q
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.array([m])
    bounds = [(float(l[i]), float(u[i])) for i in range(n)] + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', maxiter=500)
    except Exception:
        return None
    if res is None or not res.success:
        return None
    xp = res.x[:n]
    if not np.all(np.isfinite(xp)):
        return None
    xp = np.clip(xp, l, u)
    return _project(xp, l, u, m)


def _half_block(n):
    h = np.zeros(n)
    h[:n // 2] = 1.0
    if n % 2 == 1:
        h[n // 2] = 0.5
    return h


def _make_starts(base, n, m, rng):
    starts = [base.copy()]
    for sig in [0.05, 0.12, 0.25, 0.40]:
        z = rng.standard_normal(n)
        pr = _project(base + sig * z, np.zeros(n), np.ones(n), m)
        if pr is not None:
            starts.append(pr)
    b = rng.beta(0.7, 0.7, size=n)
    pr = _project(b, np.zeros(n), np.ones(n), m)
    if pr is not None:
        starts.append(pr)
    ph = rng.uniform(0, 2 * np.pi)
    idx = np.arange(n, dtype=np.float64)
    s = 0.5 + 0.45 * np.sin(2.0 * np.pi * idx / n + ph)
    pr = _project(s, np.zeros(n), np.ones(n), m)
    if pr is not None:
        starts.append(pr)
    hb = _half_block(n)
    starts.append(hb)
    good = []
    for s in starts:
        if s is None or not _feasible(s, n, m):
            continue
        dup = False
        for g in good:
            if np.max(np.abs(s - g)) < 1e-12:
                dup = True
                break
        if not dup:
            good.append(s)
    return good


def _optimize(h, n, m, dx, max_steps):
    delta = 0.12
    dmax = 0.25
    dmin = 1e-6
    C = _score(h, n)
    accepted = 0
    for _ in range(max_steps):
        q, G = _lag_data(h, n)
        if q is None:
            break
        got = False
        for attempt in range(5):
            x = _lp_step(h, delta, q, G, m, n)
            if x is not None:
                Cx = _score(x, n)
                A = C - Cx
                if A > 1e-12:
                    d = x - h
                    M = float(np.max(q + G @ d) * dx)
                    P = C - M
                    ss = float(np.max(np.abs(d)))
                    h = x
                    C = Cx
                    accepted += 1
                    if P <= 1e-12 or (A / P) < 0.25:
                        delta = max(dmin, 0.5 * delta)
                    elif (A / P) > 0.75 and ss >= 0.8 * delta:
                        delta = min(dmax, 1.5 * delta)
                    got = True
                    break
                else:
                    d = x - h
                    M = float(np.max(q + G @ d) * dx)
                    P = C - M
                    if P <= 1e-12:
                        return h, C
            delta = max(dmin, 0.5 * delta)
            if delta < dmin:
                return h, C
        if not got:
            break
    return h, C


def run(seed=42, **kwargs):
    n = 95
    base = None
    try:
        ihv = globals().get('initial_h_values', None)
        if ihv is not None:
            ihv = np.asarray(ihv, dtype=np.float64)
            if ihv.ndim == 1 and 2 <= len(ihv) < 1000:
                n = len(ihv)
                if np.all(np.isfinite(ihv)):
                    base = ihv.copy()
                else:
                    tmp = ihv.copy()
                    tmp[~np.isfinite(tmp)] = 0.0
                    base = tmp
    except Exception:
        pass
    if base is None:
        nk = kwargs.get('n_points', None)
        if nk is not None and isinstance(nk, (int, np.integer)) and 2 <= int(nk) < 1000:
            n = int(nk)
        base = _half_block(n)
    m = n / 2.0
    dx = 2.0 / n
    if not _feasible(base, n, m):
        pr = _project(base, np.zeros(n), np.ones(n), m)
        if pr is not None and _feasible(pr, n, m):
            base = pr
        else:
            base = _half_block(n)
    protected = base.copy()
    C_prot = _score(protected, n)
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    starts = _make_starts(base, n, m, rng)
    best = protected.copy()
    best_c = C_prot
    for s in starts:
        cs = _score(s, n)
        if cs < best_c - 1e-12:
            best = s.copy()
            best_c = cs
    B0 = max(8, min(30, math.ceil(30.0 * 95 / n)))
    B1 = max(5, min(14, math.ceil(14.0 * 95 / n)))
    for i, s in enumerate(starts):
        budget = B0 if i == 0 else B1
        try:
            ho, co = _optimize(s.copy(), n, m, dx, budget)
            if co < best_c - 1e-12:
                best = ho.copy()
                best_c = co
        except Exception:
            pass
    if not _feasible(best, n, m):
        pr = _project(best, np.zeros(n), np.ones(n), m)
        if pr is not None and _feasible(pr, n, m):
            best = pr
        else:
            best = protected.copy()
    c5 = _score(best, n)
    return best, float(c5), n
