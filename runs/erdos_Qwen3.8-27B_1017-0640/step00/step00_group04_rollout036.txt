            mask[rng.integers(n)] = True
            u = np.where(mask, v, snap[i])
            cand = project_capped(u, n)
            if cand is None or not is_feasible(cand, n):
                continue
            sc = score_h(cand, n)
            if sc is None:
                continue
            if sc < snap_sc[i] - 1e-12:
                nxt[i] = cand
                nxt_sc[i] = sc
            if sc < best_score - 1e-12:
                best_score = sc
                best_vec = np.array(cand, dtype=np.float64)

        pop = nxt
        scores = nxt_sc

        if best_score < float(np.min(scores)) + 1e-12:
            stagnant = 0
        else:
            stagnant += 1

        if stagnant >= 20 and injections < 3:
            order = np.argsort(scores)
            n_rep = max(1, math.ceil(0.20 * NP))
            worst = order[-n_rep:]
            for ki in range(len(worst)):
                ii = int(worst[ki])
                if ki % 2 == 0:
                    cand = project_capped(best_vec + rng.standard_normal(n) * inj_ns[ki % 4], n)
                else:
                    cand = project_capped(rng.uniform(-0.25, 1.25, n), n)
                if cand is not None and is_feasible(cand, n):
                    sc = score_h(cand, n)
                    if sc is not None:
                        pop[ii] = cand
                        scores[ii] = sc
                        if sc < best_score - 1e-12:
                            best_score = sc
                            best_vec = np.array(cand, dtype=np.float64)
            injections += 1
            stagnant = 0
        elif stagnant >= 40 and injections >= 3:
            break

    h_out = np.array(best_vec, dtype=np.float64)
    c_out = score_h(h_out, n)
    if c_out is None or c_out > fb_score + 1e-12:
        h_out = np.array(fallback, dtype=np.float64)
        c_out = fb_score
    else:
        h_out = np.clip(h_out, 0.0, 1.0)
        r5 = n / 2.0 - float(np.sum(h_out))
        if r5 > 0:
            for i in range(n):
                if r5 <= 0:
                    break
                a = min(r5, 1.0 - h_out[i])
                h_out[i] += a
                r5 -= a
        elif r5 < 0:
            for i in range(n):
                if r5 >= 0:
                    break
                a = min(-r5, h_out[i])
                h_out[i] -= a
                r5 += a
        c_out = score_h(h_out, n)
        if c_out is None:
            h_out = np.array(fallback, dtype=np.float64)
            c_out = fb_score

    return np.array(h_out, dtype=np.float64), float(c_out), int(n)
