import numpy as np
from scipy.optimize import linprog

def check_feasible(h, n):
    if h.ndim != 1 or len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10:
        return False
    return True

def joint_project(v, n):
    v = np.array(v, dtype=np.float64)
    if v.ndim != 1 or len(v) != n or not np.all(np.isfinite(v)):
        return None
    v = np.clip(v, -1e6, 1e6)
    target = n / 2.0
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        q = np.clip(v - mid, 0.0, 1.0)
        if np.sum(q) > target:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2.0
    q = np.clip(v - mid, 0.0, 1.0)
    # Residual correction
    for _ in range(2):
        res = target - np.sum(q)
        if abs(res) < 1e-14:
            break
        for i in range(n):
            if res > 0:
                add = min(res, 1.0 - q[i])
                q[i] += add
                res -= add
            else:
                sub = min(-res, q[i])
                q[i] -= sub
                res += sub
            if abs(res) < 1e-14:
                break
    if not np.all(np.isfinite(q)) or np.any(q < -1e-12) or np.any(q > 1 + 1e-12):

        return np.inf

def compute_lags(h, n):
    dx = 2.0 / n
    c = np.zeros(2 * n - 1)
    J = np.zeros((2 * n - 1, n))
    for d in range(n):

        v = h[d:]

            # Jacobian for A_d
