import numpy as np
import math
from scipy.optimize import minimize

def capped_simplex_project(v):
    n = len(v)
    s = n / 2.0
    v = np.asarray(v, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return None
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(80):
        mid = (lo + hi) / 2.0
        total = math.fsum(np.clip(v - mid, 0.0, 1.0))
        if total > s:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(200):
        r = s - math.fsum(x)
        if abs(r) <= 5e-15:
            break
        ok = False
        if r > 0:
            for i in range(n):
                if abs(r) <= 5e-15:
                    ok = True
                    break
                amt = min(r, 1.0 - x[i])
                if amt > 1e-16:
                    x[i] += amt
                    r -= amt
        else:
            for i in range(n):
                if abs(r) <= 5e-15:
                    ok = True
                    break
                amt = min(-r, x[i])
                if amt > 1e-16:
                    x[i] -= amt
                    r += amt
        if ok:
            break
    if x.min() < -1e-12 or x.max() > 1.0 + 1e-12:
        return None
    if abs(math.fsum(x) - s) > 1e-9:
        return None
    return x

def check_feasible(h, n):
    if len(h) != n or not np.all(np.isfinite(h)):
        return False
    if h.min() < -1e-12 or h.max() > 1.0 + 1e-12:
        return False
    if abs(math.fsum(h) - n / 2.0) > 1e-9:
        return False
    return True

def exact_score(h):
    n = len(h)
    if n == 0 or not np.all(np.isfinite(h)):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode="full")
    val = float(np.max(c)) * (2.0 / n)
    return val if math.isfinite(val) else float('inf')

def lags_and_jac(h):
    n = len(h)
    dx = 2.0 / n
    q = np.zeros(2 * n - 1)
    J = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        row = d + n - 1
        if d >= 0:
            src = np.arange(0, n - d)
            tgt = src + d
        else:
            src = np.arange(-d, n)
            tgt = src + d
        q[row] = math.fsum(h[src] * (1.0 - h[tgt]))
        J[row, src] += (1.0 - h[tgt])
        J[row, tgt] -= h[src]
    return q * dx, J * dx

def epig_obj(z):
    return z[-1]

def epig_obj_jac(z):
    n = len(z) - 1
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g

def eq_fun(z, n):
    return math.fsum(z[:n]) - n / 2.0

def eq_fun_jac(z, n):
    g = np.zeros(len(z))
    g[:n] = 1.0
    return g

def ineq_fun(z, n):
    h = z[:n]
    t = z[-1]
    q, _ = lags_and_jac(h)
    return t - q

def ineq_fun_jac(z, n):
    h = z[:n]
    _, J = lags_and_jac(h)
    m = 2 * n - 1
    out = np.zeros((m, n + 1))
    out[:, :n] = -J
    out[:, n] = 1.0
    return out

def solve_start(h0, n, maxiter=300):
    dx = 2.0 / n
    h0 = capped_simplex_project(h0)
    if h0 is None:
        return None
    t0 = min(1.0, exact_score(h0) + 1e-9)
    z0 = np.append(h0, t0)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_fun_jac, 'args': (n,)},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_fun_jac, 'args': (n,)}
    ]
    try:
        res = minimize(epig_obj, z0, jac=epig_obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'ftol': 1e-10, 'maxiter': maxiter, 'disp': False})
        if res.x is not None and len(res.x) == n + 1:

            h_res = np.abs(res.x[:n])
            h_res = capped_simplex_project(h_res)
