    for _ in range(80):
        mid = lo + (hi - lo) * 0.5
        if abs(hi - lo) < 1e-15:
            break
        x = np.clip(v - mid, 0.0, 1.0)
        if np.sum(x) > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - mid, 0.0, 1.0)
    r = S - np.sum(x)
    for _ in range(3):
        if abs(r) < 1e-12:
            break
        if r > 0:
            cap = 1.0 - x
            if np.sum(cap) < 1e-15:
                break
            x += np.minimum(r, cap)
        else:
            cap = x
            if np.sum(cap) < 1e-15:
                break
            x += np.maximum(r, -cap)
        r = S - np.sum(x)
    x = np.clip(x, 0.0, 1.0)
    if np.all(x >= 0) and np.all(x <= 1) and abs(np.sum(x) - S) <= 1e-10:
        return x
    return None


def check_feasible(h, n):
    if h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < 0) or np.any(h > 1):
        return False
    if abs(np.sum(h) - n * 0.5) > 1e-10:
        return False
    return True


def exact_score(h, n):
    if not check_feasible(h, n):
        return None
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(np.max(corr)) * (2.0 / n)
    if not np.isfinite(c5):
        return None
    return c5


def smooth_eval(h, tau, dx, n, with_gradient):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    cmax = np.max(c)
    u = (c - cmax) / tau
    exp_u = np.exp(u)
    s = np.sum(exp_u)
    if s <= 0 or not np.isfinite(s):
        return None, None
    F = cmax + tau * np.log(s)
    if not np.isfinite(F):
        return None, None
    w = exp_u / s
    if not with_gradient:
        return F, None
    g = np.zeros(n, dtype=np.float64)
    g += w[n - 1] * (1.0 - 2.0 * h)
    for sl in range(1, n):
        m = n - sl
        wp = w[n - 1 + sl]
        if wp > 0:
            g[sl:] += wp * (1.0 - h[:m])
            g[:m] -= wp * h[sl:]
        wn = w[n - 1 - sl]
        if wn > 0:
            g[:m] += wn * (1.0 - h[sl:])
            g[sl:] -= wn * h[:m]
    g *= dx
    if not np.all(np.isfinite(g)):
        return None, None
    return F, g


def local_search(x, n, dx, S, iter_cap, tau_schedule):
    x = x.copy()
    best_x = x.copy()
    best_score = exact_score(x, n)
    if best_score is None:
        return best_x, best_score
    for tau in tau_schedule:
        alpha = 1.0
        for _ in range(iter_cap):
            F, g = smooth_eval(x, tau, dx, n, True)
            if F is None or g is None:
                break
            accepted = False
            d = None
            for _ in range(20):
                y = project_capped(x - alpha * g, S)
                if y is None:
                    break
                if not check_feasible(y, n):
                    break
                d = y - x
                Fy, _ = smooth_eval(y, tau, dx, n, False)
                if Fy is None:
                    break
                cond = np.dot(g, d) + np.dot(d, d) / (2.0 * alpha) + 1e-12 * (1.0 + abs(F))
                if Fy <= F + cond:
                    accepted = True
                    break
                alpha *= 0.5
            if not accepted:
                break
            x = x + d
            sc = exact_score(x, n)
            if sc is not None and (best_score is None or sc < best_score - 1e-12):
                best_score = sc
                best_x = x.copy()
            if np.linalg.norm(d) < 1e-14:
                break
            if np.linalg.norm(d) / alpha <= 1e-7:
                break
            alpha = min(1.5 * alpha, 4.0)
    return best_x, best_score


def run(seed=42, **kwargs):
    n = 45
    parent = None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64)
        if iv.ndim == 1 and len(iv) >= 2:
            n = len(iv)
            if np.all(np.isfinite(iv)):
                parent = iv.copy()
    except Exception:
        pass
    S = n * 0.5
    dx = 2.0 / n
    fallback = np.full(n, 0.5, dtype=np.float64)
    first_candidate = None
    if parent is not None:
        if check_feasible(parent, n):
            first_candidate = parent.copy()
        else:
            rep = project_capped(parent, S)
            if rep is not None and check_feasible(rep, n):
                first_candidate = rep
    if first_candidate is None:
        first_candidate = fallback.copy()
    fallback = first_candidate.copy()
    fallback_score = exact_score(fallback, n)
    rng = np.random.default_rng(seed)
    starts = [first_candidate.copy(), fallback.copy()]
    base = first_candidate
    if n <= 128:
        gscales = [0.04, 0.12, 0.30]
    elif n <= 512:
        gscales = [0.08, 0.25]
    else:
        gscales = [0.12]
    for gs in gscales:
        prop = base + rng.standard_normal(n) * gs
        proj = project_capped(prop, S)
        if proj is not None:
            starts.append(proj)
    uv = rng.uniform(0.0, 1.0, size=n)
    proj = project_capped(uv, S)
    if proj is not None:
        starts.append(proj)
    if n <= 128:
        ms = [1, 2]
    else:
        ms = [1]
    iarr = np.arange(n, dtype=np.float64)
    for mv in ms:
        ph = rng.uniform(0.0, 2.0 * np.pi)
        prop = 0.5 + 0.35 * np.sin(2.0 * np.pi * mv * iarr / n + ph)
        pj = project_capped(prop, S)
        if pj is not None:
            starts.append(pj)
    deduped = []
    for st in starts:
        if not check_feasible(st, n):
            continue
        keep = True
        for dd in deduped:
            if np.linalg.norm(st - dd, np.inf) < 1e-12:
                keep = False
                break
        if keep:
            deduped.append(st)
    iter_cap = max(15, min(80, 3600 // n))
    tau_schedule = [0.02, 0.005, 0.001, 0.0003, 0.0001]
    incumbent = fallback.copy()
    incumbent_score = fallback_score
    for st in deduped:
        try:
            bx, bs = local_search(st, n, dx, S, iter_cap, tau_schedule)
            if bs is not None and (incumbent_score is None or bs < incumbent_score - 1e-12):
                incumbent = bx.copy()
                incumbent_score = bs
        except Exception:
            continue
    if n <= 128:
        rscales = [0.05, 0.015]
    elif n <= 512:
        rscales = [0.04, 0.015]
    else:
        rscales = [0.03]
    for rs in rscales:
        for _ in range(2):
            try:
                prop = incumbent + rng.standard_normal(n) * rs
                proj = project_capped(prop, S)
                if proj is None:
                    continue
                bx, bs = local_search(proj, n, dx, S, iter_cap, tau_schedule)
                if bs is not None and bs < incumbent_score - 1e-12:
                    incumbent = bx.copy()
                    incumbent_score = bs
            except Exception:
                continue
    fs = exact_score(incumbent, n)
    if fs is None:
        rep = project_capped(incumbent, S)
        if rep is not None:
            incumbent = rep
            fs = exact_score(incumbent, n)
    if fs is None:
        incumbent = fallback.copy()
        fs = exact_score(fallback, n)
    h_out = incumbent.astype(np.float64).copy()
    n_out = len(h_out)
    corr = np.correlate(h_out, 1.0 - h_out, mode='full')
    c5 = float(np.max(corr)) * (2.0 / n_out)
    return h_out, c5, n_out
