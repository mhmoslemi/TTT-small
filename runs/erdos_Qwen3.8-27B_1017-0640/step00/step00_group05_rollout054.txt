def project_box_sum(v, m):
    v = np.clip(np.asarray(v, dtype=np.float64), -1e6, 1e6)
    lo, hi = v.min() - 1.0, v.max()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > m:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(3):
        r = m - np.sum(y)
        if abs(r) < 1e-14:
            break
        if r > 0:
            for i in range(len(y)):
                if r <= 0:
                    break
                cap = 1.0 - y[i]
                d = min(r, cap)
                y[i] += d
                r -= d
        else:
            for i in range(len(y)):
                if r >= 0:
                    break
                cap = y[i]
                d = max(r, -cap)
                y[i] += d
                r -= d
    if not np.all(np.isfinite(y)) or np.any(y < -1e-12) or np.any(y > 1 + 1e-12) or abs(np.sum(y) - m) > 1e-9:
        return 0.5 * np.ones(len(v))
    y = np.clip(y, 0.0, 1.0)
    return y

def score_c5(h, N):
    dx = 2.0 / N
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(q))

def make_derivative(z, N):
    h = z[:N]
    dx = 2.0 / N
    # Returns (2N-1, N) matrix: row j corresponds to lag k = j - (N-1)
    J = np.zeros((2 * N - 1, N), dtype=np.float64)
    for j in range(2 * N - 1):
        k = j - (N - 1)  # lag
        # For column r: dx * [(1-h[r-k]) if 0<=r-k<N] - dx * [h[r+k] if 0<=r+k<N]
        # r-k in [0,N) means r in [k, k+N)
        # r+k in [0,N) means r in [-k, N-k)
        for r in range(N):
            val = 0.0
            if 0 <= r - k < N:
                val += (1.0 - h[r - k])
            if 0 <= r + k < N:
                val -= h[r + k]
            J[j, r] = dx * val
    return J

def obj_func(z, N):
    return z[N]  # t

def obj_grad(z, N):
    g = np.zeros(N + 1)
    g[N] = 1.0
    return g

def eq_fun(z, N, m):
    return np.sum(z[:N]) - m

def eq_jac(z, N, m):
    g = np.ones(N + 1)
    g[N] = 0.0
    return g

def ineq_fun(z, N):
    h = z[:N]
    t = z[N]
    q = np.correlate(h, 1.0 - h, mode='full') * (2.0 / N)
    return t - q  # should be >= 0

def ineq_jac(z, N):

    # Jacobian of (t - q_j) w.r.t. z = [-J_j, 1] for each j
    # Shape: (2N-1, N+1)
    H = np.zeros((2 * N - 1, N + 1), dtype=np.float64)
    H[:, :N] = -J
    H[:, N] = 1.0
