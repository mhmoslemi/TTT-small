import numpy as np
from scipy.optimize import minimize


def project_box_sum(v, m, N):
    v = np.clip(np.asarray(v, dtype=np.float64), -1e6, 1e6)
    lo, hi = -1e6, 1e6
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > m:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    y = np.clip(v - lam, 0.0, 1.0)
    for _ in range(3):
        r = m - y.sum()
        if abs(r) < 1e-12:
            break
        if r > 0:
            for i in range(N):
                if y[i] < 1.0:
                    adj = min(r, 1.0 - y[i])
                    y[i] += adj
                    r -= adj
                    if r < 1e-12:
                        break
        else:
            r = -r
            for i in range(N):
                if y[i] > 0.0:
                    adj = min(r, y[i])
                    y[i] -= adj
                    r -= adj
                    if r < 1e-12:
                        break
    if not np.all(np.isfinite(y)):
        return np.full(N, 0.5)
    y = np.clip(y, 0.0, 1.0)
    return y


def score_h(h, N):
    dx = 2.0 / N
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(q))


def build_grad(h, N, dx):
    J = np.zeros((2 * N - 1, N))
    for j in range(2 * N - 1):
        k = j - N + 1
        r1s = max(0, -k)
        r1e = min(N, N - k)
        for r in range(r1s, r1e):
            J[j, r] += (1.0 - h[r + k])
        r2s = max(0, k)
        r2e = min(N, N + k)
        for r in range(r2s, r2e):
            J[j, r] -= h[r - k]
    return J * dx


def _obj(z, args):
    return z[-1]


def _obj_jac(z, args):
    N = args
    g = np.zeros(N + 1)
    g[-1] = 1.0
    return g


def _eq_fun(z, args):
    N, m = args
    return np.array([np.sum(z[:N]) - m])


def _eq_jac(z, args):
    N = args
    J = np.zeros((1, N + 1))
    J[0, :N] = 1.0
    return J


def _ineq_fun(z, args):
    N, dx = args
    h = z[:N]
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    return z[-1] - q


def _ineq_jac(z, args):
    N, dx = args
    h = z[:N]
    J = build_grad(h, N, dx)
    return np.hstack([-J, np.ones((2 * N - 1, 1))])


def run(seed=42, **kwargs):
    N = 70
    dx = 2.0 / N
    m = N / 2.0
    h_parent = None

    try:
        hv = np.asarray(initial_h_values, dtype=np.float64)
        if hv.ndim == 1 and hv.size > 0 and np.all(np.isfinite(hv)):
            N = hv.size
            dx = 2.0 / N
            m = N / 2.0
            h_parent = hv.copy()
    except Exception:
        pass

    if N <= 0:
        N = 70
        dx = 2.0 / N
        m = N / 2.0

    rng = np.random.default_rng(seed)
    h_uniform = np.full(N, 0.5)

    if h_parent is not None:
        base = project_box_sum(h_parent, m, N)
    else:
        base = h_uniform.copy()

    best_h = base.copy()
    best_score = score_h(base, N)

    us = score_h(h_uniform, N)
    if us < best_score:
        best_score = us
        best_h = h_uniform.copy()

    starts = [base.copy(), h_uniform.copy()]

    if N <= 256:
        for scale in [0.03, 0.08, 0.18]:
            starts.append(project_box_sum(base + scale * rng.standard_normal(N), m, N))
        starts.append(project_box_sum(rng.uniform(0, 1, N), m, N))
        starts.append(project_box_sum(rng.beta(0.7, 0.7, N), m, N))
        x = np.arange(N, dtype=np.float64)
        for p, phi in [(1, 0.0), (2, np.pi / 3.0)]:
            sm = 0.5 + 0.45 * np.cos(2.0 * np.pi * p * (x + 0.5) / N + phi)
            starts.append(project_box_sum(sm, m, N))
    else:
        starts.append(project_box_sum(base + 0.08 * rng.standard_normal(N), m, N))
        starts.append(project_box_sum(rng.uniform(0, 1, N), m, N))
        starts.append(project_box_sum(rng.beta(0.7, 0.7, N), m, N))

    for s in starts:
        sc = score_h(s, N)
        if sc < best_score - 1e-12:
            best_score = sc
            best_h = s.copy()

    if N <= 999:
        args_eq = (N, m)
        args_ineq = (N, dx)
        args_obj = N
        bounds = [(0.0, 1.0)] * N + [(0.0, 2.0)]
        constraints = [
            {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': args_eq},
            {'type': 'ineq', 'fun': _ineq_fun, 'jac': _ineq_jac, 'args': args_ineq},
        ]

        for s in starts:
            t0 = max(score_h(s, N), 0.0)
            t0 = min(t0, 2.0)
            z0 = np.concatenate([s, [t0]])
            try:
                res = minimize(_obj, z0, jac=_obj_jac, args=args_obj,
                               method='SLSQP', bounds=bounds,
                               constraints=constraints,
                               options={'maxiter': 250, 'ftol': 1e-10, 'disp': False})
                if res.x is not None and res.x.size == N + 1:
                    if np.all(np.isfinite(res.x)):
                        hc = project_box_sum(res.x[:N], m, N)
                        sc = score_h(hc, N)
                        if sc < best_score - 1e-12:
                            best_score = sc
                            best_h = hc.copy()
            except Exception:
                continue

    final_h = project_box_sum(best_h, m, N)
    if (not np.all(np.isfinite(final_h))
            or np.any(final_h < -1e-10) or np.any(final_h > 1.0 + 1e-10)):
        final_h = h_uniform.copy()
    final_score = score_h(final_h, N)
    return (final_h, final_score, N)
