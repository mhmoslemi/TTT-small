import numpy as np

def project_capped_simplex(y, n):
    """Project y onto {0<=x<=1, sum(x)=n/2}."""
    target = n / 2.0
    low = np.min(y) - 1.0
    high = np.max(y)
    for _ in range(80):
        mid = (low + high) / 2.0
        mass = np.sum(np.clip(y - mid, 0.0, 1.0))
        if mass > target:
            low = mid
        else:
            high = mid
    lam = (low + high) / 2.0
    x = np.clip(y - lam, 0.0, 1.0)
    # Residual correction
    for _ in range(3):
        r = target - np.sum(x)
        if abs(r) < 5e-13 * max(1, n):
            break
        if r > 0:
            for i in range(n):
                if r <= 0:
                    break
                add = min(r, 1.0 - x[i])
                x[i] += add
                r -= add
        else:
            for i in range(n):
                if r >= 0:
                    break
                sub = min(-r, x[i])
                x[i] -= sub
                r += sub
    if not np.all(np.isfinite(x)) or np.any(x < -1e-12) or np.any(x > 1 + 1e-12):
