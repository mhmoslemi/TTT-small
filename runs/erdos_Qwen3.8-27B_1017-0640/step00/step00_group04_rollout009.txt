import numpy as np
from scipy.optimize import minimize


def project_feasible(y):
    n = len(y)
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(y - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    res = target - x.sum()
    if abs(res) > 1e-12:
        order = np.argsort(1.0 - x) if res > 0 else np.argsort(x)
        for j in order:
            if abs(res) < 1e-12:
                break
            if res > 0:
                d = min(1.0 - x[j], res)
                x[j] += d
                res -= d
            else:
                d = min(x[j], -res)
                x[j] -= d
                res += d
    return x


def exact_score(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)


def _obj(z):
    return z[-1]


def _obj_grad(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g


def _eq(z):
    return z[:-1].sum() - len(z[:-1]) / 2.0


def _eq_jac(z):
    n = len(z) - 1
    J = np.zeros((1, n + 1))
    J[0, :n] = 1.0
    return J


def _ineq(z):
    h = z[:-1]
    t = z[-1]
    n = len(h)
    return t - (2.0 / n) * np.correlate(h, 1.0 - h, mode='full')


def _ineq_jac(z):
    h = z[:-1]
    n = len(h)
    dx = 2.0 / n
    n_ell = 2 * n - 1
    J = np.zeros((n_ell, n + 1))
    for k in range(n_ell):
        ell = k - (n - 1)
        row = np.zeros(n)
        if ell >= 0:
            for i in range(ell, n):
                row[i] += 1.0 - h[i - ell]
                row[i - ell] -= h[i]
        else:
            e = -ell
            for i in range(0, n - e):
                row[i] += 1.0 - h[i + e]
                row[i + e] -= h[i]
        J[k, :n] = -dx * row
        J[k, n] = 1.0
    return J


def build_starts(n, anchor, rng, count):
    hb = np.zeros(n)
    hb[:n // 2] = 1.0
    if n % 2 == 1:
        hb[n // 2] = 0.5
    cands = []
    if anchor is not None:
        cands.append(anchor.copy())
        cands.append(anchor[::-1].copy())
        cands.append(1.0 - anchor.copy())
    cands.append(hb.copy())
    cands.append(np.full(n, 0.5))
    proj = []
    for s in cands:
        s = project_feasible(s)
        dup = False
        for p in proj:
            if np.max(np.abs(s - p)) < 1e-10:
                dup = True
                break
        if not dup:
            proj.append(s)
    scales = [0.08, 0.20, 0.40]
    while len(proj) < count:
        if len(proj) % 2 == 0:
            base = anchor if anchor is not None else hb
            sc = scales[(len(proj) - 2) % 3]
            s = base + rng.normal(0.0, sc, size=n)
        else:
            s = rng.uniform(-0.25, 1.25, size=n)
        s = project_feasible(s)
        proj.append(s)
    return proj


def solve_epigraph(h0, n, maxiter):
    dx = 2.0 / n
    s0 = exact_score(h0)
    t0 = min(1.0, s0 + 1e-8)
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * (n + 1)
    cons = [
        {'type': 'eq', 'fun': _eq, 'jac': _eq_jac},
        {'type': 'ineq', 'fun': _ineq, 'jac': _ineq_jac}
    ]
    try:
        res = minimize(_obj, z0, method='SLSQP', jac=_obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-10, 'disp': False})
        if res.x is not None and len(res.x) == n + 1 and np.all(np.isfinite(res.x)):
            hc = project_feasible(res.x[:n])
            if np.all(hc >= -1e-9) and np.all(hc <= 1.0 + 1e-9):
                return hc
    except Exception:
        pass
    return None


def run(seed=42, **kwargs):
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    rng = np.random.default_rng(seed)
    n_starts = 8
    maxiter = 250
    if 'n_starts' in kwargs:
        try:
            n_starts = max(4, min(10, int(kwargs['n_starts'])))
        except Exception:
            pass
    if 'maxiter' in kwargs:
        try:
            maxiter = max(100, min(400, int(kwargs['maxiter'])))
        except Exception:
            pass

    parent = None
    if 'initial_h_values' in globals():
        p = np.array(globals()['initial_h_values'], dtype=float).flatten()
        if p.size >= 2 and np.all(np.isfinite(p)):
            parent = p.copy()

    if parent is not None:
        n0 = len(parent)
    else:
        n0 = 42

    fallback = project_feasible(parent if parent is not None else np.concatenate([np.ones(n0//2), np.zeros(n0 - n0//2)]))
    best_h = fallback.copy()
    best_score = exact_score(best_h)
    best_n = len(best_h)

    def _try_improve(hcand):
        nonlocal best_h, best_score, best_n
        if hcand is not None:
            s = exact_score(hcand)
            if s < best_score - 1e-12:
                best_h = hcand.copy()
                best_score = s
                best_n = len(hcand)
            elif abs(s - best_score) <= 1e-12 and len(hcand) < best_n:
                best_h = hcand.copy()
                best_score = s
                best_n = len(hcand)

    if n0 <= 256:
        anchor0 = project_feasible(parent) if parent is not None else fallback
        starts0 = build_starts(n0, anchor0, rng, n_starts)
        for s in starts0:
            _try_improve(s)
        for s in starts0:
            hc = solve_epigraph(s, n0, maxiter)
            _try_improve(hc)

    # Double resolution
    if n0 <= 128:
        n1 = 2 * n0
        if parent is not None:
            anchor1 = np.repeat(parent, 2)
        else:
            anchor1 = np.repeat(best_h[:n0], 2) if best_n == n0 else np.repeat(fallback, 2)
        anchor1 = project_feasible(anchor1)
        starts1 = build_starts(n1, anchor1, rng, n_starts)
        for s in starts1:
            _try_improve(s)
        for s in starts1:
            hc = solve_epigraph(s, n1, maxiter)
            _try_improve(hc)

    # Final validation
    if not np.all(np.isfinite(best_h)):
        best_h = fallback.copy()
    best_h = project_feasible(best_h)
    final_score = exact_score(best_h)
    if final_score > best_score + 1e-9:
        best_h = fallback.copy()
        final_score = exact_score(best_h)
    return (best_h, final_score, int(len(best_h)))
