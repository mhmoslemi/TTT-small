import numpy as np
from scipy.optimize import minimize
from functools import partial
import math

def exact_score(h):
    n = len(h)
    if n == 0 or not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_feasible(y, n):
    m = n / 2.0
    y = np.asarray(y, dtype=np.float64)
    if len(y) != n or not np.all(np.isfinite(y)):
        return np.full(n, 0.5)
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y))
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(y - mid, 0.0, 1.0)) > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    res = m - float(np.sum(h))
    for _ in range(n + 2):
        if abs(res) < 2e-14 * max(1, n):
            break
        if res > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] += min(res, 1.0 - h[idx])
        else:
            idx = int(np.argmin(h))
            h[idx] -= min(-res, h[idx])
        res = m - float(np.sum(h))
    h = np.clip(h, 0.0, 1.0)
    if abs(float(np.sum(h)) - m) > 1e-10 * max(1, n):
        return np.full(n, 0.5)
    return h

def compute_q_grad(h, n, dx):
    q = np.zeros(2 * n - 1)
    G = np.zeros((2 * n - 1, n))
    q[n - 1] = dx * float(np.dot(h, 1.0 - h))
    G[n - 1] = dx * (1.0 - 2.0 * h)
    for s in range(1, n):
        ns = n - s
        q[n - 1 + s] = dx * float(np.dot(h[s:], 1.0 - h[:ns]))
        G[n - 1 + s, s:] = dx * (1.0 - h[:ns])
        G[n - 1 + s, :ns] -= dx * h[s:]
        q[n - 1 - s] = dx * float(np.dot(h[:ns], 1.0 - h[s:]))
        G[n - 1 - s, :ns] = dx * (1.0 - h[s:])
        G[n - 1 - s, s:] -= dx * h[:ns]
    return q, G

def _obj(z, n):
    return z[n]

def _obj_grad(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def _eq(z, n):
    return float(np.sum(z[:n]) - n / 2.0)

def _eq_grad(z, n):
    g = np.ones(n + 1)
    g[n] = 0.0
    return g

    h = z[:n]
    t = z[n]
    q, _ = compute_q_grad(h, n, 2.0 / n)

def _ineq_grad(z, n):
    h = z[:n]
    _, G = compute_q_grad(h, n, 2.0 / n)
    return np.hstack((-G, np.ones((2 * n - 1, 1))))
