```python
def corr_q_and_G(h, n):
    dx = 2.0 / n  # actually we don't need dx here, let's just return q and G
    q = np.zeros(2*n - 1)
    G = np.zeros((2*n - 1, n))
    
    # Zero lag
    q[n-1] = np.dot(h, 1.0 - h)
    G[n-1] = 1.0 - 2.0 * h
    
    # Positive and negative lags
    for s in range(1, n):
        m = n - s
        # q_{+s} = sum_{i=0}^{m-1} h[i+s] * (1-h[i])
        q[n-1+s] = np.dot(h[s:], 1.0 - h[:m])
        # q_{-s} = sum_{i=0}^{m-1} h[i] * (1-h[i+s])
        q[n-1-s] = np.dot(h[:m], 1.0 - h[s:])
        
        # Gradients
        # g^{+s}: 
        # For r in [s, n-1]: (1-h[r-s])
        # For r in [0, m-1]: -h[r+s]
        g_plus = np.zeros(n)
        g_plus[s:] = 1.0 - h[:m]
        g_plus[:m] -= h[s:]
        G[n-1+s] = g_plus
        
        # g^{-s}:
        # For r in [0, m-1]: (1-h[r+s])
        # For r in [s, n-1]: -h[r-s]
        g_minus = np.zeros(n)
        g_minus[:m] = 1.0 - h[s:]
        g_minus[s:] -= h[:m]
        G[n-1-s] = g_minus
    
    return q, G
```

```python
import numpy as np
from scipy.optimize import minimize

def project_capped(v, S):
    n = len(v)
    v = np.ascontiguousarray(v, dtype=np.float64)
    v = np.clip(v, -1e6, 1e6)
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    lam = 0.5 * (lo + hi)
    for _ in range(100):
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > S:
            lo = lam
        else:
            hi = lam
        new_lam = 0.5 * (lo + hi)
        if new_lam == lam:
            break
        lam = new_lam
    x = np.clip(v - lam, 0.0, 1.0)
    for _ in range(80):
        res = S - x.sum()
        if abs(res) < 1e-13:
            break
        if res > 0:
            mask = x < 1.0 - 1e-14
            if not mask.any():
                break
            x[mask] += res / mask.sum()
            np.minimum(x, 1.0, out=x)
        else:
            mask = x > 1e-14
            if not mask.any():
                break
            x[mask] += res / mask.sum()
            np.maximum(x, 0.0, out=x)
    x = np.clip(x, 0.0, 1.0)
    if not np.all(np.isfinite(x)) or abs(x.sum() - S) > 1e-9:
        return None
    return x

def corr_q_and_G(h, n):
    q = np.zeros(2 * n - 1, dtype=np.float64)
    G = np.zeros((2 * n - 1, n), dtype=np.float64)
    one_h = 1.0 - h
    q[n - 1] = np.dot(h, one_h)
    G[n - 1] = 1.0 - 2.0 * h
    for s in range(1, n):
        m = n - s
        q[n - 1 + s] = np.dot(h[s:], one_h[:m])
        q[n - 1 - s] = np.dot(h[:m], one_h[s:])
        gp = np.zeros(n, dtype=np.float64)
        gp[s:] = one_h[:m]
        gp[:m] -= h[s:]
        G[n - 1 + s] = gp
        gm = np.zeros(n, dtype=np.float64)
        gm[:m] = one_h[s:]
        gm[s:] -= h[:m]
        G[n - 1 - s] = gm
    return q, G

def exact_score(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def obj_fun(z, n, dx, S):
    return z[-1]

def obj_jac(z, n, dx, S):
    j = np.zeros(n + 1)
    j[-1] = 1.0
    return j

def ineq_fun(z, n, dx, S):
    h = z[:n]
    q, _ = corr_q_and_G(h, n)
    return z[-1] - dx * q

def ineq_jac(z, n, dx, S):
    h = z[:n]
    _, G = corr_q_and_G(h, n)
    njac = G.shape[0]
    J = np.zeros((njac, n + 1), dtype=np.float64)
    J[:, :n] = -dx * G
    J[:, n] = 1.0
    return J

def eq_fun(z, n, dx, S):
    return z[:n].sum() - S

def eq_jac(z, n, dx, S):
    J = np.zeros((1, n + 1))
    J[0, :n] = 1.0
    return J

def try_slsqp(h0, n, dx, S):
    sc = exact_score(h0, n)
    z0 = np.concatenate([h0, [min(max(sc, 0.0), 2.0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    args = (n, dx, S)
    cons = [
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac, 'args': args},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': args},
    ]
    try:
        res = minimize(obj_fun, z0, jac=obj_jac, args=args, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-10, 'disp': False})
    except Exception:
        return h0, sc
    if res is None or not hasattr(res, 'x') or res.x.shape[0] != n + 1:
        return h0, sc
    hx = res.x[:n]
    if not np.all(np.isfinite(hx)):
        return h0, sc
    hx = project_capped(hx, S)
    if hx is None:
        return h0, sc
    new_sc = exact_score(hx, n)
    if new_sc < sc:
        return hx, new_sc
    return h0, sc

def build_starts(parent, n, S, rng):
    starts = []
    if parent is not None:
        starts.append(parent.copy())
    const = np.full(n, 0.5, dtype=np.float64)
    starts.append(const)
    base = parent if parent is not None else const
    sigmas = [0.03, 0.08, 0.16, 0.30, 0.08, 0.20]
    for sig in sigmas:
        pert = base + rng.standard_normal(n) * sig
        p = project_capped(pert, S)
        if p is not None:
            starts.append(p)
    for _ in range(2):
        pert = rng.uniform(0.0, 1.0, n)
        p = project_capped(pert, S)
        if p is not None:
            starts.append(p)
    for m in [1, 2]:
        phase = rng.uniform(0.0, 2.0 * np.pi)
        i = np.arange(n, dtype=np.float64)
        pert = 0.5 + 0.35 * np.sin(2.0 * np.pi * m * i / n + phase)
        p = project_capped(pert, S)
        if p is not None:
            starts.append(p)
    return starts

def run(seed=42, **kwargs):

    S = n / 2.0

    parent = None
    iva = globals().get('initial_h_values', None)

        except Exception:

    if parent is None:

    best_h = parent.copy()
    best_sc = parent_score

    for sig in [0.08, 0.03, 0.01]:

                best_sc = ss
    fin = project_capped(best_h, S)
    if fin is None:

    else:
        best_h = fin
    final_sc = exact_score(best_h, n)
```
