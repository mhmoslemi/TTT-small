def calibrate_temperature(start, start_score, n, S, rng, pilot_count, p_perm, p_swap):
    best_pilot = None
    best_pilot_score = start_score
    deltas = []
    for _ in range(pilot_count):

        if prop is None:
            continue
        sc = exact_score(prop, n, S)
        if sc is None:
            continue
        if sc < best_pilot_score:
            best_pilot_score = sc
            best_pilot = prop.copy()
        d = sc - start_score
        if d > 1e-12:
            deltas.append(d)
    if deltas:
        m = float(np.median(deltas))
        T0 = max(1e-5, min(0.05, m / (-math.log(0.65))))
    else:
        T0 = 0.002

    cur_score = start_score
    bp, bps, T0 = calibrate_temperature(start, start_score, n, S, rng, pilot_count, p_perm, p_swap)
    if bp is not None and bps < incumbent_score - 1e-12:
        incumbent = bp.copy()
        incumbent_score = bps
    Tend = max(1e-7, 0.01 * T0)
    for l in range(L):

    hh = h.copy()
    hh[i] = xp
    hh[j] = c - xp
    if not feasible(hh, n, S):
        hh = project_capped(hh, S)
        if hh is None:
            return None
    return hh


def calibrate_temperature(start, start_score, n, S, rng, pilot_count, p_perm, p_swap):
    best_pilot = None
    best_pilot_score = start_score
    deltas = []
    for _ in range(pilot_count):
        prop = make_proposal(start, n, S, 0.65, rng, p_perm, p_swap)
        if prop is None:
            continue
        sc = exact_score(prop, n, S)
        if sc is None:
            continue
        if sc < best_pilot_score:
            best_pilot_score = sc
            best_pilot = prop.copy()
        d = sc - start_score
        if d > 1e-12:
            deltas.append(d)
    if deltas:
        m = float(np.median(deltas))
        T0 = max(1e-5, min(0.05, m / (-math.log(0.65))))
    else:
        T0 = 0.002
    return best_pilot, best_pilot_score, T0


def anneal_chain(start, start_score, n, S, rng, L, M, pilot_count, incumbent, incumbent_score, p_perm, p_swap):
    cur = start.copy()
    cur_score = start_score
    bp, bps, T0 = calibrate_temperature(start, start_score, n, S, rng, pilot_count, p_perm, p_swap)
    if bp is not None and bps < incumbent_score - 1e-12:
        incumbent = bp.copy()
        incumbent_score = bps
    Tend = max(1e-7, 0.01 * T0)
    for l in range(L):
        if L > 1:
            T = T0 * (Tend / T0) ** (l / (L - 1))
        else:
            T = T0
        rho = max(0.03, 0.65 * math.sqrt(T / T0))
        for _ in range(M):
            prop = make_proposal(cur, n, S, rho, rng, p_perm, p_swap)
            if prop is None:
                continue
            sc = exact_score(prop, n, S)
            if sc is None:
                continue
            if sc < incumbent_score - 1e-12:
                incumbent = prop.copy()
                incumbent_score = sc
            d = sc - cur_score
            if d <= 0:
                cur = prop
                cur_score = sc
            else:
                if d / T > 50.0:
                    continue
                if rng.random() < math.exp(-d / T):
                    cur = prop
                    cur_score = sc
    return incumbent, incumbent_score


def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)

    n = 45
    h0 = None
    try:
        raw = globals().get('initial_h_values', None)
        if raw is not None:
            arr = np.array(raw, dtype=np.float64)
            if arr.ndim == 1 and len(arr) >= 2:
                n = len(arr)
                if np.all(np.isfinite(arr)):
                    h0 = arr.copy()
    except Exception:
        pass

    S = n / 2.0
    base = None
    if h0 is not None:
        if np.all(h0 >= -1e-12) and np.all(h0 <= 1.0 + 1e-12) and abs(float(np.sum(h0)) - S) <= 1e-10:
            base = np.clip(h0, 0.0, 1.0).copy()
        else:
            base = project_capped(h0, S)
    if base is None:
        base = np.full(n, 0.5)
    fallback = base.copy()
    fallback_score = exact_score(fallback, n, S)
    if fallback_score is None:
        fallback = np.full(n, 0.5)
        fallback_score = exact_score(fallback, n, S)

    incumbent = fallback.copy()
    incumbent_score = fallback_score

    p_perm = 0.05
    p_swap = 0.15 if n <= 128 else 0.10

    starts = [base.copy()]
    comp = 1.0 - base
    if feasible(comp, n, S):
        starts.append(comp.copy())
    if n <= 128:
        starts.append(base[::-1].copy())
    starts.append(np.full(n, 0.5))
    if n <= 512:
        noise_std = 0.12 if n <= 128 else 0.10
        noisy = base + rng.normal(0.0, noise_std, size=n)
        proj = project_capped(noisy, S)
        if proj is not None:
            starts.append(proj)
    unif = rng.uniform(0.0, 1.0, size=n)
    proj = project_capped(unif, S)
    if proj is not None:
        starts.append(proj)
    if n <= 128:
        m = int(rng.integers(1, min(5, max(1, n // 2)) + 1))
        phase = rng.uniform(0.0, 2.0 * math.pi)
        i_vals = np.arange(n, dtype=np.float64)
        sinusoid = 0.5 + 0.35 * np.sin(2.0 * math.pi * m * i_vals / n + phase)
        proj = project_capped(sinusoid, S)
        if proj is not None:
            starts.append(proj)

    deduped = []
    for s in starts:
        dup = False
        for d in deduped:
            if np.max(np.abs(s - d)) < 1e-12:
                dup = True
                break
        if not dup:
            deduped.append(s)
    starts = deduped

    for s in starts:
        sc = exact_score(s, n, S)
        if sc is not None and sc < incumbent_score - 1e-12:
            incumbent = s.copy()
            incumbent_score = sc

    if n <= 64:
        L, R, M_cap = 12, 2, 600
    elif n <= 128:
        L, R, M_cap = 10, 1, 300
    elif n <= 512:
        L, R, M_cap = 8, 0, 120
    else:
        L, R, M_cap = 5, 0, 40
    Nchains = len(starts) + R
    M = max(1, min(M_cap, int(1.8e8 / (n * n * max(1, L * Nchains)))))
    pilot_count = max(8, min(48, int(1e6 / max(1, n * n))))

    for s in starts:
        sc = exact_score(s, n, S)
        if sc is None:
            continue
        try:
            incumbent, incumbent_score = anneal_chain(
                s, sc, n, S, rng, L, M, pilot_count,
                incumbent, incumbent_score, p_perm, p_swap
            )
        except Exception:
            pass

    if R > 0 and n <= 128:
        if n <= 64:
            stds = [0.10, 0.25]
        else:
            stds = [0.15]
        for std in stds[:R]:
            pert = incumbent + rng.normal(0.0, std, size=n)
            proj = project_capped(pert, S)
            if proj is None:
                continue
            sc = exact_score(proj, n, S)
            if sc is None:
                continue
            if sc < incumbent_score - 1e-12:
                incumbent = proj.copy()
                incumbent_score = sc
            try:
                incumbent, incumbent_score = anneal_chain(
                    proj, sc, n, S, rng, L, M, pilot_count,
                    incumbent, incumbent_score, p_perm, p_swap
                )
            except Exception:
                pass

    final = incumbent
    final_score = exact_score(final, n, S)
    if final_score is None or not feasible(final, n, S):
        final = fallback.copy()
        final_score = fallback_score

    return final.astype(np.float64), float(final_score), int(n)
