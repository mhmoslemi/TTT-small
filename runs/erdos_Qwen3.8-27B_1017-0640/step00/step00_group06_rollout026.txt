        if is_feasible(p, n, m):
            starts.append(p)
    except Exception:
        pass
    starts.append(fallback.copy())

    unique = [starts[0]]
    for s in starts[1:]:
        dup = False
        for u in unique:
            if np.max(np.abs(s - u)) < 1e-12:
                dup = True
                break
        if not dup and len(unique) < 8:
            unique.append(s)
    fb_found = any(np.max(np.abs(u - fallback)) < 1e-12 for u in unique)
    if not fb_found:
        unique.append(fallback.copy())

    B = max(8, min(5000, int(5000 * (95.0 / n) ** 2)))
    start_scores = []
    for s in unique:
        try:
            start_scores.append(exact_score(s))
        except Exception:
            start_scores.append(1.0)
    B -= len(start_scores)
    if B < 0:
        B = 0

    global_best = fallback.copy()
    global_best_sc = exact_score(fallback)
    for i, s in enumerate(unique):
        if start_scores[i] < global_best_sc - 1e-12:
            global_best_sc = start_scores[i]
            global_best = s.copy()

    ns = len(unique)
    per = max(1, B // ns) if ns > 0 else 0
    rem = B % ns if ns > 0 else 0
    for i in range(ns):
        bi = per + (1 if i < rem else 0)
        if bi <= 0:
            continue
        try:
            bh, bsc = one_start(unique[i], start_scores[i], rng, n, m, bi)
            if bsc < global_best_sc - 1e-12:
                global_best_sc = bsc
                global_best = bh
        except Exception:
            pass

    if not is_feasible(global_best, n, m):
        try:
            global_best = project_capped(global_best, n, m)
        except Exception:
            global_best = fallback.copy()
    if not is_feasible(global_best, n, m):
        global_best = fallback.copy()
    c5 = exact_score(global_best)
    return global_best, c5, n
