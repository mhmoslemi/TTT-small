    for j in range(2*N-1):
        k = j - (N-1)
        # term1: 1_h[r+k] valid if 0<=r+k<N
        # r+k in [0,N-1] => r in [-k, N-1-k]
        lo1 = max(0, -k)
        hi1 = min(N-1, N-1-k)

        # term2: -h[r-k] valid if 0<=r-k<N
        # r-k in [0,N-1] => r in [k, N-1+k]
        lo2 = max(0, k)
        hi2 = min(N-1, N-1+k)
        if lo2 <= hi2:

        hi2 = min(N - 1, N - 1 + k)
        if lo2 <= hi2:
            J[j, lo2:hi2 + 1] -= dx * h[lo2 - k:hi2 - k + 1]
    return J


def obj_fun(z, N, dx):
    return z[N]


def obj_jac(z, N, dx):
    g = np.zeros(z.shape)
    g[N] = 1.0
    return g


def eq_fun(z, N, m):
    return z[:N].sum() - m


def eq_jac(z, N, m):
    g = np.zeros(z.shape)
    g[:N] = 1.0
    return g


def ineq_fun(z, N, dx):
    h = z[:N]
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    return z[N] - q


def ineq_jac(z, N, dx):
    h = z[:N]
    J = compute_grad_matrix(h, N, dx)
    jac = np.empty((2 * N - 1, N + 1))
    jac[:, :N] = -J
    jac[:, N] = 1.0
    return jac


def make_starts(N, m, dx, rng, parent):
    starts = []
    base = parent.copy() if parent is not None else np.full(N, 0.5)
    starts.append(base.copy())
    if N <= 256:
        for scale in (0.03, 0.08, 0.18):
            starts.append(project_box_sum(base + scale * rng.standard_normal(N), m))
        starts.append(project_box_sum(rng.random(N), m))
        starts.append(project_box_sum(rng.beta(0.7, 0.7, N), m))
        i = np.arange(N, dtype=np.float64)
        v1 = 0.5 + 0.45 * np.cos(2.0 * np.pi * (i + 0.5) / N)
        v2 = 0.5 + 0.45 * np.cos(4.0 * np.pi * (i + 0.5) / N + np.pi / 3.0)
        starts.append(project_box_sum(v1, m))
        starts.append(project_box_sum(v2, m))
    else:
        starts.append(project_box_sum(base + 0.08 * rng.standard_normal(N), m))
        starts.append(project_box_sum(rng.random(N), m))
        starts.append(project_box_sum(rng.beta(0.7, 0.7, N), m))
    return starts


def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = None
    try:
        pv = np.asarray(initial_h_values, dtype=np.float64)
        if pv.ndim == 1 and len(pv) > 0 and np.all(np.isfinite(pv)):
            parent = pv
    except (NameError, Exception):
        pass
    N = int(parent.size) if parent is not None else 70
    if N <= 0:
        N = 70
    dx = 2.0 / N
    m = float(N) / 2.0
    h_fb = np.full(N, 0.5)
    best = h_fb.copy()
    best_score = score_c5(h_fb)
    if parent is not None:
        p_chk = parent.copy()
        if p_chk.min() >= -1e-10 and p_chk.max() <= 1.0 + 1e-10 and abs(p_chk.sum() - m) < 1e-10:
            p_chk = np.clip(p_chk, 0.0, 1.0)
        else:
            p_chk = project_box_sum(p_chk, m)
        s = score_c5(p_chk)
        if s < best_score - 1e-12:
            best = p_chk.copy()
            best_score = s
    if N > 999:
        return best, best_score, N
    starts = make_starts(N, m, dx, rng, parent)
    for h0 in starts:
        s = score_c5(h0)
        if s < best_score - 1e-12:
            best = h0.copy()
            best_score = s
    bounds = [(0.0, 1.0)] * N + [(0.0, 2.0)]
    c_eq = {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (N, m)}
    c_ieq = {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac, 'args': (N, dx)}
    constraints = [c_eq, c_ieq]
    for h0 in starts:
        q0 = np.correlate(h0, 1.0 - h0, mode='full') * dx
        t0 = float(np.clip(q0.max(), 0.0, 2.0))
        z0 = np.empty(N + 1)
        z0[:N] = h0
        z0[N] = t0
        try:
            res = minimize(
                fun=obj_fun, x0=z0, jac=obj_jac, args=(N, dx),
                bounds=bounds, constraints=constraints,
                method='SLSQP',
                options={'maxiter': 250, 'ftol': 1e-10, 'disp': False}
            )
            if res.x is not None and len(res.x) == N + 1:
                h_res = np.asarray(res.x[:N], dtype=np.float64)
                if np.all(np.isfinite(h_res)):
                    h_res = project_box_sum(h_res, m)
                    s = score_c5(h_res)
                    if s < best_score - 1e-12:
                        best = h_res.copy()
                        best_score = s
        except Exception:
            continue
    best = project_box_sum(best, m)
    if best.min() < -1e-10 or best.max() > 1.0 + 1e-10 or abs(best.sum() - m) > 1e-8:
        best = h_fb.copy()
    best_score = score_c5(best)
    return best, best_score, N
