import numpy as np

def project_capped(y, n):
    target = n / 2.0
    y = np.asarray(y, dtype=np.float64)
    lo = float(y.min()) - 1.0
    hi = float(y.max())
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    r = target - x.sum()
    for _ in range(3):
        if abs(r) < 5e-13 * max(1, n):
            break
        if r > 0:
            for i in range(n):
                if r <= 0: break
                a = min(r, 1.0 - x[i]); x[i] += a; r -= a
        else:
            for i in range(n):
                if r >= 0: break
                a = min(-r, x[i]); x[i] -= a; r += a
    if not np.all(np.isfinite(x)) or x.min() < -1e-12 or x.max() > 1.0 + 1e-12:
        return None
    if abs(x.sum() - target) > 1e-9 * max(1, n):
        return None
    return x

def exact_score(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(c)):
        return None
    return float(c.max() * (2.0 / n))

def feasible_interval(h, d):
    tlo, thi = 0.0, 0.0
    p = d > 1e-15
    m = d < -1e-15
    if np.any(p):
        tlo = max(tlo, float(np.min(-h[p] / d[p])))
        thi = min(thi, float(np.min((1.0 - h[p]) / d[p])))
    if np.any(m):
        tlo = max(tlo, float(np.min((1.0 - h[m]) / d[m])))
        thi = min(thi, float(np.min(-h[m] / d[m])))
    if not (np.isfinite(tlo) and np.isfinite(thi)) or thi - tlo < 1e-14:
        return None
    return tlo, thi

def do_line_search(h, d, q0, n):
    dx = 2.0 / n
    b = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    ac = -np.correlate(d, d, mode='full')
    if not (np.all(np.isfinite(b)) and np.all(np.isfinite(ac))):
        return None, 0.0
    iv = feasible_interval(h, d)
    if iv is None:
        return None, 0.0
    tlo, thi = iv
    cur = float(q0.max()) * dx
    ts = np.linspace(tlo, thi, 17)
    mat = q0[None, :] + ts[:, None] * b[None, :] + (ts * ts)[:, None] * ac[None, :]
    gv = dx * mat.max(axis=1)
    all_t = np.append(ts, 0.0)
    all_v = np.append(gv, cur)
    bi = int(np.argmin(all_v))
    bt, bv = float(all_t[bi]), float(all_v[bi])
    if bv >= cur - 1e-12:
        return None, 0.0
    gl = float(all_t[bi - 1]) if bi > 0 else tlo
    gr = float(all_t[bi + 1]) if bi < len(all_t) - 1 else thi
    if gr - gl > 1e-14:
        r = (math.sqrt(5.0) - 1.0) / 2.0
        c1, c2 = 1.0 - r, r
        x1 = gl + c1 * (gr - gl)
        x2 = gl + c2 * (gr - gl)
        f1 = dx * float((q0 + x1 * b + x1 * x1 * ac).max())
        f2 = dx * float((q0 + x2 * b + x2 * x2 * ac).max())
        for _ in range(14):
