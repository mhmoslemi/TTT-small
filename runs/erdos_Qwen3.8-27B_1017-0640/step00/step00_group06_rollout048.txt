import numpy as np
import math

def feasible(h, n, m):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1+1e-12):
        return False
    return abs(math.fsum(h) - m) <= 1e-10

def exact_score(h, n):
    q = np.correlate(h, 1.0-h, mode="full")
    if not np.all(np.isfinite(q)):
        return None
    return float(np.max(q) * 2.0 / n)

def project_capped(v, n, m):
    if not np.all(np.isfinite(v)):
        return None
    if np.max(np.abs(v)) > 1e12:
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = math.fsum(x)
        if s > m:
            lo = mid
        else:
            hi = mid
        if abs(s - m) <= 1e-13 or hi - lo < 1e-15:
            break
    x = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    r = m - math.fsum(x)
    for _ in range(3):
        idx = np.argsort(x) if r > 0 else np.argsort(-x)
        for i in idx:
            if r > 0:
                add = min(r, 1.0 - x[i])
                x[i] += add
                r -= add
            else:
                sub = min(-r, x[i])
                x[i] -= sub
                r += sub
            if abs(r) < 1e-14:
                break
        if abs(r) < 1e-14:
            break
    if not np.all(np.isfinite(x)):
        return None
    if not feasible(x, n, m):
        return None
    return x

def line_minimize(h, i, j, q, n):
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-15:
        return 0.0
    
    lag_offset = n - 1
    special_idx = {lag_offset, lag_offset + (j-i), lag_offset - (j-i)}
    
    # Compute B_k for all lags
    B = np.zeros(2*n-1)
    for idx in range(2*n-1):
        k = idx - lag_offset
        bk = 0.0
        if 0 <= i-k < n: bk += (1.0 - h[i-k])
        if 0 <= i+k < n: bk -= h[i+k]
        if 0 <= j-k < n: bk -= (1.0 - h[j-k])
        if 0 <= j+k < n: bk += h[j+k]
        B[idx] = bk
    
    # Special lags have A != 0
    af_mask = np.ones(2*n-1, dtype=bool)
    af_mask[list(special_idx)] = False
    
    # Build upper hull of affine functions (B[idx]*delta + D[idx]) for af_mask
    af_indices = np.where(af_mask)[0]
    if len(af_indices) > 0:
        slopes = B[af_indices]
        intercepts = q[af_indices]
        # Sort by slope, then by intercept
        order = np.lexsort((intercepts, slopes))
        sorted_slopes = slopes[order]
        sorted_intercepts = intercepts[order]
        
        # Remove duplicate slopes, keep max intercept
        unique_mask = np.ones(len(sorted_slopes), dtype=bool)
        for t in range(1, len(sorted_slopes)):
            if sorted_slopes[t] == sorted_slopes[t-1]:
                unique_mask[t] = False
        slopes_u = sorted_slopes[unique_mask]
        intercepts_u = sorted_intercepts[unique_mask]
        
        # Build upper hull
        hull = []  # list of (slope, intercept, start_x)
        for s, c in zip(slopes_u, intercepts_u):
            if len(hull) == 0:
                hull.append((s, c, -np.inf))
                continue
            s_prev, c_prev, x_start_prev = hull[-1]
            if s == s_prev:
                # Already handled by unique
                pass
            x_int = (c_prev - c) / (s - s_prev) if s != s_prev else -np.inf
            while len(hull) > 1 and x_int <= hull[-1][2]:
                s_prev, c_prev, x_start_prev = hull.pop()
                x_int = (c_prev - c) / (s - s_prev) if s != s_prev else -np.inf
            if len(hull) == 0:
                hull.append((s, c, -np.inf))
            else:
                hull.append((s, c, max(x_int, hull[-1][2])))
        
        # Extract segments
        segments = []
        for t in range(len(hull)):
            seg_start = hull[t][2]
            seg_end = hull[t+1][2] if t+1 < len(hull) else np.inf
            seg_lo = max(seg_start, lo)
            seg_hi = min(seg_end, hi)
            if seg_hi > seg_lo:

                roots = []
                if abs(dA) < 1e-15:
                    if abs(dB) > 1e-15:
                        roots.append(-dD / dB)
                else:
                    disc = dB * dB - 4.0 * dA * dD
                    dtol = 1e-12 * (abs(dB * dB) + abs(4.0 * dA * dD) + 1.0)
                    if disc > dtol:
                        sq = math.sqrt(disc)
                        z = -0.5 * (dB + math.copysign(sq, dB))
                        if abs(z) > 1e-15:
                            roots.extend([z / dA, dD / z])
                        else:
                            roots.append(-dB / (2.0 * dA))
                    elif disc >= -dtol:
                        roots.append(-dB / (2.0 * dA))
                for rt in roots:
                    if s_lo - tol <= rt <= s_hi + tol:
                        cands.append(max(s_lo, min(s_hi, rt)))
        seen = []
        for c in cands:
            dup = False
            for sc in seen:
                if abs(c - sc) < tol:
                    dup = True
                    break
            if not dup:
                seen.append(c)
        for c in seen:
            val = -1e30
            for (A, Bc, D) in polys:
                pv = A * c * c + Bc * c + D
                if pv > val:
                    val = pv
            ref = max(1.0, abs(best_val))
            if val < best_val - 1e-14 * ref:
                best_val = val
                best_delta = c
            elif abs(val - best_val) <= 1e-14 * ref and abs(c) < abs(best_delta):
                best_delta = c
    if not np.isfinite(best_val):
        return 0.0
    cur_max = float(np.max(q))
    if best_val >= cur_max - 1e-12 * max(1.0, abs(cur_max)):
        return 0.0
    return float(best_delta)

def _sweep(h, q, score, pairs, n, ks, rng):
    h = h.copy()
    q = q.copy()
    for pair in pairs:
        i, j = pair
        try:
            delta = _line_min(h, i, j, q, n, ks)
            if abs(delta) < 1e-14:
                continue
            h_try = h.copy()
            h_try[i] += delta
            h_try[j] -= delta
            h_try[i] = max(0.0, min(1.0, h_try[i]))
            h_try[j] = max(0.0, min(1.0, h_try[j]))
            res = math.fsum(h_try) - n / 2.0
            if abs(res) > 1e-13:
                if res > 0:
                    for x in range(n):
                        if h_try[x] > 0:
                            sub = min(res, h_try[x])
                            h_try[x] -= sub
                            res -= sub
                            if abs(res) < 1e-14:
                                break
                else:
                    for x in range(n):
                        if h_try[x] < 1.0:
                            add = min(-res, 1.0 - h_try[x])
                            h_try[x] += add
                            res += add
                            if abs(res) < 1e-14:
                                break
            if not _feasible(h_try, n, n / 2.0):
                continue
            q_try = np.correlate(h_try, 1.0 - h_try, mode="full")
            if not np.all(np.isfinite(q_try)):
                continue
            sc_try = float(np.max(q_try) * 2.0 / n)
            if sc_try < score - 1e-12:
                h = h_try
                q = q_try
                score = sc_try
        except Exception:
            continue
    return h, q, score

def run(seed=42, **kwargs):
    try:
        parent = initial_h_values
    except NameError:
        parent = None
    n = 95
    if parent is not None:
        try:
            p = np.array(parent, dtype=np.float64).ravel()
            if p.ndim == 1 and 2 <= p.shape[0] <= 999:
                n = int(p.shape[0])
        except Exception:
            pass
    if n is None or not (2 <= n <= 999):
        try:
            n = int(kwargs.get("n_points", 95))
            if not (2 <= n <= 999):
                n = 95
        except Exception:
            n = 95
    dx = 2.0 / n
    m = n / 2.0
    ks = np.arange(-(n - 1), n)
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    half = np.zeros(n)
    half[:int(m)] = 1.0
    if n % 2 == 1:
        half[n // 2] = 0.5
    else:
        half[:int(n // 2)] = 1.0
    half = _project(half, n, m)
    if half is None:
        half = np.zeros(n)
        half[:int(m)] = 1.0
        if n % 2 == 1:
            half[n // 2] = 0.5
    fallback = half.copy()
    starts = [half.copy()]
    if parent is not None:
        try:
            p = np.array(parent, dtype=np.float64).ravel()
            if p.shape == (n,) and np.all(np.isfinite(p)):
                if _feasible(p, n, m):
                    pcopy = p.copy()
                    if np.max(np.abs(pcopy - half)) >= 1e-12:
                        starts.append(pcopy)
                    fb = p.copy()
                else:
                    proj = _project(p, n, m)
                    if proj is not None and np.max(np.abs(proj - half)) >= 1e-12:
                        starts.append(proj)
            elif p.shape == (n,) and not np.all(np.isfinite(p)):
                pass
        except Exception:
            pass
    phase = rng.uniform(0, 2 * math.pi)
    idx_arr = np.arange(n, dtype=np.float64)
    sin_v = 0.5 + 0.45 * np.sin(2.0 * math.pi * idx_arr / n + phase)
    sin_proj = _project(sin_v, n, m)
    if sin_proj is not None:
        if all(np.max(np.abs(sin_proj - s)) >= 1e-12 for s in starts):
            starts.append(sin_proj)
    const_v = np.full(n, 0.5)
    if all(np.max(np.abs(const_v - s)) >= 1e-12 for s in starts):
        starts.append(const_v)
    starts = starts[:4]
    total = n * (n - 1) // 2
    Q = min(total, max(64, min(5000, int(5000 * (95.0 / n) ** 2))))
    if n <= 128:
        all_pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
        if len(all_pairs) > Q:
            sel = rng.choice(len(all_pairs), size=Q, replace=False)
            pairs = [all_pairs[s] for s in sel]
        else:
            pairs = all_pairs
    else:
        pairs = []
        seen = set()
        for _ in range(20 * Q):
            if len(pairs) >= Q:
                break
            a = rng.integers(0, n)
            b = rng.integers(0, n)
            if a == b:
                continue
            p = (min(a, b), max(a, b))
            if p not in seen:
                seen.add(p)
                pairs.append(p)
        if len(pairs) < Q:
            for i in range(n):
                for j in range(i + 1, n):
                    if len(pairs) >= Q:
                        break
                    if (i, j) not in seen:
                        seen.add((i, j))
                        pairs.append(pairs and (i, j) or (i, j))
                if len(pairs) >= Q:
                    break
    best_h = fallback.copy()
    best_q = np.correlate(best_h, 1.0 - best_h, mode="full")
    best_s = float(np.max(best_q) * dx)
    for start in starts:
        if not _feasible(start, n, m):
            continue
        q_s = np.correlate(start, 1.0 - start, mode="full")
        s_s = float(np.max(q_s) * dx)
        if s_s < best_s - 1e-12:
            best_h = start.copy()
            best_q = q_s.copy()
            best_s = s_s
        shuf = pairs[:]
        rng.shuffle(shuf)
        h_sw, q_sw, s_sw = _sweep(start, q_s, s_s, shuf, n, ks, rng)
        if s_sw < best_s - 1e-12:
            best_h = h_sw
            best_q = q_sw
            best_s = s_sw
    shuf = pairs[:]
    rng.shuffle(shuf)
    h_sw, q_sw, s_sw = _sweep(best_h, best_q, best_s, shuf, n, ks, rng)
    if s_sw < best_s - 1e-12:
        best_h = h_sw
        best_q = q_sw
        best_s = s_sw
    if not _feasible(best_h, n, m):
        prj = _project(best_h, n, m)
        if prj is not None and _feasible(prj, n, m):
            best_h = prj
        else:
            best_h = fallback.copy()
    final_q = np.correlate(best_h, 1.0 - best_h, mode="full")
    final_s = float(np.max(final_q) *dx)
    if not np.isfinite(final_s):
        best_h = fallback.copy()
        final_q = np.correlate(best_h, 1.0 - best_h, mode="full")
        final_s = float(np.max(final_q) * dx)
    return best_h, final_s, n
