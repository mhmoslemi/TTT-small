        sc = exact_score(s, dx)
        if sc is not None and sc < best_score - 1e-12:
            best_score = sc
            best_h = s.copy()
    for cand in [base, neutral]:
        if check_feasible(cand, N, m):
            sc = exact_score(cand, dx)
            if sc is not None and sc < best_score - 1e-12:
                best_score = sc
                best_h = cand.copy()
    if best_h is None:
        best_h = neutral.copy()
        best_score = exact_score(best_h, dx)

    if N <= 128:
        max_iter = 50
    elif N <= 256:
        max_iter = 30
    elif N <= 512:
        max_iter = 15
    else:
        max_iter = 6

    tau0 = min(0.04, max(0.008, 0.08 * best_score))
    taus = [tau0 * (0.25 ** s) for s in range(6)]

    if N > 1:
        for si in range(len(valid_starts)):
            h = valid_starts[si].copy()
            for tau in taus:
                eta = 0.5
                for it in range(max_iter):
                    q = np.correlate(h, 1.0 - h, mode='full') * dx
                    F_cur, p = smooth_stats(q, tau)
                    if F_cur is None:
                        break
                    g = compute_gradient(h, p, dx, N)
                    if g is None:
                        break
                    accepted = False
                    for bt in range(18):
                        trial = project_capped_simplex(h - eta * g / dx, N, m)
                        if trial is None:
                            eta *= 0.5
                            continue
                        d = trial - h
                        if float(np.max(np.abs(d))) <= 1e-10:
                            eta *= 0.5
                            continue
                        a = float(np.dot(g, d))
                        if not np.isfinite(a) or a >= 0:
                            eta *= 0.5
                            continue
                        q_t = np.correlate(trial, 1.0 - trial, mode='full') * dx
                        C_t = float(np.max(q_t))
                        F_t, _ = smooth_stats(q_t, tau)
                        if F_t is None:
                            eta *= 0.5
                            continue
                        if C_t < best_score - 1e-12:
                            best_score = C_t
                            best_h = trial.copy()
                        if F_t <= F_cur + 1e-4 * a + 1e-14 * max(1.0, abs(F_cur)):
                            h = trial.copy()
                            if bt == 0:
                                eta = min(8.0, eta * 1.5)
                            accepted = True
                            break
                        eta *= 0.5
                    if not accepted:
                        break
                    if eta < 1e-8:
                        break

    final_h = project_capped_simplex(best_h, N, m)
    if final_h is None or not check_feasible(final_h, N, m):
        final_h = neutral.copy()
    final_score = exact_score(final_h, dx)
    if final_score is None:
        final_h = neutral.copy()
        final_score = exact_score(final_h, dx)

    if parent is not None:
        pc = parent.copy() if check_feasible(parent, N, m) else project_capped_simplex(parent, N, m)
        if pc is not None and check_feasible(pc, N, m):
            ps = exact_score(pc, dx)
            if ps is not None and ps < final_score - 1e-12:
                final_h = pc.copy()
                final_score = ps

    ns = exact_score(neutral, dx)
    if ns is not None and ns < final_score - 1e-12:
        final_h = neutral.copy()
        final_score = ns

    return (final_h, final_score, N)
