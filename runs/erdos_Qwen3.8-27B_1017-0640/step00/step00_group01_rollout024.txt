import numpy as np
import math

def project_capped_simplex(v, n):
    s = n / 2.0
    if v.shape[0] != n or not np.all(np.isfinite(v)):
        return None
    if np.max(np.abs(v)) > 1e6:
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        m = math.fsum(x.tolist())
        if m > s:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(3):
        r = s - math.fsum(x.tolist())
        if abs(r) < 5e-15:
            break
        if r > 0:
            for i in range(n):
                if r <= 0:
                    break
                add = min(r, 1.0 - x[i])
                if add > 0:
                    x[i] += add
                    r -= add
        else:
            for i in range(n):
                if r >= 0:
                    break
                sub = min(-r, x[i])
                if sub > 0:
                    x[i] -= sub
                    r += sub
    if not np.all(np.isfinite(x)):
        return None
    if np.any(x < -1e-10) or np.any(x > 1.0 + 1e-10):
        return None
    if abs(math.fsum(np.clip(x, 0, 1).tolist()) - s) > 1e-10:
        return None
    return np.clip(x, 0.0, 1.0)

def is_feasible(h, n):
    if h.shape[0] != n or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return False
    if abs(math.fsum(h.tolist()) - n / 2.0) > 1e-10:
        return False
    return True

def exact_score(h, n):
    try:
        corr = np.correlate(h, 1.0 - h, mode='full')
        if np.all(np.isfinite(corr)):
            val = float(np.max(corr)) * (2.0 / n)
            if math.isfinite(val):
                return val
    except Exception:
        pass
    best = 0.0
    for d in range(-(n - 1), n):
        s = 0.0
        lo_r = max(0, -d)
        hi_r = min(n, n - d)
        for r in range(lo_r, hi_r):
            s += h[r] * (1.0 - h[r + d])
        if s > best:
            best = s
    return best * (2.0 / n)

def pair_coeffs(h, i, j, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    nl = 2 * n - 1
    c = corr.astype(np.float64) * dx
    a = np.zeros(nl)
    b = np.zeros(nl)
    for p in range(nl):
        d = (n - 1) - p
        bv = 0.0
        av = 0.0
        if 0 <= i + d < n:
            bv += 1.0 * (1.0 - h[i + d])
            vd = 1.0 if (i + d) == i else (-1.0 if (i + d) == j else 0.0)
            av -= 1.0 * vd
        if 0 <= j + d < n:
            bv += (-1.0) * (1.0 - h[j + d])
            vd = 1.0 if (j + d) == i else (-1.0 if (j + d) == j else 0.0)
            av -= (-1.0) * vd
        ri = i - d
        if 0 <= ri < n:
            bv -= h[ri] * 1.0
        rj = j - d
        if 0 <= rj < n:
            bv -= h[rj] * (-1.0)
        a[p] = av * dx
        b[p] = bv * dx
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if not (np.all(np.isfinite(a)) and np.all(np.isfinite(b)) and np.all(np.isfinite(c))):
        return None
    return a, b, c, lo, hi

def line_candidates_small(a, b, c, lo, hi):
    nl = len(a)
    cands = [lo, hi, max(lo, min(0.0, hi))]
    for p in range(nl):
        for q in range(p + 1, nl):
            A = a[p] - a[q]
            B = b[p] - b[q]
            C = c[p] - c[q]
            tol = 1e-14 * max(1.0, abs(A), abs(B), abs(C))
            if abs(A) < tol:
                if abs(B) > tol:
                    t = -C / B
                    if lo - 1e-10 <= t <= hi + 1e-10:
                        cands.append(max(lo, min(hi, t)))
            else:
                disc = B * B - 4.0 * A * C
                dtol = 1e-14 * max(1.0, B * B, abs(4.0 * A * C))
                if disc < -dtol:
                    continue
                if disc < 0:
                    disc = 0.0
                sq = math.sqrt(disc)
                R = -0.5 * (B + math.copysign(sq, B))
                if abs(R) > tol:
                    t1 = R / A
                    t2 = C / R
                else:
                    t1 = -B / (2.0 * A)
                    t2 = t1
                for t in (t1, t2):
                    if math.isfinite(t) and lo - 1e-10 <= t <= hi + 1e-10:
                        cands.append(max(lo, min(hi, t)))
    for p in range(nl):
        if a[p] > 1e-14:
            tv = -b[p] / (2.0 * a[p])
            if math.isfinite(tv) and lo <= tv <= hi:
                cands.append(tv)
    cands = sorted(set(cands))
    return cands

def eval_envelope(a, b, c, ts):
    out = np.empty(len(ts))
    batch = 256
    for s in range(0, len(ts), batch):
        e = min(s + batch, len(ts))
        seg = ts[s:e]
        vals = a * seg[:, None] ** 2 + b * seg[:, None] + c[None, :]
        out[s:e] = np.max(vals, axis=1)
    return out

def pair_line_search(h, score, i, j, n):
    pc = pair_coeffs(h, i, j, n)
    if pc is None:
        return h.copy(), score
    a, b, c, lo, hi = pc
    if hi - lo < 1e-14:
        return h.copy(), score
    if n <= 64:
        cands = line_candidates_small(a, b, c, lo, hi)
    else:
        cands = line_candidates_large(a, b, c, lo, hi)
    if len(cands) == 0:
        return h.copy(), score
    ts = np.array(cands, dtype=np.float64)
    env = eval_envelope(a, b, c, ts)
    order = np.argsort(env)
    best_t = -1
    best_exact = score - 1e-12
    count = 0
    for idx in order:
        t = float(ts[idx])
        prop = h.copy()
        prop[i] += t
        prop[j] -= t
        px = project_capped_simplex(prop, n)
        if px is None:
            continue
        if not is_feasible(px, n):
            continue
        try:
            s = exact_score(px, n)
        except Exception:
            continue
        if s < best_exact:
            best_exact = s
            best_t = t
        count += 1
        if count >= 12:
            break
    if best_t is None or best_exact >= score:
        return h.copy(), score
    prop = h.copy()
    prop[i] += best_t
    prop[j] -= best_t
    px = project_capped_simplex(prop, n)
    if px is None or not is_feasible(px, n):
        return h.copy(), score
    s2 = exact_score(px, n)
    if s2 < score:
        return px, s2
    return h.copy(), score
