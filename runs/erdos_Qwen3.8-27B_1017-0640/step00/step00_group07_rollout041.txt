    return p

def exact_score(h, n):
    if h.ndim != 1 or h.shape[0] != n or not np.all(np.isfinite(h)):
        return np.inf
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return np.inf
    if abs(np.sum(h) - n / 2.0) > 1e-9 * max(1, n):
        return np.inf
    try:
        return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))
    except Exception:
        return np.inf

def majorizer_q_and_g(h, r, n):
    dx = 2.0 / n
    nl = 2 * n - 1
    Q = np.zeros(nl)
    G = np.zeros((nl, n))
    Q[0] = dx * np.sum(h * (1.0 - h))
    G[0, :] = dx * (1.0 - 2.0 * r)
    d = h - r
    row = 1
    for s in range(1, n):
        p = n - s
        D = d[:p] + d[s:s + p]
        Q[row] = dx * (np.sum(h[:p] * (1.0 - h[s:s + p])) + 0.5 * np.sum(D * D))
        G[row, :p] = dx * ((1.0 - h[s:s + p]) + D)
        G[row, s:s + p] = dx * ((-h[:p]) + D)
        row += 1
        Q[row] = dx * (np.sum(h[s:s + p] * (1.0 - h[:p])) + 0.5 * np.sum(D * D))
        G[row, :p] = dx * ((-h[s:s + p]) + D)
        G[row, s:s + p] = dx * ((1.0 - h[:p]) + D)
        row += 1
    if not np.all(np.isfinite(Q)) or not np.all(np.isfinite(G)):
        return None, None
    return Q, G

def _sobj(z):
    return z[-1]

def _sobj_grad(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g

def _seq_fun(z):
    return np.sum(z[:-1]) - _n_ref / 2.0

def _seq_jac(z):
    j = np.zeros(len(z))
    j[:-1] = 1.0
    return j

def _sineq_fun(z):
    h, t = z[:-1], z[-1]
    Q, _ = majorizer_q_and_g(h, _r_ref, _n_ref)
    if Q is None:
        return np.full(2 * _n_ref - 1, 1e6)
    return t - Q

def _sineq_jac(z):
    h = z[:-1]
    _, G = majorizer_q_and_g(h, _r_ref, _n_ref)
    nl = 2 * _n_ref - 1
    J = np.zeros((nl, len(z)))
    if G is None:
        J[:, -1] = 1.0
        return J
    J[:, :-1] = -G
    J[:, -1] = 1.0
    return J

def _make_starts(base, n, rng, n_starts):
    starts = []
    base_clip = np.clip(base, 0.0, 1.0)
    starts.append(project_capped(base_clip, n))
    starts.append(project_capped(base_clip + 0.04 * rng.standard_normal(n), n))
    starts.append(project_capped(base_clip + 0.15 * rng.standard_normal(n), n))
    starts.append(project_capped(rng.random(n), n))
    K = min(6, (n - 1) // 2)
    prof = np.full(n, 0.5)
    idx = np.arange(n)
    for k in range(1, K + 1):
        ak = 0.12 / np.sqrt(k) * rng.standard_normal()
        bk = 0.12 / np.sqrt(k) * rng.standard_normal()
        prof += ak * np.cos(2.0 * np.pi * k * (idx + 0.5) / n)
        prof += bk * np.sin(2.0 * np.pi * k * (idx + 0.5) / n)
    starts.append(project_capped(prof, n))
    perm = rng.permutation(n)
    bin_p = np.zeros(n)
    nf = n // 2
    bin_p[perm[:nf]] = 1.0
    if n % 2 == 1:
        bin_p[perm[nf]] = 0.5
    starts.append(project_capped(bin_p, n))
    dedup = []
    for s in starts:
        dup = False
        for e in dedup:
            if np.max(np.abs(s - e)) < 1e-10:
                dup = True
                break
        if not dup:
            dedup.append(s)
        if len(dedup) >= n_starts:
            break
    return dedup

def run(seed=42, **kwargs):
    global _r_ref, _n_ref
    try:
        _seed = int(abs(seed))
    except Exception:
        _seed = 42
    rng = np.random.default_rng(_seed)
    n_starts = kwargs.get('n_starts', 5)
    outer_iters = kwargs.get('outer_iterations', 8)
    maxiter = kwargs.get('maxiter', 100)
    ftol = kwargs.get('ftol', 1e-9)
    try:
        n_starts = int(min(max(n_starts, 1), 6))
    except Exception:
        n_starts = 5
    try:
        outer_iters = int(min(max(outer_iters, 1), 16))
    except Exception:
        outer_iters = 8
    try:
        maxiter = int(min(max(maxiter, 40), 250))
    except Exception:
        maxiter = 100
    try:
        ftol = float(ftol)
        if not np.isfinite(ftol) or ftol <= 0:
            ftol = 1e-9
        ftol = min(max(ftol, 1e-12), 1e-6)
    except Exception:
        ftol = 1e-9
    # Determine n and base from initial_h_values
    n = 80
    base = np.full(n, 0.5)
    parent_ok = False
    try:
        ih = np.asarray(initial_h_values, dtype=np.float64)
        if ih.ndim == 1 and 2 <= ih.shape[0] <= 999:
            n = ih.shape[0]
            if np.all(np.isfinite(ih)):
                base = project_capped(ih, n)
                parent_ok = True
            else:
                base = np.full(n, 0.5)
        else:
            base = np.full(80, 0.5)
    except Exception:
        base = np.full(80, 0.5)
    parent_fallback = base.copy()
    # Build starts
    starts = _make_starts(base, n, rng, n_starts)
    # Score starts, find best
    best_h = starts[0].copy()
    best_score = exact_score(best_h, n)
    for s in starts[1:]:
        sc = exact_score(s, n)
        if sc < best_score:
            best_score = sc
            best_h = s.copy()
    # Optimize for n <= 160
    if n <= 160:
        _n_ref = n
        bounds = [(0.0, 1.0)] * n + [(0.0, np.inf)]
        cons = [
            {'type': 'eq', 'fun': _seq_fun, 'jac': _seq_jac},
            {'type': 'ineq', 'fun': _sineq_fun, 'jac': _sineq_jac}
        ]
        for start in starts:
            cur = start.copy()
            cur_score = exact_score(cur, n)
            if cur_score < best_score:
                best_score = cur_score
                best_h = cur.copy()
            for _ in range(outer_iters):
                if best_score <= 0.3808:
                    break
                _r_ref = cur.copy()
                Q0, _ = majorizer_q_and_g(cur, cur, n)
                if Q0 is None:
                    break
                t0 = float(np.max(Q0)) + 1e-8
                z0 = np.concatenate([cur, [t0]])
                try:
                    res = minimize(_sobj, z0, method='SLSQP', jac=_sobj_grad,
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
                except Exception:
                    break
                if res is None or res.x is None or res.x.shape != (n + 1,):
                    break
                if not np.all(np.isfinite(res.x)):
                    break
                proposal = res.x[:n]
                if not np.all(np.isfinite(proposal)) or np.max(np.abs(proposal)) > 10.0:
                    break
                proj = project_capped(proposal, n)
                candidates = [proj]
                for alpha in [0.5, 0.25, 0.125]:
                    candidates.append(project_capped(cur + alpha * (proj - cur), n))
                best_cand, best_cscore = None, cur_score
                for c in candidates:
                    if not is_feasible(c, n):
                        continue
                    if np.max(np.abs(c - cur)) < 1e-14:
                        continue
                    sc = exact_score(c, n)
                    if sc < best_cscore:
                        best_cscore = sc
                        best_cand = c
                if best_cand is not None and best_cscore < cur_score - 1e-12:
                    cur = best_cand.copy()
                    cur_score = best_cscore
                    if cur_score < best_score:
                        best_score = cur_score
                        best_h = cur.copy()
                else:
                    break
    # Final validation
    best_h = project_capped(best_h, n)
    pf = project_capped(parent_fallback, n)
    sb = exact_score(best_h, n)
    sp = exact_score(pf, n)
    if sp < sb:
        final_h = pf
    else:
        final_h = best_h
    if not is_feasible(final_h, n):
        final_h = np.full(n, 0.5)
    final_h = project_capped(final_h, n)
    c5 = exact_score(final_h, n)
    if not np.isfinite(c5):
        final_h = np.full(n, 0.5)
        c5 = exact_score(final_h, n)
    return (final_h, c5, int(n))
