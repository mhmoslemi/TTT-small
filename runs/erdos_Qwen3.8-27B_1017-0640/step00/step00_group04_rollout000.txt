from scipy.optimize import minimize

def compute_score(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = float(np.min(y)) - 1.0, float(np.max(y))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    r = target - x.sum()
    if r > 1e-12:
        for i in np.argsort(1.0 - x)[::-1]:
            a = min(r, 1.0 - x[i])
            x[i] += a
            r -= a
            if r <= 0:
                break
    elif r < -1e-12:
        for i in np.argsort(x):
            a = min(-r, x[i])
            x[i] -= a
            r += a
            if r >= 0:
                break
    return np.clip(x, 0.0, 1.0)

def lag_grad(h, n):
    G = np.zeros((2 * n - 1, n))
    r = np.arange(n)
    for k in range(-(n - 1), n):
        kk = k + n - 1
        idx1, m1 = r - k, (r - k >= 0) & (r - k < n)
        idx2, m2 = r + k, (r + k >= 0) & (r + k < n)
        g = np.zeros(n)
        g[m1] = 1.0 - h[idx1[m1]]
        g[m2] -= h[idx2[m2]]
        G[kk] = g
    return G

def epi_objective(z):
    return z[-1]

def epi_obj_grad(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g

def mass_fun(z):
    return z[:-1].sum() - len(z[:-1]) / 2.0

def mass_jac(z):
    n = len(z) - 1
    J = np.zeros(n + 1)
    J[:n] = 1.0
    return J

def epi_g(z):
    h, t = z[:-1], z[-1]
    n = len(h)
    return t - (2.0 / n) * np.correlate(h, 1.0 - h, mode='full')

def epi_jac(z):
    h = z[:-1]
    n = len(h)
    dx = 2.0 / n
    G = lag_grad(h, n)
    J = np.hstack([-dx * G, np.ones((2 * n - 1, 1))])
    return J

def is_feasible(h, n, tol=1e-9):
    if len(h) != n or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -tol) or np.any(h > 1.0 + tol):
        return False
    if abs(h.sum() - n / 2.0) > 1e-8 * max(1, n):
        return False
    return True

def try_accept(h, n):
    h = project_feasible(h.copy(), n)
    if not is_feasible(h, n):
        return None
    return h, compute_score(h)

    except Exception:
        seed = 42
