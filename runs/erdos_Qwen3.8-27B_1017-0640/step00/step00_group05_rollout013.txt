def solve_lp(h, N, Delta):
    q, G = get_lags_and_grad(h, N)
    if q is None:
        return None
    L = 2 * N - 1
    nv = 2 * N + 1
    # Objective
    c = np.zeros(nv)
    c[-1] = 1.0
    # Inequality rows
    # Lag: [G, 0, -1]
    lag_A = sparse.hstack([
        sparse.csr_matrix(G),
        sparse.csr_matrix((L, N)),
        sparse.csr_matrix(-np.ones((L, 1)))
    ])
    lag_b = -q
    # Abs: [I, -I, 0], [-I, -I, 0]
    Id = sparse.eye(N)
    abs1 = sparse.hstack([Id, sparse.csr_matrix(-Id), sparse.csr_matrix((N, 1))])
    abs2 = sparse.hstack([sparse.csr_matrix(-Id), sparse.csr_matrix(-Id), sparse.csr_matrix((N, 1))])
    abs_b = np.zeros(2 * N)
    # Trust: [0, ones, 0]
    trust_A = sparse.csr_matrix(np.hstack([np.zeros((1, N)), np.ones((1, N)), np.zeros(1)]))
    trust_b = np.array([Delta])
    A_ub = sparse.vstack([lag_A, abs1, abs2, trust_A], format='csr')
    b_ub = np.concatenate([lag_b, abs_b, trust_b])
    # Equality
    A_eq = sparse.csr_matrix(np.hstack([np.ones((1, N)), np.zeros((1, N + 1))]))
    b_eq = np.array([0.0])
    # Bounds
    lb = np.concatenate([-h, np.zeros(N), [-np.inf]])
    ub = np.concatenate([1.0 - h, np.full(N, np.inf), [np.inf]])
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=list(zip(lb, ub)), method='highs',
                      options={'presolve': True, 'max_iter': 500,
                               'primal_feasibility_tolerance': 1e-9,
                               'dual_feasibility_tolerance': 1e-9})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    if len(res.x) != nv or not np.all(np.isfinite(res.x)):
        return None
    d = res.x[:N]
    return d

        starts.append(neutral.copy())
    else:
        base = starts[0].copy()
        starts.append(project_capped(base + 0.08 * rng.standard_normal(N), m))
        starts.append(project_capped(rng.uniform(0, 1, N), m))
        ii = np.arange(N)
        cos_v = 0.5 + 0.45 * np.cos(2.0 * np.pi * 2.0 * (ii + 0.5) / N + 0.37)
        starts.append(project_capped(cos_v, m))
    starts = dedup_starts(starts)
    best_h = None
    best_c = 999.0
    for s in starts:
        if s is None:
            continue
        sc = exact_score(s, N)
        if sc is not None and sc < best_c - 1e-12:
            best_c = sc
            best_h = s.copy()
    if best_h is None:
        best_h = neutral.copy()
        best_c = exact_score(best_h, N)
        if best_c is None:
            best_c = 999.0
    if N <= 999:
        iters_per = 45 if N <= 256 else 15
        for s in starts:
            if s is None:
                continue
            h_out, c_out = local_search(s, N, iters_per, 0.5)
            if c_out < best_c - 1e-12:
                best_c = c_out
                best_h = h_out.copy()
        if N <= 256:
            h_pol, c_pol = local_search(best_h.copy(), N, 20, 0.1)
            if c_pol < best_c - 1e-12:
                best_c = c_pol
                best_h = h_pol.copy()
    final_h = project_capped(best_h, m)
    if final_h is None:
        final_h = neutral.copy()
    final_c = exact_score(final_h, N)
    if final_c is None:
        final_h = neutral.copy()
        final_c = exact_score(final_h, N)
        if final_c is None:
            final_c = 0.5
    return (final_h, float(final_c), int(N))
