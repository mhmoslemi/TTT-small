    q = np.correlate(h, 1.0 - h, mode='full') * dx
    L = 2 * N - 1
    G = np.zeros((L, N), dtype=np.float64)
    for p in range(L):
        k = p - (N - 1)
        if k >= 0:
            G[p, k:] += dx * (1.0 - h[:N - k])
            G[p, :N - k] -= dx * h[k:]
        else:
            G[p, :N + k] += dx * (1.0 - h[-k:])
            G[p, -k:] -= dx * h[:N + k]
    return q, G


def solve_lp(h, q, G, N, delta):
    L = 2 * N - 1
    nv = 2 * N + 1
    c = np.zeros(nv)
    c[-1] = 1.0
    I = sparse.eye(N, format='csr')
    ZN = sparse.zeros((N, N), format='csr')
    Z1 = sparse.zeros((N, 1), format='csr')
    blocks_A = [
        sparse.hstack([sparse.csr_matrix(G), sparse.zeros((L, N + 1), format='csr'), -sparse.csr_matrix(np.ones((L, 1)))]),
        sparse.hstack([I, ZN, Z1]),
        sparse.hstack([-I, ZN, Z1]),
        sparse.hstack([I, -sparse.eye(N, format='csr'), Z1]),
        sparse.hstack([-I, -sparse.eye(N, format='csr'), Z1]),
        sparse.hstack([sparse.zeros((1, N), format='csr'), sparse.csr_matrix(np.ones((1, N))), sparse.zeros((1, 1), format='csr')]
    ]
    blocks_b = [
        -q,
        1.0 - h,
        h.copy(),
        np.zeros(N),
        np.zeros(N),
        np.array([delta])
    ]
    A = sparse.vstack(blocks_A, format='csr')
    b = np.concatenate(blocks_b)
    A_eq = sparse.csr_matrix(np.hstack([np.ones((1, N)), np.zeros((1, N + 1))]))
    b_eq = np.array([0.0])
    bounds = [(-h[i], 1.0 - h[i]) for i in range(N)]
    bounds.extend([(0.0, None)] * N)
    bounds.append((None, None))
    return linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                   method='highs', options={'presolve': True,
                                            'dual_feasibility_tolerance': 1e-9,
                                            'primal_feasibility_tolerance': 1e-9})


def local_search(h0, N, dx, m, max_iter, delta0):
    h = h0.copy()
    C_cur = exact_score(h, N)
    delta = delta0
    fails = 0
    for _ in range(max_iter):
        if not np.isfinite(C_cur):
            break
        q, G = lag_gradient(h, N)
        if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):
            break
        res = solve_lp(h, q, G, N, delta)
        if res is None or res.status != 0 or not np.all(np.isfinite(res.x)):
            fails += 1
            delta = max(delta * 0.5, 1e-5)
            if fails >= 4 or delta < 1e-4:
                break
            continue
        fails = 0
        d_prop = res.x[:N]
        h_new = project_capped(h + d_prop, m, N)
        if h_new is None:
            delta *= 0.5
            if delta < 1e-4:
                break
            continue
        d_act = h_new - h
        l1 = np.sum(np.abs(d_act))
        if l1 > delta * (1.0 + 1e-6):
            delta *= 0.5
            if delta < 1e-4:
                break
            continue
        C_trial = exact_score(h_new, N)
        C_model = float(np.max(q + G @ d_act))
        p = C_cur - C_model
        a = C_cur - C_trial
        tol = 1e-12 * max(1.0, abs(C_cur))
        if a > tol:
            h = h_new
            C_cur = C_trial
        if p <= 1e-12:
            delta *= 0.5
        else:
            rho = a / p
            if rho < 0.1:
                delta *= 0.5
            elif rho > 0.75 and l1 >= 0.8 * delta:
                delta = min(delta * 2.0, 4.0)
        delta = max(1e-4, min(delta, 4.0))
    return h, C_cur


def run(seed=42, **kwargs):
    N_use, parent_valid, iv = 70, False, None
    try:
        iv = np.array(initial_h_values, dtype=np.float64).ravel()
        if iv.size > 0:
            N_use = iv.size
            if np.all(np.isfinite(iv)):
                parent_valid = True
    except Exception:
        pass
    N = N_use
    dx = 2.0 / N
    m = N / 2.0
    neutral = np.full(N, 0.5, dtype=np.float64)

    h_par = None
    if parent_valid:
        h_par = project_capped(iv, m, N)

    rng = np.random.default_rng(seed)
    starts = []
    if h_par is not None and check_feasible(h_par, N, m):
        starts.append(h_par)
    starts.append(neutral)

    idx = (np.arange(N, dtype=np.float64) + 0.5)
    cos_v = 0.5 + 0.45 * np.cos(2.0 * np.pi * 2.0 * idx / N + 0.37)

    if N <= 256:
        if h_par is not None:
            for sd in [0.04, 0.12]:
                pr = project_capped(h_par + rng.normal(0, sd, N), m, N)
                if pr is not None:
                    starts.append(pr)
        for sv in [rng.uniform(0, 1, N), rng.beta(0.7, 0.7, N), cos_v]:
            pr = project_capped(sv, m, N)
            if pr is not None:
                starts.append(pr)
    else:
        if h_par is not None:
            pr = project_capped(h_par + rng.normal(0, 0.08, N), m, N)
            if pr is not None:
                starts.append(pr)
        for sv in [rng.uniform(0, 1, N), cos_v]:
            pr = project_capped(sv, m, N)
            if pr is not None:
                starts.append(pr)

    dedup = []
    for s in starts:
        if not any(np.max(np.abs(s - d)) <= 1e-12 for d in dedup):
            dedup.append(s)
    starts = dedup

    best_h, best_c = neutral.copy(), float('inf')
    for s in starts:
        if not check_feasible(s, N, m):
            continue
        sc = exact_score(s, N)
        if sc < best_c - 1e-12:
            best_c, best_h = sc, s.copy()

    if N <= 999:
        mit = 45 if N <= 256 else 15
        for s in starts:
            if not check_feasible(s, N, m):
                continue
            h_res, c_res = local_search(s, N, dx, m, mit, 0.5)
            c_chk = exact_score(h_res, N)
            if c_chk < best_c - 1e-12:
                best_c, best_h = c_chk, h_res.copy()
        if N <= 256:
            h_pol, _ = local_search(best_h, N, dx, m, 20, 0.1)
            c_pol = exact_score(h_pol, N)
            if c_pol < best_c - 1e-12:
                best_c, best_h = c_pol, h_pol.copy()

    h_final = project_capped(best_h, m, N)
    if h_final is None or not check_feasible(h_final, N, m):
        h_final = h_par if (h_par is not None and check_feasible(h_par, N, m)) else neutral
    c5 = exact_score(h_final, N)
    if not np.isfinite(c5):
        h_final = neutral
        c5 = exact_score(neutral, N)
    return (h_final.astype(np.float64), float(c5), int(N))
