import numpy as np

def project_capped(v, S):
    v = np.asarray(v, dtype=np.float64).ravel().copy()
    v = np.clip(v, -1e6, 1e6)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        x = np.clip(v - mid, 0.0, 1.0)
        if np.sum(x) > S:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-15:
            break
    x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(3):
        r = S - np.sum(x)
        if abs(r) <= 1e-12:
            break
        if r > 0:
            room = 1.0 - x
            for i in np.argsort(-room):
                if r <= 0:
                    break
                d = min(r, room[i])
                x[i] += d
                r -= d
        else:
            for i in np.argsort(x):
                if r >= 0:
                    break
                d = min(-r, x[i])
                x[i] -= d
                r += d
    if not np.all(np.isfinite(x)) or np.any(x < -1e-12) or np.any(x > 1 + 1e-12):
        return None
    x = np.clip(x, 0.0, 1.0)
    if abs(np.sum(x) - S) > 1e-9:
        return None
    return x

def feasible(h, S):
    return (h is not None and len(h) == int(round(2 * S)) and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1 + 1e-12) and abs(np.sum(h) - S) <= 1e-10)

def build_pairs(n):
    pairs = []
    for k in range(-(n - 1), n):
        if k >= 0:
            i = np.arange(n - k)
            pairs.append((i + k, i.copy()))
        else:
            s = -k
            i = np.arange(n - s)
            pairs.append((i.copy(), i + s))
    return pairs

def exact_score(h, n):
    S = n * 0.5
    if not feasible(h, S):
        return None
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def majorizer_coeffs(h_ref, pairs):
    n = len(h_ref)
    A = np.zeros((len(pairs), n))
    B = np.zeros(len(pairs))
    for idx, (u, v) in enumerate(pairs):
        a = np.zeros(n)
        z = h_ref[u] + h_ref[v]
        np.add.at(a, u, 1.0 - 0.5 * z)
        np.add.at(a, v, -0.5 * z)
        A[idx] = a
        B[idx] = 0.25 * float(np.sum(z * z))
    return A, B

def solve_socp(n, S, dx, pairs, A, B, h_ref, t_ref):
    try:
        import cvxpy as cp
    except ImportError:
        return None
    x = cp.Variable(n)
    t = cp.Variable()
    con = [cp.sum(x) == S, x >= 0, x <= 1, t >= 0]
    for idx in range(len(pairs)):
        u, v = pairs[idx]
        expr = 0.25 * cp.sum_squares(x[u] - x[v]) + A[idx] @ x + B[idx]
        con.append(dx * expr <= t)
    prob = cp.Problem(cp.Minimize(t), con)
    for name in ['ECOS', 'CVXOPT']:
        try:
            prob.solve(solver=name, max_iters=200, abstol=1e-8, reltol=1e-8, feastol=1e-8)
            if prob.status in ['optimal', 'optimal_inaccurate'] and x.value is not None:
                xv = np.asarray(x.value, dtype=np.float64).ravel()
                if xv.shape == (n,) and np.all(np.isfinite(xv)):
                    return xv
        except Exception:
            pass
    return None

def mm_local(x0, n, S, dx, pairs, max_iter, gb):
    x = x0.copy()
    sc = exact_score(x, n)
    if sc is None:
        return gb
    if sc < gb[0]:
        gb[0] = sc
        gb[1] = x.copy()
    for _ in range(max_iter):
        A, B = majorizer_coeffs(x, pairs)
        cand = solve_socp(n, S, dx, pairs, A, B, x, sc)
        if cand is None:
            break
        cand = project_capped(cand, S)
        if cand is None or not feasible(cand, S):
            break
        sc_c = exact_score(cand, n)
        if sc_c is None:
            break
        if sc_c >= sc - 1e-10:
            break
        x = cand.copy()
        sc = sc_c
        if sc < gb[0]:
            gb[0] = sc
            gb[1] = x.copy()
        if sc < gb[0] - 1e-12:
            gb[0] = sc
            gb[1] = x.copy()
    if sc < gb[0]:
        gb[0] = sc
        gb[1] = x.copy()
    return gb

def run(seed=42, **kwargs):
    n = 45
    h_iv = globals().get('initial_h_values', None)
    if h_iv is not None:
        try:
            tmp = np.asarray(h_iv, dtype=np.float64).ravel()
            if len(tmp) >= 2:
                n = len(tmp)
        except Exception:
            pass
    S = n * 0.5
    dx = 2.0 / n
    pairs = build_pairs(n)
    const_05 = np.full(n, 0.5)
    fallback = const_05.copy()
    fb_score = exact_score(fallback, n)
    if fb_score is None:
        fb_score = float(np.max(np.correlate(const_05, 1.0 - const_05, mode='full')) * dx)

    h0 = None
    if h_iv is not None:
        try:
            tmp = np.asarray(h_iv, dtype=np.float64).ravel()
            if len(tmp) == n and np.all(np.isfinite(tmp)):
                pr = project_capped(tmp, S)
                if pr is not None and feasible(pr, S):
                    h0 = pr.copy()
                    fallback = pr.copy()
                    fb_score = exact_score(fallback, n)
        except Exception:
            pass
    if h0 is None:
        h0 = const_05.copy()

    rng = np.random.default_rng(seed)
    starts = [h0.copy(), const_05.copy()]
    for sd in [0.08, 0.22]:
        starts.append(h0 + rng.normal(0.0, sd, n))
    starts.append(rng.uniform(0.0, 1.0, n))
    phase = rng.uniform(0.0, 2.0 * np.pi)
    ii = np.arange(n, dtype=np.float64)
    starts.append(0.5 + 0.35 * np.sin(2.0 * np.pi * ii / n + phase))

    proj_starts = []
    for s in starts:
        p = project_capped(s, S)
        if p is not None and feasible(p, S):
            dup = False
            for ps in proj_starts:
                if np.linalg.norm(p - ps, np.inf) < 1e-12:
                    dup = True
                    break
            if not dup:
                proj_starts.append(p)

    max_iter = 12 if n <= 64 else (8 if n <= 128 else 4)
    gb = [fb_score, fallback.copy()]

    for ps in proj_starts:
        mm_local(ps, n, S, dx, pairs, max_iter, gb)

    for sd in [0.04, 0.015]:
        pert = gb[1] + rng.normal(0.0, sd, n)
        p = project_capped(pert, S)
        if p is not None and feasible(p, S):
            mm_local(p, n, S, dx, pairs, max_iter, gb)

    best_h = gb[1]
    best_sc = exact_score(best_h, n)
    if best_sc is None:
        best_h = fallback
        best_sc = fb_score
    if best_sc is None:
        best_h = const_05.copy()
        best_sc = float(np.max(np.correlate(best_h, 1.0 - best_h, mode='full')) * dx)

    best_h = np.clip(np.asarray(best_h, dtype=np.float64), 0.0, 1.0).copy()
    if abs(np.sum(best_h) - S) > 1e-10:
        rep = project_capped(best_h, S)
        if rep is not None:
            best_h = rep
    final_sc = exact_score(best_h, n)
    if final_sc is None:
        final_sc = float(np.max(np.correlate(best_h, 1.0 - best_h, mode='full')) * dx)
    return best_h, float(final_sc), n
