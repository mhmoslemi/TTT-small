    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2*n-1, n))
    one_minus_h = 1.0 - h
    for j in range(2*n-1):
        s = j - (n - 1)  # lag
        # First factor: h[m] * (1-h[m+s]) valid when 0<=m<n and 0<=m+s<n
        # m ranges: max(0, -s) to min(n-1, n-1-s)
        m_lo = max(0, -s)
        m_hi = min(n-1, n-1-s)
        if m_lo <= m_hi:
            G[j, m_lo:m_hi+1] += one_minus_h[m_lo+s:m_hi+s+1]
        # Second factor: -h[m-s] valid when 0<=m-s<n, i.e., s<=m<=s+n-1
        # combined with 0<=m<n: max(0,s) to min(n-1, s+n-1)
        m_lo2 = max(0, s)

        if m_lo2 <= m_hi2:

    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    omh = 1.0 - h
    for j in range(2 * n - 1):
        s = j - (n - 1)
        a, b = max(0, -s), min(n - 1, n - 1 - s)
        if a <= b:
            G[j, a:b+1] += omh[a+s:b+s+1]
        a2, b2 = max(0, s), min(n - 1, s + n - 1)
        if a2 <= b2:
            G[j, a2:b2+1] -= h[a2-s:b2-s+1]
    return G * dx

def phi_smooth(q, tau):
    qm = q.max()
    return float(qm + tau * np.log(np.sum(np.exp((q - qm) / tau))))

def phi_grad(q, G, tau):
    qm = q.max()
    e = np.exp((q - qm) / tau)
    w = e / e.sum()
    return float(qm + tau * np.log(e.sum())), G.T @ w

def optimize(x, n, max_iter):
    x = project_capped(x.copy(), n)
    taus = [0.02 * (0.25 ** k) for k in range(7)]
    for tau in taus:
        alpha = 1.0
        for _ in range(max_iter):
            q = lag_vals(x)
            G = jacobian(x)
            phi, g = phi_grad(q, G, tau)
            if not np.all(np.isfinite(g)):
                break
            steps = 0
            accepted = False
            while steps < 25:
                xt = project_capped(x - alpha * g, n)
                d = xt - x
                if np.abs(d).max() <= 1e-9:
                    return x
                gd = float(g @ d)
                if gd >= 0:
                    return x
                if phi_smooth(lag_vals(xt), tau) <= phi + 1e-4 * gd + 1e-14 * max(1.0, abs(phi)):
                    x = xt
                    alpha = min(alpha * 1.25, 4.0)
                    accepted = True
                    break
                alpha *= 0.5
                steps += 1
            if not accepted:
                break
    return x

def run(seed=42, **kwargs):
    n_starts, max_iter = 7, 80
    try:
        v = int(kwargs.get('n_starts', 7))
        if 1 <= v <= 8:
            n_starts = v
    except (ValueError, TypeError):
        pass
    try:
        v = int(kwargs.get('max_iter', 80))
        if 20 <= v <= 200:
            max_iter = v
    except (ValueError, TypeError):
        pass
    try:
        s = abs(int(seed))
    except (ValueError, TypeError):
        s = 42
    rng = np.random.default_rng(s)

    n, base = 80, np.full(80, 0.5)
    got_parent = False
    try:
        ihv = initial_h_values
        if isinstance(ihv, np.ndarray) and ihv.ndim == 1 and 2 <= len(ihv) <= 999:
            n = len(ihv)
            if np.all(np.isfinite(ihv)):
                base = project_capped(np.array(ihv, dtype=np.float64), n)
                got_parent = True
            else:
                base = np.full(n, 0.5)
    except (NameError, ValueError, TypeError):
        pass
    if not got_parent:
        try:
            nv = int(kwargs.get('n_points', 80))
            if 2 <= nv <= 999:
                n = nv
        except (ValueError, TypeError):
            pass
        base = np.full(n, 0.5)
    base = project_capped(base.astype(np.float64), n)

    best = base.copy()
    best_sc = exact_score(best)

    if n > 256:
        return best, best_sc, int(n)

    cands = [base]
    for sd in (0.02, 0.08, 0.20):
        cands.append(base + rng.standard_normal(n) * sd)
    cands.append(rng.random(n).astype(np.float64))
    K = min(6, (n - 1) // 2)
    t = (np.arange(n, dtype=np.float64) + 0.5) / n
    y = np.full(n, 0.5)
    for k in range(1, K + 1):
        y += rng.normal(0.0, 0.12 / np.sqrt(k)) * np.cos(2.0 * np.pi * k * t)
        y += rng.normal(0.0, 0.12 / np.sqrt(k)) * np.sin(2.0 * np.pi * k * t)
    cands.append(y)
    bv = np.zeros(n)
    bv[:n // 2] = 1.0
    if n % 2:
        bv[n // 2] = 0.5
    cands.append(bv[rng.permutation(n)])
    cands.append(base + rng.standard_normal(n) * 0.4)

    starts = []
    for c in cands:
        p = project_capped(np.asarray(c, dtype=np.float64), n)
        dup = False
        for s2 in starts:
            if np.abs(p - s2).max() < 1e-10:
                dup = True
                break
        if not dup:
            starts.append(p)
        if len(starts) >= n_starts:
            break

    for st in starts:
        if best_sc <= 0.3808:
            break
        x = optimize(st, n, max_iter)
        sc = exact_score(x)
        if sc < best_sc - 1e-12:
            best, best_sc = x.copy(), sc

    final = project_capped(best, n)
    sc = exact_score(final)
    if sc == float('inf'):
        final = np.full(n, 0.5)
        sc = exact_score(final)
    return final, sc, int(n)
