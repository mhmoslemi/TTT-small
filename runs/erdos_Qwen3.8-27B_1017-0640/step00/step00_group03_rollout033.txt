            m[:] = archive
        pop_scores.append(s)
        if s < archive_score:
            archive = m.copy()
            archive_score = s
    stagnation = 0
    restarts = 0
    for gen in range(G):
        gen_start_score = archive_score
        cur = [pop[i].copy() for i in range(P)]
        cur_sc = list(pop_scores)
        nxt = [cur[i].copy() for i in range(P)]
        nxt_sc = list(cur_sc)
        F = rng.uniform(0.55, 0.90) if gen < G // 2 else rng.uniform(0.35, 0.75)
        for i in range(P):
            try:
                CR = rng.uniform(0.65, 1.0)
                others = [j for j in range(P) if j != i]
                perm = rng.permutation(len(others))
                if rng.random() < 0.75:
                    r1 = others[perm[0]]
                    r2 = others[perm[1]]
                    r3 = others[perm[2]]
                    donor = cur[r1] + F * (cur[r2] - cur[r3])
                else:
                    r1 = others[perm[0]]
                    r2 = others[perm[1]]
                    donor = cur[i] + F * (archive - cur[i]) + F * (cur[r1] - cur[r2])
                mask = rng.random(n) < CR
                mask[int(rng.integers(n))] = True
                trial = np.where(mask, donor, cur[i])
                p = project_simplex(trial, n)
                if p is None:
                    continue
                s = score_h(p, n)
                if not np.isfinite(s):
                    continue
                if s < archive_score:
                    archive = p.copy()
                    archive_score = s
                if s < nxt_sc[i]:
                    nxt[i] = p
                    nxt_sc[i] = s
            except Exception:
                continue
        pop = nxt
        pop_scores = nxt_sc
        if archive_score < gen_start_score - 1e-15:
            stagnation = 0
        else:
            stagnation += 1
        if stagnation >= 18 and restarts < 2:
            sig = 0.10 if restarts == 0 else 0.035
            pop2 = [archive.copy(), pop[0].copy()]
            sc2 = [archive_score, pop_scores[0]]
            att2 = 0
            while len(pop2) < P and att2 < 5 * P:
                att2 += 1
                if len(pop2) <= int(3 * P / 4):
                    cand = archive + rng.standard_normal(n) * sig
                else:
                    cand = rng.random(n)
                p = project_simplex(cand, n)
                if p is None:
                    p = fallback.copy()
                pop2.append(p)
                sc2.append(score_h(p, n))
            while len(pop2) < P:
                pop2.append(fallback.copy())
                sc2.append(fb_score)
            for j in range(P):
                if not np.isfinite(sc2[j]):
                    sc2[j] = archive_score
                    pop2[j] = archive.copy()
                if sc2[j] < archive_score:
                    archive = pop2[j].copy()
                    archive_score = sc2[j]
            pop = pop2
            pop_scores = sc2
            stagnation = 0
            restarts += 1
    candidates = [archive, fallback, uniform]
    best_h = uniform.copy()
    best_s = score_h(best_h, n)
    for c in candidates:
        if c is None:
            continue
        cc = c.copy()
        s = score_h(cc, n)
        if not np.isfinite(s):
            p2 = project_simplex(cc, n)
            if p2 is not None:
                s = score_h(p2, n)
                cc = p2
        if np.isfinite(s) and s < best_s:
            best_s = s
            best_h = cc
    final_s = score_h(best_h, n)
    if not np.isfinite(final_s):
        best_h = uniform.copy()
        final_s = score_h(best_h, n)
    return best_h, final_s, int(n)
