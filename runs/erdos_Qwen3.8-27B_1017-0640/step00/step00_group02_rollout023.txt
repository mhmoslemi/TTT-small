            for _b in range(20):
                y = project_capped(x - alpha * g, S)
                if y is None:
                    alpha *= 0.5
                    continue
                d = y - x
                if not np.all(np.isfinite(d)):
                    alpha *= 0.5
                    continue
                F_y, _ = smooth_eval(y, tau, dx, n)
                if F_y <= F_cur + np.dot(g, d) + np.dot(d, d) / (2.0 * alpha) + 1e-12 * (1.0 + abs(F_cur)):
                    F_cur = F_y
                    x = y
                    sc = exact_score(x, n)
                    if np.isfinite(sc) and sc < best_sc - 1e-14:
                        best_sc = sc
                        best = x.copy()
                    accepted = True
                    break
                alpha *= 0.5
            if not accepted or alpha < 1e-12 or d is None:
                break
            if np.linalg.norm(d) / alpha <= 1e-7:
                break
            alpha = min(1.5 * alpha, 4.0)
            F_cur, g = smooth_eval(x, tau, dx, n)
            if g is None:
                break
    return best, best_sc

def run(seed=42, **kwargs):
    try:
        parent = np.array(initial_h_values, dtype=np.float64).ravel()
    except Exception:
        parent = None
    if parent is not None and parent.size >= 2:
        n = int(parent.size)
    else:
        n = 45
        parent = None
    dx = 2.0 / n
    S = n / 2.0
    const = np.full(n, 0.5, dtype=np.float64)
    incumbent = const.copy()
    incumbent_sc = exact_score(const, n)
    if parent is not None:
        if feasible(parent, n):
            incumbent = parent.copy()
            incumbent_sc = exact_score(incumbent, n)
        else:
            repaired = project_capped(parent, S)
            if repaired is not None and feasible(repaired, n):
                sc_r = exact_score(repaired, n)
                if np.isfinite(sc_r) and sc_r < incumbent_sc:
                    incumbent = repaired.copy()
                    incumbent_sc = sc_r
    rng = np.random.default_rng(seed)
    base = incumbent
    starts = [incumbent.copy()]
    if n <= 128:
        g_scales = [0.04, 0.12, 0.30]
    elif n <= 512:
        g_scales = [0.08, 0.25]
    else:
        g_scales = [0.12]
    for sd in g_scales:
        p = base + rng.normal(0.0, sd, n)
        r = project_capped(p, S)
        if r is not None and feasible(r, n):
            starts.append(r)
    p = rng.uniform(0.0, 1.0, n)
    r = project_capped(p, S)
    if r is not None and feasible(r, n):
        starts.append(r)
    idx = np.arange(n, dtype=np.float64)
    if n <= 128:
        sin_modes = [1, 2]
    elif n <= 512:
        sin_modes = [1]
    else:
        sin_modes = []
    for m in sin_modes:
        ph = rng.uniform(0.0, 2.0 * np.pi)
        p = 0.5 + 0.35 * np.sin(2.0 * np.pi * m * idx / n + ph)
        r = project_capped(p, S)
        if r is not None and feasible(r, n):
            starts.append(r)
    deduped = [starts[0]]
    for s in starts[1:]:
        ok = True
        for t in deduped:
            if np.linalg.norm(s - t, np.inf) < 1e-12:
                ok = False
                break
        if ok:
            deduped.append(s)
    iters = max(15, min(80, int(3600 // n)))
    for st in deduped:
        try:
            c, sc = local_search(st, n, S, dx, iters)
            if np.isfinite(sc) and sc < incumbent_sc - 1e-14:
                incumbent = c.copy()
                incumbent_sc = sc
        except Exception:
            pass
    if n <= 128:
        a_scales = [0.05, 0.015]
    elif n <= 512:
        a_scales = [0.04, 0.015]
    else:
        a_scales = [0.03]
    for scv in a_scales:
        for _ in range(2):
            try:
                p = incumbent + rng.normal(0.0, scv, n)
                r = project_capped(p, S)
                if r is not None and feasible(r, n):
                    c, sc2 = local_search(r, n, S, dx, iters)
                    if np.isfinite(sc2) and sc2 < incumbent_sc - 1e-14:
                        incumbent = c.copy()
                        incumbent_sc = sc2
            except Exception:
                pass
    if not feasible(incumbent, n):
        incumbent = const.copy()
    final_sc = exact_score(incumbent, n)
    if not np.isfinite(final_sc):
        incumbent = const.copy()
        final_sc = exact_score(const, n)
    return incumbent, final_sc, n
