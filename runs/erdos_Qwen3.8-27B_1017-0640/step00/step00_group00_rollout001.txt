def capped_simplex_project(y, m):
    n = len(y)
    y = np.clip(y.astype(np.float64), -1e6, 1e6)
    lo, hi = y.min() - 1.0, y.max()
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > m:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    r = m - x.sum()
    if r > 1e-15:
        for i in range(n):
            add = min(r, 1.0 - x[i])
            x[i] += add
            r -= add
            if r < 1e-15:
                break
    elif r < -1e-15:
        for i in range(n):
            sub = min(-r, x[i])
            x[i] -= sub
            r += sub
            if -r < 1e-15:
                break
    # Final correction
    r = m - x.sum()
    for i in range(n):
        if r > 0 and x[i] < 1.0:
            add = min(r, 1.0 - x[i])
            x[i] += add
            r -= add
            if abs(r) < 1e-14:

            sub = min(-r, x[i])
            x[i] -= sub
            r += sub
            if abs(r) < 1e-14:
                break
    x = np.clip(x, 0.0, 1.0)
    if not np.all(np.isfinite(x)) or np.abs(x.sum() - m) > 1e-9:
        x = np.full(n, m / n)
    return x

    if np.any(h < -1e-12) or np.any(h > 1 + 1e-12):

    c = np.correlate(h, 1.0 - h, mode='full') * dx

    return float(s) if np.isfinite(s) else float('inf')

    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    G[n-1] = dx * (1.0 - 2.0 * h)
    for p in range(1, n):

        G[n-1+p, p:] += dx * (1.0 - h[:n-p])

        G[n-1-p, :n-p] += dx * (1.0 - h[p:])
        G[n-1-p, p:] -= dx * h[:n-p]
    return G

    return g

        starts.append(capped_simplex_project(parent + sig * rng.standard_normal(n), m))

    maxs = 10 if n <= 100 else (7 if n <= 300 else 4)
    if len(starts) > maxs:
        starts = starts[:maxs]
    uniq = [starts[0]]
    for s in starts[1:]:
        if not any(np.max(np.abs(s - u)) < 1e-10 for u in uniq):
            uniq.append(s)
    starts = uniq

    maxiter = 250 if n <= 100 else (120 if n <= 300 else 60)
    for s in starts:
        res = solve_start(s, n, maxiter)
        if res is not None:
            sc = exact_score(res, n)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best, best_score = res, sc
        sc = exact_score(s, n)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best, best_score = s.copy(), sc

    nrounds = 2 if n <= 100 else 1
    if n <= 100:
        sigmas = [0.03, 0.10, 0.25]
    elif n <= 300:
        sigmas = [0.05, 0.18]
    else:
        sigmas = [0.08]
    for rnd in range(nrounds):
        old_best = best_score
        for sig in sigmas:
            p = best + sig * rng.standard_normal(n)
            p = capped_simplex_project(p, m)
            res = solve_start(p, n, maxiter)
            if res is not None:
                sc = exact_score(res, n)
                if np.isfinite(sc) and sc < best_score - 1e-12:
                    best, best_score = res, sc
            sc = exact_score(p, n)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best, best_score = p.copy(), sc
        if best_score > old_best - 1e-9:
            break

    best = capped_simplex_project(best, m)
    if not is_feasible(best, m):
        best = parent.copy()
    c5 = exact_score(best, n)
    if not np.isfinite(c5):
        best = np.full(n, 0.5)
        c5 = exact_score(best, n)
    return (best, c5, int(n))
