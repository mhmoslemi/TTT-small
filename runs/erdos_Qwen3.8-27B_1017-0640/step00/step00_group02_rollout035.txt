import numpy as np
import cvxpy as cp

def project_capped(v, S):
    v = np.clip(np.asarray(v, dtype=np.float64).copy(), -1e6, 1e6)
    n = len(v)
    lo, hi = float(v.min()) - 1.0, float(v.max())
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if mid == lo or mid == hi:
            break
        x = np.clip(v - mid, 0.0, 1.0)
        if x.sum() > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(3):
        r = S - x.sum()
        if abs(r) <= 1e-12:
            break
        for i in range(n):
            if r > 0:
                add = min(r, 1.0 - x[i])
                if add > 0:
                    x[i] += add
                    r -= add
                    if r <= 0:
                        break
            else:
                sub = min(-r, x[i])
                if sub > 0:
                    x[i] -= sub
                    r += sub
                    if r >= 0:
                        break
    if not np.all(np.isfinite(x)) or x.min() < -1e-10 or x.max() > 1.0 + 1e-10:
        return None
    if abs(x.sum() - S) > 1e-10:
        return None
    return np.clip(x, 0.0, 1.0)

def build_lag_pairs(n):
    pairs = []
    for k in range(-(n - 1), n):
        ilo = max(0, -k)
        ih = min(n - 1, n - 1 - k)
        u = np.arange(ilo + k, ih + k + 1, dtype=np.int64)
        v = np.arange(ilo, ih + 1, dtype=np.int64)
        pairs.append((u, v))
    return pairs

def exact_score(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n)

def is_feasible(h, n, S):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if h.min() < -1e-10 or h.max() > 1.0 + 1e-10:
        return False
    return abs(h.sum() - S) <= 1e-10

        return 12
    if n <= 128:
        return 8

def solve_subproblem(n, S, dx, pairs, h_ref):
    x = cp.Variable(n)
    t = cp.Variable(scalar=True)
    cons = [cp.sum(x) == S, x >= 0, x <= 1, t >= 0]
    for (u, v) in pairs:
        z = h_ref[u] + h_ref[v]
        a = np.zeros(n)
        a[u] += 1.0 - z / 2.0
        a[v] += -z / 2.0
        b = float(np.sum(z * z)) / 4.0
        expr = 0.25 * cp.sum_squares(x[u] - x[v]) + a @ x + b
        cons.append(dx * expr <= t)
    prob = cp.Problem(cp.Minimize(t), cons)
    for solver in [cp.ECOS, cp.CVXOPT]:
        try:
            prob.solve(solver=solver, max_iters=300, abstol=1e-9, reltol=1e-9)
        except Exception:
            continue
        if prob.status in ('optimal', 'optimal_inaccurate'):
            val = prob.variables()[0].value
            if val is not None and val.shape == (n,) and np.all(np.isfinite(val)):
                return np.asarray(val, dtype=np.float64)
    return None

def mm_local(x0, n, S, dx, pairs, max_it):
    x = x0.copy()
    cur = exact_score(x, n)
    for _ in range(max_it):
        sol = solve_subproblem(n, S, dx, pairs, x)
        if sol is None:

            ds = cur - sc
            x = sol
            cur = sc
            if dx_inf <= 1e-7 or ds < 1e-8:
                break
        else:
            break
    return x, cur
