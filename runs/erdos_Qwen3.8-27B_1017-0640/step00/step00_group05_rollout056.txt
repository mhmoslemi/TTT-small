    h_half = np.full(N, 0.5, dtype=np.float64)
    best_h = h_half.copy()
    best_score = exact_score(h_half)

    if parent is not None:
        hp = project_box_sum(parent, m)
        sp = exact_score(hp)
        if sp < best_score - 1e-12:
            best_score = sp
            best_h = hp.copy()

    if N > 999:
        return (best_h, float(best_score), int(N))

    base = parent.copy() if parent is not None else h_half.copy()
    base = project_box_sum(base, m)
    starts = [base.copy()]

    if N <= 256:
        for scale in [0.03, 0.08, 0.18]:
            noise = rng.standard_normal(N) * scale
            starts.append(project_box_sum(base + noise, m))
        starts.append(project_box_sum(rng.random(N), m))
        starts.append(project_box_sum(rng.beta(0.7, 0.7, N), m))
        idx = np.arange(N)
        s1 = 0.5 + 0.45 * np.cos(2.0 * np.pi * (idx + 0.5) / N)
        starts.append(project_box_sum(s1, m))
        s2 = 0.5 + 0.45 * np.cos(2.0 * np.pi * 2.0 * (idx + 0.5) / N + np.pi / 3.0)
        starts.append(project_box_sum(s2, m))
        # complement of parent
        if parent is not None:
            starts.append(project_box_sum(1.0 - parent, m))
    else:
        starts.append(project_box_sum(base + rng.standard_normal(N) * 0.08, m))
        starts.append(project_box_sum(rng.random(N), m))
        starts.append(project_box_sum(rng.beta(0.7, 0.7, N), m))

    for s in starts:
        sc = exact_score(s)
        if sc < best_score - 1e-12:
            best_score = sc
            best_h = s.copy()

    for s in starts:
        try:
            res = slsqp_solve(s, N, m)
            if res.x is not None and res.x.size == N + 1 and np.all(np.isfinite(res.x)):
                h_new = project_box_sum(res.x[:N], m)
                sc = exact_score(h_new)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_h = h_new.copy()
        except Exception:
            continue

    best_h = project_box_sum(best_h, m)
    final_score = exact_score(best_h)
    if not (np.all(best_h >= -1e-14) and np.all(best_h <= 1.0 + 1e-14)
            and abs(float(np.sum(best_h)) - m) < 1e-8):
        best_h = h_half
        final_score = exact_score(h_half)

    return (best_h, float(final_score), int(N))
