from scipy.optimize import minimize

def exact_score(h):
    n = len(h)
    if n < 2:
        return float('inf')
    q = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(q) * (2.0 / n))

def is_feasible(h, n, m):
    if h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return False
    if abs(math.fsum(h) - m) > 1e-9:
        return False
    return True

def project_capped(v, n, m):
    v = np.asarray(v, dtype=np.float64)
    if not np.all(np.isfinite(v)) or v.shape != (n,):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        x = np.clip(v - mid, 0.0, 1.0)
        s = math.fsum(x)
        if abs(s - m) < 1e-13:
            break
        if s > m:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    # Repair residual
    for _ in range(3):
        r = m - math.fsum(x)
        if abs(r) < 1e-14:
            break
        if r > 0:
            for i in range(n):
                add = min(r, 1.0 - x[i])
                if add > 0:
                    x[i] += add
                    r -= add
                if r <= 0:
                    break
        else:
            for i in range(n):
                sub = min(-r, x[i])
                if sub > 0:
                    x[i] -= sub
                    r += sub
                if r >= 0:
                    break
    if not np.all(np.isfinite(x)) or np.any(x < -1e-12) or np.any(x > 1.0 + 1e-12):
        return None
    return x

def smooth_value_grad(h, tau, dx):
    n = len(h)
    q = np.correlate(h, 1.0 - h, mode='full')
    qmax = np.max(q)
    shifted = (q - qmax) / tau
    exps = np.exp(shifted)
    p = exps / np.sum(exps)
    f = dx * (qmax + tau * np.log(np.sum(exps)))
    # Gradient
    rhs = np.convolve(1.0 - h, p, mode='full')
    lhs = np.convolve(h, p[::-1], mode='full')
    grad = dx * (rhs[n-1:n-1+n] - lhs[n-1:n-1+n])
    return f, grad

def mass_constraint(x, n, m):
    return np.sum(x) - m

def mass_constraint_jac(x, n, m):
    return np.ones(n)

    m = n / 2.0
    full = int(m)
    h[:full] = 1.0
    if m - full > 0.5:
        h[full] = m - full
    return h

import numpy as np
import math
from scipy.optimize import minimize


def exact_score(h):
    n = len(h)
    if n < 2:
        return float('inf')
    q = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(q) * (2.0 / n))


def is_feasible(h, n, m):
    if h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return False
    if abs(math.fsum(h) - m) > 1e-8:
        return False
    return True


def project_capped(v, n, m):
    v = np.asarray(v, dtype=np.float64)
    if v.shape != (n,) or not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        x = np.clip(v - mid, 0.0, 1.0)
        s = math.fsum(x)
        if abs(s - m) < 1e-13:
            break
        if s > m:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(3):
        r = m - math.fsum(x)
        if abs(r) < 1e-14:
            break
        if r > 0:
            for i in range(n):
                add = min(r, 1.0 - x[i])
                if add > 0:
                    x[i] += add
                    r -= add
                if r <= 0:
                    break
        else:
            for i in range(n):
                sub = min(-r, x[i])
                if sub > 0:
                    x[i] -= sub
                    r += sub
                if r >= 0:
                    break
    if not np.all(np.isfinite(x)) or np.any(x < -1e-12) or np.any(x > 1.0 + 1e-12):
        return None
    return x


def smooth_value_grad(h, tau, dx):
    q = np.correlate(h, 1.0 - h, mode='full')
    qmax = float(np.max(q))
    shifted = (q - qmax) / tau
    exps = np.exp(shifted)
    total = float(np.sum(exps))
    p = exps / total
    f = dx * (qmax + tau * math.log(total))
    rhs = np.convolve(1.0 - h, p, mode='full')
    lhs = np.convolve(h, p[::-1], mode='full')
    n = len(h)
    grad = dx * (rhs[n - 1:2 * n - 1] - lhs[n - 1:2 * n - 1])
    return f, grad


def mass_constraint(x, n, m):
    return float(np.sum(x)) - m


def mass_constraint_jac(x, n, m):
    return np.ones(n)


def fallback_h(n):
    h = np.zeros(n, dtype=np.float64)
    full = int(n // 2)
    h[:full] = 1.0
    rem = n / 2.0 - full
    if rem > 0:
        h[full] = rem
    return h
