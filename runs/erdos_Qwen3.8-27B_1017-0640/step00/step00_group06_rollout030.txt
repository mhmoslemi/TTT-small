import numpy as np
import math

def exact_score(h, n):
    if h.ndim != 1 or len(h) != n or not np.all(np.isfinite(h)):
        return float('inf')
    q = np.correlate(h, 1.0 - h, mode='full')
    s = float(np.max(q)) * (2.0 / n)
    if not math.isfinite(s):
        return float('inf')
    return s

def is_feasible(h, n, m):
    if h.ndim != 1 or len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return False
    if abs(math.fsum(h) - m) > 1e-10:
        return False
    return True

def project_capped(v, n, m):
    if v.ndim != 1 or len(v) != n or not np.all(np.isfinite(v)):
        return None
    if np.max(np.abs(v)) > 1e12:
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if mid == lo or mid == hi:
            break
        x = np.clip(v - mid, 0.0, 1.0)
        s = math.fsum(x)
        if s > m:
            lo = mid
        else:
            hi = mid
    mid = 0.5 * (lo + hi)
    x = np.clip(v - mid, 0.0, 1.0).copy()
    for _ in range(4):
        resid = m - math.fsum(x)
        if abs(resid) < 1e-14:
            break
        if resid > 0:
            for i in range(n):
                if resid <= 0:
                    break
                add = min(resid, 1.0 - x[i])
                if add > 0:
                    x[i] += add
                    resid -= add
        else:
            for i in range(n):
                if resid >= 0:
                    break
                sub = min(-resid, x[i])
                if sub > 0:
                    x[i] -= sub
                    resid += sub
    if not np.all(np.isfinite(x)) or not is_feasible(x, n, m):
        return None
    return x

def make_half_block(n):
    h = np.zeros(n)

            if p is not None:
                return p
    except Exception:
        pass
    return None

def make_noise_start(base, rng, scale, n, m):
    v = base + rng.standard_normal(n) * scale
    return project_capped(v, n, m)
