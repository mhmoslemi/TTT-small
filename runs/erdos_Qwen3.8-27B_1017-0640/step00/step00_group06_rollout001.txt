        if np.all(parent >= 0) and np.all(parent <= 1) and abs(math.fsum(parent) - n * 0.5) < 1e-9:
            parent_pt = parent.copy()
        else:
            parent_pt = project_capped_simplex(parent, n)

    fb = fallback_h(n)
    base = parent_pt if parent_pt is not None else fb

    rng = np.random.default_rng(seed)

    starts = [base.copy()]
    for sd in [0.05, 0.15, 0.30]:
        s = project_capped_simplex(base + sd * rng.standard_normal(n), n)
        if s is not None:
            starts.append(s)

    u = np.ones(n) * 0.5
    s = project_capped_simplex(u, n)
    if s is not None:
        starts.append(s)

    bv = rng.beta(0.7, 0.7, n)
    s = project_capped_simplex(bv, n)
    if s is not None:
        starts.append(s)

    phase = rng.uniform(0.0, 2.0 * math.pi)
    idx = np.arange(n, dtype=np.float64)
    sinusoid = 0.5 + 0.45 * np.sin(2.0 * math.pi * idx / n + phase)
    s = project_capped_simplex(sinusoid, n)
    if s is not None:
        starts.append(s)

    perm = base.copy()
    rng.shuffle(perm)
    s = project_capped_simplex(perm, n)
    if s is not None:
        starts.append(s)

    valid_starts = [s for s in starts if s is not None and is_feasible(s, n)]
    if not valid_starts:
        valid_starts = [fb]

    best_h = valid_starts[0].copy()
    best_score = exact_score(best_h)
    for s in valid_starts[1:]:
        sc = exact_score(s)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_h = s.copy()

    tau_schedule = [0.10, 0.03, 0.008, 0.002, 0.0005]
    tau_scale = n / 95.0
    bounds = [(0.0, 1.0)] * n
    constraint = {'type': 'eq', 'fun': _mass_fun, 'jac': _mass_jac}

    for start in valid_starts:
        cur = start.copy()
        t0 = tau_schedule[0] * tau_scale
        f0 = smooth_obj_grad(cur, t0, dx)
        if not np.isfinite(f0[0]):
            continue
        cur_f_val = f0[0]

        for tau in tau_schedule:
            tau_eff = tau * tau_scale
            try:
                res = minimize(
                    smooth_obj_grad, cur, args=(tau_eff, dx),
                    method='SLSQP', jac=True,
                    bounds=bounds,
                    constraints=[constraint],
                    options={'maxiter': 120, 'ftol': 1e-9, 'disp': False}
                )
            except Exception:
                break

            if res.x is None or res.x.shape != (n,) or not np.all(np.isfinite(res.x)):
                break

            proj = project_capped_simplex(res.x, n)
            if proj is None or not is_feasible(proj, n):
                break

            sc = exact_score(proj)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_h = proj.copy()

            nf = smooth_obj_grad(proj, tau_eff, dx)
            if not np.isfinite(nf[0]):
                break
            if nf[0] > cur_f_val + 1e-10:
                break
            cur = proj.copy()
            cur_f_val = nf[0]

    if not is_feasible(best_h, n):
        proj = project_capped_simplex(best_h, n)
        if proj is not None and is_feasible(proj, n):
            best_h = proj
        else:
            best_h = fb

    final_score = exact_score(best_h)
    return best_h, final_score, n
