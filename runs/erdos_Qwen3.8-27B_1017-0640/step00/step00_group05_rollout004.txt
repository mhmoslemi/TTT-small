        parent = None
    if n <= 0:
        n = 70
        parent = None
    m = n / 2.0
    neutral = 0.5 * np.ones(n)
    if parent is not None:
        p = project_box_sum(parent, m, n)
    else:
        p = neutral.copy()

    starts = [p]
    if n <= 256:
        for scale in [0.03, 0.08, 0.18]:
            starts.append(project_box_sum(p + scale * rng.standard_normal(n), m, n))
        starts.append(project_box_sum(rng.uniform(0.0, 1.0, n), m, n))
        starts.append(project_box_sum(rng.beta(0.7, 0.7, n), m, n))
        idx = np.arange(n, dtype=np.float64)
        smooth1 = 0.5 + 0.45 * np.cos(2.0 * np.pi * idx / n)
        smooth2 = 0.5 + 0.45 * np.cos(4.0 * np.pi * idx / n + np.pi / 3.0)
        starts.append(project_box_sum(smooth1, m, n))
        starts.append(project_box_sum(smooth2, m, n))
    else:
        starts.append(project_box_sum(p + 0.08 * rng.standard_normal(n), m, n))
        starts.append(project_box_sum(rng.uniform(0.0, 1.0, n), m, n))
        starts.append(project_box_sum(rng.beta(0.7, 0.7, n), m, n))

    inc_h = starts[0].copy()
    inc_score = exact_score(inc_h, n)
    for s in starts[1:]:
        sc = exact_score(s, n)
        if sc < inc_score - 1e-12:
            inc_score = sc
            inc_h = s.copy()

    if n <= 999:
        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [
            {'type': 'eq', 'fun': slsqp_eq_fun, 'jac': slsqp_eq_jac, 'args': (n, m)},
            {'type': 'ineq', 'fun': slsqp_ineq_fun, 'jac': slsqp_ineq_jac, 'args': (n,)},
        ]
        for h0 in starts:
            t0 = exact_score(h0, n)
            t0 = max(t0, 0.0)
            t0 = min(t0, 2.0)
            z0 = np.concatenate([h0, [t0]])
            try:
                res = minimize(slsqp_obj, z0, jac=slsqp_obj_grad, args=(n,),
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 250, 'ftol': 1e-10, 'disp': False})
                if res is not None and hasattr(res, 'x') and res.x is not None:
                    h_new = np.asarray(res.x[:n], dtype=np.float64)
                    if h_new.size == n and np.all(np.isfinite(h_new)):
                        h_new = project_box_sum(h_new, m, n)
                        sc = exact_score(h_new, n)
                        if sc < inc_score - 1e-12:
                            inc_score = sc
                            inc_h = h_new.copy()
            except Exception:
                continue

    inc_h = project_box_sum(inc_h, m, n)
    if (np.all(inc_h >= -1e-12) and np.all(inc_h <= 1.0 + 1e-12) and
            abs(float(np.sum(inc_h)) - m) < 1e-8):
        inc_h = np.clip(inc_h, 0.0, 1.0)
        final_score = exact_score(inc_h, n)
    else:
        inc_h = neutral.copy()
        final_score = exact_score(inc_h, n)
    return (inc_h, final_score, n)
