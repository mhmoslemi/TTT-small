import numpy as np

def project_capped(v, S):
    n = len(v)
    v = np.clip(v.astype(np.float64), -1e6, 1e6)
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(80):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        if x.sum() > S:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-16:
            break
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    for _ in range(4):
        r = S - x.sum()
        if abs(r) <= 1e-12:
            break
        if r > 0:
            cap = 1.0 - x
            order = np.argsort(-cap)
            for i in order:
                if r <= 0:
                    break
                a = min(r, cap[i])
                x[i] += a
                r -= a
        else:
            order = np.argsort(x)
            for i in order:
                if -r <= 0:
                    break
                a = min(-r, x[i])
                x[i] -= a
                r += a
    if not np.all(np.isfinite(x)):
        return None
    if x.min() < -1e-10 or x.max() > 1.0 + 1e-10:
        return None
    if abs(x.sum() - S) > 1e-10:
        return None
    return np.clip(x, 0.0, 1.0)

        return False
    if abs(h.sum() - n * 0.5) > 1e-10:
        return False
    return True


def exact_score(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def build_lag_pairs(n):
    pairs = []
    for k in range(-(n - 1), n):
        if k >= 0:
            idx = np.arange(n - k)
            u = (idx + k).astype(np.int64)
            v = idx.astype(np.int64)
        else:
            idx = np.arange(n + k)
            u = idx.astype(np.int64)
            v = (idx - k).astype(np.int64)
        pairs.append((u, v))
    return pairs


def build_majorizer_coeffs(h_ref, pairs, n):
    coeffs = []
    for u, v in pairs:
        z = h_ref[u] + h_ref[v]
        a = np.zeros(n, dtype=np.float64)
        np.add.at(a, u, 1.0 - 0.5 * z)
        np.add.at(a, v, -0.5 * z)
        b = float(np.sum(z * z)) * 0.25
        coeffs.append((a, b))
    return coeffs


def solve_subproblem(n, S, dx, pairs, h_ref, coeffs):
    try:
        import cvxpy as cp
    except ImportError:
        return None
    x = cp.Variable(n)
    t = cp.Variable()
    constraints = [cp.sum(x) == S, x >= 0, x <= 1, t >= 0]
    for (u, v), (a, b) in zip(pairs, coeffs):
        quad = cp.sum_squares(x[u] - x[v]) * 0.25
        expr = dx * (quad + a @ x + b)
        constraints.append(expr <= t)
    prob = cp.Problem(cp.Minimize(t), constraints)
    x.value = h_ref.copy()
    t.value = exact_score(h_ref, n)

        pass

        pass

            prob.solve(solver=solver, **kw)

                xv = np.array(x.value, dtype=np.float64)
                if xv.shape == (n,) and np.all(np.isfinite(xv)):
