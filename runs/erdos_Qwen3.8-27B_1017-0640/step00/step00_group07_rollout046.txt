        G_A[:p] += dx * ((1.0 - h[s:s + p]) + D)
        G_A[s:] += dx * ((-h[:p]) + D)
        G[row, :] = G_A
        row += 1
        
        # Q_B_s
        Q[row] = dx * (np.dot(h[s:s + p], 1.0 - h[:p]) + 0.5 * np.dot(D, D))
        G_B = np.zeros(n)
        G_B[:p] += dx * ((-h[s:s + p]) + D)
        G_B[s:] += dx * ((1.0 - h[:p]) + D)
        G[row, :] = G_B
        row += 1
    
    if not (np.all(np.isfinite(Q)) and np.all(np.isfinite(G))):
        raise ValueError("Non-finite majorizer")
    return Q, G

    return True

    m = n / 2.0
    if (y.ndim != 1 or len(y) != n or not np.all(np.isfinite(y))
            or np.max(np.abs(y)) > 1e12):
        return uniform_vec(n)

    hi = float(np.max(y)) + 1.0

            lo = mid
        else:
            hi = mid

        p[idx] = min(1.0, p[idx] + diff)
