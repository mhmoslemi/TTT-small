def check_feasible(h, n, m):

        scale = b*b + abs(4.0*a*c) + 1.0
        if disc < -1e-12 * scale:
            return roots
        if disc < 0:
            disc = 0.0
        sq = math.sqrt(disc)
        # Stable formula
        if b >= 0:
            r1 = (-b - sq) / (2.0 * a)
            r2 = c / (a * r1) if r1 != 0 else -b / (2.0 * a)
        else:
            r1 = (-b + sq) / (2.0 * a)
            r2 = c / (a * r1) if r1 != 0 else -b / (2.0 * a)
        for r in [r1, r2]:
            if np.isfinite(r) and lo - tol <= r <= hi + tol:
                roots.append(max(lo, min(hi, r)))
    return roots

def line_search(h, i, j, corr, n):
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-15:
        return 0.0
    B = compute_B(h, i, j, n)
    D = corr.copy()
    # A: zero except at specific lags
    # lag 0 -> idx n-1: A=-2
    # lag j-i -> idx (j-i)+(n-1): A=1
    # lag i-j -> idx (i-j)+(n-1): A=1
    special_idx = [n-1, (j-i)+(n-1), (i-j)+(n-1)]
    special_A = [-2.0, 1.0, 1.0]
    affine_idx = [k for k in range(2*n-1) if k not in special_idx]
    
    # Build upper hull of affine lines
    if len(affine_idx) > 0:
        slopes = B[affine_idx]
        intercepts = D[affine_idx]
        # Sort by slope, keep max intercept for same slope
        order = np.lexsort((intercepts, slopes))
        # Deduplicate slopes
        hull_slopes = []
        hull_intercepts = []
        hull_indices = []
        for o in order:
            ai = affine_idx[o]
            s = slopes[o]
            ci = intercepts[o]
            if hull_slopes and abs(s - hull_slopes[-1]) < 1e-15:
                if ci > hull_intercepts[-1]:
                    # Replace
                    hull_intercepts[-1] = ci
                    hull_indices[-1] = ai
            else:
                hull_slopes.append(s)
                hull_intercepts.append(ci)
                hull_indices.append(ai)
        
        # Build upper hull
        hs = []
        hc = []
        for s, c in zip(hull_slopes, hull_intercepts):
            while len(hs) >= 2:
                # Intersection of hs[-2],hc[-2] with hs[-1],hc[-1]
                # and hs[-1],hc[-1] with s,c
                x1 = (hc[-2] - hc[-1]) / (hs[-1] - hs[-2]) if abs(hs[-1]-hs[-2])>1e-15 else float('inf')
                x2 = (hc[-1] - c) / (s - hs[-1]) if abs(s-hs[-1])>1e-15 else float('inf')
                if x2 <= x1 + 1e-15:
                    hs.pop()
                    hc.pop()
                else:
                    break
            hs.append(s)
            hc.append(c)
        
        # Get hull segments
        segments = []
        if len(hs) == 1:
            segments.append((lo, hi, hs[0], hc[0]))
        else:
            for k in range(len(hs) - 1):
                x_cross = (hc[k] - hc[k+1]) / (hs[k+1] - hs[k]) if abs(hs[k+1]-hs[k])>1e-15 else 0.0
                seg_lo = max(lo, x_cross) if k > 0 else lo
                if k > 0:
                    prev_cross = (hc[k-1] - hc[k]) / (hs[k] - hs[k-1]) if abs(hs[k]-hs[k-1])>1e-15 else lo
                    seg_lo = max(lo, prev_cross)
                seg_hi = (hc[k] - hc[k+1]) / (hs[k+1]-hs[k]) if abs(hs[k+1]-hs[k])>1e-15 else hi
                seg_hi = min(hi, seg_hi)
                if k == 0:
                    seg_lo = lo
                if seg_lo < seg_hi:
                    segments.append((seg_lo, seg_hi, hs[k], hc[k]))
            # Last segment
            seg_lo = (hc[-2] - hc[-1]) / (hs[-1]-hs[-2]) if abs(hs[-1]-hs[-2])>1e-15 else lo
            seg_lo = max(lo, seg_lo)
            if seg_lo < hi:
                segments.append((seg_lo, hi, hs[-1], hc[-1]))
    else:
        segments = [(lo, hi, 0.0, 0.0)]  # No affine lines, dummy
    
    # Special quadratics
    quad_coefs = []
    for si, sa in zip(special_idx, special_A):
        quad_coefs.append((sa, B[si], D[si]))
    
    best_val = float('inf')
    best_delta = 0.0
    
    for (s_lo, s_hi, line_b, line_c) in segments:
        # Local polynomials: line + 3 quadratics
        # Represent as (A, B_coef, D_coef)
        polys = [(0.0, line_b, line_c)]
        for (sa, sb, sd) in quad_coefs:
            polys.append((sa, sb, sd))
        
        candidates = [s_lo, s_hi]
        # Stationary points
        for (a, b_c, d_c) in polys:
            if abs(a) > 1e-15:
                vs = -b_c / (2.0 * a)
                if s_lo - 1e-12 <= vs <= s_hi + 1e-12:
                    candidates.append(max(s_lo, min(s_hi, vs)))
        # Pairwise intersections
        for p in range(len(polys)):
            for q2 in range(p+1, len(polys)):
                a1, b1, d1 = polys[p]
                a2, b2, d2 = polys[q2]
                roots = quadratic_roots(a1, b1, d1, a2, b2, d2, s_lo, s_hi)
                candidates.extend(roots)
        
        for cx in candidates:
            cx = max(s_lo, min(s_hi, cx))
            vals = []
            for (a, b_c, d_c) in polys:
                vals.append(a*cx*cx + b_c*cx + d_c)
            v = max(vals)
            if v < best_val - 1e-14 * max(1.0, abs(best_val)):
                best_val = v
                best_delta = cx
            elif abs(v - best_val) <= 1e-14 * max(1.0, abs(best_val)):
                if abs(cx) < abs(best_delta):
                    best_delta = cx
    
    if not np.isfinite(best_delta):
        return 0.0
    # Check if improvement is meaningful
    current_max = float(np.max(corr))
    if (current_max - best_val) * (2.0/n) <= 1e-12:
        return 0.0
    return best_delta
