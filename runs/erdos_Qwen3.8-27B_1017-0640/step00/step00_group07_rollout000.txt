import numpy as np
from scipy.optimize import minimize
from functools import partial
import math

def project_feasible(y, n):
    if not np.all(np.isfinite(y)) or len(y) != n:
        return np.full(n, 0.5)
    y = np.asarray(y, dtype=np.float64).ravel()
    m = n / 2.0
    lo, hi = np.min(y) - 1.0, np.max(y)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(y - mid, 0.0, 1.0)) > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    resid = m - np.sum(h)
    tol = 2e-14 * max(1, n)
    for _ in range(n + 2):
        if abs(resid) < tol:
            break
        slacks = (1.0 - h) if resid > 0 else h.copy()
        idx = int(np.argmax(slacks))
        if slacks[idx] <= 0:
            break
        step = min(slacks[idx], abs(resid))
        h[idx] += step if resid > 0 else -step
        resid -= step if resid > 0 else -step
    if (not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12)
            or abs(np.sum(h) - m) > 1e-10 * max(1, n)):
        return np.full(n, 0.5)
    return np.clip(h, 0.0, 1.0)

def exact_score(h):
    n = len(h)
    if n < 1 or not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def corr_q_G(h):
    n = len(h)
    dx = 2.0 / n
    q = np.empty(2 * n - 1)
    G = np.zeros((2 * n - 1, n))
    q[0] = dx * np.dot(h, 1.0 - h)
    G[0] = dx * (1.0 - 2.0 * h)
    for s in range(1, n):
        ns = n - s
        a, b = h[:ns], h[s:]
        q[2 * s - 1] = dx * np.dot(a, 1.0 - b)
        G[2 * s - 1, :ns] += dx * (1.0 - b)
        G[2 * s - 1, s:] -= dx * a
        q[2 * s] = dx * np.dot(b, 1.0 - a)
        G[2 * s, s:] += dx * (1.0 - a)
        G[2 * s, :ns] -= dx * b
    return q, G

def _obj(z, n):
    return z[n]

def _obj_grad(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def _eq(z, n):
    return np.sum(z[:n]) - n / 2.0

def _eq_jac(z, n):
    J = np.zeros(1 + n + 1)
    J[:n] = 1.0
    return J

def _ineq(z, n):
    h = z[:n]
    q, _ = corr_q_G(h)
    return z[n] - q

def _ineq_jac(z, n):
    h = z[:n]
    _, G = corr_q_G(h)
    J = np.zeros((2 * n - 1, n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J

def _get_parent():
    v = globals().get('initial_h_values', None)
    if v is None:
        return None
    v = np.asarray(v, dtype=np.float64).ravel()
    if v.size < 2 or v.size > 999:
        return None
    return v

def run(seed=42, **kwargs):
    parent = _get_parent()
    if parent is not None:
        n = int(parent.size)
    else:
        n = 80
    m = n / 2.0

    n_starts = 6
    maxiter = 250
    ftol = 1e-10
    if 'n_starts' in kwargs and isinstance(kwargs['n_starts'], (int, float)):
        n_starts = int(np.clip(kwargs['n_starts'], 1, 8))
    if 'maxiter' in kwargs and isinstance(kwargs['maxiter'], (int, float)):
        maxiter = int(np.clip(kwargs['maxiter'], 50, 400))
    if 'ftol' in kwargs and isinstance(kwargs['ftol'], (int, float)):
        ftol = float(np.clip(kwargs['ftol'], 1e-12, 1e-6))

    if parent is not None and np.all(np.isfinite(parent)):
        base = project_feasible(parent, n)
        parent_fallback = project_feasible(parent, n)
        parent_valid = (np.all(parent >= -1e-12) and np.all(parent <= 1.0 + 1e-12)
                        and abs(np.sum(parent) - m) <= 1e-8 * max(1, n))
    else:
        base = np.full(n, 0.5)
        parent_fallback = base.copy()
        parent_valid = False

    rng = np.random.default_rng(seed)
    starts = [base.copy()]
    starts.append(project_feasible(base + 0.03 * rng.standard_normal(n), n))
    starts.append(project_feasible(base + 0.12 * rng.standard_normal(n), n))
    starts.append(project_feasible(rng.uniform(0, 1, n), n))
    K = min(6, (n - 1) // 2)
    fourier = np.zeros(n)
    idx = np.arange(n, dtype=np.float64) + 0.5
    for k in range(1, K + 1):
        ak = 0.12 / math.sqrt(k) * rng.standard_normal()
        bk = 0.12 / math.sqrt(k) * rng.standard_normal()
        fourier += ak * np.cos(2.0 * np.pi * k * idx / n) + bk * np.sin(2.0 * np.pi * k * idx / n)
    starts.append(project_feasible(fourier, n))
    perm = rng.permutation(n)
    binary = np.zeros(n)
    np.floor(n / 2) if n % 2 == 0 else None
    cnt = n // 2
    binary[perm[:cnt]] = 1.0
    if n % 2 == 1:
        binary[perm[cnt]] = 0.5
    starts.append(project_feasible(binary, n))
    while len(starts) < n_starts:

    starts = starts[:n_starts]

    best_score = float('inf')
    for s in starts:

            best_score = sc
            incumbent = s.copy()

            best_score = ps
            incumbent = parent_fallback.copy()

        for s in starts:
            q0, _ = corr_q_G(s)
            t0 = max(0.0, float(np.max(q0))) + 1e-8
            z0 = np.zeros(n + 1)
            z0[:n] = s
            z0[n] = t0
            cons = [
                {'type': 'eq', 'fun': partial(_eq, n=n), 'jac': partial(_eq_jac, n=n)},
                {'type': 'ineq', 'fun': partial(_ineq, n=n), 'jac': partial(_ineq_jac, n=n)},

                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'ftol': ftol, 'maxiter': maxiter, 'disp': False})
                if res is not None and hasattr(res, 'x') and len(res.x) == n + 1:

                    if cs < best_score - 1e-12:
                        best_score = cs
                        incumbent = cand.copy()
            except Exception:
                pass
            if best_score <= 0.3808:
                break

    incumbent = project_feasible(incumbent, n)
    best_score = exact_score(incumbent)
    if not np.isfinite(best_score):
        incumbent = np.full(n, 0.5)
        best_score = exact_score(incumbent)
    return incumbent, float(best_score), int(n)
