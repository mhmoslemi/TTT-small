import numpy as np

def project_v(v, n):
    v = np.clip(np.asarray(v, dtype=np.float64), -1e6, 1e6)
    target = n / 2.0
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v))
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    q = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    r = target - np.sum(q)
    for i in range(n):
        if abs(r) < 1e-13:
            break
        if r > 0:
            a = min(r, 1.0 - q[i]); q[i] += a; r -= a
        else:
            a = min(-r, q[i]); q[i] -= a; r += a
    if not np.all(np.isfinite(q)) or abs(np.sum(q) - target) > 1e-9:
        return np.full(n, 0.5)
    return np.clip(q, 0.0, 1.0)

def validate_h(h, n):
    return (h.ndim == 1 and len(h) == n and np.all(np.isfinite(h))
            and np.all(h >= 0) and np.all(h <= 1)
            and abs(np.sum(h) - n / 2.0) <= 1e-10)

def exact_score(h):
    n = len(h)
    if n < 2 or not validate_h(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if c.shape[0] != 2 * n - 1 or not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

def compute_lags(h, n, need_jac=False):
    dx = 2.0 / n
    m = 2 * n - 1
    vals = np.empty(m)
    J = np.zeros((m, n)) if need_jac else None
    idx = 0
    for d in range(n - 1, 0, -1):
        L = n - d
        vals[idx] = np.sum(h[:L] * (1.0 - h[d:])) * dx
        if need_jac:
            J[idx, :L] += (1.0 - h[d:]) * dx
            J[idx, d:] -= h[:L] * dx
        idx += 1
    for d in range(n):
        L = n - d
        vals[idx] = np.sum(h[d:] * (1.0 - h[:L])) * dx
        if need_jac:
            J[idx, d:] += (1.0 - h[:L]) * dx
            J[idx, :L] -= h[d:] * dx
        idx += 1
    return (vals, J) if need_jac else vals

def logsumexp_grad(c, tau):
    cmax = np.max(c)
    q = np.exp((c - cmax) / tau)
    sq = np.sum(q)
    F = cmax + tau * np.log(sq)
    return F, q / sq

def pg_cont_search(start, n, temps, max_iter):
    h = project_v(np.array(start, dtype=np.float64), n)
    for tau in temps:
        alpha = 0.5
        stall = 0
        for _ in range(max_iter):
            c, J = compute_lags(h, n, True)
            F, p = logsumexp_grad(c, tau)
            g = J.T @ p
            if not np.all(np.isfinite(g)):
                break
            accepted = False
            for _ in range(24):
                ht = project_v(h - alpha * g, n)
                ct = compute_lags(ht, n, False)
                Ft, _ = logsumexp_grad(ct, tau)
                d = ht - h
                armijo = F + 1e-4 * np.dot(g, d) + 1e-12 * max(1.0, abs(F))
                if Ft <= armijo:
                    h = ht
                    alpha = min(1.5 * alpha, 2.0)
                    accepted = True
                    if np.max(np.abs(d)) < 1e-8 or (F - Ft) <= 1e-12 * max(1.0, abs(F)):
                        stall += 1
                    else:
                        stall = 0
                    break
                alpha *= 0.5
            if not accepted or stall >= 8:
                break
    return h

        starts.append(project_v(0.5 + 0.15 * rng.standard_normal(n), n))
        starts.append(project_v(0.5 + 0.35 * rng.standard_normal(n), n))
    starts.append(np.full(n, 0.5))
    unique = []
    for s in starts:
        if not any(np.max(np.abs(s - u)) < 1e-10 for u in unique):
            unique.append(s)
    cap = 5 if n <= 128 else 4
    return unique[:cap]

    if n > 256:
        return incumbent, inc_score, n
    temps = [0.02, 0.005, 0.001, 0.0002, 0.00004, 0.000008]
    max_iter = 100 if n <= 128 else 40
    rng = np.random.default_rng(seed)
    pv = parent_valid
    starts = make_starts(n, seed, pv)
    for s in starts:
        try:
            res = pg_cont_search(s, n, temps, max_iter)
            sc = exact_score(res)
            if sc < inc_score - 1e-12:
                inc_score = sc
                incumbent = np.array(res)
        except Exception:
            pass
    if n <= 128:
        try:
            shake = project_v(incumbent + 0.04 * rng.standard_normal(n), n)
            if all(np.max(np.abs(shake - s)) > 1e-10 for s in starts):
                res = pg_cont_search(shake, n, temps, max_iter)
                sc = exact_score(res)
                if sc < inc_score - 1e-12:
                    inc_score = sc
                    incumbent = np.array(res)
        except Exception:
            pass
    if not validate_h(incumbent, n):
        incumbent = project_v(incumbent, n)
    fs = exact_score(incumbent)
    if fb_score < fs - 1e-12:
        incumbent = np.array(fallback)
        fs = fb_score
    return incumbent, fs, n
