        else:
            break
    x = np.clip(x, 0.0, 1.0)
    if not (np.all(np.isfinite(x)) and abs(float(np.sum(x)) - m) < 1e-8):
        return None
    return x

def feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and abs(float(np.sum(h)) - n / 2.0) < 1e-9)

def score(h, n):
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    if len(raw) != 2 * n - 1 or not np.all(np.isfinite(raw)):
        return None, None
    q = raw * dx
    return q, float(np.max(q))

def grad_coord(h, r, n):
    dx = 2.0 / n
    g = np.zeros(2 * n - 1)
    g[n - 1] = dx * (1.0 - 2.0 * h[r])
    for p in range(1, n):
        if r >= p:
            g[n - 1 + p] += dx * (1.0 - h[r - p])
            g[n - 1 - p] -= dx * h[r - p]
        if r + p < n:
            g[n - 1 + p] -= dx * h[r + p]
            g[n - 1 - p] += dx * (1.0 - h[r + p])
    return g

def line_min(c, b, A, L, U):
    nl = len(c)
    if U - L < 1e-14:
        return L
    special = [i for i in range(nl) if abs(A[i]) > 1e-14]
    aff = [i for i in range(nl) if abs(A[i]) <= 1e-14]
    cands = {L, U, max(L, min(U, 0.0))}
    for k in special:
        ts = -b[k] / (2.0 * A[k])
        if L - 1e-12 <= ts <= U + 1e-12:
            cands.add(float(np.clip(ts, L, U)))
    for i in range(len(special)):
        for j in range(i + 1, len(special)):
            ki, kj = special[i], special[j]
            da, db, dc = A[ki] - A[kj], b[ki] - b[kj], c[ki] - c[kj]
            if abs(da) < 1e-14:
                if abs(db) > 1e-14:
                    t = -dc / db
                    if L - 1e-12 <= t <= U + 1e-12:
                        cands.add(float(np.clip(t, L, U)))
            else:
                disc = db * db - 4.0 * da * dc
                if disc >= -1e-14:
                    disc = max(0.0, disc); sq = sqrt(disc)
                    for s in (1.0, -1.0):
                        t = (-db + s * sq) / (2.0 * da)
                        if L - 1e-12 <= t <= U + 1e-12:
                            cands.add(float(np.clip(t, L, U)))
    # Build upper envelope of affines, add breakpoints
    if len(aff) > 1:
        sl = sorted(aff, key=lambda i: (b[i], -c[i]))
        # Dedup slopes keeping max intercept
        ded = [sl[0]]
        for i in range(1, len(sl)):
            if abs(b[sl[i]] - b[sl[i - 1]]) < 1e-14:
                if c[sl[i]] > c[ded[-1]]:
                    ded[-1] = sl[i]
            else:
                ded.append(sl[i])
        # Upper hull with breakpoints
        hull = [ded[0]]
        breaks = [-np.inf]
        for idx in ded[1:]:
            sp, cp = b[hull[-1]], c[hull[-1]]
            sn, cn = b[idx], c[idx]
            if abs(sn - sp) < 1e-14:
                continue
            t_int = (cp - cn) / (sn - sp)
            while len(hull) > 1 and t_int <= breaks[-1] + 1e-14:
                hull.pop(); breaks.pop()
                sp, cp = b[hull[-1]], c[hull[-1]]
                t_int = (cp - cn) / (sn - sp)
            hull.append(idx); breaks.append(t_int)
        for bk in breaks[1:]:
            if L - 1e-12 <= bk <= U + 1e-12:
                cands.add(float(np.clip(bk, L, U)))
    # Quadratic-affine intersections (use hull members if available, else all affines)
    aff_check = aff
    if len(aff) > 1:
        pass  # use all affines for safety
    for k in special:
        for m in aff_check:
            da, db, dc = A[k], b[k] - b[m], c[k] - c[m]
            disc = db * db - 4.0 * da * dc
            if disc >= -1e-14:
                disc = max(0.0, disc); sq = sqrt(disc)
                for s in (1.0, -1.0):
                    t = (-db + s * sq) / (2.0 * da)
                    if L - 1e-12 <= t <= U + 1e-12:
                        cands.add(float(np.clip(t, L, U)))
    best_t, best_v = 0.0, np.inf
    for t in cands:
        t = float(np.clip(t, L, U))
        v = float(np.max(c + b * t + A * t * t))
        if v < best_v - 1e-14:
            best_v, best_t = v, t
        elif abs(v - best_v) < 1e-14 and abs(t) < abs(best_t) - 1e-14:
            best_t = t
    return best_t

def interp_to_n(src, n):
    m = len(src)
    if m == 1:
        return np.full(n, float(src[0]))
    src_pos = np.linspace(0.0, 1.0, m)
    tgt_pos = np.linspace(0.0, 1.0, n)
    return np.interp(tgt_pos, src_pos, src)

def make_starts(n, rng, has_parent, parent_vec):
    m = n / 2.0
    starts = []
    if has_parent:
        starts.append(parent_vec.copy())
    # Block
    blk = np.zeros(n); bf = int(m); frac = m - bf
    blk[:bf] = 1.0
    if frac > 1e-12:
        blk[bf] = frac
    starts.append(blk.copy())
    starts.append(blk[::-1].copy())
    if has_parent:
        starts.append((1.0 - parent_vec).copy())
        starts.append(np.roll(parent_vec, n // 3).copy())
        # Perturbed parent
        starts.append(parent_vec + rng.normal(0, 0.08, size=n))
    # Random
    starts.append(rng.uniform(0, 1, size=n))
    # Cosine
    ph = rng.uniform(0, 2 * pi)
    idx = np.arange(n)
    starts.append(np.clip(0.5 + 0.35 * np.cos(2 * pi * idx / n + ph), 0, 1))
    if not has_parent:
        starts.append(rng.uniform(0, 1, size=n))
        ph2 = rng.uniform(0, 2 * pi)
        starts.append(np.clip(0.5 + 0.35 * np.cos(4 * pi * idx / n + ph2), 0, 1))
    # Project and deduplicate
    valid = []
    seen = []
    for s in starts:
        p = project_capped(s, n)
        if p is None:
            continue
        dup = False
        for v in seen:
            if float(np.max(np.abs(p - v))) < 1e-10:
                dup = True; break
        if not dup:
            valid.append(p); seen.append(p)
    # Limit
    if n <= 64:
        lim = 6
    elif n <= 160:
        lim = 4
    else:
        lim = 3
    if len(valid) > lim:
        # Keep parent-derived (first), block (second), and at least one random/cosine
        keep = valid[:2] + valid[-1:lim - 1]
        if len(keep) > lim:
            keep = keep[:lim]
        valid = keep
    return valid if valid else [np.full(n, 0.5)]

def pair_sweep(h, q, cur_score, n, rng, n_pairs, rng2):
    dx = 2.0 / n
    improved = False
    sweep_start_score = cur_score
    pairs = []
    if n_pairs >= n * (n - 1) // 2:
        for a in range(n - 1):
            for b in range(a + 1, n):
                pairs.append((a, b))
    else:
        # Stratified by separation
        seps = []
        s = 1
        while s < n:
            seps.append(s); s *= 2
        per_sep = max(1, n_pairs // len(seps))
        seen_p = set()
        for s in seps:
            if s >= n:
                continue
            cnt = 0
            stride = max(1, (n - s) // per_sep) if per_sep > 1 else 1
            for a in range(0, n - s, stride):
                b = a + s
                if (a, b) not in seen_p:
                    seen_p.add((a, b)); pairs.append((a, b)); cnt += 1
                    if cnt >= per_sep:
                        break
        # Fill remaining
        while len(pairs) < n_pairs:
            a = int(rng2.integers(0, n - 1))
            b = int(rng2.integers(a + 1, n))
            if (a, b) not in seen_p:
                seen_p.add((a, b)); pairs.append((a, b))
    rng2.shuffle(pairs)
    for a, b in pairs:
        L = max(-h[a], h[b] - 1.0)
        U = min(1.0 - h[a], h[b])
        if U - L < 1e-13:
            continue
        ga = grad_coord(h, a, n)
        gb = grad_coord(h, b, n)
        bv = ga - gb
        cv = q.copy()
        Av = np.zeros(2 * n - 1)
        p_sep = b - a
        Av[n - 1] = -2.0 * dx
        Av[n - 1 + p_sep] = dx
        Av[n - 1 - p_sep] = dx
        if not (np.all(np.isfinite(bv)) and np.all(np.isfinite(cv))):
            continue
        t = line_min(cv, bv, Av, L, U)
        if abs(t) < 1e-13:
            continue
        trial = h.copy()
        trial[a] += t
        trial[b] -= t
        if not np.all(np.isfinite(trial)) or trial[a] < -1e-12 or trial[a] > 1.0 + 1e-12 or trial[b] < -1e-12 or trial[b] > 1.0 + 1e-12:
            continue
        if not feasible(trial, n):
            trial = project_capped(trial, n)
            if trial is None or not feasible(trial, n):
                continue
        q_new, s_new = score(trial, n)
        if q_new is None or s_new is None:
            continue
        if s_new < cur_score - 1e-12:
            h = trial; q = q_new; cur_score = s_new; improved = True
    return h, q, cur_score, improved, sweep_start_score

def run(seed=42, **kwargs):
    n = 45
    # Check kwargs for n_points
    if 'n_points' in kwargs:
        nv = kwargs['n_points']
        if isinstance(nv, (int, np.integer)) and not isinstance(nv, bool) and 1 <= nv <= 999:
            n = int(nv)
    # Get parent
    has_parent = False
    parent_vec = None
    try:
        pv = initial_h_values
        if pv is not None:
            pv = np.asarray(pv, dtype=np.float64)
            if pv.ndim == 1 and len(pv) > 0 and len(pv) <= 999 and np.all(np.isfinite(pv)):
                has_parent = True
    except (NameError, Exception):
        pass
    if has_parent:
        if len(pv) == n:
            parent_vec = pv.copy()
        else:
            parent_vec = interp_to_n(pv, n)
        parent_vec = project_capped(parent_vec, n)
        if parent_vec is None:
            parent_vec = np.full(n, 0.5)
    if not feasible(parent_vec, n):
        parent_vec = np.full(n, 0.5)
        parent_fallback = parent_vec.copy()
    else:
        parent_fallback = parent_vec.copy()
    rng = np.random.default_rng(seed)
    rng2 = np.random.default_rng(seed + 1)
    # Determine sweep params
    n_pairs = n * (n - 1) // 2 if n <= 64 else (96 if n <= 160 else 32)
    n_sweeps = 3 if n <= 64 else 2
    starts = make_starts(n, rng, has_parent, parent_vec)
    best_h = parent_fallback.copy()
    best_score = None
    _, bs = score(best_h, n)
    if bs is not None:
        best_score = bs
    else:
        best_score = 1.0
    for st in starts:
        h = st.copy()
        _, h_score = score(h, n)
        if h_score is None:
            continue
        q = None
        if h_score is not None:
            dx_t = 2.0 / n
            q = np.correlate(h, 1.0 - h, mode='full') * dx_t
        cur_score = h_score
        if cur_score < best_score - 1e-12:
            best_h = h.copy(); best_score = cur_score
        for sw in range(n_sweeps):
            h, q, cur_score, imp, ss = pair_sweep(h, q, cur_score, n, rng, n_pairs, rng2)
            if cur_score < best_score - 1e-12:
                best_h = h.copy(); best_score = cur_score
            if not imp or ss - cur_score < 1e-10:
                break
    # Shake
    if n <= 64:
        shake_rounds = [(0.05, 1), (0.18, 1)]
    elif n <= 160:
        shake_rounds = [(0.10, 1)]
    else:
        shake_rounds = [(0.15, 1)]
    for sd, nsh in shake_rounds:
        shake_start = best_score
        h = best_h + rng.normal(0, sd, size=n)
        h = project_capped(h, n)
        if h is None or not feasible(h, n):
            continue
        _, hs = score(h, n)
        if hs is None:
            continue
        q = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        if hs < best_score - 1e-12:
            best_h = h.copy(); best_score = hs
        for sw in range(max(1, n_sweeps - 1)):
            h, q, hs, imp, ss = pair_sweep(h, q, hs, n, rng, n_pairs, rng2)
            if hs < best_score - 1e-12:
                best_h = h.copy(); best_score = hs
            if not imp:
                break
        if best_score >= shake_start - 1e-9:
            break
    # Final validation
    if not feasible(best_h, n):
        best_h = project_capped(best_h, n)
    if best_h is None or not feasible(best_h, n):
        best_h = parent_fallback.copy()
    _, final_score = score(best_h, n)
    if final_score is None:
        best_h = np.full(n, 0.5)
        _, final_score = score(best_h, n)
    return best_h.astype(np.float64), float(final_score), int(n)
