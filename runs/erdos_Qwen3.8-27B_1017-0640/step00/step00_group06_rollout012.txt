import numpy as np
from scipy.optimize import linprog
import math

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    q = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(q)):
        return float('inf')
    return float(np.max(q) * (2.0 / n))

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-15) or np.any(h > 1.0 + 1e-15):
        return False
    return abs(math.fsum(h) - n / 2.0) <= 1e-9

def project_bounded_simplex(v, l, u, m):
    n = len(v)
    if not np.all(np.isfinite(v)) or not np.all(np.isfinite(l)) or not np.all(np.isfinite(u)):
        return None
    sl = math.fsum(l)
    su = math.fsum(u)
    if sl - 1e-12 > m or su + 1e-12 < m:
        return None
    lam_lo = float(np.min(v - u))
    lam_hi = float(np.max(v - l))
    x = np.clip(v - lam_hi, l, u).copy()
    for _ in range(120):
        x = np.clip(v - lam_hi, l, u)
        mass = math.fsum(x)
        if abs(mass - m) <= 1e-13:
            break

        else:
            lam_hi = lam_hi + (lab_hh(lam_lo, lam_hi)) if False else lam_lo

        lam_lo = lam_lo if mass > m else lam_lo
        mid = (lam_lo + lam_hi) / 2.0
        if lam_hi - lam_lo < 1e-15 * (1.0 + abs(mid)):
            lam_hi = mid
            x = np.clip(v - mid, l, u)
            break
        x = np.clip(v - mid, l, u)
        mass = math.fsum(x)
        if mass > m:
            lam_lo = mid
        else:
            lam_hi = mid
    x = np.clip(v - lam_hi, l, u).copy()

                    r += sub
        if abs(m - math.fsum(x)) <= 1e-12:
            break
    if not np.all(np.isfinite(x)) or np.any(x < l - 1e-12) or np.any(x > u + 1e-12):
        return None
    return x

def lab_hh(a, b):
    return (a + b) / 2.0

def lag_data(h, n):
    q = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(q)):
        return None, None
    nlag = 2 * n - 1
    G = np.zeros((nlag, n))
    for a in range(nlag):
        k = a - (n - 1)
        jlo = max(0, -k)
        jhi = min(n, n - k)
        for j in range(jlo, jhi):
            G[a, j + k] += (1.0 - h[j])
            G[a, j] -= h[j + k]
    if not np.all(np.isfinite(G)):
        return None, None
    return q, G

def lp_step(h, delta, q, G, n, m):
    l = np.maximum(0.0, h - delta)
    u = np.minimum(1.0, h + delta)
    nlag = 2 * n - 1
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((nlag, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h - q
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([m])
    bounds = [(l[i], u[i]) for i in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': 500})
    except Exception:
        return None
    if res is None or res.status != 0 or not all(np.isfinite(res.x[:n])):
        return None
    x = project_bounded_simplex(res.x[:n], l, u, m)
    if x is None:
        return None
    return x

def half_block(n):
    h = np.zeros(n, dtype=np.float64)
    h[:n // 2] = 1.0
    if n % 2 == 1:
        h[n // 2] = 0.5
    return h

def build_starts(base, n, m, rng):

def optimize_start(h0, n, m, budget, delta0=0.12, delta_max=0.25, delta_min=1e-5):
