            {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}]
    for h0 in starts:
        s0 = exact_score(h0)
        if s0 < best_s:
            best_s, best_h = s0, h0.copy()
        t0 = min(1.0, s0 + 1e-8)
        z0 = np.concatenate([h0, [t0]])
        try:
            res = minimize(slsqp_fun, z0, method='SLSQP', jac=slsqp_jac,
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-10})
            if hasattr(res, 'x') and res.x is not None:
                xc = np.asarray(res.x, dtype=np.float64)
                if xc.shape == (n + 1,) and np.all(np.isfinite(xc)):
                    hp = project_feasible(xc[:n], n)
                    if hp is not None and is_valid(hp, n):
                        sc = exact_score(hp)
                        if sc < best_s:
                            best_s, best_h = sc, hp
        except Exception:
            pass
    return best_h, best_s

def run(seed=42, **kwargs):
    try:
        seed = int(seed)
    except (ValueError, TypeError):
        seed = 42
    try:
        n_starts = max(4, min(10, int(kwargs.get('n_starts', 8))))
    except (ValueError, TypeError):
        n_starts = 8
    try:
        maxiter = max(100, min(400, int(kwargs.get('maxiter', 250))))
    except (ValueError, TypeError):
        maxiter = 250
    rng = np.random.default_rng(seed)

    parent = None
    n0 = None
    try:
        raw = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if raw.size >= 2 and np.all(np.isfinite(raw)):
            n0 = raw.size
            parent = project_feasible(raw, n0)
    except (NameError, Exception):
        pass

    if n0 is None:
        n0 = 42
    if parent is None:
        parent = make_block(n0)
        parent = project_feasible(parent, n0)
        if parent is None:
            parent = np.full(n0, 0.5)

    fallback_h = parent.copy()
    fallback_s = exact_score(fallback_h)
    global_best_h = fallback_h.copy()
    global_best_s = fallback_s

    if n0 <= 256:
        anchor = parent
        raws = [anchor, anchor[::-1].copy(), 1.0 - anchor, make_block(n0), np.full(n0, 0.5)]
        starts = dedup_starts(raws, n0)
        starts = fill_random(starts, n0, anchor, rng, n_starts)
        bh, bs = solve_resolution(n0, starts, maxiter)
        if bh is not None and bs < global_best_s - 1e-12:
            global_best_h, global_best_s = bh, bs
        elif bh is not None and abs(bs - global_best_s) <= 1e-12 and n0 < len(global_best_h):
            global_best_h, global_best_s = bh, bs
    else:
        if is_valid(parent, n0):
            global_best_h, global_best_s = parent.copy(), fallback_s

    n1 = 2 * n0
    if n0 <= 128 and n1 <= 256:
        anchor1 = project_feasible(np.repeat(parent, 2), n1)
        if anchor1 is None:
            anchor1 = make_block(n1)
        if len(global_best_h) == n0 and global_best_h is not fallback_h:
            b1 = project_feasible(np.repeat(global_best_h, 2), n1)
            if b1 is not None:
                anchor1 = b1
        raws1 = [anchor1, anchor1[::-1].copy(), 1.0 - anchor1, make_block(n1), np.full(n1, 0.5)]
        starts1 = dedup_starts(raws1, n1)
        starts1 = fill_random(starts1, n1, anchor1, rng, n_starts)
        bh1, bs1 = solve_resolution(n1, starts1, maxiter)
        if bh1 is not None and bs1 < global_best_s - 1e-12:
            global_best_h, global_best_s = bh1, bs1
        elif bh1 is not None and abs(bs1 - global_best_s) <= 1e-12 and n1 < len(global_best_h):
            global_best_h, global_best_s = bh1, bs1

    if not is_valid(global_best_h, len(global_best_h)):
        global_best_h = fallback_h
    global_best_s = exact_score(global_best_h)
    return (global_best_h, float(global_best_s), int(len(global_best_h)))
