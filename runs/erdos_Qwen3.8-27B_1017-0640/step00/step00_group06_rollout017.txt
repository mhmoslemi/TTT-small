            if nk is not None and isinstance(nk, (int, np.integer)) and 2 <= int(nk) < 1000:
                n = int(nk)
        except Exception:
            pass
    if n is None:
        n = 95
    dx = 2.0 / n
    m = n / 2.0
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)

    base = None
    fallback = None
    try:
        p = np.asarray(initial_h_values, dtype=np.float64)
        if p.ndim == 1 and p.shape[0] == n and np.all(np.isfinite(p)):
            if check_feasibility(p, n, m):
                base = p.copy()
                fallback = p.copy()
            else:
                base = project_bounded_simplex(p, np.zeros(n), np.ones(n), m)
                if base is None:
                    base = half_block(n)
    except Exception:
        pass
    if base is None:
        base = half_block(n)
    if fallback is None:
        fallback = base.copy()

    starts = [base.copy()]
    for sigma in (0.05, 0.15, 0.30):
        z = rng.standard_normal(n)
        proj = project_bounded_simplex(base + sigma * z, np.zeros(n), np.ones(n), m)
        if proj is not None:
            starts.append(proj)
    bv = rng.beta(0.7, 0.7, n)
    proj = project_bounded_simplex(bv, np.zeros(n), np.ones(n), m)
    if proj is not None:
        starts.append(proj)
    ph = rng.uniform(0, 2 * np.pi)
    idx = np.arange(n, dtype=np.float64)
    sv = 0.5 + 0.45 * np.sin(2.0 * np.pi * idx / n + ph)
    proj = project_bounded_simplex(sv, np.zeros(n), np.ones(n), m)
    if proj is not None:
        starts.append(proj)
    hb = half_block(n)
    if not np.allclose(hb, base, atol=1e-12):
        starts.append(hb)

    unique = []
    for s in starts:
        if not any(np.max(np.abs(s - u)) < 1e-12 for u in unique):
            unique.append(s)
    starts = unique

    best_h = fallback.copy()
    best_c = exact_score(best_h, n)
    for s in starts:
        if check_feasibility(s, n, m):
            c = exact_score(s, n)
            if c < best_c - 1e-12:
                best_c = c
                best_h = s.copy()

    d0, dmax, dmin = 0.12, 0.25, 1e-5
    B0 = max(8, min(28, math.ceil(28.0 * 95.0 / n)))
    B1 = max(4, min(12, math.ceil(12.0 * 95.0 / n)))

    for i, s in enumerate(starts):
        if not check_feasibility(s, n, m):
            continue
        b = B0 if i == 0 else B1
        ho, co = optimize_start(s, n, m, dx, d0, dmax, dmin, b)
        if check_feasibility(ho, n, m):
            cf = exact_score(ho, n)
            if cf < best_c - 1e-12:
                best_c = cf
                best_h = ho.copy()

    if not check_feasibility(best_h, n, m):
        proj = project_bounded_simplex(best_h, np.zeros(n), np.ones(n), m)
        if proj is not None and check_feasibility(proj, n, m):
            best_h = proj
        else:
            best_h = fallback.copy()
    c5 = exact_score(best_h, n)
    return best_h, c5, int(n)
