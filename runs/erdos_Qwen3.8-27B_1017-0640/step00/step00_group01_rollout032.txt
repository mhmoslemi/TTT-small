        parent = np.full(n, 0.5)
        psc = exact_score(parent, n)
    if 17 <= n <= 64:
        P, mg, sl, mr = 48, 240, 40, 3
    elif n <= 16:
        P, mg, sl, mr = 20, 100, 20, 2
    elif n <= 128:
        P, mg, sl, mr = 32, 140, 28, 2
    elif n <= 256:
        P, mg, sl, mr = 20, 70, 18, 1
    elif n <= 512:
        P, mg, sl, mr = 12, 35, 12, 1
    else:
        P, mg, sl, mr = 8, 12, 6, 0
    pop = build_population(rng, parent, n, P)
    try:
        h_best, c5_best = jade_search(rng, pop, n, mg, sl, mr)
    except Exception:
        h_best, c5_best = parent.copy(), psc
    if h_best is None or not check_feasible(h_best, n):
        proj = project_capped_simplex(h_best if h_best is not None else parent, s)
        h_best = proj if proj is not None and check_feasible(proj, n) else parent.copy()
    c5_final = exact_score(h_best, n)
    if c5_final is None:
        h_best, c5_final = parent.copy(), psc
    if c5_final > psc:
        h_best, c5_final = parent.copy(), psc
    return h_best.astype(np.float64), float(c5_final), int(n)
