    add_to_list(members, n, base[::-1].copy())
    add_to_list(members, n, 1.0 - base)
    add_to_list(members, n, (1.0 - base)[::-1].copy())
    add_to_list(members, n, np.full(n, 0.5))
    add_to_list(members, n, np.roll(base, 1))
    add_to_list(members, n, np.roll(base, max(1, n // 2)))
    i = np.arange(n) + 0.5
    for f in [1, 2, 3, 4]:
        phi = rng.uniform(0, 2 * math.pi)
        add_to_list(members, n, 0.5 + 0.35 * np.sin(2 * math.pi * f * i / n + phi))
    sigmas = [0.04, 0.10, 0.22, 0.45, 0.70]
    fills = 0
    while len(members) < P and fills < 30 * P:
        fills += 1
        m = fills % 3
        if m == 0:
            sig = sigmas[fills % len(sigmas)]
            cand = base + sig * rng.standard_normal(n)
        elif m == 1:
            cand = rng.random(n)
        else:
            cand = rng.beta(0.7, 0.7, n)
        add_to_list(members, n, cand)
    while len(members) < P:
        members.append(base.copy())
    return np.array(members[:P])

def run(seed=42, **kwargs):
    try:
        s = int(seed)
        if s < 0 or s > 2**32 - 1:
            s = 42
    except (ValueError, TypeError):
        s = 42
    rng = np.random.default_rng(s)
    n = 47
    parent = None
    iv = globals().get('initial_h_values')
    if iv is not None:
        iv = np.asarray(iv, dtype=np.float64).ravel()
        if 2 <= len(iv) < 1000 and np.all(np.isfinite(iv)) and np.min(iv) >= -1e-7 and np.max(iv) <= 1.0 + 1e-7:
            n = len(iv)
            p = project_capped_simplex(iv)
            if p is not None:
                parent = p
    if parent is None:
        parent = np.full(47, 0.5)
        n = 47
    if n <= 16:
        P, G, SL, MR = 20, 100, 20, 2
    elif n <= 64:
        P, G, SL, MR = 48, 240, 40, 3
    elif n <= 128:
        P, G, SL, MR = 32, 140, 28, 2
    elif n <= 256:
        P, G, SL, MR = 20, 70, 18, 1
    elif n <= 512:
        P, G, SL, MR = 12, 35, 12, 1
    else:
        P, G, SL, MR = 8, 12, 6, 0
    pop = build_pop(rng, n, parent, P)
    scores = np.full(P, np.inf)
    for i in range(P):
        sv = exact_score(pop[i], n)
        if sv is not None:
            scores[i] = sv
    inc = parent.copy()
    inc_score = exact_score(inc, n)
    if inc_score is None:
        inc_score = float('inf')
    bi = int(np.argmin(scores))
    if scores[bi] < inc_score:
        inc_score = scores[bi]
        inc = pop[bi].copy()
    mu_F, mu_CR = 0.5, 0.8
    restarts = 0
    stall = 0
    archive = []
    k = max(2, math.ceil(0.20 * P))
    for gen in range(G):
        gen_b = inc_score
        snap = pop.copy()
        snap_sc = scores.copy()
        order = rng.permutation(P)
        rank_idx = np.argsort(snap_sc)
        npop = snap.copy()
        nsc = snap_sc.copy()
        pend = []
        sF, sCR = [], []
        for i in order:
            pool = rank_idx[:k]
            cp = pool[pool != i]
            p = int(rng.choice(cp)) if len(cp) > 0 else int(rng.choice(rank_idx[:k]))
            cr1 = np.arange(P)[(np.arange(P) != i) & (np.arange(P) != p)]
            r1 = int(rng.choice(cr1))
            cr2 = cr1[cr1 != r1]
            r2 = int(rng.choice(cr2))
            F = 0.5
            for _ in range(50):
                F = mu_F + 0.1 * rng.standard_cauchy()
                if math.isfinite(F) and F > 0:
                    F = min(max(F, 0.001), 1.0)
                    break
            CR = mu_CR + 0.1 * rng.standard_normal()
            CR = min(max(CR, 0.0), 1.0)
            if not math.isfinite(CR):
                CR = 0.8
            donor = snap[i] + F * (snap[p] - snap[i]) + F * (snap[r1] - snap[r2])
            mask = rng.random(n) < CR
            mask[int(rng.integers(n))] = True
            trial = np.where(mask, donor, snap[i])
            proj = project_capped_simplex(trial)
            if proj is None or not np.all(np.isfinite(proj)):
                continue
            if np.max(np.abs(proj - snap[i])) <= 1e-15:
                continue
            ts = exact_score(proj, n)
            if ts is None:
                continue
            better = ts < snap_sc[i] or (ts == snap_sc[i] and np.max(np.abs(proj - snap[i])) > 1e-12)
            if better:
                npop[i] = proj
                nsc[i] = ts
                if ts < snap_sc[i]:
                    pend.append(snap[i].copy())
                    sF.append(F)
                    sCR.append(CR)
                if ts < inc_score:
                    inc_score = ts
                    inc = proj.copy()
        pop = npop
        scores = nsc
        archive.extend(pend)
        if len(archive) > P:
            keep = sorted(rng.choice(len(archive), size=P, replace=False))
            archive = [archive[j] for j in keep]
        if len(sF) > 0:
            sf2 = sum(f * f for f in sF)
            sf = sum(sF)
            if sf > 0:
                mu_F = 0.9 * mu_F + 0.1 * (sf2 / sf)
            mu_CR = 0.9 * mu_CR + 0.1 * (sum(sCR) / len(sCR))
        mu_F = min(max(mu_F, 0.05), 1.0)
        mu_CR = min(max(mu_CR, 0.0), 1.0)
        if inc_score < gen_b:
            stall = 0
        else:
            stall += 1
        if stall >= SL and restarts < MR:
            restarts += 1
            stall = 0
            mu_F, mu_CR = 0.5, 0.8
            archive = []
            idx_worst = np.argsort(scores)[::-1][:max(2, P // 3)]
            pop[idx_worst[0]] = inc.copy()
            rr = 0
            for ix in idx_worst[1:]:
                rr += 1
                if rr % 4 == 0:
                    cand = rng.beta(0.7, 0.7, n)
                else:
                    sig = [0.05, 0.15, 0.35, 0.70][rr % 4]
                    cand = inc + sig * rng.standard_normal(n)
                pj = project_capped_simplex(cand)
                if pj is not None:
                    pop[ix] = pj
                    sv = exact_score(pj, n)
                    if sv is not None:
                        scores[ix] = sv
                        if sv < inc_score:
                            inc_score = sv
                            inc = pj.copy()
                break
    # Final validation
    fin = project_capped_simplex(inc)
    if fin is None:
        fin = parent.copy()
    fs = exact_score(fin, n)
    ps = exact_score(parent, n)
    if ps is not None and (fs is None or ps < fs):
        fin = parent.copy()
        fs = ps
    if fs is None:
        fin = np.full(n, 0.5)
        fs = exact_score(fin, n)
    return fin, float(fs), int(n)
