def project_capped(v, S):
    v = np.array(v, dtype=np.float64).copy()
    n = len(v)
    if n < 1 or not np.all(np.isfinite(v)):
        return None
    v = np.clip(v, -1e6, 1e6)
    lo, hi = np.min(v) - 1.0, np.max(v)
    mid = (lo + hi) / 2.0
    for _ in range(80):
        x = np.clip(v - mid, 0.0, 1.0)
        if np.sum(x) > S:
            lo = mid
        else:
            hi = mid
        nm = (lo + hi) / 2.0
        if nm == mid:
            break
        mid = nm
    x = np.clip(v - mid, 0.0, 1.0)
    for _ in range(5):
        r = S - np.sum(x)
        if abs(r) <= 1e-12:
            break
        mask = (x < 1.0 - 1e-15) if r > 0 else (x > 1e-15)
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] = np.clip(x[mask] + r / cnt, 0.0, 1.0)
    x = np.clip(x, 0.0, 1.0)
    if not (np.all(x >= 0) and np.all(x <= 1) and abs(np.sum(x) - S) <= 1e-10):
        return None
    return x

def is_feasible(h, n):
    return (h is not None and h.shape == (n,) and np.all(np.isfinite(h))
            and np.all(h >= 0.0) and np.all(h <= 1.0)
            and abs(np.sum(h) - n / 2.0) <= 1e-10)

def exact_score(h, n):
    if not is_feasible(h, n):
        return None
    q = np.correlate(h, 1.0 - h, mode='full')
    s = float(np.max(q)) * (2.0 / n)
    return s if np.isfinite(s) else None

def smooth_grad(h, tau, n):
    dx = 2.0 / n
    q = np.correlate(h, 1.0 - h, mode='full')
    c = q * dx
    cmax = np.max(c)
    u = (c - cmax) / tau
    e = np.exp(u)
    se = np.sum(e)
    F = cmax + tau * np.log(se)
    w = e / se
    g = np.zeros(n)
    g += w[n - 1] * (1.0 - 2.0 * h)
    for s in range(1, n):
        m = n - s
        g[s:] += w[n - 1 + s] * (1.0 - h[:m])
        g[:m] -= w[n - 1 + s] * h[s:]
        g[:m] += w[n - 1 - s] * (1.0 - h[s:])
        g[s:] -= w[n - 1 - s] * h[:m]
    g *= dx
    if not np.all(np.isfinite(g)) or not np.isfinite(F):
        return None, None
    return F, g
