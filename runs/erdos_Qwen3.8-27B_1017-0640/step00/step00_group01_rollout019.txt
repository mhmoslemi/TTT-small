import numpy as np

def project_capped_simplex(v):
    if v is None or len(v) < 2:
        return None
    v = np.asarray(v, dtype=np.float64).ravel()
    if v.ndim != 1 or not np.all(np.isfinite(v)):
        return None
    n = len(v)
    if np.max(np.abs(v)) > 1e6:
        return None
    s = n / 2.0
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = (lo + hi) / 2.0
        mass = math.fsum(np.clip(v - mid, 0.0, 1.0))
        if mass > s:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(3):
        r = s - math.fsum(x)
        if abs(r) <= 5e-15:
            break
        if r > 0:
            for i in range(n):
                if r <= 5e-15:
                    break
                add = min(r, 1.0 - x[i])
                if add > 0:
                    x[i] += add
                    r -= add
        else:
            neg_r = -r
            for i in range(n):
                if neg_r <= 5e-15:
                    break
                rem = min(neg_r, x[i])
                if rem > 0:
                    x[i] -= rem
                    neg_r -= rem

        return None

def lag_values(h, n):
    dx = 2.0 / n
    vals = np.empty(2 * n - 1, dtype=np.float64)
    for m in range(2 * n - 1):
        d = m - (n - 1)
        start = max(0, -d)
        stop = min(n, n - d)
        if start < stop:
            vals[m] = np.dot(h[start:stop], 1.0 - h[start + d:stop + d])
        else:
            vals[m] = 0.0
    return vals * dx


def smooth_value_gradient(h, tau, n, need_grad=True):
    a = lag_values(h, n)
    if not np.all(np.isfinite(a)):
        return None, None
    A = float(np.max(a))
    e = np.exp((a - A) / tau)
    Z = float(np.sum(e))
    if Z <= 0 or not np.isfinite(Z):
        return None, None
    F = A + tau * math.log(Z)
    if not np.isfinite(F):
        return None, None
    if not need_grad:
        return F, None
    p = e / Z
    dx = 2.0 / n
    g = np.zeros(n, dtype=np.float64)
    for md in range(2 * n - 1):
        d = md - (n - 1)
        start = max(0, -d)
        stop = min(n, n - d)

        w = dx * p[md]
        src = h[start:stop]
        tgt = h[start + d:stop + d]
        g[start:stop] += w * (1.0 - tgt)
        g[start + d:stop + d] -= w * src
    if not np.all(np.isfinite(g)):
        return None, None
    return F, g
