        if i < len(anchors):
            v = anchors[i].copy()
        else:
            j = (i - len(anchors)) % 4
            ns = noise_scales[(i - len(anchors)) % 5]
            if j == 0:
                v = fb + ns * rng.standard_normal(n)
            elif j == 1:
                v = anchors[rng.integers(0, len(anchors))] + ns * rng.standard_normal(n)
            elif j == 2:
                v = rng.uniform(0, 1, n)
            else:
                v = rng.uniform(-0.25, 1.25, n)
        pop[i] = project_simplex(v, n)
        s = score(pop[i])
        sc[i] = s if s is not None else fb_c
        if s is not None and s < best_s - 1e-12:
            best = pop[i].copy()
            best_s = s

    stagn = 0
    n_inj = 0
    inj_scales = [0.04, 0.10, 0.25, 0.40]

    for gen in range(G):
        prev_best = best_s
        snap = pop.copy()
        snap_sc = sc.copy()

        for i in range(NP):
            cands = [j for j in range(NP) if j != i]
            r1, r2, r3 = rng.choice(len(cands), 3, replace=False)
            r1, r2, r3 = cands[r1], cands[r2], cands[r3]
            F = rng.uniform(0.45, 0.90)
            CR = rng.uniform(0.65, 1.0)
            v = snap[r1] + F * (snap[r2] - snap[r3])
            mask = rng.random(n) < CR
            if not mask.any():
                mask[rng.integers(n)] = True
            u = np.where(mask, v, snap[i])
            u = project_simplex(u, n)
            if not feasible(u, n):
                continue
            s = score(u)
            if s is None:
                continue
            if s < best_s - 1e-12:
                best = u.copy()
                best_s = s
            if s < snap_sc[i] - 1e-12:
                pop[i] = u
                sc[i] = s

        if best_s < prev_best - 1e-12:
            stagn = 0
        else:
            stagn += 1

        if stagn >= 20 and n_inj < 3:
            n_inj += 1
            stagn = 0
            order = np.argsort(sc)
            n_repl = max(1, int(np.ceil(0.20 * NP)))
            for k in range(n_repl):
                idx_rep = order[-1 - k]
                ns = inj_scales[k % len(inj_scales)]
                if k % 2 == 0:
                    v = best + ns * rng.standard_normal(n)
                else:
                    v = rng.uniform(-0.25, 1.25, n)
                v = project_simplex(v, n)
                if feasible(v, n):
                    s = score(v)
                    if s is not None:
                        pop[idx_rep] = v
                        sc[idx_rep] = s
                        if s < best_s - 1e-12:
                            best = v.copy()
                            best_s = s
        elif stagn >= 40:
            break

    if feasible(best, n):
        s_final = score(best)
        if s_final is not None and s_final < fb_c - 1e-12:
            return best.copy(), s_final, n
    s_fb = score(fb)
    if s_fb is None:
        s_fb = fb_c
    return fb.copy(), s_fb, n
