def lag_values_and_jacobian(h):
    n = len(h)
    dx = 2.0 / n
    q = np.empty(2 * n - 1, dtype=np.float64)
    G = np.zeros((2 * n - 1, n), dtype=np.float64)
    q[0] = dx * np.sum(h * (1.0 - h))
    G[0] = dx * (1.0 - 2.0 * h)
    idx = 1
    for s in range(1, n):
        q[idx] = dx * np.sum(h[:n - s] * (1.0 - h[s:]))
        G[idx, :n - s] += dx * (1.0 - h[s:])
        G[idx, s:] -= dx * h[:n - s]
        q[idx + 1] = dx * np.sum(h[s:] * (1.0 - h[:n - s]))
        G[idx + 1, s:] += dx * (1.0 - h[:n - s])
        G[idx + 1, :n - s] -= dx * h[s:]
        idx += 2
    return q, G


def smooth_value(q, tau):
    qmax = float(np.max(q))
    return qmax + tau * float(np.log(np.sum(np.exp((q - qmax) / tau))))


def smooth_grad(q, G, tau):
    qmax = float(np.max(q))
    e = np.exp((q - qmax) / tau)
    w = e / float(np.sum(e))
    return w @ G


def run(seed=42, **kwargs):
    global initial_h_values
    n = 80
    base = np.full(n, 0.5)
    try:
        parent = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 2 <= len(parent) <= 999 and np.all(np.isfinite(parent)):
            n = len(parent)
            base = project_capped(parent, n)
    except Exception:
        pass
    n_starts = 7
    max_iter = 80
    try:
        v = int(kwargs.get('n_starts', 7))
        if 1 <= v <= 8:
            n_starts = v
    except Exception:
        pass
    try:
        v = int(kwargs.get('max_iter', 80))
        if 20 <= v <= 200:
            max_iter = v
    except Exception:
        pass
    try:
        s = int(seed)
        if s < 0:
            s = 0
    except Exception:
        s = 42
    rng = np.random.default_rng(s)
    best = base.copy()
    best_score = exact_score(best)
    starts = [base.copy()]
    for sd in (0.02, 0.08, 0.20):
        starts.append(base + rng.standard_normal(n) * sd)
    starts.append(rng.uniform(0.0, 1.0, n))
    K = min(6, (n - 1) // 2)
    y = np.full(n, 0.5)
    for k in range(1, K + 1):
        tk = 2.0 * np.pi * k * (np.arange(n) + 0.5) / n
        y += (0.12 / np.sqrt(k)) * (rng.standard_normal() * np.cos(tk) + rng.standard_normal() * np.sin(tk))
    starts.append(y)
    bv = np.zeros(n)
    bv[:n // 2] = 1.0
    if n % 2 == 1:
        bv[n // 2] = 0.5
    bv = bv[rng.permutation(n)]
    starts.append(bv)
    starts.append(base + rng.standard_normal(n) * 0.4)
    proj_starts = []
    for st in starts[:n_starts]:
        p = project_capped(st, n)
        dup = False
        for ex in proj_starts:
            if float(np.max(np.abs(p - ex))) < 1e-10:
                dup = True
                break
        if not dup:
            proj_starts.append(p)
    temps = [0.02, 0.005, 0.00125, 0.0003125, 0.000078125, 0.00001953125, 0.0000048828125]
    if n <= 256:
        for x in proj_starts:
            if best_score <= 0.3808:
                break
            for tau in temps:
                if best_score <= 0.3808:
                    break
                alpha = 1.0
                for it in range(max_iter):
                    q, G = lag_values_and_jacobian(x)
                    if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):
                        break
                    phi = smooth_value(q, tau)
                    g = smooth_grad(q, G, tau)
                    if not np.all(np.isfinite(g)):
                        break
                    accepted = False
                    for bt in range(25):
                        xtrial = project_capped(x - alpha * g, n)
                        d = xtrial - x
                        if float(np.max(np.abs(d))) <= 1e-9:
                            break
                        gd = float(np.sum(g * d))
                        if gd >= 0.0:
                            break
                        q2 = lag_values(xtrial)
                        phi2 = smooth_value(q2, tau)
                        if not np.isfinite(phi2):
                            break
                        if phi2 <= phi + 1e-4 * gd + 1e-14 * max(1.0, abs(phi)):
                            x = xtrial
                            alpha = min(alpha * 1.25, 4.0)
                            accepted = True
                            sc = exact_score(x)
                            if sc < best_score - 1e-12:
                                best = x.copy()
                                best_score = sc
                            break
                        alpha *= 0.5
                    if not accepted:
                        break
    best = project_capped(best, n)
    final_score = exact_score(best)
    if not np.isfinite(final_score):
        best = project_capped(base, n)
        final_score = exact_score(best)
    if not np.isfinite(final_score):
        best = np.full(n, 0.5)
        final_score = exact_score(best)
    return (best, float(final_score), int(n))
