    return np.sum(z[:-1]) - m

def eq_jac(z, m):
    g = np.zeros(len(z))
    g[:-1] = 1.0
    return g

def ineq_fun(z, dx):
    h = z[:-1]
    return z[-1] - np.correlate(h, 1.0 - h, mode='full') * dx

def ineq_jac(z, dx):
    h = z[:-1]
    N = len(h)
    J = np.zeros((2 * N - 1, N))
    for k_idx in range(2 * N - 1):
        k = k_idx - (N - 1)
        r1a, r1b = max(0, -k), min(N - 1, N - 1 - k)
        r2a, r2b = max(0, k), min(N - 1, N - 1 + k)
        if r1a <= r1b:
            idx = np.arange(r1a, r1b + 1)
            J[k_idx, idx] += 1.0 - h[idx + k]
        if r2a <= r2b:
            idx = np.arange(r2a, r2b + 1)
            J[k_idx, idx] -= h[idx - k]
    J *= dx
    out = np.zeros((2 * N - 1, N + 1))
    out[:, :N] = -J
    out[:, N] = 1.0
    return out

def try_solve(h0, N, dx, m, best_score, best_h):
    t0 = min(max(score_h(h0, N), 0.0), 2.0)
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * N + [(0.0, 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (m,)},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac, 'args': (dx,)},
    ]
    try:
        res = minimize(objective_fn, z0, method='SLSQP', jac=objective_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 250, 'ftol': 1e-10, 'disp': False})
    except Exception:
        return best_score, best_h
    h_out = res.x[:N]
    if not np.all(np.isfinite(h_out)):
        return best_score, best_h
    h_proj = project_box_sum(h_out, m, N)
    s = score_h(h_proj, N)
    if s < best_score - 1e-12:
        return s, h_proj
    return best_score, best_h

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    h_init = None
    try:
        hv = np.asarray(initial_h_values, dtype=np.float64)
        if hv.ndim == 1 and len(hv) > 0 and np.all(np.isfinite(hv)):
            h_init = hv.copy()
    except Exception:
        pass
    if h_init is not None:
        N = len(h_init)
    else:
        N = 70
    dx = 2.0 / N
    m = N / 2.0
    h_base = np.full(N, 0.5)
    if h_init is not None:
        h_p = project_box_sum(h_init, m, N)
    else:
        h_p = h_base.copy()
    best_score = score_h(h_base, N)
    best_h = h_base.copy()
    s_p = score_h(h_p, N)
    if s_p < best_score:
        best_score, best_h = s_p, h_p
    if N > 999:
        return best_h, best_score, N
    starts = [h_p.copy(), h_base.copy()]
    if N <= 256:
        for scale in [0.03, 0.08, 0.18]:
            starts.append(project_box_sum(h_p + scale * rng.standard_normal(N), m, N))
        starts.append(project_box_sum(rng.uniform(0, 1, N), m, N))
        starts.append(project_box_sum(rng.beta(0.7, 0.7, N), m, N))
        i = np.arange(N) + 0.5
        starts.append(project_box_sum(0.5 + 0.45 * np.cos(2.0 * np.pi * i / N), m, N))
        starts.append(project_box_sum(0.5 + 0.45 * np.cos(4.0 * np.pi * i / N + np.pi / 3.0), m, N))
    else:
        starts.append(project_box_sum(h_p + 0.08 * rng.standard_normal(N), m, N))
        starts.append(project_box_sum(rng.uniform(0, 1, N), m, N))
        starts.append(project_box_sum(rng.beta(0.7, 0.7, N), m, N))
    for s in starts:
        s_score = score_h(s, N)
        if s_score < best_score - 1e-12:
            best_score, best_h = s_score, s
        best_score, best_h = try_solve(s, N, dx, m, best_score, best_h)
    h_final = project_box_sum(best_h, m, N)
    final_score = score_h(h_final, N)
    if (not np.all(np.isfinite(h_final)) or np.any(h_final < -1e-9)
            or np.any(h_final > 1.0 + 1e-9)):
        h_final = h_base
        final_score = score_h(h_final, N)
    h_final = np.clip(h_final, 0.0, 1.0)
    return h_final, final_score, N
