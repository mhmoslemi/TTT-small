        return None

    lo = np.min(v) - 1.0
    hi = np.max(v)

        x = np.clip(v - mid, 0.0, 1.0)
        mass = math.fsum(x)
        if mass > s:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(v - lam, 0.0, 1.0)
