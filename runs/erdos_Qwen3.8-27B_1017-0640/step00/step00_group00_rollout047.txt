import numpy as np
import math

def exact_score(h):
    n = len(h)
    if n == 0:
        return None
    raw = np.correlate(h, 1.0 - h, mode='full')
    if len(raw) != 2 * n - 1:
        return None
    if not np.all(np.isfinite(raw)):
        return None
    q = raw * (2.0 / n)
    c = float(np.max(q))
    return c if np.isfinite(c) else None

def check_feasible(h, n, tol=1e-10):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -tol) or np.any(h > 1.0 + tol):
        return False
    if abs(float(np.sum(h)) - n / 2.0) > tol:
        return False
    return True

def project_capped(y, n):
    m = n / 2.0
    y = np.asarray(y, dtype=np.float64)
    if y.shape != (n,) or not np.all(np.isfinite(y)):
        return None
    y = np.clip(y, -1e6, 1e6)
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > m:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    resid = m - float(np.sum(x))
    for _ in range(10):
        if abs(resid) < 1e-13:
            break
        if resid > 0:
            for i in range(n):
                add = min(1.0 - x[i], resid)
                if add > 0:
                    x[i] += add
                    resid -= add
                    if abs(resid) < 1e-13:
                        break
        else:
            for i in range(n):
                sub = min(x[i], -resid)
                if sub > 0:
                    x[i] -= sub
                    resid += sub
                    if abs(resid) < 1e-13:
                        break
    if not check_feasible(x, n):
        return None
    return x

def grad_matrix(h, dx):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    G[n - 1, :] = dx * (1.0 - 2.0 * h)
    for p in range(1, n):
        G[n - 1 + p, p:] += dx * (1.0 - h[:n - p])
        G[n - 1 + p, :n - p] -= dx * h[p:]
        G[n - 1 - p, :n - p] += dx * (1.0 - h[p:])
        G[n - 1 - p, p:] -= dx * h[:n - p]
    return G

def solve_cvxpy(n, x, q, G, dx):
    try:
        import cvxpy as cp
    except ImportError:
        return None
    L = 2 * n - 1
    d = cp.Variable(n)
    s = cp.Variable(non_neg=True)
    t = cp.Variable()
    qp = cp.Parameter(L)
    Gp = cp.Parameter((L, n))
    lb = cp.Parameter(n)
    ub = cp.Parameter(n)
    qp.value = q
    Gp.value = G
    lb.value = -x
    ub.value = 1.0 - x
    cons = [d >= lb, d <= ub, cp.sum(d) == 0, cp.sum_squares(d) <= s, qp + Gp @ d + dx * s <= t]
    prob = cp.Problem(cp.Minimize(t), cons)
    for solver in ['ECOS', 'SCS', 'CVXOPT', 'CLARABEL']:
        try:
            kw = {'max_iters': 500, 'verbose': False}
            try:
                prob.solve(solver=solver, abstol=1e-9, reltol=1e-9, feastol=1e-9, max_iters=500, verbose=False)
            except Exception:
                prob.solve(solver=solver, **kw)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                dv = np.array(d.value, dtype=np.float64).flatten()
                if dv.shape == (n,) and np.all(np.isfinite(dv)):
                    return dv
        except Exception:
            continue
    return None

def solve_slsqp(n, x, q, G, dx):
    from scipy.optimize import minimize as spm
    L = 2 * n - 1
    z0 = np.concatenate([np.zeros(n), [0.0, float(np.max(q))]])
    bounds = [(-x[i], 1.0 - x[i]) for i in range(n)] + [(0.0, float(n)), (None, None)]
    def f_obj(z):
        return z[n + 1]
    def f_eq(z):
        return np.sum(z[:n])
    def j_eq(z):
        return np.concatenate([np.ones(n), [0.0, 0.0]])
    def f_inq1(z):
        return z[n] - float(np.dot(z[:n], z[:n]))
    def j_inq1(z):
        return np.concatenate([-2.0 * z[:n], [1.0, 0.0]])
    def f_inq2(z):
        return z[n + 1] - q - G @ z[:n] - dx * z[n]
    def j_inq2(z):
        J = np.zeros((L, n + 2))
        J[:, :n] = -G
        J[:, n] = -dx
        J[:, n + 1] = 1.0
        return J
    cons = [{'type': 'eq', 'fun': f_eq, 'jac': j_eq},
            {'type': 'ineq', 'fun': f_inq1, 'jac': j_inq1},
            {'type': 'ineq', 'fun': f_inq2, 'jac': j_inq2}]
    maxit = 120 if n <= 64 else 80 if n <= 160 else 40
    try:
        res = spm(f_obj, z0, method='SLSQP', bounds=bounds, constraints=cons,
                  options={'maxiter': maxit, 'ftol': 1e-10, 'disp': False})
        if res.x is not None:
            dv = res.x[:n]
            if np.all(np.isfinite(dv)):
                return dv
    except Exception:
        pass
    return None

def mm_optimize(n, x0, max_iters):
    dx = 2.0 / n
    x = x0.copy()
    c_cur = exact_score(x)
    if c_cur is None:
        return x0, None
    best_x = x.copy()
    best_c = c_cur
    prev_c = c_cur
    for it in range(max_iters):
        q = np.correlate(x, 1.0 - x, mode='full') * dx
        G = grad_matrix(x, dx)
        if not np.all(np.isfinite(G)):
            break
        dval = solve_cvxpy(n, x, q, G, dx)
        if dval is None:
            dval = solve_slsqp(n, x, q, G, dx)
        if dval is None:
            break
        y_raw = x + dval
        y = project_capped(y_raw, n)
        if y is None:
            break
        best_sc = c_cur
        best_pt = None
        for th in (1.0, 0.5, 0.25, 0.125, 0.0625):
            pt = project_capped(x + th * (y - x), n)
            if pt is None:
                continue
            sc = exact_score(pt)
            if sc is not None and sc < best_sc - 1e-12:
                best_sc = sc
                best_pt = pt
        if best_pt is None:
            break
        x = best_pt
        c_cur = best_sc
        if c_cur < best_c - 1e-12:
            best_c = c_cur
            best_x = x.copy()
        if prev_c - c_cur < 1e-11:
            break
        prev_c = c_cur
    return best_x, best_c
