    B = 0.0

def line_search_candidates(a, b, c, lo, hi, n_lags):
    # a, b, c are arrays of length n_lags
    # P_d(t) = a[d]*t^2 + b[d]*t + c[d]
    cands = [lo, hi, 0.0 if lo <= 0.0 <= hi else (lo if lo > 0 else hi)]
    cands = [max(lo, min(hi, x)) for x in cands]
    
    # Pairwise intersections
    for d1 in range(n_lags):
        for d2 in range(d1+1, n_lags):
            A = a[d1] - a[d2]
            B = b[d1] - b[d2]
            C = c[d1] - c[d2]
            tol = 1e-14 * max(1.0, abs(A), abs(B), abs(C))
            if abs(A) < tol:
                # Linear: B*t + C = 0
                if abs(B) > tol:
                    t = -C / B
                    if lo - 1e-10 <= t <= hi + 1e-10:
                        cands.append(max(lo, min(hi, t)))
            else:
                disc = B*B - 4.0*A*C
                neg_tol = -1e-14 * max(1.0, B*B, abs(4.0*A*C))
                if disc > neg_tol:
                    disc = max(0.0, disc)
                    sqrt_disc = math.sqrt(disc)
                    R = -0.5 * (B + math.copysign(sqrt_disc, B))
                    if abs(R) > tol:
                        r1 = R / A
                        r2 = C / R
                    else:
                        r1 = -B / (2.0 * A)
                        r2 = r1
                    for r in [r1, r2]:
                        if np.isfinite(r) and lo - 1e-10 <= r <= hi + 1e-10:
                            cands.append(max(lo, min(hi, r)))
    
    # Vertices of convex (a_d > 0) polynomials
    for d in range(n_lags):
        if a[d] > 1e-14:
            tv = -b[d] / (2.0 * a[d])
            if np.isfinite(tv) and lo - 1e-10 <= tv <= hi + 1e-10:
                cands.append(max(lo, min(hi, tv)))
    
    # Sort and deduplicate
    cands.sort()
    unique = [cands[0]]
    for x in cands[1:]:
        if x - unique[-1] > 1e-12:
            unique.append(x)
    return unique

    return (best_h.astype(np.float64), float(c5), int(n))
