import numpy as np
import math

def project_K(y, m):
    n = len(y)
    y = np.clip(y, -1e6, 1e6)
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > m:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    res = np.clip(y - lam, 0.0, 1.0)
    for _ in range(10):
        r = m - float(np.sum(res))
        if abs(r) < 1e-13:
            break
        if r > 0:
            slack = 1.0 - res
            idx = int(np.argmax(slack))
            if slack[idx] <= 0:
                break
            res[idx] += min(r, slack[idx])
        else:
            idx = int(np.argmax(res))
            if res[idx] <= 0:
                break
            res[idx] -= min(-r, res[idx])
    res = np.clip(res, 0.0, 1.0)
    if not np.all(np.isfinite(res)) or abs(float(np.sum(res)) - m) > 1e-9:
        return np.full(n, 0.5)
    return res

def exact_score(h, n):
    dx = 2.0 / n
    val = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
    return val if math.isfinite(val) else float('inf')

def corr_vec(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def smooth_obj(q, tau):
    Q = float(np.max(q))
    e = np.exp((q - Q) / tau)
    s = float(np.sum(e))
    if s <= 0.0 or not math.isfinite(s):
        return float('inf'), None
    return Q + tau * math.log(s), e / s

def wgrad(w, h, n):
    dx = 2.0 / n
    g = dx * w[n - 1] * (1.0 - 2.0 * h)
    for p in range(1, n):
        wp = w[n - 1 + p]
        wm = w[n - 1 - p]
        if wp > 1e-15:
            g[p:] += dx * wp * (1.0 - h[:-p])
            g[:-p] -= dx * wp * h[p:]
        if wm > 1e-15:
            g[:-p] += dx * wm * (1.0 - h[p:])
            g[p:] -= dx * wm * h[:-p]
    return g
