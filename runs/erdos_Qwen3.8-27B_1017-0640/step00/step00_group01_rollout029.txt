import numpy as np

def project_capped(v, n):
    s = n / 2.0
    v = np.asarray(v, dtype=np.float64).ravel()
    if v.shape[0] != n or not np.all(np.isfinite(v)) or np.max(np.abs(v)) > 1e6:
        return None
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v))
    for _ in range(80):
        mid = lo + (hi - lo) * 0.5
        if math.fsum(np.clip(v - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    for _ in range(3):
        r = s - math.fsum(x)
        if abs(r) <= 5e-15:
            break
        for i in range(n):
            if abs(r) <= 5e-15:
                break
            if r > 0:
                add = min(r, 1.0 - x[i])
                if add > 0:
                    x[i] += add
                    r -= add
            else:
                sub = min(-r, x[i])
                if sub > 0:
                    x[i] -= sub
                    r += sub
    if not np.all(np.isfinite(x)) or np.min(x) < -1e-14 or np.max(x) > 1 + 1e-14:
        return None
    if abs(math.fsum(x) - s) > 1e-10:
        return None
    return np.clip(x, 0.0, 1.0)


def is_feasible(h, n):
    if h.ndim != 1 or h.shape[0] != n:
        return False
    if not np.all(np.isfinite(h)) or np.min(h) < -1e-12 or np.max(h) > 1 + 1e-12:
        return False
    return abs(math.fsum(h) - n / 2.0) <= 1e-10


def exact_score(h, n):
    dx = 2.0 / n
    try:
        corr = np.correlate(h, 1.0 - h, mode='full')
        val = float(np.max(corr)) * dx
        return val if math.isfinite(val) else None
    except Exception:
        pass
    mx = 0.0
    for d in range(-(n - 1), n):
        s, e = (0, n - d) if d >= 0 else (-d, n)
        sv = float(np.sum(h[s:e] * (1.0 - h[s + d:e + d])))
        if sv > mx:
            mx = sv
    return mx * dx if math.isfinite(mx) else None


def add_start(starts, v, n):

    if proj is None:
        return
    for s in starts:
        if np.max(np.abs(proj - s)) <= 1e-9:
            return
    starts.append(proj)


def pair_line_search(h, n, i, j):
    dx = 2.0 / n
    nlags = 2 * n - 1
    idx0 = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    a = np.zeros(nlags)
    b = np.zeros(nlags)
    a[idx0] = -2.0 * dx
    a[(j - i) + idx0] = dx
    a[(i - j) + idx0] = dx
    for d in range(-(n - 1), n):
        idx = d + idx0
        bv = 0.0
        if 0 <= i + d < n:
            bv += 1.0 - h[i + d]
        if 0 <= i - d < n:
            bv -= h[i - d]
        if 0 <= j + d < n:
            bv -= 1.0 - h[j + d]
        if 0 <= j - d < n:
            bv += h[j - d]
        b[idx] = bv * dx
    if not (np.all(np.isfinite(a)) and np.all(np.isfinite(b)) and np.all(np.isfinite(c))):
        return None
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if lo >= hi:
        return None
    cands = [lo, hi]
    if lo < 0.0 < hi:
        cands.append(0.0)
    for p in range(nlags):
        for q in range(p + 1, nlags):
            A = a[p] - a[q]
            B = b[p] - b[q]
            C = c[p] - c[q]
            tol = 1e-14 * max(1.0, abs(A), abs(B), abs(C))
            if abs(A) < tol:
                if abs(B) > tol:
                    t = -C / B
                    if lo - 1e-10 <= t <= hi + 1e-10:
                        cands.append(t)
            else:
                disc = B * B - 4.0 * A * C
                dtol = 1e-14 * max(1.0, B * B, abs(4.0 * A * C))
                if disc < -dtol:
                    continue
                disc = max(0.0, disc)
                sq = math.sqrt(disc)
                R = -0.5 * (B + math.copysign(sq, B))
                if abs(R) > tol:
                    for t in (R / A, C / R):
                        if math.isfinite(t) and lo - 1e-10 <= t <= hi + 1e-10:
                            cands.append(t)
                elif abs(B) > tol:
                    t = -B / (2.0 * A)
                    if lo - 1e-10 <= t <= hi + 1e-10:
                        cands.append(t)
    for d in range(nlags):
        if a[d] > 0.0:
            tv = -b[d] / (2.0 * a[d])
            if lo <= tv <= hi:
                cands.append(tv)
    cands = np.unique(np.round(np.clip(np.array(cands), lo, hi), 12))
    t2 = cands * cands
    vals = a[None, :] * t2[:, None] + b[None, :] * cands[:, None] + c[None, :]
    env = np.max(vals, axis=1)
    best_t = float(cands[int(np.argmin(env))])
    prop = h.copy()
    prop[i] = h[i] + best_t
    prop[j] = h[j] - best_t
    prop = project_capped(prop, n)
    if prop is None:
        return None
    sc = exact_score(prop, n)
    if sc is None:
        return None
    return prop, sc


def local_search(h, n, rng, max_passes, pairs_per_pass):
    h = h.copy()
    cur_sc = exact_score(h, n)
    if cur_sc is None:
        return h, 1e10
    best_h, best_sc = h.copy(), cur_sc
    empty = 0
    for p in range(max_passes):
        improved = False
        for _ in range(pairs_per_pass):
            i, j = int(rng.integers(n)), int(rng.integers(n))

            if cur_sc < best_sc:
                best_sc, best_h = cur_sc, h.copy()
            empty = 0
        else:
            empty += 1
            if empty >= 2:
                break
    return best_h, best_sc
