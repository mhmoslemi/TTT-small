def score_h(h):
    n = len(h)
    if n < 2 or not np.all(np.isfinite(h)):
        return np.inf
    c = np.correlate(h, 1.0 - h, mode='full')
    val = float(np.max(c) * (2.0 / n))
    return val if np.isfinite(val) else np.inf


def lag_values_and_jac(h):
    n = len(h)
    dx = 2.0 / n
    c = np.zeros(2 * n - 1)
    J = np.zeros((2 * n - 1, n))
    r = np.arange(n)
    for k in range(-(n - 1), n):
        j = k + n - 1
        rlo = max(0, k)
        rhi = min(n, n + k)
        if rhi > rlo:
            idx = np.arange(rlo, rhi)
            c[j] = np.dot(h[idx], 1.0 - h[idx - k])
        m1 = (r >= k) & (r < n + k)
        if m1.any():
            J[j, m1] += 1.0 - h[r[m1] - k]
        m2 = (r >= -k) & (r < n - k)
        if m2.any():
            J[j, m2] -= h[r[m2] + k]
    return dx * c, dx * J


def slsqp_obj(z):
    return z[-1]


def slsqp_obj_jac(z):
    g = np.zeros(len(z))
    g[-1] = 1.0

def slsqp_eq(z):
    return np.sum(z[:-1]) - (len(z) - 1) / 2.0


def slsqp_eq_jac(z):
    g = np.ones(len(z))
    g[-1] = 0.0
    return g


def slsqp_ineq(z):
    h = z[:-1]
    c, _ = lag_values_and_jac(h)
    return z[-1] - c


def slsqp_ineq_jac(z):

    n = len(h)
    _, J = lag_values_and_jac(h)
    col = np.ones((2 * n - 1, 1))
