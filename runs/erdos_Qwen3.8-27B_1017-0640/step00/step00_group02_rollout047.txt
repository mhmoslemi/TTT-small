    starts.append(np.full(n, 0.5, dtype=np.float64))
    if n <= 128:
        noisy = project_capped(base + rng.normal(0, 0.12, n), S)
        if noisy is not None:
            starts.append(noisy)
        unif = project_capped(rng.uniform(0, 1, n), S)
        if unif is not None:
            starts.append(unif)
        m = int(rng.integers(1, min(5, max(1, n // 2)) + 1))
        phase = rng.uniform(0, 2 * math.pi)
        ii = np.arange(n, dtype=np.float64)
        sinus = project_capped(0.5 + 0.35 * np.sin(2 * math.pi * m * ii / n + phase), S)
        if sinus is not None:
            starts.append(sinus)
    elif n <= 512:
        noisy = project_capped(base + rng.normal(0, 0.10, n), S)
        if noisy is not None:
            starts.append(noisy)
        unif = project_capped(rng.uniform(0, 1, n), S)
        if unif is not None:
            starts.append(unif)
    else:
        unif = project_capped(rng.uniform(0, 1, n), S)
        if unif is not None:
            starts.append(unif)
    dedup = []
    for s in starts:
        if not feasible(s, n, S):
            continue
        dup = False
        for d in dedup:
            if float(np.max(np.abs(s - d))) < 1e-12:
                dup = True
                break
        if not dup:
            dedup.append(s)
    starts = dedup
    scored_starts = []
    for s in starts:
        sc = exact_score(s, n, S)
        if sc is None:
            continue
        scored_starts.append((s, sc))
        if sc < incumbent_score - 1e-12:
            incumbent = s.copy()
            incumbent_score = sc
    if n <= 64:
        L, R, Mcap = 12, 2, 600
    elif n <= 128:
        L, R, Mcap = 10, 1, 300
    elif n <= 512:
        L, R, Mcap = 8, 0, 120
    else:
        L, R, Mcap = 5, 0, 40
    Nchains = max(1, len(scored_starts) + R)
    pilot_count = max(8, min(48, int(1e6 / (n * n))))
    M = max(1, min(Mcap, int(1.8e8 / (n * n * L * Nchains))))
    for s, sc in scored_starts:
        try:
            incumbent, incumbent_score = anneal_chain(s, sc, n, S, rng, L, M, pilot_count, incumbent, incumbent_score)
        except Exception:
            pass
    if n <= 64 and R >= 1:
        rd = [0.10, 0.25]
        for std in rd:
            if R <= 0:
                break
            pert = project_capped(incumbent + rng.normal(0, std, n), S)
            if pert is None:
                continue
            sc = exact_score(pert, n, S)
            if sc is None:
                continue
            if sc < incumbent_score - 1e-12:
                incumbent = pert.copy()
                incumbent_score = sc
            try:
                incumbent, incumbent_score = anneal_chain(pert, sc, n, S, rng, L, M, pilot_count, incumbent, incumbent_score)
            except Exception:
                pass
            R -= 1
    elif n <= 128 and R >= 1:
        pert = project_capped(incumbent + rng.normal(0, 0.15, n), S)
        if pert is not None:
            sc = exact_score(pert, n, S)
            if sc is not None:
                if sc < incumbent_score - 1e-12:
                    incumbent = pert.copy()
                    incumbent_score = sc
                try:
                    incumbent, incumbent_score = anneal_chain(pert, sc, n, S, rng, L, M, pilot_count, incumbent, incumbent_score)
                except Exception:
                    pass
    if not feasible(incumbent, n, S):
        rep = project_capped(incumbent, S)
        if rep is not None and feasible(rep, n, S):
            incumbent = rep
    final_sc = exact_score(incumbent, n, S)
    if final_sc is None:
        incumbent = fallback
        final_sc = fallback_score
    elif final_sc > fallback_score + 1e-10:
        incumbent = fallback
        final_sc = fallback_score
    return incumbent.astype(np.float64), float(final_sc), int(n)
