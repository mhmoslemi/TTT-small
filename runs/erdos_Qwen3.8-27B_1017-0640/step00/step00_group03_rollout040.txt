        mid = lo + (hi - lo) / 2.0
        q = np.clip(v - mid, 0.0, 1.0)
        if float(np.sum(q)) > target:
            lo = mid
        else:
            hi = mid
    q = np.clip(v - (lo + (hi - lo) / 2.0), 0.0, 1.0)
    res = target - float(np.sum(q))
    for _ in range(2):
        if abs(res) < 1e-12:
            break
        for i in range(n):
            if res > 0.0 and q[i] < 1.0:
                add = min(res, 1.0 - q[i])
                q[i] += add
                res -= add
            elif res < 0.0 and q[i] > 0.0:
                sub = min(-res, q[i])
                q[i] -= sub
                res += sub
    q = np.clip(q, 0.0, 1.0)
    if not np.all(np.isfinite(q)) or abs(float(np.sum(q)) - target) > 1e-10:
        return None
    return q

def check_feasible(h, n):
    if h is None:
        return False
    h = np.asarray(h, dtype=float)
    if h.ndim != 1 or len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(float(np.sum(h)) - n / 2.0) > 1e-10:
        return False
    return True

def exact_score(h, n):
    if not check_feasible(h, n):
        return float('inf')
    corr = np.correlate(h, 1.0 - h, mode='full')
    if len(corr) != 2 * n - 1 or not np.all(np.isfinite(corr)):
        return float('inf')
    return float(np.max(corr * (2.0 / n)))

def build_lags(h, n, dx):
    c = np.zeros(2 * n - 1)
    J = np.zeros((2 * n - 1, n))
    for d in range(n):
        u = h[:n - d]
        v = h[d:]
        A_d = float(np.sum(v * (1.0 - u)))
        B_d = float(np.sum(u * (1.0 - v)))
        row_a = np.zeros(n)
        if d < n:
            row_a[d:] = 1.0 - h[:n - d]
        row_a[:n - d] -= h[d:]
        row_b = np.zeros(n)
        row_b[:n - d] = 1.0 - h[d:]
        row_b[d:] -= h[:n - d]
        if d == 0:
            c[n - 1] = A_d * dx
            J[n - 1, :] = row_a * dx
        else:
            c[n - 1 - d] = B_d * dx
            J[n - 1 - d, :] = row_b * dx
            c[n - 1 + d] = A_d * dx
            J[n - 1 + d, :] = row_a * dx
    if not np.all(np.isfinite(c)) or not np.all(np.isfinite(J)):
        return None, None
    return c, J

def lp_step(y, c, J, r, mu, n):
    n_lags = 2 * n - 1
    total = 2 * n + 1
    nx = n
    nu = n
    c_obj = np.zeros(total)
    c_obj[nx:2 * nx] = mu
    c_obj[2 * nx] = 1.0
    n_rows = n_lags + 2 * n
    A_ub = np.zeros((n_rows, total))
    b_ub = np.zeros(n_rows)
    for l in range(n_lags):
        A_ub[l, :nx] = J[l]
        A_ub[l, 2 * nx] = -1.0
        b_ub[l] = c[l] - float(np.dot(J[l], y))
    for i in range(n):
        A_ub[n_lags + 2 * i, i] = 1.0
        A_ub[n_lags + 2 * i, nx + i] = -1.0
        b_ub[n_lags + 2 * i] = y[i]
        A_ub[n_lags + 2 * i + 1, i] = -1.0
        A_ub[n_lags + 2 * i + 1, nx + i] = -1.0
        b_ub[n_lags + 2 * i + 1] = -y[i]
    A_eq = np.zeros((1, total))
    A_eq[0, :nx] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = []
    for i in range(nx):
        bounds.append((max(0.0, y[i] - r), min(1.0, y[i] + r)))
    for i in range(nu):
        bounds.append((0.0, None))
    bounds.append((None, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': 500})
    except Exception:
        return None
    if res is None or res.status != 0 or not np.all(np.isfinite(res.x)):
        return None
    x = res.x[:nx]
    if not np.all(np.isfinite(x)):
        return None
    return x

def local_search(y, n, dx, max_attempts=22):
    cur_y = np.copy(y)
    cur_score = exact_score(cur_y, n)
    if not np.isfinite(cur_score):
        return cur_y, cur_score
    r, r_max, r_min, mu = 0.04, 0.12, 2e-4, 5e-4
    fails = 0
    for _ in range(max_attempts):
        if fails >= 6 or r < r_min:
            break
        c, J = build_lags(cur_y, n, dx)
        if c is None:
            fails += 1
            r = max(r / 2.0, r_min)
            continue
        x_prop = lp_step(cur_y, c, J, r, mu, n)
        if x_prop is None:
            fails += 1
            r = max(r / 2.0, r_min)
            continue
        x_proj = project_onto_Q(x_prop, n)
        if x_proj is None:
            fails += 1
            r = max(r / 2.0, r_min)
            continue
        disp = float(np.max(np.abs(x_proj - cur_y)))
        if disp > r + 1e-6 or disp <= 1e-10:
            fails += 1
            r = max(r / 2.0, r_min)
            continue
        old_y = np.copy(cur_y)
        best_sc = cur_score
        best_y = None
        for j in range(9):
            alpha = 2.0 ** (-j)
            trial = old_y + alpha * (x_proj - old_y)
            tp = project_onto_Q(trial, n)
            if tp is None:
                continue
            td = float(np.max(np.abs(tp - old_y)))
            if td > alpha * r + 1e-6:
                continue
            sc = exact_score(tp, n)
            if np.isfinite(sc) and sc < best_sc - 1e-12:
                best_sc = sc
                best_y = tp
        if best_y is not None:
            M_pred = float(np.max(c + J.dot(best_y - old_y)))
            pred = cur_score - M_pred
            ared = cur_score - best_sc
            rho = ared / pred if pred > 1e-12 else 0.0
            sn = float(np.max(np.abs(best_y - old_y)))
            cur_y = best_y
            cur_score = best_sc
            fails = 0
            if rho < 0.1:
                r = max(r / 2.0, r_min)
            elif rho > 0.75 and sn >= 0.8 * r:
                r = min(1.5 * r, r_max)
            if sn <= 1e-8:
                break
        else:
            fails += 1
            r = max(r / 2.0, r_min)
    return cur_y, cur_score

def run(seed=42, **kwargs):
    n = 70
    parent = globals().get('initial_h_values')
    if parent is not None:
        try:
            p = np.asarray(parent, dtype=float)
            if p.ndim == 1 and p.size >= 2:
                n = p.size
        except Exception:
            pass
    if n == 70:
        nv = kwargs.get('n_points')
        if nv is not None and not isinstance(nv, bool):
            try:
                nv2 = int(float(nv))
                if 2 <= nv2 <= 256 and float(nv) == float(nv2):
                    n = nv2
            except Exception:
                pass
    dx = 2.0 / n
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    uniform = np.full(n, 0.5)
    feasible_parent = None
    raw_parent = None
    if parent is not None:
        try:
            p = np.asarray(parent, dtype=float).flatten()
            if p.size == n and np.all(np.isfinite(p)):
                raw_parent = p.copy()
                if min(p) >= -1e-12 and max(p) <= 1.0 + 1e-12 and abs(float(np.sum(p)) - n / 2.0) <= 1e-10:
                    feasible_parent = np.clip(p, 0.0, 1.0).copy()
                else:
                    proj = project_onto_Q(p, n)
                    if proj is not None:
                        feasible_parent = proj
        except Exception:
            pass
    inc_h = uniform.copy()
    inc_sc = exact_score(uniform, n)
    if feasible_parent is not None:
        sp = exact_score(feasible_parent, n)
        if np.isfinite(sp) and sp < inc_sc - 1e-12:
            inc_h, inc_sc = feasible_parent.copy(), sp
    if n > 128:
        return inc_h, inc_sc, n
    starts = []
    if feasible_parent is not None:
        p = feasible_parent
        cands = [p, p[::-1].copy(), 1.0 - p, uniform,
                 0.5 + 0.30 * np.sin(2.0 * np.pi * (np.arange(n) + 0.5) / n),
                 p + 0.08 * rng.standard_normal(n)]
    else:
        cands = [uniform,
                 0.5 + 0.30 * np.sin(2.0 * np.pi * (np.arange(n) + 0.5) / n),
                 0.5 + 0.20 * np.cos(4.0 * np.pi * (np.arange(n) + 0.5) / n),
                 0.5 + 0.20 * rng.standard_normal(n),
                 rng.random(n)]
    for cand in cands:
        proj = project_onto_Q(cand, n)
        if proj is None:
            continue
        dup = False
        for s in starts:
            if float(np.max(np.abs(s - proj))) < 1e-10:
                dup = True
                break
        if not dup:
            starts.append(proj)
        if len(starts) >= 6:
            break
    if not starts:
        starts = [uniform.copy()]
    for s in starts:
        sc = exact_score(s, n)
        if np.isfinite(sc) and sc < inc_sc - 1e-12:
            inc_h, inc_sc = s.copy(), sc
    for s in starts:
        try:
            ly, ls = local_search(s, n, dx)
            if ls < inc_sc - 1e-12:
                inc_h, inc_sc = ly.copy(), ls
        except Exception:
            pass
    final_candidates = [inc_h]
    if feasible_parent is not None:
        final_candidates.append(feasible_parent)
    final_candidates.append(uniform)
    best_h, best_sc = inc_h, inc_sc
    for fc in final_candidates:
        fc = np.asarray(fc, dtype=float)
        if fc.ndim != 1 or len(fc) != n or not np.all(np.isfinite(fc)):
            continue
        if not (np.all(fc >= 0.0) and np.all(fc <= 1.0)):
            fc = project_onto_Q(fc, n)
            if fc is None:
                continue
        sc = exact_score(fc, n)
        if np.isfinite(sc) and sc < best_sc - 1e-12:
            best_h, best_sc = fc, sc
    final_sc = exact_score(best_h, n)
    if not np.isfinite(final_sc):
        best_h = uniform.copy()
        final_sc = exact_score(best_h, n)
    return best_h, final_sc, n
