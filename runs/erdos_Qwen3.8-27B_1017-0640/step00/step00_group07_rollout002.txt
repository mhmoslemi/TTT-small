    iv = globals().get('initial_h_values', None)
    if iv is not None:
        try:
            iv = np.asarray(iv, dtype=np.float64).ravel()
        except Exception:
            iv = None
    if iv is not None and 2 <= len(iv) <= 999 and np.all(np.isfinite(iv)):
        n = len(iv)
        proj = project_feasible(iv, n)
        base = proj
        if (np.all(iv >= -1e-12) and np.all(iv <= 1.0 + 1e-12) and abs(float(np.sum(iv)) - n / 2.0) <= 1e-8 * max(1, n)):
            parent_valid = True
            parent_arr = proj.copy()

    rng = np.random.default_rng(seed)
    starts = [base.copy()]
    if n_starts >= 2:
        starts.append(project_feasible(base + 0.03 * rng.standard_normal(n), n))
    if n_starts >= 3:
        starts.append(project_feasible(base + 0.12 * rng.standard_normal(n), n))
    if n_starts >= 4:
        starts.append(project_feasible(rng.uniform(0, 1, n), n))
    if n_starts >= 5:
        K = min(6, (n - 1) // 2)
        y = np.full(n, 0.5)
        idx = np.arange(n) + 0.5
        for k in range(1, K + 1):
            a = 0.12 / np.sqrt(k) * rng.standard_normal()
            b = 0.12 / np.sqrt(k) * rng.standard_normal()
            y += a * np.cos(2.0 * np.pi * k * idx / n) + b * np.sin(2.0 * np.pi * k * idx / n)
        starts.append(project_feasible(y, n))
    if n_starts >= 6:
        perm = rng.permutation(n)
        yb = np.zeros(n)
        yb[perm[:n // 2]] = 1.0
        if n % 2 == 1:
            yb[perm[n // 2]] = 0.5
        starts.append(project_feasible(yb, n))
    while len(starts) < min(n_starts, 8):
        starts.append(project_feasible(base + 0.08 * rng.standard_normal(n), n))

    best_h = base.copy()
    best_score = exact_score(best_h)
    for s in starts:
        sc = exact_score(s)
        if sc < best_score - 1e-12:
            best_score = sc
            best_h = s.copy()

    if n <= 256 and n > 1:
        bounds = [(0.0, 1.0)] * n + [(0.0, 2.01)]
        cons = [
            {'type': 'eq', 'fun': partial(_eq, n=n), 'jac': partial(_eq_jac, n=n)},
            {'type': 'ineq', 'fun': partial(_ineq, n=n), 'jac': partial(_ineq_jac, n=n)}
        ]
        for st in starts:
            if best_score <= 0.3808:
                break
            q_init, _ = eval_grad_q(st, n)
            t0 = max(0.0, float(np.max(q_init))) + 1e-8
            z0 = np.concatenate([st, [t0]])
            try:
                res = minimize(
                    fun=partial(_obj, n=n), x0=z0, jac=partial(_obj_grad, n=n),
                    constraints=cons, bounds=bounds, method='SLSQP',
                    options={'ftol': ftol, 'maxiter': maxiter, 'disp': False}
                )
                if res.x is not None and len(res.x) == n + 1:
                    h_cand = project_feasible(res.x[:n], n)
                    sc = exact_score(h_cand)
                    if np.isfinite(sc) and sc < best_score - 1e-12:
                        best_score = sc
                        best_h = h_cand
            except Exception:
                pass

    best_h = project_feasible(best_h, n)
    if not (np.all(np.isfinite(best_h)) and np.all(best_h >= 0) and np.all(best_h <= 1) and abs(float(np.sum(best_h)) - n / 2.0) <= 1e-10 * max(1, n)):
        if parent_valid and parent_arr is not None:
            best_h = parent_arr.copy()
        else:
            best_h = np.full(n, 0.5)
    c5 = exact_score(best_h)
    return (best_h, float(c5), int(n))
