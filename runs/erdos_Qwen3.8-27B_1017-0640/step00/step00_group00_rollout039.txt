def project_simplex(y, n):

    m = n / 2.0
    y = np.clip(y, -1e6, 1e6)
    lo, hi = y.min() - 1, y.max()
    for _ in range(80):
        mid = (lo + hi) / 2
        x = np.clip(y - mid, 0, 1)
        if x.sum() > m:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2, 0, 1)
    # Fix residual
    res = m - x.sum()
    if abs(res) > 1e-14:
        # distribute residual
        for i in range(n):
            if res > 0:
                add = min(res, 1 - x[i])
                x[i] += add
                res -= add
            else:
                sub = min(-res, x[i])
                x[i] -= sub
                res += sub

def refine(h, n, max_sweeps, rng):
    # Returns best h and score from this path
    q, score = exact_score(h, n)
    if score is None:
        return h.copy(), 1e30
    n_pairs = n * (n - 1) // 2
    all_pairs = [(a, b) for a in range(n) for b in range(a+1, n)]
    for sw in range(max_sweeps):
        rng.shuffle(all_pairs)
        improved = 0
        for (a, bb) in all_pairs:
            L = max(-h[a], h[bb] - 1.0)
            U = min(1.0 - h[a], h[bb])
            if U - L < 1e-14:
                continue
            p = bb - a
            bv = coord_grad(h, n, a) - coord_grad(h, n, bb)
            A = np.zeros(2 * n - 1)
            dx = 2.0 / n
            A[n-1] = -2.0 * dx
            A[n-1+p] = dx
            A[n-1-p] = dx
            sp = [n-1, n-1+p, n-1-p]
            t = line_min_t(q, bv, A, sp, L, U, n)
            if abs(t) < 1e-12:
                continue
            trial = h.copy()
            trial[a] += t
            trial[bb] -= t
            if not np.all(np.isfinite(trial)):
                continue
            if trial.min() < -1e-10 or trial.max() > 1.0 + 1e-10:
                trial = project_simplex(trial, n)
                if trial is None:
                    continue
            tn, sn = exact_score(trial, n)
            if sn is None:
                continue
            if sn < score - 1e-12:
                h = trial; q = tn; score = sn; improved += 1
        if improved == 0:
            break
    return h, score
