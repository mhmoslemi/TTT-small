SOLVERS = ('ECOS', 'CVXOPT', 'PDLP', 'SCS')

def _solver_kw(s):
    if s == 'ECOS':
        return {'max_iters': 400, 'abstol': 1e-8, 'reltol': 1e-8, 'feastol': 1e-8}
    if s == 'CVXOPT':
        return {'max_iters': 400, 'feastol': 1e-8}
    if s == 'PDLP':
        return {'max_iter': 500}
    if s == 'SCS':
        return {'max_iters': 3000, 'eps': 1e-8}

def is_feasible(h, n, S):
    if h.ndim != 1 or len(h) != n or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-9) or np.any(h > 1 + 1e-9):
        return False
    return abs(np.sum(h) - S) <= 1e-9

def project_capped(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if not np.all(np.isfinite(v)):
        return None
    v = np.clip(v, -1e6, 1e6)
    lo, hi = np.min(v) - 1.0, np.max(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    for _ in range(5):
        r = S - np.sum(x)
        if abs(r) < 1e-12:
            break

        k = np.sum(m)
        if k == 0:
            break

    x = np.clip(x, 0.0, 1.0)
    if abs(np.sum(x) - S) > 1e-9:
        return None

    pairs = []
    for k in range(-(n - 1), n):
        m = n - abs(k)
        if k >= 0:
            u = (np.arange(m) + k).astype(np.int32)
            v = np.arange(m, dtype=np.int32)
        else:
            u = np.arange(m, dtype=np.int32)
            v = (np.arange(m) - k).astype(np.int32)
        pairs.append((u, v))

    if not is_feasible(h, n, S):
        return np.inf
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * 2.0 / n)

def build_majorizer(n, h_ref, pairs):

    A = np.zeros((nl, n), dtype=np.float64)
    B = np.zeros(nl, dtype=np.float64)
    for k, (u, v) in enumerate(pairs):
        z = h_ref[u] + h_ref[v]
        A[k, u] += 1.0 - 0.5 * z
        A[k, v] += -0.5 * z
        B[k] = np.sum(0.25 * z * z)
    return A, B
