import numpy as np
from scipy.optimize import minimize
import math

_MASS_TARGET = 0.0
_CUR_TAU = 1.0
_CUR_DX = 1.0


def exact_score(h):
    if h.ndim != 1 or not np.all(np.isfinite(h)):
        return float('inf')
    n = len(h)
    q = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(q) * (2.0 / n))


def check_feasible(h, n, m):
    if h.ndim != 1 or len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return False
    if abs(math.fsum(h) - m) > 1e-8:
        return False
    return True


def project_simplex(v, n, m):
    if len(v) != n or not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    mid = 0.5 * (lo + hi)
    x = np.clip(v - mid, 0.0, 1.0)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        x = np.clip(v - mid, 0.0, 1.0)
        s = math.fsum(x)
        if abs(s - m) < 1e-14:
            break
        if s > m:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - math.fsum(x)
    for _ in range(3):
        if abs(r) < 1e-14:
            break
        for i in range(n):
            if abs(r) < 1e-14:
                break
            if r > 0:
                d = min(r, 1.0 - x[i])
                x[i] += d
                r -= d
            else:
                d = min(-r, x[i])
                x[i] -= d
                r += d
    if not check_feasible(x, n, m):
        return None
    return x


def smooth_value_grad(h, tau, dx):
    n = len(h)
    q = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(q)):
        return float('inf'), None
    qmax = float(np.max(q))
    shifted = (q - qmax) / tau
    exp_s = np.exp(shifted)
    s_exp = float(np.sum(exp_s))
    val = dx * (qmax + tau * math.log(s_exp))
    p = exp_s / s_exp
    p_rev = p[::-1]
    c1 = np.convolve(1.0 - h, p, mode='full')
    c2 = np.convolve(h, p_rev, mode='full')
    grad = dx * (c1[n - 1:2 * n - 1] - c2[n - 1:2 * n - 1])
    return val, grad


def opt_fun(h):
    v, _ = smooth_value_grad(h, _CUR_TAU, _CUR_DX)
    return v


def opt_jac(h):
    _, g = smooth_value_grad(h, _CUR_TAU, _CUR_DX)
    if g is None:
        return np.zeros(len(h))
    return g


def constraint_fun(h):
    return math.fsum(h) - _MASS_TARGET


def constraint_jac(h):
    return np.ones(len(h))


def make_fallback(n):
    h = np.zeros(n)
    half = n // 2
    h[:half] = 1.0
    if n % 2 == 1:
        h[half] = 0.5
    return h
