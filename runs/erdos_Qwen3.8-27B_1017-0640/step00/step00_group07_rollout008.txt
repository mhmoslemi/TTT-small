import numpy as np
from scipy.optimize import minimize
import functools

def project_feasible(y, n):
    m = n / 2.0
    y = np.asarray(y, dtype=np.float64)
    if y.shape != (n,) or not np.all(np.isfinite(y)):
        return np.full(n, 0.5)
    lo = float(y.min()) - 1.0
    hi = float(y.max())
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    residual = m - h.sum()
    for _ in range(n + 2):
        if abs(residual) < 2e-14 * max(1, n):
            break
        if residual > 0:
            slack = 1.0 - h
            idx = int(np.argmax(slack))
            if slack[idx] <= 0:
                break
            d = min(residual, slack[idx])
            h[idx] += d
            residual -= d
        else:
            idx = int(np.argmax(h))
            if h[idx] <= 0:
                break
            d = min(-residual, h[idx])
            h[idx] -= d
            residual += d
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - m) > 1e-10 * max(1, n):
        return np.full(n, 0.5)
    return h

def exact_score(h):
    n = len(h)
    if n == 0 or not np.all(np.isfinite(h)):
        return np.inf
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def compute_q_and_jac(h, n):
    dx = 2.0 / n
    q = np.empty(2 * n - 1)
    G = np.zeros((2 * n - 1, n))
    q[0] = dx * np.sum(h * (1.0 - h))
    G[0] = dx * (1.0 - 2.0 * h)
    for s in range(1, n):
        hl = h[:n - s]
        hr = h[s:]
        ol = 1.0 - hl
        orr = 1.0 - hr
        q[s] = dx * np.dot(hl, orr)
        q[n + s - 1] = dx * np.dot(hr, ol)
        ga = np.zeros(n)
        ga[:n - s] = orr
        ga[s:] -= hl
        G[s] = dx * ga
        gb = np.zeros(n)
        gb[s:] = ol
        gb[:n - s] -= hr
        G[n + s - 1] = dx * gb
    return q, G

def _obj(z, n):
    return z[n]

def _obj_jac(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def _eq_fun(z, n):
    return z[:n].sum() - n / 2.0

def _eq_jac(z, n):
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g

def _ineq_fun(z, n):
    h = z[:n]
    q, _ = compute_q_and_jac(h, n)
    return z[n] - q

def _ineq_jac(z, n):
    h = z[:n]
    _, G = compute_q_and_jac(h, n)
    jac = -G.copy()
    jac[:, n] = 1.0
    return jac

def run(seed=42, **kwargs):
    n = 80
    parent = None
    if 'initial_h_values' in globals():
        pv = globals()['initial_h_values']
        if pv is not None:
            pv = np.asarray(pv, dtype=np.float64)
            if pv.ndim == 1 and 2 <= len(pv) <= 999:
                n = len(pv)
                parent = pv.copy()

    n_starts = kwargs.get('n_starts', 6)
    maxiter = kwargs.get('maxiter', 250)
    ftol = kwargs.get('ftol', 1e-10)
    n_starts = max(1, min(8, int(n_starts)))
    maxiter = max(50, min(400, int(maxiter)))
    if not (1e-12 <= ftol <= 1e-6):
        ftol = 1e-10

    try:
        rng = np.random.default_rng(int(seed))
    except (ValueError, TypeError):
        rng = np.random.default_rng(42)

    base = project_feasible(np.full(n, 0.5) if parent is None else parent, n)
    base_score = exact_score(base)
    incumbent = base.copy()
    incumbent_score = base_score
    parent_valid = False

    if parent is not None:
        if (np.all(parent >= -1e-12) and np.all(parent <= 1.0 + 1e-12)
                and abs(parent.sum() - n / 2.0) <= 1e-8 * max(1, n)):
            parent_valid = True
            parent_proj = project_feasible(parent, n)

                incumbent = parent_proj.copy()
                incumbent_score = ps

    starts.append(project_feasible(base + 0.03 * rng.standard_normal(n), n))

    starts.append(project_feasible(rng.uniform(0, 1, n), n))

            yf += (0.12 / np.sqrt(k)) * rng.standard_normal() * np.sin(2.0 * np.pi * k * (x + 0.5) / n)

            incumbent_score = sc

            incumbent = np.full(n, 0.5)

    incumbent_score = exact_score(incumbent)
    return (incumbent, float(incumbent_score), int(n))
