    best_pilot_score = start_score
    deltas = []

            continue

                pert = inc + rng.normal(0, sd, n)
                p = _project_capped(pert, S)
                if p is None or not _feasible(p, n, S):
                    continue
                sc = _exact_score(p, n)
                if sc is None:
                    continue
                inc, inc_score = _anneal_chain(p, sc, n, S, rng, L, M, inc, inc_score, p_perm, p_swap)
            except Exception:
                pass
    elif R >= 1 and n <= 128:
        try:
            pert = inc + rng.normal(0, 0.15, n)
            p = _project_capped(pert, S)
            if p is None or not _feasible(p, n, S):
                pass
            else:
                sc = _exact_score(p, n)
                if sc is not None:
                    inc, inc_score = _anneal_chain(p, sc, n, S, rng, L, M, inc, inc_score, p_perm, p_swap)
        except Exception:
            pass
    # Final validation
    if not _feasible(inc, n, S):
        rep = _project_capped(inc, S)
        if rep is not None and _feasible(rep, n, S):
            rs = _exact_score(rep, n)
            if rs is not None and rs < fallback_score:
                inc, inc_score = rep, rs
            else:
                inc, inc_score = fallback, fallback_score
        else:
            inc, inc_score = fallback, fallback_score
    final_score = _exact_score(inc, n)
    if final_score is None:
        inc, final_score = fallback.copy(), fallback_score
    return inc.copy(), final_score, n
