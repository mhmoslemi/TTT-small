                best_score = sc
        except Exception:
            continue
    for sc_r in restarts:
        try:
            v = best + sc_r * rng.standard_normal(n)
            p = project_capped_simplex(v)
            if p is None:
                continue
            sc = exact_score(p, n)
            if sc is not None and sc < best_score - 1e-12:
                best = p.copy()
                best_score = sc
            res = projected_smooth_search(p, temps, iters)
            sc = exact_score(res, n)
            if sc is not None and sc < best_score - 1e-12:
                best = res.copy()
                best_score = sc
        except Exception:
            continue
    final = project_capped_simplex(best)
    if final is None:
        final = parent.copy()
    fs = exact_score(final, n)
    if fs is None:
        final = const.copy()
        fs = exact_score(final, n)
    if fs is None:
        fs = 1.0
    return final, fs, n
