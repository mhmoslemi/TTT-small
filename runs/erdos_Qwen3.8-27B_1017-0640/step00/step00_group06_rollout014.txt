                    elif rho > 0.75 and step_sz >= 0.8 * delta:
                        delta = min(dmax, 1.5 * delta)
                    q, G = compute_lag_data(h_cur, n)
                    accepted_this = True
                    break
                else:
                    if predicted <= 1e-12:
                        q = None
                        G = None
                        break
                    delta = max(0.5 * delta, dmin)
                    if delta < dmin:
                        break
            if q is None or G is None:
                break
            if delta < dmin:
                break
    if not is_feasible(best, n):
        proj = bounded_simplex_project(best, zeros_n, ones_n, m)
        if proj is not None and is_feasible(proj, n):
            best = proj
        else:
            best = protected
    c5 = exact_score(best, n)
    if c5 > best_score + 1e-12 and is_feasible(protected, n):
        best = protected
        c5 = exact_score(best, n)
    return best.astype(np.float64), float(c5), int(n)
