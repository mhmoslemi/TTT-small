            x[idx] = np.maximum(0.0, x[idx] + res / len(idx))
    x = np.clip(x, 0.0, 1.0)
    if abs(float(np.sum(x)) - S) > 1e-10:
        return None
    return x

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_qG(h):
    n = len(h)
    mt = 2 * n - 1
    q = np.zeros(mt)
    G = np.zeros((mt, n))
    oh = 1.0 - h
    q[n - 1] = float(np.dot(h, oh))
    G[n - 1, :] = 1.0 - 2.0 * h
    for s in range(1, n):
        m = n - s
        q[n - 1 + s] = float(np.dot(h[s:], oh[:m]))
        g1 = np.zeros(n)
        g1[s:] = oh[:m]
        g1[:m] -= h[s:]
        G[n - 1 + s, :] = g1
        q[n - 1 - s] = float(np.dot(h[:m], oh[s:]))
        g2 = np.zeros(n)
        g2[:m] = oh[s:]
        g2[s:] -= h[:m]
        G[n - 1 - s, :] = g2
    return q, G

def _obj(z):
    return z[-1]

def _obj_j(z):
    n = len(z) - 1
    g = np.zeros(n + 1)
    g[-1] = 1.0
    return g

def _ineq(z):
    n = len(z) - 1
    h = z[:n]
    t = z[-1]
    dx = 2.0 / n
    q, _ = compute_qG(h)
    return t - dx * q

def _ineq_j(z):
    n = len(z) - 1
    h = z[:n]
    dx = 2.0 / n
    _, G = compute_qG(h)
    mt = 2 * n - 1
    J = np.zeros((mt, n + 1))
    J[:, :n] = -dx * G
    J[:, n] = 1.0
    return J

def _eq(z):
    n = len(z) - 1
    return float(np.sum(z[:n])) - 0.5 * n

def _eq_j(z):
    n = len(z) - 1
    j = np.zeros(n + 1)
    j[:n] = 1.0
    return j

def run(seed=42, **kwargs):
    n = 45
    parent = None
    try:
        h_init = initial_h_values
        if h_init is not None:
            h_arr = np.array(h_init, dtype=np.float64).ravel()
            if h_arr.size >= 2 and np.all(np.isfinite(h_arr)):
                n = h_arr.size
                parent = h_arr.copy()
    except Exception:
        pass

    S = 0.5 * n
    dx = 2.0 / n
    rng = np.random.default_rng(seed)
    starts = []

    if parent is not None:
        if (np.all(parent >= -1e-12) and np.all(parent <= 1.0 + 1e-12)
                and abs(float(np.sum(parent)) - S) < 1e-7):
            starts.append(np.clip(parent, 0.0, 1.0))
        else:
            proj = project_capped(parent, S)
            if proj is not None:
                starts.append(proj)

    starts.append(np.full(n, 0.5))
    base = starts[0]
    for sig in [0.03, 0.08, 0.16, 0.30, 0.08, 0.20]:
        p = project_capped(base + rng.standard_normal(n) * sig, S)
        if p is not None:
            starts.append(p)
    for _ in range(2):
        p = project_capped(rng.uniform(0, 1, n), S)
        if p is not None:
            starts.append(p)
    for m in [1, 2]:
        ph = rng.uniform(0.0, 2.0 * math.pi)
        ii = np.arange(n, dtype=float)
        pert = 0.5 + 0.35 * np.sin(2.0 * math.pi * m * ii / n + ph)
        p = project_capped(pert, S)
        if p is not None:
            starts.append(p)

    unique_starts = []
    for s in starts:
        dup = False
        for u in unique_starts:
            if np.linalg.norm(s - u) < 1e-12:
                dup = True
                break
        if not dup:
            unique_starts.append(s)
    starts = unique_starts

    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'ineq', 'fun': _ineq, 'jac': _ineq_j},
        {'type': 'eq', 'fun': _eq, 'jac': _eq_j},
    ]
    opts = {'maxiter': 400, 'ftol': 1e-12, 'disp': False}

    best_h = starts[0].copy()
    best_score = exact_score(best_h)

    def _solve(h0):
        z0 = np.concatenate([h0, [min(2.0, max(0.0, exact_score(h0)))]])
        try:
            res = minimize(_obj, z0, jac=_obj_j, bounds=bounds,
                           constraints=cons, method='SLSQP', options=opts)
        except Exception:
            return None
        if res.x is None or len(res.x) != n + 1 or not np.all(np.isfinite(res.x)):
            return None
        h_cand = project_capped(res.x[:n], S)
        if h_cand is None:
            return None
        return h_cand

    for s in starts:
        sc = exact_score(s)
        if sc < best_score - 1e-14:
            best_h = s.copy()
            best_score = sc
        cand = _solve(s)
        if cand is not None:
            cs = exact_score(cand)
            if cs < best_score - 1e-14:
                best_h = cand
                best_score = cs

    # Adaptive restarts
    for sig in [0.08, 0.03, 0.01]:
        for _ in range(2):
            pert = best_h + rng.standard_normal(n) * sig
            proj = project_capped(pert, S)
            if proj is None:
                continue
            cand = _solve(proj)
            if cand is not None:
                cs = exact_score(cand)
                if cs < best_score - 1e-14:
                    best_h = cand
                    best_score = cs

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    if abs(float(np.sum(best_h)) - S) > 1e-10:
        fixed = project_capped(best_h, S)
        if fixed is not None:
            best_h = fixed
    c5 = exact_score(best_h)

    # Fallback check
    if not np.all(np.isfinite(best_h)) or c5 > 0.5:
        fallback = np.full(n, 0.5)
        c5 = exact_score(fallback)
        best_h = fallback

    return (best_h, c5, int(n))
