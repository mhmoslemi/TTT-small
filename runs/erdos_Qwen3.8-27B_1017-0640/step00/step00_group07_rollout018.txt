import numpy as np
import math

def project_capped(y, n):
    m = n / 2.0
    if len(y) != n or not np.all(np.isfinite(y)) or np.any(np.abs(y) > 1e12):
        return np.full(n, 0.5)
    lo = np.min(y) - 1.0
    hi = np.max(y)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > m:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    p = np.clip(y - lam, 0.0, 1.0)
    # Correct residual
    for _ in range(n + 2):
        resid = m - np.sum(p)
        if abs(resid) <= 1e-12 * max(1, n):
            break
        if resid > 0:
            slack = 1.0 - p
            idxs = np.where(slack > 1e-15)[0]
            if len(idxs) == 0:
                break
            add = min(resid / len(idxs), np.min(slack[idxs]))
            p[idxs] += add
        else:
            slack = p.copy()
            idxs = np.where(slack > 1e-15)[0]
            if len(idxs) == 0:
                break
            sub = min(-resid / len(idxs), np.min(slack[idxs]))
            p[idxs] -= sub
    if not np.all(np.isfinite(p)) or np.any(p < -1e-14) or np.any(p > 1 + 1e-14):
        return np.full(n, 0.5)
    p = np.clip(p, 0.0, 1.0)
    if abs(np.sum(p) - m) > 1e-10 * max(1, n):
        return np.full(n, 0.5)
    return p

def is_feasible(h, n):
    if len(h) != n or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-14) or np.any(h > 1 + 1e-14):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def exact_score(h):
    n = len(h)
    if not is_feasible(h, n):
        return float('inf')
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def lag_values(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def smooth_logsumexp(q, tau):
    qmax = np.max(q)
    val = qmax + tau * np.log(np.sum(np.exp((q - qmax) / tau)))
    return val

def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    q = lag_values(h)
    qmax = np.max(q)
    exps = np.exp((q - qmax) / tau)
    w = exps / np.sum(exps)
    # Build gradient: sum_l w_l * dq_l/dh
    # q order: [B_{n-1}, ..., B_1, q0, A_1, ..., A_{n-1}]
    grad = np.zeros(n)
    # Zero lag (index n-1): dq0/dh[j] = dx*(1-2h[j])
    grad += w[n-1] * dx * (1.0 - 2.0 * h)
    # B_s at index n-1-s (s=1..n-1)
    for s in range(1, n):
        idx = n - 1 - s
        ws = w[idx]
        if ws < 1e-18:
            continue
        # dB_s/dh[j] = dx*(1-h[j-s])*[j>=s] - dx*h[j+s]*[j<n-s]
        if s <= n - 1:
            # j >= s: dx*(1-h[j-s])
            grad[s:] += ws * dx * (1.0 - h[:n-s])
            # j < n-s: -dx*h[j+s]
            grad[:n-s] -= ws * dx * h[s:]
    # A_s at index n-1+s (s=1..n-1)
    for s in range(1, n):
        idx = n - 1 + s
        ws = w[idx]
        if ws < 1e-18:
            continue
        # dA_s/dh[j] = dx*(1-h[j+s])*[j<n-s] - dx*h[j-s]*[j>=s]
        grad[:n-s] += ws * dx * (1.0 - h[s:])
        grad[s:] -= ws * dx * h[:n-s]
    return q, grad

def make_starts(base, n, rng, n_starts):
    starts = [base.copy()]
    # Noise starts
    for sd in [0.02, 0.08, 0.20]:
        starts.append(base + sd * rng.standard_normal(n))
    # Uniform random
    starts.append(rng.uniform(0, 1, n))
    # Low-frequency
    K = min(6, (n - 1) // 2)
    t = (np.arange(n) + 0.5) / n
    y = np.full(n, 0.5)
    for k in range(1, K + 1):
        a = 0.12 / math.sqrt(k) * rng.standard_normal()
        b = 0.12 / math.sqrt(k) * rng.standard_normal()
        y += a * np.cos(2 * np.pi * k * t) + b * np.sin(2 * np.pi * k * t)
    starts.append(y)
    # Binary
    binv = np.zeros(n)
    half = n // 2
    idx = rng.choice(n, half, replace=False)
    binv[idx] = 1.0
    if n % 2 == 1:
        remaining = np.setdiff1d(np.arange(n), idx)
        binv[rng.choice(remaining)] = 0.5
    starts.append(binv)
    # Large noise
    starts.append(base + 0.4 * rng.standard_normal(n))
    
    # Project and deduplicate
    proj_starts = []
    for s in starts[:n_starts]:
        ps = project_capped(np.asarray(s, dtype=np.float64), n)
        is_dup = False
        for ps_existing in proj_starts:
            if np.max(np.abs(ps - ps_existing)) < 1e-10:
                is_dup = True
                break
        if not is_dup:
            proj_starts.append(ps)
        if len(proj_starts) >= n_starts:
            break
    return proj_starts

def optimize_one_start(x0, n, max_iter, rng):
    x = x0.copy()
    temperatures = [0.02, 0.005, 0.00125, 0.0003125, 0.000078125, 0.00001953125, 0.0000048828125]
    for tau in temperatures:
        alpha = 1.0
        for it in range(max_iter):
            q, grad = smooth_grad(x, tau)
            if not np.all(np.isfinite(grad)):
                break
            phi = smooth_logsumexp(q, tau)
            for _attempt in range(25):
                xtrial = project_capped(x - alpha * grad, n)
                d = xtrial - x
                if np.max(np.abs(d)) <= 1e-9:
                    break
                gd = np.dot(grad, d)
                if gd >= 0:
                    break
                q_trial = lag_values(xtrial)
                phi_trial = smooth_logsumexp(q_trial, tau)
                if phi_trial <= phi + 1e-4 * gd + 1e-14 * max(1.0, abs(phi)):
                    x = xtrial
                    break
                alpha *= 0.5
            else:
                break
            alpha = min(alpha * 1.25, 4.0)
    return x

def run(seed=42, **kwargs):

def project_capped(y, n):
    m = n / 2.0
    if len(y) != n or not np.all(np.isfinite(y)) or np.any(np.abs(y) > 1e12):
        return np.full(n, 0.5)
    lo = np.min(y) - 1.0
    hi = np.max(y)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(y - mid, 0.0, 1.0)) > m:
            lo = mid
        else:
            hi = mid
    p = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(n + 2):
        resid = m - np.sum(p)
        if abs(resid) <= 1e-12 * max(1, n):
            break
        if resid > 0:
            slack = 1.0 - p
            idxs = np.where(slack > 1e-15)[0]
            if len(idxs) == 0:
                break
            add = min(resid / len(idxs), np.min(slack[idxs]))
            p[idxs] += add
        else:

    return p

def is_feasible(h, n):
    return (len(h) == n and np.all(np.isfinite(h)) and np.all(h >= -1e-14)
            and np.all(h <= 1 + 1e-14) and abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1, n))

def exact_score(h):

    if not is_feasible(h, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lag_values(h):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))

def smooth_lse(q, tau):
    qmax = np.max(q)
    return qmax + tau * np.log(np.sum(np.exp((q - qmax) / tau)))

def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    q = lag_values(h)
    qmax = np.max(q)
    exps = np.exp((q - qmax) / tau)
    w = exps / np.sum(exps)
    grad = np.zeros(n)
    grad += w[n - 1] * dx * (1.0 - 2.0 * h)
    for s in range(1, n):
        wb = w[n - 1 - s]
        if wb > 1e-18:
            grad[s:] += wb * dx * (1.0 - h[:n - s])
            grad[:n - s] -= wb * dx * h[s:]
        wa = w[n - 1 + s]
        if wa > 1e-18:
            grad[:n - s] += wa * dx * (1.0 - h[s:])
            grad[s:] -= wa * dx * h[:n - s]
    return q, grad
