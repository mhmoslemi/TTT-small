        starts.append(project_onto_K(np.roll(h0, n // 3), m))
    starts.append(np.full(n, 0.5))
    starts.append(project_onto_K(make_block(m, n), m))
    starts.append(project_onto_K(make_block(m, n)[::-1], m))
    n_rand = 3 if parent is None else 1
    for _ in range(n_rand):
        starts.append(project_onto_K(rng.random(n), m))
    for r in range(1, 3):
        ph = rng.uniform(0.0, 2.0 * math.pi)
        hc = 0.5 + 0.35 * np.cos(2.0 * math.pi * r * np.arange(n) / n + ph)
        starts.append(project_onto_K(hc, m))
    if parent is not None:
        for sd in [0.05, 0.15]:
            starts.append(project_onto_K(h0 + sd * rng.standard_normal(n), m))

    if n <= 64:
        max_s = 10
    elif n <= 160:
        max_s = 6
    elif n <= 400:
        max_s = 4
    else:
        max_s = 2
    unique = []
    for s in starts:
        if not is_feasible(s, m):
            s = project_onto_K(s, m)
        dup = False
        for u in unique:
            if float(np.max(np.abs(s - u))) <= 1e-10:
                dup = True
                break
        if not dup:
            unique.append(s)
        if len(unique) >= max_s:
            break
    starts = unique

    best = [float('inf'), np.full(n, 0.5)]
    for s in starts:
        sc = exact_score(s, n)
        if math.isfinite(sc) and sc < best[0]:
            best[0] = sc
            best[1] = s.copy()
    for s in starts:
        local_optimize(s, n, m, best, rng)

    if n <= 64:
        prounds = [[0.05, 0.18], [0.05, 0.18]]
    elif n <= 160:
        prounds = [[0.08, 0.20]]
    else:
        prounds = [[0.10]]
    for pset in prounds:
        sc0 = best[0]
        for sd in pset:
            p = project_onto_K(best[1] + sd * rng.standard_normal(n), m)
            sc = exact_score(p, n)
            if math.isfinite(sc) and sc < best[0]:
                best[0] = sc
                best[1] = p.copy()
            local_optimize(p, n, m, best, rng)
        if best[0] >= sc0 - 1e-9:
            break

    hf = best[1]
    if not is_feasible(hf, m):
        hf = project_onto_K(hf, m)
    c5 = exact_score(hf, n)
    if not math.isfinite(c5):
        hf = np.full(n, 0.5)
        c5 = exact_score(hf, n)
    return (hf, c5, n)
