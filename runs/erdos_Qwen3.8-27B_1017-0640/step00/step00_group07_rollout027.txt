        t = exchange_proposal(s2, n, rng, 0.18, 1)
        if t is not None:
            s2 = t
    if not is_feasible(s2, n):
        s2 = project_capped(s2, n)
    starts.append(s2)
    s3 = base.copy()
    for _ in range(min(24, max(4, math.ceil(n / 3.0)))):
        t = exchange_proposal(s3, n, rng, 0.45, 1)
        if t is not None:
            s3 = t
    if not is_feasible(s3, n):
        s3 = project_capped(s3, n)
    starts.append(s3)
    s4 = np.zeros(n, dtype=np.float64)
    idx = rng.permutation(n)
    s4[idx[:n // 2]] = 1.0
    if n % 2 == 1:
        s4[idx[n // 2]] = 0.5
    starts.append(s4)
    starts.append(project_capped(rng.random(n), n))
    deduped = [starts[0]]
    for s in starts[1:]:
        if not is_feasible(s, n):
            s = project_capped(s, n)
        dup = False
        for d in deduped:
            if float(np.max(np.abs(s - d))) < 1e-12:
                dup = True
                break
        if not dup:
            deduped.append(s)
    return deduped[:max_starts]

def run(seed=42, **kwargs):
    n = 80
    base = np.full(n, 0.5, dtype=np.float64)
    try:
        parent = np.asarray(initial_h_values, dtype=np.float64)
        if parent.ndim == 1 and 2 <= parent.shape[0] <= 999:
            n = int(parent.shape[0])
            if np.all(np.isfinite(parent)):
                base = project_capped(parent, n)
            else:
                base = np.full(n, 0.5, dtype=np.float64)
    except Exception:
        n = 80
        base = np.full(n, 0.5, dtype=np.float64)
    try:
        sv = int(seed)
        if sv < 0:
            sv = 42
    except Exception:
        sv = 42
    rng = np.random.default_rng(sv)
    max_starts = 5
    try:
        ns = int(kwargs.get('n_starts', 5))
        max_starts = max(1, min(5, ns))
    except Exception:
        max_starts = 5
    if n > 256:
        max_starts = 1
    parent_fallback = base.copy()
    best_h = base.copy()
    best_score = exact_score(best_h)
    eval_count = 1
    starts = build_starts(base, n, rng, max_starts)
    base_score = exact_score(base)
    eval_count += 1
    for s in starts:
        sc = exact_score(s)
        eval_count += 1
        if sc < best_score - 1e-12:
            best_score = sc
            best_h = s.copy()
    if best_score <= 0.3808:
        fh = project_capped(best_h, n)
        if is_feasible(fh, n):
            return fh, exact_score(fh), n
        return base, exact_score(base), n
    B = min(6000, max(40, int(40000000 / (n * n))))
    P = min(24, max(4, B // 20))
    K_p = max(1, n // 20)
    pos_diffs = []
    for _ in range(P):
        prop = exchange_proposal(base, n, rng, 0.18, K_p)
        if prop is None:
            continue
        sc = exact_score(prop)
        eval_count += 1
        if sc < best_score - 1e-12:
            best_score = sc
            best_h = prop.copy()
        diff = sc - base_score
        if diff > 1e-14:
            pos_diffs.append(diff)
    if pos_diffs:
        T0 = max(1e-5, min(0.02, float(np.median(pos_diffs)) / math.log(2.0)))
    else:
        T0 = 0.001
    if best_score <= 0.3808:
        fh = project_capped(best_h, n)
        if is_feasible(fh, n):
            return fh, exact_score(fh), n
        return base, exact_score(base), n
    S = len(starts)
    A = max(0, (B - eval_count) // (9 * S))
    for ci in range(S):
        current = starts[ci].copy()
        cur_sc = exact_score(current)
        eval_count += 1
        if cur_sc < best_score - 1e-12:
            best_score = cur_sc
            best_h = current.copy()
        for level in range(9):
            T = max(1e-7, T0 * (0.5 ** level))
            rho_l = max(0.0025, 0.25 * (0.55 ** level))
            K_l = max(1, int(round(8 * (0.65 ** level))))
            for _ in range(A):
                if eval_count >= B or best_score <= 0.3808:
                    break
                try:
                    prop = exchange_proposal(current, n, rng, rho_l, K_l)
                    if prop is None or np.array_equal(prop, current):
                        continue
                    sc = exact_score(prop)
                    eval_count += 1
                    if not np.isfinite(sc):
                        continue
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_h = prop.copy()
                    delta = sc - cur_sc
                    if delta <= 0:
                        current = prop.copy()
                        cur_sc = sc
                    else:
                        ratio = delta / T
                        ap = 0.0 if ratio > 745.0 else math.exp(-ratio)
                        if rng.random() < ap:
                            current = prop.copy()
                            cur_sc = sc
                except Exception:
                    continue
            if best_score <= 0.3808:
                break
        if best_score <= 0.3808:
            break
    fh = project_capped(best_h, n)
    sc_f = exact_score(fh)
    sc_p = exact_score(parent_fallback)
    if is_feasible(fh, n) and sc_f <= sc_p:
        selected = fh
    elif is_feasible(parent_fallback, n):
        selected = parent_fallback
    else:
        selected = np.full(n, 0.5, dtype=np.float64)
    if not is_feasible(selected, n):
        selected = np.full(n, 0.5, dtype=np.float64)
    return selected, exact_score(selected), n
