            add = min(1.0 - x[idx], res)
            if add < 1e-16:
                break
            x[idx] += add
        else:
            idx = int(np.argmax(x))
            sub = min(x[idx], -res)
            if sub < 1e-16:
                break
            x[idx] -= sub
    if x.min() < -1e-12 or x.max() > 1.0 + 1e-12:
        return None
    if abs(math.fsum(x) - s) > 1e-9:
        return None
    return x

def is_feasible(h, n):
    if len(h) != n or not np.all(np.isfinite(h)):
        return False
    if h.min() < -1e-12 or h.max() > 1.0 + 1e-12:
        return False
    return abs(math.fsum(h) - n / 2) <= 1e-9

def exact_score(h):
    n = len(h)
    if n == 0 or not np.all(np.isfinite(h)):
        return float('inf')
    val = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))
    return val if np.isfinite(val) else float('inf')

def lag_val_jac(h):
    n = len(h)
    nl = 2 * n - 1
    q = np.zeros(nl)
    J = np.zeros((nl, n))
    for k in range(nl):
        d = k - (n - 1)
        jlo = max(0, -d)
        jhi = min(n, n - d)
        for j in range(jlo, jhi):
            q[k] += h[j + d] * (1.0 - h[j])
            J[k, j + d] += (1.0 - h[j])
            J[k, j] -= h[j + d]
    return q, J

def epig_obj(z):
    return z[-1]

def epig_obj_jac(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g

def epig_eq(z):
    n = len(z) - 1
    return math.fsum(z[:n]) - n / 2

def epig_eq_jac(z):
    n = len(z) - 1
    g = np.zeros(len(z))
    g[:n] = 1.0
    return g

def epig_inq(z):
    n = len(z) - 1
    h = z[:n]
    q, _ = lag_val_jac(h)
    return z[-1] - (2.0 / n) * q

def epig_inq_jac(z):
    n = len(z) - 1
    h = z[:n]
    _, J = lag_val_jac(h)
    nl = 2 * n - 1
    G = np.zeros((nl, n + 1))
    G[:, :n] = -(2.0 / n) * J
    G[:, -1] = 1.0
    return G

def solve_start(h0, n, maxiter):
    s = n / 2
    h0 = np.clip(h0, 0.0, 1.0)
    hp = project_capped(h0, s)
    if hp is None:
        hp = np.full(n, 0.5)
    t0 = min(1.0, exact_score(hp) + 1e-9)
    z0 = np.concatenate([hp, [t0]])
    bounds = [(0.0, 1.0)] * (n + 1)
    cons = [
        {'type': 'eq', 'fun': epig_eq, 'jac': epig_eq_jac},
        {'type': 'ineq', 'fun': epig_inq, 'jac': epig_inq_jac}
    ]
    try:
        res = minimize(epig_obj, z0, method='SLSQP', jac=epig_obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'ftol': 1e-10, 'maxiter': maxiter, 'disp': False})
        if res is not None and hasattr(res, 'x') and len(res.x) == n + 1:
            hr = res.x[:n]
            if np.all(np.isfinite(hr)):
                hf = project_capped(hr, s)
                if hf is not None and is_feasible(hf, n):
                    return hf
    except Exception:
        pass
    return hp.copy()

def make_starts(n, parent, rng):
    starts = []
    s = n / 2
    base = parent
    if base is not None:
        p = project_capped(base, s)
        if p is not None:
            starts.append(p.copy())
        starts.append(np.full(n, 0.5))
    else:
        starts.append(np.full(n, 0.5))
    if base is None:
        base = np.full(n, 0.5)
    for sigma in [0.03, 0.10, 0.25]:
        v = base + sigma * rng.standard_normal(n)
        p = project_capped(v, s)
        if p is not None:
            starts.append(p)
    n_uni = 1 if parent is not None else 2
    for _ in range(n_uni):
        v = rng.random(n)
        p = project_capped(v, s)
        if p is not None:
            starts.append(p)
    for q_h in [1, 2]:
        phi = 2.0 * math.pi * rng.random()
        idx = np.arange(n)
        v = 0.5 + 0.25 * np.sin(2.0 * math.pi * q_h * (idx + 0.5) / n + phi)
        p = project_capped(v, s)
        if p is not None:
            starts.append(p)
    return starts

def run(seed=42, **kwargs):
    try:
        seed = int(abs(seed))
    except (ValueError, TypeError, OverflowError):
        seed = 42
    rng = np.random.default_rng(seed)
    parent = None
    n = 47
    try:
        pv = np.array(initial_h_values, dtype=np.float64).ravel()
        if 2 <= len(pv) <= 999 and np.all(np.isfinite(pv)) and pv.min() >= -1e-7 and pv.max() <= 1.0 + 1e-7:
            parent = pv
            n = len(pv)
    except Exception:
        pass
    s = n / 2
    if parent is not None:
        fb = project_capped(parent, s)
    else:
        fb = None
    if fb is None:
        fb = np.full(n, 0.5)
    best_h = fb.copy()
    best_score = exact_score(best_h)
    if n <= 64:
        max_starts, n_adaptive, maxiter = 8, 2, 300
    elif n <= 128:
        max_starts, n_adaptive, maxiter = 6, 1, 180
    else:
        max_starts, n_adaptive, maxiter = 3, 0, 80
    starts = make_starts(n, parent, rng)[:max_starts]
    for hs in starts:
        sc = exact_score(hs)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score, best_h = sc, hs.copy()
        ho = solve_start(hs, n, maxiter)
        sc = exact_score(ho)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score, best_h = sc, ho.copy()
    for sigma in [0.02, 0.06][:n_adaptive]:
        v = best_h + sigma * rng.standard_normal(n)
        p = project_capped(v, s)
        if p is None:
            continue
        sc = exact_score(p)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score, best_h = sc, p.copy()
        ho = solve_start(p, n, maxiter)
        sc = exact_score(ho)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score, best_h = sc, ho.copy()
    if not is_feasible(best_h, n):
        best_h = fb.copy()
    bf = project_capped(best_h, s)
    if bf is None:
        bf = np.full(n, 0.5)
    c5 = exact_score(bf)
    if not np.isfinite(c5):
        bf = np.full(n, 0.5)
        c5 = exact_score(bf)
    return (bf, c5, n)
