def capped_simplex_project(v, n):
    if not np.all(np.isfinite(v)) or v.size != n:
        return np.full(n, 0.5)
    v = np.clip(v, -1e6, 1e6)
    target = n / 2.0
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        q = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(q)
        if s > target:
            lo = mid
        else:
            hi = mid
    q = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    r = target - np.sum(q)

            break
        acc = False
        for _ in range(24):
            ht = cs_proj(h - alpha * g, n)
            if not feas(ht, n):
                alpha *= 0.5; continue
            ct = np.correlate(ht, 1.0 - ht, mode='full') * (2.0 / n)
            Ft, _ = surr(ct, None, tau)
            if Ft is None:
                alpha *= 0.5; continue
            dlt = ht - h
            if Ft <= F + 1e-4 * np.dot(g, dlt) + 1e-12 * max(1.0, abs(F)):
                h = ht; acc = True
                alpha = min(1.5 * alpha, 2.0)
                sc = escore(h)
                if sc < inc_sc - 1e-12:
                    inc = h.copy(); inc_sc = sc
                if np.max(np.abs(dlt)) <= 1e-8:
                    return h, inc, inc_sc
                if (F - Ft) <= 1e-12 * max(1.0, abs(F)):
                    stall += 1
                    if stall >= 8:
                        return h, inc, inc_sc
                else:
                    stall = 0
                break
            alpha *= 0.5
        if not acc:
            break
    return h, inc, inc_sc

def add_s(starts, v, n):
    v = cs_proj(v.copy(), n)
    for s in starts:
        if np.max(np.abs(v - s)) < 1e-10:
            return
    starts.append(v)

def run(seed=42, **kwargs):
    parent = globals().get('initial_h_values', None)
    n = 70; pv = None
    if parent is not None:
        try:
            p = np.array(parent, dtype=np.float64).ravel()
            if p.size >= 2:
                n = p.size
                if np.all(np.isfinite(p)):
                    p2 = np.clip(p, 0.0, 1.0)
                    pv = p2 if abs(np.sum(p2) - n/2.0) <= 1e-10 else cs_proj(p2, n)
        except Exception:
            pass
    if pv is None:
        if 'n_points' in kwargs:
            try:
                nv = int(kwargs['n_points'])
                if 2 <= nv <= 256: n = nv
            except Exception:
                pass
        pv = np.full(n, 0.5)
    fb = pv.copy(); fb_sc = escore(fb)
    if n > 256:
        return fb, fb_sc, n
    rng = np.random.default_rng(seed)
    starts = []
    if np.max(np.abs(pv - 0.5)) > 1e-6:
        add_s(starts, pv, n)
        add_s(starts, pv[::-1], n)
        add_s(starts, pv + 0.05 * rng.standard_normal(n), n)
        add_s(starts, pv + 0.15 * rng.standard_normal(n), n)
    else:
        idx = np.arange(n) + 0.5
        add_s(starts, 0.5 + 0.3 * np.sin(2 * np.pi * idx / n), n)
        add_s(starts, 0.5 + 0.15 * rng.standard_normal(n), n)
        add_s(starts, 0.5 + 0.35 * rng.standard_normal(n), n)
    add_s(starts, np.full(n, 0.5), n)
    starts = starts[:5 if n <= 128 else 4]
    inc, inc_sc = fb.copy(), fb_sc
    for s in starts:
        sc = escore(s)
        if sc < inc_sc - 1e-12:
            inc, inc_sc = s.copy(), sc
    temps = [0.02, 0.005, 0.001, 0.0002, 0.00004, 0.000008]
