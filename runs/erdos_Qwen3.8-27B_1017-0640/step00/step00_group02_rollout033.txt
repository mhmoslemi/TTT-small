        pairs.append((u, v))
    return pairs

def build_majorizer_coeffs(h_ref, pairs):
    N = h_ref.shape[0]
    coeffs = []
    for u, v in pairs:
        a = np.zeros(N)
        b = 0.0
        for uu, vv in zip(u, v):
            z = h_ref[uu] + h_ref[vv]
            a[uu] += 1.0 - z / 2.0
            a[vv] += -z / 2.0
            b += z * z / 4.0
        coeffs.append((a, b))
    return coeffs

def try_solve_solver(N, S, dx, pairs, h_ref, solver_name):
    try:
        import cvxpy as cp
    except ImportError:
        return None
    try:
        coeffs = build_majorizer_coeffs(h_ref, pairs)
        x = cp.Variable(N)
        t = cp.Variable()
        constraints = [x >= 0, x <= 1, cp.sum(x) == S, t >= 0]
        for (a, b), (u, v) in zip(coeffs, pairs):
            quad = cp.sum_squares(x[u] - x[v]) / 4.0
            constraints.append(dx * (quad + a @ x + b) <= t)
        prob = cp.Problem(cp.Minimize(t), constraints)
        x.value = h_ref
        t.value = exact_score(h_ref, N)
        prob.solve(solver=solver_name, max_iters=200)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            xval = x.value
            if xval is not None and xval.shape == (N,) and np.all(np.isfinite(xval)):
                return xval
    except Exception:
        pass
    return None

def local_mm(h_start, N, S, dx, pairs, max_iter):
    h_cur = h_start.copy()
    score_cur = exact_score(h_cur, N)
    if not np.isfinite(score_cur):
        return h_cur, score_cur
    for _ in range(max_iter):
        x_prop = try_solve_solver(N, S, dx, pairs, h_cur, 'ECOS')
        best_prop = None
        best_prop_score = np.inf
        if x_prop is not None:
            hp = project_capped(x_prop, S)
            if hp is not None and is_feasible(hp, N):
                sp = exact_score(hp, N)
                if np.isfinite(sp) and sp < score_cur - 1e-10:
                    best_prop = hp
                    best_prop_score = sp
        if best_prop is None:
            x_prop2 = try_solve_solver(N, S, dx, pairs, h_cur, 'CVXOPT')
            if x_prop2 is not None:
                hp2 = project_capped(x_prop2, S)
                if hp2 is not None and is_feasible(hp2, N):
                    sp2 = exact_score(hp2, N)
                    if np.isfinite(sp2) and sp2 < score_cur - 1e-10:
                        best_prop = hp2
                        best_prop_score = sp2
        if best_prop is None:
            break
        change = float(np.max(np.abs(best_prop - h_cur)))
        h_cur = best_prop
        score_cur = best_prop_score
        if change <= 1e-7:
            break
    return h_cur, score_cur

def run(seed=42, **kwargs):
    N = 45
    h_parent = None
    try:
        arr = np.asarray(initial_h_values, dtype=np.float64)
        if arr.ndim == 1 and arr.shape[0] >= 2:
            N = int(arr.shape[0])
            h_parent = arr.copy()
    except (NameError, Exception):
        pass
    S = N / 2.0
    dx = 2.0 / N
    rng = np.random.default_rng(seed)
    h_const = np.full(N, 0.5)
    h_fallback = h_const.copy()
    if h_parent is not None and np.all(np.isfinite(h_parent)):
        if is_feasible(h_parent, N):
            h_fallback = h_parent.copy()
        else:
            proj = project_capped(h_parent, S)
            if proj is not None and is_feasible(proj, N):
                h_fallback = proj
    fallback_score = exact_score(h_fallback, N)
    if N > 256:
        return (h_fallback.copy(), fallback_score, int(N))
    pairs = build_lag_pairs(N)
    starts = [h_fallback.copy(), h_const.copy()]
    for std in [0.08, 0.22]:
        hn = h_fallback + rng.normal(0.0, std, N)
        p = project_capped(hn, S)
        if p is not None and is_feasible(p, N):
            starts.append(p)
    hu = rng.uniform(0.0, 1.0, N)
    p = project_capped(hu, S)
    if p is not None and is_feasible(p, N):
        starts.append(p)
    phase = rng.uniform(0.0, 2.0 * np.pi)
    ia = np.arange(N)
    hs = 0.5 + 0.35 * np.sin(2.0 * np.pi * ia / N + phase)
    p = project_capped(hs, S)
    if p is not None and is_feasible(p, N):
        starts.append(p)
    deduped = []
    for st in starts:
        dup = False
        for d in deduped:
            if float(np.max(np.abs(st - d))) < 1e-12:
                dup = True
                break
        if not dup:
            deduped.append(st)
    starts = deduped
    if N <= 64:
        max_iter = 12
    elif N <= 128:
        max_iter = 8
    else:
        max_iter = 4
    h_best = h_fallback.copy()
    score_best = fallback_score
    for st in starts:
        h_loc, s_loc = local_mm(st, N, S, dx, pairs, max_iter)
        if np.isfinite(s_loc) and s_loc < score_best - 1e-12:
            h_best = h_loc
            score_best = s_loc
    for std in [0.04, 0.015]:
        hn = h_best + rng.normal(0.0, std, N)
        p = project_capped(hn, S)
        if p is not None and is_feasible(p, N):
            h_loc, s_loc = local_mm(p, N, S, dx, pairs, max_iter)
            if np.isfinite(s_loc) and s_loc < score_best - 1e-12:
                h_best = h_loc
                score_best = s_loc
    if not is_feasible(h_best, N):
        h_best = h_fallback.copy()
        score_best = fallback_score
    final_score = exact_score(h_best, N)
    if not np.isfinite(final_score):
        h_best = h_fallback.copy()
        final_score = fallback_score
    return (h_best.copy(), float(final_score), int(N))
