import numpy as np

def project_simplex(v):
    n = len(v)
    s = n / 2.0
    if n < 2 or not np.all(np.isfinite(v)) or max(abs(float(v.min())), abs(float(v.max()))) > 1e6:
        return None
    lo_b, hi_b = float(v.min()) - 1.0, float(v.max())
    for _ in range(80):
        mid = (lo_b + hi_b) / 2.0
        if math.fsum(np.clip(v - mid, 0.0, 1.0)) > s:
            lo_b = mid
        else:
            hi_b = mid
    x = np.clip(v - (lo_b + hi_b) / 2.0, 0.0, 1.0)
    for _ in range(3):
        r = s - math.fsum(x)
        if abs(r) <= 5e-15:
            break
        for k in range(n):
            if r > 0:
                add = min(r, 1.0 - x[k])
                x[k] += add; r -= add
            else:
                sub = min(-r, x[k])
                x[k] -= sub; r += sub
            if abs(r) <= 5e-15:
                break
    if not np.all(np.isfinite(x)) or x.min() < -1e-12 or x.max() > 1.0 + 1e-12:
        return None
    if abs(math.fsum(x) - s) > 1e-10:
        return None
    return np.clip(x, 0.0, 1.0)

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    try:
        c = np.correlate(h, 1.0 - h, mode='full')
        if np.all(np.isfinite(c)):
            return float(np.max(c) * dx)
    except Exception:
        pass
    mx = 0.0
    for d in range(-(n - 1), n):
        r0, r1 = max(0, -d), min(n, n - d)
        for r in range(r0, r1):
            v = h[r] * (1.0 - h[r + d])
            if v > mx:
                mx = v
    return mx * dx

def coord_deriv(h, r, d, n):
    val = 0.0
    rd = r + d
    if 0 <= rd < n:
        val += 1.0 - h[rd]
    rm = r - d
    if 0 <= rm < n:
        val -= h[rm]
    return val

def pair_line_search(h, score, i, j):
    n = len(h)
    dx = 2.0 / n
    nlag = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(corr)):
        return h.copy(), score
    ci = 2 * (n - 1) - np.arange(nlag)
    c = dx * corr[ci]
    a = np.zeros(nlag)
    a[n - 1] = -2.0 * dx
    a[i - j + n - 1] += dx
    a[j - i + n - 1] += dx
    b = np.zeros(nlag)
    for k in range(nlag):
        L = k - (n - 1)
        b[k] = dx * (coord_deriv(h, i, L, n) - coord_deriv(h, j, L, n))
    if not (np.all(np.isfinite(a)) and np.all(np.isfinite(b)) and np.all(np.isfinite(c))):
        return h.copy(), score
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-14:
        return h.copy(), score
    cands = [lo, hi]
    if lo <= 0.0 <= hi:
        cands.append(0.0)
    if n <= 64:

def local_search(h, score, rng, n_passes, pairs_pp):
    n = len(h)
    empty_passes = 0
    for p in range(n_passes):
        if n <= 128:
            all_pairs = [(k, k + 1) for k in range(n - 1)]
            rng.shuffle(all_pairs)
            pairs = all_pairs[:pairs_pp]
        else:
            pairs, seen = [], set()
            att = 0
            while len(pairs) < pairs_pp and att < pairs_pp * 20:
                x, y = int(rng.integers(0, n)), int(rng.integers(0, n))
                if x < y and (x, y) not in seen:
                    pairs.append((x, y)); seen.add((x, y))
                att += 1
        for (i, j) in pairs:
            h, score = pair_line_search(h, score, i, j)
        empty_passes = 0
    return h, score

def make_starts(parent, n, rng):

        starts.append(0.5 + amp * np.sin(2.0 * math.pi * f * idx / n + ph))
    starts.append(0.5 + rng.standard_normal(n) * 0.12)
