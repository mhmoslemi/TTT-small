    while len(starts) < count:
        r = rng.random()
        if r < 0.5:
            sc = scales[i % 3]
            v = anchor + rng.standard_normal(n) * sc
        else:
            v = rng.uniform(-0.25, 1.25, n)
        p = project_feasible(v, n)
        if not starts or np.min([np.max(np.abs(p - s)) for s in starts]) > 1e-10:
            starts.append(p)
        i += 1
        if i > 200:
            break
    return starts[:count]

def solve_resolution(n, anchor, rng, n_starts, maxiter):
    best_h, best_s = None, np.inf
    starts = make_starts(n, anchor, rng, n_starts)
    for h0 in starts:
        s0 = exact_score(h0)
        if s0 < best_s:
            best_s, best_h = s0, h0.copy()
        t0 = min(1.0, s0 + 1e-8)
        z0 = np.concatenate([h0, [t0]])
        bnds = [(0.0, 1.0)] * (n + 1)
        cons = [
            {'type': 'eq', 'fun': slsqp_eq_fun, 'jac': slsqp_eq_jac},
            {'type': 'ineq', 'fun': slsqp_ineq_fun, 'jac': slsqp_ineq_jac},
        ]
        try:
            res = minimize(slsqp_obj, z0, method='SLSQP', jac=slsqp_obj_grad,
                           bounds=bnds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-10})
            if hasattr(res, 'x') and res.x is not None and len(res.x) == n + 1:
                h_new = project_feasible(res.x[:-1], n)
                if is_feasible(h_new):
                    s_new = exact_score(h_new)
                    if s_new < best_s - 1e-14:
                        best_s, best_h = s_new, h_new
        except Exception:
            pass
    return best_h, best_s

def run(seed=42, **kwargs):
    try:
        seed = int(seed)
    except (ValueError, TypeError):
        seed = 42
    rng = np.random.default_rng(seed)
    n_starts = kwargs.get('n_starts', 8)
    maxiter = kwargs.get('maxiter', 250)
    try:
        n_starts = max(4, min(10, int(n_starts)))
    except (ValueError, TypeError):
        n_starts = 8
    try:
        maxiter = max(100, min(400, int(maxiter)))
    except (ValueError, TypeError):
        maxiter = 250

    parent = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        p = np.asarray(initial_h_values, dtype=np.float64).flatten()
        if p.size >= 2 and np.all(np.isfinite(p)):
            if np.all(p >= -1e-9) and np.all(p <= 1.0 + 1e-9) and abs(np.sum(p) - p.size / 2.0) <= 1e-8 * p.size:
                parent = project_feasible(p, p.size)
            else:
                parent = project_feasible(np.clip(p, 0, 1), p.size)
    if parent is None:
        n0 = 42
        parent = np.zeros(n0)
        parent[:n0 // 2] = 1.0
    n0 = parent.size
    fallback_h = parent.copy()
    fallback_s = exact_score(fallback_h)

    best_h, best_s = fallback_h.copy(), fallback_s
    if n0 <= 256:
        h_nat, s_nat = solve_resolution(n0, parent, rng, n_starts, maxiter)
        if h_nat is not None and s_nat < best_s - 1e-14:
            best_h, best_s = h_nat, s_nat

    if n0 <= 128:
        n1 = 2 * n0
        anchor1 = np.repeat(parent, 2)
        transfer_best = None
        if best_h.size == n0:
            transfer_best = np.repeat(best_h, 2)
        anchor_use = transfer_best if transfer_best is not None else anchor1
        h_hi, s_hi = solve_resolution(n1, anchor_use, rng, n_starts, maxiter)
        if h_hi is not None:
            if s_hi < best_s - 1e-14:
                best_h, best_s = h_hi.copy(), s_hi
            elif abs(s_hi - best_s) <= 1e-12 and h_hi.size < best_h.size:
                best_h, best_s = h_hi.copy(), s_hi

    if not is_feasible(best_h):
        best_h = fallback_h
    best_h = project_feasible(best_h, best_h.size)
    best_s = exact_score(best_h)
    return best_h, best_s, int(len(best_h))
