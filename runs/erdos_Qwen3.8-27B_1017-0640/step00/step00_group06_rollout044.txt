                if a_seg - 1e-12 <= st <= b_seg + 1e-12:
                    cands.append(max(a_seg, min(b_seg, st)))
        for p1 in range(len(polys)):
            for p2 in range(p1 + 1, len(polys)):
                da = polys[p1][0] - polys[p2][0]
                db = polys[p1][1] - polys[p2][1]
                dc = polys[p1][2] - polys[p2][2]
                if abs(da) < 1e-15:
                    if abs(db) > 1e-15:
                        rt = -dc / db
                        if a_seg - 1e-12 <= rt <= b_seg + 1e-12:
                            cands.append(max(a_seg, min(b_seg, rt)))
                else:
                    disc = db * db - 4.0 * da * dc
                    tol = 1e-12 * (db * db + abs(4.0 * da * dc) + 1.0)
                    if disc > -tol:
                        disc = max(0.0, disc)
                        sq = math.sqrt(disc)
                        z = -0.5 * (db + math.copysign(sq, db))
                        if abs(z) > 1e-15:
                            r1, r2 = z / da, dc / z
                        else:
                            r1 = r2 = -db / (2.0 * da)
                        for rt in (r1, r2):
                            if math.isfinite(rt) and a_seg - 1e-12 <= rt <= b_seg + 1e-12:
                                cands.append(max(a_seg, min(b_seg, rt)))
        if a_seg - 1e-12 <= 0.0 <= b_seg + 1e-12:
            cands.append(0.0)
        for cand in cands:
            if not math.isfinite(cand):
                continue
            val = max(A * cand * cand + Bc * cand + D for (A, Bc, D) in polys)
            if val < best_val - 1e-14 * max(1.0, abs(best_val)):
                best_val = val
                best_d = cand
            elif abs(val - best_val) <= 1e-14 * max(1.0, abs(best_val)) and abs(cand) < abs(best_d):
                best_d = cand
    cur_max = float(np.max(q))
    if best_val >= cur_max - 1e-12 * n / 2.0:
        return 0.0
    return best_d if math.isfinite(best_d) else 0.0

def gauss_seidel_sweep(h, q, score, n, m, pair_list, rng):
    h_cur = h.copy()
    q_cur = q.copy()
    sc_cur = score
    rng.shuffle(pair_list)
    for (i, j) in pair_list:
        try:
            delta = line_minimize(h_cur, q_cur, n, i, j)
            if abs(delta) < 1e-14:
                continue
            h_new = h_cur.copy()
            h_new[i] += delta
            h_new[j] -= delta
            h_new[i] = max(0.0, min(1.0, h_new[i]))
            h_new[j] = max(0.0, min(1.0, h_new[j]))
            r = m - math.fsum(h_new.astype(float))
            for _ in range(2):
                if r > 1e-14:
                    for k in range(n):
                        a = min(r, 1.0 - h_new[k])
                        h_new[k] += a
                        r -= a
                        if r <= 1e-14:
                            break
                elif r < -1e-14:
                    for k in range(n):
                        s = min(-r, h_new[k])
                        h_new[k] -= s
                        r += s
                        if -r <= 1e-14:
                            break
                else:
                    break
            if not feasible_check(h_new, n, m):
                continue
            q_new = np.correlate(h_new, 1.0 - h_new, mode="full")
            if not np.all(np.isfinite(q_new)):
                continue
            sc_new = float(np.max(q_new)) * 2.0 / n
            if sc_new < sc_cur - 1e-12:
                h_cur, q_cur, sc_cur = h_new, q_new, sc_new
        except Exception:
            continue
    return h_cur, q_cur, sc_cur

def build_pair_list(n, rng):
    total = n * (n - 1) // 2
    Q = min(total, max(64, min(5000, int(5000 * (95.0 / n) ** 2))))
    if total <= Q:
        return [(i, j) for i in range(n) for j in range(i + 1, n)]
    pairs = []
    seen = set()
    for _ in range(20 * Q):
        if len(pairs) >= Q:
            break
        a = int(rng.integers(0, n))
        b = int(rng.integers(0, n))
        if a == b:
            continue
        if a > b:
            a, b = b, a
        if (a, b) not in seen:
            seen.add((a, b))
            pairs.append((a, b))
    if len(pairs) < Q:
        for i in range(n):
            for j in range(i + 1, n):
                if len(pairs) >= Q:
                    break
                if (i, j) not in seen:
                    pairs.append((i, j))
                    seen.add((i, j))
    return pairs

def run(seed=42, **kwargs):
    n = 95
    try:
        if 'initial_h_values' in dir() and initial_h_values is not None:
            arr = np.asarray(initial_h_values, dtype=np.float64)
            if arr.ndim == 1 and 2 <= arr.shape[0] <= 999:
                n = arr.shape[0]
        elif 'n_points' in kwargs:
            nn = int(kwargs['n_points'])
            if 2 <= nn <= 999:
                n = nn
    except Exception:
        n = 95
    dx = 2.0 / n
    m = n / 2.0
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    fallback = make_half_block(n)
    starts = []
    parent_h = None
    try:
        if 'initial_h_values' in dir() and initial_h_values is not None:
            arr = np.asarray(initial_h_values, dtype=np.float64)
            if arr.ndim == 1 and arr.shape[0] == n and np.all(np.isfinite(arr)):
                if feasible_check(arr, n, m):
                    parent_h = arr.copy()
                else:
                    parent_h = capped_simplex_project(arr, n, m)
    except Exception:
        pass
    if parent_h is not None:
        starts.append(parent_h)
    hb = make_half_block(n)
    if not starts or all(np.max(np.abs(s - hb)) > 1e-12 for s in starts):
        starts.append(hb)
    phase = float(rng.uniform(0.0, 2.0 * math.pi))
    idx_arr = np.arange(n, dtype=np.float64)
    sinc = 0.5 + 0.45 * np.sin(2.0 * math.pi * idx_arr / n + phase)
    sinc_p = capped_simplex_project(sinc, n, m)
    if sinc_p is not None and (not starts or all(np.max(np.abs(s - sinc_p)) > 1e-12 for s in starts)):
        starts.append(sinc_p)
    starts = starts[:3]
    best_h = fallback.copy()
    best_sc = exact_score(fallback, n)
    for s in starts:
        sc = exact_score(s, n)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_h = s.copy()
    pair_list = build_pair_list(n, rng)
    for s in starts:
        q_s = np.correlate(s, 1.0 - s, mode="full")
        sc_s = float(np.max(q_s)) * 2.0 / n
        h_r, q_r, sc_r = gauss_seidel_sweep(s, q_s, sc_s, n, m, pair_list, rng)
        if sc_r < best_sc - 1e-12:
            best_sc = sc_r
            best_h = h_r.copy()
    q_best = np.correlate(best_h, 1.0 - best_h, mode="full")
    sc_best = float(np.max(q_best)) * 2.0 / n
    h_f, q_f, sc_f = gauss_seidel_sweep(best_h, q_best, sc_best, n, m, pair_list, rng)
    if sc_f < best_sc - 1e-12:
        best_sc = sc_f
        best_h = h_f.copy()
    if not feasible_check(best_h, n, m):
        proj = capped_simplex_project(best_h, n, m)
        if proj is not None:
            best_h = proj
        else:
            best_h = fallback.copy()
    best_sc = exact_score(best_h, n)
    if not math.isfinite(best_sc):
        best_h = fallback.copy()
        best_sc = exact_score(fallback, n)
    return best_h, float(best_sc), n
