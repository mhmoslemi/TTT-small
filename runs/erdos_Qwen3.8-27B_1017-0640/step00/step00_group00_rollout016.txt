    dx = 2.0 / n
    g = np.zeros(n)
    g += dx * w[n - 1] * (1.0 - 2.0 * h)
    for p in range(1, n):
        wp = w[n - 1 + p]
        wm = w[n - 1 - p]
        if wp != 0.0:
            g[p:] += dx * wp * (1.0 - h[:-p])
            g[:-p] -= dx * wp * h[p:]
        if wm != 0.0:
            g[:-p] += dx * wm * (1.0 - h[p:])
            g[p:] -= dx * wm * h[:-p]
    return g


def local_optimize(x, n, m, tau_schedule, max_iters):
    best_x = x.copy()
    best_score = exact_score(x, n)
    if not math.isfinite(best_score):
        best_score = 1.0
    for tau in tau_schedule:
        alpha = 1.0
        for it in range(max_iters):
            q = compute_correlations(x, n)
            if not np.all(np.isfinite(q)):
                break
            F_val, w = compute_surrogate_weights(q, tau)
            if F_val is None:
                break
            g = compute_gradient(x, w, n)
            if not np.all(np.isfinite(g)):
                break
            accepted = False
            for bt in range(20):
                y = project_onto_K(x - alpha * g, m)
                d = y - x
                gd = float(np.dot(g, d))
                if not math.isfinite(gd) or gd >= 0.0:
                    alpha *= 0.5
                    continue
                q_y = compute_correlations(y, n)
                if not np.all(np.isfinite(q_y)):
                    alpha *= 0.5
                    continue
                F_y, _ = compute_surrogate_weights(q_y, tau)
                if F_y is None:
                    alpha *= 0.5
                    continue
                if F_y <= F_val + 1e-4 * gd + 1e-14:
                    inf_norm = float(np.max(np.abs(d)))
                    if inf_norm <= 1e-12:
                        accepted = True
                        break
                    if alpha > 0 and inf_norm / alpha <= 1e-7:
                        x = y
                        sc = exact_score(y, n)
                        if math.isfinite(sc) and sc < best_score - 1e-12:
                            best_score = sc
                            best_x = y.copy()
                        break
                    x = y
                    sc = exact_score(y, n)
                    if math.isfinite(sc) and sc < best_score - 1e-12:
                        best_score = sc
                        best_x = y.copy()
                    alpha = min(2.0, 1.25 * alpha)
                    accepted = True
                    break
                alpha *= 0.5
            if not accepted:
                break
    return best_x, best_score


def interpolate_to_n(h_src, n_tgt):
    n_s = len(h_src)
    if n_s == n_tgt:
        return h_src.copy()
    x_s = np.linspace(0.0, 1.0, n_s)
    x_t = np.linspace(0.0, 1.0, n_tgt)
    return np.interp(x_t, x_s, h_src)


def make_block(n, m):
    h = np.zeros(n)
    k = int(math.floor(m))
    h[:k] = 1.0
    frac = m - k
    if frac > 1e-14:
        h[k] = frac
    return h


def generate_starts(n, m, parent, seed):
    rng = np.random.default_rng(seed)
    starts = []
    if parent is not None:
        hp = project_onto_K(parent, m)
        starts.append(hp)
        starts.append(project_onto_K(1.0 - hp, m))
        starts.append(project_onto_K(hp[::-1].copy(), m))
        starts.append(project_onto_K(np.roll(hp, n // 3), m))
        starts.append(np.full(n, m / n))
        hb = make_block(n, m)
        starts.append(project_onto_K(hb, m))
        starts.append(project_onto_K(hb[::-1].copy(), m))
        starts.append(project_onto_K(rng.uniform(0, 1, n), m))
        phase = rng.uniform(0, 2 * math.pi)
        i = np.arange(n, dtype=np.float64)
        hcos = 0.5 + 0.35 * np.cos(2.0 * math.pi * i / n + phase)
        starts.append(project_onto_K(hcos, m))
        starts.append(project_onto_K(hp + 0.05 * rng.standard_normal(n), m))
        starts.append(project_onto_K(hp + 0.15 * rng.standard_normal(n), m))
    else:
        starts.append(np.full(n, m / n))
        hb = make_block(n, m)
        starts.append(project_onto_K(hb, m))
        starts.append(project_onto_K(hb[::-1].copy(), m))
        for _ in range(3):
            starts.append(project_onto_K(rng.uniform(0, 1, n), m))
        for r in [1, 2]:
            phase = rng.uniform(0, 2 * math.pi)
            i = np.arange(n, dtype=np.float64)
            hcos = 0.5 + 0.35 * np.cos(2.0 * math.pi * r * i / n + phase)
            starts.append(project_onto_K(hcos, m))
    unique = []
    for s in starts:
        dup = False
        for u in unique:
            if float(np.max(np.abs(s - u))) <= 1e-10:
                dup = True
                break
        if not dup:
            unique.append(s)
    return unique


def run(seed=42, **kwargs):
    n = 45
    parent = None
    if 'n_points' in kwargs:
        nv = kwargs['n_points']
        if not isinstance(nv, bool):
            try:
                nii = int(nv)
            except (TypeError, ValueError):
                nii = -1
            if 1 <= nii <= 999 and float(nii) == float(nv):
                n = nii
    try:
        pr = initial_h_values
        if pr is not None:
            pa = np.asarray(pr, dtype=np.float64)
            if pa.ndim == 1 and len(pa) > 0 and np.all(np.isfinite(pa)) and 1 <= len(pa) <= 999:
                parent = pa.copy()
                if 'n_points' not in kwargs:
                    n = len(parent)
    except (NameError, Exception):
        pass
    m = n / 2.0
    if parent is not None and len(parent) != n:
        parent = interpolate_to_n(parent, n)
    if parent is not None:
        parent_feas = project_onto_K(parent, m)
    else:
        parent_feas = np.full(n, 0.5)
    best_x = parent_feas.copy()
    best_score = exact_score(best_x, n)
    tau_schedule = [0.02, 0.005, 0.001, 0.0002, 0.00005, 0.00001]
    if n <= 64:
        max_iters, max_starts, pert_rounds = 80, 10, 2
        pert_sigs = [0.05, 0.18]
    elif n <= 160:
        max_iters, max_starts, pert_rounds = 50, 6, 1
        pert_sigs = [0.08, 0.20]
    elif n <= 400:
        max_iters, max_starts, pert_rounds = 24, 4, 1
        pert_sigs = [0.10]
    else:
        max_iters, max_starts, pert_rounds = 12, 2, 1
        pert_sigs = [0.10]
    starts = generate_starts(n, m, parent, seed)
    starts = starts[:max_starts]
    rng = np.random.default_rng(seed)
    for s in starts:
        xo, sc = local_optimize(s, n, m, tau_schedule, max_iters)
        if math.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_x = xo.copy()
    for rnd in range(pert_rounds):
        score_start = best_score
        for sig in pert_sigs:
            xp = project_onto_K(best_x + sig * rng.standard_normal(n), m)
            xo, sc = local_optimize(xp, n, m, tau_schedule, max_iters)
            if math.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_x = xo.copy()
        if best_score > score_start - 1e-9:
            break
    best_x = np.clip(best_x, 0.0, 1.0)
    if abs(float(np.sum(best_x)) - m) > 1e-9:
        best_x = project_onto_K(best_x, m)
    c5_bound = exact_score(best_x, n)
    if not math.isfinite(c5_bound):
        best_x = np.full(n, 0.5)
        c5_bound = exact_score(best_x, n)
    return best_x, c5_bound, n
