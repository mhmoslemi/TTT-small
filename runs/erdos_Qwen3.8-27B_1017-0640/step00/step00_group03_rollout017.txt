import numpy as np

def project_capped_simplex(v, n):
    if v.shape != (n,) or not np.all(np.isfinite(v)):
        return np.full(n, 0.5)
    v = np.clip(v, -1e6, 1e6)
    target = n / 2.0
    lo, hi = v.min() - 1.0, v.max()
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    q = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    r = target - q.sum()
    if abs(r) > 1e-13:
        for i in range(n):
            if abs(r) <= 1e-13:
                break
            if r > 0:
                inc = min(r, 1.0 - q[i])
                q[i] += inc
                r -= inc
            else:
                dec = min(-r, q[i])
                q[i] -= dec
                r += dec
    if (not np.all(np.isfinite(q)) or q.min() < -1e-14 or q.max() > 1.0 + 1e-14
            or abs(q.sum() - target) > 1e-10):
        return np.full(n, 0.5)
    return np.clip(q, 0.0, 1.0)

def validate_h(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and h.min() >= -1e-14
            and h.max() <= 1.0 + 1e-14 and abs(h.sum() - n / 2.0) <= 1e-10)

def exact_score(h, n):
    if not validate_h(h, n):
        return float('inf')
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    if c.shape != (2 * n - 1,) or not np.all(np.isfinite(c)):
        return float('inf')
    return float(c.max())

def compute_lags(h, n):
    dx = 2.0 / n
    vals = np.empty(2 * n - 1)
    for d in range(n):
        m = n - d
        vals[n - 1 - d] = np.dot(h[:m], 1.0 - h[d:]) * dx
        vals[n - 1 + d] = np.dot(h[d:], 1.0 - h[:m]) * dx
    return vals

def compute_jac(h, n):
    dx = 2.0 / n
    J = np.zeros((2 * n - 1, n))
    J[n - 1, :] = (1.0 - 2.0 * h) * dx
    for d in range(1, n):
        m = n - d
        gA = np.zeros(n)
        gA[d:] += 1.0 - h[:m]
        gA[:m] -= h[d:]
        J[n + d - 1, :] = gA * dx
        gB = np.zeros(n)
        gB[:m] += 1.0 - h[d:]
        gB[d:] -= h[:m]
        J[n - 1 - d, :] = gB * dx
    return J

def surrogate_eval(c, tau):
    cmax = c.max()
    q = np.exp((c - cmax) / tau)
    qs = q.sum()
    if not np.isfinite(qs) or qs <= 0.0:
        return None
    return cmax + tau * np.log(qs)

def pgd_search(h0, n, inc_h, inc_score, rng):
    temps = [0.02, 0.005, 0.001, 0.0002, 4e-5, 8e-6]
    niter = 100 if n <= 128 else 40
    h = h0.copy()
    for tau in temps:
        alpha = 0.5
        stall = 0
        for _ in range(niter):
            c = compute_lags(h, n)
            J = compute_jac(h, n)
            F = surrogate_eval(c, tau)
            if F is None:
                break
            p = np.exp((c - c.max()) / tau)
            p /= p.sum()
            g = J.T @ p
            if not np.all(np.isfinite(g)):
                break
            done = False
            for _ in range(24):
                ht = project_capped_simplex(h - alpha * g, n)
                if not np.all(np.isfinite(ht)):
                    alpha *= 0.5
                    continue
                Ft = surrogate_eval(compute_lags(ht, n), tau)
                if Ft is None:
                    alpha *= 0.5
                    continue
                dlt = ht - h
                if Ft <= F + 1e-4 * (g @ dlt) + 1e-12 * max(1.0, abs(F)):
                    h = ht
                    alpha = min(1.5 * alpha, 2.0)
                    if np.abs(dlt).max() <= 1e-8:
                        done = True
                    elif abs(F - Ft) <= 1e-12 * max(1.0, abs(F)):
                        stall += 1
                    else:
                        stall = 0
                    s = exact_score(h, n)
                    if s < inc_score - 1e-12:
                        inc_h[:] = h
                        inc_score = s
                    break
                alpha *= 0.5
            if done or stall >= 8:
                break
    return inc_h, inc_score
