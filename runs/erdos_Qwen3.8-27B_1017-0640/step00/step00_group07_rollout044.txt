        Q[2*s] = dx * np.sum(h[s:]*(1.0-h[:p]) + 0.5*(d[:p]**2 + d[s:]**2))
    return Q

        rb = np.zeros(n)
        rb[:p] += dx * (-h[s:] + d[:p])
        rb[s:] += dx * (1.0 - h[:p] + d[s:])
