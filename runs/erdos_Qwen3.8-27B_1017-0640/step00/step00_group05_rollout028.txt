            trial = fix_residual(trial, m)
            if trial is None:
                continue
            ts = score_h(trial, N)
            if ts is not None and ts < bts - 1e-12:
                bts = ts
                bt = trial.copy()
        if bt is not None:
            h = bt
            cur_score = bts
    return h, cur_score


def run(seed=42, **kwargs):
    N = 70
    base = None
    try:
        _arr = np.array(initial_h_values, dtype=np.float64)
        if _arr.ndim == 1 and _arr.size > 0:
            N = int(_arr.size)
            if np.all(np.isfinite(_arr)):
                m_tmp = N / 2.0
                if is_feasible(_arr, N):
                    base = _arr.copy()
                else:
                    p = project_box_sum(_arr, m_tmp)
                    if p is not None and is_feasible(p, N):
                        base = p
    except Exception:
        N = 70
        base = None
    if N < 1:
        N = 70
    m = N / 2.0
    neutral = np.full(N, 0.5)
    if base is None:
        base = neutral.copy()
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    starts = make_starts(N, m, base, rng)
    gh = neutral.copy()
    gs = score_h(gh, N)
    if gs is None:
        gs = 1.0
    ps = score_h(base, N)
    if ps is not None and ps < gs - 1e-12:
        gh, gs = base.copy(), ps
    for s in starts:
        ss = score_h(s, N)
        if ss is not None and ss < gs - 1e-12:
            gh, gs = s.copy(), ss
    if 1 < N <= 999:
        if N <= 128:
            B = 800
        elif N <= 256:
            B = 400
        elif N <= 512:
            B = 150
        else:
            B = 40
        for s in starts:
            hr, sr = anneal_trajectory(s, N, m, B, rng)
            if hr is not None and sr is not None and sr < gs - 1e-12:
                gh, gs = hr.copy(), sr
        hp, sp = pattern_poll(gh, N, m, rng)
        if hp is not None and sp is not None and sp < gs - 1e-12:
            gh, gs = hp.copy(), sp
    proj = project_box_sum(gh, m)
    if proj is not None:
        fs = score_h(proj, N)
        if fs is not None:
            gh, gs = proj, fs
    for fb in [base, neutral]:
        fs = score_h(fb, N)
        if fs is not None and fs < gs - 1e-12:
            gh, gs = fb.copy(), fs
    fs = score_h(gh, N)
    if fs is None:
        gh = neutral.copy()
        fs = score_h(gh, N)
    if fs is None:
        fs = 1.0
    return gh, float(fs), N
