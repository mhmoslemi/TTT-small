                    fallback = np.clip(arr, 0.0, 1.0)
                    if abs(float(np.sum(fallback)) - n/2.0) > 5e-13 * max(1, n):
                        rep = project_capped_simplex(fallback, n)
                        if rep is not None:
                            fallback = rep
                else:
                    rep = project_capped_simplex(arr, n)
                    if rep is not None:
                        fallback = rep
        except Exception:
            fallback = None
    if fallback is None:
        n = 42
        fallback = np.zeros(n)
        half = n // 2
        fallback[:half] = 1.0
        if n % 2 == 1:
            fallback[half] = 0.5
    fb_score = exact_score(fallback, n)
    if fb_score is None:
        fb_score = 1.0
    best_h = fallback.copy()
    best_score = fb_score
    starts = []
    seen = []
    def try_add(vec):
        if len(starts) >= n_starts: return
        v = project_capped_simplex(vec.copy(), n) if not (v is not None and np.all(np.isfinite(vec)) and np.all(vec >= 0) and np.all(vec <= 1) and abs(float(np.sum(vec)) - n/2.0) < 1e-10*max(1,n)) else vec.copy()
        if v is None or not np.all(np.isfinite(v)): return
        for s in seen:
            if np.max(np.abs(v - s)) < 1e-9: return
        starts.append(v.copy())
        seen.append(v.copy())
    try_add(fallback.copy())
    try_add(1.0 - fallback.copy())
    try_add(fallback[::-1].copy())
    blk = np.zeros(n); blk[:n//2] = 1.0; try_add(blk)
    try_add(np.ones(n) * 0.5)
    try_add(0.5 + 0.25 * np.cos(2.0 * math.pi * np.arange(n) / n))
    try_add(fallback + 0.15 * rng.standard_normal(n))
    try_add(rng.uniform(-0.25, 1.25, size=n))
    for sd in [0.05, 0.20, 0.35, 0.50]:
        try_add(fallback + sd * rng.standard_normal(n))
    attempts = 0
    while len(starts) < n_starts and attempts < 8 * n_starts + 8:
        attempts += 1
        try_add(rng.standard_normal(n))
    for sv in starts:
        try:
            sc = exact_score(sv, n)
            if sc is None: continue
            if sc < best_score:
                best_score = sc; best_h = sv.copy()
            corr = np.correlate(sv, 1.0 - sv, mode='full')
            nh, nc, ns = local_search(sv.copy(), corr, sc, n, rng, max_sweeps)
            if ns < best_score:
                best_score = ns; best_h = nh.copy()
        except Exception:
            continue
    final = best_h.copy()
    fs = exact_score(final, n)
    if fs is None:
        rep = project_capped_simplex(final, n)
        if rep is not None:
            final = rep
            fs = exact_score(final, n)
    if fs is None:
        final = fallback.copy(); fs = fb_score
    return (final, float(fs), int(n))
