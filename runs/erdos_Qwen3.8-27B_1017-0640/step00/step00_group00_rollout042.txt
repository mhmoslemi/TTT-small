    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n), dtype=np.float64)
    G[n - 1] = dx * (1.0 - 2.0 * h)
    for p in range(1, n):
        G[n - 1 + p, p:] += dx * (1.0 - h[:-p])
        G[n - 1 + p, :-p] -= dx * h[p:]
        G[n - 1 - p, :-p] += dx * (1.0 - h[p:])
        G[n - 1 - p, p:] -= dx * h[:-p]
    return G


def solve_mm_cvxpy(x, q, G, dx, n, L):
    D = cp.Variable(n)
    s_v = cp.Variable(non_negative=True)
    t_v = cp.Variable()
    q_p = cp.Parameter((L,))
    G_p = cp.Parameter((L, n))
    lo_p = cp.Parameter((n,))
    hi_p = cp.Parameter((n,))
    cons = [D >= lo_p, D <= hi_p, cp.sum(D) == 0,
            s_v >= cp.sum_squares(D), q_p + G_p @ D + dx * s_v <= t_v]
    prob = cp.Problem(cp.Minimize(t_v), cons)
    for solver in ['ECOS', 'SCS', 'CVXOPT']:
        try:
            q_p.value = q
            G_p.value = G
            lo_p.value = -x
            hi_p.value = 1.0 - x
            prob.solve(solver=solver, verbose=False,
                       abstol=1e-8, reltol=1e-8, feastol=1e-8, max_iters=200)
            if prob.status in ('optimal', 'optimal_inaccurate') and D.value is not None:
                d = np.asarray(D.value, dtype=np.float64)
                if d.shape == (n,) and np.all(np.isfinite(d)):
                    return d
        except Exception:
            pass
    return None


def _slsqp_obj(z, args):
    return z[-1]


def _slsqp_eq(z, args):
    return np.sum(z[:args[4]])


def _slsqp_ineq(z, args):
    x, q, G, dx, n, L = args
    d = z[:n]
    s_val = z[n]
    t_val = z[n + 1]
    return np.concatenate(([s_val - d.dot(d)], t_val - q - G.dot(d) - dx * s_val))


def solve_mm_slsqp(x, q, G, dx, n, L):
    m_n = n + 2
    z0 = np.zeros(m_n)
    z0[n + 1] = float(np.max(q))
    bounds = [(-x[i], 1.0 - x[i]) for i in range(n)]
    bounds.append((0.0, float(n)))
    bounds.append((None, None))
    args = (x, q, G, dx, n, L)
    cons = [{'type': 'eq', 'fun': _slsqp_eq, 'args': args},
            {'type': 'ineq', 'fun': _slsqp_ineq, 'args': args}]
    maxit = 100 if n <= 64 else (60 if n <= 160 else 35)
    try:
        res = minimize(_slsqp_obj, z0, args=args, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxit, 'ftol': 1e-9, 'disp': False})
        if res.success and np.all(np.isfinite(res.x[:n])):
            return res.x[:n]
    except Exception:
        pass
    return None


def mm_optimize(x0, n, max_iter):
    x = x0.copy()
    c_best = exact_score(x, n)
    if c_best is None:
        return x, None
    for _ in range(max_iter):
        q = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        G = compute_grad(x, n)
        L = 2 * n - 1
        dx = 2.0 / n
        d = solve_mm_cvxpy(x, q, G, dx, n, L) if HAVE_CP else None
        if d is None:
            d = solve_mm_slsqp(x, q, G, dx, n, L)
        if d is None:
            break
        y = project_simplex(x + d, n)
        if y is None:
            break
        best_y, best_c = None, None
        for theta in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            cand = project_simplex(x + theta * (y - x), n)
            if cand is None:
                continue
            c = exact_score(cand, n)
            if c is not None and (best_c is None or c < best_c):
                best_c = c
                best_y = cand
        if best_y is not None and best_c < c_best - 1e-12:
            if c_best - best_c < 1e-10:
                x = best_y
                c_best = best_c
                break
            x = best_y
            c_best = best_c
        else:
            break
    return x, c_best


def make_starts(n, seed):
    rng = np.random.default_rng(seed)
    m = n / 2.0
    starts = []
    try:
        parent = np.asarray(initial_h_values, dtype=np.float64)
    except (NameError, TypeError, ValueError):
        parent = None
    if parent is not None:
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            if parent.size == n:
                p = project_simplex(parent, n)
                if p is not None:
                    starts.append(p)
                comp = 1.0 - parent
                starts.append(project_simplex(comp, n))
                pert = project_simplex(parent + 0.10 * rng.standard_normal(n), n)
                if pert is not None:
                    starts.append(pert)
            else:
                idx = np.linspace(0, 1, parent.size)
                new_idx = np.linspace(0, 1, n)
                interp = np.interp(new_idx, idx, parent)
                p = project_simplex(interp, n)
                if p is not None:
                    starts.append(p)
    nb = int(np.floor(m))
    blk = np.zeros(n)
    blk[:nb] = 1.0
    if m - nb > 1e-12:
        blk[nb] = m - nb
    blk = project_simplex(blk, n)
    if blk is not None:
        starts.append(blk)
    starts.append(project_simplex(1.0 - blk if False else blk[::-1], n))
    rv = project_simplex(rng.uniform(size=n), n)
    if rv is not None:
        starts.append(rv)
    seen = []
    for s in starts:
        if s is None:
            continue
        dup = False
        for t in seen:
            if np.max(np.abs(s - t)) < 1e-10:
                dup = True
                break
        if not dup:
            seen.append(s)
    lim = 6 if n <= 64 else (4 if n <= 160 else (3 if n <= 400 else 2))
    return seen[:lim]


def run(seed=42, **kwargs):
    n = 45
    try:
        parent = np.asarray(initial_h_values, dtype=np.float64)
        if parent.ndim == 1 and 1 <= parent.size <= 999:
            n = parent.size
    except (NameError, TypeError, ValueError):
        pass
    if 'n_points' in kwargs:
        nv = kwargs['n_points']
        if not isinstance(nv, bool) and isinstance(nv, (int, np.integer)) and 1 <= int(nv) <= 999:
            n = int(nv)
    n = int(n)
    dx = 2.0 / n
    starts = make_starts(n, seed)
    if not starts:
        starts.append(np.full(n, 0.5))
    best_h = None
    best_c = None
    max_iter = 40 if n <= 64 else (25 if n <= 160 else (12 if n <= 400 else 5))
    for s in starts:
        x, c = mm_optimize(s, n, max_iter)
        if c is not None and (best_c is None or c < best_c - 1e-12):
            best_h = x
            best_c = c
    # restart perturbations
    if best_h is not None and best_c is not None:
        r = best_c
        rng2 = np.random.default_rng(seed + 1)
        scales = [0.04, 0.15] if n <= 64 else ([0.08] if n <= 160 else ([0.12] if n <= 400 else [0.15]))
        for sc in scales:
            pert = project_simplex(best_h + sc * rng2.standard_normal(n), n)
            if pert is not None:
                x, c = mm_optimize(pert, n, max_iter)
                if c is not None and c < best_c - 1e-12:
                    best_h = x
                    best_c = c
        if best_c > r - 1e-9:
            pass
    if best_h is None:
        best_h = np.full(n, 0.5)
    best_h = project_simplex(best_h, n)
    if best_h is None:
        best_h = np.full(n, 0.5)
    c5 = exact_score(best_h, n)
    if c5 is None:
        best_h = np.full(n, 0.5)
        c5 = exact_score(best_h, n)
    return (best_h, c5, n)
