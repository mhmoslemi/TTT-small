    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-14:
        return 0.0
    nn = n - 1
    # B for all lags
    B = np.zeros(2 * n - 1)
    for p in range(2 * n - 1):
        k = p - nn
        b = 0.0
        a1 = i - k
        if 0 <= a1 < n:
            b += 1.0 - h[a1]
        a2 = j - k
        if 0 <= a2 < n:
            b -= 1.0 - h[a2]
        a3 = i + k
        if 0 <= a3 < n:
            b -= h[a3]
        a4 = j + k
        if 0 <= a4 < n:
            b += h[a4]
        B[p] = b
    k0, kp, km = nn, nn + (j - i), nn + (i - j)
    # Filter valid special lags
    sp = [p for p in (k0, kp, km) if 0 <= p < 2 * n - 1]
    sp_set = set(sp)
    aff = [p for p in range(2 * n - 1) if p not in sp_set]
    # Build upper hull of affine lines
    if aff:
        slopes = B[aff]
        intercepts = q[aff]
        order = np.lexsort((intercepts, slopes))
        ls = slopes[order]
        cs = intercepts[order]
        hull = []
        for idx in range(len(ls)):
            s, c = float(ls[idx]), float(cs[idx])
            if hull and abs(s - hull[-1][0]) < 1e-15:
                if c > hull[-1][1]:
                    hull.pop()
                else:
                    continue
            while hull:
                s2, c2, x2 = hull[-1]
                if abs(s - s2) < 1e-15:
                    if c > c2:
                        hull.pop()
                    else:
                        break
                    continue
                xi = (c2 - c) / (s - s2)
                if xi > x2 + 1e-14:
                    break
                hull.pop()
            if not hull:
                hull.append((s, c, float('-inf')))
            else:
                s2, c2, x2 = hull[-1]
                xi = (c2 - c) / (s - s2)
                hull.append((s, c, max(xi, x2)))
    else:
        hull = [(0.0, 0.0, float('-inf'))]
    # Special polynomials: (A, B, D)
    polys_special = []
    for p in sp:
        if p == k0:
            A = -2.0
        else:
            A = 1.0
        polys_special.append((A, B[p], q[p]))
    # Evaluate on each hull segment
    best_val = float('inf')
    best_d = 0.0
    nseg = len(hull)
    for si in range(nseg):
        s_a, c_a, xs = hull[si]
        xe = hull[si + 1][2] if si + 1 < nseg else float('inf')
        a_lo = max(lo, xs)
        a_hi = min(hi, xe)
        if a_lo >= a_hi:
            continue
        # Polynomials on this segment: active line + specials
        polys = [(0.0, s_a, c_a)] + polys_special
        # Candidate points
        cands = [a_lo, a_hi]
        if lo <= 0.0 <= hi:
            cands.append(0.0)
        for (A, Bv, D) in polys:
            if abs(A) > 1e-15:
                st = -Bv / (2.0 * A)
                if a_lo - 1e-12 <= st <= a_hi + 1e-12:
                    cands.append(max(a_lo, min(a_hi, st)))
        for ai in range(len(polys)):
            for bi in range(ai + 1, len(polys)):
                da = polys[ai][0] - polys[bi][0]
                db = polys[ai][1] - polys[bi][1]
                dd = polys[ai][2] - polys[bi][2]
                if abs(da) < 1e-15:
                    if abs(db) > 1e-15:
                        r = -dd / db
                        if a_lo - 1e-12 <= r <= a_hi + 1e-12:
                            cands.append(max(a_lo, min(a_hi, r)))
                else:
                    disc = db * db - 4.0 * da * dd
                    scale = abs(db * db) + abs(4.0 * da * dd) + 1.0
                    if disc < -1e-12 * scale:
                        continue
                    disc = max(0.0, disc)
                    sq = math.sqrt(disc)
                    z = -0.5 * (db + math.copysign(sq, db))
                    if abs(z) > 1e-15:
                        for r in (z / da, dd / z):
                            if a_lo - 1e-12 <= r <= a_hi + 1e-12:
                                cands.append(max(a_lo, min(a_hi, r)))
                    else:
                        r = -db / (2.0 * da)
                        if a_lo - 1e-12 <= r <= a_hi + 1e-12:
                            cands.append(max(a_lo, min(a_hi, r)))
        # Evaluate
        for d in cands:
            d = max(a_lo, min(a_hi, d))
            mx = -float('inf')
            for (A, Bv, D) in polys:
                v = A * d * d + Bv * d + D
                if v > mx:
                    mx = v
            if mx < best_val - 1e-14 or (abs(mx - best_val) <= 1e-14 and abs(d) < abs(best_d)):
                best_val = mx
                best_d = d
    if not np.isfinite(best_val):
        return 0.0
    # Check predicted improvement
    cur_max = float(np.max(q))
    if (cur_max - best_val) * (2.0 / n) <= 1e-12:
        return 0.0
    return best_d

def _sweep(h, q, sc, pair_list, n, m, rng):
    h = h.copy()
    for pi in range(len(pair_list)):
        try:
            i, j = pair_list[pi]
            d = _line_min(h, i, j, q, n)
            if abs(d) < 1e-13:
                continue
            h[i] += d
            h[j] -= d
            if not np.all(np.isfinite(h)):
                h[i] -= d; h[j] += d
                continue
            h[i] = max(0.0, min(1.0, h[i]))
            h[j] = max(0.0, min(1.0, h[j]))
            r = m - math.fsum(h.tolist())
            for _ in range(3):
                if abs(r) < 1e-12:
                    break
                for idx in range(n):
                    if r > 0:
                        a = min(r, 1.0 - h[idx])
                        h[idx] += a; r -= a
                    else:
                        a = min(-r, h[idx])
                        h[idx] -= a; r += a
            if not _feasible(h, n, m):
                # revert
                h[i] = max(0.0, min(1.0, h[i] - d))
                h[j] = max(0.0, min(1.0, h[j] + d))
                r = m - math.fsum(h.tolist())
                for _ in range(2):
                    if abs(r) < 1e-12: break
                    for idx in range(n):
                        if r > 0:
                            a = min(r, 1.0 - h[idx]); h[idx] += a; r -= a
                        else:
                            a = min(-r, h[idx]); h[idx] -= a; r += a
                continue
            q_new = np.correlate(h, 1.0 - h, mode='full')
            if not np.all(np.isfinite(q_new)):
                continue
            sc_new = float(np.max(q_new)) * (2.0 / n)
            if sc_new < sc - 1e-12:
                q = q_new
                sc = sc_new
        except Exception:
            continue
    return h, q, sc
