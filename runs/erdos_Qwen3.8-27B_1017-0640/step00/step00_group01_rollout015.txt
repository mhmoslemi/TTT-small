                x[i] -= adj
                r += adj
    if not np.all(np.isfinite(x)) or np.any(x < -1e-12) or np.any(x > 1.0 + 1e-12):
        return None
    if abs(math.fsum(x) - s) > 1e-9:
        return None
    return x

def is_feasible(h, n):
    if h.ndim != 1 or h.size != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(math.fsum(h) - n / 2.0) > 1e-9:
        return False
    return True

def exact_score(h, n):
    try:
        corr = np.correlate(h, 1.0 - h, mode='full')
        val = float(np.max(corr) * (2.0 / n))
        return val if math.isfinite(val) else None
    except Exception:
        return None

def lag_values(h, n):
    dx = 2.0 / n
    vals = np.empty(2 * n - 1)
    for idx, d in enumerate(range(-(n - 1), n)):
        start = max(0, -d)
        stop = min(n, n - d)
        vals[idx] = np.dot(h[start:stop], 1.0 - h[start + d:stop + d])
    return vals * dx

def smooth_value_grad(h, tau, need_grad, n):
    a = lag_values(h, n)
    if not np.all(np.isfinite(a)):
        return None, None
    A = float(np.max(a))
    e = np.exp((a - A) / tau)
    Z = float(np.sum(e))
    if not math.isfinite(Z) or Z <= 0.0:
        return None, None
    F = A + tau * math.log(Z)
    if not math.isfinite(F):
        return None, None
    if not need_grad:
        return F, None
    p = e / Z
    g = np.zeros(n)
    dx = 2.0 / n
    for idx, d in enumerate(range(-(n - 1), n)):
        start = max(0, -d)
        stop = min(n, n - d)
        w = dx * p[idx]
        src = h[start:stop]
        tgt = h[start + d:stop + d]
        g[start:stop] += w * (1.0 - tgt)
        g[start + d:stop + d] -= w * src
    if not np.all(np.isfinite(g)):
        return None, None
    return F, g

def projected_smooth_search(start, temps, iters_per_temp, n):
    current = np.array(start, dtype=np.float64, copy=True)
    best = np.array(start, dtype=np.float64, copy=True)
    best_score = exact_score(best, n)
    if best_score is None:
        best_score = 1e10
    for tau in temps:
        current = np.array(best, dtype=np.float64, copy=True)
        F_cur, g = smooth_value_grad(current, tau, True, n)
        if F_cur is None or g is None:
            continue
        eta = 1.0
        delta = np.zeros(n)
        for _it in range(iters_per_temp):
            updated = False
            for _bt in range(25):
                y = project_capped_simplex(current - eta * g)
                if y is None or not is_feasible(y, n):
                    eta /= 2.0
                    continue
                s_y = exact_score(y, n)
                if s_y is not None and s_y < best_score - 1e-12:
                    best_score = s_y
                    best = np.array(y, dtype=np.float64, copy=True)
                delta = y - current
                d2 = float(np.dot(delta, delta))
                if d2 <= 1e-24:
                    updated = True
                    break
                gd = float(np.dot(g, delta))
                F_y, _ = smooth_value_grad(y, tau, False, n)
                if F_y is None:
                    eta /= 2.0
                    continue
                rhs = F_cur + 1e-4 * min(gd, 0.0) + 1e-14 * max(1.0, abs(F_cur))
                if F_y <= rhs:
                    current = y
                    F_cur = F_y
                    F_new, g = smooth_value_grad(current, tau, True, n)
                    if F_new is None or g is None:
                        break
                    F_cur = F_new
                    eta = min(1.5 * eta, 10.0)
                    updated = True
                    break
                else:
                    eta /= 2.0
            if not updated:
                break
            if np.max(np.abs(delta)) <= 1e-9:
                break
    return best, best_score

def run(seed=42, **kwargs):
    try:
        seed = int(seed)
        if seed < 0 or seed > 2**32 - 1:
            seed = 42
    except (ValueError, TypeError, OverflowError):
        seed = 42
    rng = np.random.default_rng(seed)
    parent = None
    raw = globals().get('initial_h_values', None)
    if raw is not None:
        try:
            arr = np.array(raw, dtype=np.float64, copy=True)
            if arr.ndim == 1 and 2 <= arr.size < 1000:
                if np.all(np.isfinite(arr)) and arr.min() >= -1e-7 and arr.max() <= 1.0 + 1e-7:
                    np_ = arr.size
                    proj = project_capped_simplex(arr)
                    if proj is not None and is_feasible(proj, np_):
                        if exact_score(proj, np_) is not None:
                            parent = proj
        except Exception:
            parent = None
    n = parent.size if parent is not None else 47
    const = np.full(n, 0.5, dtype=np.float64)
    const_proj = project_capped_simplex(const)
    if const_proj is None:
        const_proj = const.copy()
    const_score = exact_score(const_proj, n)
    if const_score is None:
        const_score = 0.5
    if n <= 64:
        temps = (0.04, 0.01, 0.0025, 0.0005, 0.0001)
        iper = 80
        mxs = 8
        rsts = (0.04, 0.12)
    elif n <= 128:
        temps = (0.04, 0.01, 0.0025, 0.0005)
        iper = 45
        mxs = 5
        rsts = (0.06,)
    elif n <= 512:
        temps = (0.03, 0.006, 0.001)
        iper = 25
        mxs = 3
        rsts = ()
    else:
        temps = (0.02, 0.003)
        iper = 12
        mxs = 2
        rsts = ()
    base = parent if parent is not None else const_proj
    starts = []
    if parent is not None:
        starts.append(parent)
    starts.append(const_proj)
    idx = np.arange(n, dtype=np.float64)
    for freq in [1, 2]:
        phi = rng.uniform(0.0, 2.0 * math.pi)
        v = 0.5 + 0.30 * np.sin(2.0 * math.pi * freq * (idx + 0.5) / n + phi)
        p = project_capped_simplex(v)
        if p is not None and is_feasible(p, n):
            starts.append(p)
    for sc in [0.06, 0.18]:
        v = base + sc * rng.standard_normal(n)
        p = project_capped_simplex(v)
        if p is not None and is_feasible(p, n):
            starts.append(p)
    for _ in range(2):
        v = rng.random(n)
        p = project_capped_simplex(v)
        if p is not None and is_feasible(p, n):
            starts.append(p)
    if parent is None:
        phi = rng.uniform(0.0, 2.0 * math.pi)
        v = 0.5 + 0.30 * np.cos(2.0 * math.pi * 3.0 * (idx + 0.5) / n + phi)
        p = project_capped_simplex(v)
        if p is not None and is_feasible(p, n):
            starts.append(p)
    unique_starts = []
    for s in starts:
        dup = False
        for us in unique_starts:
            if np.max(np.abs(np.asarray(s) - np.asarray(us))) <= 1e-9:
                dup = True
                break
        if not dup:
            unique_starts.append(s)
            if len(unique_starts) >= mxs:
                break
    global_best = const_proj.copy()
    global_best_score = const_score
    if parent is not None:
        ps = exact_score(parent, n)
        if ps is not None and ps < global_best_score - 1e-12:
            global_best = parent.copy()
            global_best_score = ps
    for s in unique_starts:
        sc = exact_score(s, n)
        if sc is not None and sc < global_best_score - 1e-12:
            global_best = s.copy()
            global_best_score = sc
        try:
            lb, lbs = projected_smooth_search(s, temps, iper, n)
            if lbs < global_best_score - 1e-12:
                global_best = lb
                global_best_score = lbs
        except Exception:
            pass
    for rs in rsts:
        try:
            v = global_best + rs * rng.standard_normal(n)
            proj = project_capped_simplex(v)
            if proj is not None and is_feasible(proj, n):
                ps = exact_score(proj, n)
                if ps is not None and ps < global_best_score - 1e-12:
                    global_best = proj.copy()
                    global_best_score = ps
                lb, lbs = projected_smooth_search(proj, temps, iper, n)
                if lbs < global_best_score - 1e-12:
                    global_best = lb
                    global_best_score = lbs
        except Exception:
            pass
    final = project_capped_simplex(global_best)
    if final is None or not is_feasible(final, n):
        final = global_best
    fs = exact_score(final, n)
    if fs is None or not math.isfinite(fs):
        if parent is not None:
            ps = exact_score(parent, n)
            if ps is not None and math.isfinite(ps):
                final = parent
                fs = ps
        if fs is None or not math.isfinite(fs):
            final = const_proj
            fs = exact_score(final, n)
    if fs is None or not math.isfinite(fs):
        final = const_proj
        fs = 0.5
    return (np.array(final, dtype=np.float64), float(fs), int(n))
