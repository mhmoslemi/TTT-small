def capped_simplex_project(v, n):
    v = np.clip(np.asarray(v, dtype=np.float64).ravel(), -1e6, 1e6)
    target = n / 2.0
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = (lo + hi) / 2.0
        q = np.clip(v - mid, 0.0, 1.0)
        if np.sum(q) > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # residual correction
    for _ in range(200):
        resid = target - np.sum(h)
        if abs(resid) < 1e-13:
            break
        if resid > 0:
            cap = 1.0 - h
            idx = np.where(cap > 1e-12)[0]
            if len(idx) == 0:
                break
            amt = min(resid, cap[idx].sum())
            h[idx] += amt / len(idx)
        else:
            idx = np.where(h > 1e-12)[0]
            if len(idx) == 0:
                break
            amt = min(-resid, h[idx].sum())
            h[idx] -= amt / len(idx)
    h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
from scipy.optimize import minimize

def capped_simplex_project(v, n):
    v = np.clip(np.asarray(v, dtype=np.float64).ravel(), -1e6, 1e6)
    if v.size != n:
        v = v[:n] if v.size > n else np.pad(v, (0, n - v.size))
    target = n / 2.0
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = (lo + hi) / 2.0
        q = np.clip(v - mid, 0.0, 1.0)
        if np.sum(q) > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(300):
        resid = target - np.sum(h)
        if abs(resid) < 1e-13:
            break
        if resid > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if len(idx) == 0:
                break
            amt = min(resid, (1.0 - h[idx]).sum())
            h[idx] += amt / len(idx)
        else:
            idx = np.where(h > 1e-12)[0]
            if len(idx) == 0:
                break
            amt = min(-resid, h[idx].sum())
            h[idx] -= amt / len(idx)
    return np.clip(h, 0.0, 1.0)

def exact_score(h):
    n = len(h)
    if n == 0:
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full')
    if len(c) != 2 * n - 1:
        return float('inf')
    s = float(np.max(c) * (2.0 / n))
    return s if np.isfinite(s) else float('inf')

def validity_check(h, n, tol=1e-10):
    if len(h) != n or not np.all(np.isfinite(h)):
        return False
    if np.min(h) < -tol or np.max(h) > 1.0 + tol:
        return False
    if abs(np.sum(h) - n / 2.0) > tol:
        return False
    return True

def lag_jacobian(h):
    n = len(h)
    dx = 2.0 / n
    c = np.zeros(2 * n - 1)
    J = np.zeros((2 * n - 1, n))
    for j in range(2 * n - 1):
        k = j - (n - 1)
        if k >= 0:
            d = k
            c[j] = np.sum(h[d:] * (1.0 - h[:n - d]))
            if d == 0:
                J[j, :] = 1.0 - 2.0 * h
            else:
                J[j, d:n] = (1.0 - h[:n - d])
                J[j, 0:n - d] -= h[d:]
        else:
            d = -k
            c[j] = np.sum(h[:n - d] * (1.0 - h[d:]))
            if d == 0:
                J[j, :] = 1.0 - 2.0 * h
            else:
                J[j, 0:n - d] = (1.0 - h[d:])
                J[j, d:n] -= h[:n - d]
    c *= dx
    J *= dx
    return c, J

def slsqp_obj(z):
    return z[-1]

def slsqp_obj_grad(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g

def slsqp_eq(z):
    return np.sum(z[:-1]) - (len(z) - 1) / 2.0

def slsqp_eq_grad(z):
    g = np.zeros(len(z))
    g[:-1] = 1.0
    return g

def slsqp_ineq(z):
    h = z[:-1]
    c, _ = lag_jacobian(h)
    return z[-1] - c

def slsqp_ineq_grad(z):
    h = z[:-1]
    n = len(h)
    _, J = lag_jacobian(h)
    Gr = np.zeros((2 * n - 1, n + 1))
    Gr[:, :n] = -J
    Gr[:, n] = 1.0
    return Gr

def solve_slsqp(start, n, maxiter, ftol):
    t0 = exact_score(start) + 1e-8
    z0 = np.concatenate([start, [t0]])

            fun=slsqp_obj, x0=z0, jac=slsqp_obj_grad,
            method='SLSQP', bounds=bounds,
            constraints=[
                {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_grad},
                {'type': 'ineq', 'fun': slsqp_ineq, 'jac': slsqp_ineq_grad}

            s_cand = exact_score(h_cand)

    return None, float('inf')
