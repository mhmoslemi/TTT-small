def _score(h):
    n = len(h)
    try:
        c = np.correlate(h, 1.0 - h, mode='full')
        s = float(np.max(c)) * (2.0 / n)
        if not math.isfinite(s):
            return None
        return s
    except Exception:
        n = len(h)
        best = 0.0
        for d in range(-n + 1, n):
            lo = max(0, -d)
            hi = min(n, n - d)
            q = 0.0
            for i in range(lo, hi):
                q += h[i] * (1.0 - h[i + d])
            if q > best:
                best = q
        s = best * (2.0 / n)
        return s if math.isfinite(s) else None

def _feasible(h, n):
    if h.ndim != 1 or len(h) != n or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-9) or np.any(h > 1.0 + 1e-9):
        return False
    if abs(math.fsum(h) - n / 2.0) > 1e-9:
        return False
    return True

def _proj_capped(v):

    s = n / 2.0

    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = lo + (hi - lo) / 2.0
        m = math.fsum(np.clip(v - mid, 0.0, 1.0))
        if m > s:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(4):
        r = s - math.fsum(x)
        if abs(r) <= 5e-15:

                    r -= add
                    changed = True

    return x

def _sdp_solve(n, rng):
    try:
        import cvxpy as cp
    except ImportError:
        return None
    try:
        h = cp.Variable(n)
        X = cp.Variable((n, n), symmetric=True)
        t = cp.Variable()
        s = n / 2.0
        constraints = [h >= 0, h <= 1, cp.sum(h) == s, t >= 0, t <= 1]
        M = cp.bmat([[cp.Constant(1), h.T], [h, X]])
        constraints.append(M >> 0)
        for i in range(n):
            for j in range(n):
                constraints.append(X[i, j] >= 0)
                constraints.append(X[i, j] >= h[i] + h[j] - 1)
                constraints.append(X[i, j] <= h[i])
                constraints.append(X[i, j] <= h[j])
            constraints.append(cp.sum(X[i, :]) == s * h[i])

            hi = min(n, n - d)

def _moments(sdp_result, n):
    hv, Xv, tv = sdp_result
    P_ = np.eye(n) - np.ones((n, n)) / n
    S = Xv - np.outer(hv, hv)
    S = P_ @ S @ P_
    S = 0.5 * (S + S.T)
    evals, evecs = np.linalg.eigh(S)
    mx = max(1.0, np.max(np.abs(evals)))
    if evals[0] < -1e-4 * mx:
        return hv, None
    good = evals > 1e-12 * mx
    eig_pos = evals[good]
    vecs = evecs[:, good]
    L = vecs @ np.diag(np.sqrt(np.maximum(eig_pos, 0)))
    L = L - np.mean(L, axis=0)
    return hv, L

        if np.max(np.abs(x - c)) <= 1e-10:
            return
    if len(cands) < cap:
        cands.append(x.copy())
        scores.append(sc)
