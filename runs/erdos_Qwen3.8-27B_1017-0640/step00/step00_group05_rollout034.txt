        max_iter = 15
    elif N <= 999:
        max_iter = 6
    else:
        max_iter = 0
    starts = [base.copy(), base[::-1].copy(), (1.0 - base).copy()]
    if N <= 128:
        starts.append(base + rng.normal(0, 0.05, N))
        starts.append(base + rng.normal(0, 0.15, N))
        starts.append(rng.uniform(0, 1, N))
        starts.append(rng.beta(0.7, 0.7, N))
        idx = np.arange(N, dtype=np.float64)
        starts.append(0.5 + 0.45 * np.cos(4.0 * math.pi * (idx + 0.5) / N + 0.37))
    elif N <= 256:
        starts.append(base + rng.normal(0, 0.08, N))
        starts.append(rng.uniform(0, 1, N))
        idx = np.arange(N, dtype=np.float64)
        starts.append(0.5 + 0.45 * np.cos(4.0 * math.pi * (idx + 0.5) / N + 0.37))
    elif N <= 512:
        starts.append(base + rng.normal(0, 0.08, N))
        idx = np.arange(N, dtype=np.float64)
        starts.append(0.5 + 0.45 * np.cos(4.0 * math.pi * (idx + 0.5) / N + 0.37))
    elif N <= 999:
        starts.append(base + rng.normal(0, 0.05, N))
    starts.append(neutral.copy())
    feasible_starts = []
    seen = []
    for s in starts:
        if s.shape != (N,) or not np.all(np.isfinite(s)):
            continue
        if check_feasible(s, N, m):
            proj = s.copy()
        else:
            proj = project_capped(s, N, m)
        if proj is None:
            continue
        dup = False
        for prev in seen:
            if np.max(np.abs(proj - prev)) <= 1e-12:
                dup = True
                break
        if not dup:
            feasible_starts.append(proj)
            seen.append(proj)
    if not feasible_starts:
        feasible_starts = [neutral.copy()]
        seen = [neutral.copy()]
    incumbent = [feasible_starts[0].copy(), exact_score(feasible_starts[0], N, dx)]
    for s in feasible_starts[1:]:
        sc = exact_score(s, N, dx)
        if sc is not None and sc < incumbent[1] - 1e-12:
            incumbent = [s.copy(), sc]
    if max_iter > 0:
        for s in feasible_starts:
            try:
                h_out, incumbent, incumbent_score = gradient_descent(
                    s, N, dx, m, tau_s, max_iter, incumbent, incumbent[1])
            except Exception:
                pass
    final = project_capped(incumbent[0], N, m)
    if final is not None:
        final_score = exact_score(final, N, dx)
    else:
        final = incumbent[0].copy()
        final_score = incumbent[1]
    for cand in [parent_fallback, neutral]:
        sc = exact_score(cand, N, dx)
        if sc is not None and sc < final_score - 1e-12:
            final = cand.copy()
            final_score = sc
    if final_score is None:
        final = neutral.copy()
        final_score = exact_score(final, N, dx)
    return final, float(final_score), int(N)
