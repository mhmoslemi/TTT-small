import math

def project_capped_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1 or v.size < 2:
        return None
    if not np.all(np.isfinite(v)) or np.any(np.abs(v) > 1e6):
        return None
    n = v.size
    s = n / 2.0
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = lo + (hi - lo) / 2.0
        if math.fsum(np.clip(v - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(3):
        r = s - math.fsum(x)
        if abs(r) <= 5e-15:
            break
        if r > 0:
            for i in range(n):
                adj = min(r, 1.0 - x[i])
                x[i] += adj
                r -= adj
                if abs(r) <= 5e-15:
                    break
        else:
            for i in range(n):
                adj = min(-r, x[i])
                x[i] -= adj
                r += adj
                if abs(r) <= 5e-15:
                    break
    if not np.all(np.isfinite(x)) or np.any(x < -1e-12) or np.any(x > 1.0 + 1e-12):
        return None
    if abs(math.fsum(x) - s) > 1e-9:
        return None
    return x

def exact_score(h):
    h = np.asarray(h, dtype=np.float64)
    if h.ndim != 1 or h.size < 2 or not np.all(np.isfinite(h)):
        return None
    n = h.size
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    if abs(math.fsum(h) - n / 2.0) > 1e-9:
        return None
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = float(np.max(corr) * (2.0 / n))
    return val if math.isfinite(val) else None

def lag_values(h, dx):
    n = h.size
    vals = np.empty(2 * n - 1, dtype=np.float64)
    for idx, d in enumerate(range(-n + 1, n)):
        st = max(0, -d)
        sp = n - abs(d)
        vals[idx] = dx * np.dot(h[st:st + sp], 1.0 - h[st + d:st + d + sp])
    return vals

def smooth_val_grad(h, tau, need_grad=True):
    n = h.size
    dx = 2.0 / n
    a_d = lag_values(h, dx)
    if not np.all(np.isfinite(a_d)):
        return None, None
    A = float(np.max(a_d))
    e_d = np.exp((a_d - A) / tau)
    Z = float(np.sum(e_d))
    F = A + tau * math.log(Z)
    if not math.isfinite(F):
        return None, None
    if not need_grad:
        return F, None
    p_d = e_d / Z
    g = np.zeros(n, dtype=np.float64)
    for idx, d in enumerate(range(-n + 1, n)):
        w = dx * p_d[idx]
        st = max(0, -d)
        sp = n - abs(d)
        src = h[st:st + sp]
        tgt = h[st + d:st + d + sp]
        g[st:st + sp] += w * (1.0 - tgt)
        g[st + d:st + d + sp] -= w * src
    if not np.all(np.isfinite(g)):
        return None, None
    return F, g

def projected_smooth_search(start, temps, iters):
    current = start.copy()
    local_best = start.copy()
    lbs = exact_score(local_best)
    if lbs is None:
        return local_best
    for tau in temps:
        current = local_best.copy()
        F_cur, g = smooth_val_grad(current, tau, True)
        if F_cur is None or g is None:
            break
        eta = 1.0
        for it in range(iters):
            accepted = False
            dmax_last = 0.0
            for bt in range(25):
                y = project_capped_simplex(current - eta * g)
                if y is None:
                    eta *= 0.5
                    continue
                sy = exact_score(y)
                if sy is not None and sy < lbs - 1e-12:
                    local_best = y.copy()
                    lbs = sy
                delta = y - current
                dmax_last = float(np.max(np.abs(delta)))
                d2 = float(np.dot(delta, delta))
                if d2 <= 1e-24:
                    break
                gd = float(np.dot(g, delta))
                Fy, _ = smooth_val_grad(y, tau, False)
                if Fy is None:
                    eta *= 0.5
                    continue
                if Fy <= F_cur + 1e-4 * min(gd, 0.0) + 1e-14 * max(1.0, abs(F_cur)):
                    current = y.copy()
                    F_cur = Fy
                    eta = min(1.5 * eta, 10.0)
                    accepted = True
                    break
                eta *= 0.5
            if dmax_last <= 1e-9:
                break
            if not accepted:
                F_cur, g = smooth_val_grad(current, tau, True)
                if F_cur is None or g is None:
                    break
                if eta < 1e-12:
                    break
            else:
                F_cur, g = smooth_val_grad(current, tau, True)
                if F_cur is None or g is None:
                    break
    return local_best

def run(seed=42, **kwargs):
    try:
        seed = int(seed)
        if not (0 <= seed < (1 << 32)):
            seed = 42
    except (ValueError, TypeError):
        seed = 42
    rng = np.random.default_rng(seed)

    parent_valid = False
    parent_vec = None
    n = 47
    parent = globals().get('initial_h_values', None)
    if parent is not None:
        try:
            p = np.asarray(parent, dtype=np.float64).ravel()
            if 2 <= p.size < 1000 and np.all(np.isfinite(p)) and p.min() >= -1e-7 and p.max() <= 1.0 + 1e-7:
                n = p.size
                pv = project_capped_simplex(p)
                if pv is not None and exact_score(pv) is not None:
                    parent_vec = pv
                    parent_valid = True
        except Exception:
            pass

    constant = 0.5 * np.ones(n, dtype=np.float64)
    if not parent_valid:
        parent_vec = constant.copy()

    starts = [parent_vec.copy()]
    if parent_valid:
        starts.append(constant.copy())

    base = parent_vec
    idx = np.arange(n, dtype=np.float64)
    for freq in [1, 2]:
        phi = rng.uniform(0, 2 * math.pi)
        v = 0.5 + 0.30 * np.sin(2.0 * math.pi * freq * (idx + 0.5) / n + phi)
        pv = project_capped_simplex(v)
        if pv is not None and all(float(np.max(np.abs(pv - s0))) > 1e-9 for s0 in starts):
            starts.append(pv)
    for scale in [0.06, 0.18]:
        v = base + scale * rng.standard_normal(n)
        pv = project_capped_simplex(v)
        if pv is not None and all(float(np.max(np.abs(pv - s0))) > 1e-9 for s0 in starts):
            starts.append(pv)
    for _ in range(2):
        pv = project_capped_simplex(rng.random(n))
        if pv is not None and all(float(np.max(np.abs(pv - s0))) > 1e-9 for s0 in starts):
            starts.append(pv)
    if not parent_valid:
        phi = rng.uniform(0, 2 * math.pi)
        v = 0.5 + 0.30 * np.cos(2.0 * math.pi * 3.0 * (idx + 0.5) / n + phi)
        pv = project_capped_simplex(v)
        if pv is not None and all(float(np.max(np.abs(pv - s0))) > 1e-9 for s0 in starts):
            starts.append(pv)

    if n <= 64:
        temps = [0.04, 0.01, 0.0025, 0.0005, 0.0001]
        iters_pt = 80
        ms = 8
        rscales = [0.04, 0.12]
    elif n <= 128:
        temps = [0.04, 0.01, 0.0025, 0.0005]
        iters_pt = 45
        ms = 5
        rscales = [0.06]
    elif n <= 512:
        temps = [0.03, 0.006, 0.001]
        iters_pt = 25
        ms = 3
        rscales = []
    else:
        temps = [0.02, 0.003]
        iters_pt = 12
        ms = 2
        rscales = []

    gbest = parent_vec.copy()
    gbs = exact_score(gbest)
    if gbs is None:
        gbest = constant.copy()
        gbs = exact_score(gbest)

    for s in starts[:ms]:
        try:
            sc = exact_score(s)
            if sc is not None and sc < gbs - 1e-12:
                gbest = s.copy()
                gbs = sc
            res = projected_smooth_search(s, temps, iters_pt)
            rs = exact_score(res)
            if rs is not None and rs < gbs - 1e-12:
                gbest = res.copy()
                gbs = rs
        except Exception:
            continue

    for rsc in rscales:
        try:
            v = gbest + rsc * rng.standard_normal(n)
            pv = project_capped_simplex(v)
            if pv is not None:
                sc = exact_score(pv)
                if sc is not None and sc < gbs - 1e-12:
                    gbest = pv.copy()
                    gbs = sc
                res = projected_smooth_search(pv, temps, iters_pt)
                rs = exact_score(res)
                if rs is not None and rs < gbs - 1e-12:
                    gbest = res.copy()
                    gbs = rs
        except Exception:
            continue

    final = project_capped_simplex(gbest)
    if final is None:
        final = (parent_vec if parent_valid else constant).copy()
    fsc = exact_score(final)
    if fsc is None:
        final = constant.copy()
        fsc = exact_score(final)

    return (final, fsc, n)
