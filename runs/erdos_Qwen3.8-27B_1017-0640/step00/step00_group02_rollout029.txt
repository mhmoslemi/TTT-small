def smooth_eval(h, tau, dx, with_gradient):
    q = np.correlate(h, 1.0 - h, mode='full')
    c = q * dx
    cmax = c.max()
    u = (c - cmax) / tau
    exp_u = np.exp(u)
    su = exp_u.sum()
    F = cmax + tau * np.log(su)
    if not np.isfinite(F):
        return F, None
    w = exp_u / su
    if not with_gradient:
        return F, None
    n = len(h)
    g = w[n - 1] * (1.0 - 2.0 * h)
    for s in range(1, n):
        m = n - s
        wp = w[n - 1 + s]
        wn = w[n - 1 - s]
        g[s:] += wp * (1.0 - h[:m])
        g[:m] -= wp * h[s:]
        g[:m] += wn * (1.0 - h[s:])
        g[s:] -= wn * h[:m]
    g *= dx
    if not np.all(np.isfinite(g)):
        return F, None
    return F, g

import numpy as np

def project_capped(v, S):
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1 or len(v) == 0 or not np.all(np.isfinite(v)):
        return None
    n = len(v)
    v = np.clip(v, -1e6, 1e6)
    lo, hi = float(v.min()) - 1.0, float(v.max())
    mid = (lo + hi) * 0.5
    for _ in range(80):
        mid = lo + (hi - lo) * 0.5
        if np.clip(v - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
        if abs(hi - lo) < 1e-15:
            break
    x = np.clip(v - mid, 0.0, 1.0)
    for _ in range(3):
        r = S - x.sum()
        if abs(r) < 1e-12:
            break
        m = (x < 1.0) if r > 0 else (x > 0.0)
        nf = int(m.sum())
        if nf == 0:
            return None
        x[m] = np.clip(x[m] + r / nf, 0.0, 1.0)
    if abs(x.sum() - S) > 1e-10:
        return None
    return x

def feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            (h >= 0.0).all() and (h <= 1.0).all() and
            abs(h.sum() - n / 2.0) <= 1e-10)

def escore(h, n):
    if not feasible(h, n):
        return float('inf')
    c = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)
    return c if np.isfinite(c) else float('inf')

def seval(h, tau, dx, grad):
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = c.max()
    u = (c - cm) / tau
    eu = np.exp(u)
    su = eu.sum()
    F = cm + tau * np.log(su)
    if not np.isfinite(F):
        return F, None
    if not grad:
        return F, None
    w = eu / su
    n = len(h)
    g = w[n - 1] * (1.0 - 2.0 * h)
    for s in range(1, n):
        m = n - s
        g[s:] += w[n - 1 + s] * (1.0 - h[:m])
        g[:m] -= w[n - 1 + s] * h[s:]
        g[:m] += w[n - 1 - s] * (1.0 - h[s:])
        g[s:] -= w[n - 1 - s] * h[:m]
    g *= dx
    return (F, g) if np.all(np.isfinite(g)) else (F, None)

def lsearch(x, n, S, dx, mi):
    bx, bs = x.copy(), escore(x, n)
    for tau in [0.02, 0.005, 0.001, 0.0003, 0.0001]:
        alpha = 1.0
        x_f, g = seval(x, tau, dx, True)
        if g is None:
            continue
        for _ in range(mi):
            ok, d = False, None
            for _ in range(20):
                y = project_capped(x - alpha * g, S)
                if y is None:
                    break
                d = y - x
                y_f, _ = seval(y, tau, dx, False)
                if np.isfinite(y_f) and y_f <= x_f + g @ d + d @ d / (2.0 * alpha) + 1e-12 * (1.0 + abs(x_f)):
                    ok = True
                    break
                alpha *= 0.5
                if alpha < 1e-12:
                    break
            if not ok:
                break
            x = y
            x_f, g = seval(x, tau, dx, True)
            if g is None:
                break
            sc = escore(x, n)
            if sc < bs - 1e-12:
                bs, bx = sc, x.copy()
            if np.linalg.norm(d) / alpha < 1e-7:
                break
            alpha = min(1.5 * alpha, 4.0)
    return bx, bs
