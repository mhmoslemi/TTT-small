    try:
        p = np.array(initial_h_values, dtype=np.float64)
        if p.ndim != 1 or p.shape[0] < 2 or p.shape[0] > 1024:
            p = None
        elif not np.all(np.isfinite(p)) or np.max(np.abs(p)) > 1e6:
            p = None
        else:
            p = p.copy()
    except Exception:
        p = None
    if p is not None:
        n = len(p)
        if feasible(p, n):
            fallback = p
        else:
            pr = project_capped_simplex(p, n / 2.0)
            if pr is not None and feasible(pr, n):
                fallback = pr
    if fallback is None:
        n = 42
        fallback = np.zeros(n, dtype=np.float64)
        fallback[: n // 2] = 1.0
        if n % 2:
            fallback[n // 2] = 0.5
    fallback = fallback.copy()
    fb_score = score(fallback)
    if fb_score is None:
        fb_score = 1.0

    if n > 64:
        return fallback, float(fb_score), n

    NP = max(32, min(6 * n, 256))
    n_gen = 80
    try:
        NP = max(32, min(int(kwargs.get('population_size', NP)), 320))
    except Exception:
        pass
    try:
        n_gen = max(20, min(int(kwargs.get('generations', n_gen)), 160))
    except Exception:
        pass

    tm = n / 2.0
    pop = np.zeros((NP, n))
    sc = np.full(NP, 1.0)
    pop[0] = fallback
    sc[0] = fb_score

    anchors = []
    anchors.append(fallback[::-1].copy())
    anchors.append(1.0 - fallback.copy())
    anchors.append(np.full(n, 0.5))
    bv = np.zeros(n)
    bv[: n // 2] = 1.0
    if n % 2:
        bv[n // 2] = 0.5
    anchors.append(bv)
    ii = np.arange(n, dtype=np.float64)
    anchors.append(0.5 + 0.25 * np.cos(2.0 * np.pi * ii / n))

    idx = 1
    for a in anchors:
        if idx >= NP:
            break
        pr = project_capped_simplex(a.copy(), tm)
        if pr is not None and feasible(pr, n):
            pop[idx] = pr
            s = score(pr)
            sc[idx] = s if s is not None else 1.0
        else:
            pop[idx] = fallback
            sc[idx] = fb_score
        idx += 1

    ns = [0.03, 0.08, 0.15, 0.30, 0.45]
    ni = 0
    while idx < NP:
        mode = (idx - 1) % 4
        s2 = ns[ni % 5]
        ni += 1
        if mode == 0:
            x = fallback + rng.normal(0, s2, n)
        elif mode == 1:
            ai = int(rng.integers(0, len(anchors)))
            x = anchors[ai] + rng.normal(0, s2, n)
        elif mode == 2:
            x = rng.uniform(0.0, 1.0, n)
        else:
            x = rng.uniform(-0.25, 1.25, n)
        pr = project_capped_simplex(x.copy(), tm)
        if pr is not None and feasible(pr, n):
            pop[idx] = pr
            sv = score(pr)
            sc[idx] = sv if sv is not None else 1.0
        else:
            pop[idx] = fallback
            sc[idx] = fb_score
        idx += 1

    best_i = int(np.argmin(sc))
    incumbent = pop[best_i].copy()
    inc_score = sc[best_i]

    no_imp = 0
    inj = 0
    inj_scales = [0.04, 0.10, 0.25, 0.40]

    for gen in range(n_gen):
        snap = pop.copy()
        snap_sc = sc.copy()
        prev_inc = inc_score

        for i in range(NP):
            others = [j for j in range(NP) if j != i]
            picks = rng.choice(others, size=3, replace=False)
            r1, r2, r3 = int(picks[0]), int(picks[1]), int(picks[2])
            F = rng.uniform(0.45, 0.90)
            CR = rng.uniform(0.65, 1.0)
            v = snap[r1] + F * (snap[r2] - snap[r3])
            mask = rng.random(n) < CR
            mask[int(rng.integers(0, n))] = True
            u = np.where(mask, v, snap[i])
            pr = project_capped_simplex(u.copy(), tm)
            if pr is None or not feasible(pr, n):
                continue
            sv = score(pr)
            if sv is None:
                continue
            if sv < snap_sc[i] - 1e-12:
                pop[i] = pr
                sc[i] = sv
            if sv < inc_score - 1e-12:
                incumbent = pr.copy()
                inc_score = sv

        if inc_score < prev_inc - 1e-12:
            no_imp = 0
        else:
            no_imp += 1

        if no_imp >= 20 and inj < 3:
            order = np.argsort(sc)
            nr = int(np.ceil(0.20 * NP))
            for k in range(nr):
                ii2 = int(order[-(k + 1)])
                if k % 2 == 0:
                    x = incumbent + rng.normal(0, inj_scales[(inj * 4 + k) % 4], n)
                else:
                    x = rng.uniform(-0.25, 1.25, n)
                pr = project_capped_simplex(x.copy(), tm)
                if pr is not None and feasible(pr, n):
                    sv = score(pr)
                    if sv is not None:
                        pop[ii2] = pr
                        sc[ii2] = sv
                        if sv < inc_score - 1e-12:
                            incumbent = pr.copy()
                            inc_score = sv
            inj += 1
            no_imp = 0
        if no_imp >= 20 and inj >= 3:
            break

    final = incumbent.copy()
    fs = score(final)
    if fs is None:
        pr = project_capped_simplex(final, tm)
        if pr is not None and feasible(pr, n):
            final = pr
            fs = score(final)
    if fs is None:
        final = fallback
        fs = fb_score
    elif fs > fb_score + 1e-12:
        final = fallback
        fs = fb_score

    return final.astype(np.float64), float(fs), n
