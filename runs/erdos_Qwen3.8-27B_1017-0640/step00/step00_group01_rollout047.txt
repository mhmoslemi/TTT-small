    scales = [0.12, 0.07, 0.035, 0.016, 0.006][:5]
    nprop = 100 if n <= 64 else 50
    if n > 128:
        temps = temps[:2]; scales = scales[:2]; nprop = 24 if n <= 512 else 10
    best_h = parent.copy()
    best_sc = exact_score(best_h, n)
    for cs in chain_starts:
        try:
            ah, asc = do_anneal(cs, n, L, rng, temps, scales, nprop, 0.65)
            if np.isfinite(asc) and asc < best_sc - 1e-12:
                best_h = ah.copy()
                best_sc = asc
        except Exception:
            pass
    for i, c in enumerate(cands):
        if np.isfinite(scs[i]) and scs[i] < best_sc - 1e-12:
            best_h = c.copy()
            best_sc = scs[i]
    fh = project_capped_simplex(best_h)
    if fh is not None and check_feasible(fh, n):
        best_h = fh
    else:
        best_h = parent.copy()
    best_sc = exact_score(best_h, n)
    if not np.isfinite(best_sc):
        best_h = parent.copy()
        best_sc = exact_score(best_h, n)
    return (np.asarray(best_h, dtype=np.float64).copy(), float(best_sc), int(n))
