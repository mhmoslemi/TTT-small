        try:
            pf = np.array(parent, dtype=np.float64).ravel()
            if pf.ndim == 1 and pf.size >= 2:
                n = pf.size
                if np.all(np.isfinite(pf)):
                    if np.all(pf >= 0) and np.all(pf <= 1) and abs(np.sum(pf) - n / 2.0) <= 1e-10:
                        parent_vec = pf.copy()
                    else:
                        parent_vec = project_capped_simplex(pf, n)
        except Exception:
            pass
    if parent_vec is None:
        parent_vec = np.full(n, 0.5)

    fb_score = exact_score(parent_vec, n)
    if not np.isfinite(fb_score):
        parent_vec = np.full(n, 0.5)
        fb_score = exact_score(parent_vec, n)

    incumbent = parent_vec.copy()
    inc_score = fb_score

    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)

    starts = []
    if np.all(np.isfinite(parent_vec)):
        starts.append(parent_vec.copy())
        starts.append(parent_vec[::-1].copy())
        starts.append(project_capped_simplex(parent_vec + 0.05 * rng.standard_normal(n), n))
        starts.append(project_capped_simplex(parent_vec + 0.15 * rng.standard_normal(n), n))
        starts.append(np.full(n, 0.5))
    else:
        starts.append(np.full(n, 0.5))
        idx = np.arange(n, dtype=np.float64)
        starts.append(project_capped_simplex(0.5 + 0.3 * np.sin(2.0 * np.pi * (idx + 0.5) / n), n))
        starts.append(project_capped_simplex(0.5 + 0.15 * rng.standard_normal(n), n))
        starts.append(project_capped_simplex(0.5 + 0.35 * rng.standard_normal(n), n))

    unique_starts = []
    for s in starts:
        dup = False
        for us in unique_starts:
            if np.max(np.abs(s - us)) < 1e-10:
                dup = True
                break
        if not dup:
            unique_starts.append(s)

    max_starts = 5 if n <= 128 else 4
    unique_starts = unique_starts[:max_starts]

    for s in unique_starts:
        sc = exact_score(s, n)
        if np.isfinite(sc) and sc < inc_score - 1e-12:
            incumbent = s.copy()
            inc_score = sc

    if n <= 256:
        for s in unique_starts:
            incumbent, inc_score = gradient_search(s, incumbent, inc_score, n)
        if n <= 128:
            shake = project_capped_simplex(incumbent + 0.04 * rng.standard_normal(n), n)
            if np.max(np.abs(shake - incumbent)) >= 1e-10:
                incumbent, inc_score = gradient_search(shake, incumbent, inc_score, n)

    h_final = project_capped_simplex(incumbent, n)
    sc_final = exact_score(h_final, n)
    if not np.isfinite(sc_final):
        h_final = parent_vec.copy()
        sc_final = fb_score
    sc_fb = exact_score(parent_vec, n)
    if np.isfinite(sc_fb) and sc_fb < sc_final - 1e-12:
        h_final = parent_vec.copy()
        sc_final = sc_fb
    uni = np.full(n, 0.5)
    sc_uni = exact_score(uni, n)
    if np.isfinite(sc_uni) and sc_uni < sc_final - 1e-12:
        h_final = uni
        sc_final = sc_uni

    h_final = np.clip(h_final, 0.0, 1.0)
    h_final = project_capped_simplex(h_final, n)
    sc_final = exact_score(h_final, n)
    if not np.isfinite(sc_final):
        h_final = np.full(n, 0.5)
        sc_final = float(np.max(np.correlate(h_final, 1.0 - h_final, mode='full') * (2.0 / n)))

    return h_final, sc_final, n
