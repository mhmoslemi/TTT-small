import numpy as np
from scipy.optimize import minimize


def is_feasible(h, n):
    if h.ndim != 1 or len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1, n)


def project_capped(y, n):
    if y.ndim != 1 or len(y) != n or not np.all(np.isfinite(y)) or np.any(np.abs(y) > 1e12):
        return np.full(n, 0.5)
    m = n / 2.0
    lo, hi = np.min(y) - 1.0, np.max(y)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(y - mid, 0.0, 1.0)) > m:
            lo = mid
        else:
            hi = mid
    p = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    res = m - np.sum(p)
    for _ in range(3):
        if abs(res) <= 2e-14 * max(1, n):
            break
        if res > 0:
            for i in range(n):
                if res <= 0:
                    break
                a = min(res, 1.0 - p[i])
                p[i] += a
                res -= a
        else:
            for i in range(n):
                if res >= 0:
                    break
                a = min(-res, p[i])
                p[i] -= a
                res += a
    return p if is_feasible(p, n) else np.full(n, 0.5)


def exact_score(h, n):
    if h.ndim != 1 or len(h) != n or not np.all(np.isfinite(h)):
        return float('inf')
    try:
        return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)
    except Exception:
        return float('inf')


def majorizer_vals_jac(h, r, n):
    dx = 2.0 / n
    nl = 2 * n - 1
    Q = np.zeros(nl)
    G = np.zeros((nl, n))
    d = h - r
    Q[0] = dx * (np.sum(r * (1.0 - r)) + np.dot(1.0 - 2.0 * r, d))
    G[0] = dx * (1.0 - 2.0 * r)
    row = 1
    for s in range(1, n):
        p = n - s
        D = d[:p] + d[s:]
        D2 = 0.5 * np.dot(D, D)
        Q[row] = dx * (np.dot(h[:p], 1.0 - h[s:]) + D2)
        GA = np.zeros(n)
        GA[:p] += dx * (1.0 - h[s:] + D)
        GA[s:] += dx * (-h[:p] + D)
        G[row] = GA
        row += 1
        Q[row] = dx * (np.dot(h[s:], 1.0 - h[:p]) + D2)
        GB = np.zeros(n)
        GB[:p] += dx * (-h[s:] + D)
        GB[s:] += dx * (1.0 - h[:p] + D)
        G[row] = GB
        row += 1
    if not (np.all(np.isfinite(Q)) and np.all(np.isfinite(G))):
        return None, None
    return Q, G


def _obj(z, *a):
    return z[-1]


def _obj_g(z, *a):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g


def _eq(z, *a):
    return np.sum(z[:a[0]]) - a[0] / 2.0


def _eq_j(z, *a):
    j = np.zeros(len(z))
    j[:a[0]] = 1.0
    return j


def _ineq(z, *a):
    n, r = a[0], a[1]
    Q, _ = majorizer_vals_jac(z[:n], r, n)
    if Q is None:
        return np.ones(2 * n - 1) * 1e6
    return z[-1] - Q


def _ineq_j(z, *a):
    n, r = a[0], a[1]
    _, G = majorizer_vals_jac(z[:n], r, n)
    if G is None:
        return np.ones((2 * n - 1, n + 1))
    J = np.zeros((2 * n - 1, n + 1))
    J[:, :n] = -G
    J[:, -1] = 1.0
    return J
