    for f in [1, 2, 3]:
        if f < N:
            cands.append(0.5 + 0.45 * np.cos(2.0 * np.pi * f * (ii + 0.5) / N))
            cands.append(0.5 + 0.45 * np.cos(2.0 * np.pi * f * (ii + 0.5) / N + np.pi / 3.0))
    for _ in range(3):
        cands.append(base[rng.permutation(N)])
    for c in cands:
        proj = c if _is_feasible(c, N, m) else _project_simplex(c, N, m)
        if proj is None:
            continue
        dup = False
        for j in range(len(pop)):
            if _is_dup(pop[j], proj):
                dup = True
                break
        if dup:
            continue
        pop.append(proj)
        s = _exact_score(proj, N)
        pop_scores.append(999.0 if s is None else s)
        if s is not None and s < inc_score:
            inc_score = s
            inc_vec = proj.copy()
    _fill_pop(pop, pop_scores, base, neutral, N, m, P, rng)
    for j in range(P):
        if pop_scores[j] < inc_score:
            inc_score = pop_scores[j]
            inc_vec = pop[j].copy()

    stall = 0
    stall_lim = max(10, G // 4)
    restarts = 0

    for g in range(G):
        old_pop = [x.copy() for x in pop]
        old_scores = list(pop_scores)
        sc_start = inc_score
        new_pop = []
        new_scores = []
        prog = g / max(G - 1, 1)
        F = 0.70 - 0.30 * prog
        CR = 0.90 - 0.30 * prog
        others_all = list(range(P))

        for i in range(P):
            others = [j for j in others_all if j != i]
            r1, r2, r3 = rng.choice(others, size=3, replace=False)
            r1, r2, r3 = int(r1), int(r2), int(r3)
            mr = old_pop[r1] + F * (old_pop[r2] - old_pop[r3])
            mn = _project_simplex(mr, N, m)
            if mn is None:
                new_pop.append(old_pop[i])
                new_scores.append(old_scores[i])
                continue
            msk = rng.random(N) < CR
            msk[int(rng.integers(0, N))] = True
            tr = np.where(msk, mn, old_pop[i])
            tr = _project_simplex(tr, N, m)
            if tr is None:
                new_pop.append(old_pop[i])
                new_scores.append(old_scores[i])
                continue
            ts = _exact_score(tr, N)
            if ts is None:
                new_pop.append(old_pop[i])
                new_scores.append(old_scores[i])
                continue
            if ts < old_scores[i]:
                new_pop.append(tr)
                new_scores.append(ts)
            else:
                new_pop.append(old_pop[i])
                new_scores.append(old_scores[i])
            if ts < inc_score:
                inc_score = ts
                inc_vec = tr.copy()

        pop = new_pop
        pop_scores = new_scores

        if inc_score < sc_start:
            stall = 0
        else:
            stall += 1

        if stall >= stall_lim and restarts < 2 and g < G - 1:
            restarts += 1
            pop = [inc_vec.copy()]
            pop_scores = [inc_score]
            order = np.argsort(pop_scores)
            ecol = max(2, P // 8)
            added = 0
            for oi in order:
                if added >= ecol:
                    break
                if not _is_dup(pop[0], old_pop[oi]):
                    pop.append(old_pop[oi].copy())
                    pop_scores.append(old_scores[oi])
                    added += 1
            _fill_pop(pop, pop_scores, inc_vec, neutral, N, m, P, rng)
            for j in range(P):
                if pop_scores[j] < inc_score:
                    inc_score = pop_scores[j]
                    inc_vec = pop[j].copy()
            stall = 0

    final_vec = _project_simplex(inc_vec, N, m)
    if final_vec is None:
        final_vec = neutral.copy()
    final_score = _exact_score(final_vec, N)
    if final_score is None:
        final_vec = neutral.copy()
        final_score = _exact_score(neutral, N)
    if parent is not None:
        ps = _exact_score(parent, N)
        if ps is not None and ps < final_score:
            final_vec = parent.copy()
            final_score = ps
    if final_score is None:
        final_score = 1.0
    return (final_vec, final_score, N)
