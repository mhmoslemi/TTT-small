            p[ix] -= min(-resid / len(ix), np.min(p[ix]))
        resid = m - np.sum(p)
    p = np.clip(p, 0.0, 1.0)
    if not np.all(np.isfinite(p)) or abs(np.sum(p) - m) > 1e-10 * max(1, n):
        return np.full(n, 0.5)
    return p

def check_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1 + 1e-12)
            and abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1, n))

def exact_score(h):
    n = len(h)
    if not check_feasible(h, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lag_values(h):
    n = len(h)
    dx = 2.0 / n
    q = np.empty(2 * n - 1)
    q[0] = dx * np.sum(h * (1.0 - h))
    for s in range(1, n):
        q[s] = dx * np.sum(h[:n - s] * (1.0 - h[s:]))
        q[n + s - 1] = dx * np.sum(h[s:] * (1.0 - h[:n - s]))
    return q

def lag_jacobian(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    G[0] = dx * (1.0 - 2.0 * h)
    for s in range(1, n):
        G[s, :n - s] += dx * (1.0 - h[s:])
        G[s, s:] -= dx * h[:n - s]
        G[n + s - 1, s:] += dx * (1.0 - h[:n - s])
        G[n + s - 1, :n - s] -= dx * h[s:]
    return G

def smooth_val(q, tau):
    qm = np.max(q)
    return qm + tau * np.log(np.sum(np.exp((q - qm) / tau)))

def smooth_val_grad(q, G, tau):
    qm = np.max(q)
    e = np.exp((q - qm) / tau)
    s = np.sum(e)
    w = e / s
    return qm + tau * np.log(s), G.T @ w

def run(seed=42, **kwargs):
    n = 80
    base = None
    try:
        pv = initial_h_values
        if pv is not None:
            pv = np.asarray(pv, dtype=np.float64)
            if pv.ndim == 1 and 2 <= len(pv) <= 999:
                n = len(pv)
                base = pv.copy()
    except Exception:
        pass
    if base is None:
        try:
            nn = int(kwargs.get('n_points', 80))
            if 2 <= nn <= 999:
                n = nn
        except Exception:
            pass
        base = np.full(n, 0.5)
    n_starts, max_iter = 7, 80
    try:
        v = int(kwargs.get('n_starts', 7))
        if 1 <= v <= 8:
            n_starts = v
    except Exception:
        pass
    try:
        v = int(kwargs.get('max_iter', 80))
        if 20 <= v <= 200:
            max_iter = v
    except Exception:
        pass
    try:
        seed = max(0, int(seed))
    except Exception:
        seed = 42
    rng = np.random.default_rng(seed)
    base_proj = project_capped(base, n)
    best = base_proj.copy()
    best_score = exact_score(best)

    if n > 256:
        h = project_capped(best, n)
        sc = exact_score(h)
        return (h, float(sc), n)

    starts = [base_proj.copy()]
    for std in [0.02, 0.08, 0.20]:
        starts.append(project_capped(base + std * rng.standard_normal(n), n))
    starts.append(project_capped(rng.uniform(0, 1, n), n))
    K = min(6, (n - 1) // 2)
    y = np.full(n, 0.5)
    t = (np.arange(n) + 0.5)
    for k in range(1, K + 1):
        sd = 0.12 / np.sqrt(k)
        y += sd * rng.standard_normal() * np.cos(2.0 * np.pi * k * t / n)
        y += sd * rng.standard_normal() * np.sin(2.0 * np.pi * k * t / n)
    starts.append(project_capped(y, n))
    nb = np.zeros(n)
    ix = rng.permutation(n)[:n // 2]
    nb[ix] = 1.0
    if n % 2 == 1:
        r = [j for j in range(n) if j not in set(ix.tolist())]
        nb[r[0]] = 0.5
    starts.append(project_capped(nb, n))
    starts.append(project_capped(base + 0.4 * rng.standard_normal(n), n))

    deduped = [starts[0]]
    for s in starts[1:]:
        if all(np.max(np.abs(s - d)) >= 1e-10 for d in deduped):
            deduped.append(s)
    starts = deduped[:n_starts]

    taus = [0.02, 0.005, 0.00125, 0.0003125, 7.8125e-05, 1.953125e-05, 4.8828125e-06]

    for start in starts:
        x = start.copy()
        for tau in taus:
            alpha = 1.0
            for _ in range(max_iter):
                q = lag_values(x)
                if not np.all(np.isfinite(q)):
                    break
                G = lag_jacobian(x)
                phi, g = smooth_val_grad(q, G, tau)
                if not np.all(np.isfinite(g)) or not np.isfinite(phi):
                    break
                accepted = False
                for _bt in range(25):
                    xt = project_capped(x - alpha * g, n)
                    d = xt - x
                    if np.max(np.abs(d)) <= 1e-9:
                        break
                    gd = np.dot(g, d)
                    if gd >= 0:
                        break
                    qt = lag_values(xt)
                    if not np.all(np.isfinite(qt)):
                        break
                    pt = smooth_val(qt, tau)
                    thr = phi + 1e-4 * gd + 1e-14 * max(1.0, abs(phi))
                    if pt <= thr:
                        x = xt
                        accepted = True
                        break
                    alpha *= 0.5
                    if alpha < 1e-12:
                        break
                if accepted:
                    alpha = min(alpha * 1.25, 4.0)
                    sc = exact_score(x)
                    if sc < best_score - 1e-12:
                        best = x.copy()
                        best_score = sc
                else:
                    break
            if best_score <= 0.3808:
                break
        if best_score <= 0.3808:
            break

    h = project_capped(best, n)
    sc = exact_score(h)
    if sc == float('inf'):
        h = project_capped(base_proj, n)
        sc = exact_score(h)
    if sc == float('inf'):
        h = np.full(n, 0.5)
        sc = exact_score(h)
    return (h, float(sc), n)
