            pj = project_capped(s, n, m)
            if pj is None:
                continue
            s = pj
        sc = exact_score(s)
        ev += 1
        scored.append((s, sc))
        if ev >= B:
            break
    if not scored:
        return (hfall, float(exact_score(hfall)), n)
    gbest_h = scored[0][0].copy()
    gbest_s = scored[0][1]
    for s, sc in scored:
        if sc < gbest_s - 1e-12:
            gbest_s = sc
            gbest_h = s.copy()
    rem = B - ev
    ns = len(scored)
    per = rem // ns if ns > 0 else 0
    extra = rem % ns if ns > 0 else 0
    for i in range(ns):
        qi = per + (1 if i < extra else 0)
        if qi <= 0:
            continue
        try:
            _, _, rb_h, rb_s = one_start_search(scored[i][0], scored[i][1], rng, n, m, qi)
            if rb_s < gbest_s - 1e-12:
                gbest_s = rb_s
                gbest_h = rb_h.copy()
        except Exception:
            pass
    if not feasible(gbest_h, n, m):
        pj = project_capped(gbest_h, n, m)
        if pj is not None:
            gbest_h = pj
        else:
            gbest_h = hfall.copy()
    gbest_s = exact_score(gbest_h)
    return (gbest_h, float(gbest_s), n)
