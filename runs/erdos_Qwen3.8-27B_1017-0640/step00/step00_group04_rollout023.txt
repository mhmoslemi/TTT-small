    cur = float(mvals[np.argmin(np.abs(grid))])
    if best_f >= cur - 1e-12: return None
    hc = project_capped_simplex(h + best_t * d, n)
    if hc is None or not is_feasible(hc, n):
        hc = h.copy()
    sc = exact_score(hc, n)
    if sc < cur - 1e-12:
        return (hc, np.correlate(hc, 1.0-hc, mode='full'), sc)
    return None

def local_search(h, n, rng, max_sweeps):
    corr = np.correlate(h, 1.0-h, mode='full')
    sc = exact_score(h, n)
    for _ in range(max_sweeps):
        improved = False
        if n <= 64:
            pairs = [(i,j) for i in range(n) for j in range(i+1, n)]
            rng.shuffle(pairs)
        else:
            pairs = list(rng.choice(n*(n-1)//2, size=min(n*(n-1)//2, 4*n), replace=False))
        for pi in range(len(pairs)):
            if n <= 64:
                i, j = pairs[pi]
            else:
                i = int(pairs[pi]) // n
                j = i+1 + int(pairs[pi]) % (n-i-1)
                if j >= n: j = n - 1 - (i // 2)
            d = np.zeros(n); d[i] = 1.0; d[j] = -1.0
            res = search_dir(h, d, corr, n)
            if res is not None:
                h, corr, sc = res; improved = True
        nrand = max(4, n // 4)
        for _ in range(nrand):
            z = rng.standard_normal(n)
            z -= z.mean()
            z[-1] = -z[:-1].sum()
            mx = np.max(np.abs(z))
            if mx < 1e-14: continue
            d = z / mx
            res = search_dir(h, d, corr, n)
            if res is not None:
                h, corr, sc = res; improved = True
        if not improved: break
    return h, sc

def run(seed=42, **kwargs):
    try: s = int(seed)
    except: s = 42
    rng = np.random.default_rng(s)
    n_starts = kwargs.get('n_starts', 6)
    max_sweeps = kwargs.get('max_sweeps', 3)
    try: n_starts = max(4, min(8, int(n_starts)))
    except: n_starts = 6
    try: max_sweeps = max(1, min(5, int(max_sweeps)))
    except: max_sweeps = 3
    # Get fallback
    fallback = None
    try:
        ph = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 2 <= len(ph) <= 4096 and np.all(np.isfinite(ph)) and np.max(np.abs(ph)) < 1e6:
            n = len(ph)
            if np.min(ph) >= -1e-10 and np.max(ph) <= 1.0+1e-10 and abs(ph.sum()-n/2.0) <= 1e-10*max(1,n):
                if abs(ph.sum()-n/2.0) > 5e-13*max(1,n):
                    fix = project_capped_simplex(ph, n)
                    if fix is not None: ph = fix
                fallback = ph.copy()
            elif np.min(ph) > -1.0 and np.max(ph) < 2.0:
                fix = project_capped_simplex(ph, n)
                if fix is not None and is_feasible(fix, n):
                    fallback = fix
    except:
        pass
    if fallback is None:
        n = 42
        fallback = np.zeros(n)
        fallback[:n//2] = 1.0
        if n % 2: fallback[n//2] = 0.5
    n = len(fallback)
    fb_score = exact_score(fallback, n)
    # Generate starts
    starts = [fallback.copy()]
    props = [fallback[::-1].copy(), (1.0-fallback).copy(), 0.5*np.ones(n),
             fallback + 0.15*rng.standard_normal(n),
             fallback + 0.05*rng.standard_normal(n),
             fallback + 0.35*rng.standard_normal(n),
             0.5 + 0.25*np.cos(2.0*math.pi*np.arange(n)/float(n)),
             rng.uniform(-0.25, 1.25, n),
             fallback + 0.20*rng.standard_normal(n),
             fallback + 0.50*rng.standard_normal(n)]
    for p in props:
        if len(starts) >= n_starts: break
        c = project_capped_simplex(p, n)
        if c is None or not is_feasible(c, n): continue
        dup = False
        for s0 in starts:
            if np.max(np.abs(c - s0)) < 1e-9: dup = True; break
        if not dup: starts.append(c)
    # Local search from each start
    best_h, best_sc = fallback.copy(), fb_score
    for st in starts:
        try:
            hs, sc = local_search(st.copy(), n, rng, max_sweeps)
            if sc < best_sc - 1e-12:
                best_h, best_sc = hs.copy(), sc
        except:
            pass
    # Final validation
    if not is_feasible(best_h, n):
        fix = project_capped_simplex(best_h, n)
        if fix is not None: best_h = fix
    if not is_feasible(best_h, n):
        best_h = fallback.copy()
    best_sc = exact_score(best_h, n)
    if not math.isfinite(best_sc):
        best_h = fallback.copy()
        best_sc = exact_score(best_h, n)
    best_h = np.asarray(best_h, dtype=np.float64)
    return (best_h, float(best_sc), int(n))
