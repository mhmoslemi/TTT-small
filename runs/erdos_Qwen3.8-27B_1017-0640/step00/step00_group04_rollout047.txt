            fallback = parent.copy()
        else:
            proj = capped_project(parent, n)
            if proj is not None and is_feasible(proj, n):
                fallback = proj
            else:
                h_blk = np.zeros(n, dtype=np.float64)
                h_blk[:n // 2] = 1.0
                if n % 2 == 1:
                    h_blk[n // 2] = 0.5
                fallback = h_blk
                parent = h_blk.copy()
    else:
        h_blk = np.zeros(n, dtype=np.float64)
        h_blk[:n // 2] = 1.0
        if n % 2 == 1:
            h_blk[n // 2] = 0.5
        fallback = h_blk
        parent = h_blk.copy()

    fb_score = exact_score(fallback)
    if fb_score is None:
        fb_score = 1.0
    global_best = fallback.copy()
    global_score = fb_score

    if n > 64:
        fs = exact_score(global_best)
        if fs is None:
            fs = fb_score
        return global_best.astype(np.float64), float(fs), int(n)

    g = min(2.0, max(0.5, 42.0 / float(n)))
    T_hi = 0.004 * g
    T_lo = 2e-5 * g
    n_blocks = 24
    n_prop = max(50, int(round(160.0 * min(1.0, (42.0 / n) ** 2))))
    D_vals = [64, 256, 1024]
    prev_best_k = None

    for di, D in enumerate(D_vals):
        starts = []
        k1 = encode_lattice(parent, D)
        if k1 is not None:
            starts.append(k1)
        k2 = encode_lattice(global_best, D)
        if k2 is not None:
            starts.append(k2)
        if prev_best_k is not None and di > 0:
            ratio = D // D_vals[di - 1]
            k3 = (prev_best_k * ratio).astype(np.int64)
            if int(np.sum(k3)) == D * n // 2 and np.all(k3 >= 0) and np.all(k3 <= D):
                starts.append(k3)
            else:
                kr = encode_lattice(parent[::-1].copy(), D)
                if kr is not None:
                    starts.append(kr)
        else:
            kr = encode_lattice(parent[::-1].copy(), D)
            if kr is not None:
                starts.append(kr)
        comp = 1.0 - parent.copy()
        k4 = encode_lattice(comp, D)
        if k4 is not None:
            starts.append(k4)
        y_r = rng.uniform(-0.25, 1.25, size=n)
        pr = capped_project(y_r, n)
        if pr is not None:
            k5 = encode_lattice(pr, D)
            if k5 is not None:
                starts.append(k5)
        if not starts:
            kf = encode_lattice(fallback, D)
            if kf is not None:
                starts.append(kf)
            else:
                continue

        best_k_D = None
        best_s_D = None
        for sk in starts:
            sh = decode_lattice(sk, D)
            if sh is None:
                continue
            ss = exact_score(sh)
            if ss is None:
                continue
            if best_k_D is None or ss < best_s_D - 1e-12:
                best_k_D = sk.copy()
                best_s_D = ss
            if ss < global_score - 1e-12:
                global_score = ss
                global_best = sh.copy()

        for sk in starts:
            try:
                bk, bs = anneal_chain(sk, D, n, rng, T_hi, T_lo, n_blocks, n_prop)
                if bk is not None and bs is not None:
                    if best_k_D is None or bs < best_s_D - 1e-12:
                        best_k_D = bk.copy()
                        best_s_D = bs
                    if bs < global_score - 1e-12:
                        bh = decode_lattice(bk, D)
                        if bh is not None:
                            global_score = bs
                            global_best = bh.copy()
            except Exception:
                continue
        prev_best_k = best_k_D

    final = global_best.copy()
    fs = exact_score(final)
    if fs is None or not is_feasible(final, n):
        proj = capped_project(final, n)
        if proj is not None:
            final = proj
            fs = exact_score(final)
    if fs is None or not is_feasible(final, n):
        final = fallback.copy()
        fs = fb_score
    if fs > fb_score:
        final = fallback.copy()
        fs = fb_score
    final = np.clip(final.copy(), 0.0, 1.0)
    r = n / 2.0 - np.sum(final)
    if abs(r) > 1e-12:
        for i in range(n):
            if abs(r) <= 1e-12:
                break
            if r > 0 and final[i] < 1.0:
                a = min(r, 1.0 - final[i])
                final[i] += a
                r -= a
            elif r < 0 and final[i] > 0:
                s = min(-r, final[i])
                final[i] -= s
                r += s
    final = np.clip(final, 0.0, 1.0)
    fs = exact_score(final)
    if fs is None:
        final = fallback.copy()
        fs = exact_score(final)
    if fs is None:
        fs = 0.38085
    return final.astype(np.float64), float(fs), int(n)
