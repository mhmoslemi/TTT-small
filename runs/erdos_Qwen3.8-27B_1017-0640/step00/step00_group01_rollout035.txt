    return np.clip(x, 0.0, 1.0)

def is_feasible(h, n):
    return (h.ndim == 1 and h.size == n and np.all(np.isfinite(h))
            and np.all(h >= -1e-15) and np.all(h <= 1.0 + 1e-15)
            and abs(math.fsum(h) - n * 0.5) <= 1e-10)

def exact_score(h, n):
    if not is_feasible(h, n):
        return None
    try:
        v = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)
        if math.isfinite(v):
            return v
    except Exception:
        pass
    v = 0.0
    for d in range(-(n - 1), n):
        t = 0.0
        i0, i1 = max(0, -d), min(n, n - d)
        for i in range(i0, i1):
            t += h[i] * (1.0 - h[i + d])
        v = max(v, t)
    v *= 2.0 / n
    return v if math.isfinite(v) else None

def make_constant(n):
    return np.full(n, 0.5, dtype=np.float64)

def add_candidate(pop, seen, h):
    for s in seen:
        if np.max(np.abs(h - s)) <= 1e-10:
            return
    seen.append(h)
    pop.append(h)

def build_population(n, rng, parent):
    if n <= 64: P = 48
    elif n <= 128: P = 32
    elif n <= 256: P = 20
    elif n <= 512: P = 12
    else: P = 8
    pop, seen = [], []
    base = parent if parent is not None else make_constant(n)
    add_candidate(pop, seen, base.copy())
    if parent is not None:
        for c in (parent[::-1], 1.0 - parent, (1.0 - parent)[::-1],
                  np.roll(parent, 1), np.roll(parent, max(1, n // 2))):
            p = project_capped_simplex(c)
            if p is not None and is_feasible(p, n):
                add_candidate(pop, seen, p)
    add_candidate(pop, seen, make_constant(n))
    for f in range(1, 5):
        ph = rng.uniform(0.0, 2.0 * math.pi)
        t = (np.arange(n, dtype=np.float64) + 0.5) / n
        p = project_capped_simplex(0.5 + 0.35 * np.sin(2.0 * math.pi * f * t + ph))
        if p is not None and is_feasible(p, n):
            add_candidate(pop, seen, p)
    sig = [0.04, 0.10, 0.22, 0.45, 0.70]
    att = 0
    while len(pop) < P and att < 30 * P:
        att += 1
        if att % 3 == 0:
            v = base + sig[att % 5] * rng.standard_normal(n)
        elif att % 3 == 1:
            v = rng.uniform(0.0, 1.0, n)
        else:
            v = rng.beta(0.7, 0.7, n)
        p = project_capped_simplex(v)
        if p is not None and is_feasible(p, n):
            add_candidate(pop, seen, p)
    while len(pop) < P:
        pop.append(pop[0].copy())
    return np.array(pop, dtype=np.float64)

def jade_search(pop, n, rng, max_gen, stall_lim, max_rest):
    P = pop.shape[0]
    sc = np.empty(P)
    for i in range(P):
        v = exact_score(pop[i], n)
        sc[i] = v if v is not None and math.isfinite(v) else 1e10
    bi = int(np.argmin(sc))
    best_h, best_s = pop[bi].copy(), sc[bi]
    archive, muF, muCR, stall, rests = [], 0.5, 0.8, 0, 0
    for gen in range(max_gen):
        prev_best = best_s
        snap = pop.copy(); snap_sc = sc.copy()
        order = np.argsort(snap_sc)
        k = max(2, math.ceil(0.20 * P))
        pb = order[:k]
        new_pop = pop.copy(); new_sc = sc.copy()
        pend_arch, sF2, sF, lCR, imp = [], 0.0, 0.0, [], False
        for i in rng.permutation(P):
            cp = [j for j in pb if j != i] or [j for j in range(P) if j != i]
            p = cp[int(rng.integers(len(cp)))]
            ex = {i, p}
            cr1 = [j for j in range(P) if j not in ex]
            r1 = cr1[int(rng.integers(len(cr1)))]
            ex.add(r1)
            cr2 = [j for j in range(P) if j not in ex]
            r2 = cr2[int(rng.integers(len(cr2)))] if cr2 else r1
            F = None
            for _ in range(50):
                F = muF + 0.1 * rng.standard_cauchy()
                if math.isfinite(F) and F > 0:
                    F = float(np.clip(F, 0.001, 1.0)); break
            if F is None: F = 0.5
            CR = float(np.clip(rng.normal(muCR, 0.1), 0.0, 1.0))
            v = snap[i] + F * (snap[p] - snap[i]) + F * (snap[r1] - snap[r2])
            mask = rng.random(n) < CR
            if not mask.any(): mask[int(rng.integers(n))] = True
            tr = np.where(mask, v, snap[i])
            trp = project_capped_simplex(tr)
            if trp is None or not is_feasible(trp, n): continue
            if np.max(np.abs(trp - snap[i])) <= 1e-15: continue
            ts = exact_score(trp, n)
            if ts is None or not math.isfinite(ts): continue
            if ts < snap_sc[i] or (ts == snap_sc[i] and np.max(np.abs(trp - snap[i])) > 1e-12):
                new_pop[i] = trp; new_sc[i] = ts
                if ts < snap_sc[i]:
                    imp = True
                    pend_arch.append(snap[i].copy())
                    sF2 += F * F; sF += F; lCR.append(CR)
                if ts < best_s:
                    best_h, best_s = trp.copy(), ts
        pop, sc, archive = new_pop, new_sc, archive + pend_arch
        while len(archive) > P: archive.pop(0)
        if imp and sF > 0:
            muF = float(np.clip(0.9 * muF + 0.1 * sF2 / sF, 0.0, 1.0))
            muCR = float(np.clip(0.9 * muCR + 0.1 * sum(lCR) / len(lCR), 0.0, 1.0))
        stall = 0 if best_s < prev_best else stall + 1
        if stall >= stall_lim and rests < max_rest:
            rests += 1; stall = 0; muF, muCR, archive = 0.5, 0.8, []
            ow = np.argsort(sc)[::-1][:max(2, P // 3)]
            rs = [0.05, 0.15, 0.35, 0.70]
            for cnt, idx in enumerate(ow):
                if cnt == 0:
                    pop[idx], sc[idx] = best_h.copy(), best_s
                elif cnt % 4 == 3:
                    v = rng.beta(0.7, 0.7, n)
                else:
                    v = best_h + rs[cnt % 4] * rng.standard_normal(n)
                p2 = project_capped_simplex(v)
                if p2 is not None and is_feasible(p2, n):
                    psc = exact_score(p2, n)
                    if psc is not None and math.isfinite(psc):
                        pop[idx], sc[idx] = p2, psc
                        if psc < best_s: best_h, best_s = p2.copy(), psc
    return best_h, best_s
