    lag_block = sparse.hstack([sparse.csr_matrix(G), sparse.csr_matrix(np.zeros((L, N))),
                               sparse.csr_matrix(-np.ones((L, 1)))])
    Id = sparse.eye(N, format='csr')
    du_block = sparse.hstack([Id, -Id, sparse.csr_matrix(np.zeros((N, 1)))])
    ndu_block = sparse.hstack([-Id, -Id, sparse.csr_matrix(np.zeros((N, 1)))])
    su_block = sparse.hstack([sparse.csr_matrix(np.zeros((1, N))),
                              sparse.csr_matrix(np.ones((1, N))),
                              sparse.csr_matrix(np.zeros((1, 1)))])
    A_ub = sparse.vstack([lag_block, du_block, ndu_block, su_block], format='csc')
    b_ub = np.concatenate([-q, np.zeros(N), np.zeros(N), [delta]])
    A_eq = sparse.hstack([sparse.csr_matrix(np.ones((1, N))),
                          sparse.csr_matrix(np.zeros((1, N + 1)))])
    bounds = [(-h[i], 1.0 - h[i]) for i in range(N)]
    bounds += [(0.0, None)] * N
    bounds += [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.array([0.0]),
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'maxiter': 500,
                               'primal_feasibility_tolerance': 1e-9,
                               'dual_feasibility_tolerance': 1e-9})
    except Exception:
        return None
    if res is None or res.status != 0 or res.x is None:
        return None
    x = res.x
    if x.shape[0] != nv or not np.all(np.isfinite(x)):
        return None
    return x[:N]

def local_search(h0, delta0, max_iter, dx, N, m):
    h = h0.copy()
    c_cur = exact_score(h, dx)
    if c_cur is None:
        return h0, None
    delta = delta0
    consec_fail = 0
    for _ in range(max_iter):
        if delta < 1e-4:
            break
        q, G = compute_grads(h, dx, N)
        if G is None:
            break
        d = solve_lp(h, q, G, delta, N)
        if d is None:
            delta /= 2.0
            consec_fail += 1
            if consec_fail >= 4:
                break
            continue
        consec_fail = 0
        h_trial = project_capped(h + d, m, N)
        if h_trial is None:
            delta /= 2.0
            continue
        d_actual = h_trial - h
        l1 = np.abs(d_actual).sum()
        if l1 > delta * (1.0 + 1e-6):
            delta /= 2.0
            continue
        c_trial = exact_score(h_trial, dx)
        if c_trial is None:
            delta /= 2.0
            continue
        c_model = float(np.max(q + G @ d_actual))
        p = c_cur - c_model
        a = c_cur - c_trial
        if a > 1e-12 * max(1.0, abs(c_cur)):
            h = h_trial
            c_cur = c_trial
        if p <= 1e-12:
            delta /= 2.0
        else:
            rho = a / p
            if rho < 0.1:
                delta /= 2.0
            elif rho > 0.75 and l1 >= 0.8 * delta:
                delta = min(2.0 * delta, 4.0)
        delta = max(1e-4, min(delta, 4.0))
    return h, c_cur

def add_start(starts, v, m, N):
    p = project_capped(v, m, N)
    if p is None or not check_feasible(p, N, m):
        return
    for s in starts:
        if np.max(np.abs(p - s)) <= 1e-12:
            return
    starts.append(p)

def run(seed=42, **kwargs):
    N = 70
    parent = None
    try:
        arr = np.asarray(initial_h_values, dtype=np.float64)
        if arr.ndim == 1 and len(arr) > 0:
            N = len(arr)
            if np.all(np.isfinite(arr)):
                parent = arr.copy()
    except Exception:
        pass
    dx = 2.0 / N
    m = N / 2.0
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    neutral = np.full(N, 0.5)
    starts = []
    add_start(starts, neutral, m, N)
    if parent is not None:
        add_start(starts, parent, m, N)
    if N <= 256:
        base = parent if parent is not None else neutral
        add_start(starts, base + rng.normal(0, 0.04, N), m, N)
        add_start(starts, base + rng.normal(0, 0.12, N), m, N)
        add_start(starts, rng.uniform(0, 1, N), m, N)
        add_start(starts, rng.beta(0.7, 0.7, N), m, N)
        t = np.arange(N) + 0.5
        add_start(starts, 0.5 + 0.45 * np.cos(2.0 * np.pi * 2.0 * t / N + 0.37), m, N)
        add_start(starts, 0.5 + 0.35 * np.cos(2.0 * np.pi * 3.0 * t / N + 1.1), m, N)
    elif N <= 999:
        base = parent if parent is not None else neutral
        add_start(starts, base + rng.normal(0, 0.08, N), m, N)
        add_start(starts, rng.uniform(0, 1, N), m, N)
        t = np.arange(N) + 0.5
        add_start(starts, 0.5 + 0.45 * np.cos(2.0 * np.pi * 2.0 * t / N + 0.37), m, N)
    best_h = neutral.copy()
    best_c = exact_score(best_h, dx)
    for s in starts:
        c = exact_score(s, dx)
        if c is not None and (best_c is None or c < best_c - 1e-12):
            best_c = c
            best_h = s.copy()
    if N <= 999:
        max_iter = 45 if N <= 256 else 15
        for s in starts:
            h_out, c_out = local_search(s, 0.5, max_iter, dx, N, m)
            if c_out is not None and c_out < best_c - 1e-12:
                best_c = c_out
                best_h = h_out.copy()
    if N <= 256:
        h_pol, c_pol = local_search(best_h, 0.1, 20, dx, N, m)
        if c_pol is not None and c_pol < best_c - 1e-12:
            best_c = c_pol
            best_h = h_pol.copy()
    final = project_capped(best_h, m, N)
    if final is None or not check_feasible(final, N, m):
        final = neutral
    c_final = exact_score(final, dx)
    if c_final is None:
        final = neutral
        c_final = exact_score(final, dx)
    return final, c_final, N
