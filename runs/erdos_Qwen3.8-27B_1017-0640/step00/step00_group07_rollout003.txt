        y += a * np.cos(2.0 * np.pi * k * i / n) + b * np.sin(2.0 * np.pi * k * i / n)
    return y

def _make_binary_start(n, rng):
    y = np.zeros(n)
    perm = rng.permutation(n)
    y[perm[:n // 2]] = 1.0
    if n % 2 == 1:
        y[perm[n // 2]] = 0.5
    return y

def run(seed=42, **kwargs):
    n_starts = kwargs.get('n_starts', 6)
    maxiter = kwargs.get('maxiter', 250)
    ftol = kwargs.get('ftol', 1e-10)
    if not isinstance(n_starts, (int, np.integer)) or n_starts < 1:
        n_starts = 6
    n_starts = min(max(int(n_starts), 1), 8)
    if not isinstance(maxiter, (int, np.integer)) or maxiter < 50:
        maxiter = 250
    maxiter = min(max(int(maxiter), 50), 400)
    try:
        ftol = float(ftol)
        if ftol <= 0 or not np.isfinite(ftol):
            ftol = 1e-10
        ftol = min(max(ftol, 1e-12), 1e-6)
    except (TypeError, ValueError):
        ftol = 1e-10
    try:
        seed = int(seed)
    except (TypeError, ValueError):
        seed = 42
    rng = np.random.default_rng(seed)

    # Determine n and base
    n = 80
    base = np.full(n, 0.5)
    parent_valid = False
    iv = globals().get('initial_h_values', None)
    if iv is not None:
        iv = np.asarray(iv, dtype=np.float64)
        if iv.ndim == 1 and 2 <= len(iv) <= 999:
            n = len(iv)
            if np.all(np.isfinite(iv)) and iv.min() >= -1e-12 and iv.max() <= 1.0 + 1e-12:
                if abs(iv.sum() - n // 2.0) <= 1e-8 * max(1, n):
                    parent_valid = True
                base = np.clip(iv, 0.0, 1.0)
            else:
                base = np.full(n, 0.5)
    base = _project(base, n)
    fallback = base.copy()

    # Build starts
    starts = [base.copy()]
    starts.append(_project(base + 0.03 * rng.standard_normal(n), n))
    starts.append(_project(base + 0.12 * rng.standard_normal(n), n))
    starts.append(_project(rng.random(n), n))
    starts.append(_project(_make_fourier_start(n, rng), n))
    starts.append(_project(_make_binary_start(n, rng), n))
    for _ in range(max(0, n_starts - 6)):
        starts.append(_project(base + 0.08 * rng.standard_normal(n), n))
    starts = starts[:n_starts]

    # Evaluate starts
    incumbent = None
    best_score = np.inf
    for s in starts:
        sc = _score(s)
        if sc < best_score:
            best_score = sc
            incumbent = s.copy()
    if parent_valid:
        ps = _score(fallback)
        if ps <= best_score:
            best_score = ps
            incumbent = fallback.copy()

    # SLSQP optimization (only for n <= 256)
    if n <= 256 and best_score > 0.3808:
        bounds = [(0.0, 1.0)] * n + [(0.0, 2.01)]
        c_eq = {'type': 'eq', 'fun': partial(_eq_fun, n), 'jac': partial(_eq_jac, n)}
        c_in = {'type': 'ineq', 'fun': partial(_in_fun, n), 'jac': partial(_in_jac, n)}
        for start in starts:
            if best_score <= 0.3808:
                break
            q0, _ = _corr_qG(start, n)
            t0 = max(0.0, np.max(q0)) + 1e-8
            z0 = np.append(start, t0)
            try:
                res = minimize(
                    partial(_obj, n), z0, method='SLSQP',
                    jac=partial(_obj_grad, n),
                    constraints=[c_eq, c_in], bounds=bounds,
                    options={'ftol': ftol, 'maxiter': maxiter, 'disp': False}
                )
                if res is not None and res.x is not None:
                    h_cand = _project(res.x[:n], n)
                    sc = _score(h_cand)
                    if sc < best_score - 1e-12:
                        best_score = sc
                        incumbent = h_cand
            except Exception:
                pass

    # Final validation
    if incumbent is None:
        incumbent = fallback.copy()
    incumbent = _project(incumbent, n)
    c5 = _score(incumbent)
    if not np.isfinite(c5) or c5 > 2.0:
        incumbent = _project(np.full(n, 0.5), n)
        c5 = _score(incumbent)
    # Ensure exact mass
    resid = 0.5 * n - incumbent.sum()
    if abs(resid) > 1e-10 * max(1, n):
        slack = 1.0 - incumbent if resid > 0 else incumbent
        idxs = np.argsort(-slack if resid > 0 else slack)
        for idx in idxs:
            if slack[idx] > 1e-12:
                step = min(abs(resid), 0.5 * slack[idx])
                incumbent[idx] += step if resid > 0 else -step
                break
        incumbent = np.clip(incumbent, 0.0, 1.0)
        c5 = _score(incumbent)
    return incumbent, float(c5), int(n)
