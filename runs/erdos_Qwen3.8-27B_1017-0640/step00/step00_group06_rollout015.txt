    hi_g = np.ones(n)
    for sig in [0.05, 0.15, 0.30]:
        v = base + sig * rng.standard_normal(n)
        pr = project_bounded(v, lo_g, hi_g, m)
        if pr is not None:
            starts.append(pr)
    bv = rng.beta(0.7, 0.7, n)
    pr = project_bounded(bv, lo_g, hi_g, m)
    if pr is not None:
        starts.append(pr)
    ph = rng.uniform(0.0, 2.0 * math.pi)
    sv = 0.5 + 0.45 * np.sin(2.0 * math.pi * np.arange(n, dtype=np.float64) / n + ph)
    pr = project_bounded(sv, lo_g, hi_g, m)
    if pr is not None:
        starts.append(pr)
    if np.max(np.abs(fb - base)) > 1e-12:
        starts.append(fb.copy())

    starts = [starts[0]]
    for s in starts[1:]:
        if not any(np.max(np.abs(s - u)) < 1e-12 for u in starts):
            starts.append(s)

    best_h = protected.copy()
    best_C = exact_score(best_h, n)

    for idx, s in enumerate(starts):
        Cs = exact_score(s, n)
        if Cs < best_C - 1e-12:
            best_h = s.copy()
            best_C = Cs
        if idx == 0:
            bud = max(8, min(28, math.ceil(28.0 * 95.0 / n)))
        else:
            bud = max(4, min(12, math.ceil(12.0 * 95.0 / n)))
        bh, bC = optimize_start(s, n, m, bud)
        if bC < best_C - 1e-12:
            best_h = bh.copy()
            best_C = bC

    if not is_feasible(best_h, n):
        pr = project_bounded(best_h, np.zeros(n), np.ones(n), m)
        best_h = pr if pr is not None else protected.copy()
    best_C = exact_score(best_h, n)
    return best_h, float(best_C), int(n)
