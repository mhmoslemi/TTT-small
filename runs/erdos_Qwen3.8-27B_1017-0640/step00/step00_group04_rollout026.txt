        if f1 <= f2:
            lo = c1
            c1, f1 = c2, f2
            c2 = lo + r * (hi - lo)
            f2 = lv(c2, q0, a, b, n)
        else:
            hi = c2
            c2, f2 = c1, f1
            c1 = hi - r * (hi - lo)
            f1 = lv(c1, q0, a, b, n)
        if f1 < bf:
            bt, bf = c1, f1
        elif f2 < bf:
            bt, bf = c2, f2
    return bt, bf

def line_search(h, d, n, q0, cur):
    lo, hi = fiv(h, d)
    if not (math.isfinite(lo) and math.isfinite(hi)) or hi - lo < 1e-12:
        return h, q0, cur
    b = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    a = -np.correlate(d, d, 'full')
    if not (np.all(np.isfinite(a)) and np.all(np.isfinite(b))):
        return h, q0, cur
    ng = 17
    ts = np.linspace(lo, hi, ng)
    vals = np.array([lv(t, q0, a, b, n) for t in ts])
    cset = {0, ng - 1, int(np.argmin(vals))}
    for i in range(1, ng - 1):
        if vals[i] <= vals[i - 1] and vals[i] <= vals[i + 1]:
            cset.add(i)
    pairs = [(vals[i], i) for i in cset]
    pairs.sort()
    pairs = pairs[:2]
    best_t, best_f = 0.0, cur
    for _, i in pairs:
        if 0 < i < ng - 1:
            lo_b, hi_b = ts[i - 1], ts[i + 1]
        elif i == 0:
            lo_b, hi_b = ts[0], ts[1]
        else:
            lo_b, hi_b = ts[i - 1], ts[i]
        rt, rf = gold(lo_b, hi_b, q0, a, b, n)
        if rf < best_f:
            best_t, best_f = rt, rf
    v0 = lv(0.0, q0, a, b, n)
    if v0 < best_f:
        best_t, best_f = 0.0, v0
    if best_f >= cur - 1e-12:
        return h, q0, cur
    hn = h + best_t * d
    hn = proj(hn, n)
    if hn is None:
        return h, q0, cur
    sn = sc(hn, n)
    if sn is None or sn >= cur - 1e-12:
        return h, q0, cur
    qn = np.correlate(hn, 1.0 - hn, 'full')
    return hn, qn, sn

def gen_starts(n, rng, fallback):
    starts = []
    seen = []
    def try_add(v):
        if v is None:
            return
        for s in seen:
            if np.max(np.abs(v - s)) < 1e-9:
                return
        seen.append(v.copy())
        starts.append(v)
    try_add(fallback.copy())
    blk = np.zeros(n)
    half = n // 2
    blk[:half] = 1.0
    if n % 2 == 1:
        blk[half] = 0.5
    try_add(blk)
    try_add(blk[::-1].copy())
    try_add(proj(1.0 - fallback, n))
    for s in (0.15, 0.05, 0.20, 0.35, 0.50):
        try_add(proj(fallback + s * rng.standard_normal(n), n))
    try_add(proj(rng.uniform(-0.25, 1.25, n), n))
    cos_v = 0.5 + 0.25 * np.cos(2.0 * np.pi * np.arange(n) / n)
    try_add(proj(cos_v, n))
    try_add(proj(np.full(n, 0.5), n))
    return starts

def run(seed=42, **kwargs):
    try:
        s = int(seed)

    rng = np.random.default_rng(s)

        if 2 <= len(p) <= 4096 and np.all(np.isfinite(p)) and np.max(np.abs(p)) < 1e6:
            n = len(p)
            if p.min() >= -1e-10 and p.max() <= 1 + 1e-10 and abs(p.sum() - n / 2.0) <= 1e-10 * n:
                fb = p.copy()

    except NameError:

    except Exception:
        pass
    if fb is None:

        fb_sc = sc(fb, n)
    best_h = fb.copy()
    best_sc = fb_sc
    starts = gen_starts(n, rng, fb)
    if len(starts) > n_starts:
        starts = starts[:n_starts]
    for st in starts:
        try:
            h = st.copy()
            c = sc(h, n)
            if c is None:
                continue
            q0 = np.correlate(h, 1.0 - h, 'full')
            if c < best_sc - 1e-12:
                best_h, best_sc = h.copy(), c
            improved = True
            for sw in range(max_sweeps):
                if not improved:
                    break
                improved = False
                if n <= 64:
                    pairs = [(i, j) for i in range(n - 1) for j in range(i + 1, n)]
                    rng.shuffle(pairs)
                    for (i, j) in pairs:
                        d = np.zeros(n)
                        d[i], d[j] = 1.0, -1.0
                        h, q0, c = line_search(h, d, n, q0, c)
                        if c < best_sc - 1e-12:
                            best_h, best_sc = h.copy(), c

                            pass
                elif n <= 128:
                    npairs = min(n * (n - 1) // 2, 4 * n)
                    allpairs = [(i, j) for i in range(n - 1) for j in range(i + 1, n)]
                    rng.shuffle(allpairs)
                    for (i, j) in allpairs[:npairs]:
                        d = np.zeros(n)
                        d[i], d[j] = 1.0, -1.0
                        h, q0, c = line_search(h, d, n, q0, c)
                        if c < best_sc - 1e-12:
                            best_h, best_sc = h.copy(), c
                nrnd = max(4, n // 4)
                for _ in range(nrnd):
                    z = rng.standard_normal(n)
                    z -= z.mean()
                    z[-1] = -z[:-1].sum()
                    mx = np.max(np.abs(z))
                    if mx < 1e-15:
                        continue
                    d = z / mx
                    h, q0, c = line_search(h, d, n, q0, c)
                    if c < best_sc - 1e-12:
                        best_h, best_sc = h.copy(), c
                if not improved:
                    c_check = sc(h, n)
                    if c_check is not None and c_check < best_sc - 1e-12:
                        best_h, best_sc = h.copy(), c_check
        except Exception:

    if not ok(best_h, n):
        best_h = proj(best_h, n)
    if best_h is None or not ok(best_h, n):
        best_h = fb.copy()
        best_sc = fb_sc

        best_sc = sc(best_h, n)
        if best_sc is None:
            best_h = fb.copy()
            best_sc = sc(fb, n)
    best_h = np.asarray(best_h, dtype=np.float64).ravel()
    return best_h, best_sc, int(n)
