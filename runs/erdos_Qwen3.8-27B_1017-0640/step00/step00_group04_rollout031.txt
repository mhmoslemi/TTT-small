        if rem < cnt:

        rem -= cnt
        i += 1
    return (n - 2, n - 1)

def run(seed=42, **kwargs):
    try:
        seed = int(seed)
    except (TypeError, ValueError):
        seed = 42
    rng = np.random.default_rng(seed)
    n_starts, max_sweeps = 6, 3
    if 'n_starts' in kwargs:
        try:
            n_starts = max(4, min(8, int(kwargs['n_starts'])))
        except (TypeError, ValueError):
            pass
    if 'max_sweeps' in kwargs:
        try:
            max_sweeps = max(1, min(5, int(kwargs['max_sweeps'])))
        except (TypeError, ValueError):
            pass
    parent, n = None, 42
    try:
        ph = np.array(initial_h_values, dtype=np.float64).ravel()
        if 2 <= len(ph) <= 4096 and np.all(np.isfinite(ph)) and float(np.max(np.abs(ph))) <= 1e6:
            pn = len(ph)
            if float(np.min(ph)) >= 0.0 and float(np.max(ph)) <= 1.0:
                mev = abs(float(np.sum(ph)) - pn / 2.0)
                if mev <= 1e-10 * max(1, pn):
                    p = ph.copy() if mev <= 5e-13 * max(1, pn) else capped_simplex_project(ph, pn, pn / 2.0)
                    if p is not None and is_feasible(p, pn):
                        parent, n = p, pn
            else:
                p = capped_simplex_project(ph, pn, pn / 2.0)
                if p is not None and is_feasible(p, pn):
                    parent, n = p, pn
    except (NameError, TypeError, ValueError, IndexError):
        pass
    if parent is None:
        fb = np.zeros(n)
        fb[:n // 2] = 1.0
        if n % 2 == 1:
            fb[n // 2] = 0.5
        parent = fb
    target = n / 2.0
    fb_score = exact_score(parent, n)
    if fb_score is None:
        fb_score = 1.0
    gbest, gbest_s = parent.copy(), fb_score
    all_s = build_starts(n, rng, parent)
    raw = [parent.copy()]
    raw.extend(all_s[1:])
    proc = []
    for rs in raw:
        if is_feasible(rs, n):
            proc.append(rs)
        else:
            p = capped_simplex_project(rs, n, target)
            if p is not None and is_feasible(p, n):
                proc.append(p)
    starts = dedup_starts(proc)[:n_starts]
    for si in range(len(starts)):
        h = starts[si].copy()
        cs = exact_score(h, n)
        if cs is None:
            continue
        hc = np.correlate(h, 1.0 - h, mode='full')
        for sweep in range(max_sweeps):
            acc = False
            if n <= 128:
                np_total = n * (n - 1) // 2
                if n <= 64:
                    pidx = np.arange(np_total)
                    rng.shuffle(pidx)
                    plist = pidx
                else:
                    plist = rng.choice(np_total, size=min(np_total, 4 * n), replace=False)
                for pi in plist:
                    ii, jj = pair_to_ij(pi, n)
                    d = np.zeros(n)
                    d[ii], d[jj] = 1.0, -1.0
                    try:
                        bt = do_line_search(h, d, hc, n, cs)
                        if bt is not None:
                            res = accept_candidate(h, bt, d, n, target, cs)
                            if res is not None:
                                h, cs = res
                                hc = np.correlate(h, 1.0 - h, mode='full')
                                acc = True
                                if cs < gbest_s - 1e-12:
                                    gbest, gbest_s = h.copy(), cs
                    except (FloatingPointError, OverflowError, ValueError):
                        pass
            for _ in range(max(4, n // 4)):
                z = rng.standard_normal(n)
                z -= np.mean(z)
                z[-1] = -np.sum(z[:-1])
                mx = float(np.max(np.abs(z)))
                if mx < 1e-14:
                    continue
                d = z / mx
                try:
                    bt = do_line_search(h, d, hc, n, cs)
                    if bt is not None:
                        res = accept_candidate(h, bt, d, n, target, cs)
                        if res is not None:
                            h, cs = res
                            hc = np.correlate(h, 1.0 - h, mode='full')
                            acc = True
                            if cs < gbest_s - 1e-12:
                                gbest, gbest_s = h.copy(), cs
                except (FloatingPointError, OverflowError, ValueError):
                    pass
            if not acc:
                break
    result = gbest.copy()
    if not is_feasible(result, n):
        p = capped_simplex_project(result, n, target)
        result = p if (p is not None and is_feasible(p, n)) else parent.copy()
    fs = exact_score(result, n)
    if fs is None:
        result, fs = parent.copy(), fb_score
    return (result.astype(np.float64), float(fs), int(n))
