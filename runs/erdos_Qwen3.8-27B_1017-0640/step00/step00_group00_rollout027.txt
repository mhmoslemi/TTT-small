def _project(y, m, n):

def _pair_sweep(h, q, n, n_sweeps, pair_list_func, rng):
    """Run pair sweeps. Returns updated h, q, score."""
    m = n / 2.0
    dx = 2.0 / n
    score, _ = _score(h, n)
    if score is None:
        score = np.inf
    best_h = h.copy()
    best_score = score
    best_q = q.copy()
    
    for sw in range(n_sweeps):
        pairs = pair_list_func(n, rng)
        sweep_start_score = score
        any_accepted = False
        for (a, bb) in pairs:
            p = bb - a
            if p <= 0:
                continue
            L = max(-h[a], h[bb] - 1.0)
            U = min(1.0 - h[a], h[bb])
            if U - L < 1e-14:
                continue
            # Compute b vector
            Ga = _coord_grad(h, n, a)
            Gb = _coord_grad(h, n, bb)
            bv = Ga - Gb
            # A vector
            A = np.zeros(2 * n - 1)
            special = [n - 1, n - 1 + p, n - 1 - p]
            A[n - 1] = -2.0 * dx
            A[n - 1 + p] = dx
            A[n - 1 - p] = dx
            if not np.all(np.isfinite(bv)) or not np.all(np.isfinite(A)):
                continue
            # Minimize
            t = _line_min(q, bv, A, L, U, special)
            # Check predicted improvement
            pred_max = np.max(q + bv * t + A * t * t)
            if pred_max >= score - 1e-12 or abs(t) < 1e-12:
                continue
            # Form trial
            trial = h.copy()
            trial[a] = h[a] + t
            trial[bb] = h[bb] - t
            if not np.all(np.isfinite(trial)):
                continue
            if trial.min() < -1e-10 or trial.max() > 1.0 + 1e-10:
                # Project
                proj, ok = _project(trial, m, n)
                if not ok:
                    continue
                trial = proj
            if not _feasible(trial, n):
                continue
            sc, q_new = _score(trial, n)
            if sc is None or sc >= score - 1e-12:
                continue
            # Accept
            h = trial
            score = sc
            q = q_new
            any_accepted = True
            if score < best_score - 1e-12:
                best_score = score
                best_h = h.copy()
                best_q = q.copy()
        if not any_accepted or sweep_start_score - score < 1e-10:
            break
    return best_h, best_q, best_score

def _make_pairs(n, rng, budget):
    """Generate a list of (a,b) pairs with a<b."""
    pairs = []
    if n <= 64:
        for a in range(n):
            for b in range(a + 1, n):
                pairs.append((a, b))
        rng.shuffle(pairs)
    else:
        # Stratified by separation
        seps = []
        s = 1
        while s < n:
            seps.append(s)
            s *= 2
        used = set()
        per_sep = max(1, budget // len(seps))
        for sep in seps:
            count = 0
            candidates = []
            for a in range(0, n - sep, max(1, (n - sep) // per_sep)):
                b = a + sep
                if (a, b) not in used:
                    candidates.append((a, b))
            rng.shuffle(candidates)
            for pr in candidates[:per_sep]:
                if len(pairs) >= budget:
                    break
                pairs.append(pr)
                used.add(pr)
        # Fill remaining
        while len(pairs) < budget:
            a = rng.integers(0, n - 1)
            b = rng.integers(a + 1, n)
            if (a, b) not in used:
                pairs.append((a, b))
                used.add((a, b))
        rng.shuffle(pairs)
    return pairs
