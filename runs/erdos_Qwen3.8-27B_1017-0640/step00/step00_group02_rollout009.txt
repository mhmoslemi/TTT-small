import numpy as np
from scipy.optimize import minimize

def project_capped(v, S):
    n = len(v)
    v = np.clip(v.astype(np.float64), -1e6, 1e6)
    lo, hi = v.min() - 1.0, v.max() + 1.0
    lam = (lo + hi) * 0.5
    for _ in range(100):
        x = np.clip(v - lam, 0.0, 1.0)
        if x.sum() > S:
            lo = lam
        else:
            hi = lam
        nm = (lo + hi) * 0.5
        if nm == lam:
            break
        lam = nm
    x = np.clip(v - lam, 0.0, 1.0)
    for _ in range(100):
        r = S - x.sum()
        if abs(r) < 1e-14:
            break
        if r > 0:
            idx = np.where(1.0 - x > 1e-12)[0]
            if not idx.size:
                break
            x[idx] = np.minimum(x[idx] + r / idx.size, 1.0)
        else:
            idx = np.where(x > 1e-12)[0]
            if not idx.size:
                break
            x[idx] = np.maximum(x[idx] + r / idx.size, 0.0)
    return x

def exact_score(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def compute_qG(h, n):
    q = np.zeros(2 * n - 1)
    G = np.zeros((2 * n - 1, n))
    q[n - 1] = np.dot(h, 1.0 - h)
    G[n - 1] = 1.0 - 2.0 * h
    for s in range(1, n):
        m = n - s
        q[n - 1 + s] = np.dot(h[s:], 1.0 - h[:m])
        g = np.zeros(n)
        g[s:] = 1.0 - h[:m]
        g[:m] -= h[s:]
        G[n - 1 + s] = g
        q[n - 1 - s] = np.dot(h[:m], 1.0 - h[s:])
        g2 = np.zeros(n)
        g2[:m] = 1.0 - h[s:]
        g2[s:] -= h[:m]
        G[n - 1 - s] = g2
    return q, G

def obj_fun(z):
    return z[-1]

def obj_jac(z):
    j = np.zeros_like(z)
    j[-1] = 1.0
    return j

def ineq_fun(z, n, dx):
    h = z[:-1]
    q, _ = compute_qG(h, n)
    return z[-1] - dx * q

def ineq_jac(z, n, dx):
    h = z[:-1]
    _, G = compute_qG(h, n)
    jl = np.zeros((2 * n - 1, len(z)))
    jl[:, :-1] = -dx * G
    jl[:, -1] = 1.0
    return jl

def eq_fun(z, n, S):
    return z[:n].sum() - S

def eq_jac(z, n, S):
    j = np.zeros(len(z))
    j[:n] = 1.0
    return j

def attempt_solve(h0, n, dx, S, bounds, cons, incumbent, incumbent_score):
    z0 = np.append(h0.copy(), min(max(exact_score(h0, n), 0.0), 2.0))
    try:
        res = minimize(obj_fun, z0, jac=obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 400, 'ftol': 1e-10, 'disp': False})
        if res.x is not None and len(res.x) == n + 1 and np.all(np.isfinite(res.x)):
            h_cand = project_capped(res.x[:n], S)
            if np.all(np.isfinite(h_cand)) and np.all(h_cand >= -1e-10) and np.all(h_cand <= 1.0 + 1e-10):
                h_cand = np.clip(h_cand, 0.0, 1.0)
                sc = exact_score(h_cand, n)
                if sc < incumbent_score - 1e-12:
                    return h_cand, sc
    except Exception:
        pass
    return incumbent, incumbent_score

    rng = np.random.default_rng(seed)
    n = 45
    parent = None

    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]

    incumbent_score = exact_score(incumbent, n)
    return (incumbent, incumbent_score, n)
