                if F_y_val is None:
                    mu *= 0.5
                    continue
                if F_y_val <= F_cur + 1e-4 * min(gd, 0.0) + 1e-14 * max(1.0, abs(F_cur)):
                    current = y
                    F_cur = F_y_val
                    _, g = smooth_val_grad(current, tau, dx, True)
                    if g is None:
                        stage_done = True
                        break
                    accepted = True
                    eta = min(1.5 * mu, 10.0)
                    break
                mu *= 0.5
            if stage_done or not accepted:
                break
    return local_best, best_score

def run(seed=42, **kwargs):
    try:
        seed = int(seed)
        if seed < 0 or seed > 2**32 - 1:
            seed = 42
    except (ValueError, TypeError, OverflowError):
        seed = 42
    rng = np.random.default_rng(seed)

    parent_h = None
    n = 47
    parent = globals().get('initial_h_values', None)
    if parent is not None:
        try:
            p = np.array(parent, dtype=np.float64).ravel()
            if p.size >= 2 and p.size < 1000 and np.all(np.isfinite(p)):
                if np.min(p) >= -1e-7 and np.max(p) <= 1.0 + 1e-7:
                    proj = project_capped_simplex(p)
                    if proj is not None:
                        parent_h = proj
                        n = p.size
        except Exception:
            pass

    if parent_h is None:
        parent_h = np.full(n, 0.5, dtype=np.float64)
    if n <= 64:
        temps = [0.04, 0.01, 0.0025, 0.0005, 0.0001]
        iters = 80
        max_starts = 8
        restart_scales = [0.04, 0.12]
    elif n <= 128:
        temps = [0.04, 0.01, 0.0025, 0.0005]
        iters = 45
        max_starts = 5
        restart_scales = [0.06]
    elif n <= 512:
        temps = [0.03, 0.006, 0.001]
        iters = 25
        max_starts = 3
        restart_scales = []
    else:
        temps = [0.02, 0.003]
        iters = 12
        max_starts = 2
        restart_scales = []
    dx = 2.0 / n
    starts = [parent_h.copy(), np.full(n, 0.5, dtype=np.float64)]
    base = parent_h.copy()
    for f in [1, 2]:
        phi = rng.uniform(0.0, 2.0 * math.pi)
        idx = np.arange(n, dtype=np.float64) + 0.5
        v = 0.5 + 0.30 * np.sin(2.0 * math.pi * f * idx / n + phi)
        proj = project_capped_simplex(v)
        if proj is not None:
            starts.append(proj)
    for sc in [0.06, 0.18]:
        v = base + sc * rng.standard_normal(n)
        proj = project_capped_simplex(v)
        if proj is not None:
            starts.append(proj)
    for _ in range(2):
        v = rng.uniform(0.0, 1.0, n)
        proj = project_capped_simplex(v)
        if proj is not None:
            starts.append(proj)
    if parent_h is None or parent_h.tolist() != [0.5]*n:
        phi = rng.uniform(0.0, 2.0 * math.pi)
        idx = np.arange(n, dtype=np.float64) + 0.5
        v = 0.5 + 0.30 * np.sin(2.0 * math.pi * 3.0 * idx / n + phi)
        proj = project_capped_simplex(v)
        if proj is not None:
            starts.append(proj)
    unique_starts = [starts[0]]
    for s in starts[1:]:
        dup = False
        for u in unique_starts:
            if np.max(np.abs(s - u)) <= 1e-9:
                dup = True
                break
        if not dup:
            unique_starts.append(s)
    starts = unique_starts[:max_starts]
    global_best = parent_h.copy()
    global_best_score = exact_score(global_best)
    if global_best_score is None:
        global_best_score = 1e10
    for s in starts:
        try:
            sc = exact_score(s)
            if sc is not None and sc < global_best_score - 1e-12:
                global_best_score = sc
                global_best = s.copy()
            res, res_score = projected_smooth_search(s, temps, iters, dx)
            if res_score is not None and res_score < global_best_score - 1e-12:
                global_best_score = res_score
                global_best = res.copy()
        except Exception:
            continue
    for sc in restart_scales:
        try:
            v = global_best + sc * rng.standard_normal(n)
            proj = project_capped_simplex(v)
            if proj is None:
                continue
            sc2 = exact_score(proj)
            if sc2 is not None and sc2 < global_best_score - 1e-12:
                global_best_score = sc2
                global_best = proj.copy()
            res, res_score = projected_smooth_search(proj, temps, iters, dx)
            if res_score is not None and res_score < global_best_score - 1e-12:
                global_best_score = res_score
                global_best = res.copy()
        except Exception:
            continue
    final = project_capped_simplex(global_best)
    if final is not None and check_feasible(final):
        global_best = final
    score = exact_score(global_best)
    if score is None:
        global_best = parent_h.copy()
        score = exact_score(global_best)
    if score is None:
        global_best = np.full(n, 0.5, dtype=np.float64)
        score = exact_score(global_best)
        if score is None:
            score = 1.0
    return global_best, score, n
