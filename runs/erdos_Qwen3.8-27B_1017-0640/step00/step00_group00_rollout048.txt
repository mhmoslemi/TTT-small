def project_capped_simplex(y, n, m):

        else:
            hi = mid
    lam = 0.5 * (lo + hi)

def try_cvxpy_mp(x, q, G, dx, n, L):

        t = cp.Variable()

        upper = 1.0 - x

            d >= lower,
            d <= upper,
            cp.sum(d) == 0,
            cp.sum_squares(d) <= s,
            q + G @ d + dx * s <= t * np.ones(L)

import numpy as np

def project_capped_simplex(y, n, m):
    y = np.clip(y.astype(np.float64), -1e6, 1e6)
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > m:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    x = np.clip(y - lam, 0.0, 1.0)
    for _ in range(10):
        res = m - float(np.sum(x))
        if abs(res) < 1e-12:
            break
        if res > 0:

                if room > 1e-15:
                    add = min(room, res)
                    x[i] += add
                    res -= add
                    if abs(res) < 1e-12:
                        break
        else:

def exact_score(h):
    n = len(h)
    if n == 0 or not np.all(np.isfinite(h)):
        return None
    if np.min(h) < -1e-10 or np.max(h) > 1.0 + 1e-10:
        return None

        return None

    raw = np.correlate(h, 1.0 - h, mode='full')
    if len(raw) != 2 * n - 1 or not np.all(np.isfinite(raw)):
        return None
    c = float(np.max(raw * dx))
    if not math.isfinite(c):
        return None
    return c

def compute_gradients(h, dx):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n), dtype=np.float64)
    G[n - 1] = dx * (1.0 - 2.0 * h)
    for p in range(1, n):

        G[n - 1 + p, p:] -= dx * h[:-p]
        G[n - 1 - p, :n - p] -= dx * h[p:]
        G[n - 1 - p, p:] += dx * (1.0 - h[:-p])
    if not np.all(np.isfinite(G)):
        return None
    return G

def try_cvxpy_mp(x, q, G, dx, n, L):
    try:
        import cvxpy as cp
        d = cp.Variable(n)
        s = cp.Variable()
        t = cp.Variable()
        ones_L = np.ones(L)
        cons = [d >= -x, d <= 1.0 - x, cp.sum(d) == 0,
                cp.sum_squares(d) <= s,
                q + G @ d + dx * s <= t * ones_L]
        prob = cp.Problem(cp.Minimize(t), cons)
        for solver in ['ECOS', 'CVXOPT']:
            try:
                prob.solve(solver=solver, abstol=1e-8, reltol=1e-8, feastol=1e-8)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    dv = d.value
                    if dv is not None and np.all(np.isfinite(dv)):
                        if np.min(dv) >= -x - 1e-4 and np.max(dv) <= 1.0 - x + 1e-4:

        pass
    return None

        y = project_capped_simplex(cur + d, n, n / 2.0)

        best_sc = cur_score

            break
