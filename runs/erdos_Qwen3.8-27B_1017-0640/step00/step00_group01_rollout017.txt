def project_capped_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1 or v.size < 2:
        return None
    if not np.all(np.isfinite(v)) or float(np.max(np.abs(v))) > 1e6:
        return None
    n = v.size
    s = n / 2.0
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = lo + (hi - lo) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(3):
        r = s - float(np.sum(x))
        if abs(r) <= 5e-15:
            break
        if r > 0:
            for i in range(n):
                add = min(r, 1.0 - x[i])
                if add > 0:
                    x[i] += add
                    r -= add
        else:
            for i in range(n):
                sub = min(-r, x[i])
                if sub > 0:
                    x[i] -= sub
                    r += sub
    if not np.all(np.isfinite(x)) or float(np.min(x)) < -1e-12 or float(np.max(x)) > 1.0 + 1e-12:
        return None
    if abs(float(np.sum(x)) - s) > 1e-9:
        return None
    return x

        n = h.size
        if n < 2 or not np.all(np.isfinite(h)):
            return None
        corr = np.correlate(h, 1.0 - h, mode='full')
        val = float(np.max(corr)) * (2.0 / n)
        return val if math.isfinite(val) else None
    except Exception:
        return None

def lag_vals(h):
    n = h.size
    dx = 2.0 / n
    vals = np.empty(2 * n - 1, dtype=np.float64)
    for k in range(2 * n - 1):
        d = k - (n - 1)
        if d >= 0:
            vals[k] = float(np.dot(h[:n - d], 1.0 - h[d:n])) * dx
        else:
            vals[k] = float(np.dot(h[-d:], 1.0 - h[:-d])) * dx
    return vals

def smooth_surr(h, tau, need_grad):
    a = lag_vals(h)
    if not np.all(np.isfinite(a)):
        return None, None
    A = float(np.max(a))
    ea = np.exp((a - A) / tau)
    Z = float(np.sum(ea))
    if not math.isfinite(Z) or Z <= 0:
        return None, None
    F = A + tau * math.log(Z)
    p = ea / Z
    if not need_grad:
        return F, None
    n = h.size
    dx = 2.0 / n
    g = np.zeros(n, dtype=np.float64)
    for k in range(2 * n - 1):
        d = k - (n - 1)
        w = dx * p[k]
        if w < 1e-15:
            continue
        if d >= 0:
            g[:n - d] += w * (1.0 - h[d:n])
            g[d:n] -= w * h[:n - d]
        else:
            g[-d:] += w * (1.0 - h[:-d])
            g[:-d] -= w * h[-d:]
    if not np.all(np.isfinite(g)):
        return None, None
    return F, g

    if lb_s is None:
        lb_s = 1e9
    for tau in temps:
        cur = lb.copy()
        eta = 1.0
        for _it in range(iters):
            F_c, g = smooth_surr(cur, tau, True)
            if F_c is None or g is None:
                break

                    cur = y

                    accepted = True
                    break
                eta *= 0.5

                break
