        if quotas[i] <= 0:
            continue
        _, _, sb, sbs = one_start_search(scored[i][0], scored[i][1], rng, n, m, quotas[i])
        if sbs < global_best_s - 1e-12:
            global_best = sb.copy()
            global_best_s = sbs
    final = global_best
    if not is_feasible(final, n, m):
        rep = project_capped(final, n, m)
        if rep is not None and is_feasible(rep, n, m):
            final = rep
        else:
            final = fallback.copy()
    c5 = exact_score(final, n)
    return (final, c5, n)
