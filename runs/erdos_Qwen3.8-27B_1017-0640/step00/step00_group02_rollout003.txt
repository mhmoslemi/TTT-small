    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    q = np.zeros(2 * n - 1)
    G = np.zeros((2 * n - 1, n))
    q[n - 1] = np.sum(h * (1.0 - h))
    G[n - 1, :] = 1.0 - 2.0 * h
    for s in range(1, n):
        m = n - s
        q[n - 1 + s] = np.dot(h[s:], 1.0 - h[:m])
        G[n - 1 + s, s:] += 1.0 - h[:m]
        G[n - 1 + s, :m] -= h[s:]
        q[n - 1 - s] = np.dot(h[:m], 1.0 - h[s:])
        G[n - 1 - s, :m] += 1.0 - h[s:]
        G[n - 1 - s, s:] -= h[:m]

    return z[-1]

def obj_jac(z, n):
    j = np.zeros(n + 1)

def ineq_jac(z, n):
