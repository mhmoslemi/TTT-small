    temps, scales, nprop, nchains, cap = anneal_params(n)
    cands = []
    scs = []
    insert_cand(parent, cands, scs, cap, n)
    insert_cand(parent[::-1].copy(), cands, scs, cap, n)
    insert_cand(1.0 - parent, cands, scs, cap, n)
    insert_cand(1.0 - parent[::-1].copy(), cands, scs, cap, n)
    insert_cand(np.full(n, 0.5), cands, scs, cap, n)
    for f in range(1, 5):
        phi = rng.uniform(0, 2 * math.pi)
        xv = np.arange(n, dtype=np.float64) + 0.5
        insert_cand(0.5 + 0.30 * np.sin(2 * math.pi * f * xv / n + phi), cands, scs, cap, n)
    for sc in [0.03, 0.10, 0.25, 0.50]:
        insert_cand(parent + sc * rng.standard_normal(n), cands, scs, cap, n)
    for _ in range(4):
        insert_cand(rng.uniform(0, 1, n), cands, scs, cap, n)
    for _ in range(4):
        insert_cand(rng.beta(0.7, 0.7, n), cands, scs, cap, n)
    Lfac = None
    if n <= 64:
        hr, Xr = solve_sdp(n)
        if hr is not None and np.all(np.abs(hr) <= 1.5) and abs(math.fsum(hr.tolist()) - n / 2.0) <= 1e-3:
            insert_cand(hr, cands, scs, cap, n)
            try:
                S = 0.5 * (Xr + Xr.T) - np.outer(hr, hr)
                Pm = np.eye(n) - np.ones((n, n)) / n
                Sc = Pm @ S @ Pm
                Sc = 0.5 * (Sc + Sc.T)
                ev, evec = np.linalg.eigh(Sc)
                mx = max(1.0, float(np.max(np.abs(ev))))
                if float(np.min(ev)) >= -1e-4 * mx:
                    ev = np.clip(ev, 0, None)
                    thr = 1e-12 * mx
                    mk = ev > thr
                    if np.any(mk):
                        kk = np.where(mk)[0]
                        Lfac = evec[:, kk] * np.sqrt(ev[kk])
                        Lfac -= Lfac.mean(axis=0)
                        for ei in range(min(8, len(kk))):
                            for sgn in [1, -1]:
                                for ds in [0.75, 1.5]:
                                    vv = hr + sgn * ds * math.sqrt(max(ev[kk[ei]], 0)) * evec[:, kk[ei]]
                                    insert_cand(vv, cands, scs, cap, n)
                        for sc2 in [0.75, 1.5]:
                            for _ in range(12):
                                zz = rng.standard_normal(Lfac.shape[1])
                                insert_cand(hr + sc2 * (Lfac @ zz), cands, scs, cap, n)
                        for _ in range(16):
                            bv = (rng.random(n) < np.clip(hr, 0, 1)).astype(np.float64)
                            insert_cand(bv, cands, scs, cap, n)
            except Exception:
                pass
    # Chain starts
    chain_ids = []
    if len(cands) > 0:
        ord_idx = np.argsort(scs).tolist()
        if 0 in ord_idx:
            ord_idx.remove(0)
        ord_idx.insert(0, 0)
        for ii in ord_idx:
            if len(chain_ids) >= nchains:
                break
            ok = True
            for cs in chain_ids:
                if np.max(np.abs(cands[ii] - cands[cs])) <= 1e-8:
                    ok = False
                    break
            if ok:
                chain_ids.append(ii)
    best_h = None
    best_sc = None
    for i in range(len(cands)):
        sc = exact_score(cands[i], n)
        if sc is not None and (best_sc is None or sc < best_sc - 1e-12):
            best_h = cands[i].copy()
            best_sc = sc
    Lnorm = None
    if Lfac is not None:
        try:
            ds = float(np.sum(np.clip(Lfac ** 2, 0, None)) / n)
            if ds > 1e-12:
                Lnorm = Lfac / math.sqrt(ds)
        except Exception:
            pass
    for ci in chain_ids:
        try:
            cur = cands[ci].copy()
            cur_sc = exact_score(cur, n)
            if cur_sc is None:
                continue
            for st in range(len(temps)):
                T = temps[st]
                sg = scales[st]
                for _ in range(nprop):
                    if Lnorm is not None and rng.random() < 0.65:
                        zz = rng.standard_normal(Lnorm.shape[1])
                        delta = Lnorm @ zz
                    else:
                        delta = rng.standard_normal(n)
                    delta -= delta.mean()
                    ds = math.fsum(delta.tolist())
                    if not math.isfinite(ds):
                        continue
                    delta[-1] -= ds
                    y = cur + sg * delta
                    if not np.all(np.isfinite(y)) or np.min(y) < 0 or np.max(y) > 1:
                        continue
                    if abs(math.fsum(y.tolist()) - n / 2.0) > 1e-9:
                        continue
                    ys = exact_score(y, n)
                    if ys is None:
                        continue
                    d = ys - cur_sc
                    if d <= 0 or rng.random() < math.exp(-min(d / T, 700.0)):
                        cur = y.copy()
                        cur_sc = ys
                        if best_sc is None or cur_sc < best_sc - 1e-12:
                            best_h = cur.copy()
                            best_sc = cur_sc
        except Exception:
            continue
    if best_h is None:
        best_h = parent
    best_h = project_capped_simplex(best_h)
    if best_h is None or not check_feas(best_h, n):
        best_h = project_capped_simplex(parent)
    if best_h is None or not check_feas(best_h, n):
        best_h = np.full(n, 0.5, dtype=np.float64)
    fs = exact_score(best_h, n)
    if fs is None:
        fs = 1.0
    return best_h.astype(np.float64).copy(), float(fs), int(n)
