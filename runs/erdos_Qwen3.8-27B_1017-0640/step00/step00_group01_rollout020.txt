        scores = np.array([eval_envelope(a, b, c, np.array([x])) for x in cands])
        order = np.argsort(scores)[:20]
        cands = [cands[k] for k in sorted(order)]
        for x in (lo, hi, 0.0 if lo <= 0.0 <= hi else lo):
            if x not in cands: cands.append(x)
    return cands

def pair_line_search(h, i, j, n):
    cur_s = exact_score(h, n)
    cc = build_coeffs(h, i, j, n)
    if cc is None:
        return h.copy(), cur_s
    a, b, c, lo, hi = cc
    if hi - lo < 1e-14:
        return h.copy(), cur_s
    cands = line_candidates(a, b, c, lo, hi, n)
    best_h, best_s = h.copy(), cur_s
    for t in cands:
        if t < lo - 1e-12 or t > hi + 1e-12:
            continue
        t = max(lo, min(hi, t))
        prop = h.copy()
        prop[i] = h[i] + t
        prop[j] = h[j] - t
        proj = project_capped_simplex(prop, n)
        if proj is None or not is_feasible(proj, n):
            continue
        try:
            sc = exact_score(proj, n)
        except Exception:
            continue
        if math.isfinite(sc) and sc < best_s - 1e-12:
            best_h, best_s = proj.copy(), sc
    return best_h, best_s

def local_search(h, n, rng, n_passes, n_pairs):
    best_h, best_s = h.copy(), exact_score(h, n)
    nm1 = n - 1
    for _ in range(n_passes):
        improved = False
        if n <= 128:
            pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
            rng.shuffle(pairs)
            pairs = pairs[:min(n_pairs, len(pairs))]
        else:
            pairs = []
            for _ in range(min(n_pairs, 200)):
                x, y = rng.integers(0, n, 2)
                if x != y and (x, y) not in pairs:
                    pairs.append((min(x,y), max(x,y)))
                if len(pairs) >= n_pairs: break
        for (pi, pj) in pairs:
            nh, ns = pair_line_search(best_h, pi, pj, n)
            if ns < best_s - 1e-12:
                best_h, best_s = nh, ns
                improved = True
        if not improved:
            break
    return best_h, best_s

def build_starts(n, rng):
    starts = []
    def add(v):
        v = np.asarray(v, dtype=np.float64).copy()
        p = project_capped_simplex(v, n)
        if p is None or not is_feasible(p, n):
            return
        for ex in starts:
            if np.max(np.abs(p - ex)) <= 1e-9:
                return
        starts.append(p)
    add(np.full(n, 0.5))
    xs = np.arange(n) + 0.5
    for f in (1, 2, 3):
        phase = rng.uniform(0, 2 * math.pi)
        add(0.5 + 0.3 * np.sin(2 * math.pi * f * xs / n + phase))
    add(np.full(n, 0.5) + 0.2 * rng.standard_normal(n))
    iv = globals().get('initial_h_values')
    if iv is not None:
        try:
            pv = np.asarray(iv, dtype=np.float64).copy()
            if pv.ndim == 1 and 2 <= pv.shape[0] < 1000 and np.all(np.isfinite(pv)) \
               and np.all(pv >= -1e-7) and np.all(pv <= 1.0 + 1e-7):
                pn = pv.shape[0]
                pp = project_capped_simplex(pv, pn)
                if pp is not None and is_feasible(pp, pn) and pn == n:
                    adds = [pp, pp[::-1].copy(), 1.0 - pp,
                            pp + 0.12 * rng.standard_normal(n)]
                    for av in adds:
                        add(av)
        except Exception:
            pass
    return starts

def run(seed=42, **kwargs):
    try:
        seed = int(seed) & 0xFFFFFFFF
    except Exception:
        seed = 42
    rng = np.random.default_rng(seed)
    iv = globals().get('initial_h_values')
    n, parent = 47, None
    if iv is not None:
        try:
            pv = np.asarray(iv, dtype=np.float64).copy()
            if pv.ndim == 1 and 2 <= pv.shape[0] < 1000 and np.all(np.isfinite(pv)) \
               and np.all(pv >= -1e-7) and np.all(pv <= 1.0 + 1e-7):
                n = pv.shape[0]
                parent = project_capped_simplex(pv, n)
                if parent is None or not is_feasible(parent, n):
                    parent = None
        except Exception:
            parent = None
    if parent is None:
        n = 47
        parent = np.full(n, 0.5)
    best_h = parent.copy()
    best_s = exact_score(best_h, n)
    starts = build_starts(n, rng)
    # Cap starts
    cap = 7 if n <= 64 else (5 if n <= 128 else 3)
    starts = starts[:cap]
    npass = 3 if n <= 64 else (2 if n <= 128 else 2)
    npairs = 20 if n <= 64 else (12 if n <= 128 else 6)
    for st in starts:
        try:
            sc = exact_score(st, n)
            if math.isfinite(sc) and sc < best_s - 1e-12:
                best_h, best_s = st.copy(), sc
            lh, ls = local_search(st, n, rng, npass, npairs)
            if ls < best_s - 1e-12:
                best_h, best_s = lh, ls
        except Exception:
            pass
    # Adaptive restarts
    if n <= 64:
        for scale in (0.06, 0.16):
            try:
                rv = best_h + scale * rng.standard_normal(n)
                rp = project_capped_simplex(rv, n)
                if rp is None or not is_feasible(rp, n):
                    continue
                ls = exact_score(rp, n)
                rh, rs = local_search(rp, n, rng, 2, npairs)
                if rs < best_s - 1e-12:
                    best_h, best_s = rh, rs
            except Exception:
                pass
    elif n <= 128:
        try:
            rv = best_h + 0.08 * rng.standard_normal(n)
            rp = project_capped_simplex(rv, n)
            if rp is not None and is_feasible(rp, n):
                rh, rs = local_search(rp, n, rng, 2, 12)
                if rs < best_s - 1e-12:
                    best_h, best_s = rh, rs
        except Exception:
            pass
    # Final validation
    fin = project_capped_simplex(best_h, n)
    if fin is None or not is_feasible(fin, n):
        fin = parent if (parent is not None and is_feasible(parent, n)) else np.full(n, 0.5)
        fin = project_capped_simplex(fin, n)
        if fin is None:
            fin = np.full(n, 0.5)
    try:
        final_s = exact_score(fin, n)
    except Exception:
        final_s = 1.0
    if not math.isfinite(final_s):
        fin = np.full(n, 0.5)
        final_s = exact_score(fin, n)
    return fin.astype(np.float64), float(final_s), int(n)
