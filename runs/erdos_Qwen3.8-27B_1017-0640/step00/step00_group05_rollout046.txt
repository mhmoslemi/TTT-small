            proj = _project(mutant_raw, N, m)
            if proj is None:
                continue
            mask = rng.random(N) < CR
            mask[rng.integers(N)] = True
            trial = np.where(mask, proj, pop_arr[i])
            trial = _project(trial, N, m)
            if trial is None:
                continue
            ts = _score(trial, N)
            if ts is None:
                continue
            if ts < ps_arr[i]:
                new_pop[i] = trial
                new_ps[i] = ts
            if ts < best_ref[1]:
                best_ref[0] = trial.copy()
                best_ref[1] = ts
                improved = True

        pop_arr, ps_arr = new_pop, new_ps
        stall = 0 if improved else stall + 1

        if stall >= stall_limit and restarts < 2 and g < G - 1:
            restarts += 1
            stall = 0
            sd = 0.12 if restarts == 1 else 0.24
            pop_arr[0] = best_ref[0]
            ps_arr[0] = best_ref[1]
            for i in range(1, P):
                if i % 2 == 0:
                    pop_arr[i] = best_ref[0] + sd * rng.standard_normal(N)
                else:
                    pop_arr[i] = rng.uniform(0, 1, N)
                pp = _project(pop_arr[i], N, m)
                pop_arr[i] = pp if pp is not None else neutral
                s = _score(pop_arr[i], N)
                ps_arr[i] = s if s is not None else 1.0
                if s is not None and s < best_ref[1]:
                    best_ref[0] = pop_arr[i].copy()
                    best_ref[1] = s

    final = best_ref[0].copy()
    fp = _project(final, N, m)
    if fp is not None:
        final = fp
    fs = _score(final, N)
    if fs is None:
        final = neutral.copy()
        fs = _score(final, N)
    if fs is None:
        fs = 0.5
    return final, fs, N
