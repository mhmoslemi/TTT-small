def is_feasible(h, n):
    if h.ndim != 1 or len(h) != n:
        return False
    if not np.all(np.isfinite(h)):

def project_capped(y, n):
    if y.ndim != 1 or len(y) != n:
        return np.ones(n) * 0.5
    if not np.all(np.isfinite(y)):
        return np.ones(n) * 0.5
    if np.any(np.abs(y) > 1e12):
        return np.ones(n) * 0.5
    m = n / 2.0
    lo = np.min(y) - 1
    hi = np.max(y)
    for _ in range(80):
        mid = (lo + hi) / 2
        p = np.clip(y - mid, 0, 1)
        if np.sum(p) > m:
            lo = mid
        else:
            hi = mid
    p = np.clip(y - (lo + hi) / 2, 0, 1)
    # Fix residual
    for _ in range(2):
        residual = m - np.sum(p)
        if abs(residual) <= 2e-14 * max(1, n):
            break
        if residual > 0:
            for i in range(n):
                if p[i] < 1.0:
                    add = min(residual, 1.0 - p[i])
                    p[i] += add
                    residual -= add
                    if abs(residual) < 1e-15:
                        break
        else:
            for i in range(n):
                if p[i] > 0.0:
                    sub = min(-residual, p[i])
                    p[i] -= sub
                    residual += sub
                    if abs(residual) < 1e-15:
                        break
    if not is_feasible(p, n):
        return np.ones(n) * 0.5
    return p

def exact_score(h, n):
    if h.ndim != 1 or len(h) != n:

    if not np.all(np.isfinite(h)):

    try:

    if n % 2 == 1:
        bv[idx[n // 2]] = 0.5
    lst.append(project_capped(bv, n))
    uniq = []
    for s in lst:
        if all(np.max(np.abs(s - u)) >= 1e-10 for u in uniq):
            uniq.append(s)
        if len(uniq) >= ns:
            break
    return uniq

def run(seed=42, **kwargs):
    parent = _get_parent()
    n = int(parent.shape[0]) if parent is not None else 80
    try: n_starts = max(1, min(6, int(kwargs.get('n_starts', 5))))
    except: n_starts = 5
    try: outer_it = max(1, min(16, int(kwargs.get('outer_iterations', 8))))
    except: outer_it = 8
    try: maxit = max(40, min(250, int(kwargs.get('maxiter', 100))))
    except: maxit = 100
    try:
        ft = float(kwargs.get('ftol', 1e-9))
        if not np.isfinite(ft) or ft <= 0: ft = 1e-9
        ft = max(1e-12, min(1e-6, ft))
    except: ft = 1e-9
    try:
        sd = int(seed)
        if sd < 0: sd = 42
    except: sd = 42
    rng = np.random.default_rng(sd)
    unif = np.ones(n) * 0.5
    base = project_capped(parent.copy(), n) if parent is not None else unif.copy()
    fallback = base.copy()
    starts = _make_starts(base, n, rng, n_starts)
    best_h = None
    best_sc = float('inf')
    for s in starts:
        sc = exact_score(s, n)
        if sc < best_sc:
            best_sc = sc; best_h = s.copy()
    fsc = exact_score(fallback, n)
    if fsc < best_sc:
        best_sc = fsc; best_h = fallback.copy()
    if n <= 160:
        for st in starts:
            if best_sc <= 0.3808:
                break
            cur = st.copy()
            csc = exact_score(cur, n)
            for _ in range(outer_it):
                if best_sc <= 0.3808:
                    break
                r = cur.copy()
                Qr, _ = build_Q_G(r, r, n)
                if not np.all(np.isfinite(Qr)):
                    break
                t0 = float(np.max(Qr)) + 1e-8
                z0 = np.concatenate([r, [t0]])
                bnds = [(0.0, 1.0)] * n + [(0.0, None)]
                ceq = {'type': 'eq', 'fun': _eq_f, 'jac': _eq_j}
                cin = {'type': 'ineq', 'fun': _ineq_f, 'jac': _ineq_j}
                try:
                    res = minimize(_obj, z0, method='SLSQP', jac=_obj_j,
                                   bounds=bnds, constraints=[ceq, cin],
                                   args=(n, r),
                                   options={'maxiter': maxit, 'ftol': ft, 'disp': False})
                except Exception:
                    break
                if res is None or not hasattr(res, 'x') or res.x is None:
                    break
                rx = np.asarray(res.x, dtype=np.float64)
                if rx.shape[0] != n + 1 or not np.all(np.isfinite(rx)):
                    break
                cands = [project_capped(rx[:n], n)]
                for al in (0.5, 0.25, 0.125):
                    cands.append(project_capped(r + al * (rx[:n] - r), n))
                bc = None; bcs = csc
                for c in cands:
                    if not is_feasible(c, n):
                        continue
                    sc = exact_score(c, n)
                    if sc < bcs - 1e-12:
                        bcs = sc; bc = c
                if bc is None:
                    break
                cur = bc.copy(); csc = bcs
                if csc < best_sc - 1e-12:
                    best_sc = csc; best_h = cur.copy()
    fin = project_capped(best_h, n) if best_h is not None else unif
    fb = project_capped(fallback, n)
    fs1 = exact_score(fin, n)
    fs2 = exact_score(fb, n)
    if fs2 < fs1:
        fin = fb; fs1 = fs2
    if not is_feasible(fin, n):
        fin = unif
    c5 = float(np.max(np.correlate(fin, 1.0 - fin, mode='full'))) * (2.0 / n)
    return (fin, c5, n)
