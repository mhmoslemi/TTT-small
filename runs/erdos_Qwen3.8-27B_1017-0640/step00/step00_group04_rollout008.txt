            if abs(res) <= tol:
                break
            adj = min(-res, x[i])
            x[i] -= adj; res += adj
    return x

def feasible(h):
    n = len(h)
    return (np.all(np.isfinite(h)) and np.all(h >= -1e-10) and np.all(h <= 1.0 + 1e-10)
            and abs(h.sum() - 0.5 * n) <= 1e-10 * max(1, n))

def _ineq_fun(z):
    n = len(z) - 1
    h, t, dx = z[:n], z[-1], 2.0 / n
    out = np.empty(2 * n - 1)
    for idx in range(2 * n - 1):
        ell = idx - (n - 1)
        jlo, jhi = max(0, ell), min(n, n + ell)
        out[idx] = t - dx * np.sum(h[jlo:jhi] * (1.0 - h[jlo - ell:jhi - ell]))
    return out

def _ineq_jac(z):
    n = len(z) - 1
    h, dx = z[:n], 2.0 / n
    J = np.zeros((2 * n - 1, n + 1))
    J[:, -1] = 1.0
    for idx in range(2 * n - 1):
        ell = idx - (n - 1)
        r1lo, r1hi = max(0, ell), min(n, n + ell)
        if r1lo < r1hi:
            rs = np.arange(r1lo, r1hi)
            J[idx, rs] += (1.0 - h[rs - ell])
        r2lo, r2hi = max(0, -ell), min(n, n - ell)
        if r2lo < r2hi:
            rs = np.arange(r2lo, r2hi)
            J[idx, rs] -= h[rs + ell]
    J[:, :-1] *= -dx
    return J

def _eq_fun(z):
    return np.sum(z[:-1]) - 0.5 * (len(z) - 1)

def _eq_jac(z):
    n = len(z) - 1
    v = np.zeros(n + 1)
    v[:n] = 1.0
    return v

def _obj_fun(z):
    return z[-1]

def _obj_grad(z):
    v = np.zeros(len(z))
    v[-1] = 1.0
    return v

def try_solve(h0, n, maxiter):
    dx = 2.0 / n
    s0 = exact_score(h0)
    z0 = np.concatenate([h0, [min(1.0, s0 + 1e-8)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [
        {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac},
        {'type': 'ineq', 'fun': _ineq_fun, 'jac': _ineq_jac},
    ]
    try:
        res = minimize(_obj_fun, z0, jac=_obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-10, 'disp': False})
        if res.x is not None and len(res.x) == n + 1 and np.all(np.isfinite(res.x)):
            hc = project_simplex(res.x[:n], n)
            if hc is not None and feasible(hc):
                return hc
    except Exception:
        pass
    return None

def make_starts(n, base, rng, count, scales):
    starts = []
    def add(h):
        p = project_simplex(h, n)
        if p is not None and feasible(p):
            for existing in starts:
                if np.max(np.abs(p - existing)) < 1e-10:
                    return
            starts.append(p)
    if base is not None and feasible(base):
        add(base.copy())
    add(base[::-1].copy())
    hblk = np.zeros(n)
    hblk[:n // 2] = 1.0
    if n % 2 == 1:
        hblk[n // 2] = 0.5
    add(hblk.copy())
    add(np.full(n, 0.5))
    i = 0
    while len(starts) < count:
        sc = scales[i % len(scales)]
        if i % 2 == 0 and base is not None:
            cand = base + rng.standard_normal(n) * sc
        else:
            cand = rng.uniform(-0.25, 1.25, size=n)
        add(cand)
        i += 1
    return starts

def run(seed=42, **kwargs):
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    rng = np.random.default_rng(seed)
    n_starts = 8
    maxiter = 250
    try:
        ns = int(kwargs.get('n_starts', 8))
        n_starts = max(4, min(10, ns))
    except Exception:
        pass
    try:
        mi = int(kwargs.get('maxiter', 250))
        maxiter = max(100, min(400, mi))
    except Exception:
        pass
    scales = [0.08, 0.20, 0.40]
    # Get parent
    parent = globals().get('initial_h_values', None)
    parent_valid = None
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64).ravel()
        if p.size >= 2 and np.all(np.isfinite(p)):
            if feasible(p):
                parent_valid = p.copy()
            else:
                pp = project_simplex(p, p.size)
                if pp is not None and feasible(pp):
                    parent_valid = pp
    n0 = parent_valid.size if parent_valid is not None else 42
    # Fallback
    if parent_valid is None:
        fb = np.zeros(n0)
        fb[:n0 // 2] = 1.0
        if n0 % 2 == 1:
            fb[n0 // 2] = 0.5
        parent_valid = fb
    best_h = parent_valid.copy()
    best_score = exact_score(parent_valid)
    # Optimize at n0
    if n0 <= 256:
        anchor = parent_valid
        starts = make_starts(n0, anchor, rng, n_starts, scales)
        for s in starts:
            sc = exact_score(s)
            if sc < best_score - 1e-12:
                best_score = sc
                best_h = s.copy()
            h_new = try_solve(s, n0, maxiter)
            if h_new is not None:
                sc2 = exact_score(h_new)
                if sc2 < best_score - 1e-12:
                    best_score = sc2
                    best_h = h_new.copy()
    # Try 2*n0
    n1 = 2 * n0
    if n0 <= 128 and n1 <= 256:
        anchor1 = np.repeat(best_h, 2)
        if parent_valid is not None:
            parent1 = np.repeat(parent_valid, 2)
        else:
            parent1 = anchor1
        starts1 = make_starts(n1, anchor1, rng, n_starts, scales)
        # Also add parent1 explicitly
        p1 = project_simplex(parent1, n1)
        if p1 is not None and feasible(p1) and not any(np.max(np.abs(p1 - s)) < 1e-10 for s in starts1):
            starts1.append(p1)
        for s in starts1:
            sc = exact_score(s)
            if sc < best_score - 1e-12:
                best_score = sc
                best_h = s.copy()
            h_new = try_solve(s, n1, maxiter)
            if h_new is not None:
                sc2 = exact_score(h_new)
                if sc2 < best_score - 1e-12:
                    best_score = sc2
                    best_h = h_new.copy()
    # Final validation
    nf = len(best_h)
    hf = project_simplex(best_h, nf)
    if hf is not None and feasible(hf):
        best_h = hf
    else:
        best_h = parent_valid.copy()
    best_score = exact_score(best_h)
    return (best_h, best_score, len(best_h))
