        proj = project_capped(final, n, m)
        if proj is not None and check_feasible(proj, n, m):
            final = proj
        else:
            final = protected
    final = np.clip(np.asarray(final, dtype=np.float64), 0.0, 1.0)
    c5 = exact_score(final, n)
    if c5 is None:
        final = np.clip(protected, 0.0, 1.0)
        c5 = exact_score(final, n)
    return (final, float(c5), int(len(final)))
