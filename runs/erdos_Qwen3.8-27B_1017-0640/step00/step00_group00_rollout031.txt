            z -= z.mean()
            rms = math.sqrt(float(np.mean(z * z)))
            if rms > 0:
                z = z / rms
            trial = project_capped_simplex(best_h + scale * z, n)
            if trial is None:
                continue
            sc = exact_score(trial, n)
            if sc is not None and sc < best_s - 1e-12:
                best_s = sc
                best_h = trial
                improved = True
        if improved:
            for _ in range(pcnt):
                z = rng.standard_normal(n)
                z -= z.mean()
                rms = math.sqrt(float(np.mean(z * z)))
                if rms > 0:
                    z = z / rms
                trial = project_capped_simplex(best_h + scale * z, n)
                if trial is None:
                    continue
                sc = exact_score(trial, n)
                if sc is not None and sc < best_s - 1e-12:
                    best_s = sc
                    best_h = trial

    if not is_feasible(best_h, n):
        rep = project_capped_simplex(best_h, n)
        if rep is not None:
            best_h = rep
        else:
            best_h = fallback.copy()
    best_s = exact_score(best_h, n)
    if best_s is None:
        best_h = np.full(n, 0.5)
        best_s = exact_score(best_h, n)
    return (best_h, best_s, int(n))
