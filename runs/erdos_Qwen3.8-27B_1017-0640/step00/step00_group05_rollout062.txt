import numpy as np
from scipy.optimize import minimize

def compute_q(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def score_h(h, dx):
    q = compute_q(h, dx)
    return float(np.max(q))

def project_box_sum(v, m, N):
    v = np.clip(v.astype(np.float64), -1e6, 1e6)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > m:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(3):
        r = m - np.sum(y)
        if abs(r) < 1e-13:
            break
        if r > 0:
            for i in range(N):
                s = 1.0 - y[i]
                if s > 0:
                    u = min(r, s)
                    y[i] += u
                    r -= u
                    if abs(r) < 1e-13:
                        break
        else:
            for i in range(N):
                if y[i] > 0:
                    u = min(-r, y[i])
                    y[i] -= u
                    r += u
                    if abs(r) < 1e-13:
                        break
    return y

    return float(np.sum(z[:N]) - m)

    return g

            j = -k
            row[:N - j] = 1.0 - h[j:]
            row[j:] -= h[:N - j]
        full_J[k_idx, :N] = -row * dx
    return full_J
