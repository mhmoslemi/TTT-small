    except (NameError, AttributeError, ValueError):
        pass
    if base is None:
        try:
            np_ = int(kwargs.get('n_points', 80))
            if 2 <= np_ <= 999:
                n = np_
        except (ValueError, TypeError):
            pass
        base = project_capped(np.full(n, 0.5), n)
    try:
        n_starts = max(1, min(8, int(kwargs.get('n_starts', 7))))
    except (ValueError, TypeError):
        n_starts = 7
    try:
        max_iter = max(20, min(200, int(kwargs.get('max_iter', 80))))
    except (ValueError, TypeError):
        max_iter = 80
    try:
        s = int(seed)
        seed = abs(s) if s != 0 else 42
    except (ValueError, TypeError):
        seed = 42
    rng = np.random.default_rng(seed)
    if not check_feasible(base, n):
        base = project_capped(np.full(n, 0.5), n)
    best = base.copy()
    best_score = exact_score(best, n)
    all_starts = make_starts(base, n, rng)
    use_starts = all_starts[:n_starts]
    temps = [0.02, 0.005, 0.00125, 0.0003125, 0.000078125, 0.00001953125, 0.0000048828125]
    if n <= 256:
        for start in use_starts:
            if not check_feasible(start, n):
                start = project_capped(start, n)
            x = optimize_start(start.copy(), n, max_iter, temps)
            if check_feasible(x, n):
                sc = exact_score(x, n)
                if sc < best_score - 1e-12:
                    best = x.copy()
                    best_score = sc
            if best_score <= 0.3808:
                break
    best = project_capped(best, n)
    if not check_feasible(best, n):
        fb = project_capped(base, n)
        if not check_feasible(fb, n):
            fb = np.full(n, 0.5)
        best = fb
    c5 = exact_score(best, n)
    if not np.isfinite(c5):
        best = np.full(n, 0.5)
        c5 = exact_score(best, n)
    return (best, float(c5), int(n))
