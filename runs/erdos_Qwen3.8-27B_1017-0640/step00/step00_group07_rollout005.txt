            if len(idx) == 0:
                break
            slack = h[idx]
            j = idx[np.argmax(slack)]
            amt = min(-resid, slack[np.argmax(slack)])
            h[j] -= amt
            resid += amt
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(h.sum() - m) > 1e-10 * max(1, n):
        return np.full(n, 0.5)
    return h

def exact_score(h):
    n = len(h)
    if n < 2 or not np.all(np.isfinite(h)):
        return np.inf
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def compute_q_and_grad(h, n):
    dx = 2.0 / n
    m = 2 * n - 1
    q = np.zeros(m)
    G = np.zeros((m, n))
    q[0] = dx * np.dot(h, 1.0 - h)
    G[0] = dx * (1.0 - 2.0 * h)
    for s in range(1, n):
        ia = s
        q[ia] = dx * np.dot(h[:n - s], 1.0 - h[s:])
        row = np.zeros(n)
        row[:n - s] += (1.0 - h[s:])
        row[s:] -= h[:n - s]
        G[ia] = dx * row
        ib = s + n - 1
        q[ib] = dx * np.dot(h[s:], 1.0 - h[:n - s])
        row2 = np.zeros(n)
        row2[s:] += (1.0 - h[:n - s])
        row2[:n - s] -= h[s:]
        G[ib] = dx * row2
    return q, G

def _obj(z, n):
    return z[n]

def _obj_grad(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def _eq(z, n):
    return z[:n].sum() - n / 2.0

def _eq_jac(z, n):
    J = np.zeros((1, n + 1))
    J[0, :n] = 1.0
    return J

def _ineq(z, n):
    h = z[:n]
    q, _ = compute_q_and_grad(h, n)
    return z[n] - q

def _ineq_jac(z, n):
    h = z[:n]
    _, G = compute_q_and_grad(h, n)
    m = 2 * n - 1
    J = np.zeros((m, n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J

def _make_starts(base, n, seed):
    rng = np.random.default_rng(seed)
    starts = [base.copy()]
    starts.append(project_feasible(base + 0.03 * rng.standard_normal(n), n))
    starts.append(project_feasible(base + 0.12 * rng.standard_normal(n), n))
    starts.append(project_feasible(rng.uniform(0.0, 1.0, n), n))
    K = min(6, (n - 1) // 2)
    idx = np.arange(n, dtype=np.float64) + 0.5
    y = np.full(n, 0.5)
    for k in range(1, K + 1):
        a = rng.normal(0.0, 0.12 / np.sqrt(k))
        b = rng.normal(0.0, 0.12 / np.sqrt(k))
        y += a * np.cos(2.0 * np.pi * k * idx / n) + b * np.sin(2.0 * np.pi * k * idx / n)
    starts.append(project_feasible(y, n))
    perm = rng.permutation(n)
    yb = np.zeros(n)
    yb[perm[:n // 2]] = 1.0
    if n % 2 == 1:
        yb[perm[n // 2]] = 0.5
    starts.append(project_feasible(yb, n))
    return starts

def run(seed=42, **kwargs):
    n_starts = kwargs.get('n_starts', 6)
    maxiter = kwargs.get('maxiter', 250)
    ftol = kwargs.get('ftol', 1e-10)
    try:
        n_starts = max(1, min(8, int(n_starts)))
    except (ValueError, TypeError):
        n_starts = 6
    try:
        maxiter = max(50, min(400, int(maxiter)))
    except (ValueError, TypeError):
        maxiter = 250
    try:
        ftol = float(ftol)
        if not np.isfinite(ftol) or ftol <= 0:
            ftol = 1e-10
        ftol = max(1e-12, min(1e-6, ftol))
    except (ValueError, TypeError):
        ftol = 1e-10
    try:
        seed = int(seed)
    except (ValueError, TypeError):
        seed = 42

    parent = None
    parent_valid = None
    ihv = globals().get('initial_h_values', None)
    if ihv is not None:
        ihv = np.asarray(ihv, dtype=np.float64)
        if ihv.ndim == 1 and 2 <= len(ihv) <= 999 and np.all(np.isfinite(ihv)):
            n = len(ihv)
            parent = ihv.copy()
            if (np.all(parent >= -1e-12) and np.all(parent <= 1.0 + 1e-12)
                    and abs(parent.sum() - n / 2.0) <= 1e-8 * max(1, n)):
                parent_valid = np.clip(parent, 0.0, 1.0)
        else:
            n = 80
    else:
        n = 80

    if parent is None:
        parent = np.full(n, 0.5)
        parent_valid = parent.copy()

    base = project_feasible(parent, n)
    best_h = base.copy()
    best_score = exact_score(best_h)
    if not np.isfinite(best_score):
        best_h = np.full(n, 0.5)
        best_score = exact_score(best_h)

    if best_score <= 0.3808 or n > 256:
        return (best_h, float(best_score), int(n))

    starts = _make_starts(base, n, seed)
    if n_starts < len(starts):
        starts = starts[:n_starts]

    obj_f = partial(_obj, n=n)
    obj_g = partial(_obj_grad, n=n)
    eq_c = partial(_eq, n=n)
    eq_j = partial(_eq_jac, n=n)
    ineq_c = partial(_ineq, n=n)
    ineq_j = partial(_ineq_jac, n=n)
    constraints = [
        {'type': 'eq', 'fun': eq_c, 'jac': eq_j},
        {'type': 'ineq', 'fun': ineq_c, 'jac': ineq_j}
    ]
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]

    for h0 in starts:
        q0, _ = compute_q_and_grad(h0, n)
        t0 = max(0.0, float(np.max(q0))) + 1e-8
        z0 = np.concatenate([h0, [t0]])
        try:
            res = minimize(obj_f, z0, jac=obj_g, bounds=bounds,
                           constraints=constraints, method='SLSQP',
                           options={'ftol': ftol, 'maxiter': maxiter, 'disp': False})
            if res.x is not None and len(res.x) == n + 1:
                h_cand = project_feasible(res.x[:n], n)
                if np.all(np.isfinite(h_cand)):
                    s_cand = exact_score(h_cand)
                    if np.isfinite(s_cand) and s_cand < best_score - 1e-12:
                        best_h = h_cand
                        best_score = s_cand
                        if best_score <= 0.3808:
                            break
        except Exception:
            pass

    best_h = project_feasible(best_h, n)
    if (not np.all(np.isfinite(best_h)) or np.any(best_h < -1e-12)
            or np.any(best_h > 1.0 + 1e-12)
            or abs(best_h.sum() - n / 2.0) > 1e-10 * max(1, n)):
        if parent_valid is not None:
            best_h = project_feasible(parent_valid, n)
        else:
            best_h = np.full(n, 0.5)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_score = exact_score(best_h)
    return (best_h, float(best_score), int(n))
