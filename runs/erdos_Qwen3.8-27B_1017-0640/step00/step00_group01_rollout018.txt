```python
import numpy as np
import math

def project_capped_simplex(v):
    n = len(v)
    if n < 2:
        return None
    s = n / 2.0
    v = np.asarray(v, dtype=np.float64)
    if not np.all(np.isfinite(v)) or np.max(np.abs(v)) > 1e6:
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = (lo + hi) / 2.0
        mass = math.fsum(np.clip(v - mid, 0.0, 1.0).tolist())
        if mass > s:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(3):
        r = s - math.fsum(x.tolist())
        if abs(r) <= 5e-15:
            break
        if r > 0:
            for i in range(n):
                adj = min(r, 1.0 - x[i])
                x[i] += adj
                r -= adj
        else:
            for i in range(n):
                adj = min(-r, x[i])
                x[i] -= adj
                r += adj
    if not np.all(np.isfinite(x)) or np.min(x) < -1e-12 or np.max(x) > 1.0 + 1e-12:
        return None
    if abs(math.fsum(x.tolist()) - s) > 1e-9:
        return None
    return x

def exact_score(h, n):
    try:
        h = np.asarray(h, dtype=np.float64)
        if h.shape != (n,) or not np.all(np.isfinite(h)):
            return None
        if np.min(h) < -1e-12 or np.max(h) > 1.0 + 1e-12:
            return None
        if abs(math.fsum(h.tolist()) - n / 2.0) > 1e-9:
            return None
        corr = np.correlate(h, 1.0 - h, mode='full')
        val = float(np.max(corr) * (2.0 / n))
        return val if np.isfinite(val) else None
    except Exception:
        return None

def smooth_value_grad(h, tau, n, need_grad=True):
    dx = 2.0 / n
    a = np.zeros(2 * n - 1, dtype=np.float64)
    for k in range(2 * n - 1):
        d = k - (n - 1)
        if d >= 0:
            a[k] = np.dot(h[d:n], 1.0 - h[0:n - d]) * dx
        else:
            m = -d
            a[k] = np.dot(h[0:n - m], 1.0 - h[m:n]) * dx
    if not np.all(np.isfinite(a)):
        return None, None
    A = float(np.max(a))
    shifted = (a - A) / tau
    e = np.exp(shifted)
    Z = float(np.sum(e))
    if Z <= 0 or not np.isfinite(Z):
        return None, None
    F = A + tau * math.log(Z)
    if not np.isfinite(F):
        return None, None
    if not need_grad:
        return F, None
    p = e / Z
    g = np.zeros(n, dtype=np.float64)
    for k in range(2 * n - 1):
        d = k - (n - 1)
        w = dx * p[k]
        if d >= 0:
            g[d:n] += w * (1.0 - h[0:n - d])
            g[0:n - d] -= w * h[d:n]
        else:
            m = -d
            g[0:n - m] += w * (1.0 - h[m:n])
            g[m:n] -= w * h[0:n - m]
    if not np.all(np.isfinite(g)):
        return None, None
    return F, g
```

```python
import numpy as np
import math

def project_capped_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n < 2 or not np.all(np.isfinite(v)) or np.max(np.abs(v)) > 1e6:
        return None
    s = n / 2.0
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        mass = math.fsum(np.clip(v - mid, 0.0, 1.0).tolist())
        if mass > s:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(3):
        r = s - math.fsum(x.tolist())
        if abs(r) <= 5e-15:
            break
        if r > 0:
            for i in range(n):
                adj = min(r, 1.0 - x[i])
                x[i] += adj
                r -= adj
        else:
            for i in range(n):
                adj = min(-r, x[i])
                x[i] -= adj
                r += adj
    if not np.all(np.isfinite(x)) or np.min(x) < -1e-12 or np.max(x) > 1.0 + 1e-12:
        return None
    if abs(math.fsum(x.tolist()) - s) > 1e-9:
        return None
    return x

def exact_score(h, n):
    try:
        h = np.asarray(h, dtype=np.float64)
        if h.shape != (n,) or not np.all(np.isfinite(h)):
            return None
        if np.min(h) < -1e-12 or np.max(h) > 1.0 + 1e-12:
            return None
        if abs(math.fsum(h.tolist()) - n / 2.0) > 1e-9:
            return None
        corr = np.correlate(h, 1.0 - h, mode='full')
        val = float(np.max(corr) * (2.0 / n))
        return val if np.isfinite(val) else None
    except Exception:
        return None

def smooth_value_grad(h, tau, n, need_grad):
    dx = 2.0 / n
    a = np.correlate(h, 1.0 - h, mode='full') * dx
    if not np.all(np.isfinite(a)):
        return None, None
    A = float(np.max(a))
    e = np.exp((a - A) / tau)
    Z = float(np.sum(e))
    if Z <= 0 or not np.isfinite(Z):
        return None, None
    F = A + tau * math.log(Z)
    if not np.isfinite(F):
        return None, None
    if not need_grad:
        return F, None
    p = e / Z
    g = np.zeros(n, dtype=np.float64)
    for k in range(2 * n - 1):
        d = k - (n - 1)
        w = dx * p[k]
        if d >= 0:
            g[d:n] += w * (1.0 - h[0:n - d])
            g[0:n - d] -= w * h[d:n]
        else:
            m = -d
            g[0:n - m] += w * (1.0 - h[m:n])
            g[m:n] -= w * h[0:n - m]
    if not np.all(np.isfinite(g)):
        return None, None
    return F, g

def projected_search(h0, temps, iters_per_temp, n):
    current = h0.copy()
    res = exact_score(current, n)
    if res is None:
        return current
    best = current.copy()
    best_score = res
    for tau in temps:
        F_cur, g = smooth_value_grad(current, tau, n, True)
        if F_cur is None:
            break
        eta = 1.0
        for _ in range(iters_per_temp):
            accepted = False
            for _bt in range(25):
                y = project_capped_simplex(current - eta * g)
                if y is None:
                    eta *= 0.5
                    continue
                s_y = exact_score(y, n)
                if s_y is not None and s_y < best_score - 1e-12:
                    best_score = s_y
                    best = y.copy()
                delta = y - current
                d2 = float(np.dot(delta, delta))
                if d2 <= 1e-24:
                    accepted = True
                    break
                gd = float(np.dot(g, delta))
                F_y, _ = smooth_value_grad(y, tau, n, False)
                if F_y is None:
                    eta *= 0.5
                    continue
                tol = 1e-4 * min(gd, 0.0) + 1e-14 * max(1.0, abs(F_cur))
                if F_y <= F_cur + tol:
                    current = y
                    F_cur = F_y
                    eta = min(1.5 * eta, 10.0)
                    accepted = True
                    break
                eta *= 0.5
            if not accepted:
                break
            if np.max(np.abs(current - best)) < 1e-9 and F_cur - best_score < 1e-10:
                break
            F_cur, g = smooth_value_grad(current, tau, n, True)
            if F_cur is None:
                break
    return best

def make_starts(n, rng, parent):
    starts = []
    if parent is not None:
        starts.append(parent)
    const = np.full(n, 0.5, dtype=np.float64)
    starts.append(const)
    base = parent if parent is not None else const
    for freq in (1, 2):
        phi = rng.uniform(0.0, 2.0 * math.pi)
        idx = np.arange(n, dtype=np.float64)
        v = 0.5 + 0.30 * np.sin(2.0 * math.pi * freq * (idx + 0.5) / n + phi)
        p = project_capped_simplex(v)
        if p is not None:
            starts.append(p)
    for scale in (0.06, 0.18):
        v = base + scale * rng.standard_normal(n)
        p = project_capped_simplex(v)
        if p is not None:
            starts.append(p)
    for _ in range(2):
        v = rng.uniform(0.0, 1.0, size=n)
        p = project_capped_simplex(v)
        if p is not None:
            starts.append(p)
    if parent is None:
        phi = rng.uniform(0.0, 2.0 * math.pi)
        idx = np.arange(n, dtype=np.float64)
        v = 0.5 + 0.30 * np.sin(2.0 * math.pi * 3.0 * (idx + 0.5) / n + phi)
        p = project_capped_simplex(v)
        if p is not None:
            starts.append(p)
    filt = [starts[0]]
    for s in starts[1:]:
        if all(np.max(np.abs(s - f)) > 1e-9 for f in filt):
            filt.append(s)
    return filt

def run(seed=42, **kwargs):
    try:
        s_val = int(seed)
        if s_val < 0 or s_val > 2**32 - 1:
            s_val = 42
    except Exception:
        s_val = 42
    rng = np.random.default_rng(s_val)
    parent = None
    n = 47
    raw = globals().get('initial_h_values', None)
    if raw is not None:
        try:
            arr = np.array(raw, dtype=np.float64)
            if arr.ndim == 1 and 2 <= arr.size < 1000:
                if np.all(np.isfinite(arr)) and np.min(arr) >= -1e-7 and np.max(arr) <= 1.0 + 1e-7:
                    n = arr.size
                    p = project_capped_simplex(arr)
                    if p is not None:
                        parent = p
        except Exception:
            pass
    const = np.full(n, 0.5, dtype=np.float64)
    best = parent if parent is not None else const
    best_score = exact_score(best, n)
    if best_score is None:
        best_score = float(np.max(np.correlate(const, 1.0 - const, mode='full')) * (2.0 / n))
        best = const.copy()
    if n <= 64:
        temps = [0.04, 0.01, 0.0025, 0.0005, 0.0001]
        itp = 80
        max_starts = 8
        restart_scales = [0.04, 0.12]
    elif n <= 128:
        temps = [0.04, 0.01, 0.0025, 0.0005]
        itp = 45
        max_starts = 5
        restart_scales = [0.06]
    elif n <= 512:
        temps = [0.03, 0.006, 0.001]
        itp = 25
        max_starts = 3
        restart_scales = []
    else:
        temps = [0.02, 0.003]
        itp = 12
        max_starts = 2
        restart_scales = []
    starts = make_starts(n, rng, parent)
    starts = starts[:max_starts]
    for st in starts:
        try:
            sc = exact_score(st, n)
            if sc is not None and sc < best_score - 1e-12:
                best_score = sc
                best = st.copy()
            res = projected_search(st, temps, itp, n)
            if res is not None:
                sc2 = exact_score(res, n)
                if sc2 is not None and sc2 < best_score - 1e-12:
                    best_score = sc2
                    best = res.copy()
        except Exception:
            pass
    for sig in restart_scales:
        try:
            v = best + sig * rng.standard_normal(n)
            p = project_capped_simplex(v)
            if p is None:
                continue
            sc = exact_score(p, n)
            if sc is not None and sc < best_score - 1e-12:
                best_score = sc
                best = p.copy()
            res = projected_search(p, temps, itp, n)
            if res is not None:
                sc2 = exact_score(res, n)
                if sc2 is not None and sc2 < best_score - 1e-12:
                    best_score = sc2
                    best = res.copy()
        except Exception:
            pass
    final = project_capped_simplex(best)
    if final is None:
        final = parent if parent is not None else const
    fs = exact_score(final, n)
    if fs is None:
        final = const
        fs = exact_score(final, n)
    if fs is None:
        fs = float(np.max(np.correlate(final, 1.0 - final, mode='full')) * (2.0 / n))
    return (final, float(fs), n)
```
