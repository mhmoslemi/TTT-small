        pass

    n = n_sel if n_sel else (len(parent) if parent is not None else 45)
    m = n / 2.0
    rng = np.random.default_rng(seed)

    starts = []
    if parent is not None:
        if len(parent) == n:
            for pv in [parent, 1.0 - parent, parent[::-1], np.roll(parent, n // 3)]:
                s = project_K(pv, m)
                if s is not None:
                    starts.append(s)
        else:
            for pv in [parent, 1.0 - parent, parent[::-1]]:
                s = interp_parent(pv, n, m)
                if s is not None:
                    starts.append(s)
        base = starts[0] if starts else 0.5 * np.ones(n)
        for sd in (0.05, 0.15):
            s = project_K(base + sd * rng.standard_normal(n), m)
            if s is not None:
                starts.append(s)

    starts.append(0.5 * np.ones(n))
    blk = make_block(n, m)
    for bv in [blk, blk[::-1]]:
        s = project_K(bv, m)
        if s is not None:
            starts.append(s)

    n_rand = 1 if parent is not None else 3
    for _ in range(n_rand):
        s = project_K(rng.random(n), m)
        if s is not None:
            starts.append(s)

    cos_ms = [1] if parent is not None else [1, 2]
    i_arr = np.arange(n)
    for rc in cos_ms:
        ph = float(rng.standard_normal())
        s = project_K(0.5 + 0.35 * np.cos(2.0 * math.pi * rc * i_arr / n + ph), m)
        if s is not None:
            starts.append(s)

    max_st = 10 if n <= 64 else (6 if n <= 160 else (4 if n <= 400 else 2))
    starts = dedup_starts(starts, max_st)

    budget = 80 if n <= 64 else (50 if n <= 160 else (24 if n <= 400 else 12))
    taus = [0.02, 0.005, 0.001, 0.0002, 0.00005, 0.00001]

    best_s = exact_score(0.5 * np.ones(n), n)
    if best_s < 0:
        best_s = 1.0
    best_h = 0.5 * np.ones(n)

    for s in starts:
        sc = exact_score(s, n)
        if sc > 0 and sc < best_s - 1e-12:
            best_s, best_h = sc, s.copy()
        _, ls, lh = optimize(s, n, m, taus, budget)
        if lh is not None and ls > 0 and ls < best_s - 1e-12:
            best_s, best_h = ls, lh

    n_rnd = 2 if n <= 64 else 1
    perts = [0.05, 0.18] if n <= 64 else ([0.08, 0.20] if n <= 160 else [0.10])
    for _ in range(n_rnd):
        sb = best_s
        for sd in perts:
            p = project_K(best_h + sd * rng.standard_normal(n), m)
            if p is None:
                continue
            sc = exact_score(p, n)
            if sc > 0 and sc < best_s - 1e-12:
                best_s, best_h = sc, p.copy()
            _, ls, lh = optimize(p, n, m, taus, budget)
            if lh is not None and ls > 0 and ls < best_s - 1e-12:
                best_s, best_h = ls, lh
        if best_s >= sb - 1e-9:
            break

    if not feasible(best_h, n):
        best_h = project_K(best_h, m)
        if best_h is None:
            best_h = 0.5 * np.ones(n)

    c5 = exact_score(best_h, n)
    if c5 <= 0:
        c5 = 0.5
    return (best_h, c5, n)
