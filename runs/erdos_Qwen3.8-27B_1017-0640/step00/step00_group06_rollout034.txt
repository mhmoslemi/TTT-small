                    found = True
                    break
            if not found:
                w = int(np.argmax(scores))
                pop[w] = global_best.copy()
                scores[w] = global_best_score

    h_final = global_best.copy()
    if not is_feasible(h_final, n, m):
        proj = project_capped(h_final, n, m)
        h_final = proj if proj is not None else fallback.copy()
    if not is_feasible(h_final, n, m):
        h_final = fallback.copy()
    c5 = exact_score(h_final, n)
    if not math.isfinite(c5):
        h_final = fallback.copy()
        c5 = exact_score(h_final, n)
    return h_final, float(c5), int(n)
