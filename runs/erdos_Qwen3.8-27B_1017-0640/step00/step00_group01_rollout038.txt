                if r <= 0:
                    break

                    r -= add

                    r += sub

    mu_F, mu_CR = 0.5, 0.8
    best_idx = int(np.argmin(scores))
    best_val = scores[best_idx]

                    stall_count = 0; improved = True

            mu_F = float(np.clip(mu_F, 0.1, 1.0))
            mu_CR = float(np.clip(mu_CR, 0.0, 1.0))

import numpy as np

def project_capped_simplex(v):
    n = len(v)
    s = n / 2.0
    if not np.all(np.isfinite(v)) or np.max(np.abs(v)) > 1e6:
        return None
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v))
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if math.fsum(np.clip(v - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0).copy()
    for _ in range(4):
        r = s - math.fsum(x)
        if abs(r) <= 5e-15:
            break
        if r > 0:
            for i in range(n):
                if r <= 0: break
                a = min(r, 1.0 - x[i])
                if a > 0: x[i] += a; r -= a
        else:
            for i in range(n):
                if r >= 0: break
                a = min(-r, x[i])
                if a > 0: x[i] -= a; r += a
    if np.min(x) < -1e-14 or np.max(x) > 1.0 + 1e-14: return None
    if abs(math.fsum(x) - s) > 1e-10: return None
    return x

def exact_score(h, n):
    try:
        val = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))
        return val if np.isfinite(val) else None
    except Exception:
        mq = 0.0
        for d in range(-(n - 1), n):
            i0, i1 = max(0, -d), min(n, n - d)
            q = float(np.sum(h[i0:i1] * (1.0 - h[i0 + d:i1 + d])))
            if q > mq: mq = q
        val = mq * (2.0 / n)
        return val if np.isfinite(val) else None

def is_feasible(h, n):

    for p in lst:
