    g[-1] = 1.0
    return g

def _eq(z):
    n = len(z) - 1
    return np.sum(z[:n]) - n / 2.0

def _eq_jac(z):
    n = len(z) - 1
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g

def _ineq(z):
    n = len(z) - 1
    h = z[:n]
    c, _ = lag_values_and_jacobian(h)
    return z[-1] - c

def _ineq_jac(z):
    n = len(z) - 1
    _, J = lag_values_and_jacobian(z[:n])
    m = 2 * n - 1
    G = np.zeros((m, n + 1))
    G[:, :n] = -J
    G[:, n] = 1.0
    return G

def _solves(s, n, maxiter, ftol):
    t0 = exact_score(s) + 1e-8
    z0 = np.concatenate([s, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'eq', 'fun': _eq, 'jac': _eq_jac},
        {'type': 'ineq', 'fun': _ineq, 'jac': _ineq_jac},
    ]
    try:
        return minimize(_obj, z0, method='SLSQP', jac=_obj_jac,
                        bounds=bounds, constraints=cons,
                        options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
    except Exception:
        return None

def _screen(res, n, h_inc, best_score):
    if res is None or res.x is None or len(res.x) != n + 1:
        return h_inc, best_score
    hp = res.x[:n]
    if not np.all(np.isfinite(hp)):
        return h_inc, best_score
    hj = project_capped_simplex(hp, n)
    if np.max(np.abs(hj - np.clip(hj, 0, 1))) > 1e-10:
        return h_inc, best_score
    if abs(np.sum(hj) - n / 2.0) > 1e-9:
        return h_inc, best_score
    s = exact_score(hj)
    if np.isfinite(s) and s < best_score - 1e-12:
        return hj, s
    return h_inc, best_score

def run(seed=42, **kwargs):
    n = 70
    parent = None
    try:
        pv = np.asarray(initial_h_values, dtype=float).ravel()
        if len(pv) >= 2 and np.all(np.isfinite(pv)) and len(pv) <= 256:
            n = len(pv)
            parent = pv.copy()
    except Exception:
        pass
    if parent is None:
        v = kwargs.get('n_points', None)
        if isinstance(v, (int, np.integer)) and 2 <= int(v) <= 256:
            n = int(v)

    if parent is not None:
        h_inc = project_capped_simplex(parent, n)
    else:
        h_inc = np.full(n, 0.5)
    best_score = exact_score(h_inc)

    if n > 256:
        return h_inc, best_score, n

    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)

    i = np.arange(n)
    starts = [
        h_inc.copy(),
        h_inc[::-1].copy(),
        np.full(n, 0.5),
        project_capped_simplex(0.5 + 0.35 * np.sin(2.0 * np.pi * (i + 0.5) / n), n),
        project_capped_simplex(0.5 + 0.25 * rng.standard_normal(n), n),
        project_capped_simplex(rng.uniform(0.0, 1.0, n), n),
    ]
    unique = [starts[0]]
    for s in starts[1:]:
        dup = False
        for us in unique:
            if np.max(np.abs(s - us)) < 1e-10:
                dup = True
                break
        if not dup:
            unique.append(s)

    maxiter = 250 if n <= 128 else 120
    ftol = 1e-10

    for s in unique:
        res = _solves(s, n, maxiter, ftol)
        h_inc, best_score = _screen(res, n, h_inc, best_score)

    if n <= 128:
        for sd in [0.03, 0.08]:
            hp = project_capped_simplex(h_inc + sd * rng.standard_normal(n), n)
            res = _solves(hp, n, maxiter, ftol)
            h_inc, best_score = _screen(res, n, h_inc, best_score)

    h_inc = project_capped_simplex(h_inc, n)
    if not (np.all(h_inc >= -1e-10) and np.all(h_inc <= 1.0 + 1e-10)):
        if parent is not None:
            h_inc = project_capped_simplex(parent, n)
        else:
            h_inc = np.full(n, 0.5)
    if abs(np.sum(h_inc) - n / 2.0) > 1e-9:
        if parent is not None:
            h_inc = project_capped_simplex(parent, n)
        else:
            h_inc = np.full(n, 0.5)
    best_score = exact_score(h_inc)
    return h_inc, best_score, n
