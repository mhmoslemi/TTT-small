import numpy as np

def project_capped(y, target):
    n = len(y)
    low = float(np.min(y)) - 1.0
    high = float(np.max(y))
    for _ in range(64):
        mid = 0.5 * (low + high)
        x = np.clip(y - mid, 0.0, 1.0)
        s = float(np.sum(x))
        if s > target:
            low = mid
        else:
            high = mid
    x = np.clip(y - 0.5 * (low + high), 0.0, 1.0)
    for _ in range(3):
        r = target - float(np.sum(x))
        if abs(r) <= 5e-13 * max(1, n):
            break
        if r > 0:
            for i in range(n):
                a = min(r, 1.0 - x[i])
                x[i] += a
                r -= a
                if r <= 0:
                    break
        else:
            r = -r
            for i in range(n):
                a = min(r, x[i])
                x[i] -= a
                r -= a
                if r <= 0:
                    break
    return x

def check_feasible(h, n, tol=1e-10):
    if not isinstance(h, np.ndarray) or h.ndim != 1 or len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-15) or np.any(h > 1.0 + 1e-15):
        return False
    if abs(float(np.sum(h)) - n / 2.0) > tol * max(1, n):
        return False
    return True

def compute_score(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    if len(c) != 2 * n - 1 or not np.all(np.isfinite(c)):
        return None
    return float(np.max(c)) * (2.0 / n)

def make_block(n):
    h = np.zeros(n, dtype=np.float64)
    h[:n // 2] = 1.0
    if n % 2 == 1:
        h[n // 2] = 0.5
    return h

            pass
    if fallback is None or not check_feasible(fallback, n):
