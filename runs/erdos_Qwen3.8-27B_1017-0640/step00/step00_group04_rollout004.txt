    target = n / 2.0
    if not np.all(np.isfinite(y)):
        return None
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y))

        return False
    if np.any(h < -tol) or np.any(h > 1.0 + tol):
        return False

    return z[-1]

    return np.concatenate([np.ones(n), [0.0]])

    h, t = z[:-1], z[-1]
    n = len(h)
    return t - (2.0 / n) * np.correlate(h, 1.0 - h, mode='full')

def _epi_jac(z):
    h = z[:-1]
    n = len(h)
    dx = 2.0 / n
    nl = 2 * n - 1
    J = np.zeros((nl, n + 1))

                d += 1.0 - h[r - lag]
            if 0 <= r + lag < n:
                d -= h[r + lag]
            J[k, r] = -dx * d
        J[k, n] = 1.0
    return J
