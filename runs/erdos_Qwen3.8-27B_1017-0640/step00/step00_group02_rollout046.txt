        add_start(starts, project_capped(unif, S), n)
        m = int(rng.integers(1, min(5, max(1, n // 2)) + 1))
        phase = rng.uniform(0.0, 2.0 * math.pi)
        idx = np.arange(n, dtype=np.float64)
        sin_h = 0.5 + 0.35 * np.sin(2.0 * math.pi * m * idx / n + phase)
        add_start(starts, project_capped(sin_h, S), n)
    elif n <= 512:
        noisy = base + rng.normal(0.0, 0.10, n)
        add_start(starts, project_capped(noisy, S), n)
        unif = rng.uniform(0.0, 1.0, n)
        add_start(starts, project_capped(unif, S), n)
    else:
        unif = rng.uniform(0.0, 1.0, n)
        add_start(starts, project_capped(unif, S), n)

    for st in starts:
        if not feasible(st, n, S):
            rep = project_capped(st, S)
            if rep is None:
                continue
            st = rep
        sc = exact_score(st, n, S)
        if sc is not None and sc < incumbent_score - 1e-12:
            incumbent = st.copy()
            incumbent_score = sc

    if n <= 64:
        L, R, cap = 12, 2, 600
    elif n <= 128:
        L, R, cap = 10, 1, 300
    elif n <= 512:
        L, R, cap = 8, 0, 120
    else:
        L, R, cap = 5, 0, 40

    Nchains = len(starts) + R
    M = max(1, min(cap, int(1.8e8 / max(1, n * n * L * max(1, Nchains)))))
    p_perm = 0.05
    p_swap = 0.15 if n <= 128 else 0.10

    for st in starts:
        if not feasible(st, n, S):
            continue
        sc = exact_score(st, n, S)
        if sc is None:
            continue
        try:
            incumbent, incumbent_score = anneal_chain(st, sc, n, S, rng, L, M, p_perm, p_swap, incumbent, incumbent_score)
        except Exception:
            pass

    if n <= 64:
        restart_sds = [0.10, 0.25]
    elif n <= 128:
        restart_sds = [0.15]
    else:
        restart_sds = []

    for sd in restart_sds:
        try:
            pert = incumbent + rng.normal(0.0, sd, n)
            proj = project_capped(pert, S)
            if proj is None:
                continue
            sc = exact_score(proj, n, S)
            if sc is None:
                continue
            incumbent, incumbent_score = anneal_chain(proj, sc, n, S, rng, L, M, p_perm, p_swap, incumbent, incumbent_score)
        except Exception:
            pass

    if not feasible(incumbent, n, S):
        rep = project_capped(incumbent, S)
        if rep is not None and feasible(rep, n, S):
            sc = exact_score(rep, n, S)
            if sc is not None and sc <= incumbent_score:
                incumbent = rep
                incumbent_score = sc
            else:
                incumbent = fallback
                incumbent_score = fallback_score
        else:
            incumbent = fallback
            incumbent_score = fallback_score

    final_score = exact_score(incumbent, n, S)
    if final_score is None:
        incumbent = fallback
        final_score = fallback_score

    incumbent = np.clip(incumbent, 0.0, 1.0)
    return incumbent.astype(np.float64), float(final_score), int(n)
