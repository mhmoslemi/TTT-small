def compute_lag_data(h, n):

    for a in range(n_lag):
        k = a - (n - 1)

        # valid r: 0 <= r-k < n => k <= r < n+k
        r_lo1 = max(0, k)
        r_hi1 = min(n, n + k)  # exclusive

            if not feasible(x, n):
                delta = max(dmin, delta * 0.5)
                continue
            Cx = exact_score(x, n)
            if not np.isfinite(Cx):
                delta = max(dmin, delta * 0.5)
                continue
            A = C_cur - Cx
            Gdx = G @ (x - h_cur)
            model_score = dx * float(np.max(q + Gdx))
            P = C_cur - model_score
            if A > 1e-12:
                s = float(np.max(np.abs(x - h_cur)))
                h_cur = x
                C_cur = Cx
                accepted += 1
                if P <= 1e-12:
                    delta = max(dmin, delta * 0.5)
                else:
                    rho = A / P
                    if rho < 0.25:
                        delta = max(dmin, delta * 0.5)
                    elif rho > 0.75 and s >= 0.8 * delta:
                        delta = min(dmax, 1.5 * delta)
                step_done = True
                break
            else:
                if P <= 1e-12:
                    return h_cur, C_cur
                delta = max(dmin, delta * 0.5)
        if not step_done and delta < dmin:
            break
    return h_cur, C_cur

def run(seed=42, **kwargs):
    global initial_h_values
    # Determine n
    n = None
    try:
        if initial_h_values is not None:
            ph = np.asarray(initial_h_values)
            if ph.ndim == 1 and 2 <= ph.shape[0] < 1000:
                n = ph.shape[0]
    except Exception:
        pass
    if n is None:
        try:
            nk = int(kwargs.get('n_points', 0))
            if 2 <= nk < 1000:
                n = nk
        except Exception:
            pass
    if n is None:
        n = 95
    dx = 2.0 / n
    m = n / 2.0

    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)

    # Base start from parent or fallback
    fallback = make_half_block(n)
    base = fallback.copy()
    has_parent = False
    try:
        ph = np.asarray(initial_h_values, dtype=float).ravel()
        if len(ph) == n and np.all(np.isfinite(ph)):
            if feasible(ph, n):
                base = ph.copy()
                has_parent = True
            else:
                proj = bounded_simplex_project(ph, np.zeros(n), np.ones(n), m)
                if proj is not None and feasible(proj, n):
                    base = proj.copy()
                    has_parent = True
    except Exception:
        pass
    if not has_parent:
        base = fallback.copy()
    protected = base.copy()

    # Build starts
    starts = [base.copy()]
    sigmas = [0.05, 0.15, 0.30]
    for sig in sigmas:
        z = rng.standard_normal(n)
        s = base + sig * z
        p = bounded_simplex_project(s, np.zeros(n), np.ones(n), m)
        if p is not None and feasible(p, n):
            starts.append(p)
    # Beta start
    bv = rng.beta(0.7, 0.7, n)
    p = bounded_simplex_project(bv, np.zeros(n), np.ones(n), m)
    if p is not None and feasible(p, n):
        starts.append(p)
    # Sinusoid start
    phase = rng.uniform(0, 2 * np.pi)
    sv = 0.5 + 0.45 * np.sin(2.0 * np.pi * np.arange(n) / n + phase)
    p = bounded_simplex_project(sv, np.zeros(n), np.ones(n), m)
    if p is not None and feasible(p, n):
        starts.append(p)
    # Half-block if not already there
    hb = make_half_block(n)
    if not np.allclose(hb, base, atol=1e-12):
        starts.append(hb.copy())

    # Deduplicate
    deduped = [starts[0]]
    for s in starts[1:]:
        is_dup = False
        for d in deduped:
            if float(np.max(np.abs(s - d))) < 1e-12:
                is_dup = True; break
        if not is_dup:
            deduped.append(s)
    starts = deduped

    # Score all starts
    best_h = protected.copy()
    best_c5 = exact_score(best_h, n)
    for s in starts:
        c = exact_score(s, n)
        if np.isfinite(c) and c < best_c5 - 1e-12:
            best_h = s.copy()
            best_c5 = c

    # Optimize each start
    scale = 95.0 / n
    b0 = max(8, min(28, int(math.ceil(28.0 * 95.0 / n))))
    b1 = max(4, min(12, int(math.ceil(12.0 * 95.0 / n))))
    d0 = 0.12; dmax = 0.25; dmin = 1e-5

    for idx, s in enumerate(starts):
        b = b0 if idx == 0 else b1
        try:
            h_opt, c_opt = optimize_start(s, n, m, dx, b, d0, dmax, dmin)
            if np.isfinite(c_opt) and c_opt < best_c5 - 1e-12:
                best_h = h_opt.copy()
                best_c5 = c_opt
        except Exception:
            pass

    # Final validation
    if not feasible(best_h, n):
        p = bounded_simplex_project(best_h, np.zeros(n), np.ones(n), m)
        if p is not None and feasible(p, n):
            best_h = p
        else:
            best_h = protected.copy()
    best_c5 = exact_score(best_h, n)
    return best_h.astype(float), float(best_c5), n
