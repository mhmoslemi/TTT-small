import numpy as np
import math

def check_feasible(h, n, tol=1e-10):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            float(np.min(h)) >= -tol and float(np.max(h)) <= 1.0 + tol and
            abs(float(np.sum(h)) - n * 0.5) <= tol)

def project_simplex(y, n):
    m = n * 0.5
    if y.shape != (n,) or not np.all(np.isfinite(y)):
        return None
    y = np.clip(y.astype(np.float64), -1e6, 1e6)
    lo, hi = float(np.min(y)) - 1.0, float(np.max(y))
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if float(np.sum(np.clip(y - mid, 0, 1))) > m:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    for _ in range(20):
        res = m - float(np.sum(x))
        if abs(res) < 1e-12:
            break
        for i in range(n):
            if abs(res) < 1e-12:
                break
            if res > 0 and x[i] < 1.0:
                v = min(1.0 - x[i], res)
                x[i] += v; res -= v
            elif res < 0 and x[i] > 0.0:
                v = min(x[i], -res)
                x[i] -= v; res += v
    return x if check_feasible(x, n) else None

def score(h, n):
    if not check_feasible(h, n):
        return None
    r = np.correlate(h, 1.0 - h, mode='full')
    if r.shape[0] != 2 * n - 1 or not np.all(np.isfinite(r)):
        return None
    return float(r.max()) * (2.0 / n)

def grads(h, n):
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n), dtype=np.float64)
    G[n - 1] = dx * (1.0 - 2.0 * h)
    for p in range(1, n):
        G[n - 1 + p, p:] += dx * (1.0 - h[:n - p])
        G[n - 1 + p, :n - p] -= dx * h[p:]
        G[n - 1 - p, :n - p] += dx * (1.0 - h[p:])
        G[n - 1 - p, p:] -= dx * h[:n - p]
    return G

def solve_mm_sub(x, n):
    try:
        import cvxpy as cp
    except ImportError:
        return None
    dx = 2.0 / n
    L = 2 * n - 1
    q = np.correlate(x, 1.0 - x, mode='full').astype(np.float64) * dx
    G = grads(x, n)
    d = cp.Variable(n)
    s = cp.Variable(non_neg=True)
    t = cp.Variable()
    cons = [d >= -x, d <= 1.0 - x, cp.sum(d) == 0,
            cp.sum_squares(d) <= s,
            q + G @ d + dx * s <= t]
    prob = cp.Problem(cp.Minimize(t), cons)
    for sol in [cp.ECOS, cp.CVXOPT, cp.SCS, cp.OSQP]:
        try:
            kw = {}
            if sol == cp.ECOS:
                kw = {'abstol': 1e-8, 'reltol': 1e-8, 'feastol': 1e-8, 'max_iters': 300}
            elif sol == cp.CVXOPT:
                kw = {'abstol': 1e-8, 'reltol': 1e-8, 'max_iters': 300}
            prob.solve(solver=sol, verbose=False, **kw)
            if prob.status in ('optimal', 'optimal_inaccurate') and d.value is not None:
                dv = np.array(d.value, dtype=np.float64).flatten()
                if dv.shape == (n,) and np.all(np.isfinite(dv)):
                    return dv
        except Exception:
            continue
    return None

def mm_optimize(x0, n, max_iter, seed):
    x = x0.copy().astype(np.float64)
    cur = score(x, n)
    if cur is None:
        return x0, cur
    rng = np.random.default_rng(seed)

            break
        y = project_simplex(x + dv, n)
        if y is None:
            break
        best_cand = y
        best_sc = score(y, n)
        if best_sc is not None:
            for th in [0.5, 0.25, 0.125]:

                if pt is not None:

                        best_sc = sc
                        best_cand = pt
        if best_sc is None or best_sc >= cur - 1e-12:
            break
        x = best_cand.copy()
        cur = best_sc
        if cur < 0.300:
            break
    return x, cur
