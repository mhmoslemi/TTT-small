            K_l = max(1, int(round(8 * (0.65 ** l))))
            for _ in range(A):
                if best_score <= 0.3808 or score_count >= B:
                    break
                prop = exchange_proposal(current, n, rng, rho_l, K_l)
                if prop is None:
                    continue
                if np.array_equal(prop, current):
                    continue
                sc = exact_score(prop)
                score_count += 1
                if not np.isfinite(sc):
                    continue
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_h = prop.copy()
                delta = sc - cur_score
                if delta <= 0:
                    current = prop
                    cur_score = sc
                else:
                    ratio = delta / T_l if T_l > 0 else float('inf')
                    p_acc = math.exp(-ratio) if ratio < 745.0 else 0.0
                    if rng.random() < p_acc:
                        current = prop
                        cur_score = sc

    return _finalize(best_h, parent_fallback, n)

def _finalize(best_h, parent_fallback, n):
    candidates = [best_h.copy(), parent_fallback.copy()]
    best_final = None
    best_final_score = float('inf')
    for c in candidates:
        fc = project_capped(c, n)
        if is_feasible(fc, n):
            sc = exact_score(fc)
            if sc < best_final_score - 1e-14:
                best_final_score = sc
                best_final = fc
    if best_final is None:
        best_final = np.full(n, 0.5, dtype=np.float64)
        best_final_score = exact_score(best_final)
    if not is_feasible(best_final, n):
        best_final = np.full(n, 0.5, dtype=np.float64)
    c5_bound = float(exact_score(best_final))
    return (best_final, c5_bound, int(n))
