    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return False
    return abs(float(np.sum(h)) - S) <= 1e-10

def project_capped(v, S):
    n = v.shape[0]
    if n < 2:
        return None
    w = np.clip(np.asarray(v, dtype=np.float64), -1e6, 1e6)
    lo, hi = float(np.min(w)) - 1.0, float(np.max(w)) + 1.0
    x = np.zeros(n)
    for _ in range(80):
        mid = (lo + hi) * 0.5
        x = np.clip(w - mid, 0.0, 1.0)
        s = float(np.sum(x))
        if abs(s - S) < 1e-13:
            break
        if s > S:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-14:
            break
    x = np.clip(w - (lo + hi) * 0.5, 0.0, 1.0)
    res = S - float(np.sum(x))
    for _ in range(3):
        if abs(res) < 1e-12:
            break
        if res > 0:
            cap = 1.0 - x
            idx = np.where(cap > 1e-12)[0]
        else:
            cap = x
            idx = np.where(cap > 1e-12)[0]
        if len(idx) == 0:
            break
        per = res / len(idx)
        adj = np.minimum(per, cap[idx]) if res > 0 else np.maximum(per, -cap[idx])
        x[idx] += adj
        res = S - float(np.sum(x))
    if not np.all(np.isfinite(x)) or np.any(x < -1e-10) or np.any(x > 1.0 + 1e-10):
        return None
    if abs(float(np.sum(x)) - S) > 1e-10:
        return None
    return np.clip(x, 0.0, 1.0)

def exact_score(h, n):
    S = float(n) / 2.0
    if not feasible(h, n, S):
        return None
    dx = 2.0 / n
    try:
        corr = np.correlate(h, 1.0 - h, mode='full')
        sc = float(np.max(corr) * dx)
    except Exception:
        return None
    return sc if np.isfinite(sc) else None

def make_proposal(h, n, S, rho, rng, p_perm, p_swap):
    u = rng.random()
    if u < p_perm:
        return h.copy()[np.asarray(rng.permutation(n))]
    if u < p_perm + p_swap:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        while j == i:
            j = int(rng.integers(n))
        out = h.copy()
        out[i], out[j] = out[j], out[i]
        return out
    i = int(rng.integers(n))
    j = int(rng.integers(n))
    while j == i:
        j = int(rng.integers(n))
    c = float(h[i] + h[j])
    L, U = max(0.0, c - 1.0), min(1.0, c)
    W = U - L
    if W < 1e-14:
        return h.copy()
    xi = rng.uniform(-rho * W, rho * W)
    r = (float(h[i]) + xi - L) % (2.0 * W)
    xp = L + r if r <= W else U - (r - W)
    out = h.copy()
    out[i] = xp
    out[j] = c - xp
    out = np.clip(out, 0.0, 1.0)
    if abs(float(np.sum(out)) - S) > 1e-9:
        return project_capped(out, S)
    return out

def add_start(starts, v, n, S):
    if v is None:
        return
    if not feasible(v, n, S):
        v = project_capped(v, S)
        if v is None:
            return
    for s in starts:
        if np.max(np.abs(v - s)) < 1e-12:
            return
    starts.append(np.array(v, dtype=np.float64))

def anneal_chain(start, n, S, rng, L, M, p_perm, p_swap, incumbent, incumbent_score):
    cs = exact_score(start, n)
    if cs is None:
        return incumbent, incumbent_score
    curr = np.array(start, dtype=np.float64)
    curr_score = cs
    pc = max(8, min(48, 1000000 // (n * n)))
    deltas = []
    for _ in range(pc):
        p = make_proposal(curr, n, S, 0.65, rng, p_perm, p_swap)
        if p is None:
            continue
        sc = exact_score(p, n)
        if sc is None:
            continue
        d = sc - curr_score
        if d > 1e-12:
            deltas.append(d)
        if sc < incumbent_score - 1e-12:
            incumbent, incumbent_score = p.copy(), sc
    if len(deltas) > 0:
        med = float(np.median(deltas))
        T0 = max(1e-5, min(0.05, med / (-math.log(0.65))))
    else:
        T0 = 0.002
    Tend = max(1e-7, 0.01 * T0)
    for level in range(L):
        denom = max(1, L - 1)
        T_l = T0 * (Tend / T0) ** (float(level) / float(denom))
        rho_l = max(0.03, 0.65 * math.sqrt(T_l / T0))
        for _ in range(M):
            p = make_proposal(curr, n, S, rho_l, rng, p_perm, p_swap)
            if p is None:
                continue
            sc = exact_score(p, n)
            if sc is None:
                continue
            if sc < incumbent_score - 1e-12:
                incumbent, incumbent_score = p.copy(), sc
            d = sc - curr_score
            if d <= 0:
                curr, curr_score = p.copy(), sc
            else:
                ratio = d / T_l
                if ratio < 700.0 and rng.random() < math.exp(-ratio):
                    curr, curr_score = p.copy(), sc
    return incumbent, incumbent_score

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    n = 45
    base = None
    try:
        arr = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if arr.size >= 2 and np.all(np.isfinite(arr)):
            n = int(arr.size)
            S = float(n) / 2.0
            if np.all(arr >= -1e-10) and np.all(arr <= 1.0 + 1e-10) and abs(float(np.sum(arr)) - S) <= 1e-12:
                base = np.clip(arr, 0.0, 1.0)
            else:
                pr = project_capped(arr, S)
                if pr is not None and feasible(pr, n, S):
                    base = pr
    except Exception:
        pass
    S = float(n) / 2.0
    if base is None:
        base = np.full(n, 0.5, dtype=np.float64)
    fb = base.copy()
    fb_score = exact_score(fb, n)
    if fb_score is None:
        fb = np.full(n, 0.5, dtype=np.float64)
        fb_score = exact_score(fb, n)
    incumbent = fb.copy()
    incumbent_score = fb_score
    starts = [incumbent.copy()]
    add_start(starts, 1.0 - base, n, S)
    if n <= 128:
        add_start(starts, base[::-1].copy(), n, S)
    add_start(starts, np.full(n, 0.5, dtype=np.float64), n, S)
    if n <= 128:
        add_start(starts, base + rng.normal(0, 0.12, size=n), n, S)
        add_start(starts, rng.uniform(0, 1, size=n), n, S)
        m = int(rng.integers(1, max(2, n // 2)))
        ph = rng.uniform(0, 2 * math.pi)
        ia = np.arange(n, dtype=np.float64)
        add_start(starts, 0.5 + 0.35 * np.sin(2.0 * math.pi * m * ia / n + ph), n, S)
    elif n <= 512:
        add_start(starts, base + rng.normal(0, 0.10, size=n), n, S)
        add_start(starts, rng.uniform(0, 1, size=n), n, S)
    else:
        add_start(starts, rng.uniform(0, 1, size=n), n, S)
    for st in starts:
        sc = exact_score(st, n)
        if sc is not None and sc < incumbent_score - 1e-12:
            incumbent, incumbent_score = st.copy(), sc
    if n <= 64:
        L_chain, R_cnt, cap_m = 12, 2, 600
    elif n <= 128:
        L_chain, R_cnt, cap_m = 10, 1, 300
    elif n <= 512:
        L_chain, R_cnt, cap_m = 8, 0, 120
    else:
        L_chain, R_cnt, cap_m = 5, 0, 40
    Nch = len(starts) + R_cnt
    M_steps = max(1, min(cap_m, int(180000000 // (n * n * max(1, L_chain) * max(1, Nch)))))
    p_perm = 0.05
    p_swap = 0.15 if n <= 128 else 0.10
    try:
        for st in starts:
            incumbent, incumbent_score = anneal_chain(st, n, S, rng, L_chain, M_steps, p_perm, p_swap, incumbent, incumbent_score)
    except Exception:
        pass
    try:
        if R_cnt >= 2:
            for sd in [0.10, 0.25]:
                pert = incumbent + rng.normal(0, sd, size=n)
                pr = project_capped(pert, S)
                if pr is not None and feasible(pr, n, S):
                    sc = exact_score(pr, n)
                    if sc is None:
                        continue
                    if sc < incumbent_score - 1e-12:
                        incumbent, incumbent_score = pr.copy(), sc
                    incumbent, incumbent_score = anneal_chain(pr, n, S, rng, L_chain, M_steps, p_perm, p_swap, incumbent, incumbent_score)
        elif R_cnt == 1:
            pert = incumbent + rng.normal(0, 0.15, size=n)
            pr = project_capped(pert, S)
            if pr is not None and feasible(pr, n, S):
                sc = exact_score(pr, n)
                if sc is None:
                    continue
                if sc < incumbent_score - 1e-12:
                    incumbent, incumbent_score = pr.copy(), sc
                incumbent, incumbent_score = anneal_chain(pr, n, S, rng, L_chain, M_steps, p_perm, p_swap, incumbent, incumbent_score)
    except Exception:
        pass
    if not feasible(incumbent, n, S):
        pr = project_capped(incumbent, S)
        if pr is not None and feasible(pr, n, S):
            incumbent = pr
    final_score = exact_score(incumbent, n)
    if final_score is None:
        incumbent, final_score = fb, fb_score
    return incumbent.copy(), float(final_score), int(n)
