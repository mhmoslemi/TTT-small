                cands = [j for j in range(P) if j != i]
                r1 = int(cands[rng.integers(0, len(cands))])
                c2 = [j for j in cands if j != r1]
                r2 = int(c2[rng.integers(0, len(c2))])
                c3 = [j for j in c2 if j != r2]
                r3 = int(c3[rng.integers(0, len(c3))])
                donor = snap[r1] + F * (snap[r2] - snap[r3])
            mask = rng.random(n) < CR
            if not np.any(mask):
                mask[rng.integers(0, n)] = True
            trial = np.where(mask, donor, snap[i])
            proj = project_capped_simplex(trial, n)
            if proj is None or np.max(np.abs(proj - snap[i])) <= 1e-15:
                continue
            s = exact_score(proj, n)
            if s is None or s >= snap_scores[i] - 1e-12:
                continue
            pop[i] = proj
            scores[i] = s
            if s < best_score - 1e-12:
                best_score = s
                best = proj.copy()

        if not any(np.max(np.abs(pop[i] - best)) <= 1e-12 for i in range(P)):
            w = int(np.argmax(scores))
            pop[w] = best
            scores[w] = best_score

        if float(np.min(snap_scores)) - float(np.min(scores)) >= 1e-10:
            stagnation = 0
        else:
            stagnation += 1

        if stagnation >= 18:
            if restarts < 4:
                n_rep = max(1, P // 3)
                idxs = np.argsort(scores)
                rids = [int(j) for j in idxs[n_rep - 1:]]
                rids = [j for j in rids if np.max(np.abs(pop[j] - best)) > 1e-12]
                sd = restart_sds[restarts]
                for ki, ri in enumerate(rids[:n_rep]):
                    if ki % 2 == 0 or n_rep == 1:
                        cand = project_capped_simplex(best + rng.normal(0, sd, size=n), n)
                    else:
                        cand = project_capped_simplex(rng.uniform(-0.5, 1.5, size=n), n)
                    if cand is not None and is_feasible(cand, n):
                        pop[ri] = cand
                        s = exact_score(cand, n)
                        scores[ri] = s if s is not None else 1e30
                        if s is not None and s < best_score - 1e-12:
                            best_score = s
                            best = cand.copy()
                restarts += 1
                stagnation = 0
            else:
                break

    for scale in [0.12, 0.05, 0.02, 0.008, 0.003]:
        imp = False
        for _ in range(poll_per):
            z = rng.standard_normal(n)
            z -= z.mean()
            zr = math.sqrt(float(z @ z))
            if zr < 1e-15:
                continue
            cand = project_capped_simplex(best + scale * z / zr, n)
            if cand is None or not is_feasible(cand, n):
                continue
            s = exact_score(cand, n)
            if s is not None and s < best_score - 1e-12:
                best_score = s
                best = cand.copy()
                imp = True
        if imp:
            for _ in range(poll_per):
                z = rng.standard_normal(n)
                z -= z.mean()
                zr = math.sqrt(float(z @ z))
                if zr < 1e-15:
                    continue
                cand = project_capped_simplex(best + scale * z / zr, n)
                if cand is None or not is_feasible(cand, n):
                    continue
                s = exact_score(cand, n)
                if s is not None and s < best_score - 1e-12:
                    best_score = s
                    best = cand.copy()

    if not is_feasible(best, n):
        p = project_capped_simplex(best, n)
        best = p if (p is not None and is_feasible(p, n)) else parent_fallback
    best = np.clip(best, 0.0, 1.0)
    sf = exact_score(best, n)
    if sf is None:
        best = parent_fallback if is_feasible(parent_fallback, n) else fallback
        sf = exact_score(best, n)
    if sf is None:
        sf = 0.0
    return best.astype(np.float64), sf, int(n)
