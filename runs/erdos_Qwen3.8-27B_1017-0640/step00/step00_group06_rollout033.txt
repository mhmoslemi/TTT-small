                        if rs < global_score - 1e-12:
                            global_best = proj.copy()
                            global_score = rs
                    replaced += 1
                if replaced > 0:
                    restarts += 1
                    stagnation = 0

    h = global_best
    if not is_feasible(h, n, m):
        proj = project_capped(h, n, m)
        h = proj if proj is not None else fallback.copy()
    if not is_feasible(h, n, m):
        h = make_half_block(n)
    c5 = exact_score(h, n)
    if not np.isfinite(c5):
        h = fallback.copy()
        c5 = exact_score(h, n)
    return (h, c5, n)
