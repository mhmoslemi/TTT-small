        if abs(r) < 1e-13:
            break
        if r > 0:
            for i in range(N):
                add = min(r, 1.0 - y[i])
                if add > 0:
                    y[i] += add
                    r -= add
                    if abs(r) < 1e-13:
                        break
        else:
            for i in range(N):
                sub = min(-r, y[i])
                if sub > 0:
                    y[i] -= sub
                    r += sub
                    if abs(r) < 1e-13:
                        break
    return y

def compute_score(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def corr_grad_mat(h, dx):
    N = len(h)
    J = np.zeros((2 * N - 1, N), dtype=np.float64)
    r = np.arange(N)
    for l in range(2 * N - 1):
        k = l - (N - 1)
        v1 = (r - k >= 0) & (r - k < N)
        v2 = (r + k >= 0) & (r + k < N)
        J[l] = dx * (np.where(v1, 1.0 - h[r - k], 0.0) - np.where(v2, h[r + k], 0.0))
    return J

def ep_obj(z, N):
    return z[N]

def ep_obj_grad(z, N):
    g = np.zeros(N + 1)
    g[N] = 1.0
    return g

def ep_eq(z, m):
    return float(np.sum(z)) - m

def ep_eq_grad(z, m):
    g = np.zeros(1)
    g[0] = 1.0
    return g

def ep_eq_h(z, N, m):
    return float(np.sum(z[:N])) - m

def ep_eq_h_grad(z, N):
    g = np.zeros(N + 1)
    g[:N] = 1.0
    return g

def ep_ineq(z, N, dx):
    h = z[:N]
    return z[N] - np.correlate(h, 1.0 - h, mode='full') * dx

def ep_ineq_jac(z, N, dx):
    h = z[:N]
    J = corr_grad_mat(h, dx)
    out = np.zeros((2 * N - 1, N + 1))
    out[:, :N] = -J
    out[:, N] = 1.0
    return out

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = None
    try:
        p = np.asarray(initial_h_values, dtype=np.float64)
        if p.ndim == 1 and len(p) > 0 and np.all(np.isfinite(p)):
            parent = p
    except Exception:
        pass
    N = len(parent) if parent is not None else 70
    dx = 2.0 / N
    m = N / 2.0
    h_uni = np.full(N, 0.5)
    best_h = h_uni.copy()
    best_c = compute_score(h_uni, dx)
    if parent is not None:
        hp = project_box_sum(parent, m)
        cp = compute_score(hp, dx)
        if cp < best_c - 1e-12:
            best_h, best_c = hp, cp
    base = best_h.copy()
    starts = [base.copy()]
    if N <= 256:
        for sc in [0.03, 0.08, 0.18]:
            starts.append(project_box_sum(base + rng.normal(0, sc, size=N), m))
        starts.append(project_box_sum(rng.uniform(0, 1, size=N), m))
        starts.append(project_box_sum(rng.beta(0.7, 0.7, size=N), m))
        x = np.arange(N) + 0.5
        for pc, ph in [(1, 0.0), (2, np.pi / 3)]:
            starts.append(project_box_sum(0.5 + 0.45 * np.cos(2.0 * np.pi * pc * x / N + ph), m))
        starts.append(project_box_sum(1.0 - base, m))
    else:
        starts.append(project_box_sum(base + rng.normal(0, 0.08, size=N), m))
        starts.append(project_box_sum(rng.uniform(0, 1, size=N), m))
    for s in starts:
        cs = compute_score(s, dx)
        if np.isfinite(cs) and cs < best_c - 1e-12:
            best_c, best_h = cs, s.copy()
    bounds = [(0.0, 1.0)] * N + [(0.0, 2.0)]
    cons = [
        {'type': 'eq', 'fun': ep_eq_h, 'jac': ep_eq_h_grad, 'args': (N, m)},
        {'type': 'ineq', 'fun': ep_ineq, 'jac': ep_ineq_jac, 'args': (N, dx)},
    ]
    for h0 in starts:
        try:
            c0 = np.max(np.correlate(h0, 1.0 - h0, mode='full') * dx)
            z0 = np.concatenate([h0, [float(np.clip(c0, 0.0, 2.0))]])
            res = minimize(ep_obj, z0, method='SLSQP', jac=ep_obj_grad, args=(N,),
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 250, 'ftol': 1e-10, 'disp': False})
            if res.x is not None and len(res.x) == N + 1 and np.all(np.isfinite(res.x)):
                h_sol = project_box_sum(res.x[:N], m)
                h_sol = np.clip(h_sol, 0.0, 1.0)
                c_sol = compute_score(h_sol, dx)
                if np.isfinite(c_sol) and c_sol < best_c - 1e-12:
                    best_c, best_h = c_sol, h_sol
        except Exception:
            continue
    # Projected gradient refinement on best solution
    for _ in range(200):
        h = best_h.copy()
        cq = np.correlate(h, 1.0 - h, mode='full') * dx
        k_star = int(np.argmax(cq))
        lag = k_star - (N - 1)
        r = np.arange(N)
        v1 = (r - lag >= 0) & (r - lag < N)
        v2 = (r + lag >= 0) & (r + lag < N)
        grad_k = dx * (np.where(v1, 1.0 - h[r - lag], 0.0) - np.where(v2, h[r + lag], 0.0))
        grad_k[0] += grad_k[-1] * 0  # no-op for clarity
        # Full gradient of max is gradient of the achieving component
        # But we need to handle ties: use average of active gradients
        active = np.where(np.abs(cq - np.max(cq)) < 1e-12)[0]
        if len(active) > 1:
            J = corr_grad_mat(h, dx)
            grad_k = J[active].mean(axis=0)
        step = 0.01 / np.sqrt(N)
        h_new = h - step * grad_k
        h_new = project_box_sum(h_new, m)
        c_new = compute_score(h_new, dx)
        if np.isfinite(c_new) and c_new < best_c - 1e-14:
            best_c, best_h = c_new, h_new
        else:
            step = 0.005 / np.sqrt(N)
            h_new = h - step * grad_k
            h_new = project_box_sum(h_new, m)
            c_new = compute_score(h_new, dx)
            if np.isfinite(c_new) and c_new < best_c - 1e-14:
                best_c, best_h = c_new, h_new
    best_h = project_box_sum(best_h, m)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c = compute_score(best_h, dx)
    if not np.isfinite(final_c) or np.abs(float(np.sum(best_h)) - m) > 1e-8:
        best_h, final_c = h_uni, compute_score(h_uni, dx)
    return (best_h, float(final_c), N)
