    return np.interp(np.linspace(0, 1, n), np.linspace(0, 1, pn), pv)

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    n = 45
    parent = None
    if 'n_points' in kwargs:
        nv = kwargs['n_points']
        if not isinstance(nv, bool) and isinstance(nv, (int, np.integer)) and 1 <= nv <= 999:
            n = int(nv)
    try:
        pv = initial_h_values
        if pv is not None:
            pv = np.asarray(pv, dtype=np.float64)
            if pv.ndim == 1 and len(pv) > 0 and np.all(np.isfinite(pv)) and len(pv) <= 999:
                if 'n_points' not in kwargs or not (isinstance(kwargs.get('n_points'), (int, np.integer)) and not isinstance(kwargs.get('n_points'), bool)):
                    n = len(pv)
                parent = transfer_parent(pv, n)
    except Exception:
        parent = None
    m = n / 2.0
    if parent is not None:
        if np.any(parent < -1e-12) or np.any(parent > 1.0 + 1e-12) or abs(np.sum(parent) - m) > 1e-9:
            parent = project_onto_k(parent, m)
    fallback = np.full(n, m / n)
    best = fallback.copy()
    best_sc = exact_score(best, n)
    if best_sc is None:
        best_sc = 0.5
    starts = build_starts(n, m, parent, rng)
    if n <= 64:
        max_start = 10
    elif n <= 160:
        max_start = 6
    elif n <= 400:
        max_start = 4
    else:
        max_start = 2
    for s in starts[:max_start]:
        sc = exact_score(s, n)
        if sc is not None and sc < best_sc - 1e-12:
            best, best_sc = s.copy(), sc
        h_new, sc_new = local_opt(s, n)
        if sc_new is not None and sc_new < best_sc - 1e-12:
            best, best_sc = h_new.copy(), sc_new
    if n <= 64:
        scales_list = [[0.05, 0.18], [0.05, 0.18]]
    elif n <= 160:
        scales_list = [[0.08, 0.20]]
    else:
        scales_list = [[0.10]]
    for scales in scales_list:
        sc0 = best_sc
        for sd in scales:
            pert = project_onto_k(best + sd * rng.standard_normal(n), m)
            if pert is None:
                continue
            sc = exact_score(pert, n)
            if sc is not None and sc < best_sc - 1e-12:
                best, best_sc = pert.copy(), sc
            h_new, sc_new = local_opt(pert, n)
            if sc_new is not None and sc_new < best_sc - 1e-12:
                best, best_sc = h_new.copy(), sc_new
        if sc0 - best_sc < 1e-9:
            break
    if best is None or best.shape != (n,) or not np.all(np.isfinite(best)):
        best = fallback.copy()
    if np.any(best < -1e-12) or np.any(best > 1.0 + 1e-12) or abs(np.sum(best) - m) > 1e-9:
        rep = project_onto_k(best, m)
        best = rep if rep is not None else fallback.copy()
    best = np.clip(best, 0.0, 1.0).astype(np.float64)
    c5 = exact_score(best, n)
    if c5 is None:
        best = fallback.copy()
        c5 = exact_score(best, n)
    if c5 is None:
        c5 = 0.5
    return best, c5, int(n)
