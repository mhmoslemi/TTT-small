def project_capped(v, n, m):
    if v.shape != (n,) or not np.all(np.isfinite(v)):
        return None
    if float(np.max(np.abs(v))) > 1e12:
        return None
    low = float(np.min(v)) - 1.0
    high = float(np.max(v))
    for _ in range(80):
        lam = (low + high) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = math.fsum(x.tolist())
        if s > m:
            low = lam
        else:
            high = lam
        if abs(s - m) <= 1e-13:
            break
    x = np.clip(v - (low + high) / 2.0, 0.0, 1.0)
    residual = m - math.fsum(x.tolist())
    for _ in range(4):
        if abs(residual) < 1e-14:
            break
        for i in range(n):
            if residual == 0.0:
                break
            if residual > 0:
                add = min(residual, 1.0 - x[i])
                x[i] += add
                residual -= add
            else:
                sub = min(-residual, x[i])
                x[i] -= sub
                residual += sub
    if not np.all(np.isfinite(x)) or not is_feasible(x, n, m):
        return None
    return x

def half_block(n):
    h = np.zeros(n, dtype=np.float64)
    h[:n // 2] = 1.0
    if n % 2 == 1:
        h[n // 2] = 0.5
    return h

def vec_dist(a, b):
    return float(np.max(np.abs(a - b)))

def build_population(base, protected, n, m, P, rng):
    pop = [protected.copy()]
    scores = [exact_score(protected)]
    if vec_dist(base, protected) > 1e-10:
        pop.append(base.copy())
        scores.append(exact_score(base))
    scales = [0.02, 0.05, 0.12, 0.25, 0.40]
    candidates = []
    for sc in scales:
        candidates.append(base + rng.standard_normal(n) * sc)
    candidates.append(np.full(n, 0.5))
    candidates.append(rng.uniform(0.0, 1.0, n))
    candidates.append(rng.beta(0.7, 0.7, n))
    phase = rng.uniform(0.0, 2.0 * math.pi)
    idx = np.arange(n, dtype=np.float64)
    candidates.append(0.5 + 0.45 * np.sin(2.0 * math.pi * idx / n + phase))
    candidates.append(base[rng.permutation(n)])
    hb = half_block(n)
    if vec_dist(hb, protected) > 1e-10:
        candidates.append(hb.copy())
    for v in candidates:
        if len(pop) >= P:
            break
        p = project_capped(v, n, m)
        if p is None:
            continue
        distinct = True
        for existing in pop:
            if vec_dist(p, existing) < 1e-10:
                distinct = False
                break
        if distinct:
            pop.append(p)
            scores.append(exact_score(p))
    attempts = 0
    while len(pop) < P and attempts < 30 * P:
        sc = scales[attempts % len(scales)]
        v = base + rng.standard_normal(n) * sc
        p = project_capped(v, n, m)
        attempts += 1
        if p is None:
            continue
        distinct = True
        for existing in pop:
            if vec_dist(p, existing) < 1e-10:
                distinct = False
                break
        if distinct:
            pop.append(p)
            scores.append(exact_score(p))
    while len(pop) < P:
        pop.append(protected.copy())
        scores.append(scores[0])
    return pop, scores

def run(seed=42, **kwargs):
    if seed is None:
        seed = 42
    rng = np.random.default_rng(seed)
    n = 95
    try:
        arr = np.array(initial_h_values, dtype=np.float64)
        if arr.ndim == 1 and 2 <= arr.shape[0] <= 999:
            n = arr.shape[0]
    except Exception:
        pass
    m = n / 2.0
    P = max(8, min(24, int(20 * math.sqrt(95.0 / n) + 0.5)))
    protected = half_block(n)
    base = protected.copy()
    try:
        arr = np.array(initial_h_values, dtype=np.float64)
        if arr.ndim == 1 and arr.shape[0] == n and np.all(np.isfinite(arr)):
            if float(np.max(np.abs(arr))) < 1e12:
                clipped = np.clip(arr, 0.0, 1.0)
                res = abs(math.fsum(clipped.tolist()) - m)
                if res <= 1e-13:
                    base = clipped
                    protected = clipped
                else:
                    p = project_capped(clipped, n, m)
                    if p is not None:
                        base = p
                        protected = p
    except Exception:
        pass
    B = max(P, min(2500, int(2500.0 * (95.0 / n) ** 2)))
    max_gen = min(120, (B - P) // P) if B > P else 0
    pop, scores = build_population(base, protected, n, m, P, rng)
    score_calls = len(scores)
    best_idx = int(np.argmin(scores))
    global_best = pop[best_idx].copy()
    global_best_score = scores[best_idx]
    stagnation = 0
    restarts_done = 0
    for gen in range(max_gen):
        if score_calls + P > B:
            break
        next_pop = [x.copy() for x in pop]
        next_scores = list(scores)
        improved_gen = False
        best_idx = int(np.argmin(scores))
        for i in range(P):
            if score_calls >= B:
                break
            F_val = rng.uniform(0.35, 0.85)
            others = [j for j in range(P) if j != i]
            if rng.random() < 0.8:
                donors = rng.choice(others, size=3, replace=False)
                v = pop[donors[0]] + F_val * (pop[donors[1]] - pop[donors[2]])
            else:
                if best_idx in others:
                    elig = [j for j in others if j != best_idx]
                    if len(elig) >= 2:
                        donors = rng.choice(elig, size=2, replace=False)
                    else:
                        donors = rng.choice(others, size=2, replace=False)
                else:
                    donors = rng.choice(others, size=2, replace=False)
                v = pop[i] + F_val * (pop[best_idx] - pop[i]) + F_val * (pop[donors[0]] - pop[donors[1]])
            if rng.random() < 0.2:
                CR = rng.uniform(0.65, 0.95)
            else:
                CR = rng.uniform(0.05, 0.45)
            mask = rng.random(n) < CR
            mask[int(rng.integers(0, n))] = True
            trial = np.where(mask, v, pop[i])
            p = project_capped(trial, n, m)
            if p is None:
                continue
            score_calls += 1
            try:
                s = exact_score(p)
            except Exception:
                continue
            if not math.isfinite(s):
                continue
            if s < scores[i] - 1e-12:
                next_pop[i] = p
                next_scores[i] = s
                if s < global_best_score - 1e-12:
                    global_best = p.copy()
                    global_best_score = s
                    improved_gen = True
        pop = next_pop
        scores = next_scores
        found = False
        for j in range(P):
            if vec_dist(pop[j], global_best) < 1e-12:
                found = True
                break
        if not found:
            worst = int(np.argmax(scores))
            pop[worst] = global_best.copy()
            scores[worst] = global_best_score
        if improved_gen:
            stagnation = 0
        else:
            stagnation += 1
        if stagnation >= 12 and restarts_done < 4:
            R = max(1, P // 3)
            if score_calls + R + P <= B:
                rscales = [0.03, 0.08, 0.20, 0.35]
                order = np.argsort(scores)
                replace_idx = []
                for j in order:
                    ji = int(j)
                    if ji != best_idx and len(replace_idx) < R:
                        replace_idx.append(ji)
                for k, idx_rep in enumerate(replace_idx):
                    if k % 2 == 0:
                        sc = rscales[restarts_done % len(rscales)]
                        v = global_best + rng.standard_normal(n) * sc
                    else:
                        v = rng.beta(0.7, 0.7, n)
                    p = project_capped(v, n, m)
                    if p is not None:
                        score_calls += 1
                        try:
                            s = exact_score(p)
                        except Exception:
                            s = scores[idx_rep]
                        pop[idx_rep] = p
                        scores[idx_rep] = s
                        if s < global_best_score - 1e-12:
                            global_best = p.copy()
                            global_best_score = s
                restarts_done += 1
                stagnation = 0
    result = global_best
    if not is_feasible(result, n, m):
        p = project_capped(result, n, m)
        if p is not None:
            result = p
        else:
            result = protected.copy()
    c5 = exact_score(result)
    return result, c5, n
