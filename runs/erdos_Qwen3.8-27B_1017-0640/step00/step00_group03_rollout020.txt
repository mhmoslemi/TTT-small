import numpy as np
import math


def check_feasible(h, n):
    if h.ndim != 1 or len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10:
        return False
    return True


def project_capped(v, n):
    if v.ndim != 1 or len(v) != n:
        return None
    v = np.asarray(v, dtype=np.float64).copy()
    if not np.all(np.isfinite(v)):
        return None
    v = np.clip(v, -1e6, 1e6)
    target = n / 2.0
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        q = np.clip(v - mid, 0.0, 1.0)
        if np.sum(q) > target:
            lo = mid
        else:
            hi = mid
    q = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(q)
    if abs(residual) > 1e-13:
        i = 0
        while abs(residual) > 1e-13 and i < n:
            if residual > 0:
                add = min(residual, 1.0 - q[i])
                q[i] += add
                residual -= add
            else:
                sub = min(-residual, q[i])
                q[i] -= sub
                residual += sub
            i += 1
    if not np.all(np.isfinite(q)):
        return None
    if np.any(q < -1e-12) or np.any(q > 1.0 + 1e-12):
        return None
    if abs(np.sum(q) - target) > 1e-10:
        return None
    q = np.clip(q, 0.0, 1.0)
    return q


def exact_score(h, n):
    if not check_feasible(h, n):
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if len(corr) != 2 * n - 1:
        return float('inf')
    if not np.all(np.isfinite(corr)):
        return float('inf')
    s = float(np.max(corr * dx))
    if not np.isfinite(s):
        return float('inf')
    return s


def make_proposal(h, radius, n, rng):
    for _ in range(8):
        p = int(rng.integers(n))
        q = int(rng.integers(n))
        if p == q:
            continue
        a_min = max(-h[p], h[q] - 1.0)
        a_max = min(1.0 - h[p], h[q])
        a_min = max(a_min, -radius)
        a_max = min(a_max, radius)
        if a_max - a_min <= 1e-12:
            continue
        alpha = float(rng.uniform(a_min, a_max))
        if abs(alpha) <= 1e-12:
            continue
        hp = h.copy()
        hp[p] += alpha
        hp[q] -= alpha
        proj = project_capped(hp, n)
        if proj is not None and check_feasible(proj, n):
            return proj
    return None


def anneal_chain(start, n, rng, global_best, global_score):
    cur = start.copy()
    cur_score = exact_score(cur, n)
    if np.isfinite(cur_score) and cur_score < global_score - 1e-12:
        global_score = cur_score
        global_best = cur.copy()
    radii = [0.18, 0.12, 0.07, 0.035, 0.015]
    temps = [0.006, 0.002, 0.0007, 0.00025, 0.00008]
    n_attempts = 240 if n <= 128 else 80
    for r, T in zip(radii, temps):
        for _ in range(n_attempts):
            prop = make_proposal(cur, r, n, rng)
            if prop is None:
                continue
            ps = exact_score(prop, n)
            if not np.isfinite(ps):
                continue
            delta = ps - cur_score
            if delta <= 0:
                accept = True
            else:
                ratio = delta / T
                if ratio >= 745.0:
                    accept = False
                else:
                    accept = (float(rng.random()) < math.exp(-ratio))
            if accept:
                cur = prop
                cur_score = ps
            if np.isfinite(ps) and ps < global_score - 1e-12:
                global_score = ps
                global_best = prop.copy()
    return global_best, global_score


def polish(best, n, rng):
    cur = best.copy()
    cur_score = exact_score(cur, n)
    radii = [0.02, 0.01, 0.005, 0.002, 0.001]
    n_att = 100 if n <= 128 else 40
    for r in radii:
        for _ in range(n_att):
            prop = make_proposal(cur, r, n, rng)
            if prop is None:
                continue
            ps = exact_score(prop, n)
            if np.isfinite(ps) and ps < cur_score - 1e-12:
                cur = prop
                cur_score = ps
    return cur, cur_score


def construct_starts(n, parent, rng):
    starts = []
    seen = []

    def add_feasible(vec):
        if vec is None:
            return
        proj = project_capped(vec, n)
        if proj is None:
            return
        if not check_feasible(proj, n):
            return
        for s in seen:
            if np.max(np.abs(s - proj)) < 1e-10:
                return
        seen.append(proj.copy())
        starts.append(proj.copy())

    count = 5 if n <= 128 else 3

    uniform = np.full(n, 0.5, dtype=np.float64)
    add_feasible(uniform)

    if parent is not None and len(parent) == n:
        p = np.asarray(parent, dtype=np.float64).copy()
        add_feasible(p)
        if len(starts) < count:
            add_feasible(p[::-1].copy())
        if len(starts) < count:
            add_feasible(1.0 - p)

    idx = np.arange(n, dtype=np.float64)
    add_feasible(0.5 + 0.35 * np.sin(2.0 * np.pi * (idx + 0.5) / n))
    if len(starts) < count:
        add_feasible(0.5 + 0.30 * rng.standard_normal(n))
    if parent is None and len(starts) < count:
        add_feasible(rng.random(n))

    return starts[:count]


def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)

    n = 70
    parent = None

                else:
                    parent = None
        except Exception:
            parent = None

        try:
            kv = kwargs['n_points']
            if not isinstance(kv, bool) and int(kv) == kv and 2 <= int(kv) <= 256:
                n = int(kv)
        except Exception:
            pass

    uniform = np.full(n, 0.5, dtype=np.float64)

            fallback = parent.copy()
        else:
            proj = project_capped(parent, n)
            if proj is not None:
                fallback = proj

    if n > 256:
        s = exact_score(fallback, n)
        return fallback, s, n

    starts = construct_starts(n, fallback if parent is not None else parent, rng)
    if not starts:
        starts = [uniform.copy()]

    global_best = fallback.copy()
    global_score = exact_score(global_best, n)

    for st in starts:
        try:
            global_best, global_score = anneal_chain(st, n, rng, global_best, global_score)
        except Exception:
            pass

    try:
        polished, ps = polish(global_best, n, rng)
        if np.isfinite(ps) and ps < global_score - 1e-12:
            global_best = polished.copy()
            global_score = ps
    except Exception:
        pass

    candidates = [global_best, fallback, uniform]
    best_vec = uniform.copy()
    best_sc = exact_score(uniform, n)
    for c in candidates:
        if check_feasible(c, n):
            sc = exact_score(c, n)
            if np.isfinite(sc) and sc < best_sc - 1e-12:
                best_vec = c.copy()
                best_sc = sc
            elif np.isfinite(sc) and sc < best_sc:
                best_vec = c.copy()
                best_sc = sc
    if not check_feasible(best_vec, n):
        proj = project_capped(best_vec, n)
        if proj is not None and check_feasible(proj, n):
            best_vec = proj
            best_sc = exact_score(proj, n)
        else:
            best_vec = fallback.copy()
            best_sc = exact_score(fallback, n)
    final_sc = exact_score(best_vec, n)
    return best_vec, final_sc, n
