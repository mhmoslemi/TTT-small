import numpy as np
import math

def project_capped_simplex(y, n):
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0).copy()
    r = target - float(np.sum(x))
    for _ in range(3):
        if abs(r) <= 5e-13 * max(1, n):
            break
        if r > 0:
            for i in range(n):
                if r <= 0:
                    break
                add = min(r, 1.0 - x[i])
                x[i] += add
                r -= add
        else:
            for i in range(n):
                if r >= 0:
                    break
                sub = min(-r, x[i])
                x[i] -= sub
                r += sub
    x = np.clip(x, 0.0, 1.0)
    if not np.all(np.isfinite(x)) or abs(float(np.sum(x)) - target) > 1e-10 * max(1, n):
        return None
    return x

    return (h.shape == (n,) and bool(np.all(np.isfinite(h))) and
            bool(np.all(h >= -1e-12)) and bool(np.all(h <= 1.0 + 1e-12)) and
            abs(float(np.sum(h)) - n / 2.0) <= 1e-10 * max(1, n))

def exact_score(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    if corr.shape[0] != 2 * n - 1:
        return None
    v = float(np.max(corr)) * (2.0 / n)
    return v if math.isfinite(v) else None

def get_interval(h, d):
    t_lo, t_hi = 0.0, 0.0
    for i in range(len(h)):
        di = d[i]
        if di > 1e-15:
            t_lo = max(t_lo, -h[i] / di)
            t_hi = min(t_hi, (1.0 - h[i]) / di)
        elif di < -1e-15:
            t_lo = max(t_lo, (1.0 - h[i]) / di)
            t_hi = min(t_hi, -h[i] / di)
    if not (math.isfinite(t_lo) and math.isfinite(t_hi)) or t_hi - t_lo <= 1e-12:
        return None
    return (t_lo, t_hi)

def line_val(t, q0, b, a, n):
    return float(np.max(q0 + t * b + t * t * a)) * (2.0 / n)

def golden_min(lo, hi, q0, b, a, n):
    r = (math.sqrt(5.0) - 1.0) / 2.0
    x1 = hi - r * (hi - lo)
    x2 = lo + r * (hi - lo)
    f1 = line_val(x1, q0, b, a, n)
    f2 = line_val(x2, q0, b, a, n)
    bt = x1 if f1 <= f2 else x2
    bv = min(f1, f2)
    for _ in range(12):
        if f1 < f2:
            hi = x2
            x2, f2 = x1, f1
            x1 = hi - r * (hi - lo)
            f1 = line_val(x1, q0, b, a, n)
        else:
            lo = x1
            x1, f1 = x2, f2
            x2 = lo + r * (hi - lo)
            f2 = line_val(x2, q0, b, a, n)
        if f1 < bv:
            bv, bt = f1, x1
        if f2 < bv:
            bv, bt = f2, x2
    return bt, bv

        return h, q0, cur_score
    lo, hi = iv

    a_arr = -np.correlate(d, d, mode='full')
