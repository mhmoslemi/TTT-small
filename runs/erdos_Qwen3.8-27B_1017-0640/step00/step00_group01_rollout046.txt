def project_capped_simplex(v):

        return None
    s = n / 2.0
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v))
    for _ in range(80):
        mid = (lo + hi) / 2.0
        c = np.clip(v - mid, 0.0, 1.0)
        if math.fsum(c) > s:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0).copy()
    for _ in range(4):
        r = s - math.fsum(x)
        if abs(r) <= 5e-15:
            break
        changed = False
        idx = np.arange(n) if r > 0 else np.arange(n - 1, -1, -1)
        for i in idx:
            if r > 0:
                d = min(r, 1.0 - x[i])
                if d > 0:

                    r -= d
                    changed = True

                if d > 0:

                    changed = True
            if abs(r) <= 5e-15:

            break

        return None
    if abs(math.fsum(x) - s) > 1e-9:
        return None
    return np.clip(x, 0.0, 1.0)

def is_feasible(h, n):
    if h is None or h.ndim != 1 or h.shape[0] != n:
        return False
    if not np.all(np.isfinite(h)) or h.min() < -1e-9 or h.max() > 1.0 + 1e-9:
        return False
    return abs(math.fsum(h) - n / 2.0) <= 1e-9

def exact_score(h, n):
    if h is None or h.ndim != 1 or h.shape[0] != n or not np.all(np.isfinite(h)):
        return None
    dx = 2.0 / n
    try:
        val = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx
    except Exception:
        mx = 0.0
        for d in range(-(n - 1), n):
            lo = max(0, -d)
            hi = min(n, n - d)
            mx = max(mx, float(np.sum(h[lo:hi] * (1.0 - h[lo + d:hi + d]))))
        val = mx * dx
    if not np.isfinite(val):
        return None
    return val

        return
    p = project_capped_simplex(v)
    if p is None or not is_feasible(p, n):
        return

        if np.max(np.abs(p - ex)) <= 1e-10:
            return
    sc = exact_score(p, n)
    if sc is None:
        return
    if len(cands) >= cap:
        return
    cands.append(p.copy())
    scores.append(sc)

def solve_relaxation(n):
    try:
        import cvxpy as cp
    except ImportError:
        return None, None
    try:
        s = n / 2.0
        h = cp.Variable(n)
        X = cp.Variable((n, n), symmetric=True)
        t = cp.Variable()
        M = cp.bmat([[cp.Constant(1.0), h], [h, X]])
        cons = [M >> 0, h >= 0, h <= 1, cp.sum(h) == s, t >= 0, t <= 1]
        for i in range(n):
            cons.append(cp.sum(X[i, :]) == s * h[i])
        for i in range(n):
            for j in range(n):
                cons.append(X[i, j] >= 0)
                cons.append(X[i, j] >= h[i] + h[j] - 1)
                cons.append(X[i, j] <= h[i])
                cons.append(X[i, j] <= h[j])
        dx = 2.0 / n

            hi = min(n, n - d)
            ss = cp.sum(h[lo:hi]) - cp.sum(X[np.ix_(lo:hi, lo + d:hi + d)])
            cons.append(t >= dx * ss)
        prob = cp.Problem(cp.Minimize(t), cons)
        prob.solve(solver=cp.CVXOPT, max_iters=80, abstol=1e-7, reltol=1e-7, feastol=1e-7, verbose=False)
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None, None

        Xv = np.array(X.value, dtype=np.float64)

def make_cov_factor(h_raw, X_mat, n):
    try:
        P = np.eye(n) - np.ones((n, n)) / n
        S = 0.5 * (X_mat + X_mat.T) - np.outer(h_raw, h_raw)
        S = P @ S @ P
        S = 0.5 * (S + S.T)
        evals, evecs = np.linalg.eigh(S)
        tol = 1e-4 * max(1.0, float(np.max(np.abs(evals))))
        if evals[0] < -tol:
            return None
        evals = np.clip(evals, 0.0, None)
        thresh = 1e-12 * max(1.0, float(np.max(np.abs(evals))))
        idx = np.where(evals > thresh)[0]
        if len(idx) == 0:
            return None
        L = evecs[:, idx] * np.sqrt(evals[idx])
        L = L - np.mean(L, axis=0, keepdims=True)
        return L
    except Exception:
        return None

    if cur_sc is None:
        return cur, cur_sc
    Lnorm = None
    if L is not None and L.size > 0:
        se = float(np.sum(np.diag(L @ L.T)))
        if se > 1e-12:
            Lnorm = L / math.sqrt(se / n)
    for ti, T in enumerate(temps):
        sigma = scales[ti]
        for _ in range(n_prop):
            if Lnorm is not None and rng.random() < prob:
                z = rng.standard_normal(Lnorm.shape[1])
                delta = Lnorm @ z
            else:
                delta = rng.standard_normal(n)
            delta -= np.mean(delta)
            delta[-1] -= math.fsum(delta)
            if not np.all(np.isfinite(delta)):
                continue
            y = cur + sigma * delta
            if y.min() < 0.0 or y.max() > 1.0 or not np.all(np.isfinite(y)):
                continue
            if abs(math.fsum(y) - n / 2.0) > 1e-9:

                cur = y.copy()
                cur_sc = y_sc
    return cur, cur_sc
