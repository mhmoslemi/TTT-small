def _check_feasible(h, n):
    if h.ndim != 1 or len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False

        return False
    if abs(float(np.sum(h)) - n / 2.0) > 1e-9:
        return False
    return True

def _project_q(v, n):

        return None

    hi = float(v.max())
    for _ in range(80):
        mid = (lo + hi) / 2.0
        q = np.clip(v - mid, 0.0, 1.0)
        if float(np.sum(q)) > target:
            lo = mid
        else:
            hi = mid
    q = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    r = target - float(np.sum(q))
    for _ in range(2):
        idx = np.arange(n)
        for j in idx:

                break

                q[j] += add
                r -= add
            else:
                sub = min(-r, q[j])
                q[j] -= sub
                r += sub

        if abs(r_check) < 1e-12:

        return None
    return q

def _score(h, n):
    if not _check_feasible(h, n):
        return float('inf')
    try:
        corr = np.correlate(h, 1.0 - h, mode='full')
        if len(corr) != 2 * n - 1:
            return float('inf')
        if not np.all(np.isfinite(corr)):
            return float('inf')
        return float(np.max(corr)) * (2.0 / n)
    except Exception:
        return float('inf')

def _make_uniform(n):
    return np.full(n, 0.5, dtype=np.float64)

def _make_sine_profile(n, phi):
    i = np.arange(n, dtype=np.float64)
    v = 0.5 + 0.35 * np.sin(2.0 * np.pi * (i + 0.5) / n + phi)
    return _project_q(v, n)

def _make_cos_profile(n):
    i = np.arange(n, dtype=np.float64)
    v = 0.5 + 0.2 * np.cos(4.0 * np.pi * (i + 0.5) / n)
    return _project_q(v, n)

def _de_step(rng, pop, scores, arch_h, arch_s, n, P, F_range, CR_range):

    CR = rng.uniform(CR_range[0], CR_range[1])

    next_pop = [pop[k].copy() for k in range(P)]
    next_scores = list(scores)

        mask = rng.random(n) < CR
        j_rand = int(rng.integers(0, n))

            arch_h = proj.copy()
            arch_s = s

            next_pop[i] = proj
            next_scores[i] = s
