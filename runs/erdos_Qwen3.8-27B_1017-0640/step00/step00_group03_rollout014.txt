def project_capped_simplex(v, target):
    n = len(v)
    if not np.all(np.isfinite(v)):
        return np.full(n, target / n)
    v = np.clip(v, -1e6, 1e6)
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(80):
        mid = lo + (hi - lo) / 2.0
        q = np.clip(v - mid, 0.0, 1.0)
        if q.sum() > target:
            lo = mid
        else:
            hi = mid
    q = np.clip(v - (lo + (hi - lo) / 2.0), 0.0, 1.0)
    r = target - q.sum()
    if abs(r) > 1e-13:
        for i in range(n):
            if abs(r) < 1e-13:
                break
            if r > 0:
                add = min(r, 1.0 - q[i])
                q[i] += add
                r -= add
            else:
                sub = min(-r, q[i])
                q[i] -= sub
                r += sub

import numpy as np


def project_capped_simplex(v, target):
    n = len(v)
    if not np.all(np.isfinite(v)):
        return np.full(n, target / n)
    v = np.clip(v, -1e6, 1e6)
    lo = float(v.min()) - 1.0
    hi = float(v.max())
    for _ in range(80):
        mid = lo + (hi - lo) * 0.5
        q = np.clip(v - mid, 0.0, 1.0)
        if q.sum() > target:
            lo = mid
        else:
            hi = mid
    q = np.clip(v - (lo + (hi - lo) * 0.5), 0.0, 1.0)
    r = target - q.sum()
    if abs(r) > 1e-13:
        for i in range(n):
            if abs(r) < 1e-13:
                break
            if r > 0:
                add = min(r, 1.0 - q[i])
                q[i] += add
                r -= add
            else:
                sub = min(-r, q[i])
                q[i] -= sub
                r += sub
    if not np.all(np.isfinite(q)) or q.min() < -1e-12 or q.max() > 1.0 + 1e-12 or abs(q.sum() - target) > 1e-10:
        return np.full(n, target / n)
    return q


def validate_h(h, n):
    if h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if h.min() < -1e-10 or h.max() > 1.0 + 1e-10:
        return False
    if abs(h.sum() - n / 2.0) > 1e-10:
        return False
    return True


def exact_score(h):
    n = len(h)
    if not validate_h(h, n):
        return float('inf')
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    if c.shape[0] != 2 * n - 1 or not np.all(np.isfinite(c)):
        return float('inf')
    return float(c.max())


def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    c = np.empty(2 * n - 1)
    c[n - 1] = float(np.sum(h * (1.0 - h))) * dx
    for d in range(1, n):
        u = h[:n - d]
        v = h[d:]
        c[n - 1 - d] = float(np.sum(u * (1.0 - v))) * dx
        c[n - 1 + d] = float(np.sum(v * (1.0 - u))) * dx
    return c


def compute_jacobian(h):
    n = len(h)
    dx = 2.0 / n
    J = np.zeros((2 * n - 1, n))
    J[n - 1, :] = (1.0 - 2.0 * h) * dx
    for d in range(1, n):
        row_b = np.zeros(n)
        row_b[:n - d] += (1.0 - h[d:]) * dx
        row_b[d:] -= h[:n - d] * dx
        J[n - 1 - d, :] = row_b
        row_a = np.zeros(n)
        row_a[d:] += (1.0 - h[:n - d]) * dx
        row_a[:n - d] -= h[d:] * dx
        J[n - 1 + d, :] = row_a
    return J


def surrogate_eval(c, tau):
    cmax = float(c.max())
    q = np.exp((c - cmax) / tau)
    s = float(q.sum())
    if s <= 0.0 or not np.isfinite(s):
        return None, None
    p = q / s
    val = cmax + tau * np.log(s)
    if not np.isfinite(val):
        return None, None
    return float(val), p


def run_pgd(h_start, tau_seq, max_iter_per_tau):
    n = len(h_start)
    h = h_start.copy()
    for tau in tau_seq:
        alpha = 0.5
        stall = 0
        for _ in range(max_iter_per_tau):
            c = compute_lags(h)
            J = compute_jacobian(h)
            f_val, p = surrogate_eval(c, tau)
            if f_val is None:
                return h
            g = J.T @ p
            if not np.all(np.isfinite(g)):
                return h
            accepted = False
            for _bt in range(24):
                h_trial = project_capped_simplex(h - alpha * g, n / 2.0)
                c_trial = compute_lags(h_trial)
                f_trial, _ = surrogate_eval(c_trial, tau)
                if f_trial is None:
                    alpha *= 0.5
                    continue
                delta = h_trial - h
                gdot = float(g @ delta)
                rhs = f_val + 1e-4 * gdot + 1e-12 * max(1.0, abs(f_val))
                if f_trial <= rhs:
                    disp = float(np.abs(delta).max())
                    h = h_trial
                    accepted = True
                    if disp <= 1e-8:
                        stall += 1
                    else:
                        stall = 0
                    break
                alpha *= 0.5
            if not accepted:
                break
            alpha = min(1.5 * alpha, 2.0)
            if stall >= 8:
                break
    return h


def run(seed=42, **kwargs):
    parent = globals().get('initial_h_values', None)
    n = 70
    parent_copy = None
    if parent is not None:
        try:
            p = np.array(parent, dtype=np.float64).ravel()
            if p.size >= 2:
                n = p.size
                if np.all(np.isfinite(p)) and p.min() >= -1e-10 and p.max() <= 1.0 + 1e-10:
                    if abs(p.sum() - n / 2.0) <= 1e-10:
                        parent_copy = p.copy()
                    else:
                        parent_copy = project_capped_simplex(p, n / 2.0)
                else:
                    parent_copy = None
        except Exception:
            parent_copy = None
    if parent is None:
        try:
            npts = int(kwargs.get('n_points', None) or 70)
            if 2 <= npts <= 256:
                n = npts
        except Exception:
            pass

    fallback = parent_copy if parent_copy is not None else np.full(n, 0.5)
    if not validate_h(fallback, n):
        fallback = project_capped_simplex(fallback, n / 2.0)
    fb_score = exact_score(fallback)

    if n > 256:
        return fallback, fb_score, n

    rng = np.random.default_rng(seed)
    starts = []
    if parent_copy is not None:
        starts.append(project_capped_simplex(parent_copy, n / 2.0))
        starts.append(project_capped_simplex(parent_copy[::-1], n / 2.0))
        starts.append(project_capped_simplex(parent_copy + 0.05 * rng.standard_normal(n), n / 2.0))
        starts.append(project_capped_simplex(parent_copy + 0.15 * rng.standard_normal(n), n / 2.0))
    else:
        idx = np.arange(n, dtype=np.float64)
        starts.append(np.full(n, 0.5))
        starts.append(project_capped_simplex(0.5 + 0.3 * np.sin(2.0 * np.pi * (idx + 0.5) / n), n / 2.0))
        starts.append(project_capped_simplex(0.5 + 0.15 * rng.standard_normal(n), n / 2.0))
        starts.append(project_capped_simplex(0.5 + 0.35 * rng.standard_normal(n), n / 2.0))
    starts.append(project_capped_simplex(np.full(n, 0.5), n / 2.0))

    unique_starts = []
    max_starts = 5 if n <= 128 else 4
    for s in starts:
        if len(unique_starts) >= max_starts:
            break
        dup = any(np.abs(s - u).max() < 1e-10 for u in unique_starts)
        if not dup:
            unique_starts.append(s)
    starts = unique_starts

    inc_h = fallback.copy()
    inc_score = fb_score
    tau_seq = [0.02, 0.005, 0.001, 0.0002, 0.00004, 0.000008]
    max_iter = 100 if n <= 128 else 40

    for s in starts:
        try:
            res = run_pgd(s, tau_seq, max_iter)
            if validate_h(res, n):
                sc = exact_score(res)
                if sc < inc_score - 1e-12:
                    inc_h = res.copy()
                    inc_score = sc
        except Exception:
            pass

    if n <= 128:
        try:
            shake = project_capped_simplex(inc_h + 0.04 * rng.standard_normal(n), n / 2.0)
            if np.abs(shake - inc_h).max() >= 1e-10:
                res = run_pgd(shake, tau_seq, max_iter)
                if validate_h(res, n):
                    sc = exact_score(res)
                    if sc < inc_score - 1e-12:
                        inc_h = res.copy()
                        inc_score = sc
        except Exception:
            pass

    if not validate_h(inc_h, n):
        inc_h = project_capped_simplex(inc_h, n / 2.0)
    final_score = exact_score(inc_h)

    if fb_score < final_score - 1e-12 and validate_h(fallback, n):
        inc_h = fallback.copy()
        final_score = fb_score

    if not validate_h(inc_h, n):
        inc_h = np.full(n, 0.5)
        final_score = exact_score(inc_h)

    return inc_h, final_score, n
