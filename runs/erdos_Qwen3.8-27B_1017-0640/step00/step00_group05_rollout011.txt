            if proj is not None:
                starts.append(proj)
        except Exception:
            pass
    idx = np.arange(n, dtype=np.float64)
    cos_v = 0.5 + 0.45 * np.cos(2.0 * np.pi * 2.0 * (idx + 0.5) / n + 0.37)
    proj = project_capped_simplex(cos_v, m, n)
    if proj is not None:
        starts.append(proj)
    unique_starts = []
    for s in starts:
        dup = False
        for u in unique_starts:
            if np.max(np.abs(s - u)) <= 1e-12:
                dup = True
                break
        if not dup and check_feasible(s, n, m):
            unique_starts.append(s)
    h_inc = None
    c_inc = float('inf')
    for s in unique_starts:
        c_s = exact_score(s, dx)
        if c_s < c_inc - 1e-12:
            c_inc = c_s
            h_inc = s.copy()
    if n <= 999:
        max_iter = 45 if n <= 256 else 15
        for s in unique_starts:
            h_res, c_res = local_search(s, dx, n, m, max_iter, 0.5)
            if c_res < c_inc - 1e-12:
                c_inc = c_res
                h_inc = h_res.copy()
        if n <= 256 and h_inc is not None:
            h_pol, c_pol = local_search(h_inc, dx, n, m, 20, 0.1)
            if c_pol < c_inc - 1e-12:
                c_inc = c_pol
                h_inc = h_pol.copy()
    if h_inc is None:
        h_inc = h_neutral
    h_final = project_capped_simplex(h_inc, m, n)
    if h_final is None or not check_feasible(h_final, n, m):
        h_final = h_neutral
    c5 = exact_score(h_final, dx)
    if not np.isfinite(c5):
        h_final = h_neutral
        c5 = exact_score(h_final, dx)
    return h_final, c5, n
