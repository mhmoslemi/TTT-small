        return None
    return x


def smooth_obj(h, tau, dx):
    n = len(h)
    q = np.correlate(h, 1.0 - h, mode='full')
    qmax = float(np.max(q))
    ex = np.exp((q - qmax) / tau)
    s = float(np.sum(ex))
    F = dx * (qmax + tau * math.log(s))
    p = ex / s
    conv1 = np.convolve(1.0 - h, p, mode='full')
    conv2 = np.convolve(h, p[::-1], mode='full')
    sl = slice(n - 1, 2 * n - 2)
    grad = dx * (conv1[sl] - conv2[sl])
    if not np.all(np.isfinite(grad)) or not np.isfinite(F):
        return F, None
    return F, grad


def mass_fun(h):
    return math.fsum(h) - len(h) / 2.0


def mass_jac(h):
    return np.ones(len(h))


def make_fallback(n):
    h = np.zeros(n, dtype=np.float64)
    h[:n // 2] = 1.0
    if n % 2 == 1:
        h[n // 2] = 0.5
    return h


def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    n = 95
    parent = None
    try:
        pv = np.asarray(initial_h_values, dtype=np.float64)
        if pv.ndim == 1 and 2 <= pv.shape[0] < 1000:
            n = pv.shape[0]
            if np.all(np.isfinite(pv)):
                parent = pv.copy()
    except (NameError, TypeError, ValueError):
        pass
    if parent is None:
        try:
            np2 = int(kwargs['n_points'])
            if 2 <= np2 < 1000:
                n = np2
        except (TypeError, ValueError, KeyError):
            pass
    dx = 2.0 / n
    m = n / 2.0

    if parent is not None:
        if is_feasible(parent, n, m):
            base = parent.copy()
        else:
            base = project_capped_simplex(parent, n, m)
            if base is None:
                base = make_fallback(n)
    else:
        base = make_fallback(n)

    starts = [base.copy()]
    for sd in (0.05, 0.15, 0.30):
        noisy = base + sd * rng.standard_normal(n)
        pr = project_capped_simplex(noisy, n, m)
        starts.append(pr if pr is not None else base.copy())
    u = rng.uniform(0.0, 1.0, n)
    pr = project_capped_simplex(u, n, m)
    starts.append(pr if pr is not None else base.copy())
    bv = rng.beta(0.7, 0.7, n)
    pr = project_capped_simplex(bv, n, m)
    starts.append(pr if pr is not None else base.copy())
    phase = rng.uniform(0.0, 2.0 * math.pi)
    idx = np.arange(n, dtype=np.float64)
    sv = 0.5 + 0.45 * np.sin(2.0 * math.pi * idx / n + phase)
    pr = project_capped_simplex(sv, n, m)
    starts.append(pr if pr is not None else base.copy())
    perm = rng.permutation(n)
    pv2 = base[perm]
    pr = project_capped_simplex(pv2, n, m)
    starts.append(pr if pr is not None else base.copy())

    best_h = base.copy()
    best_c = exact_score(best_h)
    for sv in starts:
        c = exact_score(sv)
        if c < best_c:
            best_c = c
            best_h = sv.copy()

    tau_scale = n / 95.0
    taus = [0.10 * tau_scale, 0.03 * tau_scale, 0.008 * tau_scale,
            0.002 * tau_scale, 0.0005 * tau_scale]
    bounds = [(0.0, 1.0)] * n
    constraint = {'type': 'eq', 'fun': mass_fun, 'jac': mass_jac}

    for si in range(len(starts)):
        current = starts[si].copy()
        for tau in taus:
            try:
                res = minimize(smooth_obj, current, args=(tau, dx),
                               method='SLSQP', jac=True, bounds=bounds,
                               constraints=[constraint],
                               options={'maxiter': 120, 'ftol': 1e-9,
                                          'disp': False})
                if res.x is not None and len(res.x) == n and \
                        np.all(np.isfinite(res.x)):
                    proj = project_capped_simplex(res.x, n, m)
                    if proj is not None:
                        F_cur, _ = smooth_obj(current, tau, dx)
                        F_new, _ = smooth_obj(proj, tau, dx)
                        if F_new is not None and np.isfinite(F_new) and \
                                F_new <= F_cur + 1e-10:
                            current = proj.copy()
                        c = exact_score(current)
                        if c < best_c - 1e-12:
                            best_c = c
                            best_h = current.copy()
            except Exception:
                break

    if not is_feasible(best_h, n, m):
        fixed = project_capped_simplex(best_h, n, m)
        best_h = fixed if fixed is not None else make_fallback(n)
    best_c = exact_score(best_h)
    return best_h, best_c, int(n)
