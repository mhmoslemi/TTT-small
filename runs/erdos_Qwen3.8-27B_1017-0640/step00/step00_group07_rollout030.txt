import numpy as np

def is_feasible(h, n):
    if h.ndim != 1 or h.shape[0] != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-15) or np.any(h > 1.0 + 1e-15):
        return False
    mass_eff = n / 2.0
    if abs(float(np.sum(h)) - mass_eff) > 1e-10 * max(1, n):
        return False
    return True


def project_capped(y, n):
    m = n / 2.0
    if y.ndim != 1 or y.shape[0] != n:
        return np.full(n, 0.5)

    lo = np.min(yy) - 1.0
    hi = np.max(yy)

        s = np.sum(np.clip(yy - mid, 0.0, 1.0))
        if s > m:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    p = np.clip(yy - lam, 0.0, 1.0)
    r = m - float(np.sum(p))

            break

                r += sub
                if abs(r) <= 2e-14 * max(1, n):
                    break
    if not is_feasible(p, n):
        return np.full(n, 0.5)
    return p


def exact_score(h):
    n = len(h)
    if n < 2 or n > 999:
        return float('inf')
    if not is_feasible(h, n):
        return float('inf')
    dx = 2.0 / n
    try:
        corr = np.correlate(h, 1.0 - h, mode='full')
        if not np.all(np.isfinite(corr)):
            raise ValueError
        return float(np.max(corr) * dx)
    except Exception:
        try:
            q0 = dx * float(np.sum(h * (1.0 - h)))
            terms = [q0]
            one_vec = 1.0 - h
            for s in range(1, n):
                a = dx * float(np.dot(h[:n-s], one_vec[s:]))
                b = dx * float(np.dot(h[s:], one_vec[:n-s]))
                terms.append(a)
                terms.append(b)
            return float(max(terms))
        except Exception:
            return float('inf')

            y += a * np.cos(2.0 * math.pi * k * t) + b * np.sin(2.0 * math.pi * k * t)
    return y


def make_balanced_binary(n, rng):

    half = n // 2
    y[idx[:half]] = 1.0
