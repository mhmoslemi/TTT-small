    elif n <= 128: P, mg, sl, mr = 32, 140, 28, 2
    elif n <= 256: P, mg, sl, mr = 20, 70, 18, 1
    elif n <= 512: P, mg, sl, mr = 12, 35, 12, 1
    else: P, mg, sl, mr = 8, 12, 6, 0
    fb_score = exact_score(parent, n) or 1.0
    fb = parent.copy()
    pop = build_population(n, rng, parent, P)
    try:
        incumbent, inc_score = jade_search(pop, n, rng, mg, P, sl, mr)
    except Exception:
        incumbent, inc_score = fb, fb_score
    x = project_capped_simplex(incumbent)
    if x is None or not check_feasibility(x, n):
        x = fb
    if not check_feasibility(x, n):
        x = np.full(n, 0.5)
    fs = exact_score(x, n)
    if fs is None:
        try: fs = float(np.max(np.correlate(x, 1.0 - x, mode='full'))) * (2.0 / n)
        except Exception: fs = 1.0
    if fb_score < fs:
        x, fs = fb, fb_score
    return x, float(fs), int(n)
