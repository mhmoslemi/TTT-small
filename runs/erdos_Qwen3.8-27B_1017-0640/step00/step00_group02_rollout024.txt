        if lo == hi:
            break
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    for _ in range(3):
        r = S - np.sum(x)
        if abs(r) <= 1e-12:
            break
        if r > 0:
            mask = x < 1.0
            if not np.any(mask):
                break
            adj = np.minimum(r, 1.0 - x)
            x[mask] += adj[mask]
        else:
            mask = x > 0.0
            if not np.any(mask):
                break
            adj = np.minimum(-r, x)
            x[mask] -= adj[mask]
    x = np.clip(x, 0.0, 1.0)
    if abs(np.sum(x) - S) > 1e-10:
        return None
    if np.any(x < 0.0) or np.any(x > 1.0):
        return None
    return x

def exact_score(h, n):
    if len(h) != n or not np.all(np.isfinite(h)):
        return None
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return None
    if abs(np.sum(h) - 0.5 * n) > 1e-9:
        return None
    c = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)
    if not np.isfinite(c):
        return None
    return c

def smooth_eval(h, tau, with_gradient, n):
    dx = 2.0 / n
    q = np.correlate(h, 1.0 - h, mode='full')
    c = q * dx
    cmax = float(np.max(c))
    u = (c - cmax) / tau
    eu = np.exp(u)
    su = float(np.sum(eu))
    if not np.isfinite(su) or su <= 0.0:
        return None, None
    F = cmax + tau * np.log(su)
    if not with_gradient:
        return F, None
    w = eu / su
    g = w[n - 1] * (1.0 - 2.0 * h)
    for s in range(1, n):
        m = n - s
        g[s:] += w[n - 1 + s] * (1.0 - h[:m])
        g[:m] -= w[n - 1 + s] * h[s:]
        g[:m] += w[n - 1 - s] * (1.0 - h[s:])
        g[s:] -= w[n - 1 - s] * h[:m]
    g *= dx
    if not np.all(np.isfinite(g)):
        return None, None
    return F, g

def local_search(start, n, iter_cap):
    S = 0.5 * n
    temps = [0.02, 0.005, 0.001, 0.0003, 0.0001]
    x = start.copy()
    best_x = x.copy()
    best_s = exact_score(x, n)
    if best_s is None:
        best_s = 1.0
    for tau in temps:
        alpha = 1.0
        for _ in range(iter_cap):
            F, g = smooth_eval(x, tau, True, n)
            if F is None:
                break
            accepted = False
            y = None
            d = None
            for _ in range(20):
                y = project_capped(x - alpha * g, S)
                if y is None:
                    break
                d = y - x
                Fy, _ = smooth_eval(y, tau, False, n)
                if Fy is None:
                    break
                rhs = F + np.dot(g, d) + np.dot(d, d) / (2.0 * alpha) + 1e-12 * (1.0 + abs(F))
                if Fy <= rhs:
                    accepted = True
                    break
                alpha *= 0.5
            if not accepted:
                break
            x = y
            if np.linalg.norm(d) / alpha <= 1e-7:
                break
            alpha = min(1.5 * alpha, 4.0)
            s = exact_score(x, n)
            if s is not None and s < best_s - 1e-12:
                best_s = s
                best_x = x.copy()
    return best_x, best_s

def run(seed=42, **kwargs):
    n = 45
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if len(p) >= 2:
            n = len(p)
    except Exception:
        p = None
    S = 0.5 * n
    fallback = np.full(n, 0.5, dtype=np.float64)
    incumbent = fallback.copy()
    incumbent_score = exact_score(incumbent, n)
    if incumbent_score is None:
        incumbent_score = 0.5
    try:
        pp = np.array(p, dtype=np.float64).ravel()
        if len(pp) == n and np.all(np.isfinite(pp)):
            if np.all(pp >= 0.0) and np.all(pp <= 1.0) and abs(np.sum(pp) - S) <= 1e-10:
                sc = exact_score(pp, n)
                if sc is not None:
                    incumbent = pp.copy()
                    incumbent_score = sc
            else:
                rp = project_capped(pp, S)
                if rp is not None:
                    sc = exact_score(rp, n)
                    if sc is not None:
                        incumbent = rp
                        incumbent_score = sc
    except Exception:
        pass
    rng = np.random.default_rng(seed)
    idx = np.arange(n, dtype=np.float64)
    starts = [incumbent.copy()]
    if n <= 128:
        gs_list = [0.04, 0.12, 0.30]
        n_sin = 2
    elif n <= 512:
        gs_list = [0.08, 0.25]
        n_sin = 1
    else:
        gs_list = [0.12]
        n_sin = 0
    for gs in gs_list:
        noise = rng.normal(0.0, gs, n)
        pr = project_capped(incumbent + noise, S)
        if pr is not None:
            starts.append(pr)
    uu = rng.uniform(0.0, 1.0, n)
    pr = project_capped(uu, S)
    if pr is not None:
        starts.append(pr)
    for m in range(1, n_sin + 1):
        ph = rng.uniform(0.0, 2.0 * np.pi)
        sv = 0.5 + 0.35 * np.sin(2.0 * np.pi * m * idx / n + ph)
        pr = project_capped(sv, S)
        if pr is not None:
            starts.append(pr)
    deduped = [starts[0]]
    for s in starts[1:]:
        if all(float(np.max(np.abs(s - d))) >= 1e-12 for d in deduped):
            deduped.append(s)
    starts = deduped
    iter_cap = max(15, min(80, 3600 // n))
    for st in starts:
        sc = exact_score(st, n)
        if sc is not None and sc < incumbent_score - 1e-12:
            incumbent = st.copy()
            incumbent_score = sc
        try:
            bx, bs = local_search(st, n, iter_cap)
            if bs < incumbent_score - 1e-12:
                incumbent = bx.copy()
                incumbent_score = bs
        except Exception:
            pass
    if n <= 128:
        rs_list = [0.05, 0.015]
    elif n <= 512:
        rs_list = [0.04, 0.015]
    else:
        rs_list = [0.03]
    for rs in rs_list:
        for _ in range(2):
            noise = rng.normal(0.0, rs, n)
            pr = project_capped(incumbent + noise, S)
            if pr is None:
                continue
            sc = exact_score(pr, n)
            if sc is not None and sc < incumbent_score - 1e-12:
                incumbent = pr.copy()
                incumbent_score = sc
            try:
                bx, bs = local_search(pr, n, iter_cap)
                if bs < incumbent_score - 1e-12:
                    incumbent = bx.copy()
                    incumbent_score = bs
            except Exception:
                pass
    fs = exact_score(incumbent, n)
    if fs is None:
        incumbent = fallback
        fs = exact_score(fallback, n)
        if fs is None:
            fs = 0.5
    return incumbent.astype(np.float64), float(fs), int(n)
