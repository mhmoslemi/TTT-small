- For r in [s, n-1]: G[n-1+s, r] += (1 - h[r-s])

G[n-1-s, :n-s] += (1.0 - h[s:])

import numpy as np
from scipy.optimize import minimize

def project_capped(v, S):
    v = np.clip(np.asarray(v, dtype=np.float64), -1e6, 1e6)
    lo, hi = np.min(v) - 1.0, np.max(v) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if mid == lo or mid == hi:
            break
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    res = S - np.sum(x)
    if res > 0:
        room = np.maximum(1.0 - x, 1e-15)
        x += res * room / np.sum(room)
    elif res < 0:
        cap = np.maximum(x, 1e-15)
        x += res * cap / np.sum(cap)
    x = np.clip(x, 0.0, 1.0)
    res = S - np.sum(x)
    if abs(res) > 1e-14:
        idx = int(np.argmin(np.abs(x - 0.5)))
        x[idx] = float(np.clip(x[idx] + res, 0.0, 1.0))
    if np.any(x < -1e-12) or np.any(x > 1 + 1e-12) or abs(np.sum(x) - S) > 1e-10:
        return None
    return x

def corr_q(h, n):
    m = 2 * n - 1
    q = np.zeros(m)
    q[n - 1] = np.dot(h, 1.0 - h)
    for s in range(1, n):
        m_ = n - s
        q[n - 1 + s] = np.dot(h[s:], 1.0 - h[:m_])
        q[n - 1 - s] = np.dot(h[:m_], 1.0 - h[s:])
    return q

def corr_G(h, n):
    m = 2 * n - 1
    G = np.zeros((m, n))
    G[n - 1, :] = 1.0 - 2.0 * h
    for s in range(1, n):
        m_ = n - s
        G[n - 1 + s, s:] += (1.0 - h[:m_])
        G[n - 1 + s, :m_] -= h[s:]
        G[n - 1 - s, :m_] += (1.0 - h[s:])
        G[n - 1 - s, s:] -= h[:m_]
    return G

def exact_score(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def obj_fun(z, n, dx, S):
    return z[-1]

def obj_jac(z, n, dx, S):
    j = np.zeros(len(z))
    j[-1] = 1.0
    return j

def ineq_fun(z, n, dx, S):
    h = z[:n]
    return z[-1] - dx * corr_q(h, n)

def ineq_jac(z, n, dx, S):
    h = z[:n]
    G = corr_G(h, n)
    m = 2 * n - 1
    J = np.zeros((m, n + 1))
    J[:, :n] = -dx * G
    J[:, n] = 1.0
    return J

def eq_fun(z, n, dx, S):
    return np.sum(z[:n]) - S

def eq_jac(z, n, dx, S):
    j = np.zeros(n + 1)
    j[:n] = 1.0
    return j

def try_slsqp(h0, n, dx, S):
    sc = exact_score(h0, n)
    z0 = np.concatenate([h0, [min(max(sc, 0.0), 2.0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    args = (n, dx, S)
    cons = [
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]
    try:
        res = minimize(obj_fun, z0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons, args=args,
                       options={'maxiter': 400, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return None
    if res.x is None or len(res.x) != n + 1 or not np.all(np.isfinite(res.x)):
        return None
    h_cand = project_capped(res.x[:n], S)
    if h_cand is None:
        return None
    return h_cand

                        p = project_capped(p, S)
                    if p is not None:
                        parent = p
                else:
                    p2 = project_capped(p, S)
                    if p2 is not None:
                        parent = p2
        except Exception:
            pass
    rng = np.random.default_rng(seed)
    if parent is not None:
        starts = [parent.copy()]
    else:
        starts = []
    starts.append(np.full(n, 0.5))
    base = parent if parent is not None else np.full(n, 0.5)
    sigmas = [0.03, 0.08, 0.16, 0.30, 0.08, 0.20]
    for sg in sigmas:
        p = project_capped(base + sg * rng.standard_normal(n), S)
        if p is not None:
            starts.append(p)
    for _ in range(2):
        p = project_capped(rng.uniform(0, 1, n), S)
        if p is not None:
            starts.append(p)

        x = np.arange(n)
        p = project_capped(0.5 + 0.35 * np.sin(2 * np.pi * m_ * x / n + ph), S)
        if p is not None:
            starts.append(p)
    seen = []
    unique_starts = []
    for s in starts:
        dup = False
        for u in seen:
            if np.linalg.norm(s - u) < 1e-12:
                dup = True
                break
        if not dup:
            seen.append(s)
            unique_starts.append(s)
    best_h = None
    best_sc = np.inf
    if parent is not None:

        best_sc = exact_score(parent, n)

                best_sc = sc

                if sc2 < best_sc - 1e-12:
                    best_h = cand.copy()
                    best_sc = sc2
                    base_h = best_h.copy()
    if best_h is None:
        best_h = np.full(n, 0.5)
        best_sc = exact_score(best_h, n)
    else:
        final = project_capped(best_h, S)
        if final is not None:
            best_h = final
            best_sc = exact_score(best_h, n)
    return (best_h, float(best_sc), n)
