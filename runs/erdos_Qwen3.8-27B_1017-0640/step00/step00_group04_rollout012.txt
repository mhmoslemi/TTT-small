            return

            return
        for s in starts:
            if np.max(np.abs(p - s)) < 1e-9:
                return
        starts.append(p)
    for a in anchors:
        try_add(a.copy())

        try_add(a + 0.15 * rng.standard_normal(n))
        try_add(a[::-1])
        try_add(1.0 - a)

    try_add(blk)
    try_add(np.full(n, 0.5))
    try_add(0.5 + 0.25 * np.cos(2.0 * np.pi * np.arange(n) / n))
    scales = [0.05, 0.15, 0.30, 0.40]

    fallback = parent.copy()
    fallback_sc = exact_score(fallback)
    inc_h = fallback.copy()
    inc_sc = fallback_sc
    # Native search
    if n0 <= 128:
        anchors = [parent.copy()]
        starts = build_starts(n0, anchors, rng, n_starts)
        if not starts:
            starts = [fallback.copy()]
        for st in starts:
            dh, dsc = anneal_pgd(st, n0, max_iter)
            if np.isfinite(dsc) and (dsc < inc_sc - 1e-12 or (abs(dsc - inc_sc) <= 1e-12 and n0 < len(inc_h))):
                inc_h = dh.copy(); inc_sc = dsc
        if n0 <= 128:
            nh_sc = inc_sc
    else:
        nh_sc = fallback_sc
    # Doubled search
    if use_doubled and n0 <= 64:
        n1 = 2 * n0
        anchors_hi = [np.repeat(parent, 2)]
        if len(inc_h) == n0:
            anchors_hi.append(np.repeat(inc_h, 2))
        elif len(inc_h) == n1:
            anchors_hi.append(inc_h.copy())
        starts_hi = build_starts(n1, anchors_hi, rng, n_starts)
        if not starts_hi:
            starts_hi = [np.repeat(fallback, 2)]
        for st in starts_hi:
            dh, dsc = anneal_pgd(st, n1, max_iter)
            if np.isfinite(dsc) and (dsc < inc_sc - 1e-12 or (abs(dsc - inc_sc) <= 1e-12 and n1 < len(inc_h))):
                inc_h = dh.copy(); inc_sc = dsc
    # Final validation
    nf = len(inc_h)
    if not is_feasible(inc_h, nf):
        p = project_capped(inc_h, nf)
        if p is not None and is_feasible(p, nf):
            inc_h = p
        else:
            inc_h = fallback.copy(); nf = len(fallback)
    inc_h = np.clip(inc_h, 0.0, 1.0).astype(np.float64)
    final_sc = exact_score(inc_h)
    if not np.isfinite(final_sc):
        inc_h = fallback.astype(np.float64)
        final_sc = exact_score(inc_h)
    return inc_h, float(final_sc), int(inc_h.shape[0])
