import numpy as np
from scipy.optimize import minimize


def _project_capped(v, n):
    target = n / 2.0
    v = np.clip(v, -1e6, 1e6)
    lo = min(float(v.min()) - 1.0, -1.0)
    hi = max(float(v.max()), 2.0)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        q = np.clip(v - mid, 0.0, 1.0)
        if q.sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    residual = target - h.sum()
    if residual > 1e-13:
        order = np.argsort(1.0 - h)
        for i in order:
            cap = 1.0 - h[i]
            if cap <= 0.0:
                continue
            add = min(residual, cap)
            h[i] += add
            residual -= add
            if abs(residual) < 1e-13:
                break
    elif residual < -1e-13:
        order = np.argsort(h)
        for i in order:
            if h[i] <= 0.0:
                continue
            sub = min(-residual, h[i])
            h[i] -= sub
            residual += sub
            if abs(residual) < 1e-13:
                break
    return h


def _score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    if c.shape[0] != 2 * n - 1:
        return np.inf
    if not np.all(np.isfinite(c)):
        return np.inf
    return float(np.max(c))


def _lag_jac(h):
    n = len(h)
    dx = 2.0 / n
    J = np.zeros((2 * n - 1, n))
    for d in range(n):
        j = n - 1 + d
        if d == 0:
            J[j, :] = dx * (1.0 - 2.0 * h)
        else:
            J[j, d:] += dx * (1.0 - h[:n - d])
            J[j, :n - d] -= dx * h[d:]
    for d in range(1, n):
        j = n - 1 - d
        J[j, :n - d] += dx * (1.0 - h[d:])
        J[j, d:] -= dx * h[:n - d]
    return J


def _lag_values(h):
    n = len(h)
    dx = 2.0 / n
    c = np.zeros(2 * n - 1)
    for d in range(n):
        j = n - 1 + d
        u = h[:n - d]
        v = h[d:]
        c[j] = np.sum(v * (1.0 - u)) * dx
    for d in range(1, n):
        j = n - 1 - d
        u = h[:n - d]
        v = h[d:]
        c[j] = np.sum(u * (1.0 - v)) * dx
    return c


def _obj(z):
    return z[-1]


def _obj_jac(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g


def _eq(z):
    return z[:-1].sum() - (len(z) - 1) / 2.0


def _eq_jac(z):
    j = np.zeros(len(z))
    j[:-1] = 1.0
    return j


def _ineq(z):
    n = len(z) - 1
    h = z[:n]
    c = _lag_values(h)
    return z[-1] - c


def _ineq_jac(z):
    n = len(z) - 1
    h = z[:n]
    J = _lag_jac(h)
    col = np.ones(J.shape[0])
    return np.column_stack((-J, col))


def _solve_start(h0, n):
    t0 = _score(h0) + 1e-8
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'eq', 'fun': _eq, 'jac': _eq_jac},
        {'type': 'ineq', 'fun': _ineq, 'jac': _ineq_jac},
    ]
    maxiter = 250 if n <= 128 else 120
    try:
        res = minimize(_obj, z0, method='SLSQP', jac=_obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-10, 'disp': False})
    except Exception:
        return None
    if not hasattr(res, 'x') or res.x is None:
        return None
    x = np.asarray(res.x, dtype=np.float64)
    if x.size != n + 1 or not np.all(np.isfinite(x)):
        return None
    hp = _project_capped(x[:n], n)
    if not np.all(np.isfinite(hp)):
        return None
    if abs(hp.sum() - n / 2.0) > 1e-10:
        return None
    return hp
