def compute_q_and_grad(h, n, tau):
    dx = 2.0 / n
    q = np.empty(2*n - 1)
    # q[0] = q0
    q[0] = dx * np.sum(h * (1 - h))
    # q[1] = A_1, q[2] = B_1, ..., q[2s-1] = A_s, q[2s] = B_s
    for s in range(1, n):
        q[2*s-1] = dx * np.sum(h[:n-s] * (1 - h[s:]))
        q[2*s] = dx * np.sum(h[s:] * (1 - h[:n-s]))
    # log-sum-exp
    qmax = np.max(q)
    exp_q = np.exp((q - qmax) / tau)
    sum_exp = np.sum(exp_q)
    w = exp_q / sum_exp
    phi = qmax + tau * np.log(sum_exp)
    # gradient
    grad = np.zeros(n)
    # zero lag contribution
    grad += w[0] * dx * (1 - 2*h)
    for s in range(1, n):
        n_s = n - s
        # A_s gradient: w[2s-1] * dx * (1-h[s:])[:n_s] for indices 0..n_s-1, -dx*h[:n_s] for indices s..n-1

        wb = w[2*s] * dx
        grad[:n_s] += -wb * h[s:s+n_s]
        grad[s:] += wb * (1 - h[:n_s])
    return q, phi, grad

        final_score = exact_score(best)
    return (best, float(final_score), int(n))
