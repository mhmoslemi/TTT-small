from scipy.optimize import linprog
from scipy.sparse import csr_matrix, hstack, vstack, diags

    if q.size != 2 * len(h) - 1 or not np.all(np.isfinite(q)):
        return None
    return float(np.max(q))

def check_feasible(h, N, m):
    return (h.ndim == 1 and len(h) == N and np.all(np.isfinite(h))
            and np.all(h >= 0) and np.all(h <= 1) and abs(np.sum(h) - m) <= 1e-10)

def project_capped(v, m):
    v = np.asarray(v, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return None
    v = np.clip(v, -1e6, 1e6)
    N = len(v)
    lo, hi = np.min(v) - 1.0, np.max(v)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > m:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

        if abs(r) < 1e-14:
            break

                r -= add
            else:

    if not np.all(np.isfinite(y)) or abs(np.sum(y) - m) > 1e-10:

    y = np.clip(y, 0.0, 1.0)
    return y

def compute_lags_grad(h, dx):
    N = len(h)
    q = np.correlate(h, 1.0 - h, mode="full") * dx
    L = 2 * N - 1
    G = np.zeros((L, N))
    ks = np.arange(-(N - 1), N)
    for r in range(N):
        ic = r - ks  # r-k for all k
        id2 = r + ks  # r+k for all k
        m1 = (ic >= 0) & (ic < N)
        m2 = (id2 >= 0) & (id2 < N)
        tmp = np.zeros(L)
        tmp[m1] = 1.0 - h[ic[m1]]
        tmp[m2] -= h[id2[m2]]
        G[:, r] = dx * tmp
    if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):
        return None, None
    return q, G

def solve_lp(h, q, G, delta, dx):
    N = len(h)
    L, _ = G.shape
    nvar = 2 * N + 1
    c = np.zeros(nvar)
    c[-1] = 1.0
    # Inequality: G@d - t <= -q  -> [G, 0, -1] [d;u;t] <= -q
    A1 = csr_matrix(hstack([csr_matrix(G), csr_matrix((L, N), (L, N)),
                            (-np.ones(L)).reshape(-1, 1)]))
    b1 = -q
    # d <= u: I*d - I*u <= 0
    A2 = csr_matrix(hstack([csr_matrix(np.eye(N)), csr_matrix(-np.eye(N)),
                            csr_matrix((N, 1))]))
    b2 = np.zeros(N)
    # -d - u <= 0
    A3 = csr_matrix(hstack([csr_matrix(-np.eye(N)), csr_matrix(-np.eye(N)),
                            csr_matrix((N, 1))]))
    b3 = np.zeros(N)
    # sum(u) <= delta
    A4 = csr_matrix(hstack([csr_matrix((1, N)), csr_matrix(np.ones((1, N))),
                            csr_matrix((1, 1))]))
    b4 = np.array([delta])
    A_ub = vstack([A1, A2, A3, A4])
    b_ub = np.concatenate([b1, b2, b3, b4])
    A_eq = csr_matrix(hstack([csr_matrix(np.ones((1, N))), csr_matrix((1, N + 1))]))
    b_eq = np.array([0.0])
    lbs = np.concatenate([-h, np.zeros(N), [-np.inf]])
    ubs = np.concatenate([1.0 - h, np.full(N, np.inf), [np.inf]])
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=list(zip(lbs, ubs)), method="highs",
                  options={"presolve": True, "maxiter": 500,
                           "primal_feasibility_tolerance": 1e-9,
                           "dual_feasibility_tolerance": 1e-9})
    if res.status != 0 or res.x is None or len(res.x) != nvar:
        return None
    if not np.all(np.isfinite(res.x)):
        return None
    return res.x[:N]

def local_search(h0, dx, m, n_iters, delta0):

    h = h0.copy()
    best = h.copy()
    best_s = evaluate_exact(h, dx)
    delta = delta0
    fails = 0
    for _ in range(n_iters):

        if q is None:
            break
        d = solve_lp(h, q, G, delta, dx)
        if d is None:

            fails += 1

            continue

            continue

            delta /= 2.0
            continue
        s_trial = evaluate_exact(h_prop, dx)
        if s_trial is None:
            delta /= 2.0
            continue
        C_cur = evaluate_exact(h, dx)

        C_model = float(np.max(q + G @ d_act))
        p = C_cur - C_model
        a = C_cur - s_trial
        if a > 1e-12 * max(1.0, abs(C_cur)):
            h = h_prop
            if s_trial < best_s:

                best = h.copy()
            if p <= 1e-12:
                delta /= 2.0

                    delta = min(2.0 * delta, 4.0)
        else:
            delta /= 2.0
