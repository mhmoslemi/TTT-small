    if not check_feasible(h, n):
        return float('inf')
    corr = np.correlate(h, 1.0 - h, mode='full')
    if len(corr) != 2 * n - 1 or not np.all(np.isfinite(corr)):
        return float('inf')
    s = float(np.max(corr * (2.0 / n)))
    return s if np.isfinite(s) else float('inf')

def compute_lags_jac(h, n):
    dx = 2.0 / n
    c = np.zeros(2 * n - 1)
    J = np.zeros((2 * n - 1, n))
    for d in range(n):
        nu = n - d
        u = h[:nu]
        v = h[d:]
        A_d = float(np.sum(v * (1.0 - u)))
        B_d = float(np.sum(u * (1.0 - v)))
        ra = n - 1 + d
        c[ra] = A_d * dx
        J[ra, d:] += dx * (1.0 - u)
        J[ra, :nu] -= dx * v
        if d > 0:
            rb = n - 1 - d
            c[rb] = B_d * dx
            J[rb, :nu] += dx * (1.0 - v)
            J[rb, d:] -= dx * u
    if not np.all(np.isfinite(c)) or not np.all(np.isfinite(J)):
        return None, None
    return c, J

def solve_lp(y, c, J, r, mu, n):
    m = 2 * n + 1
    c_obj = np.zeros(m)
    c_obj[n:2 * n] = mu
    c_obj[2 * n] = 1.0
    nl = 2 * n - 1
    nr = 4 * n - 1
    A_ub = np.zeros((nr, m))
    b_ub = np.zeros(nr)
    for l in range(nl):
        A_ub[l, :n] = J[l]
        A_ub[l, 2 * n] = -1.0
        b_ub[l] = float(J[l] @ y) - c[l]
    row = nl
    for i in range(n):
        A_ub[row, i] = 1.0
        A_ub[row, n + i] = -1.0
        b_ub[row] = y[i]
        row += 1
        A_ub[row, i] = -1.0
        A_ub[row, n + i] = -1.0
        b_ub[row] = -y[i]
        row += 1
    A_eq = np.zeros((1, m))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = []
    for i in range(n):
        bounds.append((max(0.0, y[i] - r), min(1.0, y[i] + r)))
    for _ in range(n):
        bounds.append((0.0, None))
    bounds.append((None, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'max_iter': 500})
    except Exception:
        return None
    if res is None or res.status != 0 or res.x is None:
        return None
    if not np.all(np.isfinite(res.x)):
        return None
    return res.x[:n]

def local_search(y, n):
    r = 0.04
    r_min, r_max = 2e-4, 0.12
    mu = 5e-4
    failures = 0
    score = exact_score(y, n)
    for _ in range(22):
        if failures >= 6 or r <= r_min:
            break
        c, J = compute_lags_jac(y, n)
        if c is None:
            failures += 1
            r = max(r * 0.5, r_min)
            continue
        xp = solve_lp(y, c, J, r, mu, n)
        if xp is None:
            failures += 1
            r = max(r * 0.5, r_min)
            continue
        xp = joint_project(xp, n)
        if xp is None:
            failures += 1
            r = max(r * 0.5, r_min)
            continue
        disp = float(np.max(np.abs(xp - y)))
        if disp > r + 1e-6 or disp <= 1e-10:
            failures += 1
            r = max(r * 0.5, r_min)
            continue
        best_trial, best_ts = None, score
        for j in range(9):
            a = 2.0 ** (-j)
            tr = y + a * (xp - y)
            tr = joint_project(tr, n)
            if tr is None:
                continue
            if float(np.max(np.abs(tr - y))) > a * r + 1e-6:
                continue
            ts = exact_score(tr, n)
            if ts < best_ts - 1e-12:
                best_ts, best_trial = ts, tr
        if best_trial is None:
            failures += 1
            r = max(r * 0.5, r_min)
            continue
        step_inf = float(np.max(np.abs(best_trial - y)))
        y = best_trial
        score = best_ts
        failures = 0
        if step_inf > 0.8 * r and r < r_max:
            r = min(1.5 * r, r_max)
        if step_inf <= 1e-8:
            break
    return y, score

def dedup_starts(starts, new, n):
    if new is None:
        return starts
    pr = joint_project(new, n)
    if pr is None:
        return starts
    for s in starts:
        if float(np.max(np.abs(pr - s))) < 1e-10:
            return starts
    return starts + [pr]

def run(seed=42, **kwargs):
    n = 70
    parent_val = globals().get('initial_h_values')
    use_parent = False
    if parent_val is not None:
        try:
            p_tmp = np.asarray(parent_val, dtype=np.float64)
            if p_tmp.ndim == 1 and len(p_tmp) >= 2:
                n = len(p_tmp)
                use_parent = True
        except Exception:
            pass
    if not use_parent:
        nv = kwargs.get('n_points', None)
        if nv is not None:
            try:
                if not isinstance(nv, bool) and 2 <= int(nv) <= 256:
                    n = int(nv)
            except Exception:
                pass
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    uniform = np.full(n, 0.5)
    parent_vec = None
    if use_parent:
        try:
            p = np.asarray(parent_val, dtype=np.float64).copy()
            if p.ndim == 1 and len(p) == n and np.all(np.isfinite(p)):
                if check_feasible(p, n):
                    parent_vec = p.copy()
                else:
                    parent_vec = joint_project(p, n)
        except Exception:
            pass
    best_h = uniform.copy()
    best_score = exact_score(uniform, n)
    if parent_vec is not None:
        ps = exact_score(parent_vec, n)
        if ps < best_score - 1e-12:
            best_h, best_score = parent_vec.copy(), ps
    if n > 128:
        return best_h, best_score, int(n)
    starts = []
    if parent_vec is not None:
        starts = dedup_starts(starts, parent_vec, n)
        starts = dedup_starts(starts, parent_vec[::-1].copy(), n)
        starts = dedup_starts(starts, 1.0 - parent_vec, n)
    starts = dedup_starts(starts, uniform, n)
    idx = np.arange(n, dtype=np.float64)
    starts = dedup_starts(starts, 0.5 + 0.30 * np.sin(2.0 * np.pi * (idx + 0.5) / n), n)
    if parent_vec is not None:
        starts = dedup_starts(starts, parent_vec + 0.08 * rng.standard_normal(n), n)
    else:
        starts = dedup_starts(starts, 0.5 + 0.20 * np.cos(4.0 * np.pi * (idx + 0.5) / n), n)
        starts = dedup_starts(starts, 0.5 + 0.20 * rng.standard_normal(n), n)
        starts = dedup_starts(starts, rng.random(n), n)
    starts = starts[:6]
    for st in starts:
        try:
            s0 = exact_score(st, n)
            if s0 < best_score - 1e-12:
                best_h, best_score = st.copy(), s0
            y_new, s_new = local_search(st.copy(), n)
            if s_new < best_score - 1e-12:
                best_h, best_score = y_new.copy(), s_new
        except Exception:
            continue
    if not check_feasible(best_h, n):
        if parent_vec is not None and check_feasible(parent_vec, n):
            best_h = parent_vec.copy()
        else:
            best_h = uniform.copy()
    fs = exact_score(best_h, n)
    if not np.isfinite(fs):
        best_h = uniform.copy()
        fs = exact_score(best_h, n)
    return best_h, fs, int(n)
