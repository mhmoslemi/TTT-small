import numpy as np

def project_to_feasible(y, m):
    n = len(y)
    if n == 0:
        return None
    y = np.clip(y.astype(np.float64).copy(), -1e6, 1e6)
    if not np.all(np.isfinite(y)):
        return np.full(n, m / n)
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > m:
            lo = mid
        else:
            hi = mid
    result = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(10):
        r = m - float(np.sum(result))
        if abs(r) < 1e-12:
            break
        if r > 0:
            slack = 1.0 - result
            idx = int(np.argmax(slack))
            result[idx] += min(r, slack[idx])
        else:
            idx = int(np.argmin(result))
            result[idx] -= min(-r, result[idx])
    result = np.clip(result, 0.0, 1.0)
    if abs(float(np.sum(result)) - m) > 1e-9:
        return np.full(n, m / n)
    return result

def exact_score(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

def check_feasible(h, n, m):
    if h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-9) or np.any(h > 1.0 + 1e-9):
        return False
    return abs(float(np.sum(h)) - m) <= 1e-9

def smooth_surrogate(q, tau):
    qmax = float(np.max(q))
    e = np.exp((q - qmax) / tau)
    s = float(np.sum(e))
    if s <= 0 or not np.isfinite(s):
        return None, None
    F = qmax + tau * math.log(s)
    w = e / s
    return F, w

def compute_grad(h, w, dx, n):
    g = dx * w[n - 1] * (1.0 - 2.0 * h)
    for p in range(1, n):
        wp = w[n - 1 + p]
        wm = w[n - 1 - p]
        if wp > 1e-15:
            g[p:] += dx * wp * (1.0 - h[:-p])
            g[:-p] -= dx * wp * h[p:]
        if wm > 1e-15:
            g[:-p] += dx * wm * (1.0 - h[p:])
            g[p:] -= dx * wm * h[:-p]
    return g

def local_optimize(x0, n, m, dx, best, best_score, rng):

def add_start(starts, seen, v, m):
    v = project_to_feasible(v, m)
    if v is None:
        return
    for s in seen:

def transfer_parent(parent, n):
    ln = len(parent)
    if ln == n:
        return parent.copy()
    xs = np.linspace(0, 1, ln)
    xt = np.linspace(0, 1, n)
    return np.interp(xt, xs, parent).astype(np.float64)
