def lag_values_and_jac(h):
    n = len(h)
    dx = 2.0 / n
    c = np.zeros(2 * n - 1)
    J = np.zeros((2 * n - 1, n))
    idx = 0
    for d in range(n):
        # A_d = sum_{i=0}^{n-d-1} h[i+d] * (1 - h[i])
        L = n - d
        u = h[:L]
        v = h[d:d+L]
        A_d = np.sum(v * (1.0 - u))
        # Jacobian of A_d
        # ∂A_d/∂h_j = (1-h[j-d]) * 1_{d<=j<=n-1} - h[j+d] * 1_{0<=j<=n-d-1}
        row_A = np.zeros(n)
        if L > 0:
            # first term: for j in [d, n-1], contribution (1-h[j-d])
            row_A[d:] += (1.0 - h[:L])  # h[j-d] for j=d..n-1 is h[0..L-1]
            # second term: for j in [0, L-1], contribution -h[j+d]
            row_A[:L] -= h[d:]  # h[j+d] for j=0..L-1 is h[d..n-1]
        c[idx] = A_d * dx
        J[idx] = row_A * dx
        idx += 1
        # B_d for d >= 1
        if d > 0:
            B_d = np.sum(u * (1.0 - v))
            # ∂B_d/∂h_j = (1-h[j+d]) * 1_{0<=j<=n-d-1} - h[j-d] * 1_{d<=j<=n-1}

            row_B[d:] -= h[:L]  # for j in [d,n-1], h[j-d] = h[0..L-1]
            c[idx] = B_d * dx
            J[idx] = row_B * dx
            idx += 1
    return c, J
