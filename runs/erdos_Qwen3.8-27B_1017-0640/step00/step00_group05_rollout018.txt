from scipy.optimize import linprog
from scipy.sparse import csr_matrix, identity, vstack, hstack, diags

def project_capped(v, m):
    v = np.asarray(v, dtype=np.float64).copy()
    v = np.clip(v, -1e6, 1e6)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if float(np.sum(y)) > m:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - float(np.sum(y))
    for _ in range(4):
        if abs(r) < 1e-14:
            break
        if r > 0:
            for i in range(len(y)):
                if r < 1e-14:
                    break
                sp = 1.0 - y[i]
                if sp > 1e-14:
                    a = min(sp, r)
                    y[i] += a
                    r -= a
        else:
            for i in range(len(y)):
                if -r < 1e-14:
                    break
                sp = y[i]
                if sp > 1e-14:
                    a = min(sp, -r)
                    y[i] -= a
                    r += a
    if not np.all(np.isfinite(y)):
        return None
    y = np.clip(y, 0.0, 1.0)
    if abs(float(np.sum(y)) - m) > 1e-10:
        return None
    return y

def exact_score(h, dx):
    N = len(h)
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    if len(q) != 2 * N - 1 or not np.all(np.isfinite(q)):
        return None
    s = float(np.max(q))
    if not np.isfinite(s):
        return None
    return s

def compute_grad(h, dx):
    N = len(h)
    L = 2 * N - 1
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    G = np.zeros((L, N), dtype=np.float64)
    for rlag in range(L):
        k = rlag - (N - 1)
        for r in range(N):
            g = 0.0
            i1 = r - k
            if 0 <= i1 < N:
                g += 1.0 - h[i1]
            i2 = r + k
            if 0 <= i2 < N:
                g -= h[i2]
            G[rlag, r] = dx * g
    return q, G

def feasible(h, N, m):
    if len(h) != N:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.min(h) < -1e-10 or np.max(h) > 1.0 + 1e-10:
        return False
    if abs(float(np.sum(h)) - m) > 1e-10:
        return False
    return True

def solve_lp(h, q, G, dmr):
    N = len(h)
    L = 2 * N - 1
    nvar = 2 * N + 1  # d(N) + u(N) + t
    c = np.zeros(nvar)
    c[-1] = 1.0  # minimize t
    
    # Build A_ub, b_ub
    rows = []
    # Lag constraints: G @ d - t <= -q  → [G | 0 | -1] x <= -q
    A_lag = np.zeros((L, nvar))
    A_lag[:, :N] = G
    A_lag[:, -1] = -1.0
    rows.append(A_lag)
    
    # |d| <= u: d - u <= 0, -d - u <= 0
    A_abs = np.zeros((2 * N, nvar))
    for i in range(N):
        A_abs[2*i, i] = 1.0
        A_abs[2*i, N+i] = -1.0
        A_abs[2*i+1, i] = -1.0
        A_abs[2*i+1, N+i] = -1.0
    rows.append(A_abs)
    
    # Trust: sum(u) <= dmr
    A_trust = np.zeros((1, nvar))
    A_trust[0, N:2*N] = 1.0
    rows.append(A_trust)
    
    A_ub = np.vstack(rows)
    b_ub = np.concatenate([-q, np.zeros(2*N), [dmr]])
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, nvar))
    A_eq[0, :N] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    lb = np.maximum(-h, -np.inf)  # -h[i]
    ub = np.minimum(1.0 - h, np.inf)  # 1-h[i]
    bounds_d = list(zip(-h, 1.0 - h))
    bounds_u = [(0, None)] * N
    bounds_t = [(None, None)]
    bounds = bounds_d + bounds_u + bounds_t
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9, 'maxiter': 500})
    except Exception:
        return None
    
    if res is None or not res.success:
        return None
    x = res.x
    if len(x) != nvar or not np.all(np.isfinite(x)):
        return None
    return x[:N]  # return d

def lp_search(h0, N, m, dx, max_iter=45, dmr0=0.5):

    best_h = h.copy()
    best_s = exact_score(h, dx)
    if best_s is None:
        return best_h, None
    dmr = dmr0
    consec_fail = 0
    for it in range(max_iter):

        if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):

            consec_fail += 1
            if consec_fail >= 4:

            continue
        d = solve_lp(h, q, G, dmr)
        if d is None:
            dmr *= 0.5
            consec_fail += 1
            if consec_fail >= 4:
                break
            continue
        consec_fail = 0
        h_trial_raw = h + d
        h_trial = project_capped(h_trial_raw, m)
        if h_trial is None:
            dmr *= 0.5
            continue
        luc = float(np.sum(np.abs(h_trial - h)))
        if luc > dmr * (1 + 1e-6):
            dmr *= 0.5
            continue
        s_trial = exact_score(h_trial, dx)
        if s_trial is None:
            dmr *= 0.5
            continue
        # Model score
        model_vals = q + G @ (h_trial - h)
        s_model = float(np.max(model_vals))
        p = best_s - s_model  # predicted reduction (from current, not best)
        # Actually use current score
        s_cur = exact_score(h, dx)
        if s_cur is None:

        p = s_cur - s_model
        a = s_cur - s_trial
        improved = a > 1e-12 * max(1.0, abs(s_cur))
        if improved:
            h = h_trial.copy()
            if s_trial < best_s:
                best_s = s_trial
                best_h = h_trial.copy()
            # Trust region update

                if rho < 0.1:

                if rho < 0.1:
                    dmr *= 0.5
                # else leave unchanged

import numpy as np
from scipy.optimize import linprog

def project_capped(v, m):
    v = np.asarray(v, dtype=np.float64).copy()
    v = np.clip(v, -1e6, 1e6)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if float(np.sum(y)) > m:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - float(np.sum(y))
    for _ in range(4):
        if abs(r) < 1e-14:
            break
        if r > 0:
            for i in range(len(y)):
                if r < 1e-14:
                    break
                sp = 1.0 - y[i]
                if sp > 1e-14:
                    a = min(sp, r)
                    y[i] += a
                    r -= a
        else:
            for i in range(len(y)):
                if -r < 1e-14:
                    break
                sp = y[i]
                if sp > 1e-14:
                    a = min(sp, -r)
                    y[i] -= a
                    r += a
    if not np.all(np.isfinite(y)):
        return None
    y = np.clip(y, 0.0, 1.0)
    if abs(float(np.sum(y)) - m) > 1e-10:
        return None
    return y

def exact_score(h, dx):
    N = len(h)
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    if len(q) != 2 * N - 1 or not np.all(np.isfinite(q)):
        return None
    s = float(np.max(q))
    return s if np.isfinite(s) else None

def compute_grad(h, dx):
    N = len(h)
    L = 2 * N - 1
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    G = np.zeros((L, N), dtype=np.float64)
    for rlag in range(L):
        k = rlag - (N - 1)
        r1 = max(0, k)
        r2 = min(N, N + k)
        if r2 > r1:
            G[rlag, r1:r2] += dx * (1.0 - h[r1 - k:r2 - k])
        r3 = max(0, -k)
        r4 = min(N, N - k)
        if r4 > r3:
            G[rlag, r3:r4] -= dx * h[r3 + k:r4 + k]
    return q, G

def is_feasible(h, N, m):
    if len(h) != N or not np.all(np.isfinite(h)):
        return False
    if float(np.min(h)) < -1e-10 or float(np.max(h)) > 1.0 + 1e-10:
        return False
    return abs(float(np.sum(h)) - m) <= 1e-10

def solve_lp(h, q, G, dmr):
    N = len(h)
    L = 2 * N - 1
    nv = 2 * N + 1
    c = np.zeros(nv)
    c[-1] = 1.0
    A_ub = np.zeros((L + 2 * N + 1, nv))
    A_ub[:L, :N] = G
    A_ub[:L, -1] = -1.0
    for i in range(N):
        A_ub[L + 2 * i, i] = 1.0
        A_ub[L + 2 * i, N + i] = -1.0
        A_ub[L + 2 * i + 1, i] = -1.0
        A_ub[L + 2 * i + 1, N + i] = -1.0
    A_ub[-1, N:2 * N] = 1.0
    b_ub = np.concatenate([-q, np.zeros(2 * N), [dmr]])
    A_eq = np.zeros((1, nv))
    A_eq[0, :N] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-h[i], 1.0 - h[i]) for i in range(N)] + [(0, None)] * N + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'maxiter': 500,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
    except Exception:
        return None
    if res is None or not res.success:
        return None
    x = res.x
    if len(x) != nv or not np.all(np.isfinite(x)):
        return None
    return x[:N]

    if best_s is None:
        return best_h, None
    dmr = dmr0
    fails = 0
    for _ in range(max_iter):
        if dmr < 1e-4:
            break
        q, G = compute_grad(h, dx)
        if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):

            fails += 1

            continue
        d = solve_lp(h, q, G, dmr)

            continue
        fails = 0
        ht = project_capped(h + d, m)

        dt = ht - h
        luc = float(np.sum(np.abs(dt)))
        if luc > dmr * (1 + 1e-6):
            dmr *= 0.5
            continue
        st = exact_score(ht, dx)
        if st is None:
            dmr *= 0.5
            continue
        s_cur = exact_score(h, dx)
        if s_cur is None:
            break
        model = float(np.max(q + G @ dt))
        pred = s_cur - model
        act = s_cur - st
        if act > 1e-12 * max(1.0, abs(s_cur)):
            h = ht.copy()
            if st < best_s:
                best_s = st
                best_h = ht.copy()
            if pred <= 1e-12:

            else:

                if rho < 0.1:
                    dmr *= 0.5

                dmr *= 0.5
            else:
                if act / pred < 0.1:
                    dmr *= 0.5

    return best_h, best_s
