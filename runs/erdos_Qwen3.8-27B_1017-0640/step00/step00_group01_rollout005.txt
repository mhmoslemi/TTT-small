        phi = rng.uniform(0, 2 * math.pi)
        v = 0.5 + 0.25 * np.sin(2 * math.pi * qq * (idx_arr + 0.5) / n + phi)
        p = capped_simplex_project(v, s)
        if p is not None:
            starts.append(p)

    if n <= 64:
        max_starts, n_adapt, maxiter = 8, 2, 300
    elif n <= 128:
        max_starts, n_adapt, maxiter = 6, 1, 180
    else:
        max_starts, n_adapt, maxiter = 3, 0, 80

    starts = starts[:max_starts]
    for start in starts:
        sc = exact_score(start)
        if sc is not None and sc < best_score - 1e-12:
            best_score = sc
            best_h = start.copy()
        result = solve_start(start, maxiter)
        sc = exact_score(result)
        if sc is not None and sc < best_score - 1e-12:
            best_score = sc
            best_h = result.copy()

    for ns_val in [0.02, 0.06][:n_adapt]:
        v = best_h + rng.standard_normal(n) * ns_val
        p = capped_simplex_project(v, s)
        if p is not None:
            sc = exact_score(p)
            if sc is not None and sc < best_score - 1e-12:
                best_score = sc
                best_h = p.copy()
            result = solve_start(p, maxiter)
            sc = exact_score(result)
            if sc is not None and sc < best_score - 1e-12:
                best_score = sc
                best_h = result.copy()

    if not is_feasible(best_h):
        best_h = capped_simplex_project(best_h, s)
    if best_h is None or not is_feasible(best_h):
        best_h = parent_h.copy()
    best_score = exact_score(best_h)
    if best_score is None:
        best_score = 1.0
    return (best_h, float(best_score), n)
