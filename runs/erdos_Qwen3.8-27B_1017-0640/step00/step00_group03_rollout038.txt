            arch_score = s

    attempts = 0
    while len(pop) < P and attempts < 10 * P:
        attempts += 1
        c = arch_h + noise_scales[attempts % 4] * rng.standard_normal(n)
        proj = project_simplex(c, n)
        if proj is None or not check_feasible(proj, n):
            continue
        if is_dup(proj, pop, 1e-10):
            continue
        s = exact_score(proj, n)
        pop.append(proj)
        pop_s.append(s)
        if s < arch_score:
            arch_h = proj.copy()
            arch_score = s

    while len(pop) < P:
        pop.append(arch_h.copy())
        pop_s.append(arch_score)

    stagnation = 0
    restarts = 0
    prev_as = arch_score

    for gen in range(G):
        pop_snap = [p.copy() for p in pop]
        F = rng.uniform(0.55, 0.90) if gen < G // 2 else rng.uniform(0.35, 0.75)
        next_pop = list(pop)
        next_s = list(pop_s)
        Pn = len(pop_snap)
        for i in range(Pn):
            CR = rng.uniform(0.65, 1.0)
            if rng.random() < 0.75:
                avail = np.array([j for j in range(Pn) if j != i], dtype=int)
                r1, r2, r3 = (int(x) for x in rng.choice(avail, size=3, replace=False))
                donor = pop_snap[r1] + F * (pop_snap[r2] - pop_snap[r3])
            else:
                avail = np.array([j for j in range(Pn) if j != i], dtype=int)
                r1, r2 = (int(x) for x in rng.choice(avail, size=2, replace=False))
                donor = pop_snap[i] + F * (arch_h - pop_snap[i]) + F * (pop_snap[r1] - pop_snap[r2])
            mask = rng.random(n) < CR
            mask[int(rng.integers(n))] = True
            trial = np.where(mask, donor, pop_snap[i])
            proj = project_simplex(trial, n)
            if proj is None or not check_feasible(proj, n):
                continue
            s = exact_score(proj, n)
            if s < arch_score:
                arch_h = proj.copy()
                arch_score = s
            if s < next_s[i]:
                next_pop[i] = proj
                next_s[i] = s
        pop = next_pop
        pop_s = next_s
        if arch_score < prev_as - 1e-14:
            stagnation = 0
        else:
            stagnation += 1
        prev_as = arch_score
        if stagnation >= 18 and restarts < 2:
            sigma = 0.10 if restarts == 0 else 0.035
            npop = [arch_h.copy()]
            ns = [arch_score]
            if len(pop) > 0:
                npop.append(pop[0].copy())
                ns.append(pop_s[0])
            for k in range(3 * P // 4):
                c = arch_h + sigma * rng.standard_normal(n)
                proj = project_simplex(c, n)
                if proj is None or not check_feasible(proj, n):
                    proj = arch_h.copy()
                if not is_dup(proj, npop, 1e-10):
                    npop.append(proj)
                    ns.append(exact_score(proj, n))
            for k in range(P // 4):
                proj = project_simplex(rng.random(n), n)
                if proj is None or not check_feasible(proj, n):
                    proj = arch_h.copy()
                if not is_dup(proj, npop, 1e-10):
                    npop.append(proj)
                    ns.append(exact_score(proj, n))
            while len(npop) < P:
                npop.append(arch_h.copy())
                ns.append(arch_score)
            for j in range(len(ns)):
                if ns[j] < arch_score:
                    arch_h = npop[j].copy()
                    arch_score = ns[j]
            pop = npop[:P]
            pop_s = ns[:P]
            stagnation = 0
            restarts += 1

    best_h = arch_h.copy()
    best_s = exact_score(best_h, n)
    if fb_score < best_s:
        best_h = fallback_h.copy()
        best_s = fb_score
    if not check_feasible(best_h, n) or not np.isfinite(best_s):
        best_h = uniform.copy()
        best_s = exact_score(best_h, n)
    final_s = exact_score(best_h, n)
    return (best_h, final_s, int(n))
