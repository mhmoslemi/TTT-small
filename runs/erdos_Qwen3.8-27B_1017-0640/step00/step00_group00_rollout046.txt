        G[n - 1 - p, :n - p] += dx * (1.0 - h[p:])
        G[n - 1 - p, p:] -= dx * h[:-p]
    return G

def _block(n, m):
    h = np.zeros(n)
    h[:int(m)] = 1.0
    f = m - int(m)
    if f > 1e-15 and int(m) < n:
        h[int(m)] = f
    return h

def _so_obj(z, *a):
    return z[len(z) - 1]

def _so_objj(z, *a):
    j = np.zeros(len(z)); j[-1] = 1.0; return j

def _so_eqf(z, nn, G, q, dx):
    return float(np.sum(z[:nn]))

def _so_eqj(z, nn, G, q, dx):
    j = np.zeros(nn + 2); j[:nn] = 1.0; return j

def _so_nf(z, nn, G, q, dx):
    d = z[:nn]
    return z[nn] - float(np.dot(d, d))

def _so_nj(z, nn, G, q, dx):
    j = np.zeros(nn + 2); j[:nn] = -2.0 * z[:nn]; j[nn] = 1.0; return j

def _so_lf(z, nn, G, q, dx, k):
    return z[nn + 1] - q[k] - float(G[k] @ z[:nn]) - dx * z[nn]

def _so_lj(z, nn, G, q, dx, k):
    j = np.zeros(nn + 2); j[:nn] = -G[k]; j[nn] = -dx; j[nn + 1] = 1.0; return j

def _slsqp(x, q, G, dx, n, maxiter):
    try:
        from scipy.optimize import minimize
        L = 2 * n - 1
        z0 = np.zeros(n + 2); z0[n + 1] = float(np.max(q))
        bnds = [(-x[i], 1.0 - x[i]) for i in range(n)] + [(0.0, float(n)), (-1e10, 1e10)]
        cs = [{'type': 'eq', 'fun': _so_eqf, 'jac': _so_eqj, 'args': (n, G, q, dx)},
              {'type': 'ineq', 'fun': _so_nf, 'jac': _so_nj, 'args': (n, G, q, dx)}]
        for k in range(L):
            cs.append({'type': 'ineq', 'fun': _so_lf, 'jac': _so_lj, 'args': (n, G, q, dx, k)})
        res = minimize(_so_obj, z0, jac=_so_objj, bounds=bnds, constraints=cs,
                       method='SLSQP', options={'ftol': 1e-9, 'maxiter': maxiter, 'disp': False})
        if np.all(np.isfinite(res.x[:n])):
            return res.x[:n]
    except Exception:
        pass
    return None

def _cvx_solve(x, q, G, n, cvx_vars, solver):
    d, s, t, qp, Gp, lp, hp, prob = cvx_vars
    try:
        qp.value = q; Gp.value = G
        lp.value = -x; hp.value = 1.0 - x
        prob.solve(solver=solver, verbose=False)
        if d.value is not None and np.all(np.isfinite(d.value)):
            dv = np.array(d.value).flatten().astype(np.float64)
            if (np.all(dv >= -x - 1e-4) and np.all(dv <= 1.0 - x + 1e-4)
                    and abs(float(np.sum(dv))) < 1e-3 and float(s.value) > -1e-4
                    and float(s.value) - float(np.dot(dv, dv)) > -1e-3):
                return dv
    except Exception:
        pass
    return None

def _cvx_setup(n):
    if not HAS_CVX:
        return None
    try:
        L = 2 * n - 1; dx = 2.0 / n
        d = cp.Variable(n); s = cp.Variable(non_neg=True); t = cp.Variable()
        qp = cp.Parameter(L); Gp = cp.Parameter((L, n))
        lp = cp.Parameter(n); hp = cp.Parameter(n)
        cs = [d >= lp, d <= hp, cp.sum(d) == 0, cp.sum_squares(d) <= s,
              qp + Gp @ d + dx * s <= t]
        prob = cp.Problem(cp.Minimize(t), cs)
        return (d, s, t, qp, Gp, lp, hp, prob)
    except Exception:
        return None

def _mm_optimize(x0, n, m, cvx_vars, max_iter):
    dx = 2.0 / n
    x = x0.copy()
    cur_sc = _score(x, n)
    if cur_sc is None:
        return x0
    best = x.copy(); best_sc = cur_sc
    for _ in range(max_iter):
        q_raw = np.correlate(x, 1.0 - x, mode='full') * dx
        G = _grads(x, n, dx)
        d = None
        if cvx_vars is not None:
            d = _cvx_solve(x, q_raw, G, n, cvx_vars, 'ECOS')
            if d is None:
                d = _cvx_solve(x, q_raw, G, n, cvx_vars, 'CVXOPT')
        if d is None:
            cap = 100 if n <= 64 else (60 if n <= 160 else (35 if n <= 400 else 20))
            d = _slsqp(x, q_raw, G, dx, n, cap)
        if d is None:
            break
        y = _proj(x + d, n, m)
        if y is None:
            break
        best_y = x.copy(); best_ys = cur_sc
        for th in (1.0, 0.5, 0.25, 0.125, 0.0625):
            pt = x + th * (y - x)
            sc = _score(pt, n)
            if sc is not None and sc < best_ys - 1e-14:
                best_y = pt.copy(); best_ys = sc
        if best_ys < cur_sc - 1e-12:
            impr = cur_sc - best_ys
            x = best_y; cur_sc = best_ys
            if cur_sc < best_sc - 1e-12:
                best = x.copy(); best_sc = cur_sc
            if impr < 1e-10:
                break
        else:
            break
    return best

def _try_add(cand, n, m, seen, starts):
    if cand is None:
        return
    c = _proj(cand, n, m)
    if c is None or not _feas(c, n, m):
        return
    for s in seen:
        if float(np.max(np.abs(c - s))) < 1e-10:
            return
    sc = _score(c, n)
    if sc is not None:
        seen.append(c.copy())
        starts.append((c.copy(), sc))

def run(seed=42, **kwargs):
    n = 45
    has_override = False
    if 'n_points' in kwargs:
        nv = kwargs['n_points']
        if not isinstance(nv, bool) and isinstance(nv, (int, np.integer)) and 1 <= int(nv) <= 999:
            n = int(nv); has_override = True
    m = n / 2.0
    parent = None
    try:
        pv = initial_h_values
        if pv is not None:
            pv = np.array(pv, dtype=np.float64).flatten()
            if pv.size > 0 and pv.size <= 999 and bool(np.all(np.isfinite(pv))):
                parent = pv
    except Exception:
        parent = None
    if not has_override and parent is not None:
        n = len(parent); m = n / 2.0
    dx = 2.0 / n
    p_at_n = None
    if parent is not None:
        if len(parent) == n:
            p_at_n = _proj(parent, n, m)
        elif len(parent) == 1:
            p_at_n = _proj(np.full(n, float(parent[0])), n, m)
        else:
            xp = np.linspace(0, 1, len(parent)); xn = np.linspace(0, 1, n)
            p_at_n = _proj(np.interp(xn, xp, parent), n, m)
    fallback = _proj(np.full(n, 0.5), n, m)
    fb_sc = _score(fallback, n)
    rng = np.random.default_rng(seed)
    seen = []; starts = []
    _try_add(p_at_n, n, m, seen, starts)
    if p_at_n is not None:
        _try_add(1.0 - p_at_n, n, m, seen, starts)
    blk = _block(n, m)
    _try_add(blk, n, m, seen, starts)
    _try_add(rng.random(n), n, m, seen, starts)
    _try_add(blk[::-1], n, m, seen, starts)
    if p_at_n is not None:
        _try_add(p_at_n + 0.10 * rng.standard_normal(n), n, m, seen, starts)
        _try_add(p_at_n[::-1], n, m, seen, starts)
        _try_add(np.roll(p_at_n, n // 3), n, m, seen, starts)
    ii = np.arange(n)
    for r in (1, 2):
        for ph in (0.0, math.pi / 3.0):
            _try_add(0.5 + 0.4 * np.cos(2.0 * math.pi * r * ii / n + ph), n, m, seen, starts)
    _try_add(rng.random(n), n, m, seen, starts)
    max_starts = 6 if n <= 64 else (4 if n <= 160 else (3 if n <= 400 else 2))
    starts = starts[:max_starts]
    best_h = fallback.copy(); best_c5 = fb_sc
    for (sh, sc) in starts:
        if sc < best_c5 - 1e-12:
            best_h = sh.copy(); best_c5 = sc
    cvx_vars = _cvx_setup(n)
    max_it = 40 if n <= 64 else (25 if n <= 160 else (12 if n <= 400 else 5))
    for (sh, sc) in starts:
        res = _mm_optimize(sh, n, m, cvx_vars, max_it)
        rs = _score(res, n)
        if rs is not None and rs < best_c5 - 1e-12:
            best_h = res.copy(); best_c5 = rs
    if n <= 64:
        sds = (0.04, 0.15)
    elif n <= 160:
        sds = (0.08,)
    elif n <= 400:
        sds = (0.12,)
    else:
        sds = (0.15,)
    round_best = best_c5
    for sd in sds:
        pert = _proj(best_h + sd * rng.standard_normal(n), n, m)
        if pert is not None:
            res = _mm_optimize(pert, n, m, cvx_vars, max_it)
            rs = _score(res, n)
            if rs is not None and rs < best_c5 - 1e-12:
                best_h = res.copy(); best_c5 = rs
        if best_c5 < round_best - 1e-9:
            round_best = best_c5
        else:
            break
    if not _feas(best_h, n, m):
        best_h = _proj(best_h, n, m)
    if not _feas(best_h, n, m):
        best_h = fallback
    best_c5 = _score(best_h, n)
    if best_c5 is None:
        best_h = fallback; best_c5 = _score(fallback, n)
    return (np.array(best_h, dtype=np.float64), float(best_c5), n)
