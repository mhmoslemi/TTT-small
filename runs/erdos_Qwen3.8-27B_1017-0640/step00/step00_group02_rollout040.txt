import numpy as np
import math

def feasible(h, n, S):
    if h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < 0.0) or np.any(h > 1.0):
        return False
    if abs(float(np.sum(h)) - S) > 1e-10:
        return False
    return True

def project_capped(v, S):
    v = np.asarray(v, dtype=np.float64).ravel()
    if v.size < 2 or not np.all(np.isfinite(v)):
        return None
    v = np.clip(v, -1e6, 1e6)
    n = v.size
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        x = np.clip(v - mid, 0.0, 1.0)
        s = float(np.sum(x))
        if abs(s - S) < 1e-14:
            break
        if s > S:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-15:
            break
    x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(3):
        r = S - float(np.sum(x))
        if abs(r) < 1e-12:
            break
        if r > 0:
            cap = 1.0 - x
            idx = np.where(cap > 1e-14)[0]
            if len(idx) == 0:
                break
            share = r / float(cap[idx].sum())
            x[idx] = np.minimum(1.0, x[idx] + share * cap[idx])
        else:
            cap = x
            idx = np.where(cap > 1e-14)[0]
            if len(idx) == 0:
                break
            share = (-r) / float(cap[idx].sum())
            x[idx] = np.maximum(0.0, x[idx] - share * cap[idx])
    if not np.all((x >= 0.0) & (x <= 1.0)):
        return None
    if abs(float(np.sum(x)) - S) > 1e-10:
        return None
    return x

def exact_score(h, n):
    dx = 2.0 / n
    S = n / 2.0
    if not feasible(h, n, S):
        return None
    try:
        c = np.correlate(h, 1.0 - h, mode='full')
        val = float(np.max(c)) * dx
        if not np.isfinite(val):
            return None
        return val
    except Exception:
        return None

def make_proposal(h, n, S, rho, rng, p_perm, p_swap):
    u = rng.uniform()
    if u < p_perm:
        return h.copy()[rng.permutation(n)]
    if u < p_perm + p_swap:
        i, j = rng.choice(n, 2, replace=False)
        nh = h.copy()
        nh[i], nh[j] = h[j], h[i]
        return nh
    i, j = rng.choice(n, 2, replace=False)
    c = h[i] + h[j]
    L = max(0.0, c - 1.0)
    U = min(1.0, c)
    W = U - L
    if W < 1e-15:
        return None
    z = h[i] + rng.uniform(-rho * W, rho * W)
    r = (z - L) % (2.0 * W)
    if r <= W:
        xp = L + r
    else:
        xp = U - (r - W)
    nh = h.copy()
    nh[i] = xp
    nh[j] = c - xp
    if not feasible(nh, n, S):
        fixed = project_capped(nh, S)
        if fixed is None:
            return None
        nh = fixed
    return nh

def calibrate_temperature(start, start_score, n, S, rng, pilot_count, p_perm, p_swap):
    best_pilot = None
    best_pilot_score = start_score
    deltas = []
    for _ in range(pilot_count):
        p = make_proposal(start, n, S, 0.65, rng, p_perm, p_swap)
        if p is None:
            continue
        sc = exact_score(p, n)
        if sc is None:
            continue
        if sc < best_pilot_score - 1e-12:
            best_pilot_score = sc
            best_pilot = p.copy()
        if sc > start_score + 1e-12:
            deltas.append(sc - start_score)
    if deltas:
        med = float(np.median(deltas))
        T0 = med / (-math.log(0.65))
        T0 = max(1e-5, min(0.05, T0))
    else:
        T0 = 0.002
    return best_pilot, best_pilot_score, T0

def anneal_chain(start, start_score, n, S, rng, L, M, pilot_count, p_perm, p_swap,
                 incumbent, incumbent_score):
    best_p, best_ps, T0 = calibrate_temperature(start, start_score, n, S, rng,
                                                pilot_count, p_perm, p_swap)
    if best_ps < incumbent_score - 1e-12:
        incumbent = best_p.copy()
        incumbent_score = best_ps
    current = start.copy()
    current_score = start_score
    Tend = max(1e-7, 0.01 * T0)
    for l in range(L):
        if L == 1:
            Tl = Tend
        else:
            Tl = T0 * (Tend / T0) ** (l / (L - 1))
        rho_l = max(0.03, 0.65 * math.sqrt(Tl / T0))
        for _ in range(M):
            try:
                prop = make_proposal(current, n, S, rho_l, rng, p_perm, p_swap)
                if prop is None:
                    continue
                ps = exact_score(prop, n)
                if ps is None:
                    continue
                if ps < incumbent_score - 1e-12:
                    incumbent = prop.copy()
                    incumbent_score = ps
                delta = ps - current_score
                if delta <= 0:
                    current = prop
                    current_score = ps
                else:
                    exp_arg = -delta / Tl
                    if exp_arg < -50.0:
                        pass
                    else:
                        if rng.uniform() < math.exp(exp_arg):
                            current = prop
                            current_score = ps
            except Exception:
                continue
    return incumbent, incumbent_score
