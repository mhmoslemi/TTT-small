import math

    pop = []
    cands = []
    cands.append(np.full(n, 0.5))
    if center is not None:
        cands.append(center.copy())
        cands.append(center[::-1].copy())
        cands.append(1.0 - center.copy())
        s1 = n // 4
        if s1 > 0:
            cands.append(np.roll(center, -s1).copy())
        cands.append(np.roll(center, -n // 2).copy())
    idx = np.arange(n)
    for phi in [0.0, math.pi / 3, 2.0 * math.pi / 3]:
        cands.append(0.5 + 0.35 * np.sin(2.0 * np.pi * (idx + 0.5) / n + phi))
    cands.append(0.5 + 0.2 * np.cos(4.0 * np.pi * (idx + 0.5) / n))
    base = center if center is not None else np.full(n, 0.5)
    scales = [0.025, 0.06, 0.12, 0.20]
    for s_i in range(8):
        sig = scales[s_i % 4]
        cands.append(base + sig * rng.standard_normal(n))
    for _ in range(4):
        cands.append(rng.uniform(0.0, 1.0, n))
    for c in cands:
        p = project_capped(c, n)
        if p is not None:
            dup = False
            for e in pop:
                if np.max(np.abs(e - p)) < 1e-10:
                    dup = True
                    break
            if not dup:
                pop.append(p)
        if len(pop) >= P:
            break
    while len(pop) < P:
        pop.append(cands[0].copy() if is_feasible(cands[0], n) else np.full(n, 0.5))
    return pop[:P]

def de_generate(pop, pop_scores, archive_h, archive_score, n, rng, P, gen, G, max_restarts, restart_count, stagnant):

    next_scores = list(pop_scores)
    improved = False
    half = G // 2
    if gen < half:
        F = rng.uniform(0.55, 0.90)
    else:
        F = rng.uniform(0.35, 0.75)
    for i in range(P):
        CR = rng.uniform(0.65, 1.0)
        others = [j for j in range(P) if j != i]
        if rng.random() < 0.75:
            idxs = rng.choice(others, size=3, replace=False)
            r1, r2, r3 = idxs
            donor = pop[r1].astype(np.float64) + F * (pop[r2].astype(np.float64) - pop[r3].astype(np.float64))
        else:

            r1, r2 = idxs
            donor = pop[i].astype(np.float64) + F * (archive_h.astype(np.float64) - pop[i].astype(np.float64)) + F * (pop[r1].astype(np.float64) - pop[r2].astype(np.float64))
        mask = rng.random(n) < CR
        j_rand = rng.integers(0, n)
        mask[j_rand] = True
        trial_vec = pop[i].copy()
        trial_vec[mask] = donor[mask]
        trial = project_capped(trial_vec, n)
        if trial is None:
            continue
        s = score_c5(trial, n)
        if s < archive_score:
            archive_h = trial.copy()
            archive_score = s
            improved = True
        if s < next_scores[i]:
            next_pop[i] = trial
            next_scores[i] = s
    return next_pop, next_scores, archive_h, archive_score, improved

    fallback_score = score_c5(fallback, n)

        else:

    if center is not None:

    archive_score = fallback_score
    if n > 256:

    P = 32 if n <= 128 else 24
    G = 80 if n <= 128 else 45

    pop_scores = [score_c5(p, n) for p in pop]

            archive_h = pop[i].copy()
            archive_score = pop_scores[i]
