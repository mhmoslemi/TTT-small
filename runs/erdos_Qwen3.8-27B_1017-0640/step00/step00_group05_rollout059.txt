def obj_func(z, N):
    return z[N]


def obj_grad(z, N):
    g = np.zeros(N + 1)
    g[N] = 1.0
    return g


def eq_fun(z, m, N):
    return z[:N].sum() - m


def eq_jac(z, m, N):
    g = np.zeros(N + 1)
    g[:N] = 1.0
    return g


def ineq_fun(z, N, dx):
    h = z[:N]
    return z[N] - np.correlate(h, 1.0 - h, mode='full') * dx


def ineq_jac(z, N, dx):
    h = z[:N]
    J = corr_grad_matrix(h, N)
    jac = np.zeros((2 * N - 1, N + 1))
    jac[:, :N] = -J
    jac[:, N] = 1.0
    return jac


def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)

    try:
        parent = np.array(initial_h_values, dtype=np.float64).ravel()
    except Exception:
        parent = None

    if parent is not None and len(parent) > 0 and np.all(np.isfinite(parent)):
        N = int(len(parent))
    else:
        N = 70
        parent = np.full(N, 0.5)

    if N <= 0:
        N = 70
        parent = np.full(N, 0.5)

    dx = 2.0 / N
    m = N / 2.0

    h_half = np.full(N, 0.5)
    h_parent_proj = project_box_sum(parent, m)

    starts = [h_parent_proj.copy(), h_half.copy()]

    if N <= 256:
        for scale in [0.03, 0.08, 0.18]:
            noisy = h_parent_proj + rng.normal(0, scale, N)
            starts.append(project_box_sum(noisy, m))
        starts.append(project_box_sum(rng.uniform(0, 1, N), m))
        starts.append(project_box_sum(rng.beta(0.7, 0.7, N), m))
        i = np.arange(N)
        for (p, phi) in [(1, 0.0), (2, np.pi / 3.0)]:
            smooth = 0.5 + 0.45 * np.cos(2.0 * np.pi * p * (i + 0.5) / N + phi)
            starts.append(project_box_sum(smooth, m))
    else:
        starts.append(project_box_sum(h_parent_proj + rng.normal(0, 0.08, N), m))
        starts.append(project_box_sum(rng.uniform(0, 1, N), m))
        starts.append(project_box_sum(rng.beta(0.7, 0.7, N), m))

    best_h = None
    best_c = float('inf')
    for hs in starts:
        if len(hs) == N and np.all(np.isfinite(hs)):
            c = exact_score(hs)
            if c < best_c - 1e-12:
                best_c = c
                best_h = hs.copy()

    if N <= 999:
        bounds = [(0.0, 1.0)] * N + [(0.0, 2.0)]
        constraints = [
            {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (m, N)},
            {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac, 'args': (N, dx)},
        ]
        for hs in starts:
            try:
                t0 = min(max(exact_score(hs), 0.0), 2.0)
                z0 = np.concatenate([hs, [t0]])
                res = minimize(obj_func, z0, method='SLSQP', jac=obj_grad,
                               bounds=bounds, constraints=constraints,
                               options={'maxiter': 250, 'ftol': 1e-10, 'disp': False})
                if res is not None and hasattr(res, 'x') and len(res.x) == N + 1:
                    h_try = project_box_sum(np.asarray(res.x[:N], dtype=np.float64), m)
                    if np.all(np.isfinite(h_try)):
                        c_try = exact_score(h_try)
                        if c_try < best_c - 1e-12:
                            best_c = c_try
                            best_h = h_try.copy()
            except Exception:
                continue

    if best_h is None or not np.all(np.isfinite(best_h)) or len(best_h) != N:
        best_h = h_half

    best_h = project_box_sum(best_h, m)
    if best_h.min() < -1e-9 or best_h.max() > 1.0 + 1e-9 or abs(best_h.sum() - m) > 1e-8:
        best_h = h_half

    best_c = exact_score(best_h)
    return (best_h, best_c, N)
