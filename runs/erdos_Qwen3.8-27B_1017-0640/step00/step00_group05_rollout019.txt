def _solve_lp(h, q, G, Delta):
    n = len(h)
    L = len(q)
    nv = 2 * n + 1
    c = np.zeros(nv)
    c[-1] = 1.0
    
    # Build A_ub, b_ub
    rows = []
    rhs = []
    
    # Lag constraints: G @ d - t <= -q
    A_lag = np.zeros((L, nv))
    A_lag[:, :n] = G
    A_lag[:, -1] = -1.0
    rows.append(A_lag)
    rhs.append(-q)
    
    # d - u <= 0
    A_du = np.zeros((n, nv))
    A_du[np.arange(n), np.arange(n)] = 1.0
    A_du[np.arange(n), n + np.arange(n)] = -1.0
    rows.append(A_du)
    rhs.append(np.zeros(n))
    
    # -d - u <= 0
    A_ndu = np.zeros((n, nv))
    A_ndu[np.arange(n), np.arange(n)] = -1.0
    A_ndu[np.arange(n), n + np.arange(n)] = -1.0
    rows.append(A_ndu)
    rhs.append(np.zeros(n))
    
    # sum(u) <= Delta
    A_u = np.zeros((1, nv))
    A_u[0, n:2*n] = 1.0
    rows.append(A_u)
    rhs.append(np.array([Delta]))
    
    A_ub = np.vstack(rows)
    b_ub = np.concatenate(rhs)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds
    bounds = []
    for i in range(n):
        bounds.append((-h[i], 1.0 - h[i]))
    for i in range(n):
        bounds.append((0.0, None))
    bounds.append((None, None))
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                           'primal_feasibility_tolerance': 1e-9})
    
    if res.status != 0 or not np.all(np.isfinite(res.x)):

def _qG(h, dx):
    n = len(h)
    q = np.correlate(h, 1.0 - h, mode="full") * dx
    L = 2 * n - 1
    G = np.zeros((L, n), dtype=np.float64)
    for j in range(L):
        k = j - (n - 1)
        for r in range(n):
            g = 0.0
            i1 = r - k
            if 0 <= i1 < n:
                g += 1.0 - h[i1]
            i2 = r + k
            if 0 <= i2 < n:
                g -= h[i2]
            G[j, r] = dx * g
    if not np.all(np.isfinite(G)) or not np.all(np.isfinite(q)):
        return None, None

    A4[0, n:2*n] = 1.0

    bounds.append((None, None))

                  options={'presolve': True,
                           'dual_feasibility_tolerance': 1e-9,
                           'primal_feasibility_tolerance': 1e-9})

    return final, final_c, N
