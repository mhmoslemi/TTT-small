    b *= dx
    c *= dx

    candidates = [lo, hi, 0.0]
    # Pairwise intersections
    for p in range(n_lags):
        for q in range(p + 1, n_lags):
            A = a[p] - a[q]
            B = b[p] - b[q]
            C = c[p] - c[q]
            tol = 1e-14 * max(1.0, abs(A), abs(B), abs(C))
            if abs(A) < tol:
                if abs(B) > tol:
                    t = -C / B
                    if lo - 1e-10 <= t <= hi + 1e-10:
                        candidates.append(max(lo, min(hi, t)))
            else:
                disc = B*B - 4.0*A*C
                neg_tol = 1e-14 * max(1.0, B*B, abs(4.0*A*C))
                if disc < -neg_tol:
                    continue
                if disc < 0:
                    disc = 0.0
                sq = math.sqrt(disc)
                R = -0.5 * (B + math.copysign(sq, B))
                if abs(R) > tol:
                    t1 = R / A
                    t2 = C / R
                else:
                    t1 = t2 = -B / (2.0 * A)
                for t in [t1, t2]:
                    if lo - 1e-10 <= t <= hi + 1e-10:
                        candidates.append(max(lo, min(hi, t)))
        # Vertices of convex polynomials
        if a[p] > tol:
            tv = -b[p] / (2.0 * a[p])
            if lo - 1e-10 <= tv <= hi + 1e-10:
                candidates.append(max(lo, min(hi, tv)))
    
    # Sort and deduplicate
    candidates = sorted(set(round(ct, 14) for ct in candidates))
    candidates = [lo, hi, 0.0] + [ct for ct in candidates if ct not in (lo, hi, 0.0)]
    
    # Evaluate envelope at candidates, pick best
    best_t = 0.0
    best_val = score + 1.0
    for t in candidates:
        t = max(lo, min(hi, t))
        val = 0.0
        for k in range(n_lags):
            pk = a[k]*t*t + b[k]*t + c[k]
            if pk > val:
                val = pk
        if val < best_val:
            best_val = val
            best_t = t

    proposal = h.copy()
    proposal[i] = h[i] + best_t
    proposal[j] = h[j] - best_t
    proj = project_capped_simplex(proposal, n)
    if proj is None or not is_feasible(proj, n):
        return h.copy(), score
    s = exact_score(proj, n)
    if s < score - 1e-12:
        return proj, s
    return h.copy(), score
