    return best_h, best_s

def add_start(starts, arr, n, cap):
    if len(starts) >= cap:
        return
    if len(arr) != n or not np.all(np.isfinite(arr)):
        return
    proj = project_capped_simplex(arr, n)
    if proj is None:
        return
    for s in starts:
        if np.max(np.abs(proj - s)) <= 1e-9:
            return
    starts.append(proj.copy())

def make_starts(rng, n, parent, cap):
    starts = []
    x = np.arange(n) + 0.5
    if parent is not None:
        add_start(starts, parent, n, cap)
        add_start(starts, 1.0 - parent, n, cap)
        add_start(starts, parent[::-1].copy(), n, cap)
    add_start(starts, np.full(n, 0.5), n, cap)
    add_start(starts, 0.5 + 0.30 * np.sin(2 * np.pi * x / n + rng.random() * 2 * np.pi), n, cap)
    add_start(starts, 0.5 + 0.30 * np.sin(4 * np.pi * x / n + rng.random() * 2 * np.pi), n, cap)
    if parent is not None:
        add_start(starts, parent + 0.12 * rng.standard_normal(n), n, cap)
    else:
        add_start(starts, 0.5 + 0.30 * np.sin(6 * np.pi * x / n + rng.random() * 2 * np.pi), n, cap)
        add_start(starts, 0.5 + 0.20 * rng.standard_normal(n), n, cap)
        add_start(starts, 0.25 + 0.5 * rng.random(n), n, cap)
    return starts

def run(seed=42, **kwargs):
    try:
        seed = int(seed)
        if seed < 0 or seed > 2**32 - 1:
            seed = 42
    except Exception:
        seed = 42
    rng = np.random.default_rng(seed)
    parent_raw = globals().get('initial_h_values', None)
    parent = None
    if parent_raw is not None:
        try:
            p = np.asarray(parent_raw, dtype=np.float64).ravel()
            if 2 <= len(p) < 1000 and np.all(np.isfinite(p)) and np.all(p >= -1e-7) and np.all(p <= 1 + 1e-7):
                n_p = len(p)
                proj = project_capped_simplex(p, n_p)
                if proj is not None:
                    parent = proj
        except Exception:
            pass
    if parent is None:
        n = 47
        fallback = np.full(n, 0.5)
    else:
        n = len(parent)
        fallback = parent.copy()
    cap = 7 if n <= 64 else (5 if n <= 128 else 3)
    n_passes = 3 if n <= 64 else (2 if n <= 128 else 2)
    n_pairs = 20 if n <= 64 else (12 if n <= 128 else 6)
    n_restarts = 2 if n <= 64 else (1 if n <= 128 else 0)
    best_h = parent.copy() if parent is not None else fallback.copy()
    best_s = exact_score(best_h)
    starts = make_starts(rng, n, parent, cap)
    for st in starts:
        try:
            sc = exact_score(st)
            if math.isfinite(sc) and sc < best_s - 1e-12:
                best_h, best_s = st.copy(), sc
            ph, ps = local_search(st, sc, n, rng, n_passes, n_pairs)
            if ps < best_s - 1e-12:
                best_h, best_s = ph, ps
        except Exception:
            continue
    for rs in range(n_restarts):
        try:
            scale = 0.06 if n <= 64 else 0.08
            perturbed = best_h + scale * rng.standard_normal(n)
            proj = project_capped_simplex(perturbed, n)
            if proj is None:
                continue
            sc = exact_score(proj)
            if not math.isfinite(sc):
                continue
            ph, ps = local_search(proj, sc, n, rng, n_passes, n_pairs)
            if ps < best_s - 1e-12:
                best_h, best_s = ph, ps
        except Exception:
            continue
    final = project_capped_simplex(best_h, n)
    if final is None:
        final = fallback.copy()
    fs = exact_score(final)
    if not math.isfinite(fs):
        final = fallback.copy()
        fs = exact_score(final)
    return final.copy(), float(fs), int(n)
