        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (n,)}
    ]
    base = parent if parent is not None else np.full(n, 0.5)
    starts = [parent.copy(), np.full(n, 0.5)]
    sigmas = [0.03, 0.08, 0.16, 0.30, 0.12, 0.20]
    for sig in sigmas:
        p = project_capped(base + rng.standard_normal(n) * sig, S)
        if p is not None:
            starts.append(p)
    for _ in range(2):
        u = rng.random(n)
        p = project_capped(u, S)
        if p is not None:
            starts.append(p)
    for m_f in [1, 2]:
        phase = rng.uniform(0.0, 2.0 * np.pi)
        idx = np.arange(n)
        v = 0.5 + 0.35 * np.sin(2.0 * np.pi * m_f * idx / n + phase)
        p = project_capped(v, S)
        if p is not None:
            starts.append(p)
    unique_starts = [starts[0]]
    for s in starts[1:]:
        if all(np.linalg.norm(s - u) > 1e-12 for u in unique_starts):
            unique_starts.append(s)
    best_h = parent.copy()
    best_score = exact_score(best_h)
    for start in unique_starts:
        sc = exact_score(start)
        if sc < best_score - 1e-12:
            best_h = start.copy()
            best_score = sc
        z0 = np.append(start.copy(), np.clip(sc, 0.0, 2.0))
        h_raw = attempt_solve(z0, n, bounds, constraints)
        if h_raw is not None:
            h_proj = project_capped(h_raw, S)
            if h_proj is not None:
                sc2 = exact_score(h_proj)
                if sc2 < best_score - 1e-12:
                    best_h = h_proj.copy()
                    best_score = sc2
    for sig in [0.08, 0.03, 0.01]:
        for _ in range(2):
            pert = project_capped(best_h + rng.standard_normal(n) * sig, S)
            if pert is None:
                continue
            sc = exact_score(pert)
            if sc < best_score - 1e-12:
                best_h = pert.copy()
                best_score = sc
            z0 = np.append(pert.copy(), np.clip(sc, 0.0, 2.0))
            h_raw = attempt_solve(z0, n, bounds, constraints)
            if h_raw is not None:
                h_proj = project_capped(h_raw, S)
                if h_proj is not None:
                    sc2 = exact_score(h_proj)
                    if sc2 < best_score - 1e-12:
                        best_h = h_proj.copy()
                        best_score = sc2
    final_h = project_capped(best_h, S)
    if final_h is None:
        final_h = fallback_h
        final_score = fallback_score
    else:
        final_score = exact_score(final_h)
        if final_score > fallback_score:
            final_h = fallback_h
            final_score = fallback_score
    final_h = np.clip(final_h, 0.0, 1.0)
    return final_h, final_score, n
