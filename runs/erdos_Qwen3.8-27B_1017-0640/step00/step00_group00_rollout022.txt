    starts.append(blk.copy())
    starts.append(blk[::-1].copy())
    if parent is not None:
        starts.append(1.0 - parent)
        starts.append(np.roll(parent, n // 3))
    c = np.full(n, 0.5)
    if c not in starts:
        starts.append(c)
    starts.append(rng.uniform(0, 1, n))
    ph = rng.uniform(0, 2 * math.pi)
    starts.append(0.5 + 0.35 * np.cos(2 * math.pi * np.arange(n) / n + ph))
    if parent is not None:
        starts.append(parent[::-1].copy())
    starts.append(rng.uniform(0, 1, n))
    rng.shuffle(starts)
    max_s = 6 if n <= 64 else (4 if n <= 160 else 3)
    proj = []
    for s in starts[:max_s]:
        p = _project(s, n)
        if p is not None and _feasible(p, n):
            dup = False
            for q in proj:
                if np.max(np.abs(p - q)) < 1e-10:
                    dup = True
                    break
            if not dup:
                proj.append(p)
    return proj

def run(seed=42, **kwargs):
    n_override = None
    if 'n_points' in kwargs:
        np_val = kwargs['n_points']
        if not isinstance(np_val, bool) and isinstance(np_val, (int, np.integer)):
            if 1 <= int(np_val) <= 999:
                n_override = int(np_val)
    parent = None
    try:
        ph = initial_h_values
        if ph is not None:
            ph = np.asarray(ph, dtype=float).ravel()
            if ph.size > 0 and ph.size <= 999 and np.all(np.isfinite(ph)):
                parent = ph
    except NameError:
        pass
    if n_override is not None:
        n = n_override
        if parent is not None and len(parent) != n:
            if len(parent) == 1:
                parent = np.full(n, parent[0])
            else:
                src = np.arange(len(parent)) / max(1, len(parent) - 1)
                dst = np.arange(n) / max(1, n - 1)
                parent = np.interp(dst, src, parent)
    elif parent is not None:
        n = len(parent)
    else:
        n = 45
    parent_proj = _project(parent, n) if parent is not None else None
    fallback = parent_proj.copy() if (parent_proj is not None and _feasible(parent_proj, n)) else np.full(n, 0.5)
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    starts = _make_starts(n, parent_proj, rng)
    if not starts:
        starts = [np.full(n, 0.5)]
    best_h, best_sc = fallback.copy(), _score(fallback, n)[0]
    npair = 1000 if n <= 64 else (96 if n <= 160 else 32)
    nsweep = 3 if n <= 64 else 2
    for s in starts:
        sc0, q0 = _score(s, n)
        h_s, q_s, sc_s = s.copy(), q0, sc0
        h_s, q_s, sc_s = _sweep(h_s, n, q_s, sc_s, rng, npair, nsweep)
        if sc_s < best_sc - 1e-12:
            best_sc, best_h = sc_s, h_s
    for std in (0.05, 0.18) if n <= 64 else [0.10 if n <= 160 else 0.15]:
        h_p = _project(best_h + rng.normal(0, std, n), n)
        if h_p is not None and _feasible(h_p, n):
            sc_p, q_p = _score(h_p, n)
            h_p, q_p, sc_p = _sweep(h_p, n, q_p, sc_p, rng, npair, max(1, nsweep - 1))
            if sc_p < best_sc - 1e-12:
                best_sc, best_h = sc_p, h_p
    if not _feasible(best_h, n):
        best_h = _project(best_h, n)
        if best_h is None:
            best_h = fallback.copy()
    best_h = np.clip(best_h, 0.0, 1.0)
    best_sc_exact, _ = _score(best_h, n)
    if not np.all(np.isfinite(best_h)):
        best_h = fallback.copy()
        best_sc_exact, _ = _score(best_h, n)
    return best_h.astype(np.float64), float(best_sc_exact), int(n)
