        e4 = encode_lattice(comp, D, n)
        starts.append(e4 if e4 is not None else (D // 2) * np.ones(n, dtype=np.int64))
        rh = capped_simplex_projection(rng.uniform(-0.25, 1.25, size=n), n)
        e5 = encode_lattice(rh, D, n)
        starts.append(e5 if e5 is not None else (D // 2) * np.ones(n, dtype=np.int64))

        best_D_k = None
        best_D_s = float('inf')

        for sk in starts:
            if sk is None:
                continue
            if int(np.sum(sk)) != M or np.any(sk < 0) or np.any(sk > D):
                continue
            try:
                bk, bs = anneal_chain(sk, D, n, rng, T_hi, T_lo, n_blocks, ppb)
                if bs < best_D_s - 1e-12:
                    best_D_s = bs
                    best_D_k = bk.copy()
                if bs < best_gs - 1e-12:
                    best_gs = bs
                    best_global = decode_lattice(bk, D).copy()
            except Exception:
                continue

        if best_D_k is not None:
            prev_best_k = best_D_k
            if best_D_s < best_gs - 1e-12:
                best_gs = best_D_s
                best_global = decode_lattice(best_D_k, D).copy()

    final_h = np.array(best_global, dtype=np.float64)
    if not is_feasible(final_h, n):
        final_h = capped_simplex_projection(final_h, n)
    fs = exact_score(final_h, n)
    if not math.isfinite(fs) or fs > fb_score + 1e-12:
        final_h = fallback.copy()
        fs = fb_score
    return np.array(final_h, dtype=np.float64), float(fs), int(n)
