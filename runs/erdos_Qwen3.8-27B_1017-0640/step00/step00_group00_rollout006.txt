import numpy as np
from scipy.optimize import minimize

def capped_simplex_project(y, n, m):
    y = np.clip(np.asarray(y, dtype=np.float64), -1e6, 1e6)
    lo, hi = y.min() - 1.0, y.max()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > m:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - x.sum()
    if r > 0:
        room = 1.0 - x
        for i in range(n):
            add = min(room[i], r)
            x[i] += add
            r -= add
            if r <= 1e-12:
                break
    elif r < 0:
        r = -r
        for i in range(n):
            sub = min(x[i], r)
            x[i] -= sub
            r -= sub
            if r <= 1e-12:
                break
    if not np.all(np.isfinite(x)) or x.min() < -1e-12 or x.max() > 1 + 1e-12 or abs(x.sum() - m) > 1e-9:
        return np.full(n, m / n)
    x = np.clip(x, 0.0, 1.0)
    return x

def is_feasible(h, n, m):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if h.min() < -1e-12 or h.max() > 1.0 + 1e-12:

def exact_score(h, n):

def ineq_jac(z):
    n = z.shape[0] - 1
    h = z[:n]
    dx = 2.0 / n
    nlag = 2 * n - 1
    jac = np.zeros((nlag, n + 1))
    jac[:, -1] = 1.0  # d/dt = 1 for all rows
    for k in range(nlag):
        lag = k - (n - 1)
        # d(c_k)/d(h[r]) = [0<=r-lag<n]*(1-h[r-lag]) - [0<=r+lag<n]*h[r+lag]
        # d(ineq)/d(h[r]) = -dx * d(c_k)/d(h[r])
        # = -dx*[0<=r-lag<n]*(1-h[r-lag]) + dx*[0<=r+lag<n]*h[r+lag]
        
        # First term: r-lag in [0,n) → r in [lag, n+lag), intersected with [0,n)
        r_start1 = max(0, lag)
        r_end1 = min(n, n + lag)
        if r_start1 < r_end1:
            idx = np.arange(r_start1, r_end1)
            jac[k, idx] += -dx * (1.0 - h[idx - lag])
        
        # Second term: r+lag in [0,n) → r in [-lag, n-lag), intersected with [0,n)
        r_start2 = max(0, -lag)
        r_end2 = min(n, n - lag)
        if r_start2 < r_end2:
            idx = np.arange(r_start2, r_end2)
            jac[k, idx] += dx * h[idx + lag]
    return jac

import numpy as np
from scipy.optimize import minimize

def capped_simplex_project(y, n, m):
    y = np.clip(np.asarray(y, dtype=np.float64).flatten(), -1e6, 1e6)
    lo, hi = y.min() - 1.0, y.max()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - x.sum()
    if r > 0:
        for i in range(n):
            add = min(1.0 - x[i], r)
            x[i] += add
            r -= add
            if r <= 1e-12:
                break
    elif r < 0:
        r = -r
        for i in range(n):
            sub = min(x[i], r)
            x[i] -= sub
            r -= sub
            if r <= 1e-12:
                break
    if not np.all(np.isfinite(x)) or x.min() < -1e-12 or x.max() > 1.0 + 1e-12 or abs(x.sum() - m) > 1e-9:
        return np.full(n, m / n)
    return np.clip(x, 0.0, 1.0)

def is_feasible(h, n, m):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and h.min() >= -1e-12
            and h.max() <= 1.0 + 1e-12 and abs(h.sum() - m) <= 1e-9)

def exact_score(h, n):
    h = np.asarray(h, dtype=np.float64)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(c)):
        return np.inf
    return float(c.max())

def obj_fn(z):
    return z[-1]

def obj_jac(z):
    g = np.zeros(z.shape[0])
    g[-1] = 1.0
    return g

def eq_con(z):
    n = z.shape[0] - 1
    return z[:n].sum() - 0.5 * n

def eq_jac(z):
    n = z.shape[0] - 1
    g = np.ones(n + 1)
    g[-1] = 0.0
    return g

def ineq_con(z):
    n = z.shape[0] - 1
    h = z[:n]
    return z[-1] - np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def ineq_jac(z):
    n = z.shape[0] - 1
    h = z[:n]
    dx = 2.0 / n
    nlag = 2 * n - 1
    jac = np.zeros((nlag, n + 1))
    jac[:, -1] = 1.0
    for k in range(nlag):
        lag = k - (n - 1)
        s1, e1 = max(0, lag), min(n, n + lag)
        if s1 < e1:
            idx = np.arange(s1, e1)
            jac[k, idx] += -dx * (1.0 - h[idx - lag])
        s2, e2 = max(0, -lag), min(n, n - lag)
        if s2 < e2:
            idx = np.arange(s2, e2)
            jac[k, idx] += dx * h[idx + lag]
    return jac

def block_vector(n, m):
    b = np.zeros(n)
    nf = int(m)
    b[:nf] = 1.0
    if m - nf > 1e-12:
        b[nf] = m - nf
    return b

def run(seed=42, **kwargs):
    raw = globals().get('initial_h_values', None)
    if raw is not None and np.ndim(raw) == 1:
        n = len(raw)
    else:
        n = 45
    if 'n_points' in kwargs:
        try:
            no = int(kwargs['n_points'])
            if 1 <= no <= 999:
                n = no
        except (ValueError, TypeError):
            pass
    n = int(n)
    m = 0.5 * n
    rng = np.random.default_rng(seed)

    parent = None
    if raw is not None and np.ndim(raw) == 1:
        pr = np.array(raw, dtype=np.float64).flatten()
        if pr.shape[0] != n:
            xo = np.linspace(0, 1, pr.shape[0])
            xn = np.linspace(0, 1, n)
            pr = np.interp(xn, xo, pr)
        if np.all(np.isfinite(pr)):
            parent = capped_simplex_project(pr, n, m)
    if parent is None:
        parent = np.full(n, m / n)

    blk = block_vector(n, m)
    starts = [parent.copy(),
              capped_simplex_project(1.0 - parent, n, m),
              parent[::-1].copy(),
              capped_simplex_project(np.roll(parent, n // 3), n, m),
              np.full(n, m / n).copy(),
              blk.copy(),
              blk[::-1].copy()]
    for _ in range(3):
        starts.append(capped_simplex_project(rng.random(n), n, m))
    for sg in [0.05, 0.18]:
        starts.append(capped_simplex_project(parent + sg * rng.standard_normal(n), n, m))
    for rfreq in [1, 3]:
        ph = rng.uniform(0, 2 * np.pi)
        idx = np.arange(n)
        sv = 0.5 + 0.35 * np.cos(2.0 * np.pi * rfreq * idx / n + ph)
        starts.append(capped_simplex_project(sv, n, m))

    uniq = [starts[0]]
    for s in starts[1:]:
        if not any(np.allclose(s, u, atol=1e-10) for u in uniq):
            uniq.append(s)

    if n <= 100:
        mx, mit, rnds = 10, 250, 2
    elif n <= 300:
        mx, mit, rnds = 7, 120, 1
    else:
        mx, mit, rnds = 4, 60, 1
    if len(uniq) > mx:
        uniq = [uniq[i] for i in range(min(mx, len(uniq)))]

    bounds = [(0.0, 1.0)] * (n + 1)
    cons = [{'type': 'eq', 'fun': eq_con, 'jac': eq_jac},
            {'type': 'ineq', 'fun': ineq_con, 'jac': ineq_jac}]

    best = parent.copy()
    best_sc = exact_score(parent, n)

    for s in uniq:
        if not is_feasible(s, n, m):
            s = capped_simplex_project(s, n, m)
            if not is_feasible(s, n, m):
                continue
        t0 = min(1.0, exact_score(s, n) + 1e-9)
        z0 = np.append(s, t0)
        try:
            res = minimize(obj_fn, z0, jac=obj_jac, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': mit, 'ftol': 1e-10, 'disp': 0})
            if res is not None and hasattr(res, 'x') and np.all(np.isfinite(res.x)):
                ht = capped_simplex_project(res.x[:n], n, m)
                if is_feasible(ht, n, m):
                    sc = exact_score(ht, n)
                    if sc < best_sc - 1e-12:
                        best, best_sc = ht, sc
        except Exception:
            pass

    if n <= 100:
        sg_list, n_rnd = [0.03, 0.10, 0.25], rnds
    elif n <= 300:
        sg_list, n_rnd = [0.05, 0.18], rnds
    else:
        sg_list, n_rnd = [0.08], rnds

    for _ in range(n_rnd):
        sc0 = best_sc
        for sg in sg_list:
            y = best + sg * rng.standard_normal(n)
            x0 = capped_simplex_project(y, n, m)
            if not is_feasible(x0, n, m):
                continue
            t0 = min(1.0, exact_score(x0, n) + 1e-9)
            z0 = np.append(x0, t0)
            try:
                res = minimize(obj_fn, z0, jac=obj_jac, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': mit, 'ftol': 1e-10, 'disp': 0})
                if res is not None and hasattr(res, 'x') and np.all(np.isfinite(res.x)):
                    ht = capped_simplex_project(res.x[:n], n, m)
                    if is_feasible(ht, n, m):
                        sc = exact_score(ht, n)
                        if sc < best_sc - 1e-12:
                            best, best_sc = ht, sc
            except Exception:
                pass
        if best_sc > sc0 - 1e-9:
            break

    best = capped_simplex_project(best, n, m)
    if not is_feasible(best, n, m):
        best = parent.copy()
        if not is_feasible(best, n, m):
            best = np.full(n, m / n)
    best_sc = exact_score(best, n)
    return (best, best_sc, n)
