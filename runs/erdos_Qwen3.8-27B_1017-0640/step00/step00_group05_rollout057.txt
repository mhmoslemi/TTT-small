    except Exception:
        pass
    if N < 4:
        N = 70
        parent = None
    m = N / 2.0
    neutral = np.full(N, 0.5)
    c_neut = compute_score(neutral, N)
    if parent is not None:
        if parent.min() >= -1e-10 and parent.max() <= 1.0 + 1e-10 and abs(parent.sum() - m) < 1e-8:
            p0 = np.clip(parent, 0.0, 1.0)
        else:
            p0 = project_box_sum(parent, m)
    else:
        p0 = neutral.copy()
    best_h = p0.copy()
    best_c = compute_score(best_h, N)
    if c_neut < best_c:
        best_h = neutral.copy()
        best_c = c_neut
    rng = np.random.default_rng(seed)
    starts = [p0.copy()]
    if N <= 256:
        for sc in (0.03, 0.08, 0.18):
            starts.append(project_box_sum(p0 + sc * rng.standard_normal(N), m))
        starts.append(project_box_sum(rng.uniform(0, 1, N), m))
        starts.append(project_box_sum(rng.beta(0.7, 0.7, N), m))
        i = np.arange(N) + 0.5
        starts.append(project_box_sum(0.5 + 0.45 * np.cos(2.0 * np.pi * i / N), m))
        starts.append(project_box_sum(0.5 + 0.45 * np.cos(2.0 * np.pi * 2.0 * i / N + np.pi / 3.0), m))
    elif N <= 999:
        starts.append(project_box_sum(p0 + 0.08 * rng.standard_normal(N), m))
        starts.append(project_box_sum(rng.uniform(0, 1, N), m))
        starts.append(project_box_sum(rng.beta(0.7, 0.7, N), m))
    for s in starts:
        c = compute_score(s, N)
        if c < best_c - 1e-12:
            best_h = s.copy()
            best_c = c
    if N <= 999:
        bounds = [(0.0, 1.0)] * N + [(0.0, 2.0)]
        con_eq = {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (m, N)}
        con_ineq = {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac, 'args': (N,)}
        for s in starts:
            t0 = float(np.clip(corr_q(s, N).max(), 0.0, 2.0))
            z0 = np.concatenate((s, [t0]))
            try:
                res = minimize(obj_func, z0, jac=obj_grad, method='SLSQP', bounds=bounds,
                               constraints=[con_eq, con_ineq],
                               options={'maxiter': 250, 'ftol': 1e-10, 'disp': False})
                if res.x is not None and res.x.size == N + 1 and np.all(np.isfinite(res.x)):
                    h_c = project_box_sum(res.x[:N], m)
                    c_c = compute_score(h_c, N)
                    if c_c < best_c - 1e-12:
                        best_h = h_c.copy()
                        best_c = c_c
            except Exception:
                continue
    best_h = project_box_sum(best_h, m)
    final_c = compute_score(best_h, N)
    if not np.all(np.isfinite(best_h)) or best_h.min() < -1e-10 or best_h.max() > 1.0 + 1e-10:
        best_h = neutral
        final_c = c_neut
    return best_h, final_c, N
