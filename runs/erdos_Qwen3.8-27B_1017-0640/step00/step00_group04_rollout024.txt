import numpy as np

def project_capped_simplex(y, n):
    target = n / 2.0
    y = np.asarray(y, dtype=np.float64)
    low = float(np.min(y)) - 1.0
    high = float(np.max(y))
    for _ in range(80):
        mid = (low + high) / 2.0
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) / 2.0
    x = np.clip(y - lam, 0.0, 1.0)
    # Correct residual
    for _ in range(3):
        r = target - float(np.sum(x))
        if abs(r) <= 5e-13 * max(1, n):
            break
        if r > 0:
            for i in range(n):
                if r <= 0: break
                add = min(r, 1.0 - x[i])
                x[i] += add
                r -= add
        else:
            for i in range(n):
                if r >= 0: break
                sub = min(-r, x[i])
                x[i] -= sub
                r += sub
    return x

def is_feasible(h, n):

def line_val(t, q0, b, a, n):

    for sw in range(max_sweeps):
        improved = False
        rng.shuffle(pairs)
        nuse = npairs if n <= 64 else min(npairs, 4 * n)
        for idx in range(nuse):
            i, j = pairs[idx]
            d = np.zeros(n, dtype=np.float64)
            d[i] = 1.0
            d[j] = -1.0
            res = linsearch(h, d, q0, n, gpts, ngold)
            if res is None:
                continue
            bt, bv = res[0], res[1]
            if bv >= sc - 1e-12:
                continue
            cand = h + bt * d
            cand = proj(cand, n)
            if not feasible(cand, n):
                continue
            cs = esc(cand, n)
            if cs is None or cs >= sc - 1e-12:
                continue
            h = cand
            sc = cs
            q0 = np.correlate(h, 1.0 - h, mode='full')
            improved = True
        for _ in range(nrand):
            z = rng.standard_normal(n)
            z -= z.mean()
            z[-1] = -z[:-1].sum()
            zmax = np.max(np.abs(z))
            if zmax < 1e-15:
                continue
            z /= zmax
            res = linsearch(h, z, q0, n, gpts, ngold)
            if res is None:
                continue
            bt, bv = res[0], res[1]
            if bv >= sc - 1e-12:
                continue
            cand = h + bt * z
            cand = proj(cand, n)
            if not feasible(cand, n):
                continue
            cs = esc(cand, n)
            if cs is None or cs >= sc - 1e-12:
                continue
            h = cand
            sc = cs
            q0 = np.correlate(h, 1.0 - h, mode='full')
            improved = True
        if not improved:
            break
    return h, sc

def run(seed=42, **kwargs):
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    rng = np.random.default_rng(seed)
    ns = 6
    ms = 3
    try:
        ns = max(4, min(8, int(kwargs.get('n_starts', 6))))
    except Exception:
        ns = 6
    try:
        ms = max(1, min(5, int(kwargs.get('max_sweeps', 3))))
    except Exception:
        ms = 3
    n = 42
    fallback = None
    iva = globals().get('initial_h_values', None)
    if iva is not None:
        try:
            iva = np.asarray(iva, dtype=np.float64).ravel()
            if 2 <= len(iva) <= 4096 and np.all(np.isfinite(iva)) and np.all(np.abs(iva) <= 1e6):
                n = len(iva)
                if feasible(iva, n):
                    fallback = iva.copy()
                    if abs(float(fallback.sum()) - n / 2.0) > 5e-13 * max(1, n):
                        fallback = proj(fallback, n)
                else:
                    rep = proj(iva, n)
                    if feasible(rep, n):
                        fallback = rep
        except Exception:
            fallback = None
    if fallback is None:
        n = 42
        fallback = np.zeros(n, dtype=np.float64)
        fallback[:n // 2] = 1.0
        if n % 2 == 1:
            fallback[n // 2] = 0.5
    fallback = proj(fallback, n)
    fb_score = esc(fallback, n)
    if fb_score is None:
        fb_score = 1.0
    best_h = fallback.copy()
    best_sc = fb_score
    gpts = 17
    ngold = 12
    starts = [fallback]
    bl = np.zeros(n, dtype=np.float64)
    bl[:n // 2] = 1.0
    if n % 2 == 1:
        bl[n // 2] = 0.5
    v = addstart(bl, n, starts)
    if v is not None:
        starts.append(v)
    v = addstart(bl[::-1].copy(), n, starts)
    if v is not None:
        starts.append(v)
    v = addstart(1.0 - fallback, n, starts)
    if v is not None:
        starts.append(v)
    for sd in [0.15, 0.05, 0.20, 0.35, 0.50]:
        v = addstart(fallback + sd * rng.standard_normal(n), n, starts)
        if v is not None:
            starts.append(v)
        if len(starts) >= ns:
            break
    if len(starts) < ns:
        v = addstart(rng.uniform(-0.25, 1.25, size=n), n, starts)
        if v is not None:
            starts.append(v)
    if len(starts) < ns:
        tv = np.arange(n, dtype=np.float64)
        v = addstart(0.5 + 0.25 * np.cos(2.0 * np.pi * tv / n), n, starts)
        if v is not None:
            starts.append(v)
    trim = n * (n - 1) // 2
    for si in range(min(ns, len(starts))):
        try:
            h0 = starts[si]
            h1, s1 = localsearch(h0, n, rng, ms, gpts, ngold)
            if s1 < best_sc - 1e-12:
                best_h = h1
                best_sc = s1
        except Exception:
            pass
    if not feasible(best_h, n):
        best_h = proj(best_h, n)
    fs = esc(best_h, n)
    if fs is None or not feasible(best_h, n):
        best_h = fallback.copy()
        fs = esc(best_h, n)
        if fs is None:
            fs = fb_score
    return best_h, float(fs), int(n)
