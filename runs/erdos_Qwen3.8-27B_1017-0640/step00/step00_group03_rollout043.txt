def check_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):

def joint_project(v, n):
    v = np.array(v, dtype=float).ravel()
    if v.size != n or not np.all(np.isfinite(v)):
        return None
    v = np.clip(v, -1e6, 1e6)
    target = n / 2.0
    lo, hi = v.min() - 1.0, v.max()
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    q = np.clip(v - (lo + hi) / 2.0, 0, 1)

    for _ in range(2):
        if res > 1e-14:
            for i in range(n):
                if res <= 0:
                    break
                a = min(1.0 - q[i], res)
                q[i] += a; res -= a
        elif res < -1e-14:
            for i in range(n):
                if res >= 0:
                    break
                s = min(q[i], -res)
                q[i] -= s; res += s
    q = np.clip(q, 0, 1)
    if abs(q.sum() - target) > 1e-8:
        return None
    return q


def exact_score(h):
    n = len(h)
    if not check_feasible(h, n):
        return float('inf')
    corr = np.correlate(h, 1.0 - h, mode='full')
    if corr.shape != (2 * n - 1,) or not np.all(np.isfinite(corr)):
        return float('inf')
    return float(np.max(corr * (2.0 / n)))


def compute_lags_J(h):
    n = len(h)
    dx = 2.0 / n
    c = np.zeros(2 * n - 1)
    J = np.zeros((2 * n - 1, n))
    for d in range(1, n):
        u, v = h[:n - d], h[d:]
        iB = n - 1 - d
        c[iB] = np.sum(u * (1.0 - v)) * dx
        J[iB, :n - d] += (1.0 - v)
        J[iB, d:] -= u
        iA = n - 1 + d
        c[iA] = np.sum(v * (1.0 - u)) * dx
        J[iA, d:] += (1.0 - u)
        J[iA, :n - d] -= v
    c[n - 1] = np.sum(h * (1.0 - h)) * dx
    J[n - 1, :] = (1.0 - 2.0 * h) * dx
    if not np.all(np.isfinite(c)) or not np.all(np.isfinite(J)):
        return None, None
    return c, J


def lp_step(y, c, J, r, mu):
    n = len(y)
    nl = J.shape[0]
    c_obj = np.concatenate([np.zeros(n), mu * np.ones(n), [1.0]])

    b_ub = np.zeros(nl + 2 * n)
    A_ub[:nl, :n] = J
    A_ub[:nl, 2 * n] = -1.0
    b_ub[:nl] = J @ y - c
    for i in range(n):
        row = nl + 2 * i
        A_ub[row, i] = 1.0
        A_ub[row, n + i] = -1.0
        b_ub[row] = y[i]
        A_ub[row + 1, i] = -1.0
        A_ub[row + 1, n + i] = -1.0
        b_ub[row + 1] = -y[i]
    A_eq = np.zeros((1, 2 * n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    lb = np.concatenate([np.maximum(0.0, y - r), np.zeros(n), [0.0]])
    ub = np.concatenate([np.minimum(1.0, y + r), np.full(n, np.inf), [np.inf]])
