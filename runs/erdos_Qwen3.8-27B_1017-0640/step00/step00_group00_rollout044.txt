    G = np.zeros((2*n-1, n))
    G[n-1] = dx*(1.0-2.0*h)
    for p in range(1, n):
        G[n-1+p, p:] += dx*(1.0-h[:n-p])
        G[n-1+p, :n-p] -= dx*h[p:]
        G[n-1-p, :n-p] += dx*(1.0-h[p:])
        G[n-1-p, p:] -= dx*h[:n-p]
    return G

def _mm_cvxpy(x, q, G, dx, n):
    try:
        import cvxpy as cp
        d = cp.Variable(n)
        s = cp.Variable(nonneg=True)
        t = cp.Variable()
        L = 2*n-1
        cons = [d >= -x, d <= 1.0-x, cp.sum(d) == 0]
        cons.append(cp.sum_squares(d) <= s)
        cons.append(q + G@d + dx*s <= t)
        prob = cp.Problem(cp.Minimize(t), cons)

                    if np.all(np.isfinite(dv)) and dv.shape == (n,):

def _project(y, n, m):
    if y is None or len(y) != n:
        return None
    y = np.asarray(y, dtype=np.float64)
    if not np.all(np.isfinite(y)):
        return None

    for _ in range(80):
        mid = 0.5 * (lo_l + hi_l)
        if np.sum(np.clip(y - mid, 0.0, 1.0)) > m:
            lo_l = mid
        else:
            hi_l = mid
    x = np.clip(y - 0.5 * (lo_l + hi_l), 0.0, 1.0)
    for _ in range(5):
        res = m - x.sum()
        if abs(res) < 1e-13:
            break
        if res > 0:
            for i in range(n):
                a = min(1.0 - x[i], res)
                x[i] += a; res -= a
                if abs(res) < 1e-13:
                    break
        else:
            for i in range(n):
                a = min(x[i], -res)
                x[i] -= a; res += a

    if h.shape != (n,) or not np.all(np.isfinite(h)):

    raw = np.correlate(h, 1.0 - h, mode='full')
    if raw.shape[0] != 2 * n - 1 or not np.all(np.isfinite(raw)):
        return None
    return float(np.max(raw * dx))

def _grad(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    G[n - 1] = dx * (1.0 - 2.0 * h)
    for p in range(1, n):
        G[n - 1 + p, p:] += dx * (1.0 - h[:n - p])
        G[n - 1 + p, :n - p] -= dx * h[p:]
        G[n - 1 - p, :n - p] += dx * (1.0 - h[p:])
        G[n - 1 - p, p:] -= dx * h[:n - p]
    return G
