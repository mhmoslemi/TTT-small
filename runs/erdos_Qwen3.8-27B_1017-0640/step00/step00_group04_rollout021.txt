import numpy as np
from numpy import correlate
import math

def project_to_simplex(y, n):
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        x = np.clip(y - mid, 0.0, 1.0)
        if np.sum(x) > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    x = np.clip(y - lam, 0.0, 1.0)
    r = target - np.sum(x)
    for _ in range(3):
        if abs(r) <= 5e-13 * max(1, n):
            break
        if r > 0:
            for i in range(n):
                add = min(r, 1.0 - x[i])
                x[i] += add
                r -= add
                if abs(r) <= 5e-13:
                    break
        else:
            for i in range(n):
                sub = min(-r, x[i])
                x[i] -= sub
                r += sub
                if abs(r) <= 5e-13:
                    break
    if not np.all(np.isfinite(x)) or np.any(x < -1e-12) or np.any(x > 1.0 + 1e-12):
        return None
    x = np.clip(x, 0.0, 1.0)
    if abs(np.sum(x) - target) > 1e-10 * max(1, n):
        return None
    return x

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1, n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return None
    corr = correlate(h, 1.0 - h, mode='full')
    if corr.shape != (2 * n - 1,):
        return None
    s = float(np.max(corr)) * (2.0 / n)
    if not np.isfinite(s):
        return None
    return s

def feasible_interval(h, d):
    t_lo, t_hi = 0.0, 0.0
    for i in range(len(h)):
        di = d[i]
        if di > 1e-15:
            t_lo = max(t_lo, -h[i] / di)
            t_hi = min(t_hi, (1.0 - h[i]) / di)
        elif di < -1e-15:
            t_lo = max(t_lo, (1.0 - h[i]) / di)
            t_hi = min(t_hi, -h[i] / di)
    if not (np.isfinite(t_lo) and np.isfinite(t_hi)):
        return None
    if t_hi - t_lo <= 1e-12:
        return None
    return (t_lo, t_hi)

def line_value(t, q0, b, a, n):
    vals = q0 + t * b + t * t * a
    return float(np.max(vals)) * (2.0 / n)

def golden_section(fn, lo, hi, n_iter=12):
    r = (math.sqrt(5.0) - 1.0) / 2.0
    x1 = hi - r * (hi - lo)
    x2 = lo + r * (hi - lo)
    f1, f2 = fn(x1), fn(x2)
    best_t, best_v = (x1, f1) if f1 < f2 else (x2, f2)
    for _ in range(n_iter):
        if f1 < f2:
            hi = x2
            x2 = x1
            f2 = f1
            x1 = hi - r * (hi - lo)
            f1 = fn(x1)
        else:
            lo = x1
            x1 = x2
            f1 = f2
            x2 = lo + r * (hi - lo)
            f2 = fn(x2)
        if f1 < best_v:
            best_t, best_v = x1, f1
        if f2 < best_v:
            best_t, best_v = x2, f2
    return best_t, best_v

def search_direction(h, d, n, cur_score, cur_corr):

    if interval is None:
        return None
    t_lo, t_hi = interval
    q0 = cur_corr
    b = correlate(d, 1.0 - h, mode='full') - correlate(h, d, mode='full')
    a = -correlate(d, d, mode='full')
    if not (np.all(np.isfinite(q0)) and np.all(np.isfinite(b)) and np.all(np.isfinite(a))):
        return None

    cands = list(range(len(grid)))
    for k in range(1, len(grid) - 1):
        if vals[k] < vals[k - 1] and vals[k] < vals[k + 1]:
            if k not in cands:
                cands.append(k)
    gk = int(np.argmin(vals))
    if 0 < gk < len(grid) - 1 and gk not in cands:
        cands.append(gk)
    cands.sort(key=lambda k: (vals[k], k))
    best_t, best_v = 0.0, cur_score
    for ci in cands[:2]:

        hi_b = grid[min(ci + 1, len(grid) - 1)]
        if hi_b - lo_b <= 1e-14:
            lo_b, hi_b = t_lo, t_hi
        ft, fv = golden_section(lambda t: line_value(t, q0, b, a, n), lo_b, hi_b, 12)
        if fv < best_v:

    for k in cands[:2]:
        if vals[k] < best_v:
            best_t, best_v = float(grid[k]), float(vals[k])
    if best_v >= cur_score - 1e-12:
        return None
    hnew = h + best_t * d
    hnew = project_to_simplex(hnew, n)
    if hnew is None:

        return None

    new_corr = correlate(hnew, 1.0 - hnew, mode='full')

def build_starts(n, rng, fallback):
    starts = [fallback.copy()]
    proposals = []
    block = np.zeros(n)
    block[:n // 2] = 1.0
    if n % 2 == 1:
        block[n // 2] = 0.5
    proposals.append(block)
    proposals.append(block[::-1].copy())
    proposals.append(1.0 - fallback)
    proposals.append(fallback + 0.15 * rng.standard_normal(n))
    proposals.append(rng.uniform(-0.25, 1.25, size=n))
    cs = 0.5 + 0.25 * np.cos(2.0 * np.pi * np.arange(n) / n)
    proposals.append(cs)
    proposals.append(np.full(n, 0.5))
    for sd in [0.05, 0.20, 0.35, 0.50]:
        proposals.append(fallback + sd * rng.standard_normal(n))
    for p in proposals:
        if len(starts) >= 6:
            break
        x = project_to_simplex(p.astype(float64) if hasattr(float, '64') else p.astype(np.float64), n)

        dup = False
        for s in starts:
            if np.max(np.abs(x - s)) < 1e-9:
                dup = True
                break
        if not dup:
            starts.append(x)

def local_search(h, n, rng, max_sweeps):
    sc = exact_score(h, n)
    if sc is None:
        return h, 1e9
    corr = correlate(h, 1.0 - h, mode='full')
    for sweep in range(max_sweeps):
        accepted = False
        if n <= 64:

            idx = list(range(len(pairs)))
            rng.shuffle(idx)
            for ii in idx:
                i, j = pairs[ii]
                d = np.zeros(n)
                d[i], d[j] = 1.0, -1.0

        else:

            res = search_direction(h, z, n, sc, corr)

                accepted = True

            break
    h = np.clip(h, 0.0, 1.0)
    h = project_to_simplex(h, n)
    if h is None or not is_feasible(h, n):
        return None, sc
    final_sc = exact_score(h, n)

    return h, final_sc
