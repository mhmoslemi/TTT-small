def compute_qk_grads(h, N, dx):
    # Returns (2N-1, N) matrix where row k_offset corresponds to lag k = k_offset - (N-1)
    Nl = 2*N - 1
    J = np.zeros((Nl, N))
    for ko in range(Nl):
        k = ko - (N - 1)
        # First term: (1-h[r-k]) for 0<=r-k<N
        # r-k >= 0 => r >= k, r-k < N => r < N+k
        lo1 = max(0, k)
        hi1 = min(N, N + k)
        if lo1 < hi1:
            J[ko, lo1:hi1] += dx * (1.0 - h[lo1-k:hi1-k])
        # Second term: -h[r+k] for 0<=r+k<N
        # r+k >= 0 => r >= -k, r+k < N => r < N-k
        lo2 = max(0, -k)
        hi2 = min(N, N - k)
        if lo2 < hi2:
            J[ko, lo2:hi2] -= dx * h[lo2+k:hi2+k]
    return J

                        best_h = h_new.copy()
            except Exception:
                continue

    best_h = project_box_sum(best_h, m, N)
    best_c = compute_score(best_h, N)
    if (not np.all(np.isfinite(best_h)) or np.any(best_h < -1e-10)
            or np.any(best_h > 1.0 + 1e-10) or abs(best_h.sum() - m) > 1e-8):
        best_h = h_fallback
        best_c = compute_score(best_h, N)

    return best_h, best_c, N
