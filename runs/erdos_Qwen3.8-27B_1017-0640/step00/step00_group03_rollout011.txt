    if parent is not None and np.all(np.isfinite(parent)) and np.all(parent >= -1e-12) and np.all(parent <= 1.0 + 1e-12) and abs(np.sum(parent) - n / 2.0) <= 1e-10:
        fb = np.clip(parent.copy(), 0.0, 1.0)
    elif parent is not None and parent.size == n:
        fb = project_capped_simplex(parent, n)
    else:
        fb = np.full(n, 0.5)
    fb_score = exact_score(fb, n)

    inc_h = fb.copy()
    inc_score = fb_score

    if n > 256:
        return inc_h, inc_score, n

    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)

    starts = make_starts(n, parent, rng)
    for s in starts:
        sc = exact_score(s, n)
        if sc < inc_score - 1e-12:
            inc_score = sc
            inc_h = s.copy()

    for s in starts:
        inc_h, inc_score = grad_search(s, n, inc_h, inc_score)

    if n <= 128:
        shake = project_capped_simplex(inc_h + 0.04 * rng.standard_normal(n), n)
        dup = any(np.max(np.abs(shake - s)) < 1e-10 for s in starts)
        if not dup and validate(shake, n):
            inc_h, inc_score = grad_search(shake, n, inc_h, inc_score)

    if not validate(inc_h, n):
        inc_h = project_capped_simplex(inc_h, n)
    fs = exact_score(inc_h, n)
    if fb_score < fs - 1e-12:
        inc_h, fs = fb.copy(), fb_score
    if not validate(inc_h, n):
        inc_h = np.full(n, 0.5)
        fs = exact_score(inc_h, n)
    return inc_h, fs, n
