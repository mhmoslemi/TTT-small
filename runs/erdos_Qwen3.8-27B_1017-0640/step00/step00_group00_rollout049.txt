def is_feasible(h, n):

def cap_proj(y, n):
    if y.shape != (n,) or not np.all(np.isfinite(y)):
        return None
    m = n/2.
    y = np.clip(y, -1e6, 1e6)
    lo, hi = y.min()-1., y.max()

        if np.sum(np.clip(y-mid, 0., 1.)) > m: lo = mid
        else: hi = mid
    x = np.clip(y-(lo+hi)/2., 0., 1.).astype(np.float64).copy()

    if x.min()<-1e-10 or x.max()>1.+1e-10 or abs(x.sum()-m)>1e-10: return None
    return x

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)): return None
    raw = np.correlate(h, 1.-h, 'full')
    if raw.size != 2*n-1 or not np.all(np.isfinite(raw)): return None
    return float(np.max(raw*(2./n)))

def lag_grads(h, n):
    dx = 2./n
    L = 2*n-1
    G = np.zeros((L, n))
    G[n-1,:] = dx*(1.-2.*h)
    for p in range(1, n):
        G[n-1+p, p:] += dx*(1.-h[:n-p])
        G[n-1+p, :n-p] -= dx*h[p:]
        G[n-1-p, :n-p] += dx*(1.-h[p:])
        G[n-1-p, p:] -= dx*h[:n-p]
    return G

def build_cvx(n):
    L = 2*n-1; dx = 2./n
    d = cp.Variable(n); s = cp.Variable(nonneg=True); t = cp.Variable()
    q_p = cp.Parameter(L); G_p = cp.Parameter((L, n))
    lo_p = cp.Parameter(n); hi_p = cp.Parameter(n)
    cons = [d >= lo_p, d <= hi_p, cp.sum(d)==0, cp.sum_squares(d) <= s,
            q_p + G_p @ d + dx*s <= t]
    return cp.Problem(cp.Minimize(t), cons), d, s, t, q_p, G_p, lo_p, hi_p

def _o(z, dt): return z[-1]
def _og(z, dt):
    g = np.zeros(len(z)); g[-1] = 1.; return g

def _eq(z, dt): return float(z[:dt[0]].sum())
def _eqj(z, dt):
    j = np.zeros(len(z)); j[:dt[0]] = 1.; return j

def _in(z, dt):
    n, q, G, dx = dt
    d, s, t = z[:n], z[n], z[-1]
    out = np.empty(len(q)+1)
    out[0] = s - d.dot(d)
    out[1:] = t - q - G.dot(d) - dx*s
    return out

def _inj(z, dt):
    n, q, G, dx = dt
    L = len(q)
    out = np.zeros((L+1, n+2))
    out[0, :n] = -2.*z[:n]
    out[0, n] = 1.
    out[1:, :n] = -G
    out[1:, n] = -dx
    out[1:, -1] = 1.
    return out

def _mm_scipy(x, n):
    dx = 2./n
    q = np.correlate(x, 1.-x, 'full')*dx
    G = lag_grads(x, n)
    dat = (n, q, G, dx)
    bounds = [(-x[i], 1.-x[i]) for i in range(n)] + [(0., float(n)), (-1e10, 1e10)]
    cons = [{'type':'eq','fun':_eq,'jac':_eqj,'args':dat},
            {'type':'ineq','fun':_in,'jac':_inj,'args':dat}]
    maxi = 100 if n<=64 else 60 if n<=160 else 35 if n<=400 else 20
    z0 = np.zeros(n+2); z0[-1] = float(q.max())
    try:
        res = minimize(_o, z0, jac=_og, method='SLSQP', bounds=bounds,
                       constraints=cons, args=dat, options={'maxiter':maxi,'ftol':1e-9,'disp':False})
        if res.x[:n].shape == (n,) and np.all(np.isfinite(res.x[:n])):
            return res.x[:n].copy()
    except Exception:
        pass
    return None

def _mm_cvx(x, n, cvx):

    q = np.correlate(x, 1.-x, 'full')*(2./n)
