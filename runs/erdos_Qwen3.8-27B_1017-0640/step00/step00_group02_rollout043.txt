    pilot_count = max(8, min(48, int(1e6 / (n * n))))
    M = max(1, min(M_cap, int(1.8e8 / (n * n * L * max(1, Nchains)))))
    for s in deduped:
        ss = exact_score(s, n, S)
        if ss is None:
            continue
        try:
            incumbent, incumbent_score = anneal_chain(s, ss, n, S, rng, L, M, pilot_count, incumbent, incumbent_score, p_perm, p_swap)
        except Exception:
            pass
    if n <= 64:
        rstds = [0.10, 0.25]
    elif n <= 128:
        rstds = [0.15]
    else:
        rstds = []
    for std in rstds:
        try:
            rp = project_capped(incumbent + rng.normal(0, std, n), S)
            if rp is None or not feasible(rp, n, S):
                continue
            rsc = exact_score(rp, n, S)
            if rsc is None:
                continue
            incumbent, incumbent_score = anneal_chain(rp, rsc, n, S, rng, L, M, pilot_count, incumbent, incumbent_score, p_perm, p_swap)
        except Exception:
            pass
    if not feasible(incumbent, n, S):
        rep = project_capped(incumbent, S)
        if rep is not None and feasible(rep, n, S):
            incumbent = rep
    final_score = exact_score(incumbent, n, S)
    if final_score is None or final_score > fallback_score:
        incumbent = fallback
        final_score = fallback_score
    return incumbent.astype(np.float64), final_score, n
