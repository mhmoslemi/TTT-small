    if np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):

    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def exact_score(h, n):
    if not feasible_check(h, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lag_values(h, n):
    dx = 2.0 / n
    q = np.empty(2 * n - 1, dtype=np.float64)
    q[n - 1] = dx * np.sum(h * (1.0 - h))
    for s in range(1, n):
        ov = n - s
        q[n - 1 - s] = dx * np.sum(h[:ov] * (1.0 - h[s:]))
        q[n - 1 + s] = dx * np.sum(h[s:] * (1.0 - h[:ov]))
    return q

def lag_jacobian(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n), dtype=np.float64)
    G[n - 1, :] = dx * (1.0 - 2.0 * h)
    for s in range(1, n):
        ov = n - s
        row = np.zeros(n)
        row[:ov] += dx * (1.0 - h[s:s + ov])
        row[s:] -= dx * h[:ov]
        G[n - 1 - s, :] = row
        row2 = np.zeros(n)
        row2[s:] += dx * (1.0 - h[:ov])
        row2[:ov] -= dx * h[s:s + ov]
        G[n - 1 + s, :] = row2
    return G

def smooth_val(q, tau):
    qmax = np.max(q)
    return qmax + tau * np.log(np.sum(np.exp((q - qmax) / tau)))

def smooth_val_grad(q, G, tau):
    qmax = np.max(q)
    e = np.exp((q - qmax) / tau)
    se = np.sum(e)
    w = e / se
    phi = qmax + tau * np.log(se)
    grad = G.T @ w
    return phi, grad

def make_starts(base, n, rng, n_starts):
    starts = [base.copy()]
    for sd in (0.02, 0.08, 0.20):
        starts.append(base + sd * rng.standard_normal(n))
    starts.append(rng.uniform(0.0, 1.0, size=n))
    K = min(6, (n - 1) // 2)
    y = np.full(n, 0.5)
    t = (np.arange(n) + 0.5) / n
    for k in range(1, K + 1):
        a = 0.12 / np.sqrt(k) * rng.standard_normal()
        b = 0.12 / np.sqrt(k) * rng.standard_normal()
        y += a * np.cos(2.0 * np.pi * k * t) + b * np.sin(2.0 * np.pi * k * t)
    starts.append(y)
    bv = np.zeros(n)
    bv[:n // 2] = 1.0
    if n % 2 == 1:
        bv[n // 2] = 0.5
    rng.shuffle(bv)
    starts.append(bv.copy())
    starts.append(base + 0.4 * rng.standard_normal(n))
    projected = []
    for s in starts:
        ps = project_capped(s, n)
        dup = False
        for ex in projected:
            if np.max(np.abs(ps - ex)) < 1e-10:
                dup = True
                break
        if not dup:
            projected.append(ps)
        if len(projected) >= n_starts:
            break
    return projected

def run(seed=42, **kwargs):
    n_starts, max_iter = 7, 80
    try:
        ns = int(kwargs.get('n_starts', 7))
        if 1 <= ns <= 8:
            n_starts = ns
    except (ValueError, TypeError):
        pass
    try:
        mi = int(kwargs.get('max_iter', 80))
        if 20 <= mi <= 200:
            max_iter = mi
    except (ValueError, TypeError):
        pass
    try:
        s = int(seed)
        if s < 0:
            s = 42
    except (ValueError, TypeError):
        s = 42
    rng = np.random.default_rng(s)
    n = 80
    base = np.full(n, 0.5)
    try:
        p = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 2 <= len(p) <= 999:
            n = len(p)
            proj = project_capped(p, n)
            if feasible_check(proj, n):
                base = proj
    except (NameError, ValueError, TypeError):
        pass
    if n == 80:
        try:
            npv = int(kwargs.get('n_points', 80))
            if 2 <= npv <= 999:
                n = npv
        except (ValueError, TypeError):
            pass
    base = project_capped(base, n)
    best = base.copy()
    best_score = exact_score(best, n)
    if n > 256:
        return (best, best_score, n)
    temps = [0.02, 0.005, 0.00125, 0.0003125, 0.000078125, 0.00001953125, 0.0000048828125]
    starts = make_starts(base, n, rng, n_starts)
    for start in starts:
        x = start.copy()
        for tau in temps:
            alpha = 1.0
            stage_done = False
            for _it in range(max_iter):
                q = lag_values(x, n)
                if not np.all(np.isfinite(q)):
                    stage_done = True
                    break
                G = lag_jacobian(x, n)
                if not np.all(np.isfinite(G)):
                    stage_done = True
                    break
                phi_x, g = smooth_val_grad(q, G, tau)
                if not np.all(np.isfinite(g)):
                    stage_done = True
                    break
                accepted = False
                for _bt in range(25):
                    xtrial = project_capped(x - alpha * g, n)
                    d = xtrial - x
                    if np.max(np.abs(d)) <= 1e-9:
                        stage_done = True
                        break
                    gd = np.dot(g, d)
                    if not (gd < 0.0):
                        stage_done = True
                        break
                    qtr = lag_values(xtrial, n)
                    phi_tr = smooth_val(qtr, tau)
                    if phi_tr <= phi_x + 1e-4 * gd + 1e-14 * max(1.0, abs(phi_x)):
                        x = xtrial
                        alpha = min(alpha * 1.25, 4.0)
                        accepted = True
                        break
                    alpha *= 0.5
                if stage_done:
                    break
                if not accepted:
                    stage_done = True
            sc = exact_score(x, n)
            if sc < best_score - 1e-12:
                best_score = sc
                best = x.copy()
        if best_score <= 0.3808:
            break
    best = project_capped(best, n)
    if not feasible_check(best, n):
        best = base.copy()
        if not feasible_check(best, n):
            best = np.full(n, 0.5)
    c5 = exact_score(best, n)
    return (best, c5, n)
