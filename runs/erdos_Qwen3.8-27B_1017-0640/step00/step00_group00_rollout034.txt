                    cp = top
                p = int(rng.choice(np.array(cp)))
                others = np.array([j for j in range(POP) if j != i and j != p])
                r1, r2 = int(others[0]), int(others[1])
                if len(others) > 2:
                    sel = rng.choice(others, size=2, replace=False)
                    r1, r2 = int(sel[0]), int(sel[1])
                donor = snap[i] + F * (snap[p] - snap[i]) + F * (snap[r1] - snap[r2])
            else:
                others = np.array([j for j in range(POP) if j != i])
                sel = rng.choice(others, size=3, replace=False)
                r1, r2, r3 = int(sel[0]), int(sel[1]), int(sel[2])
                donor = snap[r1] + F * (snap[r2] - snap[r3])
            mask = rng.random(n) < CR
            mask[int(rng.integers(0, n))] = True
            trial = np.where(mask, donor, snap[i])
            proj = project_capped(trial, n)
            if proj is None or np.max(np.abs(proj - snap[i])) <= 1e-15:
                continue
            s = exact_score(proj, n)
            if s is None or s >= snap_sc[i] - 1e-12:
                continue
            pop[i] = proj
            pop_sc[i] = s
            if s < best_score - 1e-12:
                best = proj.copy()
                best_score = s
        if best_score < gen_start - 1e-10:
            stagnation = 0
        else:
            stagnation += 1
        if stagnation >= 18 and restarts < 4:
            restarts += 1
            stagnation = 0
            nsd = [0.03, 0.08, 0.20, 0.40][restarts - 1]
            n_repl = max(1, POP // 3)
            worst_rows = sorted(range(POP), key=(lambda j: pop_sc[j]), reverse=True)
            replaced = 0
            alt = 0
            for w in worst_rows:
                if replaced >= n_repl:
                    break
                if alt > 0 and alt % 2 == 0 and n_repl >= 2:
                    nc = project_capped(rng.uniform(-0.5, 1.5, n), n)
                else:
                    nc = project_capped(best + nsd * rng.standard_normal(n), n)
                alt += 1
                if nc is not None:
                    s = exact_score(nc, n)
                    if s is not None:
                        pop[w] = nc
                        pop_sc[w] = s
                        replaced += 1
                        if s < best_score - 1e-12:
                            best = nc.copy()
                            best_score = s
        elif stagnation >= 36 and restarts >= 4:
            break
    NPROP = 16 if n <= 64 else (10 if n <= 160 else 5)
    for scale in [0.12, 0.05, 0.02, 0.008, 0.003]:
        accepted = False
        for _ in range(NPROP):
            z = rng.standard_normal(n)
            z = z - z.mean()
            nz = math.sqrt(float(np.dot(z, z)))
            if nz > 0:
                z = z / nz
            t = project_capped(best + scale * z, n)
            if t is None:
                continue
            s = exact_score(t, n)
            if s is not None and s < best_score - 1e-12:
                best = t.copy()
                best_score = s
                accepted = True
        if accepted:
            for _ in range(NPROP):
                z = rng.standard_normal(n)
                z = z - z.mean()
                nz = math.sqrt(float(np.dot(z, z)))
                if nz > 0:
                    z = z / nz
                t = project_capped(best + scale * z, n)
                if t is None:
                    continue
                s = exact_score(t, n)
                if s is not None and s < best_score - 1e-12:
                    best = t.copy()
                    best_score = s
    if not check_feasible(best, n):
        rep = project_capped(best, n)
        if rep is not None:
            best = rep
        else:
            best = parent_fb.copy()
    final_score = exact_score(best, n)
    if final_score is None:
        best = parent_fb.copy()
        final_score = exact_score(best, n)
    if final_score is None:
        best = np.full(n, 0.5)
        final_score = exact_score(best, n)
    return (best, final_score, n)
