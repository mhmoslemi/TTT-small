def project_capped(v, N, m):
    if len(v) != N or not np.all(np.isfinite(v)):
        return None
    v = np.clip(v.astype(np.float64), -1e6, 1e6).copy()

    hi = float(np.max(v)) + 1.0

        else:
            hi = mid
    y = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    r = m - np.sum(y)

                y[i] += add; r -= add
            else:
                sub = min(-r, y[i])
                y[i] -= sub; r += sub

        return None
    return y

    c = np.correlate(h, 1.0 - h, mode='full') * dx
    if len(c) != 2 * len(h) - 1 or not np.all(np.isfinite(c)):
        return None
    return float(np.max(c))

def smooth_stats(q, tau):
    qmax = float(np.max(q))

def smooth_grad(h, p, dx, N):
    g = np.zeros(N, dtype=np.float64)

        k = j - (N - 1)
        lo = max(0, -k)

        g[lo:hi + 1] += coeff * (1.0 - h[lo + k:hi + 1 + k])
        g[lo + k:hi + 1 + k] -= coeff * h[lo:hi + 1]
    return g
