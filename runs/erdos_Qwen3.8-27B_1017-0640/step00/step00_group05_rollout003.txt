def q_and_grad(h, n, dx):
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    # Gradient: (2n-1) x n
    # For row k (lag = k - (n-1)):
    # J[k, r] = dx * (1_{0<=r-lag<n}*(1-h[r-lag]) - 1_{0<=r+lag<n}*h[r+lag])

    for r in range(n):
        lags = np.arange(-(n-1), n)

        i1 = r - lags

def obj_f(z):
    return z[-1]

def obj_grad(z):
    n = len(z) - 1
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g

def eq_f(z, m):
    return np.sum(z[:-1]) - m

def eq_grad(z, m):
    n = len(z) - 1
    g = np.ones(n)
    g = np.concatenate([g, [0.0]])
    return g

def ineq_f(z, n, dx):
    h = z[:n]
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    return z[n] - q

def ineq_jac(z, n, dx):
    h = z[:n]
    q, J = q_and_grad(h, n, dx)
    # J is (2n-1) x n
    # Jacobian of ineq: each row k has -J[k, :] for h, and 1 for t
    jac = np.concatenate([-J, np.ones((2*n-1, 1))], axis=1)
    return jac

            t0 = float(np.clip(np.max(q0), 0.0, 2.0))
            z0 = np.concatenate([h0, [t0]])
            try:
                res = minimize(obj_f, z0, jac=obj_grad, method='SLSQP',
                               bounds=bnds, constraints=cons,
                               options={'maxiter': 250, 'ftol': 1e-10, 'disp': False})
                if res.x is not None and res.x.size == n + 1:
                    h_new = res.x[:n]
                    if np.all(np.isfinite(h_new)):
                        h_new = project_box_sum(h_new, m)
                        s = exact_score(h_new, n)
                        if s < best_score - 1e-12:
                            best_h = h_new.copy()
                            best_score = s
            except Exception:
                continue

    best_h = project_box_sum(best_h, m)
    best_h = np.clip(best_h, 0.0, 1.0)
    if not np.all(np.isfinite(best_h)) or abs(np.sum(best_h) - m) > 1e-8:
        best_h = np.full(n, 0.5)
    best_score = exact_score(best_h, n)
    return (best_h, best_score, n)
