        add(1.0 - parent)
    add(np.full(n, 0.5, dtype=np.float64))
    for f in (1, 2):
        phi = float(rng.uniform(0, 2.0 * math.pi))
        x = np.arange(n, dtype=np.float64)
        add(0.5 + 0.30 * np.sin(2.0 * math.pi * f * (x + 0.5) / n + phi))
    if parent is not None:
        add(parent + 0.12 * rng.standard_normal(n))
    else:
        add(0.5 + 0.20 * rng.standard_normal(n))
        add(rng.uniform(0.0, 1.0, n))
    return starts

def run(seed=42, **kwargs):
    try:
        s = int(seed)
        if s < 0 or s > 0xFFFFFFFF:
            s = 42
    except (ValueError, TypeError, OverflowError):
        s = 42
    rng = np.random.default_rng(s)
    n = 47
    parent = None
    pv = globals().get('initial_h_values', None)
    if pv is not None:
        try:
            pv = np.array(pv, dtype=np.float64).ravel()
            if 2 <= len(pv) < 1000 and np.all(np.isfinite(pv)) and np.all(pv >= -1e-7) and np.all(pv <= 1 + 1e-7):
                n = len(pv)
                p = _proj(pv, n)
                if p is not None and _feasible(p, n):
                    parent = p
        except Exception:
            pass
    if parent is None:
        n = 47
        parent = np.full(n, 0.5, dtype=np.float64)
    best = parent.copy()
    best_sc = _score(best, n)
    starts = _make_starts(parent, n, rng)
    cap = 7 if n <= 64 else (5 if n <= 128 else 3)
    starts = starts[:cap]
    n_pass = 3 if n <= 64 else 2
    n_pairs = 20 if n <= 64 else (12 if n <= 128 else 6)
    for st in starts:
        try:
            if not _feasible(st, n):
                continue
            ss = _score(st, n)
            if math.isfinite(ss) and ss < best_sc - 1e-12:
                best, best_sc = st.copy(), ss
            ls, ls_sc = _local(best, best_sc, n, rng, n_pass, n_pairs)
            if math.isfinite(ls_sc) and ls_sc < best_sc - 1e-12:
                best, best_sc = ls, ls_sc
        except Exception:
            continue
    if n <= 64:
        for scale in (0.06, 0.16):
            try:
                rv = _proj(best + scale * rng.standard_normal(n), n)
                if rv is None or not _feasible(rv, n):
                    continue
                rs = _score(rv, n)
                if math.isfinite(rs) and rs < best_sc - 1e-12:
                    best, best_sc = rv, rs
                ls, ls_sc = _local(best, best_sc, n, rng, 2, n_pairs)
                if math.isfinite(ls_sc) and ls_sc < best_sc - 1e-12:
                    best, best_sc = ls, ls_sc
            except Exception:
                continue
    elif n <= 128:
        try:
            rv = _proj(best + 0.08 * rng.standard_normal(n), n)
            if rv is not None and _feasible(rv, n):
                rs = _score(rv, n)
                if math.isfinite(rs) and rs < best_sc - 1e-12:
                    best, best_sc = rv, rs
                ls, ls_sc = _local(best, best_sc, n, rng, 2, n_pairs)
                if math.isfinite(ls_sc) and ls_sc < best_sc - 1e-12:
                    best, best_sc = ls, ls_sc
        except Exception:
            pass
    try:
        fp = _proj(best, n)
        if fp is not None and _feasible(fp, n):
            best = fp
            best_sc = _score(best, n)
    except Exception:
        pass
    if not math.isfinite(best_sc) or not _feasible(best, n):
        best = parent.copy()
        best_sc = _score(best, n)
    if not math.isfinite(best_sc):
        best = np.full(n, 0.5, dtype=np.float64)
        best_sc = _score(best, n)
    return best.copy(), float(best_sc), int(n)
