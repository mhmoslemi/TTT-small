        if float(np.sum(np.clip(y - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    tol = 1e-12 * max(1, n)
    for _ in range(n + 1):
        if abs(res) < tol:
            break
        avail = (1.0 - x) if res > 0 else (-x)
        idx = int(np.argmax(avail))
        d = min(avail[idx], abs(res))
        if d < 1e-15:
            break
        x[idx] += d if res > 0 else -d
        res -= d if res > 0 else -d
    return x

def build_lags(n):
    data = []
    for l in range(-(n - 1), n):
        lo_i = max(0, l)
        hi_i = min(n, n + l)
        iv = np.arange(lo_i, hi_i)
        data.append((iv, iv - l))
    return data

def _obj(z, n, ld):
    return z[-1]

def _obj_j(z, n, ld):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g

def _mass_f(z, n, ld):
    return float(np.sum(z[:n]) - n / 2.0)

def _mass_j(z, n, ld):
    J = np.zeros(len(z))
    J[:n] = 1.0
    return J

def _epis_f(z, n, ld):
    h, t = z[:n], z[n]
    dx = 2.0 / n
    out = np.empty(len(ld))
    for k, (iv, jv) in enumerate(ld):
        out[k] = t - dx * float(np.sum(h[iv] * (1.0 - h[jv])))
    return out

def _epis_j(z, n, ld):
    h = z[:n]
    dx = 2.0 / n
    J = np.zeros((len(ld), n + 1))
    for k, (iv, jv) in enumerate(ld):
        g = np.zeros(n)
        np.add.at(g, iv, 1.0 - h[jv])
        np.add.at(g, jv, -h[iv])
        J[k, :n] = -dx * g
        J[k, n] = 1.0
    return J

def _block_start(n):
    h = np.zeros(n)
    h[:n // 2] = 1.0
    if n % 2 == 1:
        h[n // 2] = 0.5
    return h

def _solve_one(h0, n, ld, maxiter):
    s0 = exact_score(h0)
    z0 = np.concatenate([h0, [min(1.0, s0 + 1e-8)]])
    bounds = [(0.0, 1.0)] * (n + 1)
    cons = [
        {'type': 'eq', 'fun': _mass_f, 'jac': _mass_j, 'args': (n, ld)},
        {'type': 'ineq', 'fun': _epis_f, 'jac': _epis_j, 'args': (n, ld)},
    ]
    try:
        res = minimize(_obj, z0, method='SLSQP', jac=_obj_j, bounds=bounds,
                       constraints=cons, args=(n, ld), maxiter=maxiter, ftol=1e-10)
        if res.x is not None and len(res.x) == n + 1 and np.all(np.isfinite(res.x)):
            h_new = project_feasible(res.x[:n])
            if check_feasible(h_new):
                return h_new
    except Exception:
        pass
    return h0

def run(seed=42, **kwargs):
    try:
        s = int(seed)
    except (TypeError, ValueError):
        s = 42
    rng = np.random.default_rng(s)
    try:
        n_starts = max(4, min(10, int(kwargs.get('n_starts', 8))))
    except (TypeError, ValueError):
        n_starts = 8
    try:
        maxiter = max(100, min(400, int(kwargs.get('maxiter', 250))))
    except (TypeError, ValueError):
        maxiter = 250

    parent = None
    if 'initial_h_values' in dir() and initial_h_values is not None:
        p = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if p.size >= 2 and np.all(np.isfinite(p)):
            parent = p.copy()

    n0 = len(parent) if parent is not None else 42

    fb_h = project_feasible(parent) if parent is not None else _block_start(n0)
    if not check_feasible(fb_h):
        fb_h = _block_start(n0)
    fb_score = exact_score(fb_h)
    best_h, best_score, best_n = fb_h.copy(), fb_score, n0

    # --- Search at native resolution ---
    if n0 <= 256:
        ld0 = build_lags(n0)
        anchor = fb_h.copy()
        starts = [anchor.copy()]
        starts.append(anchor[::-1].copy())
        starts.append(1.0 - anchor.copy())
        starts.append(_block_start(n0))
        starts.append(np.full(n0, 0.5))
        for _ in range(n_starts - len(starts)):
            sc = [0.08, 0.20, 0.40][_ % 3]
            h_r = project_feasible(anchor + rng.standard_normal(n0) * sc)
            starts.append(h_r)
        seen = []
        for h_s in starts:
            h_s = project_feasible(h_s)
            if not check_feasible(h_s):
                continue
            dup = False
            for h_prev in seen:
                if float(np.max(np.abs(h_s - h_prev))) < 1e-10:
                    dup = True
                    break
            if dup:
                continue
            seen.append(h_s)
            sc = exact_score(h_s)
            if sc < best_score - 1e-12:
                best_score, best_h, best_n = sc, h_s.copy(), n0
            res_h = _solve_one(h_s, n0, ld0, maxiter)
            if check_feasible(res_h):
                sc2 = exact_score(res_h)
                if sc2 < best_score - 1e-12:
                    best_score, best_h, best_n = sc2, res_h.copy(), n0

    # --- Search at doubled resolution ---
    if n0 <= 128:
        n1 = 2 * n0
        ld1 = build_lags(n1)
        h_trans = np.repeat(best_h, 2).copy()
        if parent is not None and len(parent) == n0:
            h_trans_p = np.repeat(fb_h, 2).copy()
            if not check_feasible(h_trans_p):
                h_trans_p = project_feasible(h_trans_p)
        else:
            h_trans_p = h_trans.copy()
        starts1 = [h_trans.copy(), h_trans_p.copy()]
        starts1.append(np.repeat(_block_start(n0), 2))
        for _ in range(n_starts - len(starts1)):
            sc = [0.08, 0.20, 0.40][_ % 3]
            h_r = project_feasible(h_trans + rng.standard_normal(n1) * sc)
            starts1.append(h_r)
        seen1 = []
        for h_s in starts1:
            h_s = project_feasible(h_s)
            if not check_feasible(h_s):
                continue
            dup = False
            for h_prev in seen1:
                if float(np.max(np.abs(h_s - h_prev))) < 1e-10:
                    dup = True
                    break
            if dup:
                continue
            seen1.append(h_s)
            sc = exact_score(h_s)
            if sc < best_score - 1e-12 or (abs(sc - best_score) <= 1e-12 and n1 < best_n):
                best_score, best_h, best_n = sc, h_s.copy(), n1
            res_h = _solve_one(h_s, n1, ld1, maxiter)
            if check_feasible(res_h):
                sc2 = exact_score(res_h)
                if sc2 < best_score - 1e-12 or (abs(sc2 - best_score) <= 1e-12 and n1 < best_n):
                    best_score, best_h, best_n = sc2, res_h.copy(), n1

    # --- Final validation ---
    if not check_feasible(best_h):
        best_h = fb_h
        best_score = exact_score(fb_h)
        best_n = len(fb_h)
    else:
        best_h = project_feasible(best_h)
        best_score = exact_score(best_h)
        best_n = len(best_h)

    return best_h, best_score, int(best_n)
