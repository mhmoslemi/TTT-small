def capped_simplex_project(v, n):
    if n < 1:
        return None
    v = np.asarray(v, dtype=np.float64)
    if v.shape != (n,) or not np.all(np.isfinite(v)):
        return None
    s = n / 2.0
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        tot = math.fsum(x)
        if tot > s:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(v - lam, 0.0, 1.0)
    # Fix residual
    res = s - math.fsum(x)
    for idx in range(n):
        if abs(res) < 5e-15:
            break
        if res > 0:
            cap = 1.0 - x[idx]
            adj = min(cap, res)
            if adj > 0:
                x[idx] += adj
                res -= adj
        else:
            adj = min(x[idx], -res)
            if adj > 0:
                x[idx] -= adj
                res += adj

        return None
    if np.min(x) < -1e-12 or np.max(x) > 1 + 1e-12:
        return None
    return x

def exact_score(h):
    n = len(h)
    if n < 1 or not np.all(np.isfinite(h)):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full')
    val = float(np.max(c)) * (2.0 / n)
    return val if np.isfinite(val) else float('inf')

def compute_lags_and_jac(h):
    n = len(h)
    dx = 2.0 / n
    num_lags = 2 * n - 1
    q = np.zeros(num_lags)
    J = np.zeros((num_lags, n))
    for k in range(num_lags):
        d = k - (n - 1)  # d ranges from -(n-1) to n-1
        # Source indices: i where 0<=i<n and 0<=i+d<n
        i_lo = max(0, -d)
        i_hi = min(n, n - d)  # exclusive
        for i in range(i_lo, i_hi):
            j = i + d
            q[k] += h[i] * (1.0 - h[j])
            J[k, i] += 1.0 - h[j]
            J[k, j] -= h[i]
    return q, J

    h = z[:n]
    t = z[-1]
