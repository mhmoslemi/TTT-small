    c = np.correlate(h, 1.0 - h, mode="full")
    v = float(c.max()) * (2.0 / n)
    return v if math.isfinite(v) else float('inf')

def _feasible(h):
    n = len(h)
    return (n >= 2 and np.isfinite(h).all() and h.min() >= -1e-12
            and h.max() <= 1.0 + 1e-12 and abs(math.fsum(h) - n/2.0) <= 1e-9)

def _lags(h):
    n = _opt_n
    qlen = 2 * n - 1
    q = np.zeros(qlen)
    for d in range(-(n-1), n):
        li = n - 1 + d
        if d >= 0:
            s, e = 0, n - d
        else:
            s, e = -d, n
        for i in range(s, e):
            q[li] += h[i] * (1.0 - h[i + d])
    return q

def _lags_jac(h):
    n = _opt_n
    qlen = 2 * n - 1
    q = np.zeros(qlen)
    J = np.zeros((qlen, n))
    for d in range(-(n-1), n):
        li = n - 1 + d
        if d >= 0:
            s, e = 0, n - d
        else:
            s, e = -d, n
        for i in range(s, e):
            j = i + d
            q[li] += h[i] * (1.0 - h[j])
            J[li, i] += 1.0 - h[j]
            J[li, j] -= h[i]
    return q, J

def _obj(z):
    return z[-1]

def _obj_j(z):
    j = np.zeros(_opt_n + 1)
    j[-1] = 1.0
    return j

def _eq(z):
    return math.fsum(z[:_opt_n]) - _opt_n / 2.0

def _eq_j(z):
    j = np.zeros(_opt_n + 1)
    j[:_opt_n] = 1.0
    return j

def _ineq(z):
    n = _opt_n
    return z[-1] - (2.0/n) * _lags(z[:n])

def _ineq_j(z):
    n = _opt_n
    q, J = _lags_jac(z[:n])
    dx = 2.0 / n
    out = np.zeros((2*n-1, n+1))
    out[:, :n] = -dx * J
    out[:, n] = 1.0
    return out

def _solve(h0, n, maxiter):

    _opt_n = n
    hp = _cap_proj(h0, n/2.0)
    if hp is None:

    t0 = min(1.0, _score(hp) + 1e-9)
    z0 = np.append(hp, t0)
    bnds = [(0.0, 1.0)] * (n + 1)
    cons = [{'type':'eq','fun':_eq,'jac':_eq_j},
            {'type':'ineq','fun':_ineq,'jac':_ineq_j}]
    try:
        r = minimize(_obj, z0, method='SLSQP', jac=_obj_j,
                     bounds=bnds, constraints=cons,
                     options={'ftol':1e-10,'maxiter':maxiter,'disp':False})
        if r is not None and r.x is not None and len(r.x)==n+1:
            hx = r.x[:n]
            if np.isfinite(hx).all():
                fx = _cap_proj(hx, n/2.0)
                if fx is not None:
                    return fx
    except Exception:
        pass
    return hp
