                if r <= 0:
                    break
                add = min(r, 1.0 - x[i])
                x[i] += add
                r -= add
        else:
            for i in range(n):
                if r >= 0:
                    break
                sub = min(-r, x[i])
                x[i] -= sub
                r += sub
    if not np.all(np.isfinite(x)) or np.any(x < -1e-12) or np.any(x > 1.0 + 1e-12):
        return None
    if abs(math.fsum(x) - s) > 1e-9:
        return None
    return np.clip(x, 0.0, 1.0)

def exact_score(h, n):
    try:
        val = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)
        return val if math.isfinite(val) else None
    except Exception:
        return None

def is_feasible(h, n):
    if h.ndim != 1 or h.shape[0] != n or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(math.fsum(h) - n / 2.0) <= 1e-9

def lag_values(h, n):
    dx = 2.0 / n
    out = np.empty(2 * n - 1, dtype=np.float64)
    for j in range(2 * n - 1):
        d = -(n - 1) + j
        st = max(0, -d)
        sp = min(n, n - d)
        out[j] = np.dot(h[st:sp], 1.0 - h[st + d:sp + d]) * dx
    return out

def smooth_obj_grad(h, tau, n, need_grad=True):
    a = lag_values(h, n)
    if not np.all(np.isfinite(a)):
        return None, None
    A = float(np.max(a))
    e = np.exp((a - A) / tau)
    Z = math.fsum(e)
    if Z <= 0 or not math.isfinite(Z):
        return None, None
    F = A + tau * math.log(Z)
    if not math.isfinite(F):
        return None, None
    if not need_grad:
        return F, None
    p = e / Z
    dx = 2.0 / n
    g = np.zeros(n, dtype=np.float64)
    for j in range(2 * n - 1):
        d = -(n - 1) + j
        wj = dx * p[j]
        if abs(wj) < 1e-16:
            continue
        st = max(0, -d)
        sp = min(n, n - d)
        g[st:sp] += wj * (1.0 - h[st + d:sp + d])
        g[st + d:sp + d] -= wj * h[st:sp]
    if not np.all(np.isfinite(g)):
        return None, None
    return F, g

def smooth_search(h0, temps, iters_pt, n):
    best_h = np.copy(h0)
    best_sc = exact_score(best_h, n)
    if best_sc is None:
        best_sc = 1e9
    for tau in temps:
        cur = np.copy(best_h)
        Fc, g = smooth_obj_grad(cur, tau, n, True)
        if Fc is None:
            break
        eta = 1.0
        for _ in range(iters_pt):
            acc = False
            y = None
            Fy = None
            for bt in range(25):
                y = project_capped_simplex(cur - eta * g, n)
                if y is None or not is_feasible(y, n):
                    eta *= 0.5
                    continue
                sc = exact_score(y, n)
                if sc is not None and sc < best_sc - 1e-12:
                    best_h = np.copy(y)
                    best_sc = sc
                delta = y - cur
                d2 = float(np.dot(delta, delta))
                if d2 <= 1e-24:
                    acc = True
                    break
                gd = float(np.dot(g, delta))
                Fy, _ = smooth_obj_grad(y, tau, n, False)
                if Fy is None:
                    eta *= 0.5
                    continue
                tol = 1e-4 * min(gd, 0.0) + 1e-14 * max(1.0, abs(Fc))
                if Fy <= Fc + tol:
                    acc = True
                    break
                eta *= 0.5
            if not acc:
                break
            cur = y
            Fc = Fy
            eta = min(1.5 * eta, 10.0)
            if float(np.max(np.abs(delta))) <= 1e-9:
                break
    return np.copy(best_h), best_sc

def run(seed=42, **kwargs):
    try:
        seed = int(seed)
        if seed < 0 or seed >= 4294967296:
            seed = 42
    except (TypeError, ValueError, OverflowError):
        seed = 42
    rng = np.random.default_rng(seed)

    parent = None
    raw = globals().get('initial_h_values', None)
    if raw is not None:
        try:
            p = np.array(raw, dtype=np.float64).ravel()
            if p.ndim == 1 and 2 <= len(p) < 1000 and np.all(np.isfinite(p)):
                if p.min() >= -1e-7 and p.max() <= 1.0 + 1e-7:
                    parent = p
        except Exception:
            pass

    n = len(parent) if parent is not None else 47

    if n <= 64:
        temps = [0.04, 0.01, 0.0025, 0.0005, 0.0001]
        iters_pt = 80
        ns = 8
        rest = [0.04, 0.12]
    elif n <= 128:
        temps = [0.04, 0.01, 0.0025, 0.0005]
        iters_pt = 45
        ns = 5
        rest = [0.06]
    elif n <= 512:
        temps = [0.03, 0.006, 0.001]
        iters_pt = 25
        ns = 3
        rest = []
    else:
        temps = [0.02, 0.003]
        iters_pt = 12
        ns = 2
        rest = []

    starts = []
    if parent is not None:
        pr = project_capped_simplex(parent, n)
        if pr is not None:
            starts.append(pr.copy())
    const = np.full(n, 0.5, dtype=np.float64)
    starts.append(const.copy())
    base = starts[0]

    idx = np.arange(n, dtype=np.float64)
    for f in [1, 2]:
        phi = rng.uniform(0, 2 * math.pi)
        v = 0.5 + 0.30 * np.sin(2.0 * math.pi * f * (idx + 0.5) / n + phi)
        pr = project_capped_simplex(v, n)
        if pr is not None:
            starts.append(pr)

    for sc_ in [0.06, 0.18]:
        pr = project_capped_simplex(base + sc_ * rng.standard_normal(n), n)
        if pr is not None:
            starts.append(pr)

    for _ in range(2):
        pr = project_capped_simplex(rng.uniform(0, 1, n), n)
        if pr is not None:
            starts.append(pr)

    if parent is None:
        phi = rng.uniform(0, 2 * math.pi)
        v = 0.5 + 0.30 * np.cos(2.0 * math.pi * 3 * (idx + 0.5) / n + phi)
        pr = project_capped_simplex(v, n)
        if pr is not None:
            starts.append(pr)

    deduped = [starts[0]]
    for s in starts[1:]:
        if all(float(np.max(np.abs(s - d))) > 1e-9 for d in deduped):
            deduped.append(s)
    starts = deduped[:ns]

    best_h, best_sc = None, 1e9
    for s in starts:
        sc = exact_score(s, n)
        if sc is not None and sc < best_sc - 1e-12:
            best_sc = sc
            best_h = np.copy(s)
    if best_h is None:
        best_h, best_sc = const.copy(), exact_score(const, n) or 1.0

    for s in starts:
        try:
            rh, rs = smooth_search(s, temps, iters_pt, n)
            if rs < best_sc - 1e-12:
                best_sc, best_h = rs, np.copy(rh)
        except Exception:
            pass

    for sig in rest:
        try:
            v = best_h + sig * rng.standard_normal(n)
            pr = project_capped_simplex(v, n)
            if pr is not None:
                rh, rs = smooth_search(pr, temps, iters_pt, n)
                if rs < best_sc - 1e-12:
                    best_sc, best_h = rs, np.copy(rh)
        except Exception:
            pass

    best_h = np.asarray(best_h, dtype=np.float64)
    if not is_feasible(best_h, n):
        pr = project_capped_simplex(best_h, n)
        best_h = pr if pr is not None else const.copy()
    fs = exact_score(best_h, n)
    if fs is None:
        best_h, fs = const.copy(), 1.0
    return best_h, float(fs), n
