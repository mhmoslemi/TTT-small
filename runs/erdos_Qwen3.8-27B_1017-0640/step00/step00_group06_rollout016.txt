    fb = half_block(n)
    protected = fb.copy()
    if parent is not None and feasible_check(parent, n, m):
        base = parent.copy()
        protected = parent.copy()
    elif parent is not None:
        proj = bounded_simplex_project(parent, np.zeros(n), np.ones(n), m)
        base = proj if proj is not None else fb.copy()
    else:
        base = fb.copy()
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    starts = [base.copy()]
    for sigma in (0.05, 0.15, 0.30):
        g = base + sigma * rng.standard_normal(n)
        p = bounded_simplex_project(g, np.zeros(n), np.ones(n), m)
        starts.append(p if p is not None else base.copy())
    g = rng.beta(0.7, 0.7, size=n)
    p = bounded_simplex_project(g, np.zeros(n), np.ones(n), m)
    starts.append(p if p is not None else base.copy())
    g = 0.5 + 0.45 * np.sin(2.0 * np.pi * np.arange(n) / n + rng.uniform(0, 2.0 * np.pi))
    p = bounded_simplex_project(g, np.zeros(n), np.ones(n), m)
    starts.append(p if p is not None else base.copy())
    if np.max(np.abs(fb - base)) > 1e-12:
        starts.append(fb.copy())
    dedup = [starts[0]]
    for s in starts[1:]:
        dup = any(np.max(np.abs(s - d)) < 1e-12 for d in dedup)
        if not dup:
            dedup.append(s)
    starts = dedup
    B0 = max(8, min(28, math.ceil(28.0 * 95.0 / n)))
    B1 = max(4, min(12, math.ceil(12.0 * 95.0 / n)))
    best_h = protected.copy()
    best_score = exact_score(best_h, n)
    for idx, s in enumerate(starts):
        if s is None or not np.all(np.isfinite(s)):
            continue
        budget = B0 if idx == 0 else B1
        h_opt, sc = optimize_start(s, n, m, budget)
        if sc < best_score - 1e-12:
            best_score = sc
            best_h = h_opt
    if not feasible_check(best_h, n, m):
        proj = bounded_simplex_project(best_h, np.zeros(n), np.ones(n), m)
        if proj is not None and feasible_check(proj, n, m):
            best_h = proj
        else:
            best_h = protected.copy()
    c5 = exact_score(best_h, n)
    if not np.isfinite(c5):
        best_h = protected.copy()
        c5 = exact_score(best_h, n)
    return best_h, c5, n
