def project_capped(y, target):

    if not np.all(np.isfinite(y)):
        return None

    hi = float(np.max(y))
    for _ in range(64):

        s = np.sum(np.clip(y - mid, 0.0, 1.0))

            lo = mid
        else:
            hi = mid
