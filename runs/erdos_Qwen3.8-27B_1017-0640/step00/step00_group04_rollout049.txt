                d = min(-r, x[i])
                x[i] -= d
                r += d
            if abs(r) <= 5e-13 * max(1, n):
                break
    return x

def check_feasible(h, n):
    if h.ndim != 1 or len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if float(np.min(h)) < -1e-10 or float(np.max(h)) > 1.0 + 1e-10:
        return False
    if abs(float(np.sum(h)) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def score(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    if len(corr) != 2 * n - 1:
        return float('inf')
    if not np.all(np.isfinite(corr)):
        return float('inf')
    return float(np.max(corr) * (2.0 / n))

def encode_lattice(h, D):
    n = len(h)
    M = D * n // 2
    raw = D * np.asarray(h, dtype=np.float64)
    base = np.floor(raw).astype(np.int64)
    frac = raw - base
    base = np.clip(base, 0, D)
    R = M - int(np.sum(base))
    if R > 0:
        order = np.argsort(-frac, kind='stable')
        cnt = 0
        for i in order:
            if cnt >= R:
                break
            if base[i] < D:
                base[i] += 1
                cnt += 1
        if cnt < R:
            return None
    elif R < 0:
        order = np.argsort(frac, kind='stable')
        cnt = 0
        for i in order:
            if cnt >= -R:
                break
            if base[i] > 0:
                base[i] -= 1
                cnt += 1
        if cnt < -R:
            return None
    if int(np.sum(base)) != M or np.min(base) < 0 or np.max(base) > D:
        return None
    return base

def decode_lattice(k, D):
    return k.astype(np.float64) / D

def anneal_chain(k_start, D, n, rng, n_blocks, ppb, T_hi, T_lo, bs, bsc, inc, incs):
    k = k_start.copy()
    cur_s = score(decode_lattice(k, D))
    if not math.isfinite(cur_s):
        return
    if cur_s < bsc[0] - 1e-12:
        bsc[0] = cur_s
        bs[0][:] = k
    if cur_s < incs[0] - 1e-12:
        incs[0] = cur_s
        inc[0][:] = decode_lattice(k, D)
    mx0 = 1
    mx1 = max(2, D // 32)
    mx2 = max(2, D // 8)
    for j in range(n_blocks):
        T = T_hi * (T_lo / T_hi) ** (j / float(n_blocks - 1))
        for _ in range(ppb):
            a = int(rng.integers(n))
            b = int(rng.integers(n))
            if a == b:
                continue
            r = rng.random()
            if r < 0.70:
                mx = mx0
            elif r < 0.90:
                mx = mx1
            else:
                mx = mx2
            m = int(rng.integers(1, mx + 1))
            if k[a] + m > D or k[b] < m:
                continue
            k[a] += m
            k[b] -= m
            h_t = decode_lattice(k, D)
            s_t = score(h_t)
            if not math.isfinite(s_t):
                k[a] -= m
                k[b] += m
                continue
            if s_t < bsc[0] - 1e-12:
                bsc[0] = s_t
                bs[0][:] = k
            if s_t < incs[0] - 1e-12:
                incs[0] = s_t
                inc[0][:] = h_t
            delta = s_t - cur_s
            if delta <= 0:
                cur_s = s_t
            else:
                if T > 1e-9 and rng.random() < math.exp(-delta / T):
                    cur_s = s_t
                else:
                    k[a] -= m
                    k[b] += m

def run(seed=42, **kwargs):
    try:
        seed = int(seed)
    except (ValueError, TypeError):
        seed = 42
    seed = seed % (2 ** 32)
    rng = np.random.default_rng(seed)
    n = 42
    parent = None
    if 'initial_h_values' in globals():
        try:
            p = np.asarray(initial_h_values, dtype=np.float64)
            if p.ndim == 1 and 2 <= len(p) <= 1024 and np.all(np.isfinite(p)) and np.all(np.abs(p) <= 1e6):
                n = len(p)
                if p.min() >= -1e-10 and p.max() <= 1.0 + 1e-10:
                    if abs(float(np.sum(p)) - n / 2.0) > 5e-13 * max(1, n):
                        p = project_capped(p, n)
                    if check_feasible(p, n):
                        parent = p.copy()
                else:
                    p = project_capped(p, n)
                    if check_feasible(p, n):
                        parent = p.copy()
        except Exception:
            parent = None
    if parent is None:
        parent = np.zeros(n, dtype=np.float64)
        parent[:n // 2] = 1.0
        if n % 2 == 1:
            parent[n // 2] = 0.5
    fallback = parent.copy()
    fb_score = score(fallback)
    if n > 64:
        h_out = fallback.copy()
        if check_feasible(h_out, n):
            return h_out, fb_score, n
        fb = np.zeros(n, dtype=np.float64)
        fb[:n // 2] = 1.0
        if n % 2 == 1:
            fb[n // 2] = 0.5
        return fb, score(fb), n
    inc = [parent.copy()]
    incs = [fb_score]
    g = min(2.0, max(0.5, 42.0 / n))
    T_hi = 0.004 * g
    T_lo = 2e-5 * g
    n_blocks = 24
    ppb = max(50, int(round(160.0 * min(1.0, (42.0 / n) ** 2))))
    D_list = [64, 256, 1024]
    best_prev = None
    for di, D in enumerate(D_list):
        starts = []
        if best_prev is not None and di > 0:
            e = encode_lattice(best_prev, D)
            if e is not None:
                starts.append(e.copy())
        e = encode_lattice(parent, D)
        if e is not None:
            starts.append(e.copy())
        e = encode_lattice(inc[0], D)
        if e is not None:
            starts.append(e.copy())
        if best_prev is not None:
            starts.append(encode_lattice(parent[::-1].copy(), D))
        else:
            starts.append(encode_lattice(parent[::-1].copy(), D))
        comp = np.clip(1.0 - parent, 0.0, 1.0)
        e = encode_lattice(comp, D)
        if e is not None:
            starts.append(e.copy())
        rv = rng.uniform(-0.25, 1.25, size=n)
        rv = project_capped(rv, n)
        e = encode_lattice(rv, D)
        if e is not None:
            starts.append(e.copy())
        while len(starts) < 5:
            e = encode_lattice(parent, D)
            if e is not None:
                starts.append(e.copy())
            else:
                break
        bs = [np.zeros(n, dtype=np.int64)]
        bsc = [float('inf')]
        for s in starts:
            h_s = decode_lattice(s, D)
            s_s = score(h_s)
            if math.isfinite(s_s):
                if s_s < bsc[0] - 1e-12:
                    bsc[0] = s_s
                    bs[0][:] = s
                if s_s < incs[0] - 1e-12:
                    incs[0] = s_s
                    inc[0][:] = h_s
            try:
                anneal_chain(s, D, n, rng, n_blocks, ppb, T_hi, T_lo, bs, bsc, inc, incs)
            except Exception:
                pass
        if math.isfinite(bsc[0]):
            best_prev = decode_lattice(bs[0], D).copy()
        else:
            best_prev = None
    h_final = inc[0].copy()
    if not check_feasible(h_final, n):
        h_final = project_capped(h_final, n)
    if not check_feasible(h_final, n):
        h_final = fallback.copy()
    s_final = score(h_final)
    if not math.isfinite(s_final) or s_final > fb_score + 1e-12:
        h_final = fallback.copy()
        s_final = fb_score
    return h_final, s_final, n
