import numpy as np
import math

def check_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(h.sum() - n / 2.0) <= 1e-10)

def project_capped(v, n):
    if v.shape != (n,) or not np.all(np.isfinite(v)):
        return None
    v = np.clip(v, -1e6, 1e6)
    tgt = n / 2.0
    lo, hi = v.min() - 1.0, v.max()
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > tgt:
            lo = mid
        else:
            hi = mid
    q = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = tgt - q.sum()
    if res > 0:
        for i in range(n):
            a = min(res, 1.0 - q[i])
            q[i] += a
            res -= a
            if abs(res) < 1e-13:
                break

        for i in range(n):
            a = min(-res, q[i])
            q[i] -= a
            res += a

    return q

    return s if np.isfinite(s) else float('inf')

def pair_transfer(h, radius, n, rng):

        p, q = int(idx[0]), int(idx[1])
        if p == q:

    if not np.isfinite(s):
        return h, s, inc_best, inc_score
    if s < inc_score - 1e-12:

        inc_score = s

            prop = pair_transfer(h, r, n, rng)

        except Exception:
            pass
