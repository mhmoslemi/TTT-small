def project_capped_simplex(v, m):

    v = np.clip(v, -1e6, 1e6)
    lo = v.min() - 1.0
    hi = v.max()

        mid = (lo + hi) * 0.5
        if np.clip(v - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    r = m - y.sum()

            for i in range(y.size):
                if r <= 0:
                    break
                a = min(r, 1.0 - y[i])
                y[i] += a
                r -= a
        elif r < -1e-15:
            for i in range(y.size):
                if r >= 0:
                    break
                a = min(-r, y[i])
                y[i] -= a
                r += a
    if not np.all(np.isfinite(y)) or abs(y.sum() - m) > 1e-10:
        return None
    return np.clip(y, 0.0, 1.0)

    q = np.correlate(h, 1.0 - h, mode='full') * dx

        return None

        return None, None

    b_ub = np.zeros(nrows)

    A_ub[:L, -1] = -1.0
    b_ub[:L] = -q
    A_ub[L:L+N, 0:N] = np.eye(N)
    A_ub[L:L+N, N:2*N] = -np.eye(N)
    A_ub[L+N:2*L+N, 0:N] = -np.eye(N)
    A_ub[L+N:2*L+N, N:2*N] = -np.eye(N)
    A_ub[L+2*N, N:2*N] = 1.0
