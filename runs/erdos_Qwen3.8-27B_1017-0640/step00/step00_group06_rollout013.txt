def lp_step(h, delta, q, G, m, n):

    u = np.minimum(1.0, h + delta)

    for si, st in enumerate(starts):
        budget = B0 if si == 0 else B1
        s = st.copy()
        C_s = exact_score(s, n)
        if C_s is not None and (best_score is None or C_s < best_score - 1e-12):
            best = s.copy()
            best_score = C_s
        try:
            h_opt, C_opt = optimize_start(s, n, m, budget, 0.12, 0.25, 1e-5)
        except Exception:
            continue
        if C_opt is not None and C_opt < best_score - 1e-12:
            best = h_opt.copy()
            best_score = C_opt
    if not is_feasible(best, n):
        best = project_bounded_simplex(best, l0, u1, m)
        if best is None or not is_feasible(best, n):
            best = fallback.copy()
    c5 = exact_score(best, n)
    if c5 is None:
        best = fallback.copy()
        c5 = exact_score(best, n)
    return best, c5, n
