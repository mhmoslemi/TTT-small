        scales = [0.03]
    for sc_scale in scales:
        for _ in range(2):
            try:
                p = incumbent + rng.normal(0, sc_scale, n)
                q = project_capped(p, S, n)
                if q is None or not check_feasible(q, n, S):
                    continue
                sc = exact_score(q, n)
                if sc < incumbent_score - 1e-12:
                    incumbent = q.copy()
                    incumbent_score = sc
                bx, bsc = local_search(q, n, S, iters)
                if bsc < incumbent_score - 1e-12:
                    incumbent = bx.copy()
                    incumbent_score = bsc
            except Exception:
                continue

    if not check_feasible(incumbent, n, S):
        q = project_capped(incumbent, S, n)
        if q is not None and check_feasible(q, n, S):
            sc = exact_score(q, n)
            incumbent = q if sc < fallback_score else fallback
            incumbent_score = sc if sc < fallback_score else fallback_score
        else:
            incumbent = fallback
            incumbent_score = fallback_score
    else:
        incumbent_score = exact_score(incumbent, n)

    return incumbent.astype(np.float64), incumbent_score, int(n)
