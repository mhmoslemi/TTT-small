def project_k(y, m):

    lo, hi = y.min() - 1.0, y.max()
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > m: lo = mid
        else: hi = mid

        if abs(res) < 1e-13: break
        if res > 0 and x[i] < 1.0:
            a = min(res, 1.0 - x[i]); x[i] += a; res -= a
        elif res < 0 and x[i] > 0.0:
            a = min(-res, x[i]); x[i] -= a; res += a
    if not np.all(np.isfinite(x)) or abs(x.sum() - m) > 1e-8:

def comp_score(h, n):
    dx = 2.0 / n
    q = np.correlate(h, 1.0 - h, mode='full')
    if len(q) != 2*n-1 or not np.all(np.isfinite(q)):
        return None, None
    q = q * dx
    return q, float(q.max())

def cgrad(h, n, r):
    dx = 2.0 / n
    g = np.zeros(2*n - 1)
    for k in range(-(n-1), n):
        if 0 <= r+k < n: g[n-1+k] -= dx * h[r+k]
        if 0 <= r-k < n: g[n-1+k] += dx * (1.0 - h[r-k])
    return g

def line_min(h, a, b, q, n):
    dx = 2.0 / n
    L = max(-h[a], h[b] - 1.0)
    U = min(1.0 - h[a], h[b])
    if U - L < 1e-14: return 0.0, q.max()
    p = b - a
    ga, gb = cgrad(h, n, a), cgrad(h, n, b)
    bv = ga - gb
    Av = np.zeros(2*n - 1)
    Av[n-1] = -2.0*dx
    Av[n-1+p] = dx
    Av[n-1-p] = dx

            ded.append((sl, ic, idx))

            if abs(sl - ps) < 1e-14:

                break
            t = (pi - ic) / (sl - ps)
            if t <= pb + 1e-10:
                hull.pop()
            else:
                hull.append([sl, ic, t])

            hull.append([sl, ic, float('-inf')])

def make_starts(n, m, rng, parent=None):
    starts = []
    def add(v):
        x = project_k(v, m)
        if x is not None:
            d = np.all([np.linalg.norm(x - s, np.inf) > 1e-10 for s in starts])
            if d: starts.append(x)
    if parent is not None:
        add(parent.copy())
    blk = np.zeros(n)
    blk[:int(m)] = 1.0
    if m - int(m) > 1e-12: blk[int(m)] = m - int(m)
    add(blk); add(blk[::-1])
    if parent is not None:
        add(parent[::-1].copy()); add(1.0 - parent)
        add(np.roll(parent, n//3))
    add(np.full(n, 0.5))
    for _ in range(2):
        add(rng.uniform(0, 1, n))
    ph = rng.uniform(0, 2*math.pi)
    add(0.5 + 0.35*np.cos(2*np.pi*np.arange(n)/n + ph))
    if parent is not None:
        add(parent + rng.normal(0, 0.1, n))
    return starts

def refine(h, n, m, rng, n_sweeps, max_pairs):
    q, sc = comp_score(h, n)
    if q is None: return h, float('inf')
    best_h, best_sc = h.copy(), sc
    all_pairs = [(a, b) for a in range(n) for b in range(a+1, n)]
    n_total = len(all_pairs)
    for sw in range(n_sweeps):
        improved = False
        if max_pairs < n_total:
            # stratified selection
            pairs = []
            seps = [s for s in [1,2,4,8,16,32,64] if s < n]
            per = max(1, max_pairs // len(seps))
            for s in seps:
                valid_a = range(n - s)
                step = max(1, len(valid_a) // per)
                for a in range(0, n-s, step):
                    pairs.append((a, a+s))
                    if len(pairs) >= max_pairs: break
                if len(pairs) >= max_pairs: break
            while len(pairs) < max_pairs:
                a = rng.integers(0, n-1)
                b = rng.integers(a+1, n)
                if (a,b) not in pairs: pairs.append((a,b))
            rng.shuffle(pairs)
        else:
            pairs = all_pairs[:]
            rng.shuffle(pairs)
        sw_start = sc
        for a, b in pairs:
            t, pred = line_min(h, a, b, q, n)
            if abs(t) < 1e-13: continue
            if pred >= sc - 1e-12: continue
            trial = h.copy()
            trial[a] += t; trial[b] -= t
            if not np.all(np.isfinite(trial)): continue
            if trial.min() < -1e-10 or trial.max() > 1.0+1e-10:
                fx = project_k(trial, m)
                if fx is None: continue
                trial = fx
            tq, tsc = comp_score(trial, n)
            if tq is None or tsc >= sc - 1e-12: continue
            h, q, sc = trial, tq, tsc
            improved = True
            if sc < best_sc - 1e-12:
                best_h, best_sc = h.copy(), sc
        if not improved or sc > sw_start - 1e-10:
            break
    return best_h, best_sc
