        return np.inf
    N = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / N))

def corr_vals_grad(h):
    N = len(h)
    dx = 2.0 / N
    nc = 2 * N - 1
    mid = N - 1
    q = np.zeros(nc)
    G = np.zeros((nc, N))
    q[mid] = dx * np.sum(h * (1.0 - h))
    G[mid, :] = dx * (1.0 - 2.0 * h)
    for s in range(1, N):
        q[mid + s] = dx * np.sum(h[s:] * (1.0 - h[:N - s]))
        g = np.zeros(N)
        g[s:] += (1.0 - h[:N - s])
        g[:N - s] += -h[s:]
        G[mid + s, :] = dx * g
        q[mid - s] = dx * np.sum(h[:N - s] * (1.0 - h[s:]))
        g = np.zeros(N)
        g[:N - s] += (1.0 - h[s:])
        g[s:] += -h[:N - s]
        G[mid - s, :] = dx * g
    return q, G

def obj_fn(z):
    return z[-1]

def obj_grad(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g

def eq_fn(z):
    N = len(z) - 1
    return np.sum(z[:N]) - N / 2.0

def eq_jac(z):
    N = len(z) - 1
    j = np.zeros(len(z))
    j[:N] = 1.0
    return j

def ineq_fn(z):
    N = len(z) - 1
    q, _ = corr_vals_grad(z[:N])
    return z[-1] - q

def ineq_jac(z):
    N = len(z) - 1
    _, G = corr_vals_grad(z[:N])
    nc = 2 * N - 1
    J = np.zeros((nc, N + 1))
    J[:, :N] = -G
    J[:, N] = 1.0
    return J

def build_fourth_start(N, rng, K):
    t = (np.arange(N) + 0.5) / N
    y = np.full(N, 0.5)
    for k in range(1, K + 1):
        a = rng.normal(0, 0.12 / np.sqrt(k))
        b = rng.normal(0, 0.12 / np.sqrt(k))
        y += a * np.cos(2.0 * np.pi * k * t) + b * np.sin(2.0 * np.pi * k * t)
    return y

def build_binary_start(N, rng):
    idx = np.arange(N)
    rng.shuffle(idx)
    y = np.zeros(N)
    n_ones = N // 2
    y[idx[:n_ones]] = 1.0
    if N % 2 == 1:
        y[idx[n_ones]] = 0.5
    return y

def run(seed=42, **kwargs):
    n_starts = kwargs.get('n_starts', 6)
    maxiter = kwargs.get('maxiter', 250)
    ftol = kwargs.get('ftol', 1e-10)
    if not isinstance(n_starts, (int, np.integer)) or not (1 <= n_starts <= 8):
        n_starts = 6
    if not isinstance(maxiter, (int, np.integer)) or not (50 <= maxiter <= 400):
        maxiter = 250
    if not isinstance(ftol, (int, float, np.floating)) or not (1e-12 <= float(ftol) <= 1e-6):
        ftol = 1e-10

    try:
        _s = int(seed)
    except (TypeError, ValueError, OverflowError):
        _s = 42
    rng = np.random.default_rng(_s)

    # Determine N and base vector
    N = 80
    base = np.full(N, 0.5)
    parent_valid = False
    if 'initial_h_values' in globals():
        iv = np.asarray(globals()['initial_h_values'], dtype=np.float64)
        if iv.ndim == 1 and 2 <= len(iv) <= 999 and np.all(np.isfinite(iv)):
            N = len(iv)
            base_p = project_feasible(iv.copy(), N / 2.0)
            if np.all(iv >= -1e-12) and np.all(iv <= 1.0 + 1e-12) and abs(np.sum(iv) - N / 2.0) <= 1e-8 * max(1, N):
                parent_valid = True
            base = base_p
    fallback = project_feasible(np.full(N, 0.5), N / 2.0)
    incumbent = base.copy()
    incumbent_score = exact_score(incumbent)

    # Build starts
    starts = [incumbent.copy()]
    starts.append(project_feasible(base + 0.03 * rng.standard_normal(N), N / 2.0))
    starts.append(project_feasible(base + 0.12 * rng.standard_normal(N), N / 2.0))
    starts.append(project_feasible(rng.uniform(0, 1, N), N / 2.0))
    K = min(6, (N - 1) // 2)
    starts.append(project_feasible(build_fourth_start(N, rng, K), N / 2.0))
    starts.append(project_feasible(build_binary_start(N, rng), N / 2.0))
    if n_starts < 6:
        starts = starts[:n_starts]
    elif n_starts > 6:
        extra = n_starts - 6
        for _ in range(extra):
            starts.append(project_feasible(base + 0.05 * rng.standard_normal(N), N / 2.0))

    # Optimize if N <= 256
    if N <= 256:
        bounds = [(0.0, 1.0)] * N + [(0.0, 2.5)]
        cons = [
            {'type': 'eq', 'fun': eq_fn, 'jac': eq_jac},
            {'type': 'ineq', 'fun': ineq_fn, 'jac': ineq_jac},
        ]
        for h0 in starts:
            if incumbent_score <= 0.3808:
                break
            s0 = exact_score(h0)
            if s0 < incumbent_score - 1e-12:
                incumbent = h0.copy()
                incumbent_score = s0
            t0 = max(0.0, np.max(np.correlate(h0, 1.0 - h0, mode='full'))) + 1e-8
            z0 = np.concatenate([h0, [t0]])
            try:
                res = minimize(obj_fn, z0, jac=obj_grad, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'ftol': ftol, 'maxiter': maxiter, 'disp': 0})
            except Exception:
                continue
            if res is not None and hasattr(res, 'x') and len(res.x) == N + 1:
                h_new = np.asarray(res.x[:N], dtype=np.float64)
                if np.all(np.isfinite(h_new)):
                    h_repaired = project_feasible(h_new, N / 2.0)
                    s_new = exact_score(h_repaired)
                    if s_new < incumbent_score - 1e-12:
                        incumbent = h_repaired
                        incumbent_score = s_new
    else:
        for h0 in starts:
            s0 = exact_score(h0)
            if s0 < incumbent_score - 1e-12:
                incumbent = h0.copy()
                incumbent_score = s0

    # Final validation
    final_h = project_feasible(incumbent, N / 2.0)
    final_score = exact_score(final_h)
    if not np.isfinite(final_score) or final_score > incumbent_score + 1e-6:
        final_h = project_feasible(base, N / 2.0)
        final_score = exact_score(final_h)
    if not np.isfinite(final_score):
        final_h = project_feasible(np.full(N, 0.5), N / 2.0)
        final_score = exact_score(final_h)
    return (final_h, float(final_score), int(N))
