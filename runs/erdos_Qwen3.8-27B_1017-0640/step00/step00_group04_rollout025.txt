    f2 = line_val(x2, q0, b, a, n)
    bt = x1 if f1 <= f2 else x2
    bv = f1 if f1 <= f2 else f2
    for _ in range(iters):
        if f1 < f2:
            right = x2; x2 = x1; f2 = f1
            x1 = right - phi * (right - left); f1 = line_val(x1, q0, b, a, n)
        else:
            left = x1; x1 = x2; f1 = f2
            x2 = left + phi * (right - left); f2 = line_val(x2, q0, b, a, n)
        if f1 < bv: bv, bt = f1, x1
        if f2 < bv: bv, bt = f2, x2
    return bt, bv

def do_line(h, d, n, cur_score):
    t_lo, t_hi = feasible_interval(h, d)
    if not (math.isfinite(t_lo) and math.isfinite(t_hi)) or t_hi - t_lo <= 1e-12:
        return None
    q0 = np.correlate(h, 1.0 - h, mode='full')
    b = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    a = -np.correlate(d, d, mode='full')
    if not (np.all(np.isfinite(b)) and np.all(np.isfinite(a))):
        return None
    grid = np.linspace(t_lo, t_hi, 17)
    ng = len(grid)
    vals = np.empty(ng)
    for i in range(ng):
        vals[i] = line_val(grid[i], q0, b, a, n)
    gmin = int(np.argmin(vals))
    bt, bv = grid[gmin], vals[gmin]
    cands = []
    for k in range(1, ng - 1):
        if vals[k] <= vals[k - 1] and vals[k] <= vals[k + 1]:
            cands.append((vals[k], k))
    cands.append((vals[gmin], gmin))
    cands.sort()
    for idx in range(min(2, len(cands))):
        k = cands[idx][1]
        if k < 1 or k >= ng - 1 or grid[k + 1] - grid[k - 1] < 1e-14:
            continue
        rt, rv = golden_cs(q0, b, a, n, grid[k - 1], grid[k + 1], 12)
        if rv < bv:
            bv, bt = rv, rt
    if bv >= cur_score - 1e-12:
        return None
    hn = h + bt * d
    hn = project_capped(hn, n)
    if not is_feasible(hn, n):
        return None
    s = exact_score(hn, n)
    if not math.isfinite(s) or s >= cur_score - 1e-12:
        return None
    return hn, s

def make_starts(n, rng, fallback):
    starts = [fallback.copy()]
    fb = fallback.copy()
    block = np.zeros(n); block[:n // 2] = 1.0
    if n % 2 == 1:
        block[n // 2] = 0.5
    proposals = [block, block[::-1].copy(), 1.0 - fb,
                 fb + 0.15 * rng.standard_normal(n),
                 rng.uniform(-0.25, 1.25, n),
                 0.5 + 0.25 * np.cos(2.0 * np.pi * np.arange(n) / n),
                 np.full(n, 0.5)]
    for sd in [0.05, 0.20, 0.35, 0.50]:
        proposals.append(fb + sd * rng.standard_normal(n))
    for p in proposals:
        if len(starts) >= 6:
            break
        if not np.all(np.isfinite(p)):
            continue
        proj = project_capped(p, n)
        if not is_feasible(proj, n):
            continue
        dup = False
        for s in starts:
            if float(np.max(np.abs(proj - s))) < 1e-9:
                dup = True
                break
        if not dup:
            starts.append(proj)
        if len(starts) >= 6:
            break
    return starts

def local_search(h, n, rng, max_sweeps):
    cur = h.copy()
    cs = exact_score(cur, n)
    if not math.isfinite(cs):
        return cur, 1.0
    for _ in range(max_sweeps):
        moved = False
        if n <= 64:
            pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
            rng.shuffle(pairs)
            for (i, j) in pairs:
                d = np.zeros(n); d[i] = 1.0; d[j] = -1.0
                res = do_line(cur, d, n, cs)
                if res is not None:
                    cur, cs = res; moved = True
        else:
            npairs = min(n * (n - 1) // 2, 4 * n)
            all_p = [(i, j) for i in range(n) for j in range(i + 1, n)]
            rng.shuffle(all_p)
            for (i, j) in all_p[:npairs]:
                d = np.zeros(n); d[i] = 1.0; d[j] = -1.0
                res = do_line(cur, d, n, cs)
                if res is not None:
                    cur, cs = res; moved = True
        nr = max(4, n // 4)
        for _ in range(nr):
            z = rng.standard_normal(n)
            z -= np.mean(z)
            z[-1] = -np.sum(z[:-1])
            mx = float(np.max(np.abs(z)))
            if mx < 1e-12 or not np.all(np.isfinite(z)):
                continue
            z /= mx
            res = do_line(cur, z, n, cs)
            if res is not None:
                cur, cs = res; moved = True
        if not moved:
            break
    return cur, cs

def run(seed=42, **kwargs):
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    n_starts = 6
    max_sweeps = 3
    try:
        n_starts = max(4, min(8, int(kwargs.get('n_starts', 6))))
    except Exception:
        pass
    try:
        max_sweeps = max(1, min(5, int(kwargs.get('max_sweeps', 3))))
    except Exception:
        pass
    rng = np.random.default_rng(seed)
    fb = None
    try:
        raw = initial_h_values
        arr = np.array(raw, dtype=np.float64).ravel()
        if 2 <= len(arr) <= 4096 and np.all(np.isfinite(arr)) and float(np.max(np.abs(arr))) <= 1e6:
            n = len(arr)
            tol = 1e-10 * max(1, n)
            if float(np.min(arr)) >= -1e-12 and float(np.max(arr)) <= 1.0 + 1e-12 and abs(float(np.sum(arr)) - n / 2.0) <= tol:
                if abs(float(np.sum(arr)) - n / 2.0) > 5e-13 * max(1, n):
                    arr = project_capped(arr, n)
                fb = arr.copy()
            else:
                proj = project_capped(arr, n)
                if is_feasible(proj, n):
                    fb = proj
    except Exception:
        pass
    if fb is None:
        n = 42
        fb = np.zeros(n)
        fb[:n // 2] = 1.0
        if n % 2 == 1:
            fb[n // 2] = 0.5
    n = len(fb)
    fb_score = exact_score(fb, n)
    if not math.isfinite(fb_score):
        fb_score = 1.0
    best_h = fb.copy()
    best_s = fb_score
    starts = make_starts(n, rng, fb)
    for st in starts:
        try:
            ss = exact_score(st, n)
            if not math.isfinite(ss):
                continue
            if ss < best_s - 1e-12:
                best_h, best_s = st.copy(), ss
            if n <= 128:
                rh, rs = local_search(st, n, rng, max_sweeps)
                if math.isfinite(rs) and rs < best_s - 1e-12:
                    best_h, best_s = rh.copy(), rs
        except Exception:
            continue
    try:
        if not is_feasible(best_h, n):
            best_h = project_capped(best_h, n)
        if not is_feasible(best_h, n):
            best_h = fb.copy()
        final_s = exact_score(best_h, n)
        if not math.isfinite(final_s):
            best_h, final_s = fb.copy(), fb_score
    except Exception:
        best_h, final_s = fb.copy(), fb_score
    return best_h, float(final_s), int(n)
