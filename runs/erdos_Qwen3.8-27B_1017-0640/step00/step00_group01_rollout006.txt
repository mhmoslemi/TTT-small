    for ns in [0.03, 0.10, 0.25]:
        p = capped_project(base + rng.standard_normal(n) * ns, n)
        if p is not None:
            starts.append(p)

    for _ in range(2):
        p = capped_project(rng.random(n), n)
        if p is not None:
            starts.append(p)

    for freq, amp in [(1, 0.25), (2, 0.25)]:
        phi = rng.uniform(0.0, 2.0 * math.pi)
        i = np.arange(n)
        v = 0.5 + amp * np.sin(2.0 * math.pi * freq * (i + 0.5) / n + phi)
        p = capped_project(v, n)
        if p is not None:
            starts.append(p)

    maxiter = 300 if n <= 64 else (180 if n <= 128 else 80)

    for start in starts:
        h0 = start.copy()
        sc = exact_score(h0)
        if sc < best_score - 1e-12:
            best_h = h0.copy()
            best_score = sc
        t0 = min(1.0, sc + 1e-9)
        z0 = np.append(h0, t0)
        res = solve_start(z0, n, maxiter)
        if res is not None:
            sc2 = exact_score(res)
            if sc2 < best_score - 1e-12:
                best_h = res.copy()
                best_score = sc2

    for ns in [0.02, 0.06]:
        p = capped_project(best_h + rng.standard_normal(n) * ns, n)
        if p is None:
            continue
        sc = exact_score(p)
        if sc < best_score - 1e-12:
            best_h = p.copy()
            best_score = sc
        t0 = min(1.0, sc + 1e-9)
        z0 = np.append(p, t0)
        res = solve_start(z0, n, maxiter)
        if res is not None:
            sc2 = exact_score(res)
            if sc2 < best_score - 1e-12:
                best_h = res.copy()
                best_score = sc2

    if not is_feasible(best_h, n):
        p = capped_project(best_h, n)
        if p is not None:
            best_h = p
        else:
            best_h = parent if parent is not None else np.full(n, 0.5)

    final_score = exact_score(best_h)
    if not np.isfinite(final_score):
        best_h = parent if parent is not None else np.full(n, 0.5)
        final_score = exact_score(best_h)

    return (best_h, float(final_score), n)
