                npop[i] = trial; nscores[i] = s
                if s < best_score - 1e-12:
                    best_score = s; best = trial.copy()
        pop, scores = npop, nscores
        fl = any(np.max(np.abs(pop[i] - best)) <= 1e-12 for i in range(P))
        if not fl:
            wi = int(np.argmax(scores)); pop[wi] = best.copy(); scores[wi] = best_score
        stagnant = 0 if best_score < gstart - 1e-10 else stagnant + 1
        if stagnant >= 18:
            if restarts >= 4:
                break
            nr = max(1, P // 3)
            sd = r_sds[min(restarts, 3)]
            ei = int(np.argmin(scores))
            others = sorted([i for i in range(P) if i != ei], key=lambda x: scores[x], reverse=True)
            for k, idx in enumerate(others[:nr]):
                if k < nr - 1:
                    cand = best + sd * rng.standard_normal(n)
                else:
                    cand = rng.uniform(-0.5, 1.5, n)
                cand = _project(cand, n)
                if cand is not None:
                    s = _score(cand, n)
                    if s is not None:
                        pop[idx] = cand; scores[idx] = s
                        if s < best_score - 1e-12:
                            best_score = s; best = cand.copy()
            stagnant = 0; restarts += 1

    # Final polling
    for scale in [0.12, 0.05, 0.02, 0.008, 0.003]:
        improved = False; cycles = 0
        while True:
            cycles += 1
            if cycles > 2:
                break
            for _ in range(PC):
                z = rng.standard_normal(n)
                z -= z.mean()
                zn = np.sqrt(np.mean(z * z))
                if zn > 1e-14:
                    z /= zn
                    tr = _project(best + scale * z, n)
                    if tr is not None:
                        s = _score(tr, n)
                        if s is not None and s < best_score - 1e-12:
                            best_score = s; best = tr.copy(); improved = True
            if not improved:
                break

    # Validate and return
    if not _feasible(best, n):
        best = _project(best, n)
    if best is None:
        best = h_fb
    best = np.clip(best, 0.0, 1.0)
    c5 = _score(best, n)
    if c5 is None:
        best = h_fb
        c5 = _score(best, n)
    if c5 is None:
        c5 = 0.5
    return (best, c5, int(n))
