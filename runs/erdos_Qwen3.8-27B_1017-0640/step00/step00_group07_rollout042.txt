import numpy as np
from scipy.optimize import minimize

def is_feasible(h, n):
    if h.ndim != 1 or len(h) != n or not np.all(np.isfinite(h)):
        return False
    if np.any(h < 0) or np.any(h > 1):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def project_capped(y, n):
    if y.ndim != 1 or len(y) != n or not np.all(np.isfinite(y)):
        return np.full(n, 1.0 / n)
    if np.any(np.abs(y) > 1e12):
        return np.full(n, 1.0 / n)
    m = n / 2.0
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > m:
            lo = mid
        else:
            hi = mid
    p = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    residual = m - np.sum(p)
    for _ in range(2):
        if abs(residual) <= 2e-14 * max(1, n):
            break
        if residual > 0:
            slack = 1.0 - p
        else:
            slack = p
        avail = np.sum(slack)
        if avail < 1e-15:
            break
        frac = residual / avail
        p = p + frac * slack
        p = np.clip(p, 0.0, 1.0)
        residual = m - np.sum(p)
    if is_feasible(p, n):
        return p
    return np.full(n, 1.0 / n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    try:
        corr = np.correlate(h, 1.0 - h, mode='full')
        return float(np.max(corr) * (2.0 / n))
    except Exception:
        return float('inf')

def majorizer_vals_jac(h, r, n):
    dx = 2.0 / n
    n_lags = 2 * n - 1
    Q = np.zeros(n_lags)
    G = np.zeros((n_lags, n))
    # Zero lag
    Q[0] = dx * np.sum(r * (1 - r) + (1 - 2 * r) * (h - r))
    G[0, :] = dx * (1 - 2 * r)
    row = 1
    for s in range(1, n):
        p = n - s
        D = (h[:p] - r[:p]) + (h[s:] - r[s:])
        quad = 0.5 * D ** 2
        Q[row] = dx * np.sum(h[:p] * (1 - h[s:]) + quad)
        gA = np.zeros(n)
        gA[:p] += dx * (1 + h[:p] - r[:p] - r[s:])
        gA[s:] += dx * (h[s:] - r[:p] - r[s:])
        G[row, :] = gA
        row += 1
        Q[row] = dx * np.sum(h[s:] * (1 - h[:p]) + quad)
        gB = np.zeros(n)
        gB[:p] += dx * (h[:p] - r[:p] - r[s:])
        gB[s:] += dx * (1 - r[:p] - r[s:] + h[s:])
        G[row, :] = gB
        row += 1
    return Q, G

def _slobjective(z, n):
    return z[-1]

def _slobjective_grad(z, n):
    g = np.zeros(n + 1)
    g[-1] = 1.0
    return g

def _sl_eq(z, n):
    return np.sum(z[:n]) - n / 2.0

def _sl_eq_jac(z, n):
    j = np.zeros(n + 1)
    j[:n] = 1.0
    return j

def _sl_ineq(z, n, r):
    h = z[:n]
    t = z[-1]
    Q, _ = majorizer_vals_jac(h, r, n)
    return t - Q

def _sl_ineq_jac(z, n, r):
    h = z[:n]
    _, G = majorizer_vals_jac(h, r, n)
    J = np.column_stack([-G, np.ones(2 * n - 1)])
    return J

def make_starts(base, n, rng, n_starts):
    starts = [project_capped(base.copy(), n)]
    # Noisy versions
    for scale in [0.04, 0.15]:
        noisy = base.copy() + scale * rng.standard_normal(n)
        starts.append(project_capped(noisy, n))
    # Uniform random
    starts.append(project_capped(rng.uniform(size=n), n))
    # Low-frequency
    K = min(6, (n - 1) // 2)
    y = np.full(n, 0.5)
    for k in range(1, K + 1):
        a = 0.12 / np.sqrt(k) * rng.standard_normal()
        b = 0.12 / np.sqrt(k) * rng.standard_normal()
        x = np.arange(n) + 0.5
        y += a * np.cos(2 * np.pi * k * x / n) + b * np.sin(2 * np.pi * k * x / n)
    starts.append(project_capped(y, n))
    # Binary
    perm = rng.permutation(n)
    yb = np.zeros(n)
    yb[perm[:n // 2]] = 1.0
    if n % 2 == 1:
        yb[perm[n // 2]] = 0.5
    starts.append(project_capped(yb, n))
    # Deduplicate
    unique = [starts[0]]
    for s in starts[1:]:
        dup = False
        for u in unique:
            if np.max(np.abs(s - u)) < 1e-10:
                dup = True
                break
        if not dup:
            unique.append(s)
    return unique[:n_starts]

def slsqp_subproblem(r, n, maxiter, ftol):
    dx = 2.0 / n
    Q0, _ = majorizer_vals_jac(r, r, n)
    t0 = np.max(Q0) + 1e-8
    z0 = np.concatenate([r, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'eq', 'fun': _sl_eq, 'jac': _sl_eq_jac, 'args': (n,)},
        {'type': 'ineq', 'fun': _sl_ineq, 'jac': _sl_ineq_jac, 'args': (n, r)}
    ]
    try:
        res = minimize(_slobjective, z0, jac=_slobjective_grad, args=(n,),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
        if res.x is None or len(res.x) != n + 1 or not np.all(np.isfinite(res.x)):
            return None
        return res.x[:n].copy()
    except Exception:
        return None

import numpy as np
from scipy.optimize import minimize

def is_feasible(h, n):
    if h.ndim != 1 or len(h) != n or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def project_capped(y, n):
    if y.ndim != 1 or len(y) != n or not np.all(np.isfinite(y)):
        return np.full(n, 1.0 / n)
    if np.any(np.abs(y) > 1e12):
        return np.full(n, 1.0 / n)
    m = n / 2.0
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(y - mid, 0.0, 1.0)) > m:
            lo = mid
        else:
            hi = mid
    p = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    residual = m - np.sum(p)
    for _ in range(3):
        if abs(residual) <= 2e-13 * max(1, n):
            break
        slack = (1.0 - p) if residual > 0 else p.copy()
        avail = np.sum(slack)
        if avail < 1e-15:
            break
        p = np.clip(p + (residual / avail) * slack, 0.0, 1.0)
        residual = m - np.sum(p)
    if is_feasible(p, n):
        return p
    return np.full(n, 1.0 / n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    try:
        return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))
    except Exception:

def majorizer_vals_jac(h, r, n):
    dx = 2.0 / n
    nl = 2 * n - 1
    Q = np.zeros(nl)
    G = np.zeros((nl, n))
    Q[0] = dx * np.sum(r * (1.0 - r) + (1.0 - 2.0 * r) * (h - r))
    G[0, :] = dx * (1.0 - 2.0 * r)
    row = 1
    for s in range(1, n):
        p = n - s
        D = (h[:p] - r[:p]) + (h[s:] - r[s:])
        quad = 0.5 * D * D
        Q[row] = dx * np.sum(h[:p] * (1.0 - h[s:]) + quad)
        gA = np.zeros(n)
        gA[:p] += dx * (1.0 + h[:p] - r[:p] - r[s:])
        gA[s:] += dx * (h[s:] - r[:p] - r[s:])
        G[row, :] = gA
        row += 1
        Q[row] = dx * np.sum(h[s:] * (1.0 - h[:p]) + quad)
        gB = np.zeros(n)
        gB[:p] += dx * (h[:p] - r[:p] - r[s:])
        gB[s:] += dx * (1.0 - r[:p] - r[s:] + h[s:])
        G[row, :] = gB
        row += 1
    return Q, G

def _sl_obj(z, n):
    return z[-1]

def _sl_obj_grad(z, n):
    g = np.zeros(n + 1)
    g[-1] = 1.0
    return g

def _sl_eq(z, n):
    return np.sum(z[:n]) - n / 2.0

def _sl_eq_jac(z, n):
    j = np.zeros(n + 1)
    j[:n] = 1.0
    return j

def _sl_ineq(z, n, r):
    Q, _ = majorizer_vals_jac(z[:n], r, n)
    return z[-1] - Q

def _sl_ineq_jac(z, n, r):
    _, G = majorizer_vals_jac(z[:n], r, n)
    return np.column_stack([-G, np.ones(2 * n - 1)])
