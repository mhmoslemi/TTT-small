import numpy as np
from scipy.optimize import minimize

def project_capped_simplex(y, n):
    y = np.asarray(y, dtype=np.float64).ravel()
    if y.shape[0] != n or not np.all(np.isfinite(y)):
        return None
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(y - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(200):
        resid = target - np.sum(x)
        if abs(resid) < 1e-12 * max(1, n):
            break
        if resid > 0:
            caps = 1.0 - x
            i = int(np.argmax(caps))
            if caps[i] <= 0:
                break
            x[i] += min(resid, caps[i])
        else:
            i = int(np.argmax(x))
            if x[i] <= 0:
                break
            x[i] += max(resid, -x[i])
    return np.clip(x, 0.0, 1.0)

def exact_score(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / len(h)))

def is_feasible(h, tol=1e-10):
    n = len(h)
    if h.ndim != 1 or n < 2 or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -tol) or np.any(h > 1.0 + tol):
        return False
    return abs(np.sum(h) - n / 2.0) <= tol * max(1, n)

def epi_con_fun(z):
    n = len(z) - 1
    dx = 2.0 / n
    h = z[:n]
    return z[-1] - dx * np.correlate(h, 1.0 - h, mode='full')

def epi_con_jac(z):
    n = len(z) - 1
    dx = 2.0 / n
    h = z[:n]
    J = np.zeros((2 * n - 1, n + 1))
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        for r in range(n):
            d = 0.0
            if 0 <= r - lag < n:
                d += 1.0 - h[r - lag]
            if 0 <= r + lag < n:
                d -= h[r + lag]
            J[k, r] = -dx * d
        J[k, n] = 1.0
    return J

def mass_fun(z):
    return np.sum(z[:-1]) - (len(z) - 1) / 2.0

def mass_jac(z):
    n = len(z) - 1
    J = np.zeros(n + 1)
    J[:n] = 1.0
    return J

def obj_fun(z):
    return z[-1]

def obj_grad(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g

def solve_epigraph(h0):
    n = len(h0)
    t0 = min(1.0, exact_score(h0) + 1e-8)
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * (n + 1)
    cons = [
        {'type': 'eq', 'fun': mass_fun, 'jac': mass_jac},
        {'type': 'ineq', 'fun': epi_con_fun, 'jac': epi_con_jac}
    ]
    try:
        res = minimize(obj_fun, z0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 250, 'ftol': 1e-10, 'disp': False})
    except Exception:
        return None
    if hasattr(res, 'x') and res.x is not None:
        x = np.asarray(res.x, dtype=np.float64)
        if x.shape[0] == n + 1 and np.all(np.isfinite(x)):
            hx = project_capped_simplex(x[:n], n)
            if hx is not None and is_feasible(hx):
                return hx
    return None

def construct_starts(n, anchor, rng, n_starts=8):
    starts = []
    block = np.zeros(n)
    block[:n // 2] = 1.0
    if n % 2 == 1:
        block[n // 2] = 0.5
    cand = [anchor, anchor[::-1], 1.0 - anchor, block, np.full(n, 0.5)]
    for c in cand:
        c = np.asarray(c, dtype=np.float64).ravel()
        if c.shape[0] != n or not np.all(np.isfinite(c)):
            continue
        p = project_capped_simplex(c, n)
        if p is None or not is_feasible(p):
            continue
        if not any(np.allclose(p, s, atol=1e-10) for s in starts):
            starts.append(p)
    scales = [0.08, 0.20, 0.40]
    base = anchor if anchor is not None else block
    while len(starts) < n_starts:
        if len(starts) % 2 == 0 and base is not None:

        if p is not None and is_feasible(p):

    return starts

def search_resolution(n, anchor, rng, n_starts=8):

    starts = construct_starts(n, anchor, rng, n_starts)

            sc2 = exact_score(h_sol)
            if sc2 < best_s - 1e-12:
                best_s, best_h = sc2, h_sol
    return best_h, best_s

        native_best = parent.copy()
