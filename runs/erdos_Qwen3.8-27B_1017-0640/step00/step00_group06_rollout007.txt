def project_capped_simplex(v, m):
    n = len(v)
    if not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))

        x = np.clip(v - mid, 0.0, 1.0)

        else:
            hi = mid

    return True


def project_capped_simplex(v, m):
    n = len(v)
    if not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
