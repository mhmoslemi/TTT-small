import numpy as np
from math import exp, log

def feasible_check(h, n):
    if h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10:
        return False
    return True

def project_capped(v, n):
    v = np.array(v, dtype=np.float64)
    if v.shape != (n,) or not np.all(np.isfinite(v)):
        return None
    v = np.clip(v, -1e6, 1e6)
    target = n / 2.0
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(80):
        mid = lo + (hi - lo) / 2.0
        q = np.clip(v - mid, 0.0, 1.0)
        if np.sum(q) > target:
            lo = mid
        else:
            hi = mid
    mid = lo + (hi - lo) / 2.0
    q = np.clip(v - mid, 0.0, 1.0)
    resid = target - np.sum(q)
    for i in range(n):
        if abs(resid) < 1e-13:
            break
        if resid > 0:
            add = min(resid, 1.0 - q[i])
            q[i] += add
            resid -= add
        else:
            sub = min(-resid, q[i])
            q[i] -= sub
            resid += sub
    if not feasible_check(q, n):
        return None
    return q

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    corr = np.correlate(h, 1.0 - h, mode='full')
    if corr.shape[0] != 2 * n - 1:
        return float('inf')
    if not np.all(np.isfinite(corr)):
        return float('inf')
    dx = 2.0 / n
    s = float(np.max(corr * dx))
    if not np.isfinite(s):
        return float('inf')
    return s

def pair_proposal(h, r, n, rng):
    for _ in range(8):
        p = int(rng.integers(0, n))
        q = int(rng.integers(0, n))
        if p == q:
            continue
        a_min = max(-h[p], h[q] - 1.0)
        a_max = min(1.0 - h[p], h[q])
        a_min = max(a_min, -r)
        a_max = min(a_max, r)
        if a_max - a_min <= 1e-12:
            continue
        alpha = rng.uniform(a_min, a_max)
        if abs(alpha) <= 1e-12:
            continue
        proposal = h.copy()
        proposal[p] += alpha
        proposal[q] -= alpha
        fp = project_capped(proposal, n)
        if fp is not None and feasible_check(fp, n):
            return fp
    return None

    if cur_c5 < best_c5 - 1e-12:
        best_h[:] = cur
        best_c5 = cur_c5

    temps = [0.006, 0.002, 0.0007, 0.00025, 0.00008]

        r = radii[stage]
        T = temps[stage]

            prop = pair_proposal(cur, r, n, rng)
            if prop is None:
                continue
            prop_c5 = exact_score(prop, n)
            if not np.isfinite(prop_c5):
                continue

                cur_c5 = prop_c5

                    cur_c5 = prop_c5

                best_c5 = cur_c5

def greedy_polish(h, c5, n, rng):
