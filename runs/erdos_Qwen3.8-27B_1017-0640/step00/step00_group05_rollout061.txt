    dx = 2.0 / N
    for j in range(2 * N - 1):
        k = j - (N - 1)
        lo1 = max(0, k)
        hi1 = min(N, N + k)
        if lo1 < hi1:
            G[j, lo1:hi1] += dx * (1.0 - h[lo1 - k:hi1 - k])
        lo2 = max(0, -k)
        hi2 = min(N, N - k)
        if lo2 < hi2:
            G[j, lo2:hi2] += dx * (-h[lo2 + k:hi2 + k])
    return G


def obj_fun(z, N):
    return z[N]


def obj_jac(z, N):
    g = np.zeros(N + 1)
    g[N] = 1.0
    return g


def eq_fun(z, m, N):
    return float(np.sum(z[:N])) - m


def eq_jac(z, m, N):
    g = np.zeros(N + 1)
    g[:N] = 1.0
    return g


def ineq_fun(z, N):
    h = z[:N]
    return z[N] - q_values(h, N)


def ineq_jac(z, N):
    h = z[:N]
    G = build_gradient(h, N)
    J = np.zeros((2 * N - 1, N + 1))
    J[:, :N] = -G
    J[:, N] = 1.0
    return J


def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)

    N = 70
    parent = None
    try:
        pv = np.array(initial_h_values, dtype=np.float64).flatten()
        if len(pv) > 0 and np.all(np.isfinite(pv)):
            N = len(pv)
            parent = pv.copy()
    except Exception:
        pass

    if N <= 0:
        N = 70
    m = float(N) / 2.0
    dx = 2.0 / N

    h_fallback = np.full(N, 0.5)

    if parent is not None:
        if np.all(parent >= -1e-10) and np.all(parent <= 1.0 + 1e-10) and abs(float(np.sum(parent)) - m) < 1e-8:
            h_base = np.clip(parent, 0.0, 1.0)
        else:
            h_base = project_box_sum(parent, m, N)
    else:
        h_base = np.full(N, 0.5)

    best_h = h_fallback.copy()
    best_c = exact_score(h_fallback, N)

    c = exact_score(h_base, N)
    if c < best_c - 1e-12:
        best_c = c
        best_h = h_base.copy()

    starts = [h_base.copy()]

    if N <= 256:
        for s in [0.03, 0.08, 0.18]:
            starts.append(project_box_sum(h_base + s * rng.standard_normal(N), m, N))
        starts.append(project_box_sum(rng.uniform(0.0, 1.0, N), m, N))
        h_comp = np.clip(1.0 - h_base, 0.0, 1.0)
        if abs(float(np.sum(h_comp)) - m) > 1e-8:
            h_comp = project_box_sum(h_comp, m, N)
        starts.append(h_comp)
        idx = (np.arange(N) + 0.5) / N
        starts.append(project_box_sum(0.5 + 0.45 * np.cos(2.0 * np.pi * idx), m, N))
        starts.append(project_box_sum(0.5 + 0.45 * np.cos(4.0 * np.pi * idx + np.pi / 3.0), m, N))
    elif N <= 999:
        starts.append(project_box_sum(h_base + 0.08 * rng.standard_normal(N), m, N))
        starts.append(project_box_sum(rng.uniform(0.0, 1.0, N), m, N))
        starts.append(project_box_sum(rng.beta(0.7, 0.7, N), m, N))

    for hs in starts:
        c = exact_score(hs, N)
        if c < best_c - 1e-12:
            best_c = c
            best_h = hs.copy()

    if N <= 999:
        bounds = [(0.0, 1.0)] * N + [(0.0, 2.0)]
        constraints = [
            {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (m, N)},
            {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac, 'args': (N,)},
        ]
        for h0 in starts:
            try:
                q0 = q_values(h0, N)
                t0 = float(np.clip(np.max(q0), 0.0, 2.0))
                z0 = np.empty(N + 1)
                z0[:N] = h0
                z0[N] = t0
                res = minimize(
                    func=obj_fun, x0=z0, args=(N,),
                    jac=obj_jac, method='SLSQP',
                    bounds=bounds, constraints=constraints,
                    options={'maxiter': 250, 'ftol': 1e-10, 'disp': False}
                )
                if res.x is not None and len(res.x) == N + 1:
                    h_new = np.array(res.x[:N], dtype=np.float64)
                    if np.all(np.isfinite(h_new)):
                        h_new = project_box_sum(h_new, m, N)
                        c_new = exact_score(h_new, N)
                        if c_new < best_c - 1e-12:
                            best_c = c_new
                            best_h = h_new.copy()
            except Exception:
                continue

    best_h = project_box_sum(best_h, m, N)
    final_c = exact_score(best_h, N)

    if not np.all(np.isfinite(best_h)) or np.any(best_h < -1e-10) or np.any(best_h > 1.0 + 1e-10) or abs(float(np.sum(best_h)) - m) > 1e-6:
        best_h = h_fallback
        final_c = exact_score(h_fallback, N)

    return (best_h, final_c, N)
