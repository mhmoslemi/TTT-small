    n = len(y)
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y))
    for _ in range(80):
        mid = (lo + hi) * 0.5

        else:

        x[i] += d * step
        res -= d * step
    return x

import numpy as np
from scipy.optimize import minimize


def exact_score(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))


def is_feasible(h, tol=1e-10):
    n = len(h)
    if not np.all(np.isfinite(h)):
        return False
    if np.min(h) < -tol or np.max(h) > 1.0 + tol:
        return False
    if abs(np.sum(h) - n / 2.0) > tol * max(1, n):
        return False
    return True


def project_onto_simplex(y):
    n = len(y)
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y))
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        step_dir = 1.0 if res > 0 else -1.0
        order = np.argsort(x) if res > 0 else np.argsort(x)[::-1]
        for i in order:
            if abs(res) < 1e-14:
                break
            if res > 0:
                d = min(1.0 - x[i], res)
            else:
                d = min(x[i], -res)
            x[i] += d * step_dir
            res -= d * step_dir
    return x


def lag_jac_row(h, l):
    n = len(h)
    g = np.zeros(n)
    if l >= 0:
        i0, i1 = l, n
    else:
        i0, i1 = 0, n + l
    if i0 < i1:
        idx = np.arange(i0, i1)
        g[idx] += (1.0 - h[idx - l])
    r0 = max(0, -l)
    r1 = min(n, n - l)
    if r0 < r1:
        ridx = np.arange(r0, r1)
        g[ridx] -= h[ridx + l]
    return g


def slsqp_obj(z):
    return z[-1]


def slsqp_obj_grad(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g


def slsqp_eq(z):
    n = len(z) - 1
    return np.sum(z[:n]) - n / 2.0


def slsqp_eq_jac(z):
    n = len(z) - 1
    j = np.zeros(n + 1)
    j[:n] = 1.0
    return j


def slsqp_ineq(z):
    n = len(z) - 1
    h = z[:n]
    t = z[n]
    return t - (2.0 / n) * np.correlate(h, 1.0 - h, mode='full')


def slsqp_ineq_jac(z):
    n = len(z) - 1
    h = z[:n]
    dx = 2.0 / n
    nl = 2 * n - 1
    j = np.zeros((nl, n + 1))
    j[:, n] = 1.0
    for k in range(nl):
        l = k - (n - 1)
        j[k, :n] = -dx * lag_jac_row(h, l)
    return j


def build_starts(n, anchor, rng, n_starts):
    starts = []
    if anchor is not None:
        starts.append(project_onto_simplex(anchor))
        starts.append(project_onto_simplex(anchor[::-1]))
        starts.append(project_onto_simplex(1.0 - anchor[::-1]))
    hb = np.zeros(n)
    hb[:n // 2] = 1.0
    if n % 2 == 1:
        hb[n // 2] = 0.5
    starts.append(hb.copy())
    starts.append(np.full(n, 0.5))
    base = anchor if anchor is not None else hb
    scales = [0.08, 0.20, 0.40]
    while len(starts) < n_starts:
        xi = len(starts)
        if xi % 2 == 0:
            s = scales[min(xi // 2, len(scales) - 1)]
            raw = base + s * rng.standard_normal(n)
        else:
            raw = rng.uniform(-0.25, 1.25, size=n)
        starts.append(project_onto_simplex(raw))
    unique = []
    for s in starts:
        dup = False
        for u in unique:
            if np.max(np.abs(s - u)) < 1e-10:
                dup = True
                break
        if not dup:
            unique.append(s)
    return unique


def optimize_at_resolution(n, starts, maxiter):
    best_h = None
    best_score = float('inf')
    for h0 in starts:
        h0 = project_onto_simplex(h0)
        if not is_feasible(h0):
            continue
        s0 = exact_score(h0)
        if s0 < best_score:
            best_score = s0
            best_h = h0.copy()
        t0 = min(1.0, s0 + 1e-8)
        z0 = np.concatenate([h0, [t0]])
        bounds = [(0.0, 1.0)] * (n + 1)
        cons = [
            {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac},
            {'type': 'ineq', 'fun': slsqp_ineq, 'jac': slsqp_ineq_jac},
        ]
        try:
            res = minimize(slsqp_obj, z0, method='SLSQP', jac=slsqp_obj_grad,
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-10})
            if hasattr(res, 'x') and res.x is not None:

                        if s_sol < best_score - 1e-12:
                            best_score = s_sol
                            best_h = h_sol
        except Exception:
            pass
    return best_h, best_score
