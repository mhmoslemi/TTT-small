import numpy as np
from scipy.optimize import minimize

def project_capped(y, n):
    m = n / 2.0
    y = np.clip(np.asarray(y, dtype=np.float64), -1e6, 1e6)
    if y.shape != (n,):
        return np.full(n, 0.5)
    lo, hi = y.min() - 1.0, y.max()
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(y - mid, 0, 1)) > m:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    r = m - x.sum()
    if r > 1e-12:
        for i in range(n):
            if x[i] < 1.0:
                d = min(r, 1.0 - x[i]); x[i] += d; r -= d
                if r < 1e-12: break
    elif r < -1e-12:
        r = -r
        for i in range(n):
            if x[i] > 0.0:
                d = min(r, x[i]); x[i] -= d; r -= d
                if r < 1e-12: break
    x = np.clip(x, 0.0, 1.0)
    if not np.all(np.isfinite(x)) or abs(x.sum() - m) > 1e-9:
        return np.full(n, 0.5)
    return x

def is_feasible(h, n):
    m = n / 2.0
    if h.shape != (n,) or not np.all(np.isfinite(h)): return False
    if h.min() < -1e-12 or h.max() > 1 + 1e-12: return False
    return abs(h.sum() - m) < 1e-9

def exact_score(h, n):
    v = np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)
    return float(v) if np.isfinite(v) else None

def build_G(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    G[n - 1] = dx * (1.0 - 2.0 * h)
    for p in range(1, n):
        G[n - 1 - p, p:] += dx * (1.0 - h[:-p])
        G[n - 1 - p, :-p] -= dx * h[p:]
        G[n - 1 + p, :-p] += dx * (1.0 - h[p:])
        G[n - 1 + p, p:] -= dx * h[:-p]
    return G

def _obj(z):
    return z[-1]

def _obj_j(z):
    g = np.zeros(len(z)); g[-1] = 1.0
    return g

def _eqf(z):
    return z[:-1].sum() - (len(z) - 1) / 2.0

def _eqj(z):
    n = len(z) - 1
    return np.append(np.ones(n), 0.0)

def _ineqf(z):
    n = len(z) - 1
    return z[-1] - np.correlate(z[:-1], 1.0 - z[:-1], mode='full') * (2.0 / n)

def _ineqj(z):
    n = len(z) - 1
    J = np.zeros((2 * n - 1, n + 1))
    J[:, :-1] = -build_G(z[:-1], n)
    J[:, -1] = 1.0
    return J

def make_bnds(n):
    return [(0.0, 1.0)] * (n + 1)

def make_cons():
    return [{'type': 'eq', 'fun': _eqf, 'jac': _eqj},
            {'type': 'ineq', 'fun': _ineqf, 'jac': _ineqj}]

def block_vec(n):
    m = n / 2.0
    x = np.zeros(n)
    k = int(m)
    x[:k] = 1.0
    if n % 2 == 1:
        x[k] = 0.5
    return x

def try_optimize(x0, n, maxiter, ftol):
    x0 = project_capped(x0, n)
    sc = exact_score(x0, n)
    t0 = min(1.0, (sc if sc is not None else 1.0) + 1e-9)
    z0 = np.append(x0, t0)
    try:
        res = minimize(_obj, z0, method='SLSQP', jac=_obj_j,
                       bounds=make_bnds(n), constraints=make_cons(),
                       maxiter=maxiter, ftol=ftol, disp=False)
    except Exception:
        return None
    if res is None or not hasattr(res, 'x'):
        return None
    hx = res.x[:-1]
    if len(hx) != n or not np.all(np.isfinite(hx)):
        return None
    hx = project_capped(hx, n)
    if not is_feasible(hx, n):
        return None
    return hx

def build_starts(parent, n, rng):
    starts = []
    p = project_capped(parent.copy(), n)
    starts.append(p)
    starts.append(project_capped(1.0 - p, n))
    starts.append(project_capped(p[::-1].copy(), n))
    starts.append(project_capped(np.roll(p, n // 3), n))
    starts.append(project_capped(p + 0.05 * rng.standard_normal(n), n))
    starts.append(project_capped(p + 0.18 * rng.standard_normal(n), n))
    starts.append(np.full(n, 0.5))
    bv = block_vec(n)
    starts.append(bv)
    starts.append(bv[::-1].copy())
    starts.append(project_capped(rng.random(n), n))
    starts.append(project_capped(0.5 + 0.35 * np.cos(2 * np.pi * rng.random(n) * 3 + rng.uniform(0, 6.28)), n))
    return starts

def dedup(starts, tol=1e-6):
    out = []
    for s in starts:
        if not any(np.max(np.abs(s - o)) < tol for o in out):
            out.append(s)
    return out

def run(seed=42, **kwargs):
    n = 45
    if 'n_points' in kwargs:
        try:
            nt = int(kwargs['n_points'])
            if 1 <= nt <= 999:
                n = nt
        except (ValueError, TypeError):
            pass
    parent = None
    try:
        ihv = globals().get('initial_h_values', None)
        if ihv is not None:
            ihv = np.asarray(ihv, dtype=np.float64)
            if ihv.ndim == 1 and ihv.size > 0 and np.all(np.isfinite(ihv)):
                if ihv.size == n:
                    parent = ihv.copy()
                else:
                    parent = np.interp(np.linspace(0, 1, n),
                                       np.linspace(0, 1, ihv.size), ihv)
    except Exception:
        parent = None
    if parent is None:
        parent = np.full(n, 0.5)
    parent = project_capped(parent, n)
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    if n <= 100:
        max_starts, maxiter, ftol = 10, 250, 1e-10
        refine_sigs, refine_rounds = [0.03, 0.10, 0.25], 2
    elif n <= 300:
        max_starts, maxiter, ftol = 7, 120, 1e-10
        refine_sigs, refine_rounds = [0.05, 0.18], 1
    else:
        max_starts, maxiter, ftol = 4, 60, 1e-9
        refine_sigs, refine_rounds = [0.08], 1
    starts = dedup(build_starts(parent, n, rng))[:max_starts]
    best = parent.copy()
    best_s = exact_score(best, n)
    if best_s is None:
        best_s = 1.0
    for s0 in starts:
        hx = try_optimize(s0, n, maxiter, ftol)
        if hx is None:
            continue
        sc = exact_score(hx, n)
        if sc is not None and sc < best_s - 1e-12:
            best, best_s = hx, sc
    for _rnd in range(refine_rounds):
        s0val = best_s
        for sig in refine_sigs:
            pert = project_capped(best + sig * rng.standard_normal(n), n)
            hx = try_optimize(pert, n, maxiter, ftol)
            if hx is None:
                continue
            sc = exact_score(hx, n)
            if sc is not None and sc < best_s - 1e-12:
                best, best_s = hx, sc
        if best_s > s0val - 1e-9:
            break
    best = project_capped(best, n)
    if not is_feasible(best, n):
        best = project_capped(parent, n)
    final_s = exact_score(best, n)
    if final_s is None:
        best = np.full(n, 0.5)
        final_s = exact_score(best, n)
    return (best, final_s, n)
