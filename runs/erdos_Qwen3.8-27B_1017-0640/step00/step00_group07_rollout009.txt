import numpy as np
from scipy.optimize import minimize
from functools import partial

def project_feasible(y, n):
    m = n / 2.0
    y = np.asarray(y, dtype=np.float64)
    if len(y) != n or not np.all(np.isfinite(y)):
        return np.full(n, 0.5)
    lo = y.min() - 1.0
    hi = y.max()
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    res = m - h.sum()
    if abs(res) > 1e-13:
        if res > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for i in order:
            if abs(res) < 1e-14:
                break
            d = min(abs(res), (1.0 - h[i]) if res > 0 else h[i])
            if d > 0:
                h[i] += np.sign(res) * d
                res -= np.sign(res) * d
    if not np.all(np.isfinite(h)) or abs(h.sum() - m) > 1e-10 * max(1, n):
        return np.full(n, 0.5)
    np.clip(h, 0.0, 1.0, out=h)
    return h

def exact_score(h):
    n = len(h)
    if n < 1 or not np.all(np.isfinite(h)):
        return np.inf
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def get_q_and_jac(h, n):
    dx = 2.0 / n
    nlags = 2 * n - 1
    q = np.zeros(nlags)
    G = np.zeros((nlags, n))
    q[0] = dx * np.dot(h, 1.0 - h)
    G[0, :] = dx * (1.0 - 2.0 * h)
    for s in range(1, n):
        hp = h[:n - s]
        hs = h[s:]
        q[s] = dx * np.dot(hp, 1.0 - hs)
        G[s, :n - s] += dx * (1.0 - hs)
        G[s, s:] += dx * (-hp)
        q[n + s - 1] = dx * np.dot(hs, 1.0 - hp)
        G[n + s - 1, s:] += dx * (1.0 - hp)
        G[n + s - 1, :n - s] += dx * (-hs)
    return q, G

def slsqp_obj(z, n):
    return z[n]

def slsqp_obj_jac(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def slsqp_ineq(z, n):
    q, _ = get_q_and_jac(z[:n], n)
    return z[n] - q

def slsqp_ineq_jac(z, n):
    _, G = get_q_and_jac(z[:n], n)
    jac = np.zeros((2 * n - 1, n + 1))
    jac[:, :n] = -G
    jac[:, n] = 1.0
    return jac

def slsqp_eq(z, n):
    return np.sum(z[:n]) - n / 2.0

def slsqp_eq_jac(z, n):
    g = np.ones(n + 1)
    g[n] = 0.0
    return g

    con_eq = {'type': 'eq', 'fun': partial(slsqp_eq, n=n), 'jac': partial(slsqp_eq_jac, n=n)}
    con_ineq = {'type': 'ineq', 'fun': partial(slsqp_ineq, n=n), 'jac': partial(slsqp_ineq_jac, n=n)}
    constraints = (con_eq, con_ineq)

        best_score = exact_score(incumbent)
