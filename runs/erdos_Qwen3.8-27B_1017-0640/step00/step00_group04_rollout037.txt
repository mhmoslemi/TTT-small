        elif m == 2:
            v = rng.uniform(0.0, 1.0, n)
        else:
            v = rng.uniform(-0.25, 1.25, n)
        p = project_capped(v, n)
        if p is not None and is_feasible(p, n):
            pop[ai] = p
        else:
            pop[ai] = fallback
        ai += 1
    scores = np.full(NP, 1e18, dtype=np.float64)
    for i in range(NP):
        s = exact_score(pop[i])
        scores[i] = s if s is not None else fs
    best_score = float(scores.min())
    best_vec = pop[int(np.argmin(scores))].copy()
    stag = 0
    inj_count = 0
    for gen in range(gens):
        bs0 = best_score
        snap = pop.copy()
        ss = scores.copy()
        npop = pop.copy()
        nscores = scores.copy()
        for i in range(NP):
            idx = list(range(NP))
            idx.pop(i)
            r1, r2, r3 = rng.choice(idx, size=3, replace=False)
            F = rng.uniform(0.45, 0.90)
            CR = rng.uniform(0.65, 1.0)
            v = snap[int(r1)] + F * (snap[int(r2)] - snap[int(r3)])
            mask = rng.random(n) < CR
            mask[rng.integers(n)] = True
            u = np.where(mask, v, snap[i])
            p = project_capped(u, n)
            if p is None or not is_feasible(p, n):
                continue
            s = exact_score(p)
            if s is None:
                continue
            if s < ss[i] - 1e-12:
                npop[i] = p
                nscores[i] = s
            if s < best_score - 1e-12:
                best_score = s
                best_vec = p.copy()
        pop = npop
        scores = nscores
        if best_score < bs0 - 1e-12:
            stag = 0
        else:
            stag += 1
        if stag >= 20 and inj_count < 3:
            order = np.argsort(scores)
            nrep = max(1, math.ceil(0.20 * NP))
            reps = order[-nrep:]
            isc = [0.04, 0.10, 0.25, 0.40]
            for j in range(len(reps)):
                ix = int(reps[j])
                if j % 2 == 0:
                    v = best_vec + rng.normal(0.0, isc[j % 4], n)
                else:
                    v = rng.uniform(-0.25, 1.25, n)
                p = project_capped(v, n)
                if p is not None and is_feasible(p, n):
                    s = exact_score(p)
                    if s is not None:
                        pop[ix] = p
                        scores[ix] = s
                        if s < best_score - 1e-12:
                            best_score = s
                            best_vec = p.copy()
            inj_count += 1
            stag = 0
        if inj_count >= 3 and stag >= 20:
            break
    cs = exact_score(best_vec)
    if best_vec is not None and cs is not None and is_feasible(best_vec, n) and cs < fs - 1e-12:
        return best_vec.copy(), cs, n
    return fallback.copy(), fs, n
