                    r1, r2, r3 = pick_distinct(rng, others, 3)
                donor = snap[r1] + F * (snap[r2] - snap[r3])

            mask = rng.random(n) < CR
            mask[int(rng.integers(0, n))] = True
            trial = np.where(mask, donor, tgt)
            proj = project_capped_simplex(trial, n)
            if proj is None:
                continue
            if np.max(np.abs(proj - tgt)) <= 1e-12:
                continue
            sc = exact_score(proj, n)
            if sc < tgt_sc - 1e-12:
                new_pop[i] = proj
                new_sc[i] = sc
                if sc < best_score - 1e-12:
                    best = proj.copy()
                    best_score = sc

        pop = new_pop
        pop_sc = new_sc

        best_in_pop = any(np.max(np.abs(p - best)) <= 1e-12 for p in pop)
        if not best_in_pop:
            wi = int(np.argmax(pop_sc))
            pop[wi] = best.copy()
            pop_sc[wi] = best_score

        if best_score < gen_best_start - 1e-10:
            stagnant = 0
        else:
            stagnant += 1

        if stagnant >= 18 and restarts < 4:
            stagnant = 0
            n_rep = max(1, floor(P // 3))
            scale = restart_scales[restarts]
            w_sorted = list(np.argsort(np.array(pop_sc))[::-1])
            for ri in range(n_rep):
                idx_r = w_sorted[ri]
                if ri % 2 == 0:
                    pert = best + scale * rng.standard_normal(n)
                    cand = project_capped_simplex(pert, n)
                else:
                    cand = project_capped_simplex(rng.uniform(-0.5, 1.5, size=n), n)
                if cand is not None:
                    pop[idx_r] = cand
                    ps = exact_score(cand, n)
                    pop_sc[idx_r] = ps
                    if ps < best_score - 1e-12:
                        best = cand.copy()
                        best_score = ps
            restarts += 1
        elif stagnant >= 36 and restarts >= 4:
            break

    # Final polling
    n_prop = 16 if n <= 64 else (10 if n <= 160 else 5)
    for scale in [0.12, 0.05, 0.02, 0.008, 0.003]:
        improved = False
        for cyc in range(2):
            for _ in range(n_prop):
                z = rng.standard_normal(n)
                z = z - z.mean()
                rms = sqrt(np.mean(z ** 2))
                if rms > 1e-15:
                    z = z / rms
                trial = project_capped_simplex(best + scale * z, n)
                if trial is None:
                    continue
                sc = exact_score(trial, n)
                if sc < best_score - 1e-12:
                    best = trial.copy()
                    best_score = sc
                    improved = True
            if not improved:
                break

    if not is_feasible(best, n):
        p = project_capped_simplex(best, n)
        best = p if p is not None else (parent_fb if parent_fb is not None else np.full(n, 0.5))
    best = np.clip(best, 0.0, 1.0)
    c5 = exact_score(best, n)
    if not np.isfinite(c5):
        best = parent_fb if parent_fb is not None else np.full(n, 0.5)
        c5 = exact_score(best, n)
    return best.astype(np.float64), c5, int(n)
