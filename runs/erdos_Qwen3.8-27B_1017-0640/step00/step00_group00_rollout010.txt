            try_add(starts, 0.5 + 0.35 * np.cos(2.0 * math.pi * r * ii / n + ph), n, m)
        try_add(starts, rng.standard_normal(n), n, m)
        try_add(starts, rng.standard_normal(n), n, m)
    starts = starts[:max_starts]
    best_h = None
    best_c5 = float('inf')
    for st in starts:
        sc = exact_score(st, n)
        if sc < best_c5:
            best_c5 = sc
            best_h = st.copy()
        lh, ls = local_optimize(st, n, m)
        if ls < best_c5 - 1e-12:
            best_c5 = ls
            best_h = lh.copy()
    if best_h is None:
        best_h = np.full(n, m / n, dtype=np.float64)
        best_c5 = exact_score(best_h, n)
    if n <= 64:
        rounds, sigs = 2, [0.05, 0.18]
    elif n <= 160:
        rounds, sigs = 1, [0.08, 0.20]
    else:
        rounds, sigs = 1, [0.10]
    for rd in range(rounds):
        sc0 = best_c5
        for sig in sigs:
            pert = project_to_K(best_h + sig * rng.standard_normal(n), m)
            if not is_feasible(pert, n, m):
                continue
            sc = exact_score(pert, n)
            if sc < best_c5 - 1e-12:
                best_c5 = sc
                best_h = pert.copy()
            lh, ls = local_optimize(pert, n, m)
            if ls < best_c5 - 1e-12:
                best_c5 = ls
                best_h = lh.copy()
        if best_c5 >= sc0 - 1e-9:
            break
    if not is_feasible(best_h, n, m):
        best_h = project_to_K(best_h, m)
        if not is_feasible(best_h, n, m):
            best_h = np.full(n, m / n, dtype=np.float64)
    c5 = exact_score(best_h, n)
    if not np.isfinite(c5):
        best_h = np.full(n, m / n, dtype=np.float64)
        c5 = exact_score(best_h, n)
    return (best_h, c5, int(n))
