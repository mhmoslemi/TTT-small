import numpy as np
from scipy.optimize import minimize
from functools import partial


def project_feasible(y, n):
    m = n / 2.0
    y = np.asarray(y, dtype=np.float64)
    if not np.all(np.isfinite(y)):
        return np.full(n, 0.5)
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y))
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(y - mid, 0.0, 1.0)) > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    res = m - np.sum(h)
    for _ in range(n + 2):
        if abs(res) <= 2e-14 * n:
            break
        if res > 0:
            idx = np.where(h < 1.0 - 1e-15)[0]
        else:
            idx = np.where(h > 1e-15)[0]
        if len(idx) == 0:
            break
        h[idx] = np.clip(h[idx] + res / len(idx), 0.0, 1.0)
        res = m - np.sum(h)
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(np.sum(h) - m) > 1e-10 * n:
        return np.full(n, 0.5)
    return h


def exact_score(h):
    n = len(h)
    if n == 0 or not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))


def q_and_jac(h, n):
    dx = 2.0 / n
    m = 2 * n - 1
    q = np.empty(m)
    G = np.empty((m, n))
    q[0] = dx * np.sum(h * (1.0 - h))
    G[0] = dx * (1.0 - 2.0 * h)
    idx = 1
    for s in range(1, n):
        nms = n - s
        q[idx] = dx * np.dot(h[:nms], 1.0 - h[s:])
        q[idx + 1] = dx * np.dot(h[s:], 1.0 - h[:nms])
        ga = np.zeros(n)
        ga[:nms] = dx * (1.0 - h[s:])
        ga[s:] -= dx * h[:nms]
        G[idx] = ga
        gb = np.zeros(n)
        gb[s:] = dx * (1.0 - h[:nms])
        gb[:nms] -= dx * h[s:]
        G[idx + 1] = gb
        idx += 2
    return q, G


def _obj(z, n):
    return z[n]


def _obj_g(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g


def _eq(z, n):
    return np.sum(z[:n]) - n / 2.0


def _eq_g(z, n):
    g = np.ones(n + 1)
    g[n] = 0.0
    return g


def _ineq(z, n):
    return z[n] - q_and_jac(z[:n], n)[0]


def _ineq_g(z, n):
    _, G = q_and_jac(z[:n], n)
    g = np.zeros((2 * n - 1, n + 1))
    g[:, :n] = -G
    g[:, n] = 1.0
    return g


def run(seed=42, **kwargs):
    try:
        s = int(seed)
    except (TypeError, ValueError):
        s = 42
    n_starts = 6
    maxiter = 250
    ftol = 1e-10
    if 'n_starts' in kwargs:
        try:
            n_starts = max(1, min(8, int(kwargs['n_starts'])))
        except (TypeError, ValueError):
            pass
    if 'maxiter' in kwargs:
        try:
            maxiter = max(50, min(400, int(kwargs['maxiter'])))
        except (TypeError, ValueError):
            pass
    if 'ftol' in kwargs:
        try:
            ftol = max(1e-12, min(1e-6, float(kwargs['ftol'])))
        except (TypeError, ValueError):
            pass

    n = 80
    base = np.full(n, 0.5)
    parent_valid = False
    if 'initial_h_values' in globals():
        pv = np.array(initial_h_values, dtype=np.float64, copy=True)
        if pv.ndim == 1 and 2 <= len(pv) <= 999 and np.all(np.isfinite(pv)):
            n = len(pv)
            base = project_feasible(pv, n)
            if (np.all(pv >= -1e-12) and np.all(pv <= 1.0 + 1e-12)
                    and abs(np.sum(pv) - n / 2.0) <= 1e-8 * n):
                parent_valid = True

    rng = np.random.default_rng(s)
    starts = [base.copy()]
    starts.append(project_feasible(base + 0.03 * rng.standard_normal(n), n))
    starts.append(project_feasible(base + 0.12 * rng.standard_normal(n), n))
    starts.append(project_feasible(rng.uniform(0.0, 1.0, n), n))
    K = min(6, (n - 1) // 2)
    fourier = np.full(n, 0.5)
    for k in range(1, K + 1):
        freq = 2.0 * np.pi * k * (np.arange(n, dtype=np.float64) + 0.5) / n
        fourier += (rng.normal(0.0, 0.12 / np.sqrt(k)) * np.cos(freq)
                    + rng.normal(0.0, 0.12 / np.sqrt(k)) * np.sin(freq))
    starts.append(project_feasible(fourier, n))
    perm = rng.permutation(n)
    binary = np.zeros(n)
    binary[perm[:n // 2]] = 1.0
    if n % 2 == 1:
        binary[perm[n // 2]] = 0.5
    starts.append(project_feasible(binary, n))
    for _ in range(max(0, n_starts - 6)):
        starts.append(project_feasible(base + 0.06 * rng.standard_normal(n), n))
    starts = starts[:n_starts]

    incumbent = starts[0].copy()
    best_score = exact_score(incumbent)

    if n <= 256:
        bounds = [(0.0, 1.0)] * n + [(0.0, 2.01)]
        cons = [

        for h0 in starts:

            if sc < best_score - 1e-12:
                best_score = sc
                incumbent = h0.copy()
            q0 = q_and_jac(h0, n)[0]
            t0 = max(0.0, float(np.max(q0))) + 1e-8
            z0 = np.concatenate([h0, [t0]])
            try:
                res = minimize(_obj, z0, jac=partial(_obj_g, n=n), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'ftol': ftol, 'maxiter': maxiter, 'disp': False})
                hr = np.asarray(res.x[:n], dtype=np.float64)
                if len(hr) == n and np.all(np.isfinite(hr)):
                    hr = project_feasible(hr, n)
                    sc = exact_score(hr)
                    if sc < best_score - 1e-12:
                        best_score = sc
                        incumbent = hr
            except Exception:
                pass
            if best_score <= 0.3808:
                break

    incumbent = project_feasible(incumbent, n)
    c5 = exact_score(incumbent)
    if (not np.all(np.isfinite(incumbent)) or np.any(incumbent < 0.0)
            or np.any(incumbent > 1.0) or abs(np.sum(incumbent) - n / 2.0) > 1e-10 * n):
        if parent_valid:
            incumbent = base.copy()
        else:
            incumbent = np.full(n, 0.5)
        c5 = exact_score(incumbent)
    return (incumbent, c5, int(n))
