    for _ in range(3):
        r = target - float(np.sum(x))
        if abs(r) <= 5e-13 * max(1, n):
            break
        if r > 0:
            for i in range(n):
                add = min(r, 1.0 - x[i])
                if add > 0:
                    x[i] += add
                    r -= add
                if abs(r) <= 5e-13 * max(1, n):
                    break
        else:
            r2 = -r
            for i in range(n):
                sub = min(r2, x[i])
                if sub > 0:
                    x[i] -= sub
                    r2 -= sub
                if abs(r2) <= 5e-13 * max(1, n):
                    break
    return x


def check_feasible(h, n):
    if h.ndim != 1 or len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(float(np.sum(h)) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True


def exact_score(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    if len(corr) != 2 * n - 1 or not np.all(np.isfinite(corr)):
        return float('inf')
    return float(np.max(corr) * (2.0 / n))


def encode_lattice(h, D):
    n = len(h)
    M = D * n // 2
    raw = D * h
    base = np.floor(raw).astype(np.int64)
    frac = raw - base
    base = np.clip(base, 0, D)
    R = M - int(np.sum(base))
    if R > 0:
        order = np.argsort(-frac, kind='stable')
        for idx in order:
            if R <= 0:
                break
            if base[idx] < D:
                base[idx] += 1
                R -= 1
    elif R < 0:
        order = np.argsort(frac, kind='stable')
        for idx in order:
            if R >= 0:
                break
            if base[idx] > 0:
                base[idx] -= 1
                R += 1
    if R != 0 or np.any(base < 0) or np.any(base > D):
        return None
    return base


def decode_lattice(k, D):
    n = len(k)
    h = k.astype(np.float64) / float(D)
    if not np.all(np.isfinite(h)) or np.any(h < -1e-15) or np.any(h > 1.0 + 1e-15):
        return None
    if abs(float(np.sum(h)) - n / 2.0) > 1e-9 * max(1, n):
        return None
    return h


def construct_fallback(n):
    h = np.zeros(n, dtype=np.float64)
    h[:n // 2] = 1.0
    if n % 2 == 1:
        h[n // 2] = 0.5
    return h


def make_chain_starts(parent, global_h, prev_k_scaled, rng, D, n):
    fb_k = encode_lattice(construct_fallback(n), D)
    if fb_k is None:
        fb_k = np.zeros(n, dtype=np.int64)
        fb_k[:n // 2] = D
        if n % 2 == 1:
            fb_k[n // 2] = D // 2
    starts = []
    k0 = encode_lattice(parent, D)
    starts.append(k0 if k0 is not None else fb_k.copy())
    k1 = encode_lattice(global_h, D)
    starts.append(k1 if k1 is not None else starts[0].copy())
    if prev_k_scaled is not None:
        starts.append(prev_k_scaled.copy())
    else:
        k2 = encode_lattice(parent[::-1].copy(), D)
        starts.append(k2 if k2 is not None else starts[0].copy())
    k3 = encode_lattice(1.0 - parent, D)
    starts.append(k3 if k3 is not None else starts[0].copy())
    raw_r = rng.uniform(-0.25, 1.25, size=n)
    pr = capped_simplex_project(raw_r, n)
    k4 = encode_lattice(pr, D)
    starts.append(k4 if k4 is not None else starts[0].copy())
    return starts


def anneal_chain(k, D, n, T_hi, T_lo, n_blocks, props, rng):
    curr = k.copy()
    curr_h = decode_lattice(curr, D)
    curr_s = exact_score(curr_h, n) if curr_h is not None else float('inf')
    best_k = curr.copy()
    best_s = curr_s
    for j in range(n_blocks):
        T = T_hi * (T_lo / T_hi) ** (j / float(max(1, n_blocks - 1)))
        for _ in range(props):
            a = int(rng.integers(n))
            b = int(rng.integers(n))
            if a == b:
                continue
            u = rng.random()
            if u < 0.70:
                mx = 1
            elif u < 0.90:
                mx = max(2, D // 32)
            else:
                mx = max(2, D // 8)
            m = int(rng.integers(1, mx + 1))
            if curr[a] + m > D or curr[b] < m:
                continue
            nk = curr.copy()
            nk[a] += m
            nk[b] -= m
            nh = nk.astype(np.float64) / float(D)
            ns = exact_score(nh, n)
            if not np.isfinite(ns):
                continue
            if ns < best_s - 1e-12:
                best_s = ns
                best_k = nk.copy()
            d = ns - curr_s
            if d <= 0.0:
                curr = nk
                curr_s = ns
            elif T > 1e-15 and (-d / T) < 700.0:
                if rng.random() < math.exp(-d / T):
                    curr = nk
                    curr_s = ns
    return best_k, best_s


def run(seed=42, **kwargs):
    try:
        seed = int(seed)
    except (ValueError, TypeError):
        seed = 42
    seed = seed % (2 ** 32)
    rng = np.random.default_rng(seed)

    parent = None
    n = 42
    try:
        pv = np.array(initial_h_values, dtype=np.float64)
        if pv.ndim > 1:
            pv = pv.ravel()
        if 2 <= len(pv) <= 1024 and np.all(np.isfinite(pv)) and np.all(np.abs(pv) < 1e6):
            n = len(pv)
            parent = pv.copy()
    except (NameError, Exception):
        pass

    if parent is not None:
        if not check_feasible(parent, n):
            parent = capped_simplex_project(parent, n)
        if not check_feasible(parent, n):
            parent = None

    if parent is None:
        parent = construct_fallback(n)

    fallback = parent.copy()
    fb_score = exact_score(fallback, n)
    ginc_h = fallback.copy()
    ginc_s = fb_score

    if n > 64:
        return fallback, fb_score, n

    D_list = [64, 256, 1024]
    g = min(2.0, max(0.5, 42.0 / float(n)))
    Th = 0.004 * g
    Tl = 2e-5 * g
    nb = 24
    ppb = max(50, int(round(160.0 * min(1.0, (42.0 / float(n)) ** 2))))
    nc = 5

    prev_k = None
    prev_D = None

    for di in range(len(D_list)):
        D = D_list[di]
        if prev_k is not None:
            prev_k_scaled = prev_k * (D // prev_D)
        else:
            prev_k_scaled = None

        starts = make_chain_starts(parent, ginc_h, prev_k_scaled, rng, D, n)

        for si in range(nc):
            hs = decode_lattice(starts[si], D)
            if hs is not None:
                ss = exact_score(hs, n)
                if np.isfinite(ss) and ss < ginc_s - 1e-12:
                    ginc_s = ss
                    ginc_h = hs.copy()

        den_best_k = None
        den_best_s = float('inf')
        for si in range(nc):
            try:
                bk, bs = anneal_chain(starts[si], D, n, Th, Tl, nb, ppb, rng)
                if bs < ginc_s - 1e-12:
                    bh = decode_lattice(bk, D)
                    if bh is not None and check_feasible(bh, n):
                        ginc_s = bs
                        ginc_h = bh.copy()
                if bs < den_best_s:
                    den_best_s = bs
                    den_best_k = bk.copy()
            except Exception:
                continue

        prev_k = den_best_k
        prev_D = D

    fh = ginc_h.copy()
    if not check_feasible(fh, n):
        fh = capped_simplex_project(fh, n)
    if not check_feasible(fh, n):
        fh = fallback.copy()
    fs = exact_score(fh, n)
    if not np.isfinite(fs) or fs > fb_score + 1e-12:
        fh = fallback.copy()
        fs = fb_score
    return fh, fs, n
