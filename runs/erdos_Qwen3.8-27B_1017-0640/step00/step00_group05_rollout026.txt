                    continue
                tr = poll.copy(); tr[ii] += a; tr[jj] -= a
                tr = small_resid_fix(tr, m)
                if tr is None or not is_feasible(tr, N, m):
                    continue
                tsc = score_c5(tr, N)
                if tsc is not None and tsc < best_tr_sc - 1e-12:
                    best_tr_sc, best_tr = tsc, tr.copy()
            if best_tr is not None:
                gb, gb_sc = best_tr, best_tr_sc
                if gb_sc < gbest_s - 1e-12:
                    gbest_s, gbest_h = gb_sc, gb.copy()

    fh = project_box_sum(gbest_h.copy(), m) if gbest_h is not None else None
    if fh is None or not is_feasible(fh, N, m):
        fh = parent.copy()
    fs = score_c5(fh, N)
    for cand in [parent, neutral]:
        cs = score_c5(cand, N)
        if cs is not None and (fs is None or cs < fs - 1e-12):
            fs, fh = cs, cand.copy()
    if fs is None:
        fh, fs = neutral.copy(), score_c5(neutral, N)
    return fh, fs, N
