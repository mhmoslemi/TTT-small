def feasible(h, n, m):
    if h.ndim != 1 or h.shape[0] != n:
        return False
    if not np.all(np.isfinite(h)):
        return False

        return False
    return True

def exact_score(h):

        return float('inf')

    q = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(q) * (2.0 / n))
