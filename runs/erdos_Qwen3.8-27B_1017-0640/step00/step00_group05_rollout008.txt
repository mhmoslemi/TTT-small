import numpy as np
from scipy.optimize import minimize

def project_box_sum(v, m, n):
    v = np.clip(np.asarray(v, dtype=np.float64), -1e6, 1e6)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > m:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(3):
        r = m - float(np.sum(y))
        if abs(r) < 1e-13:
            break
        if r > 0:
            for i in range(n):
                if r <= 1e-15:
                    break
                a = min(r, 1.0 - y[i])
                y[i] += a
                r -= a
        else:
            for i in range(n):
                if abs(r) <= 1e-15:
                    break
                a = min(-r, y[i])
                y[i] -= a
                r += a
    if not np.all(np.isfinite(y)):
        return np.full(n, 0.5)
    return np.clip(y, 0.0, 1.0)

def exact_score(h, N):
    dx = 2.0 / N
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(q))

def corr_gradient(h, N):
    dx = 2.0 / N
    Nl = 2 * N - 1
    J = np.zeros((Nl, N))
    idx = np.arange(N)
    for lag in range(Nl):
        k = lag - (N - 1)
        row = np.zeros(N)
        s1 = idx - k
        m1 = (s1 >= 0) & (s1 < N)
        if np.any(m1):
            row[m1] = dx * (1.0 - h[s1[m1]])
        s2 = idx + k
        m2 = (s2 >= 0) & (s2 < N)
        if np.any(m2):
            row[m2] -= dx * h[s2[m2]]
        J[lag] = row
    return J

def _obj(z):
    return z[-1]

def _obj_grad(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g

def _eq_fun(z, m):
    return float(np.sum(z[:-1])) - m

def _eq_jac(z, m):
    g = np.ones(len(z))
    g[-1] = 0.0
    return g

def _ineq_fun(z, N):
    h = z[:N]
    dx = 2.0 / N
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    return z[N] - q

def _ineq_jac(z, N):
    h = z[:N]
    Nl = 2 * N - 1
    J = corr_gradient(h, N)
    Jf = np.zeros((Nl, N + 1))
    Jf[:, :N] = -J
    Jf[:, N] = 1.0
    return Jf

    starts = [base.copy()]
    if N <= 256:
        for s in [0.03, 0.08, 0.18]:
            starts.append(project_box_sum(base + s * rng.standard_normal(N), m, N))
        starts.append(project_box_sum(rng.uniform(0.0, 1.0, N), m, N))
        starts.append(project_box_sum(rng.beta(0.7, 0.7, N), m, N))
        i = np.arange(N) + 0.5
        starts.append(project_box_sum(0.5 + 0.45 * np.cos(2.0 * np.pi * i / N), m, N))
        starts.append(project_box_sum(0.5 + 0.45 * np.cos(4.0 * np.pi * i / N + np.pi / 3), m, N))
    else:

        starts.append(project_box_sum(rng.uniform(0.0, 1.0, N), m, N))
        starts.append(project_box_sum(rng.beta(0.7, 0.7, N), m, N))
    return starts
