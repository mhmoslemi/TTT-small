                        v = rng.beta(0.7, 0.7, n)
                    else:
                        v = inc + [0.05, 0.15, 0.35, 0.70][(j - 1) % 4] * rng.standard_normal(n)
                    pv = project_capped_simplex(v)
                    if pv is not None and check_feasible(pv, n):
                        sc = score_c5(pv, n)
                        if sc is not None:
                            pop[jx] = pv; scores[jx] = sc
                            if sc < inc_score:
                                inc_score, inc = sc, pv.copy()
    fin = project_capped_simplex(inc)
    if fin is None or not check_feasible(fin, n):
        fin = parent.copy()
    fs = score_c5(fin, n)
    if fs is None:
        fin = np.full(n, 0.5); fs = score_c5(fin, n)
    if fs is None:
        fs = 0.5
    fin = np.clip(fin, 0.0, 1.0)
    return (fin.astype(np.float64), float(fs), int(n))
