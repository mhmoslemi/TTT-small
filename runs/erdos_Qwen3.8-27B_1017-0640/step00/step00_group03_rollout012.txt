    fallback = uniform.copy()
    best_score = exact_score(uniform, n)
    if parent is not None and parent.size == n and np.all(np.isfinite(parent)):
        pp = parent.copy()
        if np.all(pp >= 0) and np.all(pp <= 1) and abs(pp.sum() - n / 2.0) <= 1e-10:
            s = exact_score(pp, n)
            if np.isfinite(s) and s < best_score:
                fallback = pp.copy()
                best_score = s
        else:
            pj = capped_simplex_project(pp, n)
            s = exact_score(pj, n)
            if np.isfinite(s) and s < best_score:
                fallback = pj.copy()
                best_score = s

    incumbent = fallback.copy()
    if n > 256:
        return incumbent, best_score, n

    rng = np.random.default_rng(seed)
    starts = []
    if parent is not None and np.all(np.isfinite(parent)):
        starts.append(capped_simplex_project(parent, n))
        starts.append(capped_simplex_project(parent[::-1].copy(), n))
        starts.append(capped_simplex_project(parent + 0.05 * rng.standard_normal(n), n))
        starts.append(capped_simplex_project(parent + 0.15 * rng.standard_normal(n), n))
    else:
        idx = np.arange(n, dtype=np.float64)
        starts.append(uniform.copy())
        starts.append(capped_simplex_project(0.5 + 0.3 * np.sin(2.0 * np.pi * (idx + 0.5) / n), n))
        starts.append(capped_simplex_project(0.5 + 0.15 * rng.standard_normal(n), n))
        starts.append(capped_simplex_project(0.5 + 0.35 * rng.standard_normal(n), n))
    starts.append(uniform.copy())

    max_starts = 5 if n <= 128 else 4
    unique_starts = []
    for s in starts:
        dup = False
        for u in unique_starts:
            if np.max(np.abs(s - u)) < 1e-10:
                dup = True
                break
        if not dup:
            unique_starts.append(s)
        if len(unique_starts) >= max_starts:
            break

    for s in unique_starts:
        sc = exact_score(s, n)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            incumbent = s.copy()
            best_score = sc
        incumbent, best_score = gradient_search(s, n, incumbent, best_score, rng)

    if n <= 128:
        shake = capped_simplex_project(incumbent + 0.04 * rng.standard_normal(n), n)
        dup = False
        for u in unique_starts:
            if np.max(np.abs(shake - u)) < 1e-10:
                dup = True
                break
        if not dup:
            incumbent, best_score = gradient_search(shake, n, incumbent, best_score, rng)

    result = capped_simplex_project(incumbent, n)
    final_score = exact_score(result, n)
    fb_score = exact_score(fallback, n)
    if fb_score < final_score - 1e-12:
        result = fallback
        final_score = fb_score
    if not np.isfinite(final_score):
        result = uniform
        final_score = exact_score(uniform, n)
    return result, final_score, n
