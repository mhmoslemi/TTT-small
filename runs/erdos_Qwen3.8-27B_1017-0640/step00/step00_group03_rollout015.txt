def project_capped_simplex(v, n):
    """Project v onto {x: 0<=x[i]<=1, sum(x)=n/2}"""
    if not np.all(np.isfinite(v)):
        return np.full(n, 0.5)
    v = np.clip(v, -1e6, 1e6)
    target = n / 2.0
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        q = np.clip(v - mid, 0.0, 1.0)
        if np.sum(q) > target:
            lo = mid
        else:
            hi = mid
    q = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    r = target - np.sum(q)
    for i in range(n):
        if abs(r) <= 1e-13:
            break
        if r > 0:
            add = min(r, 1.0 - q[i])
            q[i] += add
            r -= add
        else:
            sub = min(-r, q[i])
            q[i] -= sub
            r += sub
    if not np.all(np.isfinite(q)) or np.any(q < -1e-12) or np.any(q > 1.0 + 1e-12):
        return np.full(n, 0.5)
    if abs(np.sum(q) - target) > 1e-10:
        return np.full(n, 0.5)
    q = np.clip(q, 0.0, 1.0)
    return q

def exact_score(h, n):
    """Compute C5 = max(correlate(h, 1-h, full) * dx)"""
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return float('inf')

        return float('inf')
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    if len(c) != 2 * n - 1 or not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

def lag_values(h, n, need_j(False)):

    Order: B_{n-1}, ..., B_1, A_0, A_1, ..., A_{n-1}

    vals = np.empty(2 * n - 1)
    J = None
    if need_j:

    idx = 0
    # B_d for d = n-1 down to 1
    for d in range(n - 1, 0, -1):
        u = h[:n - d]
        v = h[d:]
        B = np.sum(u * (1.0 - v)) * dx
        vals[idx] = B
        if need_j:

            gB[:n - d] += (1.0 - h[d:])
            gB[d:] -= h[:n - d]
            J[idx] = gB * dx
        idx += 1
    # A_0 (center)
    A0 = np.sum(h * (1.0 - h)) * dx
    vals[idx] = A0
    if need_j:
        J[idx] = (1.0 - 2.0 * h) * dx
    idx += 1
    # A_d for d = 1 to n-1
    for d in range(1, n):
        u = h[:n - d]
        v = h[d:]
        A = np.sum(v * (1.0 - u)) * dx
        vals[idx] = A
        if need_j:
            gA = np.zeros(n)
            gA[d:] += (1.0 - h[:n - d])
            gA[:n - d] -= h[d:]
            J[idx] = gA * dx
        idx += 1
    return vals, J

def surrogate_val(c, tau):
    """Compute log-sum-exp surrogate F_tau = tau * log(sum(exp(c/tau)))"""
    cmax = np.max(c)
    q = np.exp((c - cmax) / tau)
    s = np.sum(q)
    if not np.isfinite(s) or s <= 0:
        return float('inf'), None
    F = cmax + tau * np.log(s)
    p = q / s
    return F, p

    incumb_vec = start.copy()
    incumb_score = exact_score(start, n)

    for tau in temps:

        stalled = 0

            F, p = surrogate_val(c, tau)
            if not np.isfinite(F):

            if not np.all(np.isfinite(g)):

            accepted = False

                trial = project_capped_simplex(trial_raw, n)

                F_t, _ = surrogate_val(c_t, tau)
                delta = trial - incumb_vec
                if not np.isfinite(F_t):
                    alpha *= 0.5
                    continue
                lhs = F_t
                rhs = F + 1e-4 * (g @ delta) + 1e-12 * max(1.0, abs(F))
                if lhs <= rhs:
                    # Check improvement
                    ns = exact_score(trial, n)
                    if ns < incumb_score - 1e-12:
                        incumb_score = ns
                    incumb_vec = trial
                    disc = np.max(np.abs(delta))
                    alpha = min(1.5 * alpha, 2.0)
                    accepted = True

                    if F_t <= F - 1e-12 * max(1.0, abs(F)):
                        stalled = 0
                    else:

                    break
                alpha *= 0.5
            if stalled >= 8:

            if not accepted and it > 0:

    return incumb_vec, incumb_score

import numpy as np
import math

def project_capped_simplex(v, n):
    if not np.all(np.isfinite(v)):
        return np.full(n, 0.5)
    v = np.clip(np.asarray(v, dtype=np.float64), -1e6, 1e6)
    target = n / 2.0
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        q = np.clip(v - mid, 0.0, 1.0)
        if np.sum(q) > target:
            lo = mid
        else:
            hi = mid
    q = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    r = target - np.sum(q)
    for i in range(n):
        if abs(r) <= 1e-13:
            break
        if r > 0:
            add = min(r, 1.0 - q[i])
            q[i] += add
            r -= add
        else:
            sub = min(-r, q[i])
            q[i] -= sub
            r += sub
    if not np.all(np.isfinite(q)) or np.any(q < -1e-12) or np.any(q > 1.0 + 1e-12):
        return np.full(n, 0.5)
    if abs(np.sum(q) - target) > 1e-10:
        return np.full(n, 0.5)
    return np.clip(q, 0.0, 1.0)

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return float('inf')
    if abs(np.sum(h) - n / 2.0) > 1e-10:
        return float('inf')
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    if len(c) != 2 * n - 1 or not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

def lag_values(h, n, need_j):
    dx = 2.0 / n
    vals = np.empty(2 * n - 1)
    J = np.zeros((2 * n - 1, n)) if need_j else None
    idx = 0
    for d in range(n - 1, 0, -1):
        u = h[:n - d]
        v = h[d:]
        vals[idx] = np.sum(u * (1.0 - v)) * dx
        if need_j:
            gB = np.zeros(n)
            gB[:n - d] += (1.0 - v)
            gB[d:] -= u
            J[idx] = gB * dx
        idx += 1
    vals[idx] = np.sum(h * (1.0 - h)) * dx
    if need_j:
        J[idx] = (1.0 - 2.0 * h) * dx
    idx += 1
    for d in range(1, n):
        u = h[:n - d]
        v = h[d:]
        vals[idx] = np.sum(v * (1.0 - u)) * dx
        if need_j:
            gA = np.zeros(n)
            gA[d:] += (1.0 - u)
            gA[:n - d] -= v
            J[idx] = gA * dx
        idx += 1
    return vals, J

def surrogate(c, tau):
    cmax = np.max(c)
    q = np.exp((c - cmax) / tau)
    s = np.sum(q)
    if not np.isfinite(s) or s <= 0:
        return float('inf'), None
    F = cmax + tau * math.log(s)
    p = q / s
    if not np.isfinite(F):
        return float('inf'), None
    return F, p

def run_gradient(start, n, rng, max_iter):
    h = start.copy()
    best_vec = start.copy()
    best_sc = exact_score(start, n)
    temps = [0.02, 0.005, 0.001, 0.0002, 0.00004, 0.000008]
    for tau in temps:
        alpha = 0.5
        stalled = 0
        for _ in range(max_iter):
            c, J = lag_values(h, n, True)
            if not np.all(np.isfinite(c)):
                break
