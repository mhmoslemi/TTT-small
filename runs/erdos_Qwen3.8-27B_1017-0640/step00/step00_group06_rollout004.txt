    high = float(np.max(v))

        mid = (low + high) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = math.fsum(x)
        if s > m:
            low = mid
        else:
            high = mid
        if abs(s - m) <= 1e-13:
            break
    x = np.clip(v - (low + high) / 2.0, 0.0, 1.0)
    for _ in range(3):
        r = m - math.fsum(x)
        if abs(r) < 1e-12:
            break
        for i in range(n):
            if abs(r) < 1e-12:
                break
            if r > 0:
                add = min(r, 1.0 - x[i])
                x[i] += add
                r -= add
            else:
                sub = min(-r, x[i])
                x[i] -= sub
                r += sub
    if not np.all(np.isfinite(x)) or not is_feasible(x, n):
        return None
    return x


def constraint_fun(x):
    return math.fsum(x) - len(x) / 2.0


def constraint_jac(x):
    return np.ones(len(x))


def smooth_value_and_grad(h, tau, dx):
    n = len(h)
    q = np.correlate(h, 1.0 - h, mode='full')
    qmax = float(np.max(q))
    shifted = np.clip((q - qmax) / tau, -500.0, 500.0)
    exp_s = np.exp(shifted)
    sum_exp = float(np.sum(exp_s))
    if not np.isfinite(sum_exp) or sum_exp <= 0.0:
        return None, None
    p = exp_s / sum_exp
    fval = dx * (qmax + tau * math.log(sum_exp))
    conv1 = np.convolve(1.0 - h, p, mode='full')
    conv2 = np.convolve(h, p[::-1], mode='full')
    grad = dx * (conv1[n - 1:2 * n - 1] - conv2[n - 1:2 * n - 1])
    if not np.all(np.isfinite(grad)):
        return None, None
    return fval, grad


def fallback_h(n):
    h = np.zeros(n, dtype=np.float64)
    n_full = n // 2
    for i in range(n_full):
        h[i] = 1.0
    if n % 2 == 1:
        h[n_full] = 0.5
    return h


def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    n = 95
    parent = None
    try:
        arr = np.array(initial_h_values, dtype=np.float64, copy=True)
        if arr.ndim == 1 and 2 <= len(arr) < 1000:
            n = len(arr)
            if np.all(np.isfinite(arr)):
                parent = arr
    except (NameError, TypeError, ValueError):
        parent = None

    if parent is None and 'n_points' in kwargs:
        try:
            n = int(kwargs['n_points'])
            if not (2 <= n < 1000):
                n = 95
        except (TypeError, ValueError):
            n = 95

    dx = 2.0 / n
    m = n / 2.0

    if parent is not None and is_feasible(parent, n):
        base = parent.copy()
    elif parent is not None:
        proj = cap_simplex_project(parent, n)
        base = proj if proj is not None else fallback_h(n)
    else:
        base = fallback_h(n)

    starts = [base.copy()]
    for std in [0.05, 0.15, 0.30]:
        noised = base + rng.standard_normal(n) * std
        proj = cap_simplex_project(noised, n)
        starts.append(proj if proj is not None else base.copy())

    proj = cap_simplex_project(rng.uniform(0.0, 1.0, n), n)
    starts.append(proj if proj is not None else base.copy())

    proj = cap_simplex_project(rng.beta(0.7, 0.7, n), n)
    starts.append(proj if proj is not None else base.copy())

    phase = rng.uniform(0.0, 2.0 * math.pi)
    idx = np.arange(n, dtype=np.float64)
    sine = 0.5 + 0.45 * np.sin(2.0 * math.pi * idx / n + phase)
    proj = cap_simplex_project(sine, n)
    starts.append(proj if proj is not None else base.copy())

    perm = rng.permutation(n)
    proj = cap_simplex_project(base[perm], n)
    starts.append(proj if proj is not None else base.copy())

    best_h = None
    best_score = float('inf')
    for s in starts:
        sc = exact_score(s)
        if 0.0 <= sc < best_score:
            best_score = sc
            best_h = s.copy()

    tau_scale = n / 95.0
    taus = [0.10 * tau_scale, 0.03 * tau_scale, 0.008 * tau_scale,
            0.002 * tau_scale, 0.0005 * tau_scale]

    bounds = [(0.0, 1.0)] * n
    constraint = {'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}

    for start_arr in starts:
        current = start_arr.copy()
        for tau in taus:
            try:
                res = minimize(
                    smooth_value_and_grad, current,
                    args=(tau, dx),
                    method='SLSQP',
                    jac=True,
                    bounds=bounds,
                    constraints=constraint,
                    options={'maxiter': 120, 'ftol': 1e-9, 'disp': False}
                )
            except Exception:
                break
            if res.x is None or len(res.x) != n or not np.all(np.isfinite(res.x)):
                break
            proj = cap_simplex_project(res.x, n)
            if proj is None:
                break
            f_new, _ = smooth_value_and_grad(proj, tau, dx)
            f_old, _ = smooth_value_and_grad(current, tau, dx)
            if f_new is None or f_old is None or f_new > f_old + 1e-10:
                pass
            else:
                current = proj
            sc = exact_score(current)
            if sc >= 0.0 and sc < best_score - 1e-12:
                best_score = sc
                best_h = current.copy()

    if best_h is None or not is_feasible(best_h, n):
        proj = cap_simplex_project(best_h if best_h is not None else base, n)
        if proj is not None:
            best_h = proj
        else:
            best_h = fallback_h(n)

    c5 = exact_score(best_h)
    if c5 < 0.0:
        best_h = fallback_h(n)
        c5 = exact_score(best_h)

    return (best_h, float(c5), n)
