import numpy as np
import math

def project_capped_simplex(y):
    n = len(y)
    target = n / 2.0
    low = min(y) - 1.0
    high = max(y)
    for _ in range(80):
        mid = (low + high) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            low = mid
        else:
            high = mid
    x = np.clip(y - (low + high) / 2.0, 0.0, 1.0)
    for _ in range(3):
        r = target - np.sum(x)
        if abs(r) < 5e-13 * max(1, n):
            break
        if r > 0:
            for i in range(n):
                d = min(r, 1.0 - x[i])
                x[i] += d
                r -= d
        else:
            for i in range(n):
                d = min(-r, x[i])
                x[i] -= d
                r += d
    if not np.all(np.isfinite(x)) or np.any(x < -1e-12) or np.any(x > 1.0 + 1e-12):
        return None
    x = np.clip(x, 0.0, 1.0)
    if abs(np.sum(x) - target) > 1e-10 * max(1, n):
        return None
    return x

def check_feasible(h, n):
    if not isinstance(h, np.ndarray) or h.ndim != 1 or len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def exact_score(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    if len(corr) != 2 * n - 1 or not np.all(np.isfinite(corr)):
        return None
    return float(np.max(corr) * (2.0 / n))

def encode_lattice(h, D):
    n = len(h)
    M = D * n // 2
    raw = D * h
    base = np.floor(raw).astype(np.int64)
    frac = raw - base
    base = np.clip(base, 0, D)
    R = M - int(np.sum(base))
    idx = np.arange(n)
    if R > 0:
        order = idx[np.argsort(-frac, kind='stable')]
        for i in order[:R]:
            if base[i] < D:
                base[i] += 1
    elif R < 0:
        order = idx[np.argsort(frac, kind='stable')]
        for i in order[:(-R)]:
            if base[i] > 0:
                base[i] -= 1
    if int(np.sum(base)) != M or np.any(base < 0) or np.any(base > D):
        proj = project_capped_simplex(h.copy())
        if proj is None:
            return None
        raw = D * proj
        base = np.floor(raw).astype(np.int64)
        frac = raw - base
        base = np.clip(base, 0, D)
        R = M - int(np.sum(base))
        if R > 0:
            order = idx[np.argsort(-frac, kind='stable')]
            for i in order[:R]:
                if base[i] < D:
                    base[i] += 1
        elif R < 0:
            order = idx[np.argsort(frac, kind='stable')]
            for i in order[:(-R)]:
                if base[i] > 0:
                    base[i] -= 1
        if int(np.sum(base)) != M:
            return None
    return base

def decode_lattice(k, D):
    h = k.astype(np.float64) / float(D)
    n = len(h)
    if not np.all(np.isfinite(h)) or np.any(h < -1e-15) or np.any(h > 1.0 + 1e-15):
        return None
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return None
    return h

def anneal_chain(k, D, n, T_hi, T_lo, n_blocks, props_per_block, rng):
    M = D * n // 2
    k_cur = k.copy()
    h_cur = decode_lattice(k_cur, D)
    if h_cur is None:
        return k_cur, None
    score_cur = exact_score(h_cur)
    if score_cur is None:
        return k_cur, None
    best_k = k_cur.copy()
    best_score = score_cur
    class_maxs = (1, max(2, D // 32), max(2, D // 8))
    for j in range(n_blocks):
        T = T_hi * (T_lo / T_hi) ** (j / float(n_blocks - 1)) if n_blocks > 1 else T_lo
        for _ in range(props_per_block):
            a = rng.integers(n)
            b = rng.integers(n - 1)
            if b >= a:
                b += 1
            r = rng.random()
            if r < 0.70:
                cmax = class_maxs[0]
            elif r < 0.90:
                cmax = class_maxs[1]
            else:
                cmax = class_maxs[2]
            m = int(rng.integers(1, cmax + 1))
            if k_cur[a] + m > D or k_cur[b] < m:
                continue
            k_trial = k_cur.copy()
            k_trial[a] += m
            k_trial[b] -= m
            if int(np.sum(k_trial)) != M or np.any(k_trial < 0) or np.any(k_trial > D):
                continue
            h_trial = decode_lattice(k_trial, D)
            if h_trial is None:
                continue
            score_trial = exact_score(h_trial)
            if score_trial is None:
                continue
            if score_trial < best_score - 1e-12:
                best_score = score_trial
                best_k = k_trial.copy()
            delta = score_trial - score_cur
            if delta <= 0 or (T > 0 and rng.random() < math.exp(-delta / T)):
                k_cur = k_trial
                score_cur = score_trial
                h_cur = h_trial
    return best_k, best_score

def run(seed=42, **kwargs):
    try:
        seed = int(seed)
    except (TypeError, ValueError):
        seed = 42
    seed = seed % (2**32)
    rng = np.random.default_rng(seed)

    n = 42
    h_parent = None
    ihv = globals().get('initial_h_values', None)
    if ihv is not None:
        try:
            h_parent = np.array(ihv, dtype=np.float64).ravel()
            if h_parent.ndim != 1 or len(h_parent) < 2 or len(h_parent) > 1024:
                h_parent = None
            elif not np.all(np.isfinite(h_parent)) or np.any(np.abs(h_parent) > 1e6):
                h_parent = None
            else:
                n = len(h_parent)
                if np.any(h_parent < -1e-12) or np.any(h_parent > 1.0 + 1e-12):
                    h_parent = project_capped_simplex(h_parent.copy())
                elif abs(np.sum(h_parent) - n / 2.0) > 5e-13 * max(1, n):
                    h_parent = project_capped_simplex(h_parent.copy())
                if h_parent is not None and not check_feasible(h_parent, n):
                    h_parent = None
        except Exception:
            h_parent = None

    if h_parent is None:
        n = 42
        h_parent = np.zeros(n, dtype=np.float64)
        h_parent[:n // 2] = 1.0
        if n % 2 == 1:
            h_parent[n // 2] = 0.5
    h_fallback = h_parent.copy()
    score_fallback = exact_score(h_fallback)
    if score_fallback is None:
        score_fallback = 0.5

    global_best = h_parent.copy()
    global_score = score_fallback

    if n > 64:
        return global_best.copy(), global_score, n

    g = min(2.0, max(0.5, 42.0 / float(n)))
    T_hi = 0.004 * g
    T_lo = 2e-5 * g
    n_blocks = 24
    props_per_block = max(50, int(round(160 * min(1.0, (42.0 / n) ** 2))))
    Ds = (64, 256, 1024)

    prev_best_k = None
    for di, D in enumerate(Ds):
        starts = []
        h_enc = h_parent.copy()
        k0 = encode_lattice(h_enc, D)
        if k0 is not None:
            starts.append(k0)
        k1 = encode_lattice(global_best.copy(), D)
        if k1 is not None:
            starts.append(k1)
        if prev_best_k is not None:
            k2 = (prev_best_k * 4).astype(np.int64)
            if int(np.sum(k2)) == D * n // 2:
                starts.append(k2)
            else:
                hr = h_parent[::-1].copy()
                k2 = encode_lattice(hr, D)
                if k2 is not None:
                    starts.append(k2)
        else:
            hr = h_parent[::-1].copy()
            k2 = encode_lattice(hr, D)
            if k2 is not None:
                starts.append(k2)
        hc = 1.0 - h_parent
        if np.any(hc < 0):
            hc = np.clip(hc, 0, 1)
        hc = project_capped_simplex(hc)
        if hc is not None:
            k3 = encode_lattice(hc, D)
            if k3 is not None:
                starts.append(k3)
        y_rand = rng.uniform(-0.25, 1.25, size=n)
        y_rand = project_capped_simplex(y_rand)
        if y_rand is not None:
            k4 = encode_lattice(y_rand, D)
            if k4 is not None:
                starts.append(k4)
        if len(starts) == 0:
            k_def = np.full(n, D // 2, dtype=np.int64)
            for i in range(n):
                k_def[i] = 1 if i < n // 2 else 0
            M = D * n // 2
            excess = D * n // 2 - int(np.sum(k_def))
            for i in range(min(excess, n // 2)):
                k_def[i] += D - 1
            starts.append(k_def)

        for k_start in starts[:5]:
            try:
                bk, bs = anneal_chain(k_start, D, n, T_hi, T_lo, n_blocks, props_per_block, rng)
                if bs is not None and bs < global_score - 1e-12:
                    h_dec = decode_lattice(bk, D)
                    if h_dec is not None:
                        s = exact_score(h_dec)
                        if s is not None and s < global_score - 1e-12:
                            global_score = s
                            global_best = h_dec.copy()
                if bk is not None:
                    prev_best_k = bk
            except Exception:
                continue
            prev_best_k = starts[-1]

    h_final = global_best.copy()
    if not check_feasible(h_final, n):
        h_final = project_capped_simplex(h_final)
    if h_final is None or not check_feasible(h_final, n):
        h_final = h_fallback.copy()
    s_final = exact_score(h_final)
    if s_final is None:
        h_final = h_fallback.copy()
        s_final = exact_score(h_final)
    if s_final is None:
        s_final = 0.5
    if s_final > score_fallback + 1e-12:
        h_final = h_fallback.copy()
        s_final = score_fallback
    h_final = np.clip(h_final, 0.0, 1.0)
    return h_final, float(s_final), int(len(h_final))
