        u = 0.2 + 0.8 * rng.random()
        c = min(p[i], 1.0 - p[j])
        delta = rho * u * c
        p[i] -= delta
        p[j] += delta
    if not is_feasible(p, n):
        p = project_capped(p, n)
        if not is_feasible(p, n):
            return None
    return p

def run(seed=42, **kwargs):
    try:
        s = int(seed)
        if s < 0:
            s = 42
    except Exception:
        s = 42
    rng = np.random.default_rng(s)
    n = 80
    base = np.full(n, 0.5)
    try:
        ih = np.array(initial_h_values, dtype=np.float64).ravel()
        if 2 <= len(ih) <= 999:
            n = len(ih)
            base = project_capped(ih, n)
    except Exception:
        pass
    target = 0.3808
    parent_fallback = base.copy()
    best_h = base.copy()
    best_score = exact_score(best_h)
    score_count = 1
    n_s = 1 if n > 256 else 5
    if 'n_starts' in kwargs:
        try:
            n_s = max(1, min(5, int(kwargs['n_starts'])))
        except Exception:
            pass
    B = min(6000, max(40, int(40000000 / (n * n))))
    starts = [base.copy()]
    if n_s > 1:
        for pr, pk in [(0.18, min(8, max(2, math.ceil(n / 10.0)))),
                       (0.45, min(24, max(4, math.ceil(n / 3.0))))]:
            pp = base.copy()
            for _ in range(int(pk)):
                d_ = np.where(pp > 1e-14)[0]
                r_ = np.where(pp < 1.0 - 1e-14)[0]
                if len(d_) == 0 or len(r_) == 0:
                    break
                ii = int(d_[rng.integers(len(d_))])
                jj = int(r_[rng.integers(len(r_))])
                if ii != jj:
                    u = 0.2 + 0.8 * rng.random()
                    c = min(pp[ii], 1.0 - pp[jj])
                    dt = pr * u * c
                    pp[ii] -= dt
                    pp[jj] += dt
            pp = project_capped(pp, n)
            if is_feasible(pp, n) and all(np.max(np.abs(pp - st)) > 1e-12 for st in starts):
                starts.append(pp)
        idx = rng.permutation(n)
        pb = np.zeros(n)
        pb[idx[:n // 2]] = 1.0
        if n % 2 == 1:
            pb[idx[n // 2]] = 0.5
        pb = project_capped(pb, n)
        if is_feasible(pb, n) and all(np.max(np.abs(pb - st)) > 1e-12 for st in starts):
            starts.append(pb)
        pu = project_capped(rng.random(n), n)
        if is_feasible(pu, n) and all(np.max(np.abs(pu - st)) > 1e-12 for st in starts):
            starts.append(pu)
    starts = starts[:n_s]
    for st in starts:
        sc = exact_score(st)
        score_count += 1
        if sc < best_score - 1e-12:
            best_score = sc
            best_h = st.copy()
    T0 = 0.001
    if best_score > target and score_count < B:
        P = min(24, max(4, B // 20))
        K_p = max(1, n // 20)
        bs = exact_score(starts[0])
        score_count += 1
        diffs = []
        for _ in range(P):
            if score_count >= B:
                break
            prop = exchange_proposal(starts[0], n, rng, 0.18, K_p)
            score_count += 1
            if prop is None:
                continue
            sc = exact_score(prop)
            score_count += 1
            if sc < best_score - 1e-12:
                best_score = sc
                best_h = prop.copy()
            d = sc - bs
            if d > 1e-14:
                diffs.append(d)
        if diffs:
            D = float(np.median(diffs))
            T0 = min(0.02, max(1e-5, D / math.log(2.0)))
    if best_score > target:
        S = len(starts)
        A = max(1, (B - score_count) // (9 * S))
        for l in range(9):
            T_l = T0 * (0.5 ** l)
            rho_l = max(0.0025, 0.25 * (0.55 ** l))
            K_l = max(1, int(round(8 * (0.65 ** l))))
            for si in range(S):
                cur = starts[si].copy()
                cur_sc = exact_score(cur)
                score_count += 1
                for _ in range(A):
                    if score_count >= B or best_score <= target:
                        break
                    prop = exchange_proposal(cur, n, rng, rho_l, K_l)
                    score_count += 1
                    if prop is None:
                        continue
                    if np.max(np.abs(prop - cur)) < 1e-14:
                        continue
                    sc = exact_score(prop)
                    score_count += 1
                    if not np.isfinite(sc):
                        continue
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_h = prop.copy()
                    delta = sc - cur_sc
                    if delta <= 0:
                        cur = prop
                        cur_sc = sc
                    else:
                        ratio = delta / T_l
                        if ratio < 745.0 and rng.random() < math.exp(-ratio):
                            cur = prop
                            cur_sc = sc
    best_h = project_capped(best_h, n)
    bf = exact_score(best_h)
    pf_h = project_capped(parent_fallback, n)
    pf = exact_score(pf_h)
    if pf <= bf:
        final_h = pf_h
    else:
        final_h = best_h
    if not is_feasible(final_h, n):
        final_h = np.full(n, 0.5)
    final_h = np.clip(final_h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(final_h)
    if abs(diff) > 1e-12:
        final_h += diff / n
        final_h = np.clip(final_h, 0.0, 1.0)
    c5 = float(np.max(np.correlate(final_h, 1.0 - final_h, mode='full')) * (2.0 / n))
    if not np.isfinite(c5):
        final_h = np.full(n, 0.5)
        c5 = float(np.max(np.correlate(final_h, 1.0 - final_h, mode='full')) * (2.0 / n))
    return (final_h, c5, int(n))
