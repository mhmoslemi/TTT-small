    return h

def run(seed=42, **kwargs):
    N = 70
    base = None
    try:
        arr = np.array(initial_h_values, dtype=np.float64).ravel()
        if len(arr) > 0:
            N = len(arr)
            m_tmp = N / 2.0
            if np.all(np.isfinite(arr)):
                if is_feasible(arr, m_tmp):
                    base = arr.copy()
                else:
                    base = project_box_sum(arr, m_tmp)
    except Exception:
        N = 70
        base = None
    m = N / 2.0
    if base is None:
        base = np.full(N, 0.5)
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    neutral = np.full(N, 0.5)
    neutral_score = score_c5(neutral, N)
    global_best = base.copy()
    global_best_score = score_c5(global_best, N)
    if global_best_score is None:
        global_best = neutral.copy()
        global_best_score = neutral_score
    if neutral_score is not None and neutral_score < global_best_score - 1e-12:
        global_best = neutral.copy()
        global_best_score = neutral_score
    if N > 999 or N < 2:
        pass
    else:
        starts = make_starts(base, N, m, rng)
        if N <= 128:
            budget = 800
        elif N <= 256:
            budget = 400
        elif N <= 512:
            budget = 150
        else:
            budget = 40
        for st in starts:
            if not is_feasible(st, m):
                continue
            sc = score_c5(st, N)
            if sc is not None and sc < global_best_score - 1e-12:
                global_best = st.copy()
                global_best_score = sc
            res = anneal_trajectory(st, N, m, rng, budget)
            if is_feasible(res, m):
                rs = score_c5(res, N)
                if rs is not None and rs < global_best_score - 1e-12:
                    global_best = res.copy()
                    global_best_score = rs
        final = pair_poll(global_best, N, m, rng)
        if is_feasible(final, m):
            fs = score_c5(final, N)
            if fs is not None and fs < global_best_score - 1e-12:
                global_best = final.copy()
                global_best_score = fs
    fh = project_box_sum(global_best, m)
    if fh is None or not is_feasible(fh, m):
        if is_feasible(base, m):
            fh = base.copy()
        else:
            fh = neutral.copy()
    fs = score_c5(fh, N)
    if fs is None:
        fh = neutral.copy()
        fs = score_c5(fh, N)
    if fs is None:
        fh = np.full(N, 0.5)
        fs = float(N)
    return (fh, fs, N)
