        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def project_capped(y, n):
    m = n / 2.0
    uniform = np.full(n, 0.5)
    if y is None:
        return uniform.copy()
    y = np.asarray(y, dtype=np.float64).ravel()
    if len(y) != n or not np.all(np.isfinite(y)) or np.max(np.abs(y)) > 1e12:
        return uniform.copy()
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(y - mid, 0.0, 1.0)) > m:
            lo = mid
        else:
            hi = mid
    p = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    resid = m - np.sum(p)
    for _ in range(2):
        if abs(resid) <= 2e-14 * max(1, n):
            break
        if resid > 0:
            for i in range(n):
                if resid <= 0:
                    break
                add = min(resid, 1.0 - p[i])
                p[i] += add
                resid -= add
        else:
            for i in range(n):
                if resid >= 0:
                    break
                sub = min(-resid, p[i])
                p[i] -= sub
                resid += sub
    if is_feasible(p, n):
        return p
    return uniform.copy()

def exact_score(h, n):
    if h.ndim != 1 or len(h) != n or not np.all(np.isfinite(h)):
        return np.inf
    if np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return np.inf
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return np.inf
    try:
        val = max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)
        if not np.isfinite(val):
            return np.inf
        return float(val)
    except Exception:
        return np.inf

def majorizer_values_jac(h, r, n):
    dx = 2.0 / n
    d = h - r
    nlag = 2 * n - 1
    Q = np.zeros(nlag)
    G = np.zeros((nlag, n))
    # Zero lag
    q0_val = dx * np.sum(r * (1.0 - r))
    Q[0] = q0_val + dx * np.sum((1.0 - 2.0 * r) * d)
    G[0, :] = dx * (1.0 - 2.0 * r)
    row = 1
    for s in range(1, n):
        p = n - s
        D = d[:p] + d[s:]
        # A_s
        qA = dx * (np.sum(h[:p] * (1.0 - h[s:])) + 0.5 * np.sum(D * D))
        Q[row] = qA
        G[row, :p] += dx * (1.0 - h[s:p+s] + D)  # wait, h[s:] has length n-s = p
        # Actually h[s:] has length n-s = p, so h[s:s+p] = h[s:n] which is h[s:]
        G[row, :p] += dx * (1.0 - h[s:] + D)
        G[row, s:] += dx * (-h[:p] + D)
        row += 1
        # B_s
        qB = dx * (np.sum(h[s:] * (1.0 - h[:p])) + 0.5 * np.sum(D * D))
        Q[row] = qB
        G[row, :p] += dx * (-h[s:] + D)
        G[row, s:] += dx * (1.0 - h[:p] + D)
        row += 1
    return Q, G

def epigraph_fun(z, n, r):
    return z[n]

def epigraph_grad(z, n, r):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def epigraph_equality(z, n):
    return np.sum(z[:n]) - n / 2.0

def epigraph_eq_jac(z, n):
    J = np.zeros(n + 1)
    J[:n] = 1.0
    return J

def epigraph_ineq(z, n, r):
    h = z[:n]
    t = z[n]
    Q, _ = majorizer_values_jac(h, r, n)
    return t - Q

def epigraph_ineq_jac(z, n, r):
    h = z[:n]
    _, G = majorizer_values_jac(h, r, n)
    nlag = 2 * n - 1
    J = np.zeros((nlag, n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J
