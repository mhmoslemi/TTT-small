        for i in range(n):
            if r > 0:
                add = min(1.0 - q[i], r)
                q[i] += add
                r -= add
            elif r < 0:
                sub = min(q[i], -r)
                q[i] -= sub
                r += sub
    q = np.clip(q, 0.0, 1.0)
    if not np.all(np.isfinite(q)) or np.any(q < 0.0) or np.any(q > 1.0):
        return None
    if abs(float(np.sum(q)) - target) > 1e-10:
        return None
    return q

def score_exact(h, n):
    try:
        if h.ndim != 1 or len(h) != n:
            return float('inf')
        if not np.all(np.isfinite(h)):
            return float('inf')
        if np.any(h < 0.0) or np.any(h > 1.0):
            return float('inf')
        if abs(float(np.sum(h)) - n / 2.0) > 1e-10:
            return float('inf')
        corr = np.correlate(h, 1.0 - h, mode='full')
        if len(corr) != 2 * n - 1:
            return float('inf')
        if not np.all(np.isfinite(corr)):
            return float('inf')
        return float(np.max(corr) * (2.0 / n))
    except Exception:
        return float('inf')

def try_add(pop, hc, n, max_size):
    if hc is None:
        return
    hc = np.asarray(hc, dtype=np.float64)
    if hc.ndim != 1 or len(hc) != n or not np.all(np.isfinite(hc)):
        return
    for ex in pop:
        if float(np.max(np.abs(hc - ex))) < 1e-10:
            return
    if len(pop) < max_size:
        pop.append(hc.copy())

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)

    n = 70
    parent = None
    parent_feasible = None

    ihv = globals().get('initial_h_values', None)
    if ihv is not None:
        try:
            arr = np.asarray(ihv, dtype=np.float64).ravel()
            if arr.size >= 2:
                n = arr.size
                if np.all(np.isfinite(arr)):
                    if is_feasible(arr, n):
                        parent = arr.copy()
                        parent_feasible = arr.copy()
                    else:
                        pr = project_feasible(arr, n)
                        if pr is not None:
                            parent = pr
                            parent_feasible = pr
        except Exception:
            pass

    uniform = np.full(n, 0.5, dtype=np.float64)

    if n > 256:
        if parent_feasible is not None:
            s1 = score_exact(parent_feasible, n)
            s2 = score_exact(uniform, n)
            if s1 <= s2:
                return (parent_feasible, s1, n)
        return (uniform, score_exact(uniform, n), n)

    if n <= 128:
        P = 32
        G = 80
    else:
        P = 24
        G = 45

    fb = parent_feasible if parent_feasible is not None else uniform

    pop = []
    try_add(pop, fb, n, P)

    if parent is not None:
        try_add(pop, parent[::-1].copy(), n, P)
        try_add(pop, 1.0 - parent, n, P)
        try_add(pop, np.roll(parent, n // 4), n, P)
        try_add(pop, np.roll(parent, n // 2), n, P)

    try_add(pop, uniform, n, P)

    idx = 0.5 + np.arange(n, dtype=np.float64)
    for phi in [0.0, math.pi / 3.0, 2.0 * math.pi / 3.0]:
        try_add(pop, project_feasible(0.5 + 0.35 * np.sin(2.0 * math.pi * idx / n + phi), n), n, P)
    try_add(pop, project_feasible(0.5 + 0.2 * np.cos(4.0 * math.pi * idx / n), n), n, P)

    center = parent if parent is not None else uniform
    scales = [0.025, 0.06, 0.12, 0.20]
    attempts = 0
    while len(pop) < P and attempts < 10 * P:
        attempts += 1
        sc = scales[attempts % 4]
        try_add(pop, project_feasible(center + rng.standard_normal(n) * sc, n), n, P)
        if len(pop) < P and attempts % 3 == 0:
            try_add(pop, project_feasible(rng.random(n), n), n, P)

    while len(pop) < P:
        pop.append(fb.copy())

    P = len(pop)
    scores = []
    for i in range(P):
        s = score_exact(pop[i], n)
        if not math.isfinite(s):
            s = score_exact(fb, n)
        scores.append(s)

    arch_idx = int(np.argmin(scores))
    archive = pop[arch_idx].copy()
    arch_score = scores[arch_idx]

    stagnant = 0
    restarts = 0
    prev_arch = arch_score

    for gen in range(G):
        pop_snap = [x.copy() for x in pop]
        scores_snap = list(scores)
        next_pop = [x.copy() for x in pop]
        next_scores = list(scores)

        for i in range(P):
            try:
                if rng.random() < 0.75:
                    others = [j for j in range(P) if j != i]
                    r1, r2, r3 = rng.choice(others, size=3, replace=False)
                    if gen < G // 2:
                        F = float(rng.uniform(0.55, 0.90))
                    else:
                        F = float(rng.uniform(0.35, 0.75))
                    donor = pop_snap[int(r1)] + F * (pop_snap[int(r2)] - pop_snap[int(r3)])
                else:
                    others = [j for j in range(P) if j != i]
                    r1, r2 = rng.choice(others, size=2, replace=False)
                    if gen < G // 2:
                        F = float(rng.uniform(0.55, 0.90))
                    else:
                        F = float(rng.uniform(0.35, 0.75))
                    donor = pop_snap[i] + F * (archive - pop_snap[i]) + F * (pop_snap[int(r1)] - pop_snap[int(r2)])

                CR = float(rng.uniform(0.65, 1.0))
                mask = rng.random(n) < CR
                j_rand = int(rng.integers(0, n))
                mask[j_rand] = True
                cross = np.where(mask, donor, pop_snap[i])

                trial = project_feasible(cross, n)
                if trial is None:
                    continue
                ts = score_exact(trial, n)
                if not math.isfinite(ts):
                    continue
                if ts < arch_score:
                    archive = trial.copy()
                    arch_score = ts
                if ts < scores_snap[i]:
                    next_pop[i] = trial
                    next_scores[i] = ts
            except Exception:
                continue

        pop = next_pop
        scores = next_scores

        if arch_score < prev_arch - 1e-15:
            stagnant = 0
            prev_arch = arch_score
        else:
            stagnant += 1

        if stagnant >= 18 and restarts < 2:
            stagnant = 0
            restarts += 1
            sigma = 0.10 if restarts == 1 else 0.035
            bi = int(np.argmin(scores))
            new_pop = [archive.copy(), pop[bi].copy()]
            new_scores = [arch_score, scores[bi]]
            for _ in range(P - 2):
                if rng.random() < 0.75:
                    cand = project_feasible(archive + rng.standard_normal(n) * sigma, n)
                else:
                    cand = project_feasible(rng.random(n), n)
                if cand is None:
                    cand = fb.copy()
                new_pop.append(cand)
                new_scores.append(score_exact(cand, n))
            pop = new_pop
            scores = new_scores
            prev_arch = arch_score

    candidates = [(fb, score_exact(fb, n)), (uniform, score_exact(uniform, n)), (archive, arch_score)]
    best_h = None
    best_s = float('inf')
    for hc, hs in candidates:
        if hc is not None and math.isfinite(hs) and hs < best_s:
            best_s = hs
            best_h = hc.copy()

    if best_h is None or not is_feasible(best_h, n):
        best_h = uniform.copy()
        best_s = score_exact(uniform, n)

    final_score = score_exact(best_h, n)
    if not math.isfinite(final_score):
        best_h = uniform.copy()
        final_score = score_exact(uniform, n)

    return (best_h, final_score, n)
