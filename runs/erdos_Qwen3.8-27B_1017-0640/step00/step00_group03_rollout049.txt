    uni = np.full(n, 0.5)
    pf = None
    if parent is not None:
        if check_feasible(parent, n):
            pf = parent.copy()
        else:
            pf = project_q(parent, n)
    cands = []
    if pf is not None:
        add_cand(cands, pf.copy(), n)
        add_cand(cands, np.flip(pf).copy(), n)
        add_cand(cands, (1.0 - pf).copy(), n)
    add_cand(cands, uni.copy(), n)
    idx = np.arange(n) + 0.5
    add_cand(cands, project_q(0.5 + 0.3 * np.sin(2 * np.pi * idx / n), n), n)
    if pf is not None:
        add_cand(cands, project_q(pf + 0.08 * rng.standard_normal(n), n), n)
    else:
        add_cand(cands, project_q(0.5 + 0.2 * rng.standard_normal(n), n), n)
    best_h, best_s = uni.copy(), score_c5(uni, n)
    if pf is not None:
        sp = score_c5(pf, n)
        if sp < best_s - 1e-12:
            best_h, best_s = pf.copy(), sp
    for s in cands:
        sh = project_q(s, n)
        if sh is None:
            continue
        if n > 128:
            ss = score_c5(sh, n)
            if ss < best_s - 1e-12:
                best_h, best_s = sh.copy(), ss
            continue
        try:
            ys, cs_ = search(sh, n)
            if cs_ < best_s - 1e-12:
                best_h, best_s = ys.copy(), cs_
        except Exception:
            pass
    fh = project_q(best_h, n)
    if fh is None:
        fh = uni
    fs = score_c5(fh, n)
    if not np.isfinite(fs):
        fh = pf if pf is not None else uni
        fs = score_c5(fh, n)
    if pf is not None:
        sp = score_c5(pf, n)
        if sp < fs - 1e-12:
            fh, fs = pf, sp
    return fh, fs, int(n)
