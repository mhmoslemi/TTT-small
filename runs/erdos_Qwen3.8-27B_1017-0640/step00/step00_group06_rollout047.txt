                best_delta = c

def _sweep(h, q_arr, score, pairs, n):
    dx = 2.0 / n
    for idx in pairs:
        p, q = idx
        try:
            delta = _line_minimize(h, p, q, q_arr, n)
            if abs(delta) < 1e-15:
                continue
            trial = h.copy()
            trial[p] += delta
            trial[q] -= delta
            # Clip tiny roundoff
            trial[p] = max(0.0, min(1.0, trial[p]))
            trial[q] = max(0.0, min(1.0, trial[q]))
            # Repair mass
            r = n / 2.0 - math.fsum(trial.tolist())
            if abs(r) > 1e-13:
                for _ in range(3):
                    if abs(r) < 1e-15:
                        break
                    if r > 0:
                        for i in range(n):
                            if r <= 0:
                                break
                            cap = min(r, 1.0 - trial[i])
                            trial[i] += cap
                            r -= cap
                    else:
                        for i in range(n):
                            if r >= 0:
                                break
                            cap = min(-r, trial[i])
                            trial[i] -= cap
                            r += cap
            if not _feasible(trial, n):
                continue
            q_t = np.correlate(trial, 1.0 - trial, mode="full")
            if not np.all(np.isfinite(q_t)):
                continue
            s_t = float(np.max(q_t) * dx)
            if s_t < score - 1e-12:
                h = trial
                q_arr = q_t
                score = s_t
        except Exception:
            continue
    return h, q_arr, score

def _exact_score(h, n):
    dx = 2.0 / n
    q = np.correlate(h, 1.0 - h, mode="full")
    if not np.all(np.isfinite(q)):
        return float('inf')
    return float(np.max(q) * dx)

def _feasible(h, n):
    if h.ndim != 1 or h.shape[0] != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return False
    if abs(math.fsum(h.tolist()) - n / 2.0) > 1e-10:
        return False
    return True

def _project(v, n, target):

        return None

            break

            break

    r = target - math.fsum(x.tolist())

def _compute_B(h, p, q, n):

    gq = np.zeros(total)

    qk2 = q + ks
    m4 = (qk2 >= 0) & (qk2 < n)
    gq[m4] -= h[qk2[m4]]

def _sweep(h, q_arr, score, pairs, n, rng):

    perm = rng.permutation(len(pairs))

            tr[p] += delta
            tr[q] -= delta
            tr[p] = max(0.0, min(1.0, tr[p]))
            tr[q] = max(0.0, min(1.0, tr[q]))
            r = n / 2.0 - math.fsum(tr.tolist())
            for _ in range(3):
                if abs(r) < 1e-15:
                    break

                            break

                        r -= c
