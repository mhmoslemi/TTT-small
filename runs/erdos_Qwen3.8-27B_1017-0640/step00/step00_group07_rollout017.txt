def project_capped(y, n):
    m = n * 0.5
    y = np.asarray(y, dtype=np.float64)
    if y.shape != (n,) or not np.all(np.isfinite(y)) or np.max(np.abs(y)) > 1e12:
        return np.full(n, 0.5)
    lo, hi = y.min() - 1.0, y.max()
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(y - mid, 0, 1)) > m:
            lo = mid
        else:
            hi = mid
    p = np.clip(y - (lo + hi) * 0.5, 0, 1)
    resid = m - p.sum()
    if abs(resid) > 1e-11:
        mask = (p > 1e-12) & (p < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            p[mask] += resid / cnt
            p = np.clip(p, 0, 1)
    if not np.all(np.isfinite(p)) or p.min() < -1e-9 or p.max() > 1.0 + 1e-9:
        return np.full(n, 0.5)
    if abs(p.sum() - m) > 1e-9 * max(1, n):
        return np.full(n, 0.5)

def exact_score(h):
    n = len(h)
    if not np.all(np.isfinite(h)) or h.min() < -1e-9 or h.max() > 1.0 + 1e-9:
        return np.inf

        return np.inf
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lag_values(h):
    n = len(h)
    dx = 2.0 / n
    q = np.empty(2 * n - 1)
    q[n - 1] = dx * np.dot(h, 1.0 - h)
    for s in range(1, n):
        q[n - 1 + s] = dx * np.dot(h[:n - s], 1.0 - h[s:])
        q[n - 1 - s] = dx * np.dot(h[s:], 1.0 - h[:n - s])
    return q

def smooth_obj_grad(h, tau):
    q = lag_values(h)
    n = len(h)
    dx = 2.0 / n
    qmax = q.max()
    exps = np.exp((q - qmax) / tau)
    s = exps.sum()
    w = exps / s
    phi = qmax + tau * np.log(s)
    grad = w[n - 1] * dx * (1.0 - 2.0 * h)
    for si in range(1, n):
        wa = w[n - 1 + si]
        wb = w[n - 1 - si]
        if wa < 1e-15 and wb < 1e-15:
            continue
        ns = n - si
        grad[:ns] += dx * (wa * (1.0 - h[si:]) - wb * h[si:])
        grad[si:] += dx * (-wa * h[:ns] + wb * (1.0 - h[:ns]))
    return phi, grad

def smooth_obj(h, tau):
    q = lag_values(h)
    qmax = q.max()
    exps = np.exp((q - qmax) / tau)
    s = exps.sum()
    return qmax + tau * np.log(s)

                break
