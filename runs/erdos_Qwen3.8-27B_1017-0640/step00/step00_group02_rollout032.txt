def exact_score(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(c))


def build_lag_pairs(n):
    pairs = []
    for tau in range(-(n - 1), n):
        u_list, v_list = [], []
        for i in range(n):
            u = i + tau
            v = i
            if 0 <= u < n and 0 <= v < n:
                u_list.append(u)
                v_list.append(v)
        pairs.append((np.array(u_list, dtype=np.int64), np.array(v_list, dtype=np.int64)))
    return pairs


def mm_optimize(x0, n, S, dx):
    pairs = build_lag_pairs(n)
    current = x0.copy()
    cur_score = exact_score(current, n)
    max_iter = 12 if n <= 64 else (8 if n <= 128 else 4)
    try:
        import cvxpy as cp
    except Exception:
        return current, cur_score
    for it in range(max_iter):
        a_list, b_list = [], []
        for u_idx, v_idx in pairs:
            a = np.zeros(n)
            b = 0.0
            for j in range(len(u_idx)):
                ui, vi = int(u_idx[j]), int(v_idx[j])
                z = current[ui] + current[vi]
                a[ui] += 1.0 - z / 2.0
                a[vi] += -z / 2.0
                b += z * z / 4.0
            a_list.append(a)
            b_list.append(b)
        x_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [x_var >= 0, x_var <= 1, cp.sum(x_var) == S, t_var >= 0]
        for a_k, b_k, (u_idx, v_idx) in zip(a_list, b_list, pairs):
            quad = cp.sum_squares(x_var[u_idx] - x_var[v_idx]) / 4.0
            expr = quad + a_k @ x_var + b_k
            constraints.append(dx * expr <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        x_var.value = current
        t_var.value = cur_score
        accepted = False
        for solver_name in [cp.ECOS, cp.CVXOPT]:
            if accepted:
                break
            try:
                prob.solve(solver=solver_name, max_iters=200, abstol=1e-8, reltol=1e-8, feastol=1e-8)
            except Exception:
                try:
                    prob.solve(solver=solver_name)
                except Exception:
                    continue
            if prob.status in ('optimal', 'optimal_inaccurate') and x_var.value is not None:
                cand = np.array(x_var.value, dtype=np.float64).ravel()
                if cand.shape == (n,) and np.all(np.isfinite(cand)):
                    proj = project_capped(cand, S)
                    if proj is not None and feasible(proj, n):
                        sc = exact_score(proj, n)
                        if sc < cur_score - 1e-10:
                            current = proj
                            cur_score = sc
                            accepted = True
        if not accepted:
            break
    return current, cur_score


def run(seed=42, **kwargs):
    n, S, dx = 45, 22.5, 2.0 / 45
    fallback = np.full(n, 0.5)
    fb_score = exact_score(fallback, n)
    ihv = globals().get('initial_h_values', None)
    if ihv is not None:
        try:
            arr = np.array(ihv, dtype=np.float64).ravel()
            if len(arr) >= 2:
                n, S, dx = len(arr), len(arr) / 2.0, 2.0 / len(arr)
                if np.all(np.isfinite(arr)):
                    if feasible(arr, n):
                        fallback = arr.copy()
                    else:
                        proj = project_capped(arr, S)
                        if proj is not None and feasible(proj, n):
                            fallback = proj
                        else:
                            fallback = np.full(n, 0.5)
                else:
                    fallback = np.full(n, 0.5)
                fb_score = exact_score(fallback, n)
        except Exception:
            n, S, dx = 45, 22.5, 2.0 / 45
            fallback = np.full(n, 0.5)
            fb_score = exact_score(fallback, n)
    if n > 256:
        return fallback.copy(), fb_score, int(n)
    rng = np.random.default_rng(seed)
    starts = [fallback.copy(), np.full(n, 0.5)]
    starts.append(fallback + 0.08 * rng.standard_normal(n))
    starts.append(fallback + 0.22 * rng.standard_normal(n))
    starts.append(rng.uniform(0.0, 1.0, n))
    phase = float(rng.uniform(0, 2 * math.pi))
    starts.append(0.5 + 0.35 * np.sin(2 * math.pi * np.arange(n) / n + phase))
    dedup = []
    for s in starts:
        proj = project_capped(s, S)
        if proj is None or not feasible(proj, n):
            continue
        dup = False
        for d in dedup:
            if float(np.max(np.abs(proj - d))) < 1e-12:
                dup = True
                break
        if not dup:
            dedup.append(proj)
    best = fallback.copy()
    best_score = fb_score
    for s in dedup:
        x_opt, sc = mm_optimize(s, n, S, dx)
        if sc < best_score - 1e-12:
            best = x_opt
            best_score = sc
    for std in [0.04, 0.015]:
        noisy = best + std * rng.standard_normal(n)
        proj = project_capped(noisy, S)
        if proj is not None and feasible(proj, n):
            x_opt, sc = mm_optimize(proj, n, S, dx)
            if sc < best_score - 1e-12:
                best = x_opt
                best_score = sc
    if not feasible(best, n):
        proj = project_capped(best, S)
        if proj is not None and feasible(proj, n):
            best = proj
        else:
            best = fallback
    best = np.clip(best, 0.0, 1.0)
    best_score = exact_score(best, n)
    return best.astype(np.float64), float(best_score), int(n)
