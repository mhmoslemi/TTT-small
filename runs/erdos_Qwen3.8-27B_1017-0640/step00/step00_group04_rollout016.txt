    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(corr))
    s = float(np.sum(np.exp((corr - m) / tau)))
    return m + tau * np.log(s)

def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    L = 2 * n - 1
    m = float(np.max(corr))
    exps = np.exp((corr - m) / tau)
    s = float(np.sum(exps))
    w = exps / s
    grad = np.zeros(n)
    for idx in range(L):
        wi = w[idx]
        if abs(wi) < 1e-15:
            continue
        ell = idx - (n - 1)
        if ell == 0:
            grad += dx * wi * (1.0 - 2.0 * h)
        elif ell > 0:
            grad[ell:n] += dx * wi * (1.0 - h[0:n - ell])
            grad[0:n - ell] -= dx * wi * h[ell:n]
        else:
            k = -ell
            grad[0:n - k] += dx * wi * (1.0 - h[k:n])
            grad[k:n] -= dx * wi * h[0:n - k]
    return grad

def try_add_start(cand, n, starts, seen):
    if len(starts) >= 8:
        return
    p = project_capped(cand, n)
    if p is None or not is_feasible(p):
        return
    for s in seen:
        if float(np.max(np.abs(p - s))) < 1e-9:
            return
    starts.append(p)
    seen.append(p.copy())

def build_starts(n, anchors, rng, n_starts):
    starts = []
    seen = []
    for a in anchors:
        if a is not None and len(a) == n:
            try_add_start(a, n, starts, seen)
    for a in anchors:
        if a is not None and len(a) == n:
            try_add_start(a + 0.15 * rng.standard_normal(n), n, starts, seen)
            try_add_start(a[::-1].copy(), n, starts, seen)
            try_add_start(1.0 - a, n, starts, seen)
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    try_add_start(blk, n, starts, seen)
    try_add_start(0.5 * np.ones(n), n, starts, seen)
    ii = np.arange(n, dtype=float)
    try_add_start(0.5 + 0.25 * np.cos(2.0 * np.pi * ii / n), n, starts, seen)
    a0 = anchors[0] if anchors else None
    for draw in range(4 * n_starts + 8):
        if len(starts) >= n_starts:
            break
        if a0 is not None and len(a0) == n:
            sc = 0.05 + 0.10 * (draw % 4)
            try_add_start(a0 + sc * rng.standard_normal(n), n, starts, seen)
        if len(starts) >= n_starts:
            break
        try_add_start(rng.uniform(-0.25, 1.25, n), n, starts, seen)
    if not starts:
        starts = [np.zeros(n)]
        starts[0][:n // 2] = 1.0
        starts[0] = project_capped(starts[0], n) if starts[0] is not None else starts[0]
    return starts

def pgd_search(h0, n, max_iter):
    h = h0.copy()
    best_h = h0.copy()
    best_score = exact_score(h0)
    taus = [0.02, 0.005, 0.00125, 0.0003125, 0.0001]
    for tau in taus:
        alpha = 1.0
        F_curr = smooth_val(h, tau)
        g = smooth_grad(h, tau)
        if not np.all(np.isfinite(g)):
            break
        max_d = 0.0
        for it in range(max_iter):
            accepted = False
            d = np.zeros(n)
            for bt in range(24):
                y = project_capped(h - alpha * g, n)
                if y is None:
                    alpha *= 0.5
                    continue
                d = y - h
                directional = float(np.dot(g, d))
                if not np.isfinite(directional) or directional >= 0.0:
                    alpha *= 0.5
                    continue
                F_y = smooth_val(y, tau)
                if not np.isfinite(F_y):
                    alpha *= 0.5
                    continue
                if F_y <= F_curr + 1e-4 * directional + 1e-14 * max(1.0, abs(F_curr)):
                    h = y
                    F_curr = F_y
                    alpha = min(1.2 * alpha, 10.0)
                    accepted = True
                    sc = exact_score(h)
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_h = h.copy()
                    max_d = float(np.max(np.abs(d)))
                    break
                alpha *= 0.5
            if not accepted:
                break
            if max_d <= 1e-8:
                break
            g = smooth_grad(h, tau)
            if not np.all(np.isfinite(g)):
                break
    return best_h, best_score

def run(seed=42, **kwargs):
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    rng = np.random.default_rng(seed)
    n_starts = 6
    max_iter_ps = 120
    use_doubled = True
    try:
        n_starts = max(3, min(8, int(kwargs.get('n_starts', 6))))
    except Exception:
        n_starts = 6
    try:
        max_iter_ps = max(40, min(180, int(kwargs.get('max_iter_per_stage', 120))))
    except Exception:
        max_iter_ps = 120
    try:
        use_doubled = bool(kwargs.get('use_doubled', True))
    except Exception:
        use_doubled = True

    parent = None
    n0 = 42
    try:
        p = np.array(initial_h_values, dtype=float).ravel()
        if p.size >= 2 and p.size <= 1024 and np.all(np.isfinite(p)) and float(np.max(np.abs(p))) < 1e6:
            n0 = p.size
            if is_feasible(p):
                parent = p.copy()
            else:
                proj = project_capped(p, n0)
                if proj is not None and is_feasible(proj):
                    parent = proj
    except Exception:
        parent = None

    if parent is None:
        n0 = 42
        fb = np.zeros(n0)
        fb[:n0 // 2] = 1.0
        if n0 % 2 == 1:
            fb[n0 // 2] = 0.5
        fb = project_capped(fb, n0)
        if fb is None:
            fb = np.zeros(n0)
            fb[:n0 // 2] = 1.0
        fallback = fb
        anchors0 = [fb]
    else:
        fallback = parent.copy()
        anchors0 = [parent, fallback]

    fb_score = exact_score(fallback)
    gb_h = fallback.copy()
    gb_score = fb_score

    if n0 <= 128:
        sts = build_starts(n0, anchors0, rng, n_starts)
        for s in sts:
            bh, bs = pgd_search(s, n0, max_iter_ps)
            if bs < gb_score - 1e-12:
                gb_score = bs
                gb_h = bh.copy()

    if use_doubled and n0 <= 64:
        n1 = 2 * n0
        anchors1 = []
        if parent is not None:
            anchors1.append(np.repeat(parent, 2))
        anchors1.append(np.repeat(gb_h, 2))
        sts1 = build_starts(n1, anchors1, rng, n_starts)
        for s in sts1:
            bh, bs = pgd_search(s, n1, max_iter_ps)
            if bs < gb_score - 1e-12:
                gb_score = bs
                gb_h = bh.copy()

    sel = gb_h.copy()
    nn = len(sel)
    if abs(float(np.sum(sel)) - 0.5 * nn) > 1e-12 * max(1, nn):
        proj = project_capped(sel, nn)
        if proj is not None and is_feasible(proj):
            sel = proj
    if not is_feasible(sel):
        sel = fallback
    fs = exact_score(sel)
    return sel, fs, len(sel)
