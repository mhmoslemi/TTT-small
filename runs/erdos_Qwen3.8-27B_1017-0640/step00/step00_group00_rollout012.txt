        ph = rng.random() * 2.0 * math.pi
        cands.append(0.5 + 0.35 * np.cos(2.0 * math.pi * r * i / n + ph))
    if has_parent:
        cands.append(h + 0.05 * rng.standard_normal(n))
        cands.append(h + 0.15 * rng.standard_normal(n))
    proj = []
    for c in cands:
        p = capped_simplex_project(c, m)
        if p is None:
            continue
        dup = False
        for q in proj:
            if float(np.max(np.abs(p - q))) <= 1e-10:
                dup = True
                break
        if not dup:
            proj.append(p)
    return proj

def run(seed=42, **kwargs):
    n = None
    if 'n_points' in kwargs:
        v = kwargs['n_points']
        if not isinstance(v, bool):
            try:
                iv = int(v)
                if iv == float(v) and 1 <= iv <= 999:
                    n = iv
            except (TypeError, ValueError):
                pass
    parent = None
    try:
        parent = np.asarray(initial_h_values, dtype=np.float64)
    except (NameError, TypeError, ValueError):
        parent = None
    if parent is not None:
        if parent.ndim != 1 or len(parent) < 1 or len(parent) > 999 or not np.all(np.isfinite(parent)):
            parent = None
    if parent is not None and n is None:
        n = len(parent)
    if n is None:
        n = 45
    m = n / 2.0
    dx = 2.0 / n
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    if parent is not None:
        if len(parent) != n:
            parent = interp(parent, n)
        p = capped_simplex_project(parent, m)
        if p is None:
            p = np.full(n, m / n)
    else:
        p = np.full(n, m / n)
    best_h = p.copy()
    best_sc = exact_score(best_h)
    nstart = 10 if n <= 64 else (6 if n <= 160 else (4 if n <= 400 else 2))
    starts = make_starts(p, n, m, rng, parent is not None)[:nstart]
    for s in starts:
        sh, sc = local_opt(s, n, m, dx)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_h = sh.copy()
    if n <= 64:
        rzs = [0.05, 0.18]
        nr = 2
    elif n <= 160:
        rzs = [0.08, 0.20]
        nr = 1
    else:
        rzs = [0.10]
        nr = 1
    for rr in range(nr):
        sb = best_sc
        for sg in rzs:
            pt = best_h + sg * rng.standard_normal(n)
            pf = capped_simplex_project(pt, m)
            if pf is None:
                continue
            sh, sc = local_opt(pf, n, m, dx)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_h = sh.copy()
        if best_sc > sb - 1e-9:
            break
    if not is_feasible(best_h, m):
        fp = capped_simplex_project(best_h, m)
        best_h = fp if fp is not None else np.full(n, m / n)
    best_sc = exact_score(best_h)
    return best_h, float(best_sc), int(n)
