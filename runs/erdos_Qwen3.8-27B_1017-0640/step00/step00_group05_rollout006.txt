    n_lags = 2 * N - 1
    J = np.zeros((n_lags, N), dtype=np.float64)
    for k in range(-(N-1), N):

        # First term: r-k in [0,N) → r in [k, N+k), intersect [0,N)
        lo1 = max(0, k)
        hi1 = min(N, N + k)
        if hi1 > lo1:

        # Second term: r+k in [0,N) → r in [-k, N-k), intersect [0,N)

        hi2 = min(N, N - k)
        if hi2 > lo2:

        hi1 = min(n, n + k)
        if hi1 > lo1:
            J[ko, lo1:hi1] += dx * (1.0 - h[lo1 - k:hi1 - k])
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if hi2 > lo2:
            J[ko, lo2:hi2] -= dx * h[lo2 + k:hi2 + k]
    return J


def obj_fun(z):
    return z[-1]


def obj_grad(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g


def eq_fun(z, target_sum):
    return z[:-1].sum() - target_sum


def eq_grad(z, target_sum):
    g = np.zeros(len(z))
    g[:-1] = 1.0
    return g


def ineq_fun(z, n):
    h = z[:n]
    return z[n] - np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)


def ineq_jac(z, n):
    h = z[:n]
    J = q_grad_matrix(h, n)
    n_lags = 2 * n - 1
    Jac = np.zeros((n_lags, n + 1), dtype=np.float64)
    Jac[:, :n] = -J
    Jac[:, n] = 1.0
    return Jac


def slsqp_one_start(h0, n, target_sum):
    q0 = q_all(h0, n)
    t0 = float(np.max(q0))
    t0 = float(np.clip(t0, 0.0, 2.0))
    z0 = np.concatenate([h0, [t0]])
    constraints = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_grad, 'args': (target_sum,)},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac, 'args': (n,)}
    ]
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    res = minimize(obj_fun, z0, method='SLSQP', jac=obj_grad,
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': 250, 'ftol': 1e-10, 'disp': False})
    if res.x is not None and res.x.shape[0] == n + 1:
        h_sol = project_box_sum(res.x[:n], target_sum)
        if np.all(np.isfinite(h_sol)):
            return h_sol
    return None


def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    n = 70
    parent = None
    try:
        pv = np.array(initial_h_values, dtype=np.float64).ravel()
        if pv.size > 0 and np.all(np.isfinite(pv)):
            parent = pv.copy()
            n = pv.size
    except (NameError, AttributeError, Exception):
        pass

    if n <= 0:
        n = 70
    target_sum = n / 2.0
    dx = 2.0 / n

    h_uniform = np.full(n, 0.5)

    if parent is not None and parent.size == n:
        h_par = project_box_sum(parent, target_sum)
    else:
        h_par = h_uniform.copy()

    starts = [h_par.copy()]

    if n <= 256:
        for scale in [0.03, 0.08, 0.18]:
            pert = h_par + scale * rng.standard_normal(n)
            starts.append(project_box_sum(pert, target_sum))
        starts.append(project_box_sum(rng.uniform(0.0, 1.0, n), target_sum))
        starts.append(project_box_sum(rng.beta(0.7, 0.7, n), target_sum))
        idx = np.arange(n)
        for p_val, phi in [(1, 0.0), (2, np.pi / 3.0)]:
            s = 0.5 + 0.45 * np.cos(2.0 * np.pi * p_val * (idx + 0.5) / n + phi)
            starts.append(project_box_sum(s, target_sum))
    elif n <= 999:
        starts.append(project_box_sum(h_par + 0.08 * rng.standard_normal(n), target_sum))
        starts.append(project_box_sum(rng.uniform(0.0, 1.0, n), target_sum))
        starts.append(project_box_sum(rng.beta(0.7, 0.7, n), target_sum))

    best_h = None
    best_c5 = np.inf
    for s in starts:
        s = project_box_sum(s, target_sum)
        c = exact_score(s, n)
        if np.isfinite(c) and c < best_c5:
            best_c5 = c
            best_h = s.copy()

    if n <= 999 and best_h is not None:
        for s in starts:
            try:
                h_sol = slsqp_one_start(s, n, target_sum)
                if h_sol is not None:
                    c = exact_score(h_sol, n)
                    if np.isfinite(c) and c < best_c5 - 1e-12:
                        best_c5 = c
                        best_h = h_sol.copy()
            except Exception:
                continue

    if best_h is None or not np.all(np.isfinite(best_h)):
        best_h = h_uniform
        best_c5 = exact_score(best_h, n)
    else:
        best_h = project_box_sum(best_h, target_sum)
        best_c5 = exact_score(best_h, n)

    if (not (best_h.min() >= -1e-12) or best_h.max() > 1.0 + 1e-12
            or abs(best_h.sum() - target_sum) > 1e-8):
        best_h = h_uniform
        best_c5 = exact_score(best_h, n)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = exact_score(best_h, n)

    return (best_h, best_c5, n)
