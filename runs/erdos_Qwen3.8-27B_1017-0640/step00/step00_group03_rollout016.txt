import numpy as np

def is_close_to_any(v, lst):
    for u in lst:
        if np.max(np.abs(v - u)) < 1e-10:
            return True
    return False

def project_capped_simplex(v, n):
    if v.shape != (n,) or not np.all(np.isfinite(v)):
        return np.full(n, 0.5)
    v = np.clip(v, -1e6, 1e6)
    target = n / 2.0
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = lo + (hi - lo) / 2.0
        q = np.clip(v - mid, 0.0, 1.0)
        if np.sum(q) > target:
            lo = mid
        else:
            hi = mid
    q = np.clip(v - (lo + (hi - lo) / 2.0), 0.0, 1.0)
    r = target - np.sum(q)
    if abs(r) > 1e-13:
        for i in range(n):
            if abs(r) <= 1e-13:
                break
            if r > 0:
                add = min(r, 1.0 - q[i])
                q[i] += add
                r -= add
            else:
                sub = min(-r, q[i])
                q[i] -= sub
                r += sub
    if not np.all(np.isfinite(q)) or np.any(q < -1e-12) or np.any(q > 1.0 + 1e-12):
        return np.full(n, 0.5)
    q = np.clip(q, 0.0, 1.0)
    if abs(np.sum(q) - target) > 1e-10:
        return np.full(n, 0.5)
    return q

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return np.inf
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return np.inf
    if abs(np.sum(h) - n / 2.0) > 1e-8:
        return np.inf
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    if c.shape != (2 * n - 1,) or not np.all(np.isfinite(c)):
        return np.inf
    return float(np.max(c))

def lag_values(h, n, need_j):
    dx = 2.0 / n
    m = 2 * n - 1
    c = np.zeros(m)
    J = np.zeros((m, n)) if need_j else None
    for d in range(1, n):
        u = h[:n - d]
        v = h[d:]
        c[n - 1 - d] = np.sum(u * (1.0 - v)) * dx
        c[n - 1 + d] = np.sum(v * (1.0 - u)) * dx
        if need_j:
            gB = np.zeros(n)
            gB[:n - d] += (1.0 - v)
            gB[d:] -= u
            J[n - 1 - d] = gB * dx
            gA = np.zeros(n)
            gA[d:] += (1.0 - u)
            gA[:n - d] -= v
            J[n - 1 + d] = gA * dx
    c[n - 1] = np.sum(h * (1.0 - h)) * dx
    if need_j:
        J[n - 1] = (1.0 - 2.0 * h) * dx
    return c, J

def surrogate_eval(c, tau, J):
    cmax = float(np.max(c))
    q = np.exp((c - cmax) / tau)
    s = float(np.sum(q))
    F = cmax + tau * np.log(s)
    if not np.isfinite(F):
        return np.inf, None
    p = q / s
    if J is not None:
        g = J.T @ p
        if not np.all(np.isfinite(g)):
            return np.inf, None
        return float(F), g
    return float(F), None

def pgc_search(start_h, inc_h, inc_score, n, rng, max_iter):
    h = start_h.copy()
    taus = [0.02, 0.005, 0.001, 0.0002, 0.00004, 0.000008]
    for tau in taus:
        alpha = 0.5
        stall = 0
        F_prev = np.inf
        for _ in range(max_iter):
            c, J = lag_values(h, n, True)
            F, g = surrogate_eval(c, tau, J)
            if not np.isfinite(F) or g is None:
                return h, inc_h, inc_score
            a = alpha
            accepted = False
            F_t = np.inf
            for _ in range(24):
                h_t = project_capped_simplex(h - a * g, n)
                c_t, _ = lag_values(h_t, n, False)
                F_t, _ = surrogate_eval(c_t, tau, None)
                if not np.isfinite(F_t):
                    a *= 0.5
                    continue
                delta = h_t - h
                bound = F + 1e-4 * float(g @ delta) + 1e-12 * max(1.0, abs(F))
                if F_t <= bound:
                    h = h_t
                    alpha = min(1.5 * a, 2.0)
                    accepted = True
                    break
                a *= 0.5
            if not accepted:
                break
            s = exact_score(h, n)
            if np.isfinite(s) and s < inc_score - 1e-12:
                inc_score = s
                inc_h = h.copy()
            if F_t < F_prev:
                stall = 0
            else:
                stall += 1
            F_prev = F_t
            if stall >= 8:
                break
    return h, inc_h, inc_score

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    n = 70
    parent = None
    ihv = globals().get('initial_h_values', None)
    if ihv is not None:
        try:
            arr = np.array(ihv, dtype=np.float64).flatten()
            if arr.shape[0] >= 2:
                n = arr.shape[0]
                if np.all(np.isfinite(arr)):
                    if np.all(arr >= 0.0) and np.all(arr <= 1.0) and abs(np.sum(arr) - n / 2.0) <= 1e-10:
                        parent = arr.copy()
                    else:
                        parent = project_capped_simplex(arr, n)
        except Exception:
            pass
    if parent is None:
        try:
            np_pts = int(kwargs.get('n_points', 70))
            if 2 <= np_pts <= 256:
                n = np_pts
        except Exception:
            pass
        parent = np.full(n, 0.5)
    parent = project_capped_simplex(parent, n)
    p_score = exact_score(parent, n)
    if not np.isfinite(p_score):
        parent = np.full(n, 0.5)
        p_score = exact_score(parent, n)
    inc_h = parent.copy()
    inc_score = p_score
    if n > 256:
        return inc_h, inc_score, n
    starts = [parent.copy(), parent[::-1].copy()]
    starts.append(project_capped_simplex(parent + 0.05 * rng.standard_normal(n), n))
    starts.append(project_capped_simplex(parent + 0.15 * rng.standard_normal(n), n))
    starts.append(np.full(n, 0.5))
    max_iter = 100 if n <= 128 else 40
    max_starts = 5 if n <= 128 else 4
    unique = []
    for s in starts:
        if not np.all(np.isfinite(s)):
            continue
        if not is_close_to_any(s, unique):
            unique.append(s)
    unique = unique[:max_starts]
    for s in unique:
        ss = exact_score(s, n)
        if np.isfinite(ss) and ss < inc_score - 1e-12:
            inc_score = ss
            inc_h = s.copy()
        _, inc_h, inc_score = pgc_search(s, inc_h, inc_score, n, rng, max_iter)
    if n <= 128:
        shake = project_capped_simplex(inc_h + 0.04 * rng.standard_normal(n), n)
        if not is_close_to_any(shake, unique):
            _, inc_h, inc_score = pgc_search(shake, inc_h, inc_score, n, rng, max_iter)
    inc_h = project_capped_simplex(inc_h, n)
    final_s = exact_score(inc_h, n)
    if not np.isfinite(final_s) or (np.isfinite(p_score) and p_score < final_s - 1e-12):
        inc_h = parent.copy()
        final_s = p_score
    if not np.isfinite(final_s):
        inc_h = np.full(n, 0.5)
        final_s = exact_score(inc_h, n)
    return inc_h, final_s, n
