import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog


def project_capped(v, m, N):
    v = np.asarray(v, dtype=np.float64)
    if v.shape != (N,) or not np.all(np.isfinite(v)):
        return None
    v = np.clip(v, -1e6, 1e6)
    lo = float(v.min()) - 1.0
    hi = float(v.max())
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(v - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    for _ in range(8):
        r = m - y.sum()
        if abs(r) < 1e-13:
            break
        if r > 0:
            idx = np.where(y < 1.0)[0]
            if len(idx) == 0:
                break
            y[idx] = np.minimum(1.0, y[idx] + r / len(idx))
        else:
            idx = np.where(y > 0.0)[0]
            if len(idx) == 0:
                break
            y[idx] = np.maximum(0.0, y[idx] + r / len(idx))
    y = np.clip(y, 0.0, 1.0)
    if not np.all(np.isfinite(y)) or abs(y.sum() - m) > 1e-10:
        return None
    return y


def exact_score(h, N):
    dx = 2.0 / N
    q = np.correlate(h, 1.0 - h, mode="full") * dx
    if len(q) != 2 * N - 1 or not np.all(np.isfinite(q)):
        return None
    return float(q.max())


def check_feasible(h, N, m):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return False
    if h.min() < -1e-10 or h.max() > 1.0 + 1e-10:
        return False
    return abs(h.sum() - m) <= 1e-10


def build_lag_grad(h, N):
    dx = 2.0 / N
    q = np.correlate(h, 1.0 - h, mode="full") * dx
    L = 2 * N - 1
    r = np.arange(N, dtype=np.float64)
    k = np.arange(-(N - 1), N, dtype=np.float64)
    d1 = r[None, :] - k[:, None]
    d2 = r[None, :] + k[:, None]
    v1 = (d1 >= 0) & (d1 < N)
    v2 = (d2 >= 0) & (d2 < N)
    i1 = np.clip(d1, 0, N - 1).astype(int)
    i2 = np.clip(d2, 0, N - 1).astype(int)
    g = (np.where(v1, 1.0 - h[i1], 0.0) - np.where(v2, h[i2], 0.0)) * dx
    if not np.all(np.isfinite(g)) or not np.all(np.isfinite(q)):
        return None, None
    return q, g


def solve_sublp(h, q, g, delta, N):
    L = 2 * N - 1
    nvar = 2 * N + 1
    c = np.zeros(nvar)
    c[-1] = 1.0
    Gs = sp.csr_matrix(g)
    Z_LN = sp.csr_matrix((L, N))
    nI = sp.csr_matrix((-np.ones(L), (np.arange(L), np.zeros(L, dtype=int))))
    A_lag = sp.hstack([Gs, Z_LN, nI], format='csr')
    I_N = sp.identity(N, format='csr')
    Z1 = sp.csr_matrix((N, 1))
    A_t1 = sp.hstack([I_N, -I_N, Z1], format='csr')
    A_t2 = sp.hstack([-I_N, -I_N, Z1], format='csr')
    b_t = np.zeros(2 * N)
    A_d = sp.csr_matrix((np.ones(N), (np.zeros(N, dtype=int), np.arange(N))), shape=(1, nvar))
    b_d = np.array([delta])
    A_ub = sp.vstack([A_lag, A_t1, A_t2, A_d], format='csr')
    b_ub = np.concatenate([-q, b_t, b_d])
    A_eq = sp.csr_matrix((np.ones(N), (np.zeros(N, dtype=int), np.arange(N))), shape=(1, nvar))
    bounds = [(max(-float(h[i]), -1e10), min(1.0 - float(h[i]), 1e10)) for i in range(N)]
    bounds += [(0.0, None)] * N
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs', options={'presolve': True})
    except Exception:
        return None
    if res is None or not res.success or res.x is None:
        return None
    if len(res.x) != nvar or not np.all(np.isfinite(res.x)):
        return None
    return res.x[:N]


def local_search(h0, N, m, max_iter, delta0):
    h = h0.copy()
    C_cur = exact_score(h, N)
    if C_cur is None:
        return h, C_cur
    delta = delta0
    fails = 0
    for _ in range(max_iter):
        q, G = build_lag_grad(h, N)
        if q is None:
            break
        d = solve_sublp(h, q, G, delta, N)
        if d is None:
            fails += 1
            delta *= 0.5
            if fails >= 4 or delta < 1e-4:
                break
            continue
        fails = 0
        h_proj = project_capped(h + d, m, N)
        if h_proj is None:
            delta *= 0.5
            if delta < 1e-4:
                break
            continue
        d_act = h_proj - h
        l1 = float(np.abs(d_act).sum())
        if l1 > delta * (1.0 + 1e-6):
            delta *= 0.5
            if delta < 1e-4:
                break
            continue
        C_trial = exact_score(h_proj, N)
        if C_trial is None:
            delta *= 0.5
            if delta < 1e-4:
                break
            continue
        C_model = float((q + G @ d_act).max())
        pred = C_cur - C_model
        act = C_cur - C_trial
        if act > 1e-12 * max(1.0, abs(C_cur)):
            h = h_proj
            C_cur = C_trial
        if pred <= 1e-12:
            delta *= 0.5
        else:
            rho = act / pred
            if rho < 0.1:
                delta *= 0.5
            elif rho > 0.75 and l1 >= 0.8 * delta:
                delta = min(2.0 * delta, 4.0)
        delta = max(delta, 1e-4)
    return h, C_cur
