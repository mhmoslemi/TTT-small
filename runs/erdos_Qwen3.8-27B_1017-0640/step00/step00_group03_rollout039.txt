    for _ in range(80):
        mid = lo + (hi - lo) * 0.5
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = lo + (hi - lo) * 0.5
    q = np.clip(v - lam, 0.0, 1.0)
    r = target - float(np.sum(q))
    for _ in range(2):
        if r > 0:
            for i in range(n):
                if r <= 0:
                    break
                a = min(1.0 - q[i], r)
                q[i] += a
                r -= a
        elif r < 0:
            for i in range(n):
                if r >= 0:
                    break
                a = min(q[i], -r)
                q[i] -= a
                r += a
    q = np.clip(q, 0.0, 1.0)
    if abs(float(np.sum(q)) - target) > 1e-10:
        return None
    if not np.all(np.isfinite(q)):
        return None
    return q


def score_exact(h, n):
    if not check_feasible(h, n):
        return float('inf')
    try:
        c = np.correlate(h, 1.0 - h, mode='full')
        if c.size != 2 * n - 1:
            return float('inf')
        if not np.all(np.isfinite(c)):
            return float('inf')
        return float(np.max(c)) * (2.0 / n)
    except Exception:
        return float('inf')


def try_add(pop, scores, n, hc):
    if hc is None:
        return float('inf')
    if not check_feasible(hc, n):
        proj = project_simplex(hc, n)
        if proj is None:
            return float('inf')
        hc = proj
    for e in pop:
        if float(np.max(np.abs(e - hc))) < 1e-10:
            return float('inf')
    s = score_exact(hc, n)
    if s >= float('inf'):
        return float('inf')
    pop.append(hc)
    scores.append(s)
    return s


def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)

    n = 70
    parent = None
    _iv = globals().get('initial_h_values', None)
    if _iv is not None:
        try:
            _a = np.asarray(_iv, dtype=np.float64)
            if _a.ndim == 1 and _a.size >= 2:
                n = int(_a.size)
                if np.all(np.isfinite(_a)):
                    parent = _a.copy()
        except Exception:
            pass

    uni = np.full(n, 0.5, dtype=np.float64)

    if n > 256:
        best_h, best_s = uni, score_exact(uni, n)
        if parent is not None and check_feasible(parent, n):
            ps = score_exact(parent, n)
            if ps < best_s:
                best_h, best_s = parent, ps
        return (best_h, best_s, n)

    P = 32 if n <= 128 else 24
    G = 80 if n <= 128 else 45

    fallback = None
    fb_score = score_exact(uni, n)
    if parent is not None:
        if check_feasible(parent, n):
            ps = score_exact(parent, n)
            if ps < fb_score:
                fallback, fb_score = parent.copy(), ps
        else:
            proj = project_simplex(parent, n)
            if proj is not None:
                ps = score_exact(proj, n)
                if ps < fb_score:
                    fallback, fb_score = proj, ps

    archive_h = fallback.copy() if fallback is not None else uni.copy()
    archive_score = score_exact(archive_h, n)

    pop, pscores = [], []
    try_add(pop, pscores, n, archive_h)

    if parent is not None:
        p = parent.astype(np.float64)
        try_add(pop, pscores, n, p[::-1].copy())
        try_add(pop, pscores, n, 1.0 - p)
        try_add(pop, pscores, n, np.roll(p, n // 4))
        try_add(pop, pscores, n, np.roll(p, n // 2))

    try_add(pop, pscores, n, uni)

    idx = np.arange(n, dtype=np.float64)
    for phi in (0.0, math.pi / 3.0, 2.0 * math.pi / 3.0):
        try_add(pop, pscores, n, 0.5 + 0.35 * np.sin(2.0 * math.pi * (idx + 0.5) / n + phi))
    try_add(pop, pscores, n, 0.5 + 0.2 * np.cos(4.0 * math.pi * (idx + 0.5) / n))

    center = parent.astype(np.float64) if parent is not None else uni
    sigmas = (0.025, 0.06, 0.12, 0.20)
    si = 0
    while len(pop) < P and si < 4 * P:
        try_add(pop, pscores, n, center + rng.standard_normal(n) * sigmas[si % 4])
        si += 1
    while len(pop) < P:
        try_add(pop, pscores, n, rng.random(n).astype(np.float64))
    while len(pop) < P:
        pop.append(archive_h.copy())
        pscores.append(archive_score)

    pop, pscores = pop[:P], pscores[:P]

    restarts = 0
    stagnation = 0

    for gen in range(G):
        snap = [p.copy() for p in pop]
        snap_s = list(pscores)
        cur_best = archive_score
        next_pop = list(snap)
        next_s = list(snap_s)

        if gen < G // 2:
            F = 0.55 + 0.35 * rng.random()
        else:
            F = 0.35 + 0.40 * rng.random()

        for i in range(P):
            CR = 0.65 + 0.35 * rng.random()
            others = [j for j in range(P) if j != i]

            if rng.random() < 0.75:
                ix = rng.choice(others, size=3, replace=False)
                donor = snap[ix[0]].astype(np.float64) + F * (snap[ix[1]].astype(np.float64) - snap[ix[2]].astype(np.float64))
            else:
                ix = rng.choice(others, size=2, replace=False)
                donor = snap[i].astype(np.float64) + F * (archive_h.astype(np.float64) - snap[i].astype(np.float64)) + F * (snap[ix[0]].astype(np.float64) - snap[ix[1]].astype(np.float64))

            trial = snap[i].copy()
            mask = rng.random(n) < CR
            j_rand = int(rng.integers(n))
            mask[j_rand] = True
            trial[mask] = donor[mask]

            proj = project_simplex(trial, n)
            if proj is None:
                continue
            ts = score_exact(proj, n)
            if ts >= float('inf'):
                continue

            if ts < archive_score:
                archive_h = proj.copy()
                archive_score = ts
            if ts < next_s[i]:
                next_pop[i] = proj
                next_s[i] = ts

        pop, pscores = next_pop, next_s

        if archive_score >= cur_best - 1e-15:
            stagnation += 1
        else:
            stagnation = 0

        if stagnation >= 18 and restarts < 2:
            sigma = 0.10 if restarts == 0 else 0.035
            pop2, ps2 = [], []
            try_add(pop2, ps2, n, archive_h)
            if pop:
                bi = int(np.argmin(pscores))
                try_add(pop2, ps2, n, pop[bi])
            remaining = P - len(pop2)
            n_noisy = int(remaining * 0.75)
            for k in range(remaining):
                if k < n_noisy:
                    nc = archive_h + rng.standard_normal(n) * sigma
                else:
                    nc = rng.random(n).astype(np.float64)
                try_add(pop2, ps2, n, nc)
            while len(pop2) < P:
                pop2.append(archive_h.copy())
                ps2.append(archive_score)
            pop, pscores = pop2[:P], ps2[:P]
            stagnation = 0
            restarts += 1

    cands = [(archive_h, archive_score)]
    if fallback is not None:
        cands.append((fallback, score_exact(fallback, n)))
    cands.append((uni, score_exact(uni, n)))

    best_h, best_s = None, float('inf')
    for ch, _cs in cands:
        if check_feasible(ch, n):
            rs = score_exact(ch, n)
            if rs < best_s and rs < float('inf'):
                best_h, best_s = ch, rs
        else:
            proj = project_simplex(ch, n)
            if proj is not None and check_feasible(proj, n):
                rs = score_exact(proj, n)
                if rs < best_s and rs < float('inf'):
                    best_h, best_s = proj, rs

    if best_h is None:
        best_h, best_s = uni, score_exact(uni, n)

    final_s = score_exact(best_h, n)
    return (best_h, final_s, n)
