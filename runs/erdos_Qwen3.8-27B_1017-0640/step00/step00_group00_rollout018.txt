import numpy as np
import math

def project_capped(y, m):
    n = len(y)
    if n == 0:
        return None
    y = np.asarray(y, dtype=np.float64)
    if not np.all(np.isfinite(y)):
        return None
    y = np.clip(y, -1e6, 1e6)
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(y - mid, 0.0, 1.0)) > m:
            lo = mid
        else:
            hi = mid
    res = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(12):
        r = m - res.sum()
        if abs(r) < 1e-13:
            break
        if r > 0:
            idx = np.where(1.0 - res > 1e-14)[0]
            if not len(idx):
                break
            amt = min(r, 0.5) / len(idx)
            res[idx] = np.minimum(res[idx] + amt, 1.0)
        else:
            idx = np.where(res > 1e-14)[0]
            if not len(idx):
                break
            amt = min(-r, 0.5) / len(idx)
            res[idx] = np.maximum(res[idx] - amt, 0.0)
    res = np.clip(res, 0.0, 1.0)
    if not np.all(np.isfinite(res)) or abs(res.sum() - m) > 1e-9:
        return np.full(n, 2.0 / n)
    return res

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    s = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))
    return s if np.isfinite(s) else float('inf')

def lse(q, tau):
    qm = float(np.max(q))
    e = np.exp((q - qm) / tau)
    s = float(np.sum(e))
    w = e / s
    return qm + tau * math.log(s), w

def wgrad(h, n, w, dx):
    g = dx * w[n - 1] * (1.0 - 2.0 * h)
    for p in range(1, n):
        wp, wm = w[n - 1 + p], w[n - 1 - p]
        if wp > 1e-15:
            g[p:] += dx * wp * (1.0 - h[:n - p])
            g[:n - p] -= dx * wp * h[p:]
        if wm > 1e-15:
            g[:n - p] += dx * wm * (1.0 - h[p:])
            g[p:] -= dx * wm * h[:n - p]
    return g

def interp_parent(par, n):
    xn = np.linspace(0.0, 1.0, len(par))
    xnew = np.linspace(0.0, 1.0, n)
    return np.interp(xnew, xn, par)

def make_starts(n, m, par, rng):
    starts = []
    ii = np.arange(n)
    const = np.full(n, m / n)
    blk = np.zeros(n)
    fi = int(m)
    blk[:fi] = 1.0
    if fi < n and m - fi > 1e-12:
        blk[fi] = m - fi
    if par is not None:
        starts.append(par)
        starts.append(1.0 - par)
        starts.append(par[::-1].copy())
        starts.append(np.roll(par, n // 3))
        starts.append(const)
        starts.append(blk)
        starts.append(blk[::-1].copy())
        starts.append(rng.standard_normal(n) * 0.3 + 0.5)
        ph = rng.uniform(0, 2 * math.pi)
        starts.append(0.5 + 0.35 * np.cos(2 * math.pi * ii / n + ph))
        starts.append(par + 0.05 * rng.standard_normal(n))
        starts.append(par + 0.15 * rng.standard_normal(n))
    else:
        starts.append(const)
        starts.append(blk)
        starts.append(blk[::-1].copy())

            ph = rng.uniform(0, 2 * math.pi)

    out = []
    for s in starts:

        if p is None:
            continue
        dup = False
        for o in out:
            if np.max(np.abs(p - o)) < 1e-10:
                dup = True
                break
        if not dup:
            out.append(p)
    return out

def local_opt(x, n, m, temps, budget):
    best_h = x.copy()
    best_s = exact_score(x, n)
    dx = 2.0 / n
    for tau in temps:
        alpha = 1.0
        for _ in range(budget):
            corr = np.correlate(x, 1.0 - x, mode='full') * dx
            if not np.all(np.isfinite(corr)):
                break
            Fv, w = lse(corr, tau)
            g = wgrad(x, n, w, dx)
            if not (np.isfinite(Fv) and np.all(np.isfinite(g))):
                break
            acc = False
            for _ in range(20):
                y = project_capped(x - alpha * g, m)
                if y is None:
                    alpha *= 0.5
                    continue
                d = y - x
                gd = float(np.dot(g, d))
                if not np.isfinite(gd) or gd >= 0:
                    alpha *= 0.5
                    continue
                cy = np.correlate(y, 1.0 - y, mode='full') * dx
                Fy, _ = lse(cy, tau)
                if not np.isfinite(Fy):
                    alpha *= 0.5
                    continue
                if Fy <= Fv + 1e-4 * gd + 1e-14:
                    acc = True
                    break
                alpha *= 0.5
            if not acc:
                break

            dec = Fv - Fy
            x = y

                break

                break
