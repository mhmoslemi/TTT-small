    starts = build_starts(base, n, S, rng)
    Nch = max(len(starts) + R, 1)
    M = max(1, min(Mcap, int(1.8e8 / (n * n * L * Nch))))
    inc, inc_sc = base.copy(), bs
    for st in starts:
        ss = exact_score(st, n, S)
        if ss is None:
            continue
        if ss < inc_sc - 1e-12:
            inc, inc_sc = st.copy(), ss
        try:
            inc, inc_sc = anneal_chain(st, ss, n, S, rng, L, M, pmm, psw, inc, inc_sc)
        except Exception:
            pass
    if R > 0 and n <= 64:
        for sd in [0.10, 0.25][:R]:
            pj = project_capped(inc + rng.normal(0, sd, n), S)
            if pj is None:
                continue
            ps = exact_score(pj, n, S)
            if ps is None:
                continue
            if ps < inc_sc - 1e-12:
                inc, inc_sc = pj.copy(), ps
            try:
                inc, inc_sc = anneal_chain(pj, ps, n, S, rng, L, M, pmm, psw, inc, inc_sc)
            except Exception:
                pass
    elif R > 0 and n <= 128:
        pj = project_capped(inc + rng.normal(0, 0.15, n), S)
        if pj is not None:
            ps = exact_score(pj, n, S)
            if ps is not None:
                if ps < inc_sc - 1e-12:
                    inc, inc_sc = pj.copy(), ps
                try:
                    inc, inc_sc = anneal_chain(pj, ps, n, S, rng, L, M, pmm, psw, inc, inc_sc)
                except Exception:
                    pass
    if not feasible(inc, n, S):
        rp = project_capped(inc, S)
        if rp is not None and feasible(rp, n, S):
            inc = rp
        else:
            inc = fallback
    fs = exact_score(inc, n, S)
    if fs is None or fs > fb_sc:
        inc, fs = fallback, fb_sc
    return inc.astype(np.float64), float(fs), int(n)
