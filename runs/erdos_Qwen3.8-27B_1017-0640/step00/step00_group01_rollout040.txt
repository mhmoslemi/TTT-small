def project_capped_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1 or len(v) < 2 or not np.isfinite(v).all() or np.max(np.abs(v)) > 1e6:
        return None
    n = len(v)
    s = n / 2.0
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = lo + (hi - lo) / 2.0
        m = math.fsum(np.clip(v - mid, 0.0, 1.0))
        if m > s:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(v - lam, 0.0, 1.0)
    for _ in range(4):
        r = s - math.fsum(x)
        if abs(r) <= 5e-15:
            break
        changed = False
        if r > 0:
            for i in range(n):
                if r <= 0:
                    break
                d = min(r, 1.0 - x[i])
                x[i] += d
                r -= d
                changed = True
        else:
            for i in range(n):
                if r >= 0:
                    break
                d = min(-r, x[i])
                x[i] -= d
                r += d
                changed = True
        if not changed:
            break
    if not np.isfinite(x).all() or np.any(x < 0.0) or np.any(x > 1.0):
        return None
    if abs(math.fsum(x) - s) > 1e-9:
        return None
    return x

def is_feasible(h, n):
    if not isinstance(h, np.ndarray) or h.ndim != 1 or len(h) != n:
        return False
    if not np.isfinite(h).all():
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(math.fsum(h) - n / 2.0) <= 1e-9

def exact_score(h, n):
    h = np.asarray(h, dtype=np.float64)
    if not np.isfinite(h).all():
        return None
    try:
        c = np.correlate(h, 1.0 - h, mode='full')
        val = float(np.max(c)) * (2.0 / n)
    except Exception:
        qmax = 0.0
        for d in range(-n + 1, n):
            qs = 0.0
            for i in range(max(0, -d), min(n, n - d)):
                qs += h[i] * (1.0 - h[i + d])
            qmax = max(qmax, qs)
        val = float(qmax) * (2.0 / n)
    if not np.isfinite(val):
        return None
    return val
