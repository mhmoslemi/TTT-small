            per = max(per, -np.min(p[idx]))
        p[idx] += per
        residual = m - np.sum(p)
    if not np.all(np.isfinite(p)) or np.any(p < -1e-14) or np.any(p > 1.0 + 1e-14):
        return np.full(n, 0.5)
    p = np.clip(p, 0.0, 1.0)
    if abs(np.sum(p) - m) > 1e-10 * max(1, n):
        return np.full(n, 0.5)
    return p

def check_feasible(h, n):
    if len(h) != n or not np.all(np.isfinite(h)):
        return False
    if np.any(h < 0.0) or np.any(h > 1.0):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def exact_score(h):
    n = len(h)
    if not check_feasible(h, n):
        return float('inf')
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def lag_values(h):
    n = len(h)
    dx = 2.0 / n
    q = np.empty(2 * n - 1)
    q[0] = dx * np.sum(h * (1.0 - h))
    for s in range(1, n):
        q[s] = dx * np.sum(h[:n - s] * (1.0 - h[s:]))
        q[2 * n - 1 - s] = dx * np.sum(h[s:] * (1.0 - h[:n - s]))
    return q

def lag_grad(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    G[0, :] = dx * (1.0 - 2.0 * h)
    for s in range(1, n):
        G[s, :n - s] += dx * (1.0 - h[s:])
        G[s, s:] -= dx * h[:n - s]
        G[2 * n - 1 - s, s:] += dx * (1.0 - h[:n - s])
        G[2 * n - 1 - s, :n - s] -= dx * h[s:]
    return G

def smooth_phi(q, tau):
    qmax = np.max(q)
    e = np.exp((q - qmax) / tau)
    return qmax + tau * np.log(np.sum(e))

def smooth_grad(h, tau):
    q = lag_values(h)
    qmax = np.max(q)
    e = np.exp((q - qmax) / tau)
    w = e / np.sum(e)
    G = lag_grad(h)
    return smooth_phi(q, tau), G.T @ w, q

def run(seed=42, **kwargs):
    n = 80
    base = None
    try:
        parent = globals().get('initial_h_values', None)
        if parent is not None:
            parent = np.asarray(parent, dtype=np.float64)
            if parent.ndim == 1 and 2 <= len(parent) <= 999:
                n = len(parent)
                base = parent.copy()
    except Exception:
        pass
    if base is None:
        try:
            nk = int(kwargs.get('n_points', 80))
            if 2 <= nk <= 999:
                n = nk
        except Exception:
            pass
        base = np.full(n, 0.5)
    n_starts = 7
    try:
        v = int(kwargs.get('n_starts', 7))
        if 1 <= v <= 8:
            n_starts = v
    except Exception:
        pass
    max_iter = 80
    try:
        v = int(kwargs.get('max_iter', 80))
        if 20 <= v <= 200:
            max_iter = v
    except Exception:
        pass
    try:
        s = int(seed)
        if s < 0:
            s = 42
    except Exception:
        s = 42
    rng = np.random.default_rng(s)
    x0 = project_capped(base, n)
    best_h = x0.copy()
    best_score = exact_score(x0)
    starts = [x0.copy()]
    for sd in [0.02, 0.08, 0.20]:
        starts.append(project_capped(x0 + rng.standard_normal(n) * sd, n))
    starts.append(project_capped(rng.uniform(0.0, 1.0, n), n))
    K = min(6, (n - 1) // 2)
    y = np.full(n, 0.5)
    t = np.arange(n, dtype=np.float64)
    for k in range(1, K + 1):
        a = rng.standard_normal() * 0.12 / np.sqrt(k)
        b = rng.standard_normal() * 0.12 / np.sqrt(k)
        y += a * np.cos(2.0 * np.pi * k * (t + 0.5) / n)
        y += b * np.sin(2.0 * np.pi * k * (t + 0.5) / n)
    starts.append(project_capped(y, n))
    nb = np.zeros(n)
    half = n // 2
    idx = rng.choice(n, size=half, replace=False)
    nb[idx] = 1.0
    if n % 2 == 1:
        rem = np.setdiff1d(np.arange(n), idx)
        if len(rem) > 0:
            nb[rem[0]] = 0.5
    starts.append(project_capped(nb, n))
    starts.append(project_capped(x0 + rng.standard_normal(n) * 0.4, n))
    starts = starts[:8]
    unique_starts = [starts[0]]
    for cand in starts[1:]:
        dup = False
        for u in unique_starts:
            if np.max(np.abs(cand - u)) < 1e-10:
                dup = True
                break
        if not dup:
            unique_starts.append(cand)
    unique_starts = unique_starts[:n_starts]
    temps = np.array([0.02, 0.005, 0.00125, 0.0003125, 7.8125e-05, 1.953125e-05, 4.8828125e-06])
    if n <= 256:
        for si in range(len(unique_starts)):
            x = unique_starts[si].copy()
            for tau in temps:
                alpha = 1.0
                for it in range(max_iter):
                    phi, grad, q_curr = smooth_grad(x, tau)
                    if not np.isfinite(phi) or not np.all(np.isfinite(grad)):
                        break
                    if np.max(np.abs(grad)) < 1e-14:
                        break
                    accepted = False
                    for attempt in range(25):
                        xtrial = project_capped(x - alpha * grad, n)
                        d = xtrial - x
                        if np.max(np.abs(d)) <= 1e-9:
                            break
                        gd = np.sum(grad * d)
                        if not gd < 0:
                            break
                        q_trial = lag_values(xtrial)
                        phi_trial = smooth_phi(q_trial, tau)
                        if phi_trial <= phi + 1e-4 * gd + 1e-14 * max(1.0, abs(phi)):
                            x = xtrial
                            alpha = min(alpha * 1.25, 4.0)
                            accepted = True
                            break
                        alpha *= 0.5
                    if not accepted:
                        break
                    if check_feasible(x, n):
                        sc = exact_score(x)
                        if sc < best_score - 1e-12:
                            best_score = sc
                            best_h = x.copy()
            if check_feasible(x, n):
                sc = exact_score(x)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_h = x.copy()
            if best_score <= 0.3808:
                break
    if not check_feasible(best_h, n):
        best_h = project_capped(base, n)
        best_score = exact_score(best_h)
    if not check_feasible(best_h, n):
        best_h = np.full(n, 0.5)
        best_score = exact_score(best_h)
    best_h = project_capped(best_h, n)
    best_score = exact_score(best_h)
    return (best_h, float(best_score), int(n))
