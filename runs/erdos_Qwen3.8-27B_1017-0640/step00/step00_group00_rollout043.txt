import numpy as np
import cvxpy as cp
from scipy.optimize import minimize

_mm_data = {}

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return None
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    if abs(float(np.sum(h)) - n / 2.0) > 1e-10:
        return None
    raw = np.correlate(h, 1.0 - h, mode='full')
    if raw.shape != (2 * n - 1,) or not np.all(np.isfinite(raw)):
        return None
    return float(np.max(raw) * (2.0 / n))

def project_simplex(y, n):
    if y.shape != (n,) or not np.all(np.isfinite(y)):
        return None
    y = np.clip(y, -1e6, 1e6)
    m = n / 2.0
    lo, hi = float(np.min(y)) - 1.0, float(np.max(y))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if float(np.sum(np.clip(y - mid, 0.0, 1.0))) > m:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(10):
        res = m - float(np.sum(x))
        if abs(res) < 1e-13:
            break
        if res > 0:
            for i in range(n):
                r = 1.0 - x[i]
                if r > 1e-13:
                    a = min(res, r); x[i] += a; res -= a
                    if abs(res) < 1e-13: break
        else:
            for i in range(n):
                r = x[i]
                if r > 1e-13:
                    a = min(-res, r); x[i] -= a; res += a
                    if abs(res) < 1e-13: break
    if not np.all(np.isfinite(x)) or np.any(x < -1e-10) or np.any(x > 1.0 + 1e-10):
        return None
    if abs(float(np.sum(x)) - m) > 1e-10:
        return None
    return x

def grad_matrix(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n), dtype=np.float64)
    G[n - 1, :] = dx * (1.0 - 2.0 * h)
    for p in range(1, n):
        G[n - 1 + p, p:] += dx * (1.0 - h[:-p])
        G[n - 1 + p, :n - p] -= dx * h[p:]
        G[n - 1 - p, :n - p] += dx * (1.0 - h[p:])
        G[n - 1 - p, p:] -= dx * h[:-p]
    return G

def _eq_fun(z):
    return float(np.sum(z[:_mm_data['n']]))

def _eq_jac(z):
    j = np.zeros(len(z)); j[:_mm_data['n']] = 1.0; return j

def _in_fun(z):
    n, q, G, dx = _mm_data['n'], _mm_data['q'], _mm_data['G'], _mm_data['dx']
    d, s, t = z[:n], z[n], z[-1]
    v = np.empty(1 + 2 * n - 1)
    v[0] = s - float(np.dot(d, d))
    v[1:] = t - q - G @ d - dx * s
    return v

def _in_jac(z):
    n, q, G, dx = _mm_data['n'], _mm_data['q'], _mm_data['G'], _mm_data['dx']
    L = 2 * n - 1
    jac = np.zeros((1 + L, len(z)))
    jac[0, :n] = -2.0 * z[:n]; jac[0, n] = 1.0
    jac[1:, :n] = -G; jac[1:, n] = -dx; jac[1:, -1] = 1.0
    return jac

def _obj_fun(z):
    return z[-1]

def _obj_jac(z):
    g = np.zeros(len(z)); g[-1] = 1.0; return g

def _solve_scipy(x, n):
    dx = 2.0 / n
    q = np.correlate(x, 1.0 - x, mode='full') * dx
    G = grad_matrix(x, n)
    _mm_data.update(n=n, q=q, G=G, dx=dx)
    z0 = np.zeros(n + 2); z0[n + 1] = float(np.max(q))
    bnds = [( -x[i], 1.0 - x[i]) for i in range(n)]
    bnds.append((0.0, float(n))); bnds.append((-np.inf, np.inf))
    maxit = 100 if n <= 64 else (60 if n <= 160 else (35 if n <= 400 else 20))
    try:
        res = minimize(_obj_fun, z0, jac=_obj_jac, bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac},
                                   {'type': 'ineq', 'fun': _in_fun, 'jac': _in_jac}],
                       method='SLSQP', options={'ftol': 1e-9, 'maxiter': maxit, 'disp': False})
    except Exception:
        return None
    d = res.x[:n]
    if not np.all(np.isfinite(d)):
        return None
    return d

def _solve_mm(x, n, cvxprob, dx):
    if cvxprob is not None:
        try:
            q = np.correlate(x, 1.0 - x, mode='full') * dx
            G = grad_matrix(x, n)
            cvxprob.update(q=q, G=G, lower=-x, upper=1.0 - x)
            cvxprob.solve(solver=cp.ECOS, abstol=1e-8, reltol=1e-8, feastol=1e-8, max_iters=200)
            if cvxprob.status not in ('optimal', 'optimal_inaccurate'):
                cvxprob.solve(solver=cp.CVXOPT, maxiter=200)
            if cvxprob.status in ('optimal', 'optimal_inaccurate'):
                d = np.array(cvxprob.d_var.value, dtype=np.float64)
                if d.shape == (n,) and np.all(np.isfinite(d)):
                    if not (np.any(d < -x - 1e-4) or np.any(d > 1.0 - x + 1e-4)):
                        if abs(float(np.sum(d))) < 1e-4:
                            sv = float(cvxprob.s_var.value)
                            if sv >= 0 and np.isfinite(sv) and float(np.dot(d, d)) <= sv + 1e-4:
                                return d
        except Exception:
            pass
    return _solve_scipy(x, n)

def run(seed=42, **kwargs):
    n = 45
    if 'n_points' in kwargs:
        v = kwargs['n_points']
        if not isinstance(v, bool) and isinstance(v, (int, np.integer)) and 1 <= int(v) <= 999:
            n = int(v)
    dx = 2.0 / n
    parent = None
    try:
        pv = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= pv.size <= 999 and np.all(np.isfinite(pv)):
            parent = pv
    except Exception:
        pass
    if parent is not None and parent.size != n:
        if parent.size == 1:
            parent = np.full(n, parent[0])
        else:
            parent = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
    pp = project_simplex(parent.copy(), n) if parent is not None else None
    if pp is None:
        pp = np.full(n, 0.5)
    cvxprob = None
    L = 2 * n - 1
    try:
        dv = cp.Variable(n); sv = cp.Variable(nonneg=True); tv = cp.Variable()
        qp = cp.Parameter(L); Gp = cp.Parameter((L, n)); lp = cp.Parameter(n); up = cp.Parameter(n)
        cvxprob = cp.Problem(cp.Minimize(tv),
            [dv >= lp, dv <= up, cp.sum(dv) == 0, cp.sum_squares(dv) <= sv,
             qp + Gp @ dv + dx * sv <= tv])
        cvxprob.update(q=qp, G=Gp, lower=lp, upper=up)
        cvxprob.d_var = dv; cvxprob.s_var = sv
    except Exception:
        cvxprob = None
    rng = np.random.default_rng(seed)
    m = n / 2.0
    blk = np.zeros(n); blk[:int(np.floor(m))] = 1.0
    if m - int(np.floor(m)) > 1e-12:
        blk[int(np.floor(m))] = m - int(np.floor(m))
    cands = [pp.copy(), 1.0 - pp.copy(), blk.copy(),
             project_simplex(rng.uniform(0, 1, n), n), blk[::-1].copy(),
             project_simplex(pp + 0.10 * rng.standard_normal(n), n)]
    starts = []
    for c in cands:
        if c is not None and c.shape == (n,) and np.all(np.isfinite(c)):
            if all(np.max(np.abs(c - v)) > 1e-10 for v in starts):
                starts.append(c)
    if n <= 64: ms, mi = 6, 40
    elif n <= 160: ms, mi = 4, 25
    elif n <= 400: ms, mi = 3, 12
    else: ms, mi = 2, 5
    starts = starts[:ms]
    best = pp.copy(); bs = exact_score(best, n)
    if bs is None:
        best = np.full(n, 0.5); bs = exact_score(best, n)
    for s in starts:
        sc = exact_score(s, n)
        if sc is not None and sc < bs - 1e-12:
            best, bs = s.copy(), sc
    def _mm_opt(x, cur_sc):
        for _ in range(mi):
            d = _solve_mm(x, n, cvxprob, dx)
            if d is None: break
            y = project_simplex(x + d, n)
            if y is None: break
            by, bsc = x.copy(), cur_sc
            for th in [1.0, 0.5, 0.25, 0.125, 0.0625]:
                cp2 = project_simplex(x + th * (y - x), n)
                if cp2 is None: continue
                sc2 = exact_score(cp2, n)
                if sc2 is not None and sc2 < bsc - 1e-12:
                    by, bsc = cp2.copy(), sc2
            if bsc < cur_sc - 1e-12:
                x, cur_sc = by, bsc
                if cur_sc > bs + 1e-12: break
            else: break
        return x, cur_sc
    for s in starts:
        cs = exact_score(s, n)
        if cs is None: continue
        x, cs = _mm_opt(s.copy(), cs)
        if cs < bs - 1e-12:
            best, bs = x.copy(), cs
    perts = [0.04, 0.15] if n <= 64 else ([0.08] if n <= 160 else ([0.12] if n <= 400 else [0.15]))
    obs = bs
    for std in perts:
        xp = project_simplex(best + std * rng.standard_normal(n), n)
        if xp is None: continue
        x, cs = _mm_opt(xp.copy(), exact_score(xp, n))
        if cs < bs - 1e-12:
            best, bs = x.copy(), cs
        if bs > obs - 1e-9: break
    final = project_simplex(best, n)
    if final is None:
        final = np.full(n, 0.5)
    c5 = exact_score(final, n)
    if c5 is None:
        final = np.full(n, 0.5); c5 = exact_score(final, n)
    return (final, c5, int(n))
