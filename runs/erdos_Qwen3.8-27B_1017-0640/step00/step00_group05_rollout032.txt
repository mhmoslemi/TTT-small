                            ss_y = smooth_stats(q_y, tau)
                            if ss_y is None:
                                eta *= 0.5
                                continue
                            F_y = ss_y[0]
                        except Exception:
                            eta *= 0.5
                            continue
                        if C_y < incumbent_score - 1e-12:
                            incumbent = trial.copy()
                            incumbent_score = C_y
                        cond = F_y <= F_tau + 1e-4 * a + 1e-14 * max(1.0, abs(F_tau))
                        if cond:
                            h = trial.copy()
                            accepted = True
                            if bt == 0:
                                eta = min(eta * 1.5, 8.0)
                            break
                        eta *= 0.5
                    if not accepted:
                        break
                    if np.max(np.abs(h - prev_h)) <= 1e-9:
                        break
            if is_feasible(h, N, m):
                sc = exact_score(h, N)
                if sc is not None and sc < incumbent_score - 1e-12:
                    incumbent = h.copy()
                    incumbent_score = sc

    candidates = [incumbent]
    if parent_fallback is not None and is_feasible(parent_fallback, N, m):
        candidates.append(parent_fallback)
    candidates.append(neutral)
    best = incumbent
    best_sc = incumbent_score
    for c in candidates:
        if not is_feasible(c, N, m):
            continue
        sc = exact_score(c, N)
        if sc is not None and sc < best_sc - 1e-12:
            best = c.copy()
            best_sc = sc
    fp = project_capped(best, N, m)
    if fp is not None and is_feasible(fp, N, m):
        best = fp
    final_sc = exact_score(best, N)
    if final_sc is None:
        best = neutral
        final_sc = exact_score(best, N)
    return (best.astype(np.float64), float(final_sc), int(N))
