    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(y - mid, 0.0, 1.0)) > m:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    r = m - np.sum(x)
    if abs(r) > 1e-14:
        for i in range(n):
            if r > 0:
                adj = min(1.0 - x[i], r)
                x[i] += adj
                r -= adj
            else:
                adj = min(x[i], -r)
                x[i] -= adj
                r += adj
            if abs(r) < 1e-14:
                break
    return x


def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-9


def score_exact(h, n):
    h = np.asarray(h, dtype=np.float64)
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return np.inf
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return np.inf
    if abs(np.sum(h) - n / 2.0) > 1e-9:
        return np.inf
    dx = 2.0 / n
    c = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
    return c if np.isfinite(c) else np.inf


def lag_values(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)


def lag_gradient(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    G[n - 1] = dx * (1.0 - 2.0 * h)
    for p in range(1, n):
        G[n - 1 + p, p:] += dx * (1.0 - h[:-p])
        G[n - 1 + p, :-p] -= dx * h[p:]
        G[n - 1 - p, :-p] += dx * (1.0 - h[p:])
        G[n - 1 - p, p:] -= dx * h[:-p]
    return G


def obj_func(z):
    return z[-1]


def obj_grad(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g


def eq_func(z):
    return np.sum(z[:-1]) - (len(z) - 1) / 2.0


def eq_grad(z):
    n = len(z) - 1
    g = np.ones(n + 1)
    g[-1] = 0.0
    return g


def ineq_func(z):
    n = len(z) - 1
    return z[-1] - lag_values(z[:-1], n)


def ineq_grad(z):
    n = len(z) - 1
    G = lag_gradient(z[:-1], n)
    J = np.zeros((2 * n - 1, n + 1))
    J[:, :-1] = -G
    J[:, -1] = 1.0
    return J


def make_block(n):
    m = n / 2.0
    x = np.zeros(n)
    k = int(m)
    x[:k] = 1.0
    if m - k > 0.5e-14:
        x[k] = m - k
    return x


def optimize_candidate(x0, n, maxiter, ftol):
    sc = score_exact(x0, n)
    t0 = min(1.0, sc + 1e-9) if np.isfinite(sc) else 0.5
    z0 = np.concatenate([x0, [t0]])
    bounds = [(0.0, 1.0)] * (n + 1)
    cons = [
        {'type': 'eq', 'fun': eq_func, 'jac': eq_grad},
        {'type': 'ineq', 'fun': ineq_func, 'jac': ineq_grad}
    ]
    try:
        res = minimize(obj_func, z0, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
    except Exception:
        return None
    if res is None or not hasattr(res, 'x') or res.x is None:
        return None
    hx = res.x[:n]
    if not np.all(np.isfinite(hx)):
        return None
    return project_feasible(hx, n)


def run(seed=42, **kwargs):
    n = 45
    if 'n_points' in kwargs:
        try:
            nn = int(kwargs['n_points'])
            if 1 <= nn <= 999:
                n = nn
        except (ValueError, TypeError):
            pass

    par = globals().get('initial_h_values', None)
    parent = None
    if par is not None:
        par = np.asarray(par, dtype=np.float64)
        if par.ndim == 1 and par.shape[0] > 0 and np.all(np.isfinite(par)):
            if par.shape[0] == n:
                parent = par.copy()
            else:
                xi = np.linspace(0, 1, par.shape[0])
                yi = np.linspace(0, 1, n)
                parent = np.interp(yi, xi, par)
                parent = project_feasible(parent, n)
    if parent is None:
        parent = np.full(n, 0.5)
    elif not is_feasible(parent, n):
        parent = project_feasible(parent, n)

    rng = np.random.default_rng(seed)

    starts = []
    starts.append(parent.copy())
    starts.append(1.0 - parent)
    starts.append(parent[::-1].copy())
    starts.append(np.roll(parent, n // 3))
    starts.append(np.full(n, 0.5))
    blk = make_block(n)
    starts.append(blk)
    starts.append(blk[::-1].copy())
    for _ in range(3):
        starts.append(project_feasible(rng.standard_normal(n), n))
    starts.append(project_feasible(parent + 0.05 * rng.standard_normal(n), n))
    starts.append(project_feasible(parent + 0.18 * rng.standard_normal(n), n))

    seen = []
    unique_starts = []
    for s in starts:
        dup = False
        for v in seen:
            if np.allclose(s, v, atol=1e-6):
                dup = True
                break
        if not dup:
            seen.append(s.copy())
            unique_starts.append(s)

    if n <= 100:
        max_starts, maxiter, n_rounds = 10, 250, 2
        ref_sigs = [0.03, 0.10, 0.25]
    elif n <= 300:
        max_starts, maxiter, n_rounds = 7, 120, 1
        ref_sigs = [0.05, 0.18]
    else:
        max_starts, maxiter, n_rounds = 4, 60, 1
        ref_sigs = [0.08]

    ftol = 1e-10
    unique_starts = unique_starts[:max_starts]

    best = unique_starts[0].copy()
    best_score = score_exact(best, n)
    if not np.isfinite(best_score):
        best = parent.copy()
        best_score = score_exact(best, n)

    for s in unique_starts:
        if not is_feasible(s, n):
            s = project_feasible(s, n)
        hx = optimize_candidate(s, n, maxiter, ftol)
        if hx is not None and is_feasible(hx, n):
            sc = score_exact(hx, n)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best = hx.copy()
                best_score = sc

    for rnd in range(n_rounds):
        round_start = best_score
        for sig in ref_sigs:
            pert = project_feasible(best + sig * rng.standard_normal(n), n)
            if not is_feasible(pert, n):
                continue
            hx = optimize_candidate(pert, n, maxiter, ftol)
            if hx is not None and is_feasible(hx, n):
                sc = score_exact(hx, n)
                if np.isfinite(sc) and sc < best_score - 1e-12:
                    best = hx.copy()
                    best_score = sc
        if best_score > round_start - 1e-9:
            break

    if not is_feasible(best, n):
        best = project_feasible(best, n)
    best_score = score_exact(best, n)
    if not np.isfinite(best_score):
        best = project_feasible(parent, n)
        best_score = score_exact(best, n)

    return (best, best_score, n)
