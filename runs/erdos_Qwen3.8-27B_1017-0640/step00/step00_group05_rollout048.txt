    for g in range(G):
        prog = g / max(G - 1, 1)
        F = 0.70 - 0.30 * prog
        CR = 0.90 - 0.30 * prog
        new_pop = [pop_arr[i].copy() for i in range(P)]
        new_sc = score_arr.copy()
        imp = False
        for i in range(P):
            others = [j for j in range(P) if j != i]
            r = rng.choice(others, size=3, replace=False)
            r1, r2, r3 = int(r[0]), int(r[1]), int(r[2])
            pm = _project(pop_arr[r1] + F * (pop_arr[r2] - pop_arr[r3]), N, m)
            if pm is None:
                continue
            cr = rng.random(N) < CR
            cr[int(rng.integers(0, N))] = True
            pt = _project(np.where(cr, pm, pop_arr[i]), N, m)
            if pt is None:
                continue
            st = _score(pt, N)
            if st is None:
                continue
            if st < new_sc[i]:
                new_pop[i] = pt
                new_sc[i] = st
            if st < incumbent_score:
                incumbent, incumbent_score = pt.copy(), st
                imp = True
        pop_arr = np.array(new_pop)
        score_arr = np.array(new_sc)
        stall = 0 if imp else stall + 1
        if stall >= stall_limit and restarts < 2 and g < G - 1:
            restarts += 1
            pop_arr, score_arr = _build_restart(incumbent, incumbent_score, rng, N, m, P, restarts, pop_arr, score_arr)
            stall = 0

    fi = _project(incumbent.copy(), N, m)
    fs = _score(fi, N) if fi is not None else None
    if fs is None or fs > incumbent_score:
        fi, fs = incumbent, incumbent_score
    if parent is not None:
        sp = _score(parent, N)
        if sp is not None and sp < fs:
            fi, fs = parent.copy(), sp
    sn = _score(neutral, N)
    if sn is not None and sn < fs:
        fi, fs = neutral.copy(), sn
    return (fi, fs, N)
