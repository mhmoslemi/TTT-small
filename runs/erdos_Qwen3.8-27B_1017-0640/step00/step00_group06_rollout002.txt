    grad = dx * (conv1[n1:n1 + n] - conv2[n1:n1 + n])
    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n)
    return val, grad

def run(seed=42, **kwargs):
    global _GLOBAL_M
    rng = np.random.default_rng(seed)
    n = 95
    parent = None
    try:
        ih = np.asarray(initial_h_values, dtype=np.float64)
        if ih.ndim == 1 and 2 <= len(ih) < 1000 and np.all(np.isfinite(ih)):
            n = len(ih)
            parent = ih.copy()
    except Exception:
        pass
    if parent is None:
        npk = kwargs.get('n_points', None)
        if npk is not None:
            try:
                n = int(npk)
            except (ValueError, TypeError):
                n = 95
            if n < 2 or n >= 1000:
                n = 95
    dx = 2.0 / n
    m = n / 2.0
    _GLOBAL_M = m
    # Fallback half-block
    h_fb = np.zeros(n)
    if n % 2 == 0:
        h_fb[:n // 2] = 1.0
    else:
        h_fb[:n // 2] = 1.0
        h_fb[n // 2] = 0.5
    best_h = h_fb.copy()
    best_score = exact_score(h_fb)
    # Use parent if available
    if parent is not None:
        p = project_capped_simplex(parent, n, m)
        if is_feasible(p, n, m):
            s = exact_score(p)
            parent = p
        else:
            parent = h_fb.copy()
        if exact_score(parent) < best_score:
            best_score = exact_score(parent)
            best_h = parent.copy()
    base = parent if parent is not None else h_fb.copy()
    # Generate 8 starts
    starts = [base.copy()]
    for sig in [0.05, 0.15, 0.30]:
        pert = base + sig * rng.standard_normal(n)
        starts.append(project_capped_simplex(pert, n, m))
    starts.append(project_capped_simplex(rng.uniform(0.0, 1.0, n), n, m))
    starts.append(project_capped_simplex(rng.beta(0.7, 0.7, n), n, m))
    phase = rng.uniform(0.0, 2.0 * np.pi)
    idx = np.arange(n, dtype=np.float64)
    sin_v = 0.5 + 0.45 * np.sin(2.0 * np.pi * idx / n + phase)
    starts.append(project_capped_simplex(sin_v, n, m))
    perm = base.copy()
    rng.shuffle(perm)
    starts.append(project_capped_simplex(perm, n, m))
    # Score starts
    for si in range(len(starts)):
        if is_feasible(starts[si], n, m):
            sc = exact_score(starts[si])
            if sc < best_score - 1e-12:
                best_score = sc
                best_h = starts[si].copy()
    # Continuation SLSQP
    tau_scale = n / 95.0
    taus = [0.10 * tau_scale, 0.03 * tau_scale, 0.008 * tau_scale,
            0.002 * tau_scale, 0.0005 * tau_scale]
    constraints = [{'type': 'eq', 'fun': _mass_fun, 'jac': _mass_jac}]
    bounds = [(0.0, 1.0)] * n
    for si in range(len(starts)):
        x0 = starts[si].copy()
        if not is_feasible(x0, n, m):
            x0 = project_capped_simplex(x0, n, m)
        if not is_feasible(x0, n, m):
            continue
        for tau in taus:
            try:
                res = minimize(smooth_obj_grad, x0, args=(tau, dx),
                               method='SLSQP', jac=True, bounds=bounds,
                               constraints=constraints,
                               options={'maxiter': 120, 'ftol': 1e-9, 'disp': False})
            except Exception:
                break
            if res.x is None or len(res.x) != n or not np.all(np.isfinite(res.x)):
                break
            xn = project_capped_simplex(res.x, n, m)
            if not is_feasible(xn, n, m):
                break
            f_old, _ = smooth_obj_grad(x0, tau, dx)
            f_new, _ = smooth_obj_grad(xn, tau, dx)
            if not np.isfinite(f_new) or f_new > f_old + 1e-8:
                break
            x0 = xn
            sc = exact_score(x0)
            if sc < best_score - 1e-12:
                best_score = sc
                best_h = x0.copy()
    # Final validation
    if not is_feasible(best_h, n, m):
        best_h = project_capped_simplex(best_h, n, m)
    if not is_feasible(best_h, n, m):
        best_h = h_fb.copy()
    best_h = np.clip(best_h, 0.0, 1.0)
    final_score = exact_score(best_h)
    return best_h, final_score, n
