    return float(np.max(q))

    return z[N]


def objective_jac(z, N):
    g = np.zeros(N + 1)
    g[N] = 1.0
    return g


def eq_fun(z, N, m):
    return np.sum(z[:N]) - m


def eq_jac(z, N, m):
    g = np.zeros(N + 1)
    g[:N] = 1.0
    return g


def ineq_fun(z, N, m):
    h = z[:N]
    t = z[N]
    q = np.correlate(h, 1.0 - h, mode='full') * (2.0 / N)
    return t - q


def ineq_jac(z, N, m):
    h = z[:N]
    J = grad_q_matrix(h, N)
    aug = np.zeros((2 * N - 1, N + 1), dtype=np.float64)
    aug[:, :N] = -J
    aug[:, N] = 1.0
    return aug


def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)

    N = 70
    h_parent = None
    try:
        global initial_h_values
        if initial_h_values is not None:
            pv = np.asarray(initial_h_values, dtype=np.float64).ravel()
            if pv.size > 0 and np.all(np.isfinite(pv)):
                N = pv.size
                h_parent = pv.copy()
    except Exception:
        pass

    if N <= 0:
        N = 70
    dx = 2.0 / N
    m = N / 2.0

    h_half = np.full(N, 0.5, dtype=np.float64)
    best_h = h_half.copy()
    best_c = exact_score(h_half, N)

    if h_parent is not None:
        if (np.all(h_parent >= -1e-12) and np.all(h_parent <= 1.0 + 1e-12)
                and abs(np.sum(h_parent) - m) < 1e-10):
            h_feasible = np.clip(h_parent, 0.0, 1.0)
        else:
            h_feasible = project_box_sum(h_parent, m)
        c_par = exact_score(h_feasible, N)
        if c_par < best_c - 1e-12:
            best_c = c_par
            best_h = h_feasible.copy()
        base = h_feasible
    else:
        base = h_half.copy()

    starts = [base.copy()]

    if N <= 256:
        for scale in [0.03, 0.08, 0.18]:
            starts.append(project_box_sum(base + scale * rng.standard_normal(N), m))
        starts.append(project_box_sum(rng.random(N), m))
        starts.append(project_box_sum(rng.beta(0.7, 0.7, N), m))
        i_arr = np.arange(N, dtype=np.float64) + 0.5
        sm1 = 0.5 + 0.45 * np.cos(2.0 * np.pi * i_arr / N)
        sm2 = 0.5 + 0.45 * np.cos(4.0 * np.pi * i_arr / N + np.pi / 3.0)
        starts.append(project_box_sum(sm1, m))
        starts.append(project_box_sum(sm2, m))
    else:
        starts.append(project_box_sum(base + 0.08 * rng.standard_normal(N), m))
        starts.append(project_box_sum(rng.random(N), m))
        starts.append(project_box_sum(rng.beta(0.7, 0.7, N), m))

    for hs in starts:
        c = exact_score(hs, N)
        if c < best_c - 1e-12:
            best_c = c
            best_h = hs.copy()

    if N <= 999:
        bounds = [(0.0, 1.0)] * N + [(0.0, 2.0)]
        for h0 in starts:
            try:
                q0 = np.correlate(h0, 1.0 - h0, mode='full') * dx
                t0 = float(np.clip(np.max(q0), 0.0, 2.0))
                z0 = np.empty(N + 1)
                z0[:N] = h0
                z0[N] = t0
                res = minimize(
                    fun=objective_fn,
                    x0=z0,
                    args=(N,),
                    jac=objective_jac,
                    bounds=bounds,
                    constraints=[
                        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (N, m)},
                        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac, 'args': (N, m)},
                    ],
                    method='SLSQP',
                    options={'maxiter': 250, 'ftol': 1e-10, 'disp': False},
                )
                if res.x is not None and len(res.x) == N + 1 and np.all(np.isfinite(res.x[:N])):
                    h_r = project_box_sum(res.x[:N], m)
                    c_r = exact_score(h_r, N)
                    if c_r < best_c - 1e-12:
                        best_c = c_r
                        best_h = h_r.copy()
            except Exception:
                continue

    best_h = project_box_sum(best_h, m)
    if (np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12)
            and abs(np.sum(best_h) - m) < 1e-8):
        final_c = exact_score(best_h, N)
    else:
        best_h = np.full(N, 0.5, dtype=np.float64)
        final_c = exact_score(best_h, N)

    return best_h, final_c, int(N)
