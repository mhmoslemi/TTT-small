import numpy as np
import math

def is_feasible(h, n):
    if h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def project_capped(y, n):
    try:
        y = np.asarray(y, dtype=np.float64)
        if y.shape != (n,):
            return np.full(n, 0.5, dtype=np.float64)
        if not np.all(np.isfinite(y)) or np.max(np.abs(y)) > 1e12:
            return np.full(n, 0.5, dtype=np.float64)
        target = n / 2.0
        lo = np.min(y) - 1.0
        hi = np.max(y)
        for _ in range(80):
            mid = (lo + hi) / 2.0
            s = np.sum(np.clip(y - mid, 0.0, 1.0))
            if s > target:
                lo = mid
            else:
                hi = mid
        p = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
        for _ in range(n + 2):
            r = target - np.sum(p)
            if abs(r) <= 2e-14 * max(1, n):
                break
            if r > 0:
                slack = 1.0 - p
                for i in range(n):
                    if r <= 0:
                        break
                    d = min(r, slack[i])
                    p[i] += d
                    r -= d
            else:
                for i in range(n):
                    if r >= 0:
                        break
                    d = min(-r, p[i])
                    p[i] -= d
                    r += d
        if not np.all(np.isfinite(p)):
            return np.full(n, 0.5, dtype=np.float64)
        if not is_feasible(p, n):
            return np.full(n, 0.5, dtype=np.float64)
        return p
    except Exception:
        return np.full(n, 0.5, dtype=np.float64)

def exact_score(h):
    try:
        n = len(h)
        if not is_feasible(h, n):
            return float('inf')
        val = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)
        if not np.isfinite(val):
            return float('inf')
        return val
    except Exception:
        return float('inf')

def exchange_proposal(x, n, rng, rho, K):
    try:
        y = x.copy()
        for _ in range(K):
            donors = np.where(y > 1e-14)[0]
            receivers = np.where(y < 1.0 - 1e-14)[0]
            if len(donors) == 0 or len(receivers) == 0:
                break
            found = False
            for attempt in range(16):
                i = donors[rng.integers(len(donors))]
                j = receivers[rng.integers(len(receivers))]
                if i != j:
                    found = True
                    break
            if not found:
                for i in donors:
                    for j in receivers:
                        if i != j:
                            break
                    if found:
                        break
                if not found:
                    continue
            c = min(y[i], 1.0 - y[j])
            if c <= 0:
                continue
            u = rng.uniform(0.2, 1.0)
            delta = rho * u * c
            y[i] -= delta
            y[j] += delta
        if not is_feasible(y, n):
            y = project_capped(y, n)
            if not is_feasible(y, n):
                return None
        return y
    except Exception:
        return None

def build_starts(base, n, rng, max_starts):
    starts = [base.copy()]
    if n > 256 or max_starts <= 1:
        return starts
    K1 = min(8, max(2, math.ceil(n / 10.0)))
    s1 = exchange_proposal(base, n, rng, 0.18, K1)

        starts.append(s1)

        dup = any(np.max(np.abs(s2 - s)) < 1e-12 for s in starts)

            starts.append(s2)

        idx = rng.permutation(n)

        s3[idx[:n // 2]] = 1.0

        if is_feasible(s3, n):
            dup = any(np.max(np.abs(s3 - s)) < 1e-12 for s in starts)
            if not dup:
                starts.append(s3)
    if len(starts) < max_starts:
        s4 = project_capped(rng.uniform(0, 1, n), n)
        if is_feasible(s4, n):
            dup = any(np.max(np.abs(s4 - s)) < 1e-12 for s in starts)
            if not dup:
                starts.append(s4)
