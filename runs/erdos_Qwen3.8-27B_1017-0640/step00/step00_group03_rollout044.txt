            for i in range(n):
                sub = min(-resid, q[i])
                q[i] -= sub
                resid += sub
                if resid >= 0:
                    break
    if not np.all(np.isfinite(q)) or np.any(q < -1e-15) or np.any(q > 1.0 + 1e-15):
        return None
    q = np.clip(q, 0.0, 1.0)
    if abs(np.sum(q) - target) > 1e-10:
        return None
    return q

def check_feasible(h, n):
    if h is None or h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-15) or np.any(h > 1.0 + 1e-15):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10:
        return False
    return True

def exact_score(h, n):
    if not check_feasible(h, n):
        return float('inf')
    corr = np.correlate(h, 1.0 - h, mode='full')
    if len(corr) != 2 * n - 1 or not np.all(np.isfinite(corr)):
        return float('inf')
    return float(np.max(corr) * (2.0 / n))

def compute_lags_and_jac(h, n):
    """Return c (2n-1 vector) and J ((2n-1) x n Jacobian) for correlation entries."""
    m = 2 * n - 1
    c = np.zeros(m)
    J = np.zeros((m, n))
    for k in range(m):
        d = n - 1 - k
        if d > 0:
            # f = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
            idx = np.arange(n - d)
            c[k] = np.sum(h[idx] * (1.0 - h[idx + d]))
            # grad: (1-h[j+d]) for j in [0, n-1-d], -h[j-d] for j in [d, n-1]
            J[k, :n-d] += (1.0 - h[d:])
            J[k, d:] -= h[:n-d]
        elif d == 0:
            c[k] = np.sum(h * (1.0 - h))
            J[k, :] = 1.0 - 2.0 * h
        else:
            e = -d
            # f = sum_{j=0}^{n-1-e} h[j+e]*(1-h[j])
            idx = np.arange(n - e)
            c[k] = np.sum(h[idx + e] * (1.0 - h[idx]))
            # grad: (1-h[j-e]) for j in [e, n-1], -h[j+e] for j in [0, n-1-e]
            J[k, e:] += (1.0 - h[:n-e])
            J[k, :n-e] -= h[e:]
    if not np.all(np.isfinite(c)) or not np.all(np.isfinite(J)):
        return None, None
    return c, J

def lp_solve(y, c, J, n, r, mu):
    """Solve trust-region LP. Returns x proposal or None."""
    m = 2 * n - 1
    nz = 2 * n + 1  # x (n) + u (n) + t (1)
    # Objective: [0, 0, ..., 0, mu, mu, ..., mu, 1]
    c_obj = np.zeros(nz)
    c_obj[n:2*n] = mu
    c_obj[2*n] = 1.0
    # A_ub: (4n-1) x nz
    n_rows = m + 2 * n
    A_ub = np.zeros((n_rows, nz))
    b_ub = np.zeros(n_rows)
    # Lag constraints: grad[k] @ x - t <= grad[k] @ y - c[k]
    A_ub[:m, :n] = J
    A_ub[:m, n:2*n] = 0.0
    A_ub[:m, 2*n] = -1.0
    b_ub[:m] = J @ y - c
    # u constraints
    row = m
    for i in range(n):
        A_ub[row, i] = 1.0
        A_ub[row, n + i] = -1.0
        b_ub[row] = y[i]
        row += 1
        A_ub[row, i] = -1.0
        A_ub[row, n + i] = -1.0
        b_ub[row] = -y[i]
        row += 1
    # A_eq: sum(x) = n/2
    A_eq = np.zeros((1, nz))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    # Bounds
    lb = np.zeros(nz)
    ub = np.full(nz, np.inf)
    for i in range(n):
        lb[i] = max(0.0, y[i] - r)
        ub[i] = min(1.0, y[i] + r)
    lb[n:2*n] = 0.0
    ub[n:2*n] = np.inf
    lb[2*n] = -np.inf
    ub[2*n] = np.inf
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=list(zip(lb, ub)), method='highs',
                      options={'presolve': True, 'max_iter': 500})
    except Exception:
        return None
    if res is None or not res.success or res.x is None:
        return None
    x_prop = res.x[:n]
    if not np.all(np.isfinite(x_prop)):
        return None
    return x_prop
