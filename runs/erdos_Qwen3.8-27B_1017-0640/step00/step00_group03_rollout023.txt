import numpy as np
import math

def check_feasible(h, n):
    if h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(float(np.sum(h)) - n / 2.0) > 1e-10:
        return False
    return True

def project_capped(v, n):
    if v.shape != (n,) or not np.all(np.isfinite(v)):
        return None
    v = np.clip(v, -1e6, 1e6).copy()
    target = n / 2.0
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = (lo + hi) / 2.0
        q = np.clip(v - mid, 0.0, 1.0)
        s = float(np.sum(q))
        if s > target:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2.0
    q = np.clip(v - mid, 0.0, 1.0).copy()
    res = target - float(np.sum(q))
    if abs(res) > 1e-13:
        for i in range(n):
            if abs(res) <= 1e-13:
                break
            if res > 0:
                add = min(res, 1.0 - q[i])
                q[i] += add
                res -= add
            else:
                sub = min(-res, q[i])
                q[i] -= sub
                res += sub
    if not np.all(np.isfinite(q)) or np.any(q < -1e-12) or np.any(q > 1.0 + 1e-12):
        return None
    if abs(float(np.sum(q)) - target) > 1e-10:
        return None
    q = np.clip(q, 0.0, 1.0)
    return q

def exact_score(h, n):
    if not check_feasible(h, n):
        return float('inf')
    corr = np.correlate(h, 1.0 - h, mode='full')
    if corr.shape != (2 * n - 1,):
        return float('inf')
    if not np.all(np.isfinite(corr)):
        return float('inf')
    s = float(np.max(corr * (2.0 / n)))
    if not math.isfinite(s):
        return float('inf')
    return s

def make_proposal(h, radius, n, rng):
    for _ in range(8):
        p = int(rng.integers(0, n))
        q = int(rng.integers(0, n))
        if p == q:
            continue
        a_min = max(-h[p], h[q] - 1.0)
        a_max = min(1.0 - h[p], h[q])
        a_min = max(a_min, -radius)
        a_max = min(a_max, radius)
        if a_max - a_min <= 1e-12:
            continue
        alpha = float(rng.uniform(a_min, a_max))
        if abs(alpha) <= 1e-12:
            continue
        hp = h.copy()
        hp[p] += alpha
        hp[q] -= alpha
        proj = project_capped(hp, n)
        if proj is not None and check_feasible(proj, n):
            return proj
    return None

def anneal_chain(start, start_score, n, rng, incumbent, inc_score):
    radii = (0.18, 0.12, 0.07, 0.035, 0.015)
    temps = (0.006, 0.002, 0.0007, 0.00025, 0.00008)
    attempts = 240 if n <= 128 else 80
    cur = start.copy()
    cur_s = start_score
    if cur_s < inc_score - 1e-12:
        incumbent = cur.copy()
        inc_score = cur_s
    for radius, temp in zip(radii, temps):
        for _ in range(attempts):

                continue
            ps = exact_score(prop, n)

                cur_s = ps

                inc_score = cur_s

        inc_score = cur_s

def polish(h_best, score_best, n, rng, incumbent, inc_score):

                cur_s = ps

                    inc_score = cur_s
