            if rng.random() < 0.7:
                p = int(order[rng.integers(pc)])
                if p == i and P > 1:
                    alts = [int(order[j]) for j in range(pc)]
                    p = alts[[a for ai, a in enumerate(alts) if a != i][0]] if len(alts) > 1 else (i+1) % P
                ot = [j for j in range(P) if j != i and j != p]
                r1 = int(ot[rng.integers(len(ot))])
                ot.remove(r1)
                r2 = int(ot[rng.integers(len(ot))])
                donor = snap[i] + Fv*(snap[p]-snap[i]) + Fv*(snap[r1]-snap[r2])
            else:
                ot = [j for j in range(P) if j != i]
                r1 = int(ot[rng.integers(len(ot))])
                ot.remove(r1)
                r2 = int(ot[rng.integers(len(ot))])
                ot.remove(r2)
                r3 = int(ot[rng.integers(len(ot))])
                donor = snap[r1] + Fv*(snap[r2]-snap[r3])
            mask = rng.random(n) < CRv
            mask[int(rng.integers(n))] = True
            trial = np.where(mask, donor, snap[i])
            pr = _project(trial, n)
            if pr is None or np.max(np.abs(pr - snap[i])) <= 1e-15:
                continue
            sc = _score(pr, n)
            if sc is None:
                continue
            if sc < sss[i] - 1e-12:
                pop[i], ps[i] = pr, sc
                if sc < bs - 1e-12:
                    bs, best = sc, pr.copy()
        if not any(np.max(np.abs(pop[j]-best)) <= 1e-12 for j in range(P)):
            wi = int(np.argmax(ps))
            pop[wi], ps[wi] = best.copy(), bs
        if gbs - float(np.min(ps)) >= 1e-10:
            stag = 0
        else:
            stag += 1
        if stag >= 18:
            if rest >= 4:
                break
            rc = max(1, P//3)
            sd_r = rds[rest]
            bj = [j for j in range(P) if np.max(np.abs(pop[j]-best)) <= 1e-12]
            ct = sorted([j for j in range(P) if j not in bj], key=lambda j: ps[j], reverse=True)
            for k, j in enumerate(ct[:rc]):
                if k % 2 == 0 or rc == 1:
                    t = best + rng.normal(0, sd_r, n)
                else:
                    t = rng.uniform(-0.5, 1.5, n)
                pr = _project(t, n)
                if pr is not None:
                    sc = _score(pr, n)
                    if sc is not None:
                        pop[j], ps[j] = pr, sc
                        if sc < bs - 1e-12:
                            bs, best = sc, pr.copy()
            rest, stag = rest + 1, 0
    pct = 16 if n <= 64 else (10 if n <= 160 else 5)
    for sc_l in [0.12, 0.05, 0.02, 0.008, 0.003]:
        acc = 0
        for _ in range(pct):
            z = rng.standard_normal(n).astype(np.float64)
            z -= np.mean(z)
            rm = math.sqrt(np.mean(z*z))
            if rm > 1e-15:
                z /= rm
            pr = _project(best + sc_l * z, n)
            if pr is None:
                continue
            sc = _score(pr, n)
            if sc is not None and sc < bs - 1e-12:
                bs, best, acc = sc, pr.copy(), acc + 1
        if acc > 0:
            for _ in range(pct):
                z = rng.standard_normal(n).astype(np.float64)
                z -= np.mean(z)
                rm = math.sqrt(np.mean(z*z))
                if rm > 1e-15:
                    z /= rm
                pr = _project(best + sc_l * z, n)
                if pr is None:
                    continue
                sc = _score(pr, n)
                if sc is not None and sc < bs - 1e-12:
                    bs, best = sc, pr.copy()
    fs = _score(best, n)
    if fs is None:
        pr = _project(best, n)
        if pr is None:
            pr = fallback if fallback.shape == (n,) else np.full(n, 0.5)
        best = pr
        fs = _score(best, n)
        if fs is None:
            best = np.full(n, 0.5, dtype=np.float64)
            fs = _score(best, n)
    return (best, float(fs), n)
