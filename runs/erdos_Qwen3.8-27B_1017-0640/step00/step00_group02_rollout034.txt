def project_capped(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if not np.all(np.isfinite(v)):
        return None
    v = np.clip(v, -1e6, 1e6)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        x = np.clip(v - mid, 0.0, 1.0)
        if x.sum() > S:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-14:
            break
    x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = S - float(x.sum())
    for _ in range(4):
        if abs(r) < 1e-12:
            break
        if r > 0:
            cap = 1.0 - x
            tc = float(cap.sum())
            if tc < 1e-14:
                break
            x += cap * min(1.0, r / tc)
        else:
            tm = float(x.sum())
            if tm < 1e-14:
                break

        return None

        return None
    return x

def exact_score(h, n):
