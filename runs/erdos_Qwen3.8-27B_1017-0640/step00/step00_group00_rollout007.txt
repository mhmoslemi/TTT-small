def cs_project(y, n):
    m = n / 2.0
    y = np.clip(y.astype(np.float64), -1e6, 1e6)
    lo, hi = y.min() - 1.0, y.max()
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(y - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    r = m - x.sum()
    if r > 0:
        for i in range(n):
            if r <= 0: break
            a = min(1.0 - x[i], r); x[i] += a; r -= a
    elif r < 0:
        for i in range(n):
            if r >= 0: break
            a = min(x[i], -r); x[i] -= a; r += a
    x = np.clip(x, 0.0, 1.0)
    if not np.all(np.isfinite(x)) or abs(x.sum() - m) > 1e-9:
        return np.full(n, 0.5)

def ok(h, n):

            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(h.sum() - n / 2.0) <= 1e-9)

def sc(h, n):
    try:
        v = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))
        return v if np.isfinite(v) else 1e18
    except Exception:
        return 1e18

def lag_grads(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    G[n - 1] = dx * (1.0 - 2.0 * h)
    for p in range(1, n):
        G[n - 1 + p, p:] += dx * (1.0 - h[:n - p])
        G[n - 1 + p, :n - p] -= dx * h[p:]
        G[n - 1 - p, :n - p] += dx * (1.0 - h[p:])
        G[n - 1 - p, p:] -= dx * h[:n - p]
    return G

def fo(z):
    return z[-1]

    g = np.zeros(len(z)); g[-1] = 1.0; return g

    g = np.zeros(len(z)); g[:-1] = 1.0; return g

    return z[-1] - np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

    J[:, :-1] = -G; J[:, -1] = 1.0
    return J
