                            local = t
                            lscore = sv
                            if lscore < inc_score - 1e-12:
                                inc_score = lscore
                                incumbent = local.copy()
    if not check_feasible(incumbent, n, S):
        pj = project_capped(incumbent, n, S)
        if pj is not None and exact_score(pj, n) <= exact_score(fallback, n):
            incumbent = pj
        else:
            incumbent = fallback
    final_s = exact_score(incumbent, n)
    fb_s = exact_score(fallback, n)
    if not np.isfinite(final_s) or final_s > fb_s:
        incumbent = fallback
        final_s = fb_s
    h_out = incumbent.astype(np.float64).copy()
    c5 = float(np.max(np.correlate(h_out, 1.0 - h_out, mode='full')) * dx)
    if not np.isfinite(c5):
        h_out = fallback.astype(np.float64).copy()
        c5 = float(np.max(np.correlate(h_out, 1.0 - h_out, mode='full')) * dx)
    return h_out, c5, int(n)
