import numpy as np
import math


def project_capped_simplex(v, n):
    v = np.asarray(v, dtype=np.float64)
    if v.shape != (n,) or not np.all(np.isfinite(v)):
        return None
    v = np.clip(v, -1e6, 1e6)
    target = n / 2.0
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = (lo + hi) / 2.0
        q = np.clip(v - mid, 0.0, 1.0)
        if float(np.sum(q)) > target:
            lo = mid
        else:
            hi = mid
    q = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    resid = target - float(np.sum(q))
    if resid > 0:
        for i in range(n):
            add = min(1.0 - q[i], resid)
            q[i] += add
            resid -= add
            if abs(resid) <= 1e-13:
                break
    elif resid < 0:
        for i in range(n):
            sub = min(q[i], -resid)
            q[i] -= sub
            resid += sub
            if abs(resid) <= 1e-13:
                break
    if not np.all(np.isfinite(q)):
        return None
    if np.any(q < -1e-12) or np.any(q > 1.0 + 1e-12):
        return None
    if abs(float(np.sum(q)) - target) > 1e-10:
        return None
    q = np.clip(q, 0.0, 1.0)
    return q


def is_feasible(h, n):
    if h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(float(np.sum(h)) - n / 2.0) > 1e-10:
        return False
    return True


def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = 1.0 - h
    corr = np.correlate(h, c, mode='full')
    if corr.shape[0] != 2 * n - 1:
        return float('inf')
    if not np.all(np.isfinite(corr)):
        return float('inf')
    return float(np.max(corr * (2.0 / n)))


def propose_pair_transfer(h, radius, n, rng):
    h = np.asarray(h, dtype=np.float64)
    for _ in range(8):
        p = int(rng.integers(0, n))
        q = int(rng.integers(0, n))
        if p == q:
            continue
        a_lo = max(-h[p], h[q] - 1.0)
        a_hi = min(1.0 - h[p], h[q])
        a_lo = max(a_lo, -radius)
        a_hi = min(a_hi, radius)
        if a_hi - a_lo <= 1e-12:
            continue
        alpha = float(rng.uniform(a_lo, a_hi))
        if abs(alpha) <= 1e-12:
            continue
        new_h = h.copy()
        new_h[p] += alpha
        new_h[q] -= alpha
        proj = project_capped_simplex(new_h, n)
        if proj is not None and is_feasible(proj, n):
            return proj
    return None


def anneal_chain(start, n, rng, inc_h, inc_score):
    radii = [0.18, 0.12, 0.07, 0.035, 0.015]
    temps = [0.006, 0.002, 0.0007, 0.00025, 0.00008]
    attempts = 240 if n <= 128 else 80
    cur = start.copy()
    cur_s = exact_score(cur, n)
    if cur_s < inc_score - 1e-12:
        inc_h[:] = cur
        inc_score = cur_s
    for stage in range(5):
        r = radii[stage]
        T = temps[stage]
        for _ in range(attempts):
            prop = propose_pair_transfer(cur, r, n, rng)
            if prop is None:
                continue
            ps = exact_score(prop, n)
            if not np.isfinite(ps):
                continue
            delta = ps - cur_s
            if delta <= 0:
                cur = prop
                cur_s = ps
            else:
                ratio = delta / T

                    cur_s = ps

                inc_h[:] = cur
                inc_score = cur_s
            if ps < inc_score - 1e-12:
                inc_h[:] = prop
                inc_score = ps
    return cur, cur_s


def greedy_polish(h, score, n, rng, inc_h, inc_score):
    radii = [0.02, 0.01, 0.005, 0.002, 0.001]
    ntry = 100 if n <= 128 else 40
    cur = h.copy()
    cur_s = score
    for r in radii:
        for _ in range(ntry):
            prop = propose_pair_transfer(cur, r, n, rng)
            if prop is None:
                continue
            ps = exact_score(prop, n)
            if not np.isfinite(ps):
                continue
            if ps < cur_s - 1e-12:
                cur = prop
                cur_s = ps
            if cur_s < inc_score - 1e-12:
                inc_h[:] = cur
                inc_score = cur_s
    return cur, cur_s
