    sc = exact_score(h, N)
    if sc is None:
        return h0, None
    T0 = 0.15 * dx
    T1 = 0.002 * dx
    s0 = 0.25
    s1 = 0.01
    for t in range(B):
        f = t / max(B - 1, 1)
        T = T0 * (T1 / T0) ** f
        s = s0 * (s1 / s0) ** f
        pair = None
        for _ in range(20):
            j = int(rng.integers(N))
            i = int(rng.integers(N))
            if i != j and h[j] > 1e-14 and h[i] < 1.0 - 1e-14:
                pair = (i, j); break
        if pair is None:
            for j in range(N):
                if h[j] > 1e-14:
                    for i in range(N):
                        if i != j and h[i] < 1.0 - 1e-14:
                            pair = (i, j); break
                    if pair: break
        if pair is None:
            break
        i, j = pair
        a_max = min(h[j], 1.0 - h[i])
        U = rng.random()
        a = min(a_max, s * (0.25 + 0.75 * U))
        if a <= 1e-14:
            continue
        trial = h.copy()
        trial[i] += a
        trial[j] -= a
        fix_residual(trial, m)
        if not check_feasible(trial, N):
            continue
        ts = exact_score(trial, N)
        if ts is None:
            continue
        dC = ts - sc
        if dC <= 0:
            h = trial; sc = ts
        else:
            if dC / T >= 700:
                continue
            if rng.random() < math.exp(-dC / T):
                h = trial; sc = ts
    return h, sc

def pattern_poll(h0, N, rng):
    m = N / 2.0
    h = h0.copy()
    sc = exact_score(h, N)
    if sc is None or N < 2:
        return h0, sc
    if N <= 256:
        radii = [0.08, 0.04, 0.02, 0.01]
    else:
        radii = [0.04, 0.02, 0.01]
    for radius in radii:
        pairs = []
        if N <= 80:
            for i in range(N):
                for j in range(N):
                    if i != j:
                        pairs.append((i, j))
            rng.shuffle(pairs)
        else:
            total = N * (N - 1)
            if N <= 128:
                n_sample = min(2000, total)
            elif N <= 256:
                n_sample = min(600, total)
            else:
                n_sample = 150
            if n_sample >= total and N <= 256:
                ids = rng.choice(total, size=n_sample, replace=False)
            else:
                ids = rng.choice(total, size=n_sample, replace=(N > 256))
            for pid in ids:
                ii = int(pid) // (N - 1)
                rr = int(pid) % (N - 1)
                jj = rr if rr < ii else rr + 1
                pairs.append((ii, jj))
        best_trial = None
        best_sc = sc
        for (i, j) in pairs:
            a = min(radius, 1.0 - h[i], h[j])
            if a <= 1e-14:
                continue
            trial = h.copy()
            trial[i] += a
            trial[j] -= a
            fix_residual(trial, m)
            if not check_feasible(trial, N):
                continue
            ts = exact_score(trial, N)
            if ts is not None and ts < best_sc - 1e-12:
                best_sc = ts
                best_trial = trial.copy()
        if best_trial is not None:
            h = best_trial
            sc = best_sc
    return h, sc

def run(seed=42, **kwargs):
    N = 70
    base = None
    try:
        raw = np.array(initial_h_values, dtype=np.float64)
        if raw.ndim == 1 and raw.shape[0] > 0:
            N = int(raw.shape[0])
            if np.all(np.isfinite(raw)):
                m_tmp = N / 2.0
                if check_feasible(raw, N):
                    base = raw.copy()
                else:
                    p = project_box_sum(raw, m_tmp)
                    if p is not None and check_feasible(p, N):
                        base = p
    except Exception:
        pass
    m = N / 2.0
    neutral = np.full(N, 0.5)
    if base is None:
        base = neutral.copy()
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    starts = []
    i_arr = np.arange(N)
    if N <= 999 and N >= 2:
        try_add_start(starts, base, N, m)
        try_add_start(starts, base[::-1].copy(), N, m)
        try_add_start(starts, 1.0 - base, N, m)
        if N <= 128:
            try_add_start(starts, base + rng.normal(0, 0.05, N), N, m)
            try_add_start(starts, base + rng.normal(0, 0.15, N), N, m)
            try_add_start(starts, rng.uniform(0, 1, N), N, m)
            try_add_start(starts, rng.beta(0.7, 0.7, N), N, m)
            try_add_start(starts, 0.5 + 0.45 * np.cos(4.0 * np.pi * (i_arr + 0.5) / N + 0.37), N, m)
        elif N <= 256:
            try_add_start(starts, base + rng.normal(0, 0.08, N), N, m)
            try_add_start(starts, rng.uniform(0, 1, N), N, m)
            try_add_start(starts, rng.beta(0.7, 0.7, N), N, m)
            try_add_start(starts, 0.5 + 0.45 * np.cos(4.0 * np.pi * (i_arr + 0.5) / N + 0.37), N, m)
        elif N <= 512:
            try_add_start(starts, base + rng.normal(0, 0.08, N), N, m)
            try_add_start(starts, rng.uniform(0, 1, N), N, m)
            try_add_start(starts, 0.5 + 0.45 * np.cos(4.0 * np.pi * (i_arr + 0.5) / N + 0.37), N, m)
        else:
            try_add_start(starts, base + rng.normal(0, 0.05, N), N, m)
        try_add_start(starts, neutral, N, m)
    if not starts:
        starts = [neutral.copy()]
    global_best = starts[0].copy()
    global_best_sc = exact_score(global_best, N)
    for s in starts[1:]:
        sc = exact_score(s, N)
        if sc is not None and (global_best_sc is None or sc < global_best_sc - 1e-12):
            global_best = s.copy()
            global_best_sc = sc
    if N <= 999 and N >= 2:
        if N <= 128:
            B = 800
        elif N <= 256:
            B = 400
        elif N <= 512:
            B = 150
        else:
            B = 40
        for s in starts:
            h_out, sc_out = anneal_start(s, N, B, rng)
            if sc_out is not None and (global_best_sc is None or sc_out < global_best_sc - 1e-12):
                global_best = h_out.copy()
                global_best_sc = sc_out
    if N <= 999 and N >= 2:
        h_poll, sc_poll = pattern_poll(global_best, N, rng)
        if sc_poll is not None and sc_poll < global_best_sc - 1e-12:
            global_best = h_poll.copy()
            global_best_sc = sc_poll
    final_p = project_box_sum(global_best, m)
    if final_p is not None and check_feasible(final_p, N):
        global_best = final_p
    final_sc = exact_score(global_best, N)
    if final_sc is None or not check_feasible(global_best, N):
        if base is not None:
            cb = project_box_sum(base, m)
            if cb is not None and check_feasible(cb, N):
                global_best = cb
                final_sc = exact_score(global_best, N)
    if final_sc is None or not check_feasible(global_best, N):
        global_best = neutral.copy()
        final_sc = exact_score(global_best, N)
    if final_sc is None:
        final_sc = 0.25
        global_best = neutral.copy()
    return global_best.astype(np.float64), float(final_sc), int(N)
