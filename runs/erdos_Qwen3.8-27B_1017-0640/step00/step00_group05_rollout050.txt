import numpy as np
from scipy.optimize import minimize

def project_box_sum(v, m):
    v = np.clip(v.astype(np.float64), -1e6, 1e6)
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if y.sum() > m:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(3):
        r = m - y.sum()
        if abs(r) < 1e-14:
            break
        if r > 0:
            order = np.arange(len(y))
            for idx in order:
                cap = 1.0 - y[idx]
                add = min(r, cap)
                y[idx] += add
                r -= add
                if abs(r) < 1e-14:
                    break
        else:
            order = np.arange(len(y))
            for idx in order:
                cap = y[idx]
                sub = min(-r, cap)
                y[idx] -= sub
                r += sub
                if abs(r) < 1e-14:
                    break
    if not np.all(np.isfinite(y)) or y.min() < -1e-12 or y.max() > 1 + 1e-12:
        return 0.5 * np.ones(len(v))
    y = np.clip(y, 0.0, 1.0)
    return y

def score_h(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr * dx)), corr * dx

def build_gradient(h, N):
    dx = 2.0 / N
    K = 2 * N - 1
    J = np.zeros((K, N), dtype=np.float64)
    for j in range(K):
        lag = j - (N - 1)
        for r in range(N):
            val = 0.0
            a = r - lag
            if 0 <= a < N:
                val += (1.0 - h[a])
            b = r + lag
            if 0 <= b < N:
                val -= h[b]
            J[j, r] = dx * val
    return J

def _objective(z):
    return z[-1]

def _objective_grad(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g

def make_eq_constraint(N, m):
    def eq(z):
        return z[:N].sum() - m
    def eq_grad(z):
        g = np.zeros(N + 1)
        g[:N] = 1.0
        return g
    return eq, eq_grad

def make_ineq_constraint(N, J):
    K = 2 * N - 1
    def ineq(z):
        h = z[:N]
        t = z[N]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / N)
        return t - corr
    def ineq_grad(z):
        h = z[:N]
        Jh = build_gradient(h, N)
        Jc = np.zeros((K, N + 1))
        Jc[:, :N] = -Jh
        Jc[:, N] = 1.0
        return Jc
    return ineq, ineq_grad

    parent_ok = False
    try:

    except Exception:
        pass
    if N <= 0:
        N = 70
        parent_ok = False
    dx = 2.0 / N
    m = N / 2.0
    h_base = 0.5 * np.ones(N)
    best_h = h_base.copy()
    best_score, _ = score_h(h_base, dx)
    if parent_ok:

        except Exception:
            pass
    rng = np.random.default_rng(seed)
