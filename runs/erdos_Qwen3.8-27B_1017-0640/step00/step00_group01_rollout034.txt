        phi = rng.uniform(0, 2*math.pi)
        add_cand(pops, seen, 0.5 + 0.35*np.sin(2*math.pi*f*(idx+0.5)/n + phi), n, s)
    sigs = [0.04, 0.10, 0.22, 0.45, 0.70]
    att = 0; mx = 30*P
    while len(pops) < P and att < mx:
        att += 1; k = att % 3
        if k == 0: add_cand(pops, seen, base + sigs[att%5]*rng.standard_normal(n), n, s)
        elif k == 1: add_cand(pops, seen, rng.uniform(0,1,n), n, s)
        else: add_cand(pops, seen, rng.beta(0.7,0.7,n), n, s)
    while len(pops) < P:
        c = base.copy(); pops.append(c); seen.append(c)
    return np.array(pops[:P])

def run(seed=42, **kwargs):
    try:
        s = int(seed)
        if s < 0 or s > 2**32-1: s = 42
    except (ValueError, TypeError, OverflowError):
        s = 42
    rng = np.random.default_rng(s)
    n = 47; parent = None
    ihv = globals().get('initial_h_values', None)
    if ihv is not None:
        try:
            p = np.array(ihv, dtype=np.float64).ravel()
            if 2 <= p.size < 1000 and np.all(np.isfinite(p)):
                if p.min() >= -1e-7 and p.max() <= 1.0+1e-7:
                    n = p.size
                    pr = project_capped_simplex(np.clip(p, 0, 1), n*0.5)
                    if pr is not None and check_feasible(pr, n): parent = pr
        except Exception: pass
    if parent is None: parent = np.full(n, 0.5)
    sm = n * 0.5
    P, max_gen, stall_lim, max_rest = get_params(n)
    pop = build_pop(rng, parent, n)
    scores = np.empty(P)
    for i in range(P):
        sc = exact_score(pop[i], n)
        scores[i] = sc if sc is not None else 1e30
    incumbent = parent.copy()
    inc_sc = exact_score(incumbent, n)
    if inc_sc is None: inc_sc = 1e30
    for i in range(P):
        if scores[i] < inc_sc: inc_sc = scores[i]; incumbent = pop[i].copy()
    mu_F = 0.5; mu_CR = 0.8; stall = 0; n_rest = 0
    for gen in range(max_gen):
        snap = pop.copy(); snap_sc = scores.copy()
        k_pb = max(2, math.ceil(0.20 * P))
        order = np.argsort(snap_sc); k_idx = order[:k_pb]
        perm = rng.permutation(P)
        new_pop = pop.copy(); new_sc = scores.copy()
        pend = []; ok_F = []; ok_CR = []; strong = False
        for ii in perm:
            i = int(ii)
            p_i = int(k_idx[rng.integers(0, len(k_idx))])
            if p_i == i: p_i = int(k_idx[rng.integers(0, len(k_idx))])
            pool1 = [j for j in range(P) if j != i and j != p_i]
            r1 = int(pool1[rng.integers(0, len(pool1))])
            pool2 = [j for j in range(P) if j != i and j != p_i and j != r1]
            r2 = int(pool2[rng.integers(0, len(pool2))])
            F = 0.5
            for _ in range(50):
                ft = mu_F + 0.1 * float(rng.standard_t(1))
                if np.isfinite(ft) and ft > 0:
                    F = float(np.clip(ft, 0.001, 1.0)); break
            CR = float(np.clip(rng.normal(mu_CR, 0.1), 0.0, 1.0))
            donor = snap[i] + F*(snap[p_i]-snap[i]) + F*(snap[r1]-snap[r2])
            trial = snap[i].copy()
            mask = rng.random(n) < CR
            if not mask.any(): mask[int(rng.integers(0, n))] = True
            trial[mask] = donor[mask]
            pr = project_capped_simplex(trial, sm)
            if pr is None or not check_feasible(pr, n): continue
            if np.max(np.abs(pr - snap[i])) <= 1e-15: continue
            sc = exact_score(pr, n)
            if sc is None: continue
            if sc < snap_sc[i] - 1e-15:
                new_pop[i] = pr; new_sc[i] = sc
                pend.append(snap[i].copy()); ok_F.append(F); ok_CR.append(CR)
                if sc < inc_sc: inc_sc = sc; incumbent = pr.copy(); strong = True
            elif abs(sc - snap_sc[i]) <= 1e-15 and np.max(np.abs(pr - snap[i])) > 1e-12:
                new_pop[i] = pr; new_sc[i] = sc; pend.append(snap[i].copy())
        pop = new_pop; scores = new_sc
        for a in pend:
            if check_feasible(a, n):
                pop[len(pop)-1+len(a)]  # no-op, just to avoid error
        if ok_F:
            c = 0.1
            sF2 = sum(f*f for f in ok_F); sF = sum(ok_F)
            if sF > 0: mu_F = (1-c)*mu_F + c*(sF2/sF)
            mu_CR = (1-c)*mu_CR + c*(sum(ok_CR)/len(ok_CR))
        if strong: stall = 0
        else: stall += 1
        if stall >= stall_lim and n_rest < max_rest:
            stall = 0; n_rest += 1
            m = max(2, P // 3)
            worst = np.argsort(-scores)
            rest_sigs = [0.05, 0.15, 0.35, 0.70]
            for wi in range(m):
                j = int(worst[wi])
                if wi == 0:
                    pop[j] = incumbent.copy()
                elif wi % 4 == 0:
                    cand = rng.beta(0.7, 0.7, n)
                    pr2 = project_capped_simplex(cand, sm)
                    if pr2 is not None and check_feasible(pr2, n): pop[j] = pr2
                else:
                    sig = rest_sigs[wi % 4]
                    cand = incumbent + sig * rng.standard_normal(n)
                    pr2 = project_capped_simplex(cand, sm)
                    if pr2 is not None and check_feasible(pr2, n): pop[j] = pr2
                sc2 = exact_score(pop[j], n)
                if sc2 is not None: scores[j] = sc2
                if sc2 is not None and sc2 < inc_sc: inc_sc = sc2; incumbent = pop[j].copy()
            mu_F = 0.5; mu_CR = 0.8
    final = incumbent.copy()
    pr = project_capped_simplex(final, sm)
    if pr is not None and check_feasible(pr, n): final = pr
    sc = exact_score(final, n)
    if sc is None or not np.isfinite(sc):
        final = parent; sc = exact_score(parent, n)
        if sc is None: final = np.full(n, 0.5); sc = exact_score(final, n)
    if sc is None: sc = 0.5
    return final, sc, n
