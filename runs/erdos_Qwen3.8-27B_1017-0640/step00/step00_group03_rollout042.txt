                    r = max(r / 2.0, min_radius)
                    fail_count += 1
                    if fail_count >= 6 or r <= min_radius:
                        break
                    continue
                old_y = y.copy()
                old_score = cur_score
                y = best_trial.copy()
                cur_score = best_trial_score
                fail_count = 0
                delta = y - old_y
                model_val = float(np.max(c + J.dot(delta)))
                pred = old_score - model_val
                ared = old_score - cur_score
                if pred > 1e-12:
                    rho = ared / pred
                else:
                    rho = 0.0
                step_inf = float(np.max(np.abs(delta)))
                if rho < 0.1:
                    r = max(r / 2.0, min_radius)
                elif rho > 0.75 and step_inf >= 0.8 * r:
                    r = min(1.5 * r, max_radius)
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_h = y.copy()
                if step_inf <= 1e-8:
                    break
        except Exception:
            continue
    if not check_feasible(best_h, n):
        fp = joint_project(best_h, n)
        if fp is not None:
            best_h = fp
        else:
            best_h = uniform.copy()
    c5 = exact_score(best_h)
    if not np.isfinite(c5):
        best_h = uniform.copy()
        c5 = exact_score(best_h)
    if parent_fb is not None:
        ps = exact_score(parent_fb)
        if np.isfinite(ps) and ps < c5 - 1e-12:
            best_h = parent_fb.copy()
            c5 = ps
    return best_h, c5, int(n)
