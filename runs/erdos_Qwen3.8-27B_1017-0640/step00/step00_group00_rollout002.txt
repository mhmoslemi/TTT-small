        return None
    h = np.array(res.x[:n], dtype=np.float64)
    if not np.all(np.isfinite(h)):
        return None
    h = _project(h, n)
    if _feasible(h, n):
        return h
    return None

def run(seed=42, **kwargs):
    n = 45
    if 'n_points' in kwargs:
        try:
            nn = int(kwargs['n_points'])
            if 1 <= nn <= 999:
                n = nn
        except (ValueError, TypeError):
            pass
    parent = None
    if 'initial_h_values' in globals():
        pv = globals()['initial_h_values']
        if pv is not None:
            pv = np.asarray(pv, dtype=np.float64)
            if pv.ndim == 1 and len(pv) == n and np.all(np.isfinite(pv)):
                parent = _project(pv, n)
            elif pv.ndim == 1 and len(pv) > 1 and np.all(np.isfinite(pv)):
                idx_old = np.linspace(0, 1, len(pv))
                idx_new = np.linspace(0, 1, n)
                parent = _project(np.interp(idx_new, idx_old, pv), n)
            elif pv.ndim == 1 and np.all(np.isfinite(pv)):
                parent = None
    if parent is None:
        parent = np.full(n, 0.5)

    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)

    if n <= 100:
        n_starts, maxiter, ftol, n_rounds = 10, 250, 1e-10, 2
        sigma_s = [0.03, 0.10, 0.25]
    elif n <= 300:
        n_starts, maxiter, ftol, n_rounds = 7, 120, 1e-10, 1
        sigma_s = [0.05, 0.18]
    else:
        n_starts, maxiter, ftol, n_rounds = 4, 60, 1e-10, 1
        sigma_s = [0.08]

    starts = _make_starts(parent, n, rng, n_starts)
    best_h = parent.copy()
    best_s = _score_exact(best_h)

    for x0 in starts:
        h_new = _slsqp_solve(x0, n, maxiter, ftol)
        if h_new is not None:
            s = _score_exact(h_new)
            if s < best_s - 1e-12:
                best_s, best_h = s, h_new

    for rnd in range(n_rounds):
        s_at_round_start = best_s
        for sig in sigma_s:
            pert = _project(best_h + sig * rng.standard_normal(n), n)
            h_new = _slsqp_solve(pert, n, maxiter, ftol)
            if h_new is not None:
                s = _score_exact(h_new)
                if s < best_s - 1e-12:
                    best_s, best_h = s, h_new
        if best_s >= s_at_round_start - 1e-9:
            break

    best_h = _project(best_h, n)
    if not _feasible(best_h, n):
        best_h = parent.copy() if _feasible(parent, n) else np.full(n, 0.5)
    c5 = _score_exact(best_h)
    if not np.isfinite(c5):
        best_h = np.full(n, 0.5)
        c5 = _score_exact(best_h)
    return best_h, c5, int(n)
