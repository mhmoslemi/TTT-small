            anchors.insert(0, parent)
        search_at(n0, anchors)

    # Doubled resolution
    if use_doubled and n0 <= 64:
        n1 = 2 * n0
        d_anchors = [np.repeat(fb, 2)]
        if parent is not None and check_feasible(parent):
            d_anchors.append(np.repeat(parent, 2))
        if incident_sc is not None and len(incident) == n0:
            d_anchors.append(np.repeat(incident, 2))
        search_at(n1, d_anchors)

    # Final validation
    h_out = incident
    sc_out = incident_sc
    if h_out is None or not check_feasible(h_out):
        h_out = fb.copy()
        sc_out = fb_sc
    if sc_out is None or not np.isfinite(sc_out):
        sc_out = exact_score(h_out)
        if sc_out is None:
            h_out = fb.copy()
            sc_out = exact_score(fb)
    h_out = np.array(h_out, dtype=np.float64)
    sc_out = exact_score(h_out)
    if sc_out is None:
        h_out = fb.copy()
        sc_out = exact_score(fb)
    return h_out, float(sc_out), int(len(h_out))
