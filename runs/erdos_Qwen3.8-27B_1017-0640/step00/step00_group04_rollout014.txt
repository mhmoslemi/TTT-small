            if bs < nat_sc - 1e-12:
                nat_sc, nat_h = bs, bh
            if bs < inc_sc - 1e-12 or (abs(bs - inc_sc) <= 1e-12 and len(bh) < len(inc_h)):
                inc_sc, inc_h = bs, bh.copy()

    if use_dbl and n0 <= 64:
        n1 = 2 * n0
        anc1 = [np.repeat(fb, 2), np.repeat(nat_h, 2)]
        st1 = build_starts(n1, anc1, rng, n_starts)
        for sh in st1:
            bh, bs = pg_search(sh, taus, max_it)
            if bs < inc_sc - 1e-12 or (abs(bs - inc_sc) <= 1e-12 and len(bh) < len(inc_h)):
                inc_sc, inc_h = bs, bh.copy()

    hf = inc_h.copy()
    nf = len(hf)
    if abs(float(np.sum(hf)) - nf / 2.0) > 1e-12 * max(1, nf):
        hf = project_capped(hf, nf)
    if hf is None or not feasible(hf, nf):
        hf = fb.copy()
        nf = len(hf)
    sf = exact_score(hf)
    if sf is None:
        hf = fb.copy()
        sf = exact_score(fb)
        nf = len(hf)
    return hf, float(sf), int(nf)
