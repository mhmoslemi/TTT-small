We have q_k(h) = dx * sum_i h[i+k](1-h[i]) for the overlap indices.

Using np.correlate(h, 1-h, mode='full'), the result has length 2n-1, indexed from lag -(n-1) to +(n-1).

For q_0 = dx * sum_i h[i](1-h[i]):
∂q_0/∂h[r] = dx * (1 - 2*h[r])

For q_p (p > 0): q_p = dx * sum_{i=0}^{n-1-p} h[i+p](1-h[i])
∂q_p/∂h[r]:
- If r appears as the second factor (h[i+p] where i+p=r, so i=r-p, valid when p≤r≤n-1): contribution is dx*(1-h[r-p])
- If r appears as the first factor (1-h[i] where i=r, valid when 0≤r≤n-1-p): contribution is dx*(-h[r+p])

So ∂q_p/∂h[r] = dx * (1_{p≤r≤n-1}(1-h[r-p]) - 1_{0≤r≤n-1-p}h[r+p])

For q_{-p} (p > 0): q_{-p} = dx * sum_{i=0}^{n-1-p} h[i](1-h[i+p])
∂q_{-p}/∂h[r]:
- If r appears as h[i] where i=r, valid when 0≤r≤n-1-p: contribution is dx*(1-h[r+p])
- If r appears as (1-h[i+p]) where i+p=r, so i=r-p, valid when p≤r≤n-1: contribution is dx*(-h[r-p])

So ∂q_{-p}/∂h[r] = dx * (1_{0≤r≤n-1-p}(1-h[r+p]) - 1_{p≤r≤n-1}h[r-p])

    else:
        iters = 12
    x = x0.copy()
    for tau in taus:
        alpha = 1.0
        for _ in range(iters):
            q = np.correlate(x, 1.0 - x, mode='full') * dx
            if not np.isfinite(q).all():
                return x
            Fx, w = compute_wh(q, tau)
            if Fx is None:
                return x
            g = compute_gradient(x, w, dx)
            if not np.isfinite(g).all():
                return x
            accepted = False
            for _ in range(20):
                y = project_to_k(x - alpha * g, m)
                if y is None:
                    alpha *= 0.5
                    continue
                d = y - x
                gd = float(np.dot(g, d))
                if gd >= 0:
                    alpha *= 0.5
                    continue
                qy = np.correlate(y, 1.0 - y, mode='full') * dx
                Fy, _ = compute_wh(qy, tau)
                if Fy is None:
                    alpha *= 0.5
                    continue
                if Fy <= Fx + 1e-4 * gd + 1e-14:
                    if abs(Fy - Fx) <= 1e-12 * (1.0 + abs(Fx)):
                        return x
                    x = y
                    alpha = min(2.0, 1.25 * alpha)
                    accepted = True
                    break
                alpha *= 0.5
            if not accepted:
                break
            if float(np.max(np.abs(d))) <= 1e-12:
                break
    return x

def make_starts(n, m, parent, rng):
    starts = []
    fl = int(math.floor(m))
    frac = m - fl
    blk = np.zeros(n)
    blk[:fl] = 1.0
    if frac > 0 and fl < n:
        blk[fl] = frac
    cos_i = np.arange(n, dtype=np.float64)
    if parent is not None:
        starts.append(parent.copy())
        starts.append(1.0 - parent.copy())
        starts.append(parent[::-1].copy())
        starts.append(np.roll(parent, n // 3))
        starts.append(np.full(n, m / n))
        starts.append(blk)
        starts.append(blk[::-1].copy())
        starts.append(rng.random(n))
        ph = rng.uniform(0, 2 * math.pi)
        starts.append(0.5 + 0.35 * np.cos(2 * math.pi * cos_i / n + ph))
        starts.append(parent + 0.05 * rng.standard_normal(n))
        starts.append(parent + 0.15 * rng.standard_normal(n))
    else:
        starts.append(np.full(n, m / n))
        starts.append(blk)
        starts.append(blk[::-1].copy())
        for _ in range(3):
            starts.append(rng.random(n))
        ph1 = rng.uniform(0, 2 * math.pi)
        starts.append(0.5 + 0.35 * np.cos(2 * math.pi * cos_i / n + ph1))
        ph2 = rng.uniform(0, 2 * math.pi)
        starts.append(0.5 + 0.35 * np.cos(4 * math.pi * cos_i / n + ph2))
    projected = []
    for s in starts:
        p = project_to_k(np.asarray(s, dtype=np.float64), m)
        if p is None:
            continue
        dup = False
        for sp in projected:
            if float(np.max(np.abs(p - sp))) <= 1e-10:
                dup = True
                break
        if not dup:
            projected.append(p)
    if n <= 64:
        mx = 10
    elif n <= 160:
        mx = 6
    elif n <= 400:
        mx = 4
    else:
        mx = 2
    return projected[:mx]

def run(seed=42, **kwargs):
    n = 45
    parent = None
    use_kw_n = False
    if 'n_points' in kwargs:
        npk = kwargs['n_points']
        if not isinstance(npk, bool) and isinstance(npk, (int, float)):
            if npk == int(npk) and 1 <= int(npk) <= 999:
                n = int(npk)
                use_kw_n = True
    try:
        iv = initial_h_values
        if iv is not None:
            iv = np.asarray(iv, dtype=np.float64)
            if iv.ndim == 1 and len(iv) > 0 and np.isfinite(iv).all() and 1 <= len(iv) <= 999:
                if not use_kw_n:
                    n = len(iv)
                parent = iv.copy()
    except Exception:
        pass
    m = n / 2.0
    dx = 2.0 / n
    if parent is not None and len(parent) != n:
        xp = np.linspace(0, 1, len(parent))
        xn = np.linspace(0, 1, n)
        parent = np.interp(xn, xp, parent)
    rng = np.random.default_rng(seed)
    fallback = np.full(n, m / n)
    best = fallback.copy()
    best_score = exact_score(best, n)
    if best_score is None:
        best_score = 1e9
    if parent is not None:
        pv = project_to_k(parent.copy(), m)
        if pv is not None:
            parent = pv
        else:
            parent = None
    starts = make_starts(n, m, parent, rng)
    for s in starts:
        sf = project_to_k(s.copy(), m)
        if sf is None:
            continue
        sc = exact_score(sf, n)
        if sc is not None and sc < best_score - 1e-12:
            best = sf.copy()
            best_score = sc
        x_opt = local_optimize(sf, n, m, dx)
        so = exact_score(x_opt, n)
        if so is not None and so < best_score - 1e-12:
            best = x_opt.copy()
            best_score = so
    if n <= 64:
        rounds, sigmas = 2, [0.05, 0.18]
    elif n <= 160:
        rounds, sigmas = 1, [0.08, 0.20]
    else:
        rounds, sigmas = 1, [0.10]
    for rd in range(rounds):
        sc_before = best_score
        for sig in sigmas:
            pert = project_to_k(best + sig * rng.standard_normal(n), m)
            if pert is None:
                continue
            sc = exact_score(pert, n)
            if sc is not None and sc < best_score - 1e-12:
                best = pert.copy()
                best_score = sc
            xo = local_optimize(pert, n, m, dx)
            so = exact_score(xo, n)
            if so is not None and so < best_score - 1e-12:
                best = xo.copy()
                best_score = so
        if best_score >= sc_before - 1e-9:
            break
    if best.shape != (n,) or not np.isfinite(best).all() or np.any(best < -1e-12) or np.any(best > 1.0 + 1e-12) or abs(float(np.sum(best)) - m) > 1e-9:
        best = project_to_k(best, m)
        if best is None:
            best = fallback.copy()
    c5 = exact_score(best, n)
    if c5 is None:
        best = fallback.copy()
        c5 = exact_score(best, n)
    if c5 is None:
        c5 = 0.5
    return (best.astype(np.float64), c5, int(n))
