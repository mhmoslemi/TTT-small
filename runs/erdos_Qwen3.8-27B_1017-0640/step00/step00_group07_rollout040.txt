        maxiter = int(kwargs.get('maxiter', 100)); maxiter = max(40, min(250, maxiter))
    except Exception:
        pass
    try:
        ftol = float(kwargs.get('ftol', 1e-9))
        if not np.isfinite(ftol) or ftol <= 0:
            ftol = 1e-9
        ftol = max(1e-12, min(1e-6, ftol))
    except Exception:
        pass

    n, base, parent_fb = 80, np.full(80, 0.5), np.full(80, 0.5)
    try:
        ph = np.array(initial_h_values, dtype=np.float64)
        if ph.ndim == 1 and 2 <= ph.shape[0] <= 999:
            n = ph.shape[0]
            if np.all(np.isfinite(ph)):
                if is_feasible(ph, n):
                    base = ph.copy(); parent_fb = ph.copy()
                else:
                    base = project_capped(ph, n); parent_fb = base.copy()
            else:
                base = np.full(n, 0.5); parent_fb = base.copy()
    except Exception:
        pass

    starts = generate_starts(base, n, rng, n_starts)
    best_h, best_score = None, float('inf')
    for c in starts:
        sc = exact_score(c, n)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc; best_h = c.copy()
    psc = exact_score(parent_fb, n)
    if np.isfinite(psc) and psc < best_score:
        best_score = psc; best_h = parent_fb.copy()
    if best_h is None:
        best_h = np.full(n, 0.5); best_score = exact_score(best_h, n)

    if n <= 160:
        for start in starts:
            cur = start.copy()
            cur_sc = exact_score(cur, n)
            if not np.isfinite(cur_sc):
                continue
            for _ in range(outer_iter):
                if best_score <= 0.3808:
                    break
                r = cur.copy()
                Q0, _ = majorizer_eval(r, r, n)
                t0 = float(np.max(Q0)) + 1e-8
                z0 = np.concatenate([r, [t0]])
                bounds = [(0.0, 1.0)] * n + [(0.0, None)]
                constraints = [
                    {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac, 'args': (r, n)},
                    {'type': 'ineq', 'fun': slsqp_ineq, 'jac': slsqp_ineq_jac, 'args': (r, n)},
                ]
                try:
                    res = minimize(slsqp_objective, z0, jac=slsqp_grad, args=(r, n),
                                   method='SLSQP', bounds=bounds, constraints=constraints,
                                   options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
                except Exception:
                    break
                if res is None or not hasattr(res, 'x') or res.x.shape[0] != n + 1:
                    break
                if not np.all(np.isfinite(res.x)):
                    break
                prop = res.x[:n]
                tested = []
                for alpha in [1.0, 0.5, 0.25, 0.125]:
                    cand = project_capped(r + alpha * (prop - r), n)
                    tested.append(cand)
                tested.append(project_capped(prop, n))
                best_local, best_local_sc = None, float('inf')
                seen = []
                for cand in tested:
                    dup = any(np.max(np.abs(cand - s)) < 1e-12 for s in seen)
                    if dup:
                        continue
                    seen.append(cand)
                    sc = exact_score(cand, n)
                    if np.isfinite(sc) and sc < best_local_sc:
                        best_local_sc = sc; best_local = cand.copy()
                if best_local is not None and best_local_sc < cur_sc - 1e-12:
                    cur = best_local; cur_sc = best_local_sc
                else:
                    break
                if cur_sc < best_score - 1e-12:
                    best_score = cur_sc; best_h = cur.copy()
                if best_score <= 0.3808:
                    break
    final_h = project_capped(best_h, n)
    final_sc = exact_score(final_h, n)
    fb_sc = exact_score(project_capped(parent_fb, n), n)
    if np.isfinite(fb_sc) and fb_sc < final_sc:
        final_h = project_capped(parent_fb, n); final_sc = fb_sc
    if not is_feasible(final_h, n):
        final_h = np.full(n, 0.5); final_sc = exact_score(final_h, n)
    final_sc = exact_score(final_h, n)
    return (final_h, final_sc, int(n))
