            idx = np.arange(lo2, hi2)
            J[j, idx] -= dx * h[idx + k]
    return J


def _obj(z, N):
    return z[N]


def _obj_grad(z, N):
    g = np.zeros(N + 1)
    g[N] = 1.0
    return g


def _eq_fun(z, N, m):
    return z[:N].sum() - m


def _eq_jac(z, N, m):
    g = np.zeros(N + 1)
    g[:N] = 1.0
    return g


def _ueq_fun(z, N, dx):
    h = z[:N]
    t = z[N]
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - q


def _ueq_jac(z, N, dx):
    h = z[:N]
    J = _corr_grad(h, dx)
    out = np.zeros((2 * N - 1, N + 1))
    out[:, :N] = -J
    out[:, N] = 1.0
    return out


def _make_starts(N, m, dx, rng, parent):
    starts = []
    base = parent.copy()
    starts.append(_project_box_sum(base, m))
    for s in (0.03, 0.08, 0.18):
        starts.append(_project_box_sum(base + rng.normal(0, s, N), m))
    starts.append(_project_box_sum(rng.uniform(0, 1, N), m))
    starts.append(_project_box_sum(rng.beta(0.7, 0.7, N), m))
    i = np.arange(N) + 0.5
    starts.append(_project_box_sum(0.5 + 0.45 * np.cos(2 * np.pi * i / N), m))
    starts.append(_project_box_sum(0.5 + 0.45 * np.cos(4 * np.pi * i / N + np.pi / 3.0), m))
    return starts


def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = None
    try:
        pv = np.array(initial_h_values, dtype=np.float64).ravel().copy()
        if len(pv) > 0 and np.all(np.isfinite(pv)):
            parent = pv
    except Exception:
        pass

    if parent is not None and len(parent) > 0:
        N = len(parent)
    else:
        N = 70
        parent = np.full(N, 0.5)

    if N <= 0 or N > 999:
        h = np.full(70, 0.5)
        dx70 = 2.0 / 70
        return h, _score(h, dx70), 70

    dx = 2.0 / N
    m = N / 2.0
    h_fallback = np.full(N, 0.5)
    p_parent = _project_box_sum(parent, m)

    if N <= 256:
        starts = _make_starts(N, m, dx, rng, p_parent)
    else:
        starts = [p_parent,
                  _project_box_sum(p_parent + rng.normal(0, 0.08, N), m),
                  _project_box_sum(rng.uniform(0, 1, N), m),
                  _project_box_sum(rng.beta(0.7, 0.7, N), m)]

    best_h = p_parent.copy()
    best_c = _score(best_h, dx)
    c_fb = _score(h_fallback, dx)
    if c_fb < best_c:
        best_h = h_fallback.copy()
        best_c = c_fb

    for h0 in starts:
        if not np.all(np.isfinite(h0)):
            continue
        c0 = _score(h0, dx)
        if c0 < best_c - 1e-12:
            best_h = h0.copy()
            best_c = c0
        t0 = min(max(c0, 0.0), 2.0)
        z0 = np.concatenate([h0, [t0]])
        try:
            res = minimize(
                _obj, z0, args=(N,),
                jac=_obj_grad,
                method='SLSQP',
                bounds=[(0.0, 1.0)] * N + [(0.0, 2.0)],
                constraints=[
                    {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (N, m)},
                    {'type': 'ineq', 'fun': _ueq_fun, 'jac': _ueq_jac, 'args': (N, dx)}
                ],
                options={'maxiter': 250, 'ftol': 1e-10, 'disp': False}
            )
        except Exception:
            continue
        if res.x is None or len(res.x) != N + 1:
            continue
        h_new = _project_box_sum(res.x[:N], m)
        if not np.all(np.isfinite(h_new)):
            continue
        c_new = _score(h_new, dx)
        if c_new < best_c - 1e-12:
            best_h = h_new.copy()
            best_c = c_new

    best_h = _project_box_sum(best_h, m)
    if not (np.all(best_h >= -1e-10) and np.all(best_h <= 1.0 + 1e-10)
            and abs(best_h.sum() - m) < 1e-8):
        best_h = h_fallback.copy()
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c = _score(best_h, dx)
    return best_h, final_c, N
