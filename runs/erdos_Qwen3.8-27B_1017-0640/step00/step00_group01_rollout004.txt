        result = solve_start(s, n, maxiter)
        if result is not None:
            sc2 = exact_score(result)
            if math.isfinite(sc2) and sc2 < best_score - 1e-12:
                best_h, best_score = result.copy(), sc2
    for ns in [0.02, 0.06][:n_adaptive]:
        proj = project_capped_simplex(best_h + ns * rng.standard_normal(n), n / 2.0)
        if proj is None:
            continue
        sc = exact_score(proj)
        if sc < best_score - 1e-12:
            best_h, best_score = proj.copy(), sc
        result = solve_start(proj, n, maxiter)
        if result is not None:
            sc2 = exact_score(result)
            if math.isfinite(sc2) and sc2 < best_score - 1e-12:
                best_h, best_score = result.copy(), sc2
    if not check_feasible(best_h, n):
        best_h = parent_proj.copy()
        best_score = exact_score(best_h)
    if not check_feasible(best_h, n):
        best_h = project_capped_simplex(np.full(n, 0.5), n / 2.0)
        best_score = exact_score(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    c5 = exact_score(best_h)
    if not math.isfinite(c5):
        best_h = np.full(n, 0.5)
        c5 = exact_score(best_h)
    return (best_h, float(c5), int(n))
