    t0 = min(1.0, exact_score(h0) + 1e-9)

    cons = [

        {'type': 'ineq', 'fun': lag_ineq, 'jac': lag_ineq_jac}

        x = np.clip(v - mid, 0.0, 1.0)
        total = math.fsum(x)
        if total > s:
            lo = mid
        else:
            hi = mid
    lm = 0.5 * (lo + hi)
    x = np.clip(v - lm, 0.0, 1.0)
    resid = s - math.fsum(x)
    for i in range(n):
        if abs(resid) < 5e-15:
            break
        if resid > 0:
            cap = 1.0 - x[i]
            adj = min(resid, cap)
            if adj > 0:
                x[i] += adj
                resid -= adj
        else:
            adj = min(-resid, x[i])
            if adj > 0:
                x[i] -= adj
                resid += adj
    if abs(math.fsum(x) - s) > 1e-9 or float(np.min(x)) < -1e-12 or float(np.max(x)) > 1.0 + 1e-12:
        return None
    return x


def is_feasible(h, n):
    if h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if float(np.min(h)) < -1e-12 or float(np.max(h)) > 1.0 + 1e-12:
        return False
    if abs(math.fsum(h) - n / 2.0) > 1e-9:
        return False
    return True


def exact_score(h):
    n = len(h)
    if n < 2 or not np.all(np.isfinite(h)):
        return float('inf')
    vals = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
    sc = float(np.max(vals))
    return sc if math.isfinite(sc) else float('inf')


def lag_q_and_jac(h):
    n = len(h)
    n_lags = 2 * n - 1
    q = np.zeros(n_lags)
    J = np.zeros((n_lags, n))
    for d in range(-(n - 1), n):
        idx = d + (n - 1)
        if d >= 0:
            iStart, iEnd = 0, n - d
        else:
            iStart, iEnd = -d, n
        for i in range(iStart, iEnd):
            j = i + d
            q[idx] += h[i] * (1.0 - h[j])
            J[idx, i] += 1.0 - h[j]
            J[idx, j] -= h[i]
    return q, J


def epigraph_obj(z):
    return float(z[-1])


def epigraph_obj_grad(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g


def mass_eq(z):
    return float(np.sum(z[:-1])) - (len(z) - 1) / 2.0


def mass_eq_grad(z):
    g = np.ones(len(z))
    g[-1] = 0.0
    return g


def lag_ineq(z):
    n = len(z) - 1
    h = z[:n]
    t = z[-1]
    q, _ = lag_q_and_jac(h)
    return t - (2.0 / n) * q


def lag_ineq_jac(z):
    n = len(z) - 1
    h = z[:n]
    _, Jq = lag_q_and_jac(h)
    n_lags = 2 * n - 1
    J = np.zeros((n_lags, n + 1))
    J[:, :n] = -(2.0 / n) * Jq
    J[:, n] = 1.0
    return J


def solve_start(h0, n, maxiter=300, ftol=1e-10):
    h0 = capped_simplex_project(h0.astype(np.float64).copy(), n)
    if h0 is None:
        return None
    t0 = min(1.0, exact_score(h0) + 1e-9)
    z0 = np.empty(n + 1)
    z0[:n] = h0
    z0[n] = t0
    bounds = [(0.0, 1.0)] * (n + 1)
    cons = [
        {'type': 'eq', 'fun': mass_eq, 'jac': mass_eq_grad},
        {'type': 'ineq', 'fun': lag_ineq, 'jac': lag_ineq_jac},
    ]
    try:
        res = minimize(epigraph_obj, z0, method='SLSQP', jac=epigraph_obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'ftol': ftol, 'maxiter': maxiter, 'disp': False})
        h_res = res.x[:n]
        if h_res.shape == (n,) and np.all(np.isfinite(h_res)):
            h_rep = capped_simplex_project(h_res.astype(np.float64), n)
            if h_rep is not None:
                return h_rep
    except Exception:
        pass
    return h0


def run(seed=42, **kwargs):
    try:
        seed = int(seed)
        if seed < 0:
            seed = 42
    except (ValueError, TypeError):
        seed = 42
    rng = np.random.default_rng(seed)

    n = 47
    parent_h = None
    try:
        gh = globals().get('initial_h_values', None)
        if gh is not None:
            arr = np.array(gh, dtype=np.float64).ravel()
            if 2 <= len(arr) <= 999 and np.all(np.isfinite(arr)):
                if float(np.min(arr)) >= -1e-7 and float(np.max(arr)) <= 1.0 + 1e-7:
                    n = len(arr)
                    parent_h = capped_simplex_project(arr, n)
    except Exception:
        parent_h = None

    if parent_h is None:
        n = 47
        parent_h = np.full(n, 0.5)

    best_h = parent_h.copy()
    best_score = exact_score(best_h)

    if n <= 64:
        n_starts, n_adaptive, maxiter = 8, 2, 300
    elif n <= 128:
        n_starts, n_adaptive, maxiter = 6, 1, 180
    else:
        n_starts, n_adaptive, maxiter = 3, 0, 80

    base = parent_h.copy()
    starts = [base.copy()]

    c05 = np.full(n, 0.5)
    if not np.allclose(base, c05, atol=1e-10):
        starts.append(c05.copy())

    for sigma in [0.03, 0.10, 0.25]:
        proj = capped_simplex_project(base + rng.standard_normal(n) * sigma, n)
        if proj is not None:
            starts.append(proj)

    n_unif = 1 if parent_h is not None else 2
    for _ in range(n_unif):
        proj = capped_simplex_project(rng.random(n), n)
        if proj is not None:
            starts.append(proj)

    idx = np.arange(n, dtype=np.float64)
    for qq in [1, 2]:
        phi = rng.uniform(0.0, 2.0 * math.pi)
        sv = 0.5 + 0.25 * np.sin(2.0 * math.pi * qq * (idx + 0.5) / n + phi)
        proj = capped_simplex_project(sv, n)
        if proj is not None:
            starts.append(proj)

    for st in starts[:n_starts]:
        if is_feasible(st, n):
            sc = exact_score(st)
            if math.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_h = st.copy()
        sol = solve_start(st, n, maxiter=maxiter)
        if sol is not None and is_feasible(sol, n):
            sc = exact_score(sol)
            if math.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_h = sol.copy()

    for sigma in [0.02, 0.06][:n_adaptive]:
        proj = capped_simplex_project(best_h + rng.standard_normal(n) * sigma, n)
        if proj is None:
            continue
        if is_feasible(proj, n):
            sc = exact_score(proj)
            if math.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_h = proj.copy()
        sol = solve_start(proj, n, maxiter=maxiter)
        if sol is not None and is_feasible(sol, n):
            sc = exact_score(sol)
            if math.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_h = sol.copy()

    if not is_feasible(best_h, n):
        best_h = parent_h.copy()
    final_score = exact_score(best_h)
    if not math.isfinite(final_score):
        best_h = parent_h.copy()
        final_score = exact_score(best_h)
    if not math.isfinite(final_score):
        best_h = np.full(n, 0.5)
        final_score = exact_score(best_h)

    if not is_feasible(best_h, n):
        rep = capped_simplex_project(best_h, n)
        if rep is not None:
            best_h = rep
    final_score = exact_score(best_h)
    if not math.isfinite(final_score):
        final_score = 1.0

    return (best_h, float(final_score), int(n))
