                best_h = h_new.copy()

    if n <= 128:
        for sd in [0.03, 0.08]:
            h_pert = capped_simplex_project(best_h + sd * rng.standard_normal(n), n)
            zx = optimize_start(h_pert, n)
            if zx is not None and len(zx) == n + 1 and np.all(np.isfinite(zx[:n])):
                h_new = capped_simplex_project(zx[:n], n)
                c_new = exact_score(h_new)
                if c_new < best_c - 1e-12:
                    best_c = c_new
                    best_h = h_new.copy()

    best_h = capped_simplex_project(best_h, n)
    final_c = exact_score(best_h)
    if not np.isfinite(final_c):
        if parent is not None:
            best_h = capped_simplex_project(parent, n)
            final_c = exact_score(best_h)
        else:
            best_h = np.full(n, 0.5, dtype=np.float64)
            final_c = exact_score(best_h)
    return best_h, final_c, n
