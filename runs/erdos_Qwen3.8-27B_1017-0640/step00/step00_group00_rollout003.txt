import numpy as np
from scipy.optimize import minimize

def project_simplex(y, n):
    m = n / 2.0
    y = np.clip(np.asarray(y, dtype=np.float64), -1e6, 1e6).copy()
    lo, hi = np.min(y) - 1.0, np.max(y)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(y - mid, 0.0, 1.0)) > m:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    r = m - np.sum(x)
    for i in range(n):
        if r > 1e-15:
            d = min(r, 1.0 - x[i]); x[i] += d; r -= d
        elif r < -1e-15:
            d = min(-r, x[i]); x[i] -= d; r += d
    x = np.clip(x, 0.0, 1.0)
    if not np.isfinite(x).all() or abs(np.sum(x) - m) > 1e-7:
        return np.full(n, 0.5)
    return x

def is_feasible(h, n):
    return (h.shape == (n,) and np.isfinite(h).all()
            and np.all(h >= -1e-10) and np.all(h <= 1.0 + 1e-10)
            and abs(np.sum(h) - n / 2.0) <= 1e-9)

def exact_score(h, n):
    v = np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)
    return float(v) if np.isfinite(v) else np.inf

def _obj(z):
    return z[-1]

def _obj_jac(z):
    g = np.zeros(len(z)); g[-1] = 1.0; return g

def _eq(z):
    n = len(z) - 1
    return np.sum(z[:n]) - n / 2.0

def _eq_jac(z):
    n = len(z) - 1
    g = np.ones(n + 1); g[-1] = 0.0; return g

def _ineq(z):
    n = len(z) - 1
    h = z[:n]
    return z[-1] - np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _ineq_jac(z):
    n = len(z) - 1
    h = z[:n]
    dx = 2.0 / n
    M = np.zeros((2 * n - 1, n))
    M[n - 1, :] = dx * (1.0 - 2.0 * h)
    for p in range(1, n):
        M[n - 1 + p, p:] += dx * (1.0 - h[:n - p])
        M[n - 1 + p, :n - p] -= dx * h[p:]
        M[n - 1 - p, :n - p] += dx * (1.0 - h[p:])
        M[n - 1 - p, p:] -= dx * h[:n - p]
    J = np.zeros((2 * n - 1, n + 1))
    J[:, :n] = M
    J[:, n] = 1.0
    return J

def _block(n):
    m = n / 2.0
    h = np.zeros(n)
    k = int(m)
    h[:k] = 1.0
    if m - k > 1e-12:
        h[k] = m - k
    return h

def _solve(x0, n, maxiter, ftol):
    x0 = project_simplex(x0, n)
    if not is_feasible(x0, n):
        return None
    s0 = exact_score(x0, n)
    z0 = np.append(x0, min(1.0, s0 + 1e-9))
    bounds = [(0.0, 1.0)] * (n + 1)
    cons = [{'type': 'eq', 'fun': _eq, 'jac': _eq_jac},
            {'type': 'ineq', 'fun': _ineq, 'jac': _ineq_jac}]
    try:
        res = minimize(_obj, z0, method='SLSQP', jac=_obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
        if res is None or not hasattr(res, 'x') or len(res.x) != n + 1:
            return None
        x = project_simplex(res.x[:n], n)
        if is_feasible(x, n):
            return x
    except Exception:
        pass
    return None

def _interp_to_n(arr, n):
    old = arr.size
    if old == n:
        return arr.copy()
    t_old = np.linspace(0.0, 1.0, old)
    t_new = np.linspace(0.0, 1.0, n)
    return np.interp(t_new, t_old, arr)
