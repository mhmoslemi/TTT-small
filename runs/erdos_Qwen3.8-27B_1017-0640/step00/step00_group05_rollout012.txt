        cs = 0.5 + 0.45 * np.cos(2.0 * np.pi * 2.0 * (idx + 0.5) / N + 0.37)
        pr = project_simplex(cs, m)
        if pr is not None:
            starts.append(pr)
    else:
        pr = project_simplex(parent_proj + rng.standard_normal(N) * 0.08, m)
        if pr is not None:
            starts.append(pr)
        uni = rng.uniform(0, 1, N)
        pr = project_simplex(uni, m)
        if pr is not None:
            starts.append(pr)
        idx = np.arange(N)
        cs = 0.5 + 0.45 * np.cos(2.0 * np.pi * 2.0 * (idx + 0.5) / N + 0.37)
        pr = project_simplex(cs, m)
        if pr is not None:
            starts.append(pr)
    uniq = []
    for s in starts:
        if not check_feasible(s, N):
            continue
        dup = False
        for u in uniq:
            if float(np.max(np.abs(s - u))) <= 1e-12:
                dup = True
                break
        if not dup:
            uniq.append(s)
    best_h = neutral.copy()
    best_c = np.inf
    for s in uniq:
        c = exact_score(s, N)
        if c is not None and c < best_c:
            best_c = c
            best_h = s.copy()
    if N <= 999:
        max_it = 45 if N <= 256 else 15
        for s in uniq:
            h_out, c_out = lp_search(s, N, 0.5, max_it)
            if c_out is not None and c_out < best_c - 1e-12:
                best_c = c_out
                best_h = h_out.copy()
        if N <= 256:
            h_pol, c_pol = lp_search(best_h, N, 0.1, 20)
            if c_pol is not None and c_pol < best_c - 1e-12:
                best_c = c_pol
                best_h = h_pol.copy()
    final_h = project_simplex(best_h, m)
    if final_h is None:
        final_h = parent_proj.copy()
    c_final = exact_score(final_h, N)
    if c_final is None:
        final_h = neutral.copy()
        c_final = exact_score(final_h, N)
    if c_final is None:
        c_final = 0.5
    return final_h, float(c_final), N
