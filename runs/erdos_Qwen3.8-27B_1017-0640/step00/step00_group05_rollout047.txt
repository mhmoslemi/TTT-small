            np2 = list(elites)
            sd = 0.12 if restarts == 1 else 0.24
            att = 0
            while len(np2) < P and att < 10 * P:
                att += 1
                k = att % 3
                if k == 0:
                    raw = incumbent + sd * rng.standard_normal(N)
                elif k == 1:
                    raw = rng.random(N)
                else:
                    raw = rng.beta(0.7, 0.7, N)
                p2 = _project(raw, N, m)
                if p2 is not None and not any(_is_dup(p2, e) for e in np2):
                    np2.append(p2)
            while len(np2) < P:
                np2.append(np2[0].copy())
            pop = np2[:P]
            scores = np.array([(_score(v, N) if _score(v, N) is not None else 1e10) for v in pop], dtype=np.float64)
            for i in range(P):
                if scores[i] < inc_score:
                    inc_score = scores[i]
                    incumbent = pop[i].copy()
    fp = _project(incumbent, N, m)
    best_h = fp if fp is not None else incumbent
    best_s = _score(best_h, N)
    s_n = _score(neutral, N)
    s_p = _score(parent.copy(), N) if parent is not None else None
    if s_n is not None and (best_s is None or s_n < best_s):
        best_h, best_s = neutral, s_n
    if s_p is not None and (best_s is None or s_p < best_s):
        best_h, best_s = parent.copy(), s_p
    if best_s is None:
        best_h, best_s = neutral, _score(neutral, N)
    return (np.asarray(best_h, dtype=np.float64), float(best_s), int(N))
