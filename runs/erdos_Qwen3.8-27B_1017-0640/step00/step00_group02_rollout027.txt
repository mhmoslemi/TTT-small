def smooth_eval(h, n, tau, with_grad):
    dx = 2.0 / n
    q = np.correlate(h, 1.0 - h, mode='full')

        return None, None
    F = cmax + tau * np.log(s)
    if not with_grad:
        return F, None
    w = exp_u / s
