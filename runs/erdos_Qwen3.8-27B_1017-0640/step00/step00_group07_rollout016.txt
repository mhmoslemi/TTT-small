def project_capped(y, n):
    m = n / 2.0
    if len(y) != n or not np.all(np.isfinite(y)) or np.any(np.abs(y) > 1e12):
        return np.full(n, 0.5)
    lo, hi = np.min(y) - 1.0, np.max(y)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > m:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    p = np.clip(y - lam, 0.0, 1.0)
    resid = m - np.sum(p)
    for _ in range(n + 2):
        if abs(resid) <= 1e-12 * max(1, n):
            break
        if resid > 0:
            idx = np.where(p < 1.0 - 1e-15)[0]
        else:
            idx = np.where(p > 1e-15)[0]

            break
        per = resid / len(idx)
        for i in idx:
            if resid > 0:
                p[i] = min(1.0, p[i] + per)
            else:
                p[i] = max(0.0, p[i] + per)

def lag_values(h):
    n = len(h)
    dx = 2.0 / n
    q = np.empty(2 * n - 1)
    q[0] = dx * np.dot(h, 1.0 - h)  # zero lag
    for s in range(1, n):
        q[2 * s] = dx * np.dot(h[:n-s], 1.0 - h[s:])  # A_s
        q[2 * s - 1] = dx * np.dot(h[s:], 1.0 - h[:n-s])  # B_s
    return q

def smooth_phi(q, tau):
    qm = np.max(q)
    e = np.exp((q - qm) / tau)
    return qm + tau * np.log(np.sum(e))

def smooth_grad(q, G, tau):
    qm = np.max(q)
    e = np.exp((q - qm) / tau)
    w = e / np.sum(e)
    phi = qm + tau * np.log(np.sum(e))
    return phi, G.T @ w

def build_lag_jacobian(h):
    n = len(h)
    dx = 2.0 / n
    nlags = 2 * n - 1
    G = np.zeros((nlags, n))
    # Row 0: zero lag
    G[0] = dx * (1.0 - 2.0 * h)
    for s in range(1, n):
        # Row 2s: A_s = dx * sum_{i=0}^{n-s-1} h[i](1-h[i+s])
        # d/dh[j]: +dx*(1-h[j+s]) for j in [0, n-s-1], -dx*h[j-s] for j in [s, n-1]
        row_a = 2 * s
        if n - s > 0:

        if s < n:
            G[row_a, s:] -= dx * h[:n-s]
        # Row 2s-1: B_s = dx * sum_{i=0}^{n-s-1} h[i+s](1-h[i])

        if n - s > 0:

    return G

import numpy as np

def project_capped(y, n):
    m = n / 2.0
    y = np.asarray(y, dtype=np.float64)
    if y.shape[0] != n or not np.all(np.isfinite(y)) or np.any(np.abs(y) > 1e12):
        return np.full(n, 0.5)
    lo, hi = np.min(y) - 1.0, np.max(y)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(y - mid, 0.0, 1.0)) > m:
            lo = mid
        else:
            hi = mid
    p = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    resid = m - np.sum(p)
    for _ in range(n + 2):
        if abs(resid) <= 1e-12 * max(1, n):
            break
        idx = np.where(p < 1.0 - 1e-14)[0] if resid > 0 else np.where(p > 1e-14)[0]
        if len(idx) == 0:
            break
        per = resid / len(idx)
        p[idx] = np.clip(p[idx] + per, 0.0, 1.0)
        resid = m - np.sum(p)
    if not np.all(np.isfinite(p)) or np.any(p < -1e-14) or np.any(p > 1.0 + 1e-14):
        return np.full(n, 0.5)
    p = np.clip(p, 0.0, 1.0)
    r = m - np.sum(p)
    if abs(r) > 1e-14:
        idx = np.where(p < 0.999)[0] if r > 0 else np.where(p > 0.001)[0]
        if len(idx) > 0:
            p[idx[0]] = np.clip(p[idx[0]] + r, 0.0, 1.0)
    if abs(np.sum(p) - m) > 1e-10 * max(1, n):
        return np.full(n, 0.5)
    return p

def exact_score(h):
    n = len(h)
    if n < 2 or not np.all(np.isfinite(h)) or np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lag_values(h):
    n = len(h)
    dx = 2.0 / n
    q = np.empty(2 * n - 1)
    q[0] = dx * np.dot(h, 1.0 - h)
    for s in range(1, n):
        q[2 * s] = dx * np.dot(h[:n - s], 1.0 - h[s:])
        q[2 * s - 1] = dx * np.dot(h[s:], 1.0 - h[:n - s])
    return q

def smooth_phi(q, tau):
    qm = np.max(q)
    e = np.exp((q - qm) / tau)
    return qm + tau * np.log(np.sum(e))

def smooth_grad(q, G, tau):
    qm = np.max(q)
    e = np.exp((q - qm) / tau)
    w = e / np.sum(e)
    return qm + tau * np.log(np.sum(e)), G.T @ w

def build_lag_jacobian(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    G[0] = dx * (1.0 - 2.0 * h)
    for s in range(1, n):
        ra, rb = 2 * s, 2 * s - 1
        G[ra, :n - s] += dx * (1.0 - h[s:])
        G[ra, s:] -= dx * h[:n - s]
        G[rb, s:] += dx * (1.0 - h[:n - s])
        G[rb, :n - s] -= dx * h[s:]
    return G

def make_starts(base, n, rng, n_starts):
    cands = [base.copy()]
    for sd in [0.02, 0.08, 0.20]:
        cands.append(base + sd * rng.standard_normal(n))
    cands.append(rng.random(n))
    K = min(6, (n - 1) // 2)
    y = np.full(n, 0.5)
    idx = (np.arange(n) + 0.5) / n
    for k in range(1, K + 1):
        sd = 0.12 / np.sqrt(k)
        y += sd * rng.standard_normal() * np.cos(2 * np.pi * k * idx)
        y += sd * rng.standard_normal() * np.sin(2 * np.pi * k * idx)
    cands.append(y.copy())
    nb = np.zeros(n)
    nb[:n // 2] = 1.0
    if n % 2 == 1:
        nb[n // 2] = 0.5
    rng.shuffle(nb)
    cands.append(nb.copy())
    cands.append(base + 0.4 * rng.standard_normal(n))
    proj = [project_capped(c, n) for c in cands[:n_starts]]
    unique = [proj[0]]
    for p in proj[1:]:
        if not any(np.max(np.abs(p - u)) < 1e-10 for u in unique):
            unique.append(p)
    return unique[:n_starts]

def optimize_start(x, n, max_iter, temps):
    for tau in temps:
        alpha = 1.0
        for _ in range(max_iter):
            q = lag_values(x)
            if not np.all(np.isfinite(q)):
                break
            G = build_lag_jacobian(x)
            phi, g = smooth_grad(q, G, tau)
            if not np.all(np.isfinite(g)):
                break
            accepted = False
            for _ in range(25):
                xtrial = project_capped(x - alpha * g, n)
                d = xtrial - x
                if np.max(np.abs(d)) <= 1e-9:
                    break
                gd = np.dot(g, d)
                if gd >= 0.0:
                    break
                phit = smooth_phi(lag_values(xtrial), tau)
                if phit <= phi + 1e-4 * gd + 1e-14 * max(1.0, abs(phi)):
                    x = xtrial
                    alpha = min(4.0, alpha * 1.25)
                    accepted = True
                    break
                alpha *= 0.5
            if not accepted:
                break
    return x
