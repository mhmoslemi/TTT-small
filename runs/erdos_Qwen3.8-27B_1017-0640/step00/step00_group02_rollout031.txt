        if change <= 1e-7 or (prev_score - cur_score) < 1e-8:
            break
    return local_best_h, local_best_score


def run(seed=42, **kwargs):
    n = 45
    parent_fallback = None
    ih = globals().get('initial_h_values', None)
    if ih is not None:
        try:
            ih = np.array(ih, dtype=np.float64)
            if ih.ndim == 1 and ih.size >= 2:
                n = ih.size
                if np.all(np.isfinite(ih)):
                    S_tmp = n / 2.0
                    p = project_capped(ih, S_tmp)
                    if p is not None and exact_score(p, n) is not None:
                        parent_fallback = p.copy()
                    elif exact_score(ih, n) is not None:
                        parent_fallback = ih.copy()
        except Exception:
            pass
    S = n / 2.0
    dx = 2.0 / n
    if parent_fallback is None:
        parent_fallback = np.full(n, 0.5, dtype=np.float64)
    pairs = build_lag_pairs(n)
    if n > 256:
        return parent_fallback, exact_score(parent_fallback, n), n
    rng = np.random.default_rng(seed)
    starts_raw = [parent_fallback.copy(), np.full(n, 0.5, dtype=np.float64)]
    for sig in [0.08, 0.22]:
        starts_raw.append(parent_fallback + sig * rng.standard_normal(n))
    starts_raw.append(rng.uniform(0.0, 1.0, n))
    phase = float(rng.uniform(0.0, 2.0 * np.pi))
    ia = np.arange(n, dtype=np.float64)
    starts_raw.append(0.5 + 0.35 * np.sin(2.0 * np.pi * ia / n + phase))
    feasible_starts = []
    for s in starts_raw:
        p = project_capped(s, S)
        if p is None or exact_score(p, n) is None:
            continue
        is_dup = False
        for fs in feasible_starts:
            if float(np.max(np.abs(p - fs))) < 1e-12:
                is_dup = True
                break
        if not is_dup:
            feasible_starts.append(p)
    max_iters = 12 if n <= 64 else (8 if n <= 128 else 4)
    best_h = parent_fallback.copy()
    best_score = exact_score(best_h, n)
    if best_score is None:
        best_score = 1.0
    for st in feasible_starts:
        lb_h, lb_sc = local_mm(st, n, S, dx, pairs, max_iters)
        if lb_sc is not None and lb_sc < best_score:
            best_h = lb_h.copy()
            best_score = lb_sc
    for sig in [0.04, 0.015]:
        pert = best_h + sig * rng.standard_normal(n)
        p = project_capped(pert, S)
        if p is None or exact_score(p, n) is None:
            continue
        lb_h, lb_sc = local_mm(p, n, S, dx, pairs, max_iters)
        if lb_sc is not None and lb_sc < best_score:
            best_h = lb_h.copy()
            best_score = lb_sc
    final_score = exact_score(best_h, n)
    if final_score is None:
        fb_score = exact_score(parent_fallback, n)
        if fb_score is None:
            final_h = np.full(n, 0.5, dtype=np.float64)
            final_score = exact_score(final_h, n)
            return final_h, final_score, n
        best_h = parent_fallback
        final_score = fb_score
    return np.array(best_h, dtype=np.float64), float(final_score), int(n)
