import numpy as np
import math

def is_feasible(h, n):
    if h.ndim != 1 or len(h) != n or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-15) or np.any(h > 1.0 + 1e-15):
        return False
    if abs(float(np.sum(h)) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def project_capped(y, n):
    uniform = np.full(n, 0.5)
    if y.ndim != 1 or len(y) != n or not np.all(np.isfinite(y)):
        return uniform
    if np.max(np.abs(y)) > 1e6:
        return uniform
    m = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > m:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    p = np.clip(y - lam, 0.0, 1.0)
    r = m - float(np.sum(p))
    for _ in range(3):
        if abs(r) <= 2e-14 * max(1, n):
            break
        for i in range(n):
            if abs(r) <= 2e-14 * max(1, n):
                break
            if r > 0:
                d = min(r, 1.0 - p[i])
                p[i] += d
                r -= d
            else:
                d = min(-r, p[i])
                p[i] -= d
                r += d
    if not is_feasible(p, n):
        return uniform
    return p

def exact_score(h):
    n = len(h)
    if not is_feasible(h, n):
        return float('inf')
    dx = 2.0 / n
    try:
        corr = np.correlate(h, 1.0 - h, mode='full')
        if not np.all(np.isfinite(corr)):
            raise ValueError("nonfinite")
        return float(np.max(corr)) * dx
    except Exception:
        q0 = dx * float(np.sum(h * (1.0 - h)))
        best = q0
        for s in range(1, n):
            a = h[:n-s] * (1.0 - h[s:])
            b = h[s:] * (1.0 - h[:n-s])
            va = dx * float(np.sum(a))
            vb = dx * float(np.sum(b))
            if va > best:
                best = va
            if vb > best:
                best = vb
        if not np.isfinite(best):
            return float('inf')
        return best
