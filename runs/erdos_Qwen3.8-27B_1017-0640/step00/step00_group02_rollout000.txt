    one_minus_h = 1.0 - h
    q = np.empty(2 * n - 1)

    for s in range(1, n):

        q[n - 1 + s] = np.dot(h[s:], one_minus_h[:m])
        q[n - 1 - s] = np.dot(h[:m], one_minus_h[s:])

import numpy as np
from scipy.optimize import minimize

def project_capped(v, S):
    n = len(v)
    v = np.clip(v.astype(np.float64), -1e6, 1e6)
    lo = min(v.min() - 1.0, -1.0)
    hi = max(v.max() + 1.0, 2.0)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        if x.sum() > S:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-15:
            break
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(200):
        res = S - x.sum()
        if abs(res) < 1e-14:
            break
        if res > 0:
            mask = x < 1.0
            if not mask.any():
                break
            add = min(res, (1.0 - x[mask]).min())
            x[mask] += add
        else:
            mask = x > 0.0
            if not mask.any():
                break
            sub = min(-res, x[mask].min())
            x[mask] -= sub
    return x

def compute_q(h):
    n = len(h)
    omh = 1.0 - h
    q = np.empty(2 * n - 1)
    q[n - 1] = np.dot(h, omh)
    for s in range(1, n):
        m = n - s
        q[n - 1 + s] = np.dot(h[s:], omh[:m])
        q[n - 1 - s] = np.dot(h[:m], omh[s:])
    return q

def compute_G(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    G[n - 1] = 1.0 - 2.0 * h
    for s in range(1, n):
        m = n - s
        G[n - 1 + s, s:] = 1.0 - h[:m]
        G[n - 1 + s, :m] -= h[s:]
        G[n - 1 - s, :m] = 1.0 - h[s:]
        G[n - 1 - s, s:] -= h[:m]
    return G

def exact_score(h):
    n = len(h)
    val = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(val) * (2.0 / n))

def check_feasible(h, S, tol=1e-9):
    if h.shape != (len(h),) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -tol) or np.any(h > 1.0 + tol):
        return False
    if abs(h.sum() - S) > tol:
        return False
    return True

def obj_func(z, n, S):
    return z[-1], np.zeros(n + 1)

def ineq_fun(z, n, S, dx):
    h = z[:n]
    q = compute_q(h)
    return z[-1] - dx * q

def ineq_jac(z, n, S, dx):
    h = z[:n]
    G = compute_G(h)
    J = np.empty((2 * n - 1, n + 1))
    J[:, :n] = -dx * G
    J[:, n] = 1.0
    return J

def eq_fun(z, n, S):
    return np.array([z[:n].sum() - S])

def eq_jac(z, n, S):
    J = np.zeros((1, n + 1))
    J[0, :n] = 1.0
    return J

def run(seed=42, **kwargs):
    n = 45
    dx = 2.0 / n
    S = n / 2.0
    rng = np.random.default_rng(seed)
    fallback = np.full(n, 0.5)
    parent = None
    ival = globals().get('initial_h_values', None)
    if ival is not None:
        try:
            arr = np.array(ival, dtype=np.float64).ravel()
            if arr.size >= 2 and np.all(np.isfinite(arr)):
                _n = arr.size
                if _n != n:
                    n = _n
                    dx = 2.0 / n
                    S = n / 2.0
                    fallback = np.full(n, 0.5)
                if check_feasible(arr, S, 1e-8):
                    parent = arr.copy()
                else:
                    parent = project_capped(arr, S)
                fallback = parent.copy()
        except Exception:
            pass
    parent = np.clip(parent, 0.0, 1.0)
    parent = project_capped(parent, S)
    starts = [parent.copy(), fallback.copy()]
    sds = [0.03, 0.08, 0.16, 0.30, 0.08, 0.20]
    base = parent if parent is not None else fallback
    for sd in sds:
        cand = base + rng.standard_normal(n) * sd
        cand = project_capped(cand, S)
        starts.append(cand)
    for _ in range(2):
        cand = project_capped(rng.uniform(0, 1, n), S)
        starts.append(cand)
    for m_freq in [1, 2]:
        phase = rng.uniform(0, 2 * np.pi)
        idx = np.arange(n)
        cand = 0.5 + 0.35 * np.sin(2 * np.pi * m_freq * idx / n + phase)
        starts.append(project_capped(cand, S))
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac, 'args': (n, S, dx)},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (n, S)},
    ]
    incumbent = fallback.copy()
    incumbent_score = exact_score(incumbent)
    seen = [incumbent]
    def try_candidate(h_new):
        nonlocal incumbent, incumbent_score
        h_new = project_capped(h_new, S)
        if not check_feasible(h_new, S, 1e-9):
            return
        sc = exact_score(h_new)
        if sc < incumbent_score - 1e-12:
            incumbent = h_new.copy()
            incumbent_score = sc
    for st in starts:
        st = project_capped(st, S)
        if not check_feasible(st, S, 1e-9):
            continue
        if any(np.linalg.norm(st - s) < 1e-12 for s in seen):
            continue
        seen.append(st.copy())
        sc0 = exact_score(st)
        if sc0 < incumbent_score - 1e-12:
            incumbent = st.copy()
            incumbent_score = sc0
        z0 = np.concatenate([st, [min(max(sc0, 0.0), 2.0)]])
        try:
            res = minimize(obj_func, z0, args=(n, S), jac=True,
                           bounds=bounds, constraints=cons,
                           method='SLSQP',
                           options={'maxiter': 300, 'ftol': 1e-10, 'disp': False})
            if res.x is not None and len(res.x) == n + 1 and np.all(np.isfinite(res.x)):
                h_new = res.x[:n]
                try_candidate(h_new)
        except Exception:
            pass
    for sigma in [0.08, 0.03, 0.01]:
        for _ in range(2):
            pert = incumbent + rng.standard_normal(n) * sigma
            h_new = project_capped(pert, S)
            if not check_feasible(h_new, S, 1e-9):
                continue
            sc0 = exact_score(h_new)
            z0 = np.concatenate([h_new, [min(max(sc0, 0.0), 2.0)]])
            try:
                res = minimize(obj_func, z0, args=(n, S), jac=True,
                               bounds=bounds, constraints=cons,
                               method='SLSQP',
                               options={'maxiter': 400, 'ftol': 1e-11, 'disp': False})
                if res.x is not None and len(res.x) == n + 1 and np.all(np.isfinite(res.x)):
                    h_new2 = res.x[:n]
                    try_candidate(h_new2)
            except Exception:
                pass
    if not check_feasible(incumbent, S, 1e-9):
        incumbent = project_capped(incumbent, S)
    if not check_feasible(incumbent, S, 1e-9):
        incumbent = fallback.copy()
    c5 = exact_score(incumbent)
    return (incumbent, c5, n)
