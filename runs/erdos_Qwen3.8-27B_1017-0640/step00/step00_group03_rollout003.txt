def capped_simplex_project(v, n):
    v = np.asarray(v, dtype=np.float64).ravel()
    if v.size != n or not np.all(np.isfinite(v)):
        return np.full(n, 0.5)
    target = n / 2.0
    v = np.clip(v, -1e6, 1e6)
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        q = np.clip(v - mid, 0.0, 1.0)
        if np.sum(q) > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    q = np.clip(v - lam, 0.0, 1.0)
    residual = target - np.sum(q)
    if residual > 1e-13:
        mask = q < 1.0 - 1e-14
        if np.any(mask):
            add = min(residual * 0.999, np.sum(1.0 - q[mask]) * 0.999)
            q[mask] += add / np.sum(mask)
    elif residual < -1e-13:
        mask = q > 1e-14
        if np.any(mask):
            sub = min(-residual * 0.999, np.sum(q[mask]) * 0.999)
            q[mask] -= sub / np.sum(mask)
    q = np.clip(q, 0.0, 1.0)
    if not np.all(np.isfinite(q)) or abs(np.sum(q) - target) > 1e-8:
        return np.full(n, 0.5)
    return q


def exact_score(h):
    n = len(h)
    if n < 2 or not np.all(np.isfinite(h)):
        return float('inf')
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(np.max(corr) * (2.0 / n))
    return c5 if np.isfinite(c5) else float('inf')


def lag_values_and_jacobian(h):
    n = len(h)
    dx = 2.0 / n
    c = np.zeros(2 * n - 1)
    J = np.zeros((2 * n - 1, n))
    for d in range(1, n):
        idx = n - 1 - d
        u = h[:n - d]
        v = h[d:]
        c[idx] = np.sum(u * (1.0 - v)) * dx
        J[idx, :n - d] += (1.0 - v) * dx
        J[idx, d:] -= u * dx
    for d in range(n):
        idx = n - 1 + d
        u = h[:n - d]
        v = h[d:]
        c[idx] = np.sum(v * (1.0 - u)) * dx
        if d == 0:
            J[idx, :] = (1.0 - 2.0 * h) * dx
        else:
            J[idx, d:] += (1.0 - u) * dx
            J[idx, :n - d] -= v * dx
    return c, J


def obj_fn(z):
    return z[-1]


def obj_jac(z):
    n = len(z) - 1
    g = np.zeros(len(z))

def eq_fn(z):

def eq_jac(z):
    n = len(z) - 1
    g = np.zeros(n + 1)

    return g


def ineq_fn(z):

    h = z[:-1]

    n = len(z) - 1
    h = z[:-1]
    _, J = lag_values_and_jacobian(h)
    return np.hstack([-J, np.ones((2 * n - 1, 1))])
