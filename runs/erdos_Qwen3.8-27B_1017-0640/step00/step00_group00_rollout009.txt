    y = np.clip(np.asarray(y, dtype=np.float64).copy(), -1e6, 1e6)

        else:
            hi = mid

            d = min(r, 1.0 - x[i])

    elif r < 0:

            d = min(-r, x[i])
            if d > 0:
                x[i] -= d; r += d
                if r > -1e-15: break
