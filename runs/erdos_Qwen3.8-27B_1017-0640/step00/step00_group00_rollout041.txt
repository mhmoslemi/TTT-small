        ]
        maxit = 100 if n <= 64 else (60 if n <= 160 else (35 if n <= 400 else 20))
        r = minimize(_so, z0, method='SLSQP', jac=_soj, bounds=bnd, constraints=cons, args=(params,),
                     options={'maxiter': maxit, 'ftol': 1e-9, 'disp': False})
        if r.success and np.all(np.isfinite(r.x)):
            return r.x[:n]
    except Exception:
        pass
    return None

def mm_optimize(x, n, m, dx, max_iter):
    x_score = exact_score(x, n)
    if x_score is None:
        return x, x_score
    for _ in range(max_iter):
        raw = np.correlate(x, 1.0 - x, mode='full')
        q = raw * dx
        G = lag_gradients(x, n)
        if not np.all(np.isfinite(G)):
            break
        d = solve_subproblem(x, q, G, n, m, dx)
        if d is None:
            break
        y = project_capped(x + d, n, m)
        if y is None:
            break
        improved = False
        for th in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            pt = project_capped(x + th * (y - x), n, m)
            if pt is None:
                continue
            sc = exact_score(pt, n)
            if sc is not None and sc < x_score - 1e-12:
                x, x_score, improved = pt, sc, True
                break
        if not improved:
            break
    return x, x_score

def make_starts(n, m, rng, par):
    lst = []
    if par is not None:
        lst.append(par.copy())
        pc = project_capped(1.0 - par, n, m)
        if pc is not None:
            lst.append(pc)
    blk = np.zeros(n)
    blk[:int(m)] = 1.0
    if m - int(m) > 1e-14:
        blk[int(m)] = m - int(m)
    lst.append(blk.copy())
    lst.append(blk[::-1].copy())
    pr = project_capped(rng.random(n), n, m)
    if pr is not None:
        lst.append(pr)
    idx = np.arange(n, dtype=np.float64)
    for r in [1, 2]:
        cv = 0.5 + 0.4 * np.cos(2.0 * np.pi * r * idx / n + rng.random() * 2.0 * np.pi)
        pcv = project_capped(cv, n, m)
        if pcv is not None:
            lst.append(pcv)
    if par is not None:
        pp = project_capped(par + 0.10 * rng.standard_normal(n), n, m)
        if pp is not None:
            lst.append(pp)
        prl = project_capped(np.roll(par, n // 3), n, m)
        if prl is not None:
            lst.append(prl)
    return lst

def run(seed=42, **kwargs):
    n_override = None
    if 'n_points' in kwargs and kwargs['n_points'] is not None:
        try:
            nv = int(kwargs['n_points'])
            if not isinstance(kwargs['n_points'], bool) and 1 <= nv <= 999:
                n_override = nv
        except (ValueError, TypeError, OverflowError):
            pass
    parent = None
    try:
        pv = np.asarray(initial_h_values, dtype=np.float64).flatten()
        if 0 < len(pv) <= 999 and np.all(np.isfinite(pv)):
            parent = pv.copy()
    except Exception:
        pass
    if n_override is not None:
        n = n_override
    elif parent is not None:
        n = len(parent)
    else:
        n = 45
    m = n / 2.0
    dx = 2.0 / n
    par = None
    if parent is not None:
        if len(parent) == n:
            par = parent
        elif len(parent) == 1:
            par = np.full(n, parent[0])
        else:
            par = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        par = project_capped(par, n, m)
        if par is None:
            par = np.full(n, 0.5)
    fb = par.copy() if par is not None else np.full(n, 0.5)
    fb_sc = exact_score(fb, n)
    if fb_sc is None:
        fb, fb_sc = np.full(n, 0.5), exact_score(np.full(n, 0.5), n)
    best_h, best_c = fb.copy(), fb_sc
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    if n <= 64:
        n_starts, mx_it = 6, 40
    elif n <= 160:
        n_starts, mx_it = 4, 25
    elif n <= 400:
        n_starts, mx_it = 3, 12
    else:
        n_starts, mx_it = 2, 5
    raw_starts = make_starts(n, m, rng, par)
    starts = []
    for s in raw_starts:
        ps = project_capped(s, n, m)
        if ps is None:
            continue
        if any(np.max(np.abs(ps - e)) <= 1e-10 for e in starts):
            continue
        starts.append(ps)
        if len(starts) >= n_starts:
            break
    for st in starts:
        ox, sc = mm_optimize(st.copy(), n, m, dx, mx_it)
        if sc is not None and sc < best_c - 1e-12:
            best_c, best_h = sc, ox.copy()
    if n <= 64:
        sds = [0.04, 0.15]
    elif n <= 160:
        sds = [0.08]
    elif n <= 400:
        sds = [0.12]
    else:
        sds = [0.15]
    for sd in sds:
        pt0 = project_capped(best_h + sd * rng.standard_normal(n), n, m)
        if pt0 is None:
            continue
        ox, sc = mm_optimize(pt0, n, m, dx, mx_it)
        if sc is not None and sc < best_c - 1e-12:
            best_c, best_h = sc, ox.copy()
    fin = project_capped(best_h, n, m)
    if fin is None:
        fin = np.full(n, 0.5)
    fc = exact_score(fin, n)
    if fc is None:
        fin, fc = np.full(n, 0.5), exact_score(np.full(n, 0.5), n)
    return (fin, fc, int(n))
