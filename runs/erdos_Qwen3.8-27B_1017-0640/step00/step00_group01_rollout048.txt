def project_capped_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1 or v.size < 2:
        return None
    if not np.all(np.isfinite(v)) or np.max(np.abs(v)) > 1e6:
        return None
    n = v.size

    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))

        if m > s:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    x = np.clip(v - lam, 0.0, 1.0)
    for _ in range(4):
        r = s - math.fsum(x)
        if abs(r) <= 5e-15:
            break
        if r > 0:
            for i in range(n):
                if r <= 0:
                    break
                add = min(r, 1.0 - x[i])
                if add > 0:
                    x[i] += add
                    r -= add
        else:
            for i in range(n - 1, -1, -1):

                    break

                    r += sub

        return None
    if abs(math.fsum(x) - s) > 1e-9:
        return None
    return np.clip(x, 0.0, 1.0)


def check_feasible(h, n):
    h = np.asarray(h, dtype=np.float64)
    if h.ndim != 1 or h.size != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.min(h) < -1e-9 or np.max(h) > 1.0 + 1e-9:
        return False
    if abs(math.fsum(h) - n / 2.0) > 1e-9:
        return False
    return True


def exact_score(h, n):
    h = np.asarray(h, dtype=np.float64)
    if h.ndim != 1 or h.size != n or not np.all(np.isfinite(h)):
        return None
    try:
        corr = np.correlate(h, 1.0 - h, mode='full')
        val = float(np.max(corr)) * (2.0 / n)
    except Exception:
        val = 0.0
        for d in range(-(n - 1), n):
            q = 0.0
            for i in range(max(0, -d), min(n, n + d)):
                q += h[i] * (1.0 - h[i + d])
            if q > val:
                val = q
        val *= (2.0 / n)
    return val if np.isfinite(val) else None


def insert_candidate(v, cands, scores, cap, n):
    x = project_capped_simplex(v)
    if x is None or not check_feasible(x, n):
        return None
    sc = exact_score(x, n)
    if sc is None:
        return None
    for c in cands:
        if np.max(np.abs(x - c)) <= 1e-10:
            return None
    if len(cands) < cap:
        cands.append(x.copy())
        scores.append(sc)
    return sc


def solve_relaxation(n):
    try:
        s = n / 2.0
        h = cp.Variable(n)
        X = cp.Variable((n, n), symmetric=True)
        t = cp.Variable()
        cons = []
        M = cp.hstack([cp.hstack([1, h.T]), cp.vstack([h, X])])
        cons.append(M >> 0)
        cons.append(h >= 0)
        cons.append(h <= 1)
        cons.append(cp.sum(h) == s)
        cons.append(t >= 0)
        cons.append(t <= 1)
        for i in range(n):
            for j in range(i, n):
                cons.append(X[i, j] >= 0)
                cons.append(X[i, j] >= h[i] + h[j] - 1)
                cons.append(X[i, j] <= h[i])
                cons.append(X[i, j] <= h[j])
            cons.append(cp.sum(X[i, :]) == s * h[i])
        dx = 2.0 / n
        for d in range(-(n - 1), n):
            q = 0
            for i in range(max(0, -d), min(n, n + d)):
                q = q + (h[i] - X[i, i + d])
            cons.append(t >= dx * q)
        prob = cp.Problem(cp.Minimize(t), cons)
        prob.solve(solver=cp.CVXOPT, verbose=False, max_iters=80,
                   abstol=1e-7, reltol=1e-7, feastol=1e-7)
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None
        hv = np.array(h.value, dtype=np.float64).flatten()
        Xv = np.array(X.value, dtype=np.float64)
        if hv.shape != (n,) or Xv.shape != (n, n):
            return None
        if not np.all(np.isfinite(hv)) or not np.all(np.isfinite(Xv)):
            return None
        return (hv, Xv)
    except Exception:
        return None

    cur = start.copy()
    cur_sc = exact_score(cur, n)
    if cur_sc is None:
        return cur, None
    for T, sig in zip(temps, scales):
        for _ in range(nprops):
            if Lnorm is not None and rng.random() < mixp:

                delta = Lnorm @ z

                delta = rng.standard_normal(n)

            delta[n - 1] -= math.fsum(delta)
            if not np.all(np.isfinite(delta)):
                continue
            y = cur + sig * delta
            if np.min(y) < -1e-12 or np.max(y) > 1.0 + 1e-12:

            if dsp <= 0 or rng.random() < math.exp(-min(dsp / T, 700.0)):
                cur = y.copy()
                cur_sc = y_sc
    return cur, cur_sc
