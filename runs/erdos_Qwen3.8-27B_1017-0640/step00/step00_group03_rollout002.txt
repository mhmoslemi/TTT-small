        q = np.clip(v - mid, 0.0, 1.0)

            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    q = np.clip(v - lam, 0.0, 1.0)
    residual = target - np.sum(q)
    if abs(residual) > 1e-14:
        if residual > 0:
            cap = 1.0 - q
        else:
            cap = q
        tc = np.sum(cap)
        if tc > 1e-14:
            q = np.clip(q + (residual / tc) * cap, 0.0, 1.0)
    return q


def exact_score(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if len(c) != 2 * n - 1 or not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))


def lag_vals_jac(h):
    n = len(h)
    dx = 2.0 / n
    nv = 2 * n - 1
    c = np.zeros(nv)
    J = np.zeros((nv, n))
    for d in range(n):
        u = h[:n - d]
        v = h[d:]
        c[n - 1 + d] = np.sum(v * (1.0 - u))
        JA = np.zeros(n)
        JB = np.zeros(n)
        if d == 0:
            JA[:] = 1.0 - 2.0 * h
            JB[:] = 1.0 - 2.0 * h
        else:
            JA[d:] = 1.0 - h[:n - d]
            JA[:n - d] -= h[d:]
            JB[:n - d] = 1.0 - h[d:]
            JB[d:] -= h[:n - d]
        J[n - 1 + d, :] = JA
        if d > 0:
            c[n - 1 - d] = np.sum(u * (1.0 - v))
            J[n - 1 - d, :] = JB
    c *= dx
    J *= dx
    return c, J


def obj_fn(z):
    return z[-1]


def obj_grad(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g


def eq_fn(z):
    return np.sum(z[:-1]) - (len(z) - 1) / 2.0


def eq_grad(z):
    g = np.ones(len(z))
    g[-1] = 0.0
    return g


def ineq_fn(z):
    c, _ = lag_vals_jac(z[:-1])
    return z[-1] - c


def ineq_grad(z):
    n = len(z) - 1
    nv = 2 * n - 1
    _, J = lag_vals_jac(z[:-1])
    G = np.zeros((nv, n + 1))
    G[:, :n] = -J
    G[:, n] = 1.0
    return G


def verify_h(h, n, tol=1e-10):
    if len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.min(h) < -tol or np.max(h) > 1.0 + tol:
        return False
    if abs(np.sum(h) - n / 2.0) > tol:
        return False
    return True


def solve_slsqp(s, n, maxiter):
    t0 = exact_score(s) + 1e-8
    z0 = np.concatenate([s, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    eq_con = {'type': 'eq', 'fun': eq_fn, 'jac': eq_grad}
    ineq_con = {'type': 'ineq', 'fun': ineq_fn, 'jac': ineq_grad}
    res = minimize(obj_fn, z0, jac=obj_grad, method='SLSQP',
                   bounds=bounds, constraints=[eq_con, ineq_con],
                   options={'maxiter': maxiter, 'ftol': 1e-10, 'disp': False})
    if res.x is not None and len(res.x) == n + 1:
        if np.all(np.isfinite(res.x[:n])):
            return res.x[:n]
    return None


def run(seed=42, **kwargs):
    n = 70
    h_init = None
    parent_valid = False

    iv = globals().get('initial_h_values', None)
    if iv is not None:
        try:
            iv = np.asarray(iv, dtype=float).ravel()
        except Exception:
            iv = None
        if iv is not None and iv.size >= 2 and np.all(np.isfinite(iv)):
            n = iv.size
            h_init = iv.copy()
            if verify_h(h_init, n):
                parent_valid = True
            h_init = capped_simplex_project(h_init, n)

    if h_init is None:
        h_init = np.full(n, 0.5)

    if n > 256:
        sc = exact_score(h_init)
        return (h_init, sc, n)

    rng = np.random.default_rng(seed)
    maxiter = 250 if n <= 128 else 120

    starts = [h_init.copy(), h_init[::-1].copy(), np.full(n, 0.5)]
    idx = np.arange(n, dtype=float)
    starts.append(0.5 + 0.35 * np.sin(2.0 * np.pi * (idx + 0.5) / n))
    starts.append(0.5 + 0.25 * rng.standard_normal(n))
    starts.append(rng.uniform(0.0, 1.0, size=n))

    proj_starts = [capped_simplex_project(s, n) for s in starts]
    unique_starts = [proj_starts[0]]
    for ps in proj_starts[1:]:
        dup = False
        for us in unique_starts:
            if np.max(np.abs(ps - us)) < 1e-10:
                dup = True
                break
        if not dup:
            unique_starts.append(ps)

    if n <= 128:
        nsolve = len(unique_starts)
    else:
        nsolve = min(4, len(unique_starts))

    best_h = capped_simplex_project(h_init, n)
    best_score = exact_score(best_h)

    for s in unique_starts[:nsolve]:
        try:
            h_cand = solve_slsqp(s, n, maxiter)
        except Exception:
            continue
        if h_cand is not None:
            h_cand = capped_simplex_project(h_cand, n)
            if verify_h(h_cand, n):
                sc = exact_score(h_cand)
                if sc < best_score - 1e-12:
                    best_h = h_cand
                    best_score = sc

    if n <= 128:
        for sigma in (0.03, 0.08):
            perturbed = best_h + sigma * rng.standard_normal(n)
            ps = capped_simplex_project(perturbed, n)
            try:
                h_cand = solve_slsqp(ps, n, maxiter)
            except Exception:
                continue
            if h_cand is not None:
                h_cand = capped_simplex_project(h_cand, n)
                if verify_h(h_cand, n):
                    sc = exact_score(h_cand)
                    if sc < best_score - 1e-12:
                        best_h = h_cand
                        best_score = sc

    best_h = capped_simplex_project(best_h, n)
    if not verify_h(best_h, n):
        if parent_valid:
            best_h = capped_simplex_project(h_init, n)
        else:
            best_h = np.full(n, 0.5)
    best_score = exact_score(best_h)
    return (best_h, best_score, n)
