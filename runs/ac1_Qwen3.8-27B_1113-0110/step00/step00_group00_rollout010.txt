        s = a_new.sum()
        if s < 1e-12:
            a = a + rng.normal(0, 0.02, n)
            a = np.maximum(a, 0.0)
            s = a.sum()
            a = a / max(s, 1e-12)
        else:
            a = a_new / s

        iters += 1
        no_improve += 1

        # Periodic exact evaluation
        check_interval = 30
        if iters % check_interval == 0 or time.monotonic() > deadline - 3.0:
            score = evaluate_sequence([float(x) for x in a])
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_a = a.copy()
                last_improve_iter = iters
                no_improve = 0
                if verbose_prefix:
                    print(f"  {verbose_prefix} iter={iters} score={best_score:.7f}")

        # Adaptive parameters
        if no_improve > 300:
            eta *= 0.75
            beta = min(beta * 1.25, 500.0)
            no_improve = 0
            if eta < 1e-5:
                # Inject noise to escape
                a = a * (1.0 + rng.normal(0, 0.05, n))
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-12:
                    a = a / s
                eta = eta0 * 0.5
                beta = beta0

    return best_a, best_score, iters


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-restart projected gradient optimization to minimize
    2n*max(convolve(a,a))/(sum(a))^2 over non-negative sequences.
    Uses smooth-max surrogates, FFT gradients, and diverse initializations.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(1.0, budget_s) - 8.0

    # Gather candidate initializations
    initializations = []

    # From height_sequence_1
    try:
        h1 = np.asarray(height_sequence_1, dtype=np.float64).flatten()
        h1 = np.clip(h1, 0.0, 1000.0)
        if h1.size > 10 and h1.sum() > 0.01 and np.isfinite(h1).all():
            initializations.append(_standardize_seq(h1))
            # Perturbed version
            p = h1.copy()
            p *= np.exp(rng.normal(0, 0.1, p.size))
            initializations.append(_standardize_seq(p))
    except Exception:
        pass

    # Various structured non-uniform starts
    for n_try in set([1024, 2048, 4096]):
        initializations.append(_make_perturbed_uniform(n_try, rng, 0.2))
        initializations.append(_make_cosine_seq(n_try, freq=rng.integers(1, 5)))
        initializations.append(_make_two_block_seq(n_try, rng.uniform(0.3, 0.7)))
        initializations.append(_make_chirp_seq(n_try))

    # Add perturbed versions of each
    extra = []
    for a in initializations[:]:
        pert = a * (1.0 + rng.normal(0, 0.08, a.size))
        extra.append(_standardize_seq(pert))
    initializations.extend(extra)

    print(f"Total initializations: {len(initializations)}")

    # Allocate time: 70% for full runs, rest distributed
    n_full = min(6, len(initializations))
    time_per_full = 0.7 * max(0, (total_deadline - time.monotonic())) / max(n_full, 1)
    time_per_quick = max(2.0, time_per_full / 4.0)

    best_overall = None
    best_score_overall = np.inf

    # Phase 1: Quick screening of many initializations
    quick_idx = list(range(len(initializations)))
    rng.shuffle(quick_idx)

    for idx in quick_idx:
        if time.monotonic() > total_deadline - 60:
            break
        a_init = initializations[idx].copy()
        dl = time.monotonic() + time_per_quick
        best_a, score, _ = _run_gradient_descent(
            a_init, dl, rng, beta0=30.0, eta0=0.12, verbose_prefix=""
        )
        if score < best_score_overall:
            best_score_overall = score
            best_overall = best_a.copy()
            if best_score_overall < 2.0:
                print(f"  Quick: n={best_a.size} score={best_score_overall:.7f}")

    print(f"After quick phase: best={best_score_overall:.7f}")

    # Phase 2: Full optimization on top candidates
    # Use the best found + a few diverse ones
    full_candidates = []
    if best_overall is not None:
        full_candidates.append(best_overall)
    for idx in quick_idx[:4]:
        full_candidates.append(initializations[idx].copy())

    # Deduplicate by length
    seen_lengths = set()
    unique_full = []
    for c in full_candidates:
        if c.size not in seen_lengths:
            unique_full.append(c)
            seen_lengths.add(c.size)

    for i, fc in enumerate(unique_full):
        if time.monotonic() > total_deadline - 30:
            break
        remaining = total_deadline - time.monotonic()
        alloc = remaining / (len(unique_full) - i)
        dl = time.monotonic() + alloc
        best_a, score, iters = _run_gradient_descent(
            fc.copy(), dl, rng, beta0=50.0, eta0=0.15,
            verbose_prefix=f"Full[{i}]"
        )
        if score < best_score_overall:
            best_score_overall = score
            best_overall = best_a.copy()
            print(f"  Full[{i}] improved: score={best_score_overall:.7f} (n={best_a.size}, {iters} iters)")

    print(f"After full phase: best={best_score_overall:.7f}")

    # Phase 3: Fine-tuning with local perturbations around best
    if best_overall is not None and time.monotonic() < total_deadline - 10:
        base = best_overall.copy()
        for attempt in range(200):
            if time.monotonic() > total_deadline - 5:
                break
            sigma = rng.uniform(0.002, 0.02)
            a_pert = base + rng.normal(0, sigma, base.size)
            a_pert = np.maximum(a_pert, 0.0)
            s = a_pert.sum()
            if s > 0.01:
                a_pert = a_pert / s
                score = evaluate_sequence([float(x) for x in a_pert])
                if np.isfinite(score) and score < best_score_overall:
                    best_score_overall = score
                    base = a_pert.copy()
                    best_overall = a_pert.copy()
                    print(f"  Fine-tune: score={best_score_overall:.7f}")

    # Fallback: ensure we have something valid
    if best_overall is None or not np.isfinite(best_score_overall):
        best_overall = np.ones(1024) / 1024
        best_score_overall = evaluate_sequence([float(x) for x in best_overall])
        print(f"Fallback score: {best_score_overall:.7f}")

    final_score = evaluate_sequence([float(x) for x in best_overall])
    if not np.isfinite(final_score) or final_score > best_score_overall:
        final_score = best_score_overall

    print(f"FINAL: n={len(best_overall)}, score={final_score:.7f}")
    return [float(x) for x in best_overall]
