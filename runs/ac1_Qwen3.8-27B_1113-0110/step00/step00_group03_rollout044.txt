```python
def optimize_sequence(a, rng, deadline):
    """Subgradient descent to minimize 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    a = np.maximum(0, a).astype(np.float64)
    if a.sum() < 1e-10:
        a[:] = 1.0 / n
    a = a / a.sum()  # normalize to sum = 1
    
    best = a.copy()
    best_score = 2 * n * np.max(np.convolve(a, a))
    # (since sum=1, score = 2n*max_b)
    
    lr = 1.0 / n  # initial learning rate
    no_improve = 0
    
    while time.monotonic() < deadline - 5:
        # Compute autocorrelation
        b = np.convolve(a, a)
        sum_a = a.sum()
        score = 2 * n * b.max() / (sum_a ** 2)
        
        if score < best_score:
            best_score = score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Find argmax (and possibly top-k)
        # Use top-k for smoother gradient
        k = max(1, min(5, n // 100))
        top_indices = np.argsort(b)[-k:]
        
        # Compute subgradient: average over top-k
        g = np.zeros(n)
        for m in top_indices:
            # g[j] += 2 * a[m-j] (for valid m-j)
            # a[m-j] for j in [0, n-1]: valid when 0 <= m-j <= n-1
            j_start = max(0, m - (n-1))
            j_end = min(n-1, m)
            g[j_start:j_end+1] += 2 * a[m - (j_start:j_end+1)][::-1]
        
        g /= k  # average
        
        # Update
        a = a - lr * g
        a = np.maximum(0, a)
        s = a.sum()
        if s < 1e-10:
            a[:] = 1.0 / n
        else:
            a = a / s
        
        # Decay learning rate
        lr *= 0.999
        
        # Periodic random perturbation to escape local optima
        if no_improve > 100:
            # Random perturbation
            pert = rng.random(n) * 0.1
            a = a + pert
            a = np.maximum(0, a)
            a = a / a.sum()
            no_improve = 0
            lr = 1.0 / n  # reset learning rate
    
    return best
```

```python
Subgradient descent with multiple asymmetric initializations to minimize
the autocorrelation peak ratio score = 2n * max(conv(a,a)) / sum(a)^2.
Key insight: asymmetric sequences can achieve score < 2.

import numpy as np
import time


def compute_score(a):
    """Fast score computation (same as evaluate_sequence but on numpy array)."""
    n = len(a)
    b = np.convolve(a, a)
    max_b = np.max(b)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2 * n * max_b / (sum_a ** 2))


def simplex_project(v):
    """Project v onto the simplex {x : x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    # Find lambda such that sum(max(v - lambda, 0)) = 1
    # Sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv[-1] - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def subgradient_step(a, b, top_k=3):
    """Compute subgradient of max(b) and return updated a."""
    n = len(a)
    # Find top_k indices of b
    if top_k >= len(b):
        top_idx = np.arange(len(b))
    else:
        top_idx = np.argsort(b)[-top_k:]
    
    # Compute subgradient: g[j] = (2/top_k) * sum_{m in top_idx} a[m-j]
    g = np.zeros(n)
    for m in top_idx:
        # a[m-j] for j in [max(0, m-n+1), min(m, n-1)]
        j_lo = max(0, m - n + 1)
        j_hi = min(m, n - 1)
        if j_lo <= j_hi:
            g[j_lo:j_hi+1] += 2.0 * a[m - np.arange(j_lo, j_hi + 1)][::-1]
    
    g /= top_k
    return g

    n = len(a)

    lr = 0.5
    decay = 0.9995
    no_improve = 0
    top_k = 3
    
    iteration = 0
    while time.monotonic() < deadline:
        b = np.convolve(a, a)
        g = subgradient_step(a, b, top_k)
        
        # Update
        a = a - lr * g
        a = simplex_project(a)
        
        # Optional: small smoothing (helps convergence)
        # (skipped for now)
        
        iteration += 1

            if score < best_score:
                best_score = score
                best = a.copy()
                no_improve = 0
                if verbose:
                    print(f"    iter {iteration}: score = {best_score:.6f}")
            else:
                no_improve += 1

            lr *= decay
            
            # Random restart if stuck
            if no_improve > 50:

                pert = rng.random(n)
                a = 0.9 * a + 0.1 * pert
                a = simplex_project(a)
                lr = 0.5
                no_improve = 0
    
    # Final check
    final_score = compute_score(best)
    return best, final_score


def make_initialization(n, init_type, rng):

        smooth = np.cumsum(steps)
        smooth = (smooth - smooth.min()) / (smooth.max() - smooth.min() + 1e-10)
        a = np.exp(smooth)
        return a / a.sum()
    elif init_type == 4:

        center = n // 2
        a = np.zeros(n)
        left_decay = rng.uniform(0.5, 3.0)
        right_decay = rng.uniform(0.5, 3.0)
        for i in range(n):

    elif init_type == 5:

    elif init_type == 6:

    else:
```

```python
score = 2n * max(conv(a,a)) / sum(a)^2.
Key insight: asymmetric sequences break the Cauchy-Schwarz barrier (score < 2).
Uses exponential-weight updates, top-k subgradients, and multi-start strategy.

import numpy as np
import time


def compute_score(a):
    """Compute the score for a numpy array (fast, no list conversion)."""
    n = len(a)
    b = np.convolve(a, a)
    max_b = np.max(b)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


def subgradient_direction(a, b, top_k=5):
    """Compute averaged subgradient of max(b) w.r.t. a using top-k peaks."""
    n = len(a)
    k = min(top_k, len(b))
    top_idx = np.argpartition(b, -k)[-k:]
    
    g = np.zeros(n, dtype=np.float64)
    for m in top_idx:
        j_lo = max(0, m - n + 1)
        j_hi = min(m, n - 1)
        if j_lo <= j_hi:
            # g[j] += 2 * a[m-j] for j in [j_lo, j_hi]
            # a[m-j] for j in [j_lo, j_hi] = a[m-j_hi : m-j_lo-1 : -1]
            g[j_lo:j_hi+1] += 2.0 * a[m - (j_hi:j_lo-1:-1)]
    
    g /= float(k)
    return g


def exponential_update(a, g, lr):
    """Multiplicative update: a <- a * exp(-lr * g), then normalize."""
    # Clip g to prevent numerical overflow
    g_clipped = np.clip(g, -5.0 / max(lr, 1e-10), 5.0 / max(lr, 1e-10))
    a_new = a * np.exp(-lr * g_clipped)
    s = a_new.sum()
    if s < 1e-300 or not np.isfinite(s):

    return a_new / s

    # Determine primary length
    if parent is not None and parent.size >= 100:
        n_primary = parent.size
    else:
        n_primary = 2047
    
    print(f"Primary length: {n_primary}, parent available: {parent is not None}")
    
    # Phase 1: Quick exploration of different initializations (20% of time)
    phase1_budget = min(120.0, (total_deadline - time.time()) * 0.2)
    phase1_deadline = time.time() + phase1_budget
    
    n_inits = 10
    init_time_per = phase1_budget / n_inits
    
    for i in range(n_inits):
        if time.time() > phase1_deadline:
            break
        
        init_deadline = time.time() + init_time_per + 2
        a = make_init(n_primary, i, rng, parent)
        a, score = optimize_run(a, n_primary, rng, init_deadline, verbose_prefix=f"  init{i}: ")
        
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  Phase 1, init {i}: new best score = {best_score:.6f}")
    
    # Phase 2: Deep optimization on best found (remaining time)
    if best_seq is not None:
        deep_deadline = total_deadline
        print(f"\nPhase 2: deep optimization, current best = {best_score:.6f}")
        
        # Run multiple deep optimizations with different random seeds
        n_deep_runs = 3
        deep_budget_per = (deep_deadline - time.time()) / n_deep_runs
        
        for run_i in range(n_deep_runs):
            if time.time() > deep_deadline - 10:
                break
            
            run_deadline = time.time() + deep_budget_per
            # Slightly perturb the best to explore different basins
            if run_i == 0:
                a_start = best_seq.copy()
            else:
                a_start = best_seq * (1.0 + 0.1 * rng.normal(0, 1, n_primary))
                a_start = np.maximum(a_start, 0)
                a_start = a_start / a_start.sum()
            
            a, score = optimize_run(a_start, n_primary, rng, run_deadline,
                                     verbose_prefix=f"  deep{run_i}: ")
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"  Phase 2, run {run_i}: new best score = {best_score:.6f}")
    
    # Phase 3: Try a different length if time permits
    remaining = total_deadline - time.time()
    if remaining > 60:
        n_alt = n_primary // 2 if n_primary > 512 else n_primary * 2
        n_alt = max(256, min(n_alt, 4095))
        if n_alt != n_primary:
            print(f"\nPhase 3: trying length {n_alt}")
            for i in range(4):
                if time.time() > total_deadline - 10:
                    break
                a = make_init(n_alt, i + 1, rng, None)
                a, score = optimize_run(a, n_alt, rng, total_deadline - 5,
                                         verbose_prefix=f"  alt{i}: ")
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"  Phase 3, init {i}: new best score = {best_score:.6f}")
    
    # Final validation
    if best_seq is not None:
        final_val = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(final_val) and final_val < best_score:
            best_score = final_val
        print(f"\nFinal score: {best_score:.6f}, length: {len(best_seq)}")
    else:
        # Fallback: uniform sequence
        best_seq = np.ones(256)
        print("FAM: using fallback uniform sequence")
    
    return [float(x) for x in best_seq]
```
