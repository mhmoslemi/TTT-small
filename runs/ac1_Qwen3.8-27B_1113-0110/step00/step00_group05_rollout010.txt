    best_peak = np.convolve(a, a).max()
    
    t0 = time.monotonic()
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = np.convolve(a, a)
        peak = conv.max()
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()
        
        # Find peak location
        k_star = conv.argmax()
        
        # Compute gradient-like direction
        # For the peak at k_star, the "gradient" w.r.t. a[i] is a[k_star-i]
        # (and by symmetry, a[i] also contributes to a[k_star-i])
        
        # Direction: reduce a[i] where a[i]*a[k_star-i] is large
        # and increase where it's small
        
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[j]
        
        # Normalize contrib to have sum 0
        contrib -= contrib.mean()
        
        # Step: a_new = a - eta * contrib, clipped to >= 0
        eta = 0.01
        a_new = a - eta * contrib
        a_new = np.clip(a_new, 0, None)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-10:
            break
        a_new = a_new / a_new_sum
        
        new_peak = np.convolve(a_new, a_new).max()
        if new_peak < peak:
            a = a_new
        else:
            # Try smaller step
            eta = 0.001
            a_new = a - eta * contrib
            a_new = np.clip(a_new, 0, None)
            a_new_sum = a_new.sum()
            if a_new_sum < 1e-10:
                break
            a_new = a_new / a_new_sum
            new_peak = np.convolve(a_new, a_new).max()
            if new_peak < peak:
                a = a_new
            else:
                # Give up this direction, try random
                a = a + np.random.normal(0, 0.001, n)
                a = np.clip(a, 0, None)
                a = a / a.sum()
    
    return best_a

import time
from scipy.optimize import linprog
from scipy.sparse import csc_matrix

def build_conv_matrix(a, n):
    """Fast construction of (2n-1) x n convolution matrix."""
    rows = np.empty(n * n, dtype=np.int64)
    cols = np.empty(n * n, dtype=np.int64)
    vals = np.empty(n * n, dtype=np.float64)
    
    for j in range(n):
        start = j * n
        rows[start:start+n] = np.arange(j, j + n)
        cols[start:start+n] = j
        vals[start:start+n] = a
    
    return csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def solve_lp_direction(a, n, time_limit=30):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Return g or None."""
    # Variables: [g_0, ..., g_{n-1}, M]
    # Total: n+1 variables
    total_vars = n + 1
    m_col = n  # index of M variable
    
    # Build A_ub (inequality constraints): conv(a,g) - M <= 0
    # A_ub has shape (2n-1, n+1)
    # For the g part: use the convolution matrix
    # For the M part: -1 in each row
    
    rows_g = np.empty(n * n, dtype=np.int64)
    cols_g = np.empty(n * n, dtype=np.int64)
    vals_g = np.empty(n * n, dtype=np.float64)
    
    for j in range(n):
        start = j * n
        rows_g[start:start+n] = np.arange(j, j + n)
        cols_g[start:start+n] = j
        vals_g[start:start+n] = a
    
    # M part: -1 in column m_col for each row
    rows_m = np.arange(2*n-1)
    cols_m = np.full(2*n-1, m_col)
    vals_m = np.full(2*n-1, -1.0)
    
    # Combine
    all_rows = np.concatenate([rows_g, rows_m])
    all_cols = np.concatenate([cols_g, cols_m])
    all_vals = np.concatenate([vals_g, vals_m])
    
    A_ub = csc_matrix((all_vals, (all_rows, all_cols)), shape=(2*n-1, n+1))
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Objective: minimize M
    c_obj = np.zeros(n+1)
    c_obj[m_col] = 1.0
    
    bounds = [(0, None)] * (n+1)
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs', options={'time_limit': time_limit})
    
    if result.success and result.x is not None:

    return None

```python
def proj_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    k = ind[cond].max()
    theta = cssv[k-1] / k
    return np.maximum(v - theta, 0)
```
