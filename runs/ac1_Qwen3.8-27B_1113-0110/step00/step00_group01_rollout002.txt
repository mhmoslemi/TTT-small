```python
import numpy as np
from scipy.optimize import linprog  # or use cvxpy

def solve_lp(a, M):
    # Maximize sum(g) subject to (a*g)[k] <= M for all k, g >= 0
    # (a*g)[k] = sum_{i: 0<=i<n, 0<=k-i<n} a[i]*g[k-i]
    # For each k, the constraint involves g[j] for j in [max(0, k-(n-1)), min(n-1, k)]
    # with coefficient a[k-j]
    
    n = len(a)
    # c = -1 (we minimize -sum(g) to maximize sum(g))
    c = -np.ones(n)
    
    # A_ub @ g <= b_ub
    # For each k in range(2*n-1):
    #   sum_j a[k-j] * g[j] <= M
    #   where j ranges over valid indices
    
    # Build A_ub (2n-1) x n matrix
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k - (n-1))
        hi = min(n-1, k)
        for j in range(lo, hi+1):
            A_ub[k, j] = a[k-j]
    
    b_ub = np.full(2*n-1, M)
    
    # Solve
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    return result.x
```
