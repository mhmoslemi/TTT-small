def compute_smooth_max_gradient(a, T):
    """Compute the gradient of the smooth max of conv(a,a)."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = c.max()
    # Softmax weights
    shifted = (c - max_c) / T
    # Numerical safety
    shifted = np.clip(shifted, -50, 50)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    # Gradient: g[j] = 2 * sum_k w[k] * a[k-j] (for valid k-j)
    # = 2 * cross_correlation(w, a)[j]
    a_flipped = a[::-1]
    conv_full = np.convolve(w, a_flipped, mode='full')
    g = 2.0 * conv_full[n-1 : 2*n-1]
    return g, c, max_c
