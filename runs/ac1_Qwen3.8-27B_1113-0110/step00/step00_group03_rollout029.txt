import time
from scipy.optimize import linprog

def propose_candidate(budget_s=1800):
    # Initialize
    n = 2535
    f = np.ones(n, dtype=np.float64)
    # Normalize: sum = sqrt(2n)
    target_sum = np.sqrt(2 * n)
    f = f * target_sum / np.sum(f)
    
    M = np.max(np.convolve(f, f))
    best_score = float(evaluate_sequence(f.tolist()))
    best_f = f.copy()
    
    deadline = time.monotonic() + budget_s - 10
    
    t = 0.5  # step size
    
    while time.monotonic() < deadline:
        # Solve LP: maximize sum b subject to (f*b)[k] <= M, b >= 0
        # Variables: b_0, ..., b_{n-1}
        # Objective: minimize -sum b
        c = -np.ones(n)
        
        # Constraints: (f*b)[k] <= M for k = 0, ..., 2n-2
        # (f*b)[k] = sum_i f[i] * b[k-i] for valid i
        # This is a sparse matrix A * b <= M
        # A has shape (2n-1) x n
        # A[k, j] = f[k-j] if 0 <= k-j <= n-1, else 0
        #           = f[j-k+n-1] ... hmm let me think
        
        # Actually: (f*b)[k] = sum_{j} b[j] * f[k-j] where 0 <= j <= n-1 and 0 <= k-j <= n-1
        # So A[k, j] = f[k-j] if 0 <= k-j <= n-1, else 0
        
        # For efficiency, build A as a sparse matrix
        from scipy.sparse import csr_matrix
        rows = []
        cols = []
        vals = []
        for j in range(n):
            # b[j] contributes to (f*b)[k] for k = j, j+1, ..., j+n-1
            # A[k, j] = f[k-j] for k = j, ..., j+n-1
            k_range = range(j, j + n)
            rows.extend(k_range)
            cols.extend([j] * n)
            vals.extend(f)
        
        A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
        b_ub = np.full(2*n-1, M)
        
        # Solve LP
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
        
        if not res.success:
            break
        
        g = res.x
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        
        # Rescale g to match target_sum
        g = g * target_sum / g_sum
        
        # Update: f <- (1-t)*f + t*g
        f = (1 - t) * f + t * g
        
        # Check score
        score = float(evaluate_sequence(f.tolist()))
        if np.isfinite(score) and score < best_score:
            best_f = f.copy()
            best_score = score
        
        # Update M
        M = np.max(np.convolve(f, f))
        
        # Maybe reduce step size
        # t = max(0.1, t * 0.99)
    
    return best_f.tolist()

def lp_step(f, M, n):
    """One LP step: find g maximizing sum(g) s.t. (f@g) <= M, g >= 0."""
    c = -np.ones(n)
    A = build_convolve_sparse(n, f)
    b_ub = np.full(2*n-1, M)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
    if res.success:
        return np.clip(res.x, 0, None)
    return None

def run_lp_optimization(f_init, n, max_time=60.0):
    """Run LP-based optimization."""
    f = np.clip(np.asarray(f_init, dtype=np.float64), 0, 1000).copy()
    if np.sum(f) < 1e-10:
        f = np.ones(n)
        n = f.size
    
    n = f.size
    target_sum = np.sqrt(2.0 * n)
    f = f * target_sum / np.sum(f)
    
    M = np.max(np.convolve(f, f))
    start = time.monotonic()
    t = 0.3
    stagnant = 0
    
    for it in range(50000):
        if time.monotonic() - start > max_time:
            break
        
        g = lp_step(f, M, n)
        if g is None:
            break
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g * target_sum / g_sum
        
        f_new = (1-t)*f + t*g
        f_new = np.clip(f_new, 0, None)
        new_M = np.max(np.convolve(f_new, f_new))
        
        if new_M < M * (1 - 1e-9):
            f = f_new
            M = new_M
            stagnant = 0
        else:
            stagnant += 1
            t *= 0.5
            if t < 1e-10:
                break
            if stagnant > 20:
                break
    
    score = float(evaluate_sequence(f.tolist()))
    return f, score

def local_polish(f, n, max_time=30.0):
    """Local perturbation to polish the solution."""
    f = np.clip(f.astype(np.float64), 0, None).copy()
    target_sum = np.sqrt(2.0 * n)
    f = f * target_sum / np.sum(f)
    best_score = float(evaluate_sequence(f.tolist()))
    best_f = f.copy()
    rng = np.random.default_rng(123)
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        # Perturb a random block
        width = rng.integers(max(1, n//50), max(2, n//10))
        s = rng.integers(0, n - width + 1)
        candidate = best_f.copy()
        candidate[s:s+width] *= np.exp(rng.normal(0, 0.1, size=width))
        candidate = np.clip(candidate, 0, None)
        score = float(evaluate_sequence(candidate.tolist()))
        if score < best_score:
            best_score = score
            best_f = candidate
    
    return best_f, best_score
