"""
Minimize 2n*max(conv(a,a))/(sum(a))^2 using smoothed gradient descent (Adam)
with FFT-accelerated convolution, log-sum-exp smoothing with beta annealing,
and periodic random restarts for diversification.
"""
import numpy as np
from scipy.fft import rfft, irfft
import time


def _fft_conv_self(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_n = 1 << int(np.ceil(np.log2(size)))
    A = rfft(a, fft_n)
    C = irfft(A * A, fft_n)[:size]
    return C


def _fast_score(a):
    """Fast computation of 2n*max(conv(a,a))/(sum(a))^2 using FFT."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    C = _fft_conv_self(a)
    max_C = np.max(C)
    return 2.0 * n * max_C / (s * s)


def _softmax_grad(a, C, beta):
    """Compute gradient of (1/beta)*log(sum exp(beta*C)) w.r.t. a."""
    n = len(a)
    max_C = np.max(C)
    shifted = beta * (C - max_C)
    shifted = np.clip(shifted, -500, 500)
    exps = np.exp(shifted)
    p = exps / exps.sum()
    # grad[m] = 2 * sum_k p[k] * a[k-m] (valid range) = 2 * correlate(p, a, 'valid')
    g = 2.0 * np.correlate(p, a, 'valid')
    return g


def _adam_step(a, g, m, v, t, b1, b2, lr, eps):
    """Perform one Adam step with non-negativity projection and normalization."""
    n = len(a)
    m = b1 * m + (1.0 - b1) * g
    v = b2 * v + (1.0 - b2) * g * g
    m_hat = m / (1.0 - b1 ** t)
    v_hat = v / (1.0 - b2 ** t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = np.maximum(a_new, 0.0)
    s = np.sum(a_new)
    if s < 1e-10:
        a_new = a
    else:
        a_new = a_new / s
    return a_new, m, v


def _make_initialization(n, rng, method):
    """Create various non-trivial initializations."""
    j = np.arange(n, dtype=np.float64)
    if method == 0:  # Uniform with small noise
        a = np.ones(n) + rng.normal(0, 0.05, n)
    elif method == 1:  # Cosine wave
        a = 1.0 + 0.3 * np.cos(2 * np.pi * j / n)
    elif method == 2:  # Linear ramp
        a = 0.5 + 1.0 * j / (n - 1)
    elif method == 3:  # Sine wave
        a = 1.0 + 0.4 * np.sin(2 * np.pi * j / n)
    elif method == 4:  # Random positive
        a = np.abs(rng.normal(1.0, 0.5, n))
    elif method == 5:  # Two-tone
        a = 1.0 + 0.2 * np.cos(4 * np.pi * j / n) + 0.1 * np.sin(6 * np.pi * j / n)
    else:  # Exponential decay modulated
        a = np.exp(-2.0 * j / n) + 0.3 * np.cos(2 * np.pi * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a /= s
    return a


def propose_candidate(seed=42, budget_s=1800):

    deadline = time.monotonic() + float(budget_s) - 15.0

    # Get initial sequence from global
    try:
        init_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000.0)
        n = init_seq.size
    except:
        n = 6121
        init_seq = np.ones(n)

    # Determine n
    n = int(n)
    
    # Initialize multiple candidates
    # We'll run multiple "runs" with different initializations, time-sliced
    
    best_a = init_seq.copy()
    s = best_a.sum()
    if s > 0:
        best_a = best_a / s
    best_score = _fast_score(best_a)
    # Verify with actual evaluator
    actual_score = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(actual_score):
        best_score = actual_score
    print(f"Initial score: {best_score:.6f}, n={n}")

    # Split time among different strategies
    t_start = time.monotonic()
    
    # Strategy: Run gradient descent with periodic restarts
    # Use multiple beta values and initializations
    
    # Pre-allocate
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    # Phase 1: Run gradient descent from the provided sequence
    # with symmetry-breaking perturbation
    a = best_a.copy()

    a = np.maximum(a, 0)
