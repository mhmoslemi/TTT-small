import numpy as np
from scipy.fft import fft, ifft
import time

def project_onto_simplex(v, target_sum=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    # Sort in descending order
    u = np.sort(v)[::-1]
    # Find rho
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def auto_convolve(f):
    """Compute auto-convolution of f using FFT."""
    n = len(f)
    size = 2 * n - 1
    F = fft(f, n=size)
    return ifft(F * F).real

def smooth_max_and_weights(C, beta):
    """Compute smooth max and softmax weights."""
    C_max = np.max(C)
    exp_C = np.exp(beta * (C - C_max))
    sum_exp = np.sum(exp_C)
    smooth_max = C_max + np.log(sum_exp) / beta
    weights = exp_C / sum_exp
    return smooth_max, weights

def compute_gradient(f, beta):
    """Compute gradient of smooth max of auto-convolution."""
    n = len(f)
    C = auto_convolve(f)
    _, p = smooth_max_and_weights(C, beta)
    
    # Gradient: g_j = 2 * sum_m f_m * p[m+j]
    # This is 2 * cross-correlation of f and p
    size = 2 * n - 1  # sufficient for linear cross-correlation
    # Actually, we need size such that no wrap-around occurs
    # f has support [0, n-1], p has support [0, 2n-2]
    # cross-correlation at lag j uses f[m] * p[m+j] for m in [0, n-1]
    # max index of p used: (n-1) + (n-1) = 2n-2, which is within p's support
    # So size = 2n-1 is sufficient
    
    F_full = fft(f, n=size)
    P_full = fft(p, n=size)
    cross_corr = ifft(np.conj(F_full) * P_full).real
    grad = 2.0 * cross_corr[:n]
    return grad, C
