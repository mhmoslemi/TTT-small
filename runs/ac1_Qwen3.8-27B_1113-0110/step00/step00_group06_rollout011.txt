def compute_score_and_grad(a):
    """Compute score and gradient for a non-negative sequence."""
    n = len(a)
    conv = np.convolve(a, a)
    max_idx = np.argmax(conv)
    max_val = conv[max_idx]
    S = np.sum(a)
    if S < 1e-10:
        return np.inf, None
    score = 2.0 * n * max_val / (S * S)

    # d(score)/da_i = 2n * (g_i * S - 2 * max_val) / S^3
