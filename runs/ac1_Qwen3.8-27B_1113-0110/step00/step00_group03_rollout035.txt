"""
Minimize score = 2*n*max(conv(a,a))/sum(a)^2 using FFT-accelerated subgradient
descent with multiplicative updates, combined with block perturbation refinement
and multiple restart strategies. Exploits the fact that non-symmetric non-negative
sequences can achieve convolution peaks below sum(a)^2/n, yielding scores < 2.
"""

import numpy as np
import time


def fast_convolve(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Pad to next power of 2 for speed
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    fa = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(fa * fa, fft_size)[:size]
    return conv


def compute_score_fast(a):
    """Compute score using FFT (may have tiny numerical errors vs np.convolve)."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    conv = fast_convolve(a)
    max_b = conv.max()
    return 2.0 * n * max_b / (s * s)


def subgradient_direction(a, conv, n, num_peaks=10, tolerance=0.999):
    """Compute subgradient of max(conv) w.r.t. a."""
    max_val = conv.max()
    # Find all near-max positions
    threshold = max_val * tolerance
    peak_indices = np.where(conv >= threshold)[0]
    # Limit to top peaks
    if len(peak_indices) > num_peaks:
        # Pick the num_peaks largest
        peak_vals = conv[peak_indices]
        top_idx = np.argsort(peak_vals)[-num_peaks:]
        peak_indices = peak_indices[top_idx]
    
    subgrad = np.zeros(n)
    weights = np.zeros(len(peak_indices))
    for idx, k in enumerate(peak_indices):
        w = (conv[k] - threshold) / max(1e-15, max_val - threshold) + 0.1
        weights[idx] = w
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # subgrad[i] += 2 * a[k - i] for i in [lo, hi]
        # a[k - i] for i in [lo, hi] is a[k-hi : k-lo+1] reversed
        subgrad[lo:hi+1] += 2.0 * weights[idx] * a[k - hi:k - lo + 1][::-1]
    
    return subgrad


def multiplicative_descent(a, n, initial_lr, max_iters, deadline, verbose_interval=5000):
    """Run multiplicative gradient descent to minimize max convolution."""
    lr = initial_lr
    best = a.copy()
    best_score = compute_score_fast(best)
    no_improve = 0
    max_no_improve = 2000
    
    size = 2 * n - 1
    fft_size = 1
    while fft_size < 1024:
        fft_size *= 2
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute convolution via FFT
        fa = np.fft.rfft(best, fft_size)
        conv = np.fft.irfft(fa * fa, fft_size)[:size]
        
        # Get subgradient direction
        subgrad = subgradient_direction(best, conv, n, num_peaks=10)
        
        # Normalize subgradient for stable updates
        sg_max = subgrad.max()
        if sg_max < 1e-15:
            break
        subgrad_normed = subgrad / sg_max
        
        # Multiplicative update: a[i] *= exp(-lr * subgrad_normed[i])
        update = np.exp(-lr * subgrad_normed)
        candidate = best * update
        candidate = np.maximum(candidate, 1e-15)
        s = candidate.sum()
        candidate = candidate / s
        
        # Fast score check
        fca = np.fft.rfft(candidate, fft_size)
        conv_new = np.fft.irfft(fca * fca, fft_size)[:size]
        new_score = 2.0 * n * conv_new.max() / 1.0  # sum is 1
        
        if new_score < best_score * (1 - 1e-8):
            best = candidate
            best_score = new_score
            no_improve = 0
        else:
            no_improve += 1
            # Still move slightly for exploration
            t = 0.3
            candidate = (1 - t) * best + t * candidate
            candidate = np.maximum(candidate, 1e-15)
            candidate = candidate / candidate.sum()
            best = candidate
            # Recompute best_score occasionally
            if no_improve % 50 == 0:
                best_score = compute_score_fast(best)
            if no_improve > max_no_improve:
                lr *= 0.7
                no_improve = 0
                if lr < 1e-8:
                    break
        
        if it % verbose_interval == 0 and it > 0:
            print(f"  MD iter {it}: score={best_score:.8f}, lr={lr:.6f}")
    
    return best, best_score


def block_perturb_refine(a, n, deadline, rng, max_evals=5000):
    """Refine using block multiplicative perturbations."""
    best = a.copy()
    best_score = compute_score_fast(best)
    eval_count = 0
    no_improve = 0
    
    while time.monotonic() < deadline and eval_count < max_evals:
        candidate = best.copy()
        # Random number of blocks to perturb
        n_blocks = int(rng.integers(1, 4))
        for _ in range(n_blocks):
            min_w = max(1, n // 500)
            max_w = max(min_w + 1, n // 20)
            width = int(rng.integers(min_w, max_w))
            start = int(rng.integers(0, max(1, n - width + 1)))
            # Multiplicative perturbation
            factor = np.exp(rng.normal(0, 0.08))
            candidate[start:start + width] *= factor
        
        candidate = np.maximum(candidate, 0)
        s = candidate.sum()
        if s < 0.01:
            eval_count += 1
            continue
        candidate = candidate / s
        eval_count += 1
        
        score = compute_score_fast(candidate)
        if score < best_score:
            best = candidate
            best_score = score
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:

    return best, best_score


def try_initialization(init_type, n, rng):
    """Generate different types of initial sequences."""
    if init_type == 0:  # Try parent

        except:
            pass
        return np.ones(n) / n
    elif init_type == 1:  # Uniform
        return np.ones(n) / n
    elif init_type == 2:  # Linear ramp
        a = np.arange(1, n + 1, dtype=np.float64)
        return a / a.sum()
    elif init_type == 3:  # Inverse ramp
        a = np.arange(n, 0, -1, dtype=np.float64)
        return a / a.sum()
    elif init_type == 4:  # Sinusoidal modulation
        a = 1.0 + 0.95 * np.sin(2 * np.pi * np.arange(n) / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif init_type == 5:  # Cosine (peaked center)
        a = np.cos(np.pi * np.arange(n) / n) ** 2
        a = a + 0.01
        return a / a.sum()
    elif init_type == 6:  # Exponential decay
        a = np.exp(-np.arange(n, dtype=np.float64) / (n * 0.3))
        return a / a.sum()
    elif init_type == 7:  # Random exponential
        a = rng.exponential(1.0, size=n)
        return a / a.sum()
    elif init_type == 8:  # Two-level random (Barker-like)
        a = rng.choice([1.0, 0.3], size=n).astype(np.float64)
        return a / a.sum()
    elif init_type == 9:  # Smooth random (low-pass noise)
        a = rng.standard_normal(n)
        # Low-pass filter
        fft_a = np.fft.rfft(a)
        freqs = np.arange(len(fft_a))
        cutoff = len(fft_a) // (2 * n)
        fft_a[freqs > cutoff] *= 0.01
        a = np.fft.irfft(fft_a, n)
        a = np.abs(a) + 0.1
        return a / a.sum()
    else:  # Mixed
        a = 0.5 * np.ones(n) + 0.5 * rng.standard_normal(n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
