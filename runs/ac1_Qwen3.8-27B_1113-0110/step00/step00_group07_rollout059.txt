    inits.append((rng.random(n) * 0.5 + 0.5) * (1.0 - 0.6 * x))
    inits.append(np.abs(rng.standard_normal(n)) + 0.1)

    best = None
    best_score = np.inf
    best_mc = np.inf

    print("Evaluating initializations...")
    for i, init in enumerate(inits):
        if deadline - time.monotonic() < 2.0:
            break
        init = np.clip(init.astype(np.float64), 0.0, 1000.0)
        ssum = init.sum()
        if ssum < 0.01:
            continue
        init = init * (S / ssum)
        mc = fast_conv(init).max()
        score = 2.0 * n * mc
        if mc < best_mc:
            best_mc = mc
            best = init.copy()
            best_score = score
            print(f"  Init {i}: score={score:.6f}")

    if best is None:
        best = np.ones(n) / n
        best_score = 2.0
        best_mc = 1.0 / n

    print(f"Best initial score: {best_score:.6f}")

    # Phase 1: Optimize best init
    sub_deadline = deadline - 30.0
    best, best_mc = optimize_from(best, n, S, sub_deadline, rng)
    if best_mc < best_mc:
        pass
    print(f"After phase 1: fast_score={1.0/(best_mc/(2.0*n)): .6f} -> actual target {2.0*n*best_mc:.6f}")
    print(f"After phase 1: maxconv={best_mc:.6e}, score={2.0*n*best_mc:.6f}")

    # Verify
    verify_score = evaluate_sequence(best.tolist())
    if np.isfinite(verify_score) and verify_score < best_score:
        best_score = verify_score

    # Phase 2: Try other inits with remaining time
    remaining = deadline - time.monotonic()
    if remaining > 60:
        for i, init in enumerate(inits):
            if i == 0 and len(inits) > 1:
                pass  # already tried
            if deadline - time.monotonic() < 30:
                break
            init = np.clip(init.astype(np.float64), 0.0, 1000.0)
            ssum = init.sum()
            if ssum < 0.01:
                continue
            init = init * (S / ssum)
            mc = fast_conv(init).max()
            if mc >= best_mc * 1.1:
                continue  # not promising
            # Optimize this one with limited time
            sub_dl = min(deadline - 10.0, time.monotonic() + 60.0)
            a_opt, mc_opt = optimize_from(init, n, S, sub_dl, rng)
            if mc_opt < best_mc:
                best = a_opt
                best_mc = mc_opt
                print(f"  Init {i} improved: score={2.0*n*mc_opt:.6f}")
                verify_score = evaluate_sequence(best.tolist())
                if np.isfinite(verify_score) and verify_score < best_score:
                    best_score = verify_score

    # Phase 3: Final polish with small steps
    if deadline - time.monotonic() > 20:
        a = best.copy()
        f_lb = 1.0 / (2.0 * n - 1.0)
        for polyak_scale in [0.1, 0.3, 0.5, 1.0]:
            if deadline - time.monotonic() < 10:
                break
            aa = best.copy()
            for _ in range(100):
                if deadline - time.monotonic() < 8:
                    break
                conv = fast_conv(aa)
                mc = conv.max()
                g = topk_grad(aa, conv, 1)
                gn = np.linalg.norm(g)
                if gn < 1e-18:
                    break
                step = polyak_scale * (mc - f_lb) / (gn * gn)
                step = min(step, 0.1 * S / n)
                v = aa - step * g
                aa = proj_simplex(v, S)
            mc_final = fast_conv(aa).max()
            if mc_final < best_mc:
                best = aa
                best_mc = mc_final
                print(f"  Polish (scale={polyak_scale}): score={2.0*n*mc_final:.6f}")

    # Final verification
    final_score = evaluate_sequence(best.tolist())
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"FINAL: score={best_score:.6f}, maxconv={best_mc:.6e}")
    return [float(xi) for xi in best]
