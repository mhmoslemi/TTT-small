Softmax gradient descent with FFT-based convolution to minimize
2*n*max(conv(a,a))/sum(a)^2. Uses multiple initializations, adaptive
hyperparameters, momentum, and population mixing to beat score 2.0.
"""

import numpy as np
import time
import sys

def fft_autocorr(a):
    """Compute full linear autocorrelation a*a using FFT."""
    n = len(a)
    L = 1
    while L < 2 * n - 1:
        L *= 2
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)
    return conv[:2 * n - 1]

def compute_grad(a, w):
    """Compute gradient of softmax surrogate. w has length 2n-1, a has length n.
    Returns 2*n array: grad[j] = 2 * sum_m a[m] * w[m+j] for j=0..n-1."""
    n = len(a)
    L = 1
    while L < 3 * n - 1:
        L *= 2
    A = np.fft.rfft(a, L)
    Zf = np.fft.rfft(w[::-1], L)
    conv_az = np.fft.irfft(A * Zf, L)
    return 2.0 * conv_az[n - 1:2 * n - 1][::-1]
