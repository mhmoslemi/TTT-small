Minimizes 2*n*max(conv(a,a))/sum(a)^2 using smooth-max (log-sum-exp) 
projected gradient descent. The score is scale-invariant, so we normalize 
sum(a)=1 and minimize max(conv(a,a)).

import numpy as np
import time


def compute_smooth_gradient(a, beta, n_fft):
    """Compute gradient of smooth max of conv(a,a) w.r.t. a."""
    n = len(a)
    c = np.convolve(a, a)
    # Log-sum-exp softmax
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    p = exp_bc / exp_bc.sum()
    # Gradient: grad[j] = 2 * sum_m a[m] * p[j+m]
    # Using FFT: ifft(conj(FFT(a)) * FFT(p))
    A = np.fft.rfft(a, n=n_fft)
    P = np.fft.rfft(p, n=n_fft)
    grad = 2.0 * np.real(np.fft.irfft(np.conj(A) * P, n=n_fft)[:n])
    return p, grad
