import numpy as np
import time

def _self_convolve_fft(a):
    """Compute self-convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def _compute_score(C, n):
    """Compute score from convolution and sequence length. Assumes sum(a)=1."""
    return 2.0 * n * C.max()

def _peak_shaving_iteration(a, C, K=5, eta=0.3):
    """One iteration of peak shaving. Returns updated a."""
    n = len(a)
    # Find top K peaks
    # Use argpartition for efficiency
    if K >= len(C):
        hot_idx = np.arange(len(C))
    else:
        # Get indices of top K values
        part = np.argpartition(C, -K)[-K:]
        hot_idx = part
    
    # Weight by how much above mean
    C_mean = C.mean()
    C_std = C.std() + 1e-15
    weights = np.maximum(0, C[hot_idx] - C_mean) / C_std
    
    # Compute accumulated gradient
    grad = np.zeros(n)
    for k in hot_idx:
        w = weights[np.where(hot_idx == k)[0][0]]
        if w <= 0:
            continue
        j_min = max(0, k - (n-1))
        j_max = min(n-1, k)
        idx = k - np.arange(j_min, j_max + 1)
        grad[j_min:j_max+1] += w * 2.0 * a[idx]
    
    # Multiplicative update
    max_grad = grad.max()
    if max_grad > 0:
        a = a * np.exp(-eta * grad / max_grad)
    
    # Clip and normalize
    a = np.clip(a, 0.0, None)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / S
    
    return a

    inits = []
    
    # 1. Parent (normalized)

    if S_p > 0:

    # 2. Random

def _fft_self_conv(a):
    """Fast self-convolution via FFT."""
    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    if size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def _multi_peak_grad(a, C, K, alpha):
    """Compute weighted gradient from top-K peaks.
    
    alpha controls how much weight to give to sub-peaks vs the max.
    """
    n = len(a)
    C_sorted = np.sort(C)[::-1]
    # Weight: linear decay from max
    max_C = C_sorted[0]
    # Use top K peaks

        idx = k - np.arange(j_min, j_max + 1)

Let me also think about an alternative approach: **iterative proportional fitting (IPF) / Sinkhorn-style.**

We want C[k] ≈ c for all k (constant). This is a system of quadratic equations:
sum_i a[i]*a[k-i] = c for all k

This is like a "matrix factorization" problem: the convolution matrix C = a ⊛ a should be as flat as possible.

One approach: alternate between fixing the "rows" and "columns" (positions i and k-i) to satisfy the constraints. But the quadratic nature makes this tricky.

Another idea: **use the structure of the convolution to design the sequence.**

The convolution C[k] = sum_i a[i]*a[k-i] can be seen as: for each "diagonal" k in the outer product a ⊗ a, the sum is C[k]. We want all diagonals to have equal sum.

This is similar to the "magic square" problem but for 1D outer products. For a 1D problem, it's the "Pandigital" or "balanced" sequence problem.

An exact solution: if we could make C[k] = c for all k, then the DFT of C is zero for all non-DC frequencies, meaning |A(f)|² = 0 for f ≠ 0. This means A(f) = 0 for f ≠ 0, so a is a constant (all a[i] = c). But then C[0] = a[0]² = c² and C[n-1] = n*c², which are not equal unless n=1. Contradiction!

So a perfectly flat convolution is impossible. The best we can do is minimize the variance of C.

import numpy as np
import time

"""
Iterative FFT-based peak-shaving for minimizing 2n*max(self-conv(a))/sum(a)^2.
Uses multi-peak gradient descent with adaptive steps, noise for symmetry-breaking,
and multiple initializations.
"""

def _fft_conv(a):
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def _grad_from_peaks(a, C, K):
    n = len(a)
    if K >= len(C):
        peaks = np.arange(len(C))
    else:
        peaks = np.argpartition(C, -K)[-K:]
    grad = np.zeros(n)
    for k in peaks:
        w = C[k]
        jmn = max(0, k - n + 1)
        jmx = min(n - 1, k)
        idx = k - np.arange(jmn, jmx + 1)
        grad[jmn:jmx + 1] += w * 2.0 * a[idx]
    return grad

def _step(a, C, eta, K, noise, rng):
    n = len(a)
    grad = _grad_from_peaks(a, C, K)
    # Centered gradient (preserves sum)
    mg = grad.mean()
    gs = grad.std()

import numpy as np
import time
from scipy.signal import convolve as _sp_conv

"""
FFT-based iterative peak-shaving to minimize 2n*max(self-conv(a))/sum(a)^2.
Uses multi-peak centered-gradient descent, adaptive step/noise, multiple
initializations (uniform, random, structured), partial-randomization restarts,
and a final refinement pass on the best candidate found.
"""

def _fft_conv(a):
    """Self-convolution via FFT (size chosen as next power of 2 >= 2n-1)."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    out = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    return np.maximum(out, 0.0)


def _grad_peaks(a, C, K):
    """Weighted sum of gradients from top-K peaks of C."""
    n = len(a)
    K = min(K, len(C))
    peaks = np.argpartition(C, -K)[-K:]
    grad = np.zeros(n)
    for k in peaks:
        w = C[k]
        jmn = max(0, k - n + 1)
        jmx = min(n - 1, k)
        idx = k - np.arange(jmn, jmx + 1)
        grad[jmn : jmx + 1] += 2.0 * w * a[idx]
    return grad


def _shave_step(a, C, eta, K, noise, rng):
    """One centered-gradient + noise step; keeps a >= 0, sum(a)=1."""
    n = len(a)
    grad = _grad_peaks(a, C, K)
    mg = grad.mean()
    gs = grad.std()
    if gs > 1e-18:
        a = a - eta * (grad - mg) / gs
    a += noise * rng.standard_normal(n)
    np.maximum(a, 0.0, out=a)
    s = a.sum()
    if s > 1e-15:
        a /= s
    else:
        a[:] = 1.0 / n
    return a
