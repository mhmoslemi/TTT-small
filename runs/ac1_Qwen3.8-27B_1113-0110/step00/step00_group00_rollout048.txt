def _compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Next power of 2 for efficient FFT
    next_p2 = 1
    while next_p2 < size:
        next_p2 <<= 1
    f = np.fft.rfft(a, n=next_p2)
    conv = np.fft.irfft(f * f)[:size]
    return conv


def _grad_conv_at_lag(a, m):
    """
    Gradient of conv(a,a)[m] with respect to a.
    conv(a,a)[m] = sum_j a[j]*a[m-j] for valid j.
    d/d(a[j]) = 2*a[m-j] if m-j != j, else a[m-j] (= a[j]).
    """
    n = len(a)
    g = np.zeros(n)
    lo = max(0, m - n + 1)
    hi = min(n - 1, m)
    if lo > hi:
        return g
    idx = np.arange(lo, hi + 1)
    g[idx] = 2.0 * a[m - idx]
    if m % 2 == 0:
        mid = m // 2
        if lo <= mid <= hi:
            g[mid] = a[mid]
    return g


def _project_simplex(x):
    """
    Project x onto the probability simplex {a >= 0, sum(a) = 1}.
    Uses the standard algorithm: sort descending, find threshold via cumulative sums.
    """
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    # Find largest rho such that u[rho] > (cssv[rho] - 1) / (rho + 1)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    # Find last True
    true_indices = np.where(cond)[0]
    if len(true_indices) == 0:
        return np.ones(n) / n
    rho = true_indices[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(x - theta, 0.0)
