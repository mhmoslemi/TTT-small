```python
    data = np.tile(a, n)
    C = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return C

def lp_iteration(a, M):
    """Solve the LP to find the best direction b."""
    n = len(a)
    C = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = M * np.ones(2*n-1)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

        M = np.max(conv)
        current_score = 2 * n * M / np.sum(a)**2
        
        if current_score < best_score:
            best_score = current_score
            best_seq = a.copy()
        
        # LP iterations
        t_start = time.monotonic()
        iters = 0
        while time.monotonic() < deadline - 30:

            if b is None:
                break
            
            sum_b = np.sum(b)
            sum_a = np.sum(a)
            if sum_b < 1e-10:
                break
            
            # Rescale b to match sum of a
            b = b * sum_a / sum_b
            
            # Check if b is significantly different from a
            if np.allclose(b, a, atol=1e-6):
                break
            
            # Update with step size
            t = 0.5  # step size
            a_new = (1-t)*a + t*b
            a_new = np.clip(a_new, 0, 1000)

                a = a_new
                current_score = score_new

                    best_score = current_score
                    best_seq = a.copy()
                    print(f"n={n}, iter={iters}, score={current_score:.6f}")
```

Optimization: since the LP constraint matrix depends on `a`, and `a` changes each iteration, we need to rebuild it each time. But we can pre-allocate the structure (rows, cols) and only update the data.

Actually, the structure (rows, cols) is FIXED! It only depends on `n`. Only the data changes (it's `a` tiled). So we can pre-build the structure.

```python
def build_conv_structure(n):
    i_vals = np.tile(np.arange(n), n)
    j_vals = np.repeat(np.arange(n), n)
    rows = i_vals + j_vals
    cols = i_vals
    return rows, cols

def make_conv_matrix(a, rows, cols):
    n = len(a)
    data = np.tile(a, n)
    return sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
```
