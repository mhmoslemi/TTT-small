"""
Dense local search for low-peak nonnegative autocorrelation.

The program combines:
- smooth softmax/log-sum-exp objectives on the simplex,
- L-BFGS-B refinement of discretely sampled step functions,
- low-dimensional smooth ansatz searches,

- active-peak mirror/Euclidean line searches using the exact evaluator-compatible score.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize as _MINIMIZE
except Exception:
    _MINIMIZE = None


def score_key(item):
    return item[1]


def strict_score(a):
    """Exact direct score, used for final candidate selection."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    if n == 0:
        return float("inf")
    s = float(np.sum(a))
    if s < 1e-12 or not np.isfinite(s):
        return float("inf")
    C = np.convolve(a, a)
    m = float(np.max(C))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m / (s * s))


def normalize_and_score(a):
    """Normalize to sum 1 and return (a, score)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    if a.size == 0:
        return np.ones(1, dtype=np.float64), float("inf")
    s = float(np.sum(a))
    if s < 1e-12 or not np.isfinite(s):
        a = np.ones_like(a, dtype=np.float64) / float(a.size)
        s = 1.0
    else:
        a = a / s
    n = a.size
    C = linear_convolve(a)
    m = float(np.max(C))
    if not np.isfinite(m):
        return a, float("inf")
    return a, float(2.0 * n * m)


def linear_convolve(a):
    """Linear full convolution a*a. Direct for modest sizes, FFT for larger sizes."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    L = 2 * n - 1
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n <= 1200:
        c = np.convolve(a, a)
    else:
        N = 1
        while N < L:
            N <<= 1
        fa = np.fft.rfft(a, N)
        fc = fa * fa
        c = np.fft.irfft(fc, N)[:L]
    return np.where(c > 0.0, c, 0.0)


def gradient_corr(q, a):
    """
    h_j = sum_{d=0}^{n-1} a_d q_{j+d}, for j=0..n-1.

    This is the gradient kernel for C = convolve(a, a):
    d sum_k q_k C_k / d a_j = 2 h_j.
    """
    n = a.size
    if n == 0 or q.size != 2 * n - 1:
        return np.zeros(n, dtype=np.float64)
    if n <= 2000:
        h = np.correlate(q, a, mode="valid")
        h = np.asarray(h, dtype=np.float64)
        if h.size != n:
            ar = a[::-1]
            r = np.convolve(q, ar, mode="full")
            h = r[n - 1:n - 1 + n]
    else:
        ar = a[::-1]
        N = 1
        while N < 3 * n - 1:
            N <<= 1
        fq = np.fft.rfft(q, N)
        far = np.fft.rfft(ar, N)
        r = np.fft.irfft(fq * far, N)[:3 * n - 2]
        h = r[n - 1:n - 1 + n]
    h = np.asarray(h, dtype=np.float64)
    return np.where(h > 0.0, h, 0.0)


def make_a_from_u(u):
    """
    Convert softmax log-parameters with u[0] fixed to 0 into a probability vector.
    u has length n-1.
    """
    u = np.asarray(u, dtype=np.float64).ravel()
    n = u.size + 1
    if n <= 1:
        return np.ones(1, dtype=np.float64)
    theta = np.empty(n, dtype=np.float64)
    theta[0] = 0.0
    theta[1:] = u
    mx = float(np.max(theta))
    theta -= mx
    a = np.exp(theta)
    s = float(np.sum(a))
    if s <= 1e-300 or not np.isfinite(s):
        a = np.ones(n, dtype=np.float64) / float(n)
    else:
        a /= s
    return a


def softmax_lbfun(u, n, tau):
    """
    Smooth objective:
        J = n * (tau * logsumexp(C/tau)) = n * (M + tau*log Z)
    with gradient scaled to O(1).
    """
    u = np.asarray(u, dtype=np.float64).ravel()
    if u.size + 1 != n:
        n = u.size + 1
    if not np.all(np.isfinite(u)):
        return 1e100, np.zeros(u.size, dtype=np.float64)

    a = make_a_from_u(u)
    C = linear_convolve(a)
    if C.size == 0:
        return 1e100, np.zeros(u.size, dtype=np.float64)

    M = float(np.max(C))
    if not np.isfinite(M) or M <= 0.0:
        return 1e100, np.zeros(u.size, dtype=np.float64)

    tau = float(tau)
    if tau <= 1e-300 or not np.isfinite(tau):
        tau = 1e-12

    x = (C - M) / tau
    x = np.clip(x, -745.0, 0.0)
    ex = np.exp(x)
    Z = float(np.sum(ex))
    if Z <= 1e-300 or not np.isfinite(Z):
        return 1e100, np.zeros(u.size, dtype=np.float64)

    val = float(n) * (M + tau * float(np.log(Z)))
    q = ex / Z
    h = gradient_corr(q, a)
    grad_a = 2.0 * float(n) * h
    dot = float(np.dot(a, grad_a))
    grad_theta = grad_a - a * dot
    g = grad_theta[1:]
    if not np.all(np.isfinite(g)):
        g = np.zeros(u.size, dtype=np.float64)
    return val, g


def lrefine(a, maxiter, deadline=None):
    """L-BFGS-B refinement with tau annealing on the simplex."""
    if _MINIMIZE is None:
        return normalize_and_score(a)

    best, best_score = normalize_and_score(a)
    n = best.size
    if n <= 1:
        return best, best_score

    C = linear_convolve(best)
    Mref = max(float(np.max(C)), 1.0 / float(n))
    bounds = [(-20.0, 20.0)] * (n - 1)
    tau_fracs = (0.50, 0.20, 0.08, 0.03, 0.01)

    for frac in tau_fracs:
        if deadline is not None:
            if time.monotonic() > float(deadline) - 5.0:
                break
        tau = max(1e-12, Mref * float(frac))
        loga = np.log(np.maximum(best, 1e-300))
        u0 = np.clip(loga - loga[0], -20.0, 20.0)
        try:
            res = _MINIMIZE(
                softmax_lbfun,
                u0,
                method="L-BFGS-B",
                jac=True,
                args=(n, tau),
                bounds=bounds,
                options={
                    "maxiter": int(max(1, int(maxiter))),
                    "maxls": 30,
                    "ftol": 1e-12,
                    "gtol": 1e-6,
                    "disp": False,
                },
            )
            if res is not None and res.x is not None:
                b = make_a_from_u(res.x)
                b, bs = normalize_and_score(b)
                if np.isfinite(bs) and bs < best_score:
                    best = b
                    best_score = float(bs)
                C = linear_convolve(best)
                Mref = max(float(np.max(C)), 1.0 / float(n))
        except Exception:
            pass

    return best, float(best_score)


def resample_a(a, new_n):
    """Interpolate log-density to a new length and renormalize."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(max(1, int(new_n)), dtype=np.float64)
    if a.size == int(new_n):
        a, _ = normalize_and_score(a)
        return a

    old_n = a.size
    loga = np.log(np.maximum(a, 1e-300))
    xold = (np.arange(old_n, dtype=np.float64) + 0.5) / float(old_n)
    xnew = (np.arange(int(new_n), dtype=np.float64) + 0.5) / float(int(new_n))
    newlog = np.interp(xnew, xold, loga)
    b = np.exp(newlog - float(np.max(newlog)))
    s = float(np.sum(b))
    if s <= 1e-300 or not np.isfinite(s):
        b = np.ones(int(new_n), dtype=np.float64) / float(int(new_n))
    else:
        b /= s
    return b


def add_candidate(cand, a, n):
    """Normalize, score, and append if finite."""
    a, sc = normalize_and_score(a)
    if np.isfinite(sc):
        cand.append((a, float(sc), int(a.size)))
    return cand


def gen_inits(n, rng):
    """Generate diverse smooth nonnegative initial sequences of length n."""
    n = int(n)
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    out = []

    out.append(np.ones(n, dtype=np.float64))

    for d in (-2.0, -1.5, -1.0, -0.5, 0.0, 0.5, 1.0, 1.5, 2.0, 3.0, 4.0):
        out.append(np.exp(d * x))

    quad_params = (
        (-0.5, 0.2), (0.5, -0.2), (1.0, -1.0), (-1.0, 1.0),
        (2.0, 1.0), (-2.0, -1.0), (1.5, -2.0), (-1.5, 2.0),
        (0.25, -0.25), (0.75, -0.75)
    )
    for p1, p2 in quad_params:
        out.append(np.exp(p1 * x + p2 * x * x))

    width = max(3, n // 20)
    kern = np.ones(width, dtype=np.float64) / float(width)
    for _ in range(10):
        v = rng.normal(0.0, 0.7, size=n)
        v = np.convolve(v, kern, mode="same")
        out.append(np.exp(v))

    for _ in range(10):
        c = rng.normal(0.0, 0.6, size=5)
        v = (
            c[0] * x + c[1] * x * x
            + c[2] * np.sin(2.0 * np.pi * x + c[3])
            + c[4] * np.cos(4.0 * np.pi * x + c[1])
        )
        out.append(np.exp(v))

    return out


def low_dim_a(theta, n):
    """Low-dimensional smooth ansatz: polynomial plus Fourier modes."""
    n = int(n)
    t = np.asarray(theta, dtype=np.float64).ravel()
    if t.size < 12:
        tp = np.zeros(12, dtype=np.float64)
        tp[:t.size] = t
        t = tp
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    v = t[0] * x + t[1] * x * x + t[2] * x * x * x + t[3] * (x ** 4)
    idx = 4
    for k in range(1, 5):
        A = float(t[idx])
        B = float(t[idx + 1])
        idx += 2
        w = 2.0 * np.pi * float(k) * x
        v += A * np.cos(w) + B * np.sin(w)
    v = np.clip(v, -50.0, 50.0)
    a = np.exp(v)
    s = float(np.sum(a))
    if s <= 1e-300 or not np.isfinite(s):
        a = np.ones(n, dtype=np.float64) / float(n)
    else:
        a /= s
    return a


def low_dim_score(theta, n):
    a = low_dim_a(theta, int(n))
    _, sc = normalize_and_score(a)
    if not np.isfinite(sc):
        return 1e100
    return float(sc)


def low_dim_optimize(n, rng, deadline):
    """Nelder-Mead search over a low-dimensional smooth family."""
    n = int(n)
    dim = 12
    results = []
    best_fun = float("inf")

    starts = []
    starts.append(np.array([1.0, -0.2, 0.1, 0.0], dtype=np.float64))
    starts.append(np.array([2.0, -1.0, 0.4, 0.0], dtype=np.float64))
    starts.append(np.array([3.0, -1.6, 0.7, 0.0], dtype=np.float64))
    starts.append(np.array([-1.0, 0.7, -0.4, 0.0], dtype=np.float64))
    starts.append(np.array([0.5, -0.4, 0.5, -0.2], dtype=np.float64))
    starts.append(rng.normal(0.0, 1.0, size=4))
    starts.append(rng.normal(0.0, 1.2, size=4))

    for base in starts:
        if time.monotonic() > float(deadline):
            break
        x0 = np.zeros(dim, dtype=np.float64)
        x0[:4] = base[:4]
        x0[4:] = rng.normal(0.0, 0.45, size=dim - 4)

        a0 = low_dim_a(x0, n)
        a0, sc0 = normalize_and_score(a0)
        if np.isfinite(sc0) and sc0 < best_fun:
            best_fun = float(sc0)
            results.append(a0)

        if _MINIMIZE is not None:
            try:
                res = _MINIMIZE(
                    low_dim_score,
                    x0,
                    args=(n,),
                    method="Nelder-Mead",
                    options={
                        "maxiter": 800,
                        "maxfev": 1600,
                        "xatol": 1e-6,
                        "fatol": 1e-8,
                        "disp": False,
                    },
                )
                if res is not None:
                    fun = float(res.fun)
                    if np.isfinite(fun) and fun < best_fun:
                        best_fun = fun
                        a = low_dim_a(res.x, n)
                        a, sc = normalize_and_score(a)
                        if np.isfinite(sc):
                            results.append(a)
            except Exception:
                pass

    return results


def strict_tune_length(a, factors=(0.5, 0.75, 1.0, 1.25, 1.5, 2.0)):
    """Try nearby lengths using exact scores."""
    a, _ = normalize_and_score(a)
    best = a
    best_sc = strict_score(a)
    n = int(a.size)

    if not np.isfinite(best_sc):
        return best, 2.0

    for f in factors:
        tn = int(round(float(n) * float(f)))
        if tn < 64 or tn > 4096:
            continue
        if tn == n:
            continue
        b = resample_a(a, tn)
        b, fs = normalize_and_score(b)
        try:
            sc = strict_score(b)
        except Exception:
            sc = fs
        if np.isfinite(sc) and sc < best_sc:
            best = b
            best_sc = float(sc)

    return best, float(best_sc)


def mirror_refine(a, deadline, seed):
    """
    Final non-smooth local search.
    Uses active-peak weights and exact scoring with mirror and projected-Euclidean steps.
    """
    a, best_score = normalize_and_score(a)
    best = a
    n = int(best.size)
    L = 2 * n - 1
    rng = np.random.default_rng(seed)
    no_improve = 0
    deadline = float(deadline)

    while time.monotonic() < deadline:
        C = linear_convolve(best)
        M = float(np.max(C))
        if not np.isfinite(M):
            break

        tol = max(1e-14, 1e-6 * M)
        act = np.flatnonzero(C >= M - tol)
        need = max(5, n // 200)
        if act.size < need:
            order = np.argsort(C)[-max(1, need):]
            act = np.unique(np.concatenate((act, order)))
        if act.size == 0:
            act = np.array([int(np.argmax(C))], dtype=int)

        q = np.zeros(L, dtype=np.float64)
        q[act] = 1.0 / float(act.size)

        if no_improve % 25 == 10:
            tauq = max(1e-12, 0.10 * M)
            xq = (C - M) / tauq
            xq = np.clip(xq, -745.0, 0.0)
            ex = np.exp(xq)
            Z = float(np.sum(ex))
            if Z > 1e-300:
                qs = ex / Z
                q = 0.5 * q + 0.5 * qs

        h = gradient_corr(q, best)
        hmax = float(np.max(np.abs(h)))
        improved = False

        if hmax > 1e-25:
            if no_improve > 120:
                targets = (0.02, 0.05, 0.10, 0.20, 0.35, 0.60, 1.0, 2.0)
            else:
                targets = (0.001, 0.003, 0.006, 0.01, 0.02, 0.04, 0.08, 0.15, 0.30)

            for tt in targets:
                eta = float(tt) / hmax
                w = np.exp(-eta * h)
                b = best * w
                s = float(np.sum(b))
                if s > 1e-300 and np.isfinite(s):
                    b /= s
                    b, sc = normalize_and_score(b)
                    if np.isfinite(sc) and sc < best_score:
                        best = b
                        best_score = float(sc)
                        improved = True

        if not improved and hmax > 1e-25:
            g = 2.0 * h
            g = g - float(np.mean(g))
            gmax = float(np.max(np.abs(g)))
            if gmax > 1e-25:
                for tt in targets:
                    step = (float(tt) / gmax) * g
                    b = best - step
                    b = np.clip(b, 1e-300, None)
                    s = float(np.sum(b))
                    if s > 1e-300 and np.isfinite(s):
                        b /= s
                        b, sc = normalize_and_score(b)
                        if np.isfinite(sc) and sc < best_score:
                            best = b
                            best_score = float(sc)
                            improved = True
                            break

        if not improved:
            width = max(3, n // 20)
            kern = np.ones(width, dtype=np.float64) / float(width)
            noise = rng.normal(0.0, 0.18, size=n)
            noise = np.convolve(noise, kern, mode="same")
            for amp in (0.02, 0.05, 0.10, 0.20, 0.40, 0.80):
                b = best * np.exp(amp * noise)
                s = float(np.sum(b))
                if s > 1e-300 and np.isfinite(s):
                    b /= s
                    b, sc = normalize_and_score(b)
                    if np.isfinite(sc) and sc < best_score:
                        best = b
                        best_score = float(sc)
                        improved = True
                        break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 800:
                break

    return best, float(best_score)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Entry point invoked by the grader.
    Returns a list of nonnegative floats minimizing evaluate_sequence.
    """
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1800.0
    budget = max(60.0, budget)

    start = time.monotonic()
    deadline = start + max(60.0, budget - 30.0)
    rng = np.random.default_rng(seed)

    cand = []

    parent = globals().get("height_sequence_1", None)
    if parent is not None:
        try:
            p = np.asarray(parent, dtype=np.float64).ravel()
            p = np.clip(p, 0.0, 1000.0)
            ps = float(np.sum(p))
            if p.size > 0 and np.all(np.isfinite(p)) and ps > 1e-12:
                p = p / ps
                for tn in (128, 256, 512, 1024):
                    add_candidate(cand, resample_a(p, tn), tn)
        except Exception:
            pass

    base_lengths = (128, 256, 512, 1024)
    for n in base_lengths:
        for a in gen_inits(n, rng):
            add_candidate(cand, a, n)

    if cand:
        cand.sort(key=score_key)

    if _MINIMIZE is not None and time.monotonic() < deadline - 300.0:
        ld_deadline = min(deadline - 300.0, time.monotonic() + 140.0)
        for n in (128, 256):
            res = low_dim_optimize(n, rng, ld_deadline)
            for a in res[:3]:
                add_candidate(cand, a, n)
        if cand:
            cand.sort(key=score_key)
            if len(cand) > 0 and time.monotonic() < deadline - 250.0:
                topa = cand[0][0]
                for tn in (512, 1024):
                    add_candidate(cand, resample_a(topa, tn), tn)
                cand.sort(key=score_key)

    if _MINIMIZE is not None and time.monotonic() < deadline - 220.0:
        maxiter_map = {128: 160, 256: 120, 512: 90, 1024: 55, 2048: 35}
        counts = {}
        for a, sc, n in cand[:80]:
            if time.monotonic() > deadline - 180.0:
                break
            if n > 2048:
                continue
            if counts.get(n, 0) >= 3:
                continue
            counts[n] = counts.get(n, 0) + 1
            b, bs = lrefine(a, maxiter_map.get(n, 60), deadline=deadline - 180.0)
            add_candidate(cand, b, int(b.size))
        if cand:
            cand.sort(key=score_key)

        if cand and time.monotonic() < deadline - 200.0:
            topa = cand[0][0]
            tn = min(2048, max(int(topa.size) + 1, int(topa.size) * 2))
            if tn > int(topa.size):
                b = resample_a(topa, tn)
                b, bs = lrefine(b, 45, deadline=deadline - 140.0)
                add_candidate(cand, b, int(b.size))
                if cand:
                    cand.sort(key=score_key)

    if not cand:
        best = np.ones(256, dtype=np.float64) / 256.0
    else:
        best = cand[0][0].copy()

    best, best_score = strict_tune_length(best)
    add_candidate(cand, best, int(best.size))

    if time.monotonic() < deadline - 10.0:
        best, best_score = mirror_refine(best, deadline - 5.0, seed + 101)
        best, best_score2 = strict_tune_length(best)
        if np.isfinite(best_score2) and best_score2 < best_score:
            best_score = best_score2
        add_candidate(cand, best, int(best.size))

    if cand:
        cand.sort(key=score_key)
        best = cand[0][0].copy()

    seq = [float(x) for x in best]
    try:
        ev = evaluate_sequence(seq)
    except Exception:
        ev = float("inf")

    if not np.isfinite(ev):
        seq = [1.0] * 256

    return seq
