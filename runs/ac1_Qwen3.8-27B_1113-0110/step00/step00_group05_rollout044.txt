import numpy as np
import time

"""
Algorithm: Multi-phase optimization of 2*n*max(f*f)/(sum(f))^2
- Phase 1: Gradient descent with exponential peak weighting (FFT-based)
- Phase 2: LP-based refinement using scipy linprog
- Phase 3: Local search with random perturbations
"""

def _weighted_correlation(w, f):
    """Compute g[m] = sum_j w[m+j] * f[j] for m = 0,...,n-1 using FFT."""
    n = len(f)
    N = 1
    while N < 3 * n:
        N *= 2
    A = np.zeros(N)
    A[:len(w)] = w
    B = np.zeros(N)
    B[:n] = f[::-1]  # flipped
    result = np.fft.irfft(np.fft.rfft(A) * np.fft.rfft(B))
    return result[n-1 : n-1+n]


def _compute_score_std(f, n):
    """Compute the evaluation score without calling evaluate_sequence."""
    conv = np.convolve(f, f)
    s = np.sum(f)
    return 2.0 * n * conv.max() / (s * s)

    col_idx = []
