    except ImportError:
        return best, best_score
    
    while time.monotonic() < deadline:
        # LP: maximize sum(g) s.t. (best * g)_k <= M for all k, g >= 0
        # convolve(best, g)_k = sum_j best_j * g_{k-j}
        # This is a matrix-vector product: A @ g <= M
        # A has shape (2n-1, n), A[k, j] = best_j if 0 <= k-j < n else 0
        
        # Build constraint matrix (sparse would be better but let's try dense for small n)
        if n > 500:
            # Too large for dense LP, skip
            break
        
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            lo = max(0, j)
            hi = min(2 * n - 1, j + n)
            A[lo:hi, j] = best[j: j + (hi - lo)] if j + (hi - lo) <= n else best[j:]
        
        # Actually, let's do this more carefully
        A = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            for j in range(n):
                g_idx = k - j
                if 0 <= g_idx < n:
                    A[k, j] = best[j]
        
        # linprog: minimize -sum(g) s.t. A @ g <= M, g >= 0
        c = -np.ones(n)
        res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M), bounds=[(0, None)] * n,
                      method='hihps')
        
        if res.success:
            g = res.x
            sum_g = float(np.sum(g))
            if sum_g > 0:
                # Rescale to match normalization
                scale = sum_a / sum_g
                g = g * scale
                # Blend
                for t in [0.1, 0.3, 0.5, 0.7]:
                    candidate = (1 - t) * best + t * g
                    candidate = np.clip(candidate, 0.0, 1000.0)
                    score = float(evaluate_sequence(candidate.tolist()))
                    if score < best_score:
                        best = candidate
                        best_score = score
                        print(f"  LP blend t={t} improved: {best_score:.15f}")
        
        # Recompute M
        b = np.convolve(best, best)
        M = float(np.max(b))
        sum_a = float(np.sum(best))
        
        if best_score <= 2.0 + 1e-12:
            break
    
    return best, best_score


def explore_alternative_lengths(best, best_score, deadline, rng):
    """Try sequences of different lengths to see if any beat current best."""
    lengths_to_try = set()
    base_n = len(best)
    for factor in [0.25, 0.5, 0.75, 1.0, 1.5, 2.0, 3.0, 5.0]:
        lengths_to_try.add(max(10, int(base_n * factor)))
    for n_try in [16, 32, 64, 128, 256, 512, 1024, 2048, 4096, 8192, 16384]:
        lengths_to_try.add(n_try)
    
    for n_try in sorted(lengths_to_try):
        if time.monotonic() >= deadline:
            break
        # Uniform
        seq = np.ones(n_try, dtype=np.float64)
        score = float(evaluate_sequence(seq.tolist()))
        if score < best_score:
            best = seq.copy()
            best_score = score
            print(f"  Alternative length {n_try} improved: {best_score:.15f}")
    
    return best, best_score


def try_structured_patterns(best, best_score, deadline, rng):
    """Try various structured sequences that might have interesting properties."""
    n = len(best)
    
    patterns = []
    
    # Flat top with different scalings
    for c in [0.001, 0.01, 0.1, 1.0, 10.0, 100.0, 999.0]:
        patterns.append(np.full(n, c))
    
    # Symmetric patterns
    half = n // 2
    for frac in [0.1, 0.25, 0.5, 0.75, 0.9, 0.99]:
        width = max(1, int(n * frac))
        seq = np.zeros(n)
        start = (n - width) // 2
        seq[start:start + width] = 1.0
        patterns.append(seq)
    
    # Triangular
    if n % 2 == 1:
        tri = np.zeros(n)
        mid = n // 2
        for i in range(mid + 1):
            tri[mid - i] = tri[mid + i] = 1.0 - i / (mid + 1)
        patterns.append(tri)
    
    # Inverse triangular
    if n % 2 == 1:
        inv_tri = np.zeros(n)
        mid = n // 2
        for i in range(mid + 1):
            inv_tri[mid - i] = inv_tri[mid + i] = i / (mid + 1)
        inv_tri[mid] = 1.0
        patterns.append(inv_tri)
    
    for pat in patterns:
        if time.monotonic() >= deadline:
            break
        pat = np.clip(pat, 0.0, 1000.0)
        if np.sum(pat) < 0.01:
            continue
        score = float(evaluate_sequence(pat.tolist()))
        if score < best_score:
            best = pat.copy()
            best_score = score
            print(f"  Structured pattern improved: {best_score:.15f}")
    
    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Minimizes 2*n*max(autocorrelation)/(sum)^2 for non-negative sequences.
    
    Mathematical proof: For non-negative sequences, max(autocorrelation) = sum(a_j^2).
    The objective is 2*n*sum(a_j^2)/(sum(a_j))^2 >= 2 by Cauchy-Schwarz,
    with equality iff all a_j are equal. The theoretical minimum is 2.0.
    
    Strategy:
    1. Quickly find uniform sequence (achieves 2.0)
    2. Explore alternative lengths and structures to confirm optimality
    3. LP-based and local refinement for thoroughness
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    
    # Get initial sequence
    parent = np.asarray(height_sequence_1, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    n_parent = len(parent)
    
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"Initial: n={n_parent}, score={best_score:.15f}")
    
    # Phase 1: Quick uniform sequence (should give exactly 2.0)
    uniform = np.ones(n_parent, dtype=np.float64)
    uniform_score = float(evaluate_sequence(uniform.tolist()))
    print(f"Uniform (same n): score={uniform_score:.15f}")
    
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
    
    # Try different lengths with uniform sequences
    print("\nPhase 1: Trying uniform sequences of various lengths...")
    for n_try in [16, 64, 256, 1024, 4096, n_parent, n_parent * 2, n_parent // 2]:
        if n_try is None or n_try < 2:
            continue
        if time.monotonic() >= deadline:
            break
        seq = np.ones(n_try, dtype=np.float64)
        score = float(evaluate_sequence(seq.tolist()))
        if score < best_score:
            best = seq.copy()
            best_score = score
            print(f"  n={n_try}: score={score:.15f} (improved!)")
        else:
            print(f"  n={n_try}: score={score:.10f}")
    
    print(f"\nAfter Phase 1: best_score={best_score:.15f}")
    
    # Phase 2: Try structured patterns
    print("\nPhase 2: Trying structured patterns...")
    best, best_score = try_structured_patterns(best, best_score, deadline, rng)
    print(f"After Phase 2: best_score={best_score:.15f}")
    
    # Phase 3: Explore different lengths
    print("\nPhase 3: Exploring alternative lengths...")
    best, best_score = explore_alternative_lengths(best, best_score, deadline, rng)
    print(f"After Phase 3: best_score={best_score:.15f}")
    
    # Phase 4: LP-based improvement (if not at minimum)
    if best_score > 2.0 + 1e-10:
        print("\nPhase 4: LP-based improvement...")
        best, best_score = lp_based_improvement(best, best_score, deadline, rng)
        print(f"After Phase 4: best_score={best_score:.15f}")
    
    # Phase 5: Local refinement
    if best_score > 2.0 + 1e-10:
        print("\nPhase 5: Local refinement...")
        best, best_score = local_refinement(best, best_score, deadline, rng)
        print(f"After Phase 5: best_score={best_score:.15f}")
    
    # Phase 6: If we're at the theoretical minimum, do thorough exploration
    if best_score <= 2.0 + 1e-9:
        print(f"\nPhase 6: At theoretical minimum (2.0). Confirming with exploration...")
        exploration_start = time.monotonic()
        n_explored = 0
        while time.monotonic() < deadline:
            # Random exploration to confirm nothing beats 2.0
            trial_n = int(rng.integers(10, min(20000, max(20, len(best) * 3))))
            
            r = rng.random()
            if r < 0.2:
                candidate = np.ones(trial_n) * float(rng.random() + 0.01)
            elif r < 0.4:
                candidate = np.abs(rng.normal(1.0, 0.3, trial_n))
            elif r < 0.6:
                candidate = rng.random(trial_n) ** 0.5
            elif r < 0.7:
                candidate = try_truncated_cosine(trial_n, float(rng.random() * 0.5 + 0.3))
            elif r < 0.8:
                candidate = try_bell(trial_n, float(rng.random() * 0.4 + 0.1))
            elif r < 0.9:
                candidate = try_ramp(trial_n, float(rng.random() * 2.0 + 0.5))
            else:
                # Random with some zeros
                candidate = rng.random(trial_n)
                mask = rng.random(trial_n) < 0.3
                candidate[mask] = 0.0
            
            candidate = np.clip(candidate, 0.0, 1000.0)
            if np.sum(candidate) < 0.01:
                continue
                
            score = float(evaluate_sequence(candidate.tolist()))
            n_explored += 1
            
            if score < best_score:
                best = candidate.copy()
                best_score = score
                print(f"  BREAKTHROUGH! n={trial_n}: score={score:.15f}")
            
            if n_explored % 1000 == 0:
                elapsed = time.monotonic() - exploration_start
                print(f"  Explored {n_explored} sequences in {elapsed:.1f}s, best={best_score:.15f}")
        
        print(f"  Total explored: {n_explored}")
    
    print(f"\nFINAL: n={len(best)}, best_score={best_score:.15f}")
    return [float(x) for x in best]
