Wait, is `ifft(fft(a, N) * fft(a, N))` the same as `convolve(a, a)`? Let me verify.

`fft(a, N)` computes `A[k] = sum_{n=0}^{N-1} a[n]*exp(-2*pi*i*k*n/N)` (padding a to length N with zeros).

`= (1/N) * (sum_n a[n]) * (sum_m a[m]) * ... `

For linear convolution, we need N >= len(x) + len(y) - 1 = 2n-1. If N >= 2n-1, the circular convolution equals the linear convolution.

So yes, `np.fft.ifft(np.fft.fft(a, N) * np.fft.fft(a, N)).real[:2*n-1]` gives `np.convolve(a, a)` when N >= 2n-1. ✓

def fast_autocorr(a):
    """Compute full linear convolution a*a using FFT. O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.fft(a, fft_size)
    C = np.fft.ifft(A * A).real
    return C[:size]

def compute_objective(a):
    """Compute 2*n*max(autocorr)/sum^2."""
    n = len(a)
    c = fast_autocorr(a)
    max_c = np.max(c)
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_c / (sum_a * sum_a)

def gradient_of_c_at_k(a, k, n):
    """Compute dc[k]/da[m] for all m. Result is 2*a[k-m] if 0<=k-m<n else 0."""
    grad = np.zeros(n)
    # For each m in [0, n-1], grad[m] = 2*a[k-m] if 0<=k-m<n
    # k-m in [0, n-1] means k-n+1 <= m <= k
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        # grad[lo:hi+1] = 2 * a[k-hi : k-lo+1][::-1]
        indices = k - np.arange(lo, hi + 1)  # k-lo, k-lo-1, ..., k-hi
        grad[lo:hi+1] = 2.0 * a[indices]
    return grad

def project_onto_simplex_nonneg(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum} (approximate)."""
    v = np.clip(v, 0, None)
    s = v.sum()
    if s < 1e-15:
        return np.full(len(v), target_sum / len(v))
    return v * (target_sum / s)

def raised_cosine(n):
    t = np.arange(n)
    return 0.5 * (1 + np.cos(2 * np.pi * t / n))

def hann_window(n):
    t = np.arange(n)
    return 0.5 * (1 + np.cos(2 * np.pi * t / (n - 1)))

def triangular_window(n):
    t = np.arange(n)
    peak = (n - 1) / 2.0
    return 1 - np.abs(t - peak) / peak

def u_shaped(n):
    """U-shaped: high at edges, low in middle."""
    t = np.arange(n) / max(1, n - 1)
    return 1.0 + np.cos(np.pi * t)  # Goes from 2 to 0

def v_shaped(n):
    """V-shaped: low at edge, high in middle."""
    t = np.arange(n)
    peak = (n - 1) / 2.0
    return np.abs(t - peak) / peak

"""
Minimizes 2*n*max(autocorr(a,a))/sum(a)^2 using gradient-based local search
with simulated annealing, multiple initializations, and LP-based refinement.
"""
import numpy as np
import time
import math


def _fft_selfconv(a):
    """Compute np.convolve(a, a) using FFT. O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    fft_len = 1 << (size - 1).bit_length()
    F = np.fft.rfft(a, fft_len)
    c = np.fft.irfft(F * F, fft_len)[:size]
    return c


def _obj_from_conv(c, sum_a):
    """Compute objective from convolution and sum."""
    n = (len(c) + 1) // 2
    max_c = np.max(c)
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_c / (sum_a * sum_a)


def _grad_c_at_k(a, k, n):
    """Gradient of c[k] w.r.t. a. Returns array of length n."""
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        idx = k - np.arange(lo, hi + 1)
        grad[lo:hi + 1] = 2.0 * a[idx]
    return grad


def _local_search(a, rng, time_limit, verbose=False):
    """Gradient-based local search with simulated annealing."""
    n = len(a)
    a = a.copy()
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s  # normalize to sum=1
    
    start = time.monotonic()
    best_a = a.copy()
    
    # Compute initial objective
    c = _fft_selfconv(a)
    max_c = np.max(c)
    best_obj = 2.0 * n * max_c  # sum_a=1, so objective = 2n*max_c
    
    temperature = 0.001
    alpha = 0.05
    momentum = np.zeros(n)
    beta = 0.9
    
    no_improve = 0
    max_no_improve = 5000
    
    while time.monotonic() - start < time_limit:
        c = _fft_selfconv(a)
        c = np.maximum(c, 0)
        max_c = np.max(c)
        obj = 2.0 * n * max_c
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            # Reset: add random perturbation
            a = best_a + rng.normal(0, 0.02, n)
            a = np.clip(a, 0, None)
            a = a / a.sum()
            temperature = max(temperature * 0.5, 1e-6)
            no_improve = 0
            continue
        
        # Find top lags
        k_star = int(np.argmax(c))
        
        # Compute gradient
        grad = _grad_c_at_k(a, k_star, n)
        
        # Also consider average of top few lags for smoother gradient
        top_k = np.argsort(c)[-min(5, len(c)):]
        if len(top_k) > 1:
            grad_smooth = np.zeros(n)
            for k_t in top_k:
                grad_smooth += _grad_c_at_k(a, int(k_t), n)
            grad_smooth /= len(top_k)
            grad = 0.7 * grad + 0.3 * grad_smooth
        
        # Momentum update
        momentum = beta * momentum + (1 - beta) * grad
        alpha = min(alpha * 1.001, 0.5)  # slowly increase step size
        
        # Update
        a_new = a - alpha * momentum + alpha * momentum.mean()  # maintain sum≈1
        a_new = np.clip(a_new, 0, None)
        
        if a_new.sum() < 1e-10:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / a_new.sum()
        
        # Simulated annealing acceptance
        c_new = _fft_selfconv(a_new)
        c_new = np.maximum(c_new, 0)
        obj_new = 2.0 * n * np.max(c_new)

        if delta < 0 or rng.random() < math.exp(-delta / max(temperature, 1e-10)):
            a = a_new
        else:
            alpha = max(alpha * 0.9, 0.001)
        
        temperature *= 0.9999
        temperature = max(temperature, 1e-8)
    
    return best_a, best_obj / (1.0 * 1.0)  # sum_a was 1, so this is the actual objective


def _lp_refine(a, rng, time_limit):
    """LP-based refinement inspired by state of the art."""
    try:
        from scipy.optimize import linprog
    except ImportError:
        return a, _obj_from_conv(_fft_selfconv(a), a.sum())
    
    n = len(a)
    a = a.copy()
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a = a / s
    
    start = time.monotonic()
    best_a = a.copy()
    best_obj = 2.0 * n * np.max(_fft_selfconv(a))
    
    t_blend = 0.1
    
    while time.monotonic() - start < time_limit:
        c = _fft_selfconv(a)
        M = np.max(c)
        
        # LP: max sum(b) s.t. b>=0, (f*b)[k] <= M for all k
        # (f*b)[k] = sum_i f[i]*b[k-i]
        # Constraint matrix A: A[k,j] = f[k-j] if 0<=k-j<n else 0
        # A is (2n-1) x n
        
        size = 2 * n - 1
        # Build A sparsely
        # Row k, col j: A[k,j] = a[k-j] if 0<=k-j<n
        # For row k: j ranges from max(0, k-n+1) to min(k, n-1)
        # A[k, j] = a[k-j]
        
        from scipy.sparse import csr_matrix
        rows = []
        cols = []
        vals = []
        for k in range(size):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            for j in range(lo, hi + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
        
        A = csr_matrix((vals, (rows, cols)), shape=(size, n))
        
        # c_vec = -1 (we want to maximize sum(b))
        c_vec = np.ones(n)
        b_vec = np.full(size, M)
        
        # Bounds: b[i] >= 0
        bounds = [(0, None)] * n
        
        res = linprog(c_vec, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs')
        
        if res.success:
            b = np.clip(res.x, 0, 1000)
            b_sum = b.sum()
            if b_sum > 1e-10:
                b = b / b_sum  # normalize
                a_new = (1 - t_blend) * a + t_blend * b
                a_new = np.clip(a_new, 0, None)
                a_new = a_new / a_new.sum()
                
                obj_new = 2.0 * n * np.max(_fft_selfconv(a_new))
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    t_blend = min(t_blend * 1.1, 0.5)
                a = a_new
        else:
            break
    
    return best_a, best_obj
