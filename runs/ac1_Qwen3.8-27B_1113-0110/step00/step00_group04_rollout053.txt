    remaining = deadline - time.monotonic()
    n_inits = len(scored)

    if n_inits <= 1:
        fracs = [1.0] * max(1, n_inits)
    elif n_inits == 2:
        fracs = [0.6, 0.4]
    elif n_inits == 3:
        fracs = [0.45, 0.3, 0.25]
    elif n_inits == 4:
        fracs = [0.35, 0.25, 0.2, 0.2]
    elif n_inits == 5:
        fracs = [0.3, 0.25, 0.2, 0.15, 0.1]
    else:
        base = 0.35 / n_inits
        fracs = [base * (n_inits - i) for i in range(n_inits)]
        fracs = [f / sum(fracs) for f in fracs]

    global_best_a = scored[0][2].copy()
    global_best_score = scored[0][0]

    for i in range(n_inits):
        if time.monotonic() >= deadline:
            break
        alloc = remaining * fracs[i]
        local_deadline = min(time.monotonic() + alloc, deadline)
        score_i, name_i, a_init = scored[i]
        if local_deadline - time.monotonic() < 2.0:
            continue
        print(f"Descent '{name_i}' (init={score_i:.4f}, t={alloc:.1f}s)")
        t0 = time.monotonic()
        global_best_a, global_best_score = _mirror_descent(
            a_init.copy(), n, fft_size, fft_size2,
            local_deadline, global_best_a, global_best_score, rng
        )
        elapsed = time.monotonic() - t0
        print(f"  '{name_i}' done in {elapsed:.1f}s, best={global_best_score:.6f}")

    # Final polish
    remaining = deadline - time.monotonic()
    if remaining > 10 and global_best_a is not None:
        print(f"Final polish: {remaining:.1f}s")
        global_best_a, global_best_score = _mirror_descent(
            global_best_a.copy(), n, fft_size, fft_size2,
            deadline, global_best_a, global_best_score, rng
        )

    print(f"FINAL SCORE: {global_best_score:.6f}")
    return [float(x) for x in global_best_a]
