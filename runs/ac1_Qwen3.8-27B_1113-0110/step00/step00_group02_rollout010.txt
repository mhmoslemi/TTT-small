import numpy as np
import time

"""
Minimizes 2n*max(conv(f,f))/(sum(f))^2 using multiplicative pressure updates.
Iteratively moves mass away from positions contributing to convolution peaks,
with adaptive learning rates, symmetry enforcement, random restarts, and
coordinate descent refinement. Tries multiple sequence lengths.
"""


def _one_flatten_step(f, eta, n, hot_fraction=0.5, weight_power=8.0):
    """One multiplicative pressure update step. Returns updated f and peaking factor."""
    z = np.convolve(f, f)
    M = np.max(z)
    mean_z = 1.0 / (2 * n - 1)
    peaking = M / mean_z

    threshold = hot_fraction * M
    hot = np.where(z > threshold)[0]
    if len(hot) == 0:
        hot = np.array([np.argmax(z)])

    w = np.zeros(2 * n - 1)
    peak_vals = z[hot]
    weights = (peak_vals / M) ** weight_power
    w[hot] = weights

    pressure = np.correlate(w, f, mode='valid')

    p_max = np.max(pressure)
    if p_max > 1e-15:
        f = f * np.exp(-eta * pressure / p_max)

    f = np.clip(f, 1e-12, 1000.0)
    s = np.sum(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    return f, peaking

        if time.monotonic() > deadline:
            break
        xi = f[i]
        if xi < 1e-14:
            continue

        # Try candidate values: 0, half, quarter, and a few scaling factors
        candidates = [0.0, xi * 0.5, xi * 0.25, xi * 0.75]
        best_x = xi
        best_m = current_max

        for xv in candidates:
            # Incrementally update z
            # Affected k: i <= k <= i+n-1 (where f[k-i] is valid)
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            if k_lo > k_hi:
                continue

            # Save old values
            old_z = z[k_lo:k_hi + 1].copy()

            # Apply change: z[k] += 2*(xv-xi)*f[k-i] for k != 2i, z[2i] += xv^2 - xi^2
            for k in range(k_lo, k_hi + 1):
                j = k - i
                if j < 0 or j >= n:
                    continue
                if k == 2 * i:
                    z[k] = old_z[k - k_lo] - xi * xi + xv * xv
                else:
                    z[k] = old_z[k - k_lo] + 2.0 * (xv - xi) * f[j]

            new_max = np.max(z)
            if new_max < best_m:
                best_m = new_max
                best_x = xv

            # Restore z
            z[k_lo:k_hi + 1] = old_z

        f[i] = best_x
        if best_x != xi:
            current_max = best_m
            improved = True

        # Recompute z periodically to avoid drift
        if (i + 1) % 200 == 0:
            z = np.convolve(f, f)
            current_max = np.max(z)

    return f, improved


def _try_sequence(f, n):
    """Evaluate sequence and return score."""
    try:
        score = float(evaluate_sequence(f.tolist()))
        return score
    except Exception:
        return np.inf


def propose_candidate(budget_s=1800):

    n_options = [1024, 2048, 4096, 6962]

        parent = None

    mul_deadline = time.monotonic() + 1400.0

    for n in n_options:
        if time.monotonic() > mul_deadline:
            break
        print(f"\n=== Trying n={n} ===")

        # Build initializations for this n
        inits = []

        # 1. Uniform (gives score 2.0)
        inits.append(np.ones(n) / n)

        # 2. Raised cosine
        x = np.arange(n, dtype=np.float64)
        fc = 0.5 * (1.0 + np.cos(2.0 * np.pi * (x + 0.5) / n))
        inits.append(fc / np.sum(fc))

        # 3. Parent (resized if needed)
        if parent is not None and len(parent) == n:
            fp = np.clip(parent, 0, 1000)
            s = np.sum(fp)
            if s > 1e-6:
                fp = fp / s
                fp = 0.5 * (fp + fp[::-1])
                inits.append(fp / np.sum(fp))
        elif parent is not None:
            # Resize parent to n via interpolation
            from numpy import interp
            xp = np.linspace(0, 1, len(parent))
            xn = np.linspace(0, 1, n)
            fp = interp(xn, xp, parent)
            fp = np.clip(fp, 0, 1000)
            s = np.sum(fp)
            if s > 1e-6:
                fp = fp / s
                fp = 0.5 * (fp + fp[::-1])
                inits.append(fp / np.sum(fp))

        # 4. Smooth random
        fr = 1.0 + 0.5 * np.sin(np.arcsin(np.clip(rng.standard_normal(n), -1, 1)))
        fr = np.abs(fr) + 0.1
        fr = 0.5 * (fr + fr[::-1])
        inits.append(fr / np.sum(fr))

        # 5. Two-peak (to break symmetry and encourage flat conv)
        t = np.arange(n) / (n - 1)
        f5 = np.exp(-50 * (t - 0.3) ** 2) + np.exp(-50 * (t - 0.7) ** 2)
        inits.append(f5 / np.sum(f5))

        # Evaluate all inits and pick best
        for fi in inits:
            score = _try_sequence(fi, n)
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_f = fi.copy()
                best_n = n
                print(f"  New best init: score={score:.8f}")

        # Pick the best init for this n and run multiplicative updates
        f = best_f.copy() if (best_n == n) else inits[0].copy()
        s_init = _try_sequence(f, n)
        local_best = f.copy()
        local_best_score = s_init

        eta = 2.0
        eta_min = 0.005

                        f = 0.5 * (f + f[::-1])

            best_n = n

    print(f"\nAfter multiplicative phase: best_score={best_score:.8f}, best_n={best_n}")

        f = best_f.copy()

        cd_rng = np.random.default_rng(777)

            if time.monotonic() > cd_deadline:

            f, improved = _coordinate_descent(f, n, cd_deadline, cd_rng)
            if cd_round % 10 == 0:

                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_f = f.copy()
                    if cd_round % 30 == 0:
                        print(f"  CD round {cd_round}: score={best_score:.8f}, improved={improved}")

    # Phase 3: Quick local refinement with small random perturbations
    if time.monotonic() < deadline - 5:
        print("\n=== Final Local Refinement ===")
        f = best_f.copy()
        n = best_n
        final_deadline = deadline - 2.0
        local_rng = np.random.default_rng(999)

        for _ in range(5000):
            if time.monotonic() > final_deadline:
                break
            # Small multiplicative perturbation
            width = max(1, n // 50)
            start = int(local_rng.integers(0, n - width + 1))
            f[start:start + width] *= np.exp(local_rng.normal(0, 0.05, width))
            f = np.clip(f, 0, 1000)
            s = np.sum(f)
            if s < 1e-6:
                f = best_f.copy()
                continue
            f = f / s

            score = _try_sequence(f, n)
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_f = f.copy()

    # Final validation
    final_score = _try_sequence(best_f, best_n)
    if not np.isfinite(final_score):
        # Fallback to uniform
        best_f = np.ones(best_n if best_n > 0 else 256)
        final_score = _try_sequence(best_f, best_n if best_n > 0 else 256)

    print(f"\n=== FINAL ANSWER: score={final_score:.10f}, n={len(best_f)} ===")

    return [float(x) for x in best_f]
