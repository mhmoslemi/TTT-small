```python
import numpy as np
import time
from scipy import sparse
from scipy.optimize import linprog

def solve_lp_direction(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    # Build constraint matrix C where C[k,j] = a[j] if 0 <= k-j < n
    # C is (2n-1) x n
    row_indices = []
    col_indices = []
    data = []
    for k in range(2*n-1):
        j_start = max(0, k - (n-1))
        j_end = min(n-1, k)
        for j in range(j_start, j_end+1):
            row_indices.append(k)
            col_indices.append(j)
            data.append(a[j])
    
    C = sparse.csc_matrix((data, (row_indices, col_indices)), shape=(2*n-1, n))
    
    # LP: max c^T g s.t. C g <= b, g >= 0
    c = -np.ones(n)  # negate for minimization
    b = M * np.ones(2*n-1)
    
    res = linprog(c, A_ub=C, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return -res.fun  # this is max sum(g)
    return None
```

```python
    """Build the convolution matrix C where C[k,j] = a[j] if 0 <= k-j < n."""
    n = len(a)
    rows = np.repeat(np.arange(n), n) + np.tile(np.arange(n), n)
    cols = np.repeat(np.arange(n), n)
    data = np.repeat(a, n)
    return sparse.csc_matrix((data, (rows, cols)), shape=(2*n-1, n))

def convolve_self(a):
    """Compute self-convolution of a."""
    return np.convolve(a, a)

        s = g.sum()
```

```python
    n = len(a)
    g = np.ones(n)
    for _ in range(n_iter):
        # Compute C*g = convolve(a, g)
        cg = np.convolve(a, g)
        # Find violations
        violations = np.maximum(0, cg - M)
        if violations.sum() == 0:
            break

        # C[k,j] = a[j] if 0 <= k-j < n

        # = a[j] * (cumulative sum of violations at j to j+n-1)

        # For each j, we want sum of violations[j] to violations[j+n-1]
        # This is a convolution of violations with a window of ones of length n

        # But we need to be careful with boundaries.

        # sum_k violations[k] * C[k,j] = a[j] * sum_{k=max(j,0)}^{min(j+n-1, 2n-2)} violations[k]

        # So sum_{k=j}^{j+n-1} violations[k]
```
