"""
Minimize  2 * n * max(convolve(a,a)) / (sum a)^2  for a nonnegative sequence a.

Method: an outer "water-filling" / Fenchel-style fixed-point iteration.
For the current a (normalized to sum 1) we form M = max(a*a).  We then solve the
LP   max sum(b')  s.t.  b' >= 0  and  conv(a,b')[k] <= M  for every lag k.
The LP solution b'_*, renormalized to have the same mass as a, is the optimal
"response" direction: it pushes mass exactly where the convolution is currently
allowed to be smallest.  We then take a convex step a <- (1-t)a + t b'_* with a
line search over t, which non-increases the peak and drives the convolution toward
a flat profile.  Several non-symmetric initialisations of the (provided) length are
run in parallel on a shared time budget and the global best is returned.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


# --------------------------------------------------------------------------- #
# low level helpers
# --------------------------------------------------------------------------- #
def _my_score(a):
    """Exact replica of evaluate_sequence (a is already in a benign range)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, None)
    n = a.size
    if n == 0:
        return np.inf
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _build_conv_matrix(a):
    """Sparse (2n-1) x n matrix A with A[k,l] = a[k-l] for valid l.
       Then A @ b' == convolve(a, b')."""
    n = a.size
    data = []
    indices = []
    indptr = [0]
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        cols = np.arange(lo, hi + 1, dtype=np.int32)
        vals = a[k - cols]
        data.append(vals)
        indices.append(cols)
        indptr.append(indptr[-1] + cols.size)
    A = csr_matrix((np.concatenate(data), np.concatenate(indices), indptr),
                   shape=(2 * n - 1, n))
    return A


def _solve_response(a, M):
    """Solve the LP for the response direction b'.  Returns b' (sum 1) or None."""
    n = a.size
    A = _build_conv_matrix(a)
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0.0, None)] * n, method="highs")
    except Exception:
        return None
    if res is None or res.status != 0:
        return None
    x = np.asarray(res.x, dtype=np.float64)
    x = np.clip(x, 0.0, None)
    s = x.sum()
    if s <= 1e-12:
        return None
    return x / s


def _line_search(a, g, cur_score):
    """Best convex step from a toward g (both sum 1)."""
    best_t = 0.0
    best_a = a
    best_s = cur_score
    for t in (0.05, 0.1, 0.2, 0.3, 0.5, 0.75, 1.0):
        cand = (1.0 - t) * a + t * g
        sc = _my_score(cand)
        if sc < best_s - 1e-12:
            best_s = sc
            best_t = t
            best_a = cand
    return best_a, best_s, best_t


# --------------------------------------------------------------------------- #
# optimiser for a single initialisation
# --------------------------------------------------------------------------- #
def _optimize_start(a0, deadline, state):
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, None)
    s = a.sum()
    if s <= 1e-9:
        return
    a = a / s                       # normalise to sum 1  (score is scale free)
    best = a
    best_s = _my_score(a)
    if best_s < state["score"]:
        state["score"] = best_s
        state["a"] = best.copy()
        print("  [init %.6f]" % best_s, flush=True)

    stall = 0
    it = 0
    while time.monotonic() < deadline:
        c = np.convolve(a, a)
        M = float(c.max())
        g = _solve_response(a, M)
        if g is None:
            stall += 1
            if stall > 3:
                break
            a = a + 0.01 * np.random.default_rng(it).normal(0, 0.02, a.size)
            a = np.clip(a, 0.0, None)
            a = a / a.sum()
            continue
        na, ns, t = _line_search(a, g, best_s)
        it += 1
        if ns < best_s - 1e-10:
            a, best_s = na, ns
            stall = 0
            if ns < state["score"]:
                state["score"] = ns
                state["a"] = a.copy()
                print("  [iter %d] score=%.6f  t=%.2f" % (it, ns, t), flush=True)
        else:
            stall += 1
            if stall > 25:        # local stagnation for this start
                break
    return


# --------------------------------------------------------------------------- #
# entry point
# --------------------------------------------------------------------------- #
def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Return the best nonnegative sequence found within the time budget."""
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 6.0)

    # base length: prefer the provided construction when available
    n = 2535
    try:
        base = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, None)
        if base.ndim == 1 and base.size > 1 and np.isfinite(base).all():
            n = int(base.size)
    except Exception:
        base = None

    # ---- build a diverse set of non-symmetric initialisations (length n) ----
    inits = []
    if base is not None:
        b = base.copy()
        b = b / b.sum()
        inits.append(b)

    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp / ramp.sum())

    sq = (np.arange(1, n + 1, dtype=np.float64) ** 2)
    inits.append(sq / sq.sum())

    for alpha in (0.02, 0.06, 0.15):
        e = np.exp(alpha * np.arange(n, dtype=np.float64))
        inits.append(e / e.sum())

    # a "two lobe" non-symmetric bump
    grid = np.arange(n, dtype=np.float64)
    two = np.exp(-((grid - 0.25 * n) / (0.12 * n)) ** 2) \
        + 0.6 * np.exp(-((grid - 0.75 * n) / (0.10 * n)) ** 2)
    inits.append(two / two.sum())

    for _ in range(3):
        r = rng.dirichlet(np.ones(n) * 1.5)
        inits.append(r / r.sum())

    state = {"score": np.inf, "a": None}

    # smoke check: make sure we always have *some* valid sequence to return
    fallback = inits[0]
    fb_score = _my_score(fallback)
    state["score"] = fb_score
    state["a"] = fallback.copy()
    print("[start] n=%d  base_score=%.6f" % (n, fb_score), flush=True)

    # ---- run each initialisation on a time slice, always keep global best ----
    m = len(inits)
    for i, a0 in enumerate(inits):
        # each start gets the whole remaining window but we also rotate slices
        slice_end = min(deadline, t_start + float(budget_s - 6.0) * (i + 1) / m + 60.0)
        if time.monotonic() >= deadline:
            break
        print("== start %d / %d ==" % (i, m), flush=True)
        _optimize_start(a0, min(deadline, slice_end), state)
        print("[best so far] %.6f   elapsed=%.1fs"
              % (state["score"], time.monotonic() - t_start), flush=True)

    best = np.clip(np.asarray(state["a"], dtype=np.float64), 0.0, 1000.0)
    # keep it safely away from the sum<0.01 / clamp edges by scaling to a mean of 1
    best = best / best.sum() * n          # mean 1  ->  values O(1), well inside [0,1000]
    best = np.clip(best, 0.0, 1000.0)
    print("[final] returned score=%.6f  length=%d" % (_my_score(best), best.size),
          flush=True)
    return [float(x) for x in best]
