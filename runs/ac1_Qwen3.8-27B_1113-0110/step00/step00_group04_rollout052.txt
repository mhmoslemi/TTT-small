"""
Minimize  2 * n * max(convolve(a, a)) / (sum a)^2  over non-negative a.

Key facts exploited here:
  * The objective is scale invariant, so we fix sum(a)=1 and optimize the shape.
  * Palindromic shapes are forced to have score >= 2 (center lag dominated by
    sum a_j^2).  Asymmetric (ramp / decaying) shapes push the correlator peak
    off the center, which is the only way to beat 2.
  * We differentiate a smooth (log-sum-exp) surrogate for the max, giving a
    valid subgradient that is a weighted average of the hard-max gradients.

Per iteration (all O(n log n) via FFT):
  * c = a*a (full convolution) via FFT.
  * w = softmax(T*c); soft_max = <w,c>;  g[j] = sum_m a[m] w[j+m] (FFT).
  * grad = (4n/S^2) * (g - soft_max/S).
  * projected step, clip >=0, renormalize sum=1.

Multiple asymmetric initial shapes and multiple temperature schedules are run;
the authoritative evaluate_sequence is sampled periodically to track the true
best.  A guaranteed-valid fallback sequence is always returned.
"""

import numpy as np
import time


# ---------------------------------------------------------------------------
# small utilities
# ---------------------------------------------------------------------------
def _next_pow2(x):
    if x <= 1:
        return 1
    return 1 << (int(x) - 1).bit_length()


def _full_conv(a):
    """Full linear convolution of a with itself using FFT."""
    n = a.size
    need = 2 * n - 1
    N = _next_pow2(need)
    F = np.fft.rfft(a, N)
    out = np.fft.irfft(F * F, N)[:need]
    return out


def _correl_a_sg(a, sg):
    """
    Return g[j] = sum_m a[m]*sg[j+m] for j=0..n-1, sg length 2n-1, via FFT.
    (linear cross-correlation slice, no wrap for N >= 3n-2)
    """
    n = a.size
    N = _next_pow2(3 * n - 2)
    F = np.fft.fft(a, N)
    G = np.fft.fft(sg, N)
    R = np.fft.ifft(F * np.conj(G * 0 + G)).real  # (F * conj(G))
    # correct: ifft( F * conj(G) )
    R = np.fft.ifft(F * np.conj(G)).real
    return R[:n]


def _self_test():
    """Validate FFT helpers against direct computation on a small vector."""
    rng = np.random.default_rng(0)
    n = 17
    a = rng.random(n)
    a = np.clip(a, 0.0, None)
    c_fft = _full_conv(a)
    c_ref = np.convolve(a, a)
    if not np.allclose(c_fft, c_ref, rtol=1e-8, atol=1e-10):
        print("WARN: convolve self-test failed")
        return False
    sg = rng.random(2 * n - 1)
    g_fft = _correl_a_sg(a, sg)
    g_ref = np.empty(n)
    for j in range(n):
        g_ref[j] = np.dot(a, sg[j:j + n])
    if not np.allclose(g_fft, g_ref, rtol=1e-8, atol=1e-10):
        print("WARN: correlation self-test failed")
        return False
    return True


def _soft_max_and_w(c, T):
    """log-sum-exp soft max of c and the softmax weights (vector)."""
    cm = c.max()
    Tcm = T * cm
    if Tcm > 500.0:
        T = max(1e-9, 500.0 / max(cm, 1e-30))
    e = np.exp(T * (c - cm))
    s = e.sum()
    w = e / s
    soft = cm + np.log(s) / T
    return soft, w


def _soft_obj_and_grad(a, T):
    """Soft objective and gradient w.r.t. a (sum not assumed == 1)."""
    n = a.size
    S = a.sum()
    if S < 0.01:
        return np.inf, np.zeros(n)
    c = _full_conv(a)
    soft, w = _soft_max_and_w(c, T)
    g = _correl_a_sg(a, w)  # sum_m a[m] w[j+m]
    grad = (4.0 * n / (S * S)) * (g - soft / S)
    val = 2.0 * n * soft / (S * S)
    return val, grad


def _normalize(a):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s <= 0:
        a = np.ones_like(a)
        s = a.sum()
    return a / s

def _init_geometric(n, r):
    j = np.arange(n, dtype=np.float64)
    return np.exp(np.log(r) * j)


def _init_ramp(n, p):
    j = np.arange(n, dtype=np.float64)
    x = 1.0 - j / float(max(1, n - 1))
    return np.clip(x, 0.0, 1.0) ** p + 1e-9


def _init_double_ramp(n, p):
    j = np.arange(n, dtype=np.float64)
    # increasing ramp (mirror of decreasing) - same objective
    x = j / float(max(1, n - 1))
    return np.clip(x, 0.0, 1.0) ** p + 1e-9

# core optimizer: projected gradient descent on the soft surrogate
# ---------------------------------------------------------------------------
def _optimize(a0, deadline, rng, schedule):
    """
    schedule: list of (T, lr, n_iter) blocks applied in order until deadline.
    Returns (best_a, best_softval).
    """
    a = _normalize(a0.copy())
    best_a = a.copy()
    best_val = _soft_obj_and_grad(a, schedule[0][0])[0]
    val, grad = _soft_obj_and_grad(a, schedule[0][0])
    if not np.isfinite(val):
        return best_a, best_val

    for T, lr, niters in schedule:
        for _ in range(niters):
            if time.monotonic() > deadline - 4.0:
                return best_a, best_val
            val, grad = _soft_obj_and_grad(a, T)
            if not np.isfinite(val):
                break
            if val < best_val:
                best_val = val
                best_a = a.copy()

            # backtracking line search on the soft objective
            step = lr
            a_prev = a
            improved = False
            for _b in range(6):
                cand = _normalize(a_prev - step * grad)
                cv, _ = _soft_obj_and_grad(cand, T)
                if cv < val - 1e-12:
                    a = cand
                    val = cv
                    if val < best_val:
                        best_val = val
                        best_a = a.copy()
                    improved = True
                    break
                step *= 0.5
            if not improved:
                # accept the base (no line-search gain) to keep moving
                a = _normalize(a_prev - lr * 0.25 * grad)
    return best_a, best_val
