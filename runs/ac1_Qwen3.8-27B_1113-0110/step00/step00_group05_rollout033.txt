Intuitively, a sequence that's "as flat as possible" in its auto-convolution is optimal. This is like wanting the sequence to be "as random as possible" in some sense - its self-convolution should look like white noise (flat spectrum).

A sequence whose auto-convolution is flat would have a flat power spectrum (in the sense that |FFT(a)|² is flat). But |FFT(a)|² being flat means |FFT(a)| is constant, which means a is a "perfect" sequence. For real non-negative sequences, this is not exactly achievable, but we can get close.

The ideal would be: FFT(a) has constant magnitude. Then the auto-convolution (which is IFFT(|FFT(a)|²)) would be a delta function (all energy at lag 0). But that's the autocorrelation, not the auto-convolution!

I keep confusing auto-correlation and auto-convolution. Let me be clear:
- Auto-correlation: R[k] = sum_i a[i] * a[i+k]
- Auto-convolution: C[k] = sum_i a[i] * a[k-i]

They're different! The auto-convolution is the "product" in the Fourier domain:
FFT(C) = FFT(a) * FFT(a) = (FFT(a))²

While auto-correlation has:
FFT(R) = FFT(a) * conj(FFT(a)) = |FFT(a)|²

So for the auto-convolution to be flat, we need (FFT(a))² to be flat, which means FFT(a) has constant magnitude AND constant phase (or phase that's 0 or π everywhere, i.e., the FFT is real).

For a real, non-negative sequence, FFT(a) is real at frequency 0 (it's the sum) and has conjugate symmetry. For FFT(a) to be approximately constant in magnitude and real, we'd need a to be approximately a rectangular function in the frequency domain, which corresponds to a sinc-like function in the time domain. import numpy as np
import time

"""
Minimizes 2*n*max(auto_conv) / (sum)^2 using projected gradient descent
with soft-max smoothing and alpha annealing. Employs multiple diverse
initializations, Adam optimizer, and periodic exact evaluations.
"""


def exact_score_fft(a, fft_size):
    """Compute exact objective using FFT-based convolution."""
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    fa = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(fa * fa, fft_size)[: 2 * n - 1]
    max_b = c.max()
    return 2.0 * n * max_b / (S * S)


def compute_grad_and_value(a, alpha, fft_size):
    """Compute gradient of soft_max(conv(a,a)) and its value. Returns (grad, soft_max_value)."""
    n = len(a)
    fa = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(fa * fa, fft_size)[: 2 * n - 1]

    M = c.max()
    shifted = alpha * (c - M)
    exp_shifted = np.exp(shifted)
    w_sum = exp_shifted.sum()
    w = exp_shifted / w_sum

    # Gradient: 2 * cross_correlation(a, w) via FFT
    fw = np.fft.rfft(w, fft_size)
    grad = 2.0 * np.fft.irfft(np.conj(fa) * fw, fft_size)[:n]

    sm = M + np.log(w_sum) / alpha
    return grad, sm


def run_optimization(a_init, n, fft_size, deadline, seed, max_alphas=None):
    """Run Adam gradient descent with alpha annealing from a given init."""
    rng = np.random.default_rng(seed)
    a = a_init.copy()
    a = np.clip(a, 0.0, 1000.0)
    S = a.sum()
    if S < 1e-12:
        return None, np.inf
    a = a / S

    local_best = exact_score_fft(a, fft_size)
    best_a = a.copy()

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8

    # Alpha schedule: start smooth, end sharp
    if max_alphas is None:
        alphas = [2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0, 1000.0, 2000.0]
    else:
        alphas = max_alphas

    # Learning rate schedule (decreases as alpha increases)
    base_lr = 0.05

    stall_count = 0
    stall_threshold = 3000

    for alpha_idx, alpha in enumerate(alphas):
        if time.monotonic() > deadline:
            break

        lr = base_lr / max(1.0, alpha * 0.1)
        # Scale lr so that effective step is reasonable
        lr = min(lr, 5.0 / n)

        n_iter = 5000
        # Use more iterations for lower alpha (easier landscape)
        if alpha < 10:
            n_iter = 3000

        for it in range(n_iter):
            if time.monotonic() > deadline:
                break

            grad, sm = compute_grad_and_value(a, alpha, fft_size)
            t += 1

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # Project onto probability simplex
            a_new = np.clip(a_new, 0.0, None)
            S_new = a_new.sum()
            if S_new < 1e-15:
                # Reset if we lost all mass
                a = a_init.copy()
                a = np.clip(a, 0.0, 1000.0)
                a = a / a.sum()
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                break
            a = a_new / S_new

            # Check score periodically
            if (it + 1) % 500 == 0:
                score = exact_score_fft(a, fft_size)
                if score < local_best - 1e-10:
                    local_best = score
                    best_a = a.copy()
                    stall_count = 0
                else:
                    stall_count += 500

                # If stalled for too long, add perturbation
                if stall_count > stall_threshold and alpha > 5:
                    # Add small random perturbation
                    noise = rng.normal(0, 0.01, n)
                    a = a + noise * a
                    a = np.clip(a, 0.0, None)
                    S_a = a.sum()
                    if S_a > 1e-15:
                        a = a / S_a
                    stall_count = 0

    # Final evaluation
    score = exact_score_fft(a, fft_size)
    if score < local_best:
        local_best = score
        best_a = a.copy()

    return best_a, local_best

    inits = []

    # 1. Given sequence (normalized)
    a0 = np.clip(init_raw, 0.0, 1000.0)
    if a0.sum() > 1e-12:
        inits.append(("given", a0 / a0.sum()))

    # 2. Constant
    inits.append(("const", np.ones(n) / n))

    # 3. Random Dirichlet
    inits.append(("random", rng.dirichlet(np.ones(n) * 0.5)))

    # 4. U-shaped (high at edges)
    x = np.linspace(0, 1, n)
    u = (x - 0.5) ** 2 + 0.01
    inits.append(("u_shape", u / u.sum()))

    # 5. Bimodal peaks
    peaks = np.exp(-((x - 0.25) / 0.08) ** 2) + np.exp(-((x - 0.75) / 0.08) ** 2)
    peaks += 0.01
    inits.append(("bimodal", peaks / peaks.sum()))

    # 6. Raised cosine (triangle)
    tri = 1.0 - np.abs(2.0 * x - 1.0) + 0.05
    inits.append(("triangle", tri / tri.sum()))

    # 7. Exponential decay
    exp_dec = np.exp(-3.0 * x) + 0.01
    inits.append(("exponential", exp_dec / exp_dec.sum()))

    # 8. Multi-frequency structure
    multi = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * x) + 0.2 * np.cos(2 * np.pi * 7 * x)
    multi = np.abs(multi) + 0.1
    inits.append(("multifreq", multi / multi.sum()))

    # 9. Polynomial (quartic)
    poly = x**4 + (1 - x)**4 + 0.1
    inits.append(("quartic", poly / poly.sum()))

    # 10. Random with structure (low-pass filtered noise)
    noise = rng.standard_normal(n)
    # Simple low-pass: moving average
    kernel = np.ones(20) / 20

    best_a = None

        remaining = deadline - time.monotonic()
        if remaining < 10.0:

            best_a = result_a

            best_a = result_a

    print(f"Final best score: {best_score:.6f}")

    if best_a is None:
        return [1.0 / n] * n

    return [float(x) for x in best_a]
