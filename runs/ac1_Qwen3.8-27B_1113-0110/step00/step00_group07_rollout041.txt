def compute_score_vec(a):
    """Compute the objective value from a numpy array."""
    n = len(a)
    c = np.convolve(a, a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    return 2.0 * n * np.max(c) / (S * S)

def solve_lp(f, M):
    """Solve the LP: max sum(g) s.t. (f*g)_k <= M, g >= 0."""
    n = len(f)
    # Constraint: for each lag k = 0, ..., 2n-2
    # sum_j f_j * g_{k-j} <= M
    # Coefficient of g_m in constraint k is f_{k-m} if 0 <= k-m <= n-1
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        # g_m appears when 0 <= k-m <= n-1, i.e., max(0, k-n+1) <= m <= min(k, n-1)
        m_start = max(0, k - n + 1)
        m_end = min(k, n - 1)
        for m in range(m_start, m_end + 1):
            rows.append(k)
            cols.append(m)
            vals.append(f[k - m])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M)

    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')

        return res.x
    return None
