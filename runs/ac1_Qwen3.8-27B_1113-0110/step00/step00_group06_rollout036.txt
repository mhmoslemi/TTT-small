def propose_candidate(seed=42, budget_s=1800):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 5
    
    # Start with given sequence
    a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
    a = a / a.sum()  # normalize to sum = 1
    
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    
    while time.monotonic() < deadline:
        M = np.max(np.convolve(a, a))
        
        # Solve LP
        g = solve_lp(a, M)
        if g is None:
            break
        g = np.clip(g, 0, None)
        if g.sum() < 1e-10:
            break
        g = g / g.sum()
        
        # Line search
        best_t = 0
        best_obj = np.inf
        for t in [0.01, 0.05, 0.1, 0.2, 0.5, 0.8, 1.0]:
            candidate = (1-t)*a + t*g
            candidate = np.clip(candidate, 0, None)
            score = evaluate_sequence(candidate.tolist())
            if score < best_obj:
                best_obj = score
                best_t = t
        
        if best_t > 0 and best_obj < best_score:
            a = (1-best_t)*a + best_t*g
            a = np.clip(a, 0, None)
            a = a / a.sum()
            best_a = a.copy()
            best_score = best_obj
        else:
            # No improvement, try different step sizes or break
            # Try smaller steps
            improved = False
            for t in np.linspace(0.001, 0.1, 20):
                candidate = (1-t)*a + t*g
                candidate = np.clip(candidate, 0, None)
                score = evaluate_sequence(candidate.tolist())
                if score < best_score:
                    a = candidate
                    a = a / a.sum()
                    best_a = a.copy()
                    best_score = score
                    improved = True
                    break
            if not improved:
                # Stuck, try a random restart or perturbation
                # Add small random perturbation
                a = a + rng.normal(0, 0.001, len(a))
                a = np.clip(a, 0, None)
                a = a / a.sum()
    
    return best_a.tolist()
