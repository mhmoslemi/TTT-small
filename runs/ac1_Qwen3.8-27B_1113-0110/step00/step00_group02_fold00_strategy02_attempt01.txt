                score_new = float(evaluate_sequence(new_seq))
                if score_new < best_score:
                    best_score = score_new
                    best_candidate = new_seq
                    improved = True
                    break
            else:
                # Directed update based on gradient sign
                grad_dir = 1.0 if gradients[idx] > update_min_effect else -1.0 if gradients[idx] < -update_min_effect else 0.0
                
                # Apply gradient direction with scaling to maintain boundedness
                factor = np.exp(-np.abs(gradients[idx]) * 0.001) if grad_dir > 0 else 1 + grad_dir * 0.02
                new_val_base = best_candidate[idx] * factor
                
                # Set lower/upper bounds based on gradient and current values
                if new_val_base < 0 or new_val_base > 1000:
                    # Fix to bounds if extreme move
                    new_val = min(max(new_val_base, 0), 1000)
                else:
                    # Stay within feasible domain
                    new_val = new_val_base
                
                # Small perturbation around the new value
                new_val_final = new_val * np.exp(random.uniform(-0.1, 0.1))
                new_seq = best_candidate.copy()
                new_seq[idx] = np.clip(new_val_final, 0, 1000)
                
                score_new = float(evaluate_sequence(new_seq))
                if score_new < best_score:
                    best_score = score_new
                    best_candidate = new_seq
                    improved = True
                    break
                
                # Also check the opposite direction
                if not improved and grad_dir == -1:
                    new_val = best_candidate[idx] * (1 / factor) * np.exp(random.uniform(-0.1, 0.1))
                    new_seq[idx] = np.clip(new_val, 0, 1000)
                    score_new = float(evaluate_sequence(new_seq))
                    if score_new < best_score:
                        best_score = score_new
                        best_candidate = new_seq
                        improved = True
                        break
        
        # Periodic full reconvolution and sniffer for binding constraints
        if time.monotonic() - start_time > 60:
            full_conv = np.convolve(best_candidate, best_candidate, mode='full')
            current_max = np.max(full_conv)
            current_sum = np.sum(best_candidate)
            
            # Down-scaled evaluations if needed
            cand1 = best_candidate.copy()
            cand1[0] = min(max(cand1[0] * 1.1, 0), 1000)
            cand1_score = float(evaluate_sequence(cand1))
            
            cand2 = best_candidate.copy()
            cand2[-1] = min(max(cand2[-1] * 1.1, 0), 1000)
            cand2_score = float(evaluate_sequence(cand2))
            
            cand3 = best_candidate.copy()
            # Balance first and last elements to S and current_max
            cand3[0] = best_candidate[0] * (current_sum / np.sqrt(2*n)) * 0.9
            cand3[-1] = best_candidate[-1] * (current_sum / np.sqrt(2*n)) * 0.9
            cand3_score = float(evaluate_sequence(cand3))
            
            cand4 = best_candidate.copy()
            # Increase number of zero elements (hopefully there's few)
            if any(val == 0 for val in cand4):
                balancer = sum([best_candidate[i]*0.9 for i in range(n)]) / current_sum
                cand4 = best_candidate.copy()
                cand4[np.random.choice([i for i in range(n) if best_candidate[i] < 0.05])] *= (balancer * 5)
            cand4_score = float(evaluate_sequence(cand4))
            
            # Update final value comparison
            best_score = min(cand1_score, cand2_score, cand3_score, cand4_score, best_score)
            if best_score < cand1_score: best_candidate = cand1
            if best_score < cand2_score: best_candidate = cand2
            if best_score < cand3_score: best_candidate = cand3
            if best_score < cand4_score: best_candidate = cand4
        
        num_evals -= 1
        
    # Terminal normalization and rounding if needed
    final_sequence = [float(x) for x in best_candidate]
    final_conv = np.convolve(final_sequence, final_sequence, mode='full')
    final_max = max(final_conv) if not np.isnan(final_conv).any() else 1e6
    final_sum = np.sum(final_sequence)
    
    # Return the best sequence
    return final_sequence
