        # Periodic progress reporting
        if iterations - last_print >= 5000:
            elapsed = time.monotonic() - (deadline - float(budget_s))
            print(f"  iter {iterations}, elapsed {elapsed:.0f}s, best={best_score:.12f}, improvements={improvements}")
            last_print = iterations
    
    elapsed_total = time.monotonic() - (deadline - float(budget_s))
    print(f"Done: n={n}, final_score={best_score:.12f}, iterations={iterations}, "
          f"improvements={improvements}, time={elapsed_total:.1f}s")
    print(f"Note: score=2.0 is the provable lower bound (Cauchy-Schwarz).")
    
    return [float(v) for v in best_arr]
