"""
Minimizes the evaluation function using a subgradient optimization method
with multiple restarts, momentum, and adaptive step sizes.
The key insight is to start from non-uniform initializations and use
projected subgradient descent on the probability simplex.
"""

import numpy as np
import time

def proj_simplex(v):
    """Project v onto the probability simplex {(x>=0, sum(x)=1)}."""
    # Simple approach: clip and normalize
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s > 0:
        return v / s
    return np.ones(len(v)) / len(v)

def grad_max_conv(a, p=100, beta=5.0):
    """
    Compute the smoothed subgradient of max_k (a*a)_k w.r.t. a.
    Uses top-p positions with softmax weights.
    """
    n = len(a)
    conv = np.convolve(a, a)
    L = len(conv)  # 2n-1
    
    if p >= L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(conv, -p)[-p:]
    
    top_vals = conv[idx]
    tmax = top_vals.max()
    exp_vals = np.exp((top_vals - tmax) * beta)
    w = exp_vals / exp_vals.sum()
    
    grad = np.zeros(n)
    for i in range(len(idx)):
        wk = w[i]
        if wk < 1e-10:
            continue
        k = idx[i]
        # First term: a[k-j] for j where 0 <= k-j < n
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            # a[k-j] for j from j_lo to j_hi
            # = a[k-j_hi : k-j_lo+1] reversed
            grad[j_lo:j_hi+1] += wk * a[k-j_hi:k-j_lo+1][::-1]
        # Second term: a[j-k] for j where 0 <= j-k < n
        if 0 <= k < n:
            grad[k:n] += wk * a[:n-k]
    
    return grad, conv.max()

def run_subgradient(a, rng, max_time=None, deadline=None, p=100, beta=5.0, 
                    momentum=0.9, eta0=None, verbose=False):
    """Run subgradient optimization on a (assumed normalized to sum 1)."""
    n = len(a)
    a = a.copy()
    a = proj_simplex(a)
    v = np.zeros(n)  # Momentum
    best_a = a.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))
    
    if eta0 is None:
        eta0 = 1.0 / np.sqrt(n)
    
    t = 0
    while True:
        if deadline is not None and time.monotonic() >= deadline:
            break
        if max_time is not None and t * 0.01 > max_time:  # Rough estimate
            break
        t += 1
        
        # Compute gradient
        grad, M = grad_max_conv(a, p=p, beta=beta)
        
        # Objective on simplex: 2n * M (since sum=1)
        # Gradient of objective: 2n * grad
        # Project to tangent space of simplex
        obj_grad = 2.0 * n * grad
        obj_grad = obj_grad - obj_grad.mean()  # Project to tangent space
        
        # Momentum update
        v = momentum * v + obj_grad
        
        # Adaptive step size
        eta = eta0 / np.sqrt(t + 1)
        
        # Update
        a_new = a - eta * v
        a_new = proj_simplex(a_new)
        
        # Check improvement (use cheap estimate)
        conv = np.convolve(a_new, a_new)
        score_est = 2.0 * n * conv.max()
        
        if score_est < best_score * 0.999:  # Relative improvement
            best_score = min(best_score, score_est)
            best_a = a_new.copy()
            # Verify with exact evaluation
            exact_score = float(evaluate_sequence(best_a.tolist()))
            if exact_score < best_score:
                best_score = exact_score
            elif exact_score < best_score + 0.01:
                best_score = exact_score
                if verbose:
                    print(f"  t={t}: exact score = {exact_score:.6f}")
        
        a = a_new
    
    return best_a, best_score

    """Project v onto probability simplex via clip + normalize."""
    v = np.clip(v, 0.0, None)
