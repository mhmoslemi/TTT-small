def compute_conv(a):
    return np.convolve(a, a)

def objective(a):
    n = len(a)
    b = np.convolve(a, a)
    return 2 * n * max(b) / (np.sum(a))**2

def projected_subgradient(a, step_size):
    n = len(a)
    b = np.convolve(a, a)
    # Find positions where b is near max
    max_b = max(b)
    # Top positions (within some fraction of max)
    threshold = max_b * 0.95
    active = np.where(b >= threshold)[0]
    
    # Compute gradient of sum of active b[k] w.r.t. a
    grad = np.zeros(n)
    for k in active:
        for j in range(n):
            other = k - j
            if 0 <= other < n:
                if other != j:
                    grad[j] += 2 * a[other]
                else:
                    grad[j] += a[other]
    
    # Projected gradient step
    a_new = a - step_size * grad
    a_new = np.maximum(a_new, 0)
    a_new = a_new / np.sum(a_new)  # normalize
    return a_new, max_b
