```python
    n = len(a)
```

```python
"""
Minimizes 2n*max(conv(a,a)) / (sum a)^2 using log-sum-exp (LSE) gradient
descent with FFT-accelerated convolutions and cross-correlations.
Employs multiple diverse initializations, adaptive LSE sharpness,
momentum, and Barzilai-Borwein step-size adaptation.
"""

import numpy as np
import time


def fft_conv2(a):
    """Linear convolution of a with itself via FFT. Returns length 2n-1."""
    n = a.size
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.real(np.fft.irfft(A * A, M)[:2 * n - 1])


def lse_subgrad(a, b, beta, Mcorr):
    """
    Subgradient of the log-sum-exp of b, w.r.t. a.
    g_j = 2 * sum_k w_k * a_{k-j}  (cross-correlation of a and w).
    """
    n = a.size
    sb = b - b.max()
    eb = np.exp(beta * sb)
    w = eb / eb.sum()
    Fa = np.fft.rfft(a, Mcorr)
    Fw = np.fft.rfft(w, Mcorr)
    g = np.real(np.fft.irfft(Fa * np.conj(Fw), Mcorr))[:n] * 2.0
    return g


def optimize_lse(a0, deadline, tag=""):
    """
    Run LSE gradient descent to minimize max(a*a).
    Returns (best_a, best_score) where best_score = 2n * max_b (sum a = 1).
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    n = a.size
    s = a.sum()
    if s < 1e-14:
        a[:] = 1.0 / n
        s = 1.0
    a /= s

    Mconv = 1 << (2 * n - 2).bit_length()
    Mcorr = 1 << (3 * n - 2).bit_length()

    b = fft_conv2(a)
    best_a = a.copy()
    best_mb = b.max()
    score = 2.0 * n * best_mb

    beta = 5.0
    lr = 0.5
    mom = 0.8
    v = np.zeros(n)

    # BB step size
    s_prev = np.zeros(n)
    g_prev = None
    bb_lr = None

    it = 0
    while time.monotonic() < deadline:
        # LSE gradient
        g = lse_subgrad(a, b, beta, Mcorr)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g_normed = g / gn

        # Barzilai-Borwein
        if g_prev is not None:
            y = g_normed - g_prev
            yn = np.linalg.norm(y)
            if yn > 1e-14 and np.dot(s_prev, y) > 1e-14:
                bb_lr = np.dot(s_prev, y) / (yn * yn)
                lr = np.clip(bb_lr, 0.01, 5.0)
        
        # Momentum step
        v = mom * v - lr * g_normed
        step_mag = lr * 0.5 / np.sqrt(n)  # scale step
        a_new = a + v * step_mag
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-12:
            a_new = a.copy()
            s_new = 1.0
            v = np.zeros(n)
        a_new /= s_new

        # Store for BB
        s_prev = a_new - a
        g_prev = g_normed
        a = a_new

        # Recompute convolution
        b = fft_conv2(a)
        mb = b.max()

        if mb < best_mb:
            best_mb = mb

    return best_a, 2.0 * n * best_mb
```
