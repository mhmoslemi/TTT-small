def compute_conv(f, n):
    """Compute linear convolution f*f of length 2n-1 using FFT."""
    size = 1
    while size < 2*n - 1:
        size *= 2
    F = np.fft.rfft(f, n=size)
    conv = np.fft.irfft(F * F, n=size)
    return conv[:2*n-1]

    w = np.exp(c_scaled - c_max)

import numpy as np
import time

def _next_pow2(n):
    """Next power of 2 >= n."""
    return 1 << (n - 1).bit_length()

def _convolve_fft(a, n):
    """Compute linear convolution a * a of length 2n-1 using real FFT."""
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    return np.fft.irfft(A * A, n=size)[:2*n-1]

def _gradient_fft(w, f, n):
    """Compute gradient[i] = 2 * sum_{j=0}^{n-1} w[i+j] * f[j]."""
    size = _next_pow2(3 * n - 2)
    W = np.fft.rfft(w, n=size)
    F = np.fft.rfft(f, n=size)
    corr = np.fft.irfft(np.conj(W) * F, n=size)
    return 2.0 * corr[:n]
