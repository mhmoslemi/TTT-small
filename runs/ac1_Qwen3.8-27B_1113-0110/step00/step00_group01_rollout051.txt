import numpy as np
import time

def fft_convolve(a, b):
    """Linear convolution using FFT."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

def fft_cross_corr(w, a):
    """Compute cross-correlation (w ⋆ a)[i] = sum_j w[i+j]*a[j] for i=0..len(a)-1."""
    n = len(w) + len(a) - 1
    size = 1 << (n - 1).bit_length()
    W = np.fft.rfft(w, size)
    A = np.fft.rfft(a, size)
    corr = np.fft.irfft(W * np.conj(A), size)
    return corr[:len(a)]

def compute_J_and_grad(a, p):
    """Compute J (using power mean) and its gradient."""
    n = len(a)
    conv = fft_convolve(a, a)
    S = np.sum(a)
    
    # Power mean
    conv_p = conv ** p
    C = np.sum(conv_p)
    M_p = C ** (1.0 / p)
    
    J = 2.0 * n * M_p / (S * S)
    
    # Gradient
    # dM_p/da[i] = M_p^(1-p) * G[i] where G[i] = 2*sum_j conv[i+j]^(p-1) * a[j]
    w = conv ** (p - 1)
    G = 2.0 * fft_cross_corr(w, a)
    
    dM_p = (M_p ** (1 - p)) * G
    
    grad = 2.0 * n * dM_p / (S * S) - 4.0 * n * M_p / (S * S * S)
    
    return J, grad

def gradient_descent(a_init, time_limit, p=8, seed=42):
    """Run gradient descent to minimize J."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() <= 0:
        a = np.ones(n)
    a = a / a.sum()  # normalize to sum=1
    
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + time_limit
    
    lr = 0.1
    best_a = a.copy()
    best_J = 2.0 * n * np.max(fft_convolve(a, a))  # sum=1 so J = 2n*max_conv
    
    iter_count = 0
    plateau = 0
    
    while time.monotonic() < deadline:
        J, grad = compute_J_and_grad(a, p)
        
        # Normalize gradient to unit norm for step size control
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:

        # Actually, let me just use the raw gradient since J is scale-invariant
        # The gradient already accounts for the scale through the S terms
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        
        if a_new.sum() < 1e-10:
            a_new = a.copy()
            lr *= 0.5
            a = a_new
            continue
        
        # Evaluate the true objective (with actual max, not power mean)
        conv_new = fft_convolve(a_new, a_new)
        S_new = np.sum(a_new)
        true_J = 2.0 * n * np.max(conv_new) / (S_new * S_new)
        
        if true_J < best_J:
            best_J = true_J
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.02, 1.0)
            plateau = 0
        else:

            lr *= 0.7
            plateau += 1

                a = np.maximum(a, 0)
                a = a / a.sum()
                lr = 0.05

        if iter_count % 500 == 0:
            print(f"    GD iter {iter_count}, best_J={best_J:.6f}, lr={lr:.6f}")
