Uses structured initialization + asymmetric-aware local search + SA.

import numpy as np
import time


def _safe_eval(seq):
    """Evaluate sequence, returning np.inf on failure."""
    try:
        return float(evaluate_sequence(seq))
    except Exception:
        return float('inf')


def _convolve_max(s):
    """Compute max of full self-convolution of s."""
    c = np.convolve(s, s, mode='full')
    return float(np.max(c))


def _evaluate_fast(s):
    """Fast evaluation without security checks (s already validated)."""
    n = len(s)
    S = float(np.sum(s))
    if S < 0.01:
        return float('inf')
    c = np.convolve(s, s, mode='full')
    max_c = float(np.max(c))
    return float(2.0 * n * max_c / (S * S))


def _init_constant(n):
    return np.ones(n, dtype=np.float64)


def _init_linear_slope(n, a):
    """s[i] = 1 + a*(2i/(n-1) - 1), anti-symmetric."""
    i = np.arange(n, dtype=np.float64)
    d = 2.0 * i / (n - 1) - 1.0
    s = 1.0 + a * d
    return np.clip(s, 0.0, 1000.0)


def _init_two_block(n, ratio):
    """First half = 1+ratio, second half = 1-ratio."""
    s = np.ones(n, dtype=np.float64)
    half = n // 2
    s[:half] = 1.0 + ratio
    s[half:] = 1.0 - ratio
    return np.clip(s, 0.0, 1000.0)


def _init_cosine(n, freq, amp):
    """s[i] = 1 + amp*cos(2*pi*freq*i/n)."""
    i = np.arange(n, dtype=np.float64)
    s = 1.0 + amp * np.cos(2.0 * np.pi * freq * i / n)
    return np.clip(s, 0.0, 1000.0)


def _init_quadratic(n, a):

    i = np.arange(n, dtype=np.float64)

    return np.clip(s, 0.0, 1000.0)


def _init_from_sequence(base_seq, n):
    """Resample/trim a given sequence to length n."""
    base = np.asarray(base_seq, dtype=np.float64)
    base = np.clip(base, 0.0, 1000.0)
    if len(base) == n:
        return base
    if len(base) < n:
        out = np.zeros(n, dtype=np.float64)
        out[:len(base)] = base
        out[len(base):] = base[len(base) % max(1, len(base))]
        return out
    # Interpolate down
    old_idx = np.linspace(0, 1, len(base))
    new_idx = np.linspace(0, 1, n)
    return np.interp(new_idx, old_idx, base).astype(np.float64)


def _antisymmetric_pair_move(s, rng, n, strength):
    """s[i] += delta, s[n-1-i] -= delta. Preserves sum."""
    out = s.copy()
    i = int(rng.integers(0, n))
    j = n - 1 - i
    if i == j:
        return out
    delta = rng.normal(0, strength)
    out[i] += delta
    out[j] -= delta
    np.clip(out, 0.0, 1000.0, out=out)
    # Renormalize to preserve sum
    S = np.sum(out)
    S_orig = np.sum(s)
    if S > 0:
        out *= S_orig / S
    return np.clip(out, 0.0, 1000.0)


def _block_perturbation(s, rng, n, width, sigma):
    """Multiply a random block by exp(N(0, sigma)), then renormalize."""
    out = s.copy()
    w = int(rng.integers(max(1, n // 200), max(2, min(width, n))))
    start = int(rng.integers(0, max(1, n - w)))
    out[start:start + w] *= np.exp(rng.normal(0, sigma))
    S = np.sum(out)
    S_orig = np.sum(s)
    if S > 0:
        out *= S_orig / S
    return np.clip(out, 0.0, 1000.0)


def _antisymmetric_block_move(s, rng, n, width, strength):
    """s[i:i+w] *= (1+eps), s[n-i-w:n-i] *= (1-eps). Roughly preserves sum."""
    out = s.copy()
    w = int(rng.integers(max(1, n // 50), max(2, min(width, n // 2))))
    i = int(rng.integers(0, max(1, n // 2 - w)))
    j = n - i - w
    eps = rng.normal(0, strength)
    out[i:i + w] *= (1.0 + eps)
    out[j:j + w] *= (1.0 - eps)
    np.clip(out, 0.0, 1000.0, out=out)
    S = np.sum(out)
    S_orig = np.sum(s)
    if S > 0:
        out *= S_orig / S
    return np.clip(out, 0.0, 1000.0)


def _tilt_move(s, rng, n, width, strength):
    """Add a linear ramp to a block, subtract from mirrored block."""
    out = s.copy()
    w = int(rng.integers(max(2, n // 50), max(3, min(width, n // 2))))
    i = int(rng.integers(0, max(1, n // 2 - w)))
    j = n - i - w
    ramp = np.linspace(-strength, strength, w)
    out[i:i + w] += ramp
    out[j:j + w] -= ramp
    np.clip(out, 0.0, 1000.0, out=out)
    S = np.sum(out)
    S_orig = np.sum(s)
    if S > 0:
        out *= S_orig / S
    return np.clip(out, 0.0, 1000.0)

        candidates.append(_init_quadratic(n, a))

    best_seq = None
    best_score = float('inf')

    for idx, c in enumerate(candidates):
        if time.monotonic() > phase1_deadline:
            break
        c = np.clip(c, 0.0, 1000.0)
        score = _evaluate_fast(c)
        if score < best_score:
            best_score = score
            best_seq = c.copy()
        if (idx + 1) % 10 == 0:
            print(f"  Phase 1: iter {idx+1}/{len(candidates)}, best so far: {best_score:.6f}")

    # Verify with actual evaluator
    best_score_verified = _safe_eval(best_seq.tolist())
    if not np.isfinite(best_score_verified) or best_score_verified > best_score + 0.01:
        best_score = best_score_verified
        if not np.isfinite(best_score):
            best_score = 2.0
            best_seq = np.ones(n, dtype=np.float64)

    print(f"  Phase 1 done. Best score: {best_score:.6f}")
    t_phase1_end = time.monotonic()
    print(f"  Phase 1 time: {t_phase1_end - t_phase1_start:.1f}s")

    # ===================== PHASE 2: Local Search =====================
    print(f"Phase 2: Local search with asymmetric-aware moves")
    t_phase2_start = time.monotonic()

    current = best_seq.copy()
    current_score = best_score
    global_best = best_seq.copy()
    global_best_score = best_score

    # SA parameters
    T_start = 0.05
    T_end = 0.001
    # We'll cool over the phase 2 duration
    phase2_duration = min(1500.0, max(0.0, (deadline - time.monotonic())))

    n_evals = 0
    n_accept = 0
    stall_count = 0
    PATIENCE = 5000

    move_types = ['antisym_pair', 'block', 'antisym_block', 'tilt', 'block2']

    while time.monotonic() < deadline:
        # Compute cooling factor
        elapsed = time.monotonic() - t_phase2_start
        frac = min(1.0, elapsed / max(1.0, phase2_duration))
        T = T_start * (T_end / T_start) ** frac
        # Strength also cools
        strength_base = 0.05 + 0.5 * (1.0 - frac)

        # Choose move type
        mt = move_types[int(rng.integers(0, len(move_types)))]

        if mt == 'antisym_pair':
            candidate = _antisymmetric_pair_move(current, rng, n, strength_base * 0.3)
        elif mt == 'block':
            w = int(n ** 0.5 * rng.uniform(0.5, 3.0))
            candidate = _block_perturbation(current, rng, n, min(w, n // 3), strength_base * 0.2)
        elif mt == 'antisym_block':
            w = int(n ** 0.5 * rng.uniform(1.0, 5.0))
            candidate = _antisymmetric_block_move(current, rng, n, min(w, n // 3), strength_base * 0.15)
        elif mt == 'tilt':
            w = int(n ** 0.5 * rng.uniform(1.0, 5.0))
            candidate = _tilt_move(current, rng, n, min(w, n // 3), strength_base * 0.1)
        else:  # block2 - larger random perturbation
            w = int(n * rng.uniform(0.05, 0.3))
            candidate = _block_perturbation(current, rng, n, min(w, n // 2), strength_base * 0.3)

        # Evaluate
        cand_score = _evaluate_fast(candidate)
        n_evals += 1

        # Accept/reject (SA)
        if cand_score < current_score:
            current = candidate
            current_score = cand_score
            n_accept += 1
            stall_count = 0
            if cand_score < global_best_score:
                global_best = candidate.copy()
                global_best_score = cand_score
                if n_evals % 5000 == 0:
                    print(f"  New best: {global_best_score:.6f} (eval {n_evals})")
        else:
            # SA acceptance
            delta = cand_score - current_score
            if delta > 0 and T > T_end:
                if rng.random() < np.exp(-delta / T):
                    current = candidate
                    current_score = cand_score
            stall_count += 1

            # If stalled too long, try a bigger kick
            if stall_count > PATIENCE:
                stall_count = 0
                # Large perturbation to escape
                w = int(n * rng.uniform(0.1, 0.4))
                current = _block_perturbation(current, rng, n, min(w, n // 2), 0.5)
                current_score = _evaluate_fast(current)
                n_evals += 1
                # Reset to global best if we're too far off
                if current_score > global_best_score * 1.5:
                    current = global_best.copy()
                    current_score = global_best_score

        # Periodic reporting
        if n_evals % 10000 == 0:
            elapsed_t = time.monotonic() - t_start
            remaining = max(0, deadline - time.monotonic())
            print(f"  Phase 2: eval={n_evals}, curr={current_score:.6f}, "
                  f"best={global_best_score:.6f}, T={T:.5f}, "
                  f"time={elapsed_t:.0f}s, remain={remaining:.0f}s")

    t_phase2_end = time.monotonic()
    print(f"  Phase 2 done. {n_evals} evals, {n_accept} accepts. "
          f"Best: {global_best_score:.6f}. Time: {t_phase2_end - t_phase2_start:.1f}s")

    # ===================== PHASE 3: Polish =====================
    print(f"Phase 3: Fine-grained polish")
    t_phase3_start = time.monotonic()
    phase3_deadline = min(deadline, t_phase3_start + 100.0)

    current = global_best.copy()
    current_score = global_best_score

    # Fine coordinate descent on random positions + small antisymmetric tweaks
    n_polish = 0
    small_strength = 0.01

    while time.monotonic() < phase3_deadline:
        # Pick 3-5 positions and nudge them
        num_pos = int(rng.integers(1, 6))
        idxs = rng.integers(0, n, size=num_pos)
        candidate = current.copy()
        for idx in idxs:
            candidate[idx] += rng.normal(0, small_strength)
        # Also try antisymmetric mirror
        if rng.random() < 0.5:
            for idx in idxs:
                mirror = n - 1 - idx
                candidate[mirror] -= (candidate[idx] - current[idx])
        np.clip(candidate, 0.0, 1000.0, out=candidate)
        # Renormalize
        S = np.sum(candidate)
        S_orig = np.sum(current)
        if S > 0:
            candidate *= S_orig / S
        np.clip(candidate, 0.0, 1000.0, out=candidate)

        cand_score = _evaluate_fast(candidate)
        n_polish += 1

        if cand_score < current_score:
            current = candidate
            current_score = cand_score
            if cand_score < global_best_score:
                global_best = candidate.copy()
                global_best_score = cand_score

    n_polish_total = n_polish
    print(f"  Phase 3 done. {n_polish_total} polish evals. Best: {global_best_score:.6f}")

    # ===================== FINAL VERIFICATION =====================
    final_seq = global_best
    final_score = _safe_eval(final_seq.tolist())
    if not np.isfinite(final_score):
        final_seq = np.ones(n, dtype=np.float64)
        final_score = _safe_eval(final_seq.tolist())

    print(f"\nFINAL: score={final_score:.6f}, length={len(final_seq)}")
    print(f"Total time: {time.monotonic() - t_start:.1f}s")

    return [float(x) for x in final_seq]
