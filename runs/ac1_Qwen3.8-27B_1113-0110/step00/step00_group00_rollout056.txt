"""
Minimize autocorrelation peak ratio using projected subgradient descent
with backtracking line search, multiple restarts, and soft-max smoothing.
Exploits scale invariance to fix sum(a)=1 and minimize 2*n*max(conv(a,a)).
"""
import numpy as np
import time

def _compute_conv(a):
    """Compute full linear convolution of a with itself."""
    return np.convolve(a, a, mode='full')

def _compute_score(a):
    """Compute 2*n*max_b/sum_a^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    conv = np.convolve(a, a, mode='full')
    M = conv.max()

    for k in k_positions:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            w_mirror[j_start:j_end+1] += a[k - np.arange(j_start, j_end + 1)]
    w_mirror /= len(k_positions)
    
    # For sum=1: grad of 2*n*M = 4*n*(w_mirror - M*a)
    # We want to DESCEND, so the descent direction is -grad = 4*n*(M*a - w_mirror)
    # But for projected gradient descent, we compute: a_new = a - lr * grad
    # where grad = 4*n*(w_mirror - M*a)
    grad = 4.0 * n * (w_mirror - M * a)
    return grad

def _score_normalized(a):
    """Compute 2*n*max(conv) for a sequence with sum=1."""
    n = len(a)

def _subgradient_direction(a, conv):
    """Compute subgradient of 2*n*max(conv) for a with sum=1.
    Returns gradient (descent is a - lr*grad).
    Uses average over all argmax positions for a valid subgradient."""
    n = len(a)
    M = conv.max()
    tol = 1e-9 * max(1.0, M)
    k_positions = np.where(conv >= M - tol)[0]
