def compute_convolution(a):
    """Compute full convolution of a with itself."""
    return np.convolve(a, a)

def compute_smooth_max(b, tau):
    """Compute smooth approximation of max(b) and weights."""
    b_shifted = b - np.max(b)
    exp_b = np.exp(b_shifted / tau)
    weights = exp_b / np.sum(exp_b)
    smooth_max = np.max(b) + tau * np.log(np.sum(exp_b))
    return smooth_max, weights

    # grad_i = 2 * sum_k w_k * a_{k-i} for valid k-i

def smooth_max_and_weights(b, tau):
    """Compute log-sum-exp smooth max and its weights."""
    b_max = np.max(b)

    exp_b = np.exp(b_shifted / tau)

def smooth_max(b, tau):
    """Log-sum-exp based smooth max."""
    b_max = np.max(b)
    shifted = b - b_max
    log_sum = np.log(np.sum(np.exp(shifted / tau)))
    return b_max + tau * log_sum

def p_norm(b, p):
    """Lp norm of b."""
    return np.sum(b ** p) ** (1.0 / p)
