    return g

def mw_step(a, g, eta):
    """Multiplicative weights step."""
    a_new = a * np.exp(-eta * g)
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S < 1e-10:
        return a.copy()
    return a_new / S
