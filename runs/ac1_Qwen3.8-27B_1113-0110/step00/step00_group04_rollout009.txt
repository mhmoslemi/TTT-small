```python
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp
import time

def solve_lp(f, M):
    """Solve: max sum(g) s.t. g >= 0, (f*g)_k <= M for all k."""
    n = len(f)
    # Constraint matrix A: (2n-1) x n
    # A[k, i] = f[k-i] if 0 <= k-i < n else 0
    # Use sparse construction
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            i = k - j  # index of g
            rows.append(k)
            cols.append(i)
            vals.append(f[j])
    
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2*n - 1, M)
    
    # Objective: maximize sum(g) = minimize -sum(g)
    c = -np.ones(n)
    
    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if res.status != 0:
        return None
    return res.x
```
