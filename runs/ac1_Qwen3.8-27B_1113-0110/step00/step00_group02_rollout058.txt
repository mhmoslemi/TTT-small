    p = 1

def project_nonneg_sum1(x):
    x = np.asarray(x, dtype=np.float64)
    n = x.size
    if n == 0:
        return x
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)

def project_decreasing_nonneg_sum1(x):
    x = np.asarray(x, dtype=np.float64)

            break
        xk = (C - s) / denom
        if not np.isfinite(xk) or xk < 1e-12:
            break
        x[k] = xk
        used_limit = k + 1

    if used_limit >= n:
        last = n - 1
    else:
        pos = np.nonzero(x[used_limit:] > 0.0)[0]
        if pos.size:
            last = used_limit + int(pos[-1])
        else:
            last = used_limit - 1
    if last < 0:
        x[0] = 1.0
        last = 0

    val = max(float(x[last]), 1e-12)
    for idx in range(last + 1, n):
        x[idx] = val * (0.82 ** (idx - last - 1))

    x = np.maximum(x, 0.0)
    s = float(np.sum(x))
    if not np.isfinite(s) or s <= 0.0:
        x[0] = 1.0
        s = x[0]
    return x / s


def make_inits(n, rng):
    inits = []
    if n <= 0:
        return inits

    ones = np.ones(n, dtype=np.float64) / float(n)
    inits.append(ones)

    if n > 1:
        i = np.arange(n, dtype=np.float64)
        denom = float(n - 1)

        for beta in (0.1, 0.35, 0.8, 1.6, 3.0):
            x = np.exp(-beta * i / denom)
            y = clean_init(x)
            if y is not None:
                inits.append(y)
            x = np.exp(beta * i / denom)
            y = clean_init(x)
            if y is not None:
                inits.append(y)

        for p in (0.2, 0.45, 0.8, 1.3, 2.0):
            x = 1.0 / np.power(i + 1.0, p)
            y = clean_init(x)
            if y is not None:
                inits.append(y)
            x = 1.0 / np.power(n - i, p)
            y = clean_init(x)
            if y is not None:
                inits.append(y)

        x = i + 1.0
        y = clean_init(x)
        if y is not None:
            inits.append(y)
        x = n - i
        y = clean_init(x)
        if y is not None:
            inits.append(y)

        z = i / denom - 0.5
        x = z * z + 0.08
        y = clean_init(x)
        if y is not None:
            inits.append(y)
        x = np.abs(z) + 0.04
        y = clean_init(x)
        if y is not None:
            inits.append(y)
        x = 1.0 + np.cos(2.0 * np.pi * i / float(n))
        y = clean_init(np.maximum(x, 1e-6))
        if y is not None:
            inits.append(y)

        x = np.power(1.0 + i / denom, 1.5)
        y = clean_init(x)
        if y is not None:
            inits.append(y)
        x = np.power(1.0 + (n - i) / denom, 1.5)
        y = clean_init(x)
        if y is not None:
            inits.append(y)

    for alpha in (0.2, 0.5, 1.0, 3.0, 10.0):
        try:
            y = rng.dirichlet(np.full(n, alpha, dtype=np.float64))
            y = clean_init(y)
            if y is not None:
                inits.append(y)
        except Exception:
            pass

    try:
        y = flat_prefix_init(n, rng)
        y = clean_init(y)
        if y is not None:
            inits.append(y)
    except Exception:
        pass

    if not inits:
        inits.append(np.ones(n, dtype=np.float64) / float(n))
    return inits


def fast_score(x):
    x = np.asarray(x, dtype=np.float64)
    n = x.size
    if n == 0:
        return float("inf")
    x = np.maximum(x, 0.0)
    s = float(np.sum(x))
    if not np.isfinite(s) or s < 0.01:
        return float("inf")
    y = x / s
    N = next_power_of_two(2 * n - 1)
    X = np.fft.rfft(y, N)
    b = np.fft.irfft(X * X, N)[: 2 * n - 1]
    b = np.maximum(b, 0.0)
    if b.size == 0:
        return float("inf")
    return float(2.0 * float(n) * float(np.max(b)))


def max_conv(x, N):
    X = np.fft.rfft(x, N)
    b = np.fft.irfft(X * X, N)
    b = np.maximum(b, 0.0)
    if b.size == 0:
        return 0.0
    return float(np.max(b))


def softmax_theta(theta):
    theta = np.asarray(theta, dtype=np.float64)
    n = theta.size
    if not np.all(np.isfinite(theta)):
        return np.ones(n, dtype=np.float64) / float(n)
    z = theta - float(np.max(theta))
    e = np.exp(z)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(n, dtype=np.float64) / float(n)
    return e / s


def log_from_x(x):
    x = np.asarray(x, dtype=np.float64)
    tiny = np.finfo(np.float64).tiny
    return np.log(np.maximum(x, tiny))


def smoothed_logsumexp_x(x, N, t):
    n = x.size
    X = np.fft.rfft(x, N)
    b = np.fft.irfft(X * X, N)[: 2 * n - 1]
    b = np.maximum(b, 0.0)
    if b.size == 0:
        return 0.0, np.zeros_like(x, dtype=np.float64)
    c = float(np.max(b))
    if not np.isfinite(c) or c <= 0.0:
        return 0.0, np.zeros_like(x, dtype=np.float64)

    z = (b - c) / t
    expz = np.exp(z)
    zsum = float(np.sum(expz))
    if not np.isfinite(zsum) or zsum <= 0.0:
        w = np.full_like(b, 1.0 / float(b.size), dtype=np.float64)
        L = 0.0
    else:
        w = expz / zsum
        L = c / t + float(np.log(zsum))

    if n <= 256:
        xpad = np.empty(n + 1, dtype=np.float64)
        xpad[:n] = x
        xpad[n] = 0.0
        corr = np.correlate(w, xpad, mode="valid")
        gradx = 2.0 * corr[:n]
    else:
        Wr = np.fft.rfft(w, N)
        gradx = 2.0 * np.fft.irfft(Wr * np.conj(X), N)[:n]

    gradx = np.where(np.isfinite(gradx), gradx, 0.0)
    return float(L), gradx


def lbfgs_fun(theta, n, N, t):
    theta = np.asarray(theta, dtype=np.float64)
    if theta.size != n or not np.all(np.isfinite(theta)):
        return 1e100, np.zeros(n, dtype=np.float64)

    x = softmax_theta(theta)
    L, gradx = smoothed_logsumexp_x(x, N, t)
    scale = 2.0 * float(n) * t
    val = scale * L
    s = float(np.dot(x, gradx))
    gradtheta = scale * x * (gradx - s)
    gradtheta = np.where(np.isfinite(gradtheta), gradtheta, 0.0)

    if not np.isfinite(val):
        val = 1e100
    return float(val), gradtheta.astype(np.float64, copy=False)


def spectral_shaping(x0, n, iters, alpha, mu0, mu1, rng=None, monotone=False):
    N = next_power_of_two(2 * n - 1)
    target = np.ones(2 * n - 1, dtype=np.float64)
    target /= float(2 * n - 1)
    Ft = np.fft.rfft(target, N)

    x = project_nonneg_sum1(x0)
    bestx = x.copy()

    Xs = np.fft.rfft(x, N)
    b = np.fft.irfft(Xs * Xs, N)[: 2 * n - 1]
    bestscore = float(2.0 * n * float(np.max(np.maximum(b, 0.0))))

    cur_alpha = float(alpha)
    last_score = bestscore
    stuck = 0

    for it in range(max(1, int(iters))):
        if mu1 <= 0.0 or mu0 <= 0.0:
            mu = max(1e-6, mu1)
        else:
            frac = float(it) / float(max(1, iters - 1))
            mu = mu0 * (mu1 / mu0) ** frac
            mu = float(np.clip(mu, 1e-6, 1e2))

        X = np.fft.rfft(x, N)
        den = np.abs(X) ** 2 + mu * mu
        Y = Ft * np.conj(X) / den
        Xn = (1.0 - cur_alpha) * X + cur_alpha * Y
        xn = np.fft.irfft(Xn, N)[:n]

        if monotone:
            x = project_decreasing_nonneg_sum1(xn)
        else:
            x = project_nonneg_sum1(xn)

        if (it % 16 == 0) or (it == iters - 1):
            Xs = np.fft.rfft(x, N)
            b = np.fft.irfft(Xs * Xs, N)[: 2 * n - 1]
            sc = float(2.0 * n * float(np.max(np.maximum(b, 0.0))))
            if np.isfinite(sc):
                if sc < bestscore - 1e-12:
                    bestscore = sc
                    bestx = x.copy()
                    last_score = sc
                    stuck = 0
                    cur_alpha = min(1.0, cur_alpha * 1.04)
                else:
                    stuck += 1
                    cur_alpha = max(0.08, cur_alpha * 0.88)
                    if stuck >= 3 and rng is not None:
                        perturbed = x + 0.04 * rng.normal(size=n)
                        if monotone:
                            x = project_decreasing_nonneg_sum1(perturbed)
                        else:
                            x = project_nonneg_sum1(perturbed)
                        stuck = 0
                    last_score = sc
            else:
                x = bestx.copy()
                cur_alpha = max(0.08, cur_alpha * 0.7)

    Xs = np.fft.rfft(x, N)
    b = np.fft.irfft(Xs * Xs, N)[: 2 * n - 1]
    sc = float(2.0 * n * float(np.max(np.maximum(b, 0.0))))
    if np.isfinite(sc) and sc < bestscore:
        bestscore = sc
        bestx = x.copy()

    return bestx, bestscore


def tiled_sequence(base, factor, phase=0):
    base = np.asarray(base, dtype=np.float64)
    m = base.size
    factor = int(factor)
    if m <= 0 or factor <= 0:
        return np.array([], dtype=np.float64)
    base = project_nonneg_sum1(base)
    idx = (int(phase) + np.arange(m * factor, dtype=np.int64)) % m
    y = base[idx] / float(factor)
    return project_nonneg_sum1(y)


def add_candidate(candidates, x, score):
    x = np.asarray(x, dtype=np.float64)
    if x.size == 0 or not np.isfinite(score):
        return
    if np.any(np.isnan(x)) or np.any(np.isinf(x)):
        return
    x = project_nonneg_sum1(x)
    score = fast_score(x)
    if not np.isfinite(score):
        return
    candidates.append((float(score), x, int(x.size)))


def lp_refine_sequence(x0, max_seconds, deadline):
    x0 = np.asarray(x0, dtype=np.float64)
    n = x0.size
    if n <= 1 or n > 512 or not HAVE_LP:
        return project_nonneg_sum1(x0), fast_score(x0)

    start = time.monotonic()
    limit_time = min(start + max(0.0, float(max_seconds)), deadline)
    bestx = project_nonneg_sum1(x0)
    bestscore = fast_score(bestx)
    f = bestx.copy()
    current = bestscore

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)

    steps = 0
    while time.monotonic() < limit_time - 2.0 and steps < 18:
        A_ub = np.zeros((2 * n - 1, n + 1), dtype=np.float64)
        for k in range(2 * n - 1):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if hi >= lo:
                A_ub[k, lo:hi + 1] = f[hi:lo - 1:-1]
        A_ub[:, -1] = -1.0

        c = np.zeros(n + 1, dtype=np.float64)
        c[-1] = 1.0
        b_ub = np.zeros(2 * n - 1, dtype=np.float64)
        bounds = [(0.0, None)] * n + [(0.0, None)]

        res = None
        try:
            res = linprog(
                c,
                A_ub=A_ub,
                b_ub=b_ub,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
                method="highs",
                options={"presolve": True},
            )
        except Exception:
            res = None

        if res is None or not getattr(res, "success", False):
            try:
                res = linprog(
                    c,
                    A_ub=A_ub,
                    b_ub=b_ub,
                    A_eq=A_eq,
                    b_eq=b_eq,
                    bounds=bounds,
                    method="interior-point",
                )
            except Exception:
                res = None

        if res is None or not getattr(res, "success", False):
            break

        g = np.asarray(res.x[:n], dtype=np.float64)
        g = project_nonneg_sum1(g)

        local_f = f.copy()
        local_score = current

        for lam in (0.15, 0.3, 0.5, 0.7, 1.0):
            cand = (1.0 - lam) * f + lam * g
            cand = project_nonneg_sum1(cand)
            sc = fast_score(cand)
            if np.isfinite(sc) and sc < local_score - 1e-12:
                local_f = cand.copy()
                local_score = sc

        if local_score < current - 1e-12:
            f = local_f.copy()
            current = local_score
            if current < bestscore:
                bestscore = current
                bestx = local_f.copy()
            steps += 1
        else:
            break

    finalscore = fast_score(bestx)
    return bestx, finalscore


def run_lbfgs_polish(x0, max_seconds, rng, deadline):
    x0 = np.asarray(x0, dtype=np.float64)
    n = x0.size
    if n <= 1:
        return project_nonneg_sum1(x0), fast_score(x0)

    N = next_power_of_two(2 * n - 1)
    start = time.monotonic()
    limit_time = min(start + max(0.0, float(max_seconds)), deadline)

    x = project_nonneg_sum1(x0)
    bestx = x.copy()
    bestscore = fast_score(x)
    theta = log_from_x(x)
    besttheta = theta.copy()
    no_improve = 0

    t_schedule = (0.25, 0.16, 0.10, 0.065, 0.042, 0.028, 0.018, 0.012, 0.008, 0.005)
    if n <= 256:
        chunk_iters = 20
    elif n <= 1024:
        chunk_iters = 12
    elif n <= 3000:
        chunk_iters = 8
    else:
        chunk_iters = 6

    idx = 0
    used_sgd = False

    while time.monotonic() < limit_time - 2.0:
        xcur = softmax_theta(theta)
        bmax = max_conv(xcur, N)
        factor = t_schedule[min(idx, len(t_schedule) - 1)]
        t = max(float(bmax) * factor, 1e-12)

        if HAVE_SCIPY:
            try:
                res = minimize(
                    lbfgs_fun,
                    theta,
                    jac=True,
                    args=(n, N, t),
                    method="L-BFGS-B",
                    options={
                        "maxiter": chunk_iters,
                        "ftol": 1e-12,
                        "gtol": 1e-5,
                        "maxls": 12,
                    },
                )
                if res.x is not None and np.all(np.isfinite(res.x)):
                    theta = res.x.astype(np.float64, copy=False)
            except Exception:
                used_sgd = True
        else:
            used_sgd = True

        if used_sgd:
            lr = 0.02 / max(1.0, float(n) ** (-0.3))
            for _ in range(8):
                fval, g = lbfgs_fun(theta, n, N, t)
                if not np.isfinite(fval):
                    break
                gnorm = float(np.linalg.norm(g, ord=np.inf))
                if not np.isfinite(gnorm) or gnorm <= 0.0:
                    break
                th2 = theta - (lr / gnorm) * g
                f2, g2 = lbfgs_fun(th2, n, N, t)
                if np.isfinite(f2) and f2 < fval:
                    theta = th2
                    lr *= 1.05
                else:
                    lr *= 0.5
                    break

        x = softmax_theta(theta)
        score = fast_score(x)

        if np.isfinite(score):
            if score < bestscore - 1e-12:
                bestscore = score
                bestx = x.copy()
                besttheta = theta.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve >= 4:
                    if rng is not None:
                        theta = besttheta + 0.04 * rng.normal(size=n)
                    else:
                        theta = besttheta.copy()
                    no_improve = 0
        else:
            theta = besttheta.copy()
            no_improve = 0

        idx += 1

    finalscore = fast_score(bestx)
    return bestx, finalscore


def refine_candidate(x, budget, rng, deadline):
    x = project_nonneg_sum1(x)
    n = int(x.size)
    startc = time.monotonic()
    bestx = x.copy()
    bestscore = fast_score(x)

    if HAVE_LP and n <= 384 and budget > 25.0:
        lp_budget = min(90.0, budget * 0.38, deadline - time.monotonic() - 5.0)
        if lp_budget > 12.0:
            try:
                x2, sc2 = lp_refine_sequence(x, lp_budget, deadline)
                if sc2 < bestscore:
                    bestscore = sc2
                    bestx = x2.copy()
            except Exception:
                pass

    rem = deadline - time.monotonic()
    used = time.monotonic() - startc
    lb_budget = min(max(0.0, rem - 2.0), max(0.0, budget - used))
    if lb_budget > 12.0:
        try:
            x3, sc3 = run_lbfgs_polish(bestx, lb_budget, rng, deadline)
            if sc3 < bestscore:
                bestscore = sc3
                bestx = x3.copy()
        except Exception:
            pass

    finalscore = fast_score(bestx)
    return bestx, finalscore


def propose_candidate(seed=42, budget_s=1800.0, **kwargs):
    """
    Hybrid search for nonnegative step functions minimizing
    2*n*max(x*x)/(sum(x))^2.

    Uses:
      - spectral target-flat convolution shaping,
      - LP response-shaping for small bases,
      - softmax L-BFGS smoothing of the log-sum-exp max,
      - multiple structured and random starts,
      - tiling of good short sequences to longer outputs.
    """
    budget_s = float(budget_s)
    if not np.isfinite(budget_s) or budget_s <= 10.0:
        budget_s = 10.0

    start = time.monotonic()
    deadline = start + max(10.0, budget_s - 5.0)
    rng = np.random.default_rng(int(seed) if np.isfinite(seed) else 42)

    parent = np.asarray(globals().get("height_sequence_1", []), dtype=np.float64)
    if parent.ndim != 1 or parent.size == 0 or not np.all(np.isfinite(parent)):
        parent = np.ones(64, dtype=np.float64)
    parent = project_nonneg_sum1(parent)

    best_x = parent.copy()
    best_score = fast_score(best_x)
    candidates = []
    add_candidate(candidates, best_x, best_score)

    phase_a_deadline = start + max(25.0, min(300.0, 0.20 * budget_s, budget_s - 90.0))
    phase_a_deadline = min(phase_a_deadline, deadline - 120.0)

    if budget_s < 300.0:
        n_list = [8, 16, 24, 32, 48, 64, 96, 128, 256, 512]
    else:
        n_list = [
            8, 12, 16, 24, 32, 48, 64, 96, 128, 192, 256,
            384, 512, 768, 1024, 1536, 2048, 3072, 4096,
        ]

    print(f"[{time.monotonic()-start:7.1f}s] initial score = {best_score:.6f}, n = {best_x.size}", flush=True)

    for n in n_list:
        if time.monotonic() > phase_a_deadline:
            break
        count = 5 if n <= 64 else (3 if n <= 256 else 2)
        inits = make_inits(n, rng)
        alphas = (0.35, 0.55, 0.75, 0.90)
        mus = (0.18, 0.09, 0.05, 0.08)

        for s in range(count):
            if time.monotonic() > phase_a_deadline:
                break
            x0 = inits[s % len(inits)]
            alpha = alphas[s % len(alphas)]
            mu0 = mus[s % len(mus)]
            if n <= 128:
                iters = 320
            elif n <= 512:
                iters = 220
            else:
                iters = 120
            monotone = (s == 1)
            try:
                x, sc = spectral_shaping(x0, n, iters, alpha, mu0, 1e-4, rng, monotone)
            except Exception:
                continue
            if np.isfinite(sc):
                add_candidate(candidates, x, sc)
                if sc < best_score:
                    best_score = sc
                    best_x = x.copy()
                    print(f"[{time.monotonic()-start:7.1f}s] phase A improved: score = {best_score:.6f}, n = {n}", flush=True)

    # Add tiled variants of the currently best short sequences.
    if candidates:
        sorted_cands = sorted(candidates, key=lambda z: z[0])
        for sc, cand_x, cand_n in sorted_cands[:6]:
            if cand_n > 1024:
                continue
            for factor in (2, 3, 4):
                for phase in (0, cand_n // 3, 2 * cand_n // 3):
                    y = tiled_sequence(cand_x, factor, phase)
                    if y.size > 4096:
                        continue
                    ysc = fast_score(y)
                    if np.isfinite(ysc):
                        add_candidate(candidates, y, ysc)
                        if ysc < best_score:
                            best_score = ysc
                            best_x = y.copy()
                            print(f"[{time.monotonic()-start:7.1f}s] tiled improved: score = {best_score:.6f}, n = {y.size}", flush=True)

    # Select distinct lengths for refinement.
    selected = []
    seen_n = set()
    if candidates:
        sorted_cands = sorted(candidates, key=lambda z: z[0])
        for sc, cand_x, cand_n in sorted_cands:
            if cand_n in seen_n:
                continue
            if cand_n > 4096:
                continue
            selected.append((sc, cand_x, cand_n))
            seen_n.add(cand_n)
            if len(selected) >= 8:
                break

    # Always refine the global best if not already included.
    if not any(abs(cand_n - best_x.size) < 1 for _, _, cand_n in selected):
        selected.append((best_score, best_x.copy(), int(best_x.size)))

    # Add the original parent if it is not too large and not selected.
    if parent.size > 0 and int(parent.size) <= 6962 and int(parent.size) not in seen_n:
        selected.append((fast_score(parent), parent.copy(), int(parent.size)))

    print(f"[{time.monotonic()-start:7.1f}s] phase A done, best = {best_score:.6f}, candidates = {len(candidates)}", flush=True)

    remaining = deadline - time.monotonic()
    if remaining > 60.0 and selected:
        # Spread budget with emphasis on the best candidates.
        weights = [max(1.0, 2.5 - 0.25 * k) for k in range(len(selected))]
        wsum = float(sum(weights))
        base_share = max(0.0, (remaining - 20.0) * 0.92 / wsum)

        for k, (sc, cand_x, cand_n) in enumerate(selected):
            if time.monotonic() > deadline - 25.0:
                break
            share = base_share * weights[k]
            share = min(360.0, max(20.0, share))
            share = min(share, deadline - time.monotonic() - 25.0)
            if share < 10.0:
                break

            try:
                x2, sc2 = refine_candidate(cand_x, share, rng, deadline)
            except Exception:
                x2 = cand_x
                sc2 = sc

            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_x = x2.copy()
                print(f"[{time.monotonic()-start:7.1f}s] refinement improved: score = {best_score:.6f}, n = {int(x2.size)}", flush=True)

                # Try a quick tiling of the new best if it is short.
                if x2.size <= 512:
                    for factor in (2, 3, 4, 6):
                        y = tiled_sequence(x2, factor, 0)
                        if y.size > 8192:
                            continue
                        ysc = fast_score(y)
                        if np.isfinite(ysc) and ysc < best_score:
                            best_score = ysc
                            best_x = y.copy()
                            print(f"[{time.monotonic()-start:7.1f}s] final tiled improvement: score = {best_score:.6f}, n = {y.size}", flush=True)

    # Final polish with remaining time on the best candidate only.
    if best_x.size > 0 and time.monotonic() < deadline - 45.0:
        final_budget = min(deadline - time.monotonic() - 5.0, 420.0)
        if final_budget > 30.0:
            try:
                xfin, scfin = refine_candidate(best_x, final_budget, rng, deadline)
                if scfin < best_score:
                    best_score = scfin
                    best_x = xfin.copy()
                    print(f"[{time.monotonic()-start:7.1f}s] final polish improved: score = {best_score:.6f}, n = {int(xfin.size)}", flush=True)
            except Exception:
                pass

    # Make absolutely sure the returned object is valid.
    best_x = np.asarray(best_x, dtype=np.float64)
    if best_x.size == 0:
        best_x = np.ones(64, dtype=np.float64)
    best_x = project_nonneg_sum1(best_x)

    internal_score = fast_score(best_x)
    try:
        exact_score = float(evaluate_sequence(best_x.tolist()))
    except Exception:
        exact_score = internal_score

    if not np.isfinite(exact_score):
        fallback = np.ones(64, dtype=np.float64)
        try:
            exact_score = float(evaluate_sequence(fallback.tolist()))
        except Exception:
            exact_score = float("inf")
        print(f"[{time.monotonic()-start:7.1f}s] fallback used, score = {exact_score:.6f}", flush=True)
        return fallback.tolist()

    print(f"[{time.monotonic()-start:7.1f}s] FINAL score = {exact_score:.6f}, n = {int(best_x.size)}", flush=True)
    return [float(v) for v in best_x.tolist()]
