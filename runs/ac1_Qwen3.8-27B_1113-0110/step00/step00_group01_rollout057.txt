   auto-convolution profile.

The program always returns the best valid sequence found before the budget
expires.
"""

import time
import math
import numpy as np


def next_pow2(x):
    p = 1
    while p < int(x):
        p *= 2
    return p


def normalize_nonnegative(v):
    """Return a normalized non-negative float vector, or None if invalid."""
    v = np.asarray(v, dtype=np.float64).reshape(-1)
    v = np.where(np.isfinite(v), v, 0.0)
    v = np.maximum(v, 0.0)
    s = float(np.sum(v))
    if not math.isfinite(s) or s <= 1e-300:
        return None
    return v / s


def convolution_coeffs(a):
    """Full linear convolution a*a, clipped to non-negative values."""
    n = int(np.size(a))
    if n <= 0:
        return np.array([], dtype=np.float64)
    L = 2 * n - 1
    if n < 128:
        c = np.convolve(a, a)
    else:
        p = next_pow2(L)
        fa = np.fft.rfft(a, p)
        c = np.fft.irfft(fa * fa, p)[:L]
    c = np.asarray(c, dtype=np.float64)
    np.maximum(c, 0.0, out=c)
    return c


def fft_score(a):
    """Fast score equivalent to evaluate_sequence for non-negative normalized a."""
    a = normalize_nonnegative(a)
    if a is None:
        return np.inf
    c = convolution_coeffs(a)
    if c.size == 0:
        return np.inf
    s = float(np.sum(a))
    if s <= 0.0 or not math.isfinite(s):
        return np.inf
    return float(2.0 * len(a) * float(np.max(c)) / (s * s))


def update_fft_best(a, score, best_seq, best_score):
    if a is None:
        return best_seq, best_score
    if not np.isfinite(score):
        return best_seq, best_score
    if score < best_score:
        return np.array(a, dtype=np.float64, copy=True), float(score)
    return best_seq, best_score


def update_best_exact(a, best_seq, best_score):
    """Evaluate with the authoritative global evaluate_sequence if available."""
    if a is None:
        return best_seq, best_score
    try:
        lst = [float(x) for x in a]
        sc = evaluate_sequence(lst)
    except Exception:
        return best_seq, best_score
    try:
        sc = float(sc)
    except Exception:
        return best_seq, best_score
    if np.isfinite(sc) and sc < best_score:
        return np.array(a, dtype=np.float64, copy=True), sc
    return best_seq, best_score


def cross_corr_b(a, w):
    """
    Compute b_i = sum_{m=0}^{n-1} a_m * w_{m+i}, for i=0..n-1.
    This is the gradient multiplier for log-sum-exp auto-convolution.
    """
    n = int(np.size(a))
    L = 2 * n - 1
    if n < 128:
        b = np.empty(n, dtype=np.float64)
        for i in range(n):
            b[i] = float(np.dot(a, w[i:i + n]))
        np.maximum(b, 0.0, out=b)
        return b

    p = next_pow2(n + L - 1)
    fa = np.fft.rfft(a, p)
    fw = np.fft.rfft(w[::-1], p)
    conv = np.fft.irfft(fa * fw, p)
    b = conv[n - 1:2 * n - 1][::-1]
    b = np.asarray(b, dtype=np.float64)
    np.maximum(b, 0.0, out=b)
    return b


def multiplicative_optimize(a, deadline, rng, gamma_start=8.0, gamma_max=80.0,
                            floor_frac=1e-7):
    """
    Multiplicative gradient-style optimization.

    It maintains non-negativity and normalizes after each accepted step.
    The step is accepted only if the fast score improves.
    """
    a = normalize_nonnegative(a)
    if a is None:
        return None, np.inf

    best = np.array(a, dtype=np.float64, copy=True)
    best_score = fft_score(a)
    current = np.array(a, dtype=np.float64, copy=True)
    n = len(best)
    floor = max(1e-15, floor_frac * (1.0 / n))

    gamma = float(gamma_start)
    step = 0.12
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        c = convolution_coeffs(current)
        if c.size == 0:
            break
        maxc = float(np.max(c))
        if maxc <= 1e-300 or not math.isfinite(maxc):
            break

        # Relative deficit weights: peak has weight 1.
        rel = (maxc - c) / maxc
        w = np.exp(-gamma * rel)
        b = cross_corr_b(current, w)
        maxb = float(np.max(b))
        if maxb <= 1e-300 or not math.isfinite(maxb):
            break

        x = b / maxb
        xmin = float(np.min(x))
        improved = False

        steps = []
        for st in (step, 0.5 * step, 2.0 * step, 0.05, 0.25, 0.8):
            st = float(st)
            if 0.01 <= st <= 2.5:
                if st not in steps:
                    steps.append(st)

        for st in steps:
            factor = np.exp(-st * (x - xmin))
            cand = current * factor
            cand = np.maximum(cand, floor)
            cand = normalize_nonnegative(cand)
            if cand is None:
                continue
            sc = fft_score(cand)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best = cand
                best_score = sc
                current = cand
                step = min(1.7, max(0.03, st * 1.18))
                improved = True
                no_improve = 0
                break

        if improved:
            gamma = min(float(gamma_max), gamma * 1.04 + 0.05)
        else:
            no_improve += 1
            step = max(0.02, step * 0.72)
            if no_improve >= 25:
                escaped = False
                for sig in (0.02, 0.05, 0.10, 0.20, 0.35):
                    noise = np.exp(rng.normal(0.0, sig, n))
                    cand = current * noise
                    cand = np.maximum(cand, floor)
                    cand = normalize_nonnegative(cand)
                    if cand is None:
                        continue
                    sc = fft_score(cand)
                    if np.isfinite(sc) and sc < best_score:
                        best = cand
                        best_score = sc
                        current = cand
                        escaped = True
                        break
                if escaped:
                    no_improve = 0
                    gamma = max(5.0, gamma * 0.75)
                    step = 0.08
                else:
                    if step < 0.025:
                        break
        it += 1
        if it % 500 == 0:
            current = np.array(best, dtype=np.float64, copy=True)

    return best, best_score


def theta_to_a(theta, powers):
    """Convert exponent-polynomial coefficients to a non-negative sequence."""
    theta = np.asarray(theta, dtype=np.float64).reshape(-1)
    if powers is None or powers.size == 0:
        return None
    logh = powers @ theta
    logh = np.where(np.isfinite(logh), logh, -1e30)
    m = float(np.max(logh))
    if not math.isfinite(m):
        return None
    h = np.exp(logh - m)
    return normalize_nonnegative(h)


def theta_score(theta, powers):
    a = theta_to_a(theta, powers)
    if a is None:
        return np.inf
    return fft_score(a)


def shape_search(n, deadline, rng):
    """
    Stochastic population search over theta in exp(sum theta_j x^j).
    Returns a sorted list of (theta, score) and the powers matrix for n.
    """
    x = np.linspace(0.0, 1.0, n)
    x2 = x * x
    x3 = x2 * x
    x4 = x3 * x
    x5 = x4 * x
    powers = np.column_stack((x, x2, x3, x4, x5))

    pop = []
    scores = []

    inits = []
    inits.append(np.zeros(5, dtype=np.float64))
    inits.append(np.array([2.0, 0.0, 0.0, 0.0, 0.0], dtype=np.float64))
    inits.append(np.array([-2.0, 0.0, 0.0, 0.0, 0.0], dtype=np.float64))
    inits.append(np.array([3.0, -3.0, 0.0, 0.0, 0.0], dtype=np.float64))
    inits.append(np.array([-3.0, 3.0, 0.0, 0.0, 0.0], dtype=np.float64))
    inits.append(np.array([1.5, -2.0, 1.0, 0.0, 0.0], dtype=np.float64))
    inits.append(np.array([-1.5, 2.0, -1.0, 0.0, 0.0], dtype=np.float64))
    for _ in range(24):
        inits.append(np.clip(rng.normal(0.0, 1.5, 5), -5.0, 5.0))

    for th in inits:
        th = np.asarray(th, dtype=np.float64).reshape(-1)
        if th.size != 5:
            continue
        sc = theta_score(th, powers)
        if np.isfinite(sc):
            pop.append(th.copy())
            scores.append(float(sc))

    if not pop:
        z = np.zeros(5, dtype=np.float64)
        pop.append(z.copy())
        scores.append(2.0)

    sigma = 1.0
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        if len(pop) <= 1:
            th = np.clip(rng.normal(0.0, 2.0, 5), -5.0, 5.0)
            sc = theta_score(th, powers)
            if np.isfinite(sc):
                pop.append(th)
                scores.append(float(sc))
            it += 1
            continue

        # Occasionally replace the worst member to escape local basins.
        if rng.random() < 0.25:
            idx = int(np.argmax(scores))
        else:
            idx = int(rng.integers(len(pop)))

        base = pop[idx]
        new_th = base + sigma * rng.normal(0.0, 1.0, 5)
        new_th = np.clip(new_th, -5.0, 5.0)
        sc = theta_score(new_th, powers)

        if np.isfinite(sc) and sc < scores[idx]:
            pop[idx] = new_th
            scores[idx] = float(sc)
            best_s = float(np.min(scores))
            if sc <= best_s + 1e-12:
                no_improve = 0
                sigma = min(2.5, sigma * 1.02)
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 180:
            sigma = max(0.25, sigma * 0.75)
            no_improve = 0
            if len(pop) < 36:
                th = np.clip(rng.normal(0.0, 2.5, 5), -5.0, 5.0)
                sc = theta_score(th, powers)
                if np.isfinite(sc):
                    pop.append(th)
                    scores.append(float(sc))

        if it % 250 == 0 and len(pop) > 20:
            order = np.argsort(np.asarray(scores, dtype=np.float64))
            keep = int(order[:20])
            pop = [pop[i] for i in keep]
            scores = [float(scores[i]) for i in keep]

        it += 1

    order = np.argsort(np.asarray(scores, dtype=np.float64))
    out = []
    seen = set()
    for i in order:
        th = pop[i]
        key = tuple(np.round(th, 6).tolist())
        if key in seen:
            continue
        seen.add(key)
        out.append((th, float(scores[i])))
        if len(out) >= 12:
            break
    return out, powers


def make_starts_for_length(n, theta_list, rng, best_fft_seq, best_fft_score, max_starts=4):
    """Create a small set of good initial sequences for a requested length."""
    n = int(n)
    if n <= 0:
        return []

    starts = []
    scs = []

    def add_sequence(a):
        a = normalize_nonnegative(a)
        if a is None:
            return
        sc = fft_score(a)
        if np.isfinite(sc):
            starts.append(a)
            scs.append(float(sc))

    # Theta-generated shapes.
    if theta_list is not None:
        x = np.linspace(0.0, 1.0, n)
        x2 = x * x
        x3 = x2 * x
        x4 = x3 * x
        x5 = x4 * x
        powers = np.column_stack((x, x2, x3, x4, x5))
        for th in theta_list[:10]:
            add_sequence(theta_to_a(th, powers))

        # Slightly perturbed theta shapes for diversity.
        for _ in range(4):
            if theta_list:
                base = theta_list[int(rng.integers(min(5, len(theta_list))))]
                th = np.clip(base + 0.25 * rng.normal(0.0, 1.0, 5), -5.0, 5.0)
                add_sequence(theta_to_a(th, powers))

    # Pure exponentials.
    x = np.linspace(0.0, 1.0, n)
    for lam in (-3.0, -1.5, -0.7, 0.7, 1.5, 3.0):
        h = np.exp(lam * x)
        add_sequence(h)

    # Current global best, if compatible.
    if best_fft_seq is not None and len(best_fft_seq) == n:
        add_sequence(best_fft_seq)

    # For short lengths, add random Dirichlet starts.
    if n < 160:
        for _ in range(8):
            a = rng.dirichlet(np.ones(n))
            add_sequence(a)

    if not starts:
        add_sequence(np.ones(n))

    order = np.argsort(np.asarray(scs, dtype=np.float64))
    out = []
    for i in order[:max_starts]:
        out.append(starts[i])
    return out


def search_small(deadline, rng, best_seq, best_exact_score, best_fft_seq, best_fft_score):
    """
    Focus on short sequences, where the minimax can sometimes be much lower.
    """
    for n in (2, 3, 4, 5, 6, 7, 8, 9, 10, 12):
        if time.monotonic() >= deadline:
            break

        inits = []
        if n == 2:
            inits.append(np.array([1.0 / 3.0, 2.0 / 3.0], dtype=np.float64))
        if n == 3:
            inits.append(np.array([3.0 / 15.0, 4.0 / 15.0, 8.0 / 15.0], dtype=np.float64))

        if n > 1:
            x = np.linspace(0.0, 1.0, n)
        else:
            x = np.zeros(n)

        inits.append(np.ones(n, dtype=np.float64))
        for lam in (-3.0, -1.5, -0.7, 0.7, 1.5, 3.0):
            inits.append(np.exp(lam * x))
        for _ in range(12):
            inits.append(rng.dirichlet(np.ones(n)))

        evals_a = []
        evals_s = []
        for arr in inits:
            a = normalize_nonnegative(arr)
            if a is None:
                continue
            sc = fft_score(a)
            if np.isfinite(sc):
                evals_a.append(a)
                evals_s.append(float(sc))
                old_fft = best_fft_score
                best_fft_seq, best_fft_score = update_fft_best(a, sc, best_fft_seq, best_fft_score)
                if best_fft_score != old_fft and (best_seq is None or best_fft_score < best_exact_score - 0.02):
                    best_seq, best_exact_score = update_best_exact(a, best_seq, best_exact_score)

        if not evals_a:
            continue

        order = np.argsort(np.asarray(evals_s, dtype=np.float64))
        for o in order[:3]:
            if time.monotonic() >= deadline:
                break
            sub_dead = min(deadline - 1.0, time.monotonic() + 10.0)
            res_a, res_s = multiplicative_optimize(evals_a[o], sub_dead, rng)
            if res_a is not None and np.isfinite(res_s):
                old_fft = best_fft_score
                best_fft_seq, best_fft_score = update_fft_best(res_a, res_s, best_fft_seq, best_fft_score)
                if best_fft_score != old_fft and (best_seq is None or best_fft_score < best_exact_score - 0.02):
                    best_seq, best_exact_score = update_best_exact(res_a, best_seq, best_exact_score)

    return best_seq, best_exact_score, best_fft_seq, best_fft_score


def suggest_fallback():
    return normalize_nonnegative(np.ones(4, dtype=np.float64))


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main entry point. Returns the best sequence found before budget expires."""
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1800.0
    if budget <= 30.0:
        budget = 30.0

    start_time = time.monotonic()
    deadline = start_time + max(30.0, budget - 25.0)
    rng = np.random.default_rng(int(seed))

    best_seq = None
    best_exact_score = np.inf
    best_fft_seq = None
    best_fft_score = np.inf

    # Seed from the supplied state-of-the-art construction if present.
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a = normalize_nonnegative(raw)
        if a is not None:
            best_fft_seq, best_fft_score = update_fft_best(a, fft_score(a), best_fft_seq, best_fft_score)
            best_seq, best_exact_score = update_best_exact(a, best_seq, best_exact_score)
    except Exception:
        pass

    # Guaranteed simple candidates.
    for cand in (
        np.array([1.0 / 3.0, 2.0 / 3.0], dtype=np.float64),
        np.array([3.0 / 15.0, 4.0 / 15.0, 8.0 / 15.0], dtype=np.float64),
        np.ones(4, dtype=np.float64) / 4.0,
    ):
        a = normalize_nonnegative(cand)
        if a is None:
            continue
        sc = fft_score(a)
        best_fft_seq, best_fft_score = update_fft_best(a, sc, best_fft_seq, best_fft_score)
        best_seq, best_exact_score = update_best_exact(a, best_seq, best_exact_score)

    # 1) Small-length search.
    t_small = min(deadline, start_time + 90.0)
    best_seq, best_exact_score, best_fft_seq, best_fft_score = search_small(
        t_small, rng, best_seq, best_exact_score, best_fft_seq, best_fft_score
    )

    # 2) Continuous-shape search.
    t_shape = min(deadline, max(t_small, start_time + 230.0))
    if t_shape > time.monotonic() + 12.0:
        theta_list, powers_shape = shape_search(1024, t_shape, rng)
        if theta_list:
            for th, _sc in theta_list[:6]:
                a = theta_to_a(th, powers_shape)
                if a is None:
                    continue
                sc = fft_score(a)
                best_fft_seq, best_fft_score = update_fft_best(a, sc, best_fft_seq, best_fft_score)
                if best_seq is None or best_fft_score < best_exact_score - 0.02:
                    best_seq, best_exact_score = update_best_exact(a, best_seq, best_exact_score)
    else:
        theta_list = [np.zeros(5, dtype=np.float64)]

    # 3) Discrete optimization across several lengths.
    preferred = []
    parent_n = 0
    try:
        arr = np.asarray(height_sequence_1)
        if arr.size > 0:
            parent_n = int(arr.size)
    except Exception:
        parent_n = 0

    if 16 <= parent_n <= 20000:
        preferred.append(parent_n)

    for L in (4096, 6121, 2048, 5, 8, 8192, 1024, 6, 4, 512, 16):
        if L not in preferred:
            preferred.append(L)

    # Build a modest queue of diverse optimization attempts.
    queue = []
    max_queue = 26
    for i in range(max_queue):
        L = preferred[i % len(preferred)]
        if L <= 0:
            continue
        starts = make_starts_for_length(
            L, theta_list, rng, best_fft_seq, best_fft_score, max_starts=1
        )
        if starts:
            queue.append((L, starts[0]))

    qpos = 0
    while qpos < len(queue) and time.monotonic() < deadline - 12.0:
        L, a0 = queue[qpos]
        a0 = normalize_nonnegative(a0)
        if a0 is None:
            qpos += 1
            continue

        remaining = max(1.0, deadline - time.monotonic() - 12.0)
        left = max(1, len(queue) - qpos)
        if L < 160:
            run_time = min(10.0, remaining / 2.0)
        else:
            run_time = min(95.0, remaining / 4.0)
        run_time = max(4.0, run_time)

        sub_dead = time.monotonic() + run_time
        res_a, res_s = multiplicative_optimize(a0, sub_dead, rng)

        if res_a is not None and np.isfinite(res_s):
            old_fft = best_fft_score
            best_fft_seq, best_fft_score = update_fft_best(res_a, res_s, best_fft_seq, best_fft_score)
            if best_fft_score != old_fft:
                if best_seq is None or best_fft_score < best_exact_score - 0.02:
                    best_seq, best_exact_score = update_best_exact(res_a, best_seq, best_exact_score)

        qpos += 1

    # Final authoritative evaluation of the best fast-search candidate.
    if best_fft_seq is not None:
        best_seq, best_exact_score = update_best_exact(best_fft_seq, best_seq, best_exact_score)

    # Safety fallback.
    if best_seq is None:
        best_seq = suggest_fallback()
        best_exact_score = evaluate_sequence([float(x) for x in best_seq])

    # Ensure a valid Python list of floats.
    out = []
    for x in best_seq:
        try:
            y = float(x)
        except Exception:
            y = 0.0
        if not math.isfinite(y):
            y = 0.0
        if y < 0.0:
            y = 0.0
        if y > 1000.0:
            y = 1000.0
        out.append(y)

    if not out or sum(out) <= 0.01:
        fallback = suggest_fallback()
        out = [float(max(0.0, x)) for x in fallback]
        if sum(out) <= 0.01:
            out = [1.0, 1.0, 1.0, 1.0]

    return out
