def compute_grad_fft(a, b, beta):
    """Compute softmax-weighted gradient of max autocorrelation."""
    n = len(a)
    # Softmax weights on b (length 2n-1)
    b_shift = b - b.max()
    exp_b = np.exp(beta * b_shift)
    weights = exp_b / exp_b.sum()
    
    # grad[i] = 2 * sum_k weights[k] * a[k-i] (a[j]=0 outside [0,n-1])

def _softmax_grad(a, b, beta):
    """Compute gradient of soft-max of autocorrelation w.r.t. a via FFT."""
    n = len(a)
    b_shift = b - b.max()
    ew = np.exp(beta * b_shift)
    w = ew / ew.sum()
    # grad[i] = 2 * sum_k w[k] * a[k-i], a[j]=0 outside [0,n-1]
    # = 2 * convolve(a, w_rev, full)[2n-2-i] for i=0..n-1
    w_rev = w[::-1]
    nfft = 1
    need = n + (2 * n - 1) - 1  # 3n-2
    while nfft < need:

    A = np.fft.rfft(a, nfft)
    W = np.fft.rfft(w_rev, nfft)
    conv = np.fft.irfft(A * W, nfft)
    return 2.0 * conv[n - 1:2 * n - 1][::-1]
