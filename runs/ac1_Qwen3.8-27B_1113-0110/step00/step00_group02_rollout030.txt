import numpy as np
from scipy.optimize import linprog
import time

def solve_lp(a, M):
    """Solve the LP: max sum(g) s.t. g >= 0, |conv(a,g)| <= M"""
    n = len(a)
    # Variables: g_0, ..., g_{n-1}
    # Objective: minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: A_ub @ g <= b_ub
    # For each k = 0, ..., 2n-2:
    #   conv(a,g)[k] = sum_i a[i] * g[k-i] <= M
    #   -conv(a,g)[k] <= M (i.e., conv(a,g)[k] >= -M)
    # But since a >= 0 and g >= 0, conv(a,g) >= 0 always, so we only need:
    #   conv(a,g)[k] <= M
    
    # Build A_ub: for each k, the row is the convolution mask
    # Row k: (A_ub[k, j]) = a[j] if k-j is in [0, n-1], i.e., j in [k-(n-1), k]
    #        = a[j] * 1[0 <= k-j <= n-1]
    #        = a[k-j] if j in [max(0, k-(n-1)), min(n-1, k)]
    
    # Actually, let me think again.
    # conv(a, g)[k] = sum_{i=0}^{n-1} a[i] * g[k-i] where g[j] = 0 for j not in [0,n-1]
    # = sum_{j=0}^{n-1} g[j] * a[k-j] where a[m] = 0 for m not in [0,n-1]
    # So (A_ub[k, j]) = a[k-j] if 0 <= k-j <= n-1, else 0
    
    # This is a banded matrix!
    # I'll construct it as a dense matrix for simplicity (for n up to ~10000)
    # A_ub has 2n-1 rows and n columns
    
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(max(0, k-(n-1)), min(n, k+1)):
            A_ub[k, j] = a[k-j]
    
    A_ub = A_ub  # only upper bound constraints (lower bound is automatic)
    b_ub = np.full(2*n-1, M)
    
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return res

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _compute_score(a):
    """Compute the evaluation score for a non-negative sequence."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    conv = np.convolve(a, a)
    return 2.0 * n * float(conv.max()) / (s * s)


def _lp_direction(a, M, n_active=12):
    """
    Find LP direction: max sum(g) s.t. g >= 0, conv(a,g)[k] <= M
    for the top n_active positions k of conv(a,a).
    Returns optimal g vector or None.
    """
    n = len(a)
    conv = np.convolve(a, a)
    # Get top n_active indices
    if n_active < len(conv):
        idx = np.argpartition(conv, -n_active)[-n_active:]
    else:
        idx = np.arange(len(conv))

    rows = []
    cols = []
    vals = []
    for i, k in enumerate(idx):
        ki = int(k)
        j0 = max(0, ki - n + 1)
        j1 = min(n - 1, ki)
        for j in range(j0, j1 + 1):
            rows.append(i)
            cols.append(j)
            vals.append(float(a[ki - j]))

    if not vals:
        return None

    A = csr_matrix((vals, (rows, cols)), shape=(len(idx), n))
    b = np.full(len(idx), M)
    c = -np.ones(n)
    try:
        r = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                    method='highs', options={'time_limit': 5, 'presolve': True})
        if r.success and r.x is not None:
            return np.asarray(r.x, dtype=np.float64)
    except Exception:
        pass
    return None


def _grad_step(a, beta):
    """Compute gradient of softmax-smoothed max of conv(a,a)."""
    n = len(a)
    conv = np.convolve(a, a)
    mx = conv.max()
    w = np.exp(beta * (conv - mx))
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        if k < n:
            grad[:k + 1] += 2.0 * w[k] * a[k::-1]
        else:
            grad[k - n + 1:] += 2.0 * w[k] * a[n - 1:k - n:-1]
    return grad


def _gradient_polish(a, time_limit_s):
    """Polish using adaptive gradient descent on smoothed objective."""
    deadline = time.monotonic() + time_limit_s
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    beta = 15.0
    lr = 2.0 / n
    best_local = a.copy()
    best_score = _compute_score(a)
    ni = 0

    while time.monotonic() < deadline:
        gr = _grad_step(a, beta)
        if np.all(gr == 0):
            break
        a2 = np.clip(a - lr * gr, 0.0, None)
        s2 = a2.sum()
        if s2 < 1e-15:
            ni += 1
            lr *= 0.99
            continue
        a2 /= s2
        sc2 = _compute_score(a2)
        if sc2 < best_score:
            best_score = sc2
            best_local = a2.copy()
            a = a2
            ni = 0
            beta = min(beta * 1.005, 150.0)
            lr = lr * 1.001
        else:
            ni += 1
            lr *= 0.998
            beta = max(beta * 0.995, 5.0)
            if ni > 5000:
                break
    return best_local, best_score
