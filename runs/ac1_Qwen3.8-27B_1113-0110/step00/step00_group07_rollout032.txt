import numpy as np
import time

def fast_convolve(a, b):
    """Compute full linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    fb = np.fft.rfft(b, fft_size)
    return np.fft.irfft(fa * fb, fft_size)[:size]

def compute_subgradient(a, k_star, n):
    """Compute subgradient of (a*a)[k_star] w.r.t. a."""
    grad = np.zeros(n)
    # grad[i] = 2 * a[k_star - i] if 0 <= k_star - i < n, else 0

        indices = k_star - np.arange(i_min, i_max + 1)
        grad[i_min:i_max + 1] = 2.0 * a[indices]
    return grad
