import numpy as np

def _compute_smoothed_grad(a, t):
    """
    Compute gradient of smoothed max of conv(a,a) wrt a.
    Returns (gradient, max_value).
    """
    n = len(a)
    # Convolution
    c = np.convolve(a, a)  # length 2n-1
    # Softmax weights
    tc = t * c
    tc -= tc.max()
    e = np.exp(tc)
    w = e / e.sum()
    # Cross-correlation: (a ⊛ w)_j = sum_{m=0}^{n-1} a_m * w_{m+j}
    # = (reverse(a) * w)_{j + n - 1}
    rev_a = a[::-1]
    cc = np.convolve(rev_a, w)  # length 3n-2
    cross = cc[n-1:2*n-1]  # (a ⊛ w)_0, ..., (a ⊛ w)_{n-1}
    # w at even indices: w_0, w_2, ..., w_{2n-2}
    w_even = w[::2]  # length n (since w has length 2n-1, even indices are 0,2,...,2n-2)
    g = 2.0 * cross - a * w_even
    return g, c.max()
