def compute_conv(a):
    """Linear convolution a*a via FFT."""
    n = len(a)
    L = 1
    while L < 2 * n - 1:
        L *= 2
    A = np.fft.rfft(a, L)
    B = np.fft.irfft(A * A, L)[:2 * n - 1]

def simplex_project(x):

    n = len(x)
    s = np.sort(x)[::-1]
    cv = np.cumsum(s)
    rho = np.arange(1, n + 1)
    cond = (s - (cv - 1.0) / rho) > 0
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    i = idx[-1]
    theta = cv[i] / (i + 1.0)

def smooth_grad(a, tau):
    """Gradient of 2n*logsumexp(conv(a,a)/tau)."""

def lp_mix_step(a, M, max_iter_lp=200):
    """LP-based direction: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    # Constraint matrix: T[k, i] = a[k-i] if 0 <= k-i < n
    # A_ub has shape (2n-1, n)
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            A_ub[k, i] = a[k - i]
    # Objective: minimize -sum(b)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs',
                  options={'time_limit': 60, 'presolve': True})
    if not res.success:
        return None
    b = res.x.copy()
    if b.sum() <= 0:
        return None
    b /= b.sum()
    return b
