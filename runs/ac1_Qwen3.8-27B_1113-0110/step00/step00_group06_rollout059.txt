```
import numpy as np

def _pow2(x):
    if x <= 1: return 1

def _fft_square(a):
    n = int(a.size)
    L = 2*n - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, n=N)
    b = np.fft.irfft(fa * fa, n=N).real[:L]
    # clean
    if b.size and b.has? np.min(b)<0: b=np.maximum(b,0)
```

```python
"""
Optimize a non-negative step function sequence to minimize
    2 * n * max(convolve(a, a)) / (sum(a))**2.

The main search uses a structured anti-palindromic ansatz:
    a_i + a_{n-1-i} = 2/n,
parameterized by x in [0,1]^{n/2}.  This enforces a zero center
convolution coefficient and exposes low-dimensional variables.  In addition,
a general full-vector exponentiated-gradient optimizer is used for refinement
and diversity.

Iterative updates use softmax-type weights placed on large convolution
coefficients, FFT convolution, and FFT-based gradients.  Greedy construction
gives good binary initial anti-palindromic supports.
"""

import numpy as np
import time


def _pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _fft_square(a):
    n = int(a.size)
    L = 2 * n - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, n=N)
    b = np.fft.irfft(fa * fa, n=N).real[:L]
    b = np.maximum(b, 0.0)
    return b


def _fft_grad(weights, a):
    n = int(a.size)
    L = int(weights.size)
    N = _pow2(L + n - 1)
    fw = np.fft.rfft(weights, n=N)
    fa = np.fft.rfft(a, n=N)
    c = np.fft.irfft(fw * np.conj(fa), n=N)[:n]
    return 2.0 * c


def _internal_score(a, b=None):
    if b is None:
        b = _fft_square(a)
    s = float(a.sum())
    if s <= 1e-300:
        return float("inf"), b, 0.0
    maxb = float(b.max())
    score = float(2.0 * a.size * maxb / (s * s))
    return score, b, maxb


def _make_weights(b, focus, eps=0.02):
    L = int(b.size)
    maxb = float(b.max())
    if maxb <= 1e-300:
        return np.full(L, 1.0 / L, dtype=np.float64), maxb
    with np.errstate(over="ignore", under="ignore"):
        z = (maxb - b) / maxb
        p = np.exp(-z * float(focus))
    s = float(p.sum())
    if s <= 1e-12:
        p = np.zeros(L, dtype=np.float64)
        p[int(np.argmax(b))] = 1.0
    else:
        p = (1.0 - eps) * (p / s) + eps / L
    return p, maxb


def _random_positive(n, rng, amp=3.0):
    n = int(n)
    z = rng.normal(0.0, amp, size=n)
    z -= z.max()
    a = np.exp(z)
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    return a


def _anti_log_init(n, rng, amp=3.0, smooth=False):
    n = int(n)
    m = n // 2
    if m < 1:
        return np.ones(n, dtype=np.float64) / max(1, n)
    x = np.arange(m, dtype=np.float64)
    if smooth:
        v = np.zeros(m, dtype=np.float64)
        for f in (1, 3, 5, 7, 11):
            v += (rng.normal(0.0, 1.0) / max(1, f)) * np.sin(
                2.0 * np.pi * f * x / m + rng.uniform(0.0, 2.0 * np.pi)
            )
        v += 0.1 * rng.normal(0.0, 1.0, m)
    else:
        v = rng.normal(0.0, 1.0, m)
    v = v - float(v.mean())
    std = float(v.std())
    if std > 1e-12:
        v = (float(amp) * v) / std

    z = np.zeros(n, dtype=np.float64)
    z[:m] = v
    if n % 2 == 0:
        z[m:] = -v[::-1]
    else:
        z[m + 1:] = -v[::-1]
        z[m] = 0.0

    z = z - z.max()
    a = np.exp(z)
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    return a


def _greedy_support(n, rng, shuffle=True):
    n = int(n)
    m = n // 2
    counts = np.zeros(2 * n - 1, dtype=np.int64)
    selected = np.empty(m, dtype=np.int64)
    order = np.arange(m, dtype=np.int64)
    if shuffle:
        rng.shuffle(order)

    t = 0
    for idx_val in order:
        idx = int(idx_val)
        l = idx
        r = n - 1 - idx
        if t == 0:
            y = l
        else:
            s = selected[:t]
            affected_l = int(np.max(counts[s + l]))
            self_l = int(counts[2 * l])
            metric_l = max(affected_l + 2, self_l + 1)

            affected_r = int(np.max(counts[s + r]))
            self_r = int(counts[2 * r])
            metric_r = max(affected_r + 2, self_r + 1)

            y = l if metric_l <= metric_r else r

        if t > 0:
            counts[selected[:t] + y] += 2
        counts[2 * y] += 1
        selected[t] = y
        t += 1

    return selected


def _greedy_x(n, rng, shuffle=True):
    sel = _greedy_support(n, rng, shuffle=shuffle)
    m = n // 2
    x = np.full(m, 1e-6, dtype=np.float64)
    left = sel < m
    if np.any(left):
        x[sel[left]] = 1.0 - 1e-6
    return x


def _a_from_x(x, n=None):
    x = np.asarray(x, dtype=np.float64).ravel()
    m = x.size
    if n is None:
        n = 2 * m
    t = 1.0 / m
    a = np.empty(n, dtype=np.float64)
    a[:m] = t * x
    a[m:] = t * (1.0 - x[::-1])
    return a


def _optimize_pair(x0, time_limit, rng, verbose=False, focus_start=5.0,
                   max_focus=800.0, label=""):
    x = np.asarray(x0, dtype=np.float64).copy()
    m = int(x.size)
    n = 2 * m
    x_eps = 1e-8
    x = np.clip(x, x_eps, 1.0 - x_eps)

    a = _a_from_x(x, n)
    b = None
    score = float("inf")

    best_score = float("inf")
    best_x = x.copy()

    focus = float(focus_start)
    focus_inc = 1.02
    eta = 0.3
    eta_min = 0.02
    eta_max = 8.0

    no_improve = 0
    max_no = 1200
    resets = 0
    max_resets = 25
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < float(time_limit):
        if b is None:
            b = _fft_square(a)
            score, _, _ = _internal_score(a, b)

        if score < best_score - 1e-12:
            best_score = score
            best_x = x.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and (it % 1000 == 0):
            print(
                f"[{label}] pair it={it} t={time.monotonic()-t0:.1f}s "
                f"score={score:.6f} best={best_score:.6f} "
                f"focus={focus:.2f} eta={eta:.3f} resets={resets}"
            )

        if no_improve > max_no:
            resets += 1
            if resets > max_resets:
                break
            no_improve = 0
            max_no = int(max_no * 1.3)
            x = 0.7 * best_x + 0.3 * rng.normal(0.5, 0.25, size=m)
            x = np.clip(x, x_eps, 1.0 - x_eps)
            a = _a_from_x(x, n)
            b = None
            score = float("inf")
            focus = float(rng.choice([2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0]))
            eta = 0.3
            continue

        p, _ = _make_weights(b, focus)
        ga = _fft_grad(p, a)
        gx = ga[:m] - ga[m:][::-1]
        maxg = float(np.max(np.abs(gx)))
        if maxg < 1e-18:
            break

        gx_norm = gx / maxg
        x_new = x - eta * gx_norm
        x_new = np.clip(x_new, x_eps, 1.0 - x_eps)

        if not np.any(x_new != x):
            x_new = np.clip(x + rng.normal(0.0, 0.05, size=m), x_eps, 1.0 - x_eps)

        a_new = _a_from_x(x_new, n)
        b_new = _fft_square(a_new)
        score_new, _, _ = _internal_score(a_new, b_new)

        wt_old = float(np.dot(p, b))
        wt_new = float(np.dot(p, b_new))

        accept = False
        if score_new < best_score - 1e-12:
            accept = True
        elif score_new < score - 1e-12:
            accept = True
        elif score_new < score * 1.001 and wt_new < wt_old - 1e-10:
            accept = True

        if accept:
            x = x_new
            a = a_new
            b = b_new
            score = score_new
            eta = min(eta * 1.05, eta_max)
        else:
            eta = max(eta * 0.75, eta_min)
            if eta < 0.05:
                x = np.clip(x + rng.normal(0.0, 0.05, size=m), x_eps, 1.0 - x_eps)
                a = _a_from_x(x, n)
                b = None
                score = float("inf")

        focus = min(focus * focus_inc, float(max_focus))
        it += 1

    return _a_from_x(best_x, n), float(best_score)


def _optimize_from_a(a0, time_limit, rng, verbose=False, focus_start=5.0,
                     max_focus=800.0, label=""):
    a = np.asarray(a0, dtype=np.float64).copy()
    n = int(a.size)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-300:
        a = np.ones(n, dtype=np.float64) / max(1, n)
    else:
        a /= s
    a = np.maximum(a, 1e-15)
    a /= a.sum()

    b = None
    score = float("inf")

    best_score = float("inf")
    best_a = a.copy()

    focus = float(focus_start)
    focus_inc = 1.02
    eta = 0.3
    eta_min = 0.02
    eta_max = 8.0

    no_improve = 0
    max_no = 1200
    resets = 0
    max_resets = 25
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < float(time_limit):
        if b is None:
            b = _fft_square(a)
            score, _, _ = _internal_score(a, b)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and (it % 1000 == 0):
            print(
                f"[{label}] full it={it} t={time.monotonic()-t0:.1f}s "
                f"score={score:.6f} best={best_score:.6f} "
                f"focus={focus:.2f} eta={eta:.3f} resets={resets}"
            )

        if no_improve > max_no:
            resets += 1
            if resets > max_resets:
                break
            no_improve = 0
            max_no = int(max_no * 1.3)
            base = best_a.copy()
            if resets % 2 == 0:
                if n % 2 == 0:
                    fresh = _anti_log_init(n, rng, amp=3.0, smooth=False)
                else:
                    fresh = _random_positive(n, rng, amp=3.0)
                a = 0.8 * base + 0.2 * fresh
            else:
                a = base * np.exp(rng.normal(0.0, 0.3, size=n))
            a = np.maximum(a, 1e-15)
            a /= a.sum()
            b = None
            score = float("inf")
            focus = float(rng.choice([2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0]))
            eta = 0.3
            continue

        p, _ = _make_weights(b, focus)
        grad = _fft_grad(p, a)
        gd = grad - float(np.dot(a, grad))
        maxg = float(np.max(np.abs(gd)))
        if maxg < 1e-18:
            break

        gd = gd / maxg
        a_new = a * np.exp(-eta * gd)
        a_new = np.maximum(a_new, 1e-300)
        ss = float(a_new.sum())
        if ss <= 1e-300:
            break
        a_new /= ss

        b_new = _fft_square(a_new)
        score_new, _, _ = _internal_score(a_new, b_new)

        wt_old = float(np.dot(p, b))
        wt_new = float(np.dot(p, b_new))

        accept = False
        if score_new < best_score - 1e-12:
            accept = True
        elif score_new < score - 1e-12:
            accept = True
        elif score_new < score * 1.001 and wt_new < wt_old - 1e-10:
            accept = True

        if accept:
            a = a_new
            b = b_new
            score = score_new
            eta = min(eta * 1.05, eta_max)
        else:
            eta = max(eta * 0.75, eta_min)
            if eta < 0.05:
                a = a * np.exp(rng.normal(0.0, 0.05, size=n))
                a = np.maximum(a, 1e-15)
                a /= a.sum()
                b = None
                score = float("inf")

        focus = min(focus * focus_inc, float(max_focus))
        it += 1

    return best_a, float(best_score)


def _safe_evaluate(seq):
    ev = globals().get("evaluate_sequence")
    if ev is None or not callable(ev):
        return float("inf")
    try:
        return float(ev(seq))
    except Exception:
        return float("inf")


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    seed = int(seed)
    total = max(60.0, float(budget_s))

    now0 = time.monotonic()
    final_deadline = now0 + total - 30.0
    task_deadline = now0 + max(60.0, total - 240.0)

    print("starting step-function convolution optimizer")
    best_a = None
    best_score = float("inf")

    given_a = None
    try:
        given = globals().get("height_sequence_1")
        if given is not None:
            g = np.asarray(given, dtype=np.float64).ravel()
            g = np.nan_to_num(g, nan=0.0, posinf=1000.0, neginf=0.0)
            g = np.clip(g, 0.0, 1000.0)
            s = float(g.sum())
            if s > 0.01 and g.size > 10:
                gn = g / s
                sc, _, _ = _internal_score(gn)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = gn.copy()
                if g.size <= 12000:
                    given_a = gn
                print(f"given internal score: {sc:.6f}, n={gn.size}")
    except Exception:
        given_a = None

    tasks = []
    if given_a is not None:
        tasks.append((given_a.size, "given", 0, 90.0))

    tasks.append((4096, "pair", 0, 120.0))
    tasks.append((4096, "pair", 1, 120.0))
    tasks.append((8192, "pair", 0, 360.0))
    tasks.append((8192, "pair", 1, 300.0))
    tasks.append((8192, "pair", 2, 300.0))
    tasks.append((8192, "anti", 0, 180.0))
    tasks.append((8192, "anti", 1, 180.0))

    for idx, (n, kind, task_seed, max_alloc) in enumerate(tasks):
        now = time.monotonic()
        if now >= task_deadline:
            break
        alloc = min(max_alloc, task_deadline - now)
        if alloc < 25.0:
            break

        ng_seed = (seed * 1000003 + idx * 7919 + task_seed * 131) % (2**32)
        task_rng = np.random.default_rng(ng_seed)

        print(
            f"start task idx={idx} kind={kind} n={n} seed={task_seed} "
            f"alloc={alloc:.1f}s best={best_score:.6f}"
        )
        t0 = time.monotonic()
        cand_a = None
        cand_score = float("inf")

        try:
            if kind == "pair":
                shuffle = bool(task_seed != 0)
                x0 = _greedy_x(n, task_rng, shuffle=shuffle)
                cand_a, cand_score = _optimize_pair(
                    x0, alloc, task_rng,
                    verbose=(n == 8192 and task_seed == 0),
                    focus_start=5.0,
                    label=f"pair{n}-{task_seed}",
                )
            elif kind == "anti":
                amp = 3.0 + 0.5 * float(task_seed)
                a0 = _anti_log_init(n, task_rng, amp=amp, smooth=(task_seed % 2 == 0))
                cand_a, cand_score = _optimize_from_a(
                    a0, alloc, task_rng,
                    verbose=False,
                    focus_start=4.0,
                    label=f"anti{n}-{task_seed}",
                )
            elif kind == "given":
                a0 = given_a.copy() if given_a is not None else _random_positive(n, task_rng)
                cand_a, cand_score = _optimize_from_a(
                    a0, alloc, task_rng,
                    verbose=False,
                    focus_start=4.0,
                    label="given",
                )
            else:
                a0 = _anti_log_init(n, task_rng, amp=3.0, smooth=False)
                cand_a, cand_score = _optimize_from_a(
                    a0, alloc, task_rng,
                    verbose=False,
                    focus_start=4.0,
                    label="misc",
                )

            if cand_a is not None:
                cand_a = np.nan_to_num(cand_a, nan=0.0, posinf=0.0, neginf=0.0)
                cand_a = np.clip(cand_a, 0.0, 1000.0)
                cc = float(cand_a.sum())
                if cc > 0.01:
                    cand_a /= cc
                    internal, _, _ = _internal_score(cand_a)
                    if np.isfinite(internal) and internal < cand_score:
                        cand_score = internal
                    if np.isfinite(cand_score) and cand_score < best_score:
                        best_score = cand_score
                        best_a = cand_a.copy()
        except Exception as exc:
            print("task failed:", repr(exc))

        print(
            f"end task idx={idx} kind={kind} n={n} used={time.monotonic()-t0:.1f}s "
            f"candidate={cand_score:.6f} best={best_score:.6f}"
        )

        if best_score <= 1.32:
            print("strong score found, moving to final refinement")
            break

    if best_a is not None:
        remaining = final_deadline - time.monotonic()
        if remaining > 70.0:
            refine_time = min(170.0, remaining - 50.0)
            if refine_time > 20.0:
                task_rng = np.random.default_rng((seed + 999999) % (2**32))
                print(f"final refine time={refine_time:.1f}s")
                refined_a, refined_score = _optimize_from_a(
                    best_a, refine_time, task_rng,
                    verbose=True,
                    focus_start=10.0,
                    label="final",
                )
                if refined_a is not None and np.isfinite(refined_score):
                    best_a = refined_a
                    best_score = min(best_score, refined_score)

    if best_a is None:
        best_a = np.ones(1024, dtype=np.float64) / 1024.0

    best_a = np.nan_to_num(best_a, nan=0.0, posinf=1000.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    ss = float(best_a.sum())
    if ss <= 0.01:
        best_a = np.ones(1024, dtype=np.float64) / 1024.0
        ss = 1.0
    else:
        best_a /= ss

    final_seq = [float(x) for x in best_a]

    if best_a.size <= 12000 and time.monotonic() < final_deadline - 5.0:
        exact = _safe_evaluate(final_seq)
        if np.isfinite(exact):
            print(f"final internal score={best_score:.6f} exact score={exact:.6f} n={best_a.size}")
        else:
            print(f"final internal score={best_score:.6f} exact unavailable n={best_a.size}")
    else:
        print(f"final internal score={best_score:.6f} n={best_a.size}")

    return final_seq
```
