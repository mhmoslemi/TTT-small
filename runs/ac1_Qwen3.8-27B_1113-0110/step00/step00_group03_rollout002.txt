import numpy as np
import time

"""
Minimizes 2n*max(convolve(a,a))/sum(a)^2 using:
1. Parametric family sweep for good initialization
2. FFT-based gradient descent with momentum and adaptive step size
3. Periodic exact evaluation, restarts, and multi-peak gradient handling
"""


def fft_self_conv(a):
    """Compute full self-convolution using FFT for speed."""
    n = len(a)
    if n <= 1:
        return a * a
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:2 * n - 1]


def fast_score(a):
    """Fast approximate score (may differ from exact by tiny FFT errors)."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_self_conv(a)
    M = c.max()
    return 2.0 * n * M / (S * S)


def normalize_sum(a, target):
    """Normalize array to have sum=target, keeping non-negativity."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-10:
        return np.full_like(a, target / len(a))
    return a / S * target


def compute_gradient(a, c, k_stars):
    """
    Compute gradient of score w.r.t. a for given convolution c and
    one or more peak lags k_stars. Returns gradient vector of length n.
    Assumes sum(a) = n for simplification.
    """
    n = len(a)
    M = c.max()
    S = n  # we normalize to sum=n
    grad = np.zeros(n)
    for k in k_stars:
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        if j_max >= j_min:
            s_start = k - j_max
            s_end = k - j_min + 1
            grad[j_min:j_max + 1] += a[s_start:s_end][::-1]
    grad = grad / len(k_stars)
    # Full gradient: (4n/S^2)*a[k-j] - 4nM/S^3 = (4/n)*a[k-j] - 4M/n^2
    grad = (4.0 / n) * grad - (4.0 * M / (n * n))
    return grad
