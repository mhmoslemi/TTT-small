_adam_m = {}
_adam_v = {}
_adam_t = {}

def fft_auto_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(fa * fa, fft_size)[:size]
    return np.maximum(result, 0.0)

def subgradient_smooth(a, topk=10, temperature=None):
    """Compute a smoothed subgradient of max h_k."""
    n = len(a)
    h = fft_auto_convolve(a)
    
    if temperature is not None:
        # Softmax weighting
        h_shifted = h - h.max()
        weights = np.exp(topk * h_shifted)  # topk used as 1/T
        weights /= weights.sum()
        # Compute weighted subgradient
        g = np.zeros(n)
        # For efficiency, only use lags where weight is significant
        significant = np.where(weights > weights.max() * 0.01)[0]
        for ki in significant:
            ki = int(ki)
            lo = max(0, ki - n + 1)
            hi = min(n - 1, ki)
            i_range = np.arange(lo, hi + 1)
            g[i_range] += weights[ki] * 2.0 * a[ki - i_range]
        return g, h
    
    # Top-K approach
    if topk >= len(h):
        idx = np.arange(len(h))
    else:
        idx = np.argsort(h)[::-1][:topk]
    g = np.zeros(n)
    for ki in idx:
        ki = int(ki)
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        i_range = np.arange(lo, hi + 1)
        g[i_range] += 2.0 * a[ki - i_range]
    g /= len(idx)
    return g, h

def adam_update(a, g, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer step (additive), with projection to non-negative."""
    key = len(a)
    if key not in _adam_m:
        _adam_m[key] = np.zeros(len(a))
        _adam_v[key] = np.zeros(len(a))
        _adam_t[key] = 0
    _adam_t[key] += 1
    t = _adam_t[key]
    m = _adam_m[key]
    v = _adam_v[key]
    
    m = beta1 * m + (1 - beta1) * g
    v = beta2 * v + (1 - beta2) * g * g
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = np.maximum(a_new, 0.0)
    return a_new, m, v

Optimizes step function heights to minimize 2*n*max_conv/(sum_a)^2.
Uses FFT-accelerated smoothed subgradient (log-sum-exp) with Adam optimizer,
simulated annealing on temperature, and multi-length search.

import numpy as np
import time


def fft_auto_convolve(a):
    """Compute linear auto-convolution using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(fa * fa, fft_size)[:size]
    return np.maximum(result, 0.0)


def compute_objective_fast(a):
    """Fast objective computation using FFT convolution."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    h = fft_auto_convolve(a)
    return 2.0 * n * h.max() / (s * s)


def compute_smoothed_gradient_fft(a, w):
    """
    Compute g[i] = 2 * sum_k w[k] * a[k-i] using FFT.
    w has length 2n-1, a has length n. Returns g of length n.
    """
    n = len(a)
    w_rev = w[::-1]
    # conv(w_rev, a) has length (2n-1) + n - 1 = 3n - 2
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    fw = np.fft.rfft(w_rev, fft_size)
    fa = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(fw * fa, fft_size)
    # g[i] = 2 * result[2n-2-i] for i = 0,...,n-1
    g = 2.0 * result[n - 1:2 * n - 1][::-1]
    return g


def gradient_logsumexp(a, temperature):
    """
    Compute gradient of log-sumexp(T * h_k) where h = conv(a,a).
    Returns (gradient, h, max_h).
    """
    n = len(a)
    h = fft_auto_convolve(a)
    max_h = h.max()
    # Compute softmax weights with temperature
    log_w = temperature * (h - max_h)
    log_w -= log_w.max()
    w = np.exp(log_w)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(h))
        w[int(np.argmax(h))] = 1.0
    else:
        w /= w_sum
    g = compute_smoothed_gradient_fft(a, w)
    return g, h, max_h


class AdamState:
    def __init__(self, n):
        self.m = np.zeros(n)
        self.v = np.zeros(n)
        self.t = 0

    def step(self, g, lr, beta1=0.9, beta2=0.999, eps=1e-8):
        self.t += 1
        self.m = beta1 * self.m + (1 - beta1) * g
        self.v = beta2 * self.v + (1 - beta2) * g * g
        m_hat = self.m / (1 - beta1 ** self.t)
        v_hat = self.v / (1 - beta2 ** self.t)
        return lr * m_hat / (np.sqrt(v_hat) + eps)


def optimize_sequence(a0, deadline, seed=0):
    """
    Optimize sequence using Adam + log-sumexp smoothed gradient with
    temperature annealing. Returns best (a, obj) found.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n

    best_a = a.copy()
    best_obj = compute_objective_fast(a)

    adam = AdamState(n)
    lr = 0.01
    temp_high = 50.0
    temp_low = 200.0

    no_improve = 0
    patience = 300
    it = 0
    total_iters = 20000

    while time.monotonic() < deadline and it < total_iters:
        # Anneal temperature
        progress = it / total_iters
        temperature = temp_high + (temp_low - temp_high) * progress

        g, h, max_h = gradient_logsumexp(a, temperature)

        # Adam step
        update = adam.step(g, lr)
        a_new = a - update

        # Project: non-negative, sum=1
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-30:
            a_new = a_new / s
        else:
            a_new = a.copy()

        a = a_new
        it += 1

        # Evaluate periodically
        if it % 10 == 0 or it == total_iters - 1:
            obj = compute_objective_fast(a)
            if obj < best_obj - 1e-10:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 10

        # If stuck, perturb and restart Adam
        if no_improve >= patience:
            a = best_a * (1.0 + 0.05 * rng.standard_normal(n))
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0:
                a = a / s
            else:
                a = best_a.copy()
            adam = AdamState(n)
            no_improve = 0
            lr = max(0.001, lr * 0.5)

    return best_a, best_obj

        # Cycle through topk values

        if topk >= len(h):

        g = np.zeros(n)

            hi = min(n - 1, ki)

        n_best = len(best_seq)
        a_norm = best_seq / (best_seq.sum() + 1e-30)

        # Adam with lower temperature for fine-tuning
        sub_deadline = min(deadline, time.monotonic() + (budget_s - 15) * 0.15)
        a_opt, obj = optimize_sequence(a_norm, sub_deadline, seed=999)
        try_sequence(a_opt, f"Refine Adam n={n_best}")

        if time.monotonic() < deadline - 10:
            # Momentum refinement
            sub_deadline = min(deadline, time.monotonic() + (budget_s - 15) * 0.1)
            a_opt, obj = optimize_with_momentum(a_norm, sub_deadline, seed=999)
            try_sequence(a_opt, f"Refine Mom n={n_best}")

        print(f"  Phase4 time: {time.monotonic()-t0:.1f}s", flush=True)

    # Final verification
    final_score = float(evaluate_sequence([float(x) for x in best_seq]))
    print(f"FINAL: score={final_score:.6f}, n={len(best_seq)}", flush=True)

    return [float(x) for x in best_seq]
