```
def _proj_simplex(v, s):
    # map v (finite) to closest point (in L2) of {x >= 0, sum x = s}.
    n = len(v)
    if s <= 0: return zeros
    # reduce to unit sum, back-scale by s:
    v = v / s  # now sum >= 0 ... but may have negatives
    x = np.sort(v)[::-1]
    cssv = np.cumsum(x)
    cond = x > (cssv - 1.0)/np.arange(1, n+1)
    rho = int(np.argmax(cond))
    theta = (cssv[rho] - 1.0) / rho
    out = np.clip(v - theta, 0, None)
    out = out / np.sum(out)  # ensure sum 1
    return out * s
```

import time

# Low-ihheno (utility) helpers
# ===========================================================

def _nrr(n):
    # smallest power of two >= n
    if n <= 1:
        return 1
    r = 1
    while r < n:
        r <<= 1
    return r

def _normalize(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s <= 1e-14:
        return np.ones_like(a) / float(a.size)
    return a / s

def _convolve_fft(a):
    a = np.asarray(a, float64)
    n = int(a.shape[0])
    N = _nrr(2 * n - 1)
    F = np.fft.rfft(a, N)
    return np.fft.irfft(F * F, N)

def _score_loc(a):
    a = np.asarray(a, float64)
    n = int(a.shape[0])
    S = float(a.sum())
    if S <= 1e-12:
        return float('inf')
    b = _convolve_fft(a)
    bmax = float(b.max())
    if not np.isfinite(bmax):
        return float('inf')
    return float(2.0 * n * bmax / (S * S))

```python
def _gre_sidon(n, m):
    S = [0]
    mark = np.zeros(n, dtype=bool)
    mark[0] = True
    used = np.zeros(2 * n - 1, dtype=bool)
    used[0] = True
    for step in range(1, m):
        found = False
        for cand in range(1, n):
            if mark[cand]:
                continue
            # collisions?
            arr = np.asarray([cand + s for s in S])
            if np.any(used[arr]):
                continue
            if 2 * cand >= 2 * n - 1:
                continue
            if used[2 * cand]:
                continue
            S.append(cand)
            mark[cand] = True
            used[np.asarray([cand + s for s in S])] = True
            found = True
            break
        if not found:
            break
    return S
```

"""
Minimize F = 2*n*max(conv(a))/sum(a)^2 with non-negative a.
Main drivers: Sidon-tress-seed generation (greedy + root/tune weights),
with top-level-loop FFT-based score and projected-root local search with block/"chunk"
moves and evaluation-over-lag-gradient optimization. Final returned sequence
is non-negative, sums to 1, normalized.
"""

import numpy as np
import time

# ---- utilities ----

def _nrr(n):
    r = 1
    n = int(n)
    if n <= 1:
        return 1
    while r < n:
        r <<= 1
    return r

def _normalize(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s <= 1e-14:
        return np.ones_like(a, dtype=np.float64) / float(a.size)
    return a / s

def _convolve_fft(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.shape[0])
    N = _nrr(2*n - 1)
    F = np.fft.rfft(a, N)
    return np.fft.irfft(F * F, N)

def _score_a_full(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.shape[0])
    S = float(a.sum())
    if S <= 1e-12 or not np.isfinite(S):
        return float('inf'), None, None
    b = _convolve_fft(a)
    bmax = float(b.max())
    if not np.isfinite(bmax):
        return float('inf'), b, None
    F = float(2.0 * n * bmax / (S * S))
    thr = bmax * (1.0 - 1e-3)
    A = np.where(b >= thr)[0]
    if A.size > 64:
        A = A[0:64]
    return F, b, A

def _score_loc(a):
    F, _, _ = _score_a_full(a)
    return F

# ---- seed builders ----

def _gre_sidon(n, m, rng=None):
    S = [0]
    mark = np.zeros(n, dtype=bool)
    mark[0] = True
    used = np.zeros(2*n - 1, dtype=bool)
    used[0] = True
    for step in range(1, m):
        found = False
        for cand in range(1, n):
            if mark[cand]:
                continue
            # check collisions:
            # cand+s (for all s in S), 2*cand
            arr = np.asarray([cand + s for s in S], dtype=np.int64)
            if np.any(used[arr]):
                continue
            if (2*cand >= 2*n - 1):
                continue
            if used[2*cand]:
                continue
            # add
            S.append(cand)
            mark[cand] = True
            used[np.asarray([cand + s for s in S], dtype=np.int64)] = True
            found = True
            break
        if not found:
            break
    return S

def _weights_m(m, mode='equal', rng=None):
    if m <= 1:
        return np.array([1.0])
    if mode == 'soct':
        w = np.ones(m)
    elif mode == 'lin':
        idx = np.arange(m)
        w = (m - idx).astype(np.float64)
    elif mode == 'exp':
        idx = np.arange(m)
        w = np.exp(-(idx) / (m * 0.2))
    elif mode == 'rot':
        idx = np.arange(m)
        w = 0.5 * (1.0 + np.cos(2.0*np.pi*idx/m))
        w = np.clip(w + 1e-9, 0.0, None)
    elif mode == 'sinq':
        idx = np.arange(m)
        w = 0.5 * (1.0 + 0.6 * np.cos(2.0*np.pi*idx/m)) + 0.3 * np.cos(4.0*np.pi*idx/m)
        w = np.clip(w + 0.5, 0.0, None)
    else:
        w = np.ones(m)
    wsum = float(w.sum())
    if wsum <= 1e-12:
        w = np.ones(m)
        wsum = float(m)
    return w / wsum

def _build_seed(n, m, wmode, bg=0.0, rng=None):
    S = _gre_sidon(n, m, rng)
    S = np.asarray(S, dtype=np.int64)
    m_eff = int(S.size)
    w = _weights_m(m_eff, wmode, rng)
    J = np.zeros(n, dtype=np.float64)
    cnt = min(m_eff, w.size)
    J[S[0:cnt]] = w[0:cnt]
    if bg > 0.0 and bg < 1.0:
        J[:] += bg * m_eff
    J = _normalize(J)
    return J

# ---- local search moves ----

def _subg_step(a, active, eta):
    # a: (n,), active: array of lag indices, eta: step size
    n = int(a.shape[0])
    G = np.zeros(n, dtype=np.float64)
    for k in active:
        k = int(k)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        src = k - idx
        G[idx] += a[src]
    G *= 2.0
    a_new = a - eta * G
    a_new = np.clip(a_new, 0.0, None)
    a_new = _normalize(a_new)
    return a_new

def _block_mult(a, rng):
    n = int(a.shape[0])
    wmax = max(2, n // 8)
    wmin = 2
    width = int(np.clip(rng.integers(wmin, wmax + 1), wmin, n))
    if width >= n:
        width = n
    start = int(rng.integers(0, max(1, n - width + 1)))
    alpha = float(np.exp(rng.normal(0.0, 0.30)))
    m = a.copy()
    m[start:start + width] *= alpha
    return _normalize(m)

def _chunk_move(a, rng):
    n = int(a.shape[0])
    if n < 4:
        return a.copy()
    src = int(rng.integers(0, n))
    dst = int(rng.integers(0, n))
    if src == dst:
        return a.copy()
    amt = float(rng.uniform(0.01, 0.05)) * a[src]
    m = a.copy()
    m[src] = m[src] - amt
    m[dst] = m[dst] + amt
    m = np.clip(m, 0.0, None)
    return _normalize(m)

def _reflect_move(a, rng):
    n = int(a.shape[0])
    if n < 6:
        return a.copy()
    l = int(rng.integers(0, n // 2 + 1))
    r = int(rng.integers(n // 2 + 1, n))
    m = a.copy()
    m[l:r] = a[r-1:l-1:-1]
    return _normalize(m)

"""
Minimize F = 2*n*max(conv(a))/sum(a)^2 over non-negative sequences a.
Scale-invariant, so we work with sum(a)=1 and minimize 2*n*max(a*a).

Strategy:
  * Build structured seeds: "Sidon-like" (distinct pair-sum) equal-weight
    delta trains with tunable density m ~ c*sqrt(n), plus smooth baselines.
  * Score via FFT-based linear convolution (fast, no O(n^2) evaluator).
  * Improve with a projected sub-gradient/fdergy local search that
    targets the active (max) lags, combined with block-multiplier,
    mass-move, and reflect-type moves.
  * Run several (length, seed) trials within a 1800s budget and
    return the best valid sequence.
Top-level helpers only, no lambdas, no nested closures.
"""

import numpy as np
import time

# utilities

def _nrr(n):
    """Smallest power of two >= n."""
    n = int(n)
    r = 1
    if n <= 1:
        return 1
    while r < n:
        r <<= 1
    return r


def _normalize(a):
    """Clip to >=0 and rescale to sum exactly 1 (scale-invariance)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s <= 1e-14:
        return np.ones_like(a, dtype=np.float64) / float(a.size)
    return a / s


def _convolve_fft(a):
    """Full linear convolution a*a (a non-negative, real) via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.shape[0])
    N = _nrr(2 * n - 1)
    F = np.fft.rfft(a, N)
    return np.fft.irfft(F * F, N)


def _score_a_full(a):
    """Return (F, b, active_lags) where F=2n*max(b)/S^2; S=sum(a)."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.shape[0])
    S = float(a.sum())
    if S <= 1e-12 or not np.isfinite(S):
        return float("inf"), None, None
    b = _convolve_fft(a)
    bmax = float(b.max())
    if not np.isfinite(bmax):
        return float("inf"), b, None
    F = float(2.0 * n * bmax / (S * S))
    thr = bmax * (1.0 - 1e-3)
    A = np.where(b >= thr)[0]
    if A.size > 64:
        A = A[0:64]
    return F, b, A


def _score_loc(a):
    F, _, _ = _score_a_full(a)
    return F

def _gre_sidon(n, m):
    """Greedy distinct-pair-sum (Sidon) set on [0, n): S(0)+S(1)|= ... unique."""
    S = [0]
    mark = np.zeros(n, dtype=bool)
    mark[0] = True
    used = np.zeros(2 * n - 1, dtype=bool)
    used[0] = True
    for step in range(1, m):
        found = False
        # candidate new position
        for cand in range(1, n):
            if mark[cand]:
                continue
            if 2 * cand >= 2 * n - 1:
                continue
            if used[2 * cand]:
                continue
            # check against existing sums
            arr = np.asarray([cand + s for s in S], dtype=np.int64)
            if np.any(used[arr]):
                continue
            # accept
            S.append(cand)
            mark[cand] = True
            used[np.asarray([cand + s for s in S], dtype=np.int64)] = True
            found = True
            break
        if not found:
            break
    return S


def _weights_m(m, mode):
    """Non-negative weight vector summing to 1 at m positions."""
    if m <= 1:
        w = np.array([1.0])
        return w
    if mode == "eq":
        w = np.ones(m, dtype=np.float64)
    elif mode == "lin":
        idx = np.arange(m)
        w = (m - idx).astype(np.float64)
    elif mode == "rot":
        idx = np.arange(m)
        w = 0.5 * (1.0 + np.cos(2.0 * np.pi * idx / m))
        w = np.clip(w + 1e-9, 0.0, None)
    else:
        w = np.ones(m, dtype=np.float64)
    s = float(w.sum())
    if s <= 1e-14:
        w = np.ones(m, dtype=np.float64)
        s = float(m)
    return w / s


def _build_seed(n, m, wmode, bg_frac):
    s = _gre_sidon(n, m)
    n_eff = int(len(s))
    # if greedy failed to reach m positions, fall back to even-spread
    if n_eff < 2:
        step = max(1, n // m)
        s = list(range(0, n, step))[: m]
        s = [int(v) for v in s]
        n_eff = len(s)
    assert n_eff >= 1
    w = _weights_m(n_eff, wmode)
    w = w[: n_eff]
    J = np.zeros(n, dtype=np.float64)
    for i in range(n_eff):
        J[int(s[i])] = float(w[i])
    if bg_frac and bg_frac > 0:
        total_bg = bg_frac
        # keep total ~1: add uniform background small
        per = total_bg / n
        J[:] += per
    return _normalize(J)

def _subg_step(a, active, eta):
    """Projected sub-gradient step that reduces the active peaks' sum."""
    n = int(a.shape[0])
    G = np.zeros(n, dtype=np.float64)
    for k in active:
        kk = int(k)
        lo = max(0, kk - (n - 1))
        hi = min(n - 1, kk)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        src = kk - idx
        G[idx] += a[src]
    G *= 2.0
    a_new = a - eta * G
    a_new = np.clip(a_new, 0.0, None)
    return _normalize(a_new)


def _block_mult(a, rng):
    n = int(a.shape[0])
    if n < 4:
        return a.copy()
    lo_w = 2
    hi_w = max(3, n // 12)
    w = int(rng.integers(lo_w, hi_w + 1))
    if w >= n:
        w = n
    start = int(rng.integers(0, max(1, n - w + 1)))
    alpha = float(np.exp(rng.normal(0.0, 0.30)))
    m = a.copy()
    m[start: start + w] *= alpha
    return _normalize(m)


def _mass_move(a, rng):
    n = int(a.shape[0])
    if n < 4:
        return a.copy()
    src = int(rng.integers(0, n))
    dst = int(rng.integers(0, n))
    if src == dst:
        return a.copy()
    amt = float(rng.uniform(0.01, 0.05)) * a[src]
    m = a.copy()
    m[src] -= amt
    m[dst] += amt
    m = np.clip(m, 0.0, None)
    return _normalize(m)


def _reflect_move(a, rng):
    n = int(a.shape[0])
    if n < 8:
        return a.copy()
    l = int(rng.integers(0, n // 2 + 1))
    r = int(rng.integers(n // 2 + 1, n))
    if r - l < 2:
        # still fine
        pass
    m = a.copy()
    seg = m[l:r]
    m[l:r] = seg[::-1]
    return _normalize(m)


def _local_search(a0, budget_s, seed=0):
    a = _normalize(a0)
    best_a = a.copy()
    best_F = _score_loc(a)
    rng = np.random.default_rng(seed)
    t0 = time.time()
    t_end = t0 + max(0.1, float(budget_s))
    ic = 0
    while time.time() < t_end:
        ic += 1
        frac = (time.time() - t0) / max(1.0, (float(budget_s)))
        frac = min(1.0, max(0.0, frac))
        eta_base = 0.03 * (1.0 - frac) + 0.004
        r = rng.random()
        F, b, A = _score_a_full(a)
        if not np.isfinite(F) or A is None:
            F = best_F
            A = np.array([0])
        if r < 0.50:
            a_new = _subg_step(a, A, eta=eta_base)
        elif r < 0.72:
            a_new = _block_mult(a, rng)
        elif r < 0.86:
            a_new = _mass_move(a, rng)
        else:
            a_new = _reflect_move(a, rng)
        F_new = _score_loc(a_new)
        if np.isfinite(F_new) and F_new < best_F:
            best_F = F_new
            best_a = a_new
            a = a_new
    return best_a, best_F

    try:
        hs = np.asarray(height_sequence_1, dtype=np.float64)  # noqa: F821
        hs = np.clip(hs, 0.0, 1000.0)
        base_seq = _normalize(hs)
    except Exception:
        base_seq = None

    rng0 = np.random.default_rng(int(seed) ^ 0x9E37_79B9)

    # safe starting point
    n_s = 2048
    start = np.ones(n_s, dtype=np.float64) / n_s
    best_F = float(_score_loc(start))
    best_a = _normalize(start)

    # If provided sequence is already quite good, use it as a seed too.
    seeds = []
    if base_seq is not None:
        try:
            sF = _score_loc(base_seq)
        except Exception:
            sF = float("inf")
        if np.isfinite(sF) and sF < 2.05:
            seeds.append((base_seq, sF, "given"))

    # Create train seeds over (n, m density, wmode)
    # Structural hope: m ~ 1.6*sqrt(n) with equal weights gives ~ 1.5
    grid_ns = [64, 128, 256, 512, 1024, 2048, 4096, 8192]
    m_factors = [1.2, 1.4, 1.6]
    wmodes = ["eq", "lin"]
    for n_g in grid_ns:
        for f in m_factors:
            m_g = max(3, int(round(f * np.sqrt(n_g))))
            if m_g >= n_g:
                continue
            for wm in wmodes:
                seedJ = None
                try:
                    if time.time() < t_end and (t_end - time.time()) > 4:
                        seedJ = _build_seed(n_g, m_g, wm, bg_frac=0.0)
                except Exception:
                    seedJ = None
                if seedJ is None:
                    continue
                try:
                    Fv = _score_loc(seedJ)
                except Exception:
                    Fv = float("inf")
                if np.isfinite(Fv) and Fv < best_F:
                    best_F = Fv
                    best_a = seedJ
                    print("[search] improved seed to %.4f (n=%d, m=%d, wm=%s)" % (Fv, n_g, m_g, wm))
                elif np.isfinite(Fv) and Fv < 2.03 * (n_g / 255.0 + 0.5):
                    seeds.append((seedJ, Fv, "grid"))
                if time.time() > t0 + 0.55 * bgt:
                    break
        if time.time() > t0 + 0.55 * bgt:
            break

    # Ensure best_a is in the seeds list for local searches
    # (avoid duplicate local searches on same (n, F))
    uniq = []
    seen_key = set()
    # start with best seed (by best_F)
    # include the best_a itself
    uniq.append(("best", best_a, np.int64(best_a.shape[0])))

    # pick up to two more varied target lengths (to escape local minima)
    try:
        cur_n = int(best_a.shape[0])
    except Exception:
        cur_n = int(best_a.size)
    for (sd, sdF, tag) in seeds:
        if tag == "best":
            continue
        try:
            n_d = int(sd.shape[0])
        except Exception:
            continue
        key = n_d
        if key in seen_key:
            continue
        # prefer seeds not equal to current best_n
        if n_d == cur_n:
            # allow one duplicate-n but not too many
            dup = sum(1 for k in seen_key if k == cur_n)
            if dup >= 1:
                continue
        seen_key.add(key)
        uniq.append((tag + "_alt", _normalize(sd), n_d))
        if len(uniq) >= 3:
            break

    # Add a couple random-shapish seeds for robustness (different n)
    for extra_n in [512, 1024, 2048, 4096]:
        if extra_n == cur_n:
            continue
        if extra_n in seen_key:
            continue
        try:
            ax = np.zeros(extra_n, dtype=np.float64)
            # a smooth bump to explore "dense" structure
            x = np.linspace(0.0, 1.0, extra_n)
            ax = 0.6 * np.exp(-4.0 * (x - 0.35) ** 2) + 0.4 * np.exp(-4.0 * (x - 0.75) ** 2)
            ax = _normalize(ax)
        except Exception:
            continue
        seen_key.add(extra_n)
        uniq.append(("rnd_bump", ax, extra_n))
        if len(uniq) >= 5:
            break

    # allocate budget for local search runs
    rem = max(0.5, t_end - time.time())
    n_runs = len(uniq)
    # We will spend ~75% of remaining on local searches; remainder for final polish
    base_alloc = 0.70 * rem / max(1, n_runs)

    for (name, seed_arr, n_run) in uniq:
        # remaining guard
        if time.time() >= t_end - 2.0:
            break
        alloc = min(base_alloc, max(1.0, (t_end - time.time()) - 3.0))
        if alloc < 1.5:
            continue
        try:
            la, lf = _local_search(_normalize(seed_arr), alloc, seed=int(seed ^ 0x7B + n_run))
        except Exception:
            continue
        if np.isfinite(lf) and lf < best_F:
            best_F = lf
            best_a = la
            print("[local] %.4f from tag=%s n=%d (alloc=%.1fs)" % (lf, name, n_run, alloc))

    # Final polish (short) on the current best, if time left
    if time.time() < t_end - 6:
        pf = min(30.0, max(1.0, (t_end - time.time()) - 6.0))
        try:
            lp, lf = _local_search(best_a, pf, seed=int(seed) ^ 0x1555)
            if np.isfinite(lf) and lf < best_F:
                best_F = lf
                best_a = lp
                print("[polish] %.4f" % lf)
        except Exception:
            pass

    # Normalize and safety clip
    out = np.clip(best_a.astype(np.float64), 0.0, 1000.0)
    s = float(out.sum())
    if s < 0.011:
        out = _normalize(out) * 0.011
    out = np.clip(out, 0.0, 1000.0)
    # final validity gate with the real evaluator (we must be robust)
    try:
        g = float(evaluate_sequence(out.tolist()))  # noqa: F821
    except Exception:
        g = float("inf")
    if not np.isfinite(g):
        # fallback: evenly-spaced tiny train to keep valid
        out = _normalize(np.ones(512, dtype=np.float64))
    # unless our out gave inf, keep our best_out as the returned list
    ret = [float(x) for x in out]
    return ret
