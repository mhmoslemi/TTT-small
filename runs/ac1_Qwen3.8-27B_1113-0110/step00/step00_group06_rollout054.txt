import numpy as np
import time
import math

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    # Use the sorting-based O(n log n) algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
        # All v_i are below the threshold, project to uniform
        return np.ones(n) / n
    k = rho[-1]
    theta = (cssv[k] - 1.0) / (k + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_score(a):
    """Compute the objective value without calling evaluate_sequence (for speed)."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def subgradient_step(a, eta, rng):
    """One step of projected subgradient descent."""
    n = len(a)
    conv = np.convolve(a, a)
    k_star = int(np.argmax(conv))

    lo = max(0, k_star - n + 1)

    # are absorbed into the step size eta
    
    a_new = a - eta * grad
    a_new = project_simplex(a_new)
    return a_new, k_star, conv
