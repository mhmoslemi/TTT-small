def _fast_conv(a):
    """Linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def _project(a):
    """Project onto {x >= 0, sum(x) = 1}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        return a / s
    return np.ones_like(a) / len(a)

def _soft_grad(a, conv, topk, temp):
    """Weighted gradient of soft-max over top lags of conv."""
    n = len(a)
    if len(conv) > topk:
        idx = np.argpartition(conv, -topk)[-topk:]
    else:
        idx = np.arange(len(conv))
    vals = conv[idx]
    w = np.exp(np.clip((vals - vals.max()) / max(temp, 1e-18), -60, 0))
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(len(idx)) / len(idx)
    else:
        w /= ws
    g = np.zeros(n)
    for wi, ki in zip(w, idx):
        if wi < 1e-12:
            continue
        ki = int(ki)
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if lo > hi:
            continue
        j = np.arange(lo, hi + 1)
        g[j] += wi * 2.0 * a[ki - j]
    return g
