```python
import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft

def compute_conv(f):
    """Compute full linear convolution of f with itself using FFT."""
    n = len(f)
    size = 2 * n - 1
    # Next power of 2 for efficiency
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    conv = irfft(F * F, fft_size)[:size]
    return conv

def solve_lp(f, M, active_indices):
    """Solve max sum(g) s.t. (f*g)[k] <= M for k in active_indices, g >= 0."""
    n = len(f)
    # Build constraint matrix: A[k, i] = f[k-i] if 0 <= k-i <= n-1
    K = len(active_indices)
    A = np.zeros((K, n))
    for idx, k in enumerate(active_indices):
        # (f*g)[k] = sum_{i} f[k-i] * g[i] where 0 <= k-i <= n-1
        # i ranges from max(0, k-n+1) to min(k, n-1)
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        # A[idx, i] = f[k - i] for i in [i_start, i_end]
        js = k - np.arange(i_start, i_end + 1)  # js = k - i, ranges from k-i_end to k-i_start
        A[idx, i_start:i_end+1] = f[js]
    
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    b = np.full(K, M)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    
    if res.status != 0:
        return None
    return res.x
```
