For S = {0, 1, ..., n-1}, max pairs = n (at the middle), m = n, V = 2n*n/n^2 = 2.
For S = {0, 2, 4, ..., 2(n-1)} (even numbers), m = n, but the "convolution" is of length 4n. Hmm, this doesn't directly apply.

def build_conv_matrix(f):
    """Build the (2n-1) x n convolution matrix A where (A@x)[k] = convolve(f, x)[k]."""
    n = len(f)
    # A[k, j] = f[k-j] if 0 <= k-j < n else 0
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = coo_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A.tocsr()

def build_conv_matrix(f):
    n = len(f)
    # For row k, non-zero cols are max(0,k-n+1) to min(n-1,k)
    # A[k,j] = f[k-j]

    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            rows.extend([k] * (j_end - j_start + 1))

            data.extend(f[k-j_end:k-j_start+1])
    A = coo_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return A.tocsr()
