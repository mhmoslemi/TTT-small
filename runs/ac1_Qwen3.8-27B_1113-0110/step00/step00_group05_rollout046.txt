import numpy as np
import time
import cvxpy as cp

def compute_convolution(f):
    """Compute linear convolution of f with itself using FFT."""
    n = len(f)
    size = 2 * n
    F = np.fft.rfft(f, size=size)
    conv = np.fft.irfft(F * F, size=size)
    return conv[:2*n-1]

def lp_direction(f, M):
    """Solve the LP to find the best direction g."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    objective = cp.Maximize(cp.sum(g))
    # Constraints: (f*g)_k <= M for all k
    # (f*g)_k = sum_j f_j * g_{k-j} for valid indices
    # This is a linear constraint: for each k, the dot product of a row with g <= M
    # Row k of the "convolution matrix" has f_j at position (k-j) for valid j
    
    constraints = []
    for k in range(2*n-1):
        # (f*g)_k = sum_{j} f_j * g_{k-j} where 0 <= j < n and 0 <= k-j < n
        # = sum_{j} f_j * g_{k-j}
        # Let's build the vector: for each j in [max(0, k-n+1), min(k, n-1)],
        # the term is f_j * g_{k-j}
        # So the constraint is: sum_j f_j * g_{k-j} <= M
        # This is a row of the Toeplitz structure
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            # g indices are k - j for j in [j_start, j_end]
            # g_{k-j_start}, ..., g_{k-j_end}
            g_start = k - j_end
            g_end = k - j_start
            # The coefficients are f_j for j in [j_start, j_end]
            coeffs = f[j_start:j_end+1]
            # But we need them in the right order: g_{g_start} gets coeff f_{j_end}, etc.
            # Actually: sum_{j=j_start}^{j_end} f_j * g_{k-j}
            # g index = k-j, so as j goes from j_start to j_end, g index goes from k-j_start to k-j_end
            # So the vector is: g[g_end], g[g_end-1], ..., g[g_start] dotted with f[j_start], ..., f[j_end]
            # = sum_{i=0}^{j_end-j_start} f[j_start+i] * g[k-j_start-i]
            # Let g_idx = k - j_start - i, so i = k - j_start - g_idx
            # = sum g[g_idx] * f[j_start + k - j_start - g_idx] = sum g[g_idx] * f[k - g_idx]
            # So the coefficient of g[g_idx] is f[k-g_idx] where g_idx in [g_start, g_end]
            # = sum_{g_idx=g_start}^{g_end} f[k-g_idx] * g[g_idx]
            g_indices = np.arange(g_start, g_end+1)
            coeffs = f[k - g_indices]
            constraints.append(coeffs @ g <= M)
    
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.SCS)  # or another solver
        if g.value is not None:
            return np.clip(g.value, 0, np.inf)
    except:
        pass
    return None

**Practical LP-based approach:**

The key idea: at each step, given current $f$ with peak $M = \|f*f\|_\infty$, find a direction $g \geq 0$ such that $f*g \leq M$ (linearized constraint) and $\sum g$ is maximized. Then update $f \leftarrow (1-\alpha)f + \alpha g$ (with $g$ normalized).

The LP:
$\max \sum_j g_j$
s.t. $(f*g)_k \leq M$ for $k = 0, ..., 2n-2$
$g_j \geq 0$ for $j = 0, ..., n-1$

The constraint $(f*g)_k = \sum_i f_i g_{k-i} \leq M$ is linear in $g$.

For $n = 4096$, this LP has 4096 variables and $2 \cdot 4096 - 1 = 8191$ inequality constraints (plus 4096 non-negativity constraints). The constraint matrix is $8191 \times 4096$ with a special banded Toeplitz structure.

Total non-zeros in the constraint matrix: each row $k$ has non-zeros at positions $i$ where $0 \leq k-i < n$, i.e., $i \in [k-n+1, k] \cap [0, n-1]$. The number of non-zeros per row is at most $n$, and the total is $\sum_k \text{nnz}_k = n(2n-1) - \text{corrections} \approx n^2 \approx 1.68 \times 10^7$.

This is a large but not crazy-sized LP. With a solver like HiGHS (included in cvxpy), it should be solvable in a few seconds.

def solve_direction_lp(f, M, active_indices, max_iterations=50):
    """
    Solve the direction-finding LP with constraint generation.
    
    Returns g >= 0 that approximately satisfies f*g <= M on all indices.
    """
    import cvxpy as cp
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    obj = cp.Maximize(cp.sum(g))
    constraints = []
    
    for k in active_indices:
        # (f*g)_k = sum_i f_i * g_{k-i} for valid i
        # g index = k - i, valid when 0 <= k-i < n and 0 <= i < n
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_start <= i_end:
            g_indices = k - np.arange(i_start, i_end + 1)
            coeffs = f[i_start:i_end + 1]
            constraints.append(coeffs @ g[g_indices] <= M)
    
    prob = cp.Problem(obj, constraints)
    
    # Constraint generation: check if solution violates other constraints
    for _ in range(max_iterations):
        try:
            prob.solve(solver=cp.HIGHS)
        except:
            prob.solve(solver=cp.SCS)
        
        if g.value is None:
            return None
        
        g_val = np.clip(g.value, 0, np.inf)
        
        # Check all constraints
        F = np.fft.rfft(f, n=2*n)
        G = np.fft.rfft(g_val, n=2*n)
        fg = np.fft.irfft(F * G, n=2*n)[:2*n-1]
        
        violated = np.where(fg > M * (1 + 1e-6))[0]
        if len(violated) == 0:
            return g_val
        
        # Add a few violated constraints
        # Add the most violated ones
        if len(violated) > 10:
            fg_violated = fg[violated]
            top_violated = violated[np.argsort(fg_violated)[-10:]]
        else:
            top_violated = violated
        
        for k in top_violated:
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            if i_start <= i_end:
                g_indices = k - np.arange(i_start, i_end + 1)
                coeffs = f[i_start:i_end + 1]
                constraints.append(coeffs @ g[g_indices] <= M)

def simplex_project(v):

    n = v.size
