"""
LP-based iterative refinement to minimize 2*n*max(f*f)/(sum(f))^2.
Uses convolution-matrix LP to find improving directions, with line search
for safe blending. Falls back to subgradient/perturbation if LP stalls.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags as sparse_diags
import time


def build_conv_matrix(f, n):
    """Build sparse matrix A (shape (2n-1, n)) such that (A@g)[k] = sum_j f[j]*g[k-j]."""
    diagonals = []
    offsets = []
    for d in range(n):
        length = n - d
        if length <= 0:
            break
        diagonals.append(np.full(length, f[d]))
        offsets.append(-d)
    return sparse_diags(diagonals, offsets, shape=(2 * n - 1, n), format='csr')


def compute_score(f, n):
    """Compute the evaluation score directly."""
    s = float(np.sum(f))
    if s < 0.01:
        return float('inf')
    conv = np.convolve(f, f)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize_to(f, target_sum):
    """Scale f so that sum equals target_sum."""
    s = float(np.sum(f))
    if s < 1e-12:
        return np.ones_like(f) * target_sum / len(f)
    return f * (target_sum / s)


def subgradient_step(f, n, lr):
    """Take a subgradient step to reduce max(f*f)."""
    conv = np.convolve(f, f)
    k_star = int(np.argmax(conv))
    # Gradient of (f*f)[k_star] with respect to f[j]
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        if 0 <= k_star - j < n:
            g += f[k_star - j]
        if 0 <= j - k_star < n:
            g += f[j - k_star]
        grad[j] = g
    # Projected gradient descent onto sum(f) = target, f >= 0
    target = np.sqrt(2.0 * n)
    f_new = f - lr * grad
    f_new = np.maximum(f_new, 0.0)
    f_new = normalize_to(f_new, target)
    return f_new

        f_new[start:end] *= scale
    # Symmetrize occasionally
    if rng.random() < 0.3:
        f_new = 0.5 * (f_new + f_new[::-1])
    f_new = np.maximum(f_new, 0.0)
    return f_new
