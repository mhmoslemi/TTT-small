Minimize 2*n*max(convolve(a,a))/(sum(a))^2 using log-sum-exp smoothed
gradient descent with adaptive beta, multiple sequence lengths, and
LP-based refinement via cvxpy.
"""

import numpy as np
import time
import sys


def compute_objective(a):
    """Compute the objective value directly (no clipping, assumes valid input)."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_c / (sum_a ** 2))


def gradient_step(a, softmax_weights, eta):
    """
    Perform one gradient step on the log-sum-exp smoothed objective.
    a: normalized non-negative array (sum=1)
    softmax_weights: length 2*n-1 array (softmax of beta*c)
    Returns updated a (normalized, non-negative).
    """
    # grad[j] = 2 * sum_m a[m] * softmax[j+m] for m=0..n-1, j=0..n-1
    # This is 2 * np.correlate(softmax, a, mode='valid')
    grad = 2.0 * np.correlate(softmax_weights, a, mode='valid').astype(np.float64)
    
    # Center the gradient to preserve sum
    grad_centered = grad - grad.mean()
    
    a_new = a - eta * grad_centered
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 0:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new


def optimize_sequence(a_init, beta_init, beta_max, n_iterations, deadline, seed):

def lp_refine(a, deadline, max_iters=5):
    """
    LP-based refinement: find g >= 0 with (a*g)[k] <= M, maximize sum(g).
    Then blend: a_new = (1-t)*a + t*g_normalized.
    """
    from scipy.optimize import linprog
    
    a = a.astype(np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        return a
    a = a / s
    n = len(a)
    
    c = np.convolve(a, a)
    M = c.max()
    
    best_a = a.copy()
    best_score = compute_objective(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # LP: maximize sum(g) subject to (a*g)[k] <= M for all k, g >= 0
        # (a*g)[k] = sum_i a[i] * g[k-i] (with g[m]=0 outside [0,n-1])
        # This is a matrix A @ g <= M where A[k,j] = a[j] if 0<=k-j<n else 0
        # A[k,j] = a[j] * (1 if max(0,k-n+1) <= j <= min(k,n-1) else 0)
        # Actually A[k,j] = a[k-j] if 0 <= k-j < n else 0 ... 
        # Wait: (a*g)[k] = sum_j a[j]*g[k-j] = sum_j g[j]*a[k-j]
        # So the constraint matrix has A[i,j] = a[k-i] where k is the lag... 
        # Let me be more careful.
        # (a*g)[k] = sum_{j=0}^{n-1} a[j] * g[k-j]  (g[m]=0 if m outside [0,n-1])
        # = sum_{j: 0<=k-j<=n-1} a[j] * g[k-j]
        # Let m = k-j, j = k-m:
        # = sum_{m: 0<=k-m<=n-1, 0<=m<=n-1} a[k-m] * g[m]
        # = sum_{m} A[k,m] * g[m] where A[k,m] = a[k-m] if 0<=k-m<=n-1 and 0<=m<=n-1
        #
        # So for variable g[m], its coefficient in constraint k is a[k-m] (if valid)
        # This means the constraint matrix is a "convolution matrix" - it's banded.
        #
        # For large n, this matrix is sparse (bandwidth n). Using scipy's linprog
        # with sparse matrix should work.
        
        from scipy.sparse import lil_matrix
        
        # Build constraint matrix A (size (2n-1) x n), sparse
        # A[k, m] = a[k-m] if 0 <= k-m <= n-1 and 0 <= m <= n-1
        A_sp = lil_matrix((2 * n - 1, n), dtype=np.float64)
        for m in range(n):
            # For variable g[m], it appears in constraints k where 0 <= k-m <= n-1
            # i.e., m <= k <= m + n - 1
            k_start = m
            k_end = min(m + n - 1, 2 * n - 2)
            for k in range(k_start, k_end + 1):
                A_sp[k, m] = a[k - m]
        
        A_sp = A_sp.tocsr()
        
        # Maximize sum(g) = minimize -sum(g)
        c_obj = -np.ones(n)
        
        # Constraints: A_sp @ g <= M
        b_vec = np.full(2 * n - 1, M)
        
        # Bounds: g >= 0
        bounds = [(0, None)] * n
        
        try:
            result = linprog(c_obj, A_ub=A_sp, b_ub=b_vec, bounds=bounds,
                            method='highs', options={'presolve': True, 'time_limit': 60})
            if result.success:
                g = result.x
                g_sum = g.sum()
                if g_sum > 1e-10:
                    g_norm = g / g_sum
                    # Blend
                    t = 0.3
                    a_blend = (1 - t) * a + t * g_norm
                    a_blend = np.maximum(a_blend, 0.0)
                    a_blend = a_blend / a_blend.sum()
                    
                    score = compute_objective(a_blend)
                    if np.isfinite(score) and score < best_score:
                        best_a = a_blend.copy()
                        best_score = score
                        a = a_blend.copy()
                        # Recompute M
                        c_new = np.convolve(a, a)
                        M = c_new.max()
                else:
                    break
            else:
                break
        except Exception as e:
            print(f"  LP error: {e}")
            break
    
    return best_a
