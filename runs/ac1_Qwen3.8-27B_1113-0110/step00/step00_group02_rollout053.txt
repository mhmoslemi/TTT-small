"""
Peak-shaving via subgradient descent for non-negative step function coefficients.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-based convolution,
subgradient descent with simulated annealing, and structural modifications.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """FFT-based linear convolution of two 1-D arrays."""
    n, m = len(a), len(b)
    size = n + m - 1
    fft_size = 1
    while fft_size < size:
        fft_size <<= 1
    Fa = np.fft.rfft(a, fft_size)
    Fb = np.fft.rfft(b, fft_size)
    out = np.fft.irfft(Fa * Fb, fft_size)[:size]
    # Clean up tiny negatives from numerical error
    out[np.abs(out) < 1e-12] = 0.0
    return out


def compute_objective_fast(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2 without full validation."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    b = fft_convolve(a, a)
    max_b = float(np.max(b))
    if max_b < 0:
        max_b = 0.0
    return 2.0 * n * max_b / (sum_a * sum_a)


def compute_subgrad(a, n_top=5):
    """Compute weighted subgradient direction to reduce max(conv(a,a))."""
    n = len(a)
    b = fft_convolve(a, a)
    m = len(b)
    k_top = min(n_top, m)
    top_idx = np.argpartition(b, -k_top)[-k_top:]
    peak_vals = b[top_idx]
    peak_sum = float(np.sum(peak_vals))
    if peak_sum < 1e-30:
        return np.zeros(n), b, top_idx, 0.0
    weights = peak_vals / peak_sum

    grad = np.zeros(n)
    for wt, k in zip(weights, top_idx):
        lo = max(0, int(k) - (n - 1))
        hi = min(n - 1, int(k))
        if lo > hi:
            continue
        vi = np.arange(lo, hi + 1)
        partners = int(k) - vi
        grad[vi] += 2.0 * wt * a[partners]
    return grad, b, top_idx, float(b[top_idx[0]])


def make_step(a, lr, rng, temperature=0.0, n_peaks=5):
    """One subgradient + noise step."""
    n = len(a)
    grad, b, top_idx, max_val = compute_subgrad(a, n_peaks)
    if max_val < 1e-30:
        return a.copy()
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return a.copy()
    a_new = a - lr * grad / gn
    if temperature > 1e-9:
        scale = temperature * max(float(np.mean(a)), 1e-9)
        a_new = a_new + rng.normal(0.0, scale, size=n)
    a_new = np.maximum(a_new, 0.0)
    s_new = float(np.sum(a_new))
    s_old = float(np.sum(a))
    if s_new < 1e-12:
        a_new = a.copy()
    else:
        a_new = a_new * (s_old / s_new)
    a_new = np.clip(a_new, 0.0, 1000.0)
    return a_new


def try_structural_change(a, rng):
    """Random structural modification to escape local minima."""
    n = len(a)
    if n < 4:
        return a.copy()
    op = int(rng.integers(0, 5))
    a_new = a.copy()
    if op == 0:
        a_new = 0.5 * (a + a[::-1])
    elif op == 1:
        shift = int(rng.integers(1, n))
        a_new = np.roll(a, shift)
    elif op == 2:
        w = max(2, n // 25)
        if n - 2 * w <= 0:
            return a.copy()
        i1 = int(rng.integers(0, n - 2 * w))
        i2 = int(rng.integers(0, n - w))
        a_new[i1:i1 + w], a_new[i2:i2 + w] = a_new[i2:i2 + w].copy(), a_new[i1:i1 + w].copy()
    elif op == 3:
        w = int(rng.integers(max(2, n // 40), max(3, n // 4)))
        start = int(rng.integers(0, n - w))
        a_new[start:start + w] *= float(np.exp(rng.normal(0.0, 0.5)))
    else:
        # Multiply by a low-freq cosine to reshape
        freq = float(rng.uniform(0.5, 3.0))
        i = np.arange(n, dtype=np.float64)
        mult = 1.0 + rng.uniform(0.2, 0.7) * np.cos(2.0 * np.pi * freq * i / n)
        a_new = a * mult
    a_new = np.maximum(a_new, 0.0)
    s = float(np.sum(a_new))
    s_old = float(np.sum(a))
    if s > 1e-12:
        a_new = a_new * (s_old / s)
    return np.clip(a_new, 0.0, 1000.0)
