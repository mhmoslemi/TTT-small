import numpy as np
import time

def project_onto_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    # Sort in descending order
    u = np.sort(v)[::-1]
    # Find rho
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u > (cssv - 1) / rho
    rho_star = np.max(rho[cond])
    nu = (cssv[rho_star - 1] - 1) / rho_star
    return np.maximum(v - nu, 0)

def convolve_fft(a, b):
    """Compute linear convolution using FFT."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()  # next power of 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

        num_active = max(10, min(1000, n // 5))

Algorithm: Convex optimization via projected subgradient method with FFT acceleration.
Minimizes 2n*max(a*a)/(sum_a)^2 over non-negative sequences.

def project_onto_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u > (cssv - 1.0) / rho
    if not np.any(cond):
        return np.ones(n) / n
    rho_star = int(np.max(rho[cond]))
    nu = (cssv[rho_star - 1] - 1.0) / rho_star
    return np.maximum(v - nu, 0.0)

def convolve_fft(a, b):
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    result = np.fft.irfft(fa * fb, size)
    return result[:n]

def self_convolve_fft(a):
    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]

def optimize_sequence(a_init, time_budget, verbose=False):

    if s <= 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s
    a = np.maximum(a, 0.0)
    
    best_a = a.copy()
    b = self_convolve_fft(a)
    best_score = 2.0 * n * b.max()
    
    alpha = 1.0 / np.sqrt(n)
    start = time.monotonic()
    iters = 0
    stagnant = 0
    
    while time.monotonic() - start < time_budget:
        b = self_convolve_fft(a)
        M = b.max()
        current_score = 2.0 * n * M
        
        if current_score < best_score - 1e-12:
            best_score = current_score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        # Compute subgradient using top active constraints
        num_top = max(5, min(200, n // 10))
        # Use argpartition for efficiency
        if num_top >= len(b):
            top_idx = np.arange(len(b))
            weights = b / b.sum()
        else:
            top_vals = np.argpartition(b, -num_top)[-num_top:]
            top_idx = np.sort(top_vals)
            # Weight by how close to max
            w = b[top_idx] / M
            w = w ** 10  # sharpen weights
            weights = w / w.sum()
        
        g = np.zeros(n)
        for k, w in zip(top_idx, weights):
            i_s = max(0, k - n + 1)
            i_e = min(n, k + 1)
            g[i_s:i_e] += w * a[k - np.arange(i_s, i_e)]
        
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        
        # Subgradient step
        a_try = a - alpha * g
        a_try = project_onto_simplex(a_try)
        
        # Check improvement
        b_try = self_convolve_fft(a_try)
        M_try = b_try.max()

            a = a_try

                a_try = a + noise

                a = a_try

            print(f"    iter={iters}, score={best_score:.6f}, alpha={alpha:.2e}, elapsed={elapsed:.1f}s")
