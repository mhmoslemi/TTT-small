def compute_subgradient(a, k_star):
    """Compute subgradient of conv(a,a)[k_star] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    # For j where 0 <= k_star - j <= n-1, i.e., k_star - (n-1) <= j <= k_star
    j_start = max(0, k_star - n + 1)
    j_end = min(n - 1, k_star)
    if j_start <= j_end:
        # g[j] += a[k_star - j] for j in [j_start, j_end]
        idx = k_star - np.arange(j_start, j_end + 1)
        g[j_start:j_end+1] += a[idx]
    # For j where 0 <= j - k_star <= n-1, i.e., k_star <= j <= k_star + (n-1)
    j_start2 = max(0, k_star)
    j_end2 = min(n - 1, k_star + n - 1)
    if j_start2 <= j_end2:
        # g[j] += a[j - k_star] for j in [j_start2, j_end2]
        idx2 = np.arange(j_start2, j_end2 + 1) - k_star
        g[j_start2:j_end2+1] += a[idx2]
    return g
