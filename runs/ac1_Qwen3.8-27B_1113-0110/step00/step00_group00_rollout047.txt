import numpy as np
import time
from scipy.optimize import linprog  # if needed

def compute_conv(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    # Use next power of 2 for FFT
    size = 2 * n - 1
    ft = np.fft.rfft(a, size=2 * size)
    conv = np.fft.irfft(ft * ft)[:size]
    return conv

def compute_grad_max_conv(a):
    """Compute gradient of max_k conv(a,a)[k] with respect to a."""
    n = len(a)
    conv = compute_conv(a)
    m = np.argmax(conv)
    # Gradient: g[j] = a[m-j] (if 0<=m-j<n) + a[m-j] (if 0<=m-j<n and m-j != j)
    g = np.zeros(n)
    for j in range(n):
        k = m - j
        if 0 <= k < n:
            g[j] += a[k]
            if k != j:
                g[j] += a[k]
    return g, m

def project_simplex(x):
    """Project x onto the simplex {a >= 0, sum(a) = 1}."""
    # ... bisection on tau ...
    n = len(x)
    lo, hi = x.min() - 1, x.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.maximum(x - mid, 0))
        if s > 1:
            lo = mid
        else:
            hi = mid
    return np.maximum(x - mid, 0)

def optimize_sequence(n, a0, max_iters=100000, tol=1e-10):
    """Minimize max(conv(a,a)) subject to sum(a)=1, a>=0."""
    a = a0.copy()
    a = a / a.sum()  # normalize
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    best_conv_max = np.max(compute_conv(a))
    best_a = a.copy()
    
    step = 0.01
    for it in range(max_iters):
        g, m = compute_grad_max_conv(a)
        a_new = a - step * g
        a_new = project_simplex(a_new)
        conv = compute_conv(a_new)
        conv_max = np.max(conv)
        if conv_max < best_conv_max - tol:
            best_conv_max = conv_max
            best_a = a_new.copy()
            a = a_new
            # Maybe increase step
            step = min(step * 1.05, 0.1)
        else:
            a = best_a.copy()
            step *= 0.8
    
    return best_a, best_conv_max
