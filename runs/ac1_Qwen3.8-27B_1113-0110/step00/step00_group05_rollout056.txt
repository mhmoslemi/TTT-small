import numpy as np
import time
import scipy.sparse
import scipy.optimize

def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    rho = np.max(np.where(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_grad_and_C(s, beta):
    """Compute soft-gradient and C array."""
    n = len(s)
    C = np.convolve(s, s)
    C_max = C.max()
    w = np.exp(beta * (C - C_max))
    w /= w.sum()
    a_flip = s[::-1]
    c = np.convolve(a_flip, w)
    grad = 2.0 * c[n - 1:2 * n - 1]
    return grad, C, C_max

    s = np.maximum(s, 0)
    ssum = s.sum()
    if ssum < 1e-12:
        s = np.full(n, 1.0 / n)
    else:
        s /= ssum
    
    C = np.convolve(s, s)
    C_max = C.max()
    best_s = s.copy()
    best_Cmax = C_max
    
    eta = eta_init
    s_prev = s.copy()
    t_k = 1.0
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        grad, C, C_max = compute_grad_and_C(s, beta)

def run_lp_step(s, M_multiplier=1.0):
    """One LP-based improvement step."""
    n = len(s)
    C = np.convolve(s, s)
    M = C.max() * M_multiplier
    
    # Build sparse constraint matrix A where A[k,i] = s[i] if 0<=k-i<n
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            rows.append(k)
            cols.append(i)
            vals.append(s[i])
    
    A = scipy.sparse.coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    c = np.full(n, -1.0)  # minimize -sum(g) = maximize sum(g)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    result = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 30})
    
    if result.success:
        g = result.x
        g_sum = g.sum()
        if g_sum > 1e-10:
            g /= g_sum
            return g
    return None

    s = (2.0 * t - 1.0) ** 2 + 0.01
    inits.append(s / s.sum())
    for fr in [1, 2, 3]:
        for al in [0.4, 0.7, 0.9]:
            tt = np.arange(n)
            s = 1.0 + al * np.cos(2.0 * np.pi * fr * tt / n)
            s = np.maximum(s, 0.0)
            sm = s.sum()
            if sm > 1e-10:
                inits.append(s / sm)
    s = rng.random(n)
    inits.append(s / s.sum())
    return inits

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 5.0)
    n = 4498
    base = None
    try:
        b = np.asarray(height_sequence_1, dtype=np.float64)
        if b.ndim == 1 and b.size > 0:
            n = b.size
            bc = np.clip(b, 0.0, 1000.0)
            if bc.sum() > 1e-10:
                base = bc / bc.sum()
    except Exception:
        pass
    if base is None:
        base = np.ones(n) / n
    inits = make_inits(n, base, rng)
    print(f"n={n}, #inits={len(inits)}, deadline={deadline - time.monotonic():.1f}s")
    best_s = None
    best_score = np.inf
    for bi, s_init in enumerate(inits):
        t_rem = deadline - time.monotonic()
        if t_rem < 15:
            break
        t_alloc = min(8.0, 0.5 * t_rem / max(1, len(inits) - bi))
        s_opt, C_opt = soft_descend(s_init, 20.0, t_alloc, rng, eta0=0.003)
        sc = evaluate_sequence(s_opt.tolist())
        print(f"  init {bi:2d}: Cmax={C_opt:.8f} score={sc:.6f}  ({time.monotonic() - (deadline - t_rem):.1f}s)")
        if sc < best_score:
            best_score = sc
            best_s = s_opt.copy()
    t_rem = deadline - time.monotonic()
    if t_rem > 60 and best_s is not None:
        s_opt, C_opt = soft_descend(best_s, 20.0, 0.4 * t_rem, rng, eta0=0.001)
        sc = evaluate_sequence(s_opt.tolist())
        print(f"  refine-soft: Cmax={C_opt:.8f} score={sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_s = s_opt
    t_rem = deadline - time.monotonic()
    if t_rem > 40 and best_s is not None:
        for beta in [10, 50]:
            t_a = min(60.0, t_rem * 0.25)
            s_opt, C_opt = soft_descend(best_s, beta, t_a, rng, eta0=0.0005)
            sc = evaluate_sequence(s_opt.tolist())
            print(f"  refine-soft beta={beta}: score={sc:.6f}")
            if sc < best_score:
                best_score = sc
                best_s = s_opt
            t_rem = deadline - time.monotonic()
            if t_rem < 20:
                break
    t_rem = deadline - time.monotonic()
    if t_rem > 20 and best_s is not None:
        s_opt, C_opt = hard_descend(best_s, t_rem - 5, rng)
        sc = evaluate_sequence(s_opt.tolist())
        print(f"  refine-hard: Cmax={C_opt:.8f} score={sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_s = s_opt
    result = [float(x) for x in best_s]
    final_score = float(evaluate_sequence(result))
    print(f"FINAL: {final_score:.8f}")
    return result
