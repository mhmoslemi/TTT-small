```python
def frank_wolfe_step(p, active_set, gamma):
    """One Frank-Wolfe step for min max_k (p*p)[k] s.t. p in simplex."""
    n = len(p)
    # Compute subgradient
    g = np.zeros(n)
    for k in active_set:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                g[i] += 2.0 * p[j]
    g /= len(active_set)
    
    # Find vertex with minimum gradient
    i_star = np.argmin(g)
    q = np.zeros(n)
    q[i_star] = 1.0
    
    # Update
    p_new = (1 - gamma) * p + gamma * q
    return p_new
```

```python
import numpy as np
import time

def project_simplex(v):
    """Project vector v onto the probability simplex {x: sum x = 1, x >= 0}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_autoconv(p):
    """Compute the full autoconvolution of p."""
    return np.convolve(p, p)
```
