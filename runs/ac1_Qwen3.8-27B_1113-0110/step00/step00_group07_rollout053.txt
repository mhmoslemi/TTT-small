"""
Minimize f(x) = 2*n*max(convolve(x,x))/sum(x)**2 for a non-negative step
function.  The objective is scale invariant, so the search works on the unit
simplex (sum(x)=1) and is equivalent to driving max(convolve(x,x)) toward the
theoretical floor 1/(2n-1).

Method: FFT-accelerated gradient descent on the unit simplex.
  * The convolution is computed with an O(n log n) FFT convolution.
  * Two gradient flavours are mixed:
      - exact sub-gradient of the active peak  d c[k*]/dx = 2 * x[k*-i]
      - smoothed Lp gradient  d sum(c^p)/dx = p*2 * corr(x, c^(p-1))
    where corr is an FFT cross-correlation.
  * A momentum term, a centred gradient (to avoid wasted scale moves),
    projected non-negativity, and renormalisation to sum 1 are used.
  * A multi-phase schedule (lr, noise, peak-fraction, p) goes from coarse
    exploration to fine polishing, with random restarts to escape local minima.
  * The internal O(n log n) objective is tracked every iteration as a faithful
    proxy; the authoritative evaluate_sequence() is called periodically and on
    exit to confirm / update the true best sequence.

Top-level helpers carry no closures and no lambdas.
"""

import time

import numpy as np
from scipy.signal import fftconvolve


def _next_pow2(m):
    """Smallest power of two >= m (m>=1)."""
    k = 1
    while k < m:
        k <<= 1
    return k


def _fft_conv(x):
    """Linear self-convolution via FFT."""
    return fftconvolve(x, x, mode='full')


def _internal_obj(x):
    """Cheap O(n log n) proxy for evaluate_sequence (same formula)."""
    c = _fft_conv(x)
    s = float(np.sum(x))
    if s < 1e-12:
        return np.inf
    return 2.0 * len(x) * float(np.max(c)) / (s * s)


def _smooth_grad(x, c, p, N):
    """Gradient of sum(c^p) w.r.t. x via FFT cross-correlation."""
    cpow = c ** (p - 1)
    fx = np.fft.rfft(x, N)
    fc = np.fft.rfft(cpow, N)
    r = np.fft.irfft(np.conj(fx) * fc, N)
    return (p * 2.0) * r[: len(x)]


def _peak_grad(x, k):
    """Exact gradient of the single lag c[k] w.r.t. x."""
    n = len(x)
    g = np.zeros(n, dtype=np.float64)
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    idx = np.arange(lo, hi)
    g[idx] = 2.0 * x[k - idx]
    return g


def _project(x):
    """Project onto the non-negative unit simplex (sum == 1)."""
    x = np.maximum(x, 0.0)
    s = float(np.sum(x))
    if s < 1e-12:
        return np.ones_like(x, dtype=np.float64) / len(x)
    return x / s


def _center(g):
    """Remove the mean so the step mainly changes shape, not scale."""
    return g - float(np.mean(g))


def _make_inits(n, rng, base):
    """A handful of reasonable starting points (all normalised)."""
    idx = np.arange(n, dtype=np.float64)
    inits = []

    if base is not None:
        b = np.asarray(base, dtype=np.float64)
        if b.ndim == 1 and b.size == n and np.isfinite(b).all():
            b = np.clip(b, 0.0, 1000.0)
            if float(np.sum(b)) > 0.01:
                inits.append(_project(b.copy()))

    inits.append(np.ones(n, dtype=np.float64))                              # uniform
    inits.append(np.maximum(0.5 * (1.0 + np.cos(2.0 * np.pi * idx / n)), 1e-10))  # Hann
    inits.append(np.maximum(1.0 + 0.5 * np.cos(2.0 * np.pi * idx / (n // 3)), 1e-10))
    inits.append(np.exp(-0.5 * (2.0 * (idx - n / 2.0) / n) ** 2) + 0.01)     # gaussian
    inits.append(np.maximum(1.0 + 0.4 * np.sin(2.0 * np.pi * idx / (n // 5)), 1e-10))

    r = rng.random(n)
    ker = np.ones(min(11, n)) / min(11, n)
    r = np.convolve(r, ker, mode='same')
    inits.append(np.maximum(r, 0.01))                                       # smoothed random

    return [_project(a.copy()) for a in inits]
