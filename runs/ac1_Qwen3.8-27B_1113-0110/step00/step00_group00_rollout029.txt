import cvxpy as cp

def lp_iteration(f, top_k=2000):
    """One LP-based iteration. Returns new f."""
    n = len(f)
    M = np.max(np.convolve(f, f))
    
    # Compute auto-convolution
    ff = np.convolve(f, f)
    
    # Get top K indices
    if top_k < len(ff):
        top_indices = np.argsort(ff)[-top_k:]
    else:
        top_indices = np.arange(len(ff))
    
    # Build constraint matrix A (top_k x n)
    # A[i, j] = f[top_indices[i] - j] if 0 <= top_indices[i] - j < n
    A = np.zeros((len(top_indices), n))
    for idx, k in enumerate(top_indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[idx, lo:hi+1] = f[k-hi:k-lo+1][::-1]
    
    # Solve: max 1^T g s.t. A g <= M, g >= 0
    # Using cvxpy
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS)
    
    if g.value is None:
        return f
    
    g_val = np.maximum(g.value, 0)
    g_val = g_val / np.sum(g_val) * np.sum(f)  # rescale to match sum
    
    # Blend
    t = 0.5
    new_f = (1 - t) * f + t * g_val
    return new_f / np.sum(new_f) * np.sum(f)
