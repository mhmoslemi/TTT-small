    size = 2 * n - 1
    fsize = 1

    A = np.fft.rfft(a, fsize)
    ac = np.fft.irfft(A * np.conj(A), fsize)[:size]
    return ac

    return np.where(ac >= M - tol)[0]

    n = len(a)
    # dM/da[i] = (1/|K*|) * sum_{k in K*} dc[k]/da[i]
    # dc[k]/da[i] = 2*a[k-i] if 0 <= k-i < n, else 0

"""
Gradient-based projected descent to minimize peak linear autocorrelation.
Objective: score = 2n * max(conv(a,a)) / sum(a)^2, minimize over non-negative a.
Uses FFT for O(n log n) per-step cost. Combines subgradient at peak lags,
adaptive learning rate, periodic structured perturbations, and multi-start.

def fast_autocorr(a):
    """Compute full linear autocorrelation (a convolved with itself) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1
    while fsize < size:
        fsize <<= 1
    A = np.fft.rfft(a, fsize)
    ac = np.fft.irfft(A * np.conj(A), fsize)[:size]
    return ac

def fast_gradient_M(a, peak_lags):
    """
    Compute subgradient of M(a)=max_k ac[k] at given peak lags.
    dM/da[i] = (1/|K*|) * sum_{k in K*} 2*a[k-i] for 0<=k-i<n
    Uses FFT cross-correlation for O(n log n) computation.
    """
    n = len(a)
    m = len(peak_lags)

        return np.zeros(n)

    # = (2/m) * ifft(conj(FFT(a)) * FFT(v))[i]  (linear cross-correlation)
    fsize = 1
    while fsize < n + size:

    A = np.fft.rfft(a, fsize)
    V = np.fft.rfft(v, fsize)
    cc = np.fft.irfft(np.conj(A) * V, fsize)
    return (2.0 / m) * cc[:n]
