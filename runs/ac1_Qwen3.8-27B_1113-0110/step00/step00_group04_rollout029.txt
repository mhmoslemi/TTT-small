import numpy as np

def project_onto_simplex(a, target_sum=1.0):
    """Project a onto {x >= 0, sum(x) = target_sum}."""
    # This is the Euclidean projection onto the simplex
    # Standard algorithm
    n = len(a)
    if n == 0:
        return a
    # Sort in descending order
    u = np.sort(a)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    w = np.maximum(a - theta, 0)
    return w

def compute_score_gradient(a):
    """Compute score, max convolution value, and gradient of score w.r.t. a."""
    n = len(a)
    c = np.convolve(a, a)  # length 2n-1
    max_c = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return np.inf, max_c, np.zeros_like(a)
    score = 2.0 * n * max_c / (S * S)
    
    # Gradient of max_c / S^2
    # d(max_c)/d(a[i]) = 2*a[k*-i] (or a[i] if 2i=k*)
    active = np.where(np.abs(c - max_c) < 1e-10)[0]
    # For simplicity, use just the argmax
    k_star = np.argmax(c)
    grad = np.zeros_like(a)

        if 0 <= k_star - i < n:
            if k_star - i != i:
                grad[i] += 2.0 * a[k_star - i]
            else:

def fast_convolve(a):
    """Compute np.convolve(a, a) using FFT for O(n log n) speed."""

    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2*n-1]

    """Compute max of convolve(a,a) efficiently."""
    c = fast_convolve(a)
    return c, np.max(c)

def gradient_max_c(a, k_star):
    """Compute dc[k_star]/da as a vector."""
    n = len(a)
    idx = np.arange(n)
    valid = (k_star - idx >= 0) & (k_star - idx < n)
    grad = np.zeros(n)
    grad[valid] = 2.0 * a[k_star - idx[valid]]

        if 0 <= i_mid < n:

"""
Minimize 2n*max(convolve(a,a))/sum(a)^2 using projected gradient descent
with active-set smoothing. Key insight: asymmetric sequences break the
symmetric lower bound of 2. Uses FFT-based convolution for efficiency.
"""

import numpy as np
import time
from scipy.fft import next_fast_len


def fast_convolve(a):
    """Compute np.convolve(a, a) using FFT."""
    n = len(a)
    size = next_fast_len(2 * n - 1)
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]


def grad_c_at_k(a, k):
    """Gradient of c[k]=convolve(a,a)[k] w.r.t. a, vector of length n."""
    n = len(a)
    idx = np.arange(n)
    valid = (k - idx >= 0) & (k - idx < n)
    grad = np.zeros(n)
    grad[valid] = 2.0 * a[k - idx[valid]]
    return grad


def make_geometric(n, r):
    a = r ** np.arange(n, dtype=np.float64)
    return a / np.sum(a)


def make_power_law(n, p):
    a = (np.arange(n, dtype=np.float64) + 1.0) ** p
    return a / np.sum(a)


def make_ramp(n, alpha):
    """Two-level: first (1-alpha) fraction is low, rest is high."""
    a = np.ones(n)
    split = int(n * (1 - alpha))
    a[:split] = 0.3
    return a / np.sum(a)


def make_bump(n, center_frac, width):
    x = np.arange(n, dtype=np.float64) / n
    a = np.exp(-((x - center_frac) ** 2) / (2 * width ** 2))
    return a / np.sum(a)


def optimize_from_init(a_init, time_limit=30.0, top_k=5, verbose=False):
    """Optimize using projected gradient with active-set averaging."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return a_init.copy(), np.inf
    a = a / S

    c, current_max = fast_convolve(a), None
    c = fast_convolve(a)
    current_max = np.max(c)
    best_a = a.copy()
    best_max = current_max

    lr = 0.3
    start_time = time.monotonic()
    stagnant = 0

    for it in range(200000):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit:
            break

        # Find active set
        k_star = np.argmax(c)
        threshold = current_max * 0.9
        active = np.where(c >= threshold)[0]
        if len(active) > top_k:
            active_vals = c[active]
            top_idx = np.argpartition(active_vals, -top_k)[-top_k:]
            active = active[top_idx]
        elif len(active) == 0:
            active = np.array([k_star])

        # Average gradient over active set
        grad = np.zeros(n)
        for k in active:
            grad += grad_c_at_k(a, k)
        grad /= len(active)

        # Project onto sum=0 subspace
        grad = grad - np.mean(grad)

        # Take step with backtracking
        improved = False
        for _back in range(10):
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 0.0)
            S_new = np.sum(a_new)
            if S_new < 1e-10:
                lr *= 0.5
                break
            a_new = a_new / S_new

            c_new = fast_convolve(a_new)
            new_max = np.max(c_new)
            if new_max < current_max * (1.0 - 1e-10):
                a = a_new
                c = c_new
                current_max = new_max
                improved = True
                break
            lr *= 0.3

        if improved:
            if current_max < best_max:
                best_max = current_max
                best_a = a.copy()
                stagnant = 0
            else:
                stagnant += 1
            lr = min(lr * 1.1, 2.0)
            # Occasional random perturbation to escape local minima
            if stagnant > 200 and it % 500 == 0:
                pert = np.random.standard_normal(n) * 0.05 * np.sqrt(best_max)
                a_pert = np.maximum(a + pert, 0.0)
                S_p = np.sum(a_pert)
                if S_p > 1e-10:
                    a_pert = a_pert / S_p
                    c_pert = fast_convolve(a_pert)
                    if np.max(c_pert) < best_max:
                        a = a_pert
                        c = c_pert
                        current_max = np.max(c)
                        stagnant = 0
        else:
            lr *= 0.7
            stagnant += 1
            if lr < 1e-14:
                break
            # Reset lr occasionally
            if stagnant % 500 == 0:
                lr = 0.3

        if verbose and it % 20000 == 0 and it > 0:
            print(f"      iter={it}, score={2*n*best_max:.6f}, lr={lr:.4f}, active={len(active)}")

    return best_a, 2 * n * best_max
