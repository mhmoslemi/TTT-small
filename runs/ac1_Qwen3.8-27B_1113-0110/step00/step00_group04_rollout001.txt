def evaluate_local(sequence):

    seq = np.asarray(sequence, dtype=np.float64)
    n = seq.size
    if n == 0:
        return np.inf
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)
    max_b = np.max(conv)
    return float(2.0 * n * max_b / (s * s))

        start = int(rng.integers(0, n - width))
        candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.1))
    return np.clip(candidate, 0.0, 1000.0)
