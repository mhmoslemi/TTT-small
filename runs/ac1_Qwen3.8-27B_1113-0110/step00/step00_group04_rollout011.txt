    m = conv.max()
    return m + np.log(np.sum(np.exp(beta * (conv - m)))) / beta

Optimization of non-negative sequence to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses smooth-max L-BFGS-B optimization in Fourier domain + targeted local search.
"""

import numpy as np
import time
from scipy.fft import fft, ifft


def fast_conv(a):
    """Compute linear self-convolution via FFT. Returns array of length 2*n-1."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    A = fft(a, N)
    return np.real(np.clip(ifft(A * A)[:2 * n - 1], 0.0, None))


def smooth_max(vals, beta):
    """Numerically stable log-sum-exp based smooth max."""
    m = vals.max()
    if m < 1e-300:
        return 0.0
    return m + np.log(np.sum(np.exp(beta * (vals - m)))) / beta


def compute_gradient(a, beta):
    """
    Gradient of smooth_max(conv(a,a), beta) w.r.t. a.
    grad_i = 2 * sum_k w_k * a[k-i] for valid k-i in [0, n-1]
    where w = softmax(beta * conv(a,a)).
    """
    n = len(a)
    conv = fast_conv(a)
    # Softmax weights
    m = conv.max()
    exp_vec = np.exp(beta * (conv - m))
    w = exp_vec / exp_vec.sum()
    # Cross-correlation: cross[i] = sum_j w[i+j] * a[j] for j=0..n-1
    N2 = 1
    while N2 < (2 * n - 1) + n - 1:
        N2 <<= 1
    W = fft(w, N2)
    Af = fft(a, N2)
    cross = np.real(ifft(W * np.conj(Af)))
    # cross[i] = sum_j w[j+i]*a[j] (circular, but N2 is large enough)
    grad = 2.0 * cross[:n]
    return grad, conv, w
