"""
Minimizes 2n*max(auto_conv)/sum^2 via subgradient descent on the simplex
with FFT convolution, top-K smoothed subgradients, multiple starting points,
and a coordinate-descent polishing phase.
"""
import numpy as np
import time


def _convolve_full(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    out = np.fft.irfft(fa * fa, fsize)[:size]
    np.maximum(out, 0.0, out=out)
    return out


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u * np.arange(1, n + 1) > cssv - 1.0)[0]
    if len(rho_arr) == 0:
        return np.full(n, 1.0 / n)
    r = int(rho_arr[-1])
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def _score_norm(a):
    """Score for sum(a)=1: 2n * max(conv)."""
    n = len(a)
    c = _convolve_full(a)
    return 2.0 * n * float(c.max())


def _grad_k(a, k):
    """
    Gradient of (a*a)[k] w.r.t. a.
    Result[i] = 2*a[k-i] if 0 <= k-i < n, else 0.
    """
    n = len(a)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    g = np.zeros(n)
    if lo <= hi:
        g[lo:hi + 1] = 2.0 * a[k - lo: k - hi + 1][::-1]
    return g


def _smoothed_subgrad(a, K=30):
    """
    Smoothed subgradient: softmax-weighted average of top-K active gradients.
    Returns (gradient, max_conv_value).
    """
    n = len(a)
    c = _convolve_full(a)
    m = len(c)
    K_actual = min(K, m)
    idxs = np.argpartition(c, -K_actual)[-K_actual:]
    cv = c[idxs]
    # Use temperature based on spread
    tau = max(0.01 * cv.max(), 1e-10)
    w = np.exp((cv - cv.max()) / tau)
    w /= w.sum()
    g = np.zeros(n)
    for i in range(K_actual):
        k = int(idxs[i])
        g += w[i] * _grad_k(a, k)
    return g, float(cv.max())


def _subgrad_optimize(a0, budget, seed=0):

    Returns (best_a, best_score).

    else:
        a = a / s

    best = a.copy()
    best_sc = _score_norm(a)
    deadline = time.monotonic() + budget
    n = len(a)

    lr = 1.0 / n
    stagnant = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        g, _ = _smoothed_subgrad(a, K=30)

        # Line search over a few step multipliers
        improved = False
        for f in [1.0, 0.5, 0.25, 0.125, 0.0625, 2.0]:
            alpha = lr * f
            trial = _project_simplex(a - alpha * g)
            sc = _score_norm(trial)
            if sc < best_sc:
                best_sc = sc
                best = trial.copy()
                a = trial
                improved = True
                break

            lr *= 0.9998

                stagnant = 0

def _coord_polish(a0, budget, seed=0):
    """
    Coordinate descent: try moving mass from elements with high gradient
    contribution to elements with low contribution.
    """
    rng = np.random.default_rng(seed)
    a = a0.copy()
    n = len(a)
    best = a.copy()
    best_sc = _score_norm(a)

        c = _convolve_full(a)
        k_star = int(np.argmax(c))
        g = _grad_k(a, k_star)

        # Find best donor (highest g) and best receiver (lowest g, with a>0)
        # Restrict to positions where a > 0 for donors
        mask_active = a > 1e-12
        if mask_active.sum() < 2:
            break

        # Sample a few random pairs for speed
        n_trials = 50
        improved_any = False
        for _ in range(n_trials):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))

                continue

            trial = a.copy()
            trial[i] -= epsilon
            trial[j] += epsilon

            s = trial.sum()

                continue

                best = trial.copy()
                a = trial
                improved_any = True
        if not improved_any:
            # Try smaller epsilon
            epsilon *= 0.5
            if epsilon < 1e-8:
                break

    return best, best_sc


def _make_inits(n, seed=42):
    """Generate diverse initial sequences."""
    rng = np.random.default_rng(seed)
    inits = []
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    # Uniform
    inits.append(np.ones(n) / n)

    # Gaussian (narrow)
    g1 = np.exp(-0.5 * ((x - center) / (n * 0.05)) ** 2)
    inits.append(g1 / g1.sum())

    # Gaussian (medium)
    g2 = np.exp(-0.5 * ((x - center) / (n * 0.15)) ** 2)
    inits.append(g2 / g2.sum())

    # Gaussian (wide)
    g3 = np.exp(-0.5 * ((x - center) / (n * 0.4)) ** 2)
    inits.append(g3 / g3.sum())

    # Cosine window
    cw = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
    cw = np.maximum(cw, 1e-12)
    inits.append(cw / cw.sum())

    # Random smooth
    r = rng.random(n)
    k = np.ones(21) / 21.0
    r_sm = np.convolve(r, k, mode='same')
    r_sm = np.maximum(r_sm, 1e-12)
    inits.append(r_sm / r_sm.sum())

    # Binary 50%
    b = (rng.random(n) < 0.5).astype(np.float64)
    inits.append(b / b.sum())

    # Two blocks
    tb = np.zeros(n)
    tb[:n // 4] = 1.0
    tb[3 * n // 4:] = 1.0
    inits.append(tb / tb.sum())

    return inits
