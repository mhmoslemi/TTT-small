def project_onto_simplex(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.maximum(v, 0)
    if v.sum() < target_sum:
        return v + (target_sum - v.sum()) / len(v)
    # Bisection for the threshold
    lo, hi = 0.0, v.max()
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.maximum(v - mid, 0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2
    return np.maximum(v - mid, 0)
