def solve_lp_direction(f, M, n):
    """
    Solve LP: maximize sum(b) subject to b >= 0 and ||f*b||_∞ <= M.
    Returns the optimal b.
    """
    # Variables: b[0], b[1], ..., b[n-1]
    # Objective: maximize sum(b) -> minimize -sum(b)
    c = -np.ones(n)
    
    # Constraints: (f*b)[k] <= M for k = 0, ..., 2n-2
    # (f*b)[k] = sum_{i: 0<=i<n, 0<=k-i<n} f[i] * b[k-i]

    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2*n-1, n))

            idx = k - j
            if 0 <= idx < n:
                A[k, j] = f[idx]
    
    b_vec = np.full(2*n-1, M)
    
    # Solve LP
    from scipy.optimize import linprog
    result = linprog(c, A_ub=A, b_ub=b_vec, bounds=[(0, None)]*n, method='highs')
    
    if result.success:
        return result.x
    return None

"""
Minimizes the peak self-convolution of a non-negative sequence using
smoothed-max subgradient descent with adaptive parameters and multiple restarts.
"""

import numpy as np
import time
from scipy.fft import fft, ifft


def full_convolve(a):
    """Compute full linear self-convolution using FFT."""
    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    fa = fft(a, size)
    result = ifft(fa * fa, size)
    return np.real(result[:2 * n - 1])


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.all(v <= 0):
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(np.nonzero(cond))[0]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_max_subgradient(a, alpha):
    """Compute subgradient of smoothed max of self-convolution.
    
    Returns (gradient, convolution, max_value).
    """
    n = len(a)
    conv = full_convolve(a)
    
    max_conv = np.max(conv)
    shifted = alpha * (conv - max_conv)
    shifted = np.clip(shifted, -100, 100)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    
    g = 2.0 * np.correlate(w, a, mode='valid')

        lo = max(0, k_star - n + 1)

    fa = fft(a, size)
    result = ifft(fa * fa, size)
    return np.real(result[:2 * n - 1])


def project_simplex(v):
    n = len(v)
    if np.all(v <= 0):
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(np.nonzero(cond))[0]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
