```python
import numpy as np
import time

try:
    import scipy.optimize as scipy_opt
except Exception:
    scipy_opt = None

_FFT_LEN_CACHE = {}


def _fft_len(need: int) -> int:
    need = int(need)
    cached = _FFT_LEN_CACHE.get(need, None)
    if cached is not None:
        return int(cached)
    n = 1
    while n < need:
        n <<= 1
    _FFT_LEN_CACHE[need] = n
    return n


def fft_convolve(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    u = np.asarray(u, dtype=np.float64)
    v = np.asarray(v, dtype=np.float64)
    if u.size == 0 or v.size == 0:
        return np.array([], dtype=np.float64)
    need = int(u.size + v.size - 1)
    n = _fft_len(need)
    fu = np.fft.rfft(u, n)
    fv = np.fft.rfft(v, n)
    out = np.fft.irfft(fu * fv, n)[:need]
    out = np.asarray(out, dtype=np.float64)
    out = np.where(out < 0.0, 0.0, out)
    return out


def convolve_fast(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    u = np.asarray(u, dtype=np.float64)
    v = np.asarray(v, dtype=np.float64)
    if u.size == 0 or v.size == 0:
        return np.array([], dtype=np.float64)
    if int(u.size) * int(v.size) > 1_000_000:
        out = fft_convolve(u, v)
    else:
        out = np.asarray(np.convolve(u, v), dtype=np.float64)
    out = np.where(out < 0.0, 0.0, out)
    return out


def stable_softmax(y: np.ndarray) -> np.ndarray:
    y = np.asarray(y, dtype=np.float64)
    if y.size == 0:
        return np.array([], dtype=np.float64)
    y = np.where(np.isfinite(y), y, 0.0)
    y = y - float(np.max(y))
    z = np.exp(y)
    s = float(z.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full(y.shape, 1.0 / float(y.size), dtype=np.float64)
    return z / s


def normalize_positive(a: np.ndarray) -> np.ndarray:
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1.0e6)
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full(a.shape, 1.0 / float(a.size), dtype=np.float64)
    return a / s


def score_normalized(a: np.ndarray) -> float:
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.inf
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1.0e6)
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.inf
    a = a / s
    c = convolve_fast(a, a)
    if c.size == 0:
        return np.inf
    cm = float(c.max())
    if not np.isfinite(cm):
        return np.inf
    return float(2.0 * a.size * cm)


def grad_conv(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    s1 = convolve_fast(a[::-1], w)
    start = n - 1
    out = np.zeros(n, dtype=np.float64)
    if s1.size >= start + n:
        out[:] = s1[start:start + n]
    else:
        seg = s1[start:]
        k = min(seg.size, n)
        if k > 0:
            out[:k] = seg[:k]
    return 2.0 * out


def objective_y(y: np.ndarray, n: int, beta: float):
    y = np.asarray(y, dtype=np.float64)
    if y.size != n:
        y = np.resize(y, n)
    a = stable_softmax(y)
    c = convolve_fast(a, a)
    if c.size == 0:
        return np.inf, np.zeros(n, dtype=np.float64)
    m = float(c.max())
    if not np.isfinite(m):
        return np.inf, np.zeros(n, dtype=np.float64)
    z = beta * (c - m)
    ez = np.exp(z)
    t = float(ez.sum())
    if t <= 0.0 or not np.isfinite(t):
        w = np.ones(c.size, dtype=np.float64) / float(c.size)
    else:
        w = ez / t
    G = grad_conv(w, a)
    val = (m + float(np.log(t))) / beta
    dot = float(np.dot(a, G))
    grad = a * (G - dot)
    return float(n * val), grad


def try_y_sequence(y: np.ndarray, cur_a: np.ndarray, cur_score: float):
    y = np.asarray(y, dtype=np.float64)
    if y.size == 0:
        return cur_a, cur_score
    a = stable_softmax(y)
    sc = score_normalized(a)
    if np.isfinite(sc) and sc < cur_score:
        return a, sc
    return cur_a, cur_score


def shape_y(n: int, rng: np.random.Generator) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    y = np.zeros(n, dtype=np.float64)
    if rng.random() < 0.45:
        center = float(rng.uniform(0.25, 0.75))
        width = float(rng.uniform(0.12, 0.55))
        height = float(rng.uniform(0.4, 4.0))
        skew = float(rng.uniform(-2.5, 2.5))
        z = (x - center) / width
        y = -height * z * z + skew * (x - center)
    else:
        y += float(rng.uniform(-2.0, 2.0)) * (x - 0.5)
        y += float(rng.uniform(-1.0, 1.0)) * (x - 0.5) * (x - 0.5)
        y += float(rng.uniform(-1.0, 1.0)) * ((x - 0.5) ** 3)
    mmax = max(1, min(8, n // 10))
    for m in range(1, mmax + 1):
        amp = float(rng.normal(0.0, 1.0 / np.sqrt(m)))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        y += amp * np.cos(2.0 * np.pi * m * x + phase)
        amp2 = float(rng.normal(0.0, 1.0 / np.sqrt(m)))
        phase2 = float(rng.uniform(0.0, 2.0 * np.pi))
        y += amp2 * np.sin(2.0 * np.pi * m * x + phase2)
    xe = np.clip(x, 1.0e-4, 1.0 - 1.0e-4)
    y += float(rng.uniform(-1.5, 1.5)) * np.log(xe)
    y += float(rng.uniform(-1.5, 1.5)) * np.log(1.0 - xe)
    y += rng.normal(0.0, 0.05, n)
    y = y - float(np.mean(y))
    return np.clip(y, -40.0, 40.0)


def quick_shape_search(n: int, max_time: float, rng: np.random.Generator, init_a: np.ndarray = None):
    deadline = time.monotonic() + max(0.0, float(max_time))
    a = np.full(n, 1.0 / float(n), dtype=np.float64)
    sc = score_normalized(a)
    if init_a is not None:
        ai = normalize_positive(init_a)
        si = score_normalized(ai)
        if np.isfinite(si) and si < sc:
            a, sc = ai, si

    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    x_safe = np.clip(x, 1.0e-6, 1.0 - 1.0e-6)

    if n <= 1024:
        exp_vals = [-4.0, -2.0, -1.5, -1.0, -0.5, -0.25, 0.25, 0.5, 1.0, 1.5, 2.0, 4.0]
        p_vals = [-1.0, -0.5, 0.0, 0.5, 1.0, 2.0, 4.0]
        q_vals = [-1.0, -0.5, 0.0, 0.5, 1.0, 2.0, 4.0]
        centers = [0.30, 0.40, 0.50, 0.60, 0.70]
        widths = [0.15, 0.30, 0.50]
        skews = [-2.0, -1.0, 0.0, 1.0, 2.0]
        max_samples = 90
    else:
        exp_vals = [-3.0, -1.5, -0.75, -0.25, 0.25, 0.75, 1.5, 3.0]
        p_vals = [-0.5, 0.0, 0.5, 1.0, 2.0]
        q_vals = [-0.5, 0.0, 0.5, 1.0, 2.0]
        centers = [0.35, 0.50, 0.65]
        widths = [0.20, 0.40]
        skews = [-1.0, 0.0, 1.0]
        max_samples = 25

    for sv in exp_vals:
        if time.monotonic() > deadline:
            break
        y = sv * (x - 0.5)
        a, sc = try_y_sequence(y, a, sc)

    for p in p_vals:
        if time.monotonic() > deadline:
            break
        for q in q_vals:
            if time.monotonic() > deadline:
                break
            if p == q:
                continue
            y = p * np.log(x_safe) + q * np.log(1.0 - x_safe)
            a, sc = try_y_sequence(y, a, sc)

    for center in centers:
        if time.monotonic() > deadline:
            break
        for width in widths:
            if time.monotonic() > deadline:
                break
            for skew in skews:
                if time.monotonic() > deadline:
                    break
                if center == 0.5 and skew == 0.0:
                    continue
                z = (x - center) / width
                y = -1.5 * z * z + skew * (x - center)
                a, sc = try_y_sequence(y, a, sc)

    samples = 0
    while samples < max_samples and time.monotonic() < deadline:
        y = shape_y(n, rng)
        a, sc = try_y_sequence(y, a, sc)
        samples += 1

    return a, float(sc)


def mirror_optimize(a: np.ndarray, max_time: float, rng: np.random.Generator) -> np.ndarray:
    a = normalize_positive(a)
    n = int(a.size)
    if n <= 1 or max_time <= 0.0:
        return a
    deadline = time.monotonic() + max(0.0, float(max_time))
    y = np.log(a + 1.0e-300)
    y = np.where(np.isfinite(y), y, 0.0)
    y = y - float(np.mean(y))
    y = np.clip(y, -40.0, 40.0)

    best_a = a.copy()
    best_score = score_normalized(a)
    alpha = 0.5
    fails = 0
    it = 0
    verbose = n <= 200

    while time.monotonic() < deadline:
        it += 1
        a = stable_softmax(y)
        c = convolve_fast(a, a)
        if c.size == 0:
            break
        m = float(c.max())
        if not np.isfinite(m):
            break
        cur_score = float(2.0 * n * m)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        beta = min(1.0e6, max(10.0, 50.0 / max(m, 1.0e-8)))
        z = beta * (c - m)
        ez = np.exp(z)
        t = float(ez.sum())
        if t <= 0.0 or not np.isfinite(t):
            w = np.ones(c.size, dtype=np.float64) / float(c.size)
        else:
            w = ez / t

        G = grad_conv(w, a)
        d = G - float(np.mean(G))
        sd = float(np.std(d))
        if sd < 1.0e-14:
            break
        d = d / sd

        if n <= 200:
            alphas = [alpha * 0.5, alpha, alpha * 2.0]
        elif n <= 2000:
            alphas = [alpha, alpha * 1.5]
        else:
            alphas = [alpha]

        best_trial_score = np.inf
        best_trial_y = None
        for al in alphas:
            y_trial = y - al * d
            y_trial = np.where(np.isfinite(y_trial), y_trial, 0.0)
            y_trial = y_trial - float(np.mean(y_trial))
            y_trial = np.clip(y_trial, -40.0, 40.0)
            v = stable_softmax(y_trial)
            s_trial = score_normalized(v)
            if np.isfinite(s_trial) and s_trial < best_trial_score:
                best_trial_score = s_trial
                best_trial_y = y_trial

        if best_trial_score < best_score - 1.0e-12 and best_trial_y is not None:
            best_score = best_trial_score
            y = best_trial_y
            best_a = stable_softmax(y)
            alpha *= 1.05
            if alpha > 5.0:
                alpha = 5.0
            fails = 0
        else:
            alpha *= 0.8
            if alpha < 1.0e-3:
                alpha = 1.0e-3
            fails += 1
            if fails > 60:
                if rng.random() < 0.35:
                    y = shape_y(n, rng)
                else:
                    y = np.log(best_a + 1.0e-300)
                    y = np.where(np.isfinite(y), y, 0.0)
                y = y - float(np.mean(y))
                y = np.clip(y, -40.0, 40.0)
                fails = 0
            elif fails > 12:
                scale = 0.05 if n <= 500 else 0.02
                y = y + rng.normal(0.0, scale, n)
                y = y - float(np.mean(y))
                y = np.clip(y, -40.0, 40.0)
                fails = 0

        if verbose and it % 700 == 0:
            print(f"mirror n={n} it={it} best={best_score:.6f}")

    return best_a


def lbfgs_polish_y(y: np.ndarray, n: int, beta: float, max_time: float, rng: np.random.Generator) -> np.ndarray:
    y = np.asarray(y, dtype=np.float64).ravel()
    if y.size != n:
        y = np.resize(y, n)
    y = np.where(np.isfinite(y), y, 0.0)
    y = y - float(np.mean(y))
    y = np.clip(y, -40.0, 40.0)

    deadline = time.monotonic() + max(0.0, float(max_time))
    best_a = stable_softmax(y)
    best_score = score_normalized(best_a)

    while time.monotonic() < deadline:
        try:
            res = scipy_opt.minimize(
                objective_y,
                y,
                jac=True,
                args=(n, beta),
                method="L-BFGS-B",
                options={"maxiter": 40, "ftol": 1.0e-12, "gtol": 1.0e-6},
            )
            if res.x is not None:
                y = np.asarray(res.x, dtype=np.float64)
        except Exception:
            break

        y = np.where(np.isfinite(y), y, 0.0)
        y = y - float(np.mean(y))
        y = np.clip(y, -40.0, 40.0)
        a = stable_softmax(y)
        sc = score_normalized(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            y = np.log(best_a + 1.0e-300)
            y = np.where(np.isfinite(y), y, 0.0)
            y = y - float(np.mean(y))
            y = np.clip(y, -40.0, 40.0)

        if time.monotonic() > deadline:
            break
        y = y + rng.normal(0.0, 0.10, n)

    return best_a


def polish_with_lbfgs(a: np.ndarray, max_time: float, rng: np.random.Generator) -> np.ndarray:
    a = normalize_positive(a)
    if scipy_opt is None or max_time <= 0.1:
        return a
    n = int(a.size)
    c = convolve_fast(a, a)
    m = float(c.max()) if c.size else 1.0e-3
    if not np.isfinite(m) or m <= 0.0:
        m = 1.0e-3
    beta = min(1.0e6, max(10.0, 50.0 / max(m, 1.0e-8)))
    y = np.log(a + 1.0e-300)
    y = np.where(np.isfinite(y), y, 0.0)
    y = y - float(np.mean(y))
    y = np.clip(y, -40.0, 40.0)
    return lbfgs_polish_y(y, n, beta, max_time, rng)


def refine_sequence(n: int, a: np.ndarray, max_time: float, rng: np.random.Generator):
    t0 = time.monotonic()
    a = normalize_positive(a)
    sc = score_normalized(a)
    if max_time <= 0.05:
        return a, sc
    mirror_time = max(0.05, max_time * 0.85)
    a2 = mirror_optimize(a, mirror_time, rng)
    sc2 = score_normalized(a2)
    if np.isfinite(sc2) and sc2 < sc:
        a, sc = a2, sc2
    rem = float(max_time) - (time.monotonic() - t0)
    if rem > 8.0 and n <= 600 and sc < 1.70:
        lb_time = min(30.0, max(1.0, rem * 0.45))
        a3 = polish_with_lbfgs(a, lb_time, rng)
        sc3 = score_normalized(a3)
        if np.isfinite(sc3) and sc3 < sc:
            a, sc = a3, sc3
    return a, sc


def resize_sequence(a: np.ndarray, n: int) -> np.ndarray:
    a = normalize_positive(a)
    m = int(a.size)
    n = int(n)
    if m == n:
        return a
    x_old = np.linspace(0.0, 1.0, m, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, n, dtype=np.float64)
    vals = np.log(a + 1.0e-15)
    best = np.interp(x_new, x_old, vals)
    best = np.exp(best - float(np.max(best))))
    s = float(best.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    return best / s


def cand_key(item):
    return item[0]


def propose_candidate(seed: int = 42, budget_s: float = 1800.0):
    """Search nonnegative step-function heights with mirror descent, bounded
    deterministic/random smooth initializations, occasional L-BFGS polish, and
    length transfer.  Returns the best sequence found before the time budget.
    """
    budget_s = max(10.0, float(budget_s))
    deadline = time.monotonic() + max(5.0, budget_s - 10.0)
    rng = np.random.default_rng(int(seed))

    init_seq = None
    try:
        init_seq = height_sequence_1  # noqa: F821
    except Exception:
        init_seq = None
    init_array = None
    init_len = 0
    if init_seq is not None:
        tmp = np.asarray(init_seq, dtype=np.float64).ravel()
        tmp = tmp[np.isfinite(tmp)]
        if tmp.size > 0:
            init_array = np.clip(tmp, 0.0, 1.0e6)
            init_len = int(init_array.size)

    best_a = np.ones(4, dtype=np.float64)
    best_a /= 4.0
    best_score = score_normalized(best_a)

    if init_array is not None:
        ia = normalize_positive(init_array)
        iscore = score_normalized(ia)
        if np.isfinite(iscore) and iscore < best_score:
            best_a, best_score = ia, iscore
            print(f"initial height_sequence_1 len={init_len} score={best_score:.6f}")

    base_lengths = [
        3, 4, 5, 6, 8, 10, 12, 16, 20, 25, 32, 40, 50, 64, 80, 100,
        128, 160, 200, 256, 320, 400, 512, 640, 800, 1024, 1280,
        1600, 2048, 2560, 3200, 4096, 4806
    ]
    if 3 <= init_len <= 6000:
        base_lengths.append(init_len)
    lengths = sorted(set(int(x) for x in base_lengths))

    phase1_deadline = min(deadline, time.monotonic() + 180.0)
    candidates = []
    for n in lengths:
        if time.monotonic() >= phase1_deadline:
            break
        if time.monotonic() >= deadline:
            break
        if n <= 20:
            qt = 0.18
        elif n <= 200:
            qt = 0.35
        elif n <= 2048:
            qt = 0.70
        else:
            qt = 1.00
        init_a_for_n = None
        if init_array is not None and init_array.size <= 100000:
            init_a_for_n = resize_sequence(init_array, n)
        a, sc = quick_shape_search(n, qt, rng, init_a_for_n)
        if np.isfinite(sc):
            candidates.append((float(sc), int(n), a))
            if sc < best_score:
                best_score = sc
                best_a = a
    candidates = [c for c in candidates if np.isfinite(c[0])]
    candidates.sort(key=cand_key)

    if len(candidates) > 0:
        print(f"phase1 best score={best_score:.6f} n={best_a.size}")

    sel = candidates[:12]
    for item in sel:
        if time.monotonic() >= deadline:
            break
        sc, n, a = item
        rem = deadline - time.monotonic()
        if n <= 100:
            tt = max(6.0, min(80.0, rem * 0.25))
        elif n <= 1000:
            tt = max(12.0, min(120.0, rem * 0.20))
        else:
            tt = max(30.0, min(180.0, rem * 0.18))
        if abs(sc - best_score) <= 1.0e-7:
            tt = max(tt, min(240.0, rem * 0.40))
        if rem < 2.0:
            break
        tt = min(tt, max(0.5, rem * 0.85))
        a2, sc2 = refine_sequence(n, a, tt, rng)
        if np.isfinite(sc2) and sc2 < best_score:
            best_score = sc2
            best_a = a2
            print(f"refined n={n} score={best_score:.6f}")

    tried_lengths = set()
    tried_lengths.add(int(best_a.size))
    expansion_start = time.monotonic()
    expansion_deadline = deadline - 80.0
    while time.monotonic() < expansion_deadline:
        rem = deadline - time.monotonic()
        if rem < 60.0:
            break
        best_len = int(best_a.size)
        options = []
        if best_len > 0 and best_len < 6000:
            options.append(min(6000, int(best_len * 1.25) + 1))
            options.append(min(6000, best_len + max(1, best_len // 5)))
        if best_len > 10:
            options.append(max(3, int(best_len * 0.75)))
        if 4806 not in options:
            options.append(4806)
        chosen = None
        for nn in options:
            nn = int(nn)
            if 3 <= nn <= 6000 and nn not in tried_lengths:
                chosen = nn
                break
        if chosen is None:
            break
        tried_lengths.add(chosen)
        tt = min(120.0, max(20.0, rem * 0.15))
        init_a = resize_sequence(best_a, chosen)
        a2 = mirror_optimize(init_a, tt, rng)
        sc2 = score_normalized(a2)
        if np.isfinite(sc2) and sc2 < best_score:
            best_score = sc2
            best_a = a2
            print(f"length transfer n={chosen} score={best_score:.6f}")
        elif time.monotonic() - expansion_start > 360.0 and best_score < 1.80:
            break

    if best_a.size <= 600 and time.monotonic() < deadline - 30.0:
        lb_time = min(60.0, max(5.0, (deadline - time.monotonic()) * 0.15))
        a2 = polish_with_lbfgs(best_a, lb_time, rng)
        sc2 = score_normalized(a2)
        if np.isfinite(sc2) and sc2 < best_score:
            best_score = sc2
            best_a = a2

    restarts = 0
    while time.monotonic() < deadline - 8.0 and best_score < 1.90:
        restarts += 1
        if restarts > 10:
            break
        scale = 0.02 / max(1, int(best_a.size))
        pert = normalize_positive(best_a + rng.normal(0.0, scale, best_a.size))
        tt = min(30.0, max(5.0, (deadline - time.monotonic()) * 0.15))
        a2 = mirror_optimize(pert, tt, rng)
        sc2 = score_normalized(a2)
        if np.isfinite(sc2) and sc2 < best_score:
            best_score = sc2
            best_a = a2
            print(f"restart polish score={best_score:.6f}")
        else:
            break

    best_a = normalize_positive(best_a)
    final_list = [float(v) for v in best_a]
    print(f"final n={len(final_list)} local_score={score_normalized(best_a):.6f}")
    try:
        ev = evaluate_sequence(final_list)  # noqa: F821
        print(f"final evaluator score={ev}")
    except Exception:
        pass
    return final_list
```
