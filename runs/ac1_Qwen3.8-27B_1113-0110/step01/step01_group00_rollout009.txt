import numpy as np

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]

def _score_fast(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    c = _fft_convolve(a)
    return 2.0 * n * float(np.max(c)) / (s * s)

def _interp(src, n_target):
    n_src = len(src)
    if n_src == n_target:
        return src.copy()
    xs = np.linspace(0, 1, n_src)
    xt = np.linspace(0, 1, n_target)
    out = np.interp(xt, xs, src)
    out = np.maximum(out, 0.0)
    s = float(np.sum(out))
    if s > 1e-10:
        out /= s
    else:
        out = np.ones(n_target) / n_target
    return out

    n = len(a_init)
    a = a_init.copy().astype(np.float64)

    if S < 1e-10:
        a = np.ones(n)/n
        S = 1.0
    a /= S
    best_a = a.copy()

    rng = np.random.default_rng(seed)
    stuck_count = 0
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        conv = _fft_convolve(a)
        conv_max = float(np.max(conv))
        
        if conv_max < best_m - 1e-12:

            stuck_count = 0
        else:
            stuck_count += 1
        
        # Restart if stuck
        if stuck_count > restart_every:
            a = best_a + noise_scale * rng.standard_normal(n)
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a /= s
            else:
                a = best_a.copy()
            stuck_count = 0
            conv = _fft_convolve(a)
        
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break
            ai = a[i]
            k2i = 2 * i
            cl = len(conv)
            
            # Sensitivity vector
            shift = np.arange(cl) - i
            mask = (shift >= 0) & (shift < n)
            w = np.zeros(cl)
            w[mask] = 2.0 * a[shift[mask]]
            has_quad = (0 <= k2i < cl)
            if has_quad:
                w[k2i] = 0.0
            
            # Ternary search
            dlo = -ai
            dhi = 5.0 / n
            for _ in range(30):
                d1 = dlo + (dhi - dlo) / 3.0
                d2 = dhi - (dhi - dlo) / 3.0
                # f(d1)
                v1 = conv + d1 * w
                if has_quad:
                    v1[k2i] = conv[k2i] + 2.0*ai*d1 + d1*d1
                f1 = float(np.max(v1))
                # f(d2)
                v2 = conv + d2 * w
                if has_quad:
                    v2[k2i] = conv[k2i] + 2.0*ai*d2 + d2*d2
                f2 = float(np.max(v2))
                if f1 < f2:
                    dhi = d2
                else:
                    dlo = d1
            
            delta = (dlo + dhi) / 2.0
            if abs(delta) > 1e-14:
                new_ai = max(0.0, ai + delta)
                a[i] = new_ai
                conv[mask] += delta * w[mask]
                if has_quad:
                    conv[k2i] += 2.0 * ai * delta + delta * delta
        
        # Renormalize
        Sn = float(np.sum(a))
        if Sn > 1e-10 and abs(Sn - 1.0) > 1e-6:
            a /= Sn
            conv = _fft_convolve(a)
        
        if sweep % 300 == 0 and sweep > 0:
            print(f"      sweep {sweep}: score~{2*n*float(np.max(conv)):.6f}")
    
    conv = _fft_convolve(best_a)
    S = float(np.sum(best_a))
    score = 2.0 * n * float(np.max(conv)) / (S * S)
    return best_a, score

Minimize J(a) = 2n*max(conv(a,a))/sum(a)^2 using coordinate descent
with in-place convolution updates, subgradient polish, multi-start,
and block-coordinate refinement.
"""
import numpy as np
import time


def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]


def _score_fast(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    c = _fft_convolve(a)
    return 2.0 * n * float(np.max(c)) / (s * s)


def _interp(src, n_target):
    n_src = len(src)
    if n_src == n_target:
        return src.copy()
    xs = np.linspace(0, 1, n_src)
    xt = np.linspace(0, 1, n_target)
    out = np.interp(xt, xs, src)
    out = np.maximum(out, 0.0)
    s = float(np.sum(out))
    if s > 1e-10:
        out /= s
    else:
        out = np.ones(n_target) / n_target
    return out
