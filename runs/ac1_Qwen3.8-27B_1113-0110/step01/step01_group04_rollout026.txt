def _pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _clean(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    return np.maximum(v, 0.0)


def _normalize(v):
    v = _clean(v)
    s = float(v.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.full(v.size, 1.0 / max(1, v.size))
    return v / s


def _project_simplex(v):
    v = _clean(v)
    n = v.size
    if n == 0:
        return v.copy()
    x = np.sort(v)[::-1]
    css = np.cumsum(x)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = x > (css - 1.0) / idx
    if not np.any(cond):
        return np.full(n, 1.0 / float(n))
    r = int(np.nonzero(cond)[0][-1])
    tau = (css[r] - 1.0) / float(r + 1)
    out = np.maximum(v - tau, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.full(n, 1.0 / float(n))
    return out / s


def _score(a, L=None):

    n = v.size
    if n == 0:
        return np.inf
    s = float(v.sum())
    if s < 1e-12:

    if L is None:

    A = np.fft.rfft(v, n=L)

    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    mx = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(mx):

    return 2.0 * float(n) * mx

    cmax = float(c.max())
    if cmax <= 1e-300:
        return 0.0, np.zeros_like(x)
    z = (c - cmax) / eps
    z = np.clip(z, -745.0, 0.0)
    e = np.exp(z)
    sume = float(e.sum())
    if not np.isfinite(sume) or sume <= 0:
        return 1e300, np.zeros_like(x)
    f = cmax + eps * float(np.log(sume))
    w = e / sume
    W = np.fft.rfft(w, n=L)
    ga = 2.0 * np.fft.irfft(W * np.conj(A), n=L)[:n]
    dot = float(np.dot(a, ga))
    gfull = a * (ga - dot)
    gy = gfull[:-1]
    if not np.all(np.isfinite(gy)):
        gy = np.zeros_like(x)
    return f, gy


def _exact_subgrad_search(a0, end_time, seed, momentum=0.9):
    """Exact subgradient descent on the max with Nesterov momentum."""
    a = _project_simplex(a0)
    n = a.size
    L = _pow2(2 * n - 1)
    Lconv = _pow2(3 * n - 2)

    best = a.copy()
    best_score = _score(best, L)
    rng = np.random.default_rng(seed)

    v = np.zeros(n)  # momentum velocity
    alpha = 0.5
    last_improve = time.monotonic()
    start = time.monotonic()

    while time.monotonic() < end_time:
        c = _convolve_fft(a, Lconv)
        cmax = float(c.max())
        score = 2.0 * float(n) * cmax / (float(a.sum()) ** 2)

        if score < best_score - 1e-12:
            best_score = score
            best = a.copy()
            last_improve = time.monotonic()
            alpha = min(2.0, alpha * 1.02)

        elapsed = time.monotonic() - start
        if time.monotonic() - last_improve > 8.0:
            alpha = max(0.01, alpha * 0.8)
            if time.monotonic() - last_improve > 20.0 and elapsed > 30.0:
                a = _project_simplex(best * np.exp(rng.normal(0, 0.05, n)))
                v.fill(0)
                continue

        # Get subgradient: position(s) of max in c
        k_star = int(np.argmax(c))

        # Gradient of (a*a)_{k_star} w.r.t. a:
        # d/d a_i [(a*a)_{k_star}] = 2 * a_{k_star - i} (for valid indices)
        # with the convention that if i == k_star - i, it's a_i (not 2*a_i)
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                if i == j:
                    grad[i] = a[i]  # derivative of a_i^2 is 2*a_i, but we have one term
                else:
                    grad[i] = 2.0 * a[j]

        # Actually let me redo this properly:
        # (a*a)_k = sum_i a_i * a_{k-i}
        # d/d a_i [(a*a)_k] = a_{k-i} + a_{k-i} = 2*a_{k-i} (if i != k-i)
        #                              = 2*a_i (if i == k-i, i.e. k = 2i)
        # Wait: if i = k-i, then the term is a_i * a_i = a_i^2, derivative is 2*a_i.
        # If i != k-i, there are two terms: a_i*a_{k-i} (from j=i) and a_{k-i}*a_i (from j=k-i).
        # So d/d a_i = a_{k-i} + a_{k-i} = 2*a_{k-i}.
        # In both cases it's 2*a_{k-i} (where k-i might equal i).
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]

        # For the case i == j (k_star = 2*i): the term a_i*a_i appears only once
        # in the sum, so its derivative is 2*a_i. But 2*a_{k-i} = 2*a_i. Same thing!
        # So the formula grad[i] = 2*a[k_star - i] is correct for all valid i.

        # Project gradient onto tangent space of simplex (subtract mean component)
        grad_dot_a = float(np.dot(grad, a))
        grad_tangent = grad - (grad_dot_a / float(n))  # project onto sum=0 hyperplane
        # Actually for simplex, the projection is: g_proj = g - (sum(g*a)/sum(a)) * ones
        # But since we're on sum(a)=1, the tangent space is {h: sum(h)=0}
        # Project: g - mean(g) * ones... no, the correct projection for the simplex
        # with the constraint sum(a)=1 is: g_t = g - (sum_i g_i * a_i / sum_i a_i) * ...
        # Actually for L2 projection onto {h: sum(h) = 0}: h - mean(h)*1
        # But for the logistic/simplex parameterization, it's different.
        # Let's just use the simple: subtract the weighted mean
        g_mean = float(np.mean(grad))
        grad_flat = grad - g_mean

        # Nesterov momentum step
        v = momentum * v - alpha * grad_flat / (float(np.linalg.norm(grad_flat)) + 1e-12)
        a_new = _project_simplex(a + v)

        if not np.all(np.isfinite(a_new)):
            a = best.copy()
            v.fill(0)
        else:
            a = a_new

    return best, best_score


def _peak_shaving_search(a0, end_time, seed):
    """Directly target the top peaks by moving mass from contributing pairs."""
    a = _project_simplex(a0)
    n = a.size
    L = _pow2(2 * n - 1)
    Lconv = _pow2(3 * n - 2)

    best = a.copy()
    best_score = _score(best, L)
    rng = np.random.default_rng(seed)
    last_improve = time.monotonic()

    while time.monotonic() < end_time:
        c = _convolve_fft(a, Lconv)
        csum = float(a.sum())
        score = 2.0 * float(n) * float(c.max()) / (csum * csum)

        if score < best_score - 1e-12:
            best_score = score
            best = a.copy()
            last_improve = time.monotonic()

        if time.monotonic() - last_improve > 15.0:
            break

        # Find top peaks
        top_k = min(5, 2 * n - 1)
        c_sorted = np.sort(c)[::-1]
        threshold = c_sorted[min(top_k - 1, len(c_sorted) - 1)] * 0.95
        hot_positions = np.where(c >= threshold)[0]

        if len(hot_positions) == 0:
            break

        # For each hot position, find the "hot pairs"
        # Try to reduce the peak by moving mass
        improved = False
        for kp in hot_positions[:3]:
            # Contribution: c[kp] = sum_i a[i] * a[kp - i]
            # Find the i with largest a[i] * a[kp-i]
            contribs = np.zeros(n)
            for i in range(n):
                j = int(kp) - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]

            # The top contributor is at index i*
            i_star = int(np.argmax(contribs))
            j_star = int(kp) - i_star
            if j_star < 0 or j_star >= n:
                continue

            # Try moving a small amount of mass from i_star (or j_star) to a "cold" position
            # Cold positions: those that don't contribute to any hot peak
            cold_mask = np.ones(n, dtype=bool)
            for kp2 in hot_positions[:5]:
                for i2 in range(n):
                    j2 = int(kp2) - i2
                    if 0 <= j2 < n:
                        if a[i2] * a[j2] > 0.1 * float(c.max()):
                            cold_mask[i2] = False
                            cold_mask[j2] = False

            if not cold_mask.any():
                cold_mask = np.ones(n, dtype=bool)
                cold_mask[:n//4] = False
                cold_mask[3*n//4:] = False

            cold_indices = np.where(cold_mask)[0]
            if len(cold_indices) == 0:
                continue

            # Move a fraction of mass
            frac = 0.05 + 0.1 * rng.random()
            source = i_star if a[i_star] <= a[j_star] else j_star
            target = cold_indices[rng.integers(0, len(cold_indices))]

            a_trial = a.copy()
            delta = frac * a[source]
            a_trial[source] -= delta
            a_trial[target] += delta
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = _project_simplex(a_trial)

            sc_trial = _score(a_trial, L)
            if np.isfinite(sc_trial) and sc_trial < best_score - 1e-10:
                a = a_trial
                best = a.copy()
                best_score = sc_trial
                last_improve = time.monotonic()
                improved = True

        if not improved and rng.random() < 0.1:
            # Random perturbation to escape local minimum
            a_trial = _project_simplex(a * np.exp(rng.normal(0, 0.03, n)))
            sc_trial = _score(a_trial, L)
            if np.isfinite(sc_trial) and sc_trial < best_score:
                a = a_trial
                best = a.copy()
                best_score = sc_trial
                last_improve = time.monotonic()

    return best, best_score


def _coordinate_descent(a0, end_time, seed):
    """Cycle through coordinates, optimizing each one (1D problem)."""
    a = _project_simplex(a0)
    n = a.size
    L = _pow2(2 * n - 1)
    Lconv = _pow2(3 * n - 2)

    best = a.copy()
    best_score = _score(best, L)
    rng = np.random.default_rng(seed)
    last_improve = time.monotonic()
    order = list(range(n))

    while time.monotonic() < end_time:
        rng.shuffle(order)
        for i in order:
            if time.monotonic() > end_time:
                break

            # Save current value
            a_i = a[i]
            if a_i <= 1e-15:
                # Try setting it to a small positive value
                for trial in [0.5 / n, 1.0 / n, 2.0 / n]:
                    a_trial = a.copy()
                    a_trial[i] = trial
                    a_trial = _project_simplex(a_trial)
                    sc = _score(a_trial, L)
                    if np.isfinite(sc) and sc < best_score - 1e-10:
                        a = a_trial
                        best = a.copy()
                        best_score = sc
                        last_improve = time.monotonic()
                        break
                continue

            # For coordinate i, find the best value given others fixed.
            # We need: (a*a)_k <= t for all k, minimize t.
            # (a*a)_k depends linearly (or quadratically if k=2i) on a_i.
            # Let's do a golden-section search over a_i in [0, 3*current].
            a_rest_sum = float(a.sum()) - a_i

            def _score_with_ai(val_i):
                a_trial = a.copy()
                a_trial[i] = max(0.0, val_i)
                # Renormalize to keep sum = 1 (adjust i only is not quite right)
                # Better: scale all other elements to maintain sum = 1
                s = val_i + a_rest_sum
                if s <= 0:
                    return np.inf
                a_trial = a.copy()
                a_trial[i] = val_i
                a_trial = _project_simplex(a_trial)
                return _score(a_trial, L)

            # Golden section search
            lo = 0.0
            hi = max(3.0 * a_i, 5.0 / n)
            gr = 0.618
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = _score_with_ai(c1)
            f2 = _score_with_ai(c2)

            for _ in range(20):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    f1 = _score_with_ai(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    f2 = _score_with_ai(c2)

            best_val = (lo + hi) / 2.0
            a_trial = a.copy()
            a_trial[i] = best_val
            a_trial = _project_simplex(a_trial)
            sc = _score(a_trial, L)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                a = a_trial
                best = a.copy()
                best_score = sc
                last_improve = time.monotonic()

        if time.monotonic() - last_improve > 30.0:
            break

    return best, best_score


def _lbfgs_polish(a0, end_time, seed):
    """L-BFGS-B polish on log-sum-exp objective with adaptive temperature."""
    if _minimize is None:
        _exact_subgrad_search(a0, end_time, seed)
        return

    a = _project_simplex(a0)
    n = a.size
    L = _pow2(2 * n - 1)

    best = a.copy()
    best_score = _score(best, L)

    x = _ratios_from_a(a)
    bounds = [(-80.0, 80.0)] * max(1, n - 1)

    # Multi-temperature schedule
    c = _convolve_fft(a, _pow2(3 * n - 2))
    eps0 = 0.05 * float(c.max())
    eps = eps0
    eps_min = 1e-7 * float(c.max())

    last_improve = time.monotonic()
    chunk_count = 0

    while time.monotonic() < end_time:
        remaining = end_time - time.monotonic()
        if remaining < 2.0:
            break
        chunk_time = min(remaining, max(5.0, 0.15 * remaining))

        def _obj_and_grad(x_inner):
            f, g = _soft_obj_grad(x_inner, eps)
            return f, g

        maxiter = max(20, int(chunk_time * 3))
        options = {
            "maxiter": maxiter,
            "disp": 0,
            "maxfun": max(100, 6 * maxiter),
            "ftol": 1e-20,
            "gtol": 1e-14,
        }

        try:
            res = _minimize(_obj_and_grad, x, method="L-BFGS-B", jac=True,
                           bounds=bounds, options=options)
            x = np.clip(np.asarray(res.x, dtype=np.float64), -80.0, 80.0)
        except Exception:
            pass

        a_new = _from_ratios(x)
        a_new = _project_simplex(a_new)
        sc = _score(a_new, L)

        chunk_count += 1
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best = a_new.copy()
            last_improve = time.monotonic()
            eps = max(eps_min, eps * 0.6)
        else:
            # No improvement: reduce eps more aggressively or restart
            eps = max(eps_min, eps * 0.7)
            if chunk_count > 3 and eps <= eps_min * 1.1:
                # Add small noise and try again
                x = np.clip(x + np.random.default_rng(chunk_count * 7).normal(0, 0.02, x.size), -80, 80)
                eps = max(eps_min, eps * 0.5)
            if chunk_count > 8 and eps <= eps_min * 1.05:
                break

    return best, best_score


def _refine_single(a0, end_time, seed):
    """Full refinement pipeline for one starting point."""
    a = _project_simplex(a0)
    n = a.size
    L = _pow2(2 * n - 1)
    best = a.copy()
    best_score = _score(best, L)

    t_start = time.monotonic()
    total = max(1.0, end_time - t_start)
    if total < 5.0:
        return best, best_score

    print(f"  [refine] starting score = {best_score:.6f}, budget = {total:.1f}s")

    # Phase 1: L-BFGS polish (50% of time)
    t1_end = t_start + 0.50 * total
    if time.monotonic() < t1_end:
        best, best_score = _lbfgs_polish(best, t1_end, seed)
        print(f"  [phase1 L-BFGS] score = {best_score:.6f}, time = {time.monotonic()-t_start:.1f}s")

    # Phase 2: Exact subgradient (25% of time)
    t2_end = t_start + 0.75 * total
    if time.monotonic() < t2_end:
        best, best_score = _exact_subgrad_search(best, t2_end, seed + 100)
        print(f"  [phase2 subgrad] score = {best_score:.6f}, time = {time.monotonic()-t_start:.1f}s")

    # Phase 3: Peak shaving (15% of time)
    t3_end = t_start + 0.90 * total
    if time.monotonic() < t3_end:
        best, best_score = _peak_shaving_search(best, t3_end, seed + 200)
        print(f"  [phase3 peak-shave] score = {best_score:.6f}, time = {time.monotonic()-t_start:.1f}s")

    # Phase 4: Final L-BFGS + coordinate descent (10% of time)
    if time.monotonic() < end_time:
        t_split = time.monotonic() + 0.5 * (end_time - time.monotonic())
        best, best_score = _lbfgs_polish(best, t_split, seed + 300)
        if time.monotonic() < end_time:
            best, best_score = _coordinate_descent(best, end_time, seed + 400)

    print(f"  [final] score = {best_score:.6f}")
    return best, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Return the best non-negative sequence found within the time budget,
    minimizing 2*n*max(convolve(a,a))/(sum(a))^2.
    """
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 15.0)

    n = 3809
    try:
        h = height_sequence_1
        if isinstance(h, (list, np.ndarray)):
            arr = np.asarray(h, dtype=np.float64)
            if arr.ndim == 1 and 100 <= arr.size <= 20000:
                n = int(arr.size)
    except Exception:
        pass
    if n < 32:
        n = 3809

    print(f"Starting search with n = {n}, budget = {budget_s}s")
    print(f"Deadline: {deadline - t0:.1f}s of effective time")

    # Generate and score candidates
    cands = _generate_candidates(n)
    print(f"Generated {len(cands)} initial candidates")
    scored = _score_candidates(cands)
    if not scored:
        return [1.0 / float(n)] * n

    print(f"Best initial candidate score: {scored[0][0]:.6f}")
    for i, (sc, idx, a) in enumerate(scored[:5]):
        print(f"  Candidate {i}: score = {sc:.6f}")

    global_best_a = scored[0][2].copy()
    global_best_score = scored[0][0]

    # Allocate time to top candidates
    # Top 1 gets 45%, top 2 gets 25%, top 3 gets 15%, remaining 15% for final polish
    allocs = [
        (0.45, 0),
        (0.25, 1),
        (0.15, 2),
    ]

    used_frac = 0.0
    for frac, idx in allocs:
        if idx >= len(scored):
            break
        alloc_start = t0 + used_frac * (deadline - t0)
        alloc_end = t0 + (used_frac + frac) * (deadline - t0)
        alloc_end = min(alloc_end, deadline)

        if alloc_end - alloc_start < 10.0:
            break

        print(f"\n=== Refining candidate {idx} (score={scored[idx][0]:.6f}), time={alloc_end-alloc_start:.0f}s ===")
        a_start = scored[idx][2].copy()
        a_out, sc_out = _refine_single(a_start, alloc_end, int(seed) + 1000 * (idx + 1))

        if np.isfinite(sc_out) and sc_out < global_best_score:
            global_best_score = sc_out
            global_best_a = a_out.copy()
            print(f"  New global best: {global_best_score:.6f}")

        used_frac += frac

    # Final polish on the global best
    final_start = max(time.monotonic(), t0 + used_frac * (deadline - t0))
    if deadline - final_start > 5.0:
        print(f"\n=== Final polish, {deadline - final_start:.0f}s ===")
        # Quick L-BFGS
        t_split = final_start + 0.5 * (deadline - final_start)
        global_best_a, global_best_score = _lbfgs_polish(global_best_a, t_split, int(seed) + 9999)
        # Quick peak shaving
        if time.monotonic() < deadline:
            global_best_a, global_best_score = _peak_shaving_search(global_best_a, deadline, int(seed) + 9998)

    global_best_a = _project_simplex(global_best_a)
    out = [float(max(0.0, min(1000.0, x))) for x in global_best_a]

    # Validate
    try:
        val = evaluate_sequence(out)
        if not np.isfinite(val):
            print(f"WARNING: final score not finite ({val}), using fallback")
            out = [float(x) for x in _raised_cosine(len(out), 1.0)]
        else:
            print(f"\nFinal validated score: {val:.6f}")
    except Exception as e:
        print(f"Validation error: {e}")

    elapsed = time.monotonic() - t0
    print(f"Total time: {elapsed:.1f}s")
    return out
