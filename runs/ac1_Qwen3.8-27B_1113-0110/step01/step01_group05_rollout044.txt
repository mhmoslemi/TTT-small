Minimizer for 2*n*max(conv(a,a))/(sum(a))^2 using soft-max L-BFGS in log-space,
followed by coordinate descent polishing and LP-based escape.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog

def _norm_pos(a):
    """Normalize to sum=1, clip non-negative."""
    a = np.asarray(a, dtype=np.float64)
    np.clip(a, 0.0, None, out=a)
    s = np.sum(a)
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s

def _hard_score(a):
    """Compute the hard score 2n*max(conv)/(sum)^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)

def _soft_max(c, beta):
    """Numerically stable soft-max."""
    c_max = np.max(c)
    if not np.isfinite(c_max):
        return c_max
    e = np.exp(beta * (c - c_max))
    return c_max + np.log(np.sum(e)) / beta

def _soft_max_weights(c, beta):
    """Get the softmax weights for each lag."""
    c_max = np.max(c)
    e = np.exp(beta * (c - c_max))
    s = np.sum(e)
    return e / s

def _obj_grad(t, beta):
    """Compute soft-max objective and gradient in log-space.
    t: log of the sequence (a = exp(t))
    Returns (objective_value, gradient)
    """
    a = np.exp(t)
    n = len(a)
    S = np.sum(a)
    c = np.convolve(a, a)
    
    # Soft max and weights
    c_max = np.max(c)
    e = np.exp(beta * (c - c_max))
    e_sum = np.sum(e)
    m = c_max + np.log(e_sum) / beta
    w = e / e_sum
    
    score = 2.0 * n * m / (S * S)
    
    # Compute b_i = sum_j w_{i+j} * a_j for i = 0, ..., n-1
    # Using convolution: b = conv(a_rev, w)[n-1 : 2n-1]
    a_rev = a[::-1]
    conv_aw = np.convolve(a_rev, w)  # length 3n-2
    b = conv_aw[n-1 : 2*n-1]  # length n
    
    # w_{2i}
    idx_2i = 2 * np.arange(n)
    w_2i = np.zeros(n)
    valid = idx_2i < (2 * n - 1)
    w_2i[valid] = w[idx_2i[valid]]
    
    # dm/dt_i = a_i * (2*b_i - w_{2i} * a_i)
    dm_dt = a * (2.0 * b - w_2i * a)
    
    # dF/dt_i = (2n/S^2) * dm/dt_i - 4nm/S^3 * a_i
    grad = (2.0 * n / (S * S)) * dm_dt - (4.0 * n * m / (S**3)) * a
    
    return score, grad

    """Run L-BFGS on the soft-max objective for a given beta."""
    n = len(t0)
    
    # Normalize t so that sum(exp(t)) = 1 (breaks scale invariance)
    t = t0.copy()
    t -= np.logaddexp.reduce(t)  # now sum(exp(t)) = 1
    
    result = minimize(
        _obj_grad, t,
        method='L-BFGS-B',
        jac=True,

def _norm_pos(a):

def _hard_score(a):

    if s < 1e-30:

def _soft_max_obj_grad(t, beta):

    if S < 1e-300:
        return 1e30, np.zeros_like(t)

    c_max = float(np.max(c))
    e = np.exp(beta * (c - c_max))
    e_sum = float(np.sum(e))
    if e_sum < 1e-300:
        return 1e30, np.zeros_like(t)
    m = c_max + np.log(e_sum) / beta
    w = e / e_sum
    score = 2.0 * n * m / (S * S)
    a_rev = a[::-1]
    conv_aw = np.convolve(a_rev, w)
    b = conv_aw[n-1 : 2*n-1]
    idx_2i = 2 * np.arange(n)
    w_2i = np.zeros(n)
    valid = idx_2i < (2 * n - 1)
    w_2i[valid] = w[idx_2i[valid]]
    dm_dt = a * (2.0 * b - w_2i * a)
    grad = (2.0 * n / (S * S)) * dm_dt - (4.0 * n * m / (S * S * S)) * a

    rng = np.random.default_rng(int(seed))
    final_len = 3809

    best_seq = None
    best_score = float('inf')

    def _try_update(seq, sc_override=None):
        nonlocal best_seq, best_score
        if seq is None:
            return
        a = _norm_pos(seq)
        sc = sc_override if sc_override is not None else _hard_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_seq = a.copy()
            print(f"  New best: {sc:.8f} at t={time.monotonic()-t_start:.1f}s")

    # Seed from provided height_sequence_1
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if h.size > 0:
            h = _norm_pos(h)
            sc = _hard_score(h)
            if np.isfinite(sc):
                _try_update(h, sc)
                print(f"Starting from height_sequence_1: score={sc:.8f}, len={h.size}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = _two_level(final_len)
        best_score = _hard_score(best_seq)
        print(f"Default two-level start: score={best_score:.8f}")

    # If very little time, return immediately
    if time.monotonic() > deadline - 30:
        return [float(x) for x in best_seq]

    # Phase 1: L-BFGS with soft-max in log-space
    # Work on the full-length sequence for direct optimization
    work_len = final_len
    if best_seq.size != work_len:
        # Expand or truncate to work_len
        if best_seq.size < work_len:
            # Tile and truncate
            reps = (work_len // best_seq.size) + 1
            tiled = np.tile(best_seq, reps)[:work_len]
            best_seq = _norm_pos(tiled)
        else:
            best_seq = _norm_pos(best_seq[:work_len])
        best_score = _hard_score(best_seq)

    print(f"\n=== Phase 1: L-BFGS soft-max optimization (len={work_len}) ===")
    t0 = np.log(np.maximum(best_seq, 1e-30))

    # Adaptive beta schedule
    beta_schedule = [3.0, 8.0, 20.0, 50.0, 100.0, 200.0]
    phase1_deadline = deadline - max(400.0, 0.25 * budget)
    per_beta_time = max(30.0, (phase1_deadline - time.monotonic()) / len(beta_schedule))

    current_a = best_seq.copy()
    for bi, beta in enumerate(beta_schedule):
        if time.monotonic() > phase1_deadline:
            break
        beta_deadline = min(phase1_deadline, time.monotonic() + per_beta_time * 2.5)
        print(f"  L-BFGS beta={beta:.1f}, time_budget={per_beta_time*2.5:.0f}s")
        t_init = np.log(np.maximum(current_a, 1e-30))
        a_new, sc_new = _lbfgs_run(t_init, beta, beta_deadline, max_iter=200)
        if sc_new < best_score:
            current_a = a_new
            _try_update(a_new, sc_new)
        else:
            # Try small perturbation before next beta
            t_pert = np.log(np.maximum(current_a, 1e-30)) + rng.normal(0, 0.1, work_len)
            a_pert = _norm_pos(np.exp(t_pert))
            sc_pert = _hard_score(a_pert)
            if sc_pert < best_score:
                current_a = a_pert
                _try_update(a_pert, sc_pert)
        print(f"    After beta={beta:.1f}: score={best_score:.8f}")

    # Also try a few random restarts with L-BFGS
    restart_deadline = min(phase1_deadline, time.monotonic() + 120)
    for rs in range(3):
        if time.monotonic() > restart_deadline:
            break
        strat = int(rng.integers(0, 4))
        a_init = _make_start(work_len, strat, rng)
        # Blend with current best
        a_init = _norm_pos(0.7 * current_a + 0.3 * a_init)
        t_init = np.log(np.maximum(a_init, 1e-30))
        a_new, sc_new = _lbfgs_run(t_init, 30.0, restart_deadline, max_iter=100)
        if sc_new < best_score:
            _try_update(a_new, sc_new)
            current_a = a_new

    print(f"\nPhase 1 complete. Best score: {best_score:.8f}")

    # Phase 2: Coordinate descent polishing
    print(f"\n=== Phase 2: Coordinate descent (len={work_len}) ===")
    phase2_deadline = deadline - max(100.0, 0.05 * budget)
    a_cd, sc_cd = _coord_descent(current_a, phase2_deadline, rng, max_perturb=5)
    if sc_cd < best_score:
        _try_update(a_cd, sc_cd)
        current_a = a_cd
    print(f"Phase 2 complete. Best score: {best_score:.8f}")

    # Phase 3: LP escape on a shorter sub-sequence, then expand
    if time.monotonic() < deadline - 60:
        print(f"\n=== Phase 3: LP escape ===")
        lp_len = min(work_len, 192)
        # Take a slice of the best sequence
        a_short = _norm_pos(current_a[:lp_len])
        a_lp = _lp_direction(a_short, deadline - 60, max_iters=3)
        sc_lp = _hard_score(a_lp)
        # Expand and check
        a_expanded = _norm_pos(np.tile(a_lp, work_len // lp_len + 1)[:work_len])
        sc_exp = _hard_score(a_expanded)
        if sc_exp < best_score:
            _try_update(a_expanded, sc_exp)
            current_a = a_expanded
        # Also try LP on the full sequence if it's not too long
        if work_len <= 200:
            a_lp2 = _lp_direction(current_a, deadline - 60, max_iters=2)
            sc_lp2 = _hard_score(a_lp2)
            if sc_lp2 < best_score:
                _try_update(a_lp2, sc_lp2)
                current_a = a_lp2

    # Phase 4: Final short coordinate descent refinement
    if time.monotonic() < deadline - 10:
        print(f"\n=== Phase 4: Final polish ===")
        a_final, sc_final = _coord_descent(current_a, deadline - 5, rng, max_perturb=2)
        if sc_final < best_score:
            _try_update(a_final, sc_final)

    # Final: evaluate with external function and choose best
    print(f"\nFinal best internal score: {best_score:.8f}")
    try:
        ext_score = float(evaluate_sequence([float(x) for x in best_seq]))
        print(f"External evaluator score: {ext_score:.8f}")
    except Exception as e:
        print(f"External eval error: {e}")
        ext_score = best_score

    out = _norm_pos(best_seq)
    return [float(x) for x in out]
