import numpy as np
import time

def _pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()

def _clean(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    return np.maximum(v, 0.0)

def _project_simplex(v):
    v = _clean(v)
    n = v.size
    if n == 0:
        return v.copy()
    # Standard projection
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (css - 1))[0][-1]
    theta = (css[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s <= 0:
        return np.full(n, 1.0/n)
    return w / s

def _score(a):
    """Compute 2n * max(a*a) / (sum a)^2"""
    a = _clean(a)
    n = a.size
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a = a / s
    L = _pow2(2*n-1)
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A*A, L)[:2*n-1]
    c = np.maximum(c, 0)
    return 2.0 * n * c.max()

def _convolve(a):
    """Compute linear convolution a*a"""
    n = a.size
    L = _pow2(2*n-1)
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A*A, L)[:2*n-1]
    return np.maximum(c, 0)

def _subgradient_from_peaks(a, c, top_frac=0.01, abs_tol=0.0):
    """
    Compute subgradient of max(c) where c = a*a.
    Uses a set of peaks (top_frac of entries or those within abs_tol of max).
    Returns gradient g (length n) such that for any a':
    max(c(a')) >= max(c(a)) + g . (a' - a) (approximately)
    """
    n = a.size
    cmax = c.max()
    if cmax <= 0:
        return np.zeros(n)

    threshold = max(cmax * (1 - top_frac), cmax - abs_tol)
    S = np.where(c >= threshold)[0]

import time

    from scipy.optimize import minimize as _scipy_minimize
except Exception:
    _scipy_minimize = None


def _pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _clean(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    return np.maximum(v, 0.0)


def _project_simplex(v):
    v = _clean(v).copy()
    n = v.size
    if n == 0:
        return v
    s = v.sum()
    if s <= 1e-15:
        return np.full(n, 1.0 / n)
    v = v / s  # normalize first for numerical stability
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u > (css - 1.0) / idx
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    r = int(np.nonzero(cond)[0][-1])
    theta = (css[r] - 1.0) / float(r + 1)
    w = np.maximum(v - theta, 0.0)
    s2 = w.sum()
    if s2 <= 1e-15:
        return np.full(n, 1.0 / n)
    return w / s2


def _convolve_fft(a):
    n = a.size
    L = _pow2(2 * n - 1)
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score_fast(a):
    a = _clean(a)
    n = a.size
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a = a / s
    c = _convolve_fft(a)
    return 2.0 * float(n) * float(c.max())


def _subgrad_from_peaks(a, c, rel_tol=0.05, use_weights=True):
    """
    Compute subgradient of max(c) using peaks within rel_tol of max.
    Returns g (length n) where g_j = sum_m a_m * b_{j+m}.

    n = a.size
    cmax = float(c.max())
    if cmax <= 1e-300:
        return np.zeros(n)

    threshold = cmax * (1.0 - rel_tol)
    S = np.where(c >= threshold)[0]
    if len(S) == 0:
        S = np.array([int(np.argmax(c))])

    b = np.zeros(2 * n - 1)
    if use_weights:
        wts = c[S] - threshold + 1e-15
        b[S] = wts / wts.sum()

    L = _pow2(3 * n)
    A = np.fft.rfft(a, L)
    B = np.fft.rfft(b, L)
    g = np.fft.irfft(np.conj(A) * B, L)[:n]
    np.maximum(g, 0.0, g)
    return g

    best_score = _score_fast(best)

    rng = np.random.default_rng(seed)

    # Nesterov parameters
    y = a.copy()
    t_val = 1.0
    lr = 0.05
    lr_min = 1e-5
    lr_max = 0.5
    rel_tol = 0.08  # start wide, narrow over time
    last_improve = time.monotonic()
    start_time = time.monotonic()

    while time.monotonic() < end_time:
        elapsed = max(1e-6, time.monotonic() - start_time)
        total_est = max(1e-6, end_time - start_time)
        frac = min(1.0, elapsed / total_est)

        # Narrow the peak selection as we progress
        cur_rel_tol = max(0.01, rel_tol * (1.0 - 0.6 * frac))

        c = _convolve_fft(y)
        cmax = float(c.max())
        score = 2.0 * n * cmax

        if score < best_score - 1e-12:
            best_score = score
            best = y.copy()
            last_improve = time.monotonic()

        # Compute subgradient
        g = _subgrad_from_peaks(y, c, rel_tol=cur_rel_tol)
        dot = float(np.dot(y, g))

        # Check if we're at a stationary point
        g_spread = float(np.max(g) - np.min(g))
        if g_spread < 1e-12:
            break

        # Multiplicative update in log-space:
        # a_new_i = a_i * exp(-lr * (g_i - dot)) normalized
        ratio = g - dot
        ratio = np.clip(ratio, -10.0, 10.0)
        step = np.exp(-lr * ratio)
        a_candidate = y * step
        a_candidate = _project_simplex(a_candidate)

        # Nesterov momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        beta = (t_val - 1.0) / t_new
        y = a_candidate + beta * (a_candidate - y)
        y = _project_simplex(y)
        t_val = t_new

        # Adaptive learning rate
        sc_candidate = 2.0 * float(n) * float(_convolve_fft(a_candidate).max())
        if sc_candidate < best_score - 1e-12:
            lr = min(lr_max, lr * 1.03)
        else:
            lr = max(lr_min, lr * 0.92)

        # Periodic restart to escape local minima
        if time.monotonic() - last_improve > 8.0:
            # Small perturbation
            pert = np.exp(rng.normal(0.0, 0.02 + 0.03 * frac, size=n))
            y = _project_simplex(best * pert)
            t_val = 1.0
            lr = 0.05
            last_improve = time.monotonic()

            if time.monotonic() - last_improve > 2.0:
                # Try a larger perturbation
                pert_big = np.exp(rng.normal(0.0, 0.08, size=n))
                a_trial = _project_simplex(a0 * pert_big)
                sc_trial = _score_fast(a_trial)
                if sc_trial < best_score:
                    best = a_trial
                    best_score = sc_trial
                    y = best.copy()
                    t_val = 1.0

        # Very long stall: break
        if time.monotonic() - last_improve > 45.0:
            break

    return best, best_score


def _ratios_from_a(a):
    """Convert simplex vector to log-ratio parameterization."""
    a = _clean(a).copy()
    n = a.size
    s = a.sum()
    if s <= 1e-15:
        return np.zeros(max(0, n - 1))
    a = a / s
    if n <= 1:
        return np.zeros(0)
    ref = float(a[-1])
    if ref < 1e-200:
        a[-1] = 1e-10
        a = a / a.sum()
        ref = float(a[-1])
    x = np.log(np.maximum(a[:-1], 1e-300) / max(ref, 1e-300))
    return np.clip(x, -80.0, 80.0)


def _a_from_ratios(x):
    """Convert log-ratio parameterization back to simplex vector."""
    x = np.asarray(x, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    n = x.size + 1
    if n == 1:
        return np.ones(1)
    mx = float(np.max(x)) if x.size else 0.0
    e = np.exp(np.clip(x - mx, -700.0, 700.0))
    e_last = float(np.exp(-mx))
    s = float(e.sum()) + e_last
    if s <= 0 or not np.isfinite(s):
        return np.full(n, 1.0 / n)
    a = np.empty(n)
    a[:-1] = e / s
    a[-1] = e_last / s
    return a


def _lse_objective_grad(x, eps):
    """
    Log-sum-exp objective with gradient, in log-ratio space.
    f(x) = eps * log(sum exp(c_k/eps)) where c = a*a, a = softmax_params(x)
    Returns (f, grad_x).
    """
    a = _a_from_ratios(x)
    n = a.size
    if n <= 1:
        return 0.0, np.zeros_like(x)

    L = _pow2(2 * n - 1)
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)[:2 * n - 1]
    c = np.maximum(c, 0.0)

    cmax = float(c.max())
    if cmax < 1e-300:
        return 0.0, np.zeros_like(x)

    eps = max(float(eps), 1e-12)
    z = (c - cmax) / eps
    z = np.clip(z, -700.0, 0.0)
    exp_z = np.exp(z)
    sum_e = float(exp_z.sum())
    if sum_e < 1e-300 or not np.isfinite(sum_e):
        return 1e300, np.zeros_like(x)

    f = cmax + eps * float(np.log(sum_e))
    w = exp_z / sum_e

    # Gradient: dw/d a_j = 2 * (w * a)_shifted
    # ga_j = 2 * sum_k w_k * a_{k-j} (with bounds)
    L2 = _pow2(3 * n)
    W = np.fft.rfft(w, L2)
    Av = np.fft.rfft(a, L2)
    ga = 2.0 * np.fft.irfft(np.conj(Av) * W, L2)[:n]
    ga = np.maximum(ga, 0.0)

    dot = float(np.dot(a, ga))
    gfull = a * (ga - dot)
    gy = gfull[:-1]

    if not np.all(np.isfinite(gy)):
        gy = np.zeros_like(x)
    if not np.isfinite(f):
        f = 1e300

    return f, gy


def _q2_objective_grad(x):
    """
    L2 surrogate: sum(c_k^2) in log-ratio space.
    """
    a = _a_from_ratios(x)
    n = a.size
    if n <= 1:
        return 0.0, np.zeros_like(x)

    L = _pow2(2 * n - 1)
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)[:2 * n - 1]
    c = np.maximum(c, 0.0)

    f = float(np.dot(c, c))
    w = c  # derivative of sum c^2 is 2c

    L2 = _pow2(3 * n)
    W = np.fft.rfft(w, L2)
    Av = np.fft.rfft(a, L2)
    ga = 4.0 * np.fft.irfft(np.conj(Av) * W, L2)[:n]
    ga = np.maximum(ga, 0.0)

    dot = float(np.dot(a, ga))
    gfull = a * (ga - dot)
    gy = gfull[:-1]

    if not np.all(np.isfinite(gy)):
        gy = np.zeros_like(x)
    return f, gy


def _run_lbgfs(fun, x0, args, maxiter, bounds=None):
    """Single L-BFGS-B call with safety."""
    if _scipy_minimize is None:
        return x0, 99
    x = np.asarray(x0, dtype=np.float64)
    if x.size == 0:
        return x, 1
    if bounds is None:
        bounds = [(-80.0, 80.0)] * x.size
    opts = {"maxiter": maxiter, "disp": 0, "maxfun": max(200, 10 * maxiter),
            "ftol": 1e-20, "gtol": 1e-14}
    try:
        res = _scipy_minimize(fun, x, args=args, method="L-BFGS-B",
                              jac=True, bounds=bounds, options=opts)
        x_new = np.asarray(res.x, dtype=np.float64)
        x_new = np.where(np.isfinite(x_new), x_new, 0.0)
        x_new = np.clip(x_new, -80.0, 80.0)
        return x_new, int(res.status)
    except Exception:
        return x, 99


def _lbgfs_smooth(a0, end_time, seed, eps_start=0.05, eps_end=1e-7):
    """
    L-BFGS on log-sum-exp with geometrically decreasing eps.
    """
    if _scipy_minimize is None:
        return a0, _score_fast(a0)

    a = _project_simplex(a0)
    n = a.size
    best = a.copy()
    best_score = _score_fast(best)

    x = _ratios_from_a(a)
    bounds = [(-80.0, 80.0)] * max(0, n - 1)

    # First: L2 surrogate for a few iterations
    if time.monotonic() < end_time - 5.0 and x.size > 0:
        l2_end = min(end_time, time.monotonic() + 0.08 * (end_time - time.monotonic()))
        while time.monotonic() < l2_end:
            x, st = _run_lbgfs(_q2_objective_grad, x, (), 60, bounds)
            if st == 1:
                break
        a = _a_from_ratios(x)
        sc = _score_fast(a)
        if np.isfinite(sc) and sc < best_score:
            best, best_score = a, sc
            x = _ratios_from_a(best)

    # Then: log-sum-exp with decreasing eps
    eps = eps_start
    eps_min = eps_end
    last_improve = time.monotonic()
    stall_count = 0

    while time.monotonic() < end_time - 1.0:
        x, st = _run_lbgfs(_lse_objective_grad, x, (eps,), 80, bounds)
        a = _a_from_ratios(x)
        sc = _score_fast(a)

        now = time.monotonic()
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best = a.copy()
            x = _ratios_from_a(best)
            last_improve = now
            stall_count = 0
            eps = max(eps_min, eps * 0.6)
        else:
            stall_count += 1
            if stall_count >= 3:
                # Perturb in x-space
                rng = np.random.default_rng(seed + int(now * 1000))
                x = np.clip(x + rng.normal(0, 0.03, x.size), -80, 80)
                eps = max(eps_min, eps * 0.5)
                stall_count = 0
            elif eps > eps_min:
                eps = max(eps_min, eps * 0.7)

        if eps <= eps_min * 1.01 and stall_count >= 5:
            break

    return best, best_score


def _local_search(a0, end_time, seed):
    """Block perturbation local search."""
    a = _project_simplex(a0)
    n = a.size
    best = a.copy()
    best_score = _score_fast(best)

    rng = np.random.default_rng(seed)
    sigma = 0.10
    last_improve = time.monotonic()
    start = time.monotonic()

    while time.monotonic() < end_time:
        cand = best.copy()
        nblocks = 1 + (1 if rng.random() < 0.3 else 0)
        for _ in range(nblocks):
            maxw = max(3, n // 6)
            w = int(np.exp(rng.uniform(np.log(2.0), np.log(float(maxw)))))
            w = max(1, min(w, n))
            s = int(rng.integers(0, n - w + 1))
            cand[s:s + w] *= np.exp(rng.normal(0.0, sigma))

        cand = _project_simplex(cand)
        sc = _score_fast(cand)

        if np.isfinite(sc) and sc < best_score - 1e-12:
            best = cand
            best_score = sc
            last_improve = time.monotonic()
            sigma = min(0.30, sigma * 1.05)
        else:
            if time.monotonic() - last_improve > 4.0:
                sigma = max(0.02, sigma * 0.7)
                if time.monotonic() - last_improve > 20.0:
                    # Try re-centering
                    pert = np.exp(rng.normal(0, 0.04, n))
                    cand2 = _project_simplex(best * pert)
                    sc2 = _score_fast(cand2)
                    if sc2 < best_score:
                        best, best_score = cand2, sc2
                        last_improve = time.monotonic()
                    else:
                        break

    return best, best_score


def _power_seq(n, p=0.5, offset=1.0):
    j = np.arange(n, dtype=np.float64)
    x = np.maximum(j + offset, 1e-8)
    return np.power(x, -p)


def _window_seq(n, gamma=1.0):
    """Raised-cosine-like window, one-sided emphasis."""
    j = np.arange(n, dtype=np.float64)
    t = j / max(1.0, n - 1.0)
    return np.power(1.0 - t, gamma) + 0.1


def _two_phase_seq(n, p1=0.3, p2=0.7, split=0.6):
    """Two-phase decay: fast then slow."""
    j = np.arange(n, dtype=np.float64)
    mid = int(n * split)
    v = np.empty(n)
    if mid > 0:
        x1 = np.arange(mid, dtype=np.float64) + 1.0
        v[:mid] = np.power(x1, -p1)
    if n - mid > 0:
        x2 = np.arange(n - mid, dtype=np.float64) + 1.0
        v[mid:] = np.power(x1[-1] if mid > 0 else 1.0, -p1) * np.power(x2, -(p2 - p1))
    return v


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Returns best sequence found within time budget.
    """
    t0 = time.monotonic()
    deadline = t0 + max(10.0, float(budget_s) - 15.0)

    # Determine length
    n = 3809
    try:
        h = height_sequence_1
        if isinstance(h, (list, np.ndarray)):
            arr = np.asarray(h, dtype=np.float64)
            if arr.ndim == 1 and 100 <= arr.size <= 20000:
                n = int(arr.size)
    except Exception:
        pass

    print(f"n = {n}, budget = {budget_s}s, deadline in {deadline - t0:.1f}s")

    # Generate initial candidates
    cands = []
    try:
        h = height_sequence_1
        if isinstance(h, (list, np.ndarray)):
            arr = _clean(np.asarray(h, dtype=np.float64))
            if arr.size == n and arr.sum() > 0:
                cands.append(_project_simplex(arr))
    except Exception:
        pass

    cands.append(_project_simplex(_power_seq(n, 0.5, 1.0)))
    cands.append(_project_simplex(_power_seq(n, 0.5, 0.5)))
    cands.append(_project_simplex(_power_seq(n, 0.4, 1.0)))
    cands.append(_project_simplex(_power_seq(n, 0.6, 1.0)))
    cands.append(_project_simplex(_window_seq(n, 1.0)))
    cands.append(_project_simplex(_window_seq(n, 1.5)))
    cands.append(_project_simplex(_window_seq(n, 2.0)))
    cands.append(_project_simplex(_two_phase_seq(n, 0.3, 0.7, 0.6)))
    cands.append(_project_simplex(_two_phase_seq(n, 0.5, 0.5, 0.5)))

    # Score and sort
    scored = []
    for i, c in enumerate(cands):
        sc = _score_fast(c)
        if np.isfinite(sc):
            scored.append((sc, i, c))
    scored.sort(key=lambda x: x[0])

    print(f"Best initial candidate score: {scored[0][0]:.6f}")

    global_best = scored[0][2].copy()
    global_best_score = scored[0][0]

    # Phase 1: Subgradient on best candidate (25% time)
    phase1_end = t0 + 0.25 * (deadline - t0)
    if time.monotonic() < phase1_end:
        a_sg, sc_sg = _subgrad_optimize(global_best, phase1_end, int(seed) + 1)
        if sc_sg < global_best_score:
            global_best, global_best_score = a_sg, sc_sg
        print(f"After phase 1 (subgrad): {global_best_score:.6f}")

    # Phase 2: L-BFGS smoothing (25% time)
    phase2_end = t0 + 0.50 * (deadline - t0)
    if time.monotonic() < phase2_end and _scipy_minimize is not None:
        a_lb, sc_lb = _lbgfs_smooth(global_best, phase2_end, int(seed) + 2,
                                    eps_start=0.03, eps_end=1e-7)
        if sc_lb < global_best_score:
            global_best, global_best_score = a_lb, sc_lb
        print(f"After phase 2 (L-BFGS): {global_best_score:.6f}")

    # Phase 3: Subgradient refinement (25% time)
    phase3_end = t0 + 0.75 * (deadline - t0)
    if time.monotonic() < phase3_end:
        a_sg2, sc_sg2 = _subgrad_optimize(global_best, phase3_end, int(seed) + 3)
        if sc_sg2 < global_best_score:
            global_best, global_best_score = a_sg2, sc_sg2
        print(f"After phase 3 (subgrad2): {global_best_score:.6f}")

    # Phase 4: L-BFGS fine polish (15% time)
    phase4_end = t0 + 0.90 * (deadline - t0)
    if time.monotonic() < phase4_end and _scipy_minimize is not None:
        a_lb2, sc_lb2 = _lbgfs_smooth(global_best, phase4_end, int(seed) + 4,
                                      eps_start=0.01, eps_end=1e-8)
        if sc_lb2 < global_best_score:
            global_best, global_best_score = a_lb2, sc_lb2
        print(f"After phase 4 (L-BFGS fine): {global_best_score:.6f}")

    # Phase 5: Local search (10% time)
    if time.monotonic() < deadline:
        a_ls, sc_ls = _local_search(global_best, deadline, int(seed) + 5)
        if sc_ls < global_best_score:
            global_best, global_best_score = a_ls, sc_ls
        print(f"After phase 5 (local): {global_best_score:.6f}")

    # Also try second-best initial candidate if time was well used
    # (skip - time is tight)

    global_best = _project_simplex(global_best)
    out = [float(x) for x in global_best]

    # Validate
    try:
        val = evaluate_sequence(out)
        if not np.isfinite(val):
            print("WARNING: invalid sequence, using fallback")
            out = [float(x) for x in _project_simplex(_power_seq(n, 0.5, 1.0))]
        else:
            print(f"Validated score: {val:.6f}")
    except Exception as e:
        print(f"Validation error: {e}")

    print(f"Final answer score: {global_best_score:.6f}")
    return out
