"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 with a >= 0.
Combines coordinate descent with peak-focused ordering, LP-based escape,
and multi-scale initialization to improve upon the current best of ~1.555.

import numpy as np
import time


def _clean(arr):
    """Clean an array: flatten, handle NaN/inf, clip to [0, 1000]."""
    a = np.asarray(arr, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    np.clip(a, 0.0, 1000.0, out=a)
    return a


def _norm(a):
    """Normalize a to have sum 1."""
    a = _clean(a)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def _score(p, c):
    """Score = 2*n*max(c) / (sum(p))^2, assuming c = conv(p,p)."""
    n = int(p.size)
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 0.0:
        return float('inf')
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return float('inf')
    return (2.0 * n * max(0.0, mx)) / (s * s)


def _score_from(p):
    """Compute score from raw array."""
    p = _norm(p)
    c = np.convolve(p, p)
    return _score(p, c)


def _two_level(n, ratio=2.0/3.0):
    """Two-level half-block initialization."""
    n = int(n)
    if n <= 1:
        return np.ones(max(1, n), dtype=np.float64)
    m = n // 2
    p = np.zeros(n, dtype=np.float64)
    if m > 0:
        p[:m] = ratio / float(m)
    rest = n - m
    if rest > 0:
        p[m:] = (1.0 - ratio) / float(rest)
    s = float(np.sum(p))
    if s > 0:
        p /= s
    else:
        p = np.ones(n, dtype=np.float64) / float(n)
    return p


def _periodic_init(n, period, rng):
    """Create a periodic initialization with given period."""
    n = int(n)
    period = int(period)
    if period <= 1:
        return np.ones(n, dtype=np.float64) / float(n)
    if period > n:
        period = n
    base = rng.dirichlet(np.full(period, 1.0))
    p = np.tile(base, n // period + 1)[:n]
    return _norm(p)


def _make_init(n, strategy, rng):

    n = int(n)
    if n <= 1:
        return np.ones(max(1, n), dtype=np.float64)
    st = int(strategy) % 16
    if st == 0:

    elif st == 1:
        return rng.dirichlet(np.full(n, 1.0))
    elif st == 2:
        return rng.dirichlet(np.full(n, 0.5))
    elif st == 3:

    elif st == 5:

        return _norm(p)
    elif st == 6:

        for r in range(m):

            b = n - 1 - r
            w = rng.dirichlet(np.full(2, 0.5))
            p[a] = w[0]
            p[b] = w[1]
        return _norm(p)
    elif st == 7:

        return _two_level(n, ratio)

    elif st == 12:

    elif st == 13:

        p = np.exp(x_smooth)

def _expand(proto, target_len):
    """Expand a prototype to target length by block replication."""
    target_len = int(target_len)
    p = _norm(proto)
    R = int(p.size)
    if R <= 0 or target_len <= 0:
        return np.ones(max(1, target_len), dtype=np.float64)
    if R == target_len:
        return p.copy()
    if R > target_len:
        # Downsample by averaging
        out = np.zeros(target_len, dtype=np.float64)
        for j in range(target_len):
            lo = (j * R) // target_len
            hi = max(lo + 1, ((j + 1) * R) // target_len)
            out[j] = float(np.mean(p[lo:hi]))
        return _norm(out)
    # Upsample by block replication
    base = target_len // R
    rem = target_len % R
    lengths = np.full(R, base, dtype=np.int64)
    if rem > 0:
        lengths[:rem] += 1
    out = np.zeros(target_len, dtype=np.float64)
    idx = 0
    for i in range(R):
        L = int(lengths[i])
        if L > 0:
            out[idx:idx+L] = p[i] / float(L)
            idx += L
    return _norm(out)
