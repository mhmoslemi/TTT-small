"""
Hybrid optimizer for non-negative step function convolution minimization.
Uses targeted peak reduction with top-K weighting, anti-palindromic structure,
and multi-scale exploration to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Key innovations: adaptive top-K peak targeting, backtracking line search,
correlated restarts, and frequency-domain-aware initialization.
"""

import numpy as np
import time


def _pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _fft_square(a):
    n = int(a.size)
    L = 2 * n - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, n=N)
    b = np.fft.irfft(fa * fa, n=N).real[:L]
    b = np.maximum(b, 0.0)
    return b


def _fft_grad(weights, a):
    n = int(a.size)
    L = int(weights.size)
    N = _pow2(L + n - 1)
    fw = np.fft.rfft(weights, n=N)
    fa = np.fft.rfft(a, n=N)
    c = np.fft.irfft(fw * np.conj(fa), n=N)[:n]
    return 2.0 * c


def _internal_score(a, b=None):
    if b is None:
        b = _fft_square(a)
    s = float(a.sum())
    if s <= 1e-300:
        return float("inf"), b, 0.0
    maxb = float(b.max())
    score = float(2.0 * a.size * maxb / (s * s))
    return score, b, maxb


def _make_weights_topk(b, focus, K=1, eps=0.01):
    """Create weights emphasizing top-K peaks."""
    L = int(b.size)
    maxb = float(b.max())
    if maxb <= 1e-300:
        return np.full(L, 1.0 / L, dtype=np.float64), maxb
    
    # Top-K indices
    if K < L:
        topk_idx = np.argpartition(b, -K)[-K:]
    else:
        topk_idx = np.arange(L)
    
    p = np.zeros(L, dtype=np.float64)
    for idx in topk_idx:
        p[int(idx)] = 1.0
    
    # Add soft background
    with np.errstate(over="ignore", under="ignore"):
        z = (maxb - b) / maxb
        bg = np.exp(-z * float(focus))
    bg_sum = float(bg.sum())
    if bg_sum > 1e-30:
        bg = bg / bg_sum
    else:
        bg = np.full(L, 1.0 / L)
    
    # Combine: top-K get high weight, rest gets soft background
    alpha = 0.8  # fraction of weight on top-K
    p = alpha * (p / K) + (1 - alpha) * bg
    
    # Add small uniform component for exploration
    p = (1.0 - eps) * p + eps / L
    return p, maxb


def _make_weights_softmax(b, focus, eps=0.01):
    """Standard soft-max weighting."""
    L = int(b.size)
    maxb = float(b.max())
    if maxb <= 1e-300:
        return np.full(L, 1.0 / L, dtype=np.float64), maxb
    with np.errstate(over="ignore", under="ignore"):
        z = (maxb - b) / maxb
        p = np.exp(-z * float(focus))
    s = float(p.sum())
    if s <= 1e-12:
        p = np.zeros(L, dtype=np.float64)
        p[int(np.argmax(b))] = 1.0
    else:
        p = p / s
    p = (1.0 - eps) * p + eps / L
    return p, maxb


def _random_positive(n, rng, amp=3.0):
    n = int(n)
    z = rng.normal(0.0, amp, size=n)
    z -= z.max()
    a = np.exp(z)
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    return a


def _anti_log_init(n, rng, amp=3.0, smooth=False):
    n = int(n)
    m = n // 2
    if m < 1:
        return np.ones(n, dtype=np.float64) / max(1, n)
    x = np.arange(m, dtype=np.float64)
    if smooth:
        v = np.zeros(m, dtype=np.float64)
        for f in (1, 3, 5, 7, 11, 15, 21):
            v += (rng.normal(0.0, 1.0) / max(1, f)) * np.sin(
                2.0 * np.pi * f * x / m + rng.uniform(0.0, 2.0 * np.pi)
            )
        v += 0.1 * rng.normal(0.0, 1.0, m)
    else:
        v = rng.normal(0.0, 1.0, m)
    v = v - float(v.mean())
    std = float(v.std())
    if std > 1e-12:
        v = (float(amp) * v) / std
    z = np.zeros(n, dtype=np.float64)
    z[:m] = v
    if n % 2 == 0:
        z[m:] = -v[::-1]
    else:
        z[m + 1:] = -v[::-1]
        z[m] = 0.0
    z = z - z.max()
    a = np.exp(z)
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    return a


def _greedy_support(n, rng, shuffle=True):
    n = int(n)
    m = n // 2
    counts = np.zeros(2 * n - 1, dtype=np.int64)
    selected = np.empty(m, dtype=np.int64)
    order = np.arange(m, dtype=np.int64)
    if shuffle:
        rng.shuffle(order)
    t = 0
    for idx_val in order:
        idx = int(idx_val)
        l = idx
        r = n - 1 - idx
        if t == 0:
            y = l
        else:
            s = selected[:t]
            affected_l = int(np.max(counts[s + l]))
            self_l = int(counts[2 * l])
            metric_l = max(affected_l + 2, self_l + 1)
            affected_r = int(np.max(counts[s + r]))
            self_r = int(counts[2 * r])
            metric_r = max(affected_r + 2, self_r + 1)
            y = l if metric_l <= metric_r else r
        if t > 0:
            counts[selected[:t] + y] += 2
        counts[2 * y] += 1
        selected[t] = y
        t += 1
    return selected


def _greedy_x(n, rng, shuffle=True):
    sel = _greedy_support(n, rng, shuffle=shuffle)
    m = n // 2
    x = np.full(m, 1e-6, dtype=np.float64)
    left = sel < m
    if np.any(left):
        x[sel[left]] = 1.0 - 1e-6
    return x


def _a_from_x(x, n=None):
    x = np.asarray(x, dtype=np.float64).ravel()
    m = x.size
    if n is None:
        n = 2 * m
    t = 1.0 / m
    a = np.empty(n, dtype=np.float64)
    a[:m] = t * x
    a[m:] = t * (1.0 - x[::-1])
    return a


def _project_antipal(a):
    """Project a onto the anti-palindromic subspace a_i + a_{n-1-i} = 2/n."""
    n = int(a.size)
    m = n // 2
    t = 2.0 / n
    a_mirror = a[n-1::-1].copy()
    # a_i + a_{n-1-i} should equal t
    # Let a_i = t/2 + d_i, a_{n-1-i} = t/2 - d_i
    # From the given a, estimate d_i:
    # a_i + a_{n-1-i} = sum should be t, but might not be
    # We want to find d_i that minimizes change from a
    # d_i = (a_i - a_{n-1-i}) / 2
    d = (a[:m] - a[n-1-m:n-1][::-1]) / 2.0
    # But we need a_i >= 0, so d_i <= t/2
    d = np.clip(d, -t/2 + 1e-10, t/2 - 1e-10)
    result = np.empty(n, dtype=np.float64)
    result[:m] = t/2 + d
    if n % 2 == 0:
        result[m:] = t/2 - d[::-1]
    else:
        result[m+1:] = t/2 - d[::-1]
        result[m] = t/2
    result = np.maximum(result, 1e-15)
    return result


def _optimize_full(a0, time_limit, rng, verbose=False, focus_start=5.0,
                   max_focus=800.0, label="", use_topk=True, topk_K=3):
    """Full-vector exponentiated gradient optimizer with top-K peak targeting."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = int(a.size)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-300:
        a = np.ones(n, dtype=np.float64) / max(1, n)
    else:
        a /= s
    a = np.maximum(a, 1e-15)
    a /= a.sum()

    b = None
    score = float("inf")
    best_score = float("inf")
    best_a = a.copy()

    focus = float(focus_start)
    focus_inc = 1.015
    eta = 1.0
    eta_min = 0.01
    eta_max = 10.0

    no_improve = 0
    max_no = 500
    resets = 0
    max_resets = 30
    t0 = time.monotonic()
    it = 0

    # Adaptive K: start small, grow over time
    K = 1
    K_growth = 200  # iterations between K increases

    while time.monotonic() - t0 < float(time_limit):
        if b is None:
            b = _fft_square(a)
            score, _, _ = _internal_score(a, b)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and (it % 500 == 0):
            print(f"[{label}] it={it} t={time.monotonic()-t0:.1f}s "
                  f"score={score:.6f} best={best_score:.6f} "
                  f"focus={focus:.1f} eta={eta:.3f} K={K} resets={resets}")

        if no_improve > max_no:
            resets += 1
            if resets > max_resets:
                break
            no_improve = 0
            max_no = int(max_no * 1.2)
            # Correlated restart: perturb the best solution
            base = best_a.copy()
            if resets % 3 == 0:
                # Large perturbation
                fresh = _random_positive(n, rng, amp=4.0)
                a = 0.5 * base + 0.5 * fresh
            elif resets % 3 == 1:
                # Medium perturbation
                a = base * np.exp(rng.normal(0.0, 1.0, size=n))
            else:
                # Small perturbation
                a = base * np.exp(rng.normal(0.0, 0.3, size=n))
            a = np.maximum(a, 1e-15)
            a /= a.sum()
            b = None
            score = float("inf")
            focus = float(rng.choice([2.0, 5.0, 10.0, 20.0, 50.0, 100.0]))
            eta = 1.0
            K = 1
            continue

        # Make weights
        if use_topk and K < L if (L := int(b.size)) else 1:
            pass  # Python 3.8 compatible
        
        if use_topk:
            p, _ = _make_weights_topk(b, focus, K=max(K, 1))
        else:
            p, _ = _make_weights_softmax(b, focus)
        
        # Gradient
        grad = _fft_grad(p, a)
        gd = grad - float(np.dot(a, grad))  # center (for simplex)
        maxg = float(np.max(np.abs(gd)))
        if maxg < 1e-18:
            break

        gd = gd / maxg

        # Backtracking line search
        accepted = False
        for attempt in range(5):
            a_new = a * np.exp(-eta * gd)
            a_new = np.maximum(a_new, 1e-300)
            ss = float(a_new.sum())
            if ss <= 1e-300:
                break
            a_new /= ss
            b_new = _fft_square(a_new)
            score_new, _, _ = _internal_score(a_new, b_new)
            
            if score_new < score - 1e-10:
                a = a_new
                b = b_new
                score = score_new
                eta = min(eta * 1.1, eta_max)
                accepted = True
                break
            else:
                eta = eta * 0.5
        
        if not accepted:
            eta = max(eta * 0.7, eta_min)
            if eta < 0.05:
                a = a * np.exp(rng.normal(0.0, 0.1, size=n))
                a = np.maximum(a, 1e-15)
                a /= a.sum()
                b = None
                score = float("inf")
                eta = 1.0

        # Grow K periodically
        if it % K_growth == 0:
            K = min(K + 1, 50)
        
        focus = min(focus * focus_inc, float(max_focus))
        it += 1

    return best_a, float(best_score)


def _optimize_pair(x0, time_limit, rng, verbose=False, focus_start=5.0,
                   max_focus=800.0, label=""):
    """Anti-palindromic pair optimizer."""
    x = np.asarray(x0, dtype=np.float64).copy()
    m = int(x.size)
    n = 2 * m
    x_eps = 1e-8
    x = np.clip(x, x_eps, 1.0 - x_eps)

    a = _a_from_x(x, n)
    b = None
    score = float("inf")

    best_score = float("inf")
    best_x = x.copy()

    focus = float(focus_start)
    focus_inc = 1.015
    eta = 1.0
    eta_min = 0.01
    eta_max = 10.0

    no_improve = 0
    max_no = 500
    resets = 0
    max_resets = 25
    t0 = time.monotonic()
    it = 0

    K = 1
    K_growth = 200

    while time.monotonic() - t0 < float(time_limit):
        if b is None:
            b = _fft_square(a)
            score, _, _ = _internal_score(a, b)

        if score < best_score - 1e-12:
            best_score = score
            best_x = x.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and (it % 500 == 0):
            print(f"[{label}] it={it} t={time.monotonic()-t0:.1f}s "
                  f"score={score:.6f} best={best_score:.6f} "
                  f"focus={focus:.1f} eta={eta:.3f} K={K} resets={resets}")

        if no_improve > max_no:
            resets += 1
            if resets > max_resets:
                break
            no_improve = 0
            max_no = int(max_no * 1.2)
            if resets % 2 == 0:
                x = 0.6 * best_x + 0.4 * rng.uniform(0.0, 1.0, size=m)
            else:
                x = np.clip(best_x + rng.normal(0.0, 0.2, size=m), x_eps, 1.0 - x_eps)
            x = np.clip(x, x_eps, 1.0 - x_eps)
            a = _a_from_x(x, n)
            b = None
            score = float("inf")
            focus = float(rng.choice([2.0, 5.0, 10.0, 30.0, 100.0]))
            eta = 1.0
            K = 1
            continue

        if K < int(b.size):
            p, _ = _make_weights_topk(b, focus, K=max(K, 1))
        else:
            p, _ = _make_weights_softmax(b, focus)

        ga = _fft_grad(p, a)
        gx = ga[:m] - ga[m:n-m][::-1] if n % 2 == 0 else ga[:m] - ga[m+1:][::-1]
        maxg = float(np.max(np.abs(gx)))
        if maxg < 1e-18:
            break

        gx = gx / maxg

        # Backtracking
        accepted = False
        for attempt in range(5):
            x_new = x - eta * gx
            x_new = np.clip(x_new, x_eps, 1.0 - x_eps)
            a_new = _a_from_x(x_new, n)
            b_new = _fft_square(a_new)
            score_new, _, _ = _internal_score(a_new, b_new)

            if score_new < score - 1e-10:
                x = x_new
                a = a_new
                b = b_new
                score = score_new
                eta = min(eta * 1.1, eta_max)
                accepted = True
                break
            else:
                eta = eta * 0.5

        if not accepted:
            eta = max(eta * 0.7, eta_min)
            if eta < 0.05:
                x = np.clip(x + rng.normal(0.0, 0.05, size=m), x_eps, 1.0 - x_eps)
                a = _a_from_x(x, n)
                b = None
                score = float("inf")
                eta = 1.0

        if it % K_growth == 0:
            K = min(K + 1, 30)

        focus = min(focus * focus_inc, float(max_focus))
        it += 1

    return _a_from_x(best_x, n), float(best_score)


def _safe_evaluate(seq):
    ev = globals().get("evaluate_sequence")
    if ev is None or not callable(ev):
        return float("inf")
    try:
        return float(ev(seq))
    except Exception:
        return float("inf")


def _coordinate_descent(a, time_limit, rng, max_peaks=5):
    """Targeted coordinate descent on the top peaks."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = int(a.size)
    a /= a.sum()
    
    best_a = a.copy()
    best_score, _, _ = _internal_score(a)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < float(time_limit):
        b = _fft_square(a)
        score, _, maxb = _internal_score(a, b)
        
        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
        
        # Find top peaks
        L = 2 * n - 1
        topk = np.argpartition(b, -max_peaks)[-max_peaks:]
        topk = topk[np.argsort(b[topk])[::-1]]  # sorted by value descending
        
        for k in topk:
            k = int(k)
            # Variables that affect b_k: a_i where i is in [max(0,k-n+1), min(n-1,k)]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            
            # Try to reduce b_k by adjusting a_i
            # The contribution of a_i to b_k is a_i * a_{k-i}
            # Gradient of b_k w.r.t. a_i is a_{k-i} (plus a_i if 2i=k)
            
            # Simple move: reduce a_i proportionally to its contribution
            for trial in range(3):
                # Pick a random variable in the range
                i = int(rng.integers(lo, hi + 1))
                move = 0.01 * a[i] * a[k - i]
                a[i] -= move
                if i != k - i:
                    a[k - i] -= 0  # only adjust one at a time for simplicity
                
                a = np.maximum(a, 0.0)
                s = float(a.sum())
                if s <= 1e-300:
                    a = best_a.copy()
                    break
                a_new = a / s
                b_new = _fft_square(a_new)
                score_new, _, _ = _internal_score(a_new, b_new)
                
                if score_new < score - 1e-10:
                    a = a_new
                    score = score_new
                else:
                    a[i] += move  # revert
                    break
        
        if time.monotonic() - t0 > float(time_limit):
            break
    
    return best_a, float(best_score)


def propose_candidate(seed=42, budget_s=1800):
    """Main entry point for the optimization."""
    seed = int(seed)
    total = max(60.0, float(budget_s))

    now0 = time.monotonic()
    final_deadline = now0 + total - 10.0
    task_deadline = now0 + total - 120.0

    print("Starting hybrid step-function convolution optimizer")
    best_a = None
    best_score = float("inf")

    # Load given sequence if available
    given_a = None
    try:
        given = globals().get("height_sequence_1")
        if given is not None:
            g = np.asarray(given, dtype=np.float64).ravel()
            g = np.nan_to_num(g, nan=0.0, posinf=1000.0, neginf=0.0)
            g = np.clip(g, 0.0, 1000.0)
            s = float(g.sum())
            if s > 0.01 and g.size > 10:
                gn = g / s
                gn = np.maximum(gn, 1e-15)
                gn /= gn.sum()
                sc, _, _ = _internal_score(gn)
                if np.isfinite(sc):
                    best_score = sc
                    best_a = gn.copy()
                    given_a = gn
                print(f"Given: n={gn.size}, score={sc:.6f}")
    except Exception:
        pass

    # Task schedule: (n, type, seed, time_alloc)
    # Type: "full" = full optimization, "pair" = anti-palindromic
    tasks = []
    
    if given_a is not None:
        # First: improve the given sequence
        tasks.append((given_a.size, "full_given", 0, 200.0))
    
    # Anti-palindromic on 8192
    tasks.append((8192, "pair", 0, 300.0))
    tasks.append((8192, "pair", 1, 300.0))
    
    # Full optimization on 8192 with different inits
    tasks.append((8192, "full_anti", 0, 300.0))
    tasks.append((8192, "full_anti", 1, 300.0))
    
    # Full optimization on 4096
    tasks.append((4096, "full_random", 0, 200.0))
    
    # More full on 8192
    tasks.append((8192, "full_greedy", 0, 300.0))

    for idx, (n, kind, task_seed, max_alloc) in enumerate(tasks):
        now = time.monotonic()
        if now >= task_deadline:
            print("Reached task deadline, stopping search phase")
            break
        alloc = min(max_alloc, task_deadline - now)
        if alloc < 30.0:
            print(f"Insufficient time for task {idx}, skipping")
            break

        ng_seed = (seed * 1000003 + idx * 7919 + task_seed * 131) % (2**32)
        task_rng = np.random.default_rng(ng_seed)

        print(f"\nTask {idx}: n={n} type={kind} seed={task_seed} alloc={alloc:.0f}s "
              f"current_best={best_score:.6f}")
        t0 = time.monotonic()
        cand_a = None
        cand_score = float("inf")

        try:
            if kind == "full_given":
                a0 = given_a.copy()
                cand_a, cand_score = _optimize_full(
                    a0, alloc, task_rng, verbose=True,
                    focus_start=10.0, max_focus=1000.0,
                    label=f"given{n}", use_topk=True, topk_K=3)
                
            elif kind == "pair":
                shuffle = (task_seed % 2 == 1)
                x0 = _greedy_x(n, task_rng, shuffle=shuffle)
                cand_a, cand_score = _optimize_pair(
                    x0, alloc, task_rng, verbose=True,
                    focus_start=5.0, max_focus=800.0,
                    label=f"pair{n}-{task_seed}")
                
            elif kind == "full_anti":
                amp = 2.0 + 1.0 * float(task_seed)
                a0 = _anti_log_init(n, task_rng, amp=amp, smooth=(task_seed % 2 == 0))
                cand_a, cand_score = _optimize_full(
                    a0, alloc, task_rng, verbose=True,
                    focus_start=5.0, max_focus=800.0,
                    label=f"anti{n}-{task_seed}", use_topk=True)
                
            elif kind == "full_random":
                a0 = _random_positive(n, task_rng, amp=2.0)
                cand_a, cand_score = _optimize_full(
                    a0, alloc, task_rng, verbose=False,
                    focus_start=5.0, max_focus=800.0,
                    label=f"rand{n}", use_topk=True)
                
            elif kind == "full_greedy":
                # Use greedy support to initialize
                sel = _greedy_support(n, task_rng, shuffle=(task_seed % 2 == 1))
                m = n // 2
                a0 = np.full(n, 0.1, dtype=np.float64)
                for s_idx in sel:
                    a0[int(s_idx)] = 1.0
                    a0[n - 1 - int(s_idx)] = 1.0
                a0 += 0.01 * task_rng.random(n)
                a0 = np.maximum(a0, 1e-6)
                a0 /= a0.sum()
                cand_a, cand_score = _optimize_full(
                    a0, alloc, task_rng, verbose=True,
                    focus_start=5.0, max_focus=800.0,
                    label=f"greedy{n}-{task_seed}", use_topk=True)
            else:
                a0 = _random_positive(n, task_rng)
                cand_a, cand_score = _optimize_full(
                    a0, alloc, task_rng, verbose=False,
                    focus_start=5.0, label="misc")

            if cand_a is not None:
                cand_a = np.nan_to_num(cand_a, nan=0.0, posinf=0.0, neginf=0.0)
                cand_a = np.clip(cand_a, 0.0, 1000.0)
                cc = float(cand_a.sum())
                if cc > 0.01:
                    cand_a /= cc
                    internal, _, _ = _internal_score(cand_a)
                    if np.isfinite(internal) and internal < cand_score:
                        cand_score = internal
                    if np.isfinite(cand_score) and cand_score < best_score:
                        best_score = cand_score
                        best_a = cand_a.copy()
                        print(f"  NEW BEST: {cand_score:.6f} (n={cand_a.size})")
        except Exception as exc:
            print(f"  Task failed: {repr(exc)}")

        used = time.monotonic() - t0
        print(f"  Done task {idx}: used={used:.1f}s, candidate={cand_score:.6f}, "
              f"best={best_score:.6f}")

    # Final refinement
    if best_a is not None:
        remaining = final_deadline - time.monotonic()
        if remaining > 60.0:
            refine_time = min(100.0, remaining - 30.0)
            if refine_time > 20.0:
                task_rng = np.random.default_rng((seed + 777777) % (2**32))

The standard soft-max: `p_k = exp(-focus * (M - b_k) / M) / sum(exp(...))`

This gives high weight to peaks close to the maximum. With high focus, it concentrates on the single maximum.

The top-K approach: gives uniform weight to the top K peaks, plus a soft background.

The top-K approach might be better when there are multiple peaks at similar heights (which is common in the later stages of optimization). It prevents the optimizer from "chasing" a single peak while others are just as bad.

However, the top-K approach has a disadvantage: it's not smooth (discontinuous when a peak moves in or out of the top K). This can cause issues with the optimization.

A compromise: use a "soft top-K" where the weight transitions smoothly:
```
p_k = 1 / (1 + ((b_max - b_k) / (b_max * 1/K))^alpha)
```
This gives weight ~1 to the top K and decreases smoothly for the rest.
