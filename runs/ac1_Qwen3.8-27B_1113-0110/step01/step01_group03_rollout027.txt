This is: minimize ||a - v||^2 s.t. a >= 0, sum(a) = S.

The solution is: a[i] = max(v[i] - tau, 0) where tau is chosen so that sum(a) = S.

To find tau, sort v in decreasing order: v_(1) >= v_(2) >= ... >= v_(n).
For each k, compute tau_k = (sum_{i=1}^{k} v_(i) - S) / k.
The correct k is the largest such that v_(k) > tau_k.
Then tau = tau_k and a[i] = max(v[i] - tau, 0).

Uses smooth max approximation with FFT-based gradients (O(n log n) per iteration),
momentum-based descent with adaptive line search, simulated annealing for escape,

import numpy as np
import time
import math


def _next_pow2(m):
    s = 1
    while s < m:
        s *= 2
    return s


def _fft_self_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:2 * n - 1]


def _smooth_grad(a, c, T):
    """
    Compute gradient of smooth max of self-convolution via FFT cross-correlation.
    Returns (gradient, smooth_value).
    """
    n = len(a)
    c_max = c.max()
    shifted = T * (c - c_max)
    # Clip for numerical stability
    shifted = np.maximum(shifted, -50.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()

    Fa = np.fft.rfft(a, L)

    grad = 2.0 * cross[:n].copy()

def _project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v_clipped = np.maximum(v, 0.0)
    sv = np.sort(v_clipped)[::-1]
    cssv = np.cumsum(sv)
    idx = np.arange(1, n + 1)
    cond = sv * idx > (cssv - S)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.full(n, S / n)
    rho = rho[-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _fast_score(a):
    """Fast score via FFT (may differ from exact by tiny floating point errors)."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = _fft_self_conv(a)
    M = c.max()
    return 2.0 * n * M / (S * S)


def _optimize_one(a0, n, budget, rng, callback):
    """
    Optimize a sequence of length n starting from a0.
    Uses smooth max + FFT gradient + momentum + adaptive step + simulated annealing.
    Calls callback(a) periodically to report progress.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    S0 = a.sum()
    if S0 < 0.01:
        a = np.ones(n)
        S0 = n
    a = a / S0 * n

    momentum = np.zeros(n)
    beta_m = 0.85
    lr = 0.3
    T = 10.0  # smooth max temperature
    T_anneal_rate = 1.00001

    best_local = a.copy()
    best_local_score = _fast_score(a)
    no_improve = 0
    it = 0
    eval_interval = 20
    start_t = time.monotonic()

    while time.monotonic() - start_t < budget:
        # Ensure valid state
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 0.01:
            a = best_local.copy()
            S = a.sum()
        a = a / S * n

        # Compute convolution and gradient
        c = _fft_self_conv(a)
        M = c.max()
        score = 2.0 * M / n  # with sum=n normalization

        grad, _ = _smooth_grad(a, c, T)

        # Momentum update
        momentum = beta_m * momentum - lr * grad
        a_new = a + momentum
        a_new = _project_simplex(a_new, n)

        # Check improvement (fast)
        score_new = _fast_score(a_new)

        if score_new < score - 1e-10:
            a = a_new
            no_improve = 0
        else:
            # Simulated annealing: accept worse with some probability
            delta = score_new - score
            T_sa = max(0.001, 0.01 * math.exp(-it / 5000.0))
            if rng.random() < math.exp(-delta / T_sa):
                a = a_new
            else:
                # Reject: reduce step
                lr *= 0.9
                momentum *= 0.5
                no_improve += 1

        # Adaptive learning rate
        if no_improve >= 20:
            lr = max(lr * 0.5, 1e-5)
            momentum *= 0.3
            no_improve = 0
            # Occasionally do a random perturbation to escape
            if lr < 0.01 and rng.random() < 0.3:
                # Larger perturbation
                n_perturb = max(5, n // 20)
                idx = rng.choice(n, size=n_perturb, replace=False)
                a[idx] = a[idx] * np.exp(rng.normal(0, 0.5, n_perturb))
                a = _project_simplex(a, n)
                momentum = np.zeros(n)
                lr = 0.1

        # Very stuck: bigger escape
        if no_improve >= 50:
            # Blend back toward best_local with noise
            t_blend = 0.3
            a = (1 - t_blend) * best_local + t_blend * a
            a = a * np.exp(rng.normal(0, 0.2, n))
            a = _project_simplex(a, n)
            momentum = np.zeros(n)
            lr = 0.05
            no_improve = 0

        # Anneal temperature
        T *= T_anneal_rate
        T = min(T, 100.0)

        it += 1

        # Periodic evaluation and best tracking
        if it % eval_interval == 0:
            exact_sc = float(evaluate_sequence(a.tolist()))
            if np.isfinite(exact_sc):
                if exact_sc < best_local_score:
                    best_local_score = exact_sc
                    best_local = a.copy()
                    callback(a)

    # Final
    callback(a)
    callback(best_local)
    return best_local, best_local_score


def _fine_tune(a, budget, rng):
    """Very fine optimization with small steps and low temperature."""
    n = len(a)
    a = a.copy()
    a = _project_simplex(a, n)
    momentum = np.zeros(n)
    lr = 0.002
    T = 50.0
    best_a = a.copy()
    best_score = _fast_score(a)
    start_t = time.monotonic()

    while time.monotonic() - start_t < budget:
        a = _project_simplex(a, n)
        c = _fft_self_conv(a)
        grad, _ = _smooth_grad(a, c, T)
        momentum = 0.7 * momentum - lr * grad
        a = _project_simplex(a + momentum, n)

        sc = _fast_score(a)
        if sc < best_score - 1e-8:
            best_score = sc
            best_a = a.copy()

    return best_a


def propose_candidate(seed=42, **kwargs):
    """
    Search for best non-negative sequence minimizing 2n*max(convolve(a,a))/sum(a)^2.
    Uses smooth max + FFT gradients + momentum + simulated annealing + multi-n exploration.
    """
    total_budget = 2360.0
    start_time = time.monotonic()
    rng = np.random.default_rng(seed)

    def time_left():
        return total_budget - (time.monotonic() - start_time)

    # Get parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        parent = None

    best_score = np.inf
    best_seq = None

    def report(a):
        nonlocal best_score, best_seq
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 0.01:
            return
        sc = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            if best_score < 1.52:
                print(f"  New best: {best_score:.8f} (n={len(a)})")

    # Phase 1: Try parent's length with optimization
    n_parent = len(parent) if parent is not None else 2048

    if parent is not None and time_left() > 300:
        print(f"Phase 1: Optimizing n={n_parent} from parent, budget={min(900, time_left()-100):.0f}s")
        a_opt, sc_opt = _optimize_one(parent.copy(), n_parent, min(900.0, time_left() - 200), rng, report)
    elif time_left() > 300:
        n_parent = 2048
        print(f"Phase 1: No parent, trying n={n_parent}")
        a_init = np.ones(n_parent)
        a_opt, sc_opt = _optimize_one(a_init, n_parent, min(900.0, time_left() - 200), rng, report)

    # Phase 2: Try other lengths
    other_ns = [2048, 3072, 4096, 5120]
    other_ns = [x for x in other_ns if x != n_parent]

    for n_try in other_ns:
        if time_left() < 120:
            break
        alloc = min(400.0, time_left() - 60)
        print(f"Phase 2: Trying n={n_try}, budget={alloc:.0f}s")

        # Generate diverse initializations
        best_init = None
        best_init_sc = np.inf
        t = np.linspace(0, 1, n_try)
        inits = [
            np.ones(n_try),
            np.maximum(1.0 + 0.5 * np.sin(2 * np.pi * 3 * t), 0.1),
            np.maximum(1.0 - 0.7 * np.abs(2 * t - 1), 0.05),
            np.maximum(rng.normal(1.0, 0.4, n_try), 0.05),
            np.maximum(rng.exponential(1.0, n_try), 0.01),
            1.0 + 0.3 * np.cos(2 * np.pi * 5 * t),
        ]
        for init in inits:
            init = np.maximum(init, 0.0)
            init = init / init.sum() * n_try
            sc = _fast_score(init)
            if sc < best_init_sc:
                best_init_sc = sc
                best_init = init.copy()

        if best_init is not None:
            a_opt2, sc_opt2 = _optimize_one(best_init, n_try, alloc, rng, report)

    # Phase 3: Fine-tune the best found
    if best_seq is not None and time_left() > 30:
        print(f"Phase 3: Fine-tuning best (score={best_score:.6f}), budget={time_left()-10:.0f}s")
        a_ft = _fine_tune(best_seq.copy(), time_left() - 15, rng)
        report(a_ft)

    # Final validation
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        S = final.sum()
        if S > 0.01:
            report(final)

    print(f"FINAL: score={best_score:.8f}, n={len(best_seq) if best_seq is not None else 0}")

    if best_seq is None:
        # Fallback: uniform sequence
        best_seq = np.ones(256)
        best_score = 2.0

    return [float(x) for x in best_seq]
