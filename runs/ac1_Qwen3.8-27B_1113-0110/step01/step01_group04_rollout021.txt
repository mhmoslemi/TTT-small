Optimization of non-negative step function heights minimizing
2*n*max(convolve(a,a))/(sum(a))^2.

Uses LP-based iterative refinement (dual feasible direction),
smooth gradient descent with continuation, and local search.
"""

import numpy as np

def _pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _clean(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    return np.maximum(v, 0.0)


def _normalize(v):
    v = _clean(v)
    s = float(v.sum())
    if s <= 1e-15:
        return np.full(v.size, 1.0/v.size)
    return v / s


def _project_simplex(v):
    v = _clean(v)
    n = v.size
    if n == 0:
        return v
    x = np.sort(v)[::-1]
    css = np.cumsum(x)
    idx = np.arange(1, n+1, dtype=np.float64)
    cond = x > (css - 1.0) / idx
    r = int(np.nonzero(cond)[0][-1]) if np.any(cond) else 0
    tau = (css[r] - 1.0) / (r + 1)
    out = np.maximum(v - tau, 0.0)
    s = out.sum()
    if s <= 1e-15:
        return np.full(n, 1.0/n)
    return out / s


def _score(a):
    """Compute the score 2*n*max(a*a)/(sum(a))^2 using FFT."""
    v = _normalize(a)
    n = v.size
    L = _pow2(2 * n - 1)
    F = np.fft.rfft(v, n=L)
    c = np.fft.irfft(F * F, n=L)[:2*n-1]
    mx = max(0.0, float(c.max()))
    return 2.0 * n * mx


def _max_conv(a):
    """max of linear convolution of a with itself."""
    v = _clean(a)
    n = v.size
    L = _pow2(2 * n - 1)
    F = np.fft.rfft(v, n=L)
    c = np.fft.irfft(F * F, n=L)[:2*n-1]
    return max(0.0, float(c.max()))


def _convolve_full(a, b):
    """Full linear convolution via FFT."""
    n = a.size
    L = _pow2(2 * max(a.size, b.size) - 1)
    FA = np.fft.rfft(a, n=L)
    FB = np.fft.rfft(b, n=L)
    return np.fft.irfft(FA * FB, n=L)[:a.size + b.size - 1]


def _build_conv_matrix(f, n):
    """Build sparse matrix A where A[k,j] = f[k-j] if 0<=k-j<n."""
    # For each (i,j) pair, the entry goes to position (i+j, j) with value f[i]
    i_idx = np.tile(np.arange(n), n)
    j_idx = np.repeat(np.arange(n), n)
    rows = i_idx + j_idx
    cols = j_idx
    vals = np.tile(f, n)
    return csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))


def _power_seq(n, p=0.5, offset=1.0):
    j = np.arange(n, dtype=np.float64)
    v = np.power(j + offset, -p)
    return _project_simplex(v)

def _smooth_obj_grad(x, eps):
    """Smoothed max of a*a, parameterized by log ratios."""
    x = np.where(np.isfinite(x), x, 0.0)
    n = x.size + 1
    
    # Reconstruct a from log ratios (reference = last element)
    m = max(0.0, float(x.max())) if x.size else 0.0
    e = np.exp(np.clip(x - m, -700, 700))
    e_last = np.exp(-m)
    s = e.sum() + e_last
    a = np.empty(n)
    a[:-1] = e / s
    a[-1] = e_last / s
    
    # Compute a*a
    L = _pow2(2 * n - 1)
    F = np.fft.rfft(a, n=L)
    c = np.fft.irfft(F * F, n=L)[:2*n-1]
    c = np.maximum(c, 0.0)
    
    cmax = float(c.max())
    if cmax <= 1e-300:
        return 0.0, np.zeros_like(x)
    
    z = (c - cmax) / eps
    z = np.clip(z, -700, 0)
    w = np.exp(z)
    sw = float(w.sum())
    if sw <= 0:
        return 1e300, np.zeros_like(x)
    
    f_val = cmax + eps * np.log(sw)
    w = w / sw
    
    # Gradient: d/dx [eps * logsumexp(c/eps)] where c = a*a
    # dc/da = 2 * f (circular conv of a with delta at 0, adjusted)
    # Actually: c_k = sum_i a_i * a_{k-i}
    # dc_k/da_j = a_{k-j} + a_{j-k} (with appropriate bounds)
    # This is the full convolution of a with a, evaluated at specific lags
    # Gradient w.r.t. a: 2 * (w convolved with a) (full conv, truncated to n)
    
    W = np.fft.rfft(w, n=L)
    grad_a = 2.0 * np.fft.irfft(W * np.conj(F), n=L)[:n]
    
    # Project to simplex tangent space (subtract mean weighted by a)
    mean_grad = float(np.dot(a, grad_a))
    grad_full = a * (grad_a - mean_grad)
    grad_x = grad_full[:-1]
    
    if not np.all(np.isfinite(grad_x)):
        grad_x = np.zeros_like(x)
    
    return float(f_val), grad_x
