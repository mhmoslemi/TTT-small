def _pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _clean(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    return np.maximum(v, 0.0)


def _project_simplex(v):
    v = _clean(v)
    n = v.size
    if n == 0:
        return v.copy()
    x = np.sort(v)[::-1]
    css = np.cumsum(x)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = x > (css - 1.0) / idx
    if not np.any(cond):
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    r = int(np.nonzero(cond)[0][-1])
    tau = (css[r] - 1.0) / float(r + 1)
    out = v - tau
    np.maximum(out, 0.0, out=out)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    out *= 1.0 / s
    return out


def _score(a, L=None):
    v = _clean(a)
    n = v.size
    if n == 0:
        return np.inf
    s = float(v.sum())
    if s < 1e-12:
        return np.inf
    v = v / s
    if L is None:
        L = _pow2(2 * n - 1)
    A = np.fft.rfft(v, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    mx = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(mx):
        return np.inf
    return 2.0 * float(n) * mx


def _convolve(a, L=None):
    v = _clean(a)
    n = v.size
    if L is None:
        L = _pow2(2 * n - 1)
    A = np.fft.rfft(v, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _power_sequence(n, p, offset):
    j = np.arange(n, dtype=np.float64)
    x = np.maximum(j + float(offset), 1e-8)
    return _project_simplex(np.power(x, -float(p)))


def _gaussian_sequence(n, sigma_frac, center_frac):
    j = np.arange(n, dtype=np.float64)
    c = center_frac * (n - 1)
    s = max(1.0, sigma_frac * n)
    return _project_simplex(np.exp(-(j - c) ** 2 / (2.0 * s * s)))


def _parametric_sequence(n, alpha, beta, gamma):
    j = np.arange(n, dtype=np.float64)
    t = j / max(1, n - 1)
    return _project_simplex(np.power(1.0 + alpha * t, -beta) * np.exp(-gamma * t))

def _softmax_from_ratios(x):
    x = np.asarray(x, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    n = x.size + 1
    if n == 1:
        return np.ones(1, dtype=np.float64)
    m = 0.0
    if x.size:
        mx = float(np.max(x))
        if mx > m:
            m = mx
    e = np.exp(np.clip(x - m, -745.0, 709.0))
    e_last = float(np.exp(-m))
    s = float(e.sum()) + e_last
    if not np.isfinite(s) or s <= 0.0:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    a = np.empty(n, dtype=np.float64)
    a[:-1] = e / s
    a[-1] = e_last / s
    return a


def _ratios_from_a(a):
    a = _clean(a).copy()
    n = a.size
    if n == 0:
        return a
    s = float(a.sum())
    if s <= 0.0:
        a = np.full(n, 1.0 / float(n))
    else:
        a = a / s
    if n == 1:
        return a
    ref = float(a[-1])
    if ref <= 1e-200:
        a[-1] = 1e-12
        a = a / float(a.sum())
        ref = float(a[-1])
    x = np.log(np.maximum(a[:-1], 1e-300) / max(ref, 1e-300))
    return np.clip(x, -80.0, 80.0)


def _lse_objective_grad(x, eps, L):
    a = _softmax_from_ratios(x)
    n = a.size
    if n <= 1:
        return 0.0, np.zeros_like(x)
    A = np.fft.rfft(a, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    cmax = float(c.max())
    if cmax <= 1e-300:
        return 0.0, np.zeros_like(x)
    z = (c - cmax) / eps
    z = np.clip(z, -745.0, 0.0)
    e = np.exp(z)
    sume = float(e.sum())
    if not np.isfinite(sume) or sume <= 0.0:
        return 1e300, np.zeros_like(x)
    f = cmax + eps * float(np.log(sume))
    w = e / sume
    W = np.fft.rfft(w, n=L)
    ga = 2.0 * np.fft.irfft(W * np.conj(A), n=L)[:n]
    dot = float(np.dot(a, ga))
    gfull = a * (ga - dot)
    gy = gfull[:-1]
    if not np.all(np.isfinite(gy)):
        gy = np.zeros_like(x)
    if not np.isfinite(f):
        f = 1e300
    return f, gy


def _run_lbfgs(fun, x, args, maxiter, bounds):
    if _minimize is None:
        return x, 99
    x = np.asarray(x, dtype=np.float64)
    if x.size == 0:
        return x, 1
    options = {"maxiter": int(maxiter), "disp": 0, "maxfun": int(max(400, 20 * maxiter)), "ftol": 1e-18, "gtol": 1e-14}
    try:
        res = _minimize(fun, x, args=args, method="L-BFGS-B", jac=True, bounds=bounds, options=options)
    except Exception:
        return x, 99
    x_new = np.asarray(res.x, dtype=np.float64)
    x_new = np.where(np.isfinite(x_new), x_new, 0.0)
    x_new = np.clip(x_new, -80.0, 80.0)
    try:
        status = int(res.status)
    except Exception:
        status = 0
    return x_new, status


def _adaptive_lse_optimize(a0, end_time, L, seed):
    """Adaptive log-sum-exp optimization with decreasing temperature."""
    a = _project_simplex(a0)
    n = a.size
    best = a.copy()
    best_score = _score(a, L)
    x = _ratios_from_a(a)
    bounds = [(-80.0, 80.0)] * max(0, n - 1)
    rng = np.random.default_rng(seed)

    # Initial epsilon based on current peak relative to mean
    c = _convolve(a / a.sum(), L)
    cmax = float(c.max())
    cmean = float(c.mean())
    eps = max(1e-6, 0.3 * (cmax - cmean))
    eps_min = 1e-8

    no_improve = 0
    last_improve = time.monotonic()

    while time.monotonic() < end_time:
        remaining = end_time - time.monotonic()
        if remaining < 5.0:
            break
        chunk_iter = min(100, max(30, int(remaining / 2.0)))
        x, status = _run_lbfgs(_lse_objective_grad, x, (eps, L), chunk_iter, bounds)

        a = _softmax_from_ratios(x)
        sc = _score(a, L)
        now = time.monotonic()

        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best = a.copy()
            last_improve = now
            no_improve = 0
            eps = max(eps_min, eps * 0.85)
        else:
            no_improve += 1
            if no_improve >= 3:
                x = np.clip(x + rng.normal(0, 0.05, size=max(0, x.size)), -80.0, 80.0)
                if eps <= eps_min * 1.5:
                    break
                eps = max(eps_min, eps * 0.6)
            if now - last_improve > 30.0:
                break

    return best, best_score


def _subgradient_optimize(a0, end_time, L, seed):
    """Projected subgradient descent on max(c) with momentum."""
    a = _project_simplex(a0)
    n = a.size
    best = a.copy()
    best_score = _score(a, L)
    rng = np.random.default_rng(seed)

    m = np.zeros(n)
    t = 0
    beta = 0.9
    lr = 0.1
    last_improve = time.monotonic()
    start = time.monotonic()

    while time.monotonic() < end_time:
        elapsed = time.monotonic() - start
        c = _convolve(a, L)
        cmax = float(c.max())
        K = np.where(c >= cmax - 1e-12)[0]
        K = K[K < 2 * n - 1]
        if K.size == 0:
            break

        # Subgradient: average over active constraints
        g = np.zeros(n)
        for k in K:
            # grad of c_k wrt a: g_j = a_{k-j} + a_{k-j} (or a_j if 2j=k)
            for j in range(n):
                kj = k - j
                if 0 <= kj < n:
                    g[j] += a[kj]
                    if kj != j:
                        g[j] += a[kj]
        g /= float(K.size)

        # Center gradient (project out constraint sum(a)=1)
        g -= float(np.mean(g))

        # Adaptive step size
        score = 2.0 * float(n) * cmax
        now = time.monotonic()
        if score < best_score - 1e-12:
            best_score = score
            best = a.copy()
            last_improve = now
            lr = min(0.5, lr * 1.02)
        if now - last_improve > 5.0:
            lr = max(0.001, lr * 0.8)
            if now - last_improve > 20.0:
                a = _project_simplex(best * np.exp(rng.normal(0, 0.03, size=n)))
                m = np.zeros(n)
                t = 0

        t += 1
        m = beta * m + (1.0 - beta) * g
        mh = m / (1.0 - beta ** t)
        step = lr * mh / (np.sqrt(np.mean(mh * mh)) + 1e-10)
        a = _project_simplex(a - step)

    return best, best_score


def _local_search(a0, end_time, L, seed):

    a = _project_simplex(a0)
    n = a.size
    best = a.copy()
    best_score = _score(a, L)
    rng = np.random.default_rng(seed)
    sigma = 0.1
    last_improve = time.monotonic()
    start = last_improve

    while time.monotonic() < end_time:
        cand = best.copy()
        nblocks = 1 + int(rng.integers(0, 3))

            w = int(np.exp(rng.uniform(np.log(2.0), np.log(float(maxw)))))
            w = max(1, min(w, n))
            s = int(rng.integers(0, n - w + 1))
            cand[s:s + w] *= float(np.exp(rng.normal(0.0, sigma)))
        cand = _project_simplex(cand)
        sc = _score(cand, L)

        if np.isfinite(sc) and sc < best_score - 1e-13:
            best = cand
            best_score = sc
            last_improve = time.monotonic()
            sigma = min(0.3, sigma * 1.05)
        else:

                    sc2 = _score(cand, L)
                    if np.isfinite(sc2) and sc2 < best_score:

                        best_score = sc2

    if deadline <= t0 + 10.0:

        return [float(x) for x in a]

    # Generate and score candidates
    cands = _generate_candidates(n, seed)
    scored = []
    for i, a in enumerate(cands):
        sc = _score(a, L)
        if np.isfinite(sc):
            scored.append((sc, i, a))
    scored.sort()
    print(f"Generated {len(cands)} candidates, top score: {scored[0][0]:.8f}" if scored else "No valid candidates!")

    if not scored:
        a = np.full(n, 1.0 / float(n))
        return [float(x) for x in a]

    # Deduplicate: keep only distinct top candidates
    top = []
    seen_scores = []
    for sc, i, a in scored:
        if len(top) >= 6:
            break
        if not any(np.allclose(a, top[j][2], atol=1e-6) for j in range(len(top))):
            top.append((sc, i, a))

    global_best = np.asarray(top[0][2], dtype=np.float64).copy()
    global_best_score = float(top[0][0])
    print(f"Initial best: {global_best_score:.8f}")

    # Allocate time among top candidates
    # Give more time to better candidates
    fracs = [0.45, 0.25, 0.15, 0.10, 0.03, 0.02]
    n_top = min(len(top), 6)

    for idx in range(n_top):
        cur_time = time.monotonic()
        if cur_time > deadline - 5.0:
            break
        frac = fracs[idx] if idx < len(fracs) else 0.02
        alloc_end = min(deadline - 3.0, t0 + (sum(fracs[:idx + 1]) * budget))
        if alloc_end <= cur_time + 3.0:
            continue

        print(f"\nRefining candidate {idx+1} (score={top[idx][0]:.8f}), time budget {alloc_end - cur_time:.1f}s")
        a = np.asarray(top[idx][2], dtype=np.float64).copy()
        a, sc = _refine(a, alloc_end, L, int(seed) + 100 * (idx + 1))
        if np.isfinite(sc) and sc < global_best_score:
            global_best_score = sc
            global_best = np.asarray(a, dtype=np.float64).copy()
            print(f"  New global best: {global_best_score:.8f}")

    # Final polish on global best
    if time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining > 10.0:
            print(f"\nFinal polish, {remaining:.1f}s remaining")
            a, sc = _local_search(global_best, deadline, L, int(seed) + 9999)
            if np.isfinite(sc) and sc < global_best_score:
                global_best_score = sc
                global_best = np.asarray(a, dtype=np.float64).copy()
                print(f"  Final score: {global_best_score:.8f}")

    global_best = _project_simplex(global_best)
    out = [float(x) for x in global_best]

    # Validate
    try:
        val = evaluate_sequence(out)
        print(f"\nFinal evaluation: {val:.8f}")
        if not np.isfinite(val):
            out = [float(x) for x in _power_sequence(len(out), 0.5, 1.0)]
            print("Fallback to power sequence")
    except Exception as e:
        print(f"Validation error: {e}")

    return out
