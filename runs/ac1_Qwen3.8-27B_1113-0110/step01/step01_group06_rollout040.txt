"""
Optimize a non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses full-space mirror descent with adaptive top-K focusing, momentum,
and multiple restart strategies. Starts from the given sequence when available.
"""

import numpy as np
import time


def _pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _fft_square(a):
    n = int(a.size)
    L = 2 * n - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, n=N)
    b = np.fft.irfft(fa * fa, n=N).real[:L]
    b = np.maximum(b, 0.0)
    return b


def _fft_corr(w, a):
    """Compute correlation: IFFT(FFT(w) * conj(FFT(a)))[:len(a)]"""
    n = int(a.size)
    L = int(w.size)
    N = _pow2(L + n - 1)
    fw = np.fft.rfft(w, n=N)
    fa = np.fft.rfft(a, n=N)
    c = np.fft.irfft(fw * np.conj(fa), n=N)[:n]
    return 2.0 * c


def _score(a):
    s = float(a.sum())
    if s < 1e-300:
        return float('inf')
    b = _fft_square(a)
    return float(2.0 * a.size * b.max() / (s * s))


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-300:
        return np.ones_like(a) / a.size
    return a / s


def _top_k_weights(b, K, tail_weight=0.01):
    """Create weights that focus on the top K coefficients."""
    L = int(b.size)
    if K >= L:
        return np.ones(L) / L
    idx = np.argsort(b)[::-1][:K]
    w = np.full(L, tail_weight, dtype=np.float64)
    w[idx] = 1.0
    w /= w.sum()
    return w


def _softmax_weights(b, temperature):
    """Create softmax-based weights."""
    L = int(b.size)
    maxb = b.max()
    if maxb < 1e-300:
        return np.ones(L) / L
    with np.errstate(over='ignore', under='ignore'):
        z = (b - maxb) / (temperature * maxb + 1e-300)
        p = np.exp(z)
    s = p.sum()
    if s < 1e-300:
        p = np.zeros(L)
        p[np.argmax(b)] = 1.0
    else:
        p /= s
    return p


def _optimize_mirror(a0, time_limit, rng, verbose=False, label=""):
    """Mirror descent with adaptive focusing and momentum."""
    a = _normalize(a0.copy()).astype(np.float64)
    n = a.size
    b = _fft_square(a)
    score = float(2.0 * n * b.max() / (a.sum()**2))
    best_a = a.copy()
    best_score = score
    
    # Momentum buffer
    momentum = np.zeros(n, dtype=np.float64)
    beta = 0.9
    
    eta = 1.0
    eta_min = 0.01
    eta_max = 50.0
    
    t = time.monotonic()
    t0 = t
    it = 0
    no_improve = 0
    max_no_improve = 2000
    
    # Adaptive K
    K = 1
    K_target = 1
    K_schedule = [(0, 1), (500, 2), (1500, 3), (3000, 5), (5000, 8), (8000, 13), (12000, 21)]
    
    while t - t0 < time_limit:
        # Update K based on iteration count
        for threshold, k_val in K_schedule:
            if it >= threshold:
                K_target = k_val
        K = K_target
        
        # Compute weights
        if K <= 1:
            w = _softmax_weights(b, 50.0)  # Very sharp focus on max
        else:
            # Mix: top-K weights + smooth background
            w_top = _top_k_weights(b, K)
            w_smooth = _softmax_weights(b, max(1.0, K * 2.0))
            w = 0.7 * w_top + 0.3 * w_smooth
        
        # Compute gradient
        grad = _fft_corr(w, a)
        
        # Remove the component along the "all-ones" direction (since we normalize)
        grad = grad - float(np.dot(a, grad) / a.sum()) * a
        
        # Normalize gradient
        gmax = float(np.max(np.abs(grad)))
        if gmax < 1e-20:
            break
        grad_norm = grad / gmax
        
        # Momentum update
        momentum = beta * momentum + (1 - beta) * grad_norm
        
        # Mirror descent step
        a_new = a * np.exp(-eta * momentum)
        a_new = np.maximum(a_new, 1e-300)
        a_new = _normalize(a_new)
        
        # Evaluate
        b_new = _fft_square(a_new)
        score_new = float(2.0 * n * b_new.max() / (a_new.sum()**2))
        
        # Accept/reject
        if score_new < score:
            a = a_new
            b = b_new
            score = score_new
            eta = min(eta * 1.1, eta_max)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            eta = max(eta * 0.8, eta_min)
            no_improve += 1
            # If stuck, add noise
            if eta < 0.1:
                a = a * np.exp(rng.normal(0, 0.1, n))
                a = _normalize(a)
                b = _fft_square(a)
                score = float(2.0 * n * b.max() / (a.sum()**2))
                momentum = np.zeros(n, dtype=np.float64)
                eta = 1.0
        
        if no_improve > max_no_improve:
            # Big restart
            a = 0.5 * best_a + 0.5 * _normalize(np.exp(rng.normal(0, 1, n)))
            a = _normalize(a)
            b = _fft_square(a)
            score = float(2.0 * n * b.max() / (a.sum()**2))
            momentum = np.zeros(n, dtype=np.float64)
            eta = 1.0
            no_improve = 0
            max_no_improve = int(max_no_improve * 1.5)
        
        it += 1
        t = time.monotonic()
        
        if verbose and it % 5000 == 0:
            print(f"  [{label}] it={it} t={t-t0:.1f}s score={score:.6f} best={best_score:.6f} K={K} eta={eta:.2f}")
    
    return best_a, best_score


def _optimize_anti(a0, time_limit, rng, verbose=False, label=""):
    """Optimize in the anti-palindromic subspace."""
    a = _normalize(a0.copy()).astype(np.float64)
    n = a.size
    if n % 2 != 0:
        return _optimize_mirror(a, time_limit, rng, verbose, label)
    
    m = n // 2
    # Extract anti-palindromic parameter: x[i] = a[i] * m / 2 for i < m
    # Constraint: a[i] + a[n-1-i] = m * x[i] * (2/n) = x[i]
    # Actually let's use: a[i] = (x[i]/n) * 2, a[n-1-i] = (1 - x[i])/n * 2

    # So a[i] = 2*x[i]/n for i<m, a[n-1-i] = 2*(1-x[i])/n for i<m
    # where x[i] in [0, 1]

    x = np.clip(x, 1e-10, 1.0 - 1e-10)
    
    def x_to_a(x):
        aa = np.empty(n, dtype=np.float64)
        aa[:m] = 2.0 * x / n
        aa[m:] = 2.0 * (1.0 - x[::-1]) / n
        return aa
    
    a = x_to_a(x)
    b = _fft_square(a)
    score = float(2.0 * n * b.max() / (a.sum()**2))
    best_x = x.copy()
    best_score = score

    eta_max = 50.0

    temp = 50.0

        b_new = _fft_square(a_new)
        score_new = float(2.0 * n * b_new.max() / (a_new.sum()**2))

            a = a_new
            b = b_new
            score = score_new
            eta = min(eta * 1.05, eta_max)

                best_score = score
                best_x = x.copy()

            else:

            no_improve += 1
            if eta < 0.1:
                x = np.clip(x + rng.normal(0, 0.1, m), 1e-10, 1-1e-10)
                a = x_to_a(x)
                b = _fft_square(a)
                score = float(2.0 * n * b.max() / (a.sum()**2))
                momentum = np.zeros(m)
                eta = 1.0
        
        if no_improve > max_no:
            x = np.clip(0.5 * best_x + 0.5 * rng.uniform(0.1, 0.9, m), 1e-10, 1-1e-10)
            a = x_to_a(x)
            b = _fft_square(a)
            score = float(2.0 * n * b.max() / (a.sum()**2))
            momentum = np.zeros(m)
            eta = 1.0
            no_improve = 0
