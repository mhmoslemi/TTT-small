def _next_pow2(m):

def _normalize(a, floor=0.0):

def _convolve_full(a):
    n = a.size
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)

def _build_basis(n, n_modes=12):
    x = np.arange(n, dtype=np.float64) / n
    cols = [np.ones(n)]
    for m in range(1, n_modes + 1):
        cols.append(np.sin(2 * np.pi * m * x))
        cols.append(np.cos(2 * np.pi * m * x))
    # Add some polynomial modes
    for m in range(2, 5):
        p = x ** m - (x ** m).mean()
        cols.append(p)
    # Add antisymmetric modes (key for reducing center coefficient)
    center = 0.5
    for m in range(1, 6):
        cols.append(np.sin(np.pi * m * (x - 0.5) * 2))
    B = np.column_stack(cols)
    # Orthogonalize
    for i in range(B.shape[1]):
        for j in range(i):
            B[:, i] -= (B[:, i] @ B[:, j]) / max(1e-14, B[:, j] @ B[:, j]) * B[:, j]
    norms = np.linalg.norm(B, axis=0)
    norms[norms < 1e-10] = 1.0
    B = B / norms
    return B

            old = loga[idx].copy()
            loga[idx] += delta_mag * rng.uniform(0.5, 1.0, size)
            
            aa = np.exp(np.clip(loga - loga.max(), -60, 0))
            aa = np.maximum(aa, 1e-300)
            aa = aa / aa.sum()
            c = _conv_with_fft(aa, N)
            sc = 2.0 * n * float(c.max())
            
            if sc < best_sc * (1 - 1e-10):
                best_sc = sc
                best_a = aa.copy()
                no_improve = 0
                break
            else:
                loga[idx] = old
        else:
            no_improve += 1
            if no_improve > 2000:
                # Random restart from best
                loga = np.log(best_a + 1e-20) - np.log(best_a + 1e-20).max()
                no_improve = 0
        
        it += 1
        if it % 10000 == 0:
            print(f"  coord n={n} it={it} best={best_sc:.8f}", flush=True)
    
    return best_a, best_sc


def _resample(a, n_target):
    a = _normalize(a, 0)
    n_src = a.size
    if n_src == n_target:
        return a.copy()
    old_x = np.linspace(0, 1, n_src)
    new_x = np.linspace(0, 1, n_target)
    b = np.interp(new_x, old_x, a)
    return _normalize(b, 1e-12)


def _get_initial_sequences(n, seed):
    """Generate diverse initial sequences."""
    rng = np.random.default_rng(seed)
    seqs = []
    
    # Uniform
    seqs.append(np.ones(n) / n)
    
    # Exponential / anti-symmetric profiles
    for alpha in (1, 2, 3, 5, 8, 1, 2, 3, 5, 8):
        x = np.linspace(0, 1, n)
        if alpha <= 3:
            a = np.exp(alpha * x)
        else:
            a = np.exp(-alpha * x)
        seqs.append(_normalize(a, 1e-12))
    
    # Sine-based
    for m in range(1, 8):
        for amp in (0.3, 0.6, 0.9):
            x = np.linspace(0, 1, n)
            a = 1 + amp * np.sin(2 * np.pi * m * x)
            seqs.append(_normalize(a, 1e-12))
    
    # Anti-symmetric (key for reducing center coefficient)
    for m in range(1, 6):
        x = np.linspace(0, 1, n)
        a = 1 + 0.8 * (-1)**m * np.sin(np.pi * m * (2*x - 1))
        seqs.append(_normalize(a, 1e-12))
    
    # Gaussian blobs (off-center)
    for mu in (0.25, 0.35, 0.65, 0.75):
        for sig in (0.05, 0.1, 0.2):
            x = np.linspace(0, 1, n)
            a = np.exp(-(x-mu)**2 / (2*sig**2))
            seqs.append(_normalize(a, 1e-12))
    
    # Random smooth
    for _ in range(5):
        a = rng.random(n)
        # Smooth it
        for _ in range(3):
            a = np.convolve(a, np.ones(5)/5, mode='same')
        seqs.append(_normalize(a, 1e-12))
    
    # Costas-like (binary with specific structure)
    for step in (2, 3, 5, 7):
        a = np.zeros(n)
        for i in range(0, n//2, 2):
            a[i] = 1.0
            a[(i*step) % n] = 1.0
        if a.sum() > 0:
            seqs.append(_normalize(a, 1e-12))
    
    return seqs


def propose_candidate(budget_s=2340, **kwargs):
    """Main search function. Returns the best sequence found."""
    start_time = time.monotonic()
    deadline = start_time + float(budget_s)
    
    # Get initial sequence if available
    a_given = None
    try:
        hs = globals().get('height_sequence_1', None)
        if hs is not None:
            a_given = _normalize(np.asarray(hs, dtype=np.float64).ravel(), 0)
            if a_given.size < 2:
                a_given = None
    except Exception:
        a_given = None
    
    best_a = None
    best_score = np.inf
    
    def update_best(a, sc):
        nonlocal best_a, best_score
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"NEW BEST: score={sc:.10f}, n={a.size}, t={time.monotonic()-start_time:.1f}s", flush=True)
    
    # === PHASE 1: Quick multi-start to get a good initial point ===
    t_phase1_end = start_time + 60
    
    for n in (512, 1024, 2048):
        if time.monotonic() > t_phase1_end:
            break
        seqs = _get_initial_sequences(n, seed=42 + n)
        if a_given is not None:
            seqs.insert(0, _resample(a_given, n))
        for s in seqs:
            sc = _score_internal(s)
            update_best(s, sc)
    
    print(f"Phase 1 done: best={best_score:.8f}, t={time.monotonic()-start_time:.1f}s", flush=True)
    
    # === PHASE 2: Low-dimensional smooth optimization ===
    t_phase2_end = start_time + 180
    
    for n in (1024, 2048, 4096):
        if time.monotonic() > t_phase2_end - 30:
            break
        B = _build_basis(n, n_modes=min(15, n // 64))
        a_init = _resample(best_a, n)
        sub_deadline = min(t_phase2_end, time.monotonic() + 50)
        a2, sc2 = _lowdim_optimize(n, B, sub_deadline, a_init=a_init, n_starts=5, seed=100+n)
        update_best(a2, sc2)
    
    print(f"Phase 2 done: best={best_score:.8f}, t={time.monotonic()-start_time:.1f}s", flush=True)
    
    # === PHASE 3: Mirror descent (main optimization) ===
    t_phase3_end = start_time + 900
    
    for n in (1024, 2048, 4096):
        if time.monotonic() > t_phase3_end - 60:
            break
        a_init = _resample(best_a, n)
        allow = min(200.0, t_phase3_end - time.monotonic() - 30)
        if allow < 30:
            break
        a3, sc3 = _mirror_descent(a_init, time.monotonic() + allow, seed=200+n)
        update_best(a3, sc3)
        
        # Second round with different momentum
        if time.monotonic() < t_phase3_end - 60 and n == 2048:
            a3b, sc3b = _mirror_descent(a_init * (1 + 0.05 * np.random.default_rng(7).random(n)), 
                                         time.monotonic() + min(100, t_phase3_end - time.monotonic() - 30),
                                         seed=999, momentum_decay=0.8)
            update_best(a3b, sc3b)
    
    print(f"Phase 3 done: best={best_score:.8f}, t={time.monotonic()-start_time:.1f}s", flush=True)
    
    # === PHASE 4: Simulated annealing for exploration ===
    t_phase4_end = start_time + 1800
    
    for n in (2048, 4096):
        if time.monotonic() > t_phase4_end - 60:
            break
        a_init = _resample(best_a, n)
        allow = min(300.0, t_phase4_end - time.monotonic() - 30)
        if allow < 30:
            break
        a4, sc4 = _sim_anneal_v2(a_init, time.monotonic() + allow, seed=500+n)
        update_best(a4, sc4)
        
        # Also try from a perturbed version
        if time.monotonic() < t_phase4_end - 60:
            a_pert = _normalize(a_init + 0.02 * np.random.default_rng(n).random(n), 1e-12)
            allow2 = min(150.0, t_phase4_end - time.monotonic() - 30)
            if allow2 > 20:
                a4b, sc4b = _sim_anneal_v2(a_pert, time.monotonic() + allow2, seed=700+n)
                update_best(a4b, sc4b)
    
    print(f"Phase 4 done: best={best_score:.8f}, t={time.monotonic()-start_time:.1f}s", flush=True)
    
    # === PHASE 5: Final polish ===
    t_phase5_end = deadline - 20
    
    for n in (2048, 4096):
        if time.monotonic() > t_phase5_end - 20:
            break
        a_init = _resample(best_a, n)
        allow = min(120.0, t_phase5_end - time.monotonic() - 10)
        if allow < 15:
            break
        # Mirror descent final
        a5, sc5 = _mirror_descent(a_init, time.monotonic() + allow * 0.6, seed=1000+n, momentum_decay=0.95)
        update_best(a5, sc5)
        # Coord polish
        if time.monotonic() < t_phase5_end - 10:
            a5b, sc5b = _coord_polish(_resample(best_a, n), time.monotonic() + allow * 0.4, seed=1100+n)
            update_best(a5b, sc5b)
    
    # Try a few more sizes
    for n in (8192,):
        if time.monotonic() > t_phase5_end:
            break
        a_init = _resample(best_a, n)
        allow = min(60.0, t_phase5_end - time.monotonic())
        if allow > 10:
            a6, sc6 = _mirror_descent(a_init, time.monotonic() + allow, seed=2000)
            update_best(a6, sc6)
    
    # === FINAL: Verify and return ===
    if best_a is None:
        best_a = np.ones(1024) / 1024
    
    best_a = _normalize(best_a, 0.0)
    out_list = [float(x) for x in best_a]
    
    # Verify with actual evaluator
    try:
        final_eval = float(evaluate_sequence(out_list))
    except Exception:
        final_eval = np.inf
    
    if not np.isfinite(final_eval) or final_eval > best_score * 1.01:
        # Fallback to a safe sequence
        best_a = np.ones(1024) / 1024
        out_list = [float(x) for x in best_a]
        try:
            final_eval = float(evaluate_sequence(out_list))
        except Exception:
            final_eval = 2.0
    
    print(f"\nFINAL: internal={best_score:.10f}, evaluated={final_eval:.10f}, n={len(out_list)}, time={time.monotonic()-start_time:.1f}s", flush=True)
    return out_list
