import numpy as np

def _pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _fft_square(a):
    n = int(a.size)
    L = 2 * n - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, n=N)
    b = np.fft.irfft(fa * fa, n=N).real[:L]
    b = np.maximum(b, 0.0)
    return b


def _fft_grad(weights, a):
    n = int(a.size)
    L = int(weights.size)
    N = _pow2(L + n - 1)
    fw = np.fft.rfft(weights, n=N)
    fa = np.fft.rfft(a, n=N)
    c = np.fft.irfft(fw * np.conj(fa), n=N)[:n]
    return 2.0 * c


def _score(a, b=None):

        b = _fft_square(a)
    s = float(a.sum())
    if s < 1e-300:
        return float('inf'), b
    maxb = float(b.max())
    sc = 2.0 * a.size * maxb / (s * s)
    return sc, b


def _weights_multi(b, focuses):

    L = int(b.size)

        w_total += w

"""
Optimize non-negative step function sequence to minimize
    2 * n * max(convolve(a, a)) / (sum(a))**2.

Strategy:
- Use L_p norm gradient (finite p) for smooth optimization, transitioning
  to direct max-targeting as convergence approaches.
- Multi-start: anti-palindromic initialization at large n, plus optimization
  of the given sequence.
- Adaptive time allocation: invest more time in the most promising candidates.
- Periodic "spectral flattening" steps to escape local minima.
- Momentum-based updates for faster convergence.

import numpy as np
import time


def _pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _fft_square(a):
    n = int(a.size)
    L = 2 * n - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, n=N)
    b = np.fft.irfft(fa * fa, n=N).real[:L]
    b = np.maximum(b, 0.0)
    return b


def _fft_grad(weights, a):
    n = int(a.size)
    L = int(weights.size)
    N = _pow2(L + n - 1)
    fw = np.fft.rfft(weights, n=N)
    fa = np.fft.rfft(a, n=N)
    c = np.fft.irfft(fw * np.conj(fa), n=N)[:n]
    return 2.0 * c


def _compute_score(a, b=None):
    if b is None:
        b = _fft_square(a)
    s = float(a.sum())
    if s < 1e-300:
        return float('inf'), b, 0.0
    maxb = float(b.max())
    sc = 2.0 * a.size * maxb / (s * s)
    return sc, b, maxb


def _lp_weights(b, p, eps=0.01):
    """Compute weights proportional to b^(p-1), normalized."""
    L = int(b.size)
    maxb = float(b.max())
    if maxb < 1e-300:
        return np.full(L, 1.0 / L)
    # Use (b/maxb)^(p-1) for numerical stability
    with np.errstate(over='ignore', under='ignore', invalid='ignore'):
        w = np.power(b / maxb, float(p) - 1.0)
    w = np.nan_to_num(w, nan=0.0, posinf=0.0)
    s = float(w.sum())
    if s < 1e-300:
        w = np.zeros(L)
        w[int(np.argmax(b))] = 1.0
    else:
        w = w / s
    # Add small uniform component for stability
    w = (1.0 - eps) * w + eps / L
    return w


def _softmax_weights(b, focus, eps=0.01):
    """Classic softmax weights on (max - b)."""
    L = int(b.size)
    maxb = float(b.max())
    if maxb < 1e-300:
        return np.full(L, 1.0 / L)
    with np.errstate(over='ignore', under='ignore'):
        z = (maxb - b) / maxb
        p = np.exp(-z * float(focus))
    s = float(p.sum())
    if s < 1e-300:
        p = np.zeros(L)
        p[int(np.argmax(b))] = 1.0
    else:
        p = p / s
    p = (1.0 - eps) * p + eps / L
    return p


def _mixed_weights(b, focus, p_lp):
    """Mix softmax and L_p weights for robust gradient."""
    w1 = _softmax_weights(b, focus, eps=0.01)
    w2 = _lp_weights(b, p_lp, eps=0.01)
    # Blend: weight the L_p more as we converge
    blend = 0.4
    w = (1.0 - blend) * w1 + blend * w2
    return w


def _spectral_flat_step(a, alpha=0.3, rng=None):
    """Attempt to flatten the power spectrum and recover a non-negative sequence."""
    n = int(a.size)
    N = _pow2(2 * n - 1)
    A = np.fft.rfft(a, n=N)
    P = A.real**2 + A.imag**2
    # Target: geometric mean of P (flat spectrum)
    logP = np.log(P + 1e-300)
    target_logP = float(np.mean(logP))
    # Blend toward target
    new_logP = (1.0 - alpha) * logP + alpha * target_logP
    new_P = np.exp(new_logP)
    # Recover complex spectrum preserving phase
    phase = A / (np.abs(A) + 1e-300)
    new_A = np.sqrt(new_P + 0j) * phase
    # Inverse FFT
    a_new = np.fft.irfft(new_A, n=N)[:n]
    # Project to non-negative
    a_new = np.maximum(a_new, 0.0)
    s = float(a_new.sum())
    if s < 1e-300:
        return a.copy()
    a_new /= s
    return a_new

def _make_anti_pal_init(n, rng, amp=2.0):
    """Create an anti-palindromic initial sequence."""
    m = n // 2
    if m < 1:
        return np.ones(n) / n

    v = rng.normal(0, amp, size=m)
    v -= v.mean()
    std = v.std()
    if std > 1e-12:
        v = (amp * v) / std

    a = np.empty(n)
    a[:m] = (1.0 + v) / n
    if n % 2 == 0:
        a[m:] = (1.0 - v[::-1]) / n
    else:
        a[m + 1:] = (1.0 - v[::-1]) / n
        a[m] = 1.0 / n

    a = np.maximum(a, 1e-15)
    a /= a.sum()
    return a


def _greedy_anti_pal_init(n, rng):
    """Greedy construction of anti-palindromic sequence with low autocorrelation."""
    m = n // 2
    if m < 1:
        return np.ones(n) / n

    # Greedy: place "mass" to minimize peak autocorrelation
    # Work in the anti-palindromic parameterization
    # x_i in [0,1], a_i = x_i/n, a_{n-1-i} = (1-x_i)/n
    # Start with all x = 0.5 (uniform)
    x = np.full(m, 0.5)
    a = _x_to_a(x, n)
    b = _fft_square(a)
    max_b = b.max()

    # Iteratively adjust x to reduce peaks
    for _ in range(200):
        peak_idx = int(np.argmax(b))
        # Determine which x_i affect b[peak_idx] most
        # b[peak_idx] = sum_i a_i * a[peak_idx - i]
        # In anti-palindromic param, this is complex. Just do random perturbation
        # on the x that are "responsible"
        
        # Simpler: just do a random coordinate descent step
        idx = rng.integers(0, m)
        delta = rng.uniform(-0.1, 0.1)
        x_new = x.copy()
        x_new[idx] = np.clip(x_new[idx] + delta, 0.01, 0.99)
        a_new = _x_to_a(x_new, n)
        b_new = _fft_square(a_new)
        if b_new.max() < max_b:
            x = x_new
            a = a_new
            b = b_new
            max_b = b.max()
        else:
            # Try opposite direction
            x_new[idx] = np.clip(x[idx] - delta, 0.01, 0.99)
            a_new = _x_to_a(x_new, n)
            b_new = _fft_square(a_new)
            if b_new.max() < max_b:
                x = x_new
                a = a_new
                b = b_new
                max_b = b.max()

    return a

    for n_trial in n_candidates:
        remaining = phase2_deadline - time.monotonic()
        if remaining < 60:
            break
        alloc = min(180.0, remaining - 30.0)
        if alloc < 40:
            break

        rng2 = np.random.default_rng(seed + 100 + n_trial)
        print(f"\n[Phase 2] Anti-palindromic n={n_trial} (alloc={alloc:.0f}s)")
        t0 = time.monotonic()
        try:
            a_init = _make_anti_pal_init(n_trial, rng2, amp=3.0)
            a2, s2 = _optimize_advanced(a_init, alloc, rng2, label=f"anti{n_trial}")
            if s2 < best_score:
                best_score = s2
                best_a = a2
                print(f"  New best from anti-pal: {s2:.6f}")
        except Exception as e:
            print(f"Phase 2 (n={n_trial}) failed: {e}")
        print(f"Phase 2 (n={n_trial}) done: best={best_score:.6f} (time={time.monotonic()-t0:.1f}s)")

    # Phase 2b: Try full random initialization at a good size
    remaining = phase2_deadline - time.monotonic()
    if remaining > 90:
        n_rand = 8192
        alloc = min(150.0, remaining - 30.0)
        if alloc > 50:
            rng3 = np.random.default_rng(seed + 200)
            print(f"\n[Phase 2b] Random full init n={n_rand} (alloc={alloc:.0f}s)")
            t0 = time.monotonic()
            try:
                a_rand = np.exp(rng3.normal(0, 2.0, n_rand))
                a_rand = np.maximum(a_rand, 1e-15)
                a_rand /= a_rand.sum()
                a3, s3 = _optimize_advanced(a_rand, alloc, rng3, label=f"rand{n_rand}")
                if s3 < best_score:
                    best_score = s3
                    best_a = a3
                    print(f"  New best from random: {s3:.6f}")
            except Exception as e:
                print(f"Phase 2b failed: {e}")
            print(f"Phase 2b done: best={best_score:.6f} (time={time.monotonic()-t0:.1f}s)")

    # Phase 3: Deep refinement of best candidate
    if best_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            alloc = remaining - 30.0
            print(f"\n[Phase 3] Deep refinement (alloc={alloc:.0f}s, n={best_a.size})")
            rng4 = np.random.default_rng(seed + 500)
            t0 = time.monotonic()
            try:
                a_final, s_final = _optimize_advanced(best_a, alloc, rng4, label="refine")
                if s_final < best_score:
                    best_score = s_final
                    best_a = a_final
            except Exception as e:
                print(f"Phase 3 failed: {e}")
            print(f"Phase 3 done: best={best_score:.6f} (time={time.monotonic()-t0:.1f}s)")

    # Final: try one more short round if time permits
    if best_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 40:
            alloc = remaining - 20.0
            rng5 = np.random.default_rng(seed + 999)
            print(f"\n[Phase 4] Final polish (alloc={alloc:.0f}s)")
            try:
                a_pol, s_pol = _optimize_advanced(best_a, alloc, rng5, label="polish")
                if s_pol < best_score:
                    best_score = s_pol
                    best_a = a_pol
            except Exception as e:
                print(f"Phase 4 failed: {e}")

    # Prepare final output
    if best_a is None:
        best_a = np.ones(1024, dtype=np.float64) / 1024.0

    best_a = np.nan_to_num(best_a, nan=0.0, posinf=1000.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    ss = float(best_a.sum())
    if ss <= 0.01:
        best_a = np.ones(1024, dtype=np.float64) / 1024.0
        ss = 1.0
    else:
        best_a /= ss

    final_seq = [float(x) for x in best_a]

    # Verify with exact evaluation if possible
    if best_a.size <= 20000 and time.monotonic() < deadline:
        exact = _safe_evaluate(final_seq)
        if np.isfinite(exact):
            print(f"\n=== FINAL: internal={best_score:.6f}, exact={exact:.6f}, n={best_a.size} ===")
        else:
            print(f"\n=== FINAL: internal={best_score:.6f}, exact=inf, n={best_a.size} ===")
    else:
        print(f"\n=== FINAL: internal={best_score:.6f}, n={best_a.size} ===")

    return final_seq
