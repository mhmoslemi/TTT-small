def next_pow2(m):
    if m <= 1:

    return 1 << (int(m) - 1).bit_length()


def normalize_positive(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, float(floor), None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def conv_full(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()

    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2*n-1], fa, N

        c, fa, _ = fft_conv_full(a, N)
        cm = float(c.max())
        sc = float(2.0 * n * cm)
        if sc < bests:
            bests = sc
            best = a.copy()

        # Find the worst lag and try to reduce it by adjusting coordinates
        kstar = int(np.argmax(c))
        # Gradient at kstar: for position m, d/d a_m of c[kstar]
        # = a[kstar - m] (if valid) + a[m - kstar] (if valid)
        g = np.zeros(n)
        for m in range(n):
            v = 0.0
            if 0 <= kstar - m < 2 * n - 1:
                idx = kstar - m
                if idx < n:
                    v += a[idx]
            if 0 <= m - kstar < 2 * n - 1:
                idx = m - kstar
                if idx < n:
                    v += a[idx]
            g[m] = 2.0 * v

        # Try adjusting the coordinate with highest gradient
        # Use a small multiplicative update
        for trial in range(8):
            # Pick random coordinate
            m = int(rng.integers(0, n))
            old = a[m]
            # Try scaling
            for factor in (0.7, 0.85, 1.15, 1.3):
                if m == n - 1:
                    a[m] = max(0, old * factor)
                elif m == 0:
                    a[m] = max(0, old * factor)
                else:
                    a[m] = max(0, old * factor)
                # Renormalize
                a = norm_pos(a, 0.0)
                ct = np.fft.irfft(np.fft.rfft(a, N) ** 2, N)[:2 * n - 1]
                sct = float(2.0 * n * ct.max())
                if sct < sc * (1 - 1e-10):
                    sc = sct
                    if sc < bests:
                        bests = sc
                        best = a.copy()
                    break
                else:
                    a[m] = old
                    a = norm_pos(a, 0.0)

    return best, bests


def build_basis(n, n_params=40):
    """Build a basis for low-dimensional parameterization."""
    i = np.arange(n, dtype=np.float64)
    x = (i - 0.5 * (n - 1)) / max(1.0, 0.5 * n)
    cols = []
    for m in range(1, 12):
        cols.append(np.sin(np.pi * m * x))
        cols.append(np.cos(np.pi * m * x))
    cols.append(x.copy())
    cols.append(x * x)
    cols.append(x ** 3 - x * (x ** 2).mean())
    for m in range(1, 8):
        t = 2.0 * np.pi * m * i / n
        cols.append(np.sin(t))
        cols.append(np.cos(t))
    # Random smooth kernels
    rng = np.random.default_rng(7 + n)
    for _ in range(6):
        center = rng.uniform(0, 1)
        width = rng.uniform(0.05, 0.3)
        cols.append(np.exp(-((x - center) ** 2) / (2 * width * width)))
    B = np.column_stack(cols[:n_params])
    B -= B.mean(axis=0)
    norms = np.linalg.norm(B, axis=0)
    norms[norms < 1e-14] = 1.0
    B /= norms
    return B


def softmax_a(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = z - z.max()
    z = np.clip(z, -60, 0)
    e = np.exp(z)
    e = np.maximum(e, 1e-300)
    s = e.sum()
    if s <= 0:
        return np.full(z.size, 1.0 / z.size)
    return e / s


def lowdim_opt(n, deadline, seed, init_a=None, n_params=30):
    """Optimize in low-dim log-param space."""
    B = build_basis(n, n_params)
    D = B.shape[1]
    N = next_pow2(3 * n - 2)

    # Initialize theta
    if init_a is not None:
        z0 = np.log(np.maximum(init_a, 1e-12))
        z0 -= z0.mean()
        theta, _, _, _ = np.linalg.lstsq(B, z0, rcond=None)
        theta = theta.astype(np.float64)
    else:
        rng = np.random.default_rng(seed)
        theta = rng.normal(0, 0.3, D)

    best_a = softmax_a(B @ theta)
    bests = my_score(best_a, N)

    if _so is None:
        return best_a, bests

    def obj_f(theta_v):
        a_v = softmax_a(B @ theta_v)
        return my_score(a_v, N)

    for tau_scale in (0.3, 0.1, 0.04, 0.015):
        if time.monotonic() > deadline:
            break
        # Use smooth approximation via L-BFGS
        maxit = 15 if n <= 2048 else 10
        try:
            res = _so.minimize(obj_f, theta, method='L-BFGS-B',
                                options={'maxiter': maxit, 'ftol': 1e-10, 'gtol': 1e-5, 'maxls': 10})
            if res.x is not None and np.all(np.isfinite(res.x)):
                theta = res.x
        except Exception:
            pass
        a_new = softmax_a(B @ theta)
        sc = my_score(a_new, N)
        if np.isfinite(sc) and sc < bests:
            bests = sc
            best_a = a_new.copy()

    # Multi-start
    rng = np.random.default_rng(seed + 999)
    for trial in range(3):
        if time.monotonic() > deadline:
            break
        th2 = theta + rng.normal(0, 0.5, D)
        try:
            res = _so.minimize(obj_f, th2, method='L-BFGS-B',
                                options={'maxiter': 8, 'ftol': 1e-10, 'gtol': 1e-5, 'maxls': 8})
            if res.x is not None and np.all(np.isfinite(res.x)):
                a_new = softmax_a(B @ res.x)
                sc = my_score(a_new, N)
                if np.isfinite(sc) and sc < bests:
                    bests = sc
                    best_a = a_new.copy()
        except Exception:
            pass

    return best_a, bests


def binary_search_seq(a0, deadline, seed=0):
    """Binary local search: swap 1s and 0s to reduce max convolution."""
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    m = max(2, int(round(0.5 * n)))
    # Threshold to binary
    order = np.argsort(a0, kind='stable')
    b = np.zeros(n)
    b[order[-m:]] = 1.0
    N = next_pow2(2 * n - 1)
    c, _, _ = fft_conv_full(b, N)
    cmax = float(c.max())
    sc = float(2.0 * n * cmax / (m * m))
    best_b = b.copy()
    bests = sc

    rng = np.random.default_rng(seed)
    temp = 0.3
    stag = 0
    it = 0
    two_b = 2.0 * b

    while time.monotonic() < deadline:
        if it % 500 == 0:
            c, _, _ = fft_conv_full(b, N)
            cmax = float(c.max())
            sc = float(2.0 * n * cmax / (m * m))
            if sc < bests:
                bests = sc
                best_b = b.copy()

        # Find positions with 1 and 0
        one_idx = np.nonzero(b > 0.5)[0]
        zero_idx = np.nonzero(b < 0.5)[0]
        if one_idx.size == 0 or zero_idx.size == 0:
            break

        # Weighted random selection biased toward high-convolution areas
        # Compute pressure per position
        # For a swap of 1 at i to 0, and 0 at j to 1:
        # delta c[k] = -(a[k-i] + a[k+i] ... ) for removing, +... for adding
        # Simplified: just try a few random swaps
        for trial in range(4):
            i = int(one_idx[rng.integers(0, one_idx.size)])
            j = int(zero_idx[rng.integers(0, zero_idx.size)])
            if i == j:
                continue
            # Perform swap
            # c changes: remove contribution of b[i]=1, add contribution of b[j]=1
            # New c = old c - 2*b[k-i] - b[2k-i type terms] ... complex, just recompute for small arrays
            if n <= 4096:
                old_ci = c[i:i + n].copy()
                old_cj = c[j:j + n].copy()
                b[i] = 0.0
                b[j] = 1.0
                two_b = 2.0 * b
                c[i:i + n] -= 2.0 * b
                c[2 * i] -= 1.0 if 2 * i < len(c) else 0
                c[j:j + n] += 2.0 * b
                c[2 * j] += 1.0 if 2 * j < len(c) else 0
                new_max = float(c.max())
                new_sc = float(2.0 * n * new_max / (m * m))

                accept = False
                if new_sc < sc * (1 - 1e-12):
                    accept = True
                elif new_sc <= sc * 1.01 and rng.random() < np.exp(-(new_sc - sc) / temp):
                    accept = True

                if accept:
                    sc = new_sc
                    if sc < bests:
                        bests = sc
                        best_b = b.copy()
                    stag = 0
                    temp = max(0.01, temp * 0.999)
                else:
                    # Revert
                    c[i:i + n] += 2.0 * b
                    c[2 * i] += 1.0 if 2 * i < len(c) else 0
                    c[j:j + n] -= 2.0 * b
                    c[2 * j] -= 1.0 if 2 * j < len(c) else 0
                    b[i] = 1.0
                    b[j] = 0.0
                    two_b = 2.0 * b
                    stag += 1
                    temp = max(0.005, temp * 0.97)

        it += 1
        if stag > 300:
            # Re-heat
            temp = min(0.5, temp * 3.0)
            stag = 0
            if it % 2000 == 0:
                # Full recompute to fix drift
                c, _, _ = fft_conv_full(b, N)
                cmax = float(c.max())
                sc = float(2.0 * n * cmax / (m * m))
                if sc < bests:
                    bests = sc
                    best_b = b.copy()

    return best_b, bests


def log_anneal(a, deadline, seed=77):
    """Simulated annealing in log-space with block moves."""
    a = norm_pos(a, 1e-15)
    n = a.size
    N = next_pow2(3 * n - 2)
    best = a.copy()
    bests = my_score(a, N)
    rng = np.random.default_rng(seed)
    temp = 0.003
    sc = bests
    it = 0
    stag = 0

    while time.monotonic() < deadline:
        # Block move
        bw = int(rng.integers(max(1, n // 5000), max(2, n // 20) + 1))
        start = int(rng.integers(0, n - bw + 1))
        sigma = temp * 5.0

        old = a[start:start + bw].copy()
        pert = np.exp(rng.normal(0, sigma, bw))
        a[start:start + bw] = old * pert
        a = norm_pos(a, 1e-15)
        c, _, _ = fft_conv_full(a, N)
        sc_new = float(2.0 * n * c.max())

        if sc_new < sc:
            sc = sc_new
            if sc < bests:
                bests = sc
                best = a.copy()
            stag = 0
            temp = max(0.0001, temp * 0.999)
        else:
            a[start:start + bw] = old
            a = norm_pos(a, 1e-15)
            stag += 1
            temp = max(0.00005, temp * 0.995)
            if rng.random() < np.exp(-(sc_new - sc) / max(0.001, temp * 30)):
                a[start:start + bw] = old * pert
                a = norm_pos(a, 1e-15)
                sc = sc_new

        it += 1
        if stag > 800:
            # Big kick
            idx = int(rng.integers(0, n))
            a[idx] *= 2.0
            a = norm_pos(a, 1e-15)
            c, _, _ = fft_conv_full(a, N)
            sc = float(2.0 * n * c.max())
            if sc < bests:
                bests = sc
                best = a.copy()
            stag = 0
            temp = 0.002

    return best, bests


def propose_candidate(seed=42, budget_s=2370, **kwargs):
    start = time.monotonic()
    deadline = start + max(100.0, float(budget_s) - 25.0)

    # Get initial best
    best_a = None
    best_score = np.inf

    given = get_given()
    if given is not None and given.size > 8:
        sc = my_score(given)
        if np.isfinite(sc):
            best_score = sc
            best_a = given.copy()
            print("Given sequence: n={} score={:.10f}".format(given.size, best_score), flush=True)

    if best_a is None:
        best_a = np.ones(2048) / 2048.0
        best_score = 2.0
        print("Fallback uniform n=2048 score=2.0", flush=True)

    n_base = best_a.size

    # Phase 1: Quick mirror descent from given (n_base)
    t1_end = min(deadline - 1200, start + 180)
    if t1_end > start + 30:
        a1, sc1 = mirror_desc(best_a, t1_end, verbose=True, seed=seed)
        if np.isfinite(sc1) and sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()
            print("Phase1 mirror n={} score={:.10f}".format(n_base, best_score), flush=True)

    # Phase 2: Low-dim optimization at multiple sizes
    for n_try in (1024, 2048, 4096):
        if time.monotonic() > deadline - 900:
            break
        allow = min(120.0, deadline - time.monotonic() - 900)
        if allow < 30:
            break
        a_init = resample(best_a, n_try)
        a2, sc2 = lowdim_opt(n_try, time.monotonic() + allow, seed=seed + n_try, init_a=a_init, n_params=min(35, n_try // 2))
        if np.isfinite(sc2) and sc2 < best_score:
            best_score = sc2
            best_a = a2.copy()
            print("Phase2 lowdim n={} score={:.10f}".format(n_try, best_score), flush=True)

    # Phase 3: Mirror descent on best at its native size
    n_cur = best_a.size
    t3_end = min(deadline - 600, time.monotonic() + 250)
    if t3_end > time.monotonic() + 30:
        a3, sc3 = mirror_desc(best_a, t3_end, verbose=True, seed=seed + 500)
        if np.isfinite(sc3) and sc3 < best_score:
            best_score = sc3
            best_a = a3.copy()
            print("Phase3 mirror n={} score={:.10f}".format(n_cur, best_score), flush=True)

    # Phase 4: Try larger n
    for n_big in (4096, 8192):
        if time.monotonic() > deadline - 400:
            break
        allow = min(180.0, deadline - time.monotonic() - 400)
        if allow < 40:
            break
        a_big = resample(best_a, n_big)
        a4, sc4 = mirror_desc(a_big, time.monotonic() + allow, verbose=False, seed=seed + n_big)
        if np.isfinite(sc4) and sc4 < best_score:
            best_score = sc4
            best_a = a4.copy()
            print("Phase4 big n={} score={:.10f}".format(n_big, best_score), flush=True)

    # Phase 5: Binary refinement
    n_bin = min(best_a.size, 4096)
    if n_bin >= 256:
        if time.monotonic() < deadline - 200:
            allow = min(150.0, deadline - time.monotonic() - 200)
            if allow > 20:
                a_bin_init = resample(best_a, n_bin)
                a5, sc5 = binary_search_seq(a_bin_init, time.monotonic() + allow, seed=seed + 9000)
                if np.isfinite(sc5) and sc5 < best_score:
                    best_score = sc5
                    best_a = a5.copy()
                    print("Phase5 binary n={} score={:.10f}".format(n_bin, best_score), flush=True)

    # Phase 6: Log annealing polish
    if time.monotonic() < deadline - 60:
        allow = min(120.0, deadline - time.monotonic() - 40)
        if allow > 15:
            a6, sc6 = log_anneal(best_a, time.monotonic() + allow, seed=seed + 8000)
            if np.isfinite(sc6) and sc6 < best_score:
                best_score = sc6
                best_a = a6.copy()
                print("Phase6 anneal score={:.10f}".format(best_score), flush=True)

    # Phase 7: Final coordinate descent polish
    if time.monotonic() < deadline - 20:
        allow = min(60.0, deadline - time.monotonic() - 10)
        if allow > 10:
            a7, sc7 = coord_descent(best_a, time.monotonic() + allow, seed=seed + 12345)
            if np.isfinite(sc7) and sc7 < best_score:
                best_score = sc7
                best_a = a7.copy()
                print("Phase7 coord score={:.10f}".format(best_score), flush=True)

    # Finalize
    best_a = norm_pos(best_a, 0.0)
    out_list = [float(x) for x in best_a]

    try:
        feval = float(evaluate_sequence(out_list))
    except Exception:
        feval = float('inf')

    if not np.isfinite(feval):
        # Fallback
        best_a = np.ones(2048) / 2048.0
        out_list = [float(x) for x in best_a]
        feval = 2.0

    print("FINAL: internal={:.12f} evaluated={:.12f} n={}".format(best_score, feval, len(out_list)), flush=True)
    return out_list
