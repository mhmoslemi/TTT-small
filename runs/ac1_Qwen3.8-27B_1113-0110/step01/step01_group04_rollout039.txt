2. Find the k* that maximizes c[k*]
3. The gradient of c[k*] with respect to a[i] is: `sum_j a[k*-i] * delta(j, k*-i)` ... actually, `dc[k*]/da[i] = sum_j delta(i+j, k*) * a[j] = a[k*-i]` (if k*-i is valid).
4. So the gradient of the "peak" is `a[k*-i]` for valid i.
5. A subgradient step would be: decrease a[i] proportionally to a[k*-i] for the i that contribute to the peak.
6. But we need to maintain sum(a) = 1 and a >= 0.

This is essentially what the smoothed max objective does, but peak-shaving is more targeted.

Let me implement a "weighted" version: instead of just the max, use a weighted sum of the top few peaks, with weights based on how close they are to the max.

"""
Search for optimal non-negative step function heights minimizing 2*n*max(a*a)/(sum(a))^2.
Uses diverse initial candidates (power laws, modulated, two-sided), parameter optimization,
FFT-based gradient methods, and peak-shaving local search targeting convolution peaks.

import numpy as np

from scipy.optimize import minimize, minimize_scalar


def _pow2(m):
    m = int(max(1, m))
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _clean(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    return np.maximum(v, 0.0)


def _project_simplex(v):
    v = _clean(v)
    n = v.size
    if n == 0:
        return v.copy()
    x = np.sort(v)[::-1]
    css = np.cumsum(x)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = x > (css - 1.0) / idx
    if not np.any(cond):
        return np.full(n, 1.0 / float(n))
    r = int(np.nonzero(cond)[0][-1])
    tau = (css[r] - 1.0) / float(r + 1)
    out = v - tau
    np.maximum(out, 0.0, out=out)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.full(n, 1.0 / float(n))
    return out / s


def _score(a, L=None):
    v = _clean(a)
    n = v.size
    if n == 0:
        return np.inf
    s = float(v.sum())
    if s < 1e-12:
        return np.inf
    v = v / s
    if L is None:
        L = _pow2(2 * n - 1)
    A = np.fft.rfft(v, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return np.inf
    return 2.0 * float(n) * mx


def _convolve_a(a, L):
    A = np.fft.rfft(a, n=L)
    C = A * A
    return np.fft.irfft(C, n=L)


def _gen_power(n, p, offset=1.0):
    j = np.arange(n, dtype=np.float64)
    x = np.maximum(j + offset, 1e-8)
    return _project_simplex(np.power(x, -p))


def _gen_modulated_power(n, p, offset, alpha, freq, phase=0.0):
    j = np.arange(n, dtype=np.float64)
    x = np.maximum(j + offset, 1e-8)
    base = np.power(x, -p)
    t = j / max(1, n - 1)
    mod = 1.0 + alpha * np.cos(2.0 * np.pi * freq * t + phase)
    return _project_simplex(np.maximum(base * mod, 0.0))


def _gen_two_sided(n, p1, p2, w):
    j = np.arange(n, dtype=np.float64)
    left = np.power(np.maximum(j + 1.0, 1e-8), -p1)
    right = np.power(np.maximum(n - j, 1e-8), -p2)
    v = left + w * right
    return _project_simplex(v)


def _gen_tapered_power(n, p, offset, taper_beta, taper_pow):
    j = np.arange(n, dtype=np.float64)
    x = np.maximum(j + offset, 1e-8)
    base = np.power(x, -p)
    t = j / max(1, n - 1)
    window = 1.0 - taper_beta * np.power(t, taper_pow)
    return _project_simplex(np.maximum(base * window, 0.0))


def _gen_exp_power(n, beta, gamma):
    j = np.arange(n, dtype=np.float64)
    t = j / max(1, n - 1)
    v = np.exp(-beta * t) * np.power(j + 1.0, -gamma)
    return _project_simplex(v)


def _gen_bump(n, center_frac, width_frac, sharpness):
    j = np.arange(n, dtype=np.float64)
    t = j / max(1, n - 1)
    center = center_frac
    width = max(0.01, width_frac)
    v = np.exp(-sharpness * ((t - center) / width) ** 2)
    return _project_simplex(v)

def _soft_max_obj_grad(x, a_ref, eps):
    """
    Smoothed max of convolution, parameterized as log-ratios to last element of a_ref.
    """
    a = _a_from_ratios(x, a_ref)
    n = a.size
    if n <= 1:
        return 0.0, np.zeros_like(x)

    L = _pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    c = np.maximum(c, 0.0)

    cmax = float(c.max())
    if cmax < 1e-300:
        return 0.0, np.zeros_like(x)

    z = (c - cmax) / eps
    z = np.clip(z, -700.0, 0.0)
    e = np.exp(z)
    se = float(e.sum())
    if se < 1e-300:
        return 0.0, np.zeros_like(x)

    f = cmax + eps * float(np.log(se))
    w = e / se

    W = np.fft.rfft(w, n=L)
    ga = 2.0 * np.fft.irfft(W * np.conj(A), n=L)[:n]
    dot = float(np.dot(a, ga))
    gfull = a * (ga - dot)
    gy = gfull[:-1]

    if not np.all(np.isfinite(gy)):
        gy = np.zeros_like(x)
    return f, gy


def _l2_obj_grad(x, a_ref):
    """L2 norm squared of convolution as surrogate."""
    a = _a_from_ratios(x, a_ref)
    n = a.size
    if n <= 1:
        return 0.0, np.zeros_like(x)

    L = _pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    c = np.maximum(c, 0.0)

    f = float(np.dot(c, c))
    W = np.fft.rfft(c, n=L)
    ga = 4.0 * np.fft.irfft(W * np.conj(A), n=L)[:n]
    dot = float(np.dot(a, ga))
    gfull = a * (ga - dot)
    gy = gfull[:-1]

    if not np.all(np.isfinite(gy)):
        gy = np.zeros_like(x)
    return f, gy


def _weighted_peak_obj_grad(x, a_ref, top_k=5):
    """
    Objective: sum of top-k peaks of convolution (smoothed with soft assignment).
    """
    a = _a_from_ratios(x, a_ref)
    n = a.size
    if n <= 1:
        return 0.0, np.zeros_like(x)

    L = _pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    c = np.maximum(c, 0.0)

    # Soft top-k weights
    cmax = float(c.max())
    if cmax < 1e-300:
        return 0.0, np.zeros_like(x)
    
    # Use a soft threshold: weight = sigmoid((c - threshold)/tau)
    threshold = cmax * 0.8
    tau = cmax * 0.1 + 1e-10
    w = 1.0 / (1.0 + np.exp(-(c - threshold) / tau))
    w = w / max(float(w.sum()), 1e-300)

    f = float(np.dot(w, c))
    W = np.fft.rfft(w, n=L)
    ga = 2.0 * np.fft.irfft(W * np.conj(A), n=L)[:n]
    dot = float(np.dot(a, ga))
    gfull = a * (ga - dot)
    gy = gfull[:-1]

    if not np.all(np.isfinite(gy)):
        gy = np.zeros_like(x)
    return f, gy


def _ratios_from_a(a):
    a = _clean(a).copy()
    n = a.size
    if n == 0:
        return a
    s = float(a.sum())
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    ref = float(a[-1])
    if ref < 1e-200:
        a[-1] = 1e-12
        a = a / float(a.sum())
        ref = float(a[-1])
    x = np.log(np.maximum(a[:-1], 1e-300) / max(ref, 1e-300))
    return np.clip(x, -100.0, 100.0)


def _a_from_ratios(x, ref_val=None):
    x = np.asarray(x, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    n = x.size + 1
    if n == 1:
        return np.ones(1)
    m = 0.0
    if x.size:
        mx = float(np.max(x))
        if mx > m:
            m = mx
    e = np.exp(np.clip(x - m, -700.0, 700.0))
    e_last = float(np.exp(-m))
    s = float(e.sum()) + e_last
    if s < 1e-300:
        return np.full(n, 1.0 / n)
    a = np.empty(n)
    a[:-1] = e / s
    a[-1] = e_last / s
    return a


def _peak_shaving(a, end_time, n_iter_max=20000):
    """
    Targeted peak shaving: repeatedly find the convolution peak and apply
    a localized reduction to the contributing elements.
    """
    a = _project_simplex(a.copy())
    n = a.size
    L = _pow2(2 * n - 1)
    best = a.copy()
    best_score = _score(best, L)
    lr = 0.05
    rng = np.random.default_rng(12345)
    last_improve = time.monotonic()
    no_improve = 0

    for it in range(n_iter_max):
        if time.monotonic() > end_time:
            break
        if it % 500 == 0 and time.monotonic() - last_improve > 30.0:
            break

        c = _convolve_a(a, L)[:2 * n - 1]
        c = np.maximum(c, 0.0)
        k_star = int(np.argmax(c))
        peak_val = c[k_star]

        if peak_val < 1e-15:
            break

        # Compute gradient at the peak: dc[star]/da[i] = a[star - i] (if valid)
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = a[j]

        # Normalize gradient
        gnorm = float(np.sqrt(np.dot(grad, grad)))
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        # Subtract gradient scaled by learning rate (projected)
        step = lr * grad
        a_new = a - step
        a_new = _project_simplex(a_new)

        sc = _score(a_new, L)
        if np.isfinite(sc) and sc < best_score:
            best = a_new.copy()
            best_score = sc
            a = a_new
            last_improve = time.monotonic()
            no_improve = 0
            lr = min(0.15, lr * 1.05)
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                lr = max(0.001, lr * 0.8)
            if no_improve > 2000:
                # Perturb and reset
                a = _project_simplex(best * np.exp(rng.normal(0, 0.05, size=n)))
                lr = 0.02
                no_improve = 0

    return best, best_score


def _adam_simplex_search(a0, end_time, seed=42):
    """
    Adam-style optimizer on the simplex with weighted-peak gradient.
    Progressively sharpens the peak weighting.
    """
    a = _project_simplex(a0.copy())
    n = a.size
    L = _pow2(2 * n - 1)
    best = a.copy()
    best_score = _score(best, L)

    rng = np.random.default_rng(seed)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = 5e-4
    beta1, beta2 = 0.9, 0.999
    last_improve = time.monotonic()
    start = time.monotonic()
    total_time = max(1.0, end_time - start)

    while time.monotonic() < end_time:
        fr = min(1.0, (time.monotonic() - start) / total_time)

        c = _convolve_a(a, L)[:2 * n - 1]
        c = np.maximum(c, 0.0)
        cmax = float(c.max())
        if cmax < 1e-15:
            break

        # Progressive sharpening: start with broad weighting, narrow to peak
        tau_param = 0.3 * (1.0 - 0.7 * fr)
        threshold = cmax * (1.0 - tau_param)
        tau = cmax * 0.05 * (1.0 - fr) + 1e-10
        w = 1.0 / (1.0 + np.exp(-(c - threshold) / tau))
        w = w / max(float(w.sum()), 1e-300)

        W = np.fft.rfft(w, n=L)
        ga = 2.0 * np.fft.irfft(W * np.conj(A := np.fft.rfft(a, n=L)), n=L)[:n]
        dot = float(np.dot(a, ga))
        grad = a * (ga - dot)

        # Normalize
        rms = float(np.sqrt(np.mean(grad * grad))) + 1e-12
        grad = grad / rms

        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        mh = m / (1 - beta1 ** t)
        vh = v / (1 - beta2 ** t)
        step = lr * mh / (np.sqrt(vh) + 1e-8)

        a_new = _project_simplex(a - step)
        a_new = _clean(a_new)
        s = float(a_new.sum())
        if s < 1e-10:
            a = best.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            continue
        a_new = a_new / s
        a = a_new

        sc = _score(a, L)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = a.copy()
            last_improve = time.monotonic()
            lr = min(2e-3, lr * 1.03)
        else:
            gap = time.monotonic() - last_improve
            if gap > 5.0:
                lr = max(1e-5, lr * 0.8)
            if gap > 30.0:
                a = _project_simplex(best * np.exp(rng.normal(0, 0.03, size=n)))
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                lr = 3e-4

    return best, best_score


def _refine_with_lbfgs(a0, end_time, seed):
    """
    Multi-stage L-BFGS-B refinement using smoothed surrogates.
    """
    a = _project_simplex(a0.copy())
    n = a.size
    L = _pow2(2 * n - 1)
    best = a.copy()
    best_score = _score(best, L)
    x = _ratios_from_a(a)
    bounds = [(-100.0, 100.0)] * max(1, n - 1)
    start = time.monotonic()
    total = max(1.0, end_time - start)
    rng = np.random.default_rng(seed)

    # Stage 1: L2 surrogate (broad smoothing)
    t1_end = start + 0.15 * total
    while time.monotonic() < t1_end:
        try:
            res = minimize(_l2_obj_grad, x, args=(a,), method='L-BFGS-B',
                          jac=True, bounds=bounds,
                          options={'maxiter': 50, 'disp': 0, 'ftol': 1e-18, 'gtol': 1e-12})
            x = np.clip(np.asarray(res.x, dtype=np.float64), -100, 100)
            x = np.where(np.isfinite(x), x, 0.0)
        except Exception:
            break

    a = _a_from_ratios(x)
    sc = _score(a, L)
    if np.isfinite(sc) and sc < best_score:
        best, best_score = a, sc

    # Stage 2: Soft max with decreasing epsilon
    t2_end = end_time - max(10.0, 0.1 * total)
    eps = max(1e-6, 0.05 * (best_score / (2.0 * n)))
    eps_min = 1e-8
    no_improve = 0
    last_imp = time.monotonic()

    while time.monotonic() < t2_end:
        chunk_end = min(time.monotonic() + 10.0, t2_end)
        while time.monotonic() < chunk_end:
            try:
                res = minimize(_soft_max_obj_grad, x, args=(a, eps), method='L-BFGS-B',
                              jac=True, bounds=bounds,
                              options={'maxiter': 40, 'disp': 0, 'ftol': 1e-18, 'gtol': 1e-12})
                x = np.clip(np.asarray(res.x, dtype=np.float64), -100, 100)
                x = np.where(np.isfinite(x), x, 0.0)
            except Exception:
                break

        a = _a_from_ratios(x)
        sc = _score(a, L)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best, best_score = a.copy(), sc
            last_imp = time.monotonic()
            no_improve = 0
            eps = max(eps_min, eps * 0.7)
        else:
            no_improve += 1
            if no_improve >= 3 and eps > eps_min * 1.1:
                eps = max(eps_min, eps * 0.5)
            elif no_improve >= 5:
                x = np.clip(x + rng.normal(0, 0.03, size=max(1, x.size)), -100, 100)
                no_improve = 0
            if no_improve >= 8:
                break

    return best, best_score


def _block_local_search(a0, end_time, seed):
    """Random multiplicative block perturbations."""
    a = _project_simplex(a0.copy())
    n = a.size
    L = _pow2(2 * n - 1)
    best = a.copy()
    best_score = _score(best, L)
    rng = np.random.default_rng(seed)
    sigma = 0.1
    last_imp = time.monotonic()
    start = time.monotonic()

    while time.monotonic() < end_time:
        cand = best.copy()
        nb = int(rng.integers(1, 3))
        for _ in range(nb):
            maxw = max(3, n // 10)
            w = int(np.exp(rng.uniform(np.log(2.0), np.log(float(maxw)))))
            w = max(1, min(w, n))
            s = int(rng.integers(0, n - w + 1))
            cand[s:s + w] *= float(np.exp(rng.normal(0, sigma)))
        cand = _project_simplex(cand)
        sc = _score(cand, L)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best, best_score = cand, sc
            last_imp = time.monotonic()
            sigma = min(0.3, sigma * 1.08)
        else:
            if time.monotonic() - last_imp > 8.0:
                sigma = max(0.01, sigma * 0.7)
                if time.monotonic() - last_imp > 30.0:
                    cand = _project_simplex(best * np.exp(rng.normal(0, 0.04, size=n)))
                    cs = _score(cand, L)
                    if np.isfinite(cs) and cs < best_score:
                        best, best_score = cand, cs
                        last_imp = time.monotonic()

    return best, best_score
