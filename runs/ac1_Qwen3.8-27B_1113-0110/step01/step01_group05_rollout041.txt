def _clean(arr):

    return a / s


def _score(a, c=None):

def _coord_descent(p, c, score, deadline, rng, grid_size=30, refine=8, max_perturb=8):
    """Coordinate descent with periodic perturbations."""
    n = int(p.size)
    best_p = p.copy()
    best_c = c.copy()
    best_score = float(score)

    p = p.copy()
    c = c.copy()
    score = float(score)

    grid = np.logspace(-8.0, 2.0, grid_size)

    stagnant = 0
    perts = 0
    noise_scales = [0.8, 0.5, 0.3, 0.15, 0.08, 0.04]

    while time.monotonic() < deadline - 0.5:
        order = rng.permutation(n)
        improved = False
        for i in order:
            if time.monotonic() > deadline - 0.3:
                break
            p, c, score, imp = _coord_update(p, c, int(i), grid, refine)
            if imp:
                improved = True

        # Renormalize
        s = float(np.sum(p))
        if not np.isfinite(s) or s <= 0:
            break
        p = p / s
        c = np.convolve(p, p)
        np.clip(c, 0.0, None, out=c)
        score = _score_norm(p, c)

        tol = 1e-12 * max(1.0, best_score)
        if score < best_score - tol:
            best_score = score
            best_p = p.copy()
            best_c = c.copy()
            stagnant = 0 if improved else stagnant
        else:
            if improved:
                stagnant = 0
            else:
                stagnant += 1

        # Perturbation
        if stagnant >= 3 and perts < max_perturb:
            scale = noise_scales[min(perts, len(noise_scales) - 1)]
            # Structured perturbation: low-frequency + random
            freq = rng.integers(1, 5)
            phase = rng.uniform(0, 2 * np.pi)
            j = np.arange(n, dtype=np.float64)
            perturb = np.sin(2 * np.pi * freq * j / n + phase) * 0.1 + rng.normal(0, scale, n)
            p_new = best_p * np.exp(perturb)
            p_new = np.clip(p_new, 0.0, None)
            s_new = float(np.sum(p_new))
            if np.isfinite(s_new) and s_new > 0:
                p_new = p_new / s_new
                c_new = np.convolve(p_new, p_new)
                sc_new = _score_norm(p_new, c_new)
                if np.isfinite(sc_new) and sc_new < best_score:
                    p, c, score = p_new, c_new, sc_new
                    best_p = p_new.copy()
                    best_c = c_new.copy()
                    best_score = sc_new
                else:
                    p, c, score = p_new, c_new, sc_new
            perts += 1
            stagnant = 0
        elif stagnant >= 8:
            # Try a bigger restart
            p = best_p * np.exp(rng.normal(0, 0.5, n))
            p = np.clip(p, 0.0, None)
            s = float(np.sum(p))
            if s > 0:
                p = p / s
                c = np.convolve(p, p)
                score = _score_norm(p, c)
            stagnant = 0
            perts += 1

        if perts >= max_perturb and stagnant >= 5:
            break

    return best_p, best_c, best_score


def _block_descent(p, n_blocks, deadline, rng):
    """Block-level coordinate descent: optimize block means."""
    n = int(p.size)
    block_sizes = np.full(n_blocks, n // n_blocks, dtype=np.int64)

    block_sizes[:rem] += 1

    best_score = _score(p)

    # Extract block means
    block_means = np.zeros(n_blocks)
    idx = 0
    for b in range(n_blocks):
        bs = int(block_sizes[b])
        block_means[b] = float(np.mean(p[idx:idx + bs]))
        idx += bs

    grid = np.logspace(-6.0, 1.5, 25)

    stagnant = 0
    for _ in range(200):
        if time.monotonic() > deadline:
            break

        improved = False
        for b in rng.permutation(n_blocks):
            # Try different values for block b
            old_mean = block_means[b]
            idx = int(np.sum(block_sizes[:b]))
            bs = int(block_sizes[b])

            best_val = best_score
            best_x = old_mean
            for x in grid:
                # Set block b to value x
                p_trial = p.copy()
                p_trial[idx:idx + bs] = x
                p_trial = np.clip(p_trial, 0.0, None)
                s = float(np.sum(p_trial))
                if s <= 0:
                    continue
                p_trial = p_trial / s
                sc = _score(p_trial)
                if sc < best_val:
                    best_val = sc
                    best_x = x

            if best_x != old_mean:
                p[idx:idx + bs] = best_x
                p = np.clip(p, 0.0, None)
                s = float(np.sum(p))
                if s > 0:
                    p = p / s
                    best_score = _score(p)
                    block_means[b] = best_x
                    improved = True

        if improved:
            stagnant = 0
        else:
            stagnant += 1
            if stagnant >= 3:
                break

    # Update best
    sc = _score(p)
    if sc < best_score:
        best_score = sc
    return p, best_score


def _expand_prototype(proto, target_len):
    """Expand a short prototype to target length by repeating blocks."""
    target_len = int(target_len)
    p = _norm(proto)
    R = int(p.size)
    if R <= 0 or target_len <= 0:
        return np.ones(max(1, target_len), dtype=np.float64)
    if R == target_len:
        return p.copy()
    if R > target_len:
        # Downsample
        out = np.zeros(target_len, dtype=np.float64)
        for j in range(target_len):
            lo = (j * R) // target_len
            hi = max(lo + 1, ((j + 1) * R) // target_len)
            out[j] = float(np.mean(p[lo:hi]))
        return _norm(out)

    base = target_len // R
    rem = target_len % R
    lengths = np.full(R, base, dtype=np.int64)
    lengths[:rem] += 1
    out = np.zeros(target_len, dtype=np.float64)
    idx = 0
    for i in range(R):
        L = int(lengths[i])
        if L > 0:
            out[idx:idx + L] = p[i] / float(L)
            idx += L
    return _norm(out)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Return the best non-negative coefficient sequence found within budget_s.
    Multi-phase: parametric search -> coordinate descent -> block descent -> polish.
    """
    budget = max(1.0, float(budget_s))
    start_time = time.monotonic()
    deadline = start_time + max(0.0, budget - 5.0)
    rng = np.random.default_rng(int(seed))

    FINAL_LEN = 3809
    best_seq = None
    best_score = float('inf')

    def try_update(seq, sc):
        nonlocal best_seq, best_score
        if seq is None or not np.isfinite(sc) or sc >= best_score:
            return
        best_score = float(sc)
        best_seq = _norm(seq).copy()
        print(f"  New best: score={sc:.6f}, len={seq.size}, t={time.monotonic()-start_time:.1f}s")

    # Seed from height_sequence_1 if available
    try:
        hseq = _clean(height_sequence_1)
        if hseq.size > 0:
            sc = _score(hseq)
            if np.isfinite(sc):
                try_update(hseq, sc)
                print(f"  Start from height_sequence_1: score={sc:.6f}, len={hseq.size}")
    except Exception:
        pass

    # Baseline: two-level
    try:
        base = _make_two_level(FINAL_LEN, FINAL_LEN // 2, 1.5)
        sc = _score(base)
        try_update(base, sc)
    except Exception:
        pass

    if time.monotonic() > deadline - 30:
        if best_seq is None:
            best_seq = _make_two_level(FINAL_LEN, FINAL_LEN // 2, 1.5)
        return [float(x) for x in best_seq]

    # === Phase 1: Parametric search on short prototypes (0 - 300s) ===
    print("=" * 60)
    print("Phase 1: Parametric search")
    print("=" * 60)
    phase1_deadline = min(deadline - 300.0, start_time + 280.0)
    proto_len = 96  # prototype length for parametric search

    proto_results = []

    # Try various parametric families on short prototypes
    parametric_tries = [
        ('linear_ramp', lambda n, a: _make_linear_ramp(n, a), np.array([0.5]), [(0.0, 1.5)]),
        ('power_ramp', lambda n, a, b: _make_power_ramp(n, a, b), np.array([0.5, 2.0]), [(0.0, 1.5), (0.5, 5.0)]),
        ('two_level', lambda n, m, r: _make_two_level(n, m, r), np.array([0.3, 2.0]), [(0.1, 0.9), (0.5, 5.0)]),
        ('three_level', lambda n, m1, m2, r1, r3: _make_three_level(n, m1, m2, r1, r3),
         np.array([0.2, 0.5, 0.5, 2.0]), [(0.05, 0.4), (0.3, 0.7), (0.2, 2.0), (0.5, 5.0)]),
        ('gaussian', lambda n, s, m: _make_gaussian_tilt(n, s, m), np.array([0.5, 0.3]), [(0.1, 2.0), (0.0, 1.0)]),
        ('exponential', lambda n, r: _make_exponential(n, r), np.array([0.95]), [(0.9, 1.0)]),
    ]

    for name, func, x0, bounds in parametric_tries:
        if time.monotonic() > phase1_deadline:
            break
        try:
            a_opt, sc_opt = _evaluate_parametric(proto_len, func, x0, bounds)
            if np.isfinite(sc_opt):
                try_update(a_opt, sc_opt)
                proto_results.append((sc_opt, a_opt))
                print(f"  {name}: score={sc_opt:.6f}")
        except Exception as e:
            print(f"  {name}: error {e}")

    # Also try random starts on prototype length
    for trial in range(20):
        if time.monotonic() > phase1_deadline:
            break
        strategy = trial % 6
        if strategy == 0:
            p = rng.dirichlet(np.full(proto_len, 1.0))
        elif strategy == 1:
            p = rng.dirichlet(np.full(proto_len, 0.3))
        elif strategy == 2:
            p = np.exp(rng.normal(0, 0.5, proto_len))
            p = p / p.sum()
        elif strategy == 3:
            p = _make_linear_ramp(proto_len, rng.uniform(0.3, 1.2))
        elif strategy == 4:
            p = _make_two_level(proto_len, int(rng.integers(10, proto_len - 10)), rng.uniform(1.2, 4.0))
        else:
            p = _make_gaussian_tilt(proto_len, rng.uniform(0.3, 1.5), rng.uniform(0.2, 0.7))
        p = _norm(p)
        sc = _score(p)
        if np.isfinite(sc):
            try_update(p, sc)
            proto_results.append((sc, p))

    print(f"  Phase 1 done. Best proto score: {best_score:.6f}")

    # === Phase 2: Coordinate descent on final length (300s - 1800s) ===
    print("=" * 60)
    print("Phase 2: Coordinate descent on length", FINAL_LEN)
    print("=" * 60)

    phase2_deadline = deadline - 200.0
    if phase2_deadline < start_time + 100:
        phase2_deadline = deadline - 50.0

    # Generate diverse starting sequences for the full length
    full_starts = []

    # Start 1: Expand best prototype
    if proto_results:
        proto_results.sort(key=lambda x: x[0])
        for sc_p, p_p in proto_results[:5]:
            p_full = _expand_prototype(p_p, FINAL_LEN)
            sc_full = _score(p_full)
            if np.isfinite(sc_full):
                full_starts.append((sc_full, p_full))
                try_update(p_full, sc_full)

    # Start 2: From height_sequence_1 (if length matches)
    try:
        hseq = _clean(height_sequence_1)
        if hseq.size == FINAL_LEN:
            full_starts.append((_score(hseq), hseq))
        elif hseq.size > 0:
            p_exp = _expand_prototype(hseq, FINAL_LEN)
            sc = _score(p_exp)
            if np.isfinite(sc):
                full_starts.append((sc, p_exp))
    except Exception:
        pass

    # Start 3: Parametric shapes at full length
    full_parametric = [
        _make_linear_ramp(FINAL_LEN, 0.5),
        _make_linear_ramp(FINAL_LEN, 0.8),
        _make_linear_ramp(FINAL_LEN, 1.0),
        _make_two_level(FINAL_LEN, FINAL_LEN // 3, 2.0),
        _make_two_level(FINAL_LEN, FINAL_LEN // 2, 1.5),
        _make_two_level(FINAL_LEN, 2 * FINAL_LEN // 3, 0.7),
        _make_gaussian_tilt(FINAL_LEN, 0.6, 0.3),
        _make_gaussian_tilt(FINAL_LEN, 1.0, 0.4),
        _make_exponential(FINAL_LEN, 0.999),
    ]
    for p_fp in full_parametric:
        sc = _score(p_fp)
        if np.isfinite(sc):
            full_starts.append((sc, p_fp))
            try_update(p_fp, sc)

    # Sort by score
    full_starts.sort(key=lambda x: x[0])
    print(f"  Generated {len(full_starts)} full-length starts. Best: {full_starts[0][0]:.6f}")

    # Run coordinate descent from the best starts
    # Allocate time proportional to the number of starts
    n_starts_to_try = min(5, len(full_starts))
    time_per_start = max(20.0, (phase2_deadline - time.monotonic()) / n_starts_to_try)

    for start_idx in range(n_starts_to_try):
        if time.monotonic() > phase2_deadline:
            break
        sc_start, p_start = full_starts[start_idx]
        c_start = np.convolve(p_start, p_start)
        np.clip(c_start, 0.0, None, out=c_start)

        # Choose grid size based on time available
        if time_per_start > 60:
            grid_size, refine = 35, 10
        else:
            grid_size, refine = 25, 6

        start_deadline = min(phase2_deadline - 5.0, time.monotonic() + time_per_start)
        print(f"  Start {start_idx+1}/{n_starts_to_try}: score={sc_start:.6f}, time={time_per_start:.0f}s")

        try:
            p_best, c_best, sc_best = _coord_descent(
                p_start, c_start, sc_start, start_deadline, rng,
                grid_size=grid_size, refine=refine, max_perturb=6
            )
            if np.isfinite(sc_best):
                try_update(p_best, sc_best)
        except Exception as e:
            print(f"  Error in coord descent: {e}")

    print(f"  Phase 2 done. Best score: {best_score:.6f}")

    # === Phase 3: Block descent for coarse optimization (next 200s) ===
    print("=" * 60)
    print("Phase 3: Block descent")
    print("=" * 60)
    phase3_deadline = deadline - 100.0
    if phase3_deadline < time.monotonic() + 20:
        phase3_deadline = deadline - 20.0

    if best_seq is not None and time.monotonic() < phase3_deadline:
        p_block = best_seq.copy()
        best_block_score = _score(p_block)

        # Try different block sizes
        for n_blocks in [50, 100, 200, 380, 760]:
            if time.monotonic() > phase3_deadline:
                break
            block_deadline = min(phase3_deadline - 5.0, time.monotonic() + 40.0)
            try:
                p_bd, sc_bd = _block_descent(p_block, n_blocks, block_deadline, rng)
                if np.isfinite(sc_bd) and sc_bd < best_block_score:
                    best_block_score = sc_bd
                    p_block = p_bd.copy()
                    print(f"  Block descent ({n_blocks} blocks): {sc_bd:.6f}")
            except Exception as e:
                print(f"  Block descent error: {e}")

        if best_block_score < best_score:
            best_score = best_block_score
            best_seq = _norm(p_block).copy()

    # === Phase 4: Final polish (last 100s) ===
    print("=" * 60)
    print("Phase 4: Final polish")
    print("=" * 60)

    if best_seq is not None:
        p_polish = best_seq.copy()
        if p_polish.size != FINAL_LEN:
            p_polish = _expand_prototype(p_polish, FINAL_LEN)
        c_polish = np.convolve(p_polish, p_polish)
        sc_polish = _score(p_polish)

        polish_deadline = deadline - 5.0
        if time.monotonic() < polish_deadline - 10:
            try:
                p_final, c_final, sc_final = _coord_descent(
                    p_polish, c_polish, sc_polish, polish_deadline, rng,
                    grid_size=30, refine=8, max_perturb=3
                )
                if np.isfinite(sc_final) and sc_final < best_score:
                    best_score = sc_final
                    best_seq = _norm(p_final).copy()
                    print(f"  Final polish improved to: {sc_final:.6f}")
            except Exception as e:
                print(f"  Polish error: {e}")

    # Final evaluation with external evaluator
    if best_seq is None:
        best_seq = _make_two_level(FINAL_LEN, FINAL_LEN // 2, 1.5)

    try:
        ext_score = evaluate_sequence([float(x) for x in best_seq])
        if np.isfinite(ext_score):
            print(f"  External evaluation: {ext_score:.6f}")
            if ext_score < best_score:
                best_score = ext_score
    except Exception:
        pass

    # Ensure output is valid
    out = _norm(best_seq)
    out = np.clip(out, 0.0, 1000.0)
    final_score = _score(out)
    print(f"\nFinal result: score={final_score:.6f}, length={out.size}")
    print(f"Total time: {time.monotonic()-start_time:.1f}s")

    return [float(x) for x in out]
