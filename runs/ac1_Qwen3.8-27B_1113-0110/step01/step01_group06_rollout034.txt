"""

Uses full-vector exponentiated gradient with adaptive focus, multi-restart strategy,
and portfolio search across multiple sequence lengths (4096, 8192, 16384).
Key improvements: oscillating focus schedule, diverse restarts, aggressive
final refinement on the single best candidate.

import numpy as np
import time

_T0 = time.monotonic()


def _pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _fft_square(a):
    n = int(a.size)
    L = 2 * n - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, n=N)
    b = np.fft.irfft(fa * fa, n=N).real[:L]
    return np.maximum(b, 0.0)


def _fft_grad(weights, a):
    n = int(a.size)
    L = int(weights.size)
    N = _pow2(L + n - 1)
    fw = np.fft.rfft(weights, n=N)
    fa = np.fft.rfft(a, n=N)
    c = np.fft.irfft(fw * np.conj(fa), n=N)[:n]
    return 2.0 * c


def _score_from_b(a, b):
    s = float(a.sum())
    if s < 1e-300:
        return float('inf'), b, 0.0
    maxb = float(b.max())
    return float(2.0 * a.size * maxb / (s * s)), b, maxb


def _make_weights(b, focus, eps=0.01):
    L = int(b.size)
    maxb = float(b.max())
    if maxb < 1e-300:
        return np.full(L, 1.0 / L), maxb
    z = (maxb - b) / maxb
    with np.errstate(over='ignore', under='ignore'):
        p = np.exp(-z * float(focus))
    s = float(p.sum())
    if s < 1e-12:
        p = np.zeros(L)
        p[int(np.argmax(b))] = 1.0
    else:
        p = (1.0 - eps) * (p / s) + eps / L
    return p, maxb


def _normalize(a):
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-300:
        return np.full(a.size, 1.0 / a.size)
    return a / s


def _init_uniform(n):
    return np.full(n, 1.0 / n)


def _init_smooth_random(n, rng, freqs=(1, 2, 3, 5, 7, 11, 13)):
    x = np.arange(n, dtype=np.float64)
    v = np.zeros(n)
    for f in freqs:
        v += rng.normal(0, 1.0 / f) * np.sin(2 * np.pi * f * x / n + rng.uniform(0, 2 * np.pi))
    v = np.maximum(v, -3 * np.abs(v).max() if v.max() != 0 else -3)
    a = np.exp(v)
    return _normalize(a)


def _init_log_random(n, rng, amp=2.5):
    v = rng.normal(0, amp, n)
    a = np.exp(v - v.max())
    a = np.maximum(a, 1e-15)
    return _normalize(a)


def _init_bimodal(n, rng, gap_frac=0.3):
    # Two blobs of mass
    c1 = int(n * (1 - gap_frac) / 2)
    c2 = int(n * gap_frac / 2)
    a = np.zeros(n)
    a[:c1] = np.exp(rng.normal(0, 0.5, c1))
    a[n - c2:] = np.exp(rng.normal(0, 0.5, c2))
    return _normalize(a)

            focus_growth=1.01, eta0=0.1, eta_min=0.005, eta_max=1.0,
            label="polish"
        )
        _update_best(a, sc, "polish")

    # Final output
    if best_a is None:
        best_a = _init_uniform(8192)
        best_score = 2.0
        print("Warning: no valid candidate found, using uniform")

    best_a = _normalize(best_a)
    final_score, _, _ = _score_from_b(best_a, _fft_square(best_a))
    print(f"\nFinal: n={best_a.size}, internal_score={final_score:.6f}, "
          f"total_time={time.monotonic()-t_start:.0f}s")

    final_seq = [float(x) for x in best_a]

    # Verify with evaluate_sequence if possible
    try:
        ev = globals().get('evaluate_sequence')
        if ev is not None and callable(ev):
            exact = ev(final_seq)
            print(f"Exact evaluation: {exact}")
            if not np.isfinite(exact) or exact > final_score * 1.01:
                print("Warning: exact score differs from internal!")
    except Exception as e:
        print(f"Could not evaluate: {e}")

    return final_seq
