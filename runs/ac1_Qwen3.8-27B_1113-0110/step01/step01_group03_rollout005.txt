def fft_self_conv(a):

    n = len(a)

def compute_grad_soft(a, c, tau):

    Compute gradient of soft-max surrogate w.r.t. a.

        # Anneal tau
        frac = (time.monotonic() - t_start) / allocated_time
        tau = max(tau_end, tau_start * (1.0 - frac) ** 2.5)
        
        # Compute gradient
        if tau > tau_end * 5:
            g_c = compute_grad_soft(a, c, tau)
            grad = (2.0 / n) * g_c
        else:
            g_c = compute_grad_hard(a, c, topk=3)
            grad = (2.0 / n) * g_c
        
        # Nesterov momentum update
        velocity = beta * velocity - lr * grad
        a_candidate = a + velocity
        a_candidate = np.maximum(a_candidate, 0.0)
        S_cand = a_candidate.sum()
        if S_cand < 0.01:
            a_candidate = a.copy()
            velocity *= 0.3
        else:
            a_candidate = a_candidate / S_cand * n
        a = a_candidate
        
        # Simulated annealing: random jump
        if it % 200 == 0 and rng.random() < 0.15:
            # Jump magnitude decreases over time
            jump_mag = 0.1 * (1.0 - frac)
            a = a * np.exp(rng.normal(0, jump_mag, n))
            a = normalize(a)
            velocity *= 0.3
        
        # Periodic exact evaluation and adaptive lr
        if it % eval_every == 0:
            exact_sc = float(evaluate_sequence(a.tolist()))
            if np.isfinite(exact_sc) and exact_sc < local_best_score:
                local_best_score = exact_sc
                local_best = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            if no_improve >= patience:
                lr *= 0.7
                velocity *= 0.4
                no_improve = 0
                if lr < 1e-7:
                    # Bigger restart
                    a = local_best * np.exp(rng.normal(0, 0.06, n))
                    a = normalize(a)
                    velocity = np.zeros(n)
                    lr = 0.04
                    no_improve = 0
                if it % (eval_every * 50) == 0:
                    print(f"    it={it}: best={local_best_score:.8f}, lr={lr:.6f}, tau={tau:.2f}")
    
    # Final fine-tune with hard max, tiny steps
    a = local_best.copy()
    lr_ft = 0.001
    velocity_ft = np.zeros(n)
    t_ft = time.monotonic()
    ft_count = 0
    while time.monotonic() - t_ft < min(20, allocated_time * 0.05):
        a = normalize(a)
        c = fft_self_conv(a)
        g_c = compute_grad_hard(a, c, topk=2)
        grad = (2.0 / n) * g_c
        velocity_ft = 0.7 * velocity_ft - lr_ft * grad
        a_new = np.maximum(a + velocity_ft, 0.0)
        a_new = a_new / a_new.sum() * n
        a = a_new
        ft_count += 1
        if ft_count % 5 == 0:
            sc = float(evaluate_sequence(a.tolist()))
            if np.isfinite(sc) and sc < local_best_score:
                local_best = a.copy()
                local_best_score = sc
    
    return local_best, local_best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for low-score non-negative sequence minimizing 2n*max(a*a)/sum(a)^2.
    Uses FFT-based soft-max gradient descent with Nesterov acceleration,
    temperature annealing, random jumps, and multi-start over different n.
    """
    rng = np.random.default_rng(seed)
    wall_start = time.monotonic()
    deadline = wall_start + budget_s - 10.0
    
    # Get parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        parent = np.ones(512, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    if parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(512, dtype=np.float64)
    
    n_parent = len(parent)
    best_a = parent.copy()
    best_score = float(evaluate_sequence(parent.tolist()))
    print(f"Parent: n={n_parent}, score={best_score:.8f}")
    
    # Strategy: focus 55% on parent n, 25% on a different n, 20% on exploring
    # Phase 1: Optimize parent size
    t1 = (deadline - time.monotonic()) * 0.55
    print(f"Phase 1: n={n_parent}, time={t1:.0f}s")
    a1, s1 = optimize_sequence(best_a, n_parent, t1, rng=rng)
    if s1 < best_score:
        best_score = s1
        best_a = a1
        print(f"  Phase 1 result: score={best_score:.8f}")
    
    # Phase 2: Try a different n (power of 2 for FFT efficiency)
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Choose an alternative n that differs from parent
        alt_ns = [2048, 4096, 3072, 3584, 5120]
        alt_n = None
        for c_n in alt_ns:
            if c_n != n_parent and 512 <= c_n <= 6000:
                alt_n = c_n
                break
        if alt_n is None:
            alt_n = 2048 if n_parent != 2048 else 4096
        
        t2 = min(remaining * 0.3, 300)
        print(f"Phase 2: n={alt_n}, time={t2:.0f}s")
        
        # Initialize: spread the best sequence
        a_init2 = np.zeros(alt_n)
        if n_parent <= alt_n:
            a_init2[:n_parent] = best_a
        else:
            a_init2 = best_a[:alt_n]
        # Also try a random init
        if rng.random() < 0.5:
            a_init2 = np.maximum(a_init2, 0)
        a_init2 = a_init2 + rng.exponential(1.0, alt_n)
        a_init2 = normalize(a_init2)
        
        a2, s2 = optimize_sequence(a_init2, alt_n, t2, rng=rng)
        if s2 < best_score:
            best_score = s2
            best_a = a2
            print(f"  Phase 2 result: score={best_score:.8f}")
    
    # Phase 3: Explore with random restarts on the best n
    remaining = deadline - time.monotonic()
    if remaining > 20:
        n_best = len(best_a)
        # Do a short exploration run with random init
        t3 = min(remaining * 0.4, 150)
        print(f"Phase 3: exploration n={n_best}, time={t3:.0f}s")
        a_rand = rng.exponential(1.0, n_best)
        # Add some structure: try a sine-modulated initial
        x = np.linspace(0, 1, n_best)
        a_rand = a_rand + np.maximum(np.cos(2 * np.pi * x * 3), 0) * 0.5
        a_rand = normalize(a_rand)
        
        a3, s3 = optimize_sequence(a_rand, n_best, t3, rng=rng)
        if s3 < best_score:
            best_score = s3
            best_a = a3
            print(f"  Phase 3 result: score={best_score:.8f}")
    
    # Phase 4: Final intensive polishing on the best sequence
    remaining = deadline - time.monotonic()
    if remaining > 10:
        n_best = len(best_a)
        t4 = remaining - 5
        print(f"Phase 4: polishing n={n_best}, time={t4:.0f}s")
        # Use very low tau start (almost hard max) for precision
        a4, s4 = optimize_sequence(best_a, n_best, t4, tau_start=10.0, tau_end=0.5, rng=rng)
        if s4 < best_score:
            best_score = s4
            best_a = a4
    
    # Final validation
    final_a = np.maximum(best_a, 0.0)
    final_a = final_a / final_a.sum() * len(final_a)
    final_score = float(evaluate_sequence([float(x) for x in final_a]))
    if np.isfinite(final_score):
        best_score = min(best_score, final_score)
        best_a = final_a
    
    elapsed = time.monotonic() - wall_start
    print(f"Total time: {elapsed:.1f}s")
    print(f"Final score: {best_score:.8f}, n={len(best_a)}")
    return [float(x) for x in best_a]
