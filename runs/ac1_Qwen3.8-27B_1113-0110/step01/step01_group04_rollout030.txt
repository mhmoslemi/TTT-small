def _pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()

def _clean(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    return np.maximum(v, 0.0)

def _project_simplex(v):
    v = _clean(v)
    n = v.size
    if n == 0:
        return v
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = v_sorted > (css - 1.0) / idx
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    r = int(np.nonzero(cond)[0][-1])
    tau = (css[r] - 1.0) / (r + 1)
    out = np.maximum(v - tau, 0.0)
    s = out.sum()
    if s <= 0:
        return np.full(n, 1.0 / n)
    return out / s

def _score(a, L=None):
    v = _clean(a)
    n = v.size
    if n == 0:
        return np.inf
    s = v.sum()
    if s < 1e-15:
        return np.inf
    v = v / s
    if L is None:
        L = _pow2(2 * n - 1)
    A = np.fft.rfft(v, n=L)
    c = np.fft.irfft(A * A, n=L)[:2*n-1]
    mx = c.max()
    return 2.0 * n * mx

def _spectral_flatten(a0, n_iters=2000, L=None, alpha_start=0.15, alpha_end=0.01, seed=0):
    """
    Iteratively flatten the power spectrum.
    At each step:
    1. Compute DFT of a.
    2. Compute target (flattened) power spectrum.
    3. Interpolate between current and target.
    4. Inverse DFT, clip to non-negative, normalize.
    """
    a = _project_simplex(a0)
    n = a.size
    if L is None:
        L = _pow2(2 * n - 1)
    
    rng = np.random.default_rng(seed)
    best = a.copy()
    best_score = _score(a, L)
    
    n_half = L // 2  # Number of rfft bins
    
    for it in range(n_iters):
        frac = it / max(1, n_iters - 1)
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        
        A = np.fft.rfft(a, n=L)
        power = np.abs(A) ** 2  # Length n_half + 1
        
        # Target: distribute power more evenly
        # Keep DC fixed, flatten the rest
        dc_power = power[0]
        non_dc_power = power[1:]
        target_non_dc = (non_dc_power.sum() / non_dc_power.size)
        
        # Interpolate
        new_power = power.copy()
        new_power[1:] = (1 - alpha) * power[1:] + alpha * target_non_dc
        
        # Ensure non-negative
        new_power = np.maximum(new_power, 0)
        
        # Reconstruct
        new_mag = np.sqrt(new_power)
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)
        
        a_new = np.fft.irfft(A_new, n=L)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        sc = _score(a_new, L)
        if sc < best_score:
            best = a_new.copy()
            best_score = sc
        a = a_new
    
    return best, best_score

def _lse_objective(a, eps, L):
    """Compute the log-sum-exp of the convolution coefficients."""
    v = _clean(a)
    n = v.size
    s = v.sum()
    v = v / s
    A = np.fft.rfft(v, n=L)
    c = np.fft.irfft(A * A, n=L)[:2*n-1]
    c = np.maximum(c, 0)
    cmax = c.max()
    if cmax < 1e-300:
        return 0.0
    z = (c - cmax) / eps
    z = np.clip(z, -700, 0)
    e = np.exp(z)
    return cmax + eps * np.log(e.sum())

def _lse_grad(x, eps, L, n):
    """
    Gradient of the log-sum-exp objective w.r.t. the log-ratio parameters.
    x has length n-1, representing log(a[i]/a[n-1]) for i = 0..n-2.
    """
    # Reconstruct a from log-ratios
    x = np.clip(x, -100, 100)
    m = max(0, x.max()) if x.size else 0
    e = np.exp(np.clip(x - m, -745, 700))
    e_last = np.exp(-m)
    s = e.sum() + e_last
    a = np.empty(n)
    a[:-1] = e / s
    a[-1] = e_last / s
    
    # Compute gradient
    A = np.fft.rfft(a, n=L)
    c = np.fft.irfft(A * A, n=L)[:2*n-1]
    c = np.maximum(c, 0)
    cmax = c.max()
    if cmax < 1e-300:
        return 0.0, np.zeros(n-1)
    
    z = (c - cmax) / eps
    z = np.clip(z, -700, 0)
    e_c = np.exp(z)
    w = e_c / e_c.sum()
    
    # Gradient of c w.r.t. a: (2 * w * a) in convolution sense
    W = np.fft.rfft(w, n=L)
    gc = 2.0 * np.fft.irfft(W * np.conj(A), n=L)[:n]
    
    # Project gradient to simplex tangent space
    dot = np.dot(a, gc)
    g = a * (gc - dot)
    gx = g[:-1]  # Remove last component (reparameterized)
    
    f = cmax + eps * np.log(e_c.sum())
    gx = np.where(np.isfinite(gx), gx, 0.0)
    return f, gx

def _lse_grad_direct(a, eps, L):
    """
    Compute f(a) and its gradient, where a is on the simplex.
    f(a) = logsumexp(c/eps) * eps, c = a*a.
    """
    n = a.size
    A = np.fft.rfft(a, n=L)
    c = np.fft.irfft(A * A, n=L)[:2*n-1]
    c = np.maximum(c, 0)
    
    cmax = c.max()
    if cmax < 1e-300:
        return 0.0, np.zeros(n)
    
    z = (c - cmax) / eps
    z = np.clip(z, -700, 0)
    e = np.exp(z)
    sume = e.sum()
    w = e / sume
    
    f = cmax + eps * np.log(sume)
    
    # Gradient: g[j] = 2 * sum_k w[k] * a[k-j] (with range restrictions)
    # This is 2 * (w * a)[j] in the "cross-correlation" sense
    # Actually, it's the derivative of c[k] w.r.t. a[j], weighted by w[k]:
    # g[j] = 2 * sum_k w[k] * a[k-j] (where k-j is in [0,n-1] and j is in [0,n-1])
    # This is 2 * (w cross-correlated with a)[j] = 2 * (flip(w) * a)[j] in convolution sense
    # Or: g = 2 * ifft(conj(fft(w)) * fft(a))[:n]  ... let me think more carefully.
    
    # c[k] = sum_i a[i] * a[k-i]
    # d(c[k])/d(a[j]) = 2 * a[k-j] (if 0 <= k-j <= n-1)
    # g[j] = sum_k w[k] * d(c[k])/d(a[j]) = 2 * sum_k w[k] * a[k-j] (with range)
    # Let m = k-j, so k = m+j:
    # g[j] = 2 * sum_m w[m+j] * a[m] (where m+j is in [0, 2n-2] and m in [0, n-1])
    # This is 2 * (w_shifted * a)[j] where w_shifted[m] = w[m+j]... 
    # Actually this is the cross-correlation of w and a:
    # g[j] = 2 * sum_m a[m] * w[m+j] = 2 * (a ⋆ w)[j] (cross-correlation)
    # In terms of convolution: (a ⋆ w)[j] = sum_m a[m] * w[m+j] = (flip(a) * w)[j] + n - 1 ... 
    # Hmm, let me just use the FFT formula.
    
    # Cross-correlation: (a ⋆ w)[j] = sum_m a[m] * conj(w[m-j]) = sum_m a[m] * w[m-j] (for real w)
    # Actually for real sequences: (a ⋆ w)[j] = sum_m a[m] * w[m - j]
    # In DFT: DFT(a ⋆ w) = conj(A) * W (or A * conj(W), depending on convention)
    
    # Let me just use: g = 2 * irfft(conj(rfft(a)) * rfft(w_padded))[:n]
    # where w_padded is w zero-padded to length L.
    
    W = np.fft.rfft(w, n=L)
    g = 2.0 * np.fft.irfft(np.conj(A) * W, n=L)[:n]
    
    # Project gradient to simplex tangent space (remove component in direction of all-ones)
    g -= g.mean()
    
    return f, g

def _lse_and_grad(a, eps, L):
    """
    Compute LSE(a*a/eps) and its gradient.
    a is assumed to be non-negative (not necessarily normalized).
    """
    n = a.size
    # Normalize
    s = a.sum()
    if s < 1e-15:
        return np.inf, np.zeros(n)
    a_norm = a / s
    
    A = np.fft.rfft(a_norm, n=L)
    c = np.fft.irfft(A * A, n=L)[:2*n-1]
    c = np.maximum(c, 0)
    
    cmax = c.max()
    if cmax < 1e-300:
        return 0.0, np.zeros(n)
    
    z = (c - cmax) / eps
    z = np.minimum(z, 0)  # clip from above
    z = np.maximum(z, -700)  # clip from below
    e = np.exp(z)
    sume = e.sum()
    w = e / sume
    
    f = cmax + eps * np.log(sume)

    # f = LSE(c/eps) * eps + cmax = LSE of c with temperature eps
    # df/da = (2/eps) * (a ⋆ w)  [cross-correlation]
    # But we want gradient w.r.t. a_norm (since we'll project onto simplex)

    W = np.fft.rfft(w, n=L)

    # This is the gradient of f w.r.t. a_norm.
    # For the simplex constraint, we project this gradient.
    
    return f, g

def _pow2(m):

def _clean(v):

def _project_simplex(v):
    v = _clean(v)
    n = v.size
    if n == 0:
        return v
    v_s = np.sort(v)[::-1].copy()
    css = np.cumsum(v_s)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = v_s > (css - 1.0) / idx
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    r = int(np.nonzero(cond)[0][-1])
    tau = (css[r] - 1.0) / (r + 1)
    out = np.maximum(v - tau, 0.0)
    s = out.sum()
    if s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s

    return a
