def evaluate_sequence(seq: List[float]) -> float:
    n = len(seq)
    # Check input type and validity
    if not isinstance(seq, list):
        return np.inf
    # Reject empty lists
    if len(seq) == 0:
        return np.inf

    for x in seq:
        # Reject boolean types (as they are a subclass of int)
        if isinstance(x, bool):
            return np.inf
        # Reject any other non-integer/non-float types (like strings, complex, etc.)
        if not isinstance(x, (int, float)) or isinstance(x, bool):
            return np.inf

        if np.isnan(x) or np.isinf(x):
            return np.inf
