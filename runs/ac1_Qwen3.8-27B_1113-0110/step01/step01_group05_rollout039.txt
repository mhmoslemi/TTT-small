def _smooth_max_grad(conv, tau):
    """Compute softmax weights and gradient of smooth max of conv."""
    # conv has length 2n-1
    max_conv = np.max(conv)
    exp_vals = np.exp((conv - max_conv) / tau)
    denom = np.sum(exp_vals)
    softmax = exp_vals / denom
    return softmax

def _convolve_4fft(a, n):
    """Compute full convolution using FFT."""
    fft_size = 1
    target = 2 * n - 1

    return result

"""
Minimizer for 2*n*max(convolve(a,a)) / (sum(a))^2 with a >= 0.

Strategy:
  * Annealed smooth-max gradient descent using FFT-accelerated cross-correlation
    for the gradient computation.
  * Multiple diverse initializations: existing best, Gaussian-shaped, Fourier-flat,
    paired structures, and random.
  * Adam optimizer with adaptive learning rate.
  * Periodic coordinate descent polish for local refinement.
  * Random restarts from perturbed best to escape local minima.
  * All convolutions and gradients computed in O(n log n) via FFT.

import numpy as np
import time


def _clean_array(arr):
    a = np.asarray(arr, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    np.clip(a, 0.0, 1000.0, out=a)
    return a


def _normalize(a):
    a = _clean_array(a)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(max(1, a.size), dtype=np.float64)
    return a / s


def _next_pow2(m):
    m = int(m)
    v = 1
    while v < m:
        v *= 2
    return v


def _compute_conv(a, fft_size=None):
    n = int(a.size)
    if n <= 0:
        return np.array([0.0])
    if fft_size is None:
        fft_size = _next_pow2(2 * n - 1)
    F = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(F * F, fft_size)[:2 * n - 1]
    np.maximum(conv, 0.0, out=conv)
    return conv


def _score_array(a):
    a = _normalize(a)
    c = _compute_conv(a)
    n = int(a.size)
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return float('inf')
    return float(2.0 * n * mx)


def _gradient_of_smooth_max(a, h, fft_size):
    """Compute 2 * cross_corr(h, a) where cross_corr(h,a)[j] = sum_k h[k]*a[k-j].
    a: length n, h: length 2n-1, fft_size >= 3n-2.
    Returns array of length n.
    """
    n = int(a.size)
    h_pad = np.zeros(fft_size)
    a_pad = np.zeros(fft_size)
    h_pad[:len(h)] = h
    a_pad[:n] = a
    H = np.fft.rfft(h_pad)
    A = np.fft.rfft(a_pad)
    G = H * np.conj(A)
    g = np.fft.irfft(G, fft_size)
    return 2.0 * g[:n]

            idx2 = int(np.argmin(vals2))

        return p, c, score, False

        return p, c, score, False

    p_new = p_new / snew

    score_new = 2.0 * n * float(np.max(c_new))

                improved = True

        s = float(np.sum(p))
        if not np.isfinite(s) or s <= 0:
            break
        p = p / s
        c = _compute_conv(p)
        score = 2.0 * n * float(np.max(c))

        tol = 1e-12 * max(1.0, best_score)
        if score < best_score - tol:
            best_score = score
            best_p = p.copy()
            best_c = c.copy()
            stagnant = 0
        else:
            stagnant += 1

        if improved:
            stagnant = 0
        else:
            if stagnant >= 2 and perts < max_perturb:
                scale = noise_scales[min(perts, len(noise_scales) - 1)]
                p = best_p * np.exp(rng.normal(0.0, scale, n))
                np.maximum(p, 0.0, out=p)
                s = float(np.sum(p))
                if np.isfinite(s) and s > 0:
                    p = p / s
                    c = _compute_conv(p)
                    score = 2.0 * n * float(np.max(c))
                    if score < best_score:
                        best_score = score
                        best_p = p.copy()
                        best_c = c.copy()
                perts += 1
                stagnant = 0
            elif stagnant >= 6:
                break

    return best_p, best_c, best_score


def _make_gaussian_init(n, rng, beta_range=(0.5, 5.0)):
    """Gaussian-shaped initialization."""
    n = int(n)
    center = (n - 1) / 2.0
    beta = rng.uniform(*beta_range)
    x = np.arange(n) - center
    p = np.exp(-beta * x * x / (n * n))
    p += 0.01 * rng.random(n)
    return _normalize(p)


def _make_fourier_init(n, rng, taper=0.5):
    """Initialize from a near-flat spectrum."""
    n = int(n)
    # Create a spectrum that's flat with slight taper
    k = np.fft.rfftfreq(n) * n  # frequency indices
    k_abs = np.abs(k)
    k_norm = k_abs / (n / 2.0)
    target_power = np.exp(-taper * k_norm ** 2)
    target_power[0] = 1.0  # DC fixed
    # Take square root as amplitude
    A = np.sqrt(target_power)
    # Random phase (conjugate symmetric for real signal)
    phase = np.zeros_like(A)
    phase[1:] = rng.uniform(0, 2 * np.pi, A.size - 1)
    A = A * np.exp(1j * phase)
    # IFFT
    a = np.fft.irfft(A, n)
    np.maximum(a, 0.0, out=a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _make_paired_init(n, rng, alpha=0.5):
    """Paired anti-correlated initialization."""
    n = int(n)
    m = (n + 1) // 2
    weights = rng.dirichlet(np.full(m, 1.0))
    p = np.zeros(n)
    for r in range(m):
        a = r
        b = n - 1 - r
        if a == b:
            p[a] = weights[r]
        else:
            if rng.random() < 0.6:
                if rng.random() < 0.5:
                    p[a] = weights[r]
                else:
                    p[b] = weights[r]
            else:
                q = rng.dirichlet(np.full(2, alpha))
                p[a] = weights[r] * q[0]
                p[b] = weights[r] * q[1]
    return _normalize(p)


def _expand_to_length(proto, target_len):
    """Expand a prototype to target length by block replication."""
    target_len = int(target_len)
    p = _normalize(proto)
    R = int(p.size)
    if R <= 0 or target_len <= 0:
        return np.ones(max(1, target_len))
    if R == target_len:
        return p.copy()
    if R > target_len:
        out = np.ones(target_len)
        for j in range(target_len):
            lo = (j * R) // target_len
            hi = max(lo + 1, ((j + 1) * R) // target_len)
            out[j] = float(np.mean(p[lo:hi]))
        return _normalize(out)
    base = target_len // R
    rem = target_len % R
    lengths = np.full(R, base, dtype=int)
    if rem > 0:
        lengths[:rem] += 1
    out = np.zeros(target_len)
    idx = 0
    for i in range(R):
        L = int(lengths[i])
        if L > 0:
            out[idx:idx + L] = p[i] / float(L)
            idx += L
    return _normalize(out)


def _lp_polish(p, c, score, deadline, max_iters=3):
    """LP-based direction finding for escaping local minima."""
    try:
        from scipy.optimize import linprog
        from scipy import sparse
    except Exception:
        return p, c, score

    n = int(p.size)
    if n < 2:
        return p, c, score

    p = _normalize(p)
    c = _compute_conv(p)
    score = 2.0 * n * float(np.max(c))

    best_p = p.copy()
    best_score = score

    for _ in range(int(max_iters)):
        if time.monotonic() > deadline - 10.0:
            break
        M = float(np.max(c))
        if M <= 0:
            break

        # Build constraint matrix: (K a)[k] = sum_i a[i] * a[k-i] <= M
        # This is quadratic, but LP approximates by fixing one factor:
        # (K a)[k] = sum_i a[i] * a[k-i]  ->  sum_i (a[k-i]) * a[i] <= M
        # Treat a[k-i] as coefficients (fixed), optimize a[i]
        i_idx = np.repeat(np.arange(n), n)
        j_idx = np.tile(np.arange(n), n)
        k_idx = i_idx + j_idx
        vals = np.tile(p, n)
        K = sparse.csr_matrix((vals, (k_idx, i_idx)), shape=(2 * n - 1, n))
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        obj = -np.ones(n)

        try:
            res = linprog(c=obj, A_ub=K, b_ub=b_ub,
                          bounds=[(0, None)] * n, method='highs',
                          options={"presolve": True})
        except Exception:
            break

        if res is None or not getattr(res, "success", False):
            break
        g = np.asarray(res.x, dtype=np.float64)
        np.maximum(g, 0.0, out=g)
        gs = float(np.sum(g))
        if gs <= 1:
            break
        g = g / gs

        cg = _compute_conv(g)
        cross = _compute_conv(p, None)  # just p*p, we need cross-correlation
        # Actually we need convolve(p, g)
        fft_sz = _next_pow2(2 * n - 1)
        P = np.fft.rfft(p, fft_sz)
        G = np.fft.rfft(g, fft_sz)
        cross = np.fft.irfft(P * G, fft_sz)[:2 * n - 1]
        np.maximum(cross, 0.0, out=cross)

        # Line search on t in [0,1]
        best_t = 0.0
        best_val = float(score)
        for t in np.linspace(0, 1, 33):
            at = (1 - t) * p + t * g
            at = _normalize(at)
            val = _score_array(at)
            if val < best_val:
                best_val = val
                best_t = t

        if best_val >= score - 1e-10:
            break

        p = (1 - best_t) * p + best_t * g
        p = _normalize(p)
        c = _compute_conv(p)
        score = 2.0 * n * float(np.max(c))
        if score < best_score:
            best_score = score
            best_p = p.copy()

    return best_p, c, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Return the best nonnegative coefficient sequence found within budget_s."""
    budget = max(1.0, float(budget_s))
    deadline = time.monotonic() + max(0.0, budget - 10.0)
    rng = np.random.default_rng(int(seed))
    final_len = 3809

    best_seq = None
    best_score = float('inf')

    def _update_best(seq, sc):
        nonlocal best_seq, best_score
        if seq is None or not np.isfinite(sc):
            return
        if sc < best_score:
            best_score = float(sc)
            best_seq = _normalize(seq).copy()

    # Load existing best
    try:
        h1 = _clean_array(height_sequence_1)
        if h1.size > 0:
            sc1 = _score_array(h1)
            if np.isfinite(sc1):
                _update_best(h1, sc1)
                print(f"Loaded height_sequence_1: score={sc1:.6f}, len={h1.size}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(final_len) / final_len
        best_score = _score_array(best_seq)
        print(f"Default uniform: score={best_score:.6f}")

    n = final_len
    fft_conv = _next_pow2(2 * n - 1)
    fft_xcorr = _next_pow2(3 * n - 2)

    # Phase 1: Smooth-max gradient descent from existing best
    print(f"\nPhase 1: Smooth-max GD from existing best (n={n})")
    t0 = time.monotonic()
    phase1_deadline = deadline - max(600.0, 0.4 * budget)
    if phase1_deadline > time.monotonic() + 60:
        a1, sc1 = _smooth_max_optimize(best_seq, phase1_deadline, rng, max_steps=100000)
        _update_best(a1, sc1)
        print(f"  Phase 1 done: score={sc1:.6f}, time={time.monotonic()-t0:.1f}s")

    # Phase 2: Diverse initializations + smooth-max GD
    print(f"\nPhase 2: Diverse initializations")
    init_deadline = deadline - max(400.0, 0.3 * budget)
    inits = []

    # Gaussian inits with different betas
    for beta in [0.3, 0.7, 1.5, 3.0, 6.0]:
        if time.monotonic() > init_deadline:
            break
        p = _make_gaussian_init(n, rng, beta_range=(beta, beta * 1.1))
        sc = _score_array(p)
        inits.append((sc, p))
        _update_best(p, sc)

    # Fourier inits
    for taper in [0.3, 0.8, 1.5, 3.0]:
        if time.monotonic() > init_deadline:
            break
        p = _make_fourier_init(n, rng, taper=taper)
        sc = _score_array(p)
        inits.append((sc, p))
        _update_best(p, sc)

    # Paired inits
    for alpha in [0.3, 0.7, 1.0]:
        if time.monotonic() > init_deadline:
            break
        p = _make_paired_init(n, rng, alpha=alpha)
        sc = _score_array(p)
        inits.append((sc, p))
        _update_best(p, sc)

    # Random inits
    for _ in range(3):
        if time.monotonic() > init_deadline:
            break
        p = np.exp(rng.normal(0, 0.5, n))
        p = _normalize(p)
        sc = _score_array(p)
        inits.append((sc, p))
        _update_best(p, sc)

    inits.sort(key=lambda x: x[0])
    print(f"  Generated {len(inits)} initializations, best init score={inits[0][0]:.6f}")

    # Run smooth-max on top diverse inits
    sm_deadline = min(deadline - 300.0, init_deadline + 400.0)
    for i, (sc_init, p_init) in enumerate(inits[:6]):
        if time.monotonic() > sm_deadline:
            break
        if sc_init > best_score * 1.5:  # skip terrible inits
            continue
        t0 = time.monotonic()
        a_opt, sc_opt = _smooth_max_optimize(p_init, sm_deadline, rng, max_steps=50000)
        _update_best(a_opt, sc_opt)
        print(f"  Init {i} (init_score={sc_init:.4f}) -> {sc_opt:.6f}, time={time.monotonic()-t0:.1f}s")

    # Phase 3: Coordinate descent polish on best
    print(f"\nPhase 3: Coordinate descent polish")
    cd_deadline = deadline - 30.0
    if time.monotonic() < cd_deadline - 60:
        grid_count = 20
        refine_count = 5
        grid = np.concatenate(([0.0], np.logspace(-10.0, 2.0, grid_count)))
        p_cd = best_seq.copy()
        c_cd = _compute_conv(p_cd, fft_conv)
        sc_cd = 2.0 * n * float(np.max(c_cd))
        t0 = time.monotonic()
        p_cd, c_cd, sc_cd = _coordinate_optimize(p_cd, c_cd, sc_cd, cd_deadline, rng,
                                                  grid, refine_count, max_perturb=5)
        _update_best(p_cd, sc_cd)
        print(f"  CD done: score={sc_cd:.6f}, time={time.monotonic()-t0:.1f}s")

    # Phase 4: LP polish
    if time.monotonic() < deadline - 30:
        print(f"\nPhase 4: LP polish")
        t0 = time.monotonic()
        p_lp, c_lp, sc_lp = _lp_polish(best_seq, None, best_score, deadline, max_iters=3)
        _update_best(p_lp, sc_lp)
        print(f"  LP done: score={sc_lp:.6f}, time={time.monotonic()-t0:.1f}s")

    # Final: verify with evaluate_sequence
    print(f"\nFinal best score (internal): {best_score:.6f}")
    try:
        ext_score = float(evaluate_sequence([float(x) for x in best_seq]))
        print(f"Final best score (external): {ext_score:.6f}")
    except Exception as e:
        print(f"External eval failed: {e}")

    out = _normalize(best_seq)
    return [float(x) for x in out]
