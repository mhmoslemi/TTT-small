import numpy as np
import time
from scipy.optimize import minimize as _minimize

def _pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()

def _clean(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    return np.maximum(v, 0.0)

def _normalize(v):
    v = _clean(v)
    s = v.sum()
    if s < 1e-15:
        return np.full(v.size, 1.0/v.size)
    return v / s

def _project_simplex(v):
    v = _clean(v)
    n = v.size
    if n == 0:
        return v
    x = np.sort(v)[::-1]
    css = np.cumsum(x)
    idx = np.arange(1, n+1, dtype=float)
    cond = x > (css - 1.0) / idx
    if not np.any(cond):
        return np.full(n, 1.0/n)
    r = int(np.nonzero(cond)[0][-1])
    tau = (css[r] - 1.0) / (r + 1)
    out = np.maximum(v - tau, 0.0)
    s = out.sum()
    if s < 1e-15:
        return np.full(n, 1.0/n)
    return out / s

def _score(a):
    v = _normalize(a)
    n = v.size
    L = _pow2(2*n-1)
    A = np.fft.rfft(v, n=L)
    c = np.fft.irfft(A*A, n=L)[:2*n-1]
    c = np.maximum(c, 0.0)
    mx = c.max()
    return 2.0*n*mx

def _conv_fft(a, L=None):
    n = a.size
    if L is None:
        L = _pow2(2*n-1)
    A = np.fft.rfft(a, n=L)
    c = np.fft.irfft(A*A, n=L)[:2*n-1]
    return np.maximum(c, 0.0)

            eta = 0.5 / (4.0 * cmax + 1e-10)
        else:
            eta = min(eta, 0.5 / (4.0 * cmax + 1e-10))

        # Nesterov / FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new

        # Momentum update
        y = a + theta * (a - a_prev)
        y = _project_simplex(y)

        # Gradient step from y
        f_y, grad_y, _, _ = _soft_obj_grad(y, eps, L)
        grad_y_t = grad_y - float(np.sum(grad_y) / n)

        a_new = _project_simplex(y - eta * grad_y_t)

        if not np.all(np.isfinite(a_new)):
            a_new = a.copy()

        a_prev = a
        a = a_new
        t = t_new
        step += 1

        # Periodic restart of momentum if stuck
        if no_improve > 2000 and (step % 100 == 0):
            # Slight random perturbation
            perturbation = rng.normal(0.0, 0.002, size=n)
            a = _project_simplex(a + perturbation)
            a_prev = a.copy()
            t = 1.0
            no_improve = 0
            last_improve_time = now

        # If very stuck, do a bigger restart
        if now - last_improve_time > 30.0 and frac > 0.3:
            perturbation = rng.normal(0.0, 0.005 * (1.0 - frac), size=n)
            a = _project_simplex(best_a + perturbation)
            a_prev = a.copy()
            t = 1.0
            no_improve = 0
            last_improve_time = now
            eta = 0.5 / (4.0 * best_score / (2.0 * n) + 1e-10)

    return best_a, best_score


def _subgradient_polish(a_init, deadline, L, seed):
    """
    Polish using exact / averaged subgradients of max(c).
    FISTA-style with the non-smooth objective.
    """
    a = _normalize(a_init.copy())
    n = a.size
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    total = max(0.1, deadline - start)

    t = 1.0
    a_prev = a.copy()
    eta = 0.1
    no_improve = 0
    last_improve = start

    # Average subgradient buffer
    avg_grad = np.zeros(n, dtype=np.float64)
    buffer_count = 0
    buffer_size = 100

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / total)

        c = _convolve_fft(a, L)
        cmax = float(c.max())
        exact_score = 2.0 * n * cmax

        now = time.monotonic()
        if exact_score < best_score - 1e-14:
            best_score = exact_score
            best_a = a.copy()
            last_improve = now
            no_improve = 0
        else:
            no_improve += 1

        # Computed averaged subgradient
        grad, c, cmax = _averaged_subgrad_steps(a, L)
        avg_grad = 0.95 * avg_grad + 0.05 * grad
        buffer_count += 1

        # FISTA step
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        y = a + theta * (a - a_prev)
        y = _project_simplex(y)

        # Step with averaged gradient
        gnorm = float(np.sqrt(np.sum(avg_grad * avg_grad))) + 1e-300
        eta_adapt = min(eta, 0.5 / (4.0 * cmax * gnorm / (gnorm * gnorm + 1e-300) + 1e-10))
        eta_adapt = min(eta, 1.0 / (4.0 * cmax + 1e-10))

        a_new = _project_simplex(y - eta_adapt * avg_grad)

        if not np.all(np.isfinite(a_new)):
            a_new = a.copy()
            avg_grad = avg_grad * 0.5

        # Check improvement
        c_new = _convolve_fft(a_new, L)
        s_new = 2.0 * n * float(c_new.max())

        if s_new < exact_score - 1e-15:
            a = a_new
            eta = min(eta * 1.05, 1.0 / (2.0 * cmax + 1e-10))
        else:
            eta = eta * 0.9
            a = a_new  # still move, but reduce step

        a_prev = a
        t = t_new

        # Restart if stuck
        if no_improve > 5000:
            scale = 0.003 * max(0.1, 1.0 - frac)
            a = _project_simplex(best_a + rng.normal(0.0, scale, size=n))
            a_prev = a.copy()
            t = 1.0
            avg_grad = np.zeros(n)
            no_improve = 0
            last_improve = time.monotonic()
            eta = 1.0 / (4.0 * best_score / (2.0 * n) + 1e-10)

    return best_a, best_score


def _coordinate_descent(a_init, deadline, L, seed):
    """
    Coordinate descent: for each coordinate, try several perturbation levels
    and keep the best one.
    """
    a = _normalize(a_init.copy())
    n = a.size
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    order = np.arange(n)

    # How many coordinates to try per sweep
    c_per_sweep = max(50, n // 20)

    while time.monotonic() < deadline:
        np.random.default_rng(seed).shuffle(order)
        for i in order:
            if time.monotonic() >= deadline:
                break

            orig = a[i]
            if orig < 1e-15:
                continue

            # Try several multipliers
            best_local = orig
            best_local_score = best_score

            for mult in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 0.3]:
                a[i] = orig * mult
                # Renormalize approximately (fast)
                s = a.sum()
                if s < 1e-15:
                    a[i] = orig
                    break
                a_norm = a / s

                c = _convolve_fft(a_norm, L)
                sc = 2.0 * n * float(c.max())

                if sc < best_local_score - 1e-15:
                    best_local_score = sc
                    best_local = orig * mult

            a[i] = best_local
            s = a.sum()
            a = a / s

            if best_local_score < best_score - 1e-15:
                best_score = best_local_score
                best_a = a.copy()

    return best_a, best_score


def _local_block_search(a_init, deadline, L, seed):
    """
    Structured block local search: perturb groups of adjacent elements.
    Focus on regions where the convolution is large.
    """
    a = _normalize(a_init.copy())
    n = a.size
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    sigma = 0.05

    # Identify important regions (where a is large)
    a_sorted = np.sort(a)[::-1]
    cutoff = a_sorted[max(0, n // 4)]
    important = np.where(a > cutoff * 0.5)[0]

    while time.monotonic() < deadline:
        # Choose a block position, preferring important regions
        if len(important) > 0 and rng.random() < 0.7:
            center = int(important[rng.integers(0, len(important))])
        else:
            center = int(rng.integers(0, n))

        w = int(np.exp(rng.uniform(np.log(2.0), np.log(max(4, n // 10)))))
        w = max(2, min(w, n // 2))
        s = max(0, min(center - w // 2, n - w))

        # Multiplicative perturbation
        block = a[s:s + w].copy()
        a[s:s + w] = block * np.exp(rng.normal(0.0, sigma, size=w))
        a = _project_simplex(a)

        sc = _score(a)
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a.copy()
            sigma = min(0.15, sigma * 1.03)
        else:
            a = best_a.copy()
            if time.monotonic() - start > 10:
                sigma = max(0.01, sigma * 0.98)

    return best_a, best_score


def _generate_power(n, p, offset):
    j = np.arange(n, dtype=np.float64)
    x = np.maximum(j + offset, 1e-8)
    v = np.power(x, -p)
    return _project_simplex(v)


def _generate_gaussian(n, center_frac, width):
    j = np.arange(n, dtype=np.float64)
    center = center_frac * (n - 1)
    v = np.exp(-0.5 * ((j - center) / width) ** 2)
    return _project_simplex(v)


def _generate_mixed(n, p1, p2, blend):
    v1 = _generate_power(n, p1, 1.0)
    v2 = _generate_power(n, p2, 1.0)
    v = blend * v1 + (1 - blend) * v2
    return _project_simplex(v)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the best non-negative sequence minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses multi-stage optimization: FISTA with log-sum-exp, subgradient polish,
    coordinate descent, and structured local search.
    """
    t0 = time.monotonic()
    budget = max(0.0, float(budget_s) - 30.0)
    deadline = t0 + budget

    # Determine sequence length
    n = 3809
    try:
        h = height_sequence_1
        if isinstance(h, (list, np.ndarray)):
            arr = np.asarray(h, dtype=np.float64)
            if arr.ndim == 1 and 100 <= arr.size <= 20000:
                n = int(arr.size)
    except Exception:
        pass

    if n < 32:
        n = 3809

    L = _pow2(2 * n - 1)

    # Get initial candidates
    a_init = None
    try:
        h = height_sequence_1
        if isinstance(h, (list, np.ndarray)):
            arr = np.asarray(h, dtype=np.float64)
            arr = _clean(arr)
            if arr.size == n and arr.sum() > 0.0:
                a_init = _project_simplex(arr)
    except Exception:
        pass

    if a_init is None:
        a_init = _generate_power(n, 0.5, 1.0)

    print(f"Starting with n={n}, L={L}")
    init_score = _score(a_init)
    print(f"Initial score: {init_score:.8f}")

    best_a = a_init.copy()
    best_score = init_score

    # Phase 1: FISTA with log-sum-exp (40% of time)
    p1_end = t0 + 0.40 * budget
    if time.monotonic() < p1_end:
        eps_start = 0.15 * init_score / (2.0 * n)
        eps_end = 1e-8
        a, sc = _fista_optimize(a_init, p1_end, L, int(seed) + 7, eps_start, eps_end)
        print(f"After FISTA: score={sc:.8f} (time={time.monotonic()-t0:.1f}s)")
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # Phase 2: Subgradient polish (30% of time)
    p2_end = t0 + 0.70 * budget
    if time.monotonic() < p2_end:
        a, sc = _subgradient_polish(best_a, p2_end, L, int(seed) + 107)
        print(f"After subgrad: score={sc:.8f} (time={time.monotonic()-t0:.1f}s)")
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # Phase 3: Coordinate descent (20% of time)
    p3_end = t0 + 0.90 * budget
    if time.monotonic() < p3_end:
        a, sc = _coordinate_descent(best_a, p3_end, L, int(seed) + 2003)
        print(f"After coord desc: score={sc:.8f} (time={time.monotonic()-t0:.1f}s)")
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # Phase 4: Local block search (remaining time)
    if time.monotonic() < deadline:
        a, sc = _local_block_search(best_a, deadline, L, int(seed) + 4001)
        print(f"After local search: score={sc:.8f} (time={time.monotonic()-t0:.1f}s)")
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # Try a few diverse restarts if time permits
    remaining = deadline - time.monotonic()
    if remaining > 60.0:
        diverse_starts = [
            _generate_power(n, 0.45, 0.5),
            _generate_power(n, 0.55, 1.5),
            _generate_gaussian(n, 0.0, n * 0.1),
            _generate_mixed(n, 0.4, 0.6, 0.5),
        ]
        for di, ds in enumerate(diverse_starts):
            if time.monotonic() >= deadline - 30:
                break
            ds_score = _score(ds)
            if ds_score >= best_score:
                continue
            seg_end = min(deadline - 5.0, time.monotonic() + remaining / (len(diverse_starts) - di))
            a, sc = _fista_optimize(ds, min(seg_end, time.monotonic() + 0.3 * (deadline - time.monotonic())), L, int(seed) + 5000 + di * 100, 0.1, 1e-8)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"  Diverse start {di}: improved to {sc:.8f}")

    # Final verification
    best_a = _project_simplex(best_a)
    out = [float(x) for x in best_a]

    # Ensure sum > 0.01 for the evaluator
    if sum(out) < 0.01:
        out = [float(x) for x in np.full(n, 1.0 / n)]

    try:
        val = evaluate_sequence(out)
        print(f"Final evaluate_sequence score: {val:.8f}")
        if not np.isfinite(val):
            out = [float(x) for x in _generate_power(n, 0.5, 1.0)]
    except Exception:
        pass

    print(f"FINAL: score={best_score:.8f}, time={time.monotonic()-t0:.1f}s")
    return out
