"""
L-BFGS + subgradient hybrid search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Strategy:

from scipy.optimize import minimize


def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def normalize_positive(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s


def fast_score(a):
    p = normalize_positive(a)
    if p is None:
        return np.inf
    n = p.size
    if n <= 0:
        return np.inf
    L = next_pow2(2 * n - 1)
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0.0:
        return np.inf
    return float(2.0 * n * cm)


def softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = np.where(np.isfinite(z), z, -50.0)
    z = z - z.max()
    z = np.clip(z, -60.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(z.size) / z.size
    return e / s


def make_smooth_max_fn(n, L, tau):
    """Return (fun, jac) pair for L-BFGS operating on log-space z."""
    def fun(z):
        a = softmax(z)
        F = np.fft.rfft(a, L)
        c = np.fft.irfft(F * F, L)[:2 * n - 1]
        c = np.where(np.isfinite(c), c, 0.0)
        c = np.maximum(c, 0.0)
        c_max = c.max()
        if not np.isfinite(c_max) or c_max <= 0:
            return 1e10
        zc = (c - c_max) / tau
        exp_zc = np.exp(zc)
        smooth_val = c_max + tau * np.log(exp_zc.sum())
        return 2.0 * n * smooth_val

    def jac(z):
        a = softmax(z)
        F = np.fft.rfft(a, L)
        c = np.fft.irfft(F * F, L)[:2 * n - 1]
        c = np.where(np.isfinite(c), c, 0.0)

        if not np.isfinite(c_max) or c_max <= 0:
            return np.zeros(n)
        zc = (c - c_max) / tau
        exp_zc = np.exp(zc)
        w = exp_zc / exp_zc.sum()
        Wtau = tau * w
        Wref = Wtau[::-1]

        g_z = a * (g_a - mean_g)
        g_z = np.where(np.isfinite(g_z), g_z, 0.0)

def lbfgs_run(z, n, L, tau, max_iter, ftol, gtol):
    fun, jac = make_smooth_max_fn(n, L, tau)
    try:
        res = minimize(
            fun, z, jac=jac, method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': ftol, 'gtol': gtol,
                     'maxls': 50, 'maxfun': max_iter * 10}
        )
        return res.x, res.fun
    except Exception:
        return z, np.inf


def subgradient_polish(p, n, L, steps, lr, rng):
    """Nesterov-like subgradient descent on the exact max objective."""
    a = p.copy()
    z = np.log(np.maximum(a, 1e-30))
    z -= z.mean()
    v = np.zeros(n)
    best = a.copy()
    best_s = fast_score(a)
    mu = 0.9

    for it in range(steps):
        a = softmax(z)
        F = np.fft.rfft(a, L)
        c = np.fft.irfft(F * F, L)[:2 * n - 1]
        c = np.where(np.isfinite(c), c, 0.0)
        c = np.maximum(c, 0.0)
        c_max = c.max()
        if not np.isfinite(c_max) or c_max <= 0:
            break

        # Find all positions within 1% of max (top-K subgradient averaging)
        threshold = c_max * 0.999
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        # Compute subgradient: average over active peaks
        # dc_k/da_i = 2*a_{k-i} (approx, ignoring boundary)
        # g_a_i = (1/|active|) * sum_{k in active} 2*a_{k-i}
        # = 2/|active| * (w_active * a) where w_active is indicator of active set
        w_active = np.zeros(2 * n - 1)
        w_active[active] = 1.0 / len(active)
        Wref = w_active[::-1]
        H = np.fft.irfft(F * np.fft.rfft(Wref, L), L)
        g_a = 2.0 * H[n - 1:2 * n - 1][::-1]
        g_a = np.where(np.isfinite(g_a), g_a, 0.0)

        mean_g = float(np.dot(a, g_a))
        g_z = a * (g_a - mean_g)

        # Nesterov momentum
        v = mu * v + lr * g_z
        z = z - v
        z = np.where(np.isfinite(z), z, 0.0)
        z -= z.mean()
        z = np.clip(z, -60.0, 60.0)

        s = fast_score(softmax(z))
        if np.isfinite(s) and s < best_s:
            best_s = s
            best = softmax(z).copy()

        # Adaptive decay
        if (it + 1) % 1000 == 0:
            lr *= 0.85

    return best, best_s


def cheb_basis(n, D):
    x = np.linspace(0.0, 1.0, n)
    t = 2.0 * x - 1.0
    t = np.clip(t, -1.0, 1.0)
    ang = np.arccos(t)
    out = np.empty((n, D), dtype=np.float64)
    for m in range(1, D + 1):
        out[:, m - 1] = np.cos(m * ang)
    return out


def resize_sequence(seq, m):
    p = normalize_positive(seq)
    m = int(m)
    if p is None:
        return np.ones(m, dtype=np.float64) / float(m)
    n = p.size
    if n == m:
        return p.copy()
    if n == 1:
        return np.full(m, 1.0 / float(m), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, m)
    out = np.interp(x_new, x_old, p)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(m, dtype=np.float64) / float(m)
    return out / s

        # Try several structured initializations
        inits = [
            np.ones(N) / N,
            np.linspace(0.1, 1.0, N) / N,
            np.linspace(1.0, 0.1, N) / N,
            np.exp(-2.0 * (np.linspace(0, 1, N) - 0.5)**2) ,
            1.0 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, N)),
        ]
        for init in inits:
            p_init = normalize_positive(init)
            if p_init is not None:
                s_init = fast_score(p_init)
                if np.isfinite(s_init) and s_init < best_s:
                    best_s = s_init
                    best_p = p_init

        # Random Chebyshev scan
        for _ in range(200):
            theta = rng.normal(0.0, 0.5, D)
            val = B.dot(theta)
            h = np.exp(np.clip(val, -5, 5))
            p_init = normalize_positive(h)
            if p_init is not None:
                s_init = fast_score(p_init)
                if np.isfinite(s_init) and s_init < best_s:
                    best_s = s_init
                    best_p = p_init

        print(f"Best initial: score={best_s:.6f}")

    if best_p is None:
        best_p = np.ones(N) / float(N)
        best_s = 2.0

    # --- L-BFGS with tau annealing ---
    # Multiple full annealing cycles with noise between them
    n_cycles = 8
    tau_schedule = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]

    z_current = get_initial_z(best_p, rng)

    for cycle in range(n_cycles):
        if time.monotonic() >= deadline - 120:
            break

        cycle_start = time.monotonic()
        print(f"\n=== Annealing cycle {cycle + 1}/{n_cycles} ===")

        # Add noise for restarts (except first cycle)
        if cycle > 0:
            noise_scale = 0.3 / (cycle + 1)
            z_current = z_current + rng.normal(0.0, noise_scale, N)
            z_current = np.clip(z_current, -60, 60)
            z_current -= z_current.mean()

        for tau_frac in tau_schedule:
            if time.monotonic() >= deadline - 120:
                break

            tau = tau_frac / N
            max_iter = 300 if tau_frac >= 0.01 else 500

            t0 = time.monotonic()
            z_new, obj_val = lbfgs_run(z_current, N, L, tau, max_iter,
                                        ftol=1e-12, gtol=1e-10)
            elapsed = time.monotonic() - t0

            # Evaluate true score
            a_new = softmax(z_new)
            s_new = fast_score(a_new)

            if np.isfinite(s_new) and s_new < best_s:
                best_s = s_new
                best_p = a_new.copy()
                z_current = z_new.copy()
                print(f"  tau={tau_frac:>8.4f}  score={s_new:.6f}  "
                      f"obj={obj_val:.6f}  time={elapsed:.1f}s  [NEW BEST]")
            else:
                # Still update z_current to the L-BFGS solution (it may be useful)
                z_current = z_new.copy()
                print(f"  tau={tau_frac:>8.4f}  score={s_new:.6f}  "
                      f"obj={obj_val:.6f}  time={elapsed:.1f}s")

        cycle_time = time.monotonic() - cycle_start
        print(f"  Cycle {cycle + 1} done in {cycle_time:.1f}s, best={best_s:.6f}")

    # --- Subgradient polish ---
    rem = deadline - time.monotonic()
    if rem > 60:
        print(f"\n=== Subgradient polish (remaining {rem:.0f}s) ===")
        t0 = time.monotonic()
        for sg_round in range(5):
            if time.monotonic() >= deadline - 30:
                break
            p_sg, s_sg = subgradient_polish(best_p, N, L,
                                            steps=20000, lr=0.05, rng=rng)
            if np.isfinite(s_sg) and s_sg < best_s:
                best_s = s_sg
                best_p = p_sg.copy()
                print(f"  SG round {sg_round + 1}: score={s_sg:.6f} [NEW BEST]")
            else:
                print(f"  SG round {sg_round + 1}: score={s_sg:.6f}")
        print(f"Subgradient polish done in {time.monotonic() - t0:.1f}s")

    # --- Final L-BFGS at very small tau on the best found ---
    rem = deadline - time.monotonic()
    if rem > 30 and best_p is not None:
        print(f"\n=== Final L-BFGS polish (remaining {rem:.0f}s) ===")
        z_final = get_initial_z(best_p, rng)
        for tau_frac in [0.0001, 0.00005, 0.00002]:
            if time.monotonic() >= deadline - 10:
                break
            tau = tau_frac / N
            z_final, _ = lbfgs_run(z_final, N, L, tau, max_iter=500,
                                    ftol=1e-14, gtol=1e-12)
            a_final = softmax(z_final)
            s_final = fast_score(a_final)
            if np.isfinite(s_final) and s_final < best_s:
                best_s = s_final
                best_p = a_final.copy()
                print(f"  Final L-BFGS tau={tau_frac}: score={s_final:.6f} [NEW BEST]")
            else:
                print(f"  Final L-BFGS tau={tau_frac}: score={s_final:.6f}")

    # --- Validation with evaluate_sequence ---
    print(f"\n=== Final validation ===")
    candidates = [best_p]
    # Also try a few shorter lengths as fallback
    for fallback_n in [4096, 2048, 1024]:
        if best_p is not None:
            candidates.append(resize_sequence(best_p, fallback_n))
    candidates.append(np.ones(1024) / 1024.0)

    final_seq = None
    final_score = np.inf

    for cand in candidates:
        if cand is None:
            continue
        lst = [float(x) for x in np.clip(np.asarray(cand, dtype=np.float64), 0.0, 1000.0)]
        if not lst or len(lst) < 2:
            continue
        try:
            sc = evaluate_sequence(lst)
        except Exception:
            sc = np.inf
        if np.isfinite(sc) and sc < final_score:
            final_score = float(sc)
            final_seq = lst
            print(f"  n={len(lst)}: score={sc:.6f}")

    if final_seq is None:
        final_seq = [1.0 / 1024.0] * 1024
        print("  WARNING: Using uniform fallback")

    total_time = time.monotonic() - start_time
    print(f"\nTotal time: {total_time:.1f}s")
    print(f"Best fast_score: {best_s:.6f}")
    print(f"Best evaluate_sequence score: {final_score:.6f}")

    return final_seq
