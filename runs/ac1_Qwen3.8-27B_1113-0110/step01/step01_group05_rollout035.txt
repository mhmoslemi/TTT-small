3. The values a[i] and a[k*-i] contribute to conv[k*]
4. Reduce the "product" a[i]*a[k*-i] by adjusting a[i] and/or a[k*-i]
5. Renormalize

Specifically, for the peak at k*:
conv[k*] = sum_i a[i] * a[k*-i]

To reduce this, we can:
- For each pair (i, k*-i) with both in [0, n-1], reduce the product
- The most effective way is to reduce the larger of the two values (to minimize the sum of squares increase)

This is similar to "peak picking" in signal processing.

A concrete algorithm:
1. Find k* = argmax conv
2. For each i in [0, n-1] where k*-i is also in [0, n-1]:
   - Let x = a[i], y = a[k*-i]
   - We want to reduce x*y. The optimal adjustment (minimizing the increase in x^2+y^2) is to set:
     x_new = x - delta, y_new = y - delta (if we reduce both equally)
     Or: reduce only the larger one.
   - Actually, the optimal is to equalize: set both to (x+y)/2. But this might not be feasible.

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Coordinate descent with dense log grids
2. LP-based direction finding (max sum(g) s.t. cross-corr(a,g) <= M, g >= 0)
3. Subgradient steps for escape from local minima
4. Random restarts with structured initializations

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog


def _clean(arr):
    a = np.asarray(arr, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    np.clip(a, 0.0, 1000.0, out=a)
    if a.size == 0:
        a = np.ones(1)
    return a


def _normalize(a):
    a = _clean(a).copy()
    s = a.sum()
    if s <= 0:
        return np.ones(a.size) / a.size
    return a / s


def _score(a):
    a = _normalize(a)
    c = np.convolve(a, a)
    return 2.0 * len(a) * float(np.max(c))


def _coord_descent(a, deadline, rng, max_rounds=100, grid_size=20, refine=8):
    """Coordinate descent with log-scale grid search per coordinate."""
    n = len(a)
    a = _normalize(a).copy()
    c = np.convolve(a, a)
    s = a.sum()  # should be 1
    best_score = 2.0 * n * float(np.max(c))
    best_a = a.copy()
    
    # Precompute grid
    grid = np.concatenate(([0.0], np.logspace(-10, 2, grid_size)))
    
    rounds = 0
    stagnant = 0
    
    while time.monotonic() < deadline and rounds < max_rounds:
        rounds += 1
        improved = False
        
        for i in range(n):
            if time.monotonic() > deadline:
                break
            
            old = a[i]
            D = 1.0 - old  # since sum = 1, delta = new - old, new_sum = 1 + delta
            
            # conv[i : i+n] is affected
            # c_new[k] = (c[k] + (new-old)*(a[k-i] + a[i-(k-i)] if valid) + (new-old)^2 if k == 2i) / (1+delta)^2
            # Actually: c[k] = sum_j a[j]*a[k-j]
            # c_new[k] = sum_j a[j]*a'[k-j] where a' differs from a only at position i
            # = c[k] - a[i]*a[k-i] + a'[i]*a[k-i] (if k-i valid, k-i != i)
            #   + a[i]*a[i-k]... 
            # More carefully:
            # c[k] = sum_j a[j]*a[k-j]
            # The terms involving a[i] are: j=i gives a[i]*a[k-i], and j=k-i gives a[k-i]*a[i]
            # If i != k-i: two terms, each a[i]*a[k-i], so total contribution is 2*a[i]*a[k-i]
            # If i == k-i (k=2i): one term a[i]^2
            # 
            # After changing a[i] to a'[i] = a[i] + delta:
            # c_new[k] = c[k] - (contribution of old a[i]) + (contribution of new a[i])
            # For k where both i and k-i are valid and i != k-i:
            #   c_new[k] = c[k] - 2*a[i]*a[k-i] + 2*a'[i]*a[k-i] = c[k] + 2*delta*a[k-i]
            # For k = 2i:
            #   c_new[k] = c[k] - a[i]^2 + a'[i]^2 = c[k] + 2*delta*a[i] + delta^2
            # For k where only one of i, k-i is valid (the other is out of bounds):
            #   c_new[k] = c[k] + delta * a[valid_one]
            # For k where neither is valid: no change.
            #
            # But we also need to renormalize: a' = a_new / sum(a_new)
            # sum(a_new) = 1 + delta
            # So the final normalized c is: c_new / (1+delta)^2
            
            # For simplicity, let's compute the score for each candidate value
            # a[i] candidate values: grid values
            # new_sum = 1 - old + candidate = D + candidate
            
            candidates = grid
            # Also include the current value
            candidates = np.unique(np.concatenate((candidates, [old])))
            candidates = candidates[(candidates >= 0) & (candidates <= 100)]
            
            best_val = best_score
            best_cand = old
            
            for cand in candidates:
                delta = cand - old
                S = 1.0 + delta
                if S <= 1e-15:
                    continue
                
                # Compute new max conv
                # Affected lags: k where i or k-i is in [0, n-1]
                # k ranges from i to i+n-1 (for the a[i]*a[k-i] term)
                # and from k-i to k-i... actually k-i in [0,n-1] means k in [i, i+n-1]
                # and i in [0,n-1] is always true for our i
                # Also the a[k-i]*a[i] term: k-i in [0,n-1] means k in [i, i+n-1]
                # So affected range is k in [i, i+n-1] (intersected with [0, 2n-2])
                
                k_lo = max(0, i)
                k_hi = min(2*n-2, i+n-1)
                
                # Unaffected max
                if k_lo > 0:
                    unaffected = np.max(c[:k_lo])
                else:
                    unaffected = 0.0
                if k_hi < 2*n-2:
                    tail = np.max(c[k_hi+1:])
                    if tail > unaffected:
                        unaffected = tail
                else:
                    unaffected = 0.0
                
                # Affected part
                # c_affected[k] for k in [k_lo, k_hi]
                # c_affected_new[k] = c[k] + 2*delta*a[k-i] (if i != k-i and both valid)
                #   + delta*a[k-i] (if only k-i valid, i not... but i is always valid)
                #   Actually let me just compute it directly
                # 
                # c[k] = sum_j a[j]*a[k-j]
                # c_new[k] = c[k] - 2*a[i]*a[k-i] + 2*cand*a[k-i] (for k-i valid, k-i != i)
                #          = c[k] - a[i]^2 + cand^2 (for k = 2i)
                #          = c[k] - a[i]*a[k-i] + cand*a[k-i] (for k-i valid, i not valid - impossible since i is valid)
                #          = c[k] - a[k-i]*a[i] + a[k-i]*cand (for i valid, k-i not valid - k-i out of bounds)
                # 
                # Wait, I need to be more careful.
                # c[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j] * a[k-j]
                # The terms involving a[i]:
                #   j = i: a[i] * a[k-i] (valid if 0 <= k-i < n)
                #   j = k-i: a[k-i] * a[i] (valid if 0 <= k-i < n, same condition)
                # So if 0 <= k-i < n:
                #   If i != k-i: two terms, total 2*a[i]*a[k-i]
                #   If i == k-i: one term, a[i]^2
                # If not (0 <= k-i < n): no terms involving a[i]
                #
                # After change to cand:
                # If 0 <= k-i < n:
                #   If i != k-i: 2*cand*a[k-i]
                #   If i == k-i: cand^2
                #
                # Delta in c[k]:
                # If 0 <= k-i < n and i != k-i: 2*(cand - a[i])*a[k-i] = 2*delta*a[k-i]
                # If 0 <= k-i < n and i == k-i (k=2i): cand^2 - a[i]^2 = 2*delta*a[i] + delta^2
                # Otherwise: 0
                
                # For the affected range k in [i, i+n-1] (where k-i is in [0, n-1]):
                k_range = np.arange(k_lo, k_hi+1)
                ki = k_range - i  # = k-i, ranges from k_lo-i to k_hi-i, which is [max(0,i)-i, min(2n-2,i+n-1)-i]
                # ki ranges from max(0, i)-i to min(2n-1-i, n-1)
                # = max(-i, 0) to min(2n-1-i, n-1)
                # Since i in [0, n-1]: ki ranges from 0 to min(n-1, 2n-1-i)
                # = 0 to n-1 (since 2n-1-i >= n for i <= n-1)
                # So ki = 0, 1, ..., n-1 for the full range k = i, i+1, ..., i+n-1
                
                # c_new[k] = c[k] + 2*delta*a[k-i] for k-i != i (i.e., k != 2i)
                # c_new[k] = c[k] + 2*delta*a[i] + delta^2 for k = 2i
                
                ki_vals = k_range - i  # these are in [0, n-1]
                c_aff = c[k_range].copy()
                c_aff += 2.0 * delta * a[ki_vals]
                # Correct the k=2i term
                idx_2i = 2*i
                if k_lo <= idx_2i <= k_hi:
                    c_aff[idx_2i - k_lo] = c[idx_2i] + 2.0*delta*a[i] + delta*delta
                
                max_aff = np.max(c_aff)
                total_max = max(unaffected, max_aff)
                
                val = 2.0 * n * total_max / (S * S)
                if val < best_val:
                    best_val = val
                    best_cand = cand
            
            if best_cand != old:
                a[i] = best_cand
                s_new = a.sum()
                if s_new > 0:
                    a = a / s_new
                    c = np.convolve(a, a)
                    new_score = 2.0 * n * float(np.max(c))
                    if new_score < best_score - 1e-12:
                        best_score = new_score
                        best_a = a.copy()
                        improved = True
        
        if not improved:
            stagnant += 1
            if stagnant >= 3:
                break
        else:
            stagnant = 0
    
    return best_a, best_score

def _build_lp_matrix(a):
    n = len(a)
    # A has shape (2n-1, n)
    # A[k, i] = a[k-i] if 0 <= k-i < n
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n-1, k)
        for i in range(i_min, i_max + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A
