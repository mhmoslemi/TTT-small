def resize_sequence(seq, m):
    p = np.asarray(seq, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)

    s = p.sum()
    if s <= 1e-300:
        return np.ones(m) / m
    p = p / s
    n = p.size
    if n == m:
        return p.copy()
    if n == 1:
        return np.full(m, 1.0 / m)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, m)
    out = np.interp(x_new, x_old, p)
    out = np.maximum(out, 0.0)
    so = out.sum()
    if so <= 0:
        return np.ones(m) / m
    return out / so
