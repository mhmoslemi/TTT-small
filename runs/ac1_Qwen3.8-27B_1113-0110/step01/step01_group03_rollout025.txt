        return np.inf
    c = fft_self_conv(a)
    return 2.0 * n * c.max() / (S*S)

def normalize(a, target):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), target/len(a))
    return a / s * target

def solve_1d_opt(B, a, i, S, n, x0):
    """Given B (background), find optimal x for a[i]."""
    kq = 2 * i
    ks_min = max(0, i)
    ks_max = min(2*n - 2, i + n - 1)
    
    # Precompute constant part
    active = np.zeros(2*n-1, dtype=bool)
    active[ks_min:ks_max+1] = True
    if kq < 2*n-1:
        active[kq] = True
    const_part = B[~active].max() if np.any(~active) else 0.0
    
    # Linear part
    lin_mask = active.copy()
    if kq < 2*n-1:
        lin_mask[kq] = False
    lin_idx = np.where(lin_mask)[0]
    if len(lin_idx) > 0:
        lin_B = B[lin_idx]
        lin_a = a[lin_idx - i]
    else:
        lin_B = np.zeros(1)
        lin_a = np.zeros(1)
    
    quad_B = B[kq] if kq < 2*n-1 else -np.inf
    has_quad = kq < 2*n-1
    
    def score_x(x):
        total = S + x
        if total < 1e-15:
            return 1e10
        M = const_part
        if has_quad:
            M = max(M, quad_B + x*x)
        if len(lin_idx) > 0:
            M = max(M, (lin_B + 2.0*x*lin_a).max())
        return 2.0*n*M / (total*total)
    
    # Determine search bounds
    x_hi = max(2.0*x0 + S/n*0.5, S/n)
    x_hi = min(x_hi, S * 0.9)  # don't let x exceed total sum
    x_hi = max(x_hi, 0.01)
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_b, b_b = 0.0, x_hi
    c_pt = b_b - gr * (b_b - a_b)
    d_pt = a_b + gr * (b_b - a_b)
    fc = score_x(c_pt)
    fd = score_x(d_pt)
    
    for _ in range(45):
        if b_b - a_b < 1e-9 * max(1.0, S/n):
            break
        if fc < fd:
            b_b = d_pt; d_pt = c_pt; fd = fc
            c_pt = b_b - gr * (b_b - a_b); fc = score_x(c_pt)
        else:
            a_b = c_pt; c_pt = d_pt; fc = fd
            d_pt = a_b + gr * (b_b - a_b); fd = score_x(d_pt)
    
    return (a_b + b_b) / 2.0

def coord_descent(a, n, num_sweeps, time_limit):
    """Run coordinate descent. Returns best (a, score)."""
    a = a.copy()
    a = normalize(a, n)
    c = fft_self_conv(a)
    best_a = a.copy()
    best_score = fast_score(a)
    
    start = time.monotonic()
    sweep = 0
    
    while sweep < num_sweeps and time.monotonic() - start < time_limit:
        S_full = a.sum()
        for i in range(n):
            if time.monotonic() - start > time_limit:
                break
            S = S_full - a[i]
            x0 = a[i]
