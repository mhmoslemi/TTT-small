def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)

import numpy as np
import time

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]

def _fft_cross_corr(w, a):
    n = len(a)
    a_rev = a[::-1]
    size = len(w) + n - 1
    fft_size = _next_pow2(size)
    fw = np.fft.rfft(w, fft_size)
    fa = np.fft.rfft(a_rev, fft_size)
    result = np.fft.irfft(fw * fa, fft_size)
    return result[n-1 : 2*n-1]

def _project_simplex(v):
    n = len(v)
    s = v.sum()
    if s < 1e-30:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _smooth_score(a, beta):
    """Compute smooth approximation of the score."""
    n = len(a)
    c = _fft_convolve(a)
    mc = np.max(c)
    shifted = np.clip(beta * (c - mc), -50, 50)
    exps = np.exp(shifted)
    val = mc + np.log(np.sum(exps)) / beta
    return 2.0 * n * val, c, exps / np.sum(exps)

def _exact_score(a):
    n = len(a)
    c = _fft_convolve(a)
    return 2.0 * n * float(np.max(c))

    n = len(a0)

    beta = beta_start

    momentum = np.zeros(n)
    mu = 0.3
    
    it = 0
    no_improve = 0
    last_print = time.monotonic()

        c = _fft_convolve(a)
        score = 2.0 * n * float(np.max(c))

            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute smooth gradient
        mc = np.max(c)
        shifted = np.clip(beta * (c - mc), -50, 50)
        exps = np.exp(shifted)
        w = exps / np.sum(exps)
        
        grad = 2.0 * _fft_cross_corr(w, a)
        
        # Normalize gradient
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:

        grad = grad / g_norm  # normalize to unit norm

        momentum = mu * momentum + (1.0 - mu) * grad

        a_new = _project_simplex(a_new)

                eta = 0.1
