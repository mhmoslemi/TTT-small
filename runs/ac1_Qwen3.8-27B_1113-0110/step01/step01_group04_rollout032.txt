def _clean_nonnegative(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    return np.maximum(v, 0.0)

def _project_simplex(v):
    v = _clean_nonnegative(v)
    n = v.size
    if n == 0:
        return v.copy()
    x = np.sort(v)[::-1]
    css = np.cumsum(x)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = x > (css - 1.0) / idx
    if not np.any(cond):
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    r = int(np.nonzero(cond)[0][-1])
    tau = (css[r] - 1.0) / float(r + 1)
    out = v - tau
    np.maximum(out, 0.0, out=out)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    out *= 1.0 / s
    return out

def _score_vector(a, L=None):
    v = _clean_nonnegative(a)
    n = v.size
    if n == 0:
        return np.inf
    s = float(v.sum())
    if s < 1e-12:
        return np.inf
    v = v / s
    if L is None:
        L = _pow2(2 * n - 1)
    A = np.fft.rfft(v, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    mx = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(mx):
        return np.inf
    if mx < 0.0:
        mx = 0.0
    return 2.0 * float(n) * mx

def _power_sequence(n, p=0.5, offset=1.0):
    j = np.arange(n, dtype=np.float64)
    x = np.maximum(j + float(offset), 1e-8)
    return _project_simplex(np.power(x, -float(p)))

def _two_tail_sequence(n, p=0.5, tw=0.1):
    j = np.arange(n, dtype=np.float64)
    base = np.power(np.maximum(j + 1.0, 1e-8), -float(p))
    rev = np.power(np.maximum((n - 1.0 - j) + 1.0, 1e-8), -float(p))
    return _project_simplex(base + tw * rev)

def _generate_candidates(n, prev_seq=None):
    cands = []
    if prev_seq is not None:
        arr = np.asarray(prev_seq, dtype=np.float64)
        arr = _clean_nonnegative(arr)
        if arr.size == n and arr.sum() > 0:
            cands.append(_project_simplex(arr))
    cands.append(np.full(n, 1.0 / float(n), dtype=np.float64))
    for p in (0.45, 0.475, 0.5, 0.525, 0.55, 0.6):
        for off in (0.5, 1.0, 1.5, 2.0):
            cands.append(_power_sequence(n, p, off))
    for tw in (0.05, 0.1, 0.2, 0.5):
        cands.append(_two_tail_sequence(n, 0.5, tw))
    for seed in (7, 13, 31, 57, 83):
        rng = np.random.default_rng(seed)
        # Fourier-based: random phases with 1/k amplitude
        k = np.arange(n, dtype=np.float64)
        phases = rng.uniform(0, 2 * np.pi, size=n)
        amps = 1.0 / (1.0 + k * 0.3)
        v = 1.0 + 0.5 * np.sum(amps * np.cos(2 * np.pi * k[None, :] * np.arange(n)[:, None] / n + phases[None, :]), axis=1)
        v = _clean_nonnegative(v)
        if v.sum() > 0:
            cands.append(_project_simplex(v))
    # Random smooth sequences
    for seed in (101, 202, 303):
        rng = np.random.default_rng(seed)
        raw = rng.normal(0, 1, size=n)
        # Smooth with a large kernel
        from scipy.ndimage import gaussian_filter1d
        v = gaussian_filter1d(raw, sigma=n * 0.05)
        v = _clean_nonnegative(v)
        if v.sum() > 0:
            cands.append(_project_simplex(v))
    return cands

def _solve_lp_direction(f, M, time_limit=120):
    """Solve max sum(b) s.t. (f*b)_k <= M for all k, b >= 0.
    Returns optimal b vector or None if infeasible/too slow."""
    n = f.size
    num_constraints = 2 * n - 1
    # Build sparse constraint matrix
    # Row k, col j: f[k-j] if 0 <= k-j <= n-1
    rows = []
    cols = []
    vals = []
    for k in range(num_constraints):
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            idx = k - j
            if f[idx] > 1e-15:
                rows.append(k)
                cols.append(j)
                vals.append(f[idx])
    if not rows:
        return None
    A_sparse = csr_matrix((vals, (rows, cols)), shape=(num_constraints, n))
    # LP: max c^T b s.t. A b <= M, b >= 0
    # linprog minimizes, so: minimize -sum(b) s.t. A b <= M, b >= 0
    c_obj = -np.ones(n)
    try:
        res = linprog(
            c_obj,
            A_ub=A_sparse,
            b_ub=np.full(num_constraints, M),
            method='highs',
            options={'time_limit': time_limit, 'presolve': True}
        )
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def _lp_refine(f, end_time, max_iterations=200, lp_time_limit=60):
    """Iteratively refine f using LP directions with line search."""
    n = f.size
    L = _pow2(2 * n - 1)
    best = _project_simplex(f).copy()
    best_score = _score_vector(best, L)
    current = best.copy()
    last_improve = time.monotonic()
    
    for it in range(max_iterations):
        if time.monotonic() > end_time - 30:
            break
        elapsed = time.monotonic() - last_improve
        if elapsed > 120:
            break
        
        # Compute M
        s = float(current.sum())
        if s < 1e-12:
            break
        c = current / s
        conv = np.convolve(c, c, mode='full')
        M = float(np.max(conv))
        if M < 1e-15:
            break
        
        # Solve LP
        b_opt = _solve_lp_direction(c, M, time_limit=lp_time_limit)
        if b_opt is None:
            break
        sum_b = float(b_opt.sum())
        if sum_b < 1e-12:
            break
        
        g = b_opt / sum_b  # normalize to sum=1
        
        # Line search: try (1-t)*current + t*g
        best_t = 0.0
        best_sc = best_score
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            candidate = (1 - t) * current + t * g
            candidate = _project_simplex(candidate)
            sc = _score_vector(candidate, L)
            if sc < best_sc - 1e-10:
                best_sc = sc
                best_t = t
        
        if best_t > 0:
            current = (1 - best_t) * current + best_t * g
            current = _project_simplex(current)
            current_score = _score_vector(current, L)
            if current_score < best_score - 1e-12:
                best_score = current_score
                best = current.copy()
                last_improve = time.monotonic()
        
        if it % 10 == 0:
            print(f"  LP iter {it}: score={best_score:.6f}, M={M:.6f}")
    
    return best, best_score

def _soft_objective_grad(x, a_ref, eps, L):
    """Compute f = eps*logsumexp(c/eps) and gradient where a = softmax(x) on simplex."""
    n = a_ref.size
    # Reconstruct a from x (log-ratios to last element)
    m = 0.0
    if x.size:
        mx = float(np.max(x))
        if mx > m:
            m = mx
    e = np.exp(np.clip(x - m, -745.0, 709.0))
    e_last = float(np.exp(-m))
    s = float(e.sum()) + e_last
    if not np.isfinite(s) or s <= 0:
        return 1e300, np.zeros_like(x)
    a = np.empty(n, dtype=np.float64)
    a[:-1] = e / s
    a[-1] = e_last / s
    
    A = np.fft.rfft(a, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    cmax = float(c.max())
    if cmax < 1e-300:
        return 0.0, np.zeros_like(x)
    
    z = (c - cmax) / eps
    z = np.clip(z, -745.0, 0.0)
    e2 = np.exp(z)
    sume = float(e2.sum())
    if not np.isfinite(sume) or sume <= 0:
        return 1e300, np.zeros_like(x)
    f = cmax + eps * np.log(sume)
    w = e2 / sume
    
    W = np.fft.rfft(w, n=L)
    ga = 2.0 * np.fft.irfft(W * np.conj(A), n=L)[:n]
    dot = float(np.dot(a, ga))
    gfull = a * (ga - dot)
    gy = gfull[:-1]
    if not np.all(np.isfinite(gy)):
        gy = np.zeros_like(x)
    return float(f), gy

def _q_objective_grad(x, q, L):
    n = x.size + 1
    m = 0.0
    if x.size:
        mx = float(np.max(x))
        if mx > m:
            m = mx
    e = np.exp(np.clip(x - m, -745.0, 709.0))
    e_last = float(np.exp(-m))
    s = float(e.sum()) + e_last
    if not np.isfinite(s) or s <= 0:
        return 1e300, np.zeros_like(x)
    a = np.empty(n, dtype=np.float64)
    a[:-1] = e / s
    a[-1] = e_last / s
    
    A = np.fft.rfft(a, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    
    if q == 2.0:
        f = float(np.dot(c, c))
        w = c
    else:
        f = float(np.sum(np.power(c, q)))
        w = np.power(c, q - 1.0)
    
    W = np.fft.rfft(w, n=L)
    ga = (2.0 * q) * np.fft.irfft(W * np.conj(A), n=L)[:n]
    dot = float(np.dot(a, ga))
    gfull = a * (ga - dot)
    gy = gfull[:-1]
    if not np.all(np.isfinite(gy)):
        gy = np.zeros_like(x)
    return f, gy

def _ratios_from_a(a):
    a = _clean_nonnegative(np.asarray(a, dtype=np.float64)).copy()
    n = a.size
    if n == 0:
        return a
    s = float(a.sum())
    if s <= 0 or not np.isfinite(s):
        a = np.full(n, 1.0 / float(n), dtype=np.float64)
        s = 1.0
    else:
        a = a / s
    if n == 1:
        return a
    ref = float(a[-1])
    if ref <= 1e-200:
        a[-1] = 1e-12
        a = a / float(a.sum())
        ref = float(a[-1])
    x = np.log(np.maximum(a[:-1], 1e-300) / max(ref, 1e-300))
    return np.clip(x, -100.0, 100.0)

def _softmax_from_ratios(x):
    x = np.asarray(x, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    n = x.size + 1
    if n == 1:
        return np.ones(1, dtype=np.float64)
    m = 0.0
    if x.size:
        mx = float(np.max(x))
        if mx > m:
            m = mx
    e = np.exp(np.clip(x - m, -745.0, 709.0))
    e_last = float(np.exp(-m))
    s = float(e.sum()) + e_last
    if not np.isfinite(s) or s <= 0:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    a = np.empty(n, dtype=np.float64)
    a[:-1] = e / s
    a[-1] = e_last / s
    return a

def _lbfgs_refine(a0, end_time, L):
    """Polish a sequence using L-BFGS on decreasing-temperature soft-max objective."""
    n = a0.size
    x = _ratios_from_a(a0)
    bounds = [(-60.0, 60.0)] * max(0, n - 1)
    
    best = _project_simplex(a0).copy()
    best_score = _score_vector(best, L)
    
    if x.size == 0:
        return best, best_score
    
    # Start with moderate temperature, decrease over time
    eps = max(1e-6, 0.05 * best_score / (2.0 * n))
    eps_min = 1e-8
    
    def fun_and_grad(x_curr):
        return _soft_objective_grad(x_curr, a0, eps, L)
    
    no_improve = 0
    last_improve = time.monotonic()
    
    while time.monotonic() < end_time - 10:
        # Run L-BFGS for a while
        chunk_end = min(time.monotonic() + 30, end_time - 10)
        while time.monotonic() < chunk_end:
            try:
                res = minimize(
                    fun_and_grad, x, method='L-BFGS-B', jac=True,
                    bounds=bounds,
                    options={'maxiter': 50, 'maxfun': 500, 'ftol': 1e-16, 'gtol': 1e-10}
                )
                x = np.clip(np.asarray(res.x, dtype=np.float64), -60, 60)
                x = np.where(np.isfinite(x), x, 0.0)
            except Exception:
                break
        
        a_new = _softmax_from_ratios(x)
        sc = _score_vector(a_new, L)
        now = time.monotonic()
        
        if sc < best_score - 1e-13:
            best_score = sc
            best = a_new.copy()
            last_improve = now
            no_improve = 0
            eps = max(eps_min, eps * 0.75)
        else:
            no_improve += 1
            # Perturb and/or decrease eps
            rng = np.random.default_rng(int(now * 1000) % 100000)
            x = np.clip(x + rng.normal(0, 0.05, size=x.size), -60, 60)
            if no_improve >= 4:
                if eps > eps_min:
                    eps = max(eps_min, eps * 0.5)
                else:
                    break
            elif eps > eps_min:
                eps = max(eps_min, eps * 0.6)
    
    return best, best_score

def _local_search_annealed(a0, end_time, seed, L):
    """Local search with simulated annealing for escaping local minima."""
    a = _project_simplex(a0)
    best = a.copy()
    best_score = _score_vector(best, L)
    rng = np.random.default_rng(seed)
    sigma = 0.15
    temperature = 0.01
    last_best = time.monotonic()
    start = time.monotonic()
    total_budget = max(1.0, end_time - start)
    
    while time.monotonic() < end_time - 5:
        elapsed = time.monotonic() - start
        frac = elapsed / total_budget
        # Cool down: allow worse moves early, reject them later
        temperature = 0.01 * (1.0 - frac) ** 2
        
        cand = best.copy()
        n = best.size
        # Perturb random blocks
        nblocks = 1 + int(rng.integers(0, 3))
        for _ in range(nblocks):
            w = int(np.exp(rng.uniform(0, np.log(max(2, n // 4)))))
            w = max(1, min(w, n))
            s = int(rng.integers(0, n - w + 1))
            cand[s:s + w] *= np.exp(rng.normal(0, sigma))
        
        cand = _project_simplex(cand)
        sc = _score_vector(cand, L)
        
        if sc < best_score - 1e-12:
            best = cand
            best_score = sc
            last_best = time.monotonic()
            sigma = min(0.4, sigma * 1.05)
        elif np.random.rand() < np.exp(-(sc - best_score) / max(temperature, 1e-10)):
            # Accept worse move (annealing)
            a = cand
        else:
            if time.monotonic() - last_best > 10:
                sigma = max(0.02, sigma * 0.85)
                if time.monotonic() - last_best > 40:
                    # Reset to best with small perturbation
                    best = _project_simplex(best * np.exp(rng.normal(0, 0.03, size=n)))
                    best_score = _score_vector(best, L)
                    last_best = time.monotonic()
    
    return best, best_score

def _q_polish(a0, end_time, L, q_start=2.0, q_end=4.0):
    """Polish using L-BFGS on Lq surrogates with increasing q."""
    n = a0.size
    if n <= 1:
        return a0, _score_vector(a0, L)
    
    x = _ratios_from_a(a0)
    if x.size == 0:
        return a0, _score_vector(a0, L)
    
    bounds = [(-60.0, 60.0)] * max(0, n - 1)
    best = _project_simplex(a0).copy()
    best_score = _score_vector(best, L)
    
    # Progressively increase q
    n_stages = 6
    stage_time = max(5.0, (end_time - time.monotonic()) / n_stages)
    
    for stage in range(n_stages):
        q = q_start + (q_end - q_start) * stage / max(1, n_stages - 1)
        t_chunk_end = time.monotonic() + stage_time
        
        def fun_q(x_curr):
            return _q_objective_grad(x_curr, q, L)
        
        while time.monotonic() < t_chunk_end and time.monotonic() < end_time - 5:
            try:
                res = minimize(
                    fun_q, x, method='L-BFGS-B', jac=True,
                    bounds=bounds,
                    options={'maxiter': 40, 'maxfun': 400, 'ftol': 1e-16, 'gtol': 1e-10}
                )
                x = np.clip(np.asarray(res.x, dtype=np.float64), -60, 60)
                x = np.where(np.isfinite(x), x, 0.0)
            except Exception:
                break
        
        a_new = _softmax_from_ratios(x)
        sc = _score_vector(a_new, L)
        if sc < best_score - 1e-13:
            best_score = sc
            best = a_new.copy()
            print(f"    Q-polish stage {stage} (q={q:.1f}): {best_score:.6f}")
    
    return best, best_score

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Returns best sequence found within time budget.
    Strategy:
      1. Generate diverse candidates
      2. LP-based directional refinement for top candidates
      3. Q-norm L-BFGS polishing
      4. Soft-max L-BFGS polishing
      5. Annealed local search
    """
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 10.0)
    
    n = 3809
    try:
        h = height_sequence_1
        if isinstance(h, (list, np.ndarray)):
            arr = np.asarray(h, dtype=np.float64)
            if arr.ndim == 1 and 100 <= arr.size <= 20000:
                n = int(arr.size)
    except Exception:
        pass
    if n < 64:
        n = 3809
    
    L = _pow2(2 * n - 1)
    
    if deadline <= t0 + 10:
        a = _power_sequence(n, 0.5, 1.0)
        return [float(x) for x in a]
    
    print(f"Starting search: n={n}, budget={deadline - t0:.1f}s")
    
    # Phase 1: Generate and score candidates
    cands = _generate_candidates(n)
    scored = []
    for i, a in enumerate(cands):
        sc = _score_vector(a, L)
        if np.isfinite(sc):
            scored.append((sc, i, a))
    scored.sort(key=lambda x: x[0])
    
    if not scored:
        a = np.full(n, 1.0 / float(n), dtype=np.float64)
        return [float(x) for x in a]
    
    global_best_a = scored[0][2].copy()
    global_best_score = scored[0][0]
    print(f"Best initial candidate: {global_best_score:.6f}")
    
    # Phase 2: LP-based refinement on top candidates
    n_top = min(5, len(scored))
    lp_budget = 0.35 * (deadline - t0)
    lp_end = t0 + lp_budget
    
    for idx in range(n_top):
        if time.monotonic() > lp_end:
            break
        a_start = scored[idx][2].copy()
        start_score = scored[idx][0]
        print(f"\nLP refinement candidate {idx}: initial score={start_score:.6f}")
        
        remaining = lp_end - time.monotonic()
        per_candidate = max(30.0, remaining / (n_top - idx))
        cand_end = min(time.monotonic() + per_candidate, lp_end)
        
        a_lp, sc_lp = _lp_refine(a_start, cand_end, max_iterations=100, lp_time_limit=45)
        
        if sc_lp < global_best_score:
            global_best_score = sc_lp
            global_best_a = a_lp.copy()
            print(f"  LP improved to: {sc_lp:.6f}")
        else:
            # Still try the Q-polish on the LP result
            a_to_polish = a_lp
    
    # Phase 3: Q-norm polishing on best
    q_budget = 0.30 * (deadline - t0)
    q_end = t0 + q_budget
    if time.monotonic() < q_end:
        print(f"\nQ-polish starting: score={global_best_score:.6f}")
        a_q, sc_q = _q_polish(global_best_a, q_end, L, q_start=2.0, q_end=5.0)
        if sc_q < global_best_score:
            global_best_score = sc_q
            global_best_a = a_q.copy()
            print(f"  Q-polish improved to: {sc_q:.6f}")
    
    # Phase 4: Soft-max L-BFGS polishing
    soft_budget = 0.20 * (deadline - t0)
    soft_end = t0 + soft_budget
    if time.monotonic() < soft_end:
        print(f"\nSoft-max polish starting: score={global_best_score:.6f}")
        a_soft, sc_soft = _lbfgs_refine(global_best_a, soft_end, L)
        if sc_soft < global_best_score:
            global_best_score = sc_soft
            global_best_a = a_soft.copy()
            print(f"  Soft-max improved to: {sc_soft:.6f}")
    
    # Phase 5: Annealed local search
    if time.monotonic() < deadline - 15:
        print(f"\nLocal search starting: score={global_best_score:.6f}")
        a_ls, sc_ls = _local_search_annealed(global_best_a, deadline - 15, int(seed) + 777, L)
        if sc_ls < global_best_score:
            global_best_score = sc_ls
            global_best_a = a_ls.copy()
            print(f"  Local search improved to: {sc_ls:.6f}")
    
    # Phase 6: One more round of Q-polish if time remains
    if time.monotonic() < deadline - 10:
        a_final, sc_final = _q_polish(global_best_a, deadline - 5, L, q_start=3.0, q_end=6.0)
        if sc_final < global_best_score:
            global_best_score = sc_final
            global_best_a = a_final.copy()
            print(f"  Final polish improved to: {sc_final:.6f}")
    
    global_best_a = _project_simplex(global_best_a)
    out = [float(x) for x in global_best_a]
    
    # Validate
    try:
        val = evaluate_sequence(out)
        if not np.isfinite(val):
            print("WARNING: final validation failed, using fallback")
            out = [float(x) for x in _power_sequence(n, 0.5, 1.0)]
        else:
            print(f"\nFinal validated score: {val:.6f}")
            if val < global_best_score:
                global_best_score = val
    except Exception as e:
        print(f"Validation exception: {e}")
    
    print(f"Best score found: {global_best_score:.6f}")
    return out
