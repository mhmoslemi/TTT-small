        return 1
    return 1 << (n - 1).bit_length()

    size = 2 * n - 1

    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)

    fb = np.fft.rfft(b, fft_size)

= sum_{m: 0<=m<n} p_{j+m} * a[m] * 1[0 <= j+m < 2n-1]

This is like a "weighted convolution" of p and a!

`dF/da = convolution of p and a` (with appropriate alignment)

This can be computed efficiently with FFT!

So the algorithm would be:
1. Compute c = a*a (FFT)
2. Compute p = softmax(tau * c)
3. Compute grad = convolution of p and a (FFT)
4. Update: a = a - eta * grad (or multiplicative)
5. Project: a = max(a, 0), normalize

This is O(n log n) per iteration, which is much faster than the O(n^2) LP solve!

"""
Optimization of step function sequences to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses a combination of:
1. Smoothed (log-sum-exp) subgradient descent with FFT-accelerated gradients
2. LP-based direction finding (SOTA approach) for occasional corrections
3. Diverse initialization strategies
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _fft_convolve(a):
    """Fast linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]


def _score(a):
    """Compute the objective value."""
    c = _fft_convolve(a)
    return 2.0 * len(a) * float(np.max(c))


def _smoothed_gradient(a, tau):
    """Compute the gradient of the log-sum-exp smoothed max of (a*a)[k]."""
    n = len(a)
    c = _fft_convolve(a)
    
    # Compute softmax weights
    c_max = c.max()
    exp_vals = np.exp(tau * (c - c_max))
    p = exp_vals / exp_vals.sum()
    
    # Compute gradient using cross-correlation via FFT
    # grad[j] = sum_m a[m] * p[j+m]
    fft_size = _next_pow2(3 * n - 2)  # need at least 3n-2 for no wrap-around
    fp = np.fft.rfft(p, fft_size)
    fa = np.fft.rfft(a, fft_size)
    grad_full = np.fft.irfft(fp * np.conj(fa), fft_size)
    grad = grad_full[:n]
    
    return grad, c


def _optimize_smoothed(a0, time_budget, rng, tau_init=10000.0):
    """Optimize using smoothed (log-sum-exp) subgradient descent."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, _score(a0)
    a = a / s
    
    c = _fft_convolve(a)
    best_score = 2.0 * n * float(np.max(c))
    best_a = a.copy()
    
    tau = tau_init
    eta = 0.05
    no_improve = 0
    deadline = time.monotonic() + time_budget
    it = 0
    last_print = time.monotonic()
    
    while time.monotonic() < deadline:
        grad, c = _smoothed_gradient(a, tau)
        
        # Check current score
        current_score = 2.0 * n * float(np.max(c))
        
        # Gradient step (we want to minimize max(a*a), so we subtract gradient)
        # The gradient points in the direction of increasing the smoothed max
        # So we subtract it
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            eta *= 0.5
            it += 1
            continue
        a_new = a_new / s_new
        
        score_new = _score(a_new)
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            no_improve = 0
            if time.monotonic() - last_print > 5.0:
                print(f"    [smooth] it={it}, score={best_score:.8f}, tau={tau:.0f}, eta={eta:.4f}")

            no_improve += 1

            tau *= 1.05  # increase tau to focus more on the max

def _build_conv_matrix(a):
    """Build the constraint matrix A for the LP.
    A[k, j] = a[j] if max(0,k-n+1) <= j <= min(n-1,k), else 0.
    """
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi+1] = a[lo:hi+1]
    return A


def _solve_lp_direction(a, M):
    """Solve LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M for all k."""
    n = len(a)
    c_obj = -np.ones(n)
    A = _build_conv_matrix(a)
    b_vec = np.full(2 * n - 1, M)
    
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_vec, 
                        bounds=[(0, None)] * n, method='highs')
        if result.status == 0:
            return np.maximum(result.x, 0)
    except Exception as e:
        print(f"    LP error: {e}")
    
    return None

    best_score = _score(a)

        M = float(np.max(c))
        
        b = _solve_lp_direction(a, M)

            it += 1

                a = a / a.sum()

"""
Optimizes step function sequences to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses LP-based direction finding (SOTA) with constraint generation,
smoothed (log-sum-exp) gradient descent, and diverse initialization.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]


def _fft_convolve_ab(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    fb = np.fft.rfft(b, fft_size)
    c = np.fft.irfft(fa * fb, fft_size)
    return c[:size]


def _score(a):
    c = _fft_convolve_self(a)
    return 2.0 * len(a) * float(np.max(c))


def _smoothed_grad(a, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*(a*a)[k]))."""
    n = len(a)
    c = _fft_convolve_self(a)
    c_shifted = c - c.max()
    exp_vals = np.exp(tau * c_shifted)
    p = exp_vals / exp_vals.sum()
    fft_size = _next_pow2(3 * n - 2)
    fp = np.fft.rfft(p, fft_size)
    fa = np.fft.rfft(a, fft_size)
    grad_full = np.fft.irfft(fp * np.conj(fa), fft_size)
    grad = grad_full[:n]
    return grad, c


def _optimize_smoothed(a0, time_budget, rng, tau_init=8000.0):

        grad, c = _smoothed_grad(a, tau)

        a_new = np.maximum(a_new, 0.0)

            eta *= 0.5
            it += 1

    """Basic subgradient descent (fast, for polishing)."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()

    best_score = _score(a)
    best_a = a.copy()
    eta = 0.1
    no_improve = 0
    deadline = time.monotonic() + time_budget
    it = 0

    while time.monotonic() < deadline:
        c = _fft_convolve_self(a)
        k_max = int(np.argmax(c))
        score = 2.0 * n * c[k_max]

        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            a = best_a * np.exp(rng.normal(0, 0.05, n))
            a = np.maximum(a, 1e-20)
            a = a / a.sum()
            no_improve = 0
            eta = 0.1
            it += 1
            continue

        # Subgradient of (a*a)[k_max]
        g = np.zeros(n)
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        if lo <= hi:
            j_range = np.arange(lo, hi + 1)
            indices = k_max - j_range
            g[lo:hi + 1] = 2.0 * a[indices]

        g_max = g.max()
        if g_max < 1e-15:
            break
        g = g / g_max
        a = a * np.exp(-eta * g)
        a = np.maximum(a, 1e-20)
        a = a / a.sum()

        if no_improve > 300:
            eta *= 0.96
        elif no_improve < 30:
            eta = min(eta * 1.005, 0.3)

        it += 1

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2300, **kwargs):
    """
    Minimizes 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
    Uses LP-based (constraint generation) + smoothed gradient + subgradient descent.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 25.0

    best_overall = None
    best_overall_score = np.inf

    # Load previous best
    prev_a = None
    try:
        hs = np.array(height_sequence_1, dtype=np.float64)
        hs = np.clip(hs, 0, None)
        if hs.size > 100 and hs.sum() > 0.01:
            prev_a = hs / hs.sum()
            sc = _score(prev_a)
            print(f"Previous best: n={len(prev_a)}, score={sc:.8f}")
            if sc < best_overall_score:
                best_overall_score = sc
                best_overall = prev_a.copy()
    except (NameError, Exception) as e:
        pass

    # Generate candidates
    candidates = []

    if prev_a is not None:
        candidates.append(("prev", prev_a.copy()))
        # Slightly perturbed versions of previous best
        for pert in [0.01, 0.03]:
            p = prev_a * np.exp(rng.normal(0, pert, len(prev_a)))
            p = np.maximum(p, 1e-10)
            p = p / p.sum()
            candidates.append((f"prev_pert{pert}", p))

    n_search = 2000
    x = np.arange(n_search, dtype=np.float64)

    # Sinusoidal variants
    for alpha in [0.3, 0.4, 0.5, 0.6]:
        for phase in [0.0, np.pi / 4, np.pi / 2, 3 * np.pi / 4]:
            a = 1.0 + alpha * np.sin(2 * np.pi * x / n_search + phase)
            a = np.maximum(a, 0.01)
            candidates.append((f"sin_a{alpha}_p{phase:.2f}", a))

    # Multi-frequency
    a = 1.0 + 0.25 * np.sin(2 * np.pi * x / n_search) + 0.2 * np.sin(4 * np.pi * x / n_search + 0.7)
    a = np.maximum(a, 0.01)
    candidates.append(("multi2", a))

    a = 1.0 + 0.2 * np.sin(2 * np.pi * x / n_search) + 0.15 * np.sin(3 * np.pi * x / n_search + 0.4) + 0.1 * np.sin(5 * np.pi * x / n_search + 1.1)
    a = np.maximum(a, 0.01)
    candidates.append(("multi3", a))

    # Decay
    for rate in [0.6, 1.0, 1.5]:
        a = np.exp(-rate * x / n_search) + 0.1
        candidates.append((f"decay_{rate}", a))

    # Cosine window (peaking at center)
    a = 0.5 * (1 + np.cos(2 * np.pi * (x / n_search - 0.5))) + 0.1
    candidates.append(("cos_center", a))

    # Larger n candidates
    for n_large in [3000, 4000]:
        x_l = np.arange(n_large, dtype=np.float64)
        a = 1.0 + 0.4 * np.sin(2 * np.pi * x_l / n_large + np.pi / 3)
        a = np.maximum(a, 0.01)
        candidates.append((f"sin_n{n_large}", a))

    # Score all candidates
    scored = []
    for name, a0 in candidates:
        a0 = np.maximum(a0, 0)
        s = a0.sum()
        if s < 1e-15:
            continue
        a0 = a0 / s
        sc = _score(a0)
        scored.append((sc, name, a0))

    scored.sort(key=lambda x: x[0])
    print(f"\n{len(scored)} candidates. Top 12:")
    for i in range(min(12, len(scored))):
        print(f"  {i+1}. {scored[i][1]}: {scored[i][0]:.8f} (n={len(scored[i][2])})")

    # Optimize top candidates
    n_top = min(6, len(scored))
    remaining = deadline - time.monotonic()

    # Time allocation (weighted by rank)
    weights = [1.0 / (i + 1) for i in range(n_top)]
    total_w = sum(weights)
    times = [remaining * w / total_w for w in weights]

    for i in range(n_top):
        if time.monotonic() > deadline:
            break
        init_score, name, a0 = scored[i]
        t_alloc = min(times[i], max(15.0, deadline - time.monotonic()))
        if t_alloc < 15:
            break

        print(f"\n--- Optimizing '{name}' (n={len(a0)}, init={init_score:.8f}, time={t_alloc:.0f}s) ---")

        # Phase 1: Smoothed gradient (fast, many iterations)
        t1 = t_alloc * 0.4
        a1, s1 = _optimize_smoothed(a0, t1, rng, tau_init=8000.0)
        e1 = evaluate_sequence(a1.tolist())
        print(f"  Smoothed: {e1:.8f}")
        if np.isfinite(e1) and e1 < best_overall_score:
            best_overall_score = e1
            best_overall = a1.copy()
            print(f"  *** NEW BEST: {e1:.8f} ***")

        # Phase 2: LP with constraint generation (slower, more principled)
        t2 = t_alloc * 0.4
        if time.monotonic() < deadline:
            a2, s2 = _optimize_lp_cg(a1, t2, rng, max_rows=60)
            e2 = evaluate_sequence(a2.tolist())
            print(f"  LP-CG: {e2:.8f}")
            if np.isfinite(e2) and e2 < best_overall_score:
                best_overall_score = e2
                best_overall = a2.copy()
                print(f"  *** NEW BEST: {e2:.8f} ***")

        # Phase 3: Subgradient polish
        t3 = t_alloc * 0.2
        if time.monotonic() < deadline:
            start3 = a2 if np.isfinite(s2) and s2 <= s1 else a1
            a3, s3 = _optimize_subgrad(start3, t3, rng)
            e3 = evaluate_sequence(a3.tolist())
            print(f"  Subgrad: {e3:.8f}")
            if np.isfinite(e3) and e3 < best_overall_score:
                best_overall_score = e3
                best_overall = a3.copy()
                print(f"  *** NEW BEST: {e3:.8f} ***")

    # Final intensive polishing on the best solution
    remaining = deadline - time.monotonic()
    if best_overall is not None and remaining > 30:
        print(f"\n=== Final polishing ({remaining:.0f}s) ===")

        # LP polish
        t_lp = min(remaining * 0.35, 300)
        if t_lp > 20:
            al, sl = _optimize_lp_cg(best_overall, t_lp, rng, max_rows=80)
            el = evaluate_sequence(al.tolist())
            print(f"  LP final: {el:.8f}")
            if np.isfinite(el) and el < best_overall_score:
                best_overall_score = el
                best_overall = al.copy()
                print(f"  *** NEW BEST: {el:.8f} ***")

        # Smoothed polish with high tau
        rem2 = deadline - time.monotonic()
        if rem2 > 20:
            t_sm = min(rem2 * 0.35, 200)
            as_, ss = _optimize_smoothed(best_overall, t_sm, rng, tau_init=30000.0)
            es = evaluate_sequence(as_.tolist())
            print(f"  Smooth final: {es:.8f}")
            if np.isfinite(es) and es < best_overall_score:
                best_overall_score = es
                best_overall = as_.copy()
                print(f"  *** NEW BEST: {es:.8f} ***")

        # Subgradient polish
        rem3 = deadline - time.monotonic()
        if rem3 > 15:
            a_sub, s_sub = _optimize_subgrad(best_overall, rem3, rng)
            e_sub = evaluate_sequence(a_sub.tolist())
            print(f"  Subgrad final: {e_sub:.8f}")
            if np.isfinite(e_sub) and e_sub < best_overall_score:
                best_overall_score = e_sub
                best_overall = a_sub.copy()
                print(f"  *** NEW BEST: {e_sub:.8f} ***")

    # Verify and return
    if best_overall is not None:
        final_score = evaluate_sequence(best_overall.tolist())
        elapsed = time.monotonic() - total_start
        print(f"\nFinal: score={final_score:.8f}, n={len(best_overall)}, time={elapsed:.1f}s")
        return [float(x) for x in best_overall]
    else:
        print("Fallback: uniform")
        return [1.0] * 1000
