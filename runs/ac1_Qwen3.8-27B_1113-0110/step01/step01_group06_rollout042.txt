2. Spectral HIO (Hybrid Input-Output) in the Fourier domain

5. Coordinate descent for fine-tuning
6. Multi-size and multi-start search strategy

import numpy as np
import time


def _pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _fft_conv(a, b):
    n = int(a.size)
    m = int(b.size)
    L = n + m - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N).real[:L]
    return np.maximum(c, 0.0)


def _fft_square(a):
    n = int(a.size)
    L = 2 * n - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, N)
    b = np.fft.irfft(fa * fa, N).real[:L]
    return np.maximum(b, 0.0)


def _score_internal(a, b=None):
    n = int(a.size)
    if b is None:
        b = _fft_square(a)
    s = float(a.sum())
    if s <= 1e-300:
        return float('inf'), b, 0.0
    maxb = float(b.max())
    sc = float(2.0 * n * maxb / (s * s))
    return sc, b, maxb


def _project_anti(a):
    """Project onto anti-palindromic constraint a[j]+a[n-1-j]=c, a[j] in [0,c]."""
    n = int(a.size)
    m = n // 2
    c = 2.0 / n
    first = a[:m]
    second_rev = a[m:][::-1]
    new_first = c / 2.0 + (first - second_rev) / 2.0
    new_first = np.clip(new_first, 0.0, c)
    out = np.empty(n, dtype=np.float64)
    out[:m] = new_first
    out[m:] = (c - new_first)[::-1]
    return out


def _anti_from_x(x):
    x = np.asarray(x, dtype=np.float64).ravel()
    m = int(x.size)
    n = 2 * m
    c = 2.0 / n
    a = np.empty(n, dtype=np.float64)
    a[:m] = c * x
    a[m:] = c * (1.0 - x[::-1])
    return a


def _x_from_anti(a):
    n = int(a.size)
    m = n // 2
    c = 2.0 / n
    x = a[:m] / c
    return np.clip(x, 0.0, 1.0)


def _greedy_binary(n, rng):
    """Greedy construction of binary anti-palindromic sequence."""
    m = n // 2
    counts = np.zeros(2 * n - 1, dtype=np.int64)
    selected = np.empty(m, dtype=np.int64)
    order = np.arange(m, dtype=np.int64)
    rng.shuffle(order)
    t = 0
    for idx_val in order:
        idx = int(idx_val)
        l = idx
        r = n - 1 - idx
        if t == 0:
            y = l
        else:
            s = selected[:t]
            affected_l = int(np.max(counts[s + l]))
            self_l = int(counts[2 * l])
            metric_l = max(affected_l + 2, self_l + 1)
            affected_r = int(np.max(counts[s + r]))
            self_r = int(counts[2 * r])
            metric_r = max(affected_r + 2, self_r + 1)
            y = l if metric_l <= metric_r else r
        if t > 0:
            counts[selected[:t] + y] += 2
        counts[2 * y] += 1
        selected[t] = y
        t += 1
    x = np.full(m, 1e-8, dtype=np.float64)
    x[selected] = 1.0 - 1e-8
    return _anti_from_x(x)


def _random_anti(n, rng, amp=3.0):
    m = n // 2
    x = rng.uniform(0.0, 1.0, size=m)
    return _anti_from_x(x)


def _smooth_anti(n, rng, n_freqs=8):
    m = n // 2
    x = np.arange(m, dtype=np.float64)
    v = np.zeros(m, dtype=np.float64)
    for f in range(1, n_freqs + 1):
        amp = rng.uniform(0.5, 2.0) / f
        phase = rng.uniform(0.0, 2.0 * np.pi)
        v += amp * np.sin(2.0 * np.pi * f * x / m + phase)
    v = (v - v.min()) / (v.max() - v.min() + 1e-12)
    return _anti_from_x(v)


def _m_sequence_init(n, rng):
    """Initialize using m-sequence-like structure."""
    m = n // 2
    # Generate a pseudorandom binary sequence with good autocorrelation
    # Use a simple LFSR for p=12 (length 4095) to seed the pattern
    p = 12
    L = (1 << p) - 1
    # Primitive polynomial for p=12: x^12 + x^6 + x^4 + x + 1
    # Taps at positions (0-indexed from LSB): 0, 1, 4, 6
    state = (1 << p) - 1
    seq = []
    for _ in range(L):
        seq.append(state & 1)
        # Feedback: XOR bits at tap positions
        feedback = 0
        for tap in [0, 1, 4, 6]:
            feedback ^= (state >> tap) & 1
        state = (state >> 1) | (feedback << (p - 1))
    seq = np.array(seq, dtype=np.float64)
    # Tile or crop to length m
    if m <= L:
        start = rng.integers(0, L)
        x = np.concatenate([seq[start:], seq[:start + (m - (L - start))]])[:m]
    else:
        reps = (m + L - 1) // L
        x = np.tile(seq, reps)[:m]
    # Convert to anti-palindromic with some randomness
    x = x * 0.7 + rng.uniform(0.0, 0.3, size=m)
    x = np.clip(x, 0.0, 1.0)
    return _anti_from_x(x)


def _hio_step(a, N, alpha, rng, soft=0.9):
    """One HIO iteration: modify spectrum, project back."""
    A = np.fft.rfft(a, N)
    A_mod = A.copy()
    # Shrink non-DC components
    A_mod[1:] *= (1.0 - alpha)
    a_new = np.fft.irfft(A_mod, N).real[:a.size]
    a_new = _project_anti(a_new)
    # Mix with old for stability
    a_new = soft * a_new + (1.0 - soft) * a
    a_new = _project_anti(a_new)
    return a_new


def _gradient_step(x, a, b, focus, eta, rng, n):
    """One gradient descent step on the anti-palindromic parameters."""
    m = n // 2
    maxb = float(b.max())
    if maxb <= 1e-300:
        return x
    z = (maxb - b) / maxb
    with np.errstate(over='ignore', under='ignore'):
        p = np.exp(-z * focus)
    s = float(p.sum())
    if s <= 1e-12:
        p = np.zeros_like(b)
        p[int(np.argmax(b))] = 1.0
    else:
        p = 0.98 * (p / s) + 0.02 / b.size
    # Gradient via FFT
    L = int(b.size)
    N = _pow2(L + m)
    fw = np.fft.rfft(p, N)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fw * np.conj(fa), N)[:n]
    ga = 2.0 * c
    gx = ga[:m] - ga[m:][::-1]
    maxg = float(np.max(np.abs(gx)))
    if maxg < 1e-18:
        return x
    gx = gx / maxg
    x_new = x - eta * gx
    x_new = np.clip(x_new, 1e-8, 1.0 - 1e-8)
    return x_new

def _coordinate_descent(a, time_limit, rng):
    """Coordinate descent on the anti-palindromic parameters."""
    a = _project_anti(a.copy())
    n = int(a.size)
    m = n // 2
    c = 2.0 / n
    x = _x_from_anti(a)

    b = _fft_square(a)

    best_x = x.copy()

    t0 = time.monotonic()
    it = 0
    step_size = 0.1

    while time.monotonic() - t0 < time_limit:
        # Pick coordinates to update (batch)
        batch_size = min(m, 100)
        idx = rng.integers(0, m, size=batch_size)
        improved = False

        for i in idx:
            if time.monotonic() - t0 > time_limit:
                    break
            # Try increasing x[i]
            x_trial = x.copy()
            x_trial[i] = min(1.0 - 1e-8, x[i] + step_size)
            a_trial = _anti_from_x(x_trial)
            b_trial = _fft_square(a_trial)
            sc_trial, b_trial, _ = _score_internal(a_trial, b_trial)

            if sc_trial < best_score - 1e-10:
                x = x_trial
                a = a_trial
                b = b_trial
                score = sc_trial
                best_score = sc_trial
                best_x = x.copy()
                improved = True

            # Try decreasing x[i]
            x_trial = x.copy()
            x_trial[i] = max(1e-8, x[i] - step_size)
            a_trial = _anti_from_x(x_trial)
            b_trial = _fft_square(a_trial)
            sc_trial, b_trial, _ = _score_internal(a_trial, b_trial)

            if sc_trial < best_score - 1e-10:
                x = x_trial
                a = a_trial
                b = b_trial
                score = sc_trial
                best_score = sc_trial
                best_x = x.copy()
                improved = True

        if not improved:
            step_size *= 0.8
            if step_size < 1e-4:
                break
        else:
            step_size = min(step_size * 1.05, 0.3)

        it += 1

    return _anti_from_x(best_x), best_score


def _lp_refine(a, time_limit, rng, top_frac=0.05):
    """LP-based refinement step."""
    try:
        from scipy.optimize import linprog
    except ImportError:
        return a, float('inf')

    a = _project_anti(a.copy())
    n = int(a.size)
    b = _fft_square(a)
    score, b, maxb = _score_internal(a, b)
    best_score = score
    best_a = a.copy()

    L = 2 * n - 1
    n_constraints = max(10, int(L * top_frac))
    # Get top n_constraints indices
    top_idx = np.argsort(b)[-n_constraints:]
    top_idx = np.sort(top_idx)

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit - 5.0:
        M = float(b.max())
        if M < 1e-10:
            break

        # Form constraint matrix C: C[k,j] = a[k-j] if valid
        # (a*g)(k) = sum_j a[k-j] * g[j] for valid k-j in [0,n-1]
        C = np.zeros((n_constraints, n), dtype=np.float64)
        for ii, k in enumerate(top_idx):
            j_start = max(0, k - n + 1)
            j_end = min(int(k), n - 1)
            if j_start <= j_end:
                C[ii, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]

        # LP: max sum(g) s.t. C@g <= M, g >= 0
        c_obj = -np.ones(n)  # minimize -sum(g)
        res = linprog(c_obj, A_ub=C, b_ub=np.full(n_constraints, M),
                      bounds=[(0, 100.0)] * n, method='highs',
                      options={'time_limit': 5.0})

        if res.status != 0 or res.x is None:
            break

        g = np.maximum(res.x, 0.0)
        sg = float(g.sum())
        if sg < 1e-10:
            break

        # Rescale and blend
        sa = float(a.sum())
        g = g * (sa / sg)
        blend = 0.1 + 0.1 * (it % 3)  # Vary blend: 0.1, 0.2, 0.3
        a_new = (1.0 - blend) * a + blend * g
        a_new = _project_anti(a_new)
        s_new = float(a_new.sum())
        if s_new <= 0.01:
            break
        a_new /= s_new

        b_new = _fft_square(a_new)
        sc_new, b_new, _ = _score_internal(a_new, b_new)

        if sc_new < best_score - 1e-10:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            b = b_new
            score = sc_new
        else:
            # Try adding random perturbation
            a_pert = a * np.exp(rng.normal(0.0, 0.3, size=n))
            a_pert = _project_anti(a_pert)
            s_p = float(a_pert.sum())
            if s_p > 0.01:
                a_pert /= s_p
            a = a_pert
            b = _fft_square(a)
            score, b, _ = _score_internal(a, b)

        it += 1
        if it > 50:
            break

    return best_a, best_score
