```python
    best_x = old
    best_max = c.max()  # current max (at x = old, S = 1)
    
    # Generate candidate x values
    candidates = set()
    candidates.add(0.0)
    candidates.add(old)
    
    # Candidate from minimizing the max constraint (k* = argmax c)
    k_star = np.argmax(c)
    km = k_star - m
    if 0 <= km < n and k_star != 2*m:
        a_val = a[km]
        if a_val > 1e-15:
            # d_opt = 1 - c[k_star]/a_val
            d_opt = 1.0 - c[k_star] / a_val
            candidates.add(max(0.0, old + d_opt))
        
        # Try a range around d_opt
        for frac in [0.5, 0.75, 1.25, 1.5, 2.0]:
            candidates.add(max(0.0, old + d_opt * frac))
    elif k_star == 2*m:
        if old < 1 - 1e-15:
            d_opt = (c[k_star] - old) / (1.0 - old)
            candidates.add(max(0.0, old + d_opt))
    
    # Candidates from equalizing top-2
    top_idx = np.argsort(c)[-4:]
    for k2 in top_idx:
        km2 = k2 - m
        if k_star != 2*m and 0 <= km < n and 0 <= km2 < n and k2 != 2*m:
            a1, a2 = a[km], a[km2]
            if abs(a1 - a2) > 1e-15:
                d_eq = (c[k2] - c[k_star]) / (2.0 * (a1 - a2))
                candidates.add(max(0.0, old + d_eq))
    
    # Small perturbations
    for eps in [0.01, 0.1, 0.5, 1.0]:
        candidates.add(max(0.0, old + eps))
        candidates.add(max(0.0, old - eps * old))
    
    # Evaluate
    for x in candidates:
        x = float(x)
        if x < 0 or not np.isfinite(x):
            continue
        S = 1.0 + x - old
        if S <= 1e-15:
            continue
        # Compute new max c without full update
        # c_new[k] = (c[k] + delta_k) / S^2
        # max of c_new = max of (c[k] + delta_k) / S^2 = max(c[k] + delta_k) / S^2
        # (since S^2 is constant)
        # So we just need max(c[k] + delta_k)
        
        c_modified = c.copy()
        k_min = max(0, m)
        k_max = min(2*n - 2, m + n - 1)
        ks = np.arange(k_min, k_max + 1)
        c_modified[ks] += 2.0 * (x - old) * a[ks - m]
        if 0 <= 2*m <= 2*n - 2:
            c_modified[2*m] -= 2.0 * (x - old) * old
            c_modified[2*m] += x*x - old*old
        
        new_max = c_modified.max() / (S * S)
        if new_max < best_max:
            best_max = new_max
            best_x = x
    
    return best_x
```

```python
def _eval_candidate(c, a, m, x, old, n):
    """Evaluate max(c_new) without modifying c."""
    S = 1.0 + x - old
    if S <= 1e-15:
        return float('inf')
    inv_S2 = 1.0 / (S * S)
    d = x - old
    
    # The modified c is: c + 2*d*a[k-m] for k in valid range (excluding 2m adjustment)
    # We need max over all k of (c[k] + 2*d*a[k-m]) / S^2
    # For k where k-m is out of range: c[k] / S^2
    # For k in valid range (k != 2m): (c[k] + 2*d*a[k-m]) / S^2
    # For k = 2m: (c[k] + x^2 - old^2) / S^2
    
    # The max is the max of:
    # 1. max_{k: k-m out of range, k != 2m} c[k]
    # 2. max_{k: 0<=k-m<n, k != 2m} (c[k] + 2*d*a[k-m])
    # 3. c[2m] + x^2 - old^2 (if 0 <= 2m <= 2n-2)
    
    k_min = max(0, m)
    k_max = min(2*n - 2, m + n - 1)
    
    # Zone 1: k < k_min or k > k_max (and k != 2m)
    # These are just c[k] scaled by 1/S^2
    zone1_max = 0.0
    if k_min > 0:
        zone1_max = max(zone1_max, c[:k_min].max())
    if k_max < 2*n - 2:
        zone1_max = max(zone1_max, c[k_max+1:].max())
    # Remove k=2m from zone1 if it's in zone1
    if 0 <= 2*m <= 2*n-2:
        if 2*m < k_min or 2*m > k_max:
            # 2m is in zone1, so it's already included in zone1_max, that's fine
            # (we'll handle 2m separately, but including it in zone1 is a slight overestimate)
            pass
    
    # Zone 2: k in [k_min, k_max], k != 2m
    zone2_max = 0.0
    if k_max >= k_min:
        ks = np.arange(k_min, k_max + 1)
        vals = c[ks] + 2.0 * d * a[ks - m]
        if 0 <= 2*m <= 2*n-2:
            mask = ks != 2*m
            if mask.any():
                zone2_max = vals[mask].max()
        else:
            zone2_max = vals.max()
    
    # Zone 3: k = 2m
    zone3_val = 0.0
    if 0 <= 2*m <= 2*n-2:
        zone3_val = c[2*m] + x*x - old*old
    
    total_max = max(zone1_max, zone2_max, zone3_val) / (S * S)
    return total_max
```

This avoids the copy and is O(n) per evaluation. Good.

```python
def _eval_all_candidates(c, a, m, candidates, old, n):
    """Evaluate max(c_new) for all candidate x values."""
    d = candidates - old  # shape (num_candidates,)
    S = 1.0 + d
    valid = S > 1e-15
    inv_S2 = np.where(valid, 1.0/(S*S), 0)
    
    k_min = max(0, m)
    k_max = min(2*n - 2, m + n - 1)
    
    # Zone 1: max c[k] for k outside [k_min, k_max]
    zone1 = np.full(len(candidates), 0.0)
    if k_min > 0:
        zone1 = np.maximum(zone1, c[:k_min].max())
    if k_max < 2*n - 2:
        zone1 = np.maximum(zone1, c[k_max+1:].max())
    
    # Zone 2: max over k in [k_min, k_max], k != 2m of (c[k] + 2*d*a[k-m])
    ks = np.arange(k_min, k_max + 1)
    a_vals = a[ks - m]  # shape (n,)
    c_vals = c[ks]  # shape (n,)
    # vals[j, i] = c_vals[j] + 2*d[i]*a_vals[j], shape (n, num_candidates)
    vals = c_vals[:, None] + 2.0 * d[None, :] * a_vals[:, None]
    # Remove k=2m
    if 0 <= 2*m <= 2*n-2:
        idx_2m = 2*m - k_min
        if 0 <= idx_2m < len(ks):
            vals = np.delete(vals, idx_2m, axis=0)
    zone2 = vals.max(axis=0)  # shape (num_candidates,)
    
    # Zone 3: c[2m] + x^2 - old^2
    zone3 = np.full(len(candidates), 0.0)
    if 0 <= 2*m <= 2*n-2:
        zone3 = c[2*m] + candidates*candidates - old*old
    
    total_max = np.maximum.reduce([zone1, zone2, zone3]) * inv_S2
    total_max[~valid] = np.inf
    return total_max
```
