LP-based optimization for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses the SOTA linear programming direction-finding approach with HiGHS solver,
combined with mirror descent and local search polishing.
"""

# Helper functions
# ============================================================

def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()

def normalize_positive(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, float(floor), None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s

def convolve_fft(a, N=None):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if N is None:
        N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]

def internal_score(a):
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.inf
    a_n = a / s
    c = convolve_fft(a_n)
    return float(2.0 * n * float(np.max(c)))

def build_convolve_matrix(n):
    """Build the sparse matrix A such that (A @ b) = convolve(f, b)."""
    # A[k, j] = 1 if f[k-j] would multiply b[j] in the convolution
    # Actually: (f*b)[k] = sum_j f[k-j] * b[j] for valid k-j
    # So A[k,j] should be f[k-j], but f varies! 
    # I need to rebuild the matrix values each iteration.
    # Just return the (row, col) structure.
    rows = []
    cols = []
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        rows.extend(range(k_start, k_end + 1))
        cols.extend([j] * (k_end - k_start + 1))
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)

# Actually, for the LP, I need A[k,j] = f[k-j] where 0 <= k-j <= n-1.

# Let me precompute the offset: for entry (k,j), the offset is k-j.
# offset = k - j, and A[k,j] = f[offset]

    """Build LP constraint matrix given f and precomputed row/col structure."""
    k, j = row_cols
    offsets = k - j
    vals = f[offsets]
    return sparse.csr_matrix((vals, (k, j)), shape=(2*n-1, n))

def compute_weights(c, cmax, tau, floor=0.02):
    z = (c - float(cmax)) / float(tau)
    z = np.clip(z, -50.0, 0.0)
    ez = np.exp(z)
    s = float(ez.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(c.size, 1.0 / float(c.size), dtype=np.float64)
    w = ez / s
    if floor > 0.0:
        w = (1.0 - floor) * w + (floor / float(c.size))
    return w

def grad_from_weights(w, fa, N, n):

    fw = np.fft.rfft(w_rev, N)

Now, the key question: is the LP feasible? We need `b >= 0` and `A b <= M * 1`. 

If `f` is the uniform distribution (all `1/n`), then `A` has all entries `1/n` in the banded region. The constraint `A b <= M` where `M = 1/n` (the max of `f*f` for uniform) becomes:

`sum_j b[j] <= 1`

def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()

def normalize_positive(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, float(floor), None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s

def convolve_fft(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if N is None:
        N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]

def convolve_fft2(a, b, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    n = a.size
    if N is None:
        N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:2 * n - 1]

def build_lp_rows_cols(n):
    """Precompute row/col/offset for the convolution matrix A[k,j]=f[k-j]."""
    rows = []
    cols = []
    offsets = []
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        for k in range(k_start, k_end + 1):
            rows.append(k)
            cols.append(j)
            offsets.append(k - j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32), np.array(offsets, dtype=np.int32)

def solve_lp(f, M, rows, cols, offsets, use_dense):
    """Solve: max sum(b) s.t. A*b <= M*1, b >= 0. Returns b or None."""
    n = len(f)
    vals = f[offsets]
    
    if use_dense:
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        A[rows, cols] = vals
        res = linprog(-np.ones(n), A_ub=A, b_ub=M * np.ones(2 * n - 1),
                      bounds=[(0, None)] * n, method='highs')
    else:
        A = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        res = linprog(-np.ones(n), A_ub=A, b_ub=M * np.ones(2 * n - 1),
                      bounds=[(0, None)] * n, method='highs')
    
    if res is not None and res.success and res.x is not None:
        return np.clip(res.x, 0, None)
    return None

def compute_weights(c, cmax, tau, floor=0.03):
    c = np.asarray(c, dtype=np.float64)
    L = c.size
    z = (c - float(cmax)) / float(tau)
    z = np.clip(z, -50.0, 0.0)
    ez = np.exp(z)
    s = float(ez.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(L, 1.0 / float(L), dtype=np.float64)
    w = ez / s
    if floor > 0.0:
        w = (1.0 - floor) * w + (floor / float(L))
    return w

def grad_from_weights(w, fa, N, n):
    w_rev = w[::-1]
    fw = np.fft.rfft(w_rev, N)
    conv = np.fft.irfft(fa * fw, N)
    return 2.0 * conv[n - 1:2 * n - 1][::-1]

def resample_seq(a, n):
    a = normalize_positive(a, floor=0.0)
    old_n = a.size
    n = max(2, int(n))
    if old_n == n:
        return a.astype(np.float64, copy=True)
    if old_n == 1:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    old_x = np.linspace(0.0, 1.0, old_n, dtype=np.float64)
    new_x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    b = np.interp(new_x, old_x, a)
    return normalize_positive(b, floor=1e-12)

        loga[start:start+width] += d
        a_try = np.exp(np.clip(loga - float(np.max(loga)), -60.0, 0.0))
        a_try = np.maximum(a_try, 1e-300)
        a_try = a_try / float(a_try.sum())
        a_try = normalize_positive(a_try, floor=1e-15)
        
        sc = internal_score(a_try)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_try.copy()
            a = a_try
            stagnant = 0
            sigma = min(0.3, sigma * 1.03)
        else:
            stagnant += 1
            sigma = max(0.02, sigma * 0.95)
            if stagnant > 80:
                break
    
    return best_a, best_score

# Main propose function
# ============================================================

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    start = time.monotonic()
    total = max(120.0, float(budget_s) - 30.0)
    deadline = start + total
    
    best_f = None
    best_score = np.inf
    
    # Load given sequence
    try:
        given = globals().get("height_sequence_1", None)
        if given is not None:
            a = np.asarray(given, dtype=np.float64).ravel()
            if a.size > 0:
                a = normalize_positive(a, floor=0.0)
                if np.all(np.isfinite(a)):
                    sc = internal_score(a)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_f = a.copy()
                        print(f"Given sequence: n={a.size}, score={sc:.10f}", flush=True)
    except Exception as e:
        print(f"Warning: could not load given sequence: {e}", flush=True)
    
    if best_f is None:
        best_f = np.ones(2048, dtype=np.float64) / 2048.0
        best_score = internal_score(best_f)
        print(f"Default uniform n=2048: score={best_score:.8f}", flush=True)
    
    # ================================================================
    # Phase 1: LP optimization on the given/best sequence length
    # ================================================================
    n1 = best_f.size
    if time.monotonic() < deadline - 500:
        print(f"\n=== Phase 1: LP optimization n={n1} ===", flush=True)
        d1 = min(deadline - 500, start + 700)
        f1, sc1 = lp_optimize(best_f, d1, verbose=True)
        if np.isfinite(sc1) and sc1 < best_score:
            best_score = sc1
            best_f = f1
            print(f"Phase 1 result: score={sc1:.10f}", flush=True)
        else:
            print(f"Phase 1 best: {sc1:.10f} (current best={best_score:.10f})", flush=True)
    
    # ================================================================
    # Phase 2: LP on other sequence lengths
    # ================================================================
    for n2 in (1024, 2048, 4096, 8192):
        if time.monotonic() > deadline - 350:
            break
        if n2 == n1:
            continue
        allow = min(350.0, deadline - time.monotonic() - 350)
        if allow < 80:
            break
        print(f"\n=== Phase 2: LP optimization n={n2} (budget={allow:.0f}s) ===", flush=True)
        a0 = resample_seq(best_f, n2)
        f2, sc2 = lp_optimize(a0, time.monotonic() + allow, verbose=True)
        if np.isfinite(sc2) and sc2 < best_score:
            best_score = sc2
            best_f = f2
            print(f"Phase 2 n={n2} result: score={sc2:.10f}", flush=True)
        else:
            print(f"Phase 2 n={n2} best: {sc2:.10f}", flush=True)
    
    # ================================================================
    # Phase 3: More LP refinement on the best length found
    # ================================================================
    if time.monotonic() < deadline - 200:
        best_n = best_f.size
        print(f"\n=== Phase 3: Continued LP on n={best_n} ===", flush=True)
        allow = min(300.0, deadline - time.monotonic() - 200)
        if allow > 60:
            f3, sc3 = lp_optimize(best_f, time.monotonic() + allow, verbose=True)
            if np.isfinite(sc3) and sc3 < best_score:
                best_score = sc3
                best_f = f3
                print(f"Phase 3 result: score={sc3:.10f}", flush=True)
    
    # ================================================================
    # Phase 4: Mirror descent polishing
    # ================================================================
    if time.monotonic() < deadline - 80:
        print(f"\n=== Phase 4: Mirror descent polish ===", flush=True)
        allow = min(200.0, deadline - time.monotonic() - 80)
        if allow > 30:
            f4, sc4 = mirror_descent(best_f, time.monotonic() + allow, verbose=True)
            if np.isfinite(sc4) and sc4 < best_score:
                best_score = sc4
                best_f = f4
                print(f"Phase 4 result: score={sc4:.10f}", flush=True)
    
    # ================================================================
    # Phase 5: Local polish
    # ================================================================
    if time.monotonic() < deadline - 30:
        print(f"\n=== Phase 5: Local polish ===", flush=True)
        allow = min(100.0, deadline - time.monotonic() - 30)
        if allow > 20:
            f5, sc5 = local_polish(best_f, time.monotonic() + allow, seed=999)
            if np.isfinite(sc5) and sc5 < best_score:
                best_score = sc5
                best_f = f5
                print(f"Phase 5 result: score={sc5:.10f}", flush=True)
    
    # ================================================================
    # Final output
    # ================================================================
    best_f = normalize_positive(best_f, floor=0.0)
    out_list = [float(x) for x in best_f]
    
    try:
        final_eval = float(evaluate_sequence(out_list))
    except Exception:
        final_eval = float("inf")
    
    if not np.isfinite(final_eval):
        print("WARNING: evaluation failed, using uniform fallback", flush=True)
        return [1.0] * 2048
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: score={final_eval:.12f}, length={len(out_list)}", flush=True)
    print(f"{'='*60}", flush=True)
    return out_list
