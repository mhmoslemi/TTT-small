    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()

def _fft_square(a):
    n = int(a.size)
    L = 2 * n - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, n=N)
    b = np.fft.irfft(fa * fa, n=N).real[:L]
    return np.maximum(b, 0.0)

def _fft_grad(a, p):
    """Compute 2 * ifft(fft(p) * conj(fft(a))).real[:n]"""
    n = int(a.size)
    L = int(p.size)
    N = _pow2(L + n - 1)
    fp = np.fft.rfft(p, n=N)
    fa = np.fft.rfft(a, n=N)
    c = np.fft.irfft(fp * np.conj(fa), n=N)[:n]
    return 2.0 * c

def _score(a, b=None):
    if b is None:
        b = _fft_square(a)
    s = float(a.sum())
    if s < 1e-300:
        return float('inf'), b, 0.0
    maxb = float(b.max())
    return float(2.0 * a.size * maxb / (s * s)), b, maxb

def _softmax_weights(b, focus, eps=0.005):
    L = int(b.size)
    maxb = float(b.max())
    if maxb < 1e-300:
        return np.ones(L) / L
    z = (maxb - b) / maxb
    with np.errstate(over='ignore', under='ignore'):
        p = np.exp(-z * focus)
    s = float(p.sum())
    if s < 1e-12:
        p = np.zeros(L)
        p[int(np.argmax(b))] = 1.0
    else:
        p = (1 - eps) * (p / s) + eps / L
    return p

def _topk_weights(b, k, eps=0.005):
    L = int(b.size)
    if k >= L:
        return np.ones(L) / L
    idx = np.argpartition(b, L - k)[-k:]
    w = np.zeros(L)
    w[idx] = 1.0 / k
    w = (1 - eps) * w + eps / L
    return w

def _hybrid_weights(b, focus, topk_frac=0.1):
    L = int(b.size)
    k = max(1, int(L * topk_frac))
    w1 = _softmax_weights(b, focus)
    w2 = _topk_weights(b, k)
    return 0.5 * w1 + 0.5 * w2

def _normalize(a):
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-300:
        return np.ones(a.size) / a.size
    return a / s

        p = 0.9 * p + 0.1 / L

        grad = _fft_grad(a, p)

            b = b_new
            score = score_new
            eta = min(eta * 1.02, 2.0)
        else:
            eta = max(eta * 0.8, 0.01)
        
        it += 1
    
    return best_a, best_score

def _optimize_scipy_small(n, time_limit, rng, label=""):
    """Use scipy.optimize for small n with smooth approximation."""
    from scipy.optimize import minimize
    
    # Start with a random positive vector
    a0 = rng.uniform(0.5, 1.5, size=n)
    a0 /= a0.sum()
    
    beta = 50.0  # softmax temperature for approximating max
    
    def objective(x):
        a = np.exp(x)
        a = np.maximum(a, 1e-15)
        a /= a.sum()
        b = _fft_square(a)
        # Smooth max
        maxb = float(b.max())
        if maxb < 1e-300:
            return 0.0
        z = b / maxb * beta
        log_sum = float(np.log(np.sum(np.exp(z - z.max()))) + z.max()) / beta
        return float(2.0 * n * log_sum)
    
    x0 = np.log(a0)
    t0 = time.monotonic()
    
    result = minimize(objective, x0, method='L-BFGS-B', 
                      options={'maxiter': 500, 'maxfun': 2000,
                               'ftol': 1e-12, 'gtol': 1e-10})
    
    x_opt = result.x
    a_opt = np.exp(x_opt)
    a_opt = np.maximum(a_opt, 1e-15)
    a_opt /= a_opt.sum()
    
    b_opt = _fft_square(a_opt)
    score_opt, _, _ = _score(a_opt, b_opt)
    
    return a_opt, float(score_opt)

def _upsample(a, factor):
    """Upsample by inserting zeros (zero-stuffing)."""
    n = int(a.size)
    new_n = n * factor
    a_new = np.zeros(new_n)
    a_new[::factor] = a * factor  # scale to maintain sum
    return a_new

def _downsample(a, factor):
    """Downsample by summing consecutive blocks."""
    n = int(a.size)
    new_n = n // factor
    a_new = np.zeros(new_n)
    for i in range(new_n):
        a_new[i] = a[i*factor:(i+1)*factor].sum()
    return a_new

Optimize a non-negative step function to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses mirror descent with momentum, adaptive focus scheduling, peak deflation,

def _pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _fft_square(a):

    fa = np.fft.rfft(a, n=N)
    b = np.fft.irfft(fa * fa, n=N).real[:L]

def _fft_grad(a, p):

    fp = np.fft.rfft(p, n=N)

    if s < 1e-300:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def _softmax_weights(b, focus, eps=0.005):
    L = int(b.size)
    maxb = float(b.max())
    if maxb < 1e-300:
        return np.ones(L) / L
    z = (maxb - b) / maxb
    with np.errstate(over='ignore', under='ignore'):
        p = np.exp(-z * focus)
    s = float(p.sum())
    if s < 1e-12:
        p = np.zeros(L)
        p[int(np.argmax(b))] = 1.0
    else:
        p = (1.0 - eps) * (p / s) + eps / L
    return p


def _topk_weights(b, frac, eps=0.005):
    L = int(b.size)
    k = max(1, int(L * frac))
    if k >= L:
        return np.ones(L) / L
    idx = np.argpartition(b, L - k)[-k:]
    w = np.zeros(L)
    w[idx] = 1.0 / k
    w = (1.0 - eps) * w + eps / L
    return w


def _combined_weights(b, focus, topk_frac=0.05, mix=0.5):
    w1 = _softmax_weights(b, focus)
    w2 = _topk_weights(b, topk_frac)
    return mix * w1 + (1.0 - mix) * w2


def _optimize_momentum(a0, time_limit, rng, label="",
                       focus_start=3.0, focus_rate=1.004, max_focus=3000.0,
                       eta_init=0.15, mom_init=0.80, mom_max=0.95,
                       use_topk=False, topk_frac=0.05):
    """Mirror descent with momentum and adaptive focus."""
    a = _normalize(np.asarray(a0, dtype=np.float64).copy())
    n = int(a.size)

    best_a = a.copy()
    best_score = float('inf')
    b = None
    score = float('inf')

    eta = eta_init
    momentum = mom_init
    v = np.zeros(n)

    focus = focus_start
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    max_no = 600
    resets = 0
    max_resets = 40

    while time.monotonic() - t0 < time_limit:
        if b is None:
            b = _fft_square(a)
            score, _, _ = _score_internal(a, b)

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no:
            resets += 1
            if resets > max_resets:
                break
            noise_scale = 0.08 + 0.04 * min(resets, 5)
            a = best_a * np.exp(rng.normal(0, noise_scale, size=n))
            a = _normalize(a)
