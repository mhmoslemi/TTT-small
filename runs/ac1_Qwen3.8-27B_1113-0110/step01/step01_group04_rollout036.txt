    n = v.size
    if n == 0:
        return v.copy()

    idx = np.arange(1, n+1, dtype=np.float64)

        return np.full(n, 1.0/float(n), dtype=np.float64)

        return np.inf
    v = v / s
    L = _pow2(2 * n - 1)
    A = np.fft.rfft(v, n=L)
    c = np.fft.irfft(A * A, n=L)[:2 * n - 1]
    mx = float(c.max())
    if not np.isfinite(mx) or mx < 0:
        return np.inf
    return 2.0 * n * mx


def _convolve(a, b):
    n = max(a.size, b.size)
    L = _pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b, n=L)
    return np.fft.irfft(A * B, n=L)[:2 * n - 1]


def _power_seq(n, p=0.5, off=1.0):
    j = np.arange(n, dtype=np.float64)
    x = np.maximum(j + off, 1e-8)
    return _project_simplex(x ** (-p))


def _symmetric_seq(n, p=1.0, off=1.0):
    mid = (n - 1) / 2.0
    j = np.arange(n, dtype=np.float64)
    x = np.abs(j - mid) + off
    return _project_simplex(x ** (-p))


def _generate_candidates(n):
    cands = []
    try:
        h = height_sequence_1
        if isinstance(h, (list, np.ndarray)):
            arr = np.asarray(h, dtype=np.float64)
            arr = _clean_v(arr)
            if arr.size == n and arr.sum() > 0:
                cands.append(_project_simplex(arr))
    except Exception:
        pass

    cands.append(np.full(n, 1.0 / float(n), dtype=np.float64))

    for p in [0.35, 0.4, 0.45, 0.5, 0.55, 0.6, 0.7, 0.8]:
        for off in [0.3, 0.5, 1.0, 2.0]:
            cands.append(_power_seq(n, p, off))

    for p in [0.5, 0.8, 1.0, 1.5, 2.0, 3.0]:
        for off in [0.5, 1.0, 2.0]:
            cands.append(_symmetric_seq(n, p, off))

    # Bimodal peaks
    for frac in [0.05, 0.1, 0.15, 0.2, 0.3]:
        v = np.full(n, 0.01)
        i1 = int(n * frac)
        i2 = n - 1 - i1
        v[i1] = 1.0
        v[i2] = 1.0
        cands.append(_project_simplex(v))

    # Edge-concentrated
    for w in [0.02, 0.05, 0.1, 0.2]:
        v = np.zeros(n)
        nw = max(1, int(n * w))
        v[:nw] = 1.0
        v[-nw:] = 1.0
        cands.append(_project_simplex(v))

    # Random smooth
    for seed in [7, 13, 42, 99]:
        rng = np.random.default_rng(seed)
        v = np.exp(rng.normal(0, 0.5, n))
        v *= (np.arange(n) + 1) ** -0.3
        cands.append(_project_simplex(v))

    return cands


def _ratios(a):
    a = _clean_v(a).copy()
    n = a.size
    s = float(a.sum())
    if s <= 0:
        return np.zeros(max(0, n - 1))
    a = a / s
    ref = float(a[-1])
    if ref <= 1e-200:
        a[-1] = 1e-12
        a = a / float(a.sum())
        ref = float(a[-1])
    x = np.log(np.maximum(a[:-1], 1e-300) / max(ref, 1e-300))
    return np.clip(x, -80.0, 80.0)


def _from_ratios(x, n):
    x = np.asarray(x, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    if n == 1:
        return np.ones(1)
    m = float(np.max(x)) if x.size else 0.0
    e = np.exp(np.clip(x - m, -700, 700))
    e_last = float(np.exp(-m))
    s = float(e.sum()) + e_last
    if s < 1e-300:
        return np.full(n, 1.0 / float(n))
    a = np.empty(n)
    a[:-1] = e / s
    a[-1] = e_last / s
    return a


def _soft_obj_grad(x, eps, n):
    a = _from_ratios(x, n)
    L = _pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    c = np.fft.irfft(A * A, n=L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    cmax = float(c.max())
    if cmax < 1e-300:
        return 0.0, np.zeros(max(0, n - 1))
    eps = max(eps, 1e-12)
    z = (c - cmax) / eps
    z = np.clip(z, -700, 0)
    w = np.exp(z)
    sume = float(w.sum())
    if sume < 1e-300:
        return 1e300, np.zeros(max(0, n - 1))
    w = w / sume
    f = cmax + eps * float(np.log(sume))
    W = np.fft.rfft(w, n=L)
    ga = 2.0 * np.fft.irfft(W * np.conj(A), n=L)[:n]
    dot = float(np.dot(a, ga))
    gfull = a * (ga - dot)
    gy = gfull[:-1]
    if not np.all(np.isfinite(gy)):
        gy = np.zeros(max(0, n - 1))
    return f, gy


def _q_obj_grad(x, q, n):
    a = _from_ratios(x, n)
    L = _pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    c = np.fft.irfft(A * A, n=L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    if q == 2.0:
        f = float(np.dot(c, c))
        w = c
    else:
        f = float(np.sum(c ** q))
        w = c ** (q - 1.0)
    W = np.fft.rfft(w, n=L)
    ga = (2.0 * q) * np.fft.irfft(W * np.conj(A), n=L)[:n]
    dot = float(np.dot(a, ga))
    gfull = a * (ga - dot)
    gy = gfull[:-1]
    if not np.all(np.isfinite(gy)):
        gy = np.zeros(max(0, n - 1))
    return f, gy


def _lp_solve(a, M, max_time=90):
    """Solve: max sum(g) s.t. g>=0, conv(a,g) <= M.
    Uses constraint selection for speed."""
    n = a.size
    n_con = 2 * n - 1

    # Build full A_ub
    A_ub_full = np.zeros((n_con, n), dtype=np.float64)
    for j in range(n):
        A_ub_full[j:j + n, j] = a

    c_obj = -np.ones(n)
    bounds = [(0, None)] * n

    # Start with top-300 constraints
    conv_aa = _convolve(a, a)
    top_k = np.argsort(conv_aa)[::-1][:min(300, n_con)]

    t0 = time.monotonic()
    g = None

    for attempt in range(8):
        elapsed = time.monotonic() - t0
        remaining = max_time - elapsed
        if remaining < 5:
            break

        A_ub = A_ub_full[top_k]
        b_ub = np.full(len(top_k), M)

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'time_limit': max(1, remaining - 2)})
            if res.success:
                g = res.x
                s = float(g.sum())
                if s < 1e-12:
                    g = None
                    break
                g = g / s
                # Verify all constraints
                conv_ag = _convolve(a, g)
                if conv_ag.max() <= M * (1 + 1e-9):
                    return g
                # Find violated constraints and add them
                violated = np.where(conv_ag > M * (1 + 1e-9))[0]
                new_topk = np.unique(np.concatenate([top_k, violated]))
                top_k = new_topk[:min(len(new_topk), 600)]
                g = None
            else:
                break
        except Exception:
            break

    return g


def _gradient_polish(a, end_time, seed, stage="soft"):
    """Run gradient-based polishing."""
    n = a.size
    x = _ratios(a)
    bounds = [(-80.0, 80.0)] * (n - 1)
    best = a.copy()
    best_sc = _score(best)
    rng = np.random.default_rng(seed)

    if stage in ("l2", "l3"):
        q = 2.0 if stage == "l2" else 3.0
        while time.monotonic() < end_time:
            try:
                res = _minimize(_q_obj_grad, x, args=(q, n), method='L-BFGS-B',
                                jac=True, bounds=bounds,
                                options={'maxiter': 60, 'ftol': 1e-16, 'gtol': 1e-14})
            except Exception:
                break
            a_new = _from_ratios(res.x, n)
            sc = _score(a_new)
            if sc < best_sc - 1e-14:
                best = a_new
                best_sc = sc
                x = _ratios(best)
            else:
                break
    else:
        eps = max(1e-7, 0.05 * best_sc / (2.0 * n))
        eps_min = 1e-8
        no_imp = 0
        while time.monotonic() < end_time:
            try:
                res = _minimize(_soft_obj_grad, x, args=(eps, n), method='L-BFGS-B',
                                jac=True, bounds=bounds,
                                options={'maxiter': 60, 'ftol': 1e-16, 'gtol': 1e-14})
            except Exception:
                break
            a_new = _from_ratios(res.x, n)
            sc = _score(a_new)
            if sc < best_sc - 1e-14:
                best = a_new
                best_sc = sc
                x = _ratios(best)
                no_imp = 0
                eps = max(eps_min, eps * 0.65)
            else:
                no_imp += 1
                if no_imp >= 4:
                    x = np.clip(x + rng.normal(0, 0.04, n - 1), -80, 80)
                    if eps <= eps_min * 1.05:
                        break
                eps = max(eps_min, eps * 0.5)

    return best, best_sc


def _local_search(a0, end_time, seed):
    a = _project_simplex(a0)
    n = a.size
    best = a.copy()
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    sigma = 0.08
    last_best = time.monotonic()
    start = last_best

    while time.monotonic() < end_time:
        cand = best.copy()
        nblk = 1 + int(rng.integers(0, 3))
        for _ in range(nblk):
            w = int(np.exp(rng.uniform(0, np.log(max(3, n // 4))))
            w = max(1, min(w, n))
            s = int(rng.integers(0, n - w + 1))
            cand[s:s + w] *= np.exp(rng.normal(0, sigma, size=w))
        cand = _project_simplex(cand)
        sc = _score(cand)
        if sc < best_sc - 1e-14:
            best = cand
            best_sc = sc
            last_best = time.monotonic()
            sigma = min(0.3, sigma * 1.04)
        else:
            el = time.monotonic() - last_best
            if el > 4:
                sigma = max(0.005, sigma * 0.75)
                if el > 15 and time.monotonic() - start > 30:
                    cand = _project_simplex(best * np.exp(rng.normal(0, 0.015, n)))
                    sc2 = _score(cand)
                    if sc2 < best_sc:
                        best = cand
                        best_sc = sc2
                        last_best = time.monotonic()
    return best, best_sc


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t0 = time.monotonic()
    budget = max(0.0, float(budget_s) - 10.0)
    deadline = t0 + budget
    n = 3809

    try:
        h = height_sequence_1
        if isinstance(h, (list, np.ndarray)):
            arr = np.asarray(h, dtype=np.float64)
            if arr.ndim == 1 and 100 <= arr.size <= 20000:
                n = int(arr.size)
    except Exception:
        pass
    if n < 32:
        n = 3809

    print(f"[t=0] n={n}, budget={budget:.0f}s")

    # Generate and score candidates
    cands = _generate_candidates(n)
    scored = []
    for i, c in enumerate(cands):
        sc = _score(c)
        if np.isfinite(sc):
            scored.append((sc, i, c))
    scored.sort()
    print(f"[t={time.monotonic()-t0:.1f}s] {len(scored)} candidates, best={scored[0][0]:.6f}")

    if not scored:
        return [float(x) for x in _power_seq(n, 0.5, 1.0)]

    best_a = scored[0][2].copy()
    best_sc = scored[0][0]

    # Phase 1: LP-based diversification (25% of time)
    lp_end = t0 + 0.25 * budget
    print(f"\n[Phase 1: LP diversification] deadline={lp_end - t0:.0f}s")
    lp_it = 0
    while time.monotonic() < lp_end:
        a_norm = best_a / float(best_a.sum())
        conv = _convolve(a_norm, a_norm)
        M = float(conv.max())
        print(f"  LP iter {lp_it}: M={M:.10f}, score={best_sc:.6f}")

        g = _lp_solve(a_norm, M, max_time=60)
        if g is None:
            print("  LP failed, stopping phase 1")
            break

        improved = False
        for t_step in [0.4, 0.25, 0.15, 0.1, 0.05]:
            a_new = _project_simplex((1 - t_step) * a_norm + t_step * g)
            sc = _score(a_new)
            if sc < best_sc - 1e-14:
                best_a = a_new
                best_sc = sc
                improved = True
                print(f"    t={t_step}: improved to {best_sc:.6f}")
                break

        lp_it += 1

        # Occasionally do a gradient step within phase 1
        if lp_it % 2 == 0 and time.monotonic() < lp_end - 30:
            a2, sc2 = _gradient_polish(best_a, time.monotonic() + 30, int(seed) + lp_it, "l2")
            if sc2 < best_sc:
                best_a = a2
                best_sc = sc2
                print(f"    Gradient: {best_sc:.6f}")

        if not improved and lp_it > 3:
            # Try a different starting point for next LP
            if lp_it < len(scored):
                best_a = scored[min(lp_it, len(scored) - 1)][2].copy()
                best_sc = _score(best_a)
                print(f"    Reset to candidate {lp_it}: {best_sc:.6f}")
            else:
                break

    # Phase 2: Multi-stage gradient polishing (50% of time)
    p2_start = max(lp_end, time.monotonic())
    p2_end = deadline - 0.15 * budget
    print(f"\n[Phase 2: Gradient polishing] {p2_start - t0:.0f}s to {p2_end - t0:.0f}s")

    # Stage 2a: L2
    q2_end = max(p2_start + 10, p2_start + 0.20 * (p2_end - p2_start))
    if time.monotonic() < q2_end:
        best_a, best_sc = _gradient_polish(best_a, q2_end, int(seed) + 100, "l2")
        print(f"  After L2: {best_sc:.6f}")

    # Stage 2b: L3
    q3_end = max(q2_end + 10, p2_start + 0.40 * (p2_end - p2_start))
    if time.monotonic() < q3_end:
        best_a, best_sc = _gradient_polish(best_a, q3_end, int(seed) + 200, "l3")
        print(f"  After L3: {best_sc:.6f}")

    # Stage 2c: Log-sum-exp with decreasing temperature
    if time.monotonic() < p2_end:
        best_a, best_sc = _gradient_polish(best_a, p2_end, int(seed) + 300, "soft")
        print(f"  After soft: {best_sc:.6f}")

    # Phase 2d: Try LP again from improved point
    if time.monotonic() < p2_end - 20:
        a_norm = best_a / float(best_a.sum())
        conv = _convolve(a_norm, a_norm)
        M = float(conv.max())
        g = _lp_solve(a_norm, M, max_time=45)
        if g is not None:
            for t_step in [0.3, 0.2, 0.1, 0.05]:
                a_new = _project_simplex((1 - t_step) * a_norm + t_step * g)
                sc = _score(a_new)
                if sc < best_sc - 1e-14:
                    best_a = a_new
                    best_sc = sc
                    print(f"  LP post-gradient: {best_sc:.6f}")
                    break

    # Phase 3: Local search (remaining 15%)
    print(f"\n[Phase 3: Local search]")
    best_a, best_sc = _local_search(best_a, deadline, int(seed) + 999)
    print(f"  Final: {best_sc:.6f}")

    # Final validation
    best_a = _project_simplex(best_a)
    out = [float(x) for x in best_a]
    try:
        val = evaluate_sequence(out)
        print(f"  evaluate_sequence: {val}")
        if not np.isfinite(val):
            out = [float(x) for x in _power_seq(n, 0.5, 1.0)]
            print(f"  Fallback used")
    except Exception:
        pass

    print(f"[t={time.monotonic()-t0:.1f}s] Done. Score={best_sc:.6f}")
    return out
