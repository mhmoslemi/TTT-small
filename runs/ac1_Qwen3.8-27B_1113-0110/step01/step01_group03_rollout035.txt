```python
def frank_wolfe_step(a, M):
    """Solve the LP to find the best direction."""
    n = len(a)
    # Build constraint matrix A (2n-1 x n)
    # A[k, j] = a[k-j] if 0 <= k-j <= n-1, else 0
    
    # For the primal: max sum(b) s.t. A @ b <= M*ones, b >= 0
    # Using scipy linprog: min -sum(b) s.t. A @ b <= M*ones, b >= 0
    
    c_obj = -np.ones(n)
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_min = max(0, k - (n-1))
        j_max = min(n-1, k)
        for j in range(j_min, j_max+1):
            A_ub[k, j] = a[k-j]
    b_ub = np.full(2*n-1, M)
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if result.success:
        b_star = result.x
        b_star = np.maximum(b_star, 0)
        S = b_star.sum()
        if S < 1e-10:
            return None
        b_star = b_star / S * n
        return b_star
    return None
```

    c = np.fft.irfft(fa * fa, size)

    """Fast score using FFT convolution."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_self_conv(a)
    return 2.0 * n * c.max() / (S*S)

def compute_grad_multi_peak(a, c, peaks, S):
    """Gradient w.r.t. a, considering multiple peak lags."""
    n = len(a)
    M = c.max()
    grad = np.zeros(n)
    for k in peaks:
        # dc_k/da_i = 2*a[k-i] for 0 <= k-i < n
        # So grad[i] += 2*a[k-i] for i in [max(0,k-n+1), min(n-1,k)]

        if i_max >= i_min:

    return grad
