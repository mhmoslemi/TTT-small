Optimize non-negative step-function sequence minimizing 2*n*max(conv(a,a))/(sum a)^2.
Uses a multi-phase strategy: coarse L-BFGS on surrogates, then direct peak-reduction
via projected subgradient with active set tracking, then gradient-guided local search

import numpy as np
import time

try:
    from scipy.optimize import minimize as _minimize
except Exception:
    _minimize = None


def _pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _clean(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    return np.maximum(v, 0.0)


def _project_simplex(v):
    v = _clean(v)
    n = v.size
    if n == 0:
        return v.copy()
    x = np.sort(v)[::-1]
    css = np.cumsum(x)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = x > (css - 1.0) / idx
    if not np.any(cond):
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    r = int(np.nonzero(cond)[0][-1])
    tau = (css[r] - 1.0) / float(r + 1)
    out = v - tau
    np.maximum(out, 0.0, out=out)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    out *= 1.0 / s
    return out


def _score(a, L=None):
    v = _clean(a)
    n = v.size
    if n == 0:
        return np.inf
    s = float(v.sum())
    if s < 1e-12:
        return np.inf
    v = v / s
    if L is None:
        L = _pow2(2 * n - 1)

    mx = float(np.max(c)) if c.size else 0.0

        return np.inf
    return 2.0 * float(n) * mx


def _convolve(a, L=None):
    v = _clean(a)
    n = v.size
    if L is None:
        L = _pow2(2 * n - 1)
    A = np.fft.rfft(v, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _power_seq(n, p=0.5, off=1.0):
    j = np.arange(n, dtype=np.float64)
    x = np.maximum(j + float(off), 1e-8)
    return _project_simplex(np.power(x, -float(p)))


def _two_tail(n, p=0.5, off=1.0, tw=0.0):
    j = np.arange(n, dtype=np.float64)
    base = np.power(np.maximum(j + float(off), 1e-8), -float(p))
    if tw:
        rev = np.maximum((n - 1.0 - j) + float(off), 1e-8)
        base = base + float(tw) * np.power(rev, -float(p))
    return _project_simplex(_clean(base))


def _modulated_seq(n, p=0.5, off=1.0, freq=0.05, amp=0.1):
    j = np.arange(n, dtype=np.float64)
    base = np.power(np.maximum(j + float(off), 1e-8), -float(p))
    mod = 1.0 + float(amp) * np.sin(2.0 * np.pi * float(freq) * j / float(n))
    return _project_simplex(np.maximum(base * mod, 0.0))


def _generate_candidates(n):
    cands = []
    try:
        h = height_sequence_1
        if isinstance(h, (list, np.ndarray)):
            arr = np.asarray(h, dtype=np.float64)
            arr = _clean(arr)
            if arr.size == n and arr.sum() > 0.0:
                cands.append(_project_simplex(arr))
    except Exception:
        pass

    cands.append(np.full(n, 1.0 / float(n), dtype=np.float64))

    for p in (0.4, 0.45, 0.5, 0.55, 0.6, 0.7):
        for off in (0.3, 0.7, 1.0, 1.5, 2.0):
            cands.append(_power_seq(n, p, off))

    for tw in (0.01, 0.05, 0.1, 0.2, 0.5):
        cands.append(_two_tail(n, 0.5, 1.0, tw))

    for freq in (0.02, 0.05, 0.1, 0.2):
        for amp in (0.05, 0.1, 0.2):
            cands.append(_modulated_seq(n, 0.5, 1.0, freq, amp))

    for seed in (7, 13, 42, 99, 123):
        rng = np.random.default_rng(seed)
        v = np.exp(rng.normal(0, 0.8, size=n))
        v *= np.power(np.arange(n, dtype=np.float64) + 1.0, -0.5)
        cands.append(_project_simplex(v))

    return cands


def _soft_obj_grad(x, eps):
    a = _softmax_from(x)
    n = a.size
    if n <= 1:
        return 0.0, np.zeros_like(x)
    L = _pow2(3 * n - 2)
    A = np.fft.rfft(a, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    cmax = float(c.max())
    if cmax <= 1e-300:
        return 0.0, np.zeros_like(x)
    eps = max(float(eps), 1e-12)
    z = (c - cmax) / eps
    z = np.clip(z, -745.0, 0.0)
    e = np.exp(z)
    sume = float(e.sum())
    if not np.isfinite(sume) or sume <= 0:
        return 1e300, np.zeros_like(x)
    f = cmax + eps * float(np.log(sume))
    w = e / sume
    W = np.fft.rfft(w, n=L)
    ga = 2.0 * np.fft.irfft(W * np.conj(A), n=L)[:n]
    dot = float(np.dot(a, ga))
    gfull = a * (ga - dot)
    gy = gfull[:-1]
    if not np.all(np.isfinite(gy)):
        gy = np.zeros_like(x)
    return f, gy


def _q_obj_grad(x, q):
    a = _softmax_from(x)
    n = a.size
    if n <= 1:
        return 0.0, np.zeros_like(x)
    L = _pow2(3 * n - 2)
    A = np.fft.rfft(a, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    qq = float(q)
    if qq == 2.0:
        f = float(np.dot(c, c))
        w = c
    else:
        f = float(np.sum(np.power(c, qq)))
        w = np.power(c, max(qq - 1.0, 0.01))
    W = np.fft.rfft(w, n=L)
    ga = (2.0 * qq) * np.fft.irfft(W * np.conj(A), n=L)[:n]
    dot = float(np.dot(a, ga))
    gfull = a * (ga - dot)
    gy = gfull[:-1]
    if not np.all(np.isfinite(gy)):
        gy = np.zeros_like(x)
    if not np.isfinite(f):
        f = 1e300
    return f, gy


def _ratios(a):
    a = _clean(np.asarray(a, dtype=np.float64)).copy()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    s = float(a.sum())
    if s <= 0 or not np.isfinite(s):
        a = np.full(n, 1.0 / n, dtype=np.float64)
        s = 1.0
    else:
        a = a / s
    if n == 1:
        return np.array([], dtype=np.float64)
    ref = float(a[-1])
    if ref <= 1e-200:
        a[-1] = 1e-12
        a = a / float(a.sum())
        ref = float(a[-1])
    x = np.log(np.maximum(a[:-1], 1e-300) / max(ref, 1e-300))
    return np.clip(x, -80.0, 80.0)


def _softmax_from(x):
    x = np.asarray(x, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    n = x.size + 1
    if n == 1:
        return np.ones(1, dtype=np.float64)
    m = 0.0
    if x.size:
        mx = float(np.max(x))
        if mx > m:
            m = mx
    e = np.exp(np.clip(x - m, -745.0, 709.0))
    e_last = float(np.exp(-m))
    s = float(e.sum()) + e_last
    if not np.isfinite(s) or s <= 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    a = np.empty(n, dtype=np.float64)
    a[:-1] = e / s
    a[-1] = e_last / s
    return a


def _lbb(fun, x, args, maxiter, bounds):
    if _minimize is None:
        return x, 99
    x = np.asarray(x, dtype=np.float64)
    if x.size == 0:
        return x, 1
    options = {"maxiter": int(maxiter), "disp": 0, "maxfun": int(max(200, 10 * maxiter)),
               "ftol": 1e-16, "gtol": 1e-10}
    try:
        res = _minimize(fun, x, args=args, method="L-BFGS-B", jac=True,
                        bounds=bounds, options=options)
    except Exception:
        return x, 99
    x_new = np.asarray(res.x, dtype=np.float64)
    x_new = np.where(np.isfinite(x_new), x_new, 0.0)
    x_new = np.clip(x_new, -80.0, 80.0)
    try:
        status = int(res.status)
    except Exception:
        status = 0
    return x_new, status


def _peak_reduction(a0, end_time, seed, top_k=5):
    """
    Direct projected subgradient method on the max of the convolution.
    Uses a blend of the top-k peaks' gradients for better signal.
    """
    a = _project_simplex(a0)
    n = a.size
    best = a.copy()
    best_score = _score(best)
    rng = np.random.default_rng(seed)

    lr = 1e-4
    lr_min = 1e-7
    lr_max = 5e-3
    last_best = time.monotonic()
    start = time.monotonic()
    bounce_count = 0

    while time.monotonic() < end_time:
        c = _convolve(a)
        csum = float(c.sum())
        if csum < 1e-30:
            break

        # Find top-k peaks
        k = min(top_k, c.size)
        top_idx = np.argsort(c)[::-1][:k]
        top_val = c[top_idx]
        cmax = float(top_val[0])

        if cmax < 1e-30:
            break

        score = 2.0 * float(n) * cmax
        now = time.monotonic()

        if score < best_score - 1e-14:
            best_score = score
            best = a.copy()
            last_best = now
            lr = min(lr_max, lr * 1.03)
            bounce_count = 0

        # Build gradient: weighted sum of gradients at top-k peaks
        # grad of c[k] w.r.t. a[i] = a[k-i] + a[i-k] (symmetric), but since
        # c[k] = sum_j a[j]*a[k-j], d c[k]/d a[i] = a[k-i] + a[i-k] for valid indices
        # Actually for the full convolution c[k] = sum_j a[j]*a[k-j],
        # dc[k]/da[i] = a[k-i] + a[i-k] (both terms when valid)
        # But we can compute it via FFT: if w[k] = 1, then grad = 2*ifft(fft(w)*conj(fft(a)))[:n]
        # Let's use the FFT approach for speed.

        # Weight the top-k peaks with softmax weights
        wpeaks = np.zeros(c.size, dtype=np.float64)
        if k >= 1:
            wmax = float(top_val[0])
            if wmax > 1e-30:
                weights = np.exp((top_val - wmax) / (wmax * 0.1 + 1e-30))
                weights /= float(weights.sum())
                wpeaks[top_idx] = weights

        L = _pow2(3 * n - 2)
        A = np.fft.rfft(a, n=L)
        W = np.fft.rfft(wpeaks, n=L)
        g = 2.0 * np.fft.irfft(W * np.conj(A), n=L)[:n]

        # Project gradient: normalize
        gnorm = float(np.linalg.norm(g))
        if gnorm < 1e-30:
            break
        g = g / gnorm

        # Projected gradient step on simplex
        a_candidate = a - lr * g
        a_candidate = _project_simplex(a_candidate)

        # Check if improvement
        sc_new = _score(a_candidate)
        if np.isfinite(sc_new) and sc_new < score + lr * 0.5:
            a = a_candidate
        else:
            # Retry with smaller step
            a_candidate = a - lr * 0.3 * g
            a_candidate = _project_simplex(a_candidate)
            sc_new2 = _score(a_candidate)
            if np.isfinite(sc_new2) and sc_new2 < score:
                a = a_candidate
                lr = max(lr_min, lr * 0.7)
            else:
                lr = max(lr_min, lr * 0.5)
                bounce_count += 1
                if bounce_count > 15:
                    # Add random perturbation
                    a = _project_simplex(a * np.exp(rng.normal(0, 0.02, size=n)))
                    bounce_count = 0
                    lr = max(lr_min, lr * 0.3)

        # Anneal based on time
        elapsed = now - start
        total = max(1e-6, end_time - start)
        frac = min(1.0, elapsed / total)
        lr_base = lr_max * (1.0 - frac) + lr_min * frac
        if lr > lr_base * 2:
            lr = min(lr, lr_base)

        # Periodic restart from best
        if now - last_best > 10.0:
            a = _project_simplex(best * np.exp(rng.normal(0, 0.03, size=n)))
            bounce_count = 0
            lr = min(lr_max, 1e-3)

    return best, best_score


def _annealed_local_search(a0, end_time, seed):
    """
    Local search with simulated annealing acceptance, gradient-guided perturbations.
    """
    a = _project_simplex(a0)
    n = a.size
    best = a.copy()
    best_score = _score(best)
    rng = np.random.default_rng(seed)

    T = 0.005  # initial temperature
    T_min = 1e-8
    T_decay = 0.995
    start = time.monotonic()
    total_time = max(1e-6, end_time - start)
    last_best = start
    step_count = 0

    while time.monotonic() < end_time:
        step_count += 1
        c = _convolve(a)
        cmax = float(c.max())
        score = 2.0 * float(n) * cmax
        now = time.monotonic()

        if score < best_score - 1e-14:
            best_score = score
            best = a.copy()
            last_best = now
            T = min(0.005, T * 1.1)

        # Generate perturbation
        # Mix of random block and gradient-guided
        if step_count % 5 == 0:
            # Gradient-guided: perturb entries that contribute most to active peaks
            top_idx = np.argsort(c)[::-1][:10]
            g = np.zeros(n, dtype=np.float64)
            for ki in top_idx:
                ki = int(ki)
                # dc[ki]/da[i] = a[ki-i] (for valid i)
                lo = max(0, ki - (n - 1))
                hi = min(n - 1, ki)
                for i in range(lo, hi + 1):
                    j = ki - i
                    if 0 <= j < n:
                        g[i] += a[j]
            # Normalize
            gnorm = float(np.linalg.norm(g))
            if gnorm > 1e-20:
                g /= gnorm
                # Perturb in direction that reduces peaks (decrease entries with high gradient)
                noise = rng.normal(0, 0.05, size=n)
                a_candidate = a * np.exp(0.05 * (noise - g))
            else:
                a_candidate = a * np.exp(rng.normal(0, 0.08, size=n))
        else:
            # Random block perturbation
            a_candidate = a.copy()
            nblocks = int(rng.integers(1, 4))
            for _ in range(nblocks):
                maxw = max(3, n // 6)
                w = int(np.exp(rng.uniform(np.log(2.0), np.log(float(maxw)))))
                w = max(1, min(w, n))
                s = int(rng.integers(0, n - w + 1))
                a_candidate[s:s + w] *= float(np.exp(rng.normal(0, 0.1)))

        a_candidate = _project_simplex(a_candidate)
        sc_new = _score(a_candidate)

        # Simulated annealing acceptance
        if np.isfinite(sc_new):
            delta = sc_new - score
            if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
                a = a_candidate
            else:
                # Try from best
                if rng.random() < 0.1:
                    a = _project_simplex(best * np.exp(rng.normal(0, 0.04, size=n)))

        T *= T_decay
        if T < T_min:
            T = T_min

        # Reset if stuck
        if now - last_best > 15.0:
            a = _project_simplex(best * np.exp(rng.normal(0, 0.05, size=n)))
            T = max(T, 0.002)

    return best, best_score


def _coarse_refine(a0, end_time, seed):
    """
    L-BFGS-B on log-sum-exp surrogate with decreasing eps.
    """
    a = _project_simplex(a0)
    n = a.size
    best = a.copy()
    best_score = _score(best)
    if _minimize is None:
        return best, best_score

    x = _ratios(a)
    bounds = [(-80.0, 80.0)] * max(0, n - 1)
    start = time.monotonic()
    total = max(1e-6, end_time - start)

    # Stage 1: L2
    end1 = start + 0.2 * total
    while time.monotonic() < end1:
        x, st = _lbb(_q_obj_grad, x, (2.0,), 60, bounds)
        if st == 1:
            break
    a = _softmax_from(x)
    sc = _score(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best = a.copy()

    # Stage 2: L4
    end2 = start + 0.4 * total
    if time.monotonic() < end2:
        while time.monotonic() < end2:
            x, st = _lbb(_q_obj_grad, x, (4.0,), 60, bounds)
            if st == 1:
                break
        a = _softmax_from(x)
        sc = _score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = a.copy()

    # Stage 3: log-sum-exp with decreasing eps
    end3 = start + 0.85 * total
    if time.monotonic() < end3:
        eps = 0.05 * (best_score / (2.0 * n))
        eps = max(eps, 1e-6)
        eps_min = 1e-9
        last_improve = time.monotonic()
        no_improve = 0

        while time.monotonic() < end3:
            x, st = _lbb(_soft_obj_grad, x, (eps,), 50, bounds)
            a = _softmax_from(x)
            sc = _score(a)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best = a.copy()
                last_improve = time.monotonic()
                no_improve = 0
                eps = max(eps_min, eps * 0.7)
            else:
                no_improve += 1
                if no_improve >= 3:
                    rng = np.random.default_rng(int(time.monotonic() * 1e6) % 100000)
                    x = np.clip(x + rng.normal(0, 0.05, size=max(1, x.size)), -80, 80)
                eps = max(eps_min, eps * 0.5)
                if eps <= eps_min * 1.01 and no_improve >= 5:
                    break

    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    1. Generate diverse candidates, pick top ones.
    2. Coarse L-BFGS refinement on top candidates.
    3. Peak-reduction via projected subgradient.
    4. Annealed local search with gradient guidance.
    Returns best sequence found.
    """
    t0 = time.monotonic()
    budget = max(0.0, float(budget_s) - 10.0)
    deadline = t0 + budget

    # Determine length
    n = 3809
    try:
        h = height_sequence_1
        if isinstance(h, (list, np.ndarray)):
            arr = np.asarray(h, dtype=np.float64)
            if arr.ndim == 1 and 100 <= arr.size <= 20000:
                n = int(arr.size)
    except Exception:
        pass
    if n < 32:
        n = 3809

    if budget < 5.0:
        a = _power_seq(n, 0.5, 1.0)
        return [float(x) for x in a]

    print(f"Starting optimization with n={n}, budget={budget:.1f}s")
    t_gen = time.monotonic()

    cands = _generate_candidates(n)
    scored = []
    for i, c in enumerate(cands):
        sc = _score(c)
        if np.isfinite(sc):
            scored.append((sc, i, c))
    scored.sort()

    if not scored:
        a = np.full(n, 1.0 / n, dtype=np.float64)
        return [float(x) for x in a]

    print(f"Generated {len(cands)} candidates in {time.monotonic()-t_gen:.1f}s")
    print(f"Best initial score: {scored[0][0]:.6f}")
    for i in range(min(5, len(scored))):
        print(f"  Candidate {i}: score={scored[i][0]:.6f}")

    global_best_a = np.asarray(scored[0][2], dtype=np.float64).copy()
    global_best_score = float(scored[0][0])

    # Allocate time:
    # Phase 1 (coarse L-BFGS): 30% split among top 3
    # Phase 2 (peak reduction): 40%
    # Phase 3 (annealed local search): 30%

    p1_end = t0 + 0.30 * budget
    p2_end = t0 + 0.70 * budget

    # Phase 1: Coarse refinement of top candidates
    top_k = min(4, len(scored))
    allocs = [(0.40, 0), (0.30, 1), (0.20, 2), (0.10, 3)]
    for frac, idx in allocs:
        if idx >= top_k:
            break
        cur_end = min(p1_end, t0 + (sum(f for f, _ in allocs[:allocs.index((frac, idx))] + [frac])) * 0.30 * budget)
        # Simpler: just split p1 time
        p1_share = 0.30 * budget * frac
        phase_end = min(p1_end, t0 + time.monotonic() - t0 + p1_share)
        # Actually let's just do it sequentially
        phase_end = p1_end  # will be naturally limited by the order
        # Let me redo: allocate time slots
        pass

    # Simpler time allocation
    p1_times = []
    remaining_p1 = p1_end - t0
    splits = [0.40, 0.30, 0.20, 0.10]
    cur = t0
    for i in range(top_k):
        seg = remaining_p1 * splits[i]
        p1_times.append((cur, cur + seg))
        cur += seg

    for i in range(top_k):
        start_t, end_t = p1_times[i]
        if time.monotonic() >= end_t - 1.0:
            break
        a = np.asarray(scored[i][2], dtype=np.float64).copy()
        a, sc = _coarse_refine(a, end_t, int(seed) + 100 * i)
        if np.isfinite(sc) and sc < global_best_score:
            global_best_score = sc
            global_best_a = np.asarray(a, dtype=np.float64).copy()
            print(f"  Phase 1, candidate {i}: score={sc:.6f} (t={time.monotonic()-t0:.0f}s)")

    # Try a second round of coarse refinement if time allows
    if time.monotonic() < p1_end:
        a, sc = _coarse_refine(global_best_a, p1_end, int(seed) + 999)
        if np.isfinite(sc) and sc < global_best_score:
            global_best_score = sc
            global_best_a = np.asarray(a, dtype=np.float64).copy()
            print(f"  Phase 1 extra: score={sc:.6f}")

    print(f"After Phase 1: best score = {global_best_score:.6f}")

    # Phase 2: Peak reduction
    if time.monotonic() < p2_end:
        # Split between peak reduction and a fresh coarse run
        pr_end = t0 + 0.55 * budget
        a, sc = _peak_reduction(global_best_a, pr_end, int(seed) + 5000, top_k=8)
        if np.isfinite(sc) and sc < global_best_score:
            global_best_score = sc
            global_best_a = np.asarray(a, dtype=np.float64).copy()
            print(f"  Peak reduction 1: score={sc:.6f} (t={time.monotonic()-t0:.0f}s)")

        # Another peak reduction round with different params
        if time.monotonic() < p2_end:
            a, sc = _peak_reduction(global_best_a, p2_end, int(seed) + 6000, top_k=3)
            if np.isfinite(sc) and sc < global_best_score:
                global_best_score = sc
                global_best_a = np.asarray(a, dtype=np.float64).copy()
                print(f"  Peak reduction 2: score={sc:.6f} (t={time.monotonic()-t0:.0f}s)")

    print(f"After Phase 2: best score = {global_best_score:.6f}")

    # Phase 3: Annealed local search
    if time.monotonic() < deadline:
        # First pass: longer run
        ls_end1 = t0 + 0.88 * budget
        a, sc = _annealed_local_search(global_best_a, ls_end1, int(seed) + 7000)
        if np.isfinite(sc) and sc < global_best_score:
            global_best_score = sc
            global_best_a = np.asarray(a, dtype=np.float64).copy()
            print(f"  Local search 1: score={sc:.6f} (t={time.monotonic()-t0:.0f}s)")

        # Second pass: shorter, different seed
        if time.monotonic() < deadline:
            a, sc = _annealed_local_search(global_best_a, deadline, int(seed) + 8000)
            if np.isfinite(sc) and sc < global_best_score:
                global_best_score = sc
                global_best_a = np.asarray(a, dtype=np.float64).copy()
                print(f"  Local search 2: score={sc:.6f}")

    # Final short peak reduction if time remains
    if time.monotonic() < deadline - 2.0:
        a, sc = _peak_reduction(global_best_a, deadline - 1.0, int(seed) + 9000, top_k=5)
        if np.isfinite(sc) and sc < global_best_score:
            global_best_score = sc
            global_best_a = np.asarray(a, dtype=np.float64).copy()

    global_best_a = _project_simplex(global_best_a)
    final_score = _score(global_best_a)
    print(f"Final score: {final_score:.6f} (time used: {time.monotonic()-t0:.1f}s)")

    # Validate
    out = [float(x) for x in global_best_a]
    try:
        val = evaluate_sequence(out)
        if not np.isfinite(val):
            print("WARNING: evaluate_sequence returned non-finite, using fallback")
            out = [float(x) for x in _power_seq(len(out), 0.5, 1.0)]
        else:
            print(f"validate_to: {val:.6f}")
    except Exception:
        pass

    return out
