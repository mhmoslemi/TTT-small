And c[m] gives the correlation at lag m. For m = 0, 1, ..., N_y-1 (where N_y = n is the length of y = a):
c[m] = sum_{n=0}^{n-1} x[n+m] * y[n]

So with x = softmax (length 2n-1) and y = a (length n), and padding to size >= 3n-2:
c[m] = sum_{n=0}^{n-1} softmax[n+m] * a[n] = g[m]

So:
```python
size = 1
while size < (2*n-1) + n - 1:  # = 3n-2
    size *= 2
X = np.fft.rfft(softmax, size)
Y = np.fft.rfft(a, size)
g = np.fft.irfft(np.conj(X) * Y, size)[:n]
```

This is O(n log n) instead of O(n^2). For n=2535, this is much faster.

import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog

"""
Minimizes 2n*max(convolve(a,a))/sum(a)^2 using:
1. Adam optimizer with soft-max smoothed objective
2. LP-based direction finding for escapes from local minima
3. Multiple sequence lengths and restarts
4. Fine-tuning with exact gradient at the end
"""


def fft_self_conv(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    if n <= 1:
        return a * a
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:2 * n - 1]


def fast_score(a):
    """Fast score using FFT."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_self_conv(a)
    M = c.max()
    return 2.0 * n * M / (S * S)


def normalize_sum(a, target):
    """Normalize array to have sum=target, keeping non-negativity."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-10:
        return np.full_like(a, target / len(a))
    return a / S * target


def compute_smoothed_gradient(a, tau):
    """
    Compute gradient of smoothed score 2n*softmax_max(conv(a,a))/S^2.
    Returns (gradient, smoothed_score).
    """
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return None, np.inf

    c = fft_self_conv(a)

    # Soft-max of c
    max_c = c.max()
    shifted = tau * (c - max_c)
    # Clip for numerical stability
    shifted = np.clip(shifted, -700, 0)
    exp_shifted = np.exp(shifted)
    exp_sum = exp_shifted.sum()
    if exp_sum < 1e-300:
        softmax = np.zeros_like(c)
        softmax[np.argmax(c)] = 1.0
    else:
        softmax = exp_shifted / exp_sum
    M_smooth = max_c + np.log(exp_sum) / tau

    # Gradient of M_smooth w.r.t. a:
    # g[i] = sum_j softmax[i+j] * a[j] for i = 0,...,n-1
    # This is cross-correlation of softmax with a
    size = 1
    while size < (2 * n - 1) + n - 1:
        size *= 2
    X = np.fft.rfft(softmax, size)
    Y = np.fft.rfft(a, size)
    grad_M = np.fft.irfft(np.conj(X) * Y, size)[:n]

    # Gradient of score = 2n*M/S^2
    # d/d a_i = 2n*grad_M[i]/S^2 - 4n*M*a[i]/S^3
    grad_score = (2.0 * n / (S * S)) * grad_M - (4.0 * n * M_smooth / (S ** 3)) * a

    score = 2.0 * n * M_smooth / (S * S)
    return grad_score, score


def exact_gradient_at_peak(a):
    """
    Compute exact gradient using the true max (not smoothed).
    Handles multiple peaks by averaging their contributions.
    """
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return None, np.inf

    c = fft_self_conv(a)
    M = c.max()

    # Find all peaks within small tolerance of M
    thresh = M - 1e-12
    peak_indices = np.where(c >= thresh)[0]

    if len(peak_indices) == 0:
        peak_indices = [int(np.argmax(c))]

    # Average gradient over all peaks
    grad_M = np.zeros(n)
    for k in peak_indices:
        # grad_M[i] += a[k-i] for valid i
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        if i_max >= i_min:
            j_start = k - i_max
            j_end = k - i_min + 1
            grad_M[i_min:i_max + 1] += a[j_start:j_end][::-1]
    grad_M /= len(peak_indices)

    grad_score = (2.0 * n / (S * S)) * grad_M - (4.0 * n * M / (S ** 3)) * a
    score = 2.0 * n * M / (S * S)
    return grad_score, score


def lp_direction(a, M=None, max_iters=500):
    """
    Find LP direction: maximize sum(g) subject to (a*g)_k <= M, g >= 0.
    Returns g (direction vector) or None if LP fails.
    """
    n = len(a)
    if M is None:
        c = fft_self_conv(a)
        M = c.max()

    # Build constraint matrix A: A[k, i] = a[k-i] if 0 <= k-i < n
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])

    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)

    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                  method='highs', options={'maxiter': max_iters, 'presolve': True})

    if res.status == 0 and res.fun > -1e-10:
        return res.x
    return None

        print(f"--- Final refinement pass for n={n_best} ---")
        result = optimize_for_length(n_best, best_overall, deadline,
                                     seed + 9999, time_left)
        if result is not None:
            a, score = result
            if score < best_score_overall:
                best_score_overall = score
                best_overall = a.copy()
                print(f"  New best after refinement: score={score:.8f}")

    # Ensure we return a valid sequence
    if best_overall is None:
        best_overall = parent.copy()

    # Final exact validation
    final_list = [float(max(0, min(1000, x))) for x in best_overall]
    final_score = float(evaluate_sequence(final_list))
    print(f"FINAL: score={final_score:.10f}, n={len(final_list)}")

    return final_list
