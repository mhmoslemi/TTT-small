def _pw2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _clean(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    return np.maximum(v, 0.0)


def _project_simplex(v):
    v = _clean(v)
    n = v.size
    if n == 0:
        return v.copy()
    x = np.sort(v)[::-1]
    css = np.cumsum(x)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = x > (css - 1.0) / idx
    if not np.any(cond):
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    r = int(np.nonzero(cond)[0][-1])
    tau = (css[r] - 1.0) / float(r + 1)
    out = v - tau
    np.maximum(out, 0.0, out=out)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    out *= 1.0 / s
    return out


def _score(a, L=None):
    v = _clean(a)
    n = v.size
    if n == 0:
        return np.inf
    s = float(v.sum())
    if s < 1e-12:
        return np.inf
    v = v / s
    if L is None:
        L = _pw2(2 * n - 1)
    A = np.fft.rfft(v, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    mx = float(c.max())
    if not np.isfinite(mx):
        return np.inf
    if mx < 0:
        mx = 0.0
    return 2.0 * float(n) * mx


def _convolve_autocorr(a, L):
    A = np.fft.rfft(a, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * len(a) - 1]
    return np.maximum(c, 0.0)


def _gen_one_sided(n, p, off):
    j = np.arange(n, dtype=np.float64)
    v = np.power(np.maximum(j + off, 1e-8), -p)
    return _project_simplex(v)


def _gen_kinked(n, p1, p2, frac):
    j = np.arange(n, dtype=np.float64)
    cut = int(frac * n)
    v = np.empty(n)
    v[:cut] = np.power(np.maximum(j[:cut] + 1.0, 1e-8), -p1)
    v[cut:] = np.power(np.maximum(j[cut:] - cut + 1.0, 1e-8), -p2)
    return _project_simplex(v)


def _gen_two_sided(n, p, off):
    j = np.arange(n, dtype=np.float64)
    left = np.power(np.maximum(j + off, 1e-8), -p)
    right = np.power(np.maximum(n - 1.0 - j + off, 1e-8), -p)
    v = left + right
    return _project_simplex(v)


def _gen_gaussian(n, sigma_frac, center_frac):
    j = np.arange(n, dtype=np.float64)
    center = center_frac * (n - 1)
    sigma = sigma_frac * (n - 1)
    v = np.exp(-0.5 * ((j - center) / max(sigma, 1.0)) ** 2)
    return _project_simplex(v)


def _gen_cosine(n, q):

    t = (j + 0.5) / n
    v = np.power(np.cos(np.pi * t / 2.0), q)
    v = np.maximum(v, 0.0)
    return _project_simplex(v)

def _ratios_from_a(a):
    a = _clean(a).copy()
    n = a.size
    if n == 0:
        return a
    s = float(a.sum())
    if s <= 0 or not np.isfinite(s):
        a = np.full(n, 1.0 / float(n))
        s = 1.0
    else:
        a = a / s
    if n == 1:
        return a
    ref = float(a[-1])
    if ref <= 1e-200:
        a[-1] = 1e-12
        a = a / float(a.sum())
        ref = float(a[-1])
    x = np.log(np.maximum(a[:-1], 1e-300) / max(ref, 1e-300))
    return np.clip(x, -100.0, 100.0)


def _a_from_ratios(x):
    x = np.asarray(x, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    n = x.size + 1
    if n == 1:
        return np.ones(1)
    m = float(np.max(x)) if x.size else 0.0
    e = np.exp(np.clip(x - m, -745.0, 709.0))
    e_last = float(np.exp(-m))
    s = float(e.sum()) + e_last
    if not np.isfinite(s) or s <= 0:
        return np.full(n, 1.0 / n)
    a = np.empty(n)
    a[:-1] = e / s
    a[-1] = e_last / s
    return a


def _soft_obj_grad(x, eps, L):
    a = _a_from_ratios(x)
    n = a.size
    if n <= 1:
        return 0.0, np.zeros_like(x)
    A = np.fft.rfft(a, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    cmax = float(c.max())
    if cmax <= 1e-300:
        return 0.0, np.zeros_like(x)
    if eps <= 1e-300:
        eps = 1e-12
    z = (c - cmax) / eps
    z = np.clip(z, -745.0, 0.0)
    e = np.exp(z)
    sume = float(e.sum())
    if not np.isfinite(sume) or sume <= 0:
        return 1e300, np.zeros_like(x)
    f = cmax + eps * float(np.log(sume))
    w = e / sume
    W = np.fft.rfft(w, n=L)
    ga = 2.0 * np.fft.irfft(W * np.conj(A), n=L)[:n]
    dot = float(np.dot(a, ga))
    gfull = a * (ga - dot)
    gy = gfull[:-1]
    if not np.all(np.isfinite(gy)):
        gy = np.zeros_like(x)
    if not np.isfinite(f):
        f = 1e300
    return f, gy

def _run_lbfgs(fun, x, args, maxiter, bounds):

    x = np.asarray(x, dtype=np.float64)
    if x.size == 0:
        return x, 1
    options = {"maxiter": int(maxiter), "disp": 0, "maxfun": int(max(200, 15 * maxiter)),
               "ftol": 1e-16, "gtol": 1e-10}
    try:
        res = _minimize(fun, x, args=args, method="L-BFGS-B", jac=True,
                        bounds=bounds, options=options)
    except Exception:
        return x, 99
    x_new = np.asarray(res.x, dtype=np.float64)
    x_new = np.where(np.isfinite(x_new), x_new, 0.0)
    x_new = np.clip(x_new, -100.0, 100.0)
    try:
        status = int(res.status)
    except Exception:
        status = 0
    return x_new, status


def _subgradient_nesterov(a0, end_time, seed=L):
    a = _project_simplex(a0)
    n = a.size
    Ls = _pw2(2 * n - 1)
    Lw = _pw2(2 * n - 1)
    best = a.copy()
    best_score = _score(best, Ls)
    rng = np.random.default_rng(seed)

    # Nesterov state
    y = a.copy()
    v = np.zeros(n)
    beta = 0.9
    lr = 0.03
    t = 0
    last_best = time.monotonic()
    start = time.monotonic()

    while time.monotonic() < end_time:
        t += 1
        # Nesterov momentum
        beta_t = (1.0 + 0.5 * np.sqrt(1.0 + 4.0 * beta * beta * t)) / (2.0 * (1.0 + beta * t))
        y_new = y + beta_t * (y - a)

        # Compute autocorrelation at y
        c = _convolve_autocorr(y, Lw)
        cmax = float(c.max())
        if cmax < 1e-15:
            break

        # Choose weights: sharp softmax on top values
        # Use a small epsilon to focus on the max
        eps_w = max(1e-6, cmax * 0.001)
        z = (c - cmax) / eps_w
        z = np.clip(z, -745.0, 0.0)
        e = np.exp(z)
        sume = float(e.sum())
        if sume < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / sume

        # Compute gradient: G_i = sum_k w_k * a_{k-i} (via FFT)
        W = np.fft.rfft(w, n=Lw)
        A = np.fft.rfft(y, n=Lw)
        G = np.fft.irfft(W * np.conj(A), n=Lw)[:n]
        G *= 2.0  # derivative factor

        # Project gradient onto simplex tangent (remove mean to preserve sum)
        G -= float(G.mean())

        # Adaptive learning rate
        gnorm = float(np.sqrt(np.mean(G * G))) + 1e-12
        G = G / gnorm

        # Step
        a_new = _project_simplex(y_new - lr * G)

        score = _score(a_new, Ls)
        now = time.monotonic()

        if np.isfinite(score) and score < best_score - 1e-14:
            best = a_new.copy()
            best_score = score
            last_best = now
            lr = min(0.1, lr * 1.02)
        else:
            # Decay learning rate if no improvement
            if now - last_best > 3.0:
                lr = max(0.001, lr * 0.9)
            if now - last_best > 15.0:
                # Reset with perturbation from best
                a_new = _project_simplex(best * np.exp(rng.normal(0, 0.03, n)))
                y = a_new.copy()
                v = np.zeros(n)
                t = 0
                lr = 0.02

        y = a_new.copy()

    return best, best_score

    print(f"Generated {len(cands)} candidates, best initial score: {scored[0][0]:.8f}")

    if not scored:
        a = np.full(n, 1.0 / float(n))
        return [float(x) for x in a]

    global_best = np.asarray(scored[0][2], dtype=np.float64).copy()
    global_best_score = float(scored[0][0])

    # Select top candidates for refinement
    top_k = min(5, len(scored))
    top = scored[:top_k]

    # Allocate time: give more time to better starting points
    total_refine_time = budget * 0.85
    remaining_check = deadline - total_refine_time

    # Time allocation: proportional to quality (better candidates get more time)
    # But ensure at least some time for the 2nd and 3rd best
    fracs = []
    weights = []
    for i in range(top_k):
        # Weight by how much better it is than the worst
        gap = scored[-1][0] - scored[i][0]
        w = 1.0 + 5.0 * max(0, gap) / (scored[-1][0] + 1e-12)
        weights.append(w)
    total_w = sum(weights)
    for w in weights:
        fracs.append(w / total_w)

    time_cursor = t0
    for i in range(top_k):
        alloc_end = time_cursor + fracs[i] * total_refine_time
        alloc_end = min(alloc_end, deadline - max(5.0, budget * 0.05))
        if alloc_end - time_cursor < 5.0:
            break
        a0 = np.asarray(top[i][2], dtype=np.float64).copy()
        print(f"\nRefining candidate {i} (initial score={top[i][0]:.8f}), "
              f"time={alloc_end - time_cursor:.1f}s")
        a, sc = _refine_start(a0, alloc_end, int(seed) + 1000 * (i + 1))
        if np.isfinite(sc) and sc < global_best_score:
            global_best_score = sc
            global_best = np.asarray(a, dtype=np.float64).copy()
            print(f"  NEW GLOBAL BEST: {global_best_score:.8f}")
        time_cursor = alloc_end
        if time_cursor >= deadline - 5.0:
            break

    # Final polish
    if time.monotonic() < deadline - 2.0:
        a, sc = _local_search(global_best, deadline - 3.0, int(seed) + 77777)
        if np.isfinite(sc) and sc < global_best_score:
            global_best_score = sc
            global_best = np.asarray(a, dtype=np.float64).copy()
            print(f"\nFinal polish: {global_best_score:.8f}")

    global_best = _project_simplex(global_best)
    out = [float(x) for x in global_best]

    # Validate
    try:
        val = evaluate_sequence(out)
        print(f"\nFinal evaluation: {val:.8f}")
        if not np.isfinite(val):
            out = [float(x) for x in _gen_one_sided(len(out), 0.5, 1.0)]
    except Exception as e:
        print(f"Validation error: {e}")
        # Try to validate manually
        try:
            arr = np.array(out)
            s = arr.sum()
            if s < 0.01:
                out = [float(x) for x in _gen_one_sided(len(out), 0.5, 1.0)]
        except Exception:
            out = [1.0] * n

    print(f"Total time used: {time.monotonic() - t0:.1f}s")
    return out
