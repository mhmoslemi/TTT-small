    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]

    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a)
    return 2.0 * n * np.max(c) / (s * s)

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]


def _score_fast(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a)
    return 2.0 * n * float(np.max(c)) / (s * s)


def _fienup_flat(a_init, n_iter, alpha, deadline, verbose=False):
    """Fienup-style iteration: flatten the magnitude spectrum to reduce peak correlation."""
    n = len(a_init)

    s = a.sum()

        return a_init, _score_fast(a_init)
    a = a / s

    best_score = _score_fast(a)
    it = 0

        a_pad = np.zeros(L)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)

        mag = np.abs(A)
        phase = np.angle(A)

        if mag.max() < 1e-30:
            break

        # Compress non-DC bins toward their geometric mean
        mag_nonzero = mag[1:].copy()
        if mag_nonzero.max() > 1e-30:
            # Geometric mean of non-DC bins
            log_mean = np.mean(np.log(np.maximum(mag_nonzero, 1e-30)))
            C = np.exp(log_mean)
            # Interpolate: new_mag = mag^alpha * C^(1-alpha)

    block_size = max(1, n // 100)
    it = 0
    while time.monotonic() < deadline and it < n_perturbs:
        start = int(rng.integers(0, n - block_size))
        end = min(start + block_size, n)

        a_trial = a.copy()
        scale = np.exp(rng.normal(0, 0.1, end - start))
        a_trial[start:end] *= scale
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-15:
            it += 1
            continue
        a_trial = a_trial / s

        sc = _score_fast(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial.copy()
        it += 1

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Minimizes 2*n*max(conv(a,a))/sum(a)^2 using Fienup spectral flattening,
    smooth-max subgradient, and local search. Tries multi-frequency initializations
    and the provided sequence across multiple lengths.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 20.0

    best_overall = None
    best_overall_score = np.inf

    def update_best(a, label=""):
        nonlocal best_overall, best_overall_score
        sc = _score_fast(a)
        if sc < best_overall_score:
            best_overall_score = sc
            best_overall = a.copy()
            print(f"  *** NEW BEST ({label}): {sc:.6f}, n={len(a)} ***")
        return sc

    # Build initializations
    inits = []

    # Provided sequence
    try:
        hs = np.array(height_sequence_1, dtype=np.float64)
        hs = np.clip(hs, 0, None)
        if hs.size > 100 and hs.sum() > 0.01:
            inits.append(("provided", hs.copy()))
    except (NameError, Exception):
        pass

    # Single sinusoid with high amplitude (near the A=1 limit)
    for n in [2000, 3000, 4000]:
        x = np.arange(n, dtype=np.float64)
        for A in [0.8, 0.95, 1.0]:
            for phi in [0.0, np.pi / 4, np.pi / n]:
                a = 1.0 + A * np.sin(2.0 * np.pi * x / n + phi)
                a = np.maximum(a, 1e-6)
                inits.append((f"sin1_{n}_A{A}_p{int(phi*100)}", a))

    # Two-frequency sinusoid
    for n in [2000, 3000, 4000]:
        x = np.arange(n, dtype=np.float64)
        for f2 in [2, 3, 5]:
            for A1, A2 in [(0.6, 0.3), (0.8, 0.2), (0.5, 0.4)]:
                a = 1.0 + A1 * np.sin(2.0 * np.pi * x / n) + A2 * np.sin(2.0 * np.pi * f2 * x / n + 0.5)
                a = np.maximum(a, 1e-6)
                inits.append((f"sin2_{n}_f{f2}_{A1}_{A2}", a))

    # Half-cosine (cos^2 taper)
    for n in [2000, 3000, 4000]:
        x = np.arange(n, dtype=np.float64)
        a = np.cos(np.pi * x / (2.0 * n)) ** 2
        a = np.maximum(a, 1e-6)
        inits.append((f"cos2_{n}", a))

    # Exponential decay (from before, worked well)
    for n in [2000, 4000]:
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-0.8 * x / n) + 0.1
        inits.append((f"decay_{n}", a))

    # Sinusoidal (from before)
    for n in [2000, 4000]:
        x = np.arange(n, dtype=np.float64)
        a = 1.0 + 0.4 * np.sin(2.0 * np.pi * x / n + np.pi / 4)
        a = np.maximum(a, 0.05)
        inits.append((f"sin04_{n}", a))

    # Random non-negative
    for n in [2000, 4000]:
        a = np.abs(rng.normal(0, 1, n)) + 0.1
        inits.append((f"rand_{n}", a))

    print(f"Total initializations: {len(inits)}")

    # Quick score all inits
    scored = []
    for name, a0 in inits:
        a0 = np.maximum(a0, 0)
        s = a0.sum()
        if s < 1e-15:
            continue
        a0n = a0 / s
        sc = _score_fast(a0n)
        scored.append((sc, name, a0n))

    scored.sort(key=lambda x: x[0])
    print("Top initializations:")
    for i, (sc, name, _) in enumerate(scored[:10]):
        print(f"  {i+1}. {name}: {sc:.6f}")

    # Phase 1: Fienup on top inits
    print("\n=== Phase 1: Fienup Spectral Flattening ===")
    t_phase1_end = min(total_start + budget_s * 0.35, deadline)
    n_top_fienup = min(len(scored), 12)

    for i in range(n_top_fienup):
        if time.monotonic() > t_phase1_end:
            break
        sc_init, name, a0 = scored[i]
        t_alloc = max(5.0, (t_phase1_end - time.monotonic()) / (n_top_fienup - i))
        t_alloc = min(t_alloc, 120.0)

        print(f"  Fienup on '{name}' (n={len(a0)}, init={sc_init:.6f}, t={t_alloc:.0f}s)")

        # Try different alpha values
        for alpha in [0.3, 0.5, 0.7]:
            if time.monotonic() > t_phase1_end:
                break
            t_sub = t_alloc / 3.0
            a_f, sc_f = _fienup_flat(a0, n_iter=10000, alpha=alpha, deadline=time.monotonic() + t_sub)
            update_best(a_f, f"fienup_a{alpha}_{name}")

        # Try spectral floor
        for floor_f in [0.1, 0.3]:
            if time.monotonic() > t_phase1_end:
                break
            t_sub = t_alloc / 3.0
            a_f, sc_f = _fienup_floor(a0, n_iter=10000, floor_frac=floor_f, deadline=time.monotonic() + t_sub)
            update_best(a_f, f"floor{floor_f}_{name}")

    # Phase 2: Subgradient refinement on best found
    print(f"\n=== Phase 2: Subgradient Refinement (best so far: {best_overall_score:.6f}) ===")
    t_phase2_end = min(total_start + budget_s * 0.65, deadline)

    # Use the best from Fienup and also try the best init directly
    candidates_for_subgrad = []
    if best_overall is not None:
        candidates_for_subgrad.append(("best_fienup", best_overall.copy()))
    candidates_for_subgrad.append(("best_init", scored[0][2].copy()))
    if len(scored) > 1:
        candidates_for_subgrad.append(("init2", scored[1][2].copy()))

    n_cands = len(candidates_for_subgrad)
    for i in range(n_cands):
        if time.monotonic() > t_phase2_end:
            break
        name, a0 = candidates_for_subgrad[i]
        t_alloc = max(5.0, (t_phase2_end - time.monotonic()) / (n_cands - i))
        t_alloc = min(t_alloc, 300.0)

        print(f"  Subgradient on '{name}' (n={len(a0)}, t={t_alloc:.0f}s)")

        # Direct subgradient
        t_sub = t_alloc * 0.5
        a_sg, sc_sg = _subgradient_direct(a0, t_sub, rng)
        update_best(a_sg, f"subgrad_{name}")

        # Smooth subgradient
        if time.monotonic() < t_phase2_end:
            t_sub2 = min(t_alloc * 0.5, t_phase2_end - time.monotonic())
            a_ss, sc_ss = _subgradient_smooth(a0, t_sub2, rng)
            update_best(a_ss, f"smooth_{name}")

    # Phase 3: Local search on best
    print(f"\n=== Phase 3: Local Search (best so far: {best_overall_score:.6f}) ===")
    t_phase3_end = min(total_start + budget_s * 0.85, deadline)

    if best_overall is not None and time.monotonic() < t_phase3_end:
        t_alloc = t_phase3_end - time.monotonic()
        print(f"  Local search with {t_alloc:.0f}s")
        a_ls, sc_ls = _local_search(best_overall, t_alloc, rng)
        update_best(a_ls, "local_search")

    # Phase 4: Final Fienup + subgradient on the best
    print(f"\n=== Phase 4: Final Polishing (best so far: {best_overall_score:.6f}) ===")
    if best_overall is not None and time.monotonic() < deadline:
        t_remaining = deadline - time.monotonic()

        # One more Fienup pass
        t_f = t_remaining * 0.3
        a_f2, sc_f2 = _fienup_flat(best_overall, n_iter=20000, alpha=0.5, deadline=time.monotonic() + t_f)
        update_best(a_f2, "final_fienup")

        # One more subgradient pass
        if time.monotonic() < deadline:
            t_s = deadline - time.monotonic() - 5.0
            if t_s > 10:
                a_s2, sc_s2 = _subgradient_direct(best_overall, t_s, rng)
                update_best(a_s2, "final_subgrad")

    # Final verification
    if best_overall is not None:
        final_score = evaluate_sequence(best_overall.tolist())
        elapsed = time.monotonic() - total_start
        print(f"\nFinal result: score={final_score:.6f}, n={len(best_overall)}, elapsed={elapsed:.1f}s")
        return [float(x) for x in best_overall]
    else:
        # Fallback: high-amplitude sinusoid
        n = 2000
        x = np.arange(n, dtype=np.float64)
        a = np.maximum(1.0 + 0.95 * np.sin(2.0 * np.pi * x / n + np.pi / 4), 1e-6)
        print(f"Fallback: score={_score_fast(a):.6f}")
        return a.tolist()
