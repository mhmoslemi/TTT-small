"""
Search for low-peak self-convolution sequences.

Main ideas:
- Work with probability vectors p = a / sum(a), since the objective is
  scale-invariant (as long as no value is clipped by the evaluator).
- Use FFT-based exact convolution for a fast objective and a focused
  top-lag gradient for multiplicative / log-domain Adam updates.
- Generate diverse low-dimensional Chebyshev / trigonometric starts.
- Run several local refinements and keep the best sequence found.
"""

import time
import numpy as np


def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def normalize_positive(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s


def softmax_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return np.zeros(0, dtype=np.float64)
    z = np.where(np.isfinite(z), z, 0.0)
    z = z - float(z.max())
    z = np.clip(z, -50.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(z.size, dtype=np.float64) / float(z.size)
    return e / s


def log_to_z(p):
    p = np.asarray(p, dtype=np.float64)
    z = np.log(np.maximum(p, 1e-300))
    z = np.where(np.isfinite(z), z, 0.0)
    z = z - float(z.mean())
    return np.clip(z, -30.0, 30.0)


def score_fast(p, L):
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return np.inf
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.inf
    if abs(s - 1.0) > 1e-12:
        p = p / s
    P = np.fft.rfft(p, L)
    c = np.fft.irfft(P * P, L)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    cmax = float(c.max())
    if not np.isfinite(cmax) or cmax <= 0.0:
        return np.inf
    return float(2.0 * n * cmax)


def resize_sequence(seq, m):
    p = normalize_positive(seq)
    m = int(m)
    if p is None or m <= 0:
        return None
    n = p.size
    if n == m:
        return p.copy()
    if n == 1:
        return np.full(m, 1.0 / float(m), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, m)
    out = np.interp(x_new, x_old, p)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(m, dtype=np.float64) / float(m)
    return out / s


def build_basis(N, D):
    B = np.zeros((N, max(0, int(D))), dtype=np.float64)
    if N <= 0 or D <= 0:
        return B
    x = (np.arange(N, dtype=np.float64) + 0.5) / float(N)
    t = 2.0 * x - 1.0

    n_cheb = min(int(D), 16)
    n_trig = int(D) - n_cheb
    j = 0

    if n_cheb >= 1:
        col = t.copy()
        col -= col.mean()
        B[:, j] = col
        j += 1
        Tprev = np.ones(N, dtype=np.float64)
        Tcurr = t
        for m in range(2, n_cheb + 1):
            col = 2.0 * t * Tcurr - Tprev
            col -= col.mean()
            B[:, j] = col
            j += 1
            Tprev = Tcurr
            Tcurr = col

    r = 1
    while j < D and n_trig > 0:
        col = np.cos(r * np.pi * x)
        col -= col.mean()
        B[:, j] = col
        j += 1
        r += 2
        n_trig -= 1

    if j < D:
        Tprev = np.ones(N, dtype=np.float64)
        Tcurr = t
        m = 1
        while j < D:
            if m == 1:
                col = t
            elif m == 2:
                col = 2.0 * t * t - 1.0
            else:
                col = 2.0 * t * Tcurr - Tprev
            col -= col.mean()
            B[:, j] = col
            j += 1
            m += 1
            Tprev = Tcurr
            Tcurr = col

    return B


def smooth_noise_p(p, rng, scale=0.15, D=12):
    p = normalize_positive(p)
    if p is None:
        return None
    N = p.size
    if N <= 0 or D <= 0:
        return p.copy()
    t = np.linspace(-1.0, 1.0, N)
    noise = np.zeros(N, dtype=np.float64)
    Tprev = np.ones(N, dtype=np.float64)
    Tcurr = t
    for m in range(1, D + 1):
        if m == 1:
            col = t
        elif m == 2:
            col = 2.0 * t * t - 1.0
        else:
            col = 2.0 * t * Tcurr - Tprev
        noise = noise + float(rng.normal(0.0, 1.0)) * col
        Tprev = Tcurr
        Tcurr = col
    z = log_to_z(p) + float(scale) * noise
    z = np.clip(np.where(np.isfinite(z), z, 0.0), -30.0, 30.0)
    return softmax_z(z)


def full_grad_top(p, K, tau_ratio, L):
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return np.inf, np.zeros(0, dtype=np.float64), 0.0
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.inf, np.zeros(n, dtype=np.float64), 0.0
    p = p / s

    m = 2 * n - 1
    K = int(K)
    if K < 1:
        K = 1
    if K > m:
        K = m

    P = np.fft.rfft(p, L)
    c = np.fft.irfft(P * P, L)[:m]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    M = float(c.max())
    if not np.isfinite(M) or M <= 0.0:
        return np.inf, np.zeros(n, dtype=np.float64), 0.0

    true_obj = float(2.0 * n * M)

    if K >= m:
        idx = np.arange(m)
    else:
        idx = np.argpartition(c, m - K)[-K:]

    vals = c[idx]
    vM = float(vals.max())
    tau = max(1e-300, float(tau_ratio) * M) if tau_ratio > 0.0 else max(1e-300, M)
    z = (vals - vM) / tau
    z = np.clip(z, -50.0, 0.0)
    e = np.exp(z)
    es = float(e.sum())
    if not np.isfinite(es) or es <= 0.0:
        e = np.zeros_like(vals)
        e[int(np.argmax(vals))] = 1.0
        es = 1.0
    wv = e / es

    # For a moderate active set, accumulate shifts directly.  For larger
    # active sets, use another FFT to avoid many Python slice operations.
    use_fft = K > 512 or n > 16384
    if use_fft:
        if K >= m:
            wfull = e / es
        else:
            wfull = np.zeros(m, dtype=np.float64)
            wfull[idx] = wv
            sfull = float(wfull.sum())
            if not np.isfinite(sfull) or sfull <= 0.0:
                wfull = np.zeros(m, dtype=np.float64)
                wfull[int(idx[int(np.argmax(vals))])] = 1.0
                sfull = 1.0
            else:
                wfull = wfull / sfull
        W = np.fft.rfft(wfull, L)
        H = np.fft.irfft(W * np.conj(P), L)
        h = H[:n]
    else:
        h = np.zeros(n, dtype=np.float64)
        for pos in range(idx.size):
            kk = int(idx[pos])
            ww = float(wv[pos])
            if ww <= 1e-16:
                continue
            i0 = kk - n + 1
            if i0 < 0:
                i0 = 0
            i1 = kk
            if i1 >= n:
                i1 = n - 1
            if i0 > i1:
                continue
            j0 = kk - i1
            j1 = kk - i0
            h[i0:i1 + 1] += ww * p[j0:j1 + 1][::-1]

    h = np.where(np.isfinite(h), h, 0.0)
    h = np.maximum(h, 0.0)
    grad_p = 2.0 * h
    mean_g = float(np.dot(p, grad_p))
    grad_z = p * (grad_p - mean_g)
    grad_z = np.where(np.isfinite(grad_z), grad_z, 0.0)
    return true_obj, grad_z, M


def refine_top(p_start, duration_s, rng, L, K=384, tau_ratio=0.09, lr=0.20,

    if duration_s <= 0.0:
        p = normalize_positive(p_start)
        if p is None:
            return None, np.inf
        return p.copy(), float(score_fast(p, L))

    p = normalize_positive(p_start)
    if p is None:
        N = int(np.asarray(p_start).size)
        p = np.ones(max(1, N), dtype=np.float64) / float(max(1, N))
    N = p.size
    best_p = p.copy()
    best_s = float(score_fast(p, L))
    if not np.isfinite(best_s):
        best_p = np.ones(N, dtype=np.float64) / float(N)
        best_s = 2.0

    z = log_to_z(p)
    m = np.zeros(N, dtype=np.float64)
    v = np.zeros(N, dtype=np.float64)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step = 0
    no_improve = 0
    Kcur = int(K)
    tau_cur = float(tau_ratio)
    lr_cur = float(lr)
    deadline = time.monotonic() + duration_s

    print(f"refine_top start N={N} score={best_s:.6f} K={Kcur} tau={tau_cur:.4f} lr={lr_cur:.4f}")

    while time.monotonic() < deadline:

        if not np.isfinite(true_obj):
            break

        if step > 0 and true_obj > best_s + 0.03:
            p = best_p.copy()
            z = log_to_z(p)
            m *= 0.1
            v *= 0.1
            no_improve = 0
            continue

        if true_obj < best_s - 1e-12:
            best_s = float(true_obj)
            best_p = p.copy()

                s2 = float(score_fast(p2, L)) if p2 is not None else np.inf
            if not np.isfinite(s2):
                p2 = best_p.copy()
                s2 = best_s
            if s2 < best_s:
                best_s = float(s2)
                best_p = p2.copy()
            p = p2.copy()
            z = log_to_z(p)
            m *= 0.1
            v *= 0.1
            no_improve = 0
            Kcur = int(rng.choice([128, 256, 512, 1024]))
            tau_cur = float(rng.choice([0.03, 0.06, 0.10, 0.20, 0.40]))
            lr_cur = float(rng.uniform(0.08, 0.30))
            continue

        if no_improve > 1000:
            lr_cur = max(0.04, lr_cur * 0.9)
            if tau_cur < 0.20:
                tau_cur = min(0.30, tau_cur * 1.2)
            else:
                tau_cur = max(0.02, tau_cur * 0.8)
            if step % 1000 == 0:
                if Kcur < 1024:
                    Kcur = int(min(1024, Kcur * 2))
                elif Kcur > 256 and rng.random() < 0.5:
                    Kcur = int(max(256, Kcur * 0.6))

        step += 1
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        b1 = 1.0 - beta1 ** step
        b2 = 1.0 - beta2 ** step
        upd = (m / b1) / (np.sqrt(v / b2) + eps)
        rms = float(np.sqrt(np.mean(upd * upd)))
        if not np.isfinite(rms) or rms <= 1e-15:
            lr_cur = max(0.02, lr_cur * 0.5)
            continue

        z_new = z - (lr_cur / rms) * upd
        z_new = np.where(np.isfinite(z_new), z_new, 0.0)
        z_new = z_new - float(z_new.mean())
        z_new = np.clip(z_new, -30.0, 30.0)
        p_new = softmax_z(z_new)
        sc_new = float(score_fast(p_new, L))

        if step % 1000 == 0:
            psym = 0.5 * (p_new + p_new[::-1])
            ssym = float(score_fast(psym, L))
            if np.isfinite(ssym):
                if ssym < best_s - 1e-12:
                    best_s = ssym
                    best_p = psym.copy()
                    no_improve = 0
                if ssym < sc_new:
                    p_new = psym
                    z_new = log_to_z(p_new)
                    sc_new = ssym

        if np.isfinite(sc_new) and sc_new <= true_obj + 1e-9 + 1e-4:
            z = z_new
            p = p_new
            if sc_new < best_s - 1e-12:
                best_s = float(sc_new)
                best_p = p.copy()
                no_improve = 0
        else:
            lr_cur = max(0.02, lr_cur * 0.75)

        if step % 2000 == 0:
            print(f"refine step={step} best={best_s:.6f} cur={true_obj:.6f} "
                  f"lr={lr_cur:.3f} K={Kcur} tau={tau_cur:.4f}")

    return best_p, float(best_s)


def add_candidate(p, s, cand_list, max_keep=8):
    if p is None or not np.isfinite(s):
        return
    arr = normalize_positive(p)
    if arr is None:
        return
    if len(cand_list) < max_keep:
        cand_list.append((float(s), arr.copy()))
        return
    worst = 0
    worst_s = float(cand_list[0][0])
    for i in range(1, len(cand_list)):
        if float(cand_list[i][0]) > worst_s:
            worst_s = float(cand_list[i][0])
            worst = i
    if float(s) < worst_s:
        cand_list[worst] = (float(s), arr.copy())


def is_new_start(p, starts, threshold=1e-3):
    if p is None:
        return False
    p = np.asarray(p, dtype=np.float64)
    for q, _s in starts:
        q = np.asarray(q, dtype=np.float64)
        if p.size != q.size:
            continue
        d = float(np.linalg.norm(p - q))
        if d < threshold:
            return False
    return True


def add_start(p, s, starts, threshold=1e-3):
    if p is None or not np.isfinite(s):
        return
    p = normalize_positive(p)
    if p is None:
        return
    if len(starts) >= 3:
        worst = 0
        worst_s = float(starts[0][1])
        for i in range(1, len(starts)):
            if float(starts[i][1]) > worst_s:
                worst_s = float(starts[i][1])
                worst = i
        if float(s) < worst_s - 1e-8:
            others = []
            for i in range(len(starts)):
                if i != worst:
                    others.append(starts[i])
            if is_new_start(p, others, threshold):
                starts[worst] = (p.copy(), float(s))
        return
    if is_new_start(p, starts, threshold):
        starts.append((p.copy(), float(s)))


def choose_params(N, s0):
    if N <= 2048:
        K = 256
        tau = 0.12
        lr = 0.25
    elif N <= 4096:
        K = 384
        tau = 0.10
        lr = 0.22
    elif N <= 8192:
        K = 512
        tau = 0.09
        lr = 0.21
    else:
        K = 384
        tau = 0.09
        lr = 0.20

    if np.isfinite(s0) and s0 > 1.85:
        K = int(min(2048, max(K, int(K * 1.8))))
        tau = max(0.12, tau * 1.5)
        lr = 0.28
    return K, float(tau), float(lr)


def propose_candidate(seed=42, budget_s=2400.0):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not np.isfinite(budget) or budget <= 0.0:
        budget = 2400.0
    deadline = t0 + max(30.0, budget - 30.0)

    if budget >= 1500.0:
        N = 16384
    elif budget >= 600.0:
        N = 8192
    else:
        N = 4096

    L = next_pow2(2 * N - 1)
    candidates = []

    # Uniform baseline.
    p_uni = np.ones(N, dtype=np.float64) / float(N)
    s_uni = float(score_fast(p_uni, L))
    add_candidate(p_uni, s_uni, candidates)

    # Previous construction, if present.
    prev_p = None
    prev_s = np.inf
    try:
        hs = height_sequence_1
        p_prev = resize_sequence(hs, N)
        if p_prev is not None:
            s_prev = float(score_fast(p_prev, L))
            prev_p = p_prev
            prev_s = s_prev
            add_candidate(p_prev, s_prev, candidates)
    except Exception:
        pass

    # Low-dimensional diverse starts: single Chebyshev/trig modes and random
    # low-order combinations.
    if time.monotonic() < deadline - 240.0:
        D = 32
        try:
            B = build_basis(N, D)
            scales = 1.0 / np.sqrt(np.arange(1, D + 1, dtype=np.float64))

            for j in [0, 1, 2, 3, 4, 5, 6, 7]:
                if j >= D:
                    break
                for a in np.linspace(-3.0, 3.0, 7):
                    z = (float(a) * float(scales[j])) * B[:, j]
                    p = softmax_z(z)
                    s = float(score_fast(p, L))
                    add_candidate(p, s, candidates)

            for _ in range(40):
                theta = rng.normal(0.0, 1.0, D) * scales
                z = B.dot(theta)
                p = softmax_z(z)
                s = float(score_fast(p, L))
                add_candidate(p, s, candidates)
        except Exception:
            pass

    starts = []
    if candidates:
        scores = np.array([float(s) for s, _p in candidates], dtype=np.float64)
        order = np.argsort(scores)

        best_idx = int(order[0])
        add_start(candidates[best_idx][1], float(candidates[best_idx][0]), starts)

        if prev_p is not None and np.isfinite(prev_s):
            add_start(prev_p, float(prev_s), starts)

        for pos in range(1, len(order)):
            if len(starts) >= 3:
                break
            idx = int(order[pos])
            add_start(candidates[idx][1], float(candidates[idx][0]), starts)

    if len(starts) < 3:
        add_start(p_uni, s_uni, starts)
    if not starts:
        starts.append((p_uni.copy(), float(s_uni)))

    best_p, best_s = starts[0]
    if not np.isfinite(best_s):
        best_s = 2.0
        best_p = p_uni.copy()
    best_p = np.asarray(best_p, dtype=np.float64).copy()

    print(f"initial best N={N} score={best_s:.6f}")

    for idx, (p0, s0) in enumerate(starts):
        rem = deadline - time.monotonic()
        if rem < 90.0:
            break
        if len(starts) == 1:
            dur = max(30.0, rem - 25.0)
        elif idx == 0:
            dur = max(60.0, (rem - 30.0) * 0.58)
        elif idx == 1:
            dur = max(60.0, (rem - 30.0) * 0.28)
        else:
            dur = max(60.0, (rem - 30.0) * 0.14)
        dur = min(float(dur), max(30.0, rem - 20.0))

        K0, tau0, lr0 = choose_params(N, s0)
        try:
            p2, s2 = refine_top(p0, dur, rng, L, K=K0, tau_ratio=tau0, lr=lr0)
        except Exception:
            p2, s2 = None, np.inf

        if p2 is not None and np.isfinite(s2) and float(s2) < float(best_s):
            best_p = np.asarray(p2, dtype=np.float64).copy()
            best_s = float(s2)
            print(f"new global best score={best_s:.6f}")

    # Final safe normalization / clipping.
    arr = np.asarray(best_p, dtype=np.float64).ravel()
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    arr = np.minimum(arr, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 0.0:
        arr = np.ones(1024, dtype=np.float64) / 1024.0
        s = 1.0
    arr = arr / s

    # If the number is small, validate with the real evaluator just in case.
    rem = deadline - time.monotonic()
    if rem > 5.0 and arr.size <= 4096:
        try:
            real = evaluate_sequence([float(x) for x in arr])
            if np.isfinite(real):
                if real < float(best_s):
                    best_s = float(real)
        except Exception:
            pass

    return [float(x) for x in arr]
