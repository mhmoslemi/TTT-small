    # Find active peaks (top-k or those within tolerance)
    tol = max(1e-12, M * rel_tol)
    active = np.where(c >= M - tol)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    elif len(active) > top_k:
        # Keep only top-k
        idx = np.argsort(c[active])[-top_k:]
        active = active[idx]
    
    g = np.zeros(n, dtype=np.float64)
    for k in active:
        k = int(k)
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_hi >= i_lo:
            p_idx = k - i_hi
            g[i_lo:i_hi + 1] += 2.0 * p[p_idx:k - i_lo + 1][::-1]
    
    g /= float(len(active))
    return g


def project_subgradient(p, g):
    """Project subgradient onto tangent space of simplex (sum constraint)."""
    return g - p * np.dot(p, g)


def make_ramp(n, direction=1):
    """Linear ramp sequence."""
    if direction == 1:
        x = np.linspace(0.0, 1.0, n)
    else:
        x = np.linspace(1.0, 0.0, n)
    p = x + 0.01
    return p / p.sum()


def make_u_shape(n, power=2.0, asym=0.0):
    """U-shaped sequence with optional asymmetry."""
    x = np.linspace(0.0, 1.0, n)
    x = x + asym
    x = np.clip(x, -0.5, 1.5)
    p = np.abs(x) ** power + np.abs(1.0 - x) ** power + 0.05
    return p / p.sum()


def make_poly(n, coeffs):
    """Polynomial-shaped sequence."""
    x = np.linspace(0.0, 1.0, n)
    p = np.ones(n)
    for c in coeffs:
        p = p * (1.0 + c * x)
    p = np.maximum(p, 1e-10)
    return p / p.sum()


def make_gpsr_init(n, rng, dim=16):
    """Random low-dimensional parameterized initialization."""
    x = np.linspace(0.0, 1.0, n)
    theta = rng.normal(0, 0.8, dim)
    p = np.ones(n)
    for i, t in enumerate(theta):
        p = p + t * np.cos((i + 1) * np.pi * x)
    p = np.maximum(p, 1e-8)
    return p / p.sum()


def resize_sequence(seq, m):
    """Resize sequence via interpolation, maintaining normalization."""
    p = np.asarray(seq, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s <= 0:
        return np.ones(m) / m
    p = p / s
    n = p.size
    if n == m:
        return p.copy()
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, m)
    out = np.interp(x_new, x_old, p)
    out = np.maximum(out, 0.0)
    s2 = out.sum()
    if s2 <= 0:
        return np.ones(m) / m
    return out / s2


def adam_optimize(p0, duration_s, seed, lr0=0.015, tau_active=3):
    """
    Adam optimization in log-domain using exact max subgradient.
    p0: initial normalized sequence
    Returns (best_p, best_score)
    """
    p = np.asarray(p0, dtype=np.float64).copy()
    n = p.size
    s = p.sum()
    if s <= 0:
        return p0, np.inf
    p = p / s
    
    L = next_pow2(2 * n - 1)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + duration_s
    
    # Log-domain parameterization
    z = np.log(np.maximum(p, 1e-300))
    z = z - z.mean()
    z = np.clip(z, -50, 50)
    
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    best_p = p.copy()
    best_score = fast_score(p)
    no_improve = 0
    lr = lr0
    
    while time.monotonic() < deadline:
        # Compute convolution and score
        c = compute_conv(p, L)
        c = np.where(np.isfinite(c), c, 0.0)
        cm = c.max()
        score = 2.0 * n * cm
        
        if score < best_score - 1e-12:
            best_score = float(score)
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute exact subgradient
        subg = exact_subgradient(p, c, top_k=tau_active)
        
        # Project to simplex tangent
        pg = project_subgradient(p, subg)
        
        # Normalize gradient magnitude
        gnorm = np.linalg.norm(pg)
        if gnorm > 0:
            pg = pg / gnorm * min(gnorm, 0.1)
        
        # Adam update
        step += 1
        m = beta1 * m + (1 - beta1) * pg
        v = beta2 * v + (1 - beta2) * (pg * pg)
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        z = np.where(np.isfinite(z), z, 0.0)
        z = np.clip(z, -50, 50)
        p = softmax_safe(z)
        
        # Adaptive learning rate
        if step % 2000 == 0:
            lr = max(0.001, lr * 0.95)
        
        # Random jump from best if stuck
        if no_improve > 3000:
            z = np.log(np.maximum(best_p, 1e-300))
            z = z - z.mean()
            z = z + rng.normal(0, 0.25, n)
            z = np.clip(z, -50, 50)
            p = softmax_safe(z)
            m *= 0.1
            v *= 0.1
            no_improve = 0
            lr = max(0.003, lr * 0.6)
            tau_active = max(1, tau_active - 1)
        
        # Periodic print
        if step % 10000 == 0:
            print(f"  adam step {step}: score={score:.6f} best={best_score:.6f} lr={lr:.5f}")
    
    return best_p, best_score


def lp_frank_wolfe(p0, duration_s, n_lp=2048):
    """
    Frank-Wolfe style LP-based improvement.
    Given p, solve LP to find x minimizing max_k conv(p,x)[k]
    subject to x>=0, sum(x)=1. Then step toward x.
    """
    if not HAVE_LINPROG:
        return p0, np.inf
    
    p = np.asarray(p0, dtype=np.float64).copy()
    n_full = p.size
    p = p / p.sum()
    
    deadline = time.monotonic() + duration_s
    current = p.copy()
    best_p = p.copy()
    best_s = fast_score(p)
    
    it = 0
    no_improve = 0
    
    while time.monotonic() < deadline and no_improve < 15:
        # Solve LP at reduced size
        q = resize_sequence(current, n_lp)
        n = n_lp
        max_c = float(fast_score(q) / (2.0 * n))  # = max(conv(q,q))
        
        if max_c <= 0:
            break
        
        # Build constraint matrix: A @ x <= max_c for all lags
        # A[k, j] = q[k-j] if 0<=k-j<n else 0
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for k in range(2 * n - 1):
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            A[k, i_lo:i_hi + 1] = q[k - i_lo : k - i_hi + 1]
        
        obj = -np.ones(n)  # maximize sum(x)
        b = np.full(2 * n - 1, max_c)
        
        try:
            res = linprog(obj, A_ub=A, b_ub=b,
                         bounds=[(0, None)] * n, method='highs',
                         options={'time_limit': max(1, deadline - time.monotonic())})
        except Exception:
            break
        
        if not res.success or res.x is None:
            break
        
        x = np.maximum(res.x, 0)
        sx = x.sum()
        if sx < 1e-10:
            break
        x = x / sx
        
        # Resize x back to full size and step
        x_full = resize_sequence(x, n_full)
        
        improved = False
        for alpha in [0.3, 0.5, 0.7, 0.8, 0.9, 1.0]:
            trial = (1 - alpha) * current + alpha * x_full
            trial = np.maximum(trial, 0)
            ts = trial.sum()
            if ts < 1e-10:
                continue
            trial = trial / ts
            s = fast_score(trial)
            if s < best_s - 1e-10:
                best_s = s
                best_p = trial.copy()
                current = trial
                improved = True
                break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
        
        it += 1
        if it % 5 == 0:
            print(f"  fw iter {it}: score={best_s:.6f}")
    
    return best_p, best_s


def propose_candidate(seed=42, budget_s=1800.0, **kwargs):
    """Main search function. Returns best sequence found."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    raw_budget = max(0.0, float(budget_s))
    deadline = start_time + max(0.0, raw_budget - 15.0)
    
    # Choose sequence length based on budget
    if raw_budget < 300:
        N = 4096
    elif raw_budget < 900:
        N = 8192
    else:
        N = 16384
    
    print(f"Budget: {raw_budget:.0f}s, N={N}")
    
    best_p = None
    best_s = np.inf
    
    def update(p, s, label=""):
        nonlocal best_p, best_s
        if p is None or not np.isfinite(s):
            return
        if best_p is None or s < best_s:
            best_p = np.asarray(p, dtype=np.float64).copy()
            best_s = float(s)
            print(f"  NEW BEST ({label}): {best_s:.6f}")
    
    # Try to use previous best as starting point
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        prev = np.where(np.isfinite(prev), prev, 0.0)
        prev = np.maximum(prev, 0.0)
        if prev.size > 0 and prev.sum() > 1e-9:
            prevN = resize_sequence(prev, N)
            s_prev = fast_score(prevN)
            update(prevN, s_prev, "prev")
    except Exception:
        pass
    
    # Generate diverse initializations
    inits = []
    labels = []
    
    # Ramps (asymmetric!)
    inits.append(make_ramp(N, 1))
    labels.append("ramp_up")
    inits.append(make_ramp(N, -1))
    labels.append("ramp_down")
    
    # U-shapes with asymmetry
    for power in [1.5, 2.0, 3.0, 4.0]:
        for asym in [-0.3, -0.15, 0.0, 0.15, 0.3]:
            inits.append(make_u_shape(N, power, asym))
            labels.append(f"u_p{power}_a{asym}")
    
    # Polynomial shapes
    inits.append(make_poly(N, [1.0, 0.0]))
    labels.append("poly_lin")
    inits.append(make_poly(N, [0.5, 0.5]))
    labels.append("poly_lin2")
    inits.append(make_poly(N, [-0.5, 1.0]))
    labels.append("poly_mix")
    
    # GPSR random inits
    for i in range(8):
        inits.append(make_gpsr_init(N, rng, dim=12))
        labels.append(f"gpsr_{i}")
    
    # Evaluate all initializations
    for p_init, label in zip(inits, labels):
        s = fast_score(p_init)
        update(p_init, s, label)
    
    if best_p is None:
        best_p = np.ones(N) / N
        best_s = 2.0
    
    print(f"After init: best={best_s:.6f}")
    
    # Phase 1: Adam optimization from best init
    rem = deadline - time.monotonic()
    if rem > 60:
        dur1 = min(rem * 0.35, 500.0)
        p1, s1 = adam_optimize(best_p, dur1, seed=seed, lr0=0.018, tau_active=3)
        update(p1, s1, "adam1")
    
    # Phase 2: Frank-Wolfe LP steps
    rem = deadline - time.monotonic()
    if rem > 80 and HAVE_LINPROG:
        n_lp = min(N, 2048)
        dur2 = min(rem * 0.2, 200.0)
        p2, s2 = lp_frank_wolfe(best_p, dur2, n_lp=n_lp)
        if p2 is not None:
            p2 = resize_sequence(p2, N)
            s2 = fast_score(p2)
            update(p2, s2, "fw")
    
    # Phase 3: Adam from LP-improved point (or best)
    rem = deadline - time.monotonic()
    if rem > 60:
        dur3 = min(rem * 0.3, 400.0)
        p3, s3 = adam_optimize(best_p, dur3, seed=seed + 1000, lr0=0.012, tau_active=2)
        update(p3, s3, "adam3")
    
    # Phase 4: Another round of diverse starts + polish
    rem = deadline - time.monotonic()
    if rem > 90:
        # Try a few more diverse inits
        for i in range(4):
            p_rand = make_gpsr_init(N, rng, dim=20)
            s_rand = fast_score(p_rand)
            if s_rand < best_s:
                update(p_rand, s_rand, f"late_gpsr_{i}")
        
        # Final polish
        rem2 = deadline - time.monotonic()
        if rem2 > 40:
            dur4 = rem2 * 0.85
            p4, s4 = adam_optimize(best_p, dur4, seed=seed + 2000, lr0=0.008, tau_active=1)
            update(p4, s4, "adam_final")
    
    # Final validation with evaluate_sequence
    candidates = []
    if best_p is not None:
        candidates.append(best_p)
    # Add a few fallbacks
    candidates.append(np.ones(1024) / 1024.0)
    candidates.append(make_ramp(1024, 1))
    
    final_seq = None
    final_score = np.inf
    
    for cand in candidates:
        cand = np.asarray(cand, dtype=np.float64)
        lst = [float(max(0.0, min(1000.0, x))) for x in cand]
        if not lst:
            continue
        try:
            sc = evaluate_sequence(lst)
        except Exception:
            sc = np.inf
        if np.isfinite(sc) and sc < final_score:
            final_score = float(sc)
            final_seq = lst
    
    if final_seq is None:
        final_seq = [1.0 / 1024.0] * 1024
    
    print(f"FINAL SCORE: {final_score:.6f}")
    return final_seq
