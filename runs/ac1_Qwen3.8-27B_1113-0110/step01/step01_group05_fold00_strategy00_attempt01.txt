def evaluate_sequence_internal(seq):
    """
    Internal version of evaluate_sequence that ensures all conditions are met.
    """
    if seq is None:
        return float('inf')
    if np.size(seq) <= 1:
        return float('inf')
    # Check for non-negative and finite
    if any(x<0 for x in seq) or any(np.isinf(seq) or not np.isfinite(x) for x in seq):
        return float('inf')
    # Ensure no NaNs and within 0-1000 bounds
    seq = clean_array(seq)
    if np.size(seq) == 0 or np.max(seq) > 1000 or np.min(seq) < 0:
        return float('inf')
    # Compute score
    conv = np.convolve(seq, seq, mode='full')
    max_b = np.max(conv)
    n = np.size(seq)
    sum_a = np.sum(seq)
    score = 2.0 * n * max_b / (sum_a ** 2)
    if score < 0 or np.isnan(score):
        return float('inf')
    return score
