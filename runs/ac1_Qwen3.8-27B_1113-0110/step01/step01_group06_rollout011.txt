2. Find the top K indices: k_1, k_2, ..., k_K (sorted by c value)
3. For each top index k_i, compute the "blame" on each a[j]: how much a[j] contributes to c[k_i]
4. Reduce the a[j] that have the most total blame
5. But don't reduce any a[j] too much (it might increase other c[k] values)

This is essentially what the exponentiated gradient does, but let me make it more explicit.

**Alternative: Coordinate descent on the active set**

1. Identify the active set S = {k : c[k] >= max(c) * (1 - ε)}
2. For the active set, we want to minimize max_{k in S} c[k]
3. This is equivalent to: find a direction d such that c[k] + t*(d*d)[k] < max(c) for all k in S
4. Take the largest such t

This is a quadratic programming problem: minimize t subject to (d*d)[k] <= -c[k]/t for all k in S... it's getting complicated.

def _next_pow2(x):

    if x <= 1:

def _fft_square(a):

    c = np.fft.irfft(fa * fa, N).real[:L]

def _corr_grad(p, a):
    """Compute 2 * cross-correlation of p and a, giving gradient of <p, a*a> w.r.t. a."""
    n = a.size
    Lp = p.size  # 2n-1
    N = _next_pow2(n + Lp - 1)  # need >= 3n-2
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    R = np.fft.irfft(np.conj(P) * A, N)[:n]
    return 2.0 * R

    if c is None:
        c = _fft_square(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    return 2.0 * a.size * c.max() / (s * s)

def _make_multi_weights(c, focus_list, eps_list):
    """Create combined weights from multiple focus levels."""
    L = c.size
    maxc = c.max()
    if maxc < 1e-30:
        return np.full(L, 1.0/L)
    p_total = np.zeros(L)
    total_weight = 0
    for focus, eps in zip(focus_list, eps_list):
        z = (maxc - c) / maxc
        with np.errstate(over='ignore', under='ignore'):
            p = np.exp(-z * focus)
        s = p.sum()
        if s < 1e-12:
            p_new = np.zeros(L)
            p_new[np.argmax(c)] = 1.0
        else:
            p_new = (1-eps) * p/s + eps/L
        p_total += p_new
        total_weight += 1

"""
Step function optimizer for minimizing 2*n*max(a*a)/(sum(a))^2.
Uses exponentiated gradient with oscillating focus, Nesterov momentum,
multi-scale peak weighting, and periodic restarts.
"""
import numpy as np
import time

def _pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()

def _fft_square(a):
    n = a.size
    L = 2 * n - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N).real[:L]
    return np.maximum(c, 0)

def _corr_grad(p, a):
    n = a.size
    N = _pow2(n + p.size - 1)
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    R = np.fft.irfft(np.conj(P) * A, N)[:n]
    return 2.0 * R

def _score_internal(a, c=None):
    if c is None:
        c = _fft_square(a)
    s = float(a.sum())
    if s < 1e-10:
        return float('inf')
    return float(2.0 * a.size * c.max() / (s * s))

def _weights_multi(c, focuses, epsilons):
    L = c.size
    maxc = float(c.max())
    if maxc < 1e-30:
        return np.full(L, 1.0 / L)
    p_total = np.zeros(L, dtype=np.float64)
    for f, eps in zip(focuses, epsilons):
        z = (maxc - c) / maxc
        with np.errstate(over='ignore', under='ignore'):
            p = np.exp(-z * f)
        s = float(p.sum())
        if s < 1e-12:
            p_n = np.zeros(L)
            p_n[int(np.argmax(c))] = 1.0
        else:
            p_n = (1.0 - eps) * p / s + eps / L
        p_total += p_n
    p_total /= len(focuses)
    return p_total

def _init_cosine(n):
    x = np.linspace(0, np.pi, n)
    a = np.cos(x / 2) ** 2
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    return a

def _init_random_spectrum(n, rng):
    N = _pow2(2 * n - 1)
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    spectrum = np.ones(N // 2 + 1)
    A = np.zeros(N // 2 + 1, dtype=np.complex128)
    A[np.arange(N // 2 + 1)] = spectrum * np.exp(1j * phases)
    # For real signal: A[N-k] = conj(A[k])
    full_A = np.zeros(N, dtype=np.complex128)
    for k in range(N // 2 + 1):
        full_A[k] = A[k]
        if N - k < N and N - k != k:
            full_A[N - k] = np.conj(A[k])
    a_full = np.fft.ifft(full_A).real[:n]
    a = np.maximum(a_full, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a /= s
    return a

def _init_uniform(n):
    return np.ones(n, dtype=np.float64) / n

def _init_gaussian(n, rng):
    sigma = n / 4.0
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a += 1e-6 * rng.standard_normal(n)
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    return a

def _optimize(a0, time_limit, rng, label=""):
    """Main optimization loop."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = a.size
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-10:
        a = _init_uniform(n)
    else:
        a /= s
    
    best_a = a.copy()
    best_score = _score_internal(a)
    
    c = None
    score = best_score
    focus_phase = 0.0
    focus_period = 3.0  # radians per ~1000 iterations
    momentum_v = np.zeros(n, dtype=np.float64)
    momentum_beta = 0.7
    eta = 0.5
    eta_min = 0.01
    eta_max = 5.0
    
    no_improve = 0
    max_no_improve = 800
    resets = 0
    max_resets = 30
    t0 = time.monotonic()
    it = 0
    eval_count = 0
    
    while time.monotonic() - t0 < time_limit:
        if c is None:
            c = _fft_square(a)
            score = _score_internal(a, c)
            eval_count += 1
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if it % 5000 == 0:
            print(f"[{label}] it={it} t={time.monotonic()-t0:.1f}s "
                  f"score={score:.6f} best={best_score:.6f} "
                  f"focus_phase={focus_phase:.2f} eta={eta:.3f} res={resets}")
        
        if no_improve > max_no_improve:
            resets += 1
            if resets > max_resets:
                break
            no_improve = 0
            max_no_improve = int(max_no_improve * 1.2)
            # Restart with perturbation
            if resets % 3 == 0:
                a = 0.6 * best_a + 0.4 * _init_cosine(n)
            elif resets % 3 == 1:
                a = best_a * np.exp(rng.normal(0, 0.3, n))
            else:
                fresh = _init_gaussian(n, rng)
                a = 0.7 * best_a + 0.3 * fresh
            a = np.maximum(a, 1e-15)
            a /= a.sum()
            c = None
            score = float('inf')
            eta = 0.5
            momentum_v = np.zeros(n, dtype=np.float64)
            continue
        
        # Oscillating focus
        focus_phase += 0.001
        base_focus = 10.0
        amp = 50.0
        current_focus = base_focus + amp * (0.5 + 0.5 * np.sin(focus_phase))
        
        # Multi-scale weights
        focuses = [current_focus * 0.2, current_focus, current_focus * 5.0]
        epsilons = [0.1, 0.02, 0.01]
        p = _weights_multi(c, focuses, epsilons)
        
        # Compute gradient
        g = _corr_grad(p, a)
        
        # Nesterov momentum
        momentum_v = momentum_beta * momentum_v + (1 - momentum_beta) * g
        g = momentum_v
        
        # Normalize gradient
        max_g = float(np.max(np.abs(g)))
        if max_g < 1e-20:
            break
        g = g / max_g
        
        # Exponentiated gradient update
        a_new = a * np.exp(-eta * g)
        a_new = np.maximum(a_new, 1e-300)
        ss = float(a_new.sum())
        if ss < 1e-30:
            eta = max(eta * 0.5, eta_min)
            continue
        a_new /= ss
        
        c_new = _fft_square(a_new)
        score_new = _score_internal(a_new, c_new)
        eval_count += 1
        
        # Accept/reject
        wt_old = float(np.dot(p, c))
        wt_new = float(np.dot(p, c_new))
        
        accept = False
        if score_new < best_score - 1e-12:
            accept = True
        elif score_new < score - 1e-12:
            accept = True
        elif score_new < score * 1.002 and wt_new < wt_old - 1e-12:
            accept = True
        
        if accept:
            a = a_new
            c = c_new
            score = score_new
            eta = min(eta * 1.05, eta_max)
        else:
            eta = max(eta * 0.7, eta_min)
            if eta < 0.05:
                # Perturb to escape
                a = a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 1e-15)
                a /= a.sum()
                c = None
                score = float('inf')
                momentum_v = np.zeros(n, dtype=np.float64)
                eta = 0.5
    
    return best_a, float(best_score)

Wait, I need to double-check my `_corr_grad` function. Let me re-derive:

We want: grad[j] = ∂/∂a[j] (sum_k p[k] * c[k]) where c[k] = sum_i a[i]*a[k-i]

As I derived: grad[j] = 2 * sum_i p[j+i] * a[i] (for i in [0, n-1] and j+i in [0, 2n-2])

This is: 2 * (p ⋆ a)[j] where (p ⋆ a)[j] = sum_i p[j+i]*a[i] (cross-correlation with shift)

Using FFT: (p ⋆ a)[j] = ifft(conj(FFT(p)) * FFT(a))[j] with zero-padding to avoid wrap-around.

The maximum value of j+i is (n-1) + (n-1) = 2n-2, and the minimum is 0+0 = 0.
The cross-correlation (p ⋆ a)[j] for j in [0, n-1] involves p[j+i] for i in [0, n-1], so p indices go up to j+n-1 <= 2n-2. ✓

For FFT: I need to pad p (length 2n-1) and a (length n) to at least (2n-1) + n - 1 = 3n-2 to avoid circular wrap-around.

So N >= 3n-2. Let me use N = _pow2(3*n - 1) to be safe.

"""
Step function optimizer minimizing 2*n*max(convolve(a,a))/(sum(a))^2.
Uses exponentiated gradient with oscillating multi-scale focus, Nesterov momentum,
adaptive step size, diverse initializations (cosine, Gaussian, random-spectrum),
and a dedicated polish phase. Handles multiple sequence lengths.
"""

import numpy as np
import time


def _pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _fft_square(a):
    """Compute full self-convolution a*a via FFT."""
    n = a.size
    L = 2 * n - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N).real[:L]
    return np.maximum(c, 0.0)


def _corr_grad(p, a):
    """
    Gradient of <p, a*a> w.r.t. a.
    grad[j] = 2 * sum_i p[j+i] * a[i]  (cross-correlation).
    Uses FFT with zero-padding.
    """
    n = a.size
    # Need N >= len(p) + len(a) - 1 = (2n-1) + n - 1 = 3n-2
    N = _pow2(3 * n - 1)
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    R = np.fft.irfft(np.conj(P) * A, N)[:n]
    return 2.0 * R


def _score_internal(a, c=None):
    if c is None:
        c = _fft_square(a)
    s = float(a.sum())
    if s < 1e-10:
        return float('inf')
    return float(2.0 * a.size * c.max() / (s * s))


def _weights_multi(c, focuses, epsilons):
    """Create combined softmax weights from multiple focus levels."""
    L = c.size
    maxc = float(c.max())
    if maxc < 1e-30:
        return np.full(L, 1.0 / L, dtype=np.float64)
    p_total = np.zeros(L, dtype=np.float64)
    count = 0
    for f, eps in zip(focuses, epsilons):
        z = (maxc - c) / maxc
        with np.errstate(over='ignore', under='ignore'):
            p = np.exp(-z * f)
        s = float(p.sum())
        if s < 1e-12:
            p_n = np.zeros(L, dtype=np.float64)
            p_n[int(np.argmax(c))] = 1.0
        else:
            p_n = (1.0 - eps) * (p / s) + eps / L
        p_total += p_n
        count += 1
    p_total /= max(1, count)
    return p_total


def _init_cosine(n):
    """Cosine-bump initialization (high center, low edges)."""
    x = np.linspace(0.0, np.pi, n, dtype=np.float64)
    a = np.cos(x * 0.5) ** 2
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    return a


def _init_gaussian(n, rng):
    """Gaussian-shaped initialization with noise."""
    sigma = n / 5.0
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a += 0.01 * rng.standard_normal(n)
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    return a


def _init_random_spectrum(n, rng):
    """Random-phase flat-spectrum initialization."""
    N = _pow2(2 * n - 1)
    half = N // 2 + 1
    phases = rng.uniform(0.0, 2.0 * np.pi, half)
    full_A = np.zeros(N, dtype=np.complex128)
    for k in range(half):
        full_A[k] = np.exp(1j * phases[k])
        if N - k < N and N - k != k:
            full_A[N - k] = np.exp(-1j * phases[k])
    a_full = np.fft.ifft(full_A).real[:n]
    a = np.maximum(a_full, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / n
    a /= s
    return a


def _init_uniform(n):
    return np.ones(n, dtype=np.float64) / n


def _optimize(a0, time_limit, rng, label=""):
    """Main exponentiated-gradient optimizer with oscillating focus and momentum."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = a.size
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-10:
        a = _init_uniform(n)
    else:
        a /= s

    best_a = a.copy()
    best_score = _score_internal(a)

    c = None
    score = best_score
    focus_phase = 0.0
    momentum_v = np.zeros(n, dtype=np.float64)
    beta = 0.65
    eta = 0.5
    eta_min = 0.02
    eta_max = 5.0

    no_improve = 0
    max_no = 600
    resets = 0
    max_resets = 35
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        if c is None:
            c = _fft_square(a)
            score = _score_internal(a, c)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if it % 4000 == 0 and it > 0:
            print(f"  [{label}] it={it} t={time.monotonic()-t0:.1f}s "
                  f"score={score:.6f} best={best_score:.6f} "
                  f"eta={eta:.3f} res={resets}")

        if no_improve > max_no:
            resets += 1
            if resets > max_resets:
                break
            no_improve = 0
            max_no = int(max_no * 1.15)
            if resets % 4 == 1:
                a = 0.5 * best_a + 0.5 * _init_cosine(n)
            elif resets % 4 == 2:
                a = best_a * np.exp(rng.normal(0.0, 0.4, n))
            elif resets % 4 == 3:
                fresh = _init_gaussian(n, rng)
                a = 0.6 * best_a + 0.4 * fresh
            else:
                fresh = _init_random_spectrum(n, rng)
                a = 0.7 * best_a + 0.3 * fresh
            a = np.maximum(a, 1e-15)
            a /= a.sum()
            c = None
            score = float('inf')
            eta = 0.5
            momentum_v = np.zeros(n, dtype=np.float64)
            continue

        # Oscillating focus: periodically explore (low focus) and exploit (high focus)
        focus_phase += 0.0008
        base_f = 8.0
        amp_f = 60.0
        cur_focus = base_f + amp_f * (0.5 + 0.5 * np.sin(focus_phase))
        cur_focus = max(2.0, cur_focus)

        # Multi-scale weights
        focuses = [max(1.0, cur_focus * 0.15), cur_focus, min(500.0, cur_focus * 4.0)]
        epsilons = [0.15, 0.03, 0.01]
        p = _weights_multi(c, focuses, epsilons)

        # Gradient with momentum
        g = _corr_grad(p, a)
        momentum_v = beta * momentum_v + (1.0 - beta) * g
        g = momentum_v

        max_g = float(np.max(np.abs(g)))
        if max_g < 1e-20:
            break
        g = g / max_g

        # Exponentiated update
        a_new = a * np.exp(-eta * g)
        a_new = np.maximum(a_new, 1e-300)
        ss = float(a_new.sum())
        if ss < 1e-30:
            eta = max(eta * 0.5, eta_min)
            continue
        a_new /= ss

        c_new = _fft_square(a_new)
        score_new = _score_internal(a_new, c_new)

        wt_old = float(np.dot(p, c))
        wt_new = float(np.dot(p, c_new))

        accept = False
        if score_new < best_score - 1e-12:
            accept = True
        elif score_new < score - 1e-12:
            accept = True
        elif score_new < score * 1.001 and wt_new < wt_old - 1e-13:
            accept = True

        if accept:
            a = a_new
            c = c_new
            score = score_new
            eta = min(eta * 1.04, eta_max)
            beta = min(beta + 0.0001, 0.8)
        else:
            eta = max(eta * 0.72, eta_min)
            beta = max(beta - 0.01, 0.3)
            if eta < 0.06:
                a = a * np.exp(rng.normal(0.0, 0.08, n))
                a = np.maximum(a, 1e-15)
                a /= a.sum()
                c = None
                score = float('inf')
                momentum_v = np.zeros(n, dtype=np.float64)
                eta = 0.5
                beta = 0.65

        it += 1

    return best_a, float(best_score)


def _polish(a0, time_limit, rng, label=""):
    """Fine-tuning: very high focus, small steps, target the single max."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = a.size
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 1e-15)
    a /= a.sum()

    best_a = a.copy()
    best_score = _score_internal(a)

    c = None
    eta = 0.15
    eta_min = 0.005
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        if c is None:
            c = _fft_square(a)
        score = _score_internal(a, c)

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 800:
            break

        if it % 3000 == 0 and it > 0:
            print(f"  [{label}] it={it} t={time.monotonic()-t0:.1f}s "
                  f"score={score:.7f} best={best_score:.7f} eta={eta:.4f}")

        # High + medium focus
        maxc = float(c.max())
        if maxc < 1e-30:
            break
        z = (maxc - c) / maxc
        focuses = [80.0, 300.0]
        epsilons = [0.05, 0.01]
        p = _weights_multi(c, focuses, epsilons)

        g = _corr_grad(p, a)
        max_g = float(np.max(np.abs(g)))
        if max_g < 1e-20:
            break
        g = g / max_g

        a_new = a * np.exp(-eta * g)
        a_new = np.maximum(a_new, 1e-300)
        a_new /= a_new.sum()

        c_new = _fft_square(a_new)
        score_new = _score_internal(a_new, c_new)

        if score_new < best_score - 1e-14:
            a = a_new
            c = c_new
            score = score_new
            eta = min(eta * 1.01, 0.8)
        else:
            eta = max(eta * 0.85, eta_min)
            if eta < 0.015:
                a = a * np.exp(rng.normal(0.0, 0.04, n))
                a = np.maximum(a, 1e-15)
                a /= a.sum()
                c = None
                eta = 0.15
                no_improve = 0

        it += 1

    return best_a, float(best_score)


def _safe_eval(seq):
    ev = globals().get("evaluate_sequence")
    if ev is None or not callable(ev):
        return float('inf')
    try:
        return float(ev(seq))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Tries multiple initializations and sequence lengths,
    optimizes each with exponentiated gradient, then polishes the best result.
    """
    total = max(120.0, float(budget_s))
    t_start = time.monotonic()
    deadline = t_start + total - 35.0

    best_a = None
    best_score = float('inf')

    # Load given sequence
    given_a = None
    try:
        g = globals().get("height_sequence_1")
        if g is not None:
            ga = np.asarray(g, dtype=np.float64).ravel()
            ga = np.nan_to_num(ga, nan=0.0, posinf=0.0, neginf=0.0)
            ga = np.clip(ga, 0.0, 1000.0)
            if ga.size > 10 and ga.sum() > 0.01:
                ga /= ga.sum()
                sc = _score_internal(ga)
                print(f"Given sequence: n={ga.size}, score={sc:.6f}")
                given_a = ga
                best_a = ga.copy()
                best_score = sc
    except Exception as e:
        print(f"Given load error: {e}")

    # Define tasks: (init_type, n, time_alloc, label)
    tasks = []
    if given_a is not None:
        tasks.append(("given", given_a.size, 300, "given"))

    tasks.append(("cosine", 8192, 500, "cos8192"))
    tasks.append(("cosine", 16384, 500, "cos16384"))
    tasks.append(("gauss", 8192, 350, "gauss8192"))
    tasks.append(("rand_spec", 16384, 350, "randspec16384"))
    tasks.append(("gauss", 32768, 400, "gauss32768"))

    for idx, (init_type, n, alloc, label) in enumerate(tasks):
        now = time.monotonic()
        if now >= deadline - 250:
            break
        actual_alloc = min(alloc, deadline - now - 250)
        if actual_alloc < 40:
            break

        task_rng = np.random.default_rng((seed * 100003 + idx * 7919 + 13) % (2 ** 32))

        if init_type == "given":
            a0 = given_a.copy() if given_a is not None else _init_uniform(n)
        elif init_type == "cosine":
            a0 = _init_cosine(n)
        elif init_type == "gauss":
            a0 = _init_gaussian(n, task_rng)
        elif init_type == "rand_spec":
            a0 = _init_random_spectrum(n, task_rng)
        else:
            a0 = _init_uniform(n)

        print(f"\nTask {idx}: [{label}] n={n} alloc={actual_alloc:.0f}s best_so_far={best_score:.6f}")
        t0 = time.monotonic()

        try:
            cand_a, cand_score = _optimize(a0, actual_alloc, task_rng, label=label)
            if cand_a is not None and np.isfinite(cand_score):
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = cand_a.copy()
                    print(f"  *** New best: {cand_score:.6f} (n={n})")
                else:
                    print(f"  score={cand_score:.6f} (best={best_score:.6f})")
        except Exception as e:
            print(f"  Task [{label}] FAILED: {repr(e)}")

    # Second pass: re-optimize the best found (if time permits)
    if best_a is not None:
        now = time.monotonic()
        remaining = deadline - now - 180
        if remaining > 200:
            refine_time = min(350.0, remaining)
            task_rng = np.random.default_rng((seed * 3 + 777) % (2 ** 32))
            print(f"\nRefine pass: n={best_a.size} time={refine_time:.0f}s")
            try:
                ref_a, ref_score = _optimize(best_a, refine_time, task_rng, label="refine")
                if np.isfinite(ref_score) and ref_score < best_score:
                    best_score = ref_score
                    best_a = ref_a.copy()
                    print(f"  Refined: {ref_score:.6f}")
            except Exception as e:
                print(f"  Refine failed: {repr(e)}")

    # Polish phase
    if best_a is not None:
        now = time.monotonic()
        remaining = deadline - now - 20
        if remaining > 50:
            pol_time = min(180.0, remaining)
            pol_rng = np.random.default_rng((seed + 99999) % (2 ** 32))
            print(f"\nPolish: n={best_a.size} time={pol_time:.0f}s")
            try:
                pol_a, pol_score = _polish(best_a, pol_time, pol_rng, label="polish")
                if np.isfinite(pol_score) and pol_score < best_score:
                    best_score = pol_score
                    best_a = pol_a.copy()
                    print(f"  Polished: {pol_score:.6f}")
                else:
                    print(f"  Polish: {pol_score:.6f} (no improvement over {best_score:.6f})")
            except Exception as e:
                print(f"  Polish failed: {repr(e)}")

    # Fallback
    if best_a is None:
        best_a = _init_uniform(8192)
        best_score = 2.0

    # Final cleanup
    best_a = np.nan_to_num(best_a, nan=0.0, posinf=0.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    ss = float(best_a.sum())
    if ss <= 0.01:
        best_a = _init_uniform(8192)
        ss = 1.0
    else:
        best_a /= ss

    result = [float(x) for x in best_a]

    # Verify with exact evaluator if time permits
    if time.monotonic() < deadline - 3:
        exact = _safe_eval(result)
        print(f"\nFINAL: n={len(result)}, internal_score={best_score:.6f}, exact_score={exact:.6f}")
    else:
        print(f"\nFINAL: n={len(result)}, internal_score={best_score:.6f}")

    return result
