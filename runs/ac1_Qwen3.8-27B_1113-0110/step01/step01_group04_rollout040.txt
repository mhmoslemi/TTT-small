"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative a of length n.

Strategy:
  - Start from strong candidates (existing best, power laws, modified shapes).
  - Main phase: projected gradient descent with adaptive peak-emphasizing weights
    (weights = (c/M)^p with p increasing over time).
  - Final phase: stochastic local search with block multiplicative perturbations.
  - Adaptive learning rate with backtracking.
"""

import numpy as np
import time


def _pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _autocorr(a, L):
    A = np.fft.rfft(a, n=L)
    c = np.fft.irfft(A * A, n=L)
    return c[:2 * len(a) - 1]


def _exact_score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-12:
        return np.inf
    L = _pow2(2 * n - 1)
    c = _autocorr(a, L)
    mx = float(np.max(c))
    return 2.0 * n * mx / (s * s)


def _project_simplex(v):
    v = np.maximum(v, 0.0)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u > (css - 1.0) / idx
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    r = int(np.nonzero(cond)[0][-1])
    tau = (css[r] - 1.0) / (r + 1)
    out = v - tau
    np.maximum(out, 0.0, out=out)
    s = out.sum()
    if s <= 1e-15:
        return np.full(n, 1.0 / n)
    return out / s


def _weighted_gradient(a, w, Lg):
    """Gradient of sum_k w[k]*(a*a)[k] w.r.t. a."""
    w_pad = np.zeros(Lg, dtype=np.float64)
    w_pad[:len(w)] = w
    A = np.fft.rfft(a, n=Lg)
    W = np.fft.rfft(w_pad, n=Lg)
    g = 2.0 * np.fft.irfft(W * np.conj(A), n=Lg)[:len(a)]
    return g


def _power_law(n, p=0.5):
    j = np.arange(n, dtype=np.float64) + 1.0
    v = j ** (-p)
    return v / v.sum()


def _modified_power_law(n, p=0.5, alpha=0.0, beta=0.0):
    j = np.arange(n, dtype=np.float64) + 1.0
    t = np.arange(n, dtype=np.float64) / max(1, n - 1)
    v = j ** (-p)
    if alpha != 0.0:
        v *= np.exp(-alpha * t)
    if beta != 0.0:
        v *= 1.0 + beta * np.cos(2.0 * np.pi * t)
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return _power_law(n, p)
    return v / s


def _generate_starts(n):
    starts = []
    # Existing best
    try:
        h = height_sequence_1
        if isinstance(h, (list, np.ndarray)):
            arr = np.asarray(h, dtype=np.float64)
            if arr.ndim == 1 and arr.size == n and arr.sum() > 0:
                starts.append(_project_simplex(arr))
    except Exception:
        pass
    # Pure power laws
    for p in (0.48, 0.49, 0.50, 0.51, 0.52):
        starts.append(_power_law(n, p))
    # Modified with exponential decay
    for alpha in (0.3, 0.7, 1.5, 3.0, 5.0):
        starts.append(_modified_power_law(n, 0.5, alpha=alpha))
    # Modified with cosine modulation
    for beta in (-0.3, -0.15, 0.15, 0.3):
        starts.append(_modified_power_law(n, 0.5, beta=beta))
    # Combined
    for alpha in (0.5, 1.0, 2.0):
        for beta in (-0.2, 0.0, 0.2):
            starts.append(_modified_power_law(n, 0.5, alpha=alpha, beta=beta))
    # Two-sided (small weight)
    for tw in (0.01, 0.03, 0.05, 0.1):
        j = np.arange(n, dtype=np.float64) + 1.0
        rev = (n - np.arange(n, dtype=np.float64))
        v = j ** (-0.5) + tw * rev ** (-0.5)
        v = np.maximum(v, 0.0)
        if v.sum() > 0:
            starts.append(v / v.sum())
    return starts


def _projected_gradient_phase(a0, end_time, seed, L, Lg):
    """Main optimization: projected gradient with adaptive weights."""
    a = _project_simplex(a0).copy()
    n = len(a)
    rng = np.random.default_rng(seed)

    best = a.copy()
    best_score = _exact_score(best)
    print(f"[PG] Start score: {best_score:.6f}")

    # Adaptive parameters
    p = 4.0  # weight power
    lr = 0.05  # learning rate
    lr_min = 1e-6
    lr_max = 0.5
    no_improve = 0
    last_improve = time.monotonic()
    iter_count = 0

    start = time.monotonic()

    while time.monotonic() < end_time:
        # Compute autocorrelation
        c = _autocorr(a, L)
        M = float(np.max(c))
        if M <= 1e-15:
            break

        # Compute weights: emphasize peaks
        w = (c / M) ** p
        w_sum = w.sum()
        if w_sum > 0:
            w = w / w_sum  # normalize weights to sum=1

        # Compute gradient
        g = _weighted_gradient(a, w, Lg)

        # The gradient is for F = sum w[k]*c[k] with normalized w.
        # Scale gradient by M to get gradient of sum (c/M)^{p-1}*c[k]*...
        # Actually, let's just use the gradient as-is and scale by lr.
        # The actual objective we want to minimize is max(c), and w approximates its subgradient.
        # Gradient of max(c) at a is approximately 2 * cross_corr(w_subgrad, a)
        # where w_subgrad = delta at argmax. Our w is a smoothed version.
        # So we scale g by M to get the right magnitude.
        g = g * M

        # Projected gradient step
        a_new = _project_simplex(a - lr * g)

        # Evaluate
        if np.all(np.isfinite(a_new)):
            sc = _exact_score(a_new)
        else:
            sc = np.inf

        iter_count += 1

        if sc < best_score - 1e-10:
            best_score = sc
            best = a_new.copy()
            a = a_new
            no_improve = 0
            last_improve = time.monotonic()
            # Slightly increase lr
            lr = min(lr_max, lr * 1.02)
        else:
            a = a  # don't update
            no_improve += 1
            if no_improve >= 5:
                lr = max(lr_min, lr * 0.8)
            if no_improve >= 15:
                # Increase weight sharpness
                p = min(p * 1.5, 500.0)
                no_improve = 0
                lr = max(lr_min, lr * 0.6)
            if no_improve >= 40:
                # Restart with perturbation
                a = _project_simplex(best * np.exp(rng.normal(0, 0.05, n)))
                p = max(4.0, p * 0.7)
                lr = 0.05
                no_improve = 0
                last_improve = time.monotonic()

        # Periodic progress
        if iter_count % 200 == 0:
            elapsed = time.monotonic() - start
            remaining = max(0, end_time - time.monotonic())
            print(f"[PG] iter={iter_count} score={best_score:.6f} p={p:.1f} lr={lr:.4f} "
                  f"elapsed={elapsed:.0f}s remaining={remaining:.0f}s")

    return best, best_score


def _accelerated_gradient_phase(a0, end_time, seed, L, Lg):
    """FISTA-style accelerated projected gradient for the smoothed objective."""
    a = _project_simplex(a0).copy()
    z = a.copy()
    n = len(a)
    rng = np.random.default_rng(seed)

    best = a.copy()
    best_score = _exact_score(best)
    print(f"[FISTA] Start score: {best_score:.6f}")

    p = 8.0  # weight power (start sharper)
    t_k = 1.0
    lr = 0.1
    no_improve = 0
    last_improve = time.monotonic()
    iter_count = 0
    start = time.monotonic()

    while time.monotonic() < end_time:
        # Compute autocorrelation at z
        c = _autocorr(z, L)
        M = float(np.max(c))
        if M <= 1e-15:
            break

        # Weights
        w = (c / M) ** p
        w_sum = w.sum()
        if w_sum > 0:
            w = w / w_sum

        # Gradient at z
        g = _weighted_gradient(z, w, Lg) * M

        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = z + ((t_k - 1.0) / t_new) * (a - z)
        a_new = _project_simplex(y - lr * g)

        if np.all(np.isfinite(a_new)):
            z = a_new
        else:
            z = a
            t_k = 1.0

        t_k = t_new

        # Evaluate
        sc = _exact_score(z)
        iter_count += 1

        if sc < best_score - 1e-10:
            best_score = sc
            best = z.copy()
            no_improve = 0
            last_improve = time.monotonic()
        else:
            no_improve += 1
            if no_improve >= 10:
                lr = max(1e-5, lr * 0.7)
            if no_improve >= 25:
                # Increase p, reset momentum
                p = min(p * 1.8, 300.0)
                t_k = 1.0
                lr = 0.05
                no_improve = 0
            if no_improve >= 60:
                # Perturb and restart
                z = _project_simplex(best * np.exp(rng.normal(0, 0.03, n)))
                a = z.copy()
                t_k = 1.0
                p = max(4.0, p * 0.5)
                lr = 0.1
                no_improve = 0
                last_improve = time.monotonic()

        if iter_count % 150 == 0:
            elapsed = time.monotonic() - start
            remaining = max(0, end_time - time.monotonic())
            print(f"[FISTA] iter={iter_count} score={best_score:.6f} p={p:.1f} lr={lr:.4f} "
                  f"elapsed={elapsed:.0f}s remaining={remaining:.0f}s")

    return best, best_score


def _local_search(a0, end_time, seed, L):
    """Stochastic local search with block multiplicative perturbations."""
    a = _project_simplex(a0).copy()
    n = len(a)
    rng = np.random.default_rng(seed)

    best = a.copy()
    best_score = _exact_score(best)
    print(f"[LS] Start score: {best_score:.6f}")

    sigma = 0.08
    last_improve = time.monotonic()
    start = time.monotonic()
    iter_count = 0

    while time.monotonic() < end_time:
        cand = best.copy()
        # 1-3 blocks
        nblocks = 1 + int(rng.integers(0, 3))
        for _ in range(nblocks):
            maxw = max(3, n // 10)
            w = int(np.exp(rng.uniform(np.log(2.0), np.log(float(maxw)))))
            w = max(1, min(w, n))
            s = int(rng.integers(0, n - w + 1))
            cand[s:s + w] *= float(np.exp(rng.normal(0.0, sigma)))

        cand = _project_simplex(cand)
        sc = _exact_score(cand)
        iter_count += 1

        if np.isfinite(sc) and sc < best_score - 1e-10:
            best = cand
            best_score = sc
            last_improve = time.monotonic()
            sigma = min(0.25, sigma * 1.03)
        else:
            if time.monotonic() - last_improve > 3.0:
                sigma = max(0.01, sigma * 0.85)
            if time.monotonic() - last_improve > 15.0 and time.monotonic() - start > 20.0:
                # Re-center
                cand = _project_simplex(best * np.exp(rng.normal(0.0, 0.02, n)))
                sc2 = _exact_score(cand)
                if np.isfinite(sc2) and sc2 < best_score:
                    best = cand
                    best_score = sc2
                    last_improve = time.monotonic()

        if iter_count % 500 == 0:
            elapsed = time.monotonic() - start
            remaining = max(0, end_time - time.monotonic())
            print(f"[LS] iter={iter_count} score={best_score:.6f} sigma={sigma:.4f} "
                  f"elapsed={elapsed:.0f}s remaining={remaining:.0f}s")

    return best, best_score


def _refine_with_subgradient(a0, end_time, seed, L, Lg):
    """
    Subgradient method directly targeting the max.
    Uses the average of top-k peaks for a smoother subgradient.
    """
    a = _project_simplex(a0).copy()
    n = len(a)
    rng = np.random.default_rng(seed)

    best = a.copy()
    best_score = _exact_score(best)
    print(f"[SG] Start score: {best_score:.6f}")

    lr = 0.02
    top_k = 5  # use top-k peaks for subgradient
    no_improve = 0
    last_improve = time.monotonic()
    iter_count = 0
    start = time.monotonic()

    while time.monotonic() < end_time:
        c = _autocorr(a, L)
        M = float(np.max(c))
        if M <= 1e-15:
            break

        # Build subgradient from top-k peaks
        # Sort c to find top-k positions
        flat_c = c.copy()
        k = min(top_k, len(flat_c))
        top_indices = np.argsort(flat_c)[-k:]
        # Weight: proportional to c value
        w_top = np.zeros_like(flat_c)
        w_top[top_indices] = flat_c[top_indices]
        w_top = w_top / w_top.sum()

        # Gradient from subgradient
        g = _weighted_gradient(a, w_top, Lg) * M

        # Polyak step size if we have a lower bound
        # Lower bound on max(c) is 1/(2n-1) (averaging argument)
        # Score lower bound = 2n/(2n-1)
        lb = 2.0 * n / (2.0 * n - 1.0)
        # step = (F(a) - F*) / ||g||^2
        F_a = M  # current "objective" (max of c, with sum=1)
        F_star_est = lb / (2.0 * n)  # lower bound on max(c)
        g_norm_sq = float(np.dot(g, g))
        if g_norm_sq > 1e-20:
            step_polyak = (F_a - F_star_est) / g_norm_sq
            step = min(lr, 1.0 * step_polyak)
        else:
            step = lr

        a_new = _project_simplex(a - step * g)

        if np.all(np.isfinite(a_new)):
            sc = _exact_score(a_new)
        else:
            sc = np.inf

        iter_count += 1

        if sc < best_score - 1e-10:
            best_score = sc
            best = a_new.copy()
            a = a_new
            no_improve = 0
            last_improve = time.monotonic()
        else:
            no_improve += 1
            if no_improve >= 10:
                lr = max(1e-6, lr * 0.8)
            if no_improve >= 30:
                top_k = min(top_k + 2, 20)
                no_improve = 0
            if no_improve >= 80:
                a = _project_simplex(best * np.exp(rng.normal(0, 0.04, n)))
                lr = 0.02
                top_k = 5
                no_improve = 0
                last_improve = time.monotonic()

        if iter_count % 200 == 0:
            elapsed = time.monotonic() - start
            remaining = max(0, end_time - time.monotonic())
            print(f"[SG] iter={iter_count} score={best_score:.6f} top_k={top_k} lr={lr:.5f} "
                  f"elapsed={elapsed:.0f}s remaining={remaining:.0f}s")

    return best, best_score

        print(f"\n[Phase 1.{si}] Starting optimization from candidate {start_idx} "
              f"(initial score={scored[start_idx][0]:.6f}), time budget={time_per_start:.0f}s")

        # Alternate between FISTA and subgradient
        half_end = phase_start + (phase_end - phase_start) * 0.5

        if time.monotonic() < half_end:
            a1, sc1 = _accelerated_gradient_phase(a0, half_end, seed + 100 * si, L, Lg)
            if sc1 < best_score:
                best_score = sc1
                best_a = a1.copy()
        else:
            a1, sc1 = a0, scored[start_idx][0]

        if time.monotonic() < phase_end:
            a2, sc2 = _refine_with_subgradient(a1, phase_end, seed + 200 * si, L, Lg)
            if sc2 < best_score:
                best_score = sc2
                best_a = a2.copy()
        else:
            a2, sc2 = a1, sc1

        if time.monotonic() < phase_end - 5:
            a3, sc3 = _projected_gradient_phase(a2, phase_end - 5, seed + 300 * si, L, Lg)
            if sc3 < best_score:
                best_score = sc3
                best_a = a3.copy()

        print(f"[Phase 1.{si}] Best after phase: {best_score:.6f}")

    # Phase 2: Local search on global best
    ls_start = time.monotonic()
    ls_end = min(deadline - 10, ls_start + local_search_time)
    if ls_end > ls_start + 10:
        print(f"\n[Phase 2] Local search, time budget={ls_end - ls_start:.0f}s")
        ls_a, ls_sc = _local_search(best_a, ls_end, seed + 1000, L)
        if ls_sc < best_score:
            best_score = ls_sc
            best_a = ls_a.copy()
        print(f"[Phase 2] Best after local search: {best_score:.6f}")

    # Phase 3: Final quick polish with smaller perturbations
    final_end = deadline - 5
    if time.monotonic() < final_end - 5:
        print(f"\n[Phase 3] Final polish")
        fp_a, fp_sc = _local_search(best_a, final_end, seed + 2000, L)
        if fp_sc < best_score:
            best_score = fp_sc
            best_a = fp_a.copy()
        print(f"[Phase 3] Final score: {best_score:.6f}")

    # Final validation
    best_a = _project_simplex(np.maximum(best_a, 0.0))
    out = [float(x) for x in best_a]

    try:
        val = evaluate_sequence(out)
        print(f"Final evaluation: {val:.6f}")
        if not np.isfinite(val):
            print("Fallback to power law!")
            out = [float(x) for x in _power_law(n, 0.5)]
    except Exception as e:
        print(f"Validation error: {e}")

    print(f"Total time used: {time.monotonic() - t0:.1f}s")
    return out
