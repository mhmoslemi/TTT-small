def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def normalize_positive(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, float(floor), None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def convolve_seq(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if N is None:
        N = next_pow2(2 * n - 1)
    else:
        need = next_pow2(2 * n - 1)
        if int(N) < int(need):
            N = need
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]


def internal_score(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.inf
    a = a / s
    c = convolve_seq(a, N)
    return float(2.0 * n * float(np.max(c)))


def binary_score(c, n, m):
    return float(2.0 * n * float(np.max(c)) / (float(m) * float(m)))


def grad_from_conv(c, cmax, tau, n):
    """Compute gradient weights from convolution."""
    L = c.size
    if L == 0 or cmax <= 0 or not np.isfinite(cmax) or tau <= 0:
        return np.full(L, 1.0 / max(1, L))
    z = (c - cmax) / tau
    z = np.clip(z, -50.0, 0.0)
    ez = np.exp(z)
    s = float(ez.sum())
    if not np.isfinite(s) or s <= 0:
        return np.full(L, 1.0 / max(1, L))
    return ez / s


def compute_binary_gradient(b, c, cmax, tau, n, N):
    """Compute the gradient for binary positions."""
    w = grad_from_conv(c, cmax, tau, n)
    # For binary b, the gradient of c w.r.t. b[i] is:
    # g[i] = sum_k w[k] * (contribution of b[i] to c[k])
    # Using FFT: g = ifft(fft(w_rev) * fft(b))*2, truncated
    w_rev = w[::-1]
    fw = np.fft.rfft(w_rev, N)
    fb = np.fft.rfft(b, N)
    conv = np.fft.irfft(fw * fb, N)
    return 2.0 * conv[n - 1:2 * n - 1][::-1]


def generate_legendre(p):
    """Generate a Legendre sequence of prime length p."""
    if p < 2 or p % 2 == 0:
        return None
    # Check primality
    if p < 3:
        return None
    for d in range(2, int(math.isqrt(p)) + 1):
        if p % d == 0:
            return None
    # Legendre symbol
    b = np.zeros(p, dtype=np.float64)
    for k in range(1, p):
        # k is quadratic residue mod p iff k^((p-1)/2) ≡ 1 (mod p)
        if pow(k, (p - 1) // 2, p) == 1:
            b[k] = 1.0
    return b


def make_binary_from_real(a, m):

    order = np.argsort(a, kind='stable')
    b = np.zeros(n, dtype=np.float64)
    b[order[-m:]] = 1.0
    return b


def resample_seq(a, n):
    a = normalize_positive(a, floor=0.0)
    old_n = a.size
    n = max(2, int(n))
    if old_n == 1:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    if old_n == n:
        return a.astype(np.float64, copy=True)
    old_x = np.linspace(0.0, 1.0, old_n, dtype=np.float64)
    new_x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    b = np.interp(new_x, old_x, a)
    return normalize_positive(b, floor=1e-12)

def optimize_binary_steepest(b0, deadline, seed=0, K=12, verbose=False):
    """
    Binary optimization using steepest descent with simulated annealing.
    Tries K_remove * K_add candidate swaps per step, picks the best.
    Uses corrected incremental convolution update.
    """
    b = b0.copy().astype(np.float64)
    n = b.size
    m = int(b.sum())
    if m < 2 or m > n - 2:
        return b.copy(), np.inf

    N = next_pow2(2 * n - 1)
    c = np.rint(convolve_seq(b, N)).astype(np.float64)
    cmax = float(np.max(c))
    best_b = b.copy()
    best_cmax = cmax
    best_score = binary_score(c, n, m)

    rng = np.random.default_rng(int(seed))
    tau = 0.15 * cmax
    temp = 0.08
    stale = 0
    it = 0

    one_idx = np.nonzero(b > 0.5)[0]
    zero_idx = np.nonzero(b < 0.5)[0]

    while time.monotonic() < deadline:
        if it % 50 == 0:
            cmax = float(np.max(c))
            score = binary_score(c, n, m)
            if score < best_score * (1.0 - 1e-12):
                best_score = score
                best_cmax = cmax
                best_b = b.copy()

            if cmax <= 0 or not np.isfinite(cmax):
                break

            tau = 0.15 * cmax
            g = compute_binary_gradient(b, c, cmax, tau, n, N)

        # Select candidates
        if one_idx.size == 0 or zero_idx.size == 0:
            break

        g_one = g[one_idx]
        g_zero = g[zero_idx]

        # Top K positions to remove (highest gradient)
        if one_idx.size <= K:
            rem_candidates = one_idx.copy()
        else:
            top_rem = np.argpartition(g_one, -K)[-K:]
            rem_candidates = one_idx[top_rem]

        # Top K positions to add (lowest gradient)
        if zero_idx.size <= K:
            add_candidates = zero_idx.copy()
        else:
            top_add = np.argpartition(g_zero, K)[:K]
            add_candidates = zero_idx[top_add]

        # Try all combinations, pick best
        best_delta = np.inf
        best_pair = None
        score_cur = binary_score(c, n, m)

        for i in rem_candidates:
            for j in add_candidates:
                if i == j:
                    continue
                # Compute new max after swap using incremental update
                # c[i:i+n] -= 2*b; c[2i] -= 1; b[i]=0; c[j:j+n] += 2*b; c[2j] += 1; b[j]=1
                # New c at affected positions:
                # Instead of actually modifying, estimate the new max
                # The max can only decrease at positions where c was high and the swap reduces them
                # Simple estimate: track the max of c after the update
                # For efficiency, just track if any position exceeds current max

                # Quick rejection: if the swap doesn't affect the current max positions, skip
                # (This is an approximation for speed)

                # Compute the change in max by checking affected ranges
                # Position i+j gets special treatment
                # The new c values in affected ranges:
                # Range [i, i+n-1]: c[k] - 2*b[k-i] (+1 if k==2i for the diagonal fix)
                # Range [j, j+n-1]: c[k] + 2*b[k-j] (+1 if k==2j for the diagonal fix)

                # For speed, just estimate: does the swap reduce the max?
                # The maximum position(s) - check if they're in affected ranges
                max_pos = np.argmax(c)
                in_rem_range = (max_pos >= i) and (max_pos < i + n)
                in_add_range = (max_pos >= j) and (max_pos < j + n)

                if in_rem_range and not in_add_range:
                    # Max position is reduced
                    k = max_pos
                    change = -2.0 * b[k - i]
                    if k == 2 * i:
                        change += 1.0
                    new_val = c[k] + change
                    # Check if any other position becomes the new max
                    # Approximate: if new_val < cmax, likely improvement
                    if new_val < cmax - 0.5:
                        best_delta = new_val - cmax
                        best_pair = (int(i), int(j))
                        if best_delta < -2.0:
                            break
                elif in_add_range and not in_rem_range:
                    # Max position is increased - bad
                    continue
                elif in_rem_range and in_add_range:
                    # Both affect max position
                    k = max_pos
                    change_rem = -2.0 * b[k - i]
                    if k == 2 * i:
                        change_rem += 1.0
                    change_add = 2.0 * b[k - j]
                    if k == 2 * j:
                        change_add += 1.0
                    # But b[k-j] should use updated b (with b[i]=0)
                    if k - j == i:
                        change_add = 0.0  # b_new[i] = 0
                    if k - j == j:
                        change_add = 2.0 * 1.0  # b_new[j] = 1
                    else:
                        change_add = 2.0 * b[k - j]
                    if k == 2 * j:
                        change_add = 1.0  # diagonal
                    elif k - j == i:
                        change_add = 0.0
                    new_val = c[k] + change_rem + change_add
                    if new_val < cmax - 0.5:
                        best_delta = new_val - cmax
                        best_pair = (int(i), int(j))
                        if best_delta < -2.0:
                            break
                else:
                    # Max position not directly affected, but other positions might become max
                    # Check positions in the affected ranges that are close to cmax
                    # For speed, skip this case in the inner loop (handle via SA)
                    # Estimate: small random probability of selecting this pair
                    if rng.random() < 0.05:
                        best_delta = 0.1 * rng.random()
                        best_pair = (int(i), int(j))

            if best_pair is not None and best_delta < -2.0:
                break

        # If no good pair found, pick randomly (exploration)
        if best_pair is None:
            if rng.random() < 0.3 or stale > 100:
                i = int(one_idx[int(rng.integers(0, one_idx.size))])
                j = int(zero_idx[int(rng.integers(0, zero_idx.size))])
                if i != j:
                    best_pair = (i, j)
                    best_delta = 0.0

        if best_pair is None:
            stale += 10
            temp = max(0.001, temp * 0.7)
            it += 1
            continue

        i, j = best_pair
        i = int(i)
        j = int(j)

        # Apply the swap with CORRECT incremental update
        c[i:i + n] -= 2.0 * b
        c[2 * i] -= 1.0
        b[i] = 0.0
        c[j:j + n] += 2.0 * b
        c[2 * j] += 1.0
        b[j] = 1.0

        # Recompute max
        new_cmax = float(np.max(c))
        new_score = binary_score(c, n, m)

        # Acceptance
        accept = False
        if new_score < score_cur * (1.0 - 1e-12):
            accept = True
        else:
            delta = new_score - score_cur
            if delta <= 0:
                accept = rng.random() < 0.3
            elif delta < 5.0:
                accept = rng.random() < math.exp(-delta / max(temp, 1e-12))

        if accept:
            if new_score < best_score:
                best_score = new_score
                best_cmax = new_cmax
                best_b = b.copy()
                if verbose and it % 200 == 0:
                    print("  binary n={} m={} iter={} score={:.8f}".format(n, m, it, best_score), flush=True)
            stale = 0
            temp = max(0.001, temp * 0.997)
        else:
            # Revert with CORRECT incremental update
            c[j:j + n] -= 2.0 * b
            c[2 * j] -= 1.0
            b[j] = 0.0
            c[i:i + n] += 2.0 * b
            c[2 * i] += 1.0
            b[i] = 1.0

            stale += 1
            temp = max(0.001, temp * 0.94)
            if stale % 300 == 0:
                temp = min(0.15, temp * 2.0)
                stale = 0

        # Update candidate indices
        if accept:
            one_idx[one_idx == i] = zero_idx[zero_idx == j]  # This won't work well
            # Simpler: just rebuild occasionally
            if it % 50 == 0:
                one_idx = np.nonzero(b > 0.5)[0]
                zero_idx = np.nonzero(b < 0.5)[0]

        it += 1

    # Final check
    c_final = convolve_seq(best_b, N)
    final_score = binary_score(c_final, n, m)
    if final_score < best_score:
        best_score = final_score
        best_cmax = float(np.max(c_final))
    return best_b, best_score

def mirror_descent_polish(a, deadline, N=None, verbose=False):
    """Mirror descent / multiplicative weight optimization."""
    a = normalize_positive(a, floor=1e-15)
    n = a.size
    if N is None:
        N = next_pow2(3 * n - 2)

    best_a = a.copy()
    best_score = internal_score(a, N)

    eta = 0.5
    beta = 0.15
    stagnant = 0
    it = 0

    while time.monotonic() < deadline:
        c = convolve_seq(a, N)
        cmax = float(np.max(c))
        score = float(2.0 * n * cmax)
        if np.isfinite(score) and score < best_score * (1.0 - 1e-12):
            best_score = score
            best_a = a.copy()
            if verbose and it % 500 == 0:
                print("    mirror n={} iter={} score={:.8f}".format(n, it, best_score), flush=True)

        if cmax <= 0 or not np.isfinite(cmax):
            break

        tau = beta * cmax
        w = grad_from_conv(c, cmax, tau, n)
        w_rev = w[::-1]
        fa = np.fft.rfft(a, N)
        fw = np.fft.rfft(w_rev, N)
        conv = np.fft.irfft(fa * fw, N)
        g = 2.0 * conv[n - 1:2 * n - 1][::-1]
        g = g - g.mean()
        scale = float(np.abs(g).max())
        if not np.isfinite(scale) or scale <= 1e-15:
            break
        g = g / scale

        loga = np.log(a + 1e-20)
        loga -= float(np.max(loga))

        accepted = False
        for e in (eta, 0.5 * eta, 0.25 * eta, 0.125 * eta):
            if e <= 1e-10:
                continue
            loga_try = loga - e * g
            loga_try -= float(np.max(loga_try))
            a_try = np.exp(np.clip(loga_try, -60.0, 0.0))
            a_try = np.maximum(a_try, 1e-300)
            a_try = a_try / float(a_try.sum())

            c_try = convolve_seq(a_try, N)
            cmax_try = float(np.max(c_try))
            score_try = float(2.0 * n * cmax_try)

            if np.isfinite(score_try) and score_try < score * (1.0 - 1e-10):
                accepted = True
                a = a_try
                score = score_try
                break

        if accepted:
            stagnant = 0
            eta = min(1.5, eta * 1.02)
            beta = max(0.005, beta * 0.998)
        else:
            stagnant += 1
            eta = max(1e-4, eta * 0.6)
            beta = max(0.003, beta * 0.85)
            if stagnant % 200 == 0:
                beta = min(0.3, beta * 1.8)
                stagnant = 0
            if eta < 1e-5 or stagnant > 1500:
                break

        it += 1

    return best_a, best_score


def make_sine_sequence(n, alphas):
    n = max(2, int(n))
    x = (np.arange(n, dtype=np.float64) - 0.5 * (n - 1)) / max(1.0, 0.5 * n)
    base = np.ones(n, dtype=np.float64)
    for m_code, alpha in enumerate(alphas, start=1):
        base = base + float(alpha) * np.sin(np.pi * float(m_code) * x)
    return normalize_positive(base, floor=1e-12)


def generate_candidates(n, seed):

    # Given sequence (if available)
    given = get_given_sequence()
    if given is not None:
        candidates.append(resample_seq(given, n))

    # Legendre sequences
    for offset in (-8, -4, -2, -1, 0, 1, 2, 4, 8):
        p = n + offset
        if p < 10:
            continue
        leg = generate_legendre(p)
        if leg is not None and leg.size == n:
            candidates.append(leg)
        elif leg is not None:
            # Adjust to length n
            adj = np.zeros(n, dtype=np.float64)
            L = min(leg.size, n)
            adj[:L] = leg[:L]
            if adj.sum() > 0:
                candidates.append(adj)

    # Structured random
    for _ in range(3):
        x = np.linspace(0, 1, n)
        vals = np.ones(n)
        for m_code in range(1, 5):
            amp = rng.uniform(0.1, 0.5) / (m_code ** 0.6)
            ph = rng.uniform(0, 2 * np.pi)
            vals += amp * np.sin(2 * np.pi * m_code * x + ph)
            vals += amp * np.cos(2 * np.pi * m_code * x + ph)
        candidates.append(normalize_positive(vals, floor=1e-12))

    # Pure random
    candidates.append(rng.random(n))

    # Quadratic residue pattern (approximate)
    qr = np.zeros(n)
    p_approx = 4 * n
    leg2 = generate_legendre(p_approx if p_approx % 2 != 0 else p_approx + 2)
    if leg2 is not None:
        adj2 = np.zeros(n)
        L = min(leg2.size, n)
        adj2[:L] = leg2[:L]
        if adj2.sum() > 0:
            candidates.append(adj2)

    return candidates

    best_a = np.ones(1024, dtype=np.float64) / 1024.0
    best_score = internal_score(best_a)
    print("fallback score = {:.9f}".format(best_score), flush=True)

    given = get_given_sequence()

                best_a = a0_bin.copy()
