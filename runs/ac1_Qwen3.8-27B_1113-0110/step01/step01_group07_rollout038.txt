    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    return a_new


def _resize(a, m):
    """Resize sequence to length m via interpolation, preserving non-negativity and sum."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(m) / m
    a = a / s
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, m)
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0.0)
    so = out.sum()
    if so < 1e-15:
        return np.ones(m) / m
    return out / so


def _optimize_at_scale(a_init, L, budget_s, rng, alpha0=0.12, lr0=0.008):
    """Run the main optimization loop at a given scale. Returns best sequence and score."""
    start = time.monotonic()
    deadline = start + budget_s
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a, L)
    alpha = alpha0
    lr = lr0
    it = 0
    no_improve = 0
    grad_every = 8
    phase = 0
    thresholds = [int(budget_s * 0.15 / 0.0004), int(budget_s * 0.45 / 0.0004),
                  int(budget_s * 0.75 / 0.0004)]

    while time.monotonic() < deadline:
        it += 1

        # Phase transitions
        if phase < len(thresholds) and it >= thresholds[phase]:
            phase += 1
            alpha = max(0.015, alpha * 0.65)
            lr = max(0.001, lr * 0.6)
            grad_every = min(100, grad_every * 2)
            if it % 100000 == 0:
                print(f"  phase {phase}: alpha={alpha:.4f} lr={lr:.5f} score={best_score:.6f}")

        # Spectral flattening (main workhorse)
        a = _spectral_flatten(a, L, alpha)

        # Score check
        if it % 3 == 0 or it < 50:
            sc = _score_norm(a, L)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        # Gradient step (periodic)
        if it % grad_every == 0:
            a = _grad_step_multi_peak(a, L, lr, n_peaks=3)
            sc = _score_norm(a, L)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                no_improve = 0

        # Occasionally try single-peak gradient for sharper reduction
        if it % (grad_every * 5) == 0:
            a2 = _grad_step_single(a, L, lr * 1.5)
            sc2 = _score_norm(a2, L)
            if sc2 < best_score:
                best_score = sc2
                best_a = a2.copy()
                a = a2
                no_improve = 0

        # Diversity: restart with noise if stuck
        if no_improve > 3000:
            noise_scale = 0.02 + 0.03 * (phase + 1)
            noise = np.exp(rng.normal(0, noise_scale, n))
            a_noisy = a * noise
            a_noisy /= a_noisy.sum()
            sc_noisy = _score_norm(a_noisy, L)
            if sc_noisy < best_score * 1.03:
                a = a_noisy
            else:
                a = best_a.copy()
            no_improve = 0
            alpha = max(0.02, alpha * 0.4)

        if it % 200000 == 0:
            print(f"    iter {it} score={best_score:.6f} no_imp={no_improve}")

    return best_a, best_score


def _lp_escape(a, L, deadline):
    """Try one LP-based Frank-Wolfe step to escape local minimum."""
    if not HAVE_LINPROG:
        return a
    n = len(a)
    if n > 2048:
        n_lp = 1024
        a_lp = _resize(a, n_lp)
    else:
        n_lp = n
        a_lp = a.copy()

    L_lp = _next_pow2(2 * n_lp - 1)
    max_c = float(np.max(np.fft.irfft(np.fft.rfft(a_lp, L_lp) ** 2, L_lp)[:2 * n_lp - 1]))
    if max_c < 1e-15:
        return a

    # Build constraint matrix: conv(a_lp, g)[k] <= max_c for all k
    # conv(a_lp, g)[k] = sum_t a_lp[t] * g[k-t]
    # For each k, this is a linear constraint on g
    A = np.zeros((2 * n_lp - 1, n_lp))
    for k in range(2 * n_lp - 1):
        for t in range(max(0, k - n_lp + 1), min(n_lp, k + 1)):
            j = k - t
            if 0 <= j < n_lp:
                A[k, j] = a_lp[t]

    obj = -np.ones(n_lp)
    try:
        res = linprog(obj, A_ub=A, b_ub=np.full(2 * n_lp - 1, max_c),
                      bounds=[(0, None)] * n_lp, method='highs')
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0)
            sg = g.sum()
            if sg > 1e-15:
                g /= sg
                # Line search
                best_t = 0
                best_sc = _score_norm(_resize(a, n_lp), L_lp)
                for t in (0.2, 0.4, 0.6, 0.8):
                    c = (1 - t) * _resize(a, n_lp) + t * g
                    c = np.maximum(c, 0)
                    sc = _score_norm(c, L_lp)
                    if sc < best_sc:
                        best_sc = sc
                        best_t = t
                if best_t > 0:
                    a_new = _resize(a, n_lp)
                    a_new = (1 - best_t) * a_new + best_t * g
                    a_new = np.maximum(a_new, 0)
                    a = _resize(a_new, len(a))
    except Exception:
        pass
    return a


def propose_candidate(seed=42, budget_s=1800.0, **kwargs):
    """
    Multi-resolution spectral optimization for minimum-peak convolution.

    Strategy:
    1. Initialize from height_sequence_1 or structured defaults.
    2. Coarse-to-fine: optimize at N=512, 2048, 4096, 8192.
    3. At each scale: iterative spectral flattening + exponentiated gradient.
    4. Periodic LP-based Frank-Wolfe for escaping local minima.
    5. Return best sequence found.
    """
    start = time.monotonic()
    total_budget = float(budget_s)
    deadline_final = start + max(60.0, total_budget - 20.0)
    rng = np.random.default_rng(seed)

    # --- Load initial sequence ---
    a_init = None
    try:
        hs = np.asarray(height_sequence_1, dtype=np.float64)
        hs = np.where(np.isfinite(hs), hs, 0.0)
        hs = np.maximum(hs, 0.0)
        if hs.size > 10 and hs.sum() > 1e-9:
            a_init = hs / hs.sum()
    except Exception:
        pass

    # --- Multi-resolution optimization ---
    # Allocate time: 20% coarse, 30% mid, 50% fine
    scales = [512, 2048, 4096, 8192]
    time_fractions = [0.15, 0.20, 0.25, 0.40]

    best_overall = None
    best_score_overall = np.inf

    for si, N in enumerate(scales):
        rem = deadline_final - time.monotonic()
        if rem < 30:
            break
        alloc = min(rem, total_budget * time_fractions[si])
        if alloc < 15:
            alloc = rem * 0.5

        L = _next_pow2(2 * N - 1)

        # Get starting point for this scale
        if si == 0:
            if a_init is not None:
                a_scale = _resize(a_init, N)
            else:
                # Structured initialization: raised cosine
                x = np.linspace(0, 1, N)
                a_scale = 0.5 * (1 + np.cos(2 * np.pi * x))
                a_scale = np.maximum(a_scale, 0.01)
                a_scale /= a_scale.sum()
        else:
            a_scale = _resize(best_overall, N)

        print(f"Scale N={N}, L={L}, time_alloc={alloc:.1f}s, init_score={_score_norm(a_scale, L):.6f}")

        # Run optimization at this scale
        a_best, s_best = _optimize_at_scale(a_scale, L, alloc, rng,
                                            alpha0=0.15 if si == 0 else 0.10,
                                            lr0=0.010 if si == 0 else 0.008)
        print(f"  Scale N={N} done: score={s_best:.6f}")

        if s_best < best_score_overall:
            best_score_overall = s_best
            best_overall = a_best

        # Try LP escape if time permits
        rem = deadline_final - time.monotonic()
        if HAVE_LINPROG and rem > 60 and N <= 4096:
            lp_deadline = time.monotonic() + min(30, rem * 0.1)
            a_lp = _lp_escape(best_overall, L, lp_deadline)
            sc_lp = _score_norm(a_lp, L)
            if sc_lp < best_score_overall:
                best_score_overall = sc_lp
                best_overall = a_lp
                print(f"  LP escape improved: {sc_lp:.6f}")

    # --- Final fine-tuning at the best scale found ---
    rem = deadline_final - time.monotonic()
    if rem > 30 and best_overall is not None:
        N_best = len(best_overall)
        L_best = _next_pow2(2 * N_best - 1)
        print(f"Final polish: N={N_best}, time={rem:.1f}s, score={best_score_overall:.6f}")
        a_fin, s_fin = _optimize_at_scale(best_overall, L_best, rem * 0.9, rng,
                                          alpha0=0.05, lr0=0.004)
        if s_fin < best_score_overall:
            best_score_overall = s_fin
            best_overall = a_fin
            print(f"  Final polish improved: {s_fin:.6f}")

    # --- Build final answer ---
    if best_overall is None:
        best_overall = np.ones(8192) / 8192

    final_list = [float(max(0.0, min(1000.0, x))) for x in best_overall]

    # Validate
    try:
        sc_val = evaluate_sequence(final_list)
        print(f"Validated: {sc_val:.6f}")
    except Exception:
        print("Validation failed, returning as-is")

    return final_list
