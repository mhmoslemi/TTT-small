def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2*n -1
    m=1
    while m < size:
        m <<= 1
    fa = np.fft.rfft(a, m)
    ca = np.fft.irfft(fa*fa, m)
    ca = ca[:size]
    # remove negative roundoff
    if ca.size:
        ca = np.maximum(ca, 0.0)
    return ca

"""
Search for non-negative step-function heights minimizing

    2 * n * max(conv(a, a)) / sum(a)^2

by combining analytic low-peak candidates (especially one-sided ramps /
power-law ramps), FFT-accelerated internal scoring, and an active-set
multiplicative subgradient polish.  The best sequence found is rescaled so
its maximum height is 1 before being returned, which avoids any effect from
the evaluator's 1000.0 height cap.
"""

import numpy as np

def _fft_convolve(a):
    """Fast linear convolution of a with itself."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    m = 1
    while m < size:
        m <<= 1
    fa = np.fft.rfft(a, m)
    ca = np.fft.irfft(fa * fa, m)
    ca = ca[:size]
    if ca.size:
        ca = np.maximum(ca, 0.0)
    return ca


def _normalize(a):
    """Normalize a non-negative array to sum 1. Returns None on failure."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    a /= s
    return a


def _internal_score(a):
    """Fast internal objective using FFT convolution and sum normalization."""
    arr = np.asarray(a, dtype=np.float64).ravel()
    n = arr.size
    if n == 0:
        return np.inf
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.inf
    c = _fft_convolve(arr / s)
    if c.size == 0:
        return np.inf
    m = float(np.max(c))
    if not np.isfinite(m) or m < 0.0:
        return np.inf
    return 2.0 * n * m


def _to_output_list(a):
    """Convert a best array to a valid Python list of floats, max height = 1."""

def _resample_array(a, n):
    """Linearly resample a non-negative array to length n."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    m = a.size

        return a.copy()
    x0 = np.linspace(0.0, 1.0, m)
    x1 = np.linspace(0.0, 1.0, n)
    return np.interp(x1, x0, a)

def _weights_for_allocation(count):
    """Geometric time weights, best candidate gets most time."""
    if count <= 0:
        return []
    ws = []
    total = 0.0
    for i in range(count):
        w = float(count - i)
        ws.append(w)
        total += w
    return [w / total for w in ws]


def _optimize_active_subgrad(a0, time_budget, rng, name=""):
    """
    Multiplicative active-set subgradient optimization.

    The gradient of the current max convolution lags is computed; when several
    lags are near maximal, they are used with weights proportional to their
    values.  The update is exponentiated and renormalized.
    """
    a = _normalize(a0)
    if a is None:
        return np.asarray(a0, dtype=np.float64), np.inf

    n = a.size
    if n < 2:
        return a, _internal_score(a)

    if time_budget < 2.0:
        return a, _internal_score(a)

    c = _fft_convolve(a)
    best_score = 2.0 * n * float(np.max(c))
    best_a = a.copy()

    momentum = np.zeros(n, dtype=np.float64)
    mu = 0.55
    eta = 0.08
    no_improve = 0
    it = 0
    deadline = time.monotonic() + max(2.0, float(time_budget))
    last_print = time.monotonic()

    if n <= 1000:
        K = 16
    else:
        K = 8

    while time.monotonic() < deadline:
        c = _fft_convolve(a)
        M = float(np.max(c))
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            if time.monotonic() - last_print > 15.0:
                print("    [%s] it=%d score=%.7f best=%.7f" % (name, it, score, best_score))
                last_print = time.monotonic()
        else:
            no_improve += 1

        if c.size == 0:
            break

        if K < c.size:
            part = np.argpartition(-c, K - 1)[:K]
            act = part[np.argsort(-c[part])]
        else:
            act = np.arange(c.size)

        tol = max(1e-12, 0.01 * M)
        filtered = []
        for k in act:
            if c[k] >= M - tol:
                filtered.append(int(k))
        if not filtered:
            filtered = [int(np.argmax(c))]

        vals = c[filtered]
        svals = float(vals.sum())
        if not np.isfinite(svals) or svals <= 0.0:
            filtered = [int(np.argmax(c))]
            vals = c[filtered]
            svals = float(vals.sum())
            if svals <= 0.0:
                vals = np.array([1.0])
                svals = 1.0
        weights = vals / svals

        grad = np.zeros(n, dtype=np.float64)
        for idx, k in enumerate(filtered):
            lo = k - n + 1
            if lo < 0:
                lo = 0
            hi = k
            if hi > n - 1:
                hi = n - 1
            if lo <= hi:
                start = k - hi
                stop = k - lo + 1
                grad[lo:hi + 1] += weights[idx] * 2.0 * a[start:stop][::-1]

        gmax = float(np.max(grad))
        if not np.isfinite(gmax) or gmax <= 1e-30:
            break

        gnorm = grad / gmax
        momentum = mu * momentum + (1.0 - mu) * gnorm
        mmax = float(np.max(momentum))
        if np.isfinite(mmax) and mmax > 0.0:
            momentum /= mmax

        if float(np.max(momentum) - np.min(momentum)) < 1e-6:
            a = best_a * np.exp(rng.normal(0.0, 0.03, n))
            a = np.maximum(a, 1e-30)
            s = float(a.sum())
            if s > 0.0:
                a /= s
            else:
                a = best_a.copy()
            momentum = np.zeros(n, dtype=np.float64)
            no_improve = 0
            eta = 0.08
            it += 1
            continue

        a = a * np.exp(-eta * momentum)
        a = np.maximum(a, 1e-30)
        s = float(a.sum())
        if not np.isfinite(s) or s <= 1e-30:
            a = best_a.copy()
            eta *= 0.5
            it += 1
            continue
        a /= s

        if no_improve == 0:
            eta = min(0.35, eta * 1.004)
        elif no_improve > 60:
            eta = max(0.005, eta * 0.985)
        elif no_improve > 300:
            eta = max(0.003, eta * 0.97)

        if no_improve > 900:
            sigma = 0.04
            a = best_a * np.exp(rng.normal(0.0, sigma, n))
            a = np.maximum(a, 1e-30)
            s = float(a.sum())
            if s > 0.0:
                a /= s
            else:
                a = best_a.copy()
            momentum = np.zeros(n, dtype=np.float64)
            no_improve = 0
            eta = 0.08

        it += 1

    final_c = _fft_convolve(best_a)
    final_score = 2.0 * n * float(np.max(final_c))
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score

            if np.isfinite(sc):
                scored.append((sc, nm, arr.copy()))
    except Exception:
        scored = []

    if not scored:
        return _to_output_list(best_a)

    scored.sort(key=itemgetter(0))

    print("Initial candidate scores:")
    for i, (sc, nm, arr) in enumerate(scored[:10]):
        print("  %d. %s n=%d score=%.7f" % (i + 1, nm, arr.size, sc))

    if scored[0][0] < best_score:
        best_score = scored[0][0]
        best_a = scored[0][2].copy()
        best_name = scored[0][1]

    selected = _select_scored(scored, 10)

    remaining = deadline - time.monotonic()
    if remaining > 90.0:
        total_opt = min(1500.0, max(60.0, remaining - 180.0))
        wts = _weights_for_allocation(len(selected))

        for idx, item in enumerate(selected):
            if time.monotonic() > deadline - 90.0:
                break
            alloc = total_opt * wts[idx]
            if alloc < 15.0:
                continue
            if alloc > 360.0:
                alloc = 360.0
            sc0, nm, arr = item
            print("Optimizing %s (n=%d, init_score=%.7f) alloc=%.1fs" %
                  (nm, arr.size, sc0, alloc))
            try:
                a_opt, s_opt = _optimize_active_subgrad(arr, alloc, rng, nm)
            except Exception:
                a_opt = arr
                s_opt = sc0
            if np.isfinite(s_opt) and s_opt < best_score:
                best_score = s_opt
                best_a = a_opt.copy()
                best_name = nm
                print("    NEW BEST %s score=%.7f" % (nm, best_score))

    # Final polish of the global best.
    remaining = deadline - time.monotonic()
    if best_a is not None and remaining > 120.0:
        alloc = min(600.0, remaining - 60.0)
        print("Final polish for %s with %.1fs" % (best_name, alloc))
        try:
            a_pol, s_pol = _optimize_active_subgrad(best_a, alloc, rng, "polish")
        except Exception:
            a_pol = best_a
            s_pol = best_score
        if np.isfinite(s_pol) and s_pol < best_score:
            best_score = s_pol
            best_a = a_pol.copy()
            best_name = "polished"
            print("    POLISHED BETTER %.7f" % best_score)

    out_list = _to_output_list(best_a)

    # Optional final validation with the external evaluator if there is ample
    # time and the sequence is not so large that direct convolution is risky.
    remaining = deadline - time.monotonic()
    if remaining > 25.0 and len(best_a) <= 16000:
        eval_score = _safe_eval(out_list)
        print("External evaluation of final: %.7f" % eval_score)
        if np.isfinite(eval_score) and eval_score < best_score:
            best_score = eval_score

    return out_list
