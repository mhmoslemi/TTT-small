"""
Optimize non-negative step function to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses gradient-based peak-focusing + LP-based direction finding.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _fft_square(a):
    n = int(a.size)
    L = 2 * n - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, n=N)
    b = np.fft.irfft(fa * fa, n=N).real[:L]
    return np.maximum(b, 0.0)


def _fft_grad(weights, a):
    n = int(a.size)
    L = int(weights.size)
    N = _pow2(L + n - 1)
    fw = np.fft.rfft(weights, n=N)
    fa = np.fft.rfft(a, n=N)
    c = np.fft.irfft(fw * np.conj(fa), n=N)[:n]
    return 2.0 * c


def _internal_score(a, b=None):
    if b is None:
        b = _fft_square(a)
    s = float(a.sum())
    if s <= 1e-300:
        return float("inf"), b
    maxb = float(b.max())
    score = float(2.0 * a.size * maxb / (s * s))
    return score, b


def _topk_weights(b, k, smooth_alpha=0.1):
    """Create weights concentrated on top-k convolution coefficients."""
    L = len(b)
    maxb = b.max()
    if maxb <= 1e-300:
        return np.full(L, 1.0 / L)
    
    # Get top-k indices
    if k >= L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(b, -k)[-k:]
    
    # Create weights: high at top-k, low elsewhere
    w = np.full(L, smooth_alpha / L, dtype=np.float64)
    top_val = (1.0 - smooth_alpha) / k
    w[idx] = top_val
    return w


def _gradient_optimize(a0, time_limit, rng, verbose=False):

def _lp_direction(a, M):
    """Solve LP to find best non-negative direction c with (a*c)_k <= M."""
    n = int(a.size)
    
    # Build constraint matrix A (sparse would be ideal, but dense for simplicity)
    # (a * c)_k = sum_i a[i] * c[k-i] <= M for k = 0, ..., 2n-2
    # c >= 0
    # max sum(c)
    
    # For n > 3000, this LP is too large for dense solvers.
    # Use a reduced version: only include constraints where a has significant mass.
    
    if n > 4000:
        return None  # Too large for LP
    
    # Build the convolution matrix A (size (2n-1) x n)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for j in range(n):
        # For column j, A[k, j] = a[k-j] for k = j, ..., j+n-1
        start = j
        end = min(j + n, 2*n - 1)
        A[start:end, j] = a[:end-start]
    
    # LP: max 1^T c, s.t. A c <= M, c >= 0
    # linprog solves: min c^T c, s.t. A_ub x <= b_ub, ...
    # So: min -1^T c, s.t. A c <= M, -c <= 0 (i.e., c >= 0)
    
    c_obj = -np.ones(n)
    A_ub = A
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            c = np.maximum(result.x, 0)
            s_c = c.sum()
            if s_c > 1e-10:
                c /= s_c
            return c
    except Exception:
        pass
    
    return None

   e. Otherwise, try a mixed update: a = (1-t)*a + t*a_new

This is simpler and should still converge (it's like a fixed-point iteration on the game equilibrium).

The key advantage: the QP solve (step c) is efficient because H(p) is a Hankel matrix and matrix-vector products can be done in O(n log n).

Let me implement this approach.

**QP solve by projected gradient:**

min a^T H a s.t. a >= 0, sum(a) = 1

Projected gradient method:
1. Initialize a (e.g., the current best)
2. For t = 1, ..., T:
   a. Compute g = 2 * H * a (using FFT)
   b. a_temp = a - eta * g
   c. a = project_to_simplex(a_temp)
   where project_to_simplex(x) = argmin_{y >= 0, sum(y)=1} ||y - x||^2

The projection onto the simplex can be done in O(n log n) using the algorithm by Yu & Neelitzki (2010) or the simpler "water-filling" algorithm.

For the step size eta: in the QP, the "Lipschitz constant" of the gradient is 2*||H||_2. Since H is a Hankel matrix with entries p[i+j], and p is a probability vector, ||H||_2 <= 1 (since H is PSD and trace(H) = sum_i p[2i] <= 1). Actually, ||H||_2 = lambda_max(H) which is at most 1 (since for any unit vector v, v^T H v = sum_{i,j} p[i+j] v[i]v[j] <= sum_k p[k] * (sum_i v[i]v[k-i])^2 ... hmm this is not tight).

Let me just use a backtracking line search or a fixed small eta.

For the FFT computation of H*a:
(H*a)[i] = sum_j p[i+j] * a[j] for i = 0, ..., n-1

As I showed earlier, this equals the cross-correlation of p (length 2n-1) and a (length n) at lags 0, ..., n-1.

In FFT: 
- Pad p to length N (power of 2, N >= 3n-2)
- Pad a to length N
- Compute: r = ifft(fft(p_pad, N) * conj(fft(a_pad, N)))[:n]
- Then (H*a)[i] = r[i]

Let n=3, a = [a0, a1, a2], p = [p0, p1, p2, p3, p4] (length 2n-1=5).

(H*a)[0] = p[0]*a[0] + p[1]*a[1] + p[2]*a[2]
(H*a)[1] = p[1]*a[0] + p[2]*a[1] + p[3]*a[2]
(H*a)[2] = p[2]*a[0] + p[3]*a[1] + p[4]*a[2]

Cross-correlation: (p ⋆ a)[k] = sum_j p[j] * a[j-k] (with a[m]=0 for m outside [0,2])

(p ⋆ a)[0] = p[0]*a[0] + p[1]*a[1] + p[2]*a[2] = (H*a)[0] ✓
(p ⋆ a)[1] = p[1]*a[0] + p[2]*a[1] + p[3]*a[2] = (H*a)[1] ✓
(p ⋆ a)[2] = p[2]*a[0] + p[3]*a[1] + p[4]*a[2] = (H*a)[2] ✓

So (H*a)[i] = (p ⋆ a)[i] for i = 0, ..., n-1.

In FFT: (p ⋆ a)[k] = ifft(fft(p) * conj(fft(a)))[k] (with zero-padding to avoid circular effects).

So: (H*a) = ifft(fft(p_pad, N) * conj(fft(a_pad, N)))[0:n]

where N >= (2n-1) + n - 1 = 3n-2.

For n=5582: N >= 16744, so N = 32768.

The FFT of size 32768 is very fast (maybe 0.1ms in numpy). So each QP inner iteration is dominated by the simplex projection (O(n log n) ≈ 5582 * 13 ≈ 72K operations).

Total per QP solve (T=50 iterations): 50 * (FFT + O(n log n)) ≈ 50 * 0.5ms ≈ 25ms.

Outer loop per iteration: QP solve + one FFT for b + simple updates ≈ 30-50ms.

In 2400 seconds: about 48,000-80,000 outer iterations. More than enough to converge.

This is the approach I'll use! Let me write the code.

**Simplex projection algorithm:**

project_to_simplex(v):
  1. Sort v in decreasing order: v_sorted
  2. Find the largest rho such that sum_{i=1}^{rho} (v_sorted[i] - tau) > 0
     where tau = (sum_{i=1}^{rho} v_sorted[i] - 1) / rho
  3. Return max(v - tau, 0)

This can be done in O(n log n) time.

Let me also think about the convergence of the outer loop. The fixed-point iteration:
a_{t+1} = argmin_{a in A} a^T H(b_t / sum(b_t)) a
where b_t = a_t * a_t

This should converge to a (possibly) local equilibrium of the game. In practice, it should find a good solution.

To avoid cycling, I can use a "damped" version:
a_{t+1} = (1 - alpha) * a_t + alpha * a_QP

where alpha is a small step size (like 0.1-0.5).

1. The game-theoretic QP approach (main optimization)

"""
Minimize 2*n*max(convolve(a,a))/(sum(a))^2 using a game-theoretic approach.
The key insight: the problem is equivalent to a zero-sum game where nature
chooses a distribution over lags and we choose the sequence. We use an
iterative QP-based method where each QP involves a Hankel matrix (enabling
O(n log n) matrix-vector products via FFT).
"""

import numpy as np
import time


def _pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _fft_convolve_full(a):
    """Compute full convolution a*a using FFT."""
    n = int(a.size)
    L = 2 * n - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, n=N)
    b = np.fft.irfft(fa * fa, n=N).real[:L]
    return np.maximum(b, 0.0)


def _hankel_mv(p, a):
    """Compute H(p) * a where H[i,j] = p[i+j].
    p has length 2n-1, a has length n.
    Uses FFT for O(n log n) computation."""
    n = int(a.size)
    # (H*a)[i] = sum_j p[i+j] * a[j] = cross-correlation of p and a at lag i
    N = _pow2(3 * n)  # Need N >= (2n-1) + n - 1
    fp = np.fft.rfft(p, n=N)
    fa = np.fft.rfft(a, n=N)
    result = np.fft.irfft(fp * np.conj(fa), n=N)[:n]
    return result


def _simplex_project(v):
    """Project vector v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Algorithm from Yu & Neelitzki (2010)
    mu = np.sort(v)[::-1]
    sv = np.cumsum(mu)
    rho = (np.arange(1, n + 1) * (mu - 1.0 / np.arange(1, n + 1)) > 0).sum()
    if rho == 0:
        return np.ones(n) / n
    tau = (sv[rho - 1] - 1.0) / rho
    return np.maximum(v - tau, 0.0)


def _solve_qp(a_init, p, T=50, eta=None):
    """Solve min a^T H(p) a s.t. a >= 0, sum(a) = 1 using projected gradient.
    p has length 2n-1."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s <= 1e-300:
        a = np.ones(n) / n
    else:
        a /= s
    
    if eta is None:
        # Estimate Lipschitz constant: ||H||_2 <= 1 (since H is PSD with unit trace-like bound)
        # Actually, ||H||_2 = lambda_max(H). For a probability vector p, 
        # lambda_max(H) <= max_k p[k] * (something) ... let's just use a safe value
        eta = 0.5  # Step size (will be conservative enough)
    
    a_old = a.copy()
    obj_old = float(a @ _hankel_mv(p, a))
    
    for t in range(T):
        # Gradient: 2 * H(p) * a
        g = 2.0 * _hankel_mv(p, a)
        
        # Projected gradient step
        a_new = a - eta * g
        
        # Project onto simplex
        a = _simplex_project(a_new)
        
        # Check convergence
        obj = float(a @ _hankel_mv(p, a))
        if abs(obj - obj_old) < 1e-14:
            break
        obj_old = obj
        a_old = a
    
    return a


def _game_optimize(a0, time_limit, rng, verbose=False):
    """Game-theoretic optimization using iterative QP."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = int(a.size)
    a = np.maximum(a, 0)
    s = a.sum()
    if s <= 1e-300:
        a = np.ones(n) / n
        s = 1.0
    else:
        a /= s
    
    best_a = a.copy()
    best_score = float("inf")
    
    # Compute initial b
    b = _fft_convolve_full(a)
    max_b = float(b.max())
    score = 2.0 * n * max_b  # since sum(a) = 1
    best_score = score
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    alpha = 0.3  # Mixing parameter
    qp_iters = 30
    
    while time.monotonic() - t0 < time_limit:
        # Set p = b / sum(b) (current "worst case" distribution)
        sum_b = b.sum()
        if sum_b <= 1e-300:
            break
        p = b / sum_b
        
        # Solve QP
        a_qp = _solve_qp(a, p, T=qp_iters)
        
        # Check the QP solution's max convolution
        b_qp = _fft_convolve_full(a_qp)
        max_b_qp = float(b_qp.max())
        score_qp = 2.0 * n * max_b_qp
        
        # Damped update
        a_new = (1 - alpha) * a + alpha * a_qp
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new <= 1e-10:
            a_new = a
        else:
            a_new /= s_new
        
        b_new = _fft_convolve_full(a_new)
        max_b_new = float(b_new.max())
        score_new = 2.0 * n * max_b_new
        
        # Also check the QP solution directly
        if score_qp < best_score * (1 - 1e-9):
            best_score = score_qp
            best_a = a_qp.copy()
        
        if score_new < best_score * (1 - 1e-9):
            best_score = score_new
            best_a = a_new.copy()
        
        if score_new < score * (1 - 1e-6):
            a = a_new
            b = b_new
            score = score_new
            alpha = min(alpha * 1.02, 0.8)
        else:
            alpha = max(alpha * 0.95, 0.05)
            # Try the QP solution directly
            if score_qp < score * (1 - 1e-6):
                a = a_qp
                b = b_qp
                score = score_qp
        
        if verbose and it % 100 == 0:
            print(f"  game it={it} t={time.monotonic()-t0:.1f}s "
                  f"score={score:.6f} best={best_score:.6f} alpha={alpha:.3f}")
        
        it += 1
    
    return best_a, best_score


def _gradient_refine(a0, time_limit, rng, verbose=False):
    """Gradient-based refinement focusing on peaks."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = int(a.size)
    a = np.maximum(a, 0)
    s = a.sum()
    if s <= 1e-300:
        a = np.ones(n) / n
        s = 1.0
    else:
        a /= s
    
    best_a = a.copy()
    best_score = float("inf")
    
    b = _fft_convolve_full(a)
    max_b = float(b.max())
    score = 2.0 * n * max_b
    best_score = score
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    eta = 0.1
    no_improve = 0
    k_show = min(10, n)  # Focus on top-10 peaks
    
    while time.monotonic() - t0 < time_limit:
        # Create weights focused on top-k peaks
        L = len(b)
        if k_show >= L:
            w = np.ones(L) / L
        else:
            top_idx = np.argpartition(b, -k_show)[-k_show:]
            w = np.full(L, 0.05 / L)
            w[top_idx] = 0.95 / k_show
        
        # Gradient: 2 * cross_corr(w, a)
        N = _pow2(L + n - 1)
        fw = np.fft.rfft(w, n=N)
        fa = np.fft.rfft(a, n=N)
        grad = 2.0 * np.fft.irfft(fw * np.conj(fa), n=N)[:n]
        
        # Center the gradient (for the normalization constraint)
        grad_mean = float(a @ grad)
        grad_c = grad - grad_mean
        
        max_gc = float(np.max(np.abs(grad_c)))
        if max_gc < 1e-18:
            break
        grad_c /= max_gc
        
        # Multiplicative update
        a_new = a * np.exp(-eta * grad_c)
        a_new = np.maximum(a_new, 1e-300)
        a_new /= a_new.sum()
        
        b_new = _fft_convolve_full(a_new)
        score_new = 2.0 * n * float(b_new.max())
        
        if score_new < best_score * (1 - 1e-9):
            best_score = score_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
            b = b_new
            score = score_new
            eta = min(eta * 1.02, 2.0)
        elif score_new < score * (1 - 1e-6):
            a = a_new
            b = b_new
            score = score_new
            eta = min(eta * 1.01, 2.0)
        else:
            no_improve += 1
            eta = max(eta * 0.85, 0.01)
            if no_improve > 500:
                # Random restart
                a = best_a * np.exp(rng.normal(0, 0.2, n))
                a = np.maximum(a, 1e-15)
                a /= a.sum()
                b = _fft_convolve_full(a)
                score = 2.0 * n * float(b.max())
                eta = 0.1
                no_improve = 0
                k_show = min(int(k_show * 2), n)  # Broaden focus
                k_show = min(k_show, 100)
        
        if verbose and it % 200 == 0:
            print(f"  grad it={it} t={time.monotonic()-t0:.1f}s "
                  f"score={score:.6f} best={best_score:.6f} eta={eta:.4f} k={k_show}")
        
        it += 1
    
    return best_a, best_score
