def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def normalize_positive(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s


def fast_score(a):
    """Compute 2*n*max(conv(a,a)) for normalized a (sum=1)."""
    p = normalize_positive(a)
    if p is None:
        return np.inf
    n = p.size
    if n <= 0:
        return np.inf
    L = next_pow2(2 * n - 1)
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0.0:
        return np.inf
    return float(2.0 * n * cm)

Optimized step function search using central binomial coefficient initialization,

Key insight: p[k] = C(2k,k)/4^k produces a nearly-flat convolution (score ~ pi/2 ≈ 1.5708).

def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def normalize_positive(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s


def fast_score(a):
    p = normalize_positive(a)
    if p is None:
        return np.inf
    n = p.size
    if n <= 0:
        return np.inf
    L = next_pow2(2 * n - 1)
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0.0:
        return np.inf
    return float(2.0 * n * cm)


def central_binom_sequence(n):
    p = np.empty(n, dtype=np.float64)
    p[0] = 1.0
    for k in range(1, n):
        p[k] = p[k - 1] * (2 * k - 1) / (2.0 * k)
    return normalize_positive(p)


def double_sided_sequence(n):
    left = central_binom_sequence(n)
    right = left[::-1].copy()
    combined = left + right
    return normalize_positive(combined)


def u_shaped_sequence(n):
    k = np.arange(n, dtype=np.float64)
    p = 1.0 / np.sqrt(k + 1.0) + 1.0 / np.sqrt(float(n) - k)
    return normalize_positive(p)

    mid = (n - 1) / 2.0
    p = 1.0 / np.sqrt(np.abs(k - mid) + 1.0)
    return normalize_positive(p)


def resize_sequence(seq, m):
    p = normalize_positive(seq)
    m = int(m)
    if p is None:
        return np.ones(m, dtype=np.float64) / float(m)
    n = p.size
    if n == m:
        return p.copy()
    if n == 1:
        return np.full(m, 1.0 / float(m), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, m)
    out = np.interp(x_new, x_old, p)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(m, dtype=np.float64) / float(m)
    return out / s


def build_conv_matrix(p):
    n = p.size
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        s = max(0, k - n + 1)
        e = min(n - 1, k)
        A[k, s:e + 1] = p[k - e:k - s + 1][::-1]
    return A


def alt_lp_step(p_fixed, q_init, t_init):
    n = p_fixed.size
    A = build_conv_matrix(p_fixed)
    c_vec = np.zeros(n + 1, dtype=np.float64)
    c_vec[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1), dtype=np.float64)])
    b_ub = np.zeros(2 * n - 1, dtype=np.float64)
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1, dtype=np.float64)
    x0 = np.concatenate([q_init, [t_init]])
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=[(0.0, None)] * n + [(None, None)],
                  x0=x0, method='highs', options={'presolve': True, 'time_limit': 30})
    if res is not None and getattr(res, 'success', False) and getattr(res, 'x', None) is not None:
        q = np.maximum(res.x[:n], 0.0)
        sq = float(q.sum())
        if sq > 1e-15:
            q = q / sq
            t = float(res.x[-1])
            return q, t
    return None, np.inf


def alternating_lp(parent, deadline, n_lp=2048, max_iter=80):
    n_lp = int(n_lp)
    p = resize_sequence(parent, n_lp)
    q = p.copy()
    t_init = float((p * p).max() + 1e-6)

    best_p = p.copy()
    best_s = fast_score(p)
    no_improve = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        q_new, t = alt_lp_step(p, q, t_init)
        if q_new is None:
            break
        q = q_new

        p_new, t2 = alt_lp_step(q, p, t_init)
        if p_new is None:
            break
        p = p_new

        mixed = 0.5 * p + 0.5 * q
        mixed = normalize_positive(mixed)
        if mixed is not None:
            s = fast_score(mixed)
            if np.isfinite(s) and s < best_s - 1e-10:
                best_s = s
                best_p = mixed.copy()
                no_improve = 0
            else:
                no_improve += 1

        s_p = fast_score(p)
        s_q = fast_score(q)
        if np.isfinite(s_p) and s_p < best_s:
            best_s = s_p
            best_p = p.copy()
            no_improve = 0
        if np.isfinite(s_q) and s_q < best_s:
            best_s = s_q
            best_p = q.copy()
            no_improve = 0

        t_init = max(t, t2, (mixed * mixed).max() if mixed is not None else t_init)

        if (it + 1) % 10 == 0:
            print(f"  ALP iter {it+1}: best={best_s:.6f}")

        if no_improve >= 15:

    # dc[k]/dp[j] = p[k-j] * 1{0<=k-j<n} + p[k-j] * 1{0<=k-j<n} = 2*p[k-j]*1{0<=k-j<n}
    # BUT when k-j is out of range, the second term doesn't exist, and the first term p[k-j] also doesn't exist (since we're differentiating w.r.t. p[j], and the term p[i]*p[k-i] with i=j requires k-j to be a valid index).
    # So actually: dc[k]/dp[j] = p[k-j] if i=j is a valid term (i.e., 0<=k-j<n), PLUS p[k-j] if i=k-j is a valid term (i.e., 0<=k-j<n AND k-j != j).
    # If k-j is valid and k-j != j: dc[k]/dp[j] = p[k-j] + p[k-j] = 2*p[k-j]
    # If k-j is valid and k-j == j: dc[k]/dp[j] = 2*p[j] = 2*p[k-j] (from d(p[j]²)/dp[j])
    # If k-j is NOT valid: dc[k]/dp[j] = 0 (the term i=j doesn't exist in the sum)
    
    # So: dc[k]/dp[j] = 2 * p[k-j] if 0 <= k-j < n, else 0.
    # This means: dc/dp (as a matrix) has (dc/dp)[k,j] = 2*p[k-j] if valid.
    # This is exactly 2 times the convolution matrix A! (A[k,j] = p[k-j] if valid)
    
    # So the gradient of sum_k w[k]*c[k] w.r.t. p is: 2 * A^T * w
    # where (A^T * w)[j] = sum_k A[k,j] * w[k] = sum_k p[k-j] * w[k] (for valid k)
    # = (w * p)[j] in convolution terms... actually it's the cross-correlation.
    # (A^T w)[j] = sum_k p[k-j] w[k] = sum_m p[m] w[k] where m = k-j, k = m+j
    # = sum_m p[m] w[m+j] = (w reversed cross-correlated with p)... 
    # Actually this is just the convolution of w with reversed p, evaluated at specific points.
    # Let me just use FFT: A^T * w can be computed as the convolution w * p_rev, shifted appropriately.
    
    # For the log-parameterization p = softmax(z), dp/dz_i = p_i * (delta_ij - p_j)
    # So dz-grad_j = sum_i (dc/dp_i) * dp_i/dz_j = sum_i 2*(A^T w)[i] * p_i * (delta_ji - p_i)
    # = 2*(A^T w)[j] * p_j - 2*p_j * sum_i (A^T w)[i] * p_i
    # = 2*p_j * ((A^T w)[j] - sum_i p_i * (A^T w)[i])
    
    # Let v = A^T * w (computed via FFT)
    # Then grad_z[j] = 2 * p[j] * (v[j] - dot(p, v))
    
    # Compute v = A^T * w using FFT:
    # v[j] = sum_k A[k,j] * w[k] = sum_k p[k-j] * w[k] (for valid k-j)
    # = sum_{k: 0<=k-j<n} p[k-j] * w[k] = sum_{m: 0<=m<n} p[m] * w[m+j] (substituting m=k-j)
    # This is the cross-correlation of w with p, at lag j.
    # xcorr(w, p)[j] = sum_m w[m+j] * p[m]
    # = ifft(FFT(w) * conj(FFT(p))) ... but that's circular.
    # For linear: conv(w_rev, p) or conv(w, p_rev) gives the cross-correlation.
    # xcorr(w, p)[j] = conv(w_flipped, p)[j + n - 1] ... this is getting complicated.
    # Let me just use: v = convolve(w, p_flipped)[n-1 : n-1+n] or something.
    # Actually: sum_m p[m] * w[m+j] for m=0..n-1 (where m+j < len(w))
    # = sum_m p[m] * w[m+j]
    # If I define w_pad as w zero-padded, this is (p * w_pad)[n-1+j] ... no.
    # Let me just compute it via FFT directly:
    # Linear conv of p (length n) and w (length 2n-1) has length 3n-2.
    # (p * w)[k] = sum_i p[i] * w[k-i]
    # I want v[j] = sum_m p[m] * w[m+j] = sum_i p[i] * w[i+j] = (p * w_rev)... 
    # Hmm, let me just use the direct convolution approach:
    # v[j] = sum_{m=0}^{n-1} p[m] * w[m+j] where m+j < 2n-1
    # This is the "cross-correlation" at lag j.
    # In numpy: np.correlate(w, p, mode='valid') gives sum_m w[m+j]*p[m] for j=0..(len(w)-n)
    # len(w) = 2n-1, so 'valid' gives 2n-1 - n + 1 = n values. 
    # But I need it for j = 0, ..., n-1.
    # np.correlate(w, p, 'valid') = [sum_m w[m]*p[m], sum_m w[m+1]*p[m], ..., sum_m w[m+n-1]*p[m]]
    # That's v[0], v[1], ..., v[n-1]. 
    # Wait, is it? np.correlate(a, v, 'valid') = sum_m a[m+k]*v[m] for k=0..(len(a)-len(v))
    # So np.correlate(w, p, 'valid')[j] = sum_m w[m+j]*p[m] for j=0..(2n-2-(n-1)) = 0..(n-1). ✓
    
    # But this is O(n²) for direct computation. For n=8192, that's 67 million ops which is fine.
    # Alternatively use FFT for speed.
    
    # Actually for n=8192, np.correlate should be fast enough (uses FFT internally for large arrays).
    # But let me use the FFT approach to be safe:
    # v = ifft(FFT(w_padded) * conj(FFT(p_padded))) ... circular correlation
    # For linear correlation, I need to zero-pad.
    
    # Let me just use a simpler approach: compute v via the convolution matrix structure.
    # v[j] = sum_k p[k-j] * w[k] (for valid k-j, i.e., 0<=k-j<n)
    # = (w * p_rev)[2n-2-j] where p_rev = p[::-1]
    # Because (w * p_rev)[k] = sum_i w[i] * p_rev[k-i] = sum_i w[i] * p[n-1-(k-i)] = sum_i w[i] * p[n-1-k+i]
    # Let j = n-1-k, so k = n-1-j: (w*p_rev)[n-1-j] = sum_i w[i] * p[j+i] ... not quite right.
    
    # I'll just compute v directly using the FFT of the full convolution structure.
    # Easiest: v = np.correlate(w, p, mode='valid')[:n] -- but this might be slow for n=8192.
    # Actually scipy uses FFT for correlate, so it should be fine.
    
    # Let me just use: v = np.correlate(w, p, 'valid')
    # This gives sum_m w[m+j]*p[m] for j=0..(len(w)-len(p)) = 0..(n-1). Exactly what I need.
    
    v = np.correlate(w, p, mode='valid')[:n]
    if v.size < n:
        v = np.pad(v, (0, n - v.size))
    
    dot_pv = float(np.dot(p, v))
    grad_z = 2.0 * p * (v - dot_pv)
    
    return grad_z


def nesterov_refine(parent, duration_s, tau0=0.02, lr0=0.01):
    p = normalize_positive(parent)
    if p is None:
        return None, np.inf
    n = p.size
    L = next_pow2(3 * n - 2)
    deadline = time.monotonic() + max(0.0, float(duration_s))

    z = np.log(np.maximum(p, 1e-300))
    z = np.where(np.isfinite(z), z, -50.0)
    z -= z.mean()
    z = np.clip(z, -50.0, 50.0)

    y = z.copy()
    momentum = 0.0
    mu = 0.9
    lr = float(lr0)
    tau = float(tau0) / float(n)

    best_p = p.copy()
    best_score = fast_score(p)
    no_improve = 0
    step = 0

    while time.monotonic() < deadline:
        # Compute current p from z
        p = _softmax(z)
        
        # Score
        true_s = _score_from_p(p)
        if not np.isfinite(true_s):
            break

        if true_s < best_score - 1e-12:
            best_score = true_s
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Compute gradient
        L = next_pow2(2 * n - 1)
        F = np.fft.rfft(p, L)
        c = np.fft.irfft(F * F, L)[:2 * n - 1]
        c = np.where(np.isfinite(c), c, 0.0)
        c = np.maximum(c, 0.0)
        cm = float(c.max())
        
        # Smooth weight: emphasize peaks
        z_c = (c - cm) / max(tau, 1e-10)
        z_c = np.clip(z_c, -40.0, 0.0)
        exp_z = np.exp(z_c)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            exp_z = np.zeros_like(c)
            exp_z[int(np.argmax(c))] = 1.0
            exp_sum = 1.0
        w = exp_z / exp_sum

        # Gradient via correlate
        v = np.correlate(w, p, mode='valid')[:n]
        if v.size < n:
            v = np.pad(v, (0, n - v.size))
        dot_pv = float(np.dot(p, v))
        grad = 2.0 * p * (v - dot_pv)
        
        # Also account for the mean constraint (z mean should stay 0)
        grad -= grad.mean()
        
        # Nesterov step
        momentum = mu * momentum + (1 - mu) * grad
        z_new = y - lr * momentum
        z_new -= z_new.mean()
        z_new = np.clip(z_new, -50.0, 50.0)
        
        # Check if this improves
        p_new = _softmax(z_new)
        s_new = _score_from_p(p_new)
        
        if np.isfinite(s_new) and s_new < true_s + 0.01:
            # Accept step
            z = z_new
            mu = min(0.999, mu + 0.001)
        else:
            # Reject: use current z
            z = y - lr * momentum  # retry without momentum update
            z -= z.mean()
            z = np.clip(z, -50.0, 50.0)
            lr = max(0.001, lr * 0.95)
            mu = max(0.5, mu - 0.05)

        y = z.copy()
        step += 1

        # Anneal
        if step % 2000 == 0:
            lr = max(0.002, lr * 0.92)
            tau = max(0.001 / n, tau * 0.85)

        if no_improve > 5000:
            # Reset with noise
            z = np.log(np.maximum(best_p, 1e-300))
            z = z + np.random.default_rng(step).normal(0, 0.1, n)
            z -= z.mean()
            z = np.clip(z, -50.0, 50.0)
            y = z.copy()
            momentum = np.zeros(n)
            lr = max(0.003, lr * 0.8)
            no_improve = 0

        if step % 2000 == 0:
            print(f"  nesting step {step}: true={true_s:.6f} best={best_score:.6f} lr={lr:.5f} tau={tau* n:.6f}")

    return best_p, best_score


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - z.max()
    z = np.clip(z, -50.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0:
        return np.ones_like(z) / z.size
    return e / s


def _score_from_p(p):
    n = p.size
    L = next_pow2(2 * n - 1)
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0:
        return np.inf
    return 2.0 * n * cm


def compare_update(p, s, best_p, best_score):
    if p is None or not np.isfinite(s):
        return best_p, best_score
    if best_p is None or not np.isfinite(best_score) or s < best_score:
        return p, float(s)
    return best_p, best_score
