"""
Optimize a non-negative step function sequence to minimize
    2 * n * max(convolve(a, a, mode='full')) / (sum(a))**2.

Strategy:
1. Start from the given sequence (n=5582) with improved gradient optimization
   using momentum, adaptive focus with annealing, and structured restarts.

3. Final local search refinement on the best candidate found.

import numpy as np
import time
import math


def _pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _fft_square(a):
    n = int(a.size)
    L = 2 * n - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, n=N)
    b = np.fft.irfft(fa * fa, n=N).real[:L]
    b = np.maximum(b, 0.0)
    return b


def _fft_grad(weights, a):
    n = int(a.size)
    L = int(weights.size)
    N = _pow2(L + n - 1)
    fw = np.fft.rfft(weights, n=N)
    fa = np.fft.rfft(a, n=N)
    c = np.fft.irfft(fw * np.conj(fa), n=N)[:n]
    return 2.0 * c


def _internal_score(a, b=None):
    if b is None:
        b = _fft_square(a)
    s = float(a.sum())
    if s <= 1e-300:
        return float("inf"), b, 0.0
    maxb = float(b.max())
    score = float(2.0 * a.size * maxb / (s * s))
    return score, b, maxb


def _make_weights(b, focus, eps=0.01):
    L = int(b.size)
    maxb = float(b.max())
    if maxb <= 1e-300:
        return np.full(L, 1.0 / L, dtype=np.float64), maxb
    with np.errstate(over="ignore", under="ignore"):
        z = (maxb - b) / maxb
        p = np.exp(-z * float(focus))
    s = float(p.sum())
    if s <= 1e-12:
        p = np.zeros(L, dtype=np.float64)
        p[int(np.argmax(b))] = 1.0
    else:
        p = (1.0 - eps) * (p / s) + eps / L
    return p, maxb


def _random_positive(n, rng, amp=3.0):

    z = rng.normal(0.0, amp, size=n)
    z -= z.max()
    a = np.exp(z)
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    return a

                2.0 * np.pi * f * x / m + rng.uniform(0.0, 2.0 * np.pi)
            )
        v += 0.1 * rng.normal(0.0, 1.0, m)
    else:
        v = rng.normal(0.0, 1.0, m)
    v = v - float(v.mean())
    std = float(v.std())
    if std > 1e-12:
        v = (float(amp) * v) / std

    z = np.zeros(n, dtype=np.float64)
    z[:m] = v
    if n % 2 == 0:
        z[m:] = -v[::-1]
    else:
        z[m + 1:] = -v[::-1]
        z[m] = 0.0

    z = z - z.max()
    a = np.exp(z)
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    return a


def _spectral_init(n, rng):
    """Initialize using a random phase spectrum approach."""
    n = int(n)
    N = _pow2(2 * n - 1)
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    mag = np.ones(N // 2 + 1, dtype=np.float64)
    F = mag * np.exp(1j * phases)
    a = np.fft.irfft(F, n=N)[:n]
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s > 1e-12:
        a /= s
    else:
        a = _random_positive(n, rng)
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    return a


def _optimize_main(a0, time_limit, rng, verbose=False, label=""):
    """
    Improved gradient optimizer with momentum, adaptive focus,
    simulated annealing, and structured restarts.
    """
    a = np.asarray(a0, dtype=np.float64).copy()
    n = int(a.size)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-300:
        a = np.ones(n, dtype=np.float64) / max(1, n)
    else:
        a /= s
    a = np.maximum(a, 1e-15)
    a /= a.sum()

    b = None
    score = float("inf")
    best_score = float("inf")
    best_a = a.copy()

    focus = 5.0
    focus_inc = 1.015
    eta = 0.5
    eta_min = 0.01
    eta_max = 10.0

    momentum = 0.7
    vm = np.zeros(n, dtype=np.float64)

    temperature = 0.1
    temp_decay = 0.9995
    temp_min = 0.001

    no_improve = 0
    max_no = 800
    resets = 0
    max_resets = 40
    t0 = time.monotonic()
    it = 0
    best_focus = 5.0

    while time.monotonic() - t0 < float(time_limit):
        if time.monotonic() - t0 > float(time_limit):
            break

        if b is None:
            b = _fft_square(a)
            score, _, _ = _internal_score(a, b)

        if score < best_score - 1e-13:
            prev_best = best_score
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and (it % 500 == 0):
            print(
                f"[{label}] it={it} t={time.monotonic()-t0:.1f}s "
                f"score={score:.6f} best={best_score:.6f} "
                f"focus={focus:.1f} eta={eta:.3f} T={temperature:.4f} resets={resets}"
            )

        if no_improve > max_no:
            resets += 1
            if resets > max_resets:
                break
            no_improve = 0
            max_no = int(max_no * 1.2)

            restart_type = rng.integers(0, 4)
            if restart_type == 0:
                # Blend with random
                fresh = _random_positive(n, rng, amp=4.0)
                alpha = 0.3 + 0.3 * rng.random()
                a = (1 - alpha) * best_a + alpha * fresh
            elif restart_type == 1:
                # Multiplicative perturbation
                a = best_a * np.exp(rng.normal(0.0, 0.5, size=n))
            elif restart_type == 2:
                # Local segment perturbation
                a = best_a.copy()
                seg_len = rng.integers(max(10, n // 50), n // 5)
                start = rng.integers(0, n - seg_len)
                a[start:start + seg_len] = _random_positive(seg_len, rng, amp=2.0) * a[start:start + seg_len].sum() / seg_len
            else:
                # Spectral re-init blended with best
                fresh = _spectral_init(n, rng)
                alpha = 0.4 + 0.3 * rng.random()
                a = (1 - alpha) * best_a + alpha * fresh

            a = np.maximum(a, 1e-15)
            a /= a.sum()
            b = None
            score = float("inf")
            focus = float(rng.choice([3.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0]))
            eta = 0.5
            vm = np.zeros(n, dtype=np.float64)
            temperature = max(temperature, 0.05)
            continue

        # Compute weights and gradient
        p, _ = _make_weights(b, focus)
        grad = _fft_grad(p, a)
        gd = grad - float(np.dot(a, grad))

        # Momentum update
        maxg = float(np.max(np.abs(gd)))
        if maxg < 1e-18:
            break
        gd = gd / maxg

        vm = momentum * vm + (1 - momentum) * gd
        direction = -vm

        # Step size with adaptive adjustment
        a_new = a * np.exp(eta * direction)
        a_new = np.maximum(a_new, 1e-300)
        ss = float(a_new.sum())
        if ss <= 1e-300:
            eta = max(eta * 0.5, eta_min)
            continue
        a_new = a_new / ss

        # Evaluate
        b_new = _fft_square(a_new)
        score_new, _, _ = _internal_score(a_new, b_new)

        # Acceptance criterion with simulated annealing
        delta = score_new - score
        accept = False
        if score_new < best_score - 1e-13:
            accept = True
        elif score_new < score - 1e-13:
            accept = True
        elif delta < 0.05 * temperature * score:
            # Simulated annealing: accept worse with probability
            if rng.random() < math.exp(-delta / max(temperature * score, 1e-12)):
                accept = True

        # Also check weighted objective
        wt_old = float(np.dot(p, b))
        wt_new = float(np.dot(p, b_new))
        if not accept and score_new < score * 1.005 and wt_new < wt_old * 0.999:
            accept = True

        if accept:
            a = a_new
            b = b_new
            score = score_new
            eta = min(eta * 1.08, eta_max)
        else:
            eta = max(eta * 0.7, eta_min)
            if eta < 0.1:
                # Small random kick
                a = a * np.exp(rng.normal(0.0, 0.03, size=n))
                a = np.maximum(a, 1e-15)
                a /= a.sum()
                b = None
                score = float("inf")
                vm = np.zeros(n, dtype=np.float64)

        # Update focus
        focus = min(focus * focus_inc, 2000.0)

        # Anneal temperature
        temperature = max(temperature * temp_decay, temp_min)

        # Periodic focus reset for exploration
        if it > 0 and it % 2000 == 0 and rng.random() < 0.3:
            focus = float(rng.choice([5.0, 10.0, 30.0, 80.0, 200.0]))
            vm = np.zeros(n, dtype=np.float64)

        it += 1

    _ = best_a.copy()
    return best_a, float(best_score)


def _optimize_topk(a0, time_limit, rng, k=5, verbose=False, label=""):
    """
    Optimizer that focuses on top-k convolution coefficients using
    sorted weights for a smoother but more targeted objective.
    """
    a = np.asarray(a0, dtype=np.float64).copy()
    n = int(a.size)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-300:
        a = np.ones(n, dtype=np.float64) / max(1, n)
    else:
        a /= s
    a = np.maximum(a, 1e-15)
    a /= a.sum()

    b = None
    score = float("inf")
    best_score = float("inf")
    best_a = a.copy()

    eta = 0.4
    eta_min = 0.01
    eta_max = 8.0
    momentum = 0.6
    vm = np.zeros(n, dtype=np.float64)

    no_improve = 0
    max_no = 1000
    resets = 0
    max_resets = 30
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < float(time_limit):
        if b is None:
            b = _fft_square(a)
            score, _, _ = _internal_score(a, b)

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and (it % 500 == 0):
            print(
                f"[{label}] topk it={it} t={time.monotonic()-t0:.1f}s "
                f"score={score:.6f} best={best_score:.6f} eta={eta:.3f} resets={resets}"
            )

        if no_improve > max_no:
            resets += 1
            if resets > max_resets:
                break
            no_improve = 0
            max_no = int(max_no * 1.3)
            a = best_a * np.exp(rng.normal(0.0, 0.3, size=n))
            a = np.maximum(a, 1e-15)
            a /= a.sum()
            b = None
            score = float("inf")
            eta = 0.4
            vm = np.zeros(n, dtype=np.float64)
            continue

        # Sort-based weights: emphasize top-k
        L = int(b.size)
        maxb = float(b.max())
        if maxb > 1e-300:
            sorted_b = np.sort(b)[::-1]
            threshold = max(sorted_b[min(k - 1, L - 1)] - maxb * 0.1, 0)
            p = np.where(b > threshold, (b - threshold) / maxb, 0.0) ** 2
            p = p + 0.01 / L
            p /= p.sum()
        else:
            p = np.full(L, 1.0 / L)

        grad = _fft_grad(p, a)
        gd = grad - float(np.dot(a, grad))
        maxg = float(np.max(np.abs(gd)))
        if maxg < 1e-18:
            break
        gd = gd / maxg

        vm = momentum * vm + (1.0 - momentum) * gd
        a_new = a * np.exp(-eta * vm)
        a_new = np.maximum(a_new, 1e-300)
        ss = float(a_new.sum())
        if ss <= 1e-300:
            break
        a_new /= ss

        b_new = _fft_square(a_new)
        score_new, _, _ = _internal_score(a_new, b_new)

        if score_new < score - 1e-13 or score_new < best_score - 1e-13:
            a = a_new
            b = b_new
            score = score_new
            eta = min(eta * 1.06, eta_max)
        else:
            eta = max(eta * 0.72, eta_min)
            if eta < 0.08:
                a = a * np.exp(rng.normal(0.0, 0.04, size=n))
                a = np.maximum(a, 1e-15)
                a /= a.sum()
                b = None
                score = float("inf")
                vm = np.zeros(n, dtype=np.float64)

        it += 1

    return best_a, float(best_score)


def _local_search(a0, time_limit, rng, label=""):
    """Perturbation-based local search on the best candidate."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = int(a.size)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        return a, float("inf")
    a /= s

    b = _fft_square(a)
    score, _, _ = _internal_score(a, b)
    best_score = score
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    base_noise = 0.05

    while time.monotonic() - t0 < float(time_limit):
        # Try different perturbation types
        pert_type = rng.integers(0, 5)
        a_try = best_a.copy()

        if pert_type == 0:
            # Single element adjustment
            idx = rng.integers(0, n)
            factor = 1.0 + rng.normal(0.0, 0.3)
            a_try[idx] = max(0.0, a_try[idx] * factor)
        elif pert_type == 1:
            # Small segment swap
            seg1 = rng.integers(0, n - 10)
            seg2 = rng.integers(0, n - 10)
            length = rng.integers(3, min(20, n // 4))
            a_try = a_try.copy()
            tmp = a_try[seg1:seg1 + length].copy()
            a_try[seg1:seg1 + length] = a_try[seg2:seg2 + length]
            a_try[seg2:seg2 + length] = tmp
        elif pert_type == 2:
            # Gaussian noise on subset
            subset = rng.choice(n, size=max(5, n // 20), replace=False)
            noise = rng.normal(1.0, 0.1, size=len(subset))
            a_try[subset] = np.maximum(a_try[subset] * noise, 0.0)
        elif pert_type == 3:
            # Two-element exchange
            i1, i2 = rng.choice(n, size=2, replace=False)
            a_try[i1], a_try[i2] = a_try[i2], a_try[i1]
        else:
            # Broad small perturbation
            a_try = a_try * np.exp(rng.normal(0.0, base_noise, size=n))

        a_try = np.maximum(a_try, 0.0)
        s_try = float(a_try.sum())
        if s_try <= 1e-300:
            continue
        a_try /= s_try

        b_try = _fft_square(a_try)
        score_try, _, _ = _internal_score(a_try, b_try)

        if score_try < best_score - 1e-13:
            best_score = score_try
            best_a = a_try.copy()
            base_noise = max(base_noise * 0.99, 0.01)
        else:
            base_noise = min(base_noise * 1.02, 0.15)

        it += 1

        if it % 5000 == 0:
            print(f"[{label}] local it={it} t={time.monotonic()-t0:.1f}s best={best_score:.6f}")

    return best_a, float(best_score)


def _optimize_pair(x0, time_limit, rng, verbose=False, label=""):
    """Optimize in the anti-palindromic pair parameterization."""
    x = np.asarray(x0, dtype=np.float64).copy()
    m = int(x.size)
    n = 2 * m
    x_eps = 1e-8
    x = np.clip(x, x_eps, 1.0 - x_eps)

    a = _a_from_x(x, n)
    b = None
    score = float("inf")
    best_score = float("inf")
    best_x = x.copy()

    focus = 5.0
    focus_inc = 1.02
    eta = 0.4
    eta_min = 0.01
    eta_max = 10.0
    momentum = 0.6
    vm = np.zeros(m, dtype=np.float64)
    temperature = 0.05
    temp_decay = 0.999

    no_improve = 0
    max_no = 1000
    resets = 0
    max_resets = 30
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < float(time_limit):
        if b is None:
            b = _fft_square(a)
            score, _, _ = _internal_score(a, b)

        if score < best_score - 1e-13:
            best_score = score
            best_x = x.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and (it % 1000 == 0):
            print(
                f"[{label}] pair it={it} t={time.monotonic()-t0:.1f}s "
                f"score={score:.6f} best={best_score:.6f} focus={focus:.1f} eta={eta:.3f}"
            )

        if no_improve > max_no:
            resets += 1
            if resets > max_resets:
                break
            no_improve = 0
            max_no = int(max_no * 1.25)
            x = 0.6 * best_x + 0.4 * rng.uniform(0.1, 0.9, size=m)
            x = np.clip(x, x_eps, 1.0 - x_eps)
            a = _a_from_x(x, n)
            b = None
            score = float("inf")
            focus = float(rng.choice([3.0, 10.0, 30.0, 100.0, 300.0]))
            eta = 0.4
            vm = np.zeros(m, dtype=np.float64)
            continue

        p, _ = _make_weights(b, focus)
        ga = _fft_grad(p, a)
        gx = ga[:m] - ga[m:][::-1]
        maxg = float(np.max(np.abs(gx)))
        if maxg < 1e-18:
            break
        gx = gx / maxg

        vm = momentum * vm + (1.0 - momentum) * gx
        x_new = x - eta * vm
        x_new = np.clip(x_new, x_eps, 1.0 - x_eps)

        a_new = _a_from_x(x_new, n)
        b_new = _fft_square(a_new)
        score_new, _, _ = _internal_score(a_new, b_new)

        delta = score_new - score
        accept = score_new < score - 1e-13 or score_new < best_score - 1e-13
        if not accept and delta < 0.03 * temperature * score:
            if rng.random() < math.exp(-delta / max(temperature * score, 1e-12)):
                accept = True

        if accept:
            x = x_new
            a = a_new
            b = b_new
            score = score_new
            eta = min(eta * 1.06, eta_max)
        else:
            eta = max(eta * 0.72, eta_min)
            if eta < 0.08:
                x = np.clip(x + rng.normal(0.0, 0.05, size=m), x_eps, 1.0 - x_eps)
                a = _a_from_x(x, n)
                b = None
                score = float("inf")
                vm = np.zeros(m, dtype=np.float64)

        focus = min(focus * focus_inc, 1500.0)
        temperature = max(temperature * temp_decay, 0.001)
        it += 1

    return _a_from_x(best_x, n), float(best_score)


def _a_from_x(x, n=None):
    x = np.asarray(x, dtype=np.float64).ravel()
    m = x.size
    if n is None:
        n = 2 * m
    t = 1.0 / m
    a = np.empty(n, dtype=np.float64)
    a[:m] = t * x
    a[m:] = t * (1.0 - x[::-1])
    return a


def _safe_evaluate(seq):
    ev = globals().get("evaluate_sequence")
    if ev is None or not callable(ev):
        return float("inf")
    try:
        return float(ev(seq))
    except Exception:
        return float("inf")


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    seed = int(seed)
    total = max(60.0, float(budget_s))
    now0 = time.monotonic()
    final_deadline = now0 + total - 10.0

    print("starting improved step-function convolution optimizer")
    best_a = None
    best_score = float("inf")

    # Load given sequence
    given_a = None
    try:
        given = globals().get("height_sequence_1")
        if given is not None:
            g = np.asarray(given, dtype=np.float64).ravel()
            g = np.nan_to_num(g, nan=0.0, posinf=1000.0, neginf=0.0)
            g = np.clip(g, 0.0, 1000.0)
            s = float(g.sum())
            if s > 0.01 and g.size > 10:
                gn = g / s
                gn = np.maximum(gn, 1e-15)
                gn /= gn.sum()
                sc, _, _ = _internal_score(gn)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = gn.copy()
                given_a = gn
                print(f"given internal score: {sc:.6f}, n={gn.size}")
    except Exception:
        given_a = None

    if given_a is None:
        given_a = _random_positive(5582, np.random.default_rng(seed))
        given_a = np.maximum(given_a, 1e-15)
        given_a /= given_a.sum()

    # Phase 1: Main optimization of given sequence (most time)
    n_given = given_a.size
    main_time = min(1200.0, total * 0.55)
    if time.monotonic() < final_deadline - main_time - 60:
        alloc = main_time
        task_rng = np.random.default_rng((seed * 7919 + 1) % (2**32))
        print(f"Phase 1: main optimizer on n={n_given}, time={alloc:.0f}s")
        t0 = time.monotonic()
        try:
            cand_a, cand_score = _optimize_main(
                given_a, alloc, task_rng,

                best_score = cand_score
                best_a = cand_a.copy()
                print(f"Phase 1 done: score={cand_score:.6f}")
        except Exception as e:
            print(f"Phase 1 failed: {repr(e)}")
        print(f"Phase 1 used {time.monotonic()-t0:.1f}s")

    # Phase 2: Top-k optimizer on best
    if best_a is not None and time.monotonic() < final_deadline - 300:
        alloc = min(400.0, final_deadline - time.monotonic() - 200)
        if alloc > 60:
            task_rng = np.random.default_rng((seed * 104729 + 2) % (2**32))

            t0 = time.monotonic()
            try:

                    best_a, alloc, task_rng, k=5,

                if cand_a is not None and np.isfinite(cand_score) and cand_score < best_score:
                    best_score = cand_score
                    best_a = cand_a.copy()
                    print(f"Phase 2 done: score={cand_score:.6f}")
            except Exception as e:
                print(f"Phase 2 failed: {repr(e)}")
            print(f"Phase 2 used {time.monotonic()-t0:.1f}s")

    # Phase 3: Try alternative approach (different init or size)
    if time.monotonic() < final_deadline - 200:
        alloc = min(300.0, final_deadline - time.monotonic() - 150)
        if alloc > 60:
            task_rng = np.random.default_rng((seed * 15485863 + 3) % (2**32))
            print(f"Phase 3: alternative approach, time={alloc:.0f}s")
            t0 = time.monotonic()

            # Try spectral init with full optimizer
            try:
                a_alt = _spectral_init(n_given, task_rng)
                cand_a, cand_score = _optimize_main(

                    print(f"Phase 3 spectral: score={cand_score:.6f}")

            remaining_alloc = alloc - (time.monotonic() - t0)
            if remaining_alloc > 60:
                try:
                    # Try anti-palindromic pair on best
                    m = n_given // 2
                    if n_given % 2 == 0:
                        a_cur = best_a.copy() if best_a is not None else given_a
                        x0 = a_cur[:m].copy()
                        x0 = np.clip(x0 * m, 1e-8, 1.0 - 1e-8)
                        x0 = np.clip(x0, 1e-8, 1.0 - 1e-8)
                        cand_a, cand_score = _optimize_pair(
                            x0, remaining_alloc * 0.6, task_rng,
                            verbose=False, label="pair"
                        )
                        if cand_a is not None and np.isfinite(cand_score) and cand_score < best_score:
                            best_score = cand_score
                            best_a = cand_a.copy()
                            print(f"Phase 3 pair: score={cand_score:.6f}")
                except Exception as e:
                    print(f"Phase 3 pair failed: {repr(e)}")

            print(f"Phase 3 used {time.monotonic()-t0:.1f}s")

    # Phase 4: Local search refinement
    if best_a is not None and time.monotonic() < final_deadline - 50:
        alloc = min(200.0, final_deadline - time.monotonic() - 30)
        if alloc > 30:
            task_rng = np.random.default_rng((seed * 2749 + 4) % (2**32))
            print(f"Phase 4: local search, time={alloc:.0f}s")
            t0 = time.monotonic()
            try:
                cand_a, cand_score = _local_search(
                    best_a, alloc, task_rng, label="local"
                )
                if cand_a is not None and np.isfinite(cand_score) and cand_score < best_score:
                    best_score = cand_score
                    best_a = cand_a.copy()
                    print(f"Phase 4 done: score={cand_score:.6f}")
            except Exception as e:
                print(f"Phase 4 failed: {repr(e)}")

    # Phase 5: Final gradient refinement
    if best_a is not None and time.monotonic() < final_deadline - 20:
        alloc = max(20.0, final_deadline - time.monotonic() - 10)
        if alloc > 30:
            task_rng = np.random.default_rng((seed * 99991 + 5) % (2**32))
            print(f"Phase 5: final refinement, time={alloc:.0f}s")
            t0 = time.monotonic()
            try:
                cand_a, cand_score = _optimize_main(
                    best_a, alloc, task_rng,
                    verbose=False, label="final"
                )
                if cand_a is not None and np.isfinite(cand_score) and cand_score < best_score:
                    best_score = cand_score
                    best_a = cand_a.copy()
                    print(f"Phase 5 done: score={cand_score:.6f}")
            except Exception as e:
                print(f"Phase 5 failed: {repr(e)}")

    # Final processing
    if best_a is None:
        best_a = np.ones(1024, dtype=np.float64) / 1024.0

    best_a = np.nan_to_num(best_a, nan=0.0, posinf=1000.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    ss = float(best_a.sum())
    if ss <= 0.01:
        best_a = np.ones(1024, dtype=np.float64) / 1024.0
        ss = 1.0
    else:
        best_a /= ss

    final_seq = [float(x) for x in best_a]
    print(f"final internal score={best_score:.6f} n={best_a.size}")

    # Verify with exact evaluation if possible
    if time.monotonic() < final_deadline - 5:
        exact = _safe_evaluate(final_seq)
        if np.isfinite(exact):
            print(f"exact evaluation: {exact:.6f}")
        else:
            print(f"exact evaluation unavailable")

    return final_seq
