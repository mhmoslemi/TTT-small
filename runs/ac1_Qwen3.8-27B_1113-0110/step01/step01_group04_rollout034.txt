def _build_conv_matrix_pattern(n):
    """Build the sparsity pattern for the convolution matrix (fixed for all a)."""
    row_idx = np.zeros(n * n, dtype=np.int32)
    col_idx = np.zeros(n * n, dtype=np.int32)
    for j in range(n):
        for i in range(n):
            k = j + i
            idx = j * n + i
            row_idx[idx] = k
            col_idx[idx] = j
    return row_idx, col_idx

def _make_conv_matrix(a, row_idx, col_idx):
    """Create the sparse convolution matrix for a given sequence a."""
    n = len(a)
    data = np.tile(a, n)  # Column j has values a[0], a[1], ..., a[n-1]
    return csc_array((data, (row_idx, col_idx)), shape=(2*n-1, n))

def _score(a):
    """Compute 2*n*max(a*a)/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    n = len(a)
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (s * s)

def _fft_score(a):
    """FFT-based score computation."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    n = len(a)
    L = 1
    while L < 2*n-1:
        L *= 2
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)[:2*n-1]
    return 2.0 * n * c.max() / (s * s)

"""
LP-based iterative optimization for minimizing 2*n*max(a*a)/(sum(a))^2.
Uses a sequence of LP subproblems (max sum(b) s.t. (a*b)<=M, b>=0) to find
improvement directions, combined with diverse initialization and local search.

import numpy as np
import time
from scipy.optimize import linprog, minimize
from scipy.sparse import csc_array, csr_array


def _pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _clean(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    return np.maximum(v, 0.0)


def _normalize(v):
    v = _clean(v)
    s = float(v.sum())
    if s < 1e-15:
        return np.full(v.size, 1.0 / v.size)
    return v / s


def _project_simplex(v):
    v = _clean(v)
    n = v.size
    if n == 0:
        return v.copy()
    x = np.sort(v)[::-1]
    css = np.cumsum(x)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = x > (css - 1.0) / idx
    if not np.any(cond):
        return np.full(n, 1.0 / float(n))
    r = int(np.nonzero(cond)[0][-1])
    tau = (css[r] - 1.0) / float(r + 1)
    out = v - tau
    np.maximum(out, 0.0, out=out)
    s = float(out.sum())
    if s < 1e-15:
        return np.full(n, 1.0 / float(n))
    return out / s


def _fft_score(a):
    v = _clean(a)
    n = v.size
    s = float(v.sum())
    if s < 1e-12:
        return np.inf
    v = v / s
    L = _pow2(2 * n - 1)
    A = np.fft.rfft(v, n=L)
    c = np.fft.irfft(A * A, n=L)[:2 * n - 1]
    mx = float(c.max()) if c.size else 0.0
    if not np.isfinite(mx):
        return np.inf
    return 2.0 * float(n) * mx


def _fft_conv(a):
    n = a.size
    L = _pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    return np.fft.irfft(A * A, n=L)[:2 * n - 1]


def _build_pattern(n):
    total = n * n
    row_idx = np.empty(total, dtype=np.int32)
    col_idx = np.empty(total, dtype=np.int32)
    for j in range(n):
        base = j * n
        for i in range(n):
            row_idx[base + i] = j + i
            col_idx[base + i] = j
    return row_idx, col_idx


def _make_conv_mat(a, row_idx, col_idx):

    data = np.tile(a, n)
    return csc_array((data, (row_idx, col_idx)), shape=(2 * n - 1, n))


def _solve_lp(a, M, row_idx, col_idx, timeout=15.0):

    A = _make_conv_mat(a, row_idx, col_idx)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': timeout})
        if res.x is not None:
            b = np.maximum(res.x, 0.0)
            s = float(b.sum())
            if s > 1e-10:
                return b / s
    except Exception:
        pass
    try:

            b = np.maximum(res.x, 0.0)
            s = float(b.sum())
            if s > 1e-10:

        pass
    return None

            dist = np.minimum(j, n - 1 - j)
            v = np.power(dist + off, -p)
            cands.append(v / v.sum())

    for freq in (1, 2, 3):
        for amp in (0.3, 0.5, 0.7, 0.9):
            v = 1.0 + amp * np.cos(2 * np.pi * freq * j / n)
            v = np.maximum(v, 0.01)
            cands.append(v / v.sum())

    for amp in (0.3, 0.5, 0.7):
        v = 1.0 + amp * np.cos(2 * np.pi * j / (2 * n))
        v = np.maximum(v, 0.01)
        cands.append(v / v.sum())

    rng = np.random.default_rng(seed)
    for s in (seed, seed + 1, seed + 2, seed + 3, seed + 4):
        rr = np.random.default_rng(s)
        v = np.exp(0.5 * rr.standard_normal(n))
        v *= np.power(j + 1.0, -rr.uniform(0.2, 0.8))
        cands.append(v / v.sum())

    return cands


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 15.0)

    n = 3809
    try:
        h = height_sequence_1
        if isinstance(h, (list, np.ndarray)):
            arr = np.asarray(h, dtype=np.float64)
            if arr.ndim == 1 and 200 <= arr.size <= 20000:
                n = int(arr.size)
    except Exception:
        pass

    print(f"n = {n}, budget = {budget_s}s")

    if deadline - t0 < 10.0:
        j = np.arange(n, dtype=np.float64)
        a = np.power(j + 1.0, -0.5)
        return [float(x) for x in a / a.sum()]

    # Build fixed sparsity pattern
    row_idx, col_idx = _build_pattern(n)
    print(f"Pattern built: {n*n} nonzeros")

    # Generate and score initial candidates
    cands = _generate_candidates(n, seed)
    scored = []
    for i, a in enumerate(cands):
        sc = _fft_score(a)
        if np.isfinite(sc):
            scored.append((sc, i, a))
    scored.sort()
    print(f"Top initial candidates: {[f'{s:.6f}' for s, _, _ in scored[:5]]}")

    global_best = np.asarray(scored[0][2], dtype=np.float64).copy()
    global_best_score = float(scored[0][0])

    # Allocate time: 60% to best candidate LP, 25% to second, 15% to third
    n_top = min(3, len(scored))
    alloc_fracs = [0.55, 0.25, 0.20]

    for i in range(n_top):
        if time.monotonic() >= deadline - 10.0:
            break
        if i == n_top - 1:
            stage_end = deadline - 10.0
        else:
            frac_used = sum(alloc_fracs[:i + 1])
            stage_end = t0 + frac_used * (deadline - 10.0 - t0)
            stage_end = min(stage_end, deadline - 10.0)

        if stage_end <= time.monotonic() + 2.0:
            continue

        a_init = np.asarray(scored[i][2], dtype=np.float64).copy()
        print(f"\nStage {i+1}: LP optimization (init score={scored[i][0]:.8f})")
        print(f"  Time window: {time.monotonic()-t0:.0f}s to {stage_end-t0:.0f}s")

        a_opt, sc_opt = _lp_iteration(a_init, stage_end, row_idx, col_idx)
        if sc_opt < global_best_score:
            global_best_score = sc_opt
            global_best = a_opt.copy()
            print(f"  New global best: {global_best_score:.8f}")

    # Final gradient polish
    if time.monotonic() < deadline - 20.0:
        pol_end = min(deadline - 20.0, time.monotonic() + 120.0)
        print(f"\nGradient polish phase...")
        a_pol, sc_pol = _grad_descent_polish(global_best, pol_end)
        if sc_pol < global_best_score:
            global_best_score = sc_pol
            global_best = a_pol.copy()
            print(f"  After polish: {global_best_score:.8f}")

    # Final local search
    if time.monotonic() < deadline - 5.0:
        print(f"\nLocal search phase...")
        a_ls, sc_ls = _local_search(global_best, deadline - 2.0, seed + 7777)
        if sc_ls < global_best_score:
            global_best_score = sc_ls
            global_best = a_ls.copy()
            print(f"  After local search: {global_best_score:.8f}")

    global_best = _project_simplex(global_best)
    out = [float(x) for x in global_best]

    try:
        val = evaluate_sequence(out)
        print(f"\nFinal evaluated score: {val:.8f}")
        if not np.isfinite(val):
            j = np.arange(n, dtype=np.float64)
            fb = np.power(j + 1.0, -0.5)
            out = [float(x) for x in fb / fb.sum()]
    except Exception:
        pass

    print(f"Total time: {time.monotonic()-t0:.0f}s")
    return out
