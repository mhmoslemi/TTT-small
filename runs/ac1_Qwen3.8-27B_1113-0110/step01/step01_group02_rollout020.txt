"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using LP-based iterations (from the
'previous state of the art'), softmax subgradient polishing, and multiple
initializations. The LP approach solves a linear program at each step to find
the best direction to move, guaranteeing non-increasing objective values.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _fft_convolve(a):
    """Fast self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]


def _fft_cross(a, b):
    """Fast cross-convolution using FFT. len(a)=n, len(b)=m, result has n+m-1 elements."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    fb = np.fft.rfft(b, fft_size)
    return np.fft.irfft(fa * fb, fft_size)[:size]


def _build_conv_matrix(f):
    """Build A where (A @ b)[k] = (f*b)[k]. A shape: (2n-1, n)."""
    n = len(f)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    mask = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[mask] = f[diff[mask]]
    return A


def _lp_iteration(f, M):
    """Solve LP: max sum(b) s.t. (f*b)[k] <= M for all k, b >= 0.
    Returns b (the optimal solution) or None if LP fails."""
    n = len(f)
    A = _build_conv_matrix(f)
    c = -np.ones(n)  # minimize -sum(b)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0)
    return None


def _optimize_lp(a0, time_budget, rng, verbose=True):
    """LP-based optimization following the 'previous state of the art' approach."""
    n = len(a0)
    target_sum = np.sqrt(2.0 * n)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, 2.0 * n * float(np.max(_fft_convolve(a0)))
    a = a / s * target_sum
    
    c = _fft_convolve(a)
    M = float(np.max(c))
    best_a = a.copy()
    best_M = M
    
    deadline = time.monotonic() + time_budget
    it = 0
    last_print = time.monotonic()
    
    while time.monotonic() < deadline:
        if time.monotonic() - last_print > 15.0 and verbose:
            print(f"    [LP] it={it}, M={M:.6f}, score={M:.6f}")
            last_print = time.monotonic()
        
        # Solve LP
        b = _lp_iteration(a, M)
        if b is None:
            if verbose:
                print(f"    [LP] LP failed at it={it}")
            break
        
        sum_b = b.sum()
        if sum_b < target_sum * (1 + 1e-8):
            # No improvement possible
            if verbose:
                print(f"    [LP] No improvement at it={it}, sum_b={sum_b:.4f} <= target={target_sum:.4f}")
            break

        g = b / sum_b * target_sum

        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            a_new = (1 - t) * a + t * g
            a_new = np.maximum(a_new, 0)
            c_new = _fft_convolve(a_new)
            M_new = float(np.max(c_new))
            if M_new < M - 1e-12:
                a = a_new
                M = M_new
                improved = True
                break
        
        if not improved:
            # Try smaller steps or perturb
            if it % 10 == 5:
                # Perturb and retry
                a = a + rng.normal(0, 0.01 * target_sum / n, n)
                a = np.maximum(a, 0)
                a = a / a.sum() * target_sum
                c = _fft_convolve(a)
                M = float(np.max(c))
                if M > best_M:
                    a = best_a.copy()
                    M = best_M
            else:
                break
        
        if M < best_M:
            best_M = M
            best_a = a.copy()
        
        # Check convergence
        if it > 0 and abs(M - best_M) < 1e-10:
            # Check if we've been stuck
            pass  # The line search not improving will break us out
        
        it += 1
    
    return best_a, best_M
