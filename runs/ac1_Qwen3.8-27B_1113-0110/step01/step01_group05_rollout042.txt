import numpy as np
import time


def _normalize(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, None)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(max(1, a.size), dtype=np.float64) / max(1, a.size)
    return a / s


def _score(a):
    n = len(a)
    b = np.convolve(a, a, mode='full')
    max_b = float(np.max(b))
    if not np.isfinite(max_b):
        return float('inf')
    return 2.0 * n * max_b


def _fft_correlate(w, a):
    """Compute g_j = sum_k w[k] * a[k-j] for j = 0,...,len(a)-1."""
    n = len(a)
    fft_size = 1
    while fft_size < 3 * n:
        fft_size <<= 1
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(W * np.conj(A), fft_size)
    return c[:n]


def _soft_max_gradient(a, b, tau):
    """Gradient of soft_max(b(a), tau) w.r.t. a."""
    n = len(a)
    b_scaled = b / tau
    b_max = np.max(b_scaled)
    exp_b = np.exp(b_scaled - b_max)
    w = exp_b / np.sum(exp_b)
    g = 2.0 * _fft_correlate(w, a)
    return g


def _adam_optimize(a, deadline, tau_init, tau_final, n_epochs, steps_per_epoch, lr):

    n = len(a)
    a = _normalize(a.copy())

    t_global = 0
    
    for epoch in range(n_epochs):
        if time.monotonic() > deadline:
            break
        # Anneal tau
        frac = epoch / max(1, n_epochs - 1)
        tau = tau_init * (tau_final / tau_init) ** frac
        
        lr_epoch = lr * (1.0 - 0.5 * frac)  # slight lr decay
        
        for step in range(steps_per_epoch):
            if time.monotonic() > deadline:
                break
            b = np.convolve(a, a, mode='full')
            g = _soft_max_gradient(a, b, tau)
            
            t_global += 1
            m = 0.9 * m + 0.1 * g
            v = 0.999 * v + 0.001 * g * g
            m_hat = m / (1.0 - 0.9 ** t_global)
            v_hat = v / (1.0 - 0.999 ** t_global)
            
            a_new = a - lr_epoch * m_hat / (np.sqrt(v_hat) + 1e-8)
            a_new = _normalize(a_new)
            
            sc = _score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            a = a_new
        
        print(f"  Epoch {epoch+1}/{n_epochs}, tau={tau:.6f}, best_score={best_score:.6f}")

"""
Minimizer for 2*n*max(convolve(a,a))/(sum(a))^2 using smooth-max gradient descent
with Adam optimizer, tau annealing, FFT-based efficient gradients, coordinate descent
refinement with incremental convolution updates, and multiple restarts.
Targets improvement from current best ~1.555 toward 1.5027.
"""

import numpy as np
import time


def _normalize(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, None)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(max(1, a.size), dtype=np.float64) / max(1, a.size)
    return a / s


def _score(a):
    n = len(a)
    b = np.convolve(a, a, mode='full')
    max_b = float(np.max(b))
    if not np.isfinite(max_b):
        return float('inf')
    s = float(np.sum(a))
    if s <= 0:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


def _fft_convolve_self(a):
    """FFT-based self-convolution of a."""
    n = len(a)
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:2 * n - 1]


def _fft_correlate(w, a):
    """Compute g_j = sum_k w[k] * a[k-j] for j = 0,...,len(a)-1.
    w has length 2n-1, a has length n."""
    n = len(a)
    fft_size = 1
    while fft_size < 3 * n:
        fft_size <<= 1
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(W * np.conj(A), fft_size)
    return c[:n]


def _soft_max_gradient(a, b, tau):
    """Gradient of soft_max(b(a), tau) w.r.t. a, where b = convolve(a,a)."""
    n = len(a)
    b_scaled = b / tau
    b_max = np.max(b_scaled)
    exp_b = np.exp(b_scaled - b_max)
    w = exp_b / np.sum(exp_b)
    g = 2.0 * _fft_correlate(w, a)
    return g


def _adam_optimize(a, deadline, tau_init, tau_final, total_steps, base_lr):
    """Adam optimization with tau annealing for smooth max objective."""
    n = len(a)
    a = _normalize(a.copy())
    b = _fft_convolve_self(a)
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0

    # Precompute fft sizes
    fft_size_b = 1
    while fft_size_b < 2 * n - 1:
        fft_size_b <<= 1
    fft_size_g = 1
    while fft_size_g < 3 * n:
        fft_size_g <<= 1

    for step in range(total_steps):
        if time.monotonic() > deadline:
            break
        t += 1
        frac = t / float(total_steps)
        tau = tau_init * (tau_final / tau_init) ** frac
        lr = base_lr * (1.0 - 0.3 * frac)

        g = _soft_max_gradient(a, b, tau)

        m = 0.9 * m + 0.1 * g
        v = 0.999 * v + 0.001 * g * g
        m_hat = m / (1.0 - 0.9 ** t)
        v_hat = v / (1.0 - 0.999 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = _normalize(a_new)
        b = _fft_convolve_self(a_new)
        sc = float(2.0 * n * np.max(b))
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

        if step % 2000 == 0:
            print(f"    Adam step {step}/{total_steps}, tau={tau:.8f}, score={best_score:.6f}")

    return best_a, best_score


def _coordinate_descent(a, deadline, max_passes=15, grid_size=15):
    """Coordinate descent with incremental b updates."""
    n = len(a)
    a = _normalize(a.copy())
    b = _fft_convolve_self(a)
    s = 1.0
    score = 2.0 * n * float(np.max(b))
    best_a = a.copy()
    best_score = score

    b_len = 2 * n - 1

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            old = float(a[i])
            D = s - old

            # Build candidate grid
            if old > 1e-12:
                ratios = np.logspace(-2.5, 2.5, grid_size)
                cands = np.unique(np.concatenate(([0.0], old * ratios, [D])))
            else:
                cands = np.logspace(-8, np.log10(max(D, 1e-8)), grid_size)
                cands = np.unique(np.concatenate(([0.0], cands, [D])))

            best_val = score
            best_x = old

            # Precompute affected indices
            j_range = np.arange(n)
            k_range = i + j_range
            valid_mask = k_range < b_len
            j_valid = j_range[valid_mask]
            k_valid = k_range[valid_mask]

            for x in cands:
                delta = float(x) - old
                if abs(delta) < 1e-15:
                    continue
                s_new = s + delta
                if s_new <= 1e-12:
                    continue

                # Incremental b update
                db = 2.0 * delta * a[j_valid]
                # Fix the self-term (k=2i)
                if 0 <= i < n:
                    idx_self = np.where(j_valid == i)[0]
                    if idx_self.size > 0:
                        db[idx_self[0]] = delta * (float(x) + old)

                b[k_valid] += db

                max_b = float(np.max(b))
                val = 2.0 * n * max_b / (s_new * s_new)

                b[k_valid] -= db

                if val < best_val:
                    best_val = val
                    best_x = float(x)

            if best_x != old:
                delta = best_x - old
                s += delta
                j_range = np.arange(n)
                k_range = i + j_range
                valid_mask = k_range < b_len
                j_valid = j_range[valid_mask]
                k_valid = k_range[valid_mask]

                db = 2.0 * delta * a[j_valid]
                if 0 <= i < n:
                    idx_self = np.where(j_valid == i)[0]
                    if idx_self.size > 0:
                        db[idx_self[0]] = delta * (best_x + old)
                b[k_valid] += db
                a[i] = best_x
                score = best_val
                improved = True

                if score < best_score - 1e-12:
                    best_score = score
                    best_a = a.copy()

        if not improved:
            break
        print(f"  CD pass {p+1}/{max_passes}, score={best_score:.6f}")

    return best_a, best_score


def _perturb_and_optimize(a, rng, deadline, tau_init, tau_final, n_steps, lr):
    """Perturb a and run Adam optimization."""
    n = len(a)
    # Multiplicative perturbation
    scale = 0.1 + 0.2 * rng.random()
    a_pert = a * np.exp(rng.normal(0.0, scale, n))
    a_pert = _normalize(a_pert)
    result_a, result_score = _adam_optimize(a_pert, deadline, tau_init, tau_final, n_steps, lr)
    return result_a, result_score


def _make_ramp(n, power, rng):
    """Create a power-ramp initialization."""
    x = np.arange(1, n + 1, dtype=np.float64)

def _make_gaussian(n, center_frac, width_frac, rng):
    """Create a Gaussian bump initialization."""
    x = np.linspace(0, 1, n)

    width = width_frac
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    a += 0.01 * rng.random(n)
    return _normalize(a)


def _make_two_peaks(n, pos1, pos2, frac, rng):
