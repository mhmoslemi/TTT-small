    """Compute np.correlate(w, a, 'valid') via FFT."""
    n = len(a)
    m = len(w)
    # We want: out[k] = sum_{j=0}^{n-1} w[j+k] * a[j] for k=0,...,m-n
    # This is the cross-correlation of w with a at lags 0,1,...,m-n
    # equivalently: conv(a_rev, w)[n-1 : n-1+(m-n+1)]
    # Or via FFT: IFFT(FFT(a) * conj(FFT(w))) but padded
    
    size = max(m, n)
    fft_size = _next_pow2(size + n - 1)  # need at least m+n-1 for linear conv
    # Actually for correlation: cross_corr [lag] = sum_j a[j]*w[j+lag]
    # = conv(a, reversed(w))[... ]
    
    # Let's use: cross_corr = ifft(fft(a, N) * conj(fft(w, N))) for N >= m+n-1
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    corr = np.fft.irfft(A * np.conj(W), fft_size)
    # corr[k] = sum_j a[j] * w[(j-k) mod N] (circular correlation)
    # For linear: corr[k] for k=0,...,n-1 gives sum_j a[j]*w[j-k] (with w out of range = 0)
    # What we want: out[k] = sum_j a[j]*w[j+k] for k=0,...,m-n
    # = corr_opp[k] where corr_opp[k] = sum_j a[j]*w[j+k]
    # 
    # np.correlate(w, a, 'valid')[k] = sum_{j=0}^{n-1} w[j+k]*a[j]

    # Let's think of it as: rev_a = a[::-1]
    # conv(rev_a, w)[n-1+k] = sum_j rev_a[j]*w[n-1+k-j] = sum_j a[n-1-j]*w[n-1+k-j]
    # Let i = n-1-j: = sum_i a[i]*w[k+i]  ✓
    
    rev_a = a[::-1]
    R = np.fft.rfft(rev_a, fft_size)
    out = np.fft.irfft(R * W, fft_size)[:m - n + 1]

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Uses L-BFGS-B with log-sum-exp smooth max approximation, LP-based diversification,
and multi-harmonic/tapered initializations. Combines gradient-based optimization
with structured perturbations to escape local minima.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]

def _score(a):
    s = a.sum()
    if s < 1e-15:
        return np.inf
    b = _fft_convolve(a)
    return 2.0 * len(a) * float(np.max(b)) / (s * s)

def _smooth_obj_and_grad(a, beta):
    b = _fft_convolve(a)
    bm = b.max()
    shifted = beta * (b - bm)
    # Clip for numerical stability
    shifted = np.clip(shifted, -700, 700)
    exp_b = np.exp(shifted)
    exp_sum = exp_b.sum()
    if exp_sum < 1e-300:
        return 1e10, np.zeros_like(a)
    w = exp_b / exp_sum
    logsum = np.log(exp_sum) + beta * bm
    n = len(a)
    # Gradient of log(sum(exp(beta*b)))/beta w.r.t. a
    # d/d a[m] = 2 * sum_j a[j] * w[j+m] (valid indices)
    # This is 2 * correlate(w, a, 'valid')
    w_corr = np.correlate(w, a, mode='valid')
    grad_inner = 2.0 * w_corr  # gradient of smoothmax w.r.t. a
    obj = logsum / beta
    S = a.sum()
    # Full gradient of 2*n*smoothmax/sum(a)^2
    # dF/da[m] = 2n * (grad_inner[m] / S^2 - 2*smoothmax * 1 / S^3)
    # = 2n * grad_inner[m] / S^2 - 4n * obj / S^3
    F = 2.0 * n * obj
    grad = 2.0 * n * grad_inner / (S * S) - 4.0 * n * obj / (S * S * S)
    return F, grad

def _lbfgs_optimize(a0, beta, time_budget, rng):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, np.inf
    a = a / s

    best_a = a.copy()
    best_score = _score(a)
    start = time.monotonic()

    def fun(x):
        x = np.maximum(x, 0.0)
        s = x.sum()
        if s < 1e-15:
            return 1e10, np.zeros_like(x)
        x = x / s
        F, g = _smooth_obj_and_grad(x, beta)
        return F, g

    def jac(x):
        x = np.maximum(x, 0.0)
        s = x.sum()
        if s < 1e-15:
            return np.zeros_like(x)
        x = x / s
        F, g = _smooth_obj_and_grad(x, beta)
        return g

    n_iter = 0
    while time.monotonic() - start < time_budget:
        res = minimize(fun, a, method='L-BFGS-B', jac=jac,
                       bounds=[(0, None)] * n,
                       options={'maxiter': 300, 'maxcor': 20,
                                'gtol': 1e-6, 'maxls': 50, 'disp': False})
        n_iter += 1
        a_new = np.maximum(res.x, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        sc = _score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
        if time.monotonic() - start >= time_budget:
            break
        if res.status in [0, 4]:
            # Converged or max iterations - try with slightly different beta
            beta = min(beta * 1.5, 200.0)
        # Add small perturbation for diversity
        if n_iter > 2 and rng.random() < 0.3:
            a = a * np.exp(rng.normal(0, 0.02, n))
            a = np.maximum(a, 1e-15)
            a = a / a.sum()

    return best_a, best_score

def _subgrad_polish(a0, time_budget, rng):
    """Final polishing using exact subgradient on the true max."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, np.inf
    a = a / s

    best_a = a.copy()
    best_score = _score(a)
    eta = 0.05
    deadline = time.monotonic() + time_budget
    momentum = np.zeros(n)

    while time.monotonic() < deadline:
        b = _fft_convolve(a)
        k_max = int(np.argmax(b))
        sc = 2.0 * n * b[k_max]
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        # Subgradient: 2 * a shifted
        g = np.zeros(n)
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        if lo <= hi:
            j_range = np.arange(lo, hi + 1)
            g[lo:hi + 1] = 2.0 * a[k_max - j_range]
        g_max = g.max()
        if g_max < 1e-15:
            break
        g = g / g_max
        momentum = 0.8 * momentum + 0.2 * g
        a = a - eta * momentum
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            break
        a = a / s
        eta = max(eta * 0.995, 0.001)

    return best_a, best_score

def _lp_step(f, M, n):
    """Solve max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    from scipy.optimize import linprog
    # Variables: g[0], ..., g[n-1]
    # max: sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    # Constraints: conv(f,g)[k] <= M for k=0,...,2n-2
    # conv(f,g)[k] = sum_j f[j]*g[k-j] (valid indices)
    A_ub = np.zeros((2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A_ub[k, j] = f[k - j]
    # Actually this is O(n^2) to build. For large n, use sparse.
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A_sp = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    res = linprog(c, A_ub=A_sp, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs', options={'disp': False})
    if res.success:
        return res.x
    return None

def _make_sinusoidal(n, amplitudes, phases, beta_taper=0.0, rng=None):
    """Multi-harmonic sinusoidal with optional edge taper."""
    x = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    for k, (A, phi) in enumerate(zip(amplitudes, phases)):
        a += A * np.sin(2.0 * np.pi * (k + 1) * x / n + phi)
    if beta_taper > 0:
        t = x / n
        a *= (1.0 - beta_taper * (t - 0.5) ** 2)
    a = np.maximum(a, 0.01)
    return a

def _make_tapered_sinusoidal(n, A1, phi1, A2, phi2, taper_power, rng):
    x = np.arange(n, dtype=np.float64)
    t = x / (n - 1.0)
    a = 1.0 + A1 * np.sin(2.0 * np.pi * x / n + phi1)
    if abs(A2) > 1e-6:
        a += A2 * np.sin(4.0 * np.pi * x / n + phi2)
    # Taper: reduce at edges
    taper = 1.0 - (1.0 - 0.3) * t ** taper_power
    a *= taper
    a = np.maximum(a, 0.01)
    return a

            best_overall = a_cand.copy()
            print(f"  [update] {label}: score={ev:.8f}, n={len(a_cand)}")

    # Phase 1: Quick L-BFGS-B on multiple initializations
    print("=== Phase 1: Multi-init L-BFGS-B ===")
    phase1_end = total_start + 500.0

    # Get provided sequence if available
    provided = None
    try:
        hs = np.array(height_sequence_1, dtype=np.float64)
        hs = np.clip(hs, 0, None)
        if hs.size > 100 and hs.sum() > 0.01:
            provided = hs / hs.sum()
    except (NameError, Exception):
        pass

    inits = []
    # Provided sequence
    if provided is not None:
        inits.append(("provided", provided))

    # Multi-harmonic sinusoids
    for n in [2000, 3000, 5000]:
        for _ in range(3):
            A1 = rng.uniform(0.2, 0.6)
            phi1 = rng.uniform(0, 2 * np.pi)
            A2 = rng.uniform(-0.2, 0.2)
            phi2 = rng.uniform(0, 2 * np.pi)
            a = _make_sinusoidal(n, [A1, A2], [phi1, phi2])
            inits.append((f"mharm_{n}_{len(inits)}", a))

    # Tapered sinusoids
    for n in [2000, 3000, 5000, 8000]:
        for taper_p in [1.5, 2.0, 3.0]:
            A1 = 0.4
            phi1 = np.pi / 4
            A2 = 0.15
            phi2 = np.pi / 3
            a = _make_tapered_sinusoidal(n, A1, phi1, A2, phi2, taper_p, rng)
            inits.append((f"taper_{n}_{taper_p}", a))

    # Ramp-like
    for n in [3000, 5000]:
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-0.5 * x / n) + 0.1
        inits.append((f"decay_{n}", a))

    # Quick score all inits
    scored = []
    for name, a0 in inits:
        a0 = np.maximum(a0, 0)
        s = a0.sum()
        if s < 1e-15:
            continue
        a0 = a0 / s
        sc = _score(a0)
        scored.append((sc, name, a0))
    scored.sort(key=lambda x: x[0])
    print(f"  {len(scored)} initializations. Top 5 scores:")
    for i, (sc, name, _) in enumerate(scored[:5]):
        print(f"    {i+1}. {name}: {sc:.6f} (n={len(_)})")

    # Time allocation for Phase 1
    p1_time = min(phase1_end - time.monotonic(), 500.0)
    n_top = min(len(scored), 6)
    # Give more time to top candidates
    weights = [max(1.0, n_top - i) for i in range(n_top)]
    total_w = sum(weights)
    times_p1 = [p1_time * w / total_w for w in weights]

    for i in range(n_top):
        if time.monotonic() > phase1_end:
            break
        sc_init, name, a0 = scored[i]
        t_alloc = min(times_p1[i], max(30.0, phase1_end - time.monotonic()))
        print(f"  L-BFGS-B on '{name}' (n={len(a0)}, init={sc_init:.6f}, t={t_alloc:.0f}s)")

        # Try with beta=30
        a_opt, sc_opt = _lbfgs_optimize(a0, 30.0, t_alloc * 0.7, rng)
        _try_update(a_opt, f"lbfgs_b30_{name}")

        if time.monotonic() < phase1_end - 10:
            # Try with higher beta
            a_opt2, sc_opt2 = _lbfgs_optimize(a_opt, 80.0, phase1_end - time.monotonic() - 5, rng)
            _try_update(a_opt2, f"lbfgs_b80_{name}")

    # Phase 2: Extended optimization on best result
    print("\n=== Phase 2: Extended L-BFGS-B on best ===")
    phase2_end = total_start + 1600.0

    if best_overall is not None:
        a_work = best_overall.copy()
        n_work = len(a_work)
        print(f"  Working with n={n_work}, score={best_overall_score:.8f}")

        # Try different betas
        for beta in [20, 50, 100, 200]:
            if time.monotonic() > phase2_end:
                break
            t_alloc = min(150.0, phase2_end - time.monotonic())
            if t_alloc < 20:
                break
            print(f"  L-BFGS-B beta={beta}, t={t_alloc:.0f}s")
            a_opt, sc = _lbfgs_optimize(a_work, float(beta), t_alloc, rng)
            _try_update(a_opt, f"p2_b{beta}")
            a_work = best_overall.copy()  # restart from best

        # Try with different n by resizing
        for n_new in [n_work * 2, n_work // 2]:
            if time.monotonic() > phase2_end:
                break
            t_alloc = min(100.0, phase2_end - time.monotonic())
            if t_alloc < 20 or n_new < 500 or n_new > 20000:
                continue
            # Resize via interpolation
            x_old = np.linspace(0, 1, n_work)
            x_new = np.linspace(0, 1, n_new)
            a_resized = np.interp(x_new, x_old, best_overall)
            a_resized = np.maximum(a_resized, 0)
            a_resized = a_resized / a_resized.sum()
            print(f"  Resize to n={n_new}, L-BFGS-B t={t_alloc:.0f}s")
            a_opt, sc = _lbfgs_optimize(a_resized, 50.0, t_alloc, rng)
            _try_update(a_opt, f"p2_resize_{n_new}")

    # Phase 3: LP-based diversification
    print("\n=== Phase 3: LP-based diversification ===")
    phase3_end = total_start + 2100.0

    if best_overall is not None and time.monotonic() < phase3_end:
        f = best_overall.copy()
        n = len(f)
        b = _fft_convolve(f)
        M = b.max()
        print(f"  LP step: n={n}, M={M:.8f}")

        t_alloc = min(200.0, phase3_end - time.monotonic())
        if t_alloc > 30:
            # Solve LP
            t0 = time.monotonic()
            g = _lp_step(f, M, n)
            t_lp = time.monotonic() - t0
            if g is not None and g.sum() > 0:
                g = g / g.sum()
                # Blend
                for t_blend in [0.1, 0.2, 0.3, 0.5]:
                    a_blend = (1 - t_blend) * f + t_blend * g
                    a_blend = np.maximum(a_blend, 0.0)
                    if a_blend.sum() > 1e-15:
                        a_blend = a_blend / a_blend.sum()
                    _try_update(a_blend, f"lp_blend_{t_blend}")
                # Polish the LP result
                if time.monotonic() < phase3_end - 30:
                    a_pol, sc_pol = _lbfgs_optimize(g, 50.0,
                                                     phase3_end - time.monotonic() - 10, rng)
                    _try_update(a_pol, "lp_polished")
            # Second LP step with tighter M
            if time.monotonic() < phase3_end - 60:
                b2 = _fft_convolve(best_overall)
                M2 = b2.max() * 0.95  # tighter
                g2 = _lp_step(best_overall, M2, n)
                if g2 is not None and g2.sum() > 0:
                    g2 = g2 / g2.sum()
                    for t_blend in [0.2, 0.4]:
                        a_blend = (1 - t_blend) * best_overall + t_blend * g2
                        a_blend = np.maximum(a_blend, 0.0)
                        if a_blend.sum() > 1e-15:
                            a_blend = a_blend / a_blend.sum()
                        _try_update(a_blend, f"lp_tight_{t_blend}")
                if time.monotonic() < phase3_end - 30:
                    a_pol2, sc_pol2 = _lbfgs_optimize(g2, 50.0,
                                                       phase3_end - time.monotonic() - 10, rng)
                    _try_update(a_pol2, "lp_tight_polished")

    # Phase 4: Final polishing
    print("\n=== Phase 4: Final polishing ===")
    if best_overall is not None:
        remaining = max(0, deadline - time.monotonic())
        print(f"  {remaining:.0f}s remaining")
        if remaining > 30:
            # Subgradient polish
            t_pol = remaining * 0.4
            a_pol, sc_pol = _subgrad_polish(best_overall, t_pol, rng)
            _try_update(a_pol, "p4_subgrad")

            # Final L-BFGS with high beta
            remaining2 = max(0, deadline - time.monotonic())
            if remaining2 > 30:
                a_fin, sc_fin = _lbfgs_optimize(best_overall, 150.0, remaining2 * 0.5, rng)
                _try_update(a_fin, "p4_final_lbfgs")

            # Local search: try small perturbations
            remaining3 = max(0, deadline - time.monotonic())
            if remaining3 > 5:
                a_ls = best_overall.copy()
                n = len(a_ls)
                for _ in range(50):
                    if time.monotonic() > deadline:
                        break
                    # Random coordinate perturbation
                    idx = rng.integers(0, n, size=max(1, n // 50))
                    factors = np.exp(rng.normal(0, 0.01, len(idx)))
                    a_try = a_ls.copy()
                    a_try[idx] *= factors
                    a_try = np.maximum(a_try, 0.0)
                    if a_try.sum() > 1e-15:
                        a_try = a_try / a_try.sum()
                    sc_try = _score(a_try)
                    if sc_try < best_overall_score:
                        best_overall_score = sc_try
                        best_overall = a_try.copy()
                        a_ls = a_try.copy()

    # Final verification
    if best_overall is not None:
        final_score = evaluate_sequence(best_overall.tolist())
        elapsed = time.monotonic() - total_start
        print(f"\n=== FINAL: score={final_score:.8f}, n={len(best_overall)}, time={elapsed:.1f}s ===")
        return [float(x) for x in best_overall]
    else:
        # Fallback
        a = np.ones(2000, dtype=np.float64)
        x = np.arange(2000, dtype=np.float64)
        a *= (1.0 + 0.4 * np.sin(2.0 * np.pi * x / 2000.0 + np.pi / 4.0))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
        print(f"Fallback uniform-sinusoidal, score={_score(a):.6f}")
        return [float(x) for x in a]
