One more thought: the key insight I keep coming back to is that `max(a*a)` for non-negative `a` is NOT necessarily at the center. For a one-sided decaying sequence, it might be at the left edge (k=0 gives a[0]^2) or at some other position. This means the "shape" of the optimal sequence depends on where the max is.

For a sequence that's large on the left and decays to the right:
- (a*a)[0] = a[0]^2 (large)
- (a*a)[n-1] = sum a[i]*a[n-1-i] (moderate, since a decays)
- (a*a)[2n-2] = a[n-1]^2 (small)

So for a one-sided sequence, the max might be at k=0, which is just a[0]^2. In that case, the objective is `2n * a[0]^2 / (sum a)^2`.

To minimize this, we want a[0] to be as small as possible relative to sum(a). But we also need to keep the center term small.

This is a trade-off! The optimal sequence balances:
- Edge terms: a[0]^2, a[n-1]^2 (and nearby)
- Center term: sum a[i]*a[n-1-i]
- Middle terms: (a*a)[k] for k between 0 and n-1

For a perfectly flat convolution, all of these would be equal to (sum a)^2 / (2n-1).

Let me also add a "dual" perspective: think of the problem as:
- Minimize max_k (a*a)[k] subject to sum(a) = 1, a >= 0
- The "dual" variables are the weights w[k] >= 0, sum(w[k]) = 1
- The KKT conditions say: at optimum, w[k] > 0 only for k where (a*a)[k] = max, and the gradient condition holds.

This suggests an "inner-outer" algorithm:
- Inner: given w, minimize sum_k w[k] * (a*a)[k] subject to sum(a)=1, a>=0 (this is a quadratic program)
- Outer: update w to put more weight on the max

The inner problem is: minimize a^T Q_w a subject to sum(a)=1, a>=0, where Q_w = sum_k w[k] * Q_k (each Q_k is the Hessian of (a*a)[k]/2).

This is a QP, solvable in O(n^2) or O(n^3) per iteration. For n=3809, that's feasible but might be slow.

"""
Search for a non-negative step-function height sequence minimizing
    2 * n * max(convolve(a, a)) / (sum(a))**2.

Strategy:
  * Use the previously best sequence as a starting point.
  * Primary optimization: multiplicative weight (exponential tilting) updates
    with a soft-max subgradient, which naturally preserves the simplex.
  * Periodic L-BFGS-B polishing with smoothed-max objective in log-ratio space.
  * Final local search with block perturbations.
  * Simulated annealing: occasionally accept worse solutions to escape local minima.
"""

import numpy as np
import time
import math

try:
    from scipy.optimize import minimize as _minimize
except Exception:
    _minimize = None


def _pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _clean_nonnegative(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    return np.maximum(v, 0.0)


def _project_simplex(v):
    v = _clean_nonnegative(v)
    n = v.size
    if n == 0:
        return v.copy()
    x = np.sort(v)[::-1]
    css = np.cumsum(x)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = x > (css - 1.0) / idx
    if not np.any(cond):
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    r = int(np.nonzero(cond)[0][-1])
    tau = (css[r] - 1.0) / float(r + 1)
    out = v - tau
    np.maximum(out, 0.0, out=out)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    out *= 1.0 / s
    return out


def _score_vector(a, L=None):
    v = _clean_nonnegative(a)
    n = v.size
    if n == 0:
        return np.inf
    s = float(v.sum())
    if s < 1e-12:
        return np.inf
    v = v / s
    if L is None:
        L = _pow2(2 * n - 1)
    A = np.fft.rfft(v, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    mx = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(mx):
        return np.inf
    if mx < 0.0:
        mx = 0.0
    return 2.0 * float(n) * mx


def _convolve_squared(a, L):
    A = np.fft.rfft(a, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)
    return np.maximum(c, 0.0)


def _soft_gradient(a, w, L):
    """
    Compute gradient of sum_k w[k] * (a*a)[k] w.r.t. a.
    The result is a vector of length n.
    """
    W = np.fft.rfft(w, n=L)
    A = np.fft.rfft(a, n=L)
    g = 2.0 * np.fft.irfft(W * np.conj(A), n=L)[:a.size]
    return g


def _exp_tilt_iterate(a, end_time, tau_start, tau_end, eta_start, eta_end, 
                       seed, momentum=0.9, use_sa=True, T_start=0.01, T_end=0.001):
    """
    Multiplicative weight iterations with soft-max subgradient.
    Returns the best a found.
    """
    n = a.size
    L = _pow2(2 * n - 1)
    Lconv = _pow2(3 * n - 2)  # larger for gradient computation safety
    
    best = a.copy()
    best_score = _score_vector(a, L)
    
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    total = max(1e-6, end_time - start)
    
    # Momentum buffers (in log-space)
    m_buf = np.zeros(n, dtype=np.float64)
    v_buf = np.zeros(n, dtype=np.float64)
    t = 0
    beta1 = 0.9
    beta2 = 0.99
    
    last_improve = start
    
    while time.monotonic() < end_time:
        elapsed = (time.monotonic() - start) / total
        frac = min(1.0, elapsed)
        
        # Learning rate schedule: start high, decay
        eta = eta_start * (eta_end / eta_start) ** frac
        tau = tau_start * (tau_end / tau_start) ** frac
        T = T_start * (T_end / T_start) ** frac if use_sa else 0.0
        
        # Compute convolution
        c = _convolve_squared(a, Lconv)
        c = c[:2*n-1]
        
        current_score = 2.0 * n * float(c.max())
        
        # Soft weights
        cmax = float(c.max())
        if cmax <= 1e-300:
            break
        z = (c - cmax) / tau
        z = np.clip(z, -50.0, 0.0)
        w = np.exp(z)
        w_sum = float(w.sum())
        if w_sum <= 0 or not np.isfinite(w_sum):
            w = np.zeros_like(c)
            w[c.argmax()] = 1.0
            w_sum = 1.0
        w = w / w_sum
        
        # Compute gradient
        g = _soft_gradient(a, w, Lconv)
        
        # Normalize gradient for stable updates
        g_mean = float(g.mean())
        g = g - g_mean  # center to preserve the sum more naturally
        g_rms = float(np.sqrt(np.mean(g*g))) + 1e-12
        g = g / g_rms
        
        # Adam-like update in multiplicative (log) space
        t += 1
        m_buf = beta1 * m_buf + (1 - beta1) * g
        v_buf = beta2 * v_buf + (1 - beta2) * (g * g)
        m_hat = m_buf / (1 - beta1 ** t)
        v_hat = v_buf / (1 - beta2 ** t)
        
        step = eta * m_hat / (np.sqrt(v_hat) + 1e-8)
        
        # Multiplicative update
        a_new = a * np.exp(-step)
        
        # Clamp and renormalize
        a_new = np.maximum(a_new, 0.0)
        s = float(a_new.sum())
        if s > 1e-12 and np.isfinite(s):
            a_new = a_new / s
        else:
            a_new = a.copy()
        
        new_score = _score_vector(a_new, L)
        
        # Simulated annealing acceptance
        if new_score < best_score - 1e-12:
            best = a_new.copy()
            best_score = new_score
            last_improve = time.monotonic()
        elif use_sa and T > 0:
            delta = new_score - best_score
            if delta < 0.5:  # only accept if not too bad
                accept_prob = np.exp(-delta / T)
                if rng.random() < accept_prob:
                    best_score = new_score  # temporarily
                    best = a_new.copy()  # temporarily (for SA purposes, but track true best separately)
        
        # Always move to a_new (even if worse, for SA purposes)
        # But keep a separate "elite" best
        a = a_new
        
        # Reset momentum if stuck too long
        if time.monotonic() - last_improve > 15.0:
            m_buf.fill(0.0)
            v_buf.fill(0.0)
            t = 0
            # Occasionally perturb
            if rng.random() < 0.3:
                a = _project_simplex(best * np.exp(rng.normal(0, 0.05, size=n)))
        
    return best, best_score


def _ratios_from_a(a):
    a = _clean_nonnegative(np.asarray(a, dtype=np.float64)).copy()
    n = a.size
    if n == 0:
        return a
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        a = np.full(n, 1.0 / float(n), dtype=np.float64)
        s = 1.0
    else:
        a = a / s
    if n == 1:
        return a
    ref = float(a[-1])
    if ref <= 1e-200:
        a[-1] = 1e-12
        a = a / float(a.sum())
        ref = float(a[-1])
    x = np.log(np.maximum(a[:-1], 1e-300) / max(ref, 1e-300))
    return np.clip(x, -100.0, 100.0)


def _softmax_from_ratios(x):
    x = np.asarray(x, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    n = x.size + 1
    if n == 1:
        return np.ones(1, dtype=np.float64)
    m = 0.0
    if x.size:
        mx = float(np.max(x))
        if mx > m:
            m = mx
    e = np.exp(np.clip(x - m, -745.0, 709.0))
    e_last = float(np.exp(-m))
    s = float(e.sum()) + e_last
    if not np.isfinite(s) or s <= 0.0:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    a = np.empty(n, dtype=np.float64)
    a[:-1] = e / s
    a[-1] = e_last / s
    return a


def _soft_objective_grad(x, eps):
    a = _softmax_from_ratios(x)
    n = a.size
    if n <= 1:
        return 0.0, np.zeros_like(x, dtype=np.float64)
    L = _pow2(3 * n - 2)
    A = np.fft.rfft(a, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    cmax = float(c.max())
    if cmax <= 1e-300:
        return 0.0, np.zeros_like(x, dtype=np.float64)
    z = (c - cmax) / eps
    z = np.clip(z, -50.0, 0.0)
    e = np.exp(z)
    sume = float(e.sum())
    if not np.isfinite(sume) or sume <= 0.0:
        return 1e300, np.zeros_like(x, dtype=np.float64)
    f = cmax + eps * float(np.log(sume))
    w = e / sume
    W = np.fft.rfft(w, n=L)
    ga = 2.0 * np.fft.irfft(W * np.conj(A), n=L)[:n]
    dot = float(np.dot(a, ga))
    gfull = a * (ga - dot)
    gy = gfull[:-1]
    if not np.all(np.isfinite(gy)):
        gy = np.zeros_like(x, dtype=np.float64)
    return f, gy


def _run_minimize_once(fun, x, args, maxiter, bounds=None):
    if _minimize is None:
        return np.asarray(x, dtype=np.float64), 99
    x = np.asarray(x, dtype=np.float64)
    if x.size == 0:
        return x, 1
    if bounds is None:
        bounds = [(-100.0, 100.0)] * int(x.size)
    options = {
        "maxiter": int(maxiter),
        "disp": 0,
        "maxfun": int(max(200, 12 * maxiter)),
        "ftol": 1e-18,
        "gtol": 1e-12,
    }
    try:
        res = _minimize(fun, x, args=args, method="L-BFGS-B", jac=True, bounds=bounds, options=options)
    except Exception:
        return x, 99
    x_new = np.asarray(res.x, dtype=np.float64)
    x_new = np.where(np.isfinite(x_new), x_new, 0.0)
    x_new = np.clip(x_new, -100.0, 100.0)
    status = 0
    try:
        status = int(res.status)
    except Exception:
        status = 0
    return x_new, status

def _local_search(a0, end_time, seed):
    """Block perturbation local search."""
    a = _project_simplex(a0)
    n = a.size
    L = _pow2(2 * n - 1)
    best = a.copy()
    best_score = _score_vector(best, L)
    rng = np.random.default_rng(seed)
    sigma = 0.10
    last_best = time.monotonic()
    
    while time.monotonic() < end_time:
        cand = best.copy()
        nblocks = int(rng.integers(1, 4))
        for _ in range(nblocks):
            maxw = max(3, n // 4)
            w = max(1, min(int(np.exp(rng.uniform(np.log(2.0), np.log(float(maxw))))), n))
            s = int(rng.integers(0, n - w + 1))
            cand[s:s+w] *= float(np.exp(rng.normal(0.0, sigma)))
        cand = _project_simplex(cand)
        sc = _score_vector(cand, L)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best = cand
            best_score = sc
            last_best = time.monotonic()
            sigma = min(0.3, sigma * 1.05)
        elif time.monotonic() - last_best > 5.0:
            sigma = max(0.01, sigma * 0.8)
            if time.monotonic() - last_best > 15.0:
                cand = _project_simplex(best * np.exp(rng.normal(0, 0.03, size=n)))
                sc2 = _score_vector(cand, L)
                if np.isfinite(sc2) and sc2 < best_score:
                    best = cand
                    best_score = sc2
                    last_best = time.monotonic()
    
    return best, best_score

def _pow2(m):
    m = int(m)

def _clean_nonnegative(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)

def _project_simplex(v):
    v = _clean_nonnegative(v)
    n = v.size
    if n == 0:
        return v.copy()
    x = np.sort(v)[::-1]
    css = np.cumsum(x)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = x > (css - 1.0) / idx
    if not np.any(cond):
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    r = int(np.nonzero(cond)[0][-1])
    tau = (css[r] - 1.0) / float(r + 1)
    out = v - tau
    np.maximum(out, 0.0, out=out)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    out *= 1.0 / s
    return out

            break
        a = _softmax_from_ratios(x)
        sc = _score_vector(a, L)
        if np.isfinite(sc) and sc < best_score:
            best = a.copy()
            best_score = sc

    # Stage 2: L3
    s2_end = start + 0.50 * total
    if time.monotonic() < s2_end:
        while time.monotonic() < s2_end:
            x, st = _run_lbfgs(_q_obj_grad, x, (3.0,), 50, bounds)
            if st == 1:
                break
            a = _softmax_from_ratios(x)
            sc = _score_vector(a, L)
            if np.isfinite(sc) and sc < best_score:
                best = a.copy()
                best_score = sc

    # Stage 3: Smoothed max with decreasing eps
    if time.monotonic() < end_time:
        eps = 1e-4
        no_improve = 0
        while time.monotonic() < end_time:
            x, st = _run_lbfgs(_soft_obj_grad, x, (eps,), 50, bounds)
            if st == 1:
                break
            a = _softmax_from_ratios(x)
            sc = _score_vector(a, L)
            if np.isfinite(sc) and sc < best_score - 1e-14:
                best = a.copy()
                best_score = sc
                no_improve = 0
                eps = max(1e-7, eps * 0.6)
            else:
                no_improve += 1
                if no_improve >= 3:
                    break
                eps = max(1e-7, eps * 0.5)

    return best, _score_vector(best, L)


def _local_search(a0, end_time, seed):
    a = _project_simplex(a0)
    n = a.size
    L = _pow2(2 * n - 1)
    best = a.copy()
    best_score = _score_vector(a, L)
    rng = np.random.default_rng(seed)
    sigma = 0.08
    last_best = time.monotonic()

    while time.monotonic() < end_time:
        cand = best.copy()
        nblocks = int(rng.integers(1, 4))
        for _ in range(nblocks):
            maxw = max(3, n // 4)
            w = max(1, min(int(np.exp(rng.uniform(np.log(2.0), np.log(float(maxw))))), n))
            s = int(rng.integers(0, n - w + 1))
            cand[s:s + w] *= float(np.exp(rng.normal(0.0, sigma)))
        cand = _project_simplex(cand)
        sc = _score_vector(cand, L)
        if np.isfinite(sc) and sc < best_score - 1e-14:
            best = cand
            best_score = sc
            last_best = time.monotonic()
            sigma = min(0.25, sigma * 1.03)
        elif time.monotonic() - last_best > 8.0:
            sigma = max(0.01, sigma * 0.85)
            if time.monotonic() - last_best > 20.0:
                cand = _project_simplex(best * np.exp(rng.normal(0, 0.04, size=n)))
                sc2 = _score_vector(cand, L)
                if np.isfinite(sc2) and sc2 < best_score:
                    best = cand
                    best_score = sc2
                    last_best = time.monotonic()

    return best, _score_vector(best, L)


def _power_sequence(n, p=0.5, offset=1.0):
    j = np.arange(n, dtype=np.float64)
    x = np.maximum(j + float(offset), 1e-8)
    return _project_simplex(np.power(x, -float(p)))


def _sym_power_sequence(n, p=0.5, offset=1.0):
    j = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(j - mid) + float(offset)
    return _project_simplex(np.power(d, -float(p)))


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search: multiplicative weights + L-BFGS + local search.
    """
    t0 = time.monotonic()
    budget = max(0.0, float(budget_s) - 15.0)
    deadline = t0 + budget

    n = 3809
    try:
        h = height_sequence_1
        if isinstance(h, (list, np.ndarray)):
            arr = np.asarray(h, dtype=np.float64)
            if arr.ndim == 1 and 100 <= arr.size <= 20000:
                n = int(arr.size)
    except Exception:
        pass

    if n < 32:
        n = 3809

    print(f"n={n}, budget={budget:.0f}s")
    L = _pow2(2 * n - 1)

    # Generate candidates
    cands = []

    # Previous best
    try:
        h = height_sequence_1
        if isinstance(h, (list, np.ndarray)):
            arr = np.asarray(h, dtype=np.float64)
            arr = _clean_nonnegative(arr)
            if arr.size == n and arr.sum() > 0:
                cands.append(_project_simplex(arr))
    except Exception:
        pass

    # Power sequences (one-sided)
    for p in (0.4, 0.45, 0.5, 0.55, 0.6, 0.7, 0.8):
        for off in (0.5, 1.0, 2.0, 3.0):
            cands.append(_power_sequence(n, p, off))

    # Symmetric power sequences
    for p in (0.3, 0.4, 0.5, 0.6, 0.8):
        cands.append(_sym_power_sequence(n, p, 1.0))

    # Random smooth
    rng0 = np.random.default_rng(99999)
    for sd in (0.3, 0.5, 0.7, 1.0):
        v = np.exp(rng0.normal(0, sd, size=n))
        v *= np.power(np.arange(n) + 1, -0.3)
        cands.append(_project_simplex(v))

    # Score
    scored = []
    for i, a in enumerate(cands):
        sc = _score_vector(a, L)
        if np.isfinite(sc):
            scored.append((sc, i, a))
    scored.sort()

    if not scored:
        a = _power_sequence(n, 0.5, 1.0)
        print(f"Fallback score: {_score_vector(a, L):.6f}")
        return [float(x) for x in a]

    print(f"Best initial: {scored[0][0]:.6f} (from candidate {scored[0][1]})")

    global_best_a = np.asarray(scored[0][2], dtype=np.float64).copy()
    global_best_score = float(scored[0][0])

    # Phase 1: Multiplicative weights (55% of time)
    p1_end = t0 + 0.55 * budget
    top_k = min(3, len(scored))
    frac_split = [0.55, 0.25, 0.20] if top_k == 3 else [0.65, 0.35] if top_k == 2 else [1.0]

    for idx in range(top_k):
        ph_start = t0 + budget * (1.0 - sum(frac_split[:idx + 1]))
        ph_end = t0 + budget * (1.0 - sum(frac_split[:idx]))
        ph_end = min(ph_end, p1_end)
        if ph_end <= time.monotonic() + 3.0:
            continue

        a_start = np.asarray(scored[idx][2], dtype=np.float64).copy()
        print(f"  MW start {idx}: score={scored[idx][0]:.6f}, time={ph_end - ph_start:.0f}s")

        a_b, s_b = _mult_weight_phase(a_start, ph_end, int(seed) + 200 * (idx + 1))
        if np.isfinite(s_b) and s_b < global_best_score:
            global_best_score = s_b
            global_best_a = a_b.copy()
            print(f"    Improved to {s_b:.6f}")

    print(f"After Phase 1: {global_best_score:.6f}")

    # Phase 2: L-BFGS (30% of time)
    p2_end = t0 + 0.85 * budget
    if time.monotonic() < p2_end:
        print(f"Phase 2: L-BFGS polish")
        a_b, s_b = _lbfgs_phase(global_best_a, p2_end, int(seed) + 5555)
        if np.isfinite(s_b) and s_b < global_best_score:
            global_best_score = s_b
            global_best_a = a_b.copy()
            print(f"  After L-BFGS: {s_b:.6f}")

    # Phase 3: Local search (15% of time)
    if time.monotonic() < deadline:
        print(f"Phase 3: Local search")
        a_b, s_b = _local_search(global_best_a, deadline, int(seed) + 8888)
        if np.isfinite(s_b) and s_b < global_best_score:
            global_best_score = s_b
            global_best_a = a_b.copy()
            print(f"  After local: {s_b:.6f}")

    # Final validation
    global_best_a = _project_simplex(global_best_a)
    out = [float(x) for x in global_best_a]

    try:
        val = evaluate_sequence(out)
        print(f"FINAL evaluate_sequence: {val:.6f}")
        if not np.isfinite(val):
            out = [float(x) for x in _power_sequence(len(out), 0.5, 1.0)]
    except Exception:
        pass

    elapsed = time.monotonic() - t0
    print(f"Done. Time={elapsed:.1f}s, internal best={global_best_score:.6f}")

    return out
