    best_a = a.copy()

Minimize J(a) = 2n*max(conv(a,a))/sum(a)^2 using:
1. Alternating LP at increasing n (500, 1024, 2048, 4096)
2. Multiplicative weight descent (novel approach)
3. Peak-shaving heuristic
4. Final LP polish

import numpy as np
import time
from scipy.optimize import linprog

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]

def _score(a):
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)

def _build_conv_matrix(f):
    n = len(f)
    # C is (2n-1) x n, C[k,j] = f[k-j] if valid
    C = np.zeros((2*n-1, n), dtype=np.float64)
    for j in range(n):
        start = j
        end = j + n
        C[start:end, j] = f
    return C

def _lp_solve(f, M):
    n = len(f)
    C = _build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2*n-1)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if result.status == 0:
            return np.maximum(result.x, 0)
    except Exception as e:
        pass
    return None

    best_score = 2.0 * n * float(np.max(_fft_convolve(a)))  # sum=1 normalization
    blend = initial_blend
    stale_count = 0

def _peak_shaving(a_init, max_iters, deadline, n_shave=5):
    """Directly reduce elements contributing to peak lags."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    s = float(np.sum(a))
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * float(np.max(_fft_convolve(a)))
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        max_c = float(np.max(c))
        score = 2.0 * n * max_c
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Find top n_shave lags
        top_lags = np.argpartition(c, -n_shave)[-n_shave:]
        
        # For each peak lag, find the elements with highest "cross product"
        reduction = np.zeros(n)
        for k in top_lags:
            k = int(k)
            if k >= n:
                k = 2*n-1-k  # use symmetry
            if k == 0:
                reduction += a * a
            else:
                m = min(n-k, n)
                prod = np.zeros(n)
                prod[:n-k] += a[:n-k] * a[k:]
                prod[k:] += a[k:] * a[:n-k]
                reduction += prod * 2
        
        # Reduce top elements
        max_r = float(np.max(reduction))
        if max_r < 1e-15:
            break
        # Scale reduction
        lr = 0.1
        a = a - lr * reduction / max_r
        a = np.maximum(a, 0)
        s_new = float(np.sum(a))
        if s_new < 1e-10:
            break
        a = a / s_new
    
    return best_a, best_score

def _interpolate_sequence(src, n_target):
    n_src = len(src)
    if n_src == n_target:
        return src.copy()
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    out = np.interp(x_tgt, x_src, src)
    out = np.maximum(out, 0.0)
    s = float(np.sum(out))
    if s > 1e-10:
        out = out / s
    return out

Minimize J(a)=2n*max(conv(a,a))/sum(a)^2 using alternating LP at escalating n,

import numpy as np

from scipy.optimize import linprog


def _fft_convolve(a):
    """Full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return np.maximum(c[:size], 0.0)


def _score(a):
    """J(a) = 2n*max(conv)/sum^2."""
    n = len(a)
    c = _fft_convolve(a)
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * float(np.max(c)) / (s * s)


def _score_norm1(a):
    """Score assuming sum(a)=1: 2n*max(conv)."""
    n = len(a)
    c = _fft_convolve(a)
    return 2.0 * n * float(np.max(c))


def _build_conv_matrix(f):
    """C such that (C@g)[k] = (f*g)[k]. Shape (2n-1, n)."""
    n = len(f)
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = f
    return C


def _lp_solve(f, M, timeout=None):
    """Solve max sum(g) s.t. f*g <= M, g >= 0. Returns g or None."""
    n = len(f)
    C = _build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    try:
        opts = {'presolve': True, 'dual_feasibility_tolerance': 1e-10,
                'primal_feasibility_tolerance': 1e-10}
        if timeout is not None:
            opts['time_limit'] = timeout
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                         method='highs', options=opts)
        if result.status == 0 and result.x is not None:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None

                a_new = np.maximum(a_new, 0)
                s_new = float(np.sum(a_new))
                a_new = a_new / s_new
                stale = 0
        a = a_new
        blend = max(0.05, blend * 0.98)

    return best_a, best_score


def _spectral_flat(a_init, max_iters, deadline, alpha_init=0.15):
    """Novel: flatten |FFT(a)| in frequency domain to flatten autocorrelation."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    s = float(np.sum(a))
    a = a / s
    best_a = a.copy()
    best_score = _score_norm1(a)
    alpha = alpha_init
    rng = np.random.default_rng(7)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        F = np.fft.rfft(a)
        mag = np.abs(F).copy()

        # Target: flat magnitude (constant across frequency bins)
        target = s / np.sqrt(2.0 * n - 1.0)
        # Don't touch DC
        mag[1:] = (1.0 - alpha) * mag[1:] + alpha * target

        phase = np.angle(F)
        F_new = mag * np.exp(1j * phase)
        a_new = np.fft.irfft(F_new, 2 * n - 1)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        # Occasionally add small random perturbation
        if it % 50 == 0 and it > 0:
            a_new += 0.005 * rng.uniform(-1, 1, n)
            a_new = np.maximum(a_new, 0)
            a_new = a_new / np.sum(a_new)

        sc = _score_norm1(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            if it % 200 == 0:
                print(f"      Spectral iter {it}: score={sc:.6f}")
        a = a_new
        alpha = max(0.02, alpha * 0.999)

    return best_a, best_score


def _multiplicative_descent(a_init, max_iters, eta_init, n_top, deadline):
    """Multiplicative weight update to reduce autocorrelation peaks."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    s = float(np.sum(a))
    a = a / s
    best_a = a.copy()
    best_score = _score_norm1(a)
    eta = eta_init

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        max_c = float(np.max(c))
        sc = 2.0 * n * max_c

        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            if it % 500 == 0:
                print(f"      MW iter {it}: score={sc:.6f}")

        if n_top >= len(c):
            top_lags = np.arange(len(c))
        else:
            top_lags = np.argpartition(c, -n_top)[-n_top:]

        pressure = np.zeros(n)
        for k in top_lags:
            k = int(k)
            if k == 0:
                pressure += a
            elif k < n:
                m = n - k
                pressure[:m] += a[k:]
                pressure[k:] += a[:m]
            else:
                k2 = 2 * n - 1 - k
                m = n - k2
                pressure[:m] += a[k2:]
                pressure[k2:] += a[:m]

        max_p = float(np.max(pressure))
        if max_p > 1e-15:
            a = a * np.exp(-eta * pressure / max_p)
            a = np.maximum(a, 0.0)
            s_new = float(np.sum(a))
            if s_new < 1e-10:
                break
            a = a / s_new
        eta *= 0.9995

    return best_a, best_score


def _peak_shaving(a_init, max_iters, deadline, n_shave=8, lr=0.08):
    """Directly reduce elements at peak-lag crossover positions."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    a = a / float(np.sum(a))
    best_a = a.copy()
    best_score = _score_norm1(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        sc = _score_norm1(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if n_shave >= len(c):
            top_lags = np.arange(len(c))
        else:
            top_lags = np.argpartition(c, -n_shave)[-n_shave:]

        reduction = np.zeros(n)
        for k in top_lags:
            k = int(k)
            if k >= n:
                k = 2 * n - 1 - k
            if k == 0:
                reduction += a * a
            else:
                m = n - k
                reduction[:m] += a[:m] * a[k:]
                reduction[k:] += a[k:] * a[:m]

        max_r = float(np.max(reduction))
        if max_r < 1e-15:
            break
        a = a - lr * reduction / max_r
        a = np.maximum(a, 0.0)
        s_new = float(np.sum(a))
        if s_new < 1e-10:
            break
        a = a / s_new

    return best_a, best_score


def _interpolate_sequence(src, n_target):
    """Linearly interpolate a sequence to a new length, normalize to sum=1."""
    n_src = len(src)
    if n_src == n_target:
        out = src.copy()
    else:
        x_src = np.linspace(0, 1, n_src)
        x_tgt = np.linspace(0, 1, n_target)
        out = np.interp(x_tgt, x_src, src)
    out = np.maximum(out, 0.0)
    s = float(np.sum(out))
    if s > 1e-10:
        out = out / s
    else:
        out = np.ones(n_target) / n_target
    return out


def _make_diverse_init(n, variant, rng):
    """Create diverse initial sequences."""
    t = np.linspace(0, 1, n)
    if variant == 0:
        a = 1.0 - t
    elif variant == 1:
        a = np.exp(-3.0 * t)
    elif variant == 2:
        a = (1.0 - t) * (1.0 + 0.5 * np.cos(2.0 * np.pi * 3.0 * t))
    elif variant == 3:
        a = (1.0 - t) * (1.0 + 0.7 * np.sin(2.0 * np.pi * 5.0 * t))
    elif variant == 4:
        a = np.abs(rng.normal(0, 1, n))
    elif variant == 5:
        a = np.exp(-((t - 0.3) ** 2) / 0.02) + 0.5 * np.exp(-((t - 0.7) ** 2) / 0.05)
    elif variant == 6:
        a = t * (1.0 - t)  # parabolic
    elif variant == 7:
        a = np.exp(-((t - 0.5) ** 2) / 0.05)  # center bump
    elif variant == 8:
        a = (1.0 - t) ** 0.5 * (1.0 + 0.4 * np.cos(2.0 * np.pi * 4.0 * t))
    elif variant == 9:
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * 2.0 * t)  # cosine window
    elif variant == 10:
        a = np.exp(-2.0 * t) * (1.0 + 0.6 * np.sin(2.0 * np.pi * 6.0 * t))
    elif variant == 11:
        a = (1.0 - t) * (1.0 - t) + 0.3 * np.exp(-(t - 0.4) ** 2 / 0.01)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a
