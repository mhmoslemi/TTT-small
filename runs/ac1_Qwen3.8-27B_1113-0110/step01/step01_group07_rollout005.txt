    return float(true_score), float(2.0 * n * smooth_val), log_grad


def cheb_basis(n, D):
    x = np.linspace(0.0, 1.0, n)
    t = 2.0 * x - 1.0
    t = np.clip(t, -1.0, 1.0)
    ang = np.arccos(t)
    out = np.empty((n, D), dtype=np.float64)
    for m in range(1, D + 1):
        out[:, m - 1] = np.cos(m * ang)
    return out

def softmax_arr(z):
    z = np.asarray(z, dtype=np.float64)
    z = np.where(np.isfinite(z), z, -50.0)
    z = z - z.max()
    z = np.clip(z, -50.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones_like(z) / z.size
    return e / s

def adam_optimize(parent, duration_s, rng, tau_init=0.5, tau_min=0.005, lr0=0.03):
    """
    Adam optimizer in log-softmax parameterization.
    parent: normalized non-negative array
    Returns: (best_p, best_score)
    """
    p = normalize_positive(parent)
    if p is None:
        return None, np.inf
    
    n = p.size
    L = next_pow2(2 * n - 1)
    L2 = next_pow2(3 * n - 2)
    deadline = time.monotonic() + max(0.0, float(duration_s))
    
    z = np.log(np.maximum(p, 1e-300))
    z = np.where(np.isfinite(z), z, -50.0)
    z -= z.mean()
    z = np.clip(z, -50.0, 50.0)
    
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    bias_step = 0
    
    tau = float(tau_init)
    lr = float(lr0)
    best_p = p.copy()
    best_score = fast_score(p)
    no_improve = 0
    step = 0
    
    while time.monotonic() < deadline:
        true_s, smooth_s, grad = smooth_max_and_grad(p, tau, L, L2)
        if not np.isfinite(true_s):
            break
        
        if true_s < best_score - 1e-12:
            best_score = true_s
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Schedule
        if step > 0 and step % 3000 == 0:
            tau = max(tau_min, tau * 0.85)
            lr = max(0.001, lr * 0.92)
        
        # Restart
        if no_improve > 5000:
            zz = np.log(np.maximum(best_p, 1e-300))
            zz = np.where(np.isfinite(zz), zz, -50.0)
            zz += rng.normal(0.0, 0.1, n)
            zz -= zz.mean()
            z = np.clip(zz, -50.0, 50.0)
            m *= 0.05
            v *= 0.05
            no_improve = 0
            tau = max(tau_min, tau * 0.6)
            lr = max(0.002, lr * 0.5)
        
        bias_step += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        b1 = 1.0 - beta1 ** bias_step
        b2 = 1.0 - beta2 ** bias_step
        z = z - lr * (m / b1) / (np.sqrt(v / b2) + eps)
        z = np.where(np.isfinite(z), z, 0.0)
        z -= z.mean()
        z = np.clip(z, -50.0, 50.0)
        p = softmax_arr(z)
        step += 1
        
        if step % 10000 == 0:
            print(f"  adam step {step} true {true_s:.6f} best {best_score:.6f} tau {tau:.4f} lr {lr:.4f}")
    
    return best_p, best_score


def resize_sequence(seq, m):
    p = normalize_positive(seq)
    m = int(m)
    if p is None or m <= 0:
        return np.ones(m, dtype=np.float64) / float(m)
    n = p.size
    if n == m:
        return p.copy()
    if n == 1:
        return np.full(m, 1.0 / float(m), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, m)
    out = np.interp(x_new, x_old, p)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(m, dtype=np.float64) / float(m)
    return out / s


def make_parents(n, D, rng):
    """Generate diverse non-palindromic parent sequences."""
    B = cheb_basis(n, D)
    x = np.linspace(0.0, 1.0, n)
    parents = []
    
    # Uniform
    parents.append(np.ones(n) / n)
    
    # Linear tilt (encourages non-palindromicity)
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5]:
        h = 1.0 + alpha * (x - 0.5)
        p = normalize_positive(h)
        if p is not None:
            parents.append(p)
    
    # Quadratic shape
    for alpha in [0.5, 1.0, 2.0]:
        h = 1.0 + alpha * (x - 0.5) ** 2
        p = normalize_positive(h)
        if p is not None:
            parents.append(p)
    
    # Chebyshev-based
    for _ in range(20):
        theta = rng.normal(0.0, 0.5, D)
        theta[0] = rng.uniform(-1.0, 1.0)
        for i in range(1, min(D, 5)):
            theta[i] *= (0.6 ** i)
        val = B.dot(theta)
        h = 1.0 + 0.15 + np.clip(val, -3, 3)
        p = normalize_positive(h)
        if p is not None:
            parents.append(p)
        # Exponential mode
        h2 = np.exp(np.clip(val * 0.5, -5, 5))
        p2 = normalize_positive(h2)
        if p2 is not None:
            parents.append(p2)
    
    # "Ramp" shapes
    for slope in [0.5, 1.0, 2.0, 3.0]:
        h = np.linspace(0.1, 0.1 + slope, n)
        p = normalize_positive(h)
        if p is not None:
            parents.append(p)
        # Inverted ramp
        h = np.linspace(0.1 + slope, 0.1, n)
        p = normalize_positive(h)
        if p is not None:
            parents.append(p)
    
    # "Bump off-center" shapes
    for center in [0.3, 0.4, 0.6, 0.7]:
        width = 0.15
        h = np.exp(-((x - center) / width) ** 2) + 0.05
        p = normalize_positive(h)
        if p is not None:
            parents.append(p)
    
    # "Two bump" shapes (non-symmetric)
    for c1, c2, w in [(0.2, 0.7, 0.1), (0.3, 0.8, 0.15), (0.25, 0.75, 0.12)]:
        h = np.exp(-((x - c1) / w) ** 2) + 0.6 * np.exp(-((x - c2) / w) ** 2) + 0.05
        p = normalize_positive(h)
        if p is not None:
            parents.append(p)
    
    return parents
