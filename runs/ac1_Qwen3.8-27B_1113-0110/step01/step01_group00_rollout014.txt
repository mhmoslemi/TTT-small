def _mirror_line_search(a,b,da,db):

   f0=float(np.dot(a,da))

             best_val=fv; best_alpha=av
   if best_alpha < 1e-8:
       return a.copy(), False
   a_new = a + best_alpha*d
   return _normalize(a_new), True

```python
"""
Search for a non-negative step-function sequence minimizing

    J(a) = 2*n*max(convolve(a,a)) / (sum(a))**2.

The program combines several complementary local/global heuristics:

1. A mirror / KKT-style alternating optimizer:
   - estimate active convolution lags by a tempered softmax of the current
     convolution profile;
   - compute the weighted bilinear gradient H_beta(a);
   - build a descent direction from the low-weighted-gradient region;
   - take an exact quadratic line search on the surrogate a^T H_beta a.

2. A spectral flattening iteration:
   - cap the current convolution peak and renormalize the target profile;
   - replace the nonzero spectral magnitudes by the Fourier magnitudes of
     this target;
   - keep the current phase, project back to the non-negative time support.

3. Optional alternating-LP polish for small sequences, and direct np.convolve
   scoring of saved candidates at the end to guard against FFT rounding.

The sequence is returned only after finite, non-negative cleaning and a final
direct verification when possible.
"""

import time
import numpy as np

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None

_fft_len_cache = {}
_ham_idx_cache = {}


def _next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _fft_length(n):
    n = int(n)
    cached = _fft_len_cache.get(n)
    if cached is not None:
        return cached
    # Need enough padding for a*a (2n-1) and for the H_beta multiplication
    # (full convolution of length n with length 2n-1, i.e. 3n-2).
    L = _next_pow2(max(1, 3 * n - 2))
    _fft_len_cache[n] = L
    return L


def _normalize(vec):
    v = np.asarray(vec, dtype=np.float64)
    if v.ndim != 1:
        v = v.reshape(-1)
    if v.size == 0:
        return np.ones(1, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    v = np.maximum(v, 0.0)
    s = float(np.sum(v))
    if not np.isfinite(s) or s < 1e-12:
        return np.ones(v.size, dtype=np.float64) / float(v.size)
    return v / s


def _consider_best(best_a, best_score, a, score):
    if a is None:
        return best_a, best_score
    try:
        score = float(score)
    except Exception:
        return best_a, best_score
    if np.isfinite(score) and score < best_score:
        av = np.asarray(a, dtype=np.float64)
        if av.size > 0 and np.all(np.isfinite(av)):
            return av.copy(), score
    return best_a, best_score


def _maybe_store_by_length(store, n, score, a):
    if a is None:
        return
    try:
        score = float(score)
    except Exception:
        return
    if not np.isfinite(score):
        return
    arr = np.asarray(a, dtype=np.float64)
    if arr.size == 0:
        return
    if not np.all(np.isfinite(arr)):
        return
    key = int(n)
    old = store.get(key)
    if old is None or score < old[0]:
        store[key] = (score, arr.copy())


def _fft_convolve_n(a, n, L):
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)
    q = 2 * int(n) - 1
    return c[:q]


def _score_n(a, n, L):
    c = _fft_convolve_n(a, n, L)
    if c.size == 0:
        return float("inf")
    M = float(np.max(c))
    if not np.isfinite(M):
        return float("inf")
    if M < 0.0:
        M = 0.0
    S = float(np.sum(a))
    if not np.isfinite(S) or S < 0.01:
        return float("inf")
    return float(2.0 * float(n) * M / (S * S))


def _score_direct(a):
    arr = np.asarray(a, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    n = arr.size
    if n == 0:
        return float("inf")
    S = float(np.sum(arr))
    if not np.isfinite(S) or S < 0.01:
        return float("inf")
    try:
        c = np.convolve(arr, arr, mode="full")
        if c.size == 0:
            return float("inf")
        M = float(np.max(c))
        if not np.isfinite(M):
            return float("inf")
        if M < 0.0:
            M = 0.0
        return float(2.0 * float(n) * M / (S * S))
    except Exception:
        return float("inf")


def _get_ham_idx(n):
    n = int(n)
    cached = _ham_idx_cache.get(n)
    if cached is not None:
        return cached
    q = 2 * n - 1
    idx = (q - 1) - np.arange(n, dtype=np.int64)
    _ham_idx_cache[n] = idx
    return idx


def _ham_mul(a, beta, L, n):
    """
    Returns H_beta a, where (H_beta a)_i = sum_k a_k beta_{i+k}.
    This is the auxiliary linear operator for weighted convolution sums.
    """
    n = int(n)
    idx = _get_ham_idx(n)
    rev_beta = beta[::-1]
    A = np.fft.rfft(a, L)
    B = np.fft.rfft(rev_beta, L)
    conv = np.fft.irfft(A * B, L)
    return conv[idx]


def _perturb(a, rng, scale):
    if rng is None:
        return a.copy()
    loga = np.log(a + 1e-12)
    noise = rng.normal(0.0, float(scale), a.size)
    v = np.exp(loga + noise)
    return _normalize(v)


def _mirror_line_search(a, b, da, db):
    """
    Exact line search for f(alpha) = (a + alpha d)^T H (a + alpha d)
    with d = b - a, da = H a, db = H b.
    """
    d = b - a
    f0 = float(np.dot(a, da))
    if not np.isfinite(f0):
        return a.copy(), False

    hud = db - da
    lin = float(np.dot(a, hud))
    quad = float(np.dot(d, hud))

    if not (np.isfinite(lin) and np.isfinite(quad)):
        return a.copy(), False

    best_val = f0
    best_alpha = 0.0
    eps = 1e-14 * max(1.0, abs(f0))

    f1 = f0 + 2.0 * lin + quad
    if np.isfinite(f1) and f1 < f0 - eps:
        best_val = f1
        best_alpha = 1.0

    if quad > 1e-16:
        av = -lin / (2.0 * quad)
        if np.isfinite(av) and 0.0 < av < 1.0:
            fv = f0 + 2.0 * lin * av + quad * av * av
            if np.isfinite(fv) and fv < best_val - eps:
                best_val = fv
                best_alpha = av

    if best_alpha < 1e-8:
        return a.copy(), False

    a_new = a + best_alpha * d
    return _normalize(a_new), True


def _mirror_optimize(a_init, n, L, max_iters, deadline,
                     beta_temp_start=1.5, beta_temp_end=0.20,
                     strength=8.0, use_inverse=True, damp=0.70,
                     uniform_mix=0.02, max_stagnation=80, rng=None):
    a = _normalize(a_init)
    best_a = a.copy()
    best_score = _score_n(a, n, L)

    start = time.monotonic()
    duration = max(1e-6, deadline - start)

    stagnation = 0
    fail_count = 0

    for _ in range(max_iters):
        now = time.monotonic()
        if now >= deadline:
            break

        c = _fft_convolve_n(a, n, L)
        if not np.all(np.isfinite(c)):
            a = best_a.copy()
            continue

        M = float(np.max(c))
        if M < 0.0:
            M = 0.0
        S = float(np.sum(a))
        if not np.isfinite(S) or S <= 1e-12:
            a = best_a.copy()
            continue

        score = float(2.0 * n * M / (S * S))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        if stagnation > int(max_stagnation):
            if rng is not None:
                esc_scale = 0.02 + 0.08 * float(rng.random())
            else:
                esc_scale = 0.05
            a = _perturb(best_a, rng, esc_scale)
            stagnation = 0
            fail_count = 0
            continue

        if M <= 1e-15:
            break

        t = c / M
        t = np.clip(t, 0.0, 1.0)

        prog = (now - start) / duration
        if prog < 0.0:
            prog = 0.0
        if prog > 1.0:
            prog = 1.0

        temp = beta_temp_start + (beta_temp_end - beta_temp_start) * prog
        if temp < 0.05:
            temp = 0.05

        arg = (t - 1.0) / temp
        w = np.exp(arg)
        w = np.where(np.isfinite(w) & (w > 1e-12), w, 1e-12)
        sw = float(np.sum(w))
        q = w.size

        if not np.isfinite(sw) or sw <= 0.0:
            beta = np.ones(q, dtype=np.float64) / q
        else:
            beta = w / sw
            if uniform_mix > 0.0:
                beta = (1.0 - uniform_mix) * beta + uniform_mix * (1.0 / q)
                sb = float(np.sum(beta))
                if np.isfinite(sb) and sb > 0.0:
                    beta = beta / sb
                else:
                    beta = np.ones(q, dtype=np.float64) / q

        p = _ham_mul(a, beta, L, n)
        if not np.all(np.isfinite(p)):
            a = _perturb(best_a, rng, 0.02)
            continue

        p_min = float(np.min(p))
        p_max = float(np.max(p))
        span = p_max - p_min

        if not np.isfinite(span) or span < 1e-14:
            a = _perturb(a, rng, 0.01)
            continue

        shift = (p - p_min) / span
        shift = np.clip(shift, 0.0, 1.0)

        if use_inverse:
            soft = 1.0 / (1.0 + strength * shift)
        else:
            ex = -strength * shift
            ex = np.clip(ex, -50.0, 0.0)
            soft = np.exp(ex)

        soft = _normalize(soft)

        if damp < 1.0:
            b = damp * soft + (1.0 - damp) * a
            b = _normalize(b)
        else:
            b = soft

        Hb = _ham_mul(b, beta, L, n)
        a_new, improved = _mirror_line_search(a, b, p, Hb)

        if not improved:
            cand = a + 0.05 * (b - a)
            cand = _normalize(cand)
            if np.all(np.isfinite(cand)):
                sc = _score_n(cand, n, L)
                if np.isfinite(sc) and sc < score:
                    a_new = cand
                    improved = True

        if improved:
            fail_count = 0
            if np.all(np.isfinite(a_new)):
                a = _normalize(a_new)
            else:
                a = best_a.copy()
        else:
            fail_count += 1
            if fail_count >= 4:
                if rng is not None:
                    esc_scale = 0.02 + 0.08 * float(rng.random())
                else:
                    esc_scale = 0.05
                a = _perturb(best_a, rng, esc_scale)
                fail_count = 0

    return best_a, best_score

    if time.monotonic() >= deadline:

def _spectral_optimize(a_init, n, L, deadline,
                       cap_start=0.95, cap_end=0.60,
                       mix=0.80, lower_frac=0.02, rng=None):
    a = _normalize(a_init)
    best_a = a.copy()
    best_score = _score_n(a, n, L)

    start = time.monotonic()
    duration = max(1e-6, deadline - start)
    q = 2 * int(n) - 1
    stagnation = 0

    for _ in range(1000000):
        now = time.monotonic()
        if now >= deadline:
            break

        A = np.fft.rfft(a, L)
        cc = np.fft.irfft(A * A, L)
        c = cc[:q]

        if not np.all(np.isfinite(c)):
            a = best_a.copy()
            continue

        M = float(np.max(c))
        if M < 0.0:
            M = 0.0
        S = float(np.sum(a))
        if not np.isfinite(S) or S <= 0.0:
            a = best_a.copy()
            continue

        score = float(2.0 * n * M / (S * S))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        if stagnation > 80:
            a = _perturb(best_a, rng, 0.05)
            stagnation = 0
            continue

        if M <= 1e-15:
            break

        prog = (now - start) / duration
        if prog < 0.0:
            prog = 0.0
        if prog > 1.0:
            prog = 1.0

        cap = cap_start + (cap_end - cap_start) * prog
        if cap < 1.0:
            lower = max(0.0, lower_frac * cap * M)
            target = np.clip(c, lower, cap * M)
        else:
            target = c.copy()

        total = float(np.sum(target))
        if not np.isfinite(total) or total <= 0.0:
            continue
        target = target / total

        B = np.fft.rfft(target, L)
        mag = np.sqrt(np.abs(B))

        absA = np.abs(A)
        phase = np.ones(A.shape, dtype=np.complex128)
        mask = absA > 1e-30
        phase[mask] = A[mask] / absA[mask]

        Anew = phase * mag
        anew = np.fft.irfft(Anew, L)
        anew = anew[:int(n)].copy()
        anew = np.maximum(anew, 0.0)

        snew = float(np.sum(anew))
        if not np.isfinite(snew) or snew < 1e-10:
            continue
        anew = anew / snew

        b = mix * anew + (1.0 - mix) * a
        a = _normalize(b)

    return best_a, best_score


def _build_conv_matrix(f):
    n = len(f)
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = f
    return C


def _lp_solve(C, M, n):
    if linprog is None or M <= 0.0:
        return None
    c_obj = -np.ones(n, dtype=np.float64)
    b_ub = M * np.ones(2 * n - 1, dtype=np.float64)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(
            c_obj,
            A_ub=C,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True}
        )
        if res.status == 0:
            return np.asarray(res.x, dtype=np.float64)
    except Exception:
        return None
    return None


def _lp_refine(a, n, L, deadline, max_iters=200, initial_blend=0.80):
    a = _normalize(a)
    best_a = a.copy()
    best_score = _score_n(a, n, L)
    blend = float(initial_blend)
    stagnation = 0

    for _ in range(max_iters):
        if time.monotonic() >= deadline:
            break

        c = _fft_convolve_n(a, n, L)
        M = float(np.max(c))
        if M < 0.0:
            M = 0.0
        S = float(np.sum(a))
        if not np.isfinite(S) or S <= 0.0:
            a = best_a.copy()
            continue

        score = float(2.0 * n * M / (S * S))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1
        if stagnation > 25:
            break

        if M <= 1e-15:
            break

        C = _build_conv_matrix(a)
        g = _lp_solve(C, M, n)
        if g is None:
            break

        g = np.where(np.isfinite(g), g, 0.0)
        g = np.maximum(g, 0.0)
        sg = float(np.sum(g))
        if not np.isfinite(sg) or sg < 1e-12:
            break
        g = g / sg

        a_new = (1.0 - blend) * a + blend * g
        a_new = _normalize(a_new)
        if not np.all(np.isfinite(a_new)):
            a = best_a.copy()
            break
        a = a_new
        blend = max(0.30, blend * 0.98)

    return best_a, best_score


def _interpolate_sequence(src, n_target):
    n_src = len(src)
    if n_src == int(n_target):
        return src.copy()
    if n_src == 1:
        out = np.full(int(n_target), float(src[0]), dtype=np.float64)
    else:
        x_src = np.linspace(0.0, 1.0, n_src)
        x_tgt = np.linspace(0.0, 1.0, n_target)
        out = np.interp(x_tgt, x_src, src)
    return np.maximum(out, 0.0)


def _make_inits(n, parent, rng):
    n = int(n)
    t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    raw = []

    if parent is not None:
        raw.append(_interpolate_sequence(parent, n))

    raw.append(1.0 - t)
    raw.append(np.exp(-3.0 * t))
    raw.append(np.maximum(0.0, (1.0 - t) * (1.0 + 0.5 * np.cos(6.0 * np.pi * t))))
    raw.append(np.maximum(0.0, (1.0 - t) * (1.0 + 0.7 * np.sin(10.0 * np.pi * t))))
    raw.append(np.exp(-((t - 0.3) ** 2) / 0.02) + 0.5 * np.exp(-((t - 0.7) ** 2) / 0.05))
    raw.append(np.maximum(0.0, (1.0 - t) ** 1.5 * (1.0 + 0.4 * np.cos(4.0 * np.pi * t + 1.0))))
    raw.append(np.maximum(0.0, (1.0 - t) * (1.0 + 0.3 * np.cos(14.0 * np.pi * t))))

    sig = float(rng.uniform(0.4, 2.0))
    raw.append(rng.lognormal(0.0, sig, n))

    p1 = float(rng.uniform(0.05, 0.45))
    p2 = float(rng.uniform(0.55, 0.95))
    w1 = float(rng.uniform(0.03, 0.08))
    w2 = float(rng.uniform(0.03, 0.08))
    amp2 = float(rng.uniform(0.3, 1.5))
    raw.append(np.exp(-((t - p1) ** 2) / (2.0 * w1 * w1)) +
               amp2 * np.exp(-((t - p2) ** 2) / (2.0 * w2 * w2)))

    alpha = float(rng.uniform(0.3, 1.8))
    beta = float(rng.uniform(0.3, 1.8))
    raw.append(np.maximum(t, 1e-6) ** alpha * np.maximum(1.0 - t, 1e-6) ** beta)

    out = []
    for v in raw:
        v = _normalize(v)
        if v.size == n and np.all(np.isfinite(v)):
            out.append(v)
    return out


def propose_candidate():
    """
    Returns the best non-negative sequence found under the time budget.
    """
    t0 = time.monotonic()
    deadline = t0 + 2280.0
    rng = np.random.default_rng(20240528)

    best_a = None
    best_score = float("inf")
    length_best = {}

    parent = None
    try:
        par = np.asarray(height_sequence_1, dtype=np.float64)
        if par.ndim != 1 or par.size == 0:
            par = None
        if par is not None:
            par = np.where(np.isfinite(par), par, 0.0)
            par = np.maximum(par, 0.0)
            s = float(np.sum(par))
            if np.isfinite(s) and s > 1e-12:
                parent = par / s
    except Exception:
        parent = None

    if parent is not None:
        pL = _fft_length(parent.size)
        ps = _score_n(parent, parent.size, pL)
        print(f"Starting parent: n={parent.size}, score={ps:.9f}")
        _maybe_store_by_length(length_best, parent.size, ps, parent)
        best_a, best_score = _consider_best(best_a, best_score, parent, ps)

    planned = [
        (500, 1100.0),
        (1000, 420.0),
        (2048, 300.0),
        (4096, 200.0),
        (256, 120.0),
    ]

    for n, budget in planned:
        now = time.monotonic()
        if now > deadline - 60.0:
            break

        rem = deadline - now
        alloc = min(float(budget), 0.98 * rem)
        if alloc <= 30.0:
            break

        n = int(n)
        L = _fft_length(n)
        inits = _make_inits(n, parent, rng)

        arrays = []
        scores = []

        for a0 in inits:
            a0 = _normalize(a0)
            arrays.append(a0)
            scores.append(_score_n(a0, n, L))

        if best_a is not None:
            a_best_n = _normalize(_interpolate_sequence(best_a, n))
            arrays.append(a_best_n)
            scores.append(_score_n(a_best_n, n, L))

        if not arrays:
            arrays = [np.ones(n, dtype=np.float64)]
            scores = [_score_n(arrays[0], n, L)]

        # Select a few initial candidates: three lowest-initial-score plus
        # a couple of random ones for diversity.
        idx_by_score = sorted(range(len(scores)), key=scores.__getitem__)
        cand = []
        for i in idx_by_score[:3]:
            if i not in cand:
                cand.append(i)

        if len(cand) < len(idx_by_score):
            remaining = [i for i in idx_by_score if i not in cand]
            m = min(2, len(remaining))
            if m > 0:
                extra_arr = np.asarray(remaining, dtype=int)
                extra = rng.choice(extra_arr, size=m, replace=False)
                if m == 1:
                    e = int(extra)
                    if e not in cand:
                        cand.append(e)
                else:
                    for e in extra.tolist():
                        e = int(e)
                        if e not in cand:
                            cand.append(e)

        cand = cand[:5]
        if not cand:
            cand = [0]

        cand_time = max(20.0, alloc / float(len(cand)))

        for ci, idx in enumerate(cand):
            now = time.monotonic()
            if now > deadline - 60.0:
                break

            a0 = arrays[idx]
            s0 = scores[idx]
            _maybe_store_by_length(length_best, n, s0, a0)
            best_a, best_score = _consider_best(best_a, best_score, a0, s0)

            cand_left = min(cand_time, deadline - now - 40.0)
            if cand_left < 15.0:
                continue

            start_cand = time.monotonic()
            spec_alloc = min(30.0, 0.18 * cand_left)

            if spec_alloc > 10.0:
                spec_deadline = start_cand + spec_alloc
                a1, s1 = _spectral_optimize(
                    a0, n, L, spec_deadline,
                    cap_start=0.95, cap_end=0.60,
                    mix=0.80, lower_frac=0.02, rng=rng
                )
                _maybe_store_by_length(length_best, n, s1, a1)
                best_a, best_score = _consider_best(best_a, best_score, a1, s1)
                start_mirror = a1 if s1 < s0 else a0
            else:
                start_mirror = a0

            mirror_deadline = min(deadline - 30.0, start_cand + 0.82 * cand_left)
            a2, s2 = _mirror_run_configs(start_mirror, n, L, mirror_deadline, rng)
            _maybe_store_by_length(length_best, n, s2, a2)
            best_a, best_score = _consider_best(best_a, best_score, a2, s2)

            print(
                f"n={n} cand={ci} mirror_score={s2:.9f} "
                f"global={best_score:.9f} length={len(best_a) if best_a is not None else 0}"
            )

    # Final polish on the current global best if time remains.
    if best_a is not None and time.monotonic() < deadline - 40.0:
        n = len(best_a)
        L = _fft_length(n)
        polish_deadline = min(deadline - 40.0, time.monotonic() + min(400.0, 0.5 * (deadline - time.monotonic())))
        a_pol, s_pol = _mirror_run_configs(best_a, n, L, polish_deadline, rng)
        _maybe_store_by_length(length_best, n, s_pol, a_pol)
        best_a, best_score = _consider_best(best_a, best_score, a_pol, s_pol)
        print(f"After mirror polish: score={best_score:.9f}, n={len(best_a)}")

    # Optional LP polish for relatively short sequences.
    if (
        linprog is not None
        and best_a is not None
        and time.monotonic() < deadline - 80.0
    ):
        n_best = len(best_a)
        # Only run LP on a manageable length. If the global best is very long,
        # downsample, polish, then check both lengths.
        if n_best <= 800:
            n_lp = n_best
            a_lp0 = best_a.copy()
        else:
            n_lp = 512
            a_lp0 = _normalize(_interpolate_sequence(best_a, n_lp))

        L_lp = _fft_length(n_lp)
        lp_deadline = min(deadline - 40.0, time.monotonic() + min(220.0, deadline - time.monotonic() - 40.0))
        a_lp, s_lp = _lp_refine(a_lp0, n_lp, L_lp, lp_deadline, max_iters=120, initial_blend=0.8)
        _maybe_store_by_length(length_best, n_lp, s_lp, a_lp)
        best_a, best_score = _consider_best(best_a, best_score, a_lp, s_lp)

        if n_best != n_lp:
            L_best = _fft_length(n_best)
            a_back = _normalize(_interpolate_sequence(a_lp, n_best))
            s_back = _score_n(a_back, n_best, L_best)
            _maybe_store_by_length(length_best, n_best, s_back, a_back)
            best_a, best_score = _consider_best(best_a, best_score, a_back, s_back)

        print(f"After LP polish: score={best_score:.9f}, n={len(best_a) if best_a is not None else 0}")

    # Choose final returned sequence using direct convolution to avoid
    # FFT-rounding issues, with a preference for shorter ties.
    final_arr = None
    final_score = float("inf")

    for key, item in length_best.items():
        n = int(key)
        if n <= 4096:
            arr = _normalize(item[1])
            ds = _score_direct(arr)
            if np.isfinite(ds):
                if (
                    ds < final_score - 1e-12
                    or (abs(ds - final_score) <= 1e-12 and
                        (final_arr is None or arr.size < final_arr.size))
                ):
                    final_score = ds
                    final_arr = arr

    if final_arr is None:
        if best_a is not None:
            final_arr = best_a
            final_score = _score_direct(best_a)
        else:
            final_arr = np.ones(100, dtype=np.float64)
            final_score = 2.0

    final_arr = np.asarray(final_arr, dtype=np.float64)
    final_arr = np.where(np.isfinite(final_arr), final_arr, 0.0)
    final_arr = np.maximum(final_arr, 0.0)
    sfinal = float(np.sum(final_arr))
    if not np.isfinite(sfinal) or sfinal <= 0.01:
        final_arr = np.ones(100, dtype=np.float64)
        sfinal = 100.0
    final_arr = final_arr / sfinal
    final_arr = np.clip(final_arr, 0.0, 1000.0)

    result = [float(x) for x in final_arr]

    print("\n=== Final candidate ===")
    print(f"n={len(result)} estimated_direct_score={final_score:.9f}")

    try:
        verified = float(evaluate_sequence(result))
        print(f"evaluate_sequence verified: {verified:.9f}")
        if not np.isfinite(verified):
            # Extremely defensive fallback.
            fallback = np.ones(100, dtype=np.float64)
            result = [float(x) for x in fallback]
            print("Fell back to length-100 uniform because verified score was not finite.")
    except Exception as e:
        print(f"Final verification exception: {e}")

    return result
```
