            it += 1
            continue

        g = g / g.sum()

        # Try blending
        improved = False
        for t in [t_blend, 0.3, 0.2, 0.1, 0.05]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            c_trial = _fft_convolve(a_trial)
            M_trial = float(np.max(c_trial))
            score_trial = 2.0 * n * M_trial
            if M_trial <= current_M + 1e-12:
                a = a_trial
                current_M = M_trial
                if score_trial < best_score:
                    best_score = score_trial
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                improved = True
                break
            no_improve += 1

        if not improved:
            no_improve += 5
            # Reduce blend factor
            t_blend = max(0.02, t_blend * 0.7)
            # Try perturbation
            if no_improve > 200:
                a = best_a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 1e-20)
                a = a / a.sum()
                c = _fft_convolve(a)
                current_M = float(np.max(c))
                no_improve = 0
                t_blend = 0.3

        if time.monotonic() - last_print > 15.0:
            print(f"    [LP] it={it}, M={current_M:.8f}, score={best_score:.6f}, no_imp={no_improve}")
            last_print = time.monotonic()

        # If converged (M not improving), switch to subgradient polishing
        if no_improve > 500:
            # Polish with subgradient
            c = _fft_convolve(best_a)
            k_max = int(np.argmax(c))
            momentum = np.zeros(n)
            eta = 0.1
            for _ in range(200):
                if time.monotonic() >= deadline:
                    break
                c = _fft_convolve(best_a)
                k_max = int(np.argmax(c))
                sc = 2.0 * n * float(c[k_max])
                if sc < best_score:
                    best_score = sc
                best_a, momentum = _subgradient_step(best_a, k_max, eta, momentum, 0.7)
                eta *= 0.999
            a = best_a.copy()
            c = _fft_convolve(a)
            current_M = float(np.max(c))
            no_improve = 0
            t_blend = 0.3

        it += 1

    return best_a, best_score


def _optimize_smooth(a0, time_budget, rng, p=8):
    """Optimize using smoothed penalty: minimize sum max(0, b[k]-M)^2 + lam*sum b[k]^2."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, np.inf
    a = a / s

    c = _fft_convolve(a)
    best_score = 2.0 * n * float(np.max(c))
    best_a = a.copy()

    deadline = time.monotonic() + time_budget
    eta = 0.05
    no_improve = 0
    momentum = np.zeros(n)
    mu = 0.8
    it = 0
    last_print = time.monotonic()

    while time.monotonic() < deadline:
        c = _fft_convolve(a)
        M = float(np.max(c))
        score = 2.0 * n * M

        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            if time.monotonic() - last_print > 15.0:
                print(f"    [smooth] it={it}, score={score:.6f}, p={p}")
                last_print = time.monotonic()
        else:
            no_improve += 1

        if no_improve > 400:
            a = best_a * np.exp(rng.normal(0, 0.06, n))
            a = np.maximum(a, 1e-20)
            a = a / a.sum()
            no_improve = 0
            momentum = np.zeros(n)
            eta = 0.05

        # Compute gradient of sum_k b[k]^p
        c_p = c ** (p - 1)
        # grad[j] = 2p * sum_m c_p[m+j] * a[m] for valid m
        # This is a cross-correlation
        # Pad c_p to length 2n-1
        # grad[j] = 2p * sum_{m=0}^{n-1} c_p[m+j] * a[m]
        # = 2p * (cross-correlation of a with c_p)[j]
        # Use FFT: cross-corr = ifft(conj(FFT(a)) * FFT(c_p_padded))
        # Actually: sum_m a[m] * c_p[m+j] = (a_flipped * c_p)[j + n - 1]
        # Let's just do it with FFT
        fft_size = _next_pow2(2 * n)
        A = np.fft.rfft(a, fft_size)
        Cp = np.fft.rfft(c_p, fft_size)
        # cross-corr: sum_m a[m] * c_p[m+j]
        # = ifft(conj(A) * Cp) evaluated at appropriate indices
        # Actually: ifft(conj(FFT(a)) * FFT(c_p)) gives sum_m a[m] * c_p[m-n+j+1]... 
        # Let me just compute it directly for clarity (O(n^2) but n=2000 is manageable)
        # Actually O(n^2) for n=2000 is 4M operations per iteration, too slow.
        # Use the convolution trick:
        # sum_m a[m] * c_p[m+j] for m=0..n-1
        # Let d = flip(a), so d[n-1-m] = a[m]
        # Then sum_m a[m] * c_p[m+j] = sum_m d[n-1-m] * c_p[m+j]
        # = (d * c_p)[n-1+j] (convolution)
        d = a[::-1]
        fft_d = np.fft.rfft(d, fft_size)
        cross = np.fft.irfft(fft_d * Cp, fft_size)
        # cross[n-1+j] for j=0..n-1
        grad = 2.0 * p * cross[n - 1:n - 1 + n]

        # Normalize gradient
        g_max = np.abs(grad).max()
        if g_max < 1e-15:
            break
        grad = grad / g_max

        momentum = mu * momentum + (1.0 - mu) * grad
        a = a * np.exp(-eta * momentum)
        a = np.maximum(a, 1e-20)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            eta *= 0.5
            it += 1
            continue
        a = a / s

        # Adaptive step
        if no_improve > 200:
            eta *= 0.96
        elif no_improve < 30:
            eta = min(eta * 1.005, 0.2)

        it += 1

    return best_a, best_score


def _optimize_subgrad(a0, time_budget, rng):
    """Standard multiplicative subgradient descent."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, np.inf
    a = a / s

    c = _fft_convolve(a)
    best_score = 2.0 * n * float(np.max(c))
    best_a = a.copy()

    eta = 0.15
    no_improve = 0
    momentum = np.zeros(n)
    mu = 0.7
    deadline = time.monotonic() + time_budget
    it = 0
    last_print = time.monotonic()

    while time.monotonic() < deadline:
        c = _fft_convolve(a)
        k_max = int(np.argmax(c))
        score = 2.0 * n * c[k_max]

        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            if time.monotonic() - last_print > 10.0:
                print(f"    [subgrad] it={it}, score={score:.6f}")
                last_print = time.monotonic()
        else:
            no_improve += 1

        if no_improve > 600:
            a = best_a * np.exp(rng.normal(0, 0.07, n))
            a = np.maximum(a, 1e-20)
            a = a / a.sum()
            no_improve = 0
            eta = 0.15
            momentum = np.zeros(n)
            it += 1
            continue

        a, momentum = _subgradient_step(a, k_max, eta, momentum, mu)

        if no_improve > 400:
            eta *= 0.97
        elif no_improve < 30:
            eta = min(eta * 1.003, 0.4)

        if it % 300 == 150 and rng.random() < 0.15:
            block = max(1, n // 40)
            start = int(rng.integers(0, n - block))
            a[start:start + block] *= np.exp(rng.normal(0, 0.12))
            a = np.maximum(a, 1e-20)
            a = a / a.sum()

        it += 1

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Minimizes 2*n*max(conv(a,a))/sum(a)^2.
    Uses LP-based iterative optimization (solving max sum(g) s.t. (a*g)<=M, g>=0),
    smoothed penalty optimization, and subgradient polishing.
    Multiple initializations and sequence lengths are tried.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 20.0

    best_overall = None
    best_overall_score = np.inf

    # Build candidate initializations
    candidates = []

    # From previous best
    try:
        hs = np.array(height_sequence_1, dtype=np.float64)
        hs = np.clip(hs, 0, None)
        if hs.size > 100 and hs.sum() > 0.01:
            candidates.append(("prev_best", hs.copy()))
    except NameError:
        pass
    except Exception:
        pass

    # Sinusoidal modulations (these worked best previously)
    for n in [2000, 3000, 4000]:
        x = np.arange(n, dtype=np.float64)
        for amp in [0.3, 0.4, 0.5, 0.6]:
            for phase in [0, np.pi / 4, np.pi / 3, np.pi / 2]:
                a = 1.0 + amp * np.sin(2 * np.pi * x / n + phase)
                a = np.maximum(a, 0.01)
                candidates.append((f"sin_{n}_{amp}_{phase:.2f}", a))

    # Double sinusoid
    for n in [2000, 3000]:
        x = np.arange(n, dtype=np.float64)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * x / n) + 0.2 * np.sin(4 * np.pi * x / n + 0.5)
        a = np.maximum(a, 0.01)
        candidates.append((f"dsin_{n}", a))

    # Exponential decay (worked well before)
    for n in [2000, 3000, 4000]:
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-0.5 * x / n) + 0.1
        candidates.append((f"decay_{n}", a))

    # Triangular
    for n in [2000, 3000]:
        x = np.arange(n, dtype=np.float64)
        mid = n / 2.0
        a = 1.0 - 0.6 * np.abs(x - mid) / mid
        a = np.maximum(a, 0.01)
        candidates.append((f"tri_{n}", a))

    # Cosine window
    for n in [2000, 3000]:
        x = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 + np.cos(2 * np.pi * x / n))
        a = np.maximum(a, 0.01)
        candidates.append((f"coswin_{n}", a))

    print(f"Total candidates: {len(candidates)}")

    # Quick initial evaluation
    scored = []
    for name, a0 in candidates:
        a0 = np.maximum(a0, 0)
        s = a0.sum()
        if s < 1e-15:
            continue
        a0 = a0 / s
        c0 = _fft_convolve(a0)
        init_score = 2.0 * len(a0) * float(np.max(c0))
        scored.append((init_score, name, a0))

    scored.sort(key=lambda x: x[0])
    print("Initial scores (sorted, top 12):")
    for i, (sc, name, _) in enumerate(scored[:12]):
        print(f"  {i + 1}. {name}: {sc:.6f} (n={len(_)})")

    # Track which n values we've precomputed structure for
    conv_structures = {}

    def get_structure(n):
        if n not in conv_structures:
            conv_structures[n] = _precompute_conv_structure(n)
        return conv_structures[n]

    # Allocate time: top 5 candidates get LP optimization, next 5 get subgradient
    n_candidates = len(scored)
    remaining = deadline - time.monotonic()
    
    # Give top 4 the LP method, rest get subgradient
    n_lp = min(4, n_candidates)
    n_sub = min(4, max(0, n_candidates - n_lp))
    
    # Time allocation
    lp_time_each = remaining * 0.55 / max(1, n_lp)
    sub_time_each = remaining * 0.30 / max(1, n_sub)
    polish_time = remaining * 0.15

    print(f"\nTime allocation: LP={lp_time_each:.0f}s x{n_lp}, Sub={sub_time_each:.0f}s x{n_sub}, Polish={polish_time:.0f}s")

    # Phase 1: LP-based optimization on top candidates
    for i in range(n_lp):
        if time.monotonic() > deadline - 30:
            break
        init_score, name, a0 = scored[i]
        n = len(a0)
        t_alloc = min(lp_time_each, max(10.0, deadline - time.monotonic() - 30))
        print(f"\n[LP] Optimizing '{name}' (n={n}, init={init_score:.6f}, time={t_alloc:.0f}s)")

        row_idx, col_idx = get_structure(n)
        a1, s1 = _optimize_lp(a0, t_alloc, rng, row_idx, col_idx)
        eval1 = evaluate_sequence(a1.tolist())
        print(f"  LP result: internal={s1:.6f}, eval={eval1:.6f}")

        if np.isfinite(eval1) and eval1 < best_overall_score:
            best_overall_score = eval1
            best_overall = a1.copy()
            print(f"  *** NEW BEST: {eval1:.6f} ***")

    # Phase 2: Subgradient on next candidates (and also from LP results)
    for i in range(n_lp, min(n_lp + n_sub, n_candidates)):
        if time.monotonic() > deadline - 30:
            break
        init_score, name, a0 = scored[i]
        t_alloc = min(sub_time_each, max(10.0, deadline - time.monotonic() - 20))
        print(f"\n[Sub] Optimizing '{name}' (n={len(a0)}, init={init_score:.6f}, time={t_alloc:.0f}s)")

        a2, s2 = _optimize_subgrad(a0, t_alloc, rng)
        eval2 = evaluate_sequence(a2.tolist())
        print(f"  Subgrad result: internal={s2:.6f}, eval={eval2:.6f}")

        if np.isfinite(eval2) and eval2 < best_overall_score:
            best_overall_score = eval2
            best_overall = a2.copy()
            print(f"  *** NEW BEST: {eval2:.6f} ***")

    # Phase 3: Smoothed optimization from best found
    remaining = deadline - time.monotonic() - 10
    if best_overall is not None and remaining > 60:
        print(f"\n[Smooth] Polishing with p=8, time={remaining * 0.5:.0f}s")
        a3, s3 = _optimize_smooth(best_overall, remaining * 0.5, rng, p=8)
        eval3 = evaluate_sequence(a3.tolist())
        print(f"  Smooth result: internal={s3:.6f}, eval={eval3:.6f}")
        if np.isfinite(eval3) and eval3 < best_overall_score:
            best_overall_score = eval3
            best_overall = a3.copy()
            print(f"  *** NEW BEST: {eval3:.6f} ***")

    # Phase 4: Final subgradient polishing
    remaining = deadline - time.monotonic() - 5
    if best_overall is not None and remaining > 30:
        print(f"\n[Polish] Final subgradient polish, time={remaining:.0f}s")
        a4, s4 = _optimize_subgrad(best_overall, remaining, rng)
        eval4 = evaluate_sequence(a4.tolist())
        print(f"  Polish result: internal={s4:.6f}, eval={eval4:.6f}")
        if np.isfinite(eval4) and eval4 < best_overall_score:
            best_overall_score = eval4
            best_overall = a4.copy()
            print(f"  *** NEW BEST: {eval4:.6f} ***")

    # Final verification
    if best_overall is not None:
        final_score = evaluate_sequence(best_overall.tolist())
        elapsed = time.monotonic() - total_start
        print(f"\nFinal result: score={final_score:.6f}, n={len(best_overall)}, elapsed={elapsed:.1f}s")
        return [float(x) for x in best_overall]
    else:
        a = np.ones(1000, dtype=np.float64)
        print("Falling back to uniform sequence")
        return a.tolist()
