```python
"""
Search for a non-negative step-function coefficient sequence minimizing
    2*n*max(conv(a,a)) / sum(a)^2

The main routine uses FFT-accelerated mirror descent on the log-height
variables, with adaptive smoothing of the maximum convolution coefficient.
It alternates between several useful weightings of the active convolution
lags:
  * softmax weights with a temperature that is increased/decreased adaptively,
  * fixed top-K active-lag weights,
  * single-largest-lag subgradients for fast polishing.

The optimizer includes line-search acceptance, adaptive step-size, momentum,
random restarts, and multiple deterministic/stochastic initializations.
"""

import numpy as np
import time
import math


LOG_MIN = -40.0
LOG_MAX = 20.0


def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _normalize_from_y(y):
    y = np.asarray(y, dtype=np.float64)
    if y.size == 0:
        return y.copy()

    m = float(np.max(y))
    if not np.isfinite(m):
        return np.full(y.shape, 1.0 / float(y.size), dtype=np.float64)

    e = np.exp(y - m)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(y.shape, 1.0 / float(y.size), dtype=np.float64)

    a = e / s
    if not np.all(np.isfinite(a)):
        return np.full(y.shape, 1.0 / float(y.size), dtype=np.float64)
    return a


def _make_initial_positive(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1000, dtype=np.float64)

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)

    s = float(a.sum())
    if s <= 1e-15:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.size)

    a = a / s
    a = np.maximum(a, 1e-30)
    a = a / a.sum()
    return a


def _y_from_a(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 1e-300)
    y = np.log(a)
    return np.clip(y, LOG_MIN, LOG_MAX)


def _self_conv_and_a(a, fsize):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    size = 2 * n - 1
    if fsize < size:
        fsize = _next_pow2(size)

    A = np.fft.rfft(a, fsize)
    c = np.fft.irfft(A * A, fsize)
    c = c[:size]
    np.maximum(c, 0.0, out=c)
    return c, A


def _self_conv(a):
    n = a.size
    fsize = _next_pow2(2 * n - 1)
    return _self_conv_and_a(a, fsize)[0]


def _score_sequence(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.inf

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)

    s = float(a.sum())
    if s <= 1e-15:
        return np.inf

    a = a / s
    c = _self_conv(a)
    if c.size == 0:
        return np.inf
    return float(2.0 * a.size * np.max(c))


def _soft_weights(c, T):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return np.array([], dtype=np.float64)

    cmax = float(np.max(c))
    if cmax <= 0.0 or not np.isfinite(cmax):
        p = np.full(c.size, 1.0 / float(c.size), dtype=np.float64)
        return p

    if T <= 1e-300:
        p = np.zeros(c.size, dtype=np.float64)
        p[int(np.argmax(c))] = 1.0
        return p

    scaled = (c - cmax) / T
    p = np.exp(scaled)
    s = float(p.sum())

    if not np.isfinite(s) or s <= 0.0:
        p = np.zeros(c.size, dtype=np.float64)
        p[int(np.argmax(c))] = 1.0
        return p

    p = p / s
    return p


def _weights_topk(c, K):
    c = np.asarray(c, dtype=np.float64)
    m = c.size
    if m == 0:
        return np.array([], dtype=np.float64)

    if K <= 1:
        p = np.zeros(m, dtype=np.float64)
        p[int(np.argmax(c))] = 1.0
        return p

    if K >= m:
        return np.full(m, 1.0 / float(m), dtype=np.float64)

    idx = np.argpartition(c, m - K)[m - K:]
    p = np.zeros(m, dtype=np.float64)
    p[idx] = 1.0 / float(K)
    return p


def _dir_from_weights_a_p(a, p, A, fsize):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    P = np.fft.rfft(p, fsize)
    d = np.fft.irfft(P * A, fsize)
    return d[n - 1:n - 1 + n].copy()


def _center_scale(g):
    g = np.asarray(g, dtype=np.float64)
    if g.size == 0:
        return None

    g = g - float(np.mean(g))
    scale = float(np.max(np.abs(g)))
    if not np.isfinite(scale) or scale <= 1e-15:
        return None
    return g / scale


def _optimize_mirror(a0, budget, rng, T_frac=0.15, topk=0, eta0=0.10,
                     beta=0.7, patience=1500, label="mirror"):
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)

    s = float(a.sum())
    if a.size == 0 or s <= 1e-15:
        return None, np.inf

    a = a / s
    n = a.size
    fsize = _next_pow2(3 * n - 2)

    y = _y_from_a(a)
    c, A = _self_conv_and_a(a, fsize)
    current = float(2.0 * n * np.max(c))
    if not np.isfinite(current):
        return a, np.inf

    best_a = a.copy()
    best_y = y.copy()
    best_score = current

    momentum = np.zeros(n, dtype=np.float64)
    eta = eta0
    no_improve = 0
    it = 0

    T = T_frac * max(1e-300, float(np.max(c)))
    T_cap_frac = max(0.30, 1.25 * float(T_frac))

    deadline = time.monotonic() + max(0.0, float(budget))
    last_print = time.monotonic()

    while time.monotonic() < deadline:
        if current < best_score - 1e-12:
            best_score = current
            best_a = a.copy()
            best_y = y.copy()
            no_improve = 0
            eta = min(0.25, eta * 1.02)
        else:
            no_improve += 1

        if current > best_score * 1.01 + 1e-9 or no_improve > patience or eta < 0.004:
            sigma = 0.025
            y = best_y + rng.normal(0.0, sigma, n)
            y = np.clip(y, LOG_MIN, LOG_MAX)
            a = _normalize_from_y(y)
            c, A = _self_conv_and_a(a, fsize)
            current = float(2.0 * n * np.max(c))

            if not np.isfinite(current):
                a = best_a.copy()
                y = best_y.copy()
                c, A = _self_conv_and_a(a, fsize)
                current = best_score

            momentum = np.zeros(n, dtype=np.float64)
            eta = eta0
            no_improve = 0
            cmax_new = max(1e-300, current / (2.0 * n))
            T = max(0.02 * cmax_new, T * 1.5)
            it += 1
            continue

        cmax_cur = max(1e-300, float(np.max(c)))

        if topk <= 0:
            T_cap = max(1e-300, T_cap_frac * cmax_cur)
            T_floor = max(1e-300, 0.002 * cmax_cur)

            if no_improve < 200:
                T = max(T_floor, min(T_cap, T * 0.998))
            elif no_improve > 800:
                T = min(T_cap, T * 1.02)
            else:
                T = max(T_floor, min(T_cap, T * 0.9995))

            if not np.isfinite(T) or T <= 0.0:
                T = T_floor * 10.0

            p = _soft_weights(c, T)
        else:
            p = _weights_topk(c, int(topk))

        direction = _dir_from_weights_a_p(a, p, A, fsize)
        g = _center_scale(direction)

        if g is None:
            y = best_y + rng.normal(0.0, 0.01, n)
            y = np.clip(y, LOG_MIN, LOG_MAX)
            a = _normalize_from_y(y)
            c, A = _self_conv_and_a(a, fsize)
            current = float(2.0 * n * np.max(c))
            if not np.isfinite(current):
                a = best_a.copy()
                y = best_y.copy()
                c, A = _self_conv_and_a(a, fsize)
                current = best_score
            momentum = np.zeros(n, dtype=np.float64)
            it += 1
            continue

        momentum = beta * momentum + (1.0 - beta) * g

        accepted = False
        base_y = y
        for factor in (1.0, 0.5, 0.25):
            y_try = np.clip(base_y - eta * momentum * factor, LOG_MIN, LOG_MAX)
            a_try = _normalize_from_y(y_try)
            c_try, A_try = _self_conv_and_a(a_try, fsize)
            s_try = float(2.0 * n * np.max(c_try))

            if not np.isfinite(s_try):
                continue

            tol = 1e-12 + 1e-11 * max(1.0, current)
            if s_try <= current + tol and s_try <= best_score * 1.005 + 1e-8:
                accepted = True
                y = y_try
                a = a_try
                c = c_try
                A = A_try
                current = s_try
                break

        if accepted:
            if current < best_score - 1e-12:
                best_score = current
                best_a = a.copy()
                best_y = y.copy()
                no_improve = 0
                eta = min(0.25, eta * 1.02)
            else:
                if no_improve < 200:
                    eta = min(0.25, eta * 1.001)
                else:
                    eta = max(0.004, eta * 0.998)
        else:
            eta = max(0.002, eta * 0.5)

        it += 1
        if it % 10 == 0 and time.monotonic() - last_print > 15.0:
            print(f"    {label}: it={it} best={best_score:.8f} cur={current:.8f} eta={eta:.4f}")
            last_print = time.monotonic()

    if not np.isfinite(best_score):
        best_score = current
    return best_a, best_score


def _optimize_max_subgrad(a0, budget, rng, eta0=0.10, beta=0.7, patience=1500):
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)

    s = float(a.sum())
    if a.size == 0 or s <= 1e-15:
        return None, np.inf

    a = a / s
    n = a.size
    fsize = _next_pow2(3 * n - 2)

    y = _y_from_a(a)
    c, _ = _self_conv_and_a(a, fsize)
    current = float(2.0 * n * np.max(c))
    if not np.isfinite(current):
        return a, np.inf

    best_a = a.copy()
    best_y = y.copy()
    best_score = current

    momentum = np.zeros(n, dtype=np.float64)
    eta = eta0
    no_improve = 0
    it = 0

    deadline = time.monotonic() + max(0.0, float(budget))
    last_print = time.monotonic()

    while time.monotonic() < deadline:
        if current < best_score - 1e-12:
            best_score = current
            best_a = a.copy()
            best_y = y.copy()
            no_improve = 0
            eta = min(0.25, eta * 1.02)
        else:
            no_improve += 1

        if current > best_score * 1.01 + 1e-9 or no_improve > patience or eta < 0.004:
            sigma = 0.025
            y = best_y + rng.normal(0.0, sigma, n)
            y = np.clip(y, LOG_MIN, LOG_MAX)
            a = _normalize_from_y(y)
            c, _ = _self_conv_and_a(a, fsize)
            current = float(2.0 * n * np.max(c))

            if not np.isfinite(current):
                a = best_a.copy()
                y = best_y.copy()
                c, _ = _self_conv_and_a(a, fsize)
                current = best_score

            momentum = np.zeros(n, dtype=np.float64)
            eta = eta0
            no_improve = 0
            it += 1
            continue

        k = int(np.argmax(c))
        g = np.zeros(n, dtype=np.float64)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            g[lo:hi + 1] = a[k - idx]

        g = _center_scale(g)
        if g is None:
            y = best_y + rng.normal(0.0, 0.01, n)
            y = np.clip(y, LOG_MIN, LOG_MAX)
            a = _normalize_from_y(y)
            c, _ = _self_conv_and_a(a, fsize)
            current = float(2.0 * n * np.max(c))
            if not np.isfinite(current):
                a = best_a.copy()
                y = best_y.copy()
                c, _ = _self_conv_and_a(a, fsize)
                current = best_score
            momentum = np.zeros(n, dtype=np.float64)
            it += 1
            continue

        momentum = beta * momentum + (1.0 - beta) * g

        accepted = False
        base_y = y
        for factor in (1.0, 0.5, 0.25):
            y_try = np.clip(base_y - eta * momentum * factor, LOG_MIN, LOG_MAX)
            a_try = _normalize_from_y(y_try)
            c_try, _ = _self_conv_and_a(a_try, fsize)
            s_try = float(2.0 * n * np.max(c_try))

            if not np.isfinite(s_try):
                continue

            tol = 1e-12 + 1e-11 * max(1.0, current)
            if s_try <= current + tol and s_try <= best_score * 1.005 + 1e-8:
                accepted = True
                y = y_try
                a = a_try
                c = c_try
                current = s_try
                break

        if accepted:
            if current < best_score - 1e-12:
                best_score = current
                best_a = a.copy()
                best_y = y.copy()
                no_improve = 0
                eta = min(0.25, eta * 1.02)
            else:
                if no_improve < 200:
                    eta = min(0.25, eta * 1.001)
                else:
                    eta = max(0.004, eta * 0.998)
        else:
            eta = max(0.002, eta * 0.5)

        it += 1
        if it % 10 == 0 and time.monotonic() - last_print > 15.0:
            print(f"    maxsub: it={it} best={best_score:.8f} cur={current:.8f} eta={eta:.4f}")
            last_print = time.monotonic()

    if not np.isfinite(best_score):
        best_score = current
    return best_a, best_score


def _scored_key(item):
    return item[0]


def _build_candidates(rng):
    cands = []

    try:
        hs = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        hs = np.where(np.isfinite(hs), hs, 0.0)
        hs = np.clip(hs, 0.0, 1000.0)
        if hs.size > 10 and hs.sum() > 0.0:
            provided = _make_initial_positive(hs)
            if provided.size > 8192:
                x_new = np.linspace(0.0, 1.0, 8192, endpoint=False)
                x_old = np.linspace(0.0, 1.0, provided.size, endpoint=False)
                provided = _make_initial_positive(np.interp(x_new, x_old, provided))

            cands.append(("provided", provided))
            cands.append(("provided_rev", provided[::-1].copy()))

            for n in (2048, 3000, 4096):
                if abs(n - provided.size) > 20:
                    x_old = np.linspace(0.0, 1.0, provided.size, endpoint=False)
                    x_new = np.linspace(0.0, 1.0, n, endpoint=False)
                    arr = np.interp(x_new, x_old, provided)
                    a = _make_initial_positive(arr)
                    cands.append((f"interp_{n}", a))

            n0 = provided.size
            x0 = np.arange(n0)
            for factor in (0.05, 0.12, 0.25):
                mod = 1.0 + factor * np.sin(2.0 * np.pi * x0 / float(n0))
                a = _make_initial_positive(provided * mod)
                cands.append((f"provided_mix_{int(factor*1000)}", a))
    except Exception:
        pass

    for n in (2000, 2048, 3000, 4096):
        x = np.arange(n, dtype=np.float64)
        t = x / float(n)

        sine_specs = (
            (1, 0.40, 0.0),
            (1, 0.50, 0.7853981633974483),
            (2, 0.40, 0.0),
            (3, 0.30, 0.7853981633974483),
        )
        for freq, amp, phase in sine_specs:
            a = 1.0 + float(amp) * np.sin(2.0 * np.pi * float(freq) * t + float(phase))
            a = np.maximum(a, 0.02)
            a = a / a.sum()
            cands.append((f"sin_{n}_f{freq}_a{int(amp*1000)}", a))

        w = 0.15 + 0.85 * np.sin(np.pi * t) ** 2
        a = (1.0 + 0.5 * np.sin(2.0 * np.pi * t)) * w
        a = np.maximum(a, 0.02)
        a = a / a.sum()
        cands.append((f"win_{n}", a))

        for frac in (0.30, 0.50):
            a = np.exp(-0.5 * ((t - frac) / 0.15) ** 2) + 0.05
            a = a / a.sum()
            cands.append((f"gauss_{n}_{int(frac*100)}", a))

        for trial in range(2):
            base = np.ones(n, dtype=np.float64)
            for m in range(1, 6):
                amp = rng.uniform(0.05, 0.35) / math.sqrt(float(m))
                phase = rng.uniform(0.0, 2.0 * np.pi)
                base += amp * np.sin(2.0 * np.pi * float(m) * t + phase)
            a = np.maximum(base, 0.02)
            a = a / a.sum()
            cands.append((f"rand_{n}_{trial}", a))

    return cands


def _consider_candidate(best_seq, best_score, cand):
    if cand is None:
        return best_seq, best_score

    arr = np.asarray(cand, dtype=np.float64).ravel().copy()
    if arr.size == 0:
        return best_seq, best_score

    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)

    internal = _score_sequence(arr)
    if not np.isfinite(internal):
        return best_seq, best_score

    if internal < best_score + 1e-8:
        try:
            ev = evaluate_sequence(arr.tolist())
        except Exception:
            ev = internal

        if np.isfinite(ev) and ev < best_score - 1e-12:
            return arr, float(ev)

    return best_seq, best_score


def _safe_finalize(seq):
    arr = np.asarray(seq, dtype=np.float64).ravel().copy()
    if arr.size == 0:
        arr = np.ones(2000, dtype=np.float64)

    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)

    s = float(arr.sum())
    if s > 0.0:
        arr = arr / s
    else:
        arr = np.ones(arr.size if arr.size > 0 else 2000, dtype=np.float64)

    if arr.size < 2:
        arr = np.ones(2000, dtype=np.float64)

    out = []
    for v in arr:
        out.append(float(v))

    try:
        ev = evaluate_sequence(out)
    except Exception:
        ev = np.inf

    if not np.isfinite(ev):
        out = [1.0] * 2000

    return out


def propose_candidate(seed=0, budget_s=2350.0):
    """
    Main search entry point.
    """
    start = time.monotonic()
    total_budget = min(float(budget_s), 2400.0)
    deadline = start + total_budget - 35.0

    rng = np.random.default_rng(seed)

    print("Building initial candidates...")
    candidates = _build_candidates(rng)
    print(f"  generated {len(candidates)} pre-ROIs")

    scored = []
    initial_deadline = min(deadline, start + 120.0)
    for name, a in candidates:
        if time.monotonic() > initial_deadline:
            break
        sc = _score_sequence(a)
        if np.isfinite(sc):
            scored.append((sc, name, a.copy()))

    if not scored:
        print("No valid initial candidates; using uniform fallback.")
        return _safe_finalize(np.ones(2000, dtype=np.float64))

    scored.sort(key=_scored_key)
    print("Best initial candidates:")
    for i in range(min(10, len(scored))):
        print(f"  {i+1:2d} {scored[i][1]}: {scored[i][0]:.8f}")

    chosen = []
    chosen_names = set()
    for item in scored:
        if len(chosen) >= 10:
            break
        name = item[1]
        if name not in chosen_names:
            chosen.append(item)
            chosen_names.add(name)

    for item in scored:
        if item[1].startswith("provided") and item[1] not in chosen_names:
            chosen.append(item)
            chosen_names.add(item[1])

    if not chosen:
        chosen = [scored[0]]

    best_seq = chosen[0][2].copy()
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        best_score = chosen[0][0]
    if not np.isfinite(best_score):
        best_score = chosen[0][0]

    print(f"Initial stored best: {best_score:.8f} ({chosen[0][1]}, n={len(best_seq)})")

    reserve = 120.0
    usable = max(0.0, (deadline - time.monotonic()) - reserve)
    total_w = 0.0
    weights = []
    for i in range(len(chosen)):
        w = float(len(chosen) - i)
        weights.append(w)
        total_w += w

    if total_w <= 0.0:
        total_w = 1.0
        weights = [1.0] * len(chosen)

    for idx, item in enumerate(chosen):
        init_score, name, a0 = item

        if time.monotonic() > deadline:
            break

        t_alloc = usable * weights[idx] / total_w
        t_alloc = min(t_alloc, max(10.0, deadline - time.monotonic() - 30.0))
        if t_alloc < 8.0:
            continue

        print(f"\n=== optimizing {name} (n={len(a0)}, init={init_score:.8f}, t={t_alloc:.0f}s) ===")

        t1 = 0.45 * t_alloc
        t2 = 0.25 * t_alloc
        t3 = 0.15 * t_alloc
        t4 = max(5.0, t_alloc - t1 - t2 - t3)

        a1, s1 = _optimize_mirror(a0, t1, rng, T_frac=0.20, topk=0,
                                  eta0=0.10, label=f"{name}.broad")
        best_seq, best_score = _consider_candidate(best_seq, best_score, a1)
        print(f"  broad: internal={s1:.8f} global={best_score:.8f}")

        a2, s2 = _optimize_mirror(a1 if a1 is not None else a0, t2, rng, T_frac=0.0,
                                  topk=64, eta0=0.07, label=f"{name}.top64")
        best_seq, best_score = _consider_candidate(best_seq, best_score, a2)
        print(f"  top64: internal={s2:.8f} global={best_score:.8f}")

        a3, s3 = _optimize_mirror(best_seq if best_seq is not None else a0, t3, rng,
                                  T_frac=0.02, topk=0, eta0=0.045,
                                  label=f"{name}.sharp")
        best_seq, best_score = _consider_candidate(best_seq, best_score, a3)
        print(f"  sharp: internal={s3:.8f} global={best_score:.8f}")

        if t4 > 5.0:
            a4, s4 = _optimize_max_subgrad(best_seq if best_seq is not None else a0, t4,
                                           rng, eta0=0.07, label=f"{name}.max")
            best_seq, best_score = _consider_candidate(best_seq, best_score, a4)
            print(f"  maxsub: internal={s4:.8f} global={best_score:.8f}")

    remaining = max(0.0, deadline - time.monotonic())
    if best_seq is not None and remaining > 45.0:
        print(f"\nFinal polishing with {remaining:.0f}s remaining")

        t_pol1 = min(120.0, 0.40 * remaining)
        t_pol2 = min(90.0, 0.35 * remaining)
        t_pol3 = max(0.0, remaining - t_pol1 - t_pol2 - 5.0)

        a1, s1 = _optimize_mirror(best_seq, t_pol1, rng, T_frac=0.015, topk=0,
                                  eta0=0.035, patience=1800, label="final.smooth")
        best_seq, best_score = _consider_candidate(best_seq, best_score, a1)
        print(f"  final.smooth: internal={s1:.8f} global={best_score:.8f}")

        a2, s2 = _optimize_mirror(best_seq, t_pol2, rng, T_frac=0.0, topk=128,
                                  eta0=0.045, patience=1800, label="final.top128")
        best_seq, best_score = _consider_candidate(best_seq, best_score, a2)
        print(f"  final.top128: internal={s2:.8f} global={best_score:.8f}")

        if t_pol3 > 10.0:
            a3, s3 = _optimize_max_subgrad(best_seq, t_pol3, rng, eta0=0.035,
                                           patience=1800, label="final.max")
            best_seq, best_score = _consider_candidate(best_seq, best_score, a3)
            print(f"  final.max: internal={s3:.8f} global={best_score:.8f}")

    final_seq = best_seq
    if final_seq is None:
        final_seq = scored[0][2]

    final_seq, final_ev = _safe_finalize(final_seq)
    elapsed = time.monotonic() - start
    print(f"\nFinal returned: internal={final_ev:.8f}, n={len(final_seq)}, elapsed={elapsed:.1f}s")
    return final_seq
```
