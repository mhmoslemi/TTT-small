"""
Search for a non-negative step-function sequence minimizing
    2 * n * max(conv(a, a)) / (sum(a))**2.

The code combines several ideas:
- Loads the previous best sequence if available.
- Uses FFT-based convolution and smooth log-sum-exp objectives.
- Runs low-dimensional DCT/parametric global searches to find better basins.
- Refines candidates with L-BFGS in log-odds variables and with a score-based
  mirror-descent polish.
- Explores several lengths (the given length, 9216, 8192, 4096) and a pair
  anti-palindromic ansatz.
"""

import time
import math
import numpy as np

try:
    import scipy.optimize as _sopt
except Exception:
    _sopt = None

_BASIS_CACHE = {}


def _pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _fft_square(a):
    n = int(np.size(a))
    if n <= 0:
        return np.empty(0, dtype=np.float64)
    L = 2 * n - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, n=N)
    b = np.fft.irfft(fa * fa, n=N).real[:L]
    return np.maximum(b, 0.0)


def _score_normalized(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n <= 0:
        return float("inf")
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-300:
        return float("inf")
    b = _fft_square(a)
    if b.size == 0:
        return float("inf")
    return float(2.0 * n * float(np.max(b)) / (s * s))


def _clean_sequence(obj):
    try:
        arr = np.asarray(obj, dtype=np.float64).ravel()
    except Exception:
        return None
    if arr.size <= 0:
        return None
    arr = np.nan_to_num(arr, nan=0.0, posinf=1000.0, neginf=0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(np.sum(arr))
    if not np.isfinite(s) or s <= 1e-12:
        return None
    return arr / s


def _make_default(n=9216):
    n = max(1, int(n))
    return np.full(n, 1.0 / float(n), dtype=np.float64)


def _softmax_normalized(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    n = int(z.size)
    if n <= 0:
        return np.empty(0, dtype=np.float64)
    m = float(np.max(z)) if n else 0.0
    if not np.isfinite(m):
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    e = np.exp(z - m)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 1e-300:
        e = np.full(n, 1.0, dtype=np.float64)
        s = float(n)
    return e / s


def _u_from_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size <= 1:
        return np.empty(0, dtype=np.float64)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-300:

    else:
        a = a / s
    a = np.maximum(a, 1e-15)
    a = a / float(np.sum(a))
    loga = np.log(a)
    u = loga[:-1] - loga[-1]
    return np.clip(u, -60.0, 60.0)


def _a_from_u(u):
    u = np.asarray(u, dtype=np.float64).ravel()
    u = np.clip(u, -60.0, 60.0)
    n = int(u.size) + 1
    phi = np.empty(n, dtype=np.float64)
    if n > 1:
        phi[:-1] = u
    phi[-1] = 0.0
    return _softmax_normalized(phi)


def _core_lse(a, beta):
    """
    Returns:
      h: logsumexp(beta * b), where b = n * conv(a,a) for normalized a
      p: normalized weights exp(beta*(b-bmax)) / sum
      ga: gradient wrt normalized a of h
      b: scaled convolution
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n <= 0:
        return 1e100, np.empty(0), np.empty(0), np.empty(0)

    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-300:
        return 1e100, np.empty(0), np.zeros(n), np.empty(0)
    if abs(s - 1.0) > 1e-8:
        a = a / s

    L = 2 * n - 1
    b = _fft_square(a)
    b = b * float(n)
    b = np.maximum(b, 0.0)

    if b.size == 0:
        return 1e100, np.empty(0), np.zeros(n), np.empty(0)

    bmax = float(np.max(b))
    if not np.isfinite(bmax) or bmax <= 1e-300:
        p = np.full(L, 1.0 / float(L), dtype=np.float64)
        h = 0.0
        ga = np.zeros(n, dtype=np.float64)
        return h, p, ga, b

    z = float(beta) * (b - bmax)
    e = np.exp(z)
    psum = float(np.sum(e))
    if not np.isfinite(psum) or psum <= 1e-300:
        p = np.zeros(L, dtype=np.float64)
        p[int(np.argmax(b))] = 1.0
        h = bmax
    else:
        p = e / psum
        h = bmax + math.log(psum)

    Ngrad = _pow2(L + n - 1)
    fw = np.fft.rfft(p, n=Ngrad)
    fa = np.fft.rfft(a, n=Ngrad)
    corr = np.fft.irfft(fw * np.conj(fa), n=Ngrad)[:n].real
    ga = (2.0 * float(n) * float(beta)) * corr

    if not np.all(np.isfinite(ga)):
        ga = np.zeros(n, dtype=np.float64)
    return float(h), p, ga, b


def _dct_basis(n, k):
    key = (int(n), int(k))
    if key in _BASIS_CACHE:
        return _BASIS_CACHE[key]
    n = int(n)
    k = int(k)
    B = np.zeros((n, 0), dtype=np.float64)
    if n > 0 and k > 0:
        x = (np.arange(n, dtype=np.float64) + 0.5)
        B = np.empty((n, k), dtype=np.float64)
        for j in range(k):
            mode = j + 1
            B[:, j] = np.cos(math.pi * float(mode) * x / float(n))
    _BASIS_CACHE[key] = B
    return B


def _project_to_basis(a, B):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size != B.shape[0]:
        return np.zeros(B.shape[1], dtype=np.float64)
    a = np.maximum(a, 1e-300)
    s = float(np.sum(a))
    if s <= 1e-300:
        return np.zeros(B.shape[1], dtype=np.float64)
    a = a / s
    loga = np.log(a)
    loga = loga - float(np.mean(loga))
    try:
        c, _, _, _ = np.linalg.lstsq(B, loga, rcond=None)
        c = np.asarray(c, dtype=np.float64)
        c = np.nan_to_num(c, nan=0.0, posinf=0.0, neginf=0.0)
        return np.clip(c, -100.0, 100.0)
    except Exception:
        return np.zeros(B.shape[1], dtype=np.float64)


def _dct_obj(theta, B, beta):
    if not np.all(np.isfinite(theta)):
        return 1e100, np.zeros(B.shape[1], dtype=np.float64)
    z = B.dot(theta)
    if not np.all(np.isfinite(z)):
        return 1e100, np.zeros(B.shape[1], dtype=np.float64)
    a = _softmax_normalized(z)
    h, _p, ga, _b = _core_lse(a, beta)
    if not np.isfinite(h):
        return 1e100, np.zeros(B.shape[1], dtype=np.float64)
    dot = float(np.dot(a, ga))
    gphi = a * (ga - dot)
    grad = B.T.dot(gphi)
    if not np.all(np.isfinite(grad)):
        grad = np.zeros(B.shape[1], dtype=np.float64)
    return float(h), grad


def _lse_full_obj(u, beta):
    u = np.asarray(u, dtype=np.float64).ravel()
    z = np.zeros_like(u)

    a = _a_from_u(u)

    dot = float(np.dot(a, ga))

    if gphi.size <= 1:

    else:
        gu = gphi[:-1] - gphi[-1]
    if not np.all(np.isfinite(gu)):
        gu = np.zeros(u.size, dtype=np.float64)
    return float(h), gu


def _sigmoid(z):
    z = np.asarray(z, dtype=np.float64)
    z = np.clip(z, -60.0, 60.0)
    return 1.0 / (1.0 + np.exp(-z))


def _pair_a_from_x(x, n):

    n = int(n)
    m = n // 2
    if m <= 0:
        return np.ones(n, dtype=np.float64) / float(max(1, n))

    s = float(np.sum(a))

def _resample_density(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    n_new = int(n_new)

    a = a / s
    f_old = a * float(n)
    x_old = np.linspace(0.0, 1.0, n, dtype=np.float64)
    x_new = np.arange(n_new, dtype=np.float64) / float(n_new)
    f_new = np.interp(x_new, x_old, f_old, left=f_old[0], right=f_old[-1])
    f_new = np.maximum(f_new, 0.0)
    out = f_new / float(n_new)
    s2 = float(np.sum(out))
    if not np.isfinite(s2) or s2 <= 1e-300:
        return np.full(n_new, 1.0 / float(n_new), dtype=np.float64)
    return out / s2


def _random_dct_sequence(n, k, rng, scale=1.0):
    n = int(n)
    k = max(1, int(k))
    B = _dct_basis(n, k)
    theta = rng.normal(0.0, float(scale), size=k)
    return _softmax_normalized(B.dot(theta))


def _random_pair_x(n, rng, sharp=3.0):
    n = int(n)
    m = max(1, n // 2)
    z = rng.normal(0.0, float(sharp), size=m)
    return _sigmoid(z)


def _project_pair_x(a, n):
    n = max(2, int(n))
    ar = _resample_density(a, n)
    m = n // 2
    left = ar[:m]
    right = ar[m:][::-1]
    den = left + right
    x = np.full(m, 0.5, dtype=np.float64)
    mask = den > 1e-300
    x[mask] = left[mask] / den[mask]
    return np.clip(x, 1e-8, 1.0 - 1e-8)


def _dct_search(a_start, time_limit, K_list=(4, 8, 14), beta_list=(20.0, 80.0),
                maxfun=80, rng=None, label="dct"):
    if rng is None:
        rng = np.random.default_rng(2024)
    t0 = time.monotonic()
    time_limit = max(0.0, float(time_limit))
    t_end = t0 + time_limit
    if time_limit <= 5.0 or _sopt is None:
        sc = _score_normalized(a_start)
        return a_start, sc

    best_a = _clean_sequence(a_start)
    if best_a is None:
        best_a = _make_default(9216)
    best_score = _score_normalized(best_a)

    target_n = int(best_a.size)
    ns = []
    if 800 <= target_n <= 13000:
        ns.append(target_n)
    for nn in (9216, 8192):
        if nn not in ns:
            ns.append(nn)
    if time_limit > 420.0 and 4096 not in ns:
        ns.append(4096)

    n_index = 0
    for n in ns:
        if time.monotonic() >= t_end:
            break
        portion_end = min(t_end, t0 + time_limit * float(n_index + 1) / float(len(ns)))
        n_index += 1
        a_n = _resample_density(best_a, n)

        for K in K_list:
            if time.monotonic() >= portion_end:
                break
            B = _dct_basis(n, K)
            starts = []
            theta0 = _project_to_basis(a_n, B)
            starts.append(theta0)
            for r in range(2):
                starts.append(rng.normal(0.0, 1.0 + 0.25 * float(r), size=K))

            for theta in starts:
                theta = np.asarray(theta, dtype=np.float64).copy()
                theta = np.clip(theta, -80.0, 80.0)
                for beta in beta_list:
                    if time.monotonic() >= portion_end:
                        break
                    options = {
                        "maxiter": 90,
                        "maxfun": int(maxfun),
                        "maxls": 20,
                        "ftol": 1e-11,
                        "gtol": 1e-6,
                        "maxcor": 12
                    }
                    try:
                        res = _sopt.minimize(_dct_obj, theta, args=(B, float(beta)),
                                             method="L-BFGS-B", jac=True, options=options)
                    except Exception:
                        res = None
                    if res is not None and hasattr(res, "x"):
                        theta_new = np.asarray(res.x, dtype=np.float64)
                        if theta_new.size == theta.size and np.all(np.isfinite(theta_new)):
                            theta = np.clip(theta_new, -80.0, 80.0)
                    theta = np.clip(theta, -80.0, 80.0)
                    a_new = _softmax_normalized(B.dot(theta))
                    sc = _score_normalized(a_new)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_a = a_new.copy()
                        print(f"[{label}] n={n} K={K} beta={beta:g} score={sc:.8f}")

    print(f"[{label}] finished score={best_score:.8f}")
    return best_a, best_score


def _simple_lse_optimize(a, time_limit, rng=None, beta_start=100.0,
                         beta_max=3000.0, label="simple"):
    if rng is None:
        rng = np.random.default_rng(7)
    t0 = time.monotonic()
    time_limit = max(0.0, float(time_limit))
    t_end = t0 + time_limit
    a = _clean_sequence(a)
    if a is None:
        a = _make_default(9216)
    best_score = _score_normalized(a)
    best_a = a.copy()
    if time_limit <= 5.0:
        return best_a, best_score

    u = _u_from_a(a)
    u_best = u.copy()
    if u.size == 0:
        return best_a, best_score

    step = 0.30
    no_accept = 0
    beta = float(beta_start)
    it = 0

    while time.monotonic() < t_end - 2.0:
        a_cur = _a_from_u(u)
        _h, _p, ga, b = _core_lse(a_cur, beta)
        score = float(2.0 * float(np.max(b))) if b.size else float("inf")
        if not np.isfinite(score):
            break

        if score < best_score - 1e-13:
            best_score = score
            best_a = a_cur.copy()
            u_best = u.copy()

        if u.size:
            dot = float(np.dot(a_cur, ga))
            gphi = a_cur * (ga - dot)
            gu = gphi[:-1] - gphi[-1]
            maxg = float(np.max(np.abs(gu))) if gu.size else 0.0
        else:
            gu = np.empty(0, dtype=np.float64)
            maxg = 0.0

        if not np.isfinite(maxg) or maxg < 1e-16:
            break

        d = -gu / maxg
        accepted = False
        accepted_u = None
        for trial in range(9):
            st = step * (0.5 ** trial)
            u_try = u + st * d
            u_try = np.clip(u_try, -60.0, 60.0)
            a_try = _a_from_u(u_try)
            st_score = _score_normalized(a_try)
            if np.isfinite(st_score) and (
                    st_score < best_score - 1e-13 or st_score < score - 1e-11):
                accepted = True
                accepted_u = u_try
                break

        if accepted:
            u = accepted_u
            step = min(step * 1.15, 2.0)
            no_accept = 0
        else:
            step *= 0.65
            no_accept += 1
            if no_accept > 25:
                u = np.clip(u + rng.normal(0.0, 0.05, size=u.size), -60.0, 60.0)
                step = 0.30
                no_accept = 0

        beta = min(beta * 1.035, float(beta_max))
        it += 1
        if label and it % 300 == 0:
            print(f"[{label}] simple it={it} score={score:.8f} best={best_score:.8f}")

    return best_a, best_score


def _refine_full_lse(a, time_limit, rng=None, beta_start=20.0, beta_max=2500.0,
                     maxfun=80, label="full"):
    if rng is None:
        rng = np.random.default_rng(424242)
    t0 = time.monotonic()
    time_limit = max(0.0, float(time_limit))
    t_end = t0 + time_limit
    a = _clean_sequence(a)
    if a is None:
        a = _make_default(9216)
    best_score = _score_normalized(a)
    best_a = a.copy()
    if time_limit <= 5.0:
        return best_a, best_score
    if _sopt is None:
        return _simple_lse_optimize(a, time_limit, rng=rng, label=label)

    u = _u_from_a(a)
    u_best = u.copy()
    beta = float(beta_start)
    no_imp = 0
    sub = 0
    if u.size == 0:
        return best_a, best_score

    while time.monotonic() < t_end - 3.0:
        sub += 1
        options = {
            "maxiter": 90,
            "maxfun": int(maxfun),
            "maxls": 20,
            "ftol": 1e-12,
            "gtol": 1e-9,
            "maxcor": 10
        }
        try:
            res = _sopt.minimize(_lse_full_obj, u, args=(float(beta),),
                                 method="L-BFGS-B", jac=True, options=options)
        except Exception:
            res = None

        u_new = None
        if res is not None and hasattr(res, "x"):
            cand = np.asarray(res.x, dtype=np.float64)
            if cand.size == u.size and np.all(np.isfinite(cand)):
                u_new = np.clip(cand, -60.0, 60.0)

        if u_new is not None:
            a_new = _a_from_u(u_new)
            sc = _score_normalized(a_new)
            if np.isfinite(sc):
                if sc < best_score - 1e-13:
                    best_score = sc
                    best_a = a_new.copy()
                    u_best = u_new.copy()
                    no_imp = 0
                    if label and sub % 50 == 0:
                        print(f"[{label}] full improved score={sc:.8f} t={time.monotonic()-t0:.1f}s")
                else:
                    no_imp += 1
                if sc <= best_score * 1.015:
                    u = u_new
                else:
                    u = u_best.copy()
            else:
                no_imp += 1
                u = u_best.copy()
        else:
            no_imp += 1
            u = u_best.copy()

        if no_imp >= 4:
            if beta < float(beta_max):
                beta = min(beta * 3.0, float(beta_max))
            else:
                u = np.clip(u_best + rng.normal(0.0, 0.04, size=u_best.size), -60.0, 60.0)
            no_imp = 0

    return best_a, best_score


def _refine_pair_lse(a, time_limit, n=9216, rng=None, beta_start=20.0,
                     beta_max=2500.0, maxfun=80, label="pair"):
    if rng is None:
        rng = np.random.default_rng(11)
    t0 = time.monotonic()
    time_limit = max(0.0, float(time_limit))
    t_end = t0 + time_limit
    a = _clean_sequence(a)
    if a is None:
        a = _make_default(n)
    x = _project_pair_x(a, n)
    a = _pair_a_from_x(x, n)
    best_score = _score_normalized(a)
    best_a = a.copy()
    if time_limit <= 5.0 or _sopt is None:
        return best_a, best_score

    y = _sigmoid(x)
    y = np.where(x >= 0.5, np.log(x) - np.log(1.0 - x), np.log(x) - np.log(1.0 - x))
    y = np.clip(y, -50.0, 50.0)
    y_best = y.copy()
    beta = float(beta_start)
    no_imp = 0
    sub = 0
    if y.size == 0:
        return best_a, best_score

    while time.monotonic() < t_end - 3.0:
        sub += 1
        options = {
            "maxiter": 90,
            "maxfun": int(maxfun),
            "maxls": 20,
            "ftol": 1e-11,
            "gtol": 1e-8,
            "maxcor": 10
        }
        try:
            res = _sopt.minimize(_pair_y_obj, y, args=(float(beta),),
                                 method="L-BFGS-B", jac=True, options=options)
        except Exception:
            res = None

        y_new = None
        if res is not None and hasattr(res, "x"):
            cand = np.asarray(res.x, dtype=np.float64)
            if cand.size == y.size and np.all(np.isfinite(cand)):
                y_new = np.clip(cand, -50.0, 50.0)

        if y_new is not None:
            x_new = _sigmoid(y_new)
            a_new = _pair_a_from_x(x_new, n)
            sc = _score_normalized(a_new)
            if np.isfinite(sc):
                if sc < best_score - 1e-13:
                    best_score = sc
                    best_a = a_new.copy()
                    y_best = y_new.copy()
                    no_imp = 0
                    if label and sub % 50 == 0:
                        print(f"[{label}] pair improved score={sc:.8f} t={time.monotonic()-t0:.1f}s")
                else:
                    no_imp += 1
                if sc <= best_score * 1.015:
                    y = y_new
                else:
                    y = y_best.copy()
            else:
                no_imp += 1
                y = y_best.copy()
        else:
            no_imp += 1
            y = y_best.copy()

        if no_imp >= 4:
            if beta < float(beta_max):
                beta = min(beta * 3.0, float(beta_max))
            else:
                y = np.clip(y_best + rng.normal(0.0, 0.05, size=y_best.size), -50.0, 50.0)
            no_imp = 0

    return best_a, best_score


def _append_candidate(pool, pid, a, label, score):
    a = _clean_sequence(a)
    if a is None:
        return None
    score = float(score)
    if not np.isfinite(score):
        return None
    pool.append([score, int(pid), a, label])
    return int(pid)


def _pick_unused(pool, used_pids, max_factor=1.20):
    if not pool:
        return None
    pool = sorted(pool, key=None)
    base = pool[0][0]
    limit = base * float(max_factor)
    for item in pool:
        if item[0] <= limit and item[1] not in used_pids:
            return item
    return None


def _safe_evaluate(seq):
    ev = globals().get("evaluate_sequence")
    if ev is None or not callable(ev):
        return float("inf")
    try:
        return float(ev(list(seq)))
    except Exception:
        return float("inf")


def _load_height_sequence():
    g = globals().get("height_sequence_1")
    if g is None:
        return None
    return _clean_sequence(g)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Returns a valid non-negative list of heights.
    """
    t_start = time.monotonic()
    total = max(120.0, float(budget_s))
    t_end = t_start + total - 90.0
    rng = np.random.default_rng(int(seed))

    a_given = _load_height_sequence()
    if a_given is None:
        a_best = _make_default(9216)
        base_label = "default"
    else:
        if a_given.size > 15000:
            a_given = _resample_density(a_given, 9216)
        a_best = a_given
        base_label = "given"
    if a_best.size > 20000:
        a_best = _resample_density(a_best, 9216)
    best_score = _score_normalized(a_best)

    print(f"initial {base_label} n={a_best.size} score={best_score:.10f}")

    pool = []
    pid = 0
    best_pid = _append_candidate(pool, pid, a_best, base_label, best_score)
    pid += 1

    # Add resampled and random/structured candidates.
    for nn in (9216, 8192, 4096):
        if nn == a_best.size:
            continue
        cand = _resample_density(a_best, nn)
        if cand is not None:
            p = _append_candidate(pool, pid, cand, f"resample{nn}", _score_normalized(cand))
            if p is not None:
                pid = p + 1

    for r in range(2):
        cand = _random_dct_sequence(9216, 14, rng, scale=1.0 + 0.4 * float(r))
        p = _append_candidate(pool, pid, cand, f"dctrand{r}", _score_normalized(cand))
        if p is not None:
            pid = p + 1

    for r in range(2):
        x = _random_pair_x(9216, rng, sharp=3.0 + 2.0 * float(r))
        cand = _pair_a_from_x(x, 9216)
        p = _append_candidate(pool, pid, cand, f"pairrand{r}", _score_normalized(cand))
        if p is not None:
            pid = p + 1

    pool.sort()
    for item in pool:
        if item[0] < best_score:
            best_score = item[0]
            best_a = item[2].copy()
            best_pid = item[1]

    # Low-dimensional DCT global/local search.
    remaining = t_end - time.monotonic()
    dct_time = min(380.0, max(0.0, 0.28 * remaining))
    if dct_time > 60.0 and _sopt is not None:
        print(f"starting DCT search time={dct_time:.1f}s")
        a_dct, sc_dct = _dct_search(
            a_best, dct_time,
            K_list=(4, 8, 14),
            beta_list=(20.0, 80.0),
            maxfun=80,
            rng=rng,
            label="dct"
        )
        p = _append_candidate(pool, pid, a_dct, "dct_search", sc_dct)
        if p is not None:
            pid = p + 1
        if np.isfinite(sc_dct) and sc_dct < best_score:
            best_score = sc_dct
            best_a = a_dct.copy()
            best_pid = p if p is not None else best_pid
            print(f"DCT search improved to {best_score:.10f}")

    # Main refinement phases.
    pool.sort()
    used_pids = set()
    if best_pid is not None:
        used_pids.add(best_pid)

    remaining = t_end - time.monotonic()
    tasks = [
        ("full_best", 0.34, 950.0),
        ("full_second", 0.24, 650.0),
        ("simple_best", 0.24, 700.0),
        ("pair", 0.10, 300.0),
        ("final_simple", 0.08, 350.0)
    ]

    for task_name, frac, max_alloc in tasks:
        now = time.monotonic()
        if now >= t_end - 30.0:
            break
        rem = t_end - now
        alloc = min(float(max_alloc), float(frac) * rem)
        if alloc < 45.0:
            break

        if task_name == "full_best":
            a_in = best_a
            a_out, sc_out = _refine_full_lse(a_in, alloc, rng=rng,
                                             beta_start=20.0, beta_max=2500.0,
                                             maxfun=90, label=f"full_best_n{a_in.size}")
            p = _append_candidate(pool, pid, a_out, "full_best_out", sc_out)
            if p is not None:
                pid = p + 1
                used_pids.add(p)
            if np.isfinite(sc_out) and sc_out < best_score:
                best_score = sc_out
                best_a = a_out.copy()
                best_pid = p

        elif task_name == "full_second":
            pool.sort()
            cand = _pick_unused(pool, used_pids, max_factor=1.30)
            if cand is None:
                break
            sc_in, a_in, pid_in = cand[0], cand[2], cand[1]
            used_pids.add(pid_in)
            a_out, sc_out = _refine_full_lse(a_in, alloc, rng=rng,
                                             beta_start=30.0, beta_max=2500.0,
                                             maxfun=80, label=f"full_second_n{a_in.size}")
            p = _append_candidate(pool, pid, a_out, "full_second_out", sc_out)
            if p is not None:
                pid = p + 1
                used_pids.add(p)
            if np.isfinite(sc_out) and sc_out < best_score:
                best_score = sc_out
                best_a = a_out.copy()
                best_pid = p

        elif task_name == "simple_best":
            a_out, sc_out = _simple_lse_optimize(best_a, alloc, rng=rng,
                                                 beta_start=120.0, beta_max=5000.0,
                                                 label=f"simple_best_n{best_a.size}")
            p = _append_candidate(pool, pid, a_out, "simple_best_out", sc_out)
            if p is not None:
                pid = p + 1
                used_pids.add(p)
            if np.isfinite(sc_out) and sc_out < best_score:
                best_score = sc_out
                best_a = a_out.copy()
                best_pid = p

        elif task_name == "pair":
            pair_n = 9216
            if best_a.size % 2 == 0 and 800 <= best_a.size <= 13000:
                pair_n = int(best_a.size)
            a_out, sc_out = _refine_pair_lse(best_a, alloc, n=pair_n, rng=rng,
                                             beta_start=30.0, beta_max=3000.0,
                                             maxfun=80, label=f"pair_n{pair_n}")
            p = _append_candidate(pool, pid, a_out, "pair_out", sc_out)
            if p is not None:
                pid = p + 1
                used_pids.add(p)
            if np.isfinite(sc_out) and sc_out < best_score:
                best_score = sc_out
                best_a = a_out.copy()
                best_pid = p

        elif task_name == "final_simple":
            a_out, sc_out = _simple_lse_optimize(best_a, alloc, rng=rng,
                                                 beta_start=800.0, beta_max=8000.0,
                                                 label=f"final_simple_n{best_a.size}")
            p = _append_candidate(pool, pid, a_out, "final_simple_out", sc_out)
            if p is not None:
                pid = p + 1
                used_pids.add(p)
            if np.isfinite(sc_out) and sc_out < best_score:
                best_score = sc_out
                best_a = a_out.copy()
                best_pid = p

    # Final sanity normalization.
    best_a = _clean_sequence(best_a)
    if best_a is None:
        best_a = _make_default(1024)
        best_score = _score_normalized(best_a)

    best_a = np.nan_to_num(best_a, nan=0.0, posinf=0.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    s = float(np.sum(best_a))
    if not np.isfinite(s) or s <= 0.01:
        best_a = _make_default(1024)
        s = 1.0
    else:
        best_a = best_a / s

    final_seq = [float(x) for x in best_a]
    print(f"final internal score={_score_normalized(best_a):.12f} n={len(final_seq)}")

    if len(final_seq) <= 6000 and time.monotonic() < t_end:
        exact = _safe_evaluate(final_seq)
        if np.isfinite(exact):
            print(f"final exact score={exact:.12f}")

    return final_seq
