    fa = np.fft.rfft(a, size)

    Returns the optimal g.

    n = len(a)

    cols = []
    vals = []

            rows.append(np.full(len(idx), k))
            cols.append(idx)
            vals.append(a[k - idx])
    if not rows:
        return sparse.csr_matrix((2 * n - 1, n))
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    vals = np.concatenate(vals)
    return sparse.coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()


def solve_lp_for_g(A, M):
    """Solve: max sum(g) s.t. A g <= M*1, g >= 0. Returns g or None."""
    n_v = A.shape[1]
    c_obj = -np.ones(n_v)
    b_vec = np.full(A.shape[0], M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_vec,
                      bounds=[(0, None)] * n_v, method='highs',
                      options={'time_limit': 10})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def compute_subgradient(a, c, peak_idx):
    """Compute subgradient of max(c) w.r.t. a for given peak indices.
    d(c[k])/da[j] = 2*a[k-j] if 0<=k-j<n, else 0.
    """
    n = len(a)
    subgrad = np.zeros(n)
    for k in peak_idx:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_max >= j_min:
            # subgrad[j] += 2*a[k-j] for j in [j_min, j_max]
            # k-j ranges from k-j_min down to k-j_max
            subgrad[j_min:j_max + 1] += 2.0 * a[k - j_max:k - j_min + 1][::-1]
    subgrad /= len(peak_idx)
    return subgrad


def gradient_descent_phase(a, n, best_score, best_a, deadline, rng, lr_init=0.1):
    """Gradient descent with momentum and adaptive learning rate."""
    momentum = np.zeros(n)
    lr = lr_init
    beta = 0.85
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        a = normalize_sum(a, n)
        c = fft_self_conv(a)
        M = c.max()

        # Find near-peak lags
        thresh = M * 0.985
        peak_mask = c >= thresh
        peak_idx = np.where(peak_mask)[0]
        if len(peak_idx) == 0:
            peak_idx = np.array([int(np.argmax(c))])
        else:
            # Take top 5 distinct peaks (at least 3 apart)
            sorted_idx = peak_idx[np.argsort(-c[peak_idx])]
            selected = []
            for idx in sorted_idx:
                if all(abs(idx - s) > 3 for s in selected):
                    selected.append(int(idx))
                if len(selected) >= 5:
                    break
            peak_idx = np.array(selected) if selected else np.array([int(np.argmax(c))])

        # Compute subgradient
        subgrad = compute_subgradient(a, c, peak_idx)

        # Full gradient of score = 2n*M/S^2 with S=n: (2/n)*subgrad
        # Project to maintain sum(a) = n: subtract mean
        grad = (2.0 / n) * subgrad
        grad -= grad.mean()

        # Momentum update
        momentum = beta * momentum - lr * grad
        a = a + momentum
        a = np.maximum(a, 0.0)
        a = normalize_sum(a, n)

        it += 1

        if it % 25 == 0:
            sc = float(evaluate_sequence(a.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
                if it % 250 == 0:
                    print(f"    GD it={it}: score={sc:.8f}, lr={lr:.6f}")
            else:
                no_improve += 1

            if no_improve >= 15:
                lr *= 0.75
                momentum *= 0.4
                no_improve = 0
                if lr < 1e-8:
                    a = best_a * (1 + 0.03 * rng.normal(0, 1, n))
                    a = normalize_sum(a, n)
                    lr = 0.05 * lr_init
                    momentum = np.zeros(n)

    return a, best_score, best_a


def propose_candidate(seed=42, budget_s=2350):
    """
    Search for low-score sequence using LP-based iteration and gradient descent.
    """
    rng = np.random.default_rng(seed)

    # Get parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        parent = np.ones(512, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    if parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(512, dtype=np.float64)

    n = len(parent)
    S_parent = parent.sum()
    if S_parent < 0.01:
        parent = np.ones(n)
        S_parent = n

    # Phase 0: Quick initialization
    a = parent / S_parent * n
    best_score = float(evaluate_sequence(a.tolist()))
    best_a = a.copy()
    print(f"Initial score: {best_score:.8f}, n={n}")

    # Try a few parametric families for better init
    t_arr = np.arange(n, dtype=np.float64)
    x = 2.0 * t_arr / max(1, n - 1) - 1.0
    families = []
    for alpha in [0.2, 0.4, 0.6, 0.8]:
        families.append(1.0 + alpha * np.cos(2.0 * np.pi * t_arr / n))
        families.append(1.0 + alpha * np.cos(4.0 * np.pi * t_arr / n + 0.3))
        families.append(1.0 - alpha * x ** 2)
        families.append(1.0 + alpha * np.sin(np.pi * x))
        families.append(1.0 + alpha * (x ** 3 - x))

    for fam in families:
        a_try = np.maximum(fam, 0.0)
        a_try = normalize_sum(a_try, n)
        sc = fast_score(a_try)
        if sc < best_score - 0.01:
            exact_sc = float(evaluate_sequence(a_try.tolist()))
            if np.isfinite(exact_sc) and exact_sc < best_score:
                best_score = exact_sc
                best_a = a_try.copy()
                print(f"  Better init: {sc:.8f}")

    a = best_a.copy()
    print(f"After init sweep: {best_score:.8f}")

    deadline = time.monotonic() + budget_s

    # Phase 1: LP-based iteration
    print("Phase 1: LP-based iteration")
    lp_iter = 0
    no_improve_lp = 0
    A_cache = None
    last_a_sum = -1

    while time.monotonic() < deadline - 300:
        a = normalize_sum(a, n)
        S = a.sum()
        c = fft_self_conv(a)
        M = c.max()
        current_score = 2.0 * n * M / (S * S)

        # Rebuild constraint matrix periodically
        current_sum = abs(float(a[0] + a[n // 2] + a[-1] + a[n // 3]))
        if A_cache is None or abs(current_sum - last_a_sum) > 0.001:
            A_cache = build_conv_matrix(a)
            last_a_sum = current_sum

        # Solve LP
        g = solve_lp_for_g(A_cache, M)

        if g is None or g.sum() < 1e-8:
            no_improve_lp += 3
            if no_improve_lp > 80:
                print(f"  LP phase stopping after {lp_iter} iters, best={best_score:.8f}")
                break
            # Small perturbation
            a = a * (1 + 0.02 * rng.normal(0, 1, n))
            a = normalize_sum(a, n)
            continue

        # Rescale g
        g = g * n / g.sum()

        # Line search
        best_t = 0.1
        best_sc = current_score
        for t_val in [0.02, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.5]:
            a_try = (1.0 - t_val) * a + t_val * g
            a_try = np.maximum(a_try, 0.0)
            a_try = normalize_sum(a_try, n)
            sc_try = fast_score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_t = t_val

        if best_sc < current_score - 1e-10:
            no_improve_lp = 0
        else:
            no_improve_lp += 1
            if no_improve_lp > 40:
                # Perturb to escape
                a = a * (1 + 0.04 * rng.normal(0, 1, n))
                a = normalize_sum(a, n)
                no_improve_lp = 0
                continue

        # Update
        a = (1.0 - best_t) * a + best_t * g
        a = np.maximum(a, 0.0)
        a = normalize_sum(a, n)

        lp_iter += 1

        if lp_iter % 15 == 0:
            sc = float(evaluate_sequence(a.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"  LP it={lp_iter}: score={sc:.8f}, t={best_t:.3f}")

    print(f"  LP done: {lp_iter} iters, best={best_score:.8f}")

    # Phase 2: Gradient descent fine-tuning from best_a
    print("Phase 2: Gradient descent fine-tuning")
    deadline_gd = deadline - 20
    a, best_score, best_a = gradient_descent_phase(
        best_a.copy(), n, best_score, best_a.copy(), deadline_gd, rng, lr_init=0.08
    )

    # Phase 3: Very fine tuning
    print("Phase 3: Ultra-fine tuning")
    deadline_fin = deadline - 5
    a, best_score, best_a = gradient_descent_phase(
        best_a.copy(), n, best_score, best_a.copy(), deadline_fin, rng, lr_init=0.005
    )

    # Final validation
    final_a = normalize_sum(best_a, n)
    final_score = float(evaluate_sequence(final_a.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
        best_a = final_a

    print(f"Final score: {best_score:.8f}")
    return [float(x) for x in best_a]
