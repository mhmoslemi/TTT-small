"""
Smooth gradient descent with continuation on power-p approximation of max-convolution,
combined with coordinate descent polishing, to minimize 2*n*max(conv(a,a))/(sum a)^2.
Uses FFT for efficient gradient computation and multiple restart strategies.
"""

import numpy as np
import time


def _norm1(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    np.clip(a, 0.0, None, out=a)
    s = float(a.sum())
    if s <= 1e-30:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def _score_of(a):
    a = _norm1(a)
    n = a.size
    c = np.convolve(a, a)
    return float(2.0 * n * np.max(c))


def _next_pow2(m):
    k = 1
    while k < m:
        k <<= 1
    return k


def _fft_conv(a, b):
    n = a.size + b.size - 1
    N = _next_pow2(n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n]


def _fft_cross(s, a):
    """Compute X[i] = sum_{j=0}^{n-1} s[i+j] * a[j] for i=0..n-1.
    s has length 2n-1, a has length n.
    """
    n = a.size
    a_rev = a[::-1]
    # conv(a_rev, s)[i + n - 1] = sum_j a_rev[j] * s[i+n-1-j]
    # = sum_j a[n-1-j] * s[i+n-1-j]
    # Let k = n-1-j: = sum_k a[k] * s[i+k]
    # Yes! That's what we want.
    N = _next_pow2(n + s.size - 1)
    fa = np.fft.rfft(a_rev, N)
    fs = np.fft.rfft(s, N)
    conv = np.fft.irfft(fa * fs, N)
    return conv[n - 1: n - 1 + n]


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    sorted_v = np.sort(v)[::-1]
    cumsum = np.cumsum(sorted_v)
    # Find largest k such that sorted_v[k-1] > (cumsum[k-1] - 1) / k
    ks = np.arange(1, n + 1, dtype=np.float64)
    vals = (cumsum - 1.0) / ks
    # Condition: sorted_v[k-1] > vals[k-1]
    valid = sorted_v > vals
    # Find largest valid k
    valid_idx = np.where(valid)[0]
    if valid_idx.size == 0:
        return np.ones(n) / n
    k = int(valid_idx[-1]) + 1
    tau = (cumsum[k - 1] - 1.0) / k
    result = np.maximum(v - tau, 0.0)
    s = result.sum()
    if s > 0:
        result /= s
    else:
        result = np.ones(n) / n
    return result


def _smooth_grad(a, p):
    """Compute gradient of sum_k c_k^p where c = convolve(a,a)."""
    n = a.size
    c = np.convolve(a, a)
    c = np.maximum(c, 0.0)
    if p == 1:
        s = np.ones_like(c)
    else:
        cmax = c.max()
        if cmax <= 0:
            return np.zeros(n)
        s = c ** (p - 1)
        if p > 2:
            smax = s.max()
            if smax > 0:
                s = s / smax
    X = _fft_cross(s, a)
    G = 2.0 * X
    # Subtract diagonal correction: a_i * s_{2i} for 2i < 2n-1
    max_2i = min(2 * n - 2, 2 * (n - 1))
    G -= a * s[:max_2i + 1]
    return G * p


def _smooth_obj(a, p):
    c = np.convolve(a, a)
    c = np.maximum(c, 0.0)
    return float(np.sum(c ** p))

        lr = min(lr, 0.1)

            G = _smooth_grad(a, p)
            Gnorm = np.linalg.norm(G)
            if Gnorm < 1e-15:
                break

            G_dir = -G / Gnorm
            # Adaptive step
            alpha = lr
            # Line search (a few steps)
            for _ in range(8):
                a_try = _project_simplex(a + alpha * G_dir * (Gnorm / max(np.max(a), 1e-10)))
                c_try = np.convolve(a_try, a_try)
                c_try = np.maximum(c_try, 0.0)
                obj_try = float(np.sum(c_try ** p))
                obj_cur = float(np.sum(np.maximum(np.convolve(a, a), 0) ** p))
                if obj_try < obj_cur * 0.999:
                    break
                alpha *= 0.5
                if alpha < 1e-12:
                    break

            a_new = _project_simplex(a + alpha * G_dir * (Gnorm / max(np.max(a), 1e-10)))
            a = a_new

            # Track actual score
            if it % 50 == 0 or it == iters_in_phase - 1:
                sc = _score_of(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

            # Decay learning rate within phase
            decay = 1.0 - 0.3 * it / max(iters_in_phase, 1)
            lr = lr_init * decay / max(Gnorm / (10.0 * np.max(a) + 1e-10), 1.0)

    sc = _score_of(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def _coord_update(a, c, score, i, grid, refine=5):
    """Single coordinate update (similar to existing code)."""
    n = a.size
    old = a[i]
    D = 1.0 - old

    ks = slice(i, i + n)
    c_k = c[ks].copy()

    coeff = 2.0 * a.copy()
    const = -2.0 * old * a.copy()
    mid = i
    if 0 <= mid < n:
        coeff[mid] = 0.0
        const[mid] = -old * old

    max_unaff = 0.0
    if i > 0:
        m = float(np.max(c[:i]))
        if m > max_unaff:
            max_unaff = m
    end = i + n
    if end < 2 * n - 1:
        m = float(np.max(c[end:]))
        if m > max_unaff:
            max_unaff = m

    xs = np.concatenate(([0.0], grid, [old]))
    xs = np.unique(xs[np.isfinite(xs) & (xs >= 0) & (xs <= 1e4)])
    if xs.size == 0:
        return a, c, score, False

    aff = c_k[None, :] + coeff[None, :] * xs[:, None] + const[None, :]
    aff[:, mid] += xs * xs
    max_aff = np.max(aff, axis=1)
    maxc = np.maximum(max_unaff, max_aff)
    S = D + xs
    vals = np.full(xs.size, float('inf'))
    valid = S > 1e-15
    vals[valid] = (2.0 * n) * maxc[valid] / (S[valid] ** 2)
    idx = int(np.argmin(vals))
    best_x = float(xs[idx])
    best_val = float(vals[idx])
    tol = 1e-12 * max(1.0, score)

    if refine > 0 and xs.size > 1:
        left = 0.0 if idx == 0 else float(xs[idx - 1])
        right = float(xs[idx]) * 10.0 if idx == xs.size - 1 else float(xs[idx + 1])
        if best_x > 1e-8:
            local = best_x * np.logspace(-1.0, 1.0, refine)
        else:
            local = np.logspace(np.log10(max(left, 1e-12)), np.log10(max(right, 1e-6)), refine)
        local = np.unique(np.concatenate((local, [left, best_x, right])))
        local = local[np.isfinite(local) & (local >= 0) & (local <= 1e4)]
        if local.size > 0:
            aff2 = c_k[None, :] + coeff[None, :] * local[:, None] + const[None, :]
            aff2[:, mid] += local * local
            max_aff2 = np.max(aff2, axis=1)
            maxc2 = np.maximum(max_unaff, max_aff2)
            S2 = D + local
            vals2 = np.full(local.size, float('inf'))
            valid2 = S2 > 1e-15
            vals2[valid2] = (2.0 * n) * maxc2[valid2] / (S2[valid2] ** 2)
            idx2 = int(np.argmin(vals2))
            if vals2[idx2] < best_val:
                best_x = float(local[idx2])
                best_val = float(vals2[idx2])

    if not np.isfinite(best_val) or best_val >= score - tol:
        return a, c, score, False

    S = D + best_x
    S2 = S * S
    c_new = c / S2
    aff_best = c_k + coeff * best_x + const
    aff_best[mid] += best_x * best_x
    c_new[ks] = aff_best / S2
    a_new = a / S
    a_new[i] = best_x / S
    a_new = np.clip(a_new, 0, None)
    c_new = np.clip(c_new, 0, None)

    snew = a_new.sum()
    if snew <= 0:
        return a, c, score, False
    a_new = a_new / snew
    c_new = c_new / (snew * snew)
    score_new = float(2.0 * n * np.max(c_new))
    if score_new >= score - tol:
        return a, c, score, False
    return a_new, c_new, score_new, True


def _coord_descent(a, deadline, rng, grid_size=20, refine=5, max_perturb=3):
    a = _norm1(a.copy())
    n = a.size
    c = np.convolve(a, a)
    c = np.clip(c, 0, None)
    score = _score_of(a)
    best_a = a.copy()
    best_score = score

    grid = np.unique(np.concatenate(([0.0], np.logspace(-6, 2, grid_size))))
    stagnant = 0
    perts = 0
    noise_scales = [0.4, 0.2, 0.1, 0.05]

    while time.monotonic() < deadline - 0.5:
        order = rng.permutation(n)
        improved = False
        for i in order:
            if time.monotonic() > deadline - 0.3:
                break
            a, c, score, imp = _coord_update(a, c, score, int(i), grid, refine)
            if imp:
                improved = True

        a = _norm1(a)
        c = np.convolve(a, a)
        c = np.clip(c, 0, None)
        score = float(2.0 * n * np.max(c))

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        if not improved:
            if stagnant >= 2 and perts < max_perturb:
                scale = noise_scales[min(perts, len(noise_scales) - 1)]
                perturbed = best_a * np.exp(rng.normal(0, scale, n))
                perturbed = np.clip(perturbed, 0, None)
                perturbed = _norm1(perturbed)
                c_p = np.convolve(perturbed, perturbed)
                sc_p = _score_of(perturbed)
                if sc_p < best_score:
                    best_score = sc_p
                    best_a = perturbed
                    a = perturbed
                    c = c_p
                    score = sc_p
                perts += 1
                stagnant = 0
            elif stagnant >= 6:
                break
        else:
            stagnant = 0

    return best_a, best_score


def _make_initial(n, strategy, rng):
    n = int(n)
    if n <= 1:
        return np.ones(max(1, n))
    st = int(strategy) % 16
    if st == 0:
        p = np.ones(n)
    elif st == 1:
        p = rng.dirichlet(np.full(n, 1.0))
    elif st == 2:
        p = rng.dirichlet(np.full(n, 0.3))
    elif st == 3:
        p = np.exp(rng.normal(0, 0.8, n))
    elif st == 4:
        # Linear ramp
        x = np.arange(n, dtype=np.float64)
        p = (x + 1) * (1 + 0.1 * rng.random(n))
    elif st == 5:
        # Reverse linear ramp
        x = np.arange(n, dtype=np.float64)
        p = (n - x) * (1 + 0.1 * rng.random(n))
    elif st == 6:
        # Exponential ramp
        x = np.arange(n, dtype=np.float64)
        rate = rng.uniform(0.3, 1.5)
        p = np.exp(rate * x / n) * (1 + 0.05 * rng.random(n))
    elif st == 7:
        # Sinusoidal
        x = np.linspace(0, np.pi, n)
        p = np.sin(x) ** rng.uniform(0.5, 3.0) + 0.01
    elif st == 8:
        # Double ramp (peaked in middle)
        x = np.abs(np.arange(n) - n / 2.0) / (n / 2.0)
        p = (1 - x) ** rng.uniform(0.5, 3.0) + 0.01
    elif st == 9:
        # Two-level with noise
        half = n // 2
        h = rng.uniform(0.8, 2.0)
        p = np.ones(n)
        p[:half] = h
        p *= (1 + 0.1 * rng.random(n))
    elif st == 10:
        # Periodic modulation
        freq = rng.uniform(1, 5)
        x = np.arange(n, dtype=np.float64)
        p = 1.0 + 0.5 * np.sin(2 * np.pi * freq * x / n)
        p = np.clip(p, 0.01, None)
    elif st == 11:
        # Paired antisymmetric
        m = (n + 1) // 2
        w = rng.dirichlet(np.full(m, 0.5))
        p = np.zeros(n)
        for r in range(m):
            a_idx = r
            b_idx = n - 1 - r
            if a_idx == b_idx:
                p[a_idx] = w[r]
            else:
                if rng.random() < 0.8:
                    p[a_idx] = w[r]
                else:
                    p[b_idx] = w[r]
    elif st == 12:
        # Smooth random via cubic spline interpolation of few points
        npts = max(4, n // 50)
        pts = np.sort(rng.random(npts))
        vals = np.exp(rng.normal(0, 0.5, npts))
        x = np.linspace(0, 1, n)
        p = np.interp(x, pts, vals)
    elif st == 13:
        # Quadratic ramp
        x = np.linspace(0, 1, n)
        p = (a_ := rng.uniform(0.5, 2.0)) * x ** rng.uniform(1, 4) + 0.01
    elif st == 14:
        # Cosine window squared
        x = np.linspace(0, 1, n)
        p = np.cos(np.pi * x / 2) ** rng.uniform(1, 4) + 0.01
    else:
        p = 1 + 0.5 * rng.random(n)
    return _norm1(p)


def _expand(proto, target_len):
    target_len = int(target_len)
    p = _norm1(proto)
    R = p.size
    if R == target_len:
        return p.copy()
    if R > target_len:
        out = np.empty(target_len)
        for j in range(target_len):
            lo = (j * R) // target_len
            hi = max(lo + 1, (j + 1) * R) // target_len
            out[j] = p[lo:hi].mean()
        return _norm1(out)
    base = target_len // R
    rem = target_len % R
    lengths = np.full(R, base, dtype=np.int64)
    lengths[:rem] += 1
    out = np.zeros(target_len)
    idx = 0
    for i in range(R):
        L = int(lengths[i])
        if L > 0:
            out[idx:idx + L] = p[i] / L
            idx += L
    return _norm1(out)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Return best non-negative coefficient sequence minimizing the evaluation score."""
    budget = max(1.0, float(budget_s))
    deadline = time.monotonic() + max(0.0, budget - 5.0)
    rng = np.random.default_rng(int(seed))
    final_len = 3809

    best_a = None
    best_score = float('inf')

    def _update(seq, sc):
        nonlocal best_a, best_score
        if seq is not None and np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = _norm1(seq).copy()

    # Seed from previous best if available
    try:
        hseq = np.asarray(height_sequence_1, dtype=np.float64)
        if hseq.size > 0 and np.all(np.isfinite(hseq)):
            sc = _score_of(hseq)
            _update(hseq, sc)
            print(f"Seeded from previous best: score={sc:.6f}, len={hseq.size}")
    except Exception:
        pass

    # Baseline: two-level half-block
    n2 = final_len // 2
    base = np.ones(final_len)
    base[:n2] = 4.0 / 3.0
    base[n2:] = 2.0 / 3.0
    base = _norm1(base)
    sc_base = _score_of(base)
    _update(base, sc_base)
    print(f"Two-level baseline: score={sc_base:.6f}")

    # Phase 1: Smooth gradient descent on short sequences
    print("Phase 1: Short sequence optimization...")
    proto_deadline = deadline - max(800.0, 0.4 * budget)
    prototype_results = []

    lengths = [32, 48, 64, 96, 128, 192, 256, 384, 512]
    for n in lengths:
        if time.monotonic() > proto_deadline:
            break
        if best_score < 1.51:
            break
        n_trials = max(2, 60 // int(np.log2(n) - 3))
        per_trial = max(2.0, (proto_deadline - time.monotonic()) / max(n_trials, 1) * 0.9)

        for trial in range(n_trials):
            if time.monotonic() > proto_deadline:
                break
            strategy = int(rng.integers(0, 16))
            p = _make_initial(n, strategy, rng)
            sc = _score_of(p)
            _update(p, sc)

            # Smooth gradient descent with continuation
            p_trial_deadline = min(deadline, time.monotonic() + per_trial)
            p_opt, sc_opt = _smooth_gradient_descent(
                p, p_schedule=[2, 4, 8, 16], max_iters=2000, lr_init=0.3,
                deadline=p_trial_deadline
            )
            _update(p_opt, sc_opt)

            # Quick coordinate descent polish
            if time.monotonic() < p_trial_deadline - 0.5:
                p_cd, sc_cd = _coord_descent(
                    p_opt, deadline=p_trial_deadline, rng=rng,
                    grid_size=16, refine=4, max_perturb=2
                )
                _update(p_cd, sc_cd)
                prototype_results.append((sc_cd, n, p_cd))

        print(f"  n={n}: best so far={best_score:.6f}")

    # Phase 2: Expand best prototypes to full length
    print("Phase 2: Expansion to full length...")
    expand_deadline = min(deadline, time.monotonic() + 120.0)
    prototype_results.sort(key=lambda x: x[0])

    full_candidates = []
    for sc, n, p in prototype_results[:10]:
        if time.monotonic() > expand_deadline:
            break
        p_exp = _expand(p, final_len)
        sc_exp = _score_of(p_exp)
        full_candidates.append((sc_exp, p_exp))
        _update(p_exp, sc_exp)

    # Also expand the current best if it's shorter
    if best_a is not None and best_a.size != final_len:
        p_exp = _expand(best_a, final_len)
        sc_exp = _score_of(p_exp)
        full_candidates.append((sc_exp, p_exp))
        _update(p_exp, sc_exp)

    # If no full-length candidate, create one
    full_best = None
    full_best_score = float('inf')
    for sc, p in full_candidates:
        if sc < full_best_score:
            full_best_score = sc
            full_best = p
    if full_best is None:
        full_best = base
        full_best_score = sc_base
    _update(full_best, full_best_score)

    # Phase 3: Full-length optimization
    print(f"Phase 3: Full-length optimization (score={full_best_score:.6f})...")
    full_deadline = deadline - 30.0
    n_iter_phase3 = int((full_deadline - time.monotonic()) / max(full_len / 5000.0, 0.1))

    # 3a: Smooth gradient descent on full length
    sg_deadline = min(full_deadline, time.monotonic() + max(300.0, 0.25 * budget))
    p_opt_full, sc_opt_full = _smooth_gradient_descent(
        full_best, p_schedule=[2, 4, 8, 16, 32], max_iters=5000, lr_init=0.2,
        deadline=sg_deadline
    )
    _update(p_opt_full, sc_opt_full)
    print(f"  After smooth gradient: {sc_opt_full:.6f}")

    # 3b: Coordinate descent polishing
    cd_deadline = full_deadline - 10
    p_cd_full, sc_cd_full = _coord_descent(
        p_opt_full, deadline=cd_deadline, rng=rng,
        grid_size=16, refine=4, max_perturb=4
    )
    _update(p_cd_full, sc_cd_full)
    print(f"  After coordinate descent: {sc_cd_full:.6f}")

    # 3c: Another round of smooth gradient + coord descent if time permits
    if time.monotonic() < cd_deadline - 120:
        sg2_deadline = min(cd_deadline - 60, time.monotonic() + 120)
        p2, sc2 = _smooth_gradient_descent(
            p_cd_full, p_schedule=[8, 16, 32], max_iters=3000, lr_init=0.15,
            deadline=sg2_deadline
        )
        _update(p2, sc2)
        if time.monotonic() < cd_deadline - 10:
            p3, sc3 = _coord_descent(
                p2, deadline=cd_deadline, rng=rng,
                grid_size=14, refine=3, max_perturb=3
            )
            _update(p3, sc3)
            print(f"  After second round: {sc3:.6f}")

    # Final: evaluate with the actual evaluator and return best
    print(f"Final internal score: {best_score:.6f}")
    if best_a is not None:
        try:
            ext_score = evaluate_sequence([float(x) for x in best_a])
            print(f"External evaluation: {ext_score:.6f}")
        except Exception as e:
            print(f"Evaluation error: {e}")
    else:
        best_a = base

    return [float(x) for x in _norm1(best_a)]
