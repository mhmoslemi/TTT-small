Actually, for the LP to be useful, M needs to be set correctly. The idea is:
- Current sequence a has max convolution value M
- We want to find x >= 0 such that a*x <= M (componentwise) and sum(x) is maximized
- Then blend: a_new = (1-t)*a + t*(x / sum(x))

This is a "projection" onto the set {x: a*x <= M, x >= 0} in the direction of increasing sum.

But wait, the constraint is a*x <= M, not a*a <= M. So x can be very different from a. The LP finds the "largest" (in sum) x that's in the "shadow" of a.

This is similar to the multiplicative weight update: x is the "best response" to a in a zero-sum game.

"""
Optimize non-negative step-function sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses L-BFGS on soft-max, subgradient steps, LP projection, and local search.
"""

import time

def _pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _clean(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    return np.maximum(v, 0.0)


def _normalize(v):
    v = _clean(v)
    s = float(v.sum())
    if s <= 1e-15 or not np.isfinite(s):
        return np.full(v.size, 1.0/v.size)
    return v / s


def _project_simplex(v):
    v = _clean(v)
    n = v.size
    if n == 0:
        return v.copy()
    x = np.sort(v)[::-1]
    css = np.cumsum(x)
    idx = np.arange(1, n+1, dtype=np.float64)
    cond = x > (css - 1.0) / idx
    if not np.any(cond):
        return np.full(n, 1.0/n)
    r = int(np.nonzero(cond)[0][-1])
    tau = (css[r] - 1.0) / (r + 1.0)
    out = v - tau
    np.maximum(out, 0.0, out=out)
    s = float(out.sum())
    if s <= 1e-15:
        return np.full(n, 1.0/n)
    return out / s


def _score(a, L=None):
    """Compute 2n * max(conv) for normalized a (sum=1)."""
    v = _normalize(a)
    n = v.size
    if n == 0:
        return np.inf
    if L is None:
        L = _pow2(2*n - 1)
    A = np.fft.rfft(v, n=L)
    c = np.fft.irfft(A*A, n=L)[:2*n-1]
    mx = float(c.max())
    if not np.isfinite(mx):
        return np.inf
    return 2.0 * n * mx


def _conv_full(a, L=None):
    """Full convolution via FFT."""
    n = a.size
    if L is None:
        L = _pow2(2*n - 1)
    A = np.fft.rfft(a, n=L)
    return np.fft.irfft(A*A, n=L)[:2*n-1]


def _soft_obj_grad(x, a, eps):
    """Soft-max objective and gradient in log-ratio space."""
    # Reconstruct a from log-ratios
    n_x = x.size
    n = n_x + 1
    m_max = float(np.max(x)) if x.size else 0.0
    e = np.exp(np.clip(x - m_max, -700, 700))
    e_last = float(np.exp(-m_max))
    s = float(e.sum()) + e_last
    a = np.empty(n)
    a[:-1] = e / s
    a[-1] = e_last / s
    
    L = _pow2(3*n - 2)
    Af = np.fft.rfft(a, n=L)
    c = np.fft.irfft(Af*Af, n=L)[:2*n-1]
    c = np.maximum(c, 0.0)
    
    cmax = float(c.max())
    z = (c - cmax) / eps
    z = np.clip(z, -700, 0)
    w = np.exp(z)
    wsum = float(w.sum())
    
    f = cmax + eps * np.log(wsum)
    w = w / wsum
    
    Wf = np.fft.rfft(w, n=L)
    ga = 2.0 * np.fft.irfft(Wf * np.conj(Af), n=L)[:n]
    
    dot = float(np.dot(a, ga))
    gfull = a * (ga - dot)
    gy = gfull[:-1]
    
    if not np.all(np.isfinite(gy)):
        gy = np.zeros(n_x)
    return f, gy


def _logratios(a):
    a = _normalize(a).copy()
    n = a.size
    if n <= 1:
        return np.array([], dtype=np.float64)
    ref = float(a[-1])
    if ref < 1e-200:
        a[-1] = 1e-12
        a /= a.sum()
        ref = float(a[-1])
    x = np.log(np.maximum(a[:-1], 1e-300) / ref)
    return np.clip(x, -80, 80)


def _from_logratios(x):
    n = x.size + 1
    m_max = float(np.max(x)) if x.size else 0.0
    e = np.exp(np.clip(x - m_max, -700, 700))
    e_last = float(np.exp(-m_max))
    s = float(e.sum()) + e_last
    a = np.empty(n)
    a[:-1] = e / s
    a[-1] = e_last / s
    return a


def _subgradient_step(a, step):
    """One subgradient step on the max convolution."""
    n = a.size
    c = _conv_full(a)
    k_star = int(np.argmax(c))
    # Subgradient: d conv[k*]/da[i] = a[k*-i] (zero-padded)
    # Construct the gradient vector
    g = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            g[i] += a[j]
    # Also the "other" term (when j and k*-j are both valid)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n and j != i:
            g[i] += a[j]
    # Actually, let me just do: g[i] = a[k*-i] if 0<=k*-i<n, else 0
    # (factor of 2 doesn't matter, just affects step size)
    g = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            g[i] = a[j]
    
    # Projected gradient step: a_new = project(a - step * g / ||g||)
    g_norm = float(np.linalg.norm(g)) + 1e-12
    a_new = _project_simplex(a - step * g / g_norm)
    return a_new, k_star


def _lp_project(a, M, max_time=60):
    """LP projection: max sum(x) s.t. a*x <= M, x >= 0."""
    if not _HAS_LINPROG:
        return None
    n = a.size
    start = time.monotonic()
    
    # Build constraint matrix: (2n-1) x n
    # A[k, j] = a[j] if 0 <= k-j < n, else 0
    # Using sparse format
    row_idx = []
    col_idx =
    vals = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k + 1, n)
        for j in range(j_start, j_end):
            row_idx.append(k)
            col_idx.append(j)
            vals.append(float(a[j]))
    
    A = _sparse.csr_matrix((vals, (row_idx, col_idx)), shape=(2*n-1, n))

    if time.monotonic() - start > max_time + 5:
        return None
    
    if res.status != 0:
        return None
    
    x = _normalize(res.x)
    return x


def _local_search(a, end_time, seed, L=None):
    """Random local search with multiplicative perturbations."""
    a = _normalize(a).copy()
    n = a.size
    if L is None:
        L = _pow2(2*n - 1)
    
    best = a.copy()
    best_sc = _score(best, L)
    rng = np.random.default_rng(seed)
    sigma = 0.1
    last_best = time.monotonic()
    
    while time.monotonic() < end_time:
        if time.monotonic() - last_best > 15.0:
            sigma = max(0.01, sigma * 0.7)
            if time.monotonic() - last_best > 60.0:
                break
        
        cand = best.copy()
        # Random block perturbation
        n_blocks = int(rng.integers(1, 4))
        for _ in range(n_blocks):
            w = int(np.exp(rng.uniform(1, np.log(max(2, n//4)))))
            w = max(1, min(w, n))
            s = int(rng.integers(0, n - w + 1))
            cand[s:s+w] *= float(np.exp(rng.normal(0, sigma)))
        
        cand = _project_simplex(cand)
        sc = _score(cand, L)
        
        if sc < best_sc - 1e-10:
            best = cand
            best_sc = sc
            last_best = time.monotonic()
            sigma = min(0.3, sigma * 1.05)
    
    return best, best_sc
