import numpy as np
from scipy.optimize import linprog
import time

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]

def _lp_step(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0"""
    # Variables: g[0], ..., g[n-1]
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: A_ub @ g <= b_ub
    # For each k = 0, ..., 2n-2: sum_j f[j] * g[k-j] <= M
    # (f*g)[k] = sum_{j} f[j] * g[k-j] where 0 <= k-j < n
    # i.e., j in [max(0, k-n+1), min(k, n-1)]
    # coefficient of g[m] in constraint k is f[k-m] when m in [max(0,k-n+1), min(k,n-1)]
    
    n_constraints = 2 * n - 1
    # Build A_ub as a sparse matrix
    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    vals = []

        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)

            rows.append(k)
            cols.append(m)

Actually, let me reconsider. The LP-based approach from the state of the art is:
1. Normalize f so sum(f) = sqrt(2n)
2. M = ||f*f||_inf
3. Solve LP: max sum(g) s.t. ||f*g||_inf <= M, g >= 0
4. Rescale g: g <- g * sqrt(2n) / sum(g)
5. Update: f <- (1-t)*f + t*g
6. Repeat

After step 5, sum(f_new) = (1-t)*sqrt(2n) + t*sqrt(2n) = sqrt(2n). Good, normalization is preserved.

The score in this normalization is: 2n * ||f*f||_inf / (sum f)^2 = 2n * ||f*f||_inf / (2n) = ||f*f||_inf.

So minimizing the score is equivalent to minimizing ||f*f||_inf in this normalization.

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]

def _build_lp_matrix(n, f):
    m_vals = np.arange(n, dtype=np.int32)
    col_idx = np.repeat(m_vals, n)
    row_idx = (m_vals[:, None] + np.arange(n, dtype=np.int32)[None, :]).ravel()
    val_idx = np.tile(f, n)
    return csr_matrix((val_idx, (row_idx, col_idx)), shape=(2*n-1, n))

def _lp_solve(f, M, n):
    c = -np.ones(n)
    A_ub = _build_lp_matrix(n, f)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0)
    return None

def _next_pow2(n):
    if n <= 1:

    return 1 << (n - 1).bit_length()


def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]


def _build_lp_matrix(n, f):
    """Build sparse matrix A where A[k,m] = f[k-m] if 0<=k-m<n, else 0."""
    m_vals = np.arange(n, dtype=np.int32)
    col_idx = np.repeat(m_vals, n)
    row_idx = (m_vals[:, None] + np.arange(n, dtype=np.int32)[None, :]).ravel()
    val_idx = np.tile(f, n)
    return csr_matrix((val_idx, (row_idx, col_idx)), shape=(2 * n - 1, n))
