    n = len(f)
    # Variables: g_0, ..., g_{n-1}
    # Objective: maximize sum(g) -> minimize -sum(g)
    c_obj = -np.ones(n)

    # (f*g)_k = sum_j f_j * g_{k-j}
    # For variable g_i: coefficient in constraint k is f_{k-i} (if 0 <= k-i <= n-1)

    b_ub = np.full(2*n-1, M)

    bounds = [(0, None)] * n

import numpy as np
from scipy.optimize import linprog
import time

"""
Minimizes 2n*max(convolve(a,a))/sum(a)^2 using:
1. Vectorized coordinate descent (main workhorse)
2. LP-based direction finding (periodic, via Frank-Wolfe/CGM)
3. Multi-peak subgradient polishing
4. Multi-n exploration and adaptive restarts
"""


def fft_self_conv(a):
    """Compute full self-convolution using FFT for speed."""
    n = len(a)
    if n <= 1:
        return a * a
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:2 * n - 1]


def normalize(a, target):
    """Normalize to sum=target, clip negatives."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.full_like(a, target / len(a))
    return a / s * target


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where A[k,i] = f[k-i] if 0<=k-i<n."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        A[k, i_min:i_max + 1] = f[k - i_max: k - i_min + 1][::-1]
    return A


def lp_direction(f, M):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    n = len(f)
    c_obj = -np.ones(n)
    A_ub = build_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual feas tol': 1e-8})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def coord_descent_pass(a, c, S, n, rng, n_candidates=5):
    """One pass of vectorized coordinate descent. Returns updated a, c, S, improved."""
    order = rng.permutation(n)
    improved = 0
    for j in order:
        x = a[j]
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        n_affected = k_end - k_start + 1
        if n_affected <= 0:
            continue

        # Current max
        cur_max = c[k_start:k_end + 1].max()
        if k_start > 0:
            cur_max = max(cur_max, c[:k_start].max())
        if k_end + 1 < 2 * n - 1:
            cur_max = max(cur_max, c[k_end + 1:].max())

        # Generate candidate y values
        if x < 1e-12:
            candidates = np.array([0.0, S / n * 0.3, S / n * 0.7, S / n * 1.3])
        else:
            candidates = np.array([0.0, x * 0.3, x * 0.7, x * 1.3, x * 1.8])
            # Also try values based on the constraint
            # For the dominant lag, optimal y reduces the peak
        candidates = np.clip(candidates, 0, 1000)
        # Remove duplicates and x itself
        candidates = candidates[~np.isin(candidates, [x], assume_unique=False)]
        if len(candidates) == 0:
            continue

        best_y = x
        best_score = 2.0 * n * cur_max / (S * S)

        ki_range = np.arange(0, n_affected, dtype=np.int64)
        for y in candidates:
            delta_y = y - x
            # Compute new values for affected lags
            new_vals = 2.0 * delta_y * a[ki_range]
            if 2 * j >= k_start and 2 * j <= k_end:
                new_vals[2 * j - k_start] = y * y - x * x
            new_affected = c[k_start:k_end + 1] + new_vals
            new_max = new_affected.max()
            if k_start > 0:
                new_max = max(new_max, c[:k_start].max())
            if k_end + 1 < 2 * n - 1:
                new_max = max(new_max, c[k_end + 1:].max())
            new_S = S + delta_y
            if new_S < 0.01:
                continue
            score = 2.0 * n * new_max / (new_S * new_S)
            if score < best_score - 1e-12:
                best_score = score
                best_y = y

        if best_y != x:
            delta_y = best_y - x
            new_vals = 2.0 * delta_y * a[ki_range]
            if 2 * j >= k_start and 2 * j <= k_end:
                new_vals[2 * j - k_start] = best_y * best_y - x * x
            c[k_start:k_end + 1] += new_vals
            a[j] = best_y
            S += delta_y
            improved += 1

    return a, c, S, improved


def subgradient_step(a, c, n, S, lr, top_k=10):
    """Multi-peak subgradient step."""
    M = c.max()
    thresh = M * 0.98
    peaks = np.where(c >= thresh)[0]
    # Pick top_k peaks that are spread out
    peaks_sorted = peaks[np.argsort(-c[peaks])]
    selected = []
    for p in peaks_sorted:
        if all(abs(p - s) > 3 for s in selected):
            selected.append(int(p))
        if len(selected) >= top_k:
            break
    if not selected:
        selected = [int(np.argmax(c))]

    # Subgradient of max_k (a*a)_k: average of gradients at selected peaks
    # d(a*a)_k/da_j = 2*a[k-j] if 0<=k-j<n, else 0
    grad = np.zeros(n)
    for k in selected:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        grad[j_min:j_max + 1] += 2.0 * a[k - j_max: k - j_min + 1][::-1]
    grad /= len(selected)

    # Score gradient (with S=n): d(score)/da_j = (2/n)*grad[j] - (4M/n^2)*sign
    # Simplified: just use grad direction projected orthogonal to 1
    grad -= grad.mean()

    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize(a_new, n)
    return a_new
