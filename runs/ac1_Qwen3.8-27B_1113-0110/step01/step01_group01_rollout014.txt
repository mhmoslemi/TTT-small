"""
Hybrid nonnegative step-function search.

Minimizes
    2 * n * max(convolve(a, a)) / (sum(a))^2
by combining:
  - a given previous candidate when available,
  - Frank-Wolfe optimized active-set weights for multiplicative mirror steps,
  - LP-dual weights on a downscaled copy (when SciPy HiGHS is available),
  - very local O(n) coordinate multiplicative updates with exact incremental
    convolution updates,
  - random smooth block perturbations to escape basins,
  - an optional longer-length stretch for extra attack surface.

The final candidate is scaled to sum sqrt(2n) before returning, which is
scale-invariant for the evaluator but gives healthier floating-point magnitudes.
"""

def next_pow2(m: int) -> int:
    m = max(1, int(m))
    return 1 << (m - 1).bit_length()


def normalize_positive(a, floor: float = 0.0) -> np.ndarray:
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, float(floor), 1e300)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.full(a.size, 1.0 / float(a.size), dtype=np.float64)
    return a / s


def make_uniform(n: int) -> np.ndarray:
    n = max(1, int(n))
    return np.full(n, 1.0 / float(n), dtype=np.float64)


def resample_seq(a, n: int) -> np.ndarray:
    a = normalize_positive(a, floor=0.0)
    n = max(2, int(n))
    old_n = int(a.size)
    if old_n == n:
        return a.astype(np.float64, copy=True)
    if old_n <= 1:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    old_x = np.linspace(0.0, 1.0, old_n, dtype=np.float64)
    new_x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    b = np.interp(new_x, old_x, a)
    return normalize_positive(b, floor=1e-14)


def resample_weights(w, target_len: int) -> np.ndarray:
    w = normalize_positive(w, floor=0.0)
    target_len = max(1, int(target_len))
    if w.size == target_len:
        return w.astype(np.float64, copy=True)
    if w.size <= 1:
        return np.full(target_len, 1.0 / float(target_len), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, w.size, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, target_len, dtype=np.float64)
    out = np.interp(x_new, x_old, w)
    return normalize_positive(out, floor=0.0)

def generate_inits(n: int, seed: int):
    n = max(16, int(n))
    rng = np.random.default_rng(int(seed))
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    seqs = [make_uniform(n)]

    # Endpoint-heavy U-shapes.
    for p in (4, 6, 8, 12):
        base = 1.0 + 8.0 * (x ** p + (1.0 - x) ** p)
        seqs.append(normalize_positive(base, floor=1e-12))

    # Asymmetric beta-like shapes.
    for aa, bb in ((2, 1), (3, 2), (5, 1), (1, 3), (2, 5), (1, 1), (8, 3), (3, 8)):
        base = (x + 0.05) ** aa * ((1.0 - x) + 0.05) ** bb
        seqs.append(normalize_positive(base, floor=1e-12))

    # Random smooth Fourier shapes.
    for _ in range(14):
        v = np.ones(n, dtype=np.float64)
        for m in range(1, 14):
            amp = rng.uniform(0.10, 1.00) / (float(m) ** 0.7)
            ph = rng.uniform(0.0, 2.0 * np.pi)
            v += amp * np.sin(2.0 * np.pi * float(m) * x + ph)
            v += 0.6 * amp * np.cos(2.0 * np.pi * float(m) * x + ph)
        seqs.append(normalize_positive(v, floor=1e-12))

    # Odd-like modes that create weak ends/internal contrast.
    for _ in range(8):
        v = np.ones(n, dtype=np.float64)
        for m in (1, 2, 3, 5, 7):
            amp = rng.uniform(0.25, 1.25)
            v += amp * np.sin(np.pi * float(m) * x + rng.uniform(0.0, np.pi))
        seqs.append(normalize_positive(v, floor=1e-12))

    return seqs


def convolution_score(a, Nconv=None):
    """Return normalized a, its normalized self-convolution, and score."""
    a = normalize_positive(a, floor=0.0)
    n = int(a.size)
    if Nconv is None:
        Nconv = next_pow2(2 * n - 1)
    else:
        need = next_pow2(2 * n - 1)
        if int(Nconv) < need:
            Nconv = need
    fa = np.fft.rfft(a, Nconv)
    c = np.fft.irfft(fa * fa, Nconv)[: 2 * n - 1]
    c = np.maximum(c, 0.0)
    sc = 2.0 * float(n) * float(c.max())
    return a, c, sc


def internal_score(a, Nconv=None) -> float:
    if a is None:
        return np.inf
    try:
        _, _, sc = convolution_score(a, Nconv)
        return float(sc)
    except Exception:
        return np.inf


def gradient_from_weights(w, a, Ng=None):
    """
    Gradient of sum_k w_k c_k(c = a*a) with respect to a.

    Returns g_i = 2 * sum_k w_k a_{k-i} over valid i.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = int(a.size)
    L = 2 * n - 1
    if w.size != L:
        w = resample_weights(w, L)
    if Ng is None:
        Ng = next_pow2(3 * n - 2)
    else:
        need = next_pow2(3 * n - 2)
        if int(Ng) < need:
            Ng = need
    wr = w[::-1]
    fa = np.fft.rfft(a, Ng)
    fw = np.fft.rfft(wr, Ng)
    z = np.fft.irfft(fa * fw, Ng)
    return 2.0 * z[n - 1:2 * n - 1][::-1]


def smooth_max(c, tau: float) -> float:
    c = np.asarray(c, dtype=np.float64).ravel()
    cmax = float(c.max()) if c.size else 0.0
    if cmax <= 0.0 or not np.isfinite(cmax):
        return 0.0
    tau = float(tau)
    if tau <= 0.0:
        return cmax
    z = np.clip((c - cmax) / tau, -50.0, 0.0)
    s = float(np.exp(z).sum())
    if not np.isfinite(s) or s <= 0.0:
        return cmax
    return float(cmax + tau * float(np.log(s)))


def weights_top(c, r: int = 128, floor: float = 0.0):
    c = np.asarray(c, dtype=np.float64).ravel()
    L = int(c.size)
    r = max(1, min(int(r), L))
    if r >= L:
        idx = np.arange(L, dtype=np.int64)
    else:
        idx = np.argpartition(c, -r)[-r:].astype(np.int64)
    w = np.zeros(L, dtype=np.float64)
    w[idx] = 1.0
    w /= float(w.sum())
    if floor and floor > 0.0:
        f = min(0.99, float(floor))
        w = (1.0 - f) * w + (f / float(L))
    return w


def weights_smooth(c, tau: float, floor: float = 0.01):
    c = np.asarray(c, dtype=np.float64).ravel()
    cmax = float(c.max())
    if not np.isfinite(cmax) or cmax <= 0.0 or tau <= 0.0:
        return weights_top(c, r=max(16, c.size // 4), floor=floor)
    z = np.clip((c - cmax) / float(tau), -50.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return weights_top(c, r=max(16, c.size // 4), floor=floor)
    w = e / s
    if floor and floor > 0.0:
        f = min(0.99, float(floor))
        w = (1.0 - f) * w + (f / float(w.size))
    return w


def weights_frankwolfe(a, c, r=None, fw_iter: int = 40, floor: float = 0.0):
    """
    Choose active convolution coefficients, then solve a small Frank-Wolfe
    problem over probability weights on those coefficients. The objective is
    the variance of the resulting weighted gradient, which pushes the weights
    toward KKT-like flattening.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    c = np.asarray(c, dtype=np.float64).ravel()
    n = int(a.size)
    L = int(c.size)
    if r is None:
        r = int(max(24, min(200, n // 10)))
    r = max(1, min(int(r), L))

    if r >= L:
        idx = np.arange(L, dtype=np.int64)
    else:
        idx = np.argpartition(c, -r)[-r:].astype(np.int64)

    imax = int(np.argmax(c))
    if imax not in idx:
        vals = c[idx]
        j = int(np.argmin(vals))
        idx = idx.copy()
        idx[j] = imax

    B = np.zeros((n, r), dtype=np.float64)
    for t in range(r):
        kk = int(idx[t])
        low = kk - (n - 1)
        if low < 0:
            low = 0
        high = kk
        if high >= n:
            high = n - 1
        if high < low:
            continue
        start = kk - high
        stop = kk - low + 1
        B[low:high + 1, t] = a[start:stop][::-1]

    Bc = B - B.mean(axis=0)
    Q = Bc.T @ Bc
    Q = 0.5 * Q
    Q += 1e-8 * np.eye(r, dtype=np.float64)

    y = np.full(r, 1.0 / float(r), dtype=np.float64)
    for it in range(max(1, int(fw_iter))):
        G = Q @ y
        s = int(np.argmin(G))
        base_val = float(y @ Q @ y)
        base_alpha = 2.0 / float(it + 2)
        best_alpha = 0.0
        best_val = base_val
        for alpha in (base_alpha, 0.5 * base_alpha, 0.25 * base_alpha, 1.0 / float(it + 2)):
            if alpha <= 1e-14 or alpha >= 1.0 - 1e-14:
                continue
            y2 = (1.0 - alpha) * y
            y2[s] += alpha
            val = float(y2 @ Q @ y2)
            if val < best_val - 1e-13:
                best_val = val
                best_alpha = alpha
        if best_alpha <= 0.0:
            break
        y = (1.0 - best_alpha) * y
        y[s] += best_alpha
        y /= float(y.sum())

    w = np.zeros(L, dtype=np.float64)
    w[idx] = y
    s = float(w.sum())
    if s <= 0.0 or not np.isfinite(s):
        return weights_top(c, r=r, floor=floor)
    w /= s
    if floor and floor > 0.0:
        f = min(0.99, float(floor))
        w = (1.0 - f) * w + (f / float(L))
    return w


def mirror_candidate(a, c, score, weights, Nconv, Ng, etas=None):
    """Try a few multiplicative mirror steps and return the first exact improvement."""
    if etas is None:
        etas = (0.08, 0.04, 0.16, 0.02, 0.32, 0.06, 0.64)
    try:
        g = gradient_from_weights(weights, a, Ng)
        g = g - float(g.mean())
        scale = float(np.max(np.abs(g)))
        if not np.isfinite(scale) or scale <= 1e-18:
            return None
        g /= scale

        loga = np.log(a + 1e-300)
        loga -= float(np.max(loga))

        for eta in etas:
            loga_try = loga - float(eta) * g
            loga_try = loga_try - float(np.max(loga_try))
            a_try = np.exp(np.clip(loga_try, -60.0, 0.0))
            a_try = np.maximum(a_try, 0.0)
            a_try, c_try, sc_try = convolution_score(a_try, Nconv)
            if np.isfinite(sc_try) and sc_try < float(score) * (1.0 - 1e-11):
                return a_try, c_try, float(sc_try)
    except Exception:
        return None
    return None

    best_c = c.copy()
    best_score = score

    tmp = np.empty(n, dtype=np.float64)

    inc_factors = (1.15, 1.35, 1.65, 2.15, 2.95, 4.0)
    zero_base = 1.0 / float(n)

    weights = None
    g = None
    high_idx = None
    low_idx = None
    high_p = None
    low_p = None

    temp = 1e-4
    it = 0
    stale = 0

    while it < int(max_trials) and (deadline is None or time.monotonic() < deadline):
        if weights is None or it % int(fw_every) == 0:

                g = np.zeros(n, dtype=np.float64)

            rpick = max(8, min(256, n // 8))
            if np.max(g) - np.min(g) > 1e-15:
                if rpick < n:
                    high_idx = np.argpartition(g, -rpick)[-rpick:].astype(np.int64)
                    low_idx = np.argpartition(g, rpick)[:rpick].astype(np.int64)
                else:
                    high_idx = np.arange(n, dtype=np.int64)
                    low_idx = np.arange(n, dtype=np.int64)
                hi = g[high_idx]
                lo = g[low_idx]
                hi_p = np.maximum(hi - float(hi.min()), 1e-12)
                hi_p = hi_p / float(hi_p.sum())
                lo_p = np.maximum(float(lo.max()) - lo, 1e-12)
                lo_p = lo_p / float(lo_p.sum())
                high_p = hi_p
                low_p = lo_p
            else:
                high_idx = np.arange(n, dtype=np.int64)
                low_idx = np.arange(n, dtype=np.int64)
                high_p = np.full(n, 1.0 / float(n), dtype=np.float64)
                low_p = high_p.copy()

        r = rng.random()
        if r < 0.55:
            i = int(high_idx[int(rng.choice(high_idx, p=high_p))])
            dec_mode = True
        elif r < 0.90:
            i = int(low_idx[int(rng.choice(low_idx, p=low_p))])
            dec_mode = False
        else:
            i = int(rng.integers(0, n))
            dec_mode = rng.random() < 0.5

        v = float(a[i])

        if v > 1e-15:
            if dec_mode:
                u_list = tuple(v * f for f in dec_factors)
            else:
                u_list = tuple(v * f for f in inc_factors)
        else:
            # Multiplicative increases do nothing at exactly zero; add small absolute mass.
            u_list = (0.05 * zero_base, 0.20 * zero_base, 0.80 * zero_base, 2.50 * zero_base)

        base = (a + v) if v != 0.0 else a

        best_delta = 0.0
        best_sc = score
        best_S = 1.0

        for u in u_list:
            if not np.isfinite(u):
                continue
            u = max(0.0, float(u))
            delta = u - v
            if abs(delta) <= 1e-15 * max(1.0, abs(v)):
                continue
            S = 1.0 + delta
            if S <= 1e-10:
                continue

            np.multiply(base, delta, out=tmp)
            np.add(c[i:i + n], tmp, out=c[i:i + n])
            d2 = delta * delta
            c[2 * i] += d2
            mx = float(c.max())
            sc = (2.0 * float(n) * mx) / (S * S)

            if np.isfinite(sc) and sc < best_sc - 1e-15:
                best_sc = sc
                best_delta = delta
                best_S = S

            np.subtract(c[i:i + n], tmp, out=c[i:i + n])
            c[2 * i] -= d2

        accept = False
        if best_delta != 0.0 and best_sc < score * (1.0 - 1e-10):
            accept = True
        elif best_delta != 0.0 and best_sc < score * (1.0 + temp) and rng.random() < 0.02:
            accept = True

        if accept:
            np.multiply(base, best_delta, out=tmp)
            np.add(c[i:i + n], tmp, out=c[i:i + n])
            d2 = best_delta * best_delta
            c[2 * i] += d2
            a[i] = float(a[i]) + best_delta
            S = float(best_S)
            if S > 1e-12 and np.isfinite(S):
                inv = 1.0 / S
                a *= inv
                c *= inv * inv
            score = float(best_sc)
            stale = 0
            if score < best_score * (1.0 - 1e-12):
                best_a = a.copy()
                best_c = c.copy()
                best_score = score
                temp = max(1e-7, temp * 0.95)
            else:
                temp = max(1e-7, temp * 0.98)
        else:
            stale += 1
            if stale > 1200:
                a = best_a.copy()
                c = best_c.copy()
                score = float(best_score)
                temp = min(1e-3, max(temp * 2.0, 1e-5))
                stale = 0

        it += 1

        if it % 2500 == 0:
            try:
                a, c, score = convolution_score(a, Nconv)
                if score < best_score * (1.0 - 1e-12):
                    best_a = a.copy()
                    best_c = c.copy()
                    best_score = score
                    stale = 0
            except Exception:
                pass

    try:
        best_a, best_c, best_score = convolution_score(best_a, Nconv)
    except Exception:
        pass
    return best_a, best_c, float(best_score)


def coordinate_search(a_in, deadline, seed=0, label="coord"):
    a = normalize_positive(a_in, floor=0.0)
    n = int(a.size)
    Nconv = next_pow2(2 * n - 1)
    Ng = next_pow2(3 * n - 2)
    a, c, score = convolution_score(a, Nconv)
    best_a = a.copy()
    best_c = c.copy()
    best_score = float(score)

    rng = np.random.default_rng(int(seed) + 17)
    stagnant_chunks = 0
    last_print = -1.0
    last_improve = time.monotonic()

    while time.monotonic() < max(start, deadline) if False else time.monotonic() < deadline - 18.0:
        chunk_deadline = min(deadline - 15.0, time.monotonic() + 8.0)
        try:
            a1, c1, s1 = coordinate_batch(
                best_a, best_c, best_score,
                Nconv=Nconv, Ng=Ng, n=n,
                deadline=chunk_deadline,
                seed=int(seed) + stagnant_chunks * 7919 + 101,
                max_trials=1000000,
                fw_every=300,
            )
        except Exception:
            break

        if np.isfinite(s1) and s1 < best_score * (1.0 - 1e-11):
            best_a, best_c, best_score = a1, c1, float(s1)
            stagnant_chunks = 0
            last_improve = time.monotonic()
        else:
            stagnant_chunks += 1

        if stagnant_chunks >= 4:
            improved = False
            for attempt in range(10):
                scale = 0.08 + 0.045 * float(attempt)
                try:
                    cand = random_block_candidate(best_a, int(seed) + attempt * 31 + 1000, scale=scale)
                    sc = internal_score(cand, Nconv)
                    if np.isfinite(sc) and sc < best_score * (1.0 - 1e-11):
                        best_a, best_c, best_score = cand, None, float(sc)
                        best_a, best_c, best_score = convolution_score(best_a, Nconv)
                        improved = True
                        break
                except Exception:
                    pass
            if not improved:
                for attempt in range(6):
                    try:
                        cand = random_smooth_seq(n, int(seed) + attempt * 91 + 20000)
                        sc = internal_score(cand, Nconv)
                        if np.isfinite(sc) and sc < best_score * (1.0 - 1e-11):
                            best_a, best_c, best_score = cand, None, float(sc)
                            best_a, best_c, best_score = convolution_score(best_a, Nconv)
                            improved = True
                            break
                    except Exception:
                        pass
            stagnant_chunks = 0

        now = time.monotonic()
        if now - last_print > 60.0 and label:
            print("{} score={:.10f}".format(label, best_score), flush=True)
            last_print = now

        if now - last_improve > 600.0 and stagnant_chunks > 20:
            break

    try:
        best_a, best_c, best_score = convolution_score(best_a, Nconv)
    except Exception:
        pass
    return best_a, float(best_score)


def random_block_candidate(a, seed, scale: float = 0.20):
    a = normalize_positive(a, floor=1e-14)
    n = int(a.size)
    rng = np.random.default_rng(int(seed))
    low = max(4, n // 128)
    high = max(low + 1, n // 12)
    width = int(rng.integers(low, high + 1))
    if width < 1:
        width = 1
    width = min(width, n)
    start = int(rng.integers(0, max(1, n - width + 1)))

    loga = np.log(a + 1e-300)
    loga -= float(np.max(loga))
    t = np.linspace(0.0, 1.0, width, dtype=np.float64)
    env = np.exp(-3.0 * ((t - 0.5) ** 2))
    d = float(scale) * env * rng.normal(size=width)
    d -= float(d.mean())
    d = np.clip(d, -3.0, 3.0)
    loga[start:start + width] += d
    out = np.exp(np.clip(loga - float(np.max(loga)), -60.0, 0.0))
    return normalize_positive(out, floor=1e-14)


def random_smooth_seq(n: int, seed: int):
    n = max(16, int(n))
    rng = np.random.default_rng(int(seed))
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    v = np.ones(n, dtype=np.float64)
    for m in range(1, 16):
        amp = rng.uniform(0.05, 1.0) / (float(m) ** 0.65)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        v += amp * np.sin(2.0 * np.pi * float(m) * x + ph)
        v += 0.5 * amp * np.cos(2.0 * np.pi * float(m) * x + ph)
    return normalize_positive(v, floor=1e-12)


def random_block_polish(a_in, deadline, seed=0):
    a = normalize_positive(a_in, floor=1e-14)
    n = int(a.size)
    Nconv = next_pow2(2 * n - 1)
    a, c, score = convolution_score(a, Nconv)
    best_a, best_c, best_score = a, c, float(score)

    loga = np.log(a + 1e-300)
    loga -= float(np.max(loga))
    rng = np.random.default_rng(int(seed))
    scale = 0.22
    stale = 0

    while time.monotonic() < deadline - 10.0:
        low = max(4, n // 256)
        high = max(low + 1, n // 8)
        width = int(rng.integers(low, high + 1))
        width = min(width, max(1, n // 2))
        if width < 1:
            width = 1
        start = int(rng.integers(0, max(1, n - width + 1)))

        t = np.linspace(0.0, 1.0, width, dtype=np.float64)
        env = np.exp(-3.5 * ((t - 0.5) ** 2))
        d = float(scale) * env * rng.normal(size=width)
        d -= float(d.mean())
        d = np.clip(d, -3.0, 3.0)

        old = loga[start:start + width].copy()
        loga[start:start + width] += d
        a_try = np.exp(np.clip(loga - float(np.max(loga)), -60.0, 0.0))
        a_try = normalize_positive(a_try, floor=1e-14)
        sc = internal_score(a_try, Nconv)

        if np.isfinite(sc) and sc < best_score * (1.0 - 1e-11):
            best_a, best_c, best_score = convolution_score(a_try, Nconv)
            loga = np.log(best_a + 1e-300)
            loga -= float(np.max(loga))
            stale = 0
            scale = min(0.75, scale * 1.04)
        else:
            loga[start:start + width] = old
            stale += 1
            scale = max(0.03, scale * 0.96)
            if stale > 120:
                scale = max(0.25, scale * 2.2)
                stale = 0

    try:
        best_a, best_c, best_score = convolution_score(best_a, Nconv)
    except Exception:
        pass
    return best_a, float(best_score)


def lp_dual_weights(a, n_target: int = 512):
    """
    Solve the dual LP
        min sum(y)
        s.t. (shifted points of a) . y >= 1, y >= 0
    on a downscaled copy of a. The returned y are useful active-lag weights.
    """
    if _scio is None:
        return None
    try:
        n = max(64, int(n_target))
        b = resample_seq(a, n)
        # Small positive floor keeps rows well conditioned.
        b = b + 1e-7 * np.ones(n, dtype=np.float64)
        b = normalize_positive(b, floor=1e-12)

        # Scale by n so entries are O(1), improving LP conditioning.
        b_scaled = b * float(n)
        L = 2 * n - 1
        B = np.zeros((n, L), dtype=np.float64)
        for i in range(n):
            B[i, i:i + n] = b_scaled

        c_obj = np.ones(L, dtype=np.float64)
        A_ub = -B
        b_ub = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None) for _ in range(L)]
        res = None
        for method in ("highs", "interior-point"):
            try:
                res = _scio.linprog(
                    c_obj,
                    A_ub=A_ub,
                    b_ub=b_ub,
                    bounds=bounds,
                    method=method,
                )
                if res is not None and getattr(res, "success", False) and res.x is not None:
                    break
            except Exception:
                res = None
        if res is None:
            return None
        if not getattr(res, "success", False):
            return None
        y = np.asarray(res.x, dtype=np.float64).ravel()
        if y.size != L:
            return None
        if not np.all(np.isfinite(y)):
            return None
        s = float(y.sum())
        if s <= 1e-12:
            return None
        y = np.maximum(y, 0.0)
        s = float(y.sum())
        if s <= 1e-12:
            return None
        return y / s
    except Exception:
        return None


def mirror_search(a_in, deadline, seed=0, lp_n: int = 512, max_lp_solves: int = 8):
    a = normalize_positive(a_in, floor=1e-14)
    n = int(a.size)
    Nconv = next_pow2(2 * n - 1)
    Ng = next_pow2(3 * n - 2)
    a, c, score = convolution_score(a, Nconv)
    best_a, best_c, best_score = a, c, float(score)

    weights = None
    it = 0
    lp_solves = 0
    lp_enabled = True
    current_lp_n = max(256, int(lp_n))
    last_improve = time.monotonic()
    last_print = -1.0

    while time.monotonic() < deadline - 18.0:
        # Periodically refresh weights. LP dual is used sparingly.
        if weights is None or it % 7 == 0:
            got_lp = False
            if (lp_enabled and lp_solves < max_lp_solves and it % 14 == 0
                    and time.monotonic() < deadline - 260.0):
                t0 = time.monotonic()
                y = lp_dual_weights(a, n_target=current_lp_n)
                elapsed = time.monotonic() - t0
                if y is not None:
                    weights = resample_weights(y, 2 * n - 1)
                    lp_solves += 1
                    got_lp = True
                    if elapsed > 120.0:
                        current_lp_n = max(256, current_lp_n // 2)
                    if elapsed > 220.0:
                        lp_enabled = False
            if not got_lp:
                try:
                    r = max(24, min(180, n // 10))
                    weights = weights_frankwolfe(a, c, r=r, fw_iter=35, floor=0.02)
                except Exception:
                    weights = weights_top(c, r=max(24, min(180, n // 10)), floor=0.02)

        cand = mirror_candidate(a, c, score, weights, Nconv, Ng)
        if cand is not None:
            a, c, score = cand
            if score < best_score * (1.0 - 1e-11):
                best_a, best_c, best_score = a, c, float(score)
                last_improve = time.monotonic()

        # Short coordinate polish after each mirror wave to remove roundoff and local ripples.
        chunk_deadline = min(deadline - 15.0, time.monotonic() + 1.6)
        try:
            a1, c1, s1 = coordinate_batch(
                best_a, best_c, best_score,
                Nconv=Nconv, Ng=Ng, n=n,
                deadline=chunk_deadline,
                seed=int(seed) + it * 101 + lp_solves * 31,
                max_trials=200000,
                fw_every=250,
            )
            if s1 < best_score * (1.0 - 1e-11):
                best_a, best_c, best_score = a1, c1, float(s1)
                last_improve = time.monotonic()
        except Exception:
            pass

        # Reset current point to best; occasional random basin jump if stale.
        if time.monotonic() - last_improve > 120.0:
            improved = False
            for attempt in range(6):
                scale = 0.12 + 0.12 * float(attempt)
                try:
                    cand2 = random_block_candidate(best_a, int(seed) + attempt * 53 + 5000, scale=scale)
                    sc2 = internal_score(cand2, Nconv)
                    if np.isfinite(sc2) and sc2 < best_score * (1.0 - 1e-11):
                        best_a, best_c, best_score = cand2, None, float(sc2)
                        best_a, best_c, best_score = convolution_score(best_a, Nconv)
                        improved = True
                        break
                except Exception:
                    pass
            a, c, score = best_a, best_c, best_score
            if not improved:
                last_improve = time.monotonic()

        it += 1
        now = time.monotonic()
        if now - last_print > 45.0:
            print("mirror n={} score={:.10f} lp={}".format(n, best_score, lp_solves), flush=True)
            last_print = now

        if it > 20000 and time.monotonic() - last_improve > 300.0:
            break

    try:
        best_a, best_c, best_score = convolution_score(best_a, Nconv)
    except Exception:
        pass
    return best_a, float(best_score)


def safe_true_eval(a) -> float:
    try:
        a = normalize_positive(a, floor=0.0)
        scale = math.sqrt(2.0 * float(a.size))
        out = [float(max(0.0, min(1000.0, float(x) * scale))) for x in a]
        if sum(out) < 0.01:
            out = [50.0 for _ in out]
        val = float(evaluate_sequence(out))
        return val if np.isfinite(val) else np.inf
    except Exception:
        return np.inf


def scale_for_return(a) -> list:
    a = normalize_positive(a, floor=0.0)
    scale = math.sqrt(2.0 * float(a.size))
    out = [float(max(0.0, min(1000.0, float(x) * scale))) for x in a]
    if sum(out) < 0.01:
        out = [50.0 for _ in out]
    return out


def propose_candidate(budget_s: float = 2400.0):
    start = time.monotonic()
    total = max(120.0, float(budget_s) - 45.0)
    deadline = start + total

    given = get_given_sequence()
    if given is not None and given.size >= 512:
        if given.size <= 4096:
            target_n = int(given.size)
        else:
            target_n = 4096
        best_a = resample_seq(given, target_n)
        print("using given initial length {}".format(target_n), flush=True)
    else:
        target_n = 2048
        best_a = None
        best_score = np.inf
        inits = generate_inits(target_n, 42)
        Nconv0 = next_pow2(2 * target_n - 1)
        for a0 in inits:
            sc = internal_score(a0, Nconv0)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a0
        if best_a is None:
            best_a = make_uniform(target_n)
        print("no given sequence; random init score={:.10f}".format(
            internal_score(best_a, Nconv0)), flush=True)

    try:
        best_score = internal_score(best_a)
        print("initial best score={:.10f} length={}".format(best_score, len(best_a)), flush=True)
    except Exception:
        best_score = np.inf

    # Phase 1: strong local coordinate search at the current length.
    phase1_deadline = min(deadline - 1500.0, start + 1150.0)
    if phase1_deadline > time.monotonic() + 30.0:
        a1, s1 = coordinate_search(best_a, phase1_deadline, seed=1234, label="coord1")
        if np.isfinite(s1) and s1 < best_score:
            best_a, best_score = a1, float(s1)
            print("after coord1 score={:.10f}".format(best_score), flush=True)

    # Phase 2: mirror descent with LP-dual/Frank-Wolfe weights plus coordinate polish.
    phase2_deadline = min(deadline - 600.0, start + max(1900.0, phase1_deadline))
    if phase2_deadline > time.monotonic() + 30.0:
        a2, s2 = mirror_search(best_a, phase2_deadline, seed=777, lp_n=512, max_lp_solves=8)
        if np.isfinite(s2) and s2 < best_score:
            best_a, best_score = a2, float(s2)
            print("after mirror score={:.10f}".format(best_score), flush=True)

    # Phase 3: continue local coordinate search.
    phase3_deadline = deadline - 220.0
    if phase3_deadline > time.monotonic() + 30.0:
        a3, s3 = coordinate_search(best_a, phase3_deadline, seed=555, label="coord3")
        if np.isfinite(s3) and s3 < best_score:
            best_a, best_score = a3, float(s3)
            print("after coord3 score={:.10f}".format(best_score), flush=True)

    # Phase 4: optional longer sequence if there is a comfortable cushion.
    longer_a = None
    longer_score = np.inf
    if time.monotonic() < deadline - 300.0:
        try:
            longer_n = 4096 if int(best_a.size) < 4096 else 8192
            longer_seed = 24601
            cand4 = resample_seq(best_a, longer_n)
            if time.monotonic() < deadline - 160.0:
                cand4, long_s4 = random_block_polish(cand4, deadline - 80.0, seed=longer_seed)
            longer_score = internal_score(cand4)
            longer_a = cand4
            print("longer candidate n={} internal={:.10f}".format(longer_n, longer_score), flush=True)
        except Exception as exc:
            print("longer phase failed: {}".format(exc), flush=True)

    # Final random block polish on main best.
    if time.monotonic() < deadline - 45.0:
        try:
            a5, s5 = random_block_polish(best_a, deadline - 35.0, seed=99991)
            if np.isfinite(s5) and s5 < best_score:
                best_a, best_score = a5, float(s5)
                print("after final polish score={:.10f}".format(best_score), flush=True)
        except Exception:
            pass

    # Choose between candidates using the real evaluator when possible.
    candidates = [best_a]
    if given is not None:
        candidates.append(given)
    if longer_a is not None and np.isfinite(longer_score):
        candidates.append(longer_a)
    candidates.append(make_uniform(2048))

    final_a = best_a
    best_true = np.inf
    for cand in candidates:
        if cand is None:
            continue
        try:
            cand = normalize_positive(cand, floor=0.0)
            tru = safe_true_eval(cand)
            if np.isfinite(tru) and tru < best_true:
                best_true = tru
                final_a = cand
        except Exception:
            pass

    final_a = normalize_positive(final_a, floor=0.0)
    out = scale_for_return(final_a)

    # As a last safeguard, if the evaluator rejects the output, return a simple valid sequence.
    if not out:
        out = scale_for_return(make_uniform(1024))

    try:
        final_check = float(evaluate_sequence(out))
        print("returned length={} evaluated={:.12f}".format(len(out), final_check), flush=True)
    except Exception:
        fallback = scale_for_return(make_uniform(1024))
        try:
            print("fallback length={} evaluated={:.12f}".format(
                len(fallback), float(evaluate_sequence(fallback))), flush=True)
        except Exception:
            pass
        out = fallback

    return out
