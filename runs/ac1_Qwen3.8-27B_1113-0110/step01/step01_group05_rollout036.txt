"""
Gradient-based optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 with a >= 0.
Uses log-sum-exp annealing, Adam optimizer, simplex projection, and coordinate descent polish.
"""

import numpy as np
import time

def _clean(arr):
    a = np.asarray(arr, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    np.clip(a, 0.0, 1000.0, out=a)
    return a

def _normalize(a):
    a = _clean(a)
    s = a.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones_like(a) / len(a)
    return a / s

def _score(a):
    n = len(a)
    if n == 0:
        return float('inf')
    c = np.convolve(a, a)
    mx = c.max()
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2.0 * n * mx / (s * s))

def _softmax_t(c, T):
    """Compute softmax(T * c) with numerical stability."""
    x = T * c
    x -= x.max()
    e = np.exp(x)
    return e / e.sum()

def _gradient(a, T):
    """Compute gradient of the smooth objective with respect to a."""
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    w = _softmax_t(c, T)
    # h[i] = sum_j w[i+j] * a[j] = cross-correlation
    h = np.correlate(w, a, mode='valid')  # shape (n,)
    M_T = float(np.dot(w, c))  # = sum_k w[k] * c[k]
    # Objective: f = 2n * M_T / s^2
    # df/da[i] = 2n * (h[i] * s - 2 * M_T) / s^3
    g = 2.0 * n * (h * s - 2.0 * M_T) / (s ** 3)
    return g, c, M_T

def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _adam_update(state, grad, lr, beta1=0.9, beta2=0.999, t=1):
    """Adam optimizer update."""
    if state['m'] is None:
        state['m'] = np.zeros_like(grad)
        state['v'] = np.zeros_like(grad)
    state['t'] = t
    b1 = 1 - beta1 ** t
    b2 = 1 - beta2 ** t
    state['m'] = beta1 * state['m'] + (1 - beta1) * grad
    state['v'] = beta2 * state['v'] + (1 - beta2) * grad ** 2
    m_hat = state['m'] / b1
    v_hat = state['v'] / b2

def _optimize_at_T(a, T, steps, lr, rng, noise_base):
    """Run Adam optimization at a fixed temperature T."""
    n = a.size

    for _ in range(steps):
        t += 1
        c = np.convolve(a, a)
        c_shift = c - c.max()
        w = np.exp(T * c_shift)
        wsum = w.sum()

        w /= wsum

        ha = float(h @ a)

        v = beta2 * v + (1.0 - beta2) * g * g

        vhat = v / b2
