```
def _smooth_core(z, tau, n):

    z = np.asarray(z, dtype=np.float64)

    z = z - np.max(z)

    p = np.exp(z)

    if s <=0: p = np.ones(n)/n
    else: p /= s
    L = 1 << ((2*n-1)-1).bit_length()
    A = np.fft.rfft(p, L)
    c = np.fft.irfft(A*A, L)[:2*n-1]
    cmax = float(np.max(c))
```

```
"""Search for non-negative step..."""
import numpy as np
import time
from scipy.optimize import linprog, minimize
from scipy.linalg import toeplitz

_CACHE = {"n":-1, "tau":None, "z":None, "h":None, "f":None, "g":None}

def _fft_convolve(a):
   a=np.asarray(a,dtype=np.float64).ravel()
   n=a.size
   if n==0: return np.array([0.0])
   size=2*n-1
   L=1 << (size-1).bit_length()
   A=np.fft.rfft(a, L)
   c=np.fft.irfft(A*A, L)[:size]
   # zero tiny
   c = np.where(np.abs(c) < 1e-15, 0.0, c)
```

```python
"""
Search for a non-negative sequence minimizing
    J(a) = 2*n*max(convolve(a,a)) / sum(a)^2.

The implementation uses a hybrid:
  1) alternating linear-programming best-response iterations;
  2) smoothed L-BFGS minimization in log-softmax coordinates;
  3) multi-scale enlarge/shrink, random restarts, and final polish.
"""

import numpy as np
import time

from scipy.optimize import linprog, minimize
from scipy.linalg import toeplitz

# Small cache so L-BFGS does not recompute the gradient immediately after the objective.
_CACHE = {"n": -1, "tau": None, "h": None, "f": None, "g": None}


def _fft_convolve(a):
    """Full linear convolution a * a via real FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)[:size]
    # Convolution of nonnegative data should be nonnegative; remove tiny FFT noise.
    c = np.where(np.abs(c) < 1e-15, 0.0, c)
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    """Objective J(a), scale invariant."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return float("inf")
    a = np.clip(a, 0.0, 1000.0)
    a = np.where(np.isfinite(a), a, 0.0)
    s = float(np.sum(a))
    if s < 0.01:
        return float("inf")
    c = _fft_convolve(a)
    if c.size == 0:
        return float("inf")
    mc = float(np.max(c))
    if not np.isfinite(mc):
        return float("inf")
    return float(2.0 * a.size * mc / (s * s))


def _score_direct(a):
    """Direct score using np.convolve, useful for exact final verification."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return float("inf")
    a = np.clip(a, 0.0, 1000.0)
    a = np.where(np.isfinite(a), a, 0.0)
    s = float(np.sum(a))
    if s < 0.01:
        return float("inf")
    c = np.convolve(a, a)
    mc = float(np.max(c))
    if not np.isfinite(mc):
        return float("inf")
    return float(2.0 * a.size * mc / (s * s))


def _normalize(a):
    """Clip to nonnegative and normalize to unit sum."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = np.where(np.isfinite(a), a, 0.0)
    s = float(np.sum(a))
    if s <= 1e-12:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def _interpolate_sequence(src, n_target):
    """Resample a sequence to a new length and normalize."""
    src = np.asarray(src, dtype=np.float64).ravel()
    if n_target <= 0:
        return np.array([1.0], dtype=np.float64)
    if src.size == 0:
        return np.ones(n_target, dtype=np.float64)
    if src.size == 1:
        return np.ones(n_target, dtype=np.float64)
    if src.size == n_target:
        out = src.copy()
    else:
        x = np.linspace(0.0, 1.0, src.size)
        y = np.linspace(0.0, 1.0, n_target)
        out = np.interp(y, x, src)
    return _normalize(out)


def _make_initial(n, variant, seed=0):
    """Several asymmetric nonnegative starting profiles."""
    rng = np.random.default_rng(seed)
    if n <= 0:
        return np.array([1.0], dtype=np.float64)
    if n == 1:
        return np.array([1.0], dtype=np.float64)

    t = np.linspace(0.0, 1.0, n)
    v = int(variant) % 16

    if v == 0:
        a = np.ones(n, dtype=np.float64)
    elif v == 1:
        a = np.exp(-1.5 * t)
    elif v == 2:
        a = np.exp(-3.0 * t)
    elif v == 3:
        a = np.exp(-5.0 * t)
    elif v == 4:
        a = (1.0 - t) ** 1.0
    elif v == 5:
        a = (1.0 - t) ** 2.5
    elif v == 6:
        a = 1.0 / (1.0 + 2.0 * t)
    elif v == 7:
        a = 1.0 / (1.0 + 8.0 * t)
    elif v == 8:
        a = np.exp(-((t - 0.15) ** 2) / 0.01) + 0.4 * np.exp(-((t - 0.70) ** 2) / 0.03)
    elif v == 9:
        a = np.exp(-((t - 0.70) ** 2) / 0.01) + 0.4 * np.exp(-((t - 0.20) ** 2) / 0.03)
    elif v == 10:
        a = t ** 0.35 + 1e-12
    elif v == 11:
        a = np.abs(rng.normal(0.0, 1.0, n))
    elif v == 12:
        a = rng.exponential(1.0, n)
    elif v == 13:
        # Sparse random positive distribution; alpha kept modest for numerical safety.
        a = rng.dirichlet(0.35 * np.ones(n, dtype=np.float64))
    elif v == 14:
        r = np.cumsum(rng.normal(0.0, 1.0, n))
        a = np.maximum(r, 0.0)
    elif v == 15:
        a = np.maximum(rng.lognormal(0.0, 1.0, n), 0.0)
    else:
        a = np.ones(n, dtype=np.float64)

    return _normalize(a)


def _softmax_from_z(z):
    """Map unconstrained z to a probability simplex point with a tiny floor."""
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return np.array([1.0], dtype=np.float64)
    if not np.all(np.isfinite(z)):
        return np.ones(z.size, dtype=np.float64) / z.size

    z = z - float(np.max(z))
    # Floor at exp(-60) is negligible for the score but prevents exact zeros / zero gradients.
    z = np.clip(z, -60.0, 0.0)
    p = np.exp(z)
    s = float(np.sum(p))
    if s <= 1e-300 or not np.isfinite(s):
        return np.ones(z.size, dtype=np.float64) / z.size
    return p / s


def _geometric_blend(f, g, u):
    """Geometric (log-linear) blend of two probability vectors."""
    f = np.asarray(f, dtype=np.float64).ravel()
    g = np.asarray(g, dtype=np.float64).ravel()
    if f.size != g.size or f.size == 0:
        return _normalize(f if f.size else g)

    f_safe = np.maximum(f, 1e-15)
    g_safe = np.maximum(g, 1e-15)
    z = (1.0 - u) * np.log(f_safe) + u * np.log(g_safe)
    return _softmax_from_z(z)


def _z_hash(z):
    """Cheap hash-like summary for the cache."""
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return (0.0, 0.0, 0.0)
    zs = np.clip(z, -1e6, 1e6)
    return (
        float(np.max(zs)),
        float(np.sum(zs)),
        float(np.sum(zs * zs)),
    )


def _smooth_core(z, tau, n):
    """
    Smoothed objective:
        J_smooth = 2*n*softmax_tau(conv(a,a))
    and its gradient with respect to z, where a = softmax(z).
    """
    if n <= 0:
        return float("inf"), np.zeros(0, dtype=np.float64)

    p = _softmax_from_z(z)
    if p.size != n:
        p = np.ones(n, dtype=np.float64) / n

    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()

    A = np.fft.rfft(p, L)
    c = np.fft.irfft(A * A, L)[:size]
    c = np.maximum(c, 0.0)

    cmax = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(tau) or tau <= 0.0:
        tau = 1e-8

    scaled = (c - cmax) / tau
    e = np.exp(scaled)
    sume = float(np.sum(e))
    if sume <= 1e-300 or not np.isfinite(sume):
        w = np.zeros(size, dtype=np.float64)
        idx = int(np.argmax(c))
        w[idx] = 1.0
        m = cmax
    else:
        w = e / sume
        m = cmax + tau * np.log(sume)

    value = float(2.0 * n * m)

    # Gradient of soft-max w.r.t. p:
    #   d M / d p_j = 2 * sum_k w_k p_{k-j}
    # so d J / d p_j = 4*n * sum_k w_k p_{k-j}.
    # The cross-correlation v_j = sum_k w_k p_{j+k} is computed by W * conj(A).
    W = np.fft.rfft(w, L)
    v = np.fft.irfft(W * np.conj(A), L)[:n]
    v = np.maximum(v, 0.0)

    grad_p = (4.0 * float(n)) * v
    mean_grad = float(np.dot(p, grad_p))
    grad_z = p * (grad_p - mean_grad)

    if not np.all(np.isfinite(grad_z)):
        grad_z = np.zeros(n, dtype=np.float64)
        value = float("inf")

    return float(value), grad_z


def _smooth_fun(z, tau, n):
    f, g = _smooth_core(z, tau, n)
    _CACHE["n"] = n
    _CACHE["tau"] = tau
    _CACHE["h"] = _z_hash(z)
    _CACHE["f"] = f
    _CACHE["g"] = g
    return f


def _smooth_grad(z, tau, n):
    zarr = np.asarray(z, dtype=np.float64).ravel()
    use_cache = (
        _CACHE["n"] == n and
        _CACHE["tau"] == tau and
        _CACHE["h"] is not None
    )
    if use_cache:
        h = _z_hash(zarr)
        try:
            use_cache = bool(np.allclose(h, _CACHE["h"], rtol=1e-10, atol=1e-12))
        except Exception:
            use_cache = False
    if use_cache:
        if np.all(np.isfinite(_CACHE["g"])):
            return _CACHE["g"]

    f, g = _smooth_core(zarr, tau, n)
    _CACHE["n"] = n
    _CACHE["tau"] = tau
    _CACHE["h"] = _z_hash(zarr)
    _CACHE["f"] = f
    _CACHE["g"] = g
    return g


def _smooth_polish(a, deadline, maxiter=200,
                   fracs=(0.12, 0.06, 0.03, 0.015, 0.008),
                   seed=0, perturb=0.02):
    """
    Polished local minimization using a smooth max approximated by softmax_tau.
    Returns the best exact sequence found during this polish.
    """
    a = _normalize(np.asarray(a, dtype=np.float64))
    n = a.size
    best_a = a.copy()
    best_score = _score(best_a)

    if n == 0:
        return best_a, best_score

    rng = np.random.default_rng(99991 + int(seed))

    try:
        for idx, tau_frac in enumerate(fracs):
            if time.monotonic() >= deadline:
                break

            c = _fft_convolve(a)
            m0 = float(np.max(c)) if c.size else 0.0
            if m0 <= 0.0 or not np.isfinite(m0):
                m0 = 1e-12
            tau = max(1e-12, float(tau_frac) * float(m0))

            z = np.log(np.maximum(a, 1e-12))
            z = z - float(np.max(z))
            if perturb > 0.0:
                z = z + rng.normal(0.0, float(perturb), n)

            _CACHE["n"] = -1
            _CACHE["tau"] = None
            _CACHE["h"] = None
            _CACHE["f"] = None
            _CACHE["g"] = None

            iters = max(40, int(maxiter) - 15 * idx)
            options = {
                "maxiter": iters,
                "gtol": 1e-8,
                "ftol": 1e-13,
                "maxfun": max(100, 4 * iters),
                "maxcor": 20,
                "maxls": 20,
            }

            try:
                res = minimize(
                    _smooth_fun,
                    z,
                    args=(tau, n),
                    jac=_smooth_grad,
                    method="L-BFGS-B",
                    options=options,
                )
            except Exception as ee:
                print(f"    smooth L-BFGS exception: {ee}")
                continue

            if res is not None and getattr(res, "x", None) is not None:
                z_new = np.asarray(res.x, dtype=np.float64).ravel()
                if z_new.size == n and np.all(np.isfinite(z_new)):
                    new_a = _softmax_from_z(z_new)
                    new_score = _score(new_a)
                    if np.isfinite(new_score) and new_score < best_score - 1e-12:
                        best_score = new_score
                        a = new_a
                        best_a = new_a.copy()
                        print(f"      smooth[{n}] tau_frac={tau_frac}: best={best_score:.7f}")
    except Exception as ee:
        print(f"    smooth polish error: {ee}")

    return best_a, best_score


def _best_response(f, M):
    """
    LP best response:
        maximize sum(g)
        subject to conv(f, g) <= M, g >= 0.
    Returns (probability_vector_g, sum_g). If no improvement, g is None.
    """
    f = np.asarray(f, dtype=np.float64).ravel()
    n = f.size
    if n == 0 or M is None or M <= 1e-15 or not np.isfinite(M):
        return None, 1.0

    m = 2 * n - 1
    first_col = np.zeros(m, dtype=np.float64)
    first_row = np.zeros(m, dtype=np.float64)
    first_col[:n] = f
    first_row[0] = f[0]

    try:
        A = toeplitz(first_col, first_row)
    except Exception:
        return None, 1.0

    c_obj = -np.ones(n, dtype=np.float64)
    b_ub = M * np.ones(m, dtype=np.float64)

    try:
        res = linprog(
            c_obj,
            A_ub=A,
            b_ub=b_ub,
            bounds=(0.0, None),
            method="highs",
            options={"presolve": True},
        )
    except Exception:
        return None, 1.0

    if res is None:
        return None, 1.0

    if res.status == 0 and getattr(res, "x", None) is not None:
        g = np.asarray(res.x, dtype=np.float64).ravel()
        g = np.maximum(g, 0.0)
        g = np.where(np.isfinite(g), g, 0.0)
        gs = float(np.sum(g))
        if gs > 1e-12:
            # LP objective is -sum(g), so S = -res.fun.
            S = float(-res.fun)
            if not np.isfinite(S) or S <= 0.0:
                S = gs
            return g / gs, S

    return None, 1.0


def _alternating_lp(f, deadline, max_iters=2000, seed=1, verbose=True):
    """
    Alternating best-response + line-searched geometric/linear blends.
    """
    f = _normalize(np.asarray(f, dtype=np.float64))
    n = f.size
    if n == 0:
        return np.array([1.0], dtype=np.float64), float("inf")

    cur = f.copy()
    cur_score = _score(cur)
    best = cur.copy()
    best_score = cur_score
    stall = 0
    last_print = -100
    rng = np.random.default_rng(50000 + int(seed))

    if n > 768:
        geom_u = (1.0, 0.6, 0.4, 0.25)
        lin_u = (0.5, 0.3)
    else:
        geom_u = (1.0, 0.75, 0.55, 0.4, 0.25, 0.15)
        lin_u = (0.6, 0.35, 0.2)

    for it in range(int(max_iters)):
        if time.monotonic() >= deadline:
            break

        c_cur = _fft_convolve(cur)
        M = float(np.max(c_cur)) if c_cur.size else 0.0
        if M <= 0.0 or not np.isfinite(M):
            break

        g, S = _best_response(cur, M)
        if g is None:
            break

        if S <= 1.0 + 1e-7:
            stall += 1
        else:
            g = _normalize(g)
            cand = cur.copy()
            cand_score = cur_score

            for u in geom_u:
                if time.monotonic() >= deadline:
                    break
                h = _geometric_blend(cur, g, u)
                sc = _score(h)
                if np.isfinite(sc) and sc < cand_score - 1e-12:
                    cand_score = sc
                    cand = h.copy()

            for u in lin_u:
                if time.monotonic() >= deadline:
                    break
                h = _normalize((1.0 - u) * cur + u * g)
                sc = _score(h)
                if np.isfinite(sc) and sc < cand_score - 1e-12:
                    cand_score = sc
                    cand = h.copy()

            if cand_score < cur_score - 1e-12:
                cur = cand
                cur_score = cand_score
                stall = 0
            else:
                stall += 1

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = cur.copy()
            if verbose and (it - last_print >= 25):
                print(f"      ALP[{n}] iter {it}: best={best_score:.7f}")
                last_print = it

        if stall > 35:
            # Mild restart: mix with a random aura to escape a shallow ALP basin.
            p = _make_initial(n, 11, 777 + it)
            mix = _normalize(0.85 * cur + 0.15 * p)
            sc_mix = _score(mix)
            if np.isfinite(sc_mix) and sc_mix < best_score + 0.03:
                cur = mix
                cur_score = sc_mix
                stall = 15
            else:
                break

    return best, best_score


def propose_candidate():
    """
    Multi-phase search for the best sequence within the wall-clock budget.
    """
    start = time.monotonic()
    hard_deadline = start + 2300.0  # leave margin before 2400s.

    best_a = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Load a parent construction if available.
    # ------------------------------------------------------------------
    try:
        parent = globals().get("height_sequence_1", None)
        if parent is not None:
            arr = np.asarray(parent, dtype=np.float64).ravel()
            if arr.size > 0 and np.all(np.isfinite(arr)):
                arr = np.clip(arr, 0.0, 1000.0)
                if arr.size > 2048:
                    arr = _interpolate_sequence(arr, 2048)
                else:
                    arr = _normalize(arr)
                s0 = _score(arr)
                if np.isfinite(s0):
                    best_a = arr
                    best_score = s0
                    print(f"Parent sequence: n={best_a.size}, score={best_score:.7f}")
    except Exception as ee:
        print(f"Could not load parent: {ee}")

    if best_a is None:
        best_a = _make_initial(512, 2, seed=1)
        best_score = _score(best_a)
        print(f"Synthetic start: n={best_a.size}, score={best_score:.7f}")

    # ------------------------------------------------------------------
    # Quick polish of the parent before major LP work.
    # ------------------------------------------------------------------
    t_now = time.monotonic()
    quick_end = min(hard_deadline, start + 60.0)
    if best_a.size <= 768 and t_now < quick_end:
        try:
            a_aug, s_aug = _smooth_polish(
                best_a,
                quick_end,
                maxiter=80,
                fracs=(0.10, 0.05, 0.02),
                seed=11,
                perturb=0.01,
            )
            if s_aug < best_score:
                best_a = a_aug
                best_score = s_aug
                print(f"After quick smooth: n={best_a.size}, score={best_score:.7f}")
        except Exception as ee:
            print(f"Quick smooth failed: {ee}")

    # ------------------------------------------------------------------
    # Main ALP at n=512 from several starts.
    # ------------------------------------------------------------------
    alp512_end = min(hard_deadline, start + 660.0)
    if time.monotonic() < alp512_end:
        n = 512
        inits = []
        labels = []

        if best_a is not None:
            inits.append(_interpolate_sequence(best_a, n))
            labels.append("best")

        for v in [2, 4, 7, 11, 12]:
            inits.append(_make_initial(n, v, seed=100 + v))
            labels.append(f"v{v}")

        print(f"\n=== ALP phase n={n} ===")
        remaining = max(0.0, alp512_end - time.monotonic())
        per = max(45.0, remaining / max(1, len(inits)))

        for label, init in zip(labels, inits):
            if time.monotonic() >= alp512_end:
                break
            budget = per
            if label == "best":
                budget = min(remaining, max(160.0, per * 1.6))
            deadline_lp = min(alp512_end, time.monotonic() + budget)
            if deadline_lp <= time.monotonic():
                continue

            print(f"  ALP start label={label}, budget={budget:.1f}s")
            try:
                a_opt, sc = _alternating_lp(init, deadline_lp, max_iters=5000, seed=n, verbose=True)
            except Exception as ee:
                print(f"  ALP {label} failed: {ee}")
                continue

            if sc < best_score:
                best_a = a_opt.copy()
                best_score = sc
                print(f"  ** ALP {label} new best: {best_score:.7f} (n={best_a.size})")

            # Short smooth follow-up.
            if time.monotonic() < hard_deadline - 900.0:
                try:
                    a_s, sc_s = _smooth_polish(
                        a_opt,
                        min(alp512_end, time.monotonic() + 18.0),
                        maxiter=60,
                        fracs=(0.05, 0.02),
                        seed=200 + n,
                        perturb=0.008,
                    )
                    if sc_s < best_score:
                        best_a = a_s.copy()
                        best_score = sc_s
                        print(f"  ** smooth after ALP {label}: {best_score:.7f}")
                except Exception as ee:
                    print(f"  smooth after ALP {label} failed: {ee}")

    # ------------------------------------------------------------------
    # ALP at n=768.
    # ------------------------------------------------------------------
    alp768_end = min(hard_deadline, start + 910.0)
    if best_a is not None and time.monotonic() < alp768_end:
        n = 768
        print(f"\n=== ALP phase n={n} ===")
        inits = [_interpolate_sequence(best_a, n)]
        inits.append(_make_initial(n, 8, seed=808))

        remaining = max(0.0, alp768_end - time.monotonic())
        budgets = [max(80.0, remaining * 0.65), max(30.0, remaining * 0.35)]

        for i, init in enumerate(inits):
            if time.monotonic() >= alp768_end:
                break
            budget = budgets[i] if i < len(budgets) else 30.0
            deadline_lp = min(alp768_end, time.monotonic() + budget)
            if deadline_lp <= time.monotonic():
                continue
            try:
                a_opt, sc = _alternating_lp(init, deadline_lp, max_iters=3000, seed=n + i, verbose=True)
            except Exception as ee:
                print(f"  ALP 768 {i} failed: {ee}")
                continue
            if sc < best_score:
                best_a = a_opt.copy()
                best_score = sc
                print(f"  ** ALP 768 new best: {best_score:.7f}")

            if time.monotonic() < hard_deadline - 650.0:
                try:
                    a_s, sc_s = _smooth_polish(
                        a_opt,
                        min(alp768_end, time.monotonic() + 18.0),
                        maxiter=60,
                        fracs=(0.04, 0.02),
                        seed=300 + n,
                        perturb=0.008,
                    )
                    if sc_s < best_score:
                        best_a = a_s.copy()
                        best_score = sc_s
                        print(f"  ** smooth after ALP 768: {best_score:.7f}")
                except Exception as ee:
                    print(f"  smooth after ALP 768 failed: {ee}")

    # ------------------------------------------------------------------
    # ALP at n=1024 (less iterations due to larger LP).
    # ------------------------------------------------------------------
    alp1024_end = min(hard_deadline, start + 1160.0)
    if best_a is not None and time.monotonic() < alp1024_end:
        n = 1024
        print(f"\n=== ALP phase n={n} ===")
        init = _interpolate_sequence(best_a, n)
        try:
            a_opt, sc = _alternating_lp(init, alp1024_end, max_iters=1500, seed=n, verbose=True)
            if sc < best_score:
                best_a = a_opt.copy()
                best_score = sc
                print(f"  ** ALP 1024 new best: {best_score:.7f}")
        except Exception as ee:
            print(f"  ALP 1024 failed: {ee}")

        if time.monotonic() < hard_deadline - 500.0:
            try:
                a_s, sc_s = _smooth_polish(
                    best_a,
                    min(alp1024_end, time.monotonic() + 25.0),
                    maxiter=80,
                    fracs=(0.03, 0.015),
                    seed=400 + n,
                    perturb=0.006,
                )
                if sc_s < best_score:
                    best_a = a_s.copy()
                    best_score = sc_s
                    print(f"  ** smooth after ALP 1024: {best_score:.7f}")
            except Exception as ee:
                print(f"  smooth after ALP 1024 failed: {ee}")

    # ------------------------------------------------------------------
    # Large scale smooth search at n=1024 / 1536 / 2048.
    # ------------------------------------------------------------------
    smooth_end = min(hard_deadline, hard_deadline - 250.0)

    large_ns = [1024, 1536, 2048]
    alloc = [320, 220, 180]

    for n, b in zip(large_ns, alloc):
        if time.monotonic() >= smooth_end - 30:
            break
        if best_a is None:
            init = _make_initial(n, 2, seed=n)
        else:
            init = _interpolate_sequence(best_a, n)

        print(f"\n=== Large smooth phase n={n} ===")
        deadline_this = min(smooth_end, time.monotonic() + float(b))
        if deadline_this <= time.monotonic():
            continue

        try:
            a_opt, sc = _smooth_polish(
                init,
                deadline_this,
                maxiter=150 if n <= 1024 else 110,
                fracs=(0.10, 0.05, 0.025, 0.012),
                seed=500 + n,
                perturb=0.012,
            )
            if sc < best_score:
                best_a = a_opt.copy()
                best_score = sc
                print(f"  ** large smooth n={n} new best: {best_score:.7f}")

            # If there is time, refocus with a smaller smoothing parameter.
            if time.monotonic() < smooth_end - 60:
                a_opt2, sc2 = _smooth_polish(
                    best_a,
                    min(smooth_end - 30, time.monotonic() + 80.0),
                    maxiter=80,
                    fracs=(0.012, 0.006),
                    seed=600 + n,
                    perturb=0.004,
                )
                if sc2 < best_score:
                    best_a = a_opt2.copy()
                    best_score = sc2
                    print(f"  ** refocused smooth n={n} new best: {best_score:.7f}")
        except Exception as ee:
            print(f"  large smooth n={n} failed: {ee}")

    # ------------------------------------------------------------------
    # Final down-polish: optimize a smaller surrogate, then restore length.
    # ------------------------------------------------------------------
    final_end = hard_deadline - 30.0
    if best_a is not None and time.monotonic() < final_end:
        n_best = best_a.size
        n_ds = int(min(512, max(256, n_best // 2)))

        if n_best != n_ds:
            a_ds = _interpolate_sequence(best_a, n_ds)
            deadline_ds = min(final_end, time.monotonic() + 80.0)
            try:
                a_lp, sc_lp = _alternating_lp(a_ds, deadline_ds, max_iters=2000, seed=6100, verbose=True)
                a_up = _interpolate_sequence(a_lp, n_best)
                sc_up = _score(a_up)
                if sc_up < best_score:
                    best_a = a_up.copy()
                    best_score = sc_up
                    print(f"  ** down/up ALP new best: {best_score:.7f}")
            except Exception as ee:
                print(f"  final down/up ALP failed: {ee}")

        # Final pure smooth polish on the current best length.
        if time.monotonic() < final_end - 20:
            max_it_final = 120 if best_a.size <= 1024 else 80
            try:
                a_fin, sc_fin = _smooth_polish(
                    best_a,
                    final_end,
                    maxiter=max_it_final,
                    fracs=(0.015, 0.008, 0.004),
                    seed=7100,
                    perturb=0.0,
                )
                if sc_fin < best_score:
                    best_a = a_fin.copy()
                    best_score = sc_fin
                    print(f"  ** final smooth new best: {best_score:.7f}")
            except Exception as ee:
                print(f"  final smooth failed: {ee}")

    # ------------------------------------------------------------------
    # Safety: ensure a reasonable length and valid output.
    # ------------------------------------------------------------------
    if best_a is None or best_a.size == 0 or not np.all(np.isfinite(best_a)):
        best_a = np.ones(128, dtype=np.float64) / 128
        best_score = _score(best_a)

    if best_a.size > 2048:
        small = _interpolate_sequence(best_a, 1536)
        small_score = _score(small)
        if np.isfinite(small_score):
            best_a = small
            best_score = small_score

    best_a = _normalize(best_a)
    best_score = _score(best_a)

    if best_a.size < 2:
        best_a = np.ones(128, dtype=np.float64) / 128
        best_score = _score(best_a)

    # Optional exact verification using the actual evaluator if available.
    result = [float(float(x)) for x in best_a]
    try:
        evaluated = float(evaluate_sequence(result))
        if np.isfinite(evaluated):
            print(f"\nFinal verified evaluate_sequence: {evaluated:.7f} (internal score={best_score:.7f}, n={len(result)})")
        else:
            print(f"\nFinal internal score={best_score:.7f}, n={len(result)}; evaluator returned non-finite")
    except Exception:
        print(f"\nFinal internal score={best_score:.7f}, n={len(result)}")

    return result
```
