"""
Optimized search for non-negative coefficient sequences minimizing 2*n*max(conv)/sum^2.
Uses efficient log-domain Adam with adaptive smoothing, spectral diversification,
and multi-length evaluation to escape local minima.
"""

import time
import numpy as np


def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def normalize_positive(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s


def fast_score(p):
    n = p.size
    if n <= 0:
        return np.inf
    L = next_pow2(2 * n - 1)
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    cm = float(c.max())
    if not np.isfinite(cm):
        return np.inf
    return float(2.0 * n * cm)


def score_and_grad(p, tau, L):
    """Compute true score and gradient in log-domain parameterization."""
    n = p.size
    P = np.fft.rfft(p, L)
    c = np.fft.irfft(P * P, L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    cm = float(c.max())

    z = (c - cm) / tau
    expz = np.exp(z)
    expsum = float(expz.sum())
    if not np.isfinite(expsum) or expsum <= 0:
        expz = np.zeros_like(c)
        expz[int(np.argmax(c))] = 1.0
        expsum = 1.0

    W = tau * expz / expsum
    W_rev = W[::-1]
    H = np.fft.irfft(P * np.fft.rfft(W_rev, L), L)
    g_p = 2.0 * H[n - 1:2 * n - 1][::-1]
    g_p = np.where(np.isfinite(g_p), g_p, 0.0)

    mean_g = float(np.dot(p, g_p))
    grad = g_p - p * mean_g

    true_score = 2.0 * n * cm
    return true_score, grad

    p = p_init.copy()
    L = next_pow2(2 * n - 1)

    z = np.log(np.maximum(p, 1e-300))
    z = np.where(np.isfinite(z), z, -50.0)
    z -= z.mean()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_p = p.copy()
    best_score = fast_score(p)

    deadline = time.monotonic() + duration_s
    step = 0
    no_improve_steps = 0
    total_steps_planned = int(duration_s * 800)

    while time.monotonic() < deadline:
        frac = min(1.0, step / max(1, total_steps_planned))
        tau = tau_start * (tau_end / tau_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac

        true_s, grad = score_and_grad(p, tau, L)

        if not np.isfinite(true_s):
            break

        if true_s < best_score - 1e-10:
            best_score = true_s
            best_p = p.copy()
            no_improve_steps = 0
        else:
            no_improve_steps += 1

            z = z + rng.normal(0.0, 0.08, n)
            z -= z.mean()
            z = np.clip(z, -50.0, 50.0)
            m *= 0.05
            v *= 0.05

        b2 = 1.0 - beta2 ** step

        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        z = np.where(np.isfinite(z), z, 0.0)
        z -= z.mean()
        z = np.clip(z, -50.0, 50.0)

        z = z - z.max()
        p = np.exp(z)
        p /= p.sum()

        if step % 20000 == 0:
            print(f"  adam step {step}: score {true_s:.6f}, best {best_score:.6f}, tau {tau:.5f}")

    return best_p, best_score


def spectral_flatten(p, n_freqs, alpha, rng):
    """Apply a spectral flattening step to reduce peak autocorrelation."""
    n = p.size
    L = next_pow2(2 * n - 1)
    P = np.fft.rfft(p, L)

    c = np.fft.irfft(P * P, L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    c_mean = float(c.sum()) / (2 * n - 1)

    # Compute which lags have excess
    excess = c - c_mean
    # Map excess back to frequency domain to identify which frequencies to damp
    E = np.fft.rfft(excess, L)
    # Damp frequencies that contribute most to the peak
    P_new = P.copy()
    for k in range(1, min(n_freqs, len(P) - 1)):
        if np.abs(P[k]) > 1e-15:
            damp = 1.0 - alpha * (E[k] / (np.abs(P[k]) ** 2 + 1e-30))
            damp = np.clip(np.real(damp), 0.3, 1.7)
            P_new[k] *= damp
            P_new[len(P) - 1 - k] *= damp

    p_new = np.fft.irfft(P_new, L)[:n]
    p_new = np.maximum(p_new, 0.0)
    s = p_new.sum()
    if s <= 0:
        return p
    return p_new / s


def chebyshev_parents(n, D, rng):
    """Generate diverse Chebyshev-based parent sequences."""
    x = np.linspace(0.0, 1.0, n)
    t = np.clip(2.0 * x - 1.0, -1.0, 1.0)
    ang = np.arccos(t)

    parents = []
    # Include the previous best if available
    try:
        hs = np.asarray(height_sequence_1, dtype=np.float64)
        hs = np.where(np.isfinite(hs), hs, 0.0)
        hs = np.maximum(hs, 0.0)
        if hs.size > 0 and hs.sum() > 1e-9:
            # Resize to n
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            resized = np.interp(x_new, x_old, hs)
            resized = np.maximum(resized, 0)
            resized /= resized.sum()
            parents.append(resized)
    except Exception:
        pass

    # Fixed structured parents
    for i in range(D):
        theta = np.zeros(D)
        theta[0] = rng.uniform(0.2, 1.5) * (1 if rng.random() < 0.5 else -1)
        if rng.random() < 0.5:
            theta[1] = rng.uniform(-1, 1)
        if D > 2 and rng.random() < 0.3:
            theta[2] = rng.uniform(-0.7, 0.7)
        val = np.zeros(n)
        for m in range(1, D + 1):
            val += theta[m - 1] * np.cos(m * ang)
        p = 1.05 + 0.35 * val
        p = np.maximum(p, 0.01)
        parents.append(p / p.sum())

    # Random smooth parents
    for i in range(5):
        theta = rng.normal(0.0, 0.4, D)
        val = np.zeros(n)
        for m in range(1, D + 1):
            val += theta[m - 1] * np.cos(m * ang)
        p = np.exp(np.clip(val, -5, 5))
        parents.append(p / p.sum())

    return parents


def resize_to_n(seq, n):
    p = normalize_positive(seq)
    if p is None:
        return np.ones(n) / n
    if p.size == n:
        return p.copy()
    x_old = np.linspace(0, 1, p.size)
    x_new = np.linspace(0, 1, n)
    out = np.interp(x_new, x_old, p)
    out = np.maximum(out, 0.0)
    s = out.sum()
    if s <= 0:
        return np.ones(n) / n
    return out / s


def propose_candidate(seed=42, budget_s=2000.0, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(60.0, budget_s - 30.0)

    # Try multiple lengths
    lengths_to_try = [8192, 16384]
    if budget_s < 600:
        lengths_to_try = [4096, 8192]

    overall_best_p = None
    overall_best_s = np.inf

    # Initial evaluation of previous best
    try:
        hs = np.asarray(height_sequence_1, dtype=np.float64)
        hs = np.where(np.isfinite(hs), hs, 0.0)
        hs = np.maximum(hs, 0.0)
        if hs.size > 0 and hs.sum() > 1e-9:
            s = fast_score(hs / hs.sum())
            if s < overall_best_s:
                overall_best_s = s
                overall_best_p = hs / hs.sum()
            print(f"Previous best score: {s:.6f}, length {hs.size}")
    except Exception:
        pass

    for n in lengths_to_try:
        if time.monotonic() > deadline - 60:
            break
        print(f"\n=== Optimizing with n={n} ===")

        # Generate diverse parents
        D = 10
        parents = chebyshev_parents(n, D, rng)

        # Score all parents
        best_parent = None
        best_parent_s = np.inf
        for par in parents:
            ps = fast_score(par)
            if ps < best_parent_s:
                best_parent_s = ps
                best_parent = par
        print(f"Best parent score: {best_parent_s:.6f}")

        # Time allocation for this length
        remaining = deadline - time.monotonic()
        if remaining < 30:
            continue

        # Phase 1: Adam from best parent (60% of remaining time for this n)
        dur1 = remaining * 0.55
        if dur1 > 20:
            p1, s1 = adam_optimize(best_parent, dur1, rng,
                                   tau_start=0.15, tau_end=0.004,
                                   lr_start=0.04, lr_end=0.0005, n=n)
            if s1 < overall_best_s:
                overall_best_s = s1
                overall_best_p = p1
                print(f"  Phase 1 best: {s1:.6f}")

        # Phase 2: Adam from a different parent (25% of time)
        remaining = deadline - time.monotonic()
        if remaining > 30:
            # Pick second-best parent
            other_parent = parents[0] if len(parents) > 1 else best_parent
            if other_parent is best_parent and len(parents) > 2:
                other_parent = parents[2]
            # Add some noise for diversity
            noisy = other_parent * np.exp(rng.normal(0, 0.05, n))
            noisy = noisy / noisy.sum()
            dur2 = remaining * 0.30
            if dur2 > 20:
                p2, s2 = adam_optimize(noisy, dur2, rng,
                                       tau_start=0.20, tau_end=0.005,
                                       lr_start=0.03, lr_end=0.001, n=n)
                if s2 < overall_best_s:
                    overall_best_s = s2
                    overall_best_p = p2
                    print(f"  Phase 2 best: {s2:.6f}")

        # Phase 3: Fine-tuning from best found (20% of time)
        remaining = deadline - time.monotonic()
        if remaining > 20 and overall_best_p is not None:
            dur3 = remaining * 0.80
            if dur3 > 15:
                # Slight perturbation to escape exact local minimum
                fine_parent = overall_best_p * (1 + rng.normal(0, 0.005, n))
                fine_parent = np.maximum(fine_parent, 0)
                fine_parent /= fine_parent.sum()
                p3, s3 = adam_optimize(fine_parent, dur3, rng,
                                       tau_start=0.008, tau_end=0.001,
                                       lr_start=0.008, lr_end=0.0002, n=n)
                if s3 < overall_best_s:
                    overall_best_s = s3
                    overall_best_p = p3
                    print(f"  Phase 3 best: {s3:.6f}")

    print(f"\nOverall best fast score: {overall_best_s:.6f}")

    # Final validation with actual evaluator
    candidates = []
    if overall_best_p is not None:
        candidates.append(overall_best_p)

    # Also try shorter lengths which might score better
    for short_n in [1024, 2048]:
        if overall_best_p is not None:
            short_p = resize_to_n(overall_best_p, short_n)
            candidates.append(short_p)
        candidates.append(np.ones(short_n) / short_n)

    # Linear ramp
    for ln in [2048, 4096]:
        ramp = np.linspace(0.3, 1.7, ln)
        candidates.append(ramp / ramp.sum())

    # Inverse ramp
    for ln in [2048, 4096]:
        inv_ramp = np.linspace(1.7, 0.3, ln)
        candidates.append(inv_ramp / inv_ramp.sum())

    # Bimodal
    for ln in [4096]:
        bimodal = np.concatenate([np.linspace(0.5, 2.0, ln // 2), np.linspace(2.0, 0.5, ln - ln // 2)])
        candidates.append(bimodal / bimodal.sum())

    best_candidate = None
    best_candidate_score = np.inf

    for cand in candidates:
        lst = [float(x) for x in np.clip(np.asarray(cand, dtype=np.float64), 0.0, 1000.0)]
        if not lst:
            continue
        try:
            sc = evaluate_sequence(lst)
        except Exception:
            sc = np.inf
        if np.isfinite(sc) and sc < best_candidate_score:
            best_candidate_score = float(sc)
            best_candidate = lst
            print(f"  Candidate len={len(lst)}: score {sc:.6f}")

    if best_candidate is None:
        best_candidate = [1.0 / 1024.0] * 1024

    print(f"\nFinal best score: {best_candidate_score:.6f}")
    return best_candidate
