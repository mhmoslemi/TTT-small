"""

import time

import math

def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()

def normalize_positive(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, float(floor), None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s

def convolve_fft(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if N is None:
        N = next_pow2(2 * n - 1)
    else:
        need = next_pow2(2 * n - 1)
        if N < need:
            N = need
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[: 2 * n - 1]

def internal_score(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.inf
    a = a / s
    c = convolve_fft(a, N)
    return float(2.0 * n * float(np.max(c)))

def compute_soft_weights(c, cmax, tau, floor=0.0):
    c = np.asarray(c, dtype=np.float64).ravel()
    L = c.size
    if L == 0 or cmax <= 0 or not np.isfinite(cmax) or tau <= 0:
        return np.full(L, 1.0 / max(1, L), dtype=np.float64)
    z = (c - cmax) / tau
    z = np.clip(z, -50.0, 0.0)
    ez = np.exp(z)
    s = float(ez.sum())
    if not np.isfinite(s) or s <= 0:
        return np.full(L, 1.0 / max(1, L), dtype=np.float64)
    w = ez / s
    if floor > 0:
        w = (1.0 - floor) * w + (floor / L)
    return w

def grad_from_weights(w, fa, N, n):
    w = np.asarray(w, dtype=np.float64).ravel()
    w_rev = w[::-1]
    fw = np.fft.rfft(w_rev, N)
    conv = np.fft.irfft(fa * fw, N)
    return 2.0 * conv[n - 1: 2 * n - 1][::-1]

    it = 0

        c = convolve_fft(a, N)

            best_score = score
            best_a = a.copy()

            break

            stale = 0
            temp = max(0.002, temp * 0.997)

            stale += 1
