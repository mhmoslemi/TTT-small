"""
LP-guided optimization for minimizing 2*n*max(convolve(a,a))/(sum(a))^2.

Primary strategy: Iterate between LP-based direction finding (solve max sum(b)
s.t. conv(f,b) <= M, b>=0) and 1-D line search along the LP direction.
Complement with coordinate descent for polishing and local escape.
Also uses structured perturbations and multi-scale initialization.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse


def _clean_array(arr):
    a = np.asarray(arr, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    np.clip(a, 0.0, 1000.0, out=a)
    return a


def _normalize(a):
    a = _clean_array(a)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(max(1, a.size), dtype=np.float64)
    return a / s


def _score_normalized(p, c):
    n = int(p.size)
    if n <= 0:
        return float('inf')
    mx = float(np.max(c)) if c.size > 0 else 0.0
    if not np.isfinite(mx):
        return float('inf')
    return float(2.0 * n * max(0.0, mx))


def _score_array(p):
    p = _normalize(p)
    c = np.convolve(p, p)
    return _score_normalized(p, c)


def _two_level(n):
    n = int(n)
    if n <= 1:
        return np.ones(max(1, n), dtype=np.float64)
    p = np.zeros(n, dtype=np.float64)
    m = n // 2
    if m > 0:
        p[:m] = (2.0 / 3.0) / float(m)
    rest = n - m
    if rest > 0:
        p[m:] = (1.0 / 3.0) / float(rest)
    s = float(np.sum(p))
    if s > 0.0:
        p /= s
    return p


def _build_conv_matrix(f):
    """Build sparse matrix A where A[k,j] = f[k-j] for valid indices. Shape (2n-1, n)."""
    n = int(f.size)
    col_lengths = np.minimum(n, 2 * n - 1 - np.arange(n, dtype=np.int64))
    total = int(np.sum(col_lengths))

    rows_list = []
    vals_list = []
    for j in range(n):
        L = int(col_lengths[j])
        rows_list.append(np.arange(j, j + L, dtype=np.int64))
        vals_list.append(f[:L])
    rows = np.concatenate(rows_list) if rows_list else np.array([], dtype=np.int64)
    vals = np.concatenate(vals_list) if vals_list else np.array([], dtype=np.float64)
    cols = np.repeat(np.arange(n, dtype=np.int64), col_lengths)
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _solve_lp_direction(f, deadline):
    """Solve LP: max sum(b) s.t. conv(f,b) <= M, b >= 0. Returns normalized b or None."""
    n = int(f.size)
    c_f = np.convolve(f, f)
    M = float(np.max(c_f))
    if M <= 0.0 or not np.isfinite(M):
        return None

    A = _build_conv_matrix(f)
    c_obj = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(
            c=c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
            method='highs', options={'time_limit': max(1.0, deadline - time.monotonic() - 60.0)}
        )
    except Exception:
        return None

    if res is None or not getattr(res, 'success', False):
        return None

    b = np.asarray(res.x, dtype=np.float64)
    np.clip(b, 0.0, None, out=b)
    bs = float(np.sum(b))
    if not np.isfinite(bs) or bs <= 0.0:
        return None
    return b / bs


def _line_search(C_f, C_fg, C_g, n, n_points=500):
    """Find best t in [0,1] minimizing 2n * max_k [(1-t)^2 C_f + 2t(1-t) C_fg + t^2 C_g]."""
    fn = float(2.0 * n)

    def phi(t):
        a1 = (1.0 - t) * (1.0 - t)
        a2 = 2.0 * t * (1.0 - t)
        a3 = t * t
        return fn * float(np.max(a1 * C_f + a2 * C_fg + a3 * C_g))

    ts = np.linspace(0.0, 1.0, n_points)
    vals = np.array([phi(t) for t in ts])
    idx = int(np.argmin(vals))
    best_t = float(ts[idx])
    best_val = float(vals[idx])

    # Refine around best
    lo = float(ts[max(0, idx - 1)])
    hi = float(ts[min(n_points - 1, idx + 1)])
    for _ in range(3):
        if hi - lo < 1e-8:
            break
        ts_fine = np.linspace(lo, hi, 50)
        vals_fine = np.array([phi(t) for t in ts_fine])
        idx_f = int(np.argmin(vals_fine))
        lo = float(ts_fine[max(0, idx_f - 1)])
        hi = float(ts_fine[min(49, idx_f + 1)])
        best_t = float(ts_fine[idx_f])
        best_val = float(vals_fine[idx_f])

    return best_t, best_val


def _coord_update(p, c, score, i, grid_vals, refine_pts):
    """Update coordinate i using grid search + refinement. Returns (p, c, score, improved)."""
    n = int(p.size)
    old = float(p[i])
    if old < 0.0:
        old = 0.0
    D = 1.0 - old
    if D < 0.0:
        D = 0.0

    ks = slice(i, i + n)
    c_k = c[ks].copy()

    # Affected convolution entries: C[k] for k in [i, i+n-1]
    # C_new[k] = C[k] + 2*(x-old)*a[k-i] for k != 2i
    # C_new[k] = C[k] + (x-old)*(x+old) for k == 2i
    aff_idx = np.arange(i, i + n)
    aff_offset = aff_idx - i  # 0 to n-1
    coeff = 2.0 * p[aff_offset]  # coefficient of (x - old) for each affected position
    const = np.zeros(n, dtype=np.float64)
    mid_pos = 2 * i
    if i <= mid_pos < i + n:
        mid_local = mid_pos - i
        coeff[mid_local] = old  # (x-old)*(x+old) = x^2 - old^2; at grid x: x^2 - old^2
        const[mid_local] = 0.0
        # Handle mid specially: C_new[mid] = C[mid] + x^2 - old^2

    # Unaffected max
    max_unaff = 0.0
    if i > 0:
        m = float(np.max(c[:i]))
        if m > max_unaff:
            max_unaff = m
    end = i + n
    if end < 2 * n - 1:
        m = float(np.max(c[end:]))
        if m > max_unaff:
            max_unaff = m

    xs = np.concatenate(([0.0], grid_vals, [old]))
    xs = xs[np.isfinite(xs) & (xs >= 0.0) & (xs <= 1.0e4)]
    if xs.size == 0:
        return p, c, score, False

    # Compute affine term for each x
    # For non-mid positions: c_k[j] + coeff[j] * (x - old)
    # For mid position: c_k[mid_local] + x^2 - old^2
    aff_base = c_k.copy()
    aff_linear = coeff.copy()

    max_aff_arr = np.empty(xs.size, dtype=np.float64)
    for xi in range(xs.size):
        x = float(xs[xi])
        vals_aff = aff_base + aff_linear * (x - old)
        if i <= 2 * i < i + n:
            mid_local = 2 * i - i
            vals_aff[mid_local] = c_k[mid_local] + x * x - old * old
        ma = float(np.max(vals_aff))
        max_aff_arr[xi] = ma

    maxc = np.maximum(max_unaff, max_aff_arr)
    S = D + xs
    vals = np.full(xs.size, float('inf'), dtype=np.float64)
    valid = S > 1e-15
    vals[valid] = (2.0 * n) * maxc[valid] / (S[valid] * S[valid])

    tol = 1e-12 * max(1.0, score)
    idx = int(np.argmin(vals))
    best_x = float(xs[idx])
    best_val = float(vals[idx])

    if not np.isfinite(best_val) or best_val >= score - tol:
        return p, c, score, False

    # Refinement
    if refine_pts > 1 and xs.size > 1:
        left = 0.0 if idx == 0 else float(xs[idx - 1])
        right = float(xs[idx]) * 1.5 if idx == xs.size - 1 else float(xs[idx + 1])
        if best_x > 0:
            local = best_x * np.logspace(-1.5, 0.5, int(refine_pts))
        else:
            base_r = right if right > 0 else 1.0
            local = np.logspace(max(-10.0, np.log10(max(left, 1e-9))), np.log10(max(base_r, 1e-9)), int(refine_pts))
        local = np.unique(np.concatenate((local, [left, best_x, right]))[np.isfinite(np.concatenate((local, [left, best_x, right]))) & (np.concatenate((local, [left, best_x, right])) >= 0.0) & (np.concatenate((local, [left, best_x, right])) <= 1e4)])

        if local.size > 0:
            max_aff_local = np.empty(local.size, dtype=np.float64)
            for xi in range(local.size):
                x = float(local[xi])
                vals_aff = aff_base + aff_linear * (x - old)
                if i <= 2 * i < i + n:
                    mid_local = 2 * i - i
                    vals_aff[mid_local] = c_k[mid_local] + x * x - old * old
                max_aff_local[xi] = float(np.max(vals_aff))
            maxc_l = np.maximum(max_unaff, max_aff_local)
            S_l = D + local
            vals_l = np.full(local.size, float('inf'), dtype=np.float64)
            valid_l = S_l > 1e-15
            vals_l[valid_l] = (2.0 * n) * maxc_l[valid_l] / (S_l[valid_l] ** 2)
            idx2 = int(np.argmin(vals_l))
            if float(vals_l[idx2]) < best_val:
                best_x = float(local[idx2])
                best_val = float(vals_l[idx2])

    if not np.isfinite(best_val) or best_val >= score - tol:
        return p, c, score, False

    # Apply update
    S = D + best_x
    if S <= 1e-15:
        return p, c, score, False

    p_new = p / S
    p_new[i] = best_x / S
    c_k_new = aff_base + aff_linear * (best_x - old)
    if i <= 2 * i < i + n:
        mid_local = 2 * i - i
        c_k_new[mid_local] = c_k[mid_local] + best_x * best_x - old * old
    c_new = c.copy()
    c_new[ks] = c_k_new

    # Renormalize
    s2 = float(np.sum(p_new))
    if s2 <= 0:
        return p, c, score, False
    p_new /= s2
    c_new /= (s2 * s2)
    np.clip(p_new, 0.0, None, out=p_new)
    np.clip(c_new, 0.0, None, out=c_new)

    score_new = _score_normalized(p_new, c_new)
    if not np.isfinite(score_new) or score_new >= score - tol:
        return p, c, score, False
    return p_new, c_new, score_new, True


def _coordinate_optimize(p, c, score, deadline, rng, grid_count, refine_count, max_passes=100):
    """Run coordinate descent until deadline or convergence."""
    n = int(p.size)
    best_p = p.copy()
    best_c = c.copy()
    best_score = score

    grid = np.concatenate(([0.0], np.logspace(-8.0, 2.0, int(grid_count))))
    stagnant = 0

    while time.monotonic() < deadline - 1.0:
        order = rng.permutation(n)
        improved = False
        for i in order:
            if time.monotonic() > deadline - 0.5:
                break
            p, c, score, imp = _coord_update(p, c, score, int(i), grid, int(refine_count))
            if imp:
                improved = True

        # Verify and renormalize
        s = float(np.sum(p))
        if s <= 0:
            break
        p = p / s
        c = np.convolve(p, p)
        np.clip(c, 0.0, None, out=c)
        score = _score_normalized(p, c)

        if score < best_score - 1e-12:
            best_score = score
            best_p = p.copy()
            best_c = c.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant >= 3:
                p = best_p.copy()
                c = best_c.copy()
                score = best_score
                # Perturb
                p = best_p * np.exp(rng.normal(0.0, 0.3, n))
                p = np.clip(p, 0.0, None)
                s = float(np.sum(p))
                if s > 0:
                    p = p / s
                    c = np.convolve(p, p)
                    score = _score_normalized(p, c)
                stagnant = 0

        if stagnant >= 8:
            break

    return best_p, best_c, best_score


def _expand_to_length(proto, target_len):
    target_len = int(target_len)
    p = _normalize(proto)
    R = int(p.size)
    if R <= 0 or target_len <= 0:
        return np.ones(max(1, target_len), dtype=np.float64)
    if R == target_len:
        return p.copy()
    if R > target_len:
        out = np.ones(target_len, dtype=np.float64)
        for j in range(target_len):
            lo = (j * R) // target_len
            hi = max(lo + 1, ((j + 1) * R) // target_len)
            chunk = p[lo:hi]
            out[j] = float(np.mean(chunk)) if chunk.size > 0 else 1.0
        return _normalize(out)
    base = target_len // R
    rem = target_len % R
    lengths = np.full(R, base, dtype=np.int64)
    if rem > 0:
        lengths[:rem] += 1
    out = np.zeros(target_len, dtype=np.float64)
    idx = 0
    for i in range(R):
        L = int(lengths[i])
        if L > 0:
            out[idx:idx + L] = p[i] / float(L)
            idx += L
    return _normalize(out)


def propose_candidate(seed=42, budget_s=1800, **kwargs):

    except Exception:
        pass

    if initial is None or initial.size != final_len:
        if initial is not None and initial.size > 0:
            initial = _expand_to_length(initial, final_len)

            sc = _score_array(initial)
            try_update(initial, sc)
            initial = _normalize(initial)

    f = initial.copy()
    c = np.convolve(f, f)
    score = _score_normalized(f, c)
    print(f"Initial score: {score:.6f}, n={f.size}")

    # Main optimization loop

    coord_count = 0

                    else:
                        print(f"  LP iter {lp_count}: no improvement (score_new={score_new:.6f})")

                    print(f"  LP iter {lp_count}: t={best_t:.4f}, no improvement")

                sc_pert = _score_array(f_pert)

    if best_seq is not None:
        print(f"Best tracked score: {best_score:.6f}")

    # Choose between local best and tracked best
    out = best_seq if best_seq is not None else f
    out = _normalize(out)
    print(f"Returning sequence of length {out.size}, score={best_score:.6f}")
    return [float(x) for x in out]
