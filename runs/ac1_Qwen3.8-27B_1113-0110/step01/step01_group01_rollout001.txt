    L = c.size

"""
Search for nonnegative step-function heights minimizing
    2*n*max(convolve(a,a)) / (sum(a))^2
using adaptive mirror descent with momentum, simulated annealing,
structured sinusoidal perturbations, multi-resolution search,
and optional LP-guided refinement via cvxpy.
"""

import time
import numpy as np

try:
    import scipy.optimize
except Exception:
    pass

try:
    import cvxpy as cp
    HAVE_CP = True
except Exception:
    HAVE_CP = False


def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (max(0, int(m) - 1)).bit_length()


def normalize_pos(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([1.0])
    a = np.clip(a, floor, None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size) / a.size
    return a / s


def conv_full(a, N=None):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if N is None:
        N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[: 2 * n - 1]


def score_fn(a, N=None):
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    c = conv_full(a_norm, N)
    return float(2.0 * n * np.max(c))


def resample(a, n):
    a = normalize_pos(a, 1e-15)
    old_n = a.size
    n = max(4, int(n))
    if old_n == n:
        return a.copy()
    if old_n == 1:
        return np.full(n, 1.0 / n)
    ox = np.linspace(0.0, 1.0, old_n)
    nx = np.linspace(0.0, 1.0, n)
    b = np.interp(nx, ox, a)
    return normalize_pos(b, 1e-15)


def soft_weights(c, cmax, tau):
    L = c.size
    if cmax <= 0 or tau <= 1e-15:
        w = np.zeros(L)
        idx = np.argmax(c)
        w[idx] = 1.0
        return w
    z = (c - cmax) / tau
    z = np.clip(z, -50.0, 0.0)
    ez = np.exp(z)
    s = float(ez.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros(L)
        w[np.argmax(c)] = 1.0
        return w
    return ez / s


def grad_from_w(w, a, N, n):
    """Gradient of soft-max objective w.r.t. a, using FFT correlation."""
    w_rev = w[::-1]
    fw = np.fft.rfft(w_rev, N)
    fa = np.fft.rfft(a, N)
    conv = np.fft.irfft(fw * fa, N)
    g = conv[n - 1: 2 * n - 1]
    return 2.0 * g


def mirror_descent_full(a, deadline, verbose=True, seed=42):
    """Main mirror descent optimizer with annealing and structured escapes."""
    a = normalize_pos(a, 1e-15)
    n = a.size
    N = next_pow2(2 * n - 1)
    rng = np.random.default_rng(seed)

    best_a = a.copy()
    best_sc = score_fn(a, N)
    c = conv_full(a, N)
    cmax = float(np.max(c))

    eta = 0.7
    tau = 0.08 * cmax
    temp = 0.015
    mom_buf = []
    mom_len = 5
    stale = 0
    it = 0
    last_print_time = time.monotonic()

    while time.monotonic() < deadline:
        it += 1
        cmax = float(np.max(c))
        sc_now = float(2.0 * n * cmax)

        if sc_now < best_sc * (1.0 - 1e-10):
            best_sc = sc_now
            best_a = a.copy()
            stale = 0

        if verbose and (time.monotonic() - last_print_time) > 30:
            print("  MD n={} it={} sc={:.9f} best={:.9f} eta={:.3f} tau={:.5f} temp={:.4f}".format(
                n, it, sc_now, best_sc, eta, tau, temp), flush=True)
            last_print_time = time.monotonic()

        if cmax <= 0 or not np.isfinite(cmax):
            break

        # Compute soft gradient
        w = soft_weights(c, cmax, tau)
        g = grad_from_w(w, a, N, n)
        g = g - g.mean()
        gnorm = float(np.abs(g).max())
        if gnorm < 1e-14:
            break
        g = g / gnorm

        # Momentum
        mom_buf.append(g.copy())
        if len(mom_buf) > mom_len:
            mom_buf.pop(0)
        g_mom = np.mean(mom_buf, axis=0)
        g_mom = g_mom - g_mom.mean()
        gm = float(np.abs(g_mom).max())
        if gm < 1e-14:
            g_mom = g
        else:
            g_mom = g_mom / gm

        # Try steps with backtracking
        accepted = False
        for e in (eta, 0.6 * eta, 0.36 * eta, 0.216 * eta, 0.13 * eta):
            if e < 1e-10:
                continue
            loga = np.log(a + 1e-30)
            loga_try = loga - e * g_mom
            loga_try = loga_try - loga_try.max()
            a_try = np.exp(np.clip(loga_try, -60.0, 0.0))
            a_try = np.maximum(a_try, 1e-300)
            a_try = a_try / float(a_try.sum())
            a_try = normalize_pos(a_try, 1e-15)

            c_try = conv_full(a_try, N)
            cmax_try = float(np.max(c_try))
            sc_try = float(2.0 * n * cmax_try)

            if sc_try < sc_now * (1.0 - 1e-11):
                accepted = True
                a = a_try
                c = c_try
                break
            else:
                delta = sc_try - sc_now
                if delta < 0.08 and rng.random() < np.exp(-delta / max(temp, 1e-12)):
                    accepted = True
                    a = a_try
                    c = c_try
                    break

        if accepted:
            if stale % 32 == 0:
                eta = min(1.4, eta * 1.03)
            tau = max(0.015 * cmax, tau * 0.997)
            temp = max(0.001, temp * 0.9995)
        else:
            eta = max(1e-4, eta * 0.5)
            tau = max(0.008 * cmax, tau * 0.8)
            temp = max(0.0005, temp * 0.88)
            stale += 1

            # Structured escape when stuck
            if stale % 100 == 0:
                escaped = False
                # Try several structured perturbations
                for amp in (0.08, 0.05, 0.03):
                    if time.monotonic() > deadline:
                        break
                    for _ in range(4):
                        freq = int(rng.integers(1, max(2, n // 8)))
                        phase = rng.uniform(0.0, 2.0 * np.pi)
                        pert = amp * np.sin(2.0 * np.pi * freq * np.arange(n) / n + phase)
                        a_try = a * (1.0 + pert)
                        a_try = normalize_pos(a_try, 1e-15)
                        c_try = conv_full(a_try, N)
                        sc_try = float(2.0 * n * np.max(c_try))
                        if sc_try < best_sc:
                            a = a_try
                            c = c_try
                            best_sc = sc_try
                            best_a = a.copy()
                            stale = 0
                            escaped = True
                            break
                    if escaped:
                        break

                if not escaped:
                    # Random multiplicative noise
                    noise = rng.normal(0.0, 0.06, n)
                    a_try = a * np.exp(np.clip(noise, -1.0, 1.0))
                    a_try = normalize_pos(a_try, 1e-15)
                    c_try = conv_full(a_try, N)
                    sc_try = float(2.0 * n * np.max(c_try))
                    if sc_try < best_sc:
                        a = a_try
                        c = c_try
                        best_sc = sc_try
                        best_a = a.copy()
                    mom_buf = []
                    stale = 0
                    eta = min(1.4, eta * 2.5)
                    temp = min(0.06, temp * 3.0)

            if stale > 800:
                break

    return best_a, best_sc


def lp_guided_step(a, N, n, M):
    """Solve LP: max sum(b) s.t. b>=0, (a*b)[k]<=M for all k. Return b direction."""
    if not HAVE_CP:
        return None
    try:
        b_var = cp.Variable(n, nonneg=True)
        obj = cp.Maximize(cp.sum(b_var))
        constraints = []
        # (a*b)[k] <= M for k=0..2n-2
        # (a*b)[k] = sum_j a[j]*b[k-j] where 0<=j<n and 0<=k-j<n
        # Use cvxpy: we can express as a @ b for each shift
        # Actually, use the fact that (a*b)[k] = (a * b)[k]
        # cvxpy can handle: cp.dot(a, b_var) is sum(a*b), but we need shifts.
        # Let's use the matrix form: C @ b <= M where C is (2n-1) x n
        # C[k, j] = a[j] if 0 <= k-j < n, else 0
        
        # This is a circulant-like matrix. Build it efficiently.
        # C[k,j] = a[j] * 1{0<=k-j<n} = a[j] * 1{j<=k<n+j}
        # So C is (2n-1) x n with C[k,j] = a[j] if k-j in [0,n-1]
        
        # Build using numpy
        C = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            C[j:j+n, j] = a[j]
        
        constraints.append(C @ b_var <= M)
        prob = cp.Problem(obj, constraints)
        sol = None
        for solver in (cp.SCS, cp.ECOS, cp.CLARABEL):
            try:
                prob.solve(solver=solver, verbose=False)
                if b_var.value is not None:
                    sol = np.asarray(b_var.value, dtype=np.float64).ravel()
                    break
            except Exception:
                continue
        
        if sol is None:
            return None
        sol = np.clip(sol, 0, None)
        s = sol.sum()
        if s < 1e-15:
            return None
        return sol / s
    except Exception:
        return None


def lp_polish(a, deadline, seed=0):
    """Use LP direction for a few iterations to guide the search."""
    if not HAVE_CP:
        return a, score_fn(a)
    
    a = normalize_pos(a, 1e-15)
    n = a.size
    N = next_pow2(2 * n - 1)
    rng = np.random.default_rng(seed)
    
    best_a = a.copy()
    best_sc = score_fn(a, N)
    
    for it in range(20):
        if time.monotonic() > deadline:
            break
        c = conv_full(a, N)
        cmax = float(np.max(c))
        # Target: try to get below current max
        M = cmax * 1.01  # small slack
        
        b_dir = lp_guided_step(a, N, n, M)
        if b_dir is None:
            break
        
        # Blend current a with LP direction
        for t in (0.3, 0.5, 0.7, 0.9):
            a_try = (1 - t) * a + t * b_dir
            a_try = normalize_pos(a_try, 1e-15)
            sc_try = score_fn(a_try, N)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try.copy()
                break
        else:
            # No improvement, try smaller t
            a_try = (1 - 0.1) * a + 0.1 * b_dir
            a_try = normalize_pos(a_try, 1e-15)
            sc_try = score_fn(a_try, N)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try.copy()
    
    return best_a, best_sc


def harmonic_search(a, n, N):
    """Try adding individual sinusoidal harmonics to find good directions."""
    a = normalize_pos(a, 1e-15)
    best_sc = score_fn(a, N)
    best_a = a.copy()
    i = np.arange(n)
    
    for k in range(1, min(n // 4, 64)):
        for amp in (0.02, 0.05, 0.1, 0.15):
            for cos_phase in (0.0, np.pi / 2, np.pi):
                a_try = a * (1.0 + amp * np.cos(2.0 * np.pi * k * i / n + cos_phase))
                a_try = normalize_pos(a_try, 1e-15)
                sc_try = score_fn(a_try, N)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a_try.copy()
    return best_a, best_sc


def multi_scale_mirror(a, deadline, seed=0):
    """Mirror descent with multi-scale: coarse tau first, then fine."""
    a = normalize_pos(a, 1e-15)
    n = a.size
    best_a = a.copy()
    best_sc = score_fn(a)
    
    # Phase A: coarse (large tau, large steps)
    dl_a = min(deadline, time.monotonic() + 0.4 * (deadline - time.monotonic()))
    a1, sc1 = mirror_descent_full(a, dl_a, verbose=True, seed=seed)
    if sc1 < best_sc:
        best_sc = sc1
        best_a = a1.copy()
    
    # Phase B: fine (small tau, small steps)
    if time.monotonic() < deadline - 10:
        a2, sc2 = mirror_descent_full(a1, deadline, verbose=True, seed=seed + 1000)
        if sc2 < best_sc:
            best_sc = sc2
            best_a = a2.copy()
    
    return best_a, best_sc
