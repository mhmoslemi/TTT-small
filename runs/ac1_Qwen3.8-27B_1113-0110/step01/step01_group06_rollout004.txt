    b = _fft_sq(a)
    return float(2*a.size * b.max() / (s*s))

def _build_sparse_toeplitz(f):
    n = f.size
    L = 2*n - 1
    # A[k,j] = f[k-j] if 0<=k-j<n
    # Column j: rows j to min(j+n-1, L-1), values f[0], f[1], ...
    nnz_per_col = []
    for j in range(n):
        k_start = j
        k_end = min(j+n, L)
        nnz_per_col.append(k_end - k_start)
    total = sum(nnz_per_col)
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    vals = np.empty(total, dtype=np.float64)
    idx = 0
    for j in range(n):
        k_start = j
        k_end = min(j+n, L)
        cnt = k_end - k_start
        rows[idx:idx+cnt] = range(k_start, k_end)
        cols[idx:idx+cnt] = j
        vals[idx:idx+cnt] = f[:cnt]
        idx += cnt
    return sp.csc_matrix((vals, (rows, cols)), shape=(L, n))

def _lp_solve(f, M):
    n = f.size
    A = _build_sparse_toeplitz(f)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'time_limit': 120, 'presolve': True})
    if res.success:
        return np.maximum(res.x, 0)
    return None

def _lp_optimize(f0, time_limit, label=""):
    f = np.asarray(f0, dtype=np.float64).copy()
    f = np.maximum(f, 1e-10)
    s = f.sum()
    f /= s
    
    best_f = f.copy()
    best_score = _score(best_f)
    
    t0 = time.monotonic()
    t_step = 0.5
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 60:
            break
        
        b = _fft_sq(f)
        M = float(b.max())
        
        g = _lp_solve(f, M)
        if g is None:
            print(f"[{label}] LP failed at iter {it}, t={elapsed:.1f}s")
            break
        
        s_g = float(g.sum())
        if s_g < 1e-10:
            break
        
        g /= s_g  # normalize to sum 1
        
        f_new = (1 - t_step) * f + t_step * g
        f_new = np.maximum(f_new, 1e-15)
        f_new /= f_new.sum()
        
        sc = _score(f_new)
        f = f_new
        
        if sc < best_score:
            best_score = sc
            best_f = f.copy()
        
        it += 1
        print(f"[{label}] LP iter={it} t={elapsed:.1f}s score={sc:.6f} best={best_score:.6f} sum_g={s_g:.4f}")
        
        # Early stop if convergence
        if s_g > 0 and abs(s_g - 1.0) < 1e-6:
            print(f"[{label}] LP converged (sum_g≈1)")
            break
    
    return best_f, best_score

def _grad_optimize(f0, time_limit, label="", focus_start=5.0, max_focus=2000.0):

        elapsed = time.monotonic() - t0

            break

            f /= f.sum()
            no_improve = 0

            eta = 1.0
            it += 1
            continue

        maxb = float(b.max())
        if maxb > 1e-300:

                f /= f.sum()

        it += 1

            print(f"[{label}] grad it={it} t={elapsed:.1f}s sc={sc:.6f} best={best_score:.6f} f={focus:.1f} eta={eta:.3f}")
    
    return best_f, best_score

"""
Optimize non-negative step function minimizing 2n·max(a∗a)/(Σa)².
Combines LP-based projection (SOTA), FFT gradient descent, and
anti-palindromic parameterized search with adaptive focus.
"""
import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog


def _fft_sq(a):
    """Compute a*a (full linear convolution) via FFT."""
    n = a.size
    L = 2 * n - 1
    N = 1
    while N < L:
        N <<= 1
    fa = np.fft.rfft(a, N)
    b = np.fft.irfft(fa * fa, N).real[:L]
    return np.maximum(b, 0.0)


def _score(a):
    """Compute 2n·max(a∗a)/(Σa)²."""
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    b = _fft_sq(a)
    return float(2.0 * a.size * b.max() / (s * s))


def _score_from_b(a, b):
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    return float(2.0 * a.size * float(b.max()) / (s * s))


def _toeplitz_structure(n):
    """Pre-compute sparsity structure for banded Toeplitz (2n-1)×n matrix."""
    L = 2 * n - 1
    col_starts = np.arange(n, dtype=np.int64)
    col_lengths = np.minimum(n, L - np.arange(n))
    total = int(col_lengths.sum())
    row_idx = np.concatenate([np.arange(s, s + l) for s, l in zip(col_starts, col_lengths)])
    col_idx = np.repeat(np.arange(n, dtype=np.int64), col_lengths)
    val_idx = np.concatenate([np.arange(l) for l in col_lengths])
    return row_idx, col_idx, val_idx, L


def _build_toeplitz(f, struct):
    """Build sparse Toeplitz matrix from pre-computed structure."""
    row_idx, col_idx, val_idx, L = struct
    n = f.size
    data = f[val_idx]
    return sp.csc_matrix((data, (row_idx, col_idx)), shape=(L, n))


def _lp_solve(f, M, struct, time_limit=60):
    """Solve max sum(g) s.t. T(f)g <= M, g>=0. Returns g or None."""
    A = _build_toeplitz(f, struct)
    n = f.size
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success:
            g = np.maximum(res.x, 0.0)
            return g
    except Exception:
        pass
    return None
