"""
Search for non-negative step-function heights minimizing
    2*n*max(convolve(a,a)) / (sum(a))^2
Using:
  - Soft-max gradient descent with annealing (FFT-based)
  - Binary sequence optimization (greedy + SA)
  - Multiplicative weight local search

import time
import numpy as np

try:
    import scipy.optimize
except Exception:
    scipy_optimize = None


def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def normalize_pos(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, float(floor), None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def fft_conv(a, N=None):
    n = a.size
    need = next_pow2(2 * n - 1)
    if N is None:
        N = need
    else:
        N = max(N, need)
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]


def score_internal(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s <= 1e-15:
        return np.inf
    a = a / s
    c = fft_conv(a)
    return float(2.0 * n * np.max(c))


def grad_weighted_conv(w, a, N):
    """Gradient of sum_k w[k]*c[k] w.r.t. a, where c=conv(a,a)."""
    n = a.size
    fa = np.fft.rfft(a, N)
    wf = np.fft.rfft(w[::-1], N)
    conv = np.fft.irfft(fa * wf, N)
    return 2.0 * conv[n - 1:2 * n - 1][::-1]


def softmax_weights(c, cmax, tau, floor=0.0):
    if cmax <= 0 or not np.isfinite(cmax) or tau <= 0:
        return np.full(c.size, 1.0 / c.size)
    z = (c - cmax) / tau
    z = np.clip(z, -50.0, 0.0)
    ez = np.exp(z)
    s = ez.sum()
    if s <= 0:
        return np.full(c.size, 1.0 / c.size)
    w = ez / s
    if floor > 0:
        w = (1.0 - floor) * w + floor / c.size
    return w


def smooth_max(c, cmax, tau):
    if cmax <= 0 or not np.isfinite(cmax):
        return 0.0
    if tau <= 0:
        return cmax
    z = (c - cmax) / tau
    z = np.clip(z, -50.0, 0.0)
    s = np.exp(z).sum()
    if s <= 0:
        return cmax
    return cmax + tau * np.log(s)


def log_softmax(z):
    z = z - z.max()
    z = np.clip(z, -60.0, 0.0)
    e = np.exp(z)
    s = e.sum()
    if s <= 0:
        return np.full(z.size, 1.0 / z.size)
    return e / s


def get_given():

        seq = globals().get("height_sequence_1", None)
        if seq is None:
            return None
        a = np.asarray(seq, dtype=np.float64).ravel()

        if not np.all(np.isfinite(a)):
            return None
        return a
    except Exception:

def resample(a, n):
    a = normalize_pos(a, 0.0)
    if a.size == n:
        return a.copy()
    old_x = np.linspace(0, 1, a.size)
    new_x = np.linspace(0, 1, n)
    b = np.interp(new_x, old_x, a)
    return normalize_pos(b, 1e-12)


def optimize_softmax(a_init, deadline, seed=0, n=None):
    """Optimize using soft-max gradient on log-space with annealed temperature."""
    if n is None:
        n = a_init.size
    a = normalize_pos(a_init, 1e-15)
    n = a.size
    N = next_pow2(3 * n - 1)
    best_a = a.copy()
    best_score = score_internal(a)
    tau = 0.15 / n
    eta = 0.8
    stale = 0
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() < deadline:
        c = fft_conv(a, N)
        cmax = float(np.max(c))
        if cmax <= 0 or not np.isfinite(cmax):
            break
        score = float(2.0 * n * cmax)
        if score < best_score * (1 - 1e-12):
            best_score = score
            best_a = a.copy()

        w = softmax_weights(c, cmax, tau, floor=0.02)
        g = grad_weighted_conv(w, a, N)
        g = g - g.mean()
        gs = np.abs(g).max()
        if gs <= 1e-18:
            break
        g = g / gs

        loga = np.log(np.maximum(a, 1e-20))
        loga -= loga.max()
        J_cur = smooth_max(c, cmax, tau)

        accepted = False
        for e in (eta, 0.5 * eta, 0.25 * eta, 0.1 * eta):
            if e < 1e-10:
                continue
            la = loga - e * g
            la = la - la.max()
            a_try = np.exp(np.clip(la, -60, 0))
            a_try = np.maximum(a_try, 1e-300)
            a_try = a_try / a_try.sum()
            a_try = normalize_pos(a_try, 1e-15)
            c_try = fft_conv(a_try, N)
            sm = float(np.max(c_try))
            J_try = smooth_max(c_try, sm, tau)
            sc_try = float(2.0 * n * sm)
            if sc_try < score * (1 - 1e-10):
                a = a_try
                score = sc_try
                accepted = True
                break
            if J_try < J_cur * (1 - 1e-7) and sc_try < score * 1.02:
                a = a_try
                score = sc_try
                accepted = True
                break

        if accepted:
            stale = 0
            eta = min(1.5, eta * 1.03)
            tau = max(0.005 / n, tau * 0.998)
        else:
            stale += 1
            eta = max(1e-5, eta * 0.6)
            if stale % 100 == 0:
                eta = min(1.5, eta * 2.5)
                tau = min(0.3 / n, tau * 1.8)
                stale = 0
            if eta < 1e-6 or stale > 1500:
                break

        it += 1
        if it % 300 == 0:
            print(f"  softmax n={n} it={it} best={best_score:.9f}", flush=True)

    return best_a, best_score


def optimize_binary(a_init, deadline, seed=0):
    """Binary 0/1 sequence optimization via greedy + simulated annealing."""
    n = a_init.size
    m = max(2, n // 2)
    # Initialize from a_init
    order = np.argsort(a_init, kind='stable')
    b = np.zeros(n, dtype=np.float64)
    b[order[-m:]] = 1.0

    N = next_pow2(2 * n - 1)
    c = fft_conv(b, N).copy()
    two_b = 2.0 * b
    cmax = float(np.max(c))
    best_b = b.copy()
    best_cmax = cmax
    best_score = float(2.0 * n * cmax / (m * m))

    rng = np.random.default_rng(seed)
    temp = 0.05
    beta = 0.6
    stale = 0
    it = 0

    while time.monotonic() < deadline:
        cmax = float(np.max(c))
        score = float(2.0 * n * cmax / (m * m))
        if score < best_score * (1 - 1e-12):
            best_score = score
            best_cmax = cmax
            best_b = b.copy()

        if cmax <= 0 or not np.isfinite(cmax):
            break

        w = softmax_weights(c, cmax, beta * cmax, floor=0.04)
        fa = np.fft.rfft(b, max(N, next_pow2(3 * n - 1)))
        N2 = max(N, next_pow2(3 * n - 1))
        fa = np.fft.rfft(b, N2)
        w_rev = w[::-1]
        fw = np.fft.rfft(w_rev, N2)
        conv_g = np.fft.irfft(fa * fw, N2)
        g = 2.0 * conv_g[n - 1:2 * n - 1][::-1]

        ones = np.nonzero(b > 0.5)[0]
        zeros = np.nonzero(b < 0.5)[0]
        if ones.size < 2 or zeros.size < 2:
            break

        # Pick positions to swap
        g_ones = g[ones]
        g_zeros = g[zeros]
        # Pick high-gradient one to remove, low-gradient zero to add
        gi = int(rng.choice(ones.size, p=_softmax_pick(g_ones, temp=0.1)))
        gj = int(rng.choice(zeros.size, p=_softmax_pick_neg(g_zeros, temp=0.1)))
        i = int(ones[gi])
        j = int(zeros[gj])
        if i == j:
            continue

        # Apply swap
        c[i:i+n] -= two_b
        c[2*i] -= 1.0
        c[j:j+n] += two_b
        c[2*j] += 1.0
        b[i] = 0.0
        b[j] = 1.0
        two_b = 2.0 * b

        new_max = float(np.max(c))
        new_score = float(2.0 * n * new_max / (m * m))
        delta = new_score - score

        accept = False
        if delta < -1e-12:
            accept = True
        elif abs(delta) < 1e-12:
            accept = rng.random() < 0.3
        elif delta < 0.5:
            accept = rng.random() < np.exp(-delta / max(temp, 1e-12))

        if accept:
            stale = 0
            temp = max(0.002, temp * 0.997)
            beta = max(0.1, beta * 0.998)
        else:
            # Revert
            c[j:j+n] -= two_b
            c[2*j] -= 1.0
            c[i:i+n] += two_b
            c[2*i] += 1.0
            b[i] = 1.0
            b[j] = 0.0
            two_b = 2.0 * b
            stale += 1
            temp = max(0.001, temp * 0.95)
            if stale % 300 == 0:
                temp = min(0.15, temp * 2.0)
                beta = min(1.5, beta * 1.5)
                stale = 0
            if temp < 0.0005 and stale > 2000:
                break

        it += 1

    return normalize_pos(best_b, 0.0), best_score


def _softmax_pick(g, temp=0.1):
    """Pick index with probability proportional to exp(g/temp) (prefer high g)."""
    g = np.asarray(g, dtype=np.float64)
    if g.size == 0:
        return np.array([1.0])
    z = (g - g.max()) / max(temp, 1e-12)
    z = np.clip(z, -50, 0)
    p = np.exp(z)
    s = p.sum()
    if s <= 0:
        return np.full(g.size, 1.0 / g.size)
    return p / s


def _softmax_pick_neg(g, temp=0.1):
    """Pick index with probability proportional to exp(-g/temp) (prefer low g)."""
    return _softmax_pick(-np.asarray(g, dtype=np.float64), temp)

    rng = np.random.default_rng(seed)

        d = np.clip(d, -4, 4)

    out_list = [float(x) for x in best_a]

        final_eval = float(evaluate_sequence(out_list))
        print(f"final evaluate_sequence = {final_eval:.12f}, n={len(out_list)}", flush=True)
    except Exception:
        final_eval = float('inf')
        print("WARNING: evaluate_sequence failed, using internal score", flush=True)

    if not np.isfinite(final_eval):
        best_a = np.ones(1024, dtype=np.float64)
        out_list = [float(x) for x in best_a]

    print(f"DONE: internal={best_score:.9f}, n={len(out_list)}", flush=True)
    return out_list
