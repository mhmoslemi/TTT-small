import cvxpy as cp
import numpy as np
import time

def lp_step(f, M, n):
    """Solve: max sum(g) s.t. conv(f,g) <= M, g >= 0, sum(g) = n"""
    g = cp.Variable(n, nonneg=True)
    # conv(f, g) has length 2n-1
    # conv(f,g)[k] = sum_j f[j]*g[k-j]
    # This is a linear expression in g
    
    constraints = [cp.sum(g) == n]
    for k in range(2*n-1):
        # conv(f,g)[k] = sum_j f[j]*g[k-j] where 0<=j<n and 0<=k-j<n
        j_min = max(0, k - (n-1))
        j_max = min(n-1, k)
        # g[k-j] for j from j_min to j_max, so g[k-j_max] to g[k-j_min]
        # = g[k-j_max], g[k-j_max+1], ..., g[k-j_min]
        idx_start = k - j_max
        idx_end = k - j_min
        constraints.append(f[j_min:j_max+1] @ g[idx_start:idx_end+1] <= M)
    
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.GLOP)  # or CLP, or CBC
    return g.value

1. Enforce non-negativity: a = max(a, 0)
2. Normalize: a = a / sum(a) * n

The normalization step changes the effective step, so we need to be careful with the step size.

Let me use an Adam-like optimizer:
- m = β1 * m + (1-β1) * grad
- v = β2 * v + (1-β2) * grad²
- a = a - lr * m / (sqrt(v) + eps)
- Then project: a = max(a, 0), normalize

This should give faster convergence than plain gradient descent with momentum.

Let me also add "annealing" steps: periodically add random noise to escape local minima.
