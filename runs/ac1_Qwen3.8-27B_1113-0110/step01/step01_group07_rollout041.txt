Multi-scale FISTA with adaptive smoothing for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines multi-resolution initialization, FISTA momentum, and direct subgradient polish.
"""
import time
import numpy as np

def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()

def normalize_positive(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s

def fast_score(a):
    p = a  # assume already normalized
    n = p.size
    if n <= 0:
        return np.inf
    L = next_pow2(2 * n - 1)
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0.0:
        return np.inf
    return float(2.0 * n * cm)

def softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = np.where(np.isfinite(z), z, -50.0)
    z = z - z.max()
    z = np.clip(z, -50.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        e = np.ones_like(z)
        s = float(e.size)
    return e / s

def resize_sequence(seq, m):
    p = normalize_positive(seq)
    m = int(m)
    if p is None:
        return np.ones(m, dtype=np.float64) / float(m)
    n = p.size
    if n == m:
        return p.copy()
    if n == 1:
        return np.full(m, 1.0 / float(m), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, m)
    out = np.interp(x_new, x_old, p)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(m, dtype=np.float64) / float(m)
    return out / s

    """
    Compute gradient of 2n * smooth_max(conv(p,p)) w.r.t. z (log-space).
    p = softmax(z), sum(p) = 1.
    L_small: FFT length for computing c = conv(p,p) (>= 2n-1)
    L_large: FFT length for gradient computation (>= 3n-2)
    Returns: (true_score, grad_z)
    """
    n = p.size
    # Compute convolution
    P = np.fft.rfft(p, L_small)
    c = np.fft.irfft(P * P, L_small)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0.0:
        return np.inf, np.zeros(n, dtype=np.float64)
    
    # Smooth max weights
    z = (c - cm) / tau
    z = np.clip(z, -50.0, 0.0)
    expz = np.exp(z)
    expsum = float(expz.sum())
    if not np.isfinite(expsum) or expsum <= 0.0:
        expz = np.zeros_like(c)
        expz[int(np.argmax(c))] = 1.0
        expsum = 1.0
    
    W = tau * expz / expsum  # length 2n-1, sums to tau
    
    # Gradient of smooth_max w.r.t. p_j = 2 * sum_k W[k] * p[k-j] (by symmetry)
    # This is 2 * cross_correlation(W, p)[j]
    # = 2 * convolution(W, p_reversed)[2n-2-j]... 
    # Let me use the FFT trick:
    # cross_corr(W, p)[j] = sum_k W[k] * p[k-j]
    # = IFFT(FFT(W) * conj(FFT(p)))[j] for circular, with zero-padding for linear

    # flip(p) has length n, W has length 2n-1
    # conv length = n + 2n-1 - 1 = 3n-2
    p_flip = p[::-1]  # length n

    # Wp[k] = sum_i W[i] * p_flip[k-i] = sum_i W[i] * p[n-1-(k-i)] = sum_i W[i] * p[n-1-k+i]
    # Let j = n-1-k, so k = n-1-j: Wp[n-1-j] = sum_i W[i] * p[j+i]
    # That's the cross-correlation at lag j: sum_i W[i] * p[j+i] = (W ⋆ p)(-j)

    # g[j] = sum_{k=0}^{2n-2} W[k] * p[k-j] * I(0 <= k-j <= n-1)
    # Let's define the full convolution of W and p (both zero-padded):
    # (W * p)[m] = sum_k W[k] * p[m-k]
    # g[j] = sum_k W[k] * p[k-j] = sum_k W[k] * p[-(j-k)]

    #     = conv(W, p_rev)[j]
    # conv(W, p_rev)[j] = sum_k W[k] * p_rev[j-k] = sum_k W[k] * p[n-1-j+k]
    # Let m = n-1-j+k, so k = m-n+1+j:
    # = sum_m W[m-n+1+j] * p[m] * I(0<=m<n, 0<=m-n+1+j<=2n-2)
    # This is getting circular. Let me just compute it directly with the FFT:
    
    # The simplest correct approach:
    # g[j] = 2 * sum_{k=max(j,0)}^{min(j+n-1, 2n-2)} W[k] * p[k-j]
    # This is 2 * (W cross-correlated with p) at lag j
    # In FFT: cross_corr = IFFT(FFT(W, L) * conj(FFT(p, L)))
    # For real signals: = IFFT(FFT(W, L) * FFT(p, L)) but with p reversed...
    
    # OK let me just use the approach from the previous code which I know works:
    P_large = np.fft.rfft(p, L_large)
    W_rev = W[::-1]  # reverse W (length 2n-1, zero-padded to L_large)
    H = np.fft.irfft(P_large * np.fft.rfft(W_rev, L_large), L_large)
    # H is the circular convolution of p and W_rev, length L_large
    # Linear convolution of p (length n) and W_rev (length 2n-1) has length 3n-2
    # H[k] = sum_i p[i] * W_rev[k-i] for the linear part
    = sum_i p[i] * W[2n-2-(k-i)] = sum_i p[i] * W[2n-2-k+i]
    # Let j = 2n-2-k, so k = 2n-2-j:
    # H[2n-2-j] = sum_i p[i] * W[j+i]
    # That's the cross-correlation! (W ⋆ p)(j) = sum_i W[j+i] * p[i]
    # So g[j] = 2 * H[2n-2-j] for j = 0, ..., n-1
    
    # So g_p[j] = 2 * H[2n-2-j] for j in [0, n-1]
    # In array indexing: g_p = 2 * H[n-1 : 2n-1][::-1]
    # Wait: H[2n-2-j] for j=0 gives H[2n-2], for j=n-1 gives H[n-1]
    # So g_p[j] = 2 * H[2n-2-j], which in numpy is:
    # g_p = 2.0 * H[n-1:2*n-1][::-1]
    # Let me check: H[n-1:2*n-1] gives H[n-1], H[n], ..., H[2n-2]
    # Reversed: H[2n-2], H[2n-3], ..., H[n-1]
    # So g_p[0] = 2*H[2n-2], g_p[1] = 2*H[2n-3], ..., g_p[n-1] = 2*H[n-1]
    # g_p[j] = 2*H[2n-2-j] ✓
    
    g_p = 2.0 * H[n - 1:2 * n - 1][::-1]
    g_p = np.where(np.isfinite(g_p), g_p, 0.0)
    
    # Now the objective is 2n * smooth_max(c) (since sum(p)=1)
    # ∂F/∂p_j = 2n * g_p[j]
    # ∂F/∂z_j = p_j * (2n * g_p[j] - sum_k p_k * 2n * g_p[k])
    #          = 2n * p_j * (g_p[j] - p·g_p)
    
    mean_g = float(np.dot(p, g_p))
    grad_z = 2.0 * n * p * (g_p - mean_g)
    
    # Normalize gradient for stability
    max_g = float(np.max(np.abs(grad_z)))
    if max_g > 1e-2:
        grad_z *= (1e-2 / max_g)
    
    true_score = 2.0 * n * cm
    return float(true_score), grad_z

"""
Multi-scale FISTA with peak-reduction polish for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses FFT-based gradients in log-space, Nesterov momentum, adaptive smoothing,
and a multiplicative peak-reduction heuristic for final polish.
"""

import time
import numpy as np


def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def normalize_positive(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s


def fast_score(a):
    p = a
    n = p.size
    if n <= 0:
        return np.inf
    L = next_pow2(2 * n - 1)
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0.0:
        return np.inf
    return float(2.0 * n * cm)


def softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = np.where(np.isfinite(z), z, -50.0)
    z = z - z.max()
    z = np.clip(z, -50.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        e = np.ones_like(z)
        s = float(e.size)
    return e / s


def resize_sequence(seq, m):
    p = normalize_positive(seq)
    m = int(m)
    if p is None:
        return np.ones(m, dtype=np.float64) / float(m)
    n = p.size
    if n == m:
        return p.copy()
    if n == 1:
        return np.full(m, 1.0 / float(m), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, m)
    out = np.interp(x_new, x_old, p)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(m, dtype=np.float64) / float(m)
    return out / s


def compute_grad_z(p, tau, L_score, L_grad):
    """Compute gradient of 2n*smooth_max(conv(p,p)) w.r.t. log-space z.
    p must be a probability vector (sum to 1)."""
    n = p.size
    P = np.fft.rfft(p, L_score)
    c = np.fft.irfft(P * P, L_score)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0.0:
        return np.inf, np.zeros(n, dtype=np.float64)

    z_shift = (c - cm) / tau
    z_shift = np.clip(z_shift, -50.0, 0.0)
    expz = np.exp(z_shift)
    expsum = float(expz.sum())
    if not np.isfinite(expsum) or expsum <= 0.0:
        expz = np.zeros_like(c)
        expz[int(np.argmax(c))] = 1.0
        expsum = 1.0

    W = tau * expz / expsum

    # Gradient w.r.t. p: g_p[j] = 2 * sum_k W[k] * p[k-j] (by symmetry of c)
    P_g = np.fft.rfft(p, L_grad)
    W_rev = W[::-1]
    H = np.fft.irfft(P_g * np.fft.rfft(W_rev, L_grad), L_grad)
    g_p = 2.0 * H[n - 1:2 * n - 1][::-1]
    g_p = np.where(np.isfinite(g_p), g_p, 0.0)

    mean_g = float(np.dot(p, g_p))
    grad_z = 2.0 * n * p * (g_p - mean_g)

    max_g = float(np.max(np.abs(grad_z)))
    if max_g > 1e-3:
        grad_z *= (1e-3 / max_g)

    true_score = 2.0 * n * cm
    return float(true_score), grad_z


def peak_reduction_step(p, eta, n_steps=50):
    """Reduce the peak of conv(p,p) by multiplicatively decreasing
    positions that contribute most to the peak lag."""
    n = p.size
    L = next_pow2(2 * n - 1)
    best_p = p.copy()
    best_s = fast_score(p)

    for _ in range(n_steps):
        P = np.fft.rfft(p, L)
        c = np.fft.irfft(P * P, L)[:2 * n - 1]
        c = np.where(np.isfinite(c), c, 0.0)
        c = np.maximum(c, 0.0)
        k_star = int(np.argmax(c))

        # Sensitivity: s[j] = p[k_star - j] + p[j - k_star] (valid indices only)
        s = np.zeros(n, dtype=np.float64)
        for j in range(n):
            idx1 = k_star - j
            idx2 = j - k_star
            val = 0.0
            if 0 <= idx1 < n:
                val += p[idx1]
            if 0 <= idx2 < n and idx2 != idx1:
                val += p[idx2]
            s[j] = val

        max_s = float(s.max())
        if max_s <= 0:
            break

        # Multiplicative update: reduce positions contributing to peak
        factor = 1.0 - eta * s / max_s
        factor = np.maximum(factor, 0.0)
        p_new = p * factor
        p_new = np.maximum(p_new, 0.0)
        sn = float(p_new.sum())
        if sn <= 1e-300:
            break
        p_new /= sn

        s_new = fast_score(p_new)
        if np.isfinite(s_new) and s_new < best_s:
            best_s = s_new
            best_p = p_new.copy()
            p = p_new
        else:
            # Halve eta if no improvement
            eta *= 0.5
            if eta < 1e-8:
                break

    return best_p, best_s


def fista_run(p_init, duration_s, rng, tau_start=0.08, tau_end=0.002,
              lr_start=0.04, name="fista", verbose=True):
    """FISTA in log-space with adaptive smoothing."""
    n = p_init.size
    L_score = next_pow2(2 * n - 1)
    L_grad = next_pow2(3 * n - 2)
    deadline = time.monotonic() + max(0.0, float(duration_s))

    z = np.log(np.maximum(p_init, 1e-300))
    z = np.where(np.isfinite(z), z, -50.0)
    z -= z.mean()
    z = np.clip(z, -50.0, 50.0)

    best_p = p_init.copy()
    best_score = fast_score(p_init)

    t_param = 1.0
    z_prev = z.copy()
    z_curr = z.copy()
    no_improve = 0
    step = 0
    total_steps = max(1, int(duration_s * 15))

    while time.monotonic() < deadline:
        progress = min(1.0, step / float(total_steps))
        tau = tau_start * (tau_end / tau_start) ** progress
        lr = lr_start / (1.0 + 0.5 * progress)

        t_next = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param))
        alpha = (t_param - 1.0) / t_next

        # Nesterov step
        z_y = z_curr + alpha * (z_curr - z_prev)
        z_y = np.clip(z_y, -50.0, 50.0)

        p_y = softmax_fn(z_y)
        true_s, grad_z = compute_grad_z(p_y, tau, L_score, L_grad)
        if not np.isfinite(true_s):
            break

        z_prev = z_curr.copy()
        z_curr = z_y - lr * grad_z
        z_curr = np.where(np.isfinite(z_curr), z_curr, z_y)
        z_curr -= z_curr.mean()
        z_curr = np.clip(z_curr, -50.0, 50.0)

        t_param = t_next
        step += 1

        # Check score periodically
        if step % 20 == 0:
            p_check = softmax_fn(z_curr)
            s_check = fast_score(p_check)
            if np.isfinite(s_check) and s_check < best_score:
                best_score = s_check
                best_p = p_check.copy()
                no_improve = 0
            else:
                no_improve += 1

        # Restart if stuck
        if no_improve > 200:
            z = np.log(np.maximum(best_p, 1e-300))
            z = np.where(np.isfinite(z), z, -50.0)
            z += rng.normal(0.0, 0.08, n)
            z -= z.mean()
            z = np.clip(z, -50.0, 50.0)
            z_prev = z.copy()
            z_curr = z.copy()
            t_param = 1.0
            no_improve = 0
            if verbose:
                print(f"{name} restart at step {step}, score {best_score:.6f}")

        if verbose and step % 5000 == 0:
            p_now = softmax_fn(z_curr)
            s_now = fast_score(p_now)
            print(f"{name} step {step} cur {s_now:.6f} best {best_score:.6f} tau {tau:.5f}")

    return best_p, best_score


def adam_run(p_init, duration_s, rng, tau_ratio=0.03, lr0=0.015, name="adam"):
    """Adam in log-space with smooth-max, as a backup to FISTA."""
    n = p_init.size
    L_score = next_pow2(2 * n - 1)
    L_grad = next_pow2(3 * n - 2)
    deadline = time.monotonic() + max(0.0, float(duration_s))

    z = np.log(np.maximum(p_init, 1e-300))
    z = np.where(np.isfinite(z), z, -50.0)
    z -= z.mean()
    z = np.clip(z, -50.0, 50.0)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    bias_step = 0

    current_tau = tau_ratio / float(n)
    lr = float(lr0)
    best_p = p_init.copy()
    best_score = fast_score(p_init)
    no_improve = 0
    step = 0

    while time.monotonic() < deadline:
        true_s, grad_z = compute_grad_z(softmax_fn(z), current_tau, L_score, L_grad)
        if not np.isfinite(true_s):
            break

        if true_s < best_score - 1e-12:
            best_score = true_s
            best_p = softmax_fn(z).copy()
            no_improve = 0
        else:
            no_improve += 1

        if step > 0 and step % 5000 == 0:
            lr = max(0.001, lr * 0.85)
            current_tau = max(0.001 / n, current_tau * 0.8)

        if no_improve > 2000:
            zz = np.log(np.maximum(best_p, 1e-300))
            zz = np.where(np.isfinite(zz), zz, -50.0)
            zz += rng.normal(0.0, 0.12, n)
            zz -= zz.mean()
            z = np.clip(zz, -50.0, 50.0)
            m *= 0.1
            v *= 0.1
            no_improve = 0
            current_tau = max(0.003 / n, current_tau * 0.6)
            lr = max(0.002, lr * 0.7)

        bias_step += 1
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        b1 = 1.0 - beta1 ** bias_step
        b2 = 1.0 - beta2 ** bias_step
        z = z - lr * (m / b1) / (np.sqrt(v / b2) + eps)
        z = np.where(np.isfinite(z), z, 0.0)
        z -= z.mean()
        z = np.clip(z, -50.0, 50.0)
        step += 1

        if step % 10000 == 0:
            p_now = softmax_fn(z)
            s_now = fast_score(p_now)
            print(f"{name} step {step} cur {s_now:.6f} best {best_score:.6f}")

    return best_p, best_score
