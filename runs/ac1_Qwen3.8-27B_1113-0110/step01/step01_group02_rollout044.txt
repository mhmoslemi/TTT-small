                for phase in [0, np.pi / 4, np.pi / 3, np.pi / 2]:
                    a = 1.0 + amp * np.cos(2 * np.pi * freq * x / n + phase)
                    a = np.maximum(a, 0.01)
                    candidates.append((f"sin_{n}_f{freq}_a{amp}", a))

    # 3. Double sinusoidal
    for n in [2000, 3000, 4000]:
        x = np.arange(n, dtype=np.float64)
        for a1 in [0.3, 0.4]:
            for a2 in [0.15, 0.2]:
                a = 1.0 + a1 * np.cos(2 * np.pi * x / n + np.pi / 4) + a2 * np.cos(4 * np.pi * x / n)
                a = np.maximum(a, 0.01)
                candidates.append((f"doublsin_{n}_{a1}_{a2}", a))

    # 4. Chirp (quadratic phase)
    for n in [2000, 3000]:
        x = np.arange(n, dtype=np.float64)
        for chirp_rate in [1.0, 2.0, 3.0]:
            a = 1.0 + 0.3 * np.cos(np.pi * chirp_rate * x ** 2 / n)
            a = np.maximum(a, 0.01)
            candidates.append((f"chirp_{n}_{chirp_rate}", a))

    # 5. Exponential decay (asymmetric)
    for n in [2000, 3000]:
        x = np.arange(n, dtype=np.float64)
        for rate in [0.5, 0.8, 1.2]:
            a = np.exp(-rate * x / n) + 0.1
            candidates.append((f"decay_{n}_{rate}", a))

    # 6. Off-center Gaussian
    for n in [2000, 3000]:
        x = np.arange(n, dtype=np.float64)
        for center in [0.3, 0.4]:
            sigma = n * 0.15
            a = np.exp(-0.5 * ((x - n * center) / sigma) ** 2) + 0.05
            candidates.append((f"gauss_{n}_{int(center*100)}", a))

    # 7. Power law
    for n in [2000, 3000]:
        x = np.arange(1, n + 1, dtype=np.float64)
        for p in [0.3, 0.5, 0.7]:
            a = x ** p
            candidates.append((f"power_{n}_{p}", a))

    # 8. Random (structured)
    for n in [2000, 3000]:
        a = 1.0 + 0.3 * rng.normal(0, 1, n)
        a = np.maximum(a, 0.01)
        # Smooth it
        from scipy.ndimage import gaussian_filter1d
        a = gaussian_filter1d(a, sigma=5)
        a = np.maximum(a, 0.01)
        candidates.append((f"smoothrand_{n}", a))

    # 9. Cosine window (raised cosine)
    for n in [2000, 3000]:
        x = np.arange(n, dtype=np.float64)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
        a += 0.1
        candidates.append((f"coswin_{n}", a))

    print(f"Total candidates: {len(candidates)}")

    # Quick initial evaluation to rank
    scored = []
    for name, a0 in candidates:
        a0 = np.maximum(a0, 0.01)
        s = a0.sum()
        if s < 1e-15:
            continue
        a0_norm = a0 / s
        c0 = _fft_convolve(a0_norm)
        init_score = _score_from_conv(a0_norm, c0)
        scored.append((init_score, name, a0_norm))

    scored.sort(key=lambda x: x[0])
    print("Top 12 initial scores:")
    for i, (sc, name, _) in enumerate(scored[:12]):
        print(f"  {i+1}. {name} (n={len(_)}): {sc:.6f}")

    # Allocate time: focus on top candidates, with diminishing returns
    n_top = min(len(scored), 12)
    remaining_time = budget_s - 30.0

    # Time allocation: geometric weights favoring top candidates
    weights = []
    for i in range(n_top):
        w = max(0.5, (n_top - i) ** 1.5)
        weights.append(w)
    total_w = sum(weights)
    times = [remaining_time * w / total_w for w in weights]

    for i in range(n_top):
        if time.monotonic() > deadline:
            break

        init_score, name, a0 = scored[i]
        t_alloc = min(times[i], max(10.0, deadline - time.monotonic()))

        if t_alloc < 15:
            break

        print(f"\nOptimizing '{name}' (n={len(a0)}, init={init_score:.6f}, time={t_alloc:.0f}s)")

        # Strategy 1: Mirror descent (main workhorse)
        t1 = t_alloc * 0.45
        a1, s1 = _optimize_mirror(a0, t1, rng)
        eval1 = evaluate_sequence(a1.tolist())
        print(f"  Mirror (top5): {eval1:.6f}")
        if np.isfinite(eval1) and eval1 < best_overall_score:
            best_overall_score = eval1
            best_overall = a1.copy()
            print(f"  *** NEW BEST: {eval1:.6f} ***")

        # Strategy 2: Mirror descent with different top_k
        if time.monotonic() < deadline - 20:
            t2 = t_alloc * 0.2
            a2, s2 = _optimize_mirror(a0, t2, rng, top_k=10, eta0=0.08)
            eval2 = evaluate_sequence(a2.tolist())
            print(f"  Mirror (top10): {eval2:.6f}")
            if np.isfinite(eval2) and eval2 < best_overall_score:
                best_overall_score = eval2
                best_overall = a2.copy()
                print(f"  *** NEW BEST: {eval2:.6f} ***")

        # Strategy 3: Additive
        if time.monotonic() < deadline - 20:
            t3 = t_alloc * 0.2
            start3 = a1 if np.isfinite(s1) and s1 < init_score else a0
            a3, s3 = _optimize_additive(start3, t3, rng)
            eval3 = evaluate_sequence(a3.tolist())
            print(f"  Additive: {eval3:.6f}")
            if np.isfinite(eval3) and eval3 < best_overall_score:
                best_overall_score = eval3
                best_overall = a3.copy()
                print(f"  *** NEW BEST: {eval3:.6f} ***")

        # Strategy 4: Smoothed gradient
        if time.monotonic() < deadline - 20:
            t4 = t_alloc * 0.15
            start4 = a1 if np.isfinite(s1) and s1 < init_score else a0
            a4, s4 = _optimize_smoothed(start4, t4, rng)
            eval4 = evaluate_sequence(a4.tolist())
            print(f"  Smoothed: {eval4:.6f}")
            if np.isfinite(eval4) and eval4 < best_overall_score:
                best_overall_score = eval4
                best_overall = a4.copy()
                print(f"  *** NEW BEST: {eval4:.6f} ***")

    # Final polishing with remaining time
    remaining = deadline - time.monotonic()
    if best_overall is not None and remaining > 30:
        print(f"\n=== Final polishing ({remaining:.0f}s remaining) ===")

        # Local search
        t_ls = min(remaining * 0.3, 300)
        a_ls, s_ls = _local_search(best_overall, t_ls, rng)
        eval_ls = evaluate_sequence(a_ls.tolist())
        print(f"  Local search: {eval_ls:.6f}")
        if np.isfinite(eval_ls) and eval_ls < best_overall_score:
            best_overall_score = eval_ls
            best_overall = a_ls.copy()
            print(f"  *** NEW BEST: {eval_ls:.6f} ***")

        # Mirror descent polish
        remaining2 = deadline - time.monotonic()
        if remaining2 > 30:
            t_m = remaining2 * 0.5
            a_m, s_m = _optimize_mirror(best_overall, t_m, rng, top_k=3, eta0=0.05)
            eval_m = evaluate_sequence(a_m.tolist())
            print(f"  Mirror polish: {eval_m:.6f}")
            if np.isfinite(eval_m) and eval_m < best_overall_score:
                best_overall_score = eval_m
                best_overall = a_m.copy()
                print(f"  *** NEW BEST: {eval_m:.6f} ***")

        # Additive polish
        remaining3 = deadline - time.monotonic()
        if remaining3 > 20:
            a_add, s_add = _optimize_additive(best_overall, remaining3 * 0.6, rng, top_k=3, eta0=0.02)
            eval_add = evaluate_sequence(a_add.tolist())
            print(f"  Additive polish: {eval_add:.6f}")
            if np.isfinite(eval_add) and eval_add < best_overall_score:
                best_overall_score = eval_add
                best_overall = a_add.copy()
                print(f"  *** NEW BEST: {eval_add:.6f} ***")

        # Another local search
        remaining4 = deadline - time.monotonic()
        if remaining4 > 15:
            a_ls2, s_ls2 = _local_search(best_overall, remaining4, rng)
            eval_ls2 = evaluate_sequence(a_ls2.tolist())
            print(f"  Local search 2: {eval_ls2:.6f}")
            if np.isfinite(eval_ls2) and eval_ls2 < best_overall_score:
                best_overall_score = eval_ls2
                best_overall = a_ls2.copy()
                print(f"  *** NEW BEST: {eval_ls2:.6f} ***")

    # Final verification
    if best_overall is not None:
        final_score = evaluate_sequence(best_overall.tolist())
        elapsed = time.monotonic() - total_start
        print(f"\nFinal result: score={final_score:.6f}, n={len(best_overall)}, elapsed={elapsed:.1f}s")
        return [float(x) for x in best_overall]
    else:
        # Fallback
        n = 2000
        x = np.arange(n, dtype=np.float64)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * x / n + np.pi / 4)
        print("Falling back to sinusoidal sequence")
        return [float(max(0, v)) for v in a]
