def build_conv_matrix(a, n):
    """Build sparse matrix A where A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    # For each j (column), the non-zeros are in rows k = j, j+1, ..., j+n-1
    # with values a[0], a[1], ..., a[n-1]
    data = np.tile(a, n)  # This is n*n, too big
    # Better: build column by column
    data = np.concatenate([a] * n)  # Still n*n
    # Actually, for each column j, the entries are a[0], a[1], ..., a[n-1] at rows j, j+1, ..., j+n-1
    # But we only want rows 0 to 2n-2
    # So for column j, rows go from j to min(j+n-1, 2n-2)
    # and values are a[0], a[1], ..., a[min(2n-2-j, n-1)]
    
    nnz = 0
    for j in range(n):
        max_k = min(2*n-1, j+n)
        nnz += max_k - j
    
    rows = np.empty(nnz, dtype=int)
    cols = np.empty(nnz, dtype=int)
    vals = np.empty(nnz)
    
    idx = 0
    for j in range(n):
        max_k = min(2*n-1, j+n)
        length = max_k - j
        rows[idx:idx+length] = range(j, max_k)
        cols[idx:idx+length] = j
        vals[idx:idx+length] = a[:length]
        idx += length
    
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A
```

def _pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _clean(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    return np.maximum(v, 0.0)


def _score(a, L=None):
    v = _clean(a)
    n = len(v)
    if n == 0:
        return np.inf
    s = float(v.sum())
    if s < 1e-12:
        return np.inf
    v = v / s
    if L is None:
        L = _pow2(2 * n - 1)
    A = np.fft.rfft(v, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return np.inf
    return 2.0 * n * mx


def _normalize_to_sum(a, target=1.0):
    v = _clean(a)
    s = v.sum()
    if s < 1e-15:
        return np.full(len(a), target / len(a))
    return v * (target / s)


def _make_conv_matrix_sparse(a, n):
    """Build sparse (2n-1) x n matrix where A[k,j] = a[k-j] if valid."""
    # Each column j has non-zeros at rows j..j+n-1 (clipped to 0..2n-2)
    # with values a[0], a[1], ..., a[n-1]
    idx = 0
    total_nnz = n * n  # exactly n non-zeros per column
    rows = np.empty(total_nnz, dtype=np.int32)
    cols = np.empty(total_nnz, dtype=np.int32)
    vals = np.empty(total_nnz, dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        length = k_end - k_start
        rows[idx:idx+length] = np.arange(k_start, k_end)
        cols[idx:idx+length] = j
        vals[idx:idx+length] = a[:length]
        idx += length
    # Remove any trailing zeros (shouldn't happen but be safe)
    mask = np.arange(total_nnz) < idx
    rows, cols, vals = rows[mask], cols[mask], vals[mask]
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A
```

def _solve_lp(a, M, n):
    """Solve max sum(g) s.t. conv(a,g,'full') <= M, g >= 0."""
    # Build constraint matrix
    cols = np.repeat(np.arange(n), n)
    offsets = np.tile(np.arange(n), n)
    rows = cols + offsets
    vals = np.tile(a, n)
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Solve with linprog (minimize -sum(g) s.t. A*g <= M, g >= 0)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if res.status == 0:
        return res.x
    return None

def _nextpow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _clean(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    return np.maximum(v, 0.0)


def _score_fft(a, L=None):
    """Score = 2n * max(conv(a,a)) / sum(a)^2, computed via FFT."""
    v = _clean(a)
    n = len(v)
    if n == 0:
        return np.inf
    s = float(v.sum())
    if s < 1e-12:
        return np.inf
    v = v / s
    if L is None:
        L = _nextpow2(2 * n - 1)
    A = np.fft.rfft(v, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return np.inf
    return 2.0 * float(n) * mx


def _score_direct(a):
    """Score using direct convolution (for small n or final check)."""
    v = _clean(a)
    n = len(v)
    if n == 0:
        return np.inf
    s = float(v.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(v, v, 'full')
    mx = float(np.max(c))
    return 2.0 * float(n) * mx / (s * s)
