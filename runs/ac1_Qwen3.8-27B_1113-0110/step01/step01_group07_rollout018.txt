def cheb_score_from_coeffs(coeffs, n=2048):
    x = np.linspace(0, 1, n)
    t = 2*x - 1
    a = np.zeros(n)
    for m, c in enumerate(coeffs):
        a += c * np.cos(m * np.arccos(np.clip(t, -1, 1)))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return 1e10
    a = a / s
    # FFT-based convolution
    L = 1 << (2*n-2).bit_length()
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F*F, L)[:2*n-1]
    return 2*n*c.max()

    """Reduce the peak of a*a by iteratively modifying a."""
    a = a.copy()
    for _ in range(n_iters):
        s = a.sum()
        if s < 1e-30: break
        a = a / s
        n = len(a)
        L = 1 << (2*n-2).bit_length()
        F = np.fft.rfft(a, L)
        c = np.fft.irfft(F*F, L)[:2*n-1]
        peak_idx = c.argmax()
        peak_val = c[peak_idx]
        # The peak is at lag peak_idx: c[peak_idx] = sum_i a_i * a[peak_idx-i]
        # Compute the contribution of each a_i to this peak
        # a_i contributes a_i * a[peak_idx - i] to the peak
        # Gradient of c[peak_idx] wrt a_i is a[peak_idx-i] (for valid indices)
        grad = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                grad[i] = a[j]
        # Reduce a in the direction of the gradient (to decrease the peak)
        step = 0.01 * peak_val / (grad.max() + 1e-30)
        a = a - step * grad
        a = np.maximum(a, 0)
    return a

def _cos_matrix(n, D):

    n = p.size
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]

    c = np.maximum(c, 0.0)
    cm = float(c.max())
    if cm <= 0 or not np.isfinite(cm):
        return np.inf, np.zeros(n)
    
    z = (c - cm) / tau
    expz = np.exp(z)
    expsum = float(expz.sum())
    if not np.isfinite(expsum) or expsum <= 0:
        expz = np.zeros_like(c)
        expz[int(np.argmax(c))] = 1.0
        expsum = 1.0
    
    value_coeff = cm + tau * float(np.log(expsum))

            lr = max(0.0008, lr * 0.85)
            current_tau = max(0.003, current_tau * 0.85)
        
        if no_improve > 5000:
            # Restart from best with noise
            zz = np.log(np.maximum(best_p, 1e-300))
            zz += rng.normal(0.0, 0.2, n)
            zz -= zz.mean()
            z = np.clip(zz, -50.0, 50.0)
            m *= 0.05
            v *= 0.05
            no_improve = 0
            current_tau = max(0.005, current_tau * 0.6)
            lr = max(0.002, lr * 0.5)
        
        step += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        b1 = 1 - beta1 ** step
        b2 = 1 - beta2 ** step
        z = z - lr * (m / b1) / (np.sqrt(v / b2) + eps)
        z = np.where(np.isfinite(z), z, 0.0)
        z -= z.mean()
        z = np.clip(z, -50.0, 50.0)
        
        # Softmax
        z_shift = z - z.max()
        z_shift = np.clip(z_shift, -50.0, 0.0)
        e = np.exp(z_shift)
        s = e.sum()
        if s < 1e-30:
            e = np.ones_like(z)
            s = n
        p = e / s
        
        if step % 10000 == 0:
            print(f"  adam step {step} score {true_s:.6f} best {best_score:.6f} tau {current_tau:.4f} lr {lr:.5f}")
    
    return best_p, best_score

def _peak_reduce(p, n_iters=200, rng=None):
    """Targeted peak reduction: identify peak lag and reduce contributing elements."""
    p = _normalize(p)
    if p is None:
        return p
    n = p.size
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    
    best_p = p.copy()
    best_s = _fast_score(p)
    
    for it in range(n_iters):
        _, c = _fast_score_full(p)
        if c is None:
            break
        peak_idx = int(c.argmax())
        peak_val = c[peak_idx]
        
        # Compute which a_i contribute to the peak
        # c[peak_idx] = sum_i a_i * a_{peak_idx - i}
        contrib = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contrib[i] = p[j]
        
        # Reduce elements proportional to their contribution
        max_c = contrib.max()
        if max_c < 1e-30:
            break
        step_size = 0.02 * peak_val / (max_c * n)
        p_new = p - step_size * contrib
        p_new = np.maximum(p_new, 0.0)
        s = p_new.sum()
        if s < 1e-30:
            break
        p_new = p_new / s
        s_new = _fast_score(p_new)
        if s_new < best_s:
            best_s = s_new
            best_p = p_new.copy()
            p = p_new
        else:
            # Try smaller step
            p_try = p - 0.005 * step_size * contrib
            p_try = np.maximum(p_try, 0.0)
            st = p_try.sum()
            if st > 1e-30:
                p_try = p_try / st
                st_new = _fast_score(p_try)
                if st_new < best_s:
                    best_s = st_new
                    best_p = p_try.copy()
                    p = p_try
                else:
                    break
            else:
                break
    
    return best_p, best_s

def _try_mixed_starts(n, D, rng, time_budget=20.0):
    """Try various structured starts for Chebyshev optimization."""
    mat = _cos_matrix(n, D)
    best_f = np.inf
    best_x = None
    deadline = time.monotonic() + time_budget
    
    # Various structured starts
    starts = []
    # Pure modes
    for m in range(min(D, 6)):
        for sign in [1.0, -1.0]:
            c = np.zeros(D)
            c[m] = sign * rng.uniform(0.5, 2.0)
            starts.append(c)
    # Pairs
    for m in range(min(D-1, 5)):
        for m2 in range(m+1, min(D, 6)):
            c = np.zeros(D)
            c[m] = rng.uniform(0.3, 1.2)
            c[m2] = rng.choice([-1, 1]) * rng.uniform(0.2, 0.8)
            starts.append(c)
    # Random
    for _ in range(20):
        c = rng.normal(0, 0.5, D)
        c[0] = rng.choice([-1, 1]) * rng.uniform(0.5, 1.5)
        starts.append(c)
    
    def objective(coeffs):
        a = mat @ coeffs
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return 1e10
        a = a / s
        return _fast_score(a)
    
    for start in starts:
        if time.monotonic() > deadline:
            break
        try:
            res = minimize(objective, start, method='L-BFGS-B',
                          options={'maxiter': 100, 'ftol': 1e-7})
            if res.fun < best_f:
                best_f = float(res.fun)
                best_x = res.x.copy()
        except Exception:
            continue
    
    return best_x, best_f

def propose_candidate(seed=42, budget_s=2400.0, **kwargs):
    """Main search function. Returns the best sequence found."""
    global _TIME0
    _TIME0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    N = 8192  # Final sequence length
    N_OPT = 2048  # Length for Chebyshev optimization
    
    best_p = None
    best_s = np.inf
    
    def update(a, s):
        nonlocal best_p, best_s
        if a is not None and np.isfinite(s) and s < best_s:
            best_p = a.copy()
            best_s = float(s)
            print(f"  NEW BEST: {s:.6f}")
    
    # Seed from previous result if available
    try:
        hs = np.asarray(height_sequence_1, dtype=np.float64)
        hs = np.where(np.isfinite(hs), hs, 0.0)
        hs = np.maximum(hs, 0.0)
        if hs.size > 0 and hs.sum() > 1e-9:
            s0 = _fast_score(hs)
            update(hs, s0)
    except Exception:
        pass
    
    # Uniform baseline
    update(np.ones(N_OPT) / N_OPT, 2.0)
    
    print(f"Starting search. Initial best: {best_s:.6f}")
    print(f"Time budget: {budget_s:.0f}s")
    
    # ======================
    # Phase 1: Chebyshev optimization
    # ======================
    print("\n=== Phase 1: Chebyshev optimization ===")
    phase1_deadline = _TIME0 + min(400.0, budget_s * 0.17)
    
    best_cheb_coeffs = None
    best_cheb_D = 0
    best_cheb_score = np.inf
    
    D_range = [4, 6, 8, 10, 12, 16, 20, 24, 32]
    
    for D in D_range:
        if time.monotonic() > phase1_deadline:
            break
        alloc = min(35.0, (phase1_deadline - time.monotonic()) / max(1, len(D_range) - D_range.index(D)))
        coeffs, score = _optimize_cheb(N_OPT, D, rng, time_budget=alloc, n_starts=8)
        if coeffs is not None and score < best_cheb_score:
            best_cheb_score = score
            best_cheb_coeffs = coeffs
            best_cheb_D = D
            print(f"  D={D}: score {score:.6f}")
            # Also evaluate at full length
            p_full = _cheb_from_coeffs(coeffs, N, D)
            if p_full is not None:
                s_full = _fast_score(p_full)
                update(p_full, s_full)
    
    # Additional structured starts
    if time.monotonic() < phase1_deadline:
        for D in [8, 12, 16]:
            if time.monotonic() > phase1_deadline:
                break
            coeffs, score = _try_mixed_starts(N_OPT, D, rng, time_budget=20.0)
            if coeffs is not None and score < best_cheb_score:
                best_cheb_score = score
                best_cheb_coeffs = coeffs
                best_cheb_D = D
                print(f"  mixed D={D}: score {score:.6f}")
    
    # ======================
    # Phase 2: Adam from best Chebyshev
    # ======================
    print(f"\n=== Phase 2: Adam refinement (budget {_remaining():.0f}s remaining) ===")
    
    # Build parent from best Chebyshev or previous best
    if best_cheb_coeffs is not None and best_cheb_D > 0:
        parent = _cheb_from_coeffs(best_cheb_coeffs, N, best_cheb_D)
    elif best_p is not None:
        parent = _resize(best_p, N)
    else:
        parent = np.ones(N) / N
    
    if parent is None:
        parent = np.ones(N) / N
    
    s_parent = _fast_score(parent)
    print(f"  Parent score: {s_parent:.6f}")
    
    # Adam phase 2a: coarse
    rem = phase1_deadline + min(500.0, budget_s * 0.20) - time.monotonic()
    if rem > 30:
        p2, s2 = _adam_run(parent, max(30.0, rem), rng, tau_ratio=0.06, lr0=0.025)
        if p2 is not None:
            update(p2, s2)
    
    # ======================
    # Phase 3: Peak reduction + Adam from different angles
    # ======================
    print(f"\n=== Phase 3: Peak reduction + diverse Adam ===")
    phase3_start = time.monotonic() if time.monotonic() > phase1_deadline else phase1_deadline
    
    # Peak reduction on current best
    if best_p is not None and _remaining() > 100:
        pr_deadline_time = 60.0
        p_pr, s_pr = _peak_reduce(best_p, n_iters=300, rng=rng)
        if p_pr is not None:
            update(p_pr, s_pr)
            print(f"  Peak reduction: {s_pr:.6f}")
    
    # Diversify: try different Chebyshev degrees as subspaces
    if _remaining() > 200:
        # Take current best, project to low-D Chebyshev, re-express, add noise
        if best_p is not None:
            p_best_N = _resize(best_p, N)
            
            for D_extra in [20, 32, 48]:
                if _remaining() < 100:
                    break
                mat = _cos_matrix(N, D_extra)
                # Project best onto Chebyshev subspace
                proj = mat.T @ p_best_N  # This gives the Chebyshev coefficients
                # Add random high-frequency components
                D_opt = min(D_extra, 32)
                mat_opt = _cos_matrix(N, D_opt)
                proj_opt = mat_opt.T @ p_best_N
                noise = rng.normal(0, 0.05, D_opt)
                noise[D_opt//2:] *= 0.3
                new_coeffs = proj_opt + noise
                p_new = _cheb_from_coeffs(new_coeffs, N, D_opt)
                if p_new is not None:
                    s_new = _fast_score(p_new)
                    update(p_new, s_new)
    
    # ======================
    # Phase 4: Extended Adam with adaptive restarts
    # ======================
    print(f"\n=== Phase 4: Extended Adam ===")
    
    # Phase 4a: Medium tau
    if _remaining() > 200:
        parent4a = _resize(best_p if best_p is not None else np.ones(N)/N, N)
        p4a, s4a = _adam_run(parent4a, min(400.0, _remaining() * 0.30), rng,
                            tau_ratio=0.03, lr0=0.015)
        if p4a is not None:
            update(p4a, s4a)
    
    # Phase 4b: Fine tau
    if _remaining() > 200:
        parent4b = _resize(best_p if best_p is not None else np.ones(N)/N, N)
        p4b, s4b = _adam_run(parent4b, min(400.0, _remaining() * 0.30), rng,
                            tau_ratio=0.012, lr0=0.008)
        if p4b is not None:
            update(p4b, s4b)
    
    # ======================
    # Phase 5: Final targeted optimization
    # ======================
    print(f"\n=== Phase 5: Final optimization ===")
    
    # Try a "smart" restart: take the best, add specific shaped perturbations
    if _remaining() > 150 and best_p is not None:
        p_best = _resize(best_p, N)
        n = N
        
        # Perturbation 1: strengthen the ramp
        x = np.linspace(0, 1, n)
        ramp = (2 * x - 1)
        for amp in [0.02, 0.05, 0.1]:
            p_pert = p_best * np.exp(amp * ramp)
            p_pert = _normalize(p_pert)
            if p_pert is not None:
                sp = _fast_score(p_pert)
                update(p_pert, sp)
        
        # Perturbation 2: add a secondary bump
        for center in [0.3, 0.5, 0.7]:
            for width in [0.05, 0.1, 0.15]:
                bump = np.exp(-((x - center) / width) ** 2)
                p_pert = _normalize(p_best + 0.05 * bump)
                if p_pert is not None:
                    sp = _fast_score(p_pert)
                    update(p_pert, sp)
        
        # Short Adam from best perturbed
        if _remaining() > 100:
            p5, s5 = _adam_run(best_p, min(120.0, _remaining() * 0.5), rng,
                              tau_ratio=0.008, lr0=0.005)
            if p5 is not None:
                update(p5, s5)
    
    # Peak reduction again on final best
    if _remaining() > 50 and best_p is not None:
        p_final_pr, s_final_pr = _peak_reduce(best_p, n_iters=200, rng=rng)
        if p_final_pr is not None:
            update(p_final_pr, s_final_pr)
    
    # ======================
    # Final: Validate with evaluate_sequence
    # ======================
    print(f"\n=== Final validation ===")
    candidates = []
    if best_p is not None:
        candidates.append(best_p)
        # Also try slightly different lengths
        candidates.append(_resize(best_p, 4096))
        candidates.append(_resize(best_p, 16384))
    candidates.append(np.ones(1024) / 1024)
    
    final_seq = None
    final_score = np.inf
    
    for cand in candidates:
        if cand is None:
            continue
        lst = [float(x) for x in np.clip(np.asarray(cand, dtype=np.float64), 0.0, 1000.0)]
        if not lst:
            continue
        try:
            sc = evaluate_sequence(lst)
        except Exception:
            sc = np.inf
        if np.isfinite(sc) and sc < final_score:
            final_score = float(sc)
            final_seq = lst
            print(f"  Candidate len={len(lst)}: score={sc:.6f}")
    
    if final_seq is None:
        final_seq = [1.0 / 1024.0] * 1024
    
    print(f"\nFinal score: {final_score:.6f}")
    print(f"Total time: {_elapsed():.1f}s")
    return final_seq
