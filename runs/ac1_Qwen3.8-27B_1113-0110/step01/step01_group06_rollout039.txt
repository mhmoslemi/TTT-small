def _fft_square(a):
    n = int(a.size)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, n=N)
    b = np.fft.irfft(fa * fa, n=N).real[:L]
    b = np.maximum(b, 0.0)
    return b

def _score(a):
    b = _fft_square(a)

    if s <= 1e-300:
