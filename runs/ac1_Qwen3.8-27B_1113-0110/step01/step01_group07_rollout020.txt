    z = np.asarray(z, dtype=np.float64)
    z = np.where(np.isfinite(z), z, -50.0)
    z = z - np.max(z)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        return np.ones(z.size) / z.size
    return e / s

    """Compute score and gradient using smooth max of convolution."""
    n = p.size
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:2*n-1]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    
    cm = float(c.max())
    if cm <= 0 or not np.isfinite(cm):
        return np.inf, np.zeros(n)

    z = c / tau
    z = z - z.max()
    expz = np.exp(z)
    expsum = float(expz.sum())
    if not np.isfinite(expsum) or expsum <= 0:
        expz = np.zeros_like(c)
        expz[np.argmax(c)] = 1.0
        expsum = 1.0

Optimized search for non-negative coefficient sequences minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-based Adam with adaptive smooth-max, top-k gradients, multi-start, and cyclic LR.
"""

import time
import numpy as np

def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()

def normalize_positive(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s

def fast_score(a):
    p = normalize_positive(a)
    if p is None:
        return np.inf
    n = p.size
    L = next_pow2(2 * n - 1)
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:2*n-1]
    c = np.where(np.isfinite(c), c, 0.0)
    return float(2.0 * n * c.max())

def softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = np.where(np.isfinite(z), z, -50.0)
    z = z - np.max(z)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.ones(z.size) / z.size
    return e / s

def score_and_grad_smooth(p, tau, L, L_cross):
    n = p.size
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:2*n-1]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    cm = float(c.max())
    if cm <= 0.0 or not np.isfinite(cm):
        return np.inf, np.zeros(n, dtype=np.float64)
    z = c / tau
    z = z - z.max()
    expz = np.exp(z)
    expsum = float(expz.sum())
    if not np.isfinite(expsum) or expsum <= 0.0:
        expz = np.zeros_like(c)
        expz[int(np.argmax(c))] = 1.0
        expsum = 1.0
    w = expz / expsum
    WF = np.fft.rfft(w, L_cross)
    PF = np.fft.rfft(p, L_cross)
    cross = np.fft.irfft(np.conj(WF) * PF, L_cross)[:n]
    grad_conv = 2.0 * cross
    mean_g = float(np.dot(p, grad_conv))
    grad = grad_conv - p * mean_g
    true_score = 2.0 * n * cm
    return true_score, grad

def score_and_grad_topk(p, k, L, L_cross):
    n = p.size
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:2*n-1]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    cm = float(c.max())
    if cm <= 0.0 or not np.isfinite(cm):
        return np.inf, np.zeros(n, dtype=np.float64)
    k = max(1, min(k, c.size))
    if k >= c.size:
        w = np.ones(c.size) / c.size
    else:
        idx = np.argpartition(c, -k)[-k:]
        thresh = c[idx].min()
        w = (c >= thresh - 1e-15).astype(np.float64)
        ws = float(w.sum())
        if ws <= 0.0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
            ws = 1.0
        w /= ws
    value = float(np.dot(w, c))
    WF = np.fft.rfft(w, L_cross)
    PF = np.fft.rfft(p, L_cross)
    cross = np.fft.irfft(np.conj(WF) * PF, L_cross)[:n]
    grad_conv = 2.0 * cross
    mean_g = float(np.dot(p, grad_conv))
    grad = grad_conv - p * mean_g
    true_score = 2.0 * n * cm
    return true_score, grad

def cheb_basis(n, D):
    x = np.linspace(0.0, 1.0, n)
    t = 2.0 * x - 1.0
    t = np.clip(t, -1.0, 1.0)
    ang = np.arccos(t)
    out = np.empty((n, D), dtype=np.float64)
    for m in range(1, D + 1):
        out[:, m - 1] = np.cos(m * ang)
    return out

def make_parent(n, theta, B, mode, eps):
    val = B.dot(theta)
    if mode == 0:
        h = 1.0 + eps + val
    else:
        h = np.exp(np.clip(val * eps, -6.0, 6.0))
    return normalize_positive(h)

def resize_seq(seq, m):
    p = normalize_positive(seq)
    m = int(m)
    if p is None:
        return np.ones(m, dtype=np.float64) / m
    n = p.size
    if n == m:
        return p.copy()
    if n == 1:
        return np.full(m, 1.0/m, dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, m)
    out = np.interp(x_new, x_old, p)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(m, dtype=np.float64) / m
    return out / s

def adam_optimize(p0, duration_s, rng, n, L, L_cross, tau_start, tau_end, lr_start, lr_end, topk_start, topk_end, use_topk):

        return None, np.inf
    z = np.log(np.maximum(p, 1e-300))
    z = np.where(np.isfinite(z), z, -50.0)
    z -= z.mean()
    z = np.clip(z, -50.0, 50.0)
    m_buf = np.zeros(n, dtype=np.float64)
    v_buf = np.zeros(n, dtype=np.float64)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    best_p = p.copy()
    best_score = fast_score(p)
    no_improve = 0
    step = 0
    start_t = time.monotonic()
    deadline = start_t + max(0.0, duration_s)
    
    while time.monotonic() < deadline:
        progress = min(1.0, (time.monotonic() - start_t) / max(1e-6, duration_s))
        tau = tau_start * (tau_end / tau_start) ** progress
        tau = max(1e-14, tau / max(1, n))
        k_top = max(1, int(round(topk_start * (topk_end / topk_start) ** progress)))
        
        if use_topk and k_top < n:
            _, grad = score_and_grad_topk(p, k_top, L, L_cross)
        else:
            _, grad = score_and_grad_smooth(p, max(1e-14, tau), L, L_cross)
        
        if not np.isfinite(grad).all():
            grad = np.zeros(n, dtype=np.float64)
        
        gmax = float(np.max(np.abs(grad)))
        if gmax > 0.05:
            grad *= 0.05 / gmax
        
        step += 1
        m_buf = beta1 * m_buf + (1.0 - beta1) * grad
        v_buf = beta2 * v_buf + (1.0 - beta2) * (grad * grad)
        b1 = 1.0 - beta1 ** step
        b2 = 1.0 - beta2 ** step
        
        lr = lr_start + (lr_end - lr_start) * (1.0 - np.cos(np.pi * progress)) * 0.5
        lr = max(1e-5, lr)
        
        z = z - lr * (m_buf / b1) / (np.sqrt(v_buf / b2) + eps_adam)
        z = np.where(np.isfinite(z), z, 0.0)
        z -= z.mean()
        z = np.clip(z, -50.0, 50.0)
        p = softmax(z)
        
        if step % 2000 == 0:
            s_now = fast_score(p)
            if s_now < best_score - 1e-12:
                best_score = s_now
                best_p = p.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 8000:
                zz = np.log(np.maximum(best_p, 1e-300))
                zz = np.where(np.isfinite(zz), zz, -50.0)
                zz += rng.normal(0.0, 0.2, n)
                zz -= zz.mean()
                z = np.clip(zz, -50.0, 50.0)
                m_buf *= 0.1
                v_buf *= 0.1
                no_improve = 0
            if step % 10000 == 0:
                print(f"  adam step {step} score={s_now:.6f} best={best_score:.6f} tau={tau:.2e} lr={lr:.4f} k={k_top}")
    
    s_final = fast_score(p)
    if s_final < best_score:
        best_score = s_final
        best_p = p.copy()
    return best_p, best_score
