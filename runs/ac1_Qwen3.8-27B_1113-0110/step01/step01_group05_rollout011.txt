"""
Minimizer for 2*n*max(convolve(a,a))/(sum(a))^2 with a >= 0.
Uses FFT-based convolution, targeted coordinate descent, peak-aware
optimization, window perturbation, and simulated-annealing-style
restarts to approach the theoretical minimum for non-negative
autocorrelation-peak sequences.
"""

import numpy as np
import time


def _fft_convolve(a, b):
    n = a.size + b.size - 1
    size = 1
    while size < n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    r = np.fft.irfft(fa * fb, size)[:n]
    np.maximum(r, 0.0, out=r)
    return r


def _score_from_conv(a, c):
    n = a.size
    s = float(np.sum(a))
    if s < 1e-15:
        return float('inf')
    return float(2.0 * n * float(np.max(c)) / (s * s))


def _score(a):
    c = _fft_convolve(a, a)
    return _score_from_conv(a, c)


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).copy()
    np.clip(a, 0.0, 1000.0, out=a)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def _coord_optimize(a, c, deadline, rng, grid, max_perturb=5):
    """
    Coordinate descent on a (sum-normalized).
    c = convolve(a, a) must be consistent with a.
    Returns best (a, c, score) found.
    """
    n = a.size
    a = _normalize(a)
    c = _fft_convolve(a, a)
    score = _score_from_conv(a, c)
    
    best_a = a.copy()
    best_c = c.copy()
    best_score = score
    
    stagnant = 0
    perturb_count = 0
    noise_scales = [0.4, 0.2, 0.1, 0.05, 0.02]
    
    while time.monotonic() < deadline - 0.5:
        order = rng.permutation(n)
        improved = False
        
        for i in order:
            if time.monotonic() > deadline - 0.3:
                break
            
            old = float(a[i])
            if old < 1e-15:
                old = 0.0
            
            # Grid of candidate values for a[i]
            # Scale relative to current value
            scale = max(old, 1.0 / n)
            candidates = np.concatenate(([0.0], scale * np.logspace(-4, 2, grid.size - 1)))
            # Add the current value
            candidates = np.unique(np.concatenate((candidates, [old])))
            candidates = candidates[(candidates >= 0) & (candidates <= 1000.0 / n * 10)]
            if candidates.size == 0:
                candidates = np.array([old])
            
            # For each candidate x, compute new score
            deltas = candidates - old
            S_new = 1.0 + deltas  # since sum(a) = 1
            
            # Affected c indices: k in [i, i+n-1]
            k_lo = i
            k_hi = min(i + n, 2 * n - 1)
            a_idx = np.arange(k_lo, k_hi) - i  # indices into a, range [0, n-1]
            
            # c_new[k] = c[k] + 2*delta*a[k-i] for each k in range
            # Shape: (num_candidates, k_hi - k_lo)
           affected = c[k_lo:k_hi][None, :] + 2.0 * deltas[:, None] * a[a_idx][None, :]
            # Fix self-term at k=2i (if in range)
            si = 2 * i
            if k_lo <= si < k_hi:
                affected[:, si - k_lo] += deltas[:, None] ** 2
            
            # Unaffected max
            if k_lo > 0:
                unaff_left = float(np.max(c[:k_lo]))
            else:
                unaff_left = 0.0
            if k_hi < 2 * n - 1:
                unaff_right = float(np.max(c[k_hi:]))
            else:
                unaff_right = 0.0
            unaff = max(unaff_left, unaff_right, 0.0)
            
            max_affected = np.max(affected, axis=1)
            max_c = np.maximum(max_affected, unaff)
            
            valid = S_new > 1e-15
            vals = np.full(candidates.size, float('inf'))
            vals[valid] = 2.0 * n * max_c[valid] / (S_new[valid] ** 2)
            
            idx_best = int(np.argmin(vals))
            if not np.isfinite(vals[idx_best]) or vals[idx_best] >= score - 1e-14:
                continue
            
            new_val = float(candidates[idx_best])
            delta = new_val - old
            S_new = 1.0 + delta
            
            # Update a and c
            a[i] = new_val
            c[k_lo:k_hi] += 2.0 * delta * a[a_idx]
            if k_lo <= si < k_hi:
                c[si] += delta * delta
            c = c / (S_new * S_new)  # renormalize
            a = a / S_new
            np.maximum(c, 0.0, out=c)
            np.maximum(a, 0.0, out=a)
            
            # Recalculate score
            score = _score_from_conv(a, c)
            improved = True
            
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                best_c = c.copy()
        
        # After full pass
        a = _normalize(a)
        c = _fft_convolve(a, a)
        score = _score_from_conv(a, c)
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_c = c.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        if not improved:
            # Try perturbation
            if stagnant >= 2 and perturb_count < max_perturb:
                scale = noise_scales[min(perturb_count, len(noise_scales) - 1)]
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = _normalize(a)
                c = _fft_convolve(a, a)
                score = _score_from_conv(a, c)
                if score < best_score - 1e-10:
                    best_score = score
                    best_a = a.copy()
                    best_c = c.copy()
                perturb_count += 1
                stagnant = 0
        
        if stagnant >= 8:
            a = best_a.copy()
            c = best_c.copy()
            score = best_score
            stagnant = 0
    
    return best_a, best_c, best_score


def _peak_target(a, c, deadline, rng, max_rounds=50):
    """
    Specifically target the lags with highest c values.
    For each top lag, try to reduce it by adjusting contributing elements.
    """
    n = a.size
    best_a = a.copy()
    best_c = c.copy()
    score = _score_from_conv(best_a, best_c)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        c_max_idx = np.argsort(c)[::-1][:5]  # top 5 lags
        
        for k_star in c_max_idx:
            k_star = int(k_star)
            # Pairs (j, k_star - j) that contribute to c[k_star]
            j_lo = max(0, k_star - (n - 1))
            j_hi = min(n - 1, k_star)
            
            if j_hi < j_lo:
                continue
            
            # Find the pair with the largest product
            js = np.arange(j_lo, j_hi + 1)
            pairs = a[js] * a[k_star - js]
            max_pair_idx = int(np.argmax(pairs))
            j_idx = int(js[max_pair_idx])
            j2 = k_star - j_idx
            
            if j_idx == j2:
                # Self-pair: try reducing a[j_idx]
                for frac in [0.7, 0.5, 0.3]:
                    new_a = a.copy()
                    delta = (frac - 1.0) * a[j_idx]
                    new_a[j_idx] = a[j_idx] * frac
                    new_a = _normalize(new_a)
                    new_c = _fft_convolve(new_a, new_a)
                    new_score = _score_from_conv(new_a, new_c)
                    if new_score < score - 1e-10:
                        a, c, score = new_a, new_c, new_score
                        if score < best_score - 1e-12:
                            best_score = score
                            best_a = a.copy()
                            best_c = c.copy()
                        break
            else:
                # Transfer mass from the larger element to a "safe" element
                if a[j_idx] >= a[j2]:
                    donor = j_idx
                else:
                    donor = j2
                
                # Find a "safe" element (one that doesn't contribute to any top lag)
                # Simple heuristic: pick a random element
                for attempt in range(5):
                    receiver = int(rng.integers(n))
                    if receiver == donor:
                        continue
                    
                    for frac in [0.7, 0.5, 0.3]:
                        new_a = a.copy()
                        transfer = (1.0 - frac) * a[donor]
                        new_a[donor] = a[donor] * frac
                        new_a[receiver] += transfer
                        new_a = _normalize(new_a)
                        new_c = _fft_convolve(new_a, new_a)
                        new_score = _score_from_conv(new_a, new_c)
                        if new_score < score - 1e-10:
                            a, c, score = new_a, new_c, new_score
                            if score < best_score - 1e-12:
                                best_score = score
                                best_a = a.copy()
                                best_c = c.copy()
                            break
                    else:
                        continue
                    break
        
        # Periodic re-optimization
        if (rnd + 1) % 10 == 0:
            a, c, score = _coord_optimize(a, c, min(deadline, time.monotonic() + 30), rng, 
                                           np.logspace(-4, 1, 12), max_perturb=1)
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                best_c = c.copy()
    
    return best_a, best_c, best_score
