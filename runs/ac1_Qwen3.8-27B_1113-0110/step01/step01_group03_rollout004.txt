Minimizes 2n*max(conv(a,a))/sum(a)^2 using:
1. LP-based direction finding (convolution constraint LP)

3. Multiple n values and restarts for diversity

import numpy as np
import time

from scipy.optimize import linprog


def fft_self_conv(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    if n <= 1:
        return a.copy()
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:2 * n - 1]


def fast_score(a):
    """Compute score = 2n*max(conv)/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_self_conv(a)
    M = c.max()
    return 2.0 * n * M / (S * S)


def normalize(a, target_sum):
    """Normalize to target sum, keeping non-negative."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return np.full_like(a, target_sum / len(a))
    return a / S * target_sum


def build_conv_matrix(n, a):
    """Build sparse matrix A where (a*b)(k) = A[k,:] dot b."""
    # A[k,j] = a[k-j] if 0 <= k-j < n, else 0
    total_nz = n * n  # each row has length proportional to triangular
    # Actually: for k=0..n-1, row length = k+1; for k=n..2n-2, row length = 2n-1-k
    # Total = sum_{k=0}^{n-1}(k+1) + sum_{k=n}^{2n-2}(2n-1-k) = n(n+1)/2 + (n-1)n/2 = n^2
    data = np.empty(n * n, dtype=np.float64)
    indices = np.empty(n * n, dtype=np.int32)
    indptr = np.empty(2 * n, dtype=np.int64)
    
    offset = 0
    indptr[0] = 0
    ones_buf = np.arange(2 * n, dtype=np.int32)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k) + 1
        length = j_end - j_start
        indices[offset:offset + length] = ones_buf[j_start:j_end]
        data[offset:offset + length] = a[k - ones_buf[j_start:j_end]]
        offset += length
        indptr[k + 1] = offset
    
    return sp.csr_matrix((data, indices, indptr), shape=(2 * n - 1, n))


def lp_direction(a, M):
    """Solve LP to find best direction b with (a*b)(k) <= M, b >= 0, max sum(b)."""
    n = len(a)
    A = build_conv_matrix(n, a)
    
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'time_limit': 30})
        if result.success and result.x is not None:
            b = result.x
            b = np.maximum(b, 0)
            S = b.sum()
            if S > 0.01:
                return b / S * n
        return None
    except Exception:
        return None

    """Gradient descent on the score with multi-peak targeting."""
    n = len(a)
    a = normalize(a, n)

    beta = 0.8
    lr = lr_init
    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))
    no_improve = 0
    eval_count = 0

        if time.monotonic() > deadline:
            break
        
        c = fft_self_conv(a)
        M = c.max()

        else:

            grad[idx] += a[k - idx]

import numpy as np

import scipy.sparse as sp

def fft_self_conv(a):
    n = len(a)
    if n <= 1:
        return a.copy()

    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)

def fast_score(a):

def normalize(a, target_sum):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:

def build_conv_matrix(n, a):
    total_nz = n * n
    data = np.empty(total_nz, dtype=np.float64)
    indices = np.empty(total_nz, dtype=np.int32)

        prev = np.asarray(height_sequence_1, dtype=np.float64)
        prev = np.clip(prev, 0.0, 1000.0)
        if prev.size > 0 and np.isfinite(prev).all() and prev.sum() > 0.01:
            prev_score = float(evaluate_sequence(prev.tolist()))
            if np.isfinite(prev_score):
                best_a = prev.copy()
                best_score = prev_score
                print(f"Previous best: score={prev_score:.8f}, n={prev.size}")
        else:
            prev = None
    except Exception:
        prev = None

    configs = [(4096, 35), (5120, 25), (3072, 40), (2048, 45)]
    time_per_config = (budget_s * 0.72) / len(configs)

    for ci, (n, n_lp) in enumerate(configs):
        cfg_start = time.monotonic()
        cfg_deadline = min(deadline - 80, cfg_start + time_per_config)
        print(f"\n=== Config {ci}: n={n}, lp_iters={n_lp} ===")

        a = None
        if prev is not None:
            x_old = np.linspace(0.0, 1.0, len(prev))
            x_new = np.linspace(0.0, 1.0, n)
            from scipy.interpolate import PchipInterpolator
            try:
                f_interp = PchipInterpolator(x_old, prev)
                a = np.maximum(f_interp(x_new), 0.0)
                a = normalize(a, n)
            except Exception:
                a = None
        if a is None:
            best_init_sc = np.inf
            for _ in range(8):
                a_try = parametric_init(n, rng)
                sc = fast_score(a_try)
                if sc < best_init_sc:
                    best_init_sc = sc
                    a = a_try
            a_uni = 1.0 + 0.25 * rng.standard_normal(n)
            a_uni = normalize(a_uni, n)
            if fast_score(a_uni) < fast_score(a):
                a = a_uni

        sc_init = float(evaluate_sequence(a.tolist()))
        print(f"  Init: score={sc_init:.8f}")
        if np.isfinite(sc_init) and sc_init < best_score:
            best_score = sc_init
            best_a = a.copy()

        # LP phase
        t_blend = 0.12
        for lp_it in range(n_lp):
            if time.monotonic() > cfg_deadline - 60:
                break
            c = fft_self_conv(a)
            M = c.max()
            b = lp_direction(a, M, time_limit=25)
            if b is None:
                print(f"  LP it={lp_it}: solve failed")
                break
            a_new = (1.0 - t_blend) * a + t_blend * b
            a = normalize(a_new, n)
            if lp_it % 5 == 0 or lp_it == n_lp - 1:
                sc = fast_score(a)
                exact_sc = float(evaluate_sequence(a.tolist()))
                c2 = fft_self_conv(a)
                print(f"  LP it={lp_it}: fast={sc:.8f}, exact={exact_sc:.8f}, M={c2.max():.4f}")
                if np.isfinite(exact_sc) and exact_sc < best_score:
                    best_score = exact_sc
                    best_a = a.copy()

        # Gradient phase
        gd_deadline = cfg_deadline - 10
        a = best_a[:n].copy() if best_a is not None and len(best_a) == n else a.copy()
        a_gd, sc_gd = gradient_descent(a, gd_deadline, peak_count=3, lr_init=0.12)
        if sc_gd < best_score:
            best_score = sc_gd
            best_a = a_gd.copy()
            print(f"  After GD: score={best_score:.8f}")

        elapsed = time.monotonic() - cfg_start
        print(f"  Config {ci} done in {elapsed:.1f}s, best={best_score:.8f}")

    # Final polish
    if best_a is not None and time.monotonic() < deadline - 20:
        print(f"\n=== Final polish ===")
        n_f = len(best_a)
        a_final = best_a.copy()
        a_final, sc_final = gradient_descent(a_final, deadline - 8, peak_count=2,
                                             lr_init=0.02, eval_every=20)
        if sc_final < best_score:
            best_score = sc_final
            best_a = a_final.copy()
        final_sc = float(evaluate_sequence(best_a.tolist()))
        if np.isfinite(final_sc) and final_sc < best_score:
            best_score = final_sc
        print(f"FINAL: score={best_score:.8f}, n={len(best_a)}")

    if best_a is None:
        best_a = np.ones(512)
    return [float(x) for x in best_a]
