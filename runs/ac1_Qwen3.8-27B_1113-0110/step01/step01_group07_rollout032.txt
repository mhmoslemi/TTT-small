def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def normalize_positive(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s


def fast_score(a):
    p = normalize_positive(a)
    if p is None:
        return np.inf
    n = p.size
    if n <= 0:
        return np.inf
    L = next_pow2(2 * n - 1)
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0.0:
        return np.inf
    return float(2.0 * n * cm)


def softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = np.where(np.isfinite(z), z, -50.0)
    z = z - z.max()
    z = np.clip(z, -50.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        e = np.ones_like(z)
        s = float(e.size)
    return e / s


def compute_gradient(p, tau, L, n):
    """Compute gradient of smooth max objective in simplex tangent space."""
    P = np.fft.rfft(p, L)
    c = np.fft.irfft(P * P, L)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)

    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0.0:
        return np.inf, np.inf, np.zeros(n, dtype=np.float64)

    z = (c - cm) / tau
    # Clip for numerical stability
    z = np.clip(z, -30.0, 0.0)
    expz = np.exp(z)
    expsum = float(expz.sum())
    if not np.isfinite(expsum) or expsum <= 0.0:
        expz = np.zeros_like(c)
        expz[int(np.argmax(c))] = 1.0
        expsum = 1.0

    value_coeff = cm + tau * float(np.log(expsum))
    w = expz / expsum  # weights for each lag

    # Compute gradient via FFT
    Wref = w[::-1]
    H = np.fft.irfft(P * np.fft.rfft(Wref, L), L)
    g_p = 2.0 * H[n - 1:2 * n - 1][::-1]
    g_p = np.where(np.isfinite(g_p), g_p, 0.0)

    # Project onto simplex tangent space (sum = 0)
    mean_g = float(np.dot(p, g_p))
    grad = g_p - p * mean_g

    # Clamp gradient for stability
    max_g = float(np.max(np.abs(grad)))
    if max_g > 0.5:
        grad *= (0.5 / max_g)

    true_score = 2.0 * n * cm
    smooth_score = 2.0 * n * value_coeff
    return float(true_score), float(smooth_score), grad


def adam_optimize(p_init, duration_s, rng, n, L, tau_ratio0, lr0,
                  tau_decay_interval, tau_decay_factor,
                  lr_decay_interval, lr_decay_factor,
                  restart_threshold, verbose=False):
    """Run Adam optimizer in log domain with adaptive tau."""
    deadline = time.monotonic() + max(1.0, float(duration_s))
    p = normalize_positive(p_init)
    if p is None:
        return None, np.inf

    z = np.log(np.maximum(p, 1e-300))
    z = np.where(np.isfinite(z), z, -50.0)
    z -= z.mean()
    z = np.clip(z, -50.0, 50.0)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    bias_step = 0

    current_tau_ratio = float(tau_ratio0)
    lr = float(lr0)
    best_p = p.copy()
    best_score = fast_score(p)
    no_improve = 0
    step = 0
    t_start = time.monotonic()

    while time.monotonic() < deadline:
        tau = max(1e-14, current_tau_ratio / float(n))
        true_s, smooth_s, grad = compute_gradient(p, tau, L, n)
        if not np.isfinite(true_s):
            break

        if true_s < best_score - 1e-12:
            best_score = true_s
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Decay schedule
        if step > 0 and (step + 1) % tau_decay_interval == 0:
            current_tau_ratio = max(0.0005, current_tau_ratio * tau_decay_factor)
        if step > 0 and (step + 1) % lr_decay_interval == 0:
            lr = max(0.0005, lr * lr_decay_factor)

        # Restart mechanism
        if no_improve >= restart_threshold:
            # Structured restart: smooth noise
            freq = rng.integers(1, 8)
            phase = rng.uniform(0, 2 * np.pi)
            amplitude = rng.uniform(0.05, 0.20)
            idx = np.arange(n)
            smooth_noise = amplitude * np.sin(2 * np.pi * freq * idx / n + phase)
            smooth_noise += rng.normal(0, 0.03, n)
            zz = np.log(np.maximum(best_p, 1e-300))
            zz = np.where(np.isfinite(zz), zz, -50.0)
            zz += smooth_noise
            zz -= zz.mean()
            z = np.clip(zz, -50.0, 50.0)
            m *= 0.1
            v *= 0.1
            no_improve = 0
            current_tau_ratio = max(0.003, current_tau_ratio * 1.5)
            lr = max(0.005, lr * 1.3)

        # Adam update
        bias_step += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        b1 = 1.0 - beta1 ** bias_step
        b2 = 1.0 - beta2 ** bias_step
        z = z - lr * (m / b1) / (np.sqrt(v / b2) + eps)
        z = np.where(np.isfinite(z), z, 0.0)
        z -= z.mean()
        z = np.clip(z, -50.0, 50.0)
        p = softmax(z)
        step += 1

        if verbose and step % 10000 == 0:
            elapsed = time.monotonic() - t_start
            print(f"  adam step {step} true {true_s:.6f} best {best_score:.6f} "
                  f"tau_r {current_tau_ratio:.4f} lr {lr:.4f} "
                  f"noimp {no_improve} elapsed {elapsed:.0f}s")

    return best_p, best_score


def symmetric_adam_optimize(p_init, duration_s, rng, tau_ratio0, lr0,
                            tau_decay_interval, tau_decay_factor,
                            lr_decay_interval, lr_decay_factor,
                            restart_threshold):
    """Adam optimization with symmetry constraint a[i] = a[n-1-i]."""
    p = normalize_positive(p_init)
    if p is None:
        return None, np.inf
    n = p.size
    if n < 4:
        return p, fast_score(p)

    half = (n + 1) // 2
    # Extract symmetric parameters: z[0:half] determines the full z
    z_full = np.log(np.maximum(p, 1e-300))
    z_full = np.where(np.isfinite(z_full), z_full, -50.0)
    z_full -= z_full.mean()
    z_full = np.clip(z_full, -50.0, 50.0)
    z_half = z_full[:half].copy()

    # Build full z from half
    def build_full_z(zh):
        if n % 2 == 0:
            return np.concatenate([zh, zh[::-1]])
        else:
            return np.concatenate([zh, zh[-2::-1]])

    L = next_pow2(2 * n - 1)
    p = softmax(build_full_z(z_half))

    m = np.zeros(half, dtype=np.float64)
    v = np.zeros(half, dtype=np.float64)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    bias_step = 0

    current_tau_ratio = float(tau_ratio0)
    lr = float(lr0)
    best_p = p.copy()
    best_score = fast_score(p)
    no_improve = 0
    step = 0
    deadline = time.monotonic() + max(1.0, float(duration_s))

    while time.monotonic() < deadline:
        tau = max(1e-14, current_tau_ratio / float(n))
        true_s, smooth_s, grad_full = compute_gradient(p, tau, L, n)
        if not np.isfinite(true_s):
            break

        if true_s < best_score - 1e-12:
            best_score = true_s
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        if step > 0 and (step + 1) % tau_decay_interval == 0:
            current_tau_ratio = max(0.0005, current_tau_ratio * tau_decay_factor)
        if step > 0 and (step + 1) % lr_decay_interval == 0:
            lr = max(0.0005, lr * lr_decay_factor)

        if no_improve >= restart_threshold:
            freq = rng.integers(1, 6)
            phase = rng.uniform(0, 2 * np.pi)
            amplitude = rng.uniform(0.03, 0.15)
            idx = np.arange(half)
            noise = amplitude * np.sin(2 * np.pi * freq * idx / half + phase)
            noise += rng.normal(0, 0.02, half)
            zh_best = np.log(np.maximum(best_p[:half], 1e-300))
            zh_best = np.where(np.isfinite(zh_best), zh_best, -50.0)
            zh_best += noise
            z_half = zh_best
            m *= 0.1
            v *= 0.1
            no_improve = 0
            current_tau_ratio = max(0.003, current_tau_ratio * 1.5)
            lr = max(0.005, lr * 1.3)

        # Get gradient for half parameters
        grad_half = grad_full[:half].copy()
        if n % 2 == 0:
            grad_half = grad_half + grad_full[n-1:n-half-1:-1]
        else:
            grad_half[-1] += grad_full[n-1]  # center element
            if half > 1:
                grad_half[1:] += grad_full[n-1:n-half:-1]

        bias_step += 1
        m = beta1 * m + (1.0 - beta1) * grad_half
        v = beta2 * v + (1.0 - beta2) * (grad_half * grad_half)
        b1 = 1.0 - beta1 ** bias_step
        b2 = 1.0 - beta2 ** bias_step
        z_half = z_half - lr * (m / b1) / (np.sqrt(v / b2) + eps)
        z_half = np.where(np.isfinite(z_half), z_half, 0.0)
        z_full_arr = build_full_z(z_half)
        z_full_arr -= z_full_arr.mean()
        z_full_arr = np.clip(z_full_arr, -50.0, 50.0)
        p = softmax(z_full_arr)
        # Re-extract half
        z_half = np.log(np.maximum(p[:half], 1e-300))
        z_half = np.where(np.isfinite(z_half), z_half, -50.0)

        step += 1

    return best_p, best_score


def resize_sequence(seq, m):
    p = normalize_positive(seq)
    m = int(m)
    if p is None:
        return np.ones(m, dtype=np.float64) / float(m)
    n = p.size
    if n == m:
        return p.copy()
    if n == 1:
        return np.full(m, 1.0 / float(m), dtype=np.float64)

    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, m)
    out = np.interp(x_new, x_old, p)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(m, dtype=np.float64) / float(m)
    return out / s


def scoped_cheb_parent(n, D=10, mode=0, eps=0.1):
    """Create a parent from Chebyshev-like basis."""
    x = np.linspace(0.0, 1.0, n)
    t = 2.0 * x - 1.0
    t = np.clip(t, -1.0, 1.0)
    # Use cos basis
    val = np.zeros(n)
    for k in range(1, D + 1):
        coeff = 0.5 / k  # decaying coefficients
        val += coeff * np.cos(k * np.arccos(t))
    if mode == 0:
        h = 1.0 + eps + val
    else:
        h = np.exp(np.clip(val * 2, -6, 6))
    return normalize_positive(h)


def try_scipy_smooth(p_init, n, L, duration_s):
    """Try scipy L-BFGS-B on the smooth max in log domain."""
    if not HAVE_SCIPY:
        return None, np.inf

    p = normalize_positive(p_init)
    if p is None:
        return None, np.inf

    deadline = time.monotonic() + max(1.0, float(duration_s))
    z0 = np.log(np.maximum(p, 1e-300))
    z0 = np.where(np.isfinite(z0), z0, -50.0)
    z0 -= z0.mean()
    z0 = np.clip(z0, -30.0, 30.0)

    tau = 0.02 / n

    def objective(z):
        if time.monotonic() > deadline:
            return np.inf
        zc = np.clip(z, -30.0, 30.0)
        pp = softmax(zc)
        tau_local = max(1e-14, tau / n)
        true_s, smooth_s, grad = compute_gradient(pp, tau_local, L, n)
        if not np.isfinite(true_s):
            return 1e10
        return smooth_s

    def grad_func(z):
        if time.monotonic() > deadline:
            return np.zeros_like(z)
        zc = np.clip(z, -30.0, 30.0)
        pp = softmax(zc)
        tau_local = max(1e-14, tau / n)
        true_s, smooth_s, grad = compute_gradient(pp, tau_local, L, n)
        return grad

    try:
        res = scipy_minimize(
            objective, z0, jac=grad_func, method='L-BFGS-B',
            options={'maxiter': 5000, 'maxfun': 10000, 'gtol': 1e-8,
                     'maxls': 50}
        )
        if res.success or hasattr(res, 'fun') and np.isfinite(res.fun):
            z_best = np.clip(res.x, -30.0, 30.0)
            p_best = softmax(z_best)
            s_best = fast_score(p_best)
            return p_best, s_best
    except Exception as e:
        print(f"  scipy failed: {e}")

    return None, np.inf

        print(f"  Adam run 2 (refinement): {alloc2:.0f}s, starting from {fast_score(p_current):.6f}")
        p2, s2 = adam_optimize(
            p_current, alloc2, rng, n_eff, L,
            tau_ratio0=0.025, lr0=0.03,
            tau_decay_interval=4000, tau_decay_factor=0.93,
            lr_decay_interval=5000, lr_decay_factor=0.96,
            restart_threshold=6000, verbose=True
        )
        if p2 is not None and s2 < global_best_score:
            global_best_score = s2
            global_best_p = p2.copy()
            print(f"  New global best: {s2:.6f}")

        # Adam run 3: Very small tau (polish)
        p_current = resize_sequence(global_best_p if global_best_p is not None else p_init, N)
        print(f"  Adam run 3 (polish): {alloc3:.0f}s, starting from {fast_score(p_current):.6f}")
        p3, s3 = adam_optimize(
            p_current, alloc3, rng, n_eff, L,
            tau_ratio0=0.01, lr0=0.01,
            tau_decay_interval=5000, tau_decay_factor=0.90,
            lr_decay_interval=4000, lr_decay_factor=0.97,
            restart_threshold=8000, verbose=True
        )
        if p3 is not None and s3 < global_best_score:
            global_best_score = s3
            global_best_p = p3.copy()
            print(f"  New global best: {s3:.6f}")

        # Symmetric variant (if time remains)
        rem_phase = deadline - time.monotonic()
        if rem_phase > 120:
            p_sym_init = resize_sequence(global_best_p if global_best_p is not None else p_init, N)
            # Symmetrize
            p_sym = 0.5 * (p_sym_init + p_sym_init[::-1])
            p_sym = normalize_positive(p_sym)
            s_sym_init = fast_score(p_sym)
            print(f"  Symmetric variant: starting from {s_sym_init:.6f}")
            alloc_sym = min(rem_phase * 0.5, 300)
            ps, ss = symmetric_adam_optimize(
                p_sym, alloc_sym, rng,
                tau_ratio0=0.03, lr0=0.04,
                tau_decay_interval=3000, tau_decay_factor=0.93,
                lr_decay_interval=5000, lr_decay_factor=0.95,
                restart_threshold=4000
            )
            if ps is not None and ss < global_best_score:
                global_best_score = ss
                global_best_p = ps.copy()
                print(f"  New global best (symmetric): {ss:.6f}")

        print(f"  Phase complete. Best so far: {global_best_score:.6f}")

    # --- Phase 2: Try scipy L-BFGS-B on the best found ---
    if HAVE_SCIPY and global_best_p is not None and time.monotonic() < deadline - 30:
        rem = deadline - time.monotonic() - 30
        if rem > 60:
            N_sc = global_best_p.size
            L_sc = next_pow2(2 * N_sc - 1)
            print(f"\n=== SciPy L-BFGS-B on N={N_sc}, budget={rem:.0f}s ===")
            p_sc, s_sc = try_scipy_smooth(global_best_p, N_sc, L_sc, rem)
            if p_sc is not None and s_sc < global_best_score:
                global_best_score = s_sc
                global_best_p = p_sc.copy()
                print(f"  New global best (scipy): {s_sc:.6f}")

    # --- Phase 3: Final validation with evaluate_sequence ---
    print(f"\n=== Final validation ===")
    candidates = []
    if global_best_p is not None:
        candidates.append(global_best_p)

    # Also validate the original seed at its native length
    try:
        hs = np.asarray(height_sequence_1, dtype=np.float64)
        hs = np.where(np.isfinite(hs), hs, 0.0)
        hs = np.maximum(hs, 0.0)
        if hs.size > 0 and float(hs.sum()) > 1e-9:
            candidates.append(hs / hs.sum())
    except Exception:
        pass

    # Uniform fallback
    candidates.append(np.ones(1024, dtype=np.float64) / 1024.0)

    final_seq = None
    final_score = np.inf

    for cand in candidates:
        cand = np.asarray(cand, dtype=np.float64)
        cand = np.where(np.isfinite(cand), cand, 0.0)
        cand = np.maximum(cand, 0.0)
        lst = [float(x) for x in cand]
        if not lst:
            continue
        try:
            sc = evaluate_sequence(lst)
        except Exception:
            sc = np.inf
        if np.isfinite(sc) and sc < final_score:
            final_score = float(sc)
            final_seq = lst
            print(f"  Candidate len={len(lst)}: score={sc:.6f}")

    if final_seq is None:
        final_seq = [1.0 / 1024.0] * 1024
        final_score = 2.0

    total_time = time.monotonic() - start_time
    print(f"\nFinal result: score={final_score:.6f}, length={len(final_seq)}, time={total_time:.0f}s")
    return final_seq
