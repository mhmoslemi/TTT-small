```python
import numpy as np
import time
from scipy.optimize import linprog, minimize

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]

def _score(a):
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)

def _sensitivity_matrix(a, top_k_indices):
    """Compute sensitivity of top-k autocorrelation lags to each coordinate."""
    n = len(a)
    j = np.arange(n)
    S = np.zeros(n)
    for k in top_k_indices:
        k = int(k)
        idx1 = k - j
        idx2 = j - k
        m1 = (idx1 >= 0) & (idx1 < n)
        m2 = (idx2 >= 0) & (idx2 < n)
        S[m1] += a[idx1[m1]]
        S[m2] += a[idx2[m2]]
    return S

def _smooth_optimize(a_init, n_anneal, T_start, T_end, T_decay, lr_init, lr_decay, deadline, verbose=True):
    """Smooth optimization using log-sum-exp approximation with annealing."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_score = _score(a)
    
    T = T_start
    lr = lr_init
    
    for it in range(n_anneal):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        # Log-sum-exp
        c_max = np.max(c)
        log_sum_exp = c_max + T * np.log(np.sum(np.exp((c - c_max) / T)))
        f_T = log_sum_exp  # This approximates max(c) + T*log(2n-1)
        
        # Score with actual max
        score = _score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            if verbose and it % 50 == 0:
                print(f"      iter {it}: T={T:.4f}, score={best_score:.6f}")
        
        # Gradient of f_T
        p = np.exp((c - c_max) / T)
        p /= np.sum(p)
        
        # Sensitivity: grad_j = sum_k p_k * s_j(k)
        # Get top contributing lags for efficiency
        n_peaks = min(50, len(c))
        top_k = np.argpartition(c, -n_peaks)[-n_peaks:]
        # Actually for the gradient we need all k weighted by p_k
        # But for large n, most p_k are negligible
        # Let's use all for accuracy
        j = np.arange(n)
        grad = np.zeros(n)
        for k in range(len(c)):
            if p[k] < 1e-10:
                continue
            idx1 = k - j
            idx2 = j - k
            m1 = (idx1 >= 0) & (idx1 < n)
            m2 = (idx2 >= 0) & (idx2 < n)
            grad[m1] += p[k] * a[idx1[m1]]
            grad[m2] += p[k] * a[idx2[m2]]
        
        # Update
        a = a - lr * grad
        a = np.maximum(a, 0.0)
        s_new = float(np.sum(a))
        if s_new < 1e-10:
            a = np.ones(n) / n
            s_new = 1.0
        a = a * s / s_new  # Maintain sum = 1
        
        T *= T_decay
        lr *= lr_decay
    
    return best_a, best_score
```

```python
import numpy as np
import time
from scipy.optimize import linprog

def _fft_convolve(a):
    """Fast full linear convolution of a with itself."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]

def _score(a):
    """J(a) = 2n * max(conv(a,a)) / sum(a)^2"""
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)

def _build_conv_matrix(f):
    """Convolution matrix for f: (C @ g)[k] = (f*g)[k]"""
    n = len(f)
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        C[j:j+n, j] = f
    return C

def _lp_solve(f, M):
    """Solve max sum(g) s.t. C_f @ g <= M, g >= 0"""
    n = len(f)
    C = _build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.status == 0:
            return result.x
    except Exception:
        pass
    return None

def _compute_grad(a, p):
    """Gradient of log-sum-exp objective w.r.t. a."""
    n = len(a)
    term1 = np.convolve(p, a[::-1], 'full')[n-1:2*n-1]
    term2 = np.convolve(p, a, 'full')[:n]
    return term1 + term2

def _smooth_optimize(a_init, max_iters, T_start, T_end, deadline, n_peaks=50):
    """Smooth optimization with log-sum-exp and annealing."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_score = _score(a)
    
    T = T_start
    # Learning rate schedule
    lr = 1.0
    decay = (T_end / T_start) ** (1.0 / max_iters)
    
    # Momentum
    v = np.zeros(n)
    mu = 0.9
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        score = _score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Softmax weights
        c_max = np.max(c)
        exp_c = np.exp((c - c_max) / T)
        p = exp_c / np.sum(exp_c)
        
        # Gradient
        grad = _compute_grad(a, p)
        
        # Update with momentum
        v = mu * v - lr * grad
        a = a + v
        a = np.maximum(a, 0.0)
        s_new = float(np.sum(a))
        if s_new < 1e-10:
            a = np.ones(n) / n
            s_new = 1.0
        a = a / s_new
        
        T *= decay
        # Adaptive lr
        if it % 100 == 0 and it > 0:
            # Check if we're making progress
            pass
    
    return best_a, best_score

def _alternating_lp(a_init, max_iters, deadline, initial_blend=0.8):
    """Alternating LP optimization."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = _score(a)
    blend = initial_blend
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        M = float(np.max(c))
        g = _lp_solve(a, M)
        if g is None:
            break
        sum_g = float(np.sum(g))
        if sum_g < 1e-12:
            break
        g = g / sum_g
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        sum_new = float(np.sum(a_new))
        if sum_new < 1e-12:
```

```python
"""
Multi-phase optimizer for minimizing J(a) = 2n*max(conv(a,a))/sum(a)^2.
Uses log-sum-exp smooth optimization for fast convergence, combined with
alternating LP refinement and dual-gradient updates for precision.
"""
import numpy as np
import time
from scipy.optimize import linprog


def _fft_convolve(a):
    """Fast full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]


def _score(a):
    """Compute J(a) = 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)


def _build_conv_matrix(f):
    """Build convolution matrix C where (C @ g)[k] = (f*g)[k]."""
    n = len(f)
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = f
    return C


def _lp_solve(f, M):
    """Solve max sum(g) s.t. C_f @ g <= M, g >= 0. Returns (g, dual_y) or (None, None)."""
    n = len(f)
    C = _build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.status == 0:
            dual_y = None
            try:
                dual_y = np.abs(result.ineqlin.marginals)
            except Exception:
                pass
            return result.x, dual_y
    except Exception:
        pass
    return None, None


def _compute_grad(a, p):
    """Compute gradient of sum_k p(k)*c(k) w.r.t. a, using FFT-based convolution."""
    n = len(a)
    # term1(j) = sum_m a(m)*p(j+m) = (p * a_rev)(j + n - 1)
    term1 = np.convolve(p, a[::-1], 'full')[n - 1:2 * n - 1]
    # term2(j) = sum_m a(m)*p(j-m) = (p * a)(j)
    term2 = np.convolve(p, a, 'full')[:n]
    return term1 + term2


def _smooth_optimize(a_init, max_iters, T_start, T_end, deadline):
    """
    Smooth optimization using log-sum-exp approximation with temperature annealing
    and momentum. Minimizes max(conv(a,a)) subject to sum(a)=1, a>=0.
    """
    a = a_init.copy().astype(np.float64)
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_score = _score(a)

    T = T_start
    lr = 0.5
    v = np.zeros(n)
    momentum = 0.85
    decay_T = (T_end / T_start) ** (1.0 / max(1, max_iters))
    decay_lr = 0.9998

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        score = 2.0 * n * float(np.max(c))  # sum(a)=1
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Softmax weights (numerically stable)
        c_max = float(np.max(c))
        exp_c = np.exp((c - c_max) / T)
        p = exp_c / np.sum(exp_c)

        # Gradient via FFT-based convolution
        grad = _compute_grad(a, p)

        # Momentum update
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
            a = a_new

        T = max(T_end, T * decay_T)
        lr *= decay_lr

        # Reduce momentum as T decreases (more aggressive steps)
        momentum = max(0.7, 0.85 - 0.1 * (1.0 - T / T_start))

    return best_a, best_score

def _dual_gradient_update(a, c, M, y, lr):

    n = len(a)

    a_opt, score = _smooth_optimize(a_start, 10000, 0.04, 0.002, phase1_deadline)
    print(f"  From parent: score={score:.6f}")
    if score < best_score:
        best_score = score
        best_a = a_opt.copy()
        print(f"  ** New best: {best_score:.6f}")

    # Diverse initializations
    for v in [2, 3, 4, 5, 6, 9]:
        if time.monotonic() > phase1_deadline:
            break
        a_init = _make_initial(n_main, v, rng)
        a_opt, score = _smooth_optimize(a_init, 6000, 0.04, 0.002, phase1_deadline)
        print(f"  Init v{v}: score={score:.6f}")
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  ** New best: {best_score:.6f}")

    # === Phase 2: LP refinement at n=500 ===
    print(f"\n=== Phase 2: LP refinement (n={n_main}) ===")
    phase2_deadline = min(total_deadline, time.monotonic() + 800.0)
    print(f"  Starting LP from score={best_score:.6f}")

    # Multiple LP runs with different blend factors
    for blend_init in [0.9, 0.7, 0.5]:
        if time.monotonic() > phase2_deadline:
            break
        a_opt, score = _lp_with_dual(best_a.copy(), 500, phase2_deadline, initial_blend=blend_init)
        print(f"  LP blend={blend_init}: score={score:.6f}")
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  ** New best: {best_score:.6f}")

    # === Phase 3: Try other sequence lengths ===
    print(f"\n=== Phase 3: Multi-n exploration ===")
    for n_try in [350, 400, 450, 550, 600, 700]:
        if time.monotonic() > total_deadline - 300:
            break
        phase3_deadline = min(total_deadline - 200, time.monotonic() + 250.0)
        a_start = _interpolate(best_a, n_try)

        # Quick smooth opt
        a_opt, score_s = _smooth_optimize(a_start, 4000, 0.03, 0.002, phase3_deadline)
        # LP polish
        a_opt, score = _alternating_lp(a_opt, 300, phase3_deadline, initial_blend=0.6)
        print(f"  n={n_try}: score={score:.6f}")
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  ** New best: {best_score:.6f}")

    # === Phase 4: Final aggressive polish ===
    print(f"\n=== Phase 4: Final polish ===")
    final_deadline = total_deadline - 20

    # Work at the best n found
    n_best = len(best_a)
    # For LP, use a size where it's efficient
    n_lp = min(n_best, 500)

    if n_lp != n_best:
        a_work = _interpolate(best_a, n_lp)
    else:
        a_work = best_a.copy()

    # Long LP run
    a_opt, score = _lp_with_dual(a_work, 800, final_deadline, initial_blend=0.6)
    if n_lp != n_best:
        a_opt = _interpolate(a_opt, n_best)
        score = _score(a_opt)
    print(f"  Final LP: score={score:.6f}")
    if score < best_score:
        best_score = score
        best_a = a_opt.copy()
        print(f"  ** New best: {best_score:.6f}")

    # One more smooth opt pass on the best
    if time.monotonic() < final_deadline - 30:
        a_opt, score = _smooth_optimize(best_a.copy(), 5000, 0.02, 0.001, final_deadline)
        print(f"  Final smooth: score={score:.6f}")
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  ** New best: {best_score:.6f}")

    # Final LP if time allows
    if time.monotonic() < final_deadline:
        n_work = min(len(best_a), 400)
        a_work = _interpolate(best_a, n_work) if n_work != len(best_a) else best_a.copy()
        a_opt, score = _alternating_lp(a_work, 300, final_deadline, initial_blend=0.5)
        if n_work != len(best_a):
            a_opt = _interpolate(a_opt, len(best_a))
            score = _score(a_opt)
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  ** Final LP best: {best_score:.6f}")

    # Final verification
    print(f"\n=== FINAL: score={best_score:.6f}, n={len(best_a)} ===")
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    verified = float(evaluate_sequence(result))
    print(f"Verified score: {verified:.6f}")
    return result
```
