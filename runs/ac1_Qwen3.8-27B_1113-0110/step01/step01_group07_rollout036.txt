"""
Optimized step function search using frequency-domain gradient descent
with adaptive tau annealing and cosine learning rate schedule.
Explores n=16384 with multiple initialization strategies.
"""

import time
import numpy as np


def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def normalize_positive(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s


def fast_score(a):
    p = normalize_positive(a)
    if p is None:
        return np.inf
    n = p.size
    L = next_pow2(2 * n - 1)
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    cm = float(c.max())
    return float(2.0 * n * cm) if np.isfinite(cm) and cm > 0 else np.inf


def compute_conv(p, L):
    n = p.size
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    return np.where(np.isfinite(c), c, 0.0)


def compute_smooth_grad(c, p, tau, L):
    """Returns (smooth_value, gradient) of tau*log(sum(exp(c/tau))) w.r.t. p."""
    n = p.size
    cm = float(c.max())
    z = (c - cm) / tau
    # Clip to avoid overflow
    z = np.clip(z, -50.0, 0.0)
    expz = np.exp(z)
    expsum = float(expz.sum())
    if not np.isfinite(expsum) or expsum <= 0:
        # Fallback to one-hot
        expz = np.zeros_like(c)
        expz[int(np.argmax(c))] = 1.0
        expsum = 1.0
    
    W = tau * expz / expsum  # probability weights
    smooth_val = cm + tau * float(np.log(expsum))
    
    # Gradient: 2 * cross_correlation(W, p)
    Wref = W[::-1]
    P = np.fft.rfft(p, L)
    Wf = np.fft.rfft(Wref, L)
    H = np.fft.irfft(P * Wf, L)
    g_p = 2.0 * H[n - 1:2 * n - 1][::-1]
    g_p = np.where(np.isfinite(g_p), g_p, 0.0)
    
    # Project onto simplex: subtract mean weighted by p
    mean_g = float(np.dot(p, g_p))
    grad = g_p - p * mean_g
    
    return smooth_val, grad


def hann_window(n, alpha=1.0):
    x = np.linspace(0.0, 1.0, n)
    w = (0.5 + 0.5 * np.cos(2 * np.pi * (x - 0.5))) ** alpha
    return w / w.sum()


def blackman_window(n):
    x = np.arange(n)
    M = n - 1
    w = 0.42 - 0.5 * np.cos(2 * np.pi * x / M) + 0.08 * np.cos(4 * np.pi * x / M)
    return w / w.sum()


def optimal_init(n):
    """Try multiple window shapes and return the best."""
    candidates = []
    for alpha in [0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 3.0]:
        candidates.append(hann_window(n, alpha))
    candidates.append(blackman_window(n))
    # Triangular
    x = np.arange(n)
    tri = 1.0 - np.abs(x - (n-1)/2.0) / ((n-1)/2.0)
    candidates.append(tri / tri.sum())
    # Bimodal
    x = np.linspace(0, 1, n)
    bimodal = 0.3 + np.sin(np.pi * x) ** 2
    candidates.append(bimodal / bimodal.sum())
    
    best_p = None
    best_s = np.inf
    for p in candidates:
        s = fast_score(p)
        if s < best_s:

            best_p = p
    return best_p, best_s

    best_p = p.copy()

def resize_to(seq, m):
    """Resize a sequence to length m using interpolation."""
    p = normalize_positive(seq)
    m = int(m)
    if p is None:
        return np.ones(m) / m
    n = p.size
    if n == m:
        return p.copy()
    if n == 1:
        return np.full(m, 1.0 / m)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, m)
    out = np.interp(x_new, x_old, p)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 0:
        return np.ones(m) / m
    return out / s

"""
Multi-phase optimizer for minimizing 2n*max(conv(p,p))/(sum p)^2.
Uses parametric global search (differential evolution) combined with
frequency-domain Adam gradient descent with adaptive tau annealing
and cosine learning rate schedules. Explores n=8192-16384.
"""

import time
import numpy as np


def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def normalize_positive(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s


def fast_score(a):
    p = normalize_positive(a)
    if p is None:
        return np.inf
    n = p.size
    L = next_pow2(2 * n - 1)
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0.0:
        return np.inf
    return float(2.0 * n * cm)


def compute_conv(p, L):
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)
    n = p.size
    c = c[:2 * n - 1]
    return np.where(np.isfinite(c), c, 0.0)


def compute_smooth_grad(c, p, tau, L):
    n = p.size
    cm = float(c.max())
    z = (c - cm) / tau
    z = np.clip(z, -50.0, 0.0)
    expz = np.exp(z)
    expsum = float(expz.sum())
    if not np.isfinite(expsum) or expsum <= 0.0:
        expz = np.zeros_like(c)
        expz[int(np.argmax(c))] = 1.0
        expsum = 1.0
    W = tau * expz / expsum
    smooth_val = cm + tau * float(np.log(expsum))
    Wref = W[::-1]
    P = np.fft.rfft(p, L)
    Wf = np.fft.rfft(Wref, L)
    H = np.fft.irfft(P * Wf, L)
    g_p = 2.0 * H[n - 1:2 * n - 1][::-1]
    g_p = np.where(np.isfinite(g_p), g_p, 0.0)
    mean_g = float(np.dot(p, g_p))
    grad = g_p - p * mean_g
    return smooth_val, grad


def resize_to(seq, m):
    p = normalize_positive(seq)
    m = int(m)
    if p is None:
        return np.ones(m, dtype=np.float64) / float(m)
    n = p.size
    if n == m:
        return p.copy()
    if n == 1:
        return np.full(m, 1.0 / float(m), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, m)
    out = np.interp(x_new, x_old, p)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(m, dtype=np.float64) / float(m)
    return out / s


def window_hann(n, alpha=1.0):
    x = np.linspace(0.0, 1.0, n)
    w = 0.5 + 0.5 * np.cos(2.0 * np.pi * (x - 0.5))
    w = w ** alpha
    w = np.maximum(w, 1e-15)
    return w / w.sum()


def window_blackman(n):
    x = np.arange(n, dtype=np.float64)
    M = float(n - 1)
    w = 0.42 - 0.5 * np.cos(2.0 * np.pi * x / M) + 0.08 * np.cos(4.0 * np.pi * x / M)
    w = np.maximum(w, 1e-15)
    return w / w.sum()


def window_optimal_flat(n):
    x = np.arange(n, dtype=np.float64)
    M = float(n - 1)
    a0, a1, a2, a3 = 0.35875, 0.48829, 0.14128, 0.01168
    w = a0 - a1 * np.cos(2.0 * np.pi * x / M) + a2 * np.cos(4.0 * np.pi * x / M) - a3 * np.cos(6.0 * np.pi * x / M)
    w = np.maximum(w, 1e-15)
    return w / w.sum()


def parametric_sequence(theta, n):
    """Generate a sequence from parametric coefficients.
    theta = [a1,...,aK, p1,...,pK, alpha] where K = n//2 - 1 effectively
    p(x) = (1 + sum_k a_k * cos(2*pi*k*x + p_k)) ** alpha
    """
    K = (len(theta) - 1) // 2
    alphas = theta[:K]
    phases = theta[K:2 * K]
    power = theta[2 * K] if len(theta) > 2 * K else 1.0
    x = np.linspace(0.0, 1.0, n)
    val = np.ones(n, dtype=np.float64)
    for k in range(K):
        val += alphas[k] * np.cos(2.0 * np.pi * (k + 1) * x + phases[k])
    val = np.maximum(val, 0.01)
    val = val ** power
    val = np.maximum(val, 1e-15)
    s = float(val.sum())
    return val / s


def parametric_score(theta, n):
    try:
        p = parametric_sequence(theta, n)
        if p is None:
            return np.inf
        return fast_score(p)
    except Exception:
        return np.inf


def differential_evolution_param(n, duration_s, rng, K=4):
    """Run simple differential evolution on parametric family."""
    deadline = time.monotonic() + duration_s
    n_params = 2 * K + 1
    # Bounds: alphas in [-0.8, 0.8], phases in [-pi, pi], power in [0.5, 4.0]
    lower = np.concatenate([-0.8 * np.ones(K), -np.pi * np.ones(K), [0.5]])
    upper = np.concatenate([0.8 * np.ones(K), np.pi * np.ones(K), [4.0]])
    
    pop_size = min(40, max(15, 2 * n_params + 10))
    pop = rng.uniform(lower, upper, size=(pop_size, n_params))
    scores = np.array([parametric_score(pop[i], n) for i in range(pop_size)])
    
    best_idx = int(np.argmin(scores))
    best_theta = pop[best_idx].copy()
    best_score = scores[best_idx]
    
    F = 0.8
    CR = 0.8
    gen = 0
    
    while time.monotonic() < deadline:
        gen += 1
        # Select three random distinct individuals
        a, b, c = rng.choice(pop_size, 3, replace=False)
        # Create trial vector
        r = rng.random()
        base = b if r < 0.5 else best_theta
        trial = base + F * (pop[a] - pop[c])
        trial = np.clip(trial, lower, upper)
        trial_score = parametric_score(trial, n)
        
        # Replace worst in population (or if better than its opponent)
        worst_idx = int(np.argmax(scores))
        if trial_score < scores[worst_idx]:
            pop[worst_idx] = trial
            scores[worst_idx] = trial_score
        
        if trial_score < best_score:
            best_score = trial_score
            best_theta = trial.copy()
        
        if gen % 100 == 0:
            print(f"    DE gen {gen}: best {best_score:.6f}")
    
    return parametric_sequence(best_theta, n), best_score


def adam_optimize(p_init, duration_s, rng, n, L,
                  tau_start=0.10, tau_end=0.003,
                  lr_max=0.035, warmup_frac=0.04,
                  grad_clip=0.04, noise_period=8000, noise_std=0.06):
    p = p_init.copy()
    z = np.log(np.maximum(p, 1e-300))
    z = np.where(np.isfinite(z), z, -25.0)
    z -= z.mean()
    z = np.clip(z, -25.0, 25.0)
    
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t = 0
    
    deadline = time.monotonic() + max(0.0, duration_s)
    best_p = p.copy()
    best_score = fast_score(p)
    steps_done = 0
    
    # Estimate steps per second for schedule
    est_sps = max(30.0, 250000.0 / n)
    est_total = max(100, int(est_sps * duration_s))
    
    while time.monotonic() < deadline:
        frac = min(1.0, t / float(est_total))
        if frac < warmup_frac:
            lr = lr_max * (frac / warmup_frac)
        else:
            progress = (frac - warmup_frac) / max(1e-10, 1.0 - warmup_frac)
            lr = lr_max * 0.5 * (1.0 + np.cos(np.pi * progress))
        lr = max(lr, 0.0005)
        
        tau = tau_start * (tau_end / tau_start) ** frac
        tau = max(tau, 1e-12)
        
        c = compute_conv(p, L)
        _, grad = compute_smooth_grad(c, p, tau, L)
        
        gmax = float(np.max(np.abs(grad)))
        if gmax > grad_clip and gmax > 0:
            grad *= grad_clip / gmax
        
        true_score = 2.0 * n * float(c.max())
        if np.isfinite(true_score) and true_score < best_score:
            best_score = true_score
            best_p = p.copy()
        
        if t > 0 and t % noise_period == 0:
            z += rng.normal(0.0, noise_std, n)
            z -= z.mean()
            z = np.clip(z, -25.0, 25.0)
            p = np.exp(z)
            p = np.where(np.isfinite(p), p, 0.0)
            s = float(p.sum())
            if s > 0:
                p /= s
        
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        b1 = 1.0 - beta1 ** t
        b2 = 1.0 - beta2 ** t
        z = z - lr * (m / b1) / (np.sqrt(v / b2) + eps)
        z = np.where(np.isfinite(z), z, 0.0)
        z -= z.mean()
        z = np.clip(z, -25.0, 25.0)
        p = np.exp(z)
        p = np.where(np.isfinite(p), p, 0.0)
        s = float(p.sum())
        if s > 0:
            p /= s
        
        steps_done += 1
        if steps_done % 20000 == 0:
            print(f"  adam t={t} tau={tau:.6f} lr={lr:.6f} score={true_score:.6f} best={best_score:.6f}")
    
    return best_p, best_score


def peak_shave(p, n, L, n_iters=50, eta=0.3):
    """Local peak shaving: reduce the convolution peak by adjusting p."""
    p = p.copy()
    best_p = p.copy()
    best_s = fast_score(p)
    
    for _ in range(n_iters):
        c = compute_conv(p, L)
        cm = float(c.max())
        peak_idx = int(np.argmax(c))
        
        # Gradient of c[peak_idx] w.r.t. p
        # dc_k/dp_j = 2 * p[k-j] (with 0<=k-j<n)
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        grad = np.zeros(n, dtype=np.float64)
        for j in range(lo, hi + 1):
            grad[j] = 2.0 * p[peak_idx - j]
        
        # Subtract mean to preserve sum
        mean_g = float(np.dot(p, grad))
        grad = grad - p * mean_g
        
        # Multiplicative update: p *= exp(-eta * grad / (p + eps))
        # Simpler: p -= eta * grad, then project
        p_new = p - eta * (eta / float(np.dot(grad, grad) + 1e-30)) * grad
        p_new = np.maximum(p_new, 0.0)
        s = float(p_new.sum())
        if s > 0:
            p_new /= s
            sc = fast_score(p_new)
            if sc < best_s:
                best_s = sc
                best_p = p_new.copy()
                p = p_new
            else:
                eta *= 0.7
        else:
            break
    
    return best_p, best_s
