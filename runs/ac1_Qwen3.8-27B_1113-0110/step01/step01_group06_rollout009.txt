def _pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _fft_square(a):
    n = int(a.size)
    L = 2 * n - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, n=N)
    b = np.fft.irfft(fa * fa, n=N).real[:L]
    return np.maximum(b, 0.0)


def _score_internal(a, b=None):
    if b is None:
        b = _fft_square(a)
    s = float(a.sum())
    if s <= 1e-300:
        return float("inf"), b, 0.0
    maxb = float(b.max())
    score = float(2.0 * a.size * maxb / (s * s))
    return score, b, maxb


def _grad_weights(w, a):
    """Compute 2*cross_correlation(w, a) for w of len 2n-1, a of len n."""
    n = int(a.size)
    L = int(w.size)
    N = _pow2(L + n - 1)
    Fw = np.fft.rfft(w, n=N)
    Fa = np.fft.rfft(a, n=N)
    c = np.fft.irfft(Fw * np.conj(Fa), n=N)[:n]
    return 2.0 * c


def _make_weights_soft(b, focus, eps=0.01):
    """Soft weights: exp(-focus * (max_b - b)/max_b)."""
    L = int(b.size)
    maxb = float(b.max())
    if maxb <= 1e-300:
        return np.full(L, 1.0 / L)
    z = (maxb - b) / maxb
    with np.errstate(over="ignore", under="ignore"):
        p = np.exp(-z * float(focus))
    s = float(p.sum())
    if s <= 1e-12:
        p = np.zeros(L)
        p[int(np.argmax(b))] = 1.0
    else:
        p = (1.0 - eps) * (p / s) + eps / L
    return p


def _make_weights_topk(b, k, temperature=1.0):
    """Weights concentrated on top-k peaks with temperature smoothing."""
    L = int(b.size)
    maxb = float(b.max())
    if maxb <= 1e-300:
        return np.full(L, 1.0 / L)
    threshold_idx = min(k, L)
    sorted_idx = np.argsort(b)[::-1][:threshold_idx]
    p = np.zeros(L)
    threshold = float(b[sorted_idx[-1]])
    for idx in sorted_idx:
        p[idx] = 1.0 + temperature * (b[idx] - threshold) / maxb
    p /= p.sum()
    return p


def _init_uniform(n):

def _init_sinusoidal(n, rng, n_freqs=8, amp_range=(0.1, 1.0)):
    """Initialize with a sum of sinusoids to create structure."""
    a = np.ones(n, dtype=np.float64)

        phi = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(*amp_range)
        x = np.arange(n, dtype=np.float64)
        a += amp * np.cos(2 * np.pi * f * x / n + phi)
    a = np.maximum(a, 1e-6)
    a /= a.sum()
    return a


def _init_block(n, rng, n_blocks=None):

    a += np.exp(rng.normal(0, 0.5, size=n)) * 0.1
    a = np.maximum(a, 1e-6)

    """Anti-palindromic initialization: a[i] + a[n-1-i] = 2/n."""
    m = n // 2
    if smooth:
        x = np.arange(m, dtype=np.float64)
        v = np.zeros(m)
        for f in (1, 3, 5, 7, 11, 13):
            v += (rng.normal(0, 1) / f) * np.sin(2 * np.pi * f * x / m + rng.uniform(0, 2 * np.pi))
    else:
        v = rng.normal(0, 1, size=m)
    v = v - v.mean()
    std = v.std()
    if std > 1e-12:
        v = 3.0 * v / std
    z = np.zeros(n)
    z[:m] = v
    if n % 2 == 0:
        z[m:] = -v[::-1]
    else:
        z[m + 1:] = -v[::-1]
        z[m] = 0.0
    z -= z.max()
    a = np.exp(z)
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    # Project to anti-palindromic constraint
    if n % 2 == 0:
        half = n // 2
        s = a[:half] + a[half:][::-1]
        target = a.sum() / n * 2
        if s.max() > 1e-12:
            a[:half] = (target - a[half:][::-1]) / 2
            a[half:] = (target - a[:half][::-1]) / 2
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def _optimize_single(a0, time_budget, rng, label="opt", verbose=False):
    """
    Main single-sequence optimizer using adaptive subgradient with
    multiple focus temperatures and periodic structural mutations.
    """
    a = np.asarray(a0, dtype=np.float64).copy()
    n = int(a.size)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        a = np.ones(n) / n
        s = 1.0
    a /= s
    a = np.maximum(a, 1e-15)
    a /= a.sum()

    best_score, _, _ = _score_internal(a)
    best_a = a.copy()

    # Multi-temperature state
    # We maintain 3 "channels" with different focus levels
    channels = [(a.copy(), 5.0), (a.copy(), 20.0), (a.copy(), 100.0)]

    t0 = time.monotonic()
    it = 0
    no_improve_global = 0
    max_no_improve = 800
    last_mutation = t0
    mutation_interval = 150.0
    eta_values = [0.5, 0.3, 0.1]

    while time.monotonic() - t0 < time_budget:
        now = time.monotonic()

        # Determine which channel to update this iteration
        ch_idx = it % 3
        ch_a, ch_focus = channels[ch_idx]
        eta = eta_values[ch_idx]

        b = _fft_square(ch_a)
        score, _, maxb = _score_internal(ch_a, b)

        if score < best_score - 1e-14:
            best_score = score
            best_a = ch_a.copy()
            no_improve_global = 0
        else:
            no_improve_global += 1

        if verbose and it % 2000 == 0:
            print(f"  [{label}] it={it} t={now-t0:.0f}s best={best_score:.6f} "
                  f"ch{ch_idx} sc={score:.6f} focus={ch_focus:.1f} eta={eta:.3f} noimp={no_improve_global}")

        # Periodic structural mutation
        if now - last_mutation > mutation_interval:
            last_mutation = now
            # Block swap mutation on the best
            n_blocks = rng.integers(2, max(3, n // 2048))
            a_mut = best_a.copy()
            block_size = n // n_blocks
            perm = rng.permutation(n_blocks)
            for i in range(n_blocks):
                s1, e1 = i * block_size, (i + 1) * block_size
                s2, e2 = perm[i] * block_size, (perm[i] + 1) * block_size
                a_mut[s1:e1], a_mut[s2:e2] = a_mut[s2:e2].copy(), a_mut[s1:e1].copy()
            a_mut = np.maximum(a_mut, 1e-14)
            a_mut /= a_mut.sum()
            # Inject into one channel
            ch_idx_mut = rng.integers(0, 3)
            channels[ch_idx_mut] = (a_mut, float(rng.choice([5.0, 20.0, 50.0, 200.0])))
            eta_values[ch_idx_mut] = 0.3

        # If stuck, reset this channel
        if no_improve_global > max_no_improve * (1 + ch_idx):
            channels[ch_idx] = (best_a * np.exp(rng.normal(0, 0.4, size=n)),
                                float(rng.choice([10.0, 50.0, 200.0])))
            channels[ch_idx][0] = np.maximum(channels[ch_idx][0], 1e-14)
            channels[ch_idx][0] /= channels[ch_idx][0].sum()
            eta_values[ch_idx] = 0.5
            no_improve_global = 0
            max_no_improve = int(max_no_improve * 1.15)
            continue

        # Compute gradient
        w = _make_weights_soft(b, ch_focus)
        grad = _grad_weights(w, ch_a)
        # Center the gradient (for multiplicative update)
        mean_grad = float(np.dot(ch_a, grad))
        grad_c = grad - mean_grad
        maxg = float(np.max(np.abs(grad_c)))
        if maxg < 1e-18:
            # Try additive update instead
            ch_a = ch_a + rng.normal(0, 0.01 * ch_a.mean(), size=n)
            ch_a = np.maximum(ch_a, 1e-14)
            ch_a /= ch_a.sum()
            channels[ch_idx] = (ch_a, ch_focus)
            it += 1
            continue

        grad_c = grad_c / maxg

        # Multiplicative (exponential) update
        a_new = ch_a * np.exp(-eta * grad_c)
        a_new = np.maximum(a_new, 1e-300)
        ss = float(a_new.sum())
        if ss < 1e-100:
            a_new = ch_a * np.exp(rng.normal(0, 0.1, size=n))
            a_new = np.maximum(a_new, 1e-300)
            a_new /= a_new.sum()
        else:
            a_new /= ss

        b_new = _fft_square(a_new)
        score_new, _, _ = _score_internal(a_new, b_new)

        # Acceptance
        wt_old = float(np.dot(w, b))
        wt_new = float(np.dot(w, b_new))

        accept = False
        if score_new < best_score - 1e-14:
            accept = True
        elif score_new < score - 1e-14:
            accept = True
        elif wt_new < wt_old * 0.999:
            accept = True

        if accept:
            ch_a = a_new
            channels[ch_idx] = (ch_a, ch_focus)
            eta_values[ch_idx] = min(eta * 1.03, 5.0)
        else:
            eta_values[ch_idx] = max(eta * 0.78, 0.01)
            if eta_values[ch_idx] < 0.02:
                ch_a = best_a * np.exp(rng.normal(0, 0.2, size=n))
                ch_a = np.maximum(ch_a, 1e-14)
                ch_a /= ch_a.sum()
                channels[ch_idx] = (ch_a, float(rng.choice([5.0, 30.0, 100.0])))
                eta_values[ch_idx] = 0.3

        # Slowly increase focus
        channels[ch_idx] = (channels[ch_idx][0], min(channels[ch_idx][1] * 1.005, 500.0))

        it += 1

    return best_a, float(best_score)


def _optimize_refine(a0, time_budget, rng, label="refine", verbose=True):
    """Final refinement: high focus, small steps, very targeted."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = int(a.size)
    a = np.maximum(a, 1e-15)
    a /= a.sum()

    best_score, _, _ = _score_internal(a)
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    focus = 50.0
    eta = 0.05
    no_imp = 0

    while time.monotonic() - t0 < time_budget:
        b = _fft_square(a)
        score, _, maxb = _score_internal(a, b)

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        if verbose and it % 1000 == 0:
            print(f"  [{label}] it={it} t={time.monotonic()-t0:.0f}s best={best_score:.6f} "
                  f"cur={score:.6f} focus={focus:.1f} eta={eta:.4f} noimp={no_imp}")

        if no_imp > 2000:
            a = best_a * np.exp(rng.normal(0, 0.05, size=n))
            a = np.maximum(a, 1e-14)
            a /= a.sum()
            focus = rng.choice([30.0, 100.0, 300.0])
            eta = 0.08
            no_imp = 0
            continue

        # Use top-k weights for targeted reduction
        topk = min(5 + it // 5000, 50)
        w = _make_weights_topk(b, topk, temperature=0.5)
        grad = _grad_weights(w, a)
        mean_g = float(np.dot(a, grad))
        grad_c = grad - mean_g
        maxg = float(np.max(np.abs(grad_c)))
        if maxg < 1e-18:
            break
        grad_c /= maxg

        a_new = a * np.exp(-eta * grad_c)
        a_new = np.maximum(a_new, 1e-300)
        a_new /= a_new.sum()

        b_new = _fft_square(a_new)
        score_new, _, _ = _score_internal(a_new, b_new)

        wt_old = float(np.dot(w, b))
        wt_new = float(np.dot(w, b_new))

        if score_new < score - 1e-14 or (wt_new < wt_old * 0.9999):
            a = a_new
            eta = min(eta * 1.02, 0.5)
        else:
            eta = max(eta * 0.85, 0.005)
            if eta < 0.01:
                a = best_a + rng.normal(0, 0.005 * best_a.mean(), size=n)
                a = np.maximum(a, 1e-14)
                a /= a.sum()

        focus = min(focus * 1.003, 2000.0)
        it += 1

    return best_a, float(best_score)


def _try_lp_step(a, rng, max_time=60.0):
    """
    Try an LP-based improvement step.
    Minimize t s.t. cross_corr(g, a) <= t, sum(g)=1, g>=0.
    Then blend: new_a = (1-alpha)*a + alpha*g.
    """
    try:
        import cvxpy as cp
    except ImportError:
        return None, None

    n = int(a.size)
    # For large n, subsample constraints
    b = _fft_square(a)
    M = float(b.max())
    # Only use top constraints
    n_constraints = min(n, len(b))
    # Sort by weight to keep most impactful constraints
    k_sort = np.argsort(b)[::-1]
    n_keep = min(n_keep if (n_keep := min(2000, len(b))) else 2000, len(b))

    # Build LP: min t s.t. a^T (shifted g) <= t for key lags, sum g = 1, g >= 0
    # Cross-correlation: (a * g)[k] = sum_i a[i] g[k-i]
    # We need: for each k, sum_i a[i] * g[i-k+n-1]... this is complex.
    # Let's use: conv(a, g)[k] = sum_i a[i] g[k-i] for valid i, k-i in [0,n-1]
    # In matrix form: C[g] = T @ g where T[k, i] = a[k-i] (if valid)
    # We only keep top-m lags

    top_lags = k_sort[:min(500, len(b))]

    g_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()

    constraints = [cp.sum(g_var) == 1]

    # For each kept lag k: sum_i a[i] * g[k-i+n-1]... 
    # Actually conv(a,g)[k] = sum_{i} a[i] * g[k-i] where 0<=i<n, 0<=k-i<n
    # = sum_{i=max(0,k-n+1)}^{min(k, n-1)} a[i] * g[k-i]
    # So coefficient of g[j] in conv[k] is a[k-j] (if 0<=k-j<n)

    for k in top_lags:
        k = int(k)
        # conv(a,g)[k] = sum_{j=max(0,k-n+1)}^{min(k,n-1)} a[k-j] * g[j]
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_lo > j_hi:
            continue
        coeffs = a[k - j_hi:k - j_lo + 1][::-1]  # a[k-j] for j=j_lo..j_hi
        constraints.append(coeffs @ g_var <= t_var)

    obj = cp.Minimize(t_var)
    prob = cp.Problem(obj, constraints)

    try:
        prob.solve(solver=cp.CBC, max_time=max_time)
        if g_var.value is not None and t_var.value is not None:
            g_sol = np.asarray(g_var.value, dtype=np.float64)
            g_sol = np.maximum(g_sol, 0)
            g_sol /= g_sol.sum()
            # Blend
            for alpha in (0.1, 0.3, 0.5):
                a_new = (1 - alpha) * a + alpha * g_sol
                a_new = np.maximum(a_new, 1e-14)
                a_new /= a_new.sum()
                sc, _, _ = _score_internal(a_new)
                if sc < float("inf"):
                    return a_new, sc
    except Exception:
        pass

    return None, None


def _safe_evaluate(seq):
    ev = globals().get("evaluate_sequence")
    if ev is None or not callable(ev):
        return float("inf")
    try:
        return float(ev(seq))
    except Exception:
        return float("inf")


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main entry point: search for the best sequence within budget."""
    seed = int(seed)
    total_budget = max(120.0, float(budget_s))
    now0 = time.monotonic()
    final_deadline = now0 + total_budget - 60.0
    search_deadline = now0 + total_budget - 300.0

    print(f"=== Step-function optimizer starting. Budget={total_budget}s ===")
    print(f"Final deadline: {total_budget-60:.0f}s, Search deadline: {total_budget-300:.0f}s")

    best_a = None
    best_score = float("inf")

    # Check for given sequence
    given_a = None
    try:
        given = globals().get("height_sequence_1")
        if given is not None:
            g = np.asarray(given, dtype=np.float64).ravel()
            g = np.nan_to_num(g, nan=0.0, posinf=1000.0, neginf=0.0)
            g = np.clip(g, 0.0, 1000.0)
            s = float(g.sum())
            if s > 0.01 and g.size > 10:
                gn = g / s
                sc, _, _ = _score_internal(gn)
                if np.isfinite(sc):
                    print(f"Given sequence: n={gn.size}, internal_score={sc:.6f}")
                    best_score = min(best_score, sc)
                    best_a = gn.copy()
                if g.size <= 20000:
                    given_a = gn
    except Exception as e:
        print(f"Given seq load failed: {e}")

    # Phase 1: Generate and quickly optimize diverse candidates
    print("\n--- Phase 1: Diverse initializations ---")
    candidates = []

    # Size choices
    sizes = []
    if given_a is not None and given_a.size <= 20000:
        sizes.append(given_a.size)
    sizes.extend([8192, 16384])
    # Remove duplicates, keep sorted
    sizes = sorted(set(sizes))

    n_inits_per_size = 3

    for n in sizes:
        if time.monotonic() > search_deadline:
            break
        time_per_init = min(80.0, (search_deadline - time.monotonic()) / max(1, len(sizes) * n_inits_per_size))
        time_per_init = max(15.0, time_per_init)

        for init_idx in range(n_inits_per_size):
            if time.monotonic() > search_deadline:
                break
            rng = np.random.default_rng((seed * 7919 + n * 131 + init_idx) % (2**32))

            # Choose initialization type
            init_type = init_idx % 4
            if init_type == 0:
                a0 = _init_uniform(n)
                a0 = a0 + rng.normal(0, 0.1, size=n)
                a0 = np.maximum(a0, 1e-6)
                a0 /= a0.sum()
            elif init_type == 1:
                a0 = _init_sinusoidal(n, rng, n_freqs=rng.integers(4, 12))
            elif init_type == 2:
                a0 = _init_block(n, rng)
            else:
                a0 = _init_antipal(n, rng, smooth=(init_idx % 2 == 0))

            # Add given sequence as blend if available
            if given_a is not None and given_a.size == n:
                a0 = 0.5 * a0 + 0.5 * given_a
                a0 = np.maximum(a0, 1e-6)
                a0 /= a0.sum()

            sc0, _, _ = _score_internal(a0)
            print(f"  n={n} init{init_idx} ({['uniform+noise','sinusoidal','block','antipal'][init_type]}): "
                  f"score={sc0:.6f}, t={time.monotonic()-now0:.0f}s")

            # Quick optimization
            t_rem = min(time_per_init, search_deadline - time.monotonic())
            if t_rem < 10.0:
                break
            a_opt, sc_opt = _optimize_single(a0, t_rem, rng, label=f"n{n}_i{init_idx}", verbose=False)
            print(f"  -> n={n} init{init_idx} optimized: score={sc_opt:.6f}, t={time.monotonic()-now0:.0f}s")

            if np.isfinite(sc_opt):
                candidates.append((sc_opt, a_opt, n))
                if sc_opt < best_score:
                    best_score = sc_opt
                    best_a = a_opt.copy()

    # Sort candidates by score
    candidates.sort(key=lambda x: x[0])
    print(f"\nTop candidates after Phase 1: best={best_score:.6f}")
    for i, (sc, a, n) in enumerate(candidates[:5]):
        print(f"  #{i}: n={n}, score={sc:.6f}")

    # Phase 2: Intensive optimization on top candidates
    print("\n--- Phase 2: Intensive optimization ---")
    top_k = min(4, len(candidates))
    remaining = search_deadline - time.monotonic()
    if remaining < 60:
        print("Not enough time for Phase 2")
    else:
        time_per_top = remaining / max(1, top_k)
        for i in range(top_k):
            if time.monotonic() > search_deadline:
                break
            sc_i, a_i, n_i = candidates[i]
            rng_i = np.random.default_rng((seed * 104729 + i * 7919) % (2**32))
            t_alloc = min(time_per_top, search_deadline - time.monotonic())
            if t_alloc < 30:
                break
            print(f"  Optimizing candidate #{i}: n={n_i}, score={sc_i:.6f}, alloc={t_alloc:.0f}s")

            a_new, sc_new = _optimize_single(a_i, t_alloc, rng_i, label=f"top{i}", verbose=True)

            # Try LP step
            if time.monotonic() < search_deadline - 30:
                lp_a, lp_sc = _try_lp_step(a_new, rng_i, max_time=min(45.0, search_deadline - time.monotonic() - 10))
                if lp_a is not None and np.isfinite(lp_sc) and lp_sc < sc_new:
                    print(f"  LP step improved: {sc_new:.6f} -> {lp_sc:.6f}")
                    a_new = lp_a
                    sc_new = lp_sc

            if np.isfinite(sc_new) and sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
            print(f"  -> candidate #{i} final: score={sc_new:.6f}, best={best_score:.6f}")

    # Phase 3: Final refinement
    print("\n--- Phase 3: Final refinement ---")
    if best_a is not None:
        remaining = final_deadline - time.monotonic()
        if remaining > 60:
            refine_time = min(250.0, remaining - 30.0)
            if refine_time > 30:
                rng_r = np.random.default_rng((seed + 42) % (2**32))
                print(f"  Refining best (score={best_score:.6f}, n={best_a.size}) for {refine_time:.0f}s")
                a_ref, sc_ref = _optimize_refine(best_a, refine_time, rng_r, label="final", verbose=True)
                if np.isfinite(sc_ref) and sc_ref < best_score:
                    best_score = sc_ref
                    best_a = a_ref.copy()
                print(f"  After refinement: {best_score:.6f}")

    # Fallback
    if best_a is None:
        print("Fallback to uniform")
        best_a = np.ones(4096, dtype=np.float64) / 4096.0
        best_score, _, _ = _score_internal(best_a)

    # Final cleanup and validation
    best_a = np.nan_to_num(best_a, nan=0.0, posinf=1000.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    ss = float(best_a.sum())
    if ss < 0.01:
        best_a = np.ones(4096, dtype=np.float64) / 4096.0
        ss = 1.0
    else:
        best_a /= ss

    final_seq = [float(x) for x in best_a]

    # Evaluate with the actual function if possible
    n_final = best_a.size
    if time.monotonic() < final_deadline - 5.0:
        exact = _safe_evaluate(final_seq)
        if np.isfinite(exact):
            print(f"\n=== FINAL: n={n_final}, internal={best_score:.6f}, exact={exact:.6f} ===")
        else:
            print(f"\n=== FINAL: n={n_final}, internal={best_score:.6f}, exact failed ===")
    else:
        print(f"\n=== FINAL: n={n_final}, internal={best_score:.6f} ===")

    total_time = time.monotonic() - now0
    print(f"Total time used: {total_time:.1f}s")

    return final_seq
