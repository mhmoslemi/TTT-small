def _pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _clean(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    return np.maximum(v, 0.0)


def _normalize(v):
    v = _clean(v)
    s = float(v.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.full(v.size, 1.0 / max(1, v.size), dtype=np.float64)
    return v / s


def _project_simplex(v):
    v = _clean(v)
    n = v.size
    if n == 0:
        return v.copy()
    x = np.sort(v)[::-1]
    css = np.cumsum(x)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = x > (css - 1.0) / idx
    if not np.any(cond):
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    r = int(np.nonzero(cond)[0][-1])
    tau = (css[r] - 1.0) / float(r + 1)
    out = v - tau
    np.maximum(out, 0.0, out=out)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    out *= 1.0 / s
    return out

def _lse_grad(x, eps):
    a = _from_ratios(x)
    n = a.size
    if n <= 1:
        return 0.0, np.zeros_like(x)
    L = _pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    cmax = float(c.max())
    if cmax <= 1e-300:
        return 0.0, np.zeros_like(x)
    z = (c - cmax) / eps
    z = np.clip(z, -745.0, 0.0)
    e = np.exp(z)
    sume = float(e.sum())
    if not np.isfinite(sume) or sume <= 0:
        return 1e300, np.zeros_like(x)
    f = cmax + eps * float(np.log(sume))
    w = e / sume
    W = np.fft.rfft(w, n=L)
    ga = 2.0 * np.fft.irfft(W * np.conj(A), n=L)[:n]
    dot = float(np.dot(a, ga))
    gfull = a * (ga - dot)
    gy = gfull[:-1]
    if not np.all(np.isfinite(gy)):
        gy = np.zeros_like(x)
    return f, gy

def _local_search(a0, end_time, seed):

def _anneal_search(a0, end_time, seed):
    """Simulated annealing with block perturbations."""
    a = _project_simplex(a0.copy())
    n = a.size
    L = _pow2(2 * n - 1)
    best = a.copy()
    best_score = _score(best, L)
    cur_score = best_score
    rng = np.random.default_rng(seed)

    T0 = 0.05 * best_score / n
    Tend = T0 * 1e-5
    start = time.monotonic()
    total = max(1.0, end_time - start)
    last_best = time.monotonic()

    while time.monotonic() < end_time:
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / total)
        T = T0 * (Tend / T0) ** frac

        cand = a.copy()
        nblocks = int(rng.integers(1, 3))
        sigma = 0.05 + 0.1 * (1 - frac)
        for _ in range(nblocks):
            w = int(np.exp(rng.uniform(np.log(2.0), np.log(max(3, n // 5)))))
            w = max(1, min(w, n))
            s = int(rng.integers(0, n - w + 1))
            cand[s:s + w] *= float(np.exp(rng.normal(0.0, sigma)))
        cand = _project_simplex(cand)
        sc = _score(cand, L)

        if not np.isfinite(sc):
            continue

        delta = sc - cur_score
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / max(T, 1e-15))):
            a = cand
            cur_score = sc
            if sc < best_score:
                best = cand
                best_score = sc
                last_best = time.monotonic()
                T = max(Tend, T * 0.95)

        if time.monotonic() - last_best > 30.0 and time.monotonic() - start > 60.0:
            a = _project_simplex(best * np.exp(rng.normal(0, 0.03, n)))
            cur_score = _score(a, L)

    return best, best_score

    for i, c in enumerate(cands):
        sc = _score(c)
        if np.isfinite(sc):
            scored.append((sc, c))
    scored.sort(key=lambda x: x[0])
    print(f"Phase 1: {len(scored)} candidates, best score = {scored[0][0]:.6f}")

    global_best = scored[0][1].copy()
    global_best_score = scored[0][0]

    # Phase 2: Multiplicative weights optimization (30% time)
    mw_end = t0 + 0.30 * budget
    if time.monotonic() < mw_end:
        print(f"Phase 2: Multiplicative weights, target end at {mw_end - t0:.0f}s")
        top_k = min(3, len(scored))
        for idx in range(top_k):
            phase_end = mw_end - (top_k - 1 - idx) * (0.30 * budget / (2 * top_k))
            if time.monotonic() >= phase_end:
                break
            a = np.asarray(scored[idx][1], dtype=np.float64).copy()
            a, sc = _multiplicative_weights_optimize(a, phase_end, int(seed) + idx)
            if np.isfinite(sc) and sc < global_best_score:
                global_best_score = sc
                global_best = a.copy()
                print(f"  MW improvement: score={sc:.6f}")

    # Phase 3: L-BFGS on log-sum-exp (40% time)
    lbfgs_end = t0 + 0.70 * budget
    if time.monotonic() < lbfgs_end:
        print(f"Phase 3: L-BFGS refinement, target end at {lbfgs_end - t0:.0f}s")
        a, sc = _lbfgs_refine(global_best, lbfgs_end, int(seed) + 100)
        if np.isfinite(sc) and sc < global_best_score:
            global_best_score = sc
            global_best = a.copy()
            print(f"  LSE improvement: score={sc:.6f}")

    # Phase 4: Annealing (15% time)
    anneal_end = t0 + 0.85 * budget
    if time.monotonic() < anneal_end:
        print(f"Phase 4: Annealing, target end at {anneal_end - t0:.0f}s")
        a, sc = _anneal_search(global_best, anneal_end, int(seed) + 200)
        if np.isfinite(sc) and sc < global_best_score:
            global_best_score = sc
            global_best = a.copy()
            print(f"  Anneal improvement: score={sc:.6f}")

    # Phase 5: Local search (remaining time)
    if time.monotonic() < deadline:
        print(f"Phase 5: Local search, {deadline - time.monotonic():.0f}s remaining")
        a, sc = _local_search(global_best, deadline - 2.0, int(seed) + 300)
        if np.isfinite(sc) and sc < global_best_score:
            global_best_score = sc
            global_best = a.copy()
            print(f"  Local search improvement: score={sc:.6f}")

    global_best = _project_simplex(global_best)
    out = [float(x) for x in global_best]

    # Final validation
    try:
        val = evaluate_sequence(out)
        if not np.isfinite(val):
            print(f"WARNING: evaluate_sequence gave {val}, using fallback")
            out = [float(x) for x in _power_seq(n, 0.5, 1.0)]
        else:
            print(f"Final validated score: {val:.6f}")
    except Exception as e:
        print(f"Validation error: {e}")

    return out
