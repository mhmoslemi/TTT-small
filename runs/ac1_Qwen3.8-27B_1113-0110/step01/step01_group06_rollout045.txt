def _next_pow2(x):

"""
Optimized step-function convolution minimizer combining:
1. Gerchberg-Saxton flat-spectrum initialization
2. Hybrid exponentiated gradient + peak-aware coordinate descent
3. Multi-scale focus schedules and diversified restarts
4. Simulated-annealing style local moves for fine-tuning
"""

import numpy as np
import time
import math


def _next_pow2(x):
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _fft_conv(a):
    n = int(a.size)
    L = 2 * n - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    b = np.fft.irfft(fa * fa, n=N).real[:L]
    np.maximum(b, 0, out=b)
    return b


def _score(a, b=None):
    n = int(a.size)
    if b is None:
        b = _fft_conv(a)
    s = float(a.sum())
    if s <= 1e-300:
        return float('inf'), b, 0.0
    maxb = float(b.max())
    return 2.0 * n * maxb / (s * s), b, maxb


def _fft_grad_conv(p, a):
    """Gradient of sum_k p[k]*(a*a)[k] w.r.t. a"""
    n = int(a.size)
    L = int(p.size)
    N = _next_pow2(L + n - 1)
    fp = np.fft.rfft(p, n=N)
    fa = np.fft.rfft(a, n=N)
    return 2.0 * np.fft.irfft(np.conj(fp) * fa, n=N).real[:n]


def _gs_init(n, rng, n_iter=150):
    """Gerchberg-Saxton for flat-spectrum initialization"""
    a = rng.random(n) + 0.01
    a /= a.sum()
    N = _next_pow2(2 * n - 1)
    for it in range(n_iter):
        fa = np.fft.rfft(a, n=N)
        mag = np.abs(fa)
        phase = np.angle(fa)
        # Target: reduce non-DC magnitude toward uniform
        target = np.ones_like(mag)
        target[0] = 1.0
        non_dc = mag[1:]
        if non_dc.size > 0 and non_dc.max() > 1e-300:
            target[1:] = np.median(non_dc)
        # Blend to avoid collapse
        blend = 0.3 + 0.7 * (it / max(1, n_iter))
        new_mag = (1 - blend) * mag + blend * target
        fa_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(fa_new, n=N).real[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-300:
            a = rng.random(n) + 0.01
            a /= a.sum()
            continue
        a = a_new / s
    return a


def _optimize_hybrid(a0, time_limit, rng, label="", focus_start=5.0):
    """
    Hybrid optimizer: alternates between FFT-based exponentiated gradient
    and local peak-aware coordinate moves.
    """
    a = np.asarray(a0, dtype=np.float64).copy()
    n = int(a.size)
    a = np.clip(a, 0, 1000)
    s = float(a.sum())
    if s < 1e-300:
        a = np.ones(n) / n
    else:
        a /= s
    a = np.maximum(a, 1e-15)
    a /= a.sum()

    b = _fft_conv(a)
    score, b, _ = _score(a, b)
    best_score = score
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    max_no = 400
    resets = 0
    max_resets = 40

    # Focus cycles through values
    focus_cycle = [5, 10, 20, 50, 100, 200, 500, 1000, 200, 50, 20, 10, 5]
    focus_idx = 0
    focus = float(focus_start)

    eta = 0.4
    eta_min = 0.01
    eta_max = 4.0

    while time.monotonic() - t0 < time_limit:
        # Update focus periodically
        if it % 80 == 0:
            focus_idx = (focus_idx + 1) % len(focus_cycle)
            focus = float(focus_cycle[focus_idx])

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no:
            resets += 1
            if resets > max_resets:
                break
            no_improve = 0
            max_no = int(max_no * 1.25)

            # Diversified restart
            rtype = resets % 5
            if rtype == 0:
                gs_a = _gs_init(n, rng, n_iter=60)
                a = 0.65 * best_a + 0.35 * gs_a
            elif rtype == 1:
                a = best_a * np.exp(rng.normal(0, 0.4, n))
            elif rtype == 2:
                # Anti-palindromic perturbation
                m = n // 2
                v = rng.normal(0, 0.3, m)
                a2 = best_a.copy()
                if n % 2 == 0:
                    a2[:m] += v
                    a2[m:] -= v[::-1]
                else:
                    a2[:m] += v
                    a2[m+1:] -= v[::-1]
                a = a2
            elif rtype == 3:
                # Blend with uniform
                a = 0.85 * best_a + 0.15 * np.ones(n) / n
            else:
                # Random restart with GS
                a = _gs_init(n, rng, n_iter=80)

            a = np.maximum(a, 1e-15)
            a /= a.sum()
            b = _fft_conv(a)
            score, b, _ = _score(a, b)
            eta = 0.4
            continue

        # --- Step 1: FFT-based exponentiated gradient ---
        # Make weights focusing on peaks
        maxb = float(b.max())
        if maxb > 1e-300:
            z = (maxb - b) / maxb
            with np.errstate(over='ignore', under='ignore'):
                p_exp = np.exp(-z * focus)
            s_p = float(p_exp.sum())
            if s_p > 1e-12:
                p = p_exp / s_p
            else:
                p = np.zeros(len(b))
                p[int(np.argmax(b))] = 1.0
        else:
            p = np.full(len(b), 1.0 / len(b))

        # Add small uniform component to avoid zeroing out
        p = 0.95 * p + 0.05 / len(p)

        grad = _fft_grad_conv(p, a)
        maxg = float(np.max(np.abs(grad)))
        if maxg > 1e-18:
            grad /= maxg
            a_new = a * np.exp(-eta * grad)
            a_new = np.maximum(a_new, 1e-300)
            s_new = float(a_new.sum())
            if s_new > 1e-300:
                a_new /= s_new
                b_new = _fft_conv(a_new)
                score_new, b_new, _ = _score(a_new, b_new)

                wt_old = float(p @ b)
                wt_new = float(p @ b_new)

                accept = (score_new < best_score - 1e-12 or
                          score_new < score - 1e-12 or
                          (score_new < score * 1.003 and wt_new < wt_old * 0.998))
                if accept:
                    a = a_new
                    b = b_new
                    score = score_new
                    eta = min(eta * 1.04, eta_max)
                else:
                    eta = max(eta * 0.82, eta_min)
                    if eta < 0.015:
                        # Small perturbation to escape
                        a = np.maximum(a * np.exp(rng.normal(0, 0.08, n)), 1e-300)
                        a /= a.sum()
                        b = _fft_conv(a)
                        score, b, _ = _score(a, b)
                        eta = 0.3
            else:
                break
        else:
            break

        # --- Step 2 (periodic): Local peak-aware coordinate move ---
        if it % 10 == 5:
            k_peak = int(np.argmax(b))
            # Find positions contributing most to the peak
            j_min = max(0, k_peak - n + 1)
            j_max = min(n - 1, k_peak)
            if j_max >= j_min:
                # blame[i] = a[i] * a[k_peak - i] for i in [j_min, j_max]
                idx = np.arange(j_min, j_max + 1)
                partners = k_peak - idx
                blame = a[idx] * a[partners]
                
                # Find the worst pair (largest blame)
                worst_local = int(np.argmax(blame))
                i_bad = int(idx[worst_local])
                j_partner = int(partners[worst_local])
                
                # Find a good place to move weight: minimize max contribution
                # Try a few random candidates
                best_move_score = score
                best_target = -1
                for _ in range(20):
                    j_cand = int(rng.integers(n))
                    if j_cand == i_bad:
                        continue
                    # Estimate impact: moving delta from i_bad to j_cand
                    # Changes (a*a)[k] by delta * (2*a[k-j_cand] - 2*a[k-i_bad])
                    # The peak increase is at most delta * 2 * max_k a[k-j_cand]
                    # and the peak decrease is at least delta * 2 * a[k_peak - i_bad] (at the peak)
                    delta = a[i_bad] * 0.05
                    # Quick estimate
                    a_test = a.copy()
                    a_test[i_bad] -= delta
                    a_test[j_cand] += delta
                    a_test = np.maximum(a_test, 0)
                    a_test /= a_test.sum()
                    # Approximate new peak using the change
                    # (a*a)[k] changes by delta*(a[k-j_cand] + a[k_j_cand] - a[k-i_bad] - a[k-i_bad])
                    # Just do a quick check at the peak and nearby
                    change_at_peak = delta * 2 * (a[j_partner] - a[k_peak - j_cand] if 0 <= k_peak - j_cand < n else 0)
                    if change_at_peak < 0:  # peak should decrease
                        best_move_score = change_at_peak
                        best_target = j_cand
                        break

                if best_target >= 0:
                    delta = a[i_bad] * rng.uniform(0.02, 0.08)
                    a[i_bad] -= delta
                    a[best_target] += delta
                    a = np.maximum(a, 0)
                    a /= a.sum()
                    b = _fft_conv(a)
                    score, b, _ = _score(a, b)
                    if score < best_score - 1e-10:
                        best_score = score
                        best_a = a.copy()
                        no_improve = 0

        it += 1

        if it % 2000 == 0:
            print(f"  [{label}] it={it} t={time.monotonic()-t0:.1f}s score={score:.6f} best={best_score:.6f} focus={focus:.0f}")

    return best_a, best_score

            no_improve = 0
            max_no = int(max_no * 1.3)
            a = 0.7 * best_a + 0.3 * _gs_init(n, rng, n_iter=50)
            a = np.maximum(a, 1e-15)
            a /= a.sum()
            b = _fft_conv(a)
            score, b, _ = _score(a, b)
            beta = float(rng.choice([10, 20, 50, 100, 200, 500]))
            eta = 0.3
            continue

        # Soft-max weights
        maxb = float(b.max())
        if maxb > 1e-300:
            z = (b - maxb) / maxb  # negative or zero, max is 0
            with np.errstate(over='ignore', under='ignore'):
                p = np.exp(z * beta)
            s_p = float(p.sum())
            if s_p > 1e-12:
                p = p / s_p
            else:
                p = np.zeros(len(b))
                p[int(np.argmax(b))] = 1.0
        else:
            p = np.full(len(b), 1.0 / len(b))

        grad = _fft_grad_conv(p, a)
        maxg = float(np.max(np.abs(grad)))
        if maxg < 1e-18:
            break
        grad /= maxg

        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-300)
        a_new /= a_new.sum()

        b_new = _fft_conv(a_new)
        score_new, b_new, _ = _score(a_new, b_new)

        if score_new < score - 1e-12:
            a = a_new
            b = b_new
            score = score_new
            eta = min(eta * 1.03, 3.0)
        elif score_new < best_score * 1.002:
            if rng.random() < 0.4:
                a = a_new
                b = b_new
                score = score_new
        else:
            eta = max(eta * 0.85, 0.01)
            if eta < 0.02:
                a = np.maximum(a * np.exp(rng.normal(0, 0.1, n)), 1e-300)
                a /= a.sum()
                b = _fft_conv(a)
                score, b, _ = _score(a, b)
                eta = 0.25

        # Slowly increase beta
        beta = min(beta * 1.005, 2000.0)
        it += 1

    return best_a, best_score


def _local_refine(a0, time_limit, rng, label=""):
    """
    Fine-grained local optimization using small random moves.
    Good for squeezing out the last bit of improvement.
    """
    a = np.asarray(a0, dtype=np.float64).copy()
    n = int(a.size)
    a = np.clip(a, 0, 1000)
    a = np.maximum(a, 1e-15)
    a /= a.sum()

    b = _fft_conv(a)
    score, b, _ = _score(a, b)
    best_score = score
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    max_no = 5000
    temp = 0.01  # "temperature" for accepting slightly worse moves

    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no:
            break

        # Make a local move: pick 2-5 random positions and redistribute
        n_move = int(rng.integers(2, 6))
        indices = rng.integers(0, n, size=n_move)
        # Compute current values at these positions
        old_vals = a[indices].copy()
        # Perturb
        new_vals = old_vals * np.exp(rng.normal(0, 0.1, n_move))
        new_vals = np.maximum(new_vals, 0)
        total_old = old_vals.sum()
        total_new = new_vals.sum()
        if total_new < 1e-300:
            continue
        # Scale to preserve total
        new_vals = new_vals * (total_old / total_new)
        a[indices] = new_vals
        a = np.maximum(a, 0)
        a /= a.sum()

        b = _fft_conv(a)
        score_new, b, _ = _score(a, b)

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a.copy()
            no_improve = 0
        elif score_new < score + temp:
            # Accept (simulated annealing)
            pass
        else:
            # Revert
            a[indices] = old_vals
            # Recompute (or keep old b)
            score = score_new  # Actually revert score
            # Need to recompute
            b = _fft_conv(a)
            score, b, _ = _score(a, b)
            no_improve += 1

        # Cool down temperature
        temp = max(temp * 0.9999, 0.001)
        it += 1

    return best_a, best_score


def _safe_evaluate(seq):
    ev = globals().get("evaluate_sequence")
    if ev is None or not callable(ev):
        return float('inf')
    try:
        return float(ev(seq))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Combines multiple optimization strategies:
    - Hybrid FFT exponentiated gradient + coordinate descent
    - Soft-max objective optimization
    - Local simulated annealing for fine-tuning
    - GS-based initialization for diversity
    """
    seed = int(seed)
    total = max(60.0, float(budget_s))
    now0 = time.monotonic()
    final_deadline = now0 + total - 8.0

    print("=== Convolution minimizer v2 ===")
    print(f"Budget: {total:.0f}s, deadline in {total-8:.0f}s")

    best_a = None
    best_score = float('inf')

    # Load given sequence
    given_a = None
    try:
        given = globals().get("height_sequence_1")
        if given is not None:
            g = np.asarray(given, dtype=np.float64).ravel()
            g = np.nan_to_num(g, nan=0.0, posinf=1000.0, neginf=0.0)
            g = np.clip(g, 0, 1000)
            s = float(g.sum())
            if s > 0.01 and g.size > 10:
                gn = g / s
                sc, _, _ = _score(gn)
                if np.isfinite(sc):
                    best_score = min(best_score, sc)
                    best_a = gn.copy()
                    given_a = gn
                    print(f"Given: n={gn.size}, score={sc:.6f}")
    except Exception as e:
        print(f"Given load failed: {e}")

    # Define task list: (type, init_source, n, time_alloc)
    # type: 'hybrid', 'softmax', 'local'
    tasks = []

    if given_a is not None:
        tasks.append(('hybrid', 'given', given_a.size, 350))
        tasks.append(('softmax', 'given', given_a.size, 250))
        tasks.append(('hybrid', 'given_mixed', given_a.size, 250))

    # GS initializations at various sizes
    for n_sz in [4096, 8192]:
        tasks.append(('hybrid', f'gs_{n_sz}', n_sz, 200))
        tasks.append(('softmax', f'gs_{n_sz}', n_sz, 150))

    # Anti-palindromic initializations (different from before - use them as starts for full optimization)
    tasks.append(('hybrid', 'anti_8192', 8192, 150))

    # Random
    tasks.append(('hybrid', 'rand_4096', 4096, 100))

    task_deadline = now0 + total - 150.0  # Reserve 150s for final refinement
    task_idx = 0

    for (ttype, init_src, n_val, alloc) in tasks:
        now = time.monotonic()
        remaining = task_deadline - now
        if remaining < 50:
            break
        alloc = min(alloc, remaining)
        if alloc < 30:
            break

        task_rng = np.random.default_rng((seed * 1000003 + task_idx * 7919) % (2**32))

        print(f"\n--- Task {task_idx}: {ttype}/{init_src} n={n_val} alloc={alloc:.0f}s best={best_score:.6f} ---")
        t0 = time.monotonic()

        try:
            # Prepare initialization
            if init_src == 'given':
                a0 = given_a.copy()
            elif init_src == 'given_mixed':
                # Mix given with GS
                gs = _gs_init(given_a.size, task_rng, n_iter=80)
                a0 = 0.7 * given_a + 0.3 * gs
                a0 = np.maximum(a0, 1e-15)
                a0 /= a0.sum()
            elif init_src.startswith('gs_'):
                n_sz = int(init_src.split('_')[1])
                a0 = _gs_init(n_sz, task_rng, n_iter=100)
            elif init_src.startswith('anti_'):
                n_sz = int(init_src.split('_')[1])
                # Anti-palindromic random init
                m = n_sz // 2
                v = rng_normal = task_rng.normal(0, 1, m)
                v = v - v.mean()
                std = v.std()
                if std > 1e-12:
                    v = 3.0 * v / std
                z = np.zeros(n_sz)
                z[:m] = v
                if n_sz % 2 == 0:
                    z[m:] = -v[::-1]
                else:
                    z[m+1:] = -v[::-1]
                a0 = np.exp(z - z.max())
                a0 = np.maximum(a0, 1e-15)
                a0 /= a0.sum()
            elif init_src.startswith('rand_'):
                n_sz = int(init_src.split('_')[1])
                a0 = task_rng.random(n_sz) + 0.01
                a0 /= a0.sum()
            else:
                a0 = np.ones(n_val) / n_val

            # Run optimizer
            if ttype == 'hybrid':
                cand_a, cand_score = _optimize_hybrid(a0, alloc, task_rng, label=f"T{task_idx}")
            elif ttype == 'softmax':
                cand_a, cand_score = _optimize_soft_max(a0, alloc, task_rng, label=f"T{task_idx}")
            elif ttype == 'local':
                cand_a, cand_score = _local_refine(a0, alloc, task_rng, label=f"T{task_idx}")
            else:
                cand_a, cand_score = _optimize_hybrid(a0, alloc, task_rng, label=f"T{task_idx}")

            # Verify
            if cand_a is not None and np.isfinite(cand_score):
                cand_a = np.nan_to_num(cand_a, nan=0.0, posinf=0.0, neginf=0.0)
                cand_a = np.clip(cand_a, 0, 1000)
                cs = float(cand_a.sum())
                if cs > 0.01:
                    cand_a = np.maximum(cand_a, 0) / cs
                    # Exact verification for small enough sequences
                    if cand_a.size <= 10000:
                        exact = _safe_evaluate([float(x) for x in cand_a])
                        if np.isfinite(exact):
                            cand_score = min(cand_score, exact)

                    if cand_score < best_score:
                        best_score = cand_score
                        best_a = cand_a.copy()
                        print(f"  *** NEW BEST: {cand_score:.6f} (n={cand_a.size}) ***")
                    else:
                        print(f"  Score: {cand_score:.6f}")
        except Exception as e:
            print(f"  Task failed: {repr(e)}")

        elapsed = time.monotonic() - t0
        print(f"  Done in {elapsed:.1f}s")
        task_idx += 1

    # === Final refinement phase ===
    if best_a is not None:
        remaining = final_deadline - time.monotonic()
        print(f"\n=== Final refinement: {remaining:.0f}s remaining ===")

        if remaining > 80:
            # Phase 1: Hybrid on best
            t1 = min(remaining * 0.35, 200)
            rng1 = np.random.default_rng((seed + 777) % (2**32))
            print(f"  Refine hybrid: {t1:.0f}s")
            a_r, s_r = _optimize_hybrid(best_a, t1, rng1, label="R1")
            if np.isfinite(s_r) and s_r < best_score:
                best_score = s_r
                best_a = a_r
                print(f"  -> {s_r:.6f}")

        if best_a is not None:
            remaining = final_deadline - time.monotonic()
            if remaining > 60:
                # Phase 2: Soft-max
                t2 = min(remaining * 0.35, 150)
                rng2 = np.random.default_rng((seed + 888) % (2**32))
                print(f"  Refine softmax: {t2:.0f}s")
                a_r, s_r = _optimize_soft_max(best_a, t2, rng2, label="R2")
                if np.isfinite(s_r) and s_r < best_score:
                    best_score = s_r
                    best_a = a_r
                    print(f"  -> {s_r:.6f}")

        if best_a is not None:
            remaining = final_deadline - time.monotonic()
            if remaining > 40:
                # Phase 3: Local fine-tuning
                t3 = min(remaining * 0.7, 100)
                rng3 = np.random.default_rng((seed + 999) % (2**32))
                print(f"  Refine local: {t3:.0f}s")
                a_r, s_r = _local_refine(best_a, t3, rng3, label="R3")
                if np.isfinite(s_r) and s_r < best_score:
                    best_score = s_r
                    best_a = a_r
                    print(f"  -> {s_r:.6f}")

    if best_a is None:
        best_a = np.ones(1024, dtype=np.float64) / 1024.0

    # Final cleanup
    best_a = np.nan_to_num(best_a, nan=0.0, posinf=0.0, neginf=0.0)
    best_a = np.clip(best_a, 0, 1000)
    best_a = np.maximum(best_a, 0)
    ss = float(best_a.sum())
    if ss <= 0.01:
        best_a = np.ones(1024, dtype=np.float64) / 1024.0
        ss = 1.0
    else:
        best_a /= ss

    final_seq = [float(x) for x in best_a]

    # Final verification
    if best_a.size <= 12000 and time.monotonic() < final_deadline - 1:
        exact = _safe_evaluate(final_seq)
        print(f"\n=== FINAL: internal={best_score:.6f}, exact={exact:.6f}, n={best_a.size} ===")
    else:
        print(f"\n=== FINAL: internal={best_score:.6f}, n={best_a.size} ===")

    return final_seq
