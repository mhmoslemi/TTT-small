            best_a = a.copy()

        idx = np.argpartition(c, -top_k)[-top_k:]
        # Weight: equal on top-k, or inverse-proportional to gap

        weights = 1.0 / top_k  # equal weight on top-k

    n = a.size

    eta = 0.05
    rng = np.random.default_rng(seed)
    stagnant = 0
    
    while time.monotonic() < deadline:
        c = convolve_seq(a, N)
        cmax = float(np.max(c))
        k_max = int(np.argmax(c))
        score = 2.0 * n * cmax
        
        if score < best_score * (1 - 1e-12):
            best_score = score
            best_a = a.copy()
        
        # Gradient of conv at position k_max
        # grad_j = p_{k_max - j} + p_{j - k_max} (with bounds)
        g = np.zeros(n)
        for j in range(n):
            val = 0.0
            i1 = k_max - j  # p[i1] contributes when i=j
            i2 = j - k_max  # p[i2] contributes when k-j=j, i.e., k=2j... no
            # Actually: d(conv_k)/dp_j = p_{k-j} + p_{j-k} (if valid)

            if 0 <= j - k_max < n:
                val += a[j - k_max]
            g[j] = val

        g = g / (np.abs(g).max() + 1e-15)

            score_new = 2.0 * n * np.max(c_new)
            
            if score_new < score:
                accepted = True
                a = a_new
                break

            stagnant = 0
            eta = min(0.2, eta * 1.02)

Frank-Wolfe for this problem:
1. Compute a subgradient g of F at the current point p.
   (A subgradient of max_k f_k is any convex combination of the gradients of the active f_k's.)
2. Solve the linear subproblem: min_{q in simplex} g^T q.

   (Since the subproblem is linear over the simplex, the optimum is at a vertex.)
3. Update: p <- (1-gamma) * p + gamma * q, where gamma is a step size.
