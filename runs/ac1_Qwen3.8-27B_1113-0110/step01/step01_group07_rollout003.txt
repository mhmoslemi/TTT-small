"""
L-BFGS + Nesterov hybrid optimizer for minimizing peak of linear self-convolution
of non-negative sequences. Uses smooth-max approximation with adaptive tau,
diverse initializations, and multi-phase refinement.
"""

import time
import numpy as np
from scipy.optimize import minimize

# Global helpers

def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def normalize_positive(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s


def softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = np.where(np.isfinite(z), z, -50.0)
    z = z - z.max()
    z = np.clip(z, -50.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        e = np.ones_like(z)
        s = float(e.size)
    return e / s


def fast_score(a):
    p = normalize_positive(a)
    if p is None:
        return np.inf
    n = p.size
    if n <= 0:
        return np.inf
    L = next_pow2(2 * n - 1)
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0.0:
        return np.inf
    return float(2.0 * n * cm)


def smooth_obj_grad(z, tau, n, L, W_cache=None):
    """Compute smooth objective and gradient w.r.t. z in log-domain."""
    p = softmax_fn(z)
    P = np.fft.rfft(p, L)
    c = np.fft.irfft(P * P, L)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0.0:
        return np.inf, np.zeros(n)
    zc = (c - cm) / tau
    zc = np.clip(zc, -50.0, 50.0)
    expz = np.exp(zc)
    expsum = float(expz.sum())
    if not np.isfinite(expsum) or expsum <= 0.0:
        expz = np.zeros_like(c)
        expz[int(np.argmax(c))] = 1.0
        expsum = 1.0
    W = expz / expsum
    obj_val = 2.0 * n * (cm + tau * float(np.log(expsum)))
    # Gradient: dS/dz_j = p_j * (g_j - sum_i g_i * p_i)
    Wref = W[::-1]
    H = np.fft.irfft(P * np.fft.rfft(Wref, L), L)
    g_p = 4.0 * n * H[n - 1:2 * n - 1][::-1]
    g_p = np.where(np.isfinite(g_p), g_p, 0.0)
    mean_g = float(np.dot(p, g_p))
    grad_z = p * (g_p - p * mean_g)
    grad_z = np.where(np.isfinite(grad_z), grad_z, 0.0)
    return float(obj_val), grad_z


def smooth_obj(z, tau, n, L):
    val, _ = smooth_obj_grad(z, tau, n, L)
    return val


def lbfms_phase(z0, tau, n, L, maxiter=300, tol=1e-12):
    """Run one L-BFGS phase at given tau."""
    def fobj(z_in):
        val, gr = smooth_obj_grad(z_in, tau, n, L)
        if not np.isfinite(val):
            return 1e10, np.zeros(n)
        return val, gr

    res = minimize(fobj, z0, method='L-BFGS-B', jac=True,
                   options=dict(maxiter=maxiter, gtol=tol, ftol=1e-14,
                                maxls=40, m=15))
    return res


def resize_sequence(seq, m):
    p = normalize_positive(np.asarray(seq, dtype=np.float64))
    m = int(m)
    if p is None:
        return np.ones(m, dtype=np.float64) / float(m)
    n = p.size
    if n == m:
        return p.copy()
    if n == 1:
        return np.full(m, 1.0 / float(m), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, m)
    out = np.interp(x_new, x_old, p)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(m, dtype=np.float64) / float(m)
    return out / s


def make_initializations(n, rng):
    """Create diverse initial sequences."""
    inits = []
    # Uniform
    inits.append(np.ones(n) / n)
    # Linear ramps
    x = np.linspace(0, 1, n)
    inits.append(normalize_positive(1.0 + 2.0 * x))
    inits.append(normalize_positive(1.0 - 2.0 * x + 0.5))
    inits.append(normalize_positive(1.0 + 3.0 * x - 1.5))
    # Quadratic
    inits.append(normalize_positive(1.0 + 2.0 * (2.0 * x - 1.0) ** 2 - 1.0))
    inits.append(normalize_positive(2.0 - 2.0 * (2.0 * x - 1.0) ** 2))
    # Exponential
    inits.append(normalize_positive(np.exp(-3.0 * x)))
    inits.append(normalize_positive(np.exp(3.0 * (x - 1.0))))
    # Cosine
    inits.append(normalize_positive(0.5 + 0.5 * np.cos(2.0 * np.pi * x)))
    inits.append(normalize_positive(0.5 - 0.5 * np.cos(2.0 * np.pi * x)))
    # Sine bump
    inits.append(normalize_positive(0.3 + 0.7 * np.sin(np.pi * x) ** 2))
    # Asymmetric sine
    inits.append(normalize_positive(0.3 + 0.7 * np.sin(np.pi * x) ** 1.5))
    # Two humps
    inits.append(normalize_positive(np.exp(-20.0 * (x - 0.25) ** 2) + np.exp(-20.0 * (x - 0.75) ** 2)))
    # Random perturbations of uniform
    for _ in range(3):
        noise = np.exp(rng.normal(0.0, 0.3, n))
        inits.append(normalize_positive(noise))
    # Gentle asymmetric
    inits.append(normalize_positive(1.0 + 0.5 * np.sin(3.0 * np.pi * x) + 0.3 * (2.0 * x - 1.0)))
    inits.append(normalize_positive(1.0 + 0.4 * np.cos(4.0 * np.pi * x - 1.0)))
    return [a for a in inits if a is not None]


def lbfms_pipeline(p0, n, L, tau_start, tau_end, phases, maxiter_per_phase=250, deadline=None):
    """Multi-phase L-BFGS with decreasing tau."""
    z = np.log(np.maximum(p0, 1e-300))
    z = np.where(np.isfinite(z), z, -10.0)
    z -= z.mean()
    z = np.clip(z, -30.0, 30.0)

    best_p = p0.copy()
    best_true = fast_score(p0)
    tau_list = np.geomspace(tau_start, tau_end, phases)

    for pi, tau in enumerate(tau_list):
        if deadline and time.monotonic() > deadline:
            break
        try:
            res = lbfms_phase(z, float(tau), n, L, maxiter=maxiter_per_phase, tol=1e-10)
        except Exception:
            continue
        if res.x is None:
            continue
        z_new = np.asarray(res.x, dtype=np.float64)
        z_new = np.where(np.isfinite(z_new), z_new, z)
        z_new -= z_new.mean()
        z_new = np.clip(z_new, -40.0, 40.0)
        p_new = softmax_fn(z_new)
        true_s = fast_score(p_new)
        if np.isfinite(true_s) and true_s < best_true - 1e-10:
            best_true = true_s
            best_p = p_new
            z = z_new
        else:
            z = z_new  # still update z for continuity
        if (pi + 1) % 5 == 0:
            print(f"  lbfms phase {pi+1}/{phases} tau={tau:.2e} true={best_true:.6f}")

    return best_p, best_true


def nesterov_polish(p0, duration_s, rng, n, L, tau_ratio=0.01, lr=0.01):
    """Nesterov-accelerated gradient descent in log-domain with small tau."""
    z = np.log(np.maximum(p0, 1e-300))
    z = np.where(np.isfinite(z), z, -10.0)
    z -= z.mean()
    z = np.clip(z, -40.0, 40.0)

    z_prev = z.copy()
    v = np.zeros(n, dtype=np.float64)
    best_p = p0.copy()
    best_true = fast_score(p0)
    deadline = time.monotonic() + max(0.0, float(duration_s))
    step = 0
    tau = max(1e-15, tau_ratio / n)
    no_improve = 0

    while time.monotonic() < deadline:
        # Nesterov lookahead
        z_look = z - v
        val, grad = smooth_obj_grad(z_look, tau, n, L)
        if not np.isfinite(val):
            break
        grad = np.clip(grad, -1.0, 1.0)
        v = 0.9 * v + lr * grad
        z = z - v
        z = np.where(np.isfinite(z), z, 0.0)
        z -= z.mean()
        z = np.clip(z, -40.0, 40.0)
        p = softmax_fn(z)
        true_s = fast_score(p)
        step += 1

        if np.isfinite(true_s) and true_s < best_true - 1e-10:
            best_true = true_s
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            # Reduce tau and restart momentum
            tau = max(1e-18, tau * 0.7)
            v *= 0.3
            no_improve = 0

        if step % 2000 == 0:
            print(f"  nesterov step {step} tau={tau:.2e} true={best_true:.6f}")

    return best_p, best_true


def cheb_init(n, D, theta, mode=0, eps=0.1):
    x = np.linspace(0.0, 1.0, n)
    t = 2.0 * x - 1.0
    t = np.clip(t, -1.0, 1.0)
    ang = np.arccos(t)
    val = np.zeros(n)
    for m in range(1, D + 1):
        val += theta[m - 1] * np.cos(m * ang)
    if mode == 0:
        h = 1.0 + eps + val
    else:
        h = np.exp(np.clip(val, -10.0, 10.0))
    return normalize_positive(h)

                best_s, best_p = s_res, p_res
                print(f"  Nesterov candidate {i}: {s_res:.6f}")
            rem = deadline - time.monotonic()
            if rem <= 30:
                break

    # Phase D: Another round of different inits if time remains
    rem = deadline - time.monotonic()
    if rem > 120:
        print(f"Phase D: Additional exploration, budget {rem:.0f}s")
        for trial in range(6):
            if rem < 60:
                break
            # New random init
            noise = np.exp(rng.normal(0.0, 0.5, N))
            p_new = normalize_positive(noise)
            if p_new is None:
                continue
            # Quick L-BFGS
            dur = min(rem * 0.2, 30.0)
            s_init = fast_score(p_new)
            p_res, s_res = lbfms_pipeline(p_new, N, L,
                                          tau_start=0.03 / N, tau_end=0.005 / N,
                                          phases=8, maxiter_per_phase=200,
                                          deadline=deadline)
            if s_res < best_s:
                best_s, best_p = s_res, p_res
                print(f"  Phase D trial {trial}: {s_res:.6f}")
            rem = deadline - time.monotonic()

    # Phase E: Final Nesterov polish with adaptive tau
    rem = deadline - time.monotonic()
    if rem > 30:
        dur = max(20.0, rem * 0.8)
        p_res, s_res = nesterov_polish(best_p, dur, rng, N, L,
                                       tau_ratio=0.003, lr=0.005)
        if s_res < best_s:
            best_s, best_p = s_res, p_res
            print(f"  Final polish: {s_res:.6f}")

    # Final validation with actual evaluator
    candidates_final = [best_p]
    candidates_final.append(np.ones(1024) / 1024.0)

    final_seq = None
    final_score = np.inf
    for cand in candidates_final:
        if cand is None:
            continue
        lst = [float(x) for x in np.clip(np.asarray(cand, dtype=np.float64), 0.0, 1000.0)]
        if not lst:
            continue
        try:
            sc = evaluate_sequence(lst)
        except Exception:
            sc = np.inf
        if np.isfinite(sc) and sc < final_score:
            final_score = float(sc)
            final_seq = lst

    if final_seq is None:
        final_seq = [1.0 / 1024.0] * 1024

    print(f"FINAL SCORE: {final_score:.6f}, length={len(final_seq)}")
    return final_seq
