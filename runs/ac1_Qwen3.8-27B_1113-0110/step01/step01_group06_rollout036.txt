Optimize a non-negative step function sequence to minimize
    2 * n * max(convolve(a, a)) / (sum(a))**2.

Uses LP-based direction finding (solve max sum(g) s.t. convolve(a,g) <= M, g >= 0)
combined with gradient-based refinement and multi-start diversity.

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import linprog

def _pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()

def _fft_square(a):
    n = int(a.size)
    L = 2 * n - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, n=N)
    b = np.fft.irfft(fa * fa, n=N).real[:L]
    b = np.maximum(b, 0.0)
    return b

def _fft_convolve(a, b):
    """Compute convolve(a, b) using FFT."""
    n = len(a)
    m = len(b)
    L = n + m - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N).real[:L]
    return c

def _internal_score(a, b=None):
    if b is None:
        b = _fft_square(a)
    s = float(a.sum())
    if s <= 1e-300:
        return float("inf"), b, 0.0
    maxb = float(b.max())
    score = float(2.0 * a.size * maxb / (s * s))
    return score, b, maxb

    ev = globals().get("evaluate_sequence")
    if ev is None or not callable(ev):
        return float("inf")
    try:

    except Exception:

def _solve_lp_direction(a, M, max_time=300):
    """
    Solve: maximize sum(g) s.t. g >= 0, convolve(a, g)_k <= M for all k.
    Returns g or None if infeasible/too slow.
    """
    n = len(a)
    # Use linprog from scipy (more reliable than cvxpy for LPs of this size)
    # Minimize -sum(g) s.t. A@g <= b, g >= 0
    # A is (2n-1) x n matrix where A[k, :] gives the coefficients of convolve(a, :)_k
    # A[k, j] = a[j] if 0 <= k-j <= n-1, i.e., k-n+1 <= j <= k

    # = sum_j A[k,j] * g[j] where A[k,j] = a[j] if max(0, k-n+1) <= j <= min(k, n-1) and k-j in [0, n-1]
    # i.e., j in [max(0, k-n+1), min(k, n-1)] and k-j in [0, n-1]
    # which means j in [max(0, k-n+1), min(k, n-1)]
    
    # This is a banded matrix! Each row has at most n nonzeros.
    # For n=8192, the matrix is 16383 x 8192, which is ~133 million entries.
    # That's too large for dense, but scipy linprog can handle sparse.
    
    from scipy.sparse import csr_matrix, diags
    
    # Build the sparse constraint matrix
    # For row k (0-indexed), the nonzeros are at columns j where max(0, k-n+1) <= j <= min(k, n-1)
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend(np.full(len(js), k))
            cols.extend(js)
            vals.extend(a[js])
    
    A_sparse = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Minimize -sum(g) = c^T g where c = -1
    c = -np.ones(n)
    b_vec = np.full(2*n-1, M)

"""

    2 * n * max(convolve(a, a)) / (sum(a))**2.

Strategy:
1. LP-based direction finding (minimize cross-correlation peak) for moderate n
2. Smart exponentiated gradient with adaptive top-k weighting for larger n

4. Final refinement with very fine steps

import numpy as np
import time
import cvxpy as cp
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def _pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _fft_square(a):
    n = int(a.size)
    L = 2 * n - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, n=N)
    b = np.fft.irfft(fa * fa, n=N).real[:L]
    return np.maximum(b, 0.0)


def _fft_convolve(a, b):
    n = len(a)
    m = len(b)
    L = n + m - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    return np.fft.irfft(fa * fb, n=N).real[:L]


def _internal_score(a, b=None):
    if b is None:
        b = _fft_square(a)
    s = float(a.sum())
    if s <= 1e-300:
        return float("inf"), b, 0.0
    maxb = float(b.max())
    score = float(2.0 * a.size * maxb / (s * s))
    return score, b, maxb


def _safe_evaluate(seq):
    ev = globals().get("evaluate_sequence")
    if ev is None or not callable(ev):
        return float("inf")
    try:
        return float(ev(seq))
    except Exception:
        return float("inf")


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        a = np.ones(a.size) / a.size
    else:
        a /= s
    return a


def _binary_anti_init(n, rng, shuffle=True):
    """Create a binary anti-palindromic sequence."""
    n = int(n)
    m = n // 2
    a = np.zeros(n, dtype=np.float64)
    left = list(range(m))
    right = list(range(n - 1, n - 1 - m, -1))
    if shuffle:
        rng.shuffle(left)
        rng.shuffle(right)
    # Greedily assign to minimize overlap
    selected = set()
    for i in left:
        j = n - 1 - i
        if i not in selected and j not in selected:
            a[i] = 1.0
            selected.add(i)
            selected.add(j)
    a = np.maximum(a, 1e-15)
    return _normalize(a)


def _anti_log_init(n, rng, amp=3.0, smooth=False):
    n = int(n)
    m = n // 2
    if m < 1:
        return np.ones(n) / max(1, n)
    x = np.arange(m, dtype=np.float64)
    if smooth:
        v = np.zeros(m, dtype=np.float64)
        for f in (1, 3, 5, 7, 11, 13, 17):
            v += (rng.normal(0.0, 1.0) / max(1, f)) * np.sin(
                2.0 * np.pi * f * x / m + rng.uniform(0.0, 2.0 * np.pi)
            )
        v += 0.1 * rng.normal(0.0, 1.0, m)
    else:
        v = rng.normal(0.0, 1.0, m)
    v -= float(v.mean())
    std = float(v.std())
    if std > 1e-12:
        v = (float(amp) * v) / std

    z = np.zeros(n, dtype=np.float64)
    z[:m] = v
    if n % 2 == 0:
        z[m:] = -v[::-1]
    else:
        z[m + 1:] = -v[::-1]
        z[m] = 0.0

    z = z - z.max()
    a = np.exp(z)
    a = np.maximum(a, 1e-15)
    return _normalize(a)


def _random_positive(n, rng, amp=2.0):
    n = int(n)
    z = rng.normal(0.0, amp, size=n)
    z -= z.max()
    a = np.exp(z)
    a = np.maximum(a, 1e-15)
    return _normalize(a)


def _solve_lp_direction(a, M, n, max_time=60):
    """
    Solve: minimize t s.t. conv(a,g)_k <= t for all k, sum(g)=1, g>=0.
    Returns (g, t_min) or (None, None).
    """
    # Variables: g_0...g_{n-1}, t  (total n+1)
    # Minimize: t  => c = [0, 0, ..., 0, 1]
    # Constraints:
    #   conv(a,g)_k - t <= 0 for k=0..2n-2  (inequality)
    #   sum(g) = 1  (equality)
    #   g_i >= 0  (bounds)
    #   t free (but t >= 0 implied by constraints if a > 0)

    # Build sparse constraint matrix for inequalities
    # Row k: [a_{max(0,k-n+1)}, ..., a_{min(k,n-1)}, -1]
    rows_list = []
    cols_list = []
    vals_list = []

    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        js = np.arange(j_start, j_end + 1)
        if len(js) > 0:
            rows_list.append(np.full(len(js), k, dtype=np.int32))
            cols_list.append(js.astype(np.int32))
            vals_list.append(a[js].astype(np.float64))
        # The -1 for t column (column index = n)
        rows_list.append(np.array([k], dtype=np.int32))
        cols_list.append(np.array([n], dtype=np.int32))
        vals_list.append(np.array([-1.0], dtype=np.float64))

    rows_all = np.concatenate(rows_list)
    cols_all = np.concatenate(cols_list)
    vals_all = np.concatenate(vals_list)

    A_ub = csr_matrix((vals_all, (rows_all, cols_all)), shape=(2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(g) = 1, t free
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    # Bounds: g_i >= 0, t free
    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': max_time})
        if res.success and res.x is not None:
            g = res.x[:n]
            t_min = res.x[n]
            g = np.maximum(g, 0.0)
            gs = float(g.sum())
            if gs > 1e-12:
                g /= gs
            return g, float(t_min)
    except Exception as e:
        print(f"  LP error: {e}")

    return None, None


def _solve_lp_max_sum(a, M, n, max_time=60):
    """
    Solve: maximize sum(g) s.t. conv(a,g)_k <= M for all k, g >= 0.
    Returns g (unnormalized) or None.
    """
    rows_list = []
    cols_list = []
    vals_list = []

    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        js = np.arange(j_start, j_end + 1)
        if len(js) > 0:
            rows_list.append(np.full(len(js), k, dtype=np.int32))
            cols_list.append(js.astype(np.int32))
            vals_list.append(a[js].astype(np.float64))

    if not rows_list:
        return None

    rows_all = np.concatenate(rows_list)
    cols_all = np.concatenate(cols_list)
    vals_all = np.concatenate(vals_list)

    A_ub = csr_matrix((vals_all, (rows_all, cols_all)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)

    c_obj = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': max_time})
        if res.success and res.x is not None:
            g = res.x
            gs = float(g.sum())
            if gs > 1e-12:
                return g
    except Exception as e:
        print(f"  LP max-sum error: {e}")

    return None


def _gradient_step_topk(a, b, k_top, eta, rng):
    """Exponentiated gradient step focusing on top-k convolution coefficients."""
    n = len(a)
    # Get top-k indices
    k_top = min(k_top, len(b))
    top_idx = np.argsort(b)[-k_top:]
    top_vals = b[top_idx]
    max_top = float(top_vals.max())

    # Weight: exponential on (max - b) for top-k, small constant for rest
    if max_top > 1e-300:
        z = (max_top - b) / max_top
        weights = np.exp(-z * 10.0)
        # Boost top-k
        weights[top_idx] = 5.0
    else:
        weights = np.ones(len(b)) / len(b)

    w_sum = float(weights.sum())
    if w_sum > 0:
        weights /= w_sum

    # Gradient: 2 * convolve(weights, a)
    grad = 2.0 * _fft_convolve(weights, a)
    # Subtract mean to preserve normalization in log-space
    gd = grad - float(np.dot(a, grad))
    maxg = float(np.max(np.abs(gd)))
    if maxg < 1e-18:
        return a.copy()

    gd = gd / maxg
    a_new = a * np.exp(-eta * gd)
    a_new = np.maximum(a_new, 1e-300)
    return _normalize(a_new)


def _gradient_step_softmax(a, b, focus, eta, rng):
    """Exponentiated gradient with softmax weights on (max_b - b)."""
    n = len(a)
    maxb = float(b.max())
    if maxb <= 1e-300:
        return a.copy()

    z = (maxb - b) / maxb
    with np.errstate(over="ignore", under="ignore"):
        weights = np.exp(-z * float(focus))
    eps = 0.01
    ws = float(weights.sum())
    if ws > 1e-12:
        weights = (1 - eps) * (weights / ws) + eps / len(b)
    else:
        weights = np.ones(len(b)) / len(b)

    grad = 2.0 * _fft_convolve(weights, a)
    gd = grad - float(np.dot(a, grad))
    maxg = float(np.max(np.abs(gd)))
    if maxg < 1e-18:
        return a.copy()

    gd /= maxg
    a_new = a * np.exp(-eta * gd)
    a_new = np.maximum(a_new, 1e-300)
    return _normalize(a_new)

def _upsample(a, target_n):
    """Upsample a sequence to target_n by interpolation."""
    n = len(a)
    if n == target_n:
        return a.copy()
    if n > target_n:
        # Downsample by averaging
        out = np.zeros(target_n)
        for i in range(target_n):
            lo = i * n // target_n
            hi = (i + 1) * n // target_n
            if hi > lo:
                out[i] = a[lo:hi].mean()
            else:
                out[i] = a[i % n]
        return _normalize(out)
    # Upsample by linear interpolation
    x_old = np.linspace(0, 1, n, endpoint=False)
    x_new = np.linspace(0, 1, target_n, endpoint=False)
    out = np.interp(x_new, x_old, a, period=1.0)
    return _normalize(out)

                    best_score = cs
                    best_a = ca.copy()
                    print(f"  Phase 1 trial {trial}: NEW BEST {cs:.6f}")
                else:
                    print(f"  Phase 1 trial {trial}: score={cs:.6f} (best={best_score:.6f})")
                print(f"  Time used: {time.monotonic()-t0:.1f}s")

    # --- Phase 2: Gradient-based for n=8192 ---
    phase2_end = now0 + 1300.0
    if time.monotonic() < phase2_end:
        alloc = min(800.0, phase2_end - time.monotonic())
        if alloc > 60:
            print(f"\n--- Phase 2: Gradient, n=8192, alloc={alloc:.0f}s ---")
            n2 = 8192
            for trial in range(4):
                if time.monotonic() >= phase2_end - 30:
                    break
                alloc_t = min(200.0, phase2_end - time.monotonic() - 10)
                if alloc_t < 40:
                    break
                rng = np.random.default_rng(seed + 5000 + trial * 13)
                if trial == 0 and best_a is not None:
                    a0 = _upsample(best_a, n2)
                elif trial == 1:
                    a0 = _binary_anti_init(n2, rng)
                elif trial == 2:
                    a0 = _anti_log_init(n2, rng, amp=3.0, smooth=False)
                else:
                    a0 = _anti_log_init(n2, rng, amp=5.0, smooth=True)

                t0 = time.monotonic()
                use_topk = (trial % 2 == 1)
                ca, cs = _gradient_optimize(a0, alloc_t, rng,
                                            verbose=(trial == 0),
                                            label=f"grad{n2}-{trial}",
                                            use_topk=use_topk)
                if np.isfinite(cs) and cs < best_score:
                    best_score = cs
                    best_a = ca.copy()
                    print(f"  Phase 2 trial {trial}: NEW BEST {cs:.6f}")
                else:
                    print(f"  Phase 2 trial {trial}: score={cs:.6f} (best={best_score:.6f})")
                print(f"  Time used: {time.monotonic()-t0:.1f}s")

    # --- Phase 3: Multi-scale + diverse starts ---
    phase3_end = now0 + 2000.0
    if time.monotonic() < phase3_end:
        alloc = min(700.0, phase3_end - time.monotonic())
        if alloc > 60:
            print(f"\n--- Phase 3: Multi-scale, alloc={alloc:.0f}s ---")
            n_options = [6144, 12288, 9216]
            for ni, n3 in enumerate(n_options):
                if time.monotonic() >= phase3_end - 30:
                    break
                alloc_t = min(200.0, (phase3_end - time.monotonic()) / max(1, len(n_options) - ni))
                if alloc_t < 40:
                    break
                rng = np.random.default_rng(seed + 9000 + ni * 17)
                if best_a is not None and abs(len(best_a) - n3) / n3 < 0.3:
                    a0 = _upsample(best_a, n3)
                else:
                    a0 = _anti_log_init(n3, rng, amp=3.0 + float(ni), smooth=(ni % 2 == 0))

                t0 = time.monotonic()
                ca, cs = _gradient_optimize(a0, alloc_t, rng,
                                            verbose=False,
                                            label=f"grad{n3}",
                                            use_topk=(ni % 2 == 1))
                if np.isfinite(cs) and cs < best_score:
                    best_score = cs
                    best_a = ca.copy()
                    print(f"  Phase 3 n={n3}: NEW BEST {cs:.6f}")
                else:
                    print(f"  Phase 3 n={n3}: score={cs:.6f} (best={best_score:.6f})")
                print(f"  Time used: {time.monotonic()-t0:.1f}s")

    # --- Phase 4: Final refinement ---
    if best_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            refine_time = min(remaining - 40, 200.0)
            if refine_time > 30:
                print(f"\n--- Phase 4: Final refinement, time={refine_time:.0f}s ---")
                rng = np.random.default_rng(seed + 77777)
                # Very fine gradient on best
                ca, cs = _gradient_optimize(best_a, refine_time * 0.6, rng,
                                            verbose=True, label="final",
                                            focus_start=50.0)
                if np.isfinite(cs) and cs < best_score:
                    best_score = cs
                    best_a = ca.copy()
                    print(f"  Final refine: NEW BEST {cs:.6f}")

    # Fallback
    if best_a is None:
        best_a = np.ones(1024) / 1024.0
        best_score = 2.0
        print("  Fallback to uniform sequence, score=2.0")

    # Final normalization and validation
    best_a = np.nan_to_num(best_a, nan=0.0, posinf=1000.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    ss = float(best_a.sum())
    if ss <= 0.01:
        best_a = np.ones(1024) / 1024.0
        ss = 1.0
    else:
        best_a /= ss

    final_seq = [float(x) for x in best_a]
    final_seq = [max(0.0, min(1000.0, x)) for x in final_seq]

    print(f"\n=== Final: n={len(final_seq)}, internal score={best_score:.6f} ===")

    # Verify with actual evaluator if possible
    if len(final_seq) <= 12000 and time.monotonic() < deadline - 5.0:
        exact = _safe_evaluate(final_seq)
        if np.isfinite(exact):
            print(f"Exact evaluation: {exact:.6f}")
        else:
            print(f"Exact evaluation returned inf (unexpected)")

    return final_seq
