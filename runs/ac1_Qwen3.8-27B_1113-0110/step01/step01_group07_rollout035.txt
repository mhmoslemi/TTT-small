    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    return np.maximum(c, 0.0)


def grad_lp(c, a, p, L2):
    """Gradient of sum c[k]^p w.r.t. a. Returns array of length n."""
    n = len(a)
    c_pow = np.maximum(c, 0.0) ** (p - 1)
    a_flip = a[::-1]
    conv = np.fft.irfft(np.fft.rfft(c_pow, L2) * np.fft.rfft(a_flip, L2), L2)
    grad = 2.0 * p * conv[n - 1:n - 1 + n]
    return grad


def softmax_z(z):
    z = z - z.max()
    z = np.clip(z, -50.0, 50.0)
    e = np.exp(z)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def z_grad(grad_a, a):
    """Convert gradient w.r.t. a to gradient w.r.t. z (log-softmax parameterization)."""
    mean_g = np.dot(a, grad_a)
    return grad_a - a * mean_g


def smooth_max_grad(c, a, tau, L, L2):
    """Gradient of tau*log(sum exp(c/tau)) w.r.t. a via chain rule."""
    n = len(a)
    c_max = c.max()
    z = (c - c_max) / tau
    exp_z = np.exp(np.clip(z, -50, 0))
    exp_sum = exp_z.sum()
    if exp_sum <= 0 or not np.isfinite(exp_sum):
        exp_z = np.zeros_like(c)
        exp_z[int(np.argmax(c))] = 1.0
        exp_sum = 1.0
    w = exp_z / exp_sum
    value = c_max + tau * np.log(exp_sum)
    # grad of smooth_max w.r.t. c is w
    # grad of c w.r.t. a: 2 * a[k-i] (cross-correlation)
    a_flip = a[::-1]
    conv = np.fft.irfft(np.fft.rfft(w, L2) * np.fft.rfft(a_flip, L2), L2)
    grad_a = 2.0 * conv[n - 1:n - 1 + n]
    return value, grad_a


def adam_optimize(z, duration_s, rng, n, L, L2, phase, lr0=0.01, tau=None, p=None,
                  p_schedule=None, tau_schedule=None, log_interval=5000):
    """
    Run Adam optimizer in z-space for the given phase.
    phase: 'L2', 'L4', 'L8', 'SMAX', 'SMAX_DES'
    Returns (best_z, best_score).
    """
    t0 = time.monotonic()
    deadline = t0 + duration_s
    z = z.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0
    best_z = z.copy()
    best_score = np.inf
    no_improve = 0
    lr = lr0

    while time.monotonic() < deadline:
        a = softmax_z(z)
        c = compute_conv(a, L)
        true_score = 2.0 * n * c.max()

        if true_score < best_score - 1e-10:
            best_score = true_score
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Compute gradient based on phase
        if phase == 'L2':
            grad_a = grad_lp(c, a, 2, L2)
        elif phase == 'L4':
            grad_a = grad_lp(c, a, 4, L2)
        elif phase == 'L8':
            grad_a = grad_lp(c, a, 8, L2)
        elif phase == 'SMAX':
            if tau is None:
                tau = 0.01 / max(c.max(), 1e-10)
            _, grad_a = smooth_max_grad(c, a, tau, L, L2)
        elif phase == 'SMAX_DES':
            if tau_schedule is not None:
                progress = step / max(1, int(duration_s * 300))  # approx
                idx = min(int(progress * len(tau_schedule)), len(tau_schedule) - 1)
                tau = tau_schedule[idx]
            else:
                tau = 0.01 / max(c.max(), 1e-10)
            _, grad_a = smooth_max_grad(c, a, tau, L, L2)
        else:
            break

        gz = z_grad(grad_a, a)
        max_gz = np.abs(gz).max()
        if max_gz > 10.0:
            gz *= 10.0 / max_gz

        step += 1
        m = beta1 * m + (1 - beta1) * gz
        v = beta2 * v + (1 - beta2) * (gz * gz)
        b1 = 1 - beta1 ** step
        b2 = 1 - beta2 ** step
        z = z - lr * (m / b1) / (np.sqrt(v / b2) + eps)
        z = np.where(np.isfinite(z), z, 0.0)
        z -= z.mean()
        z = np.clip(z, -50.0, 50.0)

        # Adaptive learning rate
        if step % 2000 == 0:
            lr *= 0.92
        if no_improve > 2000:
            lr *= 0.5
            if no_improve > 8000:
                # Restart with perturbation
                z = z + rng.normal(0, 0.3, n)
                z -= z.mean()
                z = np.clip(z, -50, 50)
                m *= 0.1
                v *= 0.1
                no_improve = 0
                lr = max(lr0 * 0.3, 0.001)

        if step % log_interval == 0:
            print(f"  phase={phase} step={step} score={true_score:.6f} best={best_score:.6f} lr={lr:.5f} noimp={no_improve}")

    return best_z, best_score


def make_initialization(n, init_type, rng):
    """Create various initializations."""
    if init_type == 'uniform':
        return np.zeros(n)
    elif init_type == 'linear':
        x = np.linspace(0, 1, n)
        z = x * 3.0
        return z - z.mean()
    elif init_type == 'quadratic':
        x = np.linspace(0, 1, n)
        z = (x - 0.5) ** 2 * 6.0
        return z - z.mean()
    elif init_type == 'cos':
        x = np.linspace(0, 1, n)
        z = np.cos(np.pi * x) * 2.0
        return z - z.mean()
    elif init_type == 'random_smooth':
        # Random smooth function via Fourier
        k = np.arange(n)
        coeffs = rng.normal(0, 1, n) / (1 + np.abs(k) * 0.1)
        z = np.fft.irfft(coeffs)
        z = z[:n]
        return z - z.mean()
    elif init_type == 'two_level':
        # Alternating pattern with noise
        z = np.zeros(n)
        z[::2] = 1.0
        z[1::2] = -1.0
        z += rng.normal(0, 0.3, n)
        return z - z.mean()
    else:
        return np.zeros(n)


def propose_candidate(seed=42, budget_s=1800.0, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)

    best_seq = None
    best_eval_score = np.inf

    # Try to load previous best
    start_z = None
    start_n = 8192
    try:
        hs = np.asarray(height_sequence_1, dtype=np.float64)
        hs = np.where(np.isfinite(hs), hs, 0.0)
        hs = np.maximum(hs, 0.0)
        if hs.size > 10 and hs.sum() > 1e-9:
            start_n = hs.size
            hs_norm = hs / hs.sum()
            start_z = np.log(np.maximum(hs_norm, 1e-30))
            start_z -= start_z.mean()
            start_z = np.where(np.isfinite(start_z), start_z, 0.0)
            start_z = np.clip(start_z, -50, 50)
            s = fast_score(hs_norm)
            best_seq = hs_norm.copy()
            best_eval_score = s
            print(f"Loaded previous best: n={start_n}, score={s:.6f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(8192) / 8192
        best_eval_score = 2.0
        start_n = 8192

    # Define lengths to try
    lengths_to_try = [8192]
    if best_eval_score > 1.53:
        lengths_to_try = [8192, 16384]
    if time.monotonic() < deadline - 600:
        lengths_to_try = [4096, 8192, 16384]

    global_results = []

    for N in lengths_to_try:
        if time.monotonic() > deadline - 60:
            break

        n = N
        L = next_pow2(2 * n - 1)
        L2 = next_pow2(3 * n - 2)

        print(f"\n=== Starting optimization for n={n} ===")

        # Get initial z for this length
        if start_z is not None and len(start_z) == n:
            z0 = start_z.copy()
        elif start_z is not None:
            # Resize
            a0 = softmax_z(start_z)
            x_old = np.linspace(0, 1, len(a0))
            x_new = np.linspace(0, 1, n)
            a_new = np.interp(x_new, x_old, a0)
            a_new = np.maximum(a_new, 1e-10)
            a_new /= a_new.sum()
            z0 = np.log(a_new)
            z0 -= z0.mean()
            z0 = np.clip(z0, -50, 50)
        else:
            z0 = np.zeros(n)

        a0 = softmax_z(z0)
        s0 = fast_score(a0)
        print(f"  Initial score: {s0:.6f}")

        # Phase 1: L2 optimization
        remaining = deadline - time.monotonic()
        if remaining > 120:
            dur = min(remaining * 0.25, 300.0)
            print(f"  Phase 1 (L2): {dur:.0f}s")
            z1, s1 = adam_optimize(z0, dur, rng, n, L, L2, 'L2', lr0=0.02)
            if s1 < s0:
                z0 = z1
                s0 = s1
                print(f"  L2 improved to {s1:.6f}")

        # Phase 2: L4 optimization
        remaining = deadline - time.monotonic()
        if remaining > 120:
            dur = min(remaining * 0.25, 300.0)
            print(f"  Phase 2 (L4): {dur:.0f}s")
            z2, s2 = adam_optimize(z0, dur, rng, n, L, L2, 'L4', lr0=0.01)
            if s2 < s0:
                z0 = z2
                s0 = s2
                print(f"  L4 improved to {s2:.6f}")

        # Phase 3: L8 optimization
        remaining = deadline - time.monotonic()
        if remaining > 120:
            dur = min(remaining * 0.20, 250.0)
            print(f"  Phase 3 (L8): {dur:.0f}s")
            z3, s3 = adam_optimize(z0, dur, rng, n, L, L2, 'L8', lr0=0.005)
            if s3 < s0:
                z0 = z3
                s0 = s3
                print(f"  L8 improved to {s3:.6f}")

        # Phase 4: Smooth max with decreasing tau
        remaining = deadline - time.monotonic()
        if remaining > 100:
            dur = min(remaining * 0.25, 300.0)
            a_curr = softmax_z(z0)
            c_curr = compute_conv(a_curr, L)
            cm = c_curr.max()
            tau_schedule = [0.05 * cm, 0.03 * cm, 0.015 * cm, 0.008 * cm, 0.004 * cm, 0.002 * cm]
            print(f"  Phase 4 (SMAX_DES): {dur:.0f}s, tau from {tau_schedule[0]:.2e} to {tau_schedule[-1]:.2e}")
            z4, s4 = adam_optimize(z0, dur, rng, n, L, L2, 'SMAX_DES', lr0=0.004,

            if s4 < s0:
                z0 = z4
                s0 = s4
                print(f"  SMAX improved to {s4:.6f}")

        # Phase 5: Very fine SMAX
        remaining = deadline - time.monotonic()
        if remaining > 60:
            dur = min(remaining * 0.30, 250.0)
            a_curr = softmax_z(z0)
            c_curr = compute_conv(a_curr, L)
            cm = c_curr.max()
            tau_schedule = [0.003 * cm, 0.001 * cm, 0.0005 * cm]
            print(f"  Phase 5 (fine SMAX): {dur:.0f}s")
            z5, s5 = adam_optimize(z0, dur, rng, n, L, L2, 'SMAX_DES', lr0=0.001,
                                   tau_schedule=tau_schedule)
            if s5 < s0:
                z0 = z5
                s0 = s5
                print(f"  Fine SMAX improved to {s5:.6f}")

        # Get the result
        a_final = softmax_z(z0)
        s_final = fast_score(a_final)
        print(f"  Final score for n={n}: {s_final:.6f}")

        if s_final < best_eval_score:
            best_eval_score = s_final
            best_seq = a_final.copy()

        # Try a restart with perturbation
        remaining = deadline - time.monotonic()
        if remaining > 180:
            print(f"  Attempting restart with perturbation...")
            z_pert = z0 + rng.normal(0, 0.5, n)
            z_pert -= z_pert.mean()
            z_pert = np.clip(z_pert, -50, 50)

            # Quick L4 then SMAX
            dur1 = min(remaining * 0.3, 120.0)
            z_r, s_r = adam_optimize(z_pert, dur1, rng, n, L, L2, 'L4', lr0=0.008)
            remaining = deadline - time.monotonic()
            if remaining > 60 and s_r < s_final:
                a_curr = softmax_z(z_r)
                c_curr = compute_conv(a_curr, L)
                cm = c_curr.max()
                tau_s = [0.02 * cm, 0.005 * cm, 0.002 * cm]
                dur2 = min(remaining * 0.5, 180.0)
                z_r2, s_r2 = adam_optimize(z_r, dur2, rng, n, L, L2, 'SMAX_DES',
                                           lr0=0.002, tau_schedule=tau_s)
                if s_r2 < s_final:
                    a_r2 = softmax_z(z_r2)
                    s_final = s_r2
                    z0 = z_r2
                    print(f"  Restart improved to {s_final:.6f}")

            if s_final < best_eval_score:
                best_eval_score = s_final
                best_seq = softmax_z(z0).copy()

    # Try a different initialization strategy: structured smooth functions
    remaining = deadline - time.monotonic()
    if remaining > 300 and best_eval_score > 1.52:
        print(f"\n=== Trying structured initializations (remaining {remaining:.0f}s) ===")
        n = 8192
        L = next_pow2(2 * n - 1)
        L2 = next_pow2(3 * n - 2)

        inits = ['cos', 'linear', 'random_smooth', 'two_level']
        for init_type in inits:
            if time.monotonic() > deadline - 100:
                break
            z_init = make_initialization(n, init_type, rng)
            a_init = softmax_z(z_init)
            s_init = fast_score(a_init)
            print(f"  {init_type}: initial score {s_init:.6f}")
            if s_init >= best_eval_score:
                continue

            # Quick optimization: L4 then SMAX
            dur = min(120.0, (deadline - time.monotonic()) * 0.5)
            if dur < 30:
                break
            z_opt, s_opt = adam_optimize(z_init, dur, rng, n, L, L2, 'L4', lr0=0.015)
            remaining = deadline - time.monotonic()
            if remaining > 60 and s_opt < best_eval_score:
                a_curr = softmax_z(z_opt)
                c_curr = compute_conv(a_curr, L)
                cm = c_curr.max()
                tau_s = [0.03 * cm, 0.01 * cm, 0.003 * cm]
                dur2 = min(remaining * 0.5, 150.0)
                z_opt2, s_opt2 = adam_optimize(z_opt, dur2, rng, n, L, L2, 'SMAX_DES',
                                               lr0=0.002, tau_schedule=tau_s)
                if s_opt2 < best_eval_score:
                    best_eval_score = s_opt2
                    best_seq = softmax_z(z_opt2).copy()
                    print(f"  {init_type} improved to {s_opt2:.6f}")

    # Final: try subgradient steps on the best solution
    remaining = deadline - time.monotonic()
    if remaining > 60 and best_seq is not None:
        print(f"\n=== Subgradient polish (remaining {remaining:.0f}s) ===")
        a = best_seq.copy()
        n = len(a)
        L = next_pow2(2 * n - 1)
        L2 = next_pow2(3 * n - 2)
        z = np.log(np.maximum(a, 1e-30))
        z -= z.mean()

        eta = 0.001
        for sg_step in range(10000):
            if time.monotonic() > deadline - 5:
                break
            a = softmax_z(z)
            c = compute_conv(a, L)
            score = 2.0 * n * c.max()

            # Subgradient: use top-5 peaks weighted
            top_k = 5
            idx = np.argsort(c)[-top_k:]
            w = np.zeros(len(c))
            max_c = c.max()
            for i in idx:
                w[i] = c[i] / max_c
            w /= w.sum()

            a_flip = a[::-1]
            conv = np.fft.irfft(np.fft.rfft(w, L2) * np.fft.rfft(a_flip, L2), L2)
            grad_a = 2.0 * conv[n - 1:n - 1 + n]
            gz = z_grad(grad_a, a)

            z = z - eta * gz
            z = np.where(np.isfinite(z), z, 0)
            z -= z.mean()
            z = np.clip(z, -50, 50)

            if sg_step % 2000 == 0:
                a_check = softmax_z(z)
                s_check = 2.0 * n * compute_conv(a_check, L).max()
                print(f"  sg step {sg_step}: score={s_check:.6f}")
                if s_check > score:
                    eta *= 0.5
                else:
                    eta = min(eta * 1.05, 0.01)
            else:
                if sg_step % 500 == 0:
                    a_check = softmax_z(z)
                    s_check = 2.0 * n * compute_conv(a_check, L).max()
                    if s_check < best_eval_score:
                        best_eval_score = s_check
                        best_seq = a_check.copy()

    # Final validation with evaluate_sequence
    print(f"\n=== Final validation ===")
    candidates = [best_seq]
    # Add a few fallbacks
    candidates.append(np.ones(1024) / 1024)

    final_seq = None
    final_score = np.inf

    for cand in candidates:
        if cand is None:
            continue
        cand = np.clip(cand, 0, 1000)
        lst = [float(x) for x in cand]
        if not lst:
            continue
        try:
            sc = evaluate_sequence(lst)
        except Exception:
            sc = np.inf
        if np.isfinite(sc) and sc < final_score:
            final_score = float(sc)
            final_seq = lst
            print(f"  Candidate length {len(lst)}: score {sc:.6f}")

    if final_seq is None:
        final_seq = [1.0 / 1024.0] * 1024

    print(f"\nFinal best score: {final_score:.6f}")
    print(f"Total time: {time.monotonic() - t_start:.1f}s")
    return final_seq
