"""
Search for non-negative step-function heights minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: 
  - Diverse starting points (power-law, symmetric, from previous best)
  - Large-p norm surrogates (p=4, 16, 32) with L-BFGS for smooth optimization
  - Log-sum-exp surrogate with decreasing temperature for peak targeting
  - Hybrid local search: time-domain multiplicative + frequency-domain phase perturbations
  - Peak-targeting moves that specifically reduce the maximum convolution lag

import numpy as np
import time
from scipy.optimize import minimize as _minimize


def _pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _clean(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    return np.maximum(v, 0.0)


def _normalize(v):
    v = _clean(v)
    s = float(v.sum())
    if not np.isfinite(s) or s <= 1e-15:
        n = v.size
        return np.full(n, 1.0 / float(n), dtype=np.float64) if n > 0 else v
    return v / s


def _project_simplex(v):
    v = _clean(v)
    n = v.size
    if n == 0:
        return v.copy()
    x = np.sort(v)[::-1]
    css = np.cumsum(x)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = x > (css - 1.0) / idx
    if not np.any(cond):
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    r = int(np.nonzero(cond)[0][-1])
    tau = (css[r] - 1.0) / float(r + 1)
    out = v - tau
    np.maximum(out, 0.0, out=out)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    out *= 1.0 / s
    return out


def _score(a, L=None):
    v = _clean(a)
    n = v.size
    if n == 0:
        return np.inf
    s = float(v.sum())
    if s < 1e-12:
        return np.inf
    v = v / s
    if L is None:
        L = _pow2(2 * n - 1)
    A = np.fft.rfft(v, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    mx = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(mx):
        return np.inf
    return 2.0 * float(n) * mx


def _convolve_fft(a, L):
    A = np.fft.rfft(a, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)
    return np.maximum(c, 0.0)


def _ratios_from_a(a):
    a = _clean(a).copy()
    n = a.size
    if n == 0:
        return a
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.zeros(max(0, n - 1), dtype=np.float64)
    a = a / s
    if n == 1:
        return np.zeros(0, dtype=np.float64)
    ref = float(a[-1])
    if ref <= 1e-200:
        a[-1] = 1e-12
        a = a / float(a.sum())
        ref = float(a[-1])
    x = np.log(np.maximum(a[:-1], 1e-300) / max(ref, 1e-300))
    return np.clip(x, -80.0, 80.0)


def _a_from_ratios(x):
    x = np.asarray(x, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    n = x.size + 1
    if n == 1:
        return np.ones(1, dtype=np.float64)
    m = 0.0
    if x.size:
        mx = float(np.max(x))
        if mx > m:
            m = mx
    e = np.exp(np.clip(x - m, -745.0, 709.0))
    e_last = float(np.exp(-m))
    s = float(e.sum()) + e_last
    if not np.isfinite(s) or s <= 0.0:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    a = np.empty(n, dtype=np.float64)
    a[:-1] = e / s
    a[-1] = e_last / s
    return a


def _p_surrogate_grad(x, p, L):
    """Objective: sum(c^p), gradient w.r.t. log-ratio parameterization."""
    a = _a_from_ratios(x)
    n = a.size
    if n <= 1:
        return 0.0, np.zeros_like(x)
    A = np.fft.rfft(a, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    c = np.maximum(c, 1e-300)
    pp = float(p)
    f = float(np.sum(np.power(c, pp)))
    w = np.power(c, pp - 1.0)
    W = np.fft.rfft(w, n=L)
    ga = 2.0 * pp * np.fft.irfft(W * np.conj(A), n=L)[:n]
    dot = float(np.dot(a, ga))
    gfull = a * (ga - dot)
    gy = gfull[:-1]
    if not np.all(np.isfinite(gy)):
        gy = np.zeros_like(x)
    if not np.isfinite(f):
        f = 1e300
    return f, gy


def _lse_surrogate_grad(x, eps, L):
    """Objective: eps * logsumexp(c/eps) (smooth max), gradient w.r.t. log-ratio params."""
    a = _a_from_ratios(x)
    n = a.size
    if n <= 1:
        return 0.0, np.zeros_like(x)
    A = np.fft.rfft(a, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2 * n - 1]
    c = np.maximum(c, 1e-300)
    cmax = float(c.max())
    if cmax <= 1e-300:
        return 0.0, np.zeros_like(x)
    eps = max(float(eps), 1e-10)
    z = (c - cmax) / eps
    z = np.clip(z, -745.0, 0.0)
    e = np.exp(z)
    sume = float(e.sum())
    if not np.isfinite(sume) or sume <= 0.0:
        return 1e300, np.zeros_like(x)
    f = cmax + eps * float(np.log(sume))
    w = e / sume
    W = np.fft.rfft(w, n=L)
    ga = 2.0 * np.fft.irfft(W * np.conj(A), n=L)[:n]
    dot = float(np.dot(a, ga))
    gfull = a * (ga - dot)
    gy = gfull[:-1]
    if not np.all(np.isfinite(gy)):
        gy = np.zeros_like(x)
    return f, gy


def _run_lbfgs(fun, x, args, maxiter, bounds):
    if x.size == 0:
        return x, 1
    options = {"maxiter": int(maxiter), "disp": 0, "maxfun": int(max(200, 15 * maxiter)),
               "ftol": 1e-16, "gtol": 1e-10}
    try:
        res = _minimize(fun, x, args=args, method="L-BFGS-B", jac=True,
                        bounds=bounds, options=options)
    except Exception:
        return x, 99
    x_new = np.asarray(res.x, dtype=np.float64)
    x_new = np.where(np.isfinite(x_new), x_new, 0.0)
    x_new = np.clip(x_new, -80.0, 80.0)
    try:
        status = int(res.status)
    except Exception:
        status = 0
    return x_new, status


def _power_seq(n, p=0.5, off=1.0):
    j = np.arange(n, dtype=np.float64)
    x = np.maximum(j + float(off), 1e-8)
    return _project_simplex(np.power(x, -float(p)))


def _sym_power_seq(n, p=0.5, off=1.0):
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    dist = np.abs(j - center)
    x = np.maximum(dist + float(off), 1e-8)
    return _project_simplex(np.power(x, -float(p)))


def _sigmoid_seq(n, sharp=2.0, center_frac=0.3):
    j = np.arange(n, dtype=np.float64)
    t = j / max(1, n - 1)
    v = 1.0 / (1.0 + np.exp(float(sharp) * (t - float(center_frac)) * 10.0))
    return _project_simplex(v)


def _generate_starts(n):

    try:

    except Exception:

    cands.append(("uniform", np.full(n, 1.0 / n)))
    for p in (0.4, 0.45, 0.5, 0.55, 0.6):

    rng = np.random.default_rng(12345)

def _local_search_hybrid(a0, end_time, seed, L):
    """Hybrid local search: time-domain block + frequency-domain phase perturbations."""
    a = _project_simplex(a0)
    n = a.size
    Ls = _pow2(2 * n - 1)
    best = a.copy()
    best_sc = _score(best, Ls)
    rng = np.random.default_rng(seed)
    sigma_t = 0.15  # time-domain sigma
    sigma_p = 0.3   # phase perturbation sigma
    last_improve = time.monotonic()
    start_t = last_improve
    n_eval = 0

    while time.monotonic() < end_time:
        elapsed = time.monotonic() - start_t
        # Increase aggressiveness if stuck
        stuck = time.monotonic() - last_improve
        if stuck > 10.0:
            sigma_t = max(0.03, sigma_t * 0.8)
            sigma_p = max(0.05, sigma_p * 0.8)
        if stuck > 30.0 and elapsed > 60.0:
            # Big restart
            a = _project_simplex(best * np.exp(rng.normal(0, 0.1, n)))
            sigma_t = 0.25
            sigma_p = 0.5

        move_type = rng.random()
        if move_type < 0.5:
            # Time-domain: multiplicative block perturbation
            cand = best.copy()
            nblocks = 1 + int(rng.integers(0, 3))
            for _ in range(nblocks):
                maxw = max(5, n // 6)
                w = int(np.exp(rng.uniform(np.log(2.0), np.log(float(maxw)))))
                w = max(1, min(w, n))
                s = int(rng.integers(0, n - w + 1))
                cand[s:s + w] *= float(np.exp(rng.normal(0, sigma_t)))
            cand = _project_simplex(cand)
        elif move_type < 0.8:
            # Frequency-domain: phase perturbation
            A = np.fft.rfft(a if np.array_equal(a, best) else best, n=L)
            # Perturb phase of positive frequency bins
            phase = np.angle(A)
            mag = np.abs(A)
            # Random subset of frequency bins
            nf = phase.size
            n_perturb = max(1, int(nf * 0.1))
            idx = rng.integers(0, nf, size=n_perturb)
            phase[idx] += rng.normal(0, sigma_p, size=n_perturb)
            A_new = mag * np.exp(1j * phase)
            a_new = np.fft.irfft(A_new, n=L)[:n]
            cand = _project_simplex(a_new)
        else:
            # Peak-targeting: reduce elements contributing most to max lag
            c = _convolve_fft(best, Ls)
            m_star = int(np.argmax(c[:2 * n - 1]))
            cand = best.copy()
            # Find top contributors
            contribs = []
            for j in range(n):
                cj = 0.0
                if 0 <= m_star - j < 2 * n - 1:
                    cj += best[max(0, min(n - 1, m_star - j))]
                if 0 <= j - m_star < 2 * n - 1:
                    cj += best[max(0, min(n - 1, j - m_star))]
                contribs.append(cj)
            contribs = np.array(contribs)
            # Perturb top contributors downward
            top_k = max(5, n // 100)
            top_idx = np.argsort(contribs)[-top_k:]
            cand[top_idx] *= np.exp(rng.normal(-sigma_t * 0.5, sigma_t * 0.3, size=top_k))
            # Compensate elsewhere
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_comp = min(20, len(other))
                comp_idx = rng.choice(other, size=n_comp, replace=False)
                cand[comp_idx] *= np.exp(rng.normal(0, sigma_t * 0.4, size=n_comp))
            cand = _project_simplex(cand)

        sc = _score(cand, Ls)
        n_eval += 1
        if np.isfinite(sc) and sc < best_sc - 1e-14:
            best = cand
            best_sc = sc
            last_improve = time.monotonic()
            sigma_t = min(0.4, sigma_t * 1.05)
            sigma_p = min(1.0, sigma_p * 1.05)

    return best, best_sc

        allocs = [(0.60, 0), (0.25, 1), (0.15, 2)]
    elif top_k == 2:
        allocs = [(0.70, 0), (0.30, 1)]
    else:
        allocs = [(1.0, 0)]

    global_best_a = np.asarray(scored[0][2], dtype=np.float64).copy()
    global_best_sc = float(scored[0][0])

    t_cursor = t0
    for frac, idx in allocs:
        alloc_end = min(deadline, t0 + (t_cursor - t0) + frac * budget * 0.92)
        if alloc_end <= time.monotonic() + 5.0:
            break
        name = scored[idx][1]
        a0 = np.asarray(scored[idx][2], dtype=np.float64).copy()
        print(f"\n=== Refining '{name}' (initial score={scored[idx][0]:.8f}) ===")
        a, sc = _refine_candidate(a0, alloc_end, int(seed) + 100 * (idx + 1), L)
        if np.isfinite(sc) and sc < global_best_sc:
            global_best_sc = sc
            global_best_a = a.copy()
            print(f"  NEW GLOBAL BEST: {global_best_sc:.8f}")
        t_cursor = max(t_cursor, time.monotonic())

    # Final: if time remains, do one more local search on global best
    remaining = deadline - time.monotonic()
    if remaining > 10.0:
        print(f"\n=== Final local search ({remaining:.1f}s remaining) ===")
        a, sc = _local_search_hybrid(global_best_a, deadline, int(seed) + 99999, L)
        if np.isfinite(sc) and sc < global_best_sc:
            global_best_sc = sc
            global_best_a = a.copy()
            print(f"  FINAL BEST: {global_best_sc:.8f}")

    global_best_a = _project_simplex(global_best_a)
    out = [float(x) for x in global_best_a]

    # Final validation
    try:
        val = evaluate_sequence(out)
        print(f"\nFinal sequence length: {len(out)}, sum={sum(out):.4f}")
        print(f"evaluate_sequence result: {val:.8f}")
        if not np.isfinite(val):
            out = [float(x) for x in _power_seq(n, 0.5, 1.0)]
    except Exception:
        pass

    elapsed = time.monotonic() - t0
    print(f"Total time: {elapsed:.1f}s, final internal score: {global_best_sc:.8f}")
    return out
