import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]

def _solve_lp(f, M, max_iter=None):
    """Solve the LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    size = 2 * n - 1
    
    # Build constraint matrix A (size x n)
    # A[k, j] = f[j] if max(0, k-n+1) <= j <= min(k, n-1)
    rows = []
    cols = []
    vals = []
    
    for k in range(size):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[j])
    
    A = sp.csr_matrix((vals, (rows, cols)), shape=(size, n))
    b = np.full(size, M)
    
    # Maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    result = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    
    if result.success:
        return result.x
    else:
        return None

def _optimize_lp(a0, time_budget, rng, t_blend=0.7):
    """Optimize using LP-based best response iteration."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, np.inf
    a = a / s
    
    c = _fft_convolve(a)
    best_score = 2.0 * n * float(np.max(c))
    best_a = a.copy()
    
    deadline = time.monotonic() + time_budget
    it = 0
    last_print = time.monotonic()
    
    while time.monotonic() < deadline:
        c = _fft_convolve(a)
        k_max = int(np.argmax(c))
        M = c[k_max]
        score = 2.0 * n * M
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            it_reset = 0
            if time.monotonic() - last_print > 5.0:
                print(f"    [LP] it={it}, score={score:.6f}")
                last_print = time.monotonic()
        else:
            it_reset += 1
        
        if it_reset > 50:
            # Perturb and restart
            a = best_a * np.exp(rng.normal(0, 0.05, n))
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            it_reset = 0
            it += 1
            continue
        
        # Solve LP
        try:
            g = _solve_lp(a, M)
        except Exception as e:
            print(f"    LP failed: {e}")
            break
        
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum  # Normalize g to sum to 1

        s_new = a_new.sum()

            break
        a = a_new / s_new

def _build_lp_matrix(f, n):
    size = 2 * n - 1
    # For each row k, the non-zero columns are max(0,k-n+1) to min(k, n-1)
    # We can build this using a banded structure

    total_nnz = nnz_per_row.sum()
    
    rows = np.zeros(total_nnz, dtype=np.int32)
    cols = np.zeros(total_nnz, dtype=np.int32)
    vals = np.zeros(total_nnz, dtype=np.float64)
    
    idx = 0
    for k in range(size):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        count = j_hi - j_lo + 1
        rows[idx:idx+count] = k
        cols[idx:idx+count] = np.arange(j_lo, j_hi + 1)
        vals[idx:idx+count] = f[j_lo:j_hi+1]
        idx += count
    
    return sp.csr_matrix((vals, (rows, cols)), shape=(size, n))

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses LP-based best-response iteration combined with subgradient descent,
with multiple initializations including multi-tone sinusoidal patterns.
"""
import numpy as np
from scipy.optimize import linprog
import scipy.sparse as sp
import time

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]

def _score(a):
    c = _fft_convolve(a)
    s = a.sum()
    return 2.0 * len(a) * float(np.max(c)) / (s * s)

def _build_lp_matrix(f, n):
    """Build sparse matrix A where A[k,j] = f[j] if max(0,k-n+1)<=j<=min(k,n-1)."""
    size = 2 * n - 1
    rows_list = []
    cols_list = []
    vals_list = []
    for k in range(size):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_lo > j_hi:
            continue
        j_range = np.arange(j_lo, j_hi + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
        vals_list.append(f[j_lo:j_hi+1])
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    vals = np.concatenate(vals_list)
    return sp.csr_matrix((vals, (rows, cols)), shape=(size, n))

def _solve_lp(f, M):
    """Solve: max sum(g) s.t. A@g <= M*ones, g >= 0."""
    n = len(f)
    size = 2 * n - 1
    A = _build_lp_matrix(f, n)
    b = np.full(size, M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if result.success:
        return result.x
    return None

def _optimize_lp(a0, time_budget, rng, t_blend=0.5):
    """LP-based best response iteration."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), np.inf
    a = a / s
    
    c = _fft_convolve(a)
    best_score = 2.0 * n * float(np.max(c))
    best_a = a.copy()
    
    deadline = time.monotonic() + time_budget
    it = 0
    no_improve = 0
    last_print = time.monotonic()
    
    # Precompute LP matrix (f changes each iteration, so we rebuild)
    # Actually f = a changes each iteration, so we can't precompute.
    # But we can cache the sparsity pattern.
    
    while time.monotonic() < deadline:
        c = _fft_convolve(a)
        M = float(np.max(c))
        score = 2.0 * n * M
        
        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            if time.monotonic() - last_print > 10.0:
                print(f"    [LP] it={it}, score={score:.6f}")
                last_print = time.monotonic()
        else:
            no_improve += 1
        
        if no_improve > 30:
            # Add noise and restart
            a = best_a * np.exp(rng.normal(0, 0.1, n))
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            no_improve = 0
            it += 1
            continue
        
        # Solve LP
        t0 = time.monotonic()
        g = _solve_lp(a, M)
        lp_time = time.monotonic() - t0
        
        if g is None:
            print(f"    LP failed at it={it}")
            break
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        # Check: does g improve things?
        # Compute (a*g) to verify constraints
        # Blend
        a_new = (1.0 - t_blend) * a + t_blend * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new
        
        it += 1
        if it % 10 == 0:
            c_check = _fft_convolve(a)
            print(f"    [LP] it={it}, score={2.0*n*c_check.max():.6f}, lp_time={lp_time:.3f}s")
    
    return best_a, best_score

def _optimize_subgrad(a0, time_budget, rng):
    """Subgradient descent with momentum and adaptive step size."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), np.inf
    a = a / s
    
    c = _fft_convolve(a)
    best_score = 2.0 * n * float(np.max(c))
    best_a = a.copy()
    
    eta = 0.1
    no_improve = 0
    momentum = np.zeros(n)
    mu = 0.7
    deadline = time.monotonic() + time_budget
    it = 0
    last_print = time.monotonic()
    
    while time.monotonic() < deadline:
        c = _fft_convolve(a)
        k_max = int(np.argmax(c))
        score = 2.0 * n * c[k_max]
        
        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            if time.monotonic() - last_print > 5.0:
                print(f"    [SGD] it={it}, score={score:.6f}")
                last_print = time.monotonic()
        else:
            no_improve += 1
        
        if no_improve > 200:
            a = best_a * np.exp(rng.normal(0, 0.05, n))
            a = np.maximum(a, 1e-20)
            a = a / a.sum()
            no_improve = 0
            momentum = np.zeros(n)
            eta = 0.1
            it += 1
            continue
        
        # Subgradient
        g = np.zeros(n)
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        j_range = np.arange(lo, hi + 1)
        indices = k_max - j_range
        g[lo:hi+1] = 2.0 * a[indices]
        
        g_max = g.max()
        if g_max < 1e-15:
            break
        g_norm = g / g_max
        momentum = mu * momentum + (1.0 - mu) * g_norm
        
        a = a * np.exp(-eta * momentum)
        a = np.maximum(a, 1e-20)
        s = a.sum()
        a = a / s
        
        if no_improve > 100:
            eta *= 0.97
        elif no_improve < 20:
            eta = min(eta * 1.01, 0.3)
        
        it += 1
    
    return best_a, best_score
