def mirror_descent_momentum(a, deadline, seed=0, momentum=0.9):
    n = a.size
    N = next_pow2(3*n - 2)
    rng = np.random.default_rng(seed)
    
    best_a = a.copy()
    best_score = internal_score(a, N)
    
    loga = np.log(a + 1e-20)
    loga -= loga.max()
    
    v = np.zeros(n)  # momentum buffer
    eta = 0.8
    tau_frac = 0.3  # temperature as fraction of max_conv
    stale = 0
    it = 0
    
    while time.monotonic() < deadline:
        c = convolve_fft(a, N)
        cmax = float(c.max())
        score = 2.0 * n * cmax
        
        if score < best_score * (1 - 1e-12):
            best_score = score
            best_a = a.copy()
        
        # Compute weights for near-peak lags
        tau = tau_frac * cmax
        z = (c - cmax) / tau
        z = np.clip(z, -30, 0)
        ez = np.exp(z)
        s = ez.sum()
        w = ez / s
        
        # Get peak lags (where w is significant)
        # w is indexed 0 to 2n-2, corresponding to lags -(n-1) to n-1
        n_half = n - 1
        peak_lags = np.arange(-(n-1), n)
        # Only keep lags with w > 0.01 * max(w)
        thresh = 0.01 * w.max()
        significant = np.nonzero(w > thresh)[0]
        
        # Compute gradient
        grad = np.zeros(n)
        for idx in significant:
            k = int(idx - n_half)
            wval = float(w[idx])
            k_abs = abs(k)
            if k_abs >= n:
                continue
            lo = max(0, k_abs - n + 1)
            hi = min(n - 1, k_abs)
            cnt = hi - lo + 1
            if cnt == 1:
                grad[lo] += 2.0 * wval * a[k_abs - lo]
            else:
                grad[lo:hi+1] += 2.0 * wval * a[k_abs-lo : k_abs-hi-1 : -1]
        
        # Center the gradient (to maintain sum constraint)
        grad -= grad.mean()
        
        # Normalize
        scale = np.abs(grad).max()
        if scale < 1e-15:
            break
        grad /= scale
        
        # Momentum
        v = momentum * v + (1 - momentum) * grad
        
        # Update
        loga_try = loga - eta * v
        loga_try -= loga_try.max()
        a_try = np.exp(np.clip(loga_try, -50, 0))
        a_try = a_try / a_try.sum()
        a_try = np.maximum(a_try, 1e-15)
        a_try = a_try / a_try.sum()
        
        c_try = convolve_fft(a_try, N)
        score_try = 2.0 * n * c_try.max()
        
        # Backtracking
        if score_try < score * (1 - 1e-10):
            a = a_try
            loga = np.log(a + 1e-20) - np.log(a + 1e-20).max()
            stale = 0
            eta = min(2.0, eta * 1.02)
        else:
            # Try smaller step
            accepted = False
            for shrink in (0.5, 0.25, 0.125):
                loga_try2 = loga - eta * shrink * v
                loga_try2 -= loga_try2.max()
                a_try2 = np.exp(np.clip(loga_try2, -50, 0))
                a_try2 = a_try2 / a_try2.sum()
                a_try2 = np.maximum(a_try2, 1e-15)
                a_try2 = a_try2 / a_try2.sum()
                c_try2 = convolve_fft(a_try2, N)
                score_try2 = 2.0 * n * c_try2.max()
                if score_try2 < score * (1 - 1e-10):
                    a = a_try2
                    loga = np.log(a + 1e-20) - np.log(a + 1e-20).max()

                    a = np.maximum(a, 1e-15)

                    stale = 0

def lp_direction(a, M, N=None):
    """
    Find g ≥ 0 maximizing sum(g) subject to (a*g)[k] ≤ M for all k.
    a: length n, sum=1
    M: the current max convolution value
    Returns: g (length n, sum=1)
    """
    n = a.size
    # Variables: g_0, ..., g_{n-1}
    # Objective: maximize sum(g_i) → minimize -sum(g_i)
    c_obj = -np.ones(n)
    
    # Constraints: (a*g)[k] ≤ M for k = 0, ..., 2n-2
    # (a*g)[k] = sum_i a_i * g_{k-i} (for valid indices)
    # This is linear in g!
    
    # Constraint matrix A_ub: (2n-1) × n
    # A_ub[k, j] = a[j] if max(0, k-n+1) ≤ j ≤ min(n-1, k), else 0
    # Wait, (a*g)[k] = sum_i a_i * g[k-i]
    # For this to involve g_j, we need k-i = j, i.e., i = k-j
    # So (a*g)[k] = sum_j a[k-j] * g_j (where a[k-j] is 0 if k-j not in [0,n))
    # A_ub[k, j] = a[k-j] if 0 ≤ k-j < n, else 0
    
    A_ub = np.zeros((2*n-1, n))
    for j in range(n):
        # g_j contributes to (a*g)[k] for k = j, j+1, ..., j+n-1
        # (a*g)[k] += a[k-j] * g_j
        # So A_ub[k, j] = a[k-j] for k in [j, j+n-1]
        if j + n - 1 < 2*n - 1:
            A_ub[j:j+n, j] = a[:n]  # a[0], a[1], ..., a[n-1]
    
    b_ub = M * np.ones(2*n-1)
    
    # Bounds: g_i ≥ 0
    bounds = [(0, None)] * n
    
    from scipy.optimize import linprog
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0)
        g = g / g.sum()
        return g
    return None

"""
Search for nonnegative step-function heights minimizing
    2*n*max(convolve(a,a)) / (sum(a))^2
Using momentum mirror descent with multi-peak smoothing, LP-based direction finding,
simulated annealing in log-space, and low-dim subspace optimization at multiple lengths.

from scipy.optimize import minimize as scipy_minimize

def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def normalize_positive(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, float(floor), None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def convolve_fft(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if N is None:
        N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2*n-1]


def internal_score(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s <= 1e-15:
        return np.inf
    a_n = a / s
    c = convolve_fft(a_n, N)
    return float(2.0 * n * float(np.max(c)))

def make_uniform(n):

def resample_seq(a, n):
    a = normalize_positive(a, floor=0.0)
    old_n = a.size
    n = max(2, int(n))
    if old_n == 1:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    if old_n == n:
        return a.astype(np.float64, copy=True)
    old_x = np.linspace(0.0, 1.0, old_n, dtype=np.float64)
    new_x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    b = np.interp(new_x, old_x, a)
    return normalize_positive(b, floor=1e-12)


def compute_grad_from_weights(a, w, N):
    """Compute gradient of sum_k w[k]*(a*a)[k] w.r.t. a using FFT.
    a: length n, sum=1
    w: length 2n-1, weights for each lag
    N: FFT size
    Returns: gradient of length n
    """
    n = a.size
    W_pad = np.zeros(N, dtype=np.float64)
    W_pad[:2*n-1] = w
    a_pad = np.zeros(N, dtype=np.float64)
    a_pad[:n] = a
    # Cross-correlation: (W * a_conj) in freq domain
    FW = np.fft.rfft(W_pad, N)
    Fa = np.fft.rfft(a_pad, N)
    C = np.fft.irfft(FW * np.conj(Fa), N)
    return 2.0 * C[:n]


def compute_grad_direct(a, w):
    """Direct (slower) gradient computation for verification or small n."""
    n = a.size
    grad = np.zeros(n, dtype=np.float64)
    n_half = n - 1
    for idx in range(2*n-1):
        wi = w[idx]
        if wi < 1e-15:
            continue
        k = idx - n_half
        k_abs = abs(k)
        if k_abs >= n:
            continue
        lo = max(0, k_abs - n + 1)
        hi = min(n - 1, k_abs)
        cnt = hi - lo + 1
        if cnt == 1:
            grad[lo] += 2.0 * wi * a[k_abs - lo]
        else:
            grad[lo:hi+1] += 2.0 * wi * a[k_abs-lo : k_abs-hi-1 : -1]
    return grad


def soft_max_value(c, cmax, tau):
    """Compute soft max: cmax + tau * log(sum(exp((c-cmax)/tau)))"""
    if cmax <= 0 or tau <= 0:
        return cmax
    z = (c - cmax) / tau
    z = np.clip(z, -50.0, 0.0)
    s = float(np.exp(z).sum())
    if s <= 0 or not np.isfinite(s):
        return cmax
    return cmax + tau * np.log(s)


def mirror_descent_momentum(a0, deadline, seed=0, N=None, momentum=0.75, use_lp=False, lp_interval=80):
    """Mirror descent with momentum and multi-peak smoothing."""
    a = normalize_positive(a0, floor=1e-15)
    n = a.size
    if N is None:
        N = next_pow2(3 * n - 2)
    rng = np.random.default_rng(seed)

    best_a = a.copy()
    best_score = internal_score(a, N)

    loga = np.log(a + 1e-20)
    loga -= loga.max()

    v = np.zeros(n, dtype=np.float64)  # momentum
    eta = 0.8
    tau_frac = 0.35
    stale = 0
    it = 0
    last_print = 0

    while time.monotonic() < deadline:
        c = convolve_fft(a, N)
        cmax = float(c.max())
        if cmax <= 0 or not np.isfinite(cmax):
            break
        score = 2.0 * n * cmax

        if np.isfinite(score) and score < best_score * (1.0 - 1e-12):
            best_score = score
            best_a = a.copy()

        # Compute weights
        tau = max(tau_frac * cmax, 1e-15)
        z = (c - cmax) / tau
        z = np.clip(z, -40.0, 0.0)
        ez = np.exp(z)
        s = float(ez.sum())
        if s <= 0 or not np.isfinite(s):
            break
        w = ez / s

        # Gradient via FFT
        grad = compute_grad_from_weights(a, w, N)
        grad -= grad.mean()
        scale = float(np.abs(grad).max())
        if scale < 1e-15:
            break
        grad /= scale

        # LP direction (periodically)
        if use_lp and it % lp_interval == 0 and it > 0:
            try:
                g_lp = lp_direction(a, cmax, n)
                if g_lp is not None:
                    # Blend LP direction with current gradient
                    lp_grad = g_lp - g_lp.mean()
                    lp_scale = float(np.abs(lp_grad).max())
                    if lp_scale > 1e-10:
                        lp_grad /= lp_scale
                        grad = 0.6 * grad + 0.4 * lp_grad
                        grad -= grad.mean()
                        scale2 = float(np.abs(grad).max())
                        if scale2 > 1e-15:
                            grad /= scale2
            except Exception:
                pass

        # Momentum update
        v = momentum * v + (1.0 - momentum) * grad
        v_scale = float(np.abs(v).max())
        if v_scale < 1e-15:
            break
        v /= v_scale

        # Try update with backtracking
        accepted = False
        for shrink in (1.0, 0.5, 0.25, 0.125):
            eff_eta = eta * shrink
            if eff_eta < 1e-8:
                break
            loga_try = loga - eff_eta * v
            loga_try -= loga_try.max()
            a_try = np.exp(np.clip(loga_try, -60.0, 0.0))
            a_try = np.maximum(a_try, 1e-15)
            a_try = a_try / a_try.sum()

            c_try = convolve_fft(a_try, N)
            score_try = float(2.0 * n * float(c_try.max()))

            if np.isfinite(score_try) and score_try < score * (1.0 - 1e-10):
                a = a_try
                loga = np.log(a + 1e-20)
                loga -= loga.max()
                score = score_try
                accepted = True
                break

        if accepted:
            stale = 0
            eta = min(2.5, eta * 1.015)
            tau_frac = max(0.05, tau_frac * 0.997)
        else:
            stale += 1
            eta = max(0.005, eta * 0.65)
            if stale % 150 == 0:
                tau_frac = min(0.6, tau_frac * 1.8)
                # Random kick to escape
                kick = rng.normal(0, 0.15, n)
                kick -= kick.mean()
                loga += 0.4 * kick
                loga -= loga.max()
                a = np.exp(np.clip(loga, -60.0, 0.0))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                v *= 0.5
                stale = 0
            if stale > 600:
                break

        it += 1
        if it % 800 == 0 and it - last_print >= 800:
            print(f"  MD n={n} it={it} score={score:.9f} best={best_score:.9f} eta={eta:.3f} tau={tau_frac:.3f}", flush=True)
            last_print = it

    return best_a, best_score


def lp_direction(a, M_max, n=None):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M_max, g >= 0. Returns g normalized."""
    if n is None:
        n = a.size
    # A_ub[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # (a*g)[k] = sum_j a[k-j] * g_j
    A_ub = np.zeros((2*n-1, n), dtype=np.float64)
    for j in range(n):
        lo_k = j
        hi_k = min(2*n-2, j+n-1)
        cnt = hi_k - lo_k + 1
        A_ub[lo_k:hi_k+1, j] = a[:cnt]

    c_obj = -np.ones(n, dtype=np.float64)
    b_ub = M_max * np.ones(2*n-1, dtype=np.float64)
    bounds = [(0.0, None)] * n

    try:
        res = _linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-15:
                return g / gs
    except Exception:
        pass
    return None


def lowdim_optimize(a0, n_target, deadline, seed=0):

    B = build_basis(n)

    try:
        theta0, _, _, _ = np.linalg.lstsq(B, z, rcond=None)
        theta0 = np.asarray(theta0, dtype=np.float64)
    except Exception:
        theta0 = np.zeros(D)

    def obj_and_grad(theta, tau):
        alpha = B @ theta
        alpha = alpha - alpha.max()
        alpha = np.clip(alpha, -55.0, 0.0)
        e = np.exp(alpha)
        a_s = e / e.sum()
        a_s = np.maximum(a_s, 1e-300)
        a_s = a_s / a_s.sum()
        c = convolve_fft(a_s, N)
        cmax = float(c.max())
        if cmax <= 0:
            return 0.0, np.zeros(D)
        smax = soft_max_value(c, cmax, tau)
        score = 2.0 * n * smax

        # Gradient
        tau_eff = max(tau, 1e-12)
        zc = (c - cmax) / tau_eff
        zc = np.clip(zc, -40.0, 0.0)
        ez = np.exp(zc)
        ws = ez / ez.sum()
        ga = compute_grad_from_weights(a_s, ws, N)
        total = float(a_s @ ga)
        aB = a_s @ B
        agaB = (a_s * ga) @ B
        gtheta = agaB - total * aB
        return score, 2.0 * n * gtheta

    best_a = a_n.copy()
    best_score = internal_score(a_n, N)
    theta = theta0.copy()

    # Multi-start
    starts = [theta.copy()]
    for std in (0.5, 1.0, 2.0):
        starts.append(theta0 + rng.normal(0, std, D))
    for i in range(min(D, 6)):
        t = np.zeros(D)
        t[i] = 1.0
        starts.append(t)

    for theta_s in starts:
        if time.monotonic() > deadline:
            break
        theta = theta_s.copy()
        for fac in (0.3, 0.12, 0.05, 0.02):

                break
            tau = fac / n
            try:

def build_basis(n):
    n = max(8, int(n))
    i = np.arange(n, dtype=np.float64)
    x = (i - 0.5*(n-1)) / max(1.0, 0.5*n)
    cols = []
    # Antisymmetric sine modes
    for m in range(1, 7):
        cols.append(np.sin(np.pi * m * x))
    # Polynomials
    for p in range(1, 5):
        cp = x**p - np.mean(x**p)
        cols.append(cp)
    # Cosine modes
    for m in range(1, 5):
        t = 2.0 * np.pi * m * i / n
        cols.append(np.cos(t) - np.mean(np.cos(t)))
        cols.append(np.sin(t) - np.mean(np.sin(t)))
    # Gaussian bumps
    for mu in (0.2, 0.35, 0.5, 0.65, 0.8):
        cols.append(np.exp(-8.0*(x-mu)**2) - np.mean(np.exp(-8.0*(x-mu)**2)))
    B = np.column_stack(cols)
    B -= B.mean(axis=0)
    norms = np.linalg.norm(B, axis=0)
    norms[norms < 1e-14] = 1.0
    B = B / norms
    # Orthonormalize
    Q, _ = np.linalg.qr(B)
    return Q


def simulated_annealing(a0, deadline, seed=0):
    """SA in log-space with block perturbations."""
    a = normalize_positive(a0, floor=1e-15)
    n = a.size
    N = next_pow2(2*n-1)
    rng = np.random.default_rng(seed)

    best_a = a.copy()
    best_score = internal_score(a, N)
    curr_a = a.copy()
    curr_score = best_score

    T = 0.4
    T_min = 0.0005
    block_sizes = [4, 8, 16, 32, 64, 128, 256, 512]
    block_sizes = [bs for bs in block_sizes if bs < n//2]
    if not block_sizes:
        block_sizes = [max(2, n//16)]
    it = 0
    stale = 0

    while time.monotonic() < deadline:
        bs = block_sizes[int(rng.integers(0, len(block_sizes)))]
        bs = max(2, min(bs, n//3))
        start = int(rng.integers(0, n - bs + 1))

        old_vals = curr_a[start:start+bs].copy()
        log_vals = np.log(old_vals + 1e-20)
        noise = T * rng.normal(1.0, 1.0, bs)
        new_log = log_vals + np.log(noise)
        new_vals = np.exp(new_log - new_log.max())
        new_vals = np.maximum(new_vals, 1e-15)
        new_vals = new_vals / new_vals.sum() * old_vals.sum()

        curr_a[start:start+bs] = new_vals
        curr_a = np.maximum(curr_a, 1e-15)
        curr_a = curr_a / curr_a.sum()

        c_try = convolve_fft(curr_a, N)
        new_score = float(2.0 * n * float(c_try.max()))
        delta = new_score - curr_score

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = curr_a.copy()
                stale = 0
            T = max(T_min, T * 0.9997)
        else:
            curr_a[start:start+bs] = old_vals
            curr_a = np.maximum(curr_a, 1e-15)
            curr_a = curr_a / curr_a.sum()
            stale += 1
            T = max(T_min, T * 0.9992)
            if stale % 300 == 0:
                T = min(0.6, T * 2.0)
                stale = 0

        it += 1
        if it % 2000 == 0:
            print(f"  SA n={n} it={it} T={T:.5f} score={curr_score:.9f} best={best_score:.9f}", flush=True)

    return best_a, best_score

                loga[int(rng.integers(0,n))]  # dummy
            # Restore single element
            stale += 1
            sigma = max(0.01, sigma * 0.97)
            if stale > 300:
                idx2 = int(rng.integers(0, n))
                loga[idx2] += 2.0 * sigma * rng.normal()
                a_try2 = np.exp(np.clip(loga - loga.max(), -60, 0))
                a_try2 = np.maximum(a_try2, 1e-15)
                a_try2 = a_try2 / a_try2.sum()
                sc2 = internal_score(a_try2, N)
                if np.isfinite(sc2) and sc2 < best_score:
                    best_a = a_try2.copy()
                    best_score = sc2
                    loga = np.log(a_try2 + 1e-20) - np.log(np.maximum(a_try2,1e-20)+1e-20).max()
                else:
                    loga[idx2] -= 2.0 * sigma * rng.normal()
                stale = 0

    return best_a, best_score


def binary_search(a0, deadline, seed=0):
    """Binary (0/1) sequence optimization via swap-based SA."""
    a = np.asarray(a0, dtype=np.float64).ravel()
    n = a.size
    m = max(2, min(n-2, int(round(0.5*n))))
    N = next_pow2(2*n-1)

    # Threshold to binary
    order = np.argsort(a, kind='stable')
    b = np.zeros(n, dtype=np.float64)
    b[order[-m:]] = 1.0

    c = np.rint(convolve_fft(b, N)).astype(np.float64)
    best_b = b.copy()
    best_max = float(c.max())
    best_score = float(2.0 * n * best_max / (m*m))

    rng = np.random.default_rng(seed)
    temp = 0.05
    stale = 0

    while time.monotonic() < deadline:
        cmax = float(c.max())
        score = float(2.0 * n * cmax / (m*m))
        if score < best_score * (1 - 1e-12):
            best_score = score
            best_max = cmax
            best_b = b.copy()

        one_idx = np.nonzero(b > 0.5)[0]
        zero_idx = np.nonzero(b < 0.5)[0]
        if one_idx.size == 0 or zero_idx.size == 0:
            break

        # Weight by convolution value for selection
        w = np.exp(np.clip(-c / (0.3 * cmax + 1e-15), -30, 0))
        w_one = w[one_idx]
        w_zero = w[zero_idx]
        ws = w_one.sum()
        if ws < 1e-15:
            ws = 1.0
        w_one /= ws
        wz = w_zero.sum()
        if wz < 1e-15:
            wz = 1.0
        w_zero /= wz

        si = int(rng.choice(len(one_idx), p=w_one))
        sj = int(rng.choice(len(zero_idx), p=w_zero))
        i = int(one_idx[si])
        j = int(zero_idx[sj])
        if i == j:
            continue

        # Swap
        c[i:i+n] -= 2.0*b
        c[2*i] -= 1.0
        c[j:j+n] += 2.0*b
        c[2*j] += 1.0
        b[i] = 0.0
        b[j] = 1.0

        new_max = float(c.max())
        new_score = float(2.0 * n * new_max / (m*m))
        delta = new_score - score

        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            stale = 0
            temp = max(0.002, temp * 0.998)
            if new_score < best_score:
                best_score = new_score
                best_max = new_max
                best_b = b.copy()
        else:
            # Revert
            c[j:j+n] -= 2.0*b
            c[2*j] -= 1.0
            c[i:i+n] += 2.0*b
            c[2*i] += 1.0
            b[j] = 0.0
            b[i] = 1.0
            stale += 1
            temp = max(0.001, temp * 0.96)
            if stale % 300 == 0:
                temp = min(0.1, temp * 2.5)
                stale = 0

    return normalize_positive(best_b, floor=0.0), best_score


def propose_candidate(seed=42, budget_s=2370, **kwargs):
    start = time.monotonic()
    deadline = start + max(120.0, float(budget_s) - 35.0)

    best_a = make_uniform(2048)
    best_score = 2.0
    best_n = 2048

    # Get given sequence
    given = get_given_sequence()
    if given is not None:
        sc = internal_score(given)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = given.copy()
            best_n = given.size
            print(f"Given sequence: n={best_n}, score={best_score:.9f}", flush=True)

    # Phase 1: Quick low-dim at various n (60s)
    phase1_deadline = min(deadline, start + 80.0)
    for n in (1024, 2048, 4096):
        if time.monotonic() > phase1_deadline:
            break
        a_n = resample_seq(best_a, n)
        a1, sc1 = lowdim_optimize(a_n, n, phase1_deadline, seed=seed+n)
        if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()
            best_n = n
            print(f"Phase1 lowdim n={n}: score={sc1:.9f}", flush=True)

    # Phase 2: Main mirror descent with momentum (800s)
    phase2_deadline = min(deadline, start + 900.0)
    print(f"\nPhase 2: Mirror descent with momentum, n={best_n}", flush=True)
    a2, sc2 = mirror_descent_momentum(
        best_a, phase2_deadline, seed=seed+100,
        momentum=0.75, use_lp=False
    )
    if a2 is not None and np.isfinite(sc2) and sc2 < best_score:
        best_score = sc2
        best_a = a2.copy()
        best_n = a2.size
        print(f"Phase2 MD: score={sc2:.9f}", flush=True)

    # Phase 3: LP-enhanced mirror descent (600s)
    if time.monotonic() < deadline - 200:
        phase3_deadline = min(deadline, start + 1550.0)
        print(f"\nPhase 3: LP-enhanced MD, n={best_n}", flush=True)
        # Use smaller n for LP feasibility
        n_lp = min(best_n, 2048)
        a3_init = resample_seq(best_a, n_lp)
        a3, sc3 = mirror_descent_momentum(
            a3_init, phase3_deadline, seed=seed+200,
            momentum=0.7, use_lp=True, lp_interval=60
        )
        if a3 is not None and np.isfinite(sc3) and sc3 < best_score:
            best_score = sc3
            best_a = a3.copy()
            best_n = a3.size
            print(f"Phase3 LP-MD: score={sc3:.9f}", flush=True)

    # Phase 4: Try different n with mirror descent
    if time.monotonic() < deadline - 150:
        phase4_deadline = min(deadline, start + 1900.0)
        for n_try in (1536, 3072, 4096, 6144, 8192):
            if time.monotonic() > phase4_deadline - 60:
                break
            if n_try == best_n:
                continue
            a4_init = resample_seq(best_a, n_try)
            allow = min(120.0, phase4_deadline - time.monotonic() - 50.0)
            if allow < 30:
                continue
            a4, sc4 = mirror_descent_momentum(
                a4_init, time.monotonic() + allow, seed=seed+300+n_try,
                momentum=0.8, use_lp=False
            )
            if a4 is not None and np.isfinite(sc4) and sc4 < best_score:
                best_score = sc4
                best_a = a4.copy()
                best_n = a4.size
                print(f"Phase4 n={n_try}: score={sc4:.9f}", flush=True)

    # Phase 5: SA escape + polish
    if time.monotonic() < deadline - 30:
        phase5_deadline = deadline
        # SA
        sa_allow = min(300.0, phase5_deadline - time.monotonic() - 30.0)
        if sa_allow > 30:
            print(f"\nPhase 5: SA, n={best_n}", flush=True)
            a5, sc5 = simulated_annealing(best_a, start + sa_allow + 0.01, seed=seed+400)
            # Fix: use absolute deadline
            if time.monotonic() < phase5_deadline - 30:
                sa_deadline = min(phase5_deadline, time.monotonic() + 280.0)
                a5, sc5 = simulated_annealing(best_a, sa_deadline, seed=seed+400)
            if a5 is not None and np.isfinite(sc5) and sc5 < best_score:
                best_score = sc5
                best_a = a5.copy()
                best_n = a5.size
                print(f"Phase5 SA: score={sc5:.9f}", flush=True)

    # Phase 6: Final polish
    if time.monotonic() < deadline - 10:
        polish_deadline = deadline - 5.0
        a6, sc6 = local_polish(best_a, polish_deadline, seed=seed+500)
        if a6 is not None and np.isfinite(sc6) and sc6 < best_score:
            best_score = sc6
            best_a = a6.copy()
            best_n = a6.size
            print(f"Phase6 polish: score={sc6:.9f}", flush=True)

    # Final
    best_a = normalize_positive(best_a, floor=0.0)
    out_list = [float(x) for x in best_a]
    try:
        final_eval = float(evaluate_sequence(out_list))
    except Exception:
        final_eval = float('inf')

    if not np.isfinite(final_eval):
        best_a = make_uniform(1024)
        out_list = [float(x) for x in best_a]
        final_eval = 2.0

    print(f"\nFinal: n={len(out_list)}, internal={best_score:.12f}, evaluated={final_eval:.12f}", flush=True)
    return out_list
