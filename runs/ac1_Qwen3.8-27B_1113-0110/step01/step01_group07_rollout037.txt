    """
    c: current convolution (length 2n-1)
    a: current sequence (length n), sum = S
    i: coordinate to update
    S: sum of a
    n: length of a
    Returns: (d_opt, new_max_c, new_S)

    k_start = i
    k_end = min(i + n, 2*n - 1)  # affected range: [k_start, k_end-1]
    
    # Unaffected max
    c_unaff = 0.0

        c_unaff = max(c_unaff, float(c[:k_start].max()))
    if k_end < 2*n - 1:

    # affected k: k_start to k_end-1, which is exactly n positions
    # a[k-i] for these k is a[0:n] = a (the whole array)

"""

import time
import numpy as np


def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def normalize_positive(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s


def fast_score(a):
    p = normalize_positive(a)
    if p is None:
        return np.inf
    n = p.size
    if n <= 0:
        return np.inf
    L = next_pow2(2 * n - 1)
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0.0:
        return np.inf
    return float(2.0 * n * cm)


def compute_conv(a):
    n = a.size
    L = next_pow2(2 * n - 1)
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    return np.where(np.isfinite(c), c, 0.0)


def softmax_stable(z):
    z = np.asarray(z, dtype=np.float64)
    z = np.where(np.isfinite(z), z, -50.0)
    z = z - z.max()
    z = np.clip(z, -60.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones_like(z) / z.size
    return e / s


def smooth_max_grad(c, tau):
    """Compute smooth max value and subgradient weights for convolution c."""
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0.0:
        return cm, np.ones(c.size) / c.size
    z = (c - cm) / tau
    z = np.clip(z, -50.0, 0.0)
    expz = np.exp(z)
    expsum = float(expz.sum())
    if not np.isfinite(expsum) or expsum <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = expz / expsum
    smooth_val = cm + tau * float(np.log(expsum))
    return smooth_val, w

    z = np.where(np.isfinite(z), z, -50.0)
    z -= z.mean()
    z = np.clip(z, -50.0, 50.0)

    v = np.zeros(n, dtype=np.float64)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    bias_step = 0

    current_tau_ratio = float(tau_ratio)
    lr = float(lr0)
    best_p = p.copy()
    best_score = fast_score(p)
    no_improve = 0
    step = 0

    while time.monotonic() < deadline:
        batch = 50
        for _ in range(batch):
            if time.monotonic() > deadline:
                break
            tau = max(1e-14, current_tau_ratio / float(n))
            P = np.fft.rfft(p, L)
            c = np.fft.irfft(P * P, L)[:2 * n - 1]
            c = np.where(np.isfinite(c), c, 0.0)
            c = np.maximum(c, 0.0)

            smooth_val, w = smooth_max_grad(c, tau)
            Wref = w[::-1]
            H = np.fft.irfft(P * np.fft.rfft(Wref, L), L)

            g_p = np.where(np.isfinite(g_p), g_p, 0.0)
            mean_g = float(np.dot(p, g_p))
            grad = g_p - p * mean_g
            max_g = float(np.max(np.abs(grad)))
            if max_g > 0.05:
                grad *= (0.05 / max_g)

            bias_step += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            b1 = 1.0 - beta1 ** bias_step
            b2 = 1.0 - beta2 ** bias_step
            z = z - lr * (m / b1) / (np.sqrt(v / b2) + eps)
            z = np.where(np.isfinite(z), z, 0.0)
            z -= z.mean()
            z = np.clip(z, -50.0, 50.0)
            p = softmax_stable(z)
            step += 1

        true_s = fast_score(p)
        if true_s < best_score - 1e-12:

            best_p = p.copy()
            no_improve = 0
        else:

            z = np.clip(zz, -50.0, 50.0)

            v *= 0.3

def coordinate_descent(c, a, n, deadline, rng, max_passes=500):
    """
    Coordinate descent to minimize 2n*max(c)/(sum(a))^2.
    c: convolution of a with a (length 2n-1)
    a: sequence (length n)
    Mutates c and a in place. Returns best_score found.
    """
    S = float(a.sum())
    best_score = 2.0 * n * float(c.max()) / (S * S)
    best_a = a.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    order = np.arange(n)
    pass_count = 0
    no_improve_passes = 0

    while time.monotonic() < deadline and pass_count < max_passes:
        pass_count += 1
        rng.shuffle(order)
        pass_improved = False

        for i in order:
            if time.monotonic() > deadline:
                break

            ai = a[i]
            k_start = i
            k_end = min(i + n, 2 * n - 1)

            # Unaffected max
            c_unaff = 0.0
            if k_start > 0:
                v0 = float(c[:k_start].max())
                if v0 > c_unaff:
                    c_unaff = v0
            if k_end < 2 * n - 1:
                v1 = float(c[k_end:].max())
                if v1 > c_unaff:
                    c_unaff = v1

            c_affected = c[k_start:k_end]
            # a[k-i] for k in [k_start, k_end) is a[0:n] = a

            # Golden section search for d in [-ai, d_high]
            d_low = -ai

            def obj_fn(d):
                vals = c_affected + 2.0 * d * a
                idx_2i = 2 * i - k_start
                if 0 <= idx_2i < n:
                    vals = vals.copy()
                    vals[idx_2i] += d * d
                m_aff = float(vals.max())
                m = max(c_unaff, m_aff)
                new_S = S + d
                if new_S <= 1e-30:
                    return 1e30
                return 2.0 * n * m / (new_S * new_S)

            # Find bracket
            d_high = ai + 1e-8
            f_low = obj_fn(d_low)
            f_high = obj_fn(d_high)

            # Expand if needed
            expand_count = 0
            while f_high < f_low and expand_count < 10:
                d_low = d_high
                d_high += (d_high - (-ai)) * 0.5 + 1e-6
                d_high = min(d_high, 10.0 * S)
                f_low = f_high
                f_high = obj_fn(d_high)
                expand_count += 1

            if f_high >= f_low:
                d_opt = d_low
                f_opt = f_low
            else:
                # Golden section in [d_low, d_high]
                c1 = d_high - gr * (d_high - d_low)
                c2 = d_low + gr * (d_high - d_low)
                f1 = obj_fn(c1)
                f2 = obj_fn(c2)
                for _ in range(50):
                    if f1 < f2:
                        d_high = c2
                        c2 = c1
                        f2 = f1
                        c1 = d_high - gr * (d_high - d_low)
                        f1 = obj_fn(c1)
                    else:
                        d_low = c1
                        c1 = c2
                        f1 = f2
                        c2 = d_low + gr * (d_high - d_low)
                        f2 = obj_fn(c2)
                d_opt = (d_low + d_high) / 2.0
                f_opt = obj_fn(d_opt)

            if f_opt < best_score - 1e-12:
                # Apply update
                c[k_start:k_end] += 2.0 * d_opt * a
                idx_2i = 2 * i - k_start
                if 0 <= idx_2i < n:
                    c[2 * i] += d_opt * d_opt
                a[i] += d_opt
                S += d_opt
                best_score = f_opt
                best_a = a.copy()
                pass_improved = True

        if pass_improved:
            no_improve_passes = 0
        else:
            no_improve_passes += 1
            if no_improve_passes >= 8:
                break
            if no_improve_passes >= 4:
                # Restart from best
                c[:] = compute_conv(best_a)
                a[:] = best_a
                S = float(a.sum())
                no_improve_passes = 0

        if pass_count % 10 == 0:
            print(f"  CD pass {pass_count} best {best_score:.6f}")

    return best_score, best_a


def resize_sequence(seq, m):
    p = normalize_positive(seq)
    m = int(m)
    if p is None:
        return np.ones(m, dtype=np.float64) / float(m)
    n = p.size
    if n == m:
        return p.copy()
    if n == 1:
        return np.full(m, 1.0 / float(m), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, m)
    out = np.interp(x_new, x_old, p)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(m, dtype=np.float64) / float(m)
    return out / s

        if rem > 30.0:
            dur = max(20.0, rem * 0.6)
            p1, s1 = adam_fine_tune(p, dur, rng, tau_ratio=0.002, lr0=0.0012)
            if p1 is not None and s1 < cur_s:
                p, cur_s = p1, s1
                print(f"  After Adam: {cur_s:.6f}")

        # Phase 2: Coordinate descent
        rem = N_deadline - time.monotonic()
        if rem > 20.0:
            c = compute_conv(p)
            a_work = p.copy()
            s_cd, a_best = coordinate_descent(c, a_work, N, time.monotonic() + rem, rng, max_passes=300)
            if s_cd < cur_s:
                p, cur_s = a_best, s_cd
                print(f"  After CD: {cur_s:.6f}")

        # Compare with global best
        if best_p is None or cur_s < best_s:
            best_p = p.copy()
            best_s = cur_s
            best_N = N
            print(f"  New global best (N={N}): {cur_s:.6f}")

    # Final verification with evaluate_sequence
    candidates = []
    if best_p is not None:
        candidates.append(best_p)
    # Also try a few other lengths
    for N_test in [4096, 2048]:
        if best_p is not None:
            candidates.append(resize_sequence(best_p, N_test))
    candidates.append(np.ones(1024, dtype=np.float64) / 1024.0)

    final_seq = None
    final_score = np.inf

    for cand in candidates:
        lst = [float(x) for x in np.clip(np.asarray(cand, dtype=np.float64), 0.0, 1000.0)]
        if not lst:
            continue
        try:
            sc = evaluate_sequence(lst)
        except Exception:
            sc = np.inf
        if np.isfinite(sc) and sc < final_score:
            final_score = float(sc)
            final_seq = lst
            print(f"  Candidate len={len(lst)} score={sc:.6f}")

    if final_seq is None:
        final_seq = [1.0 / 1024.0] * 1024

    print(f"Final score: {final_score:.6f}")
    return final_seq
