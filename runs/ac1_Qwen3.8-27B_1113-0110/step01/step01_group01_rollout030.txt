2. By scaling invariance, set s = 1. Then minimize 2n * max(b).
3. max(b) is minimized when b is as flat as possible.
4. The constraint is a_i >= 0, sum(a) = 1.

- Parameterize: a = exp(z) / sum(exp(z)) (softmax)
- Objective: 2n * tau * log(sum exp(b_k / tau)) (smooth max)
- Gradient: computed via FFT
- Nesterov momentum for acceleration
- Temperature tau decreases over time

"""
Multi-stage optimization for minimizing 2*n*max(convolve(a,a))/(sum(a))^2
over non-negative sequences. Uses Nesterov-accelerated gradient descent on a
log-sum-exp smoothed objective with temperature annealing, combined with
low-dimensional basis optimization and diverse restarts.

import time
import numpy as np

try:
    import scipy.optimize as _scipy_opt
except Exception:
    _scipy_opt = None


def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def safe_softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = z - float(np.max(z))
    z = np.clip(z, -50.0, 0.0)
    e = np.exp(z)
    e = np.maximum(e, 1e-300)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(z.shape, 1.0 / float(z.size), dtype=np.float64)
    return e / s


def normalize_pos(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1)
    a = np.clip(a, float(floor), None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size) / float(a.size)
    return a / s


def convolve_fft(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if N is None:
        N = next_pow2(2 * n - 1)
    else:
        need = next_pow2(2 * n - 1)
        if int(N) < int(need):
            N = need
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]


def score_from_conv(b, n):
    if b.size == 0:
        return np.inf
    return float(np.max(b))


def internal_score(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.inf
    a = a / s
    b = convolve_fft(a, N)
    return float(2.0 * n * np.max(b))


def build_basis(n, n_modes=12):
    n = max(8, int(n))
    i = np.arange(n, dtype=np.float64)
    x = (i - 0.5 * (n - 1)) / max(1.0, 0.5 * n)
    cols = []
    # Sine modes (antisymmetric)
    for m in range(1, min(8, n_modes)):
        cols.append(np.sin(np.pi * float(m) * x))
    # Polynomial modes
    cols.append(x.copy())
    cols.append(x**2 - np.mean(x**2))
    cols.append(x**3 - np.mean(x**3))
    # Periodic modes
    for m in range(1, 4):
        t = 2.0 * np.pi * float(m) * i / float(n)
        cols.append(np.sin(t) - np.mean(np.sin(t)))
        cols.append(np.cos(t) - np.mean(np.cos(t)))
    B = np.column_stack(cols)
    B -= B.mean(axis=0)
    norms = np.linalg.norm(B, axis=0)
    norms[norms <= 1e-14] = 1.0
    B = B / norms
    return B


def smooth_max_val(b, cmax, tau):
    if b.size == 0 or cmax <= 0 or not np.isfinite(cmax):
        return 0.0
    if tau <= 0:
        return cmax
    z = (b - cmax) / tau
    z = np.clip(z, -50.0, 0.0)
    s = float(np.exp(z).sum())
    if not np.isfinite(s) or s <= 0:
        return cmax
    return float(cmax + tau * np.log(s))


def theta_objective(theta, n, B, tau, N):
    a = safe_softmax(B @ theta)
    fa = np.fft.rfft(a, N)
    b = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    cmax = float(np.max(b))
    if cmax <= 0 or not np.isfinite(cmax):
        return 0.0
    return float(2.0 * n * smooth_max_val(b, cmax, tau))


def theta_gradient(theta, n, B, tau, N):
    theta = np.asarray(theta, dtype=np.float64).ravel()
    a = safe_softmax(B @ theta)
    fa = np.fft.rfft(a, N)
    b = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    cmax = float(np.max(b))
    D = B.shape[1]
    if cmax <= 0 or not np.isfinite(cmax):
        return np.zeros(D)
    tau_eff = max(float(tau), 1e-12)
    z = (b - cmax) / tau_eff
    z = np.clip(z, -50.0, 0.0)
    ez = np.exp(z)
    s = float(ez.sum())
    if not np.isfinite(s) or s <= 0:
        return np.zeros(D)
    w = ez / s
    # Gradient of smooth_max w.r.t. a
    w_rev = np.zeros(2 * n - 1)
    w_rev[:] = w[::-1]
    fw = np.fft.rfft(w_rev, N)
    conv = np.fft.irfft(fa * fw, N)
    g_a = 2.0 * conv[n - 1:2 * n - 1][::-1]
    total = float(np.dot(g_a, a))
    aB = a @ B
    agaB = (a * g_a) @ B
    gtheta = agaB - total * aB
    return (2.0 * n) * gtheta

            w = (1 - w_floor) * w + w_floor / float(b.size)

        # Gradient
        w_rev = np.zeros(2 * n - 1)
        w_rev[:] = w[::-1]
        fw = np.fft.rfft(w_rev, N)
        conv = np.fft.irfft(fa * fw, N)
        g_a = 2.0 * conv[n - 1:2 * n - 1][::-1]
        g_a = g_a - g_a.mean()
        scale = float(np.abs(g_a).max())
        if not np.isfinite(scale) or scale <= 1e-15:
            break
        g_a /= scale

        # Try multiple step sizes
        accepted = False
        for e_try in (eta, 0.5 * eta, 0.25 * eta, 0.1 * eta):
            if e_try <= 1e-10:
                continue
            loga_try = loga - e_try * g_a
            loga_try = loga_try - float(np.max(loga_try))
            a_try = np.exp(np.clip(loga_try, -60.0, 0.0))
            a_try = np.maximum(a_try, 1e-300)
            a_try = a_try / float(a_try.sum())
            a_try = normalize_pos(a_try, floor=1e-15)
            score_try = internal_score(a_try, N)
            if np.isfinite(score_try) and score_try < score * (1 - 1e-10):
                accepted = True
                a = a_try
                loga = np.log(np.maximum(a, 1e-20))
                loga -= float(np.max(loga))
                score = score_try
                break

        if accepted:
            stagnant = 0
            eta = min(1.5, eta * 1.02)
            tau_frac = max(0.005, tau_frac * 0.998)
            w_floor = max(0.0, w_floor * 0.99)
        else:
            stagnant += 1
            eta = max(1e-5, eta * 0.65)
            tau_frac = max(0.003, tau_frac * 0.85)
            if stagnant % 100 == 0:
                tau_frac = min(0.3, tau_frac * 2.0)
                w_floor = min(0.1, w_floor + 0.01)
                # Random kick
                loga += rng.normal(0, 0.05, n)
                loga -= float(np.max(loga))
                stagnant = 0
            if eta < 1e-6 or stagnant > 1000:
                break

        it += 1
        if it % 2000 == 0 and verbose := True:
            print("    mirror2 n={} it={} score={:.9f}".format(n, it, best_score), flush=True)

    return best_a, best_score


def local_polish_v2(a_init, deadline, seed=0):
    """Fine-grained local optimization with block perturbations."""
    rng = np.random.default_rng(int(seed))
    n = a_init.size
    N = next_pow2(3 * n - 2)
    a = normalize_pos(a_init, floor=1e-15)
    best_a = a.copy()
    best_score = internal_score(a, N)
    loga = np.log(np.maximum(a, 1e-20))
    loga -= float(np.max(loga))
    sigma = 0.12
    stale = 0

    while time.monotonic() < deadline:
        # Choose block width
        w_max = max(3, n // 30)
        w = int(rng.integers(2, w_max + 1))
        start = int(rng.integers(0, n - w + 1))

        # Perturbation
        t = np.linspace(0, 1, w)
        env = np.exp(-3.0 * (t - 0.5)**2)
        d = sigma * env * rng.normal(size=w)
        d -= d.mean()
        d = np.clip(d, -2.5, 2.5)

        old = loga[start:start + w].copy()
        loga[start:start + w] += d
        a_try = np.exp(np.clip(loga - float(np.max(loga)), -60.0, 0.0))
        a_try = np.maximum(a_try, 1e-300)
        a_try = a_try / float(a_try.sum())

        sc_try = internal_score(a_try, N)
        if np.isfinite(sc_try) and sc_try < best_score * (1 - 1e-10):
            best_a = a_try.copy()
            best_score = sc_try
            a = a_try
            loga = np.log(np.maximum(a, 1e-20))
            loga -= float(np.max(loga))
            stale = 0
            sigma = min(0.3, sigma * 1.03)
        else:
            loga[start:start + w] = old
            stale += 1
            sigma = max(0.02, sigma * 0.97)
            if stale > 300:
                # Larger kick
                idx = rng.integers(0, n, size=min(10, n))
                loga[idx] += rng.normal(0, 1.0, size=idx.size)
                a_try = np.exp(np.clip(loga - float(np.max(loga)), -60.0, 0.0))
                a_try = np.maximum(a_try, 1e-300)
                a_try /= float(a_try.sum())
                sc_try = internal_score(a_try, N)
                if np.isfinite(sc_try) and sc_try < best_score:
                    best_a = a_try.copy()
                    best_score = sc_try
                    a = a_try
                    loga = np.log(np.maximum(a, 1e-20))
                    loga -= float(np.max(loga))
                else:
                    loga[idx] -= rng.normal(0, 1.0, size=idx.size)
                stale = 0
                sigma = 0.15

    return best_a, best_score


def resample(a, n_target):
    a = normalize_pos(a, floor=0.0)
    old_n = a.size
    n_target = max(2, int(n_target))
    if old_n == n_target:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, n_target)
    return normalize_pos(np.interp(new_x, old_x, a), floor=1e-12)


def get_parent():
    try:
        seq = globals().get("height_sequence_1", None)
        if seq is None:
            return None
        a = np.asarray(seq, dtype=np.float64).ravel()
        if a.size == 0:
            return None
        a = normalize_pos(a, floor=0.0)
        if not np.all(np.isfinite(a)):
            return None
        return a
    except Exception:
        return None


def make_initial(n, seed):
    rng = np.random.default_rng(int(seed))
    candidates = [np.ones(n) / n]
    # Sine-based
    x = (np.arange(n, dtype=np.float64) - 0.5 * (n - 1)) / max(1.0, 0.5 * n)
    for alpha in (0.3, 0.5, 0.7, 0.9, -0.3, -0.5, -0.7):
        v = 1.0 + alpha * np.sin(np.pi * x)
        candidates.append(normalize_pos(np.abs(v), floor=1e-12))
    # Two-frequency
    for m1, m2, a1, a2 in [(2,5,0.3,0.2),(3,7,0.3,0.25),(2,7,0.4,0.15),(3,4,0.5,0.3)]:
        v = 1.0 + a1*np.sin(np.pi*m1*x) + a2*np.cos(np.pi*m2*x)
        candidates.append(normalize_pos(np.abs(v), floor=1e-12))
    # Random
    for _ in range(4):
        v = 1.0 + rng.uniform(0.1, 0.5) * np.sin(2*np.pi*rng.uniform(1,6)*x + rng.uniform(0, 2*np.pi))
        v += rng.uniform(0.05, 0.3) * np.cos(2*np.pi*rng.uniform(1,8)*x + rng.uniform(0, 2*np.pi))
        candidates.append(normalize_pos(np.abs(v), floor=1e-12))
    best = min(candidates, key=lambda a: internal_score(a, next_pow2(2*a.size-1)))
    return best


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    start = time.monotonic()
    total_time = max(60.0, float(budget_s) - 30.0)
    deadline = start + total_time
    print("Total time budget: {:.1f}s".format(total_time), flush=True)

    # Get starting point
    best_a = None
    best_score = np.inf

    parent = get_parent()
    if parent is not None:
        sc = internal_score(parent)
        best_score = sc
        best_a = parent.copy()
        print("Parent score: {:.9f}, n={}".format(best_score, parent.size), flush=True)

    # Try various initializations
    for n_try in (1024, 2048, 4096):
        if time.monotonic() > deadline - 120:
            break
        a_init = make_initial(n_try, seed + n_try)
        sc = internal_score(a_init)
        if sc < best_score:
            best_score = sc
            best_a = a_init.copy()
            print("Init n={} score={:.9f}".format(n_try, best_score), flush=True)

    if best_a is None:
        best_a = np.ones(2048) / 2048
        best_score = 2.0

    # Phase 1: Low-dim optimization to get into good basin
    print("Phase 1: Low-dim optimization", flush=True)
    for n in (2048, 4096):
        if time.monotonic() > deadline - 1500:
            break
        alloc = min(60.0, deadline - time.monotonic() - 1500)
        if alloc < 10:
            break
        a0 = resample(best_a, n)
        a1, sc1 = optimize_lowdim(n, time.monotonic() + alloc, seed + n, initial_a=a0, max_starts=4)
        if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()
            print("  Lowdim n={} score={:.9f}".format(n, best_score), flush=True)

    # Phase 2: Nesterov optimization (main workhorse)
    print("Phase 2: Nesterov optimization", flush=True)
    for n in (2048, 4096, 8192):
        if time.monotonic() > deadline - 400:
            break
        alloc = min(400.0, deadline - time.monotonic() - 400)
        if alloc < 60:
            break
        a0 = resample(best_a, n)
        a1, sc1 = nesterov_optimize(a0, n, time.monotonic() + alloc, seed=seed + n, n_restarts=6)
        if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()
            print("  Nesterov n={} score={:.9f}".format(n, best_score), flush=True)

    # Phase 3: Improved mirror descent
    print("Phase 3: Mirror descent v2", flush=True)
    for n in (2048, 4096):
        if time.monotonic() > deadline - 200:
            break
        alloc = min(180.0, deadline - time.monotonic() - 200)
        if alloc < 30:
            break
        a0 = resample(best_a, n)
        a1, sc1 = mirror_descent_v2(a0, n, time.monotonic() + alloc, seed=seed + n)
        if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()
            print("  Mirror2 n={} score={:.9f}".format(n, best_score), flush=True)

    # Phase 4: Final polish
    print("Phase 4: Local polish", flush=True)
    if time.monotonic() < deadline - 10:
        a1, sc1 = local_polish_v2(best_a, deadline - 5, seed=9999)
        if a1 is not None and np.isfinite(sc1) and sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()
            print("  Polish score={:.9f}".format(best_score), flush=True)

    # Final validation
    best_a = normalize_pos(best_a, floor=0.0)
    out_list = [float(x) for x in best_a]
    try:
        final_eval = float(evaluate_sequence(out_list))
    except Exception:
        final_eval = float("inf")

    if not np.isfinite(final_eval):
        best_a = np.ones(2048) / 2048
        out_list = [float(x) for x in best_a]
        final_eval = 2.0

    elapsed = time.monotonic() - start
    print("FINAL: score={:.12f}, n={}, time={:.1f}s".format(final_eval, len(out_list), elapsed), flush=True)
    return out_list
