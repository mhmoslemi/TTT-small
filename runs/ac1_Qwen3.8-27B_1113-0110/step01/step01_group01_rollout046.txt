"""
Minimize 2*n*max(convolve(a,a))/(sum(a))^2 using symmetric smooth optimization,
KKT-inspired multiplicative weight descent, and coordinate descent polishing.
"""

import time
import numpy as np
from scipy.optimize import minimize as scipy_minimize


def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def convolve_full(a, N=None):
    n = len(a)
    if N is None:
        N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]


def score_unnormalized(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2 without normalizing a."""
    n = len(a)
    s = a.sum()
    if s <= 0:
        return np.inf
    c = convolve_full(a)
    return 2.0 * n * c.max() / (s * s)


def logsumexp_val(c, cmax, tau):
    """Smooth approximation of max(c) using log-sum-exp."""
    if tau <= 0:
        return cmax
    z = (c - cmax) / tau
    z = np.clip(z, -50, 0)
    ez = np.exp(z)
    return cmax + tau * np.log(ez.sum())


def symmetric_a(n, half):
    """Create a symmetric sequence from its first half."""
    a = np.zeros(n)
    h = len(half)
    a[:h] = half
    a[h:] = half[:n-h][::-1]
    return a


def logsumexp_obj(params, n, half_len, N):
    """Objective for optimization: smooth max of self-convolution."""
    # params: [log_values..., tau]
    log_vals = params[:half_len]
    a = symmetric_a(n, np.exp(log_vals - log_vals.max()))
    a = a / a.sum()
    c = convolve_full(a, N)
    cmax = c.max()
    tau = params[half_len]
    smooth_max = logsumexp_val(c, cmax, tau)

def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def convolve_fft(a, N=None):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if N is None:
        N = next_pow2(2 * n - 1)
    else:
        need = next_pow2(2 * n - 1)
        if N < need:
            N = need
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]


def score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s <= 1e-15:
        return np.inf
    c = convolve_fft(a)
    return 2.0 * n * float(c.max()) / (s * s)


def normalize(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, floor)
    s = a.sum()
    if s <= 1e-15:
        return np.ones_like(a) / len(a)
    return a / s * len(a)


def smooth_max(c, cmax, tau):
    if tau <= 0:
        return cmax
    z = (c - cmax) / tau
    z = np.clip(z, -50, 0)
    ez = np.exp(z)
    return cmax + tau * float(np.log(ez.sum()))

def build_basis(n, n_basis=20):
    """Build basis functions for low-dim optimization."""
    x = np.linspace(-1, 1, n)
    cols = []
    # Cosine bumps with different powers
    for p in [1, 1.5, 2, 2.5, 3, 4]:
        col = np.cos(np.pi * x / 2) ** p
        col = np.maximum(col, 0)
        col -= col.mean()
        cols.append(col)
    # Gaussian bumps
    for alpha in [1, 2, 4, 8]:
        col = np.exp(-alpha * x * x) - np.exp(-alpha * x * x).mean()
        cols.append(col)
    # Sine arches
    for p in [1, 2, 3]:
        col = np.sin(np.pi * (x + 1) / 2) ** p
        col = np.maximum(col, 0)
        col -= col.mean()
        cols.append(col)
    # Polynomial
    col = (1 - x * x) ** 2 - ((1 - x * x) ** 2).mean()
    cols.append(col)
    col = (1 - x * x) ** 3 - ((1 - x * x) ** 3).mean()
    cols.append(col)
    # Mixed modes
    col = np.cos(np.pi * x / 2) ** 2 * np.exp(-2 * x * x)
    col -= col.mean()
    cols.append(col)
    col = np.cos(np.pi * x / 2) * np.cos(np.pi * x)
    col -= col.mean()
    cols.append(col)

    B = np.column_stack(cols[:n_basis])
    norms = np.linalg.norm(B, axis=0)
    norms[norms < 1e-14] = 1.0
    B = B / norms
    return B

        # I'll just trust the previous code's formula since it worked.
        
        w_rev = w[::-1]
        fwr = np.fft.rfft(w_rev, N)
        conv_ca = np.fft.irfft(fa * fwr, N)
        g = 2.0 * conv_ca[n - 1:2 * n - 1][::-1]
        
        # Center the gradient
        g = g - g.mean()
        gmax = np.abs(g).max()
        if gmax < 1e-15:
            break
        g = g / gmax

        # Multiplicative update: a[i] *= exp(-eta * g[i])
        log_a = np.log(a + 1e-20)
        log_a_new = log_a - eta * g
        log_a_new = log_a_new - log_a_new.max()
        a_new = np.exp(np.clip(log_a_new, -60, 0))
        a_new = np.maximum(a_new, 1e-20)
        a_new = a_new / a_new.sum() * n

        sc_new = 2.0 * n * float(convolve_fft(a_new, N).max()) / (n * n)

        if sc_new < sc * (1 - 1e-10):
            a = a_new
            stagnation = 0
            eta = min(1.5, eta * 1.02)
            tau_ratio = max(0.02, tau_ratio * 0.998)
        else:
            stagnation += 1
            eta = max(0.01, eta * 0.7)
            tau_ratio = max(0.01, tau_ratio * 0.85)
            if stagnation % 200 == 0:
                # Escape: perturb a
                rng_esc = np.random.default_rng(it * 7919 + n)
                a = a * np.exp(rng_esc.normal(0, 0.05, n))
                a = a / a.sum() * n
                tau_ratio = min(0.4, tau_ratio * 2.0)
                eta = 0.3
                stagnation = 0
            if stagnation > 500:
                break

        it += 1

    return best_a, best_sc


def coordinate_descent(a, deadline, seed=0):
    """Local coordinate search: try small perturbations to individual elements."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum() * n
    N = next_pow2(2 * n - 1)

    best_a = a.copy()
    best_sc = 2.0 * n * float(convolve_fft(a, N).max()) / (n * n)
    rng = np.random.default_rng(seed)
    
    # Precompute c for fast updates
    c = convolve_fft(a, N)
    
    it = 0
    while time.monotonic() < deadline:
        # Random coordinate
        i = rng.integers(0, n)
        # Try a few step sizes
        for frac in [0.1, 0.05, 0.02, -0.1, -0.05, -0.02]:
            delta = frac * a.mean()
            if a[i] + delta < 0:
                continue
            # Compute change in c
            # c[k] changes by: 2*delta*a[k-i] - delta^2 (at k=2i)
            # More precisely: new_a[i] = a[i] + delta
            # dc[k] = 2*delta*a[k-i] + delta^2 (if k-i == i, i.e., k=2i)
            # Actually: dc[k] = delta*(a[k-i] + a[k-i]) = 2*delta*a[k-i] for k != 2i
            # For k = 2i: dc[k] = 2*delta*a[i] + delta^2
            # (because the term a[i]*a[i] becomes (a[i]+delta)^2 = a[i]^2 + 2*delta*a[i] + delta^2)
            
            # For efficiency, just compute the new max approximately
            # c_new[k] = c[k] + 2*delta*a[max(0,min(n-1,k-i))] ... this isn't quite right
            # Let me just compute the full c for this simple case
            a[i] += delta
            c_new = convolve_fft(a, N)
            sc_new = 2.0 * n * float(c_new.max()) / (n * n)
            if sc_new < best_sc * (1 - 1e-10):
                best_sc = sc_new
                best_a = a.copy()
                c = c_new
            else:
                a[i] -= delta
                break
        it += 1
        if it % 500 == 0:
            print("  CD it={} score={:.10f}".format(it, best_sc), flush=True)

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    start_time = time.monotonic()
    total_budget = max(60.0, float(budget_s) - 30.0)
    deadline = start_time + total_budget

    # Try to use the given sequence as a starting point
    best_a = None
    best_sc = np.inf
    
    try:
        given = globals().get("height_sequence_1", None)
        if given is not None:
            ag = np.asarray(given, dtype=np.float64)
            if len(ag) > 0 and np.all(np.isfinite(ag)):
                ag = np.maximum(ag, 0)
                ag = ag / ag.sum() * len(ag)
                sc_g = score(ag)
                if np.isfinite(sc_g):
                    best_a = ag
                    best_sc = sc_g
                    print("Given sequence: n={} score={:.10f}".format(len(ag), best_sc), flush=True)
    except Exception:
        pass

    if best_a is None:
        best_a = make_bump(2048, "cos2")
        best_sc = score(best_a)
        print("Fallback initial: score={:.10f}".format(best_sc), flush=True)

    # Phase 1: Try diverse initializations at various sizes
    init_budget = min(60.0, total_budget * 0.05)
    for n in [512, 1024, 2048, 4096, 8192]:
        if time.monotonic() > start_time + init_budget:
            break
        for typ in ["cos2", "cos3", "gauss", "sin2", "mix1", "poly2"]:
            a0 = make_bump(n, typ)
            sc0 = score(a0)
            if sc0 < best_sc:
                best_sc = sc0
                best_a = a0
                print("Init n={} typ={} score={:.10f}".format(n, typ, best_sc), flush=True)

    # Phase 2: Low-dimensional optimization on best sizes
    lowdim_budget = min(120.0, total_budget * 0.1)
    for n in [1024, 2048, 4096]:
        if time.monotonic() > start_time + init_budget + lowdim_budget:
            break
        a0 = best_a
        if len(a0) != n:
            # Resample
            x_old = np.linspace(0, 1, len(a0))
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, a0)
            a0 = a0 / a0.sum() * n
        a1, sc1 = lowdim_optimize(n, min(deadline, time.monotonic() + 35), seed=seed + n, init_a=a0)
        if sc1 < best_sc:
            best_sc = sc1
            best_a = a1
            print("Lowdim selected: n={} score={:.10f}".format(n, best_sc), flush=True)

    # Phase 3: Main multiplicative weight descent (multiple restarts)
    # This is where most of the time should go
    remaining = deadline - time.monotonic()
    mw_budget = max(60.0, remaining * 0.7)
    mw_deadline = time.monotonic() + mw_budget

    # Sort sizes by priority: try the size of best_a first, then neighbors
    sizes_to_try = sorted(set([len(best_a), 1024, 2048, 4096, 8192]),
                          key=lambda x: abs(x - len(best_a)))
    
    for n in sizes_to_try:
        if time.monotonic() > mw_deadline:
            break
        # Resample best_a to this size
        a0 = best_a
        if len(a0) != n:
            x_old = np.linspace(0, 1, len(a0))
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, a0)
            a0 = a0 / a0.sum() * n
        else:
            a0 = a0.copy()
        
        # Multiple restarts for this size
        n_restarts = 3 if n <= 2048 else 2
        per_restart_time = max(30.0, (mw_deadline - time.monotonic()) / max(1, n_restarts) / 1.2)
        
        for r in range(n_restarts):
            if time.monotonic() > mw_deadline - 30:
                break
            if r == 0:
                a_start = a0
            else:
                # Perturb for restart
                rng_r = np.random.default_rng(seed + n * 31 + r)
                a_start = a0 * np.exp(rng_r.normal(0, 0.1, n))
                a_start = a_start / a_start.sum() * n
            
            a1, sc1 = multiplicative_descent(a_start, min(mw_deadline, time.monotonic() + per_restart_time),
                                            seed=seed + 1000 * n + r, verbose=True)
            if sc1 < best_sc:
                best_sc = sc1
                best_a = a1
                print("MW selected: n={} restart={} score={:.10f}".format(n, r, best_sc), flush=True)
            # Update a0 to the best found so far for next restart
            if r == 0 or (sc1 < score(a0)):
                a0 = a1 if sc1 < score(a0) else a0

    # Phase 4: Coordinate descent polishing
    remaining = deadline - time.monotonic()
    cd_budget = max(30.0, min(120.0, remaining * 0.4))
    if time.monotonic() < deadline - 20:
        a1, sc1 = coordinate_descent(best_a, min(deadline, time.monotonic() + cd_budget), seed=seed + 99999)
        if sc1 < best_sc:
            best_sc = sc1
            best_a = a1
            print("CD selected: score={:.10f}".format(best_sc), flush=True)

    # Phase 5: Final brief MW pass
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a1, sc1 = multiplicative_descent(best_a, deadline, seed=seed + 77777, verbose=True)
        if sc1 < best_sc:
            best_sc = sc1
            best_a = a1
            print("Final MW: score={:.10f}".format(best_sc), flush=True)

    # Ensure non-negative and reasonable
    best_a = np.maximum(best_a, 0.0)
    best_a = np.minimum(best_a, 1000.0)
    s = best_a.sum()
    if s < 0.01:
        best_a = make_bump(1024, "cos2")
    
    out = [float(x) for x in best_a]
    try:
        final_sc = float(evaluate_sequence(out))
    except Exception:
        final_sc = np.inf
    
    print("FINAL: n={} internal={:.12f} evaluated={:.12f}".format(
        len(out), best_sc, final_sc), flush=True)
    return out
