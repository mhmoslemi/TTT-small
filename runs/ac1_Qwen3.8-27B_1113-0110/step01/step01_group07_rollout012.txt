def next_pow2(x):
    x = int(x)

def normalize_positive(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:

    return a / s


def fast_score(a):

    if p is None:

    n = p.size

    L = next_pow2(2 * n - 1)
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0.0:

    return float(2.0 * n * cm)


def softmax_from_logits(z):

    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:

        s = float(e.size)

    # The auto-convolution in frequency: C[m] = P[m]^2

"""

def compute_grad_logit(p, tau, L):
    """Compute gradient of 2*n*smooth_max(conv) w.r.t. logit parameters."""
    n = p.size
    P = np.fft.rfft(p, L)
    c = np.fft.irfft(P * P, L)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    cm = float(np.max(c))
    if not np.isfinite(cm) or cm <= 0.0:
        return np.inf, np.zeros(n, dtype=np.float64)

    # Softmax weights over convolution values
    z = (c - cm) / tau
    z = np.clip(z, -50.0, 0.0)
    expz = np.exp(z)
    expsum = float(expz.sum())
    if not np.isfinite(expsum) or expsum <= 0.0:
        k_idx = int(np.argmax(c))
        expz = np.zeros_like(c)
        expz[k_idx] = 1.0
        expsum = 1.0

    W = tau * expz / expsum

    # Gradient of smooth_max w.r.t. p:
    # d(smooth_max)/dp_j = sum_k W[k] * d(c_k)/dp_j
    # For linear conv (via circular conv with L >= 2n-1):
    # d(c_k)/dp_j = p_circ[(k-j) mod L] + p_circ[(j-k) mod L]  (symmetric)
    # = 2 * p_circ[(k-j) mod L]  (since p is real, the circ conv is symmetric)
    # Actually for linear conv: c_k = sum_i p[i]*p[k-i]
    # d(c_k)/dp_j = p[k-j] + p[k-j] if k != j, else p[k-j] + 2*p[0]... 
    # No: d(c_k)/dp_j = p[k-j] (from first factor) + p[k-j] (from second factor, index k-j)
    # Wait: c_k = sum_i p[i]*p[k-i]. d/dp_j: term i=j gives p[k-j], term k-i=j (i=k-j) gives p[k-j].
    # If j != k-j (i.e., 2j != k): both terms present, d = 2*p[k-j]
    # If 2j == k: only one term, d = p[k-j] + p[j] = 2*p[j] (since k-j=j)
    # Actually in both cases it's p[k-j] + p[k-j] = 2*p[k-j] (with the convention that p[x]=0 for x not in [0,n-1])
    # So d(c_k)/dp_j = 2 * p[(k-j)] where p is zero outside [0,n-1]
    
    # In circular form with padding: p_circ[m] = p[m] if 0<=m<n else 0
    # d(c_k)/dp_j = 2 * p_circ[(k-j) mod L]  ... but this isn't quite right for linear conv
    # For linear conv represented as circular conv of length L:
    # c_k = sum_i p[i] * q[(k-i) mod L] where q = p padded to L
    # d(c_k)/dp_j = q[(k-j) mod L] + p[(k-j) mod L] (from both the "a" and "b" sides)
    # Since a=b=p, and q is p padded: d(c_k)/dp_j = 2*q[(k-j) mod L] for the padded version
    # But we only track p[0..n-1], and q[m] = p[m] for m < n, 0 otherwise
    
    # Using FFT: H = ifft(FFT(2*p) * FFT(W_rev)) gives the gradient
    # Where W_rev is the time-reversed W placed in circular fashion
    
    W_full = np.zeros(L, dtype=np.float64)
    nw = 2 * n - 1
    W_full[:nw] = W[::-1]
    
    H = np.fft.irfft(np.fft.rfft(2.0 * np.pad(p, (0, L - n)), L) * np.fft.rfft(W_full, L), L)
    g_p = H[:n]
    g_p = np.where(np.isfinite(g_p), g_p, 0.0)
    
    # Project to logit space: grad_logit = g_p - p * (p . g_p)
    mean_g = float(np.dot(p, g_p))
    grad = g_p - p * mean_g
    
    # Scale for stability
    max_g = float(np.max(np.abs(grad)))
    if max_g > 0.005 and max_g > 1e-15:
        grad *= (0.005 / max_g)
    
    smooth_val = cm + tau * float(np.log(expsum))
    return float(2.0 * n * smooth_val), grad

    final_s = fast_score(best_p)
    p_sym = symmetrize(best_p)
    s_sym = fast_score(p_sym)
    if np.isfinite(s_sym) and s_sym < final_s:
        best_p = p_sym
        final_s = s_sym

    return best_p, final_s


def targeted_refine(p, n_trials, rng):
    """Try random perturbations focused on the peak region of the convolution."""
    n = p.size
    current_p = p.copy()
    current_s = fast_score(current_p)
    if not np.isfinite(current_s):
        return p, np.inf

    for _ in range(n_trials):
        # Random perturbation
        noise = rng.normal(0.0, 0.02, n)
        trial = current_p * np.exp(noise)
        trial = np.maximum(trial, 0.0)
        s = float(trial.sum())
        if s <= 0:
            continue
        trial /= s
        sc = fast_score(trial)
        if np.isfinite(sc) and sc < current_s - 1e-10:
            current_p = trial
            current_s = sc

        # Symmetrized perturbation
        noise2 = rng.normal(0.0, 0.015, n)
        trial2 = current_p * np.exp(noise2)
        trial2 = symmetrize(trial2)
        sc2 = fast_score(trial2)
        if np.isfinite(sc2) and sc2 < current_s - 1e-10:
            current_p = trial2
            current_s = sc2

    return current_p, current_s


def propose_candidate(seed=42, budget_s=1800.0, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    raw_budget = max(0.0, float(budget_s))
    deadline = t_start + max(0.0, raw_budget - 25.0)

    # Choose sequence length
    if raw_budget < 300:
        N = 4096
    else:
        N = 8192

    best_p = np.ones(N, dtype=np.float64) / float(N)
    best_s = 2.0

    # Try to use previous best
    try:
        hs = np.asarray(height_sequence_1, dtype=np.float64)
        hs = np.where(np.isfinite(hs), hs, 0.0)
        hs = np.maximum(hs, 0.0)
        if hs.size > 0 and float(hs.sum()) > 1e-9:
            s_hs = fast_score(hs)
            if np.isfinite(s_hs) and s_hs < best_s:
                best_p = normalize_positive(hs)
                best_s = s_hs
            # Also try resized
            pN = resize_sequence(hs, N)
            sN = fast_score(pN)
            if np.isfinite(sN) and sN < best_s:
                best_p = pN
                best_s = sN
            # Symmetrized
            p_sym = symmetrize(resize_sequence(hs, N))
            s_sym = fast_score(p_sym)
            if np.isfinite(s_sym) and s_sym < best_s:
                best_p = p_sym
                best_s = s_sym
            print(f"Starting from height_sequence_1: best={best_s:.6f}")
    except Exception:
        pass

    # Quick scan of structured initializations
    if time.monotonic() < deadline - 120:
        for kind in range(8):
            p_init = make_window(N, kind)
            if p_init is not None:
                s_init = fast_score(p_init)
                if np.isfinite(s_init) and s_init < best_s:
                    best_p = p_init
                    best_s = s_init
                    print(f"  Window {kind}: {s_init:.6f}")

        # Try the height_sequence at different sizes
        try:
            for m in [2048, 4096, N, 2 * N]:
                if time.monotonic() > deadline - 120:
                    break
                pm = resize_sequence(best_p, m)
                sm = fast_score(pm)
                if np.isfinite(sm) and sm < best_s:
                    best_p = pm
                    best_s = sm
                    N = m
                    print(f"  Resized to {m}: {sm:.6f}")
        except Exception:
            pass

    print(f"Initial best: {best_s:.6f} (N={N})")
    if time.monotonic() > deadline:
        return [float(x) for x in best_p]

    # Multi-phase Adam optimization
    # Phase 1: Broad exploration (medium tau)
    rem = deadline - time.monotonic()
    if rem > 120:
        dur = min(rem * 0.35, 300.0)
        parent = best_p.copy()
        p1, s1 = adam_phase(parent, dur, rng, 0.12, 0.04, 0.03, 0.012, N,
                           sym_period=8000, restart_patience=3000)
        if p1 is not None and np.isfinite(s1) and s1 < best_s:
            best_p = p1
            best_s = s1
        print(f"After phase 1: {best_s:.6f}")

    # Phase 2: Medium refinement
    rem = deadline - time.monotonic()
    if rem > 100:
        dur = min(rem * 0.30, 250.0)
        parent = best_p.copy()
        p2, s2 = adam_phase(parent, dur, rng, 0.04, 0.012, 0.018, 0.006, N,
                           sym_period=6000, restart_patience=2500)
        if p2 is not None and np.isfinite(s2) and s2 < best_s:
            best_p = p2
            best_s = s2
        # Also try from a noisy version
        noisy = best_p * np.exp(rng.normal(0.0, 0.08, N))
        noisy = normalize_positive(noisy)
        if noisy is not None:
            p2b, s2b = adam_phase(noisy, min(dur * 0.5, 100.0), rng, 0.035, 0.01, 0.015, 0.005, N,
                                  sym_period=5000, restart_patience=2000)
            if p2b is not None and np.isfinite(s2b) and s2b < best_s:
                best_p = p2b
                best_s = s2b
        print(f"After phase 2: {best_s:.6f}")

    # Phase 3: Fine refinement (low tau)
    rem = deadline - time.monotonic()
    if rem > 80:
        dur = min(rem * 0.30, 250.0)
        parent = best_p.copy()
        p3, s3 = adam_phase(parent, dur, rng, 0.012, 0.004, 0.008, 0.003, N,
                           sym_period=5000, restart_patience=2000)
        if p3 is not None and np.isfinite(s3) and s3 < best_s:
            best_p = p3
            best_s = s3
        print(f"After phase 3: {best_s:.6f}")

    # Phase 4: Very fine tuning
    rem = deadline - time.monotonic()
    if rem > 60:
        dur = min(rem * 0.35, 300.0)
        parent = best_p.copy()
        p4, s4 = adam_phase(parent, dur, rng, 0.004, 0.0015, 0.004, 0.002, N,
                           sym_period=4000, restart_patience=1500)
        if p4 is not None and np.isfinite(s4) and s4 < best_s:
            best_p = p4
            best_s = s4
        print(f"After phase 4: {best_s:.6f}")

    # Phase 5: Targeted local refinement
    rem = deadline - time.monotonic()
    if rem > 30:
        n_trials = int(min(rem * 3, 5000))
        p5, s5 = targeted_refine(best_p, n_trials, rng)
        if np.isfinite(s5) and s5 < best_s:
            best_p = p5
            best_s = s5
        print(f"After targeted refine: {best_s:.6f}")

    # Final symmetrization check
    p_sym = symmetrize(best_p)
    s_sym = fast_score(p_sym)
    if np.isfinite(s_sym) and s_sym < best_s:
        best_p = p_sym
        best_s = s_sym

    # Final validation
    candidates = []
    candidates.append(best_p)
    candidates.append(np.ones(1024, dtype=np.float64) / 1024.0)
    
    final_seq = None
    final_score = np.inf
    for cand in candidates:
        if cand is None:
            continue
        lst = [float(x) for x in np.clip(np.asarray(cand, dtype=np.float64), 0.0, 1000.0)]
        if not lst:
            continue
        try:
            sc = evaluate_sequence(lst)
        except Exception:
            sc = np.inf
        if np.isfinite(sc) and sc < final_score:
            final_score = float(sc)
            final_seq = lst

    if final_seq is None:
        final_seq = [1.0 / 1024.0] * 1024

    print(f"FINAL: {final_score:.6f} (N={len(final_seq)})")
    return final_seq
