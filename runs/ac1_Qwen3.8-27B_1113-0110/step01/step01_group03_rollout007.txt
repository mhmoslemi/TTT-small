import numpy as np
from scipy.sparse import coo_matrix
from scipy.optimize import linprog
import time

def fft_self_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:2 * n - 1]

def normalize_sum(a, target):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-10:
        return np.full_like(a, target / len(a))
    return a / S * target

def build_lp_indices(n):
    """Pre-compute (row, col) pairs for the LP constraint matrix."""
    rows_parts = []
    cols_parts = []
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        cnt = jmax - jmin + 1
        rows_parts.append(np.full(cnt, k, dtype=np.int32))
        cols_parts.append(np.arange(jmin, jmax + 1, dtype=np.int32))
    rows = np.concatenate(rows_parts)
    cols = np.concatenate(cols_parts)

"""
LP-based descent to minimize 2n*max(conv(a,a))/sum(a)^2.
Primary method: at each step, solve an LP to find the direction b that
maximizes sum(b) subject to (a*b)[k] <= M for all lags k. Blend in b
with adaptive step size.相结合 with fast FFT gradient polish for final refinement.
"""
import numpy as np
from scipy.sparse import coo_matrix
from scipy.optimize import linprog
import time


def fft_self_conv(a):
    """Compute full self-convolution a*a using FFT."""
    n = len(a)
    if n <= 1:
        return a * a
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:2 * n - 1]


def normalize_sum(a, target):
    """Normalize to have sum=target, clipping negatives to 0."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return np.full_like(a, target / len(a))
    return a / s * target


def build_lp_indices(n):
    """Pre-compute (row, col) index pairs for the constraint matrix A where A[k,j]=a[k-j]."""
    rows_parts = []
    cols_parts = []
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        cnt = jmax - jmin + 1
        rows_parts.append(np.full(cnt, k, dtype=np.int32))
        cols_parts.append(np.arange(jmin, jmax + 1, dtype=np.int32))
    rows = np.concatenate(rows_parts)
    cols = np.concatenate(cols_parts)
    return rows, cols


def lp_direction(a, M, rows, cols, time_limit=40):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0. Returns b or None."""
    n = len(a)
    val = a[rows.astype(np.intp) - cols.astype(np.intp)]
    A = coo_matrix((val, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': time_limit})
    if res.status == 0:
        return res.x
    return None


def gradient_polish(a, n, iterations, eta, deadline):
    """Fast FFT-based gradient descent polish. Returns updated a."""
    for _ in range(iterations):
        if time.monotonic() > deadline:
            break
        a = normalize_sum(a, n)
        c = fft_self_conv(a)
        M = c.max()
        # Use top-2 peaks for subgradient
        idx = np.argsort(-c)[:2]
        k1 = int(idx[0])
        k2 = int(idx[1])
        if k1 == k2:
            k2 = (k1 + 1) % (2 * n - 1)
        grad = np.zeros(n)
        for k in (k1, k2):
            jmin = max(0, k - n + 1)
            jmax = min(n - 1, k)
            grad[jmin:jmax + 1] += a[k - jmax:k - jmin + 1][::-1]
        grad /= 2.0
        a = a - eta * grad
        a = np.maximum(a, 0.0)
        a = normalize_sum(a, n)
    return a
