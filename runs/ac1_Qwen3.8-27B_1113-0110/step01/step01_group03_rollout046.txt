(a*b)(k) = sum_i a[k-i] * b[i] for i in [max(0, k-n+1), min(k, n-1)]

For efficiency, I can build the matrix using the structure:
- For k < n: A[k, 0:k+1] = a[k:k:-1] (reversed a[0:k+1])
- For k >= n: A[k, k-n+1:n] = a[n-1:k-n+1:-1] (reversed portion)

```python
def build_conv_matrix(a, n):
    m = 2 * n - 1
    rows = []
    cols = []
    vals = []
    
    # For k = 0 to n-1: i ranges from 0 to k
    # A[k, i] = a[k - i] for i = 0, ..., k
    # So row k has entries a[k], a[k-1], ..., a[0] in columns 0, 1, ..., k
    
    # For k = n to 2n-2: i ranges from k-n+1 to n-1
    # A[k, i] = a[k - i] for i = k-n+1, ..., n-1
    # So row k has entries a[k-(k-n+1)], ..., a[k-(n-1)] = a[n-1], ..., a[1] in columns k-n+1, ..., n-1
    
    for k in range(n):
        i_vals = np.arange(k + 1)
        rows.extend([k] * (k + 1))
        cols.extend(i_vals.tolist())
        vals.extend(a[k:-1 if k > 0 else None][::-1].tolist())
    
    for k in range(n, m):
        i_min = k - n + 1
        i_vals = np.arange(i_min, n)
        a_indices = k - i_vals  # = n-1, n-2, ..., 1
        rows.extend([k] * (n - i_min))
        cols.extend(i_vals.tolist())
        vals.extend(a[a_indices].tolist())
    
    return csr_matrix((vals, (rows, cols)), shape=(m, n))
```

```python
    n = len(a)
    if n <= 1:
        return a * a
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:2 * n - 1]
```

```python
def compute_gradient_multi_peak(a, c, n, top_k=3):
    """Compute score gradient using top-k distinct peaks."""
    sorted_idx = np.argsort(-c)
    k_stars = []

        j_max = min(n - 1, k)

            grad[idx] += a[k - idx]
```
