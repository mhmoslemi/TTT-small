```python
def _lp_direction(a, M):
    """Find direction b >= 0 maximizing sum(b) subject to convolve(a,b) <= M."""
    n = len(a)
    # Build constraint matrix: A[k, i] = a[k-i] if 0 <= k-i < n, else 0
    # This is a banded matrix (bandwidth n)
    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        for i in range(max(0, k-n+1), min(n, k+1)):
            j = k - i
            rows.append(k)
            cols.append(i)
            vals.append(a[j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    obj = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    
    from scipy.optimize import linprog
    res = linprog(c=obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    return res.x if res.success else None
```

```python
def _coord_update(p, c, score, i, grid, refine):
    """Coordinate descent update for position i."""
    n = p.size
    old = float(p[i])
    if old < 0:
        old = 0.0
    D = 1.0 - old
    
    ks = slice(i, i + n)
    c_k = c[ks].copy()
    
    coeff = 2.0 * p.copy()
    const = -2.0 * old * p.copy()
    mid = i
    if 0 <= mid < n:
        coeff[mid] = 0.0
        const[mid] = -old * old
    
    max_unaff = 0.0
    if i > 0:
        m = float(np.max(c[:i]))
        if m > max_unaff:
            max_unaff = m
    end = i + n
    if end < 2 * n - 1:
        m = float(np.max(c[end:]))
        if m > max_unaff:
            max_unaff = m
    
    aff = c_k[None, :] + coeff[None, :] * grid[:, None] + const[None, :]
    aff[:, mid] += grid * grid
    max_aff = np.max(aff, axis=1)
    maxc = np.maximum(max_unaff, max_aff)
    S = D + grid
    vals = np.full(grid.size, np.inf)
    valid = S > 1e-15
    vals[valid] = (2.0 * n) * maxc[valid] / (S[valid] ** 2)
    
    idx = int(np.argmin(vals))
    best_x, best_val = float(grid[idx]), float(vals[idx])
    tol = 1e-12 * max(1.0, score)
    
    if not np.isfinite(best_val) or best_val >= score - tol:
        return p, c, score, False
    
    # Local refinement
    if refine > 0:
        left = 0.0 if idx == 0 else float(grid[idx - 1])
        right = float(grid[idx]) * 10 if idx == grid.size - 1 else float(grid[idx + 1])
        if best_x > 0:
            local = best_x * np.logspace(-1, 1, refine)
        else:
            local = np.linspace(0, max(right, 1e-6), refine)
        local = np.unique(np.concatenate([local, [left, best_x, right]]))
        local = local[(local >= 0) & (local <= 1e4)]
        if local.size > 1:
            aff2 = c_k[None, :] + coeff[None, :] * local[:, None] + const[None, :]
            aff2[:, mid] += local * local
            max_aff2 = np.max(aff2, axis=1)
            maxc2 = np.maximum(max_unaff, max_aff2)
            S2 = D + local
            vals2 = np.full(local.size, np.inf)
            valid2 = S2 > 1e-15
            vals2[valid2] = (2.0 * n) * maxc2[valid2] / (S2[valid2] ** 2)
            idx2 = int(np.argmin(vals2))
            if float(vals2[idx2]) < best_val:
                best_x, best_val = float(local[idx2]), float(vals2[idx2])
    
    if not np.isfinite(best_val) or best_val >= score - tol:
        return p, c, score, False
    
    S = D + best_x
    S2 = S * S
    c_new = c / S2
    aff_best = c_k + coeff * best_x + const
    aff_best[mid] += best_x * best_x
    c_new[ks] = aff_best / S2
    p_new = p / S
    p_new[i] = best_x / S
    
    snew = p_new.sum()
    if not np.isfinite(snew) or snew <= 0:
        return p, c, score, False
    p_new /= snew
    c_new /= (snew * snew)
    np.clip(p_new, 0, None, out=p_new)
    np.clip(c_new, 0, None, out=c_new)
    
    s2 = p_new.sum()
    if not np.isfinite(s2) or s2 <= 0:
        return p, c, score, False
    p_new /= s2
    c_new /= (s2 * s2)
    
    score_new = 2.0 * n * np.max(c_new)
    if not np.isfinite(score_new) or score_new >= score - tol:
        return p, c, score, False
    return p_new, c_new, float(score_new), True


def _coord_descent(a, deadline, rng, grid, refine, max_perturb=4, noise_scales=(0.5, 0.3, 0.15, 0.07)):
    n = a.size
    a = _normalize(a).copy()
    c = np.convolve(a, a)
    score = 2.0 * n * np.max(c)
    best_a, best_score = a.copy(), score
    stagnant = 0
    perts = 0
    
    while time.monotonic() < deadline - 1:
        order = rng.permutation(n)
        improved = False
        for i in order:
            if time.monotonic() > deadline - 0.5:
                break
            a, c, score, imp = _coord_update(a, c, score, int(i), grid, refine)
            if imp:
                improved = True
        
        s = a.sum()
        if s <= 0:
            break
        a = a / s
        c = np.convolve(a, a)
        np.clip(c, 0, None, out=c)
        score = 2.0 * n * np.max(c)
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        elif improved:
            stagnant = 0
        else:
            stagnant += 1
            if score > best_score * 1.2:
                a = best_a.copy()
                c = np.convolve(a, a)
                score = best_score
        
        if stagnant >= 2 and perts < max_perturb:
            scale = noise_scales[min(perts, len(noise_scales) - 1)]
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.clip(a, 0, None)
            s = a.sum()
            if np.isfinite(s) and s > 0:
                a /= s
                c = np.convolve(a, a)
                score = 2.0 * n * np.max(c)
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
            perts += 1
            stagnant = 0
        elif stagnant >= 6:
            break
    
    return best_a, best_score


def _make_initial(n, strategy, rng):
    if strategy == 0:
        return np.ones(n)
    elif strategy == 1:
        return rng.dirichlet(np.ones(n))
    elif strategy == 2:
        return rng.dirichlet(np.full(n, 0.3))
    elif strategy == 3:
        # Two-level half block
        m = n // 2
        p = np.zeros(n)
        p[:m] = (2.0 / 3.0) / m
        p[m:] = (1.0 / 3.0) / (n - m)
        return p
    elif strategy == 4:
        # Exponential
        p = rng.exponential(1.0, n)
        return p
    elif strategy == 5:
        # Paired anti-correlated
        m = (n + 1) // 2
        w = rng.dirichlet(np.ones(m))
        p = np.zeros(n)
        for r in range(m):
            a_idx, b_idx = r, n - 1 - r
            if a_idx == b_idx:
                p[a_idx] = w[r]
            else:
                q = rng.random()
                if q < 0.6:
                    p[a_idx] = w[r]
                else:
                    p[b_idx] = w[r]
        return p
    elif strategy == 6:
        # Fourier-based: random phases
        ph = rng.uniform(0, 2 * np.pi, n)
        A = np.exp(1j * ph)
        a = np.real(np.fft.ifft(A))
        a = np.maximum(0, a)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.ones(n)
        return a
    elif strategy == 7:
        # Gaussian + clip
        p = np.maximum(0, rng.normal(0.5, 0.3, n))
        return p
    elif strategy == 8:
        # sqrt-shaped
        x = np.linspace(0, 1, n)
        p = np.sqrt(x) * (1 + 0.1 * rng.random(n))
        return p
    elif strategy == 9:
        # Binary noise
        p = (rng.random(n) > 0.5).astype(float)
        p += 0.1 * rng.random(n)
        return p
    else:
        return rng.dirichlet(np.ones(n))

"""
```

```python
Key improvements over previous approach:
1. LP-based direction finding on the FULL 3809-length sequence.
2. Hybrid alternating LP + coordinate descent optimization.

4. Adaptive coordinate descent with refined local search.

import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse


def _clean_array(arr):
    a = np.asarray(arr, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    np.clip(a, 0.0, 1000.0, out=a)
    return a.copy()


def _normalize(a):
    a = _clean_array(a)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(max(1, a.size), dtype=np.float64)
    return a / s


def _score_of(p):
    n = p.size
    if n <= 0:
        return float('inf')
    c = np.convolve(p, p)
    return float(2.0 * n * np.max(c))


def _two_level_full(n):
    n = int(n)
    m = n // 2
    p = np.zeros(n, dtype=np.float64)
    p[:m] = (2.0 / 3.0) / float(m)
    p[m:] = (1.0 / 3.0) / float(n - m)
    return _normalize(p)


def _build_conv_mat(n, a):
    """Build sparse matrix A where A[k,i] = a[k-i] if 0<=k-i<n else 0. Shape (2n-1, n)."""
    row_parts = []
    col_parts = []
    val_parts = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        cnt = hi - lo
        if cnt <= 0:
            continue
        row_parts.append(np.full(cnt, k, dtype=np.int32))
        col_parts.append(np.arange(lo, hi, dtype=np.int32))
        val_parts.append(a[k - np.arange(lo, hi)])
    if not row_parts:
        return sparse.csr_matrix((2 * n - 1, n))
    rows = np.concatenate(row_parts)
    cols = np.concatenate(col_parts)
    vals = np.concatenate(val_parts)
    return sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_find_direction(a, M, timeout=50):
    """Find b >= 0 maximizing sum(b) s.t. convolve(a,b)[k] <= M for all k."""
    n = a.size
    A = _build_conv_mat(n, a)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    obj = -np.ones(n, dtype=np.float64)
    try:
        res = linprog(c=obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res is not None and getattr(res, 'success', False):
            b = np.maximum(0.0, np.asarray(res.x, dtype=np.float64))
            if np.sum(b) > 1e-12:
                return b
    except Exception:
        pass
    return None


def _lp_line_search(a, b, c, M, n):
    """Find best t in [0,1] for a_new = normalize((1-t)*a + t*b)."""
    a = _normalize(a)
    b = _normalize(b)
    cross = np.convolve(a, b)
    cb = np.convolve(b, b)

    best_t, best_s = 0.0, 2.0 * n * M
    ts = np.linspace(0.0, 1.0, 33)
    for t in ts:
        at = (1.0 - t) ** 2 * c + 2.0 * t * (1.0 - t) * cross + t ** 2 * cb
        s = 2.0 * n * float(np.max(at))
        if s < best_s:
            best_s = s
            best_t = float(t)

    lo = max(0.0, best_t - 0.15)
    hi = min(1.0, best_t + 0.15)
    ts2 = np.linspace(lo, hi, 33)
    for t in ts2:
        at = (1.0 - t) ** 2 * c + 2.0 * t * (1.0 - t) * cross + t ** 2 * cb
        s = 2.0 * n * float(np.max(at))
        if s < best_s:
            best_s = s
            best_t = float(t)
    return best_t, best_s


def _lp_polish_full(a, deadline, max_iters=8, lp_timeout=45):
    """Run LP-based optimization on the full sequence."""
    n = a.size
    a = _normalize(a).copy()
    c = np.convolve(a, a)
    np.clip(c, 0.0, None, out=c)
    score = 2.0 * n * float(np.max(c))
    best_a, best_score = a.copy(), score

    for it in range(max_iters):
        if time.monotonic() > deadline - 60:
            break
        M = float(np.max(c))
        t0 = time.monotonic()
        b = _lp_find_direction(a, M, timeout=lp_timeout)
        lp_time = time.monotonic() - t0
        if b is None:
            print(f"    LP iter {it}: no direction found")
            break
        b = _normalize(b)

        best_t, best_s = _lp_line_search(a, b, c, M, n)
        if best_s >= score - 1e-10:
            print(f"    LP iter {it}: no improvement (t={best_t:.4f})")
            break

        a_new = _normalize((1.0 - best_t) * a + best_t * b)
        c_new = np.convolve(a_new, a_new)
        np.clip(c_new, 0.0, None, out=c_new)
        score_new = 2.0 * n * float(np.max(c_new))

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()

        a, c, score = a_new, c_new, score_new
        print(f"    LP iter {it}: score={score:.6f}, t={best_t:.4f}, lp_time={lp_time:.1f}s")

    return best_a, best_score


def _coord_update(p, c, score, i, grid, refine):
    """Coordinate descent update for position i with exact 1D minimization on grid."""
    n = p.size
    old = float(p[i])
    if old < 0.0:
        old = 0.0
    D = 1.0 - old
    if D < 0.0:
        D = 0.0

    ks = slice(i, i + n)
    c_k = c[ks].copy()

    coeff = 2.0 * p.copy()
    const = -2.0 * old * p.copy()
    mid = i
    coeff[mid] = 0.0
    const[mid] = -old * old

    max_unaff = 0.0
    if i > 0:
        m = float(np.max(c[:i]))
        if m > max_unaff:
            max_unaff = m
    end = i + n
    if end < 2 * n - 1:
        m = float(np.max(c[end:]))
        if m > max_unaff:
            max_unaff = m

    valid_grid = grid[(grid >= 0.0) & (grid <= 1e4) & np.isfinite(grid)]
    if valid_grid.size == 0:
        return p, c, score, False

    aff = c_k[None, :] + coeff[None, :] * valid_grid[:, None] + const[None, :]
    aff[:, mid] += valid_grid * valid_grid
    max_aff = np.max(aff, axis=1)
    maxc = np.maximum(max_unaff, max_aff)
    S = D + valid_grid
    vals = np.full(valid_grid.size, np.inf, dtype=np.float64)
    valid = S > 1e-15
    vals[valid] = (2.0 * n) * maxc[valid] / (S[valid] ** 2)

    idx = int(np.argmin(vals))
    best_x = float(valid_grid[idx])
    best_val = float(vals[idx])
    tol = 1e-12 * max(1.0, score)

    if not np.isfinite(best_val) or best_val >= score - tol:
        return p, c, score, False

    if refine > 0 and valid_grid.size > 1:
        left = 0.0 if idx == 0 else float(valid_grid[idx - 1])
        right = float(valid_grid[idx]) * 10.0 if idx == valid_grid.size - 1 else float(valid_grid[idx + 1])
        if best_x > 0.0:
            local = best_x * np.logspace(-1.0, 1.0, refine)
        else:
            base_right = max(right, 1e-8)
            local = np.linspace(0.0, base_right, refine)
        local = np.unique(np.concatenate([local, [left, best_x, right]]))
        local = local[(local >= 0.0) & (local <= 1e4) & np.isfinite(local)]
        if local.size > 1:
            aff2 = c_k[None, :] + coeff[None, :] * local[:, None] + const[None, :]
            aff2[:, mid] += local * local
            max_aff2 = np.max(aff2, axis=1)
            maxc2 = np.maximum(max_unaff, max_aff2)
            S2 = D + local
            vals2 = np.full(local.size, np.inf, dtype=np.float64)
            valid2 = S2 > 1e-15
            vals2[valid2] = (2.0 * n) * maxc2[valid2] / (S2[valid2] ** 2)
            idx2 = int(np.argmin(vals2))
            if float(vals2[idx2]) < best_val:
                best_x = float(local[idx2])
                best_val = float(vals2[idx2])

    if not np.isfinite(best_val) or best_val >= score - tol:
        return p, c, score, False

    S = D + best_x
    if S <= 1e-15:
        return p, c, score, False
    S2 = S * S

    c_new = c / S2
    aff_best = c_k + coeff * best_x + const
    aff_best[mid] += best_x * best_x
    c_new[ks] = aff_best / S2

    p_new = p / S
    p_new[i] = best_x / S

    snew = float(p_new.sum())
    if not np.isfinite(snew) or snew <= 0:
        return p, c, score, False
    p_new = p_new / snew
    c_new = c_new / (snew * snew)
    np.clip(p_new, 0.0, None, out=p_new)
    np.clip(c_new, 0.0, None, out=c_new)

    s2 = float(p_new.sum())
    if not np.isfinite(s2) or s2 <= 0:
        return p, c, score, False
    p_new = p_new / s2
    c_new = c_new / (s2 * s2)

    score_new = float(2.0 * n * np.max(c_new))
    if not np.isfinite(score_new) or score_new >= score - tol:
        return p, c, score, False
    return p_new, c_new, score_new, True


def _coord_descent(a, deadline, rng, grid, refine, max_perturb=5, noise_scales=None):
    """Full coordinate descent with periodic perturbations."""
    if noise_scales is None:
        noise_scales = (0.6, 0.35, 0.18, 0.09, 0.04)
    n = a.size
    a = _normalize(a).copy()
    c = np.convolve(a, a)
    np.clip(c, 0.0, None, out=c)
    score = 2.0 * n * float(np.max(c))
    best_a, best_score = a.copy(), score
    stagnant = 0
    perts = 0
    cycles_no_best = 0

    while time.monotonic() < deadline - 1:
        order = rng.permutation(n)
        improved = False
        for i in order:
            if time.monotonic() > deadline - 0.5:
                break
            a, c, score, imp = _coord_update(a, c, score, int(i), grid, refine)
            if imp:
                improved = True

        s = float(a.sum())
        if not np.isfinite(s) or s <= 0:
            break
        a = a / s
        c = np.convolve(a, a)
        np.clip(c, 0.0, None, out=c)
        score = 2.0 * n * float(np.max(c))

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            cycles_no_best = 0
        else:
            cycles_no_best += 1

        if improved:
            stagnant = 0
        else:
            stagnant += 1
            if score > best_score * 1.15:
                a = best_a.copy()
                c = np.convolve(a, a)
                score = best_score

        if cycles_no_best >= 5 and score > best_score * 1.01:
            a = best_a.copy()
            c = np.convolve(a, a)
            score = best_score
            cycles_no_best = 0

        if stagnant >= 2 and perts < max_perturb:
            scale = noise_scales[min(perts, len(noise_scales) - 1)]
            p = best_a * np.exp(rng.normal(0.0, scale, n))
            p = np.clip(p, 0.0, None)
            sp = float(p.sum())
            if np.isfinite(sp) and sp > 0:
                p = p / sp
                cp = np.convolve(p, p)
                np.clip(cp, 0.0, None, out=cp)
                sc = 2.0 * n * float(np.max(cp))
                if sc < best_score:
                    best_score = sc
                    best_a = p.copy()
                    a, c, score = p, cp, sc
            perts += 1
            stagnant = 0
        elif stagnant >= 8:
            break

    return best_a, best_score


def _make_initial(n, strategy, rng):
    n = int(n)
    if n <= 1:
        return np.ones(max(1, n), dtype=np.float64)
    st = int(strategy) % 12
    if st == 0:
        p = np.ones(n, dtype=np.float64)
    elif st == 1:
        p = rng.dirichlet(np.ones(n, dtype=np.float64))
    elif st == 2:
        p = rng.dirichlet(np.full(n, 0.3, dtype=np.float64))
    elif st == 3:
        m = n // 2
        p = np.zeros(n, dtype=np.float64)
        p[:m] = (2.0 / 3.0) / float(m)
        p[m:] = (1.0 / 3.0) / float(n - m)
    elif st == 4:
        p = rng.exponential(1.0, n)
    elif st == 5:
        m = (n + 1) // 2
        w = rng.dirichlet(np.ones(m, dtype=np.float64))
        p = np.zeros(n, dtype=np.float64)
        for r in range(m):
            ai, bi = r, n - 1 - r
            if ai == bi:
                p[ai] = w[r]
            else:
                if rng.random() < 0.65:
                    p[ai] = w[r]
                else:
                    p[bi] = w[r]
    elif st == 6:
        ph = rng.uniform(0.0, 2.0 * np.pi, n)
        A = np.exp(1j * ph)
        p = np.real(np.fft.ifft(A))
        p = np.maximum(0.0, p)
        sp = p.sum()
        if sp > 0:
            p = p / sp
        else:
            p = np.ones(n, dtype=np.float64) / n
    elif st == 7:
        p = np.maximum(0.0, rng.normal(0.5, 0.4, n))
        sp = p.sum()
        p = p / sp if sp > 0 else np.ones(n) / n
    elif st == 8:
        x = np.linspace(0.0, 1.0, n, dtype=np.float64)
        p = np.sqrt(x + 0.01) * (1.0 + 0.1 * rng.random(n))
    elif st == 9:
        p = (rng.random(n) > 0.5).astype(np.float64)
        p += 0.15 * rng.random(n)
    elif st == 10:
        # Two-level with random split ratio
        ratio = rng.uniform(0.3, 0.7)
        m = int(n * ratio)
        m = max(1, min(n - 1, m))
        p = np.zeros(n, dtype=np.float64)
        p[:m] = (2.0 / 3.0) / float(m)
        p[m:] = (1.0 / 3.0) / float(n - m)
    else:
        # Paired with soft split
        m = (n + 1) // 2
        w = rng.dirichlet(np.full(m, 0.8, dtype=np.float64))
        p = np.zeros(n, dtype=np.float64)
        for r in range(m):
            ai, bi = r, n - 1 - r
            if ai == bi:
                p[ai] = w[r]
            else:
                q = rng.beta(2.0, 2.0)
                p[ai] = w[r] * q
                p[bi] = w[r] * (1.0 - q)
    return _normalize(p)


def _grid_for_size(n):

        return np.concatenate(([0.0], np.logspace(-8.0, 2.0, 50))), 18
    elif n <= 256:

    elif n <= 1024:
        return np.concatenate(([0.0], np.logspace(-8.0, 2.0, 24))), 9
    else:
        return np.concatenate(([0.0], np.logspace(-10.0, 2.0, 20))), 8


def propose_candidate(seed=42, budget_s=2400, **kwargs):

    # --- Phase 2: Coordinate descent ---

        a_cd, sc_cd = _coord_descent(
            a0, cd1_deadline, rng, grid, refine,
            max_perturb=5, noise_scales=(0.5, 0.3, 0.15, 0.08, 0.04)
        )
        _try_update(a_cd, sc_cd)
        if sc_cd < best_score:
            a0 = a_cd
        print(f"  After CD: best={best_score:.6f}")

    # --- Phase 3: More LP ---
    if time.monotonic() < deadline - 120:
        lp2_deadline = min(deadline - 60, time.monotonic() + 250.0)
        print(f"\n{'='*50}")
        print(f"Phase 3: More LP ({lp2_deadline - time.monotonic():.0f}s available)")
        print(f"{'='*50}")

        a_lp2, sc_lp2 = _lp_polish_full(
            a0, lp2_deadline,
            max_iters=4, lp_timeout=40
        )
        _try_update(a_lp2, sc_lp2)
        if sc_lp2 < best_score:
            a0 = a_lp2
        print(f"  After LP2: best={best_score:.6f}")

    # --- Phase 4: Final coordinate descent ---
    if time.monotonic() < deadline - 15:
        cd2_deadline = deadline - 12
        print(f"\n{'='*50}")
        print(f"Phase 4: Final CD ({cd2_deadline - time.monotonic():.0f}s available)")
        print(f"{'='*50}")

        grid_fine, refine_fine = _grid_for_size(final_len)
        grid_fine = np.concatenate([grid_fine, np.logspace(-4.0, 0.5, 10)])
        grid_fine = np.unique(grid_fine)
        a_f, sc_f = _coord_descent(
            a0, cd2_deadline, rng, grid_fine, refine_fine,
            max_perturb=3, noise_scales=(0.3, 0.15, 0.07)
        )
        _try_update(a_f, sc_f)
        print(f"  Final: best={best_score:.6f}")

    # --- Validate with external evaluator ---
    if best_seq is not None:
        try:
            ext_score = evaluate_sequence([float(x) for x in best_seq])
            if np.isfinite(ext_score):
                print(f"\nExternal evaluation: {ext_score:.6f} (internal: {best_score:.6f})")
                if ext_score < best_score:
                    best_score = ext_score
        except Exception:
            pass

    if best_seq is None:
        best_seq = _two_level_full(final_len)

    print(f"\n{'='*50}")
    print(f"FINAL BEST SCORE: {best_score:.6f}")
    print(f"Target: 1.5027, Gap: {best_score - 1.5027:.6f}")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
```
