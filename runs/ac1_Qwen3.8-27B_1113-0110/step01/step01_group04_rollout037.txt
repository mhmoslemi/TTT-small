    a = a / a.sum()

    top_k = 3
    k_vals = np.argsort(c)[::-1][:top_k]

    g = np.zeros(n)
    for k in k_vals:
        # g[m] += a[m-k] + a[k-m] (with bounds check)
        for m in range(n):

            idx2 = int(k) - m
            val = 0.0
            if 0 <= idx1 < n:
                val += a[idx1]
            if 0 <= idx2 < n:
                val += a[idx2]
            g[m] += val
    g /= top_k
    g -= g.sum() * a.sum() / n  # center it (this isn't quite right for simplex)

    # The subgradient should be projected to the tangent space: g -= mean(g)

    e = np.exp(np.clip(x - m, -700, 700))
    e_last = float(np.exp(-m))
    s = float(e.sum()) + e_last
    if s <= 0 or not np.isfinite(s):
        return 1e100, np.zeros_like(x)
    a = np.empty(n, dtype=np.float64)
    a[:-1] = e / s
    a[-1] = e_last / s

    L = _pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    c = np.fft.irfft(A * A, n=L)[:2 * n - 1]
    c = np.maximum(c, 0.0)

    f = float(np.dot(c, c))
    W = np.fft.rfft(c, n=L)
    ga = 2.0 * np.fft.irfft(W * np.conj(A), n=L)[:n]
    dot = float(np.dot(a, ga))
    gfull = a * (ga - dot)
    gy = gfull[:-1]

    if not np.all(np.isfinite(gy)):
        gy = np.zeros_like(x)
    if not np.isfinite(f):
        f = 1e100
    return f, gy


def _lse_obj_grad_logratio(x, eps):
    """Smoothed max (log-sum-exp) of convolution, log-ratio parameterization."""
    x = np.where(np.isfinite(x), x, 0.0)
    n = x.size + 1
    if n <= 1:
        return 0.0, np.zeros_like(x)

    m = float(np.max(x)) if x.size else 0.0
    e = np.exp(np.clip(x - m, -700, 700))
    e_last = float(np.exp(-m))
    s = float(e.sum()) + e_last
    if s <= 0 or not np.isfinite(s):
        return 1e100, np.zeros_like(x)
    a = np.empty(n, dtype=np.float64)
    a[:-1] = e / s
    a[-1] = e_last / s

    L = _pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    c = np.fft.irfft(A * A, n=L)[:2 * n - 1]
    c = np.maximum(c, 0.0)

    cmax = float(c.max())
    if cmax < 1e-300:
        return 0.0, np.zeros_like(x)

    z = (c - cmax) / eps
    z = np.clip(z, -700, 0)
    ez = np.exp(z)
    sume = float(ez.sum())
    if sume <= 0 or not np.isfinite(sume):
        return 1e100, np.zeros_like(x)

    f = cmax + eps * float(np.log(sume))
    w = ez / sume
    W = np.fft.rfft(w, n=L)
    ga = 2.0 * np.fft.irfft(W * np.conj(A), n=L)[:n]
    dot = float(np.dot(a, ga))
    gfull = a * (ga - dot)
    gy = gfull[:-1]

    if not np.all(np.isfinite(gy)):
        gy = np.zeros_like(x)
    if not np.isfinite(f):
        f = 1e100
    return f, gy


def _a_to_logratio(a):
    a = _clean(a).copy()
    n = a.size
    if n == 0:
        return a
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    if n == 1:
        return a
    ref = float(a[-1])
    if ref < 1e-200:
        a[-1] = 1e-12
        a = a / a.sum()
        ref = float(a[-1])
    x = np.log(np.maximum(a[:-1], 1e-300) / max(ref, 1e-300))
    return np.clip(x, -100, 100)


def _logratio_to_a(x):
    x = np.where(np.isfinite(x), x, 0.0)
    n = x.size + 1
    if n == 1:
        return np.ones(1)
    m = float(np.max(x)) if x.size else 0.0
    e = np.exp(np.clip(x - m, -700, 700))
    e_last = float(np.exp(-m))
    s = float(e.sum()) + e_last
    if s <= 0 or not np.isfinite(s):
        return np.full(n, 1.0 / n)
    a = np.empty(n, dtype=np.float64)
    a[:-1] = e / s
    a[-1] = e_last / s
    return a


def _run_lbfgs(obj_grad, x, maxiter, bounds):
    if _minimize is None:
        return x
    try:
        res = _minimize(obj_grad, x, method='L-BFGS-B', jac=True,
                        bounds=bounds,
                        options={'maxiter': int(maxiter), 'disp': 0,
                                 'ftol': 1e-16, 'gtol': 1e-10,
                                 'maxfun': int(max(100, 8 * maxiter))})
        x_new = np.asarray(res.x, dtype=np.float64)
        x_new = np.where(np.isfinite(x_new), x_new, 0.0)
        return np.clip(x_new, -100, 100)
    except Exception:
        return x


def _subgrad_smooth_optimize(a0, end_time, seed, mode='hybrid'):
    """
    Main optimization loop: alternates subgradient steps and L-BFGS smoothing.
    """
    a = _project_simplex(a0)
    n = a.size
    best = a.copy()
    best_score = _score(best)
    n_line = 10
    n_top = 5

    start = time.monotonic()
    total_time = max(1.0, end_time - start)

    # Phased schedule:
    # - First 40%: mainly subgradient (aggressive)
    # - Next 40%: alternating subgradient + L-BFGS
    # - Last 20%: L-BFGS on LSE (fine-tuning)
    t_sub_only = start + 0.35 * total_time
    t_alter = start + 0.70 * total_time

    x = _a_to_logratio(a)
    bounds = [(-80.0, 80.0)] * max(1, n - 1)

    no_improve_steps = 0
    max_no_improve = 80
    t_max = 0.04

    while time.monotonic() < end_time:
        now = time.monotonic()
        if now >= end_time:
            break

        elapsed_frac = (now - start) / total_time

        if now < t_sub_only:
            # Pure subgradient phase
            a, sc = _subgradient_step(a, n_top=n_top, t_max=t_max, n_line=n_line)
            sc = _score(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best = a.copy()
                no_improve_steps = 0
                t_max = min(0.08, t_max * 1.02)
            else:
                no_improve_steps += 1
                if no_improve_steps % 10 == 0:
                    t_max = max(0.005, t_max * 0.85)
                    if no_improve_steps > max_no_improve:
                        # Perturb and restart
                        rng = np.random.default_rng(seed + no_improve_steps)
                        a = _project_simplex(best * np.exp(rng.normal(0, 0.03, n)))
                        no_improve_steps = 0
                        t_max = 0.04
                        seed += 1

        elif now < t_alter:
            # Alternating phase: 1 subgrad + 1 L-BFGS
            # Subgradient step
            a, _ = _subgradient_step(a, n_top=n_top, t_max=t_max, n_line=n_line)
            sc = _score(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best = a.copy()
                no_improve_steps = 0
            else:
                no_improve_steps += 1

            # L-BFGS step on L2 surrogate
            x = _a_to_logratio(a)
            x = _run_lbfgs(_l2_obj_grad_logratio, x, 30, bounds)
            a_new = _logratio_to_a(x)
            sc_new = _score(a_new)
            if sc_new < best_score - 1e-12:
                best_score = sc_new
                best = a_new.copy()
                a = a_new
                no_improve_steps = 0
            elif time.monotonic() - start > 10:
                a = best.copy()

        else:
            # Fine-tuning: LSE objective with decreasing eps
            eps = max(1e-7, 0.02 * (1.0 - elapsed_frac / total_time + 0.3))
            x = _a_to_logratio(a)
            x = _run_lbfgs(lambda z: _lse_obj_grad_logratio(z, eps), x, 50, bounds)
            a_new = _logratio_to_a(x)
            sc_new = _score(a_new)
            if sc_new < best_score - 1e-12:
                best_score = sc_new
                best = a_new.copy()
                a = a_new
                no_improve_steps = 0
            else:
                no_improve_steps += 1
                if no_improve_steps > 30:
                    # Try a small subgradient nudge
                    a, _ = _subgradient_step(best, n_top=3, t_max=0.01, n_line=8)
                    sc = _score(a)
                    if sc < best_score:
                        best_score = sc
                        best = a.copy()
                    no_improve_steps = 0

        # Periodic progress log
        if no_improve_steps % 200 == 0 and no_improve_steps > 0:
            print(f"  [opt] t={now-start:.0f}s score={best_score:.6f} t_max={t_max:.4f}")

    return best, best_score


def _pair_swap_local_search(a0, end_time, seed):
    """
    Local search: pick pairs (i,j), transfer mass, accept if score improves.
    Uses FFT-based scoring for speed.
    """
    a = _project_simplex(a0)
    n = a.size
    best = a.copy()
    best_score = _score(best)

    rng = np.random.default_rng(seed)
    n_iters = 0
    last_improve = time.monotonic()
    start = last_improve
    transfer_frac = 0.05

    while time.monotonic() < end_time:
        if time.monotonic() - last_improve > 30.0 and time.monotonic() - start > 60.0:
            break

        n_iters += 1

        # Pick a random pair
        if n_iters % 5 == 0:
            # Structured: pick from top-contributing lags
            c = _conv(a)
            k_max = int(np.argmax(c))
            # Find indices contributing to this lag
            valid_i = np.arange(max(0, k_max - n + 1), min(n, k_max + 1))
            if len(valid_i) >= 2:
                i = int(valid_i[rng.integers(0, len(valid_i))])
            else:
                i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                j = (j + 1) % n
        else:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue

        # Try transferring some mass from i to j and vice versa
        for direction in (1, -1):
            t = transfer_frac * a[i]
            if t <= 0:
                continue
            a[i] -= direction * t
            a[j] += direction * t
            if a[i] < 0:
                a[i] = 0
                a[j] += direction * 2e-9  # keep sum roughly
            a_new = _project_simplex(a)
            sc = _score(a_new)
            if sc < best_score - 1e-13:
                best = a_new.copy()
                best_score = sc
                last_improve = time.monotonic()
                break
        a = best.copy()

        # Adapt transfer fraction
        if time.monotonic() - last_improve > 5.0:
            transfer_frac = max(0.01, transfer_frac * 0.9)
        else:
            transfer_frac = min(0.1, transfer_frac * 1.01)

    return best, best_score


def _generate_candidates(n):
    """Generate diverse initial candidates."""
    cands = []
    j = np.arange(n, dtype=np.float64)

    # Standard power law 1/sqrt
    for p in (0.45, 0.48, 0.5, 0.52, 0.55):
        for off in (0.5, 1.0, 1.5, 2.0):
            v = np.power(j + off, -p)
            cands.append(_project_simplex(v))

    # Modulated power law
    for p in (0.48, 0.5, 0.52):
        for K in (8, 16, 32, 64, 128, 256):
            for eps_mod in (0.1, 0.2, 0.3, 0.5):
                v = np.power(j + 1.0, -p)
                v = v * (1.0 + eps_mod * np.cos(2.0 * np.pi * j / K))
                v = np.maximum(v, 0.0)
                cands.append(_project_simplex(v))

    # Two-sided (symmetric)
    for p in (0.48, 0.5, 0.55):
        dist = np.minimum(j, n - 1.0 - j) + 1.0
        v = np.power(dist, -p)
        cands.append(_project_simplex(v))

    # One-sided with central dip
    for p in (0.48, 0.5, 0.52):
        base = np.power(j + 1.0, -p)
        for dip_sigma in (0.1, 0.2, 0.3):
            for dip_depth in (0.1, 0.2, 0.3, 0.5):
                t = (j - n * 0.4) / (n * dip_sigma)
                v = base * (1.0 - dip_depth * np.exp(-0.5 * t * t))
                v = np.maximum(v, 0.0)
                cands.append(_project_simplex(v))

    # Exponential decay
    for rate in (0.3, 0.5, 0.7, 1.0, 1.5, 2.0):
        v = np.exp(-rate * j / n)
        cands.append(_project_simplex(v))

    # Load previous best if available
    try:
        h = height_sequence_1
        if isinstance(h, (list, np.ndarray)):
            arr = np.asarray(h, dtype=np.float64)
            if arr.ndim == 1 and arr.size == n and arr.sum() > 0:
                cands.insert(0, _project_simplex(arr))
    except Exception:
        pass

    return cands


def _score_candidates(cands):
    """Score and sort candidates."""
    scored = []
    for i, a in enumerate(cands):
        sc = _score(a)
        if np.isfinite(sc):
            scored.append((sc, i, a))
    scored.sort()
    return scored


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main entry point. Returns the best sequence found within budget_s seconds.
    """
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s) - 15.0)

    # Determine sequence length
    n = 3809
    try:
        h = height_sequence_1
        if isinstance(h, (list, np.ndarray)):
            arr = np.asarray(h, dtype=np.float64)
            if arr.ndim == 1 and 50 <= arr.size <= 20000:
                n = int(arr.size)
    except Exception:
        pass
    if n < 32:
        n = 3809

    print(f"Starting optimization with n={n}, budget={budget_s}s")

    # Very short budget fallback
    if deadline < t0 + 10.0:
        a = _project_simplex(np.power(np.arange(n, dtype=np.float64) + 1.0, -0.5))
        return [float(x) for x in a]

    # Phase 1: Generate and score candidates (5% of time)
    p1_end = t0 + 0.05 * (deadline - t0)
    cands = _generate_candidates(n)
    scored = _score_candidates(cands)

    if not scored:
        a = np.full(n, 1.0 / n)
        return [float(x) for x in a]

    global_best = scored[0][2].copy()
    global_score = scored[0][0]
    print(f"Phase 1: Best initial candidate score = {global_score:.6f}")
    for i in range(min(5, len(scored))):
        print(f"  cand[{i}]: {scored[i][0]:.6f}")

    # Phase 2: Optimize top candidates (75% of time)
    top_k = min(3, len(scored))
    p2_end = t0 + 0.82 * (deadline - t0)
    alloc_fracs = [0.55, 0.30, 0.15]

    for idx in range(top_k):
        if time.monotonic() >= p2_end:
            break
        frac = alloc_fracs[idx] if idx < len(alloc_fracs) else 0.1
        sub_end = t0 + (sum(alloc_fracs[:idx + 1]) * 0.82) * (deadline - t0)
        sub_end = min(sub_end, p2_end)

        a0 = scored[idx][2].copy()
        seed_i = int(seed) + 100 * (idx + 1)

        print(f"Phase 2: Optimizing candidate {idx}, score={scored[idx][0]:.6f}, time_to={sub_end - time.monotonic():.0f}s")

        a_opt, sc_opt = _subgrad_smooth_optimize(a0, sub_end, seed_i)

        if np.isfinite(sc_opt) and sc_opt < global_score:
            global_score = sc_opt
            global_best = a_opt.copy()
            print(f"  Improved to {global_score:.6f}")

    # Phase 3: Local search on global best (18% of time)
    p3_end = deadline
    if time.monotonic() < p3_end:
        print(f"Phase 3: Local search, remaining={p3_end - time.monotonic():.0f}s")
        a_ls, sc_ls = _pair_swap_local_search(global_best, p3_end, int(seed) + 99999)
        if np.isfinite(sc_ls) and sc_ls < global_score:
            global_score = sc_ls
            global_best = a_ls.copy()
            print(f"  Local search improved to {global_score:.6f}")

    # Final validation
    global_best = _project_simplex(global_best)
    # Scale to have sum ~ 1 (evaluate_sequence normalizes internally but let's be safe)
    out = [max(0.0, float(x)) for x in global_best]

    # Validate with evaluate_sequence
    try:
        val = evaluate_sequence(out)
        print(f"Final score via evaluate_sequence: {val}")
        if not np.isfinite(val):
            out = [float(x) for x in _project_simplex(np.power(np.arange(n, dtype=np.float64) + 1.0, -0.5))]
            val = evaluate_sequence(out)
            print(f"Fallback score: {val}")
    except Exception as ex:
        print(f"Warning: evaluate_sequence call failed: {ex}")

    elapsed = time.monotonic() - t0
    print(f"Done in {elapsed:.1f}s, best score = {global_score:.6f}")

    return out
