"""
Optimize a non-negative step function sequence to minimize
    2 * n * max(convolve(a, a)) / (sum(a))**2.

Strategy:
1. Use FFT-based convolution for fast evaluation.
2. Multiple optimization strategies: full multiplicative gradient descent with
   adaptive top-K focusing, anti-palindromic parameterization, and coordinate descent.
3. Multi-scale approach: try different sequence lengths, use given sequence as seed.
4. Final refinement with coordinate descent on individual entries.
"""

import numpy as np
import time


def _next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _fft_convolve(a, b=None):
    """Compute linear convolution a * (b or a) using FFT."""
    if b is None:
        b = a
    n = int(a.size)
    m = int(b.size)
    L = n + m - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    result = np.fft.irfft(fa * fb, n=N).real[:L]
    return np.maximum(result, 0.0)


def _fft_square(a):
    return _fft_convolve(a, None)


def _score(a, b=None):
    """Compute the objective value."""
    if b is None:
        b = _fft_square(a)
    n = a.size
    s = float(a.sum())
    if s < 1e-300:
        return float('inf')
    maxb = float(b.max())
    return float(2.0 * n * maxb / (s * s))


def _top_k_weights(b, k, eps=0.01):
    """Create weights concentrated on top-k convolution coefficients."""
    L = b.size
    if L == 0:
        return np.array([1.0]), 0.0
    maxb = float(b.max())
    if maxb < 1e-300:
        return np.full(L, 1.0 / L), maxb
    # Find top-k indices
    if k >= L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(b, -k)[-k:]
    weights = np.zeros(L)
    weights[idx] = 1.0 / k
    # Add small uniform background
    weights = (1.0 - eps) * weights + eps / L
    return weights, maxb


def _make_weights_adaptive(b, focus, top_frac):
    """Adaptive weight: blend between top-k and softmax focus."""
    L = b.size
    maxb = float(b.max())
    if maxb < 1e-300:
        return np.full(L, 1.0 / L), maxb
    
    k = max(1, int(L * top_frac))
    top_w, _ = _top_k_weights(b, k)
    
    # Softmax weights
    with np.errstate(over='ignore', under='ignore'):
        z = (maxb - b) / maxb
        p = np.exp(-z * focus)
    s = float(p.sum())
    if s < 1e-12:
        p = np.zeros(L)
        p[int(np.argmax(b))] = 1.0
    else:
        p = p / s
    
    # Blend
    blend = min(1.0, focus / 100.0)
    weights = (1.0 - blend) * top_w + blend * p
    weights /= weights.sum()
    return weights, maxb


def _init_anti_palin(n, rng, binary=False):
    """Initialize anti-palindromic sequence."""
    m = n // 2
    if m < 1:
        return np.ones(n) / n
    
    if binary:
        x = np.random.choice([0.0, 1.0], size=m, p=[0.5, 0.5], replace=True)
    else:
        x = rng.uniform(0.1, 0.9, size=m)
    
    a = np.empty(n)
    t = 1.0 / m
    a[:m] = t * x
    a[m:] = t * (1.0 - x[::-1])
    return a


def _init_random(n, rng, amp=2.0):
    z = rng.normal(0, amp, size=n)
    z -= z.max()
    a = np.exp(z)
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    return a

    """Full multiplicative gradient descent optimizer with adaptive focusing."""
    n = a0.size
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    
    b = _fft_square(a)
    score = _score(a, b)
    best_score = score
    best_a = a.copy()
    
    t0 = time.monotonic()
    eta = 1.0
    eta_min = 0.05
    focus = 5.0
    top_frac = 0.1
    no_improve = 0
    max_no_improve = 500
    it = 0
    
    while time.monotonic() - t0 < time_budget:
        # Update hyperparameters
        top_frac = max(0.01, 0.1 * np.exp(-it / 5000.0))
        focus = min(5.0 * (1.001 ** it), 500.0)
        
        # Compute weights
        weights, maxb = _make_weights_adaptive(b, focus, top_frac)
        
        # Gradient: d(max_b)/d(a_i) approximated by weighted sum
        # For our purposes, use the weighted convolution gradient
        grad = _fft_convolve(weights, a)[:n] + _fft_convolve(weights, a)[::-1][:n]
        # Actually for linear conv, the gradient of (w * a)_k wrt a_i is w_{k-i}
        # Let's use: grad_i = sum_k weights_k * (dba_k/dai)
        # For b = a*a, db_k/da_i = a_{k-i} + a_{k-i} (if i != k-i) or 2*a_{k-i}
        # More precisely: b_k = sum_j a_j * a_{k-j}, so db_k/da_i = a_{k-i} (+ a_i if 2i=k)
        # The gradient of sum_k w_k * b_k wrt a_i = sum_k w_k * a_{k-i} + (symmetry term)
        # This equals (w * a)_i + (w_rev * a)_i where w_rev[k] = w[2n-2-k]
        # Simplified: grad_i = sum_k w_k * a_{|k-i|} ... 
        # Let's just use the FFT approach from before
        N = _next_pow2(2 * n - 1 + n - 1)
        fw = np.fft.rfft(weights, n=N)
        fa = np.fft.rfft(a, n=N)
        grad = np.fft.irfft(fw * fa, n=N).real[:n] * 2.0
        
        # Normalize and clip gradient
        g_mean = float(np.dot(a, grad))
        gd = grad - g_mean
        maxg = float(np.max(np.abs(gd)))
        if maxg < 1e-18:
            break
        gd = gd / maxg
        
        # Multiplicative update
        a_new = a * np.exp(-eta * gd)
        a_new = np.maximum(a_new, 1e-300)
        ss = float(a_new.sum())
        a_new /= ss
        
        b_new = _fft_square(a_new)
        score_new = _score(a_new, b_new)
        
        # Accept/reject
        improved = score_new < best_score - 1e-10
        if improved or (score_new < score and score_new < best_score * 1.01):
            a = a_new
            b = b_new
            score = score_new
            if improved:
                best_score = score
                best_a = a.copy()
                no_improve = 0
                eta = min(eta * 1.08, 10.0)
            else:
                no_improve += 1
                eta = min(eta * 1.03, 10.0)
        else:
            no_improve += 1
            eta = max(eta * 0.85, eta_min)
            
            # Random kick if stuck

                kick = np.exp(rng.normal(0, 0.3, size=n))
                a = a * kick
                a = np.maximum(a, 1e-15)
                a /= a.sum()
                b = _fft_square(a)
                score = _score(a, b)
        
        if no_improve > max_no_improve:

            if score < best_score:

                a = best_a * np.exp(rng.normal(0, 0.5, size=n))
            else:
                # More aggressive restart
                fresh = _init_random(n, rng, amp=3.0)
                a = 0.7 * best_a + 0.3 * fresh
            a = np.maximum(a, 1e-15)
            a /= a.sum()
            b = _fft_square(a)
            score = _score(a, b)
            no_improve = 0
            max_no_improve = int(max_no_improve * 1.2)
            eta = 1.0
        
        it += 1
    
    return best_a, best_score

def _coordinate_descent(a, time_budget, rng, top_k=5):
    """Coordinate descent: optimize one (or few) entries at a time."""
    n = a.size
    a = a.copy().astype(np.float64)
    b = _fft_square(a)
    score = _score(a, b)
    best_score = score
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    stuck = 0
    
    while time.monotonic() - t0 < time_budget:
        # Pick which index to modify - focus on indices that contribute to top-k b values
        # The index i contributes to b_k for k in [i, i+n-1]
        # Find indices that appear in the top-k b positions
        top_idx = np.argsort(b)[-top_k * 2:]
        
        # For each top b position k, the contributing a indices are a[k-n+1:k+1]
        candidates = set()
        for k in top_idx:

            hi = min(n, k + 1)
            candidates.update(range(lo, hi))

            candidates = set(range(n))
        
        cand_list = list(candidates)
        # Pick a random subset to try
        n_try = min(20, len(cand_list))
        rng.shuffle(cand_list)
        cand_list = cand_list[:n_try]
        
        improved_any = False
        for i in cand_list:
            if time.monotonic() - t0 > time_budget:
                break
            
            # Try different values for a[i]
            # Current value
            old_val = a[i]
            
            # Try: reduce, set to near 0, increase slightly

                old_val * 0.8,
                old_val * 1.2,
                old_val * 1.5,
                1e-6,
                old_val + 0.01 / n,
            ]
            # Also try based on gradient direction
            b_full = b
            maxb = float(b_full.max())
            
            best_trial_val = old_val
            best_trial_score = score
            
            for val in candidates_vals:
                if val < 0:
                    continue
                a[i] = val
                # Recompute only affected b values (approximate: recompute all via FFT)
                # For speed, just recompute the full convolution
                b_trial = _fft_square(a)
                s_trial = float(a.sum())
                if s_trial < 1e-300:
                    a[i] = old_val
                    continue
                score_trial = 2.0 * n * float(b_trial.max()) / (s_trial * s_trial)
                if score_trial < best_trial_score - 1e-10:
                    best_trial_score = score_trial
                    best_trial_val = val
            
            if best_trial_val != old_val:
                a[i] = best_trial_val
                b = _fft_square(a)
                score = _score(a, b)
                improved_any = True
                if score < best_score - 1e-10:
                    best_score = score
                    best_a = a.copy()
                    stuck = 0
                else:
                    stuck += 1
            else:
                stuck += 1
        
        if not improved_any:
            stuck += 1
            if stuck > 50:
                # Random perturbation
                idx = rng.integers(0, n, size=10)
                for i in idx:
                    a[i] = max(1e-8, a[i] * np.exp(rng.normal(0, 1.0)))
                a /= a.sum()
                b = _fft_square(a)
                score = _score(a, b)
                stuck = 0
        
        it += 1
    
    return best_a, best_score


def _embed_and_optimize(a_small, n_target, rng):
    """Embed a smaller sequence into a larger one by interpolating."""
    n_small = a_small.size
    # Create a larger sequence by spreading out
    a_large = np.zeros(n_target)
    # Place the small sequence with spacing
    step = n_target // n_small
    for i in range(n_small):
        pos = i * step
        if pos < n_target:
            a_large[pos] = a_small[i] * (step / n_target) * n_small / n_target * n_target / step
    # Actually simpler: just use the small a as a starting point with noise
    # Better: create anti-palindromic from the small one
    return a_large


def _safe_evaluate(seq):
    try:
        ev = globals().get('evaluate_sequence')
        if ev is None or not callable(ev):
            return float('inf')
        return float(ev(seq))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800):
    """Main optimization driver."""
    total_budget = max(120.0, float(budget_s))
    t_start = time.monotonic()
    deadline = t_start + total_budget - 30.0
    
    print(f"Starting optimizer with budget={total_budget:.0f}s")
    
    best_a = None
    best_score = float('inf')
    
    def update_best(cand_a, cand_score, label=""):
        nonlocal best_a, best_score
        if cand_a is not None and np.isfinite(cand_score) and cand_score < best_score:
            best_a = cand_a.copy()
            best_score = cand_score
            print(f"  NEW BEST: {cand_score:.6f} (n={cand_a.size}) [{label}]")
    
    # Get given sequence
    given_a = None
    try:
        given = globals().get('height_sequence_1')
        if given is not None:
            g = np.asarray(given, dtype=np.float64).ravel()
            g = np.nan_to_num(g, nan=0.0, posinf=1000.0, neginf=0.0)
            g = np.clip(g, 0.0, 1000.0)
            s = float(g.sum())
            if s > 0.01:
                g = g / s
                gs = _score(g)
                given_a = g
                print(f"Given sequence: n={g.size}, score={gs:.6f}")
                if gs < best_score:
                    best_a = g.copy()
                    best_score = gs
    except Exception as e:
        print(f"Error reading given: {e}")
    
    rng = np.random.default_rng(seed)
    
    # Phase 1: Optimize given sequence (if available)
    elapsed = time.monotonic() - t_start
    if given_a is not None and deadline - time.monotonic() > 200:
        alloc = min(180.0, deadline - time.monotonic() - 300)
        if alloc > 30:
            print(f"\nPhase 1: Optimize given (n={given_a.size}, budget={alloc:.0f}s)")
            try:
                res_a, res_s = _optimize_full(given_a, alloc, rng, "given")
                update_best(res_a, res_s, "given-refined")
            except Exception as e:
                print(f"  Phase 1 error: {e}")
    
    # Phase 2: Anti-palindromic with various n
    for n_ap in [8192, 16384]:
        if deadline - time.monotonic() < 100:
            break
        alloc = min(200.0, deadline - time.monotonic() - 200)
        if alloc < 50:
            break
        print(f"\nPhase 2: Anti-palindromic n={n_ap}, budget={alloc:.0f}s")
        try:
            res_a, res_s = _optimize_anti_palin(n_ap, alloc, rng, f"ap{n_ap}")
            update_best(res_a, res_s, f"ap{n_ap}")
        except Exception as e:
            print(f"  Phase 2 error (n={n_ap}): {e}")
    
    # Phase 3: Full optimization from best found
    if deadline - time.monotonic() > 150:
        if best_a is None:
            best_a = _init_random(8192, rng)
        # Try with the best and also with a fresh random
        for trial in range(2):
            if deadline - time.monotonic() < 80:
                break
            alloc = min(250.0, deadline - time.monotonic() - 80)
            if alloc < 40:
                break
            if trial == 0:
                start = best_a.copy()
                label = "full-from-best"
            else:
                start = _init_random(best_a.size, rng, amp=3.0)
                label = "full-random"
            print(f"\nPhase 3: Full opt {label}, n={start.size}, budget={alloc:.0f}s")
            try:
                res_a, res_s = _optimize_full(start, alloc, rng, label)
                update_best(res_a, res_s, label)
            except Exception as e:
                print(f"  Phase 3 error: {e}")
    
    # Phase 4: Coordinate descent refinement
    if best_a is not None and deadline - time.monotonic() > 60:
        alloc = min(120.0, deadline - time.monotonic() - 30)
        if alloc > 20:
            print(f"\nPhase 4: Coordinate descent, n={best_a.size}, budget={alloc:.0f}s")
            try:
                res_a, res_s = _coordinate_descent(best_a, alloc, rng)
                update_best(res_a, res_s, "coord-descent")
            except Exception as e:
                print(f"  Phase 4 error: {e}")
    
    # Final: verify with evaluate_sequence
    if best_a is not None:
        final = np.nan_to_num(best_a, nan=0.0, posinf=1000.0, neginf=0.0)
        final = np.clip(final, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(1024) / 1024.0
            s = 1.0
        final = final / s
        final_seq = [float(x) for x in final]
        
        exact = _safe_evaluate(final_seq)
        print(f"\nFinal: internal={_score(final):.6f}, exact={exact:.6f}, n={final.size}")
        return final_seq
    else:
        fallback = np.ones(4096) / 4096.0
        return [float(x) for x in fallback]
