    v = np.where(np.isfinite(v), v, 0.0)

    s = v.sum()
    if s <= 1e-15:
        return np.full(v.size, 1.0/v.size)
    return v / s

def _project_simplex(v):
    v = _clean(v).copy()
    n = v.size
    if n == 0:
        return v
    x = np.sort(v)[::-1]
    css = np.cumsum(x)
    idx = np.arange(1, n+1, dtype=np.float64)
    cond = x > (css - 1) / idx
    if not np.any(cond):
        return np.full(n, 1.0/n)
    r = np.nonzero(cond)[0][-1]
    tau = (css[r] - 1) / (r + 1)
    out = np.maximum(v - tau, 0)
    s = out.sum()
    if s <= 1e-15:
        return np.full(n, 1.0/n)
    return out / s

    n = a.size
    L = _pow2(2*n - 1)
    A = np.fft.rfft(a, L)
    C = A * A
    return np.fft.irfft(C, L)[:2*n-1]

        return np.inf
    a = a / s
    c = _convolve(a)
    mx = c.max()
    return 2.0 * a.size * mx

    n = a0.size
    a = _project_simplex(a0.copy())
    _, c = _score_fast(a, L)
    score = 2.0 * n * c.max()
    best_a = a.copy()
    best_score = score

    if eps_init is None:
        eps_init = 0.3 * c.max()
    eps = eps_init

    lr = 1.0 / np.sqrt(n * 4.0)
    no_improve = 0
    max_no_improve = 200

    while time.monotonic() < t0 + time_limit:
        phase_start = time.monotonic()
        phase_time = min(30.0, time_limit * 0.15)

        for _ in range(50000):
            if time.monotonic() - phase_start > phase_time:
                break
            if time.monotonic() > t0 + time_limit:
                break

            val, grad, c, p = _softmax_value_grad(a, eps, L)
            gnorm = np.sqrt(np.mean(grad * grad))
            if gnorm < 1e-15:
                break

            # Try step with backtracking
            step = lr
            improved = False
            for _bt in range(12):
                a_new = _project_simplex(a - step * grad)
                val_new, _, c_new, _ = _softmax_value_grad(a_new, eps, L)
                if val_new < val - 1e-14:
                    improved = True
                    break
                step *= 0.4

            if improved:
                a = a_new
                score_now = 2.0 * n * c_new.max()
                if score_now < best_score - 1e-12:
                    best_score = score_now
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
                lr = max(1e-10, lr * 0.7)

            if no_improve > max_no_improve:
                break

        # Check actual score with exact computation
        sc, _ = _score_fast(a, L)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        sc, c_check = _score_fast(best_a, L)
        if sc < best_score:
            best_score = sc

        # Decrease eps
        eps_new = eps * decay
        if eps_new < eps_min or no_improve > max_no_improve * 2:
            break
        eps = eps_new

    return best_a, best_score


def _penalty_push(a0, L, time_limit, t0):
    """Penalty method: try to reduce the max autocorrelation below current level."""
    n = a0.size
    a = _project_simplex(a0.copy())
    sc, c = _score_fast(a, L)
    best_a = a.copy()
    best_score = sc
    target = c.max() * 0.999

    lr = 1.0 / (2.0 * np.sqrt(n))
    no_improve = 0

    while time.monotonic() < t0 + time_limit:
        for _ in range(20000):
            if time.monotonic() > t0 + time_limit:
                break

            val, grad, c = _penalty_value_grad(a, target, L)
            gnorm = np.sqrt(np.mean(grad * grad))
            if gnorm < 1e-15:
                break

            step = lr
            improved = False
            for _bt in range(10):
                a_new = _project_simplex(a - step * grad)
                val_new, _, _ = _penalty_value_grad(a_new, target, L)
                if val_new < val - 1e-14:
                    improved = True
                    break
                step *= 0.5

            if improved:
                a = a_new
                no_improve = 0
            else:
                no_improve += 1
                lr = max(1e-12, lr * 0.7)
                if no_improve > 100:
                    break

        # Check score
        sc, c = _score_fast(a, L)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()

        # Try to reduce target
        current_max = c.max()
        if current_max <= target * 1.001:
            # Feasible (approximately), reduce target
            target = current_max * 0.995
        else:
            # Not feasible, relax target slightly
            target = min(target * 1.005, current_max * 1.001)

        if no_improve > 300:
            break

    return best_a, best_score


def _multiplicative_weights(a0, L, time_limit, t0, eps=0.05):
    """Multiplicative weights (natural gradient) on the simplex."""
    n = a0.size
    a = np.maximum(a0.copy(), 1e-30)
    a = a / a.sum()
    sc, c = _score_fast(a, L)
    best_a = a.copy()
    best_score = sc

    lr = 0.5
    no_improve = 0

    while time.monotonic() < t0 + time_limit:
        val, grad, c, p = _softmax_value_grad(a, eps, L)

        # Multiplicative update
        log_ratio = -lr * grad / np.sqrt(n)
        log_ratio -= log_ratio.max()  # numerical stability
        weights = np.exp(log_ratio)
        a_new = a * weights
        s = a_new.sum()
        if s < 1e-30:
            a_new = best_a.copy()
        else:
            a_new = a_new / s
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()

        sc_new, _ = _score_fast(a_new, L)
        if sc_new < best_score - 1e-12:
            best_score = sc_new
            best_a = a_new.copy()
            no_improve = 0
            lr = min(2.0, lr * 1.02)
        elif sc_new < sc:
            no_improve = 0
            lr = min(2.0, lr * 1.01)
        else:
            no_improve += 1
            lr = max(0.01, lr * 0.9)
            if no_improve > 200:
                break

        a = a_new
        sc = sc_new

    return best_a, best_score


def _local_search(a0, L, time_limit, t0, seed=123):
    """Small random perturbations accepting only improvements."""
    n = a0.size
    best = _project_simplex(a0.copy())
    best_score, best_c = _score_fast(best, L)
    rng = np.random.default_rng(seed)

    sigma = 0.05
    last_improve = time.monotonic()

    while time.monotonic() < t0 + time_limit:
        cand = best.copy()
        # Perturb a random block
        w = int(np.exp(rng.uniform(np.log(3.0), np.log(max(5, n // 10)))))
        w = max(1, min(w, n // 4))
        s = int(rng.integers(0, n - w + 1))
        cand[s:s + w] *= np.exp(rng.normal(0.0, sigma, size=w))
        cand = _project_simplex(cand)

        sc, _ = _score_fast(cand, L)
        if sc < best_score - 1e-14:
            best_score = sc
            best = cand
            last_improve = time.monotonic()
            sigma = min(0.2, sigma * 1.05)
        else:
            if time.monotonic() - last_improve > 10.0:
                sigma = max(0.01, sigma * 0.8)
                if time.monotonic() - last_improve > 30.0:
                    break

    return best, best_score


def _refine_start(a0, L, time_budget, t0, seed=0):
    """Full refinement pipeline for one starting point."""
    a = _project_simplex(a0.copy())
    sc, _ = _score_fast(a, L)
    best_a = a.copy()
    best_score = sc

    # Phase 1: Softmax PGD (40% of budget)
    t1 = t0 + 0.40 * time_budget
    if time.monotonic() < t1:
        a, sc = _softmax_pgd(a, L, t1 - time.monotonic(), time.monotonic())
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # Phase 2: Multiplicative weights (20% of budget)
    t2 = t0 + 0.60 * time_budget
    if time.monotonic() < t2:
        a, sc = _multiplicative_weights(best_a, L, t2 - time.monotonic(), time.monotonic(),
                                        eps=0.02)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # Phase 3: Penalty push (25% of budget)
    t3 = t0 + 0.85 * time_budget
    if time.monotonic() < t3:
        a, sc = _penalty_push(best_a, L, t3 - time.monotonic(), time.monotonic())
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # Phase 4: Final softmax with very small eps (10% of budget)
    if time.monotonic() < t0 + time_budget:
        a, sc = _softmax_pgd(best_a, L, (t0 + time_budget) - time.monotonic(),
                             time.monotonic(), eps_init=1e-5, eps_min=1e-8, decay=0.7)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # Phase 5: Local search (remaining)
    if time.monotonic() < t0 + time_budget:
        a, sc = _local_search(best_a, L, (t0 + time_budget) - time.monotonic(),
                              time.monotonic(), seed=seed + 1)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Return the best sequence found before the time budget.
    Minimizes 2*n*max(convolve(a,a)) / (sum(a))^2.
    """
    t0 = time.monotonic()
    budget = max(0.0, float(budget_s) - 15.0)
    deadline = t0 + budget

    n = 3809
    try:
        h = height_sequence_1
        if isinstance(h, (list, np.ndarray)):
            arr = np.asarray(h, dtype=np.float64)
            if arr.ndim == 1 and arr.size == n:
                pass
    except Exception:
        pass

    L = _pow2(2 * n - 1)

    print(f"Starting search: n={n}, budget={budget:.1f}s, L={L}")

    # Generate and score starting points
    starts = _generate_starts(n)
    print(f"Generated {len(starts)} starting points")
    for i, (sc, _) in enumerate(starts[:5]):
        print(f"  Start {i}: score={sc:.6f}")

    if not starts:
        a = _power_seq(n, 0.5, 1.0)
        return [float(x) for x in a]

    # Allocate time to top 3 starting points
    top_k = min(3, len(starts))
    if top_k == 3:
        fracs = [0.50, 0.30, 0.20]
    elif top_k == 2:
        fracs = [0.60, 0.40]
    else:
        fracs = [1.0]

    global_best_a = starts[0][1].copy()
    global_best_score = starts[0][0]

    used_frac = 0.0
    for i in range(top_k):
        alloc_start = t0 + used_frac * budget
        alloc_end = min(deadline, t0 + (used_frac + fracs[i]) * budget)
        alloc_time = alloc_end - alloc_start

        if alloc_time < 10.0:
            used_frac += fracs[i]
            continue

        print(f"\nRefining start {i} (score={starts[i][0]:.6f}), "
              f"time={alloc_time:.1f}s, window=[{alloc_start - t0:.1f}, {alloc_end - t0:.1f}]s")

        a = starts[i][1].copy()
        a, sc = _refine_start(a, L, alloc_time, alloc_start, seed=int(seed) + 100 * (i + 1))

        print(f"  Result: score={sc:.6f}")

        if sc < global_best_score:
            global_best_score = sc
            global_best_a = a.copy()

        used_frac += fracs[i]

    # Final local search on global best
    if time.monotonic() < deadline - 5.0:
        remaining = deadline - 5.0 - time.monotonic()
        print(f"\nFinal local search, time={remaining:.1f}s, current best={global_best_score:.6f}")
        a, sc = _local_search(global_best_a, L, remaining, time.monotonic(),
                              seed=int(seed) + 9999)
        if sc < global_best_score:
            global_best_score = sc
            global_best_a = a.copy()

    global_best_a = _project_simplex(global_best_a)
    final_score, _ = _score_fast(global_best_a, L)
    print(f"\nFinal score: {final_score:.6f} (best tracked: {global_best_score:.6f})")

    if final_score < global_best_score:
        global_best_score = final_score

    # Validate
    out = [float(x) for x in global_best_a]
    try:
        val = evaluate_sequence(out)
        print(f"evaluate_sequence: {val:.6f}")
        if not np.isfinite(val):
            out = [float(x) for x in _power_seq(n, 0.5, 1.0)]
            print("Using fallback power sequence")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return out
