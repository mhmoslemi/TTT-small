import time
import numpy as np

def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()

def normalize(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s <= 0:
        return np.full_like(a, 1.0/len(a))
    return a / s

def fast_score(a):
    n = len(a)
    L = next_pow2(2*n-1)
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F*F, L)[:2*n-1]
    return 2.0 * n * c.max()

def score_and_grad(a, topk=20):
    """Returns (score, gradient) for the topk-smoothed max objective."""
    n = len(a)
    L = next_pow2(2*n-1)
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F*F, L)[:2*n-1]
    c = np.maximum(c, 0)  # numerical safety
    
    k = min(topk, 2*n-1)
    top_idx = np.argpartition(c, -k)[-k:]
    
    c_max = c[top_idx].max()
    c_min_top = c[top_idx].min()
    tau = max(1e-12, (c_max - c_min_top) / 5 + c_max * 1e-6)
    
    w = np.exp((c[top_idx] - c_max) / tau)
    w /= w.sum()
    
    W = np.zeros(2*n-1)
    W[top_idx] = w
    
    # g[i] = 2 * sum_m a[m] * W[i+m]
    # = 2 * cross_corr(a, W)[i]
    # = 2 * ifft(conj(fft(a, L2)) * fft(W, L2))[i]
    L2 = next_pow2(3*n-2)
    Fa = np.fft.fft(a, L2)
    FW = np.fft.fft(W, L2)

"""
Optimized search for non-negative coefficient sequences minimizing 2n*max(auto-conv)/sum^2.
Uses FFT-based top-k gradient with adaptive Adam in log-space, multiple scales, and restarts.
"""

import time
import numpy as np

try:
    from scipy.optimize import linprog
    HAVE_LINPROG = True
except Exception:
    HAVE_LINPROG = False


def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def normalize_pos(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        return np.full(a.shape, 1.0 / a.size)
    return a / s


def fast_score(a):
    """Score = 2n * max(auto-conv) assuming sum(a)=1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    L = next_pow2(2 * n - 1)
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0:
        return np.inf
    return 2.0 * n * cm


def compute_conv(a):
    """Compute auto-convolution via FFT."""
    n = a.size
    L = next_pow2(2 * n - 1)
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    return c


def score_and_grad(a, topk=30):
    """
    Score and gradient using top-k weighted subgradient.
    Returns (score, grad_a) where grad_a is w.r.t. a directly.
    """
    n = a.size
    c = compute_conv(a)
    c = np.maximum(c, 0.0)

    k = min(topk, 2 * n - 1)
    top_idx = np.argpartition(c, -k)[-k:]
    c_max = float(c[top_idx].max())
    c_min_top = float(c[top_idx].min())

    # Adaptive tau based on spread of top peaks
    spread = c_max - c_min_top
    tau = max(c_max * 1e-8, spread / 5.0 + c_max * 1e-8)

    w = np.exp((c[top_idx] - c_max) / tau)
    w /= w.sum()

    # Build weight vector W of length 2n-1
    W = np.zeros(2 * n - 1, dtype=np.float64)
    W[top_idx] = w

    # Gradient: g[i] = 2 * sum_m a[m] * W[i+m]
    # This is 2 * cross_correlation(a, W)[i]
    # = 2 * ifft(conj(fft(a)) * fft(W))[i] (linear, need padding)
    L2 = next_pow2(3 * n - 2)
    Fa = np.fft.rfft(a, L2)
    FW = np.fft.rfft(W, L2)

    # conj(fft(a)) * fft(W) in full FFT domain
    # For rfft: we need to reconstruct the full product carefully
    # Use full complex FFT for correctness
    Fa_c = np.fft.fft(a, L2)
    FW_c = np.fft.fft(W, L2)
    cross = np.fft.ifft(np.conj(Fa_c) * FW_c).real[:n]

    g = 2.0 * cross
    score = 2.0 * n * c_max
    return score, g


def adam_optimize(a_init, duration_s, rng, topk_start=80, topk_end=5,
                  lr0=0.03, nesterov=True):
    """
    Adam optimizer in log-space with adaptive topk (annealing) and Nesterov momentum.
    """
    a = normalize_pos(a_init)
    n = a.size

    z = np.log(np.maximum(a, 1e-300))
    z -= z.mean()
    z = np.clip(z, -50.0, 50.0)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    t = 0

    best_a = a.copy()
    best_score = fast_score(a)
    no_improve = 0
    total_steps_planned = int(duration_s * 80)  # rough estimate

    deadline = time.monotonic() + duration_s
    print(f"    adam start n={n} score={best_score:.6f} dur={duration_s:.0f}s")

    while time.monotonic() < deadline:
        t += 1
        progress = min(1.0, t / max(1, total_steps_planned))

        # Adaptive topk: start large, decrease
        topk = max(topk_end, int(topk_start * (1.0 - progress)))

        # Compute score and gradient
        score, grad_a = score_and_grad(a, topk=topk)

        # Project to log-space: grad_z = a * (grad_a - a @ grad_a)
        mean_g = float(a @ grad_a)
        grad_z = a * (grad_a - mean_g)

        # Gradient clipping
        max_g = float(np.abs(grad_z).max())
        clip_val = 0.05
        if max_g > clip_val:
            grad_z *= clip_val / max_g

        # Nesterov: look-ahead step for gradient
        if nesterov and t > 100:
            m_prev = m.copy()
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            # Look-ahead
            z_la = z - lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
            z_la -= z_la.mean()
            a_la = np.exp(z_la - z_la.max())
            a_la /= a_la.sum()
            _, grad_a_la = score_and_grad(a_la, topk=topk)
            mean_g_la = float(a_la @ grad_a_la)
            grad_z = a_la * (grad_a_la - mean_g_la)
            max_g = float(np.abs(grad_z).max())
            if max_g > clip_val:
                grad_z *= clip_val / max_g
        else:
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Cosine learning rate schedule
        lr = lr0 * 0.5 * (1.0 + np.cos(np.pi * progress))
        lr = max(lr, lr0 * 0.02)

        z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        z = np.where(np.isfinite(z), z, 0.0)
        z -= z.mean()
        z = np.clip(z, -50.0, 50.0)

        a = np.exp(z - z.max())
        a /= a.sum()

        # Track best
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Restart mechanism
        if no_improve > 8000:
            noise_scale = 0.2 + 0.3 * rng.random()
            z = np.log(np.maximum(best_a, 1e-300))
            z += rng.normal(0.0, noise_scale, n)
            z -= z.mean()
            z = np.clip(z, -50.0, 50.0)
            m *= 0.1
            v *= 0.1
            no_improve = 0

        if t % 20000 == 0:
            print(f"    adam t={t} score={score:.6f} best={best_score:.6f} topk={topk} lr={lr:.5f}")

    return best_a, best_score


def resize_seq(seq, m):
    """Resize sequence to length m via interpolation."""
    p = normalize_pos(seq)
    m = int(m)
    n = p.size
    if n == m:
        return p.copy()
    if n == 1:
        return np.full(m, 1.0 / m)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, m)
    out = np.interp(x_new, x_old, p)
    return normalize_pos(out)


def make_init(n, kind, rng):
    """Generate various initialization candidates."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        j = np.arange(n)
        a = np.cos(np.pi * j / (2.0 * n)) ** 2
        return normalize_pos(a)
    elif kind == 'cosine_half':
        j = np.arange(n)
        a = np.cos(np.pi * (j + 0.5) / (2.0 * n)) ** 2
        return normalize_pos(a)
    elif kind == 'gauss':
        j = np.arange(n)
        center = n / 2.0
        sigma = n * rng.uniform(0.2, 0.5)
        a = np.exp(-0.5 * ((j - center) / sigma) ** 2)
        return normalize_pos(a)
    elif kind == 'quadratic':
        j = np.arange(n, dtype=np.float64)
        a = (j + 1.0) * (n - j)
        return normalize_pos(a)
    elif kind == 'bilinear':
        j = np.arange(n, dtype=np.float64)
        a = 1.0 + rng.uniform(0.3, 1.0) * (2.0 * j / n - 1.0)
        return normalize_pos(np.maximum(a, 0.01))
    elif kind == 'w_shape':
        j = np.arange(n, dtype=np.float64)
        a = 1.0 + 0.5 * np.cos(4.0 * np.pi * j / n)
        return normalize_pos(np.maximum(a, 0.01))
    elif kind == 'm_shape':
        j = np.arange(n, dtype=np.float64)
        a = 1.0 - 0.4 * np.cos(2.0 * np.pi * j / n)
        return normalize_pos(np.maximum(a, 0.01))
    elif kind == 'smooth_rand':
        # Low-frequency random
        freqs = rng.integers(1, 8, size=6)
        phases = rng.uniform(0, 2 * np.pi, 6)
        amps = rng.uniform(0.1, 0.5, 6)
        j = np.arange(n, dtype=np.float64)
        a = np.ones(n)
        for f, ph, am in zip(freqs, phases, amps):
            a += am * np.cos(2 * np.pi * f * j / n + ph)
        return normalize_pos(np.maximum(a, 0.01))
    elif kind == 'asym_cos':
        j = np.arange(n, dtype=np.float64)
        # Asymmetric: more mass on one side
        alpha = rng.uniform(0.5, 1.5)
        a = np.ones(n) + alpha * np.sin(np.pi * j / n)
        return normalize_pos(np.maximum(a, 0.01))
    elif kind == 'asym_gauss':
        j = np.arange(n, dtype=np.float64)
        center = n * rng.uniform(0.3, 0.7)
        sigma = n * rng.uniform(0.15, 0.4)
        a = np.exp(-0.5 * ((j - center) / sigma) ** 2)
        return normalize_pos(a)
    else:
        return np.ones(n) / n


def lp_shape(parent, deadline, n_lp=1024):
    """LP-based Frank-Wolfe step to reshape the sequence."""
    if not HAVE_LINPROG:
        return None, np.inf

    f = resize_seq(parent, n_lp)
    best_p = f.copy()
    best_s = fast_score(f)
    no_improve = 0

    for it in range(3000):
        if time.monotonic() > deadline:
            break

        maxf = float(f.max())
        if maxf <= 1e-15:
            break

        # Build constraint matrix: (f * g)[k] <= maxf for all k
        # (f*g)[k] = sum_j f[j]*g[k-j] <= maxf
        # This is a linear constraint on g: A @ g <= maxf * ones
        n = n_lp
        r = (f / maxf)[::-1]
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for k in range(2 * n - 1):
            s_idx = max(0, k - n + 1)
            e_idx = min(n - 1, k)
            length = e_idx - s_idx + 1
            if length > 0:
                start = n - 1 - k + s_idx
                A[k, s_idx:e_idx + 1] = r[start:start + length]

        obj = -np.ones(n) - 1e-8 * np.random.uniform(size=n)
        try:
            res = linprog(obj, A_ub=A, b_ub=np.ones(2 * n - 1),
                          bounds=[(0.0, None)] * n, method='highs')
        except Exception:
            break

        if res is None or not getattr(res, 'success', False) or getattr(res, 'x', None) is None:
            break

        x = np.asarray(res.x, dtype=np.float64)
        x = np.maximum(x, 0.0)
        sx = float(x.sum())
        if sx <= 1e-12:
            break

        d = x / sx
        improved = False
        for t in (0.1, 0.3, 0.5, 0.7, 0.9):
            c = (1.0 - t) * f + t * d
            c = np.maximum(c, 0.0)
            sc = float(c.sum())
            if sc <= 0:
                continue
            c = c / sc
            s = fast_score(c)
            if np.isfinite(s) and s < best_s - 1e-10:
                best_s = s
                best_p = c.copy()
                f = c
                improved = True

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 5:
                break

    return best_p, best_s


def propose_candidate(seed=42, budget_s=2400.0, **kwargs):
    """
    Main search function. Tries multiple initializations and scales,
    refines with adaptive Adam, and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    total_budget = max(0.0, float(budget_s) - 30.0)
    deadline = start_time + total_budget

    best_p = None
    best_s = np.inf

    def update_best(p, s):
        nonlocal best_p, best_s
        if p is not None and np.isfinite(s) and (best_p is None or s < best_s):
            best_p = np.asarray(p, dtype=np.float64)
            best_s = float(s)

    # --- Phase 0: Try seed from previous construction ---
    try:
        hs = np.asarray(height_sequence_1, dtype=np.float64)
        hs = np.where(np.isfinite(hs), hs, 0.0)
        hs = np.maximum(hs, 0.0)
        if hs.size > 0 and hs.sum() > 1e-9:
            update_best(hs, fast_score(hs))
            print(f"seed from height_sequence_1: score={best_s:.6f}, n={hs.size}")
    except Exception:
        pass

    # --- Phase 1: Quick scan of analytical initializations at multiple n ---
    n_values = [4096, 8192, 16384]
    init_kinds = ['cosine', 'cosine_half', 'gauss', 'quadratic',
                  'bilinear', 'w_shape', 'm_shape', 'smooth_rand',
                  'asym_cos', 'asym_gauss', 'asym_gauss']

    phase1_deadline = start_time + total_budget * 0.08
    for n in n_values:
        if time.monotonic() > phase1_deadline:
            break
        for kind in init_kinds:
            if time.monotonic() > phase1_deadline:
                break
            a = make_init(n, kind, rng)
            s = fast_score(a)
            update_best(a, s)

    # Also try resizing the previous best
    if best_p is not None:
        for n in n_values:
            a = resize_seq(best_p, n)
            s = fast_score(a)
            update_best(a, s)

    print(f"Phase 1 done: best={best_s:.6f} at n={best_p.size if best_p is not None else 0}")

    # --- Phase 2: LP-based shaping on moderate size ---
    if HAVE_LINPROG and time.monotonic() < deadline - 600.0:
        lp_deadline = min(deadline, time.monotonic() + 120.0)
        lp_p, lp_s = lp_shape(best_p, lp_deadline, n_lp=1024)
        if lp_p is not None:
            for n in [4096, 8192, 16384]:
                if time.monotonic() > lp_deadline:
                    break
                a = resize_seq(lp_p, n)
                s = fast_score(a)
                update_best(a, s)
            print(f"Phase 2 (LP): best={best_s:.6f}")

    # --- Phase 3: Main Adam optimization at multiple scales ---
    # Allocate time: 40% to n=8192, 35% to n=16384, 25% to n=32768
    remaining = deadline - time.monotonic()

    # Determine best initial for each scale
    def get_best_for_n(n_target):
        candidates = []
        if best_p is not None:
            candidates.append(resize_seq(best_p, n_target))
        candidates.append(make_init(n_target, 'cosine', rng))
        candidates.append(make_init(n_target, 'gauss', rng))
        candidates.append(make_init(n_target, 'asym_gauss', rng))
        best_c = None
        best_cs = np.inf
        for c in candidates:
            s = fast_score(c)
            if s < best_cs:
                best_cs = s
                best_c = c
        return best_c, best_cs

    # Scale 1: n=8192
    rem1 = remaining * 0.35
    if rem1 > 120:
        a1, s1 = get_best_for_n(8192)
        dur = min(rem1, 500.0)
        p1, s1_opt = adam_optimize(a1, dur, rng, topk_start=100, topk_end=5, lr0=0.035)
        update_best(p1, s1_opt)
        print(f"Phase 3a (n=8192): best={best_s:.6f}")

    # Scale 2: n=16384
    rem2 = deadline - time.monotonic()
    rem2_alloc = min(rem2 * 0.45, 600.0)
    if rem2_alloc > 120:
        a2, s2 = get_best_for_n(16384)
        p2, s2_opt = adam_optimize(a2, rem2_alloc, rng, topk_start=80, topk_end=5, lr0=0.025)
        update_best(p2, s2_opt)
        print(f"Phase 3b (n=16384): best={best_s:.6f}")

    # Scale 3: n=32768
    rem3 = deadline - time.monotonic()
    rem3_alloc = min(rem3 * 0.5, 400.0)
    if rem3_alloc > 100:
        a3, s3 = get_best_for_n(32768)
        p3, s3_opt = adam_optimize(a3, rem3_alloc, rng, topk_start=60, topk_end=5, lr0=0.02)
        update_best(p3, s3_opt)
        print(f"Phase 3c (n=32768): best={best_s:.6f}")

    # --- Phase 4: Final polish on the best sequence ---
    rem4 = deadline - time.monotonic()
    if rem4 > 60 and best_p is not None:
        # Try a shorter refinement with very small topk on the best sequence
        n_best = best_p.size
        a4 = best_p.copy()
        dur = min(rem4 * 0.7, 200.0)
        p4, s4 = adam_optimize(a4, dur, rng, topk_start=10, topk_end=3, lr0=0.01)
        update_best(p4, s4)
        print(f"Phase 4 (polish n={n_best}): best={best_s:.6f}")

    # --- Phase 5: Validate with actual evaluator ---
    candidates = []
    if best_p is not None:
        candidates.append(best_p)

    # Also add a few fallbacks
    for n_fb in [1024, 2048]:
        candidates.append(np.ones(n_fb) / n_fb)

    if not hasattr(propose_candidate, '_seen_seeds'):
        propose_candidate._seen_seeds = set()

    final_seq = None
    final_score = np.inf

    for cand in candidates:
        lst = [float(x) for x in np.clip(np.asarray(cand, dtype=np.float64), 0.0, 1000.0)]
        if not lst:
            continue
        try:
            sc = evaluate_sequence(lst)
        except Exception:
            sc = np.inf
        if np.isfinite(sc) and sc < final_score:
            final_score = float(sc)
            final_seq = lst

    if final_seq is None:
        final_seq = [1.0 / 1024.0] * 1024

    print(f"FINAL: score={final_score:.6f}, n={len(final_seq)}")
    return final_seq
