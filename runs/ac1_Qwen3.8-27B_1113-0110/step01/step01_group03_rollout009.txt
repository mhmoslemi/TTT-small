```python
def sastao_lp(f, M, n):
    """Find g ≥ 0 maximizing sum(g) s.t. (f*g)_k ≤ M for all k."""
    # Decision variables: g_0, ..., g_{n-1}
    # Objective: maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Constraint matrix: (2n-1) × n
    # A[k, j] = f[k-j] if 0 ≤ k-j ≤ n-1, else 0
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_min = max(0, k - (n-1))
        j_max = min(n-1, k)
        for j in range(j_min, j_max+1):
            A[k, j] = f[k-j]
    
    b = np.full(2*n-1, M)
    
    # Bounds: g_j ≥ 0
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None
```

```python
import time
import scipy.sparse as sp
from scipy.optimize import linprog
```

```python
    if n <= 1:
        return a.copy()
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)

    S = a.sum()
    if S < 1e-10:

    return a / S * target

def build_conv_matrix_sparse(f, n):
    """Build sparse (2n-1) x n matrix where A[k,j] = f[k-j] if 0<=k-j<=n-1."""
    lengths = np.array([min(n, 2*n-1-j) for j in range(n)], dtype=int)

    # For speed, use repeat and concatenate
    col_idx = np.repeat(np.arange(n), lengths)

    offsets = np.concatenate([np.arange(L) for L in lengths]) if n > 10 else np.arange(0)

    # row_idx for column j is: j, j+1, ..., j+lengths[j]-1

    vals = np.concatenate(val_parts)
    
    return sp.csr_matrix((vals, (row_idx, col_idx)), shape=(2*n-1, n))

def solve_sastao_lp(f, M, n, timeout=30):
    """Solve SASTAO LP. Returns g or None."""
    try:
        A = build_conv_matrix_sparse(f, n)
        c_obj = -np.ones(n)
        b = np.full(2*n-1, M)
        bounds = [(0, None)] * n
        
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': timeout})

            g = res.x
            if g is not None and len(g) == n:
                return np.maximum(g, 0)
    except Exception as e:

    M = c.max()

    new_a = (1 - t) * a + t * g

    a = normalize(a, n)
    c = fft_self_conv(a)
    M = c.max()

    # ∂score/∂a_j = 2n/S^2 * ∂M/∂a_j - 2n*M/S^3 * 2

    grad = (2.0 * n / (S * S)) * grad_M - (4.0 * n * M / (S * S * S))
    
    momentum = beta * momentum - lr * grad
    a = a + momentum
    a = np.maximum(a, 0)
    a = normalize(a, n)
    return a, momentum, c, M
```

One approach: **iterative equalization**. 
1. Compute c = a★a
2. Identify lags where c_k is above target (say, target = mean(c))
3. For those lags, reduce the contributing a_j values
4. Redistribute to lags below target
5. Repeat

This is like the "water-filling" algorithm in information theory.

```python
def water_filling_step(a, target_ratio=1.0, n_steps=100):
    """Reduce peak by redistributing mass from high-convolution lags to low ones."""
    n = len(a)
    S = a.sum()
    target = target_ratio * S * S / (2*n-1)  # target for each lag
    
    for _ in range(n_steps):
        c = fft_self_conv(a)
        peak = c.max()
        if peak <= target * 1.001:
            break
        # Find the peak lag
        k_star = int(np.argmax(c))
        # Find contributing pairs
        j_min = max(0, k_star - (n-1))
        j_max = min(n-1, k_star)
        # Reduce a_j and a_{k_star-j} for j in [j_min, j_max]
        # where the product a_j * a_{k_star-j} is large
        products = np.array([a[j] * a[k_star-j] for j in range(j_min, j_max+1)])
        # Reduce the largest contributors
        delta = (c[k_star] - target) / (2 * products.sum() + 1e-10)
        for j in range(j_min, j_max+1):
            i = k_star - j
            red_j = min(a[j], delta * products[j] / a[j] if a[j] > 0 else 0)
            red_i = min(a[i], delta * products[j] / a[i] if a[i] > 0 else 0)
            a[j] -= red_j * 0.5
            a[i] -= red_i * 0.5
        a = np.maximum(a, 0)
        a = normalize(a, S)
    
    return a
```

```python
"""
Hybrid SASTAO-LP + Gradient + Diversification optimizer.
Minimizes 2n*max(convolve(a,a))/sum(a)^2 using:
1. SASTAO LP for structural peak reduction (LP-based direction finding)
2. Fast water-filling iterations for rapid local improvement
3. FFT gradient descent with momentum for fine refinement
4. Random diversification and multi-n exploration to escape local minima
"""
import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog


def fft_self_conv(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    if n <= 1:
        return a.copy()
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:2 * n - 1]


def fast_score(a):
    """Fast score using FFT."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_self_conv(a)
    M = c.max()
    return 2.0 * n * M / (S * S)


def normalize(a, target):
    """Normalize to sum=target, clip to non-negative."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-10:
        return np.full(len(a), target / len(a))
    return a / S * target


def build_conv_structure(n):
    """Precompute sparse structure for (2n-1)x n convolution matrix."""
    lengths = np.array([min(n, 2 * n - 1 - j) for j in range(n)], dtype=np.int64)
    col_idx = np.repeat(np.arange(n, dtype=np.int64), lengths)
    row_parts = []
    for j in range(n):
        row_parts.append(np.arange(j, j + lengths[j], dtype=np.int64))
    row_idx = np.concatenate(row_parts)
    val_parts = []
    for L in lengths:
        val_parts.append(np.arange(L))
    val_idx = np.concatenate(val_parts)
    return row_idx, col_idx, val_idx, lengths


def solve_sastao_lp(f, M, n, row_idx, col_idx, val_idx, timeout=25):
    """Solve SASTAO LP: max sum(g) s.t. A(f)@g <= M, g >= 0."""
    try:
        vals = f[val_idx]
        A = sp.csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n)
        b = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception as e:
        print(f"  LP error: {e}")
    return None


def water_filling_iterations(a, n_iter=200, S=None):
    """Fast iterative peak reduction via targeted mass redistribution."""
    n = len(a)
    if S is None:
        S = a.sum()
    a = a.copy()
    for it in range(n_iter):
        c = fft_self_conv(a)
        M = c.max()
        target = S * S / (2 * n - 1) * 0.95
        if M <= target:
            break
        k_star = int(np.argmax(c))
        j_min = max(0, k_star - (n - 1))
        j_max = min(n - 1, k_star)
        # Compute products
        js = np.arange(j_min, j_max + 1)
        is_idx = k_star - js
        products = a[js] * a[is_idx]
        total_prod = products.sum()
        if total_prod < 1e-10:
            break
        # Fraction to reduce from peak
        excess = (c[k_star] - target) / (2.0 * total_prod + 1e-10)
        excess = min(excess, 0.05)
        # Reduce contributing elements proportionally
        for idx_j, idx_i in zip(js, is_idx):
            a[idx_j] -= excess * a[idx_i] * 0.5
            a[idx_i] -= excess * a[idx_j if idx_j != idx_i else idx_i + 1] * 0.5
        a = np.maximum(a, 0)
        a = normalize(a, S)
    return a


def grad_descent_steps(a, n, n_steps, lr, beta, rng, S=None):
    """Multiple gradient descent steps with momentum."""
    if S is None:
        S = a.sum()
    momentum = np.zeros(n)
    for it in range(n_steps):
        a = normalize(a, S)
        c = fft_self_conv(a)
        M = c.max()
        # Use top-2 peaks
        thresh = M * 0.99
        peak_indices = np.where(c >= thresh)[0]
        if len(peak_indices) == 0:
            peak_indices = [int(np.argmax(c))]
        elif len(peak_indices) == 1:
            # Find second distinct peak
            c_masked = c.copy()
            c_masked[peak_indices[0]] = 0
            second = int(np.argmax(c_masked))
            peak_indices = [peak_indices[0], second]
        
        grad_M = np.zeros(n)
        for k in peak_indices:
            j_min = max(0, k - (n - 1))
            j_max = min(n - 1, k)
            grad_M[j_min:j_max + 1] += a[k - j_max:k - j_min + 1][::-1]
        
        S_norm = S
        grad = (2.0 * n / (S_norm ** 2)) * grad_M - (4.0 * n * M / (S_norm ** 3))
        grad /= max(1, len(peak_indices))
        
        momentum = beta * momentum - lr * grad
        a = np.maximum(a + momentum, 0)
        a = normalize(a, S)
    return a, momentum
```
