import time

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using smoothed mirror descent with continuation,
adaptive step sizes, diverse initializations (harmonic, m-sequence-inspired, multi-harmonic),
and local polishing. Explores multiple sequence lengths.

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _fft_convolve(a, b=None):
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    fb = np.fft.rfft(b, fft_size)
    ca = np.fft.irfft(fa * fb, fft_size)
    return ca[:size]


def _score(a):
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a = a / s
    c = _fft_convolve(a)
    return 2.0 * len(a) * float(np.max(c))


def _soft_max_grad(a, tau):
    n = len(a)
    c = _fft_convolve(a)
    c_div_tau = c / tau
    c_max = c_div_tau.max()
    exp_diff = np.exp(c_div_tau - c_max)
    w_sum = exp_diff.sum()
    weights = exp_diff / w_sum
    soft_score = tau * (c_max + np.log(w_sum))

    # grad[i] = 2 * sum_k weights[k] * a[k-i] (valid indices)
    a_rev = a[::-1]
    conv = _fft_convolve(weights, a_rev)
    grad = 2.0 * conv[n - 1: 2 * n - 1]

    return soft_score, grad, c

        t += 1

    return best_a, best_score


def _generate_initializations(rng):
    """Generate diverse initial sequences."""
    candidates = []

    # 1. Known best (from previous run)
    try:
        hs = np.array(height_sequence_1, dtype=np.float64)
        hs = np.clip(hs, 0, None)
        if hs.size > 100 and hs.sum() > 0.01:
            candidates.append(("known_best", hs.copy()))
    except (NameError, Exception):
        pass

    # 2. Single-harmonic with various parameters
    for n in [2000, 3000, 4000, 6000]:
        x = np.arange(n, dtype=np.float64)
        for c_amp in [0.3, 0.4, 0.5, 0.6]:
            for phase in [np.pi / 6, np.pi / 4, np.pi / 3, np.pi / 2]:
                for m in [1, 2]:
                    a = 1.0 + c_amp * np.cos(2 * np.pi * m * x / n + phase)
                    a = np.maximum(a, 0.01)
                    candidates.append((f"harm_n{n}_c{c_amp}_p{int(phase*4)}_m{m}", a))

    # 3. Multi-harmonic
    for n in [2000, 4000]:
        x = np.arange(n, dtype=np.float64)
        a = 1.0 + 0.35 * np.cos(2 * np.pi * x / n + np.pi / 4) + 0.15 * np.cos(4 * np.pi * x / n)
        a = np.maximum(a, 0.01)
        candidates.append((f"multi_n{n}", a))
        a2 = 1.0 + 0.3 * np.sin(2 * np.pi * x / n) + 0.1 * np.sin(6 * np.pi * x / n + 0.5)
        a2 = np.maximum(a2, 0.01)
        candidates.append((f"multi2_n{n}", a2))

    # 4. Decay (exponential)
    for n in [2000, 4000]:
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-1.2 * x / n) + 0.05
        candidates.append((f"decay_n{n}", a))

    # 5. Two-block smooth
    for n in [2000, 3000, 4000]:
        x = np.arange(n, dtype=np.float64)
        a = np.where(x < n * 0.4, 1.5, 0.5)
        # Smooth the transition
        trans = 200
        mask = (x >= n * 0.4 - trans) & (x < n * 0.4 + trans)
        t_val = (x[mask] - (n * 0.4 - trans)) / (2 * trans)
        t_val = 0.5 * (1 - np.cos(np.pi * t_val))
        a[mask] = 1.5 - t_val
        candidates.append((f"block_n{n}", a))

    # 6. Cosine-tapered window (like a raised cosine)
    for n in [2000, 3000, 5000]:
        x = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 + np.cos(2 * np.pi * x / n))
        a = a + 0.1
        candidates.append((f"coswin_n{n}", a))

    # 7. Anti-symmetric about center (sin with half-integer shift)
    for n in [2000, 3000, 4000]:
        x = np.arange(n, dtype=np.float64)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * (x + 0.5) / n)
        a = np.maximum(a, 0.01)
        candidates.append((f"antisym_n{n}", a))

    # 8. Random smooth (sum of random sinusoids)
    for n in [2000, 3000, 4000]:
        x = np.arange(n, dtype=np.float64)
        a = np.ones(n)
        for _ in range(3):
            freq = rng.integers(1, 10)
            amp = rng.uniform(0.1, 0.3)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * freq * x / n + phase)
        a = np.maximum(a, 0.01)
        candidates.append((f"rand_smooth_n{n}", a))

    return candidates


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Minimizes 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
    Uses smoothed mirror descent (soft-max gradient) with continuation,
    AdaGrad step size, diverse initializations, and local polishing.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 20.0

    best_overall = None
    best_overall_score = np.inf

    # Generate initializations
    candidates = _generate_initializations(rng)
    print(f"Total initial candidates: {len(candidates)}")

    # Quick evaluation and ranking
    scored = []
    for name, a0 in candidates:
        a0 = np.maximum(a0, 0)
        s = a0.sum()
        if s < 1e-15:
            continue
        a0 = a0 / s
        sc = _score(a0)
        scored.append((sc, name, a0))

    scored.sort(key=lambda x: x[0])
    print("Top initial scores:")
    for i, (sc, name, _) in enumerate(scored[:10]):
        print(f"  {i+1}. {name}: {sc:.6f} (n={len(_)})")

    # Phase 1: Optimize top candidates with smoothed mirror descent
    n_top = min(len(scored), 10)
    remaining = deadline - time.monotonic()

    # Allocate time: give more to better initializations
    weights = []
    for i in range(n_top):
        # Better initializations get more time
        w = max(1.0, float(n_top - i)) / (1.0 + scored[i][0])
        weights.append(w)
    total_w = sum(weights)
    times = [remaining * 0.7 * w / total_w for w in weights]

    for i in range(n_top):
        if time.monotonic() > deadline:
            break
        init_score, name, a0 = scored[i]
        t_alloc = min(times[i], max(10.0, deadline - time.monotonic()))

        print(f"\n[{i+1}/{n_top}] Optimizing '{name}' (n={len(a0)}, init={init_score:.6f}, time={t_alloc:.0f}s)")

        a_opt, s_opt = _smoothed_mirror_descent(a0, t_alloc, rng, verbose=True)
        eval_s = evaluate_sequence(a_opt.tolist())
        print(f"  Result: internal={s_opt:.6f}, eval={eval_s:.6f}")

        if np.isfinite(eval_s) and eval_s < best_overall_score:
            best_overall_score = eval_s
            best_overall = a_opt.copy()
            print(f"  *** NEW BEST: {eval_s:.6f} ***")

    # Phase 2: Polish the best solution
    remaining = deadline - time.monotonic()
    if best_overall is not None and remaining > 30:
        print(f"\n=== Polishing phase ({remaining:.0f}s) ===")

        # Multiple polishing attempts with different random seeds
        n_polish = min(3, int(remaining // 30))
        for p in range(n_polish):
            t_pol = min(remaining / (n_polish - p), max(10.0, deadline - time.monotonic()))
            a_pol, s_pol = _polish_local(best_overall, t_pol, rng)
            eval_p = evaluate_sequence(a_pol.tolist())
            print(f"  Polish {p+1}: {eval_p:.8f}")
            if np.isfinite(eval_p) and eval_p < best_overall_score:
                best_overall_score = eval_p
                best_overall = a_pol.copy()
                print(f"  *** NEW BEST: {eval_p:.8f} ***")

        # Try a fresh smoothed descent from the polished solution with different hyperparams
        remaining2 = deadline - time.monotonic()
        if remaining2 > 30:
            print(f"\n=== Fresh optimization from best ({remaining2:.0f}s) ===")
            a_fresh, s_fresh = _smoothed_mirror_descent(best_overall, remaining2 * 0.5, rng, verbose=True)
            eval_f = evaluate_sequence(a_fresh.tolist())
            print(f"  Fresh: {eval_f:.8f}")
            if np.isfinite(eval_f) and eval_f < best_overall_score:
                best_overall_score = eval_f
                best_overall = a_fresh.copy()
                print(f"  *** NEW BEST: {eval_f:.8f} ***")

            remaining3 = deadline - time.monotonic()
            if remaining3 > 10:
                a_pol2, s_pol2 = _polish_local(best_overall, remaining3, rng)
                eval_p2 = evaluate_sequence(a_pol2.tolist())
                print(f"  Final polish: {eval_p2:.8f}")
                if np.isfinite(eval_p2) and eval_p2 < best_overall_score:
                    best_overall_score = eval_p2
                    best_overall = a_pol2.copy()
                    print(f"  *** NEW BEST: {eval_p2:.8f} ***")

    # Final verification
    if best_overall is not None:
        final_score = evaluate_sequence(best_overall.tolist())
        elapsed = time.monotonic() - total_start
        print(f"\n=== FINAL: score={final_score:.8f}, n={len(best_overall)}, elapsed={elapsed:.1f}s ===")
        return [float(x) for x in best_overall]
    else:
        a = np.ones(1000, dtype=np.float64)
        print("Falling back to uniform sequence")
        return a.tolist()
