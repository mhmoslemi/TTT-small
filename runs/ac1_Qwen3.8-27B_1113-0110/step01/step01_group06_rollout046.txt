        a /= a.sum()

        a /= a.sum()

        # d/dz_i [f(a)] where a = softmax(z)
        # = sum_j (da_j/dz_i) * f'(a_j)
        # da_j/dz_i = a_j*(delta_ij - a_i)
        # So grad_z_i = a_i * (f'(a_i) - sum_j a_j f'(a_j))
        f_grad = g  # this is d f / d a
        mean_fg = float(np.dot(a, f_grad))
        return a * (f_grad - mean_fg)

    tau = 80.0
    t0 = time.monotonic()
    result = None
    tau_schedules = [50.0, 100.0, 200.0, 500.0]
    for ts in tau_schedules:
        if time.monotonic() - t0 > tlimit * 0.8:
            break
        tau = ts
        obj_fn = lambda z, _t=tau: _objective_with_tau(z, n, _t)
        grad_fn = lambda z, _t=tau: _gradient_with_tau(z, n, _t)
        try:
            res = _scipy_minimize(
                obj_fn, a_init, method='L-BFGS-B', jac=grad_fn,
                options={'maxiter': 200, 'maxfun': 500, 'gtol': 1e-6,
                         'iprint': 0, 'maxls': 20}
            )
            if res.success or res.fun < objective(a_init, tau_schedules[0]):
                a_res = np.exp(res.x - res.x.max())
                a_res /= a_res.sum()
                a_init = res.x  # continue from here
                result = a_res
        except Exception:
            pass

    if result is not None:
        score, _, _ = _score(result)
        return result, score
    return a0, _score(a0)[0]


def _objective_with_tau(z, n, tau):
    a = np.exp(z - z.max())
    a /= a.sum()
    b = _fft_conv(a)
    mb = float(b.max())
    b_norm = b / max(mb, 1e-10)
    with np.errstate(over='ignore'):
        vals = np.exp(tau * (b_norm - 1.0))
    return float(np.log(np.sum(vals)) / tau)


def _gradient_with_tau(z, n, tau):
    a = np.exp(z - z.max())
    a /= a.sum()
    b = _fft_conv(a)
    mb = float(b.max())
    b_norm = b / max(mb, 1e-10)
    with np.errstate(over='ignore'):
        p = np.exp(tau * (b_norm - 1.0))
    ps = float(p.sum())
    if ps < 1e-30:
        return np.zeros(n)
    p /= ps
    g = _fft_grad_b(a, p)
    mean_fg = float(np.dot(a, g))
    return a * (g - mean_fg)


def _greedy_init(n, rng):
    """Greedy anti-palindromic binary initialization."""
    m = n // 2
    counts = np.zeros(2 * n - 1, dtype=np.int64)
    selected = np.empty(m, dtype=np.int64)
    order = np.arange(m)
    rng.shuffle(order)
    t = 0
    for idx_val in order:
        idx = int(idx_val)
        l = idx
        r = n - 1 - idx
        if t == 0:
            y = l
        else:
            s = selected[:t]
            metric_l = max(int(np.max(counts[s + l])) + 2, int(counts[2 * l]) + 1) if t > 0 else 0
            metric_r = max(int(np.max(counts[s + r])) + 2, int(counts[2 * r]) + 1) if t > 0 else 0
            y = l if metric_l <= metric_r else r
        if t > 0:
            counts[selected[:t] + y] += 2
        counts[2 * y] += 1
        selected[t] = y
        t += 1
    x = np.full(m, 0.1, dtype=np.float64)
    x[sel_left] = 0.9 if hasattr(x, 'dtype') else 0.9
    # Simpler: just random
    x = rng.uniform(0.2, 0.8, m)
    return x


def _safe_evaluate(seq):
    ev = globals().get("evaluate_sequence")
    if ev is None or not callable(ev):
        return float('inf')
    try:
        return float(ev(seq))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Searches for a sequence minimizing
    2*n*max(convolve(a,a))/(sum(a))^2.

    Strategy:
    1. Optimize the given sequence (if available) with momentum mirror descent.
    2. Try anti-palindromic optimization at n=16384 from diverse starts.
    3. Polish the best with scipy L-BFGS-B.
    4. Return the best sequence found.
    """
    seed = int(seed)
    total_budget = min(float(budget_s), 2350.0)
    t_start = time.monotonic()
    deadline = t_start + total_budget - 40.0  # 40s safety margin

    best_a = None
    best_score = float('inf')
    master_rng = np.random.default_rng(seed)

    # Load given sequence
    given_a = None
    try:
        given = globals().get("height_sequence_1")
        if given is not None:
            g = np.asarray(given, dtype=np.float64).ravel()
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g = np.clip(g, 0.0, 1000.0)
            s = float(g.sum())
            if s > 0.01 and g.size >= 100:
                gn = g / s
                sc, _, _ = _score(gn)
                if np.isfinite(sc):
                    best_score = sc
                    best_a = gn.copy()
                    given_a = gn
                    print(f"Given sequence: n={gn.size}, score={sc:.6f}")
    except Exception as e:
        print(f"Could not load given sequence: {e}")

    if given_a is None:
        # Fallback: create a reasonable starting point
        n0 = 8192
        given_a = np.ones(n0) / n0
        best_a = given_a.copy()
        best_score, _, _ = _score(given_a)
        print(f"Fallback uniform: n={n0}, score={best_score:.6f}")

    # === Phase 1: Full-vector optimization of given sequence ===
    if time.monotonic() < deadline - 600:
        p1_time = min(700.0, deadline - time.monotonic() - 600)
        if p1_time > 60:
            print(f"\n=== Phase 1: Full-vector opt on n={given_a.size} ({p1_time:.0f}s) ===")
            t0 = time.monotonic()

            # Try multiple randomizations of the starting point
            for trial in range(3):
                if time.monotonic() >= deadline - 500:
                    break
                alloc = min(p1_time / 3.0, deadline - time.monotonic() - 500)
                if alloc < 30:
                    break
                if trial == 0:
                    a_init = given_a.copy()
                elif trial == 1:
                    a_init = given_a * np.exp(master_rng.normal(0, 0.5, given_a.size))
                    a_init = _normalize(a_init)
                else:
                    # Smooth perturbation
                    a_init = given_a.copy()
                    a_init = a_init * (1.0 + 0.3 * np.sin(np.pi * np.arange(a_init.size) / a_init.size))
                    a_init = _normalize(a_init)

                rng1 = np.random.default_rng((seed * 7 + trial * 13) % (2**32))
                cand, sc = _opt_full(a_init, alloc, rng1, tau0=5.0,
                                     label=f"p1-{trial}")
                if sc < best_score:
                    best_score = sc
                    best_a = cand.copy()
                    print(f"  Phase 1 trial {trial}: score={sc:.6f} (new best!)")
                else:
                    print(f"  Phase 1 trial {trial}: score={sc:.6f} (best={best_score:.6f})")

            print(f"Phase 1 done: best={best_score:.6f}, time={time.monotonic()-t0:.1f}s")

    # === Phase 2: Anti-palindromic at n=16384 ===
    if time.monotonic() < deadline - 400:
        p2_time = min(600.0, deadline - time.monotonic() - 400)
        if p2_time > 100:
            print(f"\n=== Phase 2: Anti-palindromic n=16384 ({p2_time:.0f}s) ===")
            t0 = time.monotonic()
            n2 = 16384
            m2 = n2 // 2

            # Initialize from best_a if possible
            if best_a is not None and best_a.size >= 100:
                # Project to anti-palindromic
                if best_a.size % 2 == 0:
                    bm = best_a.size // 2
                    bt = 1.0 / bm
                    x_init = best_a[:bm] / bt
                    # Resample to m2
                    x_init = np.interp(
                        np.linspace(0, bm - 1, m2),
                        np.arange(bm),
                        x_init
                    )
                else:
                    x_init = master_rng.uniform(0.3, 0.7, m2)
            else:
                x_init = master_rng.uniform(0.3, 0.7, m2)
            x_init = np.clip(x_init, 0.01, 0.99)

            rng2 = np.random.default_rng((seed * 11 + 3) % (2**32))
            cand2, sc2 = _opt_pair(x_init, p2_time * 0.6, rng2, tau0=10.0, label="p2-main")
            if sc2 < best_score:
                best_score = sc2
                best_a = cand2.copy()
                print(f"  Phase 2 main: score={sc2:.6f} (new best!)")
            else:
                print(f"  Phase 2 main: score={sc2:.6f} (best={best_score:.6f})")

            # Second trial with different init
            if time.monotonic() < deadline - 200:
                x_init2 = master_rng.uniform(0.1, 0.9, m2)
                # Make it anti-palindromic binary-ish
                x_init2 = (x_init2 < 0.4) * 0.1 + (x_init2 >= 0.4) * 0.9
                x_init2 = np.clip(x_init2 + master_rng.normal(0, 0.05, m2), 0.01, 0.99)
                alloc2 = min(p2_time * 0.35, deadline - time.monotonic() - 200)
                if alloc2 > 30:
                    cand2b, sc2b = _opt_pair(x_init2, alloc2, rng2, tau0=20.0, label="p2-bin")
                    if sc2b < best_score:
                        best_score = sc2b
                        best_a = cand2b.copy()
                        print(f"  Phase 2 binary: score={sc2b:.6f} (new best!)")
                    else:
                        print(f"  Phase 2 binary: score={sc2b:.6f} (best={best_score:.6f})")

            print(f"Phase 2 done: best={best_score:.6f}, time={time.monotonic()-t0:.1f}s")

    # === Phase 3: Scipy polishing ===
    if time.monotonic() < deadline - 60 and best_a is not None:
        p3_time = min(300.0, deadline - time.monotonic() - 60)
        if p3_time > 30:
            # Only polish if n is manageable for scipy
            if best_a.size <= 20000:
                print(f"\n=== Phase 3: Scipy polish n={best_a.size} ({p3_time:.0f}s) ===")
                t0 = time.monotonic()
                rng3 = np.random.default_rng((seed * 13 + 7) % (2**32))
                try:
                    polished, sc_pol = _scipy_polish(best_a, p3_time, rng3)
                    if sc_pol < best_score:
                        best_score = sc_pol
                        best_a = polished.copy()
                        print(f"  scipy polish: score={sc_pol:.6f} (new best!)")
                    else:
                        print(f"  scipy polish: score={sc_pol:.6f} (best={best_score:.6f})")
                except Exception as e:
                    print(f"  scipy polish failed: {e}")
                print(f"Phase 3 done: time={time.monotonic()-t0:.1f}s")
            else:
                print(f"  Skipping scipy polish (n={best_a.size} too large)")

    # === Phase 4: If time remains, do another round of full-vector on best ===
    remaining = deadline - time.monotonic()
    if remaining > 120 and best_a is not None:
        print(f"\n=== Phase 4: Extended full-vector opt ({remaining:.0f}s) ===")
        t0 = time.monotonic()
        rng4 = np.random.default_rng((seed * 17 + 11) % (2**32))
        # Slight perturbation
        a4 = best_a * np.exp(rng4.normal(0, 0.1, best_a.size))
        a4 = _normalize(a4)
        cand4, sc4 = _opt_full(a4, remaining - 30, rng4, tau0=20.0, label="p4")
        if sc4 < best_score:
            best_score = sc4
            best_a = cand4.copy()
            print(f"  Phase 4: score={sc4:.6f} (new best!)")
        else:
            print(f"  Phase 4: score={sc4:.6f} (best={best_score:.6f})")

    # === Final: Ensure valid output ===
    if best_a is None:
        best_a = np.ones(1024) / 1024.0

    best_a = np.nan_to_num(best_a, nan=0.0, posinf=0.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    ss = float(best_a.sum())
    if ss < 0.01:
        best_a = np.ones(1024) / 1024.0
        ss = 1.0
    else:
        best_a = best_a / ss

    final_seq = [float(x) for x in best_a]
    print(f"\n=== FINAL: n={best_a.size}, internal score={best_score:.6f} ===")

    # Verify with evaluate_sequence if available
    if best_a.size <= 12000 and time.monotonic() < deadline - 5:
        exact = _safe_evaluate(final_seq)
        if np.isfinite(exact):
            print(f"  Exact evaluation: {exact:.6f}")
            if exact < best_score:
                best_score = exact
        else:
            print("  Exact evaluation unavailable (inf or error)")

    return final_seq
