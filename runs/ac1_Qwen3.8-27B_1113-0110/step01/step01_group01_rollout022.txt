def next_pow2(m: int) -> int:
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def normalize_positive(a, floor: float = 0.0) -> np.ndarray:
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, float(floor), None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def convolve_seq(a, N=None) -> np.ndarray:
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if N is None:
        N = next_pow2(2 * n - 1)
    else:
        need = next_pow2(2 * n - 1)
        if int(N) < int(need):
            N = need
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]


def compute_score(a) -> float:
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.inf
    a = a / s
    N = next_pow2(2 * n - 1)
    c = convolve_seq(a, N)
    return float(2.0 * n * float(np.max(c)))


def softmax_weights(c, cmax, tau) -> np.ndarray:
    L = c.size
    if tau <= 1e-12:
        idx = np.nonzero(c >= cmax - 1e-10)[0]
        w = np.zeros(L, dtype=np.float64)
        if idx.size == 0:
            w[np.argmax(c)] = 1.0
        else:
            w[idx] = 1.0 / float(idx.size)
        return w
    z = (c - float(cmax)) / float(tau)
    z = np.clip(z, -60.0, 0.0)
    ez = np.exp(z)
    s = float(ez.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros(L, dtype=np.float64)
        w[np.argmax(c)] = 1.0
        return w
    return ez / s


def gradient_from_weights(w, a, N_eff) -> np.ndarray:
    """Compute gradient of sum_k w[k]*conv[k] w.r.t. a.
    
    grad[j] = 2 * sum_i w[i+j] * a[i] for j=0..n-1
    Computed via FFT: ifft(FFT(w) * conj(FFT(a)))
    """
    n = a.size
    W = np.fft.rfft(w, N_eff)
    A = np.fft.rfft(a, N_eff)
    corr = np.fft.irfft(W * np.conj(A), N_eff)
    return 2.0 * corr[:n]


def project_simplex(v) -> np.ndarray:
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v
    mu = np.zeros(n, dtype=np.float64)
    entries = np.sort(v)[::-1]
    cumulative = np.cumsum(entries)
    rho = np.nonzero(entries * np.arange(1, n + 1) > (cumulative - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    rho = rho[-1] + 1
    theta = (float(cumulative[rho - 1]) - 1.0) / float(rho)
    proj = np.maximum(v - theta, 0.0)
    s = proj.sum()
    if s > 0:
        proj = proj / s
    else:
        proj = np.ones(n, dtype=np.float64) / float(n)
    return proj


def make_uniform(n: int) -> np.ndarray:

def make_gaussian_sequence(n: int, mu: float, sigma: float) -> np.ndarray:

def make_sine_sequence(n: int, alphas) -> np.ndarray:
    n = max(2, int(n))
    x = (np.arange(n, dtype=np.float64) - 0.5 * (n - 1)) / max(1.0, 0.5 * n)
    base = np.ones(n, dtype=np.float64)
    for m, alpha in enumerate(alphas, start=1):
        base = base + float(alpha) * np.sin(np.pi * float(m) * x)
    return normalize_positive(base, floor=1e-12)


def make_power_sequence(n: int, p: float) -> np.ndarray:
    """a[i] proportional to (i+1)^p * (n-i)^p for symmetric shapes."""
    n = max(2, int(n))
    i = np.arange(n, dtype=np.float64)
    left = i + 1.0
    right = n - i
    vals = (left ** p) * (right ** p)
    return normalize_positive(vals, floor=1e-12)


def resample_seq(a, n: int) -> np.ndarray:
    a = normalize_positive(a, floor=0.0)
    old_n = a.size
    n = max(2, int(n))
    if old_n == 1:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    if old_n == n:
        return a.astype(np.float64, copy=True)
    old_x = np.linspace(0.0, 1.0, old_n, dtype=np.float64)
    new_x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    b = np.interp(new_x, old_x, a)
    return normalize_positive(b, floor=1e-12)

def build_logparam_basis(n: int) -> np.ndarray:
    n = max(8, int(n))
    i = np.arange(n, dtype=np.float64)
    x = (i - 0.5 * (n - 1)) / max(1.0, 0.5 * n)
    cols = []
    for m in range(1, 7):
        cols.append(np.sin(np.pi * float(m) * x))
    cols.append(x.copy())
    cols.append(x * x - np.mean(x * x))
    cols.append(x * x * x - np.mean(x * x * x))
    cols.append(x ** 4 - np.mean(x ** 4))
    cols.append(x ** 5 - np.mean(x ** 5))
    for m in range(1, 5):
        t = 2.0 * np.pi * float(m) * i / float(n)
        cols.append(np.sin(t) - np.mean(np.sin(t)))
        cols.append(np.cos(t) - np.mean(np.cos(t)))
    B = np.column_stack(cols)
    B -= B.mean(axis=0)
    norms = np.linalg.norm(B, axis=0)
    norms[norms <= 1e-14] = 1.0
    B = B / norms
    return B


def softmax_seq(z) -> np.ndarray:
    z = np.asarray(z, dtype=np.float64).ravel()
    z = z - float(np.max(z))
    z = np.clip(z, -60.0, 0.0)
    e = np.exp(z)
    e = np.maximum(e, 1e-300)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(z.size, 1.0 / float(z.size), dtype=np.float64)
    return e / s


def theta_value(theta, n, B, tau, N) -> float:
    theta = np.asarray(theta, dtype=np.float64).ravel()
    a = softmax_seq(B @ theta)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    cmax = float(np.max(c))
    if cmax <= 0.0 or not np.isfinite(cmax):
        return 0.0
    tau_eff = max(float(tau), 1e-12)
    z = (c - cmax) / tau_eff
    z = np.clip(z, -50.0, 0.0)
    s = float(np.exp(z).sum())
    if s <= 0.0:
        return cmax
    return float(2.0 * n * (cmax + tau_eff * np.log(s)))


def theta_grad(theta, n, B, tau, N) -> np.ndarray:
    theta = np.asarray(theta, dtype=np.float64).ravel()
    a = softmax_seq(B @ theta)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    cmax = float(np.max(c))
    D = B.shape[1]
    if cmax <= 0.0 or not np.isfinite(cmax):
        return np.zeros(D, dtype=np.float64)
    tau_eff = max(float(tau), 1e-12)
    z = (c - cmax) / tau_eff
    z = np.clip(z, -50.0, 0.0)
    ez = np.exp(z)
    s = float(ez.sum())
    if s <= 0.0:
        return np.zeros(D, dtype=np.float64)
    w = ez / s
    N_eff = next_pow2(3 * n - 2)
    ga = gradient_from_weights(w, a, N_eff)
    total = float(np.dot(a, ga))
    aB = a @ B
    agaB = (a * ga) @ B
    gtheta = agaB - total * aB
    return (2.0 * n) * gtheta

def optimize_euclidean(a, deadline, seed=0):

        if it % 500 == 0:

    return best_a, best_score


def optimize_active_set(a, deadline, seed=0):
    """NEW: Direct active-set equalization.
    
    Try to directly reduce the gap between the maximum convolution coefficient
    and other high coefficients by solving for a direction that equalizes them.
    """
    a = normalize_positive(a, floor=1e-15)
    n = a.size
    N = next_pow2(2 * n - 1)
    N_eff = next_pow2(3 * n - 2)
    rng = np.random.default_rng(seed)

    best_a = a.copy()
    best_score = compute_score(a)
    eta = 0.08
    it = 0

    while time.monotonic() < deadline:
        c = convolve_seq(a, N)
        cmax = float(np.max(c))
        score = float(2.0 * n * cmax)
        if np.isfinite(score) and score < best_score * (1.0 - 1e-12):
            best_score = score
            best_a = a.copy()

        if cmax <= 0.0 or not np.isfinite(cmax):
            break

        # Find top-K coefficients
        K = min(n, max(10, n // 16))
        top_k = np.argsort(c)[::-1][:K]
        top_vals = c[top_k]
        spread = float(top_vals[0] - top_vals[-1])

        if spread < 1e-10:
            break  # Already evenly spread

        # Use inverse-proportional weights: higher coefficients get more weight
        # to force them down
        w = np.zeros(2 * n - 1, dtype=np.float64)
        weights_top = (top_vals - float(top_vals[-1]) + 1e-10)
        weights_top = np.clip(weights_top, 1e-10, None)
        w_top = weights_top / float(weights_top.sum())
        w[top_k] = w_top

        # Also add some weight to all coefficients for overall flattening
        w = 0.7 * w + 0.3 * softmax_weights(c, cmax, 0.05 * cmax)

        g = gradient_from_weights(w, a, N_eff)
        g = g - g.mean()
        scale = float(np.std(g))
        if scale <= 1e-15:
