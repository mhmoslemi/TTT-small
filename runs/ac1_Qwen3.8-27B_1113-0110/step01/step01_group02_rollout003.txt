    return 1 << (n - 1).bit_length()

    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)

    fft_size = _next_pow2(3 * n)
    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a, fft_size)
    cross = np.fft.irfft(P * np.conj(A), fft_size)

    p = np.exp(bc)

    grad = 2.0 * beta * _fft_cross_corr(p, a)

def _next_pow2(n):
    """Return the next power of 2 >= n."""
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _fft_convolve(a):
    """Fast linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)

    """Compute cross-correlation (p convolve a_flipped)[i] = sum_j p[i+j]*a[j] for i=0..n-1."""
    n = len(a)

    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a, fft_size)

    it = 0
    no_improve = 0
    gamma = 1.0

    while time.monotonic() < deadline:
        # Compute spectrum
        A = np.fft.rfft(a, fft_size)
        A_mag2 = A.real ** 2 + A.imag ** 2
        A_phase = np.arctan2(A.imag, A.real)

        # Autocorrelation
        c = np.fft.irfft(A_mag2, fft_size)[:2 * n - 1]
        score = 2.0 * n * float(np.max(c))

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            a = best_a * np.exp(rng.normal(0, 0.08, n))
            a = np.maximum(a, 1e-20)
            a /= a.sum()
            no_improve = 0
            gamma = 1.0
            it += 1
            continue

        # Shape the spectrum: reduce frequency components that contribute to peak
        # Modify |A|^2 to push energy away from frequencies that cause peak autocorrelation
        c_norm = c - c.mean()
        c_neg = np.maximum(-c_norm, 0)  # where autocorrelation is below mean
        weight = 1.0 + gamma * c_neg / (c_neg.max() + 1e-15)
        new_mag2 = A_mag2 * weight
        new_mag2 = np.maximum(new_mag2, 1e-30)

        # Reconstruct with same phase
        new_A = np.sqrt(new_mag2) * np.exp(1j * A_phase)
        new_a = np.fft.irfft(new_A, fft_size)[:n]
        new_a = np.maximum(new_a, 0)
        s = new_a.sum()
        if s > 1e-15:
            new_a /= s
        else:
            new_a = a.copy()

        # Blend
        a = 0.5 * a + 0.5 * new_a
        a = np.maximum(a, 1e-20)
        a /= a.sum()

        # Adapt gamma
        if no_improve > 100:
            gamma *= 1.1
        else:
            gamma = max(gamma * 0.99, 0.1)

        it += 1
        if it % 1000 == 0:
            print(f"      [{label}] it={it}, score={score:.6f}, best={best_score:.6f}, gamma={gamma:.2f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Minimizes 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

    Algorithm:
    1. Multiple initializations (sinusoidal, spectral, polynomial) for n=2000.
    2. Main optimization via soft-max (log-sum-exp) gradient with mirror descent
       and beta annealing (uses ALL convolution lags, not just argmax).
    3. Spectral shaping as alternative optimization.
    4. Final polishing with pure subgradient descent.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 25.0

    best_overall = None
    best_overall_score = np.inf

    n = 2000
    x = np.arange(n, dtype=np.float64)
    candidates = []

    # Load previous best
    try:
        hs = np.array(height_sequence_1, dtype=np.float64)
        hs = np.clip(hs, 0, None)
        if len(hs) == n and hs.sum() > 1:
            candidates.append(("prev_best", hs.copy()))
    except (NameError, Exception):
        pass

    # Sinusoidal with various parameters
    for amp, phase in [(0.4, np.pi / 4), (0.45, np.pi / 5), (0.35, np.pi / 3),
                       (0.5, 0.0), (0.3, np.pi / 2), (0.55, np.pi / 6)]:
        a = 1.0 + amp * np.sin(2 * np.pi * x / n + phase)
        a = np.maximum(a, 0.01)
        candidates.append((f"sin_a{amp:.2f}_p{phase:.2f}", a))

    # Multi-harmonic
    a = 1.0 + 0.25 * np.sin(2 * np.pi * x / n) + 0.15 * np.cos(4 * np.pi * x / n + 0.5)
    a = np.maximum(a, 0.01)
    candidates.append(("multi_harm1", a))

    a = 1.0 + 0.3 * np.sin(2 * np.pi * x / n + np.pi / 4) + 0.1 * np.sin(6 * np.pi * x / n)
    a = np.maximum(a, 0.01)
    candidates.append(("multi_harm2", a))

    # Exponential decay
    a = np.exp(-0.9 * x / n) + 0.08
    candidates.append(("decay", a))

    # CCDF-like (high at one end, tail at other)
    t = x / (n - 1)
    a = 1.0 / (0.3 + t)
    candidates.append(("ccdf", a))

    # Quadratic
    a = (1.0 - t) ** 1.5 + 0.05
    candidates.append(("quad_decay", a))

    # "U-shaped" (high at both ends)
    a = 0.5 + 0.5 * (2 * t - 1) ** 2
    candidates.append(("u_shaped", a))

    # Random smooth initializations
    for trial in range(4):
        f1 = rng.integers(1, 5)
        f2 = rng.integers(2, 10)
        p1 = rng.uniform(0, 2 * np.pi)
        p2 = rng.uniform(0, 2 * np.pi)
        a1 = rng.uniform(0.15, 0.5)
        a2 = rng.uniform(0.05, 0.25)
        a = 1.0 + a1 * np.sin(2 * np.pi * f1 * x / n + p1) + a2 * np.cos(2 * np.pi * f2 * x / n + p2)
        a = np.maximum(a, 0.01)
        candidates.append((f"rand_{trial}", a))

    # Evaluate initial scores
    scored = []
    for name, a0 in candidates:
        a0 = np.maximum(a0, 0)
        s = a0.sum()
        if s < 1e-15:
            continue
        a0n = a0 / s
        c0 = _fft_convolve(a0n)
        init_score = 2.0 * n * float(np.max(c0))
        scored.append((init_score, name, a0n))

    scored.sort(key=lambda z: z[0])
    print(f"Candidates: {len(scored)}")
    for i, (sc, name, _) in enumerate(scored[:8]):
        print(f"  {i+1}. {name}: {sc:.6f}")

    # Allocate time to top candidates
    n_top = min(6, len(scored))
    avail = deadline - time.monotonic()
    # Top 1: 30%, Top 2: 20%, Top 3: 15%, Top 4: 12%, Top 5: 10%, Top 6: 8%
    fracs = [0.30, 0.20, 0.15, 0.12, 0.10, 0.08][:n_top]
    # Normalize
    fw = sum(fracs)
    fracs = [f / fw for f in fracs]

    for i in range(n_top):
        if time.monotonic() > deadline - 60:
            break
        init_score, name, a0 = scored[i]
        t_alloc = avail * fracs[i]
        t_alloc = min(t_alloc, deadline - time.monotonic())
        if t_alloc < 20:
            continue

        print(f"\n=== Optimizing '{name}' (n={n}, init={init_score:.6f}, time={t_alloc:.0f}s) ===")

        # Phase 1: Soft-max mirror descent (main)
        t_soft = t_alloc * 0.5
        a1, s1 = _optimize_soft_mw(a0, t_soft, rng, beta_init=5.0, beta_max=600.0, label=f"soft_{name}")
        eval1 = evaluate_sequence(a1.tolist())
        print(f"  Soft-MW: internal={s1:.6f}, eval={eval1:.6f}")
        if np.isfinite(eval1) and eval1 < best_overall_score:
            best_overall_score = eval1
            best_overall = a1.copy()
            print(f"  *** NEW BEST: {eval1:.6f} ***")

        # Phase 2: Spectral shaping from soft-mw result
        remaining = deadline - time.monotonic()
        t_spec = min(t_alloc * 0.25, remaining)
        if t_spec > 15:
            start_spec = a1 if s1 < init_score else a0
            a2, s2 = _optimize_spectral(start_spec, t_spec, rng, label=f"spec_{name}")
            eval2 = evaluate_sequence(a2.tolist())
            print(f"  Spectral: internal={s2:.6f}, eval={eval2:.6f}")
            if np.isfinite(eval2) and eval2 < best_overall_score:
                best_overall_score = eval2
                best_overall = a2.copy()
                print(f"  *** NEW BEST: {eval2:.6f} ***")

        # Phase 3: Subgradient polishing
        remaining = deadline - time.monotonic()
        t_sub = min(t_alloc * 0.25, remaining)
        if t_sub > 10:
            best_local = a1 if (np.isfinite(s1) and s1 < eval2 if np.isfinite(eval2) else True) else a0
            # Use the best local result
            if np.isfinite(eval2) and eval2 < s1:
                best_local = a2
            a3, s3 = _optimize_subgrad(best_local, t_sub, rng, label=f"sub_{name}")
            eval3 = evaluate_sequence(a3.tolist())
            print(f"  Subgrad: internal={s3:.6f}, eval={eval3:.6f}")
            if np.isfinite(eval3) and eval3 < best_overall_score:
                best_overall_score = eval3
                best_overall = a3.copy()
                print(f"  *** NEW BEST: {eval3:.6f} ***")

    # Final polishing rounds on the global best
    remaining = deadline - time.monotonic()
    if best_overall is not None and remaining > 40:
        print(f"\n=== Final polishing ({remaining:.0f}s remaining) ===")

        # High-beta soft-max
        t1 = remaining * 0.4
        a_f, s_f = _optimize_soft_mw(best_overall, t1, rng, beta_init=100.0, beta_max=3000.0, label="final_soft")
        eval_f = evaluate_sequence(a_f.tolist())
        print(f"  Final soft: {eval_f:.6f}")
        if np.isfinite(eval_f) and eval_f < best_overall_score:
            best_overall_score = eval_f
            best_overall = a_f.copy()

        # Subgradient
        remaining2 = deadline - time.monotonic()
        if remaining2 > 15:
            a_f2, s_f2 = _optimize_subgrad(best_overall, remaining2, rng, label="final_sub")
            eval_f2 = evaluate_sequence(a_f2.tolist())
            print(f"  Final sub: {eval_f2:.6f}")
            if np.isfinite(eval_f2) and eval_f2 < best_overall_score:
                best_overall_score = eval_f2
                best_overall = a_f2.copy()

        # One more soft from the subgrad result
        remaining3 = deadline - time.monotonic()
        if remaining3 > 20:
            a_f3, s_f3 = _optimize_soft_mw(best_overall, remaining3, rng, beta_init=200.0, beta_max=5000.0, label="final_soft2")
            eval_f3 = evaluate_sequence(a_f3.tolist())
            print(f"  Final soft2: {eval_f3:.6f}")
            if np.isfinite(eval_f3) and eval_f3 < best_overall_score:
                best_overall_score = eval_f3
                best_overall = a_f3.copy()

    if best_overall is not None:
        final_score = evaluate_sequence(best_overall.tolist())
        elapsed = time.monotonic() - total_start
        print(f"\nFinal result: score={final_score:.6f}, n={len(best_overall)}, elapsed={elapsed:.1f}s")
        return [float(x) for x in best_overall]
    else:
        print("Falling back to uniform")
        a = np.ones(1000, dtype=np.float64)
        return a.tolist()
