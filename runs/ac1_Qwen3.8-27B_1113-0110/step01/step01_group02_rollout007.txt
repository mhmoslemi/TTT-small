def _fft_convolve(a):
    """Linear autocorrelation (convolution with itself) via FFT."""
    n = len(a)
    size = 2 * n - 1
    # Use next power of 2 for FFT
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]

def _objective_and_grad(x, tau):
    """Compute logsumexp-smoothed objective and gradient w.r.t. x (log-param).
    a = softmax(x), minimize (1/tau)*log(sum exp(tau*conv(a,a)))"""
    # Softmax
    x_shifted = x - x.max()
    ex = np.exp(x_shifted)
    s = ex.sum()
    a = ex / s
    n = len(a)
    
    # Convolution
    c = _fft_convolve(a)
    
    # Log-sum-exp
    tc = tau * c
    tc_max = tc.max()
    lse = (tc_max + np.log(np.exp(tc - tc_max).sum())) / tau
    obj = 2.0 * n * lse
    
    # Weights for each lag
    p = np.exp(tc - tc_max) / np.exp(tc - tc_max).sum()
    
    # Gradient of lse w.r.t. a[j]: sum_k p[k] * a[k-j] (valid range)
    # This is cross-correlation of p and a
    # grad_a[j] = sum_k p[k] * a[k-j] where 0 <= k-j < n
    # = sum_l p[l+j] * a[l] where 0 <= l < n and 0 <= l+j < 2n-1

    # Use FFT-based approach

    fft_size = len(c)  # 2n-1

    a_fft = np.fft.rfft(a, 2*fft_size)

    # np.correlate(p, a, 'full')[j + len(a) - 1] should give this... let me verify

        hi = min(n - 1, fft_size - 1 - j)

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using log-sum-exp smoothed L-BFGS optimization,
subgradient polishing, and LP-based direction finding. Tries diverse analytical
initializations (sinusoidal modulations, power decays, asymmetric shapes) across
multiple sequence lengths to find the global minimum.
"""
import numpy as np
import time
import scipy.optimize
import scipy.signal
import scipy.special


def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)
    return ca[:size]


def _true_score(a):
    c = _fft_convolve(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * len(a) * float(c.max()) / (s * s)


def _logsumexp_objective(x, tau):
    n = len(x)
    x_shifted = x - x.max()
    ex = np.exp(x_shifted)
    s = ex.sum()
    a = ex / s
    c = _fft_convolve(a)
    tc = tau * c
    tc_max = tc.max()
    lse = tc_max + np.log(np.exp(tc - tc_max).sum())
    return 2.0 * n * lse / tau


def _logsumexp_grad(x, tau):
    n = len(x)
    x_shifted = x - x.max()
    ex = np.exp(x_shifted)
    s = ex.sum()
    a = ex / s
    c = _fft_convolve(a)
    tc = tau * c
    tc_max = tc.max()
    p = np.exp(tc - tc_max)
    p_sum = p.sum()
    p = p / p_sum
    lse = tc_max + np.log(p_sum)
    obj = 2.0 * n * lse / tau

    grad_a = corr[n - 1:n - 1 + n].copy()

    w = np.dot(grad_a, a)
    grad_x = a * (grad_a - w)

    return obj, grad_x

        'maxls': 50,
        'ftol': 1e-14,
        'gtol': 1e-10,
        'gtol_crossover': 1e-10,
        'maxcor': 30,
    }

    # Adaptive tau: start high, reduce if L-BFGS struggles
    taus = [tau, tau * 0.5, tau * 2, tau * 0.25]
    for ti, t_now in enumerate(taus):
        if time.monotonic() > deadline:
            break
        if t_now < 100:
            continue
        x_now = np.log(np.maximum(best_a, 1e-30) + 1e-30)
        x_now = x_now - x_now.mean()

        t_start = time.monotonic()
        try:
            res = scipy.optimize.minimize(
                fun_and_grad, x_now,
                method='L-BFGS-B', jac=True,
                options=opts
            )
        except Exception:
            break

        elapsed = time.monotonic() - t_start
        if res.success or res.message == 'CONVERGENCE: NORM_OF_PROJECTED_GRADIENT_<=_PGTOL':
            pass

        # Convert back and evaluate true score
        a_new = np.exp(res.x - res.x.max())
        a_new = a_new / a_new.sum()
        sc = _true_score(a_new)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_new.copy()
            # Continue with same x for next tau
            x0 = res.x - res.x.mean()
        elif ti == 0:
            # If no improvement with first tau, try different taus
            x0 = res.x - res.x.mean()

    # Verify with evaluate_sequence
    ev = evaluate_sequence(best_a.tolist())
    if np.isfinite(ev) and ev < best_score:
        best_score = ev
    elif ev < best_score:
        best_score = ev
    # Use the evaluate_sequence score as ground truth
    if np.isfinite(ev):
        best_score = ev

    return best_a, best_score


def _optimize_subgrad(a0, time_budget, rng):
    """Subgradient descent with momentum and adaptive step size."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, np.inf
    a = a / s

    best_score = _true_score(a)
    best_a = a.copy()
    eta = 0.12
    no_improve = 0
    momentum = np.zeros(n)
    mu = 0.75
    deadline = time.monotonic() + time_budget
    it = 0

    while time.monotonic() < deadline:
        c = _fft_convolve(a)
        k_max = int(np.argmax(c))
        score = 2.0 * n * c[k_max]

        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 600:
            a = best_a * np.exp(rng.normal(0, 0.1, n))
            a = np.maximum(a, 1e-20)
            a = a / a.sum()
            no_improve = 0
            eta = min(eta * 1.5, 0.3)
            momentum = np.zeros(n)
            it += 1
            continue

        # Subgradient
        g = np.zeros(n)
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        if lo <= hi:
            indices = k_max - np.arange(lo, hi + 1)
            g[lo:hi + 1] = a[indices]

        g_max = g.max()
        if g_max < 1e-18:
            break
        g = g / g_max
        momentum = mu * momentum + (1.0 - mu) * g

        a = a * np.exp(-eta * momentum)
        a = np.maximum(a, 1e-20)
        s = a.sum()
        a = a / s

        if no_improve > 300:
            eta *= 0.97
        elif no_improve < 20:
            eta = min(eta * 1.01, 0.35)

        # Perturb periodically
        if it % 500 == 250 and rng.random() < 0.3:
            blk = max(1, n // 30)
            st = int(rng.integers(0, n - blk))
            a[st:st + blk] *= np.exp(rng.normal(0, 0.2, blk))
            a = np.maximum(a, 1e-20)
            a = a / a.sum()

        it += 1

    ev = evaluate_sequence(best_a.tolist())
    if np.isfinite(ev):
        best_score = min(best_score, ev)
    return best_a, best_score


def _optimize_lp_refine(a0, time_budget, rng, n_iters=30):
    """LP-based direction finding: maximize sum(b) s.t. conv(a,b) <= M, b>=0."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, np.inf
    a = a / s

    best_score = _true_score(a)
    best_a = a.copy()
    deadline = time.monotonic() + time_budget

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = c.max() + 1e-12  # slight slack for feasibility
        score = 2.0 * n * M

        # Build constraint matrix: conv(a, b)[k] <= M for all k
        # conv(a,b)[k] = sum_j a[j]*b[k-j] (valid)
        # For each k (0..2n-2): sum_j A[k,j]*b[j] <= M
        # A[k,j] = a[k-j] if 0<=k-j<n else 0
        # So A[k,j] = a[k-j] for max(0,k-n+1) <= j <= min(n-1, k)

        # This is (2n-1) x n matrix. Build it sparsely.
        from scipy.sparse import csr_matrix
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                j_range = np.arange(j_lo, j_hi + 1)
                a_vals = a[k - j_range]
                rows.extend([k] * len(j_range))
                cols.extend(j_range.tolist())
                vals.extend(a_vals.tolist())

        A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        b_ub = np.full(2 * n - 1, M)

        # Maximize sum(b) => minimize -sum(b)
        c_obj = -np.ones(n)
        bounds = [(0, None)] * n

        try:
            res = scipy.optimize.linprog(
                c_obj, A_ub=A_ub, b_ub=b_ub,
                bounds=bounds, method='highs'
            )
        except Exception:
            break

        if not res.success:
            break

        b_dir = res.x  # should be non-negative
        if b_dir.sum() < 1e-15:
            break

        # Update: a <- (1-t)*a + t*b_dir, renormalize
        # b_dir has sum = some value; we want to blend
        b_sum = b_dir.sum()
        # Scale b_dir so that sum matches a's sum (=1)
        b_scaled = b_dir / b_sum

        for t in [0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * b_scaled
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            sc_try = _true_score(a_try)
            if sc_try < score - 1e-10:
                a = a_try
                break

        sc = _true_score(a)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a.copy()

    ev = evaluate_sequence(best_a.tolist())
    if np.isfinite(ev):
        best_score = min(best_score, ev)
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Minimizes 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
    Uses log-sum-exp smoothed L-BFGS, subgradient descent, and LP refinement
    from diverse analytical initializations across multiple lengths.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 30.0  # safety margin

    best_overall = None
    best_overall_score = np.inf

    # --- Collect initializations ---
    inits = []

    # Provided sequence
    try:
        hs = np.array(height_sequence_1, dtype=np.float64)
        hs = np.clip(hs, 0, None)
        if hs.size > 100 and hs.sum() > 0.01:
            hsn = hs / hs.sum()
            inits.append(("provided", hsn))
    except NameError:
        pass
    except Exception:
        pass

    # Diverse analytical initializations
    n_list = [1200, 1500, 2000, 2500, 3000, 4000, 5000, 8000]

    for n in n_list:
        x = np.arange(n, dtype=np.float64)

        # Sinusoidal modulations (multiple frequencies, amplitudes, phases)
        for freq in [1, 2, 3]:
            for A in [0.3, 0.5, 0.7, 0.9]:
                for phi in [0.0, np.pi / 4, np.pi / 3, np.pi / 2, 2 * np.pi / 3]:
                    a = 1.0 + A * np.sin(2 * np.pi * freq * x / n + phi)
                    a = np.maximum(a, 0.01)
                    a = a / a.sum()
                    inits.append((f"sin_{n}_{freq}_{A}_{int(np.degrees(phi))}", a))

        # Power-law decay (asymmetric)
        for p in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
            a = (1.0 - x / n) ** p + 0.005
            a = a / a.sum()
            inits.append((f"powd_{n}_{p}", a))

        # Exponential decay
        for alpha in [0.5, 1.0, 2.0, 4.0, 8.0]:
            a = np.exp(-alpha * x / n) + 0.005
            a = a / a.sum()
            inits.append((f"expd_{n}_{alpha}", a))

        # Generalized triangles
        t = np.abs(2.0 * x / (n - 1) - 1.0)
        for p in [0.5, 1.0, 2.0, 3.0]:
            a = (1.0 - t) ** p + 0.002
            a = a / a.sum()
            inits.append((f"tri_{n}_{p}", a))

        # Cosine-modulated triangle (key structure)
        t = np.abs(2.0 * x / (n - 1) - 1.0)
        base = (1.0 - t) ** 1.5 + 0.01
        for A in [0.2, 0.4, 0.6]:
            for freq in [1, 2]:
                a = base * (1.0 + A * np.cos(2 * np.pi * freq * x / n))
                a = np.maximum(a, 0.001)
                a = a / a.sum()
                inits.append((f"trisin_{n}_{freq}_{A}", a))

        # Two-hump structure
        for c1 in [0.25, 0.3]:
            for w in [0.1, 0.15]:
                a = (np.exp(-((x - n * c1) / (n * w)) ** 2) +
                     np.exp(-((x - n * (1 - c1)) / (n * w)) ** 2)) + 0.005
                a = a / a.sum()
                inits.append((f"twin_{n}_{c1}_{w}", a))

        # S-curve / sigmoid shape
        for mid in [0.3, 0.4, 0.5]:
            for sharp in [3, 5, 10]:
                a = 1.0 / (1.0 + np.exp(-sharp * (x / n - mid))) + 0.005
                a = a / a.sum()
                inits.append((f"sig_{n}_{mid}_{sharp}", a))

    # Random non-negative sequences
    for n in [2000, 3000]:
        for _ in range(3):
            a = rng.random(n) ** 2
            a = a / a.sum()
            inits.append((f"rand_{n}", a))

    # --- Quick evaluation and sorting ---
    scored = []
    for name, a in inits:
        sc = _true_score(a)
        if np.isfinite(sc):
            scored.append((sc, name, a))

    scored.sort(key=lambda x: x[0])
    print(f"Total initializations: {len(inits)}")
    print("Top 15 candidates:")
    for i, (sc, name, a) in enumerate(scored[:15]):
        print(f"  {i + 1}. {name}: {sc:.6f} (n={len(a)})")

    # --- Optimize top candidates ---
    # Time allocation: more time for better candidates
    n_opt = min(len(scored), 25)
    time_per_lbfgs = 30.0  # seconds per L-BFGS run
    time_per_subgrad = 20.0
    time_per_lp = 20.0

    # Geometric time weights
    weights = np.array([max(1.0, n_opt - i) for i in range(n_opt)])
    total_w = weights.sum()
    allocs = weights / total_w * (budget_s - 120.0)  # reserve 120s for polishing

    for i in range(n_opt):
        if time.monotonic() > deadline - 120.0:
            break

        init_sc, name, a0 = scored[i]
        t_alloc = min(allocs[i], max(10.0, deadline - 120.0 - time.monotonic()))

        if t_alloc < 15:

        t_lbfgs = t_alloc * 0.4
        t_subgrad = t_alloc * 0.35
        t_lp = t_alloc * 0.25

        current_best_sc = init_sc

        if t1 > 3:
            a1, sc1 = _optimize_lbfgs(current_best_a, t1, tau=20000, rng=rng)
            ev1 = evaluate_sequence(a1.tolist())
