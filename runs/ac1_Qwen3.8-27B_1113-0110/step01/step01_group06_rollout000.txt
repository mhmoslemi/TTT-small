Optimize non-negative step function to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses full-vector multiplicative gradient descent with adaptive focus,
multi-start exploration, and targeted peak reduction.
"""

import numpy as np
import time

def _pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()

def _fft_square(a):
    n = int(a.size)
    L = 2 * n - 1
    N = _pow2(L)
    fa = np.fft.rfft(a, n=N)
    b = np.fft.irfft(fa * fa, n=N).real[:L]
    return np.maximum(b, 0.0)

def _fft_grad(weights, a):
    n = int(a.size)
    L = int(weights.size)
    N = _pow2(L + n - 1)
    fw = np.fft.rfft(weights, n=N)
    fa = np.fft.rfft(a, n=N)
    c = np.fft.irfft(fw * np.conj(fa), n=N)[:n]
    return 2.0 * c

def _score(a, b=None):
    if b is None:
        b = _fft_square(a)
    s = float(a.sum())
    if s < 1e-300:
        return float('inf'), b
    return float(2.0 * a.size * float(b.max()) / (s * s)), b

def _normalize(a):
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s > 0:
        a /= s
    return a

    z = (maxb - b) / maxb
    p = np.exp(-z * focus)
    s = float(p.sum())
    if s < 1e-12:
        p = np.zeros(L)
        p[int(np.argmax(b))] = 1.0
    else:
        p /= s
    return p

def _optimize(a0, time_limit, rng, focus_start=5.0, max_focus=2000.0, 
              verbose=False, label=""):
    """Full-vector multiplicative gradient optimization."""
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    a = np.maximum(a, 1e-300)
    a /= a.sum()
    n = int(a.size)
    
    b = _fft_square(a)
    score, _ = _score(a, b)
    best_score = score
    best_a = a.copy()
    
    focus = focus_start
    eta = 1.0
    no_improve = 0
    max_no_improve = 500
    resets = 0
    max_resets = 50
    t0 = time.monotonic()
    it = 0
    focus_inc = 1.03
    
    while time.monotonic() - t0 < time_limit:
        # Compute weights and gradient
        p = _make_weights(b, focus)
        grad = _fft_grad(p, a)
        
        # Center the gradient (for multiplicative updates)
        gd = grad - float(np.dot(a, grad))
        maxg = float(np.max(np.abs(gd)))
        if maxg < 1e-300:
            break
        
        gd = gd / maxg
        
        # Multiplicative update
        a_new = a * np.exp(-eta * gd)
        a_new = np.maximum(a_new, 1e-300)
        a_new /= a_new.sum()
        
        b_new = _fft_square(a_new)
        score_new, _ = _score(a_new, b_new)
        
        # Accept/reject
        if score_new < best_score - 1e-15:
            best_score = score_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
            b = b_new
            score = score_new
            eta = min(eta * 1.03, 50.0)
        elif score_new < score - 1e-15:
            no_improve = max(no_improve - 1, 0)  # slight credit
            a = a_new
            b = b_new
            score = score_new
            eta = min(eta * 1.01, 50.0)
        else:
            no_improve += 1
            eta = max(eta * 0.7, 0.01)
        
        # Check for stuck state
        if no_improve > max_no_improve:
            resets += 1
            if resets > max_resets:
                break
            no_improve = 0
            max_no_improve = int(max_no_improve * 1.5)
            # Perturb from best
            a = best_a * np.exp(rng.normal(0.0, 0.3, size=n))
            a = np.maximum(a, 1e-300)
            a /= a.sum()
            b = _fft_square(a)
            score = float(a.size * b.max() / (a.sum()**2)) * 2
            focus = float(rng.choice([2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0]))
            eta = 1.0
        
        focus = min(focus * focus_inc, max_focus)
        
        if verbose and it % 5000 == 0:
            print(f"[{label}] it={it} t={time.monotonic()-t0:.1f}s "
                  f"score={score:.6f} best={best_score:.6f} "
                  f"focus={focus:.1f} eta={eta:.2f} resets={resets}")
        it += 1
    
    return best_a, best_score


def _optimize_topk(a0, time_limit, rng, topk=20, verbose=False, label=""):
    """Optimize focusing on top-k convolution terms."""
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    a = np.maximum(a, 1e-300)
    a /= a.sum()
    n = int(a.size)
    
    b = _fft_square(a)
    score, _ = _score(a, b)
    best_score = score
    best_a = a.copy()
    
    eta = 0.5
    no_improve = 0
    max_no_improve = 300
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        # Focus on top-k terms
        maxb = float(b.max())
        p = _make_weights(b, focus=100.0, topk=topk)
        grad = _fft_grad(p, a)
        gd = grad - float(np.dot(a, grad))
        maxg = float(np.max(np.abs(gd)))
        if maxg < 1e-300:
            break
        gd = gd / maxg
        
        a_new = a * np.exp(-eta * gd)
        a_new = np.maximum(a_new, 1e-300)
        a_new /= a_new.sum()
        
        b_new = _fft_square(a_new)
        score_new, _ = _score(a_new, b_new)
        
        if score_new < best_score - 1e-15:
            best_score = score_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
            b = b_new
            score = score_new
            eta = min(eta * 1.02, 10.0)
        elif score_new < score - 1e-15:
            a = a_new
            b = b_new
            score = score_new
            eta = min(eta * 1.01, 10.0)
            no_improve = max(no_improve - 1, 0)
        else:
            no_improve += 1
            eta = max(eta * 0.8, 0.01)
        
        if no_improve > max_no_improve:
            a = best_a * np.exp(rng.normal(0.0, 0.2, size=n))
            a = np.maximum(a, 1e-300)
            a /= a.sum()
            b = _fft_square(a)
            score = float(2.0 * n * b.max() / (a.sum()**2))
            no_improve = 0
            max_no_improve = int(max_no_improve * 1.3)
            eta = 0.5
            if it > 200000:
                break
        
        if verbose and it % 5000 == 0:
            print(f"[{label}] topk it={it} t={time.monotonic()-t0:.1f}s "
                  f"score={score:.6f} best={best_score:.6f} eta={eta:.2f}")
        it += 1
    
    return best_a, best_score


def _safe_evaluate(seq):
    try:
        ev = globals().get("evaluate_sequence")
        if ev is None or not callable(ev):
            return float('inf')
        return float(ev(seq))
    except Exception:
        return float('inf')
