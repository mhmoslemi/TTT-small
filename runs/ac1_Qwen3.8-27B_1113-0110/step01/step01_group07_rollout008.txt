"""
Optimized search for non-negative coefficient sequences minimizing 2*n*max(conv)/sum^2.
Uses FFT-based gradient of smooth-max convolution with log-domain Adam, multi-start,
multi-resolution, and adaptive smoothness scheduling.
"""

import time
import numpy as np

def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()

def normalize_positive(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s

def fast_score(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.inf
    p = p / s
    n = p.size
    L = next_pow2(2 * n - 1)
    P = np.fft.rfft(p, L)
    c = np.fft.irfft(P * P, L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return float(2.0 * n * c.max())

def compute_conv_and_grad(p, L, tau):
    """Compute convolution, smooth max value, and gradient w.r.t. p."""
    n = p.size
    P = np.fft.rfft(p, L)
    c = np.fft.irfft(P * P, L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    cm = float(c.max())
    z = (c - cm) / tau
    expz = np.exp(np.clip(z, -80, 0))
    expsum = float(expz.sum())
    if expsum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        smooth_val = cm
    else:
        w = expz / expsum
        smooth_val = cm + tau * float(np.log(expsum))
    # Gradient: grad[j] = 2 * sum_k w[k] * p[k-j]
    p_r = p[::-1]
    W = np.fft.rfft(w, L)
    PR = np.fft.rfft(p_r, L)
    conv_wpr = np.fft.irfft(W * PR, L)
    g_p = 2.0 * conv_wpr[n - 1:2 * n - 1]
    g_p = np.where(np.isfinite(g_p), g_p, 0.0)
    return c, cm, smooth_val, w, g_p

def log_domain_grad(g_p, p):
    """Convert gradient w.r.t. p to gradient w.r.t. log-parameters z."""
    mean_g = float(np.dot(p, g_p))
    g_z = p * (g_p - mean_g)
    return g_z

def adam_single_run(p0, duration_s, rng, n_target=None, tau_init=0.08, tau_min=0.002,
                    lr_init=0.04, lr_decay=0.97, restart_every=5000, restart_noise=0.2):
    """Run Adam optimizer on log-domain parameters for given duration."""
    p = normalize_positive(p0)
    if p is None:
        return None, np.inf
    n = p.size
    L = next_pow2(2 * n - 1)
    deadline = time.monotonic() + max(0.0, duration_s)

    z = np.log(np.maximum(p, 1e-300))
    z = np.where(np.isfinite(z), z, -30.0)
    z -= z.mean()
    z = np.clip(z, -40.0, 40.0)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step = 0
    t_step = 0  # for tau schedule
    best_p = p.copy()
    best_score = fast_score(p)
    no_improve = 0

    while time.monotonic() < deadline:
        # Schedule
        progress = min(1.0, t_step / 20000.0)
        tau = tau_init * (tau_min / tau_init) ** progress
        lr = lr_init * (lr_decay ** (t_step / 500.0))
        lr = max(lr, 0.0005)

        p = np.exp(z - z.max())
        ps = float(p.sum())
        if ps < 1e-300 or not np.isfinite(ps):
            z = np.zeros(n)
            break
        p = p / ps

        c, cm, smooth_val, w, g_p = compute_conv_and_grad(p, L, tau)
        g_z = log_domain_grad(g_p, p)

        # Clip gradient
        mg = float(np.max(np.abs(g_z)))
        if mg > 0.5:
            g_z *= 0.5 / mg

        # Exact score for tracking
        exact_score = 2.0 * n * cm
        if exact_score < best_score - 1e-9:
            best_score = float(exact_score)
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adam update
        t_step += 1
        m = beta1 * m + (1 - beta1) * g_z
        v = beta2 * v + (1 - beta2) * (g_z * g_z)
        b1 = 1 - beta1 ** t_step
        b2 = 1 - beta2 ** t_step
        z = z - lr * (m / b1) / (np.sqrt(v / b2) + eps)
        z = np.where(np.isfinite(z), z, 0.0)
        z -= z.mean()
        z = np.clip(z, -40.0, 40.0)

        # Restart if stuck
        if no_improve >= restart_every:
            z = np.log(np.maximum(best_p, 1e-300))
            z = np.where(np.isfinite(z), z, -30.0)
            z += rng.normal(0.0, restart_noise, n)
            z -= z.mean()
            z = np.clip(z, -40.0, 40.0)
            m *= 0.1
            v *= 0.1
            no_improve = 0
            restart_noise *= 0.7

        step += 1
        if step % 10000 == 0:
            print(f"  adam step {step}, score {best_score:.6f}, tau {tau:.5f}, lr {lr:.5f}")

    return best_p, best_score

def peak_shaping(p, n_iters=100, reduce_frac=0.02):
    """Simple peak-shaping: reduce contributions to the peak lag."""
    n = p.size
    L = next_pow2(2 * n - 1)
    p = p.copy()
    best_p = p.copy()
    best_s = fast_score(p)

    for _ in range(n_iters):
        P = np.fft.rfft(p, L)
        c = np.fft.irfft(P * P, L)[:2 * n - 1]
        c = np.maximum(c, 0.0)
        peak_k = int(np.argmax(c))
        peak_val = c[peak_k]
        if peak_val < 1.0 / n * 0.75:
            break

        # Find which p[j] contribute most to c[peak_k]
        # c[peak_k] = sum_i p[i]*p[peak_k-i]
        contribs = np.zeros(n)
        for j in range(n):
            k2 = peak_k - j
            if 0 <= k2 < n:
                contribs[j] = p[j] * p[k2]
            else:
                # Also check the "other side"
                k2b = j - peak_k
                if 0 <= k2b < n and k2b != j:
                    contribs[j] = p[j] * p[k2b]

        top_idx = np.argsort(contribs)[-max(1, n // 20):]
        for j in top_idx:
            p[j] *= (1.0 - reduce_frac)
        p = np.maximum(p, 0.0)
        s = float(p.sum())
        if s < 1e-300:
            break
        p /= s

        sc = 2.0 * n * float(np.fft.irfft(np.fft.rfft(p, L) ** 2, L)[:2 * n - 1].max())
        if sc < best_s:
            best_s = sc
            best_p = p.copy()
        else:
            p = best_p.copy()

    return best_p, best_s

def resize_seq(p, m):
    p = normalize_positive(p)
    m = int(m)
    if p is None or p.size == 0:
        return np.ones(m) / m
    if p.size == m:
        return p.copy()
    if p.size == 1:
        return np.full(m, 1.0 / m)
    x_old = np.linspace(0.0, 1.0, p.size)
    x_new = np.linspace(0.0, 1.0, m)
    out = np.interp(x_new, x_old, p)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s < 1e-300:
        return np.ones(m) / m
    return out / s

    if best_list is None:
        best_list = [1.0 / 1024.0] * 1024
        try:
            best_eval = evaluate_sequence(best_list)
        except Exception:
            best_eval = np.inf

    print(f"FINAL: score={best_eval:.6f}, n={len(best_list)}")
    return best_list
