def peak_reduction_step(f, conv, M, n, alpha=0.01):
    """One step of peak-directed perturbation."""
    # Identify peak set
    threshold = M - 0.1 * M  # 90% of max
    peak_mask = conv >= threshold
    S = np.where(peak_mask)[0]
    
    # Compute subgradient
    g = np.zeros(n)
    for k in S:
        # d/d f_i of (f*f)(k) = f(k-i) for all valid i
        for i in range(n):
            ki = k - i
            if 0 <= ki < n:
                g[i] += f[ki]
    g /= len(S)
    
    # Sort by gradient
    order = np.argsort(g)
    
    # Move mass from high-g to low-g
    half = n // 2
    high_idx = order[half:]
    low_idx = order[:half]
    
    f_new = f.copy()
    # Compute how much to move
    # For each pair, move alpha * f[high_idx] to f[low_idx]
    for i in range(len(high_idx)):
        h = high_idx[i]
        l = low_idx[i]
        move = alpha * f[h]
        f_new[h] -= move
        f_new[l] += move
    
    f_new = np.clip(f_new, 0, None)
    f_new /= f_new.sum()
    return f_new

import numpy as np
from scipy.fft import fft, ifft
import time

def linear_convolve_fft(a, b=None):
    """Compute linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b) if b is not a else n
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()  # next power of 2
    A = fft(a, fsize)
    B = fft(b, fsize)
    return np.real(ifft(A * B))[:size]

    """Compute 2*n*max(conv) assuming sum(f)=1."""

4. Pair-wise coordinate descent for local refinement

import numpy as np

import time

def _clean_array(arr):
    a = np.asarray(arr, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    np.clip(a, 0.0, 1000.0, out=a)
    return a

def _normalize(a):
    a = _clean_array(a)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(a.size, dtype=np.float64) / max(1, a.size)
    return a / s

def _fft_convolve(a, b=None):
    n = len(a)
    if b is None:
        b = a
        m = n
    else:
        m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    return np.real(ifft(A * B))[:size]

def _score_from_conv(f, conv):
    n = len(f)
    max_c = float(np.max(conv))
    s = float(np.sum(f))
    if s <= 0:
        return float('inf')
    return 2.0 * n * max_c / (s * s)

def _score(f):
    f = _clean_array(f)
    s = float(np.sum(f))
    if s <= 0:
        return float('inf')
    conv = _fft_convolve(f)
    return _score_from_conv(f, conv)

    best_f = f.copy()
    best_conv = _fft_convolve(f)
