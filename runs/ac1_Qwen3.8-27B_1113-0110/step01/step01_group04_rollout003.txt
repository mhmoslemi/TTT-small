def _solve_lp_for_g(f, M):
    """Solve: min M' s.t. (f*g)[k] <= M' for all k, g >= 0, sum(g) = 1."""
    n = len(f)
    # Variables: x = [g_0, ..., g_{n-1}, M'], total n+1
    # Objective: minimize M' → c = [0,...,0, 1]
    c = np.zeros(n+1)
    c[-1] = 1.0
    
    # Inequality constraints: (f*g)[k] - M' <= 0 for k = 0,...,2n-2
    # Row k: [f[k-i] for valid i, -1]
    # Build sparse matrix
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        i_min = max(0, k - (n-1))
        i_max = min(k, n-1)
        for i in range(i_min, i_max+1):
            rows.append(k)
            cols.append(i)
            data.append(f[i])
        # M' column
        rows.append(k)
        cols.append(n)
        data.append(-1.0)
    
    A_ub = scipy.sparse.coo_matrix((data, (rows, cols)), shape=(2*n-1, n+1)).tocsr()
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n+1)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if result.status != 0:
        return None, M
    
    g = result.x[:n]
    M_opt = result.x[n]
    return g, M_opt

"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 for non-negative a via
LP-based alternating optimization + gradient polishing + local search.
"""
import numpy as np
import time
import scipy.sparse
from scipy.optimize import linprog, minimize
import sys

def _pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()

def _clean(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    return np.maximum(v, 0.0)

def _project_simplex(v):
    v = _clean(v)
    n = v.size
    if n == 0:
        return v.copy()
    x = np.sort(v)[::-1]
    css = np.cumsum(x)
    idx = np.arange(1, n+1, dtype=np.float64)
    cond = x > (css - 1.0) / idx
    if not np.any(cond):
        return np.full(n, 1.0/float(n))
    r = int(np.nonzero(cond)[0][-1])
    tau = (css[r] - 1.0) / (r + 1.0)
    out = np.maximum(v - tau, 0.0)
    s = out.sum()
    if s <= 1e-15:
        return np.full(n, 1.0/float(n))
    return out / s

def _score(a):
    v = _clean(a).copy()
    n = v.size
    s = v.sum()
    if s < 1e-12:
        return np.inf
    v /= s
    L = _pow2(2*n - 1)
    A = np.fft.rfft(v, n=L)
    C = A * A
    c = np.fft.irfft(C, n=L)[:2*n-1]
    mx = max(0.0, float(c.max()))
    return 2.0 * n * mx

def _build_lp_matrix(f, n):
    """Build the A_ub matrix for the LP."""
    k = np.arange(2*n - 1, dtype=np.int64)
    i_min = np.maximum(0, k - (n - 1))
    i_max = np.minimum(k, n - 1)
    count = i_max - i_min + 1
    
    total = count.sum()
    rows = np.repeat(k, count)
    offsets = np.zeros(total, dtype=np.int64)
    starts = np.concatenate([[0], np.cumsum(count[:-1])])
    for i in range(2*n - 1):
        c = count[i]
        if c > 1:
            offsets[starts[i]:starts[i]+c] = np.arange(c)
    
    col_idx = i_min[rows] + offsets
    data = f[col_idx].astype(np.float64)
    
    # Add M column
    rows_m = k
    cols_m = np.full(2*n-1, n, dtype=np.int64)
    data_m = -np.ones(2*n-1)
    
    rows_all = np.concatenate([rows, rows_m])
    cols_all = np.concatenate([col_idx, cols_m])
    data_all = np.concatenate([data, data_m])
    
    A_ub = scipy.sparse.coo_matrix((data_all, (rows_all, cols_all)), shape=(2*n-1, n+1)).tocsr()
    return A_ub
