def anneal_binary(n, iters, temp_start, temp_end, rng):

    h = np.zeros(n, dtype=float)
    indices = rng.choice(n, n//2, replace=False)

def refine_continuous(h, rng, time_limit):
    n = len(h)
    cur_c5 = _c5_value(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step = 0.1
    it = 0
    while time.time() < time_limit:
        it += 1
        # annealing schedule
        tau = it / 500000  # large
        temp = 0.5 * (1 - tau) + 1e-5 * tau
        # propose delta via two indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # feasible delta interval: ensure 0 <= h_i+δ <=1, 0<=h_j-δ<=1
        lower = max(-h[i], -(1 - h[j]))
        upper = min(1 - h[i], h[j])
        if upper <= lower:
            continue
        # limit delta magnitude
        max_delta = min(step, upper)
        min_delta = max(-step, lower)
        if max_delta <= min_delta:
            continue
        delta = rng.uniform(min_delta, max_delta)
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_c5 = _c5_value(h_new)
        if new_c5 < cur_c5 or rng.random() < math.exp(-(new_c5 - cur_c5)/temp):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        # decay step occasionally
        if it % 5000 == 0:
            step *= 0.99
            if step < 1e-4:
                step = 1e-4
    return best_h, best_c5

def binary_anneal(n, iters, rng, time_limit):
    # initialize random binary vector with n/2 ones
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n//2, replace=False)
    h[ones] = 1.0
    cur_c5 = _c5_value(h)
    best_h = h.copy()
    best_c5 = cur_c5

    for it in range(iters):
        if time.time() >= time_limit:
            break
        # schedule temperature linearly decreasing
        tau = it / iters
        temp = 0.5 * (1 - tau) + 1e-4 * tau  # initial 0.5, final 1e-4.

        # random swap
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        # ensure they are different values
        if h[i] == h[j]:
            continue

        # swap
        h_new = h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]

        new_c5 = _c5_value(h_new)
        if new_c5 < cur_c5 or rng.random() < math.exp(-(new_c5 - cur_c5) / temp):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        # optional: occasional pairwise multiple flips? skip

    return best_h, best_c5

def _project_sum(h, target):
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # if diff > 0: we need to increase some entries (not at 1)
    # if diff < 0: decrease some entries (not at 0)
    # We'll modify in-place
    if diff > 0:
        caps = 1.0 - h
    else:
        caps = -h
    # iterate indices - maybe order random to avoid bias
    order = np.arange(len(h))
    np.random.shuffle(order)
    for idx in order:
        cap = caps[idx]
        if cap <= 0:
            continue
        take = min(abs(diff), abs(cap))
        take = math.copysign(take, diff)
        h[idx] += take
        diff -= take
        if abs(diff) < 1e-12:
            break
    return h

import time, math
import numpy as np

# ------------------------------------------------------------
# Helper functions
# ------------------------------------------------------------
def _c5_value(h: np.ndarray) -> float:
    """C5 bound for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _project_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust h in‑place so that sum(h) = target while staying in [0,1].
    Deterministic linear scan – no large literals."""
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    n = h.shape[0]
    if diff > 0:          # need to increase some entries
        for i in range(n):
            cap = 1.0 - h[i]
            if cap <= 0.0:
                continue
            take = min(diff, cap)
            h[i] += take
            diff -= take
            if diff <= 1e-12:
                break
    else:                 # need to decrease some entries
        for i in range(n):
            cap = h[i]
            if cap <= 0.0:
                continue
            take = min(-diff, cap)
            h[i] -= take
            diff += take
            if diff >= -1e-12:
                break
    return h


# ------------------------------------------------------------
# Binary (0/1) annealing
# ------------------------------------------------------------
def _binary_anneal(n: int, iters: int, rng: np.random.Generator,
                  time_limit: float) -> tuple[np.ndarray, float]:
    """Start from a random binary vector with exactly n/2 ones and anneal."""
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    cur = _c5_value(h)
    best_h, best = h.copy(), cur

    for it in range(iters):
        if time.time() >= time_limit:
            break
        # linear temperature schedule
        tau = it / iters
        temp = 0.5 * (1.0 - tau) + 1e-4 * tau

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if h[i] == h[j]:
            continue

        h_new = h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        new = _c5_value(h_new)

        if new < cur or rng.random() < math.exp(-(new - cur) / temp):
            h, cur = h_new, new
            if new < best:
                best, best_h = new, h.copy()
    return best_h, best


# ------------------------------------------------------------
# Continuous refinement (mass transfer between two points)
# ------------------------------------------------------------
def _continuous_refine(h0: np.ndarray, rng: np.random.Generator,
                       time_limit: float) -> tuple[np.ndarray, float]:
    """Projected stochastic hill‑climb that preserves total mass."""
    h = h0.copy()
    n = h.shape[0]
    cur = _c5_value(h)
    best_h, best = h.copy(), cur

    step = 0.5          # maximal delta for a move
    it = 0
    while True:
        if time.time() >= time_limit:
            break
        it += 1
        # temperature slowly decays over a huge horizon
        tau = min(it / 5_000_000.0, 1.0)
        temp = 0.5 * (1.0 - tau) + 1e-6 * tau

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        lo = max(-h[i], -(1.0 - h[j]))
        hi = min(1.0 - h[i], h[j])
        lo = max(lo, -step)
        hi = min(hi, step)
        if hi <= lo:
            continue

        delta = rng.uniform(lo, hi)
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta

        new = _c5_value(h_new)
        if new < cur or rng.random() < math.exp(-(new - cur) / temp):
            h, cur = h_new, new
            if new < best:
                best, best_h = new, h.copy()

        if (it & 1023) == 0:          # every 1024 steps
            step *= 0.99
            if step < 1e-4:
                step = 1e-4
    return best_h, best
