def subgrad(h):
    n = len(h)
    dx = 2.0/n
    corr = np.correlate(h, 1-h, mode='full') * dx
    max_val = corr.max()
    # indices close to max
    tol = 1e-8
    idxs = np.where(corr >= max_val - tol)[0]
    grads = np.zeros((len(idxs), n))
    for t, idx in enumerate(idxs):
        k = idx - (n - 1)
        p = abs(k)
        g = np.full(n, dx)
        if p < n:
            g[:n-p] -= dx * h[p:]
            g[p:] -= dx * h[:n-p]
        grads[t] = g
    g_avg = grads.mean(axis=0)
    return g_avg, max_val, idxs
```

import time
import numpy as np

def _project(v, target):
    """Project v onto the box [0,1]^n and the affine hyperplane sum(v)=target."""
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def _evaluate(h):
    """Return max_k ∫ h(x)(1-h(x+k))dx for the discrete step vector h."""
    n = h.size
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full') * dx)

def _subgradient(h):
    """Subgradient of the max‑overlap functional and its current value."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_val = corr.max()
    # treat all lags attaining the maximum (within tolerance) equally
    tol = max_val * 1e-12 + 1e-15
    idxs = np.where(corr >= max_val - tol)[0]
    g = np.full(n, dx)
    for idx in idxs:
        k = idx - (n - 1)
        p = abs(k)
        if p < n:
            g[:n - p] -= dx * h[p:]
            g[p:] -= dx * h[:n - p]
    g /= len(idxs)
    return g, max_val

        if init_vec.size >= base_n:
            n = init_vec.size
            h = init_vec.copy()
        else:
            n = base_n
            h = np.random.rand(n)
    except Exception:
        n = base_n
        h = np.random.rand(n)

    target = n / 2.0                     # because ∫h=1 ⇒ sum(h)=n/2
    h = _project(h, target)

            best_h = cand.copy()

    h = best_h.copy()

    max_time = budget_s * 0.95

        # If the subgradient is (almost) zero we are stuck at a flat point
