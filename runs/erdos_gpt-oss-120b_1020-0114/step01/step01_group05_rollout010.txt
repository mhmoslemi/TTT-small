def subgrad_max(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode='full') * dx
    max_val = conv.max()
    # get indices where conv close to max (within tolerance maybe). 
    max_idxs = np.where(conv >= max_val - 1e-12)[0] # maybe multiple.
    # average subgradient across them
    grad = np.zeros(n)
    for idx in max_idxs:
        shift = idx - (n - 1)
        # compute contributions
        # (1 - h[i+shift]) for i where i+shift in range
        # -h[i-shift] for i where i-shift in range
        # So we compute vectorized:
        if shift >= 0:
            # i+shift in range for i <= n-1-shift
            # i-shift in range for i >= shift
            # vector produce contributions
            # part1: (1 - h[i+shift]) added to grad[i] for i such that i+shift in range
            grad[:n-shift] += (1 - h[shift:])  # i from 0 to n-1-shift, i+shift = shift..n-1
            # part2: -h[i-shift] for i from shift..n-1
            grad[shift:] -= h[:n-shift]
        else:
            # shift negative: let s = -shift >0
            s = -shift
            # i+shift = i - s: in range for i >= s
            grad[s:] += (1 - h[:n-s])
            # i - shift = i + s: in range for i <= n-1-s
            grad[:n-s] -= h[s:]
    # Now divide by number of max indices to average
    grad /= len(max_idxs)
    return grad

def projected_subgrad_descent(h_init, target_sum, total_time, start_time):
    h = h_init.copy()
    best = h.copy()
    best_val = evaluate(h)
    n = len(h)
    dx = 2.0/n
    it = 0
    # learning rate schedule: lr = a / sqrt(it+1)
    a = 0.5
    while time.time() - start_time < total_time:
        it += 1
        # compute gradient
        grad = subgrad_max(h, dx=dx)
        # step size
        lr = a / np.sqrt(it)
        # compute candidate

import time
import numpy as np

def _project(v, target):
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def _corr(a, b):
    n = a.shape[0]
    N = 2 * n
    A = np.fft.rfft(np.concatenate([a, np.zeros(n)]), n=N)
    B = np.fft.rfft(np.concatenate([b, np.zeros(n)]), n=N)
    c = np.fft.irfft(A * np.conj(B), n=N)
    return c[:2*n-1]

def evaluate(h):
    n = h.size
    dx = 2.0 / n
    return (_corr(h, 1 - h) * dx).max()

def _subgrad_max(h, dx):
    n = h.shape[0]
    conv = _corr(h, 1 - h) * dx
    max_val = conv.max()
    # indices achieving max (allow tiny tolerance)
    idxs = np.where(conv >= max_val - 1e-12)[0]
    grad = np.zeros(n)
    for idx in idxs:
        shift = idx - (n - 1)
        if shift >= 0:
            # (1 - h[i+shift]) part
            grad[:n-shift] += (1 - h[shift:])
            # -h[i-shift] part
            grad[shift:] -= h[:n-shift]
        else:
            s = -shift
            grad[s:] += (1 - h[:n-s])
            grad[:n-s] -= h[s:]
    grad = grad * dx / len(idxs)
    return grad

            h = h_new

        if it % 200 == 0:
            jitter = np.random.normal(scale=0.01, size=n)
            h_j = _project(h + jitter, target)
            v_j = evaluate(h_j)
            if v_j < best_val:
                best_val, best_h = v_j, h_j.copy()
                h = h_j

    return best_h, best_val


# ----------------------------------------------------------------------
# Soft‑max (smooth max) refinement – fine tuning
def _softmax_refine(h0, target, end_time):
    h = h0.copy()
    n = h.size
    dx = 2.0 / n
    best_h = h.copy()
    best_val = evaluate(h)
    it = 0
    max_iters = 50000
    lr = 0.2
    T0, Tmin = 0.06, 0.001

    while time.time() < end_time and it < max_iters:
        it += 1
        T = max(Tmin, T0 * (1.0 - it / max_iters))
        conv = _corr(h, 1 - h) * dx
        gmax = conv.max()
        p = np.exp((conv - gmax) / T)
        p /= p.sum()
        idx = np.arange(n)[:, None] - np.arange(n)[None, :] + (n - 1)
        P = p[idx]                         # n×n probability matrix
        grad = dx * (P @ (1 - h) - P.T @ h)

        step = lr
        while True:
            h_new = _project(h - step * grad, target)
            val_new = evaluate(h_new)
            if val_new <= best_val:
                h, best_h, best_val = h_new, h_new.copy(), val_new
                lr = min(lr * 1.07, 1.0)
                break
            cur = evaluate(h)
            if val_new <= cur:
                h = h_new
                lr = max(lr * 0.95, 1e-5)
                break
            step *= 0.5
            if step < 1e-12:
                break

        if it % 500 == 0:
            jitter = np.random.normal(scale=0.02, size=n)
            h_j = _project(h + jitter, target)
            v_j = evaluate(h_j)
            if v_j < best_val:
                h, best_h, best_val = h_j, h_j.copy(), v_j

    return best_h, best_val


# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # ----- problem size -------------------------------------------------
    try:
        parent = np.array(initial_h_values, dtype=float)
        n_parent = parent.size
    except Exception:
        parent = None
        n_parent = 0

    n = max(n_parent, 100)            # at least 100 samples for flexibility
    target = n / 2.0                  # ∫h = 1  →  sum = n/2

    # ----- initial vector -----------------------------------------------
    if parent is not None and n_parent > 0:
        x_parent = np.linspace(0.0, 2.0, n_parent)
        x_new = np.linspace(0.0, 2.0, n)
        h0 = np.interp(x_new, x_parent, parent)
    else:
        h0 = np.full(n, 0.5)

    h0 = _project(h0, target)
    best_h, best_val = h0.copy(), evaluate(h0)

    # ----- Stage 1: many short sub‑gradient runs ------------------------
    stage1_end = start + budget_s * 0.85
    restart = 0
    while time.time() < stage1_end:
        rem = stage1_end - time.time()
        if rem < 1.0:
            break
        if restart < 6:
            chunk = min(5.0, rem)
            h_start = _project(np.random.rand(n), target)
        else:
            chunk = min(20.0, rem)
            h_start = _project(best_h + np.random.normal(scale=0.05, size=n), target)

        end_time = time.time() + chunk
        h_opt, val_opt = _subgrad_descent(h_start, target, end_time)
        if val_opt < best_val:
            best_h, best_val = h_opt, val_opt
        restart += 1

    # ----- Stage 2: soft‑max refinement ---------------------------------
    stage2_end = start + budget_s * 0.98
    if time.time() < stage2_end:
        h_ref, val_ref = _softmax_refine(best_h, target, stage2_end)
        if val_ref < best_val:
            best_h, best_val = h_ref, val_ref

    return best_h, best_val, n
