def compute_c5_binary(h):
    # assume binary integer array (0/1) float is fine
    n = h.shape[0]
    dx = 2.0 / n
    # correlation h * (1 - h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # maybe ignore zero shift? Actually shift zero also considered; but h_i * (1 - h_i) =0 (since either 0 or 1). So zero shift yields 0 -> not max.
    return float(corr.max())
```

def hill_climb_binary(h, max_iters, rng):
    best = h.copy()
    best_val = compute_c5_binary(best)
    n = len(h)
    ones = np.where(best==1)[0]
    zeros = np.where(best==0)[0]
    for _ in range(max_iters):
        # random pick i from ones, j from zeros
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cand = best.copy()
        cand[i] = 0
        cand[j] = 1
        val = compute_c5_binary(cand)
        if val < best_val - 1e-12:
            best = cand
            best_val = val
            # update ones/zeros

            # Simpler: compute new ones/zeros each iteration.

import time
import numpy as np
import scipy.optimize as opt

# ---------- basic evaluation ----------
def compute_c5(h):
    """C5 value for a step function h (array of values)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

# ---------- random feasible binary vector ----------
def random_feasible(n, rng):
    """Uniform random vector in [0,1]^n with exact sum n/2."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    # simple projection to keep bounds
    for _ in range(5):
        over = v > 1.0
        under = v < 0.0
        if not over.any() and not under.any():
            break
        excess = (v[over] - 1.0).sum()
        deficit = (-v[under]).sum()
        interior = (~over) & (~under)
        if excess > 0 and interior.any():
            v[interior] -= excess / interior.sum()
        if deficit > 0 and interior.any():
            v[interior] += deficit / interior.sum()
        v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - n / 2.0) > 1e-12:
        diff = n / 2.0 - cur
        mask = (v > 0.0) & (v < 1.0)
        if mask.any():
            v[mask] += diff / mask.sum()
            v = np.clip(v, 0.0, 1.0)
    return v

# ---------- continuous local improvement ----------
def hill_climb(h, budget, start_time):
    """Greedy pairwise swaps that preserve sum to lower C5."""
    n = h.shape[0]
    best = h.copy()
    best_val = compute_c5(best)
    rng = np.random.default_rng()
    while time.time() - start_time < budget:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # direction so that we stay inside [0,1]
        if best[i] > 0.0 and best[j] < 1.0:
            max_delta = min(best[i], 1.0 - best[j])
        elif best[j] > 0.0 and best[i] < 1.0:
            max_delta = min(best[j], 1.0 - best[i])
            i, j = j, i
        else:
            continue
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random()
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        val = compute_c5(cand)
        if val + 1e-12 < best_val:
            best = cand
            best_val = val
    return best, best_val

        if mask.any():
            h_opt[mask] += diff / mask.sum()
            h_opt = np.clip(h_opt, 0.0, 1.0)
        else:
            h_opt *= target / cur
    return h_opt, compute_c5(h_opt)

# ---------- binary targeted hill climbing ----------
def hill_climb_binary_targeted(h_bin, max_iters, rng):
    """Greedy swaps that directly reduce the worst shift (max C5)."""
    n = h_bin.shape[0]
    dx = 2.0 / n
    best = h_bin.copy()
    corr = np.correlate(best, 1.0 - best, mode="full") * dx
    best_val = float(corr.max())
    for _ in range(max_iters):
        max_idx = int(np.argmax(corr))
        max_val = float(corr[max_idx])
        if max_val <= best_val + 1e-12:
            break
        s = max_idx - (n - 1)  # shift corresponding to this index
        if s >= 0:
            rng_len = n - s
            a_mask = (best[:rng_len] == 1) & (best[s:] == 0)
            b_mask = (best[:rng_len] == 0) & (best[s:] == 1)
            A = np.where(a_mask)[0]
            B = np.where(b_mask)[0]
        else:
            s_abs = -s
            rng_len = n - s_abs
            a_mask = (best[s_abs:] == 1) & (best[:rng_len] == 0)
            b_mask = (best[s_abs:] == 0) & (best[:rng_len] == 1)
            A = np.where(a_mask)[0] + s_abs
            B = np.where(b_mask)[0] + s_abs
        if A.size == 0 or B.size == 0:
            break
        i = rng.choice(A)
        j = rng.choice(B)
        best[i], best[j] = best[j], best[i]
        corr = np.correlate(best, 1.0 - best, mode="full") * dx
        best_val = float(corr.max())
    return best, best_val

# ---------- main routine ----------
def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # base construction (if supplied) -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n0 = h0.size
        candidates = [(n0, h0.copy())]
    else:
        n0 = 180
        h0 = np.full(n0, 0.5, dtype=float)
        candidates = [(n0, h0.copy())]

    # up‑sampled versions -------------------------------------------------------------
    for factor in (2,):
        n = n0 * factor
        h_up = np.repeat(h0, factor)
        candidates.append((n, h_up))

    # a few random starts -------------------------------------------------------------
    candidates.append((n0, random_feasible(n0, rng)))
    candidates.append((n0 * 2, random_feasible(n0 * 2, rng)))

    best_h = None
    best_c5 = np.inf
    best_n = None

    # ----- optimisation loop over candidates ----------------------------------------
    for n, h_start in candidates:
        if time.time() - start > budget_s * 0.85:
            break

        # enforce exact integral
        target = n / 2.0
        cur = h_start.sum()
        if abs(cur - target) > 1e-12:
            h_start = h_start * (target / cur)
            h_start = np.clip(h_start, 0.0, 1.0)

        # continuous SLSQP refinement
        h_opt, c5_val = continuous_optimize_h(h_start, max_iter=2500)
        if c5_val < best_c5:
            best_h, best_c5, best_n = h_opt, c5_val, n

        # cheap continuous hill‑climb for extra gain
        remaining = budget_s - (time.time() - start)
        if remaining > 3.0:
            h_climb, c5_climb = hill_climb(best_h, remaining * 0.4, start)
            if c5_climb < best_c5:
                best_h, best_c5 = h_climb, c5_climb
                best_n = best_h.size

    # ----- binary conversion + targeted hill‑climb ----------------------------------
    if best_h is not None and (time.time() - start) < budget_s * 0.95:
        n = best_h.size
        # round to a binary vector with exactly n/2 ones
        idx = np.argsort(-best_h)
        h_bin = np.zeros(n, dtype=np.int8)
        h_bin[idx[: n // 2]] = 1

        remaining = budget_s - (time.time() - start)
        max_iters = int(min(200000, remaining * 250))
        h_bin_opt, c5_bin = hill_climb_binary_targeted(h_bin, max_iters, rng)

        if c5_bin < best_c5:
            best_h = h_bin_opt.astype(float)
            best_c5 = c5_bin
            best_n = n

    # ----- final safety checks -------------------------------------------------------
    if best_h is None:
        n = n0
        best_h = np.full(n, 0.5, dtype=float)
        best_c5 = compute_c5(best_h)
        best_n = n

    best_h = np.clip(best_h, 0.0, 1.0)
    # enforce exact integral once more
    target = best_n / 2.0
    cur = best_h.sum()
    if abs(cur - target) > 1e-9:
        best_h *= target / cur

    return (best_h.tolist(), float(best_c5), int(best_n))
