import time, math
import numpy as np

def _c5_value(h: np.ndarray) -> float:
    """Compute the C5 bound for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

def _thue_morse(n: int) -> np.ndarray:
    """Balanced Thue‑Morse sequence of length n (n must be a power of two)."""
    seq = np.empty(n, dtype=float)
    for i in range(n):
        # parity of the binary weight of i
        seq[i] = 1.0 - (bin(i).count("1") & 1)   # 1 for even parity, 0 for odd
    return seq

def _max_shift_and_mismatch(h: np.ndarray):
    """Return (max_shift, max_mismatch) for current h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    max_shift = idx - (n - 1)       # shift in [-n+1, n-1]
    max_mis = int(round(corr[idx]))  # integer count of mismatches
    return max_shift, max_mis

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c5 = float("inf")

    # ------------------------------------------------------------------
    # 0.  Use the harness' initial vector if it is already good.
    # ------------------------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        cur = _c5_value(h0)
        if cur < best_c5:
            best_c5, best_h = cur, h0.copy()
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Parameters – we will try a few even lengths.
    # ------------------------------------------------------------------
    lengths = [96, 128, 160]                 # all even, half‑ones = length//2
    max_total_iters = 40000                 # per restart
    temp_start = 0.12
    temp_end = 1e-5

    # ------------------------------------------------------------------
    # 2.  Main optimisation loop (restart until time expires).
    # ------------------------------------------------------------------
    while time.time() - start < budget_s * 0.94:
        n = int(rng.choice(lengths))
        half = n // 2

        # initialise either a random balanced vector or Thue‑Morse pattern
        if rng.random() < 0.5:
            h = np.zeros(n, dtype=float)
            h[:half] = 1.0
            rng.shuffle(h)
        else:
            # Thue‑Morse works best when n is a power of two; otherwise fall back
            if (n & (n - 1)) == 0:
                h = _thue_morse(n)
            else:
                h = np.zeros(n, dtype=float)
                h[:half] = 1.0
                rng.shuffle(h)

        cur_c5 = _c5_value(h)
        if cur_c5 < best_c5:
            best_c5, best_h = cur_c5, h.copy()

        # ------------------------------------------------------------------
        # 3.  Simulated‑annealing style swaps.
        # ------------------------------------------------------------------
        for it in range(max_total_iters):
            # time guard
            if time.time() - start > budget_s * 0.97:
                break

            # temperature (linear schedule)
            tau = it / max_total_iters
            temp = temp_start * (1.0 - tau) + temp_end * tau

            # locate the worst shift and its mismatched indices
            max_shift, max_mis = _max_shift_and_mismatch(h)
            if max_mis == 0:          # perfect solution found
                best_c5, best_h = 0.0, h.copy()
                break

            n = h.shape[0]
            if max_shift >= 0:
                mism = np.where((h[: n - max_shift] == 1) &
                                (h[max_shift:] == 0))[0]
            else:
                d = -max_shift
                mism = np.where((h[d:] == 1) &
                                (h[: n - d] == 0))[0] + d

            if mism.size == 0:        # shouldn't happen, but protect
                break

            i = rng.choice(mism)                # a 1 that becomes a mismatch
            zeros = np.where(h == 0)[0]
            j = rng.choice(zeros)                # a 0 to receive the mass

            # propose the swap
            h_new = h.copy()
            h_new[i] = 0.0
            h_new[j] = 1.0
            new_c5 = _c5_value(h_new)

            # Metropolis acceptance
            if new_c5 < cur_c5 or rng.random() < math.exp(-(new_c5 - cur_c5) / temp):
                h, cur_c5 = h_new, new_c5
                if cur_c5 < best_c5:
                    best_c5, best_h = cur_c5, h.copy()

        # end of one restart – try another if time permits
    # ------------------------------------------------------------------
    # 4.  Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_h is None:
        n = 60
        best_h = np.full(n, 0.5, dtype=float)
        best_c5 = _c5_value(best_h)

    return (best_h.tolist(), float(best_c5), int(best_h.shape[0]))
