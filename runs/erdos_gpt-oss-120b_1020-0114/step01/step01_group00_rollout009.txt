    best = h.copy()
    best_val = compute_c5(best)
    n = len(h)
    dx = 2/n
    while time.time() - start_time < time_limit:
        # compute all f_s
        corr = np.correlate(best, 1.0 - best, mode='full') * dx
        max_f = corr.max()
        # find the shift(s) achieving max
        # Get indices of max_f (there may be several)
        # Choose one shift at random among those
        s_indices = np.where(np.isclose(corr, max_f))[0]
        s_idx = np.random.choice(s_indices)
        # Convert index to shift value: shift = s_idx - (n-1)
        shift = s_idx - (n - 1)
        # compute contributions for this shift

        contributions = np.zeros(n)
        if shift >= 0:

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)
    # get n0
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n0 = h0.size
        candidates = [(n0, h0.copy())]
    else:
        n0 = 180
        h0 = np.full(n0, 0.5)
        candidates = [(n0, h0)]

    # upsample factor 2 and maybe 3
    for factor in (2, 3):
        n = n0 * factor
        h_up = np.repeat(h0, factor)
        candidates.append((n, h_up))

    # random initial for n0 and upsample
    for n in (n0, n0*2):
        candidates.append((n, random_feasible(n, rng)))

    # Maybe also random binary (0/1) vectors
    for n in (n0,):
        candidates.append((n, binary_random_feasible(n, rng)))

    # Evaluate each:
    best_h = None
    best_c5 = np.inf
    best_n = None

    for n, h_start in candidates:
        elapsed = time.time() - start
        if elapsed > budget_s * 0.9: break
        # Ensure sum exactly
        target = n / 2.0
        s = h_start.sum()
        if abs(s - target) > 1e-12:
            h_start = h_start * (target / s)
            h_start = np.clip(h_start, 0.0, 1.0)
        # SLSQP
        h_opt, c5_val = optimize_h(h_start, max_iter=1500)
        if c5_val < best_c5:
            best_h, best_c5, best_n = h_opt, c5_val, n
        # hill climbing (guided) using remaining time
        remaining = budget_s - (time.time() - start)
        if remaining > 5.0:
            climb_time = min(remaining * 0.3, 30.0)
            h_climb, c5_climb = guided_hill_climb(best_h, climb_time, start)
            if c5_climb < best_c5:
                best_h, best_c5, best_n = h_climb, c5_climb, best_h.size

    # final random hill climbing on best found for leftover time
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        h_final, c5_final = guided_hill_climb(best_h, remaining, start)
        if c5_final < best_c5:
            best_h, best_c5, best_n = h_final, c5_final, best_h.size

    # final safety: ensure constraints
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (best_n / 2.0) / best_h.sum()
    return (best_h.tolist(), float(best_c5), int(best_n))

def optimize_h(h_start, max_iter=1000):
    n = h_start.shape[0]
    dx = 2.0/n
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def fun(x):
        h = x[:n]
        t = x[n]
        conv = np.correlate(h, h, mode='full')
        # f_s = sum(h) - conv
        f = h.sum() - conv
        return t - f

    ineq = {"type": "ineq", "fun": fun}
    eq = {"type": "eq", "fun": lambda x: np.sum(x[:n]) - n/2.0}
    constraints = [eq, ineq]

    t0 = compute_c5(h_start)
    x0 = np.concatenate([h_start, [t0]])

    res = opt.minimize(lambda x: x[n], x0, method='SLSQP',
                        bounds=bounds, constraints=constraints,
                        options={'maxiter': max_iter, 'ftol':1e-9, 'disp':False})
    if not res.success:
        # fallback as before
        h_opt = np.clip(h_start, 0., 1.)
        c5 = compute_c5(h_opt)
        return h_opt, c5

    h_opt = np.clip(res.x[:n], 0., 1.)
    # renormalise sum exactly
    target = n/2.0
    cur = h_opt.sum()
    if abs(cur - target) > 1e-9:
        diff = target - cur
        mask = (h_opt > 0.) & (h_opt < 1.)
        if mask.any():
            h_opt[mask] += diff / mask.sum()
            h_opt = np.clip(h_opt, 0., 1.)
        else:
            h_opt *= target / cur
    c5 = compute_c5(h_opt)
    return h_opt, c5

import time
import numpy as np
import scipy.optimize as opt

# --------------------------------------------------------------
# basic utilities
# --------------------------------------------------------------
def compute_c5(h):
    """C5 value for a step function h (numpy array)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def random_feasible(n, rng):
    """Random vector in [0,1]^n with exact sum = n/2."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    # simple projection onto box
    for _ in range(8):
        over = v > 1.0
        under = v < 0.0
        if not over.any() and not under.any():
            break
        excess = (v[over] - 1.0).sum()
        deficit = (-v[under]).sum()
        if excess > 0:
            interior = (~over) & (~under)
            if interior.any():
                v[interior] -= excess / interior.sum()
        if deficit > 0:
            interior = (~over) & (~under)
            if interior.any():
                v[interior] += deficit / interior.sum()
        v = np.clip(v, 0.0, 1.0)
    # final correction of the sum
    cur = v.sum()
    if abs(cur - n / 2.0) > 1e-9:
        diff = n / 2.0 - cur
        mask = (v > 0.0) & (v < 1.0)
        if mask.any():
            v[mask] += diff / mask.sum()
            v = np.clip(v, 0.0, 1.0)
        else:
            v *= (n / 2.0) / cur
    return v


def binary_random_feasible(n, rng):
    """Binary vector with exactly n/2 ones (n must be even)."""
    ones = np.full(n // 2, 1.0)
    zeros = np.full(n - n // 2, 0.0)
    arr = np.concatenate([ones, zeros])
    rng.shuffle(arr)
    return arr.astype(float)


# --------------------------------------------------------------
# SLSQP optimisation (minimise the maximal overlap)
# --------------------------------------------------------------
def optimize_h(h_start, max_iter=1200):
    """Local optimisation of h using SLSQP on the max‑overlap formulation."""
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]  # last variable = t

    # equality: total mass = n/2
    eq = {"type": "eq", "fun": lambda x: np.sum(x[:n]) - n / 2.0}

    # inequality: t >= sum_i h_i*(1-h_{i+s}) for all integer shifts s
    def ineq_fun(x):
        h = x[:n]
        t = x[n]
        conv = np.correlate(h, h, mode="full")      # h_i * h_{i+s}
        f = h.sum() - conv                         # sum_i h_i - sum_i h_i*h_{i+s}
        return t - f                               # array length 2n-1, each >=0

    ineq = {"type": "ineq", "fun": ineq_fun}

    t0 = compute_c5(h_start)
    x0 = np.concatenate([h_start, [t0]])

    res = opt.minimize(
        lambda x: x[n],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=[eq, ineq],
        options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
    )
    if not res.success:
        h_opt = np.clip(h_start, 0.0, 1.0)
        return h_opt, compute_c5(h_opt)

    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    # enforce exact mass (tiny numerical drift)
    target = n / 2.0
    cur = h_opt.sum()
    if abs(cur - target) > 1e-9:
        diff = target - cur
        mask = (h_opt > 0.0) & (h_opt < 1.0)
        if mask.any():
            h_opt[mask] += diff / mask.sum()
            h_opt = np.clip(h_opt, 0.0, 1.0)
        else:
            h_opt *= target / cur
    return h_opt, compute_c5(h_opt)

# Guided hill‑climbing focusing on the worst shift

def guided_hill_climb(h_start, budget_s, start_time, rng):
    """Local improvement using moves that target the current worst shift."""
    best = h_start.copy()
    best_val = compute_c5(best)
    n = best.shape[0]
    dx = 2.0 / n

    while time.time() - start_time < budget_s:
        # correlation to locate the worst shift
        corr = np.correlate(best, 1.0 - best, mode="full")
        max_f = corr.max()
        # choose a shift attaining the maximum (random among ties)
        max_idxs = np.where(corr >= max_f - 1e-12)[0]
        s_idx = rng.choice(max_idxs)
        shift = s_idx - (n - 1)          # integer shift value

        # contribution of each index to the worst shift
        contrib = np.zeros(n)
        if shift >= 0:
            length = n - shift
            contrib[:length] = best[:length] * (1.0 - best[shift:])
        else:
            s = -shift
            length = n - s
            contrib[s:] = best[s:] * (1.0 - best[:length])

        # candidates for decreasing / increasing mass
        can_dec = np.where(best > 1e-12)[0]
        can_inc = np.where(best < 1.0 - 1e-12)[0]
        if can_dec.size == 0 or can_inc.size == 0:
            break

        i = can_dec[np.argmax(contrib[can_dec])]
        j = can_inc[np.argmin(contrib[can_inc])]

        max_delta = min(best[i], 1.0 - best[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random()
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        val = compute_c5(cand)
        if val < best_val - 1e-11:
            best, best_val = cand, val
            continue  # successful move, skip random perturbation

        # occasional random perturbation to escape plateaus
        if rng.random() < 0.15:
            i2, j2 = rng.integers(0, n, size=2)
            if i2 != j2 and best[i2] > 1e-12 and best[j2] < 1.0 - 1e-12:
                max_delta2 = min(best[i2], 1.0 - best[j2])
                if max_delta2 > 0:
                    delta2 = max_delta2 * rng.random()
                    cand2 = best.copy()
                    cand2[i2] -= delta2
                    cand2[j2] += delta2
                    val2 = compute_c5(cand2)
                    if val2 < best_val - 1e-11:
                        best, best_val = cand2, val2
    return best, best_val

        # local SLSQP optimisation if we still have time
        time_left = budget_s - (time.time() - start)
        if time_left > 30.0 and n <= 360:
            h_opt, c5_val = optimize_h(h_start, max_iter=max_iter_for_n(n))
        else:
            h_opt = h_start
            c5_val = compute_c5(h_opt)

        if c5_val < best_c5:
            best_h, best_c5, best_n = h_opt, c5_val, n

        # short guided hill climb on the current best
        remaining = budget_s - (time.time() - start)
        if remaining > 5.0:
            climb_time = min(remaining * 0.3, 20.0)
            h_climb, c5_climb = guided_hill_climb(best_h.copy(), climb_time, start, rng)
            if c5_climb < best_c5:
                best_h, best_c5, best_n = h_climb, c5_climb, best_h.size

    # ------------------------------------------------------------------
    # final polishing on the global best
    # ------------------------------------------------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0 and best_h is not None:
        h_fin, c5_fin = guided_hill_climb(best_h.copy(), remaining, start, rng)
        if c5_fin < best_c5:
            best_h, best_c5, best_n = h_fin, c5_fin, best_h.size

    # ------------------------------------------------------------------
    # enforce constraints exactly before returning
    # ------------------------------------------------------------------
    if best_h is None:
        best_n = n0
        best_h = np.full(best_n, 0.5)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (best_n / 2.0) / best_h.sum()
    best_c5 = compute_c5(best_h)

    return (best_h.tolist(), float(best_c5), int(best_n))
