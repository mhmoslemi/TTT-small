import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """C5 value for a step function sampled on a uniform grid."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1]^n with sum exactly n/2."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)

    # simple projection onto [0,1] while preserving the sum
    for _ in range(10):
        over = v > 1.0
        under = v < 0.0
        if not over.any() and not under.any():
            break
        excess = (v[over] - 1.0).sum()
        deficit = (-v[under]).sum()
        interior = (~over) & (~under)
        if interior.any():
            if excess > 0:
                v[interior] -= excess / interior.sum()
            if deficit > 0:
                v[interior] += deficit / interior.sum()
        v = np.clip(v, 0.0, 1.0)

    cur = v.sum()
    target = n / 2.0
    if abs(cur - target) > 1e-10:
        diff = target - cur
        mask = (v > 0.0) & (v < 1.0)
        if mask.any():
            v[mask] += diff / mask.sum()
            v = np.clip(v, 0.0, 1.0)
        else:
            v = np.clip(v * (target / cur), 0.0, 1.0)
    return v


def shift_climb(h: np.ndarray, end_time: float, rng: np.random.Generator) -> tuple:
    """
    Focused hill‑climb:
    repeatedly identify the shift attaining the current maximum overlap
    and move a little mass from a high‑value position to a low‑value
    position that reduces that shift.
    """
    n = h.shape[0]
    best_h = h.copy()
    best_val = compute_c5(best_h)
    cur_h = h.copy()
    cur_val = best_val

    while time.time() < end_time:
        # correlation without the dx factor – ordering of shifts is unchanged
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        max_idx = int(np.argmax(corr))
        s = max_idx - (n - 1)                     # signed shift
        if s >= 0:
            # i in [0, n-1-s] can be moved towards i+s
            mask = (cur_h[: n - s] > 0.0) & (cur_h[s:] < 1.0)
            if mask.any():
                diff = cur_h[: n - s] - cur_h[s:]
                eligible = np.where(mask)[0]
                i_local = eligible[np.argmax(diff[eligible])]
                i = i_local
                j = i + s
            else:
                # fallback random feasible pair
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or not (cur_h[i] > 0.0 and cur_h[j] < 1.0):
                    continue
        else:
            s_abs = -s
            mask = (cur_h[s_abs:] > 0.0) & (cur_h[: n - s_abs] < 1.0)
            if mask.any():
                diff = cur_h[s_abs:] - cur_h[: n - s_abs]
                eligible = np.where(mask)[0]
                i_local = eligible[np.argmax(diff[eligible])]
                i = i_local + s_abs          # larger index
                j = i + s                    # smaller index (i - s_abs)
            else:
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or not (cur_h[i] > 0.0 and cur_h[j] < 1.0):
                    continue

        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0.0:
            continue
        # take a random fraction of the feasible amount
        delta = max_delta * rng.random()
        if delta <= 0.0:
            continue

        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_val = compute_c5(new_h)

        # accept only strict improvements (tiny tolerance to avoid noise)
        if new_val < cur_val - 1e-12:
            cur_h, cur_val = new_h, new_val
            if new_val < best_val - 1e-12:
                best_h, best_val = new_h.copy(), new_val
        # very occasional uphill move to escape plateaus
        elif rng.random() < 0.0005:
            cur_h, cur_val = new_h, new_val
    return best_h, best_val

    # -------------------------------------------------------------- #
    total_budget = float(budget_s)
    per_candidate = total_budget * 0.12          # ~12 % of budget per start
    for n, h_start in candidates:
        now = time.time()
        if now - start_time > total_budget * 0.75:   # keep some time for final polish
            break

        # enforce exact integral (tiny rounding may appear after repeats)
        target = n / 2.0
        cur_sum = h_start.sum()
        if abs(cur_sum - target) > 1e-12:
            h_start = h_start * (target / cur_sum)

        # allocate a slice of the remaining budget for this run
        time_left = total_budget - (time.time() - start_time)
        limit = min(per_candidate, time_left * 0.9)
        end = time.time() + limit

        h_opt, val_opt = shift_climb(h_start, end, rng)

        if val_opt < best_val:
            best_h, best_val, best_n = h_opt.copy(), val_opt, n

    # -------------------------------------------------------------- #
    # 3.  Final intensive refinement of the current best solution
    # -------------------------------------------------------------- #
    remaining = total_budget - (time.time() - start_time)
    if best_h is not None and remaining > 5.0:
        end = time.time() + remaining * 0.9
        h_refined, val_refined = shift_climb(best_h, end, rng)
        if val_refined < best_val:
            best_h, best_val = h_refined, val_refined
            best_n = best_h.size

    # -------------------------------------------------------------- #
    # 4.  Safety net: fall back to a uniform construction if needed
    # -------------------------------------------------------------- #
    if best_h is None:
        best_n = n0
        best_h = np.full(best_n, 0.5)
        best_val = compute_c5(best_h)

    # -------------------------------------------------------------- #
    # 5.  Final projection onto the feasible set
    # -------------------------------------------------------------- #
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    cur = best_h.sum()
    if abs(cur - target) > 1e-9:
        best_h *= (target / cur)

    return (best_h.tolist(), float(best_val), int(best_n))
