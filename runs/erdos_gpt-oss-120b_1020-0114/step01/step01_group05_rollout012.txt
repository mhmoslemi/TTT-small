import time, numpy as np, math

def _evaluate(h):
    n = h.size
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def _project(v, target):
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def _softmax_grad(h, T):
    n = h.size
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    gmax = g.max()
    p = np.exp((g - gmax) / T)
    p = p / p.sum()
    idx = (np.arange(n)[:, None] - np.arange(n)[None, :]) + (n - 1)
    P = p[idx]  # n x n
    grad = dx * (P @ (1 - h) - P.T @ h)
    return grad

def _continuous_opt(n, max_iters, rng, time_limit):
    target_sum = n * 0.5
    # random initialization
    h = rng.random(n)
    h = _project(h, target_sum)
    best_h = h.copy()
    best_val = _evaluate(best_h)
    lr = 0.15
    T0 = 0.08
    Tmin = 0.001
    for it in range(max_iters):
        # time guard
        if time.time() > time_limit:
            break
        T = max(Tmin, T0 * (1 - it / max_iters))
        g = _softmax_grad(h, T)
        # simple projected gradient step (no line search)
        h_new = _project(h - lr * g, target_sum)
        val_new = _evaluate(h_new)
        if val_new < best_val:
            best_val = val_new
            best_h = h_new.copy()
        # accept step if improvement else maybe keep (if val_new <= _evaluate(h))
        if val_new <= _evaluate(h):
            h = h_new
        else:
            # maybe revert or keep original; reduce step size gradually
            lr *= 0.99
            if lr < 1e-5:
                lr = 1e-5
        # occasionally increase lr if improvement frequently
        if it % 200 == 0 and it > 0:
            lr = min(lr * 1.05, 0.5)
    return best_h, best_val

def _binary_hill_climb(n, rng, time_limit):
    target_ones = n // 2
    # random balanced binary init
    perm = rng.permutation(n)
    h = np.zeros(n, dtype=float)
    h[perm[:target_ones]] = 1.0

    # temperature schedule
    T = 5.0
    decay = 0.9995
    it = 0
    # Precompute indices each iteration via where
    while time.time() < time_limit:
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i_one = rng.choice(ones)
        i_zero = rng.choice(zeros)
        h_new = best_h.copy()
        h_new[i_one] = 0.0
        h_new[i_zero] = 1.0
        val_new = _evaluate(h_new)
        if val_new < best_c or rng.random() < math.exp(-(val_new - best_c) / T):

            best_c = val_new

        it += 1
        if it % 10000 == 0 and best_c <= 0.3810:

            break

def _binary_hill_climb(n, rng, time_limit, start_h=None):

    if start_h is None:

        h=np.zeros(n)

            idx=np.where(h==1)[0]
            drop=idx[:cur_ones-target_ones]

import time
import math
import numpy as np

def _evaluate(h: np.ndarray) -> float:
    """Return C5 for a given step function h."""
    n = h.size
    dx = 2.0 / n
    # full correlation implements the required overlap integral
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _project(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the box [0,1]^n while enforcing sum(v)=target_sum."""
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _binary_hill_climb(
    n: int,
    rng: np.random.Generator,
    time_limit: float,
    start_h: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style local search on binary vectors (0/1 values).
    Returns the best vector found before time_limit.
    """
    target_ones = n // 2

    # initialise current solution
    if start_h is None:
        perm = rng.permutation(n)
        cur = np.zeros(n, dtype=float)
        cur[perm[:target_ones]] = 1.0
    else:
        cur = start_h.copy()
        # enforce exact half‑ones constraint
        cur_sum = int(cur.sum())
        if cur_sum > target_ones:
            idx = np.where(cur == 1)[0]
            cur[idx[: cur_sum - target_ones]] = 0.0
        elif cur_sum < target_ones:
            idx = np.where(cur == 0)[0]
            cur[idx[: target_ones - cur_sum]] = 1.0

    cur_val = _evaluate(cur)
    best = cur.copy()
    best_val = cur_val

    # annealing schedule
    T = 5.0
    decay = 0.9997

    while time.time() < time_limit:
        # pick a random 1 and a random 0 to swap
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i1 = rng.choice(ones)
        i0 = rng.choice(zeros)

        cand = cur.copy()
        cand[i1] = 0.0
        cand[i0] = 1.0
        cand_val = _evaluate(cand)

        # Metropolis acceptance
        if cand_val < cur_val or rng.random() < math.exp(-(cand_val - cur_val) / T):
            cur = cand
            cur_val = cand_val
            if cur_val < best_val:
                best = cur.copy()
                best_val = cur_val

        T *= decay
        # optional early exit if we already beat the target
        if best_val <= 0.3808:
            break

    return best, best_val


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # 1. Choose a reasonably large even grid (but still well below 1 000)
    # ------------------------------------------------------------------
    n = 200                     # number of sample points (must be even)
    half = n // 2               # number of ones in a binary vector
    target_sum = n * 0.5        # integral constraint (∫h = 1)

    best_h = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 2. Random sampling phase – cheap, often already hits the target bound
    # ------------------------------------------------------------------
    max_random_iters = 3000
    time_for_random = budget_s * 0.4
    for _ in range(max_random_iters):
        if time.time() - start_time > time_for_random:
            break
        perm = rng.permutation(n)
        h = np.zeros(n, dtype=float)
        h[perm[:half]] = 1.0
        val = _evaluate(h)
        if val < best_val:
            best_val = val
            best_h = h
            if best_val <= 0.3808:
                break

    # ------------------------------------------------------------------
    # 3. If needed, improve the best candidate with a local search
    # ------------------------------------------------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 5.0:
        # give the hill‑climb roughly the rest of the budget
        t_end = time.time() + remaining * 0.9
        # start from the best binary sample we have (or from a fresh one)
        start_vec = best_h if best_h is not None else None
        h_opt, val_opt = _binary_hill_climb(n, rng, t_end, start_h=start_vec)
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt

    # ------------------------------------------------------------------
    # 4. Final projection (safeguard against numerical drift) and output
    # ------------------------------------------------------------------
    if best_h is None:                     # extremely unlikely fallback
        best_h = np.zeros(n, dtype=float)
        best_h[:half] = 1.0
    best_h = _project(best_h, target_sum)
    final_val = _evaluate(best_h)

    return best_h, final_val, n
