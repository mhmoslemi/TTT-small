def make_h(v):
    N2 = len(v)
    N = N2 * 2
    h = np.empty(N, dtype=np.float64)
    h[:N2] = v
    h[N2:] = 1.0 - v[::-1]

def compute_c5_h(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def compute_c5_v(v):
    return compute_c5_h(make_h(v))

def idx_to_v(i, half_N):
    if i < half_N:

def greedy_step(v):
    half = len(v)
    N = half * 2
    h = make_h(v)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(corr.argmax())
    s = max_idx - (N - 1)
    # contributions
    i_start = max(0, -s)
    i_end = min(N, N - s)
    if i_start >= i_end:
        return v, False, compute_c5_h(h)
    i_range = np.arange(i_start, i_end)
    j_range = i_range + s
    contrib = h[i_range] * (1.0 - h[j_range])
    # index with max contribution
    idx = int(contrib.argmax())
    i0 = int(i_range[idx])
    j0 = int(j_range[idx])
    hi = h[i0]
    a = 1.0 - h[j0]
    di_max = min(hi, a)
    if di_max <= 1e-12:
        return v, False, compute_c5_h(h)
    d_opt = (hi + a) / 2.0
    if d_opt > di_max:
        d_opt = di_max
    # apply update to v
    vi, si = idx_to_v(i0, half)
    vj, sj = idx_to_v(j0, half)
    v_new = v.copy()
    if si == 1:
        v_new[vi] = max(0.0, min(1.0, v_new[vi] - d_opt))
    else:
        v_new[vi] = max(0.0, min(1.0, v_new[vi] + d_opt))

import numpy as np, time, math, random

def _make_h(v):
    """Construct the full step function from the half‑vector v (symmetry h(x)+h(2‑x)=1)."""
    half = v.shape[0]
    h = np.empty(half * 2, dtype=np.float64)
    h[:half] = v
    h[half:] = 1.0 - v[::-1]
    return h

def _c5_of_h(h):
    """C5 value for a full array h."""
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def _c5_of_v(v):
    return _c5_of_h(_make_h(v))

def _idx_to_v(i, half):
    """Map a full‑array index to (v‑index, sign) where sign=+1 means h[i]=v[vi],
    sign=-1 means h[i]=1‑v[vi]."""
    if i < half:
        return i, 1
    else:
        return (2 * half - 1 - i), -1

def _greedy_step(v):
    """One deterministic move that reduces the current maximal correlation.
    Returns (new_v, improved, new_c5)."""
    half = v.shape[0]
    N = half * 2
    h = _make_h(v)

    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(corr.argmax())
    s = max_idx - (N - 1)                     # shift in index units

    i0 = max(0, -s)
    i1 = min(N, N - s)                       # exclusive
    if i0 >= i1:
        return v, False, _c5_of_h(h)

    i_range = np.arange(i0, i1)
    j_range = i_range + s
    contrib = h[i_range] * (1.0 - h[j_range])
    idx = int(contrib.argmax())
    i = int(i_range[idx])
    j = int(j_range[idx])

    hi = h[i]
    aj = 1.0 - h[j]
    dmax = min(hi, aj)
    if dmax <= 1e-12:
        return v, False, _c5_of_h(h)

    d_opt = (hi + aj) / 2.0
    if d_opt > dmax:
        d_opt = dmax

    vi, si = _idx_to_v(i, half)
    vj, sj = _idx_to_v(j, half)

    v_new = v.copy()
    if si == 1:
        v_new[vi] = max(0.0, min(1.0, v_new[vi] - d_opt))
    else:
        v_new[vi] = max(0.0, min(1.0, v_new[vi] + d_opt))
    if sj == 1:
        v_new[vj] = max(0.0, min(1.0, v_new[vj] + d_opt))
    else:
        v_new[vj] = max(0.0, min(1.0, v_new[vj] - d_opt))

    new_c5 = _c5_of_v(v_new)
    old_c5 = _c5_of_h(h)
    if new_c5 < old_c5 - 1e-12:
        return v_new, True, new_c5
    return v, False, old_c5

def _refine(v):
    """Repeated greedy steps until no improvement."""
    while True:
        v, improved, _ = _greedy_step(v)
        if not improved:
            break
    return v, _c5_of_v(v)

def _single_move(v, max_delta):
    """Propose a random coordinate change within [0,1]."""
    i = np.random.randint(v.shape[0])
    delta = (np.random.rand() * 2.0 - 1.0) * max_delta
    v_new = v.copy()
    v_new[i] = np.clip(v_new[i] + delta, 0.0, 1.0)
    return v_new

    target_sum = N / 2.0           # automatically satisfied by symmetry

    # start from a random symmetric vector
    cur_v = np.random.rand(half)
    cur_c5 = _c5_of_v(cur_v)
    if cur_c5 < best_c5:
        best_c5 = cur_c5
        best_h = _make_h(cur_v)

    # annealing‑style schedule (purely for move size, not for acceptance)
    T0, Tf = 0.02, 1e-6
    delta0, delta_min = 0.4, 1e-4

    # ------------------------------------------------------------------
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s - 5.0:
            break
        prog = elapsed / (budget_s - 5.0)
        # decreasing temperature and step size
        T = T0 * (Tf / T0) ** prog
        max_delta = delta0 * (1.0 - prog) + delta_min

        # random single‑coordinate move
        cand_v = _single_move(cur_v, max_delta)
        cand_c5 = _c5_of_v(cand_v)
        dC = cand_c5 - cur_c5
        if dC < 0 or (T > 0 and np.random.rand() < math.exp(-dC / T)):
            cur_v, cur_c5 = cand_v, cand_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = _make_h(cur_v)

        # occasional greedy refinement of the current state
        if np.random.rand() < 0.015:
            cur_v, cur_c5 = _refine(cur_v)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = _make_h(cur_v)

        # occasional larger perturbation (reset a few coordinates)
        if np.random.rand() < 0.004:
            k = np.random.randint(1, max(2, half // 10 + 1))
            idxs = np.random.choice(half, k, replace=False)
            v_big = cur_v.copy()
            v_big[idxs] = np.random.rand(k)
            big_c5 = _c5_of_v(v_big)
            dC = big_c5 - cur_c5
            if dC < 0 or (T > 0 and np.random.rand() < math.exp(-dC / T)):
                cur_v, cur_c5 = v_big, big_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = _make_h(cur_v)

        # occasional full restart
        if np.random.rand() < 0.002:
            cur_v = np.random.rand(half)
            cur_c5 = _c5_of_v(cur_v)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = _make_h(cur_v)

    # ------------------------------------------------------------------
    # final tidy‑up: a last deterministic refinement of the best vector
    if best_h is not None:
        # recover the symmetric half‑vector from best_h
        half_best = best_h.shape[0] // 2
        best_v = best_h[:half_best].copy()
        best_v, final_c5 = _refine(best_v)
        best_h = _make_h(best_v)
        final_c5 = _c5_of_h(best_h)
    else:
        best_h = _make_h(cur_v)
        final_c5 = _c5_of_h(best_h)

    # enforce the integral exactly (tiny numeric drift)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        # adjust the first element inside the box
        adj = np.clip(best_h[0] + diff, 0.0, 1.0) - best_h[0]
        best_h[0] += adj

    final_c5 = _c5_of_h(best_h)
    return best_h.tolist(), float(final_c5), len(best_h)
