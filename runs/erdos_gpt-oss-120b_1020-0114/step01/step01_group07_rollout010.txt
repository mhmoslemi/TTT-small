import numpy as np, time, math

def _project(v, target):
    # clip to [0,1]
    v = np.clip(v, 0.0, 1.0)
    # adjust sum to target
    diff = target - v.sum()
    # Tolerance small
    # We can implement water-filling as in earlier code
    while abs(diff) > 1e-12:
        if diff > 0:
            free = v < 1.0
            nfree = free.sum()
            if nfree == 0: break
            add = diff / nfree
            v[free] = np.minimum(v[free] + add, 1.0)
        else:
            free = v > 0.0
            nfree = free.sum()
            if nfree == 0: break
            sub = diff / nfree
            v[free] = np.maximum(v[free] + sub, 0.0)
        diff = target - v.sum()
    return v

def _c5_of(h):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / h.size)

def _gradient(h, shift):
    N = h.size
    g = np.ones(N)
    if shift >= 0:
        s = shift
        # Subtract h_{i+s}
        if s < N:
            g[:N - s] -= h[s:]
            g[s:] -= h[:N - s]
    else:
        s = -shift
        if s < N:
            g[s:] -= h[:N - s]   # h_{i-s}
            g[:N - s] -= h[s:]   # h_{i+s}
    return g

import numpy as np
import time
import math

def _project(v, target):
    """Project v onto the box [0,1] and adjust the sum to target."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    # simple water‑filling
    while abs(diff) > 1e-12:
        if diff > 0:
            free = v < 1.0
            nfree = free.sum()
            if nfree == 0:
                break
            add = diff / nfree
            v[free] = np.minimum(v[free] + add, 1.0)
        else:
            free = v > 0.0
            nfree = free.sum()
            if nfree == 0:
                break
            sub = diff / nfree
            v[free] = np.maximum(v[free] + sub, 0.0)
        diff = target - v.sum()
    return v


def _c5_of(h):
    """Maximum overlap integral for a step vector h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)


def _gradient(h, shift):
    """Gradient of C5 for the worst shift (ignoring the outer‑zero padding)."""
    N = h.size
    g = np.ones(N)
    if shift >= 0:
        s = shift
        if s < N:
            g[: N - s] -= h[s:]
            g[s:] -= h[: N - s]
    else:
        s = -shift
        if s < N:
            g[s:] -= h[: N - s]
            g[: N - s] -= h[s:]
    return g
