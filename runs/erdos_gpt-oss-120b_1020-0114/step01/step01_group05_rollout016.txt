def _project(v, target):
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
```

def _softmax_grad(h, T, idx):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    gmax = corr.max()
    # avoid overflow
    exp_vals = np.exp((corr - gmax) / T)
    p = exp_vals / exp_vals.sum()
    P = p[idx]   # shape (n,n)
    grad = dx * (P @ (1 - h) - P.T @ h)
    return grad, np.max(corr)  # also return current max value
```

import time
import numpy as np

def _project(v, target):
    """Project vector v onto the set {0 <= x <= 1, sum x = target}."""
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def _evaluate(h):
    """Maximum overlap integral for the discrete step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def _softmax_grad(h, T, idx):
    """Gradient of the log‑sum‑exp smoothed max objective."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    gmax = corr.max()
    # soft‑max probabilities
    expv = np.exp((corr - gmax) / T)
    p = expv / expv.sum()
    # matrix of probabilities indexed by i‑j
    P = p[idx]                     # shape (n, n)
    grad = dx * (P @ (1 - h) - P.T @ h)
    return grad, corr.max()
