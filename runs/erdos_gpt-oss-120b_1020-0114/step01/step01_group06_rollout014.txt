def run(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np

    rng = np.random.default_rng(seed)
    start = time.time()

    N = 80  # even
    target = N // 2

    # initial pattern using Beatty
    alpha = (math.sqrt(5) - 1) / 2  # approx 0.618
    i = np.arange(N)
    frac = (i * alpha) % 1.0
    h = (frac < 0.5).astype(np.int8)

    # adjust sum
    cur_sum = h.sum()
    if cur_sum > target:
        # turn excess 1s to 0
        excess = int(cur_sum - target)
        idx = rng.choice(np.flatnonzero(h), size=excess, replace=False)
        h[idx] = 0
    elif cur_sum < target:
        deficit = int(target - cur_sum)
        idx = rng.choice(np.flatnonzero(1 - h), size=deficit, replace=False)
        h[idx] = 1

    g = 1 - h
    c = np.convolve(h, g, mode='full')
    dx = 2.0 / N
    cur_val = c.max() * dx
    best_h = h.copy()
    best_val = cur_val

    max_iters = 2000000
    T_start = 0.3
    T_end = 1e-5

    for it in range(max_iters):
        if time.time() - start > budget_s * 0.97:
            break
        # pick random i with h=1, j with h=0
        # ensure there is at least one of each
        ones_idx = np.flatnonzero(h)
        zeros_idx = np.flatnonzero(g)
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        # compute trial correlation
        c_trial = c.copy()
        # apply delta
        c_trial[i:i+N] -= g
        c_trial[j:j+N] += g
        c_trial[i:i+N] += h
        c_trial[j:j+N] -= h
        c_trial[2*i] -= 1
        c_trial[2*j] -= 1
        c_trial[i+j] += 2

        new_val = c_trial.max() * dx

        progress = it / max_iters
        T = T_start * (1 - progress) + T_end * progress

        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if math.exp(-(new_val - cur_val) / T) > rng.random():
                accept = True

        if accept:
            c = c_trial
            cur_val = new_val
            # swap bits
            h[i] = 0
            h[j] = 1
            g[i] = 1
            g[j] = 0
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
        # optionally recompute ones/zeros if they become stale; we use fresh each iteration

    # Return best
    return (best_h.astype(float).tolist(), float(best_val), N)

import time, math, numpy as np

def _c5_from_corr(corr, n):
    """Convert a full convolution array to the C5 value."""
    return float(corr.max() * (2.0 / n))

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Simulated‑annealing search for a binary step function h on [0,2] that
    minimises  max_k ∫ h(x)(1‑h(x+k))dx  (the Erdős C5 constant).
    The function works with a discretisation of N points (N even) and
    returns the best vector found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # --------------------------------------------------------------
    # 1. Choose a modest size (small enough to be fast, large enough
    #    to give good resolution). 200 points works well for the budget.
    # --------------------------------------------------------------
    N = 200                     # number of discretisation points (must be even)
    target_ones = N // 2        # sum constraint  -> exactly half the entries are 1
    dx = 2.0 / N                # step width, used only for the final C5 value

    # --------------------------------------------------------------
    # 2. Initialise a binary pattern.
    #    Use a Beatty sequence (irrational rotation) – it gives a well‑
    #    distributed start and is deterministic.
    # --------------------------------------------------------------
    phi = (math.sqrt(5) - 1) / 2                     # ~0.618…
    idx = np.arange(N)
    frac = (idx * phi) % 1.0
    h = (frac < 0.5).astype(np.int8)                 # 0/1 pattern
    cur = int(h.sum())
    if cur > target_ones:                           # too many 1’s – turn some off
        excess = cur - target_ones
        pos = rng.choice(np.flatnonzero(h), size=excess, replace=False)
        h[pos] = 0
    elif cur < target_ones:                         # too few 1’s – turn some on
        deficit = target_ones - cur
        pos = rng.choice(np.flatnonzero(1 - h), size=deficit, replace=False)
        h[pos] = 1

    g = 1 - h                                        # complement (1‑h)
    # full convolution h * g  (same as np.correlate for the objective)
    corr = np.convolve(h, g, mode='full')
    cur_c5 = _c5_from_corr(corr, N)

    best_h = h.copy()
    best_c5 = cur_c5

    # --------------------------------------------------------------
    # 3. Simulated annealing.
    #    The update of the convolution after swapping a 1 and a 0
    #    can be performed in O(N) by simple slice arithmetic,
    #    avoiding a full recomputation each trial.
    # --------------------------------------------------------------
    T_start = 0.4
    T_end = 1e-6

    # counters for occasional restarts (if we stagnate)
    no_improve = 0
    RESTART_LIMIT = 200000          # iterations without improvement -> restart

    while True:
        elapsed = time.time() - start
        if elapsed > budget_s * 0.98:
            break

        # temperature ramps linearly with elapsed time
        progress = elapsed / (budget_s * 0.98)
        T = T_start * (1.0 - progress) + T_end * progress

        # ----------------------------------------------------------
        # a) try a move that directly attacks the currently worst shift
        # ----------------------------------------------------------
        k_max = int(corr.argmax())
        shift = k_max - (N - 1)                     # integer shift corresponding to the max entry

        # find indices that contribute to this worst shift
        if shift >= 0:
            rng_range = np.arange(N - shift)
            bad   = rng_range[(h[rng_range] == 1) & (h[rng_range + shift] == 0)]
            good  = rng_range[(h[rng_range] == 0) & (h[rng_range + shift] == 1)]
        else:                                       # shift < 0
            s = -shift
            rng_range = np.arange(N - s)
            # i = j + s  ⇒  h[i]=1  &  h[j]=0  for a “bad’’ pair
            bad   = (rng_range + s)[(h[rng_range + s] == 1) & (h[rng_range] == 0)]
            good  = (rng_range + s)[(h[rng_range + s] == 0) & (h[rng_range] == 1)]

        moved = False
        if bad.size and good.size:
            i = int(rng.choice(bad))
            j = int(rng.choice(good))

            # trial update of the convolution (copy – O(N))
            trial = corr.copy()
            trial[i:i+N] -= g
            trial[j:j+N] += g
            trial[i:i+N] += h
            trial[j:j+N] -= h
            trial[2*i]   -= 1
            trial[2*j]   -= 1
            trial[i+j]   += 2

            new_c5 = _c5_from_corr(trial, N)

            # Metropolis acceptance
            accept = False
            if new_c5 < cur_c5:
                accept = True
            else:
                if math.exp(-(new_c5 - cur_c5) / T) > rng.random():
                    accept = True

            if accept:
                # commit
                corr = trial
                cur_c5 = new_c5
                h[i] = 0
                h[j] = 1
                g[i] = 1
                g[j] = 0
                moved = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    no_improve = 0

        # ----------------------------------------------------------
        # b) if the targeted move failed, fall back to a completely random swap
        # ----------------------------------------------------------
        if not moved:
            # pick any 1‑position and any 0‑position
            pos_one  = rng.choice(np.flatnonzero(h))
            pos_zero = rng.choice(np.flatnonzero(g))

            i, j = int(pos_one), int(pos_zero)

            trial = corr.copy()
            trial[i:i+N] -= g
            trial[j:j+N] += g
            trial[i:i+N] += h
            trial[j:j+N] -= h
            trial[2*i]   -= 1
            trial[2*j]   -= 1
            trial[i+j]   += 2

            new_c5 = _c5_from_corr(trial, N)

            accept = False
            if new_c5 < cur_c5:
                accept = True
            else:
                if math.exp(-(new_c5 - cur_c5) / T) > rng.random():
                    accept = True

            if accept:
                corr = trial
                cur_c5 = new_c5
                h[i] = 0
                h[j] = 1
                g[i] = 1
                g[j] = 0
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    no_improve = 0
            else:
                no_improve += 1

        # ----------------------------------------------------------
        # c) occasional restart if we have stagnated for a long time
        # ----------------------------------------------------------
        if no_improve >= RESTART_LIMIT:
            # fresh random binary start
            h = np.zeros(N, dtype=np.int8)
            h[rng.choice(N, target_ones, replace=False)] = 1
            g = 1 - h
            corr = np.convolve(h, g, mode='full')
            cur_c5 = _c5_from_corr(corr, N)
            no_improve = 0
            # keep the best ever seen (best_h / best_c5)

        # early exit as soon as we beat the target bound
        if best_c5 <= 0.3808:
            break

    # --------------------------------------------------------------
    # 4. Return the best solution discovered.
    # --------------------------------------------------------------
    return (best_h.astype(float).tolist(), float(best_c5), int(N))
