def gradient_max_overlap(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full') * (2.0/n) # dx factor but gradient w.r.t h not scaling? Actually overlap scaled by dx, gradient also scaled by dx. We'll include factor dx later.
    i_max = np.argmax(corr)
    shift = i_max - (n - 1)
    grad = np.zeros_like(h)
    if shift >= 0:
        # left indices 0..n-shift-1
        left = slice(0, n - shift)
        right = slice(shift, n)
        grad[left] += (1 - h[right])
        grad[right] += -h[left]
    else:
        s_abs = -shift
        left = slice(s_abs, n)
        right = slice(0, n - s_abs)
        grad[left] += (1 - h[right])
        grad[right] += -h[left]
    # multiply by dx because corr scaled by dx; gradient of objective = dx * grad
    grad *= (2.0/n)
    return grad, corr[i_max], shift

import time
import numpy as np
from scipy.optimize import minimize, NonlinearConstraint

# ---------- global placeholders for the SLSQP stage ----------
_opt_n = None          # number of points
_opt_dx = None         # grid spacing 2 / n
_opt_target_sum = None # required sum of h (n/2)

# ---------- helper functions ----------
def project_to_feasible(h, target):
    """Clip to [0,1] and adjust linearly so that sum(h)=target (approx)."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == 0.0:
        # place mass uniformly
        idx = np.random.choice(len(h), int(round(target)), replace=False)
        h[idx] = 1.0
        return h
    scale = target / cur
    h = np.clip(h * scale, 0.0, 1.0)
    # simple iterative correction for remaining error
    for _ in range(10):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            i = np.random.randint(len(h))
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
            continue
        delta = diff / free.size
        h[free] += delta
        np.clip(h, 0.0, 1.0, out=h)
    return h

def project_onto_capped_simplex(v, S):
    """Projection onto {x: 0<=x<=1, sum(x)=S} using a simple bisection on lambda."""
    lo, hi = -1.0, 1.0
    # enlarge interval until the condition is satisfied
    while np.clip(v - hi, 0.0, 1.0).sum() > S:
        hi *= 2.0
    while np.clip(v - lo, 0.0, 1.0).sum() < S:
        lo *= 2.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > S:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def objective_fun(x):
    """Minimise the auxiliary variable t (last entry)."""
    return x[-1]

def eq_constraint_fun(x):
    """Integral constraint sum(h) = n/2."""
    return np.sum(x[:_opt_n]) - _opt_target_sum

def ineq_constraint_fun(x):
    """Enforce t >= overlap_k for all shifts."""
    h = x[:_opt_n]
    t = x[-1]
    corr = np.correlate(h, 1 - h, mode='full') * _opt_dx
    return t - corr  # must be >=0

def optimise_start(start_h, maxiter=800):
    """Run a single SLSQP optimisation from start_h."""
    global _opt_n, _opt_dx, _opt_target_sum
    n = int(start_h.shape[0])
    _opt_n = n
    _opt_dx = 2.0 / n
    _opt_target_sum = n / 2.0

    h0 = project_to_feasible(start_h, _opt_target_sum)

    init_corr = np.correlate(h0, 1 - h0, mode='full') * _opt_dx
    t0 = float(np.max(init_corr) + 1e-8)

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    eq_cons = {'type': 'eq', 'fun': eq_constraint_fun}
    m = 2 * n - 1
    ineq_cons = NonlinearConstraint(ineq_constraint_fun,
                                    lb=np.zeros(m),

                   x0,
                   method='SLSQP',
                   bounds=bounds,

    h_opt = project_to_feasible(h_opt, _opt_target_sum)

    try:

    except Exception:

    return h_opt, c5_val

def max_corr_and_gradient(h):
    """Return gradient of the maximal overlap, its value and the shift."""
    n = h.shape[0]
    dx = 2.0 / n
    corr_full = np.correlate(h, 1 - h, mode='full') * dx
    i_max = int(np.argmax(corr_full))
    max_val = corr_full[i_max]
    shift = i_max - (n - 1)          # shift k = i-(n-1)

    grad = np.zeros_like(h)
    if shift >= 0:
        left_len = n - shift
        grad[:left_len] += (1 - h[shift:])      # derivative w.r.t. left side
        grad[shift:]   += -h[:left_len]          # derivative w.r.t. right side
    else:
        s = -shift
        left_start = s
        left_len = n - s
        grad[left_start:] += (1 - h[:left_len])
        grad[:left_len]    += -h[left_start:]
    grad *= dx
    return grad, max_val, shift

        if val < best_c5 - 1e-9:
            best_h = h.copy()
            best_c5 = float(val)
            best_n = n

    # ----- Stage 1: many SLSQP restarts on the base grid -----
    max_restarts = 30
    for i in range(max_restarts):
        if time.time() - start_time > budget_s * 0.65:
            break
        if i == 0:
            start_vec = init_vec
        else:
            rnd = rng.random(n0)
            start_vec = project_to_feasible(rnd, n0 / 2.0)
        h_opt, _ = optimise_start(start_vec, maxiter=800)
        # local refinement (use a fraction of remaining time)
        remain = budget_s - (time.time() - start_time)
        if remain > 5.0:
            h_opt, val_opt = gradient_refine(h_opt, min(5.0, remain * 0.2))
        else:
            val_opt = float(evaluate_erdos_solution(h_opt))
        consider(h_opt, val_opt, n0)

    # ----- Stage 2: upscale to finer grids if time permits -----
    finer_grids = (128, 160, 200)
    for n2 in finer_grids:
        if time.time() - start_time > budget_s * 0.9:
            break
        if best_h is not None and best_n != n2:
            # interpolate current best to new grid
            x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
            h_start = np.interp(x_new, x_old, best_h, left=0.0, right=0.0)
        else:
            h_start = rng.random(n2)
        h_start = project_to_feasible(h_start, n2 / 2.0)

        h_opt, _ = optimise_start(h_start, maxiter=800)
        remain = budget_s - (time.time() - start_time)
        if remain > 2.0:
            h_opt, val_opt = gradient_refine(h_opt, min(3.0, remain * 0.2))
        else:
            val_opt = float(evaluate_erdos_solution(h_opt))
        consider(h_opt, val_opt, n2)

    # ----- fallback if nothing succeeded -----
    if best_h is None:
        best_h = np.full(n0, 0.5)
        best_c5 = float(evaluate_erdos_solution(best_h))
        best_n = n0

    return best_h, best_c5, best_n
