import time, math
import numpy as np
from scipy import optimize

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    # linear (zero‑padded) cross‑correlation of h with 1‑h
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _project_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Project h onto the hyperplane sum(h)=target while keeping 0≤h≤1."""
    # simple greedy redistribution – sufficient for our tiny problems
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # positive diff → need to add mass, take from entries <1
    # negative diff → need to remove mass, give to entries >0
    caps = (1.0 - h) if diff > 0 else (-h)
    for i in np.where(caps > 0)[0]:
        take = min(abs(diff), caps[i])
        take = math.copysign(take, diff)
        h[i] += take
        diff -= take
        if abs(diff) < 1e-12:
            break
    return h


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a feasible step function h, its C5 bound, and the number of points."""
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----------------------------------------------------------------------
    # 1. Initialise h (use provided construction if available)
    # ----------------------------------------------------------------------
    try:
        h = np.array(initial_h_values, dtype=float)  # type: ignore[name-defined]
    except Exception:
        n = 60
        h = np.full(n, 0.5, dtype=float)
    else:
        n = h.shape[0]

    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    h = _project_sum(h, target_sum)

    best_h = h.copy()
    best_c5 = _c5_value(best_h)

    # ----------------------------------------------------------------------
    # 2. Targeted hill‑climbing on the current maximal lag
    # ----------------------------------------------------------------------
    step = 0.3
    # allocate roughly 60 % of the budget to this phase
    phase_limit = start + budget_s * 0.60
    while time.time() < phase_limit:
        # evaluate current max lag
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_max = int(np.argmax(corr))
        cur_c5 = float(corr[idx_max] * (2.0 / n))
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
            # early stop if we beat the target record
            if best_c5 <= 0.3808:
                break

        lag = idx_max - (n - 1)          # integer shift corresponding to max lag

        if lag >= 0:
            # contributions for positive lag: h[i] * (1 - h[i+lag])
            i_range = np.arange(0, n - lag)
            contrib = h[i_range] * (1.0 - h[i_range + lag])
            i_star = int(np.argmax(contrib))
            src = i_range[i_star]
            dst = src + lag
        else:

            # contributions for negative lag: (1 - h[i]) * h[i+d]
            contrib = (1.0 - h[i_range]) * h[i_range + d]
            i_star = int(np.argmax(contrib))
            src = i_range[i_star] + d   # the later index holding the "1"

    return (best_h.tolist(), float(best_c5), int(n))
