def hill_climb(h, time_budget, start_time):
    best_h = h.copy()
    best_val = compute_c5(best_h)
    n = len(h)
    rng = np.random.default_rng()
    while time.time() - start_time < time_budget:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # pattern: move mass from i to j
        # if we allow continuous values, we need to determine max delta limited by [0,1] bounds.
        # Equivalent to earlier.
        hi, hj = best_h[i], best_h[j]
        # Determine direction: we can move delta from i to j if hi>0 and hj<1,
        # OR from j to i if hj>0 and hi<1.
        # We'll attempt both directions randomly.
        if hi > 0 and hj < 1:
            max_delta = min(hi, 1 - hj)
            direction = (i, j)
        elif hj > 0 and hi < 1:
            max_delta = min(hj, 1 - hi)
            direction = (j, i)
        else:
            continue
        if max_delta <= 0:
            continue
        # Choose delta random fraction maybe up to max_delta
        delta = max_delta * rng.random()
        # Store old values, update
        cand = best_h.copy()
        a, b = direction
        cand[a] -= delta
        cand[b] += delta
        cand_val = compute_c5(cand)
        if cand_val + 1e-12 < best_val:
            best_h = cand
            best_val = cand_val
    return best_h, best_val

import time
import numpy as np

def compute_c5(h):
    """Return C5 for a step function h given as a NumPy array."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def random_feasible(n, rng):
    """Return a random vector in [0,1]^n with exact sum = n/2."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    # simple projection to keep bounds
    for _ in range(10):
        over = v > 1.0
        under = v < 0.0
        if not over.any() and not under.any():
            break
        excess = (v[over] - 1.0).sum()
        deficit = (-v[under]).sum()
        interior = (~over) & (~under)
        if excess > 0 and interior.any():
            v[interior] -= excess / interior.sum()
        if deficit > 0 and interior.any():
            v[interior] += deficit / interior.sum()
        v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    target = n / 2.0
    if abs(cur - target) > 1e-12:
        diff = target - cur
        mask = (v > 0.0) & (v < 1.0)
        if mask.any():
            v[mask] += diff / mask.sum()
            v = np.clip(v, 0.0, 1.0)
        else:
            v *= target / cur
    return v

def hill_climb(h_start, time_budget, rng):
    """Greedy pairwise swaps (continuous values) to lower C5."""
    t0 = time.time()
    n = h_start.shape[0]
    cur_h = h_start.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    while time.time() - t0 < time_budget:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        hi, hj = cur_h[i], cur_h[j]
        # Direction: move mass from the larger to the smaller while staying in [0,1]
        if hi > 0.0 and hj < 1.0:
            max_delta = min(hi, 1.0 - hj)
            a, b = i, j
        elif hj > 0.0 and hi < 1.0:
            max_delta = min(hj, 1.0 - hi)
            a, b = j, i
        else:
            continue
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random()
        cand = cur_h.copy()
        cand[a] -= delta
        cand[b] += delta
        cand_val = compute_c5(cand)
        if cand_val + 1e-12 < cur_val:
            cur_h, cur_val = cand, cand_val
            if cur_val < best_val:
                best_h, best_val = cur_h.copy(), cur_val
    return best_h, best_val

def simulated_annealing(start_h, rng, time_limit):
    """Binary swap annealing (preserves exactly n/2 ones)."""
    t0 = time.time()
    n = start_h.shape[0]
    cur_h = start_h.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    while time.time() - t0 < time_limit:
        i = rng.integers(n)
        j = rng.integers(n)
        if cur_h[i] == cur_h[j]:
            continue
        cand = cur_h.copy()
        cand[i], cand[j] = cand[j], cand[i]
        new_val = compute_c5(cand)
        delta = new_val - cur_val
        elapsed = time.time() - t0
        frac = elapsed / time_limit
        T = max(1e-6, 0.02 * (1.0 - frac))  # linear cooling
        if delta <= 0.0 or rng.random() < np.exp(-delta / T):
            cur_h, cur_val = cand, new_val
            if cur_val < best_val:
                best_h, best_val = cur_h.copy(), cur_val
    return best_h, best_val
