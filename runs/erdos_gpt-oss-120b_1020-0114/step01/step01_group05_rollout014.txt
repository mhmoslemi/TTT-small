    grad = np.full(n, dx)  # base term from derivative of sum_h dx

    n_target = 200
    # ensure even
    if n_target % 2 != 0:
        n_target += 1
    n = n_target

            if c5_new <= cur_c5:

                h = h_new

                improved = True
                # adjust learning rate upwards if good
                lr = min(lr * 1.05, max_lr)
                break

            lr = max(lr * 0.5, min_lr)

import time
import numpy as np

def _evaluate(h):
    n = h.shape[0]
    dx = 2.0 / n
    # overlap integral for all integer shifts
    O = np.correlate(h, 1 - h, mode="full") * dx
    return O.max()


def _project(v, target):
    """Projection onto 0≤x≤1 and sum(x)=target via a simple bisection."""
    lo, hi = -1.0, 1.0
    for _ in range(35):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _subgrad_max_O(h):
    """Sub‑gradient of max_k ∫h(x)(1−h(x+k))dx."""
    n = h.shape[0]
    dx = 2.0 / n
    O = np.correlate(h, 1 - h, mode="full") * dx
    max_val = O.max()
    idx = np.where(O >= max_val - 1e-12)[0]
    shifts = idx - (n - 1)

    grad = np.full(n, dx)            # derivative of ∫h = dx
    for k in shifts:
        if k == 0:
            grad -= 2.0 * h * dx
        elif k > 0:
            grad[: n - k] -= h[k:] * dx
            grad[k:]       -= h[: n - k] * dx
        else:  # k < 0
            kk = -k
            grad[kk:]     -= h[: n - kk] * dx
            grad[: n - kk] -= h[kk:] * dx
    grad /= len(shifts)               # average if several maxima
    return grad


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # ----------------------------------------------------------------------
    # choose grid size (larger gives more freedom, still cheap)
    n = 200
    if n % 2:
        n += 1
    target_sum = n / 2.0                     # ∫h = 1  →  Σh = n/2

    # ----------------------------------------------------------------------
    # initialise h – interpolate if a parent solution is supplied
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        m = init.size
        x_old = np.linspace(0.0, 2.0, m, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x_new, x_old, init)
    else:
        h = np.random.rand(n)
    h = _project(h, target_sum)

    best_h = h.copy()
    best_c5 = _evaluate(best_h)

    # ----------------------------------------------------------------------
    # parameters for the projected sub‑gradient descent
    lr = 0.2
    min_lr = 1e-5
    max_lr = 1.0
    it = 0
    last_improve = 0

    while True:
        if time.time() - start > budget_s * 0.90:
            break

        # current objective
        cur_c5 = _evaluate(h)

        # compute sub‑gradient of the current maximal overlap
        g = _subgrad_max_O(h)

        # simple back‑tracking line search
        step = lr
        improved = False
        for _ in range(12):
            h_new = _project(h - step * g, target_sum)
            c5_new = _evaluate(h_new)
            if c5_new <= cur_c5:                # accept non‑worsening move
                h = h_new
                cur_c5 = c5_new
                improved = True
                lr = min(lr * 1.07, max_lr)
                break
            step *= 0.5
        if not improved:
            lr = max(lr * 0.5, min_lr)

        # record best solution
        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            best_h = h.copy()
            last_improve = it

        # occasional jitter to escape shallow plateaus
        if it - last_improve > 3000 and (it % 400 == 0):
            jitter = np.random.normal(scale=0.015, size=n)
            h_j = _project(h + jitter, target_sum)
            c5_j = _evaluate(h_j)
            if c5_j < best_c5:
                h = h_j
                best_h = h_j.copy()
                best_c5 = c5_j
                last_improve = it

        # very long stagnation → random re‑initialisation (keeps the best)
        if it - last_improve > 12000 and (it % 2000 == 0):
            h_rand = np.random.rand(n)
            h_rand = _project(h_rand, target_sum)
            c5_rand = _evaluate(h_rand)
            if c5_rand < best_c5:
                h = h_rand
                best_h = h_rand.copy()
                best_c5 = c5_rand
                last_improve = it
            else:
                # keep exploring from current point but with a small shock
                shock = np.random.normal(scale=0.03, size=n)
                h = _project(h + shock, target_sum)

        it += 1

    # ----------------------------------------------------------------------
    # fine‑grained local search (pairwise mass swaps)
    h = best_h.copy()
    cur_c5 = best_c5
    delta_max = 0.04
    attempts = 25000
    for _ in range(attempts):
        if time.time() - start > budget_s * 0.99:
            break
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # feasible transfer magnitude
        hi, hj = h[i], h[j]
        max_up = min(delta_max, hi, 1.0 - hj)
        max_down = min(delta_max, hj, 1.0 - hi)
        # pick signed delta (positive → move from i to j)
        if max_up + max_down == 0:
            continue
        delta = np.random.uniform(-max_down, max_up)
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        # bounds already respected by construction
        c5_new = _evaluate(h_new)
        if c5_new < cur_c5:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()

    # final projection to guarantee feasibility
    best_h = _project(best_h, target_sum)
    best_c5 = _evaluate(best_h)

    return best_h, best_c5, n
