def _project(v, target):
    v = np.clip(v, 0.0, 1.0)
    # quick check sum:
    s = v.sum()
    # if sum within tolerance -> ok
    if abs(s - target) < 1e-10:
        return v
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = (lo + hi)/2
        adj = np.clip(v + mid, 0.0, 1.0)
        s_adj = adj.sum()
        if s_adj > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi)/2, 0.0, 1.0)

def _softmax_gradient(h, T, idx):
    # compute correlation C
    n = h.shape[0]
    dx = 2.0 / n
    C = np.correlate(h, 1 - h, mode='full') * dx  # length 2n-1
    # compute softmax weights
    C_max = C.max()
    exp_vals = np.exp((C - C_max) / T)
    p = exp_vals / exp_vals.sum()
    # matrix of weights P shape n x n
    P = p[idx]
    grad = dx * (P @ (1 - h) - P.T @ h)
    return grad, C.max()

def _softmax_grad(h, T, idx):
    n = h.shape[0]
    dx = 2.0 / n
    C = np.correlate(h, 1 - h, mode='full') * dx
    Cmax = C.max()

    p = exp_vals / exp_vals.sum()
    P = p[idx]
    grad = dx * (P @ (1 - h) - P.T @ h)
    return grad, Cmax

import time
import numpy as np

# -------------------------------------------------------------
# Basic linear‐algebra helpers
# -------------------------------------------------------------
def _evaluate(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1‑h(x+k))dx approximated by correlation."""
    n = h.size
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode="full") * dx)


def _project(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0≤x≤1, sum x = target} by clipping then a water‑filling bisection."""
    v = np.clip(v, 0.0, 1.0)
    if abs(v.sum() - target) < 1e-12:
        return v
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        if adj.sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


# -------------------------------------------------------------
# Soft‑max (smooth max) gradient
# -------------------------------------------------------------
def _softmax_grad(h: np.ndarray, T: float, idx: np.ndarray) -> tuple:
    """Gradient of the smoothed max objective for temperature T."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    cmax = corr.max()
    # stable softmax
    shifted = corr - cmax
    expv = np.exp(shifted / T)
    p = expv / expv.sum()
    P = p[idx]                     # matrix of weights for each i‑j shift
    grad = dx * (P @ (1 - h) - P.T @ h)
    return grad, cmax


# -------------------------------------------------------------
# Sub‑gradient (max‑shift) utilities
# -------------------------------------------------------------
def _max_shift_gradient(h: np.ndarray, dx: float) -> tuple:
    """Gradient of the *worst* shift (any maximiser of C_k). Returns average of all active shifts."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full") * dx
    vmax = corr.max()
    active = np.where(corr >= vmax - 1e-12)[0]
    g = np.zeros_like(h)
    for idx in active:
        k = idx - (n - 1)
        if k > 0:
            g[: n - k] += (1 - h[k:])
            g[k:]     -= h[: n - k]
        elif k < 0:
            kp = -k
            g[kp:]   += (1 - h[: n - kp])
            g[: n - kp] -= h[kp:]
    if active.size:
        g = g * dx / active.size
    return g, vmax

# Optimiser: smooth‑max gradient descent with Adam + back‑track
# -------------------------------------------------------------
def _optimize_softmax(
    n: int,
    max_iters: int,
    start_h: np.ndarray = None,
    budget: float = 1e9,
) -> tuple:
    target = n / 2.0
    dx = 2.0 / n
    idx = (np.arange(n)[:, None] - np.arange(n)[None, :]) + (n - 1)

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # initialise
    if start_h is not None and start_h.shape[0] == n:
        h = _project(start_h.astype(float).copy(), target)
    else:
        h = _project(np.random.rand(n), target)

    best_h = h.copy()
    best_c = _evaluate(h)

    # hyper‑parameters
    base_lr = 0.5
    T0, Tmin = 0.06, 0.001
    last_improve = 0

    start_time = time.time()
    for it in range(max_iters):
        if time.time() - start_time > budget:
            break

        T = max(Tmin, T0 * (1.0 - it / max_iters))
        g, cur_c = _softmax_grad(h, T, idx)

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        # back‑track line search on the *actual* max‑overlap objective
        step_try = step.copy()
        for _ in range(10):
            h_new = _project(h - step_try, target)
            new_c = _evaluate(h_new)
            if new_c <= cur_c:
                h = h_new
                cur_c = new_c
                break
            step_try *= 0.5
        else:
            # if line‑search failed, keep old h
            pass

        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
            last_improve = it

        # occasional jitter to escape plateaus
        if (it - last_improve) > 3000 and (it % 500 == 0):
            jitter = np.random.normal(scale=0.02, size=n)
            h = _project(h + jitter, target)

        if best_c <= 0.3808:
            break

    return best_h, best_c

# Optimiser: pure sub‑gradient (max‑shift) descent
# -------------------------------------------------------------
def _optimize_subgrad(
    n: int,
    max_iters: int,
    start_h: np.ndarray = None,
    budget: float = 1e9,
) -> tuple:
    target = n / 2.0
    dx = 2.0 / n
    if start_h is not None and start_h.shape[0] == n:
        h = _project(start_h.astype(float).copy(), target)
    else:
        h = _project(np.random.rand(n), target)

    best_h = h.copy()
    best_c = _evaluate(h)
    last_imp = 0
    start_time = time.time()

    for it in range(max_iters):
        if time.time() - start_time > budget:
            break
        g, cur_c = _max_shift_gradient(h, dx)
        # diminishing step size
        step = 0.5 / np.sqrt(it + 1.0)
        h_new = _project(h - step * g, target)
        new_c = _evaluate(h_new)
        # accept if not worse
        if new_c <= cur_c:
            h = h_new
            cur_c = new_c
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
            last_imp = it
        if (it - last_imp) > 2500 and (it % 400 == 0):
            h = _project(h + np.random.normal(scale=0.015, size=n), target)
        if best_c <= 0.3808:
            break
    return best_h, best_c


# -------------------------------------------------------------
# Optimiser: binary simulated‑annealing (swap moves)
# -------------------------------------------------------------
def _optimize_binary(
    n: int,
    max_iters: int,
    budget: float = 1e9,
) -> tuple:
    if n % 2 != 0:
        n += 1
    target = n // 2  # integer number of ones
    dx = 2.0 / n

    # start from a random binary mask
    bits = np.zeros(n, dtype=float)
    bits[:target] = 1.0
    np.random.shuffle(bits)

    cur_c = _evaluate(bits)
    best_bits = bits.copy()
    best_c = cur_c

    # annealing schedule
    T0, Tmin = 0.08, 0.0005
    start_time = time.time()

    # pre‑compute the indices of ones/zeros for fast swaps
    for it in range(max_iters):
        if time.time() - start_time > budget:
            break
        T = max(Tmin, T0 * (1.0 - it / max_iters))
        # propose a swap
        ones = np.flatnonzero(bits == 1.0)
        zeros = np.flatnonzero(bits == 0.0)
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        bits[i], bits[j] = 0.0, 1.0      # swap
        new_c = _evaluate(bits)
        delta = new_c - cur_c
        accept = False
        if delta <= 0:
            accept = True
        else:
            accept = np.random.rand() < np.exp(-delta / T)
        if accept:
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_bits = bits.copy()
        else:
            # revert swap
            bits[i], bits[j] = 1.0, 0.0

        if best_c <= 0.3808:
            break

    return best_bits, best_c


# -------------------------------------------------------------
# Main driver
# -------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) with an improved upper bound for C₅."""
    np.random.seed(seed)
    start = time.time()
    time_budget = budget_s * 0.95          # leave a small safety margin

    # ---------------------------------------------------------
    # try a few problem sizes, larger sizes give more freedom
    # ---------------------------------------------------------
    candidates = [80, 100, 120, 150, 200]
    best_h = None
    best_c = np.inf
    best_n = None

    # Use the supplied initial vector (if any) for one run
    init_vec = None
    try:
        init_vec = np.asarray(initial_h_values, dtype=float)  # provided by harness
    except Exception:
        init_vec = None

    # allocate time for the three different optimisation strategies
    remaining = time_budget
    # 1) soft‑max + Adam
    for n in candidates:
        if remaining <= 0:
            break
        max_iters = 4000
        # allocate a slice of the remaining budget
        slice_budget = min(remaining, 0.3 * time_budget)
        h, c = _optimize_softmax(
            n, max_iters, start_h=init_vec if init_vec is not None and init_vec.size == n else None, budget=slice_budget
        )
        if c < best_c:
            best_c, best_h, best_n = c, h, n
        remaining -= slice_budget

    # 2) pure sub‑gradient descent (uses the best so far as start if sizes match)
    for n in candidates:
        if remaining <= 0:
            break
        max_iters = 8000
        slice_budget = min(remaining, 0.35 * time_budget)
        start_vec = best_h if best_h is not None and best_h.size == n else None
        h, c = _optimize_subgrad(
            n, max_iters, start_h=start_vec, budget=slice_budget
        )
        if c < best_c:
            best_c, best_h, best_n = c, h, n
        remaining -= slice_budget

    # 3) binary simulated annealing – works only on even n
    for n in candidates:
        if remaining <= 0:
            break
        max_iters = 20000
        slice_budget = min(remaining, 0.25 * time_budget)
        h, c = _optimize_binary(n, max_iters, budget=slice_budget)
        if c < best_c:
            best_c, best_h, best_n = c, h, n
        remaining -= slice_budget

    # fallback – uniform function (should never be needed)
    if best_h is None:
        best_n = 80
        best_h = np.full(best_n, 0.5)
        best_c = _evaluate(best_h)

    # final sanity checks
    best_h = np.clip(best_h, 0.0, 1.0)
    # enforce integral constraint exactly (tiny numerical drift possible)
    best_h = _project(best_h, best_n / 2.0)
    best_c = _evaluate(best_h)

    return best_h, best_c, best_n
