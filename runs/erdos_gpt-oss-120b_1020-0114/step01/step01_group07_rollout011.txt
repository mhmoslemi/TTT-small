                free = v < 1.0 - 1e-12

import numpy as np
import time
import math

def _project(v: np.ndarray, target: float) -> np.ndarray:
    """Project vector onto the intersection of the box [0,1] and the hyperplane sum=v_target."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    # water‑filling style loop – converges in a few steps
    while abs(diff) > 1e-12:
        if diff > 0:
            free = v < 1.0 - 1e-12
            nfree = free.sum()
            if nfree == 0:
                break
            add = diff / nfree
            v[free] = np.minimum(v[free] + add, 1.0)
        else:
            free = v > 1e-12
            nfree = free.sum()
            if nfree == 0:
                break
            sub = diff / nfree
            v[free] = np.maximum(v[free] + sub, 0.0)
        diff = target - v.sum()
    return v


def _c5_of(h: np.ndarray) -> float:
    """Overlap constant for a discrete step vector."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)


def _repeat_intervals(vals: np.ndarray, block_len: int) -> np.ndarray:
    """Expand a short interval vector to a full step array."""
    return np.repeat(vals, block_len)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s * 0.95

    best_val = np.inf
    best_h = None
    best_n = None
