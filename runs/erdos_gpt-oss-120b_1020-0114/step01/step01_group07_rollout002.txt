def _project(v, target):
    v = np.clip(v, 0.0, 1.0)
    # water-filling: adjust sum to target while respecting bounds.
    # We'll use simple iterative approach:
    diff = target - v.sum()
    # tolerance
    tol = 1e-12
    max_iter = 1000
    for _ in range(max_iter):
        if abs(diff) <= tol:
            break
        # Determine indices that can still be increased (if diff>0) or decreased (if diff<0)
        if diff > 0:
            free = v < 1 - tol
            nfree = free.sum()
            if nfree == 0:
                break
            add = diff / nfree
            v = np.where(free, np.minimum(v + add, 1.0), v)
        else:
            free = v > tol
            nfree = free.sum()
            if nfree == 0:
                break
            sub = diff / nfree
            v = np.where(free, np.maximum(v + sub, 0.0), v)
        diff = target - v.sum()
    # final clipping
    v = np.clip(v, 0.0, 1.0)
    # In case still off due to rounding, we could adjust by distributing residual to first entries.
    residual = target - v.sum()
    if abs(residual) > 1e-9:
        # adjust the first index with room
        for i in range(len(v)):
            if residual > 0 and v[i] < 1.0 - 1e-9:
                add = min(residual, 1.0 - v[i])
                v[i] += add
                residual -= add
            elif residual < 0 and v[i] > 1e-9:
                sub = min(-residual, v[i])
                v[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return v

import numpy as np

def _project(v: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust to obtain exactly the given sum."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    tol = 1e-12
    for _ in range(1000):
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = v < 1.0 - tol
            n = free.sum()
            if n == 0:
                break
            add = diff / n
            v = np.where(free, np.minimum(v + add, 1.0), v)
        else:
            free = v > tol
            n = free.sum()
            if n == 0:
                break
            sub = diff / n
            v = np.where(free, np.maximum(v + sub, 0.0), v)
        diff = target - v.sum()
    # tiny residual fix
    residual = target - v.sum()
    if abs(residual) > 1e-12:
        for i in range(v.size):
            if residual > 0 and v[i] < 1.0 - 1e-12:
                add = min(residual, 1.0 - v[i])
                v[i] += add
                residual -= add
            elif residual < 0 and v[i] > 1e-12:
                sub = min(-residual, v[i])
                v[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return v


def _c5_of(h: np.ndarray) -> float:
    """Overlap constant for a vector h."""
    N = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / N)

# Local descent guided by the worst shift(s)
# ----------------------------------------------------------------------
def _local_descent(
    h: np.ndarray,
    target_sum: float,
    slice_time: float,
    slice_start: float,
) -> tuple[np.ndarray, float]:
    N = h.size
    best_h = h.copy()
    best_val = _c5_of(best_h)

    max_step_initial = 0.5  # maximal mass moved in a single step
    stagnation = 0

    end_time = slice_start + slice_time
    while time.time() < end_time:
        # correlation and the worst shift(s)
        corr = np.correlate(h, 1.0 - h, mode="full")
        cur_max = corr.max()
        # mask of shifts attaining the max (within tolerance)
        mask = np.isclose(corr, cur_max, atol=1e-12)

        # build reduction potential for every index (sum over all worst shifts)
        dec = np.zeros(N)
        for idx in np.flatnonzero(mask):
            ell = idx - (N - 1)  # lag
            if ell >= 0:
                i_vals = np.arange(0, N - ell, dtype=int)
                dec[i_vals] += 1.0 - h[i_vals + ell]
            else:
                i_vals = np.arange(-ell, N, dtype=int)
                dec[i_vals] += 1.0 - h[i_vals + ell]

        # pick i (to decrease) and j (to increase)
        cand_i = np.where(h > 1e-12)[0]

        if cand_i.size == 0 or cand_j.size == 0:
            # nothing to move – jitter a bit and continue
            h = _project(h + np.random.uniform(-0.01, 0.01, size=N), target_sum)
            continue

        i_best = cand_i[np.argmax(dec[cand_i])]
        j_best = cand_j[np.argmin(dec[cand_j])]

        if i_best == j_best:
            # pick the next best j
            sorted_j = cand_j[np.argsort(dec[cand_j])]
            for jj in sorted_j:
                if jj != i_best:
                    j_best = jj
                    break

        # dynamic step bound (shrinks as time runs out)
        time_left = end_time - time.time()
        if time_left <= 0:

        new_h = h.copy()

        new_h = _project(new_h, target_sum)

            best_h = new_h.copy()
            h = new_h
