def gradient_of_softmax_max(h, alpha):
    n = len(h)
    dx = 2.0/n
    # correlation
    corr = np.correlate(h, 1.0 - h, mode='full')
    g = corr * dx
    # compute softmax weights
    maxg = np.max(g)
    exp_alpha = np.exp(alpha * (g - maxg))
    sumexp = np.sum(exp_alpha)
    w = exp_alpha / sumexp
    # compute forward and backward convolution:
    w_rev = w[::-1]
    conv_fwd = np.convolve(w_rev, h, mode='full')
    conv_bwd = np.convolve(w, h, mode='full')
    mid = n - 1
    forward = conv_fwd[mid:mid+n]
    backward = conv_bwd[mid:mid+n]
    grad = dx * (1.0 - forward - backward)
    return grad, g.max()

import time, math, numpy as np

# ---------- basic evaluation ----------
def _c5_of(h: np.ndarray) -> float:
    """Exact C5 value for a discrete step function."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)


# ---------- projection onto box‑simplex ----------
def _proj_box_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto {x | 0≤x≤1, sum x = target}."""
    lo = -np.max(v) - 2.0
    hi = 1.0 - np.min(v) + 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


# ---------- gradient of smooth max ----------
def _grad_softmax_max(h: np.ndarray, alpha: float):
    """Gradient of  (1/α)·log Σexp(α·g_s)  where g_s are the shift overlaps."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")          # length 2n‑1
    g = corr * dx
    mx = g.max()
    w = np.exp(alpha * (g - mx))
    w /= w.sum()                                          # softmax weights, sum=1

    # forward (h shifted left) and backward (right) contributions via convolution
    w_rev = w[::-1]
    conv_fwd = np.convolve(w_rev, h, mode="full")
    conv_bwd = np.convolve(w, h, mode="full")
    mid = n - 1
    forward = conv_fwd[mid:mid + n]
    backward = conv_bwd[mid:mid + n]

    grad = dx * (1.0 - forward - backward)                # ∂f/∂h
    return grad, g.max()


# ---------- continuous optimisation ----------
def _optimize_continuous(h0: np.ndarray, time_limit: float):
    """Projected stochastic gradient descent on the smooth‑max surrogate."""
    n = h0.shape[0]
    target = n / 2.0
    cur = np.clip(h0, 0.0, 1.0)
    cur = _proj_box_simplex(cur, target)
    cur_val = _c5_of(cur)

    best = cur.copy()
    best_val = cur_val

    start = time.time()
    it = 0
    step0 = 0.6
    T = 0.12
    max_it_est = 20000

    while time.time() - start < time_limit:
        alpha = 5.0 + 45.0 * min(it / max_it_est, 1.0)          # from smooth to sharp
        grad, _ = _grad_softmax_max(cur, alpha)
        gn = np.linalg.norm(grad)
        if gn == 0.0:
            break
        step = step0 / math.sqrt(it + 1) / (gn + 1e-12)
        nxt = cur - step * grad
        nxt = _proj_box_simplex(nxt, target)

        nxt_val = _c5_of(nxt)

        if nxt_val < cur_val - 1e-12:
            cur, cur_val = nxt, nxt_val
            if nxt_val < best_val:
                best, best_val = nxt.copy(), nxt_val
        else:
            # Metropolis‑type acceptance
            if T > 0.0 and np.random.rand() < math.exp(-(nxt_val - cur_val) / T):
                cur, cur_val = nxt, nxt_val
        T *= 0.9995
        it += 1

        # occasional random mass exchange to diversify
        if it % 500 == 0:
            i, j = np.random.randint(n, size=2)
            if i != j:
                max_tr = min(cur[i], 1.0 - cur[j])
                if max_tr > 0:
                    delta = max_tr * np.random.rand() * 0.5
                    cur[i] -= delta
                    cur[j] += delta
                    cur = _proj_box_simplex(cur, target)

    return best, best_val

def _refine_greedy(h0: np.ndarray, time_limit: float):

    n = h0.shape[0]
    target = n / 2.0

            if rem > 0.5:

            if v_opt < best_val:
                best_h, best_val, best_n = h_opt.copy(), v_opt, n

    # final refinement with any leftover budget
    remaining = budget_s - (time.time() - start_all)
    if remaining > 2.0:
        best_h, best_val = _refine_greedy(best_h, remaining)

    # final projection to guarantee feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _proj_box_simplex(best_h, best_n / 2.0)
    final_val = _c5_of(best_h)

    return best_h, float(final_val), int(best_n)
