import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# -------------------------------------------------------------
# Core utilities
# -------------------------------------------------------------
def compute_c5_fft(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function using FFT for speed."""
    n = h.shape[0]
    dx = 2.0 / n
    L = 2 * n                     # zero‑padding length for linear correlation
    H = np.fft.fft(h, L)
    Z = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(H * np.conj(Z)).real[: 2 * n - 1] * dx
    return float(corr.max())


def smoothed_objective(h: np.ndarray, alpha: float) -> float:
    """Log‑sum‑exp smoothed version of the max overlap (used by SLSQP)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    # logsumexp is numerically stable
    sm = logsumexp(alpha * corr) / alpha
    return sm * (2.0 / h.shape[0])


def enforce_constraints(h: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1] and renormalise to the required integral."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s != 0:
        h = h * (target_sum / s)
    h = np.clip(h, 0.0, 1.0)
    # tiny residual correction
    s = h.sum()
    diff = target_sum - s
    if abs(diff) > 1e-12:
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if s != 0:
            h = h * (target_sum / s)
    return h


def upsample(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a finer grid while preserving the integral."""
    old_n = h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, h)
    target = new_n / 2.0
    return enforce_constraints(new_h, target)


# -------------------------------------------------------------
# Main routine
# -------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    def elapsed() -> float:
        return time.time() - start

    # ---------------------------------------------------------
    # Initialise – use the supplied construction if present
    # ---------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
        target_sum = n / 2.0
        best_h = enforce_constraints(h0, target_sum)
    else:
        n = 200
        target_sum = n / 2.0
        # half‑filled binary start
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        rng.shuffle(best_h)

    best_c5 = compute_c5_fft(best_h)

    # ---------------------------------------------------------
    # Phase 1 – a few SLSQP refinements with a smoothed objective
    # ---------------------------------------------------------
    max_restarts = 5
    alpha_smooth = 80.0                     # strong smoothing → close to true max
    for _ in range(max_restarts):
        if elapsed() > budget_s * 0.45:
            break
        # perturb the current best solution
        pert = best_h + rng.uniform(-0.05, 0.05, size=n)
        pert = enforce_constraints(pert, target_sum)

        # optimisation
        cons = {"type": "eq", "fun": lambda x: np.sum(x) - target_sum}
        bounds = [(0.0, 1.0)] * n

        def obj(x):
            return smoothed_objective(x, alpha_smooth)

        res = minimize(
            obj,
            pert,
            method="SLSQP",
            bounds=bounds,
            constraints=[cons],
            options={"maxiter": 500, "ftol": 1e-9, "disp": False},
        )
        h_opt = enforce_constraints(res.x, target_sum)
        c5_opt = compute_c5_fft(h_opt)
        if c5_opt < best_c5:
            best_c5, best_h = c5_opt, h_opt.copy()

    # ---------------------------------------------------------
    # Phase 2 – simulated annealing / hill‑climbing on the true C5
    # ---------------------------------------------------------
    cur_h = best_h.copy()
    cur_c5 = best_c5
    target_sum = n / 2.0

    # annealing schedule – linear cooling, epsilon shrinks linearly
    max_steps = 500_000
    step = 0
    T0 = 0.010          # initial temperature
    eps_start = 0.10
    eps_end = 0.001

    while step < max_steps and elapsed() < budget_s * 0.90:
        progress = step / max_steps
        T = T0 * (1.0 - progress)            # cool down
        eps_factor = eps_start * (1.0 - progress) + eps_end * progress

        i = rng.integers(0, n)
        if cur_h[i] <= 0.0:
            step += 1
            continue
        j = rng.integers(0, n)
        if cur_h[j] >= 1.0 or i == j:
            step += 1
            continue

        eps_max = min(cur_h[i], 1.0 - cur_h[j])
        if eps_max <= 0.0:
            step += 1
            continue

        eps = eps_max * rng.random() * eps_factor
        if eps <= 0.0:
            step += 1
            continue

        # apply tentative move
        cur_h[i] -= eps
        cur_h[j] += eps
        new_c5 = compute_c5_fft(cur_h)
        delta = new_c5 - cur_c5

        accept = False
        if delta < 0.0:
            accept = True
        else:
            # Metropolis acceptance
            prob = np.exp(-delta / (T + 1e-16))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, cur_h.copy()
        else:
            # revert
            cur_h[i] += eps
            cur_h[j] -= eps

        step += 1

    # ---------------------------------------------------------
    # Phase 3 – pure greedy refinement (no uphill moves)
    # ---------------------------------------------------------
    extra_steps = 200_000
    step = 0
    while step < extra_steps and elapsed() < budget_s * 0.96:
        i = rng.integers(0, n)
        if cur_h[i] <= 0.0:
            step += 1
            continue
        j = rng.integers(0, n)
        if cur_h[j] >= 1.0 or i == j:
            step += 1
            continue

        eps_max = min(cur_h[i], 1.0 - cur_h[j])
        eps = eps_max * rng.random() * 0.02   # tiny adjustments
        if eps <= 0.0:
            step += 1
            continue

        cur_h[i] -= eps
        cur_h[j] += eps
        new_c5 = compute_c5_fft(cur_h)

        if new_c5 < cur_c5 - 1e-12:
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, cur_h.copy()
        else:
            # revert
            cur_h[i] += eps
            cur_h[j] -= eps

        step += 1

    # ---------------------------------------------------------
    # Optional up‑sampling to a finer grid if time permits
    # ---------------------------------------------------------
    if elapsed() < budget_s * 0.98 and n < 400:
        n2 = 400
        up_h = upsample(best_h, n2)
        target_sum2 = n2 / 2.0

        # brief local optimisation at the finer resolution
        # use the same SLSQP machinery but with fewer restarts
        for _ in range(3):
            if elapsed() > budget_s * 0.99:
                break
            pert = up_h + rng.uniform(-0.04, 0.04, size=n2)
            pert = enforce_constraints(pert, target_sum2)

            cons = {"type": "eq", "fun": lambda x: np.sum(x) - target_sum2}
            bounds = [(0.0, 1.0)] * n2

            def obj(x):
                return smoothed_objective(x, alpha_smooth)

            res = minimize(
                obj,
                pert,
                method="SLSQP",
                bounds=bounds,
                constraints=[cons],
                options={"maxiter": 300, "ftol": 1e-9, "disp": False},
            )
            cand = enforce_constraints(res.x, target_sum2)
            cand_c5 = compute_c5_fft(cand)
            if cand_c5 < best_c5:
                best_c5, best_h = cand_c5, cand.copy()
                n = n2                  # remember the new size

        # a short annealing phase on the finer grid
        cur_h = best_h.copy()
        cur_c5 = best_c5
        max_steps_fine = 150_000
        step = 0
        T0_fine = 0.008
        eps_start_fine = 0.08
        eps_end_fine = 0.0008
        while step < max_steps_fine and elapsed() < budget_s * 0.995:
            prog = step / max_steps_fine
            T = T0_fine * (1.0 - prog)
            eps_factor = eps_start_fine * (1.0 - prog) + eps_end_fine * prog

            i = rng.integers(0, n)
            if cur_h[i] <= 0.0:
                step += 1
                continue
            j = rng.integers(0, n)
            if cur_h[j] >= 1.0 or i == j:
                step += 1
                continue

            eps_max = min(cur_h[i], 1.0 - cur_h[j])
            if eps_max <= 0.0:
                step += 1
                continue
            eps = eps_max * rng.random() * eps_factor
            if eps <= 0.0:
                step += 1
                continue

            cur_h[i] -= eps
            cur_h[j] += eps
            new_c5 = compute_c5_fft(cur_h)
            delta = new_c5 - cur_c5

            accept = False
            if delta < 0.0:
                accept = True
            else:
                prob = np.exp(-delta / (T + 1e-16))
                if rng.random() < prob:
                    accept = True

            if accept:
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5, best_h = cur_c5, cur_h.copy()
            else:
                cur_h[i] += eps
                cur_h[j] -= eps
            step += 1

    # ---------------------------------------------------------
    # Final cleanup and return
    # ---------------------------------------------------------
    best_h = enforce_constraints(best_h, best_h.size / 2.0)
    final_c5 = compute_c5_fft(best_h)

    return (best_h, float(final_c5), int(best_h.size))
