def project_sum_and_bounds(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == 0:
        return h
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    # final small correction
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
    return h
```

_GLOBAL_N = None

def set_global_params(n, alpha):
    global _GLOBAL_N, _GLOBAL_ALPHA
    _GLOBAL_N = n
    _GLOBAL_ALPHA = alpha

def smooth_objective(h):
    n = _GLOBAL_N
    alpha = _GLOBAL_ALPHA
    corr = np.correlate(h, 1.0 - h, mode='full')
    # compute smoothed max
    smooth = logsumexp(alpha * corr) / alpha
    return smooth * (2.0 / n)

import time
import numpy as np
import scipy.optimize as opt
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# helper utilities
# ----------------------------------------------------------------------
_GLOBAL_N = None
_GLOBAL_ALPHA = None


def set_global_params(n, alpha):
    """store parameters used by the smooth objective."""
    global _GLOBAL_N, _GLOBAL_ALPHA
    _GLOBAL_N = n
    _GLOBAL_ALPHA = alpha


def smooth_objective(h):
    """log‑sum‑exp smooth approximation of the maximum overlap."""
    n = _GLOBAL_N
    alpha = _GLOBAL_ALPHA
    corr = np.correlate(h, 1.0 - h, mode="full")
    smooth = logsumexp(alpha * corr) / alpha
    return smooth * (2.0 / n)


def constraint_sum_eq(h):
    """equality constraint  ∫h = 1  ⇔  sum(h) = n/2."""
    n = _GLOBAL_N
    return np.sum(h) - n / 2.0


def compute_c5(h):
    """exact C₅ for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)


def project_sum_and_bounds(vec, target):
    """clip to [0,1] and enforce exact sum = target (by scaling, then tiny fix)."""
    v = np.clip(vec, 0.0, 1.0)
    s = v.sum()
    if s > 0:
        v = v * (target / s)
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) > 1e-12:
        v += diff / v.size
        v = np.clip(v, 0.0, 1.0)
    return v


# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    """search for a step function with a small C₅ bound."""
    np.random.seed(seed)
    start = time.time()
    elapsed = lambda: time.time() - start

    # ------------------------------------------------------------------
    # initialise a few diverse starting vectors
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        # fall back: triangular shape (integral = 1)
        n_tmp = 200
        dx_tmp = 2.0 / n_tmp
        xs = (np.arange(n_tmp) + 0.5) * dx_tmp
        h0 = np.where(xs <= 1.0, xs, 2.0 - xs)

    n = h0.size
    target_sum = n / 2.0
    h0 = project_sum_and_bounds(h0, target_sum)

    # a random binary vector (half ones)
    h_bin = np.zeros(n)
    h_bin[np.random.choice(n, n // 2, replace=False)] = 1.0

    # a random continuous vector
    h_rand = np.random.rand(n)
    h_rand = project_sum_and_bounds(h_rand, target_sum)

    starters = [h0, h_bin, h_rand]

    best_h = None
    best_c5 = np.inf

    # ------------------------------------------------------------------
    # primary stochastic search (annealing + targeted swaps)
    # ------------------------------------------------------------------
    for start_vec in starters:
        h = start_vec.copy()
        cur_c5 = compute_c5(h)
        if cur_c5 < best_c5:
            best_c5, best_h = cur_c5, h.copy()

        # annealing schedule
        T = 0.008
        T_min = 1e-5
        decay = 0.9994
        max_move = 0.22

        # keep iterating while time permits
        while elapsed() < budget_s * 0.85:
            # find the shift giving the current maximum overlap
            corr_full = np.correlate(h, 1.0 - h, mode="full")
            s_idx = int(np.argmax(corr_full))
            s = s_idx - (n - 1)          # signed shift

            # propose a transfer along that worst shift
            if s != 0:
                if s > 0:
                    i_range = np.arange(n - s)
                    diff = h[i_range] - h[i_range + s]
                else:
                    s_abs = -s
                    i_range = np.arange(s_abs, n)
                    diff = h[i_range] - h[i_range - s_abs]
                cand = np.where(diff > 0)[0]
                if cand.size:
                    i = i_range[np.random.choice(cand)]
                    j = i + s
                else:
                    i, j = np.random.randint(0, n, 2)
            else:
                i, j = np.random.randint(0, n, 2)

            if i == j:
                continue

            hi, hj = h[i], h[j]
            # delta such that both stay in [0,1]
            lo = max(-hi, hj - 1.0)
            hi_bound = min(1.0 - hi, hj)
            if lo >= hi_bound:
                continue
            # limit the step size for stability
            lo = max(lo, -max_move)
            hi_bound = min(hi_bound, max_move)
            if lo >= hi_bound:
                continue
            delta = np.random.uniform(lo, hi_bound)

            new_h = h.copy()
            new_h[i] += delta
            new_h[j] -= delta
            new_h = np.clip(new_h, 0.0, 1.0)

            new_c5 = compute_c5(new_h)

            # Metropolis acceptance
            accept = False
            if new_c5 < cur_c5:
                accept = True
            else:
                if T > 0 and np.random.rand() < np.exp((cur_c5 - new_c5) / T):
                    accept = True

            if accept:
                h, cur_c5 = new_h, new_c5
                if cur_c5 < best_c5:
                    best_c5, best_h = cur_c5, h.copy()

            # temperature decay
            T = max(T * decay, T_min)

    # ------------------------------------------------------------------
    # final local refinement with a smoothed SLSQP optimisation
    # ------------------------------------------------------------------
    if elapsed() < budget_s * 0.96 and best_h is not None:
        n = best_h.size
        target_sum = n / 2.0
        set_global_params(n, 25.0)          # moderate smoothing
        bounds = [(0.0, 1.0)] * n
        cons = {"type": "eq", "fun": constraint_sum_eq}
        # a few short restarts
        for _ in range(3):
            if elapsed() > budget_s * 0.98:
                break
            guess = best_h.copy()
            # tiny random jitter to help escape plateaux
            jitter = np.random.uniform(-0.02, 0.02, size=n)
            guess = project_sum_and_bounds(guess + jitter, target_sum)

            res = opt.minimize(
                smooth_objective,
                guess,
                method="SLSQP",
                bounds=bounds,
                constraints=[cons],
                options={"maxiter": 300, "ftol": 1e-9, "disp": False},
            )
            h_opt = project_sum_and_bounds(res.x, target_sum)
            val = compute_c5(h_opt)
            if val < best_c5:
                best_c5, best_h = val, h_opt.copy()

    # ------------------------------------------------------------------
    # final safety projection and result
    # ------------------------------------------------------------------
    if best_h is None:
        # extremely unlikely, but fall back to a simple feasible vector
        best_h = np.full(n, 0.5)
        best_h = project_sum_and_bounds(best_h, target_sum)
        best_c5 = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_and_bounds(best_h, target_sum)
    final_c5 = compute_c5(best_h)

    return (best_h, float(final_c5), int(best_h.size))
