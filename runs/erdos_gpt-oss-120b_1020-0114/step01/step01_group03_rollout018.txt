import time
import numpy as np

# ------------------------------------------------------------
# Helper routines
# ------------------------------------------------------------
def compute_c5(h):
    """Exact C5 value for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)


def upsample_h(old_h, new_n):
    """Linear interpolation to a finer grid while preserving the integral."""
    old_n = old_h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    # force exact integral
    target = new_n / 2.0
    cur = new_h.sum()
    if cur > 0:
        new_h *= target / cur
    new_h = np.clip(new_h, 0.0, 1.0)
    # tiny correction for accumulated rounding error
    diff = target - new_h.sum()
    if abs(diff) > 1e-12:
        new_h += diff / new_n
        new_h = np.clip(new_h, 0.0, 1.0)
        cur = new_h.sum()
        if cur > 0:
            new_h *= target / cur
    return new_h

# Core optimizer (simulated annealing + targeted moves)
# ------------------------------------------------------------
def _local_search(start_h, time_allow):
    """Improve start_h within given time budget. Returns best found."""
    n = start_h.shape[0]
    dx = 2.0 / n
    cur_h = start_h.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    t0 = time.time()
    # schedule constants
    T_start = 0.02  # initial temperature
    # probability of using a targeted shift move
    prob_target = 0.18

    while time.time() - t0 < time_allow:
        # linearly decreasing temperature
        elapsed = (time.time() - t0) / time_allow
        T = T_start * (1.0 - elapsed)
        if np.random.rand() < prob_target:
            # ----- targeted move on the worst shift -----
            corr_full = np.correlate(cur_h, 1.0 - cur_h, mode="full") * dx
            max_idx = np.argmax(corr_full)
            k = max_idx - (n - 1)          # shift amount (may be negative)

            if k >= 0:

                if length == 0:
                    continue
                prod = cur_h[:length] * (1.0 - cur_h[k:])
                i = np.argmax(prod)
                if prod[i] <= 0.0:

                max_delta = min(cur_h[i], 1.0 - cur_h[i + k])
                if max_delta <= 0.0:
                    continue
                delta = np.random.rand() * max_delta
                h_new = cur_h.copy()
                h_new[i] -= delta
                h_new[i + k] += delta

                max_delta = min(cur_h[i], 1.0 - cur_h[i - k2])
                if max_delta <= 0.0:
                    continue
                delta = np.random.rand() * max_delta
                h_new = cur_h.copy()
                h_new[i] -= delta
                h_new[i - k2] += delta

        else:
            # ----- random mass–transfer move -----
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            if np.random.rand() < 0.5:

                h_new[i] -= delta

            if s != 0:
