def _smooth_max_step(h, lr, alpha):
    compute H etc.
    corr = np.dot(h, 1 - H)   # raw values (no scaling)
    # compute smooth max m = (1/alpha) * logsumexp(alpha*corr)  # not needed for gradient but final objective
    # compute p as softmax of alpha*corr
    max_corr = np.max(corr)
    exp_vals = np.exp(alpha * (corr - max_corr))
    sum_exp = np.sum(exp_vals)
    p = exp_vals / sum_exp

    # compute gradient
    G = 1.0 - np.tensordot(p, (H + H[::-1]), axes=([0], [0]))

    # clip and sum adjust

def _gradient_optimize(h0, time_limit):
    n = len(h0)
    pad = n - 1
    target_sum = n/2.0
    h = h0.copy()
    best_h = h.copy()
    best_val = _compute_c5(h)
    cur_val = best_val
    lr = 0.1
    alpha = 10.0
    start = time.time()
    iters = 0
    while time.time() - start < time_limit:
        # compute H
        h_ext = np.concatenate([np.zeros(pad), h, np.zeros(pad)])
        i = np.arange(n)
        shifts = np.arange(-pad, pad+1)
        idx = pad + i[None, :] + shifts[:, None]  # shape (2n-1,n)
        H = h_ext[idx]  # shape (2n-1,n)
        one_minus_H = 1.0 - H
        corr = np.dot(h, one_minus_H.T)  # shape (2n-1,)
        # softmax weights
        max_c = corr.max()
        exp_vals = np.exp(alpha * (corr - max_c))
        sum_exp = exp_vals.sum()
        p = exp_vals / sum_exp
        # gradient
        G = 1.0 - (p[:, None] * (H + H[::-1,:])).sum(axis=0)
        # propose new h
        new_h = h - lr * G
        # clip and enforce sum
        new_h = np.clip(new_h, 0.0, 1.0)
        diff = target_sum - new_h.sum()
        if abs(diff) > 1e-12:
            free = np.where((new_h > 1e-12) & (new_h < 1.0 - 1e-12))[0]
            if free.size > 0:
                new_h[free] += diff / free.size
                new_h = np.clip(new_h, 0.0, 1.0)
            else:
                new_h[0] = min(1.0, max(0.0, new_h[0] + diff))
                new_h = np.clip(new_h, 0.0, 1.0)
        new_val = _compute_c5(new_h)
        if new_val < cur_val - 1e-12:
            h = new_h
            cur_val = new_val
            lr *= 1.1
            if new_val < best_val:
                best_val, best_h = new_val, new_h.copy()
            # maybe reset loop
        else:
            # reject; reduce lr
            lr *= 0.5
            if lr < 1e-6:
                # reset lr to a small value; break?
                lr = 1e-6
            # maybe try random perturbation occasionally
        iters += 1
        if iters % 200 == 0:
            # occasional random perturb to escape plateau
            if np.random.rand() < 0.3:
                # random shuffle small amount: add small gaussian noise and project
                h_noise = h + (np.random.randn(n) * 0.02)  # small noise
                h_noise = np.clip(h_noise, 0.0, 1.0)
                # enforce sum
                diff = target_sum - h_noise.sum()
                free = np.where((h_noise > 1e-12) & (h_noise < 1.0 - 1e-12))[0]
                if free.size > 0:
                    h_noise[free] += diff / free.size
                    h_noise = np.clip(h_noise, 0.0, 1.0)
                else:
                    h_noise[0] = min(1.0, max(0.0, h_noise[0] + diff))
                # Evaluate
                val_noise = _compute_c5(h_noise)
                if val_noise < cur_val:
                    h = h_noise
                    cur_val = val_noise
                    if val_noise < best_val:
                        best_val, best_h = val_noise, h_noise.copy()
        # Check overall time
        if time.time() - start > time_limit:
            break
    return best_h, best_val

import time, math, numpy as np

# --------------------------------------------------------------
#   Core utilities
# --------------------------------------------------------------
def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap value C5 for a given step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)


def _project_sum_and_bounds(x: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1] and then adjust linearly so that the sum equals target_sum."""
    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - x.sum()
    if abs(diff) < 1e-12:
        return x
    # free variables are those not on a bound
    free = np.where((x > 1e-12) & (x < 1.0 - 1e-12))[0]
    if free.size:
        x[free] += diff / free.size
        x = np.clip(x, 0.0, 1.0)
    else:
        # everything is at a bound – just move the first element (rare)
        x[0] = min(1.0, max(0.0, x[0] + diff))
        x = np.clip(x, 0.0, 1.0)
    return x


def _random_h(n: int, target_sum: float) -> np.ndarray:
    """Generate a random feasible h (values in [0,1], sum = target_sum)."""
    x = np.random.rand(n)
    x = _project_sum_and_bounds(x, target_sum)
    return x


def _thue_morse_h(n: int) -> np.ndarray:
    """Binary Thue‑Morse sequence of length n (first n bits)."""
    bits = np.fromfunction(lambda i: np.bitwise_xor.reduce(np.unpackbits(i.astype(np.uint64).view(np.uint8))), (n,), dtype=int)
    # pack bits into 0/1 using parity of popcount
    tm = np.unpackbits(np.arange(n, dtype=np.uint64).view(np.uint8), axis=1).sum(axis=1) % 2
    return tm.astype(float)

def _smooth_max_optimize(h0: np.ndarray, time_limit: float) -> (np.ndarray, float):
    """Projected gradient descent on the smooth max of the overlap."""
    n = h0.shape[0]
    pad = n - 1
    target_sum = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_val = _compute_c5(best_h)

    cur_val = best_val
    lr = 0.12                       # initial learning rate
    start = time.time()
    it = 0

    # pre‑allocate index array for fast shifted view
    i = np.arange(n)
    shifts = np.arange(-pad, pad + 1)          # length 2n‑1
    # This array will be used repeatedly; we only need its shape.
    while True:
        now = time.time()
        if now - start >= time_limit:
            break
        it += 1

        # ----- build shifted matrix H (2n‑1 × n) -----
        h_ext = np.concatenate([np.zeros(pad), h, np.zeros(pad)])
        idx = pad + i[None, :] + shifts[:, None]   # shape (2n‑1, n)
        H = h_ext[idx]                             # each row = h shifted by s

        # ----- correlation for every shift -----
        one_minus_H = 1.0 - H
        corr = np.dot(h, one_minus_H.T)           # length 2n‑1, raw (no scaling)

        # ----- soft‑max weights -----
        # α controls how close we are to the true max; 12 is a good compromise.
        alpha = 12.0
        maxc = corr.max()
        w = np.exp(alpha * (corr - maxc))
        w_sum = w.sum()
        p = w / w_sum                               # probabilities, sum to 1

        # ----- gradient of the smooth max -----
        # G_i = 1 - Σ_s p_s (h_{i+s}+h_{i-s})
        weighted = (p[:, None] * (H + H[::-1, :])).sum(axis=0)
        G = 1.0 - weighted

        # ----- propose a step -----

        new_h = _project_sum_and_bounds(new_h, target_sum)

def _slsqp_refine(h0: np.ndarray, time_limit: float) -> (np.ndarray, float):

    target_sum = n / 2.0

        return fun
