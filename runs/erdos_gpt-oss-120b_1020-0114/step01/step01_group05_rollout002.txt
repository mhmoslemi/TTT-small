def _grad_of_max_lag(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    idx = int(np.argmax(corr))
    k = idx - (n - 1)
    g = np.zeros_like(h)
    if k >= 0:
        # contributions for i from 0 to n-1-k
        g[:n - k] += (1 - h[k:]) * dx
        # contributions for i from k to n-1
        g[k:] -= h[:n - k] * dx
    else:
        kp = -k
        g[kp:] += (1 - h[:n - kp]) * dx
        g[:n - kp] -= h[kp:] * dx
    return g, corr.max()
```

import time
import numpy as np

def _project(v, target):
    """Project vector onto the box [0,1] and the affine hyperplane sum=v_target."""
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def _evaluate(h):
    """Return max_{k} ∫ h(x)(1‑h(x+k))dx for the discrete vector h."""
    n = h.size
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full') * dx)

def _grad_max(h):
    """Gradient of the lag attaining the current maximum overlap."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    idx = int(np.argmax(corr))
    k = idx - (n - 1)                     # integer shift
    g = np.zeros_like(h)

    if k >= 0:
        length = n - k
        # contribution from the “1‑h[i+k]” part (where i runs)
        g[:length] += (1 - h[k:]) * dx
        # contribution from the “‑h[i‑k]” part (where i runs)
        g[k:] -= h[:length] * dx
    else:
        kp = -k
        length = n - kp
        g[kp:] += (1 - h[:length]) * dx
        g[:length] -= h[kp:] * dx
    return g, corr.max()

def _random_feasible(n, target):
    v = np.random.rand(n)
    return _project(v, target)

def _binary_feasible(n, target):
    m = int(np.floor(target))
    frac = target - m
    h = np.zeros(n)
    if m > 0:
        idx = np.random.choice(n, m, replace=False)
        h[idx] = 1.0
    if frac > 1e-12:
        zeros = np.where(h == 0)[0]
        i = np.random.choice(zeros)
        h[i] = frac
    return h

def _sinusoid_feasible(n, target):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.5 * np.sin(np.pi * x / 2)   # values lie in [0,1]
    return _project(h, target)

def _interpolate_initial(old_h, n):
    n_old = old_h.size
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n, endpoint=False)
    return np.interp(x_new, x_old, old_h)

def _optimize(h0, target, deadline, max_iters=20000):
    """Projected momentum descent that reduces the current maximal overlap."""
    n = h0.size
    h = h0.copy()
    v = np.zeros_like(h)
    beta = 0.9
    lr = 0.2
    best_h = h.copy()
    best_c5 = _evaluate(h)
    no_improve = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break

        g, cur_c5 = _grad_max(h)

        v = beta * v - lr * g
        h_new = _project(h + v, target)
        new_c5 = _evaluate(h_new)

        if new_c5 <= cur_c5:
            h = h_new
            lr = min(lr * 1.02, 0.5)
            no_improve = 0
        else:
            lr = max(lr * 0.5, 1e-5)
            v = np.zeros_like(h)
            no_improve += 1

        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
            no_improve = 0

        if no_improve >= 2000:

            h = _project(h + jitter, target)
            v = np.zeros_like(h)
            no_improve = 0

        lr = max(lr * 0.9999, 1e-5)

    return best_h, best_c5
