def subgradient_optimize(start_h, max_iter, step0, rng):
    n = start_h.shape[0]
    dx = 2.0 / n
    h = start_h.copy()
    best_h = h.copy()
    best_val = max_overlap(h)
    for it in range(1, max_iter+1):
        # compute correlation
        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_corr = corr.max()
        # Determine indices achieving max
        idxs = np.where(np.abs(corr - max_corr) < 1e-12)[0]
        # translate indices to shifts
        shifts = idxs - (n - 1)
        # compute gradient
        g = np.zeros_like(h)
        for k in shifts:
            if k >= 0:
                # forward shift
                forward = np.concatenate([h[k:], np.zeros(k)])
                backward = np.concatenate([np.zeros(k), h[:n-k]])
            else:
                kk = -k
                forward = np.concatenate([np.zeros(kk), h[:n-kk]])
                backward = np.concatenate([h[kk:], np.zeros(kk)])
            g += dx * (1.0 - (forward + backward))
        g /= len(shifts)
        # step size
        step = step0 / np.sqrt(it)
        h_new = h - step * g
        # projection
        h_new = project_onto_capped_simplex(h_new, n/2)
        # evaluate
        val = max_overlap(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        h = h_new
    return best_h, best_val
```

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top level, no closures)
# ----------------------------------------------------------------------
def project_to_feasible(h, target, rng):
    """
    Project vector h onto the set {0 <= x_i <= 1, sum(x) = target}.
    Simple heuristic: clip, scale, then iterative correction.
    """
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == 0.0:
        # place unit mass on a random subset
        idx = rng.choice(len(h), int(round(target)), replace=False)
        h[idx] = 1.0
        return h
    scale = target / cur
    h = np.clip(h * scale, 0.0, 1.0)
    for _ in range(12):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            # no free variables – push a random index toward the needed direction
            i = rng.integers(len(h))
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
            continue
        delta = diff / free.size
        h[free] += delta
        np.clip(h, 0.0, 1.0, out=h)
    return h

def max_overlap(h):
    """Return the C5 value for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def thue_morse_seq(n):
    """Generate the first n values of the Thue–Morse sequence (0/1)."""
    # parity of popcount gives the sequence
    return np.fromiter((i.bit_count() % 2 for i in range(n)), dtype=float)

def subgradient_optimize(start_h, max_iter, step0, rng, time_limit):
    """
    Subgradient descent for the max‑overlap objective.
    Returns the best vector seen and its C5 value.
    """
    n = start_h.shape[0]
    dx = 2.0 / n
    target = n / 2.0
    h = project_to_feasible(start_h, target, rng)
    best_h = h.copy()
    best_val = max_overlap(h)

    for it in range(1, max_iter + 1):
        if time.time() > time_limit:
            break

        # correlation and maximal shift(s)
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        max_corr = corr.max()
        idxs = np.where(np.abs(corr - max_corr) < 1e-12)[0]
        shifts = idxs - (n - 1)          # convert to signed shift

        # subgradient (average over all maximal shifts)
        g = np.zeros_like(h)
        for k in shifts:
            if k >= 0:
                forward = np.concatenate([h[k:], np.zeros(k)])
                backward = np.concatenate([np.zeros(k), h[:n - k]])
            else:
                kk = -k
                forward = np.concatenate([np.zeros(kk), h[:n - kk]])
                backward = np.concatenate([h[kk:], np.zeros(kk)])
            g += dx * (1.0 - (forward + backward))
        g /= len(shifts)

        # diminishing step size
        step = step0 / np.sqrt(it)
        h_new = h - step * g
        h_new = project_to_feasible(h_new, target, rng)

        val = max_overlap(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        h = h_new

    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Utility to compute remaining time fraction
    # ------------------------------------------------------------------
    def time_left():
        return budget_s - (time.time() - start_time)

    # ------------------------------------------------------------------
    # Track the best solution seen
    # ------------------------------------------------------------------
    best_h = None
    best_val = float('inf')
    best_n = None

    # ------------------------------------------------------------------
    # Helper to update the global best
    # ------------------------------------------------------------------
    def consider(h, val, n):
        nonlocal best_h, best_val, best_n
        if val < best_val - 1e-9:
            best_h = h.copy()
            best_val = float(val)
            best_n = n

    # ------------------------------------------------------------------
    # Include the provided initial vector if it exists
    # ------------------------------------------------------------------
    try:
        init_vec = np.array(initial_h_values, dtype=float)  # supplied by harness
        n0 = init_vec.shape[0]
        init_vec = project_to_feasible(init_vec, n0 / 2.0, rng)
        consider(init_vec, max_overlap(init_vec), n0)
    except Exception:
        # no initial vector supplied
        pass

    # ------------------------------------------------------------------
    # Candidate grid sizes (all even for exact half‑ones in binary patterns)
    # ------------------------------------------------------------------
    candidate_ns = [80, 96, 100, 112, 128]

    # ------------------------------------------------------------------
    # Main optimisation loop over grid sizes
    # ------------------------------------------------------------------
    for n in candidate_ns:
        if time_left() < 5.0:
            break

        target_sum = n / 2.0
        dx = 2.0 / n

        # number of independent attempts per grid size
        attempts = 6
        for attempt in range(attempts):
            if time_left() < 5.0:
                break

            # --------------------------------------------------------------
            # Build a starting vector
            # --------------------------------------------------------------
            if attempt == 0 and (n & (n - 1)) == 0:        # power of two -> Thue–Morse
                h0 = thue_morse_seq(n)
            else:
                if rng.random() < 0.55:   # binary start (exact half ones)
                    h0 = np.zeros(n)
                    ones = rng.choice(n, n // 2, replace=False)
                    h0[ones] = 1.0
                else:                     # random continuous start
                    h0 = rng.random(n)
                    h0 = project_to_feasible(h0, target_sum, rng)

            # --------------------------------------------------------------
            # Run subgradient optimisation
            # --------------------------------------------------------------
            # Allow a generous number of iterations; the routine will stop early if out of time.
            max_iters = 5000 if n >= 112 else 4000
            step0 = 5.0                     # initial step magnitude (multiplied by subgradient ~dx)
            time_limit = start_time + budget_s * 0.93
            h_opt, val_opt = subgradient_optimize(h0, max_iters, step0, rng, time_limit)

            consider(h_opt, val_opt, n)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if best_h is None:

    best_val = max_overlap(best_h)

    return best_h, best_val, best_n
