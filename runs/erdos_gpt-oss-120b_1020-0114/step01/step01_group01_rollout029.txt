def project_to_feasible(h, target):
    # clip to [0,1]
    h = np.clip(h,0.0,1.0)
    current = h.sum()
    diff = target - current
    # iterative adjustment
    if abs(diff) < 1e-12:
        return h
    # adjust using free variables
    max_iter = 100
    for _ in range(max_iter):
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            # assign mass to random free variable
            i = np.random.randint(len(h))
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
            current = h.sum()
            diff = target - current
            if abs(diff) < 1e-12:
                break
            continue
        delta = diff / free.size
        h[free] += delta
        # clip after adjustment
        np.clip(h, 0.0, 1.0, out=h)
        current = h.sum()
        diff = target - current
        if abs(diff) < 1e-12:
            break
    # final scaling fix: if still not close, linearly scale non-boundary values
    if abs(h.sum() - target) > 1e-8:
        # simple scaling of all values within bounds ignoring extremes
        scale = target / h.sum()
        h = np.clip(h * scale, 0.0, 1.0)
    # ensure sum exactly target via minimal adjustment
    correction = target - h.sum()
    if abs(correction) > 1e-12:
        # add correction to first free variable
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size > 0:
            h[free[0]] += correction
            h[free[0]] = np.clip(h[free[0]],0.0,1.0)
    return h

def optimise_lbfgs(start_h, n, dx, target, rng):
    # parameters:
    penalty_coeff = 10.0  # weight for sum constraint
    # Phase schedule
    alpha_vals = [10.0, 40.0, 80.0]
    maxiters = [200, 300, 300]
    h = start_h.copy()
    for alpha, maxit in zip(alpha_vals, maxiters):
        # define objective closure that captures alpha, penalty, etc.
        def fun(x):
            # compute corr
            corr = dx * np.correlate(x, 1 - x, mode='full')
            max_corr = np.max(corr)
            exp_term = np.exp(alpha * (corr - max_corr))
            sum_exp = np.sum(exp_term)
            f_smooth = max_corr + np.log(sum_exp) / alpha
            penalty = penalty_coeff * (np.sum(x) - target) ** 2
            return f_smooth + penalty

        def grad(x):
            corr = dx * np.correlate(x, 1 - x, mode='full')
            max_corr = np.max(corr)
            exp_term = np.exp(alpha * (corr - max_corr))
            sum_exp = np.sum(exp_term)
            w = exp_term / sum_exp  # shape 2n-1
            # compute gradient of weighted sum corr
            # Convolution terms
            # because np.convolve expects 1D arrays.
            term1 = np.convolve(w[::-1], 1 - x, mode='full')
            term2 = np.convolve(w, x, mode='full')
            start = n - 1
            end = start + n
            grad_corr = dx * (term1[start:end] - term2[start:end])
            grad_pen = 2.0 * penalty_coeff * (np.sum(x) - target) * np.ones_like(x)
            return grad_corr + grad_pen

        bounds = [(0.0, 1.0)] * n
        # call minimize
        res = minimize(fun, h, method='L-BFGS-B', jac=grad, bounds=bounds,
                       options={'maxiter': maxit, 'ftol': 1e-9, 'disp': False})
        h = np.clip(res.x, 0.0, 1.0)  # ensure bound
        # optional: after each phase, re-project to maintain sum constraint
        h = project_to_feasible(h, target)
    return h

    max_corr = np.max(corr)

def grad_func(x, dx, target, alpha):
    corr = dx * np.correlate(x, 1 - x, mode='full')
    max_corr = np.max(corr)
    exp_term = np.exp(alpha * (corr - max_corr))

import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Helper utilities (top‑level as required)
# --------------------------------------------------------------

def project_to_feasible(h, target):
    """Clip to [0,1] and adjust linearly so that sum(h) = target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # iterative redistribution among non‑boundary entries
    for _ in range(10):
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            i = np.random.randint(len(h))
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
        else:
            delta = diff / free.size
            h[free] += delta
            np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
    # final tiny correction
    if abs(h.sum() - target) > 1e-9:
        scale = target / h.sum()
        h = np.clip(h * scale, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            i = np.argmax(h)
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def compute_corr(h, dx):
    """Full correlation h * (1‑h) times dx."""
    return np.correlate(h, 1 - h, mode='full') * dx

def obj_with_args(x, dx, target, alpha, penalty):
    """Smoothed max objective + quadratic penalty on the integral."""
    corr = compute_corr(x, dx)
    m = np.max(corr)
    exp_term = np.exp(alpha * (corr - m))
    sum_exp = np.sum(exp_term)
    f = m + np.log(sum_exp) / alpha
    f += penalty * (np.sum(x) - target) ** 2
    return f

def grad_with_args(x, dx, target, alpha, penalty):
    """Gradient of the smoothed max objective + penalty."""
    corr = compute_corr(x, dx)
    m = np.max(corr)
    exp_term = np.exp(alpha * (corr - m))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                     # soft‑max weights, length 2n‑1
    # convolution terms giving Σ w_s (1‑h_{i+s}) and Σ w_s h_{i‑s}
    term1 = np.convolve(w[::-1], 1 - x, mode='full')
    term2 = np.convolve(w, x, mode='full')
    start = len(x) - 1
    end = start + len(x)
    grad_corr = dx * (term1[start:end] - term2[start:end])
    grad = grad_corr + 2.0 * penalty * (np.sum(x) - target) * np.ones_like(x)
    return grad

def optimise_lbfgs(start_h, dx, target, penalty_coeff=15.0):
    """Two‑stage L‑BFGS‑B optimisation with increasing smoothness."""
    h = start_h.copy()
    alphas = [5.0, 20.0, 80.0]               # smoothing parameters
    maxiters = [180, 300, 300]               # iteration budgets per stage
    for alpha, it in zip(alphas, maxiters):
        res = minimize(
            obj_with_args,
            h,
            args=(dx, target, alpha, penalty_coeff),
            method='L-BFGS-B',
            jac=grad_with_args,
            bounds=[(0.0, 1.0)] * len(h),
            options={'maxiter': it, 'ftol': 1e-9, 'disp': False},
        )
        h = np.clip(res.x, 0.0, 1.0)
        h = project_to_feasible(h, target)  # keep the sum constraint exact
    return h
