def project_bounds_sum(h, target):

def project_onto_simplex_box(y, target):
    # Return projection onto {x: sum x = target, 0 <= x <= 1}
    lo = -np.max(y) - 1  # maybe small
    hi = 1 - np.min(y) + 1
    for _ in range(40):
        lam = (lo + hi) / 2.0
        x = np.clip(y + lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    x = np.clip(y + (lo+hi)/2.0, 0.0, 1.0)
    # Ensure sum exact by adjusting interior values slightly if needed

    return x

def gradient_for_shift(h, s):
    n = len(h)
    L = 0 if s >= 0 else -s
    R = n - s - 1 if s >=0 else n - 1
    g = np.zeros_like(h)
    idx1 = np.arange(L, R+1, dtype=int)

    idx_i_plus_s = idx1 + s

    g[idx1] += 1.0 - h[idx_i_plus_s]
    # term2: subtract h[i] from g[i+s]
    g[idx_i_plus_s] -= h[idx1]
    return g

        if s > target:

def hill_climb(h, budget, start_time):
    n = h.shape[0]
    best_h = h.copy()
    best_val = compute_c5(best_h)
    rng = np.random.default_rng()
    while time.time() - start_time < budget:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # direction mass from i to j
        # Determine maximum delta possible while staying within [0,1]
        hi, hj = best_h[i], best_h[j]
        if hi > 0.0 and hj < 1.0:
            max_delta = min(hi, 1.0 - hj)

import time
import numpy as np

# ---------- utility functions ----------
def compute_c5(h):
    """Maximum overlap integral for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode="full").max() * dx)


def gradient_for_shift(h, s):
    """Gradient of the correlation at shift s (integer) w.r.t. h."""
    n = h.shape[0]
    # valid i range where both i and i+s lie in [0,n)
    if s >= 0:
        L, R = 0, n - s - 1
    else:
        L, R = -s, n - 1
    if L > R:  # empty range
        return np.zeros_like(h)

    idx = np.arange(L, R + 1, dtype=int)
    idx_shift = idx + s

    g = np.zeros_like(h)
    # term from derivative of h[i] (first factor)
    g[idx] += 1.0 - h[idx_shift]
    # term from derivative of (1 - h[i+s]) (second factor)
    g[idx_shift] -= h[idx]
    return g


def proj_feasible(y, target):
    """
    Euclidean projection of y onto {x : sum(x)=target, 0<=x_i<=1}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = -np.max(y) - 2.0
    hi = 1.0 - np.min(y) + 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)

    # tiny residual correction
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        interior = (x > 0.0) & (x < 1.0)
        if interior.any():
            x[interior] += diff / interior.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            # fallback scaling (should be rare)
            x *= target / x.sum()
    return x


def random_feasible_vector(n, rng):
    """Random vector in [0,1]^n with exact sum = n/2."""
    target = n / 2.0
    v = rng.random(n)
    v = v / v.sum() * target
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target) > 1e-12:
        diff = target - cur
        mask = (v > 0.0) & (v < 1.0)
        if mask.any():
            v[mask] += diff / mask.sum()
            v = np.clip(v, 0.0, 1.0)
        else:
            v *= target / cur
    return v


def binary_start_vector(n, rng):
    """Binary {0,1} vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


# ---------- optimisation primitives ----------
def gradient_descent(h0, max_iter, time_limit, start_time):
    """Projected gradient descent on the max‑overlap objective."""
    n = h0.shape[0]
    target = n / 2.0
    h = proj_feasible(h0, target)
    best_h = h.copy()
    best_val = compute_c5(best_h)
    alpha = 0.5
    dx = 2.0 / n

    for _ in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        # current correlation and worst shift
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        s = np.argmax(corr) - (n - 1)

        # gradient of that worst shift
        g = gradient_for_shift(h, s)

        # step + projection
        h = proj_feasible(h - alpha * g, target)

        # evaluate
        cur = compute_c5(h)
        if cur < best_val - 1e-12:
            best_val = cur
            best_h = h.copy()

        # slowly decay step size
        alpha *= 0.995

    return best_h, best_val


def hill_climb(h, budget, start_time):
    """Simple stochastic improvement by moving mass between two points."""
    n = h.shape[0]
    best_h = h.copy()
    best_val = compute_c5(best_h)
    rng = np.random.default_rng()
    while time.time() - start_time < budget:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        hi, hj = best_h[i], best_h[j]
        # determine a feasible direction of transfer
        if hi > 0.0 and hj < 1.0:
            max_delta = min(hi, 1.0 - hj)
        elif hj > 0.0 and hi < 1.0:
            max_delta = min(hj, 1.0 - hi)
            i, j = j, i  # move from i to j after swap
        else:
            continue
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random()
        cand = best_h.copy()
        cand[i] -= delta
        cand[j] += delta
        val = compute_c5(cand)
        if val < best_val - 1e-12:
            best_h = cand
            best_val = val
    return best_h, best_val


# ---------- main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # initialise candidate constructions
    candidates = []
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n0 = h0.size
        candidates.append((n0, h0.copy()))
    else:
        n0 = 180
        candidates.append((n0, np.full(n0, 0.5)))

    # a binary start
    candidates.append((n0, binary_start_vector(n0, rng)))
    # a random continuous start
    candidates.append((n0, random_feasible_vector(n0, rng)))

    # up‑sampled versions of the first candidate (if not too large)
    if n0 <= 200:
        for factor in (2, 4):
            n_up = n0 * factor
            h_up = np.repeat(candidates[0][1], factor)
            candidates.append((n_up, h_up.copy()))

    best_h = None
    best_c5 = np.inf
    best_n = None

    # run optimisation on each candidate, respecting the global time budget
    for n, h_start in candidates:
        elapsed = time.time() - start
        if elapsed > budget_s * 0.93:
            break
        target = n / 2.0
        # ensure exact sum constraint before optimisation
        cur_sum = h_start.sum()
        if abs(cur_sum - target) > 1e-12:
            h_start = h_start * (target / cur_sum)
            h_start = np.clip(h_start, 0.0, 1.0)

        # allocate a time slice for this candidate
        remaining = budget_s - (time.time() - start)
        # decide iteration budget based on problem size
        max_iter = 5000 if n <= 180 else 3000 if n <= 360 else 1500

        # gradient descent
        gd_time = remaining * 0.45
        h_opt, c5_val = gradient_descent(
            h_start, max_iter, gd_time, start
        )
        if c5_val < best_c5:
            best_h, best_c5, best_n = h_opt, c5_val, n

        # short hill‑climb refinement
        remaining = budget_s - (time.time() - start)
        if remaining > 1.0:
            h_ref, c5_ref = hill_climb(best_h, remaining * 0.5, start)
            if c5_ref < best_c5:
                best_h, best_c5 = h_ref, c5_ref
                best_n = best_h.size

    # final global hill‑climb if time permits
    if best_h is not None and (time.time() - start) < budget_s * 0.97:
        final_time = budget_s - (time.time() - start)
        h_final, c5_final = hill_climb(best_h, final_time, start)
        if c5_final < best_c5:
            best_h, best_c5 = h_final, c5_final
            best_n = best_h.size

    # safety fallback
    if best_h is None:
        best_n = n0
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    # final projection to respect constraints exactly
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * ((best_n / 2.0) / best_h.sum())
    best_c5 = compute_c5(best_h)

    return (best_h.tolist(), float(best_c5), int(best_n))
