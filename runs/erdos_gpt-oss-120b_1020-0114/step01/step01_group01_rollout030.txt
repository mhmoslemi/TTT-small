def cap_simplex_projection(v, s):
    n = len(v)
    # Clip to [0,1]
    v_clipped = np.clip(v, 0, 1)
    # If sum already s, return
    current = v_clipped.sum()
    if np.isclose(current, s):
        return v_clipped
    # Use algorithm: sort v_clipped
    u = np.sort(v_clipped)[::-1]  # descending
    # Compute cumulative sums
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n+1)) > 0)[0]
    if len(rho) == 0:
        # all need to be set to lower bound? but s <= n? here s=n/2, so should have some.
        lambda_val = 0
    else:
        rho = rho[-1]
        lambda_val = cssv[rho] / (rho + 1)
    # Now projection x = np.clip(v_clipped - lambda_val, 0, 1)
    x = np.clip(v_clipped - lambda_val, 0, 1)
    # Might not reach exact sum due to numeric; adjust small diff
    diff = s - x.sum()
    if abs(diff) > 1e-10:
        # distribute diff among those not at bounds
        free = np.where((x > 0) & (x < 1))[0]
        if free.size > 0:
            x[free] += diff / free.size
            x = np.clip(x, 0, 1)
        else:
            # fallback simple scaling
            x = np.clip(v_clipped * (s / v_clipped.sum()), 0, 1)
    return x

def project_to_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == 0:
        # assign ones randomly to meet target
        idx = np.random.choice(len(h), int(round(target_sum)), replace=False)
        h[idx] = 1.0
        return h
    # If cur not equal target, adjust linearly:
    diff = target_sum - cur
    # distribute diff equally among free variables (not at bounds)
    # We'll iterate limited.
    for _ in range(10):
        free = np.where((h > 0) & (h < 1))[0]
        if free.size == 0:
            # No free variables, adjust random index:
            i = np.random.randint(len(h))
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
            break
        delta = diff / free.size
        h[free] += delta
        # clip again and recompute diff
        np.clip(h, 0.0, 1.0, out=h)
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
    # final small correction if any leftover (due to rounding) distribute to any variable not at bounds:
    if abs(diff) > 1e-12:
        # adjust random free variable(s) if any
        i = np.random.randint(len(h))
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def shift_array(arr, shift):
    n = len(arr)
    if shift > 0:
        out = np.empty(n)
        out[:n-shift] = arr[shift:]
        out[n-shift:] = 0.0
    elif shift < 0:
        s = -shift
        out = np.empty(n)
        out[s:] = arr[:n-s]
        out[:s] = 0.0
    else:
        out = arr.copy()
    return out

def worst_lag_and_gradient(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    # find lag index of max
    idx_max = np.argmax(corr)
    k = idx_max - (n - 1)   # correlation length 2n-1, zero lag at index n-1
    # compute gradient
    term1 = 1 - shift_array(h, k)   # (1 - h_{i+k}) with zero padding
    term2 = shift_array(h, -k)      # h_{i-k}
    grad = term1 - term2
    return grad, k, corr[idx_max] * (2.0 / n)

def project_to_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if abs(cur - target_sum) < 1e-12:
        return h
    # scale proportionally then clip
    if cur > 0:
        h = np.clip(h * (target_sum / cur), 0.0, 1.0)
    # iterative correction
    for _ in range(10):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 0) & (h < 1))[0]
        if free.size == 0:
            # adjust a random entry
            i = np.random.randint(len(h))
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
            break
        delta = diff / free.size
        h[free] += delta
        np.clip(h, 0.0, 1.0, out=h)
    return h

def slsqp_refine(h_initial, maxiter=300):
    global _opt_n, _opt_dx, _opt_target_sum
    n = len(h_initial)
    _opt_n = n
    _opt_dx = 2.0 / n
    _opt_target_sum = n / 2.0
    h0 = project_to_feasible(h_initial, _opt_target_sum)
    init_corr = np.correlate(h0, 1.0 - h0, mode='full') * _opt_dx
    t0 = float(np.max(init_corr) + 1e-8)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    eq = {'type': 'eq', 'fun': _eq_cons}
    ineq = NonlinearConstraint(_ineq_cons, lb=np.zeros(2 * n - 1), ub=np.full(2 * n - 1, np.inf))
    res = minimize(lambda x: x[-1], x0,
                   method='SLSQP',
                   bounds=bounds,
                   constraints=[eq, ineq],
                   options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    h_opt = project_to_feasible(h_opt, _opt_target_sum)
    return h_opt

    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    eq = {'type': 'eq', 'fun': _eq_cons}
    ineq = NonlinearConstraint(_ineq_cons,
                               lb=np.zeros(2 * n - 1),
                               ub=np.full(2 * n - 1, np.inf))

    res = minimize(lambda x: x[-1],
                   x0,
                   method='SLSQP',
                   bounds=bounds,
                   constraints=[eq, ineq],
                   options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})

    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    h_opt = project_to_feasible(h_opt, _opt_target_sum)
    return h_opt

# --------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Load the supplied initial vector if present
    # ------------------------------------------------------------------
    try:
        init_vec = np.array(initial_h_values, dtype=float)  # provided by harness
        n0 = init_vec.shape[0]
    except Exception:
        n0 = 100
        init_vec = np.full(n0, 0.5)

    init_vec = project_to_feasible(init_vec, n0 / 2.0)

    # ------------------------------------------------------------------
    # 2. Initialise the incumbent solution
    # ------------------------------------------------------------------
    try:
        best_c5 = float(evaluate_erdos_solution(init_vec))
    except Exception:
        best_c5 = fast_c5(init_vec)
    best_h = init_vec.copy()
    best_n = n0

    # ------------------------------------------------------------------
    # 3. Grid sizes to explore (always keep the original size)
    # ------------------------------------------------------------------
    grid_sizes = [n0, 120, 150, 180, 200]
    grid_sizes = [g for g in grid_sizes if g % 2 == 0]  # ensure even sizes

    # ------------------------------------------------------------------
    # 4. Main search loop over grid sizes
    # ------------------------------------------------------------------
    for n in grid_sizes:
        if time.time() - start_time > budget_s * 0.95:
            break
        target_sum = n / 2.0

        # ----- start from the best known solution, interpolated -----
        h_start = resize_h(best_h, n)
        h_start = project_to_feasible(h_start, target_sum)
        val_start = fast_c5(h_start)
        if val_start < best_c5:
            best_h, best_c5, best_n = h_start.copy(), val_start, n

        # ----- random binary restarts (quick filter) -----
        n_random = 250  # number of random candidates per grid size
        for _ in range(n_random):
            if time.time() - start_time > budget_s * 0.90:
                break
            h_rand = random_binary(n, rng)
            val_rand = fast_c5(h_rand)
            if val_rand < best_c5:
                best_h, best_c5, best_n = h_rand.copy(), val_rand, n

        # ----- refine the current best vector on this grid with SLSQP -----
        if time.time() - start_time < budget_s * 0.80:
            try:
                refined = slsqp_optimize(best_h, maxiter=600)
                refined_val = float(evaluate_erdos_solution(refined))
                if refined_val < best_c5 - 1e-12:
                    best_h, best_c5, best_n = refined.copy(), refined_val, n
            except Exception:
                pass

        # ----- extra short SLSQP runs from fresh random starts -----
        extra_trials = 3
        for _ in range(extra_trials):
            if time.time() - start_time > budget_s * 0.85:
                break
            h_rand = random_binary(n, rng)
            try:
                refined = slsqp_optimize(h_rand, maxiter=300)
                refined_val = float(evaluate_erdos_solution(refined))
                if refined_val < best_c5 - 1e-12:
                    best_h, best_c5, best_n = refined.copy(), refined_val, n
            except Exception:
                continue

    # ------------------------------------------------------------------
    # 5. Final sanity check and return
    # ------------------------------------------------------------------
    best_h = project_to_feasible(best_h, len(best_h) / 2.0)
    try:
        best_c5 = float(evaluate_erdos_solution(best_h))
    except Exception:
        best_c5 = fast_c5(best_h)

    return best_h, best_c5, len(best_h)
