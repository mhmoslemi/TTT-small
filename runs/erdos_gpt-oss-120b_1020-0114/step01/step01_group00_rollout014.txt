```
    n=len(h)
    dx = 2.0/n
    max_val = -inf
    max_ks = []
    for k in range(-(n-1), n):
        # compute f_k
        if k>=0:
            overlap = n - k
            idx_i = np.arange(overlap)
            idx_j = idx_i + k
            f = (h[idx_i] * (1 - h[idx_j])).sum()
        else:
            kk = -k
            overlap = n - kk
            idx_i = np.arange(overlap)
            idx_j = idx_i + kk
            # This corresponds to shift -k (i.e., on h(x+k) where k negative)
            # Actually f_{-kk} = sum_i h_i * (1 - h_{i-kk}) but index i-kk must be >=0.
            # So same as earlier with swapped.
            f = (h[idx_i+kk] * (1 - h[idx_i])).sum()
        if f > max_val + 1e-12:
            max_val = f
            max_ks = [k]
        elif abs(f - max_val) < 1e-12:
            max_ks.append(k)
    # compute gradient as sum of gradients for each max_k (average)
    grad = np.zeros_like(h)
    for k in max_ks:
        # compute gradient for this k
        if k >= 0:
            overlap = n - k
            # contributions:
            # For i=0..overlap-1:
            # d/dh_i: 1 - h_{i+k}
            # d/dh_{i+k}: -h_i

            grad[:overlap] += 1 - h[k:k+overlap]
```

```
def project_onto_bounds_sum(v, target, tol=1e-8, max_iter=100):
    # find lambda such that sum clip(v_i - lambda, 0,1) = target
    lo = -1.0  # if lambda too low, v_i - lambda large -> clip to 1 gets sum >= target?
    hi = 2.0   # if lambda too high, v_i - lambda small -> clip to 0 => sum <= target
    # Actually we need bounds for lambda: when lambda = -inf, x_i = 1 (since v_i - lambda -> +inf, clipped to 1), sum=n. When lambda=+inf, x_i=0, sum=0.
    # So we can start lo=-max(v)-1, hi=max(v)+1.
    lo = - (np.max(v) + 2)
    hi = np.max(v) + 2
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            # need to increase lambda to reduce sum
            lo = mid
        else:
            hi = mid
    # return approximate
    return np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
```

Check monotonic: As lambda increases, v_i - lambda decreases, clipping tends to lower, so sum decreases. So if sum > target, increase lambda (lo = mid). Good.

```
def compute_grad_max(h):
    n = h.shape[0]
    # compute correlation raw (no dx)
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = corr.max()
    # get indices with max (allow tolerance)
    tol = 1e-12
    max_idxs = np.where(np.abs(corr - max_corr) <= tol)[0]
    grad = np.zeros_like(h)
    for idx in max_idxs:
        k = idx - (n - 1)  # shift
        if k >= 0:
            overlap = n - k
            # grad contributions
            grad[:overlap] += 1 - h[k:k+overlap]
            grad[k:k+overlap] += -h[:overlap]
        else:
            kk = -k
            overlap = n - kk
            grad[kk:kk+overlap] += 1 - h[:overlap]
            grad[:overlap] += -h[kk:kk+overlap]
    grad = grad / len(max_idxs)  # average
    return max_corr, grad
```

def hill_climb(h, budget, start_time):
    n = h.shape[0]
    best = h.copy()
    best_val = compute_c5(best)
    rng = np.random.default_rng()
    while time.time() - start_time < budget:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # pick direction ensuring bounds
        if best[i] > 0.0 and best[j] < 1.0:
            max_delta = min(best[i], 1.0 - best[j])
        elif best[j] > 0.0 and best[i] < 1.0:
            max_delta = min(best[j], 1.0 - best[i])
            i, j = j, i
        else:
            continue
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random()
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        val = compute_c5(cand)
        if val < best_val - 1e-12:
            best = cand
            best_val = val
    return best, best_val

def compute_grad_max(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = corr.max()
    # tolerance for numeric equality
    eps = 1e-12
    max_idx = np.where(np.abs(corr - max_corr) <= eps)[0]
    grad = np.zeros_like(h)
    for idx in max_idx:
        k = idx - (n - 1)
        if k >= 0:
            overlap = n - k
            grad[:overlap] += 1.0 - h[k:k+overlap]
            grad[k:k+overlap] += -h[:overlap]
        else:
            kk = -k
            overlap = n - kk
            grad[kk:kk+overlap] += 1.0 - h[:overlap]
            grad[:overlap] += -h[kk:kk+overlap]
    # average over multiple max shifts
    grad /= max_idx.shape[0]
    return max_corr, grad

def subgradient_descent(h0, time_limit, start_time):
    n = h0.shape[0]
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)

    # initial step size
    step0 = 0.5
    while time.time() - start_time < time_limit:
        iteration += 1
        max_corr, grad = compute_grad_max(h)
        # convert to C5
        cur_val = max_corr * (2.0 / n)
        # normalize grad direction
        norm = np.linalg.norm(grad)
        if norm < 1e-12:
            # gradient zero, break
            break
        direction = grad / norm
        # step size decreasing
        step = step0 / np.sqrt(iteration)
        # try a few step scales
        improved = False
        for scale in (1.0, 0.5, 0.25):
            h_new = project_onto_bounds_sum(h - step * scale * direction, target)
            new_val = compute_c5(h_new)
            if new_val < cur_val - 1e-12:
                h = h_new
                cur_val = new_val
                improved = True
                if new_val < best_val - 1e-12:
                    best_h = h_new.copy()
                    best_val = new_val
                break
        if not improved and iteration % 100 == 0:
            # random perturbation
            noise = np.random.default_rng().normal(scale=0.01, size=n)
            h_pert = project_onto_bounds_sum(h + noise, target)
            pert_val = compute_c5(h_pert)
            if pert_val < best_val:
                h = h_pert
                best_h = h_pert.copy()
                best_val = pert_val
    return best_h, best_val

import time
import numpy as np


def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a step function h (discrete version)."""
    n = h.shape[0]
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode="full").max() * dx)


def project_onto_bounds_sum(v: np.ndarray, target: float, tol: float = 1e-9, max_iter: int = 80) -> np.ndarray:
    """Projection of v onto the set {x in [0,1]^n, sum x = target}."""
    lo, hi = -1e6, 1e6
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def compute_grad_max(h: np.ndarray):
    """Return raw maximal correlation value and a sub‑gradient of that max."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = corr.max()
    eps = 1e-10
    idxs = np.where(np.abs(corr - max_corr) <= eps)[0]
    grad = np.zeros_like(h)
    for idx in idxs:
        k = idx - (n - 1)          # shift
        if k >= 0:
            ov = n - k
            grad[:ov] += 1.0 - h[k:k + ov]
            grad[k:k + ov] += -h[:ov]
        else:
            kk = -k
            ov = n - kk
            grad[kk:kk + ov] += 1.0 - h[:ov]
            grad[:ov] += -h[kk:kk + ov]
    grad /= max(idxs.shape[0], 1)
    return max_corr, grad


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1]^n with exact sum n/2."""
    v = rng.random(n)
    return project_onto_bounds_sum(v, n / 2.0)


def hill_climb(h: np.ndarray, budget: float, start_time: float):
    """Greedy pairwise mass transfer."""
    n = h.shape[0]
    best = h.copy()
    best_val = compute_c5(best)
    rng = np.random.default_rng()
    while time.time() - start_time < budget:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if best[i] > 0.0 and best[j] < 1.0:
            max_delta = min(best[i], 1.0 - best[j])
        elif best[j] > 0.0 and best[i] < 1.0:
            max_delta = min(best[j], 1.0 - best[i])
            i, j = j, i
        else:
            continue
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random()
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        val = compute_c5(cand)
        if val < best_val - 1e-12:
            best, best_val = cand, val
    return best, best_val


def subgradient_descent(h0: np.ndarray, time_limit: float, start_time: float):
    """Projected sub‑gradient descent on the max‑overlap objective."""
    n = h0.shape[0]
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    rng = np.random.default_rng()
    it = 0
    step0 = 0.5
    while time.time() - start_time < time_limit:
        it += 1
        max_corr, grad = compute_grad_max(h)
        cur_val = max_corr * (2.0 / n)
        norm = np.linalg.norm(grad)
        if norm < 1e-12:
            break
        direction = grad / norm
        step = step0 / np.sqrt(it)
        improved = False
        for scale in (1.0, 0.5, 0.25):
            h_new = project_onto_bounds_sum(h - step * scale * direction, target)
            new_val = compute_c5(h_new)
            if new_val < cur_val - 1e-12:
                h, cur_val = h_new, new_val
                improved = True
                if new_val < best_val - 1e-12:
                    best_h, best_val = h_new.copy(), new_val
                break
        if not improved and it % 100 == 0:
            # occasional random jitter
            noise = rng.normal(scale=0.01, size=n)
            h_pert = project_onto_bounds_sum(h + noise, target)
            pert_val = compute_c5(h_pert)
            if pert_val < best_val:
                h, best_h, best_val = h_pert, h_pert.copy(), pert_val
    return best_h, best_val


def resample_h(h: np.ndarray, n_target: int) -> np.ndarray:
    """Linear interpolation of h onto a new grid of length n_target."""
    n0 = h.shape[0]
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n_target) + 0.5) * (2.0 / n_target)
    h1 = np.interp(x1, x0, h, left=0.0, right=0.0)
    h1 = np.clip(h1, 0.0, 1.0)
    target = n_target / 2.0
    if abs(h1.sum() - target) > 1e-8:
        h1 = project_onto_bounds_sum(h1, target)
    return h1
