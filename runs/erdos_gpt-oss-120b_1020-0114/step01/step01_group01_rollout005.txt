def propose_move(h, step):
    n = len(h)
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return None
    # Determine max allowable delta that keep within bounds
    # delta <= min(step, 1 - h[i])
    # delta >= -min(step, h[i]) similarly for j
    delta_max = min(step, 1.0 - h[i])
    delta_min = max(-step, -h[i])   # negative allowed
    # For j, delta should be opposite sign:
    # We'll adjust h[i] by delta, h[j] by -delta, so need -delta within bounds for h[j].
    # So delta must satisfy:
    # -delta <= 1 - h[j]   => delta >= -(1 - h[j])
    # -delta >= -h[j]      => delta <= h[j]
    # So combined constraints:
    lo = max(delta_min, -(1.0 - h[j]))
    hi = min(delta_max, h[j])
    if lo >= hi:
        return None
    delta = rng.uniform(lo, hi)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

def optimise_slsqp(start_h):
    n = len(start_h)
    dx = 2.0 / n
    target = n/2.0

    # Ensure feasibility (project)
    h0 = project_to_feasible(start_h, target)

    # Compute initial t
    corr0 = np.correlate(h0, 1 - h0, mode='full') * dx
    t0 = float(np.max(corr0) + 1e-8)

    # variables: [h0... h_{n-1}, t]
    x0 = np.concatenate([h0, [t0]])

    # bounds
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    # constraints
    def eq_cons(x):
        return np.sum(x[:n]) - target

    def ineq_cons(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1 - h, mode='full') * dx
        # need t >= corr for each lag
        return t - corr

    eq = {'type': 'eq', 'fun': eq_cons}
    # SLSQP can accept 'ineq' but we need vector constraints; use NonlinearConstraint
    from scipy.optimize import NonlinearConstraint
    ineq = NonlinearConstraint(ineq_cons, lb=np.zeros(2*n-1), ub=np.full(2*n-1, np.inf))

    res = minimize(
        lambda x: x[-1],
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=[eq, ineq],
        options={'maxiter': 800, 'ftol': 1e-10, 'disp': False}
    )
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    h_opt = project_to_feasible(h_opt, target)
    val = compute_c5(h_opt)
    return h_opt, val

import time, numpy as np
from scipy.optimize import minimize, NonlinearConstraint

def project_to_feasible(h, target):
    # Clip and adjust sum iteratively
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == target:
        return h
    # Determine if need to increase or decrease sum
    # Simple scaling may exceed bounds, so iterative
    # We'll use algorithm: if cur < target, add mass to free variables equally; else subtract.
    # Use up to 100 iterations
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        # find indices not at bounds (0< h <1)
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            # adjust a random index within bounds
            i = np.random.randint(len(h))
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
            continue
        delta = diff / free.size
        # add delta to free entries, clip after
        h[free] += delta
        np.clip(h, 0.0, 1.0, out=h)
    # Ensure sum close
    # Final adjustment: compute error, distribute among any positive entries not at 1
    err = target - h.sum()
    if abs(err) > 1e-9:
        # adjust using small random variation within bounds
        idx = np.arange(len(h))
        rng = np.random.default_rng()
        for _ in range(10):
            i = rng.choice(idx)
            add = np.clip(err, -h[i], 1.0 - h[i])
            h[i] += add
            err = target - h.sum()
            if abs(err) < 1e-12:
                break
    return h

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def project_to_feasible(h, target):
    """Clip to [0,1] and adjust linearly so that sum(h) == target (approximately)."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:                     # all variables at a bound
            i = np.random.randint(len(h))
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
            continue
        delta = diff / free.size
        h[free] += delta
        np.clip(h, 0.0, 1.0, out=h)
    # tiny correction if needed
    err = target - h.sum()
    if abs(err) > 1e-10:
        idx = np.arange(len(h))
        rng = np.random.default_rng()
        for _ in range(10):
            i = rng.choice(idx)
            add = np.clip(err, -h[i], 1.0 - h[i])
            h[i] += add
            err = target - h.sum()
            if abs(err) < 1e-12:
                break
    return h

    n = h.shape[0]

    return np.max(np.correlate(h, 1 - h, mode='full')) * dx
