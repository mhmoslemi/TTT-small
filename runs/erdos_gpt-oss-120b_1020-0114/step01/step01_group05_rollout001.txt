def subgrad(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_val = np.max(corr)
    # tolerance: consider all near max, use e.g., 1e-12 or relative tolerance
    idxs = np.where(np.abs(corr - max_val) <= 1e-12)[0]
    shifts = idxs - (n - 1)
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            grad[:n - s] -= dx * h[s:]
            grad[s:] -= dx * h[:n - s]
        else:
            s = -s
            grad[s:] -= dx * h[:n - s]
            grad[:n - s] -= dx * h[s:]
    return grad

import time
import numpy as np

# ----------------------------------------------------------------------
def _evaluate(h: np.ndarray) -> float:
    """Maximum overlap integral for a given step function."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h); length 2n-1, index n-1 corresponds to shift 0
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def _project(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0≤x≤1, sum x = target} using bisection on a uniform shift."""
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(30):                     # enough for double precision
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _subgrad_and_value(h: np.ndarray):
    """Sub‑gradient of the max‑overlap functional and its value."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_val = float(corr.max())
    # all shifts attaining the maximum (within a tiny tolerance)
    tol = 1e-12
    idxs = np.where(np.abs(corr - max_val) <= tol)[0]
    shifts = idxs - (n - 1)               # shift = index - (n-1)
    grad = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            grad[: n - s] -= dx * h[s:]
            grad[s:]     -= dx * h[: n - s]
        else:
            s_abs = -s
            grad[s_abs:] -= dx * h[: n - s_abs]
            grad[: n - s_abs] -= dx * h[s_abs:]
    return grad, max_val


def _subgrad_descent(h: np.ndarray, target: float,
                     max_steps: int = 2000, start_lr: float = 0.5):
    """Simple sub‑gradient descent with decreasing step size."""
    best_h = h.copy()
    grad, val = _subgrad_and_value(best_h)
    best_val = val
    for it in range(1, max_steps + 1):
        step = start_lr / np.sqrt(it)
        h_new = _project(best_h - step * grad, target)
        grad, val = _subgrad_and_value(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new
    return best_h, best_val


def _random_feasible(n: int, target: float) -> np.ndarray:
    """Generate a random vector in [0,1]^n with prescribed sum."""
    v = np.random.rand(n)
    return _project(v, target)
