def project_to_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    # adjust via scaling may keep out-of-bound entries, so iterative approach
    for _ in range(20):
        diff = target - cur
        if abs(diff) <= 1e-12:
            break
        # variables that are not at bound
        free = np.where((h > 0) & (h < 1))[0]
        if free.size == 0:
            # adjust a random free variable, or pick any and clip
            idx = np.random.randint(len(h))

def slsqp_optimize(start_h, maxiter=1200):

    # set globals
    global _opt_n, _opt_dx, _opt_target_sum
    _opt_n = n
    _opt_dx = 2.0 / n
    _opt_target_sum = n / 2.0
    h0 = project_to_feasible(start_h, _opt_target_sum)
    init_corr = np.correlate(h0, 1 - h0, mode='full') * _opt_dx
    t0 = float(np.max(init_corr) + 1e-8)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    eq_cons = {'type': 'eq', 'fun': eq_constraint_fun}

def pgd_optimize(start_h, iters=800, step0=0.8):
    h = start_h.copy()
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    for it in range(iters):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        # compute fwd and bwd arrays
        if shift >= 0:
            fwd = np.concatenate([np.zeros(shift), h[:n-shift]])
            bwd = np.concatenate([h[shift:], np.zeros(shift)])
        else:
            s = -shift
            fwd = np.concatenate([h[s:], np.zeros(s)])
            bwd = np.concatenate([np.zeros(s), h[:n-s]])
        grad = dx - dx * (fwd + bwd)
        alpha = step0 * (0.99 ** it)
        h = h - alpha * grad
        h = project_to_feasible(h, target)
    return h
```

    cur = h.sum()

        return h
    # iterative redistribution
    for _ in range(20):

        if abs(diff) < 1e-12:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            # adjust any element without breaking bounds
            # choose element with room
            # if diff > 0, increase first element <1
            # else decrease first element >0
            if diff > 0:
                idx = np.where(h < 1.0)[0]
                if idx.size == 0:

                i = idx[0]
                h[i] = min(1.0, h[i] + diff)

                if idx.size == 0:
                    break
                i = idx[0]

        else:
            delta = diff / free.size
            h[free] += delta
            np.clip(h, 0.0, 1.0, out=h)

    return h

# globals used by the SLSQP optimizer
_opt_n = None          # number of points
_opt_dx = None         # grid spacing 2 / n
_opt_target_sum = None # n / 2

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def project_to_feasible(h, target):
    """Clip to [0,1] and adjust linearly so that sum(h) = target (approximately)."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if abs(cur - target) < 1e-12:
        return h
    for _ in range(20):
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            # No interior points – adjust a boundary point that can move.
            if diff > 0:
                idx = np.where(h < 1.0)[0]
                if idx.size == 0:
                    break
                i = idx[0]
                h[i] = min(1.0, h[i] + diff)
            else:
                idx = np.where(h > 0.0)[0]
                if idx.size == 0:
                    break
                i = idx[0]
                h[i] = max(0.0, h[i] + diff)
        else:
            delta = diff / free.size
            h[free] += delta
            np.clip(h, 0.0, 1.0, out=h)
        cur = h.sum()
    return h

def objective_fun(x):
    """Objective for SLSQP – minimise the auxiliary variable t."""
    return x[-1]

def eq_constraint_fun(x):
    """Equality: integral of h must be 1 → sum(h) = n/2."""
    return np.sum(x[:_opt_n]) - _opt_target_sum

def ineq_constraint_fun(x):
    """Inequality: for every shift, t >= overlap_k."""
    h = x[:_opt_n]
    t = x[-1]
    corr = np.correlate(h, 1 - h, mode='full') * _opt_dx
    return t - corr  # should be >= 0 for all lags

def slsqp_optimize(start_h, maxiter=1500):
    """Run SLSQP from start_h and return (h_opt, c5_value)."""
    n = len(start_h)
    global _opt_n, _opt_dx, _opt_target_sum
    _opt_n = n
    _opt_dx = 2.0 / n
    _opt_target_sum = n / 2.0

    h0 = project_to_feasible(start_h, _opt_target_sum)

    init_corr = np.correlate(h0, 1 - h0, mode='full') * _opt_dx
    t0 = float(np.max(init_corr) + 1e-8)

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    eq_cons = {'type': 'eq', 'fun': eq_constraint_fun}
    m = 2 * n - 1
    ineq_cons = NonlinearConstraint(
        ineq_constraint_fun,
        lb=np.zeros(m),
        ub=np.full(m, np.inf),
    )
    x0 = np.concatenate([h0, [t0]])

    res = minimize(
        objective_fun,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=[eq_cons, ineq_cons],
        options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False},
    )

    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    h_opt = project_to_feasible(h_opt, _opt_target_sum)

    try:
        c5_val = float(evaluate_erdos_solution(h_opt))
    except Exception:
        c5_val = float(np.max(np.correlate(h_opt, 1 - h_opt, mode='full') * _opt_dx))
    return h_opt, c5_val

def pgd_optimize(start_h, iters=800, step0=0.8):
    """Projected gradient descent on the max‑overlap objective."""
    h = start_h.copy()
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0

    for it in range(iters):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        idx = np.argmax(corr)
        shift = idx - (n - 1)

        if shift >= 0:
            fwd = np.concatenate([np.zeros(shift), h[:n - shift]])
            bwd = np.concatenate([h[shift:], np.zeros(shift)])
        else:
            s = -shift
            fwd = np.concatenate([h[s:], np.zeros(s)])
            bwd = np.concatenate([np.zeros(s), h[:n - s]])

        grad = dx - dx * (fwd + bwd)
        alpha = step0 * (0.99 ** it)
        h = h - alpha * grad
        h = project_to_feasible(h, target)

    return h

def hill_climb_binary(start_h, rng, max_swaps=5000, time_limit=5.0):
    """Simple swap‑based improvement for binary (0/1) vectors."""
    h = start_h.copy()
    n = len(h)
    best = h
    best_val = np.max(np.correlate(best, 1 - best, mode='full') * (2.0 / n))
    start = time.time()

    while max_swaps > 0 and (time.time() - start) < time_limit:
        max_swaps -= 1
        ones = np.where(best > 0.5)[0]
        zeros = np.where(best < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        val = np.max(np.correlate(cand, 1 - cand, mode='full') * (2.0 / n))
        if val < best_val - 1e-9:
            best = cand
            best_val = val
    return best, best_val
