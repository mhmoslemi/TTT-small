    def repair_sum(h, target):
        # Implementation as before but more efficient
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            return h

        # We'll add mass where possible.

                break

                diff -= inc

    max_idx = np.argmax(corr)

import numpy as np, time, random, math

# ------------------------------------------------------------
# Helper utilities – must be top‑level (no closures or lambdas)
# ------------------------------------------------------------

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust h in‑place so that sum(h) == target while 0≤h≤1."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # greedy fill/empty
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = np.random.randint(len(h))
        if diff > 0.0:
            cap = 1.0 - h[i]
            if cap <= 0.0:
                continue
            inc = min(cap, diff)
            h[i] += inc
            diff -= inc
        else:
            cap = h[i]
            if cap <= 0.0:
                continue
            dec = min(cap, -diff)
            h[i] -= dec
            diff += dec
    # final correction if still off
    if abs(diff) > 1e-12:
        total = h.sum()
        if total > 0.0:
            h *= (target / total)
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0.0:
                idx = np.where(h < 1.0)[0]
                if idx.size:
                    h[idx[0]] = min(1.0, h[idx[0]] + diff)
            else:
                idx = np.where(h > 0.0)[0]
                if idx.size:
                    h[idx[0]] = max(0.0, h[idx[0]] + diff)
    return h


def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral C5 for a discrete step function."""
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Derivative of the worst correlation w.r.t. h (sign indicates increase)."""
    N = len(h)
    g = np.zeros(N, dtype=np.float64)
    if shift >= 0:
        # term (1 - h[i+shift]) for i where i+shift < N
        g[: N - shift] += (1.0 - h[shift:])
        # term -h[i-shift] for i where i-shift >= 0
        g[shift:] -= h[: N - shift]
    else:
        s = -shift
        # (1 - h[i-shift]) for i where i-shift >= 0  (i >= s)
        g[s:] += (1.0 - h[: N - s])
        # -h[i+shift] for i where i+shift < N  (i < N-s)
        g[: N - s] -= h[s:]
    return g


# ------------------------------------------------------------
# Core optimisation routine
# ------------------------------------------------------------

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for the Erdős C5 problem."""
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    # -----------------------------------------------------------------
    # initialise h
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=np.float64).copy()
        N = len(h)
    else:
        N = 200                     # default size, even for convenience
        h = np.zeros(N, dtype=np.float64)
        half = N // 2
        h[:half] = 1.0
        np.random.shuffle(h)

    target_sum = N / 2.0
    h = _repair_sum(h, target_sum)

    current_c5 = _compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5

    # parameters for the adaptive move size
    init_move = 0.20
    final_move = 0.001
    init_temp = 0.10
    final_temp = 1e-5

    # -----------------------------------------------------------------
    # main optimisation loop
    while True:
        # time check
        elapsed = time.time() - start
        if elapsed > budget_s - 2.0:
            break
        progress = elapsed / (budget_s - 2.0)        # 0 → 1

        # adaptive step size and temperature
        max_move = init_move * (1.0 - progress) + final_move * progress
        temperature = init_temp * (1.0 - progress) + final_temp * progress

        # -----------------------------------------------------------------
        # compute the worst (largest) correlation with the complement
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        shift = max_idx - (N - 1)               # signed lag producing max

        # gradient for that lag – sign tells whether increasing h[i] helps
        g = _gradient_for_shift(h, shift)

        # choose a source (positive gradient, mass to remove) and a target
        src_mask = (h > 1e-12)
        tgt_mask = (h < 1.0 - 1e-12)
        if not src_mask.any() or not tgt_mask.any():
            break

        src_candidates = np.where(src_mask)[0]
        tgt_candidates = np.where(tgt_mask)[0]

        # best source: largest positive gradient among candidates
        src_vals = g[src_candidates]
        src = src_candidates[int(np.argmax(src_vals))]
        # best target: most negative gradient among candidates
        tgt_vals = g[tgt_candidates]
        tgt = tgt_candidates[int(np.argmin(tgt_vals))]

        # ensure we have a genuine improving direction
        if g[src] <= 0.0 or g[tgt] >= 0.0:
            # fallback: a small random transfer move
            i, j = np.random.randint(N, size=2)
            if i != j:
                delta = min(max_move, h[i], 1.0 - h[j])
                if delta > 1e-12:
                    h[i] -= delta
                    h[j] += delta
            continue

        delta = min(max_move, h[src], 1.0 - h[tgt])
        if delta <= 1e-12:
            continue

        # propose move
        h[src] -= delta
        h[tgt] += delta

        new_c5 = _compute_c5(h)

        # accept if improvement; otherwise Metropolis criterion
        if new_c5 < current_c5:
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            prob = math.exp(-(new_c5 - current_c5) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                # revert
                h[tgt] -= delta
                h[src] += delta

        # occasional larger random perturbations to escape plateaus
        if np.random.rand() < 0.005:
            cand = h.copy()
            m = np.random.randint(1, max(2, N // 30 + 1))
            idxs = np.random.choice(N, m, replace=False)
            cand[idxs] = np.random.rand(m)
            cand = _repair_sum(cand, target_sum)
            cand_c5 = _compute_c5(cand)
            if cand_c5 < current_c5 or np.random.rand() < 0.2:
                h = cand
                current_c5 = cand_c5
                if cand_c5 < best_c5:
                    best_c5 = cand_c5
                    best_h = cand.copy()

    # -----------------------------------------------------------------
    # final tidy‑up
    best_h = _repair_sum(best_h, target_sum)
    final_c5 = _compute_c5(best_h)

    return best_h.tolist(), float(final_c5), len(best_h)
