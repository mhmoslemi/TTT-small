def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    def elapsed():
        return time.time() - start

    # Decide resolution
    target_n = 400  # may increase later if time remains
    # initial vector
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n0 = h0.size
        if n0 != target_n:
            h = upsample_h(h0, target_n)
        else:
            h = h0.copy()
    else:
        # generate random balanced vector: half 1/0
        h = np.random.rand(target_n)
    # enforce constraints
    target_sum = target_n/2.0
    h = np.clip(h,0,1)
    s = h.sum()
    if s>0:
        h = h * (target_sum/s)
    # ensure bounds
    h = np.clip(h,0,1)
    # track best
    best_h = h.copy()
    best_val = compute_c5(best_h)

    # Parameters for annealing
    temp0 = 0.01  # typical scale of C5 differences
    temp = temp0
    decay = 0.9995  # per iteration
    # iteration count limited by time
    iter = 0
    while elapsed() < budget_s * 0.95:
        # compute correlation
        c = np.correlate(h, 1 - h, mode='full') * (2.0 / target_n)
        max_idx = c.argmax()
        cur_val = c[max_idx]
        shift = max_idx - (target_n - 1)

        # Determine contributions
        if shift >= 0:
            i_idx = np.arange(0, target_n - shift)
            j_idx = i_idx + shift
        else:
            i_idx = np.arange(-shift, target_n)
            j_idx = i_idx + shift

        # contributions vector
        prod = h[i_idx] * (1 - h[j_idx])
        # find index of max contribution
        local_max = prod.max()
        max_pos = prod.argmax()
        i = i_idx[max_pos]
        j = j_idx[max_pos]

        # propose movement
        eps = 0.03
        eps = min(eps, h[i], 1 - h[j])
        if eps > 1e-8:
            new_h = h.copy()
            new_h[i] -= eps
            new_h[j] += eps
        else:
            # small random perturb
            i = np.random.randint(target_n)
            j = np.random.randint(target_n)
            eps = 0.02
            eps = min(eps, h[i], 1 - h[j])
            new_h = h.copy()
            new_h[i] -= eps
            new_h[j] += eps

        # ensure constraints: sum preserved automatically (by transfer)
        # clip boundaries could be off due to floating errors, enforce
        new_h = np.clip(new_h, 0.0, 1.0)
        # re-balance sum to target_sum (should be identical)
        s_new = new_h.sum()
        if s_new != target_sum:
            # scale minor correction while keeping within bounds
            diff = target_sum - s_new
            new_h += diff / target_n
            new_h = np.clip(new_h, 0.0, 1.0)
            # if still off, renormalize
            s_new = new_h.sum()
            if s_new > 0:
                new_h = new_h * (target_sum / s_new)
            new_h = np.clip(new_h, 0.0, 1.0)

        new_val = compute_c5(new_h)

        # acceptance
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            h = new_h
            cur_val = new_val

        # track best
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
        # occasional random reset to current best + random noise
        if iter % 2000 == 0 and iter>0:
            # add small perturb to best and continue
            pert = best_h + np.random.uniform(-0.02, 0.02, size=target_n)
            pert = np.clip(pert,0,1)
            # renormalize sum
            pert = pert * (target_sum / pert.sum())
            h = pert
        # update temperature
        temp *= decay
        iter += 1

_GLOBAL_ALPHA = None

def set_global_params(n, alpha):
    """Set parameters used by the smoothed objective."""
    global _GLOBAL_N, _GLOBAL_ALPHA
    _GLOBAL_N = n
    _GLOBAL_ALPHA = alpha


def set_global_target(target):
    """Set the target integral (sum = target)."""
    global _GLOBAL_TARGET
    _GLOBAL_TARGET = target


def constraint_sum_eq(x):
    """Equality constraint Σx = target."""
    return np.sum(x) - _GLOBAL_TARGET


def objective_func(h):
    """Smooth approximation of max_{k} ∫ h(x)(1-h(x+k))."""
    n = _GLOBAL_N
    a = _GLOBAL_ALPHA
    corr = np.correlate(h, 1.0 - h, mode="full")
    smooth = logsumexp(a * corr) / a
    return smooth * (2.0 / n)


def compute_c5(h):
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)


def upsample_h(old_h, new_n):
    """Linear interpolation to a finer grid, preserving the integral."""
    old_n = old_h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    target = new_n / 2.0
    s = new_h.sum()
    if s > 0:

        s = new_h.sum()
        if abs(s - target) > 1e-12:
            new_h = new_h * (target / s)

        new_val = compute_c5(new_h)

        # Metropolis acceptance
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            cur_h = new_h
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = cur_h.copy()

        temp *= decay
        it += 1

        # occasional jitter around the current best
        if it % 3000 == 0:
            cur_h = best_h + np.random.uniform(-0.02, 0.02, size=n)
            cur_h = np.clip(cur_h, 0.0, 1.0)
            cur_h = cur_h * (target / cur_h.sum())
            cur_val = compute_c5(cur_h)

    return best_h, best_val


def slsqp_refine(h, max_seconds):
    """Few SLSQP restarts with the smoothed objective."""
    start = time.time()
    n = h.shape[0]
    target = n / 2.0
    set_global_params(n, 30.0)          # moderate smoothing
    set_global_target(target)
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": constraint_sum_eq}

    best_h = h.copy()
    best_val = compute_c5(best_h)

    restarts = 0
    while time.time() - start < max_seconds and restarts < 4:
        # random perturbation of the current best before a local run
        start_h = best_h + np.random.uniform(-0.015, 0.015, size=n)
        start_h = np.clip(start_h, 0.0, 1.0)
        start_h = start_h * (target / start_h.sum())

        res = opt.minimize(
            objective_func,
            start_h,
            method="SLSQP",
            bounds=bounds,
            constraints=[cons],
            options={"maxiter": 250, "ftol": 1e-9, "disp": False},
        )
        h_opt = np.clip(res.x, 0.0, 1.0)
        s = h_opt.sum()
        if s > 0:
            h_opt = h_opt * (target / s)
            h_opt = np.clip(h_opt, 0.0, 1.0)

        val = compute_c5(h_opt)
        if val < best_val:
            best_val = val
            best_h = h_opt.copy()
        restarts += 1

    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function giving a small C5 bound."""
    np.random.seed(seed)
    start_all = time.time()
    def elapsed():
        return time.time() - start_all

    # ------------------------------------------------------------------
    # Initial construction (provided by the harness if available)
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        # fallback: random balanced vector of size 200
        n0 = 200
        h0 = np.random.rand(n0)

    n0 = h0.size
    target0 = n0 / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    s0 = h0.sum()
    if s0 > 0:
        h0 = h0 * (target0 / s0)
    h0 = np.clip(h0, 0.0, 1.0)

    # ------------------------------------------------------------------
    # Stage 1 – up‑sample to 400 points and anneal
    # ------------------------------------------------------------------
    n1 = 400
    h = upsample_h(h0, n1)
    target1 = n1 / 2.0
    h = np.clip(h, 0.0, 1.0)
    h = h * (target1 / h.sum())
    time_left = budget_s - elapsed()
    anneal_time = min(time_left * 0.65, 600.0)
    if anneal_time > 2.0:
        h, best_val = anneal_optimize(h, anneal_time)
    else:
        best_val = compute_c5(h)

    # ------------------------------------------------------------------
    # Stage 2 – local refinement with SLSQP
    # ------------------------------------------------------------------
    time_left = budget_s - elapsed()
    slsqp_time = min(time_left * 0.25, 250.0)
    if slsqp_time > 1.0:
        h, best_val = slsqp_refine(h, slsqp_time)

    # ------------------------------------------------------------------
    # Stage 3 – optional up‑sample to 800 points for a final quick tune
    # ------------------------------------------------------------------
    time_left = budget_s - elapsed()
    if time_left > 5.0:
        n2 = 800
        h2 = upsample_h(h, n2)
        target2 = n2 / 2.0
        h2 = np.clip(h2, 0.0, 1.0)
        h2 = h2 * (target2 / h2.sum())
        # a short anneal (few seconds)
        h2, val2 = anneal_optimize(h2, min(time_left * 0.7, 30.0))
        if val2 < best_val:
            h = h2
            best_val = val2

    # ------------------------------------------------------------------
    # Final consistency enforcement
    # ------------------------------------------------------------------
    n_final = h.shape[0]
    target_final = n_final / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 0 and abs(s - target_final) > 1e-9:
        h = h * (target_final / s)
        h = np.clip(h, 0.0, 1.0)

    final_c5 = float(compute_c5(h))

    return (h, final_c5, int(n_final))
