def gradient_move(h, max_move):
    N = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    idx_max = np.argmax(corr)
    shift = idx_max - (N - 1)  # shift in [-N+1, N-1]

    # neighbor contributions
    neigh = np.zeros(N, dtype=np.float64)
    if shift > 0:
        # h[i+shift]
        neigh[:N-shift] += h[shift:]
        neigh[shift:] += h[:N-shift]
    elif shift < 0:
        s = -shift
        # i+shift = i - s, i-s for shift negative
        neigh[s:] += h[:N-s]
        neigh[:N-s] += h[s:]
    else:

        neigh = 2.0 * h

    # Determine inc/dec indices
    inc_candidates = np.where(h < 1.0)[0]
    dec_candidates = np.where(h > 0.0)[0]
    if inc_candidates.size == 0 or dec_candidates.size == 0:
        # fallback random move
        i, j = np.random.randint(N, size=2)
        max_delta = min(max_move, h[i], 1.0 - h[j])

            return h  # nothing

        h[j] += delta

    inc_values = neigh[inc_candidates]

def _compute_c5(h):
    """Maximum overlap integral for the discrete step function."""
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _repair_sum(h, target):
    """Project h onto [0,1] and enforce sum == target (exactly to machine precision)."""
    np.clip(h, 0.0, 1.0, out=h)
    total = h.sum()
    if abs(total - target) < 1e-12:
        return h
    if total > 0:
        h[:] = h * (target / total)
    np.clip(h, 0.0, 1.0, out=h)
    # fix any residual error by a tiny redistribution
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idxs = np.where(h < 1.0)[0]
            if not idxs.size:
                break
            inc = diff / idxs.size
            inc = min(inc, np.min(1.0 - h[idxs]))
            h[idxs] += inc
        else:

            if not idxs.size:

    return h

def _gradient_move(h, max_move):
    """Perform a single move that tries to improve the worst shift."""
    N = h.shape[0]
    # correlation to locate the worst shift
    corr = np.correlate(h, 1 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (N - 1)          # shift value in [-N+1, N-1]

    # neighbor contribution for the chosen shift
    neigh = np.zeros(N, dtype=np.float64)
    if shift > 0:
        neigh[:N - shift] += h[shift:]      # h[i+shift]
        neigh[shift:] += h[:N - shift]      # h[i-shift]
    elif shift < 0:
        s = -shift
        neigh[s:] += h[:N - s]              # h[i+shift]  (i - s)
        neigh[:N - s] += h[s:]              # h[i-shift] (i + s)
    else:  # shift == 0
        neigh = 2.0 * h

    inc_idxs = np.where(h < 1.0)[0]
    dec_idxs = np.where(h > 0.0)[0]
    if inc_idxs.size == 0 or dec_idxs.size == 0:
        # fallback: simple random transfer
        i, j = np.random.randint(N, size=2)
        mx = min(max_move, h[i], 1.0 - h[j])
        if mx <= 0:
            return h
        delta = np.random.uniform(0.0, mx)
        h[i] -= delta
        h[j] += delta
        return h

    # choose where to increase (large neigh) and where to decrease (small neigh)
    inc_val = neigh[inc_idxs]
    dec_val = neigh[dec_idxs]
    i_inc = inc_idxs[int(np.argmax(inc_val))]
    i_dec = dec_idxs[int(np.argmin(dec_val))]

    if i_inc == i_dec:
        # pick second best if possible
        sorted_inc = inc_idxs[np.argsort(-inc_val)]
        sorted_dec = dec_idxs[np.argsort(dec_val)]
        for cand in sorted_inc:
            if cand != i_dec:
                i_inc = cand
                break
        for cand in sorted_dec:
            if cand != i_inc:
                i_dec = cand
                break

    max_inc = min(max_move, 1.0 - h[i_inc])
    max_dec = min(max_move, h[i_dec])
    delta = min(max_inc, max_dec)
    if delta <= 0:
        # fallback random transfer
        i, j = np.random.randint(N, size=2)
        mx = min(max_move, h[i], 1.0 - h[j])
        if mx <= 0:
            return h
        delta = np.random.uniform(0.0, mx)
        h[i] -= delta
        h[j] += delta
        return h

    h[i_inc] += delta
    h[i_dec] -= delta
    return h
