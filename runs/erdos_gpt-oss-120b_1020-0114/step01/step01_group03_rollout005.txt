import time
import numpy as np
import scipy.optimize as opt
from scipy.special import logsumexp

def compute_c5(h):
    """Exact C5 value for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def upsample_h(old_h, new_n):
    """Linear interpolation to a finer grid while preserving the integral."""
    old_n = old_h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    target = new_n / 2.0
    cur = new_h.sum()
    if cur > 0:
        new_h *= target / cur
    np.clip(new_h, 0.0, 1.0, out=new_h)
    # final correction
    cur = new_h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        new_h += diff / new_n
        np.clip(new_h, 0.0, 1.0, out=new_h)
        cur = new_h.sum()
        if cur > 0:
            new_h *= target / cur
    return new_h

def sum_eq(h):
    """Equality constraint: sum(h) = n/2."""
    return np.sum(h) - h.size / 2.0

def smooth_obj(h, alpha):
    """Smooth approximation of max_{k} ∫h(1-h_k)."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return logsumexp(alpha * corr) / alpha * dx

def random_h(n):
    """Generate a random feasible vector (0≤h≤1, sum = n/2)."""
    h = np.random.rand(n)
    target = n / 2.0
    h = h / h.sum() * target
    np.clip(h, 0.0, 1.0, out=h)
    cur = h.sum()
    if cur > 0:
        h *= target / cur
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    def elapsed():
        return time.time() - start

    # ----------- initial vector ---------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n0 = h0.size
        target0 = n0 / 2.0
        s = h0.sum()
        if s != 0:
            h0 = h0 * (target0 / s)
        np.clip(h0, 0.0, 1.0, out=h0)
    else:
        n0 = 200
        h0 = random_h(n0)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # ----------- resolution schedule ----------------------------------
    max_n = 400
    # ensure we start at current size and finish at max_n
    n_seq = sorted(set([best_h.size, 200, max_n]))
    # -----------------------------------------------------------------
    for n in n_seq:
        if elapsed() > budget_s * 0.95:
            break

        # up‑sample to current resolution
        if best_h.size != n:
            cur_h = upsample_h(best_h, n)
        else:
            cur_h = best_h.copy()

        # enforce feasibility
        target_sum = n / 2.0
        s = cur_h.sum()
        if s != 0:
            cur_h = cur_h * (target_sum / s)
        np.clip(cur_h, 0.0, 1.0, out=cur_h)

        # ---------- multi‑restart SLSQP --------------------------------
        best_local_h = None
        best_local_val = np.inf
        restarts = 6
        for _ in range(restarts):
            if elapsed() > budget_s * 0.95:
                break
            # random perturbation of current guess
            pert = cur_h + np.random.uniform(-0.07, 0.07, size=n)
            np.clip(pert, 0.0, 1.0, out=pert)
            s = pert.sum()
            if s != 0:
                pert = pert * (target_sum / s)
            np.clip(pert, 0.0, 1.0, out=pert)

            alpha = 120.0
            bounds = [(0.0, 1.0)] * n
            cons = {"type": "eq", "fun": sum_eq}
            res = opt.minimize(
                smooth_obj,
                pert,
                args=(alpha,),
                method="SLSQP",
                bounds=bounds,
                constraints=[cons],
                options={"maxiter": 300, "ftol": 1e-9, "disp": False},
            )
            h_opt = res.x
            # final projection onto equality constraint
            s = h_opt.sum()
            if abs(s - target_sum) > 1e-8 and s != 0:
                h_opt = h_opt * (target_sum / s)
            np.clip(h_opt, 0.0, 1.0, out=h_opt)
            val = compute_c5(h_opt)
            if val < best_local_val:
                best_local_val = val
                best_local_h = h_opt.copy()

        if best_local_h is not None:
            cur_h = best_local_h
            cur_c5 = best_local_val
        else:
            cur_c5 = compute_c5(cur_h)

        # ---------- simulated annealing --------------------------------
        cur_best_h = cur_h.copy()
        cur_best_c5 = cur_c5

        # allocate a rough number of moves (depends on remaining time)
        remaining = budget_s - elapsed()
        moves = int(min(40000, remaining * 2500))  # ~2500 moves per sec budget
        if moves < 2000:
            moves = int(min(20000, remaining * 2500))

        eps = 0.07
        T0 = 0.018
        for it in range(moves):
            if elapsed() > budget_s * 0.95:
                break
            T = T0 * (1.0 - it / moves)
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            max_move = min(eps, cur_h[i], 1.0 - cur_h[j])
            if max_move <= 0.0:
                continue
            delta = np.random.rand() * max_move
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
            new_c5 = compute_c5(new_h)
            d = new_c5 - cur_c5
            if d < 0.0 or np.random.rand() < np.exp(-d / (T + 1e-12)):
                cur_h = new_h
                cur_c5 = new_c5
                if cur_c5 < cur_best_c5:
                    cur_best_c5 = cur_c5
                    cur_best_h = cur_h.copy()
            # slowly shrink step size
            if (it + 1) % 5000 == 0:
                eps = max(0.015, eps * 0.85)

        # ---------- greedy refinement ----------------------------------
        steps = 2500
        while steps > 0 and elapsed() < budget_s * 0.98:
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                steps -= 1
                continue
            max_move = min(0.02, cur_best_h[i], 1.0 - cur_best_h[j])
            if max_move <= 0.0:
                steps -= 1
                continue
            delta = np.random.rand() * max_move
            cand = cur_best_h.copy()
            cand[i] -= delta
            cand[j] += delta
            cand_c5 = compute_c5(cand)
            if cand_c5 < cur_best_c5:
                cur_best_h = cand
                cur_best_c5 = cand_c5
            steps -= 1

        # record improvement
        if cur_best_c5 < best_c5:
            best_c5 = cur_best_c5
            best_h = cur_best_h.copy()

    # ---------- optional random restarts at final resolution ----------
    final_n = best_h.size
    extra_trials = 4
    for _ in range(extra_trials):
        if elapsed() > budget_s * 0.98:
            break
        trial = random_h(final_n)
        trial_c5 = compute_c5(trial)
        if trial_c5 < best_c5:
            best_c5 = trial_c5
            best_h = trial.copy()
        # a quick short SA on this trial
        cur_h = trial.copy()
        cur_c5 = trial_c5
        moves = 3000
        eps = 0.06
        T0 = 0.015
        for it in range(moves):
            if elapsed() > budget_s * 0.98:
                break
            T = T0 * (1.0 - it / moves)
            i, j = np.random.randint(0, final_n, size=2)
            if i == j:
                continue
            max_move = min(eps, cur_h[i], 1.0 - cur_h[j])
            if max_move <= 0.0:
                continue
            delta = np.random.rand() * max_move
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
            new_c5 = compute_c5(new_h)
            d = new_c5 - cur_c5
            if d < 0.0 or np.random.rand() < np.exp(-d / (T + 1e-12)):
                cur_h, cur_c5 = new_h, new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = cur_h.copy()

    # ---------- final sanity check ------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_h.size / 2.0
    s = best_h.sum()
    if abs(s - target) > 1e-9 and s != 0:
        best_h = best_h * (target / s)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)

    return (best_h, float(final_c5), int(best_h.size))
