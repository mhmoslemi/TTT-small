def improve_local(h, iterations=1000):
    N = len(h)
    dx = 2.0 / N
    target_sum = N/2.0
    for _ in range(iterations):
        cor = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(cor)
        max_val = cor[max_idx]
        # compute shift k: correlation index shift index - (N-1)
        k = max_idx - (N - 1)
        # If max_val <= best possible? Continue else improvement.
        # compute gradient g_i = 1 - (h[(i+k)%N] + h[(i-k)%N])  (circular).
        idx_plus = (np.arange(N) + k) % N
        idx_minus = (np.arange(N) - k) % N
        g = 1.0 - (h[idx_plus] + h[idx_minus])
        i_max = np.argmax(g)  # largest positive derivative
        i_min = np.argmin(g)  # most negative derivative
        if g[i_max] <= 0 or g[i_min] >= 0:
            # no improvement possible using this shift; break
            break
        # Choose step amount
        max_transfer = min(h[i_max], 1.0 - h[i_min])
        # The improvement in O_k when moving delta from i_max to i_min:
        # delta_change = delta * (g[i_max] - g[i_min])? Actually O_k changes by delta* (grad_i_max - grad_i_min)
        # Since O_k derivative w.r.t h_i is g_i. Changing h_i by -delta reduces O_k by -delta * g_i (since derivative positive means O_k increases with h_i).
        # We move from i_max (decrease) to i_min (increase), so ΔO_k = -delta * g[i_max] + delta * g[i_min] = delta * (g[i_min] - g[i_max]) < 0
        # So improvement relative to O_k.
        # Choose delta = min(step_fraction*max_transfer, max_transfer)
        delta = max_transfer * 0.5  # transfer half of possible
        if delta <= 1e-12:
            break
        h[i_max] -= delta
        h[i_min] += delta
    return h
```

def compute_gradient(h, k):
    N = len(h)
    g = np.ones(N)
    # subtract neighbor forward: i+k in range
    idx = np.where((np.arange(N) + k >= 0) & (np.arange(N) + k < N))[0]
    g[idx] -= h[idx + k]
    # subtract neighbor backward: i-k in range
    idx = np.where((np.arange(N) - k >= 0) & (np.arange(N) - k < N))[0]
    g[idx] -= h[idx - k]
    return g
```

def local_improve(h, max_iters=2000, step_frac=0.5):
    N = len(h)
    for _ in range(max_iters):
        cor = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(cor)
        max_val = cor[max_idx]
        k = max_idx - (N - 1)
        g = compute_gradient(h, k)
        # find indices with positive derivative (where O_k increases if we increase h)
        pos = np.where(g > 1e-12)[0]
        neg = np.where(g < -1e-12)[0]
        if pos.size == 0 or neg.size == 0:
            break
        i_max = pos[np.argmax(g[pos])]  # most positive
        i_min = neg[np.argmin(g[neg])]  # most negative
        # possible transfer amount
        delta = min(h[i_max], 1.0 - h[i_min])
        if delta <= 0:
            break
        # choose portion
        delta *= step_frac
        if delta < 1e-9:
            break
        h[i_max] -= delta
        h[i_min] += delta
        # after transfer, continue loop
        # we also could break if improvement less than tolerance
    return h
```

    N = h.size
    for _ in range(max_swaps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_val = corr[max_idx] * (2.0 / N)
        k = max_idx - (N - 1)
        g = gradient_for_shift(h, k)
        # positions where we can move mass from 1 to 0 (positive g)
        pos = np.where((g > 1e-12) & (h > 0.5))[0]

        j = neg[np.argmin(g[neg])]

            continue
        else:
            # revert swap
            h[i], h[j] = 1.0, 0.0

import numpy as np
import time
import math

def _project(v, target):
    """Project vector v onto the box [0,1] and enforce sum(v)=target."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    # water‑filling loop
    while abs(diff) > 1e-12:
        if diff > 0:
            free = v < 1.0 - 1e-12
            nfree = free.sum()
            if nfree == 0:
                break
            add = diff / nfree
            v[free] = np.minimum(v[free] + add, 1.0)
        else:
            free = v > 1e-12
            nfree = free.sum()
            if nfree == 0:
                break
            sub = diff / nfree
            v[free] = np.maximum(v[free] + sub, 0.0)
        diff = target - v.sum()
    return v


def _c5_of(h):
    """Overlap constant C₅ for a step vector h."""
    N = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / N)


def _corr_max_shift(h):
    """Return (max_corr, shift) where shift ∈ [-(N-1), N-1] gives the max."""
    N = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    return corr[idx] * (2.0 / N), idx - (N - 1)


def _gradient_for_shift(h, k):
    """Derivative of O_k = Σ h_i (1-h_{i+k}) w.r.t each h_i."""
    N = h.size
    g = np.ones(N, dtype=float)
    ind = np.arange(N) + k
    mask = (ind >= 0) & (ind < N)
    g[mask] -= h[ind[mask]]
    ind = np.arange(N) - k
    mask = (ind >= 0) & (ind < N)
    g[mask] -= h[ind[mask]]
    return g


def _greedy_optimize(h, max_swaps=200):
    """Local improvement: move mass from positions that increase O_k to ones that decrease it."""
    N = h.size
    for _ in range(max_swaps):
        # worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        cur_val = corr[max_idx] * (2.0 / N)
        k = max_idx - (N - 1)

        g = _gradient_for_shift(h, k)

        # candidates: 1→0 where g>0, 0→1 where g<0
        pos = np.where((g > 1e-12) & (h > 0.5))[0]
        neg = np.where((g < -1e-12) & (h < 0.5))[0]
        if pos.size == 0 or neg.size == 0:
            break

        i = pos[np.argmax(g[pos])]
        j = neg[np.argmin(g[neg])]

        # perform swap
        h[i], h[j] = 0.0, 1.0

        new_corr = np.correlate(h, 1.0 - h, mode="full")
        new_val = new_corr[np.argmax(new_corr)] * (2.0 / N)

        if new_val < cur_val - 1e-12:
            continue          # keep swap
        else:
            h[i], h[j] = 1.0, 0.0   # revert
            break
    return h


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function with a small C₅."""
    np.random.seed(seed)
    start = time.time()
    best_h = None
    best_val = math.inf
    best_N = None

    # candidate sizes (even numbers) – include the size of the parent if present
    sizes = [30, 36, 42, 48, 54, 60, 66, 72, 78, 84, 90, 96]
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        if n0 % 2 == 0 and n0 not in sizes:
            sizes.append(n0)

    # main stochastic search
    while time.time() - start < budget_s * 0.9:
        N = int(np.random.choice(sizes))
        half = N // 2

        # start from a random binary configuration
        idx = np.random.choice(N, half, replace=False)
        h = np.zeros(N, dtype=float)
        h[idx] = 1.0

        # occasional occasional use of the parent as a seed (10 % of tries)
        if "initial_h_values" in globals() and N == len(initial_h_values):
            if np.random.rand() < 0.1:
                h = np.asarray(initial_h_values, dtype=float).copy()
                h = _project(h, half)

        # local greedy optimisation
        h = _greedy_optimize(h, max_swaps=300)

        # short random perturbation + another greedy pass
        for _ in range(3):
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i, j = np.random.choice(ones), np.random.choice(zeros)
            h[i], h[j] = 0.0, 1.0
            h = _greedy_optimize(h, max_swaps=150)

        val = _c5_of(h)

        if val < best_val - 1e-12:
            best_val = val
            best_h = h.copy()
            best_N = N

    # safeguard – fall back to a simple feasible vector if nothing was found
    if best_h is None:
        best_N = 2
        best_h = np.array([0.5, 0.5])
        best_val = _c5_of(best_h)

    # final projection (tiny numerical drift)
    best_h = _project(best_h, best_N / 2.0)
    final_val = _c5_of(best_h)

    # exact evaluation via provided helper, if available
    if "evaluate_erdos_solution" in globals():
        try:
            final_val = evaluate_erdos_solution(best_h.tolist())
        except Exception:
            pass

    return (best_h.tolist(), float(final_val), int(best_N))
