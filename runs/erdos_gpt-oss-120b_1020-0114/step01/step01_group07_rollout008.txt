def refine(h, target_sum, budget_seconds):
    start = time.time()
    N = len(h)
    eta = 0.5
    while time.time() - start < budget_seconds:
        # compute overlap array via correlate and find max shift
        corr = np.correlate(h, 1 - h, mode='full')
        # find index of max
        idx = np.argmax(corr)  # integer across full length
        # compute L = idx - (N - 1)
        L = idx - (N - 1)
        if L == 0:
            # if zero shift, overlapping is zero; maybe pick next max?
            # but we can break; no improvement possible
            break
        # compute gradient
        g = np.ones(N)
        # shift left and right contributions
        # For i where i+L < N, subtract h[i+L]
        if L > 0:
            g[:N-L] -= h[L:]
            g[L:] -= h[:N-L]
        elif L < 0:
            # L negative: let d = -L >0
            d = -L
            g[d:] -= h[:N-d]
            g[:N-d] -= h[d:]
        # gradient = g (should be 1 - (h_{i+L}+h_{i-L}))
        # update
        new_h = h - eta * g  # because we want to move opposite to gradient to reduce O.
        # Actually g = 1 - (h_i+...), we want to move in direction -(1 - ...) = (h_i+... - 1)
        # So new_h = h + eta * (h_{i+L}+h_{i-L} - 1) = h - eta * g
        # Clip & project
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = _project(new_h, target_sum)
        # compute new C5
        new_val = _c5_of(new_h)
        if new_val < _c5_of(h) - 1e-12:
            h = new_h
            # possibly reduce eta
            eta = min(0.5, eta * 1.05)
        else:
            eta *= 0.7
            if eta < 1e-5:
                break
    return h

def _anneal(h0, target_sum, budget):
    h = h0.copy()
    best_h = h.copy()
    best_val = _c5_of(h)
    cur_val = best_val
    start = time.time()
    N = h.size
    max_temp = 0.2
    min_temp = 1e-7
    max_step = 0.8
    min_step = 0.001
    no_improve = 0
    while True:
        elapsed = time.time() - start
        if elapsed > budget * 0.95:
            break
        progress = elapsed / (budget * 0.95)
        T = max_temp * (1 - progress) + min_temp
        step = max_step * (1 - progress) + min_step
        # propose move
        i = np.random.randint(N)
        j = np.random.randint(N-1)
        if j >= i:
            j += 1
        # compute feasible delta bounds
        lower = max(-h[i], h[j] - 1.0)
        upper = min(1.0 - h[i], h[j])
        if lower >= upper:
            continue
        delta_low = max(lower, -step)
        delta_high = min(upper, step)
        if delta_low >= delta_high:
            continue
        delta = np.random.uniform(delta_low, delta_high)
        new_h = h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_val = _c5_of(new_h)
        if new_val < cur_val or np.random.rand() < math.exp(-(new_val - cur_val) / T):
            h, cur_val = new_h, new_val
            if new_val < best_val - 1e-12:
                best_h, best_val = new_h.copy(), new_val
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 200_000:
            # random restart perturb
            idxs = np.random.choice(N, size=5, replace=False)
            h[idxs] = np.clip(h[idxs] + np.random.uniform(-step, step, size=5), 0.0, 1.0)
            h = _project(h, target_sum)
            cur_val = _c5_of(h)
            no_improve = 0
    # final projection
    best_h = _project(best_h, target_sum)
    return best_h, _c5_of(best_h)

import numpy as np
import time, math

def _project(v, target):
    """Project v onto [0,1]^n with sum(v)=target."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    # water‑filling
    while abs(diff) > 1e-12:
        if diff > 0:
            free = v < 1.0 - 1e-12
            nfree = free.sum()
            if nfree == 0:
                break
            add = diff / nfree
            v[free] = np.minimum(v[free] + add, 1.0)
        else:
            free = v > 1e-12
            nfree = free.sum()
            if nfree == 0:
                break
            sub = diff / nfree
            v[free] = np.maximum(v[free] + sub, 0.0)
        diff = target - v.sum()
    return v


def _c5_of(h):
    """Maximum overlap integral for a step vector h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)


def _anneal(start_h, target_sum, budget):
    """Simulated‑annealing on a single run."""
    h = start_h.copy()
    cur_val = _c5_of(h)
    best_h, best_val = h.copy(), cur_val

    N = h.size
    max_temp, min_temp = 0.2, 1e-7
    max_step, min_step = 0.8, 0.001
    no_improve = 0
    t0 = time.time()
    while True:
        elapsed = time.time() - t0
        if elapsed > budget * 0.95:
            break
        prog = elapsed / (budget * 0.95)
        T = max_temp * (1 - prog) + min_temp
        step = max_step * (1 - prog) + min_step

        i, j = np.random.randint(N, size=2)
        if i == j:
            continue
        # feasible delta so both stay in [0,1]
        lo = max(-h[i], h[j] - 1.0)
        hi = min(1.0 - h[i], h[j])
        if lo >= hi:
            continue
        delta_low = max(lo, -step)
        delta_high = min(hi, step)
        if delta_low >= delta_high:
            continue
        delta = np.random.uniform(delta_low, delta_high)

        new_h = h.copy()
        new_h[i] += delta
        new_h[j] -= delta

        new_val = _c5_of(new_h)

        if new_val < cur_val or np.random.rand() < math.exp(-(new_val - cur_val) / T):
            h, cur_val = new_h, new_val
            if new_val < best_val - 1e-12:
                best_h, best_val = new_h.copy(), new_val
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 200_000:
            # small random perturbation then project
            idx = np.random.choice(N, size=4, replace=False)
            h[idx] = np.clip(h[idx] + np.random.uniform(-step, step, size=4), 0.0, 1.0)
            h = _project(h, target_sum)
            cur_val = _c5_of(h)
            no_improve = 0

    best_h = _project(best_h, target_sum)
    return best_h, _c5_of(best_h)


def _refine(h, target_sum, budget):
    """Gradient descent on the worst shift."""
    N = h.size
    eta = 0.5
    start = time.time()
    cur_val = _c5_of(h)
    while time.time() - start < budget:
        # correlation and worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        L = idx - (N - 1)
        # ignore the zero shift if it becomes the max
        if L == 0:
            corr[N - 1] = -np.inf
            idx = int(np.argmax(corr))
            L = idx - (N - 1)
            if L == 0:
                break

        # gradient of the worst overlap
        g = np.ones(N)
        if L > 0:
            g[:N - L] -= h[L:]
            g[L:] -= h[:N - L]
        elif L < 0:
            d = -L
            g[d:] -= h[:N - d]
            g[:N - d] -= h[d:]

        # step opposite to gradient
        new_h = h - eta * g
        np.clip(new_h, 0.0, 1.0, out=new_h)
        new_h = _project(new_h, target_sum)

        new_val = _c5_of(new_h)
        if new_val < cur_val - 1e-12:
            h, cur_val = new_h, new_val
            eta = min(0.5, eta * 1.05)
        else:
            eta *= 0.7
            if eta < 1e-6:
                break
    return h
