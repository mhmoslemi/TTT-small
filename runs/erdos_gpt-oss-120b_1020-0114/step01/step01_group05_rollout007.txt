import time
import numpy as np

def _project(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto the box 0≤x≤1 and the affine hyperplane Σx = target."""
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def _eval(h: np.ndarray) -> float:
    """C₅ value for a given h (full correlation with 1‑h)."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full') * dx)

def _grad_lag(h: np.ndarray, lag: int) -> np.ndarray:
    """Gradient of the overlap term for a fixed lag."""
    n = h.shape[0]
    g = np.zeros(n, dtype=float)
    if lag >= 0:
        k = lag
        g[: n - k] += 1.0 - h[k:]
        g[k:] -= h[: n - k]
    else:
        k = -lag
        g[k:] += 1.0 - h[: n - k]
        g[: n - k] -= h[k:]
    return g

def _subgrad(h: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the max‑overlap objective (gradient at the current worst lag)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    idx_max = int(np.argmax(corr))
    lag = idx_max - (n - 1)
    return dx * _grad_lag(h, lag)

def _local_optimize(
    h0: np.ndarray,
    target: float,
    max_iters: int,
    init_step: float,
    time_limit: float,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with simple back‑tracking."""
    h = h0.copy()
    cur_val = _eval(h)
    best_h = h.copy()
    best_val = cur_val
    stagnation = 0

    for it in range(max_iters):
        if time.time() > time_limit:
            break

        g = _subgrad(h)
        direction = -g
        step = init_step / np.sqrt(it + 1.0)

        improved = False
        for _ in range(10):
            cand = _project(h + step * direction, target)
            cand_val = _eval(cand)
            if cand_val < cur_val - 1e-12:
                h = cand
                cur_val = cand_val
                improved = True
                if cand_val < best_val:
                    best_val = cand_val
                    best_h = cand.copy()
                stagnation = 0
                break
            step *= 0.5

        if not improved:
            stagnation += 1

        if stagnation >= 200:
            jitter = np.random.normal(scale=0.02, size=h.shape[0])
            h = _project(h + jitter, target)
            cur_val = _eval(h)
            stagnation = 0

    return best_h, best_val

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # initialise problem size and a feasible start vector
    # ------------------------------------------------------------------

        h_start = np.full(n, 0.5)

    target_sum = n * 0.5          # ∫₀² h = 1  ⇒  Σh = n/2
    h_start = _project(h_start, target_sum)

    best_h = h_start.copy()
    best_val = _eval(best_h)

    # ------------------------------------------------------------------
    # main multi‑restart loop – runs until the allotted budget is exhausted
    # ------------------------------------------------------------------
    restart = 0
    while time.time() - start < budget_s * 0.93:

            h0 = best_h.copy()
        else:

                h0 = _project(best_h + jitter, target_sum)
            else:
                # completely random feasible point
                rnd = np.random.rand(n)
                h0 = _project(rnd, target_sum)

        # allocate a modest per‑run iteration budget
        time_limit = start + budget_s * 0.93
        h_opt, val_opt = _local_optimize(
            h0,

        )

        if val_opt < best_val:

            best_h = h_opt.copy()

        # stop early if we have already beaten the target record
        if best_val < 0.3808:
            break

        restart += 1

    # final sanity check (re‑evaluate after potential floating‑point drift)
    final_val = _eval(best_h)
    return best_h, final_val, n
