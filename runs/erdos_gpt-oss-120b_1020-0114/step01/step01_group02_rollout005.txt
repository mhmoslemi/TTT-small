def project_onto_sum_and_box(v, target, max_iter=1000):
    x = np.clip(v, 0, 1)
    for _ in range(max_iter):
        diff = target - x.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = np.where(x < 1)[0]
        else:
            free = np.where(x > 0)[0]
        if free.size == 0:
            # cannot adjust further, break
            break
        delta = diff / free.size
        # applying delta may push some beyond bounds, so do in two steps:
        x[free] += delta
        np.clip(x, 0, 1, out=x)
    # final correction for any tiny residual
    residual = target - x.sum()
    if abs(residual) > 1e-12:
        # adjust a random free variable if possible
        if residual > 0:
            idx = np.where(x < 1)[0]
            if idx.size:
                i = idx[0]
                x[i] = min(1.0, x[i] + residual)
        else:
            idx = np.where(x > 0)[0]
            if idx.size:
                i = idx[0]
                x[i] = max(0.0, x[i] + residual)
    return x

import numpy as np
import time, math, random

# --------------------------------------------------------------
# projection onto box [0,1] and a linear sum constraint
def _project_sum_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto {0<=x<=1, sum x = target} (L2‑projection)."""
    x = np.clip(v, 0.0, 1.0)
    for _ in range(2000):
        diff = target - x.sum()
        if abs(diff) < 1e-12:
            break
        # free variables in the direction we need to move
        if diff > 0:
            free = np.where(x < 1.0)[0]
        else:
            free = np.where(x > 0.0)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        # apply and re‑clip
        x[free] += delta
        np.clip(x, 0.0, 1.0, out=x)
    # tiny residual fix (should be extremely rare)
    residual = target - x.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.where(x < 1.0)[0]
            if idx.size:
                i = idx[0]
                x[i] = min(1.0, x[i] + residual)
        else:
            idx = np.where(x > 0.0)[0]
            if idx.size:
                i = idx[0]
                x[i] = max(0.0, x[i] + residual)
    return x

# --------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    """Exact C5 for a discretised step function."""
    N = h.shape[0]
    ov = np.correlate(h, h, mode="full")          # sum_i h_i h_{i+k}
    min_overlap = ov.min()
    return (2.0 / N) * (h.sum() - min_overlap)    #  = max_k ∫h(1‑h(·+k))

# --------------------------------------------------------------
def _gradient_for_lag(h: np.ndarray, lag: int) -> np.ndarray:
    """Gradient of the overlap S_k = Σ h_i h_{i+lag} w.r.t. h."""
    N = h.shape[0]
    grad = np.zeros(N, dtype=np.float64)
    if lag == 0:
        grad = 2.0 * h
        return grad
    k = abs(lag)
    # contributions from pairs (i,i+k)
    grad[: N - k] += h[k:]
    grad[k:] += h[: N - k]
    return grad

# --------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step function h, its C5 bound and the number of samples."""
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    # ------------------------------------------------------------------
    # problem size (increased from 95 to give the optimiser more freedom)
    N = 250                       # number of discretisation points
    dx = 2.0 / N
    target_sum = N / 2.0          # integral = 1

    # ------------------------------------------------------------------
    # initialise h – use parent construction if available, else random
    if "initial_h_values" in globals():
        old = np.array(initial_h_values, dtype=np.float64)
        Nold = old.shape[0]
        x_old = np.linspace(0.0, 2.0, Nold, endpoint=False)
        x_new = np.linspace(0.0, 2.0, N, endpoint=False)
        h = np.interp(x_new, x_old, old)
    else:
        h = np.random.rand(N)

    h = _project_sum_box(h, target_sum)

    # ------------------------------------------------------------------
    # keep the best solution found
    current_c5 = _c5_from_h(h)
    best_h = h.copy()
    best_c5 = current_c5

    # annealing schedule
    t_start = 0.12
    t_end = 1e-6
    max_move_start = 0.22
    prob_big = 0.003          # occasional large random jump
    prob_grad = 0.55          # use gradient‑guided transfer most of the time
    # ------------------------------------------------------------------
    total_time = budget_s - 5.0   # safety margin
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed > total_time:
            break

        progress = elapsed / total_time
        # temperature (geometric cooling)
        T = t_start * (t_end / t_start) ** progress
        # maximal transfer at this stage
        max_move = max_move_start * (1.0 - progress) + 1e-4

        # --------------------------------------------------------------
        # decide which kind of move to try
        r = random.random()
        if r < prob_big:
            # large random perturbation of the best known solution
            cand = best_h.copy()
            m = random.randint(1, max(2, N // 12 + 1))
            idxs = np.random.choice(N, m, replace=False)
            cand[idxs] = np.random.rand(m)
            cand = _project_sum_box(cand, target_sum)
            cand_c5 = _c5_from_h(cand)

        elif r < prob_big + prob_grad:
            # gradient‑guided transfer move aimed at the worst lag
            ov = np.correlate(h, h, mode="full")
            lag = int(np.argmin(ov) - (N - 1))
            grad = _gradient_for_lag(h, lag)

            # choose a donor (decrease) where gradient is smallest
            dec_mask = h > 1e-12
            if not np.any(dec_mask):
                continue
            i_candidates = np.where(dec_mask)[0]
            i = i_candidates[np.argmin(grad[i_candidates])]

            # choose a receiver (increase) where gradient is largest
            inc_mask = h < 1.0 - 1e-12
            if not np.any(inc_mask):
                continue
            j_candidates = np.where(inc_mask)[0]
            j = j_candidates[np.argmax(grad[j_candidates])]

            if i == j:
                continue
            max_delta = min(max_move, h[i], 1.0 - h[j])
            if max_delta <= 0.0:
                continue
            delta = random.uniform(0.0, max_delta)
            cand = h.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _project_sum_box(cand, target_sum)
            cand_c5 = _c5_from_h(cand)

        else:
            # simple unbiased transfer move (preserves sum exactly)
            i, j = random.randrange(N), random.randrange(N)
            if i == j:
                continue
            max_delta = min(max_move, h[i], 1.0 - h[j])
            if max_delta <= 0.0:
                # try the opposite direction
                max_delta = min(max_move, h[j], 1.0 - h[i])
                if max_delta <= 0.0:
                    continue
                i, j = j, i
            delta = random.uniform(0.0, max_delta)
            cand = h.copy()
            cand[i] -= delta
            cand[j] += delta
            # sum is still exact; just enforce the box
            cand = np.clip(cand, 0.0, 1.0)
            cand = _project_sum_box(cand, target_sum)
            cand_c5 = _c5_from_h(cand)

        # --------------------------------------------------------------
        # Metropolis acceptance
        delta_c5 = cand_c5 - current_c5
        if delta_c5 < 0.0 or random.random() < math.exp(-delta_c5 / max(T, 1e-12)):
            h = cand
            current_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = cand.copy()
        # optional early exit if we already beat the target
        if best_c5 < 0.3808:
            break

    # ------------------------------------------------------------------
    # final tidy‑up – ensure constraints are satisfied exactly
    best_h = _project_sum_box(best_h, target_sum)
    final_c5 = _c5_from_h(best_h)

    return best_h.tolist(), float(final_c5), N
