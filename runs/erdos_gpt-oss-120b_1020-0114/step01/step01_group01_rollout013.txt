def binary_hill_climb(n, time_limit):
    # initialize h as beatty pattern
    phi = (1+np.sqrt(5))/2
    idx = np.arange(n)

    # adjust to have exactly n//2 ones
    ones = int(h.sum())
    target = n // 2
    if ones > target:
        # turn some ones to zero
        ones_idx = np.where(h == 1)[0]
        to_flip = rng.choice(ones_idx, ones - target, replace=False)
        h[to_flip] = 0
    elif ones < target:
        zeros_idx = np.where(h == 0)[0]
        to_flip = rng.choice(zeros_idx, target - ones, replace=False)
        h[to_flip] = 1
    best_h = h.copy()
    best_c5 = compute_c5_from_h(h)
    cur_h = h.copy()
    cur_c5 = best_c5
    # start annealing/hill climbing loop
    attempts_without_improve = 0
    while time.time() < end_time:
        # compute worst shift
        corr_int = np.correlate(cur_h, 1 - cur_h, mode='full')
        max_cnt = corr_int.max()

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap C5 for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def compute_counts(h: np.ndarray) -> np.ndarray:
    """Integer cross‑correlation counts (no dx) used for fast decision making."""
    return np.correlate(h, 1 - h, mode="full")


def random_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a binary vector with exactly n/2 ones (n must be even)."""
    h = np.zeros(n, dtype=int)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1
    return h


def beatty_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Binary sequence derived from the Beatty (golden‑ratio) construction."""
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    idx = np.arange(n)
    h = ((idx * phi) % 1.0 < 0.5).astype(int)

    target = n // 2
    cur = int(h.sum())
    if cur > target:
        excess = cur - target
        ones_idx = np.where(h == 1)[0]
        to_zero = rng.choice(ones_idx, excess, replace=False)
        h[to_zero] = 0
    elif cur < target:
        deficit = target - cur
        zeros_idx = np.where(h == 0)[0]
        to_one = rng.choice(zeros_idx, deficit, replace=False)
        h[to_one] = 1
    return h


def hill_climb_one_start(start_h: np.ndarray,
                         rng: np.random.Generator,
                         max_iter: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple deterministic hill‑climb: repeatedly try a targeted swap that
    does not increase the current maximal overlap.  Stops after `max_iter`
    iterations or when `deadline` is reached.
    Returns the best vector seen during the run and its C5 value.
    """
    n = start_h.shape[0]
    cur_h = start_h.copy()
    cur_counts = compute_counts(cur_h)
    cur_max = int(cur_counts.max())
    cur_c5 = cur_max * (2.0 / n)

    best_h = cur_h.copy()
    best_c5 = cur_c5
    no_improve = 0

    for _ in range(max_iter):
        if time.time() >= deadline:
            break

        # worst shift (as integer lag)
        lag = int(np.argmax(cur_counts) - (n - 1))

        # indices producing a contribution for this lag
        i_start = max(0, -lag)
        i_end = min(n - 1, n - 1 - lag)
        i_range = np.arange(i_start, i_end + 1)
        j_range = i_range + lag
        hi = cur_h[i_range]
        hj = cur_h[j_range]

        I = i_range[(hi == 1) & (hj == 0)]   # 1‑0 pairs (contribute)
        J = i_range[(hi == 0) & (hj == 1)]   # 0‑1 pairs (safe to flip)

        if I.size and J.size:
            i_swap = rng.choice(I)
            j_swap = rng.choice(J)
        else:
            # fallback: random admissible swap
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]
            i_swap = rng.choice(ones)
            j_swap = rng.choice(zeros)

        new_h = cur_h.copy()
        new_h[i_swap] = 0
        new_h[j_swap] = 1
        new_counts = compute_counts(new_h)
        new_max = int(new_counts.max())

        # accept only if the maximal count does not increase
        if new_max <= cur_max:
            cur_h = new_h
            cur_counts = new_counts
            cur_max = new_max
            cur_c5 = new_max * (2.0 / n)

            if cur_c5 + 1e-12 < best_c5:
                best_h = cur_h.copy()
                best_c5 = cur_c5
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # occasional restart from the best seen so far
        if no_improve >= 2000:
            cur_h = best_h.copy()
            cur_counts = compute_counts(cur_h)
            cur_max = int(cur_counts.max())
            no_improve = 0

    return best_h, best_c5


def binary_hill_climb_with_restarts(n: int,
                                    deadline: float,
                                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run several independent hill‑climbs (different seeds) within the time budget.
    Returns the best binary vector found and its C5 value.
    """
    best_h = None
    best_c5 = float("inf")

    # a couple of deterministic seeds
    seed_patterns = [beatty_binary(n, rng), random_binary(n, rng)]
    # ensure at least one plain block pattern (helps early exploration)
    block = np.zeros(n, dtype=int)
    block[: n // 2] = 1
    seed_patterns.append(block)

    # iterate while time permits
    while time.time() < deadline:
        if seed_patterns:
            start_h = seed_patterns.pop(0)
        else:
            start_h = random_binary(n, rng)

        # allocate a modest iteration budget; longer runs are cheap for n≤200
        max_iter = 20000 if n <= 120 else 50000
        h_local, c5_local = hill_climb_one_start(start_h, rng, max_iter, deadline)
        if c5_local < best_c5:
            best_h, best_c5 = h_local, c5_local

        # early exit if already below record target
        if best_c5 <= 0.3808:
            break

    return best_h, best_c5


# ----------------------------------------------------------------------
# Continuous refinement (SLSQP) – copies of the earlier approach
# ----------------------------------------------------------------------
_opt_n = None
_opt_dx = None
_opt_target_sum = None


def project_to_feasible(h: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and linearly rescale so that sum(h) ≈ target."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == 0.0:
        # fill exactly `target` mass on random coordinates
        idx = np.random.choice(len(h), int(round(target)), replace=False)
        h[idx] = 1.0
        return h
    scale = target / cur
    h = np.clip(h * scale, 0.0, 1.0)
    # iterative correction for remaining drift
    for _ in range(10):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            i = np.random.randint(len(h))
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
            continue
        delta = diff / free.size
        h[free] += delta
        np.clip(h, 0.0, 1.0, out=h)
    return h


def objective_fun(x):
    return x[-1]


def eq_constraint_fun(x):
    return np.sum(x[:_opt_n]) - _opt_target_sum


def ineq_constraint_fun(x):
    h = x[:_opt_n]
    t = x[-1]
    corr = np.correlate(h, 1 - h, mode="full") * _opt_dx
    return t - corr  # should be >= 0


def optimise_continuous(start_h: np.ndarray) -> tuple[np.ndarray, float]:
    """Run a short SLSQP optimisation starting from `start_h`."""
    global _opt_n, _opt_dx, _opt_target_sum
    n = int(start_h.shape[0])
    _opt_n = n
    _opt_dx = 2.0 / n
    _opt_target_sum = n / 2.0

    h0 = project_to_feasible(start_h, _opt_target_sum)

    init_corr = np.correlate(h0, 1 - h0, mode="full") * _opt_dx
    t0 = float(np.max(init_corr) + 1e-8)

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    eq_cons = {"type": "eq", "fun": eq_constraint_fun}
    m = 2 * n - 1
    ineq_cons = NonlinearConstraint(
        ineq_constraint_fun,
        lb=np.zeros(m),
        ub=np.full(m, np.inf),
    )
    x0 = np.concatenate([h0, [t0]])

    res = minimize(
        objective_fun,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=[eq_cons, ineq_cons],
        options={"maxiter": 300, "ftol": 1e-9, "disp": False},
    )

    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    h_opt = project_to_feasible(h_opt, _opt_target_sum)

    try:
        c5_val = float(evaluate_erdos_solution(h_opt))
    except Exception:
        c5_val = float(res.x[-1])
    return h_opt, c5_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine basic grid size (use provided vector if present)
    # --------------------------------------------------------------
    try:
        init_vec = np.array(initial_h_values, dtype=float)  # supplied by harness
        n0 = int(init_vec.shape[0])
    except Exception:
        n0 = 100
        init_vec = None

    # --------------------------------------------------------------
    # Phase 1 – binary hill‑climbing on the base grid
    # --------------------------------------------------------------
    deadline_bin = start_time + budget_s * 0.60  # allocate ~60 % of total time
    best_h = None
    best_c5 = float("inf")
    best_n = None

    if n0 % 2 == 0:  # binary search only makes sense for even n
        h_bin, c5_bin = binary_hill_climb_with_restarts(n0, deadline_bin, rng)
        if h_bin is not None and c5_bin < best_c5:
            best_h, best_c5, best_n = h_bin.astype(float), c5_bin, n0

    # --------------------------------------------------------------
    # Phase 2 – upscale to 2·n0 (if time permits) and repeat
    # --------------------------------------------------------------
    if time.time() < deadline_bin and n0 % 2 == 0:
        n2 = 2 * n0
        if best_h is not None and best_h.shape[0] == n0:
            # simply repeat each entry – keeps the exact number of ones
            h_up = np.repeat(best_h, 2)
        else:
            h_up = None
        deadline_up = deadline_bin  # same overall deadline
        h_bin2, c5_bin2 = binary_hill_climb_with_restarts(n2, deadline_up, rng)
        if h_bin2 is not None and c5_bin2 < best_c5:
            best_h, best_c5, best_n = h_bin2.astype(float), c5_bin2, n2

    # --------------------------------------------------------------
    # Phase 3 – continuous refinement via SLSQP (if any time left)
    # --------------------------------------------------------------
    deadline_cont = start_time + budget_s * 0.95
    if best_h is not None and time.time() < deadline_cont:
        try:
            h_opt, c5_opt = optimise_continuous(best_h.astype(float))
            if c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt, c5_opt, h_opt.shape[0]
        except Exception:
            # any failure just falls back to the binary solution
            pass

    # --------------------------------------------------------------
    # Fallback – use the supplied vector or a uniform half‑density trial
    # --------------------------------------------------------------
    if best_h is None:
        if init_vec is not None:
            best_h = init_vec.astype(float)
        else:
            best_h = np.full(n0, 0.5, dtype=float)
        best_c5 = compute_c5(best_h)
        best_n = best_h.shape[0]

    # final sanity – enforce the integral constraint very loosely
    target_sum = best_n / 2.0
    cur_sum = best_h.sum()
    if abs(cur_sum - target_sum) > 1e-8:
        best_h = project_to_feasible(best_h, target_sum)

    return best_h, float(best_c5), int(best_n)
