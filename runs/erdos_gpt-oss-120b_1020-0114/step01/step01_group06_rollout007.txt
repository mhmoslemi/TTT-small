def gradient_descent(h, max_iter, gamma_start, gamma_factor, gamma_step_iters):
    target = len(h)/2
    best = h.copy()
    best_val = _c5_value(h)
    gamma = gamma_start
    for it in range(max_iter):
        # compute f vector
        corr = np.correlate(h, 1 - h, mode='full')
        # compute softmax probs
        max_corr = corr.max()
        exps = np.exp(gamma * (corr - max_corr))
        probs = exps / exps.sum()
        # gradient via convolution
        p = probs
        conv1 = np.convolve(p, h, mode='full')
        conv2 = np.convolve(p[::-1], h, mode='full')
        slice_start = len(h)-1  # n-1
        grad = np.ones_like(h) - (conv1[slice_start:slice_start+len(h)] + conv2[slice_start:slice_start+len(h)])
        # step size
        step = 0.05 / np.sqrt(it+1)
        h_new = h - step * grad
        # projection
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = _project_sum(h_new, target)
        # evaluate
        val = _c5_value(h_new)
        if val < best_val:
            best_val, best = val, h_new.copy()
        h = h_new
        if (it+1) % gamma_step_iters == 0:
            gamma = min(gamma * gamma_factor, 1e4)
        # time check to break if exceeding budget - we will compute externally
    return best, best_val

import time, math, numpy as np

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a step vector."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

def _project_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Project h onto the hyperplane sum(h)=target while staying in [0,1]."""
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # indices that can move in needed direction
    caps = (1.0 - h) if diff > 0 else (-h)
    for i in np.where(caps != 0)[0]:
        take = min(abs(diff), abs(caps[i]))
        take = math.copysign(take, diff)
        h[i] += take
        diff -= take
        if abs(diff) < 1e-12:
            break
    return h

def _softmax_probs(arr: np.ndarray, gamma: float) -> np.ndarray:
    """Numerically stable softmax with temperature gamma."""
    if gamma == 0:
        probs = np.ones_like(arr) / arr.size
    else:
        m = arr.max()
        ex = np.exp(gamma * (arr - m))
        probs = ex / ex.sum()
    return probs

def _gradient(h: np.ndarray, probs: np.ndarray) -> np.ndarray:
    """Gradient of the soft‑max‑smoothed max objective."""
    n = h.shape[0]
    # convolution of probs with h
    conv1 = np.convolve(probs, h, mode="full")
    conv2 = np.convolve(probs[::-1], h, mode="full")
    start = n - 1
    grad = np.ones(n) - (conv1[start:start + n] + conv2[start:start + n])
    return grad

def _optimize_one_start(h0: np.ndarray, target: float,
                        max_iters: int, time_limit: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """Gradient‑descent with continuation on the temperature."""
    n = h0.shape[0]
    h = h0.copy()
    best_h = h.copy()
    best_val = _c5_value(h)

    gamma = 1.0                     # initial temperature factor
    gamma_step = max_iters // 5 if max_iters >= 5 else 1
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break

        # correlation values (unnormalised)
        corr = np.correlate(h, 1.0 - h, mode="full")
        probs = _softmax_probs(corr, gamma)

        # gradient step
        grad = _gradient(h, probs)
        step = 0.10 / math.sqrt(it + 1.0)   # decreasing step size
        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = _project_sum(h_new, target)

        val = _c5_value(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        h = h_new

        # increase temperature (make soft‑max sharper) periodically
        if (it + 1) % gamma_step == 0:
            gamma = min(gamma * 1.5, 500.0)

    return best_h, best_val

def _resample(h_old: np.ndarray, n_new: int) -> np.ndarray:
    """Linear interpolation of a step vector to a different length."""
    n_old = h_old.shape[0]
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
