    n = len(h)
    grad = np.zeros_like(h)
    # w length = 2n-1
    # s vector: s = np.arange(-(n-1), n)
    for idx, wk in enumerate(w):
        if wk == 0:
            continue

    return grad

import time
import numpy as np

# ------------------------------------------------------------
# Helper: projection onto the capped simplex {0<=x_i<=1, sum x = target}
# ------------------------------------------------------------
def project_onto_capped_simplex(v, target, tol=1e-12, max_iter=60):
    """Euclidean projection of v onto {0<=x_i<=1, sum x = target}."""
    # Binary search on lambda such that sum(clip(v - lam,0,1)) = target
    lo = -np.max(v) - 1.0          # makes all entries >1 => sum=n
    hi = np.max(v) + 1.0           # makes all entries <0 => sum=0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
        if hi - lo < tol:
            break
    proj = np.clip(v - hi, 0.0, 1.0)
    # tiny correction to enforce the sum exactly (rarely needed)
    diff = target - proj.sum()
    if abs(diff) > 1e-10:
        # distribute the remaining mass uniformly over free variables
        free = (proj > 0.0) & (proj < 1.0)
        if free.any():
            proj[free] += diff / free.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

# ------------------------------------------------------------
# Gradient of the smooth‑max objective
# ------------------------------------------------------------
def compute_gradient(h, w, dx):
    """Gradient of f(h)=softmax_alpha(overlaps) w.r.t. h."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    # w corresponds to shifts s = -(n-1) .. (n-1)
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        s = idx - (n - 1)          # shift
        if s >= 0:
            # first term: i in [0, n-s)
            i_end = n - s
            grad[0:i_end] += wk * (1.0 - h[s:n])
            # second term: j in [s, n)
            grad[s:n]    -= wk * h[0:i_end]
        else:
            t = -s
            # first term: i in [t, n)
            grad[t:n]    += wk * (1.0 - h[0:n - t])
            # second term: j in [0, n-t)
            grad[0:n - t] -= wk * h[t:n]
    grad *= dx
    return grad

def optimise_h_gradient(h_init, max_iter=800, init_step=0.5):

    h = np.asarray(h_init, dtype=float).copy()
    n = h.shape[0]
    dx = 2.0 / n
    target = n / 2.0
    h = project_onto_capped_simplex(h, target)

    # keep track of the best encountered point
    best_h = h.copy()
    best_c5 = float(evaluate_erdos_solution(best_h))

    # schedule of smoothing parameters
    alphas = [2, 5, 10, 20, 40, 80]
    it_per_alpha = max_iter // len(alphas)
    step = init_step

    for alpha in alphas:
        for _ in range(it_per_alpha):
            # overlaps for all shifts
            overlaps = np.correlate(h, 1.0 - h, mode="full") * dx
            # stable softmax weights
            m = np.max(overlaps)
            w = np.exp(alpha * (overlaps - m))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            grad = compute_gradient(h, w, dx)

            # gradient descent step + projection
            h_new = h - step * grad
            h = project_onto_capped_simplex(h_new, target)

            step *= 0.9995      # gentle decay

        # evaluation after each smoothing stage
        c5_val = float(evaluate_erdos_solution(h))
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
    return best_h, best_c5
