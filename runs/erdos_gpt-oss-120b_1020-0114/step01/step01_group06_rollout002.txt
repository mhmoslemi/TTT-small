    n = len(initial_h)
    cur_h = initial_h.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    for it in range(max_iters):
        # time check
        if time.time() - start_time > time_limit:
            break
        # schedule
        tau = it / max_iters
        temp = init_temp * (1 - tau) + final_temp * tau
        step = init_step * (1 - tau) + final_step * tau

        # propose move
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # compute feasible delta bounds
        # delta added to h[i], sub from h[j], keep bounds
        lower = max(-cur_h[i], -(1 - cur_h[j]))
        upper = min(1 - cur_h[i], cur_h[j])
        if upper <= lower:
            continue
        delta = rng.uniform(max(lower, -step), min(upper, step))

        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta

        if new_c5 < cur_c5 or rng.random() < math.exp(-(new_c5 - cur_c5) / temp):

import time, math, numpy as np

def _c5_value(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def _project_sum(h, target):
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # Determine direction
    if diff > 0:
        caps = 1.0 - h
    else:
        caps = -h
    # Process in deterministic order
    for idx in np.where(caps > 1e-12)[0]:
        take = min(abs(diff), caps[idx])
        if diff < 0:
            take = -take
        h[idx] += take
        diff -= take
        if abs(diff) < 1e-12:
            break
    return h

def _thue_morse(n):
    # return array of 0/1 floats
    return np.fromiter((i.bit_count() & 1 for i in range(n)), dtype=float, count=n)

        return float(corr.max() * dx)

    h_rand[idx] = 1.0
    candidates.append(h_rand)

    # Evaluate candidates, keep the best

        if val < best_c5:

import time, math
import numpy as np

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _project_sum(h: np.ndarray, target: float) -> np.ndarray:
    """
    Greedy projection of h onto the set {0 ≤ h_i ≤ 1, Σ h_i = target}.
    The operation is performed in‑place and the array is returned.
    """
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        caps = 1.0 - h            # how much we can increase each entry
    else:
        caps = -h                 # how much we can decrease each entry
    for idx in np.where(caps > 1e-12)[0]:
        take = min(abs(diff), caps[idx])
        if diff < 0:
            take = -take
        h[idx] += take
        diff -= take
        if abs(diff) < 1e-12:
            break
    return h


def _thue_morse(n: int) -> np.ndarray:
    """Binary Thue‑Morse sequence of length n (0/1 as floats)."""
    return np.fromiter((i.bit_count() & 1 for i in range(n)), dtype=float, count=n)
