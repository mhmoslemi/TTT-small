def gradient_of_max(h):
    N = h.size
    dx = 2.0 / N
    # correlation
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (N - 1)

    h_plus = shift_array(h, shift)
    h_minus = shift_array(h, -shift)

import numpy as np, time, math

def _project(v, target):
   # clip and water fill
   v = np.clip(v, 0.0, 1.0)
   diff = target - v.sum()
   # loop up to 1000
   for _ in range(1000):
       if abs(diff) < 1e-12:
           break
       if diff > 0:
           free = v < 1.0 - 1e-12
           nfree = free.sum()
           if nfree == 0:
               break
           add = diff / nfree
           v[free] = np.minimum(v[free] + add, 1.0)
       else:
           free = v > 1e-12
           nfree = free.sum()
           if nfree == 0:
               break
           sub = diff / nfree
           v[free] = np.maximum(v[free] + sub, 0.0)
       diff = target - v.sum()
   # After loop maybe fix rounding errors:
   v = np.clip(v, 0.0, 1.0)
   # Adjust sum if small error left:
   cur_sum = v.sum()
   if abs(cur_sum - target) > 1e-9:
       # Distribute error uniformly among free positions
       error = target - cur_sum
       free = (v > 0) & (v < 1)
       nfree = free.sum()
       if nfree > 0:
           v[free] += error / nfree
           v = np.clip(v, 0.0, 1.0)
   return v

def _project(v, target):
    """Clip to [0,1] and adjust to meet the sum constraint."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = v < 1.0 - 1e-12
            nfree = free.sum()
            if nfree == 0:
                break
            add = diff / nfree
            v[free] = np.minimum(v[free] + add, 1.0)
        else:
            free = v > 1e-12
            nfree = free.sum()
            if nfree == 0:
                break
            sub = diff / nfree
            v[free] = np.maximum(v[free] + sub, 0.0)
        diff = target - v.sum()
    # tiny correction
    cur = v.sum()
    if abs(cur - target) > 1e-9:
        err = target - cur
        free = (v > 0) & (v < 1)
        nfree = free.sum()
        if nfree:
            v[free] += err / nfree
            v = np.clip(v, 0.0, 1.0)
    return v

def _c5_of(h):
    """Maximum overlap integral for a discrete step function."""
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / h.size)

def _shift_array(a, s):
    """Shift array by s with zero padding (no wrap)."""
    n = a.size
    if s >= 0:
        out = np.empty(n, dtype=a.dtype)
        out[:n - s] = a[s:]
        out[n - s:] = 0.0
    else:
        sh = -s
        out = np.empty(n, dtype=a.dtype)
        out[sh:] = a[:n - sh]
        out[:sh] = 0.0
    return out

def _gradient_of_max(h):
    """Gradient (sub‑gradient) of the currently maximal overlap term."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)
    hp = _shift_array(h, shift)          # h_{i+shift}
    hm = _shift_array(h, -shift)         # h_{i-shift}
    grad = dx * ((1.0 - hp) - hm)         # dx * (1 - h_{i+shift} - h_{i-shift})
    return grad, shift, idx, corr

def _deterministic_descent(h, target, max_iter=2000, step_factor=0.5):
    """Iteratively move mass from a positive‑gradient site to a negative‑gradient site."""
    cur_val = _c5_of(h)
    for _ in range(max_iter):
        grad, _, _, _ = _gradient_of_max(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if grad[i] <= 1e-12:
            break                     # no profitable direction
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 1e-12:
            break
        delta = max_delta * step_factor
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_val = _c5_of(new_h)
        if new_val < cur_val - 1e-12:
            h = new_h
            cur_val = new_val
        else:
            break
    return h, cur_val

def _perturb(h, target, swaps=5):
    """Randomly exchange small amounts of mass between pairs of indices."""
    n = h.size
    nh = h.copy()
    for _ in range(swaps):
        i, j = np.random.randint(n, size=2)
        lower = max(-nh[i], nh[j] - 1.0)
        upper = min(1.0 - nh[i], nh[j])
        if lower >= upper:
            continue
        delta = np.random.uniform(lower, upper)
        nh[i] += delta
        nh[j] -= delta
    return _project(nh, target)

        except Exception:

    return (best_h.tolist(), float(final_val), int(N))
