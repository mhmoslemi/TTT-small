def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return float(np.correlate(h, 1 - h, mode='full').max() * dx)

def project_sum(h, target):
    # clip to [0,1], then adjust diff greedily.
    h = np.clip(h,0,1)
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # If diff > 0, we need to increase some entries upward.
    # Create order: iterate over indices; could use random order to distribute.
    # Use np.argsort of random weights to randomize.
    order = np.arange(len(h))
    rng.shuffle(order)
    for idx in order:
        if diff <= 0:
            break
        cap = 1.0 - h[idx]
        inc = min(cap, diff)
        h[idx] += inc
        diff -= inc
    # If still diff negative (decrease):
    if diff < -1e-12:
        # diff negative, need to decrease entries.
        order = np.arange(len(h))
        rng.shuffle(order)
        for idx in order:
            if diff >= 0:
                break
            cap = -h[idx]   # can decrease up to -h
            dec = max(cap, diff)  # diff is negative, cap also negative
            h[idx] += dec   # dec is negative
            diff -= dec

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a discrete step vector."""
    n = h.shape[0]
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode="full").max() * dx)

def _project_sum(h: np.ndarray, target: float) -> np.ndarray:
    """
    Adjust h (still respecting 0‑1 bounds) so that sum(h) = target.
    Simple greedy redistribution – sufficient for our modest dimensions.
    """
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        for i in range(len(h)):
            if diff <= 0:
                break
            inc = min(1.0 - h[i], diff)
            if inc > 0:
                h[i] += inc
                diff -= inc
    else:  # need to decrease total mass
        for i in range(len(h)):
            if diff >= 0:
                break
            dec = max(-h[i], diff)   # both negative; dec is the larger (less negative)
            if dec < 0:
                h[i] += dec
                diff -= dec
    return h
