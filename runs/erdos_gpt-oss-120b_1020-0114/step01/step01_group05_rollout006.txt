def _subgrad_max(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    k = np.argmax(corr) - (n - 1) # shift index (negative to positive)
    # compute gradient
    g = np.empty_like(h)
    # using vectorized operations
    # term1: 1 - h shifted left by -k (i.e., h shifted by +k). Actually term1 = 1 - h_{j+k} if j+k within range else 1.
    # Let's compute shifted arrays with padding.
    if k >= 0:
        # term1: 1 - h[j+k] . For j where j+k<n -> use 1 - h[j+k]; else 1
        term1 = np.ones_like(h)
        term1[:n-k] = 1 - h[k:]  # positions j where j+k<n
        # term2 = h[j-k] for j >= k else 0
        term2 = np.zeros_like(h)
        term2[k:] = h[:n-k]
    else:
        # k negative: shift opposite direction
        k_abs = -k
        term1 = np.ones_like(h)
        term1[k_abs:] = 1 - h[:n-k_abs]
        term2 = np.zeros_like(h)
        term2[:n-k_abs] = h[k_abs:]
    g = dx * (term1 - term2)
    return g, corr.max()

import time
import numpy as np

def _project(v, target):
    """Project onto 0≤x≤1 and sum(x)=target."""
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def _evaluate(h):
    """Return C5 for a given h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _subgrad_max(h):
    """Sub‑gradient of the maximal overlap and its value."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k = np.argmax(corr) - (n - 1)          # shift producing the max
    # build the two terms of the gradient
    if k >= 0:
        term1 = np.empty(n)
        if n - k > 0:
            term1[: n - k] = 1.0 - h[k:]
        term1[n - k :] = 1.0
        term2 = np.empty(n)
        term2[:k] = 0.0
        if n - k > 0:
            term2[k:] = h[: n - k]
    else:
        p = -k
        term1 = np.empty(n)
        term1[p:] = 1.0 - h[: n - p]
        term1[:p] = 1.0
        term2 = np.empty(n)
        term2[: n - p] = h[p:]
        term2[n - p :] = 0.0
    g = dx * (term1 - term2)
    return g, np.max(corr) * dx
