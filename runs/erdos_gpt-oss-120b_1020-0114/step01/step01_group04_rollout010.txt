def subgradient_optimize(h0, time_limit):
    import time
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    best = h.copy()
    best_val = compute_c5(best)
    start = time.time()
    step = 0.2
    iteration = 0
    no_improve = 0
    while time.time() - start < time_limit:
        iteration += 1
        # worst shift
        conv = np.correlate(h, h, mode='full')
        min_prod = conv.min()
        idx = conv.argmin()
        shift = idx - (n - 1)
        # gradient
        w = np.zeros_like(h)
        s = shift
        if s >= 0:
            w[:n-s] += h[s:]
            w[s:] += h[:n-s]
        else:
            s = -shift
            w[s:] += h[:n-s]
            w[:n-s] += h[s:]
        # direction
        dir = w - w.mean()
        # if dir zero, break
        if np.allclose(dir, 0):
            break
        # line search
        alpha = step
        improved = False
        while alpha > 1e-6:
            v = h + alpha * dir
            h_new = project(v, target)
            new_val = compute_c5(h_new)
            if new_val < best_val - 1e-12:
                best = h_new.copy()
                best_val = new_val
                h = h_new
                improved = True
                break

                h = h_new

import time
import numpy as np

def _c5_from_h(h: np.ndarray) -> float:
    """C5 = 1 - (2/n) * min_{shift} sum_i h_i * h_{i+shift}."""
    n = h.shape[0]
    min_prod = np.correlate(h, h, mode="full").min()
    return 1.0 - (2.0 / n) * min_prod


def _project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto {0<=x<=1, sum x = target} by bisection on a shift."""
    lo, hi = -2.0, 2.0                     # wide enough to contain solution
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


def _worst_shift_and_conv(h: np.ndarray):
    """Return shift attaining minimal product and the full convolution."""
    n = h.shape[0]
    conv = np.correlate(h, h, mode="full")
    idx = conv.argmin()
    shift = idx - (n - 1)
    return shift, conv


def _gradient_w(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of product sum for a given shift: w_i = h_{i+shift}+h_{i-shift}."""
    n = h.shape[0]
    w = np.zeros_like(h)
    if shift >= 0:
        w[: n - shift] += h[shift:]
        w[shift:] += h[: n - shift]
    else:
        s = -shift
        w[s:] += h[: n - s]
        w[: n - s] += h[s:]
    return w


def _subgradient_optimize(h0: np.ndarray, time_limit: float):
    """Sub‑gradient ascent on min product (i.e. minimising C5)."""
    n = h0.shape[0]
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_val = _c5_from_h(best_h)

    step = 0.5
    no_improve = 0
    start = time.time()

    while time.time() - start < time_limit:
        shift, conv = _worst_shift_and_conv(h)
        w = _gradient_w(h, shift)
        direction = w - w.mean()
        if np.allclose(direction, 0.0):
            break

        cur_val = _c5_from_h(h)
        alpha = step
        improved = False
        for _ in range(12):                     # limited line‑search

def _w_vector(h: np.ndarray, shift: int):
    """Utility used by the greedy optimiser (same as previous code)."""
    n = h.shape[0]

def _worst_shift_and_corr(h: np.ndarray):

    shift = idx - (n - 1)

    else:
        h_start = np.random.rand(n)

    h_start = _project_to_sum(h_start, target)

    # --------------------------------------------------------------
    # 1️⃣ Sub‑gradient ascent (bulk of the budget)
    # --------------------------------------------------------------
    time_sub = budget_s * 0.70
    h_best, c5_best = _subgradient_optimize(h_start, time_sub)

    # --------------------------------------------------------------
    # 2️⃣ Greedy refinement with the remaining time
    # --------------------------------------------------------------
    time_left = budget_s - (time.time() - start_all)
    if time_left > 2.0:
        h_ref, c5_ref = _greedy_optimize(h_best, time_left)
        if c5_ref < c5_best:
            h_best, c5_best = h_ref, c5_ref

    # --------------------------------------------------------------
    # 3️⃣ Final sanity checks
    # --------------------------------------------------------------
    h_best = np.clip(h_best, 0.0, 1.0)
    diff = target - h_best.sum()
    if abs(diff) > 1e-12:
        free = np.where((h_best > 1e-12) & (h_best < 1.0 - 1e-12))[0]
        if free.size:
            h_best[free] += diff / free.size
        else:
            h_best[0] = np.clip(h_best[0] + diff, 0.0, 1.0)
        h_best = np.clip(h_best, 0.0, 1.0)

    final_c5 = _c5_from_h(h_best)
    return h_best, float(final_c5), int(n)
