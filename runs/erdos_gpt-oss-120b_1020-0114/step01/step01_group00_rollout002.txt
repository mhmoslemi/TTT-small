import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    """Maximum overlap integral for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    # full linear correlation of h with (1‑h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def corr_shift(h, s):
    """Overlap for a fixed integer shift s (no wrap‑around)."""
    n = h.shape[0]
    if s >= 0:
        return np.dot(h[: n - s], 1.0 - h[s:])
    else:
        s = -s
        return np.dot(h[s:], 1.0 - h[: n - s])

def random_feasible(n, rng):
    """Uniform random vector in [0,1]^n with exact sum=n/2."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)

    # simple projection to keep every coordinate inside [0,1]
    for _ in range(10):
        over = v > 1.0
        under = v < 0.0
        if not over.any() and not under.any():
            break
        excess = (v[over] - 1.0).sum()
        deficit = (-v[under]).sum()
        if excess > 0:
            interior = (~over) & (~under)
            if interior.any():
                v[interior] -= excess / interior.sum()
        if deficit > 0:
            interior = (~over) & (~under)
            if interior.any():
                v[interior] += deficit / interior.sum()
        v = np.clip(v, 0.0, 1.0)

    # final exact‑sum correction
    cur = v.sum()
    diff = n / 2.0 - cur
    if abs(diff) > 1e-10:
        mask = (v > 0.0) & (v < 1.0)
        if mask.any():
            v[mask] += diff / mask.sum()
            v = np.clip(v, 0.0, 1.0)
        else:
            v *= (n / 2.0) / cur
    return v

def optimize_h_slsqp(h_start, max_iter=600):
    """SLSQP minimise the auxiliary variable t ≥ overlap(s) for all shifts."""
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]               # h_i and t
    eq_cons = {"type": "eq", "fun": lambda x, n=n: np.sum(x[:n]) - n / 2.0}
    shifts = list(range(-(n - 1), n))
    ineq = []
    for s in shifts:
        def fun(x, s=s, n=n):
            h = x[:n]
            t = x[n]
            return t - corr_shift(h, s)
        ineq.append({"type": "ineq", "fun": fun})
    constraints = [eq_cons] + ineq

    t0 = compute_c5(h_start)
    x0 = np.concatenate([h_start, [t0]])

    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    try:
        res = opt.minimize(lambda x, n=n: x[n],
                           x0, method="SLSQP",
                           bounds=bounds,
                           constraints=constraints,
                           options=opts)
    except Exception:
        res = None

    if res is None or not res.success:
        h_opt = np.clip(h_start, 0.0, 1.0)
    else:
        h_opt = np.clip(res.x[:n], 0.0, 1.0)

    # re‑normalise the sum exactly
    target = n / 2.0
    cur = h_opt.sum()
    if abs(cur - target) > 1e-8:
        mask = (h_opt > 0.0) & (h_opt < 1.0)
        if mask.any():
            h_opt[mask] += (target - cur) / mask.sum()
            h_opt = np.clip(h_opt, 0.0, 1.0)
        else:
            h_opt *= target / cur

    return h_opt, compute_c5(h_opt)

def hill_climb_simple(h, time_budget, start_time):
    """Fast stochastic local improvement (pairwise mass swaps)."""
    n = h.shape[0]
    cur_h = h.copy()
    cur_val = compute_c5(cur_h)
    rng = np.random.default_rng()
    end = start_time + time_budget
    while time.time() < end:

            continue

        if cur_h[i] > 0.0 and cur_h[j] < 1.0:

        elif cur_h[j] > 0.0 and cur_h[i] < 1.0:
            max_delta = min(cur_h[j], 1.0 - cur_h[i])
            i, j = j, i
        else:

        cand[j] += delta

        h_ref, c5_ref = hill_climb_simple(best_h, remaining_total, time.time())
        if c5_ref < best_c5:
            best_h, best_c5 = h_ref, c5_ref
            best_n = best_h.size

    # --- enforce constraints exactly ----------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    cur = best_h.sum()
    if abs(cur - target) > 1e-10:
        diff = target - cur
        mask = (best_h > 0.0) & (best_h < 1.0)
        if mask.any():
            best_h[mask] += diff / mask.sum()
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            best_h *= target / cur

    return (best_h.tolist(), float(best_c5), int(best_n))
