            cur_val = new_val
            if cur_val < best_val:
                best = current.copy()
                best_val = cur_val
        # no acceptance of worse moves
    return best, best_val

            h[j] = 0.0

    """Clip to [0,1] and adjust linearly so that sum(h) = target (approximately)."""

    init_corr = np.correlate(h0, 1.0 - h0, mode="full") * _opt_dx
    t0 = float(np.max(init_corr) + 1e-8)

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    eq_cons = {"type": "eq", "fun": eq_constraint_fun}
    ineq_cons = NonlinearConstraint(ineq_constraint_fun,
                                    lb=np.zeros(2 * n - 1),
                                    ub=np.full(2 * n - 1, np.inf))

    x0 = np.concatenate([h0, [t0]])

    res = minimize(objective_fun,
                   x0,
                   method="SLSQP",
                   bounds=bounds,
                   constraints=[eq_cons, ineq_cons],
                   options={"maxiter": maxiter, "ftol": 1e-9, "disp": False})

    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    h_opt = project_to_feasible(h_opt, _opt_target_sum)

    try:
        c5_val = float(evaluate_erdos_solution(h_opt))
    except Exception:
        c5_val = float(res.x[-1])
    return h_opt, c5_val


def hill_climb_binary(n: int, max_swaps: int = 3000) -> tuple[np.ndarray, float]:
    """Greedy walk on binary vectors (exactly n/2 ones) trying to lower max overlap."""
    h = np.zeros(n)
    idx = _global_rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    cur_val = compute_C5(h)

    for _ in range(max_swaps):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i = _global_rng.choice(ones)
        j = _global_rng.choice(zeros)

        # propose swap
        h[i] = 0.0
        h[j] = 1.0
        new_val = compute_C5(h)

        if new_val < cur_val:
            cur_val = new_val
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
    return h, cur_val


# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) with a hopefully low C5."""
    start_time = time.time()
    global _global_rng
    _global_rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Helper to keep the best solution seen so far
    # ------------------------------------------------------------------
    best_h = None
    best_c5 = float("inf")
    best_n = None

    def consider(h, c5, n):
        nonlocal best_h, best_c5, best_n
        if c5 < best_c5 - 1e-12:
            best_h = h.copy()
            best_c5 = float(c5)
            best_n = n

    # ------------------------------------------------------------------
    # Starting from the supplied parent construction if it exists
    # ------------------------------------------------------------------
    try:
        init_vec = np.array(initial_h_values, dtype=float)  # provided by the harness
    except Exception:
        init_vec = np.full(100, 0.5)

    n0 = len(init_vec)
    init_vec = project_to_feasible(init_vec, n0 / 2.0)

    h_opt, c5_opt = optimise_start(init_vec, maxiter=2500)
    consider(h_opt, c5_opt, n0)

    # ------------------------------------------------------------------
    # Define a list of grid sizes to explore (even numbers, modest size)
    # ------------------------------------------------------------------
    grid_sizes = [80, 100, 120, 140, 160, 180]
    grid_sizes = [g for g in grid_sizes if g % 2 == 0]

    # ------------------------------------------------------------------
    # Main exploration loop
    # ------------------------------------------------------------------
    for n in grid_sizes:
        if time.time() - start_time > budget_s * 0.93:
            break

        candidates = []

        # 1) Interpolated best solution from a previous grid (if any)
        if best_h is not None and best_n is not None and best_n != n:
            x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h, left=0.0, right=0.0)
            h_interp = project_to_feasible(h_interp, n / 2.0)
            candidates.append(h_interp)

        # 2) Alternating binary pattern (exactly n/2 ones)
        alt = np.zeros(n)
        alt[::2] = 1.0
        candidates.append(alt)

        # 3) Few random binary patterns
        for _ in range(4):
            h_rand = np.zeros(n)
            idx = _global_rng.choice(n, n // 2, replace=False)
            h_rand[idx] = 1.0
            candidates.append(h_rand)

        # 4) Hill‑climbed binary patterns (a couple of independent runs)
        for _ in range(2):
            h_hc, _ = hill_climb_binary(n, max_swaps=4000)
            candidates.append(h_hc)

        # 5) Random continuous vectors (projected to the simplex)
        for _ in range(3):
            rnd = _global_rng.random(n)
            rnd = project_to_feasible(rnd, n / 2.0)
            candidates.append(rnd)

        # ------------------------------------------------------------------
        # Optimise each candidate with SLSQP
        # ------------------------------------------------------------------
        for h0 in candidates:
            if time.time() - start_time > budget_s * 0.97:
                break
            h0 = project_to_feasible(h0, n / 2.0)
            try:
                h_opt, c5_opt = optimise_start(h0, maxiter=2500)
            except Exception:
                continue
            consider(h_opt, c5_opt, n)

    # ----------------------------------------------------------------------
    # Fallback: uniform half‑filled vector if nothing was found (should not happen)
    # ----------------------------------------------------------------------
    if best_h is None:
        n_fallback = 100
        best_h = np.full(n_fallback, 0.5)
        best_c5 = compute_C5(best_h)
        best_n = n_fallback

    # ----------------------------------------------------------------------
    # Final evaluation via the provided helper (ensures consistency)
    # ----------------------------------------------------------------------
    try:
        final_c5 = float(evaluate_erdos_solution(best_h))
    except Exception:
        final_c5 = float(best_c5)

    return best_h, final_c5, best_n
