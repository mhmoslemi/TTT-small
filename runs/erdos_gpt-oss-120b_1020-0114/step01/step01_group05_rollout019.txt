import time
import numpy as np

def _evaluate(h):
    """Return max_k ∫ h(x)(1‑h(x+k)) dx for the discrete h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def _project(v, target):
    """Project onto {0≤x≤1, sum x = target}."""
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def _gradient_for_shifts(h, shifts):
    """Gradient of sum_{k in shifts} C(k) with respect to h."""
    n = h.shape[0]
    g = np.zeros_like(h)
    for k in shifts:
        if k == 0:
            g += 1 - 2 * h
        else:
            fwd = np.empty(n)
            fwd[: n - k] = h[k:]
            fwd[n - k :] = 0.0
            back = np.empty(n)
            back[k:] = h[: n - k]
            back[:k] = 0.0
            g += 1 - fwd - back
    return g

def _line_search(h, g, target, cur_val, alpha_start):
    """Back‑tracking line search using projection."""
    alpha = alpha_start
    for _ in range(12):
        h_new = _project(h - alpha * g, target)
        new_val = _evaluate(h_new)
        if new_val <= cur_val:
            return h_new, new_val, alpha
        alpha *= 0.5
    return h, cur_val, 0.0

def _random_feasible(n, target):
    """Generate a random feasible vector."""
    v = np.random.rand(n)
    return _project(v, target)

def _local_optimize(h_start, target, start_time, budget_s, global_best):
    """Projected sub‑gradient descent with simple line‑search."""
    h = h_start.copy()
    cur_val = _evaluate(h)
    best_h = h.copy()
    best_val = cur_val
    n = h.shape[0]
    dx = 2.0 / n
    step_alpha = 0.2
    no_improve = 0
    it = 0
    max_iters = 20000
    while it < max_iters:
        # time guard
        if time.time() - start_time > budget_s * 0.95:
            break
        it += 1

        # correlation and shifts achieving (near) maximum
        corr_full = np.correlate(h, 1 - h, mode='full') * dx
        max_c = corr_full.max()
        # absolute tolerance for “near‑max”
        tol = max(1e-6, 1e-8 * max_c)
        idxs = np.where(corr_full >= max_c - tol)[0]
        shifts = np.abs(idxs - (n - 1))
        shifts = np.unique(shifts)  # non‑negative, include 0

        # gradient for those shifts
        g = _gradient_for_shifts(h, shifts)

        # line search
        h_new, new_val, used_alpha = _line_search(h, g, target, cur_val, step_alpha)

        if used_alpha > 0.0:
            # accepted step
            h = h_new
            cur_val = new_val
            step_alpha = min(step_alpha * 1.03, 1.0)
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_h = h_new.copy()
                best_val = new_val
                if best_val <= 0.3805:      # early exit if very good
                    break
        else:
            # line search failed – shrink step size
            step_alpha *= 0.5
            no_improve += 1
            if no_improve > 1500:
                # random jitter to escape flat region
                i, j = np.random.randint(n, size=2)
                max_delta = min(1.0 - h[i], h[j])
                if max_delta > 0:
                    delta = max_delta * 0.1
                    h[i] += delta
                    h[j] -= delta
                    h = _project(h, target)
                no_improve = 0

        # optional early stop if we already beat the global best
        if best_val <= global_best - 1e-12:
            global_best = best_val

    return best_h, best_val

        # break if time is low
        if time.time() - start > budget_s * 0.98:
            break

    return best_h, best_val, n
