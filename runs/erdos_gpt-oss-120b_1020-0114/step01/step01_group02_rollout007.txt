def _repair_sum(h, target):
    # Guarantee sum exactly target, keep values in [0,1].
    # Use simple algorithm: compute diff, adjust proportionally where possible.
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # For small number steps loops
    # Use approach: while diff significant:
    #   if diff > 0: need to increase some entries
    #              choose indices where h < 1, increase up to diff
    #   else: decrease where h > 0.
    # We'll use simple vectorized method: fill capacity.
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        # random index
        i = np.random.randint(len(h))
        if diff > 0:
            cap = 1.0 - h[i]
            inc = min(cap, diff)
            h[i] += inc
            diff -= inc
        else:
            cap = h[i]
            dec = min(cap, -diff)
            h[i] -= dec
            diff += dec
    if abs(diff) > 1e-12:
        total = h.sum()
        if total > 0:
            h *= (target / total)
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # final tweak
            if diff > 0:
                idxs = np.where(h < 1.0)[0]

                    i = idxs[np.random.randint(len(idxs))]
                    h[i] = max(0.0, h[i] + diff)
    return h

import numpy as np, time, math, random

def _compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx  (discrete version)."""
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _repair_sum(h, target):
    """Adjust h in‑place so that sum(h)==target while keeping 0≤h≤1."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # greedy fill/empty
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = np.random.randint(len(h))
        if diff > 0.0:
            cap = 1.0 - h[i]
            if cap <= 0.0:
                continue
            inc = min(cap, diff)
            h[i] += inc
            diff -= inc
        else:
            cap = h[i]
            if cap <= 0.0:
                continue
            dec = min(cap, -diff)
            h[i] -= dec
            diff += dec
    # if still off, scale then clip and fix residual
    if abs(diff) > 1e-12:
        total = h.sum()
        if total > 0.0:
            h *= (target / total)
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0.0:
                idxs = np.where(h < 1.0)[0]

                    i = idxs[np.random.randint(idxs.size)]

                if idxs.size:
                    i = idxs[np.random.randint(idxs.size)]
                    h[i] = max(0.0, h[i] + diff)
    return h

def _gradient_move(h, k_shift, max_move):
    """Return (i,j,delta) to move mass from i to j based on gradient of worst shift."""
    N = len(h)
    # compute neighbours for shift k_shift
    left = np.zeros(N)
    right = np.zeros(N)
    if k_shift >= 0:
        if k_shift < N:
            right[:N - k_shift] = h[k_shift:]
            left[k_shift:] = h[:N - k_shift]
    else:
        ks = -k_shift
        if ks < N:
            right[ks:] = h[:N - ks]
            left[:N - ks] = h[ks:]
    g = 1.0 - (left + right)          # gradient of f_k (ignoring constant)
    src_idxs = np.where((g > 0.0) & (h > 1e-12))[0]
    tgt_idxs = np.where((g < 0.0) & (h < 1.0 - 1e-12))[0]
    if src_idxs.size == 0 or tgt_idxs.size == 0:
        return None, None, 0.0
    i = src_idxs[np.argmax(g[src_idxs])]
    j = tgt_idxs[np.argmin(g[tgt_idxs])]
    if i == j:
        return None, None, 0.0
    delta = min(max_move, h[i], 1.0 - h[j])
    return i, j, delta
