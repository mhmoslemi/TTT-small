def _grad_optimize(h0, time_limit):
    h = h0.copy()
    n = len(h)
    target_sum = n/2.0
    cur_c5 = _compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    start = time.time()
    iteration = 0
    fail_counter=0
    while time.time() - start < time_limit:
        # compute worst shift
        shift, _ = _worst_shift(h)
        # compute w and direction
        w = _w_vector(h, shift)
        d0 = w - 1.0
        # compute projection onto zero-sum hyperplane
        mean = d0.mean()
        d = d0 - mean
        # compute alpha max
        pos = d > 1e-12
        neg = d < -1e-12
        if not np.any(pos) and not np.any(neg):
            break # direction zero
        # compute bounds
        if pos.any():
            bound_pos = (1.0 - h[pos]) / d[pos]
        else:
            bound_pos = np.full(1, np.inf)
        if neg.any():
            bound_neg = h[neg] / (-d[neg])
        else:
            bound_neg = np.full(1, np.inf)
        alpha_max = min(bound_pos.min(), bound_neg.min())
        # step size: maybe fraction of alpha_max; we can adapt schedule:
        # Let step_factor maybe 0.5 initially, gradually decreasing.
        # We'll just use alpha = 0.5 * alpha_max
        alpha = 0.5 * alpha_max
        if alpha <= 0:
            # cannot move
            break
        new_h = h + alpha * d
        # ensure bounds due to numeric errors:
        new_h = np.clip(new_h, 0.0, 1.0)
        # compute new C5
        new_c5 = _compute_c5(new_h)
        # Accept if better
        if new_c5 < cur_c5 - 1e-12:
            h = new_h
            cur_c5 = new_c5
            fail_counter = 0
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = new_h.copy()
        else:
            # No improvement, maybe try random move
            fail_counter += 1
            if fail_counter > 10:
                # random perturbation
                i_up = np.random.randint(0, n)
                i_down = np.random.randint(0, n)
                while i_down == i_up:
                    i_down = np.random.randint(0, n)
                # compute max feasible delta
                max_inc = min(1.0 - h[i_up], h[i_down])
                if max_inc > 1e-12:
                    delta = np.random.rand() * max_inc
                    new_h2 = h.copy()
                    new_h2[i_up] += delta
                    new_h2[i_down] -= delta
                    new_c5_2 = _compute_c5(new_h2)
                    if new_c5_2 < cur_c5 - 1e-12:
                        h = new_h2
                        cur_c5 = new_c5_2
                        fail_counter = 0
                        if new_c5_2 < best_c5 - 1e-12:
                            best_c5 = new_c5_2
                            best_h = new_h2.copy()
                fail_counter = 0
        iteration += 1

    # final projection onto sum equality to fix any drift (should be exact)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        free = np.where((best_h > 1e-12) & (best_h < 1.0 - 1e-12))[0]
        if free.size:
            best_h[free] += diff / free.size
        else:
            # fallback: adjust first element
            best_h[0] = np.clip(best_h[0] + diff, 0, 1)
        best_h = np.clip(best_h, 0, 1)
        # slight adjustment for sum
        # might iterate to distribute residual
        # but we assume close
    final_c5 = _compute_c5(best_h)
    return best_h, final_c5

import time, math
import numpy as np

# -------------------------------------------------------------
# basic utilities
# -------------------------------------------------------------
def _compute_c5(h: np.ndarray) -> float:
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx (discretized)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)


def _worst_shift(h: np.ndarray):
    """Return (shift, full correlation array) for the maximal overlap."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)
    return shift, corr


def _w_vector(h: np.ndarray, shift: int):
    """w_i = h_{i+shift}+h_{i-shift} (zero outside)."""
    n = h.shape[0]
    if shift == 0:
        return 2.0 * h
    w = np.zeros_like(h)
    if shift > 0:
        w[: n - shift] = h[shift:]
        w[shift:] += h[: n - shift]
    else:  # shift < 0
        s = -shift
        w[s:] = h[: n - s]
        w[: n - s] += h[s:]
    return w


def _project_sum_bounds(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto 0≤x≤1 and sum(x)=target using bisection for the Lagrange multiplier."""
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - w.sum()
    if abs(diff) > 1e-10:
        free = np.where((w > 1e-12) & (w < 1.0 - 1e-12))[0]
        if free.size:
            w[free] += diff / free.size
        else:
            w[0] = np.clip(w[0] + diff, 0.0, 1.0)
        w = np.clip(w, 0.0, 1.0)
    return w

            new_h = _project_sum_bounds(new_h, target)

        new_c5 = _compute_c5(new_h)
        if new_c5 < cur_c5 - 1e-12:
            h, cur_c5 = new_h, new_c5
            if new_c5 < best_c5 - 1e-12:
                best_c5, best_h = new_c5, new_h.copy()

    # final projection (tiny numerical drift)
    best_h = _project_sum_bounds(best_h, target)
    return best_h, _compute_c5(best_h)


# -------------------------------------------------------------
# simple greedy local search (kept from the reference solution)
# -------------------------------------------------------------
def _greedy_optimize(h: np.ndarray, time_limit: float) -> (np.ndarray, float):
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_val = _compute_c5(best_h)

    cur_h = h.copy()
    cur_val = best_val
    step = 0.05
    no_improve = 0
    start = time.time()

    while time.time() - start < time_limit:
        shift, _ = _worst_shift(cur_h)
        w = _w_vector(cur_h, shift)

        if np.random.rand() < 0.7:
            up_mask = cur_h < 1.0 - 1e-12
            down_mask = cur_h > 1e-12
            if not up_mask.any() or not down_mask.any():
                break
            up_idx = np.where(up_mask)[0]
            down_idx = np.where(down_mask)[0]
            i_up = up_idx[np.argmax(w[up_idx])]
            i_down = down_idx[np.argmin(w[down_idx])]
            delta = min(step, 1.0 - cur_h[i_up], cur_h[i_down])
        else:
            i_up = np.random.randint(0, n)
            i_down = np.random.randint(0, n - 1)
            if i_down >= i_up:
                i_down += 1
            max_inc = min(1.0 - cur_h[i_up], cur_h[i_down])
            max_dec = min(cur_h[i_up], 1.0 - cur_h[i_down])
            if max_inc > 0 and (max_dec == 0 or np.random.rand() < 0.5):
                delta = np.random.rand() * max_inc
            elif max_dec > 0:
                delta = -np.random.rand() * max_dec
                i_up, i_down = i_down, i_up
            else:
                continue
        if delta <= 0:
            continue

        new_h = cur_h.copy()
        new_h[i_up] += delta
        new_h[i_down] -= delta
        new_val = _compute_c5(new_h)

        if new_val < cur_val - 1e-12:
            cur_h, cur_val = new_h, new_val
            no_improve = 0
            step = min(step * 1.05, 0.1)
            if cur_val < best_val:
                best_val, best_h = cur_val, cur_h.copy()
        else:
            T = step * 0.5
            if T > 0 and np.random.rand() < math.exp(-(new_val - cur_val) / T):
                cur_h, cur_val = new_h, new_val
            no_improve += 1
            if no_improve > 400:
                step *= 0.5
                no_improve = 0
                if step < 1e-6:
                    break

    # tiny correction to satisfy the integral exactly
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        free = np.where((best_h > 1e-12) & (best_h < 1.0 - 1e-12))[0]
        if free.size:
            best_h[free] += diff / free.size
        else:
            best_h[0] = np.clip(best_h[0] + diff, 0.0, 1.0)
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, _compute_c5(best_h)


# -------------------------------------------------------------
# main entry point
# -------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimise a step‑function h on [0,2] for the Erdős C₅ constant.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_all = time.time()

    # ----- initial vector -------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = h0.shape[0]
        target0 = n0 / 2.0
        if abs(h0.sum() - target0) > 1e-12:
            h0 = _project_sum_bounds(h0, target0)
    else:
        n0 = 45
        h0 = np.full(n0, 0.5)          # already satisfies the integral

    # ----- phase 1 : optimisation at the original resolution -------------
    time_phase1 = budget_s * 0.5
    h_best, c5_best = _grad_optimize(h0, time_phase1)

    # ----- phase 2 : up‑sample and refine if time permits ---------------
    elapsed = time.time() - start_all
    remaining = budget_s - elapsed
    if remaining > 5.0:
        # double the resolution (capped to keep evaluation cheap)
        n1 = min(180, n0 * 2)
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x1 = np.linspace(0.0, 2.0, n1, endpoint=False)
        h1 = np.interp(x1, x0, h_best)
        h1 = _project_sum_bounds(h1, n1 / 2.0)
        h_refined, c5_refined = _grad_optimize(h1, remaining)
        if c5_refined < c5_best:
            h_best, c5_best = h_refined, c5_refined
            n0 = n1

    # ----- phase 3 : short greedy polishing --------------------------------
    elapsed = time.time() - start_all
    remaining = budget_s - elapsed
    if remaining > 2.0:
        h_greedy, c5_greedy = _greedy_optimize(h_best, remaining)
        if c5_greedy < c5_best:
            h_best, c5_best = h_greedy, c5_greedy

    # ----- final safety projection -----------------------------------------
    h_best = np.clip(h_best, 0.0, 1.0)
    target = len(h_best) / 2.0
    h_best = _project_sum_bounds(h_best, target)
    final_c5 = _compute_c5(h_best)

    return h_best, float(final_c5), int(len(h_best))
