def project_onto_box_and_sum(v, target):
    # Clips to [0,1] with shift lambda
    # Find lambda such that sum(np.clip(v - lam,0,1)) = target
    lo = np.min(v) - 1.0  # ensures v - lo >= 1 for all i -> sum = n
    hi = np.max(v) + 1.0  # ensures v - hi <= -1 for all i -> sum = 0
    for _ in range(50):
        lam = (lo + hi) / 2
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v - (lo+hi)/2, 0.0, 1.0)
    # Might be slight deviation due to rounding, but we can correct:
    diff = target - proj.sum()
    if abs(diff) > 1e-9:
        # Adjust free variables (not at bounds) proportionally
        free = np.where((proj > 0) & (proj < 1))[0]
        if free.size > 0:
            proj[free] += diff / free.size
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj

def subgradient_descent(h0, max_iter, time_budget, start_time):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_val = float('inf')
    step0 = 1.0
    for it in range(max_iter):
        # time check
        if time.time() - start_time > time_budget:
            break
        # compute correlation
        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_idx = np.argmax(corr)
        val = corr[max_idx]
        if val < best_val - 1e-10:
            best_val = float(val)
            best_h = h.copy()
        # compute shift
        k = max_idx - (n - 1)
        # subgradient
        g = gradient_shift(h, k) * dx
        # step size schedule
        step = step0 / np.sqrt(it + 1)
        # gradient descent step
        new_h = h - step * g
        # project
        target_sum = n / 2.0
        h = project_onto_box_and_sum(new_h, target_sum)
    return best_h, best_val

def refine_slsqp(h_start):

    dx = 2.0 / n
    target_sum = n / 2.0

    # Ensure feasible start (project to bounds and sum)
    h0 = project_onto_box_and_sum(h_start, target_sum)

    init_corr = np.correlate(h0, 1 - h0, mode='full') * dx
    t0 = np.max(init_corr) + 1e-8

    bounds = [(0.0,1.0)]*n + [(0.0,1.0)]

    def eq_fun(x):
        return np.sum(x[:n]) - target_sum

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1-h, mode='full') * dx
        return t - corr

    eq_cons = {'type':'eq','fun':eq_fun}
    ineq_cons = NonlinearConstraint(ineq_fun, lb=np.zeros(2*n-1), ub=np.full(2*n-1, np.inf))

        c5_val = float(res.x[-1])

def project_onto_box_and_sum(v, target):
    # find lambda via bisection to satisfy sum_{i} clip(v_i - lambda,0,1) = target
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    # Ensure monotonic: as lambda increase sum decreases
    for _ in range(60):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) / 2.0
    proj = np.clip(v - lam, 0.0, 1.0)
    # fix small residual
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        free = np.where((proj > 0) & (proj < 1))[0]
        if free.size > 0:
            proj[free] += diff / free.size
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj

def gradient_shift(h, k):
    n = h.shape[0]
    g = np.zeros_like(h)
    if k >= 0:
        nk = n - k

def subgradient_descent(h0, max_iter, time_budget, start_time):
    n = h0.shape[0]
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_val = float('inf')
    step0 = 5.0
    for it in range(max_iter):
        # time check
        if time.time() - start_time > time_budget:
            break
        corr = np.correlate(h, 1 - h, mode='full') * dx
        idx = np.argmax(corr)
        val = corr[idx]
        if val < best_val:
            best_val = float(val)
            best_h = h.copy()
        # compute shift
        k = idx - (n - 1)
        # gradient of active shift
        g = gradient_shift(h, k) * dx
        # step size
        step = step0 / np.sqrt(it + 1)  # or / (it+1)
        new_h = h - step * g
        # project
        h = project_onto_box_and_sum(new_h, target_sum)
    # final evaluation
    # compute final correlation value for best_h (or last h)
    final_corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_val = float(np.max(final_corr))
    return best_h, final_val
```

def refine_slsqp(h_start):
    n = h_start.shape[0]
    dx = 2.0 / n
    target_sum = n / 2.0
    # make feasible
    h0 = project_onto_box_and_sum(h_start, target_sum)
    init_corr = np.correlate(h0, 1 - h0, mode='full') * dx
    t0 = float(np.max(init_corr) + 1e-8)

    bounds = [(0.0,1.0)]*n + [(0.0,1.0)]

    def eq_fun(x):
        return np.sum(x[:n]) - target_sum

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return t - corr

    eq_cons = {'type':'eq', 'fun':eq_fun}
    # number of lags = 2*n-1
    m = 2*n - 1
    ineq_cons = NonlinearConstraint(ineq_fun, lb=np.zeros(m), ub=np.full(m, np.inf))

    x0 = np.concatenate([h0, [t0]])

    res = minimize(objective_fun, x0, method='SLSQP', bounds=bounds,
                   constraints=[eq_cons, ineq_cons],
                   options={'maxiter': 400, 'ftol': 1e-9, 'disp': False})

    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    h_opt = project_onto_box_and_sum(h_opt, target_sum)

    try:
        c5_val = float(evaluate_erdos_solution(h_opt))
    except Exception:
        c5_val = float(res.x[-1])
    return h_opt, c5_val

import time
import numpy as np
from scipy.optimize import minimize, NonlinearConstraint

def objective_fun(x):
    """Objective for SLSQP: minimize the auxiliary variable t."""
    return x[-1]

def project_onto_box_and_sum(v, target):
    """Project vector v onto the box [0,1] and the hyperplane sum(v)=target."""
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    proj = np.clip(v - lam, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        free = np.where((proj > 0.0) & (proj < 1.0))[0]
        if free.size > 0:
            proj[free] += diff / free.size
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj

def gradient_shift(h, k):
    """Gradient of the correlation for a fixed shift k (zero‑padded)."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if k >= 0:
        nk = n - k
        g[:nk] += 1.0 - h[k:]
        g[k:] -= h[:nk]
    else:
        kp = -k
        nk = n - kp
        g[kp:] += 1.0 - h[:nk]
        g[:nk] -= h[kp:]
    return g

def subgradient_descent(h0, max_iter, start_time, deadline):
    """Projected sub‑gradient descent on max‑overlap objective."""
    n = h0.shape[0]
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_val = float('inf')
    step0 = 5.0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        idx = int(np.argmax(corr))
        val = corr[idx]
        if val < best_val:
            best_val = float(val)
            best_h = h.copy()
        k = idx - (n - 1)                 # shift corresponding to the max
        g = gradient_shift(h, k) * dx
        step = step0 / np.sqrt(it + 1)
        h = project_onto_box_and_sum(h - step * g, target)
    final_corr = np.correlate(best_h, 1.0 - best_h, mode='full') * dx
    final_val = float(np.max(final_corr))
    return best_h, final_val

def refine_slsqp(h_start):
    """Local SLSQP refinement of a feasible vector."""
    n = h_start.shape[0]
    dx = 2.0 / n
    target = n / 2.0
    h0 = project_onto_box_and_sum(h_start, target)
    init_corr = np.correlate(h0, 1.0 - h0, mode='full') * dx
    t0 = float(np.max(init_corr) + 1e-8)

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_fun(x):
        return np.sum(x[:n]) - target

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        return t - corr

    eq_cons = {'type': 'eq', 'fun': eq_fun}
    m = 2 * n - 1
    ineq_cons = NonlinearConstraint(ineq_fun, lb=np.zeros(m), ub=np.full(m, np.inf))

    x0 = np.concatenate([h0, [t0]])

    res = minimize(
        objective_fun,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=[eq_cons, ineq_cons],
        options={'maxiter': 400, 'ftol': 1e-9, 'disp': False},
    )

    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    h_opt = project_onto_box_and_sum(h_opt, target)

    try:
        c5_val = float(evaluate_erdos_solution(h_opt))
    except Exception:
        c5_val = float(res.x[-1])
    return h_opt, c5_val
