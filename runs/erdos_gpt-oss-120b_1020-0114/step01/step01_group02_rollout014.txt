    # Compute conv1 = convolution w * h (linear).
    L = 1 << ((len(w) + N - 1).bit_length())

    start = N-1
    end = start + N
    grad = conv1[start:end] + conv2[start:end]
    return grad

import numpy as np
import time, math, random

def _autocorr(h):
    """Linear autocorrelation a_k = sum_i h_i h_{i+k} for all shifts."""
    N = len(h)
    L = 1 << ((2 * N - 1).bit_length())          # power‑of‑2 ≥ 2N‑1
    H = np.fft.rfft(h, n=L)
    corr = np.fft.irfft(H * np.conj(H), n=L)
    return corr[: 2 * N - 1]                     # length 2N‑1, shifts –N+1…N‑1


def _project_sum_box(v, target):
    """Project v onto the set {0≤x≤1, sum(x)=target} (Euclidean projection)."""
    lo = -np.max(v) - 1.0     # low enough → all entries clipped to 1
    hi = np.max(v) + 1.0      # high enough → all entries clipped to 0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny adjustment for numerical drift
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(w < 1.0)[0]
            if idx.size:
                w[idx[0]] = min(1.0, w[idx[0]] + diff)
        else:
            idx = np.where(w > 0.0)[0]
            if idx.size:
                w[idx[0]] = max(0.0, w[idx[0]] + diff)
    return w


def _grad_from_weights(h, w):
    """Gradient of soft‑min  (weights w) w.r.t. h:  g = conv(w,h)+conv(w_rev,h)."""
    N = len(h)
    L = 1 << ((len(w) + N - 1).bit_length())
    H = np.fft.rfft(h, n=L)
    W = np.fft.rfft(w, n=L)
    conv1 = np.fft.irfft(H * W, n=L)
    w_rev = w[::-1]
    W_rev = np.fft.rfft(w_rev, n=L)
    conv2 = np.fft.irfft(H * W_rev, n=L)
    start = N - 1
    end = start + N
    return conv1[start:end] + conv2[start:end]


def _c5_from_h(h):
    """Exact C5 = max_x h(x)*(1-h(x+k)) dx using correlation with complement."""
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    np.random.seed(seed)
    random.seed(seed)
    start = time.time()

    # ---------- initialise ----------
    if "initial_h_values" in globals():

    h = np.clip(h, 0.0, 1.0)

    best_c5 = _c5_from_h(best_h)
