def hill_climb(h, steps, eps_range=(0.001,0.05)):
    for step in range(steps):
        # compute corr
        corr = np.correlate(h, 1-h, mode='full')
        # find max index
        max_idx = np.argmax(corr)
        # compute shift m (relative to zero shift)
        m = max_idx - (len(h)-1)
        # compute overlapping indices
        if m >= 0:
            i_range = np.arange(0, len(h)-m)
            pi = h[i_range] * h[i_range+m]
        else:
            m_abs = -m
            i_range = np.arange(m_abs, len(h))
            pi = h[i_range] * h[i_range-m_abs]
        # Choose i with high product
        if pi.size == 0:
            # no overlap? continue random.
            i = np.random.randint(len(h))
            j = np.random.randint(len(h))
        else:
            # pick index among top 10% high product
            thresh = np.percentile(pi, 90)
            candidates = np.where(pi >= thresh)[0]
            if candidates.size == 0:
                i = np.random.randint(len(h))
            else:
                # map to original index i in h
                i = i_range[np.random.choice(candidates)]
        # choose j random far away with small product
        # compute product for shift m at index j? Actually we need j such that its partner with shift m has low product.
        # Simpler: choose j random from where h[j] < 0.1 perhaps.
        j = np.random.randint(len(h))
        # ensure j not i and h[i] > eps, h[j] < 1-eps
        eps = np.random.uniform(*eps_range)
        eps = min(eps, h[i], 1.0 - h[j])
        if eps <= 0:
            continue
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_corr = np.correlate(new_h, 1-new_h, mode='full')
        new_c5 = np.max(new_corr) * dx
        if new_c5 < cur_c5:
            h = new_h
            cur_c5 = new_c5

import time
import numpy as np
import scipy.optimize as opt
from scipy.special import logsumexp

def compute_c5(h):
    """Exact C5 for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def upsample_h(old_h, new_n):
    """Linear interpolation to a finer grid while preserving total mass."""
    old_n = old_h.shape[0]
    if old_n == new_n:
        return old_h.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    target = new_n / 2.0
    cur = new_h.sum()
    if cur > 0:
        new_h *= target / cur
    new_h = np.clip(new_h, 0.0, 1.0)
    # final tiny correction
    cur = new_h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        new_h += diff / new_n
        new_h = np.clip(new_h, 0.0, 1.0)
        cur = new_h.sum()
        if cur > 0:
            new_h *= target / cur
    return new_h

def smooth_max_objective(h, alpha):
    """Smooth approximation of max_k ∫h(1−h_k) using log‑sum‑exp."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return (logsumexp(alpha * corr) / alpha) * (2.0 / len(h))

def optimize_slsqp(h0, n, alphas, restarts):
    """Run several SLSQP optimisations with a schedule of smoothing parameters."""
    target = n / 2.0
    best_val = np.inf
    best_h = None
    for _ in range(restarts):
        # random perturbation of the starting point
        h_start = h0 + np.random.uniform(-0.05, 0.05, size=n)
        h_start = np.clip(h_start, 0.0, 1.0)
        s = h_start.sum()
        if s != 0:
            h_start *= target / s
        # sequential passes with increasing α
        h_cur = h_start.copy()
        for alpha in alphas:
            cons = {"type": "eq", "fun": lambda h, t=target: np.sum(h) - t}
            bounds = [(0.0, 1.0)] * n
            res = opt.minimize(
                lambda h: smooth_max_objective(h, alpha),
                h_cur,
                method="SLSQP",
                bounds=bounds,
                constraints=[cons],
                options={"maxiter": 400, "ftol": 1e-9, "disp": False},
            )
            h_cur = np.clip(res.x, 0.0, 1.0)
            s = h_cur.sum()
            if abs(s - target) > 1e-8 and s != 0:
                h_cur *= target / s
                h_cur = np.clip(h_cur, 0.0, 1.0)
        val = compute_c5(h_cur)
        if val < best_val:
            best_val = val
            best_h = h_cur.copy()
    if best_h is None:
        best_h = h0.copy()
        best_val = compute_c5(best_h)
    return best_h, best_val

def simulated_annealing(h, time_limit):
    """Simple SA that moves mass between two points while keeping the total integral."""
    start = time.time()
    n = h.shape[0]
    target = n / 2.0
    cur_h = h.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    while True:
        if time.time() - start > time_limit:
            break
        elapsed = time.time() - start
        # linear cooling from 0.01 to ~0
        T = max(1e-9, 0.01 * (1.0 - elapsed / time_limit))
        i, j = np.random.randint(0, n, 2)
        if i == j:
            continue
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0.0:
            continue
        # larger steps early, smaller later
        eps = max_eps * np.random.rand() * (0.5 + 0.5 * (1.0 - elapsed / time_limit))
        if eps < 1e-9:
            continue
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_val = compute_c5(new_h)
        diff = new_val - cur_val
        if diff <= 0.0 or np.exp(-diff / max(T, 1e-12)) > np.random.rand():
            cur_h, cur_val = new_h, new_val
            if new_val < best_val:
                best_h, best_val = new_h.copy(), new_val
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function with a low C5 value."""
    np.random.seed(seed)
    start_total = time.time()
    # ------------------------------------------------------------------
    # initial construction
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n0 = h0.shape[0]
        target0 = n0 / 2.0
        s = h0.sum()
        if s > 0:
            h0 *= target0 / s
        h0 = np.clip(h0, 0.0, 1.0)
    else:
        n0 = 200
        dx = 2.0 / n0
        x = (np.arange(n0) + 0.5) * dx
        h0 = 0.5 * (1.0 + np.cos(np.pi * (x - 1.0)))  # cosine bump, zero at the ends
        target0 = n0 / 2.0
        s = h0.sum()
        h0 *= target0 / s
        h0 = np.clip(h0, 0.0, 1.0)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # ------------------------------------------------------------------
    # progressive refinement on larger grids
    # ------------------------------------------------------------------
    n_seq = [200, 400, 600]  # can be trimmed if time runs out
    for n in n_seq:
        if time.time() - start_total > budget_s * 0.95:
            break
        # upsample if needed
        if best_h.size != n:
            best_h = upsample_h(best_h, n)

        # ------------------------------------------------------------------
        # SLSQP stage (few restarts, increasing smoothing)
        # ------------------------------------------------------------------
        alphas = [5.0, 20.0, 80.0]       # from smooth to tight
        restarts = 5
        opt_h, opt_c5 = optimize_slsqp(best_h, n, alphas, restarts)
        if opt_c5 < best_c5:
            best_c5, best_h = opt_c5, opt_h.copy()

        # ------------------------------------------------------------------
        # Simulated annealing stage
        # ------------------------------------------------------------------
        remaining = budget_s - (time.time() - start_total)
        # allocate a modest slice for SA, at most 30 s
        sa_time = min(remaining * 0.5, 30.0)
        if sa_time > 5.0:
            sa_h, sa_c5 = simulated_annealing(best_h, sa_time)
            if sa_c5 < best_c5:
                best_c5, best_h = sa_c5, sa_h.copy()

        # ------------------------------------------------------------------
        # Simple random hill‑climbing refinements
        # ------------------------------------------------------------------
        steps = 5000
        cand_h = best_h.copy()
        cand_c5 = best_c5
        for _ in range(steps):
            if time.time() - start_total > budget_s * 0.95:
                break
            i, j = np.random.randint(0, n, 2)
            if i == j:
                continue
            eps = np.random.uniform(0.001, 0.05)
            eps = min(eps, cand_h[i], 1.0 - cand_h[j])
            if eps <= 0.0:
                continue
            new_h = cand_h.copy()
            new_h[i] -= eps
            new_h[j] += eps
            new_val = compute_c5(new_h)
            if new_val < cand_c5:
                cand_h, cand_c5 = new_h, new_val
        if cand_c5 < best_c5:
            best_c5, best_h = cand_c5, cand_h.copy()

    # ------------------------------------------------------------------
    # final feasibility enforcement
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_h.size / 2.0
    s = best_h.sum()
    if s != 0 and abs(s - target) > 1e-9:
        best_h *= target / s
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)
    return (best_h, float(final_c5), int(best_h.size))
