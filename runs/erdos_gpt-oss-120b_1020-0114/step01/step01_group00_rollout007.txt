def optimize_h(h_start, max_iter=2000):
    n=h_start.shape[0]
    dx=2.0/n

    bounds=[(0.0,1.0)]*n+[(0.0,None)]

    # equality constraint
    def eq_fun(x):
        return np.sum(x[:n]) - n/2.0

    eq_cons={'type':'eq','fun':eq_fun}

    # inequality constraints: t >= corr_shift for each shift s
    # we need an efficient function that computes shift correlation similarly to compute_c5 but without dx scaling.
    # We'll define a function for given s returns sum_i h_i*(1 - h_{i+s}) using slices.

    ineq_cons=[]
    for s in range(-(n-1), n):
        def con_fun(x, s=s):
            h=x[:n]
            t=x[n]
            if s>=0:
                a = h[:n-s]
                b = h[s:]
            else:
                ss=-s
                a = h[ss:]
                b = h[:n-ss]
            # compute sum_i a_i * (1 - b_i)
            # using vectorized: np.dot(a, 1.0 - b)
            val = t - np.dot(a, 1.0 - b)
            return val
        ineq_cons.append({'type':'ineq','fun':con_fun})
    constraints=[eq_cons]+ineq_cons

    # starting t as current C5 (dx*max) but we need t to be max of unscaled sum; unit: original sum (without dx). For constraints we use sum_i h_i*(1-h_{i+s}), which equals C(s)/dx.

    # compute initial t: max across s of sum_i h_i*(1-h_{i+s}) = compute_c5(h_start)/dx
    dx=2.0/n
    t0 = compute_c5(h_start)/dx

    x0=np.concatenate([h_start, [t0]])

    # options
    opts={'maxiter':max_iter, 'ftol':1e-12, 'disp':False}
    try:
        res=opt.minimize(lambda x: x[n], x0, method='SLSQP', bounds=bounds, constraints=constraints, options=opts)
    except Exception:
        res=None
    if res is None or not res.success:
        # fallback use start
        h_opt=np.clip(h_start,0.0,1.0)
        c5_val=compute_c5(h_opt)
        return h_opt,c5_val
    h_opt=np.clip(res.x[:n],0.0,1.0)
    # renormalize sum to exactly n/2
    target=n/2.0
    cur=h_opt.sum()
    if abs(cur-target)>1e-8:
        diff=target-cur
        mask=(h_opt>0.0)&(h_opt<1.0)
        if mask.any():
            h_opt[mask]+=diff/mask.sum()
            h_opt=np.clip(h_opt,0.0,1.0)
        else:
            h_opt*=target/cur
    c5_val=compute_c5(h_opt)
    return h_opt,c5_val

def hill_climb(h, budget, start_time):
    n=h.shape[0]
    best_h=h.copy()
    best_val=compute_c5(best_h)
    rng=np.random.default_rng()
    while time.time() - start_time < budget:
        # pick i,j distinct
        i,j = rng.integers(0, n, size=2)
        if i==j:
            continue
        # directional: want move mass from higher to lower
        # we can transfer delta from i to j ensuring bounds
        if best_h[i] > 0.0 and best_h[j] < 1.0:
            max_delta = min(best_h[i], 1.0 - best_h[j])
        elif best_h[j] > 0.0 and best_h[i] < 1.0:
            max_delta = min(best_h[j], 1.0 - best_h[i])
            i, j = j, i
        else:
            continue
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random()
        cand = best_h.copy()
        cand[i] -= delta
        cand[j] += delta
        val = compute_c5(cand)
        if val < best_val - 1e-12:
            best_h = cand
            best_val = val
    return best_h, best_val

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    """Maximum overlap for a step function h (numpy array)."""
    h = np.asarray(h, dtype=float)
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def random_binary(n, rng):
    """Even‑length binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=int)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1
    return h

def thue_morse_binary(n):
    """Thue–Morse sequence (balanced) for n = 2^k."""
    # parity of bit count of the index
    idx = np.arange(n, dtype=np.uint32)
    # count bits using numpy's vectorised popcount trick
    # (works for 32‑bit integers)
    bits = ((idx & 0x55555555) + ((idx >> 1) & 0x55555555))
    bits = ((bits & 0x33333333) + ((bits >> 2) & 0x33333333))
    bits = ((bits + (bits >> 4)) & 0x0F0F0F0F)
    bits = (bits * 0x01010101) >> 24
    parity = bits & 1
    return parity.astype(int)

def binary_sa_optimize(start_h, time_budget, global_start):
    """Simulated‑annealing on a binary vector with fixed number of ones."""
    n = start_h.shape[0]
    current = (start_h > 0).astype(int)
    curr_c5 = compute_c5(current)
    best = current.copy()
    best_c5 = curr_c5
    rng = np.random.default_rng()
    sa_start = time.time()
    T0 = 0.02  # initial temperature

    while True:
        now = time.time()
        elapsed = now - sa_start
        if elapsed >= time_budget:
            break

        # choose a 1 and a 0 to swap
        ones = np.flatnonzero(current)
        zeros = np.flatnonzero(1 - current)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        cand = current.copy()
        cand[i] = 0
        cand[j] = 1
        cand_c5 = compute_c5(cand)
        delta = cand_c5 - curr_c5

        T = T0 * (1.0 - elapsed / time_budget)  # linear cooling
        accept = False
        if delta <= 0:
            accept = True
        elif T > 0:
            if rng.random() < np.exp(-delta / T):
                accept = True

        if accept:
            current = cand
            curr_c5 = cand_c5
            if curr_c5 < best_c5:
                best = current.copy()
                best_c5 = curr_c5
                if best_c5 <= 0.3808:   # early stop if already very good
                    break
    return best, best_c5

def optimize_h(h_start, max_iter=2000):
    """SLSQP refinement of a (possibly fractional) step function."""
    n = h_start.shape[0]
    dx = 2.0 / n

    # bounds for each h_i and for the slack variable t
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    # equality: total mass = n/2
    def eq_fun(x):
        return np.sum(x[:n]) - n / 2.0
    eq_cons = {"type": "eq", "fun": eq_fun}

    # inequality: for every shift s we need t >= Σ h_i (1 - h_{i+s})
    ineq = []
    for s in range(-(n - 1), n):
        def con_fun(x, s=s):
            h = x[:n]
            t = x[n]
            if s >= 0:
                a = h[: n - s]
                b = h[s:]
            else:
                ss = -s
                a = h[ss:]
                b = h[: n - ss]
            return t - np.dot(a, 1.0 - b)
        ineq.append({"type": "ineq", "fun": con_fun})

    constraints = [eq_cons] + ineq

    # initial t = current max overlap (without dx)
    t0 = compute_c5(h_start) / dx
    x0 = np.concatenate([h_start, [t0]])

    opts = {"maxiter": max_iter, "ftol": 1e-12, "disp": False}
    try:
        res = opt.minimize(
            lambda x: x[n],
            x0,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options=opts,
        )
    except Exception:
        res = None

    if res is None or not res.success:
        h_opt = np.clip(h_start, 0.0, 1.0)
        return h_opt, compute_c5(h_opt)

    h_opt = np.clip(res.x[:n], 0.0, 1.0)

    # exact renormalisation of the integral
    target = n / 2.0
    cur = h_opt.sum()
    if abs(cur - target) > 1e-8:
        diff = target - cur
        mask = (h_opt > 0.0) & (h_opt < 1.0)
        if mask.any():
            h_opt[mask] += diff / mask.sum()
            h_opt = np.clip(h_opt, 0.0, 1.0)
        else:
            h_opt *= target / cur

    return h_opt, compute_c5(h_opt)

def hill_climb(h, budget, start_time):
    """Simple greedy pairwise mass moves (continuous values)."""
    n = h.shape[0]
    best_h = h.copy()
    best_val = compute_c5(best_h)
    rng = np.random.default_rng()
    while time.time() - start_time < budget:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # determine feasible direction
        if best_h[i] > 0.0 and best_h[j] < 1.0:
            max_delta = min(best_h[i], 1.0 - best_h[j])
        elif best_h[j] > 0.0 and best_h[i] < 1.0:
            max_delta = min(best_h[j], 1.0 - best_h[i])
            i, j = j, i
        else:
            continue
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random()
        cand = best_h.copy()
        cand[i] -= delta
        cand[j] += delta
        val = compute_c5(cand)
        if val < best_val - 1e-12:
            best_h = cand
            best_val = val
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h with a low Erdős C5 overlap."""
    start = time.time()
    rng = np.random.default_rng(seed)

    # choose the problem size
    if "initial_h_values" in globals():
        n = len(initial_h_values)
    else:
        n = 256          # a convenient even power of two

    # Initial candidates (binary)
    # ----------------------------------------------------------------------------------
    candidates = []

    while True:
