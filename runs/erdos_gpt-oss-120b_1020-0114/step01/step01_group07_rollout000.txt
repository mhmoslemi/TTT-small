import numpy as np, time, math

def _project(v, target):
    v = np.clip(v,0.0,1.0)
    # Use iterative water-filling loop
    diff = target - v.sum()
    while abs(diff) > 1e-12:
        if diff > 0:
            free = v < 1.0
            nfree = free.sum()
            if nfree == 0:
                break
            add = diff / nfree
            v[free] = np.minimum(v[free] + add, 1.0)
        else:
            free = v > 0.0
            nfree = free.sum()
            if nfree == 0:
                break
            sub = diff / nfree
            v[free] = np.maximum(v[free] + sub, 0.0)
        diff = target - v.sum()
    return v

def _c5_of(h):
    N = h.size

import numpy as np
import time
import math

def _project(v, target):
    """Project v onto [0,1] and enforce sum(v)=target."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    # water‑filling loop
    while abs(diff) > 1e-12:
        if diff > 0:
            free = v < 1.0
            nfree = free.sum()
            if nfree == 0:
                break
            add = diff / nfree
            v[free] = np.minimum(v[free] + add, 1.0)
        else:
            free = v > 0.0
            nfree = free.sum()
            if nfree == 0:
                break
            sub = diff / nfree
            v[free] = np.maximum(v[free] + sub, 0.0)
        diff = target - v.sum()
    return v


def _c5_of(h):
    """Compute C₅ for a step vector h."""
    N = h.size
    dx = 2.0 / N
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _max_shift_info(h):
    """
    Return information about the shift that attains the maximal overlap:
    (shift, i_max, j, max_product) where j = i_max+shift.
    """
    N = h.size
    # full correlation (without dx)
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (N - 1)          # integer shift in index units

    if shift >= 0:
        i_range = np.arange(N - shift)
        prod = h[i_range] * (1.0 - h[i_range + shift])
    else:
        i_range = np.arange(-shift, N)
        prod = h[i_range] * (1.0 - h[i_range + shift])
    i_max = i_range[int(np.argmax(prod))]
    j = i_max + shift
    return shift, i_max, j, prod.max()
