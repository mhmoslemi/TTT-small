    s = h.sum()

def normalize_sum(h, target_sum):

    h = np.clip(h, 0.0, 1.0)

    s = h.sum()

    if s > 0:

        h = h * (target_sum / s)

    h = np.clip(h, 0.0, 1.0)

    # final adjust

    s = h.sum()

    diff = target_sum - s

    if abs(diff) > 1e-12:

        # distribute evenly

        h += diff / len(h)

        h = np.clip(h, 0.0, 1.0)

        s = h.sum()

        if s > 0:

            h = h * (target_sum / s)

    return h

import time
import numpy as np
import scipy.optimize as opt
from scipy.special import logsumexp

def compute_c5(h):
    """Exact C₅ value for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)


def normalize_sum(h, target):
    """Clip to [0,1] and scale so that sum(h) == target (within tolerance)."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 0:
        h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    # final tiny fix
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-12:
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if s > 0:
            h = h * (target / s)
            h = np.clip(h, 0.0, 1.0)
    return h


def upsample_to_n(old_h, new_n):
    """Linear interpolation of a step function to a finer grid."""
    old_n = old_h.shape[0]
    if old_n == new_n:
        return old_h.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, old_h)


def smooth_max_corr(h, alpha):
    """Log‑sum‑exp smooth approximation of max_k ∫h(x)(1‑h(x+k))."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return logsumexp(alpha * corr) / alpha


def objective_penalized(h, alpha, target_sum, penalty):
    """Smoothed max‑overlap + quadratic penalty for the integral constraint."""
    return smooth_max_corr(h, alpha) + penalty * (np.sum(h) - target_sum) ** 2


def hill_climb(best_h, best_c5, time_limit, n_target, target_sum):
    """Simple (optionally simulated‑annealing) hill‑climbing on C₅."""
    start = time.time()
    steps = 0
    while time.time() - start < time_limit:
        i, j = np.random.randint(0, n_target, size=2)
        if i == j:
            continue
        # amount we can move from i to j
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0.0:
            continue
        # take a random fraction (up to 30%) of the allowable amount
        delta = np.random.uniform(0.0, max_delta * 0.3)
        if delta <= 0.0:
            continue

        new_h = best_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_c5 = compute_c5(new_h)

        # accept only improvements (deterministic descent)
        if new_c5 < best_c5:
            best_h, best_c5 = new_h, new_c5

        steps += 1

        # occasional global jitter to escape shallow locals
        if steps % 10000 == 0:
            pert = best_h + np.random.uniform(-0.02, 0.02, size=n_target)
            pert = np.clip(pert, 0.0, 1.0)
            pert = normalize_sum(pert, target_sum)
            pert_c5 = compute_c5(pert)
            if pert_c5 < best_c5:
                best_h, best_c5 = pert, pert_c5

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    def elapsed():
        return time.time() - start

    # ----------------------------------------------------------------------
    # Starting construction
    # ----------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        # fallback: half the points are 1, half 0, shuffled
        n0 = 200
        h0 = np.zeros(n0, dtype=float)
        h0[: n0 // 2] = 1.0
        np.random.shuffle(h0)

    # Target resolution – a modest increase gives extra freedom
    n_target = 400  # keep comfortably below 1 k
    h = upsample_to_n(h0, n_target)
    target_sum = n_target / 2.0
    h = normalize_sum(h, target_sum)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # ----------------------------------------------------------------------
    # Continuation of smoothed optimisation (L‑BFGS‑B + penalty)
    # ----------------------------------------------------------------------
    alphas = [10.0, 50.0, 200.0]          # increasing sharpness
    penalty = 1e5                         # force the integral constraint
    bounds = [(0.0, 1.0)] * n_target

    for alpha in alphas:
        if elapsed() > budget_s * 0.80:
            break
        for _ in range(6):                # a handful of restarts per alpha
            if elapsed() > budget_s * 0.80:
                break
            guess = best_h + np.random.uniform(-0.05, 0.05, size=n_target)
            guess = np.clip(guess, 0.0, 1.0)
            guess = normalize_sum(guess, target_sum)

            # minimise the penalised smooth max‑overlap
            res = opt.minimize(
                lambda x: objective_penalized(x, alpha, target_sum, penalty),
                guess,
                method="L-BFGS-B",
                bounds=bounds,
                options={"maxiter": 400, "ftol": 1e-12, "disp": False},
            )
            h_opt = normalize_sum(res.x, target_sum)
            c5_val = compute_c5(h_opt)
            if c5_val < best_c5:
                best_c5, best_h = c5_val, h_opt.copy()

    # ----------------------------------------------------------------------
    # Heavy hill‑climbing phase (many random mass moves)
    # ----------------------------------------------------------------------
    remaining_time = max(0.0, budget_s - elapsed())
    # reserve a tiny margin for finalisation
    hc_time = remaining_time * 0.95
    best_h, best_c5 = hill_climb(best_h, best_c5, hc_time, n_target, target_sum)

    # ----------------------------------------------------------------------
    # Final normalisation and reporting
    # ----------------------------------------------------------------------
    best_h = normalize_sum(best_h, target_sum)
    final_c5 = compute_c5(best_h)

    return (best_h, float(final_c5), int(n_target))
