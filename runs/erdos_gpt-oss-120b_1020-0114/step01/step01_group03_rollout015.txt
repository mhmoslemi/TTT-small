def min_overlap(h):
    n = h.size
    ft = np.fft.rfft(h)
    corr = np.fft.irfft(ft * np.conj(ft), n=n)
    # rounding to integer due to float errors
    corr = np.rint(corr).astype(np.int64)
    # corr[0] = number of ones
    min_ov = int(corr[1:].min())
    return min_ov, corr
```

    n = h.size

    # compute weights using log-sum-exp trick
    max_corr = np.max(alpha * corr)
    exps = np.exp(alpha * corr - max_corr)
    w = exps / exps.sum()
    # w corresponds to each lag index in range -(n-1)...(n-1)
    # compute gradient
    g = np.zeros_like(h)

            continue

def smooth_objective(h, alpha):
    corr = np.correlate(h, 1 - h, mode='full')
    # smooth max approx using log-sum-exp
    max_val = np.max(alpha * corr)
    exps = np.exp(alpha * corr - max_val)
    smooth_max = (max_val + np.log(exps.sum())) / alpha
    return smooth_max * (2.0 / len(h))

import scipy.optimize as opt

def compute_c5(h):
    """Exact C5 value for a step function using FFT based correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    H = np.fft.fft(h, L)
    G = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(H * np.conj(G)).real[: 2 * n - 1]
    return np.max(corr) * dx


def upsample_h(old_h, new_n):
    """Linear interpolation to a finer grid while preserving the integral."""
    old_n = old_h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)

    target_sum = new_n / 2.0
    cur_sum = new_h.sum()
    if cur_sum > 0:
        new_h = new_h * (target_sum / cur_sum)

    new_h = np.clip(new_h, 0.0, 1.0)

    # final correction for any remaining sum error
    cur_sum = new_h.sum()
    diff = target_sum - cur_sum
    if abs(diff) > 1e-12:
        new_h += diff / new_n
        new_h = np.clip(new_h, 0.0, 1.0)
        cur_sum = new_h.sum()
        if cur_sum > 0:
            new_h = new_h * (target_sum / cur_sum)
    return new_h


def smooth_objective(h, alpha):
    """Log‑sum‑exp smooth approximation of max_k ∫h(1−h(·+k))."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    maxval = np.max(alpha * corr)
    exps = np.exp(alpha * corr - maxval)
    smooth = (maxval + np.log(np.sum(exps))) / alpha
    return smooth * (2.0 / n)

        # (nothing else needed; best_h / best_c5 already updated)

    # ------------------------------------------------------------------
    # Final projection onto constraints (just to be safe)
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    target_sum = best_h.size / 2.0
    s = best_h.sum()
    if s != 0 and abs(s - target_sum) > 1e-9:
        best_h = best_h * (target_sum / s)
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)

    return (best_h, float(final_c5), int(best_h.size))
