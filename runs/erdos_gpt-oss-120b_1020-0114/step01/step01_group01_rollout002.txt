def correlation_and_grad(h):
    n = len(h)
    dx = 2.0 / n
    G = 1.0 - h
    c = np.correlate(h, G, mode='full') * dx
    idx = np.argmax(c)
    ell = idx - (n - 1)
    # compute gradient
    grad = np.zeros_like(h)
    # term1: (1 - h[j+ell]) if valid
    if ell >= 0:
        # shift positive: j+ell shift within range for j <= n-1-ell
        grad[:n-ell] = 1.0 - h[ell:]  # this corresponds to j from 0 to n-ell-1: j+ell in [ell,n-1]
        # term2: - h[j-ell] => j >= ell, j-ell index j-ell
        grad[ell:] -= h[:n-ell]  # j from ell to n-1

def project_to_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # identify free indices where 0 < h_i < 1
    free = np.where((h > 0) & (h < 1))[0]
    # If free empty, adjust a random index by diff (clipped)
    if free.size == 0:
        i = np.random.randint(len(h))
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
        return h
    # distribute diff equally
    while abs(diff) > 1e-12 and free.size > 0:
        delta = diff / free.size
        h[free] += delta
        # clip any that exceed bounds
        over = h > 1
        under = h < 0
        # adjust diff for newly clipped entries
        diff = target - h.sum()
        # update free set
        free = np.where((h > 0) & (h < 1))[0]
        if free.size == 0:
            # If no free left but still diff, adjust random index:
            i = np.random.randint(len(h))
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
            diff = target - h.sum()
    return h

def obj_fun(h):
    n = h.shape[0]
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(c)

def obj_jac(h):
    n = h.shape[0]
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    idx = np.argmax(c)
    ell = idx - (n - 1)
    grad = np.zeros_like(h)
    if ell >= 0:
        grad[:n-ell] = 1 - h[ell:]
        grad[ell:] -= h[:n-ell]
    else:
        k = -ell
        grad[k:] = 1 - h[:n-k]
        grad[:n-k] -= h[k:]
    grad *= dx
    return grad
```

def _project_to_feasible(vec, target):
    """Clip to [0,1] and adjust linearly so that sum(vec) = target (approximately)."""
    v = np.clip(vec, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) < 1e-12:
        return v
    # iterative redistribution among free entries
    for _ in range(20):
        free = np.where((v > 0.0) & (v < 1.0))[0]
        if free.size == 0:
            # no free variable – push a random index toward the needed direction
            i = np.random.randint(len(v))
            v[i] = np.clip(v[i] + diff, 0.0, 1.0)
            break

        v[free] += delta
        np.clip(v, 0.0, 1.0, out=v)
        diff = target - v.sum()
        if abs(diff) < 1e-12:
            break
    return v


def _obj_fun(h):
    """Maximum overlap value for the current step function."""
    n = h.shape[0]
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(c)


def _obj_jac(h):
    """Sub‑gradient of the maximum overlap (direction of increase)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    idx = np.argmax(corr)
    ell = idx - (n - 1)                     # shift that attains the max
    grad = np.zeros_like(h)

    if ell >= 0:
        # term (1 - h[j+ell])
        grad[: n - ell] = 1.0 - h[ell:]
        # term -h[j-ell]
        grad[ell:] -= h[: n - ell]
    else:
        k = -ell
        # term (1 - h[j- k])
        grad[k:] = 1.0 - h[: n - k]
        # term -h[j+ k]
        grad[: n - k] -= h[k:]

    grad *= dx
    return grad


def _interpolate(old_h, new_n):
    """Linear interpolation of a step function to a new grid size."""
    old_n = len(old_h)
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, old_h)


def _local_search(h, target, rng, steps=300, sigma=0.02):
    """Simple stochastic refinement that respects the constraints."""
    best_h = h.copy()
    best_val = float(evaluate_erdos_solution(best_h))
    n = len(h)

    for _ in range(steps):
        d = rng.normal(scale=sigma, size=n)
        d -= d.mean()                     # keep total sum zero
        cand = best_h + d
        cand = np.clip(cand, 0.0, 1.0)
        cand = _project_to_feasible(cand, target)
        val = float(evaluate_erdos_solution(cand))
        if val < best_val:
            best_val = val
            best_h = cand
    return best_h, best_val


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Starting point (provided by the harness if available)
    # ------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)  # type: ignore[name-defined]
        init_n = init_h.shape[0]
    except Exception:
        init_n = 100
        init_h = np.full(init_n, 0.5)

    best_h = init_h.copy()
    best_c5 = float(evaluate_erdos_solution(best_h))
    best_n = init_n

    def _update(best_candidate, val, n):
        nonlocal best_h, best_c5, best_n
        if val < best_c5 - 1e-9:
            best_h = best_candidate.copy()
            best_c5 = float(val)
            best_n = n

    # ------------------------------------------------------------------
    # Grid sizes to explore (coarse → finer)
    # ------------------------------------------------------------------
    grid_sizes = [init_n]
    for factor in (1.2, 1.5, 2.0):
        sz = int(round(init_n * factor))
        if 30 <= sz <= 500:
            grid_sizes.append(sz)
    grid_sizes = sorted(set(grid_sizes))

    # ------------------------------------------------------------------
    # Main optimisation loops
    # ------------------------------------------------------------------
    for n in grid_sizes:
        if time.time() - start_time > budget_s * 0.95:
            break

        target_sum = n / 2.0
        bounds = Bounds(0.0, 1.0)
        eq_con = LinearConstraint(np.ones((1, n)), [target_sum], [target_sum])

        # Build a small pool of initial vectors for this resolution
        inits = []

        # 1) Interpolated best known solution
        if n != best_n:
            interp = _interpolate(best_h, n)
            inits.append(interp)
        else:
            inits.append(best_h.copy())

        # 2) A few random feasible starts
        for _ in range(4):
            rand = rng.random(n)
            rand = _project_to_feasible(rand, target_sum)
            inits.append(rand)

        # ------------------------------------------------------------------
        # Optimise each start
        # ------------------------------------------------------------------
        for start_vec in inits:
            if time.time() - start_time > budget_s * 0.95:
                break

            start_vec = np.clip(start_vec, 0.0, 1.0)
            start_vec = _project_to_feasible(start_vec, target_sum)

            try:
                res = minimize(
                    _obj_fun,
                    start_vec,
                    method="trust-constr",
                    jac=_obj_jac,
                    bounds=bounds,
                    constraints=[eq_con],
                    options={"maxiter": 150, "verbose": 0},
                )
                h_opt = np.clip(res.x, 0.0, 1.0)
                h_opt = _project_to_feasible(h_opt, target_sum)
                c5_val = float(evaluate_erdos_solution(h_opt))
                _update(h_opt, c5_val, n)

                # Light stochastic refinement if time permits
                if time.time() - start_time < budget_s * 0.90:
                    refined_h, refined_val = _local_search(
                        h_opt, target_sum, rng, steps=300, sigma=0.02
                    )
                    _update(refined_h, refined_val, n)

            except Exception:
                # Safety fallback: evaluate the start vector itself
                c5_val = float(evaluate_erdos_solution(start_vec))
                _update(start_vec, c5_val, n)

    # ------------------------------------------------------------------
    # Ensure a feasible return even if nothing improved
    # ------------------------------------------------------------------
    if best_h is None or best_h.size == 0:
        best_n = init_n
        best_h = _project_to_feasible(np.full(best_n, 0.5), best_n / 2.0)
        best_c5 = float(evaluate_erdos_solution(best_h))

    return best_h, best_c5, best_n
