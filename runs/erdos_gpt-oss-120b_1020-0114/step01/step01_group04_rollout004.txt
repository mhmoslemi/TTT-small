def _smooth_min_gradient(h, beta):
    # Returns gradient of smooth min of autocorrelation.
    n = h.shape[0]
    # compute linear autocorrelation T
    T = np.correlate(h, h, mode='full')
    # compute weights p for shift costs: lower T -> higher weight (since we want to increase min)
    # p_s = exp(-beta * T_s) / sum exp(-beta * T_t)
    exp_arg = -beta * T
    # for numerical stability, subtract max
    max_arg = np.max(exp_arg)
    w = np.exp(exp_arg - max_arg)
    p = w / w.sum()
    # compute gradient
    grad = np.zeros_like(h)
    offset = n - 1
    for idx, pw in enumerate(p):
        if pw == 0.0:
            continue
        s = idx - offset
        if s == 0:
            grad += 2.0 * pw * h
        elif s > 0:
            grad[s:] += pw * h[:n-s]
            grad[:n-s] += pw * h[s:]
        else:
            t = -s
            grad[:n-t] += pw * h[t:]
            grad[t:] += pw * h[:n-t]
    return T, grad, p

import time, math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1-h(x+k))dx, discretized."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)


def _greedy_optimize(h: np.ndarray, time_limit: float):
    """Original greedy/annealing optimiser (kept for warm‑start)."""
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_val = _compute_c5(best_h)

    cur_h = h.copy()
    cur_val = best_val
    step = 0.05
    no_improve = 0
    start = time.time()

    while time.time() - start < time_limit:
        # worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)

        # w‑vector for this shift
        if shift == 0:
            w = 2.0 * cur_h
        else:
            w = np.zeros_like(cur_h)
            if shift > 0:
                w[: n - shift] += cur_h[shift:]
                w[shift:] += cur_h[: n - shift]
            else:
                s = -shift
                w[s:] += cur_h[: n - s]
                w[: n - s] += cur_h[s:]

        # pick up/down indices
        up_mask = cur_h < 1.0 - 1e-12
        down_mask = cur_h > 1e-12
        if not up_mask.any() or not down_mask.any():
            break
        up_idx = np.where(up_mask)[0]
        down_idx = np.where(down_mask)[0]
        i_up = up_idx[np.argmax(w[up_idx])]
        i_down = down_idx[np.argmin(w[down_idx])]
        delta = min(step, 1.0 - cur_h[i_up], cur_h[i_down])

        new_h = cur_h.copy()
        new_h[i_up] += delta
        new_h[i_down] -= delta
        new_val = _compute_c5(new_h)

        if new_val < cur_val - 1e-12:
            cur_h, cur_val = new_h, new_val
            no_improve = 0
            step = min(step * 1.05, 0.1)
            if cur_val < best_val:
                best_val, best_h = cur_val, cur_h.copy()
        else:
            T = step * 0.5
            if T > 0 and np.random.rand() < math.exp(-(new_val - cur_val) / T):
                cur_h, cur_val = new_h, new_val
            no_improve += 1
            if no_improve > 400:
                step *= 0.5
                no_improve = 0
                if step < 1e-6:
                    break

    # final projection onto hyperplane sum = target_sum
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:

            best_h[free] += diff / free.size
        else:
            best_h[0] = min(1.0, max(0.0, best_h[0] + diff))
        best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, _compute_c5(best_h)


def _smooth_min_gradient(h: np.ndarray, beta: float):
    """Gradient of smooth‑min of the autocorrelation T_s = Σ h_i h_{i-s}."""
    n = h.shape[0]
    # linear autocorrelation (no wrap)
    T = np.correlate(h, h, mode="full")          # length 2n‑1, index offset = n‑1
    # weights favouring small T (minimise)
    exp_arg = -beta * T
    max_arg = np.max(exp_arg)
    w = np.exp(exp_arg - max_arg)
    p = w / w.sum()

    grad = np.zeros_like(h)
    offset = n - 1
    for idx, pw in enumerate(p):
        if pw == 0.0:
            continue
        s = idx - offset
        if s == 0:
            grad += 2.0 * pw * h
        elif s > 0:
            grad[s:] += pw * h[:n - s]
            grad[:n - s] += pw * h[s:]
        else:  # s < 0
            t = -s
            grad[:n - t] += pw * h[t:]
            grad[t:] += pw * h[:n - t]
    return T, grad, p


def _project_constraints(h: np.ndarray):
    """Clip to [0,1] and enforce Σh = len(h)/2 exactly (simple water‑filling)."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size:
            h[free] += diff / free.size
            h = np.clip(h, 0.0, 1.0)
        else:
            h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h


def _gradient_optimize(h0: np.ndarray, time_limit: float):
    """Annealed gradient ascent on smooth‑min of the autocorrelation."""
    start = time.time()
    n = h0.shape[0]
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _compute_c5(best_h)

    # schedules
    beta_start, beta_end = 0.5, 35.0
    eta_start, eta_end = 0.04, 0.001

    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        beta = beta_start + (beta_end - beta_start) * frac
        eta = eta_start + (eta_end - eta_start) * frac

        _, grad, _ = _smooth_min_gradient(h, beta)

        # small stochastic perturbation (helps escape plateaux)
        if np.random.rand() < 0.015:
            sigma = 0.02 * (1 - frac)
            h_pert = h + sigma * (np.random.rand(n) - 0.5)
            h_pert = _project_constraints(h_pert)
            grad = _smooth_min_gradient(h_pert, beta)[1]  # recompute gradient briefly
            h = h_pert

        h_new = h + eta * grad
        h_new = _project_constraints(h_new)

        cur_c5 = _compute_c5(h_new)
        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            best_h = h_new.copy()
        h = h_new

    return best_h, best_c5


def _slsqp_optimize(h0: np.ndarray, time_limit: float):
    """Fine‑tune with SLSQP (non‑linear constrained optimiser)."""
    try:
        from scipy.optimize import minimize
    except Exception:
        return h0, _compute_c5(h0)

    n = h0.shape[0]
    target_sum = n / 2.0
    t0 = _compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    eq_cons = {'type': 'eq', 'fun': lambda x: np.sum(x[:n]) - target_sum}
    cons = [eq_cons]

    shifts = list(range(-(n - 1), n))

    def make_ineq(s):
        def fun(x):
            h = x[:n]
            t = x[-1]
            if s >= 0:
                val = np.dot(h[: n - s], 1.0 - h[s:])
            else:
                s2 = -s
                val = np.dot(h[s2:], 1.0 - h[: n - s2])
            return t - val
        return fun

    cons.extend({'type': 'ineq', 'fun': make_ineq(s)} for s in shifts)

    max_iter = max(10, int(min(300, time_limit * 0.25)))
    opts = {'maxiter': max_iter, 'ftol': 1e-12, 'disp': False}
    res = minimize(lambda x: x[-1], x0, method='SLSQP',
                   bounds=bounds, constraints=cons, options=opts)

    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    diff = target_sum - h_opt.sum()
    if abs(diff) > 1e-10:
        free = np.where((h_opt > 1e-12) & (h_opt < 1.0 - 1e-12))[0]
        if free.size:
            h_opt[free] += diff / free.size
            h_opt = np.clip(h_opt, 0.0, 1.0)
    c5 = _compute_c5(h_opt)
    return h_opt, c5
