def local_search(h, max_iters=5000, step=0.01, rng, dx):
    best_h = h.copy()
    best_val = evaluate_erdos_solution(best_h)
    n = len(h)
    target_sum = h.sum()  # should be constant
    for it in range(max_iters):
        # Random choose two indices i != j
        i, j = rng.choice(n, size=2, replace=False)
        # Choose delta such that we can move from i to j; delta positive if we want to increase h[j] and decrease h[i]; ensure bounds
        max_delta = min(step, best_h[i])  # can decrease i by up to its value
        max_delta = min(max_delta, 1.0 - best_h[j])  # can increase j up to 1
        if max_delta <= 0:

        delta = rng.uniform(0, max_delta)
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        val = evaluate_erdos_solution(h_new)
        if val < best_val - 1e-10:
            best_h = h_new
            best_val = val
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine initial vector
    try:
        init_vec = np.array(initial_h_values, dtype=float)
        n_init = init_vec.shape[0]
    except Exception:
        n_init = 100
        init_vec = np.full(n_init, 0.5)

    best_h = None
    best_c5 = float('inf')
    best_n = n_init

    # Helper to update best
    def update(h, c5, n):
        nonlocal best_h, best_c5, best_n
        if c5 < best_c5 - 1e-12:
            best_h = h.copy()
            best_c5 = float(c5)
            best_n = n

    # Optimize initial
    h_opt, val_opt = optimise_start(init_vec)
    update(h_opt, val_opt, n_init)

    # Random restarts
    attempts = 0
    max_attempts = 20
    while attempts < max_attempts and time.time() - start_time < budget_s * 0.7:
        attempts += 1
        rand_start = rng.random(n_init)
        rand_start = project_to_feasible(rand_start, n_init/2.0)
        h_opt, val_opt = optimise_start(rand_start)
        update(h_opt, val_opt, n_init)

    # Local refinement on best
    if best_h is not None and time.time() - start_time < budget_s * 0.9:
        steps = 2000
        delta = 0.02
        h_ref, val_ref = local_refine(best_h, steps=steps, max_delta=delta, rng=rng)
        # compute final using evaluator
        try:
            c5_val = float(evaluate_erdos_solution(h_ref))
        except Exception:
            c5_val = max_overlap(h_ref)
        update(h_ref, c5_val, best_n)

    # Grid refinement
    if time.time() - start_time < budget_s * 0.95:
        # Choose new grid size larger than best_n, maybe 1.5 times, and round to even.
        n2 = max(best_n + 20, int(best_n * 1.3))
        # Ensure not too large (limit maybe 200)
        n2 = min(n2, 200)
        # Interpolate best_h onto new grid
        x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h, left=0.0, right=0.0)
        # Project to feasible
        h_interp = project_to_feasible(h_interp, n2/2.0)
        h_opt2, val_opt2 = optimise_start(h_interp)
        update(h_opt2, val_opt2, n2)

        # local refinement on refined
        if time.time() - start_time < budget_s * 0.99:
            h_ref2, val_ref2 = local_refine(h_opt2, steps=3000, max_delta=0.02, rng=rng)
            try:
                c5_val = float(evaluate_erdos_solution(h_ref2))
            except Exception:
                c5_val = max_overlap(h_ref2)
            update(h_ref2, c5_val, n2)

    # Ensure best_h feasible (maybe reproject)
    if best_h is None:
        best_n = n_init
        best_h = project_to_feasible(np.full(best_n, 0.5), best_n/2.0)
        best_c5 = float(evaluate_erdos_solution(best_h))

    # Final sanity: clip h to [0,1] and project to sum
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_feasible(best_h, best_n/2.0)

        best_c5 = float(evaluate_erdos_solution(best_h))
    except Exception:
        best_c5 = max_overlap(best_h)

import time
import numpy as np
from scipy.optimize import minimize, NonlinearConstraint

# Global placeholders for optimisation (used in constraint functions)
_opt_n = None          # number of points
_opt_dx = None         # grid spacing 2 / n
_opt_target_sum = None # n / 2

def project_to_feasible(h, target):
    """Clip to [0,1] and adjust linearly so that sum(h) = target (approximately)."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == 0.0:
        # place unit mass arbitrarily
        idx = np.random.choice(len(h), int(round(target)), replace=False)
        h[idx] = 1.0
        return h
    scale = target / cur
    h = np.clip(h * scale, 0.0, 1.0)
    # iterative correction for any leftover error
    for _ in range(10):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            i = np.random.randint(len(h))
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
            continue
        delta = diff / free.size
        h[free] += delta
        np.clip(h, 0.0, 1.0, out=h)
    return h

def max_overlap(h):
    """Return max_k ∫ h(x)(1−h(x+k))dx approximated on the grid."""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full') * dx)

# ---------- constraint helpers for SLSQP ----------
def objective_fun(x):
    """Minimise the auxiliary variable t (last entry)."""
    return x[-1]

def eq_constraint_fun(x):
    """∑h = n/2."""
    return np.sum(x[:_opt_n]) - _opt_target_sum

def ineq_constraint_fun(x):
    """t >= overlap for every shift."""
    h = x[:_opt_n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode='full') * _opt_dx
    return t - corr  # must be ≥ 0 for all lags

def optimise_start(start_h):
    """Run SLSQP from start_h and return feasible h and its C5."""
    global _opt_n, _opt_dx, _opt_target_sum
    n = int(start_h.shape[0])
    _opt_n = n
    _opt_dx = 2.0 / n
    _opt_target_sum = n / 2.0

    h0 = project_to_feasible(start_h, _opt_target_sum)

    # initial feasible t a little above current max overlap
    init_corr = np.correlate(h0, 1.0 - h0, mode='full') * _opt_dx
    t0 = float(np.max(init_corr) + 1e-8)

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    eq_cons = {'type': 'eq', 'fun': eq_constraint_fun}

    m = 2 * n - 1
    ineq_cons = NonlinearConstraint(
        ineq_constraint_fun,
        lb=np.zeros(m),
        ub=np.full(m, np.inf),
    )

    x0 = np.concatenate([h0, [t0]])

    res = minimize(
        objective_fun,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=[eq_cons, ineq_cons],
        options={'maxiter': 1500, 'ftol': 1e-9, 'disp': False},
    )

    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    h_opt = project_to_feasible(h_opt, _opt_target_sum)

    try:
        c5_val = float(evaluate_erdos_solution(h_opt))
    except Exception:
        c5_val = max_overlap(h_opt)

    return h_opt, c5_val

def local_refine(h, steps=2000, max_delta=0.02, rng=None):
    """Greedy pairwise mass transfer to lower max overlap."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(h)
    best_h = h.copy()
    best_val = max_overlap(best_h)
    target = n / 2.0

    for _ in range(steps):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # randomly decide direction of transfer
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        # feasible amount we can move from src to dst
        max_move = min(best_h[src], 1.0 - best_h[dst], max_delta)
        if max_move <= 0.0:
            continue
        delta = rng.random() * max_move
        h_new = best_h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        val = max_overlap(h_new)
        if val < best_val - 1e-12:
            best_h = h_new
            best_val = val
    # ensure exact sum after refinement
    best_h = project_to_feasible(best_h, target)
    return best_h, best_val
