def corr_shift_gradient(h, s):
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        # term1: (1 - h[i+s]) for i in [0, n-1-s]
        g[:n-s] += 1 - h[s:]
        # term2: -h[i-s] for i in [s, n-1]
        g[s:] += -h[:n - s]
    else:
        # s negative: use s = -t, t>0
        t = -s
        # corr(s) = sum_{i=t}^{n-1} h[i] * (1 - h[i-t])
        # So first index i from t..n-1, second index i-t.
        # Derivative: for i>=t: derivative w.r.t h_i (first) -> (1 - h[i-t])
        # For i-t -> derivative w.r.t h_{i-t} -> -h_i
        g[t:] += 1 - h[:n - t]  # because i runs t..n-1, i-t runs 0..n-1-t
        g[:n - t] += -h[t:]     # derivative for second factor
    return g

    rng = np.random.default_rng()

        # compute corr

        max_c = corr.max()

        thresh = 1e-5
        inc_candidates = np.where(g < -thresh)[0]
        dec_candidates = np.where(g > thresh)[0]

            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:

                i_dec, j_inc = i, j

                i_dec, j_inc = j, i
            else:

            if max_delta <= 0:
                continue

            cand = best_h.copy()
            cand[i_dec] -= delta
            cand[j_inc] += delta
            cand_c5 = compute_c5(cand)
            if cand_c5 < best_c5 - 1e-12:
                best_h, best_c5 = cand, cand_c5

        K = 5
        inc_top = inc_candidates[np.argsort(g[inc_candidates])][:K]  # most negative
        dec_top = dec_candidates[np.argsort(-g[dec_candidates])][:K] # most positive

                best_h, best_c5 = cand, cand_c5
    return best_h, best_c5

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for discretised h."""
    n = h.shape[0]
    dx = 2.0 / n
    return float((np.correlate(h, 1.0 - h, mode="full") * dx).max())

def corr_shift_gradient(h, s):
    """Gradient of the correlation for a fixed integer shift s."""
    n = h.shape[0]
    g = np.zeros(n)
    if s >= 0:
        g[: n - s] += 1.0 - h[s:]
        g[s:] += -h[: n - s]
    else:
        t = -s
        g[t:] += 1.0 - h[: n - t]
        g[: n - t] += -h[t:]
    return g

def random_feasible(n, rng):
    """Uniform random feasible vector (sum=n/2, 0≤h_i≤1)."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    for _ in range(10):
        over = v > 1.0
        under = v < 0.0
        if not over.any() and not under.any():
            break
        excess = (v[over] - 1.0).sum()
        deficit = (-v[under]).sum()
        if excess > 0:
            interior = (~over) & (~under)
            if interior.any():
                v[interior] -= excess / interior.sum()
        if deficit > 0:
            interior = (~over) & (~under)
            if interior.any():
                v[interior] += deficit / interior.sum()
        v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - n / 2.0) > 1e-12:
        diff = n / 2.0 - cur
        mask = (v > 0.0) & (v < 1.0)
        if mask.any():
            v[mask] += diff / mask.sum()
            v = np.clip(v, 0.0, 1.0)
        else:
            v *= (n / 2.0) / cur
    return v

        i, j = rng.integers(0, n, size=2)
        if i == j:

        if best_h[i] > 0.0 and best_h[j] < 1.0:
            max_delta = min(best_h[i], 1.0 - best_h[j])
        elif best_h[j] > 0.0 and best_h[i] < 1.0:
            max_delta = min(best_h[j], 1.0 - best_h[i])

                approx = -(g[i] - g[j]) * max_delta
                if approx > best_approx:
                    best_approx = approx
                    best_pair = (i, j)
                    best_delta = max_delta
        if best_pair is None:
            continue
        i, j = best_pair
        # randomise step size a little
        delta = best_delta * (0.5 + 0.5 * rng.random())
        cand = best_h.copy()
        cand[i] -= delta
        cand[j] += delta
        val = compute_c5(cand)

        # accept improvements, otherwise occasional Metropolis move
        if val < best_c5 - 1e-12:
            best_h, best_c5 = cand, val
        else:
            elapsed = time.time() - start
            frac = max(0.0, 1.0 - elapsed / time_budget)
            temp = 1e-4 * frac          # tiny temperature
            if np.exp(-(val - best_c5) / max(temp, 1e-9)) > rng.random():
                best_h, best_c5 = cand, val
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    """Return a feasible step function, its C5 bound and the number of points."""
    start_global = time.time()
    rng = np.random.default_rng(seed)

    # obtain an initial feasible vector
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n = h0.size
    else:
        n = 180
        h0 = random_feasible(n, rng)

    # normalise sum exactly
    target = n / 2.0
    cur = h0.sum()
    if abs(cur - target) > 1e-12:
        h0 = h0 * (target / cur)
        h0 = np.clip(h0, 0.0, 1.0)

    best_h = None
    best_c5 = np.inf
    best_n = n

    # Main search loop – several restarts until time runs out or target reached
    while True:
        elapsed = time.time() - start_global
        remaining = budget_s - elapsed
        if remaining < 5.0:
            break

        # decide a start point for this restart
        if best_h is None or rng.random() < 0.4:
            h_start = h0.copy()
        else:
            # perturb the current best slightly to escape a possible plateau
            h_start = best_h.copy()
            # a tiny random redistribution
            i, j = rng.integers(0, n, size=2)
            if i != j:
                max_delta = min(h_start[i], 1.0 - h_start[j])
                if max_delta > 0:
                    delta = max_delta * rng.random() * 0.1
                    h_start[i] -= delta
                    h_start[j] += delta

        # allocate about half of the remaining time to a directed gradient phase
        phase_budget = min(remaining * 0.5, 30.0)
        h_opt, c5_opt = gradient_descent_h(h_start, phase_budget, rng)

        # use the rest for a simple hill‑climb refinement
        elapsed2 = time.time() - start_global
        remaining2 = budget_s - elapsed2
        if remaining2 > 1.0:
            h_opt, c5_opt = hill_climb(h_opt, remaining2 * 0.5, rng)

        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            best_n = n

        if best_c5 < 0.3808:
            break

    # final clean‑up – enforce the integral exactly
    cur = best_h.sum()
    diff = target - cur
    if abs(diff) > 1e-9:
        mask = (best_h > 0.0) & (best_h < 1.0)
        if mask.any():
            best_h[mask] += diff / mask.sum()
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            best_h = np.clip(best_h * (target / cur), 0.0, 1.0)

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), float(final_c5), int(best_n))
