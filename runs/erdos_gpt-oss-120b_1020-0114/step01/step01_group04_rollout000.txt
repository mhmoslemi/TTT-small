def _smooth_optimize(h0, time_limit):
    n = h0.shape[0]
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_val = _compute_c5(h)
    start = time.time()
    # schedule of alpha
    alphas = np.linspace(2.0, 30.0, 7)
    # initial step size
    lr = 0.5
    for alpha in alphas:
        # maybe schedule inner loop steps based on remaining time
        # we will loop while time left > small amount
        while True:
            now = time.time()

                break

            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
            # schedule lr decay
            lr *= 0.9995
            # maybe increase lr a bit each outer alpha

import time, math, numpy as np

def _c5(h: np.ndarray) -> float:
    n = h.shape[0]
    return np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n)


def _project(y: np.ndarray, target: float) -> np.ndarray:
    """Project y onto {0<=h<=1, sum h = target} using bisection on lambda."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        proj = np.clip(y - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)


def _smooth_optimize(h0: np.ndarray, time_limit: float):
    """Gradient descent on a log‑sum‑exp smoothing of the max overlap."""
    n = h0.shape[0]
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    best_h, best_val = h.copy(), _c5(h)
    start = time.time()
    alphas = np.linspace(2.0, 30.0, 7)
    lr = 1.0
    for a in alphas:
        for _ in range(1500):
            if time.time() - start > time_limit:
                return best_h, best_val
            C = np.correlate(h, 1.0 - h, mode="full") * dx
            m = C.max()
            w = np.exp(a * (C - m))
            w /= w.sum()
            conv = np.convolve(w, h, mode="full")
            weighted = conv[np.arange(n) + (n - 1)]
            grad = dx * (1.0 - 2.0 * weighted)
            y = h - lr * grad
            h = _project(y, target)
            cur = _c5(h)
            if cur < best_val:
                best_val, best_h = cur, h.copy()
            lr *= 0.9995
            if lr < 1e-4:
                lr = 1e-4
    return best_h, best_val


def _worst_shift(h: np.ndarray):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return idx - (n - 1), corr


def _w_vec(h: np.ndarray, s: int):
    n = h.shape[0]
    if s == 0:
        return 2.0 * h
    w = np.zeros_like(h)
    if s > 0:
        w[: n - s] += h[s:]
        w[s:] += h[: n - s]
    else:
        t = -s
        w[t:] += h[: n - t]
        w[: n - t] += h[t:]
    return w


def _greedy_optimize(h0: np.ndarray, time_limit: float):
    """Simple hill‑climbing using the worst shift."""
    n = h0.shape[0]
    target = n / 2.0
    h = h0.copy()
    best_h, best_val = h.copy(), _c5(h)
    step = 0.05
    no_imp = 0
    start = time.time()
    while time.time() - start < time_limit:
        s, _ = _worst_shift(h)
        w = _w_vec(h, s)
        up_mask = h < 1.0 - 1e-12
        down_mask = h > 1e-12
        if not up_mask.any() or not down_mask.any():
            break
        i_up = np.where(up_mask)[0][np.argmax(w[up_mask])]
        i_down = np.where(down_mask)[0][np.argmin(w[down_mask])]
        delta = min(step, 1.0 - h[i_up], h[i_down])
        if delta <= 0:
            break
        h_new = h.copy()
        h_new[i_up] += delta
        h_new[i_down] -= delta
        val = _c5(h_new)
        if val < best_val - 1e-12:
            h, best_h, best_val = h_new, h_new.copy(), val
            no_imp = 0
            step = min(step * 1.05, 0.12)
        else:
            # simulated‑annealing acceptance
            T = step * 0.5
            if np.random.rand() < math.exp(-(val - best_val) / max(T, 1e-12)):
                h = h_new
            no_imp += 1
            if no_imp > 300:
                step *= 0.5
                no_imp = 0
                if step < 1e-6:
                    break
    # final projection onto the hyperplane
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        best_h = _project(best_h, target)
    return best_h, _c5(best_h)


def _upscale(initial: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an initial vector to length n."""
    x0 = np.linspace(0.0, 2.0, initial.shape[0], endpoint=False)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x, x0, initial)
    target = n / 2.0
    return _project(h, target)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    np.random.seed(seed)
    start_all = time.time()

    # --- starting vector -------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n0 = 40
        h0 = np.zeros(n0)
        idx = np.random.choice(n0, n0 // 2, replace=False)
        h0[idx] = 1.0

    # --- choose resolution (larger gives more freedom) --------------------
    n = max(2 * h0.shape[0], 90)          # at least 90 points, even
    if n % 2:
        n += 1
    n = min(n, 140)                       # keep it modest

    # --- initial guess ----------------------------------------------------
    h_start = _upscale(h0, n)

    # --- smooth gradient descent -----------------------------------------
    time_rem = budget_s - (time.time() - start_all)
    if time_rem > 5.0:
        gd_time = time_rem * 0.6
        h_opt, c5_opt = _smooth_optimize(h_start, gd_time)
    else:
        h_opt, c5_opt = h_start, _c5(h_start)

    # --- final greedy refinement -----------------------------------------
    time_rem = budget_s - (time.time() - start_all)
    if time_rem > 2.0:
        h_opt, c5_opt = _greedy_optimize(h_opt, time_rem)

    # --- ensure exact mass ------------------------------------------------
    target = n / 2.0
    diff = target - h_opt.sum()
    if abs(diff) > 1e-12:
        h_opt = _project(h_opt, target)

    final_c5 = _c5(h_opt)
    return h_opt, float(final_c5), int(n)
