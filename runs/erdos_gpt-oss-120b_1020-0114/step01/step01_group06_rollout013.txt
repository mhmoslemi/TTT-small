        i_min = np.argmin(g)
        if i_max == i_min:

def _autocorr_min_shift(h):
   A = np.correlate(h, h, mode='full')
   idx_min = np.argmin(A)  # returning first min if tie.
   shift = idx_min - (len(h)-1)
   return shift, A

def _grad_for_shift(h, shift):
    n = len(h)
    s = abs(shift)
    if s == 0:
        return 2 * h
    g = np.zeros_like(h)
    g[:n-s] += h[s:]
    g[s:] += h[:n-s]
    return g

But we also have to handle shift=0 to avoid double counting incorrectly: For shift=0, we have g = 2*h, correct.

def _move_pair(h, i_max, i_min, delta):
    h2 = h.copy()

    h2[i_min] -= delta

    n = initial_h.shape[0]
    target_sum = n/2.0
    # Ensure feasible
    h = np.clip(initial_h, 0.0, 1.0)
    h = _project_sum(h, target_sum)
    best_h = h.copy()
    best_c5 = _c5_value(h)
    cur_h = h
    cur_c5 = best_c5
    step = 0.2
    max_iters = 2000000 # but break early.
    iter_cnt = 0
    while time.time() - start < time_limit:
        iter_cnt += 1
        # compute shift
        shift, _ = _autocorr_min_shift(cur_h)
        # gradient
        g = _grad_for_shift(cur_h, shift)
        # find extremes
        i_max = int(np.argmax(g))
        i_min = int(np.argmin(g))
        if i_max == i_min:
            # cannot improve; break
            break
        # compute feasible delta
        delta = min(step, 1.0 - cur_h[i_max], cur_h[i_min])
        if delta <= 1e-12:
            step *= 0.5
            if step < 1e-8:
                break
            continue
        # propose
        new_h = cur_h.copy()
        new_h[i_max] += delta
        new_h[i_min] -= delta
        # ensure bounds
        # Not needed due to delta constraints
        new_c5 = _c5_value(new_h)
        if new_c5 < cur_c5 - 1e-12:
            cur_h = new_h
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = new_h.copy()
            # increase step a little for momentum
            step = min(step * 1.02, 0.3)
        else:
            # reject; reduce step
            step *= 0.7
        # maybe adapt step gradually
        if step < 1e-8:
            break
        # optional: after certain fresh loops, random jitter
        if iter_cnt % 5000 == 0:
            # small random jitter but keep feasibility
            # choose random index pair and move small random amount
            i, j = np.random.randint(0, n, size=2)
            if i != j:
                # jitter magnitude small
                jitter = (np.random.rand() - 0.5) * 0.02
                # constrain
                low = max(-cur_h[i], -(1.0 - cur_h[j]))
                high = min(1.0 - cur_h[i], cur_h[j])
                jitter = np.clip(jitter, low, high)
                if abs(jitter) > 1e-12:
                    cur_h[i] += jitter
                    cur_h[j] -= jitter
                    cur_c5 = _c5_value(cur_h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = cur_h.copy()
    return best_h, best_c5

import time
import numpy as np

def _c5_value(h: np.ndarray) -> float:
    """C5 bound for a step vector h (using zero padding)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _project_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust h (still respecting 0‑1 bounds) so that sum(h)=target."""
    h = np.array(h, copy=True)
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        caps = 1.0 - h
        for i in np.where(caps > 1e-12)[0]:
            add = min(diff, caps[i])
            h[i] += add
            diff -= add
            if diff <= 1e-12:
                break
    else:
        caps = h
        for i in np.where(caps > 1e-12)[0]:
            sub = min(-diff, caps[i])
            h[i] -= sub
            diff += sub
            if diff >= -1e-12:
                break
    return h


def _max_corr_shift(h: np.ndarray):
    """Return shift that gives maximal overlap h·(1‑h)."""
    c = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(c))
    shift = idx - (len(h) - 1)
    return shift, c, idx


def _grad_max_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """
    Gradient direction d such that moving h in direction d (while keeping sum constant)
    decreases the current maximal overlap.
    d_j = h_{j‑shift} - (1‑h_{j+shift}) where the terms exist.
    """
    n = h.shape[0]
    d = np.zeros_like(h)
    if shift >= 0:
        s = shift
        # - (1 - h_{j+s}) for j = 0 .. n‑s‑1
        d[: n - s] -= (1.0 - h[s:])
        # + h_{j‑s} for j = s .. n‑1
        d[s:] += h[: n - s]
    else:
        p = -shift
        # + h_{j+p} for j = 0 .. n‑p‑1
        d[: n - p] += h[p:]
        # - (1 - h_{j‑p}) for j = p .. n‑1
        d[p:] -= (1.0 - h[: n - p])
    return d


def _optimize_start(h0: np.ndarray, time_limit: float, rng: np.random.Generator):
    """Local optimisation starting from h0, bounded by time_limit seconds."""
    start = time.time()
    n = h0.shape[0]
    target = n / 2.0

    h = np.clip(h0, 0.0, 1.0)
    h = _project_sum(h, target)

    best_h = h.copy()
    best_c5 = _c5_value(best_h)

    cur_h = h

        pos = int(np.argmax(d))
        neg = int(np.argmin(d))
