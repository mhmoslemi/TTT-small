def _project(v, target):
    # clip and adjust using water-filling
    v = np.clip(v, 0.0, 1.0)
    # compute excess
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    # If total < target, add to entries < 1
    # Use simple iterative loop
    diff = target - total
    # We'll use while loop with tolerance
    while abs(diff) > 1e-12:
        if diff > 0:
            free = v < 1.0 - 1e-12
            n = free.sum()
            if n == 0:
                break
            add = diff / n
            v[free] = np.minimum(v[free] + add, 1.0)
        else:
            free = v > 1e-12
            n = free.sum()
            if n == 0:
                break
            sub = diff / n
            v[free] = np.maximum(v[free] + sub, 0.0)
        diff = target - v.sum()
    return v

import numpy as np

def _project(v, target):
    """Project vector v onto [0,1] box and enforce sum(v)=target."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    # water-filling loop
    while abs(diff) > 1e-12:
        if diff > 0:
            free = v < 1.0 - 1e-12
            nfree = int(free.sum())
            if nfree == 0:
                break
            add = diff / nfree
            v[free] = np.minimum(v[free] + add, 1.0)
        else:
            free = v > 1e-12
            nfree = int(free.sum())
            if nfree == 0:
                break
            sub = diff / nfree
            v[free] = np.maximum(v[free] + sub, 0.0)
        diff = target - v.sum()
    return v


def _c5_of(h):
    """Return the overlap constant C₅ for a step vector h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)

        if new_val < cur_val or np.random.rand() < math.exp(-(new_val - cur_val) / T):
            cur_h, cur_val = new_h, new_val
            if new_val < best_val - 1e-12:
                best_h, best_val = new_h.copy(), new_val
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # occasional restart to escape possible stagnation
        if no_improve > 200_000:
            cur_h = best_h.copy()
            for _ in range(5):
                idx = np.random.randint(N)
                cur_h[idx] = np.clip(cur_h[idx] + np.random.uniform(-step, step), 0.0, 1.0)
            cur_h = _project(cur_h, target_sum)
            cur_val = _c5_of(cur_h)
            no_improve = 0

    # -------------------------------------------------------------
    # Final projection and evaluation (use provided helper if present)
    best_h = _project(best_h, target_sum)
    final_val = _c5_of(best_h)
    if "evaluate_erdos_solution" in globals():
        try:
            final_val = evaluate_erdos_solution(best_h.tolist())
        except Exception:
            pass

    return (best_h.tolist(), float(final_val), int(N))
