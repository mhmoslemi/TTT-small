    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    if abs(total - target) < 1e-12:
        return h
    # Use water-filling to enforce sum and caps.
    # We'll implement simple algorithm: 
    # Solve for lambda such that sum(clip(h - lambda,0,1)) = target.
    # Since domain is convex, we can binary search lambda.
    lo = -1.0
    hi = 1.0
    # Actually lambda may be positive or negative.
    for _ in range(50):
        lam = (lo + hi) / 2
        w = np.clip(h - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return w

def project_to_simplex_box(h, target):
    """Project vector h onto [0,1]^n intersect {sum = target}."""
    # Clip first
    w = np.clip(h, 0.0, 1.0)
    # If sum already target, return
    s = w.sum()
    if abs(s - target) < 1e-12:
        return w
    # Binary search for lambda
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        w = np.clip(h - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

import time
import numpy as np

def project_to_simplex_box(v, target):
    """Projection of v onto the intersection of the box [0,1]^n and the hyperplane sum=v = target."""
    # Clip to [0,1] first
    w = np.clip(v, 0.0, 1.0)
    s = w.sum()
    if abs(s - target) < 1e-12:
        return w
    # Binary search for lambda such that sum(clip(v - lambda,0,1)) = target
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        # when sum > target we need larger lambda (increase to reduce w)
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) / 2.0
    return np.clip(v - lam, 0.0, 1.0)

def compute_c5(h, dx):
    """Return the maximal overlap integral for step vector h on a grid with spacing dx."""
    return np.max(np.correlate(h, 1.0 - h, mode='full') * dx)

def improve_h(h0, rng, time_limit):
    """Hill‑climbing search that preserves the total mass.
    Returns the best vector seen and its C5 value."""
    n = h0.shape[0]
    dx = 2.0 / n
    best_h = h0.copy()
    best_score = compute_c5(best_h, dx)
    cur_h = h0.copy()
    cur_score = best_score
    start = time.time()
    # main loop
    while time.time() - start < time_limit:
        # correlation and current maximal shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full') * dx
        max_idx = int(np.argmax(corr))
        max_val = float(corr[max_idx])
        lag = max_idx - (n - 1)    # signed shift
        # indices i where i+lag is inside the array
        if lag >= 0:
            i_valid = np.arange(0, n - lag)
        else:
            i_valid = np.arange(-lag, n)
        # try to move mass from a source where h[i] is positive to a target where h[j] is not full
        mask = (cur_h[i_valid] > 1e-12) & (cur_h[i_valid + lag] < 1.0 - 1e-12)
        if mask.any():
            idx = rng.choice(np.where(mask)[0])
            i = int(i_valid[idx])
            j = i + lag
        else:
            src = np.where(cur_h > 1e-12)[0]
            tgt = np.where(cur_h < 1.0 - 1e-12)[0]
            if src.size == 0 or tgt.size == 0:
                break
            i = int(rng.choice(src))
            j = int(rng.choice(tgt))
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random()   # up to full transferable amount
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_score = compute_c5(new_h, dx)
        if new_score < max_val - 1e-9:
            cur_h = new_h
            cur_score = new_score
            if new_score < best_score - 1e-9:
                best_h = new_h.copy()
                best_score = new_score
        # otherwise reject
    return best_h, best_score

def interpolate_and_project(old_h, new_n):
    """Linear interpolation of old_h onto a finer grid and projection onto the feasible set."""
    old_n = old_h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, old_h, left=0.0, right=0.0)
    return project_to_simplex_box(h_new, new_n / 2.0)

def random_binary_search(n, rng, time_limit):
    """Search for a good binary (0/1) vector with exactly n/2 ones."""
    if n % 2 != 0:
        # fallback to continuous random start
        h = rng.random(n)
        return project_to_simplex_box(h, n/2.0), compute_c5(project_to_simplex_box(h, n/2.0), 2.0/n)
    best_h = None
    best_score = float('inf')
    dx = 2.0 / n
    start = time.time()
    half = n // 2
    while time.time() - start < time_limit:
        h = np.zeros(n)
        ones = rng.choice(n, half, replace=False)
        h[ones] = 1.0
        score = compute_c5(h, dx)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    if best_h is None:
        best_h = np.full(n, 0.5)
        best_score = compute_c5(best_h, dx)
    return best_h, best_score

import time

def project_to_simplex_box(v, target):
    """Projection of v onto the set {0≤x≤1, sum(x)=target}."""
    w = np.clip(v, 0.0, 1.0)
    if abs(w.sum() - target) < 1e-12:
        return w
    lo, hi = -1.0, 1.0            # lambda bounds: w = clip(v‑lambda,0,1)
    for _ in range(60):          # binary search for the correct lambda
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        if w.sum() > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) / 2.0
    return np.clip(v - lam, 0.0, 1.0)

    if remaining > 5.0:
        t_bin = remaining * 0.30
        h_rand, s_rand = random_binary_search(best_n, rng, t_bin)
        if s_rand < best_score - 1e-9:
            best_h, best_score = h_rand, s_rand

        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining > 5.0:
            h_imp, s_imp = improve_h(best_h, rng, remaining)
            if s_imp < best_score - 1e-9:
                best_h, best_score = h_imp, s_imp

    # ------------------------------------------------------------------
    # Ensure feasibility and compute the final official score
    # ------------------------------------------------------------------
    best_h = project_to_simplex_box(best_h, best_n / 2.0)
    try:
        final_score = float(evaluate_erdos_solution(best_h))
    except Exception:
        final_score = compute_c5(best_h, 2.0 / best_n)

    return best_h, final_score, best_n
