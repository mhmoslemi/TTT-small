def gradient_descent_step(h, alpha, lr):
    n = len(h)
    sum_h = h.sum()
    # autocorrelation linear
    a = np.convolve(h, h[::-1], mode='full')
    O = sum_h - a  # length 2n-1
    # compute weights
    # note: O could be large, multiply by alpha maybe cause overflow; subtract max
    O_max = O.max()
    w = np.exp(alpha * (O - O_max))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(w) / w.size
    else:
        w /= w_sum
    # convolution for gradient
    c2 = np.convolve(h, w, mode='full')[n-1:n-1+n]
    w_rev = w[::-1]

    g = np.ones_like(h) - (c1 + c2)
    # update
    y = h - lr * g
    return y

import time, math, numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper: evaluate C5 for a discrete step function
# ----------------------------------------------------------------------
def _c5_value(h: np.ndarray) -> float:
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

# ----------------------------------------------------------------------
# Projection onto the cube [0,1]^n while fixing the sum to target
# ----------------------------------------------------------------------
def _project_box_sum(y: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of y onto {0≤x≤1, sum(x)=target}."""
    lo = -np.max(y) - 1.0          # λ low enough => all entries hit upper bound 1
    hi = 1.0 - np.min(y) + 1.0     # λ high enough => all entries hit lower bound 0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(y - lam, 0.0, 1.0)

    # tiny correction for rounding errors
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = np.where((x > 1e-12) & (x < 1.0 - 1e-12))[0]
        if free.size:
            add = diff / free.size
            x[free] += add
            x = np.clip(x, 0.0, 1.0)
    return x

# ----------------------------------------------------------------------
# One gradient‑descent step for the smoothed max‑overlap objective
# ----------------------------------------------------------------------
def _gd_step(h: np.ndarray, alpha: float, lr: float) -> np.ndarray:
    n = h.shape[0]
    s = h.sum()
    # linear autocorrelation (zero‑padded)
    a = np.convolve(h, h[::-1], mode="full")
    O = s - a                     # O_k = sum(h) - sum_i h_i h_{i+k}
    # soft‑max weights emphasising the largest O_k
    O_max = O.max()
    w = np.exp(alpha * (O - O_max))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # convolution to obtain Σ w_k·h_{j±k}
    c2 = np.convolve(h, w, mode="full")[n - 1 : n - 1 + n]
    w_rev = w[::-1]
    c1 = np.convolve(h, w_rev, mode="full")[n - 1 : n - 1 + n]
    grad = np.ones_like(h) - (c1 + c2)       # ∂φ/∂h_j
    return h - lr * grad
