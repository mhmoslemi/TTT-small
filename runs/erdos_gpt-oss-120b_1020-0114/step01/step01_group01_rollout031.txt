    target = n/2
    best_h = h.copy()
    best_val = compute_c5(h)
    cur_h = h.copy()
    cur_val = best_val
    step = 0.5

        corr = np.correlate(cur_h, cur_h, mode='full')
        min_idx = np.argmin(corr)
        lag = min_idx - (n - 1)

            g[lag:] -= cur_h[:n-lag]

import time
import numpy as np

# -------------------------------------------------------------
# Helper utilities
# -------------------------------------------------------------
def project_feasible(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1] and project onto the hyperplane sum(v)=target_sum."""
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target_sum) < 1e-12:
        return v
    # simple water‑filling style iteration
    for _ in range(30):
        diff = target_sum - v.sum()
        if abs(diff) < 1e-12:
            break
        free = np.where((v > 0.0) & (v < 1.0))[0]
        if free.size == 0:
            # push a random coordinate toward the needed direction
            i = int(np.random.randint(v.size))
            v[i] = np.clip(v[i] + diff, 0.0, 1.0)
            continue
        delta = diff / free.size
        v[free] += delta
        np.clip(v, 0.0, 1.0, out=v)
    return v


def compute_c5(h: np.ndarray) -> float:
    """C5 = max_x  ∫ h(x)(1‑h(x+k))dx  computed on the discretisation."""
    n = h.shape[0]
    dx = 2.0 / n
    # np.correlate gives the linear cross‑correlation (full mode)
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return float(max_corr * dx)


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random point in the feasible set."""
    v = rng.random(n)
    return project_feasible(v, n / 2.0)


# -------------------------------------------------------------
# Gradient descent on the worst lag
# -------------------------------------------------------------
def gradient_descent(start_h: np.ndarray,
                    max_iter: int,
                    rng: np.random.Generator) -> (np.ndarray, float):
    """Projected gradient descent on the maximal overlap objective."""
    n = start_h.shape[0]
    target = n / 2.0
    cur_h = start_h.copy()
    cur_val = compute_c5(cur_h)
    best_h, best_val = cur_h.copy(), cur_val
    # initial step size – tuned experimentally
    step0 = 0.5
    for it in range(max_iter):
        # correlation of h with itself; the lag with minimal value drives the max overlap
        corr = np.correlate(cur_h, cur_h, mode="full")
        min_idx = int(np.argmin(corr))
        lag = min_idx - (n - 1)

        # gradient of O_lag = sum(h) - Σ h_i h_{i+lag}
        g = np.ones_like(cur_h)
        if lag >= 0:
            # forward neighbour
            g[: n - lag] -= cur_h[lag:]
            # backward neighbour
            g[lag:] -= cur_h[: n - lag]
        else:
            L = -lag
            g[L:] -= cur_h[: n - L]
            g[: n - L] -= cur_h[L:]

        # decreasing step size
        alpha = step0 / (1.0 + 0.01 * it)
        cand = cur_h - alpha * g
        cand = project_feasible(cand, target)
        cand_val = compute_c5(cand)

        if cand_val + 1e-12 < cur_val:
            cur_h, cur_val = cand, cand_val
            if cand_val < best_val:
                best_h, best_val = cand.copy(), cand_val

        # occasional tiny random jitter to escape flat regions
        if it % 100 == 0:
            jitter = (rng.random(n) - 0.5) * 0.02
            cur_h = np.clip(cur_h + jitter, 0.0, 1.0)
            cur_h = project_feasible(cur_h, target)
            cur_val = compute_c5(cur_h)
    return best_h, best_val


# -------------------------------------------------------------
# Local binary‑swap improvement
# -------------------------------------------------------------
def swap_local_search(start_h: np.ndarray,
                     max_trials: int,
                     rng: np.random.Generator) -> (np.ndarray, float):
    """Hill‑climbing by swapping a 1‑entry with a 0‑entry."""
    n = start_h.shape[0]
    cur_h = start_h.copy()
    cur_val = compute_c5(cur_h)
    best_h, best_val = cur_h.copy(), cur_val

    for _ in range(max_trials):
        ones = np.where(cur_h > 0.5)[0]
        zeros = np.where(cur_h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        cand = cur_h.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        cand_val = compute_c5(cand)
        if cand_val + 1e-12 < cur_val:
            cur_h, cur_val = cand, cand_val
            if cand_val < best_val:
                best_h, best_val = cand.copy(), cand_val
    return best_h, best_val
