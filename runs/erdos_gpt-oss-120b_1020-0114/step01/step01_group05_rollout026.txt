import time, numpy as np

def _project(v, target):
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo+hi)/2
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo+hi)/2, 0.0, 1.0)

def _evaluate(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return corr.max()

def _subgrad(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_val = corr.max()
    # identify all active shifts
    idxs = np.where(np.abs(corr - max_val) < 1e-12)[0]  # tolerance maybe 1e-12?
    grad = np.zeros_like(h)
    count = 0
    for idx in idxs:
        s = idx - (n - 1)
        if s >= 0:
            shift_forward = np.concatenate((h[s:], np.zeros(s)))
            shift_backward = np.concatenate((np.zeros(s), h[:n-s]))
        else:
            s_abs = -s
            shift_forward = np.concatenate((np.zeros(s_abs), h[:n-s_abs]))
            shift_backward = np.concatenate((h[s_abs:], np.zeros(s_abs)))
        grad += dx * (1.0 - shift_forward - shift_backward)
        count += 1
    if count > 1:
        grad /= count
    return grad
```

def _adam_optimize(h0, max_iters, time_limit, start_time):
    n = h0.shape[0]
    target = n/2.0
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha0 = 0.2  # may adjust
    best_h_local = h.copy()
    best_val_local = _evaluate(h)
    for t in range(1, max_iters+1):
        # time check
        if time.time() - start_time > time_limit:
            break
        grad = _subgrad(h)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        # decaying learning rate
        alpha = alpha0 / np.sqrt(t)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        h_new = _project(h - step, target)
        val_new = _evaluate(h_new)
        # accept if better, else keep old? continue with updated anyway
        h = h_new
        # if val improved
        if val_new < best_val_local - 1e-9:
            best_val_local = val_new
            best_h_local = h.copy()
        # occasional random jitter if stagnated
        if t % 200 == 0:
            # compute improvement over last e.g. 200?
            # To simplify, random jitter injection:
            if np.random.rand() < 0.3:
                h = _project(h + np.random.normal(scale=0.02, size=n), target)
        # could also break early if val <= desired threshold
        if best_val_local < 0.3808:  # target
            break
    return best_h_local, best_val_local

def _subgrad_single(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_idx = np.argmax(corr)
    s = max_idx - (n - 1)
    # compute shifted arrays:
    if s >= 0:
        forward = np.concatenate((h[s:], np.zeros(s)))
        backward = np.concatenate((np.zeros(s), h[:n-s]))
    else:
        s_abs = -s
        forward = np.concatenate((np.zeros(s_abs), h[:n-s_abs]))
        backward = np.concatenate((h[s_abs:], np.zeros(s_abs)))
    grad = dx * (1.0 - forward - backward)
    return grad
```

import time
import numpy as np

def _project(v, target):
    """Project v onto the set {0≤x≤1, sum x = target}."""
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _evaluate(h):
    """Return C5 for a discretised step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full') * dx)


def _subgrad(h):
    """
    Sub‑gradient of the active term max_k ∫h(x)(1‑h(x+k))dx.
    Uses the shift achieving the current maximum.
    """
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = int(np.argmax(corr))
    s = idx - (n - 1)          # integer shift corresponding to the max

    if s >= 0:
        # forward = h shifted left by s (h_{i+s}), backward = h shifted right by s (h_{i-s})
        forward = np.empty_like(h)
        forward[:n - s] = h[s:]
        forward[n - s:] = 0.0

        backward = np.empty_like(h)
        backward[:s] = 0.0
        backward[s:] = h[:n - s]
    else:
        p = -s
        forward = np.empty_like(h)
        forward[:p] = 0.0
        forward[p:] = h[:n - p]

        backward = np.empty_like(h)
        backward[:n - p] = h[p:]
        backward[n - p:] = 0.0

    grad = dx * (1.0 - forward - backward)
    return grad


def _optimize(h0, target, end_time):
    """
    Simple projected sub‑gradient descent with occasional stochastic jumps.
    Returns the best vector and its C5 value found before end_time.
    """
    h = h0.copy()
    cur_val = _evaluate(h)
    best_h = h.copy()
    best_val = cur_val
    last_improve = 0
    lr0 = 0.30                     # initial step‑size factor

    it = 0
    while True:
        it += 1
        if time.time() > end_time:
            break

        grad = _subgrad(h)
        step = lr0 / np.sqrt(it)
        h_new = _project(h - step * grad, target)
        new_val = _evaluate(h_new)

        # accept non‑worsening moves (monotone descent)
        if new_val <= cur_val:
            h = h_new
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_h = h_new.copy()
                last_improve = it
        else:
            # tiny probability to accept a worsening step (helps escape shallow plateaus)
            if np.random.rand() < 0.001:
                h = h_new
                cur_val = new_val

        # stochastic jitter if stuck for a while
        if it - last_improve > 2000 and (it % 500 == 0):
            h = _project(h + np.random.normal(scale=0.02, size=n), target)
            cur_val = _evaluate(h)

        # early stop if already very good
        if best_val < 0.3808:
            break

    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] with integral 1 that minimizes
    C5 = max_k ∫ h(x)(1‑h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start = time.time()
    # use 120 sample points – fine enough to capture useful structure while fast
    n = 120
    target = n / 2.0               # because ∫ h = 1  →  sum h_i = n/2

    best_h = None
    best_c5 = np.inf

    # end of computation will be a bit before the hard limit
    end_time = start + budget_s * 0.95

    # ----- try to improve the provided construction (if any) -----
    try:

            h0 = np.interp(xs_new, xs_old, h0)
        h0 = _project(h0, target)
