import time
import numpy as np
import scipy.optimize as opt
from scipy.special import logsumexp

# ---------- global variables for the smooth SLSQP optimizer ----------
_OPT_N = None
_OPT_ALPHA = None


def set_opt_params(n, alpha):
    """Set problem size and smoothing factor for the smooth max objective."""
    global _OPT_N, _OPT_ALPHA
    _OPT_N = n
    _OPT_ALPHA = alpha


def smooth_max_obj(h):
    """Smooth approximation of the max overlap (log‑sum‑exp)."""
    n = _OPT_N
    a = _OPT_ALPHA
    corr = np.correlate(h, 1.0 - h, mode="full")
    sm = logsumexp(a * corr) / a
    return sm * (2.0 / n)  # scale by dx


def sum_constraint(h):
    """Equality constraint: ∫h = 1  ⇔  sum(h) = n/2."""
    n = _OPT_N
    return np.sum(h) - n / 2.0


# ---------- core utilities ----------
def compute_c5(h):
    """Exact C5 value for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def worst_lag(h):
    """Return the lag (as integer shift) where the overlap is maximal."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return idx - (n - 1), corr[idx]  # lag, raw value (without dx)


def targeted_swap(h):
    """
    Perform a swap that directly reduces the worst lag.
    Returns a new array (or None if not applicable).
    """
    n = h.shape[0]
    lag, _ = worst_lag(h)

    if lag >= 0:
        # compare h[i] with h[i+lag]
        left = h[: n - lag]
        right = h[lag:]
        mask1 = (left > 0.5) & (right < 0.5)  # (1,0) pair
        mask2 = (left < 0.5) & (right > 0.5)  # (0,1) pair
        i_opts = np.flatnonzero(mask1)
        j_opts = np.flatnonzero(mask2)
        if i_opts.size == 0 or j_opts.size == 0:
            return None
        i = np.random.choice(i_opts)          # original index i
        j = np.random.choice(j_opts) + lag    # original index j = i+lag for mask2
    else:
        d = -lag
        # compare h[i+d] with h[i]
        left = h[d:]
        right = h[: n - d]
        mask1 = (left > 0.5) & (right < 0.5)   # (1,0) where left = h[i+d]
        mask2 = (left < 0.5) & (right > 0.5)   # (0,1)
        i_opts = np.flatnonzero(mask1) + d    # original index i = i_opts+d
        j_opts = np.flatnonzero(mask2)        # original index j
        if i_opts.size == 0 or j_opts.size == 0:
            return None
        i = np.random.choice(i_opts)
        j = np.random.choice(j_opts)

    new_h = h.copy()
    new_h[i] = 0.0
    new_h[j] = 1.0
    return new_h
