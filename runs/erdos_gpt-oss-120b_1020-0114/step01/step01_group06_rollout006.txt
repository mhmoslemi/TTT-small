def gradient_for_shift(h, shift):
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        k = shift
        # +1 term for i < n - k
        g[:n - k] += 1.0
        # -h_{i+k} term
        g[:n - k] -= h[k:]
        # -h_{i-k} term
        g[k:] -= h[:n - k]
    else:

        # +1 term for i >= s
        g[s:] += 1.0
        # -h[i - s] term for i >= s (i - s = i - s)
        g[s:] -= h[:n - s]
        # -h[i + s] term for i < n - s
        g[:n - s] -= h[s:]
    return g

import time, math, random
import numpy as np
from scipy import optimize

def _c5_value(h: np.ndarray) -> float:
    """Integral bound for a discrete step function."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))


def _proj_feasible(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto {0<=x<=1, sum(x)=target} (bounded simplex)."""
    # Bisection on lambda in x_i = clip(v_i - lam,0,1)
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _max_corr_and_shift(h: np.ndarray):
    """Return (max_corr_value, shift_index) where shift = idx-(n-1)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return float(corr[idx]), idx - (n - 1)


def _grad_shift(h: np.ndarray, shift: int):
    """Sub‑gradient of C_shift with respect to h (size‑n vector)."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        k = shift
        g[: n - k] += 1.0
        g[: n - k] -= h[k:]
        g[k:] -= h[: n - k]
    else:
        s = -shift
        g[s:] += 1.0
        g[s:] -= h[: n - s]
        g[: n - s] -= h[s:]
    return g


def _optimize_from(start_h: np.ndarray, n: int, target_sum: float,
                   time_limit: float, start_time: float):
    """Gradient‑guided simulated annealing starting from start_h."""
    # ensure correct length
    if start_h.shape[0] != n:
        old_n = start_h.shape[0]
        x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        start_h = np.interp(x_new, x_old, start_h)
    h = _proj_feasible(np.clip(start_h, 0.0, 1.0), target_sum)

    # current state
    cur_max, cur_shift = _max_corr_and_shift(h)
    cur_c5 = cur_max * (2.0 / n)

    best_h = h.copy()
    best_c5 = cur_c5

    # schedule parameters
    max_iters = 300_000
    step = 0.5
    step_decay = 0.99997
    temp_start, temp_end = 0.25, 1e-6
    stagnation = 0

    rng = np.random.default_rng()

    for it in range(max_iters):
        if time.time() - start_time > time_limit * 0.98:
            break

        # temperature (linear annealing)
        tau = it / max_iters
        temp = temp_start * (1.0 - tau) + temp_end * tau

        # gradient of the currently worst shift
        g = _grad_shift(h, cur_shift)

        # candidate indices (positive gradient -> decrease, negative -> increase)
        pos = np.where((g > 1e-9) & (h > 1e-12))[0]
        neg = np.where((g < -1e-9) & (h < 1.0 - 1e-12))[0]

        if pos.size == 0 or neg.size == 0:
            # fall back to a random feasible transfer
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            low = max(-h[i], -(1.0 - h[j]))
            high = min(1.0 - h[i], h[j])
            if high <= low:
                continue
            delta = rng.uniform(low, high)
            i_move, j_move = i, j
        else:
            i_move = rng.choice(pos)
            j_move = rng.choice(neg)
            delta_max = min(step, h[i_move], 1.0 - h[j_move])
            if delta_max <= 0:
                continue
            delta = rng.uniform(0.0, delta_max)

        h_new = h.copy()
        h_new[i_move] -= delta
        h_new[j_move] += delta

        new_max, new_shift = _max_corr_and_shift(h_new)
        new_c5 = new_max * (2.0 / n)

        if new_c5 < cur_c5 or rng.random() < math.exp(-(new_c5 - cur_c5) / temp):
            h = h_new
            cur_c5 = new_c5
            cur_shift = new_shift
            stagnation = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            stagnation += 1

        if (it + 1) % 1000 == 0:
            step *= step_decay

        if stagnation > 30_000:
            # fresh random start to escape possible dead‑end
            h = _proj_feasible(rng.random(n), target_sum)
            cur_max, cur_shift = _max_corr_and_shift(h)
            cur_c5 = cur_max * (2.0 / n)
            stagnation = 0
            step = 0.5

    return best_h, best_c5
