def hill_climb(h, max_iters, time_limit):
    best_h = h.copy()
    best_val = c5(h)
    for it in range(max_iters):
        # compute current O array and max
        corr = np.correlate(h, 1-h, mode='full')
        # O = corr * dx
        max_idx = corr.argmax()
        max_val = corr[max_idx] * dx
        # assign best
        if max_val < best_val:
            best_val = max_val
            best_h = h.copy()
        # get shift
        shift = max_idx - (len(h) - 1)
        if shift == 0:
            # shouldn't happen, break
            # but maybe we need random swap because O=0 means all zero?
            # For binary if O=0, all ones align perfectly for all shifts? improbable.
            # just random perturb.
            # Choose two random indices i,j and swap if feasible.
            # Continue.
            # ...
        else:
            # find bad and good indices
            if shift > 0:
                i_range = np.arange(0, len(h) - shift)
                i2 = i_range + shift
            else:
                shift_abs = -shift
                i_range = np.arange(shift_abs, len(h))
                i2 = i_range - shift_abs
            # compute boolean arrays
            h_i = h[i_range]
            h_j = h[i2]
            bad_mask = (h_i == 1) & (h_j == 0)
            good_mask = (h_i == 0) & (h_j == 1)
            bad_idxs = i_range[bad_mask]
            good_idxs = i_range[good_mask]
            if len(bad_idxs) > 0 and len(good_idxs) > 0:
                # select random indices
                i_bad = rng.choice(bad_idxs)
                i_good = rng.choice(good_idxs)
                # swap values
                h[i_bad], h[i_good] = h[i_good], h[i_bad]  # swap 1 and 0
                # continue
                continue
            else:

                # do random swap to explore

                if h[i] != h[j]:
                    # swap them (i.e., move mass)
                    h[i], h[j] = h[j], h[i]

import numpy as np

def _c5_value(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def _project_sum(h, target):
    # project onto [0,1] while adjusting sum to target
    # Use simple water-filling method: adjust by distributing delta across non-bound entries
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # Use greedy: add/subtract from elements not at bound
    idx = np.where((diff > 0) & (h < 1.0) | (diff < 0) & (h > 0.0))[0]
    # We'll iterate, but can also compute needed per element.
    # Simpler: loop through idx random order.
    for i in idx:
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            cap = 1.0 - h[i]
            inc = min(cap, diff)
        else:
            cap = -h[i]
            inc = max(cap, diff)  # diff negative, cap negative
        h[i] += inc
        diff -= inc
    # may still have tiny diff due to rounding, ignore
    return h

def _gradient_of_shift(h, shift):
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:

        # forward part
        if s < n:
            g[:n-s] += 1.0 - h[s:]
        # backward part

            g[s:] += -h[:n-s]
    else:

        # negative shift

            g[s:] += 1.0 - h[:n-s]
            g[:n-s] += -h[s:]
    return g

def _slsqp_refine(h):

import time, math
import numpy as np
from scipy import optimize

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _project_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Scale h into [0,1] while enforcing sum(h)=target (exactly)."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-14:
        return h
    # simple water‑filling: walk through entries and fix the sum
    indices = np.arange(h.size)
    # forward direction for diff>0, backward for diff<0
    if diff > 0:
        # need to add mass
        for i in indices:
            if diff <= 0:
                break
            cap = 1.0 - h[i]
            if cap <= 0:
                continue
            inc = min(cap, diff)
            h[i] += inc
            diff -= inc
    else:
        # need to remove mass
        diff = -diff
        for i in indices:
            if diff <= 0:
                break
            cap = h[i]
            if cap <= 0:
                continue
            dec = min(cap, diff)
            h[i] -= dec
            diff -= dec
    # any remaining tiny discrepancy is spread to the first free slot
    if abs(diff) > 1e-12:
        for i in indices:
            if 0.0 < h[i] < 1.0:
                h[i] += diff if diff > 0 else -diff
                break
    return h


def _gradient_of_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """
    Sub‑gradient of the overlap functional for a given shift.
    For shift >= 0 the functional is
        O(s)=sum_{i=0}^{n-1-s} h_i*(1-h_{i+s}).
    For shift < 0 the functional is computed analogously.
    """
    n = h.size
    g = np.zeros_like(h)
    if shift >= 0:
        s = shift
        if s < n:
            g[: n - s] += 1.0 - h[s:]
        if s > 0:
            g[s:] += -h[: n - s]
    else:
        s = -shift
        if s < n:
            g[s:] += 1.0 - h[: n - s]
            g[: n - s] += -h[s:]
    return g


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    safety_margin = 0.02 * budget_s  # keep a little slack
    best_h = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Choose a moderate resolution (even number => binary feasible)
    # ------------------------------------------------------------------
    n = 200
    if n % 2:
        n += 1
    target_sum = n // 2
    dx = 2.0 / n

    # --------------------------------------------------------------
    # Helper to evaluate and update global best
    # --------------------------------------------------------------
    def _maybe_update(h_vec):
        nonlocal best_h, best_val
        val = _c5_value(h_vec)
        if val < best_val:
            best_val = val
            best_h = h_vec.copy()
        return val

    # --------------------------------------------------------------
    # Try to use the parent construction (if any) as an additional start
    # --------------------------------------------------------------
    parent_candidates = []
    try:
        init = np.array(initial_h_values, dtype=float)  # comes from harness
        if init.ndim == 1 and init.size > 0:
            # interpolate to our resolution
            x_old = np.linspace(0, 2, init.size, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            interp = np.interp(x_new, x_old, init)
            parent_candidates.append(_project_sum(interp, target_sum))
    except Exception:
        pass  # no parent available or not usable

    # --------------------------------------------------------------
    # Main search loop – many cheap restarts with sub‑gradient + anneal
    # --------------------------------------------------------------
    restart = 0
    while time.time() - start_time < budget_s - safety_margin:
        restart += 1

        # ----- initialise ------------------------------------------------
        if parent_candidates and restart == 1:
            h = parent_candidates[0].copy()
        else:
            h = np.zeros(n, dtype=float)
            ones = rng.choice(n, size=target_sum, replace=False)
            h[ones] = 1.0

        cur_val = _maybe_update(h)

        # ----- sub‑gradient descent on the worst shift -------------------
        max_sub_iters = 4000
        step_start = 0.30
        for it in range(max_sub_iters):
            if time.time() - start_time >= budget_s - safety_margin:
                break

            # compute worst shift
            corr = np.correlate(h, 1.0 - h, mode="full")
            idx = corr.argmax()
            cur_val = float(corr[idx] * dx)
            shift = int(idx - (n - 1))

            # sub‑gradient for that shift
            grad = _gradient_of_shift(h, shift)

            # diminishing step size
            tau = it / max_sub_iters
            step = step_start * (1.0 - tau) + 0.002 * tau

            h_new = h - step * grad
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = _project_sum(h_new, target_sum)

            new_val = _c5_value(h_new)

            # simple acceptance rule (always accept improvements)
            if new_val < cur_val or rng.random() < 0.02:
                h = h_new
                cur_val = new_val
                _maybe_update(h)

        # ----- short simulated‑annealing phase -------------------------
        max_anneal = 8000
        temp_start, temp_end = 0.15, 1e-5
        for it in range(max_anneal):
            if time.time() - start_time >= budget_s - safety_margin:
                break
            tau = it / max_anneal
            temp = temp_start * (1.0 - tau) + temp_end * tau

            # propose swapping two opposite entries (keeps sum)
            i, j = rng.integers(0, n, size=2)
            if i == j or h[i] == h[j]:
                continue
            h_new = h.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]

            new_val = _c5_value(h_new)
            delta = new_val - cur_val
            if delta < 0 or rng.random() < math.exp(-delta / temp):
                h = h_new
                cur_val = new_val
                _maybe_update(h)

        # ----- occasional SLSQP fine‑tuning ----------------------------
        # (run only a few times to keep within budget)
        if restart % 7 == 0:
            cons = ({'type': 'eq', 'fun': lambda v: np.sum(v) - target_sum},)
            bounds = [(0.0, 1.0)] * n
            res = optimize.minimize(
                _c5_value,
                h,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 800, "ftol": 1e-10, "disp": False},
            )
            if res.success:
                h = np.clip(res.x, 0.0, 1.0)
                h = _project_sum(h, target_sum)
                _maybe_update(h)

    # ------------------------------------------------------------------
    # Final polish – one more SLSQP run on the best candidate found
    # ------------------------------------------------------------------
    if best_h is not None:
        cons = ({'type': 'eq', 'fun': lambda v: np.sum(v) - target_sum},)
        bounds = [(0.0, 1.0)] * n
        res = optimize.minimize(
            _c5_value,
            best_h,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 1200, "ftol": 1e-12, "disp": False},
        )
        if res.success:
            best_h = np.clip(res.x, 0.0, 1.0)
            best_h = _project_sum(best_h, target_sum)

    # ensure final feasibility and compute final bound
    best_h = _project_sum(best_h, target_sum)
    final_val = _c5_value(best_h)

    return (best_h.tolist(), float(final_val), int(n))
