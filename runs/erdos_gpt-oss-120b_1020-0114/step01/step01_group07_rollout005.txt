def optimise(N, time_budget):
    target_sum = N/2.0
    # starting h
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values)
        if N % base.size == 0:
            repeat = N // base.size
            h = np.tile(base, repeat)
        else:
            # random init
    else:
        h = np.random.rand(N)
    h = _project(h, target_sum)
    # compute current value
    cur_val = _c5_of(h)
    best_h = h.copy()
    best_val = cur_val
    cur_h = h.copy()
    cur_val = best_val

    start_local = time.time()
    # param schedule
    max_temp = 0.03
    min_temp = 1e-5
    max_step = 0.5
    min_step = 0.01
    no_improve = 0
    while True:
        if time.time() - start_local > time_budget*0.95:
            break
        elapsed = time.time() - start_local
        prog = elapsed / (time_budget*0.95)
        T = max_temp * (1 - prog) + min_temp
        step = max_step * (1 - prog) + min_step

        # decide move type
        if np.random.rand() < 0.3:   # targeted move
            # compute full correlation array once
            corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            # find shift with max C
            worst_idx = np.argmax(corr)
            shift = worst_idx - (N - 1)
            # compute shifted version of h
            shifted = np.zeros_like(cur_h)
            if shift >= 0:
                shifted[:N-shift] = cur_h[shift:]
            else:
                shifted[-shift:] = cur_h[:N+shift]
            mism = cur_h * (1.0 - shifted)   # mism contributions per position
            # pick i (large mism) and j (small mism)
            i = np.argmax(mism)
            j = np.argmin(mism)
            # ensure i!=j
            if i == j:
                # fallback to random swap
                i, j = np.random.randint(N, size=2)
            # compute feasible delta
            lower = max(-cur_h[i], cur_h[j] - 1.0)
            upper = min(1.0 - cur_h[i], cur_h[j])
            if lower < upper:
                delta_low = max(lower, -step)
                delta_high = min(upper, step)
                if delta_low < delta_high:
                    delta = np.random.uniform(delta_low, delta_high)
                else:
                    continue
            else:
                continue
        else:
            # random pair swap
            i, j = np.random.randint(N, size=2)
            if i == j:
                continue
            lower = max(-cur_h[i], cur_h[j] - 1.0)
            upper = min(1.0 - cur_h[i], cur_h[j])
            if lower >= upper:
                continue
            delta_low = max(lower, -step)
            delta_high = min(upper, step)
            if delta_low >= delta_high:
                continue
            delta = np.random.uniform(delta_low, delta_high)

        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        
        # evaluate
        new_val = np.max(np.correlate(new_h, 1.0 - new_h, mode="full")) * (2.0 / N)
        # acceptance
        if new_val < cur_val or np.random.rand() < math.exp(-(new_val - cur_val)/T):
            cur_h, cur_val = new_h, new_val
            if new_val < best_val - 1e-12:
                best_h, best_val = new_h.copy(), new_val
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # restart if stuck
        if no_improve > 300_000:
            # restart from best with random jitter
            cur_h = best_h.copy()
            # perturb random few entries
            for _ in range(10):
                idx = np.random.randint(N)
                cur_h[idx] = np.clip(cur_h[idx] + np.random.uniform(-step, step), 0.0, 1.0)
            # re-project to target sum (since perturbations may change sum)
            cur_h = _project(cur_h, target_sum)
            cur_val = _c5_of(cur_h)
            no_improve = 0

    # final projection
    best_h = _project(best_h, target_sum)
    final_val = _c5_of(best_h)
    # optionally evaluate using provided evaluate_erdos_solution
    if "evaluate_erdos_solution" in globals():
        try:
            final_val = evaluate_erdos_solution(best_h.tolist())
        except Exception:
            pass
    return best_h, final_val

def _project(v, target, tol=1e-12, max_iter=200):
    """Project v onto the box [0,1] while forcing sum(v)=target."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    it = 0
    while abs(diff) > tol and it < max_iter:
        it += 1
        if diff > 0:
            free = v < 1.0 - 1e-12
            if not free.any():
                break
            add = diff / free.sum()
            v[free] = np.minimum(v[free] + add, 1.0)
        else:
            free = v > 1e-12
            if not free.any():
                break
            sub = diff / free.sum()
            v[free] = np.maximum(v[free] + sub, 0.0)
        diff = target - v.sum()
    return v

def _c5_of(h):
    """Compute C5 for a step vector h (already discretised)."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)

def _thue_morse(n):
    """Return a 0/1 Thue‑Morse sequence of length n (n must be a power of two)."""
    seq = np.zeros(n, dtype=float)
    for i in range(n):
        # parity of popcount(i)
        seq[i] = (bin(i).count("1") & 1)
    return seq

def _shifted(arr, s):
    """Return arr shifted by s positions with zero padding (non‑circular)."""
    n = arr.size
    out = np.zeros_like(arr)
    if s >= 0:
        out[: n - s] = arr[s:]
    else:
        out[-s :] = arr[: n + s]
    return out
