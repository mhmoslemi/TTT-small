def _evaluate(h):
    n = h.size
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full') * dx)

def _project(v, target):
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _active_shifts(corr, eps=1e-12):
    """Indices of shifts attaining (up to tolerance) the maximal correlation."""
    mx = corr.max()
    rel = eps * max(1.0, mx)
    return np.where(np.abs(corr - mx) <= rel)[0]


def _subgradient_and_max(h):
    """
    Compute a subgradient of max_k C_k at h and also return the current max value.
    For each shift s that attains the maximum, the gradient component is
        g_i = 1 - (h_{i-s} + h_{i+s})   (zero‑padded outside)
    """
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    act = _active_shifts(corr)
    # distance from centre index (n-1) gives |shift|
    shifts = np.abs(act - (n - 1)).astype(int)
    uniq = np.unique(shifts)
    grad = np.zeros_like(h)
    for s in uniq:
        if s == 0:
            grad += 1 - 2 * h
        else:
            g = np.full(n, 1.0)
            g[s:] -= h[:-s]          # term h_{i-s}
            g[:-s] -= h[s:]          # term h_{i+s}
            grad += g
    grad /= uniq.size
    return grad, corr.max()


# --------------------------------------------------------------
# Core optimisation routine for a single problem size
# --------------------------------------------------------------
def _optimize_once(h, target, start, budget):
    """
    Projected gradient descent on the exact max objective.
    Uses a simple momentum scheme and occasional random jitters.
    """
    n = h.size
    best_h = h.copy()
    best_c = _evaluate(h)
    cur_h = h.copy()
    cur_c = best_c
    vel = np.zeros_like(h)

    mu = 0.9          # momentum coefficient
    lr0 = 0.5         # initial learning rate
    max_iter = 20000

    for it in range(max_iter):
        # time guard
        if time.time() - start > budget * 0.95:
            break

        grad, _ = _subgradient_and_max(cur_h)
        lr = lr0 / (1.0 + it * 1e-4)          # gentle decay
        vel = mu * vel - lr * grad
        h_new = _project(cur_h + vel, target)
        new_c = _evaluate(h_new)

        if new_c <= cur_c:                   # accept improvement (or equal)
            cur_h, cur_c = h_new, new_c
            if new_c < best_c:
                best_c, best_h = new_c, h_new.copy()
        else:
            # reject – damp momentum to stay close to feasible region
            vel *= 0.5

        # every 512 steps try a small random perturbation to escape plateaus
        if (it & 511) == 0 and it > 0:
            jitter = np.random.normal(scale=0.02, size=n)
            h_j = _project(cur_h + jitter, target)
            c_j = _evaluate(h_j)
            if c_j < best_c:
                best_c, best_h = c_j, h_j.copy()
                cur_h, cur_c = h_j, c_j
                vel = np.zeros_like(h)

    return best_h, best_c


# --------------------------------------------------------------
# Main entry point required by the harness
# --------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Choose a handful of problem sizes.  If the harness supplies an
    # initial construction we include its size as well.
    # ------------------------------------------------------------------
    n_candidates = [40, 44, 48, 56, 64, 80]
    try:
        h0 = np.array(initial_h_values, dtype=float)
        if h0.ndim == 1:
            n0 = h0.shape[0]
            if n0 not in n_candidates:
                n_candidates.append(n0)
    except Exception:
        h0 = None

    best_h_global = None
    best_c_global = np.inf
    best_n_global = None

    # --------------------------------------------------------------
    # Repeated independent trials (different n, different starts)
    # --------------------------------------------------------------
    while time.time() - start < budget_s * 0.94:
        n = int(np.random.choice(n_candidates))
        target = n * 0.5

        # initialise ------------------------------------------------
        if h0 is not None and n == h0.shape[0]:
            h = h0.copy()
        else:
            h = np.random.rand(n)
        h = _project(h, target)

        # a few quick random jitters to diversify the start
        for _ in range(5):
            h = _project(h + np.random.normal(scale=0.05, size=n), target)

        # optimise -------------------------------------------------
        h_opt, c_opt = _optimize_once(h, target, start, budget_s)

        # keep global best -----------------------------------------
        if c_opt < best_c_global:
            best_c_global = c_opt
            best_h_global = h_opt.copy()
            best_n_global = n

        # early exit if we already beat the target bound
        if best_c_global < 0.3805:
            break

    # ------------------------------------------------------------------
    # Final polishing of the best candidate found so far
    # ------------------------------------------------------------------
    if best_h_global is not None:
        h_pol, c_pol = _optimize_once(
            best_h_global, best_n_global * 0.5, start, budget_s
        )
        if c_pol < best_c_global:
            best_c_global = c_pol
            best_h_global = h_pol.copy()
            best_n_global = best_h_global.shape[0]

    # ------------------------------------------------------------------
    # safety net – should never happen, but guarantees a return value
    # ------------------------------------------------------------------
    if best_h_global is None:
        n = 80
        target = n * 0.5
        best_h_global = _project(np.full(n, 0.5), target)
        best_c_global = _evaluate(best_h_global)
        best_n_global = n

    return best_h_global, float(best_c_global), int(best_n_global)
