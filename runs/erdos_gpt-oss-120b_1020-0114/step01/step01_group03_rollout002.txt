import time, numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0/n
    return np.max(np.correlate(h, 1 - h, mode='full') * dx)

def upsample_h(old_h, new_n):
    old_n = old_h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    target = new_n / 2.0
    s = new_h.sum()
    if s != 0:
        new_h = new_h * (target / s)
    new_h = np.clip(new_h, 0.0, 1.0)
    # final correction
    diff = target - new_h.sum()
    if abs(diff) > 1e-12:
        new_h += diff / new_n
        new_h = np.clip(new_h, 0.0, 1.0)
    return new_h

def anneal(start_h, steps, init_temp, final_temp, time_check=None):
    # time_check: function returning True if time left
    cur = start_h.copy()
    cur_val = compute_c5(cur)
    best_local_h = cur.copy()
    best_local_val = cur_val
    n = cur.size
    for t in range(steps):
        if time_check is not None and not time_check():
            break
        # linear schedule
        T = init_temp + (final_temp - init_temp) * (t / steps)
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        max_eps = min(cur[i], 1.0 - cur[j])
        if max_eps <= 0:
            continue
        # early stage: larger moves; we adjust with temperature: use power scaling
        eps = max_eps * (np.random.rand() ** 0.5)  # bias larger
        cur[i] -= eps
        cur[j] += eps
        # compute C5
        new_val = compute_c5(cur)
        delta = new_val - cur_val
        if delta <= 0 or (T > 0 and np.random.rand() < np.exp(-delta / T)):
            cur_val = new_val
            if new_val < best_local_val:
                best_local_val = new_val
                best_local_h = cur.copy()
        else:
            # revert move
            cur[i] += eps
            cur[j] -= eps
    return best_local_h, best_local_val

import time
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Exact C₅ for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h); zero‑padding is handled by mode='full'
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def _upsample_h(old_h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a finer grid while preserving the integral."""
    old_n = old_h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)

    target_sum = new_n / 2.0
    s = new_h.sum()
    if s > 0:
        new_h = new_h * (target_sum / s)

    new_h = np.clip(new_h, 0.0, 1.0)

    # tiny correction for any remaining drift
    diff = target_sum - new_h.sum()
    if abs(diff) > 1e-12:
        new_h += diff / new_n
        new_h = np.clip(new_h, 0.0, 1.0)

    return new_h


def _anneal(start_h: np.ndarray, steps: int,
            init_temp: float, final_temp: float,
            time_ok) -> tuple[np.ndarray, float]:
    """Simulated‑annealing move set: transfer a small amount of mass."""
    cur = start_h.copy()
    cur_val = _compute_c5(cur)
    best_h = cur.copy()
    best_val = cur_val
    n = cur.size

    for t in range(steps):
        if not time_ok():
            break

        # linear temperature schedule
        T = init_temp + (final_temp - init_temp) * (t / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # maximal transferable mass while respecting bounds
        max_eps = min(cur[i], 1.0 - cur[j])
        if max_eps <= 0.0:
            continue

        # bias towards relatively larger moves early on
        eps = max_eps * (np.random.rand() ** 0.5)

        cur[i] -= eps
        cur[j] += eps

        new_val = _compute_c5(cur)
        delta = new_val - cur_val

        if delta <= 0.0 or (T > 0.0 and np.random.rand() < np.exp(-delta / T)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur.copy()
        else:
            # revert
            cur[i] += eps
            cur[j] -= eps

    return best_h, best_val
