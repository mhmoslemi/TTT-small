def compute_c5(arr):
    """Exact max overlap integral for a discrete step function."""
    n = arr.shape[0]
    dx = 2.0 / n
    corr = np.correlate(arr, 1.0 - arr, mode="full") * dx
    return float(np.max(corr))


def upsample_h(old_h, new_n):
    """Linear interpolation to a finer grid, preserving the integral."""
    old_n = old_h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    target = new_n / 2.0
    s = new_h.sum()
    if s > 0:
        new_h *= target / s
    new_h = np.clip(new_h, 0.0, 1.0)
    # final scaling for any tiny residue
    s = new_h.sum()
    if s > 0:
        new_h *= target / s
    return new_h


def smooth_obj_factory(n, alpha):
    """Factory for the smooth max-overlap objective used by SLSQP."""
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return logsumexp(alpha * corr) / alpha * (2.0 / n)
    return obj
