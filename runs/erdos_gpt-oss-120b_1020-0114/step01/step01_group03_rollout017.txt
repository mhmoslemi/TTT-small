import time, math
import numpy as np
import scipy.optimize as opt
from scipy.special import logsumexp

# ---------- global helpers for the smoothed SLSQP objective ----------
_GLOBAL_N = None
_GLOBAL_ALPHA = None


def set_global_params(n, alpha):
    """Store problem size and smoothing factor for the objective."""
    global _GLOBAL_N, _GLOBAL_ALPHA
    _GLOBAL_N = n
    _GLOBAL_ALPHA = alpha


def objective_func(h):
    """
    Smoothed approximation of max_k ∫ h(x)(1‑h(x+k))dx.
    The correlation is computed on the discrete grid; the result is scaled by dx.
    """
    n = _GLOBAL_N
    alpha = _GLOBAL_ALPHA
    corr = np.correlate(h, 1.0 - h, mode="full")
    smooth = logsumexp(alpha * corr) / alpha
    return smooth * (2.0 / n)   # multiply by dx (dx = 2/n)


def constraint_sum_eq(h):
    """Equality: ∑h = n/2  (i.e. ∫ h = 1)."""
    n = _GLOBAL_N
    return np.sum(h) - n / 2.0


def compute_c5(h):
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)


# ---------- simulated annealing -------------------------------------------------
def simulated_annealing(start_h, time_limit):
    """Return a (h, C5) pair obtained by SA within time_limit seconds."""
    n = start_h.shape[0]
    target_sum = n / 2.0
    cur_h = start_h.copy()
    cur_c5 = compute_c5(cur_h)

    best_h = cur_h.copy()
    best_c5 = cur_c5

    # schedule parameters
    T_start = 0.02
    T_end = 1e-6
    max_iter = 2_000_000  # theoretical upper bound; loop breaks on time

    it = 0
    t0 = time.time()
    while True:
        if time.time() - t0 >= time_limit:
            break
        it += 1
        # linear cooling
        frac = min(it / max_iter, 1.0)
        T = (T_start - T_end) * (1.0 - frac) + T_end

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        hi, hj = cur_h[i], cur_h[j]

        # decide direction of transfer
        max_eps = min(hi, 1.0 - hj)
        if max_eps <= 0:
            max_eps = min(hj, 1.0 - hi)
            if max_eps <= 0:
                continue
            src, dst = j, i
        else:
            src, dst = i, j

        # random amount to move
        eps = np.random.uniform(0.0, max_eps)
        if eps <= 0:
            continue

        new_h = cur_h.copy()
        new_h[src] -= eps
        new_h[dst] += eps

        # evaluate
        new_c5 = compute_c5(new_h)

        # Metropolis acceptance
        if new_c5 < cur_c5 or np.random.rand() < math.exp((cur_c5 - new_c5) / T):
            cur_h = new_h
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()

    # final projection onto the exact sum (tiny numerical drift)
    s = best_h.sum()
    if s != 0 and abs(s - target_sum) > 1e-12:
        best_h = best_h * (target_sum / s)
        best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, best_c5


# ---------- refinement via SLSQP -------------------------------------------------
def slsqp_refine(start_h, alphas, max_iter=250):
    """Run a few SLSQP refinements with increasing smoothing."""
    n = start_h.shape[0]
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": constraint_sum_eq}
    cur_h = start_h.copy()

    for a in alphas:
        set_global_params(n, a)
        res = opt.minimize(
            objective_func,
            cur_h,
            method="SLSQP",
            bounds=bounds,
            constraints=[cons],
            options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
        )
        cur_h = np.clip(res.x, 0.0, 1.0)
        s = cur_h.sum()
        if s != 0 and abs(s - target) > 1e-12:
            cur_h = cur_h * (target / s)
            cur_h = np.clip(cur_h, 0.0, 1.0)

    return cur_h


# ---------- deterministic greedy improvement ------------------------------------
def greedy_swaps(h, steps, time_limit):
    """Simple hill‑climbing: move a small amount from i to j if it improves C5."""
    n = h.shape[0]
    cur_h = h.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    t0 = time.time()
    for _ in range(steps):
        if time.time() - t0 >= time_limit:
            break
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        eps = np.random.uniform(0.01, 0.05)
        eps = min(eps, cur_h[i], 1.0 - cur_h[j])
        if eps <= 0:
            continue
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            cur_h, cur_c5 = new_h, new_c5
            if new_c5 < best_c5:
                best_c5, best_h = new_c5, new_h.copy()
    # final projection onto sum constraint
    target = n / 2.0
    s = best_h.sum()
    if s != 0 and abs(s - target) > 1e-12:
        best_h = best_h * (target / s)
        best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, best_c5
