def subgradient_refine(h0, max_iters=5000, time_limit=None, start_time=None):
    h = h0.copy()
    n = len(h)
    dx = 2.0 / n
    alpha0 = 0.5 # step size maybe
    # compute sum target
    target = n/2.0
    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        # compute overlap array
        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_val = np.max(corr)
        # choose lag attaining max (first)
        idx = np.argmax(corr)
        k = idx - (n - 1)
        # compute gradient
        # shift left by k: produce array of size n
        if k >= 0:
            shift_plus = np.concatenate([h[k:], np.zeros(k)])
            shift_minus = np.concatenate([np.zeros(k), h[:n - k]])
        else:
            kk = -k
            shift_plus = np.concatenate([np.zeros(kk), h[:n - kk]])  # shift left by negative means right shift
            shift_minus = np.concatenate([h[kk:], np.zeros(kk)])
        g = dx * (1.0 - shift_plus - shift_minus)
        # step size schedule
        alpha = alpha0 / np.sqrt(it + 1)
        h = h - alpha * g
        # project
        h = project_to_feasible(h, target)
    return h

import time
import numpy as np
from scipy.optimize import minimize, NonlinearConstraint

# --- global helpers for the SLSQP stage ---------------------------------
_opt_n = None          # number of variables
_opt_dx = None         # grid spacing 2/n
_opt_target_sum = None # required sum (n/2)

def objective_fun(x):
    """Minimise the auxiliary variable t (last entry)."""
    return x[-1]

def eq_constraint_fun(x):
    """Enforce integral constraint: sum(h) = n/2."""
    return np.sum(x[:_opt_n]) - _opt_target_sum

def ineq_constraint_fun(x):
    """t >= overlap_k for every shift."""
    h = x[:_opt_n]
    t = x[-1]
    corr = np.correlate(h, 1 - h, mode='full') * _opt_dx
    return t - corr  # all components must be >=0

def project_to_feasible(v, target, max_iter=15):
    """Euclidean projection onto {0≤x≤1, sum(x)=target}."""
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if cur == 0.0:
        # fill target indices with 1
        idx = np.random.choice(len(v), int(round(target)), replace=False)
        v[idx] = 1.0
        return v
    scale = target / cur
    v = np.clip(v * scale, 0.0, 1.0)
    for _ in range(max_iter):
        diff = target - v.sum()
        if abs(diff) < 1e-12:
            break
        free = np.where((v > 0.0) & (v < 1.0))[0]
        if free.size == 0:
            # push a random free entry
            i = np.random.randint(len(v))
            v[i] = np.clip(v[i] + diff, 0.0, 1.0)
            continue
        delta = diff / free.size
        v[free] += delta
        np.clip(v, 0.0, 1.0, out=v)
    return v

def optimise_slsqp(start_h):
    """Run a single SLSQP optimisation from start_h."""
    global _opt_n, _opt_dx, _opt_target_sum
    n = int(start_h.shape[0])
    _opt_n = n
    _opt_dx = 2.0 / n
    _opt_target_sum = n / 2.0

    h0 = project_to_feasible(start_h, _opt_target_sum)
    init_corr = np.correlate(h0, 1 - h0, mode='full') * _opt_dx
    t0 = float(np.max(init_corr) + 1e-8)

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    eq_cons = {'type': 'eq', 'fun': eq_constraint_fun}
    m = 2 * n - 1
    ineq_cons = NonlinearConstraint(
        ineq_constraint_fun,
        lb=np.zeros(m),
        ub=np.full(m, np.inf),
    )
    x0 = np.concatenate([h0, [t0]])
    res = minimize(
        objective_fun,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=[eq_cons, ineq_cons],
        options={'maxiter': 1200, 'ftol': 1e-9, 'disp': False}
    )
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    h_opt = project_to_feasible(h_opt, _opt_target_sum)
    try:
        val = float(evaluate_erdos_solution(h_opt))
    except Exception:
        val = float(res.x[-1])
    return h_opt, val

# --- sub‑gradient refinement --------------------------------------------
def shift_array(arr, k):
    """Shift left by k (positive) with zero padding; shape unchanged."""
    n = arr.shape[0]
    if k >= 0:
        return np.concatenate([arr[k:], np.zeros(k, dtype=arr.dtype)])
    else:
        kk = -k
        return np.concatenate([np.zeros(kk, dtype=arr.dtype), arr[:n - kk]])

def subgradient_refine(h0, max_iters=4000, start_time=None,
                       time_limit=None, rng=None):
    """Projected sub‑gradient descent on max‑overlap objective."""
    h = h0.copy()
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    alpha0 = 2.0
    for it in range(1, max_iters + 1):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        corr = np.correlate(h, 1 - h, mode='full') * dx
        idx = np.argmax(corr)
        k = idx - (n - 1)          # lag producing the worst overlap
        sp = shift_array(h, k)     # h_{i+k}
        sm = shift_array(h, -k)    # h_{i-k}
        grad = dx * (1.0 - sp - sm)    # ∂O_k/∂h
        alpha = alpha0 / np.sqrt(it)
        h = h - alpha * grad
        # small random perturbation to help escape plateaus
        if rng is not None and (it % 200 == 0):
            noise = rng.normal(scale=0.02 / np.sqrt(n), size=n)
            h += noise
        h = project_to_feasible(h, target)
    return h
