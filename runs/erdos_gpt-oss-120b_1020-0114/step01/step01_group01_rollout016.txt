import time
import math
import numpy as np
from scipy.optimize import minimize, NonlinearConstraint

# global placeholders used by the continuous optimiser
_opt_n = None          # number of points
_opt_dx = None         # grid spacing 2 / n
_opt_target_sum = None # n / 2

def _objective(x):
    """Minimise the auxiliary variable t (last entry)."""
    return x[-1]

def _eq_constraint(x):
    """Enforce integral constraint sum(h) = n/2."""
    return np.sum(x[:_opt_n]) - _opt_target_sum

def _ineq_constraint(x):
    """t >= correlation for every lag."""
    h = x[:_opt_n]
    t = x[-1]
    corr = np.correlate(h, 1 - h, mode='full') * _opt_dx
    return t - corr  # must be >= 0

def _project_to_feasible(h, target):
    """Clip h to [0,1] and (approximately) enforce sum(h) = target."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == 0:
        # fill with ones at random positions
        idx = np.random.choice(len(h), int(round(target)), replace=False)
        h[idx] = 1.0
        return h
    scale = target / cur
    h = np.clip(h * scale, 0.0, 1.0)
    # iterative correction for residual error
    for _ in range(10):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            i = np.random.randint(len(h))
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
            continue
        delta = diff / free.size
        h[free] += delta
        np.clip(h, 0.0, 1.0, out=h)
    return h

def _optimise_continuous(start_h):
    """Run a single SLSQP optimisation from a feasible start vector."""
    global _opt_n, _opt_dx, _opt_target_sum
    n = int(start_h.shape[0])
    _opt_n = n
    _opt_dx = 2.0 / n
    _opt_target_sum = n / 2.0

    h0 = _project_to_feasible(start_h, _opt_target_sum)

    # initial feasible upper bound t0
    init_corr = np.correlate(h0, 1 - h0, mode='full') * _opt_dx
    t0 = float(np.max(init_corr) + 1e-8)

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    eq_cons = {'type': 'eq', 'fun': _eq_constraint}
    m = 2 * n - 1
    ineq_cons = NonlinearConstraint(_ineq_constraint,
                                    lb=np.zeros(m),
                                    ub=np.full(m, np.inf))

    x0 = np.concatenate([h0, [t0]])

    res = minimize(_objective,
                   x0,
                   method='SLSQP',
                   bounds=bounds,
                   constraints=[eq_cons, ineq_cons],
                   options={'maxiter': 1500,
                            'ftol': 1e-10,
                            'disp': False})

    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    h_opt = _project_to_feasible(h_opt, _opt_target_sum)
    try:
        c5_val = float(evaluate_erdos_solution(h_opt))
    except Exception:
        c5_val = float(res.x[-1])
    return h_opt, c5_val

def _c5_of(h):
    """Return the overlap value C5 for a given vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return float(corr.max())

def _thue_morse(n):
    """Generate the first n values of the Thue–Morse sequence as 0/1 floats."""
    tm = np.array([0], dtype=int)
    while tm.size < n:
        tm = np.concatenate([tm, 1 - tm])
    return tm[:n].astype(float)
