    corr = np.correlate(h, h, mode='full')
    # corr index range: -(N-1) ... (N-1)

import numpy as np
import time, math, random

def _compute_c5(h):
    """Maximum overlap integral approximated on a discrete grid."""
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / N))

def _repair_sum(h, target):
    """Adjust h in‑place so that sum(h) == target while staying in [0,1]."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # greedy fill/empty
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = np.random.randint(len(h))
        if diff > 0:
            cap = 1.0 - h[i]
            if cap <= 0:
                continue
            inc = min(cap, diff)
            h[i] += inc
            diff -= inc
        else:
            cap = h[i]
            if cap <= 0:
                continue
            dec = min(cap, -diff)
            h[i] -= dec
            diff += dec
    # final fix if needed
    if abs(diff) > 1e-12:
        total = h.sum()
        if total > 0:
            h *= (target / total)
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                idxs = np.where(h < 1.0)[0]
                if len(idxs):
                    i = np.random.choice(idxs)
                    h[i] = min(1.0, h[i] + diff)
            else:
                idxs = np.where(h > 0.0)[0]
                if len(idxs):
                    i = np.random.choice(idxs)
                    h[i] = max(0.0, h[i] + diff)
    return h
