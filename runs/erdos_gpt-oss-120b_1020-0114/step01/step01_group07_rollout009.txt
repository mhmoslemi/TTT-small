    y = {}
    t = cp.Variable()
    # need t <= sum_{i} y_{i,k} for each k? Actually constraint sum >= t, but if we maximize t, make t <= each sum.
    # Actually we need sum_i y_{i,k} >= t for all k; equivalent to t <= sum_i y_{i,k}.
    for k in range(1, N):
        # define vector of y variables for this shift
        # size = N - k
        y_k = cp.Variable(N - k, boolean=True)
        y[k] = y_k
        # for each i: constraints linking y_k[i] <= x[i] and <= x[i+k] and >= x[i] + x[i+k] - 1.

    if x.value is None:
        return None
    x_sol = np.round(x.value).astype(int)
    t_val = t.value
    return x_sol, t_val

def _worst_shift_corr(h):
    N = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)

def _gradient_for_shift(h, shift):
    N = h.size
    dx = 2.0 / N
    grad = np.full(N, dx)  # from sum term
    s_abs = abs(shift)
    if s_abs == 0:

        grad -= 2 * dx * h

    i = np.arange(N - s_abs)
    j = i + s_abs
    # subtract contributions
    grad[i] -= dx * h[j]
    grad[j] -= dx * h[i]
    return grad

import numpy as np
import time

def _project(v, target):
    """Project vector v onto the hyperplane sum(v)=target with 0<=v<=1."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    # water‑filling loop
    while abs(diff) > 1e-12:
        if diff > 0:
            free = v < 1.0 - 1e-12
            nfree = free.sum()
            if nfree == 0:
                break
            add = diff / nfree
            v[free] = np.minimum(v[free] + add, 1.0)
        else:
            free = v > 1e-12
            nfree = free.sum()
            if nfree == 0:
                break
            sub = diff / nfree
            v[free] = np.maximum(v[free] + sub, 0.0)
        diff = target - v.sum()
    return v


def _c5_of(h):
    """Return the overlap constant C₅ for a step vector h."""
    N = h.size
    dx = 2.0 / N
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _worst_shift(h):
    """Return (shift, value) of the worst (largest) overlap."""
    N = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (N - 1)           # –(N‑1) … +(N‑1)
    val = corr[idx] * (2.0 / N)
    return shift, val, corr


def _gradient_for_shift(h, shift):
    """Gradient of the overlap for a given shift (with respect to h)."""
    N = h.size
    dx = 2.0 / N
    grad = np.full(N, dx)          # derivative of the linear Σh term
    s = abs(shift)
    if s == 0:
        # term h_i h_i appears; derivative gets -2*h_i
        grad -= 2.0 * dx * h
        return grad
    # indices where both i and i+s are inside [0,N)
    i = np.arange(N - s)
    j = i + s
    grad[i] -= dx * h[j]
    grad[j] -= dx * h[i]
    return grad


def _try_binary_milp(N, target_sum, time_limit):
    """Try to obtain a binary (0/1) solution with maximal minimum coincidence.
    Returns (h_binary, t_value) or (None, None) on failure."""
    try:
        import cvxpy as cp
    except Exception:
        return None, None

    # Create variables
    x = cp.Variable(N, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(x) == target_sum, t >= 0]

    for k in range(1, N):
        nk = N - k
        y = cp.Variable(nk, boolean=True)
        idx = np.arange(nk)
        constraints += [
            y <= x[idx],
            y <= x[idx + k],
            y >= x[idx] + x[idx + k] - 1,
            cp.sum(y) >= t,
        ]

    prob = cp.Problem(cp.Maximize(t), constraints)
    try:
        prob.solve(solver=cp.ECOS_BB, max_iters=50000, verbose=False)
    except Exception:
        return None, None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None, None
    h_bin = np.round(x.value).astype(float)
    t_val = t.value
    return h_bin, t_val
