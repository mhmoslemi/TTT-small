def max_overlap_shift_info(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    max_val = corr[max_idx] * (2.0 / n)
    shift = max_idx - (n - 1)
    # Determine term index for that shift
    if shift >= 0:
        # overlapping slice i=0..n-1-shift
        slice_len = n - shift
        terms = h[:slice_len] * (1 - h[shift:])
        idx_local = np.argmax(terms)
        i = idx_local
        j = i + shift
    else:
        p = -shift
        slice_len = n - p
        # terms for shift negative: i from p..n-1
        terms = h[p:] * (1 - h[:n-p])
        idx_local = np.argmax(terms)
        i = p + idx_local
        j = i - p
    return max_val, shift, i, j

def _project_to_feasible(h, target_sum, rng):
    """Clip to [0,1] and adjust so that sum(h) == target_sum."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if abs(cur - target_sum) < 1e-12:
        return h
    diff = target_sum - cur
    free = np.where((h > 0.0) & (h < 1.0))[0]
    if free.size == 0:
        # push a random free direction
        i = rng.integers(len(h))
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
        return h
    delta = diff / free.size
    h[free] += delta
    np.clip(h, 0.0, 1.0, out=h)
    # a final tiny correction if needed
    if abs(h.sum() - target_sum) > 1e-9:
        i = rng.integers(len(h))
        h[i] = np.clip(h[i] + (target_sum - h.sum()), 0.0, 1.0)
    return h


def _max_overlap(h):
    """Maximum of ∫ h(x)(1‑h(x+k))dx on the discrete grid."""
    n = h.shape[0]
    return np.correlate(h, 1 - h, mode="full").max() * (2.0 / n)


def _max_overlap_info(h):
    """
    Return (max_val, shift, i_src, i_dst) where shift is the lag that
    attains the maximum, and (i_src,i_dst) is a pair producing the maximal
    term h[i_src]*(1‑h[i_dst]) for that shift.
    """
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode="full")
    idx = int(np.argmax(corr))
    max_val = corr[idx] * (2.0 / n)
    shift = idx - (n - 1)               # integer shift in [-n+1,n-1]

    if shift >= 0:
        length = n - shift
        if length == 0:
            return max_val, shift, 0, 0
        terms = h[:length] * (1 - h[shift:])
        local = int(np.argmax(terms))
        i_src = local
        i_dst = i_src + shift
    else:
        p = -shift
        length = n - p
        if length == 0:
            return max_val, shift, 0, 0
        terms = h[p:] * (1 - h[:n - p])
        local = int(np.argmax(terms))
        i_src = p + local
        i_dst = i_src - p
    return max_val, shift, i_src, i_dst


def _improve_h(h_start, time_budget, start_time, rng):
    """
    Simulated‑annealing / greedy descent that preserves the total mass.
    Returns (best_h, best_val) found within the allotted budget.
    """
    h = h_start.copy()
    cur_val = _max_overlap(h)
    best_h = h.copy()
    best_val = cur_val
    n = h.shape[0]

    # Temperature schedule: start relatively high, decay to zero.
    while time.time() - start_time < time_budget:
        elapsed = time.time() - start_time
        remaining = max(time_budget - elapsed, 0.0)
        T = cur_val * 0.05 * (remaining / time_budget + 1e-9)

        # 60 % of the time we try a move that directly attacks the current worst lag.
        if rng.random() < 0.6:
            _, shift, src, dst = _max_overlap_info(h)
            if src != dst:
                max_delta = min(h[src], 1.0 - h[dst])
                if max_delta > 1e-14:
                    delta = max_delta * rng.random()
                    h[src] -= delta
                    h[dst] += delta
                    new_val = _max_overlap(h)
                    diff = new_val - cur_val
                    if diff <= 0 or rng.random() < math.exp(-diff / (T + 1e-12)):
                        cur_val = new_val
                        if new_val < best_val - 1e-12:
                            best_val = new_val
                            best_h = h.copy()
                        continue

                    h[src] += delta
                    h[dst] -= delta

        # otherwise a random transfer
        i, j = rng.integers(0, n, size=2)
        if i == j:

        src, dst = (i, j) if rng.random() < 0.5 else (j, i)
        max_delta = min(h[src], 1.0 - h[dst])
        if max_delta <= 1e-14:

        delta = max_delta * rng.random()
        h[src] -= delta
        h[dst] += delta
        new_val = _max_overlap(h)
        diff = new_val - cur_val
        if diff <= 0 or rng.random() < math.exp(-diff / (T + 1e-12)):

            if new_val < best_val - 1e-12:
                best_val = new_val
                best_h = h.copy()
        else:

            h[dst] -= delta

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Obtain an initial feasible vector
    # ------------------------------------------------------------------
    try:
        init_vec = np.array(initial_h_values, dtype=float)  # provided by harness
    except Exception:
        n0 = 100
        init_vec = np.zeros(n0)
        idx = rng.choice(n0, n0 // 2, replace=False)
        init_vec[idx] = 1.0
    n0 = init_vec.shape[0]
    target_sum0 = n0 / 2.0
    h0 = _project_to_feasible(init_vec, target_sum0, rng)

    # ------------------------------------------------------------------
    # 2. Choose a working resolution (at least 200 points if possible)
    # ------------------------------------------------------------------
    if n0 >= 200:
        work_n = n0
        work_h = h0.copy()
    else:
        work_n = 200
        factor = work_n // n0
        if work_n % n0 == 0:
            work_h = np.repeat(h0, factor)
        else:
            x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
            x_new = np.linspace(0.0, 2.0, work_n, endpoint=False)
            work_h = np.interp(x_new, x_old, h0)
        work_h = _project_to_feasible(work_h, work_n / 2.0, rng)

    # ------------------------------------------------------------------
    # 3. Local optimisation on the chosen grid
    # ------------------------------------------------------------------
    remaining = budget_s - (time.time() - start)
    # keep a margin for final evaluation and possible refinement
    main_budget = max(remaining * 0.85, 10.0)
    best_h, best_val = _improve_h(work_h, main_budget, start, rng)

    # ------------------------------------------------------------------
    # 4. If time permits, move to a finer grid (400 points) and refine
    # ------------------------------------------------------------------
    elapsed = time.time() - start
    if elapsed < budget_s * 0.8 and best_val > 0.3809 and work_n < 400:
        finer_n = 400
        if finer_n % work_n == 0:
            finer_h = np.repeat(best_h, finer_n // work_n)
        else:
            x_old = np.linspace(0.0, 2.0, work_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, finer_n, endpoint=False)
            finer_h = np.interp(x_new, x_old, best_h)
        finer_h = _project_to_feasible(finer_h, finer_n / 2.0, rng)
        remaining = budget_s - (time.time() - start) - 2.0
        if remaining > 30.0:
            best_h, best_val = _improve_h(finer_h, remaining, start, rng)

    # ------------------------------------------------------------------
    # 5. Final sanity: enforce feasibility and compute the official bound
    # ------------------------------------------------------------------
    final_n = best_h.shape[0]
    best_h = _project_to_feasible(best_h, final_n / 2.0, rng)
    # Use the official evaluator when available
    if "evaluate_erdos_solution" in globals():
        try:
            final_val = float(evaluate_erdos_solution(best_h))
        except Exception:
            final_val = float(_max_overlap(best_h))
    else:
        final_val = float(_max_overlap(best_h))

    return best_h, final_val, final_n
