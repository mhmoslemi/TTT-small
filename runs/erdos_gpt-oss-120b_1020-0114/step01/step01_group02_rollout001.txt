import numpy as np, time, random, math

def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for discrete h."""
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust h in‑place so that sum(h) == target while respecting 0≤h≤1."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # greedy fill/empty
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = np.random.randint(len(h))
        if diff > 0:
            cap = 1.0 - h[i]
            if cap <= 0:
                continue
            inc = min(cap, diff)
            h[i] += inc
            diff -= inc
        else:
            cap = h[i]
            if cap <= 0:
                continue
            dec = min(cap, -diff)
            h[i] -= dec
            diff += dec
    if abs(diff) > 1e-12:
        total = h.sum()
        if total > 0:
            h *= target / total
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                idx = np.where(h < 1.0)[0]
                if idx.size:
                    i = np.random.choice(idx)
                    h[i] = min(1.0, h[i] + diff)
            else:
                idx = np.where(h > 0.0)[0]
                if idx.size:
                    i = np.random.choice(idx)
                    h[i] = max(0.0, h[i] + diff)
    return h

def _upsample_initial(old: np.ndarray, new_n: int) -> np.ndarray:
    """Linear periodic up‑sampling of old vector to length new_n."""
    n0 = old.shape[0]
    xs0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    xs = np.linspace(0.0, 2.0, new_n, endpoint=False)
    # make periodic by adding point at 2.0 equal to first value
    xp = np.concatenate([xs0, [2.0]])
    fp = np.concatenate([old, [old[0]]])
    return np.interp(xs, xp, fp)
