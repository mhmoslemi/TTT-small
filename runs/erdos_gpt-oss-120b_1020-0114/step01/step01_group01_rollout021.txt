        w_decr = np.zeros(n)

        w_inc[j[valid]] = h[indices[valid]]
        # pick i and j
        if w_decr.sum() > 0:

        else:

            j_inc = rng.integers(n)

def slsqp_refine(h_init, target_sum, dx, maxiter=200):
    n = len(h_init)
    # variables: n h_i + t
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    def obj(x):
        return x[-1]
    def eq(x):
        return np.sum(x[:n]) - target_sum
    def ineq(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return t - corr
    # initial feasible t
    t0 = max(np.correlate(h_init, 1 - h_init, mode='full') * dx) + 1e-8
    x0 = np.concatenate([h_init, [t0]])
    # constraints
    eq_con = {'type': 'eq', 'fun': eq}
    ineq_con = NonlinearConstraint(ineq, lb=0, ub=np.inf)
    res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                   constraints=[eq_con, ineq_con],
                   options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    # enforce sum again
    h_opt = project_feasible(h_opt, target_sum, rng) # need rng
    return h_opt, float(res.x[-1]), res.success

def project_feasible(h, target_sum, rng):
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == 0.0:
        idx = rng.choice(len(h), int(round(target_sum)), replace=False)
        h[idx] = 1.0
        return h
    # scale
    scale = target_sum / cur
    h = np.clip(h * scale, 0.0, 1.0)
    # adjust residual
    for _ in range(10):
        diff = h.sum() - target_sum
        if abs(diff) < 1e-12:

            if free.size == 0:
                break
            delta = -diff / free.size
            h[free] += delta
            np.clip(h, 0.0, 1.0, out=h)

def _project_feasible(h, target_sum, rng):
    """Clip to [0,1] and adjust linearly so that sum(h)=target_sum (approximately)."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == 0.0:
        # place unit mass on a random subset
        k = int(round(target_sum))
        if k > 0:
            idx = rng.choice(len(h), k, replace=False)
            h[idx] = 1.0
        return h
    # simple scaling

    h = np.clip(h * scale, 0.0, 1.0)
    # iterative residual correction
    for _ in range(10):
        diff = h.sum() - target_sum
        if abs(diff) < 1e-12:

        if diff > 0:

    while time.time() < end_time:
        # progress based step size
        progress = (time.time() - start_time) / budget_s
        max_step = 0.05 * (1.0 - progress) + 0.001 * progress

        # Occasionally perform a completely random transfer
        if rng.random() < prob_random_move:
            i_dec, j_inc = rng.integers(n, size=2)
            if i_dec == j_inc:
                continue
            delta_max = min(h[i_dec], 1.0 - h[j_inc])
            if delta_max <= 0:
                continue
            delta = delta_max * rng.random() * max_step
            h_new = h.copy()
            h_new[i_dec] -= delta
            h_new[j_inc] += delta
        else:
            # Focus on the worst shift
            shift = cur_idx - (n - 1)
            inds = np.arange(n)
            j = inds - shift
            valid = (j >= 0) & (j < n)

            w_dec = np.zeros(n)
            w_inc = np.zeros(n)
            w_dec[valid] = 1.0 - h[j[valid]]
            w_inc[j[valid]] = h[inds[valid]]

            if w_dec.sum() > 0:
                i_dec = rng.choice(n, p=w_dec / w_dec.sum())
            else:
                i_dec = rng.integers(n)

            if w_inc.sum() > 0:
                j_inc = rng.choice(n, p=w_inc / w_inc.sum())
            else:
                j_inc = rng.integers(n)

            if i_dec == j_inc:
                j_inc = (j_inc + 1) % n

            delta_max = min(h[i_dec], 1.0 - h[j_inc])
            if delta_max <= 0:
                continue
            delta = delta_max * rng.random() * max_step
            if delta <= 0:
                continue

            h_new = h.copy()
            h_new[i_dec] -= delta
            h_new[j_inc] += delta

        # evaluate the move
        new_val, new_idx = _max_corr(h_new, dx)

        # accept only if it does not increase the maximal overlap
        if new_val <= cur_val + 1e-12:
            h = h_new
            cur_val, cur_idx = new_val, new_idx
            if new_val < best_val - 1e-12:
                best_val, best_h = new_val, h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # restart if we stagnate too long
        if no_improve >= max_no_improve:
            h = _project_feasible(rng.random(n), target_sum, rng)
            cur_val, cur_idx = _max_corr(h, dx)
            no_improve = 0

    return best_h, best_val


# ----------------------------------------------------------------------
# SLSQP refinement (single run)
# ----------------------------------------------------------------------
def _slsqp_refine(h_init, target_sum, dx, maxiter, rng):
    n = h_init.shape[0]

    def obj(x):
        return x[-1]

    def eq_con(x):
        return np.sum(x[:n]) - target_sum

    def ineq_con(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        return t - corr

    # initial feasible t
    t0 = float(np.max(np.correlate(h_init, 1.0 - h_init, mode='full') * dx)) + 1e-8
    x0 = np.concatenate([h_init, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    ineq = NonlinearConstraint(ineq_con, lb=np.zeros(2 * n - 1), ub=np.full(2 * n - 1, np.inf))

    res = minimize(
        obj,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=[{"type": "eq", "fun": eq_con}, ineq],
        options={"maxiter": maxiter, "ftol": 1e-9, "disp": False},
    )

    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    h_opt = _project_feasible(h_opt, target_sum, rng)
    t_opt = float(res.x[-1])
    return h_opt, t_opt, res.success


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Obtain or create an initial vector
    # ------------------------------------------------------------------
    init_vec = None
    if "initial_h_values" in globals():
        try:
            init_vec = np.asarray(initial_h_values, dtype=float)
        except Exception:
            init_vec = None
    if init_vec is None or init_vec.size == 0:
        n0 = 80
        init_vec = np.full(n0, 0.5)

    n0 = int(init_vec.shape[0])
    target0 = n0 / 2.0
    dx0 = 2.0 / n0
    init_vec = _project_feasible(init_vec, target0, rng)

    # ------------------------------------------------------------------
    # Phase 1: intensive local search on the original grid
    # ------------------------------------------------------------------
    phase1_end = start_time + budget_s * 0.55
    best_h, best_val = _local_search(
        init_vec, target0, dx0, phase1_end, start_time, budget_s, rng
    )
    best_n, best_dx = n0, dx0

    # ------------------------------------------------------------------
    # Phase 2: if time permits, upscale to a finer grid and continue searching
    # ------------------------------------------------------------------
    now = time.time()
    if now < start_time + budget_s * 0.80:
        n1 = min(200, n0 * 2)  # finer grid, capped at 200 points
        x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n1, endpoint=False)
        interp_h = np.interp(x_new, x_old, best_h, left=0.0, right=0.0)
        target1 = n1 / 2.0
        dx1 = 2.0 / n1
        interp_h = _project_feasible(interp_h, target1, rng)

        phase2_end = start_time + budget_s * 0.95
        h2, val2 = _local_search(
            interp_h, target1, dx1, phase2_end, start_time, budget_s, rng
        )
        if val2 < best_val:
            best_h, best_val = h2, val2
            best_n, best_dx = n1, dx1

    # ------------------------------------------------------------------
    # Final refinement with a short SLSQP run (if time remains)
    # ------------------------------------------------------------------
    remaining = start_time + budget_s - time.time()
    if remaining > 1.5:
        maxiter = max(30, int(remaining * 4))  # a few dozen iterations
        refined_h, refined_t, _ = _slsqp_refine(best_h, best_h.shape[0] / 2.0, best_dx, maxiter, rng)
        try:
            final_c5 = float(evaluate_erdos_solution(refined_h))
        except Exception:
            final_c5 = float(refined_t)
        best_h, best_val = refined_h, final_c5

    # ------------------------------------------------------------------
    # Ensure feasibility one last time and return
    # ------------------------------------------------------------------
    best_h = _project_feasible(best_h, best_h.shape[0] / 2.0, rng)
    try:
        best_val = float(evaluate_erdos_solution(best_h))
    except Exception:
        # fallback to direct computation
        best_val, _ = _max_corr(best_h, 2.0 / best_h.shape[0])

    return best_h, float(best_val), int(best_h.shape[0])
