    w = np.zeros_like(h)
    if shift > 0:
        w[: n - shift] += h[shift:]
        w[shift:] += h[: n - shift]
    else:
        s = -shift
        w[s:] += h[: n - s]
        w[: n - s] += h[s:]
    return w


# ----- simple greedy / simulated‑annealing improvement -----------------
def _greedy_optimize(h: np.ndarray, time_limit: float):
    """Move mass between two points while respecting the integral."""
    n = h.shape[0]
    target_sum = n / 2.0
    cur_h = h.copy()
    best_h = h.copy()
    best_val = _compute_c5(best_h)

    cur_val = best_val
    step = 0.05
    no_improve = 0
    start = time.time()

    while time.time() - start < time_limit:
        # worst shift and its w‑vector
        shift, _ = _worst_shift(cur_h)
        w = _w_vector(cur_h, shift)

        # choose a pair (increase / decrease) – 70 % deterministic, 30 % random
        if random.random() < 0.7:
            up_mask = cur_h < 1.0 - 1e-12
            down_mask = cur_h > 1e-12
            if not up_mask.any() or not down_mask.any():
                break
            up_idx = np.where(up_mask)[0]
            down_idx = np.where(down_mask)[0]
            i_up = up_idx[np.argmax(w[up_idx])]
            i_down = down_idx[np.argmin(w[down_idx])]
            delta = min(step, 1.0 - cur_h[i_up], cur_h[i_down])
        else:
            i_up = random.randrange(n)
            i_down = random.randrange(n - 1)
            if i_down >= i_up:
                i_down += 1
            max_inc = min(1.0 - cur_h[i_up], cur_h[i_down])
            max_dec = min(cur_h[i_up], 1.0 - cur_h[i_down])
            if max_inc > 0 and (max_dec == 0 or random.random() < 0.5):
                delta = random.random() * max_inc
            elif max_dec > 0:
                delta = -random.random() * max_dec
                i_up, i_down = i_down, i_up
            else:
                continue
        if delta <= 0:
            continue

        new_h = cur_h.copy()
        new_h[i_up] += delta
        new_h[i_down] -= delta
        new_val = _compute_c5(new_h)

        if new_val < cur_val - 1e-12:
            cur_h, cur_val = new_h, new_val
            no_improve = 0
            step = min(step * 1.05, 0.1)
            if cur_val < best_val:
                best_val, best_h = cur_val, cur_h.copy()
        else:
            T = step * 0.5
            if T > 0 and random.random() < math.exp(-(new_val - cur_val) / T):
                cur_h, cur_val = new_h, new_val
            no_improve += 1
            if no_improve > 400:
                step *= 0.5
                no_improve = 0
                if step < 1e-6:
                    break

    # project onto exact mass hyperplane
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        free = np.where((best_h > 1e-12) & (best_h < 1.0 - 1e-12))[0]
        if free.size:
            best_h[free] += diff / free.size
        else:
            best_h[0] = min(1.0, max(0.0, best_h[0] + diff))
        best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, _compute_c5(best_h)


# ----- linear‑programming relaxation ------------------------------------
def _solve_lp(n: int, time_limit: float):
    """Continuous LP relaxation using McCormick envelopes.
       Returns (h, t) where t = min shift autocorrelation (unscaled)."""
    try:
        import cvxpy as cp
    except Exception:
        return None, None

    # variables
    h = cp.Variable(n)
    t = cp.Variable()
    U = cp.Variable((n, n), nonneg=True)

    # basic constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n / 2.0,
        U <= h[:, None],
        U <= h[None, :],
        U >= h[:, None] + h[None, :] - 1,
    ]

    # each diagonal (shift) must have sum >= t
    for s in range(-n + 1, n):
        if s >= 0:
            diag_sum = sum(U[i, i + s] for i in range(n - s))
        else:
            diag_sum = sum(U[i - s, i] for i in range(n + s))
        constraints.append(diag_sum >= t)

    prob = cp.Problem(cp.Maximize(t), constraints)

    # try a fast conic solver; fall back to SCS if needed
    try:
        prob.solve(solver=cp.ECOS, max_iters=5000, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=20000, verbose=False)
        except Exception:
            return None, None

    if h.value is None or t.value is None:
        return None, None

    h_opt = np.clip(np.array(h.value).flatten(), 0.0, 1.0)

    # enforce exact mass (tiny numerical drift)
    target = n / 2.0
    diff = target - h_opt.sum()
    if abs(diff) > 1e-9:
        free = np.where((h_opt > 1e-8) & (h_opt < 1.0 - 1e-8))[0]
        if free.size:
            h_opt[free] += diff / free.size
            h_opt = np.clip(h_opt, 0.0, 1.0)
        else:
            h_opt[0] = min(1.0, max(0.0, h_opt[0] + diff))
    return h_opt, float(t.value)


# ----- orchestrator ----------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Optimise a step‑function for the Erdős C₅ constant.

    Returns (h_values, c5_bound, n_points)."""
    np.random.seed(seed)
    random.seed(seed)
    start_all = time.time()

    best_h = None
    best_c5 = np.inf
    best_n = None

    # Candidate discretisations (include the provided one if present)
    candidates = [45, 30, 40, 50, 60, 80]
    candidates = sorted(set(candidates))

    for n in candidates:
        if time.time() - start_all > budget_s - 1.0:
            break

        # ----- start point ------------------------------------------------
        if n == 45 and "initial_h_values" in globals():
            h_start = np.array(initial_h_values, dtype=float).copy()
            target = n / 2.0
            cur = h_start.sum()
            if abs(cur - target) > 1e-12:
                diff = target - cur
                free = np.where((h_start > 1e-12) & (h_start < 1.0 - 1e-12))[0]
                if free.size:
                    h_start[free] += diff / free.size
                else:
                    h_start[0] = min(1.0, max(0.0, h_start[0] + diff))
                h_start = np.clip(h_start, 0.0, 1.0)
        else:
            target = n / 2.0
            # random feasible vector (simple projection)
            rnd = np.random.rand(n)
            rnd = rnd / rnd.sum() * target
            # clip to [0,1] and redistribute any excess
            over = rnd > 1.0
            while over.any():
                excess = np.sum(rnd[over] - 1.0)
                rnd[over] = 1.0
                not_over = ~over
                if not not_over.any():
                    break
                rnd[not_over] += excess / np.count_nonzero(not_over)
                over = rnd > 1.0
            h_start = rnd

        # ----- LP relaxation ---------------------------------------------
        time_left = budget_s - (time.time() - start_all)
        if time_left < 5.0:
            break
        h_lp, _ = _solve_lp(n, time_left * 0.5)
        if h_lp is not None:
            c5_lp = _compute_c5(h_lp)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp.copy()
                best_n = n

        # ----- local greedy refinement ------------------------------------
        time_left = budget_s - (time.time() - start_all)
        if time_left < 2.0:
            break
        # allocate a modest budget for refinement (max 30 s)
        t_greedy = min(time_left * 0.6, 30.0)
        start_vec = best_h if (best_h is not None and best_h.shape[0] == n) else h_start
        h_ref, c5_ref = _greedy_optimize(start_vec.copy(), t_greedy)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()
            best_n = n

        # stop early if we already beat the target
        if best_c5 < 0.3808:
            break

    if best_h is None:
        # fallback: simple half‑ones half‑zeros on 45 points
        n = 45
        best_h = np.zeros(n)
        idx = np.random.choice(n, n // 2, replace=False)
        best_h[idx] = 1.0
        best_c5 = _compute_c5(best_h)
        best_n = n
    else:
        n = best_h.shape[0]

    # final sanity projection onto the mass hyperplane
    diff = n / 2.0 - best_h.sum()
    if abs(diff) > 1e-12:
        free = np.where((best_h > 1e-12) & (best_h < 1.0 - 1e-12))[0]
        if free.size:
            best_h[free] += diff / free.size
        else:
            best_h[0] = min(1.0, max(0.0, best_h[0] + diff))
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = _compute_c5(best_h)
    return best_h, float(final_c5), int(n)
