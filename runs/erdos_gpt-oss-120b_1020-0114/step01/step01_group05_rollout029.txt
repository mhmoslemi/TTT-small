def subgradient_step(h, step):
    # compute O(k) vector
    n = h.size
    dx = 2.0/n
    O = np.correlate(h, 1-h, mode='full') * dx
    k_star = np.argmax(O)
    lag = k_star - (n - 1)
    # compute grad: 1 - h shifted +1 - h shifted -1
    # shift forward (positive lag): h[i+lag]; shift backward (negative lag)
    up = np.zeros(n)
    down = np.zeros(n)
    if lag >= 0:
        up[:n-lag] = h[lag:]
        down[lag:] = h[:n-lag]
    else:
        lag_abs = -lag
        up[lag_abs:] = h[:n-lag_abs]
        down[:n-lag_abs] = h[lag_abs:]
    grad = 1.0 - up - down
    # step = step * grad (increase direction)
    new_h = _project(h - step * grad, target_sum)
    return new_h, O.max()

import time
import numpy as np

def _evaluate(h):
    """Maximum overlap integral for a discrete step function."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with 1-h gives the overlap integrand for each shift
    return np.max(np.correlate(h, 1 - h, mode="full") * dx)


def _project(v, target):
    """Clip to [0,1] and shift so that sum(v)=target."""
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        if adj.sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _softmax_grad(h, T):
    """Gradient of the smoothed max overlap (soft‑max approximation)."""
    n = h.size
    dx = 2.0 / n
    raw = np.correlate(h, 1 - h, mode="full") * dx       # length 2n‑1
    gmax = raw.max()
    expw = np.exp((raw - gmax) / T)
    p = expw / expw.sum()
    # indices for lag k = i‑j, shifted by (n‑1) to index p
    idx = (np.arange(n)[:, None] - np.arange(n)[None, :]) + (n - 1)
    P = p[idx]                                            # n×n weight matrix
    grad = dx * (P @ (1 - h) - P.T @ h)
    return grad
