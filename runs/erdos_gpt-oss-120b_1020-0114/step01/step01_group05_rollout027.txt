    n = h.shape[0]
    grad = np.ones_like(h)
    if shift >= 0:
        # forward part: h_{i+shift}
        if shift < n:
            grad[:n-shift] -= h[shift:]
        # backward part: h_{i-shift}
        if shift > 0:

        # backward: h_{i+s}

def gradient_shift(h, shift):
    n = h.shape[0]
    # forward: h_{i+shift}
    if shift >= 0:
        forward = np.empty(n, dtype=h.dtype)
        forward[:n-shift] = h[shift:]
        forward[n-shift:] = 0.0
        backward = np.empty(n, dtype=h.dtype)
        backward[shift:] = h[:n-shift]
        backward[:shift] = 0.0
    else:
        s = -shift
        forward = np.empty(n, dtype=h.dtype)
        forward[s:] = h[:n-s]
        forward[:s] = 0.0
        backward = np.empty(n, dtype=h.dtype)
        backward[:n-s] = h[s:]
        backward[n-s:] = 0.0
    grad = 1.0 - forward - backward
    return grad

def _run_continuous(init_h, start, budget_s):
    # Determine size, e.g., 120
    n_target = 120
    if init_h is not None:
        h = upsample_to_n(init_h, n_target)
    else:
        h = np.full(n_target, 0.5)
    target = n_target / 2
    h = _project(h, target)
    best_h = h.copy()
    best_c5 = _evaluate(h)

    init_lr = 0.5
    stagnation = 0
    last_improve = 0
    max_iters = 200000
    for it in range(max_iters):
        if time.time() - start > budget_s * 0.94:
            break
        # compute correlation
        corr = np.convolve(h, (1 - h)[::-1], mode='full')
        c5_cur = corr.max() * (2.0 / n_target)
        # compute max indices
        max_idx = np.flatnonzero(corr == corr.max())
        shift_idx = max_idx[np.random.randint(len(max_idx))]
        shift = shift_idx - (n_target - 1)
        grad = _gradient_shift(h, shift)  # shape n
        # step size schedule
        lr = init_lr / np.sqrt(it + 1)
        step = lr
        improved = False
        cur_c5 = c5_cur
        for _ in range(8):
            h_new = _project(h - step * grad, target)
            c5_new = _evaluate(h_new)
            if c5_new < cur_c5 - 1e-12:
                # accept
                h = h_new
                cur_c5 = c5_new
                improved = True
                break

        if improved and cur_c5 < best_c5:
            best_h = h.copy()
            best_c5 = cur_c5
            last_improve = it
        # stagnation check
        if it - last_improve > 2000:

def _run_binary_annealing(n, start, budget_s):
    target = n / 2
    # initialize binary vector randomly
    h = np.zeros(n)
    ones = np.random.choice(n, size=n//2, replace=False)
    h[ones] = 1.0
    cur_c5 = _evaluate(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    # temperature schedule based on iteration
    max_iters = 200000
    init_temp = 0.02
    for it in range(max_iters):
        if time.time() - start > budget_s * 0.97:
            break
        # propose swap
        zeros = np.where(h == 0)[0]
        ones_idx = np.where(h == 1)[0]
        if zeros.size == 0 or ones_idx.size == 0:
            break
        i0 = np.random.choice(zeros)
        i1 = np.random.choice(ones_idx)
        h_new = h.copy()
        h_new[i0] = 1.0
        h_new[i1] = 0.0
        c5_new = _evaluate(h_new)
        # acceptance
        # temperature decay
        T = init_temp * (1 - it / max_iters)
        if c5_new < cur_c5 or np.random.rand() < np.exp(-(c5_new - cur_c5) / (T + 1e-12)):
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()

import time
import numpy as np

# --------------------------------------------------------------
# Helper functions
# --------------------------------------------------------------

def _evaluate(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx for discrete h."""
    n = h.size
    dx = 2.0 / n
    # correlation = convolution with reversed second argument
    corr = np.convolve(h, (1 - h)[::-1], mode='full')
    return corr.max() * dx


def _project(v, target):
    """Project onto {0 ≤ x ≤ 1, sum x = target} using a simple bisection."""
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _gradient_shift(h, shift):
    """Gradient ∂/∂h of  ∫ h (1 - h_shift) dx for a given integer shift."""
    n = h.shape[0]
    if shift >= 0:
        # forward part: h_{i+shift}
        forward = np.empty(n, dtype=h.dtype)
        forward[: n - shift] = h[shift:]
        forward[n - shift :] = 0.0
        # backward part: h_{i-shift}
        backward = np.empty(n, dtype=h.dtype)
        backward[shift:] = h[: n - shift]
        backward[:shift] = 0.0
    else:
        s = -shift
        forward = np.empty(n, dtype=h.dtype)
        forward[s:] = h[: n - s]
        forward[:s] = 0.0
        backward = np.empty(n, dtype=h.dtype)
        backward[: n - s] = h[s:]
        backward[n - s :] = 0.0
    return 1.0 - forward - backward


def _upsample(h, new_n):
    """Linear interpolation of a step function to a larger number of points."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    target = new_n * 0.5
    return _project(h_new, target)

# Continuous sub‑gradient optimisation
# --------------------------------------------------------------

def _run_continuous(init_h, start_time, budget):
    # target resolution – a modest compromise between quality & speed
    N = 96
    if init_h is not None:
        h = _upsample(init_h, N)
    else:
        h = np.full(N, 0.5)
    target = N * 0.5
    h = _project(h, target)

    best_h = h.copy()
    best_c5 = _evaluate(best_h)

    # parameters
    init_lr = 0.6
    last_improve = 0
    max_iters = 30000

    for it in range(max_iters):
        # time guard
        if time.time() - start_time > budget * 0.94:
            break

        # correlation & max‑lag(s)
        corr = np.convolve(h, (1 - h)[::-1], mode='full')
        cur_c5 = corr.max() * (2.0 / N)

        max_idxs = np.flatnonzero(corr == corr.max())
        # average gradient over all maximizing lags (sub‑gradient)
        grad = np.zeros_like(h)
        for idx in max_idxs:
            shift = idx - (N - 1)
            grad += _gradient_shift(h, shift)
        grad /= len(max_idxs)

        # diminishing step size
        lr = init_lr / np.sqrt(it + 1.0)
        step = lr
        improved = False

        # simple back‑tracking line search
        for _ in range(8):
            h_new = _project(h - step * grad, target)
            new_c5 = _evaluate(h_new)
            if new_c5 <= cur_c5 + 1e-12:
                h = h_new
                cur_c5 = new_c5
                improved = True
                break
            step *= 0.5

        if improved and cur_c5 < best_c5 - 1e-12:
            best_h = h.copy()
            best_c5 = cur_c5
            last_improve = it

        # occasional jitter to escape plateaus
        if it - last_improve > 2500 and (it % 400 == 0):
            jitter = np.random.normal(scale=0.03, size=N)
            h_jit = _project(h + jitter, target)
            c5_jit = _evaluate(h_jit)
            if c5_jit < best_c5:
                h = h_jit
                best_h = h_jit.copy()
                best_c5 = c5_jit
                last_improve = it
            else:
                # a fresh random restart with small probability
                if np.random.rand() < 0.15:
                    h = _project(np.random.rand(N), target)
                    last_improve = it

    return best_h, best_c5, N

def _run_binary_annealing(N, start_time, budget):
    """Anneal a {0,1} vector with exactly N/2 ones."""
    target = N * 0.5
    # initialise a random binary feasible vector
    h = np.zeros(N, dtype=float)
    ones = np.random.choice(N, size=N // 2, replace=False)
    h[ones] = 1.0
    cur_c5 = _evaluate(h)
    best_h = h.copy()
    best_c5 = cur_c5

    max_iters = 200000
    init_temp = 0.05

    for it in range(max_iters):
        if time.time() - start_time > budget * 0.97:
            break

        # pick a random 1 and a random 0 and swap them
        zero_idxs = np.where(h == 0)[0]
        one_idxs = np.where(h == 1)[0]
        if zero_idxs.size == 0 or one_idxs.size == 0:
            break
        i0 = np.random.choice(zero_idxs)
        i1 = np.random.choice(one_idxs)

        h_new = h.copy()
        h_new[i0] = 1.0
        h_new[i1] = 0.0

        new_c5 = _evaluate(h_new)

        # temperature schedule (linear cooling)
        T = init_temp * (1.0 - it / max_iters)
        if new_c5 < cur_c5 or np.random.rand() < np.exp(-(new_c5 - cur_c5) / (T + 1e-12)):
            h = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()

        # occasional full restart to diversify search
        if it % 7000 == 0 and it > 0:
            if np.random.rand() < 0.2:
                h = np.zeros(N, dtype=float)
                ones = np.random.choice(N, size=N // 2, replace=False)
                h[ones] = 1.0
                cur_c5 = _evaluate(h)

    return best_h, best_c5, N

    best_h = _project(best_h, best_n * 0.5)
