def project_onto_capped_simplex(v, target, max_iter=100, tol=1e-9):
    lo = np.min(v) - 1.0  # ensure sum > target
    hi = np.max(v) + 1.0  # sum < target
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2, 0.0, 1.0)

import numpy as np
import time, random, math

def _compute_c5(h):
    """Return max_k ∫ h(x)(1-h(x+k))dx for a discrete step function."""
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project_onto_capped_simplex(v, target, iters=60, tol=1e-12):
    """
    Euclidean projection of v onto {x | 0<=x_i<=1, sum x_i = target}.
    Uses a simple binary search for the Lagrange multiplier.
    """
    lo = np.min(v) - 1.0          # low lambda => all entries hit 1
    hi = np.max(v) + 1.0          # high lambda => all entries hit 0
    for _ in range(iters):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
        if hi - lo < tol:
            break
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)

def _shift_array(a, k):
    """Shift array a by k positions with zero padding (k can be negative)."""
    n = a.shape[0]
    if k > 0:
        return np.concatenate((a[k:], np.zeros(k)))
    if k < 0:
        s = -k
        return np.concatenate((np.zeros(s), a[:n - s]))
    return a.copy()

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    margin = 5.0          # safety margin before hard timeout

    # ------------------------------------------------------------------
    # Choose number of sample points
    if "initial_h_values" in globals():
        init_src = np.array(initial_h_values, dtype=np.float64)
        N0 = init_src.size
    else:
        N0 = 0

    N = max(N0, 150)                 # enough points for fine tuning
    if N % 2:
        N += 1                       # make even (convenient)

    target_sum = N / 2.0

    # ------------------------------------------------------------------
    # Initialise h (either from the parent construction, or random)
    if N0:
        src_x = np.linspace(0.0, 2.0, N0, endpoint=False)
        tgt_x = np.linspace(0.0, 2.0, N, endpoint=False)
        h0 = np.interp(tgt_x, src_x, init_src, left=0.0, right=0.0)
    else:
        h0 = np.random.rand(N)

    h0 = _project_onto_capped_simplex(h0, target_sum)
    current_h = h0.copy()
    current_c5 = _compute_c5(current_h)

    best_h = current_h.copy()
    best_c5 = current_c5

    # ------------------------------------------------------------------
    # Parameters for the sub‑gradient descent
    base_alpha = 0.8          # initial step length (in the scale of h)
    itr = 0
    restart_interval = 2000   # how often to re‑initialise the current point
    perturb_prob = 0.015      # chance of a random perturbation each iteration

    # ------------------------------------------------------------------
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s - margin:
            break

        N = current_h.shape[0]
        dx = 2.0 / N

        # worst shift
        corr = np.correlate(current_h, 1.0 - current_h, mode="full")
        k_idx = int(np.argmax(corr))
        shift = k_idx - (N - 1)                     # integer lag

        # neighbour‑sum needed for the sub‑gradient
        if shift > 0:
            h_plus = np.concatenate((current_h[shift:], np.zeros(shift)))
            h_minus = np.concatenate((np.zeros(shift), current_h[:N - shift]))
        elif shift < 0:

        neigh = h_plus + h_minus

        cur = best_h.copy()
        cur_val = best_c5
        for s in range(steps):

            T = T0 * (T1 / T0) ** prog
            i, j = np.random.randint(N, size=2)
            if i == j:
                continue
            max_delta = min(0.12, cur[i], 1.0 - cur[j])
            if max_delta <= 0.0:
                continue
            delta = np.random.rand() * max_delta
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _project_onto_capped_simplex(cand, target_sum)
            cand_val = _compute_c5(cand)
            if cand_val < cur_val or np.random.rand() < math.exp(-(cand_val - cur_val) / max(T, 1e-12)):
                cur = cand
                cur_val = cand_val
                if cand_val < best_c5:
                    best_c5 = cand_val
                    best_h = cand.copy()

    # ------------------------------------------------------------------
    # Ensure the returned vector satisfies the sum constraint exactly
    best_h = _project_onto_capped_simplex(best_h, target_sum)
    final_c5 = _compute_c5(best_h)

    return best_h.tolist(), float(final_c5), len(best_h)
