def anneal_binary(h_start, time_budget, start_time, rng):
    n = len(h_start)
    best_h = h_start.copy()
    best_val = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_val = best_val

    # Determine temperature schedule: initial T = maybe 0.01? We'll compute as initial_assorted = initial best_val * 0.1
    T0 = best_val * 0.05 + 1e-4  # small but non-zero

    # iterations
    total_iter = 0
    while time.time() - start_time < time_budget:
        # propose swap
        ones = np.where(cur_h > 0.5)[0]
        zeros = np.where(cur_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_val = compute_c5(cur_h)
        # acceptance
        if new_val < cur_val:
            cur_val = new_val
            # keep change
        else:
            # revert with probability exp(-(new_val - cur_val)/T)
            # compute temperature decreasing over iterations
            T = T0 * (1 - (time.time() - start_time) / time_budget)
            if rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12)):
                cur_val = new_val
                # accept
            else:
                # revert swap
                cur_h[i] = 1.0
                cur_h[j] = 0.0
        if cur_val < best_val:
            best_val = cur_val
            best_h = cur_h.copy()
        total_iter += 1
    return best_h, best_val

import time
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------
def compute_c5(h):
    """C5 value for a discretised step function h (numpy array)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def corr_shift(h, s):
    """Return sum_i h[i]*(1-h[i+s]) for integer shift s (linear indexing)."""
    n = h.shape[0]
    if s >= 0:
        return np.dot(h[: n - s], 1.0 - h[s:])
    else:
        s = -s
        return np.dot(h[s:], 1.0 - h[: n - s])


def random_binary(n, rng):
    """Generate a random binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def round_to_binary(h):
    """Return a binary vector with the same length as h having the top n/2 entries set to 1."""
    n = h.shape[0]
    h_bin = np.zeros_like(h, dtype=float)
    # argpartition is O(n); we take the indices of the largest n/2 values
    idx = np.argpartition(h, -(n // 2))[-(n // 2) :]
    h_bin[idx] = 1.0
    return h_bin


def anneal_binary(h_start, time_budget, start_time, rng):
    """
    Simulated annealing on binary vectors.
    Swaps a 1 and a 0 at each step, accepts moves with Metropolis rule.
    """
    n = h_start.shape[0]
    cur = h_start.copy()
    cur_val = compute_c5(cur)
    best = cur.copy()
    best_val = cur_val

    # initial temperature – a small fraction of the current objective
    T0 = max(cur_val * 0.05, 1e-4)

    while time.time() - start_time < time_budget:
        # pick a random 1 position and a random 0 position
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # propose swap
        cur[i] = 0.0
        cur[j] = 1.0
        new_val = compute_c5(cur)

        if new_val < cur_val:
            cur_val = new_val
        else:
            # temperature decays linearly with elapsed fraction
            elapsed = time.time() - start_time
            frac = elapsed / time_budget
            T = T0 * (1.0 - frac)
            if rng.random() >= np.exp(-(new_val - cur_val) / max(T, 1e-12)):
                # reject – revert swap
                cur[i] = 1.0
                cur[j] = 0.0
                new_val = cur_val  # unchanged
            else:
                cur_val = new_val

        if cur_val < best_val:
            best_val = cur_val
            best = cur.copy()

    return best, best_val


def optimize_h(h_start, max_iter=800):
    """
    SLSQP optimisation of the continuous problem.
    Minimises the maximum overlap using an auxiliary variable t.
    """
    n = h_start.shape[0]
    dx = 2.0 / n

    # bounds for each h_i and the auxiliary t
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    # equality: integral constraint Σ h_i = n/2  (since dx cancels)
    eq_cons = {
        "type": "eq",
        "fun": lambda x, n=n: np.sum(x[:n]) - n / 2.0,
    }

    # inequality constraints: for each shift s,  t >= Σ h_i (1 - h_{i+s})
    ineq = []
    for s in range(-(n - 1), n):
        def fun(x, s=s, n=n):
            h = x[:n]
            t = x[n]
            return t - corr_shift(h, s)
        ineq.append({"type": "ineq", "fun": fun})

    constraints = [eq_cons] + ineq

    # initial auxiliary variable t set to current maximum correlation
    t0 = compute_c5(h_start)
    x0 = np.concatenate([h_start, [t0]])

    opts = {"maxiter": max_iter, "ftol": 1e-10, "disp": False}
    try:
        res = opt.minimize(
            lambda x, n=n: x[n],
            x0,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options=opts,
        )
    except Exception:
        res = None

    if res is None or not res.success:
        # fallback to the start point
        h_opt = np.clip(h_start, 0.0, 1.0)
        return h_opt, compute_c5(h_opt)

    h_opt = np.clip(res.x[:n], 0.0, 1.0)

    # enforce exact sum (tiny numerical drift)
    target = n / 2.0
    cur = h_opt.sum()
    if abs(cur - target) > 1e-8:
        diff = target - cur
        mask = (h_opt > 0.0) & (h_opt < 1.0)
        if mask.any():
            h_opt[mask] += diff / mask.sum()
            h_opt = np.clip(h_opt, 0.0, 1.0)
        else:
            h_opt *= target / cur

    return h_opt, compute_c5(h_opt)


def hill_climb(h, budget, start_time):
    """
    Simple greedy continuous hill climbing.
    Randomly moves a small amount of mass from one point to another.
    """
    n = h.shape[0]
    best_h = h.copy()
    best_val = compute_c5(best_h)
    rng = np.random.default_rng()

    while time.time() - start_time < budget:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # direction: move from i to j (i decreases, j increases)
        if best_h[i] > 0.0 and best_h[j] < 1.0:
            max_delta = min(best_h[i], 1.0 - best_h[j])
        elif best_h[j] > 0.0 and best_h[i] < 1.0:
            max_delta = min(best_h[j], 1.0 - best_h[i])
            i, j = j, i
        else:
            continue

        if max_delta <= 0.0:
            continue

        delta = max_delta * rng.random()
        cand = best_h.copy()
        cand[i] -= delta
        cand[j] += delta
        val = compute_c5(cand)

        if val < best_val - 1e-9:
            best_h, best_val = cand, val

    return best_h, best_val

    """
    Returns a step function (as a list), the corresponding C5 bound,
    and the number of discretisation points.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size; prefer the size of the provided initial solution
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.shape[0]
        # ensure we have an even number of points
        if n % 2 == 1:
            n -= 1
            h0 = h0[:n]
    else:
        n = 180
        h0 = None

    target_sum = n / 2.0

    best_h = None
    best_val = np.inf
    best_n = n

    # ------------------------------------------------------------------
    # 1. Initialise candidates
    # ------------------------------------------------------------------
    candidates = []

    if h0 is not None:
        # binary rounding of the provided (possibly continuous) start
        h_bin = round_to_binary(h0)
        candidates.append(h_bin)

    # a few pure random starts
    for _ in range(3):
        candidates.append(random_binary(n, rng))

    # ------------------------------------------------------------------
    # 2. Quick annealing on each candidate
    # ------------------------------------------------------------------
    for h_start in candidates:
        elapsed = time.time() - start
        remain = budget_s - elapsed
        if remain < 5.0:
            break

        # allocate a modest slice for annealing (max 12 s)
        slice_budget = min(12.0, remain * 0.4)

        h_opt, val_opt = anneal_binary(h_start, slice_budget, start, rng)

        if val_opt < best_val:
            best_h, best_val = h_opt, val_opt

    # ------------------------------------------------------------------
    # 3. Additional random restarts until time runs low
    # ------------------------------------------------------------------
    while True:
        elapsed = time.time() - start
        remain = budget_s - elapsed
        if remain < 8.0:
            break

        h_rand = random_binary(n, rng)
        # use a short annealing window (≈6 s)
        slice_budget = min(6.0, remain * 0.3)
        h_opt, val_opt = anneal_binary(h_rand, slice_budget, start, rng)

        if val_opt < best_val:
            best_h, best_val = h_opt, val_opt

    # ------------------------------------------------------------------
    # 4. Continuous refinement with SLSQP (if time permits)
    # ------------------------------------------------------------------
    elapsed = time.time() - start
    remain = budget_s - elapsed
    if remain > 30.0 and best_h is not None:
        # give SLSQP a generous iteration budget
        iters = max(400, int(remain // 0.06))  # rough 60 ms per iter
        h_cont, val_cont = optimize_h(best_h, max_iter=iters)
        if val_cont < best_val:
            best_h, best_val = h_cont, val_cont

    # ------------------------------------------------------------------
    # 5. Final local continuous hill‑climb (if any time left)
    # ------------------------------------------------------------------
    elapsed = time.time() - start
    remain = budget_s - elapsed
    if remain > 2.0 and best_h is not None:
        h_final, val_final = hill_climb(best_h, remain * 0.9, start)
        if val_final < best_val:
            best_h, best_val = h_final, val_final

    # ------------------------------------------------------------------
    # 6. Safety: enforce exact integral and bounds
    # ------------------------------------------------------------------
    if best_h is None:
        best_n = n
        best_h = np.full(best_n, 0.5)
        best_val = compute_c5(best_h)
    else:
        best_n = best_h.shape[0]
        best_h = np.clip(best_h, 0.0, 1.0)
        # renormalise sum exactly
        cur_sum = best_h.sum()
        if abs(cur_sum - target_sum) > 1e-12:
            best_h *= target_sum / cur_sum

        best_val = compute_c5(best_h)

    # Return a plain Python list as required
    return (best_h.tolist(), float(best_val), int(best_n))
