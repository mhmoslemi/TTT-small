def _project_to_sum(v, target):
    # v: ndarray, shape (n,)
    w = np.clip(v, 0.0, 1.0)
    diff = target - w.sum()
    # iterative water filling
    max_iter = 1000
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # free indices: those not at bound
        free = np.where((w > 0) & (w < 1))[0]
        if free.size == 0:
            # distribute diff proportionally (but all at bounds)
            # adjust by adding diff/(len(w)) equally and then clip again
            w += diff / w.size
            w = np.clip(w, 0.0, 1.0)
            diff = target - w.sum()
            continue
        inc = diff / free.size
        # add inc to free cells, but keep within bounds
        w[free] += inc
        # clip overshoot
        over_low = w < 0
        over_high = w > 1
        if np.any(over_low):
            w[over_low] = 0.0
        if np.any(over_high):
            w[over_high] = 1.0
        diff = target - w.sum()
    # Final rectify small error
    if abs(diff) > 1e-8:
        # distribute remaining diff on any variable within bounds
        # adjust by adding diff to the variable with most slack
        # just adjust the first free index
        idx = np.argmin(w) if diff < 0 else np.argmax(w)
        w[idx] += diff
        w[idx] = np.clip(w[idx], 0.0, 1.0)
    return w

def _project_to_sum(v, target):
    w = np.clip(v, 0.0, 1.0)
    total = w.sum()
    if total > target:
        # need to reduce excess = total - target
        excess = total - target
        # sort descending indices
        idx = np.argsort(-w)  # descending
        for i in idx:
            cap = w[i] # can reduce up to w[i] (to zero)
            reduce = min(cap, excess)
            w[i] -= reduce
            excess -= reduce
            if excess <= 1e-12:
                break
    elif total < target:
        deficit = target - total
        idx = np.argsort(w)  # ascending
        for i in idx:
            cap = 1 - w[i]
            inc = min(cap, deficit)
            w[i] += inc
            deficit -= inc
            if deficit <= 1e-12:
                break
    return w

def _project_to_sum(v, target):
    w = np.clip(v, 0.0, 1.0)
    total = w.sum()
    if total > target:
        excess = total - target
        # reduce from largest values
        idx = np.argsort(-w)  # descending
        for i in idx:
            if excess <= 1e-12:
                break
            cap = w[i]  # amount we can reduce to zero
            reduce = min(cap, excess)
            w[i] -= reduce
            excess -= reduce
    elif total < target:
        deficit = target - total
        idx = np.argsort(w)  # ascending
        for i in idx:
            if deficit <= 1e-12:
                break
            cap = 1.0 - w[i]  # amount we can increase
            inc = min(cap, deficit)
            w[i] += inc
            deficit -= inc
    return w
```

def hill_climb(h, max_iters, step):
    n = h.shape[0]
    cur = h.copy()
    cur_c5 = _c5_value(cur)
    for it in range(max_iters):
        i,j = rng.integers(0,n,size=2)
        lower = max(-cur[i], -(1 - cur[j]))
        upper = min(1 - cur[i], cur[j])
        if upper <= lower:
            continue
        delta = rng.uniform(max(lower, -step), min(upper, step))
        new = cur.copy()
        new[i] += delta
        new[j] -= delta
        new_c5 = _c5_value(new)
        if new_c5 < cur_c5:
            cur = new
            cur_c5 = new_c5
    return cur, cur_c5

import time, math, numpy as np
from scipy.optimize import differential_evolution, minimize, LinearConstraint

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project a vector onto the box [0,1]^n while forcing the sum to be exactly target.
    Implements a simple water‑filling algorithm that is O(n log n).
    """
    w = np.clip(v, 0.0, 1.0)
    total = w.sum()
    if total > target:                # need to reduce some entries
        excess = total - target
        idx = np.argsort(-w)          # largest first
        for i in idx:
            if excess <= 1e-12:
                break
            cap = w[i]                # can drop at most this much
            dec = min(cap, excess)
            w[i] -= dec
            excess -= dec
    elif total < target:              # need to increase some entries
        deficit = target - total
        idx = np.argsort(w)           # smallest first
        for i in idx:
            if deficit <= 1e-12:
                break
            cap = 1.0 - w[i]          # room to raise
            inc = min(cap, deficit)
            w[i] += inc
            deficit -= inc
    # tiny numerical drift correction
    diff = target - w.sum()
    if abs(diff) > 1e-10:
        # distribute the remaining tiny piece onto any free coordinate
        free = np.where((w > 0) & (w < 1))[0]
        if free.size == 0:
            free = np.arange(w.size)
        w[free[0]] = np.clip(w[free[0]] + diff, 0.0, 1.0)
    return w


def _interpolate_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a feasible vector of length n derived from initial_h_values if possible."""
    try:
        init = np.array(initial_h_values, dtype=float)   # provided by harness
        if init.shape[0] != n:
            x_old = np.linspace(0.0, 2.0, init.shape[0], endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            init = np.interp(x_new, x_old, init)
        init = _project_to_sum(init, n / 2.0)
        return init
    except Exception:
        # fallback: random feasible vector
        v = rng.random(n)
        return _project_to_sum(v, n / 2.0)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step function h on [0,2] that minimizes
        C5 = max_k ∫ h(x)(1‑h(x+k)) dx .
    Returns (h_values, c5_bound, n_points).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    best_h, best_c5 = None, float("inf")
    # try a few problem sizes, stop when time runs out
    for n in (80, 100, 120):
        # budget check
        if time.time() - start > budget_s * 0.97:
            break

        target_sum = n / 2.0
        bounds = [(0.0, 1.0)] * n

        # ------------------------------------------------------------------
        # 1. Build an initial population for DE (feasible vectors)
        # ------------------------------------------------------------------
        popsize = 20
        pop = np.empty((popsize, n))
        for i in range(popsize):
            pop[i] = _project_to_sum(rng.random(n), target_sum)
        # replace the first individual with the harness‑provided construction,
        # interpolated to the current length
        try:
            pop[0] = _interpolate_initial(n, rng)
        except Exception:
            pass

        # ------------------------------------------------------------------
        # 2. Differential Evolution (global search)
        # ------------------------------------------------------------------
        # objective works on the projected vector
        def obj(x):
            return _c5_value(_project_to_sum(x, target_sum))

        # allocate a modest number of generations proportional to remaining time
        remaining = budget_s - (time.time() - start)
        maxiter = max(10, int(min(250, remaining / budget_s * 250)))

        result = differential_evolution(
            obj,
            bounds,
            init=pop,
            maxiter=maxiter,
            polish=False,
            seed=seed,
            updating="deferred",
            disp=False,
        )
        cand = _project_to_sum(result.x, target_sum)
        cand_c5 = _c5_value(cand)
        if cand_c5 < best_c5:
            best_c5, best_h = cand_c5, cand.copy()

        # ------------------------------------------------------------------
        # 3. Local refinement – greedy mass swaps (hill climbing)
        # ------------------------------------------------------------------
        cur = best_h.copy()
        cur_c5 = best_c5
        step = 0.20
        max_iters = 20000
        for it in range(max_iters):
            if time.time() - start > budget_s * 0.97:
                break
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # feasible delta that keeps the vector in [0,1]
            low = max(-cur[i], -(1.0 - cur[j]))
            high = min(1.0 - cur[i], cur[j])
            if high <= low:
                continue
            low = max(low, -step)
            high = min(high, step)
            if high <= low:
                continue
            delta = rng.uniform(low, high)
            new = cur.copy()
            new[i] += delta
            new[j] -= delta
            new_c5 = _c5_value(new)
            # accept only improving moves (pure greedy)
            if new_c5 < cur_c5:
                cur, cur_c5 = new, new_c5
                if new_c5 < best_c5:
                    best_c5, best_h = new_c5, new.copy()
            # slowly reduce the maximal step size
            if (it + 1) % 2000 == 0 and step > 1e-4:
                step *= 0.85

        # ------------------------------------------------------------------
        # 4. SLSQP polishing (optional quick local optimisation)
        # ------------------------------------------------------------------
        elapsed = time.time() - start
        if elapsed < budget_s * 0.85:
            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},)
            bounds_slsqp = [(0.0, 1.0)] * n
            res = minimize(
                _c5_value,
                best_h,
                method="SLSQP",
                bounds=bounds_slsqp,
                constraints=cons,
                options={"maxiter": 300, "ftol": 1e-9, "disp": False},
            )
            if res.success:
                final = _project_to_sum(res.x, target_sum)
                final_c5 = _c5_value(final)
                if final_c5 < best_c5:
                    best_c5, best_h = final_c5, final.copy()

    # ----------------------------------------------------------------------
    # 5. Final sanity check and return
    # ----------------------------------------------------------------------
    if best_h is None:
        # fallback: constant 0.5 function
        n = 80
        best_h = np.full(n, 0.5)
        best_c5 = _c5_value(best_h)

    best_h = _project_to_sum(best_h, best_h.shape[0] / 2.0)
    return best_h.tolist(), float(best_c5), int(best_h.shape[0])
