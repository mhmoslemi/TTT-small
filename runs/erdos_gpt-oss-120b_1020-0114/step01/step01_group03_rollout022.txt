def grad_smooth_max(h, alpha):
    n = len(h)
    dx = 2.0 / n
    # compute autocorrelation r = correlate(h, h)
    r = np.correlate(h, h, mode='full')
    # compute softmax weights for -r (minimizing max of 1 - r*dx)

    corr2 = np.correlate(q[::-1], h, mode='full')
    term2 = corr2[start:start + n]
    grad = -dx * (term1 + term2)   # shape n
    return grad

def project_onto_sum_and_bounds(x, target):
    # We'll use iterative water-filling:
    # Clip to [0,1], then adjust sum by distributing diff among non-bound entries.
    x = np.clip(x, 0.0, 1.0)
    # Tolerance
    for _ in range(5):
        s = x.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # indices not at bounds
        free = (x > 0.0 + 1e-12) & (x < 1.0 - 1e-12)
        num_free = np.count_nonzero(free)
        if num_free == 0:
            # cannot adjust; break
            break
        delta = diff / num_free
        x[free] += delta
        # Clip again
        x = np.clip(x, 0.0, 1.0)
    # If still off by some leftover, distribute uniformly (may exceed bounds slightly)
    s = x.sum()
    if abs(s - target) > 1e-9:
        # apply scaling factor
        factor = target / s if s > 0 else 0
        x = np.clip(x * factor, 0.0, 1.0)
        # final clip and maybe repeat
        # (not perfect but should be close)
    return x

    # Equivalent to find λ such that x_i = min(1, max(0, y_i - λ)).

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def compute_c5(h):
    """Exact C5 value for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)


def upsample_h(old_h, new_n):
    """Linear interpolation to a finer grid while preserving the integral."""
    old_n = old_h.shape[0]
    if old_n == new_n:
        return old_h.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    # enforce integral
    target = new_n / 2.0
    cur = new_h.sum()
    if cur > 0:
        new_h *= target / cur
    new_h = np.clip(new_h, 0.0, 1.0)
    # final tiny correction (rare)
    cur = new_h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        new_h += diff / new_n
        new_h = np.clip(new_h, 0.0, 1.0)
        cur = new_h.sum()
        if cur > 0:
            new_h *= target / cur
    return new_h


def project_onto_sum_and_bounds(y, target):
    """
    Euclidean projection of y onto the set
        {x | 0 <= x_i <= 1 , sum x_i = target }.
    Implemented by a simple bisection on the Lagrange multiplier λ
    for the box‑constrained simplex.
    """
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)


def grad_smooth_max(h, alpha):
    """
    Gradient of the smooth approximation of the max overlap.
    Uses the soft‑max on the autocorrelation r = h⋆h.
    """
    n = h.shape[0]
    dx = 2.0 / n

    # autocorrelation r_k = Σ_i h_i h_{i+k}
    r = np.correlate(h, h, mode="full")          # length 2n-1

    # soft‑max weights for minimizing the maximum (i.e. maximise min r)
    shift = r.min()
    w = np.exp(-alpha * (r - shift))
    q = w / w.sum()                              # length 2n-1

    # term1_j = Σ_k q_k * h_{j+k}
    corr1 = np.correlate(q, h, mode="full")
    start = n - 1
    term1 = corr1[start:start + n]

    # term2_j = Σ_k q_k * h_{j-k}
    corr2 = np.correlate(q[::-1], h, mode="full")
    term2 = corr2[start:start + n]

    grad = -dx * (term1 + term2)
    return grad


def local_swap_refine(h, target, time_left, max_steps=3000):
    """Simple hill‑climbing: move a little mass between two points."""
    n = h.size
    best = h.copy()
    best_val = compute_c5(best)
    steps = max_steps
    while steps > 0 and time_left() > 0:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            steps -= 1
            continue
        eps = np.random.uniform(0.005, 0.02)
        eps = min(eps, best[i], 1.0 - best[j])
        if eps <= 0:
            steps -= 1
            continue
        cand = best.copy()
        cand[i] -= eps
        cand[j] += eps
        val = compute_c5(cand)
        if val < best_val:
            best, best_val = cand, val
        steps -= 1
    return best

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    elapsed = lambda: time.time() - start

    # ------------------------------------------------------------------
    # Starting construction (use the provided one if present)
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        # simple sinusoid on a 200‑point grid
        n0 = 200
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        h0 = (1.0 + np.cos(np.pi * x)) * 0.5
    n0 = h0.size
    # enforce constraints on the raw start
    target0 = n0 / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_onto_sum_and_bounds(h0, target0)

    # ------------------------------------------------------------------
    # Resolution schedule (200 → 400 → 800)
    # ------------------------------------------------------------------
    schedule = [n0, min(400, max(200, n0 * 2)), 800]
    schedule = [s for s in schedule if s <= 1000]
    best_h = None
    best_val = float("inf")

    for n in schedule:
        if elapsed() > budget_s * 0.94:
            break

        # up‑sample current best (or the initial) to the new grid
        if best_h is not None and best_h.size != n:
            h = upsample_h(best_h, n)
        elif h0.size != n:
            h = upsample_h(h0, n)
        else:
            h = h0.copy()

        target = n / 2.0
        h = np.clip(h, 0.0, 1.0)
        h = project_onto_sum_and_bounds(h, target)

        # optimisation parameters
        alpha = 1.0               # smoothing factor
        step = 0.6                # initial gradient step
        decay = 0.9995            # step‑size decay
        alpha_inc = 1.03          # gradual sharpening
        max_iters = 20000

        for it in range(max_iters):
            if elapsed() > budget_s * 0.94:
                break

            # gradient of smoothed max
            g = grad_smooth_max(h, alpha)

            # simple gradient step + projection
            h_new = h - step * g
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_onto_sum_and_bounds(h_new, target)

            # evaluate true C5 (rarely, to keep cheap)
            if it % 250 == 0:
                val = compute_c5(h_new)
                if val < best_val:
                    best_val = val
                    best_h = h_new.copy()

            # acceptance: keep improvement, otherwise keep old (no SA)
            if compute_c5(h_new) <= compute_c5(h):
                h = h_new

            # schedule updates
            step *= decay
            if (it + 1) % 200 == 0:
                alpha = min(alpha * alpha_inc, 300.0)

            # occasional random jitter to escape plateaus
            if (it + 1) % 1000 == 0:
                jitter = np.random.uniform(-0.02, 0.02, size=n)
                h = np.clip(h + jitter, 0.0, 1.0)
                h = project_onto_sum_and_bounds(h, target)

        # ------------------------------------------------------------------
        # Local fine‑tuning via mass swaps
        # ------------------------------------------------------------------
        if best_h is None:
            best_h = h.copy()
        best_h = local_swap_refine(
            best_h,
            target,
            lambda: budget_s - elapsed(),
            max_steps=3000,
        )
        cur_val = compute_c5(best_h)
        if cur_val < best_val:
            best_val = cur_val

    # final projection & evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_sum_and_bounds(best_h, best_h.size / 2.0)
    final_val = compute_c5(best_h)

    return (best_h, float(final_val), int(best_h.size))
