def compute_overlap_fft(h):
    n = h.shape[0]
    L = int(2 ** np.ceil(np.log2(2*n-1)))
    H = np.fft.fft(h, n=L)
    Hc = np.fft.fft(1-h, n=L)
    corr = np.fft.ifft(H * np.conj(Hc))[:2*n-1].real
    # rounding errors
    return np.max(corr) * (2.0 / n)

import time
import numpy as np

def compute_corr_full(h):
    n = h.shape[0]
    L = int(2 ** np.ceil(np.log2(2*n-1)))
    H = np.fft.fft(h, n=L)
    Hc = np.fft.fft(1-h, n=L)
    corr = np.fft.ifft(H * np.conj(Hc))[:2*n-1].real
    return corr * (2.0 / n)   # multiply dx

def compute_c5(h):
    return np.max(compute_corr_full(h))

def get_shifted_arrays(h, s):
    n = h.shape[0]
    if s >= 0:
        shift_plus = np.concatenate((np.zeros(s), h[:n-s]))
        shift_minus = np.concatenate((h[s:], np.zeros(s)))
    else:
        ss = -s
        shift_plus = np.concatenate((h[ss:], np.zeros(ss)))
        shift_minus = np.concatenate((np.zeros(ss), h[:n-ss]))
    return shift_plus, shift_minus

def gradient_descent(h, target_sum, time_deadline, start_lr=0.2, max_iter=20000):
    n = h.shape[0]
    lr = start_lr
    best_h = h.copy()
    best_val = compute_c5(h)
    while max_iter>0:
        max_iter -= 1
        if time.time() > time_deadline:
            break
        corr = compute_corr_full(h)
        idx = np.argmax(corr)
        s = idx - (n-1)
        shift_plus, shift_minus = get_shifted_arrays(h, s)
        delta = (shift_plus + shift_minus) - 1.0
        if np.all(np.abs(delta) < 1e-12):
            # no direction
            break
        # compute step
        step = lr * delta
        # Apply step with clipping
        h_new = h + step
        # Clip to [0,1]
        np.clip(h_new, 0.0, 1.0, out=h_new)
        # Project onto sum constraint: scaling
        cur_sum = h_new.sum()
        if cur_sum != 0:
            h_new *= (target_sum / cur_sum)
        np.clip(h_new, 0.0, 1.0, out=h_new)
        # Evaluate
        new_val = compute_c5(h_new)
        cur_val = compute_c5(h)
        # accept if improvement

            h = h_new
            best_h = h.copy()
            best_val = new_val
            lr = min(start_lr, lr * 1.05)  # slightly increase lr
        else:
            lr *= 0.5  # reduce lr
            if lr < 1e-6:

    return best_h, best_val

def random_hill_climb(h, target_sum, time_deadline, steps=5000):

def upsample_h(old_h, new_n):
    old_n = old_h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    target_sum = new_n / 2.0
    cur_sum = new_h.sum()
    if cur_sum > 0:
        new_h *= (target_sum / cur_sum)
    np.clip(new_h, 0.0, 1.0, out=new_h)
    # final correction
    cur_sum = new_h.sum()
    diff = target_sum - cur_sum
    if abs(diff) > 1e-12:
        new_h += diff / new_n
        np.clip(new_h, 0.0, 1.0, out=new_h)
        cur_sum = new_h.sum()
        if cur_sum > 0:
            new_h *= (target_sum / cur_sum)
    return new_h

import time
import numpy as np

# -----------------------------------------------------------
# Correlation helpers (fast via FFT, matching np.correlate)
# -----------------------------------------------------------
def _full_corr(h: np.ndarray) -> np.ndarray:
    """Return the full correlation np.correlate(h, 1-h, mode='full') * dx."""
    n = h.size
    # length for linear convolution
    L = 1 << int(np.ceil(np.log2(2 * n - 1)))
    # FFT of h
    H = np.fft.fft(h, n=L)
    # FFT of reversed (1-h) – this yields correlation after reversing the result
    G = np.fft.fft((1 - h)[::-1], n=L)
    conv = np.fft.ifft(H * G)[: 2 * n - 1].real
    corr = conv[::-1] * (2.0 / n)          # multiply by dx, reverse to match np.correlate order
    return corr


def compute_c5(h: np.ndarray) -> float:
    """Exact C5 value for a discrete step function."""
    return float(np.max(_full_corr(h)))


def _shift_arrays(h: np.ndarray, s: int):
    """Return (h shifted forward by s, h shifted backward by s) with zero padding."""
    n = h.size
    if s >= 0:
        plus = np.concatenate((np.zeros(s), h[: n - s]))
        minus = np.concatenate((h[s:], np.zeros(s)))
    else:
        ss = -s
        plus = np.concatenate((h[ss:], np.zeros(ss)))
        minus = np.concatenate((np.zeros(ss), h[: n - ss]))
    return plus, minus


# -----------------------------------------------------------
# Simple up‑sampling preserving the integral
# -----------------------------------------------------------
def upsample_h(old_h: np.ndarray, new_n: int) -> np.ndarray:
    old_n = old_h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)

    target = new_n / 2.0
    cur = new_h.sum()
    if cur > 0:
        new_h *= target / cur
    np.clip(new_h, 0.0, 1.0, out=new_h)

    # final correction for tiny sum error
    cur = new_h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        new_h += diff / new_n
        np.clip(new_h, 0.0, 1.0, out=new_h)
        cur = new_h.sum()
        if cur > 0:
            new_h *= target / cur
    return new_h


# -----------------------------------------------------------
# Optional symmetry (mirror about the centre of [0,2])
# -----------------------------------------------------------
def symmetrize(h: np.ndarray) -> np.ndarray:
    return (h + h[::-1]) * 0.5


# -----------------------------------------------------------
# Gradient–descent style optimisation on the worst shift
# -----------------------------------------------------------
def _grad_opt(h: np.ndarray, target_sum: float, deadline: float,
              start_lr: float = 0.2, max_iter: int = 20000):
    n = h.size
    lr = start_lr
    best_h = h.copy()
    best_val = compute_c5(best_h)

    for _ in range(max_iter):
        if time.time() > deadline:
            break

        corr = _full_corr(best_h)
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)

        plus, minus = _shift_arrays(best_h, shift)
        delta = (plus + minus) - 1.0          # -gradient of the worst overlap
        if np.all(np.abs(delta) < 1e-12):
            break

        step = lr * delta
        new_h = best_h + step
        np.clip(new_h, 0.0, 1.0, out=new_h)

        cur_sum = new_h.sum()
        if cur_sum > 0:
            new_h *= target_sum / cur_sum
        np.clip(new_h, 0.0, 1.0, out=new_h)

        new_val = compute_c5(new_h)
        cur_val = compute_c5(best_h)

        if new_val < cur_val - 1e-12:
            best_h = new_h
            best_val = new_val
            lr = min(start_lr, lr * 1.05)        # slight boost after improvement
        else:
            lr *= 0.5
            if lr < 1e-6:
                break
    return best_h, best_val


# -----------------------------------------------------------
# Random hill‑climbing fine‑tuner
# -----------------------------------------------------------
def _hill_climb(h: np.ndarray, target_sum: float, deadline: float,
                steps: int = 20000):
    n = h.size
    best_h = h.copy()
    best_val = compute_c5(best_h)

    for _ in range(steps):
        if time.time() > deadline:
            break
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        eps = np.random.uniform(0.001, 0.05)
        eps = min(eps, best_h[i], 1.0 - best_h[j])
        if eps <= 0:
            continue
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        val = compute_c5(cand)
        if val < best_val:
            best_h, best_val = cand, val
    return best_h, best_val

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
