import cvxpy as cp

def milp_optimize(n, target_sum, max_t=None, time_limit=30):
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == target_sum, t >= 0]

    # produce dictionary of z variables (binary)
    # For each shift s, create list of z for that shift
    for s in range(1, n):
        # create vector of z_i_s for i in 0..n-s-1
        sz = n - s
        z = cp.Variable(sz, boolean=True)
        # constraints linking z and h:
        constraints += [z <= h[:sz]]
        constraints += [z <= h[s:]]
        constraints += [z >= h[:sz] + h[s:] - 1]

        # O_s = sum(h[:sz]) - sum(z)
        constraints.append(cp.sum(h[:sz]) - cp.sum(z) <= t)
        # optionally O_s >= 0 automatically due to sum(z) <= sum(h), but fine.

        # store maybe
    if max_t is not None:
        constraints.append(t <= max_t)

    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)

    # solve with time limit via solver options if possible
    try:
        prob.solve(solver=cp.GLPK_MI, glpk={'tm_lim': int(time_limit*1000)})
    except Exception:
        try:
            prob.solve(solver=cp.CBC, timeLimit=time_limit)
        except Exception:
            prob.solve(solver=cp.ECOS_BB) # maybe slower
    # after solve, retrieve
    if h.value is None:
        return None, None
    h_sol = np.round(h.value).astype(float) # round to 0/1
    t_val = t.value
    return h_sol, float(t_val)
```

def _project_to_constraints(v, target_sum):
    # Clip to [0,1]
    v = np.clip(v, 0.0, 1.0)
    # Adjust sum to target
    diff = target_sum - v.sum()
    tolerance = 1e-12
    max_iter = 10
    for _ in range(max_iter):
        if abs(diff) < tolerance:
            break
        # Find free indices
        if diff > 0:
            mask = v < 1.0 - 1e-12
        else:
            mask = v > 1e-12
        free = np.where(mask)[0]
        if free.size == 0:
            # no free space, break
            break
        # Amount to distribute equally
        delta = diff / free.size
        # Ensure we don't overshoot bounds
        v[free] += delta
        # Clip again
        v = np.clip(v, 0.0, 1.0)
        diff = target_sum - v.sum()
    # final clip
    v = np.clip(v, 0.0, 1.0)
    return v

import time, math
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Core utilities
# ------------------------------------------------------------
def _compute_c5(h: np.ndarray) -> float:
    """Return the discretised C5 value for a step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)


def _project_to_constraints(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1] and adjust the sum to exactly target_sum."""
    v = np.clip(v, 0.0, 1.0)
    diff = target_sum - v.sum()
    # iterate a few times; each pass distributes the remaining deficit
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v < 1.0 - 1e-12
        else:
            mask = v > 1e-12
        free = np.where(mask)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
        diff = target_sum - v.sum()
    return np.clip(v, 0.0, 1.0)

# Soft‑max gradient descent on the full set of shifts
# ------------------------------------------------------------
def _softmax_descent(
    h0: np.ndarray,
    target_sum: float,
    time_limit: float,
    best_h: np.ndarray,
    best_c5: float,
):
    """Run many short soft‑max PGD runs, updating best_h / best_c5."""
    n = h0.shape[0]
    start = time.time()
    while time.time() - start < time_limit:
        # random start or perturb current best
        if np.random.rand() < 0.3 and best_h is not None:
            h = best_h.copy()
            h += 0.02 * (np.random.rand(n) - 0.5)
        else:
            h = np.random.rand(n)
        h = _project_to_constraints(h, target_sum)

        # hyper‑parameters (will slowly anneal)
        eta = 0.4
        alpha = 5.0
        # inner iteration budget limited by remaining time
        inner_iters = 2000
        for _ in range(inner_iters):
            if time.time() - start >= time_limit:
                break

            # ----- compute overlaps for all shifts -----
            # positive shifts (including 0)
            O_pos = np.empty(n)
            O_pos[0] = np.dot(h, 1.0 - h)
            # negative shifts (store O_neg[0] unused)
            O_neg = np.empty(n)
            O_neg[0] = 0.0
            for s in range(1, n):
                O_pos[s] = np.dot(h[: n - s], 1.0 - h[s:])
                O_neg[s] = np.dot(h[s:], 1.0 - h[: n - s])

            O_all = np.concatenate([O_neg[1:][::-1], O_pos])  # length 2n‑1
            max_O = O_all.max()
            c5 = max_O * (2.0 / n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                if best_c5 <= 0.3808:
                    return best_h, best_c5

            # ----- soft‑max weights -----
            # stability: subtract max before exponentiation
            mexp = np.max(alpha * O_all)
            w = np.exp(alpha * O_all - mexp)
            w /= w.sum()
            w0 = w[n - 1]
            w_pos = w[n:]               # size n‑1 (shifts 1..n‑1)
            w_neg = w[: n - 1][::-1]    # also size n‑1

    t0 = _compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    constraints = [
        {"type": "eq", "fun": lambda x: np.sum(x[:n]) - target},
    ]

    # one inequality per shift (including negative)
    for s in range(-(n - 1), n):
        def make_fun(sft):
            def fun(x, sft=sft):
                h = x[:n]
                t = x[-1]
                if sft >= 0:
                    val = np.dot(h[: n - sft], 1.0 - h[sft:])
                else:
                    ss = -sft
                    val = np.dot(h[ss:], 1.0 - h[: n - ss])
                return t - val
            return fun

        constraints.append({"type": "ineq", "fun": make_fun(s)})

    opts = {"maxiter": max(200, int(time_limit * 15)), "ftol": 1e-12, "disp": False}
    res = minimize(lambda x: x[-1], x0, method="SLSQP", bounds=bounds,
                   constraints=constraints, options=opts)

    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    diff = target - h_opt.sum()
    if abs(diff) > 1e-10:
        free = np.where((h_opt > 1e-12) & (h_opt < 1.0 - 1e-12))[0]
        if free.size:
            h_opt[free] += diff / free.size
            h_opt = np.clip(h_opt, 0.0, 1.0)
    return h_opt, _compute_c5(h_opt)


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Optimise a step function for the Erdős C5 constant."""
    np.random.seed(seed)
    start_all = time.time()

    # --------------------------------------------------------
    # 0. initialise problem size (use parent if available)
    # --------------------------------------------------------
    if "initial_h_values" in globals():
        h_start = np.array(initial_h_values, dtype=float).copy()
        n = h_start.shape[0]
        target_sum = n / 2.0
        h_start = _project_to_constraints(h_start, target_sum)
    else:
        n = 45
        target_sum = n / 2.0
        h_start = np.full(n, target_sum / n)
        h_start = _project_to_constraints(h_start, target_sum)

    best_h = h_start.copy()
    best_c5 = _compute_c5(best_h)

    # --------------------------------------------------------
    # 1. Soft‑max projected gradient descent (majority of budget)
    # --------------------------------------------------------
    time_for_pg = budget_s * 0.75
    best_h, best_c5 = _softmax_descent(
        h_start, target_sum, time_for_pg, best_h, best_c5
    )

    # --------------------------------------------------------
    # 2. Optional SLSQP refinement with remaining time
    # --------------------------------------------------------
    time_remaining = budget_s - (time.time() - start_all)
    if time_remaining > 5.0:
        h_ref, c5_ref = _slsqp_optimize(best_h, time_remaining)
        if c5_ref < best_c5:
            best_h, best_c5 = h_ref, c5_ref

    # --------------------------------------------------------
    # 3. Final clean‑up (ensure constraints exactly)
    # --------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_to_constraints(best_h, target_sum)
    final_c5 = _compute_c5(best_h)

    return best_h, float(final_c5), int(n)
