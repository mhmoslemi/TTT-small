def compute_corr_and_shift(h):
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_idx = np.argmax(corr)
    max_val = corr[max_idx]
    s = max_idx - (N - 1)  # shift amount
    return max_val, s

def gradient_for_shift(h, s):
    N = h.size
    g = np.ones(N)
    if s>=0:
        g[:N-s] -= h[s:]
        g[s:] -= h[:N-s]
    else:
        s2 = -s
        g[s2:] -= h[:N-s2]
        g[:N-s2] -= h[s2:]
    return g

def anneal(h, time_budget, start_time):
    N = len(h)
    target_sum = N/2.0
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_c5 = best_c5
    rng = np.random.default_rng(12345)
    T0 = 1e-3
    # schedule: exponent decay based on elapsed time fraction.
    while time.time() - start_time < time_budget:
        elapsed = time.time() - start_time
        frac = elapsed / time_budget
        T = T0 * (1 - frac)  # linear cooling
        # propose random move
        i, j = rng.integers(0,N, size=2)
        if i==j:
            continue
        # compute max allowable delta
        max_delta = 0.0
        if cur_h[i] > 0 and cur_h[j] < 1:
            max_delta = min(cur_h[i], 1-cur_h[j])
        elif cur_h[j] > 0 and cur_h[i] < 1:
            max_delta = min(cur_h[j], 1-cur_h[i])
            i, j = j, i
        else:
            continue
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random()
        # create candidate
        cand = cur_h.copy()
        cand[i] -= delta
        cand[j] += delta
        # compute c5 (use fast compute_c5)
        cand_c5 = compute_c5(cand)
        dE = cand_c5 - cur_c5
        if dE < 0 or rng.random() < np.exp(-dE / (T+1e-12)):
            cur_h, cur_c5 = cand, cand_c5
            if cur_c5 < best_c5:
                best_h, best_c5 = cur_h.copy(), cur_c5
    return best_h, best_c5

def project_h(v, target):
    v = np.clip(v, 0.0, 1.0)
    # iterative adjustment (like water-filling)
    for _ in range(10):
        cur = v.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        mask = (v > 0.0) & (v < 1.0)
        if mask.any():
            v[mask] += diff / mask.sum()
            v = np.clip(v, 0.0, 1.0)
        else:
            # all at bounds: distribute proportionally
            v = np.clip(v + diff / len(v), 0.0, 1.0)
    return v

    best_val = compute_c5(best)

            cand = cur - step * g

                cur_val = cand_val

                break

        it += 1
    return best, best_val

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h and (1-h) with zero padding
    # Use convolution: conv = np.fft.ifft(np.fft.fft(h, L) * np.fft.fft((1-h)[::-1], L)).real
    L = 2 * n - 1
    H = np.fft.rfft(h, n=L)
    B = np.fft.rfft((1 - h)[::-1], n=L)
    conv = np.fft.irfft(H * B, n=L)
    # conv length L = 2n-1, corresponds to correlation indices from -(n-1) to (n-1)
    max_val = conv.max() * dx
    return float(max_val)

def anneal_opt(h_init, target, time_limit, start_time):

    h = h_init.copy()
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val
    rng = np.random.default_rng(1)
    N = len(h)
    while time.time() - start_time < time_limit:
        # temperature based on elapsed fraction
        elapsed = time.time() - start_time
        frac = elapsed / time_limit
        T = max(1e-4, 0.01 * (1 - frac))  # decline toward zero

        move_type = rng.random()
        if move_type < 0.5:
            # swap move
            i, j = rng.integers(0,N,size=2)
            if i==j:
                continue
            # swap values
            if h[i] == h[j]:
                continue
            cand = h.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer move
            i, j = rng.integers(0,N,size=2)
            if i==j:
                continue
            # direction
            if h[i] > 0 and h[j] < 1:
                max_delta = min(h[i], 1 - h[j])
                delta = max_delta * rng.random()
                cand = h.copy()
                cand[i] -= delta
                cand[j] += delta
            elif h[j] > 0 and h[i] < 1:
                max_delta = min(h[j], 1 - h[i])
                delta = max_delta * rng.random()
                cand = h.copy()
                cand[j] -= delta
                cand[i] += delta
            else:
                continue

        val = compute_c5(cand)
        if val < cur_val:
            h = cand
            cur_val = val
            if val < best_val:
                best_h = cand.copy()
                best_val = val
        else:
            # accept with probability exp(-(val-cur_val)/T)
            if rng.random() < np.exp(-(val - cur_val) / (T + 1e-12)):
                h = cand
                cur_val = val
    return best_h, best_val

def slsqp_optimize(h_start, max_iter=2000):
    n = h_start.shape[0]
    dx = 2.0 / n

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    eq_cons = {"type": "eq", "fun": lambda x, n=n: np.sum(x[:n]) - n/2.0}

    # compute correlation shift list once to generate all possible shifts indexes, but we need to generate constraint for each s.

    ineq = []
    for s in range(-(n-1), n):
        def make_fun(s):
            def fun(x, s=s, n=n):
                h = x[:n]
                t = x[n]
                # compute correlation for shift s
                if s >= 0:
                    term = np.dot(h[: n - s], 1.0 - h[s:])
                else:
                    s2 = -s
                    term = np.dot(h[s2:], 1.0 - h[: n - s2])
                return t - term
            return fun
        ineq.append({"type": "ineq", "fun": make_fun(s)})
    constraints = [eq_cons] + ineq

    t0 = compute_c5(h_start)
    x0 = np.concatenate([h_start, [t0]])

    opts = {"maxiter": max_iter, "ftol": 1e-10, "disp": False}
    try:
        res = opt.minimize(lambda x, n=n: x[n], x0, method="SLSQP", bounds=bounds, constraints=constraints, options=opts)
    except Exception:
        res = None

    if res is None or not res.success:
        # fallback to start
        return h_start, compute_c5(h_start)
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    # renorm sum
    target = n/2.0
    cur = h_opt.sum()
    if abs(cur - target) > 1e-8:
        diff = target - cur
        mask = (h_opt > 0.0) & (h_opt < 1.0)
        if mask.any():
            h_opt[mask] += diff / mask.sum()
            h_opt = np.clip(h_opt, 0.0, 1.0)
        else:
            h_opt *= target / cur
    return h_opt, compute_c5(h_opt)

import time
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
# basic utilities
# ----------------------------------------------------------------------
def compute_c5(h):
    """C5 value for a step function represented by array h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def project_h(v, target):
    """Clip to [0,1] and adjust to obtain sum == target."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(10):
        cur = v.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        mask = (v > 0.0) & (v < 1.0)
        if mask.any():
            v[mask] += diff / mask.sum()
            v = np.clip(v, 0.0, 1.0)
        else:
            # all at bounds – scale uniformly (should stay in bounds)
            v *= target / cur
    return v


def random_feasible(n, rng):
    """Generate a random feasible vector of length n."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    # simple projection to respect bounds
    v = project_h(v, n / 2.0)
    return v

# SLSQP optimisation (same idea as the reference program but with
# a larger iteration limit)
# ----------------------------------------------------------------------
def slsqp_optimize(h_start, max_iter=2000):
    n = h_start.shape[0]

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    eq_cons = {"type": "eq", "fun": lambda x, n=n: np.sum(x[:n]) - n / 2.0}

    ineq = []
    for s in range(-(n - 1), n):
        def make_fun(s):
            def fun(x, s=s, n=n):
                h = x[:n]
                t = x[n]
                if s >= 0:
                    term = np.dot(h[: n - s], 1.0 - h[s:])
                else:
                    s2 = -s
                    term = np.dot(h[s2:], 1.0 - h[: n - s2])
                return t - term
            return fun
        ineq.append({"type": "ineq", "fun": make_fun(s)})

    constraints = [eq_cons] + ineq

    t0 = compute_c5(h_start)
    x0 = np.concatenate([h_start, [t0]])

    opts = {"maxiter": max_iter, "ftol": 1e-10, "disp": False}
    try:
        res = opt.minimize(lambda x, n=n: x[n], x0,
                           method="SLSQP", bounds=bounds,
                           constraints=constraints, options=opts)
    except Exception:
        res = None

    if res is None or not res.success:
        return h_start, compute_c5(h_start)

    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    # enforce sum exactly
    target = n / 2.0
    cur = h_opt.sum()
    if abs(cur - target) > 1e-8:
        diff = target - cur
        mask = (h_opt > 0.0) & (h_opt < 1.0)
        if mask.any():
            h_opt[mask] += diff / mask.sum()
            h_opt = np.clip(h_opt, 0.0, 1.0)
        else:
            h_opt *= target / cur
    return h_opt, compute_c5(h_opt)

# simple hill‑climb (deterministic improvements only)
# ----------------------------------------------------------------------
def hill_climb(h, budget, start_time):
    n = h.shape[0]
    best = h.copy()
    best_val = compute_c5(best)
    rng = np.random.default_rng()
    while time.time() - start_time < budget:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # direction that respects bounds
        if best[i] > 0.0 and best[j] < 1.0:
            max_delta = min(best[i], 1.0 - best[j])
        elif best[j] > 0.0 and best[i] < 1.0:
            max_delta = min(best[j], 1.0 - best[i])
            i, j = j, i
        else:
            continue
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random()
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        val = compute_c5(cand)
        if val < best_val - 1e-9:
            best = cand
            best_val = val
    return best, best_val

def anneal_opt(h_init, target_sum, time_limit, start_time):
    n = h_init.shape[0]
    cur = h_init.copy()
    cur_val = compute_c5(cur)
    best = cur.copy()
    best_val = cur_val
    rng = np.random.default_rng()
    # initial temperature (roughly the scale of typical improvements)
    T0 = 0.02

    while time.time() - start_time < time_limit:
        elapsed = time.time() - start_time
        frac = elapsed / time_limit
        T = T0 * (1.0 - frac) + 1e-8  # linear cooling, never zero

        # decide move type
        if rng.random() < 0.5:          # swap two entries
            i, j = rng.integers(0, n, size=2)
            if i == j or cur[i] == cur[j]:
                continue
            cand = cur.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:                           # transfer a mass delta
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # move from i to j if possible, otherwise opposite direction
            if cur[i] > 0.0 and cur[j] < 1.0:
                max_delta = min(cur[i], 1.0 - cur[j])
                delta = max_delta * rng.random()
                cand = cur.copy()
                cand[i] -= delta
                cand[j] += delta
            elif cur[j] > 0.0 and cur[i] < 1.0:
                max_delta = min(cur[j], 1.0 - cur[i])
                delta = max_delta * rng.random()
                cand = cur.copy()
                cand[j] -= delta
                cand[i] += delta
            else:
                continue

        val = compute_c5(cand)
        if val < cur_val:
            cur = cand
            cur_val = val
            if val < best_val:
                best = cand.copy()
                best_val = val
        else:
            # Metropolis acceptance
            if rng.random() < np.exp(-(val - cur_val) / T):
                cur = cand
                cur_val = val
    # final projection to guarantee exact sum
    best = project_h(best, target_sum)
    return best, compute_c5(best)
