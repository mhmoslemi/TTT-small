def random_feasible(n, rng):
    target = n/2
    v = rng.random(n)
    v *= target / v.sum()
    # clip and adjust
    for _ in range(10):
        over = v > 1
        under = v < 0
        if not over.any() and not under.any():
            break
        excess = v[over] - 1
        deficit = -v[under]
        # distribute excess/deficit
        interior = (~over) & (~under)
        if interior.any():
            v[interior] -= excess.sum() / interior.sum()
            v[interior] += deficit.sum() / interior.sum()

    if mask.any():
        v[mask] += diff / mask.sum()
    else:
        v += diff / n
    v = np.clip(v, 0, 1)
    return v

def project_onto_box_sum(v, target, max_iter=100):
    low = np.min(v) - 1.0
    high = np.max(v)
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) < 1e-12:
            return x
        if s > target:
            low = mid
        else:
            high = mid
    return np.clip(v - (low + high)/2, 0.0, 1.0)

    if s >= 0:
        i_range = slice(0, n - s)
        j_range = slice(s, n)
        prod = h[i_range] * (1 - h[j_range])

        j_range = slice(0, n - s_abs)
        prod = h[i_range] * (1 - h[j_range])
        w[i_range] += prod
        w[j_range] += prod
    return w

def project_onto_box_sum(v, target, tol=1e-12, max_iter=100):
    low = np.min(v) - 1.0
    high = np.max(v)
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            low = mid
        else:
            high = mid
    # final result
    return np.clip(v - (low + high) / 2.0, 0.0, 1.0)

def random_feasible(n, rng):
    target = n / 2.0
    v = rng.random(n)
    v = v * (target / v.sum())
    # Project into [0,1] while keeping sum
    for _ in range(10):
        over = v > 1.0
        under = v < 0.0
        if not over.any() and not under.any():
            break
        excess = (v[over] - 1.0).sum()
        deficit = (-v[under]).sum()
        interior = (~over) & (~under)
        if excess > 0 and interior.any():
            v[interior] -= excess / interior.sum()
        if deficit > 0 and interior.any():
            v[interior] += deficit / interior.sum()
        v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target) > 1e-12:
        diff = target - cur
        mask = (v > 0.0) & (v < 1.0)
        if mask.any():
            v[mask] += diff / mask.sum()
        else:
            v += diff / n
        v = np.clip(v, 0.0, 1.0)
    return v

def weight_for_shift(h, s):
    n = h.shape[0]
    w = np.zeros(n)
    if s >= 0:
        # indices where both i and i+s exist
        i_end = n - s
        prod = h[:i_end] * (1.0 - h[s:])
        w[:i_end] += prod
        w[s:] += prod
    else:
        s_abs = -s
        i_start = s_abs
        prod = h[i_start:] * (1.0 - h[:n - s_abs])
        w[i_start:] += prod
        w[:n - s_abs] += prod
    return w

import time
import math
import numpy as np


def compute_c5(h: np.ndarray) -> float:
    """C5 value for a step function (array of samples)."""
    n = h.shape[0]
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode="full").max() * dx)


def project_onto_box_sum(v: np.ndarray, target: float,
                         tol: float = 1e-12, max_iter: int = 100) -> np.ndarray:
    """
    Euclidean projection of v onto the set {x ∈ [0,1]^n : sum x = target}.
    Uses a simple water‑filling (binary search for the Lagrange multiplier).
    """
    low = np.min(v) - 1.0
    high = np.max(v)
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            low = mid
        else:
            high = mid
    return np.clip(v - (low + high) / 2.0, 0.0, 1.0)


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1]^n with sum = n/2."""
    target = n / 2.0
    v = rng.random(n)
    v *= target / v.sum()
    for _ in range(10):
        over = v > 1.0
        under = v < 0.0
        if not over.any() and not under.any():
            break
        excess = (v[over] - 1.0).sum()
        deficit = (-v[under]).sum()
        interior = (~over) & (~under)
        if excess > 0 and interior.any():
            v[interior] -= excess / interior.sum()
        if deficit > 0 and interior.any():
            v[interior] += deficit / interior.sum()
        v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target) > 1e-12:
        diff = target - cur
        mask = (v > 0.0) & (v < 1.0)
        if mask.any():
            v[mask] += diff / mask.sum()
        else:
            v += diff / n
        v = np.clip(v, 0.0, 1.0)
    return v


def weight_for_shift(h: np.ndarray, s: int) -> np.ndarray:
    """
    Contribution of each index to the correlation for a given integer shift `s`.
    For shift s the correlation term is sum_i h_i (1 - h_{i+s}).
    The per‑index weight is h_i (1 - h_{i+s})  +  h_{i-s} (1 - h_i) where defined.
    """
    n = h.shape[0]
    w = np.zeros(n)
    if s >= 0:
        end = n - s
        prod = h[:end] * (1.0 - h[s:])
        w[:end] += prod
        w[s:] += prod
    else:
        s_abs = -s
        start = s_abs
        prod = h[start:] * (1.0 - h[: n - s_abs])
        w[start:] += prod
        w[: n - s_abs] += prod
    return w


def guided_hill_climb(h_start: np.ndarray, rng: np.random.Generator,
                      time_limit: float) -> tuple[np.ndarray, float]:
    """
    Fast local search.
    At every step we look at the shift that currently attains the maximal
    correlation, compute the per‑index contribution (weight) and move a
    random amount of mass from a high‑weight index to a low‑weight one.
    Only improving moves are accepted.
    """
    start = time.time()
    n = h_start.shape[0]
    curr_h = h_start.copy()
    curr_val = compute_c5(curr_h)
    best_h = curr_h.copy()
    best_val = curr_val

    while time.time() - start < time_limit:
        corr = np.correlate(curr_h, 1.0 - curr_h, mode="full")
        idx = np.argmax(corr)
        smax = idx - (n - 1)

        w = weight_for_shift(curr_h, smax)

        # source: high weight, positive mass
        src_mask = (curr_h > 1e-12) & (w > 0)
        if not np.any(src_mask):
            src_mask = curr_h > 1e-12
        i = np.argmax(w * src_mask)

        # target: low weight, room to increase
        possible_j = np.where(curr_h < 1 - 1e-12)[0]
        if possible_j.size == 0:
            break
        cand_js = rng.choice(possible_j, size=min(5, possible_j.size), replace=False)
        j = cand_js[np.argmin(w[cand_js])]

        max_delta = min(curr_h[i], 1.0 - curr_h[j])
        if max_delta <= 0:
            continue

        delta = max_delta * rng.random()
        cand_h = curr_h.copy()
        cand_h[i] -= delta
        cand_h[j] += delta

        cand_val = compute_c5(cand_h)
        if cand_val < curr_val - 1e-12:
            curr_h = cand_h
            curr_val = cand_val
            if cand_val < best_val - 1e-12:
                best_h = cand_h.copy()
                best_val = cand_val

    return best_h, best_val


def hill_climb_basic(h_start: np.ndarray, rng: np.random.Generator,
                     steps: int) -> tuple[np.ndarray, float]:
    """Simple random pairwise swaps, greedy acceptance."""
    n = h_start.shape[0]
    best_h = h_start.copy()
    best_val = compute_c5(best_h)
    curr_h = best_h.copy()
    curr_val = best_val

    for _ in range(steps):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if curr_h[i] > 1e-12 and curr_h[j] < 1 - 1e-12:
            max_delta = min(curr_h[i], 1 - curr_h[j])
            src, dst = i, j
        elif curr_h[j] > 1e-12 and curr_h[i] < 1 - 1e-12:
            max_delta = min(curr_h[j], 1 - curr_h[i])
            src, dst = j, i
        else:
            continue
        delta = max_delta * rng.random()
        cand_h = curr_h.copy()
        cand_h[src] -= delta
        cand_h[dst] += delta
        cand_val = compute_c5(cand_h)
        if cand_val < curr_val - 1e-12:
            curr_h = cand_h
            curr_val = cand_val
            if cand_val < best_val - 1e-12:
                best_h = cand_h.copy()
                best_val = cand_val
    return best_h, best_val


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step function h, its C5 bound, and the number of points."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    def remaining():
        return budget_s - (time.time() - start_time)

    # ------------------------------------------------------------------
    # build a list of starting candidates (size, vector)
    # ------------------------------------------------------------------
    candidates = []

    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n0 = h0.size
        candidates.append((n0, h0.copy()))
        for factor in (2, 3):
            n = n0 * factor
            h_up = np.repeat(h0, factor)
            candidates.append((n, h_up.copy()))

    for n in (180, 360, 540):
        for _ in range(2):
            candidates.append((n, random_feasible(n, rng)))

    rng.shuffle(candidates)

    best_h = None
    best_c5 = np.inf
    best_n = None

    # ------------------------------------------------------------------
    # optimise each candidate while we still have time
    # ------------------------------------------------------------------
    for n, h0 in candidates:
        if remaining() <= 1.0:
            break

        # enforce feasibility exactly
        target = n / 2.0
        h = np.clip(h0, 0.0, 1.0)
        cur = h.sum()
        if abs(cur - target) > 1e-12:
            h = h * (target / cur)
        h = project_onto_box_sum(h, target)

        # time budget for this candidate (no more than 60 s)
        t_alloc = min(remaining() * 0.3, 60.0)

        # guided hill climb
        h_opt, c5_opt = guided_hill_climb(h, rng, t_alloc)

        # short basic hill‑climb for fine‑tuning
        if remaining() > 2.0:
            extra_steps = int(min(5000, remaining() * 100))
            h_opt, c5_opt = hill_climb_basic(h_opt, rng, extra_steps)

        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt.copy(), c5_opt, n

    # ------------------------------------------------------------------
    # fallback (should never happen)
    # ------------------------------------------------------------------
    if best_h is None:
        best_n = 180
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    # final projection to guarantee exact constraints
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    cur = best_h.sum()
    if abs(cur - target) > 1e-12:
        best_h = best_h * (target / cur)
    best_h = project_onto_box_sum(best_h, target)

    return (best_h.tolist(), float(best_c5), int(best_n))
