import numpy as np
import time, math, random

def _compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx for the discrete step function."""
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / N))      # dx = 2/N

def _repair_sum(h, target):
    """Adjust h so that sum(h)==target while keeping 0<=h<=1."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # greedy fill/empty
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = np.random.randint(len(h))
        if diff > 0:
            cap = 1.0 - h[i]
            if cap <= 0:
                continue
            inc = min(cap, diff)
            h[i] += inc
            diff -= inc
        else:
            cap = h[i]
            if cap <= 0:
                continue
            dec = min(cap, -diff)
            h[i] -= dec
            diff += dec
    if abs(diff) > 1e-12:
        total = h.sum()
        if total > 0:
            h *= (target / total)
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                idxs = np.where(h < 1.0)[0]
                if len(idxs):
                    i = np.random.choice(idxs)
                    h[i] = min(1.0, h[i] + diff)
            else:
                idxs = np.where(h > 0.0)[0]
                if len(idxs):
                    i = np.random.choice(idxs)
                    h[i] = max(0.0, h[i] + diff)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound and the number of sample points."""
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    safety = 5.0                     # leave a few seconds margin

    # --------------------------------------------------------------
    # parameters
    N = 200                          # even size → binary‑friendly
    target_sum = N / 2.0
    dx = 2.0 / N

    # initialise a random feasible h
    h = np.random.rand(N)
    h = _repair_sum(h, target_sum)
    h = np.clip(h, 0.0, 1.0)

    # current state
    corr = np.correlate(h, 1.0 - h, mode="full")
    cur_max = np.max(corr)
    best_h = h.copy()
    best_max = cur_max

    # annealing schedule
    t_start = 5.0
    t_end   = 1e-4
    max_move_initial = 0.30
    eps = 1e-10

    # --------------------------------------------------------------
    # main optimisation loop
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s - safety:
            break
        progress = elapsed / (budget_s - safety)
        T = t_start * (t_end / t_start) ** progress
        max_move = max_move_initial * (1.0 - progress) + 1e-5

        # recompute correlation for the current state
        corr = np.correlate(h, 1.0 - h, mode="full")
        cur_max = np.max(corr)
        idx_max = np.argmax(corr)
        shift = idx_max - (N - 1)          # integer shift giving the worst value

        # ----------------------------------------------------------

        if np.random.rand() < 0.001:
