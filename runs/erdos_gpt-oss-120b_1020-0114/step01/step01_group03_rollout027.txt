import time
import math
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----- helper functions -----------------------------------------------------

_GLOBAL_N = None
_GLOBAL_ALPHA = None


def set_global_params(n, alpha):
    global _GLOBAL_N, _GLOBAL_ALPHA
    _GLOBAL_N = n
    _GLOBAL_ALPHA = alpha


def objective_func(h):
    """Smooth approximation of max_{k} ∫ h(x)(1-h(x+k))dx."""
    n = _GLOBAL_N
    alpha = _GLOBAL_ALPHA
    corr = np.correlate(h, 1.0 - h, mode="full")
    smooth = logsumexp(alpha * corr) / alpha
    return smooth * (2.0 / n)


def constraint_sum_eq(h):
    """Exact integral constraint: sum(h) = n/2."""
    n = _GLOBAL_N
    return np.sum(h) - n / 2.0


def compute_c5(h):
    """Exact C₅ value for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)


def upsample_h(old_h, new_n):
    """Linear interpolation to a finer grid while preserving ∫h = 1."""
    old_n = old_h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)

    target = new_n / 2.0
    cur = new_h.sum()
    if cur > 0:
        new_h = new_h * (target / cur)

    new_h = np.clip(new_h, 0.0, 1.0)
    cur = new_h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        new_h += diff / new_n
        new_h = np.clip(new_h, 0.0, 1.0)
        cur = new_h.sum()
        if cur > 0:
            new_h = new_h * (target / cur)
    return new_h


def apply_change(h, S, i, delta, offset, idx_base, buf):
    """
    Apply a change delta to h[i] (delta may be positive or negative)
    and update the auto‑correlation array S = correlate(h, h, mode='full').
    """
    # delta * h (old values) in pre‑allocated buffer
    np.multiply(h, delta, out=buf)

    # indices for shift (j-i) and its opposite
    idx = idx_base - i + offset
    np.add.at(S, idx, buf)

    idx_rev = -idx + 2 * offset
    np.add.at(S, idx_rev, buf)

    # diagonal correction (double count of (i,i))
    S[offset] += delta * delta

    # finally update the variable
    h[i] = h[i] + delta


# ----- main routine -------------------------------------------------------

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    time_limit = budget_s * 0.98

    # ----- initialise h ---------------------------------------------------
    if "initial_h_values" in globals():

        n = h.size

        if cur_sum != 0.0:

        h = np.clip(h, 0.0, 1.0)
        # final correction of the integral
        cur_sum = h.sum()
        diff = target_sum - cur_sum
        if abs(diff) > 1e-12:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
            cur_sum = h.sum()
            if cur_sum > 0:
                h = h * (target_sum / cur_sum)
    else:
        n = 200
        h = np.zeros(n, dtype=float)
        half = n // 2
        h[:half] = 1.0
        np.random.shuffle(h)

    # ----- maybe upsample to a finer grid ---------------------------------
    target_n = 400 if n < 400 else n
    if target_n != n:
        h = upsample_h(h, target_n)
        n = target_n

    dx = 2.0 / n
    total_sum = n / 2.0

    # ----- correlation matrix (auto‑correlation) -------------------------
    S = np.correlate(h, h, mode="full")   # length 2n-1, S[k] = Σ h_i h_{i+k}
    offset = n - 1

        if abs(cur - total_sum) > 1e-9 and cur > 0:

    cur = best_h.sum()
    if abs(cur - total_sum) > 1e-9 and cur > 0:

        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)

    return (best_h, float(final_c5), int(best_h.size))
