import time
import numpy as np

def compute_c5_fft(h):
    """Fast C5 using FFT‑based cross‑correlation."""
    n = h.shape[0]
    L = 2 * n
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(H * np.conj(G), n=L)
    # linear correlation occupies the first 2*n‑1 entries
    max_corr = np.max(corr[: 2 * n - 1])
    return max_corr * (2.0 / n)


def compute_c5_exact(h):
    """Exact (np.correlate) C5, used only for the final report."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
    return np.max(corr)


def upsample_h(old_h, new_n):
    """Linear interpolation from old grid to a finer grid, preserving the integral."""
    old_n = old_h.shape[0]
    if old_n == new_n:
        return old_h.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    target = new_n / 2.0
    s = new_h.sum()
    if s != 0:
        new_h *= target / s
    new_h = np.clip(new_h, 0.0, 1.0)

    # final correction for rounding errors
    s = new_h.sum()
    diff = target - s
    if abs(diff) > 1e-12:
        new_h += diff / new_n
        new_h = np.clip(new_h, 0.0, 1.0)
        s = new_h.sum()
        if s != 0:
            new_h *= target / s
    return new_h

        if s0 != 0:
            h0 = h0 * (target0 / s0)
        h0 = np.clip(h0, 0.0, 1.0)
    else:
        n0 = 200
        h0 = np.zeros(n0, dtype=float)
        half = n0 // 2
        h0[:half] = 1.0
        rng.shuffle(h0)

    # ------------------------------------------------------------------
    # work at a finer resolution (400 points is a good compromise)
    # ------------------------------------------------------------------
    work_n = max(200, min(400, n0 * 2))
    h_work = upsample_h(h0, work_n) if work_n != n0 else h0.copy()
    n = h_work.shape[0]
    target_sum = n / 2.0
    s = h_work.sum()
    if abs(s - target_sum) > 1e-12:
        h_work = h_work * (target_sum / s)
    h_work = np.clip(h_work, 0.0, 1.0)

    # best solution seen so far
    best_h = h_work.copy()
    best_c5 = compute_c5_fft(best_h)

    # ------------------------------------------------------------------
    # simulated‑annealing routine (multiple restarts)
    # ------------------------------------------------------------------
    max_total_iters = 200_000          # hard cap
    iters_per_restart = 50_000
    restarts = 3
    max_move_initial = 0.20            # biggest chunk that may be shifted

    for restart in range(restarts):
        if elapsed() > budget_s * 0.94:
            break

        # initialise this run
        if restart == 0:
            cur_h = best_h.copy()
        else:
            # perturb the current best to escape a possible basin
            perturb = rng.normal(0.0, 0.07, size=n)
            cur_h = best_h + perturb
            cur_h = np.clip(cur_h, 0.0, 1.0)
            # re‑project to the equality constraint
            cur_h = cur_h * (target_sum / cur_h.sum())
        cur_c5 = compute_c5_fft(cur_h)

        # temperature schedule (linear cooling)
        T0 = cur_c5 * 0.02
        min_T = 1e-6

        it = 0
        while it < iters_per_restart and elapsed() < budget_s * 0.96:
            prog = it / iters_per_restart
            T = max(T0 * (1.0 - prog), min_T)

            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                it += 1
                continue
            hi = cur_h[i]
            hj = cur_h[j]
            if hi <= 1e-12 or hj >= 1.0 - 1e-12:
                it += 1
                continue

            # step size shrinks with progress
            max_move = max_move_initial * (1.0 - prog)
            delta_cap = min(hi, 1.0 - hj, max_move)
            if delta_cap <= 0:
                it += 1
                continue
            delta = rng.random() * delta_cap

            # propose move
            cur_h[i] = hi - delta
            cur_h[j] = hj + delta

            new_c5 = compute_c5_fft(cur_h)

            accept = False
            if new_c5 < cur_c5:
                accept = True
            else:
                prob = np.exp(-(new_c5 - cur_c5) / T)
                if rng.random() < prob:
                    accept = True

            if accept:
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = cur_h.copy()
            else:
                # revert
                cur_h[i] = hi
                cur_h[j] = hj

            it += 1

    # ------------------------------------------------------------------
    # final pure hill‑climbing refinements (only improvements)
    # ------------------------------------------------------------------
    steps = 5_000
    while steps > 0 and elapsed() < budget_s * 0.99:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            steps -= 1
            continue
        hi = best_h[i]
        hj = best_h[j]
        if hi <= 1e-12 or hj >= 1.0 - 1e-12:
            steps -= 1
            continue
        delta = rng.random() * min(hi, 1.0 - hj, 0.05)
        if delta <= 0:
            steps -= 1
            continue

        best_h[i] = hi - delta
        best_h[j] = hj + delta
        new_c5 = compute_c5_fft(best_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
        else:
            best_h[i] = hi
            best_h[j] = hj
        steps -= 1

    # ------------------------------------------------------------------
    # final tidy‑up and exact evaluation
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    s = best_h.sum()
    if abs(s - target_sum) > 1e-9:
        best_h = best_h * (target_sum / s)
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5_exact(best_h)

    return (best_h, float(final_c5), int(best_h.size))
