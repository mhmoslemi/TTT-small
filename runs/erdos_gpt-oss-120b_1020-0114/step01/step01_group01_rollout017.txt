def project_to_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == target:
        return h
    # If cur > target, need to reduce some values; if cur < target, increase some.
    # We'll do simple iterative approach:
    for _ in range(10):
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        # For free variables (not at bounds)
        free = ((h > 0.0) & (h < 1.0))
        free_idx = np.where(free)[0]
        if free_idx.size == 0:
            # No free variables: adjust a random variable towards target within bounds
            i = rng.integers(len(h))
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
        else:
            delta = diff / free_idx.size
            h[free_idx] += delta
            # Clip again
            h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
    # As final safeguard, rescale a bit
    cur = h.sum()
    if abs(cur - target) > 1e-8:
        # simple proportional scaling for non-zero entries
        scale = target / cur if cur > 0 else 0
        h = np.clip(h * scale, 0.0, 1.0)
        # and repeat close fix
        cur = h.sum()
        if abs(cur - target) > 1e-8:
            # adjust one index
            diff = target - cur
            i = np.argmin(np.abs(h - 0.5))  # arbitrary
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

import time
import numpy as np
from scipy.optimize import minimize

# -------------------------------------------------------------
# Helper utilities
# -------------------------------------------------------------
def _project_to_feasible(h, target):
    """Clip h to [0,1] and adjust its sum to exactly target."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == 0:
        # put all mass on arbitrary indices
        idx = np.arange(len(h))
        np.random.shuffle(idx)
        need = int(round(target))
        h[idx[:need]] = 1.0
        return h
    # scale ignoring bounds, then correct iteratively
    s = target / cur
    h = np.clip(h * s, 0.0, 1.0)
    for _ in range(10):
        cur = h.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        free = (h > 0.0) & (h < 1.0)
        free_idx = np.where(free)[0]
        if free_idx.size == 0:
            # adjust a random free variable (might be at a bound)
            i = np.random.randint(len(h))
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
        else:
            delta = diff / free_idx.size
            h[free_idx] += delta
            h = np.clip(h, 0.0, 1.0)
    return h


def _c5_value(h, dx):
    """Maximum overlap integral for a given discretisation."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))


def _simulated_annealing(h0, n, max_iter, rng, dx, target, start_time, budget):
    """Very simple SA that preserves the total sum exactly (swap mass)."""
    cur_h = h0.copy()
    cur_val = _c5_value(cur_h, dx)
    best_h, best_val = cur_h.copy(), cur_val

    # temperature schedule
    t_start = 2.5e-2
    t_end = 1e-6
    for it in range(max_iter):
        # time guard
        if time.time() - start_time > budget * 0.96:
            break
        # linear interpolation in log‑space
        frac = it / max_iter
        temp = t_start * (t_end / t_start) ** frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # allowed move limits
        max_up = min(1.0 - cur_h[i], cur_h[j])          # increase i, decrease j
        max_down = min(cur_h[i], 1.0 - cur_h[j])        # decrease i, increase j
        if max_up + max_down <= 0:
            continue
        # raw delta in the whole feasible interval
        delta_raw = rng.uniform(-max_down, max_up)
        # shrink moves as temperature drops
        delta = delta_raw * (temp / t_start)
        if abs(delta) < 1e-12:
            continue
        new_i = cur_h[i] + delta
        new_j = cur_h[j] - delta
        if not (0.0 <= new_i <= 1.0 and 0.0 <= new_j <= 1.0):
            continue
        new_h = cur_h.copy()
        new_h[i], new_h[j] = new_i, new_j
        new_val = _c5_value(new_h, dx)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (temp + 1e-12)):
            cur_h, cur_val = new_h, new_val
            if cur_val < best_val:
                best_h, best_val = cur_h.copy(), cur_val
    return best_h, best_val


def _slsqp_optimize(h_start, n, dx, target):
    """Local refinement with SLSQP (t variable for the max)."""
    # initial t just above the current max
    t0 = _c5_value(h_start, dx) + 1e-8
    x0 = np.concatenate([h_start, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def eq_cons(x):
        return np.sum(x[:n]) - target

    def ineq_cons(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        return t - corr  # vector, must be >=0

    cons = [
        {"type": "eq", "fun": eq_cons},
        {"type": "ineq", "fun": ineq_cons},
    ]

    res = minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": 2500, "ftol": 1e-10, "disp": False},
    )
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    h_opt = _project_to_feasible(h_opt, target)
    return h_opt
