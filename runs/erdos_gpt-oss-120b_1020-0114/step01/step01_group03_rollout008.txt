        new_c5 = compute_c5(new_h)

import time
import numpy as np
import scipy.optimize as opt

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------

def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))


def upsample_h(old_h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a finer grid; preserves the integral."""
    old_n = old_h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    target = new_n / 2.0
    cur = new_h.sum()
    if cur > 0:
        new_h *= target / cur
    np.clip(new_h, 0.0, 1.0, out=new_h)
    # final projection onto the exact sum
    cur = new_h.sum()
    if cur > 0:
        new_h *= target / cur
    np.clip(new_h, 0.0, 1.0, out=new_h)
    return new_h


def _neighbor_sum(h: np.ndarray, shift: int) -> np.ndarray:
    """For each index i compute h[i+|shift|] + h[i-|shift|] (out‑of‑range terms are 0)."""
    n = h.size
    s = abs(shift)
    if s == 0:
        return np.zeros_like(h)
    ns = np.empty_like(h)
    ns.fill(0.0)
    ns[: n - s] += h[s:]
    ns[s:] += h[: n - s]
    return ns


def hill_climb(h: np.ndarray, time_budget: float, max_steps: int = 8000) -> (np.ndarray, float):
    """
    Focused hill‑climbing using the worst shift.
    Moves a small amount from a donor (low neighbor sum) to a receiver (high neighbor sum)
    while preserving the integral.
    """
    start = time.time()
    n = h.shape[0]
    cur_c5 = compute_c5(h)

    steps_without_improve = 0
    for step in range(max_steps):
        if time.time() - start > time_budget:
            break

        # correlation to locate the worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)
        if shift == 0:
            break

        # neighbour information for this shift
        neigh = _neighbor_sum(h, shift)

        donor_mask = h > 1e-12
        recv_mask = h < 1.0 - 1e-12
        if not donor_mask.any() or not recv_mask.any():
            break

        donor_i = int(np.argmin(np.where(donor_mask, neigh, np.inf)))
        recv_j = int(np.argmax(np.where(recv_mask, neigh, -np.inf)))
        if donor_i == recv_j:
            # cannot transfer, try a random pair
            i, j = np.random.randint(0, n, size=2)
            eps = min(h[i], 1.0 - h[j], 0.03)
            if eps <= 0:
                continue
            new_h = h.copy()
            new_h[i] -= eps
            new_h[j] += eps
        else:
            eps = min(h[donor_i], 1.0 - h[recv_j], 0.03)
            if eps <= 0:
                continue
            new_h = h.copy()
            new_h[donor_i] -= eps
            new_h[recv_j] += eps

        new_c5 = compute_c5(new_h)

        if new_c5 < cur_c5 - 1e-12:
            h = new_h
            cur_c5 = new_c5
            steps_without_improve = 0
        else:
            # occasional random move to escape plateaus
            if np.random.rand() < 0.02:
                i, j = np.random.randint(0, n, size=2)
                eps = min(h[i], 1.0 - h[j], 0.02)
                if eps > 0:

                    rnd_h[i] -= eps

                        cur_c5 = rnd_c5
                        steps_without_improve = 0
                        continue
            steps_without_improve += 1
            if steps_without_improve >= 2000:
                break
