        if i == j:
            continue
        # epsilon region: choose from [-eps_max, eps_max] but limited by bounds
        eps = (rng.random() - 0.5) * eps_max

import time
import numpy as np
from scipy.optimize import minimize, NonlinearConstraint

# globals used by the optimizer
_opt_n = None
_opt_dx = None
_opt_target_sum = None


def _project_to_feasible(h: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust linearly so that sum(h) = target (approximately)."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == 0.0:
        # put unit mass on a random subset of indices
        need = int(round(target))
        idx = np.random.choice(len(h), need, replace=False)
        h[idx] = 1.0
        return h
    scale = target / cur
    h = np.clip(h * scale, 0.0, 1.0)
    # iterative correction for remaining error
    for _ in range(10):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            i = np.random.randint(len(h))
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
            continue
        delta = diff / free.size
        h[free] += delta
        np.clip(h, 0.0, 1.0, out=h)
    return h


def _max_overlap(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))


def _triangular_shape(n: int) -> np.ndarray:
    """Tent‑shaped start: h(x)=max(0,1‑|x‑1|) sampled on n points."""
    dx = 2.0 / n
    x = np.arange(n) * dx + dx / 2.0
    h = 1.0 - np.abs(x - 1.0)
    h = np.clip(h, 0.0, 1.0)
    return _project_to_feasible(h, n / 2.0)


def _interpolate_h(src: np.ndarray, n_target: int) -> np.ndarray:
    """Linear interpolation of a source vector onto a new grid size."""
    n_src = src.shape[0]
    xs_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    xs_tgt = np.linspace(0.0, 2.0, n_target, endpoint=False)
    tgt = np.interp(xs_tgt, xs_src, src, left=0.0, right=0.0)
    return _project_to_feasible(tgt, n_target / 2.0)


def _optimise_slsqp(start_h: np.ndarray, maxiter: int = 800) -> tuple:
    """Run the SLSQP optimisation described in the original solution."""
    global _opt_n, _opt_dx, _opt_target_sum
    n = int(start_h.shape[0])
    _opt_n = n
    _opt_dx = 2.0 / n
    _opt_target_sum = n / 2.0

    h0 = _project_to_feasible(start_h, _opt_target_sum)

    # initial feasible t slightly above current max overlap
    init_corr = np.correlate(h0, 1.0 - h0, mode="full") * _opt_dx
    t0 = float(np.max(init_corr) + 1e-8)

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    eq_cons = {"type": "eq", "fun": lambda x: np.sum(x[:_opt_n]) - _opt_target_sum}
    m = 2 * n - 1

    def _ineq_fun(x):
        h = x[:_opt_n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode="full") * _opt_dx
        return t - corr

    ineq_cons = NonlinearConstraint(_ineq_fun, lb=np.zeros(m), ub=np.full(m, np.inf))

    x0 = np.concatenate([h0, [t0]])

    res = minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=[eq_cons, ineq_cons],
        options={"maxiter": maxiter, "ftol": 1e-9, "disp": False},
    )

    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    h_opt = _project_to_feasible(h_opt, _opt_target_sum)

    try:
        val = float(evaluate_erdos_solution(h_opt))
    except Exception:
        val = _max_overlap(h_opt)

    return h_opt, val


def _hill_climb(h: np.ndarray, cur_val: float, deadline: float, rng: np.random.Generator) -> tuple:
    """Simple stochastic local search that conserves the total mass."""
    n = len(h)
    best_h = h.copy()
    best_val = cur_val
    attempts = 0
    # schedule for step size
    init_step = 0.2
    min_step = 0.001

    while time.time() < deadline:
        step = max(min_step, init_step * (0.995 ** attempts))
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_neg = best_h[i]               # we can decrease at most this
        max_pos = 1.0 - best_h[j]         # we can increase at most this
        if max_neg == 0.0 and max_pos == 0.0:
            continue
        delta = (rng.random() * (max_pos + max_neg)) - max_neg
        if abs(delta) < 1e-14:
            continue
        new_h = best_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        # bounds are already respected by construction
        val = _max_overlap(new_h)
        if val < best_val - 1e-12:
            best_h = new_h
            best_val = val
            attempts = 0
        else:
            attempts += 1
    return best_h, best_val


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Acquire an initial vector if the harness provides one.
    # ------------------------------------------------------------------
    try:
        init_vec = np.array(initial_h_values, dtype=float)  # type: ignore[name-defined]
        n0 = int(init_vec.shape[0])
    except Exception:
        n0 = 100
        init_vec = _triangular_shape(n0)

    best_h = None
    best_c5 = np.inf
    best_n = None

    # ------------------------------------------------------------------
    # Grid sizes to explore (coarse → finer). Keep them modest.
    # ------------------------------------------------------------------
    grid_sizes = [n0]
    for factor in (1.5, 2.0, 2.5):
        gn = int(round(n0 * factor))
        if gn <= 200:
            grid_sizes.append(gn)
    if n0 > 80:
        grid_sizes.append(80)
    grid_sizes = sorted(set(grid_sizes))

    # ------------------------------------------------------------------
    # Main exploration loop.
    # ------------------------------------------------------------------
    for n in grid_sizes:
        if time.time() - start_time > budget_s * 0.94:
            break

        # build a handful of starting points for this grid size
        starts = []
        # 1) interpolation of the supplied initial vector
        starts.append(_interpolate_h(init_vec, n))
        # 2) triangular shape on this grid
        starts.append(_triangular_shape(n))
        # 3) a few random feasible vectors
        for _ in range(3):
            rnd = rng.random(n)
            rnd = _project_to_feasible(rnd, n / 2.0)
            starts.append(rnd)

        for start_h in starts:
            if time.time() - start_time > budget_s * 0.94:
                break

            # SLSQP optimisation
            try:
                h_opt, val_opt = _optimise_slsqp(start_h, maxiter=900)
            except Exception:
                h_opt = _project_to_feasible(start_h, n / 2.0)
                val_opt = _max_overlap(h_opt)

            if val_opt < best_c5 - 1e-9:
                best_h, best_c5, best_n = h_opt.copy(), val_opt, n

            # Local stochastic refinement
            deadline = start_time + budget_s * 0.98
            h_ref, val_ref = _hill_climb(h_opt, val_opt, deadline, rng)
            if val_ref < best_c5 - 1e-9:
                best_h, best_c5, best_n = h_ref.copy(), val_ref, n

    # ------------------------------------------------------------------
    # If time permits, run a final pure hill‑climb from the current best.
    # ------------------------------------------------------------------
    if best_h is not None:
        final_deadline = start_time + budget_s * 0.995
        best_h, best_c5 = _hill_climb(best_h, best_c5, final_deadline, rng)

    # ------------------------------------------------------------------
    # Safety fallback – uniform half‑mass solution.
    # ------------------------------------------------------------------
    if best_h is None:
        best_n = max(80, n0)
        best_h = np.full(best_n, 0.5)
        best_c5 = _max_overlap(best_h)

    # One final evaluation via the provided evaluator (more accurate handling).
    try:
        best_c5 = float(evaluate_erdos_solution(best_h))
    except Exception:
        pass

    return best_h, best_c5, best_n
