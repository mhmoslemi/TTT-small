import time
import numpy as np

# ---------------------------------------------------------
# Helper utilities (box+sum projection, evaluation, subgrad)
# ---------------------------------------------------------
def _project(v: np.ndarray, target: float) -> np.ndarray:
    """Project vector v onto the intersection of [0,1]^n and sum(v)=target."""
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _evaluate(h: np.ndarray) -> float:
    """C5 value for a given discrete step function."""
    n = h.size
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full") * dx)


def _subgrad_max(h: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_k ∫h(1−h(·+k)) for the current maximising lag."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    s = int(np.argmax(corr))                     # index of the maximising lag
    offset = s - (n - 1)                         # signed shift
    grad = np.zeros_like(h)

    # contribution from the term h_i * (1 - h_{i+offset})
    if offset >= 0:
        i_start = 0
        i_end = n - offset
        idx = np.arange(i_start, i_end, dtype=int)
        grad[idx] += (1.0 - h[idx + offset])
        # contribution from h_{i-offset} * (1 - h_i) (negative part)
        idx2 = np.arange(offset, n, dtype=int)
        grad[idx2] -= h[idx2 - offset]
    else:  # offset < 0
        off = -offset
        i_start = off
        i_end = n
        idx = np.arange(i_start, i_end, dtype=int)
        grad[idx] += (1.0 - h[idx + offset])    # offset negative, idx+offset valid
        idx2 = np.arange(0, n - off, dtype=int)
        grad[idx2] -= h[idx2 - offset]          # idx2 - offset = idx2 + off
    return grad * dx


# ---------------------------------------------------------
# Main optimisation routine
# ---------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # -----------------------------------------------------------------
    # 1) Initialise a feasible vector (use harness value if available)
    # -----------------------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=float)    # supplied by harness
        n = h0.size
    except Exception:
        n = 60                                          # fallback size
        h0 = np.full(n, 0.5)

    target_sum = n * 0.5
    h0 = _project(h0, target_sum)

    best_h = h0.copy()
    best_c5 = _evaluate(best_h)

    # -----------------------------------------------------------------
    # 2) Simple deterministic candidates (cosine, linear ramps, etc.)
    # -----------------------------------------------------------------
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx

    # cosine half‑wave (decreasing from 1 to 0)
    h_cos = (1.0 + np.cos(np.pi * xs / 2.0)) / 2.0
    # its mirror (increasing)
    h_cos_rev = 1.0 - h_cos
    # linear ramp up and down
    h_lin = xs / 2.0
    h_lin_rev = 1.0 - h_lin

    for cand in (h_cos, h_cos_rev, h_lin, h_lin_rev):
        cand = _project(cand, target_sum)
        c5 = _evaluate(cand)
        if c5 < best_c5:
            best_c5, best_h = c5, cand.copy()

    # -----------------------------------------------------------------
    # 3) Two‑level edge‑middle patterns (grid search over edge length & value)
    # -----------------------------------------------------------------
    # Number of points per side that share the same value 'a'
    for p in range(1, n // 2):
        # feasible interval for a so that the resulting middle value stays in [0,1]
        a_low = max(0.0, 1.0 - n / (4.0 * p))
        a_high = min(1.0, n / (4.0 * p))
        if a_low >= a_high:
            continue
        # scan a with a modest resolution
        for a in np.linspace(a_low, a_high, 300):
            b = (target_sum - 2.0 * p * a) / (n - 2.0 * p)
            if not (0.0 <= b <= 1.0):
                continue
            h = np.empty(n, dtype=float)
            h[:p] = a
            h[p:n - p] = b
            h[n - p:] = a
            # tiny floating‑point drift → a final projection
            h = _project(h, target_sum)
            c5 = _evaluate(h)
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()

    # -----------------------------------------------------------------
    # 4) Random restarts + simulated annealing (mass‑transfer moves)
    # -----------------------------------------------------------------
    max_total_iters = 120000   # global budget for the stochastic phase
    it = 0
    cur_h = best_h.copy()
    cur_c5 = best_c5
    max_step = 0.25
    T0 = 0.03

    while it < max_total_iters:
        # time guard
        if time.time() - start > budget_s * 0.95:
            break

        # temperature schedule (linear decay)
        T = T0 * (1.0 - it / max_total_iters)

        # pick two distinct indices
        i, j = np.random.randint(0, n, size=2)
        while i == j:
            j = np.random.randint(0, n)

        # allowable delta keeping both entries in [0,1]
        hi = min(1.0 - cur_h[i], cur_h[j])          # positive direction bound
        lo = -min(cur_h[i], 1.0 - cur_h[j])         # negative direction bound
        if hi - lo <= 0.0:
            it += 1
            continue

        # step size slowly shrinks with iteration count
        step_sz = max_step * (1.0 - it / max_total_iters)
        delta = np.random.uniform(lo, hi)
        delta = np.clip(delta, -step_sz, step_sz)

        # propose new vector (preserves the total sum exactly)
        h_new = cur_h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        # safety projection (should be unnecessary but guards numerical drift)
        h_new = _project(h_new, target_sum)

        new_c5 = _evaluate(h_new)

        # Metropolis acceptance
        if new_c5 < cur_c5 or np.random.rand() < math.exp(-(new_c5 - cur_c5) / max(T, 1e-12)):
            cur_h, cur_c5 = h_new, new_c5
            if new_c5 < best_c5:
                best_c5, best_h = new_c5, h_new.copy()
        it += 1

        # occasional restart if stalled for a while
        if it % 20000 == 0 and cur_c5 - best_c5 > 1e-4:
            cur_h = _project(np.random.rand(n), target_sum)
            cur_c5 = _evaluate(cur_h)

    # -----------------------------------------------------------------
    # 5) Deterministic projected sub‑gradient descent (polishing)
    # -----------------------------------------------------------------
    max_pg_iters = 30000
    cur_h = best_h.copy()
    cur_c5 = best_c5
    base_lr = 0.12

    for it in range(max_pg_iters):
        if time.time() - start > budget_s * 0.975:
            break

        grad = _subgrad_max(cur_h)
        # diminishing step size
        lr = base_lr * (1.0 - it / max_pg_iters)
        h_next = _project(cur_h - lr * grad, target_sum)
        next_c5 = _evaluate(h_next)

        # accept if non‑worsening
        if next_c5 <= cur_c5:
            cur_h, cur_c5 = h_next, next_c5
            if next_c5 < best_c5:
                best_c5, best_h = next_c5, h_next.copy()
        else:
            # occasional aggressive step (re‑try with a smaller learning rate)
            lr = lr * 0.5
            h_next = _project(cur_h - lr * grad, target_sum)
            next_c5 = _evaluate(h_next)
            if next_c5 < cur_c5:
                cur_h, cur_c5 = h_next, next_c5
                if next_c5 < best_c5:
                    best_c5, best_h = next_c5, h_next.copy()

    # final safety projection (ensures exact feasibility)
    best_h = _project(best_h, target_sum)
    best_c5 = _evaluate(best_h)

    return best_h, float(best_c5), n
