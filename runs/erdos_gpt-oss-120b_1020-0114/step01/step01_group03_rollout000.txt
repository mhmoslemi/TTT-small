import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def upsample_h(old_h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a finer grid while preserving the integral."""
    old_n = old_h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)

    target = new_n / 2.0
    cur = new_h.sum()
    if cur > 0:
        new_h = new_h * (target / cur)
    new_h = np.clip(new_h, 0.0, 1.0)

    cur = new_h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        new_h += diff / new_n
        new_h = np.clip(new_h, 0.0, 1.0)
        cur = new_h.sum()
        if cur > 0:
            new_h = new_h * (target / cur)
    return new_h

def anneal_optimize(start_h: np.ndarray, max_seconds: float, start_T: float = 0.02) -> tuple:
    """
    Simulated‑annealing style optimization.
    Moves a small mass epsilon from a random index i to a random index j.
    """
    h = start_h.copy()
    n = h.shape[0]
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val

    t0 = time.time()
    while time.time() - t0 < max_seconds:
        # choose two distinct positions
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # maximal epsilon respecting bounds
        eps_max = min(0.5, h[i], 1.0 - h[j])
        if eps_max <= 0.0:
            continue
        eps = np.random.rand() * eps_max

        # propose new vector
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps

        new_val = compute_c5(new_h)
        delta = new_val - cur_val

        # temperature schedule (linear cooling)
        elapsed = time.time() - t0
        T = start_T * max(0.0, 1.0 - elapsed / max_seconds)

        if delta <= 0.0 or np.exp(-delta / (T + 1e-12)) > np.random.rand():
            h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
    return best_h, best_val

        h0 = np.full(n0, 0.5, dtype=float)

    # ------------------------------------------------------------------
    # Stage 1 : annealing on the original grid
    # ------------------------------------------------------------------
    stage1_frac = 0.55
    stage1_time = budget_s * stage1_frac
    best_h1, best_c5_1 = anneal_optimize(h0, stage1_time, start_T=0.02)

    # ------------------------------------------------------------------
    # Stage 2 : increase resolution and anneal again
    # ------------------------------------------------------------------
    # Choose a finer grid (up to 500 points)
    n1 = min(500, max(2 * n0, n0 + 20))
    if n1 % 2:
        n1 += 1
    if n1 != n0:
        h1_up = upsample_h(best_h1, n1)
    else:
        h1_up = best_h1.copy()

    stage2_time = budget_s - (time.time() - start_all) - 0.05 * budget_s
    if stage2_time < 1.0:
        stage2_time = budget_s - (time.time() - start_all) - 0.01 * budget_s
    best_h2, best_c5_2 = anneal_optimize(h1_up, max(stage2_time, 0.0), start_T=0.02)

    # ------------------------------------------------------------------
    # Pick the better of the two stages
    # ------------------------------------------------------------------
    if best_c5_2 < best_c5_1:
        final_h = best_h2
        final_c5 = best_c5_2
    else:
        final_h = best_h1
        final_c5 = best_c5_1

    # ------------------------------------------------------------------
    # Final clean‑up: guarantee constraints exactly
    # ------------------------------------------------------------------
    n_final = final_h.shape[0]
    final_h = np.clip(final_h, 0.0, 1.0)
    target = n_final / 2.0
    s = final_h.sum()
    if s > 0 and abs(s - target) > 1e-12:
        final_h = final_h * (target / s)
    final_h = np.clip(final_h, 0.0, 1.0)

    # Re‑evaluate to be safe
    final_c5 = compute_c5(final_h)

    return (final_h, float(final_c5), int(n_final))
