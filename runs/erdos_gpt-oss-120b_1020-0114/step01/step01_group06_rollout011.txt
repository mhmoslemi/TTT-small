def _refine_h_with_time(h, target_sum, time_budget, start_time, rng):

    # local variables
    n = h.shape[0]
    dx = 2.0 / n

    best_h = h.copy()
    best_val = _c5_value(best_h)
    cur_h = h.copy()

    step = 0.1
    last_improve_iter = 0
    iter_count = 0
    # while time left
    while time.time() - start_time < time_budget:
        iter_count += 1
        corr = _compute_corr(cur_h)
        max_corr = np.max(corr)
        # compute current C5
        cur_val = max_corr * dx
        # find active shifts
        eps = 1e-12 * max_corr + 1e-12

        grad = np.zeros_like(cur_h)

import time
import numpy as np
from scipy import optimize

def _project_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Project h onto the hyperplane sum(h)=target while keeping 0≤h≤1."""
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-15:
        return h
    # Determine where we can move in the needed direction
    caps = (1.0 - h) if diff > 0 else (-h)
    # Greedy fix
    for i in np.where(caps != 0)[0]:
        take = min(abs(diff), abs(caps[i]))
        take = np.copysign(take, diff)
        h[i] += take
        diff -= take
        if abs(diff) < 1e-15:
            break
    return h

def _c5_value(h: np.ndarray) -> float:
    """Compute C5 = max_k ∫ h(x)(1‑h(x+k))dx for the discrete step."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def _linear_gradient(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the overlap for a given (linear) shift."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        k = shift
        # +1 for indices where h_i appears alone
        g[: n - k] += 1.0
        # -h_{i+k} part
        g[: n - k] -= h[k:]
        # -h_{i} part for the partner index
        g[k:] -= h[: n - k]
    else:
        k = -shift
        # shift negative: O_{-k} = sum_{i=k}^{n-1} h_i (1 - h_{i-k})
        g[k:] += 1.0 - h[: n - k]
        g[: n - k] -= h[k:]
    return g

def _refine_h_with_time(h0: np.ndarray, target_sum: float, max_seconds: float, rng) -> tuple:
    """Projected gradient descent with occasional random swaps."""
    end_time = time.time() + max_seconds
    h = h0.copy()
    best_h = h.copy()
    best_val = _c5_value(best_h)
    step = 0.12

    while time.time() < end_time:
        n = h.shape[0]
        dx = 2.0 / n
        corr_full = np.correlate(h, 1.0 - h, mode='full')
        max_corr = corr_full.max()
        cur_val = max_corr * dx

        # active shifts (those attaining the maximum within tolerance)
        eps = 1e-12
        act_idx = np.where(corr_full >= max_corr - eps)[0]
        shifts = act_idx - (n - 1)

        # average sub‑gradient over active shifts
        grad = np.zeros_like(h)
        for s in shifts:
            grad += _linear_gradient(h, int(s))
        grad /= len(shifts)

        improved = False
        step_local = step
        for _ in range(5):
            h_new = h - step_local * grad
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = _project_sum(h_new, target_sum)
            new_val = _c5_value(h_new)
            if new_val + 1e-12 < cur_val:
                h = h_new
                cur_val = new_val
                improved = True
                if new_val < best_val - 1e-12:
                    best_val = new_val
                    best_h = h_new.copy()
                step_local = min(step_local * 1.2, 0.5)
                break
            else:
                step_local *= 0.5

        if not improved:
            step *= 0.7
            if step < 1e-5:

                    h[i] += delta
                    h[j] -= delta
                    h = np.clip(h, 0.0, 1.0)   # sum unchanged
