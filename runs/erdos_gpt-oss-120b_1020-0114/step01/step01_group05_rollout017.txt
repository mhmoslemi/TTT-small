def grad_max_corr(h, dx):
    n = h.size
    # correlation
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_val = corr.max()
    # tolerance
    eps = 1e-12
    ks = np.where(np.abs(corr - max_val) <= eps)[0] - (n - 1)
    # compute gradient sum
    grad = np.zeros_like(h)
    for k in ks:
        # shift positive or negative
        if k >= 0:
            # i from 0 to n-1-k: valid i+ k
            i_end = n - k
            # contributions for first term
            grad[:i_end] += (1 - h[k:])  # shape i_end
            # contributions for second term (i - k)
            grad[k:] -= h[:i_end]  # shift second term
        else:
            k2 = -k
            i_end = n - k2
            grad[k2:] += (1 - h[:i_end])  # because i+(-k) = i - k2, being >=0
            grad[:i_end] -= h[k2:]  # second term
    # multiply by dx
    grad *= dx
    return grad
```

def _grad_max(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    max_val = corr.max()
    # tolerance: relative maybe 1e-12 * max
    tol = 1e-12
    ks = np.where(np.abs(corr - max_val) <= tol)[0] - (n - 1)
    grad = np.zeros_like(h)
    for k in ks:
        if k >= 0:
            end = n - k
            # add (1 - h[i+k]) for i in [0,end)
            grad[:end] += (1 - h[k:])
            # subtract h[i-k] for i in [k, n)
            grad[k:] -= h[:end]
        else:
            k2 = -k
            end = n - k2
            # For negative shift:
            # i + k = i - k2 >=0 => i >= k2
            # contributions first term: (1 - h[i+k]) = (1 - h[i - k2]) for i >= k2
            grad[k2:] += (1 - h[:end])
            # second term: -h[i-k] = -h[i + k2] for i < n - k2 (i where i - k = i + k2)
            grad[:end] -= h[k2:]
    grad *= dx
    return grad

import time
import numpy as np

def _evaluate(h):
    """Exact C5 for the discrete step function."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def _project(v, target):
    """Project v onto 0≤x≤1 with sum(v)=target."""
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _grad_c_k(h, k, dx):
    """Gradient of the overlap term for a single shift k."""
    n = h.size
    g = np.zeros_like(h)
    if k >= 0:
        end = n - k
        # (1 - h[i+k]) part
        g[:end] += (1.0 - h[k:])
        # -h[i-k] part
        g[k:] -= h[:end]
    else:
        kp = -k
        end = n - kp
        # (1 - h[i+k]) = (1 - h[i-kp]) for i >= kp
        g[kp:] += (1.0 - h[:end])
        # -h[i-k] = -h[i+kp] for i < n-kp
        g[:end] -= h[kp:]
    return g * dx


def _grad_max(h):
    """Subgradient of max_k  ∫h(x)(1−h(x+k))dx."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = corr.max()
    # all shifts attaining the maximum (within tiny tolerance)
    ks = np.where(np.abs(corr - max_val) <= 1e-12)[0] - (n - 1)
    grad = np.zeros_like(h)
    for k in ks:
        grad += _grad_c_k(h, k, dx)
    if ks.size > 1:
        grad /= ks.size
    return grad


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------ #
    # Initialise from parent if available, otherwise start from scratch.
    # ------------------------------------------------------------------ #
    try:
        parent = np.array(initial_h_values, dtype=float)  # provided by harness
        m = parent.size
    except Exception:
        parent = None
        m = 0

    # Choose a modest but flexible size (≤200). Larger n gives more freedom.
    if m:
        n = min(200, max(80, int(m * 2.5)))
        # linear up‑sampling of the parent pattern
        x_old = np.linspace(0.0, 2.0, m, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x_new, x_old, parent)
    else:
        n = 120
        h = np.full(n, 0.5)

    target_sum = n * 0.5                # ∫h = 1  →  sum = n/2
    h = _project(h, target_sum)

    best_h = h.copy()
    best_c = _evaluate(best_h)

    # ------------------------------------------------------------------ #
    # Subgradient descent with projection, occasional jitter.
    # ------------------------------------------------------------------ #
    step = 0.8                # initial step magnitude
    it = 0
    last_improve = 0
    MAX_ITERS = 10 ** 7      # effectively limited by time
    while it < MAX_ITERS and (time.time() - start) < budget_s * 0.95:
        cur_c = _evaluate(h)

        # gradient of the *max* lag(s)
        grad = _grad_max(h)

        improved = False
        # try decreasing step sizes if needed
        for factor in (1.0, 0.5, 0.25, 0.125):
            alpha = step * factor
            h_candidate = _project(h - alpha * grad, target_sum)
            cand_c = _evaluate(h_candidate)

            if cand_c < cur_c - 1e-12:          # strict improvement
                h = h_candidate
                cur_c = cand_c
                step = min(alpha * 1.1, 2.0)    # modestly increase step
                improved = True
                if cand_c < best_c:
                    best_c = cand_c
                    best_h = h_candidate.copy()
                    last_improve = it
                break

        if not improved:
            step = max(step * 0.5, 1e-6)       # shrink step if stuck

        # ---------- occasional random perturbations ----------
        if it % 2500 == 0 and it > 0:
            jitter = np.random.normal(scale=0.03, size=n)
            h_jit = _project(h + jitter, target_sum)
            c_jit = _evaluate(h_jit)
            if c_jit < best_c:
                h = h_jit
                best_h = h_jit.copy()
                best_c = c_jit
                step = 0.7
                last_improve = it

        # ---------- restart if stagnation ----------

            last_improve = it

        # early exit if we have already crossed the target bound
        if best_c <= 0.3808:
            break

        it += 1

    return best_h, best_c, n
