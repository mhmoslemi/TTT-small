def local_search_binary(n, rng, max_iter=10000, init_h=None):
    if init_h is None:
        # ensure sum = n//2
        h = np.zeros(n, dtype=int)
        idx = rng.choice(n, n//2, replace=False)
        h[idx] = 1
    else:
        h = init_h.copy()
    best_h = h.copy()
    best_val = corr_max(h)
    cur_val = best_val
    T0 = 0.5 maybe; schedule: T = T0 * (1 - iter/max_iter)
    for iteration in range(max_iter):
        # pick random 1 and 0 indices
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) ==0 or len(zeros)==0:
           break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i], h[j] = 0, 1
        new_val = corr_max(h)
        # accept if better or with probability
        delta = new_val - cur_val
        T = T0 * (1 - iteration / max_iter)
        if delta < 0 or rng.random() < np.exp(-delta / (T+1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = 1, 0
    return best_h,best_val

import numpy as np

# globals used by the continuous optimiser
_opt_n = None          # grid size
_opt_dx = None         # spacing 2 / n
_opt_target_sum = None # n / 2

def objective_fun(x):
    """Minimise the bound variable t."""
    return x[-1]

def eq_constraint_fun(x):
    """Enforce the integral constraint sum(h)=n/2."""
    return np.sum(x[:_opt_n]) - _opt_target_sum

def ineq_constraint_fun(x):
    """Ensure t >= overlap for every shift."""
    h = x[:_opt_n]
    t = x[-1]
    corr = np.correlate(h, 1 - h, mode='full') * _opt_dx
    return t - corr  # non‑negative for all lags

def project_to_feasible(h, target):
    """Clip to [0,1] and rescale so that the sum is (approximately) target."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == 0:
        # put unit mass on a random subset
        idx = np.random.choice(len(h), int(round(target)), replace=False)
        h[idx] = 1.0
        return h
    scale = target / cur
    h = np.clip(h * scale, 0.0, 1.0)
    # simple iterative correction for any remaining error
    for _ in range(10):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            i = np.random.randint(len(h))
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
            continue
        delta = diff / free.size
        h[free] += delta
        np.clip(h, 0.0, 1.0, out=h)
    return h

def optimise_continuous(start_h, max_iter=2000):

    global _opt_n, _opt_dx, _opt_target_sum
    n = int(start_h.shape[0])

    h_opt = project_to_feasible(h_opt, _opt_target_sum)

    try:
        c5_val = float(evaluate_erdos_solution(h_opt))
    except Exception:
        c5_val = float(res.x[-1])

    return h_opt, c5_val

def compute_c5(arr):
    """Fast overlap bound for a (binary or continuous) step function."""
    n = len(arr)
    corr = np.correlate(arr, 1 - arr, mode='full') * (2.0 / n)
    return float(np.max(corr))

def hill_climb_binary_anneal(start_h, max_swaps, rng):
    """Simulated‑annealing style hill‑climb on a binary 0/1 vector."""
    h = start_h.copy()
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val
    T0 = 0.02  # initial temperature

    for it in range(max_swaps):
        T = T0 * (1 - it / max_swaps)  # linear cooling
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0
        h[j] = 1
        new_val = compute_c5(h)
        delta = new_val - cur_val
        if delta <= 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
    return best_h.astype(float), best_val

def _binary_from_continuous(v, target):
    """Round a continuous vector to 0/1 while keeping the exact sum."""
    h = (v > 0.5).astype(int)
    cur = int(h.sum())
    if cur > target:
        excess = cur - target
        idx_one = np.where(h == 1)[0]
        flip = np.random.choice(idx_one, excess, replace=False)
        h[flip] = 0
    elif cur < target:
        deficit = target - cur
        idx_zero = np.where(h == 0)[0]
        flip = np.random.choice(idx_zero, deficit, replace=False)
        h[flip] = 1
    return h

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine problem size and obtain an initial vector if supplied
    # --------------------------------------------------------------
    try:
        init_vec = np.array(initial_h_values, dtype=float)
        n = init_vec.shape[0]
    except Exception:
        n = 100
        init_vec = np.full(n, 0.5)

    if n % 2 != 0:          # binary implementation needs an even grid
        n -= 1
        init_vec = init_vec[:n]

    target = n // 2                         # required number of ones (half the mass)

    # --------------------------------------------------------------
    # Baseline evaluation (continuous start if any)
    # --------------------------------------------------------------
    best_h = init_vec.astype(float)
    try:
        best_val = float(evaluate_erdos_solution(best_h))
    except Exception:
        best_val = compute_c5(best_h)

    # --------------------------------------------------------------
    # Continuous optimisation via SLSQP (few but deeper runs)
    # --------------------------------------------------------------
    # the first run starts from the provided initial vector
    h_cont, val_cont = optimise_continuous(best_h, max_iter=2500)
    if val_cont < best_val - 1e-12:
        best_h, best_val = h_cont, val_cont

    # additional random restarts while time permits
    while time.time() < deadline * 0.7:
        # create a random feasible start (continuous)
        rand_start = rng.random(n)
        rand_start = project_to_feasible(rand_start, n / 2.0)
        h_c, v_c = optimise_continuous(rand_start, max_iter=1500)
        if v_c < best_val - 1e-12:
            best_h, best_val = h_c, v_c

    # --------------------------------------------------------------
    # Convert the best continuous solution to a binary one and
    # improve it with simulated‑annealing hill climbing.
    # --------------------------------------------------------------
    bin_start = _binary_from_continuous(best_h, target)
    time_left = deadline - time.time()
    if time_left > 5.0:
        # allocate the remaining budget roughly equally between two phases
        swaps = int(min(20000, max(2000, time_left * 500)))
        h_bin, val_bin = hill_climb_binary_anneal(bin_start, swaps, rng)
        if val_bin < best_val - 1e-12:
            best_h, best_val = h_bin, val_bin

    # --------------------------------------------------------------
    # Additional random binary hill‑climbs (if time still left)
    # --------------------------------------------------------------
    while time.time() < deadline * 0.97:
        # random binary start
        idx = rng.choice(n, target, replace=False)
        start_bin = np.zeros(n, dtype=int)
        start_bin[idx] = 1
        # allocate a modest number of swaps based on remaining time
        remaining = deadline - time.time()
        if remaining < 1.0:
            break
        swaps = int(min(5000, max(500, remaining * 300)))
        h_bin2, val_bin2 = hill_climb_binary_anneal(start_bin, swaps, rng)
        if val_bin2 < best_val - 1e-12:
            best_h, best_val = h_bin2, val_bin2
        if best_val < 0.3808:
            break

    # --------------------------------------------------------------
    # Final precise evaluation using the official helper
    # --------------------------------------------------------------
    try:
        final_val = float(evaluate_erdos_solution(best_h))
    except Exception:
        final_val = compute_c5(best_h)

    return best_h, final_val, n
