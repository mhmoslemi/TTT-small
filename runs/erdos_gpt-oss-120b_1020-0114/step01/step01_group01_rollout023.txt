def compute_c5(h, dx):
    n = len(h)
    # compute convolution of h and (1-h)[::-1] using FFT
    m = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, n=m)
    G = np.fft.rfft((1-h)[::-1], n=m)
    conv = np.fft.irfft(H * G, n=m)[:2*n-1]
    return np.max(conv) * dx

def simulated_annealing(h0, target_sum, rng, time_limit):
    # h0 is feasible float vector with sum target_sum.
    # returns best_h, best_val.
    start = time.time()
    n = len(h0)
    dx = 2.0/n
    h = h0.copy()
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val

    # Temperature schedule parameters
    # start_temp = max(cur_val * 0.1, 1e-3)
    start_temp = max(cur_val * 0.01, 0.001)  # smaller maybe
    end_temp = 1e-6
    # We will compute total possible iterations based on time budget
    # Each iteration may be cheap; we limit loops:
    iter_count = 0
    while time.time() - start < time_limit:
        # compute temperature
        # ratio of progress
        elapsed = time.time() - start
        # linear cooling:
        temp = start_temp * (1 - elapsed / time_limit) + end_temp
        # choose i, j distinct random
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        # compute max transfer
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        # choose epsilon: random fraction of max_transfer
        eps = rng.random() * max_transfer
        if eps == 0:
            continue
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        # ensure sum unchanged automatically
        # compute new val
        new_val = compute_c5(new_h)
        delta = new_val - cur_val
        if delta <= 0 or rng.random() < math.exp(-delta / temp):
            h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
        iter_count += 1
    return best_h, best_val

def run_slsqp(start_h, rng):
    n = len(start_h)
    global _opt_n, _opt_dx, _opt_target_sum
    _opt_n = n
    _opt_dx = 2.0 / n
    _opt_target_sum = n / 2.0

    # Ensure feasible
    h0 = project_to_feasible(start_h, _opt_target_sum)

    # initial t
    init_corr = np.correlate(h0, 1 - h0, mode='full') * _opt_dx
    t0 = float(np.max(init_corr) + 1e-8)

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    eq_cons = {'type': 'eq', 'fun': eq_cons_slsqp}
    # Using NonlinearConstraint
    from scipy.optimize import NonlinearConstraint

    # number of lags:
    m = 2 * n - 1
    ineq_con = NonlinearConstraint(
        ineq_cons_slsqp,
        lb=np.zeros(m),
        ub=np.full(m, np.inf)
    )

    x0 = np.concatenate([h0, [t0]])

    res = minimize(
        objective_slsqp,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=[eq_cons, ineq_con],
        options={'maxiter': 400, 'ftol': 1e-9, 'disp': False}

    try:
        c5_val = float(evaluate_erdos_solution(h_opt))
    except Exception:
        c5_val = float(res.x[-1])

import time
import math
import numpy as np
from scipy.optimize import minimize, NonlinearConstraint

# ----------------------------------------------------------------------
# Helper functions (top level, no closures)
# ----------------------------------------------------------------------
def project_to_feasible(h, target, rng=None):
    """Clip to [0,1] and adjust linearly so that sum(h) = target (approximately)."""
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) < 1e-12:
        return h
    for _ in range(20):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            # no free variables – push a random index toward the needed direction
            if rng is None:
                idx = np.random.randint(len(h))
            else:
                idx = rng.integers(len(h))
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
            continue
        delta = diff / free.size
        h[free] += delta
        np.clip(h, 0.0, 1.0, out=h)
    return h


def compute_c5(h):
    """Maximum overlap integral for a given step vector."""
    n = h.shape[0]
    dx = 2.0 / n
    # convolution via FFT (fast for many calls)
    m = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=m)
    G = np.fft.rfft((1.0 - h)[::-1], n=m)
    conv = np.fft.irfft(H * G, n=m)[: 2 * n - 1]
    return np.max(conv) * dx


def simulated_annealing(start_h, target_sum, rng, time_limit):
    """Simple SA that moves a random epsilon from one coordinate to another."""
    start = time.time()
    n = start_h.shape[0]
    dx = 2.0 / n
    h = start_h.copy()
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val

    # temperature schedule (linear decay)
    start_temp = max(cur_val * 0.2, 1e-4)

    while time.time() - start < time_limit:
        elapsed = time.time() - start
        # linear cooling to a tiny floor
        temp = max(start_temp * (1.0 - elapsed / time_limit), 1e-8)

        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0.0:
            continue
        eps = rng.random() * max_transfer
        if eps == 0.0:
            continue

        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps

        new_val = compute_c5(new_h)
        delta = new_val - cur_val
        if delta <= 0.0 or rng.random() < math.exp(-delta / temp):
            h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
    return best_h, best_val


# Global placeholders for SLSQP
_opt_n = None
_opt_dx = None
_opt_target_sum = None


def _objective_slsqp(x):
    """Objective: minimise the auxiliary variable t (last entry)."""
    return x[-1]


def _eq_cons_slsqp(x):
    """Equality: integral of h must be 1 (sum = n/2)."""
    return np.sum(x[:_opt_n]) - _opt_target_sum


def _ineq_cons_slsqp(x):
    """Inequality: t >= overlap for every shift."""
    h = x[:_opt_n]
    t = x[-1]
    corr = np.correlate(h, 1 - h, mode="full") * _opt_dx
    return t - corr  # must be >= 0


def run_slsqp(start_h, rng):
    """Refine a feasible vector with a classic SLSQP formulation."""
    global _opt_n, _opt_dx, _opt_target_sum
    n = start_h.shape[0]
    _opt_n = n
    _opt_dx = 2.0 / n
    _opt_target_sum = n / 2.0

    h0 = project_to_feasible(start_h, _opt_target_sum, rng)

    init_corr = np.correlate(h0, 1 - h0, mode="full") * _opt_dx
    t0 = float(np.max(init_corr) + 1e-8)

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    eq_cons = {"type": "eq", "fun": _eq_cons_slsqp}
    m = 2 * n - 1
    ineq_cons = NonlinearConstraint(
        _ineq_cons_slsqp,
        lb=np.zeros(m),
        ub=np.full(m, np.inf),
    )

    x0 = np.concatenate([h0, [t0]])

    res = minimize(
        _objective_slsqp,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=[eq_cons, ineq_cons],
        options={"maxiter": 500, "ftol": 1e-9, "disp": False},
    )
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    h_opt = project_to_feasible(h_opt, _opt_target_sum, rng)

    try:
        c5_val = float(evaluate_erdos_solution(h_opt))
    except Exception:
        c5_val = float(res.x[-1])
    return h_opt, c5_val


def upsample_h(h, n_new):
    """Linear interpolation to a finer grid."""
    n_old = h.shape[0]
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, left=0.0, right=0.0)
    return h_new
