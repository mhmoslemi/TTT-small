def optimise_h(start_h, n_iter, rng):
    n = len(start_h)
    dx = 2.0 / n
    target = n / 2.0
    h = start_h.copy()
    best_h = h.copy()
    best_val = evaluate(h)
    # parameters:
    alpha = 0.5   # step scaling factor for gradient
    for it in range(n_iter):
        # compute overlaps
        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_idx = np.argmax(corr)
        # shift k = max_idx - (n-1)
        k = max_idx - (n - 1)
        # compute gradient for that shift
        g = np.ones_like(h)
        if k >= 0:
            g[:n - k] -= h[k:]   # h[i + k] for i in 0..n-1-k
            g[k:] -= h[:n - k]   # h[i - k] for i in k..n-1; note same as above
        else:
            kk = -k
            g[kk:] -= h[:n - kk]
            g[:n - kk] -= h[kk:]
        # try gradient step
        step = alpha * g
        h_new = h - step
        # project to feasible
        h_new = np.clip(h_new, 0.0, 1.0)
        cur_sum = h_new.sum()
        if cur_sum != target:
            # adjust to exact sum via scaling of free variables
            # Use simple scaling factor for all values not at bounds
            # We'll use project_to_feasible similar
            h_new = project_to_feasible(h_new, target)
        val = evaluate(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        # Accept step if not worse? Could use some acceptance condition
        if val <= evaluate(h) + 1e-12:  # improvement or equal
            h = h_new
        else:
            # fallback: random swap
            i, j = rng.integers(0, n, size=2)

import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def _dx_from_n(n):
    return 2.0 / n


def _c5_value(h):
    """Maximum overlap C5 for a given step vector h."""
    n = h.shape[0]
    dx = _dx_from_n(n)
    # correlation of h with (1-h) (zero‑padded)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))


def _project_to_feasible(h, target):
    """Clip to [0,1] and enforce sum(h)=target (approximately)."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == 0.0:
        # place unit mass on a random subset
        idx = np.random.choice(len(h), int(round(target)), replace=False)
        h[idx] = 1.0
        return h
    scale = target / cur
    h = np.clip(h * scale, 0.0, 1.0)

    # iterative correction for the remaining error
    for _ in range(15):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            i = np.random.randint(len(h))
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
        else:
            delta = diff / free.size
            h[free] += delta
            np.clip(h, 0.0, 1.0, out=h)
    return h


def _interpolate_h(source, new_n):
    """Linear interpolation from source (length m) to a new length new_n."""
    m = source.shape[0]
    xs = np.linspace(0.0, 2.0, m, endpoint=False)
    xt = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(xt, xs, source, left=0.0, right=0.0)

# Local search / simulated annealing
# ------------------------------------------------------------
def _local_search(start_h, max_iters, time_limit, start_time):
    """Greedy + small‑temperature annealing on a fixed grid."""
    n = start_h.shape[0]
    target_sum = n / 2.0
    dx = _dx_from_n(n)

    # current state
    h = start_h.copy()
    cur_val = _c5_value(h)

    best_h = h.copy()
    best_val = cur_val

    # temperature schedule
    T0 = 0.015

    rng = np.random.default_rng()

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break

        # decide whether to do a targeted move (20% chance)
        if rng.random() < 0.2:
            # compute the worst shift for the current h
            corr = np.correlate(h, 1.0 - h, mode="full") * dx
            k_idx = int(np.argmax(corr))
            k_shift = k_idx - (n - 1)

            if k_shift >= 0:
                valid = np.arange(0, n - k_shift)
                contrib = h[valid] * (1.0 - h[valid + k_shift])
            else:
                kk = -k_shift
                valid = np.arange(kk, n)
                contrib = h[valid] * (1.0 - h[valid - kk])

            if contrib.size == 0:
                continue
            i = valid[int(np.argmax(contrib))]                # source (large contribution)
            # pick a destination that can accept mass
            cand = np.where(h < 1.0 - 1e-12)[0]
            if cand.size == 0:
                continue
            j = int(rng.choice(cand))
        else:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue

        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 1e-12:
            continue
        delta = max_delta * rng.random()
        if delta < 1e-12:
            continue

        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta

        new_val = _c5_value(h_new)

        # keep the best ever seen
        if new_val < best_val:
            best_val = new_val
            best_h = h_new.copy()

        # acceptance rule (accept improvements, otherwise with prob. exp(-Δ/T))
        if new_val < cur_val:
            h, cur_val = h_new, new_val
        else:
            T = T0 * (1.0 - it / max_iters)
            if T > 0 and rng.random() < np.exp(-(new_val - cur_val) / T):
                h, cur_val = h_new, new_val

        # occasional temperature decay without affecting moves
        if (it + 1) % 1000 == 0:
            T0 *= 0.95

    return best_h, best_val
