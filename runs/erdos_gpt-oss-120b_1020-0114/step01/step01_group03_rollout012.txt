    old_n = old_h.shape[0]

        if h_cur[i] > 1e-12 and h_cur[j] < 1.0 - 1e-12:
            max_tr = min(h_cur[i], 1.0 - h_cur[j])
            direction = 1
        elif h_cur[j] > 1e-12 and h_cur[i] < 1.0 - 1e-12:
            max_tr = min(h_cur[j], 1.0 - h_cur[i])
            direction = -1
        else:
            continue
        if max_tr <= 0:
            continue

        eps = max_tr * np.random.rand()
        if eps <= 0:
            continue

        h_new = h_cur.copy()
        if direction == 1:
            h_new[i] -= eps
            h_new[j] += eps
        else:
            h_new[i] += eps
            h_new[j] -= eps

        new_val = compute_c5(h_new)

        elapsed = time.time() - start
        frac = elapsed / budget
        temp = init_temp * (1.0 - frac)
        if new_val < cur_val or np.random.rand() < math.exp(-(new_val - cur_val) / (temp + 1e-12)):
            h_cur = h_new
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = h_cur.copy()

    # final projection onto constraints
    best_h = np.clip(best_h, 0.0, 1.0)
    s = best_h.sum()
    if s > 0 and abs(s - target_sum) > 1e-9:
        best_h = best_h * (target_sum / s)
        best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    # ------------------------------------------------------------------
    # Load or create initial step function
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        n0 = 200
        h0 = np.zeros(n0, dtype=float)
        half = n0 // 2
        h0[:half] = 1.0
        np.random.shuffle(h0)

    n0 = h0.shape[0]
    target_sum0 = n0 / 2.0
    s = h0.sum()
    if s > 0:
        h0 = h0 * (target_sum0 / s)
    h0 = np.clip(h0, 0.0, 1.0)
    s = h0.sum()
    if s > 0 and abs(s - target_sum0) > 1e-9:
        h0 = h0 * (target_sum0 / s)
        h0 = np.clip(h0, 0.0, 1.0)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Main simulated‑annealing optimisation on the current resolution
    # ------------------------------------------------------------------
    remaining = budget_s * 0.95 - (time.time() - start)
    if remaining > 5.0:
        best_h, best_val = sa_optimize(best_h, remaining)

    # ------------------------------------------------------------------
    # Optional refinement on a finer grid (up‑sampling)
    # ------------------------------------------------------------------
    time_used = time.time() - start
    time_left = budget_s * 0.98 - time_used
    while time_left > 30.0 and best_h.size < 400:
        new_n = min(best_h.size * 2, 400)
        h_fine = upsample_h(best_h, new_n)
        alloc = time_left * 0.5
        h_fine_opt, val_fine = sa_optimize(h_fine, alloc)
        if val_fine < best_val:
            best_h, best_val = h_fine_opt, val_fine
        time_used = time.time() - start
        time_left = budget_s * 0.98 - time_used

    # ------------------------------------------------------------------
    # Final enforce constraints and compute exact bound
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    target_sum = best_h.size / 2.0
    s = best_h.sum()
    if s > 0 and abs(s - target_sum) > 1e-9:
        best_h = best_h * (target_sum / s)
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)
    return (best_h, float(final_c5), int(best_h.size))
