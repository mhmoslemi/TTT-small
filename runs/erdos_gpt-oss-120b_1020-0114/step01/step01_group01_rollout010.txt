def local_search_binary(h_bin, max_iter, time_limit):
    best = h_bin.copy()
    best_val = compute_c5(best)
    n = len(h_bin)
    dx = 2.0 / n
    start_time = time.time()
    while time.time() - start_time < time_limit and max_iter > 0:
        # random swap of one 1 and one 0
        idx_one = np.flatnonzero(best)
        idx_zero = np.flatnonzero(1 - best) # indices where best==0
        # if no ones or zeros available break
        i = rng.choice(idx_one)
        j = rng.choice(idx_zero)
        # propose new array
        new = best.copy()
        new[i] = 0
        new[j] = 1
        val = compute_c5(new)
        if val < best_val:
            best = new
            best_val = val
            # optional: print progress
        else:
            # possibly accept with probability, but to keep simpler we can skip
            pass
        max_iter -= 1
    return best, best_val

def slsq_optimize(start_h, maxiter=600):

    n = len(start_h)
    _opt_n = n
    _opt_dx = 2.0 / n
    _opt_target_sum = n / 2.0
    h0 = project_to_feasible(start_h, _opt_target_sum)
    init_corr = np.correlate(h0, 1 - h0, mode='full') * _opt_dx
    t0 = float(np.max(init_corr) + 1e-8)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    eq_cons = {'type': 'eq', 'fun': eq_constraint_fun}
    ineq_cons = opt.NonlinearConstraint(ineq_constraint_fun, lb=np.zeros(2*n-1), ub=np.full(2*n-1, np.inf))
    x0 = np.concatenate([h0, [t0]])
    res = opt.minimize(objective_fun, x0, method='SLSQP', bounds=bounds,
                       constraints=[eq_cons, ineq_cons],
                       options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    h_opt = project_to_feasible(h_opt, _opt_target_sum)
    # compute final using evaluate_erdos_solution
    try:
        val = float(evaluate_erdos_solution(h_opt))
    except Exception:
        val = float(res.x[-1])
    return h_opt, val

def project_to_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if abs(cur - target) < 1e-12:
        return h
    # Simple proportional scaling within bounds
    # We'll apply iterative scaling: scale if cur>target else increase
    # but scaling may push outside bounds, so we need to adjust iteratively:
    # Use water-filling approach: adjust using while loop.
    # We'll implement naive approach: 
    diff = target - cur
    # Set direction:
    while abs(diff) > 1e-12:
        # Indices that can be increased if diff>0, or decreased if diff<0
        if diff > 0:
            free = np.where(h < 1 - 1e-12)[0]
        else:
            free = np.where(h > 1e-12)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        # Clip each free entry
        h[free] += delta
        # Enforce bounds again
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        diff = target - cur
        # If delta is tiny, break
        if np.abs(delta) < 1e-14:
            break
    return h

def improve_binary(a, time_limit, rng):
    n = len(a)
    best = a.copy()
    best_val = compute_c5(best)
    start = time.time()
    while time.time() - start < time_limit:
        # compute correlation counts as integer (without dx), maybe not needed
        corr_full = np.correlate(best, 1 - best, mode='full')
        max_count = np.max(corr_full)
        # get lag (could be multiple lags; take first)
        lag_index = int(np.argmax(corr_full))
        lag = lag_index - (n - 1)
        if lag < 0:
            lag = -lag
        if lag == 0 or lag >= n:
            # random swap
            i = rng.choice(np.flatnonzero(best))
            j = rng.choice(np.flatnonzero(1 - best))
            new = best.copy()
            new[i] = 0
            new[j] = 1
            val = compute_c5(new)
            if val < best_val:
                best, best_val = new, val
            continue

        # compute mismatched ones i and mismatched zeros j (positive lag)
        i_candidates = np.where((best[:n - lag] == 1) & (best[lag:] == 0))[0]
        j_candidates = np.where((best[:n - lag] == 0) & (best[lag:] == 1))[0]
        if i_candidates.size > 0 and j_candidates.size > 0:
            i = i_candidates[0]
            j = j_candidates[0]
            new = best.copy()
            new[i] = 0
            new[j] = 1
            val = compute_c5(new)
            if val < best_val:
                best, best_val = new, val
                continue
        # else fallback random swap
        i = rng.choice(np.flatnonzero(best))
        j = rng.choice(np.flatnonzero(1 - best))
        new = best.copy()
        new[i] = 0
        new[j] = 1
        val = compute_c5(new)
        if val < best_val:
            best, best_val = new, val
    return best, best_val

import time

from scipy import optimize as opt

# ----------------------------------------------------------------------
# Global placeholders for the current optimisation problem (filled in each call)
_opt_n = None          # number of points
_opt_dx = None         # grid spacing 2 / n
_opt_target_sum = None # n / 2

def objective_fun(x):
    """Objective: minimise the maximal overlap value (last variable)."""
    return x[-1]

def eq_constraint_fun(x):
    """Equality: integral of h must be 1 → sum(h) = n/2."""
    return np.sum(x[:_opt_n]) - _opt_target_sum

def ineq_constraint_fun(x):
    """Inequality: for every shift, t >= overlap_k."""
    h = x[:_opt_n]
    t = x[-1]
    corr = np.correlate(h, 1 - h, mode='full') * _opt_dx
    return t - corr  # should be >= 0 for all lags

# ----------------------------------------------------------------------
def compute_c5(h):
    """Return max_{k} ∫ h(x)(1‑h(x+k)) dx for a discrete step function."""
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_to_feasible(h, target):
    """Clip to [0,1] and adjust linearly so that sum(h)=target (approximately)."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if abs(cur - target) < 1e-12:
        return h
    # iterative water‑filling / draining
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = np.where(h < 1.0 - 1e-12)[0]
        else:
            free = np.where(h > 1e-12)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    return h

def slsq_optimize(start_h, maxiter=600):

    start = time.time()
    while time.time() - start < time_limit:
        corr_full = np.correlate(best, 1 - best, mode='full')
        max_cnt = np.max(corr_full)
        lag_index = int(np.argmax(corr_full))
        lag = lag_index - (n - 1)
        if lag < 0:
            lag = -lag
        if lag == 0 or lag >= n:
            i = rng.choice(np.flatnonzero(best))
            j = rng.choice(np.flatnonzero(1 - best))
            cand = best.copy()
            cand[i] = 0.0
            cand[j] = 1.0
            val = compute_c5(cand)
            if val < best_val:
                best, best_val = cand, val
            continue

        # mismatched ones (1,0) and candidate zeros (0,1) for this lag
        i_cand = np.where((best[:n - lag] == 1) & (best[lag:] == 0))[0]
        j_cand = np.where((best[:n - lag] == 0) & (best[lag:] == 1))[0]

        if i_cand.size and j_cand.size:
            i = i_cand[0]
            j = j_cand[0]
            cand = best.copy()
            cand[i] = 0.0
            cand[j] = 1.0
            val = compute_c5(cand)
            if val < best_val:
                best, best_val = cand, val
                continue

        # fallback random swap
        i = rng.choice(np.flatnonzero(best))
        j = rng.choice(np.flatnonzero(1 - best))
        cand = best.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        val = compute_c5(cand)
        if val < best_val:
            best, best_val = cand, val
    return best, best_val

# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    """Find a step function h on [0,2] with integral 1 that minimises C₅."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Obtain an initial vector (may be supplied by the harness)
    # ------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)
        n = init_h.shape[0]
    except Exception:
        n = 100
        init_h = np.full(n, 0.5)

    # Ensure even length – required for binary rounding later
    if n % 2 == 1:
        n += 1
        init_h = np.concatenate([init_h, np.full(1, 0.5)])

    best_h = None
    best_val = float('inf')
    best_n = n

    # ------------------------------------------------------------------
    # First optimisation from the supplied initial vector
    # ------------------------------------------------------------------
    h_opt, val_opt = slsq_optimize(init_h, maxiter=800)
    if val_opt < best_val:
        best_h, best_val, best_n = h_opt, val_opt, n

    # ------------------------------------------------------------------
    # Random restarts on the same grid size
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.55:
        rand_start = rng.random(n)
        rand_start = project_to_feasible(rand_start, n / 2.0)
        h_opt2, val_opt2 = slsq_optimize(rand_start, maxiter=600)
        if val_opt2 < best_val:
            best_h, best_val, best_n = h_opt2, val_opt2, n
        # stop early if we are out of the allocated time slice
        if time.time() - start_time > budget_s * 0.55:
            break

    # ------------------------------------------------------------------
    # Binary rounding + deterministic local improvement (if time permits)
    # ------------------------------------------------------------------
    if time.time() - start_time < budget_s * 0.90:
        bin_h = binary_round(best_h)
        remaining = budget_s - (time.time() - start_time) - 5.0
        improve_time = max(0.0, min(remaining, 300.0))
        bin_h_imp, val_bin = improve_binary(bin_h, improve_time, rng)
        if val_bin < best_val:
            best_h, best_val, best_n = bin_h_imp, val_bin, n

    # ------------------------------------------------------------------
    # Optional refinement on a finer grid (n2 = 120) if time allows
    # ------------------------------------------------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30.0:
        n2 = 120
        if n2 % 2 == 1:
            n2 += 1
        x_old = np.linspace(0.0, 2.0, n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h, left=0.0, right=0.0)
        h_interp = project_to_feasible(h_interp, n2 / 2.0)

        h_opt3, val_opt3 = slsq_optimize(h_interp, maxiter=600)
        if val_opt3 < best_val:
            best_h, best_val, best_n = h_opt3, val_opt3, n2

        # binary improvement on the finer grid
        if time.time() - start_time < budget_s * 0.98:
            bin_h3 = binary_round(best_h)
            extra = budget_s - (time.time() - start_time) - 2.0
            if extra > 5.0:
                bin_h3_imp, val_bin3 = improve_binary(bin_h3, extra, rng)
                if val_bin3 < best_val:
                    best_h, best_val, best_n = bin_h3_imp, val_bin3, n2

    # ------------------------------------------------------------------
    # Final sanity check – enforce the integral exactly once more
    # ------------------------------------------------------------------
    if best_h is None:
        best_n = n
        best_h = np.full(best_n, 0.5)
    best_h = project_to_feasible(best_h, best_n / 2.0)
    final_val = compute_c5(best_h)

    return best_h, final_val, best_n
