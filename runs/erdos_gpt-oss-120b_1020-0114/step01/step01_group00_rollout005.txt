```
def simulated_annealing(h0, time_budget, start_time):
    n = len(h0)
    h_curr = h0.copy()
    val_curr = compute_c5(h_curr)
    best_h = h_curr.copy()
    best_val = val_curr

    # Determine number of iterations based on time left, need quick loops.
    # We'll run while time < time_budget.
    # Temperature schedule: linear decay from T_initial to T_final based on elapsed fraction.
    T_initial = (best_val) * 0.1  # maybe 5% of objective
    T_final = 1e-8

    rng = np.random.default_rng()
    while time.time() - start_time < time_budget:
        # compute elapsed fraction
        frac = (time.time() - start_time) / time_budget
        T = T_initial * (1 - frac) + T_final * frac  # linear decay

        # pick i,j and delta
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # Determine direction: try moving from i to j
        # We want to ensure both remain in [0,1]
        # if h_i > 0 and h_j < 1 else swap direction
        if h_curr[i] > 0 and h_curr[j] < 1:
            max_delta = min(h_curr[i], 1.0 - h_curr[j])
            direction = (i, j)
        elif h_curr[j] > 0 and h_curr[i] < 1:
            max_delta = min(h_curr[j], 1.0 - h_curr[i])
            direction = (j, i)
        else:
            continue
        # random delta, include possibility of full transfer
        if max_delta <= 0:
            continue
        # delta drawn from uniform [0, max_delta]
        delta = max_delta * rng.random()
        # optional: sometimes use full delta with some probability to escape quickly
        # but uniform already includes near full.

        # apply perturbation
        cand = h_curr.copy()
        src, dst = direction
        cand[src] -= delta
        cand[dst] += delta

        # evaluate
        val_cand = compute_c5(cand)

        # acceptance
        if val_cand < val_curr or rng.random() < np.exp(-(val_cand - val_curr) / T):
            h_curr = cand
            val_curr = val_cand

        # record best
        if val_cand < best_val - 1e-12:
            best_h = cand.copy()
            best_val = val_cand
    return best_h, best_val
```

```
def binary_hill_climb(h_bin, max_iters):
    n = len(h_bin)
    best = h_bin.copy()
    best_val = compute_c5(best)
    curr = best.copy()
    curr_val = best_val
    rng = np.random.default_rng()
    for _ in range(max_iters):
        i = rng.integers(0, n)
        if curr[i] == 1:
            # find a zero to swap
            j = rng.integers(0, n)
            while curr[j] != 0:
                j = rng.integers(0, n)
        else:
            # i zero, find a one
            j = rng.integers(0, n)
            while curr[j] != 1:
                j = rng.integers(0, n)
        # perform swap
        cand = curr.copy()
        cand[i], cand[j] = cand[j], cand[i]
        val = compute_c5(cand)
        if val < curr_val - 1e-12: # improvement
            curr = cand
            curr_val = val
            if val < best_val:
                best = cand.copy()
                best_val = val
    return best, best_val
```

```
def continuous_simulated_annealing(h0, start_time, run_time):
    n = len(h0)
    h = h0.copy()
    val = compute_c5(h)
    best_h = h.copy()
    best_val = val
    rng = np.random.default_rng()
    # Temperature schedule: start T0 = (val) * 0.5 (or val * 0.2)
    T0 = val * 0.1  # maybe 0.04
    T_end = 1e-8
    # compute total iterations estimate: we will loop while time < start_time+run_time.
    while time.time() - start_time < run_time:
        # compute temperature based on progress
        elapsed = time.time() - start_time
        frac = elapsed / run_time
        T = T0 * (1 - frac) + T_end * frac
        # random move
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # direction
        if h[i] > 0.0 and h[j] < 1.0:
            max_delta = min(h[i], 1.0 - h[j])
            src, dst = i, j
        elif h[j] > 0.0 and h[i] < 1.0:
            max_delta = min(h[j], 1.0 - h[i])
            src, dst = j, i
        else:
            continue
        # choose delta, maybe uniform but also sometimes full
        delta = max_delta * rng.random()
        # small tolerance: if delta < 1e-12 continue...
        if delta <= 0:
            continue
        cand = h.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand_val = compute_c5(cand)
        # acceptance
        if cand_val < val or rng.random() < np.exp(-(cand_val - val) / T):
            h = cand
            val = cand_val
            if cand_val < best_val:
                best_h = cand.copy()
                best_val = cand_val
    return best_h, best_val
```

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def continuous_simulated_annealing(h0, max_time):
    n = h0.shape[0]
    h = h0.copy()
    val = compute_c5(h)
    best_h = h.copy()
    best_val = val
    rng = np.random.default_rng()
    t0 = time.time()
    T0 = max(val, 1e-3)
    T_end = 1e-9
    while time.time() - t0 < max_time:
        elapsed = time.time() - t0
        frac = elapsed / max_time
        T = T0 * (1 - frac) + T_end * frac
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if h[i] > 0.0 and h[j] < 1.0:
            max_delta = min(h[i], 1.0 - h[j])
            src, dst = i, j
        elif h[j] > 0.0 and h[i] < 1.0:
            max_delta = min(h[j], 1.0 - h[i])
            src, dst = j, i
        else:
            continue
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random()
        # apply delta
        cand = h.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand_val = compute_c5(cand)
        if cand_val < val or rng.random() < np.exp(-(cand_val - val) / T):
            h = cand
            val = cand_val
            if cand_val < best_val:

                best_val = cand_val

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def continuous_simulated_annealing(h0, max_time):
    """SA on the continuous feasible region (preserving the total mass)."""
    n = h0.shape[0]
    h = h0.copy()
    val = compute_c5(h)
    best_h, best_val = h.copy(), val
    rng = np.random.default_rng()
    t0 = time.time()
    T0, T_end = max(val, 1e-3), 1e-9
    while time.time() - t0 < max_time:
        elapsed = time.time() - t0
        frac = elapsed / max_time
        T = T0 * (1 - frac) + T_end * frac
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if h[i] > 0.0 and h[j] < 1.0:
            max_delta = min(h[i], 1.0 - h[j])
            src, dst = i, j
        elif h[j] > 0.0 and h[i] < 1.0:
            max_delta = min(h[j], 1.0 - h[i])
            src, dst = j, i
        else:
            continue
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random()
        cand = h.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand_val = compute_c5(cand)
        if cand_val < val or rng.random() < np.exp(-(cand_val - val) / T):
            h, val = cand, cand_val
            if cand_val < best_val:
                best_h, best_val = cand.copy(), cand_val
    return best_h, best_val

def binary_simulated_annealing(h0, max_time):

        T = T0 * (1 - frac) + T_end * frac
        i = rng.integers(0, n)
        if h[i] == 1:
            j = rng.integers(0, n)

def binary_hill_climb(h0, max_iters):
    """Greedy pairwise swaps on a binary vector."""
    n = h0.shape[0]
    h = h0.copy()
    val = compute_c5(h)
    best_h, best_val = h.copy(), val
    rng = np.random.default_rng()
    for _ in range(max_iters):
        i = rng.integers(0, n)
        if h[i] == 1:
            j = rng.integers(0, n)
            while h[j] != 0:
                j = rng.integers(0, n)
        else:
            j = rng.integers(0, n)
            while h[j] != 1:
                j = rng.integers(0, n)
        cand = h.copy()
        cand[i], cand[j] = cand[j], cand[i]
        cand_val = compute_c5(cand)
        if cand_val < val:
            h, val = cand, cand_val
            if cand_val < best_val:
                best_h, best_val = cand.copy(), cand_val

def random_continuous(n, rng):
    """Generate a random feasible continuous vector (sum = n/2, 0≤h≤1)."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)

    diff = n / 2.0 - cur

    if mask.any():
        v[mask] += diff / mask.sum()
        v = np.clip(v, 0.0, 1.0)
    else:
        v *= (n / 2.0) / cur
    return v
