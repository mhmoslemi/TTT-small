from scipy.optimize import minimize

def slsqp_refine(h_start, t_start, max_iter=200):
    x0 = np.concatenate([h_start, [t_start]])
    bounds = [(0.0,1.0)]*N + [(0.0, 2.0)]  # t bound maybe up to 2.
    def fun(x):
        return x[-1]  # t
    def constr_eq(x):
        return np.sum(x[:-1]) - target_sum
    def constr_ineq(x):
        h = x[:-1]
        t = x[-1]
        O = np.correlate(h, 1 - h, mode='full') * (2.0 / N)
        return t - O
    constraints = [{'type':'eq','fun': constr_eq},
                   {'type':'ineq','fun': constr_ineq}]
    res = minimize(fun, x0, method='SLSQP', bounds=bounds, constraints=constraints,
                   options={'maxiter': max_iter, 'ftol':1e-9, 'disp':False})
    return res

import numpy as np
import math, time, random

def _project(v, target):
    """Project v onto [0,1]^n while enforcing sum(v)=target."""
    v = np.clip(v, 0.0, 1.0)
    while True:
        diff = target - v.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = v < 1.0 - 1e-12
        else:
            free = v > 1e-12
        nfree = free.sum()
        if nfree == 0:
            break
        delta = diff / nfree
        if diff > 0:
            v[free] = np.minimum(v[free] + delta, 1.0)
        else:
            v[free] = np.maximum(v[free] + delta, 0.0)
    return v


def _c5_of(h):
    """Compute C₅ for a step vector h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)


def _grad_shift(h, shift):
    """Gradient of the overlap for a single shift."""
    N = h.size
    g = np.zeros(N)
    if shift >= 0:
        i = np.arange(N - shift)
        j = i + shift
    else:
        s = -shift
        i = np.arange(s, N)
        j = i - s
    g[i] += 1.0
    g[i] -= h[j]
    g[j] -= h[i]
    return g


def _shape_h(N, r):
    """Triangular‑with‑plateau shape parameterised by ramp length r∈[0,1]."""
    dx = 2.0 / N
    xs = (np.arange(N) + 0.5) * dx
    if r <= 0.0:
        return np.full(N, 0.5)
    if r >= 1.0:
        return np.minimum(xs, 2.0 - xs)   # pure triangle
    h0 = 1.0 / (2.0 - r)                 # plateau height

    h[right] = ((2.0 - xs[right]) / r) * h0
