import numpy as np

def _project(v, target):
    """Project v onto the box [0,1] and the hyperplane sum(v)=target."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    # water‑filling style loop – converges in a few passes
    while abs(diff) > 1e-12:
        if diff > 0:
            free = v < 1.0 - 1e-12
            nfree = free.sum()
            if nfree == 0:
                break
            inc = diff / nfree
            v[free] = np.minimum(v[free] + inc, 1.0)
        else:
            free = v > 1e-12
            nfree = free.sum()
            if nfree == 0:
                break
            dec = diff / nfree
            v[free] = np.maximum(v[free] + dec, 0.0)
        diff = target - v.sum()
    return v


def _c5_of(h):
    """Return the overlap constant C₅ for a step vector h."""
    N = h.size

    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

    dx = 2.0 / N
    x = np.arange(N) * dx + dx / 2.0
    h = np.abs(x - 1.0)
    return np.clip(h, 0.0, 1.0)

        i, j = np.random.randint(N, size=2)
        if i == j:
            continue
        # feasible delta keeping i,j inside [0,1]
        lower = max(-cur_h[i], cur_h[j] - 1.0)
        upper = min(1.0 - cur_h[i], cur_h[j])
        if lower >= upper:
            continue
        delta_low = max(lower, -step)
        delta_high = min(upper, step)
        if delta_low >= delta_high:
            continue
        delta = np.random.uniform(delta_low, delta_high)

        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_h[i] = np.clip(new_h[i], 0.0, 1.0)
        new_h[j] = np.clip(new_h[j], 0.0, 1.0)

        new_val = _c5_of(new_h)

        if new_val < cur_val or np.random.rand() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur_h, cur_val = new_h, new_val
            if new_val < best_val - 1e-12:
                best_h, best_val = new_h.copy(), new_val
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 50000:
            # gentle restart near the best point
            cur_h = best_h.copy()
            jitter = np.random.uniform(-step, step, size=N)
            cur_h += jitter
            cur_h = _project(cur_h, target_sum)
            cur_val = _c5_of(cur_h)
            no_improve = 0

    # final projection & exact evaluation (if helper is present)
    best_h = _project(best_h, target_sum)
    final_val = _c5_of(best_h)
    if "evaluate_erdos_solution" in globals():
        try:
            final_val = evaluate_erdos_solution(best_h.tolist())
        except Exception:
            pass

    return (best_h.tolist(), float(final_val), int(N))
