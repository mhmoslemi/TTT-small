import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _project_onto_bounded_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto {x : 0 <= x_i <= 1, sum x_i = target}."""
    # binary search for the Lagrange multiplier τ such that sum clip(v-τ) = target
    lo = np.min(v) - 1.0          # guarantees v-τ >= 1 for all i
    hi = np.max(v)                # guarantees v-τ <= 0 for all i
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    τ = (lo + hi) * 0.5
    return np.clip(v - τ, 0.0, 1.0)


def _initial_h(N: int, target: float, rng: np.random.Generator) -> np.ndarray:
    """Obtain a starting vector; use the supplied ``initial_h_values`` if possible."""
    try:
        h0 = np.array(initial_h_values, dtype=float)  # type: ignore[name-defined]
        if h0.shape[0] != N:
            # linear interpolation to required length
            old_x = np.linspace(0.0, 2.0, h0.shape[0], endpoint=False)
            new_x = np.linspace(0.0, 2.0, N, endpoint=False)
            h0 = np.interp(new_x, old_x, h0)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = _project_onto_bounded_simplex(h0, target)
        return h0
    except Exception:
        # fallback: random vector respecting the constraints
        v = rng.random(N)
        return _project_onto_bounded_simplex(v, target)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # problem size – a modest number gives many iterations within the budget
    N = 150
    target_sum = N / 2.0

    # ------------------------------------------------------------------
    # 1. Initialise best solution
    # ------------------------------------------------------------------
    h_best = _initial_h(N, target_sum, rng)
    best_c5 = _c5_value(h_best)

    # ------------------------------------------------------------------
    # 2. Optimisation loop (multiple restarts)
    # ------------------------------------------------------------------
    # Hyper‑parameters
    max_total_iters = 120000            # total inner iterations across restarts
    it = 0
    while it < max_total_iters and (time.time() - start) < budget_s * 0.96:
        # start a fresh trial (occasionally randomise)
        if it == 0 or it % 25000 == 0:
            h = _initial_h(N, target_sum, rng)

        # inner optimisation (smooth max‑overlap descent)
        inner_iters = min(3000, max_total_iters - it)
        for _ in range(inner_iters):
            if (time.time() - start) >= budget_s * 0.96:
                break
            # schedule for the smoothing parameter α and the step size η
            α = 5.0 + (120.0 - 5.0) * (it / max_total_iters)   # grows from 5 to 120
            η = 0.6 * (0.9998 ** it)                         # slowly decays

            # --------------------------------------------------------------
            # a) linear autocorrelation O(s) = Σ_i h_i h_{i+s}
            # --------------------------------------------------------------
            corr_full = np.correlate(h, h, mode="full")
            O = corr_full[N - 1 :]                     # O[0] is shift 0, length N

            # --------------------------------------------------------------
            # b) soft‑minimum weights (larger for smaller O)
            # --------------------------------------------------------------
            min_O = O.min()
            exp_arg = -α * (O - min_O)                 # min exponent = 0
            w = np.exp(exp_arg)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w /= w_sum

            # --------------------------------------------------------------
            # c) gradient of the smooth surrogate:
            #    g_i = Σ_s w_s (h_{i+s} + h_{i-s})
            # --------------------------------------------------------------
            grad = np.zeros_like(h)
            # forward contributions (i + s)
            for s in range(N):
                ws = w[s]
                if ws == 0.0:
                    continue
                if s == 0:
                    grad += 2.0 * ws * h
                else:
                    grad[: N - s] += ws * h[s:]
                    grad[s:]       += ws * h[: N - s]

            # --------------------------------------------------------------
            # d) step and projection
            # --------------------------------------------------------------
            h_new = h + η * grad
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = _project_onto_bounded_simplex(h_new, target_sum)

            # evaluate current candidate
            cur_c5 = _c5_value(h_new)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                h_best = h_new.copy()

            h = h_new
            it += 1

        # occasional random perturbation to escape possible stagnation
        if (time.time() - start) < budget_s * 0.96:
            noise = rng.normal(scale=0.03, size=N)
            h = _project_onto_bounded_simplex(np.clip(h + noise, 0.0, 1.0), target_sum)

    # ------------------------------------------------------------------
    # 3. Final safeguard – ensure feasibility of the reported vector
    # ------------------------------------------------------------------
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = _project_onto_bounded_simplex(h_best, target_sum)

    return (h_best.tolist(), float(best_c5), N)
