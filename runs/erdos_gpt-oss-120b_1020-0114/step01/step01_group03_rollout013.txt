def compute_c5_fast(h):
    n = h.shape[0]
    g = 1.0 - h
    N = int(2**np.ceil(np.log2(2*n-1)))
    fh = np.fft.rfft(h, N)
    fg_rev = np.fft.rfft(g[::-1], N)  # reversed g
    conv = np.fft.irfft(fh * fg_rev, N)
    # conv length N
    corr = conv[:2*n-1]  # linear conv
    # shift index: shift k maps to corr[n-1 + k]
    # Include negative shifts as well maybe but max over all corr
    # The max will be over all shift values, so just take max of corr.
    corr = corr * (2.0 / n)
    return np.max(corr)

def normalize(h):
    # Clip, then scale to exact sum = n/2
    n = h.size
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s > 0:
        h = h * (target / s)
    # Now ensure within bounds
    h = np.clip(h, 0.0, 1.0)
    # Minor correction if sum off due to clipping
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        # final scaling if needed
        s = h.sum()
        if s > 0:
            h = h * (target / s)
    return h
```

def upsample(h, new_n):
    old_n = h.size
    if new_n == old_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, h)
    return normalize(new_h)
```

        eps = np.random.uniform(0.01, 0.05)
        # ensure we don't violate bounds
        eps = min(eps, best_h[i], 1.0 - best_h[j])
        if eps <= 0:

        new_h = best_h.copy()
        new_h[i] -= eps

        if new_c5 < best_c5 - 1e-12:

            best_c5 = new_c5
        steps -= 1
    return best_h, best_c5

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    g = 1.0 - h
    N = int(2**np.ceil(np.log2(2*n-1)))
    # use rfft for speed
    H = np.fft.rfft(h, N)
    G_rev = np.fft.rfft(g[::-1], N)
    conv = np.fft.irfft(H * G_rev, N)
    # extract linear conv length 2n-1
    conv = conv[:2*n-1]
    # max absolute? We need max value of conv (non-negative)

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    g = 1.0 - h
    N = int(2**np.ceil(np.log2(2 * n - 1)))
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(g[::-1], N)
    conv = np.fft.irfft(H * G, N)
    conv = conv[:2*n-1]
    max_corr = np.max(conv)
    return max_corr * dx

def normalize(h):
    n = h.size
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s > 0:
        h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if s > 0:
            h = h * (target / s)
    return h

def upsample(h, new_n):
    old_n = h.size
    if new_n == old_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, h)
    return normalize(new_h)
