def gradient_for_shift(h, s):
    n = len(h)
    grad = np.empty(n)
    # term1
    if s >= 0:
        grad[:n-s] = 1 - h[s:]
        grad[n-s:] = 0
        # term2
        grad[s:] -= h[:n-s]
        # for j< s, term2 not subtracted.
    else:
        s_neg = -s
        grad[s_neg:] = 1 - h[:n-s_neg]
        grad[:s_neg] = 0
        grad[:n-s_neg] -= h[s_neg:]
    return grad
```

import time
import numpy as np

def _evaluate(h: np.ndarray) -> float:
    """Maximum overlap integral for a given step function."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return float(corr.max())

def _project(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0≤x≤1, sum x = target} using bisection on a shift."""
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def _gradient_for_shift(h: np.ndarray, s: int) -> np.ndarray:
    """Gradient of the overlap for a single shift s (integer index)."""
    n = h.size
    grad = np.zeros_like(h)
    if s >= 0:
        # (1 - h_{i+s}) term
        grad[: n - s] = 1.0 - h[s:]
        # -h_{i-s} term
        grad[s:] -= h[: n - s]
    else:
        s = -s
        # (1 - h_{i-s}) term
        grad[s:] = 1.0 - h[: n - s]
        # -h_{i+s} term
        grad[: n - s] -= h[s:]
    return grad

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # -------------------- problem size --------------------
    n = 160                     # even -> exact half‑mass achievable with binary values
    target_sum = n / 2.0        # ∫h = 1  ⇒  sum(h) = n/2

    # -------------------- initial vector --------------------
    try:
        h0 = np.array(initial_h_values, dtype=float)   # provided by harness, may be None
        # Upsample to n points via linear interpolation
        x_old = np.linspace(0.0, 2.0, h0.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x_new, x_old, h0)
    except Exception:
        h = np.random.rand(n)    # random start if no parent

    h = _project(h, target_sum)

    best_h = h.copy()
    best_c5 = _evaluate(best_h)

    # -------------------- optimisation parameters --------------------
    max_iters = 200000
    step_init = 1.0
    last_improve = 0
    jitter_scale = 0.04
    jitter_interval = 1200
    restart_interval = 20000

    it = 0
    while it < max_iters and time.time() - start < budget_s * 0.94:
        # current evaluation
        cur_c5 = _evaluate(h)

        # correlation and active shifts (those attaining the max)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = corr.max()
        # use a tiny absolute tolerance to collect all active indices
        active = np.where(corr >= max_corr - 1e-12)[0]

        # average sub‑gradient over active shifts
        grad = np.zeros_like(h)
        for idx in active:
            shift = idx - (n - 1)
            grad += _gradient_for_shift(h, shift)
        grad /= len(active)

        # ----------- back‑tracking line search -------------
        step = step_init
        for _ in range(15):
            cand = _project(h - step * grad, target_sum)
            cand_c5 = _evaluate(cand)

            if cand_c5 < best_c5 - 1e-12:
                # improvement over the best ever found
                best_h, best_c5 = cand.copy(), cand_c5
                h = cand
                last_improve = it
                step_init = min(step_init * 1.07, 2.0)
                break

            if cand_c5 <= cur_c5 + 1e-12:
                # non‑worsening step, accept
                h = cand
                step_init = max(step_init * 0.92, 1e-5)
                break

            step *= 0.5
        else:
            step_init = max(step_init * 0.5, 1e-5)

        # ------------------- jitter to escape plateaus -------------------
        if it - last_improve > jitter_interval:
            jitter = np.random.normal(scale=jitter_scale, size=n)
            h_jit = _project(h + jitter, target_sum)
            c5_jit = _evaluate(h_jit)
            if c5_jit < best_c5:
                best_h, best_c5 = h_jit.copy(), c5_jit
                h = h_jit
                last_improve = it
                # shrink jitter a little after a successful move
                jitter_scale = max(jitter_scale * 0.9, 0.005)
            else:
                # accept if it does not worsen the current point
                if c5_jit < cur_c5:
                    h = h_jit
                    last_improve = it

        # ------------------- occasional random restart -------------------
        if it - last_improve > restart_interval:
            # new random feasible point
            h_rand = np.random.rand(n)
            h = _project(h_rand, target_sum)
            # reset step size a bit
            step_init = 1.0
            # keep best so far untouched
            last_improve = it

        it += 1

    # Final projection to guarantee feasibility
    best_h = _project(best_h, target_sum)
    best_c5 = _evaluate(best_h)

    return best_h.tolist(), best_c5, n
