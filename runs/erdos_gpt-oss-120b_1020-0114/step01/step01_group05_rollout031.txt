def subgrad(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx   # length 2n-1
    idx = np.argmax(corr)
    k = idx - (n - 1)   # integer lag
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n - k] += 1 - h[k:]
        grad[k:] -= h[:n - k]
    else:
        k = -k
        grad[k:] += 1 - h[:n - k]
        grad[:n - k] -= h[k:]
    grad *= dx
    return grad, corr[idx]

def _project(v, target):
    # v is initial vector (maybe out of bounds)
    # Clip to [0,1]
    v = np.clip(v, 0.0, 1.0)
    # Then find lambda such that sum(clip(v+lambda,0,1)) = target
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) / 2
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) / 2, 0.0, 1.0)
```

import time
import numpy as np

# ---------- helper utilities -------------------------------------------------
def _project(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the set {0≤x≤1, sum x = target}."""
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _c5(h: np.ndarray) -> float:
    """Exact overlap constant for a discretised h."""
    n = h.size
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full") * dx)


def _subgrad(h: np.ndarray):
    """Sub‑gradient of the max‑overlap functional at h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    idx = int(np.argmax(corr))
    k = idx - (n - 1)          # integer lag achieving the maximum

    grad = np.zeros_like(h)
    if k >= 0:
        # O_k = Σ_{i=0}^{n-1-k} h[i]*(1-h[i+k])
        grad[: n - k] += 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    else:
        kk = -k
        # O_-kk = Σ_{i=kk}^{n-1} h[i]*(1-h[i-kk])
        grad[kk:] += 1.0 - h[: n - kk]
        grad[: n - kk] -= h[kk:]
    grad *= dx
    return grad, corr[idx]


# ---------- main optimisation routine ---------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # ----- initialise problem size -----------------------------------------
    try:
        init_vec = np.array(initial_h_values, dtype=float)  # provided by harness
        n = init_vec.size
    except Exception:
        n = 80
        init_vec = np.random.rand(n)

    target_sum = n * 0.5          # ∫_0^2 h = 1  →  Σ h = n/2

    # ----- keep track of the best solution ---------------------------------
    best_h = None
    best_c5 = np.inf

    # ----- different starting families --------------------------------------
    starters = []

    # use the supplied vector (if any)
    starters.append(init_vec.copy())

    # a simple monotone ramp
    starters.append(np.linspace(0.0, 1.0, n))

    # a Beta(0.5,0.5) “U‑shaped” sample
    starters.append(np.random.beta(0.5, 0.5, size=n))

    # a few pure uniform random vectors
    for _ in range(2):
        starters.append(np.random.rand(n))

    # ----- optimisation parameters ------------------------------------------
    max_iters_per_restart = 6000   # generous but still cheap (O(n²) each)
    total_restarts = len(starters)

    # ----- optimisation loop -----------------------------------------------
    for restart_id, start_h in enumerate(starters):
        if time.time() - start_time > budget_s * 0.95:
            break

        h = _project(start_h, target_sum)
        cur_c5 = _c5(h)

        # local best for this restart
        local_best_h = h.copy()
        local_best_c5 = cur_c5

        # learning‑rate schedule (simple decay)
        lr = 0.25
        last_improve = 0

        for it in range(max_iters_per_restart):
            if time.time() - start_time > budget_s * 0.95:

            grad, cur_c5 = _subgrad(h)

            # Polyak‑type step size (fallback to lr when no improvement yet)
