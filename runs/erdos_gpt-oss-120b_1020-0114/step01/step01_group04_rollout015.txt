def project_onto_box_and_sum(y, target_sum):
    # find lambda such that sum_i clip(y_i + lambda, 0, 1) = target_sum
    # use bisection
    lo = -1.0
    hi = 1.0
    # widen range sufficiently:
    lo = -max(y) - 1.0
    hi = 1.0 - min(y) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    x = np.clip(y + (lo+hi)/2.0, 0.0, 1.0)
    # due to rounding may be off by epsilon, enforce sum by adjusting small epsilon over free coordinates
    diff = target_sum - x.sum()
    if abs(diff) > 1e-9:
        free = np.where((x > 1e-12) & (x < 1-1e-12))[0]
        if free.size:
            x[free] += diff / free.size
            x = np.clip(x, 0.0, 1.0)
        else:
            # adjust a single coordinate within bounds
            i = np.argmin(np.abs(x - 0.5))
            x[i] = min(1.0, max(0.0, x[i] + diff))
    return x

def greedy_optimize_binary(h, time_limit):
    n = len(h)
    target_sum = n/2
    best_h = h.copy()
    best_val = _compute_c5(best_h)
    cur_h = h.copy()
    cur_val = best_val
    start = time.time()
    T0 = 0.1
    iteration = 0
    while time.time() - start < time_limit:
        # compute worst shift
        shift,_ = _worst_shift(cur_h)
        w = _w_vector(cur_h, shift)
        # find up (h=0) and down (h=1) etc.
        up_mask = cur_h < 0.5  # treat close to 0
        down_mask = cur_h > 0.5
        if not up_mask.any() or not down_mask.any():
            break
        i_up = np.where(up_mask)[0][np.argmax(w[up_mask])]
        i_down = np.where(down_mask)[0][np.argmin(w[down_mask])]
        delta = 1.0
        new_h = cur_h.copy()
        new_h[i_up] = 1.0
        new_h[i_down] = 0.0
        new_val = _compute_c5(new_h)

        # acceptance
        if new_val < cur_val - 1e-12:
            cur_h, cur_val = new_h, new_val
            if new_val < best_val:
                best_val, best_h = new_val, new_h.copy()
        else:
            # annealing
            temperature = T0 * (1 - (time.time() - start)/time_limit)
            if temperature > 0 and np.random.rand() < math.exp(-(new_val - cur_val) / temperature):
                cur_h, cur_val = new_h, new_val
        iteration += 1
    # after loop, ensure integer and sum
    cur_sum = best_h.sum()
    diff = target_sum - cur_sum
    if diff != 0:
        # adjust by flipping diff bits: if diff>0 need some zeros to ones
        zeros = np.where(best_h < 0.5)[0]
        ones = np.where(best_h > 0.5)[0]
        if diff > 0:
            # set diff zeros to 1
            chosen = np.random.choice(zeros, int(diff), replace=False)
            best_h[chosen] = 1.0
        else:
            # need to turn some ones to 0
            chosen = np.random.choice(ones, int(-diff), replace=False)
            best_h[chosen] = 0.0
    best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, _compute_c5(best_h)

import time, math, numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return the discretised C5 value for a step function h on [0,2]."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)


def _worst_shift(h: np.ndarray):
    """Return the shift giving the maximal overlap and the full correlation array."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)                     # shift = index‑(n‑1)
    return shift, corr


def _w_vector(h: np.ndarray, shift: int):
    """For a given shift, w_i = h_{i+shift}+h_{i-shift} (zero outside)."""
    n = h.shape[0]
    if shift == 0:
        return 2.0 * h
    w = np.zeros_like(h)
    if shift > 0:
        L = n - shift
        w[:L] += h[shift:]          # h_{i+shift}
        w[shift:] += h[:L]          # h_{i-shift}
    else:  # shift < 0
        s = -shift
        L = n - s
        w[s:] += h[:L]              # h_{i+shift}
        w[:L] += h[s:]              # h_{i-shift}
    return w


def _project_box_sum(y: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of y onto {x | 0≤x_i≤1, sum x_i = target }.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = -max(y) - 1.0
    hi = 1.0 - min(y) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(y + (lo + hi) / 2.0, 0.0, 1.0)
    # tiny correction if rounding left a small error
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = np.where((x > 1e-12) & (x < 1.0 - 1e-12))[0]
        if free.size:
            x[free] += diff / free.size
        else:
            i = np.argmin(np.abs(x - 0.5))
            x[i] = min(1.0, max(0.0, x[i] + diff))
        x = np.clip(x, 0.0, 1.0)
    return x


def _greedy_optimize_binary(h0: np.ndarray, time_limit: float):
    """
    Hill‑climbing / simulated‑annealing on a binary (0/1) vector.
    Swaps a 1 with a 0 guided by the worst shift.
    """
    n = h0.shape[0]
    target_sum = n / 2.0
    assert abs(h0.sum() - target_sum) < 1e-12, "initial vector must satisfy the mass constraint"

    best_h = h0.copy()
    best_val = _compute_c5(best_h)

    cur_h = h0.copy()
    cur_val = best_val
    start = time.time()
    no_improve = 0
    T0 = 0.1                     # initial temperature (relative)

    while time.time() - start < time_limit:
        shift, _ = _worst_shift(cur_h)
        w = _w_vector(cur_h, shift)

        up_mask = cur_h < 0.5 - 1e-12
        down_mask = cur_h > 0.5 + 1e-12
        if not up_mask.any() or not down_mask.any():
            break

        i_up = np.where(up_mask)[0][np.argmax(w[up_mask])]
        i_down = np.where(down_mask)[0][np.argmin(w[down_mask])]

        new_h = cur_h.copy()
        new_h[i_up] = 1.0
        new_h[i_down] = 0.0
        new_val = _compute_c5(new_h)

        if new_val < cur_val - 1e-12:
            cur_h, cur_val = new_h, new_val
            no_improve = 0
            if cur_val < best_val:
                best_val, best_h = cur_val, cur_h.copy()
        else:
            # simulated‑annealing acceptance
            elapsed = time.time() - start
            T = T0 * max(0.0, 1.0 - elapsed / time_limit)
            if T > 0 and np.random.rand() < math.exp(-(new_val - cur_val) / T):
                cur_h, cur_val = new_h, new_val
            no_improve += 1
            if no_improve > 5000:
                # restart from the best seen so far
                cur_h, cur_val = best_h.copy(), best_val
                no_improve = 0

    # final integer correction (should already be binary)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            zeros = np.where(best_h < 0.5)[0]
            if zeros.size >= diff:
                pick = np.random.choice(zeros, int(diff), replace=False)
                best_h[pick] = 1.0
        else:
            ones = np.where(best_h > 0.5)[0]
            if ones.size >= -diff:
                pick = np.random.choice(ones, int(-diff), replace=False)
                best_h[pick] = 0.0
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, _compute_c5(best_h)


def _smooth_grad_optimize(h0: np.ndarray, time_limit: float):
    """
    Gradient descent on a smooth approximation of the max‑overlap.
    Keeps the box and mass constraints via projection.
    """
    n = h0.shape[0]
    target = n / 2.0
    h = h0.copy()
    start = time.time()
    step0 = 0.05
    alpha = 1.0                     # temperature for soft‑max
    it = 0
    while time.time() - start < time_limit:
        # overlap values for every shift
        O = np.correlate(h, 1.0 - h, mode="full")
        # smooth max weights
        maxO = O.max()
        exp = np.exp(alpha * (O - maxO))
        w = exp / exp.sum()

        # gradient of each overlap (see analysis)
        grad = np.zeros_like(h)
        for idx, weight in enumerate(w):
            if weight == 0.0:
                continue
            s = idx - (n - 1)
            if s >= 0:
                L = n - s
                grad[:L] += weight
                grad[:L] -= weight * h[s:s + L]
                grad[s:s + L] -= weight * h[:L]
            else:
                r = -s
                L = n - r
                grad[r:] += weight
                grad[r:] -= weight * h[:L]
                grad[:L] -= weight * h[r:]

        # diminishing step size
        it += 1
        eta = step0 / math.sqrt(it)
        h_new = h - eta * grad
        h = _project_box_sum(h_new, target)

        # slowly increase the sharpness of the soft‑max
        if it % 500 == 0 and alpha < 30.0:
            alpha = min(30.0, alpha * 1.3)

    return h, _compute_c5(h)
