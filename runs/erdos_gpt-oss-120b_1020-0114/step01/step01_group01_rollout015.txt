    h = start_h.copy()
    cur_val = start_val
    best_h = h.copy()
    best_val = cur_val
    n = h.shape[0]
    target_sum = n/2.0

    # parameters
    temp = 0.02
    min_temp = 1e-6
    decay = 0.9998
    step_max = 0.2  # max delta per move

    while time.time() < start_time + time_limit:
        # propose move
        i = rng.integers(n)
        # ensure able to increase
        if h[i] >= 1.0:
            continue
        j = rng.integers(n)
        if h[j] <= 0.0:
            continue
        # Determine delta
        max_inc = min(step_max, 1.0 - h[i])
        max_dec = min(step_max, h[j])
        delta = min(max_inc, max_dec) * rng.random()
        if delta == 0:
            continue
        # apply
        h[i] += delta
        h[j] -= delta
        # compute val
        new_val = float(evaluate_erdos_solution(h))
        # acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / temp):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # reject: revert
            h[i] -= delta
            h[j] += delta
        # update temp
        temp = max(min_temp, temp * decay)
    return best_h, best_val

    best_h = h.copy()
    best_val = cur_val
    cur_h = h
    cur_val = cur_val
    # parameters
    temp = 0.02
    min_temp = 1e-6
    decay = 0.9997
    step_max = 0.2
    while time.time() - start_time < max_duration:
        # randomly pick inc and dec indices
        inc_candidates = np.where(cur_h < 1.0)[0]
        dec_candidates = np.where(cur_h > 0.0)[0]
        if inc_candidates.size == 0 or dec_candidates.size == 0:
            break
        i = rng.choice(inc_candidates)
        j = rng.choice(dec_candidates)
        if i == j:
            continue
        max_inc = min(step_max, 1.0 - cur_h[i])
        max_dec = min(step_max, cur_h[j])
        max_delta = min(max_inc, max_dec)
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random()
        # apply
        cur_h[i] += delta
        cur_h[j] -= delta
        # compute new val
        new_val = compute_c5(cur_h, dx)
        # acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] -= delta
            cur_h[j] += delta
        # temperature schedule
        temp = max(min_temp, temp * decay)
    return best_h, best_val

import time
import numpy as np

def _project_to_feasible(v, target, rng):
    """Clip to [0,1] and adjust so that sum(v) == target (approximately)."""
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    diff = target - cur
    for _ in range(20):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = np.where(v < 1.0)[0]
            if free.size == 0:
                break
            inc = diff / free.size
            inc = np.minimum(inc, 1.0 - v[free])
            v[free] += inc
        else:
            free = np.where(v > 0.0)[0]
            if free.size == 0:
                break
            dec = diff / free.size
            dec = np.maximum(dec, -v[free])
            v[free] += dec
        cur = v.sum()
        diff = target - cur
    if abs(diff) > 1e-12:
        idx = rng.integers(len(v))
        v[idx] = np.clip(v[idx] + diff, 0.0, 1.0)
    return v

def _compute_c5(h, dx):
    """Fast C5 evaluation using np.correlate."""
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def _anneal(start_h, start_val, duration, dx, rng):
    """Simulated‑annealing style improvement for a fixed time budget."""
    n = start_h.shape[0]
    best_h = start_h.copy()
    best_val = start_val
    cur_h = start_h.copy()
    cur_val = start_val

    # annealing parameters
    temp = 0.02
    min_temp = 1e-6
    decay = 0.9997
    step_max = 0.2

    end_time = time.time() + duration
    while time.time() < end_time:
        # choose one index to increase, one to decrease
        inc_idx = rng.choice(np.where(cur_h < 1.0)[0]) if np.any(cur_h < 1.0) else None
        dec_idx = rng.choice(np.where(cur_h > 0.0)[0]) if np.any(cur_h > 0.0) else None
        if inc_idx is None or dec_idx is None:
            break
        if inc_idx == dec_idx:
            continue
        # admissible delta
        max_inc = min(step_max, 1.0 - cur_h[inc_idx])
        max_dec = min(step_max, cur_h[dec_idx])
        max_delta = min(max_inc, max_dec)
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random()
        # apply move
        cur_h[inc_idx] += delta
        cur_h[dec_idx] -= delta

        new_val = _compute_c5(cur_h, dx)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[inc_idx] -= delta
            cur_h[dec_idx] += delta

        temp = max(min_temp, temp * decay)

    return best_h, best_val

            cur_h = best_h.copy()
        else:
            cur_h = rng.random(n)
            cur_h = _project_to_feasible(cur_h, target_sum, rng)

        cur_val = eval_h(cur_h)

        # Run a short annealing session (max 8 s or whatever is left)
        duration = min(remaining, 8.0)
        cur_h, cur_val = _anneal(cur_h, cur_val, duration, dx, rng)

        if cur_val < best_val:
            best_val = cur_val
            best_h = cur_h.copy()

    # ----------------------------------------------------------------------
    # Final projection and exact evaluation (the harness may use a stricter checker)
    # ----------------------------------------------------------------------
    best_h = _project_to_feasible(best_h, target_sum, rng)
    final_c5 = float(evaluate_erdos_solution(best_h))

    return best_h, final_c5, n
