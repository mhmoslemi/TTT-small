import time, numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def random_binary(n):
    # returns binary vector with n/2 ones (float)
    half = n // 2
    v = np.zeros(n, dtype=float)
    v[:half] = 1.0
    np.random.shuffle(v)
    return v

def mutate_swap(v, swaps=1):
    # expects binary vector
    v2 = v.copy()
    n = v2.size
    for _ in range(swaps):
        # find indices with 1 and 0
        ones = np.where(v2 > 0.5)[0]
        zeros = np.where(v2 < 0.5)[0]
        if ones.size ==0 or zeros.size==0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap
        v2[i], v2[j] = v2[j], v2[i]
    return v2

def hill_climb(v, steps, best_c5, start_time, budget):
    # simple hill climbing with small epsilon moves
    n = v.size
    dx = 2.0 / n
    cur_h = v.copy()
    cur_c5 = best_c5
    for _ in range(steps):
        if time.time() - start_time > budget*0.95:
            break
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # do mass transfer epsilon
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0:
            continue
        eps = np.random.uniform(0.01, 0.05)  # maybe adapt
        eps = min(eps, max_eps)
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            cur_h = new_h
            cur_c5 = new_c5
    return cur_h, cur_c5

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

def compute_c5(h: np.ndarray) -> float:
    """Exact C₅ value for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_bp_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Projection onto {0≤x≤1, sum x = target} via binary search on λ."""
    lo = np.min(v) - 1.0   # ensures all 1 after clipping
    hi = np.max(v)         # ensures all 0 after clipping
    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)

def random_binary(n: int) -> np.ndarray:
    """Binary vector with exactly n/2 ones (n must be even)."""
    half = n // 2
    v = np.zeros(n, dtype=float)
    v[:half] = 1.0
    np.random.shuffle(v)
    return v

def smooth_max_obj(h: np.ndarray, alpha: float = 200.0) -> float:
    """Smooth approximation of max_k ∫h(1‑h(·+k))."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    sm = logsumexp(alpha * corr) / alpha
    return float(sm * dx)

def constraint_sum_eq(h: np.ndarray) -> float:
    """Equality constraint: ∫h = 1."""
    return float(h.sum() - h.size / 2.0)
