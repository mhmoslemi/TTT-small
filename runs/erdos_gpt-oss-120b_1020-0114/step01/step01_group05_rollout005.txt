def subgradient_maxlag(h, dx, eps=1e-12):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_val = corr.max()
    # identify lags achieving max value within tolerance
    idxs = np.where(np.abs(corr - max_val) <= eps)[0]
    grads = np.zeros((len(idxs), n))
    for t, idx in enumerate(idxs):
        s = idx - (n - 1)  # shift
        grad = np.full(n, dx)  # start with 1*dx
        if s == 0:
            grad -= 2 * dx * h
        else:
            s_abs = abs(s)
            # subtract h shifted forward/backward
            grad[:n - s_abs] -= dx * h[s_abs:]
            grad[s_abs:] -= dx * h[:n - s_abs]
        grads[t] = grad
    # average
    grad_avg = grads.mean(axis=0)
    return grad_avg, max_val
```

import time
import numpy as np

# ------------------------------------------------------------
# Helper functions
# ------------------------------------------------------------
def _project(v, target):
    """Project vector v onto the box [0,1] with sum(v)=target."""
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _evaluate(h):
    """Compute C5 for a discrete h."""
    n = h.size
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode="full") * dx)


def _subgradient_topk(h, dx, top_k=3):
    """Average subgradient of the top_k worst overlaps."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full") * dx
    if top_k >= corr.size:
        idxs = np.arange(corr.size)
    else:
        idxs = np.argpartition(-corr, top_k - 1)[:top_k]
    grads = np.empty((len(idxs), n), dtype=float)
    for t, idx in enumerate(idxs):
        shift = idx - (n - 1)          # lag = shift
        grad = np.full(n, dx)
        if shift == 0:
            grad -= 2.0 * dx * h
        else:
            a = abs(shift)
            grad[: n - a] -= dx * h[a:]
            grad[a:] -= dx * h[: n - a]
        grads[t] = grad
    return grads.mean(axis=0)


def _optimize_start(h0, target_sum, start_time, budget_s):
    """Projected sub‑gradient descent from a single start."""
    h = h0.copy()
    n = h.size
    dx = 2.0 / n
    cur_c5 = _evaluate(h)
    best_h, best_c5 = h.copy(), cur_c5
    last_imp = 0
    base_lr = 0.8
    max_iters = 20000

    for it in range(max_iters):
        if time.time() - start_time > budget_s * 0.94:
            break

        # schedule for the number of lags used in the gradient
        if it < 4000:
            tk = 5
        elif it < 12000:
            tk = 3
        else:
            tk = 1

        grad = _subgradient_topk(h, dx, top_k=tk)
        lr = base_lr / np.sqrt(it + 1.0)
        step = lr
        improved = False

        for _ in range(7):          # back‑tracking line search
            h_new = _project(h - step * grad, target_sum)
            new_c5 = _evaluate(h_new)
            if new_c5 < cur_c5 - 1e-12:
                h, cur_c5 = h_new, new_c5
                improved = True
                if new_c5 < best_c5:
                    best_c5, best_h = new_c5, h_new.copy()
                    last_imp = it
                break
            step *= 0.5

        if not improved:
            # occasional random jitter to leave plateaus
            if it - last_imp > 3000 and (it % 250 == 0):
                jitter = np.random.normal(scale=0.015, size=n)
                h = _project(h + jitter, target_sum)
                cur_c5 = _evaluate(h)

    return best_h, best_c5


def _binary_refine(h_cont, target_sum, start_time, budget_s):
    """Round to a binary vector and perform random swap improvements."""
    n = h_cont.size
    k = int(round(target_sum))          # number of ones
    idx = np.argpartition(h_cont, -k)[-k:]
    h_bin = np.zeros(n, dtype=float)
    h_bin[idx] = 1.0
    best_h, best_c5 = h_bin, _evaluate(h_bin)

    max_swaps = 5000
    for _ in range(max_swaps):
        if time.time() - start_time > budget_s * 0.97:
            break
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h_new = best_h.copy()
        h_new[i], h_new[j] = 0.0, 1.0
        c5_new = _evaluate(h_new)
        if c5_new < best_c5 - 1e-13:
            best_h, best_c5 = h_new, c5_new
    return best_h, best_c5
