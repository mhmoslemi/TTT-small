def max_lag_info(h):
    n = h.size
    c = np.correlate(h, 1 - h, mode='full')
    max_idx = int(np.argmax(c))
    max_val = c[max_idx]
    lag = max_idx - (n - 1)
    # compute boolean mask for i where h[i]==1 and i+lag within bounds and h[i+lag]==0
    if lag >= 0:
        mask = (h[:n - lag] == 1) & (h[lag:] == 0)
        positions = np.nonzero(mask)[0]

        # adjust indices to original positions (i=l+idx)
        positions = np.nonzero(mask)[0] + l
    return lag, positions, max_val

import time, math, numpy as np

# ----- evaluation -----------------------------------------------------------
def _evaluate(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    # full correlation of h with (1-h)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

# ----- auxiliary info for targeted moves ------------------------------------
def _max_lag_info(h: np.ndarray):
    """
    Return (lag, positions, max_val) where
    - lag is the integer shift giving the maximal overlap,
    - positions are indices i with h[i]==1 and h[i+lag]==0 (i.e. contributing to the max),
    - max_val is the raw (unscaled) maximal count.
    """
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    max_val = corr[max_idx]
    lag = max_idx - (n - 1)          # signed lag
    if lag >= 0:
        mask = (h[: n - lag] == 1) & (h[lag:] == 0)
        pos = np.nonzero(mask)[0]              # i where i+lag is zero
    else:
        l = -lag
        mask = (h[l:] == 1) & (h[: n - l] == 0)
        pos = np.nonzero(mask)[0] + l
    return lag, pos, max_val

# ----- single annealing run -------------------------------------------------
def _anneal_one(start_h: np.ndarray, max_iters: int, T0: float, Tmin: float,
                prob_target: float = 0.7):
    """Simulated annealing from start_h, returns best (h, C5) seen."""
    cur_h = start_h.copy()
    cur_c5 = _evaluate(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    n = cur_h.size

    for it in range(max_iters):
        # temperature schedule (linear decay)
        T = max(Tmin, T0 * (1.0 - it / max_iters))

        # ----- choose move ----------------------------------------------------
        if np.random.rand() < prob_target:
            # target a pair that contributes to the current maximal lag
            lag, pos, _ = _max_lag_info(cur_h)
            if pos.size > 0:
                i = int(np.random.choice(pos))
                zeros = np.flatnonzero(cur_h == 0)
                if zeros.size == 0:
                    continue
                j = int(np.random.choice(zeros))
                if i == j:
                    continue
                new_h = cur_h.copy()
                new_h[i] = 0.0
                new_h[j] = 1.0
                new_c5 = _evaluate(new_h)
                delta = new_c5 - cur_c5
                if delta < 0 or np.random.rand() < math.exp(-delta / T):
                    cur_h, cur_c5 = new_h, new_c5
                    if new_c5 < best_c5:
                        best_c5, best_h = new_c5, new_h.copy()
                continue   # skip random branch this iteration

        # random swap of a 1 with a 0
        ones = np.flatnonzero(cur_h == 1)
        zeros = np.flatnonzero(cur_h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = int(np.random.choice(ones))
        j = int(np.random.choice(zeros))
        new_h = cur_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_c5 = _evaluate(new_h)
        delta = new_c5 - cur_c5
        if delta < 0 or np.random.rand() < math.exp(-delta / T):
            cur_h, cur_c5 = new_h, new_c5
            if new_c5 < best_c5:
                best_c5, best_h = new_c5, new_h.copy()

    return best_h, best_c5

# ----- greedy hill‑climb (only improving moves) ----------------------------
def _hill_climb(h: np.ndarray, cur_c5: float, deadline: float):
    """Refine h by accepting only moves that improve C5, until deadline."""
    n = h.size
    while time.time() < deadline:
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = int(np.random.choice(ones))
        j = int(np.random.choice(zeros))
        cand = h.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        cand_c5 = _evaluate(cand)
        if cand_c5 + 1e-12 < cur_c5:
            h, cur_c5 = cand, cand_c5
    return h, cur_c5

# ----- main entry point -----------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step function h on [0,2] with ∫h = 1,
    returning (h_values, c5_bound, n_points).
    The algorithm uses multiple simulated‑annealing restarts
    on binary vectors (0/1) of even length.
    """
    np.random.seed(seed)
    start_time = time.time()
    # fixed even grid size (adjustable if needed)
    n = 80                     # even → integer # of ones = n/2
    target_ones = n // 2

    # keep best overall solution
    best_h = None
    best_c5 = np.inf

    # parameters for each annealing run
    max_iters_per = 25000
    T0 = 0.5
    Tmin = 1e-5
    prob_target = 0.7

    # time budget for the outer loop (leave a small margin for final climb)
    outer_deadline = start_time + budget_s * 0.90

    while time.time() < outer_deadline:
        # start either from the current global best (20% chance) or fresh random
        if best_h is not None and np.random.rand() < 0.2:
            init_h = best_h.copy()
        else:
            init_h = np.zeros(n, dtype=float)
            init_h[np.random.choice(n, target_ones, replace=False)] = 1.0

        # run a short annealing phase
        local_h, local_c5 = _anneal_one(init_h, max_iters_per, T0, Tmin, prob_target)

        if local_c5 < best_c5:
            best_c5, best_h = local_c5, local_h.copy()
            # stop early if we already beat the target bound
            if best_c5 <= 0.3808:
                break

    # final deterministic hill‑climb to squeeze a bit more out of the best found
    if best_h is not None:
        final_deadline = start_time + budget_s * 0.97
        best_h, best_c5 = _hill_climb(best_h, best_c5, final_deadline)

    # fallback safety net
    if best_h is None:
        best_h = np.zeros(n, dtype=float)
        best_h[:target_ones] = 1.0
        best_c5 = _evaluate(best_h)

    return best_h, best_c5, n
