def local_search(N, target_sum, time_budget):
    h = np.zeros(N)
    ones = np.random.choice(N, target_sum, replace=False)
    h[ones] = 1.0
    best_h = h.copy()
    best_val = _c5_of(h)
    start = time.time()
    while time.time() - start < time_budget:
        # pick i where h[i]==1, j where h[j]==0
        ones_idx = np.where(h==1)[0]
        zeros_idx = np.where(h==0)[0]
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)

        h[i], h[j] = 0, 1

import numpy as np
import time
import math
import cvxpy as cp

def _project(v, target):
    """Project vector v onto [0,1] box and enforce sum(v)=target."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    while abs(diff) > 1e-12:
        if diff > 0:
            free = v < 1.0 - 1e-12
            nfree = free.sum()
            if nfree == 0:
                break
            add = diff / nfree
            v[free] = np.minimum(v[free] + add, 1.0)
        else:
            free = v > 1e-12
            nfree = free.sum()
            if nfree == 0:
                break
            sub = diff / nfree
            v[free] = np.maximum(v[free] + sub, 0.0)
        diff = target - v.sum()
    return v


def _c5_of(h):
    """Return the overlap constant C₅ for a step vector h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)


def _solve_milp(N, time_limit):
    """Attempt to solve the binary MILP for given N.
    Returns (h_vector, C_value) or (None, None) on failure."""
    target = N // 2
    # binary variables
    h = cp.Variable(N, integer=True)
    Y = cp.Variable((N, N), integer=True)   # product approximations
    C = cp.Variable()
    cons = [
        h >= 0,
        h <= 1,
        cp.sum(h) == target,
        Y >= 0,
        Y <= 1,
        C >= 0,
    ]

    # product constraints: Y[i,j] = h[i] * h[j]
    for i in range(N):
        for j in range(N):
            cons.append(Y[i, j] <= h[i])
            cons.append(Y[i, j] <= h[j])
            cons.append(Y[i, j] >= h[i] + h[j] - 1)

    # shift constraints: Σ h_i - Σ Y_{i,i+s} ≤ C
    for s in range(-N + 1, N):
        if s == 0:
            continue
        idx_i = []
        idx_j = []
        for i in range(N):
            j = i + s
            if 0 <= j < N:
                idx_i.append(i)
                idx_j.append(j)
        if not idx_i:
            continue
        sum_h = sum(h[i] for i in idx_i)
        sum_y = sum(Y[i, j] for i, j in zip(idx_i, idx_j))
        cons.append(sum_h - sum_y <= C)

    prob = cp.Problem(cp.Minimize(C), cons)

    # try a few solvers; GLPK_MI is usually fastest, fall back to ECOS_BB
    solved = False
    try:
        prob.solve(solver=cp.GLPK_MI, warm_start=True, max_iters=10**6,
                   **({"glpk": {"tm_lim": int(time_limit * 1000)}} if time_limit else {}))
        solved = prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE)
    except Exception:
        pass
    if not solved:
        try:
            prob.solve(solver=cp.CBC, warm_start=True,
                       **({"cbc": {"seconds": time_limit}} if time_limit else {}))
            solved = prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE)
        except Exception:
            pass
    if not solved:
        try:
            prob.solve(solver=cp.ECOS_BB, max_iters=5 * 10**6)
            solved = prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE)
        except Exception:
            solved = False

    if not solved or h.value is None:
        return None, None

    h_val = np.rint(np.array(h.value).flatten())
    # enforce exact feasibility (tiny numeric glitches)
    h_val = _project(h_val, target)
    C_val = float(C.value)
    return h_val, C_val


def _local_search(N, target, time_budget):
    """Simple randomized hill‑climbing for binary vectors."""
    # start from a random feasible point
    h = np.zeros(N)
    ones = np.random.choice(N, target, replace=False)
    h[ones] = 1.0
    best_h = h.copy()
    best_val = _c5_of(best_h)

    start = time.time()
    while time.time() - start < time_budget:
        # pick a 1 and a 0 and swap them
        idx_one = np.where(h == 1)[0]
        idx_zero = np.where(h == 0)[0]
        i = np.random.choice(idx_one)
        j = np.random.choice(idx_zero)
        h[i], h[j] = 0.0, 1.0
        new_val = _c5_of(h)
        if new_val < best_val:
            best_h, best_val = h.copy(), new_val
        else:
            # undo move (simple greedy, no temperature)
            h[i], h[j] = 1.0, 0.0
    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    """Find a step function with a small C₅ bound."""
    np.random.seed(seed)
    start = time.time()

    # Prefer a modest even size that gives enough resolution.
    # 48 points → target sum 24, dx = 2/48.
    N = 48
    target_sum = N // 2

    # Try the exact MILP first (allocating up to half the budget).
    milp_time = budget_s * 0.5
    h_vec, C_val = _solve_milp(N, milp_time)

    # If MILP failed, fall back to a quick combinatorial search.
    if h_vec is None:
        remaining = budget_s - (time.time() - start)
        h_vec, C_val = _local_search(N, target_sum, remaining * 0.9)

    # Ensure feasibility one last time (projection may be redundant).
    h_vec = _project(np.asarray(h_vec, dtype=float), target_sum)

    # Final evaluation (use the official helper if present).
    final_val = _c5_of(h_vec)
    if "evaluate_erdos_solution" in globals():
        try:
            final_val = evaluate_erdos_solution(h_vec.tolist())
        except Exception:
            pass

    # Return as plain Python list as required.
    return (h_vec.tolist(), float(final_val), int(N))
