def project_onto_bounds_simplex(v, s, l=0.0, u=1.0, tol=1e-9):
    # clip to bounds first
    # Use bisection to find lambda
    low = -1e6
    high = 1e6
    for _ in range(60):
        lam = (low + high) / 2
        y = np.clip(v - lam, l, u)
        cur = y.sum()
        if cur > s:
            low = lam
        else:
            high = lam
    y = np.clip(v - (low + high) / 2, l, u)
    # now adjust for potential small sum diff due to numerical
    diff = s - y.sum()

        if free.any():
            y[free] += diff / free.sum()
            y = np.clip(y, l, u)

def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a step function on [0,2]."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)


def _project_onto_capped_simplex(v: np.ndarray, target_sum: float,
                                 lower: float = 0.0,
                                 upper: float = 1.0,
                                 max_iter: int = 60,
                                 tol: float = 1e-12) -> np.ndarray:
    """
    Euclidean projection of *v* onto the set
        {x | lower <= x_i <= upper, sum(x) = target_sum }.
    Solved by bisection on the Lagrange multiplier λ in
        x_i = clip(v_i - λ, lower, upper).
    """
    low, high = -1e6, 1e6
    for _ in range(max_iter):
        lam = (low + high) / 2.0
        y = np.clip(v - lam, lower, upper)
        s = y.sum()
        if s > target_sum:
            low = lam
        else:
            high = lam
    y = np.clip(v - (low + high) / 2.0, lower, upper)
    # tiny correction for numerical drift
    diff = target_sum - y.sum()
    if abs(diff) > tol:
        free = (y > lower + 1e-12) & (y < upper - 1e-12)
        if free.any():
            y[free] += diff / free.sum()
            y = np.clip(y, lower, upper)
        else:
            # adjust a single coordinate if everything is at a bound
            idx = np.argmax(v)
            y[idx] = min(upper, max(lower, y[idx] + diff))
    return y


def _worst_shift(h: np.ndarray):
    """Return (shift index, full correlation array) where the overlap is maximal."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)
    return shift, corr


def _w_vector(h: np.ndarray, shift: int):
    """Vector w_i = h_{i+shift}+h_{i-shift} (zero outside the interval)."""
    n = h.shape[0]
    if shift == 0:
        return 2.0 * h
    w = np.zeros_like(h)
    if shift > 0:
        w[: n - shift] += h[shift:]
        w[shift:] += h[: n - shift]
    else:
        s = -shift
        w[s:] += h[: n - s]
        w[: n - s] += h[s:]
    return w


def _greedy_optimize(h0: np.ndarray, time_limit: float):
    """Deterministic / simulated‑annealing refinement used after the
    continuous gradient phase."""
    n = h0.shape[0]
    target_sum = n / 2.0
    best_h = h0.copy()
    best_val = _compute_c5(best_h)

    cur_h = h0.copy()
    cur_val = best_val
    step = 0.05
    no_improve = 0
    start = time.time()

    while time.time() - start < time_limit:
        shift, _ = _worst_shift(cur_h)
        w = _w_vector(cur_h, shift)

        if np.random.rand() < 0.7:
            up_mask = cur_h < 1.0 - 1e-12
            down_mask = cur_h > 1e-12
            if not up_mask.any() or not down_mask.any():
                break
            up_idx = np.where(up_mask)[0]
            down_idx = np.where(down_mask)[0]
            i_up = up_idx[np.argmax(w[up_idx])]
            i_down = down_idx[np.argmin(w[down_idx])]
            delta = min(step,
                        1.0 - cur_h[i_up],
                        cur_h[i_down])
        else:
            i_up = np.random.randint(0, n)
            i_down = np.random.randint(0, n - 1)
            if i_down >= i_up:
                i_down += 1
            max_inc = min(1.0 - cur_h[i_up], cur_h[i_down])
            max_dec = min(cur_h[i_up], 1.0 - cur_h[i_down])
            if max_inc > 0 and (max_dec == 0 or np.random.rand() < 0.5):
                delta = np.random.rand() * max_inc
            elif max_dec > 0:
                delta = -np.random.rand() * max_dec
                i_up, i_down = i_down, i_up
            else:
                continue
        if delta <= 0:
            continue

        new_h = cur_h.copy()
        new_h[i_up] += delta
        new_h[i_down] -= delta
        new_val = _compute_c5(new_h)

        if new_val < cur_val - 1e-12:
            cur_h, cur_val = new_h, new_val
            no_improve = 0
            step = min(step * 1.05, 0.1)
            if cur_val < best_val:
                best_val, best_h = cur_val, cur_h.copy()
        else:
            T = step * 0.5
            if T > 0 and np.random.rand() < math.exp(-(new_val - cur_val) / T):
                cur_h, cur_val = new_h, new_val
            no_improve += 1
            if no_improve > 400:
                step *= 0.5
                no_improve = 0
                if step < 1e-6:
                    break

    # Final correction of the integral
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        free = np.where((best_h > 1e-12) & (best_h < 1.0 - 1e-12))[0]
        if free.size:
            best_h[free] += diff / free.size
        else:
            best_h[0] = min(1.0, max(0.0, best_h[0] + diff))
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, _compute_c5(best_h)


def _slsqp_optimize(h0: np.ndarray, time_limit: float):
    """Optional short refinement with scipy.optimize (if scipy is available)."""
    try:
        from scipy.optimize import minimize
    except Exception:
        return h0, _compute_c5(h0)

    n = h0.shape[0]
    target_sum = n / 2.0
    t0 = _compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    eq_cons = {'type': 'eq', 'fun': lambda x: np.sum(x[:n]) - target_sum}
    cons = [eq_cons]

    shifts = list(range(-(n - 1), n))
    def make_ineq(s):
        def fun(x):
            h = x[:n]
            t = x[-1]
            if s >= 0:
                val = np.dot(h[: n - s], 1.0 - h[s:])
            else:
                s2 = -s
                val = np.dot(h[s2:], 1.0 - h[: n - s2])
            return t - val
        return fun
    cons.extend({'type': 'ineq', 'fun': make_ineq(s)} for s in shifts)

    max_iter = max(10, int(min(300, time_limit * 10)))
    opts = {'maxiter': max_iter, 'ftol': 1e-12, 'disp': False}
    res = minimize(lambda x: x[-1], x0, method='SLSQP',
                   bounds=bounds, constraints=cons, options=opts)

    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    diff = target_sum - h_opt.sum()
    if abs(diff) > 1e-10:
        free = np.where((h_opt > 1e-12) & (h_opt < 1.0 - 1e-12))[0]
        if free.size:
            h_opt[free] += diff / free.size
            h_opt = np.clip(h_opt, 0.0, 1.0)
    return h_opt, _compute_c5(h_opt)


def _gradient_optimize(h_start: np.ndarray, time_limit: float):
    """Projected gradient ascent on the smooth surrogate of min‑product."""
    h = h_start.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_val = _compute_c5(best_h)
    step = 0.2
    alpha = 0.5
    start = time.time()

    while time.time() - start < time_limit:
        # autocorrelation (product sums for all shifts)
        autocorr = np.correlate(h, h, mode="full")[n - 1:]  # length n
        P = autocorr
        P_min = P.min()
        # soft‑min weights
        exp_arg = -alpha * (P - P_min)
        # avoid overflow
        max_exp = np.max(exp_arg)
        w = np.exp(exp_arg - max_exp)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / n
        else:
            w = w / w_sum

        # gradient of the surrogate (w.r.t. h)
        grad = np.zeros_like(h)
        grad += 2.0 * h * w[0]                     # shift 0 contribution
        for s in range(1, n):
            ws = w[s]
            if ws == 0.0:
                continue
            grad[: n - s] += h[s:] * ws
            grad[s:]      += h[: n - s] * ws

        # normalised step
        step_norm = step / (np.linalg.norm(grad) + 1e-12)
        h_candidate = _project_onto_capped_simplex(h + step_norm * grad, target_sum)
        cand_val = _compute_c5(h_candidate)

        if cand_val < best_val:
            best_val = cand_val
            best_h = h_candidate.copy()

        # simple acceptance rule
        if cand_val <= _compute_c5(h) + 1e-12:
            h = h_candidate

        # schedule
        alpha = min(100.0, alpha * 1.02)
        step *= 0.9995

    return best_h, best_val


# ---------------------------------------------------------
# Main entry point required by the harness
# ---------------------------------------------------------

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimise a step‑function for the Erdős C₅ constant.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_all = time.time()

    # -----------------------------------------------------
    # Choose a resolution
    # -----------------------------------------------------
    n = 120                     # 2 seconds / 120 points → dx = 1/60
    target_sum = n / 2.0

    # -----------------------------------------------------
    # Build an initial vector
    # -----------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.shape[0]
        # linear interpolation onto the new grid
        x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n,   endpoint=False)
        h = np.interp(x_new, x_old, h0, left=0.0, right=0.0)
    else:
        # smooth cosine shape (peaked at the interval ends)
        idx = np.arange(n)
        h = (1.0 + np.cos(2.0 * np.pi * idx / n)) / 2.0

    # enforce feasibility
    h = np.clip(h, 0.0, 1.0)
    h = _project_onto_capped_simplex(h, target_sum)

    # -----------------------------------------------------
    # Phase 1 – continuous gradient ascent (≈½ of the budget)
    # -----------------------------------------------------
    time_elapsed = time.time() - start_all
    time_left = max(0.0, budget_s - time_elapsed)
    time_grad = time_left * 0.50
    if time_grad > 2.0:
        h_grad, val_grad = _gradient_optimize(h, time_grad)
    else:
        h_grad, val_grad = h, _compute_c5(h)

    # -----------------------------------------------------
    # Phase 2 – discrete greedy/annealing refinement
    # -----------------------------------------------------
    time_elapsed = time.time() - start_all
    time_left = max(0.0, budget_s - time_elapsed)
    time_greedy = time_left * 0.90
    if time_greedy > 1.0:
        h_greedy, val_greedy = _greedy_optimize(h_grad, time_greedy)
    else:
        h_greedy, val_greedy = h_grad, val_grad

    # pick the better of the two
    if val_greedy < val_grad:
        best_h, best_val = h_greedy, val_greedy
    else:
        best_h, best_val = h_grad, val_grad

    # -----------------------------------------------------
    # Phase 3 – optional short SLSQP polishing (if time permits)
    # -----------------------------------------------------
    time_elapsed = time.time() - start_all
    time_left = max(0.0, budget_s - time_elapsed)
    if time_left > 5.0:
        try:
            h_slsqp, val_slsqp = _slsqp_optimize(best_h, time_left)
            if val_slsqp < best_val:
                best_h, best_val = h_slsqp, val_slsqp
        except Exception:
            pass

    # -----------------------------------------------------
    # Final consistency check
    # -----------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        best_h = _project_onto_capped_simplex(best_h, target_sum)

    final_c5 = _compute_c5(best_h)

    return best_h, float(final_c5), int(n)
