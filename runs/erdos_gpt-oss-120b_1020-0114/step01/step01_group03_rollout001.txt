def upsample_h(old_h, new_n):

import time
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----- global parameters for constrained optimisation -----
_EQ_TARGET_SUM = None          # will hold n/2 for the current problem size
_SMOOTH_ALPHA = None           # smoothing factor for the log‑sum‑exp surrogate


def set_eq_target_sum(val):
    """Set the global target sum for the equality constraint."""
    global _EQ_TARGET_SUM
    _EQ_TARGET_SUM = val


def constraint_sum_eq(x):
    """Equality constraint: ∑x = target sum."""
    return np.sum(x) - _EQ_TARGET_SUM


def set_smooth_alpha(val):
    """Set the global smoothing factor used by the surrogate objective."""
    global _SMOOTH_ALPHA
    _SMOOTH_ALPHA = val


def smooth_objective_wrapper(x):
    """Wrapper that evaluates the smoothed objective using the global alpha."""
    n = x.shape[0]
    dx = 2.0 / n
    corr = correlate(x, 1 - x, mode="full")
    # log‑sum‑exp gives a smooth approximation of max(corr)
    return (logsumexp(_SMOOTH_ALPHA * corr) / _SMOOTH_ALPHA) * dx


def compute_c5(h):
    """Exact C₅ value for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)


def upsample_h(old_h, new_n):
    """Linear interpolation to a finer grid while preserving the integral."""
    old_n = old_h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)

    target_sum = new_n / 2.0
    cur_sum = new_h.sum()
    if cur_sum > 0:
        new_h = new_h * (target_sum / cur_sum)

    new_h = np.clip(new_h, 0.0, 1.0)

    # minor correction for any remaining sum error
    cur_sum = new_h.sum()
    diff = target_sum - cur_sum
    if abs(diff) > 1e-12:
        new_h += diff / new_n
        new_h = np.clip(new_h, 0.0, 1.0)
        cur_sum = new_h.sum()
        if cur_sum > 0:
            new_h = new_h * (target_sum / cur_sum)

    return new_h

    np.random.seed(seed)
