def evaluate(h):
    dx = 2.0 / len(h)
    return np.max(np.correlate(h, 1 - h, mode='full') * dx)

def max_lag_and_value(h):
    # compute correlation vector
    dx = 2.0 / len(h)
    conv = np.correlate(h, 1 - h, mode='full') * dx
    idx = np.argmax(conv)
    # lag = idx - (len(h) - 1)  # using numpy correlate indexing: idx - (n-1)
    max_val = conv[idx]
    lag = idx - (len(h) - 1)
    return lag, max_val

def gradient_max_lag(h, lag):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros_like(h)
    s = lag
    # first contribution: i where i and i+s are in [0,n)
    i1_start = max(0, -s)
    i1_end = min(n, n - s)  # exclusive
    i1 = np.arange(i1_start, i1_end)
    # gradient contribution +dx*(1 - h[i+s])
    grad[i1] += dx * (1 - h[i1 + s])
    # second contribution: i where i-s in [0,n)
    i2_start = max(0, s)
    i2_end = min(n, n + s)  # exclusive (for s negative, n+s < n)
    i2 = np.arange(i2_start, i2_end)
    grad[i2] -= dx * h[i2 - s]
    return grad

def project(h, target_sum):
    v = np.clip(h, 0, 1)
    n = len(v)
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0, 1)
        s = adj.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0, 1)

import time, numpy as np

def evaluate(h):
    n = h.shape[0]
    dx = 2.0/n
    conv = np.correlate(h, 1-h, mode='full') * dx
    return conv.max(), conv.argmax() - (n-1)

def project_to_feasible(v, target):
    # as per earlier
    v = np.clip(v, 0., 1.)
    lo, hi = -1., 1.
    for _ in range(50):
        mid = (lo+hi)*0.5
        adj = np.clip(v + mid, 0., 1.)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo+hi)/2., 0., 1.)

def gradient_at_lag(h, lag):
    n = h.size
    dx = 2.0/n
    s = lag
    grad = np.zeros_like(h)
    # first contribution
    i1_start = max(0, -s)
    i1_end = min(n, n - s)
    if i1_start < i1_end:
        idx = np.arange(i1_start, i1_end)
        grad[idx] += dx * (1 - h[idx + s])
    # second contribution
    i2_start = max(0, s)
    i2_end = min(n, n + s)
    if i2_start < i2_end:
        idx2 = np.arange(i2_start, i2_end)
        grad[idx2] -= dx * h[idx2 - s]
    return grad

def descent_one(start_h, max_iters, init_lr, target_sum, time_limit, start_time):

            # adapt lr

def random_binary_h(n):
    # n even
    half = n // 2
    arr = np.array([1]*half + [0]*half, dtype=float)
    np.random.shuffle(arr)
    return arr

def binary_hill_climb(start_h, max_iters, target_sum, time_limit, start_time):
    # start_h binary with sum=target_sum
    h = start_h.copy()
    cur_val = evaluate(h)[0]
    best_h = h.copy()
    best_val = cur_val
    n = h.size
    # indices of ones and zeros
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        # pick random indices to swap
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_val = evaluate(h)[0]
        if new_val < cur_val:
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0

            h = random_binary_h(n)
            cur_val = evaluate(h)[0]
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
    return best_h, best_val

import time
import numpy as np

# ---------- basic operations ----------
def _evaluate(h: np.ndarray) -> float:
    """Return C5 for step function h (max overlap)."""
    n = h.size
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode="full") * dx)


def _project(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0 ≤ x ≤ 1, sum x = target} via bisection."""
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _max_lag_and_val(h: np.ndarray):
    """Return (lag, max value) of the overlap for h."""
    n = h.size
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode="full") * dx
    idx = int(conv.argmax())
    lag = idx - (n - 1)
    return lag, float(conv[idx])


def _gradient_at_lag(h: np.ndarray, lag: int) -> np.ndarray:
    """Exact gradient of the overlap for the lag attaining the maximum."""
    n = h.size
    dx = 2.0 / n
    s = lag
    grad = np.zeros_like(h)

    # contribution from the term +h_i
    i1_start = max(0, -s)
    i1_end = min(n, n - s)
    if i1_start < i1_end:
        idx = np.arange(i1_start, i1_end, dtype=int)
        grad[idx] += dx * (1.0 - h[idx + s])

    # contribution from the product term -h_i h_{i+s}
    i2_start = max(0, s)
    i2_end = min(n, n + s)
    if i2_start < i2_end:
        idx2 = np.arange(i2_start, i2_end, dtype=int)
        grad[idx2] -= dx * h[idx2 - s]

    return grad
