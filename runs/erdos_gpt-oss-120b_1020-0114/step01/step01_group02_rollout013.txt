def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    # Choose N
    N = 200
    target_sum = N // 2
    # Initialize best_h from random binary plus maybe from initial_h_values scaled to N?
    best_h = np.concatenate([np.ones(target_sum), np.zeros(N - target_sum)], dtype=np.float64)
    np.random.shuffle(best_h)
    best_c5 = _compute_c5(best_h)

    # optional: incorporate initial_h_values if length near N: we can upsample it.
    if 'initial_h_values' in globals():
        # upsample or interpolate to size N, then repair sum.
        init = np.array(initial_h_values, dtype=np.float64)
        # simple linear interpolation to N points using np.interp
        x_old = np.linspace(0, 2, len(init), endpoint=False)
        x_new = np.linspace(0, 2, N, endpoint=False)
        h_interp = np.interp(x_new, x_old, init)
        h_interp = np.clip(h_interp, 0.0, 1.0)
        # adjust sum to target
        h_interp = _repair_sum(h_interp, target_sum)
        # evaluate
        c5_interp = _compute_c5(h_interp)
        if c5_interp < best_c5:
            best_h = h_interp.copy()
            best_c5 = c5_interp

    # Annealing parameters
    total_iters_est = 200000  # approximate; we will break earlier if time.
    # We'll use geometric cooling.
    init_T = 0.2
    final_T = 1e-5
    max_move_init = 0.5

    cur_h = best_h.copy()
    cur_c5 = best_c5

    it = 0
    while True:
        # time check
        if time.time() - start > budget_s - 2.0:
            break
        # schedule
        progress = it / total_iters_est
        if progress > 1.0:
            progress = 1.0
        T = init_T * (final_T / init_T) ** progress
        max_move = max_move_init * (1.0 - progress) + 1e-4

        # choose move
        move_type = np.random.rand()
        cand = cur_h.copy()
        if move_type < 0.7:
            # two-index transfer preserving sum exactly (like original)
            i, j = np.random.randint(N, size=2)
            if i == j:
                continue
            # limit delta
            max_delta = min(max_move, cur_h[i], 1.0 - cur_h[j])
            if max_delta <= 0:
                # try other direction
                max_delta = min(max_move, cur_h[j], 1.0 - cur_h[i])
                if max_delta <= 0:
                    continue
                i, j = j, i
            delta = np.random.uniform(0.0, max_delta)
            cand[i] -= delta
            cand[j] += delta
        elif move_type < 0.85:
            # binary swap: pick a near 1 and near 0 and exchange
            ones_idx = np.where(cand > 0.5)[0]
            zeros_idx = np.where(cand < 0.5)[0]
            if len(ones_idx) == 0 or len(zeros_idx) == 0:
                # fallback to transfer
                i, j = np.random.randint(N, size=2)
                if i==j: continue
                max_delta = min(max_move, cur_h[i], 1.0 - cur_h[j])
                if max_delta <=0:
                    continue
                delta = np.random.uniform(0.0, max_delta)
                cand[i] -= delta
                cand[j] += delta
            else:
                i = np.random.choice(ones_idx)
                j = np.random.choice(zeros_idx)
                # swap values
                val_i = cand[i]
                cand[i] = cand[j]
                cand[j] = val_i
        else:
            # random perturb + repair
            m = max(1, int(N * 0.02))  # modify ~2% of points
            idxs = np.random.choice(N, m, replace=False)
            cand[idxs] = np.random.rand(m)  # random values in [0,1)
            cand = _repair_sum(cand, target_sum)

        cand_c5 = _compute_c5(cand)
        delta_c5 = cand_c5 - cur_c5

        if delta_c5 < 0 or np.random.rand() < math.exp(-delta_c5 / max(T, 1e-12)):
            cur_h = cand
            cur_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = cand.copy()
        # occasional large restart
        if it % 20000 == 0 and it > 0:
            # if not improved for a while, re-perturb
            # spread out a fraction
            cur_h = best_h.copy()
            # random flip a proportion of bits if binary-like
            # add noise
            noise = (np.random.rand(N) - 0.5) * max_move_init
            cur_h = np.clip(cur_h + noise, 0.0, 1.0)
            cur_h = _repair_sum(cur_h, target_sum)
            cur_c5 = _compute_c5(cur_h)
        it += 1

    # final local refinement: attempt many random swaps (binary) until no improvement for some attempts
    improved = True
    attempts = 0
    max_attempts = 20000
    while attempts < max_attempts and time.time() - start < budget_s - 1.0:
        # try random swap
        ones_idx = np.where(best_h > 0.5)[0]
        zeros_idx = np.where(best_h < 0.5)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)
        cand = best_h.copy()
        cand[i], cand[j] = cand[j], cand[i]
        cand_c5 = _compute_c5(cand)
        if cand_c5 < best_c5 - 1e-8:
            best_h = cand
            best_c5 = cand_c5
            attempts = 0
        else:
            attempts += 1

    best_h = _repair_sum(best_h, target_sum)
    final_c5 = _compute_c5(best_h)

    return best_h.tolist(), float(final_c5), len(best_h)

def _compute_c5(h: np.ndarray) -> float:

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust `h` (in‑place) so that sum(h) == target while keeping 0≤h≤1."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h

        if abs(diff) < 1e-12:
            break

        zeros = np.where(best_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        cand = best_h.copy()
        cand[i], cand[j] = cand[j], cand[i]
        cand_c5 = _compute_c5(cand)
        if cand_c5 < best_c5 - 1e-9:
            best_h = cand
            best_c5 = cand_c5
            no_improve = 0
        else:
            no_improve += 1

    # --------------------------------------------------------------
    best_h = _repair_sum(best_h, target_sum)
    final_c5 = _compute_c5(best_h)

    return best_h.tolist(), float(final_c5), len(best_h)
