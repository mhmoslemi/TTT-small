def hill_climb(h_initial, n_iters, init_step, temp_start, temp_end):
    h = h_initial.copy()
    cur = compute_c5(h)
    best_h = h.copy()
    best_val = cur
    step = init_step
    T = temp_start
    for it in range(1, n_iters+1):
        # linearly decreasing T
        T = temp_start + (temp_end - temp_start) * it / n_iters
        # Choose move
        i, j = np.random.randint(0, len(h), 2)

def upsample_h(old_h, new_n):
    old_n = old_h.shape[0]
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    target_sum = new_n / 2
    cur_sum = new_h.sum()
    if cur_sum > 0:
        new_h = new_h * (target_sum / cur_sum)
    new_h = np.clip(new_h, 0, 1)
    # final correction
    cur_sum = new_h.sum()
    diff = target_sum - cur_sum
    if abs(diff) > 1e-12:
        new_h += diff / new_n
        new_h = np.clip(new_h, 0, 1)
        cur_sum = new_h.sum()
        if cur_sum > 0:
            new_h = new_h * (target_sum / cur_sum)
    return new_h

def hill_climb(h, time_limit, start_temp=0.001, end_temp=1e-7, start_step=0.05, end_step=0.001):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    start = time.time()
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        temp = start_temp * (1 - frac) + end_temp * frac
        step = start_step * (1 - frac) + end_step * frac

        # choose move type: random or targeted
        if np.random.rand() < 0.7:
            # targeted move: find worst shift
            corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            idx_max = np.argmax(corr)
            shift = idx_max - (n - 1)
            # compute pair index with minimal product
            if shift >= 0:
                i_range = np.arange(n - shift)
                prod = cur_h[i_range] * cur_h[i_range + shift]
                i_min = i_range[np.argmin(prod)]
                # Choose recipient as the lower of the pair
                if cur_h[i_min] <= cur_h[i_min + shift]:
                    rec = i_min
                else:
                    rec = i_min + shift
            else:
                s = -shift
                i_range = np.arange(n - s)
                prod = cur_h[i_range + s] * cur_h[i_range]
                i_min = i_range[np.argmin(prod)]
                # pair (i+s, i)
                if cur_h[i_min + s] <= cur_h[i_min]:
                    rec = i_min + s
                else:
                    rec = i_min

            # donor: randomly choose j not equal rec
            j = np.random.randint(0, n - 1)
            if j >= rec:
                j += 1
            # epsilon
            delta_max = min(cur_h[j], 1.0 - cur_h[rec])
            if delta_max <= 0:
                continue
            eps = np.random.rand() * min(delta_max, step)
            new_h = cur_h.copy()
            new_h[j] -= eps
            new_h[rec] += eps
        else:
            i = np.random.randint(0, n)
            j = np.random.randint(0, n)
            if i == j:
                continue
            delta_max = min(cur_h[i], 1.0 - cur_h[j])
            if delta_max <= 0:
                continue
            eps = np.random.rand() * min(delta_max, step)
            new_h = cur_h.copy()
            new_h[i] -= eps
            new_h[j] += eps

        new_c5 = compute_c5(new_h)
        delta_c = new_c5 - cur_c5
        if delta_c < 0 or np.random.rand() < np.exp(-delta_c / (temp + 1e-14)):
            cur_h = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
    return best_h, best_c5

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C₅ value for a discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def upsample_h(old_h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a finer grid while preserving the integral."""
    old_n = old_h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)

    target = new_n / 2.0
    s = new_h.sum()
    if s > 0:
        new_h = new_h * (target / s)
    new_h = np.clip(new_h, 0.0, 1.0)

    # final small correction
    s = new_h.sum()
    diff = target - s
    if abs(diff) > 1e-12:
        new_h += diff / new_n
        new_h = np.clip(new_h, 0.0, 1.0)
        s = new_h.sum()
        if s > 0:
            new_h = new_h * (target / s)
    return new_h


def triangle_h(n: int) -> np.ndarray:
    """Triangular shape (peak 1 at x=1) discretised on n points, normalized."""
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.where(x <= 1.0, x, 2.0 - x)  # height 1 at center, 0 at boundaries
    target = n / 2.0
    s = h.sum()
    if s > 0:
        h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)

    # fix possible tiny sum error
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if s > 0:
            h = h * (target / s)
    return h


# ----------------------------------------------------------------------
# Simulated–annealing neighbourhood search
# ----------------------------------------------------------------------
def hill_climb(
    h0: np.ndarray,
    time_limit: float,
    start_temp: float = 3e-3,
    end_temp: float = 1e-7,
    start_step: float = 0.07,
    end_step: float = 0.001,
    target_prob: float = 0.65,
) -> tuple[np.ndarray, float]:
    """Mass‑transfer SA that respects the sum constraint.
    Returns the best vector seen and its C₅ value.
    """
    n = h0.shape[0]
    dx = 2.0 / n

    cur = h0.copy()
    # initial correlation & cost
    corr = np.correlate(cur, 1.0 - cur, mode="full")
    cur_val = np.max(corr) * dx
    best = cur.copy()
    best_val = cur_val

    start = time.time()
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= time_limit:
            break
        # linear schedules
        frac = elapsed / time_limit
        temp = start_temp * (1.0 - frac) + end_temp * frac
        step = start_step * (1.0 - frac) + end_step * frac

        # ------------------------------------------------------------------
        # choose move
        # ------------------------------------------------------------------
        if np.random.rand() < target_prob:
            # targeted: improve the worst shift
            idx_max = int(np.argmax(corr))
            shift = idx_max - (n - 1)          # signed shift in [-n+1, n-1]

            if shift >= 0:
                i_range = np.arange(n - shift)
                prod = cur[i_range] * cur[i_range + shift]
                i_min = i_range[int(np.argmin(prod))]
                # choose the lower of the two paired entries as recipient
                rec = i_min if cur[i_min] <= cur[i_min + shift] else i_min + shift
            else:
                s = -shift
                i_range = np.arange(n - s)
                prod = cur[i_range + s] * cur[i_range]
                i_min = i_range[int(np.argmin(prod))]
                rec = i_min + s if cur[i_min + s] <= cur[i_min] else i_min

            # donor index, uniformly random and different from recipient
            donor = np.random.randint(0, n - 1)
            if donor >= rec:
                donor += 1

            delta_max = min(cur[donor], 1.0 - cur[rec])
            if delta_max <= 0.0:
                continue
            eps = np.random.rand() * min(delta_max, step)

            cand = cur.copy()
            cand[donor] -= eps
            cand[rec] += eps
        else:
            # simple random mass transfer
            i = np.random.randint(0, n)
            j = np.random.randint(0, n - 1)
            if j >= i:
                j += 1
            delta_max = min(cur[i], 1.0 - cur[j])
            if delta_max <= 0.0:
                continue
            eps = np.random.rand() * min(delta_max, step)
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        # ------------------------------------------------------------------
        # evaluate candidate
        # ------------------------------------------------------------------
        cand_corr = np.correlate(cand, 1.0 - cand, mode="full")
        cand_val = np.max(cand_corr) * dx
        diff = cand_val - cur_val

        # Metropolis acceptance
        if diff < 0.0 or np.random.rand() < np.exp(-diff / (temp + 1e-14)):
            cur = cand
            cur_val = cand_val
            corr = cand_corr
            if cur_val < best_val:
                best_val = cur_val
                best = cur.copy()

    return best, best_val
