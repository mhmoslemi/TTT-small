def project_to_bounds_and_sum(x, target):
    # find lambda s.t. sum(clip(x + lam, 0, 1)) = target
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        proj = np.clip(x + lam, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(x + (lo+hi)/2.0, 0.0, 1.0)

def multi_start_optimize(n, total_time):
    target = n/2.0

    while time.time() - start < total_time:

import time
import numpy as np

# ---------- basic utilities ----------
def _compute_c5(h: np.ndarray) -> float:
    """max_k ∫ h(x)(1‑h(x+k))dx  (discretised)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)


def _worst_shift(h: np.ndarray):
    """return (shift, correlation array) where overlap is maximal."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)
    return shift, corr


def _gradient_worst_shift(h: np.ndarray, s: int) -> np.ndarray:
    """sub‑gradient of the worst shift (the one attaining the max)."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if s >= 0:
        # part where h_i appears as first factor
        if s < n:
            g[: n - s] += 1.0 - h[s:]
        # part where h_i appears as second factor (negative)
        if s > 0:
            g[s:] -= h[: n - s]
    else:  # s < 0
        t = -s
        if t < n:
            g[t:] += 1.0 - h[: n - t]          # (1‑h_{k‑t})
            g[: n - t] -= h[t:]                # -h_{k+t}
    return g


def _project_bounds_sum(x: np.ndarray, target: float) -> np.ndarray:
    """
    Project x onto the set {0 ≤ z_i ≤ 1, Σz = target}
    using a simple bisection on the common offset lambda.
    """
    lo, hi = -2.0, 2.0          # wide enough for any feasible lambda
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(x + lam, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)


# ---------- simple gradient descent on the worst shift ----------
def _gd_worst_shift(h0: np.ndarray, time_limit: float):
    """Projected gradient descent that always follows the current worst shift."""
    n = h0.shape[0]
    target = n / 2.0
    best_h = h0.copy()
    best_val = _compute_c5(best_h)

    h = h0.copy()
    cur_val = best_val
    lr = 0.12                     # start step size
    start = time.time()
    stagn = 0

    while time.time() - start < time_limit:
        s, _ = _worst_shift(h)
        g = _gradient_worst_shift(h, s)

        improved = False
        # simple back‑tracking line search
        for fac in (1.0, 0.5, 0.25, 0.125):
            step = lr * fac
            h_new = _project_bounds_sum(h - step * g, target)
            val_new = _compute_c5(h_new)
            if val_new < cur_val - 1e-12:
                h, cur_val = h_new, val_new
                if val_new < best_val:
                    best_val, best_h = val_new, h_new.copy()
                lr = min(lr * 1.05, 0.5)
                improved = True
                stagn = 0
                break
        if not improved:
            lr *= 0.5
            stagn += 1
            if lr < 1e-6 or stagn > 40:
                break
        if best_val < 0.3808:          # early exit if we already beat the target
            break
    return best_h, best_val


# ---------- stochastic hill‑climbing (swap‑delta) ----------
def _swap_hill_climb(h0: np.ndarray, time_limit: float):
    """pairwise mass‑transfer hill climb with occasional annealed acceptances."""
    n = h0.shape[0]
    target = n / 2.0
    best_h = h0.copy()
    best_val = _compute_c5(best_h)

    cur_h = h0.copy()
    cur_val = best_val
    step = 0.07
    no_imp = 0
    start = time.time()

    while time.time() - start < time_limit:
        # pick random up/down indices
        up = np.where(cur_h < 1.0 - 1e-12)[0]
        down = np.where(cur_h > 1e-12)[0]
        if up.size == 0 or down.size == 0:
            break
        i_up = np.random.choice(up)
        i_down = np.random.choice(down)
        max_inc = min(step, 1.0 - cur_h[i_up])
        max_dec = min(step, cur_h[i_down])
        delta = np.random.rand() * min(max_inc, max_dec)
        if delta <= 0:
            continue
        new_h = cur_h.copy()
        new_h[i_up] += delta
        new_h[i_down] -= delta
        new_val = _compute_c5(new_h)

        if new_val < cur_val - 1e-12:
            cur_h, cur_val = new_h, new_val
            no_imp = 0
            step = min(step * 1.03, 0.12)
            if new_val < best_val:
                best_val, best_h = new_val, new_h.copy()
        else:
            # simple annealing
            T = step * 0.35
            if T > 0 and np.random.rand() < np.exp(-(new_val - cur_val) / T):
                cur_h, cur_val = new_h, new_val
            no_imp += 1
            if no_imp > 300:
                step *= 0.6
                no_imp = 0
                if step < 1e-6:
                    break
    # final projection onto hyperplane, just in case
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        best_h = _project_bounds_sum(best_h, target)
    return best_h, best_val

def _multi_start_optimize(n: int, total_time: float, init_vec=None):
    """run several independent attempts (GD + occasional hill‑climb)."""
    target = n / 2.0
    best_h = None
    best_val = np.inf
    start = time.time()

    # use provided init_vec once, if any
    if init_vec is not None:
        h0 = init_vec.copy()
        # make sure it is feasible
        h0 = _project_bounds_sum(h0, target)
        h_opt, v_opt = _gd_worst_shift(h0, total_time * 0.4)
        if v_opt < best_val:
            best_val, best_h = v_opt, h_opt.copy()
        # short hill‑climb afterwards
        remaining = total_time - (time.time() - start)
        if remaining > 0.5:
            h_opt2, v_opt2 = _swap_hill_climb(best_h, remaining * 0.5)
            if v_opt2 < best_val:
                best_val, best_h = v_opt2, h_opt2.copy()

    # remaining budget for fresh random starts
    while time.time() - start < total_time * 0.9:
        remain = total_time - (time.time() - start)
        if remain <= 0.4:
            break
        # random feasible start
        x = np.random.rand(n)
        x = _project_bounds_sum(x, target)

        # allocate a slice of the remaining budget
        slice_time = min(0.4, remain * 0.4)
        h_opt, v_opt = _gd_worst_shift(x, slice_time)
        if v_opt < best_val:
            best_val, best_h = v_opt, h_opt.copy()
        # a quick hill‑climb refinement
        r2 = min(0.2, remain - slice_time)
        if r2 > 0.1:
            h_opt2, v_opt2 = _swap_hill_climb(best_h, r2)
            if v_opt2 < best_val:
                best_val, best_h = v_opt2, h_opt2.copy()
        # early exit if we have already beaten the target bound
        if best_val < 0.3808:
            break
    # fall‑back: if nothing found, fall back to uniform half‑ones
    if best_h is None:
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        np.random.shuffle(best_h)
    best_h = _project_bounds_sum(best_h, target)
    return best_h, _compute_c5(best_h)
