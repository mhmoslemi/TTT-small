def shift(arr, offset):
    if offset >= 0:
        # arr shifted left by offset (i+offset)
        return np.concatenate([arr[offset:], np.zeros(offset)])
    else:
        offset = -offset
        # arr shifted right by offset (i - offset)
        return np.concatenate([np.zeros(offset), arr[:len(arr)-offset]])
```

def project_cap_simplex(v, s):
    # binary search lambda such that sum clip(v + lam, 0, 1) = s
    lo = -np.max(v) - 1.0 # ensure v+lam <=0
    hi = 1.0 - np.min(v) + 1.0 # ensure v+lam >=1

        if w_sum < s:
            lo = lam
        else:
            hi = lam

def project_cap_simplex(v, s):
    # find lambda such that sum(clip(v+lam, 0,1)) = s
    lo = -np.max(v) - 1.0  # ensure v+lam <=0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        w = np.clip(v + lam, 0.0, 1.0)
        wsum = w.sum()
        if wsum < s:
            lo = lam
        else:
            hi = lam
    w = np.clip(v + (lo+hi)/2.0, 0.0, 1.0)
    return w

    random.seed(seed)

        init_arr = np.array(initial_h_values, dtype=np.float64)
        N = len(init_arr)

    else:

    target_sum = N / 2.0

    # ensure feasibility
    h0 = project_cap_simplex(h0, target_sum)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # Gradient descent
    momentum = np.zeros_like(h0)
    beta = 0.9
    step0 = 0.5
    max_iter = 10_000_000  # large upper bound
    it = 0
    last_improve = 0
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s - 5.0:
            break
        # subgradient iteration
        N = len(h0)
        corr = np.correlate(h0, 1.0 - h0, mode='full')
        max_idx = np.argmax(corr)
        lag = max_idx - (N - 1)
        # gradient for lag
        h_plus = shift(h0, lag)
        h_minus = shift(h0, -lag)
        grad = (1.0 - h_plus) - h_minus
        # step size
        alpha = step0 / np.sqrt(it + 1.0)
        momentum = beta * momentum - alpha * grad
        v = h0 + momentum

        # occasional random perturbation
        if np.random.rand() < 0.01:
            v = v + np.random.normal(scale=0.02, size=N)

        # projection onto constraints
        h_new = project_cap_simplex(v, target_sum)

        # evaluate
        val = compute_c5(h_new)

        if val < best_val - 1e-9:
            best_val = val
            best_h = h_new.copy()
            last_improve = it
        # accept new state (deterministic) as current
        h0 = h_new
        it += 1

        # if no improvement for many iterations, add random restart
        if it - last_improve > 20000:
            # restart from best with noise
            h0 = best_h + np.random.normal(scale=0.05, size=N)

import numpy as np
import time, random

def _shift(arr: np.ndarray, offset: int) -> np.ndarray:
    """Zero‑padded shift: out[i] = arr[i+offset] if inside, else 0."""
    n = arr.shape[0]
    out = np.empty_like(arr)
    if offset >= 0:
        if offset < n:
            out[: n - offset] = arr[offset:]
            out[n - offset :] = 0.0
        else:  # shift beyond length -> all zeros
            out.fill(0.0)
    else:
        off = -offset
        if off < n:
            out[off:] = arr[: n - off]
            out[:off] = 0.0
        else:
            out.fill(0.0)
    return out


def _project_cap_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project v onto the intersection of the box [0,1]^n and the hyperplane sum = target.
    The projection is of the form w_i = clip(v_i + λ, 0, 1) for a suitable λ.
    """
    lo = -np.max(v) - 1.0   # guarantees v+lam <= 0
    hi = 1.0 - np.min(v) + 1.0  # guarantees v+lam >= 1
    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v + lam, 0.0, 1.0)
        s = w.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v + lam, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a step function h (as a list), its C5 bound, and the number
    of discretisation points.
    """
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # initialise h and the problem size
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n = len(h)
    else:
        n = 150                     # reasonable default
        h = np.random.rand(n)

    target_sum = n / 2.0
    h = _project_cap_simplex(h, target_sum)

    best_h = h.copy()
    best_val = _c5_value(best_h)

    # ------------------------------------------------------------------
    # sub‑gradient descent on the worst lag
    momentum = np.zeros_like(h)
    beta = 0.9                     # momentum coefficient
    step0 = 0.5                    # initial step size factor
    it = 0
    last_improve = 0

    while True:
        if time.time() - start > budget_s - 5.0:
            break

        # compute worst lag
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        lag = max_idx - (n - 1)       # signed shift, zero‑lag at index n‑1

        # gradient of the worst term (sub‑gradient)
        h_plus = _shift(h, lag)
        h_minus = _shift(h, -lag)
        grad = (1.0 - h_plus) - h_minus

        # diminishing step size, with momentum
        alpha = step0 / np.sqrt(it + 1.0)
        momentum = beta * momentum - alpha * grad
        v = h + momentum

        # occasional stochastic jitter to escape plateaux
        if np.random.rand() < 0.01:
            v += np.random.normal(scale=0.02, size=n)

        # project back onto feasible set
        h_new = _project_cap_simplex(v, target_sum)

        val = _c5_value(h_new)

        if val < best_val - 1e-9:
            best_val = val
            best_h = h_new.copy()
            last_improve = it

        h = h_new
        it += 1

        # if stagnation for long, re‑ignite from the current best with noise
        if it - last_improve > 20000:
            h = best_h + np.random.normal(scale=0.05, size=n)
            h = _project_cap_simplex(h, target_sum)
            momentum.fill(0.0)
            last_improve = it

    # ------------------------------------------------------------------
    # final tidy‑up
    best_h = _project_cap_simplex(best_h, target_sum)
    final_val = _c5_value(best_h)

    return best_h.tolist(), float(final_val), n
