import time
import math
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    """Maximum overlap integral for a step function h (array of values)."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h); full mode gives all integer shifts
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def corr_shift(h, s):
    """Sum_i h[i] * (1 - h[i+s]) for integer shift s (may be negative)."""
    n = h.shape[0]
    if s >= 0:
        if s >= n:
            return 0.0
        return np.dot(h[: n - s], 1.0 - h[s:])
    else:
        s = -s
        if s >= n:
            return 0.0
        return np.dot(h[s:], 1.0 - h[: n - s])

def random_feasible(n, rng):
    """Generate a random vector in [0,1]^n with exactly n/2 total mass."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    # simple projection to keep bounds
    for _ in range(10):
        over = v > 1.0
        under = v < 0.0
        if not over.any() and not under.any():
            break
        excess = (v[over] - 1.0).sum()
        deficit = (-v[under]).sum()
        if excess > 0.0:
            interior = (~over) & (~under)
            if interior.any():
                v[interior] -= excess / interior.sum()
        if deficit > 0.0:
            interior = (~over) & (~under)
            if interior.any():
                v[interior] += deficit / interior.sum()
        v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - n / 2.0) > 1e-12:
        diff = n / 2.0 - cur
        mask = (v > 0.0) & (v < 1.0)
        if mask.any():
            v[mask] += diff / mask.sum()
            v = np.clip(v, 0.0, 1.0)
        else:
            v *= (n / 2.0) / cur
    return v

def optimize_h(h_start, max_iter=1500):
    """SLSQP minimisation of the maximum correlation."""
    n = h_start.shape[0]
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]  # extra variable t
    eq_cons = {"type": "eq", "fun": lambda x: np.sum(x[:n]) - n / 2.0}
    ineq = []
    for s in range(-(n - 1), n):
        def fun(x, s=s):
            h = x[:n]
            t = x[n]
            return t - corr_shift(h, s)
        ineq.append({"type": "ineq", "fun": fun})
    constraints = [eq_cons] + ineq
    t0 = compute_c5(h_start)
    x0 = np.concatenate([h_start, [t0]])
    opts = {"maxiter": max_iter, "ftol": 1e-10, "disp": False}
    try:
        res = opt.minimize(
            lambda x: x[n],
            x0,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options=opts,
        )
    except Exception:
        res = None
    if res is None or not res.success:
        h_opt = np.clip(h_start, 0.0, 1.0)
    else:
        h_opt = np.clip(res.x[:n], 0.0, 1.0)

    # enforce the integral exactly
    target = n / 2.0
    cur = h_opt.sum()
    if abs(cur - target) > 1e-8:
        diff = target - cur
        mask = (h_opt > 0.0) & (h_opt < 1.0)
        if mask.any():
            h_opt[mask] += diff / mask.sum()
            h_opt = np.clip(h_opt, 0.0, 1.0)
        else:
            h_opt *= target / cur
    return h_opt, compute_c5(h_opt)

def hill_climb_sa(h, time_budget, start_time, rng):
    """Simulated‑annealing hill climbing on the max correlation."""
    n = h.shape[0]
    best = h.copy()
    best_val = compute_c5(best)
    cur = best.copy()
    cur_val = best_val
    # initial temperature proportional to current value
    T0 = max(0.001, best_val * 0.02)

    while time.time() - start_time < time_budget:
        remaining = time_budget - (time.time() - start_time)
        # linear cooling
        T = T0 * max(remaining, 0.0) / max(time_budget, 1e-12)
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move mass from i (must have >0) to j (must have <1)
        if cur[i] > 0.0 and cur[j] < 1.0:
            max_delta = min(cur[i], 1.0 - cur[j])
        elif cur[j] > 0.0 and cur[i] < 1.0:
            max_delta = min(cur[j], 1.0 - cur[i])
            i, j = j, i
        else:
            continue
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random()
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        val = compute_c5(cand)
        diff = val - cur_val
        if diff < 0.0 or rng.random() < math.exp(-diff / max(T, 1e-12)):
            cur, cur_val = cand, val
            if cur_val < best_val:
                best, best_val = cur.copy(), cur_val
    return best, best_val
