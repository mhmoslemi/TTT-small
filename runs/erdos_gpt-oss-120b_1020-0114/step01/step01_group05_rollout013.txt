def subgrad_descent(h, target_sum, max_iters, step0):
    best_h = h.copy()
    best_c5 = _evaluate(best_h)
    for it in range(max_iters):
        # time check inside above.
        corr = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
        gmax = corr.max()
        # find all max indices
        max_idx = np.where(np.abs(corr - gmax) < 1e-12)[0]
        grad = np.zeros_like(h)
        for idx in max_idx:
            s = idx - (n - 1)
            if s >= 0:
                # i=s..n-1, j=0..n-s-1
                # add (1 - h[j]) to grad[i]
                grad[s:] += 1 - h[:n - s]
                # subtract h[i] from grad[j]
                grad[:n - s] -= h[s:]
            else:
                s_abs = -s
                # i=0..n-s_abs-1, j=s_abs..n-1
                grad[:n - s_abs] += 1 - h[s_abs:]
                grad[s_abs:] -= h[:n - s_abs]

        if max_idx.size > 1:
            grad /= max_idx.size

        # step size schedule
        step = step0 / np.sqrt(it + 1)
        # Update with projection
        h_new = _project(h - step * grad, target_sum)
        c5_new = _evaluate(h_new)

        # Accept if c5_new <= best_c5, else maybe accept if less than current
        # We'll keep h as h_new anyway to continue descent; but if c5_new > best_c5 significantly, maybe we overshoot.
        # Let's accept always.

            best_c5 = c5_new

def _project(v, target):
    # Clip to [0,1]
    v = np.clip(v, 0.0, 1.0)
    # sum of v after clipping may already be target
    s = v.sum()
    if abs(s - target) <= 1e-12:
        return v
    # else find offset such that sum(clip(v + offset, 0,1)) = target
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

import time
import numpy as np

def _project(v, target):
    """Project onto {0≤x≤1, sum x = target}."""
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) <= 1e-12:
        return v
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _evaluate(h):
    n = h.size
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode="full") * dx)


def _grad_max_shift(h, dx):
    """Gradient of the current maximal correlation (subgradient)."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full") * dx
    gmax = corr.max()
    max_idx = np.where(np.abs(corr - gmax) < 1e-12)[0]
    grad = np.zeros_like(h)
    for idx in max_idx:
        s = idx - (n - 1)
        if s >= 0:
            grad[s:] += 1 - h[: n - s]
            grad[: n - s] -= h[s:]
        else:
            sa = -s
            grad[: n - sa] += 1 - h[sa:]
            grad[sa:] -= h[: n - sa]
    if max_idx.size > 1:
        grad /= max_idx.size
    return grad * dx, gmax


def _descent_one(h, target_sum, max_iters, step0, budget, start):
    n = h.size
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = _evaluate(h)
    it = 0
    while it < max_iters and (time.time() - start) < budget * 0.95:
        grad, _ = _grad_max_shift(h, dx)
        step = step0 / np.sqrt(it + 1)
        h_new = _project(h - step * grad, target_sum)
        c5_new = _evaluate(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        it += 1
        # occasional jitter to escape stagnation
        if it % 3000 == 0 and (time.time() - start) < budget * 0.95:
            jitter = np.random.normal(scale=0.02, size=n)
            h = _project(h + jitter, target_sum)
    return best_h, best_c5


def _swap_refine(h, target_sum, max_swaps, budget, start):
    """Swap two entries (preserving sum) if it improves the bound."""
    n = h.size
    best_h = h.copy()
    best_c5 = _evaluate(h)
    swaps = 0
    while swaps < max_swaps and (time.time() - start) < budget * 0.95:
        i, j = np.random.randint(n, size=2)
        if i == j or best_h[i] == best_h[j]:
            continue
        h_sw = best_h.copy()
        h_sw[i], h_sw[j] = best_h[j], best_h[i]
        c5_sw = _evaluate(h_sw)
        if c5_sw < best_c5 - 1e-9:
            best_h, best_c5 = h_sw, c5_sw
        swaps += 1
    return best_h, best_c5
