def project_to_feasible(h, target):
    # Clip and adjust sum to target using water-filling
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if abs(cur - target) < 1e-12:
        return h
    # Use iterative simple algorithm:
    for _ in range(100):
        diff = target - cur
        free = (h > 0.0) & (h < 1.0)
        free_idx = np.where(free)[0]
        if free_idx.size == 0:

def gradient_worst_shifts(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, h, mode='full') * dx
    min_corr = np.min(corr)
    # find shifts where corr ~ min_corr (within small tolerance)
    eps = 1e-12
    idxs = np.where(corr <= min_corr + eps)[0]
    shifts = np.abs(idxs - (n - 1))
    uniq_shifts = np.unique(shifts)
    grad = np.zeros_like(h)

            grad += 2 * h
        else:
            s_int = int(s)
            left = np.arange(0, n - s_int)
            right = left + s_int
            grad[left] += h[right]
            grad[right] += h[left]
    return grad

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def project_to_feasible(h, target_sum):
    """Clip to [0,1] and enforce sum(h) == target_sum (approximately)."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if abs(cur - target_sum) < 1e-12:
        return h
    # simple water‑filling style iterative correction
    for _ in range(100):
        diff = target_sum - cur
        free = (h > 0.0) & (h < 1.0)
        free_idx = np.where(free)[0]
        if free_idx.size == 0:
            # no free variable – push a random index toward the needed direction
            i = np.random.randint(len(h))
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
        else:
            delta = diff / free_idx.size
            h[free_idx] += delta
            np.clip(h, 0.0, 1.0, out=h)
        cur = h.sum()
        if abs(cur - target_sum) < 1e-12:
            break
    return h


def max_overlap(h):
    """Exact C5 value for a discretised step function."""
    n = len(h)
    dx = 2.0 / n
    # min correlation of h with itself
    corr = np.correlate(h, h, mode='full') * dx
    min_corr = np.min(corr)               # ∫ h(x)h(x+k)dx minimum over k
    return 1.0 - min_corr                  # max_k ∫ h(1‑h_shift)dx


def gradient_worst_shifts(h):
    """Gradient that increases the minimum self‑correlation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, h, mode='full') * dx
    min_corr = np.min(corr)
    eps = 1e-12
    idx = np.where(corr <= min_corr + eps)[0]
    shifts = np.abs(idx - (n - 1))
    uniq_shifts = np.unique(shifts.astype(int))
    grad = np.zeros_like(h)
    for s in uniq_shifts:
        if s == 0:
            grad += 2.0 * h                     # derivative of ∑ h_i^2
        else:
            left = np.arange(0, n - s)
            right = left + s
            grad[left] += h[right]
            grad[right] += h[left]
    return grad


def refine_h(h, time_allow, max_iter=20000):
    """Gradient‑ascent style refinement of a candidate."""
    start = time.time()
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_val = max_overlap(best_h)
    base_step = 0.02
    it = 0
    while it < max_iter and (time.time() - start) < time_allow:
        grad = gradient_worst_shifts(h)
        if not np.any(grad):
            break
        # adaptive step size
        step = base_step / np.sqrt(it + 1.0)
        improved = False
        for _ in range(5):
            cand = h + step * grad
            cand = project_to_feasible(cand, target)
            val = max_overlap(cand)
            if val < best_val - 1e-12:
                best_h, best_val = cand, val
                h = cand
                improved = True
                break
            step *= 0.5
        if not improved:
            # small random nudge to escape possible plateaus
            noise = (np.random.rand(n) - 0.5) * 0.02
            cand = project_to_feasible(h + noise, target)
            val = max_overlap(cand)
            if val < best_val:
                best_h, best_val = cand, val
                h = cand
        it += 1
        if best_val < 0.38085:
            break
    return best_h, best_val
