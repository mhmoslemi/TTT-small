def project_onto_box_and_sum(v, target, lo=0., hi=1., max_iter=60):
    # ensure feasible: target between lo*n and hi*n else cannot.
    n = len(v)
    # initial bounds for lambda
    l = min(v) - hi
    r = max(v) - lo
    for _ in range(max_iter):
        mid = (l + r) / 2
        proj = np.clip(v - mid, lo, hi)
        s = proj.sum()
        if s < target:
            r = mid
        else:
            l = mid
    return np.clip(v - (l+r)/2, lo, hi)

def gradient_for_shifts(h, shifts):
    n = h.shape[0]
    grad = np.full(n, len(shifts), dtype=float)
    for s in shifts:
        if s >= 0:
            # forward term: h[i+s]
            grad[:n-s] -= h[s:]
            # backward term: h[i-s] is h[i-s], which for i>=s => i-s>=0
            grad[s:]   -= h[:n-s]
        else:
            t = -s
            # forward term: h[i+s] = h[i-t], defined for i>=t
            grad[t:]   -= h[:n-t]
            # backward term: h[i-s] = h[i+t] defined for i <= n-1-t
            grad[:n-t] -= h[t:]
    return grad
```

import time, numpy as np, math, random

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def correlation_array(h):
    n = h.shape[0]
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def worst_shifts(h, eps=1e-12):
    n = h.shape[0]
    corr = correlation_array(h)
    max_val = corr.max()
    idx = np.where(corr >= max_val - eps)[0]
    return idx - (n - 1), max_val

def gradient_for_shifts(h, shifts):
    n = h.shape[0]
    k = len(shifts)
    grad = np.full(n, k, dtype=float)
    for s in shifts:
        if s >= 0:
            grad[:n - s] -= h[s:]
            grad[s:] -= h[:n - s]
        else:
            t = -s
            grad[t:] -= h[:n - t]
            grad[:n - t] -= h[t:]
    return grad

def project_onto_box_and_sum(v, target, lo=0.0, hi=1.0, max_iter=60):
    # binary search for lambda
    l = min(v) - hi
    r = max(v) - lo
    for _ in range(max_iter):
        mid = (l + r) * 0.5
        proj = np.clip(v - mid, lo, hi)
        s = proj.sum()
        if s < target:
            r = mid
        else:
            l = mid
    return np.clip(v - (l + r) * 0.5, lo, hi)

def random_feasible(n, rng):
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    # project to box and sum
    v = project_onto_box_and_sum(v, n / 2.0)
    return v

def hill_climb(h, budget, start_time):
    n = h.shape[0]
    best = h.copy()
    best_val = compute_c5(best)
    rng = np.random.default_rng()
    while time.time() - start_time < budget:
        i, j = rng.integers(0, n, size=2)
        if i==j: continue
        # compute feasible delta range
        if best[i] > 0 and best[j] < 1:
            max_delta = min(best[i], 1 - best[j])
        elif best[j] > 0 and best[i] < 1:
            max_delta = min(best[j], 1 - best[i])
            i, j = j, i
        else:
            continue
        if max_delta <= 0: continue
        delta = max_delta * rng.random()
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        val = compute_c5(cand)
        if val < best_val - 1e-12:
            best = cand
            best_val = val
    return best, best_val

def descent_opt(h, time_budget, start_time):
    n = h.shape[0]
    target = n / 2.0
    # initial step size adaptively based on average gradient magnitude
    step = 0.5
    best_h = h.copy()
    best_val = compute_c5(best_h)
    # keep iteration count for step adaptation
    iter_cnt = 0
    while time.time() - start_time < time_budget:
        shifts, cur_max = worst_shifts(best_h)
        # if current best already low and stable break?
        # compute gradient
        grad = gradient_for_shifts(best_h, shifts)
        # try line search
        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125, 0.0625):
            cand = best_h - step * factor * grad
            cand = project_onto_box_and_sum(cand, target)
            val = compute_c5(cand)
            if val < best_val - 1e-12:
                best_h, best_val = cand, val
                step *= 1.05
                improved = True
                break
        if not improved:
            step *= 0.7
            # occasional random perturb
            if iter_cnt % 50 == 0:
                # random local swap
                i, j = np.random.randint(0, n, size=2)
                delta = (np.random.rand() - 0.5) * 0.02
                cand = best_h.copy()
                cand[i] = np.clip(cand[i] + delta, 0., 1.)
                cand[j] = np.clip(cand[j] - delta, 0., 1.)
                cand = project_onto_box_and_sum(cand, target)
                val = compute_c5(cand)
                if val < best_val:
                    best_h, best_val = cand, val
        iter_cnt += 1
        # early stopping if step tiny
        if step < 1e-6:
            break
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # base construction
    if 'initial_h_values' in globals():

    else:

    best_n = None

    for factor in factors:

        # upsample by repeat
        if factor == 1:

        else:

        # ensure exact sum
        target = n / 2.0
        h0 = h0 * (target / h0.sum())
        h0 = np.clip(h0, 0., 1.)
        # two starts: upsampled and random

        if budget_s - (time.time() - start) > 10:

        for h_start in starts:
            remaining = budget_s - (time.time() - start)
            if remaining <= 0:

            # give more budget to larger n
            time_for_descent = min(remaining * 0.4, 30.0)
            h_opt, val_opt = descent_opt(h_start, time_for_descent, start)
            # finish with hill climb if time permits

            if val_opt < best_val:

        best_h = np.full(best_n, 0.5)
        best_val = compute_c5(best_h)
    # final projection and scaling
    best_h = np.clip(best_h, 0., 1.)
    best_h = project_onto_box_and_sum(best_h, best_n / 2.0)
    return (best_h.tolist(), float(best_val), int(best_n))

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """C5 value for a step function h (array)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def correlation_array(h: np.ndarray) -> np.ndarray:
    """Full correlation array (scaled) used by the optimiser."""
    n = h.shape[0]
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def top_shifts(corr: np.ndarray, k: int = 5) -> np.ndarray:
    """Indices (relative to zero shift) of the k largest correlation values."""
    n = (len(corr) + 1) // 2
    k = min(k, corr.size)
    idx = np.argpartition(corr, -k)[-k:]
    return idx - (n - 1)

def gradient_for_shifts(h: np.ndarray, shifts: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the max‑overlap functional for given shift indices."""
    n = h.shape[0]
    k = shifts.size
    grad = np.full(n, k, dtype=float)
    for s in shifts:
        if s >= 0:
            grad[: n - s] -= h[s:]
            grad[s:] -= h[: n - s]
        else:
            t = -s
            grad[t:] -= h[: n - t]
            grad[: n - t] -= h[t:]
    return grad

def project_onto_box_and_sum(v: np.ndarray, target: float,
                             lo: float = 0.0, hi: float = 1.0,
                             max_iter: int = 60) -> np.ndarray:
    """Projection of v onto {x | lo≤x≤hi, sum(x)=target} via a 1‑D bsearch."""
    l = np.min(v) - hi
    r = np.max(v) - lo
    for _ in range(max_iter):
        m = (l + r) * 0.5
        proj = np.clip(v - m, lo, hi)
        s = proj.sum()
        if s < target:
            r = m
        else:
            l = m
    proj = np.clip(v - (l + r) * 0.5, lo, hi)
    # tiny correction for numerical drift
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        interior = (proj > lo) & (proj < hi)
        if interior.any():
            proj[interior] += diff / interior.sum()
            proj = np.clip(proj, lo, hi)
    return proj

def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random vector in [0,1]^n with exact sum n/2."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return project_onto_box_and_sum(v, n / 2.0)

def hill_climb(h: np.ndarray, budget: float, start_time: float) -> tuple:
    """Simple stochastic local search improving the C5 value."""
    n = h.shape[0]
    best = h.copy()
    best_val = compute_c5(best)
    rng = np.random.default_rng()
    while time.time() - start_time < budget:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if best[i] > 0.0 and best[j] < 1.0:
            max_delta = min(best[i], 1.0 - best[j])
        elif best[j] > 0.0 and best[i] < 1.0:
            max_delta = min(best[j], 1.0 - best[i])
            i, j = j, i
        else:
            continue
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random()
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        val = compute_c5(cand)
        if val < best_val - 1e-12:
            best = cand
            best_val = val
    return best, best_val

def descent_opt(h_start: np.ndarray, budget: float, start_time: float) -> tuple:
    """Projected gradient descent using the sub‑gradient of the max‑overlap."""
    n = h_start.shape[0]
    target = n / 2.0
    h = h_start.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    step = 0.5          # initial step length (in direction‐norm units)
    it = 0
    while time.time() - start_time < budget:
        corr = correlation_array(h)
        max_val = corr.max()
        shifts = top_shifts(corr, k=5)
        grad = gradient_for_shifts(h, shifts)
        norm = np.linalg.norm(grad)
        if norm == 0.0:
            break
        direction = -grad / norm
        improved = False
        # try a few step multipliers
        for mul in (1.0, 0.5, 0.25, 0.125, 0.0625):
            cand = h + step * mul * direction
            cand = project_onto_box_and_sum(cand, target)
            val = compute_c5(cand)
            if val < best_val - 1e-12:
                h = cand
                best_h, best_val = cand, val
                step *= 1.05
                improved = True
                break
        if not improved:
            step *= 0.7
            # occasional random jitter to escape plateaus
            if it % 50 == 0:
                i, j = np.random.randint(0, n, size=2)
                delta = (np.random.rand() - 0.5) * 0.02
                cand = h.copy()
                cand[i] = np.clip(cand[i] + delta, 0.0, 1.0)
                cand[j] = np.clip(cand[j] - delta, 0.0, 1.0)
                cand = project_onto_box_and_sum(cand, target)
                val = compute_c5(cand)
                if val < best_val:
                    h = cand
                    best_h, best_val = cand, val
        it += 1
        if step < 1e-6:
            break
    return best_h, best_val

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step function h, its C5 bound, and the number of points."""
    start = time.time()
    rng = np.random.default_rng(seed)

    # base construction (provided or uniform)
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        base = np.full(180, 0.5)

    best_h = None
    best_val = np.inf
    best_n = None

    factors = [1, 2, 3, 4]                     # up‑sampling factors (≤720 points)
    for factor in factors:
        n = base.size * factor
        if factor == 1:
            h0 = base.copy()
        else:
            h0 = np.repeat(base, factor)
        target = n / 2.0
        h0 = h0 * (target / h0.sum())
        h0 = np.clip(h0, 0.0, 1.0)

        # two starting points: the upsampled one and a random one
        starts = [h0]
        if budget_s - (time.time() - start) > 10.0:
            starts.append(random_feasible(n, rng))

        for h_start in starts:
            remaining = budget_s - (time.time() - start)
            if remaining <= 0.0:
                break
            # gradient descent phase
            t_descent = min(remaining * 0.4, 30.0)
            h_opt, val_opt = descent_opt(h_start, t_descent, start)

            # finish with a short hill‑climb if time permits
            remaining2 = budget_s - (time.time() - start)
            if remaining2 > 2.0:
                h_opt, val_opt = hill_climb(h_opt, remaining2 * 0.5, start)

            if val_opt < best_val:
                best_h, best_val, best_n = h_opt, val_opt, n

        if budget_s - (time.time() - start) < 5.0:
            break

    # additional random‑restart phase using remaining time
    while True:
        now = time.time()
        if now - start >= budget_s * 0.95:
            break
        factor = rng.choice(factors)
        n = base.size * factor
        h_rand = random_feasible(n, rng)
        remaining = budget_s - (now - start)
        t_descent = min(remaining, 15.0)
        h_opt, val_opt = descent_opt(h_rand, t_descent, start)
        remaining2 = budget_s - (time.time() - start)
        if remaining2 > 2.0:
            h_opt, val_opt = hill_climb(h_opt, remaining2 * 0.4, start)
        if val_opt < best_val:
            best_h, best_val, best_n = h_opt, val_opt, n

    # fallback in the impossible case
    if best_h is None:
        best_n = base.size
        best_h = np.full(best_n, 0.5)
        best_val = compute_c5(best_h)

    # final projection to guarantee feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_and_sum(best_h, best_n / 2.0)

    return (best_h.tolist(), float(best_val), int(best_n))
