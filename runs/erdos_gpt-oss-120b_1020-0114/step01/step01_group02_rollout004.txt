def gradient_step(h, step):
    N = len(h)
    # compute correlation
    corr = np.correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(corr)  # index in full vector
    s = max_idx - (N - 1)  # shift integer
    # compute gradient g
    g = np.ones(N)
    # subtract h[i+s]
    if s >= 0:

def project_sum_box(x, target):

    for _ in range(50):

        y = np.clip(x + lam, 0.0, 1.0)

        if s < target:
            lo = lam

def _compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)
