def project_to_simplex(r, target, max_iter=50, tol=1e-9):
    lo = -1.0  # maybe min(r) - 1? we need low enough to produce sum≥target; but we can start lo = -max(r)-1 yields sum close to n.
    hi = max(r) + 1 # high enough to produce sum ≤ target.
    for _ in range(max_iter):
        mid = (lo+hi)/2
        proj = np.clip(r - mid, 0, 1)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(r - (lo+hi)/2, 0, 1)
    # adjust small residual difference
    diff = target - proj.sum()
    if abs(diff) > 1e-9:
        # add diff uniformly to elements not at bound
        free = (proj > 0 + 1e-12) & (proj < 1 - 1e-12)
        count = free.sum()
        if count > 0:
            proj[free] += diff / count
            proj = np.clip(proj, 0, 1)
    # final adjust maybe needed but sum may be close enough
    return proj

def upsample_h(old_h, new_n):
    old_n = old_h.shape[0]
    x_old = np.linspace(0,2,old_n,endpoint=False)
    x_new = np.linspace(0,2,new_n,endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    # ensure sum target
    target = new_n / 2
    cur = new_h.sum()
    if cur > 0:
        new_h = new_h * (target / cur)
    new_h = np.clip(new_h, 0,1)
    # adjust residual sum
    cur = new_h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # add diff to random index not at bound
        free = (new_h > 0) & (new_h < 1)
        if free.any():
            idx = np.where(free)[0][0]
            new_h[idx] += diff
            new_h = np.clip(new_h, 0, 1)
        else:
            new_h[0] = max(0, min(1, new_h[0] + diff))
    return new_h

import time
import numpy as np
import scipy.optimize as opt
from scipy.special import logsumexp

_GLOBAL_N = None
_GLOBAL_ALPHA = None


def set_global_params(n, alpha):
    """store problem size and smoothing parameter for the objective."""
    global _GLOBAL_N, _GLOBAL_ALPHA
    _GLOBAL_N = n
    _GLOBAL_ALPHA = alpha


def objective_func(h):
    """smooth approximation to max_{k} ∫h(1‑h∘k) using log‑sum‑exp."""
    n = _GLOBAL_N
    alpha = _GLOBAL_ALPHA
    corr = np.correlate(h, 1.0 - h, mode="full")
    smooth = logsumexp(alpha * corr) / alpha
    return smooth * (2.0 / n)  # multiply by dx


def constraint_sum_eq(h):
    """∑h = n/2  (integral = 1)."""
    n = _GLOBAL_N
    return np.sum(h) - n / 2.0


def compute_c5(h):
    """exact C₅ for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)


def project_to_bounds_and_sum(v, target, iters=60):
    """Euclidean projection of v onto [0,1]^n with prescribed sum=target."""
    lo, hi = -1.0, 1.0
    for _ in range(iters):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        free = (proj > 1e-12) & (proj < 1 - 1e-12)
        cnt = int(free.sum())
        if cnt:
            proj[free] += diff / cnt
            proj = np.clip(proj, 0.0, 1.0)
        else:
            proj[0] = np.clip(proj[0] + diff, 0.0, 1.0)
    return proj


def upsample_h(old_h, new_n):
    """linear interpolation + projection preserving the integral."""
    old_n = old_h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    target = new_n / 2.0
    new_h = project_to_bounds_and_sum(new_h, target)
    return new_h
