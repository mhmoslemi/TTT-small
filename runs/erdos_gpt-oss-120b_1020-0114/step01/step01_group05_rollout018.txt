def project_onto_simplex_box(v, target_sum):
    # Clip to [0,1], then adjust via adding a constant and clipping again.
    # Using the algorithm from previous code: bisection on offset.
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def compute_subgrad_max(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_val = corr.max()

    # Tolerance maybe 1e-12 absolute.
    idxs = np.where(corr >= max_val - 1e-12)[0]
    grad = np.zeros_like(h)

        if k >= 0:
            # use vectorized slice
            grad[:n - k] += (1 - h[k:])
            grad[k:] -= h[:n - k]

import time, numpy as np

def _project(v, target):
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def _evaluate(h):
    n = h.size
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full') * dx)

def _subgrad_max(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_val = corr.max()
    # tolerance to include all near-equal maxima
    eps = 1e-12
    idxs = np.where(corr >= max_val - eps)[0]
    grad = np.zeros_like(h)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            # first factor contribution
            grad[:n - k] += (1 - h[k:])
            # second factor contribution
            grad[k:] -= h[:n - k]
        else:
            kk = -k
            grad[kk:] += (1 - h[:n - kk])
            grad[:n - kk] -= h[kk:]
    grad /= float(len(idxs))
    grad *= dx
    return grad

            h0 = np.interp(new_mid, old_mid, init)
        else:
            h0 = init.copy()
    else:

import time
import numpy as np

def _project(v, target):
    """Project v onto the set {0≤x≤1, sum x = target}."""
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def _evaluate(h):
    """Maximum overlap integral for a discrete h."""
    n = h.size
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full') * dx)

def _subgrad_max(h):
    """Subgradient of the max‑overlap objective (scaled by dx)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_val = corr.max()
    eps = 1e-12
    idxs = np.where(corr >= max_val - eps)[0]
    grad = np.zeros_like(h)
    for idx in idxs:
        k = idx - (n - 1)          # shift amount
        if k >= 0:
            grad[:n - k] += (1 - h[k:])
            grad[k:]    -= h[:n - k]
        else:
            kk = -k
            grad[kk:]   += (1 - h[:n - kk])
            grad[:n - kk] -= h[kk:]
    grad /= float(len(idxs))
    grad *= dx
    return grad

    start = time.time()

    # ----- initialise ------------------------------------------------------
    try:
        init = np.array(initial_h_values, dtype=float)  # provided by harness
    except Exception:

    n_target = 250                     # resolution (≤1000 as requested)
    if init is not None:
        n0 = init.size
        if n0 != n_target:
            dx0 = 2.0 / n0
            old_mid = np.arange(n0) * dx0 + dx0 / 2.0
            dx1 = 2.0 / n_target
            new_mid = np.arange(n_target) * dx1 + dx1 / 2.0
            h0 = np.interp(new_mid, old_mid, init)
        else:
            h0 = init.copy()
    else:

    target_sum = n_target * 0.5        # ∫h = 1  →  sum = n/2
    h = _project(h0, target_sum)

    best_h = h.copy()
