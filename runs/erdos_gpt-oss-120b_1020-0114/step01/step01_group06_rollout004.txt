def proj_box_sum(x, target):
    lo = -np.max(x)  # x+lo <=0 => clip =>0
    hi = 1 - np.min(x) # x+hi >=1 => clip =>1
    for _ in range(60):
        mid = (lo + hi) / 2
        y = np.clip(x + mid, 0, 1)
        s = y.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(x + (lo+hi)/2, 0, 1)

import time, math
import numpy as np

# ---------- Helper functions -------------------------------------------------
def full_corr_fft(h: np.ndarray, nfft: int) -> np.ndarray:
    """Linear correlation of h with (1-h) using FFT (full length)."""
    # v = (1 - h) reversed
    v = (1.0 - h)[::-1]
    H = np.fft.rfft(h, n=nfft)
    V = np.fft.rfft(v, n=nfft)
    conv = np.fft.irfft(H * V, n=nfft)
    return conv[: 2 * h.shape[0] - 1]


def c5_value(h: np.ndarray, nfft: int) -> float:
    """Compute the C5 bound for a step vector h."""
    n = h.shape[0]
    corr = full_corr_fft(h, nfft)
    return float(corr.max() * (2.0 / n))


def proj_box_sum(x: np.ndarray, target: float) -> np.ndarray:
    """
    Project x onto the intersection {0<=y<=1, sum(y)=target}
    by finding a shift λ such that y = clip(x+λ,0,1) has the required sum.
    """
    lo = -np.max(x)               # makes all entries ≤0 after shift
    hi = 1.0 - np.min(x)          # makes all entries ≥1 after shift
    for _ in range(40):
        mid = (lo + hi) * 0.5
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)


def gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the correlation at the maximal shift."""
    n = h.shape[0]
    g = np.ones(n, dtype=np.float64)
    if shift >= 0:
        if shift < n:
            g[: n - shift] -= h[shift:]
            g[shift:] -= h[: n - shift]
    else:
        s = -shift
        if s < n:
            g[s:] -= h[: n - s]
            g[: n - s] -= h[s:]
    return g


def upsample_h(old_h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an old step vector to a new length."""
    old_n = old_h.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(new_x, old_x, old_h)


def sinusoid_start(rng: np.random.Generator, n: int) -> np.ndarray:
    """Generate a smooth sinusoidal start vector."""
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    freq = rng.uniform(0.5, 5.0)
    phase = rng.uniform(0.0, 2.0 * np.pi)
    amp = rng.uniform(0.3, 0.7)
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x + phase)
    h = np.clip(h, 0.0, 1.0)
    return proj_box_sum(h, n / 2.0)

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Choose a moderate resolution (good trade‑off between quality & speed)
    n = 200                     # must be even for exact half‑mass
    target_sum = n / 2.0
    nfft = 1 << ((2 * n - 1).bit_length())   # next power of two ≥ 2n‑1

    best_h = None
    best_c5 = float("inf")

    # ------------------------------------------------------------------------
    # Helper: evaluate a candidate and possibly update the global best
    # ------------------------------------------------------------------------
    def try_update(candidate: np.ndarray):
        nonlocal best_h, best_c5
        cand_c5 = c5_value(candidate, nfft)
        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = candidate.copy()

    # ------------------------------------------------------------------------
    # Prepare a list of starting heuristics
    # ------------------------------------------------------------------------
    start_vectors = []

    # 1. Upsampled initial solution (if supplied)
    try:
        init = np.array(initial_h_values, dtype=float)
        if init.size != n:
            up = upsample_h(init, n)
        else:

        start_vectors.append(up)
    except Exception:
        pass

    # 2. Random uniform vectors
    for _ in range(3):
        rand = rng.random(n)
        start_vectors.append(proj_box_sum(rand, target_sum))

    # 3. Random binary vectors (exactly half ones)
    for _ in range(3):
        bin_vec = np.zeros(n, dtype=float)
        bin_vec[rng.choice(n, n // 2, replace=False)] = 1.0
        start_vectors.append(bin_vec)

    # 4. Sinusoidal starts
    for _ in range(3):
        start_vectors.append(sinusoid_start(rng, n))

    # ------------------------------------------------------------------------
    # Core optimisation loop – restart while time permits
    # ------------------------------------------------------------------------
    restart_idx = 0
    while True:
        # time‑out guard
        if time.time() - start_time > budget_s * 0.99:
            break

        # select a starting point (cycle through the prepared list)
        h = start_vectors[restart_idx % len(start_vectors)].copy()
        restart_idx += 1

        # ---------- Gradient descent phase ----------
        eta = 0.6                               # initial step size
        cur_c5 = c5_value(h, nfft)
        try_update(h)

        for _ in range(800):  # modest number of gradient steps
            if time.time() - start_time > budget_s * 0.95:
                break
            # correlation and worst shift
            corr = full_corr_fft(h, nfft)
            idx = int(np.argmax(corr))
            shift = idx - (n - 1)
            # gradient for that shift
            g = gradient_for_shift(h, shift)
            # candidate update
            h_new = proj_box_sum(h - eta * g, target_sum)
            new_c5 = c5_value(h_new, nfft)

            # accept if improvement
            if new_c5 < cur_c5:
                h, cur_c5 = h_new, new_c5
                try_update(h)
                eta = min(1.0, eta * 1.03)      # slowly increase step if good
            else:
                eta = max(eta * 0.7, 1e-4)       # otherwise shrink step

        # ---------- Simulated‑annealing refinement ----------
        cur_h = h.copy()
        cur_val = cur_c5
        step_sz = 0.25
        temp_start, temp_end = 0.12, 1e-6
        anneal_iters = 25000

        for it in range(anneal_iters):
            if time.time() - start_time > budget_s * 0.99:
                break
            # temperature schedule (linear)
            tau = it / anneal_iters
            temp = temp_start * (1.0 - tau) + temp_end * tau
            # propose a feasible mass swap between two points
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            lower = max(-cur_h[i], -(1.0 - cur_h[j]))
            upper = min(1.0 - cur_h[i], cur_h[j])
            if upper <= lower:
                continue
            # bound by current max step size
            lower = max(lower, -step_sz)
            upper = min(upper, step_sz)
            if upper <= lower:
                continue
            delta = rng.uniform(lower, upper)
            h_trial = cur_h.copy()
            h_trial[i] += delta
            h_trial[j] -= delta
            trial_val = c5_value(h_trial, nfft)

            # Metropolis criterion
            if trial_val < cur_val or rng.random() < math.exp(-(trial_val - cur_val) / temp):
                cur_h, cur_val = h_trial, trial_val
                try_update(cur_h)

            # slowly shrink the maximum exchange size
            if (it + 1) % 1000 == 0 and step_sz > 5e-4:
                step_sz *= 0.985

        # early exit if we already beat the target record
        if best_c5 < 0.3808:
            break

    # ------------------------------------------------------------------------
    # Return the best construction found
    # ------------------------------------------------------------------------
    if best_h is None:
        # fallback – uniform half vector
        best_h = np.full(n, 0.5, dtype=float)
        best_h = proj_box_sum(best_h, target_sum)
        best_c5 = c5_value(best_h, nfft)

    return (best_h.tolist(), float(best_c5), int(n))
