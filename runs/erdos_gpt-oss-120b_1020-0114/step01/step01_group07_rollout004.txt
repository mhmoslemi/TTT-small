import numpy as np, time, math

def _project(v, target):
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    # iterative water-filling
    # we can try to adjust via while loop as previously
    # but to speed up we can use a simple algorithm:
    # Use a loop that distributes diff equally among non-bound variables.
    # We'll replicate earlier code.
    while abs(diff) > 1e-12:
        if diff > 0:
            free = v < 1.0 - 1e-12
            nfree = free.sum()
            if nfree == 0:
                break
            add = diff / nfree
            v[free] = np.minimum(v[free] + add, 1.0)
        else:
            free = v > 1e-12
            nfree = free.sum()
            if nfree == 0:
                break
            sub = diff / nfree
            v[free] = np.maximum(v[free] + sub, 0.0)
        diff = target - v.sum()
    return v

def _c5_of(h):
    N = h.size
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / N)

def _c5_raw(h):
    return np.max(np.correlate(h, 1.0 - h, mode='full'))

def _gradient_max(h):
    N = h.size
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(c)
    s = idx - (N - 1)
    grad = np.zeros_like(h)
    if s >= 0:
        grad[:N - s] += 1.0 - h[s:]
        grad[s:] -= h[:N - s]
    else:
        t = -s
        grad[t:] += 1.0 - h[:N - t]
        grad[:N - t] -= h[t:]
    return grad

def _optimize(N, target_sum, max_iters, lr_start, lr_end, init_h=None):
    if init_h is None:
        h = np.random.rand(N)
    else:
        h = init_h.copy()
    h = _project(h, target_sum)
    best_h = h.copy()
    best_val = _c5_raw(best_h)
    for it in range(max_iters):
        grad = _gradient_max(h)
        norm = np.linalg.norm(grad)
        if norm == 0:
            break
        # step size schedule linear
        lr = lr_start * (1 - it / max_iters) + lr_end
        # normalized direction
        step = -lr * (grad / norm)
        new_h = _project(h + step, target_sum)
        val = _c5_raw(new_h)
        if val < best_val:
            best_val = val
            best_h = new_h.copy()
        # Accept if improvement else keep old (simple hill-climbing)
        if val <= _c5_raw(h) or np.random.rand() < 0.01:
            h = new_h
    return best_h, best_val

import numpy as np, time, math

def _project(v, target):
    """Project onto [0,1] box while forcing sum(v)=target."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    # water‑filling
    while abs(diff) > 1e-12:
        if diff > 0:
            free = v < 1.0 - 1e-12
            nfree = free.sum()
            if nfree == 0:
                break
            add = diff / nfree
            v[free] = np.minimum(v[free] + add, 1.0)
        else:
            free = v > 1e-12
            nfree = free.sum()
            if nfree == 0:
                break
            sub = diff / nfree
            v[free] = np.maximum(v[free] + sub, 0.0)
        diff = target - v.sum()
    return v


def _c5_of(h):
    """Scaled overlap constant."""
    N = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / N)


def _c5_raw(h):
    """Unscaled max correlation."""
    return np.max(np.correlate(h, 1.0 - h, mode="full"))


def _gradient_max(h):
    """Sub‑gradient of the max‑overlap term."""
    N = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    s = idx - (N - 1)                # shift producing the maximum
    g = np.zeros_like(h)
    if s >= 0:
        # first part: i in [0, N-s)
        g[: N - s] += 1.0 - h[s:]
        # second part: i in [s, N)
        g[s:] -= h[: N - s]
    else:
        t = -s
        # first part: i in [t, N)
        g[t:] += 1.0 - h[: N - t]
        # second part: i in [0, N-t)
        g[: N - t] -= h[t:]
    return g


def _upsample(src, N):
    """Linear up‑sampling of a vector to length N."""
    srcN = src.size
    if srcN == N:
        return src.copy()
    x_src = np.linspace(0.0, 1.0, srcN, endpoint=False)
    x_tgt = np.linspace(0.0, 1.0, N, endpoint=False)
    return np.interp(x_tgt, x_src, src)


def _optimize_one(N, target, max_it, lr_start, lr_end, init_h=None, time_limit=None):
    """Projected gradient descent with a simple acceptance rule."""
    if init_h is None:
        h = np.random.rand(N)
    else:
        h = init_h.copy()
    h = _project(h, target)
    cur_val = _c5_raw(h)
    best_h, best_val = h.copy(), cur_val
    for it in range(max_it):
        if time_limit is not None and time.time() > time_limit:
            break
        grad = _gradient_max(h)
        lr = lr_start * (1.0 - it / max_it) + lr_end
        step = -lr * grad
        new_h = _project(h + step, target)
        new_val = _c5_raw(new_h)
        # deterministic acceptance plus a tiny random chance to escape plateaus
        if new_val <= cur_val or np.random.rand() < 0.005:
            h, cur_val = new_h, new_val
            if new_val < best_val:
                best_h, best_val = new_h.copy(), new_val
    return best_h, best_val
