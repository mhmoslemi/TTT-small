    w = np.exp((ov - max_ov) / T)

def project_constraints(h, target):
    # clip
    h = np.clip(h, 0.0, 1.0)
    # adjust sum
    diff = target - h.sum()
    # simple iterative water-filling
    # We'll do loops up to maybe 10 times
    for _ in range(20):
        if abs(diff) < 1e-12:
            break
        # free indices where we can adjust in direction of diff
        if diff > 0:
            # need to increase sum: find indices with h < 1
            free = np.where(h < 1 - 1e-12)[0]
            if free.size == 0:
                break
            delta = diff / free.size
            # limit to [0,1]
            inc = np.minimum(delta, 1 - h[free])
            h[free] += inc
        else:
            # need to decrease sum: find indices with h > 0
            free = np.where(h > 1e-12)[0]
            if free.size == 0:
                break
            delta = diff / free.size
            dec = np.maximum(delta, -h[free])  # delta negative
            h[free] += dec
        diff = target - h.sum()
    # final adjust if needed small residual:
    if abs(diff) > 1e-12:
        # adjust first free element within bounds
        for idx in range(len(h)):
            if diff > 0 and h[idx] < 1:
                step = min(diff, 1 - h[idx])
                h[idx] += step
                diff -= step
                if abs(diff) < 1e-12: break
            elif diff < 0 and h[idx] > 0:
                step = max(diff, -h[idx])
                h[idx] += step
                diff -= step
                if abs(diff) < 1e-12: break
    return h

import time, math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap for step function h (discretized)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)


def _project_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and enforce exact sum == target (simple water‑filling)."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # distribute diff over free variables iteratively
    for _ in range(30):
        free = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
        if not free.size:
            break
        delta = diff / free.size
        # respect bounds
        inc = np.minimum(delta, 1.0 - h[free])
        dec = np.maximum(delta, -h[free])
        adj = np.where(delta >= 0, inc, dec)
        h[free] += adj
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
    # tiny residual fix
    if abs(diff) > 1e-12:
        for i in range(h.size):
            if diff > 0 and h[i] < 1.0:
                step = min(diff, 1.0 - h[i])
                h[i] += step
                diff -= step
            elif diff < 0 and h[i] > 0.0:
                step = max(diff, -h[i])
                h[i] += step
                diff -= step
            if abs(diff) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _sa_optimize(h0: np.ndarray, target_sum: float, time_limit: float):

    n = h0.shape[0]

    step = 0.2          # maximal mass exchange
    T = 0.05            # temperature
    step_min = 1e-8
    T_min = 1e-8
    decay = 0.9995

    while time.time() - start < time_limit:
        # worst shift for the present configuration
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_max = int(np.argmax(corr))
        shift = idx_max - (n - 1)

        # 70 % of the time we take a gradient‑directed move
        if np.random.rand() < 0.7:
            g = np.zeros(n)
            if shift >= 0:
                L = n - shift
                i = np.arange(L)
                j = i + shift
                g[i] += 1.0
                g[i] -= h[j]
                g[j] -= h[i]
            else:

                L = n - s2
                i = np.arange(s2, n)
                j = i - s2
                g[i] += 1.0
                g[i] -= h[j]
                g[j] -= h[i]
            i_up = int(np.argmin(g))
            i_down = int(np.argmax(g))
            max_inc = min(step, 1.0 - h[i_up])
            max_dec = min(step, h[i_down])
            if max_inc > 0 and max_dec > 0:
                delta = np.random.rand() * min(max_inc, max_dec)
                new_h = h.copy()

        else:

def _slsqp_optimize(h0: np.ndarray, time_limit: float):
    """Refine with SciPy's SLSQP if available."""
    try:
        from scipy.optimize import minimize
    except Exception:
        return h0, _compute_c5(h0)

    n = h0.shape[0]
    target_sum = n / 2.0
    t0 = _compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    eq_cons = {'type': 'eq', 'fun': lambda x: np.sum(x[:n]) - target_sum}
    cons = [eq_cons]

    # inequality for each possible shift (full correlation, no dx)
    shifts = list(range(-(n - 1), n))
    def make_ineq(s):
        def fun(x):
            h = x[:n]
            t = x[-1]
            if s >= 0:
                val = np.dot(h[: n - s], 1.0 - h[s:])
            else:
                s2 = -s
                val = np.dot(h[s2:], 1.0 - h[: n - s2])
            return t - val
        return fun
    cons.extend({'type': 'ineq', 'fun': make_ineq(s)} for s in shifts)

    max_iter = max(10, int(min(500, time_limit * 20)))
    opts = {'maxiter': max_iter, 'ftol': 1e-12, 'disp': False}
    res = minimize(lambda x: x[-1], x0, method='SLSQP',
                   bounds=bounds, constraints=cons, options=opts)
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    # enforce exact sum one last time
    diff = target_sum - h_opt.sum()
    if abs(diff) > 1e-10:
        free = np.where((h_opt > 1e-12) & (h_opt < 1.0 - 1e-12))[0]
        if free.size:
            h_opt[free] += diff / free.size
    h_opt = np.clip(h_opt, 0.0, 1.0)
    return h_opt, _compute_c5(h_opt)

    """Optimise a step‑function for the Erdős C₅ constant."""
    np.random.seed(seed)
    start_all = time.time()

    # ----- initialise ----------------------------------------------------
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values, dtype=float).copy()
    else:
        # fallback: 22 ones, 22 zeros, one 0.5 entry (n = 45)
        n_def = 45
        h_start = np.zeros(n_def, dtype=float)
        ones = n_def // 2                     # 22
        idx = np.random.choice(n_def, ones, replace=False)
        h_start[idx] = 1.0
        remaining = n_def / 2.0 - ones        # 0.5
        free = np.where(h_start < 1.0)[0]
        if free.size:
            h_start[free[0]] = remaining

    n = h_start.shape[0]
    target_sum = n / 2.0
    cur = h_start.sum()
    if abs(cur - target_sum) > 1e-12:

        h_start = _project_sum(h_start, target_sum)

    # ----- simulated annealing -------------------------------------------
    time_sa = budget_s * 0.78

    # ----- optional SLSQP refinement ---------------------------------------
    time_remaining = budget_s - (time.time() - start_all)

    final_c5 = _compute_c5(h_best)
    return h_best, float(final_c5), int(n)
