def _smooth_max_optimize(h0, time_limit):
   start = time.time()
   h = h0.copy()
   n = len(h)
   target = n / 2.0
   best_h = h.copy()
   best_val = _compute_c5(h)
   # parameters:
   alpha = 10.0  # smoothing parameter
   step = 0.1  # initial
   decay = 0.99 per iteration
   iter = 0
   while time.time() - start < time_limit:
       # compute convex combination step maybe random perturbation of alpha.
       # Compute conv and weight as described.
       conv = np.correlate(h, 1 - h, mode='full')
       max_conv = conv.max()
       # smoothing weights:
       # large alpha will approximate max; limit.
       exps = np.exp(alpha * (conv - max_conv))
       w = exps / exps.sum()
       # compute grad
       grad = np.zeros_like(h)
       # loop over shift s
       for s in range(-(n-1), n):
           ws = w[s + n - 1]
           if ws == 0:
               continue
           if s >= 0:
               grad[:n-s] += ws * (1 - h[s:])
               grad[s:]   += ws * (-h[:n-s])
           else:
               s2 = -s
               grad[s2:] += ws * (1 - h[:n-s2])
               grad[:n-s2] += ws * (-h[s2:])
       # step
       h_new = h - step * grad
       # projection:
       # clip
       np.clip(h_new, 0.0, 1.0, out=h_new)
       # adjust sum
       diff = target - h_new.sum()
       if abs(diff) > 1e-12:
           # find free indices (not at bounds) to distribute diff
           free = np.where((h_new > 1e-12) & (h_new < 1.0 - 1e-12))[0]
           if free.size > 0:
               h_new[free] += diff / free.size
           else:
               # add to a random or first index within bounds
               idx = np.argmin(h_new)  # smallest value
               h_new[idx] = min(1.0, max(0.0, h_new[idx] + diff))

       # evaluate
       val = _compute_c5(h_new)
       if val < best_val - 1e-12:
           best_val = val
           best_h = h_new.copy()
       # update h
       h = h_new
       # adjust step maybe
       step *= decay
       if step < 1e-6:
           step = 1e-6
       # periodically increase alpha maybe? Keep constant.
       iter += 1

import time, math, numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Discretized C5 = max_k ∫ h(x)(1‑h(x+k))dx."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)


def _project_to_constraints(h: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce sum_i h_i = n/2."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(20):
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size == 0:
            if diff > 0:
                i = np.argmin(h)
                add = min(diff, 1.0 - h[i])
                h[i] += add
            else:
                i = np.argmax(h)
                sub = min(-diff, h[i])
                h[i] -= sub
            diff = target - h.sum()
            continue
        delta = diff / free.size
        h[free] += delta
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        i = np.argmin(h) if diff > 0 else np.argmax(h)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h


def _smooth_max_optimize(h0: np.ndarray, time_limit: float):
    """Smooth‐max gradient descent on the max‑overlap objective."""
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    best_h, best_val = h.copy(), _compute_c5(h)
    step = 0.2
    decay = 0.985
    alpha = 5.0
    max_alpha = 20.0

    while time.time() - start < time_limit:
        conv = np.correlate(h, 1 - h, mode="full")
        m = conv.max()
        # smooth‑max weights
        exps = np.exp(alpha * (conv - m))
        w = exps / exps.sum()

        grad = np.zeros_like(h)
        # accumulate weighted per‑shift gradients
        for s in range(-(n - 1), n):
            ws = w[s + n - 1]
            if ws == 0.0:
                continue
            if s >= 0:
                grad[: n - s] += ws * (1 - h[s:])
                grad[s:]      += ws * (-h[: n - s])
            else:
                s2 = -s
                grad[s2:] += ws * (1 - h[: n - s2])
                grad[: n - s2] += ws * (-h[s2:])

        h_new = h - step * grad
        h_new = _project_to_constraints(h_new)

        val = _compute_c5(h_new)
        if val < best_val - 1e-12:
            best_val, best_h = val, h_new.copy()

        h = h_new
        step *= decay
        if step < 1e-5:
            step = 1e-5
        alpha = min(max_alpha, alpha * 1.02)

    return best_h, best_val


def _greedy_optimize(h0: np.ndarray, time_limit: float):
    """Deterministic / annealing pairwise exchange (as in the reference)."""
    n = h0.shape[0]
    target = n / 2.0
    best_h = h0.copy()
    best_val = _compute_c5(best_h)

    cur_h = h0.copy()
    cur_val = best_val
    step = 0.05
    no_improve = 0
    start = time.time()
    while time.time() - start < time_limit:
        # worst shift
        corr = np.correlate(cur_h, 1 - cur_h, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        # w‑vector for this shift
        if shift == 0:
            w = 2.0 * cur_h
        else:
            w = np.zeros_like(cur_h)
            if shift > 0:
                w[: n - shift] += cur_h[shift:]
                w[shift:] += cur_h[: n - shift]
            else:
                s = -shift
                w[s:] += cur_h[: n - s]
                w[: n - s] += cur_h[s:]

        up_mask = cur_h < 1.0 - 1e-12
        down_mask = cur_h > 1e-12
        if not up_mask.any() or not down_mask.any():
            break
        i_up = np.where(up_mask)[0][np.argmax(w[up_mask])]
        i_down = np.where(down_mask)[0][np.argmin(w[down_mask])]
        delta = min(step, 1.0 - cur_h[i_up], cur_h[i_down])
        if delta <= 0:
            step *= 0.5
            continue
        new_h = cur_h.copy()
        new_h[i_up] += delta
        new_h[i_down] -= delta
        new_val = _compute_c5(new_h)
        if new_val < cur_val - 1e-12:
            cur_h, cur_val = new_h, new_val
            no_improve = 0
            step = min(step * 1.05, 0.1)
            if cur_val < best_val:
                best_val, best_h = cur_val, cur_h.copy()
        else:
            T = step * 0.5
            if T > 0 and np.random.rand() < math.exp(-(new_val - cur_val) / T):
                cur_h, cur_val = new_h, new_val
            no_improve += 1
            if no_improve > 400:
                step *= 0.5
                no_improve = 0
                if step < 1e-6:
                    break
    # final projection onto the hyperplane
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        free = np.where((best_h > 1e-12) & (best_h < 1.0 - 1e-12))[0]
        if free.size:
            best_h[free] += diff / free.size
        else:
            best_h[0] = min(1.0, max(0.0, best_h[0] + diff))
        best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, _compute_c5(best_h)


def _slsqp_optimize(h0: np.ndarray, time_limit: float):
    """SLSQP refinement, if SciPy is available."""
    try:
        from scipy.optimize import minimize
    except Exception:
        return h0, _compute_c5(h0)

    n = h0.shape[0]
    target = n / 2.0
    t0 = _compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    eq_cons = {'type': 'eq', 'fun': lambda x: np.sum(x[:n]) - target}
    cons = [eq_cons]

    shifts = list(range(-(n - 1), n))
    def make_ineq(s):
        def fun(x):
            h = x[:n]
            t = x[-1]
            if s >= 0:
                val = np.dot(h[: n - s], 1.0 - h[s:])
            else:
                s2 = -s
                val = np.dot(h[s2:], 1.0 - h[: n - s2])
            return t - val
        return fun
    cons.extend({'type': 'ineq', 'fun': make_ineq(s)} for s in shifts)

    max_iter = max(10, int(min(500, time_limit * 20)))
    opts = {'maxiter': max_iter, 'ftol': 1e-12, 'disp': False}

    res = minimize(lambda x: x[-1], x0, method='SLSQP',
                   bounds=bounds, constraints=cons, options=opts)

    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    diff = target - h_opt.sum()
    if abs(diff) > 1e-10:
        free = np.where((h_opt > 1e-12) & (h_opt < 1.0 - 1e-12))[0]
        if free.size:
            h_opt[free] += diff / free.size
        else:
            i = np.argmin(h_opt) if diff > 0 else np.argmax(h_opt)
            h_opt[i] = np.clip(h_opt[i] + diff, 0.0, 1.0)
        h_opt = np.clip(h_opt, 0.0, 1.0)
    return h_opt, _compute_c5(h_opt)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimise a step‑function h on [0,2] for the Erdős C₅ constant.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_all = time.time()
    # ----- initialise -------------------------------------------------
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values, dtype=float).copy()
        n = h_start.shape[0]
    else:
        n = 45
        h_start = np.zeros(n, dtype=float)
        ones = np.random.choice(n, n // 2, replace=False)
        h_start[ones] = 1.0
        remaining = np.setdiff1d(np.arange(n), ones)
        h_start[np.random.choice(remaining)] = 0.5

    h_start = _project_to_constraints(h_start)

    best_h, best_val = h_start, _compute_c5(h_start)

    # ----- multi‑restart smooth‑max optimisation --------------------
    total_budget = budget_s
    search_budget = total_budget * 0.70
    per_restart = 5.0
    max_restarts = max(1, int(search_budget // per_restart))
    max_restarts = min(max_restarts, 150)

    for r in range(max_restarts):
        now = time.time()
        time_left = start_all + search_budget - now
        if time_left <= 0.5:
            break
        t_limit = min(per_restart, time_left)
        if r == 0 and 'initial_h_values' in globals():
            h0 = h_start
        else:
            # random binary‑fractional start respecting the sum
            ones = np.random.choice(n, n // 2, replace=False)
            rem = np.setdiff1d(np.arange(n), ones)
            half = np.random.choice(rem)
            h0 = np.zeros(n, dtype=float)
            h0[ones] = 1.0
            h0[half] = 0.5
        h_opt, val_opt = _smooth_max_optimize(h0, t_limit)
        if val_opt < best_val:
            best_h, best_val = h_opt, val_opt
        if best_val < 0.3808:
            break

    # ----- greedy refinement -----------------------------------------
    time_spent = time.time() - start_all
    time_left = total_budget - time_spent
    if time_left > 2.0:
        h_greedy, val_greedy = _greedy_optimize(best_h, time_left * 0.5)
        if val_greedy < best_val:
            best_h, best_val = h_greedy, val_greedy

    # ----- final SLSQP refinement ------------------------------------
    time_spent = time.time() - start_all
    time_left = total_budget - time_spent
    if time_left > 2.0:
        h_ref, val_ref = _slsqp_optimize(best_h, time_left)
        if val_ref < best_val:
            best_h, best_val = h_ref, val_ref

    best_h = _project_to_constraints(best_h)
    final_c5 = _compute_c5(best_h)
    return best_h, float(final_c5), int(n)
