def upsample_h(old_h, new_n):
    old_n = len(old_h)
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    target = new_n / 2.0
    cur = new_h.sum()
    if cur != 0:
        new_h *= target / cur
    np.clip(new_h, 0., 1., out=new_h)

    cur = new_h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        new_h += diff / new_n
        np.clip(new_h, 0., 1., out=new_h)
        cur = new_h.sum()
        if cur != 0:
            new_h *= target / cur

import time
import numpy as np
import scipy.optimize as opt
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global helpers (no closures, no lambdas)
# ----------------------------------------------------------------------
_GLOBAL_ALPHA = None  # smoothing factor for the soft‑max objective


def set_global_alpha(alpha: float) -> None:
    """Set the smoothing factor used by `soft_max_obj`."""
    global _GLOBAL_ALPHA
    _GLOBAL_ALPHA = alpha


def soft_max_obj(h: np.ndarray) -> float:
    """
    Smoothed approximation of max_k ∫ h(x)(1‑h(x+k))dx.
    The correlation (without dx) is computed with np.correlate.
    """
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return logsumexp(_GLOBAL_ALPHA * corr) / _GLOBAL_ALPHA * (2.0 / n)


def constraint_sum_eq(h: np.ndarray) -> float:
    """Equality constraint: ∫h = 1  ⇔  sum(h) = n/2."""
    n = h.shape[0]
    return np.sum(h) - n / 2.0


def compute_c5(h: np.ndarray) -> float:
    """
    Exact C5 value for a discrete step function.
    Uses FFT based linear correlation (faster than np.correlate for many calls).
    """
    n = h.shape[0]
    dx = 2.0 / n
    rev = (1.0 - h)[::-1]                     # reversed second array
    m = 1
    target_len = 2 * n - 1
    while m < target_len:
        m <<= 1                                 # next power of two
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:target_len]  # linear correlation
    return float(np.max(conv) * dx)


def upsample_h(old_h: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation to a finer grid while preserving the integral.
    """
    old_n = old_h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)

    target_sum = new_n / 2.0
    cur_sum = new_h.sum()
    if cur_sum != 0:
        new_h *= target_sum / cur_sum

    np.clip(new_h, 0.0, 1.0, out=new_h)

    # tiny correction for any remaining sum error
    cur_sum = new_h.sum()
    diff = target_sum - cur_sum
    if abs(diff) > 1e-12:
        new_h += diff / new_n
        np.clip(new_h, 0.0, 1.0, out=new_h)
        cur_sum = new_h.sum()
        if cur_sum != 0:
            new_h *= target_sum / cur_sum

    return new_h


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function giving a small C₅ bound."""
    np.random.seed(seed)
    start = time.time()

    def elapsed():
        return time.time() - start

    # ------------------------------------------------------------------
    # Initial construction (the harness may provide `initial_h_values`)
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.asarray(initial_h_values, dtype=float)
        n = h.size
        target_sum = n / 2.0
        cur_sum = h.sum()
        if cur_sum != 0:
            h *= target_sum / cur_sum
        np.clip(h, 0.0, 1.0, out=h)
    else:
        n = 200
        h = np.full(n, 0.5, dtype=float)  # fallback uniform function

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Optional up‑sampling to a power‑of‑two size (256) for faster FFTs
    # ------------------------------------------------------------------
    target_n = 256
    if n != target_n:
        h = upsample_h(h, target_n)
        n = target_n
        best_h = h.copy()
        best_c5 = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Stage 1 – several SLSQP refinements with decreasing temperature
    # ------------------------------------------------------------------
    alphas = [30.0, 60.0, 120.0]          # smoothing factors (larger → sharper)
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        if elapsed() > budget_s * 0.6:    # preserve time for the annealing stage
            break
        set_global_alpha(alpha)

        for restart in range(3):
            if elapsed() > budget_s * 0.6:
                break
            # perturb current best slightly
            pert = best_h + np.random.uniform(-0.05, 0.05, size=n)
            np.clip(pert, 0.0, 1.0, out=pert)
            pert *= (n / 2.0) / pert.sum()

            res = opt.minimize(
                soft_max_obj,
                pert,
                method="SLSQP",
                bounds=bounds,
                constraints=[{"type": "eq", "fun": constraint_sum_eq}],
                options={"maxiter": 300, "ftol": 1e-9, "disp": False},
            )
            if not res.success:
                continue
            cand = np.clip(res.x, 0.0, 1.0)
            cand *= (n / 2.0) / cand.sum()
            cand_c5 = compute_c5(cand)
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = cand.copy()

    # ------------------------------------------------------------------
    # Stage 2 – simulated annealing on the current best vector
    # ------------------------------------------------------------------
    h = best_h.copy()
    cur_c5 = best_c5
    T0 = cur_c5 * 0.02                    # initial temperature (≈2 % of current C₅)

    while elapsed() < budget_s * 0.95:
        # linear cooling
        prog = (elapsed() / (budget_s * 0.95))
        T = T0 * (1.0 - prog)

        # pick two distinct indices
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # amount we can legally transfer
        eps_max = min(h[i], 1.0 - h[j])
        if eps_max <= 0.0:
            continue

        # choose a transfer amount (sometimes the full possible amount)
        if np.random.rand() < 0.1:
            eps = eps_max
        else:
            eps = eps_max * np.random.rand() * 0.5  # moderate steps

        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps

        new_c5 = compute_c5(new_h)
        delta = new_c5 - cur_c5

        if delta < 0.0 or np.random.rand() < np.exp(-delta / max(T, 1e-12)):
            h = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()

        # occasional global perturbation to shake out of deep minima
        if np.random.rand() < 0.001:
            noise = np.random.uniform(-0.02, 0.02, size=n)
            h = np.clip(h + noise, 0.0, 1.0)
            h *= (n / 2.0) / h.sum()
            cur_c5 = compute_c5(h)

    # ------------------------------------------------------------------
    # Stage 3 – final sharp SLSQP pass (large alpha)
    # ------------------------------------------------------------------
    if elapsed() < budget_s * 0.99:
        set_global_alpha(250.0)
        for _ in range(2):
            if elapsed() > budget_s * 0.99:
                break
            start_pt = best_h + np.random.uniform(-0.02, 0.02, size=n)
            np.clip(start_pt, 0.0, 1.0, out=start_pt)
            start_pt *= (n / 2.0) / start_pt.sum()

            res = opt.minimize(
                soft_max_obj,
                start_pt,
                method="SLSQP",
                bounds=bounds,
                constraints=[{"type": "eq", "fun": constraint_sum_eq}],
                options={"maxiter": 200, "ftol": 1e-10, "disp": False},
            )
            if res.success:
                cand = np.clip(res.x, 0.0, 1.0)
                cand *= (n / 2.0) / cand.sum()
                cand_c5 = compute_c5(cand)
                if cand_c5 < best_c5:
                    best_c5 = cand_c5
                    best_h = cand.copy()

    # ------------------------------------------------------------------
    # Final sanity check and return
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    tot = best_h.sum()
    target = best_h.size / 2.0
    if tot != 0 and abs(tot - target) > 1e-9:
        best_h = best_h * (target / tot)
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)
    return (best_h, float(final_c5), int(best_h.size))
