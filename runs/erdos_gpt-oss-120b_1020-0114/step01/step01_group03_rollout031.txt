_GLOBAL_N = None
_GLOBAL_ALPHA = None


def set_global_params(n, alpha):
    """Set problem size and smoothing parameter for the smooth objective."""
    global _GLOBAL_N, _GLOBAL_ALPHA
    _GLOBAL_N = n
    _GLOBAL_ALPHA = alpha


def objective_func(h):
    """Smooth approximation of the max overlap integral.
    Uses log‑sum‑exp with smoothing factor _GLOBAL_ALPHA."""
    n = _GLOBAL_N
    alpha = _GLOBAL_ALPHA
    corr = np.correlate(h, 1.0 - h, mode="full")
    smooth_max = logsumexp(alpha * corr) / alpha
    return smooth_max * (2.0 / n)


def constraint_sum_eq(h):
    """Equality constraint: ∫h = 1  ⇔  sum(h) = n/2."""
    n = _GLOBAL_N
    return np.sum(h) - n / 2.0


def compute_c5(h):
    """Exact C₅ value for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)


def compute_delta_swap(h, a, b):
    """Delta of the self‑correlation O = h⋆h for swapping a (1→0) with b (0→1)."""
    n = h.shape[0]
    L = 2 * n - 1

    delta.fill(0)

    # forward contributions (h[i] * h[i+k])
    delta[n - 1 - a : n - 1 - a + n] -= h
    delta[n - 1 - b : n - 1 - b + n] += h

    delta[shift + n - 1] -= 1          # k = b‑a

            objective_func,
            h_cont,
            method="SLSQP",
            bounds=bounds,
            constraints=[cons],
            options={"maxiter": 300, "ftol": 1e-9, "disp": False},
        )
        h_cont = np.clip(res.x, 0.0, 1.0)
        # enforce the integral exactly after each stage
        s = h_cont.sum()
        if s != 0:
            h_cont *= (target_sum / s)
        h_cont = np.clip(h_cont, 0.0, 1.0)

    # ------------------------------------------------------------------
    # Final evaluation and return.
    # ------------------------------------------------------------------
    final_c5 = compute_c5(h_cont)
    # final sanity checks
    h_cont = np.clip(h_cont, 0.0, 1.0)
    s = h_cont.sum()
    if abs(s - target_sum) > 1e-9:
        h_cont = h_cont * (target_sum / s)
        h_cont = np.clip(h_cont, 0.0, 1.0)

    return (h_cont, float(final_c5), int(n))
