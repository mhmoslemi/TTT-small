def project_feasible(v, target_sum):
    # clip to [0,1] then adjust with water-filling:
    v = np.clip(v, 0., 1.)
    # adjust constant c such that sum(np.clip(v + c,0,1)) = target_sum
    lo, hi = -1., 1.
    for _ in range(30):
        mid = (lo+hi)/2
        adj = np.clip(v+mid,0.,1.)
        s = adj.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo+hi)/2, 0., 1.)

def subgradient(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_val = corr.max()
    # indices of max (tolerance)
    idxs = np.where(np.abs(corr - max_val) <= 1e-12)[0] # integer positions in full correlation
    grads = np.zeros_like(h)
    for idx in idxs:
        lag = idx - (n-1)   # integer shift
        # compute forward/backward contributions
        if lag >= 0:
            # h_i * h_{i+lag}
            # gradient = -dx*(h_{i+lag} + h_{i-lag}) with proper handling
            # For i from 0 to n-1:
            # term h_{i+lag} exists for i+lag < n -> we can build shift arrays
            forward = np.concatenate((np.zeros(lag), h[:n-lag]))
            backward = np.concatenate((h[lag:], np.zeros(lag))) # shift -lag
        else:
            shift = -lag
            forward = np.concatenate((h[shift:], np.zeros(shift)))   # i+lag = i - shift
            backward = np.concatenate((np.zeros(shift), h[:n-shift]))
        grads -= dx * (forward + backward)
    grads /= len(idxs) # average
    return grads
```

def random_move(h, max_delta=0.1):
    n = len(h)
    i, j = np.random.choice(n, size=2, replace=False)
    # permissible delta range:
    low = max(-h[i], h[j] - 1.0)
    high = min(1.0 - h[i], h[j])
    # also limit magnitude by max_delta:
    low = max(low, -max_delta)
    high = min(high, max_delta)
    if low >= high:
        return None, None, None  # no feasible move
    delta = np.random.uniform(low, high)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new, i, j
```

import time
import numpy as np

# ---------- helper functions -------------------------------------------------
def _evaluate(h):
    """Return max overlap C5 for sample vector h."""
    n = h.size
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode="full") * dx)


def _project(v, target):
    """Euclidean projection onto {0≤x≤1, sum x = target}."""
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _corr_and_max(h):
    """Return full correlation array and its maximum."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return corr, corr.max()


def _subgradient(h, max_idx):
    """Sub‑gradient of max_{lag} C5 at current h (averaged over all max lags)."""
    n = h.size
    dx = 2.0 / n
    g = np.zeros_like(h)
    for idx in max_idx:
        lag = idx - (n - 1)                 # lag ∈ [-(n-1), n-1]
        if lag >= 0:
            forward  = np.concatenate((h[lag:], np.zeros(lag)))
            backward = np.concatenate((np.zeros(lag), h[:n - lag]))
        else:
            shift = -lag
            forward  = np.concatenate((np.zeros(shift), h[:n - shift]))
            backward = np.concatenate((h[shift:], np.zeros(shift)))
        g -= dx * (forward + backward)
    g /= len(max_idx)
    return g


def _random_pair_move(h, max_delta=0.12):
    """Propose a feasible move that changes exactly two coordinates."""
    n = h.size
    i, j = np.random.choice(n, size=2, replace=False)
    hi, hj = h[i], h[j]
    # feasible interval for delta (add to i, subtract from j)
    lo = max(-hi, hj - 1.0, -max_delta)
    hi_ = min(1.0 - hi, hj, max_delta)
    if lo >= hi_:
        return None
    delta = np.random.uniform(lo, hi_)
    h_new = h.copy()
    h_new[i] = hi + delta
    h_new[j] = hj - delta
    return h_new

    # --- initialise ---------------------------------------------------------

    n = h0.size
    target_sum = n * 0.5
    h = _project(h0, target_sum)

    cur_c5 = _evaluate(h)
    best_h = h.copy()
    best_c5 = cur_c5

    # --- optimisation loop --------------------------------------------------
    last_improve = 0
    it = 0
    # parameters
    tol = 1e-9                # tolerance for “max” lag detection
    max_iters = 200000       # large upper bound; time guard will stop earlier
    while it < max_iters and (time.time() - start) < budget_s * 0.95:
        it += 1
