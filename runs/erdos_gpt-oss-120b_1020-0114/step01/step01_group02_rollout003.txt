def gradient_worst_lag(h, shift):
    N = len(h)
    d = np.ones(N)
    if shift > 0:
        # subtract partner forward
        d[:N-shift] -= h[shift:]
        # subtract partner backward
        d[shift:] -= h[:N-shift]
    elif shift < 0:
        s = -shift
        # subtract partner backward (i-s) as forward for s
        d[s:] -= h[:N-s]
        d[:N-s] -= h[s:]
    else:
        d = 1 - 2*h
    return d

import numpy as np, time, math, random

def _compute_c5(h):
    """Maximum overlap integral for discrete step function h."""
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

def _repair_sum(h, target):
    """Adjust h in‑place so that sum(h) == target while staying in [0,1]."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # greedy random fill/empty
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = np.random.randint(len(h))
        if diff > 0:
            cap = 1.0 - h[i]
            if cap <= 0:
                continue
            inc = min(cap, diff)
            h[i] += inc
            diff -= inc
        else:
            cap = h[i]
            if cap <= 0:
                continue
            dec = min(cap, -diff)
            h[i] -= dec
            diff += dec
    if abs(diff) > 1e-12:
        total = h.sum()
        if total > 0:
            h *= (target / total)
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                idxs = np.where(h < 1.0)[0]
                if idxs.size:
                    i = np.random.choice(idxs)
                    h[i] = min(1.0, h[i] + diff)
            else:
                idxs = np.where(h > 0.0)[0]
                if idxs.size:
                    i = np.random.choice(idxs)
                    h[i] = max(0.0, h[i] + diff)
    return h

def _gradient_worst_lag(h, shift):
    """Derivative of f(shift)=∫h(x)(1−h(x+shift))dx w.r.t each component."""
    N = len(h)
    if shift == 0:
        return 1.0 - 2.0 * h
    d = np.ones(N)
    if shift > 0:
        d[:N - shift] -= h[shift:]
        d[shift:]   -= h[:N - shift]
    else:  # shift < 0
        s = -shift
        d[s:]   -= h[:N - s]
        d[:N - s] -= h[s:]
    return d

    N = len(h)
    cand = h.copy()
    idx = np.arange(N - s)
    avg = (cand[idx] + cand[idx + s]) * 0.5
    cand[idx] = avg
    cand[idx + s] = avg
    return cand

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound and number of points."""
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    total = budget_s

    # ------------------------------------------------------------------
    # problem size
    N = 200                     # fine grid
    target_sum = N / 2.0
    # initialise h
    if "initial_h_values" in globals():
        init0 = np.array(initial_h_values, dtype=np.float64)
        N0 = len(init0)
        xs0 = np.linspace(0.0, 2.0, N0, endpoint=False)
        xs = np.linspace(0.0, 2.0, N, endpoint=False)
        h = np.interp(xs, xs0, init0)
    else:
        h = np.zeros(N, dtype=np.float64)
        half = N // 2
        h[:half] = 1.0
    h = _repair_sum(h, target_sum)
    h = np.clip(h, 0.0, 1.0)

    cur_h = h.copy()
    cur_c5 = _compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    # SA parameters
    T0, T_end = 0.2, 1e-6
    max_move_initial = 0.2

    # move probabilities
    p_restart = 0.001
    p_large   = 0.05
    p_target  = 0.5
    p_equal   = 0.1   # pair equalisation of worst lag

    while True:
        elapsed = time.time() - start
        if elapsed > total - 5.0:
            break
        progress = elapsed / (total - 5.0)
        T = T0 * (T_end / T0) ** progress
        max_move = max_move_initial * (1.0 - progress) + 1e-4

        r = np.random.rand()
        cand = None

        # ------------------------------------------------------------------
        # 1) restart from fresh binary pattern
        if r < p_restart:
            cand = np.zeros(N, dtype=np.float64)
            idx = np.random.choice(N, N // 2, replace=False)
            cand[idx] = 1.0

        # 2) large random perturbation of current state
        elif r < p_restart + p_large:
            cand = cur_h.copy()
            m = np.random.randint(1, max(2, N // 20 + 1))
            idxs = np.random.choice(N, m, replace=False)
            cand[idxs] = np.random.rand(m)
            _repair_sum(cand, target_sum)

        else:
            # compute worst lag once (used by several move types)
            corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            idx_max = int(np.argmax(corr))
            shift = idx_max - (N - 1)   # integer shift corresponding to worst overlap

            # 3) gradient‑directed transfer move
            if r < p_restart + p_large + p_target:
                d = _gradient_worst_lag(cur_h, shift)
                pos = np.where(d > 0.05)[0]
                neg = np.where(d < -0.05)[0]
                if pos.size and neg.size:
                    i = int(np.random.choice(pos))
                    j = int(np.random.choice(neg))
                    delta = min(max_move, cur_h[i], 1.0 - cur_h[j])
                    if delta > 1e-12:
                        cand = cur_h.copy()
                        cand[i] -= delta
                        cand[j] += delta
                # fallback to simple transfer if gradient set empty or delta zero
                if cand is None:
                    # simple transfer (preserves sum)
                    i, j = np.random.randint(N, size=2)
                    if i != j:
                        max_delta = min(max_move, cur_h[i], 1.0 - cur_h[j])
                        if max_delta <= 0:
                            max_delta = min(max_move, cur_h[j], 1.0 - cur_h[i])
                            if max_delta > 0:
                                i, j = j, i
                        if max_delta > 0:
                            delta = np.random.uniform(0.0, max_delta)
                            cand = cur_h.copy()
                            cand[i] -= delta
                            cand[j] += delta

            # 4) equalise pair values for the worst lag
            elif r < p_restart + p_large + p_target + p_equal:
                cand = _equalize_worst_lag(cur_h, shift)
                # sum is unchanged, bounds stay within [0,1] automatically

        # if still None (unlikely), fall back to simple transfer
        if cand is None:
            i, j = np.random.randint(N, size=2)
            if i == j:
                continue
            max_delta = min(max_move, cur_h[i], 1.0 - cur_h[j])
            if max_delta <= 0:
                max_delta = min(max_move, cur_h[j], 1.0 - cur_h[i])
                if max_delta <= 0:
                    continue
                i, j = j, i
            delta = np.random.uniform(0.0, max_delta)
            cand = cur_h.copy()
            cand[i] -= delta
            cand[j] += delta

        cand_c5 = _compute_c5(cand)
        delta_c5 = cand_c5 - cur_c5

        if delta_c5 < 0 or np.random.rand() < math.exp(-delta_c5 / max(T, 1e-12)):
            cur_h = cand
            cur_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = cand.copy()
                if best_c5 < 0.3808:
                    break

            # ------------------------------------------------------------------
            # short deterministic polish using the same gradient idea
            for _ in range(3):
                corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
                idx_max = int(np.argmax(corr))
                shift = idx_max - (N - 1)
                d = _gradient_worst_lag(cur_h, shift)
                pos = np.where(d > 0.05)[0]
                neg = np.where(d < -0.05)[0]
                if not pos.size or not neg.size:
                    break
                i = int(np.random.choice(pos))
                j = int(np.random.choice(neg))
                delta = min(0.03, cur_h[i], 1.0 - cur_h[j])
                if delta <= 1e-12:
                    break
                cand2 = cur_h.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2_c5 = _compute_c5(cand2)
                if cand2_c5 < cur_c5:
                    cur_h = cand2
                    cur_c5 = cand2_c5
                    if cand2_c5 < best_c5:
                        best_c5 = cand2_c5
                        best_h = cand2.copy()
                else:
                    break

    # ------------------------------------------------------------------
    # final tidy‑up
    best_h = _repair_sum(best_h, target_sum)
    final_c5 = _compute_c5(best_h)

    return best_h.tolist(), float(final_c5), len(best_h)
