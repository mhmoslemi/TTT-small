def gradient_from_shifts(h, shifts):

    for s in shifts:

            # contributions to indices 0..n-1-s from h[s:]
            g[:n-s] += h[s:]
            # contributions to indices s..n-1 from h[:n-s]

import time
import numpy as np

# --------------------------------------------------------------
# Helper utilities
# --------------------------------------------------------------

def compute_c5(h, dx):
    """Maximum overlap integral for a given step function h."""
    arr = 1.0 - h
    corr = np.correlate(h, arr, mode="full") * dx
    return float(np.max(corr))


def project_to_capped_simplex(v, target):
    """
    Projection of vector v onto the set
        {x : 0 <= x_i <= 1,  sum_i x_i = target }.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = -np.max(v) - 1.0   # makes every entry >=1 after shifting
    hi = np.max(v) + 1.0    # makes every entry <=0 after shifting
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction to meet the equality (numerical)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if free.size:
            inc = diff / free.size
            x[free] += inc
            np.clip(x, 0.0, 1.0, out=x)
    return x


def worst_shifts_O(h, dx, tol=1e-12):
    """
    Returns the integer shifts s for which the overlap O(s) attains its
    maximum value (i.e. the worst shifts) and that maximal value.
    """
    arr = 1.0 - h
    corr = np.correlate(h, arr, mode="full") * dx
    max_val = np.max(corr)
    idx = np.where(corr >= max_val - tol)[0]
    shifts = idx - (len(h) - 1)
    return shifts, max_val


def gradient_from_shifts(h, shifts):
    """
    Sub‑gradient of the overlap O(s) for the worst shifts.
    For each shift s, the derivative of O(s) w.r.t. h_i is
        -(h_{i+s} + h_{i-s})   (zero where indices are out of range).
    We return the positive direction that *decreases* O(s).
    """
    n = h.shape[0]
    g = np.zeros_like(h)
    for s in shifts:
        if s >= 0:
            # i from 0 .. n-1-s  -> contribution h_{i+s}
            g[: n - s] += h[s:]
            # i from s .. n-1    -> contribution h_{i-s}
            g[s:] += h[: n - s]
        else:
            s2 = -s
            # symmetric handling for negative shift
            g[s2:] += h[: n - s2]
            g[: n - s2] += h[s2:]
    return g

# --------------------------------------------------------------

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Problem size – we use a finer grid (200 points) unless a parent is
    # already present with the same size.
    # ------------------------------------------------------------------
    desired_n = 200
    try:
        init_parent = np.array(initial_h_values, dtype=float)
        parent_n = init_parent.shape[0]
    except Exception:
        init_parent = None
        parent_n = 0

    if init_parent is not None and parent_n > 0:
        # linear interpolation to the desired resolution
        x_old = np.linspace(0.0, 2.0, parent_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, desired_n, endpoint=False)
        h = np.interp(x_new, x_old, init_parent)
    else:
        h = rng.random(desired_n)

    target_sum = desired_n / 2.0
    h = project_to_capped_simplex(h, target_sum)
    dx = 2.0 / desired_n

    # ------------------------------------------------------------------
    # Initial evaluation
    # ------------------------------------------------------------------
    best_h = h.copy()
    try:
        best_c5 = float(evaluate_erdos_solution(best_h))
    except Exception:
        best_c5 = compute_c5(best_h, dx)

    # ------------------------------------------------------------------
    # Stochastic improvement loop (simple simulated‑annealing / hill‑climb)
    # ------------------------------------------------------------------
    max_iters = 500_000           # generous upper bound – real stop by time
    base_step = 0.07              # magnitude of deterministic gradient step
    base_noise = 0.025            # magnitude of random perturbation
    it = 0

    while time.time() - start_time < budget_s * 0.95 and it < max_iters:
        # gradient that tries to lower the current worst overlap
        shifts, cur_max = worst_shifts_O(best_h, dx)
        g = gradient_from_shifts(best_h, shifts)

        # normalise gradient to avoid dependence on its scale
        norm = np.linalg.norm(g)
        if norm > 0:
            direction = g / norm
        else:
            direction = np.zeros_like(g)

        # exponential decay of step/noise magnitudes
        decay = 0.9997 ** it
        step = base_step * decay
        noise = base_noise * decay * rng.standard_normal(desired_n)

        cand = best_h + step * direction + noise
        cand = np.clip(cand, 0.0, 1.0)
        cand = project_to_capped_simplex(cand, target_sum)

        c5_val = compute_c5(cand, dx)

        # Accept any improvement (greedy) – also occasional random reset
        if c5_val + 1e-12 < best_c5:
            best_h = cand
            best_c5 = c5_val

        # every few thousand iterations try a fresh random feasible point
        if it % 4000 == 0:
            rnd = rng.random(desired_n)
            rnd = project_to_capped_simplex(rnd, target_sum)
            rnd_c5 = compute_c5(rnd, dx)
            if rnd_c5 + 1e-12 < best_c5:
                best_h = rnd
                best_c5 = rnd_c5

        it += 1

    # ------------------------------------------------------------------
    # Final evaluation with the official helper (more robust)
    # ------------------------------------------------------------------
    try:
        final_c5 = float(evaluate_erdos_solution(best_h))
    except Exception:
        final_c5 = best_c5

    return best_h, final_c5, desired_n
