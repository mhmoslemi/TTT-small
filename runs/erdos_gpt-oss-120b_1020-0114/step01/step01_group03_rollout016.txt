def anneal(h0, time_budget, start_temp=0.02, decay=0.9995):
    h = h0.copy()
    n = len(h)
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    temp = start_temp
    iteration = 0
    while time.time() < end_time:
        # compute lag where max occurs
        corr = np.correlate(h, 1-h, mode='full')
        idx_max = np.argmax(corr)
        lag = idx_max - (n - 1)
        # compute gradient
        grad = np.empty(n)
        # compute vectorized
        # term1 = 1 - h[i+lag] where i+lag in range else 1
        # term2 = h[i-lag] where i-lag in range else 0
        pos = np.arange(n) + lag
        term1 = np.where((pos >= 0) & (pos < n), 1.0 - h[pos], 1.0)
        neg = np.arange(n) - lag
        term2 = np.where((neg >= 0) & (neg < n), h[neg], 0.0)
        grad = term1 - term2
        # indices where we can move mass
        pos_idx = np.where((grad > 0) & (h > 1e-12))[0]
        neg_idx = np.where((grad < 0) & (h < 1.0 - 1e-12))[0]
        if pos_idx.size == 0 or neg_idx.size == 0:
            # fallback random
            i, j = np.random.randint(0,n,2)
            if i == j:
                continue
        else:
            i = np.random.choice(pos_idx)
            j = np.random.choice(neg_idx)
        # choose epsilon
        eps_max = min(h[i], 1.0 - h[j])
        if eps_max <= 0:
            continue
        # epsilon scaled by random factor maybe 0.5 of max
        eps = eps_max * np.random.rand() * 0.5 + 1e-8
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c5 = compute_c5(new_h)
        delta = new_c5 - cur_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            h = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        # update temperature
        iteration += 1
        if iteration % 1000 == 0:
            temp *= decay
            # ensure temperature not too low
            if temp < 1e-6:
                temp = 1e-6
    return best_h, best_c5

import time
import numpy as np
import scipy.optimize as opt
from scipy.special import logsumexp

def compute_c5(h):
    """Exact C5 for discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def upsample_h(old_h, new_n):
    """Linear interpolation preserving the integral."""
    old_n = old_h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    target = new_n / 2.0
    cur = new_h.sum()
    if cur > 0:
        new_h *= target / cur
    new_h = np.clip(new_h, 0.0, 1.0)
    cur = new_h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        new_h += diff / new_n
        new_h = np.clip(new_h, 0.0, 1.0)
    return new_h

def _smooth_obj_factory(n, alpha):
    """Factory returning a smoothed max objective for SLSQP."""
    def f(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return logsumexp(alpha * corr) / alpha * (2.0 / n)
    return f

def anneal(h_start, time_budget):
    """Guided simulated annealing on the exact C5."""
    start = time.time()
    deadline = start + time_budget
    h = h_start.copy()
    n = h.shape[0]
    cur_c5 = compute_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    temp = 0.015  # initial temperature
    it = 0
    while time.time() < deadline:
        # correlation to find lag of current max
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_max = int(np.argmax(corr))
        lag = idx_max - (n - 1)
        # gradient of the max-lag functional
        idx = np.arange(n)
        pos = idx + lag
        term1 = np.where((pos >= 0) & (pos < n), 1.0 - h[pos], 1.0)
        neg = idx - lag

        grad = term1 - term2

        pos_idx = np.where((grad > 0) & (h > 1e-12))[0]
        neg_idx = np.where((grad < 0) & (h < 1.0 - 1e-12))[0]
        if pos_idx.size and neg_idx.size:

        new_c5 = compute_c5(new_h)
        delta = new_c5 - cur_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / max(temp, 1e-9)):

            if cur_c5 < best_c5:

        it += 1
        # schedule temperature
        if it % 1000 == 0:
