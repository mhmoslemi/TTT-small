def step_h(h, step, shift, w):
    pos = np.where((w > 1.0) & (h < 1.0))[0]
    neg = np.where((w < 1.0) & (h > 0.0))[0]
    if len(pos)==0 or len(neg)==0: return h
    i_up = pos[np.argmax(w[pos])]  # largest w (increase)
    i_down = neg[np.argmin(w[neg])] # smallest w (decrease)
    max_inc = min(step, 1.0-h[i_up])
    max_dec = min(step, h[i_down])
    delta = min(max_inc, max_dec)
    new_h = h.copy()
    new_h[i_up] += delta
    new_h[i_down] -= delta
    return new_h

def run(seed=42, budget_s=1100):

    best_val = float('inf')
    best_n = None

import time, math, numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1‑h(x+k))dx using discrete correlation."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))


def _worst_shift_and_w(h: np.ndarray):
    """Find worst shift and the associated w_i = h_{i+shift}+h_{i-shift}."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)

    w = np.zeros_like(h)
    if shift == 0:
        w = 2.0 * h
    elif shift > 0:
        w[: n - shift] = h[shift:]
        w[shift:] += h[: n - shift]
    else:
        s = -shift
        w[s:] = h[: n - s]
        w[: n - s] += h[s:]
    return shift, w


def _projection_move(h, shift, w):
    """Project onto the hyperplane sum(h)=n/2 by ranking w."""
    n = len(h)
    target = n / 2.0
    idx = np.argsort(-w)                     # descending
    full = int(math.floor(target))
    frac = target - full
    new_h = np.zeros_like(h)
    if full:
        new_h[idx[:full]] = 1.0
    if frac > 1e-12:
        new_h[idx[full]] = frac
    return new_h


def _local_search(h_start: np.ndarray, time_limit: float):
    """Hill‑climbing / simulated‑annealing on a single start vector."""
    n = len(h_start)
    target_sum = n / 2.0
    cur_h = h_start.copy()
    cur_val = _compute_c5(cur_h)

    best_h = cur_h.copy()
    best_val = cur_val

    step = 0.2
    no_improve = 0
    start = time.time()

    while time.time() - start < time_limit:
        shift, w = _worst_shift_and_w(cur_h)
        r = np.random.rand()

        if r < 0.4:          # greedy pair based on w
            up_mask = (cur_h < 1.0 - 1e-12) & (w > 1.0 + 1e-12)
            down_mask = (cur_h > 1e-12) & (w < 1.0 - 1e-12)
            if up_mask.any() and down_mask.any():
                up_idx = np.where(up_mask)[0]
                i_up = up_idx[np.argmax(w[up_idx])]
                down_idx = np.where(down_mask)[0]
                i_down = down_idx[np.argmin(w[down_idx])]
                delta = min(step, 1.0 - cur_h[i_up], cur_h[i_down])
                if delta <= 0:
                    continue
                new_h = cur_h.copy()
                new_h[i_up] += delta
                new_h[i_down] -= delta
            else:           # fallback random pair
                up_idx = np.where(cur_h < 1.0 - 1e-12)[0]
                down_idx = np.where(cur_h > 1e-12)[0]
                if up_idx.size == 0 or down_idx.size == 0:
                    break
                i_up = np.random.choice(up_idx)
                i_down = np.random.choice(down_idx)
                max_inc = min(step, 1.0 - cur_h[i_up], cur_h[i_down])
                max_dec = min(step, cur_h[i_up], 1.0 - cur_h[i_down])
                if max_inc >= max_dec:
                    delta = np.random.rand() * max_inc
                else:
                    delta = np.random.rand() * max_dec
                    i_up, i_down = i_down, i_up
                new_h = cur_h.copy()
                new_h[i_up] += delta
                new_h[i_down] -= delta

        elif r < 0.85:       # pure random exchange
            up_idx = np.where(cur_h < 1.0 - 1e-12)[0]
            down_idx = np.where(cur_h > 1e-12)[0]
            if up_idx.size == 0 or down_idx.size == 0:
                break
            i_up = np.random.choice(up_idx)
            i_down = np.random.choice(down_idx)
            max_inc = min(step, 1.0 - cur_h[i_up], cur_h[i_down])
            max_dec = min(step, cur_h[i_up], 1.0 - cur_h[i_down])
            if np.random.rand() < 0.5 and max_inc > 0:
                delta = np.random.rand() * max_inc
            else:
                delta = np.random.rand() * max_dec
                i_up, i_down = i_down, i_up
            new_h = cur_h.copy()
            new_h[i_up] += delta
            new_h[i_down] -= delta

        else:                # projection on the worst shift
            new_h = _projection_move(cur_h, shift, w)

        new_h = np.clip(new_h, 0.0, 1.0)
        new_val = _compute_c5(new_h)

        if new_val < cur_val - 1e-12:
            cur_h, cur_val = new_h, new_val
            no_improve = 0
            step = min(step * 1.05, 0.5)
            if new_val < best_val - 1e-12:
                best_h, best_val = new_h.copy(), new_val
        else:
            T = step * 0.1
            if T > 0 and np.random.rand() < math.exp(-(new_val - cur_val) / T):
                cur_h, cur_val = new_h, new_val
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 500:
                    step *= 0.5
                    no_improve = 0
                    if step < 1e-4:
                        break

    # final projection onto exact mass
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        free = np.where((best_h > 1e-12) & (best_h < 1.0 - 1e-12))[0]
        if free.size:
            best_h[free] += diff / free.size
        else:
            best_h[0] = min(1.0, max(0.0, best_h[0] + diff))
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, _compute_c5(best_h)


def _slsqp_optimize(h0: np.ndarray, time_limit: float):
    """Optional refinement using SciPy's SLSQP."""
    try:
        from scipy.optimize import minimize
    except Exception:
        return h0, _compute_c5(h0)

    n = h0.shape[0]
    target = n / 2.0
    t0 = _compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    eq = {'type': 'eq', 'fun': lambda x: np.sum(x[:n]) - target}
    cons = [eq]

    shifts = list(range(-(n - 1), n))

    def make_ineq(s):
        def fun(x):
            h = x[:n]
            t = x[-1]
            if s >= 0:
                val = np.dot(h[: n - s], 1.0 - h[s:])
            else:
                s2 = -s
                val = np.dot(h[s2:], 1.0 - h[: n - s2])
            return t - val
        return fun

    cons.extend({'type': 'ineq', 'fun': make_ineq(s)} for s in shifts)

    max_iter = max(10, int(min(500, time_limit * 20)))
    opts = {'maxiter': max_iter, 'ftol': 1e-12, 'disp': False}
    res = minimize(lambda x: x[-1], x0, method='SLSQP',
                   bounds=bounds, constraints=cons, options=opts)

    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    diff = target - h_opt.sum()
    if abs(diff) > 1e-10:
        free = np.where((h_opt > 1e-12) & (h_opt < 1.0 - 1e-12))[0]
        if free.size:
            h_opt[free] += diff / free.size
            h_opt = np.clip(h_opt, 0.0, 1.0)

    return h_opt, _compute_c5(h_opt)
