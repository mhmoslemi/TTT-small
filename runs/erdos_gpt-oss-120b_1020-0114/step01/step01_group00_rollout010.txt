def greedy_refine(h, iterations=5000):
    n = len(h)
    best = h.copy()
    best_val = compute_c5_fft(best)
    rng = np.random.default_rng()
    for _ in range(iterations):
        i, j = rng.integers(0, n, size=2)
        if i == j: continue
        # Determine direction: if transferring from i to j (i->j) and allowed
        if best[i] > 0 and best[j] < 1:
            max_delta = min(best[i], 1 - best[j])
        elif best[j] > 0 and best[i] < 1:
            max_delta = min(best[j], 1 - best[i])
            # swap to maintain direction i->j
            i, j = j, i
        else:
            continue
        delta = max_delta * rng.random()
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        val = compute_c5_fft(cand)
        if val < best_val - 1e-12:
            best = cand
            best_val = val
    return best, best_val

def random_feasible(n, rng):
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    # Clip within [0,1] via projection iterative
    for _ in range(10):
        over = v > 1.0
        under = v < 0.0
        if not over.any() and not under.any():
            break
        excess = (v[over] - 1.0).sum()
        deficit = (-v[under]).sum()
        if excess > 0:
            interior = (~over) & (~under)
            if interior.any():
                v[interior] -= excess / interior.sum()
        if deficit > 0:
            interior = (~over) & (~under)
            if interior.any():
                v[interior] += deficit / interior.sum()
        v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - n/2.0) > 1e-9:
        diff = n/2.0 - cur
        mask = (v > 0.0) & (v < 1.0)
        if mask.any():
            v[mask] += diff / mask.sum()
            v = np.clip(v, 0.0, 1.0)
        else:
            v *= (n/2.0) / cur
    return v

def compute_corr_array(h):
    n = h.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1 - h, n=L)

    return corr

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np

    start = time.time()
    rng = np.random.default_rng(seed)

    # Determine number of points
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n0 = h0.size
        factor = 2  # upsample factor to 360 if n0=180
        n = n0 * factor
        h_start = np.repeat(h0, factor)
        # Ensure sum exactly n/2
        target_sum = n / 2.0
        cur_sum = h_start.sum()
        if abs(cur_sum - target_sum) > 1e-12:
            h_start = h_start * (target_sum / cur_sum)
            h_start = np.clip(h_start, 0.0, 1.0)
    else:
        n = 360
        h_start = random_feasible(n, rng)

    # Precompute FFT length
    L = 1 << ((2*n - 1).bit_length())
    # compute initial best
    best_h = h_start.copy()
    best_c5 = compute_c5_fft(best_h, L, n)

    # ------------------- Simulated Annealing -------------------
    total_time = budget_s * 0.80  # 80% for SA
    T0 = 0.015
    Tmin = 1e-6
    cooling = 0.9995
    T = T0
    cur_h = best_h.copy()
    cur_c5 = best_c5
    iter_cnt = 0
    while time.time() - start < total_time:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # Determine feasible direction (i -> j)
        if cur_h[i] > 0 and cur_h[j] < 1:
            max_delta = min(cur_h[i], 1 - cur_h[j])
        elif cur_h[j] > 0 and cur_h[i] < 1:
            max_delta = min(cur_h[j], 1 - cur_h[i])
            i, j = j, i
        else:
            continue
        if max_delta <= 0:
            continue
        step_factor = min(1.0, T / T0)  # shrink as T decreases
        delta = max_delta * rng.random() * step_factor
        if delta <= 0:
            continue
        cand = cur_h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_val = compute_c5_fft(cand, L, n)
        if cand_val < cur_c5 - 1e-12:
            cur_h = cand
            cur_c5 = cand_val
            if cand_val < best_c5:
                best_h = cand.copy()
                best_c5 = cand_val
        else:
            # probabilistic acceptance
            prob = np.exp(-(cand_val - cur_c5) / max(T, 1e-12))
            if rng.random() < prob:
                cur_h = cand
                cur_c5 = cand_val
        T = max(T * cooling, Tmin)
        iter_cnt += 1
        # occasional local greedy for the best
        if iter_cnt % 2000 == 0 and time.time() - start < total_time * 0.9:
            bh, bc = greedy_refine(best_h, iterations=500, rng=rng, L=L, n=n)
            if bc < best_c5:
                best_h, best_c5 = bh, bc
                cur_h, cur_c5 = bh, bc

    # ------------------- Targeted refinement -------------------
    # Use remaining time
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        # set iteration count roughly based on remaining
        iters = int(min(50000, remaining * 5000))  # approx 5000 iters per sec
        best_h, best_c5 = targeted_refine(best_h, iters, rng, L, n)

    # ------------------- Final greedy -------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 1.0:
        # final refine
        bh, bc = greedy_refine(best_h, iterations=2000, rng=rng, L=L, n=n)
        if bc < best_c5:
            best_h, best_c5 = bh, bc

    # final correction to enforce exact constraints
    best_h = np.clip(best_h, 0.0, 1.0)
    target_sum = n / 2.0
    cur = best_h.sum()
    if abs(cur - target_sum) > 1e-9:
        diff = target_sum - cur
        mask = (best_h > 0.0) & (best_h < 1.0)
        if mask.any():
            best_h[mask] += diff / mask.sum()
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            best_h *= target_sum / cur

    final_c5 = compute_c5_fft(best_h, L, n)
    return (best_h.tolist(), float(final_c5), int(n))

def targeted_refine(h, iterations, rng, L, n):
    best = h.copy()
    best_val = compute_c5_fft(best, L, n)
    for _ in range(iterations):
        # compute correlation array
        corr = corr_array_fft(best, L, n)   # length 2n-1
        max_idx = int(np.argmax(corr))
        s = max_idx - (n - 1)
        # Find candidate i positions where contribution positive
        # contributions = h_i * (1 - h_{i+s}) where valid.
        # Determine valid indices
        if s >= 0:
            valid = np.arange(0, n - s)
            contrib = best[valid] * (1 - best[valid + s])
        else:
            t = -s
            valid = np.arange(t, n)
            contrib = best[valid] * (1 - best[valid - t])
        # Choose i with probability proportional to contribution
        total_contrib = contrib.sum()
        if total_contrib <= 0:
            # random move fallback
            i = rng.integers(0, n)
        else:
            probs = contrib / total_contrib
            i = rng.choice(valid, p=probs)
        # Move mass from i to another index j
        if best[i] <= 0:
            continue
        # Choose j uniformly among indices with value < 1
        candidates_j = np.where(best < 1.0)[0]
        # exclude i
        candidates_j = candidates_j[candidates_j != i]
        if candidates_j.size == 0:
            continue
        j = rng.choice(candidates_j)
        max_delta = min(best[i], 1 - best[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.5
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        val = compute_c5_fft(cand, L, n)
        if val < best_val:
            best = cand
            best_val = val
    return best, best_val

import time
import numpy as np

def random_feasible(n, rng):
    """Return a random vector in [0,1]^n with sum = n/2."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    for _ in range(10):
        over = v > 1.0
        under = v < 0.0
        if not over.any() and not under.any():
            break
        excess = (v[over] - 1.0).sum()
        deficit = (-v[under]).sum()
        if excess > 0:
            interior = (~over) & (~under)
            if interior.any():
                v[interior] -= excess / interior.sum()
        if deficit > 0:
            interior = (~over) & (~under)
            if interior.any():
                v[interior] += deficit / interior.sum()
        v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - n / 2.0) > 1e-9:
        diff = n / 2.0 - cur
        mask = (v > 0.0) & (v < 1.0)
        if mask.any():
            v[mask] += diff / mask.sum()
            v = np.clip(v, 0.0, 1.0)
        else:
            v *= (n / 2.0) / cur
    return v

def compute_c5_fft(h, L=None, n=None):
    """Fast C5 using FFT‑based linear correlation."""
    if n is None:
        n = h.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    b_rev = (1.0 - h)[::-1]
    A = np.fft.rfft(h, n=L)
    B = np.fft.rfft(b_rev, n=L)
    conv = np.fft.irfft(A * B, n=L)[: 2 * n - 1]
    corr = conv[::-1] * (2.0 / n)
    return float(corr.max())

def corr_array_fft(h, L, n):
    """Full correlation array (scaled by dx)."""
    b_rev = (1.0 - h)[::-1]
    A = np.fft.rfft(h, n=L)
    B = np.fft.rfft(b_rev, n=L)
    conv = np.fft.irfft(A * B, n=L)[: 2 * n - 1]
    return conv[::-1] * (2.0 / n)

def greedy_refine(h, iterations, rng, L, n):
    """Small random pairwise moves that strictly improve C5."""
    best = h.copy()
    best_val = compute_c5_fft(best, L, n)
    for _ in range(iterations):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if best[i] > 0.0 and best[j] < 1.0:
            max_delta = min(best[i], 1.0 - best[j])
        elif best[j] > 0.0 and best[i] < 1.0:
            max_delta = min(best[j], 1.0 - best[i])
            i, j = j, i
        else:
            continue
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random() * 0.3
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        val = compute_c5_fft(cand, L, n)
        if val < best_val - 1e-12:
            best = cand
            best_val = val
    return best, best_val

def targeted_refine(h, iterations, rng, L, n):
    """Focus moves on indices contributing to the current worst shift."""
    best = h.copy()
    best_val = compute_c5_fft(best, L, n)
    for _ in range(iterations):
        corr = corr_array_fft(best, L, n)
        max_idx = int(np.argmax(corr))
        s = max_idx - (n - 1)          # shift giving the maximum
        if s >= 0:
            valid = np.arange(0, n - s)
            contrib = best[valid] * (1.0 - best[valid + s])
        else:
            t = -s
            valid = np.arange(t, n)
            contrib = best[valid] * (1.0 - best[valid - t])
        total = contrib.sum()
        if total <= 0.0:
            i = rng.integers(0, n)
        else:
            probs = contrib / total
            i = rng.choice(valid, p=probs)
        if best[i] <= 0.0:
            continue
        cand_j = np.where(best < 1.0)[0]
        cand_j = cand_j[cand_j != i]
        if cand_j.size == 0:
            continue
        j = rng.choice(cand_j)
        max_delta = min(best[i], 1.0 - best[j])
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random() * 0.5
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        val = compute_c5_fft(cand, L, n)
        if val < best_val:
            best = cand
            best_val = val
    return best, best_val

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound, and the number of points."""
    start = time.time()
    rng = np.random.default_rng(seed)

    # ----- initialise -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n0 = h0.size
        factor = 2
        n = n0 * factor
        h_start = np.repeat(h0, factor)
        target_sum = n / 2.0
        cur_sum = h_start.sum()
        if abs(cur_sum - target_sum) > 1e-12:
            h_start = h_start * (target_sum / cur_sum)
            h_start = np.clip(h_start, 0.0, 1.0)
    else:
        n = 360
        h_start = random_feasible(n, rng)

    L = 1 << ((2 * n - 1).bit_length())

    best_h = h_start.copy()
    best_c5 = compute_c5_fft(best_h, L, n)

    # ----- simulated annealing ----------------------------------------
    total_sa = budget_s * 0.80
    T0 = 0.015
    Tmin = 1e-6
    cooling = 0.9995
    T = T0
    cur_h = best_h.copy()
    cur_c5 = best_c5
    iter_cnt = 0

    while time.time() - start < total_sa:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if cur_h[i] > 0.0 and cur_h[j] < 1.0:
            max_delta = min(cur_h[i], 1.0 - cur_h[j])
        elif cur_h[j] > 0.0 and cur_h[i] < 1.0:
            max_delta = min(cur_h[j], 1.0 - cur_h[i])
            i, j = j, i
        else:
            continue
        if max_delta <= 0.0:
            continue
        step_factor = min(1.0, T / T0)
        delta = max_delta * rng.random() * step_factor
        if delta <= 0.0:
            continue
        cand = cur_h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_val = compute_c5_fft(cand, L, n)
        if cand_val < cur_c5 - 1e-12:
            cur_h, cur_c5 = cand, cand_val
            if cand_val < best_c5:
                best_h, best_c5 = cand.copy(), cand_val
        else:
            prob = np.exp(-(cand_val - cur_c5) / max(T, 1e-12))
            if rng.random() < prob:
                cur_h, cur_c5 = cand, cand_val
        T = max(T * cooling, Tmin)
        iter_cnt += 1
        if iter_cnt % 2000 == 0 and time.time() - start < total_sa * 0.90:
            bh, bc = greedy_refine(best_h, iterations=500, rng=rng, L=L, n=n)
            if bc < best_c5:
                best_h, best_c5 = bh, bc
                cur_h, cur_c5 = bh, bc

    # ----- targeted refinement -----------------------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        iters = int(min(50000, remaining * 5000))
        best_h, best_c5 = targeted_refine(best_h, iters, rng, L, n)

    # ----- final greedy polishing --------------------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 1.0:
        bh, bc = greedy_refine(best_h, iterations=2000, rng=rng, L=L, n=n)
        if bc < best_c5:
            best_h, best_c5 = bh, bc

    # ----- enforce exact feasibility -----------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    target = n / 2.0
    cur = best_h.sum()
    if abs(cur - target) > 1e-9:
        diff = target - cur
        mask = (best_h > 0.0) & (best_h < 1.0)
        if mask.any():
            best_h[mask] += diff / mask.sum()
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            best_h *= target / cur

    final_c5 = compute_c5_fft(best_h, L, n)
    return (best_h.tolist(), float(final_c5), int(n))
