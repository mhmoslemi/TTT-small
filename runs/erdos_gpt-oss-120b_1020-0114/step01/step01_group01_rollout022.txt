def project_to_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == 0:
        # Uniform distribution needed
        # assign ones randomly to meet target
        # but target sum is >0; just assign target_sum mass uniformly.
        idx = np.arange(len(h))
        # Actually set each to target_sum/len...
        # but sum must equal target_sum: we can fill fixed pattern.
        # Let's just fill first target positions with 1.
        n_ones = int(round(target_sum))
        h[:n_ones] = 1.0
        # rest zeros
        return h
    scale = target_sum / cur
    h = np.clip(h * scale, 0.0, 1.0)
    # adjust if small residual
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # add diff equally to free variables (0< h <1)
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size > 0:
            delta = diff / free.size
            h[free] += delta
            np.clip(h, 0.0, 1.0, out=h)
        else:
            # fallback: adjust random index
            i = np.random.randint(len(h))
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

# ----------------------------------------------------------------------
# Global placeholders used by the SLSQP refinement (mirrors the previous code)
# ----------------------------------------------------------------------
_opt_n = None          # number of points
_opt_dx = None         # grid spacing 2 / n
_opt_target_sum = None # n / 2

def objective_fun(x):
    """Objective: minimise the auxiliary variable t (last entry)."""
    return x[-1]

def eq_constraint_fun(x):
    """Equality constraint: sum(h) = n/2."""
    return np.sum(x[:_opt_n]) - _opt_target_sum

def ineq_constraint_fun(x):
    """Inequality: t >= overlap_k for every shift."""
    h = x[:_opt_n]
    t = x[-1]
    corr = np.correlate(h, 1 - h, mode='full') * _opt_dx
    return t - corr  # should be >= 0

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def project_to_feasible(h, target_sum):
    """Clip to [0,1] and (approximately) enforce sum(h) = target_sum."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == 0.0:
        # put the required mass on a random subset
        n = h.shape[0]
        need = int(round(target_sum))
        idx = np.random.choice(n, need, replace=False)
        h[idx] = 1.0
        return h
    scale = target_sum / cur
    h = np.clip(h * scale, 0.0, 1.0)
    # small residual correction
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size > 0:
            delta = diff / free.size
            h[free] += delta
            np.clip(h, 0.0, 1.0, out=h)
        else:
            # adjust a random index if nothing free
            i = np.random.randint(h.shape[0])
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def compute_overlap(h):
    """Return the full overlap array (maximum occurs at O.max())."""
    dx = 2.0 / h.shape[0]
    return np.correlate(h, 1 - h, mode='full') * dx

def gradient_for_shift(h, s):
    """Gradient dO_s/dh for the (linear‑zero‑padded) shift s."""
    n = h.shape[0]
    g = np.zeros(n)
    if s >= 0:
        # first term: (1 - h_{i+s}) for i = 0..n-s-1
        g[: n - s] += (1.0 - h[s:])
        # second term: -h_i contributes to index i+s
        g[s:] -= h[: n - s]

        # first term: (1 - h_{i-t}) for i = t..n-1
        g[t:] += (1.0 - h[: n - t])
        # second term: -h_{i+t} contributes to index i
        g[: n - t] -= h[t:]
    return g

        cur_val = float(corr[idx_max])
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                break

        shift = idx_max - (n - 1)          # shift corresponding to max overlap
        # compute gradient for that shift
        g = gradient_for_shift(h, shift)

        # indices to move mass from (positive grad) to (negative grad)
        pos_idx = int(np.argmax(g))
        neg_idx = int(np.argmin(g))

        if g[pos_idx] <= 1e-12 or g[neg_idx] >= -1e-12:
            # cannot improve using this shift; break to avoid endless loop
            break

        # feasible transfer amount
        eps_max = min(h[pos_idx], 1.0 - h[neg_idx])
        # limit step size for stability
        eps = eps_max * 0.3 * rng.random()
        if eps < 1e-9:
            break

        # propose move
        new_h = h.copy()
        new_h[pos_idx] -= eps
        new_h[neg_idx] += eps

        # quick feasibility check (bounds already respected)
        new_val = float(np.max(compute_overlap(new_h)))
        # accept only if not worsening the current maximum
        if new_val <= cur_val + 1e-14:
            h = new_h
        # otherwise discard and continue

    return best_h, best_val

def slsqp_refine(start_h, rng):
    """A short SLSQP refinement (identical to the previous program)."""
    global _opt_n, _opt_dx, _opt_target_sum
    n = len(start_h)
    _opt_n = n
    _opt_dx = 2.0 / n
    _opt_target_sum = n / 2.0

    h0 = project_to_feasible(start_h, _opt_target_sum)

    init_corr = np.correlate(h0, 1 - h0, mode='full') * _opt_dx
    t0 = float(np.max(init_corr) + 1e-8)

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    eq_cons = {'type': 'eq', 'fun': eq_constraint_fun}
    m = 2 * n - 1
    ineq_cons = NonlinearConstraint(
        ineq_constraint_fun,
        lb=np.zeros(m),
        ub=np.full(m, np.inf),
    )
    x0 = np.concatenate([h0, [t0]])

    res = minimize(
        objective_fun,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=[eq_cons, ineq_cons],
        options={'maxiter': 400, 'ftol': 1e-9, 'disp': False},
    )

    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    h_opt = project_to_feasible(h_opt, _opt_target_sum)

    try:
        c5_val = float(evaluate_erdos_solution(h_opt))
    except Exception:
        c5_val = float(res.x[-1])

    return h_opt, c5_val

def upsample_vector(v, N):
    """Linear interpolation from the points of v (defined on [0,2))."""
    n0 = len(v)
    if n0 == N:
        return v.copy()
    x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
    x_new = np.linspace(0.0, 2.0, N, endpoint=False)
    return np.interp(x_new, x_old, v)

# ----------------------------------------------------------------------
# Main driver required by the competition harness
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine an initial vector (if supplied)
    # ------------------------------------------------------------------
    try:
        init_vec = np.array(initial_h_values, dtype=float)  # supplied by harness
        init_n = init_vec.shape[0]
    except Exception:
        init_vec = None
        init_n = None

    # ------------------------------------------------------------------
    # Grid sizes to explore (small → larger, not exceeding 400)
    # ------------------------------------------------------------------
    grid_sizes = [80, 100, 120, 150, 200, 250, 300, 350, 400]
    if init_n is not None and init_n not in grid_sizes:
        grid_sizes = [init_n] + grid_sizes

    best_h = None
    best_val = float('inf')
    best_n = None

    def maybe_update(h, val, n):
        nonlocal best_h, best_val, best_n
        if val < best_val - 1e-9:
            best_h = h.copy()
            best_val = float(val)
            best_n = n

    # ------------------------------------------------------------------
    # Main exploration loop
    # ------------------------------------------------------------------
    for n in grid_sizes:
        # leave a safe margin for the final answer
        if time.time() - start_time > budget_s * 0.95:
            break

        target_sum = n / 2.0

        # Build a small pool of starting vectors for this size
        starts = []

        # 1) Interpolation of the supplied initial vector (if any)
        if init_vec is not None:
            starts.append(upsample_vector(init_vec, n))

        # 2) Interpolation of the current best solution (if it has a different size)
        if best_h is not None and best_n != n:
            starts.append(upsample_vector(best_h, n))

        # 3) Few random feasible vectors
        for _ in range(4):
            rand = rng.random(n)
            starts.append(project_to_feasible(rand, target_sum))

        # ------------------------------------------------------------------
        # Process each start
        # ------------------------------------------------------------------
        for start_h in starts:
            # Guarantee feasibility before any optimisation
            start_h = project_to_feasible(start_h, target_sum)

            # --------------------------------------------------------------
            # Phase 1 – cheap greedy local search
            # --------------------------------------------------------------
            time_left = budget_s - (time.time() - start_time)
            # allocate at most 15 % of the remaining budget to this start
            phase_time = min(0.15 * budget_s, max(0.0, time_left * 0.5))
            max_iters = int(3000 / (n / 100))   # roughly 3000 iterations at n=100
            max_iters = max(200, max_iters)

            h_loc, val_loc = local_search(start_h, rng, max_iters, phase_time)
            maybe_update(h_loc, val_loc, n)

            # --------------------------------------------------------------
            # Phase 2 – SLSQP refinement (if we still have time)
            # --------------------------------------------------------------
            if time.time() - start_time > budget_s * 0.80:
                continue
            h_ref, val_ref = slsqp_refine(h_loc, rng)
            maybe_update(h_ref, val_ref, n)

    # ------------------------------------------------------------------
    # Fallback in the extremely unlikely case nothing was found
    # ------------------------------------------------------------------
    if best_h is None:
        n = 100
        best_h = np.full(n, 0.5)
        best_val = float(evaluate_erdos_solution(best_h))
        best_n = n
    else:
        # Compute the final bound with the official evaluator for consistency
        best_val = float(evaluate_erdos_solution(best_h))

    return best_h, best_val, best_n
