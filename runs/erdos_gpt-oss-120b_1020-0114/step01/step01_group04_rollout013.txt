def random_feasible(n, M):
    h = np.zeros(n)
    idx = np.random.permutation(n)
    remaining = M
    for i in idx:
        if remaining >= 1.0:
            h[i] = 1.0
            remaining -= 1.0
        else:
            h[i] = remaining

def upsample_h(h_old, n_new):
    n_old = h_old.shape[0]
    # map positions over [0,2). Use np.interp
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    # clip and adjust sum
    target_sum = n_new/2.0
    cur_sum = h_new.sum()
    if cur_sum > 0:
        h_new = h_new * (target_sum / cur_sum)
    else:
        h_new[:] = 0.5
    h_new = np.clip(h_new, 0.0, 1.0)
    # adjust sum exactly
    diff = target_sum - h_new.sum()
    if abs(diff) > 1e-12:
        # distribute diff among interior points not at bounds
        free = np.where((h_new > 1e-12) & (h_new < 1.0 - 1e-12))[0]
        if free.size:
            h_new[free] += diff / free.size
        else:
            # adjust first point
            h_new[0] = min(1.0, max(0.0, h_new[0] + diff))
    h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def _w_vector(h, shift):
    n = h.shape[0]
    if shift == 0:
        return 2.0 * h
    w = np.zeros_like(h)
    if shift > 0:
        w[: n - shift] += h[shift:]
        w[shift:] += h[: n - shift]
    else:
        s = -shift
        w[s:] += h[: n - s]
        w[: n - s] += h[s:]
    return w

def _slsqp_optimize(h0, time_limit):
    try:
        from scipy.optimize import minimize
    except Exception:
        return h0, _compute_c5(h0)
    n = h0.shape[0]
    target_sum = n / 2.0
    # initial t bound as current c5
    t0 = _compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    # equality constraint
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x[:n]) - target_sum}]
    # inequality constraints for each shift
    # Want t >= overlap(s). We computed overlap(s) = (2/n) * sum_i h_i * (1 - h_{i+s}) = M*dx - (2/n)*sum_i h_i * h_{i+s}
    # But easier avoid scaling; use consistent: we want t >= (2/n) * np.dot(h[:n-s], 1 - h[s:]) for shift>=0.
    # We'll compute raw correlation value directly.
    # Using correlation unscaled to avoid constant factor: raw_corr = np.correlate(h, 1-h, mode='full')
    # For shift >=0, raw_corr[n-1 + s] = sum_i h_i * (1 - h_{i+s}) weighted by ones.
    # Then we enforce t >= raw_corr_max*dx . Instead we can enforce t >= raw_corr * dx.
    # Let's compute raw values each time.
    shifts = range(-(n-1), n)
    for s in shifts:
        def make_fun(s):
            def fun(x):
                h = x[:n]
                t = x[-1]
                # compute overlap raw
                # Clip for negative shift
                if s >= 0:
                    # sum over i=0..n-s-1
                    val = np.dot(h[: n - s], 1.0 - h[s:])
                else:
                    s2 = -s
                    val = np.dot(h[s2:], 1.0 - h[: n - s2])
                return t - val * (2.0 / n)
            return fun
        cons.append({'type': 'ineq', 'fun': make_fun(s)})
    max_iter = int(min(200, max(10, time_limit * 10)))
    opts = {'maxiter': max_iter, 'ftol': 1e-10, 'disp': False}
    res = minimize(lambda x: x[-1], x0, method='SLSQP', bounds=bounds, constraints=cons, options=opts)
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    # enforce sum
    diff = target_sum - h_opt.sum()
    if abs(diff) > 1e-12:
        free = np.where((h_opt > 1e-12) & (h_opt < 1.0 - 1e-12))[0]
        if free.size:
            h_opt[free] += diff / free.size
            h_opt = np.clip(h_opt, 0.0, 1.0)
    c5 = _compute_c5(h_opt)
    return h_opt, c5

import time, math, numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1-h(x+k)) dx for the discretised h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)

def _w_vector(h: np.ndarray, shift: int) -> np.ndarray:
    """Return w_i = h_{i+shift}+h_{i-shift} (zero outside the interval)."""
    n = h.shape[0]
    if shift == 0:
        return 2.0 * h
    w = np.zeros_like(h)
    if shift > 0:
        w[: n - shift] += h[shift:]
        w[shift:] += h[: n - shift]
    else:
        s = -shift
        w[s:] += h[: n - s]
        w[: n - s] += h[s:]
    return w

def _slsqp_optimize(h0: np.ndarray, time_limit: float):
    """Refine h via a short SLSQP run (if SciPy is available)."""
    try:
        from scipy.optimize import minimize
    except Exception:
        return h0, _compute_c5(h0)

    n = h0.shape[0]
    target_sum = n / 2.0
    t0 = _compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x[:n]) - target_sum}]

    # One inequality per shift (full mode of correlation)
    for s in range(-(n - 1), n):
        def make_fun(s):
            def fun(x):
                h = x[:n]
                t = x[-1]
                if s >= 0:
                    val = np.dot(h[: n - s], 1.0 - h[s:])
                else:
                    s2 = -s
                    val = np.dot(h[s2:], 1.0 - h[: n - s2])
                return t - val * (2.0 / n)
            return fun
        cons.append({'type': 'ineq', 'fun': make_fun(s)})

    max_iter = max(10, int(min(300, time_limit * 30)))
    opts = {'maxiter': max_iter, 'ftol': 1e-10, 'disp': False}
    res = minimize(lambda x: x[-1], x0, method='SLSQP',
                   bounds=bounds, constraints=cons, options=opts)

    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    diff = target_sum - h_opt.sum()
    if abs(diff) > 1e-12:
        free = np.where((h_opt > 1e-12) & (h_opt < 1.0 - 1e-12))[0]
        if free.size:
            h_opt[free] += diff / free.size
            h_opt = np.clip(h_opt, 0.0, 1.0)
    return h_opt, _compute_c5(h_opt)

def random_feasible(n: int, target_sum: float) -> np.ndarray:
    """Generate a random vector in [0,1]^n with exact sum target_sum (≤ n)."""
    h = np.zeros(n, dtype=float)
    idx = np.random.permutation(n)
    remain = target_sum
    for i in idx:
        if remain >= 1.0:
            h[i] = 1.0
            remain -= 1.0
        else:
            h[i] = remain
            remain = 0.0
            break
    return h

def upsample_h(old_h: np.ndarray, n_new: int) -> np.ndarray:
    """Linear interpolation of an existing h onto a finer grid (maintaining mass)."""
    n_old = old_h.shape[0]
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    target = n_new / 2.0
    cur = new_h.sum()
    if cur > 0:
        new_h *= target / cur
    new_h = np.clip(new_h, 0.0, 1.0)
    diff = target - new_h.sum()
    if abs(diff) > 1e-12:
        free = np.where((new_h > 1e-12) & (new_h < 1.0 - 1e-12))[0]
        if free.size:
            new_h[free] += diff / free.size
        else:
            new_h[0] = min(1.0, max(0.0, new_h[0] + diff))
        new_h = np.clip(new_h, 0.0, 1.0)
    return new_h

def tent_shape(n: int) -> np.ndarray:
    """Symmetric triangular (tent) shape with total integral 1."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + (1.0 / n) * 0.5
    h = 1.0 - np.abs(x - 1.0)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    cur = h.sum()
    if cur > 0:
        h *= target / cur
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size:
            h[free] += diff / free.size
        else:
            h[0] = min(1.0, max(0.0, h[0] + diff))
        h = np.clip(h, 0.0, 1.0)
    return h

def local_search(h_start: np.ndarray, time_limit: float,
                 step0: float = 0.2, top_k: int = 5) -> (np.ndarray, float):
    """Greedy / annealing style improvement of h."""
    n = h_start.shape[0]
    target_sum = n / 2.0
    cur_h = h_start.copy()
    cur_val = _compute_c5(cur_h)
    best_h, best_val = cur_h.copy(), cur_val

    step = step0
    no_improve = 0
    start = time.time()

    while time.time() - start < time_limit:
        # correlation and selection of several worst shifts
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        # indices of the largest (most overlapping) entries
        k = min(top_k, corr.size)
        top_idx = np.argpartition(-corr, k - 1)[:k]
        # convert to shift values, discard the zero shift if present
        shifts = [int(idx - (n - 1)) for idx in top_idx if idx != n - 1]
        if not shifts:
            shifts = [int(np.argmax(corr) - (n - 1))]

        # aggregated w‑vector over the selected shifts
        w = np.zeros_like(cur_h)
        for s in shifts:
            w += _w_vector(cur_h, s)

        up_mask = cur_h < 1.0 - 1e-12
        down_mask = cur_h > 1e-12
        if not up_mask.any() or not down_mask.any():
            break
        up_candidates = np.where(up_mask)[0]
        down_candidates = np.where(down_mask)[0]

        # with small probability make a random move to escape plateaus
        if np.random.rand() < 0.15:
            i_up = np.random.choice(up_candidates)
            i_down = np.random.choice(down_candidates)
        else:
            i_up = up_candidates[np.argmax(w[up_candidates])]
            i_down = down_candidates[np.argmin(w[down_candidates])]

        max_inc = min(step, 1.0 - cur_h[i_up], cur_h[i_down])
        if max_inc <= 0:
            break
        delta = max_inc

        new_h = cur_h.copy()
        new_h[i_up] += delta
        new_h[i_down] -= delta
        new_val = _compute_c5(new_h)

        if new_val < cur_val - 1e-12:
            cur_h, cur_val = new_h, new_val
            no_improve = 0
            step = min(step * 1.08, step0)
            if cur_val < best_val:
                best_val, best_h = cur_val, cur_h.copy()
        else:
            # simulated‑annealing acceptance
            T = step * 0.2
            if T > 0 and np.random.rand() < math.exp(-(new_val - cur_val) / T):
                cur_h, cur_val = new_h, new_val
            no_improve += 1
            if no_improve > 300:
                step = max(step * 0.5, 1e-5)
                no_improve = 0
                if step < 1e-6:
                    break

    # final tiny projection onto the hyperplane sum = target_sum
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        free = np.where((best_h > 1e-12) & (best_h < 1.0 - 1e-12))[0]
        if free.size:
            best_h[free] += diff / free.size
        else:
            best_h[0] = min(1.0, max(0.0, best_h[0] + diff))
        best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, _compute_c5(best_h)

    """

    # ------------------------------------------------------------------
    # Phase 1 – deterministic starts
    # ------------------------------------------------------------------
    best_h = None
    best_c5 = float('inf')
    time_spent = time.time() - start_all

    # ------------------------------------------------------------------

            if c5_ref < best_c5:
                best_c5, best_h = c5_ref, h_ref.copy()
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Final safety projection and return
    # ------------------------------------------------------------------
    if best_h is None:
        # fall back to a simple feasible vector
        best_h = np.full(n, 0.5)
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:

        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = _compute_c5(best_h)
    return best_h, float(final_c5), int(n)
