def _anneal_n(n, time_budget):
    # approximate time = time_budget
    dx = 2.0 / n
    target_sum = n // 2
    # initial vector: random binary with exactly target_sum ones
    # can pick random permutation of n indices, set first target_sum to 1.
    perm = np.random.permutation(n)
    h = np.zeros(n, dtype=float)
    h[perm[:target_sum]] = 1.0
    # create lists
    ones = set(np.flatnonzero(h).tolist())
    zeros = set(np.flatnonzero(1 - h).tolist())
    # evaluate
    cur_score = evaluate(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    # parameters
    while time.time() - start < time_budget * 0.95:
        # temperature schedule: linear decay
        elapsed = time.time() - start
        frac = elapsed / (time_budget * 0.95)
        T = max(1e-6, 0.05 * (1 - frac)) # start 0.05

        # propose swap
        i1 = np.random.choice(list(ones))
        i0 = np.random.choice(list(zeros))
        # swap
        h[i1] = 0.0
        h[i0] = 1.0
        # update sets
        ones.remove(i1)
        ones.add(i0)
        zeros.remove(i0)
        zeros.add(i1)

        new_score = evaluate(h)
        # decide acceptance
        if new_score < cur_score or np.random.rand() < np.exp(-(new_score - cur_score) / T):
            cur_score = new_score
            # accept: keep h and lists
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[i1] = 1.0
            h[i0] = 0.0
            ones.remove(i0)
            ones.add(i1)
            zeros.remove(i1)
            zeros.add(i0)
        # optional random restart if stagnation
        # but not needed maybe
    return best_h, best_score

def _softmax_grad_pre(h, T, dx, idx_matrix):
    n = h.size
    g = np.correlate(h, 1 - h, mode='full') * dx
    gmax = g.max()
    p = np.exp((g - gmax) / T)
    p_sum = p.sum()
    if p_sum == 0:
        p = np.full_like(p, 1.0 / p.size)
    else:
        p /= p_sum
    P = p[idx_matrix]   # shape (n,n)
    grad = dx * (P @ (1 - h) - P.T @ h)
    return grad

import time
import numpy as np

def _evaluate(h):
    """Return C5 for a discrete h."""
    n = h.size
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full") * dx)

def _project(v, target):
    """Euclidean projection onto 0≤x≤1 and sum(x)=target."""
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def _softmax_grad(h, T, dx, idx):
    """Gradient of the soft‑max approximation of max_k C_k."""
    n = h.size
    g = np.correlate(h, 1.0 - h, mode="full") * dx
    gmax = g.max()
    p = np.exp((g - gmax) / T)
    p /= p.sum()
    P = p[idx]                     # shape (n, n)
    grad = dx * (P @ (1.0 - h) - P.T @ h)
    return grad

def _binary_hill_climb(h, best_score, deadline):
    """Simple swap‑based local search on a binary vector."""
    n = h.size
    target = int(np.round(h.sum()))
    ones = list(np.flatnonzero(h))
    zeros = list(np.flatnonzero(1.0 - h))
    cur_score = best_score
    while time.time() < deadline:
        i1 = np.random.choice(ones)
        i0 = np.random.choice(zeros)
        # swap
        h[i1] = 0.0
        h[i0] = 1.0
        ones.remove(i1)
        ones.append(i0)
        zeros.remove(i0)
        zeros.append(i1)

        new_score = _evaluate(h)
        if new_score < cur_score:
            cur_score = new_score
        else:
            # revert
            h[i1] = 1.0
            h[i0] = 0.0
            ones.remove(i0)
            ones.append(i1)
            zeros.remove(i1)
            zeros.append(i0)
    return h, cur_score

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    # Use a relatively fine discretisation – gives more freedom to lower C5
    n = 200                     # even, keeps sum integer
    dx = 2.0 / n
    target_sum = n // 2

    # ---- initialise a feasible (continuous) vector -----------------
    h = np.random.rand(n)
    h = _project(h, target_sum)

    # pre‑compute index matrix for the soft‑max gradient
    idx = (np.arange(n)[:, None] - np.arange(n)[None, :]) + (n - 1)

    # ---- Adam optimiser parameters ---------------------------------
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.15
    beta1_pow, beta2_pow = 1.0, 1.0

    # soft‑max temperature schedule
    T0, Tmin = 0.10, 5e-4

    best_h = h.copy()
    best_score = _evaluate(h)

    max_iters = 300000
    it = 0
    # allocate ~90 % of the budget to the continuous optimiser
    cont_deadline = start + budget_s * 0.9

    while time.time() < cont_deadline and it < max_iters:
        it += 1
        # temperature (linear decay)
        T = max(Tmin, T0 * (1.0 - it / max_iters))

        # gradient
        grad = _softmax_grad(h, T, dx, idx)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        # cap step size to keep moves modest
        max_step = 0.2
        step_norm = np.max(np.abs(step))
        if step_norm > max_step:
            step = step * (max_step / step_norm)

        h_new = _project(h - step, target_sum)
        cur_score = _evaluate(h_new)

        if cur_score < best_score:
            best_score = cur_score
            best_h = h_new.copy()
        # always move (Adam’s direction)
        h = h_new

        # occasional small random jitter to escape flat plateaus
        if it % 5000 == 0:
            jitter = np.random.normal(scale=0.015, size=n)
            h_j = _project(h + jitter, target_sum)
            score_j = _evaluate(h_j)
            if score_j < best_score:
                best_score = score_j
                best_h = h_j.copy()
            h = h_j

    # ---- convert to a binary step function and improve it ----------
    # rounding: take the largest target_sum entries as 1
    bin_h = np.zeros(n, dtype=float)
    bin_h[np.argsort(-best_h)[:target_sum]] = 1.0
    bin_score = _evaluate(bin_h)

    if bin_score < best_score:
        best_score = bin_score
        best_h = bin_h.copy()

    # local binary hill‑climb (use remaining budget)
    remaining_deadline = start + budget_s * 0.99
    bin_h, bin_score = _binary_hill_climb(bin_h, bin_score, remaining_deadline)
    if bin_score < best_score:
        best_score = bin_score
        best_h = bin_h.copy()

    return best_h, best_score, n
