def _gradient_shift(h, shift):
    N = h.size
    g = np.zeros_like(h)
    if shift >= 0:
        n = N - shift
        i = np.arange(n)
        ip = i + shift
        g[i] = 1.0 - h[ip]
        g[ip] -= h[i]
    else:
        t = -shift
        i = np.arange(t, N)
        im = i - t
        g[i] = 1.0 - h[im]
        g[im] -= h[i]
    return g

def _project(v, target):
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    while abs(diff) > 1e-12:
        if diff > 0:
            free = v < 1.0 - 1e-12
            nfree = free.sum()
            if nfree == 0:
                break
            add = diff / nfree
            v[free] = np.minimum(v[free] + add, 1.0)
        else:
            free = v > 1e-12
            nfree = free.sum()
            if nfree == 0:
                break
            sub = diff / nfree
            v[free] = np.maximum(v[free] + sub, 0.0)
        diff = target - v.sum()
    return v

import numpy as np
import time
import math

def _project(v, target):
    """Project onto the box [0,1] and fix the sum to target."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    # water‑filling loop
    while abs(diff) > 1e-12:
        if diff > 0:
            free = v < 1.0 - 1e-12
            nfree = free.sum()
            if nfree == 0:
                break
            add = diff / nfree
            v[free] = np.minimum(v[free] + add, 1.0)
        else:
            free = v > 1e-12
            nfree = free.sum()
            if nfree == 0:
                break
            sub = diff / nfree
            v[free] = np.maximum(v[free] + sub, 0.0)
        diff = target - v.sum()
    return v

def _c5_of(h):
    """Overlap constant for a step vector."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)

def _gradient_shift(h, shift):
    """Gradient of the overlap for a given shift (the shift attaining the max)."""
    N = h.size
    g = np.zeros_like(h)
    if shift >= 0:
        n = N - shift
        i = np.arange(n)
        ip = i + shift
        g[i] = 1.0 - h[ip]
        g[ip] -= h[i]
    else:
        t = -shift
        i = np.arange(t, N)
        im = i - t
        g[i] = 1.0 - h[im]
        g[im] -= h[i]
    return g

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function with a small C₅."""
    np.random.seed(seed)

    # ----- initialise ----------------------------------------------------
    if "initial_h_values" in globals():
        h = np.asarray(initial_h_values, dtype=float)
    else:
        N = 60
        h = np.random.rand(N)

    N = h.size
    target_sum = N / 2.0
    h = _project(h, target_sum)

    best_h = h.copy()
    best_val = _c5_of(best_h)

    cur_h = best_h.copy()
    cur_val = best_val

    start = time.time()
    max_step = 0.5
    min_step = 0.001
    max_temp = 0.02
    min_temp = 1e-6
    no_improve = 0

    # ----- main optimisation loop ----------------------------------------
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s * 0.95:
            break

        progress = elapsed / (budget_s * 0.95)
        step = max_step * (1 - progress) + min_step
        T = max_temp * (1 - progress) + min_temp

        # current overlap and worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        s_idx = int(np.argmax(corr))
        S_max = corr[s_idx]
        cur_val = S_max * (2.0 / N)

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_h = cur_h.copy()
            no_improve = 0
        else:
            no_improve += 1

        shift = s_idx - (N - 1)
        g = _gradient_shift(cur_h, shift)

        # try a directed mass transfer along the gradient
        pos = np.where(g > 1e-12)[0]
        neg = np.where(g < -1e-12)[0]
        moved = False
        if pos.size and neg.size:
            i = pos[np.argmax(g[pos])]
            j = neg[np.argmin(g[neg])]
            delta_max_i = cur_h[i]               # can decrease at most this
            delta_max_j = 1.0 - cur_h[j]         # can increase at most this
            delta = min(delta_max_i, delta_max_j, step)
            if delta > 1e-12:
                new_h = cur_h.copy()
                new_h[i] -= delta
                new_h[j] += delta
                # projection only to fix tiny rounding errors
                new_h = _project(new_h, target_sum)
                new_val = _c5_of(new_h)
                if new_val < cur_val or np.random.rand() < math.exp(-(new_val - cur_val) / T):
                    cur_h, cur_val = new_h, new_val
                    moved = True

        if not moved:
            # fallback random two‑point exchange (same style as the reference SA)
            i, j = np.random.randint(N, size=2)
            if i != j:
                lower = max(-cur_h[i], cur_h[j] - 1.0)
                upper = min(1.0 - cur_h[i], cur_h[j])
                if lower < upper:
                    delta_low = max(lower, -step)
                    delta_high = min(upper, step)
                    if delta_low < delta_high:
                        delta = np.random.uniform(delta_low, delta_high)
                        new_h = cur_h.copy()
                        new_h[i] += delta
                        new_h[j] -= delta
                        new_h = _project(new_h, target_sum)
                        new_val = _c5_of(new_h)
                        if new_val < cur_val or np.random.rand() < math.exp(-(new_val - cur_val) / T):
                            cur_h, cur_val = new_h, new_val

        # occasional restart if we are stuck
        if no_improve > 50000:
            cur_h = best_h.copy()
            for _ in range(5):
                idx = np.random.randint(N)
                cur_h[idx] = np.clip(cur_h[idx] + np.random.uniform(-step, step), 0.0, 1.0)
            cur_h = _project(cur_h, target_sum)
            cur_val = _c5_of(cur_h)
            no_improve = 0

    # ----- final polishing ------------------------------------------------
    best_h = _project(best_h, target_sum)
    final_val = _c5_of(best_h)

    if "evaluate_erdos_solution" in globals():
        try:
            final_val = evaluate_erdos_solution(best_h.tolist())
        except Exception:
            pass

    return (best_h.tolist(), float(final_val), int(N))
