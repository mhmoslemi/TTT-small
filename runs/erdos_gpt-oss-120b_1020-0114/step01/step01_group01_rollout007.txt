def project_to_simplex_bounds(v, target, eps=1e-12):
    # water-filling algorithm
    n = len(v)
    x = np.clip(v, 0, 1)
    # Compute current sum
    cur = x.sum()
    if abs(cur - target) < eps:
        return x
    # We'll use binary search on lambda
    lo = -1
    hi = 1
    for _ in range(50):
        lam = (lo + hi) / 2
        x = np.clip(v + lam, 0, 1)
        s = x.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    # After binary, may not achieve exactly due to rounding; adjust a bit
    # Ensure sum close to target
    # If discrepancy remains, we can adjust within bounds.
    residual = target - x.sum()
    if abs(residual) > 1e-9:
        # distribute residual across non-bound entries
        idx = np.where((x > 0) & (x < 1))[0]
        if idx.size > 0:
            delta = residual / idx.size
            x[idx] += delta
            np.clip(x, 0, 1, out=x)
    return x
```

def project_to_feasible(v, target):
    # waterfilling binary search
    x = np.clip(v, 0.0, 1.0)
    cur = x.sum()
    if abs(cur - target) < 1e-12:
        return x
    # binary search for lambda
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    # adjust any residual
    residual = target - x.sum()
    if abs(residual) > 1e-9:
        # distribute residual among non-bound entries
        free = np.where((x > 0) & (x < 1))[0]
        if free.size > 0:
            delta = residual / free.size
            x[free] += delta
            np.clip(x, 0.0, 1.0, out=x)
    return x
```

def greedy_improve(h, start_time, budget_s):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    best_h = h.copy()
    best_c5 = c5_from_h(best_h)
    # variable step size
    step = 0.05
    # time budget small fraction of global
    max_time = start_time + budget_s * 0.7  # let it run up to 70% of budget
    while time.time() < max_time:
        # compute correlation
        conv = np.correlate(h, h, mode='full')
        min_idx = np.argmin(conv)
        min_corr = conv[min_idx]
        cur_c5 = 1.0 - min_corr * dx
        # shift
        shift = min_idx - (n - 1)
        k = abs(shift)
        # gradient
        if k == 0:
            g = 2 * h
        else:
            left = np.concatenate([h[k:], np.zeros(k)])
            right = np.concatenate([np.zeros(k), h[:-k]])
            g = left + right
        # find add and subtract indices
        add_mask = h < 1 - 1e-12
        sub_mask = h > 1e-12
        if not np.any(add_mask) or not np.any(sub_mask):
            break
        # choose indices
        i_max = np.argmax(g * add_mask)  # but need to avoid masked zeros.
        # This approach picks index with max g*mask (g where mask false become zero).
        # For i_min, we want min g among sub_mask, but we can similarly.
        # However if some masked entries have g zero and we multiply: might incorrectly pick zero. Better:
        addable_idx = np.where(add_mask)[0]
        i_max = addable_idx[np.argmax(g[addable_idx])]
        removable_idx = np.where(sub_mask)[0]
        i_min = removable_idx[np.argmin(g[removable_idx])]
        # epsilon
        eps = min(step, 1 - h[i_max], h[i_min])
        if eps <= 1e-12:
            step *= 0.5
            if step < 1e-6:
                break
            continue
        # make move
        h[i_max] += eps
        h[i_min] -= eps
        # check sum still correct (should be)
        # compute new c5
        conv_new = np.correlate(h, h, mode='full')
        min_corr_new = np.min(conv_new)
        new_c5 = 1.0 - min_corr_new * dx
        if new_c5 < best_c5 - 1e-9:
            best_c5 = new_c5
            best_h = h.copy()
            # maybe increase step
            step = min(step * 1.2, 0.2)
        else:
            # revert move
            h[i_max] -= eps
            h[i_min] += eps
            step *= 0.5
            if step < 1e-7:
                break
    # after loop, ensure best_h is feasible
    best_h = project_to_feasible(best_h, target)
    best_c5 = c5_from_h(best_h)
    return best_h, best_c5

def optimise_start(h_start, start_time, budget_s):
    global _opt_n, _opt_dx, _opt_target_sum
    n = len(h_start)
    _opt_n = n
    _opt_dx = 2.0 / n
    _opt_target_sum = n / 2.0
    h0 = project_to_feasible(h_start, _opt_target_sum)
    # init t as current max overlap + small epsilon
    init_corr = np.correlate(h0, 1 - h0, mode='full') * _opt_dx
    t0 = float(np.max(init_corr) + 1e-8)
    bounds = [(0.0,1.0)]*n + [(0.0,1.0)]
    eq_cons = {'type':'eq', 'fun':eq_constraint_fun}
    m = 2*n - 1
    ineq_cons = NonlinearConstraint(ineq_constraint_fun,
                                    lb=np.zeros(m),
                                    ub=np.full(m, np.inf))
    x0 = np.concatenate([h0, [t0]])
    maxiter = 500
    # maybe adjust maxiter based on remaining time
    remain = budget_s - (time.time() - start_time)
    if remain < 5:
        maxiter = 200
    res = minimize(objective_fun, x0, method='SLSQP', bounds=bounds,
                   constraints=[eq_cons, ineq_cons],
                   options={'maxiter': maxiter, 'ftol':1e-9, 'disp':False})
    # extract
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    h_opt = project_to_feasible(h_opt, _opt_target_sum)
    try:
        c5_val = float(evaluate_erdos_solution(h_opt))
    except Exception:
        c5_val = float(res.x[-1])

import time
import numpy as np
from scipy.optimize import minimize, NonlinearConstraint

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def project_to_feasible(v, target, eps=1e-12):
    """Project onto the box [0,1] and the hyperplane sum(v)=target."""
    x = np.clip(v, 0.0, 1.0)
    cur = x.sum()
    if abs(cur - target) <= eps:
        return x
    # water‑filling via binary search on lambda
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    # tiny residual handling
    residual = target - x.sum()
    if abs(residual) > 1e-9:
        free = np.where((x > 0) & (x < 1))[0]
        if free.size:
            delta = residual / free.size
            x[free] += delta
            np.clip(x, 0.0, 1.0, out=x)
    return x


def c5_from_h(h):
    """Fast C5 from h using the identity C5 = 1 - dx * min_corr."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, h, mode='full')
    min_corr = conv.min()
    return 1.0 - min_corr * dx


# ----------------------------------------------------------------------
# Greedy local improvement focusing on the worst shift
# ----------------------------------------------------------------------
def greedy_improve(h, time_limit):
    start = time.time()
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_c5 = c5_from_h(best_h)
    step = 0.05

    while time.time() - start < time_limit:
        conv = np.correlate(h, h, mode='full')
        min_idx = np.argmin(conv)
        min_corr = conv[min_idx]
        cur_c5 = 1.0 - min_corr * (2.0 / n)

        shift = min_idx - (n - 1)
        k = abs(shift)

        if k == 0:
            g = 2.0 * h
        else:
            left = np.concatenate([h[k:], np.zeros(k)])
            right = np.concatenate([np.zeros(k), h[:-k]])
            g = left + right

        addable = np.where(h < 1.0 - 1e-12)[0]
        removable = np.where(h > 1e-12)[0]
        if addable.size == 0 or removable.size == 0:
            break

        i_max = addable[np.argmax(g[addable])]
        i_min = removable[np.argmin(g[removable])]

        eps = min(step, 1.0 - h[i_max], h[i_min])
        if eps <= 1e-12:
            step *= 0.5
            if step < 1e-7:
                break
            continue

        h[i_max] += eps
        h[i_min] -= eps

        conv_new = np.correlate(h, h, mode='full')
        new_c5 = 1.0 - conv_new.min() * (2.0 / n)

        if new_c5 + 1e-12 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            step = min(step * 1.2, 0.2)
        else:
            h[i_max] -= eps
            h[i_min] += eps
            step *= 0.5
            if step < 1e-7:
                break

    best_h = project_to_feasible(best_h, target)
    best_c5 = c5_from_h(best_h)
    return best_h, best_c5

    """Minimise the auxiliary variable t (the maximal overlap)."""
    return x[-1]

    """Enforce ∑h = n/2."""
    return np.sum(x[:_opt_n]) - _opt_target_sum

    h = x[:_opt_n]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode="full") * _opt_dx

        init_vec = np.array(initial_h_values, dtype=float)
        n0 = init_vec.shape[0]
    except Exception:
        n0 = 100
        init_vec = np.full(n0, 0.5)

    target0 = n0 / 2.0
    h0 = project_to_feasible(init_vec.copy(), target0)

    best_h = None
    best_c5 = float("inf")
    best_n = n0

    def consider(h, val, n):
        nonlocal best_h, best_c5, best_n
        if val < best_c5 - 1e-12:
            best_h = h.copy()
            best_c5 = float(val)
            best_n = n

    # ------------------------------------------------------------------
    # Initial greedy improvement (short budget)
    # ------------------------------------------------------------------
    remain = budget_s - (time.time() - start_time)
    if remain > 2.0:
        time_greedy = min(1.0, remain * 0.2)
        h_g, c5_g = greedy_improve(h0, time_greedy)
        consider(h_g, c5_g, n0)

    # ------------------------------------------------------------------
    # SLSQP refinement of the best known (if time permits)
    # ------------------------------------------------------------------
    if time.time() - start_time < budget_s * 0.9:
        h_opt, c5_opt = optimise_start(best_h if best_h is not None else h0,
                                      start_time, budget_s)
        consider(h_opt, c5_opt, n0)

    # ------------------------------------------------------------------
    # Random restarts with cheap greedy + optional SLSQP
    # ------------------------------------------------------------------
    for _ in range(4):
        if time.time() - start_time > budget_s * 0.95:
            break
        # random feasible start
        rand = rng.random(n0)
        rand = project_to_feasible(rand, target0)

        remain = budget_s - (time.time() - start_time)
        if remain < 0.5:
            break
        # greedy phase (a small budget)
        time_greedy = min(0.5, remain * 0.15)
        h_g, c5_g = greedy_improve(rand, time_greedy)
        consider(h_g, c5_g, n0)

        # optional SLSQP if still time
        if time.time() - start_time < budget_s * 0.9:
            h_opt, c5_opt = optimise_start(h_g, start_time, budget_s)
            consider(h_opt, c5_opt, n0)

    # ------------------------------------------------------------------
    # Attempt a finer grid based on the current best solution
    # ------------------------------------------------------------------
    if time.time() - start_time < budget_s * 0.6:
        n2 = int(n0 * 1.4)          # e.g. 140 when n0=100
        n2 = max(80, min(n2, 500))  # keep it reasonable
        target2 = n2 / 2.0
        if best_h is not None and best_n == n0:
            x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h, left=0.0, right=0.0)
        else:
            h_interp = np.full(n2, 0.5)
        h_interp = project_to_feasible(h_interp, target2)

        remain = budget_s - (time.time() - start_time)
        if remain > 2.0:
            time_greedy = min(1.0, remain * 0.2)
            h_g2, c5_g2 = greedy_improve(h_interp, time_greedy)
            consider(h_g2, c5_g2, n2)

            if time.time() - start_time < budget_s * 0.9:
                h_opt2, c5_opt2 = optimise_start(h_g2, start_time, budget_s)
                consider(h_opt2, c5_opt2, n2)

    # ------------------------------------------------------------------
    # Fallback safety
    # ------------------------------------------------------------------
    if best_h is None:
        best_n = n0
        best_h = np.full(best_n, 0.5)
        best_c5 = float(evaluate_erdos_solution(best_h))

    # final projection & accurate evaluation
    best_h = project_to_feasible(best_h, best_n / 2.0)
    try:
        best_c5 = float(evaluate_erdos_solution(best_h))
    except Exception:
        pass

    return best_h, best_c5, best_n
