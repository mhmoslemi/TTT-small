def _worst_lag_gradient(h, lag):
    n = h.size
    dx = 2.0 / n
    grad = np.zeros_like(h)
    if lag >= 0:

        grad[:n-lag] += (1 - h[lag:]) * dx

    else:
        shift = -lag
        grad[shift:] += (1 - h[:n-shift]) * dx

    return grad

import time, numpy as np

def _evaluate(h):
    n = h.size
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full') * dx)

def _project(v, target):
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def _gradient_worstlag(h, lag):
    n = h.size
    dx = 2.0 / n
    g = np.zeros_like(h)
    if lag >= 0:
        m = n - lag
        if m > 0:
            g[:m] += (1 - h[lag:]) * dx
            g[lag:] -= h[:m] * dx
    else:
        shift = -lag
        m = n - shift
        if m > 0:
            g[shift:] += (1 - h[:m]) * dx
            g[:m] -= h[shift:] * dx
    return g

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    np.random.seed(seed)
    start = time.time()
    # parameters
    n = 200  # number of discretization points (even)
    target_sum = n * 0.5
    dx = 2.0 / n

    # ---- Phase 1: continuous subgradient descent ----
    # initialise h randomly and project
    h = np.random.rand(n)
    h = _project(h, target_sum)
    cur_c5 = _evaluate(h)
    best_h = h.copy()
    best_c5 = cur_c5

    max_cont_iters = 5000
    lr = 0.5  # learning rate
    for it in range(max_cont_iters):
        # check time
        if time.time() - start > budget_s * 0.4:
            break
        # correlation and worst lag
        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_idx = int(np.argmax(corr))
        lag = max_idx - (n - 1)
        # gradient for worst lag
        grad = _gradient_worstlag(h, lag)
        # backtracking line search
        step = lr
        improved = False
        for _ in range(12):
            h_new = _project(h - step * grad, target_sum)
            c_new = _evaluate(h_new)
            if c_new < cur_c5:
                h = h_new
                cur_c5 = c_new
                improved = True
                # adapt learning rate
                lr = min(lr * 1.08, 2.0)
                break
            step *= 0.5
        if not improved:

        # occasional jitter to escape plateaus

            cur_c5 = _evaluate(h)

    # ---- Phase 2: binarize ----

import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def _evaluate(h: np.ndarray) -> float:
    """Return max_{k} ∫ h(x)*(1-h(x+k)) dx for discrete h."""
    n = h.size
    dx = 2.0 / n
    # correlation with zero padding (np.correlate implements this)
    return np.max(np.correlate(h, 1.0 - h, mode="full") * dx)


def _project(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto {0≤x≤1, sum x = target}."""
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _gradient_worstlag(h: np.ndarray, lag: int) -> np.ndarray:
    """Sub‑gradient of the lag giving the current maximum overlap."""
    n = h.size
    dx = 2.0 / n
    g = np.zeros_like(h)
    if lag >= 0:
        m = n - lag
        if m > 0:
            g[:m] += (1.0 - h[lag:]) * dx          # h_i * (1 - h_{i+lag})
            g[lag:] -= h[:m] * dx                  # -h_{i-lag} * (1 - h_i)
    else:
        shift = -lag
        m = n - shift
        if m > 0:
            g[shift:] += (1.0 - h[:m]) * dx        # h_i * (1 - h_{i-shift})
            g[:m] -= h[shift:] * dx                # -h_{i+shift} * (1 - h_i)
    return g


# ------------------------------------------------------------
# Main optimisation routine
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a step function h on [0,2] (as a vector), its C5 bound,
    and the number of discretisation points.
    """
    np.random.seed(seed)
    start = time.time()

    # --------------------------------------------------------
    # Choose discretisation size (even, to allow exact binary)
    # --------------------------------------------------------
    n = 200                     # modest size, gives dx = 0.01
    target_sum = n * 0.5        # ∫ h = 1  →  sum = n/2
    dx = 2.0 / n

    # --------------------------------------------------------
    # Phase 1 – continuous sub‑gradient descent on the worst lag
    # --------------------------------------------------------
    h = np.random.rand(n)
    h = _project(h, target_sum)
    cur_c5 = _evaluate(h)
    best_h = h.copy()
    best_c5 = cur_c5

    max_cont_iters = 5000
    lr = 0.5                    # base step size

    for it in range(max_cont_iters):
        if time.time() - start > budget_s * 0.35:
            break

        # identify the lag that attains the maximum overlap
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        max_idx = int(np.argmax(corr))
        lag = max_idx - (n - 1)

        # sub‑gradient for that lag
        grad = _gradient_worstlag(h, lag)

        # back‑tracking line search
        step = lr
        improved = False
        for _ in range(12):
            h_new = _project(h - step * grad, target_sum)
            c_new = _evaluate(h_new)
            if c_new < cur_c5:
                h = h_new
                cur_c5 = c_new
                improved = True
                lr = min(lr * 1.08, 2.0)
                break
            step *= 0.5
        if not improved:
            lr = max(lr * 0.5, 1e-4)

        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()

        # occasional jitter to shake out of plateaus
        if (it + 1) % 200 == 0:
            jitter = np.random.normal(scale=0.02, size=n)
            h = _project(h + jitter, target_sum)
            cur_c5 = _evaluate(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()

    # --------------------------------------------------------
    # Phase 2 – binarise the best continuous solution
    # --------------------------------------------------------
    order = np.argsort(-best_h)                     # descending
    binary_h = np.zeros(n, dtype=float)
    binary_h[order[:int(target_sum)]] = 1.0
    cur_c5 = _evaluate(binary_h)
    if cur_c5 < best_c5:
        best_c5 = cur_c5
        best_h = binary_h.copy()

    # --------------------------------------------------------
    # Phase 3 – binary simulated annealing / hill‑climbing
    # --------------------------------------------------------
    h_bin = binary_h.copy()
    cur_c5 = _evaluate(h_bin)

    # temperature schedule parameters
    T0 = 0.02
    max_sa_iters_est = 500_000   # used for schedule; actual loop respects time budget

    it = 0
    while True:
        if time.time() - start > budget_s * 0.97:
            break
        it += 1
        # linear cooling

        j = np.random.randint(n)
        if h_bin[i] == h_bin[j]:

        h_new = h_bin.copy()
        h_new[i], h_new[j] = h_bin[j], h_bin[i]
        c_new = _evaluate(h_new)

        delta = c_new - cur_c5
        if delta < 0.0 or np.random.rand() < np.exp(-delta / T):
            h_bin = h_new
            cur_c5 = c_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_bin.copy()

        # every few ten‑thousands of steps perform a larger random perturbation

            for idx in flips:

            # repair the sum to remain exactly n/2
            diff = int(target_sum - h_bin.sum())
            if diff > 0:

                    h_bin[sel] = 1.0
            elif diff < 0:

                if ones.size >= -diff:
                    sel = np.random.choice(ones, size=-diff, replace=False)
                    h_bin[sel] = 0.0
            cur_c5 = _evaluate(h_bin)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_bin.copy()

    # --------------------------------------------------------
    # Return the best solution found
    # --------------------------------------------------------
    return best_h, float(best_c5), n
