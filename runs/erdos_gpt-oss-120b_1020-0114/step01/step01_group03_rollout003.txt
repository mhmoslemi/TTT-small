def run(seed=42,budget_s=1100,**kwargs):
    np.random.seed(seed)
    start_time = time.time()
    # read initial
    if 'initial_h_values' in globals():
        h = np.asarray(initial_h_values, dtype=float)
    else:
        n = 200
        h = np.zeros(n, dtype=float)
        half = n//2

        np.random.shuffle(h)
    
    n = h.size
    target_sum = n/2.0
    # clip & scale
    h = np.clip(h,0.0,1.0)
    s = h.sum()

    h = np.clip(h,0.0,1.0)

import time
import numpy as np

# ------------------------------------------------------------
# Core helpers
# ------------------------------------------------------------
def compute_corr_array(h):
    """Full correlation array for h with (1-h) using FFT (no dx scaling)."""
    n = h.shape[0]
    size = 2 * n
    H = np.fft.rfft(h, n=size)
    G_rev = np.fft.rfft((1.0 - h)[::-1], n=size)
    conv = np.fft.irfft(H * G_rev, n=size)
    return conv[: 2 * n - 1]  # length 2n-1, lag = idx-(n-1)

def compute_c5(h):
    """Exact C5 value for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = compute_corr_array(h)
    return np.max(corr) * dx

def upsample_h(old_h, new_n):
    """Linear interpolation to a finer grid while preserving the integral."""
    old_n = old_h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    target = new_n / 2.0
    s = new_h.sum()
    if s > 0:
        new_h = new_h * (target / s)
    new_h = np.clip(new_h, 0.0, 1.0)
    s = new_h.sum()
    diff = target - s
    if abs(diff) > 1e-12:
        new_h += diff / new_n
        new_h = np.clip(new_h, 0.0, 1.0)
        s = new_h.sum()
        if s > 0:
            new_h = new_h * (target / s)
    return new_h

def propose_random_move(h, eps_max):
    """Transfer a small amount of mass between two random positions."""
    n = h.shape[0]
    for _ in range(10):
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        mx = min(eps_max, h[i], 1.0 - h[j])
        if mx <= 0:
            continue
        eps = np.random.rand() * mx
        nh = h.copy()
        nh[i] -= eps
        nh[j] += eps
        return nh
    return None

def propose_targeted_move(h, lag, eps_max):
    """Move mass from a site that contributes to the worst lag towards its partner."""
    n = h.shape[0]
    if lag == 0:
        return None
    if lag > 0:
        start, stop = 0, n - lag
        for _ in range(10):
            i = np.random.randint(start, stop)
            if h[i] > 0 and h[i + lag] < 1.0:

                if mx <= 0:

                nh = h.copy()
                nh[i] -= eps
                nh[i + lag] += eps

                    continue
                eps = np.random.rand() * mx
