def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    best_h = None
    best_c5 = np.inf
    # Define several n values to explore
    n_candidates = [45, 90, 135]
    # Precompute shapes
    for n in n_candidates:
        target_sum = n * 0.5
        dx = 2.0 / n
        # precompute idx
        idx = (np.arange(n)[:, None] - np.arange(n)[None, :]) + (n - 1)

        # generate initial h candidates: maybe three: tent, sin2, random
        init_candidates = []

        # tent shape
        x = (np.arange(n) + 0.5) * dx  # midpoints
        tent = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
        init_candidates.append(tent)

        # sin^2 shape
        sin2 = np.sin(np.pi * x / 2) ** 2
        init_candidates.append(sin2)

        # if initial_h_values provided and size matches n, use it (maybe scale)
        try:
            h0 = np.array(initial_h_values, dtype=float)
            if h0.size == n:
                init_candidates.append(h0)
            else:
                # Interpolate to n using linear interpolation
                from scipy.interpolate import interp1d
                f = interp1d(np.linspace(0, 2, h0.size, endpoint=False) + dx0/2, h0, kind='linear', fill_value="extrapolate")
                tent2 = f(np.arange(n) * dx + dx/2)  # sample
                init_candidates.append(tent2)
        except Exception:
            pass

        # start optimization for each init candidate
        for init_h in init_candidates:
            h = _project(init_h, target_sum)
            # initial best within this n?
            cur_c5 = _evaluate(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5; best_h = h.copy(); best_n = n; best_idx = idx

        # Run gradient descent combos
        # We'll perform a fixed number of iterations, but break early if time near budget.
        max_iters = 40000
        lr0 = 0.15
        T0, Tmin = 0.15, 0.001
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        # Initialize Adam moments zero
        m = np.zeros(n)
        v = np.zeros(n)
        # Choose starting h as best found for this n so far
        h = best_h if best_n == n else _project(np.random.rand(n), target_sum)  # fallback random
        # baseline
        cur_c5 = _evaluate(h)
        for it in range(max_iters):
            # budget check
            if time.time() - start > budget_s * 0.95:
                break

            # schedule temperature
            T = max(Tmin, T0 * (1 - it / max_iters))

            # compute gradient
            # correlation vector
            g = np.correlate(h, 1 - h, mode='full') * dx
            gmax = g.max()
            # softmax probabilities
            p = np.exp((g - gmax) / T)
            p /= p.sum()
            grad = dx * (p[idx] @ (1 - h) - (p[idx].T) @ h)   # shape (n,)

            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** (it + 1))
            v_hat = v / (1 - beta2 ** (it + 1))
            step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

            # project
            h_new = _project(h - step, target_sum)
            new_c5 = _evaluate(h_new)

            # acceptance: if improvement, accept; else maybe accept if lower than current?
            if new_c5 <= cur_c5:
                h = h_new
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_new.copy()
                    best_n = n
                    best_idx = idx

            # occasional random jitter
            if it % 5000 == 0 and it > 0:
                jitter = np.random.normal(scale=0.02, size=n)

                cand_c5 = _evaluate(cand)

                    if cand_c5 < best_c5:
                        best_c5 = cand_c5
                        best_h = cand.copy()
                        best_n = n
                        best_idx = idx

        # Continue to next n

    return best_h, best_c5, best_n

def _evaluate(h):
    n = h.size
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full') * dx)
def _project(v, target):
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
```

import time
import numpy as np

def _evaluate(h):
    n = h.size
    dx = 2.0 / n
    # correlation defined by the problem statement
    return np.max(np.correlate(h, 1 - h, mode="full") * dx)

def _project(v, target):
    """Project vector onto the feasible set {0≤x≤1, sum x = target}."""
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

    # ------------------------------------------------------------------
