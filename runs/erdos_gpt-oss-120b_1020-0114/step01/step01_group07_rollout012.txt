def _random_binary(N,rng):
    x = np.zeros(N, dtype=np.int8)
    idx = rng.choice(N, size=N//2, replace=False)
    x[idx] = 1
    return x

def _corr_max(x):
    a = x.astype(np.float64)
    b = 1.0 - a
    n = a.size
    fsize = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, fsize)
    B = np.fft.rfft(b[::-1], fsize)
    conv = np.fft.irfft(A * B, fsize)
    start = n - 1
    corr = conv[start:start + 2*n - 1]
    max_idx = int(corr.argmax())
    max_val = float(corr[max_idx])
    return corr, max_val, max_idx

    no_improve = 0

        elapsed = time.time() - start
        if elapsed > budget_s * 0.95:
            break

        progress = elapsed / (budget_s * 0.95)

        if shift != 0:
            if shift > 0:
                left = np.arange(N - shift, dtype=np.int32)
                right = left + shift
            else:
                left = np.arange(-shift, N, dtype=np.int32)
                right = left + shift
            xl = x[left]
            xr = x[right]
            mask_A = (xl == 1) & (xr == 0)
            mask_B = (xl == 0) & (xr == 1)
            A = left[mask_A]
            B = left[mask_B]
        else:

        if A.size and B.size:

import numpy as np, time, math

def _rand_binary(N, rng):
    """Return a binary vector of length N with exactly N//2 ones."""
    x = np.zeros(N, dtype=np.int8)
    ones = rng.choice(N, size=N // 2, replace=False)
    x[ones] = 1
    return x


def _corr_max(x):
    """
    Linear correlation of x with 1-x (as used in the evaluation).
    Returns the full correlation array, its maximum value (raw count), and the index of that maximum.
    """
    a_rev = x[::-1].astype(np.float64)          # reverse first argument
    b = (1.0 - x).astype(np.float64)
    n = a_rev.size
    fsize = 1 << ((2 * n - 1).bit_length())      # next power of two ≥ 2n‑1
    A = np.fft.rfft(a_rev, fsize)
    B = np.fft.rfft(b, fsize)
    conv = np.fft.irfft(A * B, fsize)
    corr = conv[: 2 * n - 1]                     # length 2n‑1, matches np.correlate(...,mode='full')
    idx = int(corr.argmax())
    val = float(corr[idx])
    return corr, val, idx


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # dimension – larger values give more freedom, still well below 1000
    N = 600                         # must be even so that the integral constraint is integral
    x = _rand_binary(N, rng)        # current binary candidate (dtype int8)

    corr, cur_max_raw, cur_idx = _corr_max(x)
    cur_val = cur_max_raw * (2.0 / N)

    best_x = x.copy()
    best_val = cur_val

    start = time.time()
    no_improve = 0
    TARGET = 0.3808

    while True:
        elapsed = time.time() - start
        if elapsed > budget_s * 0.95:
            break
        # linear temperature schedule (helps escape shallow local minima)
        progress = elapsed / (budget_s * 0.95)
        T = 0.20 * (1.0 - progress) + 1e-4

        # shift corresponding to the current worst lag
        shift = cur_idx - (N - 1)

        # build candidate swap that directly reduces the worst lag if possible
        if shift != 0:
            if shift > 0:
                left = np.arange(N - shift, dtype=np.int32)
                right = left + shift
            else:
                left = np.arange(-shift, N, dtype=np.int32)
                right = left + shift
            xl = x[left]
            xr = x[right]
            mask_A = (xl == 1) & (xr == 0)   # 1→0 contributes to this lag
            mask_B = (xl == 0) & (xr == 1)   # 0→1 contributes to opposite direction
            A = left[mask_A]
            B = left[mask_B]
        else:
            A = B = np.empty(0, dtype=np.int32)

        if A.size and B.size:
            i = int(rng.choice(A))
            j = int(rng.choice(B))
        else:
            # generic random swap (preserving the total number of ones)
            ones = np.flatnonzero(x)
            zeros = np.flatnonzero(x == 0)
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))

        # perform the swap
        x[i], x[j] = 0, 1

        new_corr, new_max_raw, new_idx = _corr_max(x)
        new_val = new_max_raw * (2.0 / N)

        delta = new_val - cur_val
        if delta <= 0 or rng.random() < math.exp(-delta / T):
            # accept the move
            cur_val = new_val
            cur_idx = new_idx
            corr = new_corr
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_x = x.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # reject – revert the swap
            x[i], x[j] = 1, 0
            no_improve += 1

        # stagnation handling: a modest random reshuffle of a few bits
        if no_improve > 200_000:
            k = max(1, N // 50)                     # touch ~2 % of the entries
            ones = np.flatnonzero(x)
            zeros = np.flatnonzero(x == 0)
            if ones.size >= k and zeros.size >= k:
                i_sel = rng.choice(ones, size=k, replace=False)
                j_sel = rng.choice(zeros, size=k, replace=False)
                x[i_sel] = 0
                x[j_sel] = 1
                corr, cur_max_raw, cur_idx = _corr_max(x)
                cur_val = cur_max_raw * (2.0 / N)
            no_improve = 0

        # stop early if we already beat the target bound
        if best_val <= TARGET:
            break

    # final evaluation of the best vector found
    final_corr, final_max_raw, _ = _corr_max(best_x)
    final_val = final_max_raw * (2.0 / N)

    if "evaluate_erdos_solution" in globals():
        try:
            final_val = evaluate_erdos_solution(best_x.tolist())
        except Exception:
            pass

    return (best_x.astype(float).tolist(), float(final_val), int(N))
