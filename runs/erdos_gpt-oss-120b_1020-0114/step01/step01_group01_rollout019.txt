    best_h = current_h.copy()
    best_val = current_val

    start = time.monotonic()
    while time.monotonic() - start < time_limit:
        elapsed = time.monotonic() - start
        frac = elapsed / time_limit
        T = 0.01 * (1 - frac) + 1e-12   # temperature schedule

        # choose a random 1‑position and a random 0‑position to swap
        ones_idx = np.where(current_h == 1)[0]
        zeros_idx = np.where(current_h == 0)[0]
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break
        i_one = rng.choice(ones_idx)
        i_zero = rng.choice(zeros_idx)

        new_h = current_h.copy()
        new_h[i_one] = 0
        new_h[i_zero] = 1
        new_val = max_overlap_fft(new_h)

        delta = new_val - current_val
        accept = False
        if delta < -1e-12:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            current_h = new_h
            current_val = new_val
            if new_val < best_val:
                best_h = new_h.copy()
                best_val = new_val
    return best_h, best_val

def optimise_continuous(start_h, time_limit):
    """SLSQP optimisation starting from start_h within the given time budget."""
    n = start_h.shape[0]
    global _opt_n, _opt_dx, _opt_target_sum
    _opt_n = n
    _opt_dx = 2.0 / n
    _opt_target_sum = n / 2.0

    h0 = project_to_feasible(start_h, _opt_target_sum)
    init_corr = np.correlate(h0, 1 - h0, mode='full') * _opt_dx
    t0 = float(np.max(init_corr) + 1e-8)

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    eq_cons = {'type': 'eq', 'fun': eq_constraint_fun}
    m = 2 * n - 1
    ineq_cons = NonlinearConstraint(
        ineq_constraint_fun,
        lb=np.zeros(m),
        ub=np.full(m, np.inf),
    )

    x0 = np.concatenate([h0, [t0]])

    # Simple time guard – we split the global budget beforehand, so no callback needed
    res = minimize(
        objective_fun,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=[eq_cons, ineq_cons],
        options={'maxiter': 800, 'ftol': 1e-9, 'disp': False},
    )

    if not res.success:
        h_opt = h0
        t_opt = t0
    else:
        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        h_opt = project_to_feasible(h_opt, _opt_target_sum)
        t_opt = float(res.x[-1])

    try:
        c5_val = float(evaluate_erdos_solution(h_opt))
    except Exception:
        c5_val = float(t_opt)
    return h_opt, c5_val

def upsample(h, new_n):
    """Linear interpolation of a step function to a finer grid."""
    n = h.size
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h, left=0.0, right=0.0)

def run(seed=42, budget_s=1100, **kwargs):
    start_total = time.monotonic()
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # Base grid and initial vector
    # ----------------------------------------------------------------------
    try:
        init_vec = np.array(initial_h_values, dtype=float)
        n0 = init_vec.shape[0]
    except Exception:
        n0 = 100
        init_vec = np.full(n0, 0.5)

    best_h = None
    best_val = float('inf')
    best_n = n0

    # ----------------------------------------------------------------------
    # Helper to update the current best solution
    # ----------------------------------------------------------------------
    def update_candidate(h, val, n):
        nonlocal best_h, best_val, best_n
        if val < best_val - 1e-12:
            best_h = h.copy()
            best_val = float(val)
            best_n = n

    # ----------------------------------------------------------------------
    # Phase 1 – binary hill climbing on the original grid
    # ----------------------------------------------------------------------
    phase1_budget = budget_s * 0.45
    h_bin, val_bin = binary_hill_climb(init_vec, rng, phase1_budget)
    update_candidate(h_bin, val_bin, n0)

    # a few extra random binary restarts (if time permits)
    extra_restart_budget = budget_s * 0.05
    if time.monotonic() - start_total < budget_s - extra_restart_budget:
        for _ in range(3):
            random_bin = np.zeros(n0)
            random_bin[rng.choice(n0, n0 // 2, replace=False)] = 1.0
            h_tmp, v_tmp = binary_hill_climb(random_bin, rng, extra_restart_budget / 3)
            update_candidate(h_tmp, v_tmp, n0)

    # ----------------------------------------------------------------------
    # Phase 2 – continuous refinement with SLSQP on the same grid
    # ----------------------------------------------------------------------
    phase2_budget = budget_s * 0.30
    sub_budget = phase2_budget / 4
    for _ in range(4):
        if time.monotonic() - start_total > budget_s - 5:
            break
        # small Gaussian perturbation of the current best binary pattern
        start_h = best_h + rng.normal(scale=0.02, size=n0)
        start_h = np.clip(start_h, 0.0, 1.0)
        start_h = project_to_feasible(start_h, n0 / 2.0)
        h_opt, val_opt = optimise_continuous(start_h, sub_budget)
        update_candidate(h_opt, val_opt, n0)

    # ----------------------------------------------------------------------
    # Phase 3 – optional up‑sampling to a finer grid
    # ----------------------------------------------------------------------
    if time.monotonic() - start_total < budget_s * 0.80:
        n2 = n0 * 2
        h_up = upsample(best_h, n2)
        h_up = project_to_feasible(h_up, n2 / 2.0)
        remaining = budget_s - (time.monotonic() - start_total)
        h_opt2, val_opt2 = optimise_continuous(h_up, remaining)
        update_candidate(h_opt2, val_opt2, n2)

    # ----------------------------------------------------------------------
    # Final sanity check with the official evaluator
    # ----------------------------------------------------------------------
    try:
        best_val = float(evaluate_erdos_solution(best_h))
    except Exception:
        pass

    # final projection to be safe
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_feasible(best_h, best_n / 2.0)

    return best_h, best_val, best_n
