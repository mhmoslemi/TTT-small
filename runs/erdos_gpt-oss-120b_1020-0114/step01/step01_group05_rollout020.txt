def optimize_one_restart(n, target, dx, time_deadline):
    # initialize random binary
    h = np.zeros(n)
    ones = np.random.choice(n, target, replace=False)
    h[ones] = 1.0
    # gradient descent
    lr = 0.5
    best_h = h.copy()
    best_val = evaluate_fast(h, dx)
    max_iters = 2000
    for it in range(max_iters):
        if time.time() > time_deadline:
            break
        # compute gradient for max shift
        g = np.correlate(h, 1 - h, mode='full')
        g = g * dx
        max_g = g.max()
        # indices near max
        idxs = np.where(g >= max_g - 1e-12)[0]
        grad = np.zeros_like(h)
        for idx in idxs:
            d = idx - (n-1)
            grad += grad_shift(h, d)
        grad /= len(idxs)

        val_new = evaluate_fast(h_new, dx)
        if val_new < best_val:
            best_val = val_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_val

import time
import numpy as np

def _project(v, target):
    """Projection onto {0≤x≤1, sum x = target} using bisection on the Lagrange multiplier."""
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def _grad_shift(h, d):
    """Gradient of the overlap for a single integer shift d."""
    n = h.shape[0]
    if d >= 0:
        # term1 = (1 - h[i+d]) where i+d < n
        term1 = np.empty_like(h)
        term1[:n - d] = 1.0 - h[d:]
        term1[n - d:] = 0.0
        # term2 = h[i-d] where i-d >= 0
        term2 = np.empty_like(h)
        term2[:d] = 0.0
        term2[d:] = h[:n - d]
    else:
        d = -d
        # term1 = (1 - h[i-d]) where i-d >= 0
        term1 = np.empty_like(h)
        term1[:d] = 0.0
        term1[d:] = 1.0 - h[:n - d]
        # term2 = h[i+d] where i+d < n
        term2 = np.empty_like(h)
        term2[:n - d] = h[d:]
        term2[n - d:] = 0.0
    return term1 - term2

def _eval_fast(h, dx):
    """Fast evaluation of C5 for a discrete h."""
    g = np.correlate(h, 1.0 - h, mode='full')
    return g.max() * dx

def _optimize_once(n, dx, target, deadline):
    """Run a single projected sub‑gradient descent run."""
    # start from a random binary vector (exactly target ones)
    h = np.zeros(n, dtype=float)
    ones = np.random.choice(n, target, replace=False)
    h[ones] = 1.0

    best_h = h.copy()
    best_val = _eval_fast(h, dx)

    lr = 0.5
    max_iters = 5000
    no_improve = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break

        # correlation and indices attaining the current maximum
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        cur_max = corr.max()
        max_idx = np.where(corr >= cur_max - 1e-12)[0]

        # sub‑gradient (average over all maximal shifts)
        grad = np.zeros_like(h)
        for idx in max_idx:
            d = idx - (n - 1)
            grad += _grad_shift(h, d)
        grad /= max_idx.size

        # simple back‑tracking line search
        step = lr
        improved = False
        cur_val = cur_max
        for _ in range(12):
            h_new = _project(h - step * grad, target)
            new_val = _eval_fast(h_new, dx)
            if new_val <= cur_val:
                h = h_new
                cur_val = new_val
                improved = True
                break
            step *= 0.5

        if improved:
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_h = h.copy()
                no_improve = 0
                lr = min(lr * 1.07, 1.0)          # accelerate after success
            else:
                no_improve += 1
                lr = max(lr * 0.97, 1e-4)
        else:
            # no improvement – shrink learning rate and maybe jitter later
            lr = max(lr * 0.5, 1e-4)
            no_improve += 1

        if no_improve > 2000 and (it % 200 == 0):
            # small random perturbation to escape flat regions
            jitter = np.random.normal(scale=0.02, size=n)
            h = _project(h + jitter, target)
            no_improve = 0

    return best_h, best_val

def _hill_climb_binary(h_bin, dx, target, deadline):
    """Binary (0/1) local search by swapping a 1 with a 0."""
    cur = h_bin.copy()
    cur_val = _eval_fast(cur, dx)

    while time.time() < deadline:
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        i1 = np.random.choice(ones)
        i0 = np.random.choice(zeros)
        nxt = cur.copy()
        nxt[i1] = 0.0
        nxt[i0] = 1.0
        nxt_val = _eval_fast(nxt, dx)
        if nxt_val < cur_val - 1e-12:
            cur = nxt
            cur_val = nxt_val
        # stop after many fruitless attempts
        if np.random.rand() < 1e-4:
            break
    return cur, cur_val

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s * 0.95

    # ----- choose a working resolution -----
    n = 120                     # even size gives integer target = n/2
    target = n // 2
    dx = 2.0 / n

    best_h = None
    best_val = np.inf

    # ----- use the supplied parent if it exists (interpolate to n) -----
    try:
        parent = np.array(initial_h_values, dtype=float)
        if parent.size != n:
            # interpolate uniformly onto our grid
            x_parent = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            parent = np.interp(x_new, x_parent, parent)
        parent = _project(parent, target)
        val = _eval_fast(parent, dx)
        if val < best_val:
            best_val = val
            best_h = parent.copy()
    except Exception:
        pass  # no inherited vector, continue with random starts

    # ----- main optimization loop with many random restarts -----
    while time.time() < deadline:
        h_opt, val_opt = _optimize_once(n, dx, target, deadline)
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()

        # try a binary hill‑climb from the continuous result
        # convert to binary keeping the exact sum
        bin_h = (h_opt > 0.5).astype(float)
        diff = target - bin_h.sum()
        if diff > 0:
            zeros = np.where(bin_h == 0)[0]
            flip = np.random.choice(zeros, int(diff), replace=False)
            bin_h[flip] = 1.0
        elif diff < 0:
            ones = np.where(bin_h == 1)[0]
            flip = np.random.choice(ones, int(-diff), replace=False)
            bin_h[flip] = 0.0

        bin_h, bin_val = _hill_climb_binary(bin_h, dx, target, deadline)
        if bin_val < best_val:
            best_val = bin_val
            best_h = bin_h.copy()

    # final safeguard – ensure feasibility
    best_h = _project(best_h, target)
    best_val = _eval_fast(best_h, dx)

    return best_h, best_val, n
