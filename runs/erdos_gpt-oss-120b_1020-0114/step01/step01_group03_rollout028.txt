import time
import numpy as np

def _max_corr_and_shift(h, dx):
    """Return (max_correlation, shift) for binary h."""
    corr = np.correlate(h, 1 - h, mode="full")
    max_c = int(corr.max())
    idx = int(np.argmax(corr))
    shift = idx - (len(h) - 1)      # negative for left shift
    return max_c, shift


def _greedy_optimize(h, dx, time_limit):
    """Greedy hill‑climbing: improve the worst shift as long as possible."""
    N = len(h)
    while True:
        # stop if out of time
        if time.time() > time_limit:
            break
        max_c, shift = _max_corr_and_shift(h, dx)
        if max_c == 0:
            break
        w = abs(shift)
        if w == 0:
            break
        direction = 1 if shift > 0 else -1

        # neighbour contributions for shift w
        left = np.zeros(N, dtype=int)
        right = np.zeros(N, dtype=int)
        left[w:] = h[:-w]                # h[i‑w] for i≥w
        right[:N - w] = h[w:]            # h[i+w] for i+w<N
        neigh = left + right

        # early region depends on direction
        if direction == 1:               # positive shift
            early = np.arange(N) < N - w
        else:                            # negative shift
            early = np.arange(N) >= w
        early_i = early.astype(int)

        L = early_i - neigh               # L_i for every index
        R = early_i - neigh               # R_j for every index (same formula)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break

        L_one = L[ones]
        R_zero = R[zeros]

        i = ones[int(np.argmax(L_one))]   # 1‑position with largest L
        j = zeros[int(np.argmin(R_zero))] # 0‑position with smallest R

        delta = int(R[j]) - int(L[i])     # change in mismatches for this shift
        if delta < 0:
            # perform the swap
            h[i] = 0
            h[j] = 1
            continue
        # no improving swap for the current worst shift
        break
    return h


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function with a small C₅ value."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.97  # safety margin

    # ------------------------------------------------------------------
    # Initialise a binary pattern (exactly N/2 ones)
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)

        h0[: N // 2] = 1

    dx = 2.0 / N

    best_h = h0.copy()
    best_corr, _ = _max_corr_and_shift(best_h, dx)

    while time.time() < deadline:
