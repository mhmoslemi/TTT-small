def evaluate_c5(h):
    return float(np.max(np.correlate(h, 1 - h, mode='full') * (2.0 / len(h))))

def simulated_annealing(h, max_iters, time_limit):
    n = len(h); dx = 2.0/n
    cur_h = h.copy()
    cur_c5 = evaluate_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    start = time.time()
    iter = 0
    T0 = 0.02  # initial temperature
    # We'll scale T based on difference existing between 1 and cur_c5 maybe.
    while time.time() - start < time_limit and iter < max_iters:
        iter += 1
        # cooling schedule:
        T = T0 * (1 - iter / max_iters)
        # pick two random distinct indices
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j:
            continue
        # ensure swap changes. If both same value, skip
        if cur_h[i] == cur_h[j]:
            continue
        # propose swap
        new_h = cur_h.copy()
        new_h[i], new_h[j] = cur_h[j], cur_h[i]  # swap
        # compute new c5
        new_c5 = evaluate_c5(new_h)
        # accept?
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            cur_h = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
        # optionally, bigger moves after some iterations
    return best_h, best_c5

import time
import numpy as np
from scipy.optimize import minimize, NonlinearConstraint

# globals used by the optimizer
_opt_n = None          # number of points
_opt_dx = None         # grid spacing 2 / n
_opt_target_sum = None # n / 2


def eval_c5(h: np.ndarray) -> float:
    """Calculate max_k ∫ h(x)(1-h(x+k))dx on the discrete grid."""
    n = h.shape[0]
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))


def project_to_feasible(h: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and linearly renormalise so that sum(h)=target."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == 0.0:
        # put all mass on a random subset
        idx = np.random.choice(len(h), int(round(target)), replace=False)
        h[idx] = 1.0
        return h
    scale = target / cur
    h = np.clip(h * scale, 0.0, 1.0)
    # small iterative correction
    for _ in range(10):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            i = np.random.randint(len(h))
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
            continue
        delta = diff / free.size
        h[free] += delta
        np.clip(h, 0.0, 1.0, out=h)
    return h


def objective_fun(x):
    """Objective for SLSQP – minimise the auxiliary variable t."""
    return x[-1]


def eq_constraint_fun(x):
    """Equality constraint: sum(h) = n/2."""
    return np.sum(x[:_opt_n]) - _opt_target_sum


def ineq_constraint_fun(x):
    """Inequality: t >= overlap for every shift."""
    h = x[:_opt_n]
    t = x[-1]
    corr = np.correlate(h, 1 - h, mode="full") * _opt_dx
    return t - corr  # must be >= 0


def optimise_start(start_h: np.ndarray, maxiter: int = 1200):
    """Run a single SLSQP optimisation starting from start_h."""
    global _opt_n, _opt_dx, _opt_target_sum
    n = int(start_h.shape[0])
    _opt_n = n
    _opt_dx = 2.0 / n
    _opt_target_sum = n / 2.0

    h0 = project_to_feasible(start_h, _opt_target_sum)

    init_corr = np.correlate(h0, 1 - h0, mode="full") * _opt_dx
    t0 = float(np.max(init_corr) + 1e-8)

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    eq_cons = {"type": "eq", "fun": eq_constraint_fun}
    m = 2 * n - 1
    ineq_cons = NonlinearConstraint(
        ineq_constraint_fun,
        lb=np.zeros(m),
        ub=np.full(m, np.inf),
    )

    x0 = np.concatenate([h0, [t0]])

    res = minimize(
        objective_fun,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=[eq_cons, ineq_cons],
        options={"maxiter": maxiter, "ftol": 1e-9, "disp": False},
    )

    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    h_opt = project_to_feasible(h_opt, _opt_target_sum)

    try:
        c5_val = float(evaluate_erdos_solution(h_opt))
    except Exception:
        c5_val = float(res.x[-1])

    return h_opt, c5_val

        i = rng.choice(zeros)
        j = rng.choice(ones)

        new_h = cur_h.copy()
        new_h[i] = 1.0
        new_h[j] = 0.0
        new_c5 = float(np.max(np.correlate(new_h, 1 - new_h, mode="full") * dx))
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            cur_h = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
    return best_h, best_c5


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # Determine problem size and an initial vector (if supplied by the harness)
    # ----------------------------------------------------------------------
    try:
        init_vec = np.array(initial_h_values, dtype=float)
        n_points = int(init_vec.shape[0])
    except Exception:
        init_vec = None
        n_points = 100

    target_sum = n_points / 2.0
    best_h = None
    best_c5 = float("inf")
    best_n = n_points

    # ----------------------------------------------------------------------
    # Phase 1 – binary hill climbing (~60% of the budget)
    # ----------------------------------------------------------------------
    hill_budget = budget_s * 0.6
    hill_deadline = start_time + hill_budget

    # start from a deterministic binary version of the provided vector if possible
    if init_vec is not None:
        order = np.argsort(-init_vec)
        bin_start = np.zeros(n_points, dtype=float)
        bin_start[order[: n_points // 2]] = 1.0
    else:
        bin_start = np.zeros(n_points, dtype=float)
        bin_start[rng.choice(n_points, n_points // 2, replace=False)] = 1.0

    h_hc, c_hc = hill_climb_binary(bin_start, max(0.0, hill_deadline - time.time()), rng)
    if c_hc < best_c5:
        best_h, best_c5, best_n = h_hc, c_hc, n_points

    # additional random binary restarts while time permits
    while time.time() < hill_deadline:
        rand_bin = np.zeros(n_points, dtype=float)
        rand_bin[rng.choice(n_points, n_points // 2, replace=False)] = 1.0
        h_tmp, c_tmp = hill_climb_binary(
            rand_bin, max(0.0, hill_deadline - time.time()), rng
        )
        if c_tmp < best_c5:
            best_h, best_c5, best_n = h_tmp, c_tmp, n_points
        if best_c5 <= 0.3808:
            break

    # ----------------------------------------------------------------------
    # Phase 2 – SLSQP refinements (remaining budget)
    # ----------------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.97:
        # give each optimisation at most ~30 s, but break early if little time left
        time_left = budget_s - (time.time() - start_time)
        if time_left < 5.0:
            break

        # Choose a starting vector: best found so far, possibly perturbed
        if best_h is not None:
            start_vec = best_h.copy()
        else:
            start_vec = np.zeros(n_points, dtype=float)
            start_vec[rng.choice(n_points, n_points // 2, replace=False)] = 1.0

        # add a modest random perturbation to escape possible local minima
        perturb = rng.random(n_points) * 0.2 - 0.1  # [-0.1,0.1]
        start_vec = np.clip(start_vec + perturb, 0.0, 1.0)
        start_vec = project_to_feasible(start_vec, target_sum)

        try:
            h_opt, c_opt = optimise_start(start_vec, maxiter=1500)
        except Exception:
            # fallback: pure binary start
            alt = np.zeros(n_points, dtype=float)
            alt[rng.choice(n_points, n_points // 2, replace=False)] = 1.0
            h_opt, c_opt = optimise_start(alt, maxiter=1200)

        if c_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c_opt, n_points
        if best_c5 <= 0.3808:
            break

    # ----------------------------------------------------------------------
    # Final sanity check and return
    # ----------------------------------------------------------------------
    if best_h is None:
        best_h = np.full(n_points, 0.5, dtype=float)
        best_c5 = eval_c5(best_h)

    try:
        best_c5 = float(evaluate_erdos_solution(best_h))
    except Exception:
        best_c5 = eval_c5(best_h)

    return best_h, best_c5, best_n
