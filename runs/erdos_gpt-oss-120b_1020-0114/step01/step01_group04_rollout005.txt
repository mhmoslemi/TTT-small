def improved_greedy(h, time_limit):
    n = h.shape[0]
    target = n/2.0
    best_h = h.copy()
    best_val = _compute_c5(best_h)
    cur_h = h.copy()
    cur_val = best_val
    start = time.time()
    step = 0.2  # initial step size
    min_step = 1e-4
    # We'll use a cooling schedule: step decays if no improvement for 200 steps.
    no_improve = 0
    while time.time() - start < time_limit:
        # compute worst shift and w
        shift, _ = _worst_shift(cur_h)
        w = _w_vector(cur_h, shift)
        # find up and down indices
        up_mask = cur_h < 1 - 1e-12
        down_mask = cur_h > 1e-12
        if not up_mask.any() or not down_mask.any():  # cannot move
            break
        # choose up idx with max w among up_mask
        i_up = np.argmax(np.where(up_mask, w, -np.inf))
        # choose down idx with min w among down_mask
        i_down = np.argmin(np.where(down_mask, w, np.inf))
        # delta limited by step and bounds
        delta = min(step, 1.0 - cur_h[i_up], cur_h[i_down])
        if delta <= 0:
            # maybe try swapped role if w suggests else random
            # fallback: random feasible move
            i_up, i_down, delta = random_feasible_move(cur_h, step)
        new_h = cur_h.copy()
        new_h[i_up] += delta
        new_h[i_down] -= delta
        new_val = _compute_c5(new_h)
        if new_val < cur_val - 1e-12:
            cur_h, cur_val = new_h, new_val
            no_improve = 0
            step = min(step * 1.03, 0.3)   # increase step a bit
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
        else:
            # Simulated annealing acceptance
            T = step * 0.5
            if T > 0 and np.random.rand() < math.exp(-(new_val - cur_val)/T):
                cur_h, cur_val = new_h, new_val
            no_improve += 1
            if no_improve >= 200:
                step = max(step * 0.5, min_step)
                no_improve = 0
                if step <= min_step:
                    break
    # final projection to enforce sum
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        free = np.where((best_h > 1e-12) & (best_h < 1 - 1e-12))[0]
        if free.size:
            best_h[free] += diff / free.size
        else:
            best_h[0] += diff
        best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, _compute_c5(best_h)

def anneal_binary(h0, time_limit):
    n = h0.shape[0]
    target = n / 2
    best_h = h0.copy()
    best_val = _compute_c5(best_h)
    cur_h = h0.copy()
    cur_val = best_val
    start = time.time()
    # Temperature schedule: start T = 0.1
    T0 = 0.2
    # Let's compute approximate number of iterations we can afford: given time, we can loop without predetermined count.
    while time.time() - start < time_limit:
        # choose random 1 and 0 indices
        ones = np.flatnonzero(cur_h > 0.5)   # treat near 1
        zeros = np.flatnonzero(cur_h < 0.5)
        if ones.size == 0 or zeros.size == 0:
            break
        i_up = np.random.choice(ones)
        i_down = np.random.choice(zeros)
        # swap
        new_h = cur_h.copy()
        new_h[i_up] = 0.0
        new_h[i_down] = 1.0
        new_val = _compute_c5(new_h)
        # accept if better or with prob
        # compute temperature as decaying with elapsed time
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = T0 * (1 - frac) + 1e-4
        if new_val < cur_val or np.random.rand() < math.exp(-(new_val - cur_val) / T):
            cur_h, cur_val = new_h, new_val
            if new_val < best_val:
                best_h, best_val = new_h.copy(), new_val
    # Ensure sum exactly (should be)
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        # adjust one index
        idx = np.random.randint(0, n)
        best_h[idx] = np.clip(best_h[idx] + diff, 0, 1)
    best_h = np.clip(best_h, 0, 1)
    return best_h, best_val

import time, math
import numpy as np

# ----------------------------------------------------------------------
# Core evaluation utilities
# ----------------------------------------------------------------------
def _compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)·(1‑h(x+k)) dx  (discretized)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)


def _worst_shift(h: np.ndarray):
    """Return shift with maximal overlap and the full correlation array."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)
    return shift, corr


def _w_vector(h: np.ndarray, shift: int):
    """Weight w_i = h_{i+shift}+h_{i-shift} (zeros outside)."""
    n = h.shape[0]
    if shift == 0:
        return 2.0 * h
    w = np.zeros_like(h)
    if shift > 0:
        w[: n - shift] += h[shift:]
        w[shift:] += h[: n - shift]
    else:  # shift < 0
        s = -shift
        w[s:] += h[: n - s]
        w[: n - s] += h[s:]
    return w


def _top_shifts(h: np.ndarray, k: int = 2):
    """Indices of the k worst shifts (including sign)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    # argpartition yields unsorted but fast
    idxs = np.argpartition(corr, -k)[-k:]
    return idxs - (n - 1)


def _aggregate_w(h: np.ndarray, shifts):
    """Sum of w‑vectors for a list of shifts."""
    w = np.zeros_like(h)
    for s in shifts:
        w += _w_vector(h, s)
    return w


# ----------------------------------------------------------------------
# Heuristic optimisation (greedy + simulated annealing)
# ----------------------------------------------------------------------
def improved_greedy(start_h: np.ndarray, time_limit: float):
    """Local optimisation driven by worst‑shift weights."""
    n = start_h.shape[0]
    target = n / 2.0
    best_h = start_h.copy()
    best_val = _compute_c5(best_h)

    cur_h = start_h.copy()
    cur_val = best_val

    step = 0.4               # start with a relatively large move size
    min_step = 1e-4
    no_improve = 0
    start = time.time()

    while time.time() - start < time_limit:
        # 1) find several worst shifts and compute combined weight
        shifts = _top_shifts(cur_h, k=2)
        w = _aggregate_w(cur_h, shifts)

        # 2) masks for feasible up/down moves
        up_mask = cur_h < 1.0 - 1e-12       # can increase
        down_mask = cur_h > 1e-12           # can decrease
        if not up_mask.any() or not down_mask.any():
            break

        # choose indices guided by w
        w_up = np.where(up_mask, w, -np.inf)
        i_up = int(np.argmax(w_up))
        w_down = np.where(down_mask, w, np.inf)
        i_down = int(np.argmin(w_down))

        # feasible delta
        delta = min(step, 1.0 - cur_h[i_up], cur_h[i_down])
        if delta <= 0:
            # fall back to a random feasible pair
            up_idx = np.flatnonzero(up_mask)
            down_idx = np.flatnonzero(down_mask)
            i_up = int(np.random.choice(up_idx))
            i_down = int(np.random.choice(down_idx))
            delta = min(step, 1.0 - cur_h[i_up], cur_h[i_down])

        # propose new h
        new_h = cur_h.copy()
        new_h[i_up] += delta
        new_h[i_down] -= delta
        new_val = _compute_c5(new_h)

        # acceptance
        if new_val < cur_val - 1e-12:
            cur_h, cur_val = new_h, new_val
            no_improve = 0
            step = min(step * 1.04, 0.5)          # be a bit more aggressive after improvement
            if new_val < best_val:
                best_h, best_val = new_h.copy(), new_val
        else:
            # simulated‑annealing style acceptance
            T = step * 0.3
            if T > 0 and np.random.rand() < math.exp(-(new_val - cur_val) / T):
                cur_h, cur_val = new_h, new_val
            no_improve += 1
            if no_improve > 300:
                step = max(step * 0.6, min_step)
                no_improve = 0
                if step <= min_step:
                    break

    # final projection onto the hyperplane ∑h = target
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        free = np.where((best_h > 1e-12) & (best_h < 1.0 - 1e-12))[0]
        if free.size:
            best_h[free] += diff / free.size
        else:
            best_h[0] = np.clip(best_h[0] + diff, 0.0, 1.0)
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, _compute_c5(best_h)

def upsample_h(old_h: np.ndarray, n_new: int):

    n_old = old_h.shape[0]
    if n_old == n_new:
        return old_h.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)

        new_h = np.clip(new_h, 0.0, 1.0)

    # ------------------------------------------------------------------
    # Final refinement on the best solution found
    # ------------------------------------------------------------------
    time_left = total_budget - (time.time() - start_total)
    if time_left > 2.0 and best_h is not None:
        refined_h, refined_val = improved_greedy(best_h, time_left)
        if refined_val < best_val:
            best_h, best_val = refined_h, refined_val

    # Safety projection to respect constraints exactly
    if best_h is None:                     # fallback – should never happen
        best_h = random_binary_h(n_target)
        best_val = _compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    target_sum = best_h.shape[0] / 2.0
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        free = np.where((best_h > 1e-12) & (best_h < 1.0 - 1e-12))[0]
        if free.size:
            best_h[free] += diff / free.size
        else:
            best_h[0] = np.clip(best_h[0] + diff, 0.0, 1.0)
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = _compute_c5(best_h)
    return best_h, float(final_c5), int(best_h.shape[0])
