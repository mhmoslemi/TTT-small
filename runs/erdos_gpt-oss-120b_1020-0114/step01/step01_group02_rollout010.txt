def _proj_sum_box(y, target):
    lo = -np.max(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi)/2.0, 0.0, 1.0)
    # final correction for rounding errors
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        # distribute diff to entries not at bounds
        idxs = np.where((x > 0) & (x < 1))[0]
        if len(idxs) > 0:
            add = diff / len(idxs)
            x[idxs] += add
            x = np.clip(x, 0.0, 1.0)
    return x

def _gradient_for_lag(h, lag):
    N = len(h)
    g = np.ones(N)
    L = abs(lag)
    if L == 0:
        g = 1 - 2*h
    else:
        if lag > 0:
            g[:N-L] -= h[L:]
            g[L:]   -= h[:N-L]
        else:
            # same pattern due to symmetry
            g[:N-L] -= h[L:]
            g[L:]   -= h[:N-L]
    return g
```

import numpy as np
import time, random

def _compute_c5(h):
    """Exact C5 using full linear correlation (as required by the checker)."""
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / N))

def _max_corr_and_lag(h):
    """Return (max correlation value, lag) using FFT for speed."""
    N = len(h)
    n = 2 * N
    # linear convolution via zero‑padding
    corr = np.fft.ifft(np.fft.fft(h, n=n) *
                       np.fft.fft(1.0 - h, n=n)).real
    corr = corr[: 2 * N - 1]              # keep the linear part
    idx = int(np.argmax(corr))
    lag = idx - (N - 1)                   # zero shift is at index N‑1
    return float(corr[idx]), lag

def _gradient_for_lag(h, lag):
    """Gradient of the correlation w.r.t. h for the given lag."""
    N = len(h)
    L = abs(lag)
    if L == 0:
        return 1.0 - 2.0 * h
    g = np.ones(N)
    # i+L term
    g[: N - L] -= h[L:]
    # i‑L term
    g[L:] -= h[: N - L]
    return g

def _proj_sum_box(y, target, iters=50):
    """Projection of y onto {0≤x≤1, sum(x)=target} (water‑filling via binary search)."""
    lo = -np.max(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(iters):
        mid = (lo + hi) * 0.5
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

    # tiny correction for numerical drift
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = np.where((x > 0) & (x < 1))[0]
        if free.size:
            add = diff / free.size
            x[free] += add
            np.clip(x, 0.0, 1.0, out=x)
        else:
            # adjust a single coordinate if everything is saturated
            i = np.argmax(x) if diff > 0 else np.argmin(x)
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def _pgd_step(h, step):
    """One projected gradient step for the current worst lag."""
    max_corr, lag = _max_corr_and_lag(h)
    grad = _gradient_for_lag(h, lag)
    y = h - step * grad
    y = np.clip(y, 0.0, 1.0)
    target = len(h) / 2.0
    x = _proj_sum_box(y, target)
    return x, max_corr, lag
