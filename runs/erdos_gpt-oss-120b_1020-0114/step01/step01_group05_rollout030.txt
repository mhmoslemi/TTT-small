def neighbor_sum(h, shift):
    n = h.shape[0]
    s = np.zeros(n)
    if shift >= 0:
        s[:n-shift] += h[shift:]    # h_i + shift neighbor
        s[shift:] += h[:n-shift]    # h_i - shift neighbor
    else:
        s[-shift:] += h[:n+shift]   # shift negative, i+shift => i - |shift|
        s[:n+shift] += h[-shift:]   # i-shift => i + |shift|
    return s

Check indexes: For shift <0, define k = -shift > 0.

We need to ensure that the contributions do not double count at i where both i+shift and i-shift are inside range. For k>0, each pair counted twice: both i's neighbor contributions. That's expected.

Now for shift 0: code should result in s = 2*h, because shift >=0 path: shift=0 => s[:n] += h[0:] => s += h; and s[0:] += h[:n] => s += h again => s=2h. So works.

def _softmax_grad(h, T):
    n = h.size
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    gmax = g.max()
    p = np.exp((g - gmax) / T)
    p = p / p.sum()
    # create index matrix for convolution:
    idx = (np.arange(n)[:, None] - np.arange(n)[None, :]) + (n - 1)
    P = p[idx]  # n×n
    grad = dx * (P @ (1 - h) - P.T @ h)
    return grad

def refine_softmax(h, target_sum, time_budget):
    start = time.time()
    n = h.size
    lr = 0.2
    T0 = 0.05
    Tmin = 0.001
    max_iters = 20000
    best_h = h.copy()
    best_c5 = _evaluate(h)
    cur_h = h.copy()
    for it in range(max_iters):
        if time.time() - start > time_budget * 0.95:
            break
        T = max(Tmin, T0 * (1.0 - it / max_iters))
        g = _softmax_grad(cur_h, T)
        # line search
        step = lr
        for _ in range(8):
            h_new = _project(cur_h - step * g, target_sum)
            c5_new = _evaluate(h_new)
            if c5_new < best_c5:
                cur_h = h_new
                best_h = h_new.copy()
                best_c5 = c5_new
                lr = min(lr * 1.05, 1.0)
                break
            # accept if better than current
            cur_c5 = _evaluate(cur_h)
            if c5_new < cur_c5:
                cur_h = h_new
                lr = max(lr * 0.97, 1e-6)
                break
            step *= 0.5
        else:
            lr = max(lr * 0.5, 1e-6)
        # occasional random jitter if stuck
        if it % 3000 == 0:
            # jitter
            jitter = np.random.normal(scale=0.02, size=n)
            h_jit = _project(cur_h + jitter, target_sum)
            c_jit = _evaluate(h_jit)
            if c_jit < best_c5:
                cur_h = h_jit
                best_h = h_jit.copy()
                best_c5 = c_jit
    return best_h, best_c5

import time
import numpy as np

def _project(v, target):
    """Project v onto [0,1] and sum(v)=target (within tolerance)."""
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _evaluate(h):
    n = h.size
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full') * dx)


def _neighbor_sum(h, shift):
    """Sum of h[i+shift] + h[i-shift] for each i (missing terms omitted)."""
    n = h.shape[0]
    s = np.zeros(n)
    if shift >= 0:
        k = shift
        if k < n:
            s[: n - k] += h[k:]
            s[k:] += h[: n - k]
    else:
        k = -shift
        if k < n:
            s[k:] += h[: n - k]
            s[: n - k] += h[k:]
    return s


def _binary_start(n):
    """Random binary start (values 0,1, maybe a single 0.5 for odd n)."""
    h = np.zeros(n)
    ones = n // 2
    idx = np.random.choice(n, ones, replace=False)
    h[idx] = 1.0
    if n % 2:
        # one position gets 0.5 to meet the exact sum n/2
        remaining = np.setdiff1d(np.arange(n), idx)
        h[np.random.choice(remaining, 1)[0]] = 0.5
    return h


def _search_from(h_start, target, time_limit):
    """Local search driven by the worst shift."""
    start = time.time()
    h_cur = _project(h_start, target)
    cur_c5 = _evaluate(h_cur)
    best_h = h_cur.copy()
    best_c5 = cur_c5
    n = h_cur.size
    last_imp = 0
    it = 0
    while True:
        if time.time() - start > time_limit * 0.95:
            break
        # schedule
        frac = it / 500000.0
        step = 0.3 * (1 - min(frac, 1.0)) + 0.001 * min(frac, 1.0)
        T = 0.02 * (1 - min(frac, 1.0)) + 0.0001 * min(frac, 1.0)

        # worst shift
        corr = np.correlate(h_cur, 1 - h_cur, mode='full') * (2.0 / n)
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)
        cur_c5 = corr[idx_max]

        ns = _neighbor_sum(h_cur, shift)

        low = np.where(h_cur > 1e-12)[0]
        high = np.where(h_cur < 1 - 1e-12)[0]
        if low.size == 0 or high.size == 0:
            break
        i_low = low[np.argmin(ns[low])]
        j_high = high[np.argmax(ns[high])]
        delta = min(step, h_cur[i_low], 1 - h_cur[j_high])
        if delta > 0:
            h_new = h_cur.copy()
            h_new[i_low] -= delta
            h_new[j_high] += delta
            new_c5 = _evaluate(h_new)
            accept = False
            if new_c5 < cur_c5:
                accept = True
            elif np.random.rand() < np.exp(-(new_c5 - cur_c5) / (T + 1e-12)):
                accept = True
            if accept:
                h_cur = h_new
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_new.copy()
                    last_imp = it

        if it - last_imp > 5000:
            # random jitter to escape plateaus
            jitter = np.random.normal(scale=0.05, size=n)
            h_jit = _project(h_cur + jitter, target)
            c5_jit = _evaluate(h_jit)
            if c5_jit < best_c5:
                h_cur = h_jit
                cur_c5 = c5_jit
                best_h = h_jit.copy()
                best_c5 = c5_jit
                last_imp = it
            else:
                h_cur = best_h.copy()
                cur_c5 = best_c5
        it += 1
    return best_h, best_c5


def _softmax_grad(h, T):
    n = h.size
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    gmax = g.max()
    p = np.exp((g - gmax) / T)
    p /= p.sum()
    idx = (np.arange(n)[:, None] - np.arange(n)[None, :]) + (n - 1)
    P = p[idx]
    grad = dx * (P @ (1 - h) - P.T @ h)
    return grad


def _refine_softmax(h, target, time_budget):
    start = time.time()
    n = h.size
    lr = 0.2
    T0, Tmin = 0.05, 0.001
    max_it = 15000
    best_h = h.copy()
    best_c5 = _evaluate(h)
    cur_h = h.copy()
    for it in range(max_it):
        if time.time() - start > time_budget * 0.95:
            break
        T = max(Tmin, T0 * (1.0 - it / max_it))
        g = _softmax_grad(cur_h, T)
        step = lr
        for _ in range(8):
            h_new = _project(cur_h - step * g, target)
            c5_new = _evaluate(h_new)
            if c5_new < best_c5:
                cur_h = h_new
                best_h = h_new.copy()
                best_c5 = c5_new
                lr = min(lr * 1.05, 1.0)
                break
            cur_c5 = _evaluate(cur_h)
            if c5_new < cur_c5:
                cur_h = h_new
                lr = max(lr * 0.97, 1e-6)
                break
            step *= 0.5
        else:
            lr = max(lr * 0.5, 1e-6)
        if it % 3000 == 0:
            jitter = np.random.normal(scale=0.02, size=n)
            h_jit = _project(cur_h + jitter, target)
            c_j = _evaluate(h_jit)
            if c_j < best_c5:
                cur_h = h_jit
                best_h = h_jit.copy()
                best_c5 = c_j
    return best_h, best_c5
