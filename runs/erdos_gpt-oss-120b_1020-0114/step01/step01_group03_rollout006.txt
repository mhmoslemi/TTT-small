def project_onto_capped_simplex(v, target):
    # Ensure within [0,1] via water-filling.
    # Use binary search on lambda.
    lo = -1.0  # you can set generic lower bound: -max(v) - 1 catches all >1 after subtract.
    hi = 1.0   # upper bound: max(v) maybe 1.
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(v - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    # adjust small diff
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        # distribute diff among non-saturated entries
        free_idxs = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
        if free_idxs.size > 0:
            # distribute evenly
            add = diff / free_idxs.size
            h[free_idxs] += add
            # clip again
            h = np.clip(h, 0.0, 1.0)
        else:
            # fallback: adjust all then clip
            h += diff / v.size
            h = np.clip(h, 0.0, 1.0)
            # final scale adjust
            s = h.sum()
            if s != 0:
                h *= target / s
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def project_onto_capped_simplex(v, target):
    lo = -np.max(v) - 1.0  # enough low to push all to 1 if needed
    hi = np.max(v) + 1.0   # high enough for sum zero
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(v - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        free = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
        if free.size > 0:
            h[free] += diff / free.size
            h = np.clip(h, 0.0, 1.0)
        else:
            # fallback simple rescale
            s = h.sum()
            if s > 0:
                h = h * (target / s)
            h = np.clip(h, 0.0, 1.0)
    return h

import time
import numpy as np

# ---------- helper utilities -------------------------------------------------

def compute_c5(h: np.ndarray) -> float:
    """Exact C₅ value for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))


def project_onto_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the set {0 ≤ x ≤ 1, sum x = target}.
    Uses a simple binary search on the Lagrange multiplier λ.
    """
    lo = -np.max(v) - 1.0   # low enough → all entries become 1
    hi = np.max(v) + 1.0    # high enough → all entries become 0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

    # tiny correction for numerical drift
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-12) & (proj < 1.0 - 1e-12))[0]
        if free.size:
            proj[free] += diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
        else:
            # fallback: simple rescaling (rare)
            s = proj.sum()
            if s > 0:
                proj *= target / s
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def smooth_obj_grad(h: np.ndarray, alpha: float):
    """
    Smoothed version of the max-overlap objective and its gradient.
    Uses log‑sum‑exp with smoothing factor α.
    Returns (objective value, gradient vector).
    """
    n = h.shape[0]
    dx = 2.0 / n
    w = 1.0 - h
    corr = np.correlate(h, w, mode="full")          # length 2n‑1

    # stable log‑sum‑exp
    a_corr = alpha * corr
    max_ac = np.max(a_corr)
    exp_shift = np.exp(a_corr - max_ac)
    sum_exp = np.sum(exp_shift)
    f = dx * (max_ac + np.log(sum_exp)) / alpha   # smoothed max

    soft = exp_shift / sum_exp                     # softmax weights
    soft_rev = soft[::-1]

    # convolution gives the two linear parts of the gradient efficiently
    conv1 = np.convolve(soft_rev, w, mode="full")
    conv2 = np.convolve(soft, h, mode="full")
    start = n - 1
    g1 = conv1[start:start + n]
    g2 = conv2[start:start + n]

    grad = dx * (g1 - g2)
    return f, grad
