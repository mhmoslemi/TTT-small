    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    total_time = budget_s

    # Determine sample size
    N = 200  # aim for higher resolution
    target_sum = N/2.0

    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=np.float64)
        # Interpolate to length N
        x_old = np.linspace(0, 1, len(init))
        x_new = np.linspace(0, 1, N)
        h = np.interp(x_new, x_old, init)
    else:
        h = np.random.rand(N)
    # clip and repair sum
    np.clip(h, 0.0, 1.0, out=h)
    h = _repair_sum(h, target_sum)

    current_h = h.copy()
    current_c5 = _compute_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5

    # schedule parameters
    T0 = 0.2
    T_end = 1e-5
    max_move_initial = 0.3

    iteration = 0
    while True:
        elapsed = time.time() - start
        if elapsed > total_time - 2.0:
            break
        progress = elapsed / (total_time - 2.0)
        T = T0 * (T_end / T0) ** progress
        max_move = max_move_initial * (1 - progress) + 1e-4

        # Choose move type
        r = np.random.rand()
        if r < 0.02:
            # big random perturbation
            cand = best_h.copy()
            m = np.random.randint(1, max(2, N // 10 + 1))
            idxs = np.random.choice(N, m, replace=False)
            cand[idxs] = np.random.rand(m)
            cand = _repair_sum(cand, target_sum)
            cand_c5 = _compute_c5(cand)
        elif r < 0.12:
            # gradient move targeting worst lag
            # compute self-correlation
            corr_self = np.correlate(current_h, current_h, mode='full')
            min_idx = np.argmin(corr_self)
            shift = min_idx - (N - 1)
            cand = _gradient_move(current_h, shift)
            if cand is None:
                continue
            cand_c5 = _compute_c5(cand)
        else:
            # small transfer move
            i, j = np.random.randint(N, size=2)
            if i == j:
                continue
            max_delta = min(max_move, current_h[i], 1.0 - current_h[j])
            if max_delta <= 0:
                # reverse direction
                max_delta = min(max_move, current_h[j], 1.0 - current_h[i])
                if max_delta <= 0:
                    continue
                i, j = j, i
            delta = np.random.uniform(0.0, max_delta)
            cand = current_h.copy()
            cand[i] -= delta
            cand[j] += delta
            cand_c5 = _compute_c5(cand)

        # Acceptance
        delta_c5 = cand_c5 - current_c5
        if delta_c5 < 0 or np.random.rand() < math.exp(-delta_c5 / max(T, 1e-12)):
            current_h = cand
            current_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = cand.copy()

        # occasional restart if stuck
        if iteration % 5000 == 0 and iteration > 0:
            if best_c5 < current_c5:
                current_h = best_h.copy()
                current_c5 = best_c5
            else:
                # random restart to escape
                new_h = np.random.rand(N)
                np.clip(new_h, 0.0, 1.0, out=new_h)
                new_h = _repair_sum(new_h, target_sum)
                new_c5 = _compute_c5(new_h)
                if new_c5 < current_c5:
                    current_h = new_h
                    current_c5 = new_c5
        iteration += 1

    # final repair
    best_h = _repair_sum(best_h, target_sum)
    final_c5 = _compute_c5(best_h)
    return best_h.tolist(), float(final_c5), len(best_h)

import numpy as np
import time, math, random

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust h in‑place so that sum(h) == target while 0<=h<=1."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Fill or empty using random admissible indices
    if diff > 0:
        while diff > 1e-12:
            idxs = np.where(h < 1.0 - 1e-12)[0]
            if idxs.size == 0:
                break
            i = np.random.choice(idxs)
            inc = min(diff, 1.0 - h[i])
            h[i] += inc
            diff -= inc
    else:
        while diff < -1e-12:
            idxs = np.where(h > 1e-12)[0]
            if idxs.size == 0:
                break
            i = np.random.choice(idxs)
            dec = min(-diff, h[i])
            h[i] -= dec
            diff += dec
    # A final tiny scaling / clipping fix
    cur = h.sum()
    if abs(cur - target) > 1e-9:
        if cur > 0:
            h *= target / cur
        np.clip(h, 0.0, 1.0, out=h)
        diff2 = target - h.sum()
        if abs(diff2) > 1e-12:
            if diff2 > 0:
                i = np.argmin(h)
                h[i] = min(1.0, h[i] + diff2)
            else:
                i = np.argmax(h)
                h[i] = max(0.0, h[i] + diff2)
    return h


def _c5_from_h(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx for the discrete step function."""
    N = len(h)
    # self‑correlation; min_R = min_k Σ h_i h_{i+k}
    min_R = np.min(np.correlate(h, h, mode="full"))
    return float((h.sum() - min_R) * (2.0 / N))


def _gradient_move(h: np.ndarray, shift: int) -> np.ndarray | None:
    """Take a projected‑gradient step that raises the correlation for a given shift."""
    N = len(h)
    s = shift
    g = np.zeros(N)
    if s >= 0:
        g[: N - s] += h[s:]
        g[s:] += h[: N - s]
    else:
        s = -s
        g[s:] += h[: N - s]
        g[: N - s] += h[s:]
    # direction preserving the sum (subtract mean)
    v = g - np.mean(g)
    if np.allclose(v, 0):
        return None
    pos = v > 1e-12
    neg = v < -1e-12
    max_step = np.inf
    if np.any(pos):
        max_step = min(max_step, np.min((1.0 - h[pos]) / v[pos]))
    if np.any(neg):
        max_step = min(max_step, np.min((-h[neg]) / v[neg]))
    if not np.isfinite(max_step) or max_step <= 0:
        return None
    alpha = 0.7 * max_step            # a safe fraction of the feasible step
    h_new = h + alpha * v
    np.clip(h_new, 0.0, 1.0, out=h_new)
    target = N / 2.0
    h_new = _repair_sum(h_new, target)
    return h_new

    random.seed(seed)
    start = time.time()
    total_time = budget_s

    # problem size – a modest resolution that still fits the budget
    N = 200                       # even, gives dx = 0.01
    target_sum = N / 2.0

    h = _repair_sum(h, target_sum)

                    iteration += 1
                    continue
                i, j = j, i
            delta = np.random.uniform(0.0, max_delta)
            cand = current_h.copy()
            cand[i] -= delta
            cand[j] += delta
        # ----------------------------------------------------------
        cand_c5 = _c5_from_h(cand)
        delta_c5 = cand_c5 - current_c5

        # acceptance criterion (simulated annealing)
        if delta_c5 < 0.0 or np.random.rand() < math.exp(-delta_c5 / max(T, 1e-12)):
            current_h = cand
            current_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = cand.copy()

        # occasional random restart if progress stalls
        if iteration % 7000 == 0 and iteration > 0:
            if best_c5 < current_c5:
                current_h = best_h.copy()
                current_c5 = best_c5
            else:
                # gentle reshuffle of the current best
                new_h = best_h.copy()
                idxs = np.random.choice(N, N // 8, replace=False)
                new_h[idxs] = np.random.rand(len(idxs))
                new_h = _repair_sum(new_h, target_sum)
                new_c5 = _c5_from_h(new_h)
                current_h = new_h
                current_c5 = new_c5

        iteration += 1

    # final tidy‑up
    best_h = _repair_sum(best_h, target_sum)
    final_c5 = _c5_from_h(best_h)

    return best_h.tolist(), float(final_c5), len(best_h)
