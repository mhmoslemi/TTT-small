def _subgrad_dir(h):
    n = h.size
    dx = 2.0 / n
    O = np.correlate(h, 1 - h, mode='full') * dx
    O_max = O.max()
    eps = 1e-12
    idxs = np.where(O >= O_max - eps)[0]
    # Convert indices to shifts
    shifts = idxs - (n - 1)
    dir = np.zeros_like(h)
    for s in shifts:
        # shift h by s
        if s >= 0:
            shifted = np.concatenate((np.zeros(s), h[:n - s]))
        else:
            s_abs = -s
            shifted = np.concatenate((h[s_abs:], np.zeros(s_abs)))
        dir += shifted
        # shift by -s
        s2 = -s
        if s2 >= 0:

            s2_abs = -s2
            shifted2 = np.concatenate((h[s2_abs:], np.zeros(s2_abs)))
        dir += shifted2
    return dir

def _project(v, target):
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
```

def _softmax_grad(h, T):
    n = h.size
    dx = 2.0 / n
    # overlaps O
    O = np.correlate(h, 1 - h, mode='full') * dx  # length 2n-1
    Omax = O.max()
    exp_vals = np.exp((O - Omax) / T)
    p = exp_vals / exp_vals.sum()
    # Build matrix P where P[i,j] = p[i-j+offset]
    offset = n - 1
    idx = np.arange(n)[:, None] - np.arange(n)[None, :] + offset
    P = p[idx]  # shape n x n
    grad = dx * (P @ (1 - h) - P.T @ h)
    return grad
```

import time
import numpy as np

def _project(v, target):
    """Project vector v onto the feasible set {0≤x≤1, sum(x)=target}."""
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _evaluate(h):
    """Return C5 = max_k ∫ h(x)(1‑h(x+k)) dx for the discrete h."""
    n = h.size
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode="full") * dx)


def _shift_vec(v, s):
    """Shift vector v by integer s (positive → right) with zero padding."""
    n = v.size
    if s > 0:
        out = np.empty_like(v)
        out[:s] = 0.0
        out[s:] = v[: n - s]
        return out
    elif s < 0:
        s = -s
        out = np.empty_like(v)
        out[: n - s] = v[s:]
        out[n - s :] = 0.0
        return out
    else:
        return v.copy()


def _subgrad_dir(h):
    """Direction = sum_{j∈argmax} (shift(h,j) + shift(h,-j))."""
    n = h.size
    dx = 2.0 / n
    O = np.correlate(h, 1 - h, mode="full") * dx
    O_max = O.max()
    eps = 1e-12
    idxs = np.where(O >= O_max - eps)[0]
    shifts = idxs - (n - 1)
    d = np.zeros_like(h)
    for s in shifts:
        if s == 0:
            d += 2.0 * h
        else:
            d += _shift_vec(h, s) + _shift_vec(h, -s)
    return d


def _softmax_grad(h, T):
    """Gradient of the smoothed max overlap (soft‑max temperature T)."""
    n = h.size
    dx = 2.0 / n
    O = np.correlate(h, 1 - h, mode="full") * dx
    Omax = O.max()
    ex = np.exp((O - Omax) / T)
    p = ex / ex.sum()
    offset = n - 1
    idx = np.arange(n)[:, None] - np.arange(n)[None, :] + offset
    P = p[idx]                     # n×n matrix of probabilities
    grad = dx * (P @ (1 - h) - P.T @ h)
    return grad
