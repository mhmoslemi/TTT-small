import time, numpy as np

def _evaluate(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def _project(v, target):
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def _softmax_grad(h, T):
    n = h.size
    dx = 2.0 / n
    # g vector: correlations
    g = np.correlate(h, 1 - h, mode='full') * dx
    gmax = g.max()
    p = np.exp((g - gmax) / T)
    p /= p.sum()
    idx = (np.arange(n)[:, None] - np.arange(n)[None, :]) + (n - 1)
    P = p[idx]
    grad = dx * (P @ (1 - h) - P.T @ h)
    return grad
```

import time
import numpy as np

# ---------- Helper functions ----------
def _evaluate(h):
    """Maximum overlap C5 for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _project(v, target):
    """Clip to [0,1] and adjust by a uniform shift so that sum(v)=target."""
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _worst_shift(h):
    """Return (k, value, corr) where k gives the maximal overlap."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    idx = int(np.argmax(corr))
    k = idx - (n - 1)          # shift corresponding to idx
    return k, float(corr[idx]), corr


def _subgrad(h):
    """Sub‑gradient of the max‑overlap term (for the worst shift)."""
    n = h.shape[0]
    dx = 2.0 / n
    k, _, _ = _worst_shift(h)
    grad = np.zeros_like(h)

    if k >= 0:
        i = np.arange(n - k)
        j = i + k
        grad[i] += 1.0 - h[j]          # ∂/∂h_i from term h_i*(1‑h_{i+k})
        grad[j] += -h[i]               # ∂/∂h_{i+k} from the same product
    else:
        kk = -k
        i = np.arange(kk, n)
        j = i - kk
        grad[i] += 1.0 - h[j]
        grad[j] += -h[i]

    return grad * dx

    lr = 0.07                     # base step size
    max_it = 25000
    for it in range(max_it):
        if time.time() - start > budget_s * 0.97:
            break

        grad = _subgrad(h)               # sub‑gradient of the worst shift
        # simple back‑tracking line search
        step = lr
        cur_val = _evaluate(h)
        for _ in range(8):
            h_new = _project(h - step * grad, target_sum)
            new_val = _evaluate(h_new)
            if new_val <= cur_val:
                h = h_new
                cur_val = new_val
                break
            step *= 0.5

        # keep global best
        if cur_val < best_c5:
            best_c5 = cur_val
            best_h = h.copy()
            lr = min(lr * 1.05, 1.0)   # accelerate when improving
        else:
            lr = max(lr * 0.98, 5e-5)  # slowly decay otherwise

    final_c5 = _evaluate(best_h)
    return best_h, final_c5, n
