import time
import numpy as np

def _apply_delta(h: np.ndarray, O: np.ndarray, idx: int, delta: float) -> None:
    """Update h and the autocorrelation array O after changing h[idx] by delta.
    O[s] = sum_{i=0}^{n-1-s} h[i] * h[i+s] for s=0..n-1."""
    n = h.shape[0]
    old = h[idx]

    # s = 0 term (square)
    O[0] += 2.0 * old * delta + delta * delta

    # idx as the first index of the pair (i, i+s)
    if idx < n - 1:
        # s = 1 .. n-1-idx
        O[1:n - idx] += delta * h[idx + 1:]

    # idx as the second index of the pair (i-s, i)
    if idx > 0:
        # s = 1 .. idx
        O[1:idx + 1] += delta * h[idx - 1::-1]

    # finally update h
    h[idx] = old + delta


def _project_sum_box(a: np.ndarray, target: float) -> np.ndarray:
    """Project vector a onto the box [0,1]^n while forcing sum(a)=target."""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        p = np.clip(a + mid, 0.0, 1.0)
        s = p.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(a + (lo + hi) * 0.5, 0.0, 1.0)


def _initial_h(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a starting vector of length n."""
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=np.float64)
        x_old = np.linspace(0.0, 2.0, base.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x_new, x_old, base)
    else:
        h = rng.random(n)
    h = _project_sum_box(h, n / 2.0)
    return h


def _autocorr(h: np.ndarray) -> np.ndarray:
    """Full non‑negative lag autocorrelation O[s] = Σ h[i]h[i+s]."""
    n = h.shape[0]
    O = np.empty(n, dtype=np.float64)
    for s in range(n):
        O[s] = np.dot(h[: n - s], h[s:])
    return O


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step function h on [0,2] with a low C5 bound."""
    rng = np.random.default_rng(seed)

    # ---------- simulated‑annealing parameters ----------
    temperature = 0.006
    decay = 0.99985
    step_factor = 0.5
    iter_cnt = 0
    report_interval = 2000

        i = rng.integers(n)

        if j >= i:

            if delta_max <= 1e-12:
