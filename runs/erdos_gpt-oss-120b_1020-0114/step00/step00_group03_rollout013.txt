    corr = np.correlate(h.astype(np.float64), (1 - h).astype(np.float64), mode='full')
    max_corr = corr.max()
    best_h = h.copy()
    best_score = max_corr
    # track best global for maybe later convert to float.
    # Simulated annealing schedule:
    temperature = 1.0
    iteration = 0
    while time.time() - start < budget_s * 0.95:
        # propose swap
        i = np.random.randint(n)
        while h[i] == 0:
            i = np.random.randint(n)
        j = np.random.randint(n)
        while h[j] == 1:
            j = np.random.randint(n)
        # create new h by swapping
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1
        # evaluate
        corr_new = np.correlate(h_new.astype(np.float64), (1 - h_new).astype(np.float64), mode='full')
        max_corr_new = corr_new.max()
        # decide accept
        if max_corr_new < max_corr or np.random.rand() < np.exp(-(max_corr_new - max_corr) / (temperature + 1e-12)):
            h = h_new
            max_corr = max_corr_new
            # record best
            if max_corr_new < best_score:
                best_score = max_corr_new
                best_h = h_new.copy()
        # temperature schedule
        iteration += 1
        if iteration % 1000 == 0:
            # cool
            temperature *= 0.99
    # After loop compute final C5 bound
    dx = 2.0 / n
    c5_bound = best_score * dx
    # return
    h_values = best_h.astype(np.float64)  # convert to float
    return (h_values, c5_bound, n)

def subgradient_descent(h, target_sum, max_iters, time_left):
    n = len(h)
    best_h = h.copy()
    best_val = compute_c5(h)
    t = 0
    start = time.time()
    while t < max_iters and time.time() - start < time_left:
        t += 1
        # compute correlation and find worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        shift = max_idx - (n-1)
        # compute gradient as described:
        g = np.zeros_like(h)
        if shift >= 0:
            i = np.arange(0, n - shift)
            g[i] += 1 - h[i + shift]
            g[i + shift] -= h[i]
        else:
            s = -shift
            i = np.arange(0, n - s)
            g[i] -= h[i + s]
            g[i + s] += 1 - h[i]
        # step size schedule
        step = 0.5 / np.sqrt(t)  # arbitrary
        h = h - step * g
        # projection
        h = proj_capped(h, target_sum)
        # evaluate
        val = compute_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
        # optional early break if val < target threshold (0.3808)
        if best_val <= 0.3808:
            break
    return best_h, best_val

            perturb = np.random.uniform(-0.03, 0.03, n)
            best_h = np.clip(best_h + perturb, 0.0, 1.0)
            best_h = proj_capped(best_h, target_sum)

    # final safety projection and bound computation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = proj_capped(best_h, target_sum)
    final_c5 = compute_c5(best_h)

    return (best_h, final_c5, n)
