def max_corr_fft(h):
    n = h.shape[0]
    L = 1 << ((2*n-1)-1).bit_length()
    F = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1 - h, n=L)
    conv = np.fft.irfft(F * G, n=L)
    conv = conv[:2*n-1]
    return np.max(conv)

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return max_corr_fft(h) * dx

def project_onto_feasible(x):
    x = np.clip(x, 0, 1)
    cur_sum = x.sum()
    target = N
    # if sum too high, need to reduce: reduce from highest values >0.
    if cur_sum > target:
        excess = cur_sum - target
        # find indices where x > 0 (or > eps)
        # reduce proportionally among >0
        mask = x > 0
        total = x[mask].sum()
        # if total==0 cannot reduce; but shouldn't happen
        x[mask] -= excess * x[mask] / total
        # ensure not below zero
        x = np.clip(x, 0, 1)
    elif cur_sum < target:
        deficit = target - cur_sum
        mask = x < 1
        total = (1 - x[mask]).sum()
        x[mask] += deficit * (1 - x[mask]) / total
        x = np.clip(x, 0, 1)
    # Might still not converge due to rounding; iterate a few times
    for _ in range(5):
        cur_sum = x.sum()
        if abs(cur_sum - target) < 1e-9:
            break
        if cur_sum > target:
            excess = cur_sum - target
            mask = x > 0
            total = x[mask].sum()
            if total == 0:
                break
            x[mask] -= excess * x[mask] / total
        else:
            deficit = target - cur_sum
            mask = x < 1
            total = (1 - x[mask]).sum()
            if total == 0:
                break
            x[mask] += deficit * (1 - x[mask]) / total
        x = np.clip(x, 0, 1)
    # final clip
    return x

import numpy as np, time, math, random

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def max_corr(h):
    conv = compute_corr(h)
    return np.max(conv)

def max_corr_and_idx(h):
    conv = compute_corr(h)
    idx = np.argmax(conv)
    max_val = conv[idx]
    k = idx - (len(h) - 1)
    return max_val, k, conv

def subgradient_step(h, step):
    n = len(h)
    max_val, k, _ = max_corr_and_idx(h)
    grad = np.zeros_like(h)
    if k >= 0:
        i = np.arange(0, n - k)
        j = i + k
    else:
        i = np.arange(-k, n)
        j = i + k
    grad[i] += (1 - h[j])
    grad[j] -= h[i]
    # step update
    h_new = h - step * grad
    return h_new

def project_feasible(x, target):
    # Clip to [0,1]
    x = np.clip(x, 0, 1)
    # Water-filling approach
    for _ in range(10):
        cur = x.sum()
        diff = cur - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need reduction
            mask = x > 0
            if mask.sum() == 0:
                break
            dec = diff / mask.sum()
            x[mask] = np.maximum(0, x[mask] - dec)
        else:
            # need increase
            mask = x < 1
            if mask.sum() == 0:
                break
            inc = -diff / mask.sum()
            x[mask] = np.minimum(1, x[mask] + inc)
    return np.clip(x, 0, 1)

def binary_annealing(n, time_limit):
    start = time.time()
    N = n // 2

    # init h
    if 'initial_h_values' in globals():
        # upsample to binary
        init = initial_h_values
        m = len(init)
        x_old = np.linspace(0,2, m, endpoint=False) + (2/m)/2
        x_new = np.linspace(0,2, n, endpoint=False) + (2/n)/2
        vals = np.interp(x_new, x_old, init)
        h = (vals > 0.5).astype(int)
        cur = h.sum()
        if cur > N:
            idx = np.where(h==1)[0]
            np.random.shuffle(idx)
            h[idx[:cur-N]] = 0
        elif cur < N:
            idx = np.where(h==0)[0]
            np.random.shuffle(idx)
            h[idx[:N-cur]] = 1
    else:
        h = np.zeros(n, dtype=int)
        h[:N] = 1
        np.random.shuffle(h)

    cur_c5 = max_corr(h) * (2.0 / n)
    best_h = h.copy()
    best_c5 = cur_c5

    # Simulated annealing parameters
    it = 0
    while time.time() - start < time_limit:
        # pick random 1 and 0 indices
        i = np.random.randint(n)
        while h[i] != 1:
            i = np.random.randint(n)
        j = np.random.randint(n)
        while h[j] != 0:
            j = np.random.randint(n)
        # swap
        h[i] = 0
        h[j] = 1
        new_c5 = max_corr(h) * (2.0 / n)
        # acceptance
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert swap with probability
            # use temperature decreasing linearly
            elapsed = time.time() - start
            rem = time_limit - elapsed
            temp = max(0.01 * (rem / time_limit), 1e-4)
            if np.random.rand() < math.exp(-(new_c5 - cur_c5) / temp):
                cur_c5 = new_c5
                # keep new state
            else:
                # revert
                h[i] = 1
                h[j] = 0

import time
import math
import numpy as np

# ------------------------------------------------------------
# basic correlation utilities
def max_corr_vec(h: np.ndarray) -> float:
    """maximum of h * (1-h) correlation, without dx scaling."""
    conv = np.correlate(h, 1 - h, mode="full")
    return float(conv.max())


def max_corr_and_shift(h: np.ndarray):
    """return max correlation value, the shift (lag) producing it, and full conv."""
    conv = np.correlate(h, 1 - h, mode="full")
    idx = int(conv.argmax())
    max_val = float(conv[idx])
    shift = idx - (len(h) - 1)
    return max_val, shift, conv


# ------------------------------------------------------------
# projection onto the feasible set {0≤h≤1, sum(h)=N}
def project_feasible(x: np.ndarray, target_sum: float) -> np.ndarray:
    """clip to [0,1] then adjust to hit target_sum (water‑filling)."""
    x = np.clip(x, 0.0, 1.0)
    for _ in range(10):
        cur = float(x.sum())
        diff = cur - target_sum
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = x > 1e-12
            if not mask.any():
                break
            dec = diff / mask.sum()
            x[mask] = np.maximum(0.0, x[mask] - dec)
        else:
            mask = x < 1.0 - 1e-12
            if not mask.any():
                break
            inc = (-diff) / mask.sum()
            x[mask] = np.minimum(1.0, x[mask] + inc)
    return np.clip(x, 0.0, 1.0)


# ------------------------------------------------------------
# sub‑gradient step for the worst shift
def subgradient_step(h: np.ndarray, step: float):
    """one sub‑gradient descent step on the worst shift."""
    n = h.shape[0]
    _, k, _ = max_corr_and_shift(h)
    grad = np.zeros_like(h)

    if k >= 0:
        i = np.arange(0, n - k)
        j = i + k
    else:
        i = np.arange(-k, n)
        j = i + k

    grad[i] += (1.0 - h[j])
    grad[j] -= h[i]

    return h - step * grad


# ------------------------------------------------------------
# binary simulated annealing
def binary_annealing(n: int, time_limit: float):
    """search for a good 0/1 vector of length n."""
    start = time.time()
    N = n // 2

    # --- initial vector -------------------------------------------------
    if "initial_h_values" in globals():
        init = initial_h_values
        m = len(init)
        # linear interpolation onto the new grid
        xs_old = np.linspace(0.0, 2.0, m, endpoint=False) + (2.0 / m) / 2.0
        xs_new = np.linspace(0.0, 2.0, n, endpoint=False) + (2.0 / n) / 2.0
        vals = np.interp(xs_new, xs_old, init)
        h = (vals > 0.5).astype(int)
        cur = int(h.sum())
        if cur > N:
            idx = np.where(h == 1)[0]
            np.random.shuffle(idx)
            h[idx[: cur - N]] = 0
        elif cur < N:
            idx = np.where(h == 0)[0]
            np.random.shuffle(idx)
            h[idx[: N - cur]] = 1
    else:
        h = np.zeros(n, dtype=int)
        h[:N] = 1
        np.random.shuffle(h)

    cur_c5 = max_corr_vec(h) * (2.0 / n)
    best_h = h.copy()
    best_c5 = cur_c5

    it = 0
    while time.time() - start < time_limit:
        # pick random 1 and 0 positions
        i = np.random.randint(n)
        while h[i] != 1:
            i = np.random.randint(n)
        j = np.random.randint(n)
        while h[j] != 0:
            j = np.random.randint(n)

        # perform the swap
        h[i] = 0
        h[j] = 1
        new_c5 = max_corr_vec(h) * (2.0 / n)

        if new_c5 < cur_c5:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # simulated‑annealing acceptance
            elapsed = time.time() - start
            remaining = max(time_limit - elapsed, 0.0)
            temp = max(0.01 * (remaining / time_limit), 1e-5)
            if np.random.rand() < math.exp(-(new_c5 - cur_c5) / temp):
                cur_c5 = new_c5
            else:
                # revert
                h[i] = 1
                h[j] = 0

        it += 1
        # occasional diversification
        if it % 5000 == 0 and cur_c5 - best_c5 > 1e-4:
            h = best_h.copy()
            # apply a few random swaps to escape plateaus
            for _ in range(10):
                i2 = np.random.choice(np.where(h == 1)[0])
                j2 = np.random.choice(np.where(h == 0)[0])
                h[i2] = 0
                h[j2] = 1
            cur_c5 = max_corr_vec(h) * (2.0 / n)

    return best_h.astype(float), best_c5


# ------------------------------------------------------------
# continuous refinement using sub‑gradient steps
def continuous_refine(h_start: np.ndarray, time_limit: float):
    start = time.time()
    n = h_start.shape[0]
    N = n // 2

    h = project_feasible(h_start.copy(), float(N))
    cur_c5 = max_corr_vec(h) * (2.0 / n)

    best_h = h.copy()
    best_c5 = cur_c5

    it = 0
    while time.time() - start < time_limit:
        # linear schedule for step size
        frac = (time.time() - start) / time_limit
        step = 0.3 * (1.0 - frac)
        step = max(step, 1e-4)

        h_new = subgradient_step(h, step)
        h_new = project_feasible(h_new, float(N))
        new_c5 = max_corr_vec(h_new) * (2.0 / n)

        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()

        if new_c5 < cur_c5:
            h = h_new
            cur_c5 = new_c5

        it += 1
        # occasional small random perturbations to avoid stagnation
        if it % 2000 == 0:
            if cur_c5 - best_c5 > 1e-5:
                h = best_h.copy()
                # tiny Gaussian noise then re‑project
                noise = np.random.normal(scale=0.02, size=n)
                h = np.clip(h + noise, 0.0, 1.0)
                h = project_feasible(h, float(N))
                cur_c5 = max_corr_vec(h) * (2.0 / n)

    return best_h, best_c5


# ------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Main entry point – returns (h_values, c5_bound, n_points)."""
    np.random.seed(seed)
    total_start = time.time()
    # choose a moderate grid size (even)
    n = 200

    # split the time budget
    remaining = budget_s - (time.time() - total_start)
    binary_time = max(10.0, remaining * 0.55)
    cont_time = max(5.0, remaining * 0.35)

    # 1) binary simulated annealing
    h_bin, c5_bin = binary_annealing(n, binary_time)

    # 2) continuous refinement
    h_cont, c5_cont = continuous_refine(h_bin, cont_time)

    # pick the best of the two phases
    if c5_cont < c5_bin:
        best_h = h_cont
        best_c5 = c5_cont
    else:
        best_h = h_bin
        best_c5 = c5_bin

    # final safety projection (ensures feasibility)
    best_h = project_feasible(best_h, float(n // 2))
    best_c5 = max_corr_vec(best_h) * (2.0 / n)

    return best_h, float(best_c5), int(n)
