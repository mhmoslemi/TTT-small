def _evaluate_c5(h, dx):
    """Return max_{k} ∫ h(x)(1-h(x+k))dx using discrete correlation."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _project_onto_box_sum(v, target, max_iter=60, tol=1e-9):
    """
    Project vector v onto {x : sum(x)=target, 0<=x<=1}
    using the water-filling (clipping after adding a scalar λ) method.
    """
    lo, hi = -1.0, 1.0
    # Expand bounds until they bracket the solution
    while np.clip(v + lo, 0.0, 1.0).sum() > target:
        lo *= 2.0
    while np.clip(v + hi, 0.0, 1.0).sum() < target:
        hi *= 2.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def _neighbors_sum(h, k):
    """Return array S_i = h_{i+k}+h_{i-k} with zero padding."""

            can_inc = h_cur < 1.0 - 1e-12
            can_dec = h_cur > 1e-12
            if not np.any(can_inc) or not np.any(can_dec):
                break
            i_candidates = np.where(can_inc)[0]
            j_candidates = np.where(can_dec)[0]
            i = i_candidates[np.argmax(S[i_candidates])]
            j = j_candidates[np.argmin(S[j_candidates])]
            if i == j:
                # fallback to a random move if degeneracy occurs
                i, j = rng.integers(0, N, size=2, endpoint=False)
                while i == j:
                    j = rng.integers(0, N, endpoint=False)
                delta_low = -min(h_cur[j], 1.0 - h_cur[i])
                delta_high = min(h_cur[i], 1.0 - h_cur[j])
                if delta_low >= delta_high:
                    continue
                delta = rng.uniform(delta_low, delta_high)
            else:
                # amount we can safely move from j to i
                delta_max = min(1.0 - h_cur[i], h_cur[j])
                if delta_max <= 0:
                    # should not happen, but guard
                    continue
                # take a random fraction of the maximal feasible transfer
                delta = delta_max * rng.random()

        # --------------------------------------------------------
        # Apply the transfer.
        # --------------------------------------------------------
        h_new = h_cur.copy()
        h_new[i] += delta
        h_new[j] -= delta

        # --------------------------------------------------------
        # Evaluate the new configuration.
        # --------------------------------------------------------
        new_c = _evaluate_c5(h_new, dx)

        # --------------------------------------------------------
        # Accept if improvement, otherwise maybe accept probabilistically.
        # --------------------------------------------------------
        if new_c < cur_c:
            h_cur = h_new
            cur_c = new_c
            no_improve = 0
            if new_c < best_c:
                best_c = new_c
                best_h = h_new.copy()
        else:
            # Simulated‑annealing style occasional uphill move.
            elapsed = time.time() - start_time
            frac = elapsed / budget_s
            # temperature linearly decays from 0.04 to 0
            T = max(0.04 * (1.0 - frac), 1e-7)
            if rng.random() < math.exp(-(new_c - cur_c) / T):
                h_cur = h_new
                cur_c = new_c
            no_improve += 1

        total_iters += 1

        # --------------------------------------------------------
        # Periodic restart to escape deep local minima.
        # --------------------------------------------------------
        if no_improve >= max_no_improve:
            # start a new search from the best found so far, with a small perturbation
            noise = rng.normal(scale=0.03, size=N)
            h_restart = best_h + noise
            h_restart = np.clip(h_restart, 0.0, 1.0)
            h_restart = _project_onto_box_sum(h_restart, target_sum)
            h_cur = h_restart
            cur_c = _evaluate_c5(h_cur, dx)
            no_improve = 0

    # ------------------------------------------------------------
    # Return the best solution discovered.
    # ------------------------------------------------------------
    return best_h.tolist(), float(best_c), N
