import numpy as np
import cvxpy as cp
import time

def solve_milp_binary(n):
    h = cp.Variable(n, boolean=True)   # 0/1
    t = cp.Variable()   # scalar, non-negative
    constraints = []
    # sum h = n/2
    constraints.append(cp.sum(h) == n/2)
    # t >=0 maybe
    constraints.append(t >= 0)
    # For each shift k
    for k in range(-(n-1), n):
        # overlapping indices:
        i_start = max(0, -k)
        i_end = min(n-1, n-1-k)   # inclusive? Actually last index i such that i+k <= n-1
        # indices range i_start..i_end inclusive
        if i_start > i_end:
            continue
        indices_i = np.arange(i_start, i_end+1)
        indices_j = indices_i + k
        L = len(indices_i)
        # define w variables
        w = cp.Variable(L, boolean=True)
        # constraints linking w = h_i & h_j
        # w <= h_i
        constraints.append(w <= h[indices_i])
        # w <= h_j
        constraints.append(w <= h[indices_j])
        # w >= h_i + h_j - 1
        constraints.append(w >= h[indices_i] + h[indices_j] - 1)
        # compute a_k = sum_i h_i - sum w
        # note: sum_i h_i over region = sum of h_i for indices_i
        a_k = cp.sum(h[indices_i]) - cp.sum(w)
        constraints.append(a_k <= t)
    # objective: minimize t
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.GLPK_MI, verbose=False)   # maybe solver=cp.CBC
    # Check status
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        return None, None
    h_vals = np.round(h.value).astype(float)
    t_val = t.value
    return h_vals, t_val

def binary_hill_climb(h, budget_seconds, current_best_val):
    n = len(h)
    dx = 2.0/n
    best = h.copy()
    best_val = current_best_val
    # compute indices of ones and zeros
    ones = [i for i in range(n) if best[i] > 0.5]
    zeros = [i for i in range(n) if best[i] < 0.5]
    start = time.time()
    while time.time() - start < budget_seconds:
        if not ones or not zeros:
            break
        i = random.choice(ones)
        j = random.choice(zeros)
        # swap
        cand = best.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        val = np.max(np.correlate(cand, 1 - cand, mode='full') * dx)
        if val < best_val - 1e-9:
            best = cand
            best_val = val
            # update ones/zeros lists accordingly
            ones.remove(i); ones.append(j)
            zeros.remove(j); zeros.append(i)
            # early exit if target reached
            if best_val <= target:
                break
        # else maybe accept? We'll just continue
    return best, best_val

    best_h = None
    best_val = np.inf
    start = time.time()
    dx = 2.0 / n
    while time.time() - start < time_limit:
        # generate random binary vector with n/2 ones
        h = np.zeros(n)
        idx = np.random.choice(n, n//2, replace=False)
        h[idx] = 1.0
        val = np.max(np.correlate(h, 1 - h, mode='full') * dx)
        if val < best_val:
            best_val = val
            best_h = h
        # maybe try hill climb from this
        # limit time for each climb small
        remaining = time_limit - (time.time() - start)
        if remaining < 0.1:

        # run hill climb for a short duration
        h_opt, val_opt = binary_hill_climb(h, min(0.5, remaining), best_val)

            best_val = val_opt
            best_h = h_opt

def solve_binary_milp(n, time_left):
    import cvxpy as cp
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == n/2, t >= 0]
    for k in range(-(n-1), n):
        i_start = max(0, -k)
        i_end = min(n-1, n-1-k)
        if i_start > i_end:
            continue
        idx_i = np.arange(i_start, i_end+1)
        idx_j = idx_i + k
        L = len(idx_i)
        w = cp.Variable(L, boolean=True)
        constraints.append(w <= h[idx_i])
        constraints.append(w <= h[idx_j])
        constraints.append(w >= h[idx_i] + h[idx_j] - 1)
        constraints.append(cp.sum(h[idx_i]) - cp.sum(w) <= t)
    prob = cp.Problem(cp.Minimize(t), constraints)
    solvers = [cp.GLPK_MI, cp.ECOS_BB]
    for s in solvers:
        try:
            prob.solve(solver=s, verbose=False, warm_start=False, max_iters=100000)
            if prob.status in ("optimal", "optimal_inaccurate"):
                h_val = np.rint(h.value).astype(float)
                t_val = float(t.value)
                return h_val, t_val
        except Exception:
            continue
    return None, None
```

def simulated_annealing(h_init, dx, time_limit, target=None):
    rng = np.random.default_rng()
    h = h_init.copy()
    cur_val = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_h = h.copy()
    best_val = cur_val
    n = len(h)
    start = time.time()
    # Base temperature
    T0 = 0.02
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        T = T0 * (1 - elapsed / time_limit)
        # choose two distinct indices i (donor), j (receiver)
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # donor: i (will decrease), receiver: j (increase)
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0:
            # try opposite direction
            max_move = min(h[j], 1.0 - h[i])
            if max_move <= 0:
                continue
            # swap roles
            i, j = j, i
        delta = rng.random() * max_move * 0.5  # take part of max
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_val = np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx
        # acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12)):
            h = h_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h_new.copy()
                if target is not None and best_val <= target:
                    break
    return best_h, best_val

    start = time.time()

    random.seed(seed)

    # Determine n_points

        if h_bin is not None:

            if val < best_val:
                best_h = h_bin
                best_val = val
    except Exception:
        pass

    # If initial solution available and better
    if init_h is not None:
        val0 = np.max(np.correlate(init_h, 1 - init_h, mode='full')) * dx
        if val0 < best_val:
            best_h = init_h.copy()
            best_val = val0

    def random_binary():
        h = np.zeros(n)
        idx = rng.choice(n, n//2, replace=False)
        h[idx] = 1.0
        return h

    # If best_h is None, start from random binary
    if best_h is None:
        best_h = random_binary()
        best_val = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    # Now run simulated annealing starting from best_h
    remaining_time = budget_s - (time.time() - start)
    if remaining_time > 0:
        target = 0.3808
        best_h, best_val = simulated_annealing(best_h, dx, remaining_time, target=target)

    # Ensure feasibility: clip and adjust sum
    best_h = np.clip(best_h, 0.0, 1.0)
    desired = n / 2.0
    sum_h = best_h.sum()
    diff = desired - sum_h
    # redistribute diff uniformly within bounds
    if abs(diff) > 1e-9:
        # adjust using simple projection: add/subtract small epsilon to first element not at bound
        for _ in range(1000):
            if abs(diff) < 1e-9:
                break
            # find indices that can be increased or decreased

                # need increase sum
                idx = np.where(best_h < 1.0 - 1e-12)[0]

                    break
                inc = min(diff/len(idx), 0.5)

                    break

            sum_h = best_h.sum()
            diff = desired - sum_h

    final_val = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_val = float(final_val)

import time, random, math
import numpy as np

def _solve_binary_milp(n, time_left):
    """Try to solve the binary version of the problem (h_i ∈ {0,1}) via MILP.
    Returns (h_vals, t_val) where h_vals is a 0/1 array and t_val = max_k a_k.
    """
    try:
        import cvxpy as cp
    except Exception:
        return None, None

    # Decide which solvers are available
    solvers = []
    for name in ["GLPK_MI", "CBC", "SCIP", "ECOS_BB"]:
        if hasattr(cp, name):
            solvers.append(getattr(cp, name))
    if not solvers:
        return None, None

    # Variables
    h = cp.Variable(n, integer=True)
    t = cp.Variable()
    constr = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n / 2,
        t >= 0,
    ]

    # For every possible shift k
    for k in range(-(n - 1), n):
        i_start = max(0, -k)
        i_end = min(n - 1, n - 1 - k)
        if i_start > i_end:
            continue
        idx_i = np.arange(i_start, i_end + 1)
        idx_j = idx_i + k
        L = len(idx_i)
        w = cp.Variable(L, integer=True)
        constr += [
            w >= 0,
            w <= 1,
            w <= h[idx_i],
            w <= h[idx_j],
            w >= h[idx_i] + h[idx_j] - 1,
            cp.sum(h[idx_i]) - cp.sum(w) <= t,
        ]

    prob = cp.Problem(cp.Minimize(t), constr)

    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False, warm_start=False)
            if prob.status in ("optimal", "optimal_inaccurate"):
                hv = np.rint(h.value).astype(float)
                tv = float(t.value)
                return hv, tv
        except Exception:
            continue
    return None, None


def _simulated_annealing(h_start, dx, time_limit, target=None):
    """Simple SA that preserves the sum of h and keeps 0≤h≤1."""
    rng = np.random.default_rng()
    h = h_start.copy()
    cur_val = np.max(np.correlate(h, 1.0 - h, mode="full")) * dx
    best_h = h.copy()
    best_val = cur_val
    n = len(h)
    start = time.time()
    T0 = 0.02  # initial temperature

    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= time_limit:
            break
        # linear cooling
        T = T0 * (1.0 - elapsed / time_limit)
        if T < 1e-8:
            T = 1e-8

        # pick a donor and a receiver
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # try moving mass from i to j
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0:
            # opposite direction
            max_move = min(h[j], 1.0 - h[i])
            if max_move <= 0:
                continue
            i, j = j, i
        # propose a delta (biased to be small)
        delta = rng.random() * max_move * 0.5
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_val = np.max(np.correlate(h_new, 1.0 - h_new, mode="full")) * dx

        # acceptance
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / T):
            h = h_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h_new.copy()
                if target is not None and best_val <= target:
                    break
    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    """Find a step function h on [0,2] with ∫h=1 that minimizes
    max_k ∫ h(x)(1−h(x+k))dx. Returns (h_values, c5_bound, n_points)."""
    start_time = time.time()
    rng = np.random.default_rng(seed)
    random.seed(seed)

    # Determine problem size
    if "initial_h_values" in globals():
        init_h = np.array(initial_h_values, dtype=float)
        n = init_h.size
    else:
        n = 42
        init_h = None

    dx = 2.0 / n
    best_h = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 1) Attempt exact binary MILP (fast for n~40)
    # ------------------------------------------------------------------
    time_spent = time.time() - start_time
    time_left = max(0.0, budget_s - time_spent - 5.0)  # keep a few seconds for SA
    if time_left > 0 and n % 2 == 0:
        h_bin, t_val = _solve_binary_milp(n, time_left)
        if h_bin is not None:
            val = np.max(np.correlate(h_bin, 1.0 - h_bin, mode="full")) * dx
            best_h = h_bin
            best_val = float(val)

    # ------------------------------------------------------------------
    # 2) Use any provided initial construction
    # ------------------------------------------------------------------
    if init_h is not None:
        val0 = np.max(np.correlate(init_h, 1.0 - init_h, mode="full")) * dx
        if val0 < best_val:
            best_h = init_h.copy()
            best_val = float(val0)

    # ------------------------------------------------------------------
    # 3) If still no feasible solution, create a random binary start
    # ------------------------------------------------------------------
    if best_h is None:
        h0 = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h0[idx] = 1.0
        best_h = h0
        best_val = float(np.max(np.correlate(h0, 1.0 - h0, mode="full")) * dx)

    # ------------------------------------------------------------------
    # 4) Continuous simulated annealing refinement
    # ------------------------------------------------------------------
    time_spent = time.time() - start_time
    time_left = max(0.0, budget_s - time_spent)
    if time_left > 0.5:
        target_bound = 0.3808  # stop early if we reach a good bound
        best_h, best_val = _simulated_annealing(best_h, dx, time_left, target=target_bound)

    # ------------------------------------------------------------------
    # 5) Final feasibility correction (tiny rounding adjustments)
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    desired_sum = n / 2.0
    cur_sum = best_h.sum()
    diff = desired_sum - cur_sum
    if abs(diff) > 1e-9:
        # simple projection: distribute the error uniformly respecting bounds
        for _ in range(1000):
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                idx = np.where(best_h < 1.0 - 1e-12)[0]
                if idx.size == 0:
                    break
                inc = min(diff / len(idx), 0.5)
                best_h[idx] += inc
            else:
                idx = np.where(best_h > 1e-12)[0]
                if idx.size == 0:
                    break
                dec = min(-diff / len(idx), 0.5)
                best_h[idx] -= dec
            cur_sum = best_h.sum()
            diff = desired_sum - cur_sum
        best_h = np.clip(best_h, 0.0, 1.0)

    final_val = float(np.max(np.correlate(best_h, 1.0 - best_h, mode="full")) * dx)
    best_val = final_val

    return (best_h.tolist(), best_val, int(n))
