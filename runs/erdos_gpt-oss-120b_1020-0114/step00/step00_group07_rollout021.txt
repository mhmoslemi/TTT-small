def _compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1‑h(x+k))dx for the discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

        remain_frac = max(0.0, (budget_s - elapsed) / budget_s)
        step = max(step_min, step_initial * remain_frac)

        delta = min(step, h_cur[i], 1.0 - h_cur[j])
        if delta <= 0:
            continue

        h_new = h_cur.copy()
        h_new[i] -= delta
        h_new[j] += delta

        new_c5 = _compute_c5(h_new)

        # keep the best ever found
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            if best_c5 <= 0.3808:               # target reached – stop early
                break

        # accept improvements, otherwise count a failure
        if new_c5 < cur_c5:
            h_cur = h_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                # restart near the current best
                h_cur = best_h.copy()
                # small random jitter to escape plateaus
                a, b = rng.integers(n, size=2)
                eps = step * rng.random()
                if h_cur[a] >= eps and h_cur[b] <= 1.0 - eps:
                    h_cur[a] -= eps
                    h_cur[b] += eps
                no_improve = 0

    # --------------------------------------------------------------
    # final result
    # --------------------------------------------------------------
    return best_h.tolist(), float(best_c5), int(n)
