def project_box_simplex(v, S, tol=1e-9):
    lo = -1e6
    hi = 1e6
    for _ in range(100):
        lam = 0.5*(lo+hi)
        x = np.clip(v - lam, 0, 1)
        s = x.sum()
        if abs(s - S) < tol:
            return x
        if s > S:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0, 1)

def compute_c5_and_max_shifts(h, dx):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_val = np.max(corr)
    # Determine shift indices (zero-based) that achieve max (within tolerance)

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return corr.max() * dx
def max_lags(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = corr.max()
    idxs = np.where(corr == max_val)[0]
    lags = idxs - (n - 1)
    return lags
def project_box_simplex(v, target_sum, tol=1e-12, max_iter=100):
    # binary search lambda
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if abs(s - target_sum) <= tol:
            return x
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    # final
    x = np.clip(v - lam, 0.0, 1.0)
    # adjust small difference by scaling remainder:
    diff = target_sum - x.sum()
    if abs(diff) > 1e-9:
        # distribute diff proportionally among non-saturated indices
        mask = (x > 0.0) & (x < 1.0)
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def optimize_h(initial_h, time_budget, start_time):
    h = np.array(initial_h, copy=True)
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    step = 0.5  # initial step magnitude
    # ensure feasibility (project)
    h = project_box_simplex(h, target)
    while time.time() - start_time < time_budget:
        # compute gradient
        grads = gradient_max_lags(h)
        improved = False
        # try decreasing step scales
        for factor in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            new_h = project_box_simplex(h - step * factor * grads, target)

            h = project_box_simplex(h + perturb, target)
            step = 0.1

        if step < 1e-9:

import numpy as np
import time

# ---------- basic tools -------------------------------------------------

def compute_c5(h):
    """Compute the Erdős C5 value for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def max_lags(h, tol=1e-12):
    """Return all integer lags (relative shifts) that attain the maximal correlation."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = corr.max()
    idxs = np.where(corr >= max_val - tol)[0]
    return idxs - (n - 1)


def project_box_simplex(v, target_sum, tol=1e-12, max_iter=60):
    """
    Euclidean projection of vector v onto the intersection of the box [0,1]^n
    and the affine hyperplane sum(x)=target_sum.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = -np.max(v) - 1.0      # ensures all entries become 1 after shift
    hi = np.max(v) + 1.0       # ensures all entries become 0 after shift
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if abs(s - target_sum) <= tol:
            return x
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    # final projection (tiny residual error is tolerated)
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-9:
        mask = (x > 0.0) & (x < 1.0)
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
    return x


def gradient_max_lags(h):
    """Sub‑gradient of the maximal correlation term (averaged over all maximisers)."""
    n = h.shape[0]
    grads = np.zeros_like(h)
    lags = max_lags(h)
    for ell in lags:
        term1 = np.zeros(n)
        term2 = np.zeros(n)
        if ell >= 0:
            # (1 - h[i+ell]) part
            term1[: n - ell] = 1.0 - h[ell:]
            # h[i-ell] part
            term2[ell:] = h[: n - ell]
        else:
            d = -ell
            term1[d:] = 1.0 - h[: n - d]
            term2[: n - d] = h[d:]
        grads += term1 - term2
    grads /= max(1, len(lags))
    return grads


def upsample_h(h, n_new):
    """Linear interpolation of a step vector to a finer grid, followed by feasibility projection."""
    n_old = h.shape[0]
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = project_box_simplex(h_new, n_new / 2.0)
    return h_new


# ---------- optimisation driver -----------------------------------------

def optimise_h(initial_h, time_budget, start_time):
    """
    Simple projected sub‑gradient descent with back‑tracking.
    Returns the best vector found within the allotted time.
    """
    h = project_box_simplex(initial_h, initial_h.shape[0] / 2.0)
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_val = compute_c5(best_h)

    step = 0.5  # initial step length (in h‑space)

    while time.time() - start_time < time_budget:
        grad = gradient_max_lags(h)
        improved = False

        # try a few step scalings
        for factor in (1.0, 0.5, 0.25, 0.125, 0.0625):
            cand = project_box_simplex(h - step * factor * grad, target)
            cand_val = compute_c5(cand)
            if cand_val < best_val - 1e-12:
                h = cand
                best_h = cand.copy()
                best_val = cand_val
                step *= 1.1
                improved = True
                break

        if not improved:
            step *= 0.5

        # occasional random shake to escape flat regions
        if step < 1e-9:
            perturb = 0.02 * (np.random.rand(n) - 0.5)
            h = project_box_simplex(h + perturb, target)
            step = 0.1

        if step < 1e-12:
            break

    return best_h, best_val


# ---------- main entry point --------------------------------------------

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points).
    The algorithm refines the supplied initial construction (if any)
    and also tries a doubled resolution.
    """
    np.random.seed(seed)
    start = time.time()

    # obtain the parent construction if present
    if "initial_h_values" in globals():
        base_h = np.array(initial_h_values, dtype=float)
    else:
        # fallback: simple uniform half‑density
        base_h = np.full(45, 0.5, dtype=float)

    n0 = base_h.shape[0]
    candidates = []

    # list of grid sizes to investigate (increase if time permits)
    grid_sizes = [n0]
    elapsed = time.time() - start
    if elapsed < budget_s * 0.7:
        grid_sizes.append(2 * n0)

    # allocate time roughly equally among the grid sizes
    for n in grid_sizes:
        remaining = budget_s - (time.time() - start)
        if remaining <= 0:
            break
        # proportion of remaining time for this grid
        per_grid_budget = remaining / len(grid_sizes)

        if n == n0:
            init = base_h.copy()
        else:
            init = upsample_h(base_h, n)

        # main optimisation from the (interpolated) start point
        h_opt, c5_opt = optimise_h(init, per_grid_budget * 0.6, time.time())
        candidates.append((c5_opt, h_opt, n))

        # a couple of random restarts to explore the landscape
        for _ in range(2):
            if time.time() - start >= budget_s:
                break
            rand_start = project_box_simplex(np.random.rand(n), n / 2.0)
            h_rand, c5_rand = optimise_h(rand_start, per_grid_budget * 0.2, time.time())
            candidates.append((c5_rand, h_rand, n))

    # pick the best discovered solution
    if not candidates:
        # very unlikely, but ensure a return
        best_h = base_h
        best_c5 = compute_c5(best_h)
        best_n = n0
    else:
        best_c5, best_h, best_n = min(candidates, key=lambda t: t[0])

    return best_h, float(best_c5), int(best_n)
