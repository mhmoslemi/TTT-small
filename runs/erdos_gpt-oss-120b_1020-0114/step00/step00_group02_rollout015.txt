    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def _shift(arr, k):
    n = arr.shape[0]
    if k > 0:
        out = np.empty_like(arr)
        out[:k] = 0.0
        out[k:] = arr[:n - k]
        return out
    elif k < 0:
        k = -k
        out = np.empty_like(arr)
        out[:n - k] = arr[k:]
        out[n - k:] = 0.0
        return out
    else:
        return arr.copy()

import time
import math
import numpy as np

def _c5_of(h: np.ndarray) -> float:
    """Return the C5 value for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _shift(arr: np.ndarray, k: int) -> np.ndarray:
    """Shift array by k with zero padding (positive = right shift)."""
    n = arr.shape[0]
    if k > 0:
        out = np.empty_like(arr)
        out[:k] = 0.0
        out[k:] = arr[: n - k]
        return out
    if k < 0:
        k = -k
        out = np.empty_like(arr)
        out[: n - k] = arr[k:]
        out[n - k :] = 0.0
        return out
    return arr.copy()
