import numpy as np, time, math

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = corr.max()
    dx = 2.0 / len(h)
    return max_corr * dx

def gradient_max_shift(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = corr.max()
    shift_indices = np.where(corr == max_val)[0] - (n - 1)
    g = np.zeros(n)
    for s in shift_indices:
        g += np.ones(n)
        if s >= 0:
            if n - s > 0:
                g[:n - s] -= h[s:]
                g[s:]   -= h[:n - s]
        else:
            t = -s
            if n - t > 0:
                g[t:]   -= h[:n - t]
                g[:n - t] -= h[t:]
    g /= len(shift_indices)
    return g

def project_capsimplex(v, target):
    # find lambda such that sum clip(v+lambda,0,1) == target
    lo = -2.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + hi, 0.0, 1.0)

def optimize_one_run(n, budget, start_time, initial_h=None):

import time
import numpy as np

def _dx(n: int) -> float:
    return 2.0 / n

def _c5_value(h: np.ndarray) -> float:
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * _dx(len(h))

def _gradient_of_max(h: np.ndarray) -> np.ndarray:
    """sub‑gradient of  max_k  Σ h_i(1‑h_{i+k})  (linear correlation)."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = corr.max()
    shifts = np.where(corr == mx)[0] - (n - 1)   # index → shift
    g = np.zeros(n)
    for s in shifts:
        gs = np.ones(n)
        if s >= 0:
            if n - s:
                gs[: n - s] -= h[s:]
                gs[s:]     -= h[: n - s]
        else:
            t = -s
            if n - t:
                gs[t:]     -= h[: n - t]
                gs[: n - t] -= h[t:]
        g += gs
    return g / len(shifts)

def _project_capsimplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto {0≤x_i≤1 , Σx_i=target_sum}  (water‑filling)."""
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v + hi, 0.0, 1.0)

def _upsample(initial: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a coarse vector to length n."""
    n0 = len(initial)
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    return np.interp(x, x0, initial)

def _local_optimize(h0: np.ndarray, budget: float, start: float) -> tuple:
    """Gradient descent with simple step‑size schedule."""
    n = len(h0)
    target = n * 0.5
    h = _project_capsimplex(h0.copy(), target)
    best_h, best_c = h.copy(), _c5_value(h)
    step = 0.5
    it = 0
    while True:
        now = time.time()
        if now - start > budget:
            break
        if it >= 2000:
            break
        it += 1
        grad = _gradient_of_max(h)
        # one gradient step
        cand = _project_capsimplex(h - step * grad, target)
        cand_c = _c5_value(cand)
        if cand_c < best_c:
            best_c, best_h = cand_c, cand.copy()
        # accept if improvement, otherwise keep old
        if cand_c < _c5_value(h):
            h = cand
            step *= 0.99
        else:
            step *= 1.02
        # occasional mild random shake to escape flat regions
        if it % 100 == 0:
            for _ in range(3):
                i, j = np.random.randint(0, n, 2)
                if i == j:
                    continue
                delta = min(0.04, h[i], 1.0 - h[j])
                if delta > 0:
                    h[i] -= delta
                    h[j] += delta
            h = _project_capsimplex(h, target)
    return best_h, best_c

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Return a step function h (as a list), the resulting C5 bound,
    and the number of sample points used.
    """
    np.random.seed(seed)
    start_time = time.time()
    # sample size – a moderate grid gives a good trade‑off
    n = 180                       # 2·90 = 1 (integral constraint)
    target_sum = n * 0.5

    # initialise from the provided construction if present
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
        if len(init) != n:
            init = _upsample(init, n)
    else:
        init = np.random.rand(n)

    # keep the best solution found within the time budget
    best_h, best_c = _local_optimize(init, budget_s * 0.9, start_time)

    # a short final polishing phase with a smaller step size
    final_h, final_c = _local_optimize(best_h, budget_s * 0.1, start_time)
    if final_c < best_c:
        best_h, best_c = final_h, final_c

    # final safety projection and recomputation
    best_h = _project_capsimplex(best_h, target_sum)
    best_c = _c5_value(best_h)

    # return list for the harness, the bound, and n
    return (best_h.tolist(), float(best_c), n)
