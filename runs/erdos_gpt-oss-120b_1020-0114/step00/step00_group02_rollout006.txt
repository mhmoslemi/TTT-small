            noise -= noise.mean()

        neg = d < 0
        step_allowed = np.full_like(d, np.inf)
        step_allowed[pos] = (1 - h)[pos] / d[pos]
        step_allowed[neg] = -h[neg] / d[neg]
        max_step = step_allowed.min()

            noise = np.random.randn(n) * 0.01
            noise -= noise.mean()
            h = project_sum_bounds(h + noise)
            no_improve += 1
            continue

        h_new = h + alpha * d
        # evaluate new C
        c5_new = compute_c5(h_new)
        # accept if not worse
        if c5_new <= current_c5 + 1e-12:
            h = h_new
            # adjust step_base loosely
            step_base = max(step_base * 0.9, 1e-6)
            no_improve = 0
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            # reject, reduce step
            step_base *= 0.5
            no_improve += 1

        # occasional random perturb to escape local minima
        if no_improve >= 300:
            noise = np.random.randn(n) * 0.03
            noise -= noise.mean()
            h = project_sum_bounds(h + noise)
            no_improve = 0
            step_base = 0.8

    # final projection
    best_h = project_sum_bounds(best_h)
    final_c5 = compute_c5(best_h)

    return (best_h, float(final_c5), n)
