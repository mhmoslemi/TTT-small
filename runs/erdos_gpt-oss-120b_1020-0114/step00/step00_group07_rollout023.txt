import time, math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1‑h(x+k))dx for a step function sampled on a uniform grid."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation without wrap‑around; np.correlate implements the desired sum.
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _random_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Binary (0/1) vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h


def _anneal(start_h: np.ndarray, time_budget: float, rng: np.random.Generator):
    """Simple simulated‑annealing on binary vectors preserving the sum constraint."""
    n = start_h.shape[0]
    best_h = start_h.copy()
    best_c = _compute_c5(best_h)

    cur_h = best_h.copy()
    cur_c = best_c

    start = time.time()
    # temperature schedule (exponential decay)
    T0, Tf = 0.5, 1e-4
    it = 0
    no_improve = 0

    while time.time() - start < time_budget:
        it += 1

        frac = (time.time() - start) / time_budget
        temp = T0 * (Tf / T0) ** frac

            j += 1

        idx = rng.integers(n)
        correction = (1.0 - total) / dx
        new_val = np.clip(best_h[idx] + correction, 0.0, 1.0)
        best_h[idx] = new_val

    final_c = _compute_c5(best_h)

    return best_h, final_c, n
