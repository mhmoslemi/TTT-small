    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    d = idx - (n - 1)          # lag producing the max

    g = np.zeros_like(h)

    if d >= 0:                 # forward lag
        # contribution from first factor (1 - h[i+d])
        g[: n - d] += (1.0 - h[d:])
        # contribution from second factor -h[i-d]
        g[d:] -= h[: n - d]
    else:                      # negative lag
        nd = -d
        # (1 - h[i-d]) part
        g[nd:] += (1.0 - h[: n - nd])
        # -h[i+d] part
        g[: n - nd] -= h[nd:]

    return g * dx

def _project_bounds_sum(x, target, eps=1e-12):
    """Project x onto [0,1] with exact sum = target (simple iterative fix)."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = cur - target
    # Small adjustments are enough for our use‑case
    if abs(diff) <= eps:
        return x
    # adjust a single free coordinate if possible
    free = np.where((x > 0.0) & (x < 1.0))[0]
    if free.size == 0:
        # fallback: spread adjustment uniformly and re‑clip
        adjust = diff / len(x)
        x = x - adjust
        x = np.clip(x, 0.0, 1.0)
        # final tiny correction on a random index
        diff = x.sum() - target
        if abs(diff) > eps:
            i = np.random.randint(len(x))
            x[i] -= diff
            x[i] = np.clip(x[i], 0.0, 1.0)
        return x
    # distribute diff over free indices (keep within bounds)
    share = diff / free.size
    for i in free:
        x[i] -= share
    # re‑clip and repeat if needed (very few iterations)
    return _project_bounds_sum(x, target, eps=eps)

def _random_binary_candidate(n, target_sum):
    """Create a binary (0/1) vector with exact sum = target_sum."""
    # decide number of intervals (3..7)
    M = np.random.randint(3, 8)
    # interval lengths (integers) summing to target_sum
    raw = np.random.rand(M)
    lengths = np.floor(raw / raw.sum() * target_sum).astype(int)
    # fix rounding error
    diff = int(target_sum - lengths.sum())
    while diff > 0:
        lengths[np.random.randint(M)] += 1
        diff -= 1
    # gap lengths (M+1) summing to the remaining points
    raw = np.random.rand(M + 1)
    gaps = np.floor(raw / raw.sum() * target_sum).astype(int)
    diff = int(target_sum - gaps.sum())
    while diff > 0:
        gaps[np.random.randint(M + 1)] += 1
        diff -= 1
    # build h
    h = np.zeros(n, dtype=float)
    pos = 0
    for i in range(M):
        pos += gaps[i]                    # skip gap
        L = lengths[i]
        if L > 0:
            end = min(pos + L, n)
            h[pos:end] = 1.0
            pos = end
    # final correction if rounding slipped
    cur = int(h.sum())
    if cur != target_sum:
        diff = int(target_sum - cur)
        if diff > 0:
            zeros = np.where(h == 0)[0]
            chosen = np.random.choice(zeros, size=diff, replace=False)
            h[chosen] = 1.0
        elif diff < 0:
            ones = np.where(h == 1)[0]
            chosen = np.random.choice(ones, size=-diff, replace=False)
            h[chosen] = 0.0
    return h

def _thue_morse_candidate(n, target_sum):
    """Generate a (approximately) balanced Thue–Morse sequence."""
    # build the classic (0/1) Thue–Morse for length n (use parity of popcount)
    t = np.arange(n, dtype=np.uint32)
    # popcount parity: xor of bits
    # use builtin bit_count if available (numpy 1.20+)
    if hasattr(t, "bit_count"):
        bits = t.bit_count()
    else:
        # fallback: use vectorised bit population count
        bits = np.unpackbits(t[:, None].view(np.uint8), axis=1).sum(axis=1)
    seq = (bits % 2).astype(float)      # 0/1 pattern
    # ensure exact sum = target_sum by adjusting a few entries
    cur = seq.sum()
    diff = int(target_sum - cur)
    if diff > 0:
        zeros = np.where(seq == 0)[0]
        chosen = np.random.choice(zeros, size=diff, replace=False)
        seq[chosen] = 1.0
    elif diff < 0:
        ones = np.where(seq == 1)[0]
        chosen = np.random.choice(ones, size=-diff, replace=False)
        seq[chosen] = 0.0
    return seq

def _optimize_once(h0, target_sum, maxiter=200):
    n = h0.shape[0]
    bounds = [(0.0, 1.0)] * n
    constraints = [{
        'type': 'eq',
        'fun': lambda x: np.sum(x) - target_sum,
        'jac': lambda x: np.ones_like(x)
    }]

    opts = {'maxiter': maxiter, 'ftol': 1e-9, 'disp': False}
    try:
        res = minimize(lambda x: _calc_c5(x),
                       h0,
                       method='SLSQP',
                       jac=lambda x: _grad_c5(x),
                       bounds=bounds,
                       constraints=constraints,
                       options=opts)
    except Exception:
        return None, np.inf

    if not res.success:
        # still return the best feasible point if any
        cand = np.clip(res.x, 0.0, 1.0)
        cand = _project_bounds_sum(cand, target_sum)
        val = _calc_c5(cand)
        return cand, val

    cand = np.clip(res.x, 0.0, 1.0)
    cand = _project_bounds_sum(cand, target_sum)
    val = _calc_c5(cand)
    return cand, val

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to lower the Erdős C5 bound.
    Returns (h_vector, c5_value, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    # Target number of discretisation points (keep modest for speed)
    n_points = 200
    target_sum = n_points / 2.0

    # ---- prepare initial candidates ---------------------------------
    init_candidates = []

    # Use supplied construction if present
    try:
        init_h = initial_h_values            # provided by the harness
        old_n = len(init_h)
        # interpolate to our grid
        x_old = (np.arange(old_n) + 0.5) * (2.0 / old_n)
        x_new = (np.arange(n_points) + 0.5) * (2.0 / n_points)
        init_h = np.interp(x_new, x_old, init_h)
        # enforce exact integral
        init_h *= (target_sum / init_h.sum())
        init_h = np.clip(init_h, 0.0, 1.0)
        init_h = _project_bounds_sum(init_h, target_sum)
        init_candidates.append(init_h)
    except Exception:
        pass   # no parent construction available

    # a few hand‑crafted deterministic candidates
    init_candidates.append(np.full(n_points, 0.5))
    # Thue‑Morse (if n_points is a power of two, it is perfectly balanced)
    if (n_points & (n_points - 1)) == 0:
        init_candidates.append(_thue_morse_candidate(n_points, target_sum))

    # a few random binary configurations
    for _ in range(4):
        init_candidates.append(_random_binary_candidate(n_points, target_sum))

    best_h = None
    best_val = np.inf

    # ---- main optimisation loop ------------------------------------
    for base_h in init_candidates:
        if time.time() - start_time > budget_s * 0.90:
            break
        h_opt, val = _optimize_once(base_h, target_sum, maxiter=250)
        if h_opt is not None and val < best_val:
            best_val = val
            best_h = h_opt.copy()

        # a tiny stochastic refinement: small random perturbation then re‑optimise
        for _ in range(2):
            if time.time() - start_time > budget_s * 0.90:
                break
            pert = np.random.normal(scale=0.05, size=n_points)
            h_pert = best_h + pert
            h_pert = np.clip(h_pert, 0.0, 1.0)
            h_pert = _project_bounds_sum(h_pert, target_sum)
            h_opt, val = _optimize_once(h_pert, target_sum, maxiter=150)
            if h_opt is not None and val < best_val:
                best_val = val
                best_h = h_opt.copy()

    # ---- optional second‑stage with finer grid if time permits -----
    if best_h is not None and time.time() - start_time < budget_s * 0.70:
        # increase resolution to 300 points, keep the shape via interpolation
        finer_n = 300
        target_fine = finer_n / 2.0
        x_old = (np.arange(n_points) + 0.5) * (2.0 / n_points)
        x_fine = (np.arange(finer_n) + 0.5) * (2.0 / finer_n)
        h_fine = np.interp(x_fine, x_old, best_h)
        h_fine = np.clip(h_fine, 0.0, 1.0)
        h_fine = _project_bounds_sum(h_fine, target_fine)

        # run a quick optimisation on the finer grid
        h_opt, val = _optimize_once(h_fine, target_fine, maxiter=200)
        if h_opt is not None and val < best_val:
            best_val = val
            best_h = h_opt.copy()
            n_points = finer_n
            target_sum = target_fine

    # ---- final sanity checks ----------------------------------------
    if best_h is None:
        # fallback: constant 0.5 (guaranteed feasible)
        best_h = np.full(n_points, 0.5)
        best_val = _calc_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_bounds_sum(best_h, target_sum)
    best_val = _calc_c5(best_h)

    return best_h, float(best_val), n_points
