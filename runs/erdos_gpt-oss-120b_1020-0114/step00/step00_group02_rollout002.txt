def gradient_step(h, α):
    n = len(h)
    # compute min shift
    corr = np.correlate(h, h, mode='full')
    min_idx = np.argmin(corr)
    k = min_idx - (n-1)
    grad = np.zeros_like(h)
    if k >= 0:
        grad[:n-k] += h[k:]  # h_{i+k}
        grad[k:] += h[:n-k]  # h_{i-k}

        s = -k
        grad[s:] += h[:n-s]
        grad[:n-s] += h[s:]
    # ascent
    new_h = h + α * grad
    # projection to constraints
    new_h = project_to_simplex_box(new_h, S)
    return new_h

import time
import numpy as np

def _project_to_box_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto the set {x: 0<=x_i<=1, sum x_i = target_sum}."""
    # binary search on lambda such that sum(clip(v - λ,0,1)) = target_sum
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        if w.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """Return the C5 value for a given step vector h."""
    n = h.shape[0]
    conv = np.correlate(h, h, mode="full")
    min_corr = conv.min()
    return 1.0 - (2.0 / n) * min_corr


def _gradient_min_shift(h: np.ndarray) -> np.ndarray:
    """Gradient of the minimal autocorrelation shift (to be *ascended*)."""
    n = h.shape[0]
    conv = np.correlate(h, h, mode="full")
    k = np.argmin(conv) - (n - 1)          # shift attaining the minimum
    grad = np.zeros_like(h)
    if k >= 0:
        # term h[i] * h[i+k]
        grad[: n - k] += h[k:]               # derivative from the first factor
        grad[k:]      += h[: n - k]          # derivative from the second factor
    else:
        s = -k
        grad[s:]      += h[: n - s]
        grad[: n - s] += h[s:]
    return grad
