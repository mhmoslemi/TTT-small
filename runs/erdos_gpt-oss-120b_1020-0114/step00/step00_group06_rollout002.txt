def project_box_sum(v, target):
    lo = np.min(v) - 1  # or -np.inf

            lo = mid
        else:
            hi = mid

import time
import numpy as np

def _project_box_sum(v, target, lower=0.0, upper=1.0, iters=60):
    """L2‑projection of v onto the hypercube [lower,upper] while enforcing sum=v_target."""
    lo, hi = -1e6, 1e6
    for _ in range(iters):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, lower, upper)
        s = w.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v + (lo + hi) / 2.0, lower, upper)
    # tiny adjustment if numerical error left a gap
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = (w > lower + 1e-12) & (w < upper - 1e-12)
        cnt = free.sum()
        if cnt:
            w[free] += diff / cnt
            w = np.clip(w, lower, upper)
    return w

def _c5_value(h):
    """Return the C5 bound for a vector h (integral over [0,2])."""
    N = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / N)

def _max_corr_info(h):
    """Compute correlation array, its maximum and the active shift indices."""
    N = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxv = corr.max()
    # active shifts (indices where corr is max within a tiny tolerance)
    idxs = np.where(corr >= maxv - 1e-12)[0]
    return corr, maxv, idxs

def _gradient_of_active_shifts(h, active_idxs):
    """Sub‑gradient of the max‑correlation w.r.t. h (average over all active shifts)."""
    N = h.size
    grads = np.zeros((len(active_idxs), N), dtype=np.float64)
    for p, idx in enumerate(active_idxs):
        m = idx - (N - 1)            # shift (positive → right)
        if m >= 0:
            # part coming from the first factor
            grads[p, :N - m] += (1.0 - h[m:])
            # part coming from the second factor
            grads[p, m:]    -= h[:N - m]
        else:
            dm = -m
            grads[p, dm:]   += (1.0 - h[:N - dm])
            grads[p, :N - dm] -= h[dm:]
    # average sub‑gradient
    return grads.mean(axis=0)

    # ------------------------------------------------------------------
    # initialise h
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():

    else:

        N = 120
        half = N // 2
        extra = N - 2 * half
        h = np.concatenate([np.ones(half), np.full(extra, 0.5), np.zeros(N - half - extra)])
        rng.shuffle(h)

    target_sum = N / 2.0
    h = _project_box_sum(h, target_sum)

    # ------------------------------------------------------------------
    # optimisation loop (simulated‑annealing + gradient steps)
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_c5 = _c5_value(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5

    start = time.time()
    # parameters
    sigma = 0.5                 # scale of random perturbations
    sigma_min = 1e-4
    sigma_decay = 0.9995
    grad_step = 0.2
    grad_decay = 0.9997

    while time.time() - start < budget_s * 0.94:
        # ---- random perturbation step ---------------------------------
        if rng.random() < 0.6:                     # mostly random moves
            proposal = cur_h + sigma * rng.normal(size=N)

            else:
