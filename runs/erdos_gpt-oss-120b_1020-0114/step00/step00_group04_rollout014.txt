        success = False
        # Determine tail indices
        if shift >= 0:

            interior_indices = np.arange(0, n - shift)

        if tail_pos.size > 0 and interior_pos.size > 0:

Implement function trapezoid_h(R, n):
    dx=2/n
    a = (1 - R)/2
    P = 1 - R
    xs = np.arange(n) * dx
    h = np.zeros_like(xs)
    # left ramp region: a <= x < a+R
    mask = (xs >= a) & (xs < a+R)
    h[mask] = (xs[mask] - a) / R
    # plateau region: a+R <= x < a+R+P
    mask = (xs >= a+R) & (xs < a+R+P)
    h[mask] = 1.0
    # right ramp region: a+R+P <= x < a+2*R+P
    mask = (xs >= a+R+P) & (xs < a+2*R+P)
    h[mask] = (a+2*R+P - xs[mask]) / R
    # else zero
    return h

def quadratic_h(p, n):
    dx = 2.0/n
    w = (p+1) / (2*p)  # support half-length from center
    xs = np.arange(n) * dx
    # center = 1
    d = np.abs(xs - 1.0)
    inside = d <= w
    h = np.zeros_like(xs)
    # normalized: (1 - (d/w)^p)
    h[inside] = 1.0 - (d[inside] / w) ** p
    return h

def compute_c5(h):

def improve_via_targeted(h, iters, budget, start_time):
   for _ in range(iters):
       if time.time() - start_time > budget * 0.95: break
       corr = np.correlate(h, 1 - h, mode='full')
       max_idx = np.argmax(corr)
       shift = max_idx - (len(h)-1)
       # Determine tail indices
       if shift >= 0:
           tail = np.arange(len(h) - shift, len(h))
           interior = np.arange(0, len(h) - shift)
       else:
           s = -shift
           tail = np.arange(0, s)
           interior = np.arange(s, len(h))
       # If tail empty, break
       if tail.size == 0 or interior.size == 0:
           break
       # Choose donor with largest h in tail
       i = tail[np.argmax(h[tail])]
       # Choose recipient j in interior s.t. complement small: maximize h[j+shift] if shift>=0 else h[j+shift]
       offset = interior + shift
       # offset in range [0,n) due to interior selection.
       # Compute complement values:
       comp = 1 - h[offset]
       j_candidate_idx = np.argmin(comp)  # or argmax h
       j = interior[j_candidate_idx]
       # Determine max delta to move from i to j while staying within bounds
       delta_max = min(h[i], 1 - h[j])
       if delta_max <= 1e-12:
           continue
       # Choose delta as factor * delta_max (e.g., 0.5)
       delta = delta_max * 0.5
       new_h = h.copy()
       new_h[i] -= delta
       new_h[j] += delta

import time
import numpy as np
import math

def _project_box_sum(v, target, lo=0.0, hi=1.0, it=60):
    """Project onto [lo,hi] box and hyperplane sum(v)=target."""
    left = -np.max(v) - 1.0
    right = np.max(v) + 1.0
    for _ in range(it):
        lam = (left + right) / 2.0
        proj = np.clip(v - lam, lo, hi)
        s = proj.sum()
        if s > target:
            left = lam
        else:
            right = lam
    return np.clip(v - (left + right) / 2.0, lo, hi)


def _c5_of(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _trapezoid_h(R, n):
    """Symmetric trapezoidal shape with ramp length R (0<R<1)."""
    a = (1.0 - R) / 2.0          # zero interval at each side
    P = 1.0 - R                  # flat top length
    xs = np.arange(n, dtype=float) * (2.0 / n)
    h = np.zeros_like(xs)

    # left ramp
    mask = (xs >= a) & (xs < a + R)
    h[mask] = (xs[mask] - a) / R

    # flat top
    mask = (xs >= a + R) & (xs < a + R + P)
    h[mask] = 1.0

    # right ramp
    mask = (xs >= a + R + P) & (xs < a + 2 * R + P)
    h[mask] = (a + 2 * R + P - xs[mask]) / R

    return h


def _quadratic_h(p, n):
    """Symmetric power‑law shape: h = 1 - (|x-1|/w)^p on its support."""
    w = (p + 1.0) / (2.0 * p)          # half‑width chosen so area = 1
    xs = np.arange(n, dtype=float) * (2.0 / n)
    d = np.abs(xs - 1.0)
    mask = d <= w
    h = np.zeros_like(xs)
    h[mask] = 1.0 - (d[mask] / w) ** p
    return h

    start = time.time()

    # Choose a modest resolution; larger n gives finer shapes.
    n = 200
    target_sum = n / 2.0

    best_h = None

    for R in np.linspace(0.01, 0.99, 99):
        h = _trapezoid_h(R, n)
        h = _project_box_sum(h, target_sum)

    # Power‑law (quadratic, cubic, …) family

    best_h = _project_box_sum(best_h, target_sum)
    best_val = _c5_of(best_h)

    return (best_h.tolist(), float(best_val), n)
