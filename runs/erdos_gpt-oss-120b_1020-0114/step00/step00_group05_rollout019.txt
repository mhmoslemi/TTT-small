import time
import numpy as np

def _c5_val(h):
    """C5 value for a sample vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def _proj_feasible(y, target):
    """
    Projection of y onto {x : 0<=x<=1, sum(x)=target}.
    Solves x = clip(y - λ,0,1) with sum(x)=target via bisection on λ.
    """
    lo = y.min() - 1.0          # λ low ⇒ all entries become 1 after clipping
    hi = y.max() + 1.0          # λ high ⇒ all entries become 0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)


def _grad_shift(h, shift):
    """
    Sub‑gradient of the overlap term for a given integer shift.
    shift > 0  : uses indices i, i+shift  (i+shift < n)
    shift < 0  : uses i, i+shift  (i+shift >= 0)
    """
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        k = shift
        if k < n:
            i = np.arange(0, n - k)
            j = i + k
            # contribution from h_i * (1 - h_j)
            g[i] += 1.0 - h[j]
            # contribution from -h_i * h_j
            g[j] -= h[i]
    else:
        d = -shift
        if d < n:
            i = np.arange(d, n)
            j = i - d
            g[i] += 1.0 - h[j]
            g[j] -= h[i]
    return g


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- problem size -------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.shape[0]
    else:
        n = 200                     # default resolution
        h = np.full(n, 0.5, dtype=float)

    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    h = _proj_feasible(h, target_sum)

    best_h = h.copy()
    best_val = _c5_val(best_h)

    # ---- optimisation parameters ---------------------------------------
    step0 = 1.0
    total_iter = 0
    restart_counter = 0

    while True:
        if time.time() - start > budget_s * 0.98:
            break

        # correlation and shift giving the current worst overlap
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)            # integer lag

        # sub‑gradient of the active term
        grad = _grad_shift(h, shift)

        # move opposite to the gradient (reduce the max)
        direction = -grad
        step = step0 / np.sqrt(total_iter + 1.0)
        y = h + step * direction
        h_new = _proj_feasible(y, target_sum)

        val = _c5_val(h_new)

        if val < best_val:
            best_val = val
            best_h = h_new.copy()

        h = h_new
        total_iter += 1

        # occasional random perturbation to escape plateaus
        if total_iter % 100 == 0 and total_iter > 0:
            noise = rng.normal(scale=0.02, size=n)
            h = _proj_feasible(h + noise, target_sum)

        # periodic full restart with a fresh binary seed
        if total_iter % 2000 == 0 and total_iter > 0:
            restart_counter += 1
            # binary start: exactly half ones
            idx_one = rng.choice(n, n // 2, replace=False)
            h = np.zeros(n, dtype=float)
            h[idx_one] = 1.0
            h = _proj_feasible(h, target_sum)

    return (best_h, best_val, n)
