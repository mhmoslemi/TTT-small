def project_to_target_sum(v, target):
    N = len(v)
    # produce binary vector (0/1) satisfying sum = floor(target) or ceil.
    # If target is integer (N even), assign exactly target ones.
    # For N odd and target half integer, assign floor(target) ones + one element = 0.5.
    v_sorted_idx = np.argsort(v)[::-1]
    binary = np.zeros(N)
    ones_needed = int(np.floor(target))
    binary[v_sorted_idx[:ones_needed]] = 1.0
    remainder = target - ones_needed
    if remainder > 0:
        # assign fractional remainder to next highest element
        binary[v_sorted_idx[ones_needed]] = remainder
    return binary
```

def project_to_target_sum(v, target):
    N = len(v)
    ones_needed = int(round(target))
    idx = np.argsort(v)[::-1]
    binary = np.zeros(N)
    binary[idx[:ones_needed]] = 1.0
    # If target not integer, assign fractional remainder as remainder
    remainder = target - ones_needed
    if remainder > 1e-12:
        binary[idx[ones_needed]] = remainder
    return binary
```

def eligible_swap_positions(h, k):
    N = len(h)
    if k >= 0:
        # overlapping region indices 0..N-1-k
        overlap = np.arange(N - k)
        # Condition for overlapping indices: h[j]==0 and h[j+k]==1
        mask_overlap = (h[overlap] == 0) & (h[overlap + k] == 1)
        # out-of-overlap indices: j >= N - k
        out_of_overlap = np.arange(N - k, N)
        mask_out = (h[out_of_overlap] == 0)
        eligible = np.concatenate((overlap[mask_overlap], out_of_overlap[mask_out]))
    else:
        kpos = -k
        # overlapping region indices kpos..N-1
        overlap = np.arange(kpos, N)
        mask_overlap = (h[overlap] == 0) & (h[overlap - kpos] == 1)
        # out of overlap: j < kpos
        out_of_overlap = np.arange(0, kpos)
        mask_out = (h[out_of_overlap] == 0)
        eligible = np.concatenate((overlap[mask_overlap], out_of_overlap[mask_out]))
    return eligible
```

def propose_move(h, step):
    N = len(h)
    i,j = np.random.choice(N, 2, replace=False)
    # propose random epsilon in [-step, step] but respecting bounds
    limit_i_up = 1 - h[i]
    limit_i_down = -h[i]
    limit_j_up = 1 - h[j]
    limit_j_down = -h[j]
    # Since we need to add epsilon to i and subtract epsilon from j, epsilon must satisfy:
    # -h[i] <= epsilon <= 1 - h[i]
    # h[j] - epsilon must be within [0,1] => - (1 - h[j]) <= epsilon <= h[j]
    eps_low = max(-h[i], h[j] - 1)

    eps = np.random.uniform(eps_low, eps_high)
    new_h = h.copy()
    new_h[i] += eps
    new_h[j] -= eps
    return new_h

import time
import numpy as np
import random

# ---------------- helper functions ----------------

def project_to_target(v, target):
    """Return a vector with the same length as v, values in [0,1],
    and sum exactly equal to target (up to a tiny remainder)."""
    N = v.shape[0]
    ones_needed = int(np.floor(target))
    idx = np.argsort(v)[::-1]                    # descending order
    out = np.zeros(N, dtype=float)
    if ones_needed > 0:
        out[idx[:ones_needed]] = 1.0
    # fractional remainder (only when target is not integer)
    rem = target - ones_needed
    if rem > 1e-12:
        out[idx[ones_needed]] = rem
    return out

def max_correlation(h):
    """Return (max_corr, shift) where max_corr = max_i sum_j h[j]*(1-h[j+i])"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    i = np.argmax(corr)
    max_val = corr[i]
    shift = i - (len(h) - 1)   # shift corresponding to this index
    return max_val, shift, corr

def mismatch_positions(h, k):
    """Indices i where h[i]==1 and (i+k) in range and h[i+k]==0."""
    N = h.shape[0]
    if k >= 0:
        i = np.arange(N - k)
        mask = (h[i] == 1) & (h[i + k] == 0)
    else:
        kp = -k
        i = np.arange(kp, N)
        mask = (h[i] == 1) & (h[i - kp] == 0)
    return i[mask]

def eligible_swap_positions(h, k):
    """Indices j where h[j]==0 and swapping j to 1 does NOT increase
    the overlap for shift k."""
    N = h.shape[0]
    if k >= 0:
        # overlapping part
        overlap = np.arange(N - k)
        mask_overlap = (h[overlap] == 0) & (h[overlap + k] == 1)
        # out of overlap (no contribution for this k)
        out_part = np.arange(N - k, N)
        mask_out = (h[out_part] == 0)
        eligible = np.concatenate((overlap[mask_overlap], out_part[mask_out]))
    else:
        kp = -k
        overlap = np.arange(kp, N)
        mask_overlap = (h[overlap] == 0) & (h[overlap - kp] == 1)
        out_part = np.arange(0, kp)
        mask_out = (h[out_part] == 0)
        eligible = np.concatenate((overlap[mask_overlap], out_part[mask_out]))
    return eligible

# ---------------- main optimisation ----------------

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) achieving a low C5."""
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()

    # ---- initialise h -------------------------------------------------
    if 'initial_h_values' in globals():
        base = np.array(initial_h_values, dtype=float)
        # ensure even length for binary representation
        if base.size % 2 == 1:
            base = np.repeat(base, 2)                      # length becomes even
        N = base.size
        # project onto a feasible binary vector (sum = N/2)
        h = project_to_target(base, N / 2.0)
    else:
        N = 200                               # default even size
        h = np.zeros(N, dtype=float)
        ones = np.random.choice(N, N // 2, replace=False)
        h[ones] = 1.0

    # guarantee exact feasibility (binary + correct sum)
    h = np.clip(h, 0.0, 1.0)
    target_sum = N / 2.0
    # tiny correction if rounding created a deficit/excess
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # adjust the smallest non‑integer entry (or any zero) by the needed amount
        # keep within [0,1]
        for idx in np.argsort(h):
            if 0.0 <= h[idx] + diff <= 1.0:
                h[idx] += diff
                break

    dx = 2.0 / N

    # compute initial objective
    max_corr, worst_shift, corr = max_correlation(h)
    cur_val = max_corr * dx
    best_h = h.copy()
    best_val = cur_val

    # parameters for the search
    no_improve = 0
    restart_limit = 3000          # after so many non‑improving steps we restart
    max_time = budget_s * 0.95     # keep a safety margin

    # ---- hill‑climbing loop --------------------------------------------
    while time.time() - start_time < max_time:
        # find the shift that currently attains the maximum overlap
        max_corr, worst_shift, corr = max_correlation(h)

        # try a smart swap targeting the worst shift
        mism = mismatch_positions(h, worst_shift)
        elig = eligible_swap_positions(h, worst_shift)

        if mism.size > 0 and elig.size > 0:
            i = int(np.random.choice(mism))
            j = int(np.random.choice(elig))
        else:
            # fallback: random 1↔0 swap
            ones_idx = np.where(h == 1)[0]

        h_new[i] = 0.0

        if new_val < cur_val - 1e-12:               # strict improvement

            cur_val = new_val
            no_improve = 0

                # early exit if we already beat the target record

    diff = target_sum - best_h.sum()

        for idx in np.argsort(best_h):
            if 0.0 <= best_h[idx] + diff <= 1.0:
                best_h[idx] += diff
                break

    return (best_h, best_val, N)
