    # compute max delta magnitude allowed

import time, math

def _compute_c5(vec):
    """Return the C5 value for a given step‑function sample vector."""
    n = vec.size
    dx = 2.0 / n
    corr = np.correlate(vec, 1.0 - vec, mode="full")
    return float(corr.max() * dx)


def _make_feasible(vec, target):
    """Project a vector onto the feasible set 0≤h≤1, sum(h)=target."""
    vec = np.clip(vec, 0.0, 1.0)
    cur = vec.sum()
    diff = target - cur
    # simple correction on a random free coordinate
    if abs(diff) > 1e-12:
        free = np.where((vec > 1e-12) & (vec < 1.0 - 1e-12))[0]
        if free.size == 0:          # everything at a bound – pick any index
            free = np.arange(vec.size)
        idx = np.random.choice(free)
        vec[idx] = np.clip(vec[idx] + diff, 0.0, 1.0)
    # final safety clamp
    return np.clip(vec, 0.0, 1.0)


def _random_feasible(n, target):
    """Generate a random feasible vector (values in [0,1], correct sum)."""
    # start from a uniform Dirichlet sample, then clip & re‑project
    x = np.random.random(n)
    x = x / x.sum() * target
    x = np.clip(x, 0.0, 1.0)
    # iterate a few times to repair the sum
    for _ in range(5):
        cur = x.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        # distribute the deficit/excess uniformly among non‑saturated entries
        free = np.where((x > 1e-12) & (x < 1.0 - 1e-12))[0]
        if free.size == 0:
            break
        add = diff / free.size
        x[free] = np.clip(x[free] + add, 0.0, 1.0)
    return x


def _guided_move(h, shift, rng):
    """Propose a move that tries to decrease the overlap at the worst shift."""
    n = h.size
    if shift >= 0:
        i_idx = np.arange(0, n - shift)
        j_idx = i_idx + shift
    else:
        k = -shift
        i_idx = np.arange(k, n)
        j_idx = i_idx - k

    # contribution of each pair to the problematic correlation
    contrib = h[i_idx] * (1.0 - h[j_idx])
    if contrib.sum() == 0:
        return None

    # sample an index proportionally to its contribution
    probs = contrib / contrib.sum()
    i = rng.choice(i_idx, p=probs)
    j = i + shift

    # two possible directions: i→j or j→i
    best_new = None
    best_val = None

    for src, dst in ((i, j), (j, i)):
        if h[src] <= 0 or h[dst] >= 1:
            continue
        max_delta = min(h[src], 1.0 - h[dst])
        if max_delta <= 0:
            continue
        delta = rng.uniform(0.0, max_delta)
        new_h = h.copy()
        new_h[src] -= delta
        new_h[dst] += delta
        val = _compute_c5(new_h)
        if best_val is None or val < best_val:
            best_val = val
            best_new = new_h
    return best_new, best_val

            idx_max = int(full_corr.argmax())
            shift = idx_max - (n - 1)          # shift in [-n+1, n-1]

            guided = _guided_move(cur_h, shift, rng)
            if guided is not None:
                new_h, new_val = guided
            else:
                # fall back to a generic pair move
                i, j = rng.choice(n, size=2, replace=False)
                hi, hj = cur_h[i], cur_h[j]
                max_pos = min(1.0 - hi, hj)
                max_neg = min(hi, 1.0 - hj)
                if max_pos <= 0 and max_neg <= 0:
                    continue
                delta = rng.uniform(-max_neg, max_pos)
                new_h = cur_h.copy()
                new_h[i] = hi + delta
                new_h[j] = hj - delta
                new_val = _compute_c5(new_h)
        else:
            # generic two‑index move
            i, j = rng.choice(n, size=2, replace=False)
            hi, hj = cur_h[i], cur_h[j]
            max_pos = min(1.0 - hi, hj)
            max_neg = min(hi, 1.0 - hj)
            if max_pos <= 0 and max_neg <= 0:
                continue
            delta = rng.uniform(-max_neg, max_pos)
            new_h = cur_h.copy()
            new_h[i] = hi + delta
            new_h[j] = hj - delta
            new_val = _compute_c5(new_h)

        # ------------------------------------------------------------------
        # 5. Acceptance test (simulated annealing)
        # ------------------------------------------------------------------
        accept = False
        if new_val < cur_val - 1e-12:
            accept = True
        else:
            prob = math.exp(-(new_val - cur_val) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_h, cur_val = new_h, new_val
            if new_val < best_val - 1e-12:
                best_h, best_val = new_h.copy(), new_val
                last_improve = it

        # ------------------------------------------------------------------
        # 6. Temperature decay and occasional restart
        # ------------------------------------------------------------------
        T *= cooling
        it += 1

        if it - last_improve > 250_000:
            # generate a fresh random feasible point
            cur_h = _random_feasible(n, target_sum)
            cur_val = _compute_c5(cur_h)
            last_improve = it

    # ------------------------------------------------------------------
    # 7. Final clean‑up (tiny drift correction)
    # ------------------------------------------------------------------
    best_h = _make_feasible(best_h, target_sum)
    final_val = _compute_c5(best_h)

    return (best_h.tolist(), float(final_val), n)
