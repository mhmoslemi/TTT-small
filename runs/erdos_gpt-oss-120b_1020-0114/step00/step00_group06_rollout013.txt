import numpy as np
import time

def project_to_sum_box(v, target):
    # project v onto [0,1] with sum = target using bisection on additive shift
    lo = -2.0  # sufficiently low
    hi = 2.0   # sufficiently high
    for _ in range(60):
        lam = (lo + hi) / 2.0
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + lo, 0.0, 1.0)

def compute_c5(h, dx):
    # correlation full
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

import numpy as np
import time

def _project_to_sum_box(v, target):
    """
    Project a vector v onto the box [0,1] while forcing its sum to equal target.
    This is done by finding a scalar λ such that sum(clip(v+λ,0,1)) = target.
    """
    lo, hi = -2.0, 2.0          # bounds safely containing the solution
    for _ in range(60):        # enough for double precision
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + lo, 0.0, 1.0)


def _c5_of(h, dx):
    """return the overlap constant C5 for a discrete h (full correlation)."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def run(seed=42, budget_s=1100, **kwargs):

    Construct a step function h on [0,2] that attains a small C5 value.
    The construction is an analytically‑derived two‑level symmetric
    function; a short hill‑climbing phase optionally refines it.
    Returns (h_values, c5_bound, n_points).
    """

    np.random.seed(seed)

    # ------------------------------------------------------------
    # 1. Choose a modest even discretisation.
    # ------------------------------------------------------------

    dx = 2.0 / N
    target_sum = N / 2.0

    #    a = length of each outer block with value 1.

    b_opt = (0.5 - a_opt) / (1.0 - a_opt) if a_opt < 1.0 else 0.0

    # positions of cell centres on [0,2]
    x = (np.arange(N) + 0.5) * dx
    h = np.where((x <= a_opt) | (x >= 2.0 - a_opt), 1.0, b_opt)

    # enforce exact integral (sum = N/2) while staying in [0,1]
    h = _project_to_sum_box(h, target_sum)
