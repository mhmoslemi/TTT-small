def weight_vector(h, lag):
    n = len(h)
    w = np.zeros(n)
    if lag >= 0:
        # for i = 0..n-1-lag, h[i+lag] defined
        w[:n-lag] += h[lag:]
        # for i = lag..n-1, h[i-lag] defined
        w[lag:] += h[:n-lag]
    else:
        lag = -lag
        w[lag:] += h[:n-lag]
        w[:n-lag] += h[lag:]
    return w

import time, numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    # compute autocorrelation
    Nfft = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(h, n=Nfft)
    rev = np.fft.rfft(h[::-1], n=Nfft)
    conv = np.fft.irfft(fa * rev, n=Nfft)
    r = conv[:2*n-1]
    min_r = np.min(r)
    sum_h = np.sum(h)  # should be n/2
    return (sum_h - min_r) * dx, min_r

import time
import numpy as np

# ----- basic utilities -------------------------------------------------
def min_autocorr(h):
    """Return min of full autocorrelation of h (zero‑padded)."""
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(h, n=N)
    fb = np.fft.rfft(h[::-1], n=N)
    conv = np.fft.irfft(fa * fb, n=N)
    r = conv[: 2 * n - 1]                         # same layout as np.correlate
    return np.min(r)


def c5_from_h(h):
    """Maximum overlap integral for step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    sum_h = h.sum()                              # should be n/2
    min_r = min_autocorr(h)
    return (sum_h - min_r) * dx


def lag_of_min_autocorr(h):
    """Signed lag (−(n−1)…+(n−1)) where autocorrelation is minimal."""
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(h, n=N)
    fb = np.fft.rfft(h[::-1], n=N)
    conv = np.fft.irfft(fa * fb, n=N)
    r = conv[: 2 * n - 1]
    idx = np.argmin(r)
    return idx - (n - 1)


def weight_vector(h, lag):
    """Weight of each coordinate for increasing correlation at given lag."""
    n = h.shape[0]
    w = np.zeros(n)
    if lag >= 0:
        if lag < n:
            w[: n - lag] += h[lag:]
            w[lag:] += h[: n - lag]
    else:
        k = -lag
        if k < n:
            w[k:] += h[: n - k]
            w[: n - k] += h[k:]
    return w


def random_initial_h(n):
    """Generate a feasible random h (values in [0,1], sum = n/2)."""
    target = n / 2.0
    h = np.random.rand(n)
    h *= target / h.sum()
    # clip and fix tiny deviations
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    for _ in range(200):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1 - 1e-12)[0]
            if idx.size == 0:
                break
            i = np.random.choice(idx)
            inc = min(diff, 1.0 - h[i])
            h[i] += inc
            diff -= inc
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = np.random.choice(idx)
            dec = min(-diff, h[i])
            h[i] -= dec
            diff += dec
    return h


# ----- local optimisation -----------------------------------------------
def improve_h(h, max_iters=20000, step_factor=0.5):
    """Greedy hill‑climb: move mass from low‑weight to high‑weight sites."""
    n = h.shape[0]
    best_h = h.copy()
    best_val = c5_from_h(best_h)

    for it in range(max_iters):
        # worst lag and its weight vector
        lag = lag_of_min_autocorr(h)
        w = weight_vector(h, lag)

        # choose donor (low weight, positive) and receiver (high weight, not full)
        inc_idx = np.where(h < 1 - 1e-12)[0]
        dec_idx = np.where(h > 1e-12)[0]
        if inc_idx.size == 0 or dec_idx.size == 0:
            break

        i = inc_idx[np.argmax(w[inc_idx])]
        j = dec_idx[np.argmin(w[dec_idx])]

        max_delta = min(1.0 - h[i], h[j])
        if max_delta <= 0:
            break
        delta = max_delta * step_factor

        h[i] += delta
        h[j] -= delta

        new_val = c5_from_h(h)
        if new_val + 1e-12 < best_val:
            best_val = new_val
            best_h = h.copy()
        if new_val < best_val + 1e-12:
            # keep the move
            continue
        # undo, try a smaller step
        h[i] -= delta
        h[j] += delta
        delta_small = max_delta * 0.1
        if delta_small < 1e-12:
            break
        h[i] += delta_small
        h[j] -= delta_small
        new_val2 = c5_from_h(h)
        if new_val2 + 1e-12 < best_val:
            best_val = new_val2
            best_h = h.copy()
        else:
            h[i] -= delta_small
            h[j] += delta_small
            # no improvement – stop early
            break
    return best_h, best_val
