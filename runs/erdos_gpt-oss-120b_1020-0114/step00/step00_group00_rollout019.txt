def upsample_h(original_h, n_target):
    n0 = len(original_h)
    x0 = np.linspace(0, 2, n0, endpoint=False) # points at start of each segment? Actually positions consistent.
    x_target = np.linspace(0, 2, n_target, endpoint=False)
    # Use np.interp (linear) to fill.
    h_target = np.interp(x_target, x0, original_h, left=original_h[0], right=original_h[-1])

    target_sum = n_target / 2
    h_target = project_to_bounds_and_sum(h_target, target_sum, lo=0.0, hi=1.0)

    hi_val = hi

import time
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Discrete C5 value for a step function given by array h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _project_bounds_sum(x: np.ndarray, target: float, lo: float = 0.0, hi: float = 1.0) -> np.ndarray:
    """Project x onto [lo, hi] and enforce sum(x)=target by a simple water‑filling."""
    # binary search for lambda so that sum(clamp(x+lambda)) = target
    lo_lambda, hi_lambda = -10.0, 10.0
    for _ in range(60):
        mid = (lo_lambda + hi_lambda) * 0.5
        s = np.sum(np.clip(x + mid, lo, hi))
        if s < target:
            lo_lambda = mid
        else:
            hi_lambda = mid
    lam = (lo_lambda + hi_lambda) * 0.5
    return np.clip(x + lam, lo, hi)


def _random_feasible(n: int) -> np.ndarray:
    """Generate a random vector in [0,1] with sum = n/2."""
    x = np.random.rand(n)
    return _project_bounds_sum(x, n / 2.0)


def _upsample(old: np.ndarray, new_n: int) -> np.ndarray:
    """Linear up‑sample old vector to length new_n and re‑project onto constraints."""
    old_n = old.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    up = np.interp(x_new, x_old, old)
    return _project_bounds_sum(up, new_n / 2.0)


def _hill_climb(start_h: np.ndarray, start_val: float,
                max_iters: int, time_limit: float) -> tuple:
    """Small simulated‑annealing style local search."""
    n = start_h.shape[0]
    target_sum = n / 2.0
    cur_h = start_h.copy()
    cur_val = start_val
    best_h = cur_h.copy()
    best_val = cur_val
    T = 0.02  # initial temperature

    for _ in range(max_iters):
        if time.time() > time_limit:
            break

        # occasional global perturbation
        if np.random.rand() < 0.07:
            sigma = 0.12
            cand = cur_h + sigma * np.random.randn(n)
            cand = np.clip(cand, 0.0, 1.0)
            cand = _project_bounds_sum(cand, target_sum)
        else:
            i, j = np.random.randint(0, n, size=2)
            while i == j:
                j = np.random.randint(0, n)
            sign = 1.0 if np.random.rand() < 0.5 else -1.0
            if sign > 0.0:
                max_delta = min(0.15, 1.0 - cur_h[i], cur_h[j])
            else:
                max_delta = min(0.15, cur_h[i], 1.0 - cur_h[j])
            if max_delta <= 0.0:
                continue
            delta = sign * np.random.rand() * max_delta
            cand = cur_h.copy()
            cand[i] += delta
            cand[j] -= delta

        cand_val = _compute_c5(cand)

        if cand_val < cur_val or np.random.rand() < np.exp(-(cand_val - cur_val) / max(T, 1e-12)):
            cur_h, cur_val = cand, cand_val
            if cand_val < best_val:
                best_h, best_val = cand.copy(), cand_val

        T *= 0.9997  # slow annealing

    return best_h, best_val
