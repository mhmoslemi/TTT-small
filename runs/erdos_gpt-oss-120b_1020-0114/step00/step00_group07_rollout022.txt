def project_to_feasible(v, target_sum):
    # binary search λ such that sum(clip(v+λ, 0,1)) = target_sum
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(v + mid, 0, 1)
        s = h.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(v + (lo+hi)/2, 0, 1)
    # after binary search may be slight off; adjust final scaling small
    # ensure exactly sum: compute diff = target_sum - sum
    diff = target_sum - h.sum()
    # distribute diff uniformly among non-saturated elements
    free = (h > 0) & (h < 1)
    if np.any(free):
        # add diff/num_free to those
        h[free] += diff / free.sum()
        # clip again
        h = np.clip(h, 0, 1)
    # if still off due to rounding, adjust by small iter
    # final guarantee: clip and adjust with while loop could iterate
    # ensure within tolerance
    # ...
    # Lastly, ensure sum close to target using scaling of whole vector for minor error:
    correction = target_sum - h.sum()
    if abs(correction) > 1e-9:
        # distribute over all entries proportionally?

def run(seed=42, budget_s=1100, **kwargs):

    start = time.time()
    rng = np.random.default_rng(seed)

    # Determine size n based on budget, maybe up to 300?
    n = 200  # could be parameter
    dx = 2.0 / n
    target_sum = n / 2

    # Starting vector:
    if 'initial_h_values' in globals():
        h0 = globals()['initial_h_values']
        if isinstance(h0, (list, tuple, np.ndarray)):
            h0 = np.asarray(h0, dtype=float)
            if h0.shape[0] != n:
                x_old = np.linspace(0, 2, h0.shape[0], endpoint=False)
                x_new = np.linspace(0, 2, n, endpoint=False)
                h = np.interp(x_new, x_old, h0)
            else:
                h = h0.copy()

            h = rng.random(n)
    else:
        h = rng.random(n)

    h = project_to_feasible(h, target_sum)

    # Evaluate initial
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    current_h = h.copy()
    current_c5 = best_c5

    # Simulated annealing parameters:
    max_iters = 200000  # may be truncated by time
    # adjust based on budget: each iteration maybe ~0.0001 sec -> 200k = 20 secs. Should be okay.
    # We'll compute remaining time each loop and break if exceed budget-safety.
    # Temperature schedule:
    t_initial = 0.05
    t_final = 1e-5

    # Step size for delta:
    step_magnitude = 0.2  # max absolute delta for move

    # Pre-setup to reduce overhead:
    n = int(n)
    target_sum = float(target_sum)

    for it in range(max_iters):
        # time check
        if time.time() - start > budget_s * 0.97:  # leave margin
            break
        # temperature
        t = t_initial * ((t_final / t_initial) ** (it / max_iters))
        # propose move: swap mass between two random indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = rng.uniform(-step_magnitude, step_magnitude)
        # create proposal vector
        proposal = current_h.copy()
        proposal[i] = proposal[i] + delta
        proposal[j] = proposal[j] - delta

            if prop_c5 < best_c5:
                best_c5 = prop_c5
                best_h = proposal.copy()
    # Done: return best_h, best_c5, n
    # Ensure result is list or array convertible
    return best_h, float(best_c5), n

def project_to_feasible(v, target):
    lo = -np.max(v) - 1.0 # ensure all v+lo <= 0
    hi = 1.0 - np.min(v) + 1.0 # ensure >=1
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(v + mid, 0.0, 1.0)
        s = h.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v + (lo+hi)/2.0, 0.0, 1.0)
    # after binary search, ensure sum close:
    # If due to rounding sum not equal target, adjust by small mass distribution.
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # distribute diff among non-saturated indices
        free = (h > 1e-12) & (h < 1-1e-12)
        if free.any():
            # adjust proportionally
            adjust = diff / free.sum()
            h[free] += adjust
            # clip again maybe break bound
            h = np.clip(h, 0.0, 1.0)
        # After second clipping, we can re-binary-search again if needed
        # Ensure final sum within tolerance, if not, fallback to scaling whole vector.
        # Let's just accept minor error within tolerance 1e-9.
    return h

        # need increase some entries not at 1
        free = np.where(h < 1 - 1e-12)[0]
    else:
        free = np.where(h > 1e-12)[0]

    # To avoid endless loops, we bound iterations
    for _ in range(10):

import time
import math
import numpy as np

def compute_c5(h, dx):
    """C5 value for a given step function."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def grad_for_shift(h, k):
    """Gradient of C(k) = Σ h[i]*(1-h[i+k]) w.r.t h, for a fixed integer shift k."""
    n = h.shape[0]
    g = np.ones_like(h)
    if k >= 0:
        if k < n:
            g[: n - k] -= h[k:]          # subtract h[i+k]
            g[k:] -= h[: n - k]          # subtract h[i-k]
    else:
        k2 = -k
        if k2 < n:
            g[k2:] -= h[: n - k2]        # subtract h[i+k] (i >= k2)
            g[: n - k2] -= h[k2:]        # subtract h[i-k] (i < n-k2)
    return g


def normalize_sum(h, target):
    """Clip to [0,1] and adjust to hit the target sum (simple iterative correction)."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # distribute the difference uniformly a few times – it converges quickly for small diffs
    for _ in range(5):
        if abs(diff) < 1e-12:
            break
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    return h


def project_to_feasible(v, target):
    """
    Exact projection of a vector onto the box [0,1]^n with a fixed linear sum.
    Binary search on a global shift λ so that sum(clamp(v+λ,0,1)) = target.
    """
    lo = -np.max(v) - 2.0          # guarantees v+lo ≤ 0 ⇒ sum = 0
    hi = 1.0 - np.min(v) + 2.0    # guarantees v+hi ≥ 1 ⇒ sum = n
    for _ in range(60):
        mid = (lo + hi) * 0.5
        h = np.clip(v + mid, 0.0, 1.0)
        s = h.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # final tiny correction if needed
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = normalize_sum(h, target)
    return h


def random_move(h, rng, max_step=0.2):
    """Two‑index move that preserves the sum and respects [0,1] bounds."""
    n = h.shape[0]
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    # feasible interval for delta added to i (and subtracted from j)
    lower = max(-h[i], h[j] - 1.0)
    upper = min(1.0 - h[i], h[j])
    if lower >= upper:
        return h  # cannot move safely
    # limit magnitude to stay near current point
    bound = min(max_step, upper - lower)
    if bound <= 0:
        return h
    delta = rng.uniform(-bound, bound)
    delta = np.clip(delta, lower, upper)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

    """Gradient step for the shift producing the current maximum."""
    g = grad_for_shift(h, k)
    h_new = h - alpha * g
    h_new = normalize_sum(h_new, target)
    return h_new
