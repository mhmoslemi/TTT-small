import time, math
import numpy as np

def _eval_h(h, dx):
    """Return C5 for a given step function h."""
    # correlation of h with (1-h) over full mode, then scale by dx
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def _project_to_sum_box(v, target):
    """Project vector v onto [0,1]^n with sum == target (simple waterfilling)."""
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target) < 1e-12:
        return v
    # if we need to increase the sum
    if cur < target:
        deficit = target - cur
        # capacity to increase each entry
        cap = 1.0 - v
        while deficit > 1e-12:
            idx = np.where(cap > 1e-12)[0]
            if len(idx) == 0:
                break
            add = deficit / len(idx)
            add = np.minimum(add, cap[idx])
            v[idx] += add
            deficit -= add.sum()
            cap = 1.0 - v
    else:  # need to decrease
        excess = cur - target
        cap = v.copy()
        while excess > 1e-12:
            idx = np.where(cap > 1e-12)[0]
            if len(idx) == 0:
                break
            sub = excess / len(idx)
            sub = np.minimum(sub, cap[idx])
            v[idx] -= sub
            excess -= sub.sum()
            cap = v.copy()
    return v

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # Determine problem size
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 42
        h0 = np.full(n, 0.5)

    dx = 2.0 / n
    target_sum = n / 2.0  # =21 for n=42

    # Use a repeated‑half representation to keep the search space small
    half_n = n // 2

    # Initialise p (first half). Start from uniform 0.5 which satisfies the sum.
    p = np.full(half_n, target_sum / half_n)  # 0.5
    # If a parent construction exists, try to use its projected half‑vector.
    if 'initial_h_values' in globals():
        # project the provided vector onto the repeated‑half space
        # Take the first half; if n is even the split is exact.
        cand = h0[:half_n].copy()
        cand = _project_to_sum_box(cand, target_sum / 2.0)
        # evaluate both candidates
        h_cand = np.concatenate([cand, cand])
        if _eval_h(h_cand, dx) < _eval_h(np.concatenate([p, p]), dx):
            p = cand

    # evaluation helpers
    def eval_p(vec):
        h = np.concatenate([vec, vec])
        return _eval_h(h, dx)

    cur_val = eval_p(p)
    best_p = p.copy()
    best_val = cur_val

    start_time = time.time()
    # Simulated annealing parameters
    max_temp = 0.05
    min_temp = 1e-6
    # loop until time budget exhausted
    it = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break

        # temperature linearly decreasing
        t_ratio = max(0.0, 1.0 - elapsed / budget_s)
        T = max(min_temp, max_temp * t_ratio)

        # pick two distinct indices and transfer a small amount
        i, j = rng.integers(0, half_n, size=2)
        if i == j:
            continue
        # random direction of transfer
        if rng.random() < 0.5:
            # move mass from j → i
            max_delta = min(1.0 - p[i], p[j])
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta * 0.5
            new_i = p[i] + delta
            new_j = p[j] - delta
        else:
            # move mass from i → j
            max_delta = min(p[i], 1.0 - p[j])
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta * 0.5
            new_i = p[i] - delta
            new_j = p[j] + delta

        # construct candidate vector
        cand = p.copy()
        cand[i] = new_i
        cand[j] = new_j
        cand_val = eval_p(cand)

        # Metropolis acceptance
        if cand_val <= cur_val or rng.random() < math.exp((cur_val - cand_val) / T):
            p = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_p = cand.copy()
                best_val = cand_val

        it += 1

        # occasional random perturbation to escape stagnation
        if it % 200000 == 0:
            # small shake preserving sum
            shake = rng.uniform(-0.1, 0.1, size=half_n)
            shake = np.clip(shake, -best_p, 1.0 - best_p)
            # adjust to keep total sum
            diff = shake.sum()
            if abs(diff) > 1e-12:
                # redistribute the diff uniformly over free entries
                free = np.where((best_p > 0.0) & (best_p < 1.0))[0]

                cur_val = sh_val

    best_h = np.concatenate([best_p, best_p])
    final_val = _eval_h(best_h, dx)

    return best_h, float(final_val), n
