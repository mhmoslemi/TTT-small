def project_to_constraints(v, target, max_iter=100, tol=1e-9):
    lo = -1.0
    hi = max(v)  # ensures sum <= target maybe
    for _ in range(max_iter):
        tau = (lo + hi) / 2
        w = np.clip(v - tau, 0, 1)
        s = w.sum()
        if abs(s - target) < tol:
            return w
        if s > target:
            lo = tau
        else:
            hi = tau
    return np.clip(v - (lo+hi)/2, 0, 1)

def gradient_descent(h_init, steps, alpha, eta, target_sum):
    h = h_init.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    for s in range(steps):
        # compute smooth max with current alpha
        sum_h = h.sum()
        corr = np.correlate(h, h, mode='full')
        overlap = sum_h - corr
        max_o = overlap.max()
        exp_vals = np.exp(alpha * (overlap - max_o))
        w = exp_vals / exp_vals.sum()
        # gradient
        n = len(h)
        grad = np.ones(n)
        # loop over idx, wk
        # We can vectorize across positive k only and double weights since symmetry? But easier to include all lags.
        for idx, wk in enumerate(w):
            if wk == 0:
                continue
            k = idx - (n - 1)
            if k == 0:
                grad -= wk * 2 * h
            elif k > 0:
                grad[:n - k] -= wk * h[k:]
                grad[k:] -= wk * h[:n - k]
            else:
                kk = -k
                grad[kk:] -= wk * h[:n - kk]
                grad[:n - kk] -= wk * h[kk:]
        # update
        h = h - eta * grad
        # projection
        h = project_to_constraints(h, target_sum)
        # evaluate
        val = compute_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
        # optionally reduce eta
        eta = eta * 0.9995  # decay
    return best_h, best_val

def _project_to_constraints(v: np.ndarray, target: float) -> np.ndarray:

    lo, hi = -1.0, v.max()

        w = np.clip(v - tau, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = tau
        else:
            hi = tau

                    cur_h, cur_val = cand, cand_val
                eps = max(eps * 0.9999, 0.001)

            if cur_val < best_val:
                best_h, best_val, best_n = cur_h.copy(), cur_val, n

    # final polish on the best candidate found
    if best_h is not None:
        best_h, best_val = _smooth_max_descent(
            best_h,
            steps=2000,
            alpha=50.0,
            eta=0.1,
            target_sum=best_n / 2.0,
        )
    else:
        best_n = 48
        best_h = np.full(best_n, 0.5)
        best_val = _c5_of(best_h)

    # final safe projection and evaluation
    best_h = _project_to_constraints(best_h, best_n / 2.0)
    best_val = _c5_of(best_h)

    return best_h.tolist(), float(best_val), int(best_n)
