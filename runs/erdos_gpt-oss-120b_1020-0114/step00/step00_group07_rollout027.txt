import numpy as np
import time
import math

def _c5_value(vec: np.ndarray) -> float:
    """Compute C₅ for a step vector."""
    n = len(vec)
    dx = 2.0 / n
    # correlation of h with (1‑h) over the non‑circular domain
    corr = np.correlate(vec, 1.0 - vec, mode="full")
    return float(corr.max() * dx)


def _project_to_sum(vec: np.ndarray, target: float) -> np.ndarray:
    """
    Project `vec` onto the box [0,1] while enforcing sum(vec) == target.
    Finds λ such that x = clip(vec + λ, 0, 1) has the required sum.
    Uses a simple bisection, sufficient for the tiny dimensions used here.
    """
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(vec + lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(vec + lam, 0.0, 1.0)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Return a step function h (as an array) together with its C₅ bound.
    The routine starts from `initial_h_values` when provided; otherwise it
    falls back to a uniform ½‑function.  A simple stochastic local search
    improves the bound while respecting all constraints.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. obtain an initial feasible vector
    # ------------------------------------------------------------------
    try:
        h0 = np.asarray(initial_h_values, dtype=float)   # may be supplied by the harness
    except NameError:
        # fallback: uniform ½‑function (always feasible)
        n = 200
        h0 = np.full(n, 0.5, dtype=float)

    n = len(h0)
    target_sum = n / 2.0                     # because ∫₀² h = 1  →  sum(h)·dx = 1

    h = np.clip(h0, 0.0, 1.0)
    if not math.isclose(h.sum(), target_sum, rel_tol=1e-12, abs_tol=1e-12):
        h = _project_to_sum(h, target_sum)

    # ------------------------------------------------------------------
    # 2. simple stochastic improvement loop
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_c5 = _c5_value(best_h)

    start = time.time()
    # run until the allotted budget (leaving a safety margin)
    while time.time() - start < budget_s * 0.95:
        i, j = rng.choice(n, size=2, replace=False)

        # two‑direction move: either raise h[i] / lower h[j] or the opposite
        inc_max = min(1.0 - best_h[i], best_h[j])   # can increase i, decrease j

            new_h[i] += delta
            new_h[j] -= delta

            if delta == 0.0:

            new_h = best_h.copy()
            new_h[i] -= delta
            new_h[j] += delta

        c5 = _c5_value(new_h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = new_h
