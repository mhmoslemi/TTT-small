    v = np.clip(v, 0, 1)

def softmax_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx  # vector length M = 2n - 1
    # if we compute probability as softmax of g / T
    g_max = np.max(g)
    expg = np.exp((g - g_max) / T)
    p = expg / np.sum(expg)
    # Build matrix P
    idx = np.arange(n).reshape(-1, 1) - np.arange(n).reshape(1, -1) + (n - 1)
    P = p[idx]  # shape n x n
    grad = dx * (P @ (1 - h) - P.T @ h)
    return grad

Note: this gradient corresponds to derivative of softmax approximated max, not exact original objective. As T goes to 0, p approximates one-hot at argmax. Works.

import numpy as np

def _evaluate(h):
    n = h.size
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full') * dx)

def _project(v, target):
    """Project onto {0≤x≤1, sum x = target}."""
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        adj = np.clip(v + mid, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def _softmax_grad(h, T):
    n = h.size
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx           # length 2n‑1
    gmax = g.max()
    p = np.exp((g - gmax) / T)
    p /= p.sum()                                           # probabilities
    # matrix of probabilities indexed by i‑j
    idx = (np.arange(n)[:, None] - np.arange(n)[None, :]) + (n - 1)
    P = p[idx]                                             # n×n
    grad = dx * (P @ (1 - h) - P.T @ h)
    return grad
