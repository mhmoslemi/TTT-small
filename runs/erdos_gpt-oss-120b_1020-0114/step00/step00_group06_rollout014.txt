    rng = np.random.default_rng(seed)
    start = time.time()

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)

    # Simulated annealing parameters
    T0 = 0.1  # initial temperature
    T_end = 1e-4
    max_iters = 2000000  # high bound; we break by time.
    # We'll adapt T linearly with iteration ratio.
    it = 0
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s - 1.0:  # leave 1s margin
            break

        frac = it / (budget_s*1000) # approximate
        # Let's just keep constant step size small.

        i, j = rng.choice(n, size=2, replace=False)
        # Determine max delta

        max_dec = h[j]  # we can move from j to i
        # bound delta magnitude:
        # Choose direction: from j to i (increase i, decrease j)
        # allowable delta <= min(max_inc, max_dec)
        max_delta = min(max_inc, max_dec)
        # If max_delta is zero, skip
        if max_delta == 0:

        # random delta magnitude about step proportional to temperature
        # Temperature schedule:
        T = max(T0 * (1 - elapsed / budget_s), T_end)
        delta = rng.uniform(0, max_delta) * (T / T0)  # scale delta

        h_new[j] -= delta
        # ensure within bounds automatically
        # no need for projection since we kept within bounds, sum unchanged.
        # Evaluate C5
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5 - 1e-12:

    n = len(h)
    i, j = rng.integers(0, n, size=2, endpoint=False)

        new = h.copy()
        new[i] -= delta
        new[j] += delta
        return new
    else:

        new = h.copy()
        new[i] += delta
        new[j] -= delta
        return new

import numpy as np, time

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    conv = np.correlate(h, 1 - h, mode='full')
    max_val = conv.max() * dx
    return max_val

def project_to_sum_bounds(x, S):
    # Use water-filling binary search for λ
    lo = -1e3
    hi = 1e3
    for _ in range(50):
        mid = (lo + hi) / 2.0
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    # after binary search may be slight error, adjust simple
    diff = S - y.sum()
    if abs(diff) > 1e-8:
        # distribute diff to free entries
        free = (y > 0) & (y < 1)
        n_free = free.sum()
        if n_free > 0:
            y[free] += diff / n_free
            y = np.clip(y, 0.0, 1.0)
        else:
            # fallback: adjust uniformly
            y += diff / len(y)
            y = np.clip(y, 0.0, 1.0)
    return y

def gradient_O(h, lag):
    N = len(h)
    g = np.zeros(N)
    if lag >= 0:
        L = lag
        if L < N:
            idx1 = np.arange(N - L)  # 0 .. N-L-1
            g[idx1] += 1.0
            g[idx1] -= h[idx1 + L]
            idx2 = np.arange(L, N)  # L .. N-1
            g[idx2] -= h[idx2 - L]
    else:
        L = -lag
        if L < N:
            idx1 = np.arange(L, N)  # L .. N-1
            g[idx1] += 1.0
            idx2 = np.arange(N - L)  # 0 .. N-L-1
            g[idx2] -= h[idx2 + L]
            idx3 = np.arange(L, N)
            g[idx3] -= h[idx3 - L]
    return g

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    start = time.time()
    # initial vector
    if 'initial_h_values' in globals():

    N = h.shape[0]
    S = N / 2.0
    h = project_to_sum_bounds(h, S)
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    # Parameters
    max_iter = 2000000
    # step schedule
    initial_alpha = 1.0  # maybe large
    # Actually we need step size small to not overshoot
    # Determine scale relative to typical g values (order 1). We'll set alpha0 = 0.5
    alpha0 = 0.5
    # Determine number of iterations based on time
    it = 0
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s - 0.5:
            break
        # compute correlation

        max_val = conv.max()
        # compute lag where max occurs (first index)
        idx_max = int(np.argmax(conv))
        lag = idx_max - (N - 1)
        # compute gradient for this worst lag

        # step size decreases

        h_new = project_to_sum_bounds(h_new, S)

                cur_c5 = new_c5
        # update best
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        it += 1
    # final evaluation
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), float(final_c5), int(best_h.shape[0]))

import time, math
import numpy as np

def _compute_c5(h):
    """Return the C5 value for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(conv.max() * dx)


def _project_to_sum(v, target_sum):
    """Project v onto the box [0,1] and the hyperplane sum(v)=target_sum."""
    # water‑filling: find λ such that sum clip(v-λ,0,1)=target_sum
    lo, hi = -1e3, 1e3
    for _ in range(60):
        lam = (lo + hi) * 0.5
        y = np.clip(v - lam, 0.0, 1.0)
        s = y.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    y = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction to eliminate residual error
    diff = target_sum - y.sum()
    if abs(diff) > 1e-12:
        free = (y > 0.0) & (y < 1.0)
        cnt = free.sum()
        if cnt:
            y[free] += diff / cnt
            y = np.clip(y, 0.0, 1.0)
        else:
            y += diff / y.size
            y = np.clip(y, 0.0, 1.0)
    return y


def _gradient_O(h, lag):
    """
    Sub‑gradient of O_k = Σ h_i (1‑h_{i+lag}) for the lag that attains the maximum.
    Implements: g_i = 1 – (cond1·h_{i+lag}+cond2·h_{i‑lag}),
    where cond1/cond2 indicate that the partner index lies inside the array.
    """
    n = h.shape[0]
    g = np.ones(n, dtype=np.float64)
    if lag >= 0:
        L = lag
        if L < n:
            g[: n - L] -= h[L:]          # subtract h_{i+L}
            g[L:]      -= h[: n - L]      # subtract h_{i‑L}
    else:
        L = -lag
        if L < n:
            g[L:]      -= h[: n - L]      # subtract h_{i‑L}
            g[: n - L] -= h[L:]          # subtract h_{i+L}
    return g


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to lower the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # initialise
    if "initial_h_values" in globals():
        h = np.asarray(initial_h_values, dtype=np.float64).copy()
    else:
        # default size if no parent is given
        n = 200
        h = np.full(n, 0.5, dtype=np.float64)

    n = h.shape[0]
    target_sum = n / 2.0
    h = _project_to_sum(h, target_sum)

    cur_h = h.copy()
    cur_c5 = _compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    # ------------------------------------------------------------------
    # optimisation parameters
    alpha0 = 0.8                     # initial step size
    it = 0
    # main loop – stops when budget exhausted
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s - 0.5:
            break

        # correlation and worst lag
        conv = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        max_idx = int(np.argmax(conv))
        lag = max_idx - (n - 1)

        # sub‑gradient for that lag
        g = _gradient_O(cur_h, lag)

        # step size with linear decay
        frac = elapsed / budget_s
        alpha = alpha0 * (1.0 - frac) + 1e-6

        # propose new point, keep in [0,1] and correct the sum
        cand = cur_h - alpha * g
        cand = np.clip(cand, 0.0, 1.0)
        cand = _project_to_sum(cand, target_sum)

        cand_c5 = _compute_c5(cand)

        # simple acceptance rule (always accept improvements)
        if cand_c5 < cur_c5:
            cur_h, cur_c5 = cand, cand_c5
        else:
            # occasional uphill move (simulated‑annealing style)
            T = max(0.01, (budget_s - elapsed) / budget_s)
            if rng.random() < math.exp(-(cand_c5 - cur_c5) / T):
                cur_h, cur_c5 = cand, cand_c5

        # record global best
        if cur_c5 < best_c5:
            best_c5, best_h = cur_c5, cur_h.copy()

        # occasional random jitter to escape stagnation
        if it % 4000 == 0 and it > 0:
            jitter = rng.normal(scale=0.04, size=n)
            cur_h = np.clip(cur_h + jitter, 0.0, 1.0)
            cur_h = _project_to_sum(cur_h, target_sum)
            cur_c5 = _compute_c5(cur_h)

        it += 1

    return (best_h.tolist(), float(best_c5), int(n))
