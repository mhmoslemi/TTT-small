def project_to_constraints(h, target):
    # Clip
    h = np.clip(h, 0, 1)
    # Adjust sum with water-filling
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # need to increase
        mask = h < 1 - 1e-12
        if not np.any(mask):
            return h
        capacities = 1 - h[mask]
        total_capacity = capacities.sum()
        # If diff > total_capacity due to rounding, just set to max
        diff = min(diff, total_capacity)
        # Update increments proportionally
        inc = diff * capacities / total_capacity
        h[mask] += inc
    else:
        # need to decrease
        mask = h > 1e-12
        capacities = h[mask]
        total_capacity = capacities.sum()
        diff_abs = min(-diff, total_capacity)
        dec = diff_abs * capacities / total_capacity
        h[mask] -= dec
    return h

import numpy as np, time

def project_to_constraints(h, target):
    # clip
    h = np.clip(h, 0.0, 1.0)
    # compute difference
    diff = target - h.sum()
    # tolerance if near zero
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # increase
        mask = h < 1.0 - 1e-12
        capacities = 1.0 - h[mask]
        total_cap = capacities.sum()
        if total_cap == 0:
            # nothing to increase, just return
            return h
        # allocate proportionally
        inc = diff * capacities / total_cap
        h[mask] += inc
    else:
        # decrease
        mask = h > 0.0 + 1e-12
        capacities = h[mask]
        total_cap = capacities.sum()
        if total_cap == 0:
            return h
        dec = -diff * capacities / total_cap
        h[mask] -= dec
    return h
```

def compute_c5_and_smooth(h, alpha):
    n = h.shape[0]
    target = h.sum()
    conv = np.correlate(h, h, mode='full')
    f = target - conv  # array length 2n-1

    max_f = f.max()
    # log-sum-exp
    # scaling for stability
    exp_term = np.exp(alpha * (f - max_f))
    sum_exp = exp_term.sum()
    weights = exp_term / sum_exp  # shape 2n-1

    # gradient
    g = np.ones_like(h)
    # loop offset
    for idx, L in enumerate(range(-n+1, n)):
        w = weights[idx]
        if w == 0:
            continue
        if L >= 0:
            # forward shift
            if L < n:
                g[:n - L] -= w * h[L:]
                g[L:] -= w * h[:n - L]
        else:
            k = -L
            g[:n - k] -= w * h[k:]
            g[k:] -= w * h[:n - k]
    # compute smoothed max value
    smooth_max = max_f + np.log(sum_exp) / alpha
    c5_smooth = smooth_max * (2.0 / n)
    c5_actual = f.max() * (2.0 / n)
    grad = g * (2.0 / n)  # gradient of smooth objective
    return c5_actual, c5_smooth, grad

import numpy as np, time

def project_to_constraints(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
        cap = 1.0 - h[mask]
        total = cap.sum()
        if total > 0:
            inc = diff * cap / total
            h[mask] += inc
    else:
        mask = h > 0.0
        cap = h[mask]
        total = cap.sum()
        if total > 0:
            dec = -diff * cap / total
            h[mask] -= dec
    return h

def compute_c5_and_grad(h, alpha):
    n = h.shape[0]
    target = h.sum()
    conv = np.correlate(h, h, mode='full')
    f = target - conv        # length 2n-1
    max_f = f.max()
    # smooth max via log-sum-exp
    # shift by max_f for numerical stability
    exp_term = np.exp(alpha * (f - max_f))
    sum_exp = exp_term.sum()
    weights = exp_term / sum_exp
    # gradient of smooth max
    g = np.ones_like(h)
    # loop over shifts
    # offsets range -n+1 .. n-1
    for idx, L in enumerate(range(-n+1, n)):
        w = weights[idx]
        if w == 0.0:
            continue
        if L >= 0:
            if L < n:
                # forward
                g[:n - L] -= w * h[L:]
                # backward
                g[L:] -= w * h[:n - L]
        else:
            k = -L
            g[:n - k] -= w * h[k:]
            g[k:] -= w * h[:n - k]
    dx = 2.0 / n
    smooth_max = max_f + np.log(sum_exp) / alpha
    c5_smooth = smooth_max * dx
    c5_actual = (target - conv.min()) * dx
    grad = g * dx
    return c5_actual, c5_smooth, grad

    n = h.shape[0]
    target = h.sum()
    conv = np.correlate(h, h, mode='full')
    return (target - conv.min()) * (2.0 / n)

    target = n / 2.0
    if init_vec is not None:
        if init_vec.size != n:
            # interpolate

            h = np.interp(x_new, x_old, init_vec)
        else:
            h = init_vec.copy()
    else:
        h = rng.rand(n)
    h = project_to_constraints(h, target)
    best_h = h.copy()

    # parameters

import numpy as np
import time

def _project_to_constraints(h, target):
    """Clip to [0,1] and force sum == target by a water‑filling step."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0.0:                     # need to increase the sum
        mask = h < 1.0
        cap = 1.0 - h[mask]
        tot = cap.sum()
        if tot > 0.0:
            inc = diff * cap / tot
            h[mask] += inc
    else:                               # need to decrease the sum
        mask = h > 0.0
        cap = h[mask]
        tot = cap.sum()
        if tot > 0.0:
            dec = -diff * cap / tot
            h[mask] -= dec
    return h


def _c5_and_grad(h, alpha):
    """Return actual C5, smoothed C5 and the gradient of the smoothed value."""
    n = h.shape[0]
    target = h.sum()
    conv = np.correlate(h, h, mode='full')          # h * h
    f = target - conv                               # f(L) = Σh - Σh·h_shift
    max_f = f.max()
    # smooth approximation of max via log‑sum‑exp
    exp_term = np.exp(alpha * (f - max_f))
    sum_exp = exp_term.sum()
    w = exp_term / sum_exp                          # weights, sum to 1

    # gradient of the smoothed max
    g = np.ones_like(h)
    for idx, L in enumerate(range(-n + 1, n)):
        wl = w[idx]
        if wl == 0.0:
            continue
        if L >= 0:
            if L < n:
                g[: n - L] -= wl * h[L:]           # forward partner
                g[L:]      -= wl * h[: n - L]      # backward partner
        else:
            k = -L
            g[: n - k] -= wl * h[k:]                # forward partner (L negative)
            g[k:]      -= wl * h[: n - k]           # backward partner

    dx = 2.0 / n
    smooth_max = max_f + np.log(sum_exp) / alpha
    c5_smooth = smooth_max * dx
    c5_actual = (target - conv.min()) * dx          # true C5 = dx·max f(L)
    grad = g * dx                                    # gradient of smooth C5
    return c5_actual, c5_smooth, grad


def _c5_actual(h):
    """Exact C5 using the definition (no smoothing)."""
    n = h.shape[0]
    target = h.sum()
    conv = np.correlate(h, h, mode='full')
    return (target - conv.min()) * (2.0 / n)


def _optimize_one(n, budget, rng, init_vec=None):
    """Gradient descent with projection; returns a feasible h and its C5."""
    target = n / 2.0
    if init_vec is not None:
        if init_vec.size != n:
            x_old = np.linspace(0.0, 2.0, init_vec.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(x_new, x_old, init_vec)
        else:
            h = init_vec.copy()
    else:
        h = rng.rand(n)
    h = _project_to_constraints(h, target)

    best_h = h.copy()
    best_c5 = _c5_actual(h)

    alpha = 5.0
    step0 = 0.7
    it = 0
    t0 = time.time()
    while time.time() - t0 < budget:
        c5_act, c5_smooth, grad = _c5_and_grad(h, alpha)

        eta = step0 / np.sqrt(it + 1.0)
        h_new = _project_to_constraints(h - eta * grad, target)

        new_c5 = _c5_actual(h_new)

        if new_c5 < best_c5 - 1e-8:
            best_c5 = new_c5
            best_h = h_new.copy()

        # keep the move if it does not worsen the true objective
        if new_c5 <= c5_act + 1e-12:
            h = h_new
        else:
            if rng.rand() < 0.01:          # occasional uphill move
                h = h_new

        it += 1
        if it % 3000 == 0:

        if it % 5000 == 0 and rng.rand() < 0.3:
            # random restart to escape possible plateaus
            h = _project_to_constraints(rng.rand(n), target)

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    start = time.time()
    best_h, best_c5, best_n = None, float('inf'), None

    # possible grid sizes – small enough for fast evaluation yet expressive
    sizes = [120, 140, 160]

    # parent construction if provided

    if best_h is None:

        best_h = np.full(n, 0.5)
        best_c5 = _c5_actual(best_h)
        best_n = n

    return best_h, best_c5, best_n
