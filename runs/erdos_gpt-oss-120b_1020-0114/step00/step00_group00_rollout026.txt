def optimize_sequence(h0, max_iter, time_limit, n):
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    lr = 0.1
    alpha = 5.0
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        # compute f_s
        f = compute_f_s(h)
        max_f = f.max()
        # compute softmax weights
        exp_f = np.exp(alpha * (f - max_f))
        weights = exp_f / exp_f.sum()
        # compute gradient using vectorized loop
        grad = np.zeros_like(h)
        for s, w in enumerate(weights):
            if w == 0.0:
                continue
            L = n - s
            if L:
                grad[:L] += w * (1.0 - h[s:s+L])
            if s > 0:
                grad[s:] -= w * h[:n-s]
        # update
        h = h - lr * grad
        # project
        h = project_onto_sum_and_box(h, n/2)
        # evaluate
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        # adapt learning rate
        if (it+1) % 100 == 0:
            lr *= 0.9
            alpha = min(alpha * 1.2, 200.0)
    return best_h, best_c
```

def init_local():
    if 'initial_h_values' in globals():
        base = np.array(initial_h_values, dtype=float)
        # upsample to n by replication (if length divides) else interpolation.
        if n % base.shape[0] == 0:
            repeat = n // base.shape[0]
            h = np.repeat(base, repeat)
        else:
            # linear interpolation
            old_x = np.linspace(0, 2, base.shape[0], endpoint=False)
            new_x = np.linspace(0, 2, n, endpoint=False)
            h = np.interp(new_x, old_x, base)
        # ensure sum constraint via projection
        h = project_onto_sum_and_box(h, n/2)
    else:
        # random init
        h = np.random.rand(n)
        h = project_onto_sum_and_box(h, n/2)

def _project_onto_sum_and_box(y, target):
    """Project y onto the set {x | 0<=x<=1, sum(x)=target}."""
    lo, hi = -1.0, 1.0

        return np.clip(y - lam, 0.0, 1.0).sum()

    # expand bounds until they bracket the solution
    while _s(lo) < target:
        lo -= 1.0

        else:

                grad[s:] -= ws * h[:n - s]

        h = h - lr * grad
        h = _project_onto_sum_and_box(h, n / 2.0)

        c = _c5_of(h)
        if c < best_c - 1e-12:
            best_c, best_h = c, h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # simple schedule
        if (it + 1) % 200 == 0:
            lr *= 0.9
            alpha = min(alpha * 1.2, 200.0)
        if no_improve >= 600:
            break
        if best_c <= 0.3807:
            break

    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C₅ bound, and the number of samples."""
    np.random.seed(seed)
    n_points = 180                     # a modest size that still allows refinement
    start = time.time()
    deadline = start + budget_s - 0.2   # tiny safety margin

    best_h, best_c = None, float('inf')
    # keep trying fresh starts until time runs out or we reach the target
    while time.time() < deadline:
        h0 = _init_h(n_points)
        # allocate a generous iteration budget; the inner loop also watches the clock
        h_opt, c_opt = _optimize_one_start(h0, max_iters=5000, time_deadline=deadline)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt
        if best_c <= 0.3807:          # good enough, stop early
            break

    # fallback – should never happen, but guarantees a return value
    if best_h is None:
        best_h = _init_h(n_points)
        best_c = _c5_of(best_h)

    return best_h, best_c, n_points
