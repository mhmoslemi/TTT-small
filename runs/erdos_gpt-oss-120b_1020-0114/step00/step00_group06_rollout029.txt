def project_onto_box_and_sum(v, target):

def project_to_feasible(v, target):
    # Find lambda such that sum clip(v - lambda, 0, 1) = target
    # Use binary search.
    # initial bounds: low = -1 (ensuring all v - lambda >= 1, clipping at 1), high = 1 (ensuring all v - lambda <=0, clipping at 0).
    low = -2.0
    high = 2.0
    for _ in range(60):
        mid = (low + high)/2
        proj = np.clip(v - mid, 0, 1)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    return np.clip(v - (low+high)/2, 0, 1)

    # Ensure sum OK: compute sum; project.
    h = project_to_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    # gradient descent
    step0 = 0.5
    t = 0
    start_time = time.time()
    while time.time() - start_time < budget * 0.98:
        # compute corr and worst lag
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        lag = max_idx - (N - 1)  # shift k
        # compute gradient
        g = np.ones(N)
        if lag >= 0:
            # subtract forward shift
            g[:N - lag] -= h[lag:]
            # subtract backward shift
            g[lag:] -= h[:N - lag]
        else:
            lag = -lag
            # shift negative: same logic (since absolute)
            g[lag:] -= h[:N - lag]
            g[:N - lag] -= h[lag:]
        # step size

        # update

        # project

        # evaluate

import numpy as np, time

def project_to_feasible(v, target):
    # binary search for lambda to make sum of clipped (v - lam) = target
    low = -2.0
    high = 2.0
    for _ in range(60):
        mid = (low + high) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    return np.clip(v - (low + high) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def worst_lag_and_corr(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    lag = idx - (n - 1)
    return lag, corr[idx]

def gradient_worst_lag(h, lag):
    n = len(h)
    g = np.ones(n)
    if lag >= 0:
        k = lag
        if k < n:
            g[:n - k] -= h[k:]
            g[k:] -= h[:n - k]
    else:
        k = -lag
        if k < n:
            g[k:] -= h[:n - k]
            g[:n - k] -= h[k:]
    return g

    # target sum = n/2
    target = n / 2.0

    h = project_to_feasible(h0, target)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start = time.time()
    # Parameters
    step0 = 0.7  # initial step size
    iteration = 0
    # Main loop
    while True:
        elapsed = time.time() - start

        # compute worst lag
        lag, cur_val = worst_lag_and_corr(h)
        # gradient of that term
        g = gradient_worst_lag(h, lag)
        # step size schedule
        alpha = step0 / np.sqrt(iteration + 1)
        h_new = h - alpha * g
        # project

import time
import numpy as np

def _project_to_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto the box [0,1]^n with a fixed sum."""
    # binary search for a shift λ so that sum(clamp(v-λ,0,1)) = target_sum
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _c5_of(h: np.ndarray) -> float:
    """C₅ value for a given step vector h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _worst_lag(h: np.ndarray):
    """Return (lag, correlation value) of the worst shift."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    lag = idx - (n - 1)          # integer shift
    return lag, corr[idx]


def _gradient_worst(h: np.ndarray, lag: int) -> np.ndarray:
    """Sub‑gradient of the worst lag term (∑h_i(1‑h_{i+lag}))."""
    n = h.size
    g = np.ones(n)
    if lag >= 0:
        k = lag
        if k < n:
            g[: n - k] -= h[k:]
            g[k:] -= h[: n - k]
    else:
        k = -lag
        if k < n:
            g[k:] -= h[: n - k]
            g[: n - k] -= h[k:]
    return g


def run(seed: int = 42, budget_s: float = 1100, **_):
    rng = np.random.default_rng(seed)

    # ----- determine problem size -----
    if "initial_h_values" in globals():
        h0_raw = np.asarray(initial_h_values, dtype=float)
        n0 = h0_raw.size
        N = max(200, n0 * 2)          # a denser grid than the parent
        # interpolate parent onto the new grid
        dx0 = 2.0 / n0
        xs_old = (np.arange(n0) + 0.5) * dx0
        dx = 2.0 / N
        xs_new = (np.arange(N) + 0.5) * dx
        h0 = np.interp(xs_new, xs_old, h0_raw)
    else:
        N = 200
        dx = 2.0 / N
        xs_new = (np.arange(N) + 0.5) * dx
        h0 = (1.0 + np.cos(np.pi * xs_new)) * 0.5   # smooth start

    target = N / 2.0                     # because Σh = N/2 ⇔ integral = 1
    h0 = _project_to_sum(h0, target)

    best_h = h0.copy()
    best_val = _c5_of(best_h)

    start_time = time.time()
    restart = 0

    # main multi‑restart loop
    while time.time() - start_time < budget_s * 0.94:
        # ----- initialise a fresh candidate -----
        if restart == 0 and "initial_h_values" in globals():
            h = h0.copy()
        elif restart == 1:
            # cosine start on the same grid
            xs = (np.arange(N) + 0.5) * (2.0 / N)
            h = (1.0 + np.cos(np.pi * xs)) * 0.5
            h = _project_to_sum(h, target)
        else:
            # random start
            h = rng.random(N)
            h = _project_to_sum(h, target)

        cur_val = _c5_of(h)
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()

        # ----- inner sub‑gradient optimisation -----
        step0 = 0.7
        it = 0
        max_it = 5000
        while it < max_it and time.time() - start_time < budget_s * 0.94:
            lag, _ = _worst_lag(h)
            grad = _gradient_worst(h, lag)
            alpha = step0 / np.sqrt(it + 1.0)

            h_new = h - alpha * grad
            h_new = _project_to_sum(h_new, target)

            new_val = _c5_of(h_new)

            # keep globally best
            if new_val < best_val:
                best_val = new_val
                best_h = h_new.copy()

            # simple acceptance: keep improvement, otherwise stay
            if new_val <= cur_val or rng.random() < 0.05:
                h = h_new
                cur_val = new_val

            # occasional random perturbation to escape flat regions
            if it % 400 == 0 and it > 0:

                h_alt = h + pert
                h_alt = _project_to_sum(h_alt, target)
                alt_val = _c5_of(h_alt)
                if alt_val < cur_val:
                    h = h_alt
                    cur_val = alt_val

            it += 1

        restart += 1

    # final clean‑up
    best_val = _c5_of(best_h)
    return best_h, float(best_val), int(N)
