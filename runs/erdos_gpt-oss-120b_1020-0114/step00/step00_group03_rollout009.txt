import numpy as np, time, math, random

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    # correlation
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def optimize_h(initial_h, max_iter, time_budget):
    n = len(initial_h)
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5

    rng = np.random.default_rng()
    start = time.time()
    iter = 0
    T0 = 0.01
    T = T0
    decay = 0.9999 # adjust

    while time.time() - start < time_budget:
        # propose move:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # direction: move from j to i
        delta_max = min(1 - h[i], h[j])
        if delta_max <= 1e-12:
            continue
        delta = rng.random() * delta_max * 0.5 # limit size
        # apply
        h[i] += delta
        h[j] -= delta
        new_c5 = compute_c5(h)
        # accept?
        if new_c5 < cur_c5 or rng.random() < math.exp((cur_c5 - new_c5) / (T + 1e-12)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert
            h[i] -= delta
            h[j] += delta
        iter += 1
        # schedule temperature
        T *= decay
        # maybe occasional big move to escape
        if iter % 1000 == 0:
            # optionally increase T a bit
            pass
    return best_h, best_c5

import time, math
import numpy as np

# ------------------------------------------------------------
# Helper functions
# ------------------------------------------------------------
def compute_c5(h):
    """Return max_k ∫ h(x)*(1-h(x+k)) dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def worst_shift_info(h):
    """Return (shift, correlation array) for the current worst shift."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)          # integer shift relative to zero
    return shift, corr


def gradient_for_shift(h, shift):
    """Gradient of the worst‑shift correlation w.r.t. h."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        # (1 - h[i+shift]) part
        g[: n - shift] += 1.0 - h[shift:]
        # -h[i-shift] part
        g[shift:] -= h[: n - shift]
    else:
        s = -shift
        g[s:] += 1.0 - h[: n - s]
        g[: n - s] -= h[s:]
    return g


def project_to_feasible(a, target_sum, eps=1e-12):
    """Project vector a onto the box [0,1] with equality sum == target_sum."""
    lo, hi = -1.0, 2.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(a - lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    x = np.clip(a - (lo + hi) / 2.0, 0.0, 1.0)

    diff = target_sum - x.sum()
    if abs(diff) > eps:
        mask = (x > 0.0) & (x < 1.0)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
    return x


def upscale_h(h, new_n):
    """Linearly interpolate h to a new length new_n and re‑project."""
    old_n = h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    # enforce exact integral
    target_sum = new_n / 2.0
    h_new = project_to_feasible(h_new, target_sum)
    return h_new


# ------------------------------------------------------------
# Core optimisation routine
# ------------------------------------------------------------
def hill_climb(start_h, max_iters, rng):
    """Simple hill‑climbing using pairwise mass transfers and a
    gradient‑guided move on the worst shift."""
    h = start_h.copy()
    n = h.shape[0]
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step_factor = 0.07

    for it in range(max_iters):
        # 25 % chance to use a gradient‑guided move
        if rng.random() < 0.25:
            shift, _ = worst_shift_info(h)
            g = gradient_for_shift(h, shift)
            pos = np.where(g > 1e-12)[0]
            neg = np.where(g < -1e-12)[0]
            if pos.size and neg.size:
                i = rng.choice(pos)
                j = rng.choice(neg)
                delta_max = min(h[i], 1.0 - h[j])
                if delta_max > 0.0:
                    delta = rng.random() * delta_max * step_factor
                    h[i] -= delta
                    h[j] += delta
                    new_c5 = compute_c5(h)
                    if new_c5 <= cur_c5:
                        cur_c5 = new_c5
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = h.copy()
                    else:
                        h[i] += delta
                        h[j] -= delta
            continue   # skip the random move this iteration

        # random pair move
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # decide direction randomly
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        delta_max = min(h[src], 1.0 - h[dst])
        if delta_max <= 0.0:
            continue
        delta = rng.random() * delta_max * step_factor
        h[src] -= delta
        h[dst] += delta
        new_c5 = compute_c5(h)
        if new_c5 <= cur_c5:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[src] += delta
            h[dst] -= delta

        # slowly shrink the step size
        if (it + 1) % 4000 == 0:
            step_factor = max(step_factor * 0.7, 5e-4)

    return best_h, best_c5

    while time.time() - start_time < budget_s * 0.95:
