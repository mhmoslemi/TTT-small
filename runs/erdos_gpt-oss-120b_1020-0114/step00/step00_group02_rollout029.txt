def compute_c5(h):
    n = len(h)
    b = 1.0 - h
    b_rev = b[::-1]
    N = 1 << ((2*n-1).bit_length())
    fft_a = np.fft.rfft(h, N)  # real FFT
    fft_b = np.fft.rfft(b_rev, N)
    conv = np.fft.irfft(fft_a * fft_b, N)
    # conv length N, take first 2*n-1 elements
    corr = conv[:2*n-1]
    c5 = corr.max() * (2.0 / n)
    return c5

def compute_c5(h):
    n = h.shape[0]
    b_rev = (1.0 - h)[::-1]
    N = 1 << ((2*n-1).bit_length())
    # compute real fft
    fft_h = np.fft.rfft(h, n=N)
    fft_b = np.fft.rfft(b_rev, n=N)
    conv = np.fft.irfft(fft_h * fft_b, n=N)
    corr = conv[:2*n-1]
    return corr.max() * (2.0 / n)

    b = 1.0 - h
    N = 1 << ((2*n-1).bit_length())

    c5 = corr.max() * (2.0/n)

def compute_c5(h):
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(h, N)
    fb = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(fa * np.conj(fb), N)
    corr = corr[:2 * n - 1]
    return np.max(corr) * (2.0 / n)

import numpy as np, math, time, random

def compute_c5(h):
    n = h.shape[0]
    N = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(h, N)
    fb = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(fa * np.conj(fb), N)
    corr = corr[:2*n-1]
    return np.max(corr) * (2.0 / n)

def random_binary_vector(n):
    half = n // 2
    h = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, half, replace=False)
    h[idx] = 1.0
    return h

def build_two_level_vector(n, w):
    # w is length of each high block (number of points)
    if w == 0:
        c = 0.5
        return np.full(n, c, dtype=np.float64)
    c = (n/2 - 2*w) / (n - 2*w)
    h = np.full(n, c, dtype=np.float64)
    h[:w] = 1.0
    if w > 0:
        h[-w:] = 1.0
    return h

def project_to_sum_one_half(h):
    # Ensure sum = n/2 and clip to [0,1]
    n = h.shape[0]
    target = n/2
    h = np.clip(h, 0.0, 1.0)
    # water-filling simple
    diff = target - h.sum()
    # apply diff uniformly while respecting bounds, iterating a few times
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        add = diff / cnt
        h[mask] += add
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
    # final tiny adjustment
    if abs(diff) > 1e-12:
        idx = 0
        while True:
            if diff > 0 and h[idx] < 1.0:
                inc = min(diff, 1.0 - h[idx])
                h[idx] += inc
                diff -= inc
            elif diff < 0 and h[idx] > 0.0:
                dec = min(-diff, h[idx])
                h[idx] -= dec
                diff += dec
            idx = (idx + 1) % n
            if abs(diff) < 1e-12:
                break
    return h

            cur_h[j] += delta
            no_improve += 1

        if no_improve > 50000:

            cur_c5 = compute_c5(cur_h)
            no_improve = 0

        # early stop if reached target bound
        if best_c5 <= 0.3808:

    # Final projection to enforce constraints exactly (should already be satisfied)
    best_h = np.clip(best_h, 0.0, 1.0)
    # adjust sum exactly
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-9:
        # distribute diff to first element within bounds
        best_h[0] += diff
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n)

    n = h.shape[0]

    fa = np.fft.rfft(h, N)

    sc = sc[:2*n-1]
    # find lag with minimum sc
    # We need to map index to lag: lag = idx - (n-1)
    idx_min = np.argmin(sc)
    lag = idx_min - (n - 1)
    # compute gradient vector g_i
    g = np.zeros_like(h)

    if lag >= 0:
        g[:n-lag] += h[lag:]  # h[i+lag]
        g[lag:] += h[:n-lag]  # h[i-lag] for i>=lag? Actually second term: h[i-lag] exists for i>=lag
    else:

    # pick idx_up with max g among those h<1

def _compute_c5(h: np.ndarray) -> float:
    """C5 = max_k  ∫ h(x)*(1-h(x+k))dx  approximated on a uniform grid."""
    n = h.shape[0]
    # linear correlation of h with (1-h) via FFT
    N = 1 << ((2 * n - 1).bit_length())      # next power of two ≥ 2n-1
    fa = np.fft.rfft(h, N)
    fb = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(fa * np.conj(fb), N)
    corr = corr[: 2 * n - 1]                 # keep the linear part
    return float(corr.max() * (2.0 / n))


def _random_binary_vector(n: int) -> np.ndarray:
    """Return a feasible vector with exactly n/2 entries equal to 1."""
    half = n // 2
    h = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, half, replace=False)
    h[idx] = 1.0
    return h


def _two_level_vector(n: int, w: int) -> np.ndarray:
    """
    Construct a vector that has value 1 on two edge blocks of length w
    and a constant value c on the interior, satisfying the sum constraint.
    """
    if w == 0:
        return np.full(n, 0.5, dtype=np.float64)
    c = (n / 2 - 2 * w) / (n - 2 * w)
    h = np.full(n, c, dtype=np.float64)
    h[:w] = 1.0
    h[-w:] = 1.0
    return h


def _project_to_feasible(h: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and force sum(h) = n/2 by a short water‑filling loop."""
    n = h.shape[0]
    target = n / 2
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        add = diff / cnt
        h[mask] += add
        np.clip(h, 0.0, 1.0, out=h)
    # tiny final tweak
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # distribute the remainder sequentially
        i = 0
        while abs(diff) > 1e-12:
            if diff > 0 and h[i] < 1.0:
                inc = min(diff, 1.0 - h[i])
                h[i] += inc
                diff -= inc
            elif diff < 0 and h[i] > 0.0:
                dec = min(-diff, h[i])
                h[i] -= dec
                diff += dec
            i = (i + 1) % n
    return h


def _improve_worst_shift(h: np.ndarray, step: float) -> bool:
    """
    Deterministic move aimed at the shift with minimal self‑correlation.
    Returns True if a move was performed (regardless of improvement).
    """
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(h, N)
    sc = np.fft.irfft(fa * np.conj(fa), N)[: 2 * n - 1]

    idx_min = int(np.argmin(sc))
    lag = idx_min - (n - 1)                     # lag ∈ [-(n-1), n-1]

    # gradient of the selected correlation term
    g = np.zeros_like(h)
    if lag >= 0:
        g[: n - lag] += h[lag:]                 # h_i * h_{i+lag}
        g[lag:] += h[: n - lag]                 # symmetric part
    else:
        l = -lag
        g[l:] += h[: n - l]
        g[: n - l] += h[l:]

    # choose an index to increase (large gradient) and one to decrease (small gradient)
    up_mask = h < 1.0 - 1e-12
    down_mask = h > 1e-12
    if not up_mask.any() or not down_mask.any():
        return False

    up_idx = np.argmax(g * up_mask)            # where g is biggest among feasible up‑moves
    down_idx = np.argmin(g * down_mask)        # where g is smallest among feasible down‑moves

    inc = 1.0 - h[up_idx]
    dec = h[down_idx]
    delta = min(step, inc, dec)
    if delta <= 0.0:
        return False

    h[up_idx] += delta
    h[down_idx] -= delta
    return True


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step function h on [0,2] with ∫h = 1, 0≤h≤1 that minimises
    C₅ = maxₖ ∫ h(x)(1−h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------
    # 1. Choose a resolution (even number of points, moderately small)
    # ------------------------------------------------------------
    n = 400                         # 400 points gives dx = 0.005
    target_sum = n / 2

    # ------------------------------------------------------------
    # 2. Build an initial feasible vector
    # ------------------------------------------------------------
    if 'initial_h_values' in globals():
        # resample the provided construction onto the new grid
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n0 = h0.size
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x_new, x0, h0, left=0.0, right=0.0)
        h = _project_to_feasible(h)
    else:
        h = _random_binary_vector(n)

    best_h = h.copy()
    best_c5 = _compute_c5(best_h)

    # ------------------------------------------------------------
    # 3. Quick scan of simple two‑level families (edge blocks of 1's)
    # ------------------------------------------------------------
    for w in range(0, n // 4 + 1):
        cand = _two_level_vector(n, w)
        c5 = _compute_c5(cand)
        if c5 < best_c5:
            best_c5 = c5
            best_h = cand.copy()

    # ------------------------------------------------------------
    # 4. Simulated‑annealing style local search
    # ------------------------------------------------------------
    cur_h = best_h.copy()
    cur_c5 = best_c5

    T_start = 0.5
    T_end = 1e-4
    max_delta_initial = 0.5            # maximal mass transferred in one move
    improve_interval = 2000           # how often we try the deterministic improvement
    steps_without_improve = 0
    total_iters = 0

    while True:
        # time management
        now = time.time()
        elapsed = now - start_time
        if elapsed >= budget_s * 0.95:      # leave a small safety margin
            break

        # linear cooling schedule
        T = T_end + (T_start - T_end) * (1.0 - elapsed / budget_s)

        # ---- stochastic pairwise transfer ----
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        inc_i = 1.0 - cur_h[i]
        dec_i = cur_h[i]

        max_pos = min(inc_i, cur_h[j])           # transfer from j → i
        max_neg = min(dec_i, 1.0 - cur_h[j])     # transfer from i → j
        if max_pos + max_neg < 1e-12:
            continue

        # step size adapts with temperature
        step_limit = max_delta_initial * (T / T_start) + 1e-12
        max_pos = min(max_pos, step_limit)
        max_neg = min(max_neg, step_limit)

        delta = np.random.uniform(-max_neg, max_pos)
        if abs(delta) < 1e-14:
            continue

        # propose move
        cur_h[i] += delta
        cur_h[j] -= delta

        new_c5 = _compute_c5(cur_h)

        # Metropolis acceptance
        accept = False
        if new_c5 < cur_c5:
            accept = True
        else:
            prob = math.exp(-(new_c5 - cur_c5) / max(T, 1e-12))
            if random.random() < prob:
                accept = True

        if accept:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
                steps_without_improve = 0
            else:
                steps_without_improve += 1
        else:
            # revert
            cur_h[i] -= delta
            cur_h[j] += delta
            steps_without_improve += 1

        total_iters += 1

        # ---- deterministic improvement on the worst shift ----
        if total_iters % improve_interval == 0:
            step = step_limit
            improved = _improve_worst_shift(cur_h, step)
            if improved:
                cur_c5 = _compute_c5(cur_h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = cur_h.copy()
                    steps_without_improve = 0

        # ---- restart if stagnated for a while ----
        if steps_without_improve > 50000:
            cur_h = _random_binary_vector(n)
            cur_c5 = _compute_c5(cur_h)
            steps_without_improve = 0

        # early stop if we already beat the target record
        if best_c5 <= 0.3808:
            break

    # ------------------------------------------------------------
    # 5. Final sanity projection and bound evaluation
    # ------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    # enforce exact linear constraint (tiny numerical drift)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        best_h[0] += diff
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = _compute_c5(best_h)

    return best_h.tolist(), final_c5, n
