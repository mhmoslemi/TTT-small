def project_onto_box_simplex(x, target, tol=1e-9, max_iter=100):
    # Called after we have some initial x (real numbers). Return y.
    # Bisection on lambda.
    # Determine lower bound L and upper bound U such that sum_i clip(x_i - L,0,1) >= target and sum_i clip(x_i - U,0,1) <= target.
    # As lambda increases, clip(x_i - lambda) decreases (monotonic). So we can search.
    # For lambda very negative, x_i - lambda large >1, clip=1 for all -> sum=n. So if target <= n, which holds.
    # For lambda very large, x_i - lambda <<0 => clip=0 for all -> sum=0.
    # So we can set lo = -1e6, hi = 1e6 maybe.
    lo = -1e6
    hi = 1e6
    # tighten bounds:
    # Ensure sum at lo >= target
    sum_lo = np.sum(np.clip(x - lo, 0, 1))
    while sum_lo < target:
        lo *= 2  # make more negative
        sum_lo = np.sum(np.clip(x - lo, 0, 1))
    sum_hi = np.sum(np.clip(x - hi, 0, 1))
    while sum_hi > target:
        hi *= 2  # make more positive
        sum_hi = np.sum(np.clip(x - hi, 0, 1))
    # bisection
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        y = np.clip(x - mid, 0, 1)
        s = y.sum()
        if abs(s - target) <= tol:
            return y
        if s > target:
            lo = mid
        else:
            hi = mid
    # after loop, return y

def smooth_max_minus_conv_gradient(h, alpha):
    n = len(h)
    conv_full = np.correlate(h, h, mode='full')
    conv = conv_full[n-1:]  # length n
    # compute weights for -conv
    # exponent = -alpha * conv

    max_arg = np.max(exp_arg)
    exps = np.exp(exp_arg - max_arg)
    sum_exps = np.sum(exps)
    w = exps / sum_exps
    # gradient
    grad = np.zeros_like(h)
    # k=0 contribution: derivative of conv_0 is 2 * h_i
    # Negative sign included: grad0 = - w0 * 2 * h
    w0 = w[0]
    grad -= 2 * w0 * h
    # other ks
    for k in range(1, n):
        wk = w[k]
        if wk == 0:
            continue

        grad[:n - k] -= wk * h[k:]
        # for indices j-k >=0: -wk * h_{j-k}
        grad[k:] -= wk * h[:n - k]
    # smooth value (optional)
    smooth_val = (np.log(sum_exps) + max_arg) / alpha  # approximates max(-conv)
    return smooth_val, grad

def project_to_sum_box(x, target_sum, tol=1e-9):
    # Clip values to [0,1] initially; then adjust using water-filling.
    # If after clipping sum already equals target, return.
    x_clipped = np.clip(x, 0, 1)
    s = x_clipped.sum()
    if abs(s - target_sum) <= tol:
        return x_clipped
    # We'll find lambda such that sum(clip(x - lambda, 0, 1)) = target_sum
    # Use binary search between low and high.
    low = -np.max(x) - 1.0  # ensures all clip to 1
    high = np.max(x) + 1.0  # ensures all clip to 0
    # Ensure correct direction: sum_at_low = n (since all 1), sum_at_high = 0.
    for _ in range(60):
        mid = (low + high) / 2.0
        y = np.clip(x - mid, 0, 1)
        s_mid = y.sum()
        if s_mid > target_sum:
            low = mid
        else:
            high = mid
    y = np.clip(x - (low + high) / 2.0, 0, 1)
    return y

import numpy as np
import time
import random

def _dx(n: int) -> float:
    """grid spacing for n points on [0,2]."""
    return 2.0 / n


def compute_c5(h: np.ndarray) -> float:
    """Exact evaluation used by the harness."""
    n = h.shape[0]
    dx = _dx(n)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))


def _project_to_sum_box(x: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of x onto the set
        { y : 0 <= y_i <= 1, sum(y) = target }.
    Uses a simple bisection on the Lagrange multiplier λ for the
    operation y_i = clip(x_i - λ, 0, 1).
    """
    # Clip first – this gives a feasible point for the box.
    y = np.clip(x, 0.0, 1.0)
    s = y.sum()
    if abs(s - target) <= 1e-12:
        return y

    # bounds for λ: very negative → all ones, very positive → all zeros
    lo = -np.max(x) - 1.0  # ensures y_i = 1 for all i
    hi = np.max(x) + 1.0   # ensures y_i = 0 for all i
    for _ in range(60):
        mid = (lo + hi) * 0.5
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)


def upsample_to_n(old_h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a step function from old_n to new_n points."""
    old_n = old_h.shape[0]
    dx_old = _dx(old_n)
    dx_new = _dx(new_n)
    # cell centres for interpolation
    x_old = np.arange(old_n) * dx_old + dx_old * 0.5
    x_new = np.arange(new_n) * dx_new + dx_new * 0.5
    new_h = np.interp(x_new, x_old, old_h, left=old_h[0], right=old_h[-1])
    return new_h


def _smooth_max_minus_conv_grad(h: np.ndarray, alpha: float):
    """
    Returns (smooth_value, grad) for the smoothed max of -conv,
    where conv(k) = sum_i h_i * h_{i+k}, k≥0.
    grad = ∂/∂h of (1/α)·log Σ_i exp(-α·conv_i).
    """
    n = h.shape[0]
    # full autocorrelation, then keep non‑negative lags (including 0)
    conv_full = np.correlate(h, h, mode="full")
    conv = conv_full[n - 1 :]  # length n, conv[0] = Σ h_i²
    # weights for the smoothed max of -conv
    exp_arg = -alpha * conv
    max_arg = np.max(exp_arg)
    exps = np.exp(exp_arg - max_arg)
    w = exps / exps.sum()  # Σ w = 1

    # gradient of the smooth max
    grad = np.zeros_like(h)
    # k = 0 term: d(conv_0)/dh = 2*h
    grad -= 2.0 * w[0] * h
    # k >= 1
    for k in range(1, n):
        wk = w[k]
        if wk == 0.0:
            continue
        # forward part: contribution of h_{j+k}
        grad[: n - k] -= wk * h[k:]
        # backward part: contribution of h_{j‑k}
        grad[k:] -= wk * h[: n - k]

    # smooth max value (not needed for the optimiser but useful for diagnostics)
    smooth_val = (np.log(exps.sum()) + max_arg) / alpha
    return smooth_val, grad


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimises a step function h on [0,2] for the C_5 constant.
    Returns (h_values, c5_bound, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()

    # ---------- choose resolution ----------
    target_n = 200                     # < 1k, gives decent accuracy
    target_sum = target_n / 2.0        # because integral = 1

    # ---------- initial guess ----------
    try:
        init = initial_h_values        # provided by the harness
    except NameError:
        init = None

    if init is not None:
        # upsample (or downsample) to the target size
        h = upsample_to_n(np.asarray(init, dtype=float), target_n)
    else:
        # random start respecting box, later projected onto the sum‑box
        h = np.random.rand(target_n)

    # ensure feasibility before optimisation
    h = _project_to_sum_box(h, target_sum)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # ---------- hyper‑parameters ----------
    max_iters = 5000
    lr_start = 0.5
    alpha_start = 0.5
    alpha_end = 30.0

    # ---------- main optimisation loop ----------
    for it in range(max_iters):
        # time‑budget check
        if time.time() - start_time > budget_s - 0.5:
            break

        # continuation on the smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)

        # smoothed objective and its gradient
        _, grad = _smooth_max_minus_conv_grad(h, alpha)

        # simple learning‑rate decay
        lr = lr_start / (1.0 + it * 0.005)

        # gradient step
        h = h - lr * grad

        # occasional jitter to escape shallow plateaus
        if it % 500 == 0 and it > 0:
            h += np.random.normal(scale=0.01, size=target_n)

        # projection onto the feasible set
        h = _project_to_sum_box(h, target_sum)

        # track the best so far according to the true C5 objective
        cur_c5 = compute_c5(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()

    # final safety projection & evaluation
    best_h = _project_to_sum_box(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, float(best_c5), target_n
