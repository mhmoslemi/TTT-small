def propose(h):
    i = np.random.randint(n)
    j = np.random.randint(n)
    while j == i:
        j = np.random.randint(n)
    # random direction.
    max_increase_i = 1 - h[i]
    max_decrease_i = h[i]
    max_increase_j = 1 - h[j]
    max_decrease_j = h[j]
    # choose delta bounded by both moves: we will increase i by delta and decrease j by delta such that both stay within [0,1].
    # delta may be positive or negative.
    delta_max = min(max_increase_i, max_decrease_j)
    delta_min = -min(max_decrease_i, max_increase_j)
    # sample delta uniformly in [delta_min, delta_max]
    if delta_max <= delta_min:
        return h  # can't move?
    delta = np.random.uniform(delta_min, delta_max)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

def propose_move(h, rng):
    n = h.shape[0]
    i, j = rng.integers(n, size=2)
    while i == j:
        j = rng.integers(n)
    # random direction
    if rng.random() < 0.5:
        # increase i, decrease j
        delta_max = min(1 - h[i], h[j])
        if delta_max <= 1e-12:
            return None
        delta = rng.random() * delta_max
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta

        # increase j, decrease i

            return None
        delta = rng.random() * delta_max
        h_new = h.copy()
        h_new[j] += delta
        h_new[i] -= delta
    return h_new

def run(seed=42, budget_s=1100, **kwargs):

        # Ensure sum constraint by scaling if needed
        target = n / 2.0

        if abs(current_sum - target) > 1e-9:

            # Use iterative scaling+clipping
            h = h0.copy()
            for _ in range(10):

                h *= scale
                np.clip(h, 0, 1, out=h)
                # If sum still off, break
                if abs(h.sum() - target) < 1e-9:

    except NameError:

        h0 = np.zeros(n)

    # Evaluate initial
    best_h = h0.copy()
    best_val = evaluate_h(best_h, dx)
    cur_h = best_h.copy()
    cur_val = best_val

    start_time = time.time()
    # time budget for optimization
    time_budget = budget_s * 0.95   # leave margin
    # Set temperature parameters
    # We'll linearly decay from init_temp to final_temp
    init_temp = best_val * 0.1 + 0.005  # some small base
    final_temp = 1e-6

    iteration = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= time_budget:
            break
        # linear temperature decay
        t_frac = max(0.0, 1.0 - elapsed / time_budget)
        temp = final_temp + (init_temp - final_temp) * t_frac

        if h_new is None:

        new_val = evaluate_h(h_new, dx)
        delta = new_val - cur_val

            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h_new.copy()

        if iteration % 1000 == 0 and rng.random() < 0.02:
            cur_h = best_h.copy()
            cur_val = best_val

def project_to_sum(h, target):
    # ensure values within bounds
    h = np.clip(h, 0, 1)
    for _ in range(10):
        # compute scaling factor
        total = h.sum()
        if total == 0:
            # can't scale; distribute equally
            h[:] = target / len(h)
            break
        scale = target / total
        h *= scale
        np.clip(h, 0, 1, out=h)
        if abs(h.sum() - target) < 1e-10:
            break
    # final adjustment if needed by distributing diff
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # find indices not at bound
        free_idx = np.where((h > 0) & (h < 1))[0]
        if free_idx.size > 0:
            per = diff / free_idx.size
            # adjust small steps, ensuring not exceed bounds; clamp naive.
            for idx in free_idx:
                new_val = h[idx] + per
                if new_val < 0:
                    new_val = 0.0
                elif new_val > 1:
                    new_val = 1.0
                h[idx] = new_val
        else:
            # fallback: adjust first element
            h[0] = np.clip(h[0] + diff, 0, 1)
    return h

def propose_move(h, rng):
    n = h.shape[0]
    i, j = rng.integers(n, size=2)
    while i == j:
        j = rng.integers(n)
    # choose direction uniformly
    if rng.random() < 0.5:
        delta_max = min(1.0 - h[i], h[j])
        if delta_max <= 0:
            return None
        delta = rng.random() * delta_max
        hnew = h.copy()
        hnew[i] += delta
        hnew[j] -= delta
    else:
        delta_max = min(1.0 - h[j], h[i])
        if delta_max <= 0:
            return None
        delta = rng.random() * delta_max
        hnew = h.copy()
        hnew[i] -= delta
        hnew[j] += delta
    return hnew

    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def project_to_sum(h, target):
    # Clip to bounds
    np.clip(h, 0, 1, out=h)
    # Simple scaling+clipping iteration
    for _ in range(10):
        s = h.sum()
        if s == 0:
            break
        scale = target / s
        h *= scale
        np.clip(h, 0, 1, out=h)
        if abs(h.sum() - target) < 1e-12:
            break
    # Final diff correction
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # distribute diff to non‑boundary entries
        free = np.where((h > 0) & (h < 1))[0]
        if free.size > 0:
            per = diff / free.size
            for idx in free:
                new = h[idx] + per
                h[idx] = min(1.0, max(0.0, new))
        else:
            # fallback: adjust first element
            h[0] = min(1.0, max(0.0, h[0] + diff))
    return h

def propose_move(h, rng):
    n = h.shape[0]
    i, j = rng.integers(n, size=2)
    while i == j:
        j = rng.integers(n)
    if rng.random() < 0.5:
        delta_max = min(1.0 - h[i], h[j])
        if delta_max <= 0:
            return None
        delta = rng.random() * delta_max
        hnew = h.copy()
        hnew[i] += delta
        hnew[j] -= delta
    else:
        delta_max = min(1.0 - h[j], h[i])
        if delta_max <= 0:
            return None
        delta = rng.random() * delta_max
        hnew = h.copy()
        hnew[i] -= delta
        hnew[j] += delta
    return hnew

def binary_hill_climb(n, rng, max_iters, start_h=None):
    if start_h is None:
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, size=n//2, replace=False)
        h[ones] = 1.0
    else:
        # force to binary and correct count
        h = (start_h > 0.5).astype(float)
        cur = int(h.sum())
        target = n // 2
        if cur != target:
            diff = target - cur
            if diff > 0:
                zeros = np.where(h == 0)[0]
                flip = rng.choice(zeros, size=diff, replace=False)
                h[flip] = 1.0
            else:
                ones = np.where(h == 1)[0]
                flip = rng.choice(ones, size=-diff, replace=False)
                h[flip] = 0.0
    dx = 2.0 / n
    cur_val = evaluate_h(h, dx)
    best_h = h.copy()
    best_val = cur_val
    for _ in range(max_iters):
        i = rng.integers(n)
        j = rng.integers(n)
        while i == j or h[i] == h[j]:
            i = rng.integers(n)
            j = rng.integers(n)
        # swap
        h[i], h[j] = h[j], h[i]
        new_val = evaluate_h(h, dx)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = h[j], h[i]
    return best_h, best_val

            best_val = val_bin

    if best_h is not None:

        target = n / 2.0

        best_h = np.array([0.5, 0.5])
        final_val = evaluate_h(best_h, 2.0 / n)
        best_n = n
