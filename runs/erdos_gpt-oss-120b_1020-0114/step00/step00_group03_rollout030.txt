from scipy.optimize import minimize, NonlinearConstraint

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # Determine n based on initial_h_values or default
    try:
        init = np.array(initial_h_values, dtype=float)
        n = init.shape[0]
    except NameError:
        n = 70
        init = np.full(n, 0.5)   # fallback uniform

    # Ensure init meets sum constraint
    target_sum = n/2.0
    # Scale / adjust as needed
    # ... (function adjust)

    best_h = None
    best_val = np.inf
    best_n = n

    # Global variables placeholder
    global _n, _dx
    _n = None
    _dx = None

    def evaluate_max_overlap(h):
        # uses evaluate_erdos_solution (pre-imported)
        return float(evaluate_erdos_solution(h))

    # Optimization per given start vector
    def opt_one(h_start):
        nonlocal best_h, best_val, best_n, _n, _dx
        n_local = len(h_start)
        dx = 2.0 / n_local
        target_sum_local = n_local/2.0

        # Adjust h_start to satisfy sum constraint exactly (via simple scaling and clipping)
        h = np.clip(h_start.copy(), 0.0, 1.0)
        sum_h = h.sum()
        if abs(sum_h - target_sum_local) > 1e-8:
            # Simple linear scaling then clip again
            scaling = target_sum_local / sum_h if sum_h>0 else 0.0
            h *= scaling
            np.clip(h, 0.0, 1.0, out=h)
            # If still off, adjust by adding small amount uniformly
            diff = target_sum_local - h.sum()
            if abs(diff) > 1e-9:
                # distribute diff among unfixed indices
                free = np.where((h > 0) & (h < 1))[0]
                if free.size > 0:
                    inc = diff / free.size
                    h[free] += inc
                    np.clip(h, 0.0, 1.0, out=h)
                else:
                    # fallback binary assignment
                    idx = rng.choice(n_local, int(target_sum_local), replace=False)
                    h[:] = 0.0
                    h[idx] = 1.0

        # compute initial t
        f_vals = np.correlate(h, 1 - h, mode='full') * dx
        t0 = f_vals.max() + 1e-8

        # Prepare global values for constraint funcs
        _n = n_local
        _dx = dx

        # objective
        def obj(x):
            return x[-1]

        # equality constraint
        def eq_fun(x):
            return np.sum(x[:_n]) - target_sum_local

        # inequality vector constraint: t - f_vals >=0
        def ineq_fun(x):
            hvec = x[:_n]
            t = x[-1]
            # correlation vector
            corr = np.correlate(hvec, 1 - hvec, mode='full') * _dx
            return t - corr

        # bounds for h and t
        bounds = [(0.0, 1.0)] * _n + [(0.0, 1.0)]

        eq_con = {'type': 'eq', 'fun': eq_fun}
        ineq_con = NonlinearConstraint(ineq_fun, lb=np.zeros(2*_n - 1), ub=np.full(2*_n - 1, np.inf))

        x0 = np.concatenate([h, [t0]])

        # minimize
        res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                       constraints=[eq_con, ineq_con],
                       options={'maxiter': 500, 'ftol': 1e-9, 'disp': False})
        if not res.success:
            # maybe try with higher maxiter
            pass

        h_opt = np.clip(res.x[:_n], 0.0, 1.0)
        # adjust sum to be exact if needed
        sum_h_opt = h_opt.sum()
        if abs(sum_h_opt - target_sum_local) > 1e-6:
            # project to feasible via simple linear scaling and clip
            scaling = target_sum_local / sum_h_opt if sum_h_opt>0 else 0.0
            h_opt *= scaling
            np.clip(h_opt, 0.0, 1.0, out=h_opt)

        val = evaluate_max_overlap(h_opt)

        if val < best_val - 1e-10:

            best_val = float(val)
            best_n = _n

    # run optimization on initial value

        if abs(diff) > 1e-8:
            free = np.where((h_rand > 0) & (h_rand < 1))[0]
            if free.size > 0:
                inc = diff / free.size
                h_rand[free] += inc
                np.clip(h_rand, 0, 1, out=h_rand)
        opt_one(h_rand)

    # Optionally try a refined grid with more points if time permits
    if time.time() - start < budget_s * 0.7:
        # try n2 = 100
        n2 = 100
        # interpolate best if exists
        if best_h is not None and best_n == n:
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n2, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h, left=0, right=0)

    if best_h is None:

        n = n

        best_val = evaluate_max_overlap(best_h)

import time
import numpy as np
from scipy.optimize import minimize, NonlinearConstraint

# Global placeholders for the current optimisation problem
_opt_n = None          # number of points
_opt_dx = None         # grid spacing 2 / n
_opt_target_sum = None # n / 2

def objective_fun(x):
    """Objective: minimise the maximal overlap value (the last variable)."""
    return x[-1]

def eq_constraint_fun(x):
    """Equality: integral of h must be 1 → sum(h) = n/2."""
    return np.sum(x[:_opt_n]) - _opt_target_sum

def ineq_constraint_fun(x):
    """Inequality: for every shift, t >= overlap_k."""
    h = x[:_opt_n]
    t = x[-1]
    corr = np.correlate(h, 1 - h, mode='full') * _opt_dx
    return t - corr  # should be >= 0 for all lags

def project_to_feasible(h, target):
    """Clip to [0,1] and adjust linearly so that sum(h) = target (approximately)."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == 0:
        # put unit mass on a random subset
        idx = np.random.choice(len(h), int(round(target)), replace=False)
        h[idx] = 1.0
        return h
    scale = target / cur
    h = np.clip(h * scale, 0.0, 1.0)
    # simple iterative correction for remaining error
    for _ in range(10):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            # no free variables – push a random index toward the needed direction
            i = np.random.randint(len(h))
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
            continue
        delta = diff / free.size
        h[free] += delta
        np.clip(h, 0.0, 1.0, out=h)
    return h

def optimise_start(start_h):
    """Run a single SLSQP optimisation starting from start_h."""
    global _opt_n, _opt_dx, _opt_target_sum
    n = int(start_h.shape[0])
    _opt_n = n
    _opt_dx = 2.0 / n
    _opt_target_sum = n / 2.0

    h0 = project_to_feasible(start_h, _opt_target_sum)

    # initial feasible t (slightly above the current max overlap)
    init_corr = np.correlate(h0, 1 - h0, mode='full') * _opt_dx
    t0 = float(np.max(init_corr) + 1e-8)

    # bounds for each variable (h_i in [0,1], t in [0,1])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    # constraints
    eq_cons = {'type': 'eq', 'fun': eq_constraint_fun}
    m = 2 * n - 1
    ineq_cons = NonlinearConstraint(
        ineq_constraint_fun,
        lb=np.zeros(m),
        ub=np.full(m, np.inf),
    )

    x0 = np.concatenate([h0, [t0]])

    res = minimize(
        objective_fun,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=[eq_cons, ineq_cons],
        options={'maxiter': 600, 'ftol': 1e-9, 'disp': False},
    )

    # take the result, enforce bounds and sum again (tiny numerical drift)
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    h_opt = project_to_feasible(h_opt, _opt_target_sum)

    # final value using the provided evaluator (may be slightly different from t)
    try:
        c5_val = float(evaluate_erdos_solution(h_opt))
    except Exception:
        # fallback to the optimizer's t value
        c5_val = float(res.x[-1])

    return h_opt, c5_val
