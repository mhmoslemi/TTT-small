def project_to_feasible(x, target):
    x = np.clip(x, 0.0, 1.0)
    total = x.sum()
    diff = target - total
    eps = 1e-12
    max_iter = 1000
    for _ in range(max_iter):
        if abs(diff) < eps:
            break
        # free indices
        if diff > 0:
            free = np.where(x < 1 - 1e-12)[0]
        else:
            free = np.where(x > 1e-12)[0]
        if len(free) == 0:
            break
        delta = diff / len(free)
        # update and clamp
        for i in free:
            x[i] += delta
        # after addition, clamp again:
        x = np.clip(x, 0.0, 1.0)
        new_total = x.sum()
        diff = target - new_total
    # final clamp
    return x

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    from scipy.optimize import differential_evolution

    # get n_points
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        init_h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 42
        # simple initial like half ones half zeros
        init_h = np.concatenate([np.ones(n_points//2), np.zeros(n_points - n_points//2)])
    target_sum = n_points/2.0
    dx = 2.0/ n_points

    # define projection
    def project_to_feasible(x):
        # returns feasible vector, using iterative waterfilling
        x = np.clip(x, 0.0, 1.0)
        total = x.sum()
        diff = target_sum - total
        eps = 1e-12
        for _ in range(1000):
            if abs(diff) < eps:
                break
            if diff > 0:
                free = np.where(x < 1 - 1e-12)[0]
            else:
                free = np.where(x > 1e-12)[0]
            if len(free) == 0:
                break
            delta = diff / len(free)
            # update free entries
            x[free] += delta
            # clamp again
            x = np.clip(x, 0.0, 1.0)
            total = x.sum()
            diff = target_sum - total
        # final clip
        return x

    # objective function
    def obj(x):
        h = project_to_feasible(x)
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    # DE bounds
    bounds = [(0.0, 1.0)] * n_points

    # initial population includes the initial solution
    rng = np.random.default_rng(seed)
    popsize = 15  # population size factor (popsize * dim individuals)
    init_pop = rng.uniform(0,1,size=(popsize * n_points, n_points))
    # replace first individual with init_h (projected)
    init_pop[0] = init_h.copy()
    # we may also include a flipped version etc.
    # Use DE with init=init_pop
    # time limit monitoring: let max_iter such that evaluations < some number
    start_time = time.time()
    maxiter = 2000  # but we'll break early if time passes

    # DE returns result with x and fun
    result = differential_evolution(obj, bounds, init=init_pop, maxiter=maxiter, seed=seed,
                                    polish=False, updating='deferred', disp=False)

    # best solution
    h_best = project_to_feasible(result.x)
    c5_best = np.max(np.correlate(h_best, 1.0 - h_best, mode='full')) * dx

    # Ensure feasibility
    # maybe do a small local adjustment: use gradient-free local search but we can skip.

    return h_best.tolist(), float(c5_best), n_points

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math, random
    from math import exp

    start = time.time()
    random.seed(seed)
    np.random.seed(seed)

    # define n_points: choose moderate n like 200 (or 300) but ensure feasible within time.
    n_points = 200  # maybe increase if time left.
    dx = 2.0 / n_points
    w = n_points // 2

    # compute C5 bound evaluation
    def c5_of(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def max_mismatch(h):
        return int(np.max(np.correlate(h, 1 - h, mode='full')))

    best_h = None
    best_score = np.inf

    # optionally include initial_h_values scaled to n_points if available
    if 'initial_h_values' in globals():
        ini = np.array(initial_h_values, dtype=float)
        # if length matches n_points, maybe use directly; else ignore
        if len(ini) == n_points:
            best_h = np.clip(ini, 0, 1)
            # adjust to binary? we'll accept fractional but compute.
            best_score = c5_of(best_h)
        else:
            # Could resample to new n_points via interpolation?
            # But skip
            pass

    # main SA loop
    # set parameters
    max_iters = 2000000  # just big; will stop on time
    temperature_start = 5.0
    temperature_end = 0.01
    # schedule using exponential decay: T = temp_start * (temp_end/temp_start)^(iters / max_iters)
    # but we will compute based on elapsed fraction.

    # start with random feasible binary vector
    # we will maintain indices of ones
    def random_binary():
        # choose w distinct indices
        ones = np.random.choice(n_points, w, replace=False)
        h = np.zeros(n_points, dtype=float)
        h[ones] = 1.0
        return h

    h_current = random_binary()
    cur_score = max_mismatch(h_current)

    if cur_score < best_score:
        best_h = h_current.copy()
        best_score = cur_score

    # precompute zero indices list update each iteration; easier: maintain arrays
    # But we can compute indices each time with np.where. That's O(n) fine.

    it = 0
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s * 0.95:
            break
        # fraction of total budget used
        frac = elapsed / (budget_s * 0.95)
        # temperature schedule
        temperature = temperature_start * (temperature_end / temperature_start) ** frac
        # propose swap
        ones_idx = np.where(h_current > 0.5)[0]
        zeros_idx = np.where(h_current < 0.5)[0]
        if len(ones_idx) ==0 or len(zeros_idx)==0:
            break
        i = random.choice(ones_idx)
        j = random.choice(zeros_idx)
        # propose swap
        h_new = h_current.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        new_score = max_mismatch(h_new)
        delta = new_score - cur_score
        if delta <= 0 or random.random() < math.exp(-delta / temperature):
            # accept
            h_current = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        it += 1
        # optional break early if best_score already <= target mismatches
        # target mismatches for C5=0.3808 => max_mis <= target / dx
        target_c5 = 0.3808
        target_mis = int(math.ceil(target_c5 / dx))
        if best_score <= target_mis:
            break

    # compute final c5 bound
    if best_h is None:
        # fallback to random binary at least
        best_h = random_binary()
        best_score = max_mismatch(best_h)

    c5_final = best_score * dx

    # Convert best_h to list
    h_vals = best_h.tolist()
    # ensure sum approx w (should be exactly)
    # return
    return h_vals, float(c5_final), n_points

import time

import numpy as np

# ---------- helper utilities -------------------------------------------------


def project_to_feasible(x: np.ndarray, target_sum: float) -> np.ndarray:
    """Project vector x onto the feasible set:
       0 <= x_i <= 1  and  sum(x) == target_sum  (approximately).
    """
    x = np.clip(x, 0.0, 1.0)
    total = x.sum()
    diff = target_sum - total
    eps = 1e-12
    for _ in range(1000):
        if abs(diff) < eps:
            break
        if diff > 0:
            free = np.where(x < 1.0 - eps)[0]
        else:
            free = np.where(x > eps)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
        total = x.sum()
        diff = target_sum - total
    return x


def c5_value(h: np.ndarray, dx: float) -> float:
    """Compute the Erdős C5 value for a given step function `h`."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def gradient_step(h: np.ndarray, dx: float, target_sum: float, alpha: float):
    """Perform one projected sub‑gradient step for the max‑overlap objective."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_idx = int(np.argmax(corr))
    shift = k_idx - (n - 1)          # signed integer shift producing the max

    # term1 = 1 - h[i+shift]   (when inside the array, else 1)
    idx_plus = np.arange(n) + shift
    term1 = np.ones_like(h)
    mask_plus = (idx_plus >= 0) & (idx_plus < n)
    term1[mask_plus] = 1.0 - h[idx_plus[mask_plus]]

    # term2 = h[i-shift]       (when inside the array, else 0)
    idx_minus = np.arange(n) - shift
    term2 = np.zeros_like(h)
    mask_minus = (idx_minus >= 0) & (idx_minus < n)
    term2[mask_minus] = h[idx_minus[mask_minus]]

    grad = term1 - term2                     # sub‑gradient of the active shift
    h_new = h - alpha * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_to_feasible(h_new, target_sum)
    return h_new


def refine_h(initial_h: np.ndarray, dx: float, target_sum: float,
             max_iter: int = 5000, alpha_start: float = 0.2,
             time_limit: float = None, start_time: float = None):
    """Iteratively improve `initial_h` using projected sub‑gradient descent."""
    h = initial_h.copy()
    best_h = h.copy()
    best_score = c5_value(h, dx)
    alpha = alpha_start

    for it in range(max_iter):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break

        h_new = gradient_step(h, dx, target_sum, alpha)
        new_score = c5_value(h_new, dx)

        if new_score < best_score - 1e-12:
            best_score = new_score
            best_h = h_new.copy()
            alpha = min(alpha_start, alpha * 1.05)   # slight acceleration
        else:
            alpha *= 0.7                               # back‑off
            if alpha < 1e-6:
                break
        h = h_new
    return best_h, best_score


def make_triangular(n: int) -> np.ndarray:
    """Create a symmetric triangular step function (height 1 at the centre)."""
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    h = 1.0 - np.abs(x - 1.0)
    return np.clip(h, 0.0, 1.0)


def make_constant(n: int, value: float = 0.5) -> np.ndarray:
    """Create a constant function of given value."""
    return np.full(n, value, dtype=float)


def make_random_feasible(n: int, target_sum: float) -> np.ndarray:
    """Generate a random feasible vector (values in [0,1] with prescribed sum)."""
    h = np.random.rand(n)
    h = project_to_feasible(h, target_sum)
    return h


# ---------- main entry point -------------------------------------------------


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Find a step function h on [0,2] with ∫h = 1 that minimizes
    C5 = max_k ∫ h(x)(1‑h(x+k)) dx.
    Returns (h_values, c5_bound, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)

    start_all = time.time()
    # number of discretisation points – a balance between speed and quality
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # construct a small pool of starting vectors
    candidates = [
        project_to_feasible(make_triangular(n_points), target_sum),
        project_to_feasible(make_constant(n_points, 0.5), target_sum),
    ]
    for _ in range(4):
        candidates.append(make_random_feasible(n_points, target_sum))

    best_h = None
    best_score = float("inf")

    # allocate roughly equal time to each refinement run
    per_candidate_time = budget_s * 0.9 / max(1, len(candidates))

    for h0 in candidates:
        elapsed = time.time() - start_all
        remaining = budget_s - elapsed
        if remaining <= 0:
            break
        timelimit = min(per_candidate_time, remaining * 0.95)

        refined_h, refined_score = refine_h(
            h0,
            dx,
            target_sum,
            max_iter=8000,
            alpha_start=0.2,
            time_limit=timelimit,
            start_time=time.time(),
        )
        if refined_score < best_score:
            best_score = refined_score
            best_h = refined_h.copy()

        if time.time() - start_all > budget_s * 0.95:
            break

    # final safety projection and score
    if best_h is None:
        best_h = make_constant(n_points, 0.5)
    best_h = project_to_feasible(best_h, target_sum)
    final_score = c5_value(best_h, dx)

    return best_h.tolist(), float(final_score), n_points
