import time
import numpy as np

def _project_capped_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project vector v onto the set {w | 0 <= w_i <= 1, sum w_i = target_sum}.
    This solves w_i = min(max(v_i - lam, 0), 1) for a suitable lambda.
    """
    # binary search for lambda
    lo = np.min(v) - 1.0      # makes all entries 1
    hi = np.max(v)            # makes all entries 0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) / 2.0
    w = np.clip(v - lam, 0.0, 1.0)

    # tiny correction for rounding errors
    diff = target_sum - w.sum()
    if abs(diff) > 1e-12:
        # distribute the remaining mass among non‑boundary entries
        mask = (w > 0.0) & (w < 1.0)
        if mask.any():
            w[mask] += diff / mask.sum()
            w = np.clip(w, 0.0, 1.0)
        else:
            # virtually impossible, but fallback: add to the smallest entry
            idx = np.argmin(w)
            w[idx] = np.clip(w[idx] + diff, 0.0, 1.0)
    return w


def _c5_value(h: np.ndarray) -> float:
    """Return the C5 value for a given step function h."""
    n = h.size
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv.max() * dx


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a step function h on [0,2] with integral 1 that minimizes C5.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values : np.ndarray of shape (n_points,)  – the best found function
        c5_bound : float                         – its C5 value
        n_points : int                           – number of discretisation points
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine the problem size and an initial feasible vector.
    # --------------------------------------------------------------
    try:
        # initial_h_values is injected by the harness when available.
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
        target_sum = n / 2.0
        # Ensure feasibility (sum and bounds)
        if not np.isclose(h0.sum(), target_sum, atol=1e-9) or np.any((h0 < 0) | (h0 > 1)):
            h0 = _project_capped_simplex(h0, target_sum)
        curr_h = h0.copy()
    except NameError:
        # No provided construction – fall back to a random feasible point.
        n = 42
        target_sum = n / 2.0
        # Dirichlet‑like initialization then projection.
        y = rng.exponential(1.0, size=n)
        h0 = y / y.sum() * target_sum
        curr_h = _project_capped_simplex(h0, target_sum)

    # --------------------------------------------------------------
    # Initialise best solution tracker.
    # --------------------------------------------------------------
    best_h = curr_h.copy()
    best_c5 = _c5_value(best_h)

    # --------------------------------------------------------------
    # Simulated annealing parameters.
    # --------------------------------------------------------------
    start_time = time.time()
    # initial temperature – chosen after a quick probing of objective scale
    T0 = 0.12
    T = T0
    cooling = 0.9994          # multiplicative cooling per iteration
    max_delta = 0.6           # maximum raw change before projection
    stagnation_limit = 8000   # re‑initialise after this many non‑improving steps
    steps_without_improve = 0

    while True:
        # time check
        if time.time() - start_time > budget_s * 0.94:
            break

        # ----------------------------------------------------------
        # Propose a move: transfer a small amount of mass between two indices.
        # ----------------------------------------------------------
        i, j = rng.choice(n, size=2, replace=False)
        # scale step size with temperature
        delta = rng.uniform(-max_delta, max_delta) * (T / T0)
        trial = curr_h.copy()
        trial[i] += delta
        trial[j] -= delta
        # Project back onto the feasible capped simplex.
        trial = _project_capped_simplex(trial, target_sum)

        # ----------------------------------------------------------
        # Evaluate.
        # ----------------------------------------------------------
        trial_c5 = _c5_value(trial)
        delta_c5 = trial_c5 - _c5_value(curr_h)

        # ----------------------------------------------------------
        # Metropolis acceptance.
        # ----------------------------------------------------------
        accept = False
        if delta_c5 <= 0:
            accept = True
        else:
            prob = np.exp(-delta_c5 / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:

            # If this trial beats the global best, remember it.
            if trial_c5 < best_c5:
                best_c5 = trial_c5
                best_h = trial.copy()
                steps_without_improve = 0
            else:

        else:
            steps_without_improve += 1

            steps_without_improve = 0
            T = T0  # re‑heat a little

    # --------------------------------------------------------------
