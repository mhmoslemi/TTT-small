    rng = np.random.default_rng(seed)
    start = time.time()

        init = np.array(initial_h_values, dtype=float)

        h[:n//2] = 1.0
        rng.shuffle(h)
    # Ensure sum exactly n/2 via target
    target_sum = n/2
    # adjust sum if needed using simple scaling + clipping
    # We'll use an iterative projection onto bounded simplex (capped simplex).
    # Use cvxpy? maybe easier: we can use algorithm from "Projection onto the capped simplex"
    # Implement simple iterative water-filling:
    def project_to_bounds_and_sum(x, target):
        # Clip between 0 and 1
        x = np.clip(x, 0, 1)
        # Compute excess/deficit
        cur = x.sum()
        if abs(cur - target) < 1e-12:
            return x
        # If need to add mass
        if cur < target:
            # add mass to elements not at 1 proportionally
            while cur < target - 1e-12:
                free = np.where(x < 1 - 1e-12)[0]
                if free.size == 0:
                    break
                # distribute equally
                add = (target - cur) / free.size
                add = np.minimum(add, 1 - x[free])
                x[free] += add
                cur = x.sum()
        else:
            # cur > target, subtract from elements >0
            while cur > target + 1e-12:
                free = np.where(x > 1e-12)[0]
                if free.size == 0:
                    break
                sub = (cur - target) / free.size
                sub = np.minimum(sub, x[free])
                x[free] -= sub
                cur = x.sum()
        # final clip
        return np.clip(x, 0, 1)
    h = project_to_bounds_and_sum(h, target_sum)

    # Evaluation function
    def compute_c5(vec):
        corr = np.correlate(vec, 1 - vec, mode='full')
        return corr.max() * (2.0 / n)

    cur_h = h.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val

    total_time = budget_s * 0.95
    initial_temp = 0.02
    final_temp = 1e-5

        if elapsed >= total_time:

import time
import numpy as np

def _project_to_capped_simplex(x, target):
    """Clip to [0,1] and adjust to meet the exact sum = target."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    if abs(cur - target) < 1e-12:
        return x
    # simple water‑filling loop; converges quickly because N is modest
    while True:
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        if diff > 0:               # need to add mass
            free = np.where(x < 1.0 - 1e-12)[0]
            if free.size == 0:
                break
            inc = diff / free.size
            inc = np.minimum(inc, 1.0 - x[free])
            x[free] += inc
        else:                      # need to remove mass
            free = np.where(x > 1e-12)[0]
            if free.size == 0:
                break
            dec = (-diff) / free.size
            dec = np.minimum(dec, x[free])
            x[free] -= dec
        cur = x.sum()
    return np.clip(x, 0.0, 1.0)

    start = time.time()

    target_sum = n / 2.0
    h = _project_to_capped_simplex(h, target_sum)

    dx = 2.0 / n

    def _c5(val):
        """Compute the discrete C5 value for a candidate vector."""
        return np.correlate(val, 1.0 - val, mode="full").max() * dx

    cur_h = h.copy()
    cur_val = _c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val

            if max_delta <= 0:

                diff += sub
            if abs(diff) < 1e-12:

    best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = _c5(best_h)
    return best_h, float(final_c5), n
