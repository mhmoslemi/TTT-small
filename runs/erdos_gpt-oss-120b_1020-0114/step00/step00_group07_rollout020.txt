import numpy as np, time

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # np.correlate returns array of length 2*n-1 for 'full'
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # margin

    # Determine dimensionality and initial vector
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=np.float64)
        n = init.shape[0]
    else:
        n = 80
        init = np.full(n, 0.5, dtype=np.float64)

    target_sum = n / 2.0

    # Normalize to exact sum if necessary
    cur_sum = init.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        init = init * (target_sum / cur_sum)
        init = np.clip(init, 0.0, 1.0)
        diff = target_sum - init.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if 0.0 <= init[i] + diff <= 1.0:
                    init[i] += diff
                    break

    best_h = init.copy()
    best_val = compute_c5(best_h)

    # Add a couple of simple deterministic starting points
    # Binary random
    def random_binary():
        idx = rng.choice(n, size=n // 2, replace=False)
        arr = np.zeros(n, dtype=np.float64)
        arr[idx] = 1.0
        return arr
    # Two blocks at ends
    def two_blocks():
        arr = np.zeros(n, dtype=np.float64)
        block = n // 4
        arr[:block] = 1.0
        arr[-block:] = 1.0
        return arr

    for cand in (random_binary(), two_blocks()):
        val = compute_c5(cand)
        if val < best_val:
            best_h = cand.copy()
            best_val = val

    cur_h = best_h.copy()
    cur_val = best_val

    # Simple hill‑climbing with mass swaps
    while time.time() < deadline:
        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:

def compute_c5(h: np.ndarray) -> float:
    """
    Compute C₅ = max_k ∫ h(x)(1−h(x+k))dx
    Discrete version: max of full correlation multiplied by dx.
    """
    n = h.shape[0]
    dx = 2.0 / n
    # np.correlate with mode='full' gives exactly the desired sums.
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)
