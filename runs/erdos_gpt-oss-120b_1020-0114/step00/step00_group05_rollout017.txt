    # define functions
    def sigmoid(z):
        return 1 / (1 + np.exp(-z))

    def softmax(v):
        e = np.exp(v - np.max(v))
        return e / np.sum(e)

    def build_h_from_params(params):
        # params length = 2*m -1
        w_raw = params[:m]
        h_raw = params[m:]
        # widths
        w = softmax(w_raw) * 2.0   # sum to 2
        # heights for first m-1 intervals
        h_vals = sigmoid(h_raw)    # in (0,1)
        # compute last height to satisfy integral constraint
        # h_m = (1 - sum_{i<m} w_i * h_i) / w_m
        integral_needed = 1.0
        sum_prev = np.dot(w[:m-1], h_vals)
        h_last = (integral_needed - sum_prev) / w[m-1]
        # enforce bounds
        if h_last < 0 or h_last > 1:

            return None

                j = m-1
            h_arr[i] = heights[j]

    e = np.exp(v - np.max(v))
    return e / np.sum(e)

    # returns array or None
    w_raw = params[:m]
    h_raw = params[m:]
    w = softmax_raw(w_raw) * 2.0
    if np.min(w) < 1e-12:
        return None
    h_vals = sigmoid(h_raw)
    # compute last height
    sum_prev = np.dot(w[:m-1], h_vals)
    h_last = (1.0 - sum_prev) / w[m-1]
    if h_last < 0.0 or h_last > 1.0:

    heights = np.concatenate([h_vals, [h_last]])
    # assign to discrete points
    dx = 2.0 / n_points
    x = (np.arange(n_points) + 0.5) * dx
    cum_w = np.cumsum(np.concatenate([[0.0], w]))
    idx = np.searchsorted(cum_w, x, side='right') - 1
    idx = np.clip(idx, 0, m-1)
    h_arr = heights[idx]
    return h_arr

    return C5_val + pen

def _sigmoid(z: np.ndarray) -> np.ndarray:
    """Logistic sigmoid mapping R -> (0,1)."""
    return 1.0 / (1.0 + np.exp(-z))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Softmax that returns a probability vector summing to 1."""
    e = np.exp(v - np.max(v))
    return e / np.sum(e)


def build_h_from_params(params: np.ndarray, m: int, n_points: int):
    """
    Construct the discretised step function h from a parameter vector.

    Parameters
    ----------
    params : np.ndarray
        Length = 2*m-1. First m entries are raw widths,
        remaining m-1 entries are raw heights.
    m : int
        Number of constant intervals.
    n_points : int
        Number of samples on [0,2].

    Returns
    -------
    h_arr : np.ndarray or None
        Length n_points array with values in [0,1] or None if infeasible.
    """
    # split parameters
    w_raw = params[:m]
    h_raw = params[m:]

    # widths (positive, sum to 2)
    widths = _softmax(w_raw) * 2.0
    if np.min(widths) < 1e-12:
        return None

    # heights for first m-1 intervals (in (0,1))
    heights_part = _sigmoid(h_raw)

    # determine height of the final interval from the integral constraint
    integral_needed = 1.0
    sum_prev = np.dot(widths[: m - 1], heights_part)
    last_width = widths[m - 1]
    last_height = (integral_needed - sum_prev) / last_width
    if not (0.0 <= last_height <= 1.0):
        return None

    heights = np.concatenate([heights_part, [last_height]])

    # map to the discrete grid (mid‑point rule)
    dx = 2.0 / n_points
    x = (np.arange(n_points) + 0.5) * dx          # mid‑points
    cum_w = np.cumsum(np.concatenate([[0.0], widths]))
    idx = np.searchsorted(cum_w, x, side='right') - 1
    idx = np.clip(idx, 0, m - 1)
    h_arr = heights[idx]
    return h_arr


def objective_function(params: np.ndarray, m: int, n_points: int) -> float:
    """
    Penalised objective: max overlap + large penalty for violating constraints.
    """
    h = build_h_from_params(params, m, n_points)
    if h is None:
        return 1e3  # infeasible → huge value

    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode="full") * dx
    C5 = float(np.max(corr))

    # penalty for sum constraint (must be n_points/2)
    target_sum = n_points / 2.0
    sum_h = float(np.sum(h))
    penalty = 100.0 * abs(sum_h - target_sum)

    # safety penalty for any slight out‑of‑bounds values
    if np.any(h < -1e-9) or np.any(h > 1.0 + 1e-9):
        penalty += 100.0

    return C5 + penalty


def adjust_sum_and_clip(h: np.ndarray, n_points: int) -> np.ndarray:
    """
    Enforce the exact integral constraint on the discrete vector.
    """
    target = n_points / 2.0
    cur = float(np.sum(h))
    diff = target - cur

    if abs(diff) <= 1e-12:
        return h

    # try to fix by adjusting a single entry
    for i in range(len(h)):
        new_val = h[i] + diff
        if 0.0 <= new_val <= 1.0:
            h[i] = new_val
            return h

    # fallback: uniform scaling then tiny redistribution
    scale = target / cur
    h = np.clip(h * scale, 0.0, 1.0)
    cur = float(np.sum(h))
    diff = target - cur
    if abs(diff) > 1e-12:
        for i in range(len(h)):
            if diff == 0.0:
                break
            step = np.sign(diff) * min(abs(diff),
                                       (1.0 - h[i]) if diff > 0 else h[i])
            h[i] += step
            diff -= step
    return h


def compute_c5(h: np.ndarray) -> float:
    """Direct computation of C5 for a given discrete h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return float(np.max(corr))


# ----- main entry point -----
def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function on [0,2] to obtain a low upper bound for C5.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start = time.time()

    n_points = 70                         # fixed small grid size
    target_c5 = 0.3808                    # we aim to beat this

    best_h = None
    best_c5 = float('inf')

    # explore a few interval counts (m = 4 … 8)
    for m in (4, 5, 6, 7, 8):
        if time.time() - start > budget_s * 0.85:
            break

        dim = 2 * m - 1
        bounds = [(-5.0, 5.0)] * dim       # generous box for raw parameters

        # differential evolution – global search
        de_res = differential_evolution(
            objective_function,
            bounds,
            args=(m, n_points),
            maxiter=200,
            popsize=15,
            polish=False,
            seed=seed,
            disp=False,
        )
        params = de_res.x
        h_vec = build_h_from_params(params, m, n_points)
        if h_vec is None:
            continue

        h_vec = adjust_sum_and_clip(h_vec, n_points)

        # evaluate with the provided helper if available
        try:
            c5_val = evaluate_erdos_solution(h_vec)
        except Exception:
            c5_val = compute_c5(h_vec)

        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_vec.copy()

        if best_c5 < target_c5:
            break

        # local refinement (Nelder–Mead) starting from DE result
        try:
            loc_res = minimize(
                objective_function,
                params,
                args=(m, n_points),
                method='Nelder-Mead',
                options={'maxfev': 3000, 'fatol': 1e-12, 'xatol': 1e-12, 'disp': False},
            )
            if loc_res.success:
                h_loc = build_h_from_params(loc_res.x, m, n_points)
                if h_loc is not None:
                    h_loc = adjust_sum_and_clip(h_loc, n_points)
                    try:
                        c5_loc = evaluate_erdos_solution(h_loc)
                    except Exception:
                        c5_loc = compute_c5(h_loc)
                    if c5_loc < best_c5:
                        best_c5 = c5_loc
                        best_h = h_loc.copy()
        except Exception:
            pass

    # fallback to the supplied parent construction if optimisation failed
    if best_h is None:
        try:
            h0 = np.array(initial_h_values, dtype=float)
            h0 = adjust_sum_and_clip(h0, len(h0))
            best_h = h0
            best_c5 = compute_c5(best_h)
        except Exception:
            best_h = np.full(n_points, 0.5)
            best_c5 = compute_c5(best_h)

    # final safety check – enforce exact sum one more time
    best_h = adjust_sum_and_clip(best_h, n_points)

    return best_h, float(best_c5), n_points
