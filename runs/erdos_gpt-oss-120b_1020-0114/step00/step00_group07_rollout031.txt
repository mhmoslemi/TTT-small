def compute_c5(h):

    n = h.shape[0]

    g = 1 - h

    L = 1 << ((2*n - 1).bit_length())   # next power of two

    H = np.fft.rfft(h, n=L)

    G = np.fft.rfft(g, n=L)

    corr = np.fft.irfft(H * np.conj(G), n=L)[:2*n-1]

    c5 = corr.max() * (2.0 / n)

    return c5

Note: np.rfft returns real FFT half-length; taking conjugate works.

def compute_gradient_for_lag(h, lag):
    n = len(h)
    g = np.zeros_like(h)
    if lag >= 0:
        # first contribution: (1 - h[i+lag]) for i in [0, n-lag-1]
        g[:n - lag] += 1 - h[lag:]
        # second contribution: -h[i-lag] for i in [lag, n-1]
        g[lag:] -= h[:n - lag]
    else:

        g[k:] += 1 - h[:n - k]
        # second: -h[i + k] for i in [0, n - k -1]
        g[:n - k] -= h[k:]
    return g

import numpy as np, time, math, itertools

def _project_box_sum(x, target):
    """project x onto [0,1]^n with sum == target by bisection on lambda."""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)


def _corr_full(h, g):
    """linear correlation h * g (full mode) via FFT."""
    n = h.size
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(g, n=L)
    c = np.fft.irfft(H * np.conj(G), n=L)[: 2 * n - 1]
    return c


def _max_corr_and_lag(h):
    """return max correlation value and signed lag for h vs (1‑h)."""
    g = 1.0 - h
    c = _corr_full(h, g)
    idx = int(np.argmax(c))
    n = h.size
    lag = idx - (n - 1)          # signed lag
    return c[idx], lag


def _grad_for_lag(h, lag):
    """gradient of the correlation term at given lag (to be *subtracted*)."""
    n = h.size
    g = np.zeros_like(h)
    if lag >= 0:
        L = n - lag
        if L > 0:
            g[:L] += 1.0 - h[lag:]          # ∂/∂h_i  of h_i*(1‑h_{i+lag})
            g[lag:] -= h[:L]                # ∂/∂h_{i+lag}
    else:
        k = -lag
        L = n - k
        if L > 0:
            g[k:] += 1.0 - h[:L]            # (1‑h_{i‑k})
            g[:L] -= h[k:]                  # -h_{i+k}
    return g


def _c5_from_h(h):
    """exact C5 using the problem's definition."""
    n = h.size
    dx = 2.0 / n
    maxc, _ = _max_corr_and_lag(h)
    return maxc * dx


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # choose resolution
    if 'initial_h_values' in globals():
        init = np.asarray(globals()['initial_h_values'], dtype=float)
        n = len(init)
        h0 = _project_box_sum(init, n / 2.0)
    else:
        n = 200                     # even, modest size
        h0 = rng.random(n)
        h0 = _project_box_sum(h0, n / 2.0)

    best_h = h0.copy()
    best_c5 = _c5_from_h(best_h)
    target = 0.3808

    # optimisation parameters
    max_iters = 6000
    step_start = 0.2

    while time.time() - start < budget_s * 0.95:
        # new start (occasionally reuse best)
        h = rng.random(n) if rng.random() < 0.7 else best_h.copy()
        h = _project_box_sum(h, n / 2.0)

        for it in range(max_iters):
            # compute current worst lag
            maxc, lag = _max_corr_and_lag(h)
            cur_c5 = maxc * (2.0 / n)

            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, h.copy()
                if best_c5 <= target:
                    return (best_h.tolist(), best_c5, n)

            # gradient for the worst term (to *decrease* the value)
            g = _grad_for_lag(h, lag)
            norm = np.linalg.norm(g)
            if norm == 0:
                break
            # step size annealed
            step = step_start * (1.0 - it / max_iters)
            h_new = h - step * g / norm
            # stay feasible
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = _project_box_sum(h_new, n / 2.0)

            # accept if improvement or with small probability
            maxc_new, _ = _max_corr_and_lag(h_new)
            new_c5 = maxc_new * (2.0 / n)
            if new_c5 <= cur_c5 or rng.random() < 0.001:
                h = h_new
            # early termination if time is low
            if time.time() - start > budget_s * 0.95:
                break
        # after a run, maybe try another random start
    return (best_h.tolist(), best_c5, n)
