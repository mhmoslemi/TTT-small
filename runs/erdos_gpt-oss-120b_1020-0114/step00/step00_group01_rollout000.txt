def hill_climb(h0, rng, time_limit):
    h = h0.copy()
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_score = evaluate(best_h)
    current_score = best_score
    start = time.time()
    iter = 0
    # parameters
    max_iter = 10**7
    step_max = 0.2  # maximum absolute epsilon change
    # temperature schedule: start maybe 0.01, decays
    while time.time() - start < time_limit:
        # adaptive step size: gradually reduce
        # maybe step_size = step_max * (1 - (time elapsed / time_limit))
        elapsed = time.time() - start
        step = step_max * (1 - elapsed / time_limit) + 0.001  # ensure minimal step
        i, j = rng.choice(n, size=2, replace=False)
        # bound for eps
        lower = max(-h[i], h[j] - 1)
        upper = min(1 - h[i], h[j])
        # restrict step magnitude
        bound = min(step, upper - lower)
        if bound <= 0:
            continue
        eps = rng.uniform(-bound, bound)
        # optionally shrink near 0
        if abs(eps) < 1e-10:
            continue
        h_new = h.copy()
        h_new[i] += eps
        h_new[j] -= eps
        # evaluate
        score = evaluate(h_new)
        if score < best_score - 1e-10:
            best_score = score
            best_h = h_new.copy()
            h = h_new
            current_score = score
        else:
            # accept with probability
            # temperature schedule: maybe linear decreasing
            temperature = max(1e-4, (time_limit - elapsed) / time_limit) * 0.01
            if score < current_score: # improvement but not best
                # accept as current to continue exploring (like hill climbing).
                h = h_new
                current_score = score
            else:
                # possibly accept worse
                delta = score - current_score
                prob = np.exp(-delta / temperature) if temperature > 0 else 0
                if rng.random() < prob:
                    h = h_new
                    current_score = score
        iter += 1
    return best_h, best_score

def project_to_simplex_box(h, target_sum):
    # h within [0,1] and sum target_sum
    # Use algorithm: clamp to [0,1], then adjust to meet sum by water-filling
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    if abs(total - target_sum) < 1e-12:
        return h
    # Determine if need to increase or decrease
    if total < target_sum:
        # need to increase some values up to 1
        deficit = target_sum - total
        # Identify not at upper bound
        free = (h < 1.0)
        while deficit > 1e-12 and np.any(free):
            free_indices = np.where(free)[0]
            inc = deficit / free_indices.size
            # limit inc to not exceed 1-h
            max_inc = np.min(1.0 - h[free_indices])
            if inc > max_inc:
                inc = max_inc
            h[free_indices] += inc
            deficit = target_sum - h.sum()
            free = (h < 1.0 - 1e-12)  # recompute
    else:
        # need to decrease some values down to 0
        surplus = total - target_sum
        free = (h > 0.0)
        while surplus > 1e-12 and np.any(free):
            free_indices = np.where(free)[0]
            dec = surplus / free_indices.size
            max_dec = np.min(h[free_indices])  # amount we can subtract without negative
            if dec > max_dec:
                dec = max_dec
            h[free_indices] -= dec
            surplus = h.sum() - target_sum
            free = (h > 1e-12)
    return h

import time
import numpy as np

def _evaluate(h):
    """Return max_k ∫ h(x)(1‑h(x+k))dx approximated by discrete correlation."""
    n = h.size
    dx = 2.0 / n
    # np.correlate does ∑ h[i] * (1‑h[i+lag]) for each lag
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())

def _project_to_box_simplex(v, target):
    """Project v onto {0≤x≤1, sum x = target}."""
    x = np.clip(v, 0.0, 1.0)
    diff = target - x.sum()
    # Simple water‑filling loop
    max_iter = 2 * len(v)
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            free = (x < 1.0 - 1e-12)
        else:
            free = (x > 1e-12)
        if not np.any(free):
            break
        cnt = free.sum()
        delta = diff / cnt
        # limit delta so we stay inside the box
        if diff > 0.0:
            inc = np.minimum(delta, 1.0 - x[free])
            x[free] += inc
        else:
            dec = np.maximum(delta, -x[free])
            x[free] += dec
        diff = target - x.sum()
    # Final tiny correction on a non‑border element if needed
    if abs(diff) > 1e-12:
        idx = np.argmin(np.abs(x - 0.5))
        x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
    return x

def _random_feasible(n, rng):
    """Generate a random vector in [0,1] with sum = n/2."""
    v = rng.random(n)
    target = n * 0.5
    return _project_to_box_simplex(v, target)

def _source_dest_move(h, rng):
    """Pick a source i (decrease) and destination j (increase) respecting bounds."""
    n = h.size
    for _ in range(10):
        i, j = rng.choice(n, size=2, replace=False)
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps > 1e-12:
            eps = rng.uniform(0.0, max_eps)
            return i, eps, j
    return None  # no feasible pair found

def _shift_move(h, shift, rng):
    """Move mass from i to i+shift (i is source) for the dominant shift."""
    n = h.size
    if shift >= 0:
        idx = np.arange(0, n - shift)
    else:
        idx = np.arange(-shift, n)
    if idx.size == 0:
        return None
    # contribution to the offending correlation
    contrib = h[idx] * (1.0 - h[idx + shift])
    i = idx[np.argmax(contrib)]
    j = i + shift
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps < 1e-12:
        return None
    eps = rng.uniform(0.0, max_eps)
    return i, eps, j

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # -------------------------------------------------------------
    # initialise
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n = h0.size
    else:
        n = 70
        h0 = _random_feasible(n, rng)

    target_sum = n * 0.5
    h0 = _project_to_box_simplex(h0, target_sum)

    best_h = h0.copy()
    best_score = _evaluate(best_h)

    current_h = best_h.copy()
    current_score = best_score

    # parameters
    max_no_improve = max(2000, n * 50)
    no_improve = 0
    # leave a little margin for the return
    time_limit = budget_s * 0.96

    while time.time() - start < time_limit:
        # decide move type
        if rng.random() < 0.7:
            # shift‑focused move
            corr = np.correlate(current_h, 1.0 - current_h, mode="full")
            s_idx = int(np.argmax(corr))
            shift = s_idx - (n - 1)
            mv = _shift_move(current_h, shift, rng)
            if mv is None:
                mv = _source_dest_move(current_h, rng)
        else:
            mv = _source_dest_move(current_h, rng)

        if mv is None:
            continue  # unable to find a feasible small move

        i, eps, j = mv
        new_h = current_h.copy()
        new_h[i] -= eps
        new_h[j] += eps

        # tiny numerical drift correction
        diff = new_h.sum() - target_sum
        if abs(diff) > 1e-9:
            new_h -= diff / n
            new_h = _project_to_box_simplex(new_h, target_sum)

        new_score = _evaluate(new_h)

        if new_score + 1e-12 < current_score:
            current_h, current_score = new_h, new_score
            no_improve = 0
            if new_score + 1e-12 < best_score:
                best_h, best_score = new_h.copy(), new_score
                # early exit if already under target bound
                if best_score < 0.3808:
                    break
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            # restart from a fresh random point
            current_h = _random_feasible(n, rng)
            current_score = _evaluate(current_h)
            no_improve = 0

    # final clean‑up
    best_h = _project_to_box_simplex(best_h, target_sum)
    final_bound = _evaluate(best_h)

    return best_h.tolist(), final_bound, n
