import time, math, random
import numpy as np

def _project_box_simplex(v, target):
    """Project v onto {x | 0<=x_i<=1, sum x_i = target}."""
    # binary search for lambda such that sum_i clip(v_i - lam,0,1) = target
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _correlation_max(h):
    """Return (raw_max, shift_index, full_corr) for array h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    raw_max = corr.max()
    idx = corr.argmax()
    return raw_max, idx, corr


def _gradient_for_shifts(h, shifts):
    """Gradient of the raw correlation for each shift in ``shifts`` (list of ints)."""
    n = h.size
    g = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            g[: n - k] += (1.0 - h[k:])
            g[k:]       -= h[: n - k]
        else:
            kk = -k
            g[kk:]      += (1.0 - h[: n - kk])
            g[: n - kk] -= h[kk:]
    return g


def _upscale(old_h, n_new):
    """Linearly interpolate old_h (defined on [0,2]) to length n_new."""
    n_old = old_h.size
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.arange(n_new) * (2.0 / n_new)
    # linear interpolation; outside range repeats the endpoint values
    new_h = np.interp(x_new, x_old, old_h, left=old_h[0], right=old_h[-1])
    return new_h
