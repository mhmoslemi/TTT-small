def propose(h):
    h_new = h.copy()
    # gather indices for possible donor and receiver
    donors = np.where(h_new > 0)[0]
    receivers = np.where(h_new < 1)[0]
    if donors.size ==0 or receivers.size ==0: return h_new
    i = np.random.choice(donors)
    j = np.random.choice(receivers)
    # delta positive
    max_delta = min(h_new[i], 1 - h_new[j])
    if max_delta <= 0:
        return h_new
    delta = max_delta * np.random.rand()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

def hill_climb(h, max_iter, time_limit, start_time):
    best_h = h.copy()
    best_val = evaluate(best_h)
    N = len(h)
    # precompute maybe indices of values that are not 0 and not 1 for efficient donor/receiver.
    for it in range(max_iter):

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    # Determine N
    if 'initial_h_values' in globals():
        init = initial_h_values
        N = init.shape[0]
        h_best = np.clip(init, 0, 1)
        # adjust sum to N/2 if needed by scaling?
        current_sum = h_best.sum()
        target_sum = N/2.0
        if not np.isclose(current_sum, target_sum):
            # Rescale by adding small offset to maintain constraints
            diff = target_sum - current_sum
            # add diff uniformly across all entries while staying within bound
            # simple adjust first entries
            for i in range(N):
                can_add = 1 - h_best[i] if diff > 0 else h_best[i]
                add = np.sign(diff) * min(abs(diff), can_add)
                h_best[i] += add
                diff -= add
                if abs(diff) < 1e-12:
                    break
        best_val = evaluate(h_best)
    else:
        N = 200
        h_best = random_binary(N)
        best_val = evaluate(h_best)
    # search loop
    while time.time() - start < budget_s * 0.95:
        # time remaining
        # random start
        h = random_binary(N)
        # hill climbing with limited iterations
        h, val = hill_climb(h, max_iter=2000, time_limit=budget_s*0.95, start_time=start)
        if val < best_val:
            h_best = h
            best_val = val
    # return best solution
    return h_best, best_val, N

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    np.random.seed(seed)
    start = time.time()
    # Determine N and initial vector
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        N = h0.size
    else:
        N = 200
        h0 = None
    target_sum = N / 2.0
    # Function to project onto [0,1] and sum constraint
    def project(v):
        # projection onto hyperplane sum=target_sum intersect box
        lo, hi = -2.0, 2.0
        for _ in range(50):
            mid = (lo+hi)/2.0
            proj = np.clip(v + mid, 0.0, 1.0)
            s = proj.sum()
            if s < target_sum:
                lo = mid
            else:
                hi = mid
        return np.clip(v + (lo+hi)/2.0, 0.0, 1.0)
    # evaluation
    dx = 2.0 / N
    def evaluate(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return corr.max() * dx
    # Starting point
    if h0 is not None:
        # Ensure feasibility
        h = project(np.clip(h0, 0.0, 1.0))
    else:
        # random feasible init
        v = np.random.rand(N)
        h = project(v)
    best_h = h.copy()
    best_val = evaluate(best_h)
    # search parameters
    beta = 4.0
    lr = 0.5
    iter_count = 0
    # optimize per iteration
    while time.time() - start < budget_s * 0.95:
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        # compute smoothed max weights
        max_corr = corr.max()
        # exponent scaling
        exp_vals = np.exp(beta * (corr - max_corr))
        weights = exp_vals / exp_vals.sum()
        # gradient
        grad = np.ones(N)
        # loops over shifts
        # compute shifts range
        shift_range = np.arange(-(N-1), N)
        for w, k in zip(weights, shift_range):
            if k >= 0:
                if k < N:
                    grad[:N - k] -= w * h[k:]
                    grad[k:]   -= w * h[:N - k]
            else:
                kk = -k
                if kk < N:
                    grad[kk:]   -= w * h[:N - kk]
                    grad[:N - kk] -= w * h[kk:]
        # update step
        h_new = h - lr * grad
        # project to constraints
        h = project(h_new)
        # evaluate
        val = evaluate(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
        iter_count += 1
        # adapt parameters occasionally
        if iter_count % 200 == 0:
            lr = max(0.01, lr * 0.9)
            beta = min(20.0, beta * 1.05)
        # occasional random restart if stagnation
        if iter_count % 1000 == 0:
            # random perturbation of best
            h = best_h.copy()
            # perturb: add small noise then project
            noise = (np.random.rand(N) - 0.5) * 0.1
            h = project(h + noise)
    # final evaluate best solution
    final_val = evaluate(best_h)
    return best_h, final_val, N

import time
import numpy as np

def _project_to_box_hyperplane(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project vector v onto the intersection of the box [0,1]^n and the hyperplane
    sum(x) = target_sum.  The projection has the form clip(v + λ,0,1) for a suitable
    scalar λ, found by bisection.
    """
    lo, hi = -3.0, 3.0  # wide enough for any reasonable vector
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    λ = (lo + hi) / 2.0
    return np.clip(v + λ, 0.0, 1.0)


def _evaluate(h: np.ndarray) -> float:
    """Return C₅ for the given step function h."""
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx


def _gradient_step(h: np.ndarray, beta: float, lr: float) -> np.ndarray:
    """One gradient-descent step on the smoothed max objective."""
    N = h.shape[0]
    # correlation values (not scaled by dx – they differ by a constant factor)
    corr = np.correlate(h, 1.0 - h, mode="full")
    # numerically stable soft‑max weights
    max_corr = corr.max()
    w = np.exp(beta * (corr - max_corr))
    w /= w.sum()

    grad = np.ones(N)               # Σ_k w_k * 1
    shifts = np.arange(-(N - 1), N)  # order matches np.correlate output

    for wk, k in zip(w, shifts):
        if wk == 0.0:
            continue
        if k >= 0:
            if k < N:
                # forward term h_{j+k}
                grad[: N - k] -= wk * h[k:]
                # backward term h_{j‑k}
                grad[k:] -= wk * h[: N - k]
        else:
            kk = -k
            if kk < N:
                # now k is negative: forward term = h_{j‑kk}, backward = h_{j+kk}
                grad[kk:] -= wk * h[: N - kk]   # h_{j‑kk}
                grad[: N - kk] -= wk * h[kk:]   # h_{j+kk}
    # gradient descent update and projection
    h_new = h - lr * grad
    return h_new


def _local_hill_climb(h: np.ndarray, iters: int) -> np.ndarray:
    """Simple binary‑swap hill‑climb (works for any real values)."""
    N = h.shape[0]
    best = h.copy()
    best_val = _evaluate(best)
    for _ in range(iters):
        i, j = np.random.randint(N, size=2)
        if i == j:
            continue
        cand = best.copy()
        cand[i], cand[j] = cand[j], cand[i]          # swap preserves sum
        val = _evaluate(cand)
        if val < best_val:
            best, best_val = cand, val
    return best


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step function h, its C₅ bound, and the number of samples."""
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # initialise problem size and feasible starting vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        N = h0.shape[0]
    else:
        N = 200
        h0 = None

    target_sum = N / 2.0

    if h0 is not None:
        h = _project_to_box_hyperplane(np.clip(h0, 0.0, 1.0), target_sum)
    else:
        # random feasible start
        rand = np.random.rand(N)
        h = _project_to_box_hyperplane(rand, target_sum)

    best_h = h.copy()
    best_val = _evaluate(best_h)

    # ------------------------------------------------------------------
    # optimisation loop (mixed gradient + hill‑climb)
    # ------------------------------------------------------------------
    beta = 4.0        # smoothing parameter for the soft‑max
    lr = 0.5          # learning rate
    it = 0
    while time.time() - start < 0.95 * budget_s:
        # ----- gradient descent step -----
        h = _gradient_step(h, beta, lr)
        h = _project_to_box_hyperplane(h, target_sum)
        val = _evaluate(h)
        if val < best_val:
            best_val, best_h = val, h.copy()

        # ----- occasional hill‑climb to escape plateaus -----
        if it % 300 == 0:
            hc = _local_hill_climb(best_h, iters=500)
            hv = _evaluate(hc)
            if hv < best_val:
                best_val, best_h = hv, hc.copy()
                h = hc.copy()          # continue from this improvement

        # ----- schedule adjustments -----
        it += 1
        if it % 200 == 0:
            lr = max(0.02, lr * 0.92)   # slowly decay learning rate
            beta = min(30.0, beta * 1.04)  # sharpen the max approximation

    # final sanity check (project once more, just in case)
    best_h = _project_to_box_hyperplane(best_h, target_sum)
    final_val = _evaluate(best_h)

    return best_h, final_val, N
