import time, math, random
import numpy as np
from scipy import optimize

def _c5_value(h: np.ndarray) -> float:
    """Compute the C5 bound for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

def _project_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust h (still respecting 0‑1 bounds) so that sum(h)=target."""
    cur = h.sum()
    diff = target - cur
    # distribute the needed correction greedily
    if abs(diff) < 1e-12:
        return h
    # indices that can still move in the needed direction
    if diff > 0:
        caps = 1.0 - h
    else:
        caps = -h
    # walk through the array and fix the sum
    for i in np.where(caps != 0)[0]:
        take = min(abs(diff), abs(caps[i]))
        take = math.copysign(take, diff)
        h[i] += take
        diff -= take
        if abs(diff) < 1e-12:
            break
    return h

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Return a step function h (as a list), its C5 bound and the number of points.
    The routine tries a quick SLSQP refinement followed by simulated annealing.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # 1. Initialise h
    # ------------------------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=float)  # provided by the harness
    except Exception:                     # fallback if not supplied
        n0 = 60
        h0 = np.full(n0, 0.5, dtype=float)

    n = h0.shape[0]
    target_sum = n / 2.0

    # enforce feasibility (sum and bounds)
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = _project_sum(h0, target_sum)

    # ------------------------------------------------------------------
    # 2. Helper to keep the best solution seen so far
    # ------------------------------------------------------------------
    best_h = h0.copy()
    best_c5 = _c5_value(best_h)

    # ------------------------------------------------------------------
    # 3. Quick local optimisation with SLSQP (few restarts)
    # ------------------------------------------------------------------
    def obj(vec):
        return _c5_value(vec)

    cons = ({'type': 'eq', 'fun': lambda v: np.sum(v) - target_sum},)
    bounds = [(0.0, 1.0)] * n

    for _ in range(3):
        if time.time() - start_time > budget_s * 0.55:
            break
        # slight random perturbation of the current best
        pert = best_h + rng.uniform(-0.04, 0.04, size=n)
        pert = np.clip(pert, 0.0, 1.0)
        pert = _project_sum(pert, target_sum)

        res = optimize.minimize(obj,
                                pert,
                                method='SLSQP',
                                bounds=bounds,
                                constraints=cons,
                                options={'maxiter': 500, 'ftol': 1e-9, 'disp': False})
        if res.success:
            cand = np.clip(res.x, 0.0, 1.0)
            cand = _project_sum(cand, target_sum)
            cand_c5 = _c5_value(cand)
            if cand_c5 < best_c5:
                best_c5, best_h = cand_c5, cand.copy()

    # ------------------------------------------------------------------
    # 4. Simulated annealing / stochastic hill‑climbing
    # ------------------------------------------------------------------
    h = best_h.copy()
    cur_c5 = best_c5
    n = h.shape[0]

    # annealing schedule parameters
    max_iters = 180000
    temp_start = 0.12
    temp_end = 1e-5
    step = 0.20

    for it in range(max_iters):
        if time.time() - start_time > budget_s * 0.97:
            break

        # temperature linear in iteration index
        tau = it / max_iters
        temp = temp_start * (1.0 - tau) + temp_end * tau

        # propose a feasible swap of mass between two positions
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # feasible interval for delta (add to i, subtract from j)
        lower = max(-h[i], -(1.0 - h[j]))
        upper = min(1.0 - h[i], h[j])
        if upper <= lower:
            continue
        # limit to the current step size
        lower = max(lower, -step)
        upper = min(upper, step)
        if upper <= lower:
            continue
        delta = rng.uniform(lower, upper)

        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta

        new_c5 = _c5_value(h_new)

        # Metropolis acceptance
        if new_c5 < cur_c5 or rng.random() < math.exp(-(new_c5 - cur_c5) / temp):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5, best_h = new_c5, h.copy()

        # slowly shrink the maximum step
        if (it + 1) % 1000 == 0 and step > 5e-4:
            step *= 0.985

    # ------------------------------------------------------------------
    # 5. Return the best solution found
    # ------------------------------------------------------------------
    return (best_h.tolist(), float(best_c5), int(n))
