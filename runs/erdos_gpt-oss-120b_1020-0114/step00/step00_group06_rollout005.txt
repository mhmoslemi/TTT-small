def improve_h(h, max_iter, dt):
    n = len(h)
    dx = 2.0 / n
    for iteration in range(max_iter):
        # compute correlation
        corr = np.correlate(h, 1-h, mode='full')
        idx = corr.argmax()
        s = idx - (n-1)
        # compute improvement potential
        improved = False
        # For each i where j valid
        if s >= 0:
            i_range = range(s, n)
            for i in i_range:
                j = i - s
                diff = h[i] - h[j]
                if diff > 1e-8:
                    epsilon = min(diff/2, dt)
                    if h[i] - epsilon >= 0 and h[j] + epsilon <= 1:
                        h[i] -= epsilon
                        h[j] += epsilon
                        improved = True
        else:
            # s negative
            i_range = range(0, n + s)   # because s negative
            for i in i_range:
                j = i - s # i - (-|s|) = i + |s|
                diff = h[i] - h[j]
                if diff > 1e-8:
                    epsilon = min(diff/2, dt)
                    if h[i] - epsilon >= 0 and h[j] + epsilon <= 1:
                        h[i] -= epsilon
                        h[j] += epsilon
                        improved = True
        if not improved:
            # may break or reduce dt
            dt *= 0.5
            if dt < 1e-6:
                break

                diff = diffs[idx]
                if abs(diff) < eps:
                    # nothing useful – do a random move
                    i, j = rng.integers(0, n, size=2)
                    if i == j:
                        continue
                    source, target = (i, j) if rng.random() < 0.5 else (j, i)
                    max_delta = min(h[source], 1.0 - h[target])
                    if max_delta <= 0:
                        continue
                    delta = max_delta * rng.random()
                    h[source] -= delta
                    h[target] += delta
                else:
                    i = i_range[idx]
                    j = j_range[idx]
                    if diff > 0:
                        source, target = i, j
                    else:
                        source, target = j, i
                    max_delta = min(h[source], 1.0 - h[target])
                    if max_delta > 0:
                        # move a fraction of the imbalance
                        delta = min(abs(diff) / 2.0, max_delta) * (0.5 + 0.5 * rng.random())
                        h[source] -= delta
                        h[target] += delta
        else:
            # simple random mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            source, target = (i, j) if rng.random() < 0.5 else (j, i)
            max_delta = min(h[source], 1.0 - h[target])
            if max_delta <= 0:
                continue
            delta = max_delta * rng.random()
            h[source] -= delta
            h[target] += delta

        # ----- evaluate -------------------------------------------------------
        new_max, new_idx, new_corr = _max_corr_and_shift(h)
        new_c5 = _c5_from_corr(new_max, n)

        # ----- acceptance ------------------------------------------------------
        delta_c5 = new_c5 - cur_c5
        accept = False
        if delta_c5 <= 0:
            accept = True
        elif T > Tmin and rng.random() < math.exp(-delta_c5 / T):
            accept = True

        if accept:
            cur_c5 = new_c5
            cur_idx = new_idx
            cur_corr = new_corr
            # keep best ever
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                best_corr = cur_corr.copy()
        else:
            h[:] = old_h                     # revert

        # ----- cooling ---------------------------------------------------------
        T = max(T * decay, Tmin)

    # final sanity check – enforce constraints within tolerance
    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = best_h.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        # redistribute the tiny error uniformly on non‑boundary entries
        deficit = target_sum - cur_sum
        free = (best_h > 0) & (best_h < 1)
        if free.any():
            adj = deficit / free.sum()
            best_h[free] += adj
            best_h = np.clip(best_h, 0.0, 1.0)

    # compute final bound (use the helper if available)
    try:
        final_c5 = evaluate_erdos_solution(best_h)
    except Exception:
        final_c5 = _c5_from_corr(best_corr.max(), n)

    return best_h, float(final_c5), n
