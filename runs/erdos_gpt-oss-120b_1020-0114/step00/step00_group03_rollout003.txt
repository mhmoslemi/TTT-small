import time
import numpy as np

def compute_c5(h):
    """Return max overlap integral for discretized h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(dx * np.max(corr))


def _local_optimize(start_h, max_seconds, rng):
    """Hill‑climbing / simulated annealing on a fixed grid."""
    n = start_h.shape[0]
    dx = 2.0 / n
    cur = start_h.copy()
    cur_score = compute_c5(cur)
    best = cur.copy()
    best_score = cur_score

    start = time.time()
    eps = 1e-12
    max_step = 0.3  # maximal transferable mass at the beginning

    while time.time() - start < max_seconds:
        # worst (minimum) autocorrelation → shift that limits C5
        auto = np.correlate(cur, cur, mode="full")
        G = auto * dx
        min_idx = np.argmin(G)
        s = min_idx - (n - 1)

        # partner sum = contribution of each entry to the worst shift
        partner = np.zeros(n)
        if s >= 0:
            partner[: n - s] += cur[s:]
            partner[s:] += cur[: n - s]
        else:
            ss = -s
            partner[ss:] += cur[: n - ss]
            partner[: n - ss] += cur[ss:]

        # indices with large / small partner sums
        idx_sorted = np.argsort(partner)
        k20 = max(1, int(0.2 * n))
        src_cand = idx_sorted[:k20]
        tgt_cand = idx_sorted[-k20:]

        src = [j for j in src_cand if cur[j] > eps]
        tgt = [i for i in tgt_cand if cur[i] < 1.0 - eps]

        if src and tgt:
            i = rng.choice(tgt)
            j = rng.choice(src)
            max_delta = min(cur[j], 1.0 - cur[i])
        else:
            # random feasible pair fallback
            found = False
            for _ in range(5):
                i = rng.randint(n)
                j = rng.randint(n)
                if i != j and cur[i] < 1.0 - eps and cur[j] > eps:
                    max_delta = min(cur[j], 1.0 - cur[i])
                    found = True
                    break
            if not found:
                break

        # shrink step size as time runs out
        elapsed = time.time() - start
        remaining = max_seconds - elapsed
        step_frac = remaining / max_seconds
        limit = max_step * step_frac
        max_delta = min(max_delta, limit)
        if max_delta <= 0:
            continue

        delta = max_delta * rng.rand()
        new = cur.copy()
        new[i] += delta
        new[j] -= delta
        new_score = compute_c5(new)

        if new_score < cur_score:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
        else:
            # occasional uphill move to escape plateaus
            if rng.rand() < 1e-4:
                cur = new
                cur_score = new_score

    return best, best_score
