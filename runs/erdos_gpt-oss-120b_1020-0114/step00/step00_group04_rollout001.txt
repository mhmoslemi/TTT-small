def run(seed=42, budget_s=1100, **kwargs):

    np.random.seed(seed)
    start_time = time.time()
    # Determine n_points
    init_h = globals().get('initial_h_values', None)
    if init_h is not None:
        h0 = np.array(init_h, copy=False)
        n = h0.size
    else:
        n = 42
        h0 = np.full(n, 0.5) # feasible
    target_sum = n / 2.0
    dx = 2.0 / n
    # Ensure feasibility
    def project(v):
        return project_to_box_sum(v, target_sum)
    # If init_h might not be feasible (due rounding), project.
    h0 = project(h0.astype(float))
    best_h = h0.copy()
    best_val = compute_c5(best_h)
    # Set up parameters
    current_h = best_h.copy()
    current_val = best_val
    # Simulated annealing parameters
    T = 0.1  # initial temperature
    T_min = 1e-6
    cooling = 0.9995
    step = 0.3  # max delta for moves
    step_min = 0.001
    step_decay = 0.9999
    it = 0
    stagnation = 0
    max_stagnation = 20000
    while True:

        if elapsed > budget_s - 0.5:
            break
        # propose move
        i, j = np.random.choice(n, size=2, replace=False)
        # compute feasible delta interval
        lo = max(-step, -current_h[i], current_h[j] - 1.0)
        hi = min(step, 1.0 - current_h[i], current_h[j])

        new_h = current_h.copy()
        new_h[i] += delta
        new_h[j] -= delta

        new_val = compute_c5(new_h)

            current_val = new_val
            # Update best

                best_h = new_h.copy()
                stagnation = 0
            else:
                stagnation += 1
        else:

import numpy as np
import time
import math

def project_to_box_sum(v, target, lower=0.0, upper=1.0, max_iter=60, eps=1e-12):
    """
    Project vector v onto the box [lower,upper]^n with prescribed sum=target.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo, hi = -1e6, 1e6
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, lower, upper)
        s = x.sum()
        if abs(s - target) <= eps:
            return x
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, lower, upper)

def compute_c5(h, dx):
    """Maximum overlap integral for a discrete step function h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def propose_move(h, step):
    """Generate a feasible move that preserves sum and bounds."""
    n = h.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)
    lo = max(-step, -h[i], h[j] - 1.0)
    hi = min(step, 1.0 - h[i], h[j])
    if lo >= hi:
        return None
    delta = np.random.uniform(lo, hi)
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    return new_h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Initialise problem size and feasible start vector
    # ------------------------------------------------------------------
    init = globals().get('initial_h_values', None)
    if init is not None:
        h0 = np.asarray(init, dtype=float).copy()
        n = h0.size
    else:
        n = 42
        h0 = np.full(n, 0.5, dtype=float)

    target_sum = n * 0.5            # because integral over [0,2] must be 1
    dx = 2.0 / n

    # Ensure feasibility of the start (projection onto the bounded simplex)
    h0 = project_to_box_sum(h0, target_sum)

    best_h = h0.copy()
    best_val = compute_c5(best_h, dx)

    # ------------------------------------------------------------------
    # Simulated annealing / hill climbing
    # ------------------------------------------------------------------
    current_h = best_h.copy()
    current_val = best_val

    # annealing schedule
    T = 0.12               # initial temperature
    T_min = 1e-6
    cooling = 0.9995

    # step size schedule
    step = 0.30
    step_min = 0.001
    step_decay = 0.9999

    # stagnation handling
    stagnation = 0
    max_stagnation = 25000   # after this many non‑improving moves we restart

    iteration = 0
    # Main loop respects the time budget
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s - 0.5:
            break

        # propose a move
        new_h = propose_move(current_h, step)
        if new_h is None:
            continue

        new_val = compute_c5(new_h, dx)
        iteration += 1

        # Acceptance rule (Metropolis)
        if new_val < current_val or np.random.rand() < math.exp(-(new_val - current_val) / max(T, 1e-12)):
            current_h = new_h
            current_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
                stagnation = 0
            else:
                stagnation += 1
        else:
            stagnation += 1

        # cooling and step decay
        T = max(T * cooling, T_min)
        step = max(step * step_decay, step_min)

        # occasional restart if stuck
        if stagnation > max_stagnation:
            rand_vec = np.random.rand(n)
            current_h = project_to_box_sum(rand_vec, target_sum)
            current_val = compute_c5(current_h, dx)
            stagnation = 0
            step = max(step, 0.2)
            T = max(T, 0.05)

        # every so often, re‑seed the neighbourhood around the best solution
        if iteration % 12000 == 0:

            pert = project_to_box_sum(pert, target_sum)
            mix = 0.88 * best_h + 0.12 * pert
            current_h = project_to_box_sum(mix, target_sum)
            current_val = compute_c5(current_h, dx)

    # ------------------------------------------------------------------
    # Final pure descent (zero temperature) to polish the best found point
    # ------------------------------------------------------------------
    final_step = 0.05
    for _ in range(30000):

        if cand is None:

        cand_val = compute_c5(cand, dx)
        if cand_val < best_val:
            best_h = cand
            best_val = cand_val

    best_h = project_to_box_sum(best_h, target_sum)

    # Return as plain Python list for compatibility
    return (best_h.tolist(), float(best_val), int(n))
