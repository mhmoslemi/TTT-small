def project_onto_box_sum(v, target, lo=0.0, hi=1.0, max_iter=100):
    # Use bisection on lambda
    # Clip v - lam to [lo, hi]
    # sum = target
    # Decreasing in lam -> monotonic.
    low = np.min(v) - hi
    high = np.max(v) - lo
    for _ in range(max_iter):
        lam = (low + high) / 2.0
        x = np.clip(v - lam, lo, hi)
        s = x.sum()
        if s > target:
            low = lam
        else:
            high = lam
        # If convergence small break
        if high - low < 1e-12:
            break
    return np.clip(v - ((low + high) / 2.0), lo, hi)

import time
import numpy as np

def _project_onto_box_sum(v, target, lo=0.0, hi=1.0, max_iter=60):
    """ Euclidean projection of v onto {x: lo<=x<=hi, sum(x)=target}. """
    lo_lam = np.min(v) - hi
    hi_lam = np.max(v) - lo
    for _ in range(max_iter):
        lam = (lo_lam + hi_lam) / 2.0
        x = np.clip(v - lam, lo, hi)
        s = x.sum()
        if s > target:
            lo_lam = lam
        else:
            hi_lam = lam
        if hi_lam - lo_lam < 1e-14:
            break
    return np.clip(v - (lo_lam + hi_lam) / 2.0, lo, hi)


def _gradient_of_shift(h, k, dx):
    """Gradient of the correlation at shift k (worst shift)."""
    N = h.shape[0]
    g = np.zeros(N, dtype=h.dtype)
    if k >= 0:
        end = N - k
        if end > 0:
            g[:end] = 1.0 - h[k:]
            g[k:] -= h[:end]
    else:
        kk = -k
        end = N - kk
        if end > 0:
            g[kk:] = 1.0 - h[:end]
            g[:end] -= h[kk:]
    return dx * g


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ----- initialise -------------------------------------------------
    init_h = globals().get("initial_h_values", None)
    if init_h is not None:
        h = np.array(init_h, dtype=float)
        N = h.shape[0]
        target = N / 2.0
        h = _project_onto_box_sum(h, target)
    else:
        N = 200                     # a modest resolution
        dx = 2.0 / N
        x = (np.arange(N) + 0.5) * dx     # mid‑point sample
        # triangular “tent” shape, exact integral = 1
        h = np.minimum(x, 2.0 - x)
        target = N / 2.0
        h = _project_onto_box_sum(h, target)

    dx = 2.0 / N
    target = N / 2.0

    # ----- optimisation ------------------------------------------------
    best_h = h.copy()
    best_val = np.max(np.correlate(best_h, 1.0 - best_h, mode="full")) * dx

    start = time.time()
    it = 0
    no_improve = 0
    # parameters for step‑size schedule
    alpha0 = 1.0

    while True:
        elapsed = time.time() - start
        if elapsed > budget_s - 0.2:
            break

        # current objective
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        cur_val = corr.max()

        # keep best ever
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # worst shift (the one attaining the max)
        idx = int(np.argmax(corr))
        k = idx - (N - 1)

        # sub‑gradient step
        it += 1
        step_size = alpha0 / np.sqrt(it)
        grad = _gradient_of_shift(h, k, dx)
        h_new = h - step_size * grad

        # occasional random jitter to escape plateaus
        if it % 1000 == 0:
            jitter = rng.normal(scale=0.005, size=N)
            h_new += jitter

        # project back onto feasible set
        h = _project_onto_box_sum(h_new, target)

        # if stuck for a long time, re‑jitter around the current best
        if no_improve >= 5000:
            h = best_h + rng.normal(scale=0.01, size=N)
            h = _project_onto_box_sum(h, target)
            no_improve = 0

    # final evaluation of the best candidate found
    final_val = np.max(np.correlate(best_h, 1.0 - best_h, mode="full")) * dx
    return (best_h, float(final_val), N)
