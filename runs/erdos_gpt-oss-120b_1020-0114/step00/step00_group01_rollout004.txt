    corr = np.correlate(h, 1 - h, mode='full')

    # parameters
    T0 = 0.1

import time, random, math
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # linear cross‑correlation h * (1‑h)  (mode="full")
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_onto_constraints(v: np.ndarray) -> np.ndarray:
    """
    Projection of a vector onto the set
        {x ∈ [0,1]^n  :  sum x = n/2 }.
    Uses a binary search for the Lagrange multiplier λ in
        x_i = clip(v_i - λ, 0, 1).
    """
    n = v.shape[0]
    target = n * 0.5
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

        a = h[: n - shift]
        b = h[shift:]
        g[: n - shift] += 1.0 - b
        g[shift:] += -a
    else:
        d = -shift
        a = h[d:]
        b = h[: n - d]
        g[d:] += 1.0 - b
        g[: n - d] += -a
    return g


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start = time.time()
    total_budget = budget_s

    # ------------------------------------------------------------------
    # initialise a feasible vector h
    # ------------------------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=float)   # provided by the harness
        n = h0.shape[0]
        # ensure feasibility (tiny numerical drift may appear)
        h0 = project_onto_constraints(h0)
        h = h0.copy()
    except NameError:
        # no parent supplied – build a fresh start with n even
        n = 100
        if n % 2:
            n += 1
        # random values then project onto constraints
        v = np.random.rand(n)
        h = project_onto_constraints(v)

    # best solution seen so far
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # ------------------------------------------------------------------
    # simulated‑annealing core loop
    # ------------------------------------------------------------------
    # temperature schedule (linear decay to a small floor)
    T_start = 0.03
    T_min   = 5e-6
    move_prob_swap = 0.4          # probability of performing a pure swap

    while time.time() - start < total_budget * 0.88:
        # progress controls temperature
        elapsed = time.time() - start
        prog = elapsed / (total_budget * 0.88)
        T = max(T_start * (1.0 - prog), T_min)

        #---------------  propose a move  -------------------------------
        if random.random() < move_prob_swap:
            # pure swap of two positions
            i, j = random.sample(range(n), 2)
            if h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]
            new_c5 = compute_c5(h)
            accept = (new_c5 <= best_c5) or (random.random() < math.exp(-(new_c5 - best_c5) / T))
            if accept:
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                # revert
                h[i], h[j] = h[j], h[i]
        else:
            # transfer a small amount δ from index j to i (keeps sum exact)
            i, j = random.sample(range(n), 2)
            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], h[j])
            if lo >= hi:
                continue
            delta = random.uniform(lo, hi)
            h[i] += delta
            h[j] -= delta
            new_c5 = compute_c5(h)
            accept = (new_c5 <= best_c5) or (random.random() < math.exp(-(new_c5 - best_c5) / T))
            if accept:
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                # revert
                h[i] -= delta
                h[j] += delta

    # ------------------------------------------------------------------
    # gradient‑based polishing (uses the current best vector)
    # ------------------------------------------------------------------
    h = best_h.copy()
    # small amount of time reserved for polishing
    polish_deadline = start + total_budget * 0.96
    lr = 0.12                      # initial learning rate
    while time.time() < polish_deadline:
        cur = compute_c5(h)
        # shift achieving the maximum overlap
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        grad = gradient_for_shift(h, shift)
        # descent step
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_onto_constraints(h_new)
        new = compute_c5(h_new)
        if new < cur:
            h = h_new
            if new < best_c5:
                best_c5 = new
                best_h = h.copy()
        else:
            lr *= 0.5
            if lr < 1e-7:
                break

    # final evaluation (safety)
    final_c5 = compute_c5(best_h)

    return (best_h, final_c5, n)
