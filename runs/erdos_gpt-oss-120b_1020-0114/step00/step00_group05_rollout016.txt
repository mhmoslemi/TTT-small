def proj_simplex_box(v, target):
    # ensure target in [0, len(v)]
    lo = -1.0  # lower bound for lambda
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # final small correction:
    diff = target - w.sum()
    if abs(diff) > 1e-9:
        # adjust among non-saturated entries
        free = np.where((w > 1e-9) & (w < 1 - 1e-9))[0]
        if free.size > 0:
            w[free] += diff / free.size
            w = np.clip(w, 0.0, 1.0)
        else:
            # All entries at boundary, adjust largest slack
            # (should not happen for typical target= n/2)
            # distribute to all entries proportionally?
            w = np.clip(w + diff / len(w), 0, 1)
    return w

import numpy as np
import time
import math

def _proj_capped_simplex(v, target):
    """
    Euclidean projection of vector v onto the set {x | 0 <= x_i <= 1, sum x_i = target}.
    Binary search for the Lagrange multiplier λ such that sum_i clip(v_i-λ,0,1)=target.
    """
    lo = -1.0                           # ensures all entries are clipped to 1
    hi = np.max(v) + 1.0                # ensures all entries are clipped to 0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny adjustment to hit the target more precisely
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = np.where((w > 1e-9) & (w < 1.0 - 1e-9))[0]
        if free.size:
            w[free] += diff / free.size
            w = np.clip(w, 0.0, 1.0)
        else:
            # fallback – distribute uniformly (should rarely happen)
            w += diff / w.size
            w = np.clip(w, 0.0, 1.0)
    return w

def _c5_value(h):
    """C5 for a discretised step function."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)

def _gradient_for_lag(h, lag):
    """g_i = 1 - h_{i+lag} - h_{i-lag} (missing terms omitted)."""
    n = h.shape[0]
    g = np.ones(n)
    if lag > 0:
        # i+lag valid for i=0..n-1-lag
        g[:n - lag] -= h[lag:]
        # i-lag valid for i=lag..n-1
        g[lag:] -= h[:n - lag]
    elif lag < 0:
        lag = -lag
        # i+lag valid for i=lag..n-1  (i+lag = i-lag_abs)
        g[lag:] -= h[:n - lag]
        # i-lag valid for i=0..n-1-lag
        g[:n - lag] -= h[lag:]
    else:  # lag == 0
        g = 1.0 - 2.0 * h
    return g

def _local_optimize(start_h, target_sum, time_limit):
    """
    Gradient‑descent with occasional random jitter and restarts.
    Returns best vector and its C5 found within time_limit seconds.
    """
    n = start_h.shape[0]
    h = start_h.copy()
    best_h = h.copy()
    best_val = _c5_value(best_h)

    start = time.time()
    no_improve = 0
    # parameters
    max_step = 0.6
    min_step = 0.001
    jitter_prob = 0.02
    jitter_scale = 0.25
    restart_after = 3000

    while time.time() - start < time_limit:
        # current value and lag producing the maximum
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_val = corr.max() * (2.0 / n)
        lag = np.argmax(corr) - (n - 1)

        # step size decays with elapsed time
        elapsed = time.time() - start
        frac = elapsed / time_limit
        step = max_step * (1.0 - frac) + min_step * frac

        # gradient for the worst lag
        g = _gradient_for_lag(h, lag)

        # propose move
        new_h = h - step * g
        if rng.random() < jitter_prob:
            new_h += rng.normal(scale=jitter_scale, size=n)
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = _proj_capped_simplex(new_h, target_sum)

        new_val = _c5_value(new_h)

        # accept only if we improve
        if new_val < cur_val - 1e-12:
            h = new_h
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
        else:
            no_improve += 1

        # occasional restart to escape possible stagnation
        if no_improve >= restart_after:
            h = _proj_capped_simplex(rng.random(n), target_sum)
            no_improve = 0

    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points) where h_values is a numpy array
    representing a step function on [0,2] with integral 1 and values in [0,1].
    """
    global initial_h_values
    rng = np.random.default_rng(seed)

    # Determine dimension and starting vector
    try:
        init_arr = np.asarray(initial_h_values, dtype=float)
        n = init_arr.shape[0]
        h0 = np.clip(init_arr, 0.0, 1.0)
    except Exception:
        n = 70
        h0 = rng.random(n)

    target_sum = n / 2.0                     # because ∫h = 1  ⇒  sum(h)*dx = 1  ⇒ sum(h)=n/2
    h0 = _proj_capped_simplex(h0, target_sum)

    total_budget = budget_s * 0.95
    start_total = time.time()
    best_h = h0.copy()
    best_val = _c5_value(best_h)

    # First, improve the provided (or generated) start point
    remaining = total_budget - (time.time() - start_total)
    if remaining > 0.5:
        local_h, local_val = _local_optimize(best_h, target_sum, remaining * 0.5)
        if local_val < best_val:
            best_h, best_val = local_h, local_val

    # Use the remaining time for several random restarts
    while True:
        elapsed = time.time() - start_total
        if elapsed >= total_budget:
            break
        remaining = total_budget - elapsed
        # give each restart a modest slice of the remaining budget
        slice_time = min(5.0, remaining * 0.4)
        start_vec = _proj_capped_simplex(rng.random(n), target_sum)
        local_h, local_val = _local_optimize(start_vec, target_sum, slice_time)
        if local_val < best_val:
            best_h, best_val = local_h, local_val

    return best_h, best_val, n
