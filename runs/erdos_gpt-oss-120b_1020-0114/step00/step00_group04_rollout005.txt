def project_onto_capped_simplex(v, S):
    # v: numpy array shape (n), not necessarily feasible.
    # find λ such that w = np.clip(v + λ, 0, 1) satisfies sum(w) = S. If S outside feasible range (i.e., S < 0 or > n), but our S should be between 0 and n.
    # Use binary search.
    lo = -2.0
    hi = 2.0
    for _ in range(50):
        lam = (lo+hi)/2
        w = np.clip(v + lam, 0, 1)
        total = w.sum()
        if total > S:
            hi = lam
        else:
            lo = lam
    w = np.clip(v + (lo+hi)/2, 0, 1)
    # adjust rounding errors:
    # after projection, sum may be slightly off, we can adjust a few entries not at bounds.
    diff = S - w.sum()
    if abs(diff) > 1e-8:
        # distribute diff equally among free entries (not at bounds)
        free = (w > 1e-12) & (w < 1-1e-12)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0, 1) # reclip
        else:
            # rare case all at bounds, adjust anyway
            w[0] = np.clip(w[0] + diff, 0, 1)
    return w

            w = np.clip(w, 0, 1)
        else:
            # fallback: adjust first element
            w[0] = np.clip(w[0] + diff, 0, 1)
    return w

import time, math, random
import numpy as np

def _compute_c5(h, dx):
    """Return max_k ∫ h(x)(1‑h(x+k)) dx for the discretised h."""
    # correlation with zero padding (full mode)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_capped_simplex(v, target_sum):
    """
    Project vector v onto the set {0≤x_i≤1, Σx_i = target_sum}.
    Uses a simple bisection on the Lagrange multiplier λ for the
    shifted‑clip formulation x_i = clip(v_i+λ,0,1).
    """
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v + lam, 0.0, 1.0)
        s = w.sum()
        if s > target_sum:
            hi = lam
        else:
            lo = lam
    w = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

    # tiny correction for numerical drift
    diff = target_sum - w.sum()
    if abs(diff) > 1e-10:
        free = (w > 1e-12) & (w < 1.0 - 1e-12)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0.0, 1.0)
        else:
            w[0] = np.clip(w[0] + diff, 0.0, 1.0)
    return w

def _random_feasible(n, total):
    """Draw a random vector in [0,1]^n with prescribed sum `total`."""
    v = np.random.rand(n)
    return _project_capped_simplex(v, total)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise
        C5 = max_k ∫ h(x)(1‑h(x+k)) dx

    Returns (h_values, c5_bound, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)

    start = time.time()

    # ------------------------------------------------------------------
    # Choose a modest number of discretisation points.
    # If a parent construction is supplied we embed it, otherwise start
    # from scratch.
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
        base_n = len(init)
        # use a modest multiple (2) to give the optimiser more freedom
        n = min(200, base_n * 2)
        if n % 2:
            n += 1
        # simple up‑sampling by repetition
        reps = n // base_n
        extra = n % base_n
        h = np.concatenate([np.repeat(init, reps), init[:extra]])
        h = _project_capped_simplex(h, n / 2.0)
    else:
        n = 120                # default resolution when no parent is present
        if n % 2:
            n += 1
        h = _random_feasible(n, n / 2.0)

    dx = 2.0 / n
    target_sum = n / 2.0

    best_h = h.copy()
    best_score = _compute_c5(best_h, dx)
    cur_score = best_score

    # ------------------------------------------------------------------
    # Simulated‑annealing style optimisation.
    # Moves: pick two distinct indices i,j and transfer a feasible amount δ
    # from one to the other while respecting the box constraints.
    # ------------------------------------------------------------------
    it = 0
    stalls = 0
    while time.time() - start < budget_s * 0.95:
        # pick two distinct positions
        i, j = np.random.randint(n), np.random.randint(n)
        if i == j:
            continue
        hi, hj = h[i], h[j]

        # feasible interval for δ (add to i, subtract from j)
        lo_delta = max(-hi, hj - 1.0)
        hi_delta = min(1.0 - hi, hj)
        if lo_delta >= hi_delta:
            continue

        delta = np.random.uniform(lo_delta, hi_delta)

        # propose the move
        h[i] = hi + delta
        h[j] = hj - delta

        new_score = _compute_c5(h, dx)

        # temperature decays linearly with elapsed time
        elapsed = time.time() - start
        T = max(1e-4, 0.02 * (1.0 - elapsed / (budget_s * 0.95)))

        if new_score < cur_score or np.random.rand() < math.exp(-(new_score - cur_score) / T):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                stalls = 0
            else:
                stalls += 1
        else:
            # revert
            h[i] = hi
            h[j] = hj
            stalls += 1

        # occasional diversification when stuck
        if stalls > 5000:
            # perturb the current state a little and re‑project
            perturb = np.random.randn(n) * 0.02
            h = best_h + perturb
            h = np.clip(h, 0.0, 1.0)
            h = _project_capped_simplex(h, target_sum)
            cur_score = _compute_c5(h, dx)
            stalls = 0

        # occasional full random restart (keeps the best ever found)
        if it % 20000 == 0 and it > 0 and np.random.rand() < 0.2:
            cand = _random_feasible(n, target_sum)
            cand_score = _compute_c5(cand, dx)
            h = cand
            cur_score = cand_score

        it += 1

    # ------------------------------------------------------------------
    # Finalise the best solution found.
    # ------------------------------------------------------------------
    best_h = _project_capped_simplex(best_h, target_sum)
    final_score = _compute_c5(best_h, dx)

    return best_h.tolist(), float(final_score), n
