import numpy as np
import time
import math

def project_to_constraints(x, target_sum):
    # clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
    # adjust sum
    current = np.sum(x)
    # if already target within tolerance, return
    if abs(current - target_sum) < 1e-12:
        return x
    # water-filling method
    # We'll iteratively adjust
    while True:
        diff = target_sum - current
        if abs(diff) < 1e-12:
            break
        # indices that can be increased if diff positive, else decreased
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        delta = diff / cnt
        # apply delta but not exceed bounds
        # for each index:
        #   new = x[i] + delta
        #   if new > 1: cap to 1, adjust diff accordingly
        #   if new < 0: cap to 0
        # We can vectorize:
        if diff > 0:
            # increase
            inc = np.minimum(delta, 1.0 - x)
            inc = inc * mask
            x += inc
            current = np.sum(x)
        else:
            # decrease
            dec = np.minimum(-delta, x)
            dec = dec * mask
            x -= dec
            current = np.sum(x)
    return x
```

def project_to_constraints(x, target):
    x = np.clip(x, 0.0, 1.0)
    # iterative water-filling
    for _ in range(1000):
        s = np.sum(x)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # eligible indices
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            # cannot adjust further
            break
        delta = diff / cnt
        # update only eligible entries, but ensure they stay within bounds
        if diff > 0:
            # increase
            inc = np.minimum(delta, 1.0 - x)
            inc = inc * mask
            x += inc
        else:
            # decrease
            dec = np.minimum(-delta, x)
            dec = dec * mask
            x -= dec
    # final clip
    return np.clip(x, 0.0, 1.0)
```

def propose_move(h):

    delta = np.random.uniform(lo, hi_delta)
    new_h = h.copy()
    new_h[i] = hi + delta
    new_h[j] = hj - delta

def gradient_step(h, eta):
    n = len(h)
    # compute corr via np.correlate? We'll need array of C values:
    corr = np.correlate(h, 1 - h, mode='full')
    # find lag index of max correlation:
    lag = np.argmax(corr) - (n - 1)  # because "full" correlation length 2n-1, index offset
    # gradient contributions:
    g = np.zeros(n)
    # for i from 0 to n-1-lag-1? we need to compute for indices where i+lag in range
    if lag >= 0:
        # i range 0 .. n-1-lag
        i_range = slice(0, n - lag)
        # g[i] contribution from term (1 - h[i+lag])
        g[i_range] += (1 - h[i_range + lag])
        # also for i_range2 where i-lag >=0, i.e., i from lag .. n-1
        i_range2 = slice(lag, n)
        # subtract h[i - lag] to gradient
        g[i_range2] -= h[i_range2 - lag]
    else:
        # lag negative
        k = -lag
        # i+lag is i-k, so shift left.
        # contributions:
        # for i in k..n-1: term (1 - h[i+lag]) => (1 - h[i-k])
        i_range = slice(k, n)
        g[i_range] += (1 - h[i_range - k])
        # subtract h[i - lag] = h[i + k] for i in 0..n-k-1
        i_range2 = slice(0, n - k)
        g[i_range2] -= h[i_range2 + k]
    # Move in negative gradient direction to reduce C:
    d = -g
    # Project d onto sum-zero subspace:
    d = d - np.mean(d)
    # take step
    h_new = h + eta * d
    # Clip and project to constraints:
    target = len(h) / 2.0
    h_new = project_to_constraints(h_new, target)
    return h_new

    if abs(s - target) < 1e-12:

    # Use water-filling:
    # We will adjust values in direction of desire.
    # While loop
    max_iter = 1000
    for _ in range(max_iter):
        diff = target - s
        if abs(diff) < 1e-9:
            break
        if diff > 0:
            # Need to increase some values that are not at 1
            free = (x < 1.0 - 1e-12)
            cnt = np.count_nonzero(free)
            if cnt == 0:
                break
            delta = diff / cnt
            inc = np.minimum(delta, 1.0 - x)
            x += inc * free
        else:
            # diff < 0, need to decrease values not at 0
            free = (x > 1e-12)
            cnt = np.count_nonzero(free)

            dec = np.minimum(-delta, x)
            x -= dec * free
        s = np.sum(x)

import time, math
import numpy as np

# ---------- helpers ---------------------------------------------------------

def evaluate_c5(h):
    """Return the C5 value for a discretised step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_to_constraints(x, target):
    """Clip x to [0,1] and adjust to have sum == target (box + equality)."""
    x = np.clip(x, 0.0, 1.0)
    s = np.sum(x)
    if abs(s - target) < 1e-12:
        return x
    # water‑filling style iterative projection
    for _ in range(500):
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = x < 1.0
            cnt = np.count_nonzero(free)
            if cnt == 0:
                break
            delta = diff / cnt
            inc = np.minimum(delta, 1.0 - x[free])
            x[free] += inc
        else:
            free = x > 0.0
            cnt = np.count_nonzero(free)
            if cnt == 0:
                break
            delta = diff / cnt  # negative
            dec = np.minimum(-delta, x[free])
            x[free] -= dec
        s = np.sum(x)
    return np.clip(x, 0.0, 1.0)


def propose_move(h):
    """Swap a random amount of mass between two coordinates while keeping constraints."""
    n = h.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)
    hi, hj = h[i], h[j]
    lo = max(-hi, -(1.0 - hj))
    hi_delta = min(1.0 - hi, hj)
    if lo >= hi_delta:
        return None
    delta = np.random.uniform(lo, hi_delta)
    new_h = h.copy()
    new_h[i] = hi + delta
    new_h[j] = hj - delta
    return new_h


def get_initial_h(n):
    """Create a feasible start vector, interpolating from the provided global if present."""
    target = n / 2.0
    try:
        base = np.asarray(initial_h_values, dtype=np.float64)
        n0 = base.shape[0]
        if n0 == n:
            h = base.copy()
        else:
            xs_old = np.linspace(0.0, 2.0, n0, endpoint=False)
            xs_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(xs_new, xs_old, base)
    except Exception:
        h = np.random.rand(n)
    # scale to correct total mass, then enforce box constraints
    s = np.sum(h)
    if s > 0:
        h *= target / s
    h = project_to_constraints(h, target)
    return h


def max_corr_lag(h):
    """Return lag (as integer index offset) where C(k) attains its maximum."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    lag = idx - (n - 1)
    return lag, corr


def gradient_step(h, eta, target):
    """One projected gradient step aimed at reducing the worst overlap."""
    n = h.shape[0]
    lag, _ = max_corr_lag(h)
    g = np.zeros(n)
    if lag >= 0:
        end = n - lag
        if end > 0:
            g[:end] += (1.0 - h[lag:])
            g[lag:] -= h[:end]
    else:
        k = -lag
        end = n - k
        if end > 0:
            g[k:] += (1.0 - h[:end])
            g[:end] -= h[k:]
    d = -g
    d -= np.mean(d)               # keep sum zero
    h_new = h + eta * d
    h_new = project_to_constraints(h_new, target)
    return h_new


# ---------- main optimisation routine ----------------------------------------

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound, and the number of points used."""
    np.random.seed(seed)
    start_time = time.time()
    end_time = start_time + budget_s - 2.0   # small safety margin

    # decide number of discretisation points
    default_n = 200
    try:
        base_n = len(initial_h_values)
        n_points = max(base_n, default_n)
        n_points = min(n_points, 800)   # keep under 1000 as requested
    except Exception:
        n_points = default_n

    target = n_points / 2.0
    h_curr = get_initial_h(n_points)
    best_h = h_curr.copy()
    best_val = evaluate_c5(best_h)
    curr_val = best_val

    # annealing / temperature schedule parameters
    T0 = 0.02
    eta0 = 0.15
    it = 0

    while time.time() < end_time:
        # linear cooling
        elapsed = time.time() - start_time
        progress = elapsed / (budget_s - 2.0)
        T = max(T0 * (1.0 - progress), 1e-6)
        eta = eta0 * (1.0 - progress)

        if np.random.rand() < 0.6:
            # gradient descent oriented move on the worst shift
            h_new = gradient_step(h_curr, eta, target)
        else:
            # simple two‑index mass exchange
            h_new = propose_move(h_curr)
            if h_new is None:
                continue

        new_val = evaluate_c5(h_new)

        # Metropolis acceptance
        if new_val < curr_val or np.random.rand() < math.exp(-(new_val - curr_val) / max(T, 1e-12)):
            h_curr = h_new
            curr_val = new_val
            if new_val < best_val:
                best_h = h_new.copy()
                best_val = new_val
                # early exit if we already beat the target bound
                if best_val <= 0.3808:
                    break

        # occasional smoothing to explore different basins
        if it % 5000 == 0 and it > 0:
            kernel = np.array([0.25, 0.5, 0.25])
            smooth = np.convolve(h_curr, kernel, mode='same')
            smooth = project_to_constraints(smooth, target)
            smooth_val = evaluate_c5(smooth)
            if smooth_val < best_val:
                best_h = smooth.copy()
                best_val = smooth_val
                h_curr = smooth.copy()
                curr_val = smooth_val

        it += 1

    # final clean‑up projection and evaluation
    best_h = project_to_constraints(best_h, target)
    final_val = evaluate_c5(best_h)
    return best_h, final_val, n_points
