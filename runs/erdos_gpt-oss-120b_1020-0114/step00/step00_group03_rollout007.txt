def propose_neighbor(h, max_step=0.02):
    n = len(h)
    # choose source index i where h[i] > 0
    src_candidates = np.where(h > 1e-12)[0]
    dst_candidates = np.where(h < 1 - 1e-12)[0]
    if len(src_candidates) == 0 or len(dst_candidates) == 0:
        return h.copy() # no move
    i = np.random.choice(src_candidates)
    j = np.random.choice(dst_candidates)
    # avoid i == j
    while j == i:
        j = np.random.choice(dst_candidates)
    max_transfer = min(h[i], 1 - h[j], max_step)
    if max_transfer <= 0:
        return h.copy()
    delta = max_transfer * np.random.rand()
    # optionally also sometimes move negative direction (i gains, j loses)
    # To allow both directions, we can sometimes swap i/j.
    # We'll also optionally randomly decide direction.
    if np.random.rand() < 0.5:
        # transfer from i to j (i loses)
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
    else:
        # opposite direction but ensure feasible
        max_transfer2 = min(h[j], 1 - h[i], max_step)
        if max_transfer2 <= 0:

            new_h[j] += delta
        else:
            delta2 = max_transfer2 * np.random.rand()
            new_h = h.copy()
            new_h[i] += delta2

    return new_h

def propose_neighbor(h, max_step):
    n = len(h)
    # choose source and dest with possible transfer
    src_idx = np.random.choice(np.where(h > 0)[0])
    dst_idx = np.random.choice(np.where(h < 1)[0])
    if src_idx == dst_idx:
        # pick another dest
        # just return copy
        return h.copy()
    max_transfer = min(h[src_idx], 1 - h[dst_idx], max_step)
    if max_transfer <= 0:
        return h.copy()
    delta = np.random.rand() * max_transfer
    new_h = h.copy()
    new_h[src_idx] -= delta
    new_h[dst_idx] += delta
    # no sum change
    return new_h

def optimize_h(h_initial, time_budget):
    import numpy as np, time, math
    n = len(h_initial)
    dx = 2.0 / n
    target_sum = n / 2
    # ensure constraints
    h_current = enforce_sum_and_bounds(h_initial.copy())
    cur_score = compute_obj(h_current)
    best_h = h_current.copy()
    best_score = cur_score
    start = time.time()
    # store best
    # initial temperature parameters
    T0 = 0.02  # adjust
    max_step0 = 0.1
    # try hill climbing in steps
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        # temperature decreasing linearly
        T = T0 * max(0.0, 1 - elapsed / time_budget)
        # step size scaling
        step = max_step0 * max(0.1, T / T0)  # keep minimum 0.1 of max_step0
        # propose neighbor
        h_new = propose_neighbor(h_current, max_step=step)
        # compute objective
        new_score = compute_obj(h_new)
        # accept
        if new_score < cur_score:
            h_current = h_new
            cur_score = new_score
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score
        else:
            # accept with prob
            if T > 0 and np.random.rand() < math.exp(-(new_score - cur_score) / T):
                h_current = h_new
                cur_score = new_score
        # occasional random heal: with small prob try random perturbation
        if np.random.rand() < 0.001:
            # generate random point and project to sum constraint
            rand = np.random.rand(n)
            # scale to target sum
            rand = rand / np.mean(rand) * 0.5  # average = 0.5
            # clip
            rand = np.clip(rand, 0, 1)
            rand = enforce_sum_and_bounds(rand)
            rand_score = compute_obj(rand)
            h_current = rand
            cur_score = rand_score
            if rand_score < best_score:
                best_h = rand.copy()
                best_score = rand_score
    return best_h, best_score

def compute_obj(h):

def enforce_sum_and_bounds(h):
    n = len(h)
    target = n / 2
    # Clip first
    np.clip(h, 0, 1, out=h)
    # Compute diff
    diff = target - h.sum()
    # Use while loop to adjust diff, adjusting random indices with capacity.
    # We'll perform at most 100 tries.
    if abs(diff) < 1e-12:
        return h
    # We'll iterate up to some max attempts.
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to increase
            candidates = np.where(h < 1 - 1e-12)[0]
            if len(candidates) == 0:
                break
            i = np.random.choice(candidates)
            add = min(diff, 1 - h[i])
            h[i] += add
            diff -= add
        else:
            candidates = np.where(h > 1e-12)[0]
            if len(candidates) == 0:
                break
            i = np.random.choice(candidates)
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    # final correction using uniform distribution across all indices if still small diff (due to numerical).
    # We'll distribute residual diff equally to all indices, then clip again.
    if abs(diff) > 1e-8:
        # distribute proportionally
        # add diff/n to each entry
        h += diff / n
        np.clip(h, 0, 1, out=h)
        # Re-adjust diff iteratively maybe
        diff2 = target - h.sum()
        if abs(diff2) > 1e-9:
            # final fixed: adjust first element within bounds
            i0 = 0
            adjust = max(min(diff2, 1 - h[i0]), -h[i0])
            h[i0] += adjust
    return h

import numpy as np, time, math

def compute_obj(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def enforce_sum_and_bounds(h):
    n = h.shape[0]
    target = n / 2.0
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # try to fix via adjustments
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to increase
            cand = np.where(h < 1.0 - 1e-12)[0]
            if cand.size == 0:
                break
            i = np.random.choice(cand)
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        else:
            # need to decrease
            cand = np.where(h > 1e-12)[0]
            if cand.size == 0:
                break
            i = np.random.choice(cand)
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    # final correction if still small diff
    if abs(diff) > 1e-9:
        # distribute uniformly, then clip and correct again
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
        # one final adjustment
        final = target - h.sum()
        if abs(final) > 1e-12:
            # adjust first element within bounds
            i = 0
            adj = max(min(final, 1.0 - h[i]), -h[i])
            h[i] += adj
    return h

def gradient_of_f_at_shift(h, k):
    n = h.shape[0]
    g = np.ones(n, dtype=np.float64)
    if k >= 0:
        k = int(k)
        if k < n:
            g[:n - k] -= h[k:]
            g[k:] -= h[:n - k]
    else:
        kk = -k
        if kk < n:
            g[kk:] -= h[:n - kk]
            g[:n - kk] -= h[kk:]
    return g

def propose_neighbor(h, max_step):
    n = h.shape[0]
    # pick source with h>0, dest with h<1
    src_cand = np.where(h > 1e-12)[0]
    dst_cand = np.where(h < 1.0 - 1e-12)[0]
    if src_cand.size == 0 or dst_cand.size == 0:
        return h.copy()
    i = np.random.choice(src_cand)
    j = np.random.choice(dst_cand)
    if i == j:
        return h.copy()
    max_transfer = min(h[i], 1.0 - h[j], max_step)

        return h.copy()
    delta = max_transfer * np.random.rand()
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h

    avg = arr.mean()
    arr = arr * (0.5 / avg)

    n = h_start.shape[0]
    h = h_start.copy()
    cur_score = compute_obj(h)
    best_h = h.copy()

def _objective(h: np.ndarray) -> float:

    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _fix_sum_and_bounds(h: np.ndarray) -> np.ndarray:
    """Clamp to [0,1] and enforce Σh = n/2 exactly."""
    n = h.shape[0]
    target = n / 2.0
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # greedily fix the deficit / surplus
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:                     # need to add mass

            add = min(diff, 1.0 - h[i])

            diff -= add

                break

            sub = min(-diff, h[i])

            diff += sub

        if k < n:
            g[: n - k] -= h[k:]
            g[k:] -= h[: n - k]
    else:
        kk = -k
        if kk < n:
            g[kk:] -= h[: n - kk]
            g[: n - kk] -= h[kk:]
    return g


def _random_h(n: int) -> np.ndarray:
    """Generate a random feasible h."""
    a = np.random.rand(n)
    a = a / a.mean() * 0.5            # force average 0.5
    a = np.clip(a, 0.0, 1.0)
    return _fix_sum_and_bounds(a)


def _neighbor(h: np.ndarray, max_step: float) -> np.ndarray:
    """Small feasible perturbation preserving the total sum."""
    n = h.shape[0]
    src = np.where(h > 1e-12)[0]
    dst = np.where(h < 1.0 - 1e-12)[0]
    if src.size == 0 or dst.size == 0:
        return h.copy()
    i = np.random.choice(src)
    j = np.random.choice(dst)
    if i == j:
        return h.copy()
    mx = min(h[i], 1.0 - h[j], max_step)
    if mx <= 0:
        return h.copy()
    delta = mx * np.random.rand()
    nh = h.copy()
    nh[i] -= delta
    nh[j] += delta
    return nh


def _subgradient_optimize(h0: np.ndarray, budget: float) -> tuple[np.ndarray, float]:
    """Simple sub‑gradient descent with occasional random steps."""
    n = h0.shape[0]
    h = h0.copy()
    cur = _objective(h)
    best_h, best = h.copy(), cur
    start = time.time()
    atol = 1e-12

    while time.time() - start < budget:
        corr = np.correlate(h, 1.0 - h, mode="full")
        mx = np.max(corr)
        # indices of the active (worst) shifts
        act = np.where(corr >= mx - atol)[0]

        # aggregate gradient of the active constraints
        g = np.zeros(n, dtype=np.float64)
        for s in act:
            k = s - (n - 1)
            g += _grad_shift(h, k)

        # try a directed move: decrease where g>0, increase where g<0
        src = np.where((g > 1e-8) & (h > 1e-12))[0]
        dst = np.where((g < -1e-8) & (h < 1.0 - 1e-12))[0]
        moved = False
        if src.size and dst.size:
            i = np.random.choice(src)
            j = np.random.choice(dst)
            cap = min(h[i], 1.0 - h[j])
            if cap > 0:
                delta = cap * np.random.rand() * 0.5 + 1e-5
                nh = h.copy()
                nh[i] -= delta
                nh[j] += delta
                new = _objective(nh)
                if new <= cur:
                    h, cur = nh, new
                    moved = True
                    if new < best:
                        best_h, best = nh.copy(), new

        if not moved:
            # fallback random neighbour
            nh = _neighbor(h, max_step=0.05)
            new = _objective(nh)
            # accept if better, else very rarely accept
            if new < cur or np.random.rand() < 0.001:
                h, cur = nh, new
                if new < best:
                    best_h, best = nh.copy(), new

        # occasional full random restart to avoid stagnation
        if np.random.rand() < 0.0005:
            h = _random_h(n)
            cur = _objective(h)
            if cur < best:
                best_h, best = h.copy(), cur

    return best_h, best


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a step function h, its C5 bound, and the grid size."""
    np.random.seed(seed)
    total_time = budget_s * 0.99                     # tiny safety margin
    start = time.time()

    # ----- initialise -------------------------------------------------
    if "initial_h_values" in globals():
        base = np.array(initial_h_values, dtype=np.float64)
        n0 = base.shape[0]
        # work on a finer grid (at least 140 points)
        target_n = max(140, n0 * 2)
        if target_n != n0:
            old_x = np.arange(n0) * (2.0 / n0)
            new_x = np.arange(target_n) * (2.0 / target_n)
            base = np.interp(new_x, old_x, base, left=0.0, right=0.0)
        init = _fix_sum_and_bounds(base)
    else:
        target_n = 140
        x = np.arange(target_n) * (2.0 / target_n)
        init = (1.0 - np.cos(np.pi * x / 2.0)) / 2.0
        init = _fix_sum_and_bounds(init)

    # ----- main optimisation -----------------------------------------
    main_budget = total_time * 0.7
    best_h, best_val = _subgradient_optimize(init, main_budget)

    # ----- extra random restarts if time remains --------------------
    elapsed = time.time() - start
    remaining = total_time - elapsed
    if remaining > 5.0:
        per_restart = remaining / 3.0
        for _ in range(3):
            h0 = _random_h(best_h.shape[0])
            h_opt, val = _subgradient_optimize(h0, per_restart)
            if val < best_val:
                best_h, best_val = h_opt, val

    # final safety clamp
    best_h = _fix_sum_and_bounds(best_h)
    best_val = _objective(best_h)

    return (best_h.tolist(), float(best_val), best_h.shape[0])
