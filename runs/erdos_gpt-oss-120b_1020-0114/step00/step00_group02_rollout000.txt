def project_onto_capped_simplex(v, c, max_iter=100, tol=1e-9):
    lo = -np.max(v) - 1  # ensure v - lo large positive => all 1s
    hi = np.max(v) + 1   # ensures v - hi negative => all 0s
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0, 1)
        s = x.sum()
        if abs(s - c) < tol:
            return x
        if s > c:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2.0, 0, 1)

def grad_corr_shift(h, shift):
    n = h.shape[0]
    if shift >= 0:
        k = shift
        grad = np.zeros_like(h)
        if n - k > 0:
            grad[:n - k] += 1

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the intersection of the hyperplane sum = c and the box [0,1]
def project_capped_simplex(v: np.ndarray, c: float) -> np.ndarray:
    """
    Solve   min_x ||x - v||_2   s.t.   0 <= x_i <= 1 ,  sum(x) = c .
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo, hi = -1e6, 1e6  # wide enough bounds
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > c:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


# ----------------------------------------------------------------------
# Helper: gradient of the correlation for a given shift (integer lag)
def grad_corr_shift(h: np.ndarray, shift: int) -> np.ndarray:
    n = h.shape[0]
    if shift >= 0:
        k = shift
        if k >= n:
            return np.zeros_like(h)
        grad = np.zeros_like(h)
        # +1 term for the linear part of the correlation
        grad[: n - k] += 1.0
        # -h[i+k] term
        grad[: n - k] -= h[k:]
        # -h[i-k] term
        grad[k:] -= h[: n - k]
        return grad
    else:
        p = -shift
        if p >= n:
            return np.zeros_like(h)
        grad = np.zeros_like(h)
        # +1 term for i >= p
        grad[p:] += 1.0
        # -h[i-p] term (i is the index where h_i appears)
        grad[p:] -= h[: n - p]
        # -h[i+p] term (i is the index of the partner)
        grad[: n - p] -= h[p:]
        return grad


# ----------------------------------------------------------------------
# Helper: compute C5 value (max overlap) for a given h
def compute_c5(h: np.ndarray) -> float:
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx

    m = old.shape[0]
    x_old = np.linspace(0.0, 2.0, m, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_len, endpoint=False)
    return np.interp(x_new, x_old, old)

    n = max(60, base_n * 2)

    sum_target = n / 2.0
    dx = 2.0 / n

    best_h = None

    # Number of independent runs we will try (adjusted by remaining time)
    restart = 0
    while time.time() - start < budget_s - 5.0:
        restart += 1

        # ---- initial point -------------------------------------------------
        if base_h is not None and restart == 1:

        else:

            v = np.random.rand(n)

        # ---- sub‑gradient descent -----------------------------------------
