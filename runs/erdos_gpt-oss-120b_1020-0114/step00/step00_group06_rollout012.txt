def run(seed: int = 42, budget_s: float = 1100, **kwargs):

    Heuristic search for a step function h on [0,2] with integral 1

    Returns (h_values, c5_bound, n_points).
    """

    # Choose a problem size (must be even for an exact binary half‑density)
    # ----------------------------------------------------------------------

    h = np.zeros(n, dtype=float)

    # ----------------------------------------------------------------------
    # Helper to evaluate the objective
    # ----------------------------------------------------------------------
    def compute_c5(vec: np.ndarray) -> float:
        """return C5 for a given vector (dx is captured from outer scope)."""
        corr = np.correlate(vec, 1.0 - vec, mode="full")
        return float(np.max(corr) * dx)

    # keep the best solution seen so far
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    cur_h = h.copy()
    cur_c5 = best_c5

    # Main optimisation loop (gradient‑guided simulated annealing)
    # ----------------------------------------------------------------------
    max_no_improve = 2000          # restart after this many non‑improving steps
    no_improve = 0
    iter_count = 0

    while time.time() - start < budget_s - 0.5:
        # correlation and worst lag (the one that attains the max)
        full_corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        max_idx = int(np.argmax(full_corr))
        lag = max_idx - (n - 1)                     # lag can be negative
