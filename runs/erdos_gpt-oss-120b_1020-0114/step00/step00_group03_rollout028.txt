def project_feasible(h, target_sum):
    # clip within bounds
    h = np.clip(h, 0, 1)
    for _ in range(10):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-9:
            break
        # Determine indices where we can increase (if diff>0) or decrease (if diff<0)
        if diff > 0:
            mask = h < 1 - 1e-12
        else:
            mask = h > 1e-12
        num = mask.sum()
        if num == 0:
            break
        adjust = diff / num
        # limit adjustment to avoid crossing bounds
        if diff > 0:
            adjust = np.minimum(adjust, (1 - h[mask]).min())
        else:
            adjust = np.maximum(adjust, -h[mask].min())
        h[mask] += adjust
        h = np.clip(h, 0, 1)
    return h

import numpy as np, time, math

def evaluate(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(corr.max() * dx)

def make_feasible(h):
    # Clip and adjust to sum = n/2
    n = h.size
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Simple projection using scaling then clipping iterative
    total = h.sum()
    # If total != target, adjust.
    # Use water-filling approach:
    diff = target - total
    # If diff > 0 we need to increase some values not at 1.
    # if diff < 0 we need to decrease some values not at 0.
    # We'll distribute diff over eligible indices uniformly.
    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        # amount to change per selected index
        # cap per index to avoid crossing bounds
        cap = (1.0 - h[mask]) if diff > 0 else h[mask]
        # average change per index
        avg = diff / np.count_nonzero(mask)
        # clip avg to max possible change per index
        change = np.clip(avg, -np.max(cap), np.max(cap))
        h[mask] += change
        h = np.clip(h, 0.0, 1.0)
        total = h.sum()
        diff = target - total
    # Ensure final sum close to target
    # minor correction to enforce exact sum: adjust last element
    if abs(h.sum() - target) > 1e-8:
        # adjust first element within bounds
        diff = target - h.sum()
        # find index that can accommodate diff
        for i in range(n):
            new = h[i] + diff
            if 0 <= new <= 1:
                h[i] = new
                break
    return h

def hill_climb_binary(n, time_budget):
    # initialize with upscaled initial_h:
    if 'initial_h_values' in globals():
        # use initial to create binary by threshold median
        init = initial_h_values
        init_n = len(init)
        x = np.linspace(0,2,n,endpoint=False)
        x0 = np.linspace(0,2,init_n,endpoint=False)
        h_cont = np.interp(x, x0, init)  # interpolation
        # threshold at >0.5 for binary
        binary_h = np.where(h_cont > 0.5, 1.0, 0.0)
        # adjust count: if not exactly n/2 ones, random adjust
        ones = binary_h.sum()
        target = n // 2
        # while ones != target:
        while ones > target:
            idx = np.where(binary_h == 1)[0]
            # choose random 1 and turn to 0
            i = np.random.choice(idx)
            binary_h[i] = 0
            ones -= 1
        while ones < target:
            idx = np.where(binary_h == 0)[0]
            i = np.random.choice(idx)
            binary_h[i] = 1
            ones += 1
        cur_h = binary_h

        cur_h = np.zeros(n)
        cur_h[:n//2] = 1.0
        np.random.shuffle(cur_h)

    best_val = evaluate(best_h)
    cur_val = best_val

def refine_continuous(h, time_budget):
    start = time.time()
    cur = h.copy()
    cur_val = evaluate(cur)
    best = cur.copy()
    best_val = cur_val
    n = len(h)
    target_sum = n/2
    while time.time() - start < time_budget:
        i, j = np.random.randint(0, n, 2, endpoint=False)
        if i == j:
            continue
        # ensure we can transfer
        max_transfer = min(cur[i], 1 - cur[j])
        if max_transfer <= 1e-12:
            continue
        delta = np.random.rand() * max_transfer
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new_val = evaluate(new)
        if new_val < cur_val:
            cur, cur_val = new, new_val
            if new_val < best_val:
                best, best_val = new.copy(), new_val
        else:
            # accept with some probability
            # temperature decreasing
            elapsed = time.time() - start
            T = 0.01 * max(0., 1 - elapsed / time_budget)
            if np.random.rand() < math.exp(-(new_val - cur_val)/T):
                cur, cur_val = new, new_val
    return best, best_val
```

import numpy as np, time, math, random

def _eval_c5(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1‑h(x+k)) dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1‑h); mode='full' gives all integer shifts
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)


def _piecewise_search(n: int, max_seconds: float):
    """
    Search symmetric two‑level step functions:
        a on the left/right edge of total length 2*l,
        b on the middle part.
    Returns the best (h, value) found.
    """
    start = time.time()
    best_h = None
    best_val = 1e9
    target_sum = n / 2.0
    dx = 2.0 / n

    # candidate grid
    l_vals = np.linspace(0.0, 0.45, 91)          # edge length (total from each side)
    a_vals = np.linspace(0.0, 0.45, 91)          # edge height
    for l in l_vals:
        # number of points on one side
        n_edge = int(round(l / dx))
        if 2 * n_edge >= n:
            continue
        n_mid = n - 2 * n_edge
        for a in a_vals:
            # compute middle height b so that total integral = 1
            b = (target_sum - 2 * n_edge * a) / n_mid
            if b < 0.0 or b > 1.0:
                continue
            h = np.full(n, a, dtype=np.float64)
            h[n_edge:n_edge + n_mid] = b
            # (the construction guarantees the sum exactly)
            val = _eval_c5(h)
            if val < best_val:
                best_val = val
                best_h = h.copy()
            # time guard
            if time.time() - start > max_seconds * 0.9:
                return best_h, best_val
    return best_h, best_val


def _continuous_hill_climb(h0: np.ndarray, max_seconds: float):
    """
    Simple donor–receiver hill‑climbing with simulated‑annealing acceptance.
    Keeps the sum exactly constant by moving mass between two coordinates.
    """
    n = h0.size
    target_sum = n / 2.0
    cur = h0.copy()
    cur_val = _eval_c5(cur)
    best = cur.copy()
    best_val = cur_val
    start = time.time()
    # pre‑compute index lists for speed
    idx = np.arange(n, dtype=np.int32)

    while time.time() - start < max_seconds:
        # pick donor (i) with h[i] > 0 and receiver (j) with h[j] < 1
        i = np.random.choice(idx)
        if cur[i] <= 1e-12:
            continue
        j = np.random.choice(idx)
        if i == j or cur[j] >= 1.0 - 1e-12:
            continue
        delta_max = min(cur[i], 1.0 - cur[j])
        if delta_max <= 0.0:
            continue
        # random transfer (up to full allowed amount)
        delta = delta_max * np.random.rand()
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new_val = _eval_c5(new)
        if new_val < cur_val:
            cur, cur_val = new, new_val
            if new_val < best_val:
                best, best_val = new.copy(), new_val
        else:
            # annealing acceptance
            elapsed = time.time() - start
            T = max(0.001, 0.02 * (1.0 - elapsed / max_seconds))
            if np.random.rand() < math.exp(-(new_val - cur_val) / T):
                cur, cur_val = new, new_val
    return best, best_val


def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points) where:
    - h_values : list of float values in [0,1] (length = n_points)
    - c5_bound : the achieved C5 value (float)
    - n_points : the grid size used
    The routine respects the given time budget.
    """
    np.random.seed(seed)
    random.seed(seed)

    # grid size – balance resolution and speed
    n = 200

    # allocate time: 30 % for coarse piecewise search, rest for refinement
    t0 = time.time()
    search_time = budget_s * 0.3
    refine_time = budget_s - search_time - 0.5  # leave a margin

    # 1) coarse search over two‑level symmetric step functions
    best_h, best_val = _piecewise_search(n, search_time)
    if best_h is None:
        # fallback: simple half‑ones vector
        best_h = np.zeros(n, dtype=np.float64)
        best_h[: n // 2] = 1.0
        np.random.shuffle(best_h)
        best_val = _eval_c5(best_h)

    # 2) continuous hill‑climbing starting from the best piecewise candidate
    best_h, best_val = _continuous_hill_climb(best_h, refine_time)

    # final sanity check – enforce constraints exactly
    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = best_h.sum()
    diff = n / 2.0 - cur_sum
    if abs(diff) > 1e-10:
        # adjust the first non‑saturated entry to fix the sum
        for i in range(n):
            cand = best_h[i] + diff
            if 0.0 <= cand <= 1.0:
                best_h[i] = cand
                break
    # return as plain Python list for the harness
    return (best_h.tolist(), float(best_val), n)
