def project_to_feasible(x, target_sum):
    x = np.clip(x, 0.0, 1.0)
    # waterfilling algorithm: we find λ such that sum_i min(max(x_i - λ, 0), 1) = target_sum

            hi = mid

def compute_gradient(h, shift):
    if shift >= 0:
        left = np.concatenate([h[shift:], np.zeros(shift)])
        right = np.concatenate([np.zeros(shift), h[:len(h) - shift]])
    else:
        s = -shift
        left = np.concatenate([np.zeros(s), h[:len(h)-s]])
        right = np.concatenate([h[s:], np.zeros(s)])
    return left + right
```

import time
import numpy as np
import math

def _project_to_bounds_and_sum(v, target):
    """Project vector v onto the box [0,1]^n intersect {sum = target}."""
    # Find λ such that sum_i clip(v_i - λ,0,1) = target.
    lo = -np.max(v) - 2.0
    hi = np.max(v) + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def _c5_of_h(h, dx):
    """Exact evaluation used for final reporting."""
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def _worst_shift_and_gradient(h, shift):
    """Compute subgradient of the worst shift term."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        # term (1 - h_{i+shift}) for i = 0 .. n-1-shift
        i_end = n - shift
        g[:i_end] += 1.0 - h[shift:]
        # term -h_{i-shift} for i = shift .. n-1
        g[shift:] -= h[:n - shift]
    else:
        s = -shift
        # term (1 - h_{i-s}) for i = s .. n-1
        g[s:] += 1.0 - h[:n - s]
        # term -h_{i+s} for i = 0 .. n-1-s
        g[:n - s] -= h[s:]
    return g

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    # target number of samples (chosen to be moderate for speed)
    n = 200
    dx = 2.0 / n
    target_sum = n / 2.0

    # initialise from parent if available, otherwise random
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        # linear interpolation to the working resolution
        x_old = np.linspace(0.0, 2.0, init.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x_new, x_old, init)
    else:
        h = rng.random(n)

    # enforce feasibility
    h = _project_to_bounds_and_sum(h, target_sum)

    # current best solution
    best_h = h.copy()
    best_c = _c5_of_h(best_h, dx)

    # optimisation parameters
    start_time = time.time()
    step = 0.3               # initial step size
    min_step = 2e-5
    decay = 0.9995
    no_improve = 0

    # main loop
    while True:
        now = time.time()
        if now - start_time > budget_s - 0.2:
            break

        # evaluate current worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)

        # compute subgradient of the worst term
        grad = _worst_shift_and_gradient(h, shift)

        # gradient descent step (move opposite to subgradient)
        y = h - step * grad
        h_new = _project_to_bounds_and_sum(y, target_sum)

        # evaluate new candidate
        c_new = _c5_of_h(h_new, dx)

        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # accept only if not worsening
        if c_new < _c5_of_h(h, dx):
            h = h_new

        # step scheduling
        step = max(min_step, step * decay)

        # occasional random perturbation to escape stagnation
        if no_improve >= 4000:
            # perturb around the best known point
            h = best_h + rng.normal(scale=0.02, size=n)
            h = np.clip(h, 0.0, 1.0)
            h = _project_to_bounds_and_sum(h, target_sum)
            no_improve = 0

    # final evaluation (ensure consistency)
    final_c = _c5_of_h(best_h, dx)
    return (list(best_h), float(final_c), n)
