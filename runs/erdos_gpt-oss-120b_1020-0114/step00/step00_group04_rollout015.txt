import numpy as np, time, math, random

def _c5_of(h, dx):
    """max_k ∫ h(x)*(1-h(x+k)) dx approximated by discrete correlation."""
    # correlation h with (1-h) using zero padding (np.correlate 'full')
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def _make_shape(n, p):
    """
    Piecewise linear shape with left/right ramps of length p (0<p≤1)
    and a central plateau of height h_max = 1/(2-p).  Returns an
    array of length n with integral exactly 1 (up to floating error).
    """
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx            # mid‑point sample points
    h_max = 1.0 / (2.0 - p)                  # ensures total area =1
    # left ramp
    left = x < p
    # right ramp
    right = x > 2.0 - p
    # plateau
    mid = ~left & ~right
    h = np.empty(n, dtype=np.float64)
    h[left] = h_max * x[left] / p
    h[right] = h_max * (2.0 - x[right]) / p
    h[mid] = h_max
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Find a step function h on [0,2] with ∫h=1 that minimises C5."""
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    # number of sample points – keep it below 1000 for speed
    n = 600
    dx = 2.0 / n

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    best_h = None

        c5 = _c5_of(init, dx)
        best_h, best_c5 = init, c5
