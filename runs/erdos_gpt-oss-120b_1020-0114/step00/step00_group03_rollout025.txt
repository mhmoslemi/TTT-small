def simulated_annealing(h, max_iters, time_limit):
   n = len(h)
   cur = h.copy()
   cur_val = evaluate_c5(cur)
   best = cur.copy()
   best_val = cur_val
   start = time.time()
   T0 = 0.01 # initial temperature maybe 0.01
   T_end = 1e-6
   # schedule exponential based on iteration fraction
   for iter in range(max_iters):
        if time.time() - start > time_limit:
            break
        # compute temperature
        t = iter / max_iters
        T = T0 * (1 - t) + T_end * t  # linear decay
        # propose move
        i, j = np.random.randint(0, n, size=2)
        while i == j:
            j = np.random.randint(0, n)
        # compute allowed delta
        lo = max(-cur[i], cur[j] - 1, -0.2)  # limit step size to 0.2 maybe bigger
        hi = min(1 - cur[i], cur[j], 0.2)
        if lo >= hi:
            continue
        delta = np.random.uniform(lo, hi)
        new = cur.copy()
        new[i] += delta
        new[j] -= delta
        # evaluate
        new_val = evaluate_c5(new)
        # accept?
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best = new.copy()
                best_val = new_val
   return best, best_val

def derivative_at_lag(h, L):
    n = h.shape[0]
    D = np.zeros(n)
    if L >= 0:
        # indicator for t >= L
        D[L:] = 1.0
        # subtract h[t-L] for t >= L
        D[L:] -= h[:n - L]
        # subtract h[t+L] for t+L < n
        D[:n - L] -= h[L:]
    else:
        M = -L
        # indicator for t <= n-1-M (i.e., t < n-M)
        D[:n - M] = 1.0
        # subtract h[t+M] for t <= n-M-1

        # subtract h[t-M] for t >= M
        D[M:] -= h[:n - M]

import numpy as np
import time
import math

def evaluate_c5(arr):
    """Return the C5 value for a discretized function."""
    n = arr.shape[0]
    dx = 2.0 / n
    corr = np.correlate(arr, 1.0 - arr, mode="full")
    return np.max(corr) * dx

def derivative_at_lag(h, L):
    """Derivative of the max‑lag correlation w.r.t. h (sign only)."""
    n = h.shape[0]
    D = np.zeros(n)
    if L >= 0:
        # indicator 1 for t >= L
        D[L:] = 1.0
        # subtract h[t-L] where defined
        D[L:] -= h[: n - L]
        # subtract h[t+L] where defined
        D[: n - L] -= h[L:]
    else:
        M = -L
        # indicator 1 for t <= n-1-M
        D[: n - M] = 1.0
        # subtract h[t+M] where defined
        D[: n - M] -= h[M:]
        # subtract h[t-M] where defined
        D[M:] -= h[: n - M]
    return D

def enforce_sum(arr):
    """Clamp to [0,1] and enforce sum(arr) == len(arr)/2."""
    n = arr.shape[0]
    target = n / 2.0
    np.clip(arr, 0.0, 1.0, out=arr)
    diff = target - arr.sum()
    # simple iterative correction
    while abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(arr < 1 - 1e-12)[0]
            if idx.size == 0:
                break
            i = np.random.choice(idx)
            inc = min(diff, 1.0 - arr[i])
            arr[i] += inc
        else:
            idx = np.where(arr > 1e-12)[0]
            if idx.size == 0:
                break
            i = np.random.choice(idx)
            dec = min(-diff, arr[i])
            arr[i] -= dec
        diff = target - arr.sum()
    np.clip(arr, 0.0, 1.0, out=arr)

def run(seed=42, budget_s=1100, **kwargs):
    """Return a feasible step function beating the current record."""
    np.random.seed(seed)
    start = time.time()

    # ----- initialise ---------------------------------------------------------
    init = globals().get("initial_h_values")
    if init is not None:
        base = np.asarray(init, dtype=float)
        n0 = base.shape[0]
        factor = 2                     # up‑sample to gain flexibility
        n = n0 * factor
        x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x_new, x_old, base)
    else:
        n = 100
        h = np.zeros(n)
        idx = np.random.choice(n, n // 2, replace=False)
        h[idx] = 1.0

    enforce_sum(h)

    best_h = h.copy()
    best_val = evaluate_c5(best_h)

    # ----- phase 1 : deterministic descent ------------------------------------
    cur_h = h.copy()
    cur_val = best_val
    step = 0.5
    eps = 1e-12
    phase1_deadline = start + budget_s * 0.5   # half of the budget

    while time.time() < phase1_deadline:
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        max_idx = int(np.argmax(corr))
        max_corr = float(corr[max_idx])
        cur_val = max_corr * (2.0 / n)
        L = max_idx - (n - 1)

        D = derivative_at_lag(cur_h, L)

        high_mask = (D > eps) & (cur_h > eps)
        low_mask = (D < -eps) & (cur_h < 1.0 - eps)

        if not high_mask.any() or not low_mask.any():
            step *= 0.8
            if step < 1e-6:
                break
            continue

        high_candidates = np.where(high_mask)[0]
        high_idx = high_candidates[np.argmax(D[high_candidates])]
        low_candidates = np.where(low_mask)[0]
        low_idx = low_candidates[np.argmin(D[low_candidates])]

        delta = min(step, cur_h[high_idx], 1.0 - cur_h[low_idx])
        if delta <= eps:
            step *= 0.5
            continue

        new_h = cur_h.copy()
        new_h[high_idx] -= delta
        new_h[low_idx] += delta
        new_val = evaluate_c5(new_h)

        if new_val < cur_val - 1e-12:
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_h = new_h.copy()
                best_val = new_val
        else:
            step *= 0.7
            if step < 1e-6:
                break

    # ----- phase 2 : simulated annealing --------------------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 1.0:
        cur_h = best_h.copy()
        cur_val = best_val
        max_iters = int(5e6)
        T0 = 0.005
        T_end = 1e-6
        max_step_initial = 0.5
        it = 0
        while time.time() - start < budget_s:
            frac = it / max_iters
            T = T0 * (1.0 - frac) + T_end * frac
            max_step = max_step_initial * (1.0 - frac)

            i = np.random.randint(0, n)
            j = np.random.randint(0, n - 1)
            if j >= i:
                j += 1

            lo = max(-cur_h[i], cur_h[j] - 1.0, -max_step)
            hi = min(1.0 - cur_h[i], cur_h[j], max_step)
            if lo >= hi:
                it += 1
                continue

            delta = np.random.uniform(lo, hi)
            new_h = cur_h.copy()
            new_h[i] += delta
            new_h[j] -= delta
            new_val = evaluate_c5(new_h)

            if new_val < cur_val or np.random.rand() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
                cur_h = new_h
                cur_val = new_val
                if new_val < best_val:
                    best_h = new_h.copy()
                    best_val = new_val
            it += 1

    # ----- final sanity check -------------------------------------------------
    enforce_sum(best_h)
    best_val = evaluate_c5(best_h)

    return best_h, best_val, n
