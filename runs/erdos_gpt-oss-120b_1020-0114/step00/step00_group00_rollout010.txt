def compute_c5(h):
    c = np.correlate(h, 1 - h, mode='full')
    return np.max(c) * (2.0 / len(h))

def random_swap(h):
    n = len(h)
    i, j = np.random.choice(n, 2, replace=False)
    # swap values
    h[i], h[j] = h[j], h[i]

def simulated_anneal(n, time_remaining):
    # start random binary vector with half ones
    h = np.zeros(n)
    h[:n//2] = 1.
    np.random.shuffle(h)

    best_h = h.copy()
    best_val = compute_c5(h)

    # Determine number of iterations based on time budget
    start = time.time()
    iter = 0
    while time.time() - start < time_remaining:
        i, j = np.random.randint(0, n, size=2)
        if i == j or h[i] == h[j]:
            # swapping identical values will not change objective; skip
            continue
        # compute candidate
        # apply swap
        h[i], h[j] = h[j], h[i]
        val = compute_c5(h)
        if val <= best_val:
            best_val = val
            best_h = h.copy()
        else:
            # undo swap with probability p (annealing)
            # we could accept with some probability depending on temperature
            # For simplicity, revert always
            h[i], h[j] = h[j], h[i]
        iter += 1
        # could gradually adjust acceptance.
    return best_h, best_val, iter

            current_h[j] -= delta

def anneal(h_start, time_limit):
    import numpy as np, math, random, time
    h = h_start.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_val = compute_c5(h)
    cur_val = best_val
    start = time.time()
    T0 = 0.05  # initial temperature
    # Determine max iterations: unlimited; loop until time.
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        # Temperature schedule:
        T = T0 * (1.0 - elapsed / time_limit)
        # choose two coordinates i,j
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # propose moving a mass delta from j to i (or opposite)
        direction = random.random() < 0.5
        if direction:
            inc_idx, dec_idx = i, j
        else:
            inc_idx, dec_idx = j, i
        # compute allowed delta
        max_inc = 1.0 - h[inc_idx]
        max_dec = h[dec_idx]
        max_delta = min(max_inc, max_dec)
        if max_delta <= 0:
            continue
        # draw delta
        delta = random.random() * max_delta
        # apply
        h[inc_idx] += delta
        h[dec_idx] -= delta
        new_val = compute_c5(h)
        diff = new_val - cur_val
        if diff < 0 or random.random() < math.exp(-diff / (T + 1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert
            h[inc_idx] -= delta
            h[dec_idx] += delta
    return best_h, best_val

import time
import math
import random
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return C5 for a given step function h."""
    n = h.shape[0]
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def _compute_c5_and_shift(h: np.ndarray):
    """Return (C5, shift) where shift is the lag attaining the max."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    val = float(corr[idx] * (2.0 / n))
    shift = idx - (n - 1)          # zero shift corresponds to index n-1
    return val, shift

def _random_balanced_vector(n: int) -> np.ndarray:
    """
    Produce a feasible vector:
      * 0 <= h[i] <= 1
      * sum(h) = n/2
      * roughly half of the entries are 1 (binary), with a single 0.5 if n is odd.
    """
    h = np.zeros(n, dtype=np.float64)
    half = n // 2
    if n % 2 == 0:
        h[:half] = 1.0
    else:
        h[:half] = 1.0
        h[half] = 0.5
    np.random.shuffle(h)
    return h

def _targeted_swap(h: np.ndarray, shift: int) -> tuple:
    """
    Propose a swap that tries to lower the contribution at the worst shift.
    Returns (i_inc, i_dec, delta) where mass delta is moved from i_dec to i_inc,
    or None if no sensible pair exists.
    """
    n = h.shape[0]
    idx = np.arange(n)
    j = idx - shift                      # partner index for each i (may be out of range)

    in_range = (j >= 0) & (j < n)
    # complement = 1 - h[j] where defined, otherwise 1 (because h=0 outside)
    complement = np.where(in_range, 1.0 - h[j], 1.0)

    # contribution of each position to the current worst‑shift correlation
    contrib = h * complement

    # positions that currently contribute (bad) and can give up some mass
    bad = np.where((contrib > 1e-12) & (h > 1e-12))[0]
    # positions that do not contribute (good) and can receive mass
    good = np.where((contrib < 1e-12) & (h < 1.0 - 1e-12))[0]

    if bad.size == 0 or good.size == 0:
        return None

    i_dec = int(random.choice(bad))
    i_inc = int(random.choice(good))

    # maximal transfer respecting the box constraints
    max_dec = h[i_dec]
    max_inc = 1.0 - h[i_inc]
    delta = min(max_dec, max_inc)
    if delta <= 0:
        return None
    return i_inc, i_dec, delta

def _anneal(h_start: np.ndarray, time_limit: float) -> tuple:
    """
    Simple simulated‑annealing / hill‑climbing on a single vector.
    Returns the best vector seen and its C5 value.
    """
    h = h_start.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_val = _compute_c5(h)

    cur_val, cur_shift = _compute_c5_and_shift(h)

    start = time.time()
    T0 = 0.05          # initial temperature scale

    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break

        # linearly decreasing temperature
        T = T0 * (1.0 - elapsed / time_limit)

        # occasionally try a targeted move that directly attacks the worst shift
        if random.random() < 0.15:
            proposal = _targeted_swap(h, cur_shift)
            if proposal is not None:
                i_inc, i_dec, delta = proposal
                h[i_inc] += delta
                h[i_dec] -= delta
                new_val, new_shift = _compute_c5_and_shift(h)
                diff = new_val - cur_val
                accept = diff < 0 or random.random() < math.exp(-diff / (T + 1e-12))
                if accept:
                    cur_val, cur_shift = new_val, new_shift
                    if new_val < best_val:
                        best_val = new_val
                        best_h = h.copy()
                else:
                    # revert
                    h[i_inc] -= delta
                    h[i_dec] += delta
                continue   # go to next iteration

        # generic random delta move between two coordinates
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # decide direction: mass from dec -> inc
        if random.random() < 0.5:
            inc, dec = i, j
        else:
            inc, dec = j, i

        max_inc = 1.0 - h[inc]
        max_dec = h[dec]
        max_delta = min(max_inc, max_dec)
        if max_delta <= 0:
            continue

        delta = random.random() * max_delta
        h[inc] += delta
        h[dec] -= delta

        new_val, new_shift = _compute_c5_and_shift(h)
        diff = new_val - cur_val
        accept = diff < 0 or random.random() < math.exp(-diff / (T + 1e-12))
        if accept:
            cur_val, cur_shift = new_val, new_shift
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert
            h[inc] -= delta
            h[dec] += delta

    return best_h, best_val

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize the Erdős minimum overlap step function.
    Returns (h_values, c5_bound, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)

    start_total = time.time()
    # Determine base length
    if 'initial_h_values' in globals():
        init_vec = np.array(initial_h_values, dtype=np.float64)
        n_base = init_vec.shape[0]
    else:
        init_vec = None
        n_base = 45

    # Candidate lengths (larger gives more flexibility)
    candidates = [n_base, 80, 120, 200]
    # ensure uniqueness and reasonable bounds
    candidates = [c for c in sorted(set(candidates)) if c >= 20 and c <= 400]

    best_overall_h = None
    best_overall_val = math.inf
    best_n = None

    # Simple time budgeting: try each candidate in order, using remaining budget
    for n in candidates:
        elapsed = time.time() - start_total
        remaining = budget_s - elapsed
        if remaining <= 5.0:               # keep a little slack for final steps
            break

        # Build an initial feasible vector for this n
        if init_vec is not None and n == n_base:
            h0 = init_vec.copy()
        else:
            h0 = _random_balanced_vector(n)

        # Allocate most of the remaining time to the first (largest) candidate,
        # and a smaller share to subsequent ones.
        # Simple heuristic: 70% of remaining for the first, split remainder equally.
        if n == candidates[0]:
            alloc = remaining * 0.70
        else:
            alloc = (remaining * 0.30) / (len(candidates) - 1)

        # Run several restarts within the allocated slice
        restarts = max(1, int(alloc // 15.0))   # roughly one restart per 15 s
        for _ in range(restarts):
            # Give each restart a fair slice of time
            sub_alloc = alloc / restarts
            h_opt, val_opt = _anneal(h0, sub_alloc)
            if val_opt < best_overall_val:
                best_overall_val = val_opt
                best_overall_h = h_opt.copy()
                best_n = n
            # early exit if we already beat the target
            if best_overall_val <= 0.3808:
                break
        if best_overall_val <= 0.3808:
            break

    # If for any reason we have no solution, fall back to a uniform feasible vector
    if best_overall_h is None:
        best_n = n_base
        best_overall_h = _random_balanced_vector(best_n)
        best_overall_val = _compute_c5(best_overall_h)

    # Final tiny projection to guarantee exact sum (avoid tiny floating error)
    target_sum = best_n / 2.0
    cur_sum = best_overall_h.sum()
    diff = target_sum - cur_sum
    if abs(diff) > 1e-12:
        # distribute the missing mass to a random position that can absorb it
        for _ in range(100):
            idx = random.randint(0, best_n - 1)
            avail = (1.0 - best_overall_h[idx]) if diff > 0 else best_overall_h[idx]
            take = min(abs(diff), avail)
            if take > 0:
                best_overall_h[idx] += take if diff > 0 else -take
                diff += -take if diff > 0 else take
                if abs(diff) < 1e-12:
                    break

    return (best_overall_h, best_overall_val, int(best_n))
