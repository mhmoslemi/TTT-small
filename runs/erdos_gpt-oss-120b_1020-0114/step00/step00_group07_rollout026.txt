def simulated_annealing(h0, time_limit):
    n = len(h0)
    dx = 2.0 / n
    rng = np.random.default_rng(42)
    h = h0.copy()
    # ensure binary list of ones and zeros indices
    ones_idx = np.where(h == 1)[0].tolist()
    zeros_idx = np.where(h == 0)[0].tolist()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    start = time.time()
    max_iter = 2000000   # large limit
    for it in range(max_iter):
        elapsed = time.time() - start
        if elapsed > time_limit * 0.95:
            break
        # temperature schedule linear decreasing from 0.1 to 0
        T = 0.1 * (1 - it / max_iter)
        # propose swap
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        # swap
        h[i], h[j] = h[j], h[i]
        # update indices
        ones_idx.remove(i); zeros_idx.append(i)
        zeros_idx.remove(j); ones_idx.append(j)
        # compute c5
        cand_c5 = compute_c5(h)
        # acceptance
        if cand_c5 < cur_c5 or rng.random() < math.exp(-(cand_c5 - cur_c5) / (T + 1e-12)):
            cur_c5 = cand_c5
            # keep update lists as is (already swapped)
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]
            # revert indices
            ones_idx.remove(j); zeros_idx.append(j)
            zeros_idx.remove(i); ones_idx.append(i)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
            # early exit if below target
            if best_c5 <= 0.38085:
                break
    return best_h, best_c5

import numpy as np, time, math

def compute_c5(values):
    n = values.shape[0]
    corr = np.correlate(values, 1.0 - values, mode='full')
    return corr.max() * (2.0 / n)

def binary_from_initial(arr, rng):
    n = len(arr)
    # Clip to [0,1]
    arr = np.clip(arr, 0.0, 1.0)
    # Simple threshold
    bin_arr = (arr >= 0.5).astype(np.float64)
    target = n // 2
    cur = int(bin_arr.sum())
    if cur < target:
        zero_idx = np.where(bin_arr == 0)[0]
        rng.shuffle(zero_idx)
        bin_arr[zero_idx[:target - cur]] = 1.0
    elif cur > target:
        one_idx = np.where(bin_arr == 1)[0]
        rng.shuffle(one_idx)
        bin_arr[one_idx[:cur - target]] = 0.0
    return bin_arr

def simulated_annealing(h_start, budget, rng):
    n = len(h_start)
    h = h_start.copy()
    current = compute_c5(h)
    best = current
    best_h = h.copy()
    # maintain sets of indices for ones and zeros
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    start_time = time.time()
    max_iters = 2000000
    for it in range(max_iters):
        elapsed = time.time() - start_time
        if elapsed > budget * 0.95:
            break
        # temperature schedule
        T = 0.2 * (1.0 - it / max_iters)  # start maybe 0.2
        # propose swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        h[i], h[j] = h[j], h[i]
        # update index lists
        ones.remove(i); zeros.remove(j)
        ones.append(j); zeros.append(i)
        cand = compute_c5(h)
        if cand < current or rng.random() < math.exp(-(cand - current) / (T + 1e-12)):
            current = cand
            # keep the swap
        else:
            # revert
            h[i], h[j] = h[j], h[i]
            ones.remove(j); zeros.remove(i)
            ones.append(i); zeros.append(j)
        if current < best:
            best = current
            best_h = h.copy()
            if best <= 0.38085:
                break
    return best_h, best

import time, math
import numpy as np

def _c5_of(h: np.ndarray) -> float:
    """C5 value for a step function sampled in h."""
    n = h.shape[0]
    # correlation of h with (1-h) over the whole line (no padding)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))


def _binary_from_initial(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Turn ``arr`` into a binary vector with exactly n/2 ones."""
    n = arr.shape[0]
    # force to [0,1] and threshold
    bin_vec = (np.clip(arr, 0.0, 1.0) >= 0.5).astype(np.float64)
    target = n // 2
    cur = int(bin_vec.sum())
    if cur < target:
        zeros = np.where(bin_vec == 0)[0]
        rng.shuffle(zeros)
        bin_vec[zeros[: target - cur]] = 1.0
    elif cur > target:
        ones = np.where(bin_vec == 1)[0]
        rng.shuffle(ones)
        bin_vec[ones[: cur - target]] = 0.0
    return bin_vec


def _thue_morse_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Balanced Thue‑Morse binary vector of length n."""
    # parity of binary weight of each index
    parity = np.fromiter((bin(i).count("1") & 1 for i in range(n)), dtype=np.int8)
    # choose the colour which is closest to half ones
    zeros = np.sum(parity == 0)
    ones = n - zeros
    if zeros == n // 2:
        vec = (parity == 0).astype(np.float64)
    elif ones == n // 2:
        vec = (parity == 1).astype(np.float64)
    else:
        # start with the larger colour and flip the excess
        if zeros > n // 2:
            vec = (parity == 0).astype(np.float64)
            excess = zeros - n // 2
            idx = np.where(vec == 1)[0]
            rng.shuffle(idx)
            vec[idx[:excess]] = 0.0
        else:
            vec = (parity == 1).astype(np.float64)
            excess = ones - n // 2
            idx = np.where(vec == 1)[0]
            rng.shuffle(idx)
            vec[idx[:excess]] = 0.0
    return vec


def _simulated_annealing(h_start: np.ndarray, time_budget: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Simple Metropolis search on binary vectors with a fixed Hamming weight."""
    n = h_start.shape[0]
    h = h_start.copy()
    cur_val = _c5_of(h)

    # maintain explicit index lists for fast swaps
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])

    best_h = h.copy()
    best_val = cur_val

    start = time.time()
    # a generous upper bound – we will stop well before it if the budget expires
    max_iters = 2_000_000
    for it in range(max_iters):
        if time.time() - start > time_budget * 0.95:
            break

        # linear cooling schedule (starts at 0.2, ends at 0)
        T = 0.20 * (1.0 - it / max_iters)

        # propose a random swap of a 1 and a 0
        i = rng.choice(ones)
        j = rng.choice(zeros)

        h[i], h[j] = h[j], h[i]          # perform swap
        ones.remove(i)
        zeros.remove(j)
        ones.append(j)
        zeros.append(i)

        cand_val = _c5_of(h)

        # Metropolis acceptance
        if cand_val < cur_val or rng.random() < math.exp(-(cand_val - cur_val) / (T + 1e-12)):
            cur_val = cand_val
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]
            ones.remove(j)
            zeros.remove(i)
            ones.append(i)
            zeros.append(j)

        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
            if best_val <= 0.38085:
                break
    return best_h, best_val


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------ #
    # Determine problem size and a first binary candidate
    # ------------------------------------------------------------------ #
    if "initial_h_values" in globals():
        init_vec = np.array(initial_h_values, copy=True)
        n = init_vec.shape[0]
        start_bin = _binary_from_initial(init_vec, rng)
    else:
        n = 80                     # a modest default size
        start_bin = np.zeros(n)
        start_bin[: n // 2] = 1.0
        rng.shuffle(start_bin)

    # ------------------------------------------------------------------ #
    # Assemble a few cheap deterministic candidates
    # ------------------------------------------------------------------ #
    candidates = [(start_bin, _c5_of(start_bin))]

    # Thue‑Morse pattern (often has nice autocorrelation)
    tm_vec = _thue_morse_binary(n, rng)
    candidates.append((tm_vec, _c5_of(tm_vec)))

    # a few pure random starts (use a small fraction of the budget for evaluation)
    for _ in range(3):
        rand = np.zeros(n)
        rand[: n // 2] = 1.0
        rng.shuffle(rand)
        candidates.append((rand, _c5_of(rand)))

    # keep the best of the cheap attempts
    best_h, best_val = min(candidates, key=lambda p: p[1])

    # ------------------------------------------------------------------ #
    # If the current best already satisfies the target, return immediately
    # ------------------------------------------------------------------ #
    target = 0.38085
    if best_val <= target:
        return best_h, best_val, n

    # ------------------------------------------------------------------ #
    # Main optimisation phase – simulated annealing with time safety‑guard
    # ------------------------------------------------------------------ #
    start_time = time.time()
    time_left = budget_s - (time.time() - start_time)

    # run SA on the current best vector
    h_opt, v_opt = _simulated_annealing(best_h, time_left, rng)
    if v_opt < best_val:
        best_h, best_val = h_opt, v_opt

    # additional random‑restart SA loops while time permits
    while time.time() - start_time < budget_s * 0.90:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5.0:
            break

        # another random binary start
        rand = np.zeros(n)
        rand[: n // 2] = 1.0
        rng.shuffle(rand)

        # spend at most half of the remaining budget on this restart
        h_opt, v_opt = _simulated_annealing(rand, remaining * 0.5, rng)
        if v_opt < best_val:
            best_h, best_val = h_opt, v_opt
        if best_val <= target:
            break

    # ------------------------------------------------------------------ #
    # Final sanity checks (clip, enforce exact mass)
    # ------------------------------------------------------------------ #
    best_h = np.clip(best_h, 0.0, 1.0)
    total = best_h.sum()
    target_sum = n / 2
    if abs(total - target_sum) > 1e-9:
        # simple proportional scaling, then a tiny correction to keep bounds
        scale = target_sum / total
        best_h *= scale
        np.clip(best_h, 0.0, 1.0, out=best_h)
        # tiny redistribution if rounding broke the equality
        diff = target_sum - best_h.sum()
        if abs(diff) > 1e-12:
            free = np.where((best_h > 0) & (best_h < 1))[0]
            if free.size:
                inc = diff / free.size
                best_h[free] += inc
                np.clip(best_h, 0.0, 1.0, out=best_h)

    final_c5 = _c5_of(best_h)
    return best_h, final_c5, n
