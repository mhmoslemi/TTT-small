def compute_c5_fft(h):
    N = len(h)
    dx = 2.0/N
    a = h.astype(np.float64)
    b = 1 - a
    n = 2*N  # zero pad length for linear correlation using FFT
    A = np.fft.rfft(a, n=n)
    B = np.fft.rfft(b, n=n)
    # correlation a * b_rev (since correlation = conv(a, reverse(b))) => In frequency domain:
    # conv(a, b_rev) = ifft(fft(a) * conj(fft(b)))
    # But b_rev is reversed version of b. If we instead compute convolution of a and reversed b,
    # we can produce appropriate multiplication.
    # However using conj multiplication yields correlation without reversal if we treat conj(FFT(b)).
    # Actually for real sequences, correlation r[k] = sum_n a[n] * b[n+k] = (a * reverse(b))[k]
    # The FFT method: r = ifft(fft(a) * conj(fft(b))) gives linear correlation (no reversal).
    # So we can just compute correlation as they're defined.
    corr = np.fft.irfft(A * np.conj(B), n=n)
    # corr length = n; only first 2N-1 needed
    corr = corr[:2*N-1]
    return np.max(corr) * dx
```

def anneal(h_initial, max_iter, start_time):
   h = h_initial.copy()
   cur_c5 = compute_c5_fft(h)
   best_h = h.copy()
   best_c5 = cur_c5
   N = len(h)
   step = 0.2  # max delta per move
   T0 = 0.02  # initial temperature
   # schedule: T = T0 * (1 - iter / max_iter)
   for it in range(max_iter):
       if time.time() - start_time > budget - 0.5: break
       # gradually reduce step and T:
       progress = it / max_iter
       step = max(0.001, step * (1 - progress*0.5))  # decays to small
       T = T0 * (1 - progress)
       i, j = np.random.randint(0, N, size=2)
       while j == i: j = np.random.randint(0, N)
       direction = np.random.choice([1,-1])
       if direction == 1:
           delta_max = min(step, 1 - h[i], h[j])
       else:
           delta_max = min(step, h[i], 1 - h[j])
       if delta_max <= 0:
           continue
       delta = np.random.rand() * delta_max
       # apply move
       if direction == 1:
           h[i] += delta
           h[j] -= delta
       else:
           h[i] -= delta
           h[j] += delta
       # compute new C5
       new_c5 = compute_c5_fft(h)
       diff = new_c5 - cur_c5
       # acceptance
       if diff <= 0 or np.random.rand() < np.exp(-diff / (T+1e-12)):
           cur_c5 = new_c5
           # update best if new better
           if new_c5 < best_c5:
               best_c5 = new_c5
               best_h = h.copy()
       else:
           # revert
           if direction == 1:
               h[i] -= delta
               h[j] += delta
           else:
               h[i] += delta
               h[j] -= delta
   return best_h, best_c5
```

                    diff -= add

                    h0[idx] -= sub
