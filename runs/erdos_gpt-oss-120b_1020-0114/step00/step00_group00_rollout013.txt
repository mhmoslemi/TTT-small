def generate_triangular(N):
    dx = 2.0 / N
    # positions at midpoints of intervals
    x = (np.arange(N) + 0.5) * dx
    h = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
    # Adjustment to satisfy integral exactly:
    total = np.sum(h)
    target = N / 2.0
    diff = target - total
    # Ideally diff small; adjust at peak
    idx_peak = np.argmin(np.abs(x - 1.0))
    h[idx_peak] += diff
    # clip range
    h = np.clip(h, 0.0, 1.0)
    # iterate to correct any residual diff
    for _ in range(10):
        total = np.sum(h)
        if abs(total - target) < 1e-12:
            break
        diff = target - total
        if diff > 0:
            # find index with capacity to increase
            idx = np.argmax(1 - h)
            delta = min(diff, 1 - h[idx])
            h[idx] += delta
        else:
            idx = np.argmin(h)
            delta = max(diff, -h[idx])
            h[idx] += delta
        # Clip anyway
        h = np.clip(h, 0.0, 1.0)
    return h

import time
import numpy as np

def _triangular_h(N: int) -> np.ndarray:
    """Return a stepwise approximation of the triangular function
    h(x)=max(0,1-|x-1|) on [0,2] sampled at N midpoints.
    The vector is adjusted so that sum(h)=N/2 (i.e. ∫h=1)."""
    dx = 2.0 / N
    # mid‑point grid
    x = (np.arange(N) + 0.5) * dx
    h = np.maximum(0.0, 1.0 - np.abs(x - 1.0))

    # enforce the integral constraint exactly
    target = N / 2.0
    cur = h.sum()
    diff = target - cur
    # add the whole adjustment to the point nearest the peak
    peak = np.argmin(np.abs(x - 1.0))
    h[peak] += diff
    h = np.clip(h, 0.0, 1.0)

    # if clipping altered the total, repair by small redistributions
    for _ in range(10):
        cur = h.sum()
        if abs(cur - target) < 1e-12:
            break
        diff = target - cur
        if diff > 0:
            # increase a point that can still grow
            i = np.argmax(1.0 - h)
            step = min(diff, 1.0 - h[i])
        else:
            # decrease a point that is positive
            i = np.argmin(h)
            step = max(diff, -h[i])
        h[i] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def _c5_of(h: np.ndarray) -> float:
    """Compute C5 using the harness' definition (full correlation)."""
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)
