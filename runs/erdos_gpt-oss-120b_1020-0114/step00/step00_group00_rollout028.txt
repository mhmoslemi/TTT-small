def proj_capped_simplex(v, target, lower=0, upper=1, max_iter=100):
    lo = min(v - upper) - 1.0  # to ensure lower bound for λ
    hi = max(v - lower) + 1.0
    for _ in range(max_iter):
        lam = (lo+hi)/2
        x = np.clip(v - lam, lower, upper)
        s = x.sum()
        if abs(s - target) < 1e-10:
            return x
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo+hi)/2, lower, upper)

def project_capped_simplex(v, target, lower=0.0, upper=1.0, eps=1e-12, max_iter=100):
    low = np.min(v) - upper
    high = np.max(v) - lower
    # Expand a bit
    for _ in range(max_iter):
        lam = (low + high) / 2
        x = np.clip(v - lam, lower, upper)
        s = x.sum()
        if abs(s - target) < eps:
            return x
        if s > target:
            low = lam
        else:
            high = lam
    # final
    return np.clip(v - ((low + high) / 2), lower, upper)

import time, numpy as np

def _project_capped_simplex(v, target, lo=0.0, hi=1.0, eps=1e-10, it=50):
    """Euclidean projection of v onto {x: lo<=x<=hi, sum(x)=target}."""
    low = np.min(v) - hi
    high = np.max(v) - lo
    for _ in range(it):
        lam = (low + high) * 0.5
        x = np.clip(v - lam, lo, hi)
        s = x.sum()
        if abs(s - target) <= eps:
            return x
        if s > target:
            low = lam
        else:
            high = lam
    return np.clip(v - (low + high) * 0.5, lo, hi)


def _c5_value(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode="full") * dx)


def _upscale(src, n):
    """Linear interpolation of a step function to a new resolution."""
    m = src.shape[0]
    xs = (np.arange(m) + 0.5) * (2.0 / m)
    xt = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(xt, xs, src)

        if s >= 0:
            i1 = np.arange(n - s)
            g[i1] += (1 - cur_h[i1 + s])
            i2 = np.arange(s, n)
            g[i2] -= cur_h[i2 - s]
        else:
            a = -s
            i1 = np.arange(a, n)
            g[i1] += (1 - cur_h[i1 - a])
            i2 = np.arange(0, n - a)
            g[i2] -= cur_h[i2 + a]

        # take a step and re‑project onto the feasible set
        cand = cur_h - alpha * g
        cand = _project_capped_simplex(cand, target)
        cand_score = _c5_value(cand)

        # accept improvements
        if cand_score + 1e-12 < cur_score:
            cur_h, cur_score = cand, cand_score
            no_imp = 0
            if cand_score < best_score:
                best_h, best_score = cand.copy(), cand_score
        else:
            # occasional jitter to escape plateaus
            if rng.random() < 0.02:
                jitter = (rng.random(n) - 0.5) * 0.02
                cand2 = cur_h + jitter
                cand2 = _project_capped_simplex(cand2, target)
                cand2_score = _c5_value(cand2)
                if cand2_score < cur_score:
                    cur_h, cur_score = cand2, cand2_score
                    if cand2_score < best_score:
                        best_h, best_score = cand2.copy(), cand2_score
            no_imp += 1

        # restart if stuck too long
        if no_imp > 3000:
            cur_h = _project_capped_simplex(rng.random(n), target)
            cur_score = _c5_value(cur_h)
            no_imp = 0
            alpha = 0.12

        # slowly decay the step size
        if (no_imp % 500) == 0:
            alpha *= 0.95

    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    overall_best = None          # (h, score, n)

    # resolutions we will try; they are cheap enough for the budget
    candidates = [45, 60, 80, 100]
    start_total = time.time()

    for n in candidates:
        # allocate a share of the remaining budget
        elapsed = time.time() - start_total
        if elapsed >= budget_s * 0.98:
            break
        remaining = budget_s - elapsed
        # give each resolution at most 0.25 · remaining seconds
        sub_budget = min(remaining * 0.25, 350.0)

        h_opt, score_opt = _optimize_one(n, rng, sub_budget)

        if (overall_best is None) or (score_opt < overall_best[1]):
            overall_best = (h_opt, score_opt, n)

    # safety fallback – uniform function if everything failed
    if overall_best is None:
        n = 60
        h = np.full(n, 0.5)
        score = _c5_value(h)
        overall_best = (h, score, n)

    h_best, c5_best, n_best = overall_best
    return (h_best, float(c5_best), int(n_best))
