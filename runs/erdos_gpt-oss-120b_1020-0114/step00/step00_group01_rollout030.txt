def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with 1-h
    h_rev = h[::-1]
    # choose fft size pow2 >= 2*n-1
    L = 1 << ((2*n-1).bit_length())
    fft_h_rev = np.fft.rfft(h_rev, n=L)
    fft_one_minus_h = np.fft.rfft(1 - h, n=L)
    conv = np.fft.irfft(fft_h_rev * fft_one_minus_h, n=L)[:2*n-1]
    return conv.max() * dx
```

def construct_h(a,b,n):
    # compute c
    c = 2 - 2*a - 2*b
    u = (1 - 2*a) / (2*b) if b>0 else 0.0
    # Ensure bounds
    # Build piecewise constant h:
    # segment lengths (in terms of x length):
    # L1 = a, L2 = b, L3 = c, L4 = b, L5 = a
    # total 2
    # Convert lengths to sample counts: n_i = int(round(Li / 2 * n))? because total length 2 corresponds to n points with spacing dx = 2/n.
    # The number of points per segment is round(Li / dx) = round(Li * n / 2).
    # Use floor/ceil correction to ensure sum = n.
    seg_lengths = np.array([a,b,c,b,a]) * n / 2.0
    # Convert to integer counts using rounding while keeping sum = n
    seg_counts = np.floor(seg_lengths).astype(int)
    remainder = n - seg_counts.sum()
    # Distribute remainder to largest fractional parts
    fractions = seg_lengths - seg_counts
    if remainder > 0:
        idx = np.argsort(-fractions)[:remainder]
        seg_counts[idx] += 1
    # Build h array
    h = np.concatenate([
        np.full(seg_counts[0], 1.0),
        np.full(seg_counts[1], u),
        np.full(seg_counts[2], 0.0),
        np.full(seg_counts[3], u),
        np.full(seg_counts[4], 1.0)
    ])
    # If any rounding mismatch, adjust length
    if h.shape[0] != n:
        # pad or trim
        if h.shape[0] > n:
            h = h[:n]
        else:
            # pad zeros
            h = np.concatenate([h, np.zeros(n - h.shape[0])])
    return h
```

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # use fft correlation
    rev = h[::-1]
    L = 1 << ((2*n - 1).bit_length())
    f_rev = np.fft.rfft(rev, L)
    f_one_minus = np.fft.rfft(1 - h, L)
    conv = np.fft.irfft(f_rev * f_one_minus, L)[:2*n - 1]
    return conv.max() * dx
```

def project_to_constraints(v, target):

def _project_to_constraints(v: np.ndarray, target: float) -> np.ndarray:
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _compute_c5(h: np.ndarray) -> float:
    n = h.shape[0]
    dx = 2.0 / n
    rev = h[::-1]
    L = 1 << ((2 * n - 1).bit_length())
    fr = np.fft.rfft(rev, L)
    fo = np.fft.rfft(1.0 - h, L)
    conv = np.fft.irfft(fr * fo, L)[: 2 * n - 1]
    return conv.max() * dx


def _construct_h(a: float, b: float, n: int) -> np.ndarray:
    c = 2.0 - 2.0 * a - 2.0 * b
    u = (1.0 - 2.0 * a) / (2.0 * b) if b > 1e-12 else 0.0
    u = max(0.0, min(1.0, u))
    seg_len = np.array([a, b, c, b, a]) * n / 2.0
