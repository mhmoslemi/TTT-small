def smooth_c5_and_grad(h, dx, alpha):
    # compute auto-correlation of h with itself
    corr = np.correlate(h, h, mode='full')
    # overlap = corr * dx
    f = 1.0 - corr * dx  # shape 2N-1
    # ensure numeric stable
    max_f = np.max(f)  # may be up to 1?

    # Use w = np.exp(alpha * (f - max_f))  # to keep numbers moderate
    shifted_f = f - max_f

def project_onto_box_simplex(x, target_sum):
    lo = -np.max(x)  # ensures x+lo <=0 => clipped to zero
    hi = 1 - np.min(x)  # ensures x+hi >=1 => clipped to 1

    for _ in range(60):
        mid = (lo + hi) * 0.5
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    y = np.clip(x + (lo+hi)*0.5, 0.0, 1.0)
    # adjust rounding
    diff = target_sum - y.sum()
    if abs(diff) > 1e-12:
        free = (y > 0) & (y < 1)
        if free.any():
            y[free] += diff / free.sum()
            y = np.clip(y, 0.0, 1.0)
        else:
            # adjust all entries uniformly
            y += diff / len(y)
            y = np.clip(y, 0.0, 1.0)
    return y

def smooth_and_grad(h, dx, alpha):
    # correlation of h with itself
    corr = np.correlate(h, h, mode='full')
    f = 1.0 - corr * dx
    max_f = np.max(f)
    # stable exponentiation
    w = np.exp(alpha * (f - max_f))
    sumw = w.sum()
    w_norm = w / sumw
    # convolution using scipy.signal
    from scipy import signal
    conv_minus = signal.convolve(h, w_norm, mode='same')
    conv_plus = signal.convolve(h, w_norm[::-1], mode='same')
    grad = -dx * (conv_minus + conv_plus)
    # smooth value (not needed for optimization but helpful)
    C_smooth = (np.log(sumw) + alpha * max_f) / alpha
    return C_smooth, grad

import numpy as np
import time
from scipy import signal


def project_onto_box_simplex(x, target):
    lo = -np.max(x)          # makes all entries ≤0 after adding lo
    hi = 1.0 - np.min(x)     # makes all entries ≥1 after adding hi
    for _ in range(60):
        mid = (lo + hi) * 0.5
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        free = (y > 0) & (y < 1)
        if free.any():
            y[free] += diff / free.sum()
            y = np.clip(y, 0.0, 1.0)
        else:
            y += diff / len(y)
            y = np.clip(y, 0.0, 1.0)
    return y


def compute_c5(h, dx):
    """exact C₅ = max ∫ h(x)(1‑h(x+k))dx  (discrete version)"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def smooth_and_grad(h, dx, alpha):
    """smooth approximation of the max and its gradient."""
    corr = np.correlate(h, h, mode='full')
    f = 1.0 - corr * dx                # f_k = 1 − overlap_k
    max_f = np.max(f)
    w = np.exp(alpha * (f - max_f))    # shift to avoid overflow
    sumw = w.sum()
    w_norm = w / sumw                  # normalized weights

    conv_minus = signal.convolve(h, w_norm, mode='same')
    conv_plus = signal.convolve(h, w_norm[::-1], mode='same')
    grad = -dx * (conv_minus + conv_plus)

    C_smooth = (np.log(sumw) + alpha * max_f) / alpha
    return C_smooth, grad


def hill_climb_step(h, cur_c5, dx):
    """transfer a random amount of mass between two coordinates."""
    n = len(h)
    i, j = np.random.choice(n, size=2, replace=False)
    if np.random.rand() < 0.5:
        donor, receiver = i, j
    else:
        donor, receiver = j, i
    max_transfer = min(h[donor], 1.0 - h[receiver])
    if max_transfer <= 1e-12:
        return h, cur_c5
    delta = np.random.rand() * max_transfer
    h2 = h.copy()
    h2[donor] -= delta
    h2[receiver] += delta
    new_c5 = compute_c5(h2, dx)
    if new_c5 < cur_c5 - 1e-12:
        return h2, new_c5
    return h, cur_c5


def alpha_schedule(it):
    """gradually tighten the smooth‑max approximation."""
    if it < 200:
        return 0.5
    elif it < 800:
        return 2.0
    elif it < 2000:
        return 10.0
    elif it < 5000:
        return 30.0
    else:
        return 60.0


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # ---- initialise h ----------------------------------------------------
    init = globals().get('initial_h_values')
    if init is not None:
        h0 = np.asarray(init, dtype=float)
    else:
        n = 120
        h0 = np.zeros(n)
        half = n // 2
        h0[:half] = 1.0
        np.random.shuffle(h0)

    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0

    if not (np.all((h0 >= 0) & (h0 <= 1)) and abs(h0.sum() - target) < 1e-9):
        h0 = project_onto_box_simplex(h0, target)

    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    cur_c5 = best_c5

    it = 0
    # ---- main optimisation loop -----------------------------------------
    while time.time() - start < budget_s * 0.95:
        it += 1
        alpha = alpha_schedule(it)

        # smooth max gradient
        _, grad = smooth_and_grad(h, dx, alpha)

        # simple gradient descent with a decreasing step size
        lr = 0.2 / (1.0 + it * 0.001)          # ≈0.2 → 0.1 during the run
        h = project_onto_box_simplex(h - lr * grad, target)

        # exact objective after the gradient step
        cur_c5 = compute_c5(h, dx)
        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            best_h = h.copy()

        # occasional hill‑climbing move
        if it % 5 == 0:
            h, cur_c5 = hill_climb_step(h, cur_c5, dx)
            if cur_c5 < best_c5 - 1e-12:
                best_c5 = cur_c5
                best_h = h.copy()

        # occasional random jitter to escape plateaus
        if it % 500 == 0:
            noise = np.random.normal(0.0, 0.02, size=n)
            h = best_h + noise
            h = np.clip(h, 0.0, 1.0)
            h = project_onto_box_simplex(h, target)
            cur_c5 = compute_c5(h, dx)

    final_c5 = compute_c5(best_h, dx)
    return best_h, float(final_c5), n
