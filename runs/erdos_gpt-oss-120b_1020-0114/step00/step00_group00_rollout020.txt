    idx_max = np.argmax(c)
    shift = idx_max - (n - 1)

    w = compute_w(h, shift)
    # indices where h<1 and h>0
    up_candidates = np.where(h < 1 - 1e-12)[0]
    down_candidates = np.where(h > 1e-12)[0]
    if len(up_candidates)==0 or len(down_candidates)==0:
        return h, None
    # choose i_up with max w (prefer bigger) and i_down with min w
    i_up = up_candidates[np.argmax(w[up_candidates])]
    i_down = down_candidates[np.argmin(w[down_candidates])]

import time, math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx discretized."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)


def _worst_shift(h: np.ndarray):
    """Return (shift, correlation array) where shift gives the maximal overlap."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)          # shift = index‑(n‑1)
    return shift, corr


def _w_vector(h: np.ndarray, shift: int):
    """w_i = h_{i+shift}+h_{i-shift} (zero outside)."""
    n = h.shape[0]
    if shift == 0:
        return 2.0 * h
    w = np.zeros_like(h)
    if shift > 0:
        w[: n - shift] += h[shift:]
        w[shift:] += h[: n - shift]
    else:  # shift < 0
        s = -shift
        w[s:] += h[: n - s]
        w[: n - s] += h[s:]
    return w


def _greedy_optimize(h: np.ndarray, time_limit: float) -> (np.ndarray, float):
    """Simple deterministic / annealing search that respects the constraints."""
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_val = _compute_c5(best_h)

    cur_h = h.copy()
    cur_val = best_val
    step = 0.05                     # initial move size
    no_improve = 0
    start = time.time()

    while time.time() - start < time_limit:
        # find worst shift and the associated w‑vector
        shift, _ = _worst_shift(cur_h)
        w = _w_vector(cur_h, shift)

        # decide whether to use greedy pair or a random perturbation
        if np.random.rand() < 0.7:
            up_mask = cur_h < 1.0 - 1e-12
            down_mask = cur_h > 1e-12
            if not up_mask.any() or not down_mask.any():
                break
            up_idx = np.where(up_mask)[0]
            down_idx = np.where(down_mask)[0]
            i_up = up_idx[np.argmax(w[up_idx])]
            i_down = down_idx[np.argmin(w[down_idx])]
            delta = min(step, 1.0 - cur_h[i_up], cur_h[i_down])
        else:
            # random pair with a random feasible delta
            i_up = np.random.randint(0, n)
            i_down = np.random.randint(0, n - 1)
            if i_down >= i_up:
                i_down += 1
            max_inc = min(1.0 - cur_h[i_up], cur_h[i_down])
            max_dec = min(cur_h[i_up], 1.0 - cur_h[i_down])
            if max_inc > 0 and (max_dec == 0 or np.random.rand() < 0.5):
                delta = np.random.rand() * max_inc
            elif max_dec > 0:
                delta = -np.random.rand() * max_dec
                i_up, i_down = i_down, i_up
            else:
                continue          # nothing to move
        if delta <= 0:
            continue

        new_h = cur_h.copy()
        new_h[i_up] += delta
        new_h[i_down] -= delta
        new_val = _compute_c5(new_h)

        # acceptance

            cur_h, cur_val = new_h, new_val

        else:
            best_h[0] = min(1.0, max(0.0, best_h[0] + diff))
        best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, _compute_c5(best_h)


def _slsqp_optimize(h0: np.ndarray, time_limit: float) -> (np.ndarray, float):
    """Non‑linear constrained optimisation (SLSQP) using SciPy."""
    try:
        from scipy.optimize import minimize
    except Exception:                 # SciPy missing – fall back
        return h0, _compute_c5(h0)

    n = h0.shape[0]
    target_sum = n / 2.0
    t0 = _compute_c5(h0)               # initial upper bound guess
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    # equality: total mass
    eq_cons = {'type': 'eq', 'fun': lambda x: np.sum(x[:n]) - target_sum}
    cons = [eq_cons]

    # inequality for every possible shift
    shifts = list(range(-(n - 1), n))
    def make_ineq(s):
        def fun(x):
            h = x[:n]
            t = x[-1]
            if s >= 0:
                val = np.dot(h[: n - s], 1.0 - h[s:])
            else:
                s2 = -s
                val = np.dot(h[s2:], 1.0 - h[: n - s2])
            return t - val                     # we require t ≥ overlap(s)
        return fun
    cons.extend({'type': 'ineq', 'fun': make_ineq(s)} for s in shifts)

    # limit iteration count roughly to the available time
    max_iter = max(10, int(min(500, time_limit * 20)))
    opts = {'maxiter': max_iter, 'ftol': 1e-12, 'disp': False}

    start = time.time()
    res = minimize(lambda x: x[-1], x0, method='SLSQP',
                   bounds=bounds, constraints=cons, options=opts)
    # enforce feasibility in case of tiny violations
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    diff = target_sum - h_opt.sum()
    if abs(diff) > 1e-10:
        free = np.where((h_opt > 1e-12) & (h_opt < 1.0 - 1e-12))[0]
        if free.size:
            h_opt[free] += diff / free.size
            h_opt = np.clip(h_opt, 0.0, 1.0)
    c5 = _compute_c5(h_opt)
    return h_opt, c5

    else:
        # fallback: half ones, half zeros, shuffled
        n_def = 45
        h_start = np.zeros(n_def, dtype=float)
        idx = np.random.choice(n_def, n_def // 2, replace=False)
        h_start[idx] = 1.0

    n0 = h_start.shape[0]
    target_sum0 = n0 / 2.0
    # ensure exact sum (tiny numerical errors may appear)
    cur_sum = h_start.sum()
    if abs(cur_sum - target_sum0) > 1e-12:
        free = np.where((h_start > 1e-12) & (h_start < 1.0 - 1e-12))[0]
        if free.size:
            h_start[free] += (target_sum0 - cur_sum) / free.size
        else:
            h_start[0] = min(1.0, max(0.0, h_start[0] + target_sum0 - cur_sum))
        h_start = np.clip(h_start, 0.0, 1.0)

    # ------------------------------------------------------------------
    # 2️⃣  Greedy / annealing improvement on the original resolution
    # ------------------------------------------------------------------
    time_for_greedy = budget_s * 0.70
    h_best, c5_best = _greedy_optimize(h_start, time_for_greedy)

    # ------------------------------------------------------------------
    # 3️⃣  Optional SLSQP refinement (uses remaining budget)
    # ------------------------------------------------------------------
    time_remaining = budget_s - (time.time() - start_all)
    if time_remaining > 2.0:               # give it a few seconds at least
        h_refined, c5_refined = _slsqp_optimize(h_best, time_remaining)
        if c5_refined < c5_best:
            h_best, c5_best = h_refined, c5_refined

    # ------------------------------------------------------------------
    # 4️⃣  Final consistency check
    # ------------------------------------------------------------------
    h_best = np.clip(h_best, 0.0, 1.0)
    # enforce exact integral
    diff = target_sum0 - h_best.sum()
    if abs(diff) > 1e-12:
        free = np.where((h_best > 1e-12) & (h_best < 1.0 - 1e-12))[0]
        if free.size:
            h_best[free] += diff / free.size
        else:
            h_best[0] = min(1.0, max(0.0, h_best[0] + diff))
        h_best = np.clip(h_best, 0.0, 1.0)

    final_c5 = _compute_c5(h_best)
    return h_best, float(final_c5), int(n0)
```

This implementation

* respects all constraints (values in [0, 1] and integral = 1),
* starts from the supplied `initial_h_values`,
* performs a fast deterministic/annealing local search,
* optionally refines the result with SciPy’s SLSQP within the remaining time,
* finally returns the step‑function, its C₅ bound and the sample count.
