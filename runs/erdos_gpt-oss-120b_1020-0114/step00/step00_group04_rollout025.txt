def simulated_annealing(h0, rng, time_limit):
    h = h0.copy()
    best_h = h.copy()
    best_val = evaluate(h)
    cur_val = best_val
    N = h.size
    start = time.time()
    # parameters
    T = 0.1  # initial temperature relative to value
    # we can set schedule: after certain number of steps reduce T by factor
    step = 0
    while time.time() - start < time_limit:
        # propose move:
        i, j = rng.integers(N, size=2)
        if i == j: continue
        # maximum delta: limited to maintain h within bounds
        # compute delta range:
        max_decrease = h[i]  # cannot make negative
        max_increase = 1.0 - h[j]  # cannot exceed 1
        max_delta = min(max_decrease, max_increase, 0.1)  # limit step size
        if max_delta <= 1e-12:
            continue
        delta = rng.random() * max_delta
        # apply move
        h[i] -= delta
        h[j] += delta
        new_val = evaluate(h)
        dE = new_val - cur_val
        if dE < 0 or rng.random() < np.exp(-dE / T):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert
            h[i] += delta
            h[j] -= delta
        step += 1
        # cooling schedule: T decay per iteration
        if step % 5000 == 0:
            T *= 0.9  # reduce temperature
    return best_h, best_val

import time
import math
import numpy as np

def _c5_value(vec: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1−h(x+k))dx approximated on the grid."""
    n = vec.shape[0]
    corr = np.correlate(vec, 1.0 - vec, mode="full")
    return corr.max() * (2.0 / n)


def _project_to_simplex(vec: np.ndarray, target_sum: float) -> None:
    """Clamp to [0,1] and adjust linearly so that sum(vec) == target_sum."""
    np.clip(vec, 0.0, 1.0, out=vec)
    diff = target_sum - vec.sum()
    if abs(diff) < 1e-12:
        return
    # distribute the remaining mass greedily
    if diff > 0:
        for i in range(vec.size):
            room = 1.0 - vec[i]
            add = min(room, diff)
            vec[i] += add
            diff -= add
            if diff <= 1e-12:
                break
    else:  # diff < 0
        for i in range(vec.size):
            room = vec[i]
            sub = min(room, -diff)
            vec[i] -= sub
            diff += sub
            if diff >= -1e-12:
                break


def _thue_morse(n: int) -> np.ndarray:
    """Binary Thue–Morse sequence of length n (n must be a power of two)."""
    seq = np.empty(n, dtype=np.float64)
    for i in range(n):
        seq[i] = (bin(i).count("1") & 1)
    return seq


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----------------------------------------------------------------------
    # 1. Initial feasible vector
    # ----------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n = h.size
        target = n / 2.0
        _project_to_simplex(h, target)
    else:
        # power‑of‑two length lets us start from the Thue–Morse pattern
        n = 128
        h = _thue_morse(n)
        target = n / 2.0
        _project_to_simplex(h, target)

    # ----------------------------------------------------------------------
    # 2. Simulated‑annealing optimisation
    # ----------------------------------------------------------------------
    best_h = h.copy()
    best_val = _c5_value(best_h)
    cur_h = h.copy()
    cur_val = best_val

    max_iters = 400_000                     # generous upper bound
    T0 = 0.04                               # initial temperature
    Tf = 1e-6                               # final temperature
    prob_big = 0.015                        # chance to perform a “big” move

    for it in range(max_iters):
        # time guard
        if time.time() - start_time > budget_s * 0.94:
            break

        # linear cooling schedule (exponential in practice)
        progress = it / max_iters
        T = T0 * ((Tf / T0) ** progress)

        if rng.random() < prob_big:
            # ---------- big move: re‑distribute mass on a random subset ----------
            size = rng.integers(2, min(20, n) + 1)
            idx = rng.choice(n, size=size, replace=False)
            total = cur_h[idx].sum()
            # draw a random point in the simplex on the subset
            for _ in range(5):
                w = rng.random(size)
                if w.sum() == 0:
                    continue
                new_vals = total * (w / w.sum())
                if np.all(new_vals <= 1.0):
                    cand = cur_h.copy()
                    cand[idx] = new_vals
                    cand_val = _c5_value(cand)

            else:

                if i == j:

                max_inc = 1.0 - cur_h[j]
