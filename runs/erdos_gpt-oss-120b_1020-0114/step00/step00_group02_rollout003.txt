def anneal_n(n, total_time):
    start = time.time()
    best_val = np.inf
    best_h = None
    # global iteration count base
    while time.time() - start < total_time:
        # start a new run with random binary half ones.
        h = np.zeros(n, dtype=np.float64)
        # set exactly n//2 ones
        indices = rng.permutation(n)[:n//2]
        h[indices] = 1.0
        curr_val = compute_c5(h)
        # local best for this run:
        local_best_val = curr_val
        local_best_h = h.copy()
        # local annealing iterations
        max_local_iters = int(1e5)  # maybe adjust
        for it in range(max_local_iters):
            # propose swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones)==0 or len(zeros)==0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h[i], h[j] = 0.0, 1.0
            new_val = compute_c5(h)
            delta = new_val - curr_val
            T = 0.005 * (1 - it / max_local_iters) # schedule, may adjust
            accept = False
            if delta < 0:
                accept = True
            elif T > 0:
                if rng.random() < np.exp(-delta / T):
                    accept = True
            if accept:
                curr_val = new_val
                if new_val < local_best_val:
                    local_best_val = new_val
                    local_best_h = h.copy()
            else:
                # revert swap
                h[i], h[j] = 1.0, 0.0
            # time check maybe break
            if time.time() - start > total_time:
                break
        # after local run, compare with global best
        if local_best_val < best_val:
            best_val = local_best_val
            best_h = local_best_h.copy()
        # check time for global loop
        if time.time() - start > total_time:
            break
    return best_h, best_val

def refine_continuous(h, time_limit):
    # h shape n, values in [0,1] sum = n/2.
    n = len(h)
    dx = 2.0 / n
    curr_val = compute_c5(h)
    best_val = curr_val
    best_h = h.copy()
    start = time.time()
    # Use simple coordinate descent: each iteration pick random index and adjust up/down small step while project sum.
    while time.time() - start < time_limit:
        i = rng.integers(n)
        # propose delta = step * random sign
        step = 0.02  # adjust
        delta = step * (1 if rng.random()<0.5 else -1)
        # propose new value ensuring bounds
        new_val_i = np.clip(h[i] + delta, 0.0, 1.0)
        # To keep sum, adjust another index j to compensate the delta difference.
        diff = new_val_i - h[i]
        if abs(diff) < 1e-12:
            continue
        # pick another index j != i
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        # adjust h[j] opposite direction with same magnitude, clip
        new_val_j = np.clip(h[j] - diff, 0.0, 1.0)
        # after clipping, sum may not be exactly preserved; we can compute actual diff applied and adjust again maybe.
        # Simpler: skip this move if cannot preserve sum exactly.
        # Compute actual diff_j = new_val_j - h[j]
        actual_diff_j = new_val_j - h[j]
        # If actual_diff_j != -diff due to clipping, discard
        if abs(actual_diff_j + diff) > 1e-9:
            continue
        # apply move
        old_i = h[i]; old_j = h[j]
        h[i] = new_val_i; h[j] = new_val_j
        new_val = compute_c5(h)
        if new_val < curr_val:
            curr_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert
            h[i] = old_i; h[j] = old_j
        # maybe break if low enough
        if best_val < 0.3808:
            break
    return best_h, best_val

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    N = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(h, n=N)
    fb = np.fft.rfft(1.0 - h, n=N)
    corr = np.fft.irfft(fa * np.conj(fb), n=N)
    # Only first 2*n - 1 entries correspond to full correlation
    max_corr = np.max(corr[:2*n-1])
    return max_corr * dx
```

def refine_continuous(h0, time_limit, rng):
    h = h0.copy()
    n = len(h)
    curr_val = compute_c5(h)
    best_val = curr_val
    best_h = h.copy()
    start = time.time()
    step = 0.1  # max absolute change per move
    while time.time() - start < time_limit:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        # random delta within [-step, step]
        delta = (rng.random() * 2 - 1) * step
        # propose new values
        vi = h[i] + delta
        vj = h[j] - delta
        # clip to [0,1]

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    N = 1 << ((2*n - 1).bit_length())

import time, math
import numpy as np


def compute_c5(h: np.ndarray) -> float:
    """Return max_{k} ∫ h(x)(1-h(x+k)) dx approximated on a uniform grid."""
    n = h.size
    dx = 2.0 / n
    N = 1 << ((2 * n - 1).bit_length())           # next power of two ≥2n‑1
    fa = np.fft.rfft(h, n=N)
    fb = np.fft.rfft(1.0 - h, n=N)
    corr = np.fft.irfft(fa * np.conj(fb), n=N)
    max_corr = np.max(corr[: 2 * n - 1])
    return max_corr * dx


def thue_morse_seq(n: int) -> np.ndarray:
    """Binary Thue–Morse sequence of length n (values 0/1)."""
    seq = np.array([0], dtype=np.float64)
    while seq.size < n:
        seq = np.concatenate([seq, 1 - seq])
    return seq[:n]


def anneal_from_initial(h0: np.ndarray, time_limit: float, rng: np.random.Generator):
    """Simple simulated‑annealing on binary vectors (swap 0↔1)."""
    h = h0.copy()
    curr = compute_c5(h)
    best = curr
    best_h = h.copy()
    it = 0
    max_it = 500_000
    T0 = 0.02
    start = time.time()
    while time.time() - start < time_limit and it < max_it:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new = compute_c5(h)
        delta = new - curr
        T = T0 * (1.0 - it / max_it)
        accept = delta < 0 or (T > 0 and rng.random() < math.exp(-delta / T))
        if accept:
            curr = new
            if new < best:
                best = new
                best_h = h.copy()
                if best < 0.3808:
                    break
        else:
            h[i] = 1.0
            h[j] = 0.0
        it += 1
    return best_h, best


def refine_continuous(h0: np.ndarray, time_limit: float, rng: np.random.Generator):
    """Coordinate‑pair adjustments preserving sum and box constraints."""
    h = h0.copy()
    n = h.size
    cur = compute_c5(h)
    best = cur
    best_h = h.copy()
    step = 0.1
    it = 0
    max_it = 200_000
    start = time.time()
    while time.time() - start < time_limit and it < max_it:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = (rng.random() * 2.0 - 1.0) * step
        vi = h[i] + delta
        vj = h[j] - delta
        vi = min(1.0, max(0.0, vi))
        vj = min(1.0, max(0.0, vj))
        if abs((vi - h[i]) + (vj - h[j])) > 1e-9:
            it += 1
            continue
        old_i, old_j = h[i], h[j]
        h[i], h[j] = vi, vj
        new = compute_c5(h)
        if new < cur:
            cur = new
            if new < best:
                best = new
                best_h = h.copy()
                if best < 0.3808:
                    break
        else:
            h[i], h[j] = old_i, old_j
        it += 1
    return best_h, best
