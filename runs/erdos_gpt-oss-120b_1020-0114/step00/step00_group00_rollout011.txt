def improve_h(h, max_iter=10000, step=0.01):
    best_h = h.copy()
    best_obj = compute_c5(h)
    for it in range(max_iter):
        # compute full correlation
        corr = np.correlate(h, 1-h, mode='full')
        O = corr.max() * dx
        # early exit if no improvement possible - but continue
        # find shift index achieving max
        idx_max = np.argmax(corr)
        shift = idx_max - (len(h)-1)  # shift p
        # compute gradient
        p = shift
        # compute left neighbor: i-p index; compute arrays shifted
        left = np.zeros_like(h)
        right = np.zeros_like(h)
        if p >= 0:
            left[p:] = h[:-p] if p>0 else h
            right[:len(h)-p] = h[p:]
        else:
            # negative shift
            pneg = -p
            left[:len(h)-pneg] = h[pneg:]
            right[pneg:] = h[:-pneg]
        # gradient = 1 - left - right
        grad = 1.0 - left - right
        # identify indices with max grad (should be decreased) and min grad (to increase)
        i_max = np.argmax(grad)  # high grad -> decreasing
        i_min = np.argmin(grad)  # low grad -> increasing
        # Determine delta
        max_dec = best_h[i_max]  # can decrease at most this amount
        max_inc = 1 - best_h[i_min]  # can increase at most this amount
        delta = min(step, max_dec, max_inc)
        if delta <= 0:
            # cannot change, break?
            step *= 0.5
            continue
        new_h = best_h.copy()
        new_h[i_max] -= delta
        new_h[i_min] += delta
        # ensure sum unchanged - yes (delta cancels)
        # compute new objective
        new_obj = compute_c5(new_h)
        if new_obj < best_obj - 1e-8:
            best_h = new_h
            best_obj = new_obj
            # maybe increase step?
        else:
            # not improvement; optionally try smaller delta
            step *= 0.5
    return best_h, best_obj

def local_optimize(h, target_sum, max_iter, start_time, budget_s):
    best_h = h.copy()
    best_obj = compute_c5(best_h)
    cur_h = h.copy()
    cur_obj = best_obj
    T0 = 0.01
    for it in range(max_iter):
        if time.time() - start_time > budget_s * 0.95: break
        # linear cooling
        T = T0 * (1 - it / max_iter)
        # propose move
        i, j = np.random.randint(n, size=2)
        if i == j: continue
        max_dec = cur_h[i]
        max_inc = 1 - cur_h[j]
        if max_dec <= 0 or max_inc <= 0: continue
        delta = np.random.rand() * min(max_dec, max_inc) * 0.5  # up to half possible
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_obj = compute_c5(new_h)
        dE = new_obj - cur_obj
        if dE < 0 or np.random.rand() < np.exp(-dE / max(T, 1e-12)):
            cur_h = new_h
            cur_obj = new_obj
            if new_obj < best_obj:
                best_h = new_h.copy()
                best_obj = new_obj
    return best_h, best_obj
```

import time
import numpy as np

def compute_c5(h):
    """Return the overlap integral C5 for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx


def enforce_constraints(h, target):
    """Clip to [0,1] and adjust the sum to be exactly *target*."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # simple iterative redistribution, sufficient for the tiny mismatches
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            slack = 1.0 - h
            total = slack.sum()
            if total <= 0:
                break
            inc = np.minimum(slack, diff * slack / total)
            h += inc
        else:
            slack = h
            total = slack.sum()
            if total <= 0:
                break
            dec = np.minimum(slack, -diff * slack / total)
            h -= dec
        diff = target - h.sum()
    return h


def shape_h(d, n):
    """Symmetric piecewise‑linear shape controlled by width *d* (0<d<1)."""
    a = 1.0 / (2.0 - d)            # plateau height, guarantees integral 1
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.empty_like(x)

    left = x < d
    right = x > (2.0 - d)
    middle = ~left & ~right

    # rising slope
    h[left] = a / d * x[left]
    # falling slope (symmetric)
    h[right] = a / d * (2.0 - x[right])
    # plateau
    h[middle] = a

    # enforce exact integral constraint on the discretised grid
    h = enforce_constraints(h, n / 2.0)
    return h


def scan_simple_shapes(n, n_samples=200):
    """Search the 1‑parameter family of shapes for a good starting point."""
    best_h = None
    best_val = np.inf
    for d in np.linspace(0.001, 0.999, n_samples):
        h = shape_h(d, n)
        val = compute_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
    return best_h, best_val


def simulated_annealing(start_h, target_sum, max_iter, start_time, budget_s):
    """Simple SA that moves mass between two coordinates while keeping sum constant."""
    n = start_h.shape[0]
    cur_h = start_h.copy()
    cur_val = compute_c5(cur_h)

    best_h = cur_h.copy()
    best_val = cur_val

    T0 = 0.02  # initial temperature
    for it in range(max_iter):
        # time guard
        if time.time() - start_time > budget_s * 0.94:
            break

        # linear cooling
        T = T0 * (1.0 - it / max_iter)

        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        max_dec = cur_h[i]
        max_inc = 1.0 - cur_h[j]
        if max_dec <= 0.0 or max_inc <= 0.0:
            continue

        # propose a move that respects bounds
        delta = np.random.rand() * min(max_dec, max_inc) * 0.9
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta

        new_val = compute_c5(new_h)
        dE = new_val - cur_val

        if dE < 0.0 or np.random.rand() < np.exp(-dE / max(T, 1e-12)):
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
    return best_h, best_val


def initial_from_parent(n):
    """Interpolate the pre‑provided initial_h_values to length n."""

            best_h[up] += inc
        elif diff < 0 and down.any():
            dec = -diff / down.sum()
            best_h[down] -= dec
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
