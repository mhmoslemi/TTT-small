def compute_d(h, shift):
    n = len(h)
    d = np.zeros(n)
    if shift >= 0:
        d[:n-shift] += h[shift:]
        d[shift:] += h[:n-shift]
    else:
        s = -shift
        d[s:] += h[:n-s]
        d[:n-s] += h[s:]
    return d
```

def random_feasible(n):
    # generate random numbers
    v = np.random.rand(n)
    # scale to sum target
    target = n / 2
    v = v / v.sum() * target
    # clip to 1 maybe; but if some exceed 1 after scaling, maybe need to handle.
    # Simple approach: limit each to 1 and adjust
    v = np.clip(v, 0, 1)
    # adjust sum: if sum less than target due to clipping, add leftover to random indices not at 1.
    # While sum diff > tolerance, assign additional mass.
    diff = target - v.sum()
    while abs(diff) > 1e-12:
        # pick random index not at bound
        idx = np.random.randint(0, n)
        if diff > 0:
            add = min(diff, 1 - v[idx])
            v[idx] += add
            diff -= add
        else:
            sub = min(-diff, v[idx])
            v[idx] -= sub
            diff += sub
    return v
```

                if mx <= 0:
                    continue
                delta = mx * np.random.rand()
                h[i] += delta
                h[j] -= delta
                rev = (i, j, delta)
            else:
                mx = min(1.0 - h[j], h[i])
                if mx <= 0:
                    continue
                delta = mx * np.random.rand()
                h[j] += delta
                h[i] -= delta
                rev = (j, i, delta)
            new_c5, new_prod = _c5_from_product(h, dx)
            if new_c5 <= cur_c5:
                prod = new_prod
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                # revert
                a, b, dlt = rev
                h[a] -= dlt
                h[b] += dlt
                no_improve += 1
            continue

        # random fraction of the allowed amount
        delta = max_delta * np.random.rand()
        h[i_up] += delta
        h[i_down] -= delta

        # evaluate
        new_c5, new_prod = _c5_from_product(h, dx)

        if new_c5 <= cur_c5 + 1e-12:
            # accept move
            prod = new_prod
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # reject
            h[i_up] -= delta
            h[i_down] += delta
            no_improve += 1

        # occasional restart if stuck
        if no_improve >= max_no_improve:
            h = _random_binary_feasible(n)
            cur_c5, prod = _c5_from_product(h, dx)
            no_improve = 0

    # ---- final clean‑up ---------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)

    # enforce exact sum (tiny adjustments)
    deficit = target_sum - best_h.sum()
    if abs(deficit) > 1e-12:
        for _ in range(200):
            idx = np.random.randint(0, n)
            if deficit > 0:
                add = min(deficit, 1.0 - best_h[idx])
                best_h[idx] += add
                deficit -= add
            else:
                sub = min(-deficit, best_h[idx])
                best_h[idx] -= sub
                deficit += sub
            if abs(deficit) < 1e-12:
                break

    final_c5 = _c5_full(best_h, dx)

    return best_h.tolist(), float(final_c5), n
