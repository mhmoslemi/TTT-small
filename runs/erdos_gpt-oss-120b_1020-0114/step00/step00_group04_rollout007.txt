def project_box_sum(v, target, lower=0., upper=1., tol=1e-9, max_iter=100):
    lo = -1e5
    hi = 1e5
    for it in range(max_iter):
        mid = (lo+hi)/2
        w = np.clip(v - mid, lower, upper)
        s = w.sum()
        if abs(s - target) < tol:
            return w
        if s > target:
            # need increase mid (reduce v-mid) -> shift to make w smaller => increase mid
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2, lower, upper)

import time
import random
import numpy as np

def _project_box_sum(v: np.ndarray, target: float,
                     lower: float = 0.0, upper: float = 1.0,
                     tol: float = 1e-9, max_iter: int = 80) -> np.ndarray:
    """
    Projection of v onto the set {w : lower <= w_i <= upper, sum(w)=target}.
    Solution has the form w_i = clip(v_i - lam, lower, upper) for some λ.
    """
    lo = -1e6
    hi = 1e6
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, lower, upper)
        s = w.sum()
        if abs(s - target) <= tol:
            return w
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, lower, upper)


def _upsample(h_old: np.ndarray, n_new: int) -> np.ndarray:
    """Linear up‑sampling of a step function sampled on [0,2)."""
    n_old = h_old.size
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def _c5_of(h: np.ndarray) -> float:
    """Return the overlap constant C5 for a given discretised h."""
    conv = np.convolve(h, (1.0 - h)[::-1])
    return conv.max() * (2.0 / h.size)


def _gradient_qk(h: np.ndarray, k: int) -> np.ndarray:
    """
    Gradient of q_k = sum_i h_i * h_{i+k} (zero‑padded) w.r.t. h.
    For the worst shift k we want to increase q_k, so we move in the direction
    of this gradient.
    """
    n = h.size
    g = np.zeros_like(h)
    if k >= 0:
        g[: n - k] += h[k:]
        g[k:] += h[: n - k]
    else:
        kk = -k
        g[kk:] += h[: n - kk]
        g[: n - kk] += h[kk:]
    return g


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] (discretised) to obtain a low upper bound
    for the Erdős constant C5.

    Returns
    -------
    h_values : np.ndarray
        The best step function found (values in [0,1]).
    c5_bound : float
        Corresponding C5 value.
    n_points : int
        Number of discretisation points used.
    """
    random.seed(seed)
    np.random.seed(seed)

    # -------------------------------------------------------------
    # Parameters
    # -------------------------------------------------------------
    n = 500                     # number of sampling points (feel free to change)
    target_sum = n / 2.0        # integral constraint  ∫ h = 1  ↔  sum = n/2
    max_iters = 200_000        # safety cap
    restart_after = 20_000      # iterations without improvement → restart
    prob_jitter = 0.001         # occasional random acceptance
    base_step = 0.05            # initial step size (will decay)

    # -------------------------------------------------------------
    # Initial solution
    # -------------------------------------------------------------
    try:
        # initial_h_values is provided by the harness if a parent solution exists
        h0 = _upsample(np.array(initial_h_values, dtype=float), n)
    except Exception:
        # start from a random feasible vector
        h0 = np.random.rand(n)

    h0 = _project_box_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = _c5_of(best_h)

    # current working point
    h = best_h.copy()
    cur_c5 = best_c5
    last_improve = 0
    start_time = time.time()
    time_limit = budget_s * 0.95

    iteration = 0
    while iteration < max_iters and (time.time() - start_time) < time_limit:
        # ---------------------------------------------------------
        # 1) Find the shift with minimal autocorrelation q_k
        # ---------------------------------------------------------
        conv_hh = np.convolve(h, h[::-1])               # length 2n-1
        k_bad = h.size - 1 - np.argmin(conv_hh)        # shift giving min q_k

        # ---------------------------------------------------------
        # 2) Gradient step to increase that q_k
        # ---------------------------------------------------------
        grad = _gradient_qk(h, k_bad)

        # decaying step size
        step = base_step / np.sqrt(iteration + 1.0)
        h_candidate = _project_box_sum(h + step * grad, target_sum)

        # ---------------------------------------------------------
        # 3) Evaluate candidate
        # ---------------------------------------------------------
        cand_c5 = _c5_of(h_candidate)

        # ---------------------------------------------------------
        # 4) Acceptance
        # ---------------------------------------------------------
        if cand_c5 < cur_c5:
            h = h_candidate
            cur_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = h_candidate.copy()
                last_improve = iteration
        else:
            # occasional jump to escape plateaus
            if random.random() < prob_jitter:
                h = h_candidate
                cur_c5 = cand_c5

        # ---------------------------------------------------------
        # 5) Restart if stagnated
        # ---------------------------------------------------------
        if iteration - last_improve > restart_after:
            # perturb best solution a little and continue
            h = best_h + 0.05 * (np.random.rand(n) - 0.5)
            h = _project_box_sum(h, target_sum)
            cur_c5 = _c5_of(h)
            last_improve = iteration

        iteration += 1

    # final projection to be safe
    best_h = _project_box_sum(best_h, target_sum)
    best_c5 = _c5_of(best_h)

    return best_h, float(best_c5), int(n)
