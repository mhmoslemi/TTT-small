import time
import numpy as np

def _project_to_feasible(x, target):
    """Clip to [0,1] and adjust so that sum(x) == target."""
    x = np.clip(x, 0.0, 1.0)
    # simple iterative redistribution of the excess/deficit
    for _ in range(10):
        total = x.sum()
        diff = total - target
        if abs(diff) < 1e-12:
            break
        # indices that can still move without hitting bounds
        free = (x > 0.0) & (x < 1.0)
        cnt = free.sum()
        if cnt == 0:               # all at a bound – adjust a single element
            i = np.argmin(np.abs(x - 0.5))
            x[i] = np.clip(x[i] - diff, 0.0, 1.0)
            break
        adj = diff / cnt
        x[free] -= adj
        x = np.clip(x, 0.0, 1.0)
    # final tiny correction on a free entry (if any) to hit the target exactly
    total = x.sum()
    diff = total - target
    if abs(diff) > 1e-10:
        free = (x > 0.0) & (x < 1.0)
        if free.any():
            i = np.flatnonzero(free)[0]
            x[i] = np.clip(x[i] - diff, 0.0, 1.0)
        else:
            # fallback: tweak the first element (it must be interior because sum constraint is feasible)
            i = 0
            x[i] = np.clip(x[i] - diff, 0.0, 1.0)
    return x

def _c5_of(h):
    """Return the C5 value for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _binary_from(h):
    """Return a binary (0/1) vector having exactly floor(n/2) ones, chosen from the largest entries of h."""
    n = h.shape[0]
    k = n // 2
    idx = np.argpartition(-h, k - 1)[:k]
    binary = np.zeros(n, dtype=float)
    binary[idx] = 1.0
    # enforce exact sum (may be off by 1 when n is odd)
    target = n / 2.0
    return _project_to_feasible(binary, target)

def _hillclimb_binary(a, max_iter, time_limit, start_time):
    """Simple greedy hill‑climb on a binary vector."""
    n = a.shape[0]
    best = a.copy()
    best_c5 = _c5_of(best)
    ones = np.flatnonzero(best == 1)
    zeros = np.flatnonzero(best == 0)
    for _ in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        cand = best.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        c5 = _c5_of(cand)
        if c5 < best_c5:
            best = cand
            best_c5 = c5
            ones = np.flatnonzero(best == 1)
            zeros = np.flatnonzero(best == 0)
    return best, best_c5

def _gradient_step(h, eta):
    """One gradient‑ascent step that increases the worst‑case self‑overlap."""
    n = h.shape[0]
    conv = np.correlate(h, h, mode="full")
    idx_min = conv.argmin()
    shift = idx_min - (n - 1)           # shift s achieving minimal overlap
    g = np.zeros_like(h)

    if shift >= 0:
        s = shift
        if s < n:
            g[: n - s] += h[s:]
            g[s:] += h[: n - s]
    else:
        s = -shift
        if s < n:
            g[s:] += h[: n - s]
            g[: n - s] += h[s:]

    h_new = h + eta * g
    target = n / 2.0
    h_new = _project_to_feasible(h_new, target)
    return h_new

def _upsample(base, n):
    """Simple nearest‑neighbour up‑sampling of a shorter vector to length n."""
    if len(base) == n:
        return base.astype(float).copy()
    repeat = n // len(base)
    remainder = n % len(base)
    up = np.repeat(base, repeat)
    if remainder:
        up = np.concatenate((up, base[:remainder]))
    return up[:n].astype(float)
