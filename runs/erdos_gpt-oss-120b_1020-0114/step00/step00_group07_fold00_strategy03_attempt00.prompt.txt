```python
def project_level_down(h, n_orig, n_new):
    """Sampling to half-grid"""
    if n_new > n_orig:
        raise ValueError("Downsampling to fewer points")
    # Reshape and uniform sampling
    segments = []
    for i in range(0,n_orig,2):
        seg_ind = min(i+1, n_orig-1)
        if i == seg_ind: continue
        segments.append((h[i], h[i+1]))
    seg_pairs = int(n_orig/2)
    h_down = []
    for i in range(seg_pairs):
        h_down.append(segments[i][0] + (segments[i][1]-segments[i][0])*0.5)
    
    # Adjust integral
    total = np.sum(h_down)
    desired = n_new / 2  # because dx_new=dx_orig*2, so total应该= n_new/2 (like a normalized sum)
    factor = desired / total if total else 1.0
    h_scaled = np.array([x * factor for x in h_down])
    return h_scaled

def project_level_up(h, n_orig, n_new):
    """Upsample to twice grid points via cubic interpolation"""
    if n_new < n_orig:
        raise ValueError("Upsampling requires larger grid")
    dx_orig = 2.0 / n_orig
    dx_new = 2.0 / n_new
    # Cubic interpolation
    x_orig = np.arange(n_orig) * dx_orig
    # 创建适应插值函数的网格点
    x_new = np.array([i * dx_new for i in range(n_new)]) 
    # 注意: 不同插值要求不同的点集，在固定插值中需要连续点集
    interp_func = interpolate.interp1d(x_orig, h, kind='cubic', bounds_error=False, fill_value=0, assume_sorted=True)
    h_up = interp_func(x_new)
    # Normalization
    total_val = dx_new * np.sum(h_up)
    factor = 1.0 / total_val if total_val else 1.0
    h_normalized = h_up * factor
    return h_normalized
```

```python
def project_coefficients(coeffs, n):
    # Enforce |irfft(coeffs)| <= 1
    h = np.real(np.fft.irfft(coeffs))
    for i in range(len(h)):
        if h[i] > 1:
            coeffs = coeffs - (h[i] - 1) * (np.angle(np.fft.rfft(h)[i]) / (2*np.pi))
        elif h[i] < 0:
            coeffs = coeffs + (-h[i]) * (np.angle(np.fft.rfft(h)[i]) / (2*np.pi))
    return coeffs

def clamp_and_normalize(h, dx):
    total = dx * np.sum(h)
    if abs(total - 1) > 0.001:
        factor = 1 / total
        h = h * factor
    h = np.clip(h, 0, 1)
```
