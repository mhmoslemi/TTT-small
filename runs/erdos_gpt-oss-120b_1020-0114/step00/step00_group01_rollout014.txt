import time, numpy as np, cvxpy as cp

def _c5_of(h):
    """return C5 for a vector h (h values in [0,1])"""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def _solve_mip(n, max_time):
    """solve the binary formulation for a given length n.
    Returns (h_vector, t_value) or (None, None) on failure."""
    # binary decision variables for the function values
    h = cp.Variable(n, boolean=True)
    # t is the maximal overlap deficit (integer but we keep it continuous)
    t = cp.Variable(nonneg=True)

    constraints = [cp.sum(h) == n // 2]

    # for each positive shift we introduce auxiliary products w = h_i * h_{i+s}
    for s in range(1, n):
        length = n - s
        w = cp.Variable(length, boolean=True)
        hi = h[:length]
        hj = h[s:]
        constraints += [
            w <= hi,
            w <= hj,
            w >= hi + hj - 1,
        ]
        # mismatch count for this shift: sum(h) - sum(w) <= t   ->  t >= sum(h)-sum(w)
        constraints.append(t >= cp.sum(h) - cp.sum(w))

    prob = cp.Problem(cp.Minimize(t), constraints)

    # GLPK_MI is fast for modest sizes; fall back to ECOS_BB if unavailable.
    try:
        prob.solve(solver=cp.GLPK_MI, glpk={'tm_lim': int(max_time * 1000)},
                    verbose=False)
    except Exception:                     # pragma: no cover
        prob.solve(solver=cp.ECOS_BB, max_iters=200000,
                    verbose=False)

    if h.value is None or t.value is None:
        return None, None
    h_val = np.rint(np.array(h.value).flatten()).astype(float)   # binary
    return h_val, float(t.value)

def _hill_climb(n, time_budget):
    """simple hill‑climbing on binary vectors of length n."""
    rng = np.random.default_rng()
    # start from a random balanced vector
    idx = rng.choice(n, n // 2, replace=False)
    cur = np.zeros(n, dtype=float)
    cur[idx] = 1.0
    cur_c5 = _c5_of(cur)
    best = cur.copy()
    best_c5 = cur_c5
    start = time.time()
    while time.time() - start < time_budget:
        # propose a swap of a 1 and a 0
        ones = np.flatnonzero(cur == 1.0)
        zeros = np.flatnonzero(cur == 0.0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        nxt = cur.copy()
        nxt[i], nxt[j] = 0.0, 1.0
        nxt_c5 = _c5_of(nxt)
        if nxt_c5 < cur_c5:
            cur, cur_c5 = nxt, nxt_c5
            if nxt_c5 < best_c5:
                best, best_c5 = nxt.copy(), nxt_c5
    return best, best_c5

        # we give the MIP a modest budget (max 30 s per instance)
        mip_time = min(30.0, time_left * 0.3)

        c5_val = _c5_of(h_vec)

    # --- if still not good enough, run a fast hill‑climb -----------------
