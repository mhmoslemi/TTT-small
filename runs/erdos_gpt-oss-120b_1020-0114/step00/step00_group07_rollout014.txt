def optimize_swaps(h0, budget_seconds, start_time):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    target = 0.3808
    it = 0
    max_iter = 500000
    while time.time() - start_time < budget_seconds * 0.9 and it < max_iter:
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        # overlapping indices
        if shift >= 0:
            i_range = np.arange(0, n - shift)
            i_shifted = i_range + shift
        else:
            s = -shift
            i_range = np.arange(s, n)
            i_shifted = i_range - s
        mask = (h[i_range] == 1) & (h[i_shifted] == 0)
        mis_i = i_range[mask]
        if mis_i.size > 0:
            i_swap = mis_i[np.random.randint(mis_i.size)]
        else:
            # choose random one that is 1
            ones_idx = np.where(h == 1)[0]
            i_swap = ones_idx[np.random.randint(len(ones_idx))]
        zeros_idx = np.where(h == 0)[0]
        # filter good zeros: after swap should not increase mismatches for shift
        if shift >= 0:
            good_mask = (i_swap + shift >= n) | (h[zeros_idx + shift] == 1)
        else:
            s = -shift
            good_mask = (zeros_idx - s < 0) | (h[zeros_idx - s] == 1)
        good_zeros = zeros_idx[good_mask]
        if good_zeros.size > 0:
            j_swap = good_zeros[np.random.randint(good_zeros.size)]
        else:
            j_swap = zeros_idx[np.random.randint(len(zeros_idx))]
        # generate new candidate
        h_new = h.copy()
        h_new[i_swap] = 0.0
        h_new[j_swap] = 1.0
        c5_new = compute_c5(h_new)
        # acceptance
        # simple hill climbing with occasional random moves
        if c5_new < current_c5 or np.random.rand() < 0.001:
            h = h_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                if best_c5 <= target:
                    break
        it += 1
    return best_h, best_c5

import numpy as np
import time

# ---------- helpers ---------------------------------------------------------

def compute_c5(h: np.ndarray) -> float:
    """return C5 = max correlation * dx for a step function."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def project_onto_box_simplex(y: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of y onto the set {x | 0<=x_i<=1, sum x_i = target_sum}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(y - lam, 0.0, 1.0)


def subgradient_step(h: np.ndarray, step: float, target_sum: float) -> np.ndarray:
    """one projected sub‑gradient step for the max‑overlap objective."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)

    g = np.zeros_like(h)

    if shift >= 0:
        i = np.arange(0, n - shift)
        g[i] += 1.0 - h[i + shift]
        g[i + shift] -= h[i]
    else:
        s = -shift
        i = np.arange(s, n)
        g[i] += 1.0 - h[i - s]
        g[i - s] -= h[i]

    y = h - step * g
    return project_onto_box_simplex(y, target_sum)


def optimize_subgradient(h0: np.ndarray, rng: np.random.Generator,
                         time_limit: float, start_time: float) -> tuple:
    """projected sub‑gradient descent with a linear step schedule."""
    n = h0.size
    target = n / 2.0
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    h = h0.copy()
    cur_c5 = best_c5

    max_iters = 200_000
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        # linear schedule from 0.10 to 0.001
        t = it / max_iters
        step = 0.10 * (1.0 - t) + 0.001 * t

        h_new = subgradient_step(h, step, target)
        c5_new = compute_c5(h_new)

        # accept improvement; occasional random acceptance to escape flat spots
        if c5_new < cur_c5 or rng.random() < 0.001:
            h, cur_c5 = h_new, c5_new
            if c5_new < best_c5:
                best_h, best_c5 = h_new.copy(), c5_new
                if best_c5 <= 0.3808:
                    break
    return best_h, best_c5


def optimize_swaps(h0: np.ndarray, rng: np.random.Generator,
                   time_limit: float, start_time: float) -> tuple:
    """simple hill‑climbing on binary vectors by swapping a 1 and a 0."""
    n = h0.size
    target = n / 2.0
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    h = h0.copy()
    cur_c5 = best_c5

    max_iters = 500_000
    for _ in range(max_iters):
        if time.time() - start_time > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)

        if shift >= 0:
            i_range = np.arange(0, n - shift)
            i_shift = i_range + shift
        else:
            s = -shift
            i_range = np.arange(s, n)
            i_shift = i_range - s

        mask = (h[i_range] == 1.0) & (h[i_shift] == 0.0)
        mis_i = i_range[mask]

        if mis_i.size:
            i_swap = int(rng.choice(mis_i))
        else:
            ones = np.where(h == 1.0)[0]
            i_swap = int(rng.choice(ones))

        zeros = np.where(h == 0.0)[0]
        # choose a zero that does not create a new mismatch for the worst shift
        if shift >= 0:
            j_plus = zeros + shift
            out_of = j_plus >= n
            good = out_of.copy()
            inside = ~out_of
            good[inside] = h[j_plus[inside]] == 1.0
        else:
            j_plus = zeros + shift          # shift is negative
            out_of = j_plus < 0
            good = out_of.copy()
            inside = ~out_of
            good[inside] = h[j_plus[inside]] == 1.0
        cand = zeros[good]
        if cand.size:
            j_swap = int(rng.choice(cand))
        else:
            j_swap = int(rng.choice(zeros))

        h_new = h.copy()
        h_new[i_swap] = 0.0
        h_new[j_swap] = 1.0

        c5_new = compute_c5(h_new)

        if c5_new < cur_c5 or rng.random() < 0.001:
            h, cur_c5 = h_new, c5_new
            if c5_new < best_c5:
                best_h, best_c5 = h_new.copy(), c5_new
                if best_c5 <= 0.3808:
                    break
    return best_h, best_c5


# ---------- main entry point ------------------------------------------------

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # --------------------------------------------------------------------
    # Determine problem size and a starting vector
    # --------------------------------------------------------------------
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n = init.size
        # binary start: keep the n/2 largest entries as 1
        k = n // 2
        idx = np.argpartition(-init, k - 1)[:k]
        h_binary = np.zeros(n, dtype=float)
        h_binary[idx] = 1.0
    else:
        n = 200
        h_binary = np.zeros(n, dtype=float)
        h_binary[rng.choice(n, n // 2, replace=False)] = 1.0

    # a smooth "tent" start – useful for sub‑gradient optimisation
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    h_tent = np.minimum(x, 2.0 - x)                     # triangular shape
    h_tent = project_onto_box_simplex(h_tent, n / 2.0)

    # pick the better of the two simple starts
    best_h = h_binary if compute_c5(h_binary) <= compute_c5(h_tent) else h_tent
    best_c5 = compute_c5(best_h)

    # --------------------------------------------------------------------
    # Allocate time: half for binary swaps, half for continuous sub‑gradient
    # --------------------------------------------------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining <= 0:
        return best_h, best_c5, n

    half = remaining * 0.5
    h_bin_opt, c5_bin = optimize_swaps(best_h.copy(), rng, half + start_time, start_time)
    if c5_bin < best_c5:
        best_h, best_c5 = h_bin_opt, c5_bin

    remaining = budget_s - (time.time() - start_time)
    if remaining > 0:
        h_sub_opt, c5_sub = optimize_subgradient(best_h.copy(), rng,
                                                remaining + start_time, start_time)
        if c5_sub < best_c5:
            best_h, best_c5 = h_sub_opt, c5_sub

    # final safety projection (removes possible tiny numerical drift)
    best_h = project_onto_box_simplex(best_h, n / 2.0)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
