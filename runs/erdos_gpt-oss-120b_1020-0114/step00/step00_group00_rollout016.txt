import time
import math
import numpy as np

def evaluate_h(h: np.ndarray) -> float:
    """Return the C5 value for a step function h on [0,2] (dx = 2/n)."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h); mode='full' gives all integer shifts
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * float(corr.max())

def compute_gradient(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the maximal overlap term for a given integer shift."""
    n = h.shape[0]
    grad = np.ones(n)
    if shift > 0:
        grad[: n - shift] -= h[shift:]
        grad[shift:] -= h[: n - shift]
    elif shift < 0:
        d = -shift
        grad[d:] -= h[: n - d]
        grad[: n - d] -= h[d:]
    else:  # shift == 0
        grad -= 2.0 * h
    return grad

    n = h.shape[0]
    best_h = h.copy()
    best_c5 = evaluate_h(best_h)

    # pre‑computed constant
    target_sum = n / 2.0
    # small epsilon for picking feasible indices
    eps = 1e-12

    # main loop
    while True:
        now = time.time()
        if now - start_time >= max_seconds:
            break
        # remaining fraction for step‑size/temperature scaling
        remaining = max_seconds - (now - start_time)
        frac = remaining / max_seconds

        # correlation and shift that currently dominates
        corr_full = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(corr_full))
        max_corr = float(corr_full[max_idx])
        curr_c5 = (2.0 / n) * max_corr
        shift = max_idx - (n - 1)

        # gradient for this shift
        grad = compute_gradient(h, shift)

        # indices that can give (or take) mass
        pos_mask = h > eps
        neg_mask = h < 1.0 - eps

        moved = False
        # try a directed move that reduces the dominant term
        if pos_mask.any() and neg_mask.any():
            # candidate to decrease: largest positive gradient where we have mass
            candidate_pos = np.where(pos_mask)[0]
            i_pos = candidate_pos[np.argmax(grad[pos_mask])]
            # candidate to increase: most negative gradient where we have room
            candidate_neg = np.where(neg_mask)[0]
            i_neg = candidate_neg[np.argmin(grad[neg_mask])]

            if grad[i_pos] > grad[i_neg] + 1e-9:
                # step size limited by both bounds and a schedule
                max_step = 0.2 * frac + 1e-4
                delta = min(max_step, h[i_pos], 1.0 - h[i_neg])
                if delta > 0:
                    delta *= np.random.rand()
                    h_new = h.copy()
                    h_new[i_pos] -= delta
                    h_new[i_neg] += delta
                    new_c5 = evaluate_h(h_new)
                    if new_c5 <= curr_c5 + 1e-12:
                        h = h_new
                        curr_c5 = new_c5
                        moved = True
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = h.copy()
        # fallback: simple random transfer
        if not moved:
            # pick random feasible source and sink
            src_candidates = np.where(h > eps)[0]
            dst_candidates = np.where(h < 1.0 - eps)[0]
            if src_candidates.size == 0 or dst_candidates.size == 0:
                break
            i_src = np.random.choice(src_candidates)
            i_dst = np.random.choice(dst_candidates)
            max_step = 0.2 * frac + 1e-4
            delta = min(max_step, h[i_src], 1.0 - h[i_dst])
            if delta <= 0:
                continue
            delta *= np.random.rand()
            h_new = h.copy()
            h_new[i_src] -= delta
            h_new[i_dst] += delta
            new_c5 = evaluate_h(h_new)
            # accept if not worse, or with a tiny probability
            if new_c5 < curr_c5 or np.random.rand() < 0.005:
                h = h_new
                curr_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
        # occasionally renormalise to avoid drift caused by numerical noise
        if (now - start_time) % 30 < 0.01:
            cur_sum = best_h.sum()
            if abs(cur_sum - target_sum) > 1e-9:
                diff = target_sum - cur_sum
                # distribute uniformly over interior points
                interior = (best_h > eps) & (best_h < 1.0 - eps)
                if interior.any():
                    best_h[interior] += diff / interior.sum()
                    best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function h for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    import sys

    start_time = time.time()
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Base function (provided by the harness if available)
    # ------------------------------------------------------------------
    try:
        base_h = np.array(initial_h_values, dtype=np.float64)  # type: ignore
    except Exception:
        # fallback: simple half‑ones, half‑zeros pattern of length 45
        n0 = 45
        base_h = np.concatenate([np.ones(n0 // 2), np.zeros(n0 - n0 // 2)])
        base_h = base_h.astype(np.float64)

    n0 = base_h.shape[0]
    target_sum0 = n0 / 2.0
    # enforce the sum constraint exactly (tiny adjustments if needed)
    cur_sum0 = base_h.sum()
    if abs(cur_sum0 - target_sum0) > 1e-12:
        diff0 = target_sum0 - cur_sum0
        # distribute uniformly over all entries that are not at a bound
        free = (base_h > 0) & (base_h < 1)
        if free.any():
            base_h[free] += diff0 / free.sum()
            base_h = np.clip(base_h, 0.0, 1.0)
        else:
            base_h.fill(0.5)

    # ------------------------------------------------------------------
    # Choose a resolution – a multiple of the base size.
    # Using a larger n gives the optimizer more freedom.
    # ------------------------------------------------------------------
    factor = 4                     # 45 → 180 points (good compromise)
    n = n0 * factor
    h0 = np.repeat(base_h, factor)

    # verify constraints for the up‑sampled start
    target_sum = n / 2.0
    cur_sum = h0.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        # simple uniform correction
        diff = target_sum - cur_sum
        interior = (h0 > 0) & (h0 < 1)
        if interior.any():
            h0[interior] += diff / interior.sum()
            h0 = np.clip(h0, 0.0, 1.0)

    # ------------------------------------------------------------------
    # Run the heuristic search for the remaining time budget.
    # ------------------------------------------------------------------
    remaining_budget = budget_s - (time.time() - start_time)
    # keep a small safety margin
    search_time = max(0.0, remaining_budget - 0.2)

    best_h, best_c5 = search_h(h0, search_time, start_time)

    # final sanity checks / projection
    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = best_h.sum()
    if abs(cur_sum - target_sum) > 1e-8:
        # enforce the exact sum by a simple projection onto the hyperplane
        diff = target_sum - cur_sum
        free = (best_h > 0) & (best_h < 1)
        if free.any():
            best_h[free] += diff / free.sum()
            best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = evaluate_h(best_h)

    # ------------------------------------------------------------------
    # Return the solution
    # ------------------------------------------------------------------
    return best_h, float(final_c5), best_h.shape[0]
