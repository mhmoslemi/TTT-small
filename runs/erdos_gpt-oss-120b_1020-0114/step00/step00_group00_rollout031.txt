    def c5_val(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    best_h = np.clip(h0, 0.0, 1.0)
    # enforce sum exactly (in case of rounding)
    diff = target_sum - best_h.sum()

import math
import numpy as np
from scipy.optimize import minimize

def _c5_value(h, dx):
    """return C5 value for a step vector h (with given dx)."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """Find a step function h on [0,2] with ∫h=1 and 0≤h≤1
    that minimizes   C5 = max_k ∫ h(x)(1-h(x+k))dx .
    Returns (list_of_h, C5_value, n_points)."""
    rng = np.random.RandomState(seed)

    # ------------------------------------------------------------------
    # 1.  Starting vector (use the provided one if it exists)
    # ------------------------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=float)  # type: ignore
    except NameError:
        # fallback: uniform 0.5, length 45
        h0 = np.full(45, 0.5, dtype=float)

    n = len(h0)
    target_sum = n / 2.0          # ∫h = 1  ⇒  sum(h) = n/2
    dx = 2.0 / n

    # make sure the initial vector respects bounds and the sum exactly
    h0 = np.clip(h0, 0.0, 1.0)
    diff = target_sum - h0.sum()
    if abs(diff) > 1e-12:
        # adjust a free component (not on a bound) if possible
        free = np.where((h0 > 0) & (h0 < 1))[0]
        if free.size == 0:
            free = np.arange(n)
        idx = free[0]
        h0[idx] = np.clip(h0[idx] + diff, 0.0, 1.0)

    # ------------------------------------------------------------------
    # 2.  Helper to evaluate C5
    # ------------------------------------------------------------------
    def c5_val(vec):
        return _c5_value(vec, dx)

    # ------------------------------------------------------------------
    # 3.  Local optimisation via SLSQP (equality + box constraints)
    # ------------------------------------------------------------------
    eq_con = {"type": "eq", "fun": lambda x: np.sum(x) - target_sum}
    bounds = [(0.0, 1.0) for _ in range(n)]

    best_h = h0.copy()
    best_val = c5_val(best_h)

    # optimise the provided starting point
    res = minimize(c5_val, best_h, method="SLSQP",
                   bounds=bounds, constraints=[eq_con],
                   options={"maxiter": 500, "ftol": 1e-12, "disp": False})
    if res.success and res.fun < best_val:
        best_h = res.x
        best_val = res.fun

    start_time = time.time()
