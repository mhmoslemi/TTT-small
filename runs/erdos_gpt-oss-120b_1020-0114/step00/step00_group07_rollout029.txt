        # choose indices

        j = rng.integers(0, n-1)
        while j == i:
            j = rng.integers(0, n)
        # compute allowed delta
        max_inc = 1.0 - current_h[i]
        max_dec = current_h[j]
        if max_inc == 0 or max_dec == 0:
            continue
        max_delta = min(max_inc, max_dec)
        # step size scaling
        delta = max_delta * rng.random() * 0.5  # up to half of allowed
        # optionally adjust delta size decreasing: scale by (1-frac)**2
        delta *= (1 - frac)**2
        if delta <= 1e-12:
            continue
        new_h = current_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_val = compute_c5(new_h)
        if new_val < current_val or rng.random() < math.exp(-(new_val - current_val) / (T + 1e-12)):
            current_h = new_h
            current_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
    return best_h, best_val

def gradient_step(h, alpha, step):
    n = len(h)
    g = 1 - h
    corr = np.correlate(h, g, mode='full')
    # compute weights
    max_corr = corr.max()
    exp_vals = np.exp(alpha * (corr - max_corr))  # subtract max for stability
    w = exp_vals / exp_vals.sum()
    # w array length 2n-1 correspond to shifts from -(n-1) to (n-1)
    # compute convolution with w and w_rev
    w_full = w
    # compute back = sum_s w_s * h[i - s]
    # as we derived: back[i] = sum_s w_s * h[i - s]
    # convolution: back = np.convolve(h, w_full, mode='full')[n-1:n-1+n]
    conv_back = np.convolve(h, w_full, mode='full')
    back = conv_back[n-1:n-1+n]
    # forward:
    w_rev = w_full[::-1]
    conv_fwd = np.convolve(h, w_rev, mode='full')
    fwd = conv_fwd[n-1:n-1+n]
    grad = 1.0 - (back + fwd)
    h_new = h - step * grad
    # project onto constraints
    h_new = np.clip(h_new, 0.0, 1.0)
    # ensure sum = target
    target_sum = n / 2.0
    h_new = project_to_box_simplex(h_new, target_sum)
    return h_new, corr.max()
```

import time
import math
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a given step function h."""
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_to_box_simplex(v, target_sum, eps=1e-12, max_iter=60):
    """
    Project vector v onto the intersection of the box [0,1]^n and the affine hyperplane sum(v)=target_sum.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = np.min(v) - 1.0        # ensures all entries become 1 after clipping
    hi = np.max(v) + 1.0        # ensures all entries become 0 after clipping
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
        if hi - lo < eps:
            break
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def make_h_from_p(p, n):
    """Construct a symmetric profile from a power parameter p (0<p≤1)."""
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx          # cell centres
    s = (p + 1.0) / 2.0                     # scaling so that ∫_0^1 φ = 0.5
    h = np.empty_like(xs)
    left = xs <= 1.0
    # first half (0,1]
    h[left] = s * (xs[left] ** p)
    # reflected half (1,2]
    xr = 2.0 - xs[~left]
    h[~left] = 1.0 - s * (xr ** p)
    return h


def gradient_step(h, alpha, step, target_sum):
    """
    One projected gradient step on the smooth approximation
    of the max-overlap objective.
    """
    n = h.size
    # correlation for the current step
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    # soft‑max weights
    maxc = np.max(corr)
    w = np.exp(alpha * (corr - maxc))
    w /= w.sum()                     # shape (2n‑1)

    # two convolutions to obtain Σ_s w_s h[i±s]
    conv_back = np.convolve(h, w, mode="full")
    back = conv_back[n - 1 : n - 1 + n]          # sum_s w_s h[i‑s]

    conv_fwd = np.convolve(h, w[::-1], mode="full")
    fwd = conv_fwd[n - 1 : n - 1 + n]           # sum_s w_s h[i+s]

    grad = 1.0 - (back + fwd)        # ∂/∂h of the smooth objective
    h_new = h - step * grad
    # box‑projection + mass constraint
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_to_box_simplex(h_new, target_sum)
    new_val = np.max(corr) * (2.0 / n)   # actual C5 for this iterate
    return h_new, new_val


def simulated_annealing(h, cur_val, budget, start, target_sum, rng):
    """
    Light‑weight annealing that swaps a small amount of mass between two points.
    Returns the best vector found in the remaining time.
    """
    n = h.size
    best_h = h.copy()
    best_val = cur_val
    temperature0 = 0.02
    while True:
        elapsed = time.time() - start
        if elapsed >= budget - 0.05:
            break
        # linearly decreasing temperature
        t = temperature0 * (1.0 - elapsed / budget)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # how much can we move?
        inc = 1.0 - h[i]
        dec = h[j]
        if inc <= 0 or dec <= 0:
            continue
        # magnitude – a random fraction of the feasible amount
        delta = min(inc, dec) * rng.random() * 0.5 * (1.0 - elapsed / budget) ** 2
        if delta <= 1e-14:
            continue
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_val = compute_c5(h_new)
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (t + 1e-12)):
            h = h_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h_new.copy()
    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points)
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # 1. Choose a modest resolution – enough to capture shape details.
    # ------------------------------------------------------------------
    n = 200                     # must be even; 200 → dx=0.01
    target_sum = n / 2.0

    # ------------------------------------------------------------------
    # 2. Initialise either from a parent construction or from a simple family.
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        if init.size != n:
            # linear interpolation to required resolution
            x_old = (np.arange(init.size) + 0.5) * (2.0 / init.size)
            x_new = (np.arange(n) + 0.5) * (2.0 / n)
            h = np.interp(x_new, x_old, init)
        else:
            h = init.copy()
        h = np.clip(h, 0.0, 1.0)
        h = project_to_box_simplex(h, target_sum)
    else:
        # start from the simple symmetric power‑family with p=1 (the triangle)
        h = make_h_from_p(1.0, n)
        h = project_to_box_simplex(h, target_sum)

    best_h = h.copy()
    best_val = compute_c5(h)

    # ------------------------------------------------------------------
    # 3. Scan a one‑parameter family (power‑shapes) for a good starting point.
    # ------------------------------------------------------------------
    p_grid = np.linspace(0.1, 1.0, 91)        # coarse scan
    for p in p_grid:
        cand = make_h_from_p(p, n)
        cand = project_to_box_simplex(cand, target_sum)
        val = compute_c5(cand)
        if val < best_val:
            best_val = val
            best_h = cand.copy()

    # ------------------------------------------------------------------
    # 4. Refine with a smooth‑gradient descent (soft‑max approximation).
    # ------------------------------------------------------------------
    max_iter = 4000
    step0 = 0.30
    alpha = 30.0
    for it in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.70:      # leave time for SA
            break
        # simple linear schedule for the step size
        step = step0 * (1.0 - it / max_iter) ** 1.2
        best_h, cur_val = gradient_step(best_h, alpha, step, target_sum)
        if cur_val < best_val:
            best_val = cur_val

    # ------------------------------------------------------------------
    # 5. Final short simulated‑annealing polishing phase.
    # ------------------------------------------------------------------
    cur_val = compute_c5(best_h)
    best_h, best_val = simulated_annealing(best_h, cur_val,
                                           budget_s, start_time,
                                           target_sum, rng)

    # ------------------------------------------------------------------
    # 6. Return the best solution found.
    # ------------------------------------------------------------------
    # Ensure the integral constraint is satisfied (tiny numerical drift may appear).
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_box_simplex(best_h, target_sum)

    c5_bound = compute_c5(best_h)
    return best_h, c5_bound, n
