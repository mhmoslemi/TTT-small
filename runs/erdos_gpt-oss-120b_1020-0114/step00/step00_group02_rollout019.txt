# -*- coding: utf-8 -*-
"""
Optimisation for the Erdős minimum overlap constant C₅.
The algorithm performs a projected sub‑gradient descent on the worst
correlation lag and uses random restarts / perturbations to escape local minima.
"""

import time
import math
import random
import numpy as np


def _c5_of(h: np.ndarray) -> float:
    """Return the C₅ value of a feasible step function h."""
    # full correlation of h with (1‑h)
    full = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(full) * (2.0 / h.size))


def _grad_for_lag(h: np.ndarray, k: int) -> np.ndarray:
    """Gradient of the correlation at lag k (the one achieving the max)."""
    n = h.size
    g = np.zeros(n, dtype=float)
    # contributions from left index
    for i in range(n):
        j = i + k
        if 0 <= j < n:
            g[i] += 1.0 - h[j]
        j2 = i - k
        if 0 <= j2 < n:
            g[i] -= h[j2]
    return g


def _local_improve(h: np.ndarray, start: float, budget: float) -> tuple:
    """
    Greedy improvement of h by moving mass from a positive‑gradient index
    to a negative‑gradient one for the current worst lag.
    Returns (best_h, best_c5).
    """
    n = h.size
    best_h = h.copy()
    best_c5 = _c5_of(best_h)

    while True:
        # time guard
        if time.time() - start > budget * 0.97:
            break

        # worst lag
        full = np.correlate(best_h, 1.0 - best_h, mode="full")
        idx = int(np.argmax(full))
        k = idx - (n - 1)

        # gradient for that lag
        g = _grad_for_lag(best_h, k)

        max_g = float(np.max(g))
        if max_g <= 1e-12:          # nothing to reduce on this lag
            break

        i_max = int(np.argmax(g))   # positive gradient → decrease
        i_min = int(np.argmin(g))   # negative gradient → increase

        eps_allowed = min(best_h[i_max], 1.0 - best_h[i_min])
        if eps_allowed <= 1e-12:
            break

        step = min(0.2, eps_allowed)       # start with a reasonable step
        improved = False
        for factor in (1, 0.5, 0.25, 0.125, 0.0625, 0.03125):
            eps = step * factor
            if eps < 1e-12:
                continue
            h_new = best_h.copy()
            h_new[i_max] -= eps
            h_new[i_min] += eps
            c5_new = _c5_of(h_new)
            if c5_new < best_c5 - 1e-12:
                best_h = h_new
                best_c5 = c5_new
                improved = True
                break
        if not improved:
            break

    return best_h, best_c5


def _random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a feasible h with the required sum n/2.
    Starts from a binary half‑set and shuffles it.
    """
    h = np.zeros(n, dtype=float)
    half = n // 2
    h[:half] = 1.0
    if n % 2:
        h[half] = 0.5
    rng.shuffle(h)
    return h
