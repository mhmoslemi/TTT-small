def evaluate_h(h, dx):

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def _dx_from_n(n: int) -> float:
    """Grid spacing for a uniform grid on [0,2] with n points."""
    return 2.0 / n


def _evaluate(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx for the discrete step function."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * _dx_from_n(n)


def _full_corr(h: np.ndarray) -> np.ndarray:
    """Full correlation array (without dx scaling)."""
    return np.correlate(h, 1.0 - h, mode="full")


def _max_shift_index(h: np.ndarray) -> int:
    """Index (relative to full correlation array) attaining the maximum."""
    corr = _full_corr(h)
    return int(np.argmax(corr) - (h.shape[0] - 1))


def _gradient_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """
    Gradient of g_s(h) = Σ_i h_i (1‑h_{i+shift}) w.r.t. h,
    where shift can be negative. The gradient of the *scaled* C5
    (i.e. without the dx factor) is the same up to the same factor.
    """
    n = h.shape[0]
    grad = np.zeros_like(h)
    if shift >= 0:
        s = shift
        if s < n:
            grad[: n - s] += 1.0 - h[s:]
            grad[s:] -= h[: n - s]
    else:
        s = -shift
        if s < n:
            grad[s:] += 1.0 - h[: n - s]
            grad[: n - s] -= h[s:]
    return grad


def _project_box_sum(v: np.ndarray, target_sum: float, tol: float = 1e-12) -> np.ndarray:
    """
    Euclidean projection of v onto the set {x | 0 ≤ x_i ≤ 1, Σ x_i = target_sum }.
    This is a classic water‑filling problem solved by a binary search on the
    Lagrange multiplier λ for the equality Σ clamp(v_i+λ,0,1) = target_sum.
    """
    lo = -np.max(v) - 10.0   # safely low
    hi = 1.0 - np.min(v) + 10.0   # safely high
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    proj = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

    # tiny correction for possible numerical drift
    diff = target_sum - proj.sum()
    if abs(diff) > tol:
        free = (proj > tol) & (proj < 1.0 - tol)
        nf = free.sum()
        if nf > 0:
            proj[free] += diff / nf
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _interpolate_to_new_grid(old_h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a step function from its old grid to a finer grid."""
    old_n = old_h.shape[0]
    # grid points are taken at the left ends of sub‑intervals
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h, left=old_h[0], right=old_h[-1])
    # force feasibility on the finer grid
    target = new_n / 2.0
    new_h = _project_box_sum(new_h, target)
    return new_h


# ----------------------------------------------------------------------
# Core optimizer (projected gradient descent with simple restarts)
# ----------------------------------------------------------------------
def _pgd_optimize(
    h0: np.ndarray,
    time_limit: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Perform projected gradient descent from h0 for at most `time_limit` seconds.
    Returns the best pair (h, C5) discovered.
    """
    n = h0.shape[0]
    target_sum = n / 2.0
    dx = _dx_from_n(n)

    best_h = h0.copy()
    best_val = _evaluate(best_h)

    h = h0.copy()
    alpha = 0.5            # initial step size
    patience = 0
    max_patience = 30

    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        it += 1
        # evaluate current state
        corr = _full_corr(h)
        idx_max = int(np.argmax(corr))
        cur_val = corr[idx_max] * dx
        shift = idx_max - (n - 1)

        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
            patience = 0

        # gradient of the active shift
        grad = _gradient_shift(h, shift)

        # line search (simple backtracking)
        improved = False
        step = alpha
        for _ in range(10):
            cand = _project_box_sum(h - step * grad, target_sum)
            cand_val = _evaluate(cand)
            if cand_val < cur_val - 1e-12:
                h = cand
                improved = True
                break
            step *= 0.5

        if improved:
            alpha = min(alpha * 1.05, 1.0)
            patience = 0
            continue

        # no improvement – try a small stochastic perturbation
        noise = rng.uniform(-0.03, 0.03, size=n)
        cand = _project_box_sum(h + noise, target_sum)
        cand_val = _evaluate(cand)
        if cand_val < cur_val:
            h = cand
            patience = 0
            alpha = max(alpha * 0.9, 1e-3)
            continue

        # stagnation handling
        patience += 1
        alpha = max(alpha * 0.85, 1e-3)
        if patience >= max_patience:
            # restart from the best point seen so far
            h = best_h.copy()
            patience = 0

    return best_h, best_val


def _optimize_multiple_restarts(
    h_start: np.ndarray,
    total_time: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Run several independent projected‑gradient attempts within `total_time`.
    Each attempt begins from a perturbation of `h_start`.
    """
    n = h_start.shape[0]
    target_sum = n / 2.0

    best_h = None
    best_val = np.inf
    start_all = time.time()

    # allocate time slices (≈ 0.2 s each attempt) but respect total budget
    while time.time() - start_all < total_time:
        remaining = total_time - (time.time() - start_all)
        # very short slice for a single attempt
        slice_time = min(0.25, remaining)
        # create a perturbed start point
        h0 = h_start + rng.normal(scale=0.1, size=n)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = _project_box_sum(h0, target_sum)

        h_opt, val = _pgd_optimize(h0, slice_time, rng)

        if val < best_val:
            best_val = val
            best_h = h_opt.copy()

        # if we have very little time left, break
        if remaining < 0.3:
            break

    # fallback – if nothing succeeded, return the (projected) start point
    if best_h is None:
        best_h = _project_box_sum(h_start, target_sum)
        best_val = _evaluate(best_h)

    return best_h, best_val

# Main entry point required by the harness
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Returns a feasible step function h, its C5 bound, and the grid size.
    The algorithm works in two stages:
      1. Optimise on the original grid size (taken from `initial_h_values`
         when present, otherwise 70 points).
      2. Upscale to twice the resolution and continue optimisation.
    """
    rng = np.random.default_rng(seed)
    t0 = time.time()

    # ------------------------------------------------------------------
    # Stage 1 – base grid
    # ------------------------------------------------------------------
    init = globals().get("initial_h_values")
    if init is not None:
        h_base = np.asarray(init, dtype=float)
        n_base = h_base.shape[0]
    else:
        n_base = 70
        h_base = np.full(n_base, 0.5, dtype=float)

    # enforce exact feasibility on the base grid
    h_base = _project_box_sum(h_base, n_base / 2.0)

    # allocate a generous (but safe) slice of the budget for stage 1
    time_for_stage1 = min(400.0, budget_s * 0.35)
    h_opt1, val1 = _optimize_multiple_restarts(h_base, time_for_stage1, rng)

    # ------------------------------------------------------------------
    # Stage 2 – refined grid (double the points, if still within budget)
    # ------------------------------------------------------------------
    n_refined = min(1000, n_base * 2)          # keep < 1000 points as requested
    h_refined_start = _interpolate_to_new_grid(h_opt1, n_refined)

    # remaining time (leave a small safety margin)
    time_spent = time.time() - t0
    time_remaining = max(0.0, budget_s - time_spent - 0.5)
    # use at most half of the remaining budget for the refined optimisation
    time_for_stage2 = min(time_remaining, budget_s * 0.5)

    h_opt2, val2 = _optimize_multiple_restarts(h_refined_start, time_for_stage2, rng)

    # ------------------------------------------------------------------
    # Choose the best of the two stages
    # ------------------------------------------------------------------
    if val2 < val1:
        best_h = h_opt2
        best_val = val2
        best_n = n_refined
    else:
        best_h = h_opt1
        best_val = val1
        best_n = n_base

    # final safety projection (tiny numerical drift)
    best_h = _project_box_sum(best_h, best_n / 2.0)

    # return as required by the harness
    return best_h, float(best_val), best_n
