def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
```

def hill_climb_binary(n, max_iters, time_limit, best_val):
    h = np.zeros(n, dtype=np.float64)
    # assign first n/2 ones randomly
    indices = np.arange(n)
    np.random.shuffle(indices)
    ones_idx = indices[:n//2]
    h[ones_idx] = 1.0

    current_val = compute_c5(h)
    best_local_val = current_val
    best_local_h = h.copy()

    # maintain lists of one indices and zero indices
    ones = set(ones_idx.tolist())
    zeros = set(np.where(h==0)[0].tolist())
    # Replace sets with lists for random choice.
    ones_list = list(ones)
    zeros_list = list(zeros)

    iter = 0
    while iter < max_iters and (time.time() - start_time) < time_limit:
        iter += 1
        i = random.choice(ones_list)
        j = random.choice(zeros_list)
        # perform swap
        h[i], h[j] = 0.0, 1.0
        new_val = compute_c5(h)
        if new_val < current_val:
            # accept improvement
            current_val = new_val
            # update best local
            if new_val < best_local_val:
                best_local_val = new_val
                best_local_h = h.copy()
                # update global best if needed
                if new_val < best_val:
                    best_val = new_val
                    best_h = h.copy()
                    best_n = n

def construct_h_from_params(n, a, b, h1, h2):
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    h = np.zeros(n)
    # outer region: x < a
    mask0 = (x < a)
    # inner region: a <= x < a+b
    mask1 = (x >= a) & (x < a + b)
    # middle region: a+b <= x < 2 - (a+b)
    mask2 = (x >= a + b) & (x < 2 - (a + b))
    # inner right: (2 - (a + b)) <= x < 2 - a
    mask3 = (x >= 2 - (a + b)) & (x < 2 - a)

    mask4 = (x >= 2 - a)

        return None

def random_search(n, max_iters, time_limit):
    best_val, best_h = inf, None
    start = time.time()
    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        a = np.random.rand()  # [0,1]
        b = np.random.rand() * max(0.0, 1.0 - a)    # ensure a+b <= 1
        # can also sample b uniformly in [0, 1-a]
        # compute c = 2 - 2*(a+b)
        c = 2 - 2*(a+b)
        if c <= 1e-6:
            continue
        h1 = np.random.rand()
        h2 = np.random.rand()
        h3 = (1 - 2*a*h1 - 2*b*h2) / c
        if not (0 <= h3 <= 1):
            continue
        h = np.zeros(n)
        x = (np.arange(n) + 0.5) * (2.0 / n)
        # assign
        mask0 = x < a
        mask1 = (x >= a) & (x < a+b)
        mask2 = (x >= a+b) & (x < 2 - (a+b))
        mask3 = (x >= 2 - (a+b)) & (x < 2 - a)
        mask4 = x >= 2 - a
        h[mask0] = h1
        h[mask1] = h2
        h[mask2] = h3
        h[mask3] = h2
        h[mask4] = h1
        val = compute_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
    return best_h, best_val
```

import numpy as np
import time
import random

def compute_c5(h):
    """Return the Erdős C5 value for a step function h given as a numpy array."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx


def build_h(n, a, b, h1, h2):
    """
    Build a symmetric 5‑interval step function on [0,2] with
        - outer interval length a (each side) and height h1,
        - next interval length b (each side) and height h2,
        - middle interval length c = 2-2*a-2*b and height h3 chosen to satisfy the integral.
    Returns None if the parameters are infeasible.
    """
    c = 2.0 - 2.0 * (a + b)
    if c <= 0:
        return None
    # height of the middle interval from the area constraint
    h3 = (1.0 - 2.0 * a * h1 - 2.0 * b * h2) / c
    if not (0.0 <= h3 <= 1.0):
        return None

    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx  # mid‑point coordinates
    h = np.empty(n, dtype=np.float64)

    mask0 = x < a
    mask1 = (x >= a) & (x < a + b)
    mask2 = (x >= a + b) & (x < 2.0 - (a + b))
    mask3 = (x >= 2.0 - (a + b)) & (x < 2.0 - a)
    mask4 = x >= 2.0 - a

    h.fill(0.0)
    h[mask0] = h1
    h[mask1] = h2
    h[mask2] = h3
    h[mask3] = h2
    h[mask4] = h1

    # tiny correction to guarantee the integral exactly equals 1
    total = h.sum()
    target = n / 2.0
    if total <= 0:
        return None
    factor = target / total
    if factor < 0.999 or factor > 1.001:
        if factor * h.max() > 1.0:
            return None
        h = h * factor
    return h


def random_search(n, best_h, best_val, start, deadline):
    """Stochastic search over the 5‑interval parameter space."""
    while time.time() < deadline:
        # choose interval lengths (a,b) with a+b <= 1
        a = np.random.rand()
        b = np.random.rand() * (1.0 - a)

        # choose heights for the outer and inner intervals
        h1 = np.random.rand()
        h2 = np.random.rand()

        h = build_h(n, a, b, h1, h2)
        if h is None:
            continue

        val = compute_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
    return best_h, best_val


def local_refine(h, val, n, start, deadline):
    """Simple hill‑climbing on the vector itself (swap mass between two points)."""
    while time.time() < deadline:
        i, j = random.sample(range(n), 2)
        # move a small amount of mass from j to i (keeping bounds)
        max_transfer = min(0.02, 1.0 - h[i], h[j])
        if max_transfer <= 0:
            continue
        delta = np.random.uniform(0.0, max_transfer)
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        # evaluate
        new_val = compute_c5(h_new)
        if new_val < val:
            h, val = h_new, new_val
    return h, val


def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] with values in [0,1] that minimizes
    C5 = max_k ∫ h(x)(1‑h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    # default: use the provided initial construction if present
    best_h = None
    best_val = float('inf')
    best_n = None

    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=np.float64)
        init_val = compute_c5(init)
        best_h, best_val, best_n = init, init_val, init.size

    # we work mostly with n = 200 (enough resolution, still fast)
    n = 200
    # reserve most of the time for the stochastic search, keep a margin for refinement
    search_deadline = start_time + budget_s * 0.85
    refine_deadline = start_time + budget_s * 0.95

    # ----- random exploration over the 5‑interval family -----
    best_h, best_val = random_search(n, best_h, best_val, start_time, search_deadline)

    # ----- optional local refinement on the best vector found -----
    if best_h is not None and time.time() < refine_deadline:
        best_h, best_val = local_refine(best_h, best_val, n, start_time, refine_deadline)

    # final safety check – enforce integral exactly (tiny scaling if needed)
    if best_h is not None:
        total = best_h.sum()
        target = n / 2.0
        if abs(total - target) > 1e-8:
            factor = target / total
            # scaling never pushes values above 1 because factor ≤ 1 for all feasible cases
            best_h = best_h * factor

        # recompute the bound after the final adjustment
        best_val = compute_c5(best_h)

    return best_h, best_val, n
