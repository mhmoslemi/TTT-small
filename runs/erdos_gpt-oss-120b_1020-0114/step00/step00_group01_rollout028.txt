def local_search(n, max_iters, time_limit):
    # random initialization
    h = np.zeros(n, dtype=np.int8)
    idx = np.arange(n)
    np.random.shuffle(idx)
    h[idx[:n//2]] = 1
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    it = 0
    temp = initial_temp = 0.5
    while time.time() - start < time_limit and it < max_iters:
        # propose swap
        i = np.random.randint(n)
        while h[i] == 0:
            i = np.random.randint(n)
        j = np.random.randint(n)
        while h[j] == 1:
            j = np.random.randint(n)
        # perform swap
        h[i] = 0
        h[j] = 1
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-9 or (np.random.rand() < np.exp(-(c5 - best_c5)/temp)):
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
        # cooling
        if it % 1000 == 0:
            temp *= 0.99
        it += 1
    return best_h, best_c5
```

def improve_h(h, target_c5, end_time):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_h = h.copy()
    while time.time() < end_time:
        # compute current correlation
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        max_corr = corr.max()
        # get shift
        lag = int(np.argmax(corr) - (n - 1))

        idx = np.arange(n)

            j = np.random.choice(j_cand)

            j = np.random.choice(np.where(current_h == 0)[0])

        current_h[i], current_h[j] = current_h[j], current_h[i]

        new_corr = np.correlate(current_h, 1 - current_h, mode='full')

import time
import numpy as np

def _c5_of(h: np.ndarray) -> float:
    """Return the C5 value for a step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation h with (1‑h) using zero padding (np.correlate mode='full')
    max_corr = np.correlate(h, 1.0 - h, mode="full").max()
    return float(max_corr * dx)


def _improve_binary(h: np.ndarray, end_time: float, target: float) -> tuple:
    """
    Hill‑climbing / simulated‑annealing improvement for a binary vector.
    Keeps exactly half of the entries equal to 1.
    """
    n = h.shape[0]
    half = n // 2
    best_h = h.copy()
    best_c5 = _c5_of(best_h)
    cur_h = h.copy()

    # pre‑compute index array once
    idx = np.arange(n)

    while time.time() < end_time:
        # current correlation and the lag giving the maximum value
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        max_corr = corr.max()
        lag = int(np.argmax(corr) - (n - 1))

        # try a swap that attacks the worst lag
        valid = (idx + lag >= 0) & (idx + lag < n)
        i_cand = idx[valid & (cur_h == 1) & (cur_h[idx + lag] == 0)]
        j_cand = idx[valid & (cur_h == 0) & (cur_h[idx + lag] == 1)]

        if i_cand.size and j_cand.size:
            i = np.random.choice(i_cand)
            j = np.random.choice(j_cand)
        else:
            # fall back to a random 1/0 swap
            i = np.random.choice(np.where(cur_h == 1)[0])
            j = np.random.choice(np.where(cur_h == 0)[0])

        # perform the swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

        # evaluate new max correlation
        new_max = np.correlate(cur_h, 1.0 - cur_h, mode="full").max()

        # keep the move only if it does not increase the worst correlation
        if new_max <= max_corr:
            cur_c5 = new_max * (2.0 / n)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
                if best_c5 <= target:
                    break
        else:
            # revert the change
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

    return best_h, best_c5


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Return a step function h, its C5 bound, and the number of sample points.
    The routine first uses any supplied `initial_h_values`; if they are binary
    with an even length it tries a quick local optimisation, otherwise it
    falls back to a random binary construction.
    """
    np.random.seed(seed)
    start = time.time()
    target_c5 = 0.3808        # target we would like to reach

    # ------------------------------------------------------------------
    # 1. obtain an initial candidate
    # ------------------------------------------------------------------
    try:
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n0 = h0.size
    except Exception:          # no initial_h_values supplied
        h0 = None
        n0 = None

    # If we have a binary even‑length starter, keep it; otherwise build one.
    if h0 is not None and n0 % 2 == 0 and np.allclose(h0, np.round(h0)):
        h = h0.copy()
        n = n0
    else:
        # choose a modest even size that fits comfortably in the budget
        n = 200 if (budget_s > 30) else 100
        h = np.zeros(n, dtype=np.float64)
        perm = np.arange(n)
        np.random.shuffle(perm)
        h[perm[: n // 2]] = 1.0

    # ------------------------------------------------------------------
    # 2. ensure the integral constraint (sum(h) * dx == 1)
    # ------------------------------------------------------------------
    dx = 2.0 / n
    needed = n / 2.0
    cur_sum = h.sum()
    if not np.isclose(cur_sum, needed):
        # adjust the first component while staying in [0,1]
        delta = needed - cur_sum
        h[0] = np.clip(h[0] + delta, 0.0, 1.0)

    # ------------------------------------------------------------------
    # 3. optional local optimisation (only for binary vectors)
    # ------------------------------------------------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0 and np.allclose(h, np.round(h)):
        # give most of the remaining budget to a single improvement run
        end = time.time() + remaining * 0.95
        h_opt, c5_opt = _improve_binary(h, end, target_c5)
    else:
        h_opt = h
        c5_opt = _c5_of(h_opt)

    # ------------------------------------------------------------------
    # 4. final check – if time still left, try a few random restarts
    # ------------------------------------------------------------------
    while time.time() - start < budget_s * 0.97:
        # quick random binary restart
        n2 = n
        h2 = np.zeros(n2, dtype=np.float64)
        perm = np.arange(n2)
        np.random.shuffle(perm)
        h2[perm[: n2 // 2]] = 1.0
        end = time.time() + (budget_s - (time.time() - start)) * 0.1
        h2_opt, c5_2 = _improve_binary(h2, end, target_c5)
        if c5_2 < c5_opt:
            h_opt, c5_opt = h2_opt, c5_2
            if c5_opt <= target_c5:
                break

    return (h_opt, float(c5_opt), n)
