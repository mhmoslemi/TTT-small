def hill_climbing(N, init_h, rng, max_iter, best_global):
    h = init_h.copy()
    cur_best = evaluate(h)
    best_h_local = h.copy()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    for it in range(max_iter):
        # random swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # propose swap
        h[i], h[j] = 0, 1  # set
        new_val = evaluate(h)
        if new_val < cur_best:
            cur_best = new_val
            best_h_local = h.copy()
            # update ones and zeros sets
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
        else:
            # revert swap
            h[i], h[j] = 1, 0
        # update global best if improved
        if cur_best < best_global[0]:
            best_global[:] = [cur_best, h.copy(), N]

    cur_h = h.copy()
    cur_val = compute_c5(cur_h)
    best_local = cur_val
    best_h_local = cur_h.copy()
    ones = list(np.where(cur_h == 1)[0])
    zeros = list(np.where(cur_h == 0)[0])
    for it in range(max_iter):
        if time.time() >= time_deadline: break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0.0; cur_h[j] = 1.0
        new_val = compute_c5(cur_h)
        if new_val < cur_val:
            cur_val = new_val
            # update lists
            ones.remove(i); zeros.append(i)
            zeros.remove(j); ones.append(j)
            if cur_val < best_local:
                best_local = cur_val
                best_h_local = cur_h.copy()
            # maybe continue
        else:
            # revert
            cur_h[i] = 1.0; cur_h[j] = 0.0
        # optional early stop if cur_val equals theoretical lower bound (maybe zero?) not needed
    return best_h_local, best_local

def hillclimb_binary(h, rng, deadline):
    N = len(h)
    cur_h = h.copy()
    cur_val = compute_max_overlap(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    # get indices of 1 and 0
    ones = list(np.flatnonzero(cur_h > 0.5))
    zeros = list(np.flatnonzero(cur_h <= 0.5))
    # guarantee lengths half
    while time.time() < deadline and best_val > 0:  # while time left
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_val = compute_max_overlap(cur_h)
        if new_val <= cur_val:
            cur_val = new_val
            # update lists
            ones.remove(i); zeros.append(i)
            zeros.remove(j); ones.append(j)
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
                # early exit if best_val <= target (maybe 34)
                if best_val * (2.0 / N) <= 0.3808:
                    return best_h, best_val
        else:
            # revert
            cur_h[i] = 1.0
            cur_h[j] = 0.0
    return best_h, best_val

import time, math
import numpy as np

def _max_overlap(h: np.ndarray) -> float:
    """Maximum of sum_i h[i]*(1-h[i+shift]) over all integer shifts."""
    return np.max(np.correlate(h, 1.0 - h, mode="full"))

def _c5_value(h: np.ndarray) -> float:
    """C5 bound for a step function represented by array h."""
    n = h.shape[0]
    return _max_overlap(h) * (2.0 / n)

def _hillclimb_binary(start_h: np.ndarray,
                      rng: np.random.Generator,
                      deadline: float,
                      max_no_improve: int = 3000) -> tuple[np.ndarray, float]:
    """
    Simple greedy hill‑climb on binary vectors.
    Swaps a 1 with a 0 and accepts the swap if it does not increase the
    maximal overlap. Stops after `max_no_improve` consecutive non‑improving
    attempts or when the time limit is reached.
    Returns the best vector found and its *unscaled* maximal overlap.
    """
    n = start_h.shape[0]
    cur = start_h.copy()
    cur_val = _max_overlap(cur)
    best = cur.copy()
    best_val = cur_val

    ones = list(np.flatnonzero(cur == 1))
    zeros = list(np.flatnonzero(cur == 0))
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        if not ones or not zeros:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        # propose swap
        cur[i] = 0.0
        cur[j] = 1.0
        new_val = _max_overlap(cur)
        if new_val <= cur_val:
            # accept
            cur_val = new_val
            ones.remove(i)
            zeros.append(i)
            zeros.remove(j)
            ones.append(j)
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best = cur.copy()
        else:
            # revert
            cur[i] = 1.0
            cur[j] = 0.0
            no_improve += 1
    return best, best_val
