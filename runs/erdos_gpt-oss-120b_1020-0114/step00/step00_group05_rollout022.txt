import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper: exact C5 evaluation (fallback if evaluate_erdos_solution missing)
def _exact_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


# ----------------------------------------------------------------------
# Soft‑max of the correlation together with its weighted gradient
def _softmax_corr_and_weights(corr, beta):
    """Return (softmax value, weights) for a vector of correlations."""
    max_corr = np.max(corr)
    # stable log‑sum‑exp
    exp_term = np.exp(beta * (corr - max_corr))
    sum_exp = np.sum(exp_term)
    soft = max_corr + np.log(sum_exp) / beta
    weights = exp_term / sum_exp
    return soft, weights


def _gradient_corr(h, weights):
    """Weighted gradient of all correlation entries w.r.t. h."""
    n = h.shape[0]
    grad = np.zeros_like(h)

    # shift runs from -(n-1) .. (n-1)
    for idx, shift in enumerate(range(-(n - 1), n)):
        w = weights[idx]
        if shift >= 0:
            # part where i+shift is inside the array
            limit = n - shift
            # G[i] += w * (1 - h[i+shift])   for i = 0 .. limit-1
            grad[:limit] += w * (1.0 - h[shift:])
            # G[i] -= w * h[i-shift]   for i = shift .. n-1
            grad[shift:] -= w * h[:limit]
        else:
            t = -shift
            limit = n - t
            # G[i] += w * (1 - h[i-shift])   for i = t .. n-1  (i-shift = i-t)
            grad[t:] += w * (1.0 - h[:limit])
            # G[i] -= w * h[i-shift]   for i = 0 .. limit-1   (i-shift = i+t)
            grad[:limit] -= w * h[t:]
    return grad


# ----------------------------------------------------------------------
# Objective and gradient (soft‑max approximation of max overlap)
def _obj_and_grad(h, beta, penalty):
    n = h.shape[0]
    dx = 2.0 / n
    target_sum = n / 2.0

    corr = np.correlate(h, 1.0 - h, mode="full")
    soft, weights = _softmax_corr_and_weights(corr, beta)

    # penalty term for the integral constraint
    sum_err = np.sum(h) - target_sum
    pen = penalty * sum_err * sum_err
    value = dx * soft + pen

    # gradient
    grad_corr = _gradient_corr(h, weights)
    grad = dx * grad_corr + 2.0 * penalty * sum_err * np.ones_like(h)
    return value, grad


# ----------------------------------------------------------------------
def _objective(h, beta, penalty):
    return _obj_and_grad(h, beta, penalty)[0]


def _gradient(h, beta, penalty):
    return _obj_and_grad(h, beta, penalty)[1]
