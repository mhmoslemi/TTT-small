import numpy as np, time, random, math

def neighbor_sum(h, idx):
    n = len(h)
    L = 2*n - 1
    ns = np.zeros(L, dtype=np.float64)
    # left contribution: h[idx + shift]
    start = n - 1 - idx
    ns[start:start + n] = h
    # right contribution: h[idx - shift] -> horizon symmetrical reversed
    rev_idx = (idx + n - 1) - np.arange(n)
    ns[rev_idx] += h
    return ns
```

import time
import random
import math
import numpy as np

def neighbor_sum(h: np.ndarray, idx: int) -> np.ndarray:
    """Return vector s[l] = h[idx+l] + h[idx-l] with zero padding.
    Length of returned vector = 2*len(h)-1, l = -(n-1)..(n-1)."""
    n = h.shape[0]
    L = 2 * n - 1
    s = np.zeros(L, dtype=np.float64)

    # contribution h[idx + l]
    start = n - 1 - idx                     # position of l = 0 in the full vector
    s[start:start + n] = h                  # fills l = -idx .. (n-1-idx)

    # contribution h[idx - l]  (the reversed copy)
    rev_idx = (idx + n - 1) - np.arange(n)   # l = idx - t  => index = (idx+n-1)-t
    s[rev_idx] += h
    return s


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    random.seed(seed)

    # ----- initial vector -------------------------------------------------
    if 'initial_h_values' in globals():

    if abs(cur_sum - target_sum) > 1e-12:

    dx = 2.0 / n

    # current state

    # best ever
