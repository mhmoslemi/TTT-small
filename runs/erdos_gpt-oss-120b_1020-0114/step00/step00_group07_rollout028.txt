def hill_climb(start_h, rng, time_budget, start_time):
    n = len(start_h)
    cur_h = start_h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    T = 0.1
    decay = 0.9995

    # compute initial worst shift & contributions
    while time.time() - start_time < time_budget:
        max_val, shift, _ = max_shift_info(cur_h)
        # targeted move
        moved = False
        if shift != 0:
            if shift > 0:
                h_i = cur_h[:n-shift]
                h_j = cur_h[shift:]
                contrib = h_i * (1 - h_j)
                idx = np.argmax(contrib)
                i = idx
                j = idx + shift
            else:
                s = -shift
                h_i = cur_h[s:]
                h_j = cur_h[:n-s]
                contrib = h_i * (1 - h_j)
                idx = np.argmax(contrib)
                i = idx + s
                j = idx
            # Determine delta capacity
            delta_max = min(cur_h[i], 1 - cur_h[j])
            if delta_max > 0:
                delta = delta_max * 0.3 * rng.random() # maybe 30% of capacity
                new_h = cur_h.copy()
                new_h[i] -= delta
                new_h[j] += delta
                new_c = compute_c5(new_h)
                # accept if better or by SA
                if new_c < cur_c - 1e-12 or rng.random() < math.exp(-(new_c - cur_c) / T):
                    cur_h = new_h
                    cur_c = new_c
                    moved = True
                    if cur_c < best_c:
                        best_h = cur_h.copy()
                        best_c = cur_c
        # if not moved, do random transfer move
        if not moved:
            # random pair
            i, j = rng.integers(n, size=2, replace=False)
            # compute capacities both directions
            capacity_i_to_j = min(cur_h[i], 1 - cur_h[j])
            capacity_j_to_i = min(cur_h[j], 1 - cur_h[i])
            # choose direction with larger capacity or random if both zero
            if capacity_i_to_j > capacity_j_to_i:
                if capacity_i_to_j > 0:
                    delta = capacity_i_to_j * 0.3 * rng.random()
                    new_h = cur_h.copy()
                    new_h[i] -= delta
                    new_h[j] += delta
                else:
                    continue
            else:
                if capacity_j_to_i > 0:
                    delta = capacity_j_to_i * 0.3 * rng.random()
                    new_h = cur_h.copy()
                    new_h[i] += delta
                    new_h[j] -= delta
                else:
                    continue
            new_c = compute_c5(new_h)
            if new_c < cur_c - 1e-12 or rng.random() < math.exp(-(new_c - cur_c) / T):
                cur_h = new_h
                cur_c = new_c
                if cur_c < best_c:
                    best_h = cur_h.copy()
                    best_c = cur_c
        # reduce temperature
        T = max(T * decay, 1e-6)
    return best_h, best_c

def max_shift_info(h):
    corr = np.correlate(h, 1 - h, mode='full')
    idx = corr.argmax()
    max_val = corr[idx]
    shift = idx - (len(h) - 1)
    return max_val, shift, corr

def project_to_box_sum(v, target):
    # water-filling binary search
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # after rounding maybe sum not equal target within tiny epsilon; optionally adjust by adding to non-bound
    diff = target - w.sum()
    if abs(diff) > 1e-9:
        # distribute diff equally among entries not at bounds
        mask = (w > 1e-12) & (w < 1 - 1e-12)
        if mask.any():
            w[mask] += diff / mask.sum()
            w = np.clip(w, 0.0, 1.0)
    return w

import numpy as np
import time
import math

def _compute_c5(h):
    """Return C5 for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx

def _max_shift_info(h):
    """Return (max_corr, shift, full_corr) for vector h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(corr.argmax())
    shift = idx - (n - 1)
    return float(corr[idx]), shift, corr

def _project_to_box_sum(v, target):
    """Project v onto [0,1]^n with sum(target). Uses water‑filling."""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction if rounding left a tiny error
    diff = target - w.sum()
    if abs(diff) > 1e-9:
        mask = (w > 1e-12) & (w < 1 - 1e-12)
        if mask.any():
            w[mask] += diff / mask.sum()
            w = np.clip(w, 0.0, 1.0)
    return w

def _hill_climb(start_h, rng, deadline):
    """Simulated‑annealing style local search."""
    n = start_h.shape[0]
    cur_h = start_h.copy()
    cur_c = _compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    T = 0.12                     # initial temperature
    decay = 0.9996               # exponential cooling
    min_step = 1e-5

    while time.time() < deadline:
        # identify the worst shift
        _, shift, _ = _max_shift_info(cur_h)

        moved = False
        if shift != 0:
            # contributions for the worst shift
            if shift > 0:
                hi = cur_h[: n - shift]
                hj = cur_h[shift:]
                contrib = hi * (1.0 - hj)
                idx = int(contrib.argmax())
                i, j = idx, idx + shift
            else:
                s = -shift
                hi = cur_h[s:]
                hj = cur_h[: n - s]
                contrib = hi * (1.0 - hj)
                idx = int(contrib.argmax())
                i, j = idx + s, idx          # i > j because shift is negative

            # transfer from i (higher) to j (lower) if possible
            delta_max = min(cur_h[i], 1.0 - cur_h[j])
            if delta_max > min_step:
                delta = delta_max * 0.35 * rng.random()
                new_h = cur_h.copy()
                new_h[i] -= delta
                new_h[j] += delta
                new_c = _compute_c5(new_h)
                if new_c < cur_c - 1e-12 or rng.random() < math.exp(-(new_c - cur_c) / max(T, 1e-12)):
                    cur_h, cur_c = new_h, new_c
                    moved = True
                    if new_c < best_c:
                        best_h, best_c = new_h.copy(), new_c

        if not moved:
            # random pairwise transfer
            i, j = rng.integers(n, size=2, endpoint=False)
            while i == j:
                j = rng.integers(n)
            cap_i_to_j = min(cur_h[i], 1.0 - cur_h[j])
            cap_j_to_i = min(cur_h[j], 1.0 - cur_h[i])
            if cap_i_to_j > cap_j_to_i:
                if cap_i_to_j > min_step:
                    delta = cap_i_to_j * 0.30 * rng.random()
                    new_h = cur_h.copy()
                    new_h[i] -= delta
                    new_h[j] += delta
                else:
                    continue
            else:
                if cap_j_to_i > min_step:
                    delta = cap_j_to_i * 0.30 * rng.random()
                    new_h = cur_h.copy()
                    new_h[i] += delta
                    new_h[j] -= delta
                else:
                    continue
            new_c = _compute_c5(new_h)
            if new_c < cur_c - 1e-12 or rng.random() < math.exp(-(new_c - cur_c) / max(T, 1e-12)):
                cur_h, cur_c = new_h, new_c
                if new_c < best_c:
                    best_h, best_c = new_h.copy(), new_c

        T = max(T * decay, 1e-8)

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretised) that minimises C5.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # initialise the vector (use the parent if present)
    # ------------------------------------------------------------------
    init = globals().get("initial_h_values")
    if init is not None:
        h0 = np.asarray(init, dtype=float).copy()
        n = h0.shape[0]
        # enforce constraints exactly
        h0 = _project_to_box_sum(h0, n / 2.0)
    else:
        n = 80                         # default size when no parent is given
        h0 = np.zeros(n)
        h0[rng.choice(n, n // 2, replace=False)] = 1.0   # balanced binary start

    best_h, best_c = h0.copy(), _compute_c5(h0)

    # ------------------------------------------------------------------
    # main optimisation loop – multiple restarts with time sharing
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s - 4.0:
        remaining = budget_s - (time.time() - start_time)
        # spend at most 30 s on a single local search
        search_time = min(30.0, remaining * 0.4)
        deadline = time.time() + search_time

        # create a start point: either a jittered version of the current best
        # or a fresh random binary vector
        if rng.random() < 0.6:
            start_vec = best_h.copy()
            # small random jitter (few balanced transfers)
            for _ in range(7):
                i, j = rng.integers(n, size=2, replace=False)
                cap_i_to_j = min(start_vec[i], 1.0 - start_vec[j])
                cap_j_to_i = min(start_vec[j], 1.0 - start_vec[i])
                if cap_i_to_j > cap_j_to_i:
                    if cap_i_to_j > 0.0:
                        d = cap_i_to_j * rng.random() * 0.4
                        start_vec[i] -= d
                        start_vec[j] += d
                else:
                    if cap_j_to_i > 0.0:
                        d = cap_j_to_i * rng.random() * 0.4
                        start_vec[i] += d
                        start_vec[j] -= d
            start_vec = _project_to_box_sum(start_vec, n / 2.0)
        else:
            start_vec = np.zeros(n)
            start_vec[rng.choice(n, n // 2, replace=False)] = 1.0

        # run the local optimiser
        local_h, local_c = _hill_climb(start_vec, rng, deadline)

        if local_c < best_c:
            best_h, best_c = local_h, local_c

    # final clean‑up: enforce constraints exactly (tiny numeric drift)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_to_box_sum(best_h, n / 2.0)
    final_c = _compute_c5(best_h)

    return best_h, final_c, n
