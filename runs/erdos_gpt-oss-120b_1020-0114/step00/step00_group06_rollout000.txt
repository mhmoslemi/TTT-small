    diff = target - h.sum()

            h *= (target / total)
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                idxs = np.where(h < 1.0)[0]
                if len(idxs):
                    i = np.random.choice(idxs)
                    h[i] = min(1.0, h[i] + diff)
            else:
                idxs = np.where(h > 0.0)[0]
                if len(idxs):
                    i = np.random.choice(idxs)
                    h[i] = max(0.0, h[i] + diff)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound and the number of sample points."""
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # initialise h
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        N = len(h)
    else:
        N = 100                 # fallback size (even, convenient)
        h = np.zeros(N, dtype=np.float64)
        half = N // 2
        h[:half] = 1.0
        if N % 2:
            h[half] = 0.5
        np.random.shuffle(h)
    target_sum = N / 2.0
    h = _repair_sum(h, target_sum)
    h = np.clip(h, 0.0, 1.0)

    current_h = h.copy()
    current_c5 = _compute_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5

    # ------------------------------------------------------------------
    # simulated‑annealing parameters
    t_start = 0.04          # initial temperature
    t_end   = 1e-6          # final temperature
    max_move_initial = 0.20

    # ------------------------------------------------------------------
    # main optimisation loop
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s - 5.0:      # keep a small safety margin
            break

        progress = elapsed / (budget_s - 5.0)
        # geometric cooling
        T = t_start * (t_end / t_start) ** progress
        max_move = max_move_initial * (1.0 - progress) + 1e-4
        N = len(current_h)

        # --------------------------------------------------------------
        # occasional larger perturbation to escape plateaux
        if np.random.rand() < 0.003:
            cand = best_h.copy()
            m = np.random.randint(1, max(2, N // 20 + 1))
            idxs = np.random.choice(N, m, replace=False)
            cand[idxs] = np.random.rand(m)
            cand = _repair_sum(cand, target_sum)
            cand_c5 = _compute_c5(cand)
            delta = cand_c5 - current_c5
            if delta < 0.0 or np.random.rand() < math.exp(-delta / max(T, 1e-12)):
                current_h = cand
                current_c5 = cand_c5
                if cand_c5 < best_c5:
                    best_c5 = cand_c5
                    best_h = cand.copy()
            continue

        # --------------------------------------------------------------
        # small transfer move (preserves sum exactly)
        i, j = np.random.randint(N, size=2)
        if i == j:
            continue

        max_delta = min(max_move, current_h[i], 1.0 - current_h[j])
        if max_delta <= 0.0:
            # try the opposite direction
            max_delta = min(max_move, current_h[j], 1.0 - current_h[i])
            if max_delta <= 0.0:
                continue
            i, j = j, i

        delta = np.random.uniform(0.0, max_delta)
        cand = current_h.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_c5 = _compute_c5(cand)
        delta_c5 = cand_c5 - current_c5

        if delta_c5 < 0.0 or np.random.rand() < math.exp(-delta_c5 / max(T, 1e-12)):
            current_h = cand
            current_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = cand.copy()

    # ------------------------------------------------------------------
    # final tidy‑up (enforce the integral exactly)
    best_h = _repair_sum(best_h, target_sum)
    final_c5 = _compute_c5(best_h)

    return best_h.tolist(), float(final_c5), len(best_h)
