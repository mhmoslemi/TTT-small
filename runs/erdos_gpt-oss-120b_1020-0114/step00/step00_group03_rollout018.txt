def weighted_gradient(h, beta=10):
    n = len(h)
    f = np.correlate(h, 1 - h, mode='full')
    max_f = f.max()
    exp_f = np.exp(beta * (f - max_f))  # stable
    w = exp_f / exp_f.sum()
    grad = np.zeros_like(h)
    # loop over shifts
    for s, wk in enumerate(w):
        if wk == 0:
            continue
        k = s - (n-1)
        # overlapping region
        if k >= 0:
            i_start = 0
            i_end = n - k
            grad[i_start:i_end] += wk * (1 - h[i_start + k: i_end + k])
            grad[i_start + k: i_end + k] -= wk * h[i_start:i_end]
        else:
            k_abs = -k
            i_start = k_abs
            i_end = n
            grad[i_start:i_end] += wk * (1 - h[i_start - k_abs: i_end - k_abs])
            grad[i_start - k_abs: i_end - k_abs] -= wk * h[i_start:i_end]
    return grad

def optimize_subgradient(h0, max_time, beta=10.0, step0=0.5):

    best_h = h.copy()

    start = time.time()

    while True:
        elapsed = time.time() - start
        if elapsed >= max_time:
            break
        t += 1

        f = np.correlate(h, 1 - h, mode='full')
        c5 = f.max() * (2.0 / n)

            best_c5 = c5
            best_h = h.copy()

        w = np.exp(beta * (f - max_f))
        w_sum = w.sum()

            grad = subgradient_max_shift(h)
        else:
            w = w / w_sum

import time
import numpy as np

# ---------- helper utilities ----------

def compute_c5(h):
    """C5 for a given step function h (array of length n)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr


def project_onto_box_sum(v, target_sum, eps=1e-12):
    """
    Euclidean projection of vector v onto the set
    {x | 0 <= x_i <= 1, sum x_i = target_sum}.
    """
    lo = -np.max(v) - 1.0        # ensures v+lo <= 0 -> all zeros after clipping
    hi = 1.0 - np.min(v) + 1.0   # ensures v+hi >= 1 -> all ones after clipping
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    x = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = target_sum - x.sum()
    if abs(diff) > eps:
        # distribute the tiny residue on non‑boundary entries
        free = (x > eps) & (x < 1.0 - eps)
        if free.any():
            x[free] += diff / free.mean()
            np.clip(x, 0.0, 1.0, out=x)
        else:                # fallback – adjust the first element
            x[0] = np.clip(x[0] + diff, 0.0, 1.0)
    return x


def upsample_h(old_h, n_new):
    """Linear interpolation of an old step function onto a finer grid."""
    n_old = old_h.shape[0]
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    new_h = np.clip(new_h, 0.0, 1.0)
    new_h = project_onto_box_sum(new_h, n_new / 2.0)
    return new_h


def weighted_subgradient(h, corr, beta):
    """
    Gradient of the smooth max (log‑sum‑exp) of the overlap correlations.
    corr is the raw correlation vector (no dx factor).
    """
    n = h.shape[0]
    max_f = corr.max()
    w = np.exp(beta * (corr - max_f))
    w /= w.sum()
    grad = np.zeros_like(h)

    # iterate over all possible shifts
    for s, wk in enumerate(w):
        if wk == 0.0:
            continue
        k = s - (n - 1)                # shift value
        if k >= 0:
            i0, i1 = 0, n - k
            grad[i0:i1] += wk * (1.0 - h[i0 + k:i1 + k])
            grad[i0 + k:i1 + k] -= wk * h[i0:i1]
        else:
            ka = -k
            i0, i1 = ka, n
            grad[i0:i1] += wk * (1.0 - h[i0 - ka:i1 - ka])
            grad[i0 - ka:i1 - ka] -= wk * h[i0:i1]
    return grad


def subgradient_max_shift(h, corr):
    """
    Subgradient for the (non‑smooth) max—uses the shift attaining the max.
    """
    n = h.shape[0]
    s = np.argmax(corr) - (n - 1)
    grad = np.zeros_like(h)
    if s >= 0:
        i0, i1 = 0, n - s
        grad[i0:i1] += 1.0 - h[i0 + s:i1 + s]
        grad[i0 + s:i1 + s] -= h[i0:i1]
    else:
        sa = -s
        i0, i1 = sa, n
        grad[i0:i1] += 1.0 - h[i0 - sa:i1 - sa]
        grad[i0 - sa:i1 - sa] -= h[i0:i1]
    return grad


def optimize_h(initial_h, max_seconds, beta=12.0, step0=0.8):
    """Projected sub‑gradient descent with a soft‑max smoothing."""
    n = initial_h.shape[0]
    target_sum = n / 2.0
    h = initial_h.copy()
    best_h = h.copy()
    best_val, _ = compute_c5(h)

    start = time.time()
    it = 0
    while True:
        now = time.time()
        if now - start > max_seconds:
            break
        it += 1

        # correlation and current bound
        val, corr = compute_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()

        # gradient (smooth max)
        grad = weighted_subgradient(h, corr, beta)

        # step size with classic 1/√t decay
        step = step0 / np.sqrt(it)

        # projected update
        h = h - step * grad
        h = project_onto_box_sum(h, target_sum)

        # occasional hard‑max descent to sharpen the solution
        if it % 250 == 0:
            grad2 = subgradient_max_shift(h, corr)
            h = h - step * grad2
            h = project_onto_box_sum(h, target_sum)

    # final check
    final_val, _ = compute_c5(best_h)
    return best_h, final_val


# ---------- main entry point ----------

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points) where h_values is a numpy array.
    """
    np.random.seed(seed)

    # Desired grid resolution (trade‑off between quality and speed)
    n_points = 200

    # Build a feasible starting vector
    if 'initial_h_values' in globals():
        h0 = upsample_h(np.asarray(initial_h_values, dtype=float), n_points)
    else:
        # random start, then project onto the feasible set
        v = np.random.rand(n_points)
        h0 = project_onto_box_sum(v, n_points / 2.0)

    # Reserve a fraction of the time budget for the optimizer
    opt_time = budget_s * 0.95

    # Run the optimizer
    best_h, best_c5 = optimize_h(h0, opt_time, beta=12.0, step0=0.8)

    # Ensure the constraints are satisfied (tiny numerical adjustments)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, n_points / 2.0)

    return best_h, best_c5, n_points
