def simulated_annealing(h0, time_limit):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    start = time.time()
    T0 = 0.1  # initial temperature based on initial value range
    while time.time() - start < time_limit:
        # schedule temperature decreasing linearly
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = T0 * (1 - frac)
        # propose move
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # pick delta from normal distribution with std scaling with T
        delta = np.random.uniform(-0.05, 0.05) * (1 + T) # e.g.
        # adjust h[i] by delta within bounds
        hi = h[i] + delta
        hj = h[j] - delta
        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
            continue
        h_new = h.copy()
        h_new[i] = hi
        h_new[j] = hj
        val_new = compute_c5(h_new)
        dE = val_new - best_val
        if dE < 0 or np.random.rand() < np.exp(-dE / (T + 1e-12)):
            h = h_new
            if val_new < best_val:
                best_val = val_new
                best_h = h_new.copy()
```

import numpy as np
import time

def compute_c5(h):
    """Maximum overlap integral for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def hill_climb(h0, time_limit):
    """Simple simulated‑annealing style improvement with box and sum constraints."""
    n = h0.shape[0]
    target_sum = n / 2.0
    h = h0.copy()
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val
    start = time.time()
    # step schedule: starts larger, shrinks to a small minimum
    start_step, min_step = 0.25, 0.005

    while time.time() - start < time_limit:
        # fraction of time used
        frac = (time.time() - start) / time_limit
        step = start_step * (1.0 - frac) + min_step * frac
        # choose two distinct indices and a delta respecting bounds
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        # allowed delta range so that both stay inside [0,1]
        delta_low = max(-step, -hi, hj - 1.0)
        delta_high = min(step, 1.0 - hi, hj)
        if delta_low >= delta_high:
            continue
        delta = np.random.uniform(delta_low, delta_high)
        # propose move
        h[i] = hi + delta
        h[j] = hj - delta
        new_val = compute_c5(h)
        # accept if improvement, otherwise Metropolis step
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / (step + 1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert
            h[i] = hi
            h[j] = hj
    # final tiny projection to keep the sum exact
    cur = best_h.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-10:
        best_h += diff / n
        np.clip(best_h, 0.0, 1.0, out=best_h)
    return best_h, best_val


def init_from_prev(prev_h, prev_n, new_n):
    """Linear interpolation of a previous solution to a higher resolution."""
    x_old = np.linspace(0.0, 2.0, prev_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, prev_h)
    new_h = np.clip(new_h, 0.0, 1.0)
    target = new_n / 2.0
    cur = new_h.sum()
    # simple uniform correction
    new_h += (target - cur) / new_n
    np.clip(new_h, 0.0, 1.0, out=new_h)
    # second‑order fix if needed
    cur = new_h.sum()
    diff = target - cur
    if abs(diff) > 1e-10:
        free = np.where((new_h > 1e-12) & (new_h < 1.0 - 1e-12))[0]
        if free.size:
            inc = diff / free.size
            new_h[free] += inc
            np.clip(new_h, 0.0, 1.0, out=new_h)
    return new_h


def triangular_init(n):
    """Triangular hat function h(x)=1-|x-1| sampled at mids of n sub‑intervals."""
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    h = 1.0 - np.abs(xs - 1.0)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    cur = h.sum()
    if abs(cur - target) > 1e-12:
        h += (target - cur) / n
        np.clip(h, 0.0, 1.0, out=h)
    return h


def random_h(n):
    """Generate a random feasible vector (box + sum constraints)."""
    h = np.full(n, 0.5)
    target = n / 2.0
    # a few random swaps to break symmetry
    for _ in range(3 * n):
        i, j = np.random.randint(0, n, size=2)
        if i == j:

        delta = np.random.uniform(-0.2, 0.2)
        hi, hj = h[i] + delta, h[j] - delta
        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:

    cur = h.sum()
    if abs(cur - target) > 1e-10:
        h += (target - cur) / n
        np.clip(h, 0.0, 1.0, out=h)
    return h
