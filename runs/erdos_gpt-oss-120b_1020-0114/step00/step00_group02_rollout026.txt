def _project_to_bounds_and_sum(v, target_sum, max_iter=200):
    """Project v onto the set {0<=x_i<=1, sum x_i = target_sum}."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        current_sum = v.sum()
        diff = target_sum - current_sum
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = (v < 1 - 1e-12)
        else:
            free = (v > 1e-12)
        if not np.any(free):
            break
        delta = diff / free.sum()
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    # final tiny adjustment to hit the exact sum (should be within float tol)
    correction = target_sum - v.sum()
    if abs(correction) > 1e-12:
        # distribute the tiny correction uniformly among non‑boundary entries
        free = (v > 1e-12) & (v < 1 - 1e-12)
        if np.any(free):
            v[free] += correction / free.sum()
            v = np.clip(v, 0.0, 1.0)
    return v


def _c5_value(vec, dx):
    """Return the C5 value for a given vector."""
    corr = np.correlate(vec, 1.0 - vec, mode='full')
    return np.max(corr) * dx


def _gradient_max_corr(vec, dx):
    """
    Gradient of the maximum correlation term.
    Returns the gradient for the lag that currently attains the maximum.
    """
    n = vec.shape[0]
    corr_raw = np.correlate(vec, 1.0 - vec, mode='full')
    idx = np.argmax(corr_raw)
    lag = idx - (n - 1)          # integer shift corresponding to the max term

        term1 = np.concatenate([np.zeros(p), 1.0 - vec[: n - p]])
        term2 = np.concatenate([-vec[p:], np.zeros(p)])

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta = (random.random() - 0.5) * 0.12            # modest move size
        if not (0.0 <= h[i] + delta <= 1.0 and 0.0 <= h[j] - delta <= 1.0):
            continue
        h_candidate = h.copy()
        h_candidate[i] += delta
        h_candidate[j] -= delta

        val_cand = _c5_value(h_candidate, dx)
        if val_cand < best_val:
            best_val = val_cand
            best_h = h_candidate.copy()
            h = h_candidate
        elif val_cand < _c5_value(h, dx):
            h = h_candidate

    # ------------------------------------------------------------------
    # 5. Return the best solution discovered.
    # ------------------------------------------------------------------
    final_val = _c5_value(best_h, dx)
    return (best_h, float(final_val), n)
