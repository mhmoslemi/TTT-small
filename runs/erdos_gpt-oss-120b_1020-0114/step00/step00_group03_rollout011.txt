def compute_c5(h):

def gradient_shift(h, k):
    n = len(h)

    if k == 0:

    else:
        # forward shift: h[i+k]
        g[:n - k] += h[k:]
        # backward shift: h[i - k]
        g[k:] += h[:n - k]

import time
import random
import numpy as np

def _c5_score(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx using discrete samples."""
    n = h.shape[0]
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))

def _grad_for_shift(h, shift):
    """Gradient of  -∑ h_i h_{i+shift}  (i.e. of the overlap) w.r.t. h."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift == 0:
        g = 2.0 * h
    else:
        k = abs(shift)
        # forward part
        g[: n - k] += h[k:]
        # backward part
        g[k:] += h[: n - k]
    return g

def _project_to_constraints(x):
    """Project onto 0≤x≤1 and sum(x)=len(x)/2 (exactly)."""
    n = x.shape[0]
    target = n / 2.0
    x = np.clip(x, 0.0, 1.0)
    # simple iterative water‑filling to fix the sum
    for _ in range(10):
        cur = x.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        free = (x > 0.0) & (x < 1.0)
        if not free.any():
            # distribute uniformly (may violate bounds a little)
            x += diff / n
            x = np.clip(x, 0.0, 1.0)
            continue
        add = diff / free.sum()
        x[free] += add
        x = np.clip(x, 0.0, 1.0)
    return x

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for the Erdős C5 problem."""
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # 1. Obtain an initial feasible vector.
    # ------------------------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=float)  # provided by the harness
        n0 = h0.size
    except Exception:               # no parent construction available
        n0 = 200
        h0 = np.full(n0, 0.5, dtype=float)

    # If the initial vector is short, upscale to a larger working size.
    TARGET_N = max(200, n0)          # we like at least 200 points
    if n0 != TARGET_N:
        # linear interpolation on the interval [0,2)
        xs_src = np.linspace(0.0, 2.0, n0, endpoint=False)
        xs_dst = np.linspace(0.0, 2.0, TARGET_N, endpoint=False)
        h = np.interp(xs_dst, xs_src, h0)
    else:
        h = h0.copy()

    # Ensure feasibility exactly.
    h = _project_to_constraints(h)

    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = _c5_score(best_h)
    cur_h = h.copy()
    cur_score = best_score

    # ------------------------------------------------------------------
    # 2. Main optimisation loop (coordinate ascent with occasional random
    #    perturbations).  We maximise the minimal autocorrelation, i.e.
    #    minimise the C5 score.
    # ------------------------------------------------------------------
    max_iters = 10_000_000          # effectively limited by the time budget
    step_initial = 0.05
    K_WORST = 3                     # number of worst shifts to influence the gradient

    for it in range(max_iters):
        # time check
        if time.time() - start_time > budget_s * 0.95:

        # autocorrelation of cur_h with itself
        corr_full = np.correlate(cur_h, cur_h, mode="full")
        # indices of the K smallest values (worst correlations)
        worst_idx = np.argpartition(corr_full, K_WORST)[:K_WORST]
