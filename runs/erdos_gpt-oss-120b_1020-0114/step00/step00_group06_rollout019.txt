def project(x):
    x = np.clip(x, 0.0, 1.0)
    # enforce sum by iterative redistribution
    target = n/2.0
    for _ in range(5):
        diff = target - x.sum()
        if abs(diff) < 1e-9:
            break
        # distribute diff uniformly among non-bound elements
        free = (x > 0) & (x < 1)
        cnt = free.sum()
        if cnt == 0:
            # distribute to all and clip
            x += diff / n
            x = np.clip(x, 0.0, 1.0)
        else:
            delta = diff / cnt
            x[free] += delta
            # Clip again to avoid overshoot
            x = np.clip(x, 0.0, 1.0)
    # final adjustment: if sum slightly off due to rounding, adjust by adding small amount to a free element
    diff = target - x.sum()
    if abs(diff) > 1e-7:
        # add diff to first element not at bound
        for i in range(n):
            if (diff > 0 and x[i] < 1) or (diff < 0 and x[i] > 0):
                x[i] = np.clip(x[i] + diff, 0.0, 1.0)
                break
    return x

def compute_c5(h, dx):
    return np.max(np.correlate(h, 1 - h, mode='full') * dx)

def compute_gradient(h, shift, dx):
    n = len(h)
    grad = np.full(n, dx)  # constant part
    term1 = np.zeros(n)
    term2 = np.zeros(n)
    if shift >= 0:
        if shift < n:
            term1[:n-shift] = h[shift:]
            term2[shift:] = h[:n-shift]
    else:
        k = -shift
        if k < n:
            term1[k:] = h[:n-k]
            term2[:n-k] = h[k:]
    grad -= dx * (term1 + term2)
    return grad

def project_to_feasible(x, target_sum):
    # clip and adjust sum
    x = np.clip(x, 0.0, 1.0)
    n = len(x)
    for _ in range(10):
        diff = target_sum - x.sum()
        if abs(diff) < 1e-12:
            break
        free = (x > 0) & (x < 1)
        cnt = free.sum()
        if cnt == 0:
            # adjust uniformly all
            x += diff / n
            x = np.clip(x, 0.0, 1.0)
        else:
            delta = diff / cnt
            x[free] += delta
            x = np.clip(x, 0.0, 1.0)
    # final residual adjustment
    diff = target_sum - x.sum()
    if abs(diff) > 1e-9:
        # add/subtract to first non-bound element

            if diff > 0 and x[i] < 1 - 1e-12:

                break
            elif diff < 0 and x[i] > 1e-12:
                x[i] = max(0.0, x[i] + diff)
                break
    return x

def compute_c5(h, dx):
    # returns maximum overlap integral term
    return np.max(np.correlate(h, 1 - h, mode='full') * dx)

def compute_gradient(h, shift, dx):
    n = h.shape[0]
    grad = np.full(n, dx)  # contribution from the constant 1 term
    # term for h_{j+shift}
    if shift >= 0:
        if shift < n:
            grad[:n - shift] -= dx * h[shift:]
            grad[shift:]   -= dx * h[:n - shift]
    else:
        k = -shift
        if k < n:
            grad[k:]       -= dx * h[:n - k]
            grad[:n - k]   -= dx * h[k:]
    return grad

def project_to_feasible(x, target_sum):

    x = np.clip(x, 0.0, 1.0)
    n = x.shape[0]

            x[free] += delta
            x = np.clip(x, 0.0, 1.0)

    diff = target_sum - x.sum()

                break
            if diff < 0 and x[i] > 0.0:
                x[i] = max(0.0, x[i] + diff)
                break

import numpy as np
import time

# ---------- helper utilities ----------

def compute_c5(h, dx):
    """
    Return max_k ∫ h(x)(1‑h(x+k))dx for discrete h.
    """
    # np.correlate implements Σ_i h_i·(1‑h_{i+k}) for all integer k
    return np.max(np.correlate(h, 1 - h, mode="full") * dx)


def compute_gradient(h, shift, dx):
    """
    Gradient of the worst shift term C_k with respect to h.
    shift is an integer in [-(n‑1), n‑1] where n = len(h).
    """
    n = h.shape[0]
    grad = np.full(n, dx)                     # derivative of Σ_i h_i part
    if shift >= 0:
        if shift < n:
            # term from h_{j+shift}
            grad[: n - shift] -= dx * h[shift:]
            # term from h_{j‑shift}
            grad[shift:] -= dx * h[: n - shift]
    else:
        k = -shift
        if k < n:
            grad[k:] -= dx * h[: n - k]      # h_{j‑k}
            grad[: n - k] -= dx * h[k:]      # h_{j+k}
    return grad


def project_to_feasible(x, target_sum):
    """
    Clip to [0,1] and adjust so that Σ x_i = target_sum.
    Uses a simple water‑filling scheme.
    """
    x = np.clip(x, 0.0, 1.0)
    n = x.shape[0]

    for _ in range(10):
        diff = target_sum - x.sum()
        if abs(diff) < 1e-12:
            break
        free = (x > 0.0) & (x < 1.0)
        cnt = free.sum()
        if cnt == 0:
            x += diff / n
            x = np.clip(x, 0.0, 1.0)
        else:
            delta = diff / cnt
            x[free] += delta
            x = np.clip(x, 0.0, 1.0)

    # tiny final correction if needed
    diff = target_sum - x.sum()
    if abs(diff) > 1e-9:
        for i in range(n):
            if diff > 0 and x[i] < 1.0:
                x[i] = min(1.0, x[i] + diff)
                break
            if diff < 0 and x[i] > 0.0:
                x[i] = max(0.0, x[i] + diff)
                break
    return x


def random_feasible(n):
    """Return a random vector in [0,1]^n with sum = n/2."""
    x = np.random.rand(n)
    return project_to_feasible(x, n / 2.0)


# ---------- main optimisation routine ----------

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points):
        h_values  – list of step‑function values on [0,2]
        c5_bound  – the achieved upper bound for C₅
        n_points  – number of discrete samples used
    """
    np.random.seed(seed)

    # ---- initialise -------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=np.float64).copy()
        n = h.shape[0]
    else:
        n = 95                     # a modest size that evaluates quickly
        h = random_feasible(n)

    target_sum = n / 2.0
    dx = 2.0 / n

    best = h.copy()
    best_val = compute_c5(best, dx)

    start = time.time()
    lr = 0.12                     # learning rate for projected GD
    decay = 0.9995                # gradual decay
    it = 0

    # ---- Phase 1 : projected gradient descent on the worst shift ---
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s * 0.85:
            break

        # current worst shift
        conv = np.correlate(h, 1 - h, mode="full")
        C_arr = conv * dx
        idx = int(np.argmax(C_arr))
        cur_val = C_arr[idx]
        shift = idx - (n - 1)

        # gradient of that worst term
        grad = compute_gradient(h, shift, dx)

        # trial step + projection
        h_new = h - lr * grad
        h_new = project_to_feasible(h_new, target_sum)

        new_val = compute_c5(h_new, dx)

        # accept improvement or occasional uphill move (simple annealing)
        if new_val < cur_val - 1e-12:
            h = h_new
        else:
            temp = max(0.001, (budget_s - elapsed) / budget_s)   # linear cooling
            prob = np.exp(-(new_val - cur_val) / (temp * 0.01 + 1e-12))
            if np.random.rand() < prob:
                h = h_new

        # update overall best
        if new_val < best_val - 1e-12:
            best_val = new_val
            best = h_new.copy()

        lr *= decay
        it += 1

        # occasional random perturbation of the current best to escape flat regions
        if it % 5000 == 0:
            pert = best + np.random.normal(scale=0.02, size=n)
            h = project_to_feasible(pert, target_sum)

    # ---- Phase 2 : pairwise hill‑climbing moves (sum‑preserving) --
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s * 0.98:
            break

        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = (np.random.rand() - 0.5) * 0.15    # up to ±0.075
        hi, hj = best[i], best[j]
        if not (0.0 <= hi + delta <= 1.0):
            continue
        if not (0.0 <= hj - delta <= 1.0):
            continue

        cand = best.copy()
        cand[i] = hi + delta
        cand[j] = hj - delta

        cand_val = compute_c5(cand, dx)
        if cand_val < best_val - 1e-12:
            best_val = cand_val
            best = cand.copy()

    # final safety projection and evaluation
    best = project_to_feasible(best, target_sum)
    final_c5 = compute_c5(best, dx)

    return best.tolist(), float(final_c5), int(n)
