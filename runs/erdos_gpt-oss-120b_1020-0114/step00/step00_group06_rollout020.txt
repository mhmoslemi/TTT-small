def gradient_of_max_corr(h):
    N = len(h)
    corr_full = np.correlate(h, 1-h, mode='full')
    idx = np.argmax(corr_full)
    shift = idx - (N-1)
    # compute g
    g = np.zeros(N)
    if shift >= 0:
        # h[j+shift] valid for j in 0..N-1-shift
        g[:N-shift] += -dx * h[shift:]  # because j+shift = where h index from shift..N-1
        # h[j-shift] valid for j in shift..N-1
        g[shift:] += -dx * h[:N-shift]
    else:
        # shift negative
        s = -shift
        # h[j+shift] = h[j - s]; valid for j >= s
        g[s:] += -dx * h[:N-s]
        # h[j-shift] = h[j + s]; valid for j <= N-1-s
        g[:N-s] += -dx * h[s:]
    return g

import numpy as np, time, math, random

def compute_c5(h):
    """Maximum overlap integral for a discrete step function."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_bounds_sum(x, target):
    """
    Euclidean projection of x onto the feasible set:
        0 <= h_i <= 1  and  sum(h) = target.
    Implements a binary‑search on the shift s in h = clip(x - s,0,1).
    """
    lo = np.min(x) - 1.0      # makes all entries -> 1 after clipping
    hi = np.max(x)            # makes all entries -> 0 after clipping
    for _ in range(50):
        mid = (lo + hi) * 0.5
        h = np.clip(x - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny numerical correction
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.any():
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def gradient_of_max_corr(h):
    """Sub‑gradient of max_k ∫ h(1‑h_shift) for the shift attaining the maximum."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)          # signed lag
    g = np.zeros_like(h)
    if shift >= 0:
        g[: n - shift] += -dx * h[shift:]
        g[shift:]      += -dx * h[: n - shift]
    else:
        s = -shift
        g[s:]     += -dx * h[: n - s]
        g[: n - s] += -dx * h[s:]
    return g

    """
    Return a step function h on [0,2] (as a list), its C5 bound,
    and the number of sample points.
    """
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Initialise problem size and the starting vector
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
        target = n / 2.0
        h0 = project_bounds_sum(h0, target)          # ensure feasibility
    else:
        n = 95
        target = n / 2.0
        h0 = np.full(n, 0.5)                         # trivial feasible start

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Core simulated‑annealing / hill‑climbing loop
    # ------------------------------------------------------------------
    start_time = time.time()
    cur_h = best_h.copy()
    cur_c5 = best_c5
    it = 0
    # we keep a fraction of the budget for the final re‑projection
    time_limit = budget_s * 0.95

    while True:
        elapsed = time.time() - start_time
        if elapsed >= time_limit:
            break
        # schedule: step size shrinks linearly, temperature shrinks linearly
        frac = elapsed / time_limit
        step = 0.5 * (1.0 - frac) + 0.001          # perturbation magnitude
        temp = max(1e-5, 0.05 * (1.0 - frac))      # annealing temperature

        # ----- random perturbation -------------------------------------------------
        cand = cur_h + step * np.random.randn(n)
        cand = project_bounds_sum(cand, target)
        cand_c5 = compute_c5(cand)

        # ----- Metropolis acceptance ------------------------------------------------
        if cand_c5 < cur_c5 or np.random.rand() < math.exp(-(cand_c5 - cur_c5) / temp):
            cur_h, cur_c5 = cand, cand_c5
            if cand_c5 < best_c5:
                best_h, best_c5 = cand.copy(), cand_c5

        # ----- occasional gradient descent step ------------------------------------
        if it % 10 == 0:
            g = gradient_of_max_corr(cur_h)
