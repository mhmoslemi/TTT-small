    # generate random values uniformly [0,1]
    h = np.random.rand(n)
    # scale to target sum
    target = n/2
    total = h.sum()
    h = h * (target / total)
    # clip to [0,1] might reduce sum; adjust remaining diff equally among non-clipped indices
    # simple approach: iterate a few times
    for _ in range(5):
        excess = h - 1
        under = -h
        over_idx = excess > 0
        under_idx = under > 0
        if not over_idx.any() and not under_idx.any():
            break
        # adjust over and under to bring within [0,1] while preserving sum via redistribution
        if over_idx.any():

def upsample_h(h, new_n):
    old_n = len(h)
    old_x = np.linspace(0, 2, old_n, endpoint=False) + (2.0/old_n)/2.0
    new_x = np.linspace(0, 2, new_n, endpoint=False) + (2.0/new_n)/2.0
    new_h = np.interp(new_x, old_x, h)
    target = new_n / 2.0
    cur_sum = new_h.sum()
    if cur_sum > 0:
        new_h *= target / cur_sum
    # clip and adjust diff
    np.clip(new_h, 0.0, 1.0, out=new_h)
    diff = target - new_h.sum()
    if abs(diff) > 1e-12:
        free = (new_h > 0) & (new_h < 1)
        if free.any():
            # distribute diff uniformly among free slots
            new_h[free] += diff / free.sum()
            np.clip(new_h, 0.0, 1.0, out=new_h)
        else:
            # fallback uniform scaling
            new_h = np.clip(new_h, 0.0, 1.0)
            cur_sum = new_h.sum()
            if cur_sum != 0:
                new_h *= target / cur_sum
    return new_h

def _c5_from_h(h):
    """return C5 for a given h (dx is implicit in length)"""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode="full")) * dx

def _random_binary(n):
    """binary vector with exactly n/2 ones (n must be even)"""
    h = np.zeros(n, dtype=float)
    h[: n // 2] = 1.0
    np.random.shuffle(h)
    return h

def _upsample_h(h, new_n):
    """linear interpolation of h to a larger grid, re‑project to sum=n/2"""
    old_n = len(h)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + (2.0 / old_n) / 2.0
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + (2.0 / new_n) / 2.0
    new_h = np.interp(new_x, old_x, h, left=h[0], right=h[-1])
    target = new_n / 2.0
    # scale to the correct total mass
    s = new_h.sum()
    if s != 0:
        new_h *= target / s
    # enforce box constraints and fix any residual mass error
    np.clip(new_h, 0.0, 1.0, out=new_h)
    diff = target - new_h.sum()
    if abs(diff) > 1e-12:
        free = (new_h > 0.0) & (new_h < 1.0)
        if free.any():
            new_h[free] += diff / free.sum()
            np.clip(new_h, 0.0, 1.0, out=new_h)
        else:
            # fallback uniform scaling
            s = new_h.sum()
            if s != 0:
                new_h *= target / s
            np.clip(new_h, 0.0, 1.0, out=new_h)
    return new_h

def _hill_climb(start_h, time_budget):
    """
    Simple monotone hill climbing.
    Moves consist of transferring δ from one index to another while keeping the total sum.
    Uses the lag with maximal overlap as a guide.
    """
    n = len(start_h)
    dx = 2.0 / n
    cur_h = start_h.copy()
    cur_corr = np.correlate(cur_h, 1 - cur_h, mode="full")
    cur_c5 = np.max(cur_corr) * dx
    best_h = cur_h.copy()
    best_c5 = cur_c5

    shifts = np.arange(-n + 1, n)  # length 2n‑1
    start = time.time()
    it = 0

    while time.time() - start < time_budget:
        # lag with current maximal overlap
        idx_max = int(np.argmax(cur_corr))
        L = idx_max - (n - 1)

        # gradient of the maximal term w.r.t. each h[i]
        forward = np.zeros(n)
        backward = np.zeros(n)
        if L >= 0:
            forward[: n - L] = 1.0 - cur_h[L:]
            backward[L:] = cur_h[: n - L]
        else:
            Lneg = -L
            forward[Lneg:] = 1.0 - cur_h[: n - Lneg]
            backward[: n - Lneg] = cur_h[Lneg:]
        grad = forward - backward

        pos = np.where(grad > 1e-12)[0]   # decreasing them helps
        neg = np.where(grad < -1e-12)[0]  # increasing them helps

        moved = False
        # try a guided move (transfer from a 'pos' index to a 'neg' index)
        if pos.size and neg.size:
            inc_idx = int(neg[np.random.randint(neg.size)])   # index to increase
            dec_idx = int(pos[np.random.randint(pos.size)])   # index to decrease
            delta_max = min(1.0 - cur_h[inc_idx], cur_h[dec_idx])
            if delta_max > 0.0:
                delta = delta_max * np.random.rand()
                new_h = cur_h.copy()
                new_h[inc_idx] += delta
                new_h[dec_idx] -= delta
                new_corr = np.correlate(new_h, 1 - new_h, mode="full")
                new_c5 = np.max(new_corr) * dx
                if new_c5 <= cur_c5:
                    cur_h, cur_corr, cur_c5 = new_h, new_corr, new_c5
                    moved = True

        # fallback: random feasible transfer
        if not moved:
            i, j = np.random.randint(n, size=2)
            if i != j:
                lo = max(-cur_h[i], cur_h[j] - 1.0)
                hi = min(1.0 - cur_h[i], cur_h[j])
                if lo < hi:
                    delta = lo + np.random.rand() * (hi - lo)
                    if delta != 0.0:
                        new_h = cur_h.copy()
                        new_h[i] += delta
                        new_h[j] -= delta
                        new_corr = np.correlate(new_h, 1 - new_h, mode="full")
                        new_c5 = np.max(new_corr) * dx
                        if new_c5 < cur_c5:
                            cur_h, cur_corr, cur_c5 = new_h, new_corr, new_c5
                            moved = True

        if cur_c5 < best_c5 - 1e-12:
            best_c5, best_h = cur_c5, cur_h.copy()

        it += 1
        # occasional full recompute to keep numerical drift tiny
        if it % 20000 == 0:
            cur_corr = np.correlate(cur_h, 1 - cur_h, mode="full")
            cur_c5 = np.max(cur_corr) * dx

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)

    total_start = time.time()
    total_budget = float(budget_s)

    # ------------------------------------------------------------------
    # initial construction (provided by the harness if available)
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        # fallback: simple binary seed
        h0 = _random_binary(70)

    n0 = len(h0)
    target0 = n0 / 2.0
    # guarantee the integral constraint exactly
    diff0 = target0 - h0.sum()
    if abs(diff0) > 1e-12:
        free = (h0 > 0.0) & (h0 < 1.0)
        if free.any():
            h0[free] += diff0 / free.sum()
            np.clip(h0, 0.0, 1.0, out=h0)
        else:
            h0 *= target0 / h0.sum()
            np.clip(h0, 0.0, 1.0, out=h0)

    # ------------------------------------------------------------------
    # Stage 1 : low‑resolution refinement
    # ------------------------------------------------------------------
    low_budget = min(300.0, total_budget * 0.2)
    # two short runs (original seed + random binary) and keep the best
    half = low_budget / 2.0
    best_h_low, best_c5_low = _hill_climb(h0, half)
    rand_start = _random_binary(n0)
    h_tmp, c5_tmp = _hill_climb(rand_start, low_budget - half)
    if c5_tmp < best_c5_low:
        best_h_low, best_c5_low = h_tmp, c5_tmp

    elapsed = time.time() - total_start
    remaining = total_budget - elapsed - 2.0  # 2 s safety margin

    # ------------------------------------------------------------------
    # Stage 2 : higher resolution (if time permits)
    # ------------------------------------------------------------------
    if remaining > 5.0:
        n1 = max(140, n0 * 2)
        if n1 % 2:
            n1 += 1
        if n1 > 300:
            n1 = 300
        h1 = _upsample_h(best_h_low, n1)
        # give the higher‑resolution search the rest of the budget
        best_h_high, best_c5_high = _hill_climb(h1, remaining)

        # possibly also try a fresh binary start at the higher resolution
        if remaining > 30.0:
            h_bin = _random_binary(n1)
            bh, bc = _hill_climb(h_bin, min(30.0, remaining / 3.0))
            if bc < best_c5_high:
                best_h_high, best_c5_high = bh, bc

        if best_c5_high < best_c5_low:
            final_h = best_h_high
            final_c5 = best_c5_high
            final_n = n1
        else:
            final_h = best_h_low
            final_c5 = best_c5_low
            final_n = n0
    else:
        final_h = best_h_low
        final_c5 = best_c5_low
        final_n = n0

    # final tiny correction to satisfy the integral exactly
    target_final = final_n / 2.0
    diff = target_final - final_h.sum()
    if abs(diff) > 1e-12:
        free = (final_h > 0.0) & (final_h < 1.0)
        if free.any():
            final_h[free] += diff / free.sum()
            np.clip(final_h, 0.0, 1.0, out=final_h)
        else:
            final_h *= target_final / final_h.sum()
            np.clip(final_h, 0.0, 1.0, out=final_h)

    # compute the exact bound to return
    final_c5 = _c5_from_h(final_h)

    return (final_h, float(final_c5), int(final_n))
