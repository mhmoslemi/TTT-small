def refine_h(initial_h, max_iter=200, beta=20):
    n = len(initial_h)
    target_sum = n/2
    bounds = [(0,1)]*n
    constraints = [{'type':'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    # define objective wrapper
    def obj(x):
        # compute smoothed max
        dx = 2.0 / n
        corr = np.correlate(x, 1-x, mode='full') * dx
        m = np.max(corr)
        # smoothing
        # compute log-sum-exp
        # using mean
        exp = np.exp(beta * (corr - m))
        # return m + log(mean(exp))/beta
        return m + np.log(np.mean(exp)) / beta
    res = minimize(obj, initial_h, method='SLSQP', bounds=bounds, constraints=constraints,
                   options={'maxiter': max_iter, 'ftol': 1e-9, 'disp': False})
    x = np.clip(res.x, 0, 1)
    # enforce sum exactly by scaling small adjustment
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-9:
        # adjust uniformly
        x = x + diff / n
        x = np.clip(x, 0, 1)
        # maybe adjust again to meet sum
        # simple scaling ignoring bounds
        total = np.sum(x)
        if total != 0:
            x = x * target_sum / total
        x = np.clip(x, 0, 1)
    return x, compute_c5(x)

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper functions (top level, no closures / lambdas)
# ----------------------------------------------------------------------


def compute_c5(h: np.ndarray) -> float:
    """Exact C5 evaluation using full linear correlation."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))


def triangular_h(n: int) -> np.ndarray:
    """Triangular step function on [0,2] with area 1."""
    dx = 2.0 / n
    centers = (np.arange(n) + 0.5) * dx
    h = np.minimum(centers, 2.0 - centers)   # shape of an isosceles triangle
    # enforce the integral constraint (sum = n/2)
    target = n / 2.0
    cur = h.sum()
    if cur != 0:
        h = h * (target / cur)
    h = np.clip(h, 0.0, 1.0)
    # tiny adjustment for exact sum
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def binary_feasible_h(n: int) -> np.ndarray:
    """Binary vector with exactly n/2 ones (or as close as possible)."""
    h = np.zeros(n, dtype=float)
    ones = n // 2
    h[:ones] = 1.0
    np.random.shuffle(h)
    return h


def random_feasible_h(n: int) -> np.ndarray:
    """Random feasible vector (0≤h≤1, sum=n/2)."""
    # start from a binary pattern and add jitter
    h = binary_feasible_h(n)
    eps = 0.2
    jitter = np.random.uniform(-eps, eps, size=n)
    h = h + jitter
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    cur = h.sum()
    if cur != 0:
        h = h * (target / cur)
    h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


class ObjectiveWithPenalty:
    """Smooth approximation of max correlation + quadratic penalty for sum."""

    def __init__(self, beta: float, target: float, penalty: float):
        self.beta = beta
        self.target = target
        self.penalty = penalty

    def __call__(self, x: np.ndarray) -> float:
        n = x.size
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode="full") * dx
        m = np.max(corr)
        # log‑sum‑exp smooth max
        exp_term = np.exp(self.beta * (corr - m))
        smooth = m + np.log(np.mean(exp_term)) / self.beta
        # quadratic penalty for violating the integral constraint
        pen = self.penalty * (np.sum(x) - self.target) ** 2
        return smooth + pen


def hill_climb(h: np.ndarray, max_iters: int, time_limit: float) -> (np.ndarray, float):
    """Very cheap random pairwise perturbation with sum preservation."""
    n = h.size
    target = n / 2.0
    best = h.copy()
    best_val = compute_c5(best)
    start = time.time()
    step = 0.05
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        i, j = np.random.choice(n, 2, replace=False)
        delta = np.random.uniform(-step, step)
        new_i = best[i] + delta
        new_j = best[j] - delta
        if not (0.0 <= new_i <= 1.0 and 0.0 <= new_j <= 1.0):
            continue
        cand = best.copy()
        cand[i] = new_i
        cand[j] = new_j
        val = compute_c5(cand)
        if val < best_val:
            best = cand
            best_val = val
    # final projection onto exact sum (tiny adjustment)
    diff = target - best.sum()
    if abs(diff) > 1e-10:
        best += diff / n
        best = np.clip(best, 0.0, 1.0)
    return best, best_val
