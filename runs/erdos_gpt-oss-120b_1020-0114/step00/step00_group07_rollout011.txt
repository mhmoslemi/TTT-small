def _optimize_binary(start_h, rng, time_limit):
    N = len(start_h)
    dx = 2.0 / N
    h = start_h.astype(float)
    # compute initial correlation and scores
    corr = np.correlate(h, 1 - h, mode='full')
    cur_score = np.max(corr) * dx
    best_h = h.copy()
    best_score = cur_score

    start = time.time()
    iteration = 0
    # schedule parameters
    T0 = 0.02  # maybe adjust later based on cur_score
    T_end = 1e-6

    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        iteration += 1
        # compute T
        frac = elapsed / time_limit
        T = T0 * (1 - frac) + T_end

        # compute current correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx] * dx
        # we have max_val = cur_score? might have changed after last acceptance; but recalc to be safe.
        cur_score = max_val

        offset = max_idx - (N - 1)  # shift value (integer)
        # attempt targeted swap
        performed = False
        if offset != 0:
            if offset > 0:
                i_range = np.arange(N - offset)
                # contributions where h=1 and h[i+offset]=0
                mask = (h[i_range] == 1) & (h[i_range + offset] == 0)
                contrib_i = i_range[mask]
                if contrib_i.size > 0:
                    i1 = int(rng.choice(contrib_i))
                    i0 = i1 + offset
                    performed = True
            else:
                off = -offset
                i_range = np.arange(off, N)  # i >= off
                mask = (h[i_range] == 1) & (h[i_range - off] == 0)
                contrib_i = i_range[mask]
                if contrib_i.size > 0:
                    i1 = int(rng.choice(contrib_i))
                    i0 = i1 - off
                    performed = True

        if not performed:
            # random swap
            ones_idx = np.flatnonzero(h)
            zeros_idx = np.flatnonzero(1 - h)
            if len(ones_idx) == 0 or len(zeros_idx) == 0:
                break

            i0 = int(rng.choice(zeros_idx))

        h[i1] = 0.0
        h[i0] = 1.0

        # evaluate new cost
        new_corr = np.correlate(h, 1 - h, mode='full')
        new_score = np.max(new_corr) * dx

            h[i0] = 0.0

        # optionally, occasional re-evaluation of score beyond cur_score to update best if correlation changed due to earlier changes not captured? but we already compute.

    return best_h, best_score

import cvxpy as cp

def solve_milp(N):
    # Define variables
    h = cp.Variable(N, boolean=True)
    # define t variable, continuous
    t = cp.Variable()
    # constraints list
    constraints = [cp.sum(h) == N/2]  # N/2 integer for even N
    # For each shift s
    for s in range(-N+1, N):
        # compute indices i_range
        if s >= 0:
            i = np.arange(N - s)
            j = i + s
        else:
            offset = -s
            i = np.arange(offset, N)
            j = i - offset
        # create w_var for this shift: length len(i)
        w = cp.Variable(len(i), boolean=True)
        # constraints linking w and h
        constraints += [w <= h[i],
                        w <= h[j],
                        w >= h[i] + h[j] - 1]
        # now sum w over this shift >= t
        constraints.append(cp.sum(w) >= t)
        # store w maybe need to keep list to reuse? we won't store but only require to create variable for each shift.
    # objective: maximize t (i.e., Find max minimal overlap)
    prob = cp.Problem(cp.Maximize(t), constraints)
    prob.solve(solver=cp.CBC, verbose=False)  # might need to set max_iters, time limit? but we have budget.

    best_h = np.round(h.value).astype(int)  # may be already integer.
    t_val = t.value
    # compute C5:
    dx = 2.0 / N
    C5 = (N/2 - t_val) * dx
    return best_h, C5

def solve_milp(N, time_limit):
    import cvxpy as cp, numpy as np, time
    h = cp.Variable(N, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == N/2]
    # iterate s
    for s in range(-N+1, N):
        if s >= 0:
            i = np.arange(N - s)
            j = i + s
        else:
            off = -s
            i = np.arange(off, N)
            j = i - off
        w = cp.Variable(len(i), boolean=True)
        constraints += [w <= h[i],
                        w <= h[j],
                        w >= h[i] + h[j] - 1,
                        cp.sum(w) >= t]
    prob = cp.Problem(cp.Maximize(t), constraints)
    # Provide time limit if solver supports, else ignore.
    try:
        prob.solve(solver=cp.CBC, max_seconds=time_limit, verbose=False)
    except Exception:
        # fallback to GLPK MI
        try:
            prob.solve(solver=cp.GLPK_MI, verbose=False)
        except Exception:
            # fallback to ECOS_BB
            prob.solve(solver=cp.ECOS_BB, verbose=False)
    h_val = np.round(h.value).astype(int)
    t_val = t.value
    dx = 2.0/N
    C5 = (N/2 - t_val) * dx
    return h_val, float(C5)

def _continuous_optimize(h_init, budget_s, rng):
    h = h_init.copy()
    N = len(h)
    S = N / 2
    dx = 2.0 / N
    best_h = h.copy()

    start = time.time()

    alpha0 = 0.3  # initial step
    decay = 0.9995  # per iteration

        elapsed = time.time() - start
        if elapsed >= budget_s:
            break

        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        s = max_idx - (N - 1)
        # compute gradient for shift s
        g = np.zeros_like(h)
        if s >= 0:
            L = N - s
            g[:L] += 1 - h[s:]
            g[s:] -= h[:L]
        else:
            s_pos = -s
            L = N - s_pos
            g[s_pos:] += 1 - h[:L]
            g[:L] -= h[s_pos:]
        # step size:
        alpha = alpha0 * (decay ** iteration)
        v = h - alpha * g
        # project to feasible set
        h = project_capped_simplex(v, S)
        # evaluate
        cur_score = _c5_score(h)
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_h = h.copy()

def _c5_score(h):
    N = len(h)
    dx = 2.0 / N
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def _project_capped_simplex(v, S):
    lo = np.min(v) - 1.0
    hi = np.max(v)

import time
import numpy as np

def _c5_score(h):
    """Maximum overlap integral for a step function h."""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode="full")) * dx


def _project_capped_simplex(v, target_sum):
    """Projection of v on {0≤x≤1, sum(x)=target_sum}."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _continuous_optimize(h0, time_budget, rng):
    """Sub‑gradient descent on the shift with maximal overlap."""
    h = h0.astype(float).copy()
    n = len(h)
    half = n / 2.0
    best_h, best_score = h.copy(), _c5_score(h)
    start = time.time()
    it = 0
    alpha0, decay = 0.5, 0.9995
    while time.time() - start < time_budget:
        it += 1
        corr = np.correlate(h, 1 - h, mode="full")
        s = np.argmax(corr) - (n - 1)                # integer shift attaining the max
        g = np.zeros_like(h)
        if s >= 0:
            L = n - s
            g[:L] += 1.0 - h[s:]                     # d/dh_i for i in [0, L)
            g[s:] -= h[:L]                           # d/dh_i for i in [s, n)
        else:
            off = -s
            L = n - off
            g[off:] += 1.0 - h[:L]                   # d/dh_i for i in [off, n)
            g[:L] -= h[off:]                         # d/dh_i for i in [0, L)
        alpha = alpha0 * (decay ** it)
        h = _project_capped_simplex(h - alpha * g, half)
        cur = _c5_score(h)
        if cur < best_score - 1e-9:
            best_score, best_h = cur, h.copy()
    return best_h, best_score


def _solve_milp(N, time_limit):
    """Binary MILP: maximise the minimal self‑overlap."""
    import cvxpy as cp

    h = cp.Variable(N, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == N // 2]

    for s in range(-N + 1, N):
        if s >= 0:
            L = N - s
            i1 = np.arange(L)
            i2 = i1 + s
        else:
            off = -s
            L = N - off
            i1 = np.arange(off, N)
            i2 = i1 - off
        w = cp.Variable(L, boolean=True)
        constraints += [w <= h[i1],
                        w <= h[i2],
                        w >= h[i1] + h[i2] - 1,
                        cp.sum(w) >= t]

    prob = cp.Problem(cp.Maximize(t), constraints)

    # try a few common MILP solvers
    for solver in (cp.CBC, cp.GLPK_MI, cp.SCIP, cp.ECOS_BB):
        try:
            prob.solve(solver=solver, max_seconds=time_limit, verbose=False)
            if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
                break
        except Exception:
            continue

    if h.value is None:
        raise RuntimeError("MILP failed to solve")
    h_bin = np.rint(h.value).astype(int)
    t_val = float(t.value)
    dx = 2.0 / N
    bound = (N / 2 - t_val) * dx
    return h_bin, bound


def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C₅ bound, and the number of samples."""
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # initialise a feasible binary vector
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        N = len(init)
        h = np.rint(init).astype(float)               # force 0/1
        target = N // 2
        cur = int(h.sum())
        if cur != target:                             # balance the weight
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if cur > target:
                flip = rng.choice(ones, cur - target, replace=False)
                h[flip] = 0.0
            else:
                flip = rng.choice(zeros, target - cur, replace=False)
                h[flip] = 1.0
    else:
        N = 120                                        # a modest resolution
        target = N // 2
        h = np.zeros(N, dtype=float)
        h[rng.choice(N, target, replace=False)] = 1.0

    best_h, best_score = h.copy(), _c5_score(h)

    # ------------------------------------------------------------------
    # try a small exact MILP (if time permits)
    elapsed = time.time() - start
    if budget_s - elapsed > 70:
        try:
            h_bin, bound = _solve_milp(N, time_limit=min(300, budget_s - elapsed - 30))
            if bound < best_score:
                best_h = h_bin.astype(float)
                best_score = bound
        except Exception:
            pass

    # ------------------------------------------------------------------
    # continuous sub‑gradient descent on the current best
    elapsed = time.time() - start
    remain = budget_s - elapsed
    if remain > 5.0:
        h_cont, sc = _continuous_optimize(best_h, remain, rng)
        if sc < best_score:
            best_h, best_score = h_cont, sc

    # ------------------------------------------------------------------
    # final safety projection (keeps integral exactly 1)
    best_h = _project_capped_simplex(best_h, N / 2)
    final_score = _c5_score(best_h)

    return best_h.tolist(), float(final_score), int(N)
