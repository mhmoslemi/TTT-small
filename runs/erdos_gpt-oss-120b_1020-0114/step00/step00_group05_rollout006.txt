def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    # Determine n_points and initial vector
    try:
        init = np.array(initial_h_values, copy=True)
        n = init.shape[0]
    except Exception:
        n = 70
        init = None
    dx = 2.0 / n
    target_sum = n / 2.0

    # projection onto [0,1] and sum constraint
    def project(v):
        lo = -1.0
        hi = 1.0
        for _ in range(50):
            mu = (lo + hi) * 0.5
            x = np.clip(v - mu, 0.0, 1.0)
            s = x.sum()
            if s > target_sum:
                lo = mu
            else:
                hi = mu
        mu = (lo + hi) * 0.5
        return np.clip(v - mu, 0.0, 1.0)

    def compute_cmax(h):
        cs = np.correlate(h, 1.0 - h, mode='full')
        return cs.max() * dx, cs

    # Prepare initial best solution
    if init is not None:
        h0 = project(init)
    else:
        h0 = project(np.random.rand(n))
    best_h = h0.copy()
    best_c, _ = compute_cmax(best_h)

    # gradient descent routine
    def improve(h_start, max_steps):
        h = h_start.copy()
        c_val, cs = compute_cmax(h)
        n_local = n
        for step in range(max_steps):
            # stop if out of time
            if time.time() - start > budget_s - 0.5:
                break
            max_idx = np.argmax(cs)
            shift = max_idx - (n_local - 1)
            # compute gradient
            inds = np.arange(n_local)
            if shift >= 0:
                term1 = np.where(inds + shift < n_local, 1.0 - h[inds + shift], 1.0)
                term2 = np.where(inds - shift >= 0, h[inds - shift], 0.0)
            else:
                s = -shift
                term1 = np.where(inds - s >= 0, 1.0 - h[inds - s], 1.0)
                term2 = np.where(inds + s < n_local, h[inds + s], 0.0)
            grad = (term1 - term2) * dx
            # learning rate schedule
            lr = 0.5 * (1.0 - step / max_steps)
            cand = project(h - lr * grad)
            cand_c, cand_cs = compute_cmax(cand)
            # accept if better, else possibly accept with small prob
            if cand_c < c_val:
                h, c_val, cs = cand, cand_c, cand_cs
            else:
                # Simulated annealing acceptance
                T = 0.02 * (1.0 - step / max_steps) + 1e-4
                if np.random.rand() < math.exp(-(cand_c - c_val) / T):
                    h, c_val, cs = cand, cand_c, cand_cs
                else:
                    # keep old h but we need cs for old h for next iteration; already have cs
                    pass
        return h, c_val

    # use initial solution first
    cur_h, cur_c = improve(best_h, 5000)
    if cur_c < best_c:
        best_h, best_c = cur_h, cur_c

    # additional random restarts
    restart = 0
    while time.time() - start < budget_s - 1.0:
        restart += 1
        # random start
        rand_vec = np.random.rand(n)
        h_start = project(rand_vec)
        # run a shorter improvement
        steps = 3000

        if c_new < best_c:
            best_h, best_c = h_new, c_new

    # Final compute best_c precisely (in case)
    best_c, _ = compute_cmax(best_h)
    return best_h, float(best_c), n

        term2 = np.where(inds + d < n, h[inds + d], 0.0)

import time
import math
import numpy as np

def _project_onto_bounds(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the box [0,1] and the hyperplane sum(x)=target."""
    lo, hi = -1.0, 1.0                      # bounds for the Lagrange multiplier
    for _ in range(60):
        mu = (lo + hi) * 0.5
        x = np.clip(v - mu, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mu
        else:
            hi = mu
    mu = (lo + hi) * 0.5
    return np.clip(v - mu, 0.0, 1.0)


def _cmax_and_corr(h: np.ndarray, dx: float):
    """Return max_{k} ∫h(x)(1‑h(x+k))dx and the full correlation vector."""
    cs = np.correlate(h, 1.0 - h, mode="full")
    return cs.max() * dx, cs


def _gradient_for_shift(h: np.ndarray, shift: int, dx: float):
    """Gradient of the worst overlap term w.r.t. h."""
    n = h.shape[0]
    inds = np.arange(n)
    if shift >= 0:
        term1 = np.where(inds + shift < n, 1.0 - h[inds + shift], 1.0)
        term2 = np.where(inds - shift >= 0, h[inds - shift], 0.0)
    else:
        d = -shift
        term1 = np.where(inds - d >= 0, 1.0 - h[inds - d], 1.0)
        term2 = np.where(inds + d < n, h[inds + d], 0.0)
    return (term1 - term2) * dx


def _improve(
    h_start: np.ndarray,
    dx: float,
    target_sum: float,
    max_steps: int,
    start_time: float,
    budget: float,
):
    """Run a projected (simulated‑annealing) descent from h_start."""
    h = h_start.copy()
    c_val, cs = _cmax_and_corr(h, dx)
    n = h.shape[0]

    for step in range(max_steps):
        if time.time() - start_time > budget - 0.5:
            break

        max_idx = int(np.argmax(cs))
        shift = max_idx - (n - 1)

        grad = _gradient_for_shift(h, shift, dx)
        lr = 0.5 * (1.0 - step / max_steps)          # linear decay
        cand = _project_onto_bounds(h - lr * grad, target_sum)

        cand_c, cand_cs = _cmax_and_corr(cand, dx)

        if cand_c < c_val:
            h, c_val, cs = cand, cand_c, cand_cs
        else:
            # occasional uphill moves (simulated annealing)
            T = 0.02 * (1.0 - step / max_steps) + 1e-4
            if np.random.rand() < math.exp(-(cand_c - c_val) / T):
                h, c_val, cs = cand, cand_c, cand_cs

    return h, c_val


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step function h, its C₅ bound, and the number of sample points."""
    np.random.seed(seed)
    start = time.time()

    # --------------------------------------------------------------
    # problem size and optional parent construction
    # --------------------------------------------------------------
    try:
        init_arr = np.array(initial_h_values, copy=True)   # provided by the harness
        n = init_arr.shape[0]
    except Exception:
        n = 70
        init_arr = None

    dx = 2.0 / n
    target_sum = n / 2.0                     # because ∫₀²h = 1

    # start from the parent if available, otherwise random feasible point
    if init_arr is not None:
        h0 = _project_onto_bounds(init_arr, target_sum)
    else:
        h0 = _project_onto_bounds(np.random.rand(n), target_sum)

    best_h = h0.copy()
    best_c, _ = _cmax_and_corr(best_h, dx)

    # --------------------------------------------------------------
    # long deterministic improvement from the best known start
    # --------------------------------------------------------------
    h_opt, c_opt = _improve(
        best_h, dx, target_sum, max_steps=200_000, start_time=start, budget=budget_s
    )
    if c_opt < best_c:
        best_h, best_c = h_opt, c_opt

    # --------------------------------------------------------------
    # additional random restarts while time permits
    # --------------------------------------------------------------
    while time.time() - start < budget_s - 1.0:
        rnd = _project_onto_bounds(np.random.rand(n), target_sum)
        h_new, c_new = _improve(
            rnd, dx, target_sum, max_steps=50_000, start_time=start, budget=budget_s
        )
        if c_new < best_c:
            best_h, best_c = h_new, c_new

    # final precise evaluation
    best_c, _ = _cmax_and_corr(best_h, dx)

    return best_h, float(best_c), n
