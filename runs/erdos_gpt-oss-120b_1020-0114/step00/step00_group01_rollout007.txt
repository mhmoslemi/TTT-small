def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    start = time.time()
    # Set target length
    N = 200
    # Build initial h
    if 'initial_h_values' in globals():
        base = np.array(initial_h_values, dtype=np.float64)
        # convert to binary roughly
        base_bin = (base >= 0.5).astype(np.float64)
        # tile to length N
        repeats = int(np.ceil(N / len(base_bin)))
        h = np.tile(base_bin, repeats)[:N]
        # adjust to exact weight
        current_ones = int(h.sum())
        target_ones = N // 2
        if current_ones < target_ones:
            zeros_idxs = np.flatnonzero(h == 0)
            add = target_ones - current_ones
            choose = rng.choice(zeros_idxs, add, replace=False)
            h[choose] = 1.0
        elif current_ones > target_ones:
            ones_idxs = np.flatnonzero(h == 1)
            remove = current_ones - target_ones
            choose = rng.choice(ones_idxs, remove, replace=False)
            h[choose] = 0.0
    else:
        h = np.zeros(N, dtype=np.float64)
        ones = rng.choice(N, N//2, replace=False)
        h[ones] = 1.0

    cur = h.copy()
    cur_val = compute_c5(cur)
    best = cur.copy()
    best_val = cur_val

    # Simulated annealing parameters
    max_time = budget_s * 0.95
    # temperature schedule: start high, linearly decays
    T0 = 0.1
    iter_cnt = 0
    last_report = 0
    # we can produce a loop
    while time.time() - start < max_time:
        # progress fraction
        elapsed = time.time() - start
        progress = elapsed / max_time
        T = max(1e-4, T0 * (1 - progress))
        # choose random swap
        one_idxs = np.flatnonzero(cur == 1)
        zero_idxs = np.flatnonzero(cur == 0)
        # random pick
        i = rng.choice(one_idxs)
        j = rng.choice(zero_idxs)
        # apply swap
        cur[i] = 0.0
        cur[j] = 1.0
        new_val = compute_c5(cur)
        delta = new_val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / T):
            # accept
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cur.copy()
        else:
            # reject, revert
            cur[i] = 1.0
            cur[j] = 0.0
        iter_cnt += 1
        # optional early stop if best_val < target
        if best_val < 0.3808:
            break

    return best, best_val, N

import time, math
import numpy as np

def _compute_c5(vec: np.ndarray) -> float:
    """Return the overlap integral C5 for a discretised step function."""
    n = vec.size
    dx = 2.0 / n
    # correlation with zero padding (full mode)
    corr = np.correlate(vec, 1.0 - vec, mode='full')
    return float(np.max(corr) * dx)


def run(seed: int = 42, budget_s: int = 1100, **_):
    """
    Search for a step function h on [0,2] (discretised) that lowers the
    Erdős minimum‑overlap constant C5.

    Returns
    -------
    h_values : np.ndarray
        The best function found (values in [0,1]).
    c5_bound : float
        Corresponding C5 bound.
    n_points : int
        Number of discretisation points (len(h_values)).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    max_time = budget_s * 0.95                     # stay safe inside the budget

    # ------------------------------------------------------------------ #
    # Initialise the candidate
    # ------------------------------------------------------------------ #
    if 'initial_h_values' in globals():
        # use the supplied construction as a starting point
        h = np.asarray(initial_h_values, dtype=np.float64).copy()
        n = h.size
        target_sum = n / 2.0                         # because ∫h = 1, dx = 2/n

        diff = target_sum - h.sum()
        if abs(diff) > 1e-12:
            # redistribute the tiny deficit/excess uniformly (still respects bounds)
            # we perform a simple water‑filling adjustment
            while abs(diff) > 1e-12:
                idx = rng.integers(n)
                if diff > 0:                         # need to add mass
                    add = min(diff, 1.0 - h[idx])
                else:                                # need to remove mass
                    add = max(diff, -h[idx])
                if abs(add) < 1e-15:
                    continue
                h[idx] += add
                diff -= add
    else:
        # a generic binary start, length 200 (good trade‑off between speed and resolution)
        n = 200
        h = np.zeros(n, dtype=np.float64)
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1.0

    # ------------------------------------------------------------------ #
    # Ensure feasibility (exact integral, bounds)
    # ------------------------------------------------------------------ #
    target_sum = n / 2.0
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # simple pairwise mass redistribution to hit the exact target
        while abs(cur_sum - target_sum) > 1e-12:
