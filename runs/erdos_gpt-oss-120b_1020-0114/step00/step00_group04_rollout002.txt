def propose_move(h):
    n = len(h)
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return None
    if np.random.rand() < 0.5:
        donor, recipient = i, j
    else:
        donor, recipient = j, i
    max_delta = min(h[donor], 1 - h[recipient])
    if max_delta <= 0:
        return None
    delta = np.random.rand() * max_delta
    h_new = h.copy()
    h_new[donor] -= delta
    h_new[recipient] += delta
    return h_new
```

def random_feasible(n):
    target = n/2
    # start with all zero
    h = np.zeros(n)
    # allocate mass iteratively
    remaining = target
    indices = list(range(n))
    while remaining > 1e-12 and indices:
        # sample random weights for remaining indices
        draws = np.random.rand(len(indices))
        # scale to remaining
        draws = draws / draws.sum() * remaining
        # assign but cap at 1
        for idx, val in zip(indices[:], draws):
            add = min(val, 1 - h[idx])
            h[idx] += add
            remaining -= add
        # remove saturated indices
        indices = [i for i in indices if h[i] < 1 - 1e-12]
    # after loop may have tiny remaining due to rounding; distribute evenly among not full entries
    if remaining > 1e-12:
        for i in indices:
            add = min(remaining / len(indices), 1 - h[i])
            h[i] += add
            remaining -= add
    return h
```

import time
import numpy as np

def _c5_of(h):
    """Return the C5 value for a discrete h (numpy array)."""
    # correlation (full) between h and (1-h)
    conv = np.correlate(h, 1.0 - h, mode="full")
    # max overlap, scaled by dx = 2/n
    return conv.max() * (2.0 / h.size)


def _make_feasible(v):
    """Force a vector into the box [0,1] while keeping its sum = target."""
    n = v.size
    target = n / 2.0
    v = np.clip(v, 0.0, 1.0)
    current = v.sum()
    # simple water-filling to adjust the sum
    if abs(current - target) > 1e-12:
        diff = target - current
        # indices that can still move in the needed direction
        if diff > 0:
            # need to increase some entries
            mask = v < 1.0 - 1e-12
        else:
            mask = v > 1e-12
        inds = np.flatnonzero(mask)
        if inds.size == 0:
            return v  # should not happen
        # distribute the difference uniformly (will stay feasible because of mask)
        add = diff / inds.size
        for i in inds:
            v[i] = np.clip(v[i] + add, 0.0, 1.0)
        # final correction if rounding error remains
        residual = target - v.sum()
        if abs(residual) > 1e-12:
            # spread residual over first feasible index
            for i in inds:
                avail = 1.0 - v[i] if residual > 0 else v[i]
                step = np.clip(residual, -avail, avail)
                v[i] += step
                residual -= step
                if abs(residual) < 1e-12:
                    break
    return v


def _random_move(h, rng):
    """Return a new h after moving a random amount from one index to another."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return None
    # random direction
    if rng.random() < 0.5:
        donor, recipient = i, j
    else:
        donor, recipient = j, i
    max_delta = min(h[donor], 1.0 - h[recipient])
    if max_delta <= 0.0:
        return None
    delta = rng.random() * max_delta
    h_new = h.copy()
    h_new[donor] -= delta
    h_new[recipient] += delta
    return h_new


def _heuristic_move(h, rng):
    """Create a move guided by the currently worst shift."""
    n = h.size
    conv = np.correlate(h, 1.0 - h, mode="full")
    idx_max = int(conv.argmax())
    shift = idx_max - (n - 1)           # shift can be negative
    # identify contributors for this shift
    if shift >= 0:
        i_vals = np.arange(0, n - shift)
        contrib = h[i_vals] * (1.0 - h[i_vals + shift])
        donor = i_vals[np.argmax(contrib)]
        recipient = donor + shift
    else:
        s = -shift
        i_vals = np.arange(s, n)
        contrib = h[i_vals] * (1.0 - h[i_vals - s])
        donor = i_vals[np.argmax(contrib)]
        recipient = donor - s
    # ensure feasibility of move direction
    if h[donor] <= 0.0 or h[recipient] >= 1.0:
        return None
    max_delta = min(h[donor], 1.0 - h[recipient])
    if max_delta <= 0.0:
        return None
    delta = rng.random() * max_delta
    h_new = h.copy()
    h_new[donor] -= delta
    h_new[recipient] += delta
    return h_new
