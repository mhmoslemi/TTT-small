    # target: sum required
    # Perform projection onto [0,1] and sum target.
    n = len(v)
    # initial clipping?

    # 1. clip to [0,1].

    s = x.sum()
    if s > target:
        # need to reduce sum -> subtract from entries >0
        # sort descending
        excess = s - target
        # We'll keep indices sorted by x descending
        idx = np.argsort(-x)  # descending
        for i in idx:
            if excess <= 0:
                break
            reducible = x[i] - 0  # lower bound
            delta = min(reducible, excess)
            x[i] -= delta
            excess -= delta
    elif s < target:
        deficit = target - s
        idx = np.argsort(x)  # ascending
        for i in idx:
            if deficit <= 0:
                break
            increase = 1 - x[i]
            delta = min(increase, deficit)
            x[i] += delta
            deficit -= delta
    # now sum == target (or within epsilon)
    return x

def hill_climb_binary(N, max_iters, seed):
    rng = np.random.default_rng(seed)
    ones = N // 2
    # random initial vector
    idx = rng.choice(N, size=ones, replace=False)
    x = np.zeros(N, dtype=int)
    x[idx] = 1
    # compute c with FFT for speed
    def compute_min_corr(x):
        fx = np.fft.fft(x)
        c = np.fft.ifft(fx*np.conj(fx)).real
        # rounding may produce tiny negative values; correct
        c = np.rint(c).astype(int)
        # exclude shift 0
        min_c = int(c[1:].min())
        return min_c

    cur_min = compute_min_corr(x)
    best_x = x.copy()
    best_min = cur_min

    # lists for fast indices of ones and zeros
    ones_idx = np.flatnonzero(x)
    zeros_idx = np.flatnonzero(1 - x)

    # temperature schedule
    T0 = 5.0
    for it in range(max_iters):
        # propose swap
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        # swap in x copy
        x[i] = 0
        x[j] = 1
        # update indices
        # compute min_corr new
        new_min = compute_min_corr(x)
        delta = new_min - cur_min
        T = T0 * (1 - it / max_iters)

        if delta > 0:

        else:
            if T > 1e-8 and rng.random() < np.exp(delta / T):
                accept = True
        if accept:
            cur_min = new_min
            # update indices after swap

def project_onto_box_simplex(v, target):
    # clip to [0,1]
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if total > target + 1e-12:
        # need to reduce sum
        excess = total - target
        # reduce from largest values first
        idx_desc = np.argsort(-v)
        for i in idx_desc:
            if excess <= 1e-12:
                break
            reducible = v[i] - 0.0
            delta = min(reducible, excess)
            v[i] -= delta
            excess -= delta
    elif total < target - 1e-12:
        deficit = target - total
        idx_asc = np.argsort(v)
        for i in idx_asc:
            if deficit <= 1e-12:
                break
            increase = 1.0 - v[i]
            delta = min(increase, deficit)
            v[i] += delta
            deficit -= delta
    return v

def hill_climb_binary(N, max_iters, rng):
    ones = N // 2
    # random initial with exactly ones set
    idx = rng.choice(N, size=ones, replace=False)
    x = np.zeros(N, dtype=int)
    x[idx] = 1
    cur_min = compute_min_corr_binary(x)
    best_x = x.copy()
    best_min = cur_min

    T0 = 5.0
    for it in range(max_iters):
        i, j = propose_swap(x)
        # swap
        x[i], x[j] = x[j], x[i]
        new_min = compute_min_corr_binary(x)
        delta = new_min - cur_min
        # temperature schedule
        T = T0 * (1 - it / max_iters)
        accept = False
        if delta > 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(delta / T):
                accept = True
        if accept:
            cur_min = new_min
            if cur_min > best_min:
                best_min = cur_min
                best_x = x.copy()
        else:
            # revert swap
            x[i], x[j] = x[j], x[i]
    return best_x, best_min

def hill_climb_binary(N, max_iters, rng):
    # initialize binary vector
    ones = N // 2
    idx = rng.choice(N, size=ones, replace=False)
    x = np.zeros(N, dtype=int)
    x[idx] = 1
    # compute initial max_overlap
    cur_max = max_overlap_fft(x)   # integer
    cur_c5 = cur_max * (2.0 / N)
    best_x = x.copy()
    best_c5 = cur_c5

    T0 = 5.0
    for it in range(max_iters):
        i, j = propose_swap(x, rng)
        # swap
        x[i], x[j] = x[j], x[i]
        cand_max = max_overlap_fft(x)
        cand_c5 = cand_max * (2.0 / N)
        delta = cand_c5 - cur_c5  # improvement negative means lower C5
        T = T0 * (1 - it / max_iters)
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):  # negative exponent (since delta >0)
                accept = True
        if accept:
            cur_c5 = cand_c5
            cur_max = cand_max
            if cur_c5 < best_c5:
                best_c5 = cur_c5

        idx = np.argsort(-v)
        for i in idx:
            reducible = v[i] - 0.0
            if reducible <= 0:
                continue
            delta = min(reducible, excess)
            v[i] -= delta
            excess -= delta
            if excess <= 1e-12:
                break
    elif total < target:
        deficit = target - total
        idx = np.argsort(v)
        for i in idx:
            increase = 1.0 - v[i]
            if increase <= 0:
                continue
            delta = min(increase, deficit)
            v[i] += delta
            deficit -= delta
            if deficit <= 1e-12:
                break
    # final adjustment for small diff
    diff = target - v.sum()
    if abs(diff) > 1e-10:
        # find any index where we can adjust

            if diff > 0 and v[i] < 1.0:

                v[i] += add
                diff -= add
            elif diff < 0 and v[i] > 0.0:
                sub = min(-diff, v[i])
                v[i] -= sub
                diff += sub
            if abs(diff) <= 1e-10:
                break

import time
import numpy as np

# ----- helper functions ---------------------------------------------------


def compute_c5(h):
    """Exact C5 using the definition from the evaluator."""
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode="full")
    return float(corr.max() * dx)


def max_overlap_fft(h):
    """Maximum overlap ∫ h(x)(1‑h(x+k))dx (without dx factor) using FFT."""
    N = h.shape[0]
    L = 2 * N - 1
    hf = np.fft.fft(h, L)
    bf = np.fft.fft(1 - h, L)
    c = np.fft.ifft(hf * bf).real
    # rounding eliminates tiny numerical noise for binary vectors
    return float(np.rint(c.max()))


def project_onto_box_simplex(v, target):
    """Project v onto the set {0≤x_i≤1, sum x_i = target}."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if total > target + 1e-12:
        excess = total - target
        idx = np.argsort(-v)  # descend
        for i in idx:
            if excess <= 0:
                break
            reducible = v[i]  # lower bound is 0
            delta = min(reducible, excess)
            v[i] -= delta
            excess -= delta
    elif total < target - 1e-12:
        deficit = target - total
        idx = np.argsort(v)   # ascend
        for i in idx:
            if deficit <= 0:
                break
            inc = 1.0 - v[i]
            delta = min(inc, deficit)
            v[i] += delta
            deficit -= delta
    # tiny correction for floating errors
    diff = target - v.sum()
    if abs(diff) > 1e-10:
        for i in range(v.shape[0]):
            if diff > 0 and v[i] < 1.0:
                add = min(diff, 1.0 - v[i])
                v[i] += add
                diff -= add
            elif diff < 0 and v[i] > 0.0:
                sub = min(-diff, v[i])
                v[i] -= sub
                diff += sub
            if abs(diff) <= 1e-10:
                break
    return v


def propose_swap(x, rng):
    """Return indices i,j with x[i]!=x[j] (i has 1, j has 0)."""
    N = x.shape[0]
    while True:
        i = rng.integers(N)
        j = rng.integers(N)
        if x[i] != x[j]:
            return i, j


def hill_climb_binary(N, max_iters, rng):
    """Simulated‑annealing hill climb on a binary vector (0/1)."""
    ones = N // 2
    idx = rng.choice(N, size=ones, replace=False)
    x = np.zeros(N, dtype=int)
    x[idx] = 1

    cur_max = max_overlap_fft(x)
    cur_c5 = cur_max * (2.0 / N)
    best_x = x.copy()
    best_c5 = cur_c5

    T0 = 5.0
    for it in range(max_iters):
        i, j = propose_swap(x, rng)
        x[i], x[j] = x[j], x[i]               # perform swap
        cand_max = max_overlap_fft(x)
        cand_c5 = cand_max * (2.0 / N)

        delta = cand_c5 - cur_c5               # negative → improvement
        T = T0 * (1.0 - it / max_iters)
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True

        if accept:
            cur_c5 = cand_c5
            cur_max = cand_max
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_x = x.copy()
        else:
            x[i], x[j] = x[j], x[i]           # revert

    return best_x.astype(float), best_c5


def refine_continuous(h_init, max_iters, lr0, rng):
    """Sub‑gradient descent on the max‑overlap objective with projection."""
    N = h_init.shape[0]
    target = N / 2.0
    h = project_onto_box_simplex(h_init.copy(), target)

    best_h = h.copy()
    best_c5 = compute_c5(h)

    for it in range(max_iters):
        # correlation (full) and shift attaining the maximum
        corr = np.correlate(h, 1 - h, mode="full")
        idx_max = int(np.argmax(corr))
        shift = idx_max - (N - 1)          # shift ∈ [-(N-1), N-1]

        # gradient of the maximal term
        grad = np.zeros(N)
        if shift >= 0:
            if shift < N:
                grad[:N - shift] += 1.0 - h[shift:]
                grad[shift:] -= h[:N - shift]
        else:
            t = -shift
            grad[t:] += 1.0 - h[:N - t]
            grad[:N - t] -= h[t:]

        # step size schedule
        lr = lr0 * (1.0 - it / max_iters)
        h_new = h - lr * grad
        h_new = project_onto_box_simplex(h_new, target)

        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

        # accept‑or‑reject (allow occasional non‑improving moves)
        if c5_new < compute_c5(h) or rng.random() < 0.01:
            h = h_new

    return best_h, best_c5


def thue_morse(N):
    """Generate the Thue–Morse binary sequence of length N (N a power of two)."""
    # parity of binary digit sum
    seq = np.fromiter(((bin(i).count("1") & 1) for i in range(N)), dtype=int, count=N)
    return seq.astype(float)
