def adjust_sum(h, target, rng):
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        if diff > 0:
            # needs increase
            candidates = np.where(h < 1 - 1e-12)[0]
            if len(candidates) == 0:
                break
            i = rng.choice(candidates)
            inc = min(diff, 1 - h[i])
            h[i] += inc
            diff -= inc
        else:
            candidates = np.where(h > 1e-12)[0]
            if len(candidates) == 0:
                break
            i = rng.choice(candidates)
            dec = min(-diff, h[i])
            h[i] -= dec
            diff += dec
    return h

import time
import math
import numpy as np

def _evaluate_h(h: np.ndarray, dx: float) -> float:
    """Return max_k ∫ h(x)(1-h(x+k)) dx for the discretised h."""
    # np.correlate gives the discrete convolution sum_i h[i] * (1-h[i+lag])
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _adjust_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h in-place so that sum(h) == target while keeping 0≤h≤1."""
    # Very small adjustments; the diff after interpolation is tiny.
    diff = target - h.sum()
    # tolerance to stop the loop
    while abs(diff) > 1e-12:
        if diff > 0.0:
            # need to increase sum: pick a position not already at 1
            cand = np.where(h < 1.0 - 1e-12)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            inc = min(diff, 1.0 - h[i])
            h[i] += inc
            diff -= inc
        else:
            # need to decrease sum: pick a position not already at 0
            cand = np.where(h > 1e-12)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            dec = min(-diff, h[i])
            h[i] -= dec
            diff += dec
    return h


def _optimize_h(initial_h: np.ndarray, rng: np.random.Generator,
                time_limit: float) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing maximising the minimum autocorrelation.

    Returns the best vector encountered and its C5 value.
    """
    h = initial_h.copy()
    n = h.shape[0]
    dx = 2.0 / n
    target_sum = n / 2.0

    # current correlation and value
    corr = np.correlate(h, 1.0 - h, mode="full")
    cur_val = float(corr.max() * dx)

    best_h = h.copy()
    best_val = cur_val

    start = time.time()
    max_iter = 3_000_000          # generous upper bound; actual stops by time
    step_scale = 0.5               # maximal transferred mass in a move
    T0 = 0.001                     # initial temperature for SA

    for it in range(max_iter):
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break

        # locate the lag that currently gives the maximal overlap
        idx_max = int(np.argmax(corr))
        lag = idx_max - (n - 1)          # lag can be negative or positive

        # propose a move
        if lag != 0 and rng.random() < 0.85:
            # targeted move that directly reduces the worst lag
            if lag > 0:
                # donor i, receiver i+lag
                if n - lag <= 1:
                    continue
                i = rng.integers(0, n - lag)
                j = i + lag
            else:
                ll = -lag
                if n - ll <= 1:
                    continue
                i = rng.integers(ll, n)
                j = i - ll
            if h[i] <= 1e-12 or h[j] >= 1.0 - 1e-12:
                # fallback to a random move if limits block us
                i = j = None
        else:
            i = j = None

        if i is None:
            # generic random donor/receiver
            for _ in range(5):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j:
                    continue
                if h[i] > 1e-12 and h[j] < 1.0 - 1e-12:
                    break
            else:
                continue

        delta_max = min(step_scale, h[i], 1.0 - h[j])
        if delta_max <= 0.0:
            continue
        delta = delta_max * rng.random()

        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta

        # evaluate new candidate
        new_corr = np.correlate(h_new, 1.0 - h_new, mode="full")
        new_val = float(new_corr.max() * dx)

        # SA acceptance
        T = T0 * (1.0 - it / max_iter)
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-14)):
            h = h_new
            corr = new_corr
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()

    # final tiny correction to satisfy the exact integral constraint
    best_h = _adjust_sum(best_h, target_sum, rng)
    best_val = _evaluate_h(best_h, dx)
    return best_h, best_val


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] for the Erdős C5 problem.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Choose a modest resolution (keeps evaluation cheap while giving flexibility)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialise h:
    init = globals().get("initial_h_values", None)
    if init is not None and isinstance(init, np.ndarray):
        if init.shape[0] == n_points:
            h0 = init.astype(float).copy()
        else:
            # linear interpolation from the provided vector
            old_n = init.shape[0]
            x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
            h0 = np.interp(x_new, x_old, init.astype(float))
    else:
        # No parent construction: start from a random binary pattern
        h0 = np.zeros(n_points, dtype=float)
        ones = n_points // 2
        idx = rng.choice(n_points, ones, replace=False)
        h0[idx] = 1.0

    # Clip to ``[0,1]`` and enforce the integral constraint exactly
    np.clip(h0, 0.0, 1.0, out=h0)
    h0 = _adjust_sum(h0, target_sum, rng)

    # Allocate most of the budget to the optimisation loop
    remaining = max(0.0, budget_s - (time.time() - start_time) - 0.3)
    best_h, best_val = _optimize_h(h0, rng, remaining)

    # Final sanity checks (they should already hold)
    np.clip(best_h, 0.0, 1.0, out=best_h)
    best_h = _adjust_sum(best_h, target_sum, rng)
    best_val = _evaluate_h(best_h, dx)

    # Return a plain python list so the harness can serialise it easily
    return (best_h.tolist(), best_val, n_points)
