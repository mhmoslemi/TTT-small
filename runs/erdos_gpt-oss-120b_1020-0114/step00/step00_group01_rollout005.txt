import numpy as np
import time

def compute_c5(h: np.ndarray) -> float:
    """Return max overlap integral for a step function h sampled on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def build_h(cp: int, cq: int, cr: int, a: float, n: int) -> np.ndarray:
    """Construct symmetric five‑step vector given integer block sizes."""
    parts = []
    if cp:
        parts.append(np.zeros(cp, dtype=float))
    if cq:
        parts.append(np.full(cq, a, dtype=float))
    if cr:
        parts.append(np.ones(cr, dtype=float))
    if cq:
        parts.append(np.full(cq, a, dtype=float))
    if cp:
        parts.append(np.zeros(cp, dtype=float))
    h = np.concatenate(parts)
    # safety: ensure length matches n
    if h.shape[0] != n:
        raise ValueError("Length mismatch in build_h")
    return h


def enumerate_symmetric_shapes(n: int, time_deadline: float):
    """
    Exhaustively search symmetric five‑step shapes (0‑a‑1‑a‑0)
    respecting the integral constraint exactly.
    Returns the best h and its C5 value found before the deadline.
    """
    target = n // 2                       # required sum of h entries
    best_c5 = np.inf
    best_h = None

    for cr in range(0, target + 1):
        # minimal cq that allows a ≤ 1  (ceil((target‑cr)/2))
        min_cq = (target - cr + 1) // 2
        if min_cq < 0:
            min_cq = 0
        max_cq = (n - cr) // 2
        if min_cq > max_cq:
            continue
        for cq in range(min_cq, max_cq + 1):
            rem = n - 2 * cq - cr
            if rem < 0 or (rem & 1):
                continue
            cp = rem // 2
            if cq == 0:
                if cr != target:
                    continue
                a = 0.0
            else:

                if a < 0.0 or a > 1.0:

            h = build_h(cp, cq, cr, a, n)
            c5 = compute_c5(h)
            if c5 < best_c5:

                best_h = h

            break
    return best_h, best_c5


def hill_climb(h_init: np.ndarray, time_deadline: float):
    """
    Random two‑coordinate swaps preserving the total sum.
    Accept only improvements (simple greedy hill climbing).
    """
    h = h_init.copy()
    n = h.shape[0]
    best = compute_c5(h)
    rng = np.random.default_rng()
    while time.time() < time_deadline:
        i, j = rng.choice(n, size=2, replace=False)
        vi, vj = h[i], h[j]
        # feasible epsilon range for vi+eps, vj‑eps staying in [0,1]
        eps_lo = max(-vi, vj - 1.0)
        eps_hi = min(1.0 - vi, vj)
        if eps_lo >= eps_hi:
            continue
        eps = rng.uniform(eps_lo, eps_hi)
        h[i] = vi + eps
        h[j] = vj - eps
        c5 = compute_c5(h)
        if c5 < best:
            best = c5
        else:
            # revert
            h[i] = vi
            h[j] = vj
    return h, best
