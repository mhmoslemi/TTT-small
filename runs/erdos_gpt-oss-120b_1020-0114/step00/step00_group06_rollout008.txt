def project_onto_box_simplex(v, target):
    lo = -1e6; hi = 1e6
    for _ in range(50):
        lam = (lo+hi)/2.0
        x = np.clip(v - lam, 0, 1)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo+hi)/2.0, 0, 1)

def project_onto_box_simplex(v, target, bound_low=0.0, bound_high=1.0):
    # Clip values to bounds after adjusting with lambda?
    # Approach: find λ such that sum_i clip(v_i - λ, 0, 1) == target.
    # Since the function sum_i clip(v_i - λ, 0, 1) is monotonic decreasing in λ.
    lo = -1e6
    hi = 1e6
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, bound_low, bound_high)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    # Return final projection
    return np.clip(v - (lo+hi)/2.0, bound_low, bound_high)

def subgradient_max(h):
    corr = compute_corr(h)
    max_val = corr.max()
    idxs = np.flatnonzero(corr == max_val)
    weights = np.full(idxs.shape, 1.0 / len(idxs), dtype=float)

            grad[shift:] -= weight * h[:N-shift]

            grad[d:] += weight * w[:N-d]

def project_onto_box_simplex(v, target):
    lo, hi = -1e6, 1e6
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

    corr = compute_corr(h)
    m = corr.max()
    w = 1.0 - h
    # soft‑max weights, numerically stable
    expb = np.exp(beta * (corr - m))
    weights = expb / expb.sum()

    h0 = project_onto_box_simplex(h0, N / 2.0)
    best_h = h0.copy()
    best_val = compute_corr(best_h).max()
    dx = 2.0 / N

    start_time = time.time()

    # --- first local optimisation from the given start -------------
    remaining = budget_s - (time.time() - start_time) - 0.2
    if remaining > 0.0:
        opt_h, opt_val = improve_h(
            h0, max_iter=8000, time_limit=remaining, init_alpha=0.5
        )
        if opt_val < best_val:
            best_val, best_h = opt_val, opt_h.copy()

    # --- stochastic restarts ----------------------------------------
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 0.2:
            break

        # new random feasible start
        h_start = random_feasible(N, N / 2.0)
        remaining = budget_s - (time.time() - start_time) - 0.2
        if remaining <= 0.0:
            break

        opt_h, opt_val = improve_h(
            h_start, max_iter=4000, time_limit=remaining, init_alpha=0.5
        )
        if opt_val < best_val:
            best_val, best_h = opt_val, opt_h.copy()

    # --- final projection and objective ------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_simplex(best_h, N / 2.0)
    final_c5 = np.max(compute_corr(best_h)) * dx

    return best_h, float(final_c5), N
