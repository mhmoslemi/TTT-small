import time, math, random
import numpy as np

def _project_to_bounds(v, target):
    """Project vector v onto the hyperplane sum=v_target with box bounds [0,1]."""
    x = np.clip(v, 0.0, 1.0)
    eps = 1e-12
    for _ in range(500):
        s = x.sum()
        diff = target - s
        if abs(diff) <= eps:
            break
        free = (x > eps) & (x < 1.0 - eps)
        free_idx = np.where(free)[0]
        if free_idx.size == 0:
            # all variables are at a bound; distribute over all then clip
            free_idx = np.arange(x.size)
        delta = diff / free_idx.size
        x[free_idx] += delta
        x = np.clip(x, 0.0, 1.0)
    # final tiny fix if needed
    s = x.sum()
    if abs(s - target) > 1e-9:
        correction = target - s
        # distribute to a non‑boundary entry if possible
        for idx in range(x.size):
            if eps < x[idx] < 1.0 - eps:
                x[idx] = min(1.0, max(0.0, x[idx] + correction))
                break
    return x

def _c5_of(h, dx):
    """Return the overlap value C5 for a given step function h."""
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)
