import numpy as np, time, math

def project_to_box_sum(y, S):
    # bisection to find lambda such that sum(clip(y - lambda, 0, 1)) = S
    low = np.min(y) - 1.0
    high = np.max(y) + 1.0
    for _ in range(60):
        mid = (low + high) / 2.0
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > S:
            low = mid
        else:
            high = mid
    return np.clip(y - (low + high) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def gradient_descent(start_h, max_iter=1500, alpha=30.0, step0=0.5, time_limit=None, start_time=None):
    h = start_h.copy()
    n = h.shape[0]
    S = n / 2.0
    best_h = h.copy()
    best_val = compute_c5(h)
    step = step0
    offset = n - 1
    for it in range(max_iter):
        # time check
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        q = np.correlate(h, 1 - h, mode='full')
        max_q = np.max(q)
        w = np.exp(alpha * (q - max_q))
        w_sum = np.sum(w)
        w /= w_sum
        grad = np.ones(n)
        # loops
        for idx, wi in enumerate(w):
            if wi == 0:
                continue
            shift = idx - offset
            if shift >= 0:
                if shift < n:
                    grad[:n - shift] -= wi * h[shift:]
                    grad[shift:] -= wi * h[:n - shift]
            else:
                pos = -shift
                if pos < n:
                    grad[pos:] -= wi * h[:n - pos]
                    grad[:n - pos] -= wi * h[pos:]
        new_h = h - step * grad
        new_h = project_to_box_sum(new_h, S)
        val = compute_c5(new_h)
        if val < best_val:
            best_val = val
            best_h = new_h.copy()
        h = new_h
        step *= 0.995  # decay
        # optionally increase alpha after many iterations ...
    return best_h, best_val

def upsample_vector(v, n):
    old_n = v.shape[0]
    if old_n == n:
        return v.copy()
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, n, endpoint=False)
    new_v = np.interp(x_new, x_old, v)
    # project to sum = n/2 -> S
    S = n / 2.0
    new_v = project_to_box_sum(new_v, S)
    return new_v

import numpy as np
import time
import random

def _project_to_box_sum(y, S):
    """Euclidean projection of y onto {0<=x<=1, sum(x)=S}."""
    low = np.min(y) - 1.0   # yields all ones -> sum = len(y)
    high = np.max(y) + 1.0  # yields all zeros -> sum = 0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > S:
            low = mid
        else:
            high = mid
    return np.clip(y - (low + high) * 0.5, 0.0, 1.0)


def _c5_value(h):
    """Compute the discrete C5 bound for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def _gradient_step(h, alpha, step):
    """One gradient‑descent step using a smooth max (log‑sum‑exp)."""
    n = h.shape[0]
    S = n * 0.5
    # smooth weights
    q = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(q)
    w = np.exp(alpha * (q - m))
    w /= w.sum()
    # gradient of the smooth max
    grad = np.ones(n)
    off = n - 1
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        shift = idx - off
        if shift >= 0:
            if shift < n:
                grad[: n - shift] -= wi * h[shift:]
                grad[shift:]     -= wi * h[: n - shift]
        else:
            pos = -shift
            if pos < n:
                grad[pos:]      -= wi * h[: n - pos]
                grad[: n - pos] -= wi * h[pos:]
    new_h = h - step * grad
    return _project_to_box_sum(new_h, S)


def _optimize_h(start_h, max_iter=1500, alpha=30.0, step0=0.5):
    """Local optimisation from a start point."""
    h = start_h.copy()
    best_h = h.copy()
    best_val = _c5_value(h)
    step = step0
    for _ in range(max_iter):
        h = _gradient_step(h, alpha, step)
        val = _c5_value(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
        step *= 0.995
    return best_h, best_val


def _refine_pairwise(h, cur_val, iters=500):
    """Simple pair‑wise mass transfer hill‑climb."""
    n = h.shape[0]
    best_h, best_val = h.copy(), cur_val
    for _ in range(iters):
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta = min(best_h[i], 1.0 - best_h[j])
        if delta <= 0.0:
            continue
        delta *= np.random.rand() * 0.5
        new_h = best_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        val = _c5_value(new_h)
        if val < best_val:
            best_val, best_h = val, new_h
    return best_h, best_val


def _upsample(v, n):
    """Linear interpolation of a vector to a new length n."""
    old = v.shape[0]
    if old == n:
        return v.copy()
    x_old = np.linspace(0, 2, old, endpoint=False)
    x_new = np.linspace(0, 2, n, endpoint=False)
    nv = np.interp(x_new, x_old, v)
    return _project_to_box_sum(nv, n * 0.5)
