def project_capsimplex(v, S):
    # Clip to [0,1] after shifting.
    # Use bisection between low = min(v)-1 (ensures all clipped to 1) and high = max(v) (ensures all 0).
    low = v.min() - 1.0
    high = v.max()
    for _ in range(50):
        mid = (low + high) / 2.0
        x = np.clip(v - mid, 0, 1)
        total = x.sum()
        if total > S:
            low = mid
        else:
            high = mid
    return np.clip(v - (low+high)/2.0, 0, 1)

def gradient_softmax(h, beta):

    n = arr.shape[0]

def gradient_softmax(h, beta):
    n = h.size
    g = np.correlate(h, 1 - h, mode='full')
    # stable exponentials
    b_g = beta * g
    maxbg = np.max(b_g)
    expb = np.exp(b_g - maxbg)
    p = expb / np.sum(expb)
    grad = np.zeros_like(h)
    for idx, weight in enumerate(p):
        l = idx - (n - 1)
        pos = shift_h(h, l)
        neg = shift_h(h, -l)
        grad += weight * (1.0 - pos - neg)
    return grad

Now compute_c5:

def compute_c5(h, dx):
    g = np.correlate(h, 1 - h, mode='full')
    return np.max(g) * dx

Now project capsimplex:

def project_capsimplex(v, S):
    # projection onto [0,1] box with sum=S
    low = v.min() - 1.0
    high = v.max()
    for _ in range(50):
        mid = (low + high) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        total = x.sum()
        if total > S:
            low = mid
        else:
            high = mid
    return np.clip(v - (low + high) / 2.0, 0.0, 1.0)

            x_init = np.linspace(0, 2, m, endpoint=False) + 1.0/m
            x_target = np.linspace(0, 2, n, endpoint=False) + 1.0/n
            h_interp = np.interp(x_target, x_init, init)
            h_interp = np.clip(h_interp, 0.0, 1.0)
            h_interp = project_capsimplex(h_interp, S)
            starts.append(h_interp)

        # for each start apply continuous optimization then binary hill climbing
        for h0 in starts:
            # allocate time slice for this start
            time_elapsed = time.time() - start_all
            time_left = total_budget - time_elapsed
            if time_left <= 0:
                break
            # allocate portion
            sub_budget = time_left * 0.3  # allocate 30% for this start
            # continuous optimization
            h_cont, score_cont = optimize_continuous(h0, n, dx, sub_budget * 0.6)
            # keep best
            if score_cont < best_score:

            t_rem = sub_budget * 0.4

            if score_bin < best_score:
                best_score = score_bin
                best_h = h_bin_opt.copy()
                best_n = n

    # final projection exact sum
    if best_h is not None:
        best_h = project_capsimplex(best_h, best_n / 2.0)
        final_score = compute_c5(best_h, 2.0 / best_n)
    else:
        # fallback: zeros

            best_h = np.full(2, 0.5)
            best_n = 2
            final_score = compute_c5(best_h, 1.0)

    return best_h, float(final_score), int(best_n)

import numpy as np
import time

def _shift(arr: np.ndarray, s: int) -> np.ndarray:
    """Shift array left by s (positive) or right by -s with zero padding."""
    n = arr.shape[0]
    if s > 0:
        return np.concatenate([arr[s:], np.zeros(s, dtype=arr.dtype)])
    if s < 0:
        p = -s
        return np.concatenate([np.zeros(p, dtype=arr.dtype), arr[:n - p]])
    return arr

def _project_capsimplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project onto {0≤x≤1, sum(x)=target_sum} by bisection on a shift λ."""
    low, high = v.min() - 1.0, v.max()
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(v - lam, 0.0, 1.0)

def _c5(h: np.ndarray, dx: float) -> float:
    """Exact C5 for a discrete step function."""
    g = np.correlate(h, 1.0 - h, mode="full")
    return float(g.max() * dx)

def _grad_softmax(h: np.ndarray, beta: float) -> np.ndarray:
    """Gradient of the smooth max (softmax) of the overlap expressions."""
    n = h.size
    g = np.correlate(h, 1.0 - h, mode="full")
    b = beta * g
    b -= b.max()                     # stabilization
    p = np.exp(b)
    p /= p.sum()
    grad = np.zeros_like(h)
    for idx, w in enumerate(p):
        lag = idx - (n - 1)
        pos = _shift(h, lag)
        neg = _shift(h, -lag)
        grad += w * (1.0 - pos - neg)
    return grad

def _opt_continuous(h0: np.ndarray, n: int, dx: float, timelimit: float):
    """Projected gradient descent on the smooth‑max surrogate."""
    S = n / 2.0
    h = h0.copy()
    best_h, best_val = h.copy(), _c5(h, dx)
    start = time.time()
    beta = 0.5
    while time.time() - start < timelimit:
        g = _grad_softmax(h, beta)
        lr = 0.5 / (beta + 1.0)               # decreasing step as beta grows
        h = _project_capsimplex(h - lr * g, S)
        val = _c5(h, dx)
        if val < best_val:
            best_val, best_h = val, h.copy()
        beta = min(beta * 1.25, 30.0)          # raise smoothness
    return best_h, best_val

def _hill_binary(h_bin: np.ndarray, dx: float, timelimit: float):
    """Simple swap‑based hill‑climbing on binary vectors."""
    n = h_bin.size
    half = n // 2
    best = h_bin.copy()
    best_val = _c5(best, dx)
    start = time.time()
    while time.time() - start < timelimit:
        ones = np.where(best > 0.5)[0]
        zeros = np.where(best < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        best[i], best[j] = best[j], best[i]          # swap
        val = _c5(best, dx)
        if val < best_val:
            best_val = val
        else:
            best[i], best[j] = best[j], best[i]      # revert
    return best, best_val

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step function giving a (hopefully) improved upper bound for C5."""
    np.random.seed(seed)
    start_all = time.time()
    total_budget = float(budget_s)

    # size of the problem – try to reuse the provided construction if present
    if "initial_h_values" in globals():
        init = globals()["initial_h_values"].astype(np.float64)
        n = init.shape[0]
        h_initial = init
    else:
        n = 70
        h_initial = None

    dx = 2.0 / n
    target_sum = n / 2.0

    best_h, best_val, best_n = None, np.inf, n

    # number of independent restarts (adjustable)
    restarts = 5
    for r in range(restarts):
        # time check
        if time.time() - start_all > total_budget * 0.94:
            break
        # create a fresh start point
        if h_initial is not None:
            h0 = h_initial.copy()
            # add a modest random perturbation to escape possible plateaus
            h0 += np.random.uniform(-0.07, 0.07, size=n)
            np.clip(h0, 0.0, 1.0, out=h0)
            h0 = _project_capsimplex(h0, target_sum)
        else:
            perm = np.random.permutation(n)
            h0 = np.zeros(n, dtype=np.float64)
            h0[perm[:n // 2]] = 1.0

        # allocate a slice of the remaining time for this restart
        elapsed = time.time() - start_all
        remaining = total_budget - elapsed
        slice_time = remaining / (restarts - r) * 0.9   # 90% of fair share
        # ----- continuous optimisation -----
        cont_time = slice_time * 0.6
        h_cont, val_cont = _opt_continuous(h0, n, dx, cont_time)

        # ----- binary refinement -----
        # produce a feasible binary vector by taking the top half entries
        order = np.argsort(-h_cont)
        h_bin = np.zeros_like(h_cont)
        h_bin[order[:n // 2]] = 1.0
        bin_time = slice_time * 0.4
        h_bin_opt, val_bin = _hill_binary(h_bin, dx, bin_time)

        # keep the best among the candidates
        for cand_h, cand_val in ((h_cont, val_cont), (h_bin_opt, val_bin)):

    # final safety projection and evaluation
    if best_h is None:
        # fallback – a simple constant 0.5 function (guaranteed feasible)

        best_n = n
    else:
        best_h = _project_capsimplex(best_h, best_n / 2.0)
        best_val = _c5(best_h, 2.0 / best_n)

    return best_h, float(best_val), int(best_n)
