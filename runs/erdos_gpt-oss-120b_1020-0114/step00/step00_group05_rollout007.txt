import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Determine problem size and initial solution
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=np.float64)
        n = len(init)
        target_sum = n // 2                     # because dx = 2/n, integral =1
        # Ensure exact sum by a tiny correction on a random coordinate
        cur_h = init.copy()
        cur_sum = cur_h.sum()
        if abs(cur_sum - target_sum) > 1e-12:
            diff = target_sum - cur_sum
            # adjust the first element, staying inside [0,1]
            cur_h[0] = np.clip(cur_h[0] + diff, 0.0, 1.0)
    else:
        # No parent – work with a modest even size
        n = 140                      # 70 ones, 70 zeros → integral 1
        target_sum = n // 2
        cur_h = np.zeros(n, dtype=np.float64)
        cur_h[:target_sum] = 1.0
        rng.shuffle(cur_h)

    # ------------------------------------------------------------------
    # Helper to evaluate C5 via full correlation (dx = 2/n)
    # ------------------------------------------------------------------
    dx = 2.0 / n
    def c5_of(arr):
        conv = np.correlate(arr, 1.0 - arr, mode='full')
        return conv.max() * dx

    best_h = cur_h.copy()
    best_c5 = c5_of(best_h)

    # ------------------------------------------------------------------
    # Simulated‑annealing style local search
    # ------------------------------------------------------------------
    # temperature schedule
    T = 0.06
    decay = 0.9997
    # maximal amount we move in one step
    max_step = 0.2

    it = 0
    while True:
        # time guard
        if time.time() - start > budget_s * 0.95:
            break

        # donor – indices that can decrease a little; receiver – can increase
        donors = np.where(cur_h > 1e-12)[0]
        receivers = np.where(cur_h < 1.0 - 1e-12)[0]
        if donors.size == 0 or receivers.size == 0:
            # restart from a fresh random binary pattern
            cur_h[:] = 0.0
            cur_h[:target_sum] = 1.0
            rng.shuffle(cur_h)
            T = 0.06
            continue

        i = int(rng.choice(donors))
        j = int(rng.choice(receivers))
        # maximal transferable mass while preserving bounds
        delta_max = min(cur_h[i], 1.0 - cur_h[j])
        if delta_max <= 0:
            continue
        delta = rng.random() * min(delta_max, max_step)

        # propose move
        cur_h[i] -= delta
        cur_h[j] += delta
        new_c5 = c5_of(cur_h)

        # decide acceptance
        improve = new_c5 < best_c5
        accept = False
        if improve:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_c5 - best_c5) / T)
                if rng.random() < prob:
                    accept = True
        if accept:

                best_c5 = new_c5
                best_h = cur_h.copy()

            cur_h[i] += delta
            cur_h[j] -= delta

        it += 1

            rng.shuffle(cur_h)
