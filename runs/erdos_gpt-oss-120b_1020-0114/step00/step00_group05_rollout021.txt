import numpy as np
import random
import time
import math

def _c5_of_h(h, dx):
    """Maximum overlap integral for a discrete step function."""
    conv = np.convolve(h, (1.0 - h)[::-1], mode='full')
    return dx * float(conv.max())

def _random_binary(n, ones):
    """Generate a binary vector of length n with exactly `ones` ones."""
    h = np.zeros(n, dtype=float)
    idx = np.random.choice(n, ones, replace=False)
    h[idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Find a step function that yields a low C5 bound."""
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    best_h = None
    best_c5 = float('inf')
    best_n = None

            if abs(init_vals.sum() * dx0 - 1.0) < 1e-9:

    except NameError:
        init_vals = None
