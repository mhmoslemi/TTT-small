def _compute_c5(h):
    """Return C5 for a given step function h (array of samples)."""
    n = h.shape[0]
    # correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)


def _worst_shift(h):
    """Return the integer shift that attains the maximal overlap integral."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = corr.argmax()
    return idx - (n - 1)   # shift relative to zero


def _w_vector(h, shift):
    """For a given shift, compute w_i = h_{i+shift}+h_{i-shift} (zero outside)."""
    n = h.shape[0]
    w = np.zeros_like(h)
    if shift >= 0:
        if shift < n:
            w[: n - shift] += h[shift:]
            w[shift:] += h[: n - shift]
    else:
        s = -shift
        if s < n:
            w[s:] += h[: n - s]
            w[: n - s] += h[s:]
    return w


def _make_feasible(h):
    """Clip to [0,1] and adjust the total mass to exactly n/2."""
    n = h.shape[0]
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # Distribute the difference over interior points (not at a bound)
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if not mask.any():
            # fallback: distribute uniformly (may violate bounds slightly)
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
            diff = target - h.sum()
            continue
        adjust = diff / mask.sum()
        h[mask] += adjust
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    return h

                new_h[j] -= delta
                new_val = _compute_c5(new_h)

                if new_val <= cur_val + 1e-14:
                    cur_h, cur_val = new_h, new_val
                else:
                    temp = max_step * 0.5
                    if np.random.rand() < np.exp(-(new_val - cur_val) / (temp + 1e-14)):
                        cur_h, cur_val = new_h, new_val

        # --------------------------------------------------------------
        # 4.  Keep the best ever encountered.
        # --------------------------------------------------------------
        if cur_val < best_val - 1e-13:
            best_val = cur_val
            best_h = cur_h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # 5.  Periodic jitter if we have been stuck for a while.
        # --------------------------------------------------------------
        if no_improve > max_no_improve:
            # jitter around the current best state
            cur_h = best_h + np.random.normal(scale=0.015, size=n)
            cur_h = np.clip(cur_h, 0.0, 1.0)
            cur_h = _make_feasible(cur_h)
            cur_val = _compute_c5(cur_h)
            no_improve = 0

        # --------------------------------------------------------------
        # 6.  Early exit if we have already passed the target bound.
        # --------------------------------------------------------------
        if best_val <= 0.3808:
            break

    # --------------------------------------------------------------------
    # 7.  Final post‑processing: ensure feasibility and recompute exact C5.
    # --------------------------------------------------------------------
    best_h = _make_feasible(best_h)
    final_val = _compute_c5(best_h)

    return (best_h, float(final_val), n)
