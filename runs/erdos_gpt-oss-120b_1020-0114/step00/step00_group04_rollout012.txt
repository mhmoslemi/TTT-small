def compute_c5(h):
    n = len(h)
    nfft = 1 << (2*n - 1).bit_length()
    # using np.fft.rfftn? In 1D rfft.
    H = np.fft.rfft(h, n=nfft)
    G = np.fft.rfft(1 - h, n=nfft)
    corr = np.fft.irfft(H * np.conj(G), n=nfft)[:(2*n - 1)]
    return corr.max() * (2.0 / n)

def refine_h(h_init, best_val, time_limit):
    h = h_init.copy()
    cur_val = best_val
    best_h = h.copy()
    n = h.size
    start = time.time()
    steps = 0
    while time.time() - start < time_limit:
        steps += 1
        # temperature schedule
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = 0.01 * (1 - frac) + 1e-6
        i, j = np.random.randint(0, n, 2)
        hi, hj = h[i], h[j]
        # max delta positive to increase hi and decrease hj
        max_pos = min(1 - hi, hj)
        max_neg = min(hi, 1 - hj)
        # choose delta uniformly in [-max_neg, max_pos]
        delta = (np.random.rand() * (max_pos + max_neg) - max_neg)
        if abs(delta) < 1e-12:
            continue
        # apply move
        h[i] = hi + delta
        h[j] = hj - delta
        new_val = compute_c5(h)
        if new_val < cur_val or np.random.rand() < math.exp(-(new_val - cur_val) / T):
            cur_val = new_val
            if new_val < compute_c5(best_h):
                best_h = h.copy()
        else:
            # revert
            h[i] = hi
            h[j] = hj
    return best_h, compute_c5(best_h)

def shape_h_zero_edges(n, t):
    if t <= 0:
        return np.full(n, 0.5)
    dx = 2.0 / n
    b = 1.0 / (2.0 - t)  # plateau height
    xs = (np.arange(n) + 0.5) * dx
    h = np.empty(n, dtype=np.float64)
    # ramp up region: x < t
    ramp_up = xs < t
    h[ramp_up] = b * xs[ramp_up] / t
    # plateau region:
    plateau = (xs >= t) & (xs <= 2 - t)
    h[plateau] = b
    # ramp down region:
    ramp_down = xs > 2 - t
    h[ramp_down] = b * (2 - xs[ramp_down]) / t
    return h
```

def find_best_t(n, budget_s):
    best_val = float('inf')
    best_t = None
    # coarse grid
    for t in np.linspace(0.01,0.99,200):
        h = shape_h_zero_edges(n, t)
        h = h * ( (n/2) / h.sum() )
        val = compute_c5(h)

            best_val = val
            best_t = t
    # refine within +- step using more points
    # Use golden-section search
    lo = max(0.001, best_t - 0.05)
    hi = min(0.999, best_t + 0.05)
    for _ in range(20):
        t1 = lo + (hi - lo) / 3.0
        t2 = hi - (hi - lo) / 3.0
        h1 = shape_h_zero_edges(n, t1); h1 *= (n/2) / h1.sum()
        c1 = compute_c5(h1)
        h2 = shape_h_zero_edges(n, t2); h2 *= (n/2) / h2.sum()
        c2 = compute_c5(h2)
        if c1 < c2:

            if c1 < best_val:

            lo = t1
            if c2 < best_val:

import numpy as np, time, math, random

def compute_c5(h: np.ndarray) -> float:
    """max_{k} ∫ h(x)*(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    # length needed for full correlation
    nfft = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=nfft)
    G = np.fft.rfft(1.0 - h, n=nfft)
    corr = np.fft.irfft(H * np.conj(G), n=nfft)[: 2 * n - 1]
    return float(corr.max() * (2.0 / n))


def shape_h_zero_edges(n: int, t: float) -> np.ndarray:
    """
    Symmetric piecewise‑linear shape:
    - ramps of width t from 0 to height b,
    - flat plateau of height b on the centre,
    - edges are exactly 0.
    Parameter t∈(0,1), b = 1/(2‑t) guarantees ∫h=1.
    """
    if t <= 0.0:
        return np.full(n, 0.5)
    dx = 2.0 / n
    b = 1.0 / (2.0 - t)
    xs = (np.arange(n) + 0.5) * dx          # mid‑point sampling
    h = np.empty(n, dtype=np.float64)

    ramp_up = xs < t
    plateau = (xs >= t) & (xs <= 2.0 - t)
    ramp_down = xs > 2.0 - t

    h[ramp_up] = b * xs[ramp_up] / t
    h[plateau] = b
    h[ramp_down] = b * (2.0 - xs[ramp_down]) / t
    # renormalise to exact integral (discretisation error)
    total = h.sum()
    if total != 0.0:
        h *= (n / 2.0) / total
    return np.clip(h, 0.0, 1.0)


def grid_search_best(n: int, time_limit: float):
    """Coarse + fine grid over t to obtain a good starting shape."""
    start = time.time()
    best_val = 10.0
    best_h = None
    best_t = None

    # coarse scan
    for t in np.linspace(0.01, 0.99, 300):
        h = shape_h_zero_edges(n, t)
        val = compute_c5(h)
        if val < best_val:
            best_val, best_h, best_t = val, h, t
    # fine refinement around best_t
    lo = max(0.001, best_t - 0.05)
    hi = min(0.999, best_t + 0.05)
    for _ in range(40):
        t1 = lo + (hi - lo) / 3.0
        t2 = hi - (hi - lo) / 3.0
        h1 = shape_h_zero_edges(n, t1)
        v1 = compute_c5(h1)
        h2 = shape_h_zero_edges(n, t2)
        v2 = compute_c5(h2)
        if v1 < v2:
            hi, val, h = t2, v2, h2
            if v1 < best_val:
                best_val, best_h, best_t = v1, h1, t1
        else:
            lo, val, h = t1, v1, h1
            if v2 < best_val:
                best_val, best_h, best_t = v2, h2, t2
        if time.time() - start > time_limit:
            break
    return best_h, best_val


def simulated_annealing(start_h: np.ndarray, start_val: float, time_limit: float):
    """Simple SA that preserves the sum by swapping mass between two sites."""
    n = start_h.size
    h = start_h.copy()
    cur_val = start_val
    best_h = h.copy()
    best_val = cur_val
    start = time.time()
    it = 0
    while True:
        # timing check every 1000 iterations
        if it % 1000 == 0:
            if time.time() - start > time_limit:
                break
        i, j = np.random.randint(0, n, 2)
        hi, hj = h[i], h[j]
        # maximum admissible delta keeping both variables in [0,1]
        max_pos = min(1.0 - hi, hj)          # increase i, decrease j
        max_neg = min(hi, 1.0 - hj)          # decrease i, increase j
        if max_pos + max_neg == 0.0:
            it += 1
            continue
        delta = (np.random.rand() * (max_pos + max_neg) - max_neg)
        if abs(delta) < 1e-12:
            it += 1
            continue
        h[i] = hi + delta
        h[j] = hj - delta
        new_val = compute_c5(h)
        # temperature schedule (exponential cooling)
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = 0.02 * (0.001 ** frac)   # starts ~0.02, ends ~2e-5
        accept = new_val < cur_val or (np.random.rand() < math.exp(-(new_val - cur_val) / T))
        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val, best_h = new_val, h.copy()
        else:
            # revert move
            h[i] = hi
            h[j] = hj
        it += 1
    return best_h, best_val
