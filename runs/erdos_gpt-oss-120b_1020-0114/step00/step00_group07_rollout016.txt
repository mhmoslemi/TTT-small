import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return C5 for a given step function h (array of values in [0,1])."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def generate_alpha_pattern(n: int, alpha: float, offset: float) -> np.ndarray:
    """Binary pattern of length n with exact weight n//2 using a Beatty‑type ordering."""
    i = np.arange(n, dtype=float)
    v = (i * alpha + offset) % 1.0
    order = np.argsort(v, kind="mergesort")
    h = np.zeros(n, dtype=float)
    h[order[: n // 2]] = 1.0
    return h


def binarize_by_weight(x: np.ndarray) -> np.ndarray:
    """Convert any array to a binary one with exactly half of the entries equal to 1."""
    n = x.shape[0]
    k = n // 2
    order = np.argsort(x, kind="mergesort")
    h = np.zeros(n, dtype=float)
    h[order[:k]] = 1.0
    return h


def simulated_annealing(
    h0: np.ndarray,
    rng: np.random.Generator,
    time_budget: float,
    init_val: float | None = None,
) -> tuple[np.ndarray, float]:
    """Simple SA that swaps a 1 and a 0 while respecting the weight constraint."""
    n = h0.shape[0]
    dx = 2.0 / n
    cur = h0.copy()
    cur_val = compute_c5(cur) if init_val is None else init_val

    best = cur.copy()
    best_val = cur_val

    ones_idx = np.flatnonzero(cur > 0.5).astype(np.int64)
    zeros_idx = np.flatnonzero(cur <= 0.5).astype(np.int64)
    n_ones = ones_idx.size
    n_zeros = zeros_idx.size

    start = time.time()
    T0 = 0.02
    T_end = 1e-5

    # iterate until the allocated time expires
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        # linearised progress for temperature schedule
        prog = elapsed / time_budget
        T = T0 * (T_end / T0) ** prog

        # propose a swap
        i_pos = rng.integers(n_ones)
        j_pos = rng.integers(n_zeros)
        i = ones_idx[i_pos]
        j = zeros_idx[j_pos]

        # perform swap
        cur[i] = 0.0
        cur[j] = 1.0

        # evaluate
        new_val = compute_c5(cur)
        delta = new_val - cur_val

        # Metropolis acceptance
        accept = delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12))
        if accept:
            cur_val = new_val
            # keep the indices in sync with the new bit values
            ones_idx[i_pos] = j
            zeros_idx[j_pos] = i
            if new_val < best_val:
                best_val = new_val
                best = cur.copy()
        else:
            # revert swap
            cur[i] = 1.0
            cur[j] = 0.0

    return best, best_val

    if remaining > 0.0 and best_h is not None:
        # Ensure binary representation (some initial vectors may be fractional)
        if not np.all(np.logical_or(best_h <= 0.0, best_h >= 1.0)):
            best_h = binarize_by_weight(best_h)
        # Run SA on the current best pattern
        best_h, best_val = simulated_annealing(best_h, rng, remaining, best_val)

    # ------------------------------------------------------------------
    # 5. Final sanity fix (clip + exact weight)
    # ------------------------------------------------------------------
    n = best_h.shape[0]
    best_h = np.clip(best_h, 0.0, 1.0)

    cur_sum = best_h.sum()
    target = n / 2
    if abs(cur_sum - target) > 1e-9:
        diff = int(round(target - cur_sum))
        if diff > 0:
            # need more 1s
            zero_pos = np.where(best_h < 0.5)[0]
            if zero_pos.size:
                sel = rng.choice(zero_pos, size=min(diff, zero_pos.size), replace=False)
                best_h[sel] = 1.0
        elif diff < 0:
            # need fewer 1s
            one_pos = np.where(best_h > 0.5)[0]
            if one_pos.size:
                sel = rng.choice(one_pos, size=min(-diff, one_pos.size), replace=False)
                best_h[sel] = 0.0

    # Re‑evaluate final bound
    best_val = compute_c5(best_h)

    # ------------------------------------------------------------------
    # 6. Return result
    # ------------------------------------------------------------------
    return best_h, float(best_val), int(n)
