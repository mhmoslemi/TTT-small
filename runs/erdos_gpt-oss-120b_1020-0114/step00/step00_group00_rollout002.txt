def simulated_annealing(h_initial, max_iters, rng, max_time=None):
    h = h_initial.copy()
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val
    n = len(h)
    start = time.time()
    T0 = 0.01  # starting temperature
    T_end = 1e-6
    # schedule: exponential decay over iterations
    for it in range(1, max_iters+1):
        if max_time is not None and time.time() - start > max_time:
            break
        # temperature
        T = T0 * ((T_end / T0) ** (it / max_iters))
        # propose move
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        # compute max epsilon possible
        # random direction
        if rng.random() < 0.5:
            max_eps = min(1 - h[i], h[j])
            if max_eps <= 0:
                continue
            eps = max_eps * rng.random() * 0.5
            new_h = h.copy()
            new_h[i] += eps
            new_h[j] -= eps
        else:
            max_eps = min(1 - h[j], h[i])
            if max_eps <= 0:
                continue
            eps = max_eps * rng.random() * 0.5
            new_h = h.copy()
            new_h[i] -= eps
            new_h[j] += eps
        new_val = compute_c5(new_h)
        delta = new_val - cur_val
        if delta < -1e-12 or rng.random() < math.exp(-delta / T):
            h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
    return best_h, best_val

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level, no closures)
# ----------------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Return max overlap C5 for a step function h on [0,2] with step size dx."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv.max() * dx


def project_box_sum(x: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector x onto the intersection of the box [0,1]^n and the affine
    hyperplane sum(y)=target. Uses a simple bisection on the shift lambda.
    """
    low, high = -2.0, 2.0
    for _ in range(60):
        mid = (low + high) * 0.5
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s < target:
            low = mid
        else:
            high = mid
    y = np.clip(x + (low + high) * 0.5, 0.0, 1.0)
    # Tiny numerical fix – distribute any leftover uniformly among free components
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        free = np.where((y > 1e-12) & (y < 1.0 - 1e-12))[0]
        if free.size > 0:
            y[free] += diff / free.size
            y = np.clip(y, 0.0, 1.0)
    return y

    T0 = 0.015      # initial temperature
    T_end = 1e-6    # final temperature

    for it in range(1, max_iters + 1):
        if max_time is not None and (time.time() - start) > max_time:
            break

        # exponential temperature schedule
        T = T0 * ((T_end / T0) ** (it / max_iters))

        # pick two distinct indices
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # decide direction of transfer
        if rng.random() < 0.5:
            max_eps = min(1.0 - h[i], h[j])
            if max_eps <= 0.0:
                continue
            eps = max_eps * rng.random() * 0.5
            new_h = h.copy()
            new_h[i] += eps
            new_h[j] -= eps
        else:
            max_eps = min(1.0 - h[j], h[i])
            if max_eps <= 0.0:
                continue
            eps = max_eps * rng.random() * 0.5
            new_h = h.copy()
            new_h[i] -= eps
            new_h[j] += eps

        new_val = compute_c5(new_h)
        delta = new_val - cur_val
        if delta < -1e-12 or rng.random() < math.exp(-delta / T):
            h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
    return best_h, best_val


def upsample_h(h_src: np.ndarray, n_target: int) -> np.ndarray:
    """Linear interpolation of h_src (defined on [0,2]) to n_target points."""
    n_src = h_src.shape[0]
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_tgt = np.linspace(0.0, 2.0, n_target, endpoint=False)
    return np.interp(x_tgt, x_src, h_src)


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] to obtain a good upper bound for C5.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.96  # small safety margin

    # ------------------------------------------------------------------
    # Obtain an initial construction if the harness supplied one.
    # ------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)  # provided by harness
    except NameError:
        # Fallback: simple half‑ones / half‑zeros pattern
        n_fallback = 45
        init_h = np.concatenate([np.ones(n_fallback // 2), np.zeros(n_fallback // 2)])
        if n_fallback % 2:
            init_h = np.append(init_h, 0.5)
        rng.shuffle(init_h)

    n0 = init_h.shape[0]

    # ------------------------------------------------------------------
    # Candidate resolutions (the problem statement prefers <1000 points)
    # ------------------------------------------------------------------
    candidates = [45, 90, 180]  # try coarse to fine
    best_h = None
    best_val = np.inf
    best_n = None

    for n in candidates:
        # check time
        if time.time() > deadline:
            break

        # initialise h for this resolution
        if n == n0:
            h0 = init_h.copy()
        else:
            h0 = upsample_h(init_h, n)

        # enforce box and sum constraints
        h0 = np.clip(h0, 0.0, 1.0)
        target_sum = n / 2.0
        h0 = project_box_sum(h0, target_sum)

        # baseline value
        baseline = compute_c5(h0)
        if baseline < best_val:
            best_val, best_h, best_n = baseline, h0.copy(), n

        # Allocate iteration budget (roughly scaled with n)
        # Each iteration is cheap; we use a generous upper bound.
        max_iters = 150_000 // max(1, n // 45)

        # remaining time for this stage
        rem_time = deadline - time.time()
        if rem_time <= 0:
            break

        # Run simulated annealing with an internal time cap
        h_opt, val_opt = simulated_annealing(
            h0, max_iters, rng, max_time=rem_time * 0.9
        )
        if val_opt < best_val:
            best_val, best_h, best_n = val_opt, h_opt.copy(), n

        # early exit if we already beat the target record
        if best_val <= 0.3808:
            break

    # Final sanity check – ensure constraints are satisfied
    if best_h is not None:
        best_h = np.clip(best_h, 0.0, 1.0)
        # tiny renormalisation if necessary
        diff = best_n / 2.0 - best_h.sum()
        if abs(diff) > 1e-9:
            best_h = project_box_sum(best_h, best_n / 2.0)

    return best_h, float(best_val), int(best_n)
