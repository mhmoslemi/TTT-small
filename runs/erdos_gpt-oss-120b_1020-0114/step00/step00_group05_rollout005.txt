import time, math, random
import numpy as np

def _c5_value(arr: np.ndarray) -> float:
    """Compute the C5 value for a sampled step function."""
    # correlation of h with (1-h), full mode (aperiodic)
    corr = np.correlate(arr, 1.0 - arr, mode="full")
    return np.max(corr) * (2.0 / arr.size)


def _random_h(n: int, steps: int = 300) -> np.ndarray:
    """Generate a random feasible h (sum = n/2, values in [0,1])."""
    h = np.full(n, 0.5, dtype=float)
    for _ in range(steps):
        i, j = np.random.randint(n), np.random.randint(n)
        if i == j:
            continue
        # move mass from j to i (or opposite) while respecting bounds
        max_inc = min(1.0 - h[i], h[j])
        if max_inc <= 0:
            continue
        delta = max_inc * np.random.rand()
        h[i] += delta
        h[j] -= delta
    return h


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step function h, its C5 bound, and number of samples."""
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # 1. Load initial construction (if supplied by the harness)
    # ------------------------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=float)  # type: ignore[name-defined]
    except Exception:

        n = 70
        h0 = np.full(n, 0.5, dtype=float)

    n = h0.size
