    max_corr = conv.max()

import numpy as np, time

def compute_max_corr(h):
    conv = np.correlate(h, 1-h, mode='full')
    max_corr = conv.max()
    return max_corr, conv

def subgradient_max_corr(h):
    N = len(h)
    conv = np.correlate(h, 1-h, mode='full')
    max_corr = conv.max()
    # tolerance for active shifts due to floating error
    tol = 1e-12
    idxs = np.where(conv >= max_corr - tol)[0]
    shifts = idxs - (N - 1)
    grad = np.zeros(N)
    for k in shifts:
        if k >= 0:
            A = h[k:]
            B = h[:N - k]
            grad[k:] += 1 - B
            grad[:N - k] += -A
        else:
            kk = -k
            A = h[:N - kk]
            B = h[kk:]
            grad[:N - kk] += 1 - B
            grad[kk:] += -A
    grad /= len(shifts)
    return grad, max_corr

def project_onto_box_sum(v, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    proj = np.clip(v + (lo+hi)/2.0, 0.0, 1.0)
    return proj

import time
import numpy as np
import random

# ---------- helper functions ----------

def project_onto_box_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto the set {0<=x<=1, sum(x)=target_sum}."""
    lo, hi = -2.0, 2.0               # bounds guarantee sum after clipping = 0/len(v)
    for _ in range(60):             # binary search on the Lagrange multiplier
        mid = (lo + hi) * 0.5
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def subgradient_max_corr(h: np.ndarray):
    """Return a sub‑gradient of max_k ∫h(1‑h(·+k)) and the maximal correlation value."""
    N = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode='full')
    max_val = conv.max()
    # indices of shifts attaining the maximum (allow tiny numerical tolerance)
    tol = 1e-12
    idx = np.where(conv >= max_val - tol)[0]
    shifts = idx - (N - 1)

    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            # forward shift: contributions from pairs (i, i‑k)
            a = h[k:]            # h[i] where i>=k
            b = h[:N - k]        # h[i‑k]
            grad[k:] += 1.0 - b
            grad[:N - k] += -a
        else:
            kk = -k
            a = h[:N - kk]       # h[i]
            b = h[kk:]           # h[i+kk]
            grad[:N - kk] += 1.0 - b
            grad[kk:] += -a
    grad /= len(shifts)            # any convex combination works; average is convenient
    return grad, max_val


def optimize_h(N: int, h0: np.ndarray, max_iters: int, time_limit: float):
    """Projected sub‑gradient descent for a given length N."""
    target = N / 2.0
    h = project_onto_box_sum(h0, target)

    # initial evaluation
    _, cur_max = subgradient_max_corr(h)
    best_h, best_val = h.copy(), cur_max

    step0 = 0.8
    for it in range(max_iters):
        if time.time() > time_limit:
            break

        grad, cur_max = subgradient_max_corr(h)
        step = step0 / np.sqrt(it + 1.0)

        # back‑tracking line search (monotone improvement)
        improved = False
        for _ in range(8):
            h_new = project_onto_box_sum(h - step * grad, target)
            _, new_max = subgradient_max_corr(h_new)
            if new_max <= cur_max + 1e-12:
                h, cur_max = h_new, new_max
                improved = True
                break
            step *= 0.5

        if not improved:
            # small random perturbation to escape possible plateaus
            jitter = np.random.randn(N) * 0.01
            h = project_onto_box_sum(h + jitter, target)
            _, cur_max = subgradient_max_corr(h)

        if cur_max < best_val - 1e-12:
            best_val, best_h = cur_max, h.copy()

        # occasional jitter for diversity
        if it % 100 == 0:
            jitter = np.random.randn(N) * 0.02
            h = project_onto_box_sum(h + jitter, target)

    c5 = best_val * (2.0 / N)
    return best_h, c5


# ---------- main entry point ----------

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()

    best_h = None
    best_c5 = np.inf
    best_n = None

    # candidate lengths (odd/even both okay); order shuffled for variety
    n_candidates = [45, 60, 80, 100, 120, 150, 180, 200]
    random.shuffle(n_candidates)

    # try each candidate while time permits
    for N in n_candidates:
        if time.time() - start > budget_s * 0.95:
            break

        # initial vector
        if N == 45 and 'initial_h_values' in globals():
            h0 = np.array(initial_h_values, dtype=float)
        else:

        # ensure feasibility before optimisation
        h0 = project_onto_box_sum(h0, N / 2.0)

        # allocate a modest iteration budget; adapt if time runs low
        remaining = budget_s - (time.time() - start)
        iters = 2500
        if remaining < 30:          # very little time left
            iters = max(200, iters // 5)

        time_limit = start + budget_s * 0.95
        h_opt, c5_opt = optimize_h(N, h0, iters, time_limit)

        if c5_opt < best_c5:
            best_c5, best_h, best_n = c5_opt, h_opt, N

        if best_c5 <= 0.3808:       # target achieved – stop early
            break

    # final refinement on the current best solution (if any time left)
    if best_h is not None and time.time() - start < budget_s * 0.90:
        time_limit = start + budget_s * 0.95
        h_ref, c5_ref = optimize_h(best_n, best_h, 5000, time_limit)
        if c5_ref < best_c5:
            best_c5, best_h = c5_ref, h_ref

    # safety fallback (should never trigger)
    if best_h is None:
        best_n = 100
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_c5 = float(np.correlate(best_h, 1 - best_h, mode='full').max() * (2.0 / best_n))

    # final projection to guarantee feasibility
    best_h = project_onto_box_sum(best_h, best_n / 2.0)
    final_max = np.correlate(best_h, 1 - best_h, mode='full').max()
    best_c5 = float(final_max * (2.0 / best_n))
