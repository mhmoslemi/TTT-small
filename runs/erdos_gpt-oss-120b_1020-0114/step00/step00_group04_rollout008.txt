import numpy as np, time, math, random

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def random_binary_pattern(n):
    h = np.zeros(n)
    half = n // 2
    idx = np.random.choice(n, half, replace=False)
    h[idx] = 1.0
    return h

def hill_climb(n, time_limit):
    start = time.time()
    best_h = random_binary_pattern(n)
    best_score = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_score = best_score
    # Temperature schedule:
    T = 0.02
    cooling = 0.9999
    # Keep indexes of ones and zeros for quick selection
    # We'll maintain arrays.
    while time.time() - start < time_limit:
        # Choose a 1 index and 0 index:
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap:
        new_h = cur_h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        new_score = compute_c5(new_h)
        delta = new_score - cur_score
        if delta < 0 or np.random.rand() < math.exp(-delta / T):
            cur_h = new_h
            cur_score = new_score
            if new_score < best_score:
                best_h = new_h
                best_score = new_score
        T *= cooling
        # occasional restart to avoid stagnation:
        if np.random.rand() < 0.001:
            cur_h = random_binary_pattern(n)
            cur_score = compute_c5(cur_h)
    return best_h, best_score

import time, math, random
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return the Erdős C5 value for a discretised step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation h with (1-h); np.correlate gives the full linear correlation
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * float(np.max(corr))

def _binary_start(n: int) -> np.ndarray:
    """Generate a random binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    ones = np.random.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a step function giving a low C5 bound.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values – list of floats in [0,1] of length n_points,
        c5_bound – the achieved C5 value (float),
        n_points – number of discretisation points (int).
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # Determine problem size and an initial feasible vector
    # ------------------------------------------------------------------
    try:
        # initial_h_values may be provided by the harness
        init = np.array(initial_h_values, dtype=float)
        n = init.shape[0]
        # Produce a binary version that satisfies the integral exactly
        half = n // 2
        idx = np.argsort(-init)[:half]
        h_cur = np.zeros(n, dtype=float)
        h_cur[idx] = 1.0
    except Exception:
        # No pre‑existing vector – use a modest size (even) for speed
        n = 84                # 84 points → dx = 2/84 ≈ 0.0238
        h_cur = _binary_start(n)

    # ------------------------------------------------------------------
    # Initial evaluation
    # ------------------------------------------------------------------
    cur_score = _compute_c5(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score

    start_time = time.time()
    time_limit = budget_s * 0.96          # keep a small safety margin

    # Simulated‑annealing parameters
    T = 0.05                               # start temperature
    cooling = 0.99995
    T_min = 1e-6
    no_improve = 0
    max_no_improve = 50000                 # restart after many stagnated steps

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() - start_time < time_limit:
        # ---- propose a feasible transfer between two coordinates ----
        i, j = np.random.choice(n, 2, replace=False)
        hi, hj = h_cur[i], h_cur[j]

        # feasible delta such that 0≤hi−δ≤1 and 0≤hj+δ≤1
        delta_min = max(-hj, -(1.0 - hi))
        delta_max = min(hi, 1.0 - hj)
        if delta_max <= delta_min:
            continue
        # draw a delta; uniform works well for this problem
        delta = np.random.uniform(delta_min, delta_max)
        if abs(delta) < 1e-12:
            continue

        h_new = h_cur.copy()
        h_new[i] = hi - delta
        h_new[j] = hj + delta

        new_score = _compute_c5(h_new)

        # ---- Metropolis acceptance ----
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(T, T_min))
            accept = random.random() < prob

        if accept:
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_h = h_new.copy()
                best_score = new_score
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # temperature decay
        T = max(T * cooling, T_min)

        # occasional random restart to avoid long stagnation
        if no_improve > max_no_improve:
            h_cur = _binary_start(n)
            cur_score = _compute_c5(h_cur)
            no_improve = 0

    # ------------------------------------------------------------------
    # Final evaluation (using the harness' evaluator if it exists)
    # ------------------------------------------------------------------
    try:
        final_score = float(evaluate_erdos_solution(best_h))
    except Exception:
        final_score = _compute_c5(best_h)

    # Ensure the returned list obeys the constraints (tiny rounding noise may occur)
    best_h = np.clip(best_h, 0.0, 1.0)
    # Force exact integral by correcting the smallest numerical drift
    total = best_h.sum()
    target = n / 2
    if abs(total - target) > 1e-8:
        # Adjust the first element to compensate (will stay in [0,1] because drift is tiny)
        best_h[0] += target - total

    return best_h.tolist(), float(final_score), n
