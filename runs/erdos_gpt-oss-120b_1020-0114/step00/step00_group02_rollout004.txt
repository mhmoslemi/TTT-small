import time
import math
import random
import numpy as np

def _next_pow_two(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _project_sum(vec: np.ndarray, target: float) -> np.ndarray:
    """
    Adjust ``vec`` so that its sum equals ``target``,
    keeping all entries inside [0, 1] by moving mass between entries.
    """
    diff = target - vec.sum()
    # small tolerance – return early
    if abs(diff) < 1e-12:
        return vec
    # simple water‑filling loop (fast enough for our sizes)
    while abs(diff) > 1e-12:
        if diff > 0:
            cand = np.where(vec < 1.0)[0]
            if cand.size == 0:
                break
            i = np.random.choice(cand)
            add = min(diff, 1.0 - vec[i])
            vec[i] += add
            diff -= add
        else:
            cand = np.where(vec > 0.0)[0]
            if cand.size == 0:
                break
            i = np.random.choice(cand)
            sub = min(-diff, vec[i])
            vec[i] -= sub
            diff += sub
    return vec


def _c5_fast(h: np.ndarray, dx: float, L: int) -> float:
    """
    Compute C5 = max_k ∫ h(x)(1‑h(x+k))dx using FFT.
    ``h`` is a 1‑D array of length N, ``dx`` = 2/N,
    ``L`` is a power‑of‑two ≥ 2*N‑1 (pre‑computed).
    """
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    conv = np.fft.irfft(H * np.conj(G), n=L)
    # linear (non‑circular) correlation lives in the first 2*N‑1 entries
    max_val = conv[: 2 * h.size - 1].max()
    return max_val * dx

    Search for a step function h on [0,2] with values in [0,1] and
    integral 1 that minimises C5.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------
    # configuration
    # ------------------------------------------------------------
    n_points = 200                     # moderate size, < 1k as requested
    dx = 2.0 / n_points
    target_sum = n_points / 2.0        # because ∫h = 1  =>  sum(h)*dx = 1
    L_fft = _next_pow_two(2 * n_points - 1)

    # ------------------------------------------------------------
    # initial candidate
    # ------------------------------------------------------------
    if 'initial_h_values' in globals():
        # interpolate the provided coarse construction to our finer grid
        init = np.asarray(initial_h_values, dtype=np.float64)
        x_init = np.linspace(0.0, 2.0, init.size, endpoint=False) + 0.5 * (2.0 / init.size)
        x_target = np.linspace(0.0, 2.0, n_points, endpoint=False) + 0.5 * dx
        h = np.interp(x_target, x_init, init)
        h = np.clip(h, 0.0, 1.0)
    else:
        # simple start: constant 0.5 with a tiny random perturbation
        h = np.full(n_points, 0.5, dtype=np.float64)
        h += 0.1 * (np.random.rand(n_points) - 0.5)
        h = np.clip(h, 0.0, 1.0)

    # enforce the exact mass constraint
    h = _project_sum(h, target_sum)

    # ------------------------------------------------------------
    # helper closures (capturing the constants)
    # ------------------------------------------------------------
    def c5_of(vec):
        return _c5_fast(vec, dx, L_fft)

    # ------------------------------------------------------------
    # simulated‑annealing loop
    # ------------------------------------------------------------
    current_h = h.copy()
    current_c5 = c5_of(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
