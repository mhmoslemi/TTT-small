import numpy as np
import time
import math
import random

def _c5_value(h):
    """Return the C5 overlap value for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # full correlation gives the needed overlaps for all integer shifts
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _project_to_sum_box(v, target_sum, eps=1e-12):
    """
    Project vector v onto the set {x | 0 <= x_i <= 1, sum x_i = target_sum}
    using a simple bisection on the Lagrange multiplier.
    """
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (sampled uniformly) that gives a
    small value of C5 = max_k ∫ h(x)(1-h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()

    # --------------------------------------------------------------
    # 1. Choose a grid size and prepare the coordinate array
    # --------------------------------------------------------------
    n = 200                     # number of samples (adjustable, still fast)
    dx = 2.0 / n
    target_sum = n / 2.0        # because ∫h = 1  ->  sum(h)*dx = 1
    x = (np.arange(n) + 0.5) * dx   # mid‑point positions in [0,2]

    best_h = None
    best_c5 = np.inf
    best_params = None

    # --------------------------------------------------------------
    # 2. Coarse parametric search:
    #    shape = {edge block height = h_edge, middle height = p}
    #    edge blocks occupy [0,L) and (2-L,2]
    # --------------------------------------------------------------
    L_vals = np.linspace(1e-6, 0.5 - 1e-6, 300)      # edge half‑width
    h_edge_vals = np.linspace(0.5, 1.0, 250)        # edge height (≥ middle)

    for L in L_vals:
        # indices that belong to the left/right edge blocks
        edge_mask = (x < L) | (x > (2.0 - L))
        n_edge = int(edge_mask.sum())
        n_mid = n - n_edge
        if n_mid == 0:                     # would be a constant function
            continue

        for h_edge in h_edge_vals:
            # Determine middle height p to satisfy the integral exactly
            p = (target_sum - h_edge * n_edge) / n_mid
            if p < 0.0 or p > 1.0:
                continue

            h = np.full(n, p, dtype=np.float64)
            h[edge_mask] = h_edge

            # Quick evaluation
            c5 = np.max(np.correlate(h, 1.0 - h, mode="full")) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_params = (L, h_edge, p)

        # keep some budget for later refinements
        if time.time() - start_time > budget_s * 0.8:
            break

    # --------------------------------------------------------------
    # 3. If an initial construction is supplied, compare it
    # --------------------------------------------------------------
    if 'initial_h_values' in globals():
        try:
            init_h = np.asarray(initial_h_values, dtype=np.float64).copy()
            # Project to the feasible set (in case of rounding)
            init_h = _project_to_sum_box(init_h, target_sum)
            init_c5 = _c5_value(init_h)
            if init_c5 < best_c5:
                best_c5 = init_c5
                best_h = init_h.copy()
                best_params = ("initial",)
        except Exception:
            pass

    # --------------------------------------------------------------
    # 4. Local stochastic optimisation (mass‑transfer moves)
    # --------------------------------------------------------------
    if best_h is None:
        # fallback: uniform function (always feasible)
        best_h = np.full(n, 0.5, dtype=np.float64)
        best_c5 = _c5_value(best_h)

    cur_h = best_h.copy()
    cur_c5 = best_c5
    max_iter = 10_000_000  # large upper bound; we stop by time
    it = 0
    while time.time() - start_time < budget_s - 0.2 and it < max_iter:
        it += 1
        # pick two distinct positions
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # delta is amount moved from i to j (i loses, j gains)
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        min_delta = -min(cur_h[j], 1.0 - cur_h[i])
        if max_delta <= 0 and min_delta >= 0:
            continue
        # random step size, limited to keep moves local
        delta = min_delta + (max_delta - min_delta) * np.random.rand()
        delta = np.clip(delta, -0.05, 0.05)

        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta

        # evaluate
        new_c5 = np.max(np.correlate(new_h, 1.0 - new_h, mode="full")) * dx

        if new_c5 < cur_c5:
            cur_h, cur_c5 = new_h, new_c5
            if new_c5 < best_c5:
                best_h, best_c5 = new_h.copy(), new_c5
        else:
            # occasional uphill move (simulated‑annealing style)
            # temperature decays linearly to 0
            t = (time.time() - start_time) / budget_s
            T = 0.01 * (1.0 - t) + 1e-4
            if math.exp(-(new_c5 - cur_c5) / T) > random.random():
                cur_h, cur_c5 = new_h, new_c5

    # --------------------------------------------------------------
    # 5. Final projection onto the feasible set (tiny numeric drift)
    # --------------------------------------------------------------
    best_h = _project_to_sum_box(best_h, target_sum)

    final_c5 = _c5_value(best_h)

    # sanity check (should never fail, but keep it safe)
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12)
    assert abs(best_h.sum() - target_sum) < 1e-9

    return best_h, float(final_c5), n
