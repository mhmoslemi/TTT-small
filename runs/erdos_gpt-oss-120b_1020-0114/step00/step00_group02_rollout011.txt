    target = n/2
    import cvxpy as cp
    x = cp.Variable(n)
    constraints = [x >= 0, x <= 1, cp.sum(x) == target]

    prob.solve(solver=cp.ECOS, warm_start=True)
    return x.value

def project(v, target):
    lo = -np.max(v)
    hi = 1 - np.min(v)
    for _ in range(50):
        mid = (lo+hi)/2
        w = np.clip(v + mid, 0, 1)
        s = np.sum(w)
        if s < target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v + (lo+hi)/2, 0, 1)
    # due to numeric, scaling maybe small error; we can adjust diff by adding diff to some entries not at bound.
    diff = target - np.sum(w)
    if abs(diff) > 1e-10:
        # distribute diff equally among non-bound entries
        free = (w > 0) & (w < 1)
        cnt = np.sum(free)
        if cnt >0:
            w[free] += diff / cnt
            w = np.clip(w,0,1)
    return w

def _project_to_constraints(v, target):
    """Euclidean projection of v onto [0,1]^n with sum(v)=target."""
    lo = -np.max(v)
    hi = 1.0 - np.min(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = (w > 0.0) & (w < 1.0)
        cnt = free.sum()
        if cnt:
            w[free] += diff / cnt
            w = np.clip(w, 0.0, 1.0)
    return w


def _c5_of(h, dx):
    """Maximum overlap integral for a step vector h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _derivative(h, shift):
    """Sub‑gradient of the worst‑shift overlap w.r.t. h."""
    n = h.size
    if shift >= 0:

        h_minus = np.concatenate((np.zeros(shift, dtype=h.dtype), h[: n - shift]))

        h_minus = np.concatenate((h[s:], np.zeros(s, dtype=h.dtype)))
    return 1.0 - (h_plus + h_minus)

        # propose a deterministic improvement move
        made_move = False
        if pos_idx.size and neg_idx.size:
            i = pos_idx[int(np.argmax(d[pos_idx]))]          # most positive
            j = neg_idx[int(np.argmin(d[neg_idx]))]          # most negative
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta > 1e-12:
                # step size shrinks with time
                frac = (now - start) / budget_s
                step_frac = 0.5 * (1.0 - frac) + 0.01
                delta = max_delta * step_frac * rng.random()
                h_new = h.copy()
                h_new[i] -= delta
                h_new[j] += delta
                new_c5 = _c5_of(h_new, dx)
                if new_c5 <= cur_c5 or rng.random() < math.exp(-(new_c5 - cur_c5) / (0.02 + 0.1 * frac)):
                    h = h_new
                    cur_c5 = new_c5
                    made_move = True
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()

        # fallback: random two‑index transfer
        if not made_move:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                if rng.random() < 0.5:
                    max_delta = min(1.0 - h[i], h[j])
                    if max_delta > 1e-12:
                        delta = max_delta * 0.5 * rng.random()
                        h_new = h.copy()
                        h_new[i] += delta
                        h_new[j] -= delta
                        new_c5 = _c5_of(h_new, dx)
                        if new_c5 <= cur_c5 or rng.random() < math.exp(-(new_c5 - cur_c5) / 0.02):
                            h = h_new
                            cur_c5 = new_c5
                            if new_c5 < best_c5:
                                best_c5 = new_c5
                                best_h = h.copy()
                else:
                    max_delta = min(h[i], 1.0 - h[j])
                    if max_delta > 1e-12:
                        delta = max_delta * 0.5 * rng.random()
                        h_new = h.copy()
                        h_new[i] -= delta
                        h_new[j] += delta
                        new_c5 = _c5_of(h_new, dx)
                        if new_c5 <= cur_c5 or rng.random() < math.exp(-(new_c5 - cur_c5) / 0.02):
                            h = h_new
                            cur_c5 = new_c5
                            if new_c5 < best_c5:
                                best_c5 = new_c5
                                best_h = h.copy()

        # periodic restart to diversify search
        if now - last_restart > RESTART_EVERY:
            # keep the best so far, start a fresh random feasible vector
            h = rng.random(n)
            h = _project_to_constraints(h, target_sum)
            cur_c5 = _c5_of(h, dx)
            last_restart = now

    # final projection (to correct tiny drift) and evaluation
    best_h = _project_to_constraints(best_h, target_sum)
    final_c5 = _c5_of(best_h, dx)

    return best_h, final_c5, n
