import time
import numpy as np

# -------------------------------------------------
# Helper functions (all top‑level as required)
# -------------------------------------------------

def compute_c5(h: np.ndarray) -> float:
    """Return max overlap integral for a step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    # linear correlation h * (1-h) (zero padding outside the interval)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def normalize_h(h: np.ndarray) -> np.ndarray:
    """Project h onto the box [0,1]^n and the hyperplane sum(h)=n/2."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

    # iterative redistribution of the excess/deficit
    for _ in range(20):
        cur = h.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            if idx.size == 0:
                break
            inc = min(diff, np.sum(1.0 - h[idx]))
            add = inc / idx.size
            h[idx] += add
            h = np.clip(h, 0.0, 1.0)
        else:  # diff < 0
            idx = np.where(h > 0.0)[0]
            if idx.size == 0:
                break
            dec = min(-diff, np.sum(h[idx]))
            sub = dec / idx.size
            h[idx] -= sub
            h = np.clip(h, 0.0, 1.0)

    # final tiny correction (if any) on a free entry
    cur = h.sum()
    diff = target - cur
    if abs(diff) > 1e-9:
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size:
            h[free[0]] += diff
            h = np.clip(h, 0.0, 1.0)
    return h


def upscale(old_h: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate an old step function onto a finer grid."""
    old_n = old_h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h, period=2.0)
    return normalize_h(new_h)


def improve_h(
    h_start: np.ndarray,
    rng: np.random.Generator,
    time_limit: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Simple projected‑gradient / random‑pair descent.
    Returns the best vector seen and its C5 value.
    """
    h = h_start.copy()
    n = h.shape[0]
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val

    # initial step size (mass transferred between two positions)
    step = 0.2

    it = 0
    while True:
        # time stop
        if time.time() - start_time > time_limit * 0.95:
            break
        # schedule step reduction
        if it and it % 5000 == 0:
            step = max(step * 0.5, 0.001)

        # -------------------------------------------------
        # 1) occasional deterministic gradient step
        # -------------------------------------------------
        if it % 1000 == 0:
            # gradient of the *max* overlap term
            corr = np.correlate(h, 1.0 - h, mode="full")
            k = np.argmax(corr) - (n - 1)  # shift attaining max
            grad = np.empty(n, dtype=float)
            for i in range(n):
                g = 1.0
                if 0 <= i + k < n:
                    g -= h[i + k]
                if 0 <= i - k < n:
                    g -= h[i - k]
                grad[i] = g
            # step opposite gradient (descent)
            trial = h - step * grad
            trial = normalize_h(trial)
            trial_val = compute_c5(trial)
            if trial_val < best_val:
                best_val = trial_val
                best_h = trial.copy()
            if trial_val < cur_val:
                h = trial
                cur_val = trial_val
            it += 1
            continue

        # -------------------------------------------------
        # 2) random pairwise mass transfer
        # -------------------------------------------------
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = (rng.random() - 0.5) * 2.0 * step
        hi, hj = h[i], h[j]
        if delta > 0:
            delta = min(delta, 1.0 - hi, hj)
        else:
            delta = -min(-delta, hi, 1.0 - hj)

        if abs(delta) < 1e-14:
            it += 1
            continue

        trial = h.copy()
        trial[i] = hi + delta
        trial[j] = hj - delta
        trial_val = compute_c5(trial)

        if trial_val < best_val:
            best_val = trial_val
            best_h = trial.copy()
        if trial_val < cur_val:
            h = trial
            cur_val = trial_val
        it += 1

    return best_h, best_val

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Returns a triple (h_values, c5_bound, n_points) where
    h_values is a 1‑D numpy array satisfying the constraints,
    c5_bound = max_k ∫ h(x)(1‑h(x+k))dx, and n_points = len(h_values).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # -----------------------------------------------------------------
    # Obtain an initial construction if the harness provided one.
    # -----------------------------------------------------------------
    try:
        init = initial_h_values  # supplied by the harness
        init = np.asarray(init, dtype=float)
    except NameError:
        init = None

    # -----------------------------------------------------------------
    # Grid sizes to explore (larger grids give finer approximations).
    # -----------------------------------------------------------------
    grid_sizes = [70, 100, 150, 200]

    best_h = None
    best_val = np.inf

    for N in grid_sizes:
        # stop early if time is almost exhausted
        if time.time() - start_time > budget_s * 0.85:
            break

        # ----- starting vector ----------------------------------------
        if init is not None and init.shape[0] == N:
            h0 = init.copy()
        elif init is not None:
            h0 = upscale(init, N)
        else:
            h0 = rng.random(N)
            h0 = normalize_h(h0)

        # ----- local improvement ---------------------------------------
        h_opt, val_opt = improve_h(
            h_start=h0,
            rng=rng,
            time_limit=budget_s,
            start_time=start_time,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt

        # ----- a few random restarts at this resolution --------------
        for _ in range(2):
            if time.time() - start_time > budget_s * 0.95:
                break
            rand_h = rng.random(N)
            rand_h = normalize_h(rand_h)
            h_opt2, val_opt2 = improve_h(
                h_start=rand_h,
                rng=rng,
                time_limit=budget_s,
                start_time=start_time,
            )
            if val_opt2 < best_val:
                best_val = val_opt2
                best_h = h_opt2

    # -----------------------------------------------------------------
    # Fallback: if everything failed, use a simple uniform construction.
    # -----------------------------------------------------------------
    if best_h is None:
        n = 70
        best_h = np.full(n, 0.5)
        best_h = normalize_h(best_h)
        best_val = compute_c5(best_h)

    # final projection (just in case)
    best_h = normalize_h(best_h)
    best_val = compute_c5(best_h)

    return best_h, float(best_val), int(best_h.shape[0])
