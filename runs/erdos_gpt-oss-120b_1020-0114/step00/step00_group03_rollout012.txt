def project_to_box_and_sum(v, target, lower=0.0, upper=1.0, tol=1e-8, max_iter=100):
    # find lambda such that sum(clip(v + lam, lower, upper)) = target
    # If v already within bounds, can adjust lam quickly.
    # low bound = lower - max(v) # ensure v+lam <= lower for all?
    # Actually we need start with lam low enough that all w_i = lower
    # So low = lower - np.max(v) (makes v+low <= lower for all)
    # high = upper - np.min(v) (makes v+high >= upper for all)
    low = lower - np.max(v)
    high = upper - np.min(v)
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        w = np.clip(v + mid, lower, upper)
        s = w.sum()
        if np.abs(s - target) <= tol:
            return w
        if s > target:
            high = mid
        else:
            low = mid
    # after loops, return final w
    return np.clip(v + (low + high)/2.0, lower, upper)

import time
import numpy as np
import random

def _project_to_box_and_sum(v, target, lower=0.0, upper=1.0, tol=1e-9, max_iter=60):
    """
    Project a vector onto the set {x | lower <= x_i <= upper, sum(x)=target}.
    Uses a simple water‑filling bisection on the Lagrange multiplier.
    """
    low = lower - np.max(v)    # makes all v+low <= lower
    high = upper - np.min(v)   # makes all v+high >= upper
    for _ in range(max_iter):
        mid = (low + high) * 0.5
        w = np.clip(v + mid, lower, upper)
        s = w.sum()
        if s > target:
            high = mid
        else:
            low = mid
    w = np.clip(v + (low + high) * 0.5, lower, upper)
    # tiny residual correction (should be negligible)
    residual = target - w.sum()
    if abs(residual) > 1e-12:
        free = np.where((w > lower + 1e-12) & (w < upper - 1e-12))[0]
        if free.size:
            w[free] += residual / free.size
            w = np.clip(w, lower, upper)
    return w

def _gradient_for_lag(h, lag):
    """Sub‑gradient of  c_lag = sum_i h_i(1‑h_{i+lag})  w.r.t. h."""
    n = h.size
    g = np.zeros_like(h)
    if lag >= 0:
        L = n - lag
        if L > 0:
            g[:L] += 1.0 - h[lag:]          # (1‑h_{i+lag}) part
            g[lag:] -= h[:L]                # -h_{i‑lag} part
    else:
        d = -lag
        L = n - d
        if L > 0:
            g[d:] += 1.0 - h[:L]            # (1‑h_{i‑d}) part
            g[:L] -= h[d:]                  # -h_{i+d} part
    return g

def _max_corr_and_lag(h):
    """Return max correlation value (unscaled) and the corresponding lag."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = corr.max()
    max_idx = corr.argmax()
    lag = int(max_idx - (h.size - 1))
    return max_val, lag, corr

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)

    # ---- problem size -------------------------------------------------
    try:
        init_h_arr = np.array(initial_h_values, dtype=float)
    except Exception:
        init_h_arr = None

    n = 200                     # choose a modest size (< 1k)
    target_sum = n / 2.0
    dx = 2.0 / n

    # ---- helpers ------------------------------------------------------
    def _evaluate(h):
        """C5 (scaled) for a given h."""
        max_val, _, _ = _max_corr_and_lag(h)
        return max_val * dx

    # ---- main optimisation loop ----------------------------------------
    start_time = time.time()
    time_limit = budget_s - 5          # small safety margin
    best_h = None
    best_val = np.inf

    # a few independent restarts
    restart = 0
    while time.time() - start_time < time_limit and restart < 8:
        restart += 1

        # ---- initialise ------------------------------------------------
        if restart == 1 and init_h_arr is not None:
            # upscale / downscale initial data to length n
            if init_h_arr.size == n:
                h = init_h_arr.copy()
            else:
                x_src = np.linspace(0.0, 2.0, init_h_arr.size, endpoint=False)
                x_tgt = np.linspace(0.0, 2.0, n, endpoint=False)
                h = np.interp(x_tgt, x_src, init_h_arr)
                h = _project_to_box_and_sum(h, target_sum)
        else:
            # random start respecting constraints
            rnd = np.random.rand(n)
            h = _project_to_box_and_sum(rnd, target_sum)

        cur_val, cur_lag, _ = _max_corr_and_lag(h)
        if cur_val * dx < best_val:
            best_val = cur_val * dx
            best_h = h.copy()

        # ---- local search --------------------------------------------
        it = 0
        max_it = 20000
        while it < max_it and time.time() - start_time < time_limit:
            max_val, lag, corr = _max_corr_and_lag(h)

            # collect all lags tied for the maximum (within tiny tolerance)
            tol = 1e-12
            worst_idxs = np.where(corr >= max_val - tol)[0]
            # accumulate gradient for all worst lags
            g = np.zeros_like(h)
            for idx in worst_idxs:
                cur_lag = int(idx - (n - 1))
                g += _gradient_for_lag(h, cur_lag)

            # step size schedule (simple diminishing step)
            step = 0.5 / (1.0 + it * 0.001)

            # tentative move and projection
            h_new = _project_to_box_and_sum(h - step * g, target_sum)
