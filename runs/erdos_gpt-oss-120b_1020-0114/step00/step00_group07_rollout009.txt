import numpy as np, time, math
from math import exp

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def simulated_annealing_binary(h0, rng, time_limit, start_temp=0.02, end_temp=1e-5):
    n = len(h0)
    # ensure binary counts half ones
    h = h0.copy()
    # ensure integer 0/1
    h = np.round(h).astype(int)  # just in case
    # compute current score
    current_score = compute_c5(h)
    best_h = h.copy()
    best_score = current_score
    # precompute indices of zeros and ones
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    start_time = time.time()
    iter = 0
    # schedule: T = start_temp * (end_temp/start_temp) ** (t/total_iters)
    # as we don't know total_iters, we can use exponential decay with iteration count causing T reduction.
    # Let's set total_iters = large, compute T accordingly.
    total_iters_est = 2000000  # arbitrary large
    while True:
        elapsed = time.time() - start_time
        if elapsed >= time_limit:
            break
        # sample indices to swap
        # pick random one from ones, random zero from zeros
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        h[i], h[j] = 0, 1
        # update ones/zeros lists: swap positions
        # But we can simply recompute after each swap or maintain sets.
        # Computing list updates O(1) via swaps in arrays: we can just swap entries in ones and zeros arrays
        # but easier to recompute each time due to low n.
        # We'll recompute after acceptance
        new_score = compute_c5(h)
        # Temperature schedule
        # use iteration fraction
        T = start_temp * (end_temp / start_temp) ** (iter / total_iters_est)
        delta = new_score - current_score
        accept = False
        if delta <= 0:
            accept = True
        else:
            if T > 1e-12:
                accept = rng.random() < math.exp(-delta / T)
        if accept:
            current_score = new_score
            # update best
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
            # update ones/zeros
            # after swap, i now 0, j now 1
            # we can update arrays accordingly
            # easier: recompute
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            # revert
            h[i], h[j] = 1, 0
        iter += 1
    return best_h, best_score

import time, math
import numpy as np

# --------------------------------------------------------------
# Helper functions – must be defined at top level (no closures)
# --------------------------------------------------------------

def compute_c5(h: np.ndarray) -> float:
    """Return the C5 value for a step function represented by array h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h) using linear (zero‑padded) convolution
    corr = np.correlate(h, 1 - h, mode="full")
    return dx * np.max(corr)


def random_binary_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Binary vector of length n with exactly n//2 ones."""
    perm = rng.permutation(n)
    h = np.zeros(n, dtype=int)
    h[perm[: n // 2]] = 1
    return h


def simulated_annealing_binary(h0: np.ndarray,
                               rng: np.random.Generator,
                               time_limit: float) -> tuple[np.ndarray, float]:
    """
    Simple simulated annealing that swaps a 1 with a 0 while preserving
    the exact number of ones.  Returns the best (lowest C5) vector found.
    """
    h = h0.astype(int).copy()
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score

    # maintain index lists for fast random selection
    ones = [i for i, v in enumerate(h) if v == 1]
    zeros = [i for i, v in enumerate(h) if v == 0]

    start = time.time()
    max_iters = 10_000_000
    start_temp = 0.02
    end_temp = 1e-5

    it = 0
    while True:
        # time / iteration stop
        if time.time() - start > time_limit or it >= max_iters:
            break

        # temperature schedule (exponential cooling)
        T = start_temp * (end_temp / start_temp) ** (it / max_iters)

        # pick a random 1 and a random 0 to swap
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]

        # perform the swap
        h[i], h[j] = 0, 1
        new_score = compute_c5(h)
        delta = new_score - cur_score

        # Metropolis acceptance
        accept = False
        if delta <= 0:
            accept = True
        else:
            if T > 0:
                accept = rng.random() < math.exp(-delta / T)

        if accept:
            cur_score = new_score
            # update index lists in‑place (swap the entries)
            ones[i_idx] = j
            zeros[j_idx] = i
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert the move
            h[i], h[j] = 1, 0

        it += 1

    return best_h, best_score


def local_improve_binary(h: np.ndarray, max_passes: int = 5) -> tuple[np.ndarray, float]:
    """
    Greedy improvement: try all single swaps and keep any that lower C5.
    Repeats up to max_passes times or until no improvement occurs.
    """
    h = h.astype(int).copy()
    cur_score = compute_c5(h)
    n = h.shape[0]

    for _ in range(max_passes):
        improved = False
        ones = [i for i, v in enumerate(h) if v == 1]
        zeros = [i for i, v in enumerate(h) if v == 0]

        # random order to avoid systematic bias
        rng = np.random.default_rng()
        rng.shuffle(ones)
        rng.shuffle(zeros)

        for i in ones:
            for j in zeros:
                h[i], h[j] = 0, 1
                new_score = compute_c5(h)
                if new_score + 1e-12 < cur_score:
                    cur_score = new_score
                    improved = True
                    # keep the swap and move to next pass
                    break
                else:
                    h[i], h[j] = 1, 0
            if improved:
                break
        if not improved:
            break

    return h, cur_score

        h_sa, score_sa = simulated_annealing_binary(h_start, rng, run_time)

        # ----- greedy local improvement --------------------------------------------
        h_loc, score_loc = local_improve_binary(h_sa, max_passes=5)

        # keep the best solution seen so far
        if score_loc < best_score:
            best_score = score_loc
            best_h = h_loc.copy()
            best_n = n
        if score_sa < best_score:
            best_score = score_sa
            best_h = h_sa.copy()
            best_n = n

        # early exit if we already beat the target record
        if best_score <= 0.3808:
            break

    # ------------------------------------------------------------------
    # Fallback handling – should never be triggered, but kept for safety
    # ------------------------------------------------------------------
    if best_h is None:
        if init_h is not None:
            best_h = init_h.copy()
            best_n = best_h.shape[0]
        else:
            best_n = 80
            best_h = np.zeros(best_n, dtype=float)
        best_score = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Final post‑processing – enforce exact constraints
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    # enforce the integral constraint exactly (sum = n/2)
    desired_sum = best_n / 2
    current_sum = best_h.sum()
    if abs(current_sum - desired_sum) > 1e-9:
        # simple scaling (keeps values inside [0,1] for binary vectors)
        factor = desired_sum / current_sum if current_sum != 0 else 0.0
        best_h = np.clip(best_h * factor, 0.0, 1.0)
        # tiny correction to hit the target sum exactly
        diff = desired_sum - best_h.sum()
        if abs(diff) > 1e-12:
            # adjust the first non‑boundary entry
            idx = np.where((best_h > 0.0) & (best_h < 1.0))[0]
            if idx.size == 0:
                idx = np.arange(best_n)
            best_h[idx[0]] = np.clip(best_h[idx[0]] + diff, 0.0, 1.0)

    final_score = compute_c5(best_h)

    return best_h.astype(float), final_score, best_n
