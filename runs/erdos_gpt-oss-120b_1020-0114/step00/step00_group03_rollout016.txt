import numpy as np, time, math

def project_box_sum(v, target):
    # binary search lambda
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        y = np.clip(v - lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h, dx):
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def max_shift_and_corr(h):
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (len(h) - 1)
    max_corr = corr[idx]
    return shift, max_corr, corr

def grad_for_shift(h, k):
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        kp = k
        # part from h_i factor
        # indices 0 .. n-1-k
        end = n - kp
        grad[:end] += (1 - h[kp:])
        # part from -(h_{i-k}) factor
        grad[kp:] -= h[:end]
    else:
        d = -k
        end = n - d
        grad[d:] += (1 - h[:end])
        grad[:end] -= h[d:]
    return grad

import time
import math
import numpy as np

# -------------------------------------------------------------
# Helper functions (top level, no closures)
# -------------------------------------------------------------
def _project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the set:
        {x : 0 <= x_i <= 1,  sum(x) = target}
    Uses a simple binary search for the Lagrange multiplier.
    """
    lo, hi = -2.0, 2.0  # guarantees y=1 and y=0 at the ends
    for _ in range(60):
        lam = (lo + hi) * 0.5
        y = np.clip(v - lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


def _compute_c5(h: np.ndarray, dx: float) -> float:
    """C5 = max_k ∫ h(x)(1-h(x+k))dx  (discrete version)."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * dx


def _max_shift_corr(h: np.ndarray):
    """Return (shift, max_corr, full_corr_array)."""
    corr = np.correlate(h, 1 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (len(h) - 1)   # conversion from index to integer shift
    return shift, corr[idx], corr


def _grad_for_shift(h: np.ndarray, k: int) -> np.ndarray:
    """
    Gradient of  f_k(h) = Σ_i h_i (1 - h_{i+k})
    with respect to h, for a fixed integer shift k.
    """
    n = h.shape[0]
    grad = np.zeros_like(h)

    if k >= 0:
        end = n - k
        # contribution from the h_i factor
        grad[:end] += (1.0 - h[k:])
        # contribution from the -(h_{i}) factor inside (1-h_{i+k})
        grad[k:] -= h[:end]
    else:
        d = -k
        end = n - d
        # contribution from the h_{i+d} factor (the h term)
        grad[d:] += (1.0 - h[:end])
        # contribution from the -(h_{i}) factor
        grad[:end] -= h[d:]
    return grad


# -------------------------------------------------------------
# Main entry point
# -------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points) where
        h_values : np.ndarray  (step function sampled on [0,2])
        c5_bound : float      (the computed max overlap)
        n_points : int        (grid size used)
    """
    np.random.seed(seed)
    start_time = time.time()

    # ----- parameters -------------------------------------------------
    n = 140                      # grid size (feel free to adjust)
    dx = 2.0 / n
    target_sum = n / 2.0         # because ∫ h = 1  →  sum(h)*dx = 1

    # ----- initial vector ---------------------------------------------
    if "initial_h_values" in globals():
        init = initial_h_values                   # provided by the harness
        m = init.shape[0]
        # linear interpolation onto the new grid
        x_old = np.linspace(0.0, 2.0, m, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(x_new, x_old, init)
    else:
        # fallback: uniform random start respecting the bound
        h0 = np.random.rand(n)

    h0 = np.clip(h0, 0.0, 1.0)
    h0 = _project_box_sum(h0, target_sum)
    # enforce even symmetry (optional, speeds up convergence)
    h0 = 0.5 * (h0 + h0[::-1])

    cur_h = h0.copy()
    cur_c5 = _compute_c5(cur_h, dx)

    best_h = cur_h.copy()
    best_c5 = cur_c5

    it = 0
    # Main stochastic search -------------------------------------------------
    while time.time() - start_time < budget_s * 0.95:
        # temperature schedule for occasional uphill moves
        T = max(1e-6, 0.1 * (0.995 ** it))

        if np.random.rand() < 0.4:
            # deterministic move: gradient descent on the worst shift
            shift, _, _ = _max_shift_corr(cur_h)
            grad = _grad_for_shift(cur_h, shift) * dx
            step = 0.5 * (0.97 ** (it / 1000.0))
            proposal = cur_h - step * grad
        else:
            # random perturbation (Gaussian)
            step = 0.2 * (0.98 ** (it / 2000.0))
            proposal = cur_h + step * np.random.randn(n)

        # keep within [0,1] and satisfy the integral constraint
        proposal = np.clip(proposal, 0.0, 1.0)
        proposal = _project_box_sum(proposal, target_sum)
        # preserve symmetry
        proposal = 0.5 * (proposal + proposal[::-1])

        prop_c5 = _compute_c5(proposal, dx)

        # Metropolis acceptance
        if prop_c5 < cur_c5 or np.random.rand() < math.exp((cur_c5 - prop_c5) / T):
            cur_h = proposal
            cur_c5 = prop_c5

        # record best found so far
        if prop_c5 < best_c5:
            best_c5 = prop_c5
            best_h = proposal.copy()

        # early exit if we already beat the target record
        if best_c5 < 0.3808:
            break

        it += 1

    # ----- final local refinement with a constrained optimiser -------------
    try:
        from scipy.optimize import minimize

        bounds = [(0.0, 1.0) for _ in range(n)]

            return _compute_c5(x, dx)

            return np.sum(x) - target_sum

                best_h = refined

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_box_sum(best_h, target_sum)
