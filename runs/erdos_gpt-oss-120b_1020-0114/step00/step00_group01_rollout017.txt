import time
import numpy as np

def _project_capped_simplex(y, S, tol=1e-9, max_iter=200):
    """Project y onto {0<=x<=1, sum(x)=S} via bisection on the threshold."""
    lo = -np.max(y) - 1.0          # low enough so that all entries become 1
    hi = np.max(y) + 1.0           # high enough so that all entries become 0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if s > S:
            lo = lam
        else:
            hi = lam
        if hi - lo < tol:
            break
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)


def _correlation(h):
    """Full correlation of h with 1‑h, scaled by dx."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def _max_corr(h):
    """Maximum of the overlap integral."""
    return _correlation(h).max()


def _gradient(h, weights, dx):
    """Gradient of the smooth‑max of the correlation."""
    n = h.shape[0]
    g = np.zeros_like(h)
    # weights array corresponds to shifts -n+1 … n-1
    for s in range(-n + 1, n):
        w = weights[s + n - 1]
        if w == 0.0:
            continue
        if s >= 0:
            # first term: i = 0 … n‑s‑1
            i_end = n - s
            g[:i_end] += w * (1.0 - h[s:])
            # second term: i = s … n‑1
            g[s:] -= w * h[:i_end]
        else:
            t = -s
            # first term: i = t … n‑1
            i_end = n - t
            g[t:] += w * (1.0 - h[:i_end])
            # second term: i = 0 … n‑t‑1
            g[:i_end] -= w * h[t:]
    return g * dx


def _smooth_weights(corr, beta):
    """Soft‑max weights for a given beta."""
    z = beta * corr
    z -= np.max(z)                     # for numerical stability
    e = np.exp(z)
    return e / e.sum()


def _optimize_continuous(n, rng, time_limit):
    """Projected gradient descent on a continuous relaxation."""
    dx = 2.0 / n
    S = n / 2.0                         # required sum
    # start from a random feasible point
    y = rng.random(n)
    h = _project_capped_simplex(y, S)
    best_h = h.copy()
    best_val = _max_corr(h)

    beta = 5.0
    alpha = 0.8
    start = time.time()
    it = 0
    while time.time() - start < time_limit and it < 3000:
        corr = _correlation(h)
        # smooth‑max
        w = _smooth_weights(corr, beta)

        # gradient step
        grad = _gradient(h, w, dx)
        h_new = _project_capped_simplex(h - alpha * grad, S)

        val_new = _max_corr(h_new)

        # simple acceptance rule
        if val_new < best_val:
            best_val = val_new
            best_h = h_new.copy()

        if val_new <= _max_corr(h):
            h = h_new
        else:
            alpha *= 0.7                     # shrink step if no progress

        # slowly increase beta to focus on the true max
        beta = min(30.0, beta * 1.02)

        # occasional random perturbation to escape flat regions
        if it % 250 == 0 and it > 0:
            pert = rng.normal(scale=0.02, size=n)
            h_pert = _project_capped_simplex(h + pert, S)
            if _max_corr(h_pert) < best_val:
                h = h_pert
                best_h = h_pert.copy()
                best_val = _max_corr(h_pert)

        it += 1
    return best_h, best_val


def _binary_hill_climb(n, rng, max_iter, time_limit):
    """Simple hill‑climbing on binary vectors (0/1 values)."""
    S = n // 2
    best_h = np.zeros(n)
    best_h[:S] = 1.0
    rng.shuffle(best_h)
    best_val = _max_corr(best_h)
    start = time.time()
    it = 0
    while time.time() - start < time_limit and it < max_iter:
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # propose swap
        cand = best_h.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        val = _max_corr(cand)
        if val < best_val:
            best_h = cand
            best_val = val
        it += 1
    return best_h, best_val


def run(seed=42, budget_s=1100, **_):
    rng = np.random.default_rng(seed)
    overall_best_h = None
    overall_best_val = np.inf
    overall_best_n = None

    # Use the provided parent solution if it exists
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n0 = h0.shape[0]
        # enforce feasibility (tiny rounding errors possible)
        S0 = n0 / 2.0
        h0 = _project_capped_simplex(h0, S0)
        val0 = _max_corr(h0)
        if val0 < overall_best_val:
            overall_best_val = val0
            overall_best_h = h0.copy()
            overall_best_n = n0

    # Different resolutions to explore
    resolutions = [60, 84, 108, 144]   # modest sizes, powers of 12 help cover 2
    start_time = time.time()
    for n in resolutions:
        remaining = budget_s - (time.time() - start_time)
        if remaining <= 0:
            break

        # Time allocated for this resolution (share proportionally)
        t_alloc = remaining / (len(resolutions) - resolutions.index(n) + 1)

        # Continuous optimisation (main effort)
        h_cont, val_cont = _optimize_continuous(n, rng, t_alloc * 0.75)
        if val_cont < overall_best_val:
            overall_best_val = val_cont
            overall_best_h = h_cont.copy()
            overall_best_n = n

        # Binary hill‑climbing as a cheap complementary search
        h_bin, val_bin = _binary_hill_climb(n, rng, max_iter=4000, time_limit=t_alloc * 0.25)
        if val_bin < overall_best_val:
            overall_best_val = val_bin
            overall_best_h = h_bin.copy()
            overall_best_n = n

    # As a final polishing step, try to upscale the best-found shape
    if overall_best_h is not None and overall_best_n < 200:
        # interpolate to a higher resolution and run a few gradient steps
        target_n = min(200, overall_best_n * 2)
        x_old = np.linspace(0, 2, overall_best_n, endpoint=False)
        x_new = np.linspace(0, 2, target_n, endpoint=False)
        h_up = np.interp(x_new, x_old, overall_best_h)
        h_up = _project_capped_simplex(h_up, target_n / 2.0)
        h_up, val_up = _optimize_continuous(target_n, rng, budget_s * 0.1)
        if val_up < overall_best_val:
            overall_best_val = val_up
            overall_best_h = h_up.copy()
            overall_best_n = target_n

    # final safety projection (ensures exact feasibility)
    if overall_best_h is not None:
        overall_best_h = _project_capped_simplex(overall_best_h, overall_best_n / 2.0)

    return overall_best_h, float(overall_best_val), int(overall_best_n)
