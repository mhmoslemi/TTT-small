def hill_climb(h0, time_budget, start_time):
    h = h0.copy()
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    curr_c5 = best_c5

    # Precompute list of indices with 1s and 0s
    ones_idx = np.flatnonzero(h > 0.5)
    zeros_idx = np.flatnonzero(h < 0.5)

    # parameters
    T0 = 0.01
    Tmin = 1e-6
    decay = 0.9995  # per iteration
    T = T0
    it = 0
    while time.time() - start_time < time_budget:
        # choose random swap candidates
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)
        # swap
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h)
        d = new_c5 - curr_c5
        if d < 0 or np.random.rand() < math.exp(-d / (T + 1e-12)):
            # accept
            curr_c5 = new_c5
            # update index lists accordingly
            if h[i] > 0.5:
                # after swap, i now has 0? Actually after swap, i originally had 1, j had 0.
                # After swap, i gets 0, j gets 1.
                # So need to update ones_idx and zeros_idx accordingly.
                # We'll recompute lists each iteration maybe expensive but okay for n~200.
                pass
        else:
            # reject revert
            h[i], h[j] = h[j], h[i]
        # update best
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = h.copy()
        # temperature decay
        T = max(Tmin, T * decay)
        it += 1
        # occasionally recompute idx arrays
        if it % 200 == 0:
            ones_idx = np.flatnonzero(h > 0.5)
            zeros_idx = np.flatnonzero(h < 0.5)
    return best_h, best_c5

def anneal(h, time_limit, start_time):
    n = len(h)
    best_h = h.copy()
    best_val = compute_c5(h)
    cur_val = best_val
    cur_h = h.copy()
    T0 = 0.01
    T = T0
    it = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= time_limit:
            break
        # pick two random positions i, j where we can transfer mass
        i, j = np.random.randint(n, size=2)
        # compute feasible delta interval
        hi = cur_h[i]
        hj = cur_h[j]
        # delta to add to i and subtract from j
        delta_min = max(-hi, hj - 1.0)
        delta_max = min(1.0 - hi, hj)
        if delta_max <= delta_min:
            continue
        # limit delta magnitude
        max_step = 0.1  # can adapt based on T
        lo = max(delta_min, -max_step)
        hi_delta = min(delta_max, max_step)
        if hi_delta <= lo:
            continue
        delta = np.random.uniform(lo, hi_delta)
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_val = compute_c5(new_h)
        d = new_val - cur_val
        if d < 0 or np.random.rand() < math.exp(-d / (T + 1e-12)):
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
        # temperature schedule
        it += 1
        # decays every 1000 iter or based on elapsed
        if it % 1000 == 0:
            T = max(1e-6, T * 0.9)
    return best_h, best_val

import time, math, random
import numpy as np

def _c5_value(h: np.ndarray) -> float:
    """Return the C5 value for a step function represented by h."""
    n = h.shape[0]
    # correlation of h with (1-h) using zero padding (full mode)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _binary_anneal(h_init: np.ndarray, time_limit: float, start: float):
    """Simulated annealing on a binary (0/1) vector preserving sum."""
    h = h_init.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_val = _c5_value(h)
    cur_val = best_val

    T = 0.02            # initial temperature
    it = 0
    while time.time() - start < time_limit:
        # current sets of ones and zeros
        ones = np.flatnonzero(h > 0.5)
        zeros = np.flatnonzero(h < 0.5)
        if ones.size == 0 or zeros.size == 0:
            break

        i = np.random.choice(ones)
        j = np.random.choice(zeros)

        # swap the two entries
        h[i], h[j] = h[j], h[i]
        new_val = _c5_value(h)
        d = new_val - cur_val

        if d < 0 or np.random.rand() < math.exp(-d / (T + 1e-12)):
            # accept move
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # reject – revert swap
            h[i], h[j] = h[j], h[i]

        it += 1
        # temperature schedule
        if it % 2000 == 0:
            T = max(1e-6, T * 0.85)

    return best_h, best_val

        hi_ = min(delta_max, max_step)
        if hi_ <= lo:
            continue

        delta = np.random.uniform(lo, hi_)
        h[i] = hi + delta
        h[j] = hj - delta

        new_val = _c5_value(h)
        d = new_val - cur_val
        if d < 0 or np.random.rand() < math.exp(-d / (T + 1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert
            h[i] = hi
            h[j] = hj

        it += 1
        if it % 3000 == 0:
            T = max(1e-7, T * 0.9)

    # final projection onto the feasible hyperplane (helps against small drift)
    sum_h = np.sum(best_h)
    if abs(sum_h - target_sum) > 1e-12:
        diff = target_sum - sum_h
        # distribute the difference uniformly over non‑boundary entries
        free = np.where((best_h > 1e-8) & (best_h < 1.0 - 1e-8))[0]
        if free.size:
            adjust = diff / free.size
            best_h[free] += adjust
            # clip again
            np.clip(best_h, 0.0, 1.0, out=best_h)

    return best_h, best_val


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimise a step function h:[0,2]→[0,1] with total integral 1
    to obtain a small Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # 1. Choose the discretisation size (even number, moderate)
    # ------------------------------------------------------------------
    n = 140                     # even, gives reasonable resolution (<1k)

    # ------------------------------------------------------------------
    # 2. Build an initial vector.
    #    If a parent construction `initial_h_values` is present, interpolate it.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(initial_h_values, dtype=float)
        if parent.shape[0] != n:
            # linear interpolation to our grid
            x_parent = np.linspace(0.0, 2.0, parent.shape[0], endpoint=False)
            x_target = np.linspace(0.0, 2.0, n, endpoint=False)
            parent = np.interp(x_target, x_parent, parent)
        # Convert to a binary start: keep exactly n/2 ones (the rest zeros)
        order = np.argsort(parent)[::-1]
        h0 = np.zeros(n, dtype=float)
        h0[order[: n // 2]] = 1.0
    except NameError:
        # no parent given – start from a random half‑filled binary pattern
        h0 = np.zeros(n, dtype=float)
        h0[np.random.choice(n, n // 2, replace=False)] = 1.0

    # ------------------------------------------------------------------
    # 3. Phase 1 – binary simulated annealing (majority of the budget)
    # ------------------------------------------------------------------
    time_for_binary = budget_s * 0.68
    best_h, best_val = _binary_anneal(h0, time_for_binary, start)

    # ------------------------------------------------------------------
    # 4. Phase 2 – continuous refinement (remaining time)
    # ------------------------------------------------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 0.5:
        best_h, best_val = _continuous_anneal(best_h, remaining, start)

    # ------------------------------------------------------------------
    # 5. Return the result
    # ------------------------------------------------------------------
    # Ensure feasibility one last time (tiny numerical drift may appear)
    np.clip(best_h, 0.0, 1.0, out=best_h)
    # enforce exact sum constraint (important for the evaluation routine)
    current_sum = best_h.sum()
    target_sum = n / 2.0
    if abs(current_sum - target_sum) > 1e-9:
        diff = target_sum - current_sum
        # adjust the first element that stays inside the box
        for idx in range(n):
            cand = best_h[idx] + diff
            if 0.0 <= cand <= 1.0:
                best_h[idx] = cand
                break

    final_c5 = _c5_value(best_h)
    return (best_h, float(final_c5), n)
