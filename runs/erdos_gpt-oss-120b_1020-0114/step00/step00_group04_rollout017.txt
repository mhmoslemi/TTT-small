import numpy as np
import time

def project_to_box_sum(v, target):
    # Find lambda such that sum(clip(v - lambda,0,1)) = target
    lo = -2.0
    hi = 2.0
    for _ in range(40):
        mid = (lo + hi) / 2.0
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    return w

def evaluate_c5(h, dx):
    # correlation of h with (1 - h)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    np.random.seed(seed)
    start = time.time()
    # retrieve initial if present
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 42
        h0 = np.random.rand(n)
    target_sum = n / 2.0
    dx = 2.0 / n

    # project initial to feasible region
    h_cur = project_to_box_sum(h0, target_sum)
    cur_val = evaluate_c5(h_cur, dx)

    best_h = h_cur.copy()
    best_val = cur_val

    # parameters
    # we allow many iterations until time budget
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= budget_s * 0.96:
            break
        progress = elapsed / budget_s
        # schedule sigma from large to small
        sigma = 0.5 * (1.0 - progress) + 0.001  # start .5 shrink to .001
        # propose new candidate
        trial = h_cur + sigma * np.random.randn(n)
        trial = project_to_box_sum(trial, target_sum)
        trial_val = evaluate_c5(trial, dx)
        # hill climbing acceptance
        if trial_val < cur_val:
            h_cur = trial
            cur_val = trial_val
        # best record
        if trial_val < best_val:
            best_val = trial_val
            best_h = trial.copy()
            # early exit if good enough
            if best_val <= 0.3808:
                # we can keep searching for maybe even better, but can break after some time
                pass
    # final best_h may be floating points; ensure within [0,1]
    best_h = np.clip(best_h, 0.0, 1.0)
    # final projection onto constraints to be safe
    best_h = project_to_box_sum(best_h, target_sum)
    best_val = evaluate_c5(best_h, dx)
    return (best_h.tolist(), float(best_val), n)

import time
import numpy as np

def _project_to_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the box [0,1]^n while keeping the sum equal to target.
    Solves min ||x-v||^2 s.t. 0<=x_i<=1, sum(x)=target via a binary search on
    the Lagrange multiplier."""
    lo, hi = -5.0, 5.0          # wide enough for all reasonable values
    for _ in range(50):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def _c5_value(h: np.ndarray, dx: float) -> float:
    """Maximum overlap integral for the current step function."""
    # np.correlate implements the discrete convolution needed for the problem.
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a step function h on [0,2] giving a low C5 bound.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values – list of samples of the step function,
        c5_bound – the computed max overlap,
        n_points – number of samples (length of h_values).
    """
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # 1.  Initialise a feasible h
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 42
        h0 = np.random.rand(n)

    target_sum = n / 2.0          # because ∫_0^2 h = 1  →  sum(h)*dx = 1
    dx = 2.0 / n

    # Ensure feasibility (box and sum)
    h_best = _project_to_box_sum(h0, target_sum)
    best_val = _c5_value(h_best, dx)

    # ------------------------------------------------------------------
    # 2.  Main search loop (random perturbations + greedy local moves)
    # ------------------------------------------------------------------
    max_sigma = 0.4
    min_sigma = 0.001

    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= budget_s * 0.96:
            break

        progress = elapsed / budget_s
        sigma = max_sigma * (1.0 - progress) + min_sigma

        if np.random.rand() < 0.65:
            # ---- random perturbation around current best -----------------
            cand = h_best + sigma * np.random.randn(n)
            cand = _project_to_box_sum(cand, target_sum)
            cand_val = _c5_value(cand, dx)
            if cand_val < best_val:
                h_best = cand
                best_val = cand_val
        else:
            # ---- greedy move reducing the worst shift --------------------
            # correlation of current best
            corr = np.correlate(h_best, 1.0 - h_best, mode="full")
            idx_max = int(np.argmax(corr))
            shift = idx_max - (n - 1)                 # integer shift amount

            # overlapping index range for this shift
            i_start = max(0, -shift)
            i_end = min(n - 1, n - 1 - shift)
            if i_end >= i_start:
                i_range = np.arange(i_start, i_end + 1)
                j_range = i_range + shift
                contrib = h_best[i_range] * (1.0 - h_best[j_range])
                local_max = int(np.argmax(contrib))
                i_max = i_range[local_max]

                # small amount to move from i_max to a low‑value position
                eps = min(0.02, h_best[i_max])
                if eps <= 0:
                    continue

                # choose a destination index with the smallest h (but not full)
                candidates = np.where(h_best < 1.0)[0]
                candidates = candidates[candidates != i_max]
                if candidates.size == 0:
                    continue
                q = candidates[int(np.argmin(h_best[candidates]))]
                eps = min(eps, 1.0 - h_best[q])      # keep within bounds
                if eps <= 0:
                    continue

                cand = h_best.copy()
                cand[i_max] -= eps
                cand[q] += eps
                cand_val = _c5_value(cand, dx)
                if cand_val < best_val:
                    h_best = cand
                    best_val = cand_val

    # ------------------------------------------------------------------
    # 3.  Final correction and return
    # ------------------------------------------------------------------
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = _project_to_box_sum(h_best, target_sum)
    final_val = _c5_value(h_best, dx)

    return (h_best.tolist(), float(final_val), n)
