def project_onto_simplex_bounds(v, target_sum):
    n = len(v)
    # Clip to [0,1]
    x = np.clip(v, 0, 1)
    # Compute deficit
    current_sum = np.sum(x)
    # If sum close to target, return
    if abs(current_sum - target_sum) < 1e-12:
        return x
    # Use water-filling
    # We'll adjust free indices gradually.
    # Compute residual = target_sum - current_sum
    residual = target_sum - current_sum
    # Determine free indices where 0 < x < 1
    # We'll loop
    max_iter = 10 * n
    for _ in range(max_iter):
        free = (x > 0) & (x < 1)
        n_free = np.sum(free)
        if n_free == 0:
            break
        delta = residual / n_free
        # Add delta to free entries
        x[free] += delta
        # Clip again and recompute residual
        # But we need to handle any that exceed bounds
        # For those >1, set to 1 and adjust residual
        over = x > 1
        if np.any(over):
            excess = np.sum(x[over] - 1)
            x[over] = 1
            residual = target_sum - np.sum(x)
            continue
        under = x < 0
        if np.any(under):
            deficit = np.sum(-x[under])
            x[under] = 0
            residual = target_sum - np.sum(x)
            continue
        # If all within bounds, done
        residual = target_sum - np.sum(x)
        if abs(residual) < 1e-12:
            break
    # Finally, clip to [0,1] to handle numerical errors
    x = np.clip(x, 0, 1)
    # If still sum not exact due to rounding, adjust smallest free index
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-8:
        # distribute diff to free entries (first)
        for i in range(n):
            if diff == 0:
                break
            if diff > 0 and x[i] < 1 - 1e-12:
                inc = min(diff, 1 - x[i])
                x[i] += inc
                diff -= inc
            elif diff < 0 and x[i] > 1e-12:
                dec = min(-diff, x[i])
                x[i] -= dec
                diff += dec
    return x
```

def smoothed_max_optimize(h_init, budget, start_time):
    h = h_init.copy()
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    # Parameters
    tau = 0.1
    tau_min = 1e-5
    tau_decay = 0.9995
    step = 0.1  # step size
    step_min = 1e-5
    step_decay = 0.9999
    # Precompute shift list
    shifts = np.arange(-(n-1), n)
    while True:
        now = time.time()
        if now - start_time >= budget:
            break
        # compute overlap values
        conv = np.correlate(h, 1 - h, mode='full') * dx
        # compute weights for smoothed max
        # to avoid overflow:
        max_c = np.max(conv)
        exp_arg = (conv - max_c) / max(tau, 1e-12)
        w = np.exp(exp_arg)
        w_sum = np.sum(w)
        if w_sum == 0:
            w = np.ones_like(w) / len(w)
        else:
            w /= w_sum
        # gradient init
        grad = np.ones(n)
        # compute contributions
        # use vectorized loops
        # shift index: 0 corresponds to s = -(n-1)
        for idx, s in enumerate(shifts):
            ws = w[idx]
            if ws == 0:
                continue
            if s >= 0:

                grad[:n - s] -= ws * h[s:]
                # k such that k-s in range: k in [s, n-1]

import numpy as np
import time
import random

def compute_c5(h: np.ndarray) -> float:
    """Return the overlap constant C5 for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)

def project_onto_simplex_bounds(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project vector v onto the box [0,1]^n while enforcing sum(v) = target_sum.
    Uses a simple water-filling / iterative clipping scheme – sufficient for the
    small dimensions used here.
    """
    x = np.clip(v, 0.0, 1.0)
    cur_sum = float(np.sum(x))
    if abs(cur_sum - target_sum) < 1e-12:
        return x

    # iterative adjustment of the free (non‑boundary) entries
    max_iter = 10 * len(v)
    for _ in range(max_iter):
        free = (x > 0.0) & (x < 1.0)
        n_free = int(np.sum(free))
        if n_free == 0:
            break
        delta = (target_sum - cur_sum) / n_free
        if abs(delta) < 1e-12:
            break
        x[free] += delta
        # enforce bounds again
        over = x > 1.0
        under = x < 0.0
        if np.any(over):
            cur_sum -= np.sum(x[over] - 1.0)
            x[over] = 1.0
        if np.any(under):
            cur_sum -= np.sum(x[under])
            x[under] = 0.0
        else:
            cur_sum = float(np.sum(x))
    # final tiny correction (numerical)
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-10:
        # distribute the tiny remainder greedily
        for i in range(len(x)):
            if diff == 0:
                break
            if diff > 0 and x[i] < 1.0:
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
            elif diff < 0 and x[i] > 0.0:
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return np.clip(x, 0.0, 1.0)

def smoothed_max_optimize(h_start: np.ndarray, budget: float) -> tuple:
    """
    Gradient descent on a smooth approximation of the max overlap.
    Uses a softmax with temperature `tau` to turn the max into a weighted sum.
    """
    n = h_start.shape[0]
    target_sum = n / 2.0
    h = h_start.copy()
    best_h = h.copy()
    best_val = compute_c5(h)

    # parameters
    tau = 0.12          # initial temperature
    tau_min = 1e-5
    tau_decay = 0.9995
    step = 0.12         # initial gradient step
    step_min = 1e-5
    step_decay = 0.9997

    shifts = np.arange(-(n - 1), n, dtype=int)

    start_time = time.time()
    while time.time() - start_time < budget:
        # current overlaps
        conv = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

        # softmax weights (avoids overflow)
        m = np.max(conv)
        w_exp = np.exp((conv - m) / max(tau, 1e-12))
        w = w_exp / np.sum(w_exp)

        # gradient of the smoothed max (see analysis notes)
        grad = np.ones(n)
        for idx, s in enumerate(shifts):
            ws = w[idx]
            if ws == 0.0:
                continue
            if s >= 0:
                # indices where k+s is inside the array
                grad[: n - s] -= ws * h[s:]
                # indices where k-s is inside the array
                grad[s:] -= ws * h[: n - s]
            else:
                s_abs = -s
                grad[s_abs:] -= ws * h[: n - s_abs]
                grad[: n - s_abs] -= ws * h[s_abs:]

        # gradient step + projection
        h_candidate = h - step * grad
        h = project_onto_simplex_bounds(h_candidate, target_sum)

        # evaluate
        val = compute_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()

        # annealing / step decay
        tau = max(tau * tau_decay, tau_min)
        step = max(step * step_decay, step_min)

        # occasional random restart to escape shallow minima
        if random.random() < 0.001:
            rnd = np.random.rand(n)
            h = project_onto_simplex_bounds(rnd, target_sum)

    return best_h, best_val
