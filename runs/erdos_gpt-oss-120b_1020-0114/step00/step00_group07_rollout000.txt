import time
import numpy as np
import cvxpy as cp

def _compute_c5(h):
    """Overlap integral for a step function sampled on a uniform grid."""
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return np.max(corr) * (2.0 / h.size)

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step function h, its C5 bound, and the number of samples.

    The routine builds a linear program that maximises the minimal
    autocorrelation sum (equivalently minimises the maximal overlap)
    using McCormick envelopes for the bilinear terms.
    After solving the LP we perform a short inexpensive hill‑climb to
    polish the solution.
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size – a moderate resolution that fits comfortably in the
    #    time budget.
    # ------------------------------------------------------------------
    n = 200                     # must be even so that n/2 is integer
    dx = 2.0 / n                # grid spacing

    # ------------------------------------------------------------------
    # 2. Linear programme with variables:
    #       h_i  : function values (0 ≤ h_i ≤ 1)
    #       Y_ij : auxiliary variables approximating h_i * h_j
    #       t    : lower bound on all autocorrelation sums Σ_i Y_{i,i+s}
    # ------------------------------------------------------------------
    h = cp.Variable(n, name="h")
    Y = cp.Variable((n, n), name="Y")
    t = cp.Variable(name="t")
    constraints = []

    # integral (mass) constraint and simple bounds on h
    constraints.append(cp.sum(h) == n / 2)
    constraints.append(h >= 0)
    constraints.append(h <= 1)

    # McCormick envelope for each product h_i * h_j
    constraints.append(Y >= 0)                                   # y_ij ≥ 0
    # y_ij ≤ h_i  (row wise)
    for i in range(n):
        constraints.append(Y[i, :] <= h[i])
    # y_ij ≤ h_j  (column wise)
    for j in range(n):
        constraints.append(Y[:, j] <= h[j])
    # y_ij ≥ h_i + h_j - 1  (lower hull, matrix form)
    constraints.append(Y >= cp.reshape(h, (n, 1)) + cp.reshape(h, (1, n)) - 1)

    # Lower bound t on every diagonal sum (i,i+s)
    constraints.append(t >= 0)
    for s in range(-(n - 1), n):
        if s >= 0:
            idx = range(n - s)
            expr = cp.sum(cp.hstack([Y[i, i + s] for i in idx]))
        else:
            sneg = -s
            idx = range(n - sneg)
            expr = cp.sum(cp.hstack([Y[i + sneg, i] for i in idx]))
        constraints.append(expr >= t)

    # ------------------------------------------------------------------
    # 3. Solve the LP – we use ECOS which handles dense linear programmes
    #    reasonably fast for the chosen size.
    # ------------------------------------------------------------------
    prob = cp.Problem(cp.Maximize(t), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000,
                   abstol=1e-8, reltol=1e-8, feastol=1e-8)
    except Exception:
        # If the solver fails we fall back to a simple feasible point.
        h_opt = np.full(n, 0.5)
        best_h = np.clip(h_opt, 0, 1)
        best_c5 = _compute_c5(best_h)
        return best_h, best_c5, n

    # ------------------------------------------------------------------
    # 4. Extract the solution, enforce feasibility (tiny numeric errors)
    # ------------------------------------------------------------------
    h_raw = np.array(h.value).flatten()
    # Clip to the admissible interval
    h_raw = np.clip(h_raw, 0.0, 1.0)

    # Re‑normalise the integral exactly (preserves the bound to machine eps)
    current_sum = h_raw.sum()
    target_sum = n / 2.0
    if current_sum > 0:
        h_raw *= target_sum / current_sum
        h_raw = np.clip(h_raw, 0.0, 1.0)

    best_h = h_raw.copy()
    best_c5 = _compute_c5(best_h)

    # ------------------------------------------------------------------
    # 5. Simple hill‑climbing to improve the bound while time permits.
    #    We move a small amount of mass from one point to another,
    #    keeping the total sum constant and respecting the [0,1] box.
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s - 0.3:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # maximise the amount we can transfer from j to i without violating bounds
        max_to_i = 1.0 - best_h[i]       # free capacity at i
        max_from_j = best_h[j]           # amount we can take from j
        max_delta = min(max_to_i, max_from_j)
        if max_delta <= 1e-12:
            continue

        # propose a random transfer (biased towards larger moves)
        delta = rng.random() * max_delta
        cand = best_h.copy()
        cand[i] += delta
        cand[j] -= delta

        cand_c5 = _compute_c5(cand)
        # accept only strict improvements (tiny tolerance avoids endless loops)
        if cand_c5 + 1e-9 < best_c5:
            best_h = cand
            best_c5 = cand_c5

    # ------------------------------------------------------------------
    # 6. Return the best candidate discovered.
    # ------------------------------------------------------------------
    return best_h, float(best_c5), n
