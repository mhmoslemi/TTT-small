def project_to_feasible(v, target):
    v = np.array(v, dtype=float)
    lo = -np.max(v) - 1
    hi = np.max(v) + 1
    for _ in range(60):
        lam = (lo + hi) / 2
        x = np.clip(v - lam, 0, 1)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo+hi)/2, 0, 1)
    # Adjust small rounding error to exactly satisfy sum
    diff = target - x.sum()
    if abs(diff) > 1e-8:
        # distribute diff proportionally among non-bound elements
        free = (x > 0) & (x < 1)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0, 1)
        else:
            # adjust any element (choose random)
            idx = np.argmax(x)
            x[idx] += diff
            x = np.clip(x, 0, 1)
    return x

import time, math
import numpy as np

def _project_to_feasible(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the set {x in [0,1]^n : sum(x)=target_sum}."""
    # Clip initially to the box.
    x = np.clip(v, 0.0, 1.0)
    # Simple water‑filling: find λ such that sum(clamp(x-λ,0,1)) = target_sum.
    lo = -np.max(v) - 1.0   # low enough -> all entries become 1 after shift
    hi = np.max(v) + 1.0    # high enough -> all entries become 0 after shift
    for _ in range(60):
        lam = (lo + hi) * 0.5
        y = np.clip(v - lam, 0.0, 1.0)
        s = y.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # Fix tiny rounding error in the sum.
    diff = target_sum - x.sum()
    if abs(diff) > 1e-10:
        free = (x > 0.0) & (x < 1.0)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            # nothing free – just adjust the largest element
            idx = np.argmax(x)
            x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
    return x

def _c5_of(h: np.ndarray) -> float:
    """Compute the Erdős overlap constant for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def _max_shift_gradient(h: np.ndarray):
    """
    Sub‑gradient of the max‑overlap functional.
    Returns (grad, current_max) where grad is d/dh max_k ∫h(x)(1‑h(x+k))dx .
    """
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    kmax = int(np.argmax(corr))
    s = kmax - (n - 1)                # shift corresponding to the max entry
    grad = np.zeros_like(h)

    if s >= 0:
        i0, i1 = 0, n - s
    else:
        i0, i1 = -s, n
    inds = np.arange(i0, i1, dtype=int)
    j = inds + s                      # partner indices (always in‑range)

    # derivative of Σ h_i (1‑h_{i+s}) w.r.t. h_i  : 1‑h_{i+s}
    # derivative w.r.t. h_{i+s}                     : -h_i
    grad[inds] += 1.0 - h[j]
    grad[j]   -= h[inds]

    return grad, float(corr.max())

def run(seed: int = 42, budget_s: float = 1100.0, **kwargs):
    """
    Return a step function h, its C5 bound, and the number of samples.
    The routine performs a stochastic projected gradient / annealing search.
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------ #
    # 1. initialise the discretisation
    # ------------------------------------------------------------------ #
    init = globals().get('initial_h_values', None)
    if init is not None:
        init = np.asarray(init, dtype=float)
        n = init.shape[0]
        # optionally increase resolution modestly (max 300 points)
        if n < 200:
            n_target = min(300, n * 2)
            x_old = np.linspace(0.0, 2.0, n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
            h = np.interp(x_new, x_old, init)
            n = n_target
        else:
            h = init.copy()
    else:
        n = 200
        h = np.full(n, 0.5)

    target_sum = n / 2.0
    h = _project_to_feasible(h, target_sum)

    best_h = h.copy()
    best_val = _c5_of(best_h)

    # ------------------------------------------------------------------ #
    # 2. optimisation loop
    # ------------------------------------------------------------------ #
    cur_h = best_h.copy()
    cur_val = best_val

    # parameters for the stochastic part
    # we blend two strategies: (a) small random perturbations,
    # (b) occasional projected gradient step on the current worst shift.
    iter_count = 0
    last_improve_iter = 0

    while time.time() - start < budget_s * 0.96:
        elapsed = time.time() - start
        frac = elapsed / (budget_s * 0.96)

        # ------------------------------------------------------------------
        # a) random perturbation (simulated‑annealing style)
        # ------------------------------------------------------------------
        scale = max(0.001, 0.4 * (1.0 - frac))          # shrinking perturbation size
        candidate = cur_h + rng.normal(scale=scale, size=n)
        candidate = np.clip(candidate, 0.0, 1.0)
        candidate = _project_to_feasible(candidate, target_sum)

        cand_val = _c5_of(candidate)

        # acceptance rule (temperature decreasing with time)
        temperature = 0.02 + 0.3 * (1.0 - frac)
        accept = False
        if cand_val < cur_val:
            accept = True
        else:
            prob = math.exp(-(cand_val - cur_val) / temperature)
            accept = rng.random() < prob

        if accept:
            cur_h, cur_val = candidate, cand_val
            if cand_val < best_val - 1e-9:
                best_h, best_val = candidate.copy(), cand_val
                last_improve_iter = iter_count

        # ------------------------------------------------------------------
        # b) every few hundred iterations try a directed gradient step
        # ------------------------------------------------------------------
        if iter_count % 300 == 0:
            grad, max_val = _max_shift_gradient(cur_h)
            # step size adaptive: larger when far from optimum, smaller near
            step = 0.5 * (1.0 - frac) + 0.05
            new_h = cur_h - step * grad
            new_h = np.clip(new_h, 0.0, 1.0)
            new_h = _project_to_feasible(new_h, target_sum)
            new_val = _c5_of(new_h)

            if new_val < cur_val:
                cur_h, cur_val = new_h, new_val
                if new_val < best_val - 1e-9:
                    best_h, best_val = new_h.copy(), new_val
                    last_improve_iter = iter_count

        # ------------------------------------------------------------------
        # c) occasional restart from the current best + small noise
        # ------------------------------------------------------------------
        if iter_count - last_improve_iter > 2500:
            # re‑seed from the best solution
            cur_h = best_h + rng.normal(scale=0.03, size=n)
            cur_h = np.clip(cur_h, 0.0, 1.0)
            cur_h = _project_to_feasible(cur_h, target_sum)
            cur_val = _c5_of(cur_h)
            last_improve_iter = iter_count

        iter_count += 1

    # ---------------------------------------------------------------------- #
    # 3. final evaluation and return
    # ---------------------------------------------------------------------- #
    final_val = _c5_of(best_h)
    return best_h, final_val, n
