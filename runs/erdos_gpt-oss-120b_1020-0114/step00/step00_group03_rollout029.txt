def local_search(h_initial, max_iter, time_limit):
    h = h_initial.copy()
    best_h = h.copy()
    best_val = evaluate(h)
    n = len(h)
    step = 0.02  # typical delta magnitude; adapt?
    start = time.time()
    while time.time() - start < time_limit:
        # propose random move
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j: continue
        # decide transfer direction: from i to j
        # ensure we move some positive mass; we need both i>0 and j<1.
        if h[i] == 0 or h[j] == 1:
            continue
        delta = step * np.random.rand()
        delta = min(delta, h[i], 1 - h[j])
        if delta <= 0: continue
        h[i] -= delta
        h[j] += delta
        # Evaluate
        val = evaluate(h)
        if val < best_val - 1e-9:  # improvement
            best_val = val
            best_h = h.copy()
            # possibly increase step adaptively
        else:
            # revert move (or accept with small probability)
            # using simulated annealing temperature decreasing
            # we can compute acceptance probability: exp(-(val-best_val)/temp)
            # keep simple: revert if not improvement
            h[i] += delta
            h[j] -= delta
        # optionally reduce step adaptively
    return best_h, best_val

def _c5_from_h(h):

    n = target_n
    dx = 2.0 / n

    # ------------------------------------------------------------------
    # 2.  Prepare data structures for fast incremental updates.
    # ------------------------------------------------------------------
    # self‑correlation of h with itself (needed because
    # max_{k} h·(1‑h_{k}) = 1 – dx·min self‑corr).
    self_corr = np.correlate(h, h, mode="full")          # length 2*n‑1
    cur_min = self_corr.min()
    cur_val = 1.0 - dx * cur_min                         # current C5
    best_h = h.copy()
    best_val = cur_val

    idx = np.arange(n)                                   # constant index array

    # ------------------------------------------------------------------
    # 3.  Simulated‑annealing style local search.
    # ------------------------------------------------------------------
    # Parameters (tuned empirically for speed / quality)
    max_step = 0.07               # largest absolute mass moved in one trial
    T_start = 1e-3
    T_end = 1e-6

    # time budget for the optimizer (leave a safety margin)
    remaining = budget_s - (time.time() - start_time) - 0.5
    if remaining <= 0:
        return best_h, _c5_from_h(best_h), n
    end_time = start_time + remaining

    # Working copies.
    cur_h = h.copy()
    cur_corr = self_corr.copy()

    # Helper: recompute whole correlation occasionally to cancel drift.
    def recompute():
        nonlocal cur_corr, cur_min, cur_val
        cur_corr = np.correlate(cur_h, cur_h, mode="full")
        cur_min = cur_corr.min()
        cur_val = 1.0 - dx * cur_min

    # Main loop.
    iter_count = 0
    while time.time() < end_time:
        iter_count += 1
        # pick two distinct indices with room to move mass
        i = np.random.randint(n)
        if cur_h[i] <= 1e-12:
            continue
        j = np.random.randint(n)
        if i == j or cur_h[j] >= 1.0 - 1e-12:
            continue

        # feasible delta
        delta = np.random.rand() * max_step
        delta = min(delta, cur_h[i], 1.0 - cur_h[j])
        if delta <= 0:
            continue

        old_i = cur_h[i]
        old_j = cur_h[j]

        # ----   Apply the incremental update to the self‑correlation  ----
        offs_i = (idx - i) + (n - 1)          # indices of shifts where i participates
        offs_j = (idx - j) + (n - 1)

        # note: cur_h is still the old vector at this point
        cur_corr[offs_i] -= delta * cur_h
        cur_corr[offs_j] += delta * cur_h

        # diagonal (shift 0) needs the quadratic correction
        cur_corr[n - 1] += (old_j - old_i) * delta + 2.0 * delta * delta

        # ----   Update the vector itself  ----
        cur_h[i] = old_i - delta
        cur_h[j] = old_j + delta

        # ----   Evaluate the new state  ----
        new_min = cur_corr.min()
        new_val = 1.0 - dx * new_min

        # temperature decreasing linearly with elapsed time
        elapsed = time.time() - start_time
        frac = (elapsed) / remaining
        T = T_start * (1.0 - frac) + T_end * frac

        # Accept if better, otherwise with Metropolis probability
        improve = new_val < cur_val - 1e-12
        if improve or np.random.rand() < math.exp(-(new_val - cur_val) / T):
            cur_val = new_val
            cur_min = new_min
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_h = cur_h.copy()
        else:
            # ----   Revert the move  ----
            cur_h[i] = old_i
            cur_h[j] = old_j
            cur_corr[offs_i] += delta * cur_h   # cur_h now restored to old_i/old_j
            cur_corr[offs_j] -= delta * cur_h
            cur_corr[n - 1] -= (old_j - old_i) * delta + 2.0 * delta * delta

        # occasional full recompute to keep numerical drift tiny
        if iter_count % 2000 == 0:
            recompute()

    # ------------------------------------------------------------------
    # 4.  Post‑process the best vector and compute the final bound.
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    # final renormalisation (tiny adjustments)
    best_h *= target_sum / best_h.sum()
    best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = _c5_from_h(best_h)

    return best_h, final_c5, n
