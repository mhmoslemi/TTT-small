def compute_c5(h_vec):
    N = h_vec.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h_vec, 1 - h_vec, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def _c5_value(h):
    """Return the C5 bound for a vector h."""
    n = h.shape[0]

    # correlation of h with (1-h) on the interval [0,2]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

            h_new[i1] = 0.0
            h_new[i0] = 1.0
        else:                        # fractional transfer between two coordinates
            i, j = np.random.randint(N, size=2)
            if i == j:
                continue
            # decide direction of transfer
            if np.random.rand() < 0.5:
                max_delta = min(1.0 - h_cur[i], h_cur[j])
                if max_delta <= 0:
                    continue
                delta = np.random.rand() * max_delta
                h_new = h_cur.copy()
                h_new[i] += delta
                h_new[j] -= delta
            else:
                max_delta = min(h_cur[i], 1.0 - h_cur[j])
                if max_delta <= 0:
                    continue
                delta = np.random.rand() * max_delta
                h_new = h_cur.copy()
                h_new[i] -= delta
                h_new[j] += delta

        new_score = _c5_value(h_new)

        # --------------------------------------------------------------
        # temperature schedule & Metropolis acceptance
        # --------------------------------------------------------------
        elapsed = time.time() - start_time
        T = T0 * (1.0 - elapsed / total_budget)
        if T < 1e-8:
            T = 1e-8

        if new_score < cur_score or np.random.rand() < np.exp(-(new_score - cur_score) / T):
            h_cur = h_new
            cur_score = new_score
            no_improve = 0
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # keep the best solution seen so far
        # --------------------------------------------------------------
        if cur_score < best_score:
            best_h = h_cur.copy()
            best_score = cur_score
            no_improve = 0

        # --------------------------------------------------------------
        # occasional restart to escape deep plateaus
        # --------------------------------------------------------------
        if no_improve > max_no_improve:
            h_cur = np.zeros(N, dtype=np.float64)
            h_cur[: N // 2] = 1.0
            np.random.shuffle(h_cur)
            cur_score = _c5_value(h_cur)
            no_improve = 0

    # ------------------------------------------------------------------
    # Optional fine‑tuning with small fractional moves (pure descent)
    # ------------------------------------------------------------------
    h_cur = best_h.copy()
    cur_score = best_score
    for _ in range(20000):
        i, j = np.random.randint(N, size=2)
        if i == j:
            continue
        # try to move a tiny amount from one coordinate to another
        if np.random.rand() < 0.5:
            max_delta = min(1.0 - h_cur[i], h_cur[j])
            if max_delta <= 0:
                continue
            delta = np.random.rand() * max_delta * 0.1
            h_new = h_cur.copy()
            h_new[i] += delta
            h_new[j] -= delta
        else:
            max_delta = min(h_cur[i], 1.0 - h_cur[j])
            if max_delta <= 0:
                continue
            delta = np.random.rand() * max_delta * 0.1
            h_new = h_cur.copy()
            h_new[i] -= delta
            h_new[j] += delta

        new_score = _c5_value(h_new)
        if new_score < cur_score:
            h_cur = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_h = h_cur.copy()
                best_score = cur_score

    # ------------------------------------------------------------------
    # Final correction of the linear constraint (exact sum = N/2)
    # ------------------------------------------------------------------
    target_sum = N / 2.0
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        # distribute the tiny discrepancy without violating bounds
        for idx in range(N):
            if diff > 0 and best_h[idx] < 1.0:
                add = min(diff, 1.0 - best_h[idx])
                best_h[idx] += add
                diff -= add
            elif diff < 0 and best_h[idx] > 0.0:
                sub = min(-diff, best_h[idx])
                best_h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break

    final_score = _c5_value(best_h)
    return best_h, final_score, N
