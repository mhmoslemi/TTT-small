def project_to_feasible(h, target_sum):
    lo, hi = -1.0, 1.0
    for iteration in range(50):
        mid = (lo + hi) / 2
        clamped = np.clip(h + mid, 0, 1)
        cur_sum = clamped.sum()
        if cur_sum < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo+hi)/2, 0, 1)

            hi = mid

def eval_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    O = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(O))

def project_to_feasible(x, target_sum):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        clamped = np.clip(x + mid, 0.0, 1.0)
        s = clamped.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)

def gradient_of_worst(h):
    n = h.shape[0]
    O = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(O)
    lag = idx - (n - 1)
    G = np.zeros_like(h)
    if lag == 0:
        G = 1 - 2 * h
    elif lag > 0:
        L = lag
        G[:n-L] += 1 - h[L:]
        G[L:] -= h[:n-L]
    else:  # lag < 0
        L = -lag
        G[L:] += 1 - h[:n-L]
        G[:n-L] -= h[L:]
    return G, lag

import time
import numpy as np

def _project_to_feasible(x: np.ndarray, target_sum: float) -> np.ndarray:
    """Project x onto the box [0,1] with summed value = target_sum."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        clamped = np.clip(x + mid, 0.0, 1.0)
        s = clamped.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)


def _eval_c5(h: np.ndarray) -> float:
    """Compute the overlap bound C5 for a given step vector h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def _gradient_worst_lag(h: np.ndarray):
    """Gradient of the worst (maximal) lag of h·(1‑h(·+lag))."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    lag = idx - (n - 1)          # signed lag
    G = np.zeros_like(h)

    if lag == 0:
        # O(0) = Σ h_i (1‑h_i) => ∂/∂h_i = 1‑2h_i
        G[:] = 1.0 - 2.0 * h
    elif lag > 0:
        L = lag
        # i from 0 .. n‑L‑1 : contribution +1‑h[i+L]
        G[: n - L] += 1.0 - h[L:]
        # i from L .. n‑1 : contribution –h[i‑L]
        G[L:] -= h[: n - L]
    else:  # lag < 0
        L = -lag
        # i from L .. n‑1 : contribution +1‑h[i‑L]
        G[L:] += 1.0 - h[: n - L]
        # i from 0 .. n‑L‑1 : contribution –h[i+L]
        G[: n - L] -= h[L:]

    return G, lag


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Find a step function h on [0,2] that (approximately) minimizes

        C5 = max_k ∫ h(x) (1‑h(x+k)) dx

    using a simple projected sub‑gradient descent with occasional random
    restarts.  Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # safety margin

    # ------------------------------------------------------------------
    # initialise h
    try:
        h = np.asarray(initial_h_values, dtype=float).copy()
        n = h.size
    except NameError:
        n = 42
        h = rng.random(n)
    target_sum = n / 2.0
    h = _project_to_feasible(h, target_sum)

    best_h = h.copy()
    best_c5 = _eval_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5

    # sub‑gradient schedule
    step_initial = 0.5
    step_min = 1e-4
    decay = 1e-4

    it = 0
    while time.time() < deadline:
        it += 1

        # gradient of the worst lag
        G, _ = _gradient_worst_lag(cur_h)

        # diminishing step size
        step = max(step_min, step_initial / (1.0 + it * decay))

        # propose a new point
        cand = cur_h - step * G
        cand = _project_to_feasible(cand, target_sum)

        cand_c5 = _eval_c5(cand)

        # accept if improvement (simple hill‑climb)
        if cand_c5 < cur_c5:
            cur_h, cur_c5 = cand, cand_c5

        # update global best
        if cur_c5 < best_c5:
            best_h, best_c5 = cur_h.copy(), cur_c5

        # occasional random restart / exploration
        if it % 5000 == 0:
            rnd = rng.random(n)
            rnd = _project_to_feasible(rnd, target_sum)
            rnd_c5 = _eval_c5(rnd)
            if rnd_c5 < best_c5:
                best_h, best_c5 = rnd.copy(), rnd_c5
                cur_h, cur_c5 = rnd.copy(), rnd_c5

    # ------------------------------------------------------------------
    # a short fine‑tuning phase with a very small step
    for _ in range(800):
        G, _ = _gradient_worst_lag(best_h)
        cand = best_h - step_min * G
        cand = _project_to_feasible(cand, target_sum)
        cand_c5 = _eval_c5(cand)
        if cand_c5 < best_c5:
            best_h, best_c5 = cand, cand_c5
        else:
            break

    return (best_h.tolist(), float(best_c5), n)
