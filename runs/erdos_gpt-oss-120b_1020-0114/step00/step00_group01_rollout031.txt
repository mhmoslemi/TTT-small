import time
import math
import numpy as np

def _c5_of_h(h: np.ndarray) -> float:
    """Return the C5 value for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation h with (1-h) using the full, zero‑padded mode
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _project_onto_sum_half(x: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the feasible set:
        0 <= x_i <= 1  and  sum_i x_i = n/2
    The projection is the Euclidean projection, obtained by a simple
    binary search for a shift λ such that y_i = clip(x_i+λ,0,1) satisfies the sum.
    """
    n = x.shape[0]
    target = n / 2.0
    lo, hi = -1.0, 1.0
    # binary search for λ
    for _ in range(60):
        lam = (lo + hi) * 0.5
        y = np.clip(x + lam, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    y = np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)
    return y


def _triangle_h(n: int) -> np.ndarray:
    """
    Construct the triangular step function on [0,2] with height 1 at the centre.
    For even n the midpoint rule integrates the triangle exactly,
    giving sum_i h_i = n/2.
    """
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx          # midpoint of each subinterval
    h = np.maximum(0.0, 1.0 - np.abs(xs - 1.0))
    # enforce the linear constraint exactly (tiny numerical drift may appear)
    return _project_onto_sum_half(h)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Produce a step function h on [0,2] that yields a low C5 value.
    The construction starts from a triangular shape (which gives C5 = 1/3)
    and performs a lightweight hill‑climbing search to possibly improve it.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # --- initial construction -------------------------------------------------
    # use a reasonably fine discretisation; 200 points balances speed and quality
    n = 200
    h = _triangle_h(n)
    best_h = h.copy()
    best_val = _c5_of_h(h)

    # --- simple local optimisation -------------------------------------------
    # we keep the linear constraint exactly by moving a small amount δ from one
    # coordinate to another.  Only moves that stay inside [0,1] are considered.
    time_limit = start_time + budget_s * 0.95
    max_iters = 20000
    for it in range(max_iters):
        if time.time() >= time_limit:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # decide a direction that respects the box constraints
        if h[i] > 0.0 and h[j] < 1.0:
            src, dst = i, j
        elif h[j] > 0.0 and h[i] < 1.0:
            src, dst = j, i
        else:
            continue

        max_delta = min(h[src], 1.0 - h[dst])
        if max_delta <= 0.0:
            continue

        # step size shrinks linearly with the iteration count
        step_factor = 0.5 + 0.5 * (1.0 - it / max_iters)
        delta = rng.random() * max_delta * step_factor

        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta

        val_new = _c5_of_h(h_new)

        # accept any strict improvement; otherwise a small probabilistic uphill move
        if val_new < best_val:
            best_h, best_val = h_new, val_new
            h = h_new
        else:
            temperature = 0.02 * (1.0 - it / max_iters)
            if temperature > 0.0 and rng.random() < math.exp(-(val_new - best_val) / temperature):
                h = h_new

    # final projection to guarantee feasibility (tiny drift may have occurred)
    best_h = _project_onto_sum_half(best_h)
    final_val = _c5_of_h(best_h)

    return best_h, final_val, int(best_h.shape[0])
