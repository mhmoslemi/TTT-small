def project_box_sum(v, target, lower=0.0, upper=1.0, eps=1e-9):
    lo = lower - np.max(v) - 10.0
    hi = upper - np.min(v) + 10.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, lower, upper)
        s = w.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    # final w
    w = np.clip(v + hi, lower, upper) # hi is the largest lam s.t. sum <= target? but using hi stable.
    # Need to enforce sum exactly target: after clipping there may be slight diff due to float.
    # We'll adjust small residual by adding to indices not at boundary:
    diff = target - w.sum()
    if abs(diff) > 1e-8:
        # Distribute diff over free entries equally (but ensure no violation)
        free = (w > lower + 1e-12) & (w < upper - 1e-12)
        num_free = free.sum()
        if num_free > 0:
            w[free] += diff / num_free
            # Clip again in case overshoot
            w = np.clip(w, lower, upper)
    return w

def compute_c5_max_shift(h):
    n = h.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    # compute FFT
    A = np.fft.fft(h, m)
    B = np.fft.fft((1 - h)[::-1], m)
    corr = np.fft.ifft(A * B).real[:2*n-1]
    max_val = corr.max()
    # shift index
    shift = int(corr.argmax()) - (n - 1)
    c5 = max_val * (2.0 / n)
    return c5, shift

def gradient_wrt_h_at_shift(h, s):
    n = h.shape[0]
    if s >= 0:
        # h_plus[i] = h[i+s] if i+s < n else 0
        h_plus = np.empty_like(h)
        h_plus[:n-s] = h[s:]
        h_plus[n-s:] = 0.0
        # h_minus[i] = h[i-s] if i-s >= 0 else 0
        h_minus = np.empty_like(h)
        h_minus[s:] = h[:n-s]
        h_minus[:s] = 0.0
    else:
        ss = -s
        h_plus = np.empty_like(h)
        h_plus[ss:] = h[:n-ss]
        h_plus[:ss] = 0.0
        h_minus = np.empty_like(h)
        h_minus[:n-ss] = h[ss:]
        h_minus[n-ss:] = 0.0
    g = 1.0 - (h_plus + h_minus)
    return g

def project_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12):
    # binary search for lambda such that sum(clip(v + lam, lower, upper)) = target
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    # Ensure at lo, sum == n*lower, at hi sum == n*upper
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, lower, upper)
        s = w.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    w = np.clip(v + hi, lower, upper)
    # final correction for numeric errors
    # Might adjust a tiny amount
    diff = target - w.sum()
    if abs(diff) > 1e-9:
        # distribute diff to entries not at bound
        free = (w > lower + 1e-12) & (w < upper - 1e-12)
        num_free = free.sum()
        if num_free > 0:
            w[free] += diff / num_free
            w = np.clip(w, lower, upper)
    return w

def propose_random_move(h, max_step, rng):
    n = h.shape[0]
    i = rng.integers(n)
    j = rng.integers(n-1)
    if j >= i:
        j += 1
    # compute epsilon max
    epsilon_max = min(max_step, h[i], 1.0 - h[j])
    if epsilon_max <= 0:
        # try opposite direction if possible
        epsilon_max2 = min(max_step, h[j], 1.0 - h[i])
        if epsilon_max2 <= 0:
            return h.copy()
        epsilon = rng.random() * epsilon_max2
        h2 = h.copy()
        h2[j] -= epsilon
        h2[i] += epsilon
        return h2
    epsilon = rng.random() * epsilon_max
    h_new = h.copy()
    h_new[i] -= epsilon
    h_new[j] += epsilon
    return h_new

def compute_c5_max_shift(h):
    n = h.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    A = np.fft.fft(h, m)
    B = np.fft.fft((1 - h)[::-1], m)
    corr = np.fft.ifft(A * B).real[:2 * n - 1]
    max_idx = np.argmax(corr)
    max_val = corr[max_idx]
    shift = int(max_idx) - (n - 1)
    return max_val * (2.0 / n), shift

def multi_level_h(n, levels=5):
    # allocate equal counts for each level
    counts = n // levels
    remainder = n % levels
    values = np.linspace(0, 1, levels) # i.e., 0, 0.25, 0.5, 0.75, 1
    # adjust first remainder levels to get extra one
    h = []
    for i, v in enumerate(values):
        cnt = counts + (1 if i < remainder else 0)
        h.extend([v] * cnt)
    h = np.array(h[:n], dtype=float)
    # random shuffle to avoid structure
    np.random.shuffle(h)
    # project to sum constraint
    target = n / 2
    h = project_box_sum(h, target)
    return h

import numpy as np
import time
import math

def _project_box_sum(v: np.ndarray, target: float,
                     lower: float = 0.0, upper: float = 1.0,
                     max_iter: int = 60) -> np.ndarray:
    """Project v onto {x : lower <= x <= upper, sum(x)=target}."""
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        w = np.clip(v + mid, lower, upper)
        s = w.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    w = np.clip(v + hi, lower, upper)
    # tiny residual correction
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = (w > lower + 1e-12) & (w < upper - 1e-12)
        cnt = free.sum()
        if cnt > 0:
            w[free] += diff / cnt
            w = np.clip(w, lower, upper)
    return w


def _compute_c5_max_shift(h: np.ndarray):
    """Return (C5 value, shift) for current h using FFT correlation."""
    n = h.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    A = np.fft.fft(h, m)
    B = np.fft.fft((1.0 - h)[::-1], m)
    corr = np.fft.ifft(A * B).real[: 2 * n - 1]
    idx = int(np.argmax(corr))
    max_val = corr[idx]
    shift = idx - (n - 1)
    return max_val * (2.0 / n), shift


def _grad_wrt_shift(h: np.ndarray, s: int):
    """Gradient of f_s = ∫ h (1-h(·+s)) w.r.t. h."""
    n = h.shape[0]
    if s >= 0:
        hp = np.empty_like(h)
        hp[: n - s] = h[s:]
        hp[n - s :] = 0.0
        hm = np.empty_like(h)
        hm[s:] = h[: n - s]
        hm[: s] = 0.0
    else:
        ss = -s
        hp = np.empty_like(h)
        hp[ss:] = h[: n - ss]
        hp[: ss] = 0.0
        hm = np.empty_like(h)
        hm[: n - ss] = h[ss:]
        hm[n - ss :] = 0.0
    return 1.0 - (hp + hm)


def _step_gradient(h: np.ndarray, shift: int, eta: float):
    """Take a projected gradient step aimed at reducing the worst shift."""
    g = _grad_wrt_shift(h, shift)
    d = -g + g.mean()               # preserves sum
    h_new = h + eta * d
    target = h.shape[0] / 2.0
    return _project_box_sum(h_new, target)


def _propose_random_move(h: np.ndarray, max_step: float, rng: np.random.Generator):
    """Move ε from one coordinate to another while keeping the sum exactly."""
    n = h.shape[0]
    i = rng.integers(n)
    j = rng.integers(n - 1)
    if j >= i:
        j += 1
    eps_max = min(max_step, h[i], 1.0 - h[j])
    if eps_max <= 0.0:
        # try opposite direction
        eps_max = min(max_step, h[j], 1.0 - h[i])
        if eps_max <= 0.0:
            return h.copy()
        eps = rng.random() * eps_max
        h_new = h.copy()
        h_new[j] -= eps
        h_new[i] += eps
        return h_new
    eps = rng.random() * eps_max
    h_new = h.copy()
    h_new[i] -= eps
    h_new[j] += eps
    return h_new


def _triangle_h(n: int):
    """Triangular shape that is zero‑mean after scaling to integral 1."""
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    h = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
    target = n / 2.0
    return _project_box_sum(h, target)


def _multi_level_h(n: int, levels: int = 5):
    """Equal‑size blocks with equally spaced heights."""
    rng = np.random.default_rng(0)
    base = np.linspace(0.0, 1.0, levels)
    counts = n // levels
    remainder = n % levels
    vals = []
    for i, v in enumerate(base):
        cnt = counts + (1 if i < remainder else 0)
        vals.extend([v] * cnt)
    h = np.array(vals[:n], dtype=float)
    rng.shuffle(h)
    target = n / 2.0
    return _project_box_sum(h, target)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a step function h on [0,2] that minimizes
        C5 = max_k ∫ h(x)*(1-h(x+k)) dx
    using a simple stochastic projected‑gradient / annealing scheme.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # initialise candidates ----------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.shape[0]
        h0 = _project_box_sum(h0, n / 2.0)
        candidates = [h0.copy()]
    else:
        n = 120
        candidates = []

    # add a few more diverse starts
    candidates.append(_triangle_h(n))
    candidates.append(_multi_level_h(n, levels=5))
    for _ in range(3):
        rnd = rng.random(n)
        candidates.append(_project_box_sum(rnd, n / 2.0))

    # evaluate initial candidates ------------------------------------------------
    best_h = None
    best_c5 = float('inf')
    for h in candidates:
        c5, _ = _compute_c5_max_shift(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    if best_h is None:                     # fallback – should never happen
        best_h = _project_box_sum(rng.random(n), n / 2.0)
        best_c5, _ = _compute_c5_max_shift(best_h)

    h = best_h.copy()
    total_iters = 0

    # main optimisation loop ---------------------------------------------------
    while time.time() - start < budget_s * 0.95:
        c5, shift = _compute_c5_max_shift(h)

        # keep global best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # schedule for step size and move type
        elapsed = time.time() - start
        progress = elapsed / budget_s
        eta = 0.2 * (1.0 - progress) + 0.001          # decaying learning rate
        if rng.random() < 0.65:
            h_new = _step_gradient(h, shift, eta)
        else:
            h_new = _propose_random_move(h, max_step=0.07, rng=rng)

        c5_new, _ = _compute_c5_max_shift(h_new)

        # simple Metropolis acceptance to avoid getting stuck
        if c5_new < c5 or rng.random() < math.exp(-(c5_new - c5) / (0.01 + progress)):
            h = h_new

        total_iters += 1

    # final evaluation of the best candidate
    final_c5, _ = _compute_c5_max_shift(best_h)
    return best_h, float(final_c5), best_h.shape[0]
