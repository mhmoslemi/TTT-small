import time
import random
import math
import numpy as np

def max_corr(vec: np.ndarray) -> int:
    """Maximum value of the correlation h * (1-h) over all shifts."""
    return int(np.max(np.correlate(vec, 1 - vec, mode="full")))

def random_binary(n: int) -> np.ndarray:
    """Return a binary array of length n with exactly n//2 ones."""
    perm = np.random.permutation(n)
    h = np.zeros(n, dtype=np.uint8)
    h[perm[: n // 2]] = 1
    return h

def simulated_annealing(start_vec: np.ndarray, deadline: float,
                       global_best_vec: np.ndarray, global_best_score: int) -> tuple:
    """
    Run a simple SA on a binary vector while respecting the sum constraint.
    Returns the best vector and its score found during the run.
    """
    n = start_vec.size
    h = start_vec.copy()
    # index sets for fast sampling
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())

    cur_score = max_corr(h)
    if cur_score < global_best_score:
        global_best_score = cur_score
        global_best_vec = h.copy()

    # schedule – linear cooling from 1.0 to 0.001
    total_iters = 500_000
    T0, Tmin = 1.0, 0.001

    for it in range(total_iters):
        if time.time() >= deadline:
            break
        T = max(Tmin, T0 * (1.0 - it / total_iters))

        # propose a swap of a random 1 with a random 0
        i = random.sample(ones, 1)[0]
        j = random.sample(zeros, 1)[0]
        h[i], h[j] = h[j], h[i]

        new_score = max_corr(h)
        delta = new_score - cur_score
        accept = delta <= 0 or random.random() < math.exp(-delta / T)

        if accept:
            # keep the change, update sets and current score
            ones.remove(i)
            zeros.remove(j)
            ones.add(j)
            zeros.add(i)
            cur_score = new_score
            if cur_score < global_best_score:
                global_best_score = cur_score
                global_best_vec = h.copy()
        else:
            # revert
            h[i], h[j] = h[j], h[i]

    return global_best_vec, global_best_score

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    # even number of points – larger n gives more flexibility
    n = 120
    target_c5 = 0.3808
    target_raw = target_c5 * (n / 2)          # raw correlation threshold

    best_vec = None
    best_score = float("inf")
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # multiple SA restarts until time runs out
    while time.time() < deadline:
        start_vec = random_binary(n)
        best_vec, best_score = simulated_annealing(start_vec, deadline,
                                                   best_vec if best_vec is not None else start_vec,
                                                   best_score if best_vec is not None else float("inf"))
        if best_score <= target_raw:
            break

    # if nothing found (should not happen), fall back to the supplied construction
    if best_vec is None:
        if "initial_h_values" in globals():
            init = np.array(initial_h_values, dtype=float)
            # project onto the feasible simplex by simple scaling
            if init.size % 2 != 0:
                # make length even by truncating the last entry
                init = init[:-1]
            n = init.size
            target = n / 2.0
            s = init.sum()
            if s != 0:
                init *= target / s
            init = np.clip(init, 0.0, 1.0)
            best_vec = init
        else:
            # degenerate constant function (never optimal, just a fallback)
            n = 2
            best_vec = np.array([0.5, 0.5], dtype=float)

    # ensure exact sum constraint (tiny numerical drift possible)
    best_vec = best_vec.astype(float)
    cur_sum = best_vec.sum()

    c5_val = max_corr(best_vec) * (2.0 / best_vec.size)
    return best_vec.tolist(), float(c5_val), best_vec.size
