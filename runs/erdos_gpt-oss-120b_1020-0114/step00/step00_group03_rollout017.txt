def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    size = 2 * n - 1
    m = 1 << (size - 1).bit_length()
    # reversed (1-h)
    g_rev = (1.0 - h)[::-1]
    H = np.fft.rfft(h, n=m)
    G = np.fft.rfft(g_rev, n=m)
    conv = np.fft.irfft(H * G, n=m)[:size]
    return np.max(conv) * dx

def project_to_constraints(v, target_sum):
    v = np.clip(v, 0.0, 1.0)
    # iterative water-filling
    for _ in range(50):
        total = v.sum()
        diff = target_sum - total
        if abs(diff) < 1e-12:
            break
        free = (v > 0) & (v < 1)
        nfree = free.sum()
        if nfree == 0:
            # adjust one arbitrary element (if diff small)
            # choose element not at bound (if any)
            idx = np.argmin(v)  # smallest
            v[idx] = np.clip(v[idx] + diff, 0, 1)
            break
        delta = diff / nfree
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    # final adjust small residual
    total = v.sum()
    diff = target_sum - total
    if abs(diff) > 1e-9:
        # add diff to first element not at bound
        for idx in range(len(v)):
            if 0 < v[idx] < 1:
                v[idx] = np.clip(v[idx] + diff, 0.0, 1.0)
                break
        else:
            # all at bounds, adjust first
            v[0] = np.clip(v[0] + diff, 0.0, 1.0)
    return v

def upsample_h(h_small, n_new):
    n_small = len(h_small)
    if n_small == n_new:
        return h_small.copy()
    # positions: midpoints
    dx_small = 2.0 / n_small
    dx_new = 2.0 / n_new
    xs_small = (np.arange(n_small) + 0.5) * dx_small
    xs_new = (np.arange(n_new) + 0.5) * dx_new
    return np.interp(xs_new, xs_small, h_small)

def simulated_annealing(initial_h, total_time, start_temp=0.05, end_temp=1e-6, max_delta_frac=0.5):
    import time, random, numpy as np
    n = len(initial_h)
    target_sum = n / 2.0
    h = initial_h.copy()
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val
    start = time.time()
    while True:
        elapsed = time.time() - start
        if elapsed >= total_time:
            break
        frac = elapsed / total_time
        temp = start_temp * (1 - frac) + end_temp * frac
        # pick i (decrease) and j (increase)
        # Choose i with h[i] > 0 and j with h[j] < 1
        i_candidates = np.where(h > 1e-12)[0]
        j_candidates = np.where(h < 1 - 1e-12)[0]
        if len(i_candidates) == 0 or len(j_candidates) == 0:
            break
        i = random.choice(i_candidates)
        j = random.choice(j_candidates)
        # max possible delta
        max_delta = min(h[i], 1 - h[j])
        if max_delta <= 0:
            continue
        # propose delta
        delta = random.random() * max_delta * max_delta_frac
        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h)
        dE = new_val - cur_val
        accept = False
        if dE <= 0:
            accept = True
        else:
            prob = np.exp(-dE / max(temp, 1e-12))
            if random.random() < prob:
                accept = True
        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert
            h[i] += delta
            h[j] -= delta
    return best_h, best_val

import time
import random
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)*(1‑h(x+k))dx for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    size = 2 * n - 1
    # length for a power‑of‑two FFT (linear convolution)
    m = 1 << (size - 1).bit_length()
    g_rev = (1.0 - h)[::-1]                 # reversed (1‑h)
    H = np.fft.rfft(h, n=m)
    G = np.fft.rfft(g_rev, n=m)
    conv = np.fft.irfft(H * G, n=m)[:size]  # linear convolution
    return float(np.max(conv) * dx)


def project_to_constraints(v: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust to satisfy Σv = target (water‑filling)."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(50):
        total = v.sum()
        diff = target - total
        if abs(diff) < 1e-12:
            break
        free = (v > 0) & (v < 1)
        n_free = free.sum()
        if n_free == 0:
            # all at bounds – tweak a non‑border element if we can
            idx = np.argmin(v)
            v[idx] = np.clip(v[idx] + diff, 0.0, 1.0)
            break
        delta = diff / n_free
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    # tiny residual fix
    residual = target - v.sum()
    if abs(residual) > 1e-9:
        for i in range(v.size):
            if 0 < v[i] < 1:
                v[i] = np.clip(v[i] + residual, 0.0, 1.0)
                break
        else:
            v[0] = np.clip(v[0] + residual, 0.0, 1.0)
    return v


def upsample_h(h_small: np.ndarray, n_new: int) -> np.ndarray:
    """Linear interpolation of a step function onto a finer grid."""
    n_small = h_small.shape[0]
    if n_small == n_new:
        return h_small.copy()
    dx_small = 2.0 / n_small
    dx_new = 2.0 / n_new
    xs_small = (np.arange(n_small) + 0.5) * dx_small
    xs_new = (np.arange(n_new) + 0.5) * dx_new
    return np.interp(xs_new, xs_small, h_small)


def simulated_annealing(
    init_h: np.ndarray,
    total_time: float,
    start_temp: float = 0.05,
    end_temp: float = 1e-6,
    max_delta_frac: float = 0.5,
) -> tuple[np.ndarray, float]:
    """Simple SA that preserves the integral exactly (move mass between points)."""
    n = init_h.size
    target = n / 2.0
    h = init_h.copy()
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val
    start = time.time()
    while True:
        elapsed = time.time() - start
        if elapsed >= total_time:
            break
        frac = elapsed / total_time
        temp = start_temp * (1.0 - frac) + end_temp * frac

        # pick i (decrease) and j (increase) respecting bounds
        while True:
            i = random.randint(0, n - 1)
            if h[i] > 1e-12:
                break
        while True:
            j = random.randint(0, n - 1)
            if h[j] < 1.0 - 1e-12:
                break
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue
        delta = random.random() * max_delta * max_delta_frac
        h[i] -= delta
        h[j] += delta

        new_val = compute_c5(h)
        dE = new_val - cur_val
        accept = False
        if dE <= 0.0:
            accept = True
        else:
            prob = np.exp(-dE / max(temp, 1e-12))
            if random.random() < prob:
                accept = True
        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert move
            h[i] += delta
            h[j] -= delta
    return best_h, best_val


def greedy_refine(
    h: np.ndarray,
    time_limit: float,
    delta_frac: float = 0.1,
) -> tuple[np.ndarray, float]:
    """Hill‑climbing (only downhill moves) for a short final polish."""
    n = h.size
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val
    start = time.time()
    while time.time() - start < time_limit:
        # pick a decreasing index and an increasing index
        while True:
            i = random.randint(0, n - 1)
            if h[i] > 1e-12:
                break
        while True:
            j = random.randint(0, n - 1)
            if h[j] < 1.0 - 1e-12:
                break
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue
        delta = max_delta * delta_frac
        h[i] -= delta
        h[j] += delta
        new_val = compute_c5(h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            h[i] += delta
            h[j] -= delta
    return best_h, best_val


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Optimise a step‑function for the Erdős C₅ constant."""
    random.seed(seed)
    np.random.seed(seed)
    start_overall = time.time()

    # ---- initial construction -------------------------------------------------
    if "initial_h_values" in globals():
        h_small = np.array(initial_h_values, dtype=float)
    else:
        # fallback: half ones, half zeros on a modest grid
        n0 = 100
        h_small = np.zeros(n0, dtype=float)
        h_small[: n0 // 2] = 1.0

    # aim for a finer grid (more freedom) but keep runtime modest
    n_target = 200 if h_small.size < 150 else h_small.size
    h = upsample_h(h_small, n_target)
    target_sum = n_target / 2.0
    h = project_to_constraints(h, target_sum)

    best_h = h.copy()
    best_val = compute_c5(best_h)

    # ---- allocate time ---------------------------------------------------------
    elapsed = time.time() - start_overall
    remaining = max(0.0, budget_s - elapsed)
    if remaining <= 0.1:
        # out of time – return current best
        corr = np.correlate(best_h, 1 - best_h, mode="full")
        c5 = float(np.max(corr) * (2.0 / n_target))
        return best_h.tolist(), c5, n_target

    # majority of time for simulated annealing
    sa_time = remaining * 0.85
    h_sa, val_sa = simulated_annealing(best_h, total_time=sa_time,
                                      start_temp=0.05, end_temp=1e-6)
    if val_sa < best_val:
        best_h, best_val = h_sa, val_sa

    # a short greedy polish with the leftovers
    leftover = max(0.0, budget_s - (time.time() - start_overall) - 0.5)
    if leftover > 0.2:
        h_g, val_g = greedy_refine(best_h, time_limit=leftover)
        if val_g < best_val:
            best_h, best_val = h_g, val_g

    # final projection to guarantee the exact integral
    best_h = project_to_constraints(best_h, target_sum)

    # compute the bound with the same routine used in the evaluator
    corr = np.correlate(best_h, 1 - best_h, mode="full")
    c5_bound = float(np.max(corr) * (2.0 / n_target))

    return best_h.tolist(), c5_bound, n_target
