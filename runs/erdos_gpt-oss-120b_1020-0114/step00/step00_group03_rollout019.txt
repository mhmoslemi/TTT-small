def project_sum(arr, target_sum):
    # Adjust arr to have sum exactly target_sum, keeping bounds [0,1].
    # Simple iterative approach
    diff = target_sum - arr.sum()
    tolerance = 1e-12
    n = len(arr)
    # First attempt: distribute diff to non-boundary entries proportionally
    if abs(diff) > tolerance:
        # choose free indices
        free = np.where((arr > 0) & (arr < 1))[0]
        if len(free) == 0:
            # fallback: adjust any index
            free = np.arange(n)
        # distribute equally
        per = diff / len(free)
        for i in free:
            if per > 0:
                inc = min(per, 1.0 - arr[i])
                arr[i] += inc
                diff -= inc
            else:
                dec = min(-per, arr[i])
                arr[i] -= dec
                diff += dec
        # if residual diff remains (due to clipping), handle iteratively
        attempts = 0
        while abs(diff) > tolerance and attempts < 1000:
            i = random.randrange(n)
            if diff > 0 and arr[i] < 1.0:

                arr[i] += inc
                diff -= inc
            elif diff < 0 and arr[i] > 0.0:
                dec = min(-diff, arr[i])
                arr[i] -= dec
                diff += dec
            attempts += 1
    return arr

import numpy as np
import random
import time
import math

def _compute_c5(arr):
    """Return C5 for a given step vector."""
    n = arr.shape[0]
    dx = 2.0 / n
    corr = np.correlate(arr, 1.0 - arr, mode="full")
    return np.max(corr) * dx

def _compute_c5_and_shift(arr):
    """Return (C5, shift) where shift is the lag giving the maximum."""
    n = arr.shape[0]
    dx = 2.0 / n
    corr = np.correlate(arr, 1.0 - arr, mode="full")
    idx = int(np.argmax(corr))
    val = corr[idx] * dx
    shift = idx - (n - 1)          # zero‑lag corresponds to index n‑1
    return val, shift

def _project_sum(arr, target):
    """Adjust `arr` (elements in [0,1]) so its sum equals `target`."""
    diff = target - arr.sum()
    if abs(diff) < 1e-12:
        return arr
    n = arr.shape[0]

    # indices that can still move
    free = np.where((arr > 0.0) & (arr < 1.0))[0]
    if free.size == 0:          # everything stuck at a bound
        free = np.arange(n)

    # distribute the needed correction uniformly
    per = diff / free.size
    for i in free:
        if per > 0.0:
            inc = min(per, 1.0 - arr[i])
            arr[i] += inc
            diff -= inc
        else:
            dec = min(-per, arr[i])
            arr[i] -= dec
            diff += dec

    # tiny residual – fix greedily
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        i = random.randrange(n)
        if diff > 0.0 and arr[i] < 1.0:
            inc = min(diff, 1.0 - arr[i])
            arr[i] += inc
            diff -= inc
        elif diff < 0.0 and arr[i] > 0.0:
            dec = min(-diff, arr[i])
            arr[i] -= dec
            diff += dec
        attempts += 1
    return arr

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function with a low C5 value."""
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # 1. Starting vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        base = np.array(initial_h_values, dtype=float)
    else:
        # fallback: half ones, half zeros (integral = 1)
        n0 = 70
        base = np.concatenate(
            [np.ones(n0 // 2, dtype=float), np.zeros(n0 // 2, dtype=float)]
        )

    # ------------------------------------------------------------------
    # 2. Choose a working resolution (double the size of the parent)
    # ------------------------------------------------------------------
    target_n = max(120, len(base) * 2)          # 140 for the usual parent
    if target_n != len(base):
        x_old = np.linspace(0.0, 2.0, len(base), endpoint=False)
        x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
        h = np.interp(x_new, x_old, base)
    else:
        h = base.copy()

    # enforce feasibility
    n = target_n
    target_sum = n / 2.0
    cur_sum = h.sum()
    if cur_sum != 0.0:
        h *= (target_sum / cur_sum)
    else:
        h.fill(0.5)
    h = np.clip(h, 0.0, 1.0)
    h = _project_sum(h, target_sum)

    # ------------------------------------------------------------------
    # 3. Local optimisation (simulated annealing + directed moves)
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_val = _compute_c5(best_h)

    cur_h = h.copy()
    cur_val = best_val

    start = time.time()
    deadline = start + budget_s - 0.1          # a safety margin

    # temperature / step‑size schedules
    T0 = best_val * 0.6 + 0.005                # start temperature
    eps_start = 0.2
    eps_end = 0.001

    iter_no = 0
    last_improve = 0

    while time.time() < deadline:
        # progress‑dependent parameters
        elapsed = time.time() - start
        frac = elapsed / budget_s
        T = T0 * (1.0 - frac)
        eps = eps_start * (1.0 - frac) + eps_end * frac

        # ------------------------------------------------------------------
        # 3a – decide move type
        # ------------------------------------------------------------------
        if random.random() < 0.35:          # directed gradient‑like move
            cur_val, cur_shift = _compute_c5_and_shift(cur_h)

            # gradient of the active lag (sub‑gradient of the max term)
            g = np.zeros_like(cur_h)
            sh = cur_shift
            if sh >= 0:
                i1 = np.arange(n - sh)
                i2 = i1 + sh
                g[i1] += cur_h[i2]
                g[i2] += cur_h[i1]
            else:
                s = -sh
                i1 = np.arange(s, n)
                i2 = i1 - s
                g[i1] += cur_h[i2]
                g[i2] += cur_h[i1]

            # candidate indices
            low_idx_candidates = np.where(cur_h > 0.0)[0]
            high_idx_candidates = np.where(cur_h < 1.0)[0]
            if low_idx_candidates.size and high_idx_candidates.size:
                low_idx = low_idx_candidates[np.argmin(g[low_idx_candidates])]
                high_idx = high_idx_candidates[np.argmax(g[high_idx_candidates])]
                delta = min(eps, cur_h[low_idx], 1.0 - cur_h[high_idx])
                if delta > 0.0:
                    cur_h[low_idx] -= delta
                    cur_h[high_idx] += delta
                    new_val, _ = _compute_c5_and_shift(cur_h)
                    if new_val < cur_val or (T > 0 and random.random() < math.exp(-(new_val - cur_val) / T)):
                        cur_val = new_val
                        if new_val < best_val - 1e-9:
                            best_val = new_val
                            best_h = cur_h.copy()
                            last_improve = iter_no
                    else:
                        # revert
                        cur_h[low_idx] += delta
                        cur_h[high_idx] -= delta
        else:                               # random pair move (SA)
            i, j = random.sample(range(n), 2)
            delta = (random.random() - 0.5) * 2.0 * eps
            hi, hj = cur_h[i], cur_h[j]
            new_hi, new_hj = hi + delta, hj - delta
            if 0.0 <= new_hi <= 1.0 and 0.0 <= new_hj <= 1.0:
                cur_h[i], cur_h[j] = new_hi, new_hj
                new_val = _compute_c5(cur_h)
                accept = False
                if new_val < cur_val:
                    accept = True
                elif T > 0.0:
                    prob = math.exp(-(new_val - cur_val) / T)
                    if random.random() < prob:
                        accept = True
                if accept:
                    cur_val = new_val
                    if new_val < best_val - 1e-9:
                        best_val = new_val
                        best_h = cur_h.copy()
                        last_improve = iter_no
                else:
                    cur_h[i], cur_h[j] = hi, hj

        # ------------------------------------------------------------------
        # 3b – occasional perturbation if we have stalled
        # ------------------------------------------------------------------
        if iter_no - last_improve > 30000:
            cur_h = best_h.copy()
            cur_val = best_val
            # small random shake
            for _ in range(5):
                i, j = random.sample(range(n), 2)
                d = (random.random() - 0.5) * 2.0 * eps * 0.5
                if 0.0 <= cur_h[i] + d <= 1.0 and 0.0 <= cur_h[j] - d <= 1.0:
                    cur_h[i] += d
                    cur_h[j] -= d
            cur_val = _compute_c5(cur_h)
            last_improve = iter_no

        iter_no += 1

    # ------------------------------------------------------------------
    # 4. Final clean‑up (ensure exact feasibility)
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_sum(best_h, target_sum)
    final_c5 = _compute_c5(best_h)

    return best_h.tolist(), float(final_c5), n
