import time, numpy as np, scipy.optimize as opt, random

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def worst_shift(h):
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    # shift offset k = idx - (len(h)-1)
    k = idx - (len(h)-1)
    return k, corr[idx]

def grad_for_shift(h, k):
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        if k < n:
            grad[:n - k] = 1 - h[k:]
            grad[k:] -= h[:n - k]
        # else k >= n => no overlap, grad zero
    else:
        t = -k
        if t < n:
            grad[t:] = 1 - h[:n - t]    # because i >= t, i - t index < n
            grad[:n - t] -= h[t:]
    return grad

def hill_climb(h0, max_iters, time_limit, dx):
    h = h0.copy()
    n = len(h)
    target_sum = n/2
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    start = time.time()
    step = 0.05  # could be dynamic
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        k, val = worst_shift(h)
        grad = grad_for_shift(h, k)
        up_idx = np.where(grad < -1e-12)[0] # need increase (since gradient negative)
        down_idx = np.where(grad > 1e-12)[0] # need decrease
        if up_idx.size == 0 or down_idx.size == 0:
            # gradient zero? random perturbation
            # Add small random noise and project sum
            # simple random bump:
            delta = np.random.uniform(-step, step, size=n)
            h_new = h + delta
            # enforce bounds
            h_new = np.clip(h_new, 0, 1)
            # adjust sum to target: compute diff
            diff = target_sum - np.sum(h_new)
            # distribute diff uniformly across non-boundary entries
            # find indices not at bound
            free = np.where((h_new > 0) & (h_new < 1))[0]
            if free.size > 0:
                h_new[free] += diff / free.size
                h_new = np.clip(h_new, 0, 1)
            # after correction maybe sum off, but fine.
            h = h_new
            continue

        # pick random up and down
        i = np.random.choice(up_idx)
        j = np.random.choice(down_idx)
        # compute allowed delta to keep bounds
        max_inc = min(step, 1.0 - h[i])
        max_dec = min(step, h[j])
        # also ensure we use delta not too large; we could also compute gradient magnitude factor
        delta = min(max_inc, max_dec)
        if delta <= 1e-12:
            continue
        # Apply
        h[i] += delta
        h[j] -= delta
        # Evaluate
        cur_c5 = compute_c5(h, dx)
        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            best_h = h.copy()
        # else occasionally accept move if not improving (simulated annealing) maybe
        # we could accept with probability if random < something - but skip
        # optionally reduce step gradually
        # Decrease step size after some iterations
        if (it+1) % 500 == 0:
            step *= 0.9
    return best_h, best_c5

def slsqp_refine(h0, dx, time_remaining):
    n = len(h0)
    target_sum = n/2
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    # Use options with maxiter limited
    maxiter = 200
    res = opt.minimize(lambda x: compute_c5(x, dx), h0, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    if res.success:
        return res.x, compute_c5(res.x, dx)
    else:
        return h0, compute_c5(h0, dx)

import time
import random
import numpy as np
import scipy.optimize as _opt

# ----- Global placeholders for the optimizer -----
CURRENT_DX = None  # will be set inside run()


def compute_c5(vec, dx):
    """Return max_k ∫ h(x)(1-h(x+k)) dx for a sampled vector."""
    corr = np.correlate(vec, 1.0 - vec, mode="full")
    return np.max(corr) * dx


def worst_shift(vec):
    """Return the integer shift (relative index) where the correlation is maximal,
    together with the value of that correlation."""
    corr = np.correlate(vec, 1.0 - vec, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (len(vec) - 1)
    return shift, corr[idx]


def grad_for_shift(vec, shift):
    """Gradient of the correlation for a fixed shift w.r.t. the vector."""
    n = len(vec)
    g = np.zeros(n)
    if shift >= 0:
        k = shift
        if k < n:
            # contribution of the first factor h[i]
            g[: n - k] = 1.0 - vec[k:]
            # contribution of the second factor (when h appears as h[i+k])
            g[k:] -= vec[: n - k]
    else:
        t = -shift
        if t < n:
            g[t:] = 1.0 - vec[: n - t]
            g[: n - t] -= vec[t:]
    return g


def _project_sum_and_bounds(v, target):
    """Clip to [0,1] and then adjust the sum back to target while staying in bounds."""
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return v
    # indices that are not at a bound
    free = np.where((v > 0.0) & (v < 1.0))[0]
    if free.size > 0:
        # distribute the remaining mass uniformly over free entries
        for _ in range(5):
            v[free] += diff / free.size
            v = np.clip(v, 0.0, 1.0)
            cur = v.sum()
            diff = target - cur
            if abs(diff) < 1e-12:
                break
            free = np.where((v > 0.0) & (v < 1.0))[0]
            if free.size == 0:
                break
        return v
    # fallback: uniform scaling (rare)
    if cur > 0:
        v = np.clip(v * (target / cur), 0.0, 1.0)
    return v


def hill_climb(start_vec, max_iters, time_limit, dx):
    """Simple pairwise mass‑transfer hill‑climbing guided by the worst shift."""
    n = len(start_vec)
    target_sum = n / 2.0
    h = start_vec.copy()
    best_h = h.copy()
    best_val = compute_c5(h, dx)

    step = 0.05
    start = time.time()
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        k, cur_val = worst_shift(h)
        grad = grad_for_shift(h, k)

        # indices that want to go up (grad < 0) and down (grad > 0)
        up_idx = np.where(grad < -1e-12)[0]
        down_idx = np.where(grad > 1e-12)[0]

        if up_idx.size == 0 or down_idx.size == 0:
            # No directional information – inject a tiny random perturbation
            delta = np.random.uniform(-step, step, size=n)
            h = _project_sum_and_bounds(h + delta, target_sum)
            continue

        i = int(np.random.choice(up_idx))
        j = int(np.random.choice(down_idx))

        # amount we can move without breaking bounds
        delta = min(step, 1.0 - h[i], h[j])
        if delta <= 0:
            continue

        h[i] += delta
        h[j] -= delta

        # The transfer keeps the sum exact; only clipping errors may appear.
        # Re‑project to keep everything in [0,1] (tiny numerical drift).
        h = np.clip(h, 0.0, 1.0)

        new_val = compute_c5(h, dx)
        if new_val < best_val - 1e-12:
            best_val = new_val
            best_h = h.copy()

        # slowly reduce the step size
        if (it + 1) % 500 == 0:
            step = max(step * 0.9, 1e-4)

    return best_h, best_val


class SumConstraint:
    """Equality constraint: sum(x) == target."""
    def __init__(self, target):
        self.target = target

    def __call__(self, x):
        return np.sum(x) - self.target


def objective_func(x):
    """Wrapper for scipy optimizer – uses global CURRENT_DX."""
    return compute_c5(x, CURRENT_DX)


def slsqp_refine(x0, dx, time_allowed):
    """One SLSQP refinement step (box+equality constraints)."""
    global CURRENT_DX
    CURRENT_DX = dx
    n = len(x0)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': SumConstraint(target)}
    # give the optimizer a modest budget – it will stop early if out of time
    opts = {'maxiter': 300, 'ftol': 1e-9, 'disp': False}
    res = _opt.minimize(objective_func, x0, method='SLSQP',
                        bounds=bounds, constraints=[cons], options=opts)
    if res.success:
        return res.x, compute_c5(res.x, dx)
    return x0, compute_c5(x0, dx)


def run(seed=42, budget_s=1100, **_):
    # ------- preparation -------------------------------------------------
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()

    # initial vector (if the harness supplies one)
    if 'initial_h_values' in globals():

    else:

        n = 100

    dx = 2.0 / n

    # ------- hill‑climbing phase ----------------------------------------
    total_time = budget_s - 5.0          # leave a small slack

    remaining = total_time - elapsed
    # Use up to ~70 % of the remaining budget for a series of short climbs

    # first attempt starts from the supplied vector
    attempts = 0
    while time.time() - climb_start < climb_budget:
        attempts += 1
        # allocate a few seconds for each attempt (but not less than 5 s)

        attempt_time = min(20.0, max(5.0, time_left * 0.5))

        # generate a start point: perturb the current best a bit
        start_vec = best_vec + np.random.normal(scale=0.08, size=n)
        start_vec = np.clip(start_vec, 0.0, 1.0)

    # ------- final projection to satisfy constraints exactly ------------
    best_vec = np.clip(best_vec, 0.0, 1.0)
    best_vec = _project_sum_and_bounds(best_vec, n / 2.0)
    final_val = compute_c5(best_vec, dx)

    # Return a plain Python list as required by the harness
    return best_vec.tolist(), float(final_val), n
