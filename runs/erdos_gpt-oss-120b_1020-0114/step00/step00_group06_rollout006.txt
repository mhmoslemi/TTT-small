import numpy as np, time, math

def proj_box_sum(y, target, tol=1e-9, max_iter=60):
    low = np.min(y) - 2.0
    high = np.max(y) + 2.0
    for _ in range(max_iter):
        lam = (low + high) / 2.0
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            low = lam
        else:
            high = lam
    return np.clip(y - (low + high) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def gradient(h, beta, dx):
    n = len(h)
    M = 2 * n - 1
    c = np.correlate(h, 1 - h, mode='full') * dx
    # compute softmax weights
    # to avoid overflow, subtract max
    b_c = beta * c
    max_bc = np.max(b_c)
    w = np.exp(b_c - max_bc)
    w_sum = np.sum(w)
    w = w / w_sum
    g = np.zeros_like(h)
    N = n
    for idx, weight in enumerate(w):
        if weight == 0:
            continue
        s = idx - (N - 1)
        # indices where j+s in [0,N-1]
        start = max(0, -s)
        end = min(N, N - s)
        if start < end:
            g[start:end] += weight
            # subtract weight * h[j+s]
            g[start:end] -= weight * h[start + s:end + s]
        # indices where j-s in [0,N-1]
        start2 = max(0, s)
        end2 = min(N, N + s)
        if start2 < end2:
            g[start2:end2] -= weight * h[start2 - s:end2 - s]
    return g

import time
import math
import numpy as np

def _proj_box_sum(y, target, tol=1e-9, max_iter=60):
    """Project y onto {0<=x<=1, sum(x)=target}."""
    low = np.min(y) - 2.0
    high = np.max(y) + 2.0
    for _ in range(max_iter):
        lam = (low + high) * 0.5
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            low = lam
        else:
            high = lam
    return np.clip(y - ((low + high) * 0.5), 0.0, 1.0)


def _c5_value(h):
    """Exact C5 for a discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def _gradient_from_corr(h, corr_scaled, beta):
    """
    Gradient of the smoothed max (softmax) of the overlap values.
    corr_scaled already contains the factor dx.
    """
    n = h.shape[0]
    N = n
    # softmax weights
    bc = beta * corr_scaled
    max_bc = np.max(bc)
    w = np.exp(bc - max_bc)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    g = np.zeros_like(h)
    for idx, weight in enumerate(w):
        if weight == 0.0:
            continue
        s = idx - (N - 1)          # actual shift
        # part where j+s is inside the interval
        start = max(0, -s)
        end = min(N, N - s)
        if start < end:
            g[start:end] += weight                     # derivative of the linear term
            g[start:end] -= weight * h[start + s:end + s]   # -h[j+s] term
        # part where j-s is inside the interval
        start2 = max(0, s)
        end2 = min(N, N + s)
        if start2 < end2:
            g[start2:end2] -= weight * h[start2 - s:end2 - s]   # -h[j-s] term
    return g


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start = time.time()

    # ----------------------------------------------------------------------
    # 1.  Initialise the problem size and a feasible starting point
    # ----------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        # fallback: even length, half ones
        n = 200
        h0 = np.zeros(n, dtype=float)
        h0[: n // 2] = 1.0
        np.random.shuffle(h0)

    target_sum = n / 2.0                     # ∫ h = 1  =>  sum(h) = n/2
    h0 = _proj_box_sum(h0, target_sum)      # enforce feasibility

    best_h = h0.copy()
    best_c5 = _c5_value(best_h)

    dx = 2.0 / n

    # ----------------------------------------------------------------------
    # 2.  Main optimisation loop (gradient descent + random restarts)
    # ----------------------------------------------------------------------
    while time.time() - start < budget_s * 0.95:
        # Occasionally start from a fresh random feasible point
        if np.random.rand() < 0.15:
            y = np.random.rand(n)
            h = _proj_box_sum(y, target_sum)
        else:
            h = best_h.copy()

        beta = 1.0          # softness of the max
        step = 0.2          # initial step size

        # Inner gradient steps
        for _ in range(500):
            # correlation (scaled by dx) and current C5
            corr = np.correlate(h, 1.0 - h, mode='full') * dx
            cur_c5 = np.max(corr)
            if cur_c5 < best_c5 - 1e-12:
                best_c5 = cur_c5
                best_h = h.copy()

            # gradient of the smoothed objective
            g = _gradient_from_corr(h, corr, beta)

            # projected gradient step
            h = _proj_box_sum(h - step * g, target_sum)

            # schedules
            beta = min(beta * 1.03, 30.0)      # grow softness
            step = max(step * 0.99, 1e-5)      # shrink step

            if time.time() - start >= budget_s * 0.95:
                break

        # ------------------------------------------------------------------
        # 3.  Small random perturbations to escape shallow local minima
        # ------------------------------------------------------------------
        for _ in range(20):
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = (np.random.rand() - 0.5) * 0.2
            h_tmp = h.copy()
            h_tmp[i] = np.clip(h_tmp[i] + delta, 0.0, 1.0)
            h_tmp[j] = np.clip(h_tmp[j] - delta, 0.0, 1.0)
            h_tmp = _proj_box_sum(h_tmp, target_sum)

            tmp_c5 = _c5_value(h_tmp)
            if tmp_c5 < best_c5 - 1e-12:
                best_c5 = tmp_c5
                best_h = h_tmp.copy()
                h = h_tmp

    # final verification
    final_c5 = _c5_value(best_h)

    return best_h, float(final_c5), n
