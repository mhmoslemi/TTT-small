import time
import numpy as np

# ------------------------------------------------------------
# Helper functions
# ------------------------------------------------------------
def compute_c5(h):
    """Return the C5 value for a step function sampled by h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx


def _project_onto_constraints(x, target_sum):
    """
    Project x onto the set {0 <= y <= 1, sum(y) = target_sum}.
    Uses a simple bisection on a uniform shift lambda.
    """
    x = np.clip(x, 0.0, 1.0)
    lo, hi = -2.0, 2.0  # wide enough for any needed shift
    for _ in range(50):
        lam = (lo + hi) * 0.5
        y = np.clip(x + lam, 0.0, 1.0)
        s = y.sum()
        if s < target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)


def _gradient_for_shift(h, shift):
    """Gradient of the active overlap term w.r.t. h (ignoring the factor dx)."""
    n = h.shape[0]
    if shift >= 0:
        h_plus = np.concatenate((h[shift:], np.zeros(shift)))
        h_minus = np.concatenate((np.zeros(shift), h[:n - shift]))
    else:
        s = -shift
        h_plus = np.concatenate((np.zeros(s), h[:n - s]))
        h_minus = np.concatenate((h[s:], np.zeros(s)))
    grad = 1.0 - (h_plus + h_minus)
    return grad


def _optimize_one_start(h0, time_budget):
    """
    Gradient‑descent / projected‑gradient local search starting from h0.
    Stops when time_budget seconds have elapsed.
    """
    n = h0.shape[0]
    target_sum = n / 2.0
    h = _project_onto_constraints(h0.copy(), target_sum)

    best_h = h.copy()
    best_val = compute_c5(h)

    start = time.time()
    it = 0
    alpha = 0.2  # initial step size

    # main loop
    while time.time() - start < time_budget:
        # correlation and active shift
        corr = np.correlate(h, 1 - h, mode="full")
        max_idx = int(np.argmax(corr))
        cur_val = corr[max_idx] * (2.0 / n)

        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()

        shift = max_idx - (n - 1)

        # gradient for this shift
        grad = _gradient_for_shift(h, shift)

        # take a step
        h = h - alpha * grad

        # project back to the feasible polytope
        h = _project_onto_constraints(h, target_sum)

        it += 1
        # occasional decay of step size
        if it % 4000 == 0:
            alpha *= 0.7
        # occasional random perturbation to escape plateaus
        if it % 2500 == 0:
            noise = np.random.uniform(-0.04, 0.04, size=n)
            h = h + noise
            h = _project_onto_constraints(h, target_sum)

    return best_h, best_val


def _random_feasible(n):
    """Return a random feasible vector (sum = n/2, 0/1 values)."""
    half = n // 2
    h = np.zeros(n)
    h[:half] = 1.0
    np.random.shuffle(h)
    return h


def _smooth_initial(n, base):
    """Interpolate a base vector to length n and project to feasibility."""
    m = base.shape[0]
    x_old = np.linspace(0.0, 2.0, m, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, base)
    # clipping + projection enforce the constraints
    h = _project_onto_constraints(h, n / 2.0)
    return h


def _generate_initials(n, base_vec):
    """Return a list of diverse starting points for a given length."""
    initials = []

    # 1) Interpolated copy of the parent (if any)
    if base_vec is not None:
        initials.append(_smooth_initial(n, base_vec))

    # 2) Random binary half‑ones
    initials.append(_random_feasible(n))

    # 3) Slightly perturbed binary (shuffle + small noise)
    hb = _random_feasible(n)
    hb = hb + np.random.uniform(-0.05, 0.05, size=n)
    initials.append(_project_onto_constraints(hb, n / 2.0))

    # 4) Simple sine‑shaped guess (scaled to keep the average 0.5)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    hs = 0.5 + 0.5 * np.sin(2 * np.pi * x / 2.0)  # one full period
    initials.append(_project_onto_constraints(hs, n / 2.0))

    # 5) Triangular wave (ramp up then down)
    tri = np.abs(((x / 2.0) * 2) % 2 - 1)  # tent shape from 0 to 1
    hs2 = 1.0 - tri  # high in middle, low at edges
    initials.append(_project_onto_constraints(hs2, n / 2.0))

    return initials

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_total = time.time()

    # Grab the optional parent construction
    base_vec = None
    if "initial_h_values" in globals():
        try:
            base_vec = np.asarray(initial_h_values, dtype=float)
        except Exception:
            base_vec = None

    # Candidate sample counts (prefer modest size for speed)
    candidate_ns = []
    if base_vec is not None:
        candidate_ns.append(len(base_vec))
    # add a few handcrafted sizes (all even for binary constructions)
    candidate_ns.extend([48, 56, 64, 80, 96, 112])
    candidate_ns = sorted(set(candidate_ns))

    # Reserve a small safety margin
    budget_s = max(budget_s - 0.5, 0.0)
    per_n = budget_s / max(len(candidate_ns), 1)

    best_h = None
    best_val = np.inf
    best_n = None

    for n in candidate_ns:
        # time slice for this n
        remaining = budget_s - (time.time() - start_total)
        if remaining <= 0:
            break
        slice_time = min(per_n, remaining)

        # generate starting points
        initials = _generate_initials(n, base_vec)

        # allocate a share of the slice to each start (simple round‑robin)
        slice_per_start = slice_time / max(len(initials), 1)

        for h0 in initials:
            # run local optimisation
            h_opt, val_opt = _optimize_one_start(h0, slice_per_start)

            if val_opt < best_val:
                best_val = val_opt
                best_h = h_opt.copy()
                best_n = n

            # check global time
            if time.time() - start_total >= budget_s:
                break
        # early exit if out of time
        if time.time() - start_total >= budget_s:
            break

    # -----------------------------------------------------------------
    # If time permits, perform a quick random‑binary search on the best n
    # -----------------------------------------------------------------
    global_remaining = budget_s - (time.time() - start_total)
    if best_h is not None and global_remaining > 0.3:
        n = best_n
        target_sum = n / 2.0
        end_search = time.time() + global_remaining
        while time.time() < end_search:
            h_rand = _random_feasible(n)
            val = compute_c5(h_rand)
            if val < best_val:
                best_val = val
                best_h = h_rand.copy()

    # final safety projection (in case of tiny drift)
    if best_h is not None:
        best_h = _project_onto_constraints(best_h, best_n / 2.0)
        best_val = compute_c5(best_h)

    # fallback (should never happen)
    if best_h is None:
        best_n = 2
        best_h = np.full(best_n, 0.5)
        best_val = compute_c5(best_h)

    return best_h, float(best_val), int(best_n)
