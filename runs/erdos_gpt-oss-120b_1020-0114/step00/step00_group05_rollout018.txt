def optimize(h_initial, max_iters=500000):
    h = h_initial.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    step_max = 0.2
    T0 = 0.001
    start = time.time()
    for t in range(max_iters):
        if time.time() - start > budget - 0.1: break
        # adapt step size: reduce gradually
        step = step_max * (1 - t / max_iters)
        # propose
        i, j = np.random.randint(0, len(h), 2)
        # compute allowed delta
        delta_pos = min(step, 1 - h[i], h[j])
        delta_neg = -min(step, h[i], 1 - h[j])
        if delta_pos < delta_neg:
            continue
        delta = np.random.uniform(delta_neg, delta_pos)
        if abs(delta) < 1e-12:
            continue
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        # evaluate
        val_new = compute_c5(h_new)
        # acceptance:
        if val_new < best_val:
            best_h = h_new.copy()
            best_val = val_new
            h = h_new
        else:
            # Simulated annealing: accept with prob
            T = T0 * (1 - t / max_iters)
            if T > 0 and np.exp(-(val_new - best_val) / T) > np.random.rand():
                # accept but not update best
                h = h_new
    return best_h, best_val

import time
import math
import numpy as np

def _compute_c5(h):
    """Return the overlap bound C5 for a step function sampled in h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def _project_onto_capped_simplex(v, target_sum):
    """
    Euclidean projection of v onto the set:
        { x | 0 <= x_i <= 1,  sum x_i = target_sum }.
    Uses bisection on the threshold λ in x_i = clip(v_i - λ, 0, 1).
    """
    v = np.asarray(v, dtype=float)
    n = v.size
    # bounds for λ: low gives all ones, high gives all zeros
    low = -np.max(v) - 1.0
    high = np.max(v) + 1.0
    for _ in range(60):
        lam = (low + high) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            low = lam
        else:
            high = lam
    lam = (low + high) / 2.0
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(n, target_sum, rng):
    """Generate a random vector in [0,1]^n with prescribed sum."""
    v = rng.random(n)
    return _project_onto_capped_simplex(v, target_sum)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise h
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        # fallback: binary half‑filled vector
        n_default = 70
        h0 = np.zeros(n_default, dtype=float)
        h0[: n_default // 2] = 1.0
    n = h0.shape[0]

    # enforce feasibility
    h0 = np.clip(h0, 0.0, 1.0)
    target_sum = n / 2.0
    h0 = _project_onto_capped_simplex(h0, target_sum)

    cur_h = h0.copy()
    cur_val = _compute_c5(cur_h)

    best_h = cur_h.copy()
    best_val = cur_val

    # ------------------------------------------------------------------
    # Optimisation parameters
    # ------------------------------------------------------------------
    max_iters = 500_000
    start_time = time.time()
    T0 = 1e-3                # initial temperature for SA
    step_max_initial = 0.25  # maximal transferred mass per move
    prob_direct = 0.22       # probability to use a directed worst‑shift move

    for it in range(max_iters):
        # Time check
        if time.time() - start_time > budget_s - 0.4:
            break

        # Linear cooling for step size and temperature
        frac = it / max_iters
        step = step_max_initial * (1.0 - frac) + 1e-6
        temp = T0 * (1.0 - frac)

        # ------------------------------------------------------------------
        # Propose a new candidate
        # ------------------------------------------------------------------
        if rng.random() < prob_direct:
            # ---- directed move based on the current worst shift ----
            corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            idx_max = int(np.argmax(corr))
            shift = idx_max - (n - 1)  # can be negative, zero or positive
            if shift != 0:
                # choose a source index that can donate and a destination that can receive
                if shift > 0:
                    src = rng.integers(0, n - shift)
                    dst = src + shift
                else:
                    src = rng.integers(-shift, n)
                    dst = src + shift
                # amount that can be moved without violating bounds
                delta_max = min(step, cur_h[src], 1.0 - cur_h[dst])
                if delta_max > 0.0:
                    delta = rng.random() * delta_max
                    new_h = cur_h.copy()
                    new_h[src] -= delta
                    new_h[dst] += delta
                else:
                    continue
            else:
                # shift zero gives zero contribution, fall back to random swap
                i, j = rng.integers(0, n, size=2)
                while i == j:
                    j = rng.integers(0, n)
                delta_pos = min(step, 1.0 - cur_h[i], cur_h[j])
                delta_neg = -min(step, cur_h[i], 1.0 - cur_h[j])
                if delta_pos <= delta_neg:
                    continue
                delta = rng.uniform(delta_neg, delta_pos)
                new_h = cur_h.copy()
                new_h[i] += delta
                new_h[j] -= delta
        else:
            # ---- random pairwise exchange ----
            i, j = rng.integers(0, n, size=2)
            while i == j:
                j = rng.integers(0, n)
            delta_pos = min(step, 1.0 - cur_h[i], cur_h[j])
            delta_neg = -min(step, cur_h[i], 1.0 - cur_h[j])
            if delta_pos <= delta_neg:
                continue
            delta = rng.uniform(delta_neg, delta_pos)
            new_h = cur_h.copy()
            new_h[i] += delta
            new_h[j] -= delta

        # ------------------------------------------------------------------
        # Evaluate and accept / reject
        # ------------------------------------------------------------------
        new_val = _compute_c5(new_h)

        accept = False
        if new_val < cur_val:
            accept = True
        elif temp > 0.0:
            prob = math.exp(-(new_val - cur_val) / temp)
            if prob > rng.random():
                accept = True

        if accept:
            cur_h = new_h
            cur_val = new_val
            if cur_val < best_val:
                best_h = cur_h.copy()
                best_val = cur_val

        # occasional re‑projection to curb floating‑point drift
        if (it + 1) % 20000 == 0:
            cur_h = _project_onto_capped_simplex(cur_h, target_sum)
            cur_val = _compute_c5(cur_h)

    # Final clean‑up: ensure best solution satisfies constraints exactly
    best_h = _project_onto_capped_simplex(best_h, target_sum)
    best_val = _compute_c5(best_h)

    return best_h.tolist(), float(best_val), n
