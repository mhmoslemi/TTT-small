def generate_random_feasible(n, m=5):
    total_ones = n // 2
    # generate random lengths for intervals (positive integers)
    # ensure at most m intervals, some may be zero.
    # Use random exponential for lengths.
    lengths = np.random.exponential(scale=1.0, size=m)
    lengths = np.floor(lengths / lengths.sum() * total_ones).astype(int)
    # ensure sum equals total_ones; adjust by adding/subtracting 1 from random indices.
    diff = total_ones - lengths.sum()
    while diff != 0:
        idx = np.random.randint(m)
        if diff > 0:
            lengths[idx] += 1
            diff -= 1
        else:
            if lengths[idx] > 0:
                lengths[idx] -= 1
                diff += 1
    # generate gaps
    total_gap = n - total_ones
    gaps = np.random.exponential(scale=1.0, size=m+1)
    gaps = np.floor(gaps / gaps.sum() * total_gap).astype(int)
    diffg = total_gap - gaps.sum()
    while diffg != 0:
        idx = np.random.randint(m+1)
        if diffg > 0:
            gaps[idx] += 1
            diffg -= 1
        else:
            if gaps[idx] > 0:
                gaps[idx] -= 1
                diffg += 1
    # Build array
    h = np.zeros(n, dtype=np.int8)
    pos = 0
    for i in range(m):
        pos += gaps[i]
        if lengths[i] > 0:
            h[pos:pos+lengths[i]] = 1
            pos += lengths[i]
    # last gap after final interval not needed.
    # Ensure exactly total ones.
    assert h.sum() == total_ones
    return h
```

import time, math, numpy as np

def count_runs(arr):
    """Number of contiguous blocks of 1s."""
    start = (arr == 1) & np.concatenate(([True], arr[:-1] == 0))
    return int(start.sum())

def compute_bound(corr, total_ones, dx):
    """C5 bound from autocorrelation."""
    return (total_ones - int(corr.min())) * dx

def remove_one(corr, idx, ones_set, h):
    """Update autocorrelation when h[idx] is set from 1 to 0."""
    n = len(h)
    for p in ones_set:                     # include idx itself
        d = p - idx
        if d == 0:
            corr[n - 1] -= 1               # self‑pair
        else:
            corr[d + n - 1] -= 1
            corr[-d + n - 1] -= 1
    h[idx] = 0

def add_one(corr, idx, ones_set, h):
    """Update autocorrelation when h[idx] is set from 0 to 1."""
    n = len(h)
    for p in ones_set:
        d = p - idx
        corr[d + n - 1] += 1
        corr[-d + n - 1] += 1
    corr[n - 1] += 1                       # self‑pair
    h[idx] = 1

def apply_swap(h, corr, ones_set, zeros_set, i, j):
    """Swap a 1 at i with a 0 at j (both indices are distinct)."""
    remove_one(corr, i, ones_set, h)
    ones_set.remove(i)
    zeros_set.add(i)

    add_one(corr, j, ones_set, h)
    zeros_set.remove(j)
    ones_set.add(j)

    """Undo a previous swap."""
    apply_swap(h, corr, ones_set, zeros_set, j, i)

    # Simulated‑annealing / hill‑climbing loop
    # ------------------------------------------------------------------
    max_iters = 300000
    init_temp = 0.02
    stagnation = 0

    it = 0
    while time.time() - start_time < budget_s - 0.5 and it < max_iters:
        it += 1
        temp = init_temp * (1.0 - it / max_iters)

        i = rng.choice(list(ones_set))
        j = rng.choice(list(zeros_set))

        # perform the swap
        apply_swap(h, corr, ones_set, zeros_set, i, j)

        # keep at most five runs of 1s
        if count_runs(h) > 5:
            revert_swap(h, corr, ones_set, zeros_set, i, j)
            continue

        new_bound = compute_bound(corr, total_ones, dx)

        if new_bound < cur_bound - 1e-12:
            # improvement – keep it
            cur_bound = new_bound
            stagnation = 0
            if new_bound < best_bound - 1e-12:
                best_bound = new_bound
                best_h = h.copy()
        else:
            # maybe accept a worse move (simulated annealing)
            if temp > 0 and rng.random() < math.exp((cur_bound - new_bound) / temp):
                cur_bound = new_bound
                stagnation = 0
            else:
                # reject – undo the swap
                revert_swap(h, corr, ones_set, zeros_set, i, j)
                stagnation += 1

        # restart if stuck for a while
        if stagnation >= 20000:
            h = generate_random_feasible(n, rng=rng)
            corr = np.correlate(h, h, mode='full').astype(np.int32)
            ones_set = set(np.where(h == 1)[0].tolist())
            zeros_set = set(np.where(h == 0)[0].tolist())
            cur_bound = compute_bound(corr, total_ones, dx)
            stagnation = 0

    # ------------------------------------------------------------------
    # Return best solution found
    # ------------------------------------------------------------------
    return (best_h.astype(float), best_bound, n)
