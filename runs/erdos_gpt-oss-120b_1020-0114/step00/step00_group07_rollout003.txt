def project_to_bounds_and_sum(y, S, tol=1e-9, max_iter=100):
    lo = -1.0
    hi = 1.0
    # Actually we need range covering possible λ such that sum within [0,N].
    lo = min(y) - 1.0
    hi = max(y) + 1.0
    for _ in range(max_iter):
        mid = (lo+hi)/2.0
        x = np.clip(y - mid, 0, 1)
        total = x.sum()
        if abs(total - S) <= tol:
            return x
        if total > S:
            lo = mid
        else:
            hi = mid
    # final clamp
    x = np.clip(y - (lo+hi)/2, 0, 1)
    # Might not have exact sum due to tolerance
    # Adjust small diff by adding to interior entries
    diff = S - x.sum()
    if abs(diff) > 1e-6:
        # adjust approximate: add diff equally to all indices not at bounds.
        interior = np.where((x>1e-12) & (x<1-1e-12))[0]
        if interior.size > 0:
            x[interior] += diff / interior.size
            x = np.clip(x, 0, 1)
    return x

def gradient_for_shift(h, s):
    # s>=0 integer shift
    N = len(h)
    g = np.zeros(N)
    if s==0:
        # gradient = 2*h (since J(0) = sum_i h_i^2, derivative is 2 h_i)
        g = 2 * h
    else:
        # For each i where both i and i+s valid:
        # derivative contributions: for i < N-s: partial w.r.t h_i = h_{i+s}
        # for i >= s: partial w.r.t h_i = h_{i-s}
        # So we can compute using shifts:
        g[:N-s] += h[s:]   # h_{i+s}
        g[s:] += h[:N-s]   # h_{i-s}
    return g

def project_to_bounds_sum(y, S, tol=1e-9, max_iter=100):
    lo = -1.0
    hi = 2.0  # broad enough
    # adapt to actual range
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(y - mid, 0.0, 1.0)
        total = x.sum()
        if np.abs(total - S) <= tol:
            return x
        if total > S:
            lo = mid
        else:
            hi = mid
    # final
    x = np.clip(y - (lo+hi)/2.0, 0.0, 1.0)
    # adjust residual if needed
    diff = S - x.sum()
    if np.abs(diff) > 1e-6:
        # distribute residual uniformly among interior points (0,1)
        interior = (x > 1e-12) & (x < 1-1e-12)
        if interior.any():
            x[interior] += diff / interior.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

import numpy as np
import time

# ------------------------------------------------------------
# Projection of a vector onto the hyperplane sum(x)=S with bounds 0≤x≤1.
# ------------------------------------------------------------
def project_to_bounds_sum(y: np.ndarray, S: float, tol: float = 1e-9, max_iter: int = 100) -> np.ndarray:
    """Return the Euclidean projection of y onto {0≤x≤1, sum(x)=S}."""
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        x = np.clip(y - mid, 0.0, 1.0)
        total = x.sum()
        if abs(total - S) <= tol:
            return x
        if total > S:
            lo = mid
        else:
            hi = mid
    # final clipping, then a tiny fix for the remaining error
    x = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-8:
        interior = (x > 1e-12) & (x < 1.0 - 1e-12)
        if interior.any():
            x[interior] += diff / interior.sum()
            x = np.clip(x, 0.0, 1.0)
    return x


# ------------------------------------------------------------
# Gradient of the autocorrelation J(s)=∑ h_i h_{i+s} for a given shift s≥0.
# ------------------------------------------------------------
def gradient_for_shift(h: np.ndarray, s: int) -> np.ndarray:
    """∇_h J(s) where J(s)=∑_i h_i h_{i+s} (zero padding outside)."""
    N = h.size
    if s == 0:
        return 2.0 * h
    g = np.zeros_like(h)
    N_minus_s = N - s
    g[:N_minus_s] += h[s:]          # contribution from h_{i+s}
    g[s:] += h[:N_minus_s]          # contribution from h_{i-s}
    return g


# ------------------------------------------------------------
# C5 value for a given step function h.
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """C5 = dx * max_{k} Σ h_i (1 - h_{i+k})."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


# ------------------------------------------------------------
# Gradient ascent that tries to increase the minimal autocorrelation.
# ------------------------------------------------------------
def gradient_ascent(h0: np.ndarray, max_iters: int, time_limit: float, rng: np.random.Generator) -> tuple:
    h = h0.copy()
    N = h.size
    target_sum = N / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)

    for it in range(1, max_iters + 1):
        if time.time() > time_limit:
            break

        # step size decays like 1/√it
        step = 0.5 / np.sqrt(it)

        conv = np.correlate(h, h, mode="full")
        min_corr = conv.min()
        # all shifts achieving the minimum (within a tiny tolerance)
        idxs = np.where(np.abs(conv - min_corr) < 1e-12)[0]

        grad = np.zeros_like(h)
        for idx in idxs:
            s = idx - (N - 1)          # signed shift
            grad += gradient_for_shift(h, abs(s))

        norm = np.linalg.norm(grad)
        if norm > 0:
            grad = grad / norm

        y = h + step * grad
        y = project_to_bounds_sum(y, target_sum)

        score = compute_c5(y)
        if score < best_score - 1e-12:
            best_score = score
            best_h = y.copy()

        h = y

        # occasional random perturbation to escape plateaus
        if it % 200 == 0 and rng.random() < 0.3:
            noisy = h + rng.normal(scale=0.02, size=N)
            noisy = project_to_bounds_sum(noisy, target_sum)
            noisy_score = compute_c5(noisy)
            if noisy_score < best_score:
                best_score = noisy_score
                best_h = noisy.copy()
                h = noisy

    return best_h, best_score


# ------------------------------------------------------------
# Convert a (possibly fractional) vector to a binary one with exactly N/2 ones.
# ------------------------------------------------------------
def binarize_and_fix(h: np.ndarray) -> np.ndarray:
    N = h.size
    idx = np.argsort(-h)                 # descending order
    bin_h = np.zeros(N, dtype=float)
    bin_h[idx[: N // 2]] = 1.0
    return bin_h


# ------------------------------------------------------------
# Simple hill‑climbing on binary vectors by swapping a 1 and a 0.
# ------------------------------------------------------------
def hill_climb_swap(h0: np.ndarray, time_limit: float) -> tuple:

    target = N // 2
    h = h0.copy()

    zeros = list(np.where(h < 0.5)[0])

    best_h = h.copy()
    best_score = compute_c5(best_h)
    rng = np.random.default_rng()
    stagnation = 0

    while time.time() < time_limit:
        if not ones or not zeros:

        h[j] = 1.0
        ones.remove(i)
        zeros.append(i)

        ones.append(j)

        new_score = compute_c5(h)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_h = h.copy()
            stagnation = 0
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
            zeros.append(i)
            ones.append(j)
            ones.remove(j)
            zeros.remove(i)
            stagnation += 1

        # after many failed attempts, restart from a fresh random binary vector
        if stagnation >= 5000:
            h[:] = 0.0
            h[:target] = 1.0
            rng.shuffle(h)
            ones = list(np.where(h == 1.0)[0])
            zeros = list(np.where(h == 0.0)[0])
            stagnation = 0

    return best_h, best_score


# ------------------------------------------------------------
# Main entry point required by the harness.
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # --------------------------------------------------------------------
    # Initialise h – use the provided construction if it exists.
    # --------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        N = h0.size
    else:
        N = 80
        h0 = np.zeros(N, dtype=np.float64)
        h0[: N // 2] = 1.0
        rng.shuffle(h0)

    target_sum = N / 2.0
    h0 = project_to_bounds_sum(h0, target_sum)

    best_h = h0.copy()
    best_score = compute_c5(best_h)

    # --------------------------------------------------------------------
    # Allocate time between gradient ascent and combinatorial hill‑climb.
    # --------------------------------------------------------------------
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    grad_time = start_time + remaining * 0.65
    hill_time = start_time + budget_s - 0.5   # keep a small safety margin

    # --------------------------------------------------------------------
    # Gradient ascent (continuous optimisation)
    # --------------------------------------------------------------------
    h_grad, score_grad = gradient_ascent(best_h, max_iters=20000,
                                         time_limit=grad_time, rng=rng)
    if score_grad < best_score:
        best_h, best_score = h_grad, score_grad

    # --------------------------------------------------------------------
    # Binary hill‑climb (swap based local search)
    # --------------------------------------------------------------------
    h_bin = binarize_and_fix(best_h)
    h_hill, score_hill = hill_climb_swap(h_bin, time_limit=hill_time)
    if score_hill < best_score:
        best_h, best_score = h_hill, score_hill

    # --------------------------------------------------------------------
    # Final projection (ensures exact constraints) and final C5 value.
    # --------------------------------------------------------------------
    best_h = project_to_bounds_sum(best_h, target_sum)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, N
