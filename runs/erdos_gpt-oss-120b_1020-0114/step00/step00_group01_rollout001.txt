import time
import numpy as np

def _compute_c5(h):
    """Return the max overlap integral for a discretised step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Initialise h
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float).copy()
        n = h.shape[0]
    else:
        n = 80                     # even length for easy binary start
        h = np.zeros(n, dtype=float)
        h[: n // 2] = 1.0
        rng.shuffle(h)

    target_sum = n / 2.0            # ∑h must equal n/2 (integral = 1)

    # Project onto the feasible affine set (clip then fix the sum)
    np.clip(h, 0.0, 1.0, out=h)
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # Simple redistribution of the deficit/excess
        diff = target_sum - cur_sum
        # add/subtract diff to a single entry that can absorb it
        for idx in range(n):
            lo, hi = -h[idx], 1.0 - h[idx]
            if diff >= lo and diff <= hi:
                h[idx] += diff
                break

    best_h = h.copy()
    best_score = _compute_c5(best_h)
    current_score = best_score

    # ------------------------------------------------------------------
    # Main optimisation loop (gradient‑guided simulated annealing)
    # ------------------------------------------------------------------
    T0 = 0.01                          # initial temperature
    no_improve = 0
    iter_cnt = 0

    while True:
        iter_cnt += 1
        now = time.time()
        if now - start >= budget_s * 0.98:
            break

        # Current worst shift and its gradient‑like vector g
        n = h.shape[0]
        conv = np.correlate(h, 1.0 - h, mode='full')
        idx_max = int(conv.argmax())
        shift = idx_max - (n - 1)          # integer lag producing the max

        # g[i] = derivative of the active term w.r.t. h_i
        g = np.ones(n, dtype=float)
        if shift >= 0:
            g[: n - shift] = 1.0 - h[shift:]
        else:
            g[-shift:] = 1.0 - h[: n + shift]

        i_max = int(np.argmax(g))
        i_min = int(np.argmin(g))

        # Pick a feasible transfer direction (decrease i_max, increase i_min)
        eps_max = min(h[i_max], 1.0 - h[i_min])
        if eps_max <= 0.0:
            # fallback: pick any two distinct indices with feasible transfer
            i_max = rng.integers(n)
            i_min = rng.integers(n - 1)
            if i_min >= i_max:
                i_min += 1
            eps_max = min(h[i_max], 1.0 - h[i_min])
            if eps_max <= 0.0:
                continue

        # Random step size (uniform fraction of the feasible amount)
        eps = eps_max * rng.random()

        h_new = h.copy()
        h_new[i_max] -= eps
        h_new[i_min] += eps

        new_score = _compute_c5(h_new)

        # Simulated‑annealing acceptance
        elapsed = now - start
        T = T0 * max(0.0, 1.0 - elapsed / budget_s)
        accept = False
        if new_score < current_score - 1e-12:
            accept = True
        elif T > 0.0 and rng.random() < np.exp(-(new_score - current_score) / T):
            accept = True

        if accept:

            current_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_h = h.copy()

                    lo, hi = -h[idx], 1.0 - h[idx]
                    add = np.clip(diff, lo, hi)
                    h[idx] += add
                    diff -= add
                    if abs(diff) < 1e-12:
                        break
            current_score = _compute_c5(h)
            no_improve = 0

    # ------------------------------------------------------------------
    # Final tidy‑up (exact sum enforcement) and result packaging
    # ------------------------------------------------------------------
    # Ensure the returned vector satisfies the exact linear constraint
    diff = target_sum - best_h.sum()

            add = np.clip(diff, lo, hi)
            best_h[idx] += add
            diff -= add
            if abs(diff) < 1e-12:
                break

    final_score = _compute_c5(best_h)

    return best_h.tolist(), float(final_score), n
