def project_onto_bounds_and_sum(h, target_sum):
    # Clip after shifting by lambda as described: find λ so that sum clip(h - λ, 0, 1) = target_sum.
    # Use binary search λ within bounds: low = min(h - 1) maybe -1, high = max(h)
    low = -1.0
    high = 1.0
    # We'll perform 50 iterations of bisection for high precision.
    for _ in range(50):
        mid = (low + high) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            # need larger λ to reduce sum (making projection smaller)
            low = mid
        else:
            high = mid
    proj = np.clip(h - (low + high) / 2.0, 0.0, 1.0)
    return proj

def sturmian_binary(n, alpha= (1+np.sqrt(5))/2, rho=0.5):
    # compute fractional part
    i = np.arange(n)
    frac = (i * alpha) % 1.0
    h = (frac < rho).astype(float)
    # ensure sum exactly n/2
    current_sum = h.sum()
    target = n/2
    diff = int(target - current_sum)
    if diff > 0:
        # add 1s at smallest fractional values > rho
        indices = np.argsort(frac)  # ascending
        # find indices where currently 0, and add 1 to diff of them
        # but we need to respect no 1's exceed 1 quickly
        zeros = indices[frac[indices] >= rho]  # those > rho
        h[zeros[:diff]] = 1.0
    elif diff < 0:
        # remove some 1's at largest fractional values < rho
        indices = np.argsort(-frac)  # descending
        ones = indices[frac[indices] < rho]  # those < rho
        h[ones[: -diff]] = 0.0
    # Now sum exactly target
    return h
```

# -*- coding: utf-8 -*-
"""
Optimisation for the Erdős minimum overlap problem (C5).
Returns a step‑function h on [0,2] (as a list of samples),
its upper bound C5 and the number of sample points.
"""
import time
import math
import numpy as np

# a global random generator – set inside run()
global_rng = None


def project_onto_bounds_and_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project the vector v onto the set {0 ≤ x ≤ 1, sum(x)=target_sum}
    using a simple bisection on the Lagrange multiplier λ for the
    constraints x_i = clip(v_i-λ,0,1).
    """
    low, high = -1.0, 1.0
    # expand bounds a little in case v is far from [0,1]
    lo = v.min() - 2.0
    hi = v.max() + 2.0
    low, high = lo, hi
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            low = mid
        else:
            high = mid
    return np.clip(v - (low + high) * 0.5, 0.0, 1.0)


def compute_gradient(h: np.ndarray, k: int) -> np.ndarray:
    """Gradient of C(k)=∑h_i(1‑h_{i+k}) w.r.t. h (subgradient for the max‑term)."""
    n = h.shape[0]
    if k == 0:
        return 1.0 - 2.0 * h
    # shift forward (i+k)
    plus = np.empty_like(h)
    plus[: n - k] = h[k:]
    plus[n - k :] = 0.0
    # shift backward (i‑k)
    minus = np.empty_like(h)
    minus[:k] = 0.0
    minus[k:] = h[: n - k]
    return 1.0 - (plus + minus)


def evaluate_h(h: np.ndarray) -> float:
    """Return C5 for a given h (dx already applied)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)


def sturmian_binary(n: int) -> np.ndarray:
    """
    Generate a balanced binary word of length n using the golden‑ratio
    Beatty sequence. The sum is forced to n//2 exactly.
    """
    phi = (1.0 + np.sqrt(5.0)) * 0.5
    i = np.arange(n, dtype=np.float64)
    frac = (i * phi) % 1.0
    h = (frac < 0.5).astype(np.float64)
    target = n // 2
    cur = int(h.sum())
    diff = target - cur
    if diff > 0:
        # turn some 0→1, choose those just above 0.5
        zero_idx = np.flatnonzero(h == 0)
        # their fractional parts are ≥0.5; pick the smallest ones
        order = np.argsort(frac[zero_idx])
        h[zero_idx[order[:diff]]] = 1.0
    elif diff < 0:
        # turn some 1→0, choose those just below 0.5
        one_idx = np.flatnonzero(h == 1)
        order = np.argsort(-frac[one_idx])  # largest fractional parts first
        h[one_idx[order[: -diff]]] = 0.0
    return h


def improve_h(start_h: np.ndarray, n: int, start_time: float,
              total_budget: float, max_iters: int = 8000) -> tuple:
    """
    Local optimisation starting from start_h.
    Returns (best_h, best_value).
    """
    target_sum = n / 2.0
    h = start_h.copy()
    best_h = h.copy()
    best_val = float("inf")
    for it in range(max_iters):
        # time guard
        if time.time() - start_time > total_budget * 0.98:
            break

        # current overlap and worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        cur_val = corr[idx] * (2.0 / n)

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_h = h.copy()

        # gradient of the active shift
        k = idx - (n - 1)
        grad = compute_gradient(h, abs(k))

        # step size (simple diminishing schedule)
        eta = 0.15 / math.sqrt(it + 1.0)

        # small random noise to escape plateaus
        noise_amp = 0.02 / (it + 1.0)
        noise = (global_rng.random(n) - 0.5) * noise_amp

        # gradient step + projection
        h = h - eta * grad + noise
        h = project_onto_bounds_and_sum(h, target_sum)

    return best_h, best_val

    """
    Optimise a step function h and return (h_values, c5_bound, n_points).
    """
    global global_rng
    global_rng = np.random.default_rng(seed)

    start_time = time.time()
    best_h = None
    best_val = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # candidate sample sizes
    # ------------------------------------------------------------------
    n_candidates = []

    # Use the provided construction if it exists
    try:
        init_h = np.array(initial_h_values, dtype=float)

    except NameError:
        init_h = None

    # Additional sizes (keep them modest)
    extra = [80, 100, 120, 140, 160, 180, 200, 240, 300]
    n_candidates.extend(extra)

    # ensure uniqueness while preserving order
    seen = set()
    n_list = []
    for v in n_candidates:
        if v not in seen:
            seen.add(v)
            n_list.append(v)

    # ------------------------------------------------------------------
    # main search loop
    # ------------------------------------------------------------------
    for n in n_list:
        if time.time() - start_time > budget_s * 0.96:
            break
        if n % 2 == 1:      # we need an even count for exact sum = n/2
            n += 1

        # starting points for this n
        starts = []

        # 1) the official initial construction if dimensions match
        if init_h is not None and len(init_h) == n:
            starts.append(init_h.copy())

        # 2) Sturmian balanced binary word
        starts.append(sturmian_binary(n))

        # 3) a random binary arrangement
        rnd = np.zeros(n, dtype=float)
        ones = global_rng.choice(n, n // 2, replace=False)

        starts.append(rnd)

        uniform = np.full(n, 0.5) + 0.02 * (global_rng.random(n) - 0.5)
        starts.append(uniform)

        for s in starts:
            if time.time() - start_time > budget_s * 0.96:
                break
            # ensure feasibility before optimisation
            s = project_onto_bounds_and_sum(s, n / 2.0)
            h_opt, val_opt = improve_h(s, n, start_time, budget_s)
            if val_opt < best_val:
                best_val = val_opt
                best_h = h_opt.copy()
                best_n = n

    if best_h is None:
        best_n = 100
        best_h = np.full(best_n, 0.5)
        best_val = evaluate_h(best_h)

    # Convert to plain Python list as required by the harness
    return (best_h.tolist(), float(best_val), int(best_n))
