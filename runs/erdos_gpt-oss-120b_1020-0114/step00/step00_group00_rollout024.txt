def compute_c5(h):
    n = len(h)
    y = 1 - h
    # compute linear convolution h * y_rev
    y_rev = y[::-1]
    N = 1 << ((2*n-1).bit_length())
    # zero pad
    fh = np.fft.rfft(h, N)
    fy = np.fft.rfft(y_rev, N)
    conv = np.fft.irfft(fh * fy, N)[:2*n-1]
    c5 = np.max(conv) * (2.0 / n)  # dx
    return c5

import time, math
import numpy as np
import scipy.optimize as opt
from scipy.interpolate import interp1d

def _compute_c5(h):
    """Return the max overlap integral for a step function h."""
    n = h.shape[0]
    conv = np.convolve(h, (1.0 - h)[::-1], mode="full")
    return np.max(conv) * (2.0 / n)


def _project_to_feasible(h, target_sum):
    """Project vector h onto [0,1]^n with exact sum = target_sum."""
    a = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = (lo + hi) / 2.0
        p = np.clip(a + mid, 0.0, 1.0)
        s = p.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(a + (lo + hi) / 2.0, 0.0, 1.0)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # base resolution – try to improve on the provided construction
    base_n = (
        len(initial_h_values) if "initial_h_values" in globals() else 45
    )
    # work with a finer grid for more flexibility
    n = max(80, base_n * 2)
    if n > 500:
        n = 500
    target_sum = n / 2.0

    # ----- helper to keep time ----------
    def time_left():
        return budget_s - (time.time() - start_time)

    # ----- build a few starting vectors -----
    candidates = []

    # 1. up‑sample the parent construction if it exists
    if "initial_h_values" in globals():
        parent = np.asarray(initial_h_values, dtype=float)
        old_n = parent.size
        x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        f = interp1d(x_old, parent, kind="linear", fill_value="extrapolate")
        h_up = f(x_new)
        h_up = np.clip(h_up, 0.0, 1.0)
        h_up = _project_to_feasible(h_up, target_sum)
        candidates.append(h_up)

    # 2. uniform 0.5 (feasible by construction)
    candidates.append(np.full(n, 0.5))

    # 3. few binary random vectors (exactly half ones)
    for _ in range(2):
        h_bin = np.zeros(n)
        idx = rng.choice(n, size=n // 2, replace=False)
        h_bin[idx] = 1.0
        candidates.append(h_bin)

    # 4. random continuous vectors projected onto the simplex
    for _ in range(4):
        a = rng.random(n)
        h_rand = _project_to_feasible(a, target_sum)
        candidates.append(h_rand)

    # keep the best among the initial guesses
    best_h = None
    best_val = np.inf
    for h in candidates:
        val = _compute_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()

    # ----- local refinement with SLSQP -----
    eq_cons = {"type": "eq", "fun": lambda x: float(x.sum() - target_sum)}
    bounds = [(0.0, 1.0) for _ in range(n)]

    for h0 in candidates:
        if time_left() < budget_s * 0.15:
            break
        # ensure the starting point satisfies the equality (SLSQP likes it)
        x0 = _project_to_feasible(h0, target_sum)

        res = opt.minimize(
            _compute_c5,
            x0,
            method="SLSQP",
            bounds=bounds,
            constraints=eq_cons,
            options={"maxiter": 200, "ftol": 1e-9, "disp": False},
        )
        if res.success:
            val = _compute_c5(res.x)
            if val < best_val:
                best_val = val
                best_h = res.x.copy()

    # parameters for the annealing schedule

    Tmin = 1e-5
    decay = 0.995
    cur_h = best_h.copy()
    cur_val = best_val

        # feasible delta so that both stay in [0,1] after transfer

        d_max = min(1.0 - hi, hj)       # hi + d <= 1 ;  hj - d >= 0
        if d_min > d_max:
            continue
        d = rng.random() * (d_max - d_min) + d_min

        new_h = cur_h.copy()
        new_h[i] = hi + d
        new_h[j] = hj - d

        new_val = _compute_c5(new_h)

            if new_val < best_val:

    best_h = _project_to_feasible(best_h, target_sum)
