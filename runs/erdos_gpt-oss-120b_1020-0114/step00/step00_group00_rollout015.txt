def propose_move(h, max_delta):
    n = len(h)
    # choose i where h[i] > 0
    i_candidates = np.where(h > 1e-12)[0]
    if len(i_candidates) == 0:
        return None
    i = np.random.choice(i_candidates)
    # choose j where h[j] < 1
    j_candidates = np.where(h < 1 - 1e-12)[0]
    if len(j_candidates) == 0:
        return None
    j = np.random.choice(j_candidates)
    delta = np.random.uniform(0, max_delta)
    delta = min(delta, h[i], 1 - h[j])
    if delta == 0:
        return None
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    # Clip small numeric errors
    h_new = np.clip(h_new, 0, 1)
    return h_new
```

def compute_c5_and_offset(h):
    n = h.size
    g_rev = (1 - h)[::-1]
    size = 1 << ((2*n-1).bit_length())
    Ha = np.fft.rfft(np.concatenate([h, np.zeros(size - n)]))
    Hg = np.fft.rfft(np.concatenate([g_rev, np.zeros(size - n)]))
    conv = np.fft.irfft(Ha * Hg, n=size)[:2*n-1]
    idx = np.argmax(conv)
    max_val = conv[idx] * (2.0 / n)
    offset = idx - (n - 1)
    return max_val, offset, conv
```

import numpy as np, time, math

def _c5_and_offset(h):
    """Return (max_overlap, offset) where max_overlap = max_c5 (integral value)
    and offset is the lag giving that maximum. Uses FFT for speed."""
    n = h.size
    g_rev = (1.0 - h)[::-1]
    size = 1 << ((2 * n - 1).bit_length())
    Ha = np.fft.rfft(np.concatenate([h, np.zeros(size - n)]))
    Hg = np.fft.rfft(np.concatenate([g_rev, np.zeros(size - n)]))
    conv = np.fft.irfft(Ha * Hg, n=size)[: 2 * n - 1]
    idx = int(np.argmax(conv))
    max_val = conv[idx] * (2.0 / n)
    offset = idx - (n - 1)
    return max_val, offset


def _project(v, target):
    """Project v onto {0<=x<=1, sum(x)=target} by water‑filling."""
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _gradient_step(h, offset, step):
    """One projected gradient step for the current worst offset."""
    n = h.size
    idx = np.arange(n)
    plus = idx + offset
    term1 = np.where((plus >= 0) & (plus < n), 1.0 - h[plus], 1.0)
    minus = idx - offset
    term2 = np.where((minus >= 0) & (minus < n), h[minus], 0.0)
    grad = term1 - term2
    v = h - step * grad
    return _project(v, n * 0.5)


def _swap_move(h, rng):
    """Swap a positive amount from a non‑zero entry to a non‑full entry."""
    nz = np.where(h > 1e-12)[0]
    nf = np.where(h < 1.0 - 1e-12)[0]
    if nz.size == 0 or nf.size == 0:
        return None
    i = rng.choice(nz)
    j = rng.choice(nf)
    delta = min(h[i], 1.0 - h[j])
    if delta <= 0:
        return None
    h2 = h.copy()
    h2[i] -= delta
    h2[j] += delta
    return h2


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # decide a reasonably sized even n (binary feasible)
    n = 90  # even, enough resolution but still cheap
    target_sum = n * 0.5

    # three simple initializations
    xs = np.linspace(0, 2, n, endpoint=False)
    cos_h = 0.5 + 0.5 * np.cos(np.pi * (xs - 1.0))
    cos_h = _project(cos_h, target_sum)

    block_h = np.zeros(n)
    block_h[: n // 2] = 1.0
    block_h = _project(block_h, target_sum)

    rand_h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    rand_h[ones] = 1.0

    inits = [cos_h, block_h, rand_h]

    best_h = None
    best_val = np.inf

    start = time.time()
    time_limit = budget_s * 0.94  # safety margin

    # main loop: iterate over initial seeds, allocate time fractionally
    for init in inits:
        cur_h = init.copy()
        cur_val, cur_off = _c5_and_offset(cur_h)
        if cur_val < best_val:
            best_h, best_val = cur_h.copy(), cur_val

        no_improve = 0
        while time.time() - start < time_limit:
            elapsed = time.time() - start
            frac = elapsed / time_limit
            # schedule
            step = 0.5 * (1.0 - frac) + 0.01
            T = 0.02 * (1.0 - frac)

            # choose move type
            if rng.random() < 0.65:
                # gradient step using current worst offset
                new_h = _gradient_step(cur_h, cur_off, step)
            else:
                # random swap move
                new_h = _swap_move(cur_h, rng)
                if new_h is None:
                    continue

            new_val, new_off = _c5_and_offset(new_h)

            # acceptance
            if new_val < cur_val - 1e-12:
                cur_h, cur_val, cur_off = new_h, new_val, new_off
                no_improve = 0
                if new_val < best_val:
                    best_h, best_val = new_h.copy(), new_val
            else:
                if T > 0 and rng.random() < math.exp(-(new_val - cur_val) / T):
                    cur_h, cur_val, cur_off = new_h, new_val, new_off
                no_improve += 1

            # occasional restart if stuck
            if no_improve > 8000:
                cur_h = _project(rng.random(n), target_sum)
                cur_val, cur_off = _c5_and_offset(cur_h)
                no_improve = 0

            # quit early if we already beat the target by a comfortable margin
            if best_val < 0.3806:
                break
        if best_val < 0.3806:
            break

    # final evaluation with numpy.correlate (exact to the harness)
    # ensure the returned array obeys the linear constraints
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-9:
        # fix tiny residual by adjusting a non‑boundary element
        idx = np.where((best_h > 0) & (best_h < 1))[0]
        if idx.size:
            best_h[idx[0]] += diff
        else:
            best_h[0] += diff
        best_h = np.clip(best_h, 0.0, 1.0)

    # final precise C5 using the same definition as the evaluator
    final_c5, _ = _c5_and_offset(best_h)
    return best_h, float(final_c5), n
