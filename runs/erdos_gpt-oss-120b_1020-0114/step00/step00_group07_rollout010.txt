# Helper: compute the overlap bound C5 for a step vector h
def compute_c5(h):
    """Return C5 = dx * max_{shift} np.correlate(h, 1-h)"""

# MILP model for a given number of points N (must be even)
def solve_milp(N, time_limit):
    """Return (h, C5) or (None, None) if not solved in time."""
    # binary vector x of length N, exactly N/2 ones
    x = cp.Variable(N, boolean=True)
    # matrix of ANDs: w_ij = x_i * x_j
    w = cp.Variable((N, N), boolean=True)

    constraints = []
    # exact half ones
    constraints.append(cp.sum(x) == N // 2)

    # link w to x (and enforce symmetry)
    for i in range(N):
        constraints.append(w[i, i] == x[i])
    for i in range(N):
        for j in range(i + 1, N):
            constraints.append(w[i, j] <= x[i])
            constraints.append(w[i, j] <= x[j])
            constraints.append(w[i, j] >= x[i] + x[j] - 1)
            constraints.append(w[j, i] == w[i, j])

    # auxiliary integer t = maximal overlap count
    t = cp.Variable(integer=True, nonneg=True)

    # shift constraints (full correlation, zero padding)
    for k in range(-(N - 1), N):
        if k == 0:
            continue
        if k > 0:
            i_idx = np.arange(N - k)
            j_idx = i_idx + k
        else:
            kp = -k
            i_idx = np.arange(kp, N)
            j_idx = i_idx - kp
        # sum of x over the overlapping region
        sum_x = cp.sum(x[i_idx])
        # sum of w over the same pairs
        sum_w = cp.sum(w[i_idx, j_idx])
        constraints.append(t >= sum_x - sum_w)

    # objective
    prob = cp.Problem(cp.Minimize(t), constraints)

    # solve with a time limit (CBC accepts 'maximumSeconds')
    try:
        prob.solve(solver=cp.CBC, maximumSeconds=time_limit, log=False)
    except Exception:
        # fallback to GLPK_MI if CBC unavailable
        prob.solve(solver=cp.GLPK_MI, tm_lim=int(time_limit * 1000), msg=False)

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None, None

    # Extract solution
    xv = np.rint(x.value).astype(int)   # ensure 0/1
    # sanity check
    if xv.sum() != N // 2:
        return None, None
    c5 = compute_c5(xv.astype(float))
    return xv.astype(float), c5

# Simple local search (swap two positions) – used as fallback
def local_search(start_h, budget_seconds, target_c5):
    """Hill‑climbing with random swaps until time runs out."""
    N = start_h.shape[0]
    best_h = start_h.copy()
    best_c5 = compute_c5(best_h)
    if best_c5 <= target_c5:
        return best_h, best_c5

    # pre‑compute index lists of ones and zeros
    ones = np.where(best_h > 0.5)[0].tolist()
    zeros = np.where(best_h < 0.5)[0].tolist()
    if not ones or not zeros:
        return best_h, best_c5

    t0 = time.time()
    while time.time() - t0 < budget_seconds:
        i = random.choice(ones)
        j = random.choice(zeros)
        # propose swap
        cand = best_h.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        cand_c5 = compute_c5(cand)
        if cand_c5 < best_c5:
            best_h = cand
            best_c5 = cand_c5
            # update lists
            ones.remove(i)
            zeros.append(i)
            zeros.remove(j)
            ones.append(j)
            if best_c5 <= target_c5:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) with an improved C5."""
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    TARGET = 0.3808

    # ------------------------------------------------------------------
    # Determine an initial candidate (use provided one if present)
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.shape[0]
        # enforce exact integral constraint if rounding required
        dx0 = 2.0 / n0
        # project onto feasible binary space preserving sum
        ones_needed = n0 // 2
        idx = np.argsort(-h0)  # descending
        h_bin = np.zeros_like(h0)
        h_bin[idx[:ones_needed]] = 1.0
        cur_best_h = h_bin
        cur_best_c5 = compute_c5(cur_best_h)
        cur_n = n0
    else:
        cur_best_h = None
        cur_best_c5 = float('inf')
        cur_n = None

    # ------------------------------------------------------------------
    # Try MILP for a few even sizes; start with 100 points
    candidate_sizes = [100, 120, 140, 160]
    for N in candidate_sizes:
        # keep enough budget for later steps
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed - 10.0   # reserve a little margin
        if remaining <= 5.0:
            break
        # give each solve a modest time slice (max 150 s)
        tl = min(150.0, remaining / 2.0)

        h_sol, c5_sol = solve_milp(N, tl)
        if h_sol is None:
            continue
        # keep the best found so far
        if c5_sol < cur_best_c5:
            cur_best_h = h_sol
            cur_best_c5 = c5_sol
            cur_n = N
        if c5_sol <= TARGET:
            # reached the target – stop early
            break

    # ------------------------------------------------------------------
    # If MILP did not meet target, refine with local search (remaining time)
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 5.0 and cur_best_h is not None and cur_best_c5 > TARGET:
        # start from current best (or random if none)
        if cur_best_h is None:
            N = 100
            h0 = np.zeros(N)
            h0[:N // 2] = 1.0
            np.random.shuffle(h0)
            cur_best_h = h0
            cur_n = N
        # allocate most of the remaining time to search
        cur_best_h, cur_best_c5 = local_search(
            cur_best_h, remaining - 2.0, TARGET
        )

    # ------------------------------------------------------------------
    # Final sanity checks and return
    if cur_best_h is None:
        # fallback: simple block construction (should never happen)
        N = 80
        h = np.zeros(N)
        h[:N // 2] = 1.0
        np.random.shuffle(h)
        cur_best_h = h.astype(float)
        cur_n = N
        cur_best_c5 = compute_c5(cur_best_h)

    # Ensure exact integral constraint (tiny numerical drift)
    dx = 2.0 / cur_n
    total = cur_best_h.sum() * dx
    if abs(total - 1.0) > 1e-9:
        # force correction by fixing the number of ones
        needed = cur_n // 2
        idx = np.argsort(-cur_best_h)
        cur_best_h[:] = 0.0
        cur_best_h[idx[:needed]] = 1.0

    final_c5 = compute_c5(cur_best_h)
    return cur_best_h, final_c5, cur_n
