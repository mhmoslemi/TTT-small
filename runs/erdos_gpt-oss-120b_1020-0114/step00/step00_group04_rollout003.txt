def project_to_sum_box(v, target, tol=1e-9, max_iter=100):
    low = np.min(v) - 1  # low bound for λ, maybe v_i - 1 ensures clamp at upper bound.
    high = np.max(v)  # high bound.
    for _ in range(max_iter):
        mid = (low + high)/2
        h = np.clip(v - mid, 0, 1)
        s = h.sum()
        if abs(s - target) < tol:
            return h
        if s > target:
            low = mid
        else:
            high = mid
    return np.clip(v - (low+high)/2, 0, 1)

import numpy as np
import time, math, random

def _project_to_sum_box(v: np.ndarray, target: float, tol: float = 1e-9, max_iter: int = 100) -> np.ndarray:
    """Project v onto {h | 0≤h_i≤1 , sum h = target}."""
    low = np.min(v) - 1.0   # enough to force all entries to 1 after subtracting low
    high = np.max(v)        # enough to force all entries to 0 after subtracting high
    for _ in range(max_iter):
        mid = (low + high) * 0.5
        h = np.clip(v - mid, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) <= tol:
            return h
        if s > target:
            low = mid
        else:
            high = mid
    # final refinement
    h = np.clip(v - (low + high) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        free = (h > 0.0) & (h < 1.0)
        if free.any():
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def _c5_value(arr: np.ndarray, dx: float) -> float:
    """Compute the C5 bound for a discrete step function."""
    corr = np.correlate(arr, 1.0 - arr, mode="full")
    return np.max(corr) * dx


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for the Erdős C5 problem."""
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # --------------------------------------------------------------------
    # Problem size – 200 points gives a good trade‑off between resolution
    # and evaluation speed.
    # --------------------------------------------------------------------
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # because ∫h = 1

    # -------------------------- initial guess ---------------------------
    if "initial_h_values" in globals():
        # Upsample the parent construction to the chosen resolution.
        base = np.asarray(initial_h_values, dtype=float)
        n0 = base.shape[0]
        xs0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        xs = np.linspace(0.0, 2.0, n_points, endpoint=False)
        v = np.interp(xs, xs0, base)
        h = _project_to_sum_box(v, target_sum)
    else:
        # Pure random start.
        v = np.random.rand(n_points)
        h = _project_to_sum_box(v, target_sum)

    # --------------------------------------------------------------------
    # Simulated‑annealing style local search.
    # --------------------------------------------------------------------
    cur_c5 = _c5_value(h, dx)
    best_h = h.copy()
    best_c5 = cur_c5

    T_start = 0.02       # initial temperature
    T_end = 1e-6         # final temperature
    stagnation_limit = 60000   # steps without improvement before a perturbation
    last_improve = 0
    it = 0

    while True:
        # time budget check
        if time.time() - start_time > budget_s - 2.0:
            break

        # linear temperature schedule
        elapsed = time.time() - start_time
        frac = elapsed / (budget_s - 2.0)
        T = T_start * (1 - frac) + T_end * frac

        # pick two different indices
        i = np.random.randint(n_points)
        j = np.random.randint(n_points - 1)
        if j >= i:
            j += 1

        # feasible interval for the mass transfer delta (preserves the sum)
        lo = max(-h[i], h[j] - 1.0)
        hi = min(1.0 - h[i], h[j])
        if lo >= hi:
            continue

        delta = np.random.uniform(lo, hi)
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta

        new_c5 = _c5_value(h_new, dx)

        # Metropolis acceptance
        if new_c5 < cur_c5 or np.random.rand() < math.exp(-(new_c5 - cur_c5) / (T + 1e-12)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = h.copy()
                last_improve = it

        it += 1

        # occasional random “shake’’ if we have been stuck for a while
        if it - last_improve > stagnation_limit:
            noise = np.random.normal(scale=0.02, size=n_points)
            v = best_h + noise
            v = np.clip(v, 0.0, 1.0)
            h = _project_to_sum_box(v, target_sum)
            cur_c5 = _c5_value(h, dx)
            last_improve = it

    best_c5 = _c5_value(best_h, dx)
    return best_h, best_c5, n_points
