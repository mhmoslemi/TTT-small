            # term2 for i in 0..n-s_abs-1: - arr[i + s_abs]
            grad[:n - s_abs] -= arr[s_abs:]
    # average over shifts:
    if len(shifts) > 1:
        grad /= len(shifts)
    return grad

import time
import numpy as np

# ---------- helper functions (top level) ----------
def project_to_box_sum(v, total, max_iter=30):
    """Project vector v onto the set {0<=x<=1, sum(x)=total}."""
    lo, hi = -2.0, 2.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s < total:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)


def evaluate_c5(arr, dx):
    """Return the max overlap integral for the step function represented by arr."""
    return np.max(np.correlate(arr, 1.0 - arr, mode="full") * dx)


def compute_gradient(arr, shifts):
    """
    Gradient of the maximal overlap value with respect to arr,
    averaged over the given list of integer shifts.
    """
    n = arr.shape[0]
    grad = np.zeros_like(arr)
    for s in shifts:
        if s >= 0:
            # contribution of (1 - h[i+s])
            grad[: n - s] += 1.0 - arr[s:]
            # contribution of -h[i-s]
            grad[s:] -= arr[: n - s]
        else:
            s_abs = -s
            # (1 - h[i-s]) with s negative -> (1 - h[i+s_abs])
            grad[s_abs:] += 1.0 - arr[: n - s_abs]
            # -h[i-s] -> -h[i+s_abs]
            grad[: n - s_abs] -= arr[s_abs:]
    if len(shifts) > 1:
        grad /= len(shifts)
    return grad


# ---------- main entry point ----------

    total_sum = n / 2.0
    dx = 2.0 / n

    # make sure we start from a feasible point
    if init_h is not None:

    else:

    best_h = h_start.copy()
    best_val = evaluate_c5(best_h, dx)

    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # tiny safety margin

    # ---- main optimisation loop ----
    while time.time() < deadline:
        # pick a start: occasionally reuse the current best
        if rng.random() < 0.25:
            h = best_h.copy()
        else:
            h = project_to_box_sum(rng.random(n), total_sum)

        alpha = 0.5  # initial step size

        # inner gradient‑descent / local search
        for _ in range(20000):
