def max_corr(h):
    n = h.shape[0]
    a = h
    b = 1 - h
    b_rev = b[::-1]
    N = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b_rev, N)
    C = A * B
    corr = np.fft.irfft(C, N)[:2*n-1]  # trim to full length
    return np.max(corr) * (2.0 / n)
```

def propose_move(h, step):
    n = len(h)
    i, j = np.random.randint(0, n, size=2)
    while j == i:
        j = np.random.randint(0, n)
    # Determine allowed delta range
    # delta >0 means increase i, decrease j
    delta_max = min(step, 1 - h[i], h[j])
    delta_min = -min(step, h[i], 1 - h[j])
    delta = np.random.uniform(delta_min, delta_max)
    # apply
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    # sum unchanged and bounds satisfied
    return new_h, delta
```

import time, random, math
import numpy as np

def project_box_simplex(x, target, eps=1e-12):
    """Project x onto {0<=y_i<=1, sum y_i = target} using a binary search on lambda."""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)

def initialize_h(n):
    """Create a feasible initial h of length n."""
    target = n / 2.0
    if 'initial_h_values' in globals():
        init = initial_h_values
        m = init.shape[0]
        xs0 = np.linspace(0.0, 2.0, m, endpoint=False)
        xs = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(xs, xs0, init)
    else:
        h = np.random.rand(n)
    h = project_box_simplex(h, target)
    return h

def compute_corr(h):
    """Full correlation of h with (1-h) (unscaled)."""
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())   # next power of two
    A = np.fft.rfft(h, N)
    B = np.fft.rfft((1.0 - h)[::-1], N)
    C = A * B
    corr = np.fft.irfft(C, N)[:2 * n - 1]
    return corr

def max_corr_value(h):
    """Maximum overlap integral C5 for a given h."""
    n = h.shape[0]
    dx = 2.0 / n
    return compute_corr(h).max() * dx

def gradient_for_shift(h, shift):
    """Subgradient of the active overlap term w.r.t. h."""
    n = h.shape[0]
    g = np.ones_like(h)          # derivative of Σ h_i
    if shift >= 0:
        s = shift
        # subtract h_{i+shift}
        g[:n - s] -= h[s:]
        # subtract h_{i-shift}
        g[s:] -= h[:n - s]
    else:
        s = -shift
        # subtract h_{i+shift} = h_{i-s}
        g[s:] -= h[:n - s]
        # subtract h_{i-shift} = h_{i+s}
        g[:n - s] -= h[s:]
    return g

def propose_move(h, step):
    """Two‑coordinate move preserving the sum."""
    n = h.shape[0]
    i = np.random.randint(0, n)
    j = np.random.randint(0, n - 1)
    if j >= i:
        j += 1
    # delta >0 : increase i, decrease j
    delta_max = min(step, 1.0 - h[i], h[j])
    delta_min = -min(step, h[i], 1.0 - h[j])
    delta = np.random.uniform(delta_min, delta_max)
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    return new_h
