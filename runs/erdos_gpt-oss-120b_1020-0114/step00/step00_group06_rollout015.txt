def project_simple(y, S):

    lo, hi = -1.0, 1.0

            lo = lam
        else:
            hi = lam

def project_to_sum_and_bounds(v, target_sum):
    # ensure 0 - 1 bounds
    lo, hi = -1.0, 1.0
    for _ in range(40):
        lam = (lo + hi) / 2.0
        w = np.clip(v + lam, 0.0, 1.0)
        s = w.sum()
        if s < target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo+hi)/2.0, 0.0, 1.0)

def compute_maxC(h):
    N = len(h)
    c = np.correlate(h, 1 - h, mode='full')
    return np.max(c) * (2.0 / N)

def project_to_sum_and_bounds(v, target):
    lo, hi = -1.0, 1.0
    for _ in range(30):
        lam = (lo + hi) / 2.0
        w = np.clip(v + lam, 0.0, 1.0)
        s = w.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    return w

def shape_for_p(p, N):
    a = (p + 1.0) / (2.0 * p)
    if a > 1.0: a = 1.0
    dx = 2.0 / N
    x = (np.arange(N) + 0.5) * dx
    h = np.empty(N, dtype=float)

    target = N / 2.0
    total = h.sum()
    if abs(total - target) > 1e-9:
        h = project_to_sum_and_bounds(h, target)
    return h

def shape_linear_plateau(a, N):
    # a = ramp length on each side.
    # plateau height H
    if a <= 0:
        H = 0.5
        return np.full(N, H)
    H = 1.0 / (2.0 - a)
    dx = 2.0 / N
    x = (np.arange(N) + 0.5) * dx
    h = np.empty(N)
    left = x < a
    right = x > 2 - a
    middle = ~ (left | right)
    # left ramp linear: from 0 to H at x=a
    h[left] = (H / a) * x[left]   # x <= a
    # right ramp: from H at x=2-a down to 0
    h[right] = (H / a) * (2 - x[right])
    h[middle] = H
    # clip just in case
    h = np.clip(h, 0.0, 1.0)
    # ensure sum
    target = N / 2.0
    total = h.sum()
    if abs(total - target) > 1e-9:
        h = project_to_sum_and_bounds(h, target)
    return h

import time
import numpy as np

def compute_maxC(h):
    """Return max_{k} ∫ h(x) (1 - h(x+k)) dx approximated by correlation."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)   # dx = 2/n

def project_to_sum_and_bounds(v, target_sum, tol=1e-12, max_iter=50):
    """Project vector v onto [0,1]^n with prescribed sum using a simple bisection."""
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        w = np.clip(v + lam, 0.0, 1.0)
        s = w.sum()
        if s < target_sum:
            lo = lam
        else:
            hi = lam
        if hi - lo < tol:
            break
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def shape_linear_plateau(a, n):
    """Piecewise linear function with left/right ramps of length a and a central plateau.
    The plateau height H = 1/(2-a) guarantees ∫ h = 1.
    a ∈ [0,1]."""
    if a <= 1e-12:
        # constant 1/2
        h = np.full(n, 0.5, dtype=float)
        return h
    H = 1.0 / (2.0 - a)          # plateau height, 0.5 ≤ H ≤ 1
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    h = np.empty_like(x)
    left = x < a
    right = x > 2.0 - a
    middle = ~(left | right)
    # left ramp: from 0 to H over [0,a]
    h[left] = (H / a) * x[left]
    # right ramp: symmetric
    h[right] = (H / a) * (2.0 - x[right])
    # plateau
    h[middle] = H
    # enforce bounds and exact integral
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    if abs(h.sum() - target) > 1e-9:
        h = project_to_sum_and_bounds(h, target)
    return h

def shape_power_plateau(p, n):
    """Ramp with power p (p≥1) and a central plateau at height 1.
    The ramp length a = (p+1)/(2p) ensures ∫ h = 1."""
    if p <= 1.0:
        # fallback to linear case (a=1)
        return shape_linear_plateau(1.0, n)
    a = (p + 1.0) / (2.0 * p)   # 0.5 ≤ a ≤ 1
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    h = np.empty_like(x)
    left = x < a
    right = x > 2.0 - a
    middle = ~(left | right)
    # left ramp: x^p scaled to reach 1 at x=a
    h[left] = (x[left] / a) ** p
    # right ramp: symmetric
    h[right] = ((2.0 - x[right]) / a) ** p
    # plateau at height 1
    h[middle] = 1.0
    # bounds (should already be ≤1) and exact integral
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    if abs(h.sum() - target) > 1e-9:
        h = project_to_sum_and_bounds(h, target)
    return h
