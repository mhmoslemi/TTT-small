import time
import numpy as np

def _compute_c5(vec, dx):
    """Return max_k ∫ h(x)(1-h(x+k))dx approximated by discrete correlation."""
    corr = np.correlate(vec, 1 - vec, mode="full")
    return np.max(corr) * dx


def _project_to_sum(vec, target):
    """Simple projection onto the hyperplane sum(vec)=target while keeping 0≤vec≤1."""
    # Clip first
    vec = np.clip(vec, 0.0, 1.0)
    diff = target - vec.sum()
    if abs(diff) < 1e-12:
        return vec
    # Distribute the remaining mass over free coordinates proportionally to available room
    free_low = vec > 1e-12
    free_high = vec < 1.0 - 1e-12
    free = free_low | free_high
    if not np.any(free):
        return vec
    # capacity to increase or decrease
    inc_cap = np.sum(1.0 - vec[free_high])
    dec_cap = np.sum(vec[free_low])
    if diff > 0 and inc_cap > 0:          # need to add mass
        inc = diff * (1.0 - vec[free_high]) / inc_cap
        vec[free_high] += inc
    elif diff < 0 and dec_cap > 0:        # need to remove mass
        dec = (-diff) * vec[free_low] / dec_cap
        vec[free_low] -= dec
    # final clip just in case numerical noise happened
    return np.clip(vec, 0.0, 1.0)
