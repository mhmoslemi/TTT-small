def upscale(h, n_new):

    xs_old = np.linspace(0, 2, n_old, endpoint=False) # positions of sample centers (maybe).

    h_new = np.interp(xs_new, xs_old, h)

                hi = lam
            else:

def adjust_exact_sum(v, target):
    s = v.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return v
    # find index not at bound
    for idx in range(v.size):
        if diff > 0 and v[idx] < 1.0 - 1e-12:
            inc = min(diff, 1.0 - v[idx])
            v[idx] += inc
            diff -= inc
        elif diff < 0 and v[idx] > 1e-12:
            dec = min(-diff, v[idx])
            v[idx] -= dec
            diff += dec
        if abs(diff) < 1e-12:
            break
    return v

def project_onto_box_simplex(v, target):
    # Find lam via bisection
    lo = -np.max(v) - 1.0  # low enough
    hi = 1.0 - np.min(v) + 1.0  # high enough
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

    n = h.size
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(conv.max() * dx)


def _project_box_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector v onto the intersection of the box [0,1]^n
    and the hyperplane sum(x) = target using a simple bisection.
    """
    lo = -np.max(v) - 1.0      # low enough to clamp everything to 0
    hi = 1.0 - np.min(v) + 1.0  # high enough to clamp everything to 1
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    x = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction for rounding errors
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # spread the tiny difference onto an unfixed coordinate
        free = np.where((x > 1e-14) & (x < 1.0 - 1e-14))[0]
        if free.size == 0:
            # all entries are at a bound – pick any index
            idx = np.argmax(x) if diff > 0 else np.argmin(x)
        else:
            idx = free[0]
        x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
    return x


def enforce_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and enforce exact sum = target."""
    return _project_box_simplex(v, target)


def upscale_solution(base_h: np.ndarray, n_new: int) -> np.ndarray:
    """Linear‑interpolate a coarse solution to a finer grid and project."""
    n_old = base_h.size
    xs_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(xs_new, xs_old, base_h, left=0.0, right=0.0)
    h_new = enforce_sum(h_new, n_new / 2.0)
    return h_new


def subgradient_optimize(
    h0: np.ndarray, max_iter: int, deadline: float
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the max‑overlap objective.
    Returns the best vector encountered and its C5 value.
    """
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)

    step0 = 0.2  # initial step magnitude

    for t in range(max_iter):
        if time.time() >= deadline:
            break

        # find the shift that attains the current maximum
        conv = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(conv.argmax())
        shift = max_idx - (n - 1)

        # gradient of the active overlap term
        grad = np.zeros_like(h)

        if shift >= 0:
            s = shift
            end = n - s
            if end > 0:
                # contribution from the linear term (the “1”)
                grad[:end] = 1.0 - h[s:]
                # contribution from the bilinear term
                grad[s:] -= h[:end]
        else:
            s = -shift
            start = s
            length = n - s
            if length > 0:
                grad[start:] = 1.0 - h[:length]
                grad[:length] -= h[start:]

        # step size decays like 1/sqrt(t)
        gamma = step0 / math.sqrt(t + 1.0)
        h = h - gamma * grad
        h = enforce_sum(h, target)

        cur_score = compute_c5(h)
        if cur_score < best_score:
            best_score = cur_score
            best_h = h.copy()
            # early exit if already under the target bound
            if best_score < 0.3808:
                break

    return best_h, best_score


def hill_climb_optimize(
    h0: np.ndarray, max_iter: int, deadline: float
) -> tuple[np.ndarray, float]:
    """
    Simple pairwise‑swap hill climbing (with tiny uphill moves allowed).
    Returns the best vector found.
    """
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)

    # temperature schedule for occasional uphill moves
    temp0 = 0.05

    for t in range(max_iter):
        if time.time() >= deadline:
            break

        i = random.randrange(n)
        j = random.randrange(n - 1)
        if j >= i:
            j += 1

        # propose a bounded change that preserves the total sum
        max_delta = 0.3 * (1.0 - t / max_iter) + 0.01
        delta = (random.random() * 2.0 - 1.0) * max_delta
        new_i = h[i] + delta
        new_j = h[j] - delta
        if not (0.0 <= new_i <= 1.0 and 0.0 <= new_j <= 1.0):
            continue

        h_new = h.copy()
        h_new[i] = new_i
        h_new[j] = new_j

        new_score = compute_c5(h_new)

        # accept if better or with small probability
        cur_score = compute_c5(h)
        if new_score < cur_score:
            accept = True
        else:
            temp = temp0 * (1.0 - t / max_iter)
            accept = random.random() < math.exp(-(new_score - cur_score) / (temp + 1e-12))

        if accept:
            h = h_new

        if new_score < best_score:
            best_score = new_score
            best_h = h_new.copy()
            if best_score < 0.3808:
                break

    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step‑function h, its C5 bound and the number of samples."""
    start_time = time.time()
    random.seed(seed)
    np.random.seed(seed)

    # Obtain a base construction if supplied by the harness
    base_h = None
    if "initial_h_values" in globals():
        base_h = np.asarray(initial_h_values, dtype=float)

    # Choose a set of grid sizes to explore (keep them modest)
    n_candidates = []
    if base_h is not None:
        base_n = base_h.size
        # try a few refinements of the provided construction
        for factor in (2, 4):
            n_candidates.append(min(720, max(120, base_n * factor)))
    else:
        n_candidates = [180, 360]

    # Ensure uniqueness and a reasonable ordering (coarse → fine)
    n_candidates = sorted(set(n_candidates))

    best_overall_h = None
    best_overall_score = float("inf")
    best_overall_n = None

    for n_target in n_candidates:
        # Give each phase roughly half of the remaining budget
        time_now = time.time()
        time_left = budget_s - (time_now - start_time)
        if time_left < 5.0:
            break
        phase_deadline = time_now + 0.48 * time_left

        # Initialise a feasible h on the current grid
        if base_h is not None:
            h0 = upscale_solution(base_h, n_target)
        else:
            # simple half‑ones random start
            h0 = np.zeros(n_target, dtype=float)
            h0[: n_target // 2] = 1.0
            np.random.shuffle(h0)

        # 1) Sub‑gradient descent (fast convergence on the max term)
        max_iter_sg = 20000
        h_sg, score_sg = subgradient_optimize(h0, max_iter_sg, phase_deadline)

        # 2) If time permits, run a short hill‑climb from the result
        time_now = time.time()
        if time_now < phase_deadline:
            max_iter_hc = 30000
            h_hc, score_hc = hill_climb_optimize(
                h_sg, max_iter_hc, phase_deadline
            )
        else:
            h_hc, score_hc = h_sg, score_sg

        # Keep the best solution seen so far
        if score_hc < best_overall_score:
            best_overall_score = score_hc
            best_overall_h = h_hc.copy()
            best_overall_n = n_target

        # Stop early if we already beat the target record
        if best_overall_score < 0.3808:
            break

    # Fallback – should never happen but guard against no candidate being tried
    if best_overall_h is None:
        n_target = 200
        h_fallback = np.zeros(n_target, dtype=float)
        h_fallback[: n_target // 2] = 1.0
        np.random.shuffle(h_fallback)
        best_overall_h = enforce_sum(h_fallback, n_target / 2.0)
        best_overall_score = compute_c5(best_overall_h)
        best_overall_n = n_target

    # Final sanity projection (ensures exact integral)
    best_overall_h = enforce_sum(best_overall_h, best_overall_n / 2.0)
    best_overall_score = compute_c5(best_overall_h)

    return best_overall_h, float(best_overall_score), int(best_overall_n)
