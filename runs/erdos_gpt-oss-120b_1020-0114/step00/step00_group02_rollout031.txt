import numpy as np, time

def construct_h(alpha, q, N):
   dx = 2.0 / N
   # compute lengths
   r = 1 - q * alpha
   if r < 0:
        return None
   p = (2 - r - 2*q) / 2.0
   if p < 0:
        return None
   # create array
   xs = np.linspace(dx/2, 2.0 - dx/2, N)
   h = np.empty(N)
   # define thresholds
   # intervals: [0, p):0 ; [p, p+q):alpha ; [p+q, p+q+r):1 ; [p+q+r, p+q+r+q):alpha ; rest 0
   # but avoid floating errors using masks.
   left0_end = p
   leftAlpha_end = p + q
   center_end = leftAlpha_end + r
   rightAlpha_end = center_end + q
   # region definitions
   # xs are between 0 and 2
   h = np.where((xs >= left0_end) & (xs < leftAlpha_end), alpha,
                np.where((xs >= leftAlpha_end) & (xs < center_end), 1.0,
                         np.where((xs >= center_end) & (xs < rightAlpha_end), alpha, 0.0)))
   # compute sum
   total = h.sum()
   # scale factor
   target = N/2.0   # due to sum(h) = N/2 required (since average 0.5)
   factor = target / total
   h = np.clip(h * factor, 0, 1)
   # after clipping, adjust again maybe. but close enough.
   return h

import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def max_overlap(h: np.ndarray) -> float:
    """Return C5 for a step function h given as a 1‑D array."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1‑h); mode='full' already pads with zeros
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx


def build_five_segment(N: int, a: float, b: float, v_mid: float):
    """
    Build a symmetric 5‑segment step function on [0,2] with N samples.
    Parameters
    ----------
    N      : number of samples (integer > 0)
    a      : length of each outer segment (value = 1.0)
    b      : length of each inner segment (value = v_mid)
    v_mid  : value taken on the two inner segments (0 ≤ v_mid ≤ 1)
    Returns
    -------
    h (np.ndarray) of length N  (or None if the parameters are infeasible)
    """
    # total length constraints
    c = 2.0 - 2.0 * a - 2.0 * b          # central length
    if c <= 0.0:
        return None

    # outer value is forced to 1.0 (helps extreme lags)
    v0 = 1.0

    # solve for the central value v2 from the integral constraint:
    # 2*a*v0 + 2*b*v_mid + c*v2 = 1
    v2 = (1.0 - 2.0 * a * v0 - 2.0 * b * v_mid) / c
    if not (0.0 <= v2 <= 1.0):
        return None

    dx = 2.0 / N
    x = (np.arange(N) + 0.5) * dx  # mid‑point sampling

    # thresholds (all in the original [0,2] interval)
    t0 = a
    t1 = a + b
    t2 = a + b + c / 2.0

    h = np.full(N, v0, dtype=np.float64)

    # left inner band
    mask = (x >= t0) & (x < t1)
    h[mask] = v_mid
    # left central band
    mask = (x >= t1) & (x < t2)
    h[mask] = v2
    # right central band (mirror of left central)
    mask = (x >= 2.0 - t2) & (x < 2.0 - t1)
    h[mask] = v2
    # right inner band
    mask = (x >= 2.0 - t1) & (x < 2.0 - t0)
    h[mask] = v_mid
    # outer part already set to v0

    # project onto the linear constraint Σh = N/2  (i.e. ∫h = 1)
    target = N / 2.0
    cur = h.sum()
    if cur <= 0.0:
        return None

    factor = target / cur
    h *= factor
    # clipping may slightly break the sum; a tiny correction afterwards
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        # distribute the remaining mass uniformly, respecting the box constraints
        # this is a cheap fix; the residual will be tiny.
        incr = diff / N
        h += incr
        np.clip(h, 0.0, 1.0, out=h)

    return h


def random_five_segment(N: int, rng: np.random.Generator):
    """Generate a random feasible 5‑segment function."""
    # draw outer length a uniformly in (0,1)  (a cannot exceed 1)
    a = rng.random() * 0.9 + 0.05          # keep a away from the extremes
    # inner length b must satisfy 2a+2b ≤ 2  →  b ≤ 1‑a
    max_b = 1.0 - a
    b = rng.random() * max_b
    # inner value v_mid uniformly
    v_mid = rng.random()
    return build_five_segment(N, a, b, v_mid)

            if c5_new < best_c5:

                best_h = h_new

            else:

                if rng.random() < 0.05:
                    step_a = min(step_a * 1.05, 0.05)

    # --------------------------------------------------------

    # --------------------------------------------------------
    if best_h is None:
