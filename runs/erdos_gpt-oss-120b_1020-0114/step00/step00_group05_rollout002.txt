def _shift(arr: np.ndarray, k: int) -> np.ndarray:
    """Shift array by k positions with zero padding.
    Positive k → shift right (insert zeros on the left)."""
    n = arr.shape[0]
    if k == 0:
        return arr
    if k > 0:
        return np.concatenate((np.zeros(k, dtype=arr.dtype), arr[: n - k]))
    # k < 0
    k = -k
    return np.concatenate((arr[k:], np.zeros(k, dtype=arr.dtype)))

                    overlay = total_ones - corr
                    cur_max = int(overlay.max())
                    no_improve = 0

        if best_val <= target:
            break

    # ------------------------------------------------------------------
    # 5️⃣  Final sanity & exact evaluation --------------------------------
    if best_h is None:
        # fallback: simple constant 0.5 function (always feasible)
        N = 200
        best_h = np.full(N, 0.5, dtype=float)
        best_n = N
        best_val = float(evaluate_erdos_solution(best_h))
    else:
        # ensure exact integral = 1 (tiny rounding errors may appear)
        N = best_h.shape[0]
        dx = 2.0 / N
        total = best_h.sum() * dx
        if abs(total - 1.0) > 1e-12:
            best_h = best_h / total
        best_h = np.clip(best_h, 0.0, 1.0)
        # recompute with the official helper for safety
        try:
            best_val = float(evaluate_erdos_solution(best_h))
        except Exception:
            pass

    return best_h, float(best_val), int(best_n)
