import time
import math
import numpy as np

# ------------------- helper functions (top level) ------------------- #

def evaluate_erdos_h(h: np.ndarray) -> float:
    """Return C5 for the step function represented by h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_onto_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the box [0,1]^n intersect {sum = target}.
    The solution has the form clamp(v + λ,0,1) for a suitable λ.
    """
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

    # tiny correction so that the sum is exactly target (up to machine epsilon)
    diff = x.sum() - target
    if abs(diff) > 1e-10:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        if mask.any():
            x[mask] -= diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            # fallback: simple scaling (should be very rare)
            if x.sum() > 0:
                x = x * (target / x.sum())
    return x


def max_corr_and_grad(h: np.ndarray):
    """
    Compute the maximal correlation value max_i h[i]*(1-h[i+shift]) (zero‑padded)
    together with a sub‑gradient of that max w.r.t. h.
    Returns (max_corr, gradient_vector).
    """
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    max_corr = float(corr[idx])
    shift = idx - (n - 1)          # integer shift producing the max

    g = np.ones(n)                # derivative of Σ h[i] part
    if shift >= 0:
        if shift < n:
            # forward term: subtract h[i+shift] where i+shift < n
            g[:n - shift] -= h[shift:]
            # backward term: subtract h[i-shift] where i-shift >= 0
            g[shift:] -= h[:n - shift]
    else:
        s = -shift
        # forward term (i+shift = i-s) valid for i >= s
        g[s:] -= h[:n - s]
        # backward term (i-shift = i+s) valid for i < n-s
        g[:n - s] -= h[s:]

    return max_corr, g


# ------------------- main optimisation routine ------------------- #

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points) with
    h_values – list of step heights on a uniform grid over [0,2],
    c5_bound – the computed C5 for that function,
    n_points – number of grid points used.
    """
    rng = np.random.RandomState(seed)
    start = time.time()

    best_h = None
    best_val = math.inf
    best_n = None

    # candidate grid sizes (must be even to allow sum = n/2)
    candidates = [45, 90, 120]
    candidates = [n if n % 2 == 0 else n + 1 for n in candidates]

    # main loop over resolutions
    for n in candidates:
        target_sum = n / 2.0
        dx = 2.0 / n

        # number of independent restarts per resolution
        restarts = 8
        # iterations per gradient descent run (adjusted later if time is low)
        iters_per = 3000

        for r in range(restarts):
            # stop early if we run out of budget
            if time.time() - start > budget_s * 0.95:
                break

            # ----- initial point ----- #
            use_initial = (r == 0) and ("initial_h_values" in globals())
            if use_initial:
                init = globals()["initial_h_values"]
                init = np.asarray(init, dtype=np.float64)
                if init.shape[0] != n:
                    reps = int(np.ceil(n / init.shape[0]))
                    h = np.tile(init, reps)[:n].astype(np.float64)
                else:
                    h = init.copy()
                h = project_onto_sum(h, target_sum)
            else:
                h = project_onto_sum(rng.rand(n), target_sum)

            # ----- gradient descent on the max‑overlap objective ----- #
            step = 1.0            # initial step size
            decay = 0.9995        # geometric decay per iteration

            # keep track of the best local solution
            cur_val = evaluate_erdos_h(h)
            best_local_h = h.copy()
            best_local_val = cur_val

            for it in range(iters_per):
                if time.time() - start > budget_s * 0.95:
                    break

                max_corr, g = max_corr_and_grad(h)
                cur_val = max_corr * dx

                # store local optimum
                if cur_val < best_local_val:
                    best_local_val = cur_val
                    best_local_h = h.copy()

                # sub‑gradient step and projection back to feasible set
                h_new = h - step * g
                h = project_onto_sum(h_new, target_sum)
                step *= decay

            # occasional random “jitter” to escape shallow minima
            jitter_iters = 500
            for _ in range(jitter_iters):
                if time.time() - start > budget_s * 0.95:
                    break
                i, j = rng.randint(0, n, size=2)
                if i == j:

                # decide a direction of mass transfer
                if rng.rand() < 0.5:
                    delta = rng.uniform(0.0, min(1.0 - h[i], h[j]))
                    if delta == 0.0:

                    h2 = h.copy()
                    h2[i] += delta
                    h2[j] -= delta
                else:
                    delta = rng.uniform(0.0, min(h[i], 1.0 - h[j]))
                    if delta == 0.0:

                    h2 = h.copy()
                    h2[i] -= delta
                    h2[j] += delta
                val2 = evaluate_erdos_h(h2)
                if val2 < best_local_val:
                    best_local_val = val2
                    best_local_h = h2.copy()
                if val2 < cur_val:
                    h = h2
                    cur_val = val2

            # ----- update global best ----- #
            if best_local_val < best_val:
                best_val = best_local_val
                best_h = best_local_h.copy()
                best_n = n

    # Fallback (should never happen) – simple half‑ones block
    if best_h is None:
        n = 45
        best_h = np.zeros(n, dtype=np.float64)
        best_h[: n // 2] = 1.0
        best_val = evaluate_erdos_h(best_h)
        best_n = n

    # Return the solution as plain Python types
    return (best_h.tolist(), float(best_val), int(best_n))
