```
def gradient_for_shift(h, shift):
    n = len(h)
    g = np.zeros_like(h)
    if shift >= 0:
        k = shift
        # (1 - h[i+k]) part for i in [0, n-1-k]
        g[:n-k] += (1 - h[k:])
        # -h[i-k] part for i in [k, n-1] (i >= k)
        g[k:] -= h[:n-k]
    else:
        d = -shift
        # (1 - h[i-d]) for i >= d
        g[d:] += (1 - h[:n-d])
        # -h[i+d] for i <= n - d - 1
        g[:n-d] -= h[d:]
    return g
```

import time
import numpy as np

# ---------- helper utilities ----------
def project_onto_capped_simplex(v: np.ndarray, total: float, max_iter: int = 60) -> np.ndarray:
    """
    Euclidean projection of vector v onto the set {x | 0<=x_i<=1, sum x = total}.
    Uses a simple binary search for the Lagrange multiplier.
    """
    lo = -np.max(v) - 1.0      # ensures v - lo >= 1 for all components
    hi = np.max(v) + 1.0       # ensures v - hi <= 0 for all components
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > total:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the worst‑shift objective w.r.t. h."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        k = shift
        # (1 - h[i+k]) part
        if k < n:
            g[: n - k] += (1.0 - h[k:])
        # -h[i-k] part
        if k > 0:
            g[k:] -= h[: n - k]
    else:
        d = -shift
        # (1 - h[i-d]) part
        if d < n:
            g[d:] += (1.0 - h[: n - d])
        # -h[i+d] part
        if d > 0:
            g[: n - d] -= h[d:]
    return g


def compute_c5(h: np.ndarray) -> float:
    """Exact C5 value for a discretised step function."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv.max() * (2.0 / n)


def max_shift_info(h: np.ndarray):
    """Return (max_value, shift) for current h."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    idx = int(conv.argmax())
    max_val = float(conv[idx])
    shift = idx - (n - 1)
    return max_val, shift


def binary_round(h: np.ndarray) -> np.ndarray:
    """Force exactly half of entries to 1 (rest 0) by taking the largest values."""
    n = len(h)
    k = int(round(n / 2))
    # argsort descending and set first k to 1
    idx = np.argpartition(-h, k)[:k]
    out = np.zeros_like(h)
    out[idx] = 1.0
    return out

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    # time management
    start = time.time()
    np.random.seed(seed)

    # --- initialise vector -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
        # guarantee feasibility

    else:
        # fall back to a modest size if no parent is supplied
        n = 100
        h0 = project_onto_capped_simplex(np.random.rand(n), n / 2.0)

    total_sum = n / 2.0
    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # if the parent already beats the target we can return immediately
    TARGET = 0.3808
    if best_val <= TARGET:
        return best_h.tolist(), best_val, n

    # parameters for the local search
    max_restarts = 12
    iters_per_restart = 800
    base_step = 0.6

    # --- optimisation loop -------------------------------------------------
    for restart in range(max_restarts):
        # time check
        if time.time() - start > budget_s * 0.97:
            break

        # start from current best with a small perturbation
        if restart == 0:
            cur_h = best_h.copy()
        else:
            cur_h = project_onto_capped_simplex(
                best_h + np.random.normal(scale=0.05, size=n), total_sum
            )
        step = base_step

        for it in range(iters_per_restart):
            # time guard
            if time.time() - start > budget_s * 0.97:
                break

            max_val, shift = max_shift_info(cur_h)

            # compute gradient for the worst shift
            grad = gradient_for_shift(cur_h, shift)

            # take a step and re‑project
            cand = project_onto_capped_simplex(cur_h - step * grad, total_sum)

            cand_val = np.max(np.correlate(cand, 1.0 - cand, mode="full"))
            # accept if strictly improving the worst‑shift value
            if cand_val < max_val:
                cur_h = cand
                max_val = cand_val
                # slowly increase step after a successful move
                step = min(step * 1.07, 1.5)
            else:
                # otherwise shrink the step
                step *= 0.72
                if step < 1e-5:
                    break

            cur_c5 = max_val * (2.0 / n)
            if cur_c5 < best_val:
                best_val = cur_c5
                best_h = cur_h.copy()
                if best_val <= TARGET:
                    # early exit if we already beat the target bound
                    return best_h.tolist(), best_val, n

        # try a binary rounding of the best continuous solution obtained so far

            cb = compute_c5(hb)
            if cb < best_val:
                best_val = cb
                best_h = hb

    # final sanity projection (ensures constraints are met exactly)
    best_h = project_onto_capped_simplex(best_h, total_sum)

    return best_h.tolist(), best_val, n
