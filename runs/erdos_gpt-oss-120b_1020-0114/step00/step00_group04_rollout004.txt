def project(v, target):
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    # if diff close to zero, done
    while abs(diff) > 1e-12:
        if diff > 0:
            free = (v < 1.0)
            nfree = free.sum()
            if nfree == 0: break
            add = diff / nfree
            v[free] = np.minimum(v[free] + add, 1.0)
        else:
            free = (v > 0.0)
            nfree = free.sum()
            if nfree == 0: break
            sub = diff / nfree
            v[free] = np.maximum(v[free] + sub, 0.0) # diff negative, sub negative
        diff = target - v.sum()
    return v
```

def project(v, target):
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    # iterative water-filling
    while abs(diff) > 1e-12:
        if diff > 0:
            free = v < 1.0 - 1e-12
            nfree = free.sum()
            if nfree == 0:
                break
            add = diff / nfree
            v[free] = np.minimum(v[free] + add, 1.0)

            free = v > 1e-12
            nfree = free.sum()
            if nfree == 0:
                break
            sub = diff / nfree
            v[free] = np.maximum(v[free] + sub, 0.0)
        diff = target - v.sum()
    return v

import numpy as np
import time
import math

def _project(v, target):
    """Project vector v onto [0,1] box and enforce sum(v)=target."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    # simple water‑filling loop
    while abs(diff) > 1e-12:
        if diff > 0:
            free = v < 1.0 - 1e-12
            nfree = free.sum()
            if nfree == 0:
                break
            add = diff / nfree
            v[free] = np.minimum(v[free] + add, 1.0)
        else:
            free = v > 1e-12
            nfree = free.sum()
            if nfree == 0:
                break
            sub = diff / nfree
            v[free] = np.maximum(v[free] + sub, 0.0)
        diff = target - v.sum()
    return v


def _c5_of(h):
    """Return the overlap constant C₅ for a step vector h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)


def run(seed=42, budget_s=1100, **kwargs):

    cur_val = best_val

    start = time.time()
