import time
import numpy as np

def _compute_c5(h):
    """maximum overlap integral using the prescribed np.correlate definition."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_box_hyperplane(x, target_sum, eps=1e-12):
    """Project x onto the intersection of [0,1]^n and the hyperplane sum(x)=target_sum."""
    lo, hi = -2.0, 2.0
    # binary search for λ such that sum clip(x-λ,0,1) = target_sum
    for _ in range(60):
        mid = (lo + hi) * 0.5
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(x - lam, 0.0, 1.0)

def _smooth_weights(corr, beta):
    """soft‑max weights approximating a max over the correlation vector."""
    maxc = np.max(corr)
    expc = np.exp(beta * (corr - maxc))
    return expc / (expc.sum() + 1e-30)

def _gradient(h, w):
    """Gradient of the smooth max objective with respect to h."""
    n = h.shape[0]
    g = np.zeros_like(h)
    L = 2 * n - 1
    for m in range(L):
        lag = m - (n - 1)          # actual lag corresponding to this entry
        wm = w[m]
        if lag >= 0:
            end = n - lag
            if end <= 0:
                continue
            # contribution from h[i] * (1 - h[i+lag])
            g[:end]      += wm * (1.0 - h[lag:])   # ∂/∂h[i]
            g[lag:]      -= wm * h[:end]          # ∂/∂h[i+lag]
        else:
            d = -lag
            end = n - d
            if end <= 0:
                continue
            # contribution from h[i+d] * (1 - h[i])
            g[d:]        += wm * (1.0 - h[:end])   # ∂/∂h[i+d]
            g[:end]      -= wm * h[d:]            # ∂/∂h[i]
    return g

def _optimize(initial_h, time_budget):
    """Run a simple projected gradient descent on the smoothed max objective."""
    n = initial_h.shape[0]
    target = n / 2.0
    h = initial_h.copy()
    best_h = h.copy()
    best_val = _compute_c5(best_h)

    start = time.time()
    it = 0
    beta = 0.5                     # start with a mild smoothing
    while True:
        # time check
        if time.time() - start > time_budget * 0.94:
            break

        # current objective and its gradient (smooth version)
        corr = np.correlate(h, 1.0 - h, mode="full")
        cur_val = np.max(corr) * (2.0 / n)
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()

        # weights for smooth max
        w = _smooth_weights(corr, beta)

        # gradient of the smooth max
        g = _gradient(h, w)

        # step size (decreases slowly)
        step = 0.05 / (1.0 + np.sqrt(it + 1.0))

        # tentative move and projection
        h_new = h - step * g
        h_new = _project_box_hyperplane(h_new, target)

        # occasional random jitter to escape plateaus
        if it % 500 == 0 and it > 0:
            jitter = 0.02 * (np.random.rand(n) - 0.5)
            h_new = _project_box_hyperplane(h_new + jitter, target)

        h = h_new

        # slowly increase smoothing parameter
        if it % 200 == 0 and beta < 15.0:
            beta = min(15.0, beta * 1.08 + 0.02)

        it += 1

    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    """
    Return a step‑function that yields a low value of the overlap integral.
