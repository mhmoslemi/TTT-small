    lo = -5.0

            hi = mid

    diff = target - y.sum()

        mask = (y > 0.0) & (y < 1.0)

            pass
    return y

def subgradient_step(h, lr):
    N = h.shape[0]
    # compute correlation to find worst shift
    r = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(r)
    k = idx - (N - 1)
    g = np.zeros_like(h)
    if k >= 0:
        kk = k
        if kk < N:
            g[:N-kk] += (1.0 - h[kk:])
            g[kk:] -= h[:N-kk]
    else:
        l = -k
        if l < N:
            g[l:] += (1.0 - h[:N-l])
            g[:N-l] -= h[l:]
    h_new = h - lr * g
    h_new = project_box_sum(h_new, N/2.0)

import time
import math
import numpy as np

def _project_box_sum(x: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector x onto the set {y | 0<=y_i<=1, sum(y)=target}
    using a simple binary search for the additive offset λ such that
    y_i = clip(x_i + λ, 0, 1) and sum(y)=target.
    """
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)

    # tiny correction for numerical drift
    diff = target - y.sum()
    if abs(diff) > 1e-10:
        mask = (y > 0.0) & (y < 1.0)
        if mask.any():
            cnt = mask.sum()
            y[mask] += diff / cnt
            y = np.clip(y, 0.0, 1.0)
    return y


def _evaluate_c5(h: np.ndarray) -> float:
    """Return the C5 bound for a given step function h (discretised)."""
    n = h.shape[0]
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode='full')
    return float(r.max() * dx)


def _initialize_h(n: int) -> np.ndarray:
    """
    Initialise h with the provided global 'initial_h_values' if present,
    interpolating to the desired length, otherwise random uniform.
    The resulting h satisfies the constraints exactly.
    """
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=np.float64)
        m = base.shape[0]
        x_old = np.linspace(0.0, 2.0, m)
        x_new = np.linspace(0.0, 2.0, n)
        h = np.interp(x_new, x_old, base)
    else:
        rng = np.random.default_rng()
        h = rng.random(n)
    h = _project_box_sum(h, n / 2.0)
    return h


def _subgradient_step(h: np.ndarray, lr: float) -> np.ndarray:
    """
    Perform one sub‑gradient step for the max‑overlap objective.
    lr is the learning rate.
    """
    n = h.shape[0]
    r = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(r))
    k = idx - (n - 1)                     # shift that attains the current maximum
    g = np.zeros_like(h)

    if k >= 0:
        kk = k
        if kk < n:
            g[: n - kk] += (1.0 - h[kk:])
            g[kk:]      -= h[: n - kk]
    else:
        l = -k
        if l < n:
            g[l:]       += (1.0 - h[: n - l])
            g[: n - l]  -= h[l:]

    h_new = h - lr * g
    return _project_box_sum(h_new, n / 2.0)


def _hill_climb_step(h: np.ndarray, step_max: float, rng: np.random.Generator):
    """
    Try a tiny random mass transfer between two coordinates.
    Return (new_h, improved_flag).
    """
    n = h.shape[0]
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)

    hi, hj = h[i], h[j]
    lower = max(-hi,      hj - 1.0)   # smallest Δ so that both stay in [0,1]
    upper = min( 1.0 - hi, hj)       # largest  Δ
    if lower >= upper:
        return h, False

    delta = rng.uniform(lower, upper)
    if abs(delta) > step_max:
        delta = step_max * math.copysign(1.0, delta)
    delta = max(lower, min(delta, upper))

    if abs(delta) < 1e-12:
        return h, False

    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta

    val_cur = _evaluate_c5(h)
    val_new = _evaluate_c5(h_new)
    if val_new < val_cur:
        return h_new, True
    return h, False


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise a step function h to obtain a low Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # Choose a working resolution.  Use at least double the parent size.
    if 'initial_h_values' in globals():
        base_n = len(initial_h_values)
        n_pts = max(140, base_n * 2)
    else:
        n_pts = 200

    # Initial candidate
    h = _initialize_h(n_pts)
    best_h = h.copy()
    best_val = _evaluate_c5(best_h)

    # ---- Sub‑gradient descent phase ----
    max_iter = 2000
    for it in range(max_iter):
        if time.time() - start > budget_s * 0.88:
            break
        lr = 0.5 * (1.0 - it / max_iter) + 0.01   # monotone decreasing, never below 0.01
        h = _subgradient_step(h, lr)
        val = _evaluate_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()

    # ---- Simple random hill‑climb phase ----
    step_max = 0.05
    hc_iters = 8000
    for it in range(hc_iters):
        if time.time() - start > budget_s * 0.96:
            break
        h_candidate, improved = _hill_climb_step(best_h, step_max, rng)
        if improved:
            best_h = h_candidate
            best_val = _evaluate_c5(best_h)
            step_max *= 0.995   # slowly tighten the step size

    # Final projection (safety) and evaluation
    best_h = _project_box_sum(best_h, n_pts / 2.0)
    best_val = _evaluate_c5(best_h)

    return (best_h, float(best_val), n_pts)
