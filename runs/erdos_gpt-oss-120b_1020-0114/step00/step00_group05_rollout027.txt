import time
import numpy as np

def _normalize(v):
    """Clip to [0,1] and adjust to exact sum = len(v)/2."""
    n = v.size
    target = n / 2.0
    v = np.clip(v, 0.0, 1.0)
    for _ in range(10):
        s = v.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        free = (v > 0.0) & (v < 1.0)
        cnt = free.sum()
        if cnt == 0:
            v += diff / n
        else:
            v[free] += diff / cnt
        v = np.clip(v, 0.0, 1.0)
    return v

def _c5_fast(h):
    """Fast C5 using the same definition the harness expects."""
    n = h.size
    dx = 2.0 / n
    # np.correlate with mode='full' implements zero‑padding.
    # This matches the statement in the problem description.
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def _c5_eval(h):
    """Use the provided evaluation helper if it exists."""
    try:
        return evaluate_erdos_solution(h)   # type: ignore[name-defined]
    except Exception:
        return _c5_fast(h)

def _random_feasible(n):
    """Create a random feasible vector of length n."""
    v = np.random.rand(n)
    v = v - np.mean(v) + 0.5          # force mean 0.5
    v = _normalize(v)
    return v

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # initialise problem size and a feasible starting vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h_cur = np.asarray(initial_h_values, dtype=float).copy()
    else:
        # default size – small enough to keep optimisation cheap
        n = 70
        h_cur = _random_feasible(n)

    n = h_cur.size
    h_cur = _normalize(h_cur)                # guarantee exact feasibility
    cur_c = _c5_eval(h_cur)

    best_h = h_cur.copy()
    best_c = cur_c

    # ------------------------------------------------------------------
    # Simulated‑annealing style local search
    # ------------------------------------------------------------------
    max_no_improve = 30000          # restart after this many non‑improving steps
    no_improve = 0
    it = 0
    while time.time() - start < budget_s - 0.5:
        # temperature based on elapsed fraction
        frac = (time.time() - start) / budget_s
        temp = 0.018 * (1.0 - frac) + 0.0005   # slowly cools to near zero

        # step size shrinks with time
        step = 0.45 * (1.0 - frac) + 0.005

        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        hi, hj = h_cur[i], h_cur[j]

        # allowed delta to keep both entries in [0,1] and respect step size
        lower = max(-hi, hj - 1.0, -step)
        upper = min(1.0 - hi, hj, step)
        if lower >= upper:
            continue

        delta = np.random.uniform(lower, upper)

        cand = h_cur.copy()
        cand[i] = hi + delta
        cand[j] = hj - delta

        cand_c = _c5_eval(cand)

        # acceptance rule (Metropolis)
        if cand_c < cur_c or np.random.rand() < np.exp(-(cand_c - cur_c) / (temp + 1e-12)):
            h_cur = cand
            cur_c = cand_c
            no_improve = 0
            if cand_c < best_c:
                best_c = cand_c
                best_h = cand.copy()
        else:
            no_improve += 1

        # restart if stuck for a long time
        if no_improve >= max_no_improve:
            h_cur = _random_feasible(n)
            cur_c = _c5_eval(h_cur)
            no_improve = 0

        it += 1

    # final exact evaluation (in case the helper uses a more accurate routine)
    final_c = _c5_eval(best_h)

    return (best_h, final_c, n)
