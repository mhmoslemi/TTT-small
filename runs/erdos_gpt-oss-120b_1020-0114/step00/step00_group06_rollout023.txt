import numpy as np, time

def compute_c5(h):
    N = h.shape[0]
    dx = 2.0 / N
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def project_interval_sum(v, target):
    # bisection for lambda
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def propose_swap(h, rng):
    N = h.shape[0]
    i = rng.integers(N)
    j = rng.integers(N)
    while j == i:
        j = rng.integers(N)
    low = max(-h[i], h[j] - 1.0)
    high = min(1.0 - h[i], h[j])
    if low >= high:
        return None, False
    delta = rng.uniform(low, high)
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    return new_h, True

def gradient_step_max(h, dx):
    N = h.shape[0]
    # correlation f = h * (1-h)
    f = np.correlate(h, 1 - h, mode='full') * dx
    max_val = f.max()
    # find all shifts achieving max (within tolerance)
    tol = 1e-12
    idxs = np.where(f >= max_val - tol)[0]
    shifts = idxs - (N - 1)
    grad = np.zeros(N)
    for shift in shifts:
        if shift >= 0:
            # term 1 - (h_i+shift + h_i-shift)
            term = np.ones(N)
            term[:N-shift] -= h[shift:]
            term[shift:] -= h[:N-shift]
        else:
            s = -shift
            term = np.ones(N)
            term[s:] -= h[:N-s]
            term[:N-s] -= h[s:]
        grad += term
    # average and scale
    grad = grad / len(shifts) * dx
    step = 0.2  # can tune
    y = h - step * grad
    # project to feasible set
    y = project_interval_sum(y, N/2)
    return y

def run(seed=42, budget_s=1100, **kwargs):
    import time
    rng = np.random.default_rng(seed)
    start = time.time()
    # initial vector
    init = globals().get('initial_h_values', None)
    if init is not None:
        h = np.array(init, dtype=np.float64)
        N = h.shape[0]
    else:
        N = 200
        h = np.zeros(N, dtype=np.float64)
        half = N // 2
        h[:half] = 1.0
        rng.shuffle(h)
    # ensure feasibility
    target_sum = N / 2.0
    if not np.isclose(h.sum(), target_sum, atol=1e-9):
        h = project_interval_sum(h, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_c5 = best_c5
    dx = 2.0 / N
    # simulated annealing parameters
    temp = max(0.01, (cur_c5 - 0.38) * 0.3)
    min_temp = 1e-6
    decay = 0.9995
    iter_no_improve = 0
    # main loop
    while time.time() - start < budget_s * 0.97:
        # occasionally use gradient step
        if rng.random() < 0.01:
            new_h = gradient_step_max(cur_h, dx)
            new_c5 = compute_c5(new_h)
        else:
            new_h, ok = propose_swap(cur_h, rng)
            if not ok:
                continue
            new_c5 = compute_c5(new_h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            cur_h = new_h
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
                iter_no_improve = 0
            else:
                iter_no_improve += 1
        else:
            iter_no_improve += 1
        # temperature cooling
        temp = max(min_temp, temp * decay)
        # if no improvement for long, reinitialize from best or random
        if iter_no_improve > 20000:
            # restart from best to explore new region
            cur_h = best_h.copy()
            cur_c5 = best_c5
            iter_no_improve = 0
        # early exit if we beat target
        if best_c5 <= 0.3808:
            break
    # final ensure feasibility (project again)
    best_h = project_interval_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, float(best_c5), best_h.shape[0]

import time
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated by correlation."""
    N = h.shape[0]
    dx = 2.0 / N
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _project_interval_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the box [0,1]^n while enforcing sum(v)=target.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


def _propose_swap(h: np.ndarray, rng: np.random.Generator):
    """Perform a feasible 2‑coordinate move that preserves the sum."""
    N = h.shape[0]
    i = rng.integers(N)
    j = rng.integers(N - 1)
    if j >= i:
        j += 1
    # feasible interval for delta so that both entries stay in [0,1]
    low = max(-h[i], h[j] - 1.0)
    high = min(1.0 - h[i], h[j])
    if low >= high:
        return None, False
    delta = rng.uniform(low, high)
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    return new_h, True


def _gradient_step_max(h: np.ndarray, target_sum: float) -> np.ndarray:
    """
    One gradient step aimed at reducing the current maximal overlap.
    The gradient of the active shift(s) is averaged and a step is taken,
    then the result is projected back onto the feasible set.
    """
    N = h.shape[0]
    dx = 2.0 / N
    # correlation of h with 1‑h
    f = np.correlate(h, 1.0 - h, mode="full") * dx
    max_val = f.max()
    # all shifts attaining the maximum (within tolerance)
    idxs = np.where(f >= max_val - 1e-12)[0]
    shifts = idxs - (N - 1)
    direction = np.zeros(N)
    for shift in shifts:
        direction += 1.0
        if shift >= 0:
            direction[: N - shift] -= h[shift:]
            direction[shift:] -= h[: N - shift]
        else:
            s = -shift
            direction[s:] -= h[: N - s]
            direction[: N - s] -= h[s:]
    direction /= len(shifts)  # average direction
    step_size = 0.2
    y = h - step_size * direction
    return _project_interval_sum(y, target_sum)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # initialise h
    init = globals().get("initial_h_values", None)
    if init is not None:
        h = np.array(init, dtype=np.float64)
    else:
        # fallback: 200 points, half ones half zeros
        N = 200
        h = np.zeros(N, dtype=np.float64)
        h[: N // 2] = 1.0
        rng.shuffle(h)

    N = h.shape[0]
    target_sum = N / 2.0

    # make sure the starting vector satisfies the constraints
    if not np.isclose(h.sum(), target_sum, atol=1e-9):
        h = _project_interval_sum(h, target_sum)

    best_h = h.copy()
    best_c5 = _compute_c5(best_h)
    cur_h = best_h.copy()
    cur_c5 = best_c5

    # simulated‑annealing parameters
    temperature = max(0.01, (cur_c5 - 0.38) * 0.3)
    min_temp = 1e-6
    decay = 0.9995
    stagnation = 0

    # main optimisation loop
    while time.time() - start < budget_s * 0.97:
        # occasionally use a deterministic gradient move
        if rng.random() < 0.01:
            new_h = _gradient_step_max(cur_h, target_sum)
            new_c5 = _compute_c5(new_h)
        else:
            new_h, ok = _propose_swap(cur_h, rng)
            if not ok:
                continue
            new_c5 = _compute_c5(new_h)

        delta = new_c5 - cur_c5
        accept = False
        if delta < 0:
            accept = True
        elif temperature > 0:
            prob = np.exp(-delta / max(temperature, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_h, cur_c5 = new_h, new_c5
            if new_c5 < best_c5:
                best_c5, best_h = new_c5, new_h.copy()
                stagnation = 0
            else:
                stagnation += 1
        else:
            stagnation += 1

        temperature = max(min_temp, temperature * decay)

        # if we have been stagnating for a long time, restart from the best found
        if stagnation > 20000:
            cur_h, cur_c5 = best_h.copy(), best_c5
            stagnation = 0

        # early exit if we beat the target
        if best_c5 <= 0.3808:
            break

    # final safety projection (should already be feasible)
    best_h = _project_interval_sum(best_h, target_sum)
    best_c5 = _compute_c5(best_h)

    return best_h, float(best_c5), best_h.shape[0]
