def gradient_max_corr(h, dx):
    conv = np.correlate(h, 1 - h, mode='full')
    # conv length 2N-1
    idx = np.argmax(conv)  # index of max
    k = idx - (len(h) - 1)  # shift
    N = len(h)
    grad = np.zeros_like(h)
    # add term for i such that i+k<N.
    if k >= 0:
        # left factor exists for i = 0..N-1-k
        grad[:N-k] += dx * (1 - h[k:])  # compute 1 - h[i+k]
        # right factor exists for i = k..N-1 (i - k >=0)
        grad[k:] -= dx * h[:N-k]  # h[i-k]
    else:
        # k negative: shift left
        kk = -k
        # i+k >=0 when i >= kk
        grad[kk:] += dx * (1 - h[:N-kk])
        # i - k >= 0 when i - (-kk) = i+kk < N => i <= N-kk-1
        grad[:N-kk] -= dx * h[kk:]
    return grad, max_val, k

def project_simplex_box(v, C, tol=1e-9, max_iter=100):
    lo = -np.max(v) - 1  # ensure v+lo <= 0 for all i
    hi = 1 - np.min(v) + 1  # ensure v+hi >=1 for all i
    # Actually we can just set lo = -np.max(v) - 1, hi = 1 - np.min(v) + 1
    for _ in range(max_iter):
        mid = (lo + hi) / 2
        x = np.clip(v + mid, 0, 1)
        s = x.sum()
        if np.abs(s - C) < tol:
            return x
        if s > C:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo+hi)/2, 0, 1)

    # conv array length L = 2N-1
    # shift corresponding to index s: k = s - (N-1)

import time, math
import numpy as np

def _project_to_box_simplex(v, target_sum, max_iter=60):
    """Project v onto {0<=x_i<=1, sum x_i = target_sum}."""
    lo = -np.max(v)               # after adding lo all entries <=0 → sum 0
    hi = 1.0 - np.min(v)          # after adding hi all entries >=1 → sum = len(v)
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def _compute_c(h, dx):
    """Maximum overlap integral for a given h."""
    conv = np.correlate(h, 1.0 - h, mode='full')
    return conv.max() * dx

def _gradient_smooth_max(h, dx, alpha):
    """Gradient of the smooth-maximum of the overlap integrals."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode='full')
    max_c = conv.max()
    # stable softmax weights
    exp_arg = alpha * (conv - max_c)
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:
        # fallback to subgradient of the active shift
        idx = int(conv.argmax())
        k = idx - (n - 1)
        g = np.zeros_like(h)
        if k >= 0:
            g[:n - k] += (1.0 - h[k:])
            g[k:]     -= h[:n - k]
        else:
            kk = -k
            g[kk:]    += (1.0 - h[:n - kk])
            g[:n - kk] -= h[kk:]
        return g * dx
    w /= w_sum
    g = np.zeros_like(h)
    for s, wk in enumerate(w):
        if wk == 0.0:
            continue
        k = s - (n - 1)
        if k >= 0:
            g[:n - k] += wk * (1.0 - h[k:])
            g[k:]     -= wk * h[:n - k]
        else:
            kk = -k
            g[kk:]    += wk * (1.0 - h[:n - kk])
            g[:n - kk] -= wk * h[kk:]
    return g * dx

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the working dimension.
    # Use the provided construction if available, otherwise start from
    # a random feasible vector.
    # ------------------------------------------------------------------
    try:
        base_h = np.asarray(initial_h_values, dtype=float)
    except NameError:
        base_h = None

    if base_h is not None:
        n_base = base_h.size
        # up‑sample a bit to give the optimizer more freedom
        n = max(150, n_base * 4)
        # linearly interpolate the supplied sample onto the new grid
        dx_base = 2.0 / n_base
        x_base = np.arange(n_base) * dx_base + dx_base * 0.5
        dx = 2.0 / n
        x_new = np.arange(n) * dx + dx * 0.5
        h0 = np.interp(x_new, x_base, base_h)
    else:
        n = int(kwargs.get('n_points', 200))
        dx = 2.0 / n
        h0 = rng.random(n)

    target_sum = n / 2.0
    # enforce feasibility
    h0 = _project_to_box_simplex(h0, target_sum)

    # ------------------------------------------------------------------
    # Optimization parameters
    # ------------------------------------------------------------------
    best_h = h0.copy()
    best_val = _compute_c(best_h, dx)

    cur_h = best_h.copy()
    cur_val = best_val

    start = time.time()
    it = 0
    max_iter = 10_000_000          # safety bound
    step0 = 1.0
    alpha = 5.0
    alpha_inc = 0.001
    target = 0.3808                # problem target

    while time.time() - start < budget_s - 0.4 and it < max_iter:
        # gradient of smooth max
        grad = _gradient_smooth_max(cur_h, dx, alpha)

        # simple step size schedule
        step = step0 / math.sqrt(it + 1.0)

        # propose a new point and re‑project onto the feasible set
        h_new = _project_to_box_simplex(cur_h - step * grad, target_sum)

        new_val = _compute_c(h_new, dx)

        # metropolis‑style acceptance (allows occasional uphill moves)
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) * 10.0):
            cur_h, cur_val = h_new, new_val

        # keep the best ever seen
        if new_val < best_val:
            best_h, best_val = h_new.copy(), new_val
            # early exit if we already beat the target
            if best_val <= target:
                break

        # periodic random restart to avoid stagnation
        if it % 5000 == 0 and it > 0:
            rnd = rng.random(n)
            cur_h = _project_to_box_simplex(rnd, target_sum)
            cur_val = _compute_c(cur_h, dx)

        # slowly increase the smoothing parameter so we eventually focus on the true max
        alpha = min(200.0, alpha + alpha_inc)
        it += 1

    # ------------------------------------------------------------------
    # Return result in the required format
    # ------------------------------------------------------------------
    return (best_h, float(best_val), int(n))
