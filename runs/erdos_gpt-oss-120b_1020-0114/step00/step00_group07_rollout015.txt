def compute_c5(h):
    N = h.shape[0]

    S = P - O
    max_val = np.max(S)

def project_sum(h, target):
    # Simple scaling then clip and renormalize
    total = h.sum()
    if abs(total - target) > 1e-12:
        h = h * (target / total)
        # Clip after scaling
        h = np.clip(h, 0.0, 1.0)
        # Re-adjust for any clipped mass loss/gain iteratively
        # simple loop up to few passes
        for _ in range(5):
            total = h.sum()
            diff = target - total
            if abs(diff) < 1e-12:
                break
            # distribute diff uniformly among non-clipped entries
            free = (h > 0) & (h < 1)
            if not np.any(free):
                break
            adjust = diff / np.count_nonzero(free)
            h[free] += adjust
            h = np.clip(h, 0.0, 1.0)
    return h

import time
import random
import numpy as np

def compute_c5_and_worst(h):
    """Return (C5, full_S_array, lag_of_worst) for a given h."""
    N = h.shape[0]
    # overlap term O(k) = sum h[i] * h[i+k]
    O = np.correlate(h, h, mode="full")
    # partial sum P(k) = sum h[i] over overlapping indices
    P = np.correlate(h, np.ones(N), mode="full")
    S = P - O                      # S(k) = P(k) - O(k)
    idx = int(np.argmax(S))        # index of worst lag
    worst = float(S[idx])
    lag = idx - (N - 1)            # convert index to lag in [-N+1, N-1]
    return worst * (2.0 / N), S, lag


def compute_c5(h):
    """Simple wrapper returning only the C5 value."""
    N = h.shape[0]
    O = np.correlate(h, h, mode="full")
    P = np.correlate(h, np.ones(N), mode="full")
    S = P - O
    return float(np.max(S)) * (2.0 / N)


def project_sum(h, target):
    """Scale and clip h so that sum(h) == target while keeping 0≤h≤1."""
    h = np.asarray(h, dtype=np.float64).copy()
    total = h.sum()
    if total != 0:
        h *= target / total
    # Clip and correct any small deviation caused by clipping
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 0) & (h < 1)
        if not np.any(free):
            break
        adjust = diff / free.sum()
        h[free] += adjust
        h = np.clip(h, 0.0, 1.0)
    return h


def random_initial(N):
    """Create a random feasible h with the correct integral."""
    target = N / 2.0
    while True:
        h = np.random.rand(N)
        h = h / h.sum() * target
        if np.all(h <= 1.0):
            return h
