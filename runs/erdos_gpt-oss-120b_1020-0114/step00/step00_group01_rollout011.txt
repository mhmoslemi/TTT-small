import time
import numpy as np

def _c5_score(h: np.ndarray) -> float:
    """max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    # np.correlate gives linear (zero‑padded) correlation
    return float(np.correlate(h, 1.0 - h, mode="full").max() * dx)


def _project_simple(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project v onto {0≤x_i≤1, Σx_i = target} using a scalar shift and clipping.
    Finds theta with binary search s.t. sum(clamp(v-theta,0,1)) = target.
    """
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _random_feasible(n: int, target: float, rng: np.random.Generator) -> np.ndarray:
    """Return a random vector in [0,1]^n with the prescribed sum."""
    v = rng.random(n)
    return _project_simple(v, target)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------
    #  initialise a starting vector
    # ------------------------------------------------------------
    init = globals().get("initial_h_values")
    if init is not None:
        h = np.asarray(init, dtype=np.float64).copy()
    else:
        n = 60
        h = np.full(n, 0.5, dtype=np.float64)

    n = h.size
    target_sum = n * 0.5  # because ∫h = 1  and dx = 2/n

    # Ensure feasibility (might be necessary when a parent is provided)
    if not (np.all((0.0 <= h) & (h <= 1.0)) and abs(h.sum() - target_sum) < 1e-12):
        h = _project_simple(h, target_sum)

    best_h = h.copy()
    best_score = _c5_score(best_h)

    # ------------------------------------------------------------
    #  optimisation parameters
    # ------------------------------------------------------------
    max_iter_per_restart = 25000
    temp0 = 0.07
    cooling = 0.9995   # exponential cooling factor

    # ------------------------------------------------------------
    #  main loop – many short restarts until time runs out
    # ------------------------------------------------------------
    while time.time() - start < budget_s * 0.93:
        # occasionally use the parent solution as a fresh start
        if rng.random() < 0.2 and init is not None:
            cur = np.asarray(init, dtype=np.float64).copy()
        else:
            cur = _random_feasible(n, target_sum, rng)

        cur_score = _c5_score(cur)
        temperature = temp0

        for it in range(max_iter_per_restart):
            # pick two distinct indices and propose a pairwise transfer
            i, j = rng.integers(0, n, size=2)
            if i == j:

            hi, hj = cur[i], cur[j]

            delta = rng.uniform(low, high)
            cur[i] = hi + delta
