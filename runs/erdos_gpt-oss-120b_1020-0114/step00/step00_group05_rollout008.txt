    j = rng.integers(N-1)
    if j >= i:
        j += 1

    if max_delta > 1e-15:
        delta = rng.random() * max_delta
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        return h_new
    # try opposite

        delta = rng.random() * max_delta

        h_new[i] += delta

import numpy as np
import time

def _c5_value(h: np.ndarray) -> float:
    """Return the C₅ value for a given step function."""
    n = h.shape[0]
    return (2.0 / n) * np.max(np.correlate(h, 1.0 - h, mode="full"))


def _adjust_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Move mass between entries until the sum equals target (keeping 0‑1 bounds)."""
    diff = target - h.sum()
    # simple greedy pairwise correction
    while abs(diff) > 1e-12:
        if diff > 0:
            idx = rng.choice(np.where(h < 1.0)[0])
            inc = min(diff, 1.0 - h[idx])
            h[idx] += inc
            diff -= inc
        else:
            idx = rng.choice(np.where(h > 0.0)[0])
            dec = min(-diff, h[idx])
            h[idx] -= dec
            diff += dec
    return h


def _quadratic_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Quadratic “bump’’ h(x)=0.75·(1-(x-1)²) sampled on n points and projected to the feasible set."""
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    h = 0.75 * (1.0 - (x - 1.0) ** 2)
    h = np.clip(h, 0.0, 1.0)
    h = _adjust_sum(h, n / 2.0, rng)
    return h


def _rand_move(h: np.ndarray, eps: float, rng: np.random.Generator) -> np.ndarray | None:
    """Transfer a small amount δ (≤eps) from one index to another."""
    n = h.shape[0]
    i = rng.integers(n)
    j = rng.integers(n - 1)
    if j >= i:
        j += 1
    # i → j
    mx = min(h[i], 1.0 - h[j], eps)
    if mx > 1e-15:
        δ = rng.random() * mx
        h2 = h.copy()
        h2[i] -= δ
        h2[j] += δ
        return h2
    # j → i
    mx = min(h[j], 1.0 - h[i], eps)
    if mx > 1e-15:
        δ = rng.random() * mx
        h2 = h.copy()
        h2[i] += δ
        h2[j] -= δ
        return h2
    return None


def _targeted_move(h: np.ndarray, shift: int, eps: float, rng: np.random.Generator) -> np.ndarray | None:
    """Move mass from a high‑contributing site to its partner at the worst shift."""
    n = h.shape[0]
    if shift >= 0:
        i_idxs = np.arange(n - shift)
        j_idxs = i_idxs + shift
    else:
        s = -shift
        i_idxs = np.arange(s, n)
        j_idxs = i_idxs - s
    if i_idxs.size == 0:
        return None
    contrib = h[i_idxs] * (1.0 - h[j_idxs])
    order = np.argsort(-contrib)  # descending
    for k in order:
        i, j = i_idxs[k], j_idxs[k]
        mx = min(h[i], 1.0 - h[j], eps)
        if mx > 1e-15:
            δ = rng.random() * mx
            h2 = h.copy()
            h2[i] -= δ
            h2[j] += δ
            return h2
    return None

    """
    Optimize a step function h on [0,2] to lower the Erdős C₅ bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Acquire a feasible start vector
    # ------------------------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=np.float64)  # may be supplied by the harness
        n = h0.size
    except NameError:
        n = 70
        h0 = np.zeros(n, dtype=np.float64)
        h0[: n // 2] = 1.0
        rng.shuffle(h0)

    h0 = np.clip(h0, 0.0, 1.0)
    h0 = _adjust_sum(h0, n / 2.0, rng)

    # ------------------------------------------------------------------
    # 2. Initialise best candidate with a couple of constructions
    # ------------------------------------------------------------------
    best_h = h0.copy()
    best_c = _c5_value(best_h)

    quad = _quadratic_shape(n, rng)
    quad_c = _c5_value(quad)
    if quad_c < best_c:
        best_h, best_c = quad, quad_c

    cur_h, cur_c = best_h.copy(), best_c

    # ------------------------------------------------------------------
    # 3. Simulated‑annealing / hill‑climbing loop
    # ------------------------------------------------------------------
    start = time.time()
    eps_start, eps_end = 0.15, 0.001
    T_start, T_end = 0.02, 0.0005

            if cand_c < best_c:

    # ------------------------------------------------------------------

    final_c = _c5_value(best_h)

    return best_h, final_c, n
