def compute_c5(h):
    n = len(h)
    dx = 2.0/n
    corr = np.correlate(h, 1-h, mode='full')
    return corr.max()*dx

def hill_climb(h, max_iter, budget):
    start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n = len(h)
    # Pre-calc for speed: correlation each iteration can be computed from scratch; O(n^2). Acceptable for n ~200.
    for it in range(max_iter):
        if time.time() - start > budget*0.98:
            break
        # propose move
        # pick two indices i != j
        i, j = np.random.choice(n, 2, replace=False)
        # propose delta
        # determine max allowable delta: cannot go below 0 or above 1.
        # Let delta be positive if we transfer from i to j (i decreases, j increases). Choose direction randomly.
        if np.random.rand() < 0.5:
            # transfer from i to j
            max_delta = min(h[i], 1-h[j])
        else:
            # transfer from j to i
            max_delta = min(h[j], 1-h[i])
        if max_delta <= 1e-12:
            continue
        # random delta up to max_delta
        delta = np.random.rand()*max_delta
        # apply move
        h_new = h.copy()
        if np.random.rand() < 0.5:
            # direction as originally selected (i->j)
            h_new[i] -= delta
            h_new[j] += delta
        else:
            # opposite direction (j->i)
            h_new[i] += delta
            h_new[j] -= delta
        # ensure within [0,1] (should be)
        # compute c5
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
            h = h_new
        # optionally accept occasional worse move (simulated annealing)
        # ignore for now
    return best_h, best_c5

def adjust_to_target(p, target):
    h = p.copy()
    # Simple iterative scaling
    for _ in range(10):
        total = h.sum()
        diff = target - total
        if abs(diff) < 1e-12:
            break
        # compute free indices where not at bounds
        free = (h > 1e-12) & (h < 1-1e-12)
        if not free.any():
            # all at bounds; distribute diff to random free positions if possible
            idx = np.random.choice(len(h))
            h[idx] = np.clip(h[idx] + diff, 0, 1)
            break
        # proportionally adjust free entries
        adjust = diff / free.sum()
        h[free] += adjust
        # clip
        h = np.clip(h, 0, 1)
    # final small correction to meet target precisely within 1e-8 by adding diff to random free entry
    total = h.sum()
    diff = target - total
    if abs(diff) > 1e-9:
        # find index with capacity
        if diff > 0:
            # need increase

import numpy as np
import time
import math
import random

def _compute_c5(h):
    """Return the overlap bound C5 for a step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    # full correlation of h with (1-h); zero padding is handled automatically
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _adjust_to_target(h, target):
    """
    Project h onto the feasible set:
      0 <= h_i <= 1 and sum(h_i) = target.
    Simple water‑filling style iteration.
    """
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = np.where(h < 1.0 - 1e-12)[0]
        else:
            free = np.where(h > 1e-12)[0]
        if free.size == 0:
            break
        add = diff / free.size
        if diff > 0:
            inc = np.minimum(1.0 - h[free], add)
            h[free] += inc
        else:
            dec = np.minimum(h[free] - 0.0, -add)
            h[free] -= dec
    # tiny leftover correction on a random free coordinate
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            cand = np.where(h < 1.0 - 1e-12)[0]
        else:
            cand = np.where(h > 1e-12)[0]
        if cand.size:
            i = int(np.random.choice(cand))
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def _interpolate_h(old_h, new_n):
    """Linear interpolation of a step‑function defined by old_h onto new_n points."""
    old_n = old_h.shape[0]
    dx_old = 2.0 / old_n
    dx_new = 2.0 / new_n
    old_x = (np.arange(old_n) + 0.5) * dx_old
    new_x = (np.arange(new_n) + 0.5) * dx_new
    return np.interp(new_x, old_x, old_h)

def _hill_climb_sa(h0, time_limit):
    """
    Simulated‑annealing style hill climbing.
    Moves consist of transferring a random amount of mass from one index
    to another while preserving the total sum.
    """
    start = time.time()
    n = h0.shape[0]
    best = h0.copy()
    best_val = _compute_c5(best)
    cur = best.copy()
    cur_val = best_val

    # initial temperature (a fraction of the typical C5 scale)
    T0 = 0.02

    # stagnation counter to exit early
    stagn = 0
    while time.time() - start < time_limit:
        # linear cooling
        t_frac = (time.time() - start) / time_limit
        T = T0 * (1.0 - t_frac)

        # choose a donor (decrease) and a receiver (increase)
        donor = np.where(cur > 1e-9)[0]
        receiver = np.where(cur < 1.0 - 1e-9)[0]
        if donor.size == 0 or receiver.size == 0:
            break
        i = int(np.random.choice(donor))
        j = int(np.random.choice(receiver))
        max_delta = min(cur[i], 1.0 - cur[j])
        if max_delta <= 1e-12:
            continue
        delta = max_delta * np.random.rand()
        if delta < 1e-12:
            continue

        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new_val = _compute_c5(new)
        dval = new_val - cur_val

        if dval < 0 or np.random.rand() < math.exp(-dval / max(T, 1e-12)):
            cur, cur_val = new, new_val
            stagn = 0
            if new_val < best_val:
                best, best_val = new.copy(), new_val
        else:
            stagn += 1

        if stagn > 30000:          # give up if nothing improves for long
            break
    return best, best_val
