def compute_c5(h):
    n = len(h)
    N = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(1 - h, n=N)
    corr = np.fft.irfft(H * np.conj(G), n=N)
    corr = corr[:2*n - 1]
    return float(np.max(corr) * (2.0 / n))

def simulated_annealing(n, rng, time_limit):
    import numpy as np, math, time
    # Initialize binary vector
    h = np.zeros(n)
    ones = n // 2
    h[:ones] = 1.0
    rng.shuffle(h)
    cur_f = compute_c5(h)
    best_h = h.copy()
    best_f = cur_f
    start = time.time()
    # We'll use linear cooling
    max_iters = int(1e6)  # high but will break due to time
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        # propose move: transfer delta
        # choose i, j indices
        i = rng.integers(n)
        while h[i] <= 0.0:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] >= 1.0:
            j = rng.integers(n)
        max_delta = min(h[i], 1.0 - h[j])
        # choose delta relative size; we can use uniform fraction
        delta = max_delta * rng.random()
        # Slightly reduce small moves to speed
        if delta < 1e-6:
            continue
        cand = h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_f = compute_c5(cand)
        # Temperature schedule linearly decreasing to near zero
        T = max(0.01, 0.2 * (1 - it / max_iters))
        if cand_f < cur_f or rng.random() < math.exp(-(cand_f - cur_f) / T):
            h = cand
            cur_f = cand_f
            if cur_f < best_f:
                best_f = cur_f
                best_h = h.copy()
    return best_h, best_f

import numpy as np, time, math

def _c5_value(h: np.ndarray) -> float:
    """C5 value for a discrete step function h."""
    n = h.shape[0]
    # correlation of h with (1-h) – same as the problem statement
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def _sturmian_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Balanced 0/1 sequence from the Beatty word of the golden ratio."""
    phi = (math.sqrt(5) - 1) / 2.0          # ≈0.618034
    idx = np.arange(n, dtype=np.float64)
    h = (np.mod(idx * phi, 1.0) < 0.5).astype(np.float64)
    # enforce exactly n/2 ones (n is even)
    target = n // 2
    cur = int(h.sum())
    if cur != target:
        if cur < target:
            zeros = np.where(h == 0)[0]
            flip = rng.choice(zeros, size=target - cur, replace=False)
            h[flip] = 1.0
        else:
            ones = np.where(h == 1)[0]
            flip = rng.choice(ones, size=cur - target, replace=False)
            h[flip] = 0.0
    return h


def _binary_simulated_annealing(n: int,
                               rng: np.random.Generator,
                               timelimit: float,
                               start_vec: np.ndarray = None) -> tuple:
    """
    Binary (0/1) simulated annealing.
    Swaps a randomly chosen 1 with a randomly chosen 0 and accepts
    according to a simple temperature schedule.
    """
    half = n // 2
    if start_vec is None:
        h = np.zeros(n, dtype=np.float64)
        h[:half] = 1.0
        rng.shuffle(h)
    else:
        h = start_vec.copy()

    cur_val = _c5_value(h)
    best_h = h.copy()
    best_val = cur_val

    start = time.time()
    # a very large iteration budget – loop stops when time is exceeded
    max_iter = 10_000_000
    T0 = 0.4

    for it in range(max_iter):
        if time.time() - start > timelimit:
            break

        # pick a random 1 and a random 0
        i = rng.integers(n)
        while h[i] < 0.5:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] > 0.5:
            j = rng.integers(n)

        # propose the swap
        cand = h.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        cand_val = _c5_value(cand)

        # linear cooling schedule
        T = max(T0 * (1.0 - it / max_iter), 1e-8)
        accept = cand_val < cur_val or rng.random() < math.exp(-(cand_val - cur_val) / T)
        if accept:
            h = cand
            cur_val = cand_val
            if cur_val < best_val:
                best_val, best_h = cur_val, h.copy()

    return best_h, best_val


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a step function on [0,2] that gives a small C5 value.
    Returns (list_of_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------
    # problem size – larger gives finer granularity but costs a bit more
    # ------------------------------------------------------------
    n = 800                     # must be even so that sum = n/2 is integral
    target_sum = n // 2

    # ------------------------------------------------------------
    # initialise with a known balanced Sturmian word (binary)
    # ------------------------------------------------------------
    best_h = _sturmian_binary(n, rng)
    best_val = _c5_value(best_h)

    # ------------------------------------------------------------
    # If the harness provided an initial vector, respect it (but keep the
    # feasibility constraints)
    # ------------------------------------------------------------
    try:
        init = np.array(initial_h_values, dtype=np.float64)
        if init.size != n:
            # resample to the required length by linear interpolation
            x_old = np.linspace(0.0, 2.0, init.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            init = np.interp(x_new, x_old, init)
        init = np.clip(init, 0.0, 1.0)
        # force the exact integral constraint
        cur = init.sum()
        diff = target_sum - cur
        if abs(diff) > 1e-9:
            free = np.where((init > 0.0) & (init < 1.0))[0]
            if free.size:
                init[free] += diff / free.size
                init = np.clip(init, 0.0, 1.0)
            else:
                # fallback binary correction
                ones = np.where(init >= 0.5)[0]
                zeros = np.where(init < 0.5)[0]
                if diff > 0:
                    flip = rng.choice(zeros, size=int(diff), replace=False)
                    init[flip] = 1.0
                else:
                    flip = rng.choice(ones, size=int(-diff), replace=False)
                    init[flip] = 0.0
        val = _c5_value(init)
        if val < best_val:
            best_val, best_h = val, init.copy()
    except Exception:
        # no parent construction – ignore
        pass

    # ------------------------------------------------------------
    # Main optimisation loop – repeated short simulated‑annealing runs
    # ------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.93:
        remaining = budget_s - (time.time() - start_time)
        # allocate at most 12 seconds to a single SA run
        run_time = min(12.0, remaining)
        # start each run from a fresh Sturmian binary vector (good diversity)
        start_vec = _sturmian_binary(n, rng)
        h_candidate, val_candidate = _binary_simulated_annealing(
            n, rng, run_time, start_vec=start_vec
        )
        if val_candidate < best_val:
            best_val, best_h = val_candidate, h_candidate.copy()

    # ------------------------------------------------------------
    # final housekeeping – exact integral, tiny clipping
    # ------------------------------------------------------------
    cur_sum = best_h.sum()
    diff = target_sum - cur_sum
    if abs(diff) > 1e-9:
        free = np.where((best_h > 0.0) & (best_h < 1.0))[0]
        if free.size:
            best_h[free] += diff / free.size
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            # flip a few bits if needed
            idx = rng.choice(n, size=int(abs(diff)), replace=False)
            best_h[idx] = 1.0 - best_h[idx]

    final_val = _c5_value(best_h)

    return best_h.tolist(), final_val, n
