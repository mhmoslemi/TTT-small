def project_to_box_sum(x, target_sum, lo=0.0, hi=1.0):
    # returns y = clip(x + lam, lo, hi) s.t. sum(y)=target_sum.
    # lam found by binary search.
    lo_lam = -2.0 # enough
    hi_lam = 2.0
    for _ in range(60):
        mid = (lo_lam + hi_lam) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target_sum:
            hi_lam = mid
        else:
            lo_lam = mid
    return np.clip(x + (lo_lam+hi_lam)/2.0, lo, hi)

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def project_to_sum_box(x: np.ndarray, target_sum: float) -> np.ndarray:
    """Project x onto the box [0,1]^n while fixing the sum to target_sum.
    Finds λ such that y = clip(x + λ, 0, 1) satisfies sum(y) = target_sum."""
    lo, hi = -1.0, 1.0
    # binary search for λ
    for _ in range(30):
        mid = (lo + hi) * 0.5
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)
    return y


def evaluate_c5(h: np.ndarray) -> float:
    """Return the Erdős C5 value for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def max_corr_shift(h: np.ndarray):
    """Return (max_corr, shift, full_correlation_array)."""
    n = h.shape[0]
    full = np.correlate(h, 1.0 - h, mode="full")
    idx = int(full.argmax())
    shift = idx - (n - 1)
    return full[idx], shift, full


def gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the (unnormalised) correlation w.r.t. h for a given shift."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        L = n - shift
        g[:L] += 1.0 - h[shift : shift + L]
        g[shift : shift + L] -= h[:L]
    else:
        s = -shift
        L = n - s
        g[s : s + L] += 1.0 - h[:L]
        g[:L] -= h[s : s + L]
    return g


def upsample_h(old_h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a step function to a new number of points."""
    old_n = old_h.shape[0]
    if old_n == new_n:
        return old_h.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    target = new_n / 2.0
    new_h = project_to_sum_box(new_h, target)
    return new_h
