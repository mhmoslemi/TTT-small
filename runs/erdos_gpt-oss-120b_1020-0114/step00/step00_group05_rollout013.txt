import time, math, numpy as np

def _c5_of_h(h):
    """Compute the C5 bound for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def _vshape_h(a, p, n):
    """
    V‑shaped (symmetric) construction:
        h(x) = a + b * |x-1|^p ,   b = (1-2a)*(p+1)/2
    The integral of h on [0,2] equals 1 for any a∈[0,0.5] and p>0.
    The function is sampled at mid‑points of the uniform grid.
    """
    dx = 2.0 / n
    x = (np.arange(n, dtype=np.float64) + 0.5) * dx          # mid‑points
    b = (1.0 - 2.0 * a) * (p + 1.0) / 2.0
    h = a + b * np.abs(x - 1.0) ** p
    # clip for safety, then renormalise (tiny correction for discretisation)
    h = np.clip(h, 0.0, 1.0)
    total = h.sum() * dx
    if not math.isclose(total, 1.0, rel_tol=1e-12, abs_tol=1e-12):
        h = h / total
        h = np.clip(h, 0.0, 1.0)
        total = h.sum() * dx
        if not math.isclose(total, 1.0, rel_tol=1e-12, abs_tol=1e-12):
            # final simple renormalisation (the deviation is negligible)
            h = h * (1.0 / total)
    return h


def _random_binary_smooth(n, radius=3):
    """Return a smooth (continuous) vector built from a random 0/1 pattern."""
    # start with exactly n/2 ones
    h = np.zeros(n, dtype=np.float64)
    ones = np.random.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    # simple moving‑average smoothing
    k = max(1, radius)
    kernel = np.ones(k, dtype=np.float64) / k
    h = np.convolve(h, kernel, mode="same")
    # enforce bounds and the integral constraint
    target = n / 2.0
    cur = h.sum()
    if cur > 0:
        h = h * (target / cur)
    h = np.clip(h, 0.0, 1.0)
    # tiny redistribution to hit the sum exactly (numeric safety)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        free = np.where((h > 0) & (h < 1))[0]
        if free.size > 0:
            h[free] += diff / free.size
            h = np.clip(h, 0.0, 1.0)
    return h


def _targeted_step(h, max_step=0.2):
    """
    Produce a feasible neighbour of h.
    The move is guided by the shift having the smallest overlap
    (i.e. the current worst shift).
    """
    n = h.shape[0]
    # correlation of h with itself (zero‑padding for shifts)
    corr = np.correlate(h, h, mode="full")
    centre = n - 1
    # ignore the zero shift (max overlap)
    corr[centre] = np.inf
    worst = np.argmin(corr)        # smallest overlap
    k = worst - centre             # signed shift
    # build the gradient of the overlap for this shift
    grad = np.zeros(n, dtype=np.float64)
    if k >= 0:
        idx = np.arange(n - k)
        grad[idx] += h[idx + k]
        idx2 = np.arange(k, n)
        grad[idx2] += h[idx2 - k]
    else:
        kk = -k
        idx = np.arange(kk, n)
        grad[idx] += h[idx - kk]
        idx2 = np.arange(n - kk)
        grad[idx2] += h[idx2 + kk]

    # choose a source (decrease) with low gradient and a target (increase) with high gradient
    inc_candidates = np.where(h < 1.0 - 1e-12)[0]
    dec_candidates = np.where(h > 1e-12)[0]
    if inc_candidates.size == 0 or dec_candidates.size == 0:
        return h.copy()    # no move possible

    i_target = inc_candidates[np.argmax(grad[inc_candidates])]
    i_source = dec_candidates[np.argmin(grad[dec_candidates])]

    # allowed delta respecting the box constraints
    delta_max = min(1.0 - h[i_target], h[i_source])
    if delta_max <= 0:
        return h.copy()
    delta = delta_max * np.random.rand() * max_step
    new_h = h.copy()
    new_h[i_target] += delta
    new_h[i_source] -= delta
    return new_h


def _random_step(h, max_step=0.2):
    """Simple random mass transfer between two coordinates."""
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    if np.random.rand() < 0.5:
        # move from i to j
        delta = min(h[i], 1.0 - h[j])
        if delta <= 0:
            return h.copy()
        delta *= np.random.rand() * max_step
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
    else:
        delta = min(h[j], 1.0 - h[i])
        if delta <= 0:
            return h.copy()
        delta *= np.random.rand() * max_step
        new_h = h.copy()
        new_h[j] -= delta
        new_h[i] += delta
    return new_h
