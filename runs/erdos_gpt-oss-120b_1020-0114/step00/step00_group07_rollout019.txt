import time
import math
import numpy as np

# ---------- helper functions (top level) ----------
def compute_c5_raw(h: np.ndarray) -> float:
    """Maximum linear correlation value (without dx scaling)."""
    return np.max(np.correlate(h, 1.0 - h, mode="full"))


def compute_c5(h: np.ndarray) -> float:
    """C5 value with the proper dx factor."""
    n = len(h)
    return compute_c5_raw(h) * (2.0 / n)


def max_corr_shift(h: np.ndarray):
    """Return (max_val, shift, full_corr_vector)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    max_val = float(corr[max_idx])
    shift = max_idx - (len(h) - 1)
    return max_val, shift, corr


def targeted_candidate(h: np.ndarray, rng: np.random.Generator, step_factor: float):
    """
    Propose a move that tries to reduce the current maximal correlation.
    Returns (i, j, delta) meaning move delta from i to j,
    or None if no feasible move was found.
    """
    n = len(h)
    _, shift, _ = max_corr_shift(h)
    if shift == 0:
        return None

    if shift > 0:
        hi = h[: n - shift]
        hj = h[shift:]
        start = 0
    else:  # shift < 0
        hi = h[-shift :]
        hj = h[: n + shift]
        start = -shift

    # contribution = h[i] * (1 - h[i+shift])
    contrib = hi * (1.0 - hj)
    if contrib.size == 0:
        return None
    i_rel = int(np.argmax(contrib))
    i = start + i_rel
    j = i + shift

    # direction i -> j ?
    max_delta = min(h[i], 1.0 - h[j])
    if max_delta <= 0.0:
        # try opposite direction
        max_delta = min(h[j], 1.0 - h[i])
        if max_delta <= 0.0:
            return None
        i, j = j, i

    delta = max_delta * rng.random() * step_factor
    if delta <= 0.0:
        return None
    return int(i), int(j), float(delta)


def random_candidate(h: np.ndarray, rng: np.random.Generator, max_step: float):
    """Propose a random feasible transfer of mass."""
    n = len(h)
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1  # ensure i != j
    # try i -> j
    max_delta = min(h[i], 1.0 - h[j])
    if max_delta <= 0.0:
        # try opposite
        max_delta = min(h[j], 1.0 - h[i])
        if max_delta <= 0.0:
            return None
        i, j = j, i
    delta = max_delta * rng.random() * max_step
    if delta <= 0.0:
        return None
    return int(i), int(j), float(delta)


def apply_move(h: np.ndarray, i: int, j: int, delta: float):
    """Transfer delta from index i to index j (in‑place)."""
    h[i] -= delta
    h[j] += delta

    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 constant.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # ----- initialise h -------------------------------------------------
    try:
        init = initial_h_values.copy()          # provided by the harness
        n = len(init)
    except NameError:                           # no parent construction
        n = 80
        init = np.full(n, 0.5, dtype=float)

    cur_sum = init.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        init = init * (target_sum / cur_sum)
        init = np.clip(init, 0.0, 1.0)

                init[idx] += add
                diff -= add

    # ----- state variables -----------------------------------------------
    current_h = init.copy()
    current_raw = compute_c5_raw(current_h)

    best_h = current_h.copy()
    best_raw = current_raw

    step_start = 0.8                         # large steps early

        progress = (time.time() - start_time) / budget
        temperature = initial_temp * (1.0 - progress)
        step_factor = step_start * (1.0 - progress) + step_end * progress

        # propose a move
        cand = targeted_candidate(current_h, rng, step_factor)
        if cand is None:
            cand = random_candidate(current_h, rng, step_factor)
        if cand is None:
            iteration += 1
            continue

        i, j, delta = cand
        trial = current_h.copy()
        apply_move(trial, i, j, delta)
        trial_raw = compute_c5_raw(trial)

        # acceptance test
        accept = False
        if trial_raw <= current_raw:

            if rng.random() < prob:

            current_raw = trial_raw

            if trial_raw < best_raw - 1e-12:
                best_h = trial.copy()
                best_raw = trial_raw

        iteration += 1

    dx = 2.0 / n

    while refined and time.time() - start_time < budget:

            apply_move(trial, i, j, delta)

            if trial_raw < best_raw - 1e-12:
                best_h = trial
                best_raw = trial_raw
                refined = True

    # ----- final evaluation ---------------------------------------------
    try:
        final_c5 = evaluate_erdos_solution(best_h)
    except NameError:
        final_c5 = best_raw * dx

    return (best_h.tolist(), float(final_c5), n)
