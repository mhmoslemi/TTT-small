def project_bounds_sum(v, target):

    lo = -1.0
    hi = 1.0

        s = x.sum()

            lo = lam
        else:
            hi = lam

import numpy as np
import time, math

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    return corr.max() * (2.0 / n)

def project_bounds_sum(v, target):
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - ((lo + hi) / 2.0), 0.0, 1.0)

def gradient_shift(h, shift):
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        g[:n-shift] += (1 - h[shift:])
        g[shift:] -= h[:n-shift]
    else:
        s = -shift
        g[s:] += (1 - h[:n-s])
        g[:n-s] -= h[s:]
    return g

def local_optimize(h0, time_budget):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val
    start = time.time()
    iter_count = 0
    no_improve = 0
    while time.time() - start < time_budget:
        # compute C values
        corr = np.correlate(h, 1 - h, mode='full')
        C_vals = corr * (2.0 / n)
        idx_max = np.argmax(C_vals)
        max_shift = idx_max - (n - 1)
        # compute gradient for max shift
        g = gradient_shift(h, max_shift)
        norm = np.linalg.norm(g)
        if norm < 1e-8:

            i, j = np.random.choice(n, size=2, replace=False)

            h_new[i] += delta
            h_new[j] -= delta
        else:

import numpy as np, time, math

def compute_c5(h: np.ndarray) -> float:
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)

def project_bounds_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection onto {x | 0<=x_i<=1, sum x_i = target}."""
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def random_feasible(n: int, target: float) -> np.ndarray:
    v = np.random.rand(n)
    return project_bounds_sum(v, target)

def gradient_shift(h: np.ndarray, shift: int) -> np.ndarray:
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        g[: n - shift] += (1.0 - h[shift:])
        g[shift:] -= h[: n - shift]
    else:
        s = -shift
        g[s:] += (1.0 - h[: n - s])
        g[: n - s] -= h[s:]
    return g

def local_optimize(h0: np.ndarray, time_budget: float) -> tuple:
    n = h0.shape[0]
    target = n * 0.5
    h = h0.copy()
    cur_val = compute_c5(h)
    best_h, best_val = h.copy(), cur_val
    start = time.time()
    no_improve = 0
    while time.time() - start < time_budget:
        corr = np.correlate(h, 1.0 - h, mode="full")
        c_vals = corr * (2.0 / n)
        idx_max = int(np.argmax(c_vals))
        shift = idx_max - (n - 1)

        g = gradient_shift(h, shift)
        if np.linalg.norm(g) < 1e-9:
            # random feasible exchange move
            i, j = np.random.choice(n, 2, replace=False)
            dmin = -min(h[i], 1.0 - h[j])
            dmax = min(1.0 - h[i], h[j])
            if dmax <= dmin:
                continue
            delta = np.random.uniform(dmin, dmax)
            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta
        else:
            elapsed = time.time() - start
            frac = elapsed / time_budget
            step = 0.05 * (1.0 - frac) + 1e-5
            h_new = h - step * g
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_bounds_sum(h_new, target)

        new_val = compute_c5(h_new)

        # simple Metropolis acceptance
        if new_val < cur_val:
            h, cur_val = h_new, new_val
            no_improve = 0
            if new_val < best_val:
                best_h, best_val = h_new.copy(), new_val
        else:

            cur_val = best_val
            no_improve = 0
    return best_h, best_val
