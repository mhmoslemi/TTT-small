Define function hill_climb(h_initial, time_limit):

    h = h_initial.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n = len(h)
    # Precompute indices sets for potential move to avoid endless loops.
    start_time = time.time()
    # schedule step size.
    while time.time() - start_time < time_limit:
        # pick i, j
        i = np.random.randint(n)
        if h[i] >= 1:
            continue
        j = np.random.randint(n)
        if i == j or h[j] <= 0:
            continue
        max_delta = min(1 - h[i], h[j])
        if max_delta <= 0:
            continue
        # step factor based on remaining time (starting large).
        elapsed = time.time() - start_time
        # compute remaining proportion
        proportion = 1 - elapsed / time_limit
        # set max_transfer = proportion * max_delta  maybe
        delta = max_delta * np.random.rand() * proportion
        if delta <= 0:
            continue
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        # compute objective
        new_c5 = compute_c5(h_new)
        # simulated annealing temperature
        T0 = 0.01
        T = T0 * proportion  # decreasing
        if new_c5 < best_c5 - 1e-12 or np.random.rand() < math.exp(-(new_c5 - best_c5) / (T + 1e-12)):
            h = h_new
            if new_c5 < best_c5:
                best_c5, best_h = new_c5, h_new.copy()
    return best_h, best_c5

import time, math, numpy as np

def _compute_c5(h, dx):
    """max overlap integral for discrete step function h."""
    # correlation of h with (1-h)
    c = np.correlate(h, 1.0 - h, mode="full")
    return np.max(c) * dx

def _shape_base(d, n, dx):
    """piecewise‑linear base shape f with max value 1, integral =1."""
    if d < 1e-12:                     # pure block of length 1
        x = (np.arange(n) + 0.5) * dx
        return (x <= 1.0).astype(float)

    x = (np.arange(n) + 0.5) * dx    # centre points
    f = np.zeros_like(x)

    # left ramp 0 -> 1 over [0,d]
    left = x < d
    f[left] = x[left] / d

    # plateau of height 1 over [d,1]
    plat = (x >= d) & (x <= 1.0)
    f[plat] = 1.0

    # right ramp 1 -> 0 over [1,1+d]
    right = (x > 1.0) & (x < 1.0 + d)
    f[right] = (1.0 + d - x[right]) / d

    # elsewhere already zero
    return f

def _adjust_sum_to_target(h, target):
    """Exact integral correction while respecting [0,1] bounds."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # distribute the difference greedily
    for i in range(len(h)):
        if diff > 0:
            room = 1.0 - h[i]
            inc = min(room, diff)
            h[i] += inc
            diff -= inc
        else:
            room = h[i]
            dec = min(room, -diff)
            h[i] -= dec
            diff += dec
        if abs(diff) < 1e-12:
            break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) with a low C5 for the
    Erdős minimum‑overlap problem."""
    start_time = time.time()
    np.random.seed(seed)

    # number of samples – 100 gives a good trade‑off between speed
    # and resolution; it satisfies the “<1k samples” guideline.
    n = 100
    dx = 2.0 / n
    target_sum = n / 2.0                     # integral = 1  => sum = n/2

    best_h = None
    best_c5 = np.inf
    best_params = (None, None)

    # ------------------------------------------------------------------
    # 1) Coarse grid search over the two shape parameters:
    #    d  – length of each sloping side (0 ≤ d ≤ 1)
    #    β  – mixing coefficient with the constant 0.5 function
    # ------------------------------------------------------------------
    d_vals = np.linspace(0.0, 1.0, 51)        # step 0.02
    beta_vals = np.linspace(0.0, 1.0, 51)    # step 0.02

    for d in d_vals:
        f = _shape_base(d, n, dx)            # base shape, integral = 1

        for beta in beta_vals:
            a = (1.0 - beta) / 2.0            # constant part (≥0)
            h = a + beta * f                  # convex combo, respects [0,1]

            # tiny numerical drift → enforce exact integral
            h = _adjust_sum_to_target(h, target_sum)

            c5 = _compute_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_params = (d, beta)

    # ------------------------------------------------------------------
    # 2) Refine around the best coarse parameters.
    # ------------------------------------------------------------------
    d0, beta0 = best_params
    if d0 is not None:
        d_fine = np.linspace(max(0.0, d0 - 0.02), min(1.0, d0 + 0.02), 41)
        beta_fine = np.linspace(max(0.0, beta0 - 0.02), min(1.0, beta0 + 0.02), 41)

        for d in d_fine:
            f = _shape_base(d, n, dx)
            for beta in beta_fine:
                a = (1.0 - beta) / 2.0
                h = a + beta * f
                h = _adjust_sum_to_target(h, target_sum)
                c5 = _compute_c5(h, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    best_params = (d, beta)

    # ------------------------------------------------------------------
    # 3) Optional: short simulated‑annealing / hill‑climbing phase.
    #    Works on the current best vector, preserving the integral.
    # ------------------------------------------------------------------
    elapsed = time.time() - start_time
    remaining = max(0.0, budget_s - elapsed)
    if remaining > 5.0:                     # reserve a few seconds for the final phase
        h = best_h.copy()
        best = best_c5
        n = len(h)
        T0 = 0.02                           # initial temperature
        it_start = time.time()
        while time.time() - it_start < remaining * 0.9:
            # pick indices i (receive) and j (donor)
            i = np.random.randint(n)
            if h[i] >= 1.0:
                continue
            j = np.random.randint(n)
            if i == j or h[j] <= 0.0:
                continue

            # maximal transferable mass respecting bounds
            max_delta = min(1.0 - h[i], h[j])
            if max_delta <= 0.0:
                continue

            # gradually shrink step size as time runs out
            prog = 1.0 - (time.time() - it_start) / (remaining * 0.9)
            delta = max_delta * np.random.rand() * prog

            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta

            c5_new = _compute_c5(h_new, dx)
            # acceptance rule (simulated‑annealing)
            if c5_new < best or np.random.rand() < math.exp(-(c5_new - best) / (T0 * prog + 1e-12)):
                h = h_new
                if c5_new < best:
                    best = c5_new
                    best_h = h_new.copy()

        best_c5 = best

    # ------------------------------------------------------------------
    # 4) Final exact‑sum correction (very small tweaks if needed)
    # ------------------------------------------------------------------
    best_h = _adjust_sum_to_target(best_h, target_sum)
    best_c5 = _compute_c5(best_h, dx)

    # Ensure output is a plain list for maximal compatibility
    return (best_h.tolist(), float(best_c5), n)
