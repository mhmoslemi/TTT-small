def adjust_sum(h, target):
    # Clip
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target - cur
    # If diff > 0, increase some entries <1
    # else decrease some entries >0
    # Use random order
    indices = np.arange(len(h))
    np.random.shuffle(indices)
    for i in indices:
        if diff > 1e-12 and h[i] < 1.0:
            inc = min(diff, 1.0 - h[i])
            h[i] += inc
            diff -= inc
        elif diff < -1e-12 and h[i] > 0.0:
            dec = min(-diff, h[i])
            h[i] -= dec
            diff += dec
        if abs(diff) < 1e-12:
            break
    # If still diff not zero due to rounding, adjust last element
    if abs(diff) > 1e-9:
        # adjust uniformly across all entries to match target
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def c5_value(h: np.ndarray) -> float:
    """Return C5 for a given step function h (discrete vector)."""
    n = h.size
    dx = 2.0 / n
    # correlation h * (1-h) (full mode)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def enforce_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust (by tiny transfers) so that sum(h)==target."""
    h = np.clip(h, 0.0, 1.0)
    cur = float(np.sum(h))
    diff = target - cur
    # simple redistribution loop
    rng = np.random.default_rng()
    while abs(diff) > 1e-12:
        if diff > 0:
            # need to increase some entries
            cand = np.where(h < 1.0 - 1e-12)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            inc = min(diff, 1.0 - h[i])
            h[i] += inc
            diff -= inc
        else:
            # need to decrease some entries
            cand = np.where(h > 1e-12)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            dec = min(-diff, h[i])
            h[i] -= dec
            diff += dec
    return h


def propose_swap(h: np.ndarray, max_delta: float, rng: np.random.Generator) -> np.ndarray | None:
    """Move a small amount of mass from a positive entry to a non‑full entry."""
    n = h.size
    # indices where we can give (h>0) and receive (h<1)
    give = np.where(h > 1e-12)[0]
    take = np.where(h < 1.0 - 1e-12)[0]
    if give.size == 0 or take.size == 0:
        return None
    i = rng.choice(give)
    j = rng.choice(take)
    if i == j:
        return None
    delta = min(max_delta, h[i], 1.0 - h[j])
    if delta <= 0.0:
        return None
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h


def propose_targeted(h: np.ndarray, shift: int, max_delta: float, rng: np.random.Generator) -> np.ndarray | None:
    """Transfer mass from indices that heavily contribute to the current worst shift."""
    n = h.size
    # coefficient w_i = (1 - h[i+shift]) where defined, else 0
    w = np.zeros(n)
    if shift >= 0:
        w[: n - shift] = 1.0 - h[shift:]
    else:
        w[-shift :] = 1.0 - h[: n + shift]

    # source: large w_i * h_i (big contribution)
    src_weights = w * h
    # destination: small w_j * (1-h_j) (small contribution)
    dst_weights = (1.0 - w) * (1.0 - h)

    if src_weights.sum() == 0 or dst_weights.sum() == 0:
        return None

    src_probs = src_weights / src_weights.sum()
    dst_probs = dst_weights / dst_weights.sum()

    i = rng.choice(n, p=src_probs)
    j = rng.choice(n, p=dst_probs)
    if i == j:
        return None

    delta = min(max_delta, h[i], 1.0 - h[j])
    if delta <= 0.0:
        return None
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h


def sa_optimize(h0: np.ndarray, time_limit: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Simulated‑annealing style optimisation of C5."""
    n = h0.size
    target_sum = n / 2.0
    h = enforce_sum(h0.copy(), target_sum)

    best_h = h.copy()
    best_val = c5_value(h)
    cur_val = best_val

    start = time.time()
    max_delta = 0.2
    # temperature schedule (linear decay)
    temp_start = 0.02
    temp_end = 5e-4

    while True:
        now = time.time()
        if now - start > time_limit:
            break
        elapsed = now - start
        frac = elapsed / time_limit
        temp = temp_start * (1.0 - frac) + temp_end * frac

        # correlation to locate current worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)

        # 70% of the time try a targeted move, otherwise a generic swap
        if rng.random() < 0.7:
            cand = propose_targeted(h, shift, max_delta, rng)
            if cand is None:
                cand = propose_swap(h, max_delta, rng)
        else:
            cand = propose_swap(h, max_delta, rng)

        if cand is None:
            continue

        new_val = c5_value(cand)
        diff = new_val - cur_val
        if diff < 0.0 or rng.random() < np.exp(-diff / max(temp, 1e-12)):
            h = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cand.copy()
    # final feasibility check
    best_h = enforce_sum(np.clip(best_h, 0.0, 1.0), target_sum)
    best_val = c5_value(best_h)
    return best_h, best_val


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points) where:
      - h_values : list of step values on [0,2]
      - c5_bound : the computed C5 for the returned function
      - n_points : length of the discrete representation
    """
    rng = np.random.default_rng(seed)
    # Determine problem size
    if "initial_h_values" in globals():
        h_start = np.asarray(initial_h_values, dtype=float)
        n = h_start.size
    else:
        # reasonable default size (even for exact half‑sum)
        n = 200
        # start with a triangular “tent” function (area=1)
        x = np.linspace(0.0, 2.0, n)
        h_start = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
    target = n / 2.0
    h_start = enforce_sum(h_start, target)

    start_time = time.time()
    time_left = budget_s - (time.time() - start_time)
    if time_left <= 0.0:
        val = c5_value(h_start)
        return (h_start.tolist(), val, n)

    # Run the optimiser (single long SA run)
    best_h, best_val = sa_optimize(h_start, time_left, rng)

    # Ensure final vector obeys constraints (tiny numerical drift possible)
    best_h = enforce_sum(np.clip(best_h, 0.0, 1.0), target)
    best_val = c5_value(best_h)

    return (best_h.tolist(), best_val, n)
