    def enforce_constraints(arr):

        # Clip
        arr = np.clip(arr, 0.0, 1.0)
        # Adjust sum iteratively
        current = arr.sum()
        diff = target - current
        # if diff small return
        if abs(diff) < 1e-12:

        # Determine indices that can be increased or decreased
        # We'll perform simple loop
        # Use while diff > 0, add to random not at 1; diff<0, subtract from random not at 0.

        while abs(diff) > 1e-12:

                if len(idx) == 0:
                    break

                # We add proportionally to all idx
                # Compute possible addition total = sum(1 - arr[idx])
                possible = (1 - arr[idx]).sum()
                if possible < diff:
                    # fill all to 1
                    arr[idx] = 1.0
                    diff -= possible
                else:
                    # distribute equally
                    add = diff / len(idx)
                    # ensure not exceed 1

                    diff = 0.0

                    arr[idx] = 0.0
                    diff += possible
                else:

                    arr[idx] -= sub
                    diff = 0.0
        # finally clip
        arr = np.clip(arr, 0.0, 1.0)
        return arr

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------
    # Helper: enforce bounds [0,1] and exact integral constraint
    # ------------------------------------------------------------
    def enforce_constraints(arr):
        n = arr.shape[0]
        target = n / 2.0
        arr = np.clip(arr, 0.0, 1.0)
        cur = arr.sum()
        diff = target - cur
        # simple redistribution loop (few iterations are enough)
        while abs(diff) > 1e-12:
            if diff > 0:
                idx = np.where(arr < 1.0)[0]
                if idx.size == 0:
                    break
                free = (1.0 - arr[idx]).sum()
                if free <= diff:
                    arr[idx] = 1.0
                    diff -= free
                else:
                    add = diff / idx.size
                    arr[idx] += add
                    diff = 0.0
            else:
                idx = np.where(arr > 0.0)[0]
                if idx.size == 0:
                    break
                free = arr[idx].sum()
                if free <= -diff:
                    arr[idx] = 0.0
                    diff += free
                else:
                    sub = (-diff) / idx.size
                    arr[idx] -= sub
                    diff = 0.0
        return np.clip(arr, 0.0, 1.0)

    # ------------------------------------------------------------
    # Helper: compute C5 for a given vector
    # ------------------------------------------------------------
    def compute_c5(vec):
        dx = 2.0 / vec.size
        corr = np.correlate(vec, 1.0 - vec, mode='full')
        return corr.max() * dx

    # ------------------------------------------------------------
    # Acquire initial vector (possibly supplied by the harness)
    # ------------------------------------------------------------
    try:
        init = np.asarray(initial_h_values, dtype=float)  # type: ignore
    except NameError:
        init = None

    # Target resolution (kept modest for speed)
    target_n = 200
    if init is not None and init.size > 0:
        # Resample to target_n points by linear interpolation
        x_old = np.linspace(0.0, 2.0, init.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
        h = np.interp(x_new, x_old, init)
    else:
        # Random start respecting the integral
        h = rng.random(target_n)
        h = enforce_constraints(h)

    n = h.size
    h = enforce_constraints(h)          # guarantee exact constraints
    best_h = h.copy()
    best_val = compute_c5(best_h)

    cur_h = h.copy()
    cur_val = best_val

    # ------------------------------------------------------------
    # Parameters for the stochastic search
    # ------------------------------------------------------------
    start_time = time.time()
    eps_start = 0.20               # maximal delta magnitude at start
    eps_min = 1e-4
    decay = 0.9995                 # exponential decay of eps
    eps = eps_start
    temp_start = 0.02              # initial temperature for SA
    iter_counter = 0
    buffer = 0.5                   # leave a tiny margin before timeout

    # Main optimisation loop
    while True:
        now = time.time()
        elapsed = now - start_time
        if elapsed > budget_s - buffer:
            break

        # ---- propose a two‑point mass transfer -----------------
        i, j = rng.integers(n, size=2)
        if i == j:
            continue

        hi, hj = cur_h[i], cur_h[j]

        # delta such that hi' = hi - delta, hj' = hj + delta
        delta_min = max(hi - 1.0, -hj)
        delta_max = min(hi, 1.0 - hj)
        if delta_max <= delta_min:
            continue

        # random delta bounded by eps and feasibility range
        raw = (rng.random() * 2.0 - 1.0) * eps
        delta = max(min(raw, delta_max), delta_min)
        if abs(delta) < 1e-12:
            continue

        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta

        new_val = compute_c5(new_h)

        # ---- acceptance (simulated annealing) -----------------
        if new_val < cur_val:
            accept = True
        else:
            # temperature linearly decays to 0
            T = temp_start * max(0.0, (budget_s - elapsed) / budget_s)
            accept = (T > 0.0) and (rng.random() < np.exp(-(new_val - cur_val) / T))

        if accept:
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_h = new_h.copy()
                best_val = new_val

        # ---- schedule eps --------------------------------------
        eps = max(eps * decay, eps_min)

        # ---- occasional gradient descent on worst shift ---------
        iter_counter += 1
        if iter_counter % 2000 == 0:
            corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
            s_idx = int(np.argmax(corr))
            shift = s_idx - (n - 1)          # signed shift giving max correlation

            grad = np.zeros_like(cur_h)
            if shift >= 0:
                grad[: n - shift] += (1.0 - cur_h[shift:])
                grad[shift:]      -= cur_h[: n - shift]
            else:
                t = -shift
                grad[t:]   += (1.0 - cur_h[: n - t])
                grad[: n - t] -= cur_h[t:]

            # step size limited by bounds
            pos = grad > 0
            neg = grad < 0
            alphas = []
            if np.any(pos):
                alphas.append(np.min(cur_h[pos] / grad[pos]))
            if np.any(neg):
                alphas.append(np.min((1.0 - cur_h[neg]) / (-grad[neg])))
            max_alpha = min(alphas) if alphas else 1.0

        pass

    return best_h, final_c5, n
