def evaluate(h):
    c = np.correlate(h, 1 - h, mode='full')
    return np.max(c) * (2.0 / len(h))

def project_onto_constraints(v, target):
    lo = -10.0
    hi = 10.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        total = w.sum()
        if total > target:
            lo = lam
        else:
            hi = lam
    return w

def random_feasible(n):

    return project_onto_constraints(v, n/2)

import time
import random
import numpy as np

# ------------------------------------------------------------
# helper: evaluate the objective C5 for a candidate h
# ------------------------------------------------------------
def _c5_value(h):
    """Return max_k ∫ h(x)(1−h(x+k))dx for discrete h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


# ------------------------------------------------------------
# projection onto the capped simplex {x∈[0,1]^n, sum x = target}
# ------------------------------------------------------------
def _project_capped_simplex(v, target):
    """Euclidean projection of v onto [0,1]^n with sum = target."""
    lo, hi = -10.0, 10.0          # wide enough bounds for the bisection
    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


# ------------------------------------------------------------
# gradient of the worst‑shift term
# ------------------------------------------------------------
def _worst_shift_gradient(h):
    """Gradient of the term achieving the current maximum."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    k = idx - (n - 1)               # shift corresponding to the max
    g = np.ones_like(h)

    if k >= 0:
        g[: n - k] -= h[k:]
        g[k:]      -= h[: n - k]
    else:
        kk = -k
        g[kk:]      -= h[: n - kk]
        g[: n - kk] -= h[kk:]

    return g, k


# ------------------------------------------------------------
# a few steps of projected gradient descent
# ------------------------------------------------------------
def _grad_descent(start_h, target, max_iter, time_limit):
    h = start_h.copy()
    best_val = _c5_value(h)
    best_h = h.copy()
    step = 0.1
    start = time.time()

    for _ in range(max_iter):
        if time.time() - start > time_limit:
            break
        g, _ = _worst_shift_gradient(h)
        h_new = h - step * g
        h_new = _project_capped_simplex(h_new, target)
        val = _c5_value(h_new)

        if val < best_val - 1e-12:
            best_h, best_val = h_new, val
            h = h_new
            step = min(step * 1.2, 0.5)      # a bit larger step after success
        else:
            step *= 0.5
            if step < 1e-6:
                step = 0.1
    return best_h, best_val


# ------------------------------------------------------------
# simple random mass‑transfer hill‑climbing
# ------------------------------------------------------------
def _hill_climb(start_h, target, time_limit):
    h = start_h.copy()
    best_h = h.copy()
    best_val = _c5_value(h)
    cur_val = best_val
    n = h.shape[0]
    start = time.time()

    while time.time() - start < time_limit:
        i, j = random.randrange(n), random.randrange(n)
        if i == j:
            continue
        # move a random amount from i to j
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0.0:

        delta = random.random() * max_move
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        val = _c5_value(h_new)
        if val < cur_val - 1e-12:
            h, cur_val = h_new, val
            if val < best_val:
                best_h, best_val = h_new.copy(), val
    return best_h, best_val

    if h0 is None:
        # smooth sinusoidal guess – it satisfies the integral exactly in the limit
        xs = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = (1.0 - np.cos(np.pi * xs / 2.0)) * 0.5
    else:
        h0 = np.clip(h0, 0.0, 1.0)

    # enforce the linear constraints exactly
    h0 = _project_capped_simplex(h0, target_sum)

    best_h = h0.copy()
    best_val = _c5_value(best_h)

    start_all = time.time()

    # --------------------------------------------------------
    # iterative improvement – alternating gradient steps and
    # random hill‑climbing bursts, occasional random restarts.
    # --------------------------------------------------------
    while True:
        elapsed = time.time() - start_all
        remaining = budget_s - elapsed
        if remaining < 1.0:
            break

        # allocate a slice of the remaining budget to a sub‑routine
        slice_time = min(remaining * 0.5, 25.0)

        if random.random() < 0.6:
            # projected gradient descent burst
            cand_h, cand_val = _grad_descent(best_h, target_sum,
                                             max_iter=5000,
                                             time_limit=slice_time)
        else:
            # hill‑climbing burst
            cand_h, cand_val = _hill_climb(best_h, target_sum,
                                           time_limit=slice_time)

        if cand_val < best_val - 1e-12:
            best_h, best_val = cand_h, cand_val

        # occasional random restart to escape possible stagnation
        if random.random() < 0.07:
            rnd = np.random.rand(n)
            rnd = _project_capped_simplex(rnd, target_sum)
            val = _c5_value(rnd)
            if val < best_val:
                best_h, best_val = rnd, val

    final_val = _c5_value(best_h)

    # return plain Python list as required by the harness
    return best_h.tolist(), float(final_val), int(n)
