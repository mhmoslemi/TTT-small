        elapsed = time.time() - start
        frac = elapsed / budget_s
        temp = T0 * (1.0 - frac) + Tf * frac

        # ---- propose a tiny mass transfer between two indices ----
        # pick donor (with mass) and receiver (with room)
        i_up = rng.integers(n)
        i_down = rng.integers(n)
        # ensure feasibility of the move
        attempts = 0
        while (i_up == i_down or h[i_up] >= 1.0 or h[i_down] <= 0.0) and attempts < 10:
            i_up = rng.integers(n)
            i_down = rng.integers(n)
            attempts += 1
        if i_up == i_down or h[i_up] >= 1.0 or h[i_down] <= 0.0:
            continue

        max_delta = min(1.0 - h[i_up], h[i_down], 0.08)  # limit step size
        delta = rng.random() * max_delta

        # apply move (in‑place copy -> evaluate)
        h_new = h.copy()
        h_new[i_up] += delta
        h_new[i_down] -= delta

        new_c5 = _c5_value(h_new, dx)
        delta_c5 = new_c5 - current_c5

        accept = False
        if delta_c5 < 0:
            accept = True
        else:
            # Metropolis acceptance
            if temp > 0 and rng.random() < np.exp(-delta_c5 / temp):
                accept = True

        if accept:
            h = h_new
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                iters_since_best = 0
            else:
                iters_since_best += 1
        else:
            iters_since_best += 1

        iter_cnt += 1

        # ---- every few thousand iterations try a directed improvement ----
        if iter_cnt % 5000 == 0:
            # autocorrelation of h with itself
            q = np.correlate(h, h, mode="full")
            min_idx = int(np.argmin(q))
            shift = min_idx - (n - 1)

            if shift >= 0:
                i_range = np.arange(0, n - shift)
                j_range = i_range + shift
            else:
                i_range = np.arange(-shift, n)
                j_range = i_range + shift

            # locate pair with smallest product h[i]*h[j]
            prod = h[i_range] * h[j_range]
            pair_pos = int(np.argmin(prod))
            i_min = int(i_range[pair_pos])
            j_min = int(j_range[pair_pos])

            # two indices with largest values (excluding the chosen pair)
            order = np.argsort(-h)
            dec = []
            for idx in order:
                if idx not in (i_min, j_min):
                    dec.append(idx)
                if len(dec) == 2:
                    break
            if len(dec) == 2:
                # feasible amount to move
                inc_cap = min(1.0 - h[i_min], 1.0 - h[j_min], 0.05)
                dec_cap = min(h[dec[0]], h[dec[1]], inc_cap)
                step = min(inc_cap, dec_cap)
                if step > 1e-8:
                    h_test = h.copy()
                    h_test[i_min] += step
                    h_test[j_min] += step
                    h_test[dec[0]] -= step
                    h_test[dec[1]] -= step
                    # projection not needed – step respects bounds
                    test_c5 = _c5_value(h_test, dx)
                    if test_c5 < best_c5:
                        best_c5 = test_c5
                        best_h = h_test.copy()
                        h = h_test
                        current_c5 = test_c5
                        iters_since_best = 0

        # ---- random restart if stuck for a long time ----
        if iters_since_best >= max_iter_without_improve:
            # rebuild a fresh random balanced vector
            h = np.zeros(n, dtype=np.float64)
            ones = rng.choice(n, size=n // 2, replace=False)
            h[ones] = 1.0
            current_c5 = _c5_value(h, dx)
            iters_since_best = 0

    # ------------------------------------------------------------------
    # 5. Final feasibility fix and final C5 evaluation
    # ------------------------------------------------------------------
    best_h = _project(best_h)
    final_c5 = _c5_value(best_h, dx)

    return best_h, float(final_c5), int(n)
