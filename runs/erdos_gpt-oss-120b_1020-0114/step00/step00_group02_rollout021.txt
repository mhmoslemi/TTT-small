def binarize_initial(h, N):
    # h may be of same length N; if length not N, we can upsample and then round.
    n = len(h)
    if n != N:
        # interpolate linearly onto N points
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, N, endpoint=False)
        h_interp = np.interp(x_new, x_old, h)
    else:
        h_interp = h.copy()
    # Convert to binary: sort values, set top N/2 as 1 else 0.
    # This ensures sum = N/2 exactly.
    idx = np.argsort(h_interp)[-N//2:]
    binary = np.zeros(N, dtype=int)
    binary[idx] = 1
    return binary

import numpy as np, time, math, random

def compute_correlation(h):
    return np.correlate(h, 1 - h, mode='full')

def max_overlap_and_shift(h):
    corr = compute_correlation(h)
    max_val = int(corr.max())
    max_idx = int(np.argmax(corr))
    shift = max_idx - (len(h) - 1)
    return max_val, shift, corr

def random_swap_indices(h):
    N = len(h)
    i = np.random.randint(N)
    while h[i] != 1:
        i = np.random.randint(N)
    j = np.random.randint(N)
    while h[j] != 0:
        j = np.random.randint(N)
    return i, j

def targeted_swap_indices(h, shift):
    N = len(h)
    L = abs(shift)
    if L == 0 or L >= N:
        return None

    # Need to map j to original index: for shift>=0, j refers to index in left part (same as i). So j stays as j index.

import numpy as np
import time, math, random

def compute_correlation(h):
    """Full cross‑correlation of h with (1‑h)."""
    return np.correlate(h, 1 - h, mode='full')

def max_overlap_and_shift(h):
    """Return (max_overlap, shift_of_max, full_correlation)."""
    corr = compute_correlation(h)
    max_val = int(corr.max())
    max_idx = int(np.argmax(corr))
    shift = max_idx - (len(h) - 1)
    return max_val, shift, corr

def random_swap_indices(h):
    """Choose i with h[i]==1 and j with h[j]==0."""
    n = len(h)
    i = np.random.randint(n)
    while h[i] != 1:
        i = np.random.randint(n)
    j = np.random.randint(n)
    while h[j] != 0:
        j = np.random.randint(n)
    return i, j

def targeted_swap_indices(h, shift):

    such that swapping reduces the mismatches for this shift.

def balanced_binary_from_initial(init, n):
    """Create a binary vector of length n with exactly n/2 ones."""
    if len(init) != n:
        x_old = np.linspace(0.0, 2.0, len(init), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        init = np.interp(x_new, x_old, init)
    arr = np.zeros(n, dtype=int)
    idx = np.argsort(init)[-n // 2 :]
    arr[idx] = 1
    return arr

def irr_rotation_binary(n):
    """Sturmian‑like binary sequence using an irrational rotation."""
    alpha = (np.sqrt(5) - 1.0) / 2.0   # golden ratio conjugate, irrational
    vals = (np.arange(n) * alpha) % 1.0

        h[flip] = 1
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] with ∫ h = 1 minimizing
    max_k ∫ h(x)(1‑h(x+k)) dx.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    # time left for the optimisation loop (95 % of the budget)
    end_time = start + budget_s * 0.95

    # discretisation – 200 points is a good trade‑off between speed and quality
    n = 200

    # initialise h
    try:
        h = balanced_binary_from_initial(initial_h_values, n)
    except NameError:
        # fall back to a low‑discrepancy irrational‑rotation start
        h = irr_rotation_binary(n)

    cur_max, cur_shift, _ = max_overlap_and_shift(h)
    best_h = h.copy()
    best_max = cur_max

    # target based on the required bound 0.3808
    target_max = int(math.floor(0.3808 * n / 2.0))

    # simulated‑annealing temperature schedule
    T0 = max(1.0, cur_max * 0.5)
    Tf = 0.001

    it = 0
    while time.time() < end_time:
        progress = (time.time() - start) / (budget_s * 0.95)
        T = T0 * (1.0 - progress) + Tf * progress

        # try a targeted swap that hurts the worst shift the least
        cand = targeted_swap_indices(h, cur_shift)
        if cand is not None:
            i, j = cand
        else:
            i, j = random_swap_indices(h)

        # perform the swap
        h[i] = 0
        h[j] = 1

        new_max, new_shift, _ = max_overlap_and_shift(h)
        delta = new_max - cur_max

        accept = False
        if delta <= 0:
            accept = True
        else:
            # Metropolis acceptance
            if np.random.rand() < math.exp(-delta / (T + 1e-12)):
                accept = True

        if accept:
            cur_max, cur_shift = new_max, new_shift
        else:
            # revert the change
            h[i] = 1
            h[j] = 0

        if cur_max < best_max:
            best_max = cur_max
            best_h = h.copy()
            if best_max <= target_max:
                break

        it += 1

    # final bound
    c5 = best_max * 2.0 / n
    return (best_h.astype(float), c5, n)
