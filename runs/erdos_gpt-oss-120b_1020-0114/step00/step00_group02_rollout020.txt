        s = x.sum()

        else:
            right = mid
    x = np.clip(v - (left+right)/2, 0, 1)
    # Because result sum may deviate due to floating errors, adjust small difference.
    diff = S - x.sum()
    # Adjust by adding diff to components not at bounds, using small epsilon.
    if abs(diff) > 1e-8:

        # find indices where 0 < x_i < 1

def project_onto_capped_simplex(v, total):
    # bisection lambda.
    n = v.shape[0]
    left = v.min() - 1.0
    right = v.max()
    for _ in range(50):
        mid = (left + right) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > total:
            left = mid
        else:
            right = mid
    x = np.clip(v - (left + right)/2.0, 0.0, 1.0)
    # adjust residual
    diff = total - x.sum()
    if abs(diff) > 1e-8:
        # distribute to interior
        interior = (x > 1e-12) & (x < 1 - 1e-12)
        if interior.any():
            x[interior] += diff / interior.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            # fallback: add diff to max component limited by 1
            idx = np.argmax(v)
            x[idx] = min(1.0, max(0.0, x[idx] + diff))
    return x
```

    grad = conv1[off:off+n] + conv2[off:off+n]
    return grad

import time
import numpy as np

def project_onto_capped_simplex(v: np.ndarray, total: float) -> np.ndarray:
    """Projection onto {x: 0<=x_i<=1 , sum x_i = total}."""
    n = v.shape[0]
    # Bisection on lambda so that sum_i clip(v_i - l,0,1) = total
    left = v.min() - 1.0
    right = v.max()
    for _ in range(60):
        mid = (left + right) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > total:
            left = mid
        else:
            right = mid
    x = np.clip(v - (left + right) * 0.5, 0.0, 1.0)
    # tiny correction to hit the sum exactly
    diff = total - x.sum()
    if abs(diff) > 1e-10:
        interior = (x > 1e-12) & (x < 1.0 - 1e-12)
        if interior.any():
            add = diff / interior.sum()
            x[interior] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            # fallback: adjust a single coordinate
            idx = np.argmax(v)
            x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
    return x


def compute_c5(h: np.ndarray) -> float:
    """Exact discretised C5 value for a given h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx


def softmin_gradient(h: np.ndarray, beta: float) -> np.ndarray:
    """Gradient of the soft‑minimum of the auto‑correlation."""
    n = h.shape[0]
    # full auto‑correlation (no dx factor)
    c = np.correlate(h, h, mode="full")
    maxc = c.max()
    # numerically stable soft‑min weights
    w = np.exp(-beta * (c - maxc))
    w /= w.sum()
    # convolution to obtain Σ w_s * (h_{i+s}+h_{i-s})
    conv1 = np.convolve(w, h, mode="full")
    conv2 = np.convolve(w[::-1], h, mode="full")
    off = n - 1
    grad = conv1[off:off + n] + conv2[off:off + n]
    return grad


def local_optimize(h_start: np.ndarray, deadline: float, rng: np.random.Generator) -> tuple:
    """Projected gradient ascent on the soft‑min objective."""
    n = h_start.shape[0]
    target_sum = n * 0.5
    h = h_start.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    # schedule of beta values (increasing smoothness)
    betas = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0]
    lr = 0.15
    while time.time() < deadline:
        for beta in betas:
            for _ in range(300):
                if time.time() >= deadline:
                    break
                grad = softmin_gradient(h, beta)
                step = lr * grad
                h_new = project_onto_capped_simplex(h + step, target_sum)
                val = compute_c5(h_new)
                if val < best_val - 1e-9:
                    best_val, best_h = val, h_new.copy()
                    h = h_new
                    lr = min(lr * 1.07, 0.4)
                else:
                    # occasional stochastic acceptance to escape plateaus
                    if rng.random() < 0.005:
                        h = h_new
                    lr = max(lr * 0.72, 5e-6)
                if lr < 5e-6:
                    break
            lr *= 0.68
    return best_h, best_val


def upsample(v: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of v (defined on [0,2]) to n points."""
    m = v.shape[0]
    x_old = np.linspace(0.0, 2.0, m, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    return np.interp(x_new, x_old, v)

    start = time.time()

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    attempts = 0
    while time.time() - start < budget_s * 0.97:
        attempts += 1
        # first attempt uses the up‑sampled parent, later attempts random
        if attempts == 1 and 'init0' in locals():
            h_start = h0.copy()
        else:
            rand_vec = rng.random(n)
            h_start = project_onto_capped_simplex(rand_vec, target_sum)

        # allocate a slice of remaining time (max 30 s) for this run
        remaining = budget_s - (time.time() - start)
        if remaining < 2.0:

        h_opt, val_opt = local_optimize(h_start, deadline, rng)
        if val_opt < best_val:
            best_val, best_h = val_opt, h_opt.copy()
            if best_val <= 0.3808:

    return best_h, best_val, n
