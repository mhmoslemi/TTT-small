    start = time.time()

    # Ensure sum = n/2 exactly; adjust by scaling difference.
    target_sum = n/2

    if abs(diff) > 1e-12:
        # add diff to random element respecting bounds
        # trivial adjust: h[0] = np.clip(h[0]+diff,0,1)
        h[0] = np.clip(h[0] + diff, 0, 1)
        # then rescale or adjust others if needed.

    # compute initial C
    def compute_c5(h):
        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c = compute_c5(best_h)
    # Keep current h as cur_h, cur_c
    cur_h = best_h.copy()
    cur_c = best_c

    # parameters
    n_iter = 0
    max_iter = 2000000  # large
    # we will sample pairs each iteration
    # step magnitude scaling maybe adapt
    while time.time() - start < budget_s * 0.95:
        # choose two indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # Determine max delta transfer from i to j
        hi = cur_h[i]
        hj = cur_h[j]
        max_delta = min(hi, 1 - hj)
        if max_delta <= 0:
            continue
        # Choose delta random uniform 0..max_delta
        delta = rng.random() * max_delta

                best_h = new_h.copy()
                best_c = new_c

import time, math
import numpy as np

def _c5_of(h: np.ndarray) -> float:
    """Return the C5 value for a step function sampled by h."""
    n = h.shape[0]
    dx = 2.0 / n
    # linear correlation of h with (1‑h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

def _project_to_sum_box(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector v onto the set {x : 0≤x≤1, sum(x)=target}.
    Solved by a simple bisection on the Lagrange multiplier λ
    for the operation x = clip(v‑λ,0,1).
    """
    lo, hi = -1e6, 1e6
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def _random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random feasible vector (values in [0,1], sum = n/2)."""
    # start with a random binary vector (exactly half ones)
    half = n // 2
    arr = np.zeros(n, dtype=np.float64)
    arr[:half] = 1.0
    rng.shuffle(arr)
    # add a small random perturbation and re‑project
    perturb = rng.random(n) - 0.5
    arr += perturb * 0.1
    arr = _project_to_sum_box(arr, n / 2.0)
    return arr

    best_c = cur_c

    # ------------------------------------------------------------------
    # 2️⃣  optimisation loop (simulated‑annealing style)
    # ------------------------------------------------------------------
    T0 = 0.008               # initial temperature
    T = T0
    iter_count = 0
    # we allocate ~90 % of the budget to the main loop
    time_limit = budget_s * 0.90

    while time.time() - start < time_limit:
        # pick two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        hi, hj = h[i], h[j]
        # feasible transfer range from i → j
        delta_max = min(hi, 1.0 - hj)
        delta_min = -min(hj, 1.0 - hi)
        if delta_max - delta_min <= 1e-12:
            continue

        # occasionally take a very small step (fine‑tuning)
        if rng.random() < 0.15:
            delta = delta_min + rng.random() * (delta_max - delta_min) * 0.1
        else:
            delta = delta_min + rng.random() * (delta_max - delta_min)

        if abs(delta) < 1e-12:
            continue

        # propose new vector
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta

        # evaluate
        new_c = _c5_of(h_new)

        # Metropolis criterion
        dC = new_c - cur_c
        accept = False
        if dC < -1e-12:                     # improvement
            accept = True
        else:
            prob = math.exp(-dC / (T + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            h = h_new
            cur_c = new_c
            if new_c < best_c - 1e-12:
                best_h = h_new.copy()
                best_c = new_c

        # temperature schedule (slow exponential decay)
        iter_count += 1
        if iter_count % 500 == 0:
            T = T0 * (0.9995 ** (iter_count // 500))

        # ------------------------------------------------------------------
        # occasional large perturbation to escape deep local minima
        # ------------------------------------------------------------------
        if iter_count % 4000 == 0 and rng.random() < 0.12:
            # shake the current vector a little and re‑project
            shake = (rng.random(n) - 0.5) * 0.3
            h_shaken = h + shake
            h_shaken = _project_to_sum_box(h_shaken, target_sum)
            c_shaken = _c5_of(h_shaken)
            if c_shaken < cur_c:
                h, cur_c = h_shaken, c_shaken
                if c_shaken < best_c:
                    best_h, best_c = h_shaken.copy(), c_shaken

        # ------------------------------------------------------------------
        # give chance to restart from a fresh random point once in a while
        # ------------------------------------------------------------------
        if iter_count % 25000 == 0 and rng.random() < 0.07:
            h = _random_feasible(n, rng)
            cur_c = _c5_of(h)
            if cur_c < best_c:
                best_h, best_c = h.copy(), cur_c

    # ------------------------------------------------------------------
    # 3️⃣  final sanity projection (eliminate accumulated drift)
    # ------------------------------------------------------------------
    best_h = _project_to_sum_box(best_h, target_sum)
    final_c = _c5_of(best_h)

    # ------------------------------------------------------------------
    # return as plain Python objects
    # ------------------------------------------------------------------
    return best_h.tolist(), float(final_c), int(n)
