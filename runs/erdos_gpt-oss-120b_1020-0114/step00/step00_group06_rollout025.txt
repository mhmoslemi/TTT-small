def compute_c5(h):
    N = h.shape[0]

    H = np.fft.fft(h)

import time
import numpy as np

# ------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ------------------------------------------------------------

def _compute_O_vector(h: np.ndarray) -> np.ndarray:
    """Circular correlation O(k) = ∫ h(x)*(1-h(x+k))dx for a step function.
    The interval length is 2, dx = 2/N."""
    N = h.shape[0]
    dx = 2.0 / N
    H = np.fft.fft(h)
    G = np.fft.fft(1.0 - h)
    corr = np.real(np.fft.ifft(H * np.conj(G)))   # circular cross‑correlation
    return corr * dx


def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap value C5 for a given h."""
    return _compute_O_vector(h).max()


def _project_sum_box(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto the box [0,1] while keeping sum(x)=target.
    Simple water‑filling iteration – fast enough for N≈100."""
    x = np.clip(x, 0.0, 1.0)
    for _ in range(10):
        s = x.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            cap = 1.0 - x
            tot = cap.sum()
            if tot == 0.0:
                break
            x += diff * cap / tot
        else:
            cap = x
            tot = cap.sum()
            if tot == 0.0:
                break
            x += diff * cap / tot
        x = np.clip(x, 0.0, 1.0)
    return x
