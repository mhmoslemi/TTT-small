```python
def propose_move(h, rng):
    N = h.shape[0]
    i, j = rng.integers(0, N, size=2)
    while j == i:
         j = rng.integers(0, N)
    hi = h[i]; hj = h[j]
    # compute feasible delta interval
    delta_min = -min(hi, 1.0 - hj)
    delta_max = min(1.0 - hi, hj)
    if delta_min >= delta_max: # no feasible move, return None
        return None
    delta = rng.uniform(delta_min, delta_max)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new, delta
```

```python
def simulated_annealing(h0, rng, budget_seconds):
    start_time = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c5, _ = compute_c5(h)
    cur_c5 = best_c5

    # Estimate step size for temperature
    # Collect a sample of delta changes
    samples = []
    for _ in range(100):
        move = propose_move(h, rng)
        if move is None:
            continue
        new_h, _ = move
        new_c5, _ = compute_c5(new_h)
        samples.append(new_c5 - cur_c5)
    if samples:
        avg_abs = np.mean(np.abs(samples))
    else:
        avg_abs = 1e-3
    T0 = avg_abs * 10  # starting temp
    Tmin = 1e-6
    total_iters = 0
    # Determine number of iterations based on budget: compute time per iteration and estimate target.
    # We'll just loop until time left < small threshold
    while True:
        # time check
        if time.time() - start_time > budget_seconds:
            break
        move = propose_move(h, rng)
        if move is None:
            continue
        new_h, _ = move
        new_c5, _ = compute_c5(new_h)
        delta = new_c5 - cur_c5
        # compute temperature based on schedule
        elapsed = time.time() - start_time
        fraction = elapsed / budget_seconds
        T = T0 * (1 - fraction) + Tmin * fraction  # linear decreasing
        if delta < 0 or rng.random() < math.exp(-delta / T):
            # accept
            h = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        total_iters += 1
        # optional early stop if best low enough
        if best_c5 <= 0.3808:
            break
    return best_h, best_c5, total_iters
```

```python
def upsample_h(h, M):
    N = len(h)
    # define original x positions (midpoints) as linspace(0,2,N,endpoint=False)+dx/2 maybe
    x = np.linspace(0, 2, N, endpoint=False) + 1.0/N  # actually step dx = 2/N; positions at step centers? Not important.
    x_new = np.linspace(0, 2, M, endpoint=False) + 1.0/M
    h_new = np.interp(x_new, x, h)
    # Clip to [0,1]
    h_new = np.clip(h_new, 0, 1)
    # Adjust sum to meet constraint
    target_sum = M/2
    cur_sum = h_new.sum()
    if cur_sum == 0:
        # set to mean
        h_new[:] = 0.5
    else:
        h_new *= target_sum / cur_sum
    # clip again
    h_new = np.clip(h_new, 0, 1)
    # after scaling, the sum might not exactly equal target due to clipping; we can adjust by adding small epsilon to random entries that are not at bounds.
    cur_sum = h_new.sum()
    diff = target_sum - cur_sum
    # distribute diff to entries where we can add/subtract while keeping bounds.
    if abs(diff) > 1e-12:
        # find adjustable indices
        if diff > 0:
            # need to increase some entries, find indices with h<1
            idx = np.where(h_new < 1)[0]
        else:
            idx = np.where(h_new > 0)[0]
        if idx.size > 0:
            # distribute uniformly
            inc = diff / idx.size
            # clip each adjustment to maintain bounds
            for i in idx:
                if diff > 0:
                    max_inc = 1 - h_new[i]
                    delta = min(inc, max_inc)
                else:
                    max_dec = h_new[i]
                    delta = max(inc, -max_dec)  # inc negative
                h_new[i] += delta
                diff -= delta
            # After loop, small diff may remain < tolerance, we can ignore
    # final clip
    h_new = np.clip(h_new, 0, 1)
    # adjust final sum if small error
    cur_sum = h_new.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        # Normalize scaling factor approx
        h_new *= target_sum / cur_sum
        h_new = np.clip(h_new, 0, 1)
    return h_new
```

import numpy as np, time, math

def _c5_value(h: np.ndarray) -> float:
    """Return C5 bound for a given step vector."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _normalize(h: np.ndarray) -> np.ndarray:
    """Scale/clamp vector to satisfy 0≤h≤1 and Σh = n/2 exactly."""
    n = h.shape[0]
    target = n / 2.0
    cur = h.sum()
    if cur == 0:
        h[:] = target / n
        return h
    h *= target / cur
    np.clip(h, 0.0, 1.0, out=h)
    cur = h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        mask = (h > 0.0) & (h < 1.0)
        idx = np.where(mask)[0]
        if idx.size == 0:
            # fallback: perturb all entries equally then re‑clip
            h += diff / n
            np.clip(h, 0.0, 1.0, out=h)
        else:
            inc = diff / idx.size
            for i in idx:
                if diff > 0:
                    delta = min(inc, 1.0 - h[i])
                else:
                    delta = max(inc, -h[i])
                h[i] += delta
                diff -= delta
                if abs(diff) < 1e-12:
                    break
            np.clip(h, 0.0, 1.0, out=h)
    # final tiny correction if still off
    cur = h.sum()
    if abs(cur - target) > 1e-9:
        h *= target / cur
        np.clip(h, 0.0, 1.0, out=h)
    return h


def _propose_move(h: np.ndarray, rng: np.random.Generator):
    """Generate a feasible pair‑wise mass‑transfer move preserving the integral."""
    n = h.shape[0]
    i = rng.integers(n)
    j = rng.integers(n - 1)
    if j >= i:
        j += 1
    hi, hj = h[i], h[j]
    delta_min = -min(hi, 1.0 - hj)
    delta_max = min(1.0 - hi, hj)
    if delta_max <= delta_min:
        return None
    delta = rng.uniform(delta_min, delta_max)
    hnew = h.copy()
    hnew[i] = hi + delta
    hnew[j] = hj - delta
    return hnew, delta


def _anneal(h0: np.ndarray, rng: np.random.Generator, budget: float):
    """Simulated‑annealing search for a low C5 value."""
    n = h0.shape[0]
    h = h0.copy()
    best_h = h.copy()
    best_val = _c5_value(best_h)
    cur_val = best_val
    start = time.time()

    # Get a rough scale of objective changes to set initial temperature.
    samples = []
    for _ in range(200):
        mv = _propose_move(h, rng)
        if mv is None:
            continue
        new_h, _ = mv
        samples.append(_c5_value(new_h) - cur_val)
    avg_abs = np.mean(np.abs(samples)) if samples else 1e-3
    T0 = avg_abs * 5.0
    Tmin = 1e-6

    while True:
        if time.time() - start >= budget:
            break
        mv = _propose_move(h, rng)
        if mv is None:
            continue
        new_h, _ = mv
        new_val = _c5_value(new_h)
        delta = new_val - cur_val
        elapsed = time.time() - start
        frac = elapsed / budget
        T = T0 * (1.0 - frac) + Tmin * frac
        accept = delta < 0 or (T > 0 and rng.random() < math.exp(-delta / T))
        if accept:
            h = new_h
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
                if best_val <= 0.3808:          # target reached → stop early
                    break
    return best_h, best_val
