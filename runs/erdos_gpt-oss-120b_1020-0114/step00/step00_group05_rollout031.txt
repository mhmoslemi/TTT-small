import time
import numpy as np

def _project_onto_box_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the set
        {x | 0 <= x_i <= 1, sum(x) = target_sum}
    via bisection on the Lagrange multiplier.
    """
    lo, hi = -1.0, 1.0                # bounds guarantee sum(lo) >= target, sum(hi) <= target
    for _ in range(30):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """evaluate C₅ for a step function h (discrete version)"""
    dx = 2.0 / h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _subgradient_step(h: np.ndarray, eta: float) -> (np.ndarray, float):
    """
    One sub‑gradient step for the objective max_k ∫ h(x)(1‑h(x+k))dx.
    Returns the projected new h and its C₅ value.
    """
    n = h.size
    # correlation and index of the worst shift
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_idx = int(np.argmax(corr))
    lag = k_idx - (n - 1)                     # lag >=0  ⇒ shift to the right

    # sub‑gradient of the active correlation term
    g = np.zeros_like(h)
    if lag >= 0:
        m = n - lag
        g[:m] = (1.0 - h[lag:])               # derivative from h_i * (1‑h_{i+lag})
        g[lag:] -= h[:m]                      # derivative from -(h_{i‑lag}) * (1‑h_i)
    else:
        k = -lag
        m = n - k
        g[k:] = (1.0 - h[:m])
        g[:m] -= h[k:]

    dx = 2.0 / n
    g *= dx                                   # scale by the integration step

    # gradient descent update and projection onto feasible set
    h_new = h - eta * g
    h_proj = _project_onto_box_simplex(h_new, n * 0.5)
    return h_proj, _c5_value(h_proj)


def _random_feasible(n: int) -> np.ndarray:
    """generate a random feasible step function (uniform points, then project)."""
    v = np.random.rand(n)
    return _project_onto_box_simplex(v, n * 0.5)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Find a step function h on [0,2] with integral 1 that minimizes C₅."""
    np.random.seed(seed)
    start_time = time.time()

    # number of discretisation points – use the parent size when available
    try:
        init_arr = np.asarray(initial_h_values, dtype=float)
        n_pts = init_arr.size
        h_start = _project_onto_box_simplex(init_arr, n_pts * 0.5)
    except Exception:
        n_pts = 200                     # reasonable default when no parent solution
        h_start = np.full(n_pts, 0.5)   # constant start (later perturbed)

    best_h = h_start.copy()
    best_c = _c5_value(best_h)

    # helper to run a descent from a given start vector
    def _run_descent(h0, max_iter):
        nonlocal best_h, best_c, start_time
        h = h0.copy()
        for it in range(1, max_iter + 1):
            # respect overall time budget
            if time.time() - start_time > budget_s * 0.98:
                break
            eta = 0.5 / np.sqrt(it)          # decreasing step size
            h, cur_c = _subgradient_step(h, eta)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
                # stop early if we have already beaten the target
                if best_c < 0.3808:
                    return True
        return False

    # 1) try the (possibly) provided construction first
    _run_descent(best_h, 3000)

    # 2) several random restarts with modest iterations each
    for _ in range(12):
        if time.time() - start_time > budget_s * 0.98:
            break
        h0 = _random_feasible(n_pts)
        if _run_descent(h0, 2500):
            break

    # final sanity check (clamp to bounds and re‑project just in case)
    best_h = _project_onto_box_simplex(best_h, n_pts * 0.5)
    best_c = _c5_value(best_h)

    return best_h.tolist(), float(best_c), int(n_pts)
