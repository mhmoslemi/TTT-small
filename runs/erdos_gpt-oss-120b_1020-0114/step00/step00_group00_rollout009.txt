def obj_grad(h):
    n = len(h)
    conv = np.correlate(h, 1 - h, mode='full')
    cmax = np.max(conv)
    z = conv - cmax
    exp_z = np.exp(GLOBAL_ALPHA * z)
    sum_exp_z = np.sum(exp_z)
    w = exp_z / sum_exp_z  # weights
    grad = np.ones(n)
    # subtract contributions
    for i, wk in enumerate(w):
        if wk == 0:
            continue
        k = i - (n - 1)
        if k >= 0:
            # overlap when j+k < n
            grad[: n - k] -= wk * h[k:]
            # and when j-k >=0
            grad[k:] -= wk * h[: n - k]
        else:
            kk = -k
            grad[kk:] -= wk * h[: n - kk]
            grad[: n - kk] -= wk * h[kk:]
    return grad

import time
import numpy as np
from scipy.optimize import minimize

# ----- global parameters (updated during optimisation) -----
GLOBAL_ALPHA = 1.0          # smoothing factor for the log‑sum‑exp surrogate
GLOBAL_TARGET = 0.0         # required sum of h ( = n/2 )

# ----- projection onto the box [0,1] and hyperplane sum = target -----
def project_to_feasible(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto {0≤x≤1, sum(x)=target}."""
    lo = -np.max(v) - 1.0    # guarantees x_i = 1 for all i when λ = lo
    hi = 1.0 - np.min(v) + 1.0  # guarantees x_i = 0 for all i when λ = hi
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)

def random_feasible(n: int, target: float) -> np.ndarray:
    """Random vector in [0,1]^n with prescribed sum."""
    v = np.random.rand(n)
    return project_to_feasible(v, target)

def upsample_h(h_small: np.ndarray, n_new: int) -> np.ndarray:
    """Linear interpolation of a step function to a finer grid."""
    n_old = h_small.size
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_small)

def compute_c5(h: np.ndarray) -> float:
    """Exact overlap bound for a given discretisation."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

# ----- surrogate objective (smooth max) and its gradient -----
def obj_fun(h: np.ndarray) -> float:
    """log‑sum‑exp surrogate of max_k ∑ h_i(1‑h_{i+k})  (un‑scaled)."""
    conv = np.correlate(h, 1.0 - h, mode='full')
    cmax = conv.max()
    z = conv - cmax                     # shift for numerical stability
    w = np.exp(GLOBAL_ALPHA * z)
    return cmax + np.log(w.sum()) / GLOBAL_ALPHA

def obj_grad(h: np.ndarray) -> np.ndarray:
    """Gradient of the surrogate with respect to h."""
    n = h.size
    conv = np.correlate(h, 1.0 - h, mode='full')
    cmax = conv.max()
    z = conv - cmax
    expz = np.exp(GLOBAL_ALPHA * z)
    w = expz / expz.sum()               # softmax weights, sum to 1
    grad = np.ones(n)                   # contribution of the constant term
    # subtract the two shifted‑h terms for each lag
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        k = idx - (n - 1)               # actual shift
        if k >= 0:
            grad[: n - k] -= wk * h[k:]          # h_{j+k}
            grad[k:]     -= wk * h[: n - k]      # h_{j‑k}
        else:
            kk = -k
            grad[kk:]    -= wk * h[: n - kk]     # h_{j+k}
            grad[: n - kk] -= wk * h[kk:]        # h_{j‑k}
    return grad

# ----- equality constraint (sum = target) -----
def eq_constraint_fun(h: np.ndarray) -> float:
    return h.sum() - GLOBAL_TARGET

def eq_constraint_jac(h: np.ndarray) -> np.ndarray:
    return np.ones_like(h)

# ----- optimisation wrapper for a single starting point -----
def optimise_one(start_h: np.ndarray, alphas, maxiter=120):
    """Run a sequence of SLSQP steps with increasing smoothing."""
    global GLOBAL_ALPHA, GLOBAL_TARGET
    h = start_h.copy()
    n = h.size
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_constraint_fun,
             'jac': eq_constraint_jac}]
    for a in alphas:
        GLOBAL_ALPHA = a
        res = minimize(obj_fun, h, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter,
                                'ftol': 1e-9,
                                'disp': False})
        h = res.x if res.success else res.x
        h = project_to_feasible(h, GLOBAL_TARGET)
    return h
