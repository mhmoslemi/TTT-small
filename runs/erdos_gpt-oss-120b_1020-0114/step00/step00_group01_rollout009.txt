def project_to_box_and_sum(x, target):
    # project x onto [0,1] with sum target using water-filling.
    # We need to find λ such that sum(clip(x - λ, 0, 1)) = target.

    for _ in range(50):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0, 1)
        total = s.sum()
        if total > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0, 1)

import time
import numpy as np

# ---------- helper utilities -------------------------------------------------
def _proj_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector v onto the intersection of the box [0,1]^n and the hyperplane
    sum(x) = target using a simple binary‑search on the offset λ where
    x_i = clip(v_i - λ, 0, 1).  The search interval [-2,2] always contains the
    correct λ for 0≤v_i≤1 and 0≤target≤len(v).
    """
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _worst_lag(h: np.ndarray):
    """
    Return the lag (as an integer index offset) that realises the maximal
    overlap integral  ∫ h(x)(1‑h(x+k))dx  (up to the discretisation used by
    np.correlate) together with the raw correlation value.
    """
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    n = h.shape[0]
    lag = idx - (n - 1)
    return lag, corr[idx]


# ---------- main optimisation ------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step‑function h on [0,2] (represented by n samples) that
    minimises C₅ = max_k ∫ h(x)(1‑h(x+k))dx.
    Returns a tuple (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # keep the best solution seen so far
    best_h = None          # numpy array
    best_val = np.inf

    # length of the provided parent (if any) – otherwise start with 100 points
    base_n = len(initial_h_values) if "initial_h_values" in globals() else 100

    # a few candidate sample sizes (all stay comfortably below 1 k)
    cand_ns = [base_n + d for d in (0, 20, 40, 60, 80) if base_n + d <= 300]

    # --------------------------------------------------------------
    def _evaluate(vec):
        """wrapper for the supplied sanity checker."""
        return evaluate_erdos_solution(vec)

    # --------------------------------------------------------------
    def _optimise_from(start_vec):
        """project‑gradient + simple simulated‑annealing refinement."""
        nonlocal best_h, best_val, start

        h = start_vec.astype(float).copy()
        n = h.size
        target = n / 2.0
        h = _proj_capped_simplex(h, target)

        cur_val = _evaluate(h)
        if cur_val < best_val:
            best_val, best_h = cur_val, h.copy()

        # ---- projected sub‑gradient descent on the worst lag -------------
        max_grad_iters = 2000
        it = 0
        while time.time() - start < budget_s * 0.70 and it < max_grad_iters:
            lag, _ = _worst_lag(h)

            # construct the two shifted copies (zero‑padded outside the interval)
            if lag >= 0:
                left  = np.concatenate([h[lag:], np.zeros(lag)])
                right = np.concatenate([np.zeros(lag), h[: n - lag]])
            else:
                L = -lag
                left  = np.concatenate([np.zeros(L), h[: n - L]])
                right = np.concatenate([h[L:], np.zeros(L)])

            # gradient of the active overlap functional
            grad = 1.0 - (left + right)

            # diminishing step size
            step = 0.5 / np.sqrt(it + 1.0)

            h_new = h - step * grad
            h_new = _proj_capped_simplex(h_new, target)

            new_val = _evaluate(h_new)
            # accept only if the bound improves
            if new_val < cur_val:
                h, cur_val = h_new, new_val
                if new_val < best_val:
                    best_val, best_h = new_val, h_new.copy()
            it += 1

        # ---- simple random pairwise swaps (SA‑style) --------------------
        while time.time() - start < budget_s * 0.95:
            i, j = rng.choice(n, size=2, replace=False)

            # feasible delta while keeping both entries inside [0,1]
            delta_lo = max(-h[i], h[j] - 1.0)
            delta_hi = min(1.0 - h[i], h[j])
            if delta_lo >= delta_hi:

            delta = rng.uniform(delta_lo, delta_hi)

            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta

            new_val = _evaluate(h_new)
            if new_val < cur_val:
                h, cur_val = h_new, new_val
                if new_val < best_val:
                    best_val, best_h = new_val, h_new.copy()

    # --------------------------------------------------------------
    # run a few independent restarts while time permits
    while time.time() - start < budget_s * 0.99:

        # initialise – reuse the supplied parent when sizes match

    # fallback – uniform function if everything failed
    if best_h is None:

        best_h = np.full(n, 0.5)
