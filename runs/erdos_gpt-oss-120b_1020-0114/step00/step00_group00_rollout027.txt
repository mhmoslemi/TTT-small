import numpy as np
import time
import random
import math

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Heuristic search for a step function h on [0,2] with
    ∫ h = 1 and 0≤h≤1 that minimizes
        C5 = max_k ∫ h(x) (1-h(x+k)) dx .
    Returns tuple (h_values, c5_bound, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)

    # ----- helper utilities -------------------------------------------------
    def compute_c5(h: np.ndarray) -> float:
        """Return C5 for a given discretised h."""
        n = h.shape[0]
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    def project_onto_box_and_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
        """
        Project vector v onto the set {x : 0≤x_i≤1 , sum x_i = target_sum}.
        Uses a simple bisection on a Lagrange multiplier λ.
        """
        lo = -np.max(v) - 1.0   # ensures all entries become 1 after clipping
        hi =  np.max(v) + 1.0   # ensures all entries become 0 after clipping
        for _ in range(50):
            mid = (lo + hi) * 0.5
            x = np.clip(v - mid, 0.0, 1.0)
            s = x.sum()
            if s > target_sum:
                lo = mid      # need larger λ → smaller sum
            else:
                hi = mid
        lam = (lo + hi) * 0.5
        return np.clip(v - lam, 0.0, 1.0)

    def gradient_wrt_shift(h: np.ndarray, d: int) -> np.ndarray:
        """
        Gradient of the overlap term for a fixed integer shift d
        (d may be negative).  Implements:
            g_i =  (1 - h_{i+d})   if i+d in range
                 - h_{i-d}         if i-d in range
        """
        n = h.shape[0]
        g = np.zeros_like(h)
        if d >= 0:
            nd = n - d
            if nd > 0:
                g[:nd] += 1.0 - h[d:]      # + part
                g[d:]  -= h[:nd]           # - part
        else:
            s = -d
            ns = n - s
            if ns > 0:
                g[s:]  += 1.0 - h[:ns]     # + part (i+d valid)
                g[:ns] -= h[s:]            # - part (i-d valid)
        return g

    # ----------------------------------------------------------------------
    # Choose a working discretisation size.
    # Prefer a reasonably fine grid but stay well within the time budget.
    n = 200                     # even, dx = 0.01
    target_sum = n * 0.5        # ∫ h = 1  →  sum(h) = n/2
    dx = 2.0 / n

    # ----- generate a set of starting candidates -------------------------
    candidates = []

    # 1) Use the provided construction if available (interpolate if needed)
    try:
        init = np.asarray(initial_h_values, dtype=np.float64)
        if init.ndim == 1:
            if init.shape[0] == n:
                candidates.append(init.copy())
            else:
                # linear interpolation to current grid
                x_old = np.linspace(0.0, 2.0, init.shape[0], endpoint=False) + dx/2
                x_new = np.arange(n) * dx + dx/2
                interp = np.interp(x_new, x_old, init)
                interp = project_onto_box_and_sum(interp, target_sum)
                candidates.append(interp)
    except NameError:
        pass  # no initial_h_values supplied

    # 2) Regular patterns with m equally spaced intervals (2 ≤ m ≤ 8)
    for m in range(2, 9):
        ell = 1.0 / m               # interval length
        G   = 1.0 / (m + 1)         # gap length (including both ends)
        h_pat = np.zeros(n, dtype=np.float64)
        for k in range(m):
            start = G + k * (ell + G)
            i0 = int(round(start / dx))
            length = int(round(ell / dx))
            i1 = min(n, i0 + length)
            h_pat[i0:i1] = 1.0
        h_pat = project_onto_box_and_sum(h_pat, target_sum)
        candidates.append(h_pat)

    # 3) Pure binary random patterns (exactly n/2 ones)
    for _ in range(5):
        hb = np.zeros(n, dtype=np.float64)
        idx = np.random.choice(n, n // 2, replace=False)
        hb[idx] = 1.0
        candidates.append(hb)

    # 4) Random continuous vectors projected onto the feasible set
    for _ in range(5):
        rv = np.random.rand(n)
        rv = project_onto_box_and_sum(rv, target_sum)
        candidates.append(rv)

    # ----------------------------------------------------------------------
    # Evaluate all candidates and keep the best
    best_h = None
    best_val = float('inf')
    for h0 in candidates:
        val = compute_c5(h0)
        if val < best_val:
            best_val = val
            best_h = h0.copy()

    # ----------------------------------------------------------------------
    # Simulated‑annealing / gradient descent refinement
    cur_h = best_h.copy()
    cur_val = best_val

    start_time = time.time()
    # Temperature and step‑size schedules
    temp = 0.02

    while time.time() - start_time < budget_s * 0.95:

        max_idx = int(np.argmax(corr))

        proposal = cur_h - step * grad

    cur_h = best_h.copy()

        if i == j:

            max_transfer = min(cur_h[j], 1.0 - cur_h[i])
            if max_transfer <= 0:

            delta = np.random.rand() * max_transfer
            cur_h[j] -= delta
            cur_h[i] += delta

            cur_val = new_val

                best_h = cur_h.copy()

            if delta is not None:

                cur_h[i] += delta if cur_h[i] < cur_h[j] else -delta
                cur_h[j] -= delta if cur_h[i] < cur_h[j] else -delta

    # Ensure feasibility of the final answer
    best_h = project_onto_box_and_sum(best_h, target_sum)
    c5_bound = compute_c5(best_h)

    return best_h, c5_bound, n
