def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    np.random.seed(seed)
    # determine n: if initial_h_values defined, else default 100
    try:
        init = np.asarray(initial_h_values, dtype=np.float64)
        n = init.shape[0]
    except NameError:
        n = 100
        init = None
    total_sum = n / 2.0
    if init is not None:
        h_cur = np.clip(init, 0.0, 1.0)
        h_cur = project_sum_bounds(h_cur, total_sum)
    else:
        # random feasible vector
        rand = np.random.rand(n)
        h_cur = project_sum_bounds(rand, total_sum)

    cur_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score

    no_improve_counter = 0
    max_no_improve = int(20000) # after which restart

    # main loop
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed > budget_s * 0.97:
            break
        # update step and temperature
        fraction = elapsed / budget_s
        step = max_step * (1.0 - fraction) + min_step * fraction
        T = T_start * (1.0 - fraction) + T_end * fraction

        # propose move
        i = np.random.randint(n)
        j = np.random.randint(n-1)
        if j >= i:
            j += 1
        delta = (np.random.rand() * 2.0 - 1.0) * step
        ti = h_cur[i] + delta
        uj = h_cur[j] - delta
        if ti < 0.0 or ti > 1.0 or uj < 0.0 or uj > 1.0:
            # reject move
            continue
        # create new candidate
        h_new = h_cur.copy()
        h_new[i] = ti
        h_new[j] = uj
        new_score = compute_c5(h_new)
        # acceptance
        if new_score < cur_score - 1e-14:
            accept = True
        else:
            # metropolis
            if T > 0:
                prob = math.exp(-(new_score - cur_score) / T)
                accept = np.random.rand() < prob
            else:
                accept = False
        if accept:
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_h = h_new.copy()
                best_score = new_score
                no_improve_counter = 0
            else:
                no_improve_counter += 1
        else:
            no_improve_counter += 1
        # restart if stagnation
        if no_improve_counter > max_no_improve:
            # restart from random feasible or best?
            # Keep best overall, start from random for exploring
            rand_vec = np.random.rand(n)
            h_cur = project_sum_bounds(rand_vec, total_sum)
            cur_score = compute_c5(h_cur)
            no_improve_counter = 0

    return best_h.tolist(), float(best_score), int(n)

import time
import math
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    # full correlation of h with (1‑h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    # dx = length(2) / n_points
    return corr.max() * (2.0 / h.size)


def project_to_sum_bounds(v: np.ndarray, total: float) -> np.ndarray:
    """
    Project vector v onto the hyperplane sum(v)=total with box constraints [0,1].
    Finds λ such that sum(clip(v+λ,0,1)) = total using bisection.
    """
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s < total:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Determine dimensionality.
    try:
        init_vec = np.asarray(initial_h_values, dtype=np.float64)
        n = init_vec.size
    except NameError:
        # If no parent vector is supplied, use a modest size.
        n = 80
        init_vec = None

    total_sum = n / 2.0  # because ∫h=1 ⇒ sum h_i = n/2

    # Initialise a feasible vector.
    if init_vec is not None:
        cur = np.clip(init_vec, 0.0, 1.0)
        cur = project_to_sum_bounds(cur, total_sum)
    else:
        cur = project_to_sum_bounds(np.random.rand(n), total_sum)

    cur_score = compute_c5(cur)
    best_h = cur.copy()
    best_score = cur_score

    # Parameters for the stochastic search.
    max_step = 0.20
    min_step = 0.001
    T_start = 1.0e-3
    T_end = 1.0e-6
    guided_prob = 0.30                     # fraction of iterations using a guided move
    max_no_improve = 20000                # restart after this many non‑improving tries
    no_improve = 0

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.97:
            break

        # linearly decreasing step size and temperature
        frac = elapsed / budget_s
        step = max_step * (1.0 - frac) + min_step * frac
        temperature = T_start * (1.0 - frac) + T_end * frac

        # decide move type
        if np.random.rand() < guided_prob:
            # ----- guided move towards improving the worst lag -----
            # compute current correlation to locate worst lag
            corr_full = np.correlate(cur, 1.0 - cur, mode="full")
            worst_idx = int(np.argmax(corr_full))
            lag = worst_idx - (n - 1)           # lag can be negative

            # compute gradient‑like neighbour sums for this lag
            g = np.zeros(n, dtype=np.float64)
            if lag >= 0:
                # forward neighbour
                g[: n - lag] += cur[lag:]
                # backward neighbour
                g[lag:] += cur[: n - lag]
            else:
                l = -lag
                # forward neighbour
                g[l:] += cur[: n - l]
                # backward neighbour
                g[: n - l] += cur[l:]

            # choose donor (low g, positive h) and receiver (high g, room to grow)
            donor_mask = cur > 1e-12
            receiver_mask = cur < 1.0 - 1e-12
            if not donor_mask.any() or not receiver_mask.any():
                # fall back to random move if no feasible pair
                i, j = np.random.randint(n), np.random.randint(n - 1)
                if j >= i:
                    j += 1
                delta = (np.random.rand() * 2.0 - 1.0) * step
                ti = cur[i] + delta
                tj = cur[j] - delta
                if 0.0 <= ti <= 1.0 and 0.0 <= tj <= 1.0:
                    cand = cur.copy()
                    cand[i], cand[j] = ti, tj
                else:
                    continue
            else:
                # pick extremes
                i_max = np.argmax(g * receiver_mask)   # best to receive mass
                i_min = np.argmin(g * donor_mask)      # best to give mass
                # ensure they are different
                if i_max == i_min:
                    # small random tweak if they coincide
                    i_max = np.random.choice(np.where(receiver_mask)[0])
                    i_min = np.random.choice(np.where(donor_mask)[0])

                max_possible = min(step, 1.0 - cur[i_max])
                min_possible = min(step, cur[i_min])
                delta = np.random.rand() * min(max_possible, min_possible)
                # move delta from i_min to i_max
                cand = cur.copy()
                cand[i_max] += delta
                cand[i_min] -= delta
        else:
            # ----- simple two‑index random move -----
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1
            delta = (np.random.rand() * 2.0 - 1.0) * step
            ti = cur[i] + delta
            tj = cur[j] - delta
            if not (0.0 <= ti <= 1.0 and 0.0 <= tj <= 1.0):
                continue
            cand = cur.copy()
            cand[i], cand[j] = ti, tj

        # evaluate candidate
        cand_score = compute_c5(cand)

        # Metropolis acceptance
        accept = False
        if cand_score < cur_score - 1e-14:
            accept = True
        else:
            if temperature > 0.0:
                prob = math.exp(-(cand_score - cur_score) / temperature)
                accept = np.random.rand() < prob

        if accept:
            cur = cand
            cur_score = cand_score
            no_improve = 0
            if cand_score < best_score - 1e-12:
                best_h = cand.copy()
                best_score = cand_score
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                # restart from a fresh random feasible point
                cur = project_to_sum_bounds(np.random.rand(n), total_sum)
                cur_score = compute_c5(cur)
                no_improve = 0

    # final safety projection (should already be feasible)
    best_h = project_to_sum_bounds(best_h, total_sum)
    # ensure exact sum within tiny tolerance
    diff = total_sum - best_h.sum()
    if abs(diff) > 1e-12:
        # distribute the tiny remainder uniformly while staying in bounds
        for idx in range(n):
            add = min(1.0 - best_h[idx], max(0.0, diff))
            best_h[idx] += add
            diff -= add
            if abs(diff) < 1e-12:
                break

    return best_h.tolist(), float(best_score), int(n)
