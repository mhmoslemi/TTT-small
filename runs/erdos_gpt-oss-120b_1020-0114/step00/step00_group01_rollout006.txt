import numpy as np, time

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def hill_climb_binary(n, rng, time_limit):
    # returns best_h and best_c found within time_limit seconds
    half = n // 2
    # start random binary
    h = np.zeros(n, dtype=float)
    h[:half] = 1.0
    rng.shuffle(h)
    best_h_local = h.copy()
    best_c_local = compute_c5(h)
    cur_c = best_c_local
    # lists
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    start_local = time.time()
    while time.time() - start_local < time_limit:
        # propose swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i], h[j] = h[j], h[i]
        new_c = compute_c5(h)
        # acceptance using simulated annealing derived from global time proportion
        # get temperature based on global start maybe not needed; just accept if better
        if new_c <= cur_c or rng.random() < 0.001: # occasional accept

import time
import numpy as np

def _c5_value(h: np.ndarray) -> float:
    """Return the C5 value for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _binary_hill_climb(n: int, rng: np.random.Generator, time_limit: float) -> tuple:
    """
    Simple simulated–annealing style hill‑climb on binary vectors of length n.
    Returns the best vector found within the allotted time.
    """
    half = n // 2
    # start from a random binary pattern with exactly half ones
    h = np.zeros(n, dtype=float)
    h[:half] = 1.0
    rng.shuffle(h)

    cur_c = _c5_value(h)
    best_h = h.copy()
    best_c = cur_c

    # maintain index lists for fast sampling
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])

    start = time.time()
    while time.time() - start < time_limit:
        # propose a swap
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # perform swap
        h[i], h[j] = h[j], h[i]
        new_c = _c5_value(h)

        # simple annealing schedule based on global remaining time
        # (temperature drifts to zero, encouraging greedy behaviour)
        t_elapsed = time.time() - start
        temp = max(1e-6, 0.02 * (1.0 - t_elapsed / time_limit))

        accept = False
        if new_c <= cur_c:
            accept = True
        else:
            # Metropolis acceptance
            if rng.random() < np.exp(-(new_c - cur_c) / temp):
                accept = True

        if accept:
            cur_c = new_c
            # update index bookkeeping
            ones.remove(i)
            zeros.append(i)
            zeros.remove(j)
            ones.append(j)

            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]

    return best_h, best_c
