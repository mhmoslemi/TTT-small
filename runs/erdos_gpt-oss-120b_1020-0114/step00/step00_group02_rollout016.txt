def make_move(h, step):
    n = len(h)
    i = np.random.randint(n)
    # Ensure source > 0
    while h[i] <= 0:
        i = np.random.randint(n)
    j = np.random.randint(n)
    while h[j] >= 1:
        j = np.random.randint(n)
    max_delta = min(step, h[i], 1 - h[j])

import time
import math
import numpy as np

def _c5_of(h):
    """Return the C5 value for a given vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function with a small C₅ value.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values : np.ndarray, shape (n_points,)
        c5_bound : float, the max overlap integral for this h
        n_points : int, the discretisation size
    """
    start_time = time.time()
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Initialise h
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.asarray(initial_h_values, dtype=np.float64).copy()
        n = h.shape[0]
    else:
        n = 60                     # a modest even size when no parent is given
        h = np.full(n, 0.5, dtype=np.float64)

    # Enforce the integral constraint exactly: sum(h) = n/2
    target_sum = n * 0.5
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # simple scaling; values stay in [0,1] because target_sum/n = 0.5
        h *= target_sum / cur_sum
        # clip just in case numerical overshoot
        h = np.clip(h, 0.0, 1.0)

    # ------------------------------------------------------------------
    # SA / greedy descent loop
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_c5 = _c5_of(best_h)

    cur_h = h.copy()
    cur_c5 = best_c5

    # Annealing schedule parameters
    T_initial = 0.006
    step_initial = 0.09
    # loop until time budget is almost exhausted
    time_limit = start_time + budget_s * 0.95

    while time.time() < time_limit:
        # ----------- compute worst shift and its gradient ----------------
        corr_full = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        idx_max = int(np.argmax(corr_full))
        k_shift = idx_max - (n - 1)               # shift value (may be negative)

        # gradient of the worst shift (piecewise linear)
        g = np.zeros_like(cur_h)

        if k_shift >= 0:
            # overlapping indices i = 0 .. n-1-k_shift
            g[: n - k_shift] += 1.0 - cur_h[k_shift:]
            # indices that appear as shifted term
            g[k_shift:] -= cur_h[: n - k_shift]
        else:  # k_shift < 0
            kk = -k_shift
            # overlapping indices i = kk .. n-1
            g[kk:] += 1.0 - cur_h[: n - kk]
            # shifted term contributions
            g[: n - kk] -= cur_h[kk:]

        # ----------- choose a transfer direction -------------------------
        # Prefer moving mass from a position where increasing h raises the
        # worst overlap (positive gradient) to a position where it lowers it
        # (negative gradient).
        pos = np.where(g > 1e-14)[0]
        neg = np.where(g < -1e-14)[0]

        if pos.size and neg.size:
            i = pos[np.argmax(g[pos])]     # largest positive gradient
            j = neg[np.argmin(g[neg])]     # most negative gradient
        else:
            # fall back to random feasible pair
            src = np.where(cur_h > 1e-12)[0]
            tgt = np.where(cur_h < 1.0 - 1e-12)[0]
            if src.size == 0 or tgt.size == 0:
                break
            i = np.random.choice(src)
            j = np.random.choice(tgt)

        # ----------- step size (decreasing with time) -------------------
        elapsed = time.time() - start_time
        progress = elapsed / (budget_s * 0.95)
        temperature = max(1e-6, T_initial * (1.0 - progress))
        step = max(1e-5, step_initial * (1.0 - progress))

        # feasible delta respecting bounds and the sum constraint
        max_delta = min(step, cur_h[i], 1.0 - cur_h[j])

        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
