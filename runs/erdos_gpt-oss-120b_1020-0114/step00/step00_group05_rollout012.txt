def solve_binary(N):
    x = cp.Variable(N, boolean=True)
    # sum constraint
    constraints = [cp.sum(x) == N//2]
    # create variable m (real scalar) for max diffs
    m = cp.Variable()
    constraints += [m >= 0]

    # For each shift k from 0 to N-1:
    for k in range(N):
        # overlapping indices
        indices_i = np.arange(N - k)
        i = indices_i
        j = i + k

        # define y_ik as binary variable product
        # Use auxiliary var y_ik for each position
        # compute sum of x[i] * x[j] as sum of y_ik
        y_ik = cp.Variable(N - k, boolean=True)
        # product constraints:
        constraints += [y_ik <= x[i]]
        constraints += [y_ik <= x[j]]
        constraints += [y_ik >= x[i] + x[j] - 1]

        overlap = cp.sum(y_ik)
        # constraint m >= (N/2 - overlap)
        constraints += [m >= (N/2) - overlap]

    # objective minimize m
    obj = cp.Minimize(m)

    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.CBC, verbose=False, max_iters=500000, # optional time limit
               # might set time limit param? Not sure; but may rely on runtime budget.
              )

    return x.value, m.value

def _project_capped_simplex(v, target):
    """Project v onto {0<=x_i<=1, sum(x)=target}."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # tiny correction to hit the target exactly
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        # distribute the remainder on non‑boundary entries
        free = np.where((w > 0) & (w < 1))[0]
        if free.size:
            inc = diff / free.size
            w[free] = np.clip(w[free] + inc, 0.0, 1.0)
    return w
