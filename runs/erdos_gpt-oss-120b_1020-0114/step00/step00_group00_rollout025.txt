Now we need to implement `project_to_feasible_continuous` to adjust sum while staying within [0,1].

One standard algorithm: we want vector x such that 0 <= x_i <= 1 and sum_i x_i = S.

We can apply "project onto simplex with upper bound" algorithm: start with x = clip(original,0,1). Let S_rem = S - sum(x). If S_rem > 0, we need to add to some entries not at 1; we can do water-filling: sort indexes where x_i < 1, find amount to add to each uniformly until hitting 1 limit or meeting S_rem. Similarly, if S_rem < 0, we need to subtract from entries >0.

Implementation details:

```
def project_to_feasible_continuous(x, target_sum, eps=1e-12):
    x = np.clip(x, 0.0, 1.0)
    total = x.sum()
    if abs(total - target_sum) <= eps:
        return x
    # if need to increase sum
    if total < target_sum:
        # need to distribute delta among those <1
        deficit = target_sum - total
        # indices not at upper bound
        mask = x < 1 - eps
        while deficit > eps and mask.any():
            # compute how much each can take
            can_add = 1 - x[mask]
            # distribute equally (or proportionally)
            add = deficit / mask.sum()
            # limit to can_add per entry
            inc = np.minimum(add, can_add)
            x[mask] += inc
            deficit -= inc.sum()
            mask = x < 1 - eps
        # if still deficit due to numeric, adjust small amount
    else:
        # total > target sum, need to reduce
        excess = total - target_sum
        mask = x > eps
        while excess > eps and mask.any():
            can_sub = x[mask]
            sub = np.minimum(excess / mask.sum(), can_sub)
            x[mask] -= sub
            excess -= sub.sum()
            mask = x > eps
    # if small residual, adjust last element
    diff = x.sum() - target_sum
    if abs(diff) > eps:
        # adjust a non-boundary element
        idxs = np.where((x > 0+eps) & (x < 1-eps))[0]
        if idxs.size > 0:
            x[idxs[0]] -= diff

def project_to_feasible(x, target_sum, max_iter=100):
    # Clip
    y = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        s = y.sum()
        if abs(s - target_sum) <= 1e-12:
            break
        # compute set of indices where y is strictly between 0 and 1
        free = (y > 1e-12) & (y < 1 - 1e-12)
        n_free = free.sum()
        if n_free == 0:  # all at bounds, cannot adjust
            # distribute remaining diff proportionally?
            # just adjust one arbitrary element inside [0,1] (if any)
            diff = target_sum - s
            # find any index and adjust within bounds
            idx = np.argmin(np.abs(y - 0.5))
            y[idx] = np.clip(y[idx] + diff, 0.0, 1.0)
            break
        delta = (target_sum - s) / n_free
        y[free] += delta
        # clip again
        y = np.clip(y, 0.0, 1.0)
    # final adjustment: ensure sum exactly via small tweak
    diff = target_sum - y.sum()
    if abs(diff) > 1e-10:
        # Adjust a random free index
        idxs = np.where((y > 0) & (y < 1))[0]
        if idxs.size == 0:
            idxs = np.arange(len(y))
        idx = np.random.choice(idxs)
        y[idx] = np.clip(y[idx] + diff, 0.0, 1.0)
    return y
```

def local_search_sa(h, dx, time_budget):
    import numpy as np, math, random, time
    n = len(h)
    cur_h = h.copy()
    cur_c5 = compute_c5_vec(cur_h, dx)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    start = time.time()
    # schedule parameters
    T0 = 0.05
    step_max = 0.3
    # Precompute indices of ones? Not needed for continuous.
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = max(T0 * (1 - frac), 1e-6)
        step = max(step_max * (1 - frac), 0.001)
        # pick i and j distinct
        i, j = random.sample(range(n), 2)
        # choose direction randomly
        if random.random() < 0.5:
            # increase i, decrease j
            max_inc = 1.0 - cur_h[i]
            max_dec = cur_h[j]
            if max_inc <= 0 or max_dec <= 0:
                continue
            delta = step * random.random()
            delta = min(delta, max_inc, max_dec)
            if delta <= 0:
                continue
            cur_h[i] += delta
            cur_h[j] -= delta
            moved = (i, j, delta, 1)  # direction 1
        else:
            max_inc = 1.0 - cur_h[j]
            max_dec = cur_h[i]
            if max_inc <= 0 or max_dec <= 0:
                continue
            delta = step * random.random()
            delta = min(delta, max_inc, max_dec)
            if delta <= 0:
                continue
            cur_h[j] += delta
            cur_h[i] -= delta
            moved = (j, i, delta, -1)
        new_c5 = compute_c5_vec(cur_h, dx)
        accept = False
        if new_c5 < cur_c5:
            accept = True
        else:
            if random.random() < math.exp(-(new_c5 - cur_c5) / T):
                accept = True
        if accept:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            # revert
            a, b, delta, dir_sign = moved
            cur_h[a] -= delta
            cur_h[b] += delta
        # occasional random large jump to explore
        if random.random() < 0.001:
            # random perturbation of a few entries
            for _ in range(3):
                i2, j2 = random.sample(range(n), 2)
                # arbitrarily set to random val within bounds while preserving sum:
                # take some amount from j2 to i2:
                amount = step * random.random()
                delta = min(amount, 1.0 - cur_h[i2], cur_h[j2])
                cur_h[i2] += delta
                cur_h[j2] -= delta
    return best_h, best_c5

def project_to_feasible(x, target):
    y = np.clip(x, 0., 1.)
    for _ in range(10):
        s = y.sum()
        diff = target - s
        if abs(diff) <= 1e-12:
            break
        free = ((y > 0) & (y < 1))
        cnt = free.sum()
        if cnt == 0:
            # distribute to all entries within bounds whichever possible
            # just adjust the first element
            y[0] = np.clip(y[0] + diff, 0., 1.)
            break
        delta = diff / cnt
        y[free] += delta
        # clip again
        np.clip(y, 0., 1., out=y)
    # final fix small diff
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        # adjust a random free slot
        idxs = np.where((y > 0) & (y < 1))[0]
        if idxs.size == 0:
            idxs = np.arange(len(y))
        idx = np.random.choice(idxs)
        y[idx] = np.clip(y[idx] + diff, 0., 1.)
    return y

import time, random, math
import numpy as np

def compute_c5(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx

def project_to_feasible(x, target_sum, max_iter=12):
    """Project x onto [0,1]^n ∩ {sum = target_sum}."""
    y = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        s = y.sum()
        diff = target_sum - s
        if abs(diff) <= 1e-12:
            break
        free = (y > 0.0) & (y < 1.0)
        cnt = free.sum()
        if cnt == 0:
            # adjust a single coordinate if everything is at a bound
            idx = np.argmin(np.abs(y - 0.5))
            y[idx] = np.clip(y[idx] + diff, 0.0, 1.0)
            break
        delta = diff / cnt
        y[free] += delta
        np.clip(y, 0.0, 1.0, out=y)
    # final tiny tweak
    diff = target_sum - y.sum()
    if abs(diff) > 1e-12:
        idxs = np.where((y > 0.0) & (y < 1.0))[0]
        if idxs.size == 0:
            idxs = np.arange(y.shape[0])
        i = np.random.choice(idxs)
        y[i] = np.clip(y[i] + diff, 0.0, 1.0)
    return y

def interpolate_h(old_h, new_n):
    """Linear interpolation of an existing h to a new length, then project."""
    old_n = old_h.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    nh = np.interp(new_x, old_x, old_h)
    return project_to_feasible(nh, new_n / 2.0)

def local_search_sa(h0, time_budget):
    """Simulated‑annealing style pairwise‑delta search on continuous h."""
    n = h0.shape[0]
    h = h0.copy()
    cur = compute_c5(h)
    best_h, best = h.copy(), cur
    start = time.time()
    T0, step0 = 0.04, 0.5
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = max(T0 * (1 - frac), 1e-8)
        step = max(step0 * (1 - frac), 0.001)

        i, j = random.sample(range(n), 2)
        if random.random() < 0.5:
            max_inc = 1.0 - h[i]
            max_dec = h[j]
            if max_inc <= 0 or max_dec <= 0:
                continue
            d = step * random.random()
            d = min(d, max_inc, max_dec)
            if d <= 0:
                continue
            h[i] += d
            h[j] -= d
            moved = (i, j, d, 1)
        else:
            max_inc = 1.0 - h[j]
            max_dec = h[i]
            if max_inc <= 0 or max_dec <= 0:
                continue
            d = step * random.random()
            d = min(d, max_inc, max_dec)
            if d <= 0:
                continue
            h[j] += d
            h[i] -= d
            moved = (j, i, d, -1)

        new = compute_c5(h)
        if new < cur or random.random() < math.exp(-(new - cur) / T):
            cur = new
            if new < best:
                best, best_h = new, h.copy()
        else:
            a, b, d, _ = moved
            h[a] -= d
            h[b] += d
    return best_h, best

def binary_local_search_sa(h0, time_budget):
    """SA on binary vectors (0/1) preserving exactly n/2 ones."""
    n = h0.shape[0]
    h = h0.copy()
    cur = compute_c5(h)
    best_h, best = h.copy(), cur
    start = time.time()

            cur = new
            if new < best:
                best, best_h = new, h.copy()
        else:
            h[i], h[j] = 1, 0
    return best_h, best
