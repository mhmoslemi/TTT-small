def max_overlap(h):
    n = h.shape[0]
    dx = 2.0 / n
    a = h.astype(np.float64)
    rev = (1.0 - a)[::-1]
    L = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(rev, n=L)
    conv = np.fft.irfft(fa * fb, n=L)
    # conv length L, linear part first 2n-1 entries
    conv = conv[:2*n-1]
    # Note conv possibly slight negative due to numerical errors; use max of conv
    return np.max(conv) * dx

        # update temperature gradually

import time, math
import numpy as np

# ------------------------------------------------------------
# Helper: fast evaluation of the Erdős overlap C5 for a step
# function represented by a numpy array h on [0,2].
# ------------------------------------------------------------
def _max_overlap(h: np.ndarray) -> float:
    """Return max_{k} ∫ h(x)(1-h(x+k)) dx using FFT based correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlate h with (1-h) : linear correlation = conv(h, rev(1-h))
    a = h.astype(np.float64, copy=False)
    b = (1.0 - a)[::-1]
    L = 1 << ((2 * n - 1).bit_length())        # next power of two
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)
    conv = conv[: 2 * n - 1]                    # linear part only
    return np.max(conv) * dx


# ------------------------------------------------------------
# Produce a family of analytic shapes:  h(x) = sin^p(πx/2)
# scaled so that the integral equals 1 (i.e. sum = n/2).
# ------------------------------------------------------------
def _analytic_shape(n: int, p: float) -> np.ndarray:
    """Return a step‑function of length n based on sin^p, already mass‑normalised."""
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx                 # mid‑points of sub‑intervals
    if p == 0.0:                                   # constant 0.5 exactly satisfies the constraints
        h = np.full(n, 0.5, dtype=np.float64)
        return h
    h_raw = np.sin(np.pi * xs / 2.0) ** p
    target = n / 2.0
    scaling = target / h_raw.sum()
    h = h_raw * scaling
    # clipping never needed for p∈[0,2] but keep for safety
    h = np.clip(h, 0.0, 1.0)
    # tiny uniform correction to hit the mass constraint exactly
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


# ------------------------------------------------------------
# Simple simulated‑annealing on the feasible simplex.
# A move exchanges a small amount δ between two coordinates.
# ------------------------------------------------------------
def _anneal_continuous(h: np.ndarray,
                       max_iter: int,
                       budget_end: float,
                       rng: np.random.Generator) -> np.ndarray:
    n = h.shape[0]
    best = h.copy()
    best_val = _max_overlap(best)
    cur = h.copy()
    cur_val = best_val

    T = 0.3                 # initial temperature
    Tmin = 1e-5
    eps = 0.25              # maximal magnitude of a transferred chunk

    for it in range(max_iter):
        if time.time() >= budget_end:
            break

        i, j = rng.integers(n, size=2)
        if i == j:
            continue

        # propose a transfer δ from j to i
        delta = (rng.random() - 0.5) * 2.0 * eps
        hi = cur[i] + delta
        hj = cur[j] - delta
        if not (0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0):
            continue

        cand = cur.copy()
        cand[i] = hi
        cand[j] = hj
        cand_val = _max_overlap(cand)

        # Metropolis acceptance
        if cand_val < cur_val or rng.random() < math.exp((cur_val - cand_val) / T):
            cur = cand
            cur_val = cand_val
            if cand_val < best_val:
                best = cand.copy()
                best_val = cand_val

        # cooling
        T = max(Tmin, T * 0.9999)
        if (it + 1) % 3000 == 0:
            eps = max(eps * 0.9, 1e-3)

    return best


# ------------------------------------------------------------
# Main entry point required by the harness.
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Scan a modest set of analytic shapes (different exponents, sizes)
    # ------------------------------------------------------------------
    best_h = None
    best_val = float('inf')
    best_n = None

    n_candidates = [80, 100, 120, 140, 160, 180, 200, 250]
    exponents = [0.0, 0.5, 1.0, 1.5, 2.0]

    for n in n_candidates:
        if time.time() - start_time > 0.85 * budget_s:
            break
        for p in exponents:
            h = _analytic_shape(n, p)
            val = _max_overlap(h)
            if val < best_val:
                best_val = val
                best_h = h.copy()
                best_n = n

    # ------------------------------------------------------------------
    # 2) Refine the best analytic candidate by a short annealing run.
    # ------------------------------------------------------------------
    if best_h is not None:
        # allocate a fraction of the remaining budget for local search
        remaining = budget_s - (time.time() - start_time)
        # allow up to 30 000 proposals (or less if time runs out)
        max_iter = min(30000, int(remaining * 20))   # crude 20 iters per sec estimate
        budget_end = start_time + budget_s * 0.99   # keep a tiny slack
        refined = _anneal_continuous(best_h, max_iter, budget_end, rng)
        refined_val = _max_overlap(refined)
        if refined_val < best_val:
            best_h, best_val, best_n = refined, refined_val, best_h.shape[0]

    # ------------------------------------------------------------------
    # 3) Final clean‑up: enforce the exact mass constraint.
    # ------------------------------------------------------------------
    if best_h is None:                     # safety fallback
        raise RuntimeError("Failed to construct a feasible h.")
    n = best_h.shape[0]
    target = n / 2.0
    current = best_h.sum()
    diff = target - current
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
