import numpy as np, time, math

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def max_lag_info(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    lag = idx - (n - 1)
    max_val = corr[idx]
    return lag, max_val, corr

def mismatched_indices(h, lag):
    n = h.shape[0]
    if lag >= 0:
        left = h[:n - lag]
        right = h[lag:]
        i = np.where((right == 1) & (left == 0))[0] + lag
        j = np.where((right == 0) & (left == 1))[0] + lag
    else:
        L = -lag
        left = h[L:]
        right = h[:n - L]
        i = np.where((right == 1) & (left == 0))[0]
        j = np.where((right == 0) & (left == 1))[0]
    return i.astype(int), j.astype(int)

def random_swap(h, rng):
    n = h.shape[0]
    ones = np.flatnonzero(h > 0.5)
    zeros = np.flatnonzero(h < 0.5)
    if len(ones) == 0 or len(zeros) == 0:
        return h.copy()
    i = rng.choice(ones)
    j = rng.choice(zeros)
    h2 = h.copy()
    h2[i] = 0.0
    h2[j] = 1.0
    return h2

def targeted_swap(h, lag, rng):
    i_idxs, j_idxs = mismatched_indices(h, lag)
    if len(i_idxs) == 0 or len(j_idxs) == 0:
        return None
    i = rng.choice(i_idxs)
    j = rng.choice(j_idxs)
    h2 = h.copy()
    h2[i] = 0.0
    h2[j] = 1.0
    return h2

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # Initial vector
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=np.float64)
        # enforce binary 0/1 and sum constraint
        n = init.shape[0]
        target = n / 2.0
        # If values already binary and sum correct, keep
        if np.all((init == 0) | (init == 1)):
            cur = init.copy()
            s = cur.sum()
            if abs(s - target) > 1e-9:
                # adjust by swapping to achieve exact half ones
                cur[:] = 0.0
                cur[:int(target)] = 1.0
                rng.shuffle(cur)
        else:
            # project to binary with correct sum
            cur = np.zeros_like(init)
            idx = np.argsort(-init)
            cur[idx[:int(round(target))]] = 1.0
    else:
        n = 200
        target = n / 2.0
        cur = np.zeros(n, dtype=np.float64)
        cur[:int(target)] = 1.0
        rng.shuffle(cur)

    best_h = cur.copy()
    best_score = compute_c5(best_h)

    # Main optimization loop
    cur_score = best_score
    it = 0
    # temperature schedule
    T0 = 0.2
    T = T0
    # exponential decay factor
    decay = 0.9999
    # set of parameters
    while time.time() - start < budget_s * 0.95:
        # decide move type: give higher chance to targeted
        if rng.random() < 0.7:
            # targeted
            lag, _, _ = max_lag_info(cur)
            cand = targeted_swap(cur, lag, rng)
            if cand is None:
                # fallback random
                cand = random_swap(cur, rng)
        else:
            cand = random_swap(cur, rng)

        cand_score = compute_c5(cand)
        # Acceptance
        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_h = cand.copy()
        # temperature update
        T *= decay
        it += 1

    # Additional local refinement: greedy swaps to reduce best_score further
    # try deterministic swaps focusing on worst lag
    for _ in range(2000):
        lag, _, _ = max_lag_info(best_h)
        cand = targeted_swap(best_h, lag, rng)
        if cand is None:
            break
        cand_score = compute_c5(cand)
        if cand_score < best_score:
            best_h = cand
            best_score = cand_score
        else:
            break

    return (best_h.tolist(), float(best_score), int(best_h.shape[0]))

import numpy as np
import time
import math

def _c5_val(h):
    """Return C5 for a binary (or [0,1]) vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    # Correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def _max_lag_info(h):
    """Index of worst lag, its value and the full correlation array."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    lag = idx - (n - 1)
    return lag, corr[idx], corr

def _mismatch_indices(h, lag):
    """Indices i where h[i]==1 & h[i-lag]==0, and j where h[j]==0 & h[j-lag]==1."""
    n = h.shape[0]
    if lag >= 0:
        left = h[: n - lag]          # h[i-lag]
        right = h[lag:]              # h[i]
        i = np.where((right == 1) & (left == 0))[0] + lag
        j = np.where((right == 0) & (left == 1))[0] + lag
    else:
        L = -lag
        left = h[L:]                 # h[i+L]
        right = h[: n - L]           # h[i]
        i = np.where((right == 1) & (left == 0))[0]
        j = np.where((right == 0) & (left == 1))[0]
    return i.astype(int), j.astype(int)

def _random_swap(h, rng):
    """Swap a random 1 with a random 0 (keeps sum constant)."""
    ones = np.flatnonzero(h > 0.5)
    zeros = np.flatnonzero(h < 0.5)
    if ones.size == 0 or zeros.size == 0:
        return h.copy()
    i = rng.choice(ones)
    j = rng.choice(zeros)
    h2 = h.copy()
    h2[i] = 0.0
    h2[j] = 1.0
    return h2

def _targeted_swap(h, lag, rng):
    """Swap a mismatched 1 with a complementary 0 for the given lag."""
    i_idx, j_idx = _mismatch_indices(h, lag)
    if i_idx.size == 0 or j_idx.size == 0:
        return None
    i = rng.choice(i_idx)
    j = rng.choice(j_idx)
    h2 = h.copy()
    h2[i] = 0.0
    h2[j] = 1.0
    return h2

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step‑function h, its C5 bound, and the number of samples."""
    start = time.time()
    rng = np.random.default_rng(seed)

    # ----- initialise -------------------------------------------------
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=np.float64)
        n = init.shape[0]
        target_ones = int(round(n / 2.0))
        # force binary while preserving as much of init as possible
        cur = np.zeros_like(init)
        order = np.argsort(-init)
        cur[order[:target_ones]] = 1.0
    else:
        n = 200                     # fallback size
        target_ones = n // 2
        cur = np.zeros(n, dtype=np.float64)
        cur[:target_ones] = 1.0
        rng.shuffle(cur)

    best_h = cur.copy()
    best_score = _c5_val(best_h)
    cur_score = best_score

    # ----- simulated annealing ----------------------------------------
    T = 0.2                # initial temperature
    decay = 0.9999         # geometric cooling
    while time.time() - start < budget_s * 0.95:
        # pick move type
        if rng.random() < 0.7:
            lag, _, _ = _max_lag_info(cur)
            cand = _targeted_swap(cur, lag, rng)
            if cand is None:
                cand = _random_swap(cur, rng)
        else:
            cand = _random_swap(cur, rng)

        cand_score = _c5_val(cand)
        # Metropolis acceptance
        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_h = cand.copy()
        T *= decay

        # early exit if we already beat the target
        if best_score <= 0.3808:
            break

    # ----- greedy refinement -------------------------------------------
    # try to lower the worst lag by deterministic swaps
    for _ in range(2000):
        lag, _, _ = _max_lag_info(best_h)
        cand = _targeted_swap(best_h, lag, rng)
        if cand is None:
            break
        cand_score = _c5_val(cand)
        if cand_score < best_score:
            best_h = cand
            best_score = cand_score
        else:
            break

    return (best_h.tolist(), float(best_score), int(n))
