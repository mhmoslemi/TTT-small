import time, math, random
import numpy as np

# ------------------------------------------------------------
def compute_c5(h):
    """Return the C5 value for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h), zero‑padded, then scale by dx
    return np.max(np.correlate(h, 1.0 - h, mode="full") * dx)


def thue_morse_seq(n):
    """Generate the first n terms of the Thue–Morse binary sequence (0/1)."""
    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = (bin(i).count("1") & 1)  # 1 if odd parity, else 0
    return seq


def init_binary_solution(n):
    """Create a balanced binary vector of length n."""
    # Try to use the provided parent construction if it exists.
    try:
        init = np.asarray(initial_h_values, dtype=float)  # noqa: F821
        m = init.shape[0]
        # Interpolate parent onto the new grid.
        x_old = np.linspace(0.0, 2.0, m, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        interp = np.interp(x_new, x_old, init)
        # Keep the largest n/2 values as ones.
        idx = np.argpartition(-interp, n // 2)[: n // 2]
        h = np.zeros(n, dtype=float)
        h[idx] = 1.0
        return h
    except Exception:
        # No parent available – fall back to a deterministic low‑autocorrelation pattern.
        if n & (n - 1) == 0:                 # power of two → Thue–Morse fits perfectly
            return thue_morse_seq(n)
        # otherwise a simple random balanced vector.
        h = np.zeros(n, dtype=float)
        h[np.random.choice(n, n // 2, replace=False)] = 1.0
        return h


def anneal_h(h_start, start_time, budget_s):
    """Simulated‑annealing improvement of a binary vector."""
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c

    T = 0.02          # initial temperature
    min_T = 1e-6
    decay = 0.9995

    n = h.shape[0]
    while True:
        if time.time() - start_time > budget_s - 0.5:
            break

        # pick a random 1 and a random 0 to swap
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)

        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0

        new_c = compute_c5(h_new)
        delta = new_c - cur_c
        if delta < 0.0 or np.random.rand() < math.exp(-delta / T):
            h = h_new
            cur_c = new_c
            if cur_c < best_c:
                best_h = h.copy()
                best_c = cur_c
                # early exit if we beat the target bound
                if best_c <= 0.3808:
                    return best_h, best_c
        T = max(min_T, T * decay)

    return best_h, best_c
