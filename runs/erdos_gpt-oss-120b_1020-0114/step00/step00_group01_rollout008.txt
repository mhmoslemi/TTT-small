def project_onto_sum_and_bounds(v, target_sum):
    # returns w = clip(v - lam, 0, 1) such that sum(w) = target_sum
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

import time
import numpy as np

def _correlation_full(a, b):
    """Linear correlation a⋆b (same as np.correlate(a,b,'full'))."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())   # next power of two ≥ 2n‑1
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b, n=L)
    c = np.fft.irfft(A * np.conj(B), n=L)[:2 * n - 1]
    return c

def compute_c5(h):
    """C5 value for a step function h sampled on a uniform grid."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = _correlation_full(h, 1.0 - h)
    return np.max(corr) * dx

def _project_to_sum_and_bounds(v, target):
    """Project v onto {0≤x≤1, sum(x)=target} by water‑filling."""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def anneal_continuous(start_h, time_limit):
    """Simulated‑annealing on a continuous step function (maintains sum)."""
    n = start_h.shape[0]
    target = n / 2.0
    cur = start_h.copy()
    cur_val = compute_c5(cur)
    best = cur.copy()
    best_val = cur_val
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        # temperature schedule
        T = 0.015 * (0.99999 ** it)
        # pick two distinct positions
        i, j = np.random.choice(n, 2, replace=False)
        # feasible delta that keeps both in [0,1]
        max_inc_i = 1.0 - cur[i]
        max_dec_i = cur[i]
        max_inc_j = cur[j]
        max_dec_j = 1.0 - cur[j]
        delta_low = -min(max_dec_i, max_inc_j)
        delta_high = min(max_inc_i, max_dec_j)
        if delta_low >= delta_high:
            it += 1
            continue
        delta = np.random.uniform(delta_low, delta_high)
        cand = cur.copy()
        cand[i] += delta
        cand[j] -= delta
        val = compute_c5(cand)
        diff = val - cur_val
        if diff < 0 or np.random.rand() < np.exp(-diff / (T + 1e-12)):
            cur = cand
            cur_val = val
            if cur_val < best_val:
                best_val = cur_val
                best = cur.copy()
        it += 1
    # final projection to guarantee feasibility (numerical drift)
    best = _project_to_sum_and_bounds(best, target)
    best_val = compute_c5(best)
    return best, best_val

def hillclimb_binary(start_h, time_limit):
    """Swap‑based simulated annealing for binary (0/1) vectors."""
    n = start_h.shape[0]
    target = n / 2.0
    cur = start_h.copy()
    cur_val = compute_c5(cur)
    best = cur.copy()
    best_val = cur_val
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        T = 0.02 * (0.99999 ** it)
        # pick an index containing 1 and an index containing 0
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        cand = cur.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        val = compute_c5(cand)
        diff = val - cur_val
        if diff < 0 or np.random.rand() < np.exp(-diff / (T + 1e-12)):
            cur = cand
            cur_val = val
            if cur_val < best_val:
                best_val = cur_val
                best = cur.copy()
        it += 1
    # safety projection (should already be binary)
    best = _project_to_sum_and_bounds(best, target)
    best_val = compute_c5(best)
    return best, best_val

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    global_start = time.time()
    best_h = None
    best_val = np.inf
    best_n = None

    # ----------------------------------------------------------------------
    # Choose a list of grid sizes to explore.
    # ----------------------------------------------------------------------
    try:
        init_vec = np.asarray(initial_h_values, dtype=float)  # may not exist
        n0 = init_vec.shape[0]
    except Exception:
        init_vec = None
        n0 = None

    # grid sizes (even numbers are convenient for the binary case)
    sizes = [80, 100, 120, 140, 160]
    if n0 is not None:
        sizes = [n0] + sizes
    # make sure they are positive and not too large
    sizes = [int(n) for n in sizes if n > 0 and n <= 300]

    # ----------------------------------------------------------------------
    # Main loop over selected discretisations.
    # ----------------------------------------------------------------------
    for n in sizes:
        if time.time() - global_start > budget_s - 5:
            break

        target_sum = n / 2.0
        # ----- initial vector ------------------------------------------------
        if init_vec is not None and init_vec.shape[0] == n:
            h0 = _project_to_sum_and_bounds(init_vec, target_sum)
        else:
            # start from a smooth triangular shape – a known good seed
            xs = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
            h_tri = np.maximum(0.0, 1.0 - np.abs(xs - 1.0))
            h0 = _project_to_sum_and_bounds(h_tri, target_sum)

        # ----- time budgeting for this size ---------------------------------
        elapsed = time.time() - global_start
        remaining_budget = budget_s - elapsed
        # allocate ~60 % to continuous optimisation, 40 % to binary search
        t_cont = min(remaining_budget * 0.6, max(5.0, remaining_budget * 0.6))
        t_bin = remaining_budget - t_cont
        if t_bin < 1.0:
            t_bin = 0.0

        # ----- continuous simulated annealing --------------------------------
        cont_h, cont_val = anneal_continuous(h0, t_cont)
        if cont_val < best_val:
            best_val = cont_val
            best_h = cont_h.copy()
            best_n = n

        # ----- binary hill‑climb (only when n is even) ----------------------
        if n % 2 == 0:
            # start from a random balanced binary vector (also try the rounded cont.)
            bin_start = np.zeros(n)
            idx = np.random.choice(n, n // 2, replace=False)
            bin_start[idx] = 1.0
            # also try a binary version of the continuous result
            bin_from_cont = np.where(cont_h >= 0.5, 1.0, 0.0)
            # ensure exact half ones
            cur_ones = int(bin_from_cont.sum())
            needed = n // 2 - cur_ones
            if needed > 0:
                zero_idx = np.where(bin_from_cont == 0)[0]
                flip = np.random.choice(zero_idx, needed, replace=False)
                bin_from_cont[flip] = 1.0
            elif needed < 0:
                one_idx = np.where(bin_from_cont == 1)[0]
                flip = np.random.choice(one_idx, -needed, replace=False)
                bin_from_cont[flip] = 0.0

            # combine a few starts
            starts = [bin_start, bin_from_cont]
            for s in starts:
                if time.time() - global_start > budget_s - 2:
                    break
                h_bin, val_bin = hillclimb_binary(s, t_bin / len(starts))
                if val_bin < best_val:
                    best_val = val_bin
                    best_h = h_bin.copy()
                    best_n = n

    # ----------------------------------------------------------------------
    # Final sanity check – enforce constraints exactly.
    # ----------------------------------------------------------------------
    if best_h is None:
        # fallback trivial construction (single interval of length 1)
        best_n = 2
        best_h = np.array([1.0, 0.0])
        best_val = compute_c5(best_h)
    else:
        best_h = _project_to_sum_and_bounds(best_h, best_n / 2.0)
        best_val = compute_c5(best_h)

    # Return as list for the harness.
    return (best_h.tolist(), float(best_val), int(best_n))
