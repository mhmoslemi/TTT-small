    # Right part: 1 - h shifted forward by d (where out of range fills with 0)
    # compute array of (1 - h) and shift left (i.e., forward)

    if d >= 0:

        R[N-d:] = 1.0  # out of range
    else:
        # negative shift: j+d <0 for first -d entries.
        dneg = -d
        R[dneg:] = 1 - h[:N-dneg]  # j+d in range

def gradient_shift(h, d):
    N = h.shape[0]
    if d >= 0:
        # compute R: 1 - h[j+d] for j where j+d<N else 1
        R = np.empty(N, dtype=h.dtype)
        R[:N-d] = 1.0 - h[d:]
        R[N-d:] = 1.0
        # compute L: h[j-d] for j >= d else 0
        L = np.empty(N, dtype=h.dtype)
        L[:d] = 0.0
        L[d:] = h[:N-d]
    else:
        dneg = -d
        # R: for j<dneg, out-of-range case
        R = np.empty(N, dtype=h.dtype)
        R[:dneg] = 1.0
        R[dneg:] = 1.0 - h[:N-dneg]
        # L: for j where j-dneg >=0 => h[j+dneg]
        L = np.empty(N, dtype=h.dtype)
        L[:N-dneg] = h[dneg:]
        L[N-dneg:] = 0.0
    return R - L
```

def project_to_simplex_box(v, target):
    lo = -np.max(v) - 5.0
    hi = 1.0 - np.min(v) + 5.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        w = np.clip(v + lam, 0.0, 1.0)
        s = w.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    # due to numeric tolerance, want sum exactly target; distribute small diff
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        # adjust a few free variables (not at bounds)
        free = (w > 0) & (w < 1)
        if free.any():
            # add diff uniformly over free entries
            adj = diff / free.sum()
            w[free] += adj
            # clip again
            w = np.clip(w, 0.0, 1.0)
    return w
```

def improve(h0, dx, target, time_deadline):
    h = h0.copy()
    best_h = h.copy()
    best_val = evaluate(h, dx)
    N = len(h)
    eta = 0.2
    # iteration count may adapt.
    it = 0
    while True:
        it += 1
        if time.time() > time_deadline:
            break
        conv = np.correlate(h, 1-h, mode='full') * dx
        max_idx = np.argmax(conv)
        cur_val = conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
        # compute shift
        d = max_idx - (N - 1)
        grad = gradient_shift(h, d)
        # line search
        step = eta
        improved = False
        for _ in range(8):
            if step < 1e-6:
                break
            h_new = project_to_simplex_box(h - step * grad, target)
            val_new = evaluate(h_new, dx)
            if val_new <= cur_val:
                # improvement or not worse
                h = h_new
                improved = True
                break
            step *= 0.5
        if not improved:
            # if no step improves, reduce eta and maybe random jitter
            eta *= 0.5
            if eta < 1e-4:
                # random restart jitter
                # perturb h a bit
                perturb = np.random.randn(N) * 0.02  # small
                h = project_to_simplex_box(h + perturb, target)
                eta = 0.2
        else:
            # Successful step
            eta = min(1.0, step * 1.2)  # slightly increase
        # limit iteration count?
        if it >= 5000:
            break
    return best_h, best_val

import time
import numpy as np

def _evaluate(h, dx):
    """Return max_{shift} ∫ h(x)(1-h(x+shift))dx approximated by discrete correlation."""
    return np.max(np.correlate(h, 1.0 - h, mode='full') * dx)

def _gradient_shift(h, d):
    """Gradient of the overlap for a given integer shift d (positive = right)."""
    N = h.size
    if d >= 0:
        # R = 1 - h_{j+d} if in range, else 1
        R = np.empty(N, dtype=h.dtype)
        R[:N - d] = 1.0 - h[d:]
        R[N - d:] = 1.0
        # L = h_{j-d} if in range, else 0
        L = np.empty(N, dtype=h.dtype)
        L[:d] = 0.0
        L[d:] = h[:N - d]
    else:
        dneg = -d
        R = np.empty(N, dtype=h.dtype)
        R[:dneg] = 1.0
        R[dneg:] = 1.0 - h[:N - dneg]
        L = np.empty(N, dtype=h.dtype)
        L[:N - dneg] = h[dneg:]
        L[N - dneg:] = 0.0
    return R - L

def _project_to_capped_simplex(v, target):
    """Euclidean projection of v onto {0≤x≤1, sum x = target}."""
    lo = -np.max(v) - 5.0
    hi = 1.0 - np.min(v) + 5.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v + lam, 0.0, 1.0)
        if w.sum() < target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = (w > 0.0) & (w < 1.0)
        if free.any():
            adj = diff / free.sum()
            w[free] += adj
            w = np.clip(w, 0.0, 1.0)
    return w

def _improve(h0, deadline):
    """Simple projected gradient descent on the max-overlap objective."""
    h = h0.copy()
    N = h.size
    dx = 2.0 / N
    target = N * 0.5
    best_h = h.copy()
    best_val = _evaluate(h, dx)
    eta = 0.2
    it = 0
    while time.time() < deadline and it < 5000:
        it += 1
        conv = np.correlate(h, 1.0 - h, mode='full') * dx
        idx = int(conv.argmax())
        cur_val = conv[idx]
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
        d = idx - (N - 1)          # integer shift attaining the maximum
        grad = _gradient_shift(h, d)
        step = eta
        improved = False
        for _ in range(8):
            if step < 1e-8:
                break
            hn = _project_to_capped_simplex(h - step * grad, target)
            val = _evaluate(hn, dx)
            if val <= cur_val:
                h = hn
                improved = True
                break
            step *= 0.5
        if not improved:
            eta *= 0.5
            if eta < 1e-4:
                # random jitter to escape flat regions
                h = _project_to_capped_simplex(h + np.random.randn(N) * 0.02, target)
                eta = 0.2
        else:
            eta = min(1.0, step * 1.2)
    return best_h, best_val
