import time, math, random
import numpy as np

def _project_to_sum_and_box(z: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of z onto {x: sum(x)=target, 0<=x<=1}."""
    lo, hi = -1e3, 1e3
    for _ in range(80):
        lam = (lo + hi) * 0.5
        y = np.clip(z + lam, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)


def _c5_of(h: np.ndarray) -> float:
    """Compute the C5 value for a discrete step function h."""
    N = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / N)


def _random_feasible(N: int, target: float) -> np.ndarray:
    z = np.random.rand(N)
    return _project_to_sum_and_box(z, target)

        init[: N // 2] = 1.0

    target_sum = N / 2.0
    h = _project_to_sum_and_box(init, target_sum)

    best_h = h.copy()
    best_c5 = _c5_of(best_h)

    # ----------------------------------------------------------------------

        if elapsed >= budget_s - 3.0:
            break

        # inner loop: simulated annealing
        while True:
            now = time.time()
            if now - start >= budget_s - 3.0:
