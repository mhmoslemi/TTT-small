def subgradient_max_shift(h, shift):
    N = len(h)
    grad = np.ones(N, dtype=float)
    if shift >= 0:
        s = shift
        if s < N:
            grad[:N - s] -= h[s:]
            grad[s:]    -= h[:N - s]
    else:
        s = -shift
        if s < N:
            grad[s:]    -= h[:N - s]
            grad[:N - s] -= h[s:]
    return grad
