import time
import numpy as np
import random

def compute_c5(h):
    """Maximum overlap integral for a discrete step function."""
    dx = 2.0 / h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full") * dx)


def smooth_max_gradient(h, beta):
    """Gradient of the smooth approximation of max_k C_k."""
    N = h.size
    dx = 2.0 / N
    C = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(C)
    # stable softmax weights
    expb = np.exp(beta * (C - m))
    w = expb / expb.sum()
    offset = N - 1
    grad = np.zeros(N, dtype=float)

    for idx, wl in enumerate(w):
        d = idx - offset
        for i in range(N):
            j = i + d
            if 0 <= j < N:
                grad[i] += wl * dx * (1.0 - h[j])
            k = i - d
            if 0 <= k < N:
                grad[i] -= wl * dx * h[k]
    return grad


def project_onto_box_and_sum(v, target, tol=1e-12, max_iter=60):
    """
    Projection of v onto the intersection of the box [0,1]^n and the hyperplane sum=v.
    Uses a binary search for the Lagrange multiplier.
    """
    lo = -np.max(v)          # ensures v+lo <= 0
    hi = 1.0 - np.min(v)     # ensures v+hi >= 1
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        y = np.clip(v + lam, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    y = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction if needed
    diff = target - y.sum()
    if abs(diff) > tol:
        free = (y > 0) & (y < 1)
        if free.any():
            y[free] += diff / free.sum()
            y = np.clip(y, 0.0, 1.0)
    return y


def projected_gradient_descent(h0, beta_start, beta_end, step0, max_iter, target):
    """Projected gradient descent on the smooth max objective."""
    h = h0.copy()
    for t in range(max_iter):
        beta = beta_start + (beta_end - beta_start) * (t / max_iter)
        grad = smooth_max_gradient(h, beta)
        step = step0 / np.sqrt(t + 1.0)
        h -= step * grad
        h = project_onto_box_and_sum(h, target)
    return h


def hill_climb_improve(h, max_moves, target, start_time, budget):
    """Simple improvement by random pairwise swaps that preserve the sum."""
    cur_h = h.copy()
    cur_c5 = compute_c5(cur_h)
    N = cur_h.size
    moves = 0
    while moves < max_moves:
        if time.time() - start_time > budget * 0.96:
            break
        i, j = np.random.randint(N), np.random.randint(N)
        if i == j:
            continue
        # feasible delta range keeping both entries in [0,1]
        low = max(-cur_h[i], cur_h[j] - 1.0)
        high = min(1.0 - cur_h[i], cur_h[j])
        if low >= high:
            continue
        delta = low + (high - low) * np.random.rand()
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        # sum unchanged, bounds already respected
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            cur_h, cur_c5 = new_h, new_c5
        moves += 1
    return cur_h, cur_c5


def resample_h(old_h, new_n):
    """Linear interpolation of a step function to a different resolution."""
    old_n = old_h.size
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(new_x, old_x, old_h)
