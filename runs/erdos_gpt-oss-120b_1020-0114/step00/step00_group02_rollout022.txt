    # Equivalent to solving min ||x - v||^2 s.t. 0 <= x <= 1, sum x = target.

def optimize_h(initial_h, max_iter, time_budget):
    h = initial_h.copy()
    n = len(h)
    target_sum = n/2
    best_h = h.copy()
    best_val = compute_c5(h)
    start = time.time()
    alpha = 1.0  # initial step size maybe compute based on magnitude.
    for iter in range(max_iter):
        # time check
        if time.time() - start > time_budget - 0.1: break
        # compute correlation and find max shift(s)
        corr = np.correlate(h, 1 - h, mode='full')
        cmax = corr.max()
        # note: we don't multiply by dx yet for comparison; scaling is constant factor.
        if cmax < best_val:
            best_val = cmax
            best_h = h.copy()
        # Choose a shift with max index (could be multiple)
        idx = np.flatnonzero(corr == cmax)[0]  # take first
        shift = idx - (n-1)  # shift index
        # compute subgradient g (length n)
        g = np.zeros_like(h)
        # compute contributions for this shift
        if shift >= 0:
            i = np.arange(0, n - shift)
            j = i + shift
        else:
            shift_abs = -shift
            i = np.arange(shift_abs, n)
            j = i - shift_abs
        # contributions
        g[i] += (1 - h[j])
        g[j] -= h[i]
        # subgradient step: h = h - alpha * g
        h_new = h - alpha * g
        # project onto feasible set
        h = project_onto_capped_simplex(h_new, target_sum)
        # optionally reduce step size gradually
        alpha = 0.99 * alpha  # decay
        # occasionally random perturbation for escape
        if iter % 1000 == 0:
            # small Gaussian noise on interior entries, then project.
            noise = (np.random.rand(n) - 0.5) * 0.02  # small amplitude
            h = project_onto_capped_simplex(h + noise, target_sum)
    # compute final best c5 bound
    c5 = compute_c5(best_h)
    return best_h, c5

def compute_c5(h):

def hill_climb(h, time_limit):
    start = time.time()
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_val = compute_c5(best_h)
    # Pre-generate random indices per iteration
    while time.time() - start < time_limit:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # Determine allowable delta
        max_up_i = 1 - best_h[i]
        max_down_i = best_h[i]
        # When we increase i by delta, we must decrease j by same delta.
        # So delta must satisfy:
        #   delta <= max_up_i
        #   delta <= best_h[j] (cannot make j negative)
        #   delta >= -max_down_i (cannot make i negative)
        #   delta >= -(1 - best_h[j]) (cannot make j >1)
        # Actually if delta negative (decrease i, increase j):
        #   -delta must be <= max_down_i; and -delta <= 1 - best_h[j]
        # So combine:
        delta_max = min(max_up_i, best_h[j])
        delta_min = -min(max_down_i, 1 - best_h[j])
        if delta_max <= delta_min:
            continue
        delta = np.random.uniform(delta_min, delta_max)
        if abs(delta) < 1e-12:
            continue
        h_new = best_h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        val = compute_c5(h_new)
        if val < best_val - 1e-9:
            best_val = val
            best_h = h_new
    return best_h, best_val

import time
import numpy as np

def _c5_value(h):
    """Return C5 = max_x ∫ h(x)*(1-h(x+k)) dx  (discrete approximation)."""
    n = h.size
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode="full")) * dx


def _project_capped_simplex(v, target):
    """Euclidean projection of v onto {0≤x≤1, sum(x)=target}."""
    # Ensure the interval [lo,hi] contains a solution for λ with sum(clip(v-λ,0,1))=target.
    lo, hi = -1.0, 2.0
    # enlarge bounds if needed
    for _ in range(60):
        if np.clip(v - lo, 0, 1).sum() >= target:
            break
        lo -= 2.0
    for _ in range(60):
        if np.clip(v - hi, 0, 1).sum() <= target:
            break
        hi += 2.0
    # binary search for λ
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0, 1)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0, 1)


def _optimize_softmax(h0, time_limit, rng):
    """Minimise max‑overlap using a soft‑max surrogate."""
    start = time.time()
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_val = _c5_value(h)

    # schedule of β (sharpness) values
    betas = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0]
    beta_idx = 0
    it = 0
    while time.time() - start < time_limit:
        if it % 2000 == 0 and beta_idx < len(betas) - 1:
            beta_idx += 1
        beta = betas[beta_idx]

        # correlation (unnormalised)
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = corr.max()
        cur_val = max_corr * (2.0 / n)
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()

        # soft‑max weights
        exp_vals = np.exp(beta * (corr - max_corr))
        w = exp_vals / exp_vals.sum()

        # weighted gradient
        g = np.zeros_like(h)
        # offsets run from -(n-1) … (n-1)
        for idx, weight in enumerate(w):
            if weight == 0.0:
                continue
            offset = idx - (n - 1)
            if offset >= 0:
                i = np.arange(0, n - offset)
                j = i + offset
            else:
                sh = -offset
                i = np.arange(sh, n)
                j = i - sh
            g[i] += weight * (1 - h[j])
            g[j] -= weight * h[i]

        # step size (simple diminishing)
        step = 0.5 / np.sqrt(it + 1.0)
        h = _project_capped_simplex(h - step * g, target)

        # occasional jitter to avoid stagnation
        if it % 5000 == 0 and it > 0:
            jitter = (rng.random(n) - 0.5) * 0.01
            h = _project_capped_simplex(h + jitter, target)

        it += 1

    return best_h, best_val

    n = h0.size
    best_h = h0.copy()
    best_val = _c5_value(best_h)

    while time.time() - start < time_limit:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # allowable δ: increase i, decrease j
        delta_max = min(1.0 - best_h[i], best_h[j])
        delta_min = -min(best_h[i], 1.0 - best_h[j])
        if delta_max <= delta_min:
            continue
        delta = rng.uniform(delta_min, delta_max)
        if abs(delta) < 1e-12:
            continue
        h_new = best_h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        val = _c5_value(h_new)
        if val < best_val - 1e-10:
            best_val = val
            best_h = h_new
    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_total = time.time()

    # Determine dimension – use the provided solution if any.
    init_vals = globals().get("initial_h_values")
    if init_vals is not None:
        init_arr = np.asarray(init_vals, dtype=float)
        n = init_arr.size
        init_arr = _project_capped_simplex(init_arr, n / 2.0)
        init_vectors = [init_arr]
    else:
        n = 50                     # a modest even size when no seed is supplied
        init_vectors = []

    # add several random feasible starts
    target_sum = n / 2.0
    for _ in range(8):
        v = rng.random(n)
        v = _project_capped_simplex(v, target_sum)
        init_vectors.append(v)

    best_h = None
    best_val = float("inf")

    # allocate time to optimisation of each start
    for k, vec in enumerate(init_vectors):
        elapsed = time.time() - start_total
        remaining = budget_s - elapsed - 0.5          # keep half‑second slack
        if remaining <= 0:
            break
        # a shrinking share for later seeds
        share = 0.8 * remaining / (len(init_vectors) - k)
        h_opt, val_opt = _optimize_softmax(vec, share, rng)
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()

    # final refinement via hill climbing
    if best_h is not None:
        time_left = budget_s - (time.time() - start_total) - 0.1
        if time_left > 0:
            best_h, best_val = _hill_climb(best_h, time_left, rng)

    # safety projection and final C5 evaluation
    best_h = _project_capped_simplex(best_h, n / 2.0)
    final_val = _c5_value(best_h)

    return best_h.tolist(), final_val, n
