    # Clip to [0,1] first

def gradient_for_lag(h, lag):
    n = len(h)
    dx = 2.0 / n
    if lag == 0:
        return dx * (1 - 2*h)
    g = np.full(n, dx)  # base term
    # subtract contributions
    if lag > 0:
        # h[i - lag] exists for i >= lag
        g[lag:] -= dx * h[:-lag]
        # h[i + lag] exists for i < n-lag
        g[:n-lag] -= dx * h[lag:]

        # lag negative: let k = -lag >0
        k = -lag

        g[:n-k] -= dx * h[k:]
        # g[i + (-k)] = g[i - k] subtract h[i - k] for i >= k
        g[k:] -= dx * h[:n-k]
    return g
```

import numpy as np
import time
import random

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def gradient_for_lag(h, lag):
    n = len(h)
    dx = 2.0 / n
    if lag == 0:
        return dx * (1 - 2 * h)
    g = np.full(n, dx)
    if lag > 0:
        k = lag
        g[k:] -= dx * h[:-k]
        g[:n - k] -= dx * h[k:]
    else:
        k = -lag
        g[:n - k] -= dx * h[k:]
        g[k:] -= dx * h[:n - k]
    return g

def project_box_simplex(y, target_sum):
    # Euclidean projection onto [0,1]^n with sum target
    # using bisection on lambda
    lo = np.min(y) - 1.0
    hi = np.max(y)
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(y - lam, 0.0, 1.0)

def resample_h(h, new_n):
    old_n = len(h)
    if old_n == new_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)

def optimize_h(init_h, max_iter=2000, time_limit=None, seed=0):
    np.random.seed(seed)
    random.seed(seed)
    n = len(init_h)
    target = n / 2.0
    h = init_h.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    start = time.time()
    # Precompute indexes for shifts up to maybe limit
    for it in range(max_iter):
        if time_limit is not None and time.time() - start > time_limit:
            break
        # compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = corr.max()
        # pick top several lags
        M = 5
        # If number of lags less than M, adjust
        if corr.size <= M:
            top_idxs = np.arange(corr.size)
        else:
            # argpartition for largest M
            part = np.argpartition(corr, -M)[-M:]
            top_idxs = part
        lags = top_idxs - (n - 1)
        grad = np.zeros_like(h)
        for lag in lags:
            grad += gradient_for_lag(h, lag)
        grad /= len(lags)
        # step size schedule

        # project onto constraints

        # evaluate
        cur_val = compute_c5(h)
        if cur_val < best_val:

            best_h = h.copy()

import numpy as np
import time
import random

# ---------- helper functions ----------

def compute_c5(h):
    """Maximum overlap deficiency for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def gradient_for_lag(h, lag):
    """
    Gradient of the overlap deficiency for a single lag.
    Returns d/dh of  dx * sum_i h_i (1 - h_{i-lag}) .
    """
    n = h.shape[0]
    dx = 2.0 / n
    if lag == 0:
        return dx * (1.0 - 2.0 * h)

    g = np.full(n, dx, dtype=np.float64)

    if lag > 0:
        k = lag
        # term with h_{i-k}
        g[k:] -= dx * h[:-k]
        # term with h_{i+k}
        g[: n - k] -= dx * h[k:]
    else:
        k = -lag
        g[: n - k] -= dx * h[k:]
        g[k:] -= dx * h[: n - k]

    return g


def project_box_simplex(y, target):
    """
    Euclidean projection of vector y onto the set
    {x | 0 <= x_i <= 1, sum x_i = target }.
    """
    lo = np.min(y) - 1.0   # low enough so all x_i become 1
    hi = np.max(y)         # high enough so all x_i become 0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(y - lam, 0.0, 1.0)


def resample_h(h, new_n):
    """Linear interpolation of h from its current grid to a grid with new_n points."""
    old_n = h.shape[0]
    if old_n == new_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


def random_h(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    y = rng.random(n)
    return project_box_simplex(y, n / 2.0)


def optimize_h(init_h, max_iter=100000, time_limit=None, seed=0):
    """
    Sub‑gradient descent on the max‑overlap objective.
    Stops after max_iter steps or when time_limit seconds have elapsed.
    """
    rng = np.random.RandomState(seed)
    n = init_h.shape[0]
    target = n / 2.0
    h = init_h.copy()
    best_h = h.copy()
    best_val = compute_c5(h)

    start = time.time()
    for it in range(max_iter):
        if time_limit is not None and (time.time() - start) > time_limit:
            break

        # correlation and the most critical lags
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = np.max(corr)
        # pick a few top lags (including the absolute maximum)
        M = 5
        if corr.size <= M:
            top_idx = np.arange(corr.size)
        else:
            part = np.argpartition(corr, -M)[-M:]
            top_idx = part
        lags = top_idx - (n - 1)

        # average sub‑gradient of the selected lags
        grad = np.zeros_like(h)
        for lag in lags:
            grad += gradient_for_lag(h, int(lag))
        grad /= len(lags)

        # diminishing step size
        step = 1.0 / np.sqrt(it + 1.0)

        # gradient step + projection onto constraints
        h = project_box_simplex(h - step * grad, target)

        # keep the best seen
        cur_val = compute_c5(h)
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()

    return best_h, best_val
