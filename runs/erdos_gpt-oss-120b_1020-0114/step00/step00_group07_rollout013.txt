import time
import numpy as np

def _correlation_max(h):
    """Return the full correlation array and its maximum."""
    corr = np.correlate(h, 1 - h, mode="full")
    return corr, corr.max()


def _random_binary(rng, n, half):
    """Generate a binary vector of length n with exactly `half` ones."""
    h = np.zeros(n, dtype=np.uint8)
    h[rng.choice(n, half, replace=False)] = 1
    return h


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size and initial vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=np.float64)
        n = init.size
        half = n // 2
        # make a binary start from the best half of the given values
        idx = np.argsort(-init)[:half]
        h_curr = np.zeros(n, dtype=np.uint8)
        h_curr[idx] = 1
    else:
        n = 80
        half = n // 2
        h_curr = _random_binary(rng, n, half)

    # guarantee feasibility (should already hold)
    assert h_curr.sum() == half

    # ------------------------------------------------------------------
    # Pre‑compute target threshold (integer count of mismatches)
    # ------------------------------------------------------------------
    target_c5 = 0.3808
    dx = 2.0 / n
    target_corr = int(np.floor(target_c5 / dx + 1e-12))  # integer bound on count

    # ------------------------------------------------------------------
    # Initial evaluation
    # ------------------------------------------------------------------
    _, best_corr = _correlation_max(h_curr)
    best_h = h_curr.copy()

    # ------------------------------------------------------------------
    # Parameters for the search
    # ------------------------------------------------------------------
    start_time = time.time()
    max_time = budget_s * 0.95  # a little safety margin
    iteration = 0
    last_improve = 0
    restart_limit = 20000   # iterations without improvement before a restart
    max_temp = 2.0
    min_temp = 0.001

    # ------------------------------------------------------------------
    # Main simulated‑annealing / hill‑climbing loop
    # ------------------------------------------------------------------
    while True:
        elapsed = time.time() - start_time
        if elapsed > max_time:
            break

        # choose move: targeted if far from target, else random
        if best_corr > target_corr and rng.random() < 0.85:
            # targeted: try to fix the current worst shift
            corr_arr, curr_corr = _correlation_max(h_curr)
            max_idx = int(np.argmax(corr_arr))
            shift = max_idx - (n - 1)           # signed shift
            if shift >= 0:
                mask = (h_curr[: n - shift] == 1) & (h_curr[shift:] == 0)
                mismatched = np.flatnonzero(mask)
            else:
                s = -shift
                mask = (h_curr[s:] == 1) & (h_curr[: n - s] == 0)
                mismatched = np.flatnonzero(mask) + s

            if mismatched.size == 0:
                # fallback to a random 1
                i = rng.choice(np.flatnonzero(h_curr == 1))
            else:
                i = rng.choice(mismatched)

            # pick a zero somewhere (anywhere)
            zero_pos = np.flatnonzero(h_curr == 0)
            j = rng.choice(zero_pos)
        else:
            # simple random swap
            ones = np.flatnonzero(h_curr == 1)
            zeros = np.flatnonzero(h_curr == 0)
            i = rng.choice(ones)
            j = rng.choice(zeros)
            _, curr_corr = _correlation_max(h_curr)  # we need current max for acceptance

        # perform the swap
        h_curr[i] = 0
        h_curr[j] = 1

        # evaluate new state
        _, new_corr = _correlation_max(h_curr)

        # acceptance rule
        if new_corr <= curr_corr:
            accept = True
        else:
            # temperature schedule (linear cooling)
            temp = max(min_temp, max_temp * (1.0 - iteration / 800000.0))
            prob = np.exp(-(new_corr - curr_corr) / temp)
            accept = rng.random() < prob

        if accept:
            curr_corr = new_corr
            if new_corr < best_corr:
                best_corr = new_corr
                best_h = h_curr.copy()
                last_improve = iteration
                if best_corr <= target_corr:
                    break
        else:
            # revert swap
            h_curr[i] = 1
            h_curr[j] = 0

        iteration += 1

        # restart if stagnation persists
        if iteration - last_improve > restart_limit:
            h_curr = _random_binary(rng, n, half)
            _, curr_corr = _correlation_max(h_curr)
            last_improve = iteration

    # ------------------------------------------------------------------
    # Final result
    # ------------------------------------------------------------------
    c5_bound = best_corr * dx
    h_out = best_h.astype(np.float64)
    return (h_out, float(c5_bound), int(n))
