import time
import numpy as np

def _project_to_simplex(v, target_sum):
    """Project v onto the box [0,1] with exact sum = target_sum."""
    v = np.clip(v, 0.0, 1.0)
    diff = target_sum - v.sum()
    if abs(diff) < 1e-12:
        return v
    # water‑filling style projection
    for _ in range(10):
        free = (v > 0.0) & (v < 1.0)
        cnt = free.sum()
        if cnt == 0:
            break
        delta = diff / cnt
        # apply delta only where it stays in bounds
        add_up = np.minimum(delta, 1.0 - v, where=free)
        add_dn = np.maximum(delta, -v, where=free)
        v[free] += np.where(delta > 0, add_up, add_dn)
        diff = target_sum - v.sum()
        if abs(diff) < 1e-12:
            break
    # final clipping (in case of tiny numerical drift)
    return np.clip(v, 0.0, 1.0)


def _c5_of_h(h):
    """C5 for a discrete step function h (array) with dx = 2/len(h)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx
