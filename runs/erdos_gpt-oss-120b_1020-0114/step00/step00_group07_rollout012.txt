    if k >= 0:

        out = np.empty_like(arr)
        out[:n - k] = arr[k:]

import time
import numpy as np

def _compute_c5(h):
    """Return C5 value for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx), corr


def _shift(arr, k):
    """Shift array by k positions with zero padding. Positive k shifts left."""
    n = arr.shape[0]
    if k == 0:
        return arr.copy()
    if k > 0:
        if k >= n:
            return np.zeros_like(arr)
        out = np.empty_like(arr)
        out[: n - k] = arr[k:]
        out[n - k :] = 0.0
        return out
    else:  # k < 0
        k = -k
        if k >= n:
            return np.zeros_like(arr)
        out = np.empty_like(arr)
        out[k:] = arr[: n - k]
        out[:k] = 0.0
        return out


def _proj_box_sum(v, target):
    """
    Euclidean projection of v onto {0 <= x <= 1, sum(x) = target}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        w = np.clip(v + lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    w = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    # tiny adjustment if rounding left a small residual
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        mask = (w > 0.0) & (w < 1.0)
        cnt = int(mask.sum())
        if cnt:
            w[mask] += diff / cnt
            np.clip(w, 0.0, 1.0, out=w)
    return w

        v = start_vec.copy()
    best_h = v.copy()
    best_c5, _ = _compute_c5(v)

    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        # correlation and worst shift
        corr = np.correlate(v, 1.0 - v, mode="full")
        idx = int(corr.argmax())
        k = idx - (n - 1)

        # direction that reduces the worst‑case term
        forward = _shift(v, k)
        backward = _shift(v, -k)
        direction = forward + backward

        # diminishing step size
        eta = 0.5 / (1.0 + it / 2000.0)

        v_new = v + eta * direction
        v_new = _proj_box_sum(v_new, target)

        cur_c5, _ = _compute_c5(v_new)

        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = v_new.copy()

        v = v_new
        it += 1
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # choose a moderate resolution (fast enough, enough freedom)
    n = 200
    target_sum = n / 2.0

    best_h = None
    best_c5 = np.inf
    best_n = n

    # check for a parent construction
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        c5_0, _ = _compute_c5(h0)
        if c5_0 < best_c5:
            best_c5 = c5_0
            best_h = h0.copy()
            best_n = h0.shape[0]

    # allocate the time budget among several restarts
    remaining_budget = budget_s
    while time.time() - start_time < budget_s * 0.95:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 5.0:
            break
        # give each restart a slice of the remaining time
        slice_time = min(remaining * 0.15, 150.0)

        # occasionally start from the current best (refinement)
        if best_h is not None and best_h.shape[0] == n and rng.random() < 0.25:
            start_vec = best_h.copy()
        else:
            start_vec = None

        h_candidate, c5_candidate = _subgradient_optimize(
            n, rng, slice_time, start_vec=start_vec
        )
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate.copy()
            best_n = n

        # early exit if record is already beaten
        if best_c5 <= 0.3808:
            break

    # final sanity projection (preserve the integral exactly)
    if best_h is not None and best_h.shape[0] == n:
        cur_sum = best_h.sum()
        diff = target_sum - cur_sum
        if abs(diff) > 1e-12:
            mask = (best_h > 0.0) & (best_h < 1.0)
            cnt = int(mask.sum())
            if cnt:
                best_h[mask] += diff / cnt
                np.clip(best_h, 0.0, 1.0, out=best_h)

        final_c5, _ = _compute_c5(best_h)
        best_c5 = final_c5
        best_n = n

    return best_h, best_c5, best_n
