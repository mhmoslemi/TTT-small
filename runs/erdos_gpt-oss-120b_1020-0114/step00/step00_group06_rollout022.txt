def project_sum_box(v, target):
    # Solve for lambda such that sum(clip(v + lambda, 0, 1)) = target
    # Use binary search over lambda.
    lo = -np.max(v) - 1.0  # ensure all values at lower bound 0
    hi = 1 - np.min(v) + 1.0  # ensure all values at upper bound 1
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo+hi)/2, 0.0, 1.0)

import numpy as np
import time
import math

def _project_sum_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto the intersection {0≤x≤1, sum(x)=target} by
    shifting then clipping: x = clip(v + λ,0,1) with λ chosen by bisection."""
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(v + lam, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """C₅ = max_k ∫ h(x)[1‑h(x+k)]dx evaluated on the discrete grid."""
    n = h.shape[0]
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


def _grad_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the autocorrelation sum_{i}h_i h_{i+shift} w.r.t. h."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        g[: n - shift] += h[shift:]
        g[shift:] += h[: n - shift]
    else:
        s = -shift
        g[s:] += h[: n - s]
        g[: n - s] += h[s:]
    return g

                    best_h = cand.copy()

        # occasional perturbation of the current best to escape plateaus
        if np.random.rand() < 0.001:
            pert = np.random.normal(scale=0.05, size=n)
            cand = best_h + pert
            cand = np.clip(cand, 0.0, 1.0)
            cand = _project_sum_box(cand, target_sum)
            cand_c5 = _c5_value(cand)
            if cand_c5 < best_c5:
                best_h = cand.copy()
                best_c5 = cand_c5
            if cand_c5 < cur_c5 or np.random.rand() < 0.5:
                h = cand
                cur_c5 = cand_c5

    # final evaluation (ensuring consistency)
    final_c5 = float(_c5_value(best_h))
    return best_h, final_c5, int(n)
