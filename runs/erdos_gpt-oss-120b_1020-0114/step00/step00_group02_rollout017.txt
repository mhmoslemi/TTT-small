def project_box_sum(y, target_sum, lower=0.0, upper=1.0, tol=1e-9, max_iter=100):
    lo = y.min() - upper
    hi = y.max() - lower
    for _ in range(max_iter):
        tau = (lo + hi) / 2.0
        x = np.clip(y - tau, lower, upper)
        s = x.sum()
        if abs(s - target_sum) <= tol:
            return x
        if s > target_sum:
            lo = tau
        else:
            hi = tau
    return np.clip(y - (lo+hi)/2, lower, upper)

import time
import numpy as np

def _project_box_sum(y, target, lower=0.0, upper=1.0, tol=1e-9, max_iter=100):
    """Project y onto the box [lower,upper] with the equality sum(x)=target."""
    lo = y.min() - upper
    hi = y.max() - lower
    for _ in range(max_iter):
        tau = (lo + hi) / 2.0
        x = np.clip(y - tau, lower, upper)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            lo = tau
        else:
            hi = tau
    return np.clip(y - (lo + hi) / 2.0, lower, upper)


def _c5_value(h):
    """Return the overlap constant C5 for a vector h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)


def _gradient_max_lag(h, lag):
    """Gradient of the overlap term that attains the maximum (for given lag)."""
    n = h.size
    g = np.zeros(n, dtype=float)
    if lag >= 0:
        # first indices i where i+lag is valid
        i1 = np.arange(0, n - lag)
        g[i1] += 1.0 - h[i1 + lag]
        # second indices i where i-lag is valid
        i2 = np.arange(lag, n)
        g[i2] += -h[i2 - lag]
    else:
        d = -lag
        # first indices (i+lag >=0)
        i1 = np.arange(d, n)
        g[i1] += 1.0 - h[i1 - d]          # i1+lag = i1-d
        # second indices (i-lag < n)
        i2 = np.arange(0, n - d)
        g[i2] += -h[i2 + d]               # i2-lag = i2+d
    return g
