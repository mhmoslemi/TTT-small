def propose_move(h, rng):
    n = len(h)
    i, j = rng.integers(0,n,size=2)
    if i==j:
        return None
    # direction random
    if rng.random() < 0.5:
        # from i -> j
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0: return None
        eps = rng.random() * max_eps
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
    else:
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0: return None
        eps = rng.random() * max_eps
        new_h = h.copy()
        new_h[j] -= eps
        new_h[i] += eps
    return new_h

import time
import numpy as np

def _compute_corr(arr):
    return np.correlate(arr, 1.0 - arr, mode='full')

def _c5_from_corr(corr, n):
    return np.max(corr) * (2.0 / n)

def _adjust_sum(v):
    target = v.shape[0] / 2.0
    diff = target - v.sum()
    if abs(diff) < 1e-12:
        return v
    # simple redistribution respecting bounds
    for i in range(v.shape[0]):
        if diff > 0:
            space = 1.0 - v[i]
            delta = min(diff, space)
            v[i] += delta
            diff -= delta
        else:
            space = v[i]
            delta = min(-diff, space)
            v[i] -= delta
            diff += delta
        if abs(diff) < 1e-12:
            break
    return v

def run(seed=42, budget_s=1100, **kwargs):

    start = time.time()

    # ---------- initialise h ----------
    try:

        n = 120
        h = rng.random(n)
        h = np.clip(h, 0.0, 1.0)
        h = _adjust_sum(h)

    n = h.shape[0]
    h = _adjust_sum(h)

    # ---------- helpers ----------
    def grad_for_shift(arr, s):
        n = arr.shape[0]
        g = np.zeros(n, dtype=float)
        if s >= 0:
            if n - s > 0:
                g[:n - s] += 1.0 - arr[s:]
                g[s:]   -= arr[:n - s]
        else:
            t = -s
            if n - t > 0:
                g[t:]   += 1.0 - arr[:n - t]
                g[:n - t] -= arr[t:]
        return g

    # current state
    cur_h = h.copy()
    cur_corr = _compute_corr(cur_h)
    cur_c5 = _c5_from_corr(cur_corr, n)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    # annealing schedule
    T_start = 0.02
    T_end   = 1e-7
    max_iters = 10_000_000   # large; we stop by time
    last_imp = 0
    it = 0

    while True:
        if time.time() - start > budget_s * 0.95:

        # worst shift and its gradient
        max_idx = int(np.argmax(cur_corr))
        s = max_idx - (n - 1)

        if rng.random() < 0.7:
