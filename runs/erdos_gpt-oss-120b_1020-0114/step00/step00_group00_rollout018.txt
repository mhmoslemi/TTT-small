import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return max_x ∫ h(x)(1-h(x+x_shift)) dx approximated by full correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())


def _project_to_bounds_and_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto {x | 0≤x_i≤1, sum x = target}.
    Uses bisection on the Lagrange multiplier λ for the affine constraint.
    """
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) / 2.0
    return np.clip(v - lam, 0.0, 1.0)


def upsample_h(src: np.ndarray, n_target: int) -> np.ndarray:
    """Linearly interpolate src (defined on [0,2]) to n_target samples and enforce constraints."""
    n_src = src.shape[0]
    if n_target == n_src:
        return src.copy()
    xs = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    xt = (np.arange(n_target) + 0.5) * (2.0 / n_target)
    h_t = np.interp(xt, xs, src, left=src[0], right=src[-1])
    h_t = _project_to_bounds_and_sum(h_t, n_target / 2.0)
    return h_t


def _gradient_max_shift(h: np.ndarray):
    """
    Compute gradient of the correlation term that attains the maximum.
    Returns (gradient, value_of_max_correlation_without_dx).
    """
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(conv))
    shift = idx - (n - 1)                     # integer shift (positive → right)
    g = np.zeros_like(h)

    if shift >= 0:
        # term (1 - h[i+shift])
        i1 = np.arange(0, n - shift)
        g[i1] += 1.0 - h[i1 + shift]
        # term -h[i-shift]
        i2 = np.arange(shift, n)
        g[i2] -= h[i2 - shift]
    else:
        s = -shift
        # term (1 - h[i-shift]) where i≥s
        i1 = np.arange(s, n)
        g[i1] += 1.0 - h[i1 - s]
        # term -h[i+shift] where i≤n-1-s
        i2 = np.arange(0, n - s)
        g[i2] -= h[i2 + s]

    return g, conv[idx]


def descent(h0: np.ndarray, step0: float = 0.5, max_iter: int = 2000):
    """Projected sub‑gradient descent on the max‑overlap objective."""
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for t in range(max_iter):
        g, _ = _gradient_max_shift(h)
        step = step0 / np.sqrt(t + 1.0)
        h_new = h - step * g
        h_new = _project_to_bounds_and_sum(h_new, target)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new

    return best_h, best_c5


def local_refine(h: np.ndarray, cur_c5: float, max_iter: int,
                 rng: np.random.Generator, time_limit: float = 0.0):
    """Simple 2‑coordinate random moves keeping the sum constant."""
    start = time.time()
    n = h.shape[0]
    best_h = h.copy()
    best_val = cur_c5

    while max_iter > 0 and (time.time() - start) < time_limit:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        if rng.random() < 0.5:
            delta_max = min(1.0 - best_h[i], best_h[j])
            if delta_max > 0.0:
                delta = rng.random() * delta_max
                h_new = best_h.copy()
                h_new[i] += delta
                h_new[j] -= delta
                c5_new = compute_c5(h_new)
                if c5_new < best_val:
                    best_h, best_val = h_new, c5_new
        else:
            delta_max = min(1.0 - best_h[j], best_h[i])
            if delta_max > 0.0:
                delta = rng.random() * delta_max
                h_new = best_h.copy()
                h_new[j] += delta
                h_new[i] -= delta
                c5_new = compute_c5(h_new)
                if c5_new < best_val:
                    best_h, best_val = h_new, c5_new

        max_iter -= 1

    return best_h, best_val
