import math
import time
import numpy as np


def _compute_c5(arr: np.ndarray) -> float:
    """C5 = max_k ∫ h(x)(1−h(x+k))dx  (discretized)."""
    n = arr.shape[0]
    dx = 2.0 / n
    corr = np.correlate(arr, 1.0 - arr, mode="full")
    return float(corr.max() * dx)


def _worst_shift(arr: np.ndarray) -> int:
    """Index shift (can be negative) giving smallest ∑ h_i h_{i+shift}."""
    conv = np.correlate(arr, arr, mode="full")
    idx_min = int(conv.argmin())
    n = arr.shape[0]
    return idx_min - (n - 1)


def _gradient(arr: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the overlap ∑ h_i h_{i+shift} w.r.t. each entry of arr."""
    n = arr.shape[0]
    g = np.zeros_like(arr)
    if shift == 0:
        g = 2.0 * arr
    elif shift > 0:
        k = shift
        g[: n - k] += arr[k:]
        g[k:] += arr[: n - k]
    else:  # shift < 0
        k = -shift
        g[k:] += arr[: n - k]
        g[: n - k] += arr[k:]
    return g


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step‑function h on [0,2] (discretised) to minimise
    C5 = max_k ∫ h(x)(1−h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    rnd = np.random.RandomState(seed)

    # ------------------------------------------------------------------
    # initialise
    try:
        h = np.array(initial_h_values, dtype=np.float64)  # type: ignore
    except Exception:
        # fallback: binary vector with exactly half ones
        n_points = 200
        h = np.zeros(n_points, dtype=np.float64)
        idx = rnd.choice(n_points, n_points // 2, replace=False)
        h[idx] = 1.0
    else:
        n_points = h.shape[0]

    n = n_points
    target_sum = n / 2.0
    # tiny correction if the initial sum drifts (should not happen)
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        diff = target_sum - cur_sum
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
        # re‑balance any tiny residual
        residual = h.sum() - target_sum
        if abs(residual) > 1e-12:
            # shift a small amount from a random non‑boundary entry
            candidates = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
            if candidates.size:
                i = rnd.choice(candidates)
                h[i] -= residual

    cur_c = _compute_c5(h)
    best_h = h.copy()
    best_c = cur_c

    start = time.time()
    # temperature for simulated‑annealing part
    T0 = 0.004

    # ------------------------------------------------------------------
    # main optimisation loop
    while True:
        now = time.time()
        if now - start >= budget_s * 0.95:
            break
        elapsed = now - start
        T = T0 * max(0.0, 1.0 - elapsed / (budget_s * 0.95))

        # 60 % of the time we try a “gradient” move that targets the
        # shift with the smallest overlap; the rest we do a random transfer.
        if rnd.rand() < 0.6:
            # ------------ gradient move -----------------
            shift = _worst_shift(h)
            g = _gradient(h, shift)

            # indices that can receive / give mass
            inc_cand = np.where(h < 1.0 - 1e-12)[0]
            dec_cand = np.where(h > 1e-12)[0]
            if inc_cand.size == 0 or dec_cand.size == 0:
                continue

            i_inc = inc_cand[g[inc_cand].argmax()]   # largest gradient
            i_dec = dec_cand[g[dec_cand].argmin()]   # smallest gradient

            max_delta = min(1.0 - h[i_inc], h[i_dec])
            if max_delta <= 0.0:
                continue
            delta = max_delta * rnd.rand() * 0.5
            h_new = h.copy()
            h_new[i_inc] += delta
            h_new[i_dec] -= delta
        else:
            # ------------ random transfer move ------------
            i, j = rnd.choice(n, 2, replace=False)
            if h[i] < 1.0 - 1e-12 and h[j] > 1e-12:
                max_delta = min(1.0 - h[i], h[j])
                delta = max_delta * rnd.rand() * 0.5
                h_new = h.copy()
                h_new[i] += delta
                h_new[j] -= delta
            elif h[j] < 1.0 - 1e-12 and h[i] > 1e-12:
                max_delta = min(1.0 - h[j], h[i])
                delta = max_delta * rnd.rand() * 0.5
                h_new = h.copy()
                h_new[j] += delta
                h_new[i] -= delta
            else:
                continue

        # enforce box constraints (numerical safety)
        np.clip(h_new, 0.0, 1.0, out=h_new)

        new_c = _compute_c5(h_new)

        # accept if better, otherwise with Metropolis probability
        if new_c < cur_c - 1e-12:
            accept = True
        else:
            accept = (T > 0.0) and (rnd.rand() < math.exp((cur_c - new_c) / T))

        if accept:

            cur_c = new_c

    final_c = _compute_c5(best_h)

    return best_h.tolist(), float(final_c), n_points
