def local_search(N, init_h, time_limit):
    cur_h = init_h.copy()
    cur_cnt = compute_max_mismatch(cur_h)
    cur_c5 = cur_cnt * (2.0 / N)
    best_h_local = cur_h.copy()
    best_cnt_local = cur_cnt
    best_c5_local = cur_c5
    start = time.time()
    T = 0.5
    iteration = 0
    while time.time() - start < time_limit:
        # propose swap
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = random.choice(ones)
        j = random.choice(zeros)
        # swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_cnt = compute_max_mismatch(cur_h)
        new_c5 = new_cnt * (2.0 / N)
        delta = new_c5 - cur_c5
        accept = False
        if delta < -1e-12:  # improvement
            accept = True
        else:
            # accept with probability exp(-delta/T)
            if random.random() < math.exp(-delta / T):
                accept = True
        if accept:
            cur_cnt = new_cnt
            cur_c5 = new_c5
            if new_c5 < best_c5_local - 1e-12:
                best_h_local = cur_h.copy()
                best_cnt_local = new_cnt
                best_c5_local = new_c5
        else:
            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        # schedule T reduce
        T = max(T * 0.9995, 1e-4)
        iteration += 1
    return best_h_local, best_cnt_local, best_c5_local

import numpy as np
import random
import time
import math

def compute_c5(h):
    """Discrete C5 from correlation."""
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, "full")
    # only non‑negative shifts
    max_corr = corr[n - 1 :].max()
    return max_corr * (2.0 / n)


def max_overlap_ab(a, b):
    """Analytic max_{k} ∫ h(x)(1‑h(x+k))dx for the three‑level step."""
    if a < 0.0 or a > 1.0 or b <= 0.0 or a + b > 2.0:
        return float("inf")
    # ensure p∈[0,1]  →  b ≥ 1‑a
    if b < 1.0 - a - 1e-12:
        return float("inf")
    p = (1.0 - a) / b
    # candidate shift values (break points of the piecewise linear O(k))
    ks = [0.0, a, b, a + b, 2.0 - a, 2.0 - (a + b), 2.0]
    best = 0.0
    for k in ks:
        if k < 0.0 or k > 2.0:
            continue
        # overlaps of the four relevant interval pairs
        l12 = max(0.0, min(a, a + b - k) - max(0.0, a - k))
        l13 = max(0.0, min(a, 2.0 - k) - max(0.0, a + b - k))
        l22 = max(0.0, b - k) if k <= b else 0.0
        l23 = max(0.0, min(a + b, 2.0 - k) - max(a, a + b - k))
        O = (1.0 - p) * l12 + l13 + p * (1.0 - p) * l22 + p * l23
        if O > best:
            best = O
    return best
