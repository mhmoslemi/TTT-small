def project_to_sum_and_bounds(v, target):
    lower = 0.0
    upper = 1.0
    # Find λ such that sum( clip(v - λ, lower, upper) ) == target.
    # We'll perform binary search over λ.
    # λ low may be v.min() - upper (makes all entries <= upper after shift? Actually if λ <= v.min() - lower, all entries shift up but after clipping maybe > lower.)
    # We'll just start with lo = -1.0, hi = 1.0 (since v in [0,1] originally)
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        y = np.clip(v - mid, lower, upper)
        s = y.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - (lo+hi)/2, lower, upper)
    # Due to rounding, adjust small error:
    diff = target - y.sum()
    if abs(diff) > 1e-9:
        # distribute diff among non-extreme entries
        # simple loop: add equally to entries not at bounds
        mask = (y > lower+1e-12) & (y < upper-1e-12)
        if np.any(mask):
            y[mask] += diff / np.count_nonzero(mask)
            y = np.clip(y, lower, upper)
    return y

def compute_corr_full(h):
    n = h.shape[0]
    size = 2 * n
    H = np.fft.fft(h, size)
    G = np.fft.fft(1 - h, size)
    corr = np.fft.ifft(H * np.conj(G)).real
    # length 2n: we need only first 2n-1 entries.
    return corr[:2*n - 1]

import time
import numpy as np

def _corr_full(h: np.ndarray) -> np.ndarray:
    """Linear cross‑correlation of h with (1‑h) using FFT."""
    n = h.shape[0]
    size = 2 * n
    H = np.fft.fft(h, size)
    G = np.fft.fft(1.0 - h, size)
    corr = np.fft.ifft(H * np.conj(G)).real
    return corr[: 2 * n - 1]  # length 2n‑1 (full mode)


def _c5_value(h: np.ndarray) -> float:
    """Return the C₅ value for a given vector."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(_corr_full(h)) * dx


def _target_shift(corr: np.ndarray, n: int) -> int:
    """Shift (as an integer index offset) where the correlation is maximal."""
    idx = int(np.argmax(corr))
    return idx - (n - 1)


def _attempt_targeted_swap(
    h: np.ndarray, shift: int, rng: np.random.Generator
) -> np.ndarray | None:
    """
    For the given shift, try to swap a mismatching 1‑0 pair with a 0‑1 pair.
    Returns a new vector or None if no suitable pair exists.
    """
    n = h.shape[0]
    idx = np.arange(n)
    if shift >= 0:
        ok = idx + shift < n
        partner = idx + shift
    else:
        s = -shift
        ok = idx - s >= 0
        partner = idx - s

    # positions where (h[i]=1, h[i+shift]=0)
    mask_one_zero = (h == 1) & ok & (h[partner] == 0)
    # positions where (h[i]=0, h[i+shift]=1)
    mask_zero_one = (h == 0) & ok & (h[partner] == 1)

    i_candidates = np.flatnonzero(mask_one_zero)
    j_candidates = np.flatnonzero(mask_zero_one)
    if i_candidates.size == 0 or j_candidates.size == 0:
        return None

    i = rng.choice(i_candidates)
    j = rng.choice(j_candidates)
    cand = h.copy()
    cand[i] = 0.0
    cand[j] = 1.0
    return cand


def _random_binary_swap(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Swap a random 1 with a random 0, keeping the weight fixed."""
    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)
    i = rng.choice(ones)
    j = rng.choice(zeros)
    cand = h.copy()
    cand[i] = 0.0
    cand[j] = 1.0
    return cand


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Returns a step‑function vector `h`, the achieved C₅ bound, and the sample count.
    """
    rng = np.random.default_rng(seed)
    # resolution – small enough to stay fast, large enough to allow improvement
    n = 200                     # 200 samples → dx = 0.01, weight = n/2 = 100
    target_ones = n // 2

    # ----- initial vector -------------------------------------------------
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=np.float64)
        if init.shape[0] != n:
            # linear resampling to the required length
            x_old = np.linspace(0.0, 2.0, init.shape[0], endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            init = np.interp(x_new, x_old, init)
        # binarise and force the exact weight
        h = (init >= 0.5).astype(np.float64)
        cur_ones = int(h.sum())
        if cur_ones > target_ones:
            excess = cur_ones - target_ones
            idx = rng.choice(np.flatnonzero(h == 1), size=excess, replace=False)
            h[idx] = 0.0
        elif cur_ones < target_ones:
            deficit = target_ones - cur_ones
            idx = rng.choice(np.flatnonzero(h == 0), size=deficit, replace=False)
            h[idx] = 1.0
    else:
        h = np.zeros(n, dtype=np.float64)
        h[rng.choice(n, size=target_ones, replace=False)] = 1.0

    cur_c5 = _c5_value(h)
    best_h = h.copy()
    best_c5 = cur_c5

    # ----- optimisation loop ---------------------------------------------
    start_time = time.time()
    max_iters = 800_000                # generous upper bound
    stagnation_limit = 30_000          # iterations without improvement → restart
    last_improve = 0
    it = 0

    while True:
        # time budget check
        if time.time() - start_time > budget_s * 0.95:
            break
        if it >= max_iters:
            break

        # compute the worst shift for the current vector
        corr = _corr_full(h)
        shift = _target_shift(corr, n)

        # attempt a focused swap that directly reduces the worst‑shift count
        cand = _attempt_targeted_swap(h, shift, rng)

        # if no useful pair exists, fall back to a random 0‑1 swap
        if cand is None:
            cand = _random_binary_swap(h, rng)

        cand_c5 = _c5_value(cand)

        # accept if the new vector improves the objective
        if cand_c5 < cur_c5:
            h = cand
            cur_c5 = cand_c5
            if cand_c5 < best_c5:
                best_h = cand.copy()
                best_c5 = cand_c5
                last_improve = it
        else:
            # occasional uphill move (very low probability)
            if rng.random() < 0.001:
                h = cand
                cur_c5 = cand_c5

        # occasional restart if stuck for a long while
        if it - last_improve > stagnation_limit:
            # keep the best seen so far, restart with a fresh random binary vector
            h = np.zeros(n, dtype=np.float64)
            h[rng.choice(n, size=target_ones, replace=False)] = 1.0
            cur_c5 = _c5_value(h)
            last_improve = it

        it += 1

    # final sanity projection (still binary, weight exact)
    best_h = np.clip(best_h, 0.0, 1.0)
    # ensure exact weight – a quick correction if rounding slipped
    cur_ones = int(best_h.sum())
    if cur_ones != target_ones:
        diff = target_ones - cur_ones
        if diff > 0:
            # need more ones
            candidates = np.flatnonzero(best_h == 0)
            best_h[rng.choice(candidates, size=diff, replace=False)] = 1.0
        else:
            # too many ones
            candidates = np.flatnonzero(best_h == 1)
            best_h[rng.choice(candidates, size=-diff, replace=False)] = 0.0

    final_c5 = _c5_value(best_h)
    return best_h, final_c5, n
