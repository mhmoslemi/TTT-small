def _compute_c5(h):
    """C5 = max_k ∫ h(x)*(1-h(x+k)) dx  ;  discrete version."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _project_box_sum(v, target):
    """Euclidean projection of v onto [0,1]^n ∩ {sum = target}."""
    # Find λ such that sum(clip(v+λ,0,1)) = target (λ is monotone)
    lo, hi = -3.0, 3.0

        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()

            lo = lam
        else:
            hi = lam

            p[: n - shift] = h[shift:]
    else:
        s = -shift
        if s < n:
            p[s:] = h[: n - s]

            # simulated‑annealing style acceptance
            T = 0.02 * frac                     # temperature shrinks to 0
            if random.random() < math.exp(-(new_c5 - cur_c5) / (T + 1e-12)):
                h = h_new
                cur_c5 = new_c5

        # occasional pure random transfers to escape flat regions
        if random.random() < 0.03:
            i, j = np.random.choice(N, 2, replace=False)
            max_transfer = min(0.1, h[i], 1.0 - h[j])
            if max_transfer > 1e-12:
                delta = max_transfer * random.random()
                h_tmp = h.copy()
                h_tmp[i] -= delta
                h_tmp[j] += delta
                # projection keeps feasibility
                h_tmp = _project_box_sum(h_tmp, target_sum)
                cand_c5 = _compute_c5(h_tmp)
                if cand_c5 < cur_c5:
                    h = h_tmp
                    cur_c5 = cand_c5
                    if cand_c5 < best_c5:
                        best_c5 = cand_c5
                        best_h = h_tmp.copy()

        elapsed = time.time() - start

    # final fine‑tuning with tiny random swaps
    for _ in range(5000):
        i, j = np.random.choice(N, 2, replace=False)
        max_transfer = min(0.02, best_h[i], 1.0 - best_h[j])
        if max_transfer <= 0:
            continue
        delta = max_transfer * random.random()
        h_try = best_h.copy()
        h_try[i] -= delta
        h_try[j] += delta
        h_try = _project_box_sum(h_try, target_sum)
        cand = _compute_c5(h_try)
        if cand < best_c5:
            best_c5 = cand
            best_h = h_try

    # final projection to guarantee exact sum
    best_h = _project_box_sum(best_h, target_sum)
    final_c5 = _compute_c5(best_h)

    return (best_h, float(final_c5), N)
