def compute_c5(h):
    n = len(h)

    corr = np.correlate(h, 1 - h, mode='full')

def project_to_bounds_and_sum(x, target, lower=0., upper=1., tol=1e-9, max_iter=50):
    lo = -1.0
    hi = 1.0
    # expand range adaptively
    for _ in range(max_iter):
        mid = (lo+hi)/2.0
        proj = np.clip(x - mid, lower, upper)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo+hi)/2.0, lower, upper)

def gradient_descent_step(h, lag):
    n = len(h)
    if lag == 0:
        g = 1.0 - 2.0 * h
    else:
        g = np.ones_like(h)
        if lag > 0:
            g[:n - lag] -= h[lag:]
            g[lag:] -= h[:n - lag]
        else:
            k = -lag
            g[k:] -= h[:n - k]
            g[:n - k] -= h[k:]

    start = time.time()
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_val = compute_c5(h)

    # Already ensured by projection.

    step0 = 0.5
    while time.time() - start < time_budget:

        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)

import numpy as np, random, time

def _compute_c5(h):
    """Return the C5 value for a discretised step function h."""
    n = h.shape[0]
    return (2.0 / n) * np.max(np.correlate(h, 1.0 - h, mode='full'))

def _project_onto_box_and_sum(x, target, lo=0.0, hi=1.0, max_iter=50):
    """Project x onto the hyperplane sum(x)=target while respecting lo≤x≤hi."""
    # binary search for lambda such that sum(clip(x-lambda)) = target
    lam_low, lam_high = -10.0, 10.0
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) * 0.5
        proj = np.clip(x - lam_mid, lo, hi)
        s = proj.sum()
        if s > target:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(x - (lam_low + lam_high) * 0.5, lo, hi)

def _grad_max_overlap(h, lag):
    """Gradient of the maximal overlap term c_k for a given lag."""
    n = h.shape[0]
    if lag == 0:
        return 1.0 - 2.0 * h
    g = np.ones_like(h)
    if lag > 0:
        g[:n - lag] -= h[lag:]
        g[lag:]     -= h[:n - lag]
    else:
        k = -lag
        g[k:]      -= h[:n - k]
        g[:n - k]  -= h[k:]
    return g

def _hill_climb(start_h, time_budget):
    """Random‑pair hill climbing with occasional gradient steps."""
    start = time.time()
    n = start_h.shape[0]
    target_sum = n / 2.0
    cur_h = start_h.copy()
    cur_val = _compute_c5(cur_h)
    best_h, best_val = cur_h.copy(), cur_val

    # parameters for the random walk
    max_step = 0.25          # absolute bound for a single coordinate change
    accept_decay = 0.0002    # temperature‑like decay

    it = 0
    while time.time() - start < time_budget:
        # ---- random pair move ----
        i, j = np.random.choice(n, 2, replace=False)
        # feasible delta range keeping both coords in [0,1]
        delta_min = max(-cur_h[i], cur_h[j] - 1.0)
        delta_max = min(1.0 - cur_h[i], cur_h[j])
        if delta_min >= delta_max:
            continue
        # bound the magnitude by max_step
        lo = max(delta_min, -max_step)
        hi = min(delta_max,  max_step)
        if lo >= hi:
            continue
        delta = np.random.uniform(lo, hi)
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_val = _compute_c5(new_h)

        # acceptance rule (simulated‑annealing flavour)
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / (0.001 + it * accept_decay)):
            cur_h, cur_val = new_h, new_val
            if new_val < best_val:
                best_h, best_val = new_h.copy(), new_val

        # ---- occasional gradient refinement ----
        if it % 250 == 0 and it > 0:
            corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
            lag = np.argmax(corr) - (n - 1)
            g = _grad_max_overlap(cur_h, lag)
            step = 0.05 / (1.0 + it * 1e-4)   # diminishing step size
            trial = _project_onto_box_and_sum(cur_h - step * g, target_sum)
            trial_val = _compute_c5(trial)
            if trial_val <= cur_val:
                cur_h, cur_val = trial, trial_val
                if trial_val < best_val:
                    best_h, best_val = trial.copy(), trial_val

        # slowly shrink max_step to focus search
        if it % 1000 == 0 and it > 0:
            max_step = max(0.01, max_step * 0.95)

        it += 1

    # final projection (numerical safety)
    best_h = _project_onto_box_and_sum(best_h, target_sum)
    best_val = _compute_c5(best_h)
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) achieving a low C5 bound."""
    random.seed(seed)
    np.random.seed(seed)
    start_total = time.time()

    # obtain parent if present
    base_h = None
    if 'initial_h_values' in globals():
        base_h = np.asarray(initial_h_values, dtype=float)

    # candidate point counts (keep them modest for speed)
    cand_ns = []
    if base_h is not None:
        nb = base_h.shape[0]
        cand_ns = [nb, nb * 2, nb * 3, nb * 4]
    else:
        cand_ns = [70, 100, 150, 200, 280, 350]
    cand_ns = [n for n in cand_ns if n <= 800]
    cand_ns = sorted(set(cand_ns))

    best_h = None
    best_val = float('inf')
    best_n = None

    # allocate time roughly evenly among candidates
    for idx, n in enumerate(cand_ns):
        elapsed = time.time() - start_total
        if elapsed >= budget_s * 0.97:
            break
        remaining = budget_s - elapsed
        remaining_cands = len(cand_ns) - idx
        # give a bit less than equal share to keep margin
        time_for_this = remaining / remaining_cands * 0.93

        # initialise h for this n
        if base_h is not None and base_h.shape[0] == n:
            h0 = base_h.copy()
        elif base_h is not None:
            # linear interpolation of the parent to new resolution
            x_old = np.linspace(0.0, 2.0, base_h.shape[0], endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(x_new, x_old, base_h)
        else:
            # start from a random binary configuration with half mass
            half = n // 2
            h0 = np.concatenate([np.ones(half), np.zeros(n - half)])
            np.random.shuffle(h0)
            # add a tiny random noise to avoid exact 0/1 stagnation
            h0 += np.random.uniform(-0.02, 0.02, size=n)
        # enforce constraints before optimisation
        h0 = _project_onto_box_and_sum(h0, n / 2.0)

        # optimise
        h_opt, val_opt = _hill_climb(h0, time_for_this)

        if val_opt < best_val:
            best_val, best_h, best_n = val_opt, h_opt.copy(), n

    # fallback in the unlikely event no solution was found
    if best_h is None:
        n = 70
        best_h = np.full(n, 0.5)
        best_val = _compute_c5(best_h)
        best_n = n

    return best_h, float(best_val), int(best_n)
