            delta = rng.random() * delta_max

        cur[i] += delta
        cur[j] -= delta
        # enforce bounds amplitude; ensure within [0,1] due to delta choices

import time
import numpy as np

def _evaluate(h):
    """Return C5 for a given array h."""
    n = h.shape[0]
    # full correlation h * (1-h)
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(conv.max() * (2.0 / n))

def _make_binary_from_continuous(arr):
    """Return a binary vector with the same length and half the entries =1 (rounded)."""
    n = arr.shape[0]
    half = int(round(n / 2))
    binary = np.zeros(n, dtype=float)
    if half == 0:
        return binary
    idx = np.argpartition(arr, -half)[-half:]
    binary[idx] = 1.0
    return binary

def _project_sum_one(h, target):
    """Adjust a single element to enforce exact sum=target (keeping 0<=h<=1)."""
    diff = h.sum() - target
    if abs(diff) < 1e-12:
        return h
    # try to correct with a single coordinate
    if diff > 0:
        # need to reduce sum
        cand = np.where(h > diff)[0]
        if cand.size:
            h[cand[0]] -= diff
            return h
    else:
        # need to increase sum
        cand = np.where(h + (-diff) <= 1.0)[0]
        if cand.size:
            h[cand[0]] -= diff  # diff is negative, so this adds
            return h
    # fallback: distribute diff over two coordinates
    i = np.argmax(h)
    j = np.argmin(h)
    delta = diff / 2.0
    if h[i] - delta >= 0 and h[i] - delta <= 1:
        h[i] -= delta
        h[j] += delta
    else:
        # very unlikely, just clip
        h = np.clip(h, 0.0, 1.0)
        excess = h.sum() - target
        if abs(excess) > 1e-12:
            # distribute uniformly
            h += -excess / n
            h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] with low C5."""
    rng = np.random.default_rng(seed)
    # ------------------------------------------------------------------ #
    # initialise h ----------------------------------------------------- #
    try:
        # `initial_h_values` is supplied by the harness when available
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    except Exception:
        # fallback: a modest even size
        n = 42
        half = n // 2
        h0 = np.zeros(n, dtype=float)
        idx = rng.choice(n, size=half, replace=False)
        h0[idx] = 1.0

    target_sum = n / 2.0
    # ensure feasibility (exact sum, values in [0,1])
    if abs(h0.sum() - target_sum) > 1e-9:
        # produce a binary feasible vector based on the largest entries
        h0 = _make_binary_from_continuous(h0)

    # work with a binary representation – this is the main search space
    current = _make_binary_from_continuous(h0).astype(float)

    best_h = current.copy()
    best_val = _evaluate(best_h)

    start_time = time.time()
    max_time = budget_s * 0.95   # keep a safety margin

    # ------------------------------------------------------------------ #
    # main optimisation loop -------------------------------------------- #
    while time.time() - start_time < max_time:
        # decide initial point for this descent
        if rng.random() < 0.2:
            # fresh random start
            cur = np.zeros(n, dtype=float)
            idx = rng.choice(n, size=int(target_sum), replace=False)
            cur[idx] = 1.0
        else:

            swaps = rng.integers(1, 4)      # 1‑3 random swaps
            for _ in range(swaps):

                    break

                cur[i], cur[j] = cur[j], cur[i]

            best_h = h_cont.copy()
        else:
            # revert move
            h_cont[i] -= delta
            h_cont[j] += delta
