def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    np.random.seed(seed)
    start = time.time()
    n = 200
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=np.float64)
        n = max(n, len(init))
        if n % 2 == 1:
            n += 1
    else:
        # n already set
        pass
    dx = 2.0 / n
    target = n // 2

    # initialize binary vector
    h = np.zeros(n, dtype=np.float64)
    # use upsample from initial if present, else random
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=np.float64)
        # Interpolate to n points
        x_init = np.linspace(0, 2, len(init), endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        interp = np.interp(x_new, x_init, init)
        # Set top target values to 1
        top_idx = np.argsort(interp)[-target:]
        h[top_idx] = 1.0
    else:
        # random
        idx = np.random.choice(n, target, replace=False)
        h[idx] = 1.0

    # compute objective
    def max_corr(vec):
        return np.max(np.correlate(vec, 1.0 - vec, mode='full')) * dx

    best_h = h.copy()
    best_val = max_corr(best_h)

    # Simulated annealing parameters
    total_time = budget_s
    init_temp = best_val * 0.5
    # may also set min_temp near 1e-6
    # loop
    iteration = 0
    while True:
        if time.time() - start > total_time - 0.5:
            break
        iteration += 1
        # compute full correlation for current h
        corr_full = np.correlate(h, 1.0 - h, mode='full')
        # find shift with max correlation
        max_idx = np.argmax(corr_full)
        shift = max_idx - (n - 1)

        # try targeted swap
        indices = np.arange(n)
        shifted = indices + shift
        valid_mask = (shifted >= 0) & (shifted < n)
        i_valid = indices[valid_mask]
        j_shifted = shifted[valid_mask]

        # produce arrays of bad and good positions
        # only need where h[i]=1 & h[shifted]=0 (bad)
        # and h[i]=0 & h[shifted]=1 (good)
        h_i = h[i_valid]
        h_j = h[j_shifted]
        bad_mask = (h_i == 1.0) & (h_j == 0.0)
        good_mask = (h_i == 0.0) & (h_j == 1.0)

        if np.any(bad_mask) and np.any(good_mask):
            i_candidates = i_valid[bad_mask]
            j_candidates = i_valid[good_mask]  # note j must be index where h[i]=0 and h[shifted]=1
            i_swap = np.random.choice(i_candidates)
            j_swap = np.random.choice(j_candidates)
            new_h = h.copy()
            new_h[i_swap], new_h[j_swap] = new_h[j_swap], new_h[i_swap]
        else:
            # fallback to random swap of any 0,1
            # get indices of ones and zeros
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            i_swap = np.random.choice(ones)
            j_swap = np.random.choice(zeros)
            new_h = h.copy()
            new_h[i_swap], new_h[j_swap] = new_h[j_swap], new_h[i_swap]

        new_val = max_corr(new_h)

        # acceptance
        # compute temperature schedule: linear cooling from init_temp to 0
        elapsed = time.time() - start
        T = init_temp * max(0.0, 1.0 - elapsed / total_time)
        if new_val < best_val:
            best_val = new_val
            best_h = new_h.copy()
            h = new_h
        else:
            # Metropolis acceptance:
            if T > 1e-12 and np.random.rand() < math.exp(-(new_val - best_val) / T):
                h = new_h

        # occasionally update best_h even if not new best? no.

    return (best_h.tolist(), float(best_val), n)

import time, math
import numpy as np

def _max_corr(h, dx):
    """Maximum overlap integral for a binary step function."""
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_time = budget_s

    # ----- choose discretisation size -----
    N = 256                     # even, power of two – good trade‑off
    if N % 2:
        N += 1
    dx = 2.0 / N
    half = N // 2               # required number of 1‑values

    # ----- initialise h -----
    try:
        init_arr = np.array(initial_h_values, dtype=np.float64)
        if init_arr.ndim != 1:
            init_arr = init_arr.ravel()
    except Exception:
        init_arr = None

    if init_arr is not None:
        # Upsample to N points and binarise by keeping the half largest values
        x_old = np.linspace(0.0, 2.0, init_arr.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, N, endpoint=False)
        interp = np.interp(x_new, x_old, init_arr)
        order = np.argsort(interp)
        h = np.zeros(N, dtype=np.float64)
        h[order[-half:]] = 1.0
    else:
        # Thue‑Morse sequence (balanced binary pattern)
        tm = np.array([bin(i).count('1') % 2 for i in range(N)], dtype=np.float64)
        h = tm.copy()  # already has exactly N/2 ones

    # ----- initial evaluation -----
    corr_cur = np.correlate(h, 1.0 - h, mode='full')
    cur_val = corr_cur.max() * dx
    best_h = h.copy()
    best_val = cur_val

    # ----- simulated annealing parameters -----
    T0 = cur_val * 0.5          # initial temperature
    it = 0

    # ----- main optimisation loop -----
    while True:
        if time.time() - start > total_time - 0.5:
            break
        it += 1

        # shift that currently dominates
        shift_idx = int(corr_cur.argmax())
        shift = shift_idx - (N - 1)

        # locate “bad” (1,0) and “good” (0,1) positions for this shift
        idx = np.arange(N)
        shifted = idx + shift
        mask = (shifted >= 0) & (shifted < N)
        i_valid = idx[mask]
        j_shifted = shifted[mask]

        h_i = h[i_valid]
        h_j = h[j_shifted]

        bad = (h_i == 1.0) & (h_j == 0.0)
        good = (h_i == 0.0) & (h_j == 1.0)

        if bad.any() and good.any():
            i_swap = rng.choice(i_valid[bad])
            j_swap = rng.choice(i_valid[good])
            new_h = h.copy()
            new_h[i_swap], new_h[j_swap] = new_h[j_swap], new_h[i_swap]
        else:
            # simple random 0‑1 swap
            ones = np.flatnonzero(h)
            zeros = np.flatnonzero(1.0 - h)
            if ones.size == 0 or zeros.size == 0:
                break
            i_swap = rng.choice(ones)
            j_swap = rng.choice(zeros)
            new_h = h.copy()
            new_h[i_swap], new_h[j_swap] = 0.0, 1.0

        # evaluate candidate
        new_corr = np.correlate(new_h, 1.0 - new_h, mode='full')
        new_val = new_corr.max() * dx

        # temperature schedule (linear cooling)
        elapsed = time.time() - start
        T = T0 * max(0.0, 1.0 - elapsed / total_time)

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        elif T > 1e-12 and rng.random() < math.exp(-(new_val - cur_val) / T):
            accept = True

        if accept:
            h = new_h
            corr_cur = new_corr
            cur_val = new_val
            if new_val < best_val - 1e-15:
                best_val = new_val
                best_h = new_h.copy()

    # ----- final result -----
    return (best_h.tolist(), float(best_val), N)
