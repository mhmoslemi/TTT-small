def project_to_sum_and_bounds(x, target, max_iter=10):

    for _ in range(max_iter):
        s = x.sum()
        diff = target - s
        if abs(diff) < 1e-9:
            break
        # indices that can be moved positively (if diff >0) or negatively (if diff <0)
        if diff > 0:
            free = (x < 1)
            # total capacity to increase
            cap = np.sum(1 - x[free])
            if cap <= 0:
                break
            # distribute diff proportionally to capacity maybe equal share
            # compute increment for free indices proportionally to their capacity
            # to ensure not exceed bound, we clip after distribution iteratively.
            inc = diff / np.sum(free)  # naive equal distribution
            # We need to ensure we don't overshoot: if inc > max for some free, they'd be clipped next iteration anyway.
            x[free] += inc
        else:
            free = (x > 0)
            cap = np.sum(x[free])
            if cap <= 0:
                break
            dec = diff / np.sum(free)  # diff negative
            x[free] += dec  # dec negative

        x = np.clip(x, 0, 1)
    # final adjustment if sum still not exact due to numeric rounding: adjust small epsilon distributed among free variables that are not at bounds.

import time
import numpy as np

def _project_to_sum_and_bounds(x: np.ndarray, target: float) -> np.ndarray:
    """Clip x to [0,1] and adjust to have sum == target."""
    x = np.clip(x, 0.0, 1.0)
    # simple iterative equal‑share correction
    for _ in range(15):
        s = x.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = x < 1.0
            if not np.any(free):
                break
            inc = diff / free.sum()
            x[free] += inc
        else:
            free = x > 0.0
            if not np.any(free):
                break
            dec = diff / free.sum()
            x[free] += dec
        x = np.clip(x, 0.0, 1.0)
    # final tiny tweak if needed
    s = x.sum()
    diff = target - s
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        if np.any(mask):
            idx = int(np.flatnonzero(mask)[0])
            delta = min(max(diff, -x[idx]), 1.0 - x[idx])
            x[idx] += delta
    return x


def _evaluate(h: np.ndarray, dx: float) -> float:
    """Return max_k ∫ h(x)(1‑h(x+k)) dx approximated by correlation."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a step function h, its C5 bound, and number of samples."""
    start_time = time.time()
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Determine problem size and start vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        init_h = np.array(initial_h_values, dtype=float)
        n = init_h.size
    else:
        # fallback size (still small enough for fast search)
        n = 62
        init_h = None

    dx = 2.0 / n
    target_sum = n / 2.0

    # ------------------------------------------------------------------
    # Helper objects
    # ------------------------------------------------------------------
    def project(v):
        return _project_to_sum_and_bounds(v, target_sum)

    def fitness(v):
        return _evaluate(v, dx)

    # ------------------------------------------------------------------
    # Initialise population
    # ------------------------------------------------------------------
    pop_size = max(50, n * 2)          # modest but diverse
    pop = np.empty((pop_size, n), dtype=float)
    fit = np.empty(pop_size, dtype=float)

    # first individual = provided construction (if any)
    start_idx = 0
    if init_h is not None:
        pop[0] = project(init_h.copy())
        fit[0] = fitness(pop[0])
        start_idx = 1

    # remaining individuals = random feasible vectors
    for i in range(start_idx, pop_size):
        rnd = np.random.rand(n)
        pop[i] = project(rnd)
        fit[i] = fitness(pop[i])

    best_idx = int(np.argmin(fit))
    best_h = pop[best_idx].copy()
    best_f = float(fit[best_idx])

    # ------------------------------------------------------------------
    # Differential Evolution (global search)
    # ------------------------------------------------------------------
    F = 0.78
    Cr = 0.92
    max_gen = 5000  # will be cut by time limit
    gen = 0
    # allocate roughly 70 % of budget to DE
    de_cutoff = start_time + budget_s * 0.70

    while gen < max_gen and time.time() < de_cutoff:
        for i in range(pop_size):
            # mutation: a + F*(b - c)
            idxs = list(range(pop_size))
            idxs.remove(i)
            a, b, c = np.random.choice(idxs, 3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            mutant = project(mutant)

            # crossover
            cross = np.random.rand(n) < Cr
            if not np.any(cross):
                cross[np.random.randint(n)] = True
            trial = np.where(cross, mutant, pop[i])
            trial = project(trial)

            f_trial = fitness(trial)

            # selection
            if f_trial <= fit[i]:
                pop[i] = trial
                fit[i] = f_trial
                if f_trial < best_f:
                    best_f = f_trial
                    best_h = trial.copy()
        gen += 1

    # ------------------------------------------------------------------
    # Simple stochastic local search (refinement)
    # ------------------------------------------------------------------
    remaining = max(0.0, budget_s - (time.time() - start_time))
    if remaining > 0.0:
        cur_h = best_h.copy()
        cur_f = best_f
        # start with a modest temperature
        T = 0.05
        cooling = 0.9995
        ls_start = time.time()
        while time.time() - ls_start < remaining:
            # pick two distinct indices and exchange a small amount
            i, j = np.random.choice(n, 2, replace=False)
            # amount that can be moved from j to i
            max_inc = min(1.0 - cur_h[i], cur_h[j])
            max_dec = min(cur_h[i], 1.0 - cur_h[j])
            if max_inc <= 0 and max_dec <= 0:
                continue
            if np.random.rand() < 0.5 and max_inc > 0:
                delta = np.random.rand() * max_inc
            elif max_dec > 0:
                delta = -np.random.rand() * max_dec
            else:
                continue

            new_h = cur_h.copy()
            new_h[i] += delta
            new_h[j] -= delta

            f_new = fitness(new_h)

            # Metropolis acceptance
            if f_new < cur_f or np.random.rand() < np.exp((cur_f - f_new) / max(T, 1e-12)):
                cur_h, cur_f = new_h, f_new
                if cur_f < best_f:
                    best_f, best_h = cur_f, cur_h.copy()

            T *= cooling
            if T < 1e-12:  # keep temperature from vanishing completely
                T = 1e-12

    # final safety projection and evaluation
    best_h = project(best_h)
    final_f = fitness(best_h)

    # Return plain Python list for compatibility
    return best_h.tolist(), float(final_f), int(n)
