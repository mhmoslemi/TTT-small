def evaluate_h(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient_for_shift(h, shift):
    n = len(h)
    term_pos = np.zeros(n)
    term_neg = np.zeros(n)
    if shift >= 0:
        s = shift
        # term_pos = h_{k+shift} for k where k+shift < n
        term_pos[:n - s] = h[s:]
        term_neg[s:] = h[:n - s]
    else:
        s = -shift
        term_pos[s:] = h[:n - s]
        term_neg[:n - s] = h[s:]
    return 1.0 - (term_pos + term_neg)

def project_to_sum_bounds(v, target_sum):
    # Solve min ||w - v||^2 s.t. sum w = target_sum, 0 ≤ w ≤ 1.
    # Use binary search on lambda as described.
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    # after clipping, due to numerical precision, sum may be slightly off; do a final correction:
    diff = w.sum() - target_sum
    if abs(diff) > 1e-8:
        # adjust via distributing diff on all non-edge entries that are not at bounds
        mask = (w > 1e-8) & (w < 1 - 1e-8)
        if np.any(mask):
            w[mask] -= diff / np.sum(mask)
            w = np.clip(w, 0.0, 1.0)
        else:
            # fallback uniformly adjust all.
            w -= diff / len(w)
            w = np.clip(w, 0.0, 1.0)
    return w

def compute_gradient_for_shift(h, shift):
    n = h.shape[0]
    term = np.zeros_like(h)
    if shift >= 0:
        s = shift
        if s < n:
            term[:n - s] += h[s:]
            term[s:] += h[:n - s]
    else:
        s = -shift
        if s < n:
            term[s:] += h[:n - s]
            term[:n - s] += h[s:]
    return 1.0 - term

import time
import numpy as np

def evaluate_h(h):
    """Return C5 for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx


def project_to_sum_bounds(v, target):
    """
    Euclidean projection of v onto {w : sum(w)=target , 0<=w<=1}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo, hi = -1.0, 2.0  # bounds that force all entries to 1 or 0 respectively
    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

    # tiny correction for numerical drift
    diff = w.sum() - target
    if abs(diff) > 1e-10:
        mask = (w > 1e-8) & (w < 1 - 1e-8)
        if mask.any():
            w[mask] -= diff / mask.sum()
            w = np.clip(w, 0.0, 1.0)
        else:
            # fallback uniform adjustment
            w -= diff / w.size
            w = np.clip(w, 0.0, 1.0)
    return w


def random_feasible_h(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    v = rng.random(n)
    return project_to_sum_bounds(v, n / 2.0)


def compute_gradient_for_shift(h, shift):
    """
    Gradient of f_shift(h) = Σ h[i] (1 - h[i+shift]) w.r.t. h.
    shift can be negative; out‑of‑range indices are ignored.
    """
    n = h.shape[0]
    term = np.zeros_like(h)
    if shift >= 0:
        s = shift
        if s < n:
            term[: n - s] += h[s:]
            term[s:] += h[: n - s]
    else:
        s = -shift
        if s < n:
            term[s:] += h[: n - s]
            term[: n - s] += h[s:]
    return 1.0 - term  # ∂f/∂h


def optimize_h_gradient(h0, time_budget, rng):
    """
    Local optimisation using gradient steps on the worst shift.
    Returns the best vector found and its C5 value.
    """
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_c = evaluate_h(h)
    no_improve = 0

    while time.time() - start < time_budget:
        # ----- evaluate current worst shift -----
        corr = np.correlate(h, 1 - h, mode="full")
        idx_max = int(np.argmax(corr))
        cur_c = corr[idx_max] * (2.0 / n)

        if cur_c < best_c - 1e-12:
            best_c = cur_c
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # early escape if we already beat the target record
        if best_c < 0.3808:
            break

        # occasional random shake‑up if stuck
        if no_improve > 500:
            idxs = rng.choice(n, size=3, replace=False)
            vals = h[idxs].copy()
            # random redistribution preserving total mass
            rand = rng.random(3)
            rand /= rand.sum()
            new_vals = np.clip(vals.sum() * rand, 0.0, 1.0)
            # fix tiny sum error
            diff = vals.sum() - new_vals.sum()
            for k in range(3):
                if diff > 0 and new_vals[k] < 1.0:
                    add = min(diff, 1.0 - new_vals[k])
                    new_vals[k] += add
                    diff -= add
                elif diff < 0 and new_vals[k] > 0.0:
                    sub = min(-diff, new_vals[k])
                    new_vals[k] -= sub
                    diff += sub
            h[idxs] = new_vals
            no_improve = 0
            continue

        shift = idx_max - (n - 1)
        grad = compute_gradient_for_shift(h, shift)
        d_raw = -grad                      # direction that decreases the worst value
        d = d_raw - d_raw.mean()           # enforce zero‑sum (keeps total mass)

        # maximal step respecting [0,1] bounds
        pos = d > 0
        neg = d < 0
        if pos.any():
            alpha_pos = np.min((1.0 - h[pos]) / d[pos])
        else:
            alpha_pos = np.inf
        if neg.any():
            alpha_neg = np.min(h[neg] / (-d[neg]))
        else:
            alpha_neg = np.inf
        alpha_max = min(alpha_pos, alpha_neg)

        if not np.isfinite(alpha_max) or alpha_max <= 0:
            # fallback to a tiny random exchange
            i, j = rng.choice(n, 2, replace=False)
            direction = 1 if rng.random() < 0.5 else -1
            if direction == 1:
                limit = min(1.0 - h[i], h[j])
            else:
                limit = min(h[i], 1.0 - h[j])
                direction = -1
            if limit <= 0:
                continue
            delta = direction * rng.random() * limit
            h[i] += delta
            h[j] -= delta
            continue

        # step size: start large, shrink as time runs out
        elapsed = time.time() - start
        frac = max(0.05, 1.0 - elapsed / time_budget)
        alpha = frac * alpha_max

        h_new = h + alpha * d
        h_new = np.clip(h_new, 0.0, 1.0)
        if abs(h_new.sum() - target_sum) > 1e-9:
            h_new = project_to_sum_bounds(h_new, target_sum)

        # evaluate new candidate
        corr_new = np.correlate(h_new, 1 - h_new, mode="full")
        new_c = corr_new[np.argmax(corr_new)] * (2.0 / n)

        if new_c <= cur_c + 1e-12:
            h = h_new
        else:
            # simulated‑annealing style acceptance
            temp = max(0.0, (time_budget - elapsed) / time_budget) * 0.01
            if rng.random() < np.exp(-(new_c - cur_c) / (temp + 1e-12)):
                h = h_new

    best_c = evaluate_h(best_h)
    return best_h, float(best_c)
