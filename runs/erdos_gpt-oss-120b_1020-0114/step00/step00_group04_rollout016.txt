def project_feasible(v, target):
    v = np.clip(v, 0, 1)
    # compute need = target - sum(v)
    need = target - v.sum()
    # iterate while |need| > 1e-12:
    for _ in range(100):
        if abs(need) < 1e-12:
            break
        if need > 0:
            # capacity to increase
            cap = 1 - v
            total_cap = cap.sum()
            if total_cap == 0:
                break
            # allocate proportionally
            delta = need * (cap / total_cap)
            v += delta
        else:
            cap = v   # capacity to decrease
            total_cap = cap.sum()
            if total_cap == 0:
                break
            delta = need * (cap / total_cap)  # need negative
            v += delta  # because need negative so subtract
        need = target - v.sum()
    # final clip again
    v = np.clip(v, 0, 1)
    # possible small residual; adjust by small tweak using first index
    if abs(v.sum() - target) > 1e-9:
        # adjust first element within bounds
        diff = target - v.sum()
        # try to add/subtract to first element respecting bounds
        i = 0
        if diff > 0:
            v[i] = min(v[i] + diff, 1.0)
        else:
            v[i] = max(v[i] + diff, 0.0)
    return v

def project_feasible(v, target):
    v = np.clip(v, 0, 1)
    for _ in range(100):
        s = v.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # Determine free indices (non-bound), which can be increased (if diff>0) or decreased
        if diff > 0:
            free = v < 1
        else:
            free = v > 0
        if not free.any():
            break
        # Compute total capacity:
        if diff > 0:
            cap = 1 - v[free]
        else:
            cap = v[free]
        total_cap = cap.sum()
        # If total capacity zero, break
        if total_cap == 0:
            break
        # amount to distribute:
        amount = diff * (cap / total_cap)
        v[free] += amount
        # Clip again to ensure not exceed bounds
        v = np.clip(v, 0, 1)
    # final fix if small residual
    diff = target - v.sum()
    if abs(diff) > 1e-9:
        # adjust first element within bounds
        for i in range(len(v)):
            if diff > 0:
                add = min(diff, 1 - v[i])
                v[i] += add
                diff -= add
            else:
                sub = min(-diff, v[i])
                v[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return v

import time
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return the C5 value for a given step function vector."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(conv.max() * (2.0 / h.size))

def _project_to_bounds_and_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project a vector onto the box [0,1] while keeping the total sum equal to target.
    Simple iterative redistribution algorithm; sufficient for modest dimensions.
    """
    v = np.clip(v, 0.0, 1.0)
    for _ in range(100):
        s = v.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = v < 1.0
        else:
            free = v > 0.0
        if not free.any():
            break
        if diff > 0:
            cap = 1.0 - v[free]
        else:
            cap = v[free]
        total_cap = cap.sum()
        if total_cap == 0.0:
            break
        # distribute proportionally to capacity
        delta = diff * (cap / total_cap)
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    # final tiny correction if needed
    residual = target - v.sum()
    if abs(residual) > 1e-9:
        for i in range(v.size):
            if residual > 0:
                add = min(residual, 1.0 - v[i])
                v[i] += add
                residual -= add
            else:
                sub = min(-residual, v[i])
                v[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return v

    Optimize the Erdős minimum overlap constant C5.

    best_h = h_cur.copy()

    # Simple simulated‑annealing parameters
    T = 0.02                # initial temperature
    T_min = 1e-6

    it = 0
    # Main optimisation loop
    while time.time() - start_time < budget_s * 0.98:
        # ---- propose a zero‑sum swap move ----
        i = rng.integers(n_points)
        j = rng.integers(n_points - 1)
        if j >= i:
            j += 1  # guarantee i != j

        direction = 1 if rng.integers(2) == 0 else -1  # 1: inc i / dec j, -1: opposite
        if direction == 1:
            delta_max = min(1.0 - h_cur[i], h_cur[j])

            delta_max = min(1.0 - h_cur[j], h_cur[i])

            continue
