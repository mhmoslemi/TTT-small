import time
import math
import numpy as np

def _compute_c5(arr: np.ndarray) -> float:
    """Return the C5 value for a discretised step function."""
    n = arr.shape[0]
    dx = 2.0 / n
    # correlation of h with (1‑h); mode='full' gives all shifts
    corr = np.correlate(arr, 1.0 - arr, mode="full") * dx
    return float(corr.max())

def _make_random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1]^n whose sum is exactly n/2."""
    target = n / 2.0
    while True:
        v = rng.random(n)        # uniform in [0,1)
        cur = v.sum()
        factor = target / cur
        v = v * factor
        if (v <= 1.0).all():
            return v

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Perform a simple simulated‑annealing optimisation of the Erdős
    overlap functional.  Returns a tuple (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Initialise the discretisation.
    # ------------------------------------------------------------------
    try:
        base_h = initial_h_values                     # provided by the harness
        base_n = int(base_h.shape[0])
        repeat_factor = 2                             # double the resolution
        h = np.repeat(base_h.astype(np.float64), repeat_factor)
        n = h.shape[0]
    except Exception:                                 # no parent supplied
        n = 140                                         # a modest size (<1000)
        h = _make_random_feasible(n, rng)

    target_sum = n / 2.0
    # tiny post‑processing to guarantee the integral exactly (possible rounding)
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        diff = target_sum - cur_sum
        # try to fix a single entry
        if diff > 0:
            feasible = np.where(h < 1.0)[0]
        else:
            feasible = np.where(h > 0.0)[0]
        if feasible.size:
            idx = rng.choice(feasible)
            add = max(min(diff, 1.0 - h[idx]), -h[idx])
            h[idx] += add
        else:
            # fallback: scale and clip
            h = np.clip(h * (target_sum / cur_sum), 0.0, 1.0)
            # final tiny correction
            cur_sum = h.sum()
            diff = target_sum - cur_sum
            if abs(diff) > 1e-12:
                idx = rng.integers(n)
                h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)

    # ------------------------------------------------------------------
    # 2.  Simulated‑annealing core.
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_c5 = _compute_c5(best_h)

    cur_h = h.copy()
    cur_c5 = best_c5

    start = time.time()
    time_limit = budget_s * 0.94            # leave a safety margin

    # annealing schedule parameters
    T0 = 0.04                                # initial temperature
    T_final = 1e-6
    move0 = 0.22                              # initial maximal move
    move_final = 0.001

    while True:
        now = time.time()
        if now - start > time_limit:
            break

        # linear schedules
        progress = (now - start) / time_limit
        T = T0 * (1.0 - progress) + T_final * progress
        max_move = move0 * (1.0 - progress) + move_final * progress

        # pick two distinct indices
        i, j = rng.choice(n, size=2, replace=False)
        hi, hj = cur_h[i], cur_h[j]

        # admissible delta (increase i, decrease j)
        max_delta_pos = min(max_move, 1.0 - hi, hj)
        max_delta_neg = min(max_move, hi, 1.0 - hj)

        if max_delta_pos <= 0.0 and max_delta_neg <= 0.0:
            continue

        delta = rng.uniform(-max_delta_neg, max_delta_pos)
        if abs(delta) < 1e-14:
            continue

        # propose new state
        cur_h[i] = hi + delta
        cur_h[j] = hj - delta

        new_c5 = _compute_c5(cur_h)

        # Metropolis acceptance
        if new_c5 < cur_c5:
            accept = True
        else:
            prob = math.exp(-(new_c5 - cur_c5) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            # revert the move
            cur_h[i] = hi
            cur_h[j] = hj

    # ------------------------------------------------------------------
    # 3.  Post‑processing – enforce exact integral and bounds.
    # ------------------------------------------------------------------
    best_sum = best_h.sum()
    diff = target_sum - best_sum
    if abs(diff) > 1e-10:
        # try a simple adjustment on a feasible entry
        if diff > 0:
            feasible = np.where(best_h < 1.0)[0]
        else:
            feasible = np.where(best_h > 0.0)[0]
        if feasible.size:
            idx = rng.choice(feasible)
            add = max(min(diff, 1.0 - best_h[idx]), -best_h[idx])
            best_h[idx] += add
        else:
            # last‑resort scaling
            best_h = np.clip(best_h * (target_sum / best_sum), 0.0, 1.0)
            # tiny leftover correction
            leftover = target_sum - best_h.sum()
            if abs(leftover) > 1e-12:
                best_h[0] = np.clip(best_h[0] + leftover, 0.0, 1.0)

    final_c5 = _compute_c5(best_h)

    # ------------------------------------------------------------------
    # 4.  Return the result.
    # ------------------------------------------------------------------
    return best_h, final_c5, n
