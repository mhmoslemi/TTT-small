def project_capped_simplex(v, target, max_iter=50, tol=1e-9):
    lo = np.min(v - 1)  # could be negative large.
    hi = np.max(v)
    for _ in range(max_iter):
        mid = (lo + hi) / 2
        x = np.clip(v - mid, 0, 1)
        s = np.sum(x)
        if abs(s - target) <= tol:
            return x
        if s > target:
            lo = mid
        else:
            hi = mid
    # after loop, return x at lo
    return np.clip(v - lo, 0, 1)

def compute_c5(h):

def simulated_annealing(n, init_h, time_budget):
    h = init_h.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    start = time.time()
    iteration = 0
    while time.time() - start < time_budget:
        # choose random swap
        i = np.random.randint(n)
        j = np.random.randint(n)
        # ensure different and opposite values
        if h[i] == h[j]:
            continue
        # swap
        h[i], h[j] = h[j], h[i]
        c = compute_c5(h)
        if c < cur_c or np.random.rand() < math.exp((cur_c - c) / max(1e-6, (0.1 * (1 - iteration / max_iter)))):
            cur_c = c
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = h[j], h[i]
        iteration += 1
    return best_h, best_c

import math
import time
import numpy as np

def _correlation(h: np.ndarray) -> np.ndarray:
    """Full correlation of h with (1-h)."""
    return np.correlate(h, 1.0 - h, mode="full")


def _c5_value(corr: np.ndarray, n: int) -> float:
    """Convert correlation array to C5 value (with dx factor)."""
    dx = 2.0 / n
    return np.max(corr) * dx


def _initial_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h


def _from_initial_h(initial, n: int, rng: np.random.Generator) -> np.ndarray:
    """Upsample an optional global initial_h_values to length n and binarise."""
    arr = np.asarray(initial, dtype=float)
    if arr.size == 0:
        return _initial_binary(n, rng)

    # repeat to reach at least n, then truncate
    rep = max(1, n // arr.size)
    h = np.repeat(arr, rep)[:n]

    # binarise (threshold 0.5) and then fix the exact number of ones
    h = (h >= 0.5).astype(float)
    target_ones = n // 2
    cur_ones = int(h.sum())
    if cur_ones > target_ones:
        idx = np.where(h == 1)[0]
        rng.shuffle(idx)
        h[idx[: cur_ones - target_ones]] = 0.0
    elif cur_ones < target_ones:
        idx = np.where(h == 0)[0]
        rng.shuffle(idx)
        h[idx[: target_ones - cur_ones]] = 1.0
    return h


def _optimize_binary(init_h: np.ndarray,
                     time_limit: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing / hill‑climbing on a binary vector.
    Returns the best vector found and its C5 value.
    """
    n = init_h.size
    dx = 2.0 / n
    target_ones = n // 2

    h = init_h.copy()
    cur_corr = _correlation(h)
    cur_c = _c5_value(cur_corr, n)

    best_h = h.copy()
    best_c = cur_c

    start = time.time()
    no_improve = 0
    stagnation_limit = 5000  # after this many failures we perturb the state

    while time.time() - start < time_limit:
        # worst shift for the current state
        max_idx = int(np.argmax(cur_corr))
        shift = max_idx - (n - 1)

        # indices that are mismatched for this shift and the opposite ones
        if shift >= 0:
            rng_idx = np.arange(0, n - shift)
            mis = (h[rng_idx] == 1) & (h[rng_idx + shift] == 0)
            opp = (h[rng_idx] == 0) & (h[rng_idx + shift] == 1)
        else:
            s = -shift
            rng_idx = np.arange(s, n)
            mis = (h[rng_idx] == 1) & (h[rng_idx - s] == 0)
            opp = (h[rng_idx] == 0) & (h[rng_idx - s] == 1)

        mis_idx = rng_idx[mis]
        opp_idx = rng_idx[opp]

        if mis_idx.size and opp_idx.size:
            i = rng.choice(mis_idx)
            j = rng.choice(opp_idx)
        else:
            # generic random 1↔0 swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            i = rng.choice(ones)
            j = rng.choice(zeros)

        # propose swap
        h[i], h[j] = 0.0, 1.0
        new_corr = _correlation(h)
        new_c = _c5_value(new_corr, n)

        # temperature decreases linearly from 0.07 to 1e‑5
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = 0.07 * (1.0 - frac) + 1e-5

        if new_c < cur_c or rng.random() < math.exp(-(new_c - cur_c) / T):
            # accept
            cur_c = new_c
            cur_corr = new_corr
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # reject
            h[i], h[j] = 1.0, 0.0
            no_improve += 1

        # occasional perturbation to escape plateaus
        if no_improve > stagnation_limit:
            # start from the best known state and flip a modest number of bits
            h = best_h.copy()
            flip = max(1, n // 20)  # ~5 % of positions
            ones_idx = np.where(h == 1)[0]
            zeros_idx = np.where(h == 0)[0]
            if ones_idx.size >= flip and zeros_idx.size >= flip:
                flip_ones = rng.choice(ones_idx, size=flip, replace=False)
                flip_zeros = rng.choice(zeros_idx, size=flip, replace=False)
                h[flip_ones] = 0.0
                h[flip_zeros] = 1.0
            cur_corr = _correlation(h)
            cur_c = _c5_value(cur_corr, n)
            no_improve = 0

    return best_h, best_c


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Find a binary step function h on [0,2] (sampled uniformly) that yields a low
    C5 bound. Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # Number of sample points – small enough for fast evaluation, large enough
    # for a reasonably fine discretisation.
    n = 200                     # must be even so that integral = 1 is feasible

    # Build the initial binary vector, possibly from the supplied global.
    if "initial_h_values" in globals():
        h0 = _from_initial_h(initial_h_values, n, rng)
    else:
        h0 = _initial_binary(n, rng)

    # Allocate most of the budget to the optimisation routine.
    allotted = budget_s * 0.95
    best_h, best_c = _optimize_binary(h0, allotted, rng)

    # Sanity check: the integral constraint must hold exactly.
    assert abs(best_h.sum() - n / 2) < 1e-9, "sum constraint violated"

    return best_h.tolist(), float(best_c), n
