import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    """return the C5 value for a step function h (array of values)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def corr_shift(h, s):
    """dot product sum_i h[i] * (1 - h[i+s]) for a given integer shift s."""
    n = h.shape[0]
    if s >= 0:
        return np.dot(h[: n - s], 1.0 - h[s:])
    else:
        s = -s
        return np.dot(h[s:], 1.0 - h[: n - s])

def optimize_h(h_start, max_iter=1000):
    """SLSQP optimisation: minimise max correlation."""
    n = h_start.shape[0]
    dx = 2.0 / n

    # bounds for h_i and t
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    # equality: sum h_i = n/2
    eq_cons = {
        "type": "eq",
        "fun": lambda x, n=n: np.sum(x[:n]) - n / 2.0,
    }

    # inequality: for each shift s, t >= sum_i h_i * (1 - h_{i+s})
    ineq = []
    for s in range(-(n - 1), n):
        def fun(x, s=s, n=n):
            h = x[:n]
            t = x[n]
            return t - corr_shift(h, s)
        ineq.append({"type": "ineq", "fun": fun})

    constraints = [eq_cons] + ineq

    # start value for t is the current max correlation of the start vector
    t0 = compute_c5(h_start)
    x0 = np.concatenate([h_start, [t0]])

    opts = {"maxiter": max_iter, "ftol": 1e-10, "disp": False}
    try:
        res = opt.minimize(
            lambda x, n=n: x[n],
            x0,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options=opts,
        )
    except Exception:
        res = None

    if res is None or not res.success:
        # fallback: use the start vector
        h_opt = np.clip(h_start, 0.0, 1.0)
        c5 = compute_c5(h_opt)
        return h_opt, c5

    h_opt = np.clip(res.x[:n], 0.0, 1.0)

    # renormalise sum exactly (tiny numerical drift)
    target = n / 2.0
    cur = h_opt.sum()
    if abs(cur - target) > 1e-8:
        # distribute the residual uniformly over non‑bound entries
        diff = target - cur
        mask = (h_opt > 0.0) & (h_opt < 1.0)
        if mask.any():
            h_opt[mask] += diff / mask.sum()
            h_opt = np.clip(h_opt, 0.0, 1.0)
        else:
            # extreme case: simply scale (should keep bounds satisfied)
            h_opt *= target / cur

    c5 = compute_c5(h_opt)
    return h_opt, c5

def random_feasible(n, rng):
    """produce a random vector in [0,1]^n with sum = n/2."""
    # start with uniform Dirichlet then scale
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    # ensure bounds by simple projection (few iterations)
    for _ in range(10):
        over = v > 1.0
        under = v < 0.0
        if not over.any() and not under.any():
            break
        excess = (v[over] - 1.0).sum()
        deficit = (-v[under]).sum()
        if excess > 0:
            # move excess to interior points
            interior = (~over) & (~under)
            if interior.any():
                v[interior] -= excess / interior.sum()
        if deficit > 0:
            interior = (~over) & (~under)
            if interior.any():
                v[interior] += deficit / interior.sum()
        v = np.clip(v, 0.0, 1.0)
    # final correction of the sum
    cur = v.sum()
    if abs(cur - n / 2.0) > 1e-8:
        diff = n / 2.0 - cur
        mask = (v > 0.0) & (v < 1.0)
        if mask.any():
            v[mask] += diff / mask.sum()
            v = np.clip(v, 0.0, 1.0)

def hill_climb(h, budget, start_time):
    """simple greedy pairwise swaps to lower C5 while respecting constraints."""
    n = h.shape[0]
    best_h = h.copy()
    best_val = compute_c5(best_h)
    rng = np.random.default_rng()
    while time.time() - start_time < budget:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # we will move mass from i to j (or opposite) depending on availability
        # determine direction that keeps values in [0,1]
        if best_h[i] > 0.0 and best_h[j] < 1.0:
            max_delta = min(best_h[i], 1.0 - best_h[j])
        elif best_h[j] > 0.0 and best_h[i] < 1.0:
            max_delta = min(best_h[j], 1.0 - best_h[i])
            i, j = j, i  # swap direction
        else:
            continue
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random()
        cand = best_h.copy()
        cand[i] -= delta
        cand[j] += delta
        val = compute_c5(cand)
        if val < best_val - 1e-9:
            best_h = cand
            best_val = val
    return best_h, best_val
