def run(seed=42, budget_s=1100, **kwargs):

    import time, numpy as np, math, random

    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    
    # get initial vector
    if 'initial_h_values' in globals():
        base = np.array(initial_h_values, dtype=np.float64)
    else:
        # fallback: simple uniform 0.5 of size 60
        base = np.full(60, 0.5, dtype=np.float64)
    
    # upsample factor
    factor = 4
    h = np.repeat(base, factor)
    n = h.size
    target_sum = n / 2.0
    dx = 2.0 / n
    
    # ensure exact sum
    # At this point sum should be exactly target (due to repeat)
    diff = h.sum() - target_sum
    if abs(diff) > 1e-12:
        # adjust by shifting small epsilon to first element
        h[0] -= diff
    
    # define functions
    def compute_bound(vec):
        c = np.correlate(vec, 1 - vec, mode='full')
        return c.max() * dx

    def project(v):
        # binary search lam to make sum = target_sum after clipping
        lo = -np.max(v) - 1.0
        hi = 1 - np.min(v) + 1.0
        for _ in range(50):
            lam = (lo + hi) * 0.5
            x = np.clip(v + lam, 0.0, 1.0)
            s = x.sum()
            if s < target_sum:
                lo = lam
            else:
                hi = lam
        return np.clip(v + (lo+hi)*0.5, 0.0, 1.0)
    
    # initial bound
    best_h = h.copy()
    best_bound = compute_bound(best_h)
    current_h = best_h.copy()
    current_bound = best_bound

    # parameters
    max_iters = 200000  # Cap; may be limited by budget
    init_step = 0.2
    final_step = 0.001
    init_beta = 1.0
    final_beta = 25.0

    # iteration loop
    it = 0
    last_improve_iter = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.97:
            break
        if it >= max_iters:
            break
        # compute correlation c (not scaled)
        c = np.correlate(current_h, 1 - current_h, mode='full')
        # compute bound only occasionally maybe?
        # compute bound to track best
        bound = c.max() * dx
        if bound < best_bound:
            best_bound = bound
            best_h = current_h.copy()
            last_improve_iter = it
            # early exit if good enough
            if best_bound <= 0.3808:
                break
        # schedule beta and step
        frac = it / max_iters
        beta = init_beta + frac * (final_beta - init_beta)
        step = init_step + (1 - frac) * (init_step - final_step)  # linear decreasing from init_step to final_step

        # compute smooth weights
        maxc = c.max()
        # subtract maxc to avoid overflow
        w = np.exp(beta * (c - maxc))
        sumw = w.sum()
        if sumw == 0:
            # fallback uniform weights
            p = np.full_like(w, 1.0 / w.size)
        else:
            p = w / sumw

        # gradient
        grad = np.zeros(n, dtype=np.float64)
        # loop over shifts
        # p indexed from 0 to 2n-2, offset = n-1
        offset = n - 1
        for idx in range(len(p)):

            if wgt == 0.0:

            s = idx - offset
            if s >= 0:
                limit = n - s

                    # term2

            else:
                t = -s

                if limit > 0:
                    grad[t:] += wgt * (1 - current_h[:limit])
                    grad[:limit] -= wgt * current_h[t:]

            last_improve_iter = it

    final_bound = compute_bound(best_h)
    return (best_h, float(final_bound), int(n))

import time
import math
import random
import numpy as np

def _compute_bound(vec, dx):
    """Return C5 bound for a step function given by vec."""
    corr = np.correlate(vec, 1.0 - vec, mode="full")
    return corr.max() * dx

def _project_onto_box_and_sum(v, target_sum):
    """Project v onto the set {0≤x≤1, sum(x)=target_sum}."""
    # binary search for lambda such that sum(clip(v+lam,0,1)) == target_sum
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s < target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def _gradient_smoothmax(h, weights):
    """Gradient of smooth‑max of the correlation array."""
    n = h.size
    grad = np.zeros(n, dtype=np.float64)
    offset = n - 1
    for idx, w in enumerate(weights):
        if w == 0.0:
            continue
        s = idx - offset
        if s >= 0:
            limit = n - s
            if limit:
                grad[:limit] += w * (1.0 - h[s:])
                grad[s:]   -= w * h[:limit]
        else:
            t = -s
            limit = n - t
            if limit:
                grad[t:]  += w * (1.0 - h[:limit])
                grad[:limit] -= w * h[t:]
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    start = time.time()
    # ------------------------------------------------------------------
    # 1. Build an initial feasible step function
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=np.float64)
    else:
        # fallback: uniform 0.5 on 60 points
        base = np.full(60, 0.5, dtype=np.float64)

    # Upsample to give more degrees of freedom
    up_factor = 4                     # 45·4 = 180 points (reasonable size)
    h = np.repeat(base, up_factor)
    n = h.size
    dx = 2.0 / n
    target_sum = n / 2.0
    # tiny correction just in case
    diff = h.sum() - target_sum
    if abs(diff) > 1e-12:
        h[0] -= diff

    # ------------------------------------------------------------------
    # 2. Optimisation – projected gradient descent on a smooth‑max
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_bound = _compute_bound(best_h, dx)

    cur_h = h.copy()
    cur_bound = best_bound

    max_iters = 200_000
    init_step = 0.20
    final_step = 0.001
    init_beta = 1.0
    final_beta = 25.0

    last_improve = 0
    for it in range(max_iters):
        # time guard
        if time.time() - start > budget_s * 0.97:
            break

        # correlation and current bound
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        bound = corr.max() * dx
        if bound < best_bound:
            best_bound = bound
            best_h = cur_h.copy()
            last_improve = it
            if best_bound <= 0.3808:
                break

        # schedule of parameters
        frac = it / max_iters
        step = init_step * (1.0 - frac) + final_step * frac
        beta = init_beta * (1.0 - frac) + final_beta * frac

        # smooth‑max weights (log‑sum‑exp trick)
        maxc = corr.max()
        w = np.exp(beta * (corr - maxc))
        p = w / w.sum()
