def search_best_pattern(N, time_deadline, rng):
    best_h = None
    best_c5 = np.inf
    # precompute coprime steps
    steps = [s for s in range(1, N) if math.gcd(s, N) == 1]
    for s in steps:
        # early break if time over
        if time.time() > time_deadline:
            break
        # we can also randomize offsets order to not waste time if best found early
        offsets = list(range(N))
        rng.shuffle(offsets)

import time, math
import numpy as np

def _c5_of(h: np.ndarray) -> float:
    """Maximum overlap integral for a discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation = convolution with reversed complement
    conv = np.convolve(h, (1.0 - h)[::-1], mode='full')
    return float(np.max(conv) * dx)


def _best_pattern(N: int, deadline: float, rng: np.random.Generator):
    """Search all coprime step sizes and all offsets for binary patterns."""
    best_h = None
    best_c5 = np.inf
    # all steps coprime with N
    steps = [s for s in range(1, N) if math.gcd(s, N) == 1]
    half = N // 2
    idx_base = np.arange(half, dtype=int)
    for s in steps:
        if time.time() >= deadline:
            break
        offsets = list(range(N))
        rng.shuffle(offsets)                         # random order, may find good one early
        for off in offsets:
            if time.time() >= deadline:
                break
            pos = (idx_base * s + off) % N
            h = np.zeros(N, dtype=float)
            h[pos] = 1.0
            c5 = _c5_of(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return best_h, best_c5


def _greedy_improve(h0: np.ndarray, deadline: float, rng: np.random.Generator):
    """Hill‑climbing with random swaps, keeping the sum constraint."""
    best_h = h0.copy()
    best_c5 = _c5_of(best_h)
    cur_h = best_h.copy()
    cur_c5 = best_c5
    N = cur_h.shape[0]

    # simple stop when many consecutive failures
    fails = 0
    max_fails = 20000

    while time.time() < deadline and fails < max_fails:
        ones = np.flatnonzero(cur_h == 1.0)
        zeros = np.flatnonzero(cur_h == 0.0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_c5 = _c5_of(cur_h)
        if new_c5 < cur_c5 - 1e-14:          # improvement
            cur_c5 = new_c5
            fails = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] = 1.0
            cur_h[j] = 0.0
            fails += 1
    return best_h, best_c5

    # ------------------------------------------------------------
    N = 240
    half = N // 2

    best_h = None

    if 'initial_h_values' in globals():

            if init.shape[0] == N and np.all((init >= 0) & (init <= 1)):
