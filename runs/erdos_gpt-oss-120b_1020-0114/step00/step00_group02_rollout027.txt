def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    # compute correlation h with (1 - h) using FFT
    a = h.astype(np.float64)
    b = (1 - h).astype(np.float64)
    # reverse b for correlation (linear)
    b_rev = b[::-1]
    Nfft = 1 << (2*n - 1).bit_length()  # next power of two >= 2n-1
    A = np.fft.rfft(a, Nfft)
    B = np.fft.rfft(b_rev, Nfft)
    conv = np.fft.irfft(A * B, Nfft)
    # conv length Nfft; linear convolution first len = 2n-1 entries in conv[:2n-1]
    corr = conv[:2*n-1]
    # Max
    max_corr = corr.max()
    return max_corr * dx

def project_to_sum_and_bounds(x, target_sum):
    # Find lambda such that sum(clipped(x + lambda)) = target_sum
    lo = -np.max(x) - 2.0
    hi = 1 - np.min(x) + 2.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        y = np.clip(x + lam, 0.0, 1.0)
        s = y.sum()
        if s < target_sum:
            lo = lam
        else:
            hi = lam
    y = np.clip(x + (lo+hi)/2.0, 0.0, 1.0)

def compute_c5_fft(h):
    n = len(h)
    dx = 2.0 / n
    a = h
    b = 1 - h
    b_rev = b[::-1]
    Nfft = 1 << ((2*n - 1).bit_length())

    B = np.fft.rfft(b_rev, Nfft)

def subgradient(h, s):
    n = len(h)
    g = np.ones(n)
    # t + s indices
    if s >= 0:
        # indices 0..n-s-1 have t+s within range
        g[:n-s] -= h[s:]   # subtract h[t+s]
        # t - s indices s..n-1 have t-s within range
        g[s:] -= h[:n-s]
    else:
        s_abs = -s
        # now t + s = t - s_abs
        # valid for t >= s_abs
        g[s_abs:] -= h[:n-s_abs]
        # t - s = t + s_abs
        # valid for t < n - s_abs
        g[:n-s_abs] -= h[s_abs:]
    return g

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    np.random.seed(seed)
    start = time.time()
    # Choose n
    n = 200  # could adjust: maybe 180
    target_sum = n/2.0

    # projection function
    def project_sum_bounds(v):
        # lambda search for lambda such that sum(clip(v+lam,0,1)) = target_sum
        lo = -np.max(v) - 2.0
        hi = 1 - np.min(v) + 2.0
        for _ in range(60):
            lam = (lo + hi) / 2.0
            x = np.clip(v + lam, 0.0, 1.0)
            s = x.sum()
            if s < target_sum:
                lo = lam
            else:
                hi = lam
        return np.clip(v + (lo+hi)/2.0, 0.0, 1.0)

    # correlation function: returns value and shift
    def compute_val_shift(h):
        nlocal = h.shape[0]
        dx = 2.0 / nlocal
        a = h
        b = 1.0 - h
        b_rev = b[::-1]
        size = 1 << ((2*nlocal - 1).bit_length())
        A = np.fft.rfft(a, size)
        B = np.fft.rfft(b_rev, size)
        conv = np.fft.irfft(A * B, size)[:2*nlocal-1]
        max_idx = np.argmax(conv)
        max_corr = conv[max_idx]
        shift = int(max_idx - (nlocal - 1))
        return max_corr * dx, shift, conv

    # subgradient
    def subgrad(h, s):
        nlocal = h.shape[0]
        g = np.ones(nlocal)
        if s >= 0:
            if s < nlocal:
                g[:nlocal-s] -= h[s:]
                g[s:] -= h[:nlocal-s]
        else:
            s_abs = -s
            if s_abs < nlocal:
                g[s_abs:] -= h[:nlocal-s_abs]
                g[:nlocal-s_abs] -= h[s_abs:]
        return g

    # Initialize h
    h = np.random.rand(n)
    # project to constraints
    h = project_sum_bounds(h)
    cur_val, cur_shift, _ = compute_val_shift(h)
    best_h = h.copy()
    best_val = cur_val

    max_iter = 10**7  # large
    iter_count = 0
    # temperature schedule for SA part
    T0 = 0.005
    # gradient step schedule parameters
    max_alpha = 0.5

    while True:
        now = time.time()
        if now - start > budget_s - 0.2:
            break
        # choose move type
        if np.random.rand() < 0.5:
            # gradient step
            # compute shift for current
            _, s, _ = compute_val_shift(h)
            g = subgrad(h, s)
            # step size
            alpha = max_alpha / np.sqrt(iter_count + 1)
            y = h - alpha * g
            y = project_sum_bounds(y)
            new_val, new_shift, _ = compute_val_shift(y)
            # accept if better or by prob
            if new_val < cur_val or np.random.rand() < math.exp(-(new_val - cur_val) / max(T0 * (1 - (now-start)/budget_s), 1e-12)):
                h = y
                cur_val = new_val
                cur_shift = new_shift
                if new_val < best_val:
                    best_val = new_val
                    best_h = h.copy()
        else:
            # random pair move
            i = np.random.randint(n)
            j = np.random.randint(n)
            while j == i:
                j = np.random.randint(n)
            hi = h[i]
            hj = h[j]
            delta_min = max(-hi, hj - 1.0)
            delta_max = min(1.0 - hi, hj)
            if delta_min >= delta_max:
                continue
            delta = np.random.uniform(delta_min, delta_max)
            y = h.copy()
            y[i] = hi + delta
            y[j] = hj - delta
            new_val, new_shift, _ = compute_val_shift(y)
            # acceptance with temperature
            T = T0 * max(0.0, 1.0 - (now - start) / budget_s)
            if new_val < cur_val or np.random.rand() < math.exp(-(new_val - cur_val) / max(T,1e-12)):
                h = y
                cur_val = new_val
                cur_shift = new_shift
                if new_val < best_val:
                    best_val = new_val
                    best_h = h.copy()
        iter_count += 1
        # occasional restart from best with perturbation
        if iter_count % 50000 == 0:
            # small perturbation of best
            h = best_h.copy()
            # add random noise and reproject
            noise = np.random.normal(scale=0.02, size=n)
            h = np.clip(h + noise, 0.0, 1.0)
            h = project_sum_bounds(h)
            cur_val, cur_shift, _ = compute_val_shift(h)

    # final projection to ensure feasibility
    best_h = project_sum_bounds(best_h)
    best_val, _, _ = compute_val_shift(best_h)
    return (best_h, float(best_val), n)

def project_to_sum_bounds(v, target_sum):
    lo = -np.max(v) - 2.0
    hi = 1.0 - np.min(v) + 2.0
    for _ in range(60):

        s = x.sum()
        if s < target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo+hi)*0.5, 0.0, 1.0)

def compute_c5_and_shift(h):
    n = h.shape[0]
    dx = 2.0 / n
    a = h
    b = 1.0 - h
    b_rev = b[::-1]
    size = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b_rev, size)
    conv = np.fft.irfft(A * B, size)[:2*n - 1]
    max_idx = int(conv.argmax())
    max_corr = conv[max_idx]
    shift = max_idx - (n - 1)
    return max_corr * dx, shift, conv

def subgradient(h, s):
    n = h.shape[0]
    g = np.ones(n, dtype=np.float64)
    if s >= 0:
        if s < n:
            g[:n-s] -= h[s:]
            g[s:] -= h[:n-s]
    else:
        s_abs = -s
        if s_abs < n:
            g[s_abs:] -= h[:n-s_abs]
            g[:n-s_abs] -= h[s_abs:]
    return g

import numpy as np
import time
import math

def project_to_sum_bounds(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto the intersection
    {x | sum(x)=target_sum, 0<=x_i<=1}."""
    lo = -np.max(v) - 2.0
    hi = 1.0 - np.min(v) + 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s < target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def compute_c5_and_shift(h: np.ndarray):
    """Return (C5 value, shift attaining the max, full correlation array)."""
    n = h.shape[0]
    dx = 2.0 / n
    a = h
    b = 1.0 - h
    b_rev = b[::-1]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b_rev, size)
    conv = np.fft.irfft(A * B, size)[: 2 * n - 1]
    max_idx = int(conv.argmax())
    max_corr = conv[max_idx]
    shift = max_idx - (n - 1)                       # shift in integer steps
    return max_corr * dx, shift, conv


def subgradient(h: np.ndarray, s: int):
    """Sub‑gradient of the maximal shift term w.r.t. h."""
    n = h.shape[0]
    g = np.ones(n, dtype=np.float64)

        if s < n:
            g[: n - s] -= h[s:]
            g[s:] -= h[: n - s]
    else:
        s_abs = -s
        if s_abs < n:
            g[s_abs:] -= h[: n - s_abs]
            g[: n - s_abs] -= h[s_abs:]
    return g

            h = init.copy()
    except NameError:                          # no supplied construction
        h = np.random.rand(n)

    h = project_to_sum_bounds(h, target_sum)
    cur_val, cur_shift, _ = compute_c5_and_shift(h)

    best_h = h.copy()
    best_val = cur_val

    # -----------------------------------------------------------------------
    iter_cnt = 0
    while True:
        now = time.time()
        if now - start > budget_s - 0.2:
            break

        # 50 % gradient step, 50 % random pair move
        if np.random.rand() < 0.5:
            # ---- gradient (sub‑gradient) move --------------------------------
            _, s, _ = compute_c5_and_shift(h)
            g = subgradient(h, s)
            alpha = max_alpha / math.sqrt(iter_cnt + 1.0)
            y = h - alpha * g
            y = project_to_sum_bounds(y, target_sum)

            new_val, new_shift, _ = compute_c5_and_shift(y)

            if new_val < cur_val:
                accept = True
            else:
                T = T0 * max(0.0, 1.0 - (now - start) / budget_s)
                accept = np.random.rand() < math.exp(-(new_val - cur_val) / max(T, 1e-12))

            if accept:
                h = y
                cur_val = new_val
                cur_shift = new_shift
                if new_val < best_val:
                    best_val = new_val
                    best_h = h.copy()
        else:
            # ---- random pair move preserving sum ------------------------------
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1
            hi, hj = h[i], h[j]

            delta_min = max(-hi, hj - 1.0)
            delta_max = min(1.0 - hi, hj)
            if delta_min < delta_max:
                delta = np.random.uniform(delta_min, delta_max)

                y = h.copy()
                y[i] = hi + delta
                y[j] = hj - delta

                new_val, new_shift, _ = compute_c5_and_shift(y)

                if new_val < cur_val:
                    accept = True
                else:
                    T = T0 * max(0.0, 1.0 - (now - start) / budget_s)
                    accept = np.random.rand() < math.exp(-(new_val - cur_val) / max(T, 1e-12))

                if accept:
                    h = y
                    cur_val = new_val
                    cur_shift = new_shift
                    if new_val < best_val:
                        best_val = new_val
                        best_h = h.copy()

        iter_cnt += 1

        # occasional restart from the best solution with a little noise
        if iter_cnt % 50000 == 0:
            h = best_h.copy()
            h += np.random.normal(scale=0.02, size=n)
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum_bounds(h, target_sum)
            cur_val, cur_shift, _ = compute_c5_and_shift(h)

    # -----------------------------------------------------------------------
    # Compare with a simple analytic candidate (sin²) which often does well.
    sine_h = _sine_candidate(n, target_sum)
    sine_val, _, _ = compute_c5_and_shift(sine_h)
    if sine_val < best_val:
        best_h, best_val = sine_h, sine_val

    # final projection and evaluation for safety
    best_h = project_to_sum_bounds(best_h, target_sum)
    best_val, _, _ = compute_c5_and_shift(best_h)

    return best_h, float(best_val), n
