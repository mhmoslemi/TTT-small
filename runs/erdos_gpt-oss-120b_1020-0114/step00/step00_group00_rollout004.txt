def project_sum_box(x, target):
    # Clip to [0,1]
    x = np.clip(x, 0, 1)
    # water-filling
    while True:
        diff = target - np.sum(x)
        if abs(diff) < 1e-12:
            break
        # free indices where not at bounds (within epsilon)
        free = ( (x > 0) & (x < 1) )
        free_idx = np.where(free)[0]
        if len(free_idx) == 0:
            # can't adjust further, break
            break
        delta = diff / len(free_idx)
        x[free_idx] += delta
        # clip again
        x = np.clip(x, 0, 1)
    return x
```

def random_neighbor(h, rng, epsilon=0.05):
    n = len(h)
    # pick two distinct indices
    i, j = rng.choice(n, size=2, replace=False)
    # Transfer delta = random between -epsilon and epsilon, but must keep both within [0,1]
    # Determine max feasible delta to add to i (increase) and subtract from j.
    # delta_range = (max(-h[i], -epsilon), min(1 - h[i], epsilon))
    # Wait we want to add to i and subtract from j: delta positive means moving from j to i.
    max_increase_i = min(1 - h[i], epsilon)
    max_decrease_j = min(h[j], epsilon)
    max_delta = min(max_increase_i, max_decrease_j)
    max_decrease_i = min(h[i], epsilon)
    max_increase_j = min(1 - h[j], epsilon)
    # Random decide direction
    if rng.random() < 0.5:
        # Transfer from j to i
        delta = rng.uniform(0, max_delta)
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
    else:
        # Transfer from i to j
        delta = rng.uniform(0, min(max_decrease_i, max_increase_j))
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
    return h_new

def project_sum_box(x, target, eps=1e-12):
    # Clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
    # If sum already close to target, done
    s = np.sum(x)
    if abs(s - target) <= eps:
        return x
    n = len(x)
    # Water-filling algorithm: ensure sum = target while staying within bounds.
    # Approach: sort by values?
    # The problem of projecting onto the intersection of hyperplane sum == target and box [0,1] can be done with algorithm:
    #   Find lambda such that sum_i clip(x_i - lambda, 0, 1) = target.
    # We can solve for lambda via bisection.
    # This is standard: projection onto simplex with upper bound 1.
    # We'll implement bisection.
    lo = -1.0  # lambda low bound
    hi = 1.0   # lambda high bound
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(x - (lo+hi)/2.0, 0.0, 1.0)
    # Ensure sum approx target; adjust minor diff by adding uniformly to free entries
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        # distribute diff over entries not at bounds
        free = (x > 0 + 1e-9) & (x < 1 - 1e-9)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x
```

def gradient_max_shift(h, s):
    n = len(h)
    g = np.zeros_like(h)
    if s >= 0:
        # term when i contributes as first factor
        i1 = np.arange(0, n - s)
        g[i1] += (1 - h[i1 + s])
        # term when i contributes as second factor (via h[i-s])
        i2 = np.arange(s, n)
        g[i2] += -h[i2 - s]
    else:
        # s negative: define t = -s >0, then conv for shift -t = sum_i h[i + t] * (1 - h[i])
        # Equivalent to swap roles of i and i+s
        t = -s
        # first factor: h[i + t] corresponds to index i where i+t < n => i< n-t
        i1 = np.arange(0, n - t)
        # derivative w.r.t h[i] appears as second factor (since h[i] appears as (1 - h[i]) factor)
        # Actually formula for s negative is:
        # C_{-t} = sum_i h[i + t] * (1 - h[i]) for i from 0 to n-1-t.
        # derivative w.r.t h[i] appears in term: -h[i + t] (since derivative of (1 - h[i]) = -1) for i in range 0..n-1-t

import time, math
import numpy as np

# ---- helper functions -------------------------------------------------

def project_sum_box(x: np.ndarray, target: float, iters: int = 50) -> np.ndarray:
    """Project x onto the set {0<=y<=1, sum(y)=target}."""
    x = np.clip(x, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(iters):
        mid = (lo + hi) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 0.0 + 1e-12) & (x < 1.0 - 1e-12)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x


def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for the step function h."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(conv.max() * (2.0 / n))


def interp_to_n(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of h from its current resolution to new_n points."""
    old_n = h.shape[0]
    if old_n == new_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


def random_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Binary feasible vector: n/2 ones (plus a 0.5 if n is odd)."""
    v = np.zeros(n)
    ones = n // 2
    v[:ones] = 1.0
    if n % 2 == 1:
        v[ones] = 0.5
    rng.shuffle(v)
    return v


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point in the feasible polytope."""
    target = n / 2.0
    x = rng.random(n)
    x = x / x.sum() * target
    x = np.clip(x, 0.0, 1.0)
    return project_sum_box(x, target)


def neighbor_move(h: np.ndarray, rng: np.random.Generator, max_transfer: float) -> np.ndarray:
    """Transfer a small amount of mass between random indices."""
    n = h.shape[0]
    h_new = h.copy()
    moves = rng.integers(1, 4)  # 1‑3 elementary moves
    for _ in range(moves):
        i, j = rng.choice(n, size=2, replace=False)
        if rng.random() < 0.5:
            # i -> j
            delta = min(max_transfer, h_new[i], 1.0 - h_new[j])
            if delta <= 0:
                continue
            delta *= rng.random()
            h_new[i] -= delta
            h_new[j] += delta
        else:
            # j -> i
            delta = min(max_transfer, h_new[j], 1.0 - h_new[i])
            if delta <= 0:
                continue
            delta *= rng.random()
            h_new[j] -= delta
            h_new[i] += delta
    return h_new


def gradient_max_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the correlation value for a fixed integer shift."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        s = shift
        i1 = np.arange(0, n - s)
        g[i1] += (1.0 - h[i1 + s])
        i2 = np.arange(s, n)
        g[i2] -= h[i2 - s]
    else:
        t = -shift
        i1 = np.arange(t, n)
        g[i1] += (1.0 - h[i1 - t])
        i2 = np.arange(0, n - t)
        g[i2] -= h[i2 + t]
    return g


def gradient_step(h: np.ndarray, rng: np.random.Generator, max_alpha: float) -> np.ndarray:
    """Take a projected gradient step for the worst‑case shift."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    idx = int(conv.argmax())
    shift = idx - (n - 1)
    g = gradient_max_shift(h, shift)
    if np.allclose(g, 0):
        return h.copy()
    alpha = max_alpha * rng.random()
    h_new = h - alpha * g
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_sum_box(h_new, n / 2.0)


# ---- main optimisation ------------------------------------------------

    rng = np.random.default_rng(seed)
    start = time.time()

    # --- initialise with the provided parent if available ----------------
    try:

    except NameError:

    n0 = init_h.shape[0]
    init_h = project_sum_box(init_h, n0 / 2.0)

    # keep the best solution seen so far
    best_h = init_h.copy()
    best_val = compute_c5(best_h)
    best_n = n0
    best_per_n = {n0: (best_h.copy(), best_val)}

    # candidate resolutions (keep them modest)
    cand_ns = {n0}
    # add a few multiples, limited to 300 points total
    for mult in (2, 3, 4):
        nn = n0 * mult
        if nn <= 300:

    cand_ns = sorted(cand_ns)

    # main timed loop ----------------------------------------------------
    while time.time() - start < budget_s - 0.5:
        # pick a resolution
        n = rng.choice(cand_ns)

        # initialise current state for this resolution

        # decide number of inner steps based on remaining time

            else:

                best_per_n[n] = (best_h.copy(), best_val)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_box(best_h, best_n / 2.0)
