    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def project_onto_box_and_sum(x, target):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        p = np.clip(x - mid, 0.0, 1.0)
        s = p.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    p = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - p.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(p < 1 - 1e-12)[0]
        else:
            idx = np.where(p > 1e-12)[0]
        if idx.size:
            i = idx[0]
            p[i] = np.clip(p[i] + diff, 0.0, 1.0)
    return p

def gradient_shift(h, k):
    n = h.shape[0]
    if k >= 0:
        left = np.concatenate((h[k:], np.zeros(k)))
        right = np.concatenate((np.zeros(k), h[:n - k]))
    else:
        kk = -k
        left = np.concatenate((np.zeros(kk), h[:n - kk]))
        right = np.concatenate((h[kk:], np.zeros(kk)))
    return 1.0 - left - right

    best = cur.copy()

import time, math
import numpy as np

def _c5_val(h: np.ndarray) -> float:
    """max overlap integral for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full").max() * dx


def _project_box_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto [0,1]^n while forcing sum(x)=target."""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        p = np.clip(x - mid, 0.0, 1.0)
        s = p.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    p = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - p.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(p < 1 - 1e-12)[0]
        else:
            idx = np.where(p > 1e-12)[0]
        if idx.size:
            i = idx[0]
            p[i] = np.clip(p[i] + diff, 0.0, 1.0)
    return p


def _grad_for_shift(h: np.ndarray, k: int) -> np.ndarray:
    """Gradient of overlap C_k = Σ h_i(1‑h_{i+k}) w.r.t. h."""
    n = h.shape[0]
    if k >= 0:
        left = np.concatenate((h[k:], np.zeros(k)))
        right = np.concatenate((np.zeros(k), h[: n - k]))
    else:
        kk = -k
        left = np.concatenate((np.zeros(kk), h[: n - kk]))
        right = np.concatenate((h[kk:], np.zeros(kk)))
    return 1.0 - left - right


def _local_optimize(h0: np.ndarray, max_iter: int, rng: np.random.Generator):
    """Hill‑climb using transfers guided by the current worst shift."""
    n = h0.shape[0]
    cur = h0.copy()
    cur_val = _c5_val(cur)
    best = cur.copy()
    best_val = cur_val

    for _ in range(max_iter):
        # identify the shift that attains the maximum overlap
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        idx = int(corr.argmax())
        k = idx - (n - 1)

        # gradient for that shift
        grad = _grad_for_shift(cur, k)

        # favourable source (decrease) and destination (increase) indices
        src_cand = np.where((grad > 1e-12) & (cur > 1e-12))[0]
        dst_cand = np.where((grad < -1e-12) & (cur < 1 - 1e-12))[0]

        if src_cand.size and dst_cand.size:
            src = rng.choice(src_cand)
            dst = rng.choice(dst_cand)
        else:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            src, dst = (i, j) if rng.random() < 0.5 else (j, i)

        max_delta = min(cur[src], 1.0 - cur[dst])
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        new = cur.copy()
        new[src] -= delta
        new[dst] += delta
        new_val = _c5_val(new)

        if new_val < cur_val - 1e-12:
            cur, cur_val = new, new_val
            if new_val < best_val:
                best, best_val = new.copy(), new_val

    return best, best_val


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # start from the supplied parent if it exists
    best_h = None
    best_val = math.inf
    try:
        init = np.array(initial_h_values, dtype=float)
        best_h = init.copy()
        best_val = _c5_val(best_h)
    except Exception:
        pass

    # choose a working resolution (at least 100 points, even)
    n_target = max(100, best_h.shape[0] if best_h is not None else 0)
    if n_target % 2:
        n_target += 1

    # main stochastic search
    while time.time() - start < budget_s - 2.0:
        n = n_target
        target = n / 2.0

        # random feasible start
        rnd = rng.random(n)
        h = _project_box_sum(rnd, target)

        # allocate iterations according to remaining time
        remain = budget_s - (time.time() - start)
        max_iter = int(min(6000, max(50, remain * 150)))
        if max_iter < 10:
            break

        h_opt, val_opt = _local_optimize(h, max_iter, rng)

        if val_opt < best_val:
            best_h, best_val = h_opt, val_opt
            if best_val < 0.3808:
                break

    # final polishing on the best found vector
    if best_h is not None:
        extra = int(min(20000, max(100, (budget_s - (time.time() - start)) * 150)))
        if extra > 0:
            best_h, best_val = _local_optimize(best_h, extra, rng)

        # absolute projection to satisfy constraints (tiny numerical drift possible)
        best_h = _project_box_sum(best_h, best_h.shape[0] / 2.0)
        best_val = _c5_val(best_h)

    else:

        n = n_target
        best_h = np.full(n, 0.5)
        best_val = _c5_val(best_h)

    return best_h.tolist(), float(best_val), best_h.shape[0]
