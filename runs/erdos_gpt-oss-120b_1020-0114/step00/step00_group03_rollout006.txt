import time, numpy as np

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def shift_by(arr, s):
    n = len(arr)
    if s == 0:
        return arr.copy()
    if s > 0:
        out = np.empty_like(arr)
        if s >= n:
            out.fill(0.0)
        else:
            out[:s] = 0.0
            out[s:] = arr[:-s]
        return out
    else:
        # s < 0
        s = -s
        out = np.empty_like(arr)
        if s >= n:
            out.fill(0.0)
        else:
            out[:n - s] = arr[s:]
            out[n - s:] = 0.0
        return out

import time
import numpy as np

def _evaluate_c5(h):
    """Return the C5 value for a step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _shift(arr, s):
    """Shift arr by s steps with zero padding (linear shift, not circular)."""
    n = arr.size
    if s == 0:
        return arr.copy()
    if s > 0:  # shift right
        if s >= n:
            return np.zeros_like(arr)
        out = np.empty_like(arr)
        out[:s] = 0.0
        out[s:] = arr[: n - s]
        return out
    else:      # shift left
        s = -s
        if s >= n:
            return np.zeros_like(arr)
        out = np.empty_like(arr)
        out[: n - s] = arr[s:]
        out[n - s :] = 0.0
        return out


def _project_to_constraints(v, target_sum):
    """Project v onto the box [0,1] and enforce sum(v)=target_sum (water-filling)."""
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target_sum) < 1e-12:
        return v
    if cur > target_sum:
        excess = cur - target_sum
        # reduce from the largest entries first
        idx = np.argsort(-v)
        for i in idx:
            red = min(v[i], excess)
            v[i] -= red
            excess -= red
            if excess <= 1e-12:
                break
    else:
        deficit = target_sum - cur
        # increase from the smallest entries first
        idx = np.argsort(v)
        for i in idx:
            inc = min(1.0 - v[i], deficit)
            v[i] += inc
            deficit -= inc
            if deficit <= 1e-12:
                break
    return v

            best_c5 = c5
            best_h = cand.copy()
            # occasionally promote a new good base
            if len(base_candidates) < max_base:
                base_candidates.append(best_h.copy())

    # final safety projection (might have drifted by rounding)
    best_h = _project_to_constraints(best_h, target_sum)
    best_c5 = _evaluate_c5(best_h)

    # return as a plain list for compatibility
    return (best_h.tolist(), float(best_c5), int(n))
