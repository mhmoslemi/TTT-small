def simulated_annealing_generic(h0, time_budget, rng):
    n = len(h0)
    # Determine if binary
    is_binary = np.all(np.logical_or(np.isclose(h0,0, atol=1e-9), np.isclose(h0,1, atol=1e-9)))
    # Ensure h is float
    h = h0.copy()
    # compute initial
    best_h = h.copy()
    best_val = evaluate_c5(h)
    current_val = best_val
    start = time.time()
    # parameters
    start_temp = 0.02  # can adjust based on best value magnitude
    iteration = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        # temperature linear decreasing
        T = start_temp * (1 - elapsed / time_budget)
        if is_binary:
            # swap move
            i = rng.integers(n)
            j = rng.integers(n)
            if h[i] == h[j]:
                continue
            # swap values
            h_candidate = h.copy()
            temp = h_candidate[i]
            h_candidate[i] = h_candidate[j]
            h_candidate[j] = temp
        else:
            i, j = rng.choice(n, size=2, replace=False)
            # random sign direction
            sign = rng.choice([-1, 1])
            if sign == 1:
                max_delta = min(1.0 - h[i], h[j])
                if max_delta <= 0:
                    continue
                delta = rng.random() * max_delta
                h_candidate = h.copy()
                h_candidate[i] += delta
                h_candidate[j] -= delta
            else:
                max_delta = min(h[i], 1.0 - h[j])
                if max_delta <= 0:
                    continue
                delta = rng.random() * max_delta

import numpy as np, time, math

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def random_binary_vec(n, rng):
    vec = np.zeros(n, dtype=float)
    inds = rng.choice(n, size=n // 2, replace=False)
    vec[inds] = 1.0
    return vec

def simulated_annealing_generic(h_start, time_budget, rng):
    n = len(h_start)
    # ensure binary
    # maybe h_start not binary; we treat as binary for moves anyway?
    # For safety, treat binary flag if near-binary; else use continuous moves.
    is_binary = np.all(np.logical_or(np.isclose(h_start,0, atol=1e-9), np.isclose(h_start,1, atol=1e-9)))
    # ensure h as float
    h = h_start.copy()
    # Ensure sum constraint: if not exactly n/2, adjust
    target_sum = n / 2.0
    if not np.isclose(h.sum(), target_sum):
        # project onto binary by random fill
        h = random_binary_vec(n, rng)
    current_val = evaluate_c5(h)
    best_h = h.copy()
    best_val = current_val
    start = time.time()
    steps_without_improve = 0
    patience = max(5000, n*100)  # maybe 5000 steps
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        # temperature
        # start_temp maybe 0.04
        T = 0.04 * (1.0 - elapsed / time_budget)
        # generate move
        if is_binary:
            # swap between a 1 and a 0
            # quickly select indices
            # choose i where h[i]==1, j where h[j]==0
            # Use while to find
            while True:
                i = rng.integers(n)
                j = rng.integers(n)
                if h[i] != h[j]:
                    break
            # swap
            h_candidate = h.copy()
            # swap values
            tmp = h_candidate[i]
            h_candidate[i] = h_candidate[j]
            h_candidate[j] = tmp
        else:
            # continuous move preserving sum
            i, j = rng.choice(n, size=2, replace=False)
            sign = rng.choice([-1, 1])
            if sign == 1:

                delta = rng.random() * max_delta
                h_candidate = h.copy()
                h_candidate[i] += delta
                h_candidate[j] -= delta
            else:
                max_delta = min(h[i], 1.0 - h[j])
                if max_delta <= 0:
                    continue
                delta = rng.random() * max_delta

        cand_val = evaluate_c5(h_candidate)

                best_h = h_candidate.copy()
                steps_without_improve = 0
            else:
                steps_without_improve += 1
        else:
            steps_without_improve += 1
        # restart if stuck long

                current_val = evaluate_c5(h)
                steps_without_improve = 0
    return best_h, best_val

import time, math
import numpy as np

def _evaluate_c5(h: np.ndarray) -> float:
    """Return max overlap integral for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    # full correlation of h with (1-h)
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def _random_binary_vec(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a binary vector of length n with exactly n//2 ones."""
    vec = np.zeros(n, dtype=float)
    ones = rng.choice(n, size=n // 2, replace=False)
    vec[ones] = 1.0
    return vec

def _binarize_to_half_one(v: np.ndarray) -> np.ndarray:
    """Project any vector onto the set of binary vectors with half ones."""
    n = v.shape[0]
    k = n // 2
    # indices of the k largest entries become 1
    idx = np.argpartition(-v, k - 1)[:k]
    out = np.zeros(n, dtype=float)
    out[idx] = 1.0
    return out

def _simulated_annealing(h_start: np.ndarray,
                         time_budget: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple SA that works on binary vectors with exact half‑ones.
    Returns the best vector and its C5 value found within time_budget.
    """
    n = h_start.shape[0]
    # keep everything binary – the optimal solution is known to be extreme
    h = _binarize_to_half_one(h_start)
    best_h = h.copy()
    best_val = _evaluate_c5(best_h)

    start = time.time()
    steps_without_improve = 0
    # patience grows with dimension; we reset after many futile steps
    patience = max(5000, n * 150)
    # start temperature (large enough to accept occasional uphill moves)
    start_temp = 0.20

    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break

        # linear cooling to 0
        T = start_temp * (1.0 - elapsed / time_budget)

        # ---- propose a swap between a 1 and a 0  -----------------
        # pick two distinct indices until they differ
        i = rng.integers(n)
        j = rng.integers(n)
        while h[i] == h[j]:
            i = rng.integers(n)
            j = rng.integers(n)

        # perform swap
        cand = h.copy()
        cand[i], cand[j] = cand[j], cand[i]

        cand_val = _evaluate_c5(cand)
        delta = cand_val - best_val

        if delta < 0 or (T > 0 and rng.random() < math.exp(-delta / T)):
            # accept move
            h = cand
            if cand_val < best_val - 1e-12:
                best_val = cand_val
                best_h = cand.copy()
                steps_without_improve = 0
            else:
                steps_without_improve += 1
        else:
            steps_without_improve += 1

        # ---- restart if stuck ------------------------------------
        if steps_without_improve >= patience:
            h = _random_binary_vec(n, rng)
            steps_without_improve = 0

    return best_h, best_val

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize the Erdős minimum overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    t_start = time.time()

    # -----------------------------------------------------------------
    # initialise
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n0 = h0.shape[0]
    else:
        n0 = 42
        h0 = _random_binary_vec(n0, rng)

    # force binary half‑ones representation
    h0 = _binarize_to_half_one(h0)
    best_h_global = h0.copy()
    best_val_global = _evaluate_c5(best_h_global)

    # -----------------------------------------------------------------
    # Phase 1 : work on the original resolution
    remaining = budget_s - (time.time() - t_start)
    phase1_time = min(0.65 * remaining, remaining - 5.0)   # keep a few seconds for later
    h_best, val_best = _simulated_annealing(best_h_global, phase1_time, rng)
    if val_best < best_val_global:
        best_val_global = val_best
        best_h_global = h_best.copy()

    # early exit if we already beat the target
    if best_val_global <= 0.3808:
        return best_h_global, best_val_global, best_h_global.shape[0]

    # -----------------------------------------------------------------
    # Phase 2 : up‑sample (repeat each entry twice) and refine
    # keep trying to up‑sample while time permits, but stop before ~800 pts
    current_h = best_h_global.copy()
    while True:
        elapsed = time.time() - t_start
        remaining = budget_s - elapsed
        if remaining < 8.0:
            break
        # stop if next up‑sample would exceed a reasonable size
        if current_h.shape[0] * 2 > 800:
            break

        # up‑sample by factor 2 (preserves half‑ones property)
        current_h = np.repeat(current_h, 2)
        # time slice for this new resolution
        slice_time = min(0.30 * remaining, remaining - 4.0)

        h_best, val_best = _simulated_annealing(current_h, slice_time, rng)
        if val_best < best_val_global:
            best_val_global = val_best
            best_h_global = h_best.copy()

        # stop if we already reached the target
        if best_val_global <= 0.3808:
            break

    # -----------------------------------------------------------------
    # final sanity fix – enforce exact sum due to possible rounding
    final_h = best_h_global.copy()
    n_final = final_h.shape[0]
    target_sum = n_final / 2.0
    diff = target_sum - final_h.sum()
    if abs(diff) > 1e-9:
        # adjust a random feasible entry
        idx = rng.integers(n_final)
        new_val = np.clip(final_h[idx] + diff, 0.0, 1.0)
        final_h[idx] = new_val
        # if still off (tiny numerical), spread uniformly and re‑clip
        diff2 = target_sum - final_h.sum()
        if abs(diff2) > 1e-9:
            final_h += diff2 / n_final
            final_h = np.clip(final_h, 0.0, 1.0)
            # final tiny fix
            final_h[0] = np.clip(final_h[0] + (target_sum - final_h.sum()), 0.0, 1.0)

    final_val = _evaluate_c5(final_h)

    return final_h, final_val, n_final
