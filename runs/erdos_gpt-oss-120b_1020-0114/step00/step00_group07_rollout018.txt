import time
import numpy as np

def compute_c5(h: np.ndarray, dx: float) -> float:
    """C5 = max_k ∫ h(x)(1−h(x+k))dx approximated by full correlation."""
    return (np.correlate(h, 1.0 - h, mode="full") * dx).max()


def project_onto_capsimplex(v: np.ndarray, S: float) -> np.ndarray:
    """
    Project v onto the set {x | 0 ≤ x_i ≤ 1, sum x_i = S}
    using a simple bisection on the Lagrange multiplier λ.
    """
    lo = np.min(v) - 1.0          # guarantees all entries after shift are ≥1 → clipped to 1
    hi = np.max(v)                # guarantees all entries after shift are ≤0 → clipped to 0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - hi, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = x.sum() - S
    if abs(diff) > 1e-10:
        mask = (x > 0.0) & (x < 1.0)
        if mask.any():
            x[mask] -= diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
    return x


def gradient_step(h: np.ndarray, alpha: float, lr: float, dx: float) -> np.ndarray:
    """
    One gradient‑ascent step aiming to maximise the minimum autocorrelation.
    Uses a soft‑minimum with temperature 1/α.
    """
    N = h.size
    # full autocorrelation (including lag 0)
    corr = np.correlate(h, h, mode="full") * dx
    w = np.exp(-alpha * corr)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w.fill(1.0 / w.size)

    grad = np.zeros_like(h)
    # offsets run from -(N-1) … (N-1)
    for off in range(-N + 1, N):
        weight = w[off + N - 1]
        if weight == 0.0:
            continue
        if off == 0:
            grad += 2.0 * weight * h
            continue
        if off > 0:
            limit = N - off
            grad[:limit] += weight * h[off:]
            grad[off:]  += weight * h[:limit]
        else:  # off < 0
            pos = -off
            limit = N - pos
            grad[pos:]   += weight * h[:limit]
            grad[:limit] += weight * h[pos:]

    grad *= dx                     # derivative of corr already contains one factor dx
    h_new = h + lr * grad
    return h_new


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimise a step function h on [0,2] to obtain a low C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # -------------------------------------------------------------
    # Choose a working resolution (balance quality vs runtime)
    N = 200                     # number of discretisation points (≤1000)
    dx = 2.0 / N
    target_sum = N / 2.0        # because ∫ h = 1  ⇒  sum(h)*dx = 1

    # -------------------------------------------------------------
    # Initialise h (use parent if available, otherwise random binary)
    init = globals().get("initial_h_values", None)
    if init is not None:
        init = np.asarray(init, dtype=float)
        if init.size != N:
            # linear interpolation to the required size
            x_old = np.linspace(0.0, 2.0, init.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, N, endpoint=False)
            h = np.interp(x_new, x_old, init)
        else:
            h = init.copy()
    else:
        h = np.zeros(N, dtype=float)
        idx = rng.choice(N, N // 2, replace=False)
        h[idx] = 1.0

    # enforce constraints exactly
    h = project_onto_capsimplex(h, target_sum)

    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)

    # -------------------------------------------------------------
    # Optimisation parameters (annealing schedule)
    max_iters = 8000                # absolute ceiling
    it = 0
    while True:
        # time guard
        if time.time() - start > budget_s * 0.97:
            break
        if it >= max_iters:
            break
        it += 1

        # temperature schedule for soft‑minimum
        alpha = 5.0 + (200.0 - 5.0) * (it / max_iters)   # 5 → 200
        # learning‑rate schedule (gradual decay)
        lr = 0.6 * (1.0 - it / max_iters) + 0.04        # ~0.64 → 0.04

        # gradient ascent step
        h_candidate = gradient_step(h, alpha, lr, dx)

        # projection back onto feasible set
        h_candidate = project_onto_capsimplex(h_candidate, target_sum)

        # evaluate
        c5_val = compute_c5(h_candidate, dx)
        if c5_val < best_c5 - 1e-9:
            best_c5 = c5_val
            best_h = h_candidate.copy()

        # move on
        h = h_candidate

        # occasional random shake‑up to escape very flat plateaus
        if it % 1000 == 0:
            perturb = rng.normal(scale=0.02, size=N)
            h_shake = np.clip(h + perturb, 0.0, 1.0)
            h_shake = project_onto_capsimplex(h_shake, target_sum)
            # accept if improves
            c5_shake = compute_c5(h_shake, dx)
            if c5_shake < best_c5:
                best_c5 = c5_shake
                best_h = h_shake.copy()
                h = h_shake

    # final projection & evaluation
    best_h = project_onto_capsimplex(best_h, target_sum)
    final_c5 = compute_c5(best_h, dx)

    # Return a plain Python list for compatibility (optional)
    return (best_h, final_c5, N)
