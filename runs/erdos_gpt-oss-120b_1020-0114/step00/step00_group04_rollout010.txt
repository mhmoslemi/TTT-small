def repair(v):
    h = np.clip(v, 0, 1)
    s = h.sum()
    target = n_points / 2
    if s < target:
        deficit = target - s
        # indices where h_i < 1
        idx = np.where(h < 1)[0]
        if idx.size == 0:
            # shouldn't happen
            return h
        # add uniform distribution over these indices as possible
        # compute capacity each index can receive: 1 - h_i
        caps = 1 - h[idx]
        total_cap = caps.sum()
        if total_cap <= 0:
            # cannot add
            return h
        # allocate proportionally
        add = deficit * caps / total_cap
        h[idx] += add
        # may exceed 1 due to rounding, clip again
        h = np.clip(h, 0, 1)
    elif s > target:
        excess = s - target
        idx = np.where(h > 0)[0]
        caps = h[idx]  # amount that can be removed
        total_cap = caps.sum()
        if total_cap <= 0:
            return h
        sub = excess * caps / total_cap

import time
import math
import numpy as np
from scipy.optimize import differential_evolution


def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h (as a NumPy array) on [0,2] that gives a low
    C₅ bound.  The function tries a quick random search, a short differential
    evolution run and a simulated‑annealing refinement, all within the time
    budget.

    Returns
    -------
    (h_values, c5_bound, n_points) : tuple
        h_values – NumPy array of values in [0,1] (length n_points)
        c5_bound – the resulting C₅ bound (float)
        n_points – number of discretisation points (int)
    """
    start_time = time.time()
    rng = np.random.RandomState(seed)

    # ----------------------------------------------------------------------
    # Discretisation parameters
    # ----------------------------------------------------------------------
    n_points = 200                # even, gives dx = 0.01 and target sum = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2.0   # ∑ h[i] must equal this (integral = 1)

    # ----------------------------------------------------------------------
    # Helper: project a vector onto the box [0,1] with exact sum constraint
    # ----------------------------------------------------------------------
    def project_to_simplex(v):
        """Return w = clip(v - λ, 0, 1) such that sum(w)=target_sum."""
        # binary search for λ (lambda) – the shift that makes the sum correct
        low = v.min() - 1.0
        high = v.max()
        for _ in range(80):
            lam = (low + high) * 0.5
            w = np.clip(v - lam, 0.0, 1.0)
            s = w.sum()
            if abs(s - target_sum) <= 1e-12:
                return w
            if s > target_sum:
                low = lam
            else:
                high = lam
        # after loop, adjust the small residual linearly on a free coordinate
        w = np.clip(v - low, 0.0, 1.0)
        diff = target_sum - w.sum()
        if abs(diff) > 1e-9:
            # add/subtract the remainder on the first adjustable entry
            if diff > 0:
                idx = np.where(w < 1.0)[0]
                if idx.size:
                    i = idx[0]
                    w[i] = min(1.0, w[i] + diff)
            else:
                idx = np.where(w > 0.0)[0]
                if idx.size:
                    i = idx[0]
                    w[i] = max(0.0, w[i] + diff)
        return w

    # ----------------------------------------------------------------------
    # Compute C₅ for a feasible h
    # ----------------------------------------------------------------------
    def c5_of(h):
        # full correlation of h with (1‑h); the max entry, scaled by dx, is C₅
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    # ----------------------------------------------------------------------
    # Build a deterministic smooth starting point (sin² shape)
    # ----------------------------------------------------------------------
    xs = (np.arange(n_points) + 0.5) * dx               # mid‑point grid
    if 'initial_h_values' in globals():
        # upscale provided parent solution
        parent = initial_h_values
        m = parent.shape[0]
        xp = (np.arange(m) + 0.5) * (2.0 / m)
        h_start = np.interp(xs, xp, parent)
    else:
        # sin² wave: 0 → 1 → 0, integral = 1, values already within [0,1]
        h_start = (1.0 - np.cos(np.pi * xs)) * 0.5
    h_start = np.clip(h_start, 0.0, 1.0)
    h_start = project_to_simplex(h_start)

    best_h = h_start.copy()
    best_score = c5_of(best_h)

    # ----------------------------------------------------------------------
    # Stage 1 – quick random sampling
    # ----------------------------------------------------------------------
    max_samples = 3000
    for _ in range(max_samples):
        if time.time() - start_time > budget_s * 0.45:
            break
        raw = rng.rand(n_points)
        h = project_to_simplex(raw)
        val = c5_of(h)
        if val < best_score:
            best_score = val
            best_h = h.copy()
            if best_score < 0.3808:
                break

    # ----------------------------------------------------------------------
    # Stage 2 – short differential evolution (global search)
    # ----------------------------------------------------------------------
    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.65:
        # time left determines how many generations we may afford
        remaining = budget_s - elapsed
        max_iter = max(2, int(min(120, remaining // 0.05)))  # empirical budget
        bounds = [(0.0, 1.0)] * n_points

        # DE works on the raw (unprojected) vector; feasibility is restored
        # inside the objective.
        def de_obj(v):
            h = project_to_simplex(v)
            return c5_of(h)

        try:
            res = differential_evolution(
                de_obj,
                bounds,
                maxiter=max_iter,
                popsize=10,
                seed=seed,
                polish=False,
                updating='deferred',
                disp=False,
                workers=1,
            )
            h_de = project_to_simplex(res.x)
            val_de = c5_of(h_de)
            if val_de < best_score:
                best_score = val_de
                best_h = h_de.copy()
        except Exception:
            pass   # DE may fail if the time budget is exceeded

    # ----------------------------------------------------------------------
    # Stage 3 – simulated annealing / hill climbing refinement
    # ----------------------------------------------------------------------
    cur_h = best_h.copy()
    cur_score = best_score

    T_initial = 0.02
    T_min = 1e-6
    T = T_initial
    iteration = 0
    while time.time() - start_time < budget_s * 0.97:
        iteration += 1
        # exponential cooling
        T = max(T * 0.9993, T_min)

        # Propose a mass transfer between two positions (preserves sum)
        i, j = rng.randint(0, n_points, size=2)
        if i == j:
            continue

        # Choose direction randomly
        if rng.rand() < 0.5:
            # move from i to j
            max_delta = min(0.2, cur_h[i], 1.0 - cur_h[j])
            if max_delta <= 0:
                continue
            delta = rng.rand() * max_delta
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        else:
            # move from j to i
            max_delta = min(0.2, cur_h[j], 1.0 - cur_h[i])
            if max_delta <= 0:
                continue
            delta = rng.rand() * max_delta
            new_h = cur_h.copy()
            new_h[j] -= delta
            new_h[i] += delta

        new_score = c5_of(new_h)

        # Metropolis acceptance
        if new_score < cur_score or rng.rand() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
            cur_h = new_h
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = new_h.copy()
                if best_score < 0.3808:
                    break

        # occasional large‑scale perturbation to escape plateaus
        if iteration % 5000 == 0:
            # replace a random block by fresh random values (mass‑balanced)
            block_len = rng.randint(5, 20)
            start_idx = rng.randint(0, n_points - block_len + 1)
            block = rng.rand(block_len)
            block = np.clip(block, 0.0, 1.0)
            # adjust to keep total sum unchanged
            delta_block = block.sum() - cur_h[start_idx:start_idx + block_len].sum()
            # spread the surplus/deficit uniformly over the rest
            rest_idx = np.setdiff1d(np.arange(n_points), np.arange(start_idx, start_idx + block_len))
            if rest_idx.size:
                cur_h[rest_idx] -= delta_block / rest_idx.size
                cur_h[rest_idx] = np.clip(cur_h[rest_idx], 0.0, 1.0)
            cur_h[start_idx:start_idx + block_len] = block
            cur_h = project_to_simplex(cur_h)      # re‑enforce feasibility
            cur_score = c5_of(cur_h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = cur_h.copy()

    # final safety projection (removes any tiny drift)
    final_h = project_to_simplex(best_h)
    final_score = c5_of(final_h)

    return final_h, final_score, n_points
