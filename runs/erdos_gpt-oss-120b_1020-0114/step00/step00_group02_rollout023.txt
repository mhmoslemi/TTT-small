def gradient_step(h, dx, step):
    # Compute correlation
    corr = np.correlate(h, 1-h, mode='full')
    idx = corr.argmax()
    shift = idx - (len(h) - 1)
    k = abs(shift)
    if k == 0:
        grad = 1 - 2 * h
    else:
        fwd = np.concatenate([h[k:], np.zeros(k)])
        bwd = np.concatenate([np.zeros(k), h[:len(h)-k]])
        grad = 1 - (fwd + bwd)
    h_new = h - step * grad
    h_new = project_to_bounded_simplex(h_new, total_sum = len(h)/2)
    return h_new, corr.max() * dx, shift

def project_to_bounded_simplex(v, total):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s > total:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo+hi)/2.0, 0.0, 1.0)

import time, math, numpy as np

def _proj_to_bounded_simplex(v, total):
    """Projection of v onto {0≤x_i≤1, sum x = total} using a simple bisection."""
    lo, hi = -1.0, 1.0  # guarantees sum(w) ranges from 0 to len(v)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s > total:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)


def _c5_continuous(h):
    """C5 for a (possibly fractional) step function."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx, corr


def _c5_binary(h):
    """C5 for a binary (0/1) vector."""
    n = h.size
    dx = 2.0 / n
    m = h.sum()
    q = np.correlate(h, h, mode="full")
    return (m - q.min()) * dx, q


def _gradient_step(h, step, total):
    """One projected‑gradient step for the continuous problem."""
    n = h.size
    # current worst shift
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = corr.argmax()
    shift = idx - (n - 1)
    k = abs(shift)

    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        fwd = np.concatenate([h[k:], np.zeros(k)])
        bwd = np.concatenate([np.zeros(k), h[: n - k]])
        grad = 1.0 - (fwd + bwd)

    h_new = h - step * grad
    h_new = _proj_to_bounded_simplex(h_new, total)
    new_c5, _ = _c5_continuous(h_new)
    return h_new, new_c5, corr.max() * (2.0 / n)
