import time, math, random
import numpy as np

def _project_to_box_and_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto [0,1]^n while forcing sum(v)=target (water‑filling)."""
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0  # bounds wide enough (v∈[0,1])
    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction to hit the sum a little more accurately
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = (w > 0.0) & (w < 1.0)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0.0, 1.0)
    return w


def _max_corr_raw(h: np.ndarray) -> float:
    """Return the un‑scaled max of the correlation h ⋆ (1‑h)."""
    # np.correlate does the cross‑correlation used in the evaluator
    return np.max(np.correlate(h, 1.0 - h, mode="full"))


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function on [0,2] with small max overlap.

    Returns (list_of_h, C5_value, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)

    # -------------------------------------------------------------
    # 1. initialise a feasible vector h
    # -------------------------------------------------------------
    try:
        init = np.array(initial_h_values, dtype=np.float64)  # provided by the harness
        n = init.size
        h = _project_to_box_and_sum(init, n / 2.0)
    except Exception:                     # no global `initial_h_values`
        n = 200                           # a modest size (<<1000)
        h = np.random.rand(n)
        h = _project_to_box_and_sum(h, n / 2.0)

    dx = 2.0 / n

    # -------------------------------------------------------------
    # 2. helper to evaluate the objective
    # -------------------------------------------------------------
    def _c5(hv: np.ndarray) -> float:
        return _max_corr_raw(hv) * dx

    cur_val = _c5(h)
    best_h = h.copy()
    best_val = cur_val

    # -------------------------------------------------------------
    # 3. optimisation parameters
    # -------------------------------------------------------------
    max_iters = 800000                     # generous upper bound
    eps_start, eps_end = 0.35, 0.005      # max transfer amount (will anneal)
    T_start, T_end = 0.12, 1e-7           # temperature for Metropolis moves

    it = 0
    start_time = time.time()
    # -------------------------------------------------------------
    # 4. main loop (guided + occasional random moves)
    # -------------------------------------------------------------
    while True:
        # time guard
        if time.time() - start_time > budget_s - 0.4:
            break
        if it >= max_iters:
            break
        it += 1

        # annealing schedules (linear)
        frac = it / max_iters
        eps = eps_start * (1.0 - frac) + eps_end * frac
        T = T_start * (1.0 - frac) + T_end * frac

        # -----------------------------------------------------------------
        # 4a. pick a move: 90 % directed (attack the worst lag), 10 % random
        # -----------------------------------------------------------------
        if random.random() < 0.9:
            # --- directed move -------------------------------------------------
            raw = np.correlate(h, 1.0 - h, mode="full")
            idx_max = int(np.argmax(raw))
            lag = idx_max - (n - 1)                     # lag ∈ [-(n-1), …, n-1]

            if lag == 0:
                # no contribution – fall back to a random pair
                i, j = random.sample(range(n), 2)
                src, dst = i, j
            else:
                if lag > 0:
                    contrib = h[: n - lag] * (1.0 - h[lag:])
                    i0 = int(np.argmax(contrib))
                    src = i0
                    dst = i0 + lag
                else:  # lag < 0
                    l = -lag
                    contrib = h[l:] * (1.0 - h[: n - l])
                    i0 = int(np.argmax(contrib))
                    src = i0 + l
                    dst = i0
                # ensure the direction moves mass from a larger to a smaller entry
                if h[src] < h[dst]:
                    src, dst = dst, src
        else:
            # --- random pair move ---------------------------------------------
            src, dst = random.sample(range(n), 2)
            if h[src] < h[dst]:
                src, dst = dst, src

        # -----------------------------------------------------------------
        # 4b. propose a transfer delta from src → dst
        # -----------------------------------------------------------------
        max_pos = min(eps, h[src], 1.0 - h[dst])
        if max_pos <= 0.0:
            continue
        delta = random.random() * max_pos          # 0 < delta ≤ max_pos

        # apply tentative move
        h[src] -= delta
        h[dst] += delta

        new_val = _c5(h)

        # -----------------------------------------------------------------
        # 4c. acceptance test
        # -----------------------------------------------------------------
        accepted = False
        if new_val < cur_val:
            accepted = True
        else:
            # Metropolis acceptance
            if T > 0.0:
                prob = math.exp(-(new_val - cur_val) / T)
                if random.random() < prob:
                    accepted = True

        if accepted:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                # early stop if we beat the target record
                if best_val <= 0.3808:
                    break
        else:
            # revert the move
            h[src] += delta
            h[dst] -= delta

    # -------------------------------------------------------------
    # 5. final projection (safety) and final evaluation
    # -------------------------------------------------------------
    best_h = _project_to_box_and_sum(best_h, n / 2.0)
    best_val = _c5(best_h)

    return (best_h.tolist(), float(best_val), n)
