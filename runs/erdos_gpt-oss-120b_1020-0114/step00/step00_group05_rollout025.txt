    # choose two distinct indices
    i, j = np.random.choice(N, size=2, replace=False)
    if h[i] >= 1 and h[j] <= 0:

    # delta limited by permissible range
    max_delta_up = 1.0 - h[i]

        return None

    h_new = h.copy()

import time
import numpy as np

def _project_box_simplex(v, target, lo=0.0, hi=1.0, tol=1e-12, max_iter=60):
    """Project v onto [lo,hi]^n intersect {sum=x = target}."""
    v = np.clip(v, lo, hi)
    # binary search for lambda
    lam_low, lam_high = -1e5, 1e5
    for _ in range(max_iter):
        lam = (lam_low + lam_high) / 2.0
        x = np.clip(v - lam, lo, hi)
        s = x.sum()
        if s > target:
            lam_low = lam
        else:
            lam_high = lam
        if abs(s - target) <= tol:
            break
    return np.clip(v - (lam_low + lam_high) / 2.0, lo, hi)


def _compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k)) dx  (discrete version)."""
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * corr.max()


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    # ------------------------------------------------------------------
    # initialise h
    # ------------------------------------------------------------------
    try:
        init = np.array(initial_h_values, dtype=float)
        N = init.size
        h = init.copy()
    except Exception:
        N = 120                     # fallback size
        h = np.zeros(N, dtype=float)
        half = N // 2
        h[:half] = 1.0
        np.random.shuffle(h)

    target_sum = N / 2.0
    h = _project_box_simplex(h, target_sum)

    best_h = h.copy()
    best_c5 = _compute_c5(best_h)

    # ------------------------------------------------------------------
    # optimisation parameters
    # ------------------------------------------------------------------
    start = time.time()
    beta = 0.5          # smoothing for min‑corr
    step = 0.25         # base gradient step
    iteration = 0

    while True:
        elapsed = time.time() - start
        if elapsed > budget_s - 0.2:
            break
        iteration += 1

        # correlation of h with itself
        g = np.correlate(h, h, mode="full")      # length 2N‑1
        g_min = g.min()
        # smoothed min‑weight (larger weight for smaller correlations)
        w = np.exp(-beta * (g - g_min))
        p = w / w.sum()                          # shape (2N‑1,)

        # gradient of smoothed min‑correlation
        conv_fwd = np.convolve(p, h, mode="full")
        conv_rev = np.convolve(p[::-1], h, mode="full")
        mid = N - 1
        grad = conv_fwd[mid:mid + N] + conv_rev[mid:mid + N]

        # ascent step + projection
        h_new = _project_box_simplex(h + step * grad, target_sum)
        cur_c5 = _compute_c5(h)
        new_c5 = _compute_c5(h_new)

        # acceptance (always if improvement, otherwise SA‑like)
        if new_c5 < cur_c5:
            h = h_new
            cur_c5 = new_c5
        else:
            # temperature decreasing linearly
            T = max(1e-4, (budget_s - elapsed) / budget_s)
            if np.random.rand() < np.exp(-(new_c5 - cur_c5) / T):
                h = h_new
                cur_c5 = new_c5

        # keep track of the best vector seen
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()

        # schedule adjustments
