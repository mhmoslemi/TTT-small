    corr = np.correlate(h, 1 - h, mode='full')

import time
import math
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return C5 = max_k ∫h(x)(1−h(x+k))dx approximated on the grid."""
    # correlation of h with (1−h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    # dx = length / n_points = 2 / len(h)
    return (2.0 / h.shape[0]) * np.max(corr)


def gradient_step(h: np.ndarray, target_sum: float, T: float, rng: np.random.Generator):
    """
    Perform a single gradient-inspired move aimed at the worst shift.
    Returns the new vector and its C5 value.
    """
    n = h.shape[0]
    # worst shift (the lag giving the current maximal overlap)
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx_max = int(np.argmax(corr))
    shift = idx_max - (n - 1)          # 0‑lag is at index n‑1
    if shift == 0:
        # nothing to improve; return unchanged
        return h, compute_c5(h)

    # gradient of S(shift)=∑h_i h_{i−shift}
    grad = np.zeros_like(h)
    if shift > 0:
        grad[shift:] += h[:-shift]
        grad[:-shift] += h[shift:]
    else:
        s = -shift
        grad[: n - s] += h[s:]
        grad[s:] += h[: n - s]

    # project to zero‑mean to keep the sum unchanged (ignoring bounds)
    grad -= np.mean(grad)

    # step size proportional to temperature
    step = T * 0.5
    h_new = h + step * grad
    h_new = np.clip(h_new, 0.0, 1.0)

    # re‑enforce the linear sum constraint by fixing a single coordinate
    diff = h_new.sum() - target_sum
    if abs(diff) > 1e-12:
        k = rng.integers(0, n)
        if diff > 0.0:          # need to reduce sum
            adj = min(diff, h_new[k])
            h_new[k] -= adj
        else:                   # need to increase sum
            adj = min(-diff, 1.0 - h_new[k])
            h_new[k] += adj

    return h_new, compute_c5(h_new)


def project_initial(h: np.ndarray, target_sum: float):
    """Clip to [0,1] and adjust a single entry so the sum equals target_sum."""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # adjust first element (any element works)
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
        # if still off due to clipping, distribute the remaining error
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            # distribute over a random index
            idx = np.random.randint(0, len(h))
            if diff2 > 0.0:
                h[idx] = np.clip(h[idx] + diff2, 0.0, 1.0)
            else:
                h[idx] = np.clip(h[idx] + diff2, 0.0, 1.0)
    return h


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step function h on [0,2] with average 0.5
    that minimises C5 = max_k ∫h(x)(1−h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------
    # Initialise the candidate vector
    # ----------------------------------------------------
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=float)
        factor = 2                     # up‑sample to gain more degrees of freedom
        h = np.repeat(base, factor)
    else:
        # fallback: a purely random feasible vector of length 140
        n_points = 140
        target_sum = n_points / 2.0
        h = rng.random(n_points)
        h = project_initial(h, target_sum)

    n = h.shape[0]
    target_sum = n / 2.0
    h = project_initial(h, target_sum)

    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    # ----------------------------------------------------
    # Simulated annealing + occasional gradient moves
    # ----------------------------------------------------
    start = time.time()
    total_time = budget_s * 0.95      # keep a safety margin
    T0 = 0.02                         # initial temperature
    T_final = 1e-6
    iteration = 0

    while True:
        elapsed = time.time() - start
        if elapsed >= total_time:
            break
        # early exit if we already beat the target record
        if best_c5 <= 0.3808:
            break

        iteration += 1
        # linear cooling schedule
        T = max(T_final, T0 * (1.0 - elapsed / total_time))

        # ----------- pairwise mass transfer move ------------
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        hi, hj = h[i], h[j]

        # feasible delta: keep both entries inside [0,1] while preserving total mass
        delta_pos = min(1.0 - hi, hj)   # increase i, decrease j
        delta_neg = min(hi, 1.0 - hj)   # decrease i, increase j
        if delta_pos + delta_neg == 0.0:
            continue
        delta = rng.uniform(-delta_neg, delta_pos)

        h_new = h.copy()
        h_new[i] = hi + delta
        h_new[j] = hj - delta

        new_c5 = compute_c5(h_new)
        d = new_c5 - cur_c5
        if d <= 0.0 or rng.random() < math.exp(-d / T):
            h = h_new
            cur_c5 = new_c5

                h = h_tmp
                cur_c5 = c5_tmp
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()

    # ----------------------------------------------------
    # Return the best vector found
    # ----------------------------------------------------
    return (best_h.tolist(), float(best_c5), n)
