import time
import numpy as np

def compute_max_c(h):
    """Exact C5 value for a step function given by vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_box_simplex(v, target, iters=60, tol=1e-12):
    """
    Project vector v onto the box simplex:
        0 <= x[i] <= 1,   sum(x) = target
    using a simple bisection on the Lagrange multiplier.
    """
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(iters):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny adjustment for rounding error
    diff = target - proj.sum()
    if abs(diff) > 1e-10:
        free = np.where((proj > 1e-12) & (proj < 1.0 - 1e-12))[0]
        if free.size:
            proj[free] += diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj

        w = tmp / tmp.sum()                     # weight for each shift

        # ----- gradient of the smooth‑max objective ----------------
        grad = np.zeros_like(h)
        for idx, k in enumerate(shifts):
            wk = w[idx]
            if wk == 0.0:
                continue
            if k >= 0:
                kk = int(k)
                # contribution from (1 - h[i+kk])
                grad[:n - kk] += wk * (1.0 - h[kk:])
                # contribution from -h[i-kk]
                grad[kk:]     -= wk * h[:n - kk]
            else:
                kk = -int(k)
                # contribution from (1 - h[i-kk])
                grad[kk:]     += wk * (1.0 - h[:n - kk])
                # contribution from -h[i+kk]
                grad[:n - kk] -= wk * h[kk:]

        grad *= dx                               # chain rule (dx factor)

        # ----- descent step + projection ---------------------------
        h_new = h - lr * grad
        h_new = project_box_simplex(h_new, target_sum)

        # ----- evaluate true C5 ------------------------------------
        cur_val = compute_max_c(h_new)

        # keep the best ever seen
        if cur_val < best_val - 1e-9:
            best_val = cur_val
            best_h = h_new.copy()

        # accept the step (always) – it is already feasible
        h = h_new

        # ----- schedule adjustments --------------------------------
        it += 1
        if it % 120 == 0:           # slowly tighten the approximation
            alpha = min(alpha * 1.07, 45.0)
            lr *= 0.97

        # early exit if we already beat the target
        if best_val <= 0.3808:
            break

    final_c5 = compute_max_c(best_h)
    return (best_h.tolist(), float(final_c5), n)
