import numpy as np, time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project(h, target_sum):
    # clip
    h = np.clip(h, 0.0, 1.0)
    # enforce sum via simple iteration
    # Use water-filling: adjust while diff>tol
    tol = 1e-10
    for _ in range(10):
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) < tol:
            break
        # compute free indices where we can add/subtract without hitting bounds
        # Determine direction = sign(diff)
        if diff > 0:
            # need to add; can add at indices < 1
            free = h < 1
        else:
            # need to subtract; use indices > 0
            free = h > 0
        free_idx = np.where(free)[0]
        if free_idx.size == 0:
            break
        add_per = diff / free_idx.size
        # limit add_per to not exceed bounds
        add_per = np.clip(add_per, -np.min(h[free_idx]), np.max(1 - h[free_idx]))
        h[free_idx] += add_per
        # reclip
        h = np.clip(h, 0.0, 1.0)
    # after loop, adjust small diff by adding to first non-bound element
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-8:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def compute_min_auto_corr(h):
    auto = np.correlate(h, h, mode='full')
    # exclude central index
    n = len(h)
    mid = n - 1
    # maybe take only positive lags?
    # O_min over all lags (including negative) same as min over auto array
    # remove central.
    # compute minimum excluding central index:
    auto_ex = np.concatenate([auto[:mid], auto[mid+1:]])
    return auto_ex.min()

def gradient_for_lag(h, lag):
    n = len(h)
    if lag == 0:
        return np.zeros_like(h)
    lag_abs = abs(lag)
    grad = np.zeros_like(h)
    # forward term: i where i+lag < n
    grad[:n - lag_abs] += h[lag_abs:]
    # backward term: i where i-lag >=0
    grad[lag_abs:] += h[:n - lag_abs]
    return grad

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    # iterative water-filling
    diff = target_sum - h.sum()
    # tolerance
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        # indices that can move in direction of diff
        if diff > 0:
            free = h < 1
        else:
            free = h > 0
        free_idx = np.where(free)[0]
        if free_idx.size == 0:
            break
        # distribute evenly across free indices
        delta = diff / free_idx.size
        # apply limited by bounds
        # for each free index, compute possible room
        if diff > 0:
            max_add = 1 - h[free_idx]
            delta = min(delta, max_add.min())
        else:
            max_sub = h[free_idx]  # positive amount can subtract
            delta = max(delta, -max_sub.max())
        h[free_idx] += delta
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
    # final small correction
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-8:
        # adjust a single free index
        # choose index not at bound
        free = (h > 0) & (h < 1)
        if np.any(free):
            idx = np.where(free)[0][0]
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        else:
            # all at boundaries; adjust any with capacity
            idx = np.argmax(h) if diff > 0 else np.argmin(h)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

import numpy as np
import time
import math

def _compute_c5(h):
    """Return the C5 value for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with its complement
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _project_to_constraints(h, target_sum):
    """
    Clip h to [0,1] and adjust it so that its sum equals target_sum.
    Simple water‑filling style projection.
    """
    h = np.clip(h, 0.0, 1.0)
    # first coarse correction by uniform offset
    for _ in range(5):
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        # indices that can still move in the needed direction
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        delta = diff / cnt
        # limit delta so we do not cross bounds
        if diff > 0:
            delta = min(delta, (1.0 - h[mask]).min())
        else:
            delta = max(delta, -h[mask].min())
        h[mask] = h[mask] + delta
        h = np.clip(h, 0.0, 1.0)

    # final tiny correction on a non‑boundary element if needed
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-9:
        free = (h > 0.0) & (h < 1.0)
        idxs = np.where(free)[0]
        if idxs.size:
            idx = idxs[0]
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        else:
            # fallback – adjust the first element (should be rare)
            h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h

def _gradient_of_worst_lag(h, lag):
    """
    Gradient of the objective f_lag(h)=∑ h_i (1-h_{i+lag}) with respect to h.
    For internal points:
        ∂f/∂h_j = 1 - h_{j+lag} - h_{j-lag}
    """
    n = h.shape[0]
    grad = np.ones_like(h)   # derivative of ∑ h_i part
    if lag == 0:
        # f_0 = ∑ h_i (1-h_i) = 1 - ∑ h_i^2
        grad = 1.0 - 2.0 * h
        return grad
    L = abs(lag)
    # forward contribution: -h_{j+lag}
    grad[: n - L] -= h[L:]
    # backward contribution: -h_{j-lag}
    grad[L:] -= h[: n - L]
    return grad

    # ------------------------------------------------------------------
    # initialise h
    # ------------------------------------------------------------------
    try:
        init = initial_h_values          # provided by the harness, if any

    target_sum = n / 2.0                    # because integral of h must be 1

    start_time = time.time()
