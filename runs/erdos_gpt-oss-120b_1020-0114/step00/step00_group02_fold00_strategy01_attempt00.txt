    def repair_solution(x, n):
        """Repair the solution to be in [0,1]^n and sum to n/2 (within tolerance)"""
        x = np.copy(x)
        x_clamped = np.clip(x, 0, 1.astype(x.dtype))
        desired = n/2.0
        tolerance = 1e-5
        max_iter = 100
        # If sum is close enough, return immediately.
        s = np.sum(x_clamped)

        if np.abs(s - desired) < tolerance:
            return x_clamped

        # Adjust phase, but if we do one adjustment, we might go outside bounds, so we do iteratively.
        for i_iter in range(max_iter):
            adjusted = x_clamped + (desired - s) / n
            x_clamped = np.clip(adjusted, 0, 1)
            s_new = np.sum(x_clamped)
            if np.abs(s_new - desired) < tolerance:
                return x_clamped
            # If the sum increased (or decreased) but we didn't reach, try again.
            s = s_new
