def compute_c5_corr(h):
    # compute conv between h and 1 - h; return max*dx
    N = h.shape[0]
    M = 2 * N
    fa = np.fft.rfft(h, n=M)
    fb = np.fft.rfft(1 - h, n=M)
    corr = np.fft.irfft(fa * np.conj(fb), n=M)
    # The relevant part is first (2*N-1)
    max_corr = np.max(corr[:2*N-1])
    return max_corr * (2.0 / N)

import numpy as np, time, math

def compute_c5(h):
    n = h.shape[0]
    g = 1.0 - h
    M = 2 * n
    # use rfft
    fa = np.fft.rfft(h, n=M)
    fb = np.fft.rfft(g, n=M)
    corr = np.fft.irfft(fa * np.conj(fb), n=M)
    # get length 2n-1 relevant part
    max_corr = np.max(corr[:2*n-1])
    return max_corr * (2.0 / n)

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    # Determine base h and upsample factor
    if 'initial_h_values' in globals():
        base = np.array(initial_h_values, dtype=np.float64)
        # upsample factor to reach around 700 points but not exceed 800
        target_n = 720  # can adjust
        up_factor = max(1, target_n // base.shape[0])
        h = np.repeat(base, up_factor)
        n = h.shape[0]
        # if exceed target, truncate
        if n > target_n:
            h = h[:target_n]
            n = target_n
    else:
        # default size
        n = 720
        h = np.full(n, 0.5, dtype=np.float64)
    # Ensure sum equals n/2 (within tolerance)
    total = h.sum()
    target_sum = n / 2
    if abs(total - target_sum) > 1e-9:
        correction = (target_sum - total) / n
        h += correction
        np.clip(h, 0.0, 1.0, out=h)
        # re-balance if needed
        diff = target_sum - h.sum()
        if abs(diff) > 1e-9:
            # distribute diff uniformly then clip
            h += diff / n
            np.clip(h, 0.0, 1.0, out=h)
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    total_budget = max(0.0, budget_s - 1.0)
    start = time.time()
    # parameters
    T0 = 0.03  # initial temperature
    iteration = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= total_budget:
            break
        # linear cooling
        T = T0 * max(0.0, 1.0 - elapsed / total_budget)
        # choose move type: majority pair transfer; some local smoothing
        if rng.random() < 0.8:
            # pair transfer move
            i, j = rng.choice(n, size=2, replace=False)
            # direction
            if rng.random() < 0.5:
                donor, receiver = i, j
            else:
                donor, receiver = j, i
            max_transfer = min(h[donor], 1.0 - h[receiver])
            if max_transfer <= 0.0:
                continue
            # step size scaling
            step_scale = 0.5 + 0.5 * (T / T0)  # between 0.5 and 1
            delta = rng.random() * max_transfer * step_scale
            if delta <= 0.0:
                continue
            new_h = h.copy()
            new_h[donor] -= delta
            new_h[receiver] += delta
        else:
            # local smoothing between three consecutive points
            i = rng.integers(1, n-1)
            direction = 1 if rng.random() < 0.5 else -1
            if direction > 0:
                # increase middle, decrease neighbors
                max_inc = min(1.0 - h[i], h[i-1], h[i+1])
                if max_inc <= 0.0:
                    continue
                step_scale = 0.3 + 0.7 * (T / T0)
                delta = rng.random() * max_inc * step_scale
                new_h = h.copy()
                new_h[i] += delta
                new_h[i-1] -= delta / 2.0
                new_h[i+1] -= delta / 2.0
            else:
                max_dec = min(h[i], 1.0 - h[i-1], 1.0 - h[i+1])
                if max_dec <= 0.0:
                    continue
                step_scale = 0.3 + 0.7 * (T / T0)
                delta = rng.random() * max_dec * step_scale
                new_h = h.copy()
                new_h[i] -= delta
                new_h[i-1] += delta / 2.0
                new_h[i+1] += delta / 2.0
        # evaluate
        new_c5 = compute_c5(new_h)
        delta_c5 = new_c5 - cur_c5
        accept = False
        if delta_c5 < 0:
            accept = True
        else:
            if T > 0 and rng.random() < math.exp(-delta_c5 / T):
                accept = True
        if accept:
            h = new_h
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = new_h.copy()
        # periodic aggressive perturbation to escape stagnation
        if iteration % 10000 == 0 and iteration > 0:
            # apply small random perturbation while preserving sum
            pert = (rng.random(n) - 0.5) * 0.02
            new_h = h + pert
            np.clip(new_h, 0.0, 1.0, out=new_h)
            # adjust sum
            diff = target_sum - new_h.sum()
            # distribute diff over entries not at bounds
            mask = (new_h > 0.0) & (new_h < 1.0)
            if mask.any():
                adjust = diff / mask.sum()
                new_h[mask] += adjust
                np.clip(new_h, 0.0, 1.0, out=new_h)
            # recompute
            new_c5 = compute_c5(new_h)
            # accept if not worse
            if new_c5 < cur_c5:
                h = new_h
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = new_h.copy()
        iteration += 1

    # final projection to feasible region
    best_h = np.clip(best_h, 0.0, 1.0)
    # correct sum precisely
    sum_best = best_h.sum()
    diff = target_sum - sum_best
    if abs(diff) > 1e-9:
        # distribute diff to first non-saturated entry
        for idx in range(n):
            if diff > 0 and best_h[idx] < 1.0:
                inc = min(1.0 - best_h[idx], diff)
                best_h[idx] += inc
                diff -= inc
            elif diff < 0 and best_h[idx] > 0.0:
                dec = min(best_h[idx], -diff)
                best_h[idx] -= dec
                diff += dec
            if abs(diff) < 1e-12:
                break
        # final correction if still off
        if abs(diff) > 1e-9:
            # uniform adjustment and clip
            best_h += diff / n
            np.clip(best_h, 0.0, 1.0, out=best_h)

    final_c5 = compute_c5(best_h)

    return (best_h, final_c5, n)

import time
import math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1‑h(x+k))dx for a step function represented by h."""
    n = h.shape[0]
    # zero‑padded linear correlation via FFT
    m = 2 * n
    fa = np.fft.rfft(h, n=m)
    fb = np.fft.rfft(1.0 - h, n=m)
    corr = np.fft.irfft(fa * np.conj(fb), n=m)
    max_corr = np.max(corr[: 2 * n - 1])
    return max_corr * (2.0 / n)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
