from scipy.optimize import minimize

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def optimize_h(initial_h, max_iter=200, time_limit=None):
    n = len(initial_h)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0) for _ in range(n)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    # Use SLSQP
    res = minimize(compute_c5, initial_h, method='SLSQP', bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': max_iter, 'ftol': 1e-9, 'disp': False})
    if not res.success:
        # fallback return initial
        return initial_h, compute_c5(initial_h)
    h_opt = res.x
    # Ensure values within bounds due to numerical errors
    h_opt = np.clip(h_opt, 0.0, 1.0)
    # Project to sum exactly (adjust small)
    diff = np.sum(h_opt) - target_sum
    if abs(diff) > 1e-8:
        # Distribute difference across non-boundary entries
        non_bound = (h_opt > 1e-12) & (h_opt < 1 - 1e-12)
        if np.any(non_bound):
            h_opt[non_bound] -= diff / np.sum(non_bound)
            h_opt = np.clip(h_opt, 0.0, 1.0)
            # Readjust
            diff = np.sum(h_opt) - target_sum
        if abs(diff) > 1e-8:
            # if still off, scale
            h_opt = h_opt * (target_sum / np.sum(h_opt))
            h_opt = np.clip(h_opt, 0.0, 1.0)
    return h_opt, compute_c5(h_opt)

from scipy.optimize import minimize

def _compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1-h(x+k)) dx for a discretized step function."""
    n = h.shape[0]
    dx = 2.0 / n
    # full correlation gives all integer shifts (including negative)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Scale/shift h so that its sum equals target while staying in [0,1]."""
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if cur == 0:
        return h
    h = h * (target / cur)
    h = np.clip(h, 0.0, 1.0)
    # tiny correction if still off
    diff = np.sum(h) - target
    if abs(diff) > 1e-10:
        # distribute diff uniformly over non‑bound entries
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        if interior.any():
            h[interior] -= diff / interior.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            # fallback uniform shift
            h -= diff / h.size
            h = np.clip(h, 0.0, 1.0)
    return h

def _optimize_h(initial_h: np.ndarray, max_iter: int, time_limit: float) -> tuple:
    """Run SLSQP from initial_h within a local time budget."""
    n = initial_h.size
    target_sum = n / 2.0
    bounds = [(0.0, 1.0) for _ in range(n)]
    eq_cons = {"type": "eq", "fun": lambda x: np.sum(x) - target_sum}
    opts = {"maxiter": max_iter, "ftol": 1e-10, "disp": False}
    # minimize max overlap → minimize _compute_c5
    res = minimize(_compute_c5, initial_h, method="SLSQP",
                   bounds=bounds, constraints=[eq_cons], options=opts)
    if not res.success:
        # return the best we have
        cand = _project_sum(initial_h, target_sum)
        return cand, _compute_c5(cand)
    cand = _project_sum(res.x, target_sum)
    return cand, _compute_c5(cand)

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # --------------------------------------------------------------------
    # Base solution (provided by the harness, otherwise a simple sinusoid)
    # --------------------------------------------------------------------
    try:
        base_h = np.array(initial_h_values, dtype=float)
    except NameError:
        n0 = 45
        x0 = np.linspace(0, 2, n0, endpoint=False)
        base_h = 0.5 + 0.5 * np.cos(np.pi * x0 / 2.0)
        base_h = _project_sum(base_h, n0 / 2.0)

    best_h = base_h
    best_c5 = _compute_c5(best_h)
    best_n = best_h.size

    # --------------------------------------------------------------------
    # Resolution schedule – we try increasing numbers of samples.
    # --------------------------------------------------------------------
    schedule = [best_n, best_n * 2, best_n * 4, best_n * 8]  # 45,90,180,360
    for n in schedule:
        # leave a safety margin
        if time.time() - start_time > budget_s * 0.92:
            break

        # ---- interpolate the current best to the new grid ----------------
        x_src = np.linspace(0, 2, best_h.size, endpoint=False)
        x_tgt = np.linspace(0, 2, n, endpoint=False)
        init = np.interp(x_tgt, x_src, best_h)
        init = _project_sum(init, n / 2.0)

        # keep the best candidate for this resolution
        local_best_h = init
        local_best_c5 = _compute_c5(local_best_h)

        # ---- primary SLSQP run -----------------------------------------
        remaining = budget_s - (time.time() - start_time)
        max_iter = max(100, int(200 * (n / 180)))  # a few more iters for larger n
        h_opt, c5_opt = _optimize_h(local_best_h, max_iter, remaining)
        if c5_opt < local_best_c5:
            local_best_h, local_best_c5 = h_opt, c5_opt

        # ---- a few random restarts (binary and noisy continuous) -------
        for _ in range(4):
            if time.time() - start_time > budget_s * 0.92:
                break
            # random binary vector with exactly n/2 ones
            if np.random.rand() < 0.5:
                idx = np.arange(n)
                np.random.shuffle(idx)
                bin_vec = np.zeros(n, dtype=float)
                bin_vec[idx[: n // 2]] = 1.0
                cand0 = bin_vec
            else:
                # continuous noise around current best
                noise = np.random.uniform(-0.07, 0.07, size=n)
                cand0 = local_best_h + noise

            cand0 = _project_sum(cand0, n / 2.0)

            # brief SLSQP (fewer iterations) to polish
            h_opt2, c5_opt2 = _optimize_h(cand0, max_iter=80, time_limit=budget_s - (time.time() - start_time))
            if c5_opt2 < local_best_c5:
                local_best_h, local_best_c5 = h_opt2, c5_opt2

        # keep globally best
        if local_best_c5 < best_c5:
            best_h, best_c5, best_n = local_best_h, local_best_c5, n

    # --------------------------------------------------------------------
    # Final projection onto constraints (just in case)
    # --------------------------------------------------------------------
    best_h = _project_sum(best_h, best_n / 2.0)
    final_c5 = _compute_c5(best_h)

    # Return as plain Python list, the bound, and the number of points.
    return best_h.tolist(), float(final_c5), int(best_n)
