def _c5_value(h, dx):
    """Return max_k ∫ h(x)(1‑h(x+k))dx for a discrete step function."""
    # same as np.correlate(h, 1‑h, mode='full') * dx

    Accept a swap only if it improves the C₅ value.
    Stops after ``deadline`` or after a long streak without improvement.
    """
    h = start_h.copy()
    cur = _c5_value(h, dx)
    n = len(h)
    no_improve = 0
    MAX_STREAK = 5000          # stop after this many failed attempts

    while time.time() < deadline and no_improve < MAX_STREAK:
        # pick a random 1 and a random 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # perform the swap
        h[i] = 0.0
        h[j] = 1.0
        new = _c5_value(h, dx)

        if new < cur:                     # improvement → keep it
            cur = new
            no_improve = 0
        else:                             # reject
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
    return h, cur


def run(seed=42, budget_s=1100, **_):
    """
    Returns a step function h (as a NumPy array), its C₅ bound,
    and the number of sample points.
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # starting configuration
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        n = 80
        h0 = np.zeros(n, dtype=float)
        h0[: n // 2] = 1.0
        rng.shuffle(h0)

    n = len(h0)
    dx = 2.0 / n

    best_h = h0.copy()
    best_c5 = _c5_value(best_h, dx)

    # ------------------------------------------------------------------
    # main optimisation loop – many short hill‑climbs with random restarts
    # ------------------------------------------------------------------
    while time.time() - start < budget_s - 1.0:
        # random binary start (exactly half ones)
        m = n // 2
        cur = np.zeros(n, dtype=float)
        cur[:m] = 1.0
        rng.shuffle(cur)

        # give each climb a small time budget (≈0.3 s)
        end_climb = time.time() + 0.30
        h_loc, c5_loc = _hill_climb(cur, dx, rng, end_climb)
        if c5_loc < best_c5:
            best_c5, best_h = c5_loc, h_loc.copy()

        # polish the current best a little more
        end_polish = time.time() + 0.20
        h_pol, c5_pol = _hill_climb(best_h, dx, rng, end_polish)
        if c5_pol < best_c5:
            best_c5, best_h = c5_pol, h_pol.copy()

    # ------------------------------------------------------------------
    # final check – ensure constraints are still satisfied
    # ------------------------------------------------------------------
    # (they are by construction, but a quick assert helps catch bugs)
    assert np.all((0.0 <= best_h) & (best_h <= 1.0))
    assert abs(best_h.sum() - n / 2) < 1e-9

    return best_h, float(best_c5), n
