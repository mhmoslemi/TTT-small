def compute_c5(h):

    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def random_binary_vector(n):
    v = np.zeros(n)
    v[:n//2] = 1.0
    np.random.shuffle(v)
    return v

def compute_c5(v):
    dx = 2.0 / len(v)
    return np.max(np.correlate(v, 1 - v, mode='full')) * dx

def hill_climb(v, max_iters, start_time, time_limit):
    n = len(v)
    best_v = v.copy()
    best_score = compute_c5(v)
    current_v = v.copy()
    current_score = best_score

    iteration = 0
    # Tempering schedule maybe linear
    while iteration < max_iters and time.time() - start_time < time_limit:
        # propose swap
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if current_v[i] == current_v[j]:
            continue
        # swap
        current_v[i], current_v[j] = current_v[j], current_v[i]
        new_score = compute_c5(current_v)
        # Accept if improves else revert maybe with probability
        if new_score <= current_score or np.random.rand() < 0.001: # small random acceptance
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_v = current_v.copy()
        else:
            # revert swap
            current_v[i], current_v[j] = current_v[j], current_v[i]
        iteration += 1
    return best_v, best_score

def hill_climb(h, max_iters, start_time, time_budget):
    n = len(h)
    cur = h.copy()
    cur_score = compute_c5(cur)
    best = cur.copy()
    best_score = cur_score

    for t in range(max_iters):
        if time.time() - start_time > time_budget:
            break
        # propose swap
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if cur[i] == cur[j]:
            continue
        # swap
        cur[i], cur[j] = cur[j], cur[i]
        new_score = compute_c5(cur)
        # temperature:
        T = 0.02 * (1 - t / max_iters)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if T > 1e-6:
                if np.random.rand() < math.exp(-(new_score - cur_score) / T):
                    accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            # revert swap
            cur[i], cur[j] = cur[j], cur[i]

    return best, best_score

def hill_climb(h, max_iters, end_time):
    n = len(h)
    cur = h.copy()
    cur_score = compute_c5(cur)
    best = cur.copy()
    best_score = cur_score
    for iteration in range(max_iters):
        if time.time() > end_time:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        if cur[i] == cur[j]:
            continue
        # swap
        cur[i], cur[j] = cur[j], cur[i]
        new_score = compute_c5(cur)
        # temperature decreasing linearly
        T = max(1e-6, 0.02 * (1 - iteration / max_iters))
        if new_score < cur_score or np.random.rand() < math.exp(-(new_score - cur_score) / T):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            cur[i], cur[j] = cur[j], cur[i]
    return best, best_score

import numpy as np
import math
import time
import random

def compute_c5(h):
    """max_{k} ∫ h(x)(1-h(x+k))dx approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def generate_balanced_vector(alpha, offset, n):
    """Binary vector of length n with exactly n/2 ones.
    The ones correspond to the smallest n/2 fractional parts of i*alpha+offset."""
    i = np.arange(n, dtype=np.float64)
    s = (i * alpha + offset) % 1.0
    k = n // 2
    thresh = np.partition(s, k - 1)[k - 1]
    h = (s <= thresh).astype(np.float64)
    cur = int(h.sum())
    if cur > k:                                   # too many ones, drop some ties
        tie_idx = np.where(s == thresh)[0]
        excess = cur - k
        if excess and tie_idx.size:
            drop = np.random.choice(tie_idx, excess, replace=False)
            h[drop] = 0.0
    elif cur < k:                                 # too few ones, add the smallest above thresh
        need = k - cur
        above_idx = np.where(s > thresh)[0]
        if above_idx.size >= need:
            order = np.argpartition(s[above_idx], need - 1)[:need]
            add = above_idx[order]
            h[add] = 1.0
        else:                                     # fallback random fill
            zero_idx = np.where(h == 0)[0]
            add = np.random.choice(zero_idx, need, replace=False)
            h[add] = 1.0
    return h

def search_balanced_for_n(n, end_time):
    """Exhaustive grid search over (alpha,offset) for a given n."""
    best_h = None
    best_score = np.inf
    alphas = np.linspace(0.3, 0.9, 61)          # ≈0.01 step
    offsets = np.linspace(0.0, 0.99, 100)        # ≈0.01 step
    for alpha in alphas:
        for offset in offsets:
            if time.time() > end_time:
                return best_h, best_score
            h = generate_balanced_vector(alpha, offset, n)
            sc = compute_c5(h)
            if sc < best_score:
                best_score = sc
                best_h = h.copy()
    return best_h, best_score

def hill_climb(h0, max_iters, end_time):
    """Simple simulated‑annealing style local search on binary vectors."""
    n = h0.shape[0]
    cur = h0.copy()
    cur_sc = compute_c5(cur)
    best = cur.copy()
    best_sc = cur_sc
    T0 = 0.02
    for it in range(max_iters):
        if time.time() > end_time:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        if cur[i] == cur[j]:
            continue
        # swap a 1 and a 0
        cur[i], cur[j] = cur[j], cur[i]
        new_sc = compute_c5(cur)
        T = max(1e-8, T0 * (1.0 - it / max_iters))
        if new_sc < cur_sc or np.random.rand() < math.exp(-(new_sc - cur_sc) / T):
            cur_sc = new_sc
            if new_sc < best_sc:
                best_sc = new_sc
                best = cur.copy()
        else:
            cur[i], cur[j] = cur[j], cur[i]   # revert
    return best, best_sc
