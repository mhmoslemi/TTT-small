def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math, random
    random.seed(seed)
    np.random.seed(seed)

    # choose n_points
    n_points = 150 # maybe adjust
    dx = 2.0/n_points
    target_sum = n_points/2

    # init h
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        # ensure 1D
        init = init.ravel()
        x_old = np.linspace(0,2, len(init))
        x_new = np.linspace(0,2, n_points)
        h = np.interp(x_new, x_old, init)
    else:
        x = np.linspace(0,2,n_points)
        h = 1 - np.abs(x - 1)
    # project to sum & box
    h = project_to_sum_box(h, target_sum)

    # compute initial value
    best_h = h.copy()
    best_val = c5_of(h)
    cur_h = h.copy()
    cur_val = best_val

    start = time.time()
    # temperature schedule parameters
    T0 = best_val * 0.5 + 0.01   # initial
    # Number of steps determined dynamically
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= budget_s - 0.5:  # left small margin
            break
        # generate move
        i, j = np.random.randint(0, n_points, size=2)
        # ensure i != j
        if i == j:
            continue
        # choose direction: transfer from j to i
        if random.random() < 0.5:
            # increase i, decrease j
            max_delta = min(1 - cur_h[i], cur_h[j])
            if max_delta <= 0:
                continue
            delta = random.random() * max_delta
            h_new = cur_h.copy()
            h_new[i] += delta
            h_new[j] -= delta
        else:
            # increase j, decrease i
            max_delta = min(1 - cur_h[j], cur_h[i])
            if max_delta <= 0:
                continue
            delta = random.random() * max_delta
            h_new = cur_h.copy()
            h_new[j] += delta
            h_new[i] -= delta
        # evaluate
        val_new = c5_of(h_new)
        if val_new < cur_val:
            accept = True
        else:
            T = max(1e-8, T0 * (1 - elapsed / budget_s))
            accept = random.random() < math.exp(-(val_new - cur_val) / T)
        if accept:
            cur_h = h_new
            cur_val = val_new
            if cur_val < best_val:
                best_val = cur_val
                best_h = cur_h.copy()
    return best_h, float(best_val), int(n_points)

def project_to_sum_box(v, target):
    v = np.clip(v, 0.0, 1.0)
    # if current sum close to target, return
    s = v.sum()
    if abs(s - target) < 1e-12:
        return v
    # use binary search for lambda
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        vp = np.clip(v - mid, 0.0, 1.0)
        sz = vp.sum()
        if sz > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
```

import numpy as np, math, time, random

def _project_to_sum_box(v, target):
    """Project v onto the box [0,1]^n while enforcing sum(v)=target."""
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target) < 1e-12:
        return v
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        vp = np.clip(v - mid, 0.0, 1.0)
        s = vp.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _c5_value(h):
    """Maximum overlap with the complement, using the problem's discretisation."""
    dx = 2.0 / h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    # -------------------------------------------------------------
    # Choose a modest number of points – enough to capture shape, fast to eval
    n_points = 120
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # because ∑h_i*dx = 1  ->  ∑h_i = n/2

    # ----- initial candidate -------------------------------------------------
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float).ravel()
        x_old = np.linspace(0.0, 2.0, base.size)
        x_new = np.linspace(0.0, 2.0, n_points)
        h = np.interp(x_new, x_old, base)
    else:
        # a smooth cosine bump (average = 1/2, values in [0,1])
        x = np.linspace(0.0, 2.0, n_points)
        h = 0.5 + 0.5 * np.cos(np.pi * (x - 1.0))

    # enforce exact integral constraint and box bounds
    h = _project_to_sum_box(h, target_sum)

    # -------------------------------------------------------------
    best_h = h.copy()
    best_val = _c5_value(best_h)
    cur_h = best_h.copy()
    cur_val = best_val

    # Simple greedy + occasional stochastic perturbations
    no_improve = 0
    max_no_improve = 50000
    # temperature-like factor for occasional worse moves (annealing style)
    temp_start, temp_end = 0.02, 1e-5

    while True:
        now = time.time()
        if now - start_time >= budget_s - 0.5:
            break

        # propose a mass transfer between two distinct indices
        i, j = np.random.randint(0, n_points, size=2)
        if i == j:
            continue

        # direction: move from j to i or opposite
        if random.random() < 0.5:
            # increase i, decrease j
            max_delta = min(1.0 - cur_h[i], cur_h[j])
            if max_delta <= 0.0:
                continue
            delta = random.random() * max_delta
            new_h = cur_h.copy()
            new_h[i] += delta
            new_h[j] -= delta
        else:
            # increase j, decrease i
            max_delta = min(1.0 - cur_h[j], cur_h[i])
            if max_delta <= 0.0:
                continue
            delta = random.random() * max_delta
            new_h = cur_h.copy()
            new_h[j] += delta
            new_h[i] -= delta

        new_val = _c5_value(new_h)

        # accept if improvement, otherwise maybe accept with tiny probability
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # linear temperature schedule based on elapsed fraction
            frac = (now - start_time) / budget_s
            T = temp_start * (1.0 - frac) + temp_end * frac
            if T > 0 and random.random() < math.exp(-(new_val - cur_val) / T):
                accept = True

        if accept:
            cur_h, cur_val = new_h, new_val
            no_improve = 0
            if cur_val < best_val:
                best_val, best_h = cur_val, cur_h.copy()
        else:
            no_improve += 1

        # if stuck for a long time, shake the current solution
        if no_improve >= max_no_improve:

            cur_val = _c5_value(cur_h)
            no_improve = 0

    # final safety projection (tiny numerical drift possible)
    best_h = _project_to_sum_box(best_h, target_sum)
    best_val = _c5_value(best_h)
    return best_h, float(best_val), int(n_points)
