def project_onto_constraints(y, S):
    # clip + shift technique
    lo = -1.0
    hi = 1.0
    # Actually need lo maybe -np.max(y) - 1 and hi = 1 - np.min(y) + 1 etc. We'll just start broad.
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(y + lam, 0.0, 1.0)
        cur_sum = x.sum()
        if cur_sum > S:
            hi = lam
        else:
            lo = lam
    x = np.clip(y + (lo+hi)/2.0, 0.0, 1.0)
    # due to rounding may be off by small epsilon; we can correct by adjusting few entries within bounds.
    # Ensure sum close to S, we can adjust by small epsilon across non-bound entries.
    diff = S - x.sum()
    if abs(diff) > 1e-8:
        # distribute diff uniformly among free variables.
        free = (x > 1e-12) & (x < 1-1e-12)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            # fallback: adjust any element not at bound
            # find index with minimal distance to bound
            idx = np.argmin(np.abs(x - 0.5)) # maybe
            x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
    # finally ensure sum exactly:
    # if still off due to rounding, adjust
    # we can do simple loop:
    cur_sum = x.sum()
    if abs(cur_sum - S) > 1e-6:
        diff = S - cur_sum
        # adjust first element if not at bound else find another.
        for i in range(len(x)):
            avail = 1 - x[i] if diff > 0 else x[i]
            take = np.clip(diff, -avail, avail)
            x[i] += take
            diff -= take
            if abs(diff) < 1e-10:
                break
    return x

def compute_c5(h):
    N = len(h)
    dx = 2.0/N
    return np.max(np.correlate(h, 1-h, mode='full'))*dx

def subgradient(h, k):
    N = len(h)
    g = np.zeros_like(h)
    if k >= 0:
        L = N - k
        g[:L] += 1.0 - h[k:]
        g[k:] -= h[:L]
    else:
        s = -k
        L = N - s
        g[s:] += 1.0 - h[:L]
        g[:L] -= h[s:]
    return g

def project_onto_sum_box(y, target_sum):
    # projection onto intersection of x>=0, x<=1, sum x = target_sum
    # via bisection on lambda shift.
    lo = -2.0
    hi = 2.0
    # Expand lo,hi enough
    for _ in range(60):
        mid = (lo + hi)/2.0
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    x = np.clip(y + (lo + hi)/2.0, 0.0, 1.0)
    # fix small residual
    diff = target_sum - x.sum()
    if abs(diff) > 1e-9:
        # distribute diff to free indices
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)

            idx = np.argmin(np.abs(x - 0.5))
            x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
    return x

def anneal_binary(N, time_budget, rng):
    # initialize random binary solution
    h = np.zeros(N, dtype=np.float64)
    idx = rng.choice(N, N//2, replace=False)
    h[idx] = 1.0
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val
    start = time.time()
    t0 = 0.5
    t_final = 0.01
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        fraction = elapsed / time_budget
        T = t0 * (1 - fraction) + t_final * fraction  # linear cooling
        # propose swap
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new_val = compute_c5(h)
        delta = new_val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / T):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = 1.0, 0.0
    return best_h, best_val

import numpy as np
import time
import math

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a given discretised step function."""
    N = h.shape[0]
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / N)


def subgradient(h: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of the max‑overlap functional at the lag giving the maximum."""
    N = h.shape[0]
    g = np.zeros_like(h)
    if k >= 0:
        L = N - k
        if L > 0:
            g[:L] += 1.0 - h[k:]
            g[k:] -= h[:L]
    else:  # k < 0
        s = -k
        L = N - s
        if L > 0:
            g[s:] += 1.0 - h[:L]
            g[:L] -= h[s:]
    return g


def project_onto_sum_box(y: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of y onto {x | 0 <= x_i <= 1, sum x_i = target}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)

    # fix tiny residual error
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
        else:  # fallback: adjust any coordinate that is not at a hard bound
            i = np.argmin(np.abs(x - 0.5))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x


def subgradient_descent(h0: np.ndarray,
                        max_iter: int,
                        max_time: float,
                        step0: float = 0.8) -> tuple:
    """Projected sub‑gradient descent on the max‑overlap functional."""
    N = h0.shape[0]
    target = N / 2.0
    h = h0.copy()
    best_h, best_val = h.copy(), compute_c5(h)
    start = time.time()
    for it in range(1, max_iter + 1):
        if time.time() - start > max_time:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        k = int(np.argmax(corr) - (N - 1))
        g = subgradient(h, k)
        step = step0 / math.sqrt(it)
        y = h - step * g
        h = project_onto_sum_box(y, target)
        val = compute_c5(h)
        if val < best_val:
            best_val, best_h = val, h.copy()
    return best_h, best_val


def anneal_binary(N: int, time_budget: float, rng: np.random.Generator) -> tuple:
    """Simple simulated‑annealing on binary vectors with exactly N/2 ones."""
    assert N % 2 == 0
    target = N // 2
    # initialise
    h = np.zeros(N, dtype=np.float64)
    h[rng.choice(N, target, replace=False)] = 1.0
    cur_val = compute_c5(h)
    best_h, best_val = h.copy(), cur_val

    start = time.time()
    t0, t_end = 0.6, 0.01
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = t0 * (1 - frac) + t_end * frac

        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        h[i], h[j] = 0.0, 1.0
        new_val = compute_c5(h)
        delta = new_val - cur_val

        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val, best_h = new_val, h.copy()
        else:
            h[i], h[j] = 1.0, 0.0

    return best_h, best_val
