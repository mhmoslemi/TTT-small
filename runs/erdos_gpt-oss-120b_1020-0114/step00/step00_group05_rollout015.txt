import time, math
import numpy as np
import cvxpy as cp

def _c5_of(h):
    """Return the C5 value for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _sa_optimize(rng, n, start_h, budget_s):
    """Simple simulated‑annealing refinement; works for any n."""
    dx = 2.0 / n
    target = n / 2.0
    h = start_h.copy()
    # enforce exact integral (tiny rounding may appear)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for i in range(n):
            if 0.0 <= h[i] + diff <= 1.0:
                h[i] += diff
                break
    best_h = h.copy()
    best_val = _c5_of(best_h)
    cur_h = h.copy()
    cur_val = best_val

    start = time.time()
    max_iter = 300000
    for it in range(max_iter):
        if time.time() - start > budget_s * 0.94:
            break
        # linear schedule for temperature and step size
        prog = it / max_iter
        T = max(1e-6, best_val * (1.0 - prog) * 0.5)

        # pick a source (h>0) and a sink (h<1)
        src = rng.choice(np.where(cur_h > 1e-12)[0])
        sink = rng.choice(np.where(cur_h < 1.0 - 1e-12)[0])
        if src == sink:
            continue
        max_move = min(cur_h[src], 1.0 - cur_h[sink])
        if max_move <= 0.0:
            continue
        # step size shrinks with progress
        delta = max_move * rng.random() * (1.0 - prog) * 0.5 + 1e-9
        new_h = cur_h.copy()
        new_h[src] -= delta
        new_h[sink] += delta
        new_val = _c5_of(new_h)
        dE = new_val - cur_val
        if dE <= 0.0 or rng.random() < math.exp(-dE / T):
            cur_h, cur_val = new_h, new_val
            if cur_val < best_val:
                best_h, best_val = cur_h.copy(), cur_val
    # final projection onto the exact integral
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        for i in range(n):
            if 0.0 <= best_h[i] + diff <= 1.0:
                best_h[i] += diff
                break
    return best_h, best_val
