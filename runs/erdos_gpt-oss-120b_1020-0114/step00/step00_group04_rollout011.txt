    a = np.fft.fft(h, L)

def compute_c5(h):
    n = h.shape[0]
    L = 2*n - 1
    fa = np.fft.fft(h, L)
    fb = np.fft.fft(1 - h, L)
    corr = np.fft.ifft(fa * np.conj(fb))
    max_val = np.max(np.real(corr))
    return max_val * (2.0 / n)

def compute_c5(h):
    n = h.shape[0]
    L = 2*n -1
    # compute correlation using FFT
    # Use np.fft.rfft for speed? But need full correlation (real). Use np.fft.ifft conj.

    # Determine length of the discretization.
    try:
        n0 = len(initial_h_values)  # may be provided

        n0 = None

    if n0 is not None and n0 % 6 == 0:

    else:

    half = n // 2                  # required sum of h_i

    n1_floor = half // 3
