        i = random.randint(0, n-1)
        j = random.randint(0, n-1)
        if i == j: continue
        # Determine max transfer possible from i to j
        # Transfer amount d from i to j: reduces h_i by d, increases h_j by d.
        # Ensure new values within [0,1].
        max_transfer = min(step, cur_h[i], 1 - cur_h[j])
        if max_transfer <= 0:
            continue
        delta = random.random() * max_transfer
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta

                best_score = new_score
        T *= decay

import numpy as np
import time
import random
import math
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper functions (top level, no closures or lambdas)
# ----------------------------------------------------------------------
def compute_score_vec(h):
    """Return the C5 value using the pre‑imported evaluator."""
    return evaluate_erdos_solution(np.asarray(h, dtype=float))


def to_binary(vec):
    """Convert a vector to a binary one with exactly n/2 ones (n assumed even)."""
    n = len(vec)
    idx = np.argsort(-vec)                # descending order
    binary = np.zeros(n, dtype=int)
    binary[idx[: n // 2]] = 1
    return binary


def hill_climb_binary(start_vec, start_score, time_limit):
    """Simple simulated‑annealing on a binary {0,1} vector with fixed weight."""
    n = len(start_vec)
    cur_vec = start_vec.copy()
    cur_score = start_score
    best_vec = cur_vec.copy()
    best_score = cur_score

    ones = list(np.where(cur_vec == 1)[0])
    zeros = list(np.where(cur_vec == 0)[0])

    T = 0.1
    decay = 0.9999
    while time.time() < time_limit:
        i_one = random.choice(ones)
        i_zero = random.choice(zeros)

        new_vec = cur_vec.copy()
        new_vec[i_one] = 0
        new_vec[i_zero] = 1

        new_score = compute_score_vec(new_vec.astype(float))

        if new_score < cur_score or random.random() < math.exp(-(new_score - cur_score) / T):
            cur_vec = new_vec
            cur_score = new_score

            # update index lists
            ones.remove(i_one)
            zeros.append(i_one)
            zeros.remove(i_zero)
            ones.append(i_zero)

            if new_score < best_score:
                best_score = new_score
                best_vec = cur_vec.copy()

        T *= decay
        if T < 1e-8:
            T = 0.1
    return best_vec, best_score


# ----------------------------------------------------------------------
# Continuous (real‑valued) optimisation helpers
# ----------------------------------------------------------------------
CURRENT_DX = None          # will be set before continuous optimisation
CURRENT_TARGET = None      # desired sum (= n/2)


def objective_cont(x):
    """Penalty‑augmented max‑correlation objective for continuous optimisation."""
    c = np.correlate(x, 1 - x, mode='full')
    maxc = np.max(c) * CURRENT_DX
    penalty = 1e6 * abs(np.sum(x) - CURRENT_TARGET)
    return maxc + penalty


def continuous_simulated_anneal(start_h, time_limit):
    """Continuous simulated annealing that preserves the total sum."""
    n = len(start_h)
    cur_h = start_h.copy()
    cur_score = compute_score_vec(cur_h)

    best_h = cur_h.copy()
    best_score = cur_score

    T = 0.05
    decay = 0.9995
    step = 0.5

    while time.time() < time_limit:
        i, j = random.sample(range(n), 2)          # transfer from i to j
        max_transfer = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_transfer <= 0:
            continue
        delta = random.random() * max_transfer

        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta

        new_score = compute_score_vec(new_h)

        if new_score < cur_score or random.random() < math.exp(
                -(new_score - cur_score) / max(T, 1e-12)):
            cur_h = new_h
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = new_h.copy()

        T *= decay
        if T < 1e-7:
            T = 0.05
        step = max(0.001, step * decay)

    return best_h, best_score


def continuous_optimize(start_h):
    """Final refinement with a gradient‑based optimiser (L‑BFGS‑B)."""
    global CURRENT_DX, CURRENT_TARGET
    n = len(start_h)
    CURRENT_DX = 2.0 / n
    CURRENT_TARGET = n / 2.0

    bounds = [(0.0, 1.0)] * n
    res = minimize(
        objective_cont,
        start_h,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 200, 'ftol': 1e-12}
    )
    x = np.clip(res.x, 0.0, 1.0)
    s = np.sum(x)
    if s != 0:
        x = x * (CURRENT_TARGET / s)
    x = np.clip(x, 0.0, 1.0)
    final_score = compute_score_vec(x)
    return x, final_score
