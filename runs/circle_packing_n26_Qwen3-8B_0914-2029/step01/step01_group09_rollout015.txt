from scipy.optimize import minimize

    n = 26

    radii.append(0.1)
    centers.append([0.9, 0.9])

    def objective(x):

        # Initial radius is 0.1, so we start with 0.1 and allow increase
        # Use Lagrange multipliers to maximize the sum of radii
        # This is a constrained optimization problem - need to use scipy.optimize.minimize with constraints
        # So, reframe the objective as -sum(radii) and minimize
        return -np.sum(new_radii)

    # Constraints:
    # 1. Each circle must be inside the unit square
    # 2. Each pair of circles must not overlap
    # 3. Each radius must be non-negative

    # Create bounds for radii: minimum radius is 0 (can't be negative)
    bounds_r = [(0, 1) for _ in range(n)]
    # Create bounds for positions: each coordinate must be in [0, 1]
    bounds_pos = [(0, 1) for _ in range(2 * n)]
