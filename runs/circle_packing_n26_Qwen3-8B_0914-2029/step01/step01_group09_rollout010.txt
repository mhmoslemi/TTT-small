import scipy.optimize as optimize

def run_packing():

    n = 26
    centers = []
    radii = []
    
    # Hexagonal grid: 1 row with 1 circle, 2 rows with 2 circles, etc.
    # To approximate a hexagonal grid, we use a vertical stacking of rows
    
    # Build the grid of 7 rows and 4 columns (total of 26 circles)
    # Define the vertical spacing between rows
    row_spacing = 0.45
    col_spacing = 0.45
    
    row_indices = []
    col_indices = []
    
    for row in range(7):
        for col in range(4):
            row_indices.append(row)
            col_indices.append(col)

    for row, col in zip(row_indices, col_indices):
        x = col * col_spacing
        y = row * row_spacing
        centers.append([x, y])
    
    centers = np.array(centers)
    radii = np.full(n, 0.1)  # Initial guess
    
    # Define the loss function that penalizes overlap and boundary violations
    def objective(params):
        # Flatten parameters into positions and radii
        params_flat = params.reshape(-1)
        positions = params_flat[:2*n].reshape(n, 2)
        radii = params_flat[2*n:]
        
        # Compute pairwise distances
        overlaps = 0
        boundary_penalties = 0
        for i in range(n):
            x, y = positions[i]
            r = radii[i]
            # Boundary check
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                boundary_penalties += (x - r + 1e-12) if x - r < -1e-12 else (1 + 1e-12 - (x + r))
                boundary_penalties += (y - r + 1e-12) if y - r < -1e-12 else (1 + 1e-12 - (y + r))
            
            # Overlap check
            for j in range(i + 1, n):
                dx = positions[i, 0] - positions[j, 0]
                dy = positions[i, 1] - positions[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    overlap = radii[i] + radii[j] - dist
                    overlaps += overlap
        
        return overlaps + boundary_penalties
    
    # Define bounds for positions and radii
    bounds = []
    for i in range(n):
        # Positions must be in [0,1]^2
        bounds.append((0, 1))
        bounds.append((0, 1))
        # Radii must be >= 0
        bounds.append((0, 1))
    
    # Run optimization with simulated annealing
    result = optimize.minimize(
        objective,
        x0=np.concatenate([centers.reshape(-1), radii]),
        bounds=bounds,
        method="L-BFGS-B"
    )
    
    # Extract the final positions and radii

    for i in range(n):
