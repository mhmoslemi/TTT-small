import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Define the number of circles
    n = 26

    # Initialize the grid
    rows = 5
    cols = 6  # With some adjustments for 26 circles

    # Hexagonal grid with offset
    centers = []
    radii = []

    # Grid offset and radius scaling factor
    x_gap = 1.0 / (cols - 1)
    y_gap = 1.0 / (rows - 1)
    offset_x = 0.5 * x_gap
    radius_scale = 0.15

    for i in range(rows):
        for j in range(cols):
            x = j * x_gap
            y = i * y_gap
            if i % 2 == 1:
                x += offset_x
            centers.append([x, y])
            radii.append(radius_scale)

    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)

    # Optimize radius scaling based on distance between centers
    def objective(x):
        return -np.sum(x)  # Maximize sum of radii

    def constraint_overlap(x, i, j):
        dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
        return dist - (radii[i] + radii[j]) >= 0

    # Formulate constraints
    cons = []
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({"type": "ineq", "fun": lambda x, i=i, j=j: constraint_overlap(x, i, j)})

    # Bounds for radii (minimum 0)
    bounds = [(0, 1.0) for _ in range(n)]

    # Optimize
    result = optimize.minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=cons,
        method="SLSQP"
    )

    # Use the optimized radii
    radii = result.x

    # Final check (manual verify here for simplicity)
    # We'll now return the centers and radii

    return (centers, radii, np.sum(radii))
