import cvxpy as cp

def run_packing():

    n = 26

    y_min, y_max = 0.0, 1.0

    # First row starts aligned to the bottom

    # Second row: 6 circles (since hexagonal pattern may have alternating rows)
    # Third row: 7 circles again
    # Fourth row: 6 circles

    # Sixth row: 6 circles

    # Total circles = 7 + 6 + 7 + 6 + 7 + 6 = 39 (but we'll only use 26 from these)

    spacing = 0.3  # Adjust based on the square size

    grid_y = np.linspace(y_min + spacing / 2, y_max - spacing / 2, 6)

    # Generate a grid of positions based on hexagonal pattern
    centers = []
    for i in range(6):
        for j in range(7 if i % 2 == 0 else 6):
            x = grid_x[j]
            y = grid_y[i]
            centers.append((x, y))

    # Trim to 26 circles (approximately half of the 39)
    centers = centers[:26]
    centers = np.array(centers)

    # Radius variables
    radii = cp.Variable(n)

    # Objective: Maximize the sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints

    constraints = [
        radii >= 0,
        (centers[:, 0] - radii) >= x_min,

    problem = cp.Problem(objective, constraints)
