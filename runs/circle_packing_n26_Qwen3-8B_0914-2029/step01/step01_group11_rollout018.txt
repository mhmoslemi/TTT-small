import numpy as np
import cvxpy as cp

def run_packing():
    # Initialize 26 points in a grid
    n = 26
    grid = np.zeros((n, 2))
    grid[:n//2, 0] = np.linspace(0, 1, n//2)
    grid[n//2:, 1] = np.linspace(0, 1, n - n//2)

    # Define optimization variables
    centers = cp.Variable((n, 2))
    radii = cp.Variable(n)

    # Objective: maximize total sum of radii
    obj = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []

    # All centers are within the unit square
    for i in range(n):
        constraints.append(centers[i, 0] >= 0)
        constraints.append(centers[i, 0] <= 1)
        constraints.append(centers[i, 1] >= 0)
        constraints.append(centers[i, 1] <= 1)

    # Ensure each circle has non-negative radius
    for i in range(n):
        constraints.append(radii[i] >= 0)

    # Distance between every pair of circles is at least the sum of their radii
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            constraints.append(dist >= radii[i] + radii[j] - 1e-8)

    # Solve the optimization problem
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.SCS, verbose=False)

    if not prob.status in ["OPTIMAL", "OPTIMAL_INFEASIBLE"]:
        raise Exception("Optimization failed")

    # Extract results
    centers = centers.value
    radii = radii.value

    return np.array(centers), np.array(radii), np.sum(radii)
