import cvxpy as cp
from typing import Tuple

        [0.1, 0.1], [0.3, 0.1], [0.5, 0.1], [0.7, 0.1], [0.9, 0.1],
        [0.1, 0.3], [0.3, 0.3], [0.5, 0.3], [0.7, 0.3], [0.9, 0.3],

        [0.1, 0.9], [0.3, 0.9], [0.5, 0.9], [0.7, 0.9], [0.9, 0.9]
    ])

    # Define variables
    radii = cp.Variable(26)
    if np.isnan(centers).any():
        raise ValueError("Initial centers must not contain NaN values")

    # Objective: Maximize the sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []

    # Each circle must be inside [0,1]^2
    for i in range(26):
        x, y = centers[i]
        constraints.append(radii[i] <= 1 - x)
        constraints.append(radii[i] <= x)
        constraints.append(radii[i] <= 1 - y)

    # No two circles can overlap

            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = cp.sqrt(dx**2 + dy**2)
            constraints.append(dist >= radii[i] + radii[j])

    # No negative radii
    constraints.append(radii >= 0)

    # Create the problem
    prob = cp.Problem(objective, constraints)

    # Solve the problem
    try:
        prob.solve(solver=cp.SCS, max_iter=1000)
    except Exception as e:
        raise RuntimeError(f"Optimization failed: {e}")

    # Validity check
    if not prob.status in ("OPTIMUM", "OPTIMAL"):
        raise RuntimeError("Optimization did not reach an optimal solution")
