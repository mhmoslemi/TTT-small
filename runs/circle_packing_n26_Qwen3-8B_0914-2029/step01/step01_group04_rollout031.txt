    def hex_grid(n_circles, n_rows, offset=0.5):

                x = j * (1 / (n_circles - 1)) + offset
                y = i * np.sqrt(3) / 2
                row.append((x, y))

    def in_square(x, r):
        return (x[0] - r >= 0) and (x[0] + r <= 1) and (x[1] - r >= 0) and (x[1] + r <= 1)

    # Constraint 2: No two circles may overlap
    def non_overlap(x1, r1, x2, r2):

        return dist >= r1 + r2

        return np.concatenate([centers.flatten(), radii])

        bounds.extend([(0, 1), (0, 1), (0, 1)])

    var = flatten(initial_centers, initial_radii)
