import scipy.optimize as opt
from scipy.spatial.distance import pdist, squareform

def run_packing():
    # Step 1: Approximate hexagonal packing
    num_circles = 26
    rows = int(np.sqrt(num_circles))
    cols = (num_circles + rows - 1) // rows

    # Hexagonal grid parameters
    h = 1.0 / (cols + 1)  # horizontal spacing
    v = np.sqrt(3) / (2 * (rows + 1))  # vertical spacing

    centers = []
    radii = []

    # Distribute circles in a hexagonal grid
    for i in range(rows):
        for j in range(cols):
            x = j * h
            y = i * v
            centers.append([x, y])
            radii.append(0.05)

    # We'll need to optimize the radii and positions
    centers = np.array(centers)
    radii = np.array(radii)

    # Define objective: maximize sum of radii (i.e., minimize negative of sum)
    def objective(x):
        new_centers = x[:2 * num_circles].reshape((num_circles, 2))
        new_radii = x[2 * num_circles:]
        return -np.sum(new_radii)

    # Define constraints
    def constraint_overlap(x):
        new_centers = x[:2 * num_circles].reshape((num_circles, 2))
        new_radii = x[2 * num_circles:]
        dists = squareform(pdist(new_centers))
        overlaps = np.sum(dists < new_radii[:, np.newaxis] + new_radii[np.newaxis, :])
        return overlaps

    def constraint_boundary(x):
        new_centers = x[:2 * num_circles].reshape((num_circles, 2))
        new_radii = x[2 * num_circles:]
        return np.sum(np.abs(new_centers[:, 0] - new_radii) > 1e-12) + np.sum(
            np.abs(new_centers[:, 1] - new_radii) > 1e-12
        )

    # Initial guess for center coordinates (as a flat array)
    init_centers = centers.flatten()
    init_radii = radii

    x0 = np.concatenate([init_centers, init_radii])

    # Bounds: center coordinates in [0,1]^2, radii non-negative
    bounds = []
    for i in range(num_circles):
        bounds.append((0, 1))  # x-coordinate
        bounds.append((0, 1))  # y-coordinate
    for i in range(num_circles):
        bounds.append((0, 1))  # radius

    # Optimization settings
    cons = [
        {"type": "ineq", "fun": constraint_overlap},
        {"type": "ineq", "fun": constraint_boundary},
    ]

    # Use L-BFGS-B for bounded optimization
    result = opt.minimize(
        objective,
        x0=x0,
        bounds=bounds,
        constraints=cons,
        method="L-BFGS-B",
        options={"maxiter": 1000, "ftol": 1e-8}

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
