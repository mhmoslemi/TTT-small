import numpy as np
import scipy.optimize as opt
import random

def run_packing():
    # Define the number of circles and grid spacing
    n_circles = 26
    grid_rows = 5
    grid_cols = 5
    grid_spacing = 1.0 / grid_rows

    # Initial grid-based placement
    centers = []
    for row in range(grid_rows):
        for col in range(grid_cols):
            x = col * grid_spacing
            y = row * grid_spacing
            centers.append([x, y])
    centers = np.array(centers)

    radii = np.full(n_circles, grid_spacing / 2)

        for i in range(n_circles):
            x, y = centers[i]
            constraint.append(x - radii[i] + 1e-8)
            constraint.append(x + radii[i] - 1e-8)

            constraint.append(y + radii[i] - 1e-8)
        return constraint
