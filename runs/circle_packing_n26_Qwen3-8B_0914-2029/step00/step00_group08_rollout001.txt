import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():
    n = 26
    np.random.seed(42)
    
    # Initial guess for centers: grid of points in [0,1]^2
    grid_size = int(np.ceil(np.sqrt(n)))
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // grid_size
        col = i % grid_size
        centers[i, 0] = col / (grid_size - 1)
        centers[i, 1] = row / (grid_size - 1)
    
    # Initial guess for radii: uniform
    radii = np.full(n, 0.05)
    
    # Bounds for the problem
    bounds = Bounds(np.zeros(n), np.full(n, 1))
    
    def objective(radii):
        return -np.sum(radii)  # Minimize negative of sum to maximize it
    
    def constraints(radii):
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                distance = np.sqrt(dx*dx + dy*dy)
                constraint = distance - (radii[i] + radii[j]) >= -1e-8
                constraints.append(constraint)
        return constraints
    
    # Initial values
    x0 = radii
    
    # Optimize
    result = minimize(
        fun=objective,
        x0=x0,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraints}
        ]
    )
    
    radii = result.x
    sum_radii = np.sum(radii)
    
    return (centers, radii, sum_radii)
