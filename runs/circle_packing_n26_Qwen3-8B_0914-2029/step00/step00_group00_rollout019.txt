import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint, NonlinearConstraint

def run_packing():
    # Step 1: Create a base grid of points (Initial guess for circle centers)
    # We will create a 5x5 + 1 = 26 point grid.
    # x and y coordinates are based on a more efficient hexagonal packing pattern.
    num_rows = 6
    num_cols = 5
    centers = []
    for i in range(num_rows):
        for j in range(num_cols):
            x = j / (num_cols - 1)
            y = i / (num_rows - 1)
            centers.append([x, y])

    centers = np.array(centers)
    radii = np.full(len(centers), 0.05)  # Initial guess radii
    
    # Objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum
    
    # Constraints: Non-overlapping and boundary
    def constraint_func(radii, centers):
        n = len(centers)
        for i in range(n):
            x, y = centers[i]
            r_i = radii[i]
            if x - r_i < -1e-12 or x + r_i > 1 + 1e-12 or y - r_i < -1e-12 or y + r_i > 1 + 1e-12:
                return np.inf

                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return np.inf
        return 0
    
    # Define bounds for radii (non-negative)
    bounds = Bounds([0.0] * len(centers), [np.inf] * len(centers))
    
    # Optimization
    result = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=[NonlinearConstraint(constraint_func, 0, 0, args=(centers,))],
        method='SLSQP'
    )

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
