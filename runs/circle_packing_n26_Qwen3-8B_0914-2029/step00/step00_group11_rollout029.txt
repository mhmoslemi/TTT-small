import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Define the initial grid placement
    def grid_centers(n, boundary_offset=0.05):
        k = int(np.sqrt(n))
        if k * k < n:
            k += 1
        m = n // k + (1 if n % k else 0)
        centers = []
        for i in range(m):
            for j in range(k):
                x = j * (1 - 2*boundary_offset) / (k - 1)
                y = i * (1 - 2*boundary_offset) / (m - 1)
                centers.append([x, y])
        return np.array(centers)

    # Initial centers based on a grid
    centers = grid_centers(26)
    # Initial radii, all the same for the grid pattern
    radii = np.full(26, 0.05)

    # Fitness function to maximize sum of radii
    def objective(radii_vector):
        new_radii = np.copy(radii)
        new_radii[:] = radii_vector
        return -np.sum(new_radii)  # Minimize negative sum to maximize sum

    # Constraints:
    # 1. Each circle must be within the unit square
    # 2. The distance between any two circles must be >= sum of their radii

    def constraint1(radii_vector):
        new_radii = np.copy(radii)
        new_radii[:] = radii_vector
        for i in range(len(centers)):
            x, y = centers[i]
            r = new_radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
                return -np.inf
        return 0

    def constraint2(radii_vector):
        new_radii = np.copy(radii)
        new_radii[:] = radii_vector
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < new_radii[i] + new_radii[j] - 1e-12:
                    return -np.inf
        return 0

    # Create bounds for radii (non-negative)
    bounds = [(0, 1.0) for _ in range(26)]

    # Set up the optimization problem
    res = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint1},
            {'type': 'ineq', 'fun': constraint2}
        ],
        method='SLSQP'
    )

    if not res.success:
        raise RuntimeError("Optimization failed")

    # Final centers and radii
    final_centers = centers
    final_radii = res.x
    sum_radii = np.sum(final_radii)

    return final_centers, final_radii, sum_radii
