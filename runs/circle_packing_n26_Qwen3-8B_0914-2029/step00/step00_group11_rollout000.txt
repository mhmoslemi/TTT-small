        grid.append((x, y))

    centers = np.array(grid)
    radii = np.ones(n) * 0.05  # Initial small radius

    # Define the optimization problem for radius
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

            return (x_i - r) - 0.0  # Don't let circle go beyond left

        constraints.append(opt.Constraint(constraint_boundary(i), name=f"left_{i}"))
