import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import NonlinearConstraint

def run_packing():
    # Define initial positions (hexagonal grid in the center)
    n = 26
    grid = []
    for i in range(n):
        x = i / 5 + 0.25
        y = (i // 5) * np.sqrt(3) / 2 + 0.5
        grid.append((x, y))
    centers = np.array(grid)

    # Initial guess for radii (all equal)
    radii = np.full(n, 0.1)

    def objective(radii):
        return -np.sum(radii)

    def non_overlap_constraint(radii, centers):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -10000
        return 0

    def boundary_constraint(radii, centers):
        for i in range(n):
            x, y = centers[i]
            if x - radii[i] < -1e-8 or x + radii[i] > 1 + 1e-8:
                return -10000
            if y - radii[i] < -1e-8 or y + radii[i] > 1 + 1e-8:
                return -10000
        return 0

    # Constraints
    cons = [
        NonlinearConstraint(non_overlap_constraint, -np.inf, 0, args=(centers,)),
        NonlinearConstraint(boundary_constraint, -np.inf, 0, args=(centers,))
    ]

    bounds = Bounds([0, 0], [1, 1])

    # Optimization
    result = minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 1000, 'ftol': 1e-10, 'eps': 1e-8}
    )

    centers_final = centers
    radii_final = result.x
    sum_radii = np.sum(radii_final)

    return (centers_final, radii_final, sum_radii)
