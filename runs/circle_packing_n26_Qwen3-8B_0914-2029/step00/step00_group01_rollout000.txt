import numpy as np
import cvxpy as cp

def run_packing():
    # Define initial grid of centers for circle placement
    # We'll place 5 rows and 5 columns of circles with some vertical spacing for hexagon-like packing
    rows = 5
    cols = 5
    spacing_x = 1 / (cols - 1)  # 1/4 of unit square side
    spacing_y = 1 / (rows - 1)  # 1/4 of unit square side

    # Generate initial grid points with a hexagonal-like pattern
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = j * spacing_x
            y = i * spacing_y
            if i % 2 == 0:
                x += spacing_x / 2  # Offset for hexagonal pattern
            centers.append((x, y))

    centers = np.array(centers)

    # Add additional circles in gaps (we need a total of 26)
    # These will be manually added in "empty" space to increase total
    extra_centers = []
    for i in range(2):
        for j in range(4):
            x = j * spacing_x + 0.1
            y = i * spacing_y + 0.1
            if i % 2 == 0:
                x += spacing_x / 2
            extra_centers.append((x, y))
    extra_centers.append((0.1, 0.7))
    extra_centers.append((0.9, 0.7))
    extra_centers.append((0.1, 0.3))
    extra_centers.append((0.9, 0.3))
    extra_centers.append((0.5, 0.3))
    extra_centers.append((0.5, 0.7))
    extra_centers.append((0.4, 0.5))
    extra_centers.append((0.6, 0.5))
    centers = np.vstack([centers, extra_centers])

    # Define radii variable (primal variable to optimize)
    n = centers.shape[0]
    radii = cp.Variable(n)

    # Define constraints
    constraints = []
    for i in range(n):
        # Circle i is inside the unit square
        constraints.append(centers[i, 0] - radii[i] >= -1e-12)
        constraints.append(centers[i, 0] + radii[i] <= 1 + 1e-12)
        constraints.append(centers[i, 1] - radii[i] >= -1e-12)
        constraints.append(centers[i, 1] + radii[i] <= 1 + 1e-12)

    # No overlaps between circles
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            constraints.append(dist >= radii[i] + radii[j] - 1e-12)

    # Objective: maximize sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Solve the optimization problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS)

    # Convert results to NumPy arrays
    centers_np = centers
    radii_np = radii.value

    # Numerical stability check
    if np.isnan(radii_np).any():
        raise ValueError("NaN values in radii")

    if np.any(radii_np < 0):
        raise ValueError("Negative radii found")

    # Return the results
    return centers_np, radii_np, np.sum(radii_np)
