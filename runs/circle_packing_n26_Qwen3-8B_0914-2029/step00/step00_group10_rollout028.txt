import scipy.optimize as optimize

    ]

    for k in range(16, n):
        if k - 16 < len(corner_positions):
            x, y = corner_positions[k - 16]
            centers[k] = [x, y]

    # Initial guess for radii: all equal, small to ensure non-overlapping
    radii = np.full(n, 0.05)

    # Define objective function to maximize sum of radii
    def objective(params, centers, n):
        r = params.reshape(n)
        return -np.sum(r)  # minimize negative sum to maximize sum

    # Define constraints: all circles must stay within [0,1]^2
    def boundary_constraints(params, centers, n):
        r = params.reshape(n)
        return np.array([

        ]) + np.array([
            centers[i, 0] + r[i] <= 1 + 1e-12 for i in range(n)
        ]) + np.array([

    sum_radii = np.sum(radii)
