from scipy.optimize import minimize
from scipy.optimize.slsqpv import minimize_slsqp
from typing import Tuple, List

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Step 1: Define a hexagonal grid with seed points in a unit square
    num_circles = 26
    desired_seeds_per_row = 6  # More tightly packed
    row_spacing = 1.0 / (desired_seeds_per_row + 1)
    total_rows = 5  # Adjust based on total circles
    seeds = []

    # Generate seeds for the grid
    for row in range(total_rows):
        for col in range(desired_seeds_per_row):
            x = col * row_spacing
            y = row * row_spacing
            seeds.append([x, y])

    for i in range(len(seeds), num_circles):
        x = np.random.uniform(0, 1)
        y = np.random.uniform(0, 1)
        seeds.append([x, y])

    seeds = np.array(seeds)

    def objective(radii):

            x_i = x[i]
            x_j = x[j]
            dist = np.sqrt((x_i[0] - x_j[0])**2 + (x_i[1] - x_j[1])**2)
            return dist - (radii[i] + radii[j]) - 1e-6
        return constraint
