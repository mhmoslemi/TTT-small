import scipy.optimize as opt
import scipy.signal as signal
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(signal.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # Assume this is the starting point

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)

    n_points = 100  # Use fewer points for faster computation
    dx = 2.0 / n_points

        h = h_flat.reshape(n_points)
        h = np.clip(h, 0, 1)
        # Compute the overlap for all shifts
        corr = signal.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - n_points / 2

    # Setup bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}

            h.flatten(),
            bounds=bounds,

        if np.abs(current_c5 - best_c5) < 1e-6:
            break

    return best_h, best_c5, n_points
