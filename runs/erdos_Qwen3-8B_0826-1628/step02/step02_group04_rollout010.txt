```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Define parameters
    n_points = 100  # Start with 100 samples for efficiency
    dx = 2.0 / n_points
    
    # Initial guess
    h = initial_h_values(n_points) if n_points == 100 else np.random.rand(n_points)
    h = h * (n_points / 2) / np.sum(h)  # Scale to maintain integral constraint
    
    # Define objective function
    def objective_function(h_values):
        h = h_values.reshape(-1)
        h = np.clip(h, 0, 1)
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Define constraints
    def constraints(h_values):
        h = h_values.reshape(-1)
        return np.sum(h) - (n_points / 2)
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization settings
    max_iterations = 100
    tolerance = 1e-5
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = objective_function(best_h)
    
    for _ in range(max_iterations):
        # Local optimization
        result = optimize.minimize(
            objective_function,
            h,
            bounds=bounds,
            method='L-BFGS-B',
            jac=None,
            options={'ftol': tolerance, 'maxiter': 50}
        )
        
        # Check if improvement
        if result.success:
            h = result.x
            current_c5 = objective_function(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
            if time.time() - start_time > budget_s - 1:
                break
        
        # Add some random perturbation to avoid getting stuck
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h = np.clip(h + perturbation, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = objective_function(best_h)
    return best_h, final_c5, n_points
```
