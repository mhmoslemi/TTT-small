```python
import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, LinearConstraint
import cvxpy as cp

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    ones_minus_h = 1 - h
    corr = np.correlate(h, ones_minus_h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values():
    n_points = 45
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < 15:
            h[i] = 1
        elif i < 30:
            h[i] = 0
        else:
            h[i] = 1
    return h.tolist()

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Try different n_points starting from a lower value to save time
    n_points_list = [40, 60, 80, 100]
    best_c5 = 1.0
    best_h = None
    best_n = 0
    
    for n_points in n_points_list:
        dx = 2.0 / n_points
        # Start from initial construction
        h_initial = np.zeros(n_points)
        initial_h = initial_h_values()
        initial_h = np.resize(initial_h, n_points)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
        
        # Define the objective function
        def objective(h):
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            c5 = evaluate_erdos_solution(h)
            return c5
        
        # Define bounds and constraints
        bounds = Bounds(0, 1)
        constraints = [LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2)]
        
        # Optimization
        start_time = time.time()
        result = opt.minimize(
            fun=objective,
            x0=initial_h,
            bounds=bounds,
            constraints=constraints,
            method='L-BFGS-B',
            options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                best_n = n_points
        
        # If time is left, try a simulated annealing approach for refinement
        if time.time() - start_time < budget_s:
            result = opt.minimize(
                fun=objective,
                x0=result.x,
                bounds=bounds,
                constraints=constraints,
                method='SLSQP',
                options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
                    best_n = n_points
    
    # Ensure sum of h is correct
    best_h = best_h * (best_n / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    
    return best_h.tolist(), best_c5, best_n
```
