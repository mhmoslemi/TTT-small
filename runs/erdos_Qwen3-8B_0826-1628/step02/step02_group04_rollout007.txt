from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):

    n_points = 100  # Using 100 points for balance between speed and accuracy
    dx = 2.0 / n_points
    h = np.zeros(n_points)

    h = np.clip(h, 0, 1)  # Ensure values are within [0, 1]

        overlap = correlate(h, 1 - h, mode="full") * dx
        c5 = np.max(overlap)

        best_c5 = c5_opt

    if elapsed_time >= budget_s:
