```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.special import erf
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100  # Number of samples
    dx = 2.0 / n_points
    target_c5 = 0.380876

    # Initial guess from provided initial_h_values
    h_initial = initial_h_values[:n_points].copy()
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    # Function to compute the C5 value
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        # Ensure the integral constraint is satisfied
        if np.abs(np.sum(h) * dx - 1) > 1e-8:
            return float('inf')
        # Compute the overlap integrals
        corr = np.correlate(h, 1 - h, mode='full')
        # Only consider shifts from k=0 to k=2 (since the interval is [0,2])
        # Normalize by dx
        c5_vals = corr[1: n_points + 1] * dx
        return np.max(c5_vals)
    
    # Optimization function
    def objective_function(h):
        return compute_c5(h)
    
    # Bounds: h[i] ∈ [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial attempt using a local optimizer
    result = minimize(objective_function, h_initial, bounds=bounds, method='L-BFGS-B', tol=1e-8)
    best_h = result.x
    best_c5 = result.fun
    best_time = time.time() - start_time

    # Try a few more local optimizations with perturbations
    for _ in range(3):
        # Perturb the solution slightly
        perturb = np.random.normal(0, 0.05, size=n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
        
        result = minimize(objective_function, perturbed_h, bounds=bounds, method='L-BFGS-B', tol=1e-8)
        if result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun
    
    # If time allows, try a more intensive optimization using CVXPY
    if best_time < budget_s - 10:
        # Define variables
        h = cp.Variable(n_points)
        # Objective: minimize the maximum overlap integral
        # Compute the correlation
        corr = cp.hstack([cp.sum(h[i] * (1 - h[j + i]) for j in range(n_points)) for i in range(n_points)])
        c5 = cp.Max(corr * dx)
        # Constraint: integral of h is 1
        constraint = cp.sum(h) * dx == 1
        # Objective: minimize c5
        prob = cp.Problem(cp.Minimize(c5), [constraint])
        prob.solve(solver=cp.CVXPY_MI, eps=1e-8, verbose=False)
        
        if prob.status == 'optimal':
            best_h = h.value
            best_c5 = c5.value
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Check if we are below the target
    if best_c5 < target_c5:
        print(f"Found a better solution: C5 = {best_c5}")
    else:
        print(f"Did not improve on target: C5 = {best_c5}")
    
    # Ensure we return within the budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        return (best_h, best_c5, n_points)
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    return (best_h, best_c5, n_points)
```
