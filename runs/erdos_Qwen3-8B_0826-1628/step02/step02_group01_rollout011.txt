import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, LinearConstraint
import cvxpy as cp

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    ones_minus_h = 1 - h
    corr = np.correlate(h, ones_minus_h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values():
    n_points = 100
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < 15:
            h[i] = 1
        elif i < 30:
            h[i] = 0
        else:
            h[i] = 1
    return h.tolist()

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Use 100 points for optimization, faster than higher n
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values())
    initial_h = np.resize(initial_h, n_points)  # Ensure it's the right length
    initial_h = np.clip(initial_h, 0, 1)  # Clamp to [0, 1]
    initial_h = initial_h + np.random.uniform(-0.1, 0.1, size=n_points)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)  # Ensure sum is n_points / 2

    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    best_time = 0

    # Define the objective function
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)  # Ensure sum constraint
        c5 = evaluate_erdos_solution(h)
        return c5

    # Define bounds and constraints
    bounds = Bounds(0, 1)
    constraints = [LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2)]

    # Try multiple optimization methods and strategies
    for _ in range(3):  # Run multiple iterations with different methods
        start_time = time.time()
        # Use L-BFGS-B for global search
        result = opt.minimize(
            fun=objective,
            x0=best_h,
            bounds=bounds,
            constraints=constraints,
            method='L-BFGS-B',
            options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6}
        )
        if result.success:
            best_h = result.x
            best_c5 = result.fun
        end_time = time.time()
        best_time += end_time - start_time
        if best_time >= budget_s:
            break

    # Try SLSQP for local refinement
    if best_time < budget_s:
        start_time = time.time()
        result = opt.minimize(
            fun=objective,
            x0=best_h,
            bounds=bounds,
            constraints=constraints,
            method='SLSQP',
            options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6}
        )
        if result.success:
            best_h = result.x
            best_c5 = result.fun
        end_time = time.time()
        best_time += end_time - start_time

    # Ensure sum of h is correct
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)

    return best_h.tolist(), best_c5, n_points
