import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 100  # Start with 100 points
    h = np.array(initial_h_values(n_points))
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    norm_factor = (n_points / 2) / sum_h
    h *= norm_factor
    h = np.clip(h, 0, 1)
    
    # Optimization parameters
    max_iterations = 100
    learning_rate = 0.01
    decay_rate = 0.999
    momentum = 0.9
    epsilon = 1e-8
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Precompute all possible shifts for speed
    shifts = np.arange(-n_points + 1, n_points)
    
    for iteration in range(max_iterations):
        # Evaluate current C5
        current_c5 = evaluate_erdos_solution(h)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Check if time is up
        if time.time() - start_time > budget_s:
            break
        
        # Compute gradient approximation
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += 1e-5
            h_plus = np.clip(h_plus, 0, 1)
            h_plus_sum = np.sum(h_plus)
            norm_factor_plus = (n_points / 2) / h_plus_sum
            h_plus *= norm_factor_plus
            h_plus = np.clip(h_plus, 0, 1)
            
            h_minus = h.copy()
            h_minus[i] -= 1e-5
            h_minus = np.clip(h_minus, 0, 1)
            h_minus_sum = np.sum(h_minus)
            norm_factor_minus = (n_points / 2) / h_minus_sum
            h_minus *= norm_factor_minus
            h_minus = np.clip(h_minus, 0, 1)
            
            c5_plus = evaluate_erdos_solution(h_plus)
            c5_minus = evaluate_erdos_solution(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * 1e-5)

        grad = np.clip(grad, -1e6, 1e6)
        grad /= np.max(np.abs(grad)) + 1e-8
        
        # Update h
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        norm_factor = (n_points / 2) / sum_h
        h *= norm_factor
        h = np.clip(h, 0, 1)
        
        # Decay learning rate
        learning_rate *= decay_rate
        
        # Early stopping if improvement is minimal
        if np.abs(current_c5 - best_c5) < 1e-6:
            break

    if time.time() - start_time < budget_s - 10:
