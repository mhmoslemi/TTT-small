import numpy as np
import scipy.optimize as optimize
import time
import copy

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')  # hypothetical initial sample

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 100  # Start with 100 points for speed
    h = np.zeros(n_points)
    h[:int(n_points / 2)] = 1.0  # Initial simple step function
    
    # Optimization parameters
    max_iterations = 100
    learning_rate = 0.1
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient descent-like approach with random perturbations
    for iter in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Perturb h slightly
        h_perturbed = copy.deepcopy(h)
        for i in range(n_points):
            if np.random.random() < 0.1:  # 10% chance to perturb
                h_perturbed[i] += np.random.uniform(-0.2, 0.2)
                h_perturbed[i] = np.clip(h_perturbed[i], 0, 1)
        
        # Ensure the integral condition is approximately satisfied
        total = np.sum(h_perturbed)
        scale = (n_points / 2) / total if total != 0 else 1.0
        h_scaled = h_perturbed * scale
        
        # Evaluate the overlap integral
        c5 = evaluate_erdos_solution(h_scaled)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_scaled.copy()
        
        # Move towards better solution
        h = h_scaled.copy()
    
    # Try a higher resolution if time allows
    if time.time() - start_time < budget_s - 10:
        n_points = 200
        h = np.zeros(n_points)
        h[:int(n_points / 2)] = 1.0
        best_c5 = float('inf')
        
        for iter in range(50):
            if time.time() - start_time > budget_s - 1:
                break
            
            h_perturbed = copy.deepcopy(h)
            for i in range(n_points):
                if np.random.random() < 0.1:
                    h_perturbed[i] += np.random.uniform(-0.2, 0.2)
                    h_perturbed[i] = np.clip(h_perturbed[i], 0, 1)
            
            total = np.sum(h_perturbed)
            scale = (n_points / 2) / total if total != 0 else 1.0
            h_scaled = h_perturbed * scale
            
            c5 = evaluate_erdos_solution(h_scaled)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_scaled.copy()
            
            h = h_scaled.copy()
    
    return best_h, best_c5, n_points
