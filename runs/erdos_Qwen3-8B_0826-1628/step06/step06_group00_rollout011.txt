import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    population_size = 30
    max_iter = 150
    mutation_rate = 0.3
    crossover_rate = 0.7
    mutation_strength = 0.1
    elite_size = 5
    initial_perturbation = 0.2
    
    # Initialize a population of candidate solutions
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [np.clip(p * (n_points / 2) / sum(p), 0, 1) for p in population]
    best_solution = population[0]
    best_c5 = compute_overlap(best_solution)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness of each solution
        fitness = [compute_overlap(p) for p in population]
        
        # Select elite solutions
        elite_indices = np.argsort(fitness)[:elite_size]
        elites = [population[i] for i in elite_indices]
        
        # Create new population
        new_population = []
        
        # Add elites to new population
        new_population.extend(elites)
        
        # Generate new solutions via crossover and mutation
        while len(new_population) < population_size:
            # Select two parents
            parent1, parent2 = random.choices(population, k=2)
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
            else:
                child = parent1.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutation = np.random.uniform(-mutation_strength, mutation_strength, size=n_points)
                child += mutation
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / sum(child)
            
            new_population.append(child)
        
        # Replace population with new population
        population = new_population
        
        # Update best solution
        for p in population:
            current_c5 = compute_overlap(p)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_solution = p
    
    return (best_solution.tolist(), best_c5, n_points)
