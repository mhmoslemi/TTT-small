import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    population_size = 30
    generations = 100
    mutation_rate = 0.1
    tournament_size = 5
    
    # Initialize a population of candidate solutions
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [x * (n_points / 2) / sum(x) for x in population]
    
    best_solution = population[0]
    best_c5 = compute_overlap(best_solution)
    
    for generation in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness of all individuals
        fitness = [compute_overlap(ind) for ind in population]
        best_index = np.argmin(fitness)
        best_c5 = fitness[best_index]
        best_solution = population[best_index]
        
        # Create new population using tournament selection and mutation
        new_population = []
        for _ in range(population_size):
            # Tournament selection
            tournament = np.random.choice(range(population_size), size=tournament_size, replace=False)
            winner = tournament[np.argmin([fitness[i] for i in tournament])]
            parent = population[winner]
            
            # Apply mutation
            mutated = parent.copy()
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    mutated[i] = np.random.uniform(0, 1)
            mutated = mutated * (n_points / 2) / sum(mutated)
            
            new_population.append(mutated)
        
        population = new_population
    
    return (best_solution.tolist(), best_c5, n_points)
