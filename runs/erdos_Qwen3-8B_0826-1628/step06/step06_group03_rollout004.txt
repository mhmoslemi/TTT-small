    def compute_overlap(h_values):

        return max(corr) * (2.0 / len(h_values))

    def integral_constraint(h_values):
        return sum(h_values) - (len(h_values) / 2)

    def enforce_integral(h_values):
        integral = sum(h_values)
        if integral == 0:
            return np.zeros_like(h_values)

    # Initial configuration

    pop_size = 30

        offspring = []
