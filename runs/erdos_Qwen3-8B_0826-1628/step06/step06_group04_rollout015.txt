```python
import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, LinearConstraint
import random
from cvxopt import matrix, solvers

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    ones_minus_h = 1 - h
    corr = np.correlate(h, ones_minus_h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values():
    n_points = 60
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < 15:
            h[i] = 1
        elif i < 30:
            h[i] = 0
        else:
            h[i] = 1
    return h.tolist()

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    best_c5 = float('inf')
    best_h = None
    best_n_points = 60
    elapsed_time = 0
    time_per_n_points = budget_s / 3  # Approximate time per n_points

    # Try different values of n_points
    n_points_list = [60, 80, 100, 120, 140, 160]
    for n_points in n_points_list:
        dx = 2.0 / n_points
        # Start with the initial_h_values
        h_initial = np.array(initial_h_values())
        h_initial = np.resize(h_initial, n_points)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        # Define the objective function
        def objective(x):
            h = np.clip(x, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            c5 = evaluate_erdos_solution(h)
            return c5

        # Define the bounds and constraints
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2)]

        # Optimization using SLSQP
        start_time = time.time()
        result = opt.minimize(
            fun=objective,
            x0=h_initial,
            bounds=bounds,
            constraints=constraints,
            method='SLSQP',
            options={'maxiter': 300, 'ftol': 1e-9, 'gtol': 1e-9}
        )
        elapsed_time = time.time() - start_time

        if result.success:
            current_c5 = result.fun
            current_h = result.x
            print(f"n_points={n_points}, C5={current_c5}")
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
                best_n_points = n_points

        # Check time budget and break if exceeded
        if elapsed_time >= budget_s:
            break

    # Final clipping and normalization
    best_h = best_h * (best_n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    return best_h.tolist(), best_c5, best_n_points
```
