import numpy as np
import scipy.optimize as optimize
import time
import matplotlib.pyplot as plt

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    # Starting points
    n_points_start = len(initial_h_values)
    h_start = initial_h_values.copy()
    dx = 2.0 / n_points_start
    h_scaled = h_start * (n_points_start / 2) / sum(h_start)
    
    def compute_overlap(h_values):
        if len(h_values) != n_points_start:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points_start / 2)
    
    # Optimization: Use a hybrid approach with global perturbations and local refinements
    max_iter = 100
    best_h = h_scaled.copy()
    best_c5 = compute_overlap(best_h)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Perturb the current solution
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points_start)
        h_perturbed = np.clip(h_scaled + perturbation, 0, 1)
        h_perturbed = h_perturbed * (n_points_start / 2) / sum(h_perturbed)
        
        # Local refinement using SLSQP
        bounds = [(0, 1) for _ in range(n_points_start)]
        cons = ({'type': 'eq', 'fun': integral_constraint})
        result = optimize.minimize(lambda x: compute_overlap(x), h_perturbed, bounds=bounds, constraints=cons, method='SLSQP')
        
        if result.success and result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun
        
        # Periodically try a new starting point
        if i % 10 == 0:
            # Create a new random starting point with integral constraint
            h_random = np.random.uniform(0, 1, size=n_points_start)
            h_random = h_random * (n_points_start / 2) / sum(h_random)
            h_random = np.clip(h_random, 0, 1)
            
            # Local refinement on the random starting point
            result = optimize.minimize(lambda x: compute_overlap(x), h_random, bounds=bounds, constraints=cons, method='SLSQP')
            
            if result.success and result.fun < best_c5:
                best_h = result.x
                best_c5 = result.fun
        
    return (best_h.tolist(), best_c5, n_points_start)
