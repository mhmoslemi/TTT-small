import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, LinearConstraint
import cvxpy as cp

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    ones_minus_h = 1 - h
    corr = np.correlate(h, ones_minus_h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values():
    n_points = 40
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < 15:
            h[i] = 1
        elif i < 30:
            h[i] = 0
        else:
            h[i] = 1
    return h.tolist()

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Initial setup
    best_c5 = 1.0
    best_h = None
    best_n = 0
    time_start = time.time()
    
    # Start with initial construction
    n_points = 40
    h_initial = initial_h_values()
    h_initial = np.array(h_initial)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    # Define the objective function
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        c5 = evaluate_erdos_solution(h)
        return c5
    
    # Bounds and constraints
    bounds = Bounds(0, 1)
    constraints = [LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2)]
    
    # Simulated Annealing for global search
    result = opt.minimize(
        fun=objective,
        x0=h_initial,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6}
    )
    
    if result.success:
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
            best_n = n_points
    
    # Gradient-based refinement
    if time.time() - time_start < budget_s:
        result = opt.minimize(
            fun=objective,
            x0=result.x,
            bounds=bounds,
            constraints=constraints,
