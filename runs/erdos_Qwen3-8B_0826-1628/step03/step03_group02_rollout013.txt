```python
import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    elapsed_time = 0
    
    # Start with a high-resolution initial configuration
    n_points = 64
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    
    # Normalize to satisfy the integral constraint
    target_sum = n_points / 2
    h = h / np.sum(h) * target_sum
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Function to evaluate and update best solution
    def update_best(current_h):
        nonlocal best_h, best_c5
        current_c5 = evaluate_erdos_solution(current_h)
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
    
    # Optimization loop with multiple configurations
    for _ in range(3):  # Try different starting configurations
        # Vary the initial configuration slightly
        perturbation = np.random.normal(0, 0.1, n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * target_sum
        
        # Prepare optimization
        def objective(x):
            return evaluate_erdos_solution(x)
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        # Run optimization
        result = optimize.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            tol=1e-6,
            options={'maxiter': 100, 'ftol': 1e-9, 'gtol': 1e-9}
        )
        
        if result.success:
            update_best(result.x)
        else:
            update_best(h)
        
        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s - 5:
            break
    
    # Try a lower-resolution configuration if time permits
    if elapsed_time < budget_s - 5:
        n_points = 32
        dx = 2.0 / n_points
        h = initial_h_values[:n_points].copy()
        h = h / np.sum(h) * (n_points / 2)
        
        def objective(x):
            return evaluate_erdos_solution(x)
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - (n_points / 2)}]
        
        result = optimize.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            tol=1e-6,
            options={'maxiter': 100, 'ftol': 1e-9, 'gtol': 1e-9}
        )
        
        if result.success:
            update_best(result.x)
        else:
            update_best(h)
    
    return (best_h.tolist(), best_c5, n_points)
```
