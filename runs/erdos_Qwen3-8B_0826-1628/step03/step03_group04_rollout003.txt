import numpy as np
import time
import random
from scipy.optimize import minimize

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    ones_minus_h = 1 - h
    corr = np.correlate(h, ones_minus_h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values():
    n_points = 100
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < 15:
            h[i] = 1
        elif i < 30:
            h[i] = 0
        else:
            h[i] = 1
    return h.tolist()

def run(seed=42, budget_s=100, **kwargs):

    n_points = 100
    dx = 2.0 / n_points

    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)

    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    best_time = 0
    elapsed_time = 0

    def objective(h):

        h = h * (n_points / 2) / np.sum(h)

        return c5

    best_c5_pop = fitness[best_index]
