import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, LinearConstraint
import cvxpy as cp
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    ones_minus_h = 1 - h
    corr = np.correlate(h, ones_minus_h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values():
    n_points = 45
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < 15:
            h[i] = 1
        elif i < 30:
            h[i] = 0
        else:
            h[i] = 1
    return h.tolist()

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Use 100 points for optimization, faster than higher n
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values())
    initial_h = np.resize(initial_h, n_points)  # Ensure it's the right length
    initial_h = np.clip(initial_h, 0, 1)  # Clamp to [0, 1]
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)  # Ensure sum is n_points / 2

    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    best_time = 0

    # Define the objective function
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)  # Ensure sum constraint
        c5 = evaluate_erdos_solution(h)
        return c5

    # Define bounds and constraints
    bounds = Bounds(0, 1)
    constraints = [LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2)]

    # Use a more refined optimization approach with random restarts
    # Parameters
    max_iterations = 100
    restarts = 5
    population_size = 30
    generations = 30
    mutation_rate = 0.1
    elite_size = 5

    # Initial population
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [h * (n_points / 2) / np.sum(h) for h in population]
    population = [np.clip(h, 0, 1) for h in population]

    # Evaluate initial population
    fitness = [objective(h) for h in population]
    best_index = np.argmin(fitness)
    best_h_pop = population[best_index]
    best_c5_pop = fitness[best_index]

    # Genetic algorithm loop
    for gen in range(generations):
        # Evaluate fitness
        fitness = [objective(h) for h in population]
        best_index = np.argmin(fitness)
        best_c5_pop = fitness[best_index]

        # Select elite
        elite_indices = np.argsort(fitness)[:elite_size]
        elite = [population[i] for i in elite_indices]

        # Select parents
        parents = []
        for _ in range(population_size - elite_size):
            idx1, idx2 = np.random.choice(np.arange(len(population)), size=2, replace=False)
            if fitness[idx1] < fitness[idx2]:
                parent1 = population[idx1]
                parent2 = population[idx2]
            else:
                parent1 = population[idx2]
                parent2 = population[idx1]
            # Crossover
            child = np.zeros(n_points)
            for i in range(n_points):
                if np.random.rand() < 0.5:
                    child[i] = parent1[i]
                else:
                    child[i] = parent2[i]
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] = np.random.uniform(0, 1)
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            parents.append(child)

        # Combine elite and parents
        population = elite + parents

        # Update best solution
        if best_c5_pop < best_c5:
            best_h = best_h_pop.copy()
            best_c5 = best_c5_pop

        # Track time
        elapsed_time = time.time() - best_time
        if elapsed_time >= budget_s:
            break

    # Refinement with local optimization
    if best_time < budget_s:
        start_time = time.time()
        # Use a simulated annealing approach for local refinement
        result = opt.minimize(
            fun=objective,
            x0=best_h,
            bounds=bounds,
            constraints=constraints,
            method='SLSQP',
            options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6}
        )
        if result.success:
            best_h = result.x
            best_c5 = result.fun
        end_time = time.time()
        best_time += end_time - start_time

    # Ensure sum of h is correct
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)

    return best_h.tolist(), best_c5, n_points
