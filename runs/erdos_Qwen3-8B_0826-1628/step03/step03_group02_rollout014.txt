initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    import time
    start_time = time.time()
    
    # Define number of samples
    n_points = 128  # Increased to 128 for better precision
    dx = 2.0 / n_points
    
    # Start with a more sophisticated initial guess
    h = np.zeros(n_points)
    for i in range(n_points):
        if i % 4 == 0:
            h[i] = 0.0
        elif i % 4 == 1:
            h[i] = 0.5
        elif i % 4 == 2:
            h[i] = 1.0
        else:
            h[i] = 0.5
    
    # Normalize to satisfy the integral constraint
    target_sum = n_points / 2
    current_sum = np.sum(h)
    h = h / current_sum * target_sum
    
    # Define objective function for optimization
    def objective_function(x):
        return evaluate_erdos_solution(x)
    
    # Define constraints
    def constraint_sum(x):
        return np.sum(x) - target_sum
    
    # Define bounds for each sample (0 <= h[i] <= 1)
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization: Use a constrained optimization method with multiple restarts
    best_h = h
    best_c5 = evaluate_erdos_solution(h)
    
    # Perform multiple restarts to avoid local optima
    for _ in range(5):  # Number of restarts
        # Perturb the initial guess to avoid symmetry
        perturbation = np.random.rand(n_points) * 0.1 - 0.05
        h = best_h + perturbation
        h = np.clip(h, 0.0, 1.0)
        h = h / np.sum(h) * target_sum
        
        # Run optimization
        result = optimize.minimize(
            objective_function,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            tol=1e-6,
            options={'maxiter': 100}
        )
        
        # Check if we are running out of time
        if time.time() - start_time > budget_s:
            print("Time budget exceeded, returning best solution found so far")
            return (best_h.tolist(), best_c5, n_points)
        
        if result.success:
            current_h = result.x
            current_c5 = evaluate_erdos_solution(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
    
    # Return the best solution found
    return (best_h.tolist(), best_c5, n_points)
