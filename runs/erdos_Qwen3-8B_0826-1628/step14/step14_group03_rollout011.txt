```python
import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    h_norm = h * (n_points / 2) / sum(h)
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * (2.0 / n_points)
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    n_samples = 30
    n_iterations = 150
    perturbation = 0.2
    decay = 0.99
    local_steps = 10
    
    # Initialize a swarm of candidate solutions
    swarm = [h_norm.copy() for _ in range(n_samples)]
    best_solution = h_norm.copy()
    best_c5 = compute_overlap(best_solution)
    best_solution = best_solution.copy()
    
    for iteration in range(n_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Global exploration with perturbation and local refinement
        for i in range(n_samples):
            perturbation_vector = np.random.uniform(-perturbation, perturbation, size=n_points)
            candidate = np.clip(swarm[i] + perturbation_vector, 0, 1)
            candidate = candidate * (n_points / 2) / sum(candidate)
            bounds = [(0, 1) for _ in range(n_points)]
            cons = ({'type': 'eq', 'fun': integral_constraint})
            
            try:
                result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-9, 'maxiter': 100})
            except:
                result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='L-BFGS-B', options={'ftol': 1e-9, 'maxiter': 100})
            
            if result.success:
                swarm[i] = result.x
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_solution = result.x
        
        # Local refinement with small perturbations
        for i in range(n_samples):
            perturbation_vector = np.random.uniform(-0.05, 0.05, size=n_points)
            candidate = np.clip(swarm[i] + perturbation_vector, 0, 1)
            candidate = candidate * (n_points / 2) / sum(candidate)
            bounds = [(0, 1) for _ in range(n_points)]
            cons = ({'type': 'eq', 'fun': integral_constraint})
            
            try:
                result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-9, 'maxiter': 100})
            except:
                result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='L-BFGS-B', options={'ftol': 1e-9, 'maxiter': 100})
            
            if result.success:
                swarm[i] = result.x
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_solution = result.x
        
        # Gradually reduce perturbation range
        perturbation *= decay
    
    return (best_solution.tolist(), best_c5, n_points)
```
