import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = sum(h)
    h_norm = h * (n_points / 2) / sum_h

    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    num_iterations = 100
    population_size = 30
    mutation_rate = 0.1
    crossover_rate = 0.8
    elite_size = 2
    max_evals = 1000
    current_evals = 0
    best_c5 = float('inf')
    best_solution = h_norm.copy()
    
    # Initialize population
    def initialize_population(size):
        population = []
        for _ in range(size):
            candidate = np.random.uniform(0, 1, n_points)
            # Normalize to satisfy the integral constraint
            candidate = candidate * (n_points / 2) / np.sum(candidate)
            population.append(candidate)
        return population
    
    population = initialize_population(population_size)
    
    # Function to evaluate a candidate
    def evaluate(candidate):
        nonlocal current_evals
        if current_evals >= max_evals:
            return float('inf')
        current_evals += 1
        return compute_overlap(candidate)
    
    # Genetic algorithm
    for iteration in range(num_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [evaluate(candidate) for candidate in population]
        # Find the best solution
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_solution = population[best_idx].copy()
        
        # Select parents
        parents = np.random.choice(population_size, size=population_size, p=np.array(fitness)/np.sum(fitness))
        parents = [population[i] for i in parents]
        
        # Crossover
        offspring = []
        for i in range(0, population_size, 2):
            if i + 1 < population_size:
                parent1 = parents[i]
                parent2 = parents[i+1]
                child1 = np.zeros(n_points)
                child2 = np.zeros(n_points)
                for j in range(n_points):
                    if np.random.rand() < crossover_rate:
                        child1[j] = parent1[j]
                        child2[j] = parent2[j]
                    else:
                        child1[j] = parent2[j]
                        child2[j] = parent1[j]
                offspring.append(child1)
                offspring.append(child2)
        
        # Mutation
        for i in range(len(offspring)):
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    offspring[i][j] = np.random.uniform(0, 1)
                    # Normalize to satisfy the integral constraint
                    offspring[i] = offspring[i] * (n_points / 2) / np.sum(offspring[i])
        
        # Replace population
        population = np.array(population)
        offspring = np.array(offspring)
        population = np.vstack([population, offspring])
        # Keep the best solutions
        population = np.vstack([population[np.argsort(fitness)[:population_size - elite_size]], 
                               np.array([best_solution for _ in range(elite_size)])])
        
        # Ensure diversity
        if len(set(np.round(population, 5).flatten())) < population_size:
            population = initialize_population(population_size - len(set(np.round(population, 5).flatten())))
    
    return (best_solution.tolist(), best_c5, n_points)
