import numpy as np
import scipy.optimize as optimize
import time
import random
import math

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = sum(h)
    h = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Parameters for optimization
    n_candidates = 20
    max_iter = 50
    n_refine = 5
    perturb_scale = 0.2
    refine_scale = 0.05
    convergence_threshold = 1e-5
    
    # Initialize a set of candidate solutions
    swarm = [np.random.uniform(0, 1, n_points) for _ in range(n_candidates)]
    for i in range(n_candidates):
        # Enforce integral constraint
        swarm[i] = swarm[i] * (n_points / 2) / sum(swarm[i])
    
    best_solution = np.copy(swarm[0])
    best_c5 = compute_overlap(best_solution)
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate all candidates
        current_solutions = []
        for i in range(n_candidates):
            candidate = swarm[i].copy()
            # Apply random perturbation
            perturbation = np.random.uniform(-perturb_scale, perturb_scale, size=n_points)
            candidate += perturbation
            # Clip values to [0, 1] and normalize
            candidate = np.clip(candidate, 0, 1)
            candidate = candidate * (n_points / 2) / sum(candidate)
            current_solutions.append(candidate)
        
        # Evaluate all and find the best
        current_c5s = [compute_overlap(sol) for sol in current_solutions]
        best_idx = np.argmin(current_c5s)
        best_candidate = current_solutions[best_idx]
        best_c5 = current_c5s[best_idx]
        
        # Update the best solution
        if best_c5 < best_c5:
            best_solution = best_candidate
            best_c5 = best_c5
        
        # Refine the best candidate
        for _ in range(n_refine):
            # Apply small random perturbation
            perturbation = np.random.uniform(-refine_scale, refine_scale, size=n_points)
            candidate = best_solution + perturbation
            # Clip values to [0, 1] and normalize
            candidate = np.clip(candidate, 0, 1)
            candidate = candidate * (n_points / 2) / sum(candidate)
            
            # Local refinement with SLSQP
            bounds = [(0, 1) for _ in range(n_points)]
            cons = ({'type': 'eq', 'fun': integral_constraint})
            result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
            
            if result.success:
                best_solution = result.x
                best_c5 = result.fun
        
        # Update swarm
        for i in range(n_candidates):
            swarm[i] = best_solution + np.random.normal(0, 0.05, n_points)
            swarm[i] = np.clip(swarm[i], 0, 1)
            swarm[i] = swarm[i] * (n_points / 2) / sum(swarm[i])
        
        # Check for convergence
        if abs(best_c5 - best_c5) < convergence_threshold:
            break
    
    return (best_solution.tolist(), best_c5, n_points)
