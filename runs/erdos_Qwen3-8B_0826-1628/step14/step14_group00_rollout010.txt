import numpy as np
import scipy.optimize as optimize
import time
import random
import math

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = sum(h)
    h = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Parameters
    n_particles = 10
    max_iter = 50
    refine_steps = 3
    perturb_range = 0.3
    perturb_decay = 0.95
    refine_decay = 0.95
    elite_ratio = 0.2
    adaptive_perturb = True
    
    swarm = [h.copy() for _ in range(n_particles)]
    best_solution = h.copy()
    best_c5 = compute_overlap(best_solution)
    elite_solutions = [best_solution]
    
    def evolve_swarm(swarm, best_solution, best_c5, elite_solutions, perturb_range, perturb_decay, refine_decay, max_iter, n_points, dx, compute_overlap, integral_constraint):
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            
            # Perturb each candidate in the swarm
            for i in range(n_particles):
                # Apply random perturbation
                perturbation = np.random.uniform(-perturb_range, perturb_range, size=n_points)
                candidate = np.clip(swarm[i] + perturbation, 0, 1)
                
                # Enforce integral constraint
                if sum(candidate) == 0:
                    candidate = np.full(n_points, 0.5)
                else:
                    candidate = candidate * (n_points / 2) / sum(candidate)
                
                # Local refinement with SLSQP
                bounds = [(0, 1) for _ in range(n_points)]
                cons = ({'type': 'eq', 'fun': integral_constraint})
                result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
                
                # Update the swarm member if optimization succeeded
                if result.success:
                    swarm[i] = result.x
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_solution = result.x
                        if len(elite_solutions) < n_particles:
                            elite_solutions.append(result.x)
                        else:
                            elite_solutions.pop(0)
                            elite_solutions.append(result.x)
            
            # Reduce perturbation range
            perturb_range *= perturb_decay
            
            # Local refinement with random perturbation on elites
            for sol in elite_solutions:
                perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
                candidate = np.clip(sol + perturbation, 0, 1)
                if sum(candidate) == 0:
                    candidate = np.full(n_points, 0.5)
                else:
                    candidate = candidate * (n_points / 2) / sum(candidate)
                result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
                
                if result.success:
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_solution = result.x
            
            # Additional refinement through multi-step local optimization on best solution
            candidate = best_solution.copy()
            for _ in range(refine_steps):
                perturbation = np.random.uniform(-0.01, 0.01, size=n_points)
                candidate = np.clip(candidate + perturbation, 0, 1)
                if sum(candidate) == 0:
                    candidate = np.full(n_points, 0.5)
                else:
                    candidate = candidate * (n_points / 2) / sum(candidate)
                result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
                
                if result.success:
                    candidate = result.x
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_solution = candidate
            
        return best_solution, best_c5
    
    # Run the optimization
    best_solution, best_c5 = evolve_swarm(swarm, best_solution, best_c5, elite_solutions, perturb_range, perturb_decay, refine_decay, max_iter, n_points, dx, compute_overlap, integral_constraint)
    
    return (best_solution.tolist(), best_c5, n_points)
