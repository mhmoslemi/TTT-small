import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds, LinearConstraint

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Use a higher number of samples for better resolution, but keep it under 1000
    n_points = 100
    dx = 2.0 / n_points
    
    # Initial guess: evenly spaced segments with alternating values
    h = np.zeros(n_points)
    for i in range(n_points):
        if i % 4 == 0 or i % 4 == 1:
            h[i] = 0.5
        else:
            h[i] = 0.0
    # Normalize to satisfy the integral constraint
    total = np.sum(h)
    h = h * (n_points / 2) / total
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    max_iter = 100
    learning_rate = 0.1
    momentum = 0.9
    decay_rate = 0.995
    
    # Initialize a gradient descent optimizer
    best_solution = h.copy()
    best_c5 = compute_overlap(best_solution)
    prev_gradient = np.zeros(n_points)
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Compute the overlap and gradients
        def objective(x):
            return compute_overlap(x)
        
        # Use finite differences to approximate the gradient
        epsilon = 1e-6
        grad = np.zeros(n_points)
        for i in range(n_points):
            x_plus = x.copy()
            x_plus[i] += epsilon
            x_minus = x.copy()
            x_minus[i] -= epsilon
            grad[i] = (objective(x_plus) - objective(x_minus)) / (2 * epsilon)
        
        # Compute the negative gradient direction
        direction = -grad
        
        # Apply momentum and learning rate
        momentum_update = momentum * prev_gradient + learning_rate * direction
        prev_gradient = momentum_update
        
        # Update the solution
        new_solution = best_solution + momentum_update
        new_solution = np.clip(new_solution, 0, 1)
        # Ensure the integral constraint is satisfied
        new_solution = new_solution * (n_points / 2) / np.sum(new_solution)
        
        # Evaluate the new solution
        new_c5 = compute_overlap(new_solution)
        if new_c5 < best_c5:
            best_solution = new_solution
            best_c5 = new_c5
        
        # Decay learning rate
        learning_rate *= decay_rate
    
    return (best_solution.tolist(), best_c5, n_points)
