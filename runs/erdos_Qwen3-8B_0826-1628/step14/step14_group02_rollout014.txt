import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds, LinearConstraint

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return np.max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    max_iter = 200
    initial_perturbation = 0.3
    temp_start = 1.0
    temp_end = 0.1
    cooling_rate = 0.995
    population_size = 30
    mutation_rate = 0.15
    
    # Initialize a population of candidate solutions
    population = [h_norm.copy() for _ in range(population_size)]
    best_solution = h_norm.copy()
    best_c5 = compute_overlap(best_solution)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Compute current temperature
        temp = temp_start * (temp_end / temp_start) ** (iteration / max_iter)
        
        # Evaluate the population
        fitness = [compute_overlap(ind) for ind in population]
        best_index = np.argmin(fitness)
        best_c5 = fitness[best_index]
        best_solution = population[best_index].copy()
        
        # Apply genetic operations
        for i in range(population_size):
            # Mutation
            if np.random.rand() < mutation_rate:
                perturbation = np.random.uniform(-initial_perturbation, initial_perturbation, size=n_points)
                candidate = np.clip(population[i] + perturbation, 0, 1)
            
            else:
                # Crossover
                j = np.random.randint(0, population_size)
                candidate = np.copy(population[i])
                crossover_point = np.random.randint(1, n_points - 1)
                candidate[crossover_point:] = population[j][crossover_point:]
            
            # Enforce integral constraint
            constraint = integral_constraint(candidate)
            if abs(constraint) > 1e-6:
                # Adjust candidate to meet constraint
                if constraint > 0:
                    # Too much mass, reduce by spreading it out
                    excess = constraint / n_points
                    adjustments = np.random.uniform(-excess, 0, size=n_points)
                else:
                    # Too little mass, add by concentrating it
                    deficit = -constraint / n_points
                    adjustments = np.random.uniform(0, deficit, size=n_points)
                candidate += adjustments
                candidate = np.clip(candidate, 0, 1)
            
            # Evaluate candidate
            candidate_c5 = compute_overlap(candidate)
            
            # Accept the solution with probability based on temperature
            delta = candidate_c5 - best_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                population[i] = candidate
        
        # Gradually reduce perturbation size and temperature
        initial_perturbation *= 0.98
        temp_start *= cooling_rate
    
    return (best_solution.tolist(), best_c5, n_points)
