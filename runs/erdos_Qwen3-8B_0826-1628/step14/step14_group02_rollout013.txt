import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds, LinearConstraint

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return np.max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    max_iter = 200
    initial_perturbation = 0.4
    temp_start = 1.0
    temp_end = 0.1
    cooling_rate = 0.995
    population_size = 20
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Initialize a population of candidate solutions
    population = [h_norm.copy() for _ in range(population_size)]
    best_solution = h_norm.copy()
    best_c5 = compute_overlap(best_solution)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Compute current temperature
        temp = temp_start * (temp_end / temp_start) ** (iteration / max_iter)
        
        # Evaluate all solutions
        fitness = [compute_overlap(sol) for sol in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_solution = population[best_idx].copy()
        
        # Generate new population
        new_population = []
        for i in range(population_size):
            # Select two parents
            parent1_idx = np.random.randint(population_size)
            parent2_idx = np.random.randint(population_size)
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                child = np.zeros(n_points)
                for j in range(n_points):
                    if np.random.rand() < 0.5:
                        child[j] = parent1[j]
                    else:
                        child[j] = parent2[j]
            else:
                child = np.random.rand(n_points)
                child = np.clip(child, 0, 1)
            
            # Mutation
            if np.random.rand() < mutation_rate:
                perturbation = np.random.uniform(-initial_perturbation, initial_perturbation, size=n_points)
                child += perturbation
                child = np.clip(child, 0, 1)
            
            # Enforce integral constraint
            child_sum = np.sum(child)
            child = child * (n_points / 2) / child_sum
            
            new_population.append(child)
        
        # Replace old population with new one
        population = new_population
        
        # Adaptive perturbation and temperature
        initial_perturbation *= 0.98
        temp_start *= cooling_rate
    
    return (best_solution.tolist(), best_c5, n_points)
