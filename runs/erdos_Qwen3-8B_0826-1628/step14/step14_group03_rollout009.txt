import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    sum_h = sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * (2.0 / n_points)
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Parameters
    swarm_size = 30
    max_iter = 150
    perturbation_decay = 0.99
    mutation_rate = 0.1
    local_refinement_steps = 3
    
    # Initialize a swarm of candidate solutions
    swarm = [h_norm.copy() for _ in range(swarm_size)]
    best_solution = h_norm.copy()
    best_c5 = compute_overlap(best_solution)
    
    # Function to evolve swarm with a balance of exploration and exploitation
    def evolve_swarm(swarm, best_solution, best_c5, perturbation_decay, mutation_rate, max_iter, n_points, compute_overlap, integral_constraint):
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            
            # Exploration phase: random mutations and local optimization
            for i in range(swarm_size):
                # Apply mutations
                mutation = np.random.normal(0, mutation_rate, size=n_points)
                candidate = np.clip(swarm[i] + mutation, 0, 1)
                # Local refinement using SLSQP
                bounds = [(0, 1) for _ in range(n_points)]
                cons = ({'type': 'eq', 'fun': integral_constraint})
                try:
                    result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-9, 'maxiter': 100})
                except:
                    result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='L-BFGS-B', options={'ftol': 1e-9, 'maxiter': 100})
                
                # Update the swarm member if the optimization succeeded
                if result.success:
                    swarm[i] = result.x
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_solution = result.x
            
            # Exploitation phase: local refinement with smaller perturbations
            for i in range(swarm_size):
                # Small perturbations
                perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
                candidate = np.clip(swarm[i] + perturbation, 0, 1)
                # Ensure integral constraint is satisfied
                candidate = candidate * (n_points / 2) / sum(candidate)
                # Local refinement using SLSQP
                try:
                    result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-9, 'maxiter': 100})
                except:
                    result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='L-BFGS-B', options={'ftol': 1e-9, 'maxiter': 100})
                
                if result.success:
                    swarm[i] = result.x
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_solution = result.x
            
            # Gradually reduce mutation rate
            mutation_rate *= perturbation_decay
        
        return best_solution, best_c5
    
    # Run the optimization
    best_solution, best_c5 = evolve_swarm(swarm, best_solution, best_c5, perturbation_decay, mutation_rate, max_iter, n_points, compute_overlap, integral_constraint)
    
    return (best_solution.tolist(), best_c5, n_points)
