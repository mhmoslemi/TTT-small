import numpy as np
import scipy.optimize as optimize
import time
import random
import math

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Try different starting points
    n_points_list = [32, 64, 128]
    best_c5 = float('inf')
    best_h = None
    
    for n_points in n_points_list:
        # Initialize with the initial solution or random if not available
        if n_points == 32:
            h = initial_h_values.copy()
        else:
            h = np.random.uniform(0, 1, n_points)
        
        # Enforce integral constraint
        sum_h = sum(h)
        h = h * (n_points / 2) / sum_h
        
        def compute_overlap(h_values):
            if len(h_values) != n_points:
                return float('inf')
            corr = np.correlate(h_values, 1 - h_values, mode="full")
            return max(corr) * (2.0 / n_points)
        
        # Function to enforce the integral constraint
        def integral_constraint(h_values):
            return sum(h_values) - (n_points / 2)
        
        # Parameters
        n_particles = 10
        max_iter = 50
        refine_steps = 5
        perturb_range = 0.3
        perturb_decay = 0.95
        refine_decay = 0.95
        
        # Initialize a swarm of candidate solutions
        swarm = [h.copy() for _ in range(n_particles)]
        best_solution = h.copy()
        best_c5_current = compute_overlap(best_solution)
        
        # Function to evolve swarm with a balance of exploration and exploitation
        def evolve_swarm(swarm, best_solution, best_c5_current, perturb_range, perturb_decay, refine_decay, max_iter, n_points, compute_overlap, integral_constraint):
            for iteration in range(max_iter):
                if time.time() - start_time > budget_s:
                    break
                
                # Perturb each candidate in the swarm and refine
                for i in range(n_particles):
                    # Apply random perturbation
                    perturbation = np.random.uniform(-perturb_range, perturb_range, size=n_points)
                    candidate = np.clip(swarm[i] + perturbation, 0, 1)
                    
                    # Enforce integral constraint
                    if sum(candidate) == 0:
                        candidate = np.full(n_points, 0.5)
                    else:
                        candidate = candidate * (n_points / 2) / sum(candidate)
                    
                    # Local refinement with SLSQP
                    bounds = [(0, 1) for _ in range(n_points)]
                    cons = ({'type': 'eq', 'fun': integral_constraint})
                    result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
                    
                    # Update the swarm member if the optimization succeeded
                    if result.success:
                        swarm[i] = result.x
                        current_c5 = result.fun
                        if current_c5 < best_c5_current:
                            best_c5_current = current_c5
                            best_solution = result.x
                
                # Reduce perturbation range
                perturb_range *= perturb_decay
                
                # Local refinement with random perturbation
                for i in range(n_particles):
                    perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
                    candidate = np.clip(swarm[i] + perturbation, 0, 1)
                    if sum(candidate) == 0:
                        candidate = np.full(n_points, 0.5)
                    else:
                        candidate = candidate * (n_points / 2) / sum(candidate)
                    result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
                    
                    if result.success:
                        swarm[i] = result.x
                        current_c5 = result.fun
                        if current_c5 < best_c5_current:
                            best_c5_current = current_c5
                            best_solution = result.x
                
                # Additional refinement through multi-step local optimization
                for i in range(n_particles):
                    candidate = swarm[i].copy()
                    for _ in range(refine_steps):
                        perturbation = np.random.uniform(-0.01, 0.01, size=n_points)
                        candidate = np.clip(candidate + perturbation, 0, 1)
                        if sum(candidate) == 0:
                            candidate = np.full(n_points, 0.5)
                        else:
                            candidate = candidate * (n_points / 2) / sum(candidate)
                        result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
                        
                        if result.success:
                            candidate = result.x
                            current_c5 = result.fun
                            if current_c5 < best_c5_current:
                                best_c5_current = current_c5
                                best_solution = candidate
                    swarm[i] = candidate
            
            return best_solution, best_c5_current
        
        # Run the optimization
        best_solution, best_c5_current = evolve_swarm(swarm, best_solution, best_c5_current, perturb_range, perturb_decay, refine_decay, max_iter, n_points, compute_overlap, integral_constraint)
        
        # Update best solution
        if best_c5_current < best_c5:
            best_c5 = best_c5_current
            best_h = best_solution
    
    return (best_h.tolist(), best_c5, len(best_h))
