```python
import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds, LinearConstraint

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    h = h * (n_points / 2) / sum(h)
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Define bounds and constraints
    bounds = Bounds(0, 1)
    cons = LinearConstraint(np.ones(n_points), lb=[n_points / 2], ub=[n_points / 2])
    
    # Initial solution
    best_solution = h.copy()
    best_c5 = compute_overlap(best_solution)
    
    # Parameters for optimization
    num_candidates = 20
    max_iterations = 100
    perturbation_scale = 0.3
    decay_factor = 0.97
    exploration_rate = 0.7
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Generate candidate solutions
        candidates = []
        for _ in range(num_candidates):
            perturbation = np.random.normal(0, perturbation_scale, size=n_points)
            candidate = np.clip(h + perturbation, 0, 1)
            if sum(candidate) == 0:
                candidate = np.random.uniform(0, 1, n_points)
            else:
                candidate = candidate * (n_points / 2) / sum(candidate)
            candidates.append(candidate)
        
        # Evaluate all candidates
        candidate_c5s = []
        for candidate in candidates:
            try:
                result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-8)
                if result.success:
                    candidate_c5s.append((result.fun, result.x))
                else:
                    candidate_c5s.append((float('inf'), candidate))
            except:
                candidate_c5s.append((float('inf'), candidate))
        
        # Select best candidate
        candidate_c5s.sort()
        best_candidate_c5, best_candidate = candidate_c5s[0]
        
        # Update best solution
        if best_candidate_c5 < best_c5:
            best_c5 = best_candidate_c5
            best_solution = best_candidate
        
        # Update perturbation scale and exploration rate
        perturbation_scale *= decay_factor
        exploration_rate *= 0.99
    
    return (best_solution.tolist(), best_c5, n_points)
```
