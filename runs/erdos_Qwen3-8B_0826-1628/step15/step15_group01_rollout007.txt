def run(seed=42, budget_s=100, **kwargs):

    # Start with the initial solution (n=34)
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx

    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)

                candidate = candidate * (n_points / 2) / sum(candidate)
                # Local refinement with SLSQP
                bounds = Bounds(0, 1)
                cons = LinearConstraint(np.ones(n_points), lb=[n_points / 2], ub=[n_points / 2])

                # Update the swarm member if the optimization succeeded
                if result.success:
                    swarm[i] = result.x
                    current_c5 = result.fun

                        best_c5 = current_c5
                        best_solution = result.x
