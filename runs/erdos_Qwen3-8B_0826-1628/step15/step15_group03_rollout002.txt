def run(seed=42, budget_s=100, **kwargs):

    random.seed(seed)
    
    # Start with the initial solution (n=34)
    n_points = len(initial_h_values)

        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)

    def generate_perturbation(h_values, perturb_range):
        return np.random.uniform(-perturb_range, perturb_range, size=n_points)

    def refine_candidate(candidate, bounds, constraints, compute_overlap, max_iter=10):
        for _ in range(max_iter):
            perturbation = np.random.uniform(-0.01, 0.01, size=n_points)
            candidate = np.clip(candidate + perturbation, 0, 1)
            candidate = candidate * (n_points / 2) / sum(candidate)
            result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=constraints, method='SLSQP')
            if result.success:
                candidate = result.x
        return candidate
    
    # Optimization parameters

    max_iter = 50
    perturb_range = 0.2

    def evolve_swarm(swarm, best_solution, best_c5, perturb_range, perturb_decay, max_iter, n_points, dx, compute_overlap, integral_constraint):

            for i in range(n_particles):

                candidate = candidate * (n_points / 2) / sum(candidate)
                
                # Refine the candidate
                refined_candidate = refine_candidate(candidate, bounds, constraints, compute_overlap)

                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_solution = refined_candidate
                    swarm[i] = refined_candidate
            
            # Reduce perturbation range
            perturb_range *= perturb_decay

                    best_solution = refined_candidate
                    swarm[i] = refined_candidate
        
        return best_solution, best_c5
    
    # Run the optimization

    return (best_solution.tolist(), best_c5, n_points)
