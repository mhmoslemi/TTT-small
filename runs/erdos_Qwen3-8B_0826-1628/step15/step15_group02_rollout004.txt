import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds, minimize

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the provided initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    h = h * (n_points / 2) / np.sum(h)
    dx = 2.0 / n_points
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': integral_constraint}]
    
    # Refinement and exploration parameters
    refine_steps = 5
    refine_decay = 0.95
    perturbation_decay = 0.97
    max_refinement_attempts = 10
    
    # Initial solution
    best_h = h.copy()
    best_c5 = compute_overlap(best_h)
    
    # Optimization function
    def optimize_solution(h_initial):
        result = minimize(lambda x: compute_overlap(x), h_initial, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 100})
        if result.success:
            return result.x, result.fun
        return h_initial, float('inf')
    
    # Stochastic gradient descent with momentum and adaptive learning rates
    def stochastic_gradient_descent(h_initial, max_steps=200, learning_rate=0.1, momentum=0.9, decay=0.995):
        h = h_initial.copy()
        velocity = np.zeros_like(h)
        for step in range(max_steps):
            if time.time() - start_time > budget_s:
                break
            # Evaluate overlap
            current_c5 = compute_overlap(h)
            # Compute gradient
            perturbations = np.random.normal(0, 0.05, size=n_points)
            perturbed_h = np.clip(h + perturbations, 0, 1)
            perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
            perturbed_c5 = compute_overlap(perturbed_h)
            gradient = (perturbed_c5 - current_c5) * perturbations
            # Update velocity and parameters
            velocity = momentum * velocity - learning_rate * gradient
            h = np.clip(h + velocity, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            # Apply decay
            learning_rate *= decay
        return h, compute_overlap(h)
    
    # Refinement loop
    for _ in range(max_refinement_attempts):
        if time.time() - start_time > budget_s:
            break
        # Stochastic gradient descent refinement
        refined_h, refined_c5 = stochastic_gradient_descent(best_h)
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5
        # Local optimization
        best_h, best_c5 = optimize_solution(best_h)
        # Adaptive perturbation
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
        perturbed_c5 = compute_overlap(perturbed_h)
        if perturbed_c5 < best_c5:
            best_h = perturbed_h
            best_c5 = perturbed_c5
    
    return (best_h.tolist(), best_c5, n_points)
