import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds, LinearConstraint

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = sum(h)
    h_norm = h * (n_points / 2) / sum_h

    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx

    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Function to enforce the integral constraint
    def integral_constraint_func(h_values):
        return sum(h_values) - (n_points / 2)

    # Local search with SLSQP
    def local_search(initial_guess):
        bounds = Bounds(0, 1)
        cons = LinearConstraint(np.ones(n_points), lb=[n_points / 2], ub=[n_points / 2])
        result = optimize.minimize(lambda x: compute_overlap(x), initial_guess, bounds=bounds, constraints=cons, method='SLSQP')
        return result.x if result.success else initial_guess

    # Random perturbation and local search
    def random_perturbation_search(h_values, perturbation_range=0.3, max_iter=100):
        best_h = h_values.copy()
        best_c5 = compute_overlap(best_h)
        for _ in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            perturbation = np.random.uniform(-perturbation_range, perturbation_range, size=n_points)
            candidate = np.clip(h_values + perturbation, 0, 1)
            candidate = candidate * (n_points / 2) / sum(candidate)
            candidate = local_search(candidate)
            current_c5 = compute_overlap(candidate)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = candidate
        return best_h, best_c5

    # Start from initial solution
    best_h = h_norm.copy()
    best_c5 = compute_overlap(best_h)

    # Local refinement
    best_h, best_c5 = random_perturbation_search(best_h, perturbation_range=0.3, max_iter=50)

    # Second refinement with smaller perturbation
    best_h, best_c5 = random_perturbation_search(best_h, perturbation_range=0.1, max_iter=50)

    # Final refinement
    best_h, best_c5 = random_perturbation_search(best_h, perturbation_range=0.05, max_iter=50)

    return (best_h.tolist(), best_c5, n_points)
