import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds, LinearConstraint

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    n_particles = 50
    max_iter = 100
    inertia = 0.8
    cognitive = 1.2
    social = 1.2
    personal_best = [h_norm.copy() for _ in range(n_particles)]
    personal_best_fitness = [compute_overlap(p) for p in personal_best]
    global_best = h_norm.copy()
    global_best_fitness = compute_overlap(global_best)
    velocity = np.random.rand(n_particles, n_points) * 0.1
    
    # Main PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        for i in range(n_particles):
            # Randomly choose neighbors
            r1, r2 = np.random.rand(2)
            cognitive_component = cognitive * r1 * (personal_best[i] - np.array(h_norm))
            social_component = social * r2 * (global_best - np.array(h_norm))
            velocity[i] = inertia * velocity[i] + cognitive_component + social_component
            
            # Update position
            h_new = np.clip(np.array(h_norm) + velocity[i], 0, 1)
            # Ensure integral constraint
            if sum(h_new) == 0:
                h_new = np.random.uniform(0, 1, n_points)
            else:
                h_new = h_new * (n_points / 2) / sum(h_new)
            
            # Evaluate new fitness
            fitness_new = compute_overlap(h_new)
            
            # Update personal best
            if fitness_new < personal_best_fitness[i]:
                personal_best[i] = h_new.copy()
                personal_best_fitness[i] = fitness_new
                
                # Update global best
                if fitness_new < global_best_fitness:
                    global_best = h_new.copy()
                    global_best_fitness = fitness_new
        
        # Local refinement with SLSQP
        for i in range(n_particles):
            candidate = personal_best[i].copy()
            bounds = Bounds(0, 1)
            cons = LinearConstraint(np.ones(n_points), lb=[n_points / 2], ub=[n_points / 2])
            result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
            
            if result.success:
                personal_best[i] = result.x
                personal_best_fitness[i] = result.fun
                if result.fun < global_best_fitness:
                    global_best = result.x
                    global_best_fitness = result.fun
    
    return (global_best.tolist(), global_best_fitness, n_points)
