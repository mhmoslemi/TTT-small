import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds, LinearConstraint

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return np.max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    max_iter = 100
    temp_start = 1.0
    temp_end = 0.01
    perturbation_factor = 0.3
    n_swarm = 50
    n_local_steps = 10
    alpha = 0.95
    n_population = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Initialize a swarm of candidate solutions
    swarm = [h_norm.copy() for _ in range(n_swarm)]
    best_solution = h_norm.copy()
    best_c5 = compute_overlap(best_solution)
    
    # Hybrid approach: genetic algorithm with local optimization
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Genetic algorithm step
        population = [np.random.uniform(0, 1, size=n_points) for _ in range(n_population)]
        population = [np.clip(p, 0, 1) for p in population]
        population = [p * (n_points / 2) / np.sum(p) for p in population]
        
        # Evaluate population
        fitness = [compute_overlap(p) for p in population]
        best_idx = np.argmin(fitness)
        best_pop = population[best_idx]
        best_pop_c5 = fitness[best_idx]
        
        # Update global best
        if best_pop_c5 < best_c5:
            best_c5 = best_pop_c5
            best_solution = best_pop
        
        # Crossover and mutation
        for i in range(n_population):
            if np.random.rand() < crossover_rate:
                j = np.random.randint(0, n_population)
                crossover_point = np.random.randint(0, n_points)
                population[i] = np.concatenate(
                    (population[i][:crossover_point], population[j][crossover_point:])
                )
            if np.random.rand() < mutation_rate:
                mutation = np.random.normal(0, perturbation_factor, size=n_points)
                population[i] += mutation
                population[i] = np.clip(population[i], 0, 1)
                population[i] = population[i] * (n_points / 2) / np.sum(population[i])
        
        # Reduce perturbation size
        perturbation_factor *= alpha
    
    # Refine the best solution with additional local steps
    for _ in range(n_local_steps):
        if time.time() - start_time > budget_s:
            break
        bounds = Bounds([0.0]*n_points, [1.0]*n_points)
        cons = LinearConstraint([[1.0]*n_points], [n_points/2], [n_points/2])
        result = optimize.minimize(lambda x: compute_overlap(x), best_solution, bounds=bounds, constraints=cons, method='SLSQP')
        if result.success:
            best_c5 = result.fun
            best_solution = result.x
    
    return (best_solution.tolist(), best_c5, n_points)
