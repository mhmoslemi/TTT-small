import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds, LinearConstraint

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    swarm_size = 30
    max_iter = 150
    perturbation_decay = 0.98
    initial_perturbation = 0.3
    elite_ratio = 0.2  # Keep top 20% of swarm for next iteration
    
    # Initialize a swarm of candidate solutions
    swarm = [h_norm.copy() for _ in range(swarm_size)]
    best_solution = h_norm.copy()
    best_c5 = compute_overlap(best_solution)
    
    # Function to evolve swarm with a balance of exploration and exploitation
    def evolve_swarm(swarm, best_solution, best_c5, initial_perturbation, perturbation_decay, max_iter, n_points, dx, compute_overlap, integral_constraint):
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            
            # Evaluate and sort the swarm
            swarm_evals = [(compute_overlap(s), s) for s in swarm]
            swarm_evals.sort()
            best_swarm, _ = swarm_evals[0]
            
            # Replace worst half with random perturbations
            for i in range(len(swarm) // 2):
                perturbation = np.random.uniform(-initial_perturbation, initial_perturbation, size=n_points)
                candidate = np.clip(swarm[i] + perturbation, 0, 1)
                # Ensure integral constraint is satisfied
                candidate = candidate * (n_points / 2) / sum(candidate)
                # Local refinement with SLSQP
                bounds = Bounds(0, 1)
                cons = LinearConstraint(np.ones(n_points), lb=[n_points / 2], ub=[n_points / 2])
                result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
                
                # Update the swarm member if the optimization succeeded
                if result.success:
                    swarm[i] = result.x
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_solution = result.x
            
            # Keep elite solutions
            new_swarm = swarm[:int(elite_ratio * len(swarm))]
            # Generate new solutions by perturbing elites
            for i in range(len(new_swarm)):
                perturbation = np.random.uniform(-initial_perturbation, initial_perturbation, size=n_points)
                candidate = np.clip(new_swarm[i] + perturbation, 0, 1)
                # Ensure integral constraint is satisfied
                candidate = candidate * (n_points / 2) / sum(candidate)
                # Local refinement with SLSQP
                result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
                
                if result.success:
                    new_swarm.append(result.x)
            
            # Replace the swarm with the new_swarm
            swarm = new_swarm
            # Update best solution
            swarm_evals = [(compute_overlap(s), s) for s in swarm]
            swarm_evals.sort()
            best_swarm, _ = swarm_evals[0]
            if best_swarm < best_c5:
                best_c5 = best_swarm
                best_solution = swarm_evals[0][1]
            
            # Gradually reduce perturbation range
            initial_perturbation *= perturbation_decay
            
        return best_solution, best_c5
    
    # Run the optimization
    best_solution, best_c5 = evolve_swarm(swarm, best_solution, best_c5, initial_perturbation, perturbation_decay, max_iter, n_points, dx, compute_overlap, integral_constraint)
    
    return (best_solution.tolist(), best_c5, n_points)
