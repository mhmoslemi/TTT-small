import numpy as np
import scipy.optimize as optimize
import time
import random
import math

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution (n=34)
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = sum(h)
    h = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Define a more efficient local optimization strategy
    def local_refine(h_values, dx, n_points, compute_overlap, max_iter=20):
        bounds = [(0, 1) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': integral_constraint})
        result = optimize.minimize(lambda x: compute_overlap(x), h_values, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-9, 'maxiter': max_iter})
        if result.success:
            return result.x
        return h_values
    
    # Define a more effective global search strategy
    def global_search(h_initial, dx, n_points, compute_overlap, budget_s, max_swarm=30, max_iter=50, refine_steps=10):
        swarm = [h_initial.copy() for _ in range(max_swarm)]
        best_solution = h_initial.copy()
        best_c5 = compute_overlap(best_solution)
        best_swarm = [best_solution.copy() for _ in range(max_swarm)]
        perturb_range = 0.25
        perturb_decay = 0.98
        decay_factor = 0.99
        
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            
            # Perturb and refine the swarm
            for i in range(len(swarm)):
                # Random perturbation
                perturbation = np.random.uniform(-perturb_range, perturb_range, size=n_points)
                candidate = np.clip(swarm[i] + perturbation, 0, 1)
                candidate = candidate * (n_points / 2) / sum(candidate)
                
                # Local refinement
                refined = local_refine(candidate, dx, n_points, compute_overlap, max_iter=refine_steps)
                current_c5 = compute_overlap(refined)
                
                # Update the swarm
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_solution = refined
                swarm[i] = refined
            
            # Reduce perturbation range
            perturb_range *= perturb_decay
            # Decay the swarm diversity
            for i in range(len(swarm)):
                swarm[i] = swarm[i] * (1 - decay_factor) + best_swarm[i] * decay_factor
            
            # Update best swarm
            best_swarm = [np.copy(sol) for sol in swarm]
        
        return best_solution, best_c5
    
    # Run the global search
    best_solution, best_c5 = global_search(h, dx, n_points, compute_overlap, budget_s)
    
    return (best_solution.tolist(), best_c5, n_points)
