import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds, LinearConstraint

initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    h = h * (n_points / 2) / np.sum(h)
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return np.max(corr) * (2.0 / n_points)
    
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Function to evaluate the overlap for a given h
    def objective(h_values):
        return compute_overlap(h_values)
    
    # Function to optimize
    def optimize_h(h0):
        bounds = Bounds([0.0]*n_points, [1.0]*n_points)
        cons = LinearConstraint([[1.0]*n_points], [n_points/2], [n_points/2])
        result = optimize.minimize(objective, h0, bounds=bounds, constraints=cons, method='SLSQP')
        return result
    
    # Function to generate a new candidate solution via local perturbation
    def perturb_solution(h, perturbation_factor=0.1):
        perturbation = np.random.uniform(-perturbation_factor, perturbation_factor, size=n_points)
        candidate = np.clip(h + perturbation, 0, 1)
        return candidate
    
    # Function to run a local optimization
    def local_optimization(h):
        result = optimize_h(h)
        if result.success:
            return result.x
        else:
            return h
    
    # Function to run a greedy hill climbing
    def greedy_hill_climbing(h, max_steps=100, perturbation_factor=0.1):
        for _ in range(max_steps):
            if time.time() - start_time > budget_s:
                break
            candidate = perturb_solution(h, perturbation_factor)
            result = optimize_h(candidate)
            if result.success and result.fun < objective(h):
                h = result.x
        return h
    
    # Function to run a population-based search
    def population_search(pop_size=50, n_epochs=20, perturbation_factor=0.2):
        population = [np.random.uniform(0, 1, n_points) for _ in range(pop_size)]
        population = [h * (n_points / 2) / np.sum(h) for h in population]
        
        for _ in range(n_epochs):
            if time.time() - start_time > budget_s:
                break

            fitness = [compute_overlap(h) for h in population]
            best_idx = np.argmin(fitness)
            best_h = population[best_idx]
            best_c5 = fitness[best_idx]

            new_population = []

                # Local optimization
                optimized_h = local_optimization(h)
                new_population.append(optimized_h)
                # Perturb and optimize
                perturbed_h = perturb_solution(h, perturbation_factor)
                optimized_perturbed_h = local_optimization(perturbed_h)
                new_population.append(optimized_perturbed_h)
            
            # Replace population with new candidates
            population = new_population
            population = [np.clip(h, 0, 1) for h in population]
            population = [h * (n_points / 2) / np.sum(h) for h in population]
        
        # Return the best solution
        fitness = [compute_overlap(h) for h in population]
        best_idx = np.argmin(fitness)
        return population[best_idx], fitness[best_idx]

    h = h.copy()

    h = local_optimization(h)

    h = greedy_hill_climbing(h)
    
    # Then, run a population-based search
    best_h, best_c5 = population_search()
    
    # Final refinement with local optimization
    final_h = local_optimization(best_h)
    final_c5 = compute_overlap(final_h)
    
    # Return the best solution found
    return (final_h.tolist(), final_c5, n_points)
