import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds, LinearConstraint
from scipy.interpolate import interp1d

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)

    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = sum(h)
    h_norm = h * (n_points / 2) / sum_h

    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx

    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)

    # Function to generate a random sample with integral constraint
    def random_sample():
        h_rand = np.random.uniform(0, 1, n_points)
        h_rand = h_rand * (n_points / 2) / sum(h_rand)
        return h_rand

    # Function to generate a smooth initial guess with guided mutation
    def guided_initial_guess(best_solution):
        t = np.linspace(0, 1, n_points)
        x = np.linspace(0, 2, n_points)
        f = interp1d(t, best_solution, kind='cubic')
        h_smooth = f(x)
        h_smooth = np.clip(h_smooth, 0, 1)
        h_smooth = h_smooth * (n_points / 2) / sum(h_smooth)
        return h_smooth

    # Function to perform local optimization with bounds and constraints
    def local_optimize(candidate):
        bounds = Bounds(0, 1)
        cons = LinearConstraint(np.ones(n_points), lb=[n_points / 2], ub=[n_points / 2])
        result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
        return result.x if result.success else candidate

    # Function to generate diverse samples with different strategies
    def generate_samples(n_samples=50):
        samples = []
        for _ in range(n_samples):
            if np.random.rand() < 0.3:
                # Random mutation
                candidate = np.random.uniform(0, 1, n_points)
                candidate = candidate * (n_points / 2) / sum(candidate)
            elif np.random.rand() < 0.5:
                # Guided mutation from current best
                direction = np.random.uniform(-0.2, 0.2, n_points)
                candidate = np.clip(best_solution + direction, 0, 1)
                candidate = candidate * (n_points / 2) / sum(candidate)
            else:
                # Smooth initial guess
                candidate = guided_initial_guess(best_solution)
            samples.append(candidate)
        return samples

    # Main optimization loop
    best_solution = h_norm.copy()
    best_c5 = compute_overlap(best_solution)
    max_iter = 50
    sample_size = 30
    diversity_factor = 0.7
    cooling_rate = 0.98
    perturbation_decay = 0.95
    perturbation_max = 0.5

    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break

        # Generate diverse candidates
        candidates = generate_samples(sample_size)
        current_c5 = [compute_overlap(c) for c in candidates]

        # Select top candidates based on diversity and overlap
        top_indices = np.argsort([current_c5[i] for i in range(sample_size)])[:int(sample_size * diversity_factor)]
        top_candidates = [candidates[i] for i in top_indices]

        # Evaluate and select the best candidate
        for candidate in top_candidates:
            optimized_candidate = local_optimize(candidate)
            current_c5_opt = compute_overlap(optimized_candidate)
            if current_c5_opt < best_c5:
                best_c5 = current_c5_opt
                best_solution = optimized_candidate

        # Update perturbation parameters
        perturbation_max *= perturbation_decay
        if np.random.rand() < 0.3:
            # Occasionally reset perturbation range
            perturbation_max = 0.5
        if np.random.rand() < 0.1:
            # Occasionally use random sample
            best_solution = random_sample()

        # Apply guided mutation to current best
        direction = np.random.uniform(-0.1, 0.1, n_points)
        best_solution = np.clip(best_solution + direction, 0, 1)
        best_solution = best_solution * (n_points / 2) / sum(best_solution)

    return (best_solution.tolist(), best_c5, n_points)
