def run(seed=42, budget_s=100, **kwargs):

    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)

        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)

    bounds = Bounds(0, 1)

    # Use a global optimization algorithm to refine the initial guess
    # Use differential evolution with a small population size for speed
    
    def objective(x):
        return compute_overlap(x)

    result = optimize.differential_evolution(

        bounds,
        constraints=cons,
        seed=seed,
        maxiter=100,
