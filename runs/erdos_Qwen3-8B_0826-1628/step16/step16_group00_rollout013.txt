```python
import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds, LinearConstraint

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    h_norm = h * (n_points / 2) / sum(h)
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Define bounds and constraints
    bounds = Bounds(0, 1)
    cons = LinearConstraint(np.ones(n_points), lb=[n_points / 2], ub=[n_points / 2])
    
    # Use a more efficient optimization strategy with guided mutation
    swarm_size = 30
    max_iterations = 200
    perturbation = 0.3
    perturbation_decay = 0.98
    convergence_threshold = 1e-4
    best_c5 = float('inf')
    best_h = h_norm.copy()
    
    # Initialize swarm
    swarm = [h_norm.copy() for _ in range(swarm_size)]
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate current swarm
        for i in range(swarm_size):
            candidate = swarm[i].copy()
            # Add random perturbation
            perturb = np.random.uniform(-perturbation, perturbation, size=n_points)
            candidate += perturb
            # Clip to [0,1]
            candidate = np.clip(candidate, 0, 1)
            # Normalize to maintain integral constraint
            candidate = candidate * (n_points / 2) / sum(candidate)
            # Local refinement with SLSQP
            result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
            if result.success:
                swarm[i] = result.x
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        
        # Check for convergence
        if abs(best_c5 - evaluate_erdos_solution(best_h)) < convergence_threshold:
            break
        
        # Update perturbation
        perturbation *= perturbation_decay
    
    return (best_h.tolist(), best_c5, n_points)
```
