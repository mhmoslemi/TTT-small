import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds, LinearConstraint

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    max_iter = 200
    n_swarm = 30
    n_elite = 5
    perturbation_decay = 0.99
    initial_perturbation = 0.5
    
    # Initialize swarm
    swarm = [h_norm.copy() for _ in range(n_swarm)]
    best_solution = h_norm.copy()
    best_c5 = compute_overlap(best_solution)
    best_c5_history = [best_c5]
    
    # Function to evolve swarm with a balance of exploration and exploitation
    def evolve_swarm(swarm, best_solution, best_c5, initial_perturbation, perturbation_decay, max_iter, n_points, dx, compute_overlap, integral_constraint):
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            
            # Evaluate all candidates
            evaluations = []
            for i in range(len(swarm)):
                candidate = swarm[i].copy()
                # Random perturbation
                perturbation = np.random.uniform(-initial_perturbation, initial_perturbation, size=n_points)
                candidate += perturbation
                candidate = np.clip(candidate, 0, 1)
                # Enforce integral constraint
                if sum(candidate) == 0:
                    candidate = np.random.uniform(0, 1, n_points)
                else:
                    candidate = candidate * (n_points / 2) / sum(candidate)
                # Evaluate overlap
                try:
                    current_c5 = compute_overlap(candidate)
                except:
                    current_c5 = float('inf')
                evaluations.append((candidate, current_c5))
            
            # Sort by overlap
            evaluations.sort(key=lambda x: x[1])
            # Update best solution
            if evaluations[0][1] < best_c5:
                best_c5 = evaluations[0][1]
                best_solution = evaluations[0][0].copy()
                best_c5_history.append(best_c5)
            
            # Update swarm using elitism and local search
            for i in range(len(swarm)):
                # Elitism: replace worst with best
                if i >= len(swarm) - n_elite:
                    swarm[i] = evaluations[0][0].copy()
                else:
                    # Local search with perturbation
                    candidate = swarm[i].copy()
                    perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
                    candidate += perturbation
                    candidate = np.clip(candidate, 0, 1)
                    # Enforce integral constraint
                    if sum(candidate) == 0:
                        candidate = np.random.uniform(0, 1, n_points)
                    else:
                        candidate = candidate * (n_points / 2) / sum(candidate)
                    # Evaluate
                    try:
                        current_c5 = compute_overlap(candidate)
                    except:
                        current_c5 = float('inf')
                    if current_c5 < evaluations[i][1]:
                        swarm[i] = candidate.copy()
            
            # Reduce perturbation range
            initial_perturbation *= perturbation_decay
        
        return best_solution, best_c5
    
    # Run the optimization
    best_solution, best_c5 = evolve_swarm(swarm, best_solution, best_c5, initial_perturbation, perturbation_decay, max_iter, n_points, dx, compute_overlap, integral_constraint)
    
    return (best_solution.tolist(), best_c5, n_points)
