```python
import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds, LinearConstraint

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    max_iter = 200
    perturbation_decay = 0.98
    initial_perturbation = 0.3
    n_particles = 50
    inertia = 0.8
    cognitive_weight = 1.5
    social_weight = 1.5
    
    # Initialize a swarm of candidate solutions
    swarm = [h_norm.copy() for _ in range(n_particles)]
    velocities = np.random.uniform(-initial_perturbation, initial_perturbation, size=(n_particles, n_points))
    best_solution = h_norm.copy()
    best_c5 = compute_overlap(best_solution)
    best_c5_history = [best_c5]
    
    # Function to update velocities and positions
    def update_velocity(velocity, personal_best, global_best, inertia, cognitive_weight, social_weight):
        r1 = np.random.rand(n_points)
        r2 = np.random.rand(n_points)
        cognitive = cognitive_weight * r1 * (personal_best - velocity)
        social = social_weight * r2 * (global_best - velocity)
        return inertia * velocity + cognitive + social
    
    # Run the optimization
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Update velocities and positions
        for i in range(n_particles):
            velocity = velocities[i]
            personal_best = swarm[i]
            best_solution_current = best_solution
            velocity = update_velocity(velocity, personal_best, best_solution_current, inertia, cognitive_weight, social_weight)
            velocities[i] = velocity
            candidate = np.clip(swarm[i] + velocity, 0, 1)
            # Ensure integral constraint is satisfied
            if sum(candidate) == 0:
                candidate = np.random.uniform(0, 1, n_points)
            else:
                candidate = candidate * (n_points / 2) / sum(candidate)
            swarm[i] = candidate
        
        # Evaluate all particles and update best solution
        for i in range(n_particles):
            current_c5 = compute_overlap(swarm[i])
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_solution = swarm[i].copy()
                best_c5_history.append(best_c5)
        
        # Gradually reduce perturbation range
        initial_perturbation *= perturbation_decay
    
    return (best_solution.tolist(), best_c5, n_points)
```
