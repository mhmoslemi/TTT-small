import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds, LinearConstraint

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    population_size = 50
    generations = 150
    mutation_rate = 0.15
    crossover_rate = 0.8
    elite_size = 5
    mutation_decay = 0.98
    initial_mutation_scale = 0.3
    
    # Initialize a population of candidate solutions
    population = [h_norm.copy() for _ in range(population_size)]
    best_solution = h_norm.copy()
    best_c5 = compute_overlap(best_solution)
    
    # Function to evolve population with adaptive mutation and multi-phase refinement
    def evolve_population(population, best_solution, best_c5, initial_mutation_scale, mutation_decay, generations, n_points, dx, compute_overlap, integral_constraint):
        for generation in range(generations):
            if time.time() - start_time > budget_s:
                break
            
            # Evaluate fitness of current population
            fitness = [compute_overlap(ind) for ind in population]
            
            # Select elites
            elite_indices = np.argsort(fitness)[:elite_size]
            elites = [population[i] for i in elite_indices]
            best_c5 = min(best_c5, min(fitness))
            best_solution = population[np.argmin(fitness)].copy()
            
            # Generate offspring through crossover and mutation
            offspring = []
            for i in range(population_size - elite_size):
                # Select parents
                parent1 = population[np.random.choice(elite_indices)]
                parent2 = population[np.random.choice(population_size)]
                
                # Crossover
                if np.random.rand() < crossover_rate:
                    crossover_point = np.random.randint(1, n_points - 1)
                    child = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
                else:
                    child = parent1.copy()
                
                # Mutation
                mutation_scale = initial_mutation_scale * (mutation_decay ** generation)
                mutation = np.random.normal(0, mutation_scale, size=n_points)
                child += mutation
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / sum(child)
                
                offspring.append(child)
            
            # Combine elites and offspring
            population = elites + offspring
            
            # Local refinement with SLSQP for top candidates
            top_candidates = np.argsort(fitness)[:max(5, int(population_size * 0.1))]
            for i in top_candidates:
                candidate = population[i].copy()
                bounds = Bounds(0, 1)
                cons = LinearConstraint(np.ones(n_points), lb=[n_points / 2], ub=[n_points / 2])
                result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
                
                if result.success:
                    population[i] = result.x
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_solution = result.x
            
        return best_solution, best_c5
    
    # Run the optimization
    best_solution, best_c5 = evolve_population(population, best_solution, best_c5, initial_mutation_scale, mutation_decay, generations, n_points, dx, compute_overlap, integral_constraint)
    
    return (best_solution.tolist(), best_c5, n_points)
