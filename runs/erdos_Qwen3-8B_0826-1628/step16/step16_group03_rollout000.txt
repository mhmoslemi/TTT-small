import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds, LinearConstraint

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    n_swarm = 20
    max_iter = 150
    perturbation_range = 0.3
    cooling_rate = 0.99
    elite_ratio = 0.2
    
    # Initialize a swarm of candidate solutions
    swarm = [h_norm.copy() for _ in range(n_swarm)]
    best_solution = h_norm.copy()
    best_c5 = compute_overlap(best_solution)
    best_c5_history = [best_c5]
    
    # Function to evolve swarm with a balance of exploration and exploitation
    def evolve_swarm(swarm, best_solution, best_c5, perturbation_range, cooling_rate, elite_ratio, max_iter, n_points, dx, compute_overlap, integral_constraint):
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            
            # Evaluate all members of the swarm
            swarm_fitness = []
            for member in swarm:
                fitness = compute_overlap(member)
                swarm_fitness.append((fitness, member))
            
            # Select elite members
            sorted_swarm = sorted(swarm_fitness, key=lambda x: x[0])
            elite_count = int(elite_ratio * n_swarm)
            elite_members = [member for (fitness, member) in sorted_swarm[:elite_count]]
            
            # Perturb and refine non-elite members
            for i in range(n_swarm):
                if i < elite_count:
                    continue  # Elite members are not perturbed
                member = swarm[i].copy()
                # Perturb
                perturbation = np.random.uniform(-perturbation_range, perturbation_range, size=n_points)
                candidate = np.clip(member + perturbation, 0, 1)
                # Ensure integral constraint is satisfied
                if sum(candidate) < 1e-6:
                    candidate = np.random.uniform(0, 1, n_points)
                else:
                    candidate = candidate * (n_points / 2) / sum(candidate)
                # Local refinement with SLSQP
                bounds = Bounds(0, 1)
                cons = LinearConstraint(np.ones(n_points), lb=[n_points / 2], ub=[n_points / 2])
                result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
                
                # Update the swarm member if the optimization succeeded
                if result.success:
                    swarm[i] = result.x
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_solution = result.x
                        best_c5_history.append(best_c5)
            
            # Replace worst members with elite members
            worst_idx = np.argsort(swarm_fitness, key=lambda x: x[0])[-int((1 - elite_ratio) * n_swarm):]
            for idx in worst_idx:
                swarm[idx] = elite_members[np.random.randint(len(elite_members))]
            
            # Cool down the perturbation range
            perturbation_range *= cooling_rate
        
        return best_solution, best_c5
    
    # Run the optimization
    best_solution, best_c5 = evolve_swarm(swarm, best_solution, best_c5, perturbation_range, cooling_rate, elite_ratio, max_iter, n_points, dx, compute_overlap, integral_constraint)
    
    return (best_solution.tolist(), best_c5, n_points)
