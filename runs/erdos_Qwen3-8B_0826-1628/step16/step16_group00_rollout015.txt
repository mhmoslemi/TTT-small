import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds, LinearConstraint

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with a refined initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Use a different initial solution with more structured patterns
    # Create a step function with 4 intervals
    structured_h = np.zeros(n_points)
    interval = n_points // 4
    for i in range(n_points):
        if i < interval:
            structured_h[i] = 0.5
        elif i < 2 * interval:
            structured_h[i] = 0.75
        elif i < 3 * interval:
            structured_h[i] = 0.25
        else:
            structured_h[i] = 0.0
    structured_h = structured_h * (n_points / 2) / sum(structured_h)
    h = structured_h.copy()
    
    # Define bounds and constraints
    bounds = Bounds(0, 1)
    cons = LinearConstraint(np.ones(n_points), lb=[n_points / 2], ub=[n_points / 2])
    
    # Optimization parameters
    max_iter = 100
    perturbation = 0.3
    decay_rate = 0.99
    
    # Use a hybrid approach: random search with gradient-based refinement
    best_solution = h.copy()
    best_c5 = compute_overlap(best_solution)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Generate multiple random perturbations
        for _ in range(10):
            perturbation_vec = np.random.uniform(-perturbation, perturbation, size=n_points)
            candidate = np.clip(h + perturbation_vec, 0, 1)
            candidate = candidate * (n_points / 2) / sum(candidate)
            
            # Local optimization with SLSQP
            result = optimize.minimize(lambda x: compute_overlap(x), candidate, 
                                     bounds=bounds, constraints=cons, method='SLSQP')
            
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_solution = result.x
        
        # Gradually reduce perturbation size
        perturbation *= decay_rate
    
    return (best_solution.tolist(), best_c5, n_points)
