import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = sum(h)
    h_norm = h * (n_points / 2) / sum_h

    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    population_size = 20
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    elite_size = 2
    mutation_decay = 0.98
    crossover_decay = 0.98
    
    # Define bounds for h_values
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initialize population
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    # Scale to satisfy integral constraint
    population = [h * (n_points / 2) / sum(h) for h in population]
    
    # Evaluate initial population
    fitness = [compute_overlap(ind) for ind in population]
    best_solution = population[np.argmin(fitness)]
    best_c5 = min(fitness)
    
    # Genetic algorithm optimization
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [compute_overlap(ind) for ind in population]
        
        # Select elites
        elite_indices = np.argsort(fitness)[:elite_size]
        elites = [population[i] for i in elite_indices]
        
        # Generate offspring
        offspring = []
        for _ in range(population_size - elite_size):
            # Select parents
            parent1 = elites[np.random.choice(len(elites))]
            parent2 = elites[np.random.choice(len(elites))]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
            else:
                child = np.random.uniform(0, 1, n_points)
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutate_indices = np.random.choice(n_points, int(mutation_rate * n_points), replace=False)
                child[mutate_indices] = np.random.uniform(0, 1, len(mutate_indices))
            
            # Scale to satisfy integral constraint
            child = child * (n_points / 2) / sum(child)
            offspring.append(child)
        
        # Combine elites and offspring
        population = elites + offspring
        
        # Update best solution
        current_best_index = np.argmin(fitness)
        if fitness[current_best_index] < best_c5:
            best_c5 = fitness[current_best_index]
            best_solution = population[current_best_index]
        
        # Decay mutation and crossover rates
        mutation_rate *= mutation_decay
        crossover_rate *= crossover_decay
    
    return (best_solution.tolist(), best_c5, n_points)
