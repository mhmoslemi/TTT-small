import numpy as np
import scipy.optimize as optimize
import time
import random
import copy

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the provided initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Define the integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    max_iter = 100
    refine_steps = 10
    refine_decay = 0.95
    perturbation_decay = 0.98
    initial_perturbation = 0.2
    
    # Initialize a swarm of candidate solutions
    swarm_size = 20
    swarm = [h.copy() for _ in range(swarm_size)]
    best_solution = h.copy()
    best_c5 = compute_overlap(best_solution)
    
    # Function to evolve swarm with a balance of exploration and exploitation
    def evolve_swarm():
        nonlocal best_solution, best_c5
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            
            # Global exploration with random perturbations
            for i in range(swarm_size):
                # Generate perturbation vector
                perturbation = np.random.uniform(-initial_perturbation, initial_perturbation, size=n_points)
                candidate = np.clip(swarm[i] + perturbation, 0, 1)
                # Ensure integral constraint
                candidate = candidate * (n_points / 2) / np.sum(candidate)
                # Optimize locally with SLSQP
                bounds = [(0, 1) for _ in range(n_points)]
                cons = [{'type': 'eq', 'fun': integral_constraint}]
                result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
                if result.success:
                    swarm[i] = result.x
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_solution = result.x
            
            # Reduce perturbation range and perform refinement
            initial_perturbation *= perturbation_decay
            
            # Local refinement with multi-step optimization
            for i in range(swarm_size):
                candidate = swarm[i].copy()
                for _ in range(refine_steps):
                    perturbation = np.random.uniform(-0.01, 0.01, size=n_points)
                    candidate = np.clip(candidate + perturbation, 0, 1)
                    # Ensure integral constraint
                    candidate = candidate * (n_points / 2) / np.sum(candidate)
                    result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
                    if result.success:
                        candidate = result.x
                        current_c5 = result.fun
                        if current_c5 < best_c5:
                            best_c5 = current_c5
                            best_solution = candidate
                swarm[i] = candidate
        
        return best_solution, best_c5
    
    # Run the optimization
    best_solution, best_c5 = evolve_swarm()
    
    return (best_solution.tolist(), best_c5, n_points)
