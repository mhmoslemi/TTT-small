import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds, LinearConstraint
from scipy import sparse

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return np.max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    max_iter = 100
    temp_start = 1.0
    temp_end = 0.01
    perturbation_factor = 0.4
    
    # Hybrid optimization approach with global perturbation, local optimization, and gradient descent
    
    # Initial best solution
    best_solution = h_norm.copy()
    best_c5 = compute_overlap(best_solution)
    
    # Use a swarm approach with dynamic perturbation and local optimization
    swarm_size = 20
    swarm = [h_norm.copy() for _ in range(swarm_size)]
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Compute current temperature
        temp = temp_start * (temp_end / temp_start) ** (iteration / max_iter)
        
        # Evaluate swarm and update best solution
        for i in range(len(swarm)):
            # Apply random perturbation
            perturbation = np.random.uniform(-perturbation_factor, perturbation_factor, size=n_points)
            candidate = np.clip(swarm[i] + perturbation, 0, 1)
            
            # Local optimization with constraints
            bounds = Bounds([0.0]*n_points, [1.0]*n_points)
            cons = LinearConstraint([[1.0]*n_points], [n_points/2], [n_points/2])
            result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
            
            # Accept the solution with probability based on temperature
            if result.success:
                current_c5 = result.fun
                delta = current_c5 - best_c5
                if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                    swarm[i] = result.x
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_solution = result.x
        
        # Reduce perturbation size
        perturbation_factor *= 0.95
    
    return (best_solution.tolist(), best_c5, n_points)
