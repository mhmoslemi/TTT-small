import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = sum(h)
    h_norm = h * (n_points / 2) / sum_h

    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    swarm_size = 30
    max_iter = 100
    perturbation_decay = 0.95
    initial_perturbation = 0.2
    mutation_rate = 0.1
    crossover_rate = 0.7
    elite_ratio = 0.2
    local_refinement_steps = 10
    
    # Initialize a swarm of candidate solutions
    swarm = [h_norm.copy() for _ in range(swarm_size)]
    best_solution = h_norm.copy()
    best_c5 = compute_overlap(best_solution)
    
    # Function to evolve swarm with a balance of exploration and exploitation
    def evolve_swarm(swarm, best_solution, best_c5, initial_perturbation, perturbation_decay, max_iter, n_points, dx, compute_overlap, integral_constraint):
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            
            # Create a new population through genetic operations
            new_swarm = []
            for i in range(swarm_size):
                # Select two random parents
                parent1 = np.random.choice(swarm)
                parent2 = np.random.choice(swarm)
                
                # Crossover
                child = np.copy(parent1)
                if np.random.rand() < crossover_rate:
                    child = (parent1 * np.random.rand(n_points) + parent2 * (1 - np.random.rand(n_points)))
                
                # Mutation
                if np.random.rand() < mutation_rate:
                    mutation = np.random.uniform(-initial_perturbation, initial_perturbation, size=n_points)
                    child += mutation
                    child = np.clip(child, 0, 1)
                
                # Ensure integral constraint is satisfied
                child = child * (n_points / 2) / sum(child)
                
                # Add to new population
                new_swarm.append(child)
            
            # Evaluate new population and select elites
            new_swarm = np.array(new_swarm)
            fitness = np.array([compute_overlap(solution) for solution in new_swarm])
            elite_indices = np.argsort(fitness)[:int(elite_ratio * swarm_size)]
            swarm = new_swarm[elite_indices]
            
            # Local refinement with SLSQP for top performers
            for i in range(min(10, len(swarm))):
                candidate = swarm[i]
                bounds = [(0, 1) for _ in range(n_points)]
                cons = ({'type': 'eq', 'fun': integral_constraint})
                try:
                    result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-9, 'maxiter': local_refinement_steps})
                except:
                    result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='L-BFGS-B', options={'ftol': 1e-9, 'maxiter': local_refinement_steps})
                
                if result.success:
                    swarm[i] = result.x
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_solution = result.x
            
            # Gradually reduce perturbation range
            initial_perturbation *= perturbation_decay
            
        return best_solution, best_c5
    
    # Run the optimization
    best_solution, best_c5 = evolve_swarm(swarm, best_solution, best_c5, initial_perturbation, perturbation_decay, max_iter, n_points, dx, compute_overlap, integral_constraint)
    
    return (best_solution.tolist(), best_c5, n_points)
