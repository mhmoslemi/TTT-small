import numpy as np
import scipy.optimize as optimize
import time
import random
import copy

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the provided initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    # Ensure the integral constraint is met
    h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Define the integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    num_iterations = 100
    population_size = 20
    mutation_rate = 0.1
    mutation_decay = 0.98
    
    # Initialize a population of candidate solutions
    population = [h.copy() for _ in range(population_size)]
    best_solution = h.copy()
    best_c5 = compute_overlap(best_solution)
    
    # Genetic algorithm with SLSQP refinement
    for iteration in range(num_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate the population
        fitness = [compute_overlap(ind) for ind in population]
        best_index = np.argmin(fitness)
        best_c5 = fitness[best_index]
        best_solution = population[best_index].copy()
        
        # Create new population
        new_population = []
        
        # Elitism: keep the best solution
        new_population.append(best_solution)
        
        # Crossover and mutation
        for _ in range(population_size - 1):
            # Select two parents
            parent1, parent2 = np.random.choice(population, size=2, replace=False)
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
            # Mutation
            mutation = np.random.uniform(-mutation_rate, mutation_rate, size=n_points)
            child += mutation
            # Clip values to [0, 1]
            child = np.clip(child, 0, 1)
            # Adjust to satisfy the integral constraint
            child = child * (n_points / 2) / np.sum(child)
            new_population.append(child)
        
        # Replace the population
        population = new_population
        
        # Local refinement with SLSQP on the best solution
        if iteration % 5 == 0:
            # Refine the best solution
            bounds = [(0, 1) for _ in range(n_points)]
            cons = [{'type': 'eq', 'fun': integral_constraint}]
            result = optimize.minimize(lambda x: compute_overlap(x), best_solution, bounds=bounds, constraints=cons, method='SLSQP')
            if result.success:
                best_solution = result.x
                best_c5 = result.fun
        
        # Reduce mutation rate
        mutation_rate *= mutation_decay
    
    return (best_solution.tolist(), best_c5, n_points)
