import numpy as np
import scipy.optimize as optimize
import time
import random
import copy

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the provided initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Define the integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    initial_perturbation = 0.2
    perturbation_decay = 0.95
    max_iter = 150
    refine_steps = 5
    refine_decay = 0.9
    swarm_size = 30
    mutation_rate = 0.15
    elite_size = 5
    
    # Initialize a swarm of candidate solutions
    swarm = [h.copy() for _ in range(swarm_size)]
    best_solution = h.copy()
    best_c5 = compute_overlap(best_solution)
    
    # Function to evolve swarm with a balance of exploration and exploitation
    def evolve_swarm(swarm, best_solution, best_c5, initial_perturbation, perturbation_decay, max_iter, n_points, dx, compute_overlap, integral_constraint):
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            
            # Evaluate all candidates
            evaluated_swarm = []
            for i in range(swarm_size):
                candidate = swarm[i].copy()
                # Ensure integral constraint
                candidate = candidate * (n_points / 2) / np.sum(candidate)
                # Evaluate overlap
                current_c5 = compute_overlap(candidate)
                evaluated_swarm.append((current_c5, candidate))
            
            # Sort evaluated swarm
            evaluated_swarm.sort()
            # Update best solution
            if evaluated_swarm[0][0] < best_c5:
                best_c5 = evaluated_swarm[0][0]
                best_solution = evaluated_swarm[0][1]
            
            # Select elites
            elites = evaluated_swarm[:elite_size]
            elite_candidates = [x[1] for x in elites]
            
            # Generate new candidates
            new_swarm = []
            for i in range(swarm_size):
                # Global exploration
                if np.random.rand() < 0.3:
                    perturbation = np.random.uniform(-initial_perturbation, initial_perturbation, size=n_points)
                    candidate = np.clip(swarm[i] + perturbation, 0, 1)
                # Local refinement
                else:
                    candidate = swarm[i].copy()
                
                # Ensure integral constraint
                candidate = candidate * (n_points / 2) / np.sum(candidate)
                # Evaluate overlap
                current_c5 = compute_overlap(candidate)
                new_swarm.append((current_c5, candidate))
            
            # Add elites to new swarm
            new_swarm += elites
            
            # Sort new_swarm
            new_swarm.sort()
            # Update swarm
            swarm = [x[1] for x in new_swarm]
            
            # Reduce perturbation range
            initial_perturbation *= perturbation_decay
        
        return best_solution, best_c5
    
    # Run the optimization
    best_solution, best_c5 = evolve_swarm(swarm, best_solution, best_c5, initial_perturbation, perturbation_decay, max_iter, n_points, dx, compute_overlap, integral_constraint)
    
    return (best_solution.tolist(), best_c5, n_points)
