import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Function to compute the overlap integral
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Function to compute the integral constraint violation
    def integral_violation(h_values):
        return abs(sum(h_values) - (n_points / 2))
    
    # Initialize with the initial solution (n=34)
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    h = h * (n_points / 2) / sum(h)
    
    # Best solution so far
    best_solution = h.copy()
    best_c5 = compute_overlap(best_solution)
    
    # Particle Swarm Optimization parameters
    n_particles = 20
    max_iter = 150
    inertia = 0.8
    cognitive = 1.2
    social = 1.2
    perturb_range = 0.2
    convergence_threshold = 1e-4
    
    # Initialize swarm
    swarm = [h.copy() for _ in range(n_particles)]
    velocities = np.random.uniform(-perturb_range, perturb_range, size=(n_particles, n_points))
    
    # Personal best and global best
    personal_best = [s.copy() for s in swarm]
    global_best = best_solution.copy()
    
    # Optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        for i in range(n_particles):
            # Update velocity and position
            r1, r2 = np.random.rand(2)
            velocities[i] = inertia * velocities[i] + cognitive * r1 * (personal_best[i] - swarm[i]) + social * r2 * (global_best - swarm[i])
            swarm[i] = np.clip(swarm[i] + velocities[i], 0, 1)
            
            # Enforce integral constraint
            swarm[i] = swarm[i] * (n_points / 2) / sum(swarm[i])
            
            # Compute overlap
            curr_c5 = compute_overlap(swarm[i])
            
            # Update personal best
            if curr_c5 < compute_overlap(personal_best[i]):
                personal_best[i] = swarm[i].copy()
            
            # Update global best
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_solution = swarm[i].copy()
        
        # Early stopping if convergence
        if best_c5 < convergence_threshold:
            break
    
    # Additional local refinement using SLSQP
    def local_refine(h_values):
        bounds = [(0, 1) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': integral_constraint})
        result = optimize.minimize(lambda x: compute_overlap(x), h_values, bounds=bounds, constraints=cons, method='SLSQP')
        if result.success:
            return result.x
        return h_values
    
    best_solution = local_refine(best_solution)
    best_c5 = compute_overlap(best_solution)
    
    return (best_solution.tolist(), best_c5, n_points)
