import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds, LinearConstraint

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return np.max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    max_iter = 150
    temp_start = 1.0
    temp_end = 0.001
    perturbation_factor = 0.3
    cooling_factor = 0.995
    population_size = 30
    mutation_rate = 0.15
    mutation_strength = 0.2
    
    # Initialize a population of candidate solutions
    population = [h_norm.copy() for _ in range(population_size)]
    best_solution = h_norm.copy()
    best_c5 = compute_overlap(best_solution)
    
    # Use a hybrid approach: global perturbation and local optimization
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Compute current temperature
        temp = temp_start * (temp_end / temp_start) ** (iteration / max_iter)
        
        # Evaluate all solutions in the population
        fitness = [compute_overlap(sol) for sol in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_solution = population[best_idx].copy()
        
        # Create new population with mutations and recombination
        new_population = []
        for i in range(population_size):
            # Apply random mutation
            if np.random.rand() < mutation_rate:
                mutant = population[i].copy()
                perturbation = np.random.uniform(-mutation_strength, mutation_strength, size=n_points)
                mutant += perturbation
                mutant = np.clip(mutant, 0, 1)
                new_population.append(mutant)
            else:
                # Recombine with another solution
                j = np.random.randint(0, population_size)
                parent1 = population[i]
                parent2 = population[j]
                child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
                new_population.append(child)
        
        # Local optimization for the best solutions
        for i in range(len(new_population)):
            if np.random.rand() < 0.3:
                # Perform local optimization with constraints
                bounds = Bounds([0.0]*n_points, [1.0]*n_points)
                cons = LinearConstraint([[1.0]*n_points], [n_points/2], [n_points/2])
                result = optimize.minimize(lambda x: compute_overlap(x), new_population[i], bounds=bounds, constraints=cons, method='SLSQP')
                if result.success:
                    new_population[i] = result.x
        
        # Replace the population with the new population
        population = new_population
        
        # Reduce perturbation size
        perturbation_factor *= cooling_factor
    
    return (best_solution.tolist(), best_c5, n_points)
