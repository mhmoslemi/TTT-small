import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Define initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    max_iter = 100
    pop_size = 20
    elite_size = 5
    mutation_rate = 0.1
    mutation_strength = 0.1
    crossover_rate = 0.8
    
    # Initialize population
    population = [np.random.uniform(0, 1, n_points) for _ in range(pop_size)]
    population = [h * (n_points / 2) / sum(h) for h in population]
    best_solution = population[0]
    best_c5 = compute_overlap(best_solution)
    
    # Function to evolve population
    def evolve_population(population, best_solution, best_c5, max_iter, n_points, dx, compute_overlap, integral_constraint):
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            
            # Evaluate fitness
            fitness = [compute_overlap(ind) for ind in population]
            best_idx = np.argmin(fitness)
            best_c5 = fitness[best_idx]
            best_solution = population[best_idx].copy()
            
            # Select elites
            elite_indices = np.argsort(fitness)[:elite_size]
            elites = [population[i] for i in elite_indices]
            
            # Generate offspring
            offspring = []
            for i in range(pop_size - elite_size):
                # Select parents
                parent1 = elites[np.random.choice(elite_size)]
                parent2 = elites[np.random.choice(elite_size)]
                
                # Crossover
                if np.random.rand() < crossover_rate:
                    child = np.zeros(n_points)
                    for j in range(n_points):
                        if np.random.rand() < 0.5:
                            child[j] = parent1[j]
                        else:
                            child[j] = parent2[j]
                else:
                    child = np.random.uniform(0, 1, n_points)
                
                # Mutation
                for j in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[j] += np.random.uniform(-mutation_strength, mutation_strength)
                        child[j] = np.clip(child[j], 0, 1)
                
                # Normalize to satisfy integral constraint
                child = child * (n_points / 2) / sum(child)
                offspring.append(child)
            
            # Create new population
            population = elites + offspring
            population = [np.clip(ind, 0, 1) for ind in population]
            
            # Check if best solution improved
            if best_c5 < compute_overlap(best_solution):
                best_solution = population[best_idx].copy()
        
        return best_solution, best_c5
    
    # Run the optimization
    best_solution, best_c5 = evolve_population(population, best_solution, best_c5, max_iter, n_points, dx, compute_overlap, integral_constraint)
    
    return (best_solution.tolist(), best_c5, n_points)
