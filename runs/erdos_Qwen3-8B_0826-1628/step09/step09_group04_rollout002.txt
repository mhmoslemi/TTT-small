import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds, LinearConstraint

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return np.max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    max_iter = 150
    initial_perturbation = 0.3
    temp_start = 1.0
    temp_end = 0.1
    
    # Define a function to evaluate the overlap for a given h
    def objective(h_values):
        return compute_overlap(h_values)
    
    # Define the integral constraint
    constraints = LinearConstraint([[1.0]*n_points], [n_points/2], [n_points/2])
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    
    # Initialize a good solution
    best_solution = h_norm.copy()
    best_c5 = compute_overlap(best_solution)
    
    # Run a local optimization with the initial solution
    result = optimize.minimize(objective, best_solution, bounds=bounds, constraints=constraints, method='SLSQP')
    best_solution = result.x
    best_c5 = result.fun
    
    # Try a few random perturbations of the initial solution
    for _ in range(5):
        perturbation = np.random.uniform(-0.2, 0.2, size=n_points)
        candidate = np.clip(best_solution + perturbation, 0, 1)
        result = optimize.minimize(objective, candidate, bounds=bounds, constraints=constraints, method='SLSQP')
        if result.success and result.fun < best_c5:
            best_solution = result.x
            best_c5 = result.fun
    
    # Run a simulated annealing-style optimization
    temp = 1.0
    cooling_rate = 0.995
    for iteration in range(100):
        if time.time() - start_time > budget_s:
            break
        # Generate a random perturbation
        perturbation = np.random.normal(0, 0.1, size=n_points)
        candidate = np.clip(best_solution + perturbation, 0, 1)
        # Enforce the integral constraint
        candidate = candidate * (n_points / 2) / np.sum(candidate)
        # Compute the overlap
        current_c5 = compute_overlap(candidate)
        # Accept the solution with probability based on temperature
        delta = current_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            best_solution = candidate
            best_c5 = current_c5
        temp *= cooling_rate
    
    return (best_solution.tolist(), best_c5, n_points)
