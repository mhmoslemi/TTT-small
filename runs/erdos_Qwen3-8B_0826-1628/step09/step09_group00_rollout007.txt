import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start from the initial construction
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    
    # Normalize to satisfy the integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Define the integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    max_iter = 100
    population_size = 30
    mutation_rate = 0.1
    crossover_rate = 0.8
    elite_size = 5
    refine_steps = 5
    refine_decay = 0.9
    
    # Initial population
    population = [h.copy() for _ in range(population_size)]
    population = [np.clip(ind, 0, 1) for ind in population]
    population = [ind * (n_points / 2) / np.sum(ind) for ind in population]
    
    best_solution = h.copy()
    best_c5 = compute_overlap(best_solution)
    
    # Function to evaluate and rank a population
    def evaluate_population(population, compute_overlap, integral_constraint):
        ranked = []
        for ind in population:
            # Check constraint satisfaction
            if np.abs(integral_constraint(ind)) > 1e-6:
                continue  # Skip invalid solutions
            overlap = compute_overlap(ind)
            ranked.append((overlap, ind))
        # Sort by overlap (smaller is better)
        ranked.sort()
        return ranked
    
    # Main evolutionary algorithm loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate the current population
        ranked = evaluate_population(population, compute_overlap, integral_constraint)
        
        # Select elite solutions
        elite = ranked[:elite_size]
        if not elite:
            break
        
        # Generate offspring
        offspring = []
        for _ in range(len(population) - elite_size):
            # Select parents
            parent1 = random.choice(elite)[1]
            parent2 = random.choice(elite)[1]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                child = np.zeros(n_points)
                for i in range(n_points):
                    if np.random.rand() < 0.5:
                        child[i] = parent1[i]
                    else:
                        child[i] = parent2[i]
            else:
                child = np.random.rand(n_points)
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] = np.clip(child[i] + np.random.normal(0, 0.05), 0, 1)
            
            # Normalize
            child = child * (n_points / 2) / np.sum(child)
            offspring.append(child)
        
        # Combine elite and offspring
        new_population = [ind for overlap, ind in elite]
        new_population.extend(offspring)
        
        # Select top solutions for next iteration
        new_population.sort(key=lambda x: compute_overlap(x))
        population = new_population[:population_size]
        population = [np.clip(ind, 0, 1) for ind in population]
        population = [ind * (n_points / 2) / np.sum(ind) for ind in population]
        
        # Update best solution
        for overlap, ind in ranked:
            if overlap < best_c5:
                best_c5 = overlap
                best_solution = ind
    
    # Refine the best solution with local optimization
    if best_c5 is not None:
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': integral_constraint}]
        result = optimize.minimize(lambda x: compute_overlap(x), best_solution, bounds=bounds, constraints=cons, method='SLSQP')
        if result.success:
            best_solution = result.x
            best_c5 = result.fun
    
    return (best_solution.tolist(), best_c5, n_points)
