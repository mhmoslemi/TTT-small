import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Base parameters
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    total_integral = n_points / 2
    
    # Function to compute overlap
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - total_integral
    
    # Use the initial solution as a starting point
    h = initial_h_values.copy()
    h /= sum(h) * dx
    h = h * total_integral / sum(h)
    
    # Optimization settings
    max_iter = 100
    n_swarm = 20
    n_pop = 50
    n_elite = 5
    mutation_rate = 0.1
    crossover_rate = 0.8
    mutation_strength = 0.15
    
    # Initialize population
    def initialize_population(size):
        population = []
        for _ in range(size):
            h = np.random.uniform(0, 1, n_points)
            h = h * total_integral / sum(h)
            population.append(h)
        return np.array(population)
    
    # Fitness function
    def fitness(x):
        return compute_overlap(x)
    
    # Crossover
    def crossover(parent1, parent2):
        child = np.copy(parent1)
        for i in range(n_points):
            if np.random.rand() < crossover_rate:
                child[i] = parent2[i]
        return child
    
    # Mutation
    def mutate(individual):
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                individual[i] += np.random.uniform(-mutation_strength, mutation_strength)
                individual[i] = np.clip(individual[i], 0, 1)
        individual = individual * total_integral / sum(individual)
        return individual
    
    # Evaluate population
    def evaluate(pop):
        fitnesses = np.array([fitness(ind) for ind in pop])
        return fitnesses
    
    # Selection
    def select(pop, fitnesses):
        indices = np.argsort(fitnesses)
        return pop[indices[:n_elite]]
    
    # Genetic algorithm
    pop = initialize_population(n_pop)
    best_fitness = float('inf')
    best_solution = np.copy(pop[0])
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate population
        fitnesses = evaluate(pop)
        
        # Track best solution
        best_idx = np.argmin(fitnesses)
        if fitnesses[best_idx] < best_fitness:
            best_fitness = fitnesses[best_idx]
            best_solution = np.copy(pop[best_idx])
        
        # Select elites
        elites = select(pop, fitnesses)
        
        # Generate new population
        new_pop = []
        for _ in range(n_pop - n_elite):
            # Choose two parents
            parent1, parent2 = np.random.choice(elites, size=2, replace=False)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_pop.append(child)
        
        # Replace old population
        pop = np.vstack([elites, new_pop])
    
    return (best_solution.tolist(), best_fitness, n_points)
