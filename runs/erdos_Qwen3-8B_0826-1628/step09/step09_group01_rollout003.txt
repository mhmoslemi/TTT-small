import numpy as np
import scipy.optimize as optimize
import time
import random
from functools import partial

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    h = h * (n_points / 2) / sum(h)
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * (2.0 / n_points)
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': integral_constraint})
    
    # Optimization parameters
    max_iter = 100
    elite_size = 10
    mutation_rate = 0.1
    crossover_rate = 0.8
    mutation_strength = 0.2
    elite_perturbation = 0.05
    
    # Initialize population
    population = [np.random.uniform(0, 1, n_points) for _ in range(elite_size)]
    population = [h * (n_points / 2) / sum(h) for h in population]
    
    # Evaluate initial population
    fitness = [compute_overlap(h) for h in population]
    best_index = np.argmin(fitness)
    best_solution = population[best_index].copy()
    best_c5 = fitness[best_index]
    
    # Evolutionary algorithm
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Selection
        population = np.array(population)
        fitness = np.array(fitness)
        indices = np.argsort(fitness)[:elite_size]
        population = population[indices]
        fitness = fitness[indices]
        
        # Crossover
        new_population = []
        for _ in range(elite_size):
            parent1 = population[np.random.randint(elite_size)]
            parent2 = population[np.random.randint(elite_size)]
            child = np.zeros(n_points)
            for i in range(n_points):
                if np.random.rand() < crossover_rate:
                    child[i] = parent1[i]
                else:
                    child[i] = parent2[i]
            new_population.append(child)
        
        # Mutation
        for i in range(elite_size):
            child = new_population[i]
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[j] += np.random.uniform(-mutation_strength, mutation_strength)
                    child[j] = np.clip(child[j], 0, 1)
            new_population[i] = child
        
        # Ensure integral constraint
        for i in range(elite_size):
            new_population[i] = new_population[i] * (n_points / 2) / sum(new_population[i])
        
        # Evaluate new population
        new_fitness = [compute_overlap(h) for h in new_population]
        fitness = np.concatenate((fitness, new_fitness))
        population = np.concatenate((population, new_population))
        population = population[:elite_size * 2]
        fitness = fitness[:elite_size * 2]
        
        # Update best solution
        best_index = np.argmin(fitness)
        if fitness[best_index] < best_c5:
            best_c5 = fitness[best_index]
            best_solution = population[best_index].copy()
        
        # Local refinement for best solution
        result = optimize.minimize(lambda x: compute_overlap(x), best_solution, bounds=bounds, constraints=cons, method='SLSQP')
        if result.success:
            best_solution = result.x
            best_c5 = result.fun
    
    return (best_solution.tolist(), best_c5, n_points)
