import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds, LinearConstraint
from scipy.interpolate import interp1d

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return np.max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    max_iter = 100
    temp_start = 1.0
    temp_end = 0.1
    perturbation_scale = 0.3
    perturbation_decay = 0.995
    elite_fraction = 0.2  # Fraction of elite solutions to carry over
    
    # Initialize a population of candidate solutions
    population_size = 50
    population = [h_norm.copy() for _ in range(population_size)]
    best_solution = h_norm.copy()
    best_c5 = compute_overlap(best_solution)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Compute current temperature
        temp = temp_start * (temp_end / temp_start) ** (iteration / max_iter)
        
        # Evaluate all solutions in the population
        eval_results = []
        for sol in population:
            try:
                c5 = compute_overlap(sol)
                eval_results.append((c5, sol))
            except:
                eval_results.append((float('inf'), sol))
        
        # Sort by fitness
        eval_results.sort()
        current_best_c5, current_best_sol = eval_results[0]
        
        # Update best solution
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_solution = current_best_sol
        
        # Generate new solutions via mutation and crossover
        new_population = []
        for _ in range(len(population)):
            # Select two parents using tournament selection
            parent1, parent2 = random.choices(population, k=2)
            
            # Crossover
            crossover_point = random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            mutation_strength = perturbation_scale * (1 - iteration / max_iter)
            mutation = np.random.normal(0, mutation_strength, n_points)
            child += mutation
            child = np.clip(child, 0, 1)
            
            # Enforce integral constraint via projection
            child_sum = np.sum(child)
            child = child * (n_points / 2) / child_sum if child_sum != 0 else child
            
            new_population.append(child)
        
        # Replace population with new solutions
        population = new_population
        
        # Keep elite solutions
        elite_indices = np.argsort([compute_overlap(sol) for sol in population])[:int(elite_fraction * len(population))]
        population = [population[i] for i in elite_indices]
        
        # Occasionally perturb the best solution
        if iteration % 10 == 0:
            perturbation = np.random.uniform(-0.1, 0.1, n_points)
            best_solution += perturbation
            best_solution = np.clip(best_solution, 0, 1)
            best_solution = best_solution * (n_points / 2) / np.sum(best_solution)
    
    return (best_solution.tolist(), best_c5, n_points)
