import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Function to generate a random initial solution
    def generate_random_solution():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / sum(h)
        return h
    
    # Function to perform local refinement using SLSQP
    def local_refinement(candidate):
        bounds = [(0, 1) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': integral_constraint})
        result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
        if result.success:
            return result.x
        return candidate
    
    # Function to evolve the solution via genetic algorithm
    def evolve_solution(solution):
        # Create two random perturbations
        perturbation1 = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbation2 = np.random.uniform(-0.1, 0.1, size=n_points)
        candidate1 = np.clip(solution + perturbation1, 0, 1)
        candidate1 = candidate1 * (n_points / 2) / sum(candidate1)
        candidate2 = np.clip(solution + perturbation2, 0, 1)
        candidate2 = candidate2 * (n_points / 2) / sum(candidate2)
        
        # Refine both candidates
        refined1 = local_refinement(candidate1)
        refined2 = local_refinement(candidate2)
        
        # Select the better solution
        overlap1 = compute_overlap(refined1)
        overlap2 = compute_overlap(refined2)
        return refined1 if overlap1 < overlap2 else refined2
    
    # Main optimization loop
    best_solution = h_norm.copy()
    best_c5 = compute_overlap(best_solution)
    time_remaining = budget_s - (time.time() - start_time)
    
    # Perform a series of iterations to refine the solution
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        
        # Generate a new candidate solution
        candidate = generate_random_solution()
        refined_candidate = evolve_solution(candidate)
        
        # Update best solution if needed
        current_c5 = compute_overlap(refined_candidate)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_solution = refined_candidate
        
        # Decrease time remaining
        time_remaining = budget_s - (time.time() - start_time)
    
    return (best_solution.tolist(), best_c5, n_points)
