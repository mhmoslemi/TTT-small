import numpy as np
import scipy.optimize as optimize
import time
import random
from functools import lru_cache

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    num_particles = 50
    max_iterations = 200
    inertia = 0.7
    cognitive_coeff = 1.5
    social_coeff = 1.5
    personal_best = [h_norm.copy() for _ in range(num_particles)]
    personal_best_values = [compute_overlap(p) for p in personal_best]
    global_best = h_norm.copy()
    global_best_value = compute_overlap(global_best)
    
    # Initialize velocities
    velocities = np.random.uniform(-0.1, 0.1, (num_particles, n_points))
    
    # Particle Swarm Optimization
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        for i in range(num_particles):
            # Randomly select neighbors
            r1, r2 = np.random.rand(2)
            cognitive = cognitive_coeff * r1 * (personal_best[i] - h_norm)
            social = social_coeff * r2 * (global_best - h_norm)
            velocities[i] = inertia * velocities[i] + cognitive + social
            
            # Update position
            h_norm = np.clip(h_norm + velocities[i], 0, 1)
            h_norm = h_norm * (n_points / 2) / sum(h_norm)
            
            # Evaluate new position
            current_value = compute_overlap(h_norm)
            
            # Update personal best if improvement
            if current_value < personal_best_values[i]:
                personal_best[i] = h_norm.copy()
                personal_best_values[i] = current_value
                
                # Update global best if improvement
                if current_value < global_best_value:
                    global_best = h_norm.copy()
                    global_best_value = current_value
    
    return (global_best.tolist(), global_best_value, n_points)
