import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds, LinearConstraint

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Define a function to compute the overlap integral
    def calculate_c5(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return np.max(corr) * dx
    
    # Define a function to enforce the integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    # Optimization parameters
    max_iterations = 300
    perturbation_size = 0.3
    temp_start = 1.0
    temp_end = 0.1
    cooling_rate = 0.995
    
    # Initialize the best solution
    best_solution = h_norm.copy()
    best_c5 = calculate_c5(best_solution)
    
    # Initialize a swarm of candidate solutions
    swarm_size = 30
    swarm = [h_norm.copy() for _ in range(swarm_size)]
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Compute current temperature
        temp = temp_start * (temp_end / temp_start) ** (iteration / max_iterations)
        
        # Update perturbation size
        perturbation_size *= cooling_rate
        
        # Iterate over the swarm
        for i in range(swarm_size):
            # Perturb the current solution
            perturbation = np.random.uniform(-perturbation_size, perturbation_size, size=n_points)
            candidate = np.clip(swarm[i] + perturbation, 0, 1)
            
            # Define bounds and constraints
            bounds = Bounds([0.0] * n_points, [1.0] * n_points)
            cons = LinearConstraint([[1.0] * n_points], [n_points / 2], [n_points / 2])
            
            # Perform local optimization
            result = optimize.minimize(
                lambda x: calculate_c5(x),
                candidate,
                bounds=bounds,
                constraints=cons,
                method='SLSQP',
                tol=1e-6
            )
            
            # Accept the solution with a probability based on temperature
            if result.success:
                current_c5 = result.fun
                delta = current_c5 - best_c5
                if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                    swarm[i] = result.x
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_solution = result.x
        
        # Print progress
        if iteration % 50 == 0 and iteration > 0:
            print(f"Epoch {iteration}: Best C5 = {best_c5:.6f}")
    
    return (best_solution.tolist(), best_c5, n_points)
