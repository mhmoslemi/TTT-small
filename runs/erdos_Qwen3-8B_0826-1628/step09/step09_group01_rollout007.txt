import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    population_size = 50
    generations = 100
    mutation_rate = 0.1
    elite_size = 5
    crossover_rate = 0.8
    
    # Initialize population
    population = np.random.uniform(0, 1, (population_size, n_points))
    population = [h * (n_points / 2) / sum(h) for h in population]
    
    best_solution = population[0]
    best_c5 = compute_overlap(best_solution)
    
    # Main evolutionary loop
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [compute_overlap(h) for h in population]
        best_index = np.argmin(fitness)
        best_c5 = fitness[best_index]
        best_solution = population[best_index]
        
        # Select elites
        elite_indices = np.argsort(fitness)[:elite_size]
        elites = [population[i] for i in elite_indices]
        
        # Create new population
        new_population = elites.copy()
        
        # Generate offspring
        for _ in range(population_size - elite_size):
            # Select parents
            parent1 = random.choice(elites)
            parent2 = random.choice(elites)
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
            else:
                child = np.random.uniform(0, 1, n_points)
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutation = np.random.normal(0, 0.05, n_points)
                child += mutation
                child = np.clip(child, 0, 1)
            
            # Normalize to satisfy the integral constraint
            child = child * (n_points / 2) / sum(child)
            
            new_population.append(child)
        
        # Update population
        population = new_population
    
    return (best_solution.tolist(), best_c5, n_points)
