import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Function to enforce the integral constraint
    def integral_constraint_func(x):
        return sum(x) - (n_points / 2)
    
    # Bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': integral_constraint_func})
    
    # Optimization parameters
    n_swarm = 30
    max_iter = 150
    perturbation_decay = 0.95
    initial_perturbation = 0.2
    
    # Initialize a swarm of candidate solutions
    swarm = [h_norm.copy() for _ in range(n_swarm)]
    best_solution = h_norm.copy()
    best_c5 = compute_overlap(best_solution)
    best_indices = np.arange(n_points)
    
    def evolve_swarm(swarm, best_solution, best_c5, best_indices, initial_perturbation, perturbation_decay, max_iter, n_points, dx, compute_overlap, integral_constraint_func):
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            
            # Exploration phase with global perturbations and SLSQP
            for i in range(n_swarm):
                perturbation = np.random.uniform(-initial_perturbation, initial_perturbation, size=n_points)
                candidate = np.clip(swarm[i] + perturbation, 0, 1)
                # Local refinement with SLSQP
                result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
                
                if result.success:
                    swarm[i] = result.x
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_solution = result.x
                        best_indices = np.arange(n_points)
            
            # Exploitation phase with small perturbations
            for i in range(n_swarm):
                perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
                candidate = np.clip(swarm[i] + perturbation, 0, 1)
                # Ensure integral constraint is satisfied
                candidate = candidate * (n_points / 2) / sum(candidate)
                # Local refinement with SLSQP
                result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
                
                if result.success:
                    swarm[i] = result.x
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_solution = result.x
                        best_indices = np.arange(n_points)
            
            # Reduce perturbation range
            initial_perturbation *= perturbation_decay
        
        return best_solution, best_c5
    
    # Run the optimization
    best_solution, best_c5 = evolve_swarm(swarm, best_solution, best_c5, best_indices, initial_perturbation, perturbation_decay, max_iter, n_points, dx, compute_overlap, integral_constraint_func)
    
    return (best_solution.tolist(), best_c5, n_points)
