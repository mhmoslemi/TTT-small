import numpy as np
import scipy.optimize as optimize
import time
import random
import copy

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Function to optimize a single candidate solution
    def optimize_candidate(candidate):
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': integral_constraint}]
        result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
        if result.success:
            return result.x, result.fun
        else:
            return candidate, float('inf')
    
    # Function to generate a random initial solution
    def random_initial_solution():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Function to perform local search from a given solution
    def local_search(solution, steps=20, step_size=0.05):
        best_solution = solution.copy()
        best_c5 = compute_overlap(best_solution)
        for _ in range(steps):
            perturbation = np.random.uniform(-step_size, step_size, size=n_points)
            candidate = np.clip(solution + perturbation, 0, 1)
            candidate = candidate * (n_points / 2) / np.sum(candidate)
            candidate, c5 = optimize_candidate(candidate)
            if c5 < best_c5:
                best_solution = candidate
                best_c5 = c5
        return best_solution, best_c5
    
    # Function to perform a hill climbing search
    def hill_climbing(max_iterations=100, step_size=0.05):
        best_solution = random_initial_solution()
        best_c5 = compute_overlap(best_solution)
        for _ in range(max_iterations):
            if time.time() - start_time > budget_s:
                break
            perturbation = np.random.uniform(-step_size, step_size, size=n_points)
            candidate = np.clip(best_solution + perturbation, 0, 1)
            candidate = candidate * (n_points / 2) / np.sum(candidate)
            candidate, c5 = optimize_candidate(candidate)
            if c5 < best_c5:
                best_solution = candidate
                best_c5 = c5
        return best_solution, best_c5
    
    # Function to perform a simulated annealing search
    def simulated_annealing(max_iterations=200, initial_temp=1.0, cooling_rate=0.995):
        current_solution = random_initial_solution()
        current_c5 = compute_overlap(current_solution)
        best_solution = current_solution.copy()
        best_c5 = current_c5
        temperature = initial_temp
        for _ in range(max_iterations):
            if time.time() - start_time > budget_s:
                break
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            candidate = np.clip(current_solution + perturbation, 0, 1)
            candidate = candidate * (n_points / 2) / np.sum(candidate)
            candidate, c5 = optimize_candidate(candidate)
            delta = c5 - current_c5
            if delta < 0 or np.random.random() < np.exp(-delta / temperature):
                current_solution = candidate
                current_c5 = c5
                if current_c5 < best_c5:
                    best_solution = current_solution
                    best_c5 = current_c5
            temperature *= cooling_rate
        return best_solution, best_c5
    
    # Main optimization loop
    best_solution = random_initial_solution()
    best_c5 = compute_overlap(best_solution)
    
    # Perform hill climbing
    best_solution, best_c5 = hill_climbing(max_iterations=100, step_size=0.05)
    
    # Perform simulated annealing
    best_solution, best_c5 = simulated_annealing(max_iterations=200, initial_temp=1.0, cooling_rate=0.995)
    
    # Perform local search around the best solution
    best_solution, best_c5 = local_search(best_solution, steps=20, step_size=0.05)
    
    return (best_solution.tolist(), best_c5, n_points)
