import numpy as np
import scipy.optimize as optimize
import time
import random
import math

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = sum(h)
    h = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Use a combination of random search and gradient-based optimization
    # Start with a refined version of the initial guess
    def refine_initial_guess(h, n_points):
        h = h.copy()
        sum_h = sum(h)
        h = h * (n_points / 2) / sum_h
        bounds = [(0, 1) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': integral_constraint})
        result = optimize.minimize(lambda x: compute_overlap(x), h, bounds=bounds, constraints=cons, method='SLSQP')
        return result.x
    
    h = refine_initial_guess(h, n_points)
    best_solution = h.copy()
    best_c5 = compute_overlap(best_solution)
    
    # Optimization parameters
    n_particles = 30
    max_iter = 150
    refine_steps = 10
    refine_decay = 0.95
    perturb_range = 0.2
    perturb_decay = 0.98
    
    # Initialize a swarm of candidate solutions
    swarm = [h.copy() for _ in range(n_particles)]
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Evolution of the swarm
        for i in range(n_particles):
            # Apply random perturbation
            perturbation = np.random.uniform(-perturb_range, perturb_range, size=n_points)
            candidate = np.clip(swarm[i] + perturbation, 0, 1)
            
            # Enforce integral constraint
            candidate = candidate * (n_points / 2) / sum(candidate)
            
            # Gradient-based optimization
            bounds = [(0, 1) for _ in range(n_points)]
            cons = ({'type': 'eq', 'fun': integral_constraint})
            result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
            
            # Update swarm member if the optimization succeeded
            if result.success:
                swarm[i] = result.x
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_solution = result.x
        
        # Reduce perturbation range and refine
        perturb_range *= perturb_decay
        
        # Additional refinement through multi-step local optimization
        for i in range(n_particles):
            candidate = swarm[i].copy()
            for _ in range(refine_steps):
                perturbation = np.random.uniform(-0.01, 0.01, size=n_points)
                candidate = np.clip(candidate + perturbation, 0, 1)
                candidate = candidate * (n_points / 2) / sum(candidate)
                result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
                
                if result.success:
                    candidate = result.x
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_solution = candidate
            swarm[i] = candidate
    
    return (best_solution.tolist(), best_c5, n_points)
