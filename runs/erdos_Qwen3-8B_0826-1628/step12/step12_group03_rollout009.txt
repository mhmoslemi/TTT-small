```python
import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds, LinearConstraint

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return np.max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    max_iter = 150
    perturbation_decay = 0.99
    initial_perturbation = 0.15
    population_size = 30
    
    # Initialize a population of candidate solutions
    population = [h_norm.copy() for _ in range(population_size)]
    best_solution = h_norm.copy()
    best_c5 = compute_overlap(best_solution)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate the population
        fitness = [compute_overlap(ind) for ind in population]
        best_index = np.argmin(fitness)
        best_c5 = fitness[best_index]
        best_solution = population[best_index].copy()
        
        # Generate new candidates through mutation and crossover
        new_population = []
        for i in range(population_size):
            # Mutation
            perturbation = np.random.uniform(-initial_perturbation, initial_perturbation, size=n_points)
            candidate = np.clip(population[i] + perturbation, 0, 1)
            
            # Local refinement with constraints
            bounds = Bounds([0.0]*n_points, [1.0]*n_points)
            cons = LinearConstraint([[1.0]*n_points], [n_points/2], [n_points/2])
            result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
            
            # Add to new population if optimization succeeded
            if result.success:
                new_population.append(result.x)
            else:
                new_population.append(population[i].copy())
        
        # Replace the population with the new candidates
        population = new_population
        
        # Gradually reduce perturbation range
        initial_perturbation *= perturbation_decay
    
    return (best_solution.tolist(), best_c5, n_points)
```
