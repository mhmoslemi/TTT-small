```python
import numpy as np
import scipy.optimize as optimize
import time
import random
import copy

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the provided initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Define the integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    swarm_size = 30
    max_iter = 100
    perturbation_decay = 0.97
    initial_perturbation = 0.2
    refine_steps = 3
    refine_decay = 0.95
    batch_size = 5
    
    # Initialize a swarm of candidate solutions
    swarm = [h.copy() for _ in range(swarm_size)]
    best_solution = h.copy()
    best_c5 = compute_overlap(best_solution)
    
    # Function to evolve swarm with a balance of exploration and exploitation
    def evolve_swarm(swarm, best_solution, best_c5, initial_perturbation, perturbation_decay, max_iter, n_points, dx, compute_overlap, integral_constraint):
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            
            # Split the swarm into batches for parallel processing
            batches = [swarm[i:i + batch_size] for i in range(0, len(swarm), batch_size)]
            best_in_batch = [best_solution.copy() for _ in batches]
            best_c5_in_batch = [best_c5 for _ in batches]
            
            # Process each batch
            for batch_idx, batch in enumerate(batches):
                for i in range(len(batch)):
                    # Generate a random perturbation
                    perturbation = np.random.uniform(-initial_perturbation, initial_perturbation, size=n_points)
                    candidate = np.clip(batch[i] + perturbation, 0, 1)
                    # Ensure integral constraint
                    candidate = candidate * (n_points / 2) / np.sum(candidate)
                    
                    # Local refinement with SLSQP
                    bounds = [(0, 1) for _ in range(n_points)]
                    cons = [{'type': 'eq', 'fun': integral_constraint}]
                    result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
                    
                    # Update the batch member if the optimization succeeded
                    if result.success:
                        batch[i] = result.x
                        current_c5 = result.fun
                        if current_c5 < best_c5_in_batch[batch_idx]:
                            best_c5_in_batch[batch_idx] = current_c5
                            best_in_batch[batch_idx] = result.x
            
            # Update the swarm and best solution
            for i in range(len(swarm)):
                swarm[i] = best_in_batch[i % len(best_in_batch)]
                if best_c5_in_batch[i % len(best_c5_in_batch)] < best_c5:
                    best_c5 = best_c5_in_batch[i % len(best_c5_in_batch)]
                    best_solution = best_in_batch[i % len(best_in_batch)]
            
            # Reduce perturbation range and local refinement
            initial_perturbation *= perturbation_decay
            
            # Local refinement with small random perturbations
            for i in range(len(swarm)):
                perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
                candidate = np.clip(swarm[i] + perturbation, 0, 1)
                # Ensure integral constraint
                candidate = candidate * (n_points / 2) / np.sum(candidate)
                result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
                
                if result.success:
                    swarm[i] = result.x
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_solution = result.x
            
            # Additional refinement through multi-step local optimization
            for i in range(len(swarm)):
                candidate = swarm[i].copy()
                for _ in range(refine_steps):
                    perturbation = np.random.uniform(-0.01, 0.01, size=n_points)
                    candidate = np.clip(candidate + perturbation, 0, 1)
                    # Ensure integral constraint
                    candidate = candidate * (n_points / 2) / np.sum(candidate)
                    result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
                    
                    if result.success:
                        candidate = result.x
                        current_c5 = result.fun
                        if current_c5 < best_c5:
                            best_c5 = current_c5
                            best_solution = candidate
                swarm[i] = candidate
        
        return best_solution, best_c5
    
    # Run the optimization
    best_solution, best_c5 = evolve_swarm(swarm, best_solution, best_c5, initial_perturbation, perturbation_decay, max_iter, n_points, dx, compute_overlap, integral_constraint)
    
    return (best_solution.tolist(), best_c5, n_points)
```
