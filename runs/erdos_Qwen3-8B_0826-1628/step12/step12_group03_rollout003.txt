import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds, LinearConstraint

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return np.max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    max_iter = 200
    perturbation_decay = 0.98
    initial_perturbation = 0.15
    swarm_size = 20
    cooling_rate = 0.995
    temperature = 1.0
    
    # Initialize a swarm of candidate solutions
    swarm = [h_norm.copy() for _ in range(swarm_size)]
    best_solution = h_norm.copy()
    best_c5 = compute_overlap(best_solution)
    best_energy = best_c5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Generate new solutions with perturbations
        for i in range(len(swarm)):
            # Perturb the current solution
            perturbation = np.random.uniform(-initial_perturbation, initial_perturbation, size=n_points)
            candidate = np.clip(swarm[i] + perturbation, 0, 1)
            
            # Evaluate the candidate
            current_c5 = compute_overlap(candidate)
            
            # Metropolis-Hastings acceptance criterion
            if np.random.rand() < np.exp(-(current_c5 - best_energy) / temperature):
                swarm[i] = candidate
                current_energy = current_c5
                if current_energy < best_energy:
                    best_energy = current_energy
                    best_solution = candidate
        
        # Gradually reduce perturbation range and temperature
        initial_perturbation *= perturbation_decay
        temperature *= cooling_rate
    
    return (best_solution.tolist(), best_energy, n_points)
