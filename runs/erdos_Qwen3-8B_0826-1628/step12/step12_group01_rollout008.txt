import numpy as np
import scipy.optimize as optimize
import time
import random
import copy

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    h = h * (n_points / 2) / np.sum(h)
    dx = 2.0 / n_points
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Parameters for the algorithm
    swarm_size = 20
    max_iter = 150
    refine_steps = 5
    refine_decay = 0.9
    perturbation_scale = 0.15
    
    # Initialize swarm
    swarm = [h.copy() for _ in range(swarm_size)]
    best_solution = h.copy()
    best_c5 = compute_overlap(best_solution)
    
    # Function to refine a candidate solution with local optimization
    def refine_candidate(candidate, n_points, dx, compute_overlap, integral_constraint):
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': integral_constraint}]
        result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
        if result.success:
            return result.x
        else:
            return candidate
    
    # Function to evolve the swarm with adaptive perturbation and refinement
    def evolve_swarm(swarm, best_solution, best_c5, n_points, dx, compute_overlap, integral_constraint, perturbation_scale, refine_steps, refine_decay, max_iter):
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            
            # Global exploration with larger perturbations
            for i in range(swarm_size):
                perturbation = np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)
                candidate = np.clip(swarm[i] + perturbation, 0, 1)
                candidate = candidate * (n_points / 2) / np.sum(candidate)
                candidate = refine_candidate(candidate, n_points, dx, compute_overlap, integral_constraint)
                current_c5 = compute_overlap(candidate)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_solution = candidate
            
            # Reduce perturbation scale and do local refinement
            perturbation_scale *= 0.95
            
            # Refinement phase with local perturbations and SLSQP
            for i in range(swarm_size):
                candidate = swarm[i].copy()
                for _ in range(refine_steps):
                    perturbation = np.random.uniform(-0.02, 0.02, size=n_points)
                    candidate = np.clip(candidate + perturbation, 0, 1)
                    candidate = candidate * (n_points / 2) / np.sum(candidate)
                    candidate = refine_candidate(candidate, n_points, dx, compute_overlap, integral_constraint)
                current_c5 = compute_overlap(candidate)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_solution = candidate
                swarm[i] = candidate
        
        return best_solution, best_c5
    
    # Run the optimization
    best_solution, best_c5 = evolve_swarm(swarm, best_solution, best_c5, n_points, dx, compute_overlap, integral_constraint, perturbation_scale, refine_steps, refine_decay, max_iter)
    
    return (best_solution.tolist(), best_c5, n_points)
