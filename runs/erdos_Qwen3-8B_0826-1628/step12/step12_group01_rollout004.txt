import numpy as np
import scipy.optimize as optimize
import time
import random
import copy
from scipy.optimize import Bounds, LinearConstraint

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the provided initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Define the integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    max_iter = 100
    pop_size = 20
    refine_steps = 5
    refine_decay = 0.9
    mutation_rate = 0.1
    mutation_strength = 0.15
    crossover_rate = 0.7
    
    # Initialize a population
    population = [h.copy() for _ in range(pop_size)]
    best_solution = h.copy()
    best_c5 = compute_overlap(best_solution)
    
    # Function to evolve population
    def evolve_population(population, best_solution, best_c5, dx, compute_overlap, integral_constraint, n_points):
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            
            # Evaluate current population
            fitness = [compute_overlap(sol) for sol in population]
            best_idx = np.argmin(fitness)
            current_best = population[best_idx]
            current_best_c5 = fitness[best_idx]
            
            if current_best_c5 < best_c5:
                best_c5 = current_best_c5
                best_solution = current_best.copy()
            
            # Generate new population
            new_population = []
            for i in range(pop_size):
                # Crossover
                if np.random.rand() < crossover_rate:
                    parent1 = population[np.random.randint(pop_size)]
                    parent2 = population[np.random.randint(pop_size)]
                    child = np.copy(parent1)
                    for j in range(n_points):
                        if np.random.rand() < 0.5:
                            child[j] = parent2[j]
                else:
                    child = np.copy(current_best)
                
                # Mutation
                for j in range(n_points):
                    if np.random.rand() < mutation_rate:
                        delta = np.random.uniform(-mutation_strength, mutation_strength)
                        child[j] = np.clip(child[j] + delta, 0, 1)
                
                # Enforce integral constraint
                child = child * (n_points / 2) / np.sum(child)
                
                # Local optimization for refinement
                bounds = Bounds(0, 1)
                constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
                result = optimize.minimize(lambda x: compute_overlap(x), child, bounds=bounds, constraints=constraint, method='SLSQP')
                
                if result.success:
                    new_population.append(result.x)
                else:
                    new_population.append(child)
            
            population = new_population
        
        return best_solution, best_c5
    
    # Run the optimization
    best_solution, best_c5 = evolve_population(population, best_solution, best_c5, dx, compute_overlap, integral_constraint, n_points)
    
    return (best_solution.tolist(), best_c5, n_points)
