# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return np.max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    max_iter = 200
    perturbation_decay = 0.98
    initial_perturbation = 0.15
    population_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    elite_ratio = 0.2
    tournament_size = 5
    
    # Initialize a population of candidate solutions
    population = [h_norm.copy() for _ in range(population_size)]
    best_solution = h_norm.copy()
    best_c5 = compute_overlap(best_solution)
    
    # Precompute the values 1 - h_values for faster computation
    def compute_overlap_with_cache(h_values):
        one_minus_h = 1 - h_values
        corr = np.correlate(h_values, one_minus_h, mode="full")
        return np.max(corr) * dx
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate the fitness of the current population
        fitness = [compute_overlap_with_cache(individual) for individual in population]
        best_index = np.argmin(fitness)
        best_c5 = fitness[best_index]
        best_solution = population[best_index].copy()
        
        # Create a new population
        new_population = []
        for _ in range(population_size):
            # Select parents
            parents = []
            for _ in range(2):
                # Tournament selection
                tournament = np.random.choice(population, size=tournament_size)
                winner = tournament[np.argmin([compute_overlap_with_cache(ind) for ind in tournament])]
                parents.append(winner)
            
            # Crossover
            if np.random.random() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate((parents[0][:crossover_point], parents[1][crossover_point:]))
            else:
                child = parents[0].copy()
            
            # Mutation
            for i in range(n_points):
                if np.random.random() < mutation_rate:
                    child[i] += np.random.uniform(-0.1, 0.1)
                    child[i] = np.clip(child[i], 0.0, 1.0)
            
            # Enforce the integral constraint
            child_sum = np.sum(child)
            if child_sum != n_points / 2:
                scaling_factor = (n_points / 2) / child_sum
                child = child * scaling_factor
            
            new_population.append(child)
        
        # Replace the old population with the new one
        population = new_population
        
        # Keep the best solution from the previous generation
        if np.random.random() < elite_ratio:
            population[0] = best_solution.copy()
    
    return (best_solution.tolist(), best_c5, n_points)
