import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds, LinearConstraint
from scipy.optimize import minimize

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return np.max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    max_iter = 300
    perturbation_decay = 0.99
    initial_perturbation = 0.25
    
    # Initialize a swarm of candidate solutions
    swarm = [h_norm.copy() for _ in range(40)]
    best_solution = h_norm.copy()
    best_c5 = compute_overlap(best_solution)
    
    # Precompute the values 1 - h_values for faster computation
    def compute_overlap_with_cache(h_values):
        one_minus_h = 1 - h_values
        corr = np.correlate(h_values, one_minus_h, mode="full")
        return np.max(corr) * dx
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Shuffle the swarm for diversity
        random.shuffle(swarm)
        
        # Perform a global perturbation on each candidate in the swarm
        for i in range(len(swarm)):
            perturbation = np.random.uniform(-initial_perturbation, initial_perturbation, size=n_points)
            candidate = np.clip(swarm[i] + perturbation, 0, 1)
            bounds = Bounds([0.0]*n_points, [1.0]*n_points)
            cons = LinearConstraint([[1.0]*n_points], [n_points/2], [n_points/2])
            
            # Use a more robust optimization method
            result = minimize(compute_overlap_with_cache, candidate, bounds=bounds, constraints=cons, method='SLSQP', options={'maxiter': 50})
            
            # Update the swarm member if the optimization succeeded
            if result.success:
                swarm[i] = result.x
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_solution = result.x
        
        # Gradually reduce perturbation range
        initial_perturbation *= perturbation_decay
    
    return (best_solution.tolist(), best_c5, n_points)
