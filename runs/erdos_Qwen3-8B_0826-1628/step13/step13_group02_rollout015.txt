import numpy as np
import scipy.optimize as optimize
import time
import random
import math

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = sum(h)
    h = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Parameters
    n_particles = 10
    max_iter = 50
    refine_steps = 10
    perturb_range = 0.2
    perturb_decay = 0.98
    refine_decay = 0.98
    elite_ratio = 0.2
    population_size = 20
    
    # Initialize a swarm of candidate solutions
    swarm = [h.copy() for _ in range(population_size)]
    best_solution = h.copy()
    best_c5 = compute_overlap(best_solution)
    
    # Function to evolve swarm with a balance of exploration and exploitation
    def evolve_swarm(swarm, best_solution, best_c5, perturb_range, perturb_decay, refine_decay, max_iter, n_points, dx, compute_overlap, integral_constraint):
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            
            # Evaluate all particles
            evaluations = []
            for i in range(population_size):
                h_val = swarm[i].copy()
                result = optimize.minimize(lambda x: compute_overlap(x), h_val, bounds=[(0, 1) for _ in range(n_points)], constraints={'type': 'eq', 'fun': integral_constraint}, method='SLSQP')
                if result.success:
                    evaluations.append((result.fun, result.x))
                else:
                    evaluations.append((float('inf'), swarm[i]))
            
            # Sort evaluations by overlap
            evaluations.sort()
            # Select top elite solutions
            elite_size = int(elite_ratio * population_size)
            elite_solutions = evaluations[:elite_size]
            elite_solutions = [sol[1] for sol in elite_solutions]
            
            # Generate new solutions by perturbing elites
            new_solutions = []
            for sol in elite_solutions:
                # Apply random perturbation
                perturbation = np.random.uniform(-perturb_range, perturb_range, size=n_points)
                candidate = np.clip(sol + perturbation, 0, 1)
                # Enforce integral constraint
                candidate = candidate * (n_points / 2) / sum(candidate)
                new_solutions.append(candidate)
            
            # Replace worst solutions in swarm with new solutions
            for i in range(len(swarm)):
                if i < len(new_solutions):
                    swarm[i] = new_solutions[i]
                else:
                    # Randomly perturb other solutions
                    perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
                    candidate = np.clip(swarm[i] + perturbation, 0, 1)
                    candidate = candidate * (n_points / 2) / sum(candidate)
                    swarm[i] = candidate
            
            # Local refinement with SLSQP
            for i in range(len(swarm)):
                h_val = swarm[i].copy()
                result = optimize.minimize(lambda x: compute_overlap(x), h_val, bounds=[(0, 1) for _ in range(n_points)], constraints={'type': 'eq', 'fun': integral_constraint}, method='SLSQP')
                if result.success:
                    swarm[i] = result.x
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_solution = result.x
            
            # Reduce perturbation range
            perturb_range *= perturb_decay
            
            # Additional refinement through multi-step local optimization
            for i in range(len(swarm)):
                candidate = swarm[i].copy()
                for _ in range(refine_steps):
                    perturbation = np.random.uniform(-0.01, 0.01, size=n_points)
                    candidate = np.clip(candidate + perturbation, 0, 1)
                    candidate = candidate * (n_points / 2) / sum(candidate)
                    result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=[(0, 1) for _ in range(n_points)], constraints={'type': 'eq', 'fun': integral_constraint}, method='SLSQP')
                    
                    if result.success:
                        candidate = result.x
                        current_c5 = result.fun
                        if current_c5 < best_c5:
                            best_c5 = current_c5
                            best_solution = candidate
                swarm[i] = candidate
        
        return best_solution, best_c5
    
    # Run the optimization
    best_solution, best_c5 = evolve_swarm(swarm, best_solution, best_c5, perturb_range, perturb_decay, refine_decay, max_iter, n_points, dx, compute_overlap, integral_constraint)
    
    return (best_solution.tolist(), best_c5, n_points)
