```python
import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds, LinearConstraint

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return np.max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    max_iter = 100
    perturbation_decay = 0.98
    initial_perturbation = 0.15
    swarm_size = 20
    local_steps = 10
    local_decay = 0.95
    elite_fraction = 0.2
    mutation_rate = 0.1
    mutation_strength = 0.05
    crossover_rate = 0.8
    
    # Initialize a swarm of candidate solutions
    swarm = [h_norm.copy() for _ in range(swarm_size)]
    best_solution = h_norm.copy()
    best_c5 = compute_overlap(best_solution)
    elite_solutions = []
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate all candidates and collect elites
        evaluations = []
        for candidate in swarm:
            try:
                c5 = compute_overlap(candidate)
                evaluations.append((c5, candidate))
            except:
                evaluations.append((float('inf'), candidate))
        
        # Sort by C5 value
        evaluations.sort()
        # Select elites based on elite fraction
        elite_index = int(elite_fraction * len(evaluations))
        elite_solutions = evaluations[:elite_index]
        # Update best solution
        best_c5 = evaluations[0][0]
        best_solution = evaluations[0][1]
        
        # Generate new candidates via mutation and crossover
        new_swarm = []
        for i in range(len(swarm)):
            # Select two random elites
            j, k = np.random.choice(len(elite_solutions), 2, replace=False)
            parent1 = elite_solutions[j][1]
            parent2 = elite_solutions[k][1]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                child = np.copy(parent1)
                indices = np.random.choice(n_points, int(n_points / 2), replace=False)
                child[indices] = parent2[indices]
            else:
                child = np.copy(parent1)
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutation = np.random.uniform(-mutation_strength, mutation_strength, size=n_points)
                child += mutation
                child = np.clip(child, 0, 1)
            
            # Local refinement with constraints
            for _ in range(local_steps):
                bounds = Bounds([0.0]*n_points, [1.0]*n_points)
                cons = LinearConstraint([[1.0]*n_points], [n_points/2], [n_points/2])
                result = optimize.minimize(lambda x: compute_overlap(x), child, bounds=bounds, constraints=cons, method='SLSQP')
                
                # Update the candidate if the optimization succeeded
                if result.success:
                    child = result.x
                else:
                    break
            
            new_swarm.append(child)
        
        # Replace the swarm with new candidates
        swarm = new_swarm
        
        # Gradually reduce perturbation range
        initial_perturbation *= perturbation_decay
    
    return (best_solution.tolist(), best_c5, n_points)
```
