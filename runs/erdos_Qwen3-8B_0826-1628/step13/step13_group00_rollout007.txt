import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with a denser initial solution
    n_points = 64
    h = np.zeros(n_points)
    h[:n_points//2] = 1.0
    h[n_points//2:] = 0.0
    h = np.random.permutation(h)
    h = (h * (n_points / 2) / sum(h)).tolist()
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * (2.0 / n_points)
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    swarm_size = 50
    max_iter = 150
    perturbation_decay = 0.98
    initial_perturbation = 0.3
    refinement_steps = 5
    n_refinement = 3
    tolerance = 1e-6
    
    # Initialize a swarm of candidate solutions
    swarm = [np.random.uniform(0, 1, n_points) for _ in range(swarm_size)]
    best_solution = copy.copy(h)
    best_c5 = compute_overlap(best_solution)
    
    # Function to evolve swarm with a balance of exploration and exploitation
    def evolve_swarm(swarm, best_solution, best_c5, initial_perturbation, perturbation_decay, max_iter, n_points, dx, compute_overlap, integral_constraint):
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            
            # Global exploration phase with random perturbation
            for i in range(swarm_size):
                # Generate a candidate with random perturbation
                perturbation = np.random.uniform(-initial_perturbation, initial_perturbation, size=n_points)
                candidate = np.clip(swarm[i] + perturbation, 0, 1)
                # Ensure integral constraint is satisfied
                candidate = candidate * (n_points / 2) / sum(candidate)
                # Local refinement using SLSQP
                bounds = [(0, 1) for _ in range(n_points)]
                cons = ({'type': 'eq', 'fun': integral_constraint})
                try:
                    result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': tolerance, 'maxiter': 100})
                except:
                    result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='L-BFGS-B', options={'ftol': tolerance, 'maxiter': 100})
                
                if result.success:
                    swarm[i] = result.x
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_solution = result.x
            
            # Local refinement phase with smaller perturbations
            for i in range(swarm_size):
                for _ in range(n_refinement):
                    # Generate a candidate with small perturbation
                    perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
                    candidate = np.clip(swarm[i] + perturbation, 0, 1)
                    # Ensure integral constraint is satisfied
                    candidate = candidate * (n_points / 2) / sum(candidate)
                    # Local refinement using SLSQP
                    bounds = [(0, 1) for _ in range(n_points)]
                    cons = ({'type': 'eq', 'fun': integral_constraint})
                    try:
                        result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': tolerance, 'maxiter': 100})
                    except:
                        result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='L-BFGS-B', options={'ftol': tolerance, 'maxiter': 100})
                    
                    if result.success:
                        swarm[i] = result.x
                        current_c5 = result.fun
                        if current_c5 < best_c5:
                            best_c5 = current_c5
                            best_solution = result.x
            
            # Gradually reduce perturbation range
            initial_perturbation *= perturbation_decay
            
        return best_solution, best_c5
    
    # Run the optimization
    best_solution, best_c5 = evolve_swarm(swarm, best_solution, best_c5, initial_perturbation, perturbation_decay, max_iter, n_points, 2.0 / n_points, compute_overlap, integral_constraint)
    
    return (best_solution.tolist(), best_c5, n_points)
