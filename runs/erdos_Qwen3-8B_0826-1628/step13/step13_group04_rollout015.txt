```python
import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds, LinearConstraint

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return np.max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    max_iter = 200
    perturbation_decay = 0.98
    initial_perturbation = 0.15
    swarm_size = 30
    local_steps = 10
    local_decay = 0.95
    learning_rate = 0.05
    momentum = 0.9
    
    # Initialize a swarm of candidate solutions
    swarm = [h_norm.copy() for _ in range(swarm_size)]
    velocities = [np.zeros(n_points) for _ in range(swarm_size)]
    best_solution = h_norm.copy()
    best_c5 = compute_overlap(best_solution)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Perform a global perturbation on each candidate in the swarm
        for i in range(len(swarm)):
            # Update velocity and position
            velocities[i] = momentum * velocities[i] + np.random.uniform(-learning_rate, learning_rate, size=n_points)
            swarm[i] += velocities[i]
            swarm[i] = np.clip(swarm[i], 0, 1)
            
            # Local refinement with constraints
            for _ in range(local_steps):
                bounds = Bounds([0.0]*n_points, [1.0]*n_points)
                cons = LinearConstraint([[1.0]*n_points], [n_points/2], [n_points/2])
                result = optimize.minimize(lambda x: compute_overlap(x), swarm[i], bounds=bounds, constraints=cons, method='SLSQP')
                
                # Update the candidate if the optimization succeeded
                if result.success:
                    swarm[i] = result.x
                else:
                    break
            
            # Update the swarm member if the optimization succeeded
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_solution = swarm[i]
        
        # Gradually reduce perturbation range
        initial_perturbation *= perturbation_decay
    
    return (best_solution.tolist(), best_c5, n_points)
```
