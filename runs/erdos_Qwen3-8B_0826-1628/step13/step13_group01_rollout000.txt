import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = sum(h)
    h_norm = h * (n_points / 2) / sum_h

    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    swarm_size = 30
    max_iter = 100
    perturbation_decay = 0.95
    initial_perturbation = 0.2
    refinement_steps = 20
    n_refinement = 5
    n_points_list = [34, 68, 136]  # Try multiple resolutions

    best_solution = h_norm.copy()
    best_c5 = compute_overlap(best_solution)
    best_n_points = n_points

    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        dx = 2.0 / n_points
        # Initialize a swarm of candidate solutions
        swarm = [np.random.uniform(0, 1, size=n_points) for _ in range(swarm_size)]
        for i in range(len(swarm)):
            swarm[i] = swarm[i] * (n_points / 2) / sum(swarm[i])
        
        # Local refinement with SLSQP
        for i in range(len(swarm)):
            bounds = [(0, 1) for _ in range(n_points)]
            cons = ({'type': 'eq', 'fun': integral_constraint})
            try:
                result = optimize.minimize(lambda x: compute_overlap(x), swarm[i], bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-9, 'maxiter': 100})
            except:
                result = optimize.minimize(lambda x: compute_overlap(x), swarm[i], bounds=bounds, constraints=cons, method='L-BFGS-B', options={'ftol': 1e-9, 'maxiter': 100})
            if result.success:
                swarm[i] = result.x
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_solution = result.x
                    best_n_points = n_points
        
        # Refinement phase
        for _ in range(10):
            if time.time() - start_time > budget_s:
                break
            for i in range(len(swarm)):
                perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
                candidate = np.clip(swarm[i] + perturbation, 0, 1)
                candidate = candidate * (n_points / 2) / sum(candidate)
                bounds = [(0, 1) for _ in range(n_points)]
                cons = ({'type': 'eq', 'fun': integral_constraint})
                try:
                    result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-9, 'maxiter': 100})
                except:
                    result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='L-BFGS-B', options={'ftol': 1e-9, 'maxiter': 100})
                if result.success:
                    swarm[i] = result.x
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_solution = result.x
                        best_n_points = n_points
    
    return (best_solution.tolist(), best_c5, best_n_points)
