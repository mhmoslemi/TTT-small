import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution (n=34 samples)
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    # Normalize h to satisfy the integral constraint
    sum_h = sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    n_candidates = 20
    n_iterations = 100
    n_refinement = 30
    learning_rate = 0.1
    momentum = 0.9
    patience = 5
    min_improvement = 1e-6
    
    # Initialize candidate solutions
    candidates = [h_norm.copy() for _ in range(n_candidates)]
    best_solution = h_norm.copy()
    best_c5 = compute_overlap(best_solution)
    best_c5_history = [best_c5]
    best_c5_improvements = [0]
    
    # Function to perform gradient descent with constraints
    def optimize_with_constraints(h, n_points, dx, compute_overlap, integral_constraint):
        bounds = [(0, 1) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': integral_constraint})
        try:
            result = optimize.minimize(lambda x: compute_overlap(x), h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-9, 'maxiter': 100})
        except:
            result = optimize.minimize(lambda x: compute_overlap(x), h, bounds=bounds, constraints=cons, method='L-BFGS-B', options={'ftol': 1e-9, 'maxiter': 100})
        return result.x if result.success else h
    
    # Evolutionary refinement
    for iteration in range(n_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate all candidates
        fitness = [compute_overlap(c) for c in candidates]
        best_idx = np.argmin(fitness)
        current_c5 = fitness[best_idx]
        
        # Early stopping if no improvement
        if iteration > 0 and abs(current_c5 - best_c5_history[-1]) < min_improvement:
            if len(best_c5_improvements) > patience:
                break
        
        # Update best solution and history
        best_c5_history.append(current_c5)
        best_c5_improvements.append(current_c5 - best_c5_history[-2] if len(best_c5_history) > 2 else 0)
        
        # Generate new candidates
        new_candidates = []
        for c in candidates:
            # Perturb the solution
            perturbation = np.random.normal(0, learning_rate, size=n_points)
            perturbed = np.clip(c + perturbation, 0, 1)
            # Enforce integral constraint
            perturbed = perturbed * (n_points / 2) / sum(perturbed)
            # Refine with local optimization
            refined = optimize_with_constraints(perturbed, n_points, dx, compute_overlap, integral_constraint)
            new_candidates.append(refined)
        
        # Update the candidate pool
        candidates = new_candidates
        # Add best solution to candidates
        candidates.append(best_solution)
        
        # Replace worst candidates
        fitness = [compute_overlap(c) for c in candidates]
        indices = np.argsort(fitness)
        worst_idx = indices[-int(n_candidates / 2):]
        for idx in worst_idx:
            candidates[idx] = best_solution
    
    # Final evaluation
    best_solution = np.min(candidates, key=compute_overlap)
    best_c5 = compute_overlap(best_solution)
    
    # Final refinement with local optimization
    best_solution = optimize_with_constraints(best_solution, n_points, dx, compute_overlap, integral_constraint)
    best_c5 = compute_overlap(best_solution)
    
    return (best_solution.tolist(), best_c5, n_points)
