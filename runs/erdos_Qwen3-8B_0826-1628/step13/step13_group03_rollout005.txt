import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds, LinearConstraint
from scipy.optimize import minimize
from scipy.optimize import differential_evolution

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Define the number of points
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    required_sum = n_points / 2
    
    # Objective function to minimize the overlap
    def objective(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return np.max(corr) * dx
    
    # Constraint to ensure the integral constraint
    def constraint(h_values):
        return np.sum(h_values) - required_sum
    
    # Bounds for h values
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Use differential evolution for global optimization
    result = differential_evolution(
        objective,
        bounds,
        constraints=[{'type': 'eq', 'fun': constraint}],
        strategy='best1bin',
        maxiter=100,
        popsize=20,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        disp=False,
        polish=True,
        seed=seed
    )
    
    # Get the best solution and its C5 value
    best_h = result.x
    best_c5 = result.fun
    
    # Ensure the solution meets the integral constraint
    if abs(np.sum(best_h) - required_sum) > 1e-6:
        # Adjust the solution to meet the integral constraint
        adjustment = (required_sum - np.sum(best_h)) / n_points
        best_h = np.clip(best_h + adjustment, 0, 1)
    
    # Return the best solution, its C5 value, and the number of points
    return (best_h.tolist(), best_c5, n_points)
