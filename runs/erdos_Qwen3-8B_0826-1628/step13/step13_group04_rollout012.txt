import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds, LinearConstraint
from sklearn.cluster import KMeans

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return np.max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    max_iter = 100
    initial_perturbation = 0.25
    swarm_size = 20
    local_steps = 15
    local_decay = 0.95
    cluster_centers = 5
    
    # Initialize a swarm of candidate solutions
    swarm = [h_norm.copy() for _ in range(swarm_size)]
    best_solution = h_norm.copy()
    best_c5 = compute_overlap(best_solution)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Perform a global perturbation on each candidate in the swarm
        for i in range(len(swarm)):
            perturbation = np.random.uniform(-initial_perturbation, initial_perturbation, size=n_points)
            candidate = np.clip(swarm[i] + perturbation, 0, 1)
            
            # Local refinement with constraints
            for _ in range(local_steps):
                bounds = Bounds([0.0]*n_points, [1.0]*n_points)
                cons = LinearConstraint([[1.0]*n_points], [n_points/2], [n_points/2])
                result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
                
                # Update the candidate if the optimization succeeded
                if result.success:
                    candidate = result.x
                else:
                    break
            
            # Update the swarm member if the optimization succeeded
            if result.success:
                swarm[i] = candidate
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_solution = candidate
        
        # Cluster the swarm to guide perturbations
        kmeans = KMeans(n_clusters=cluster_centers, n_init=10)
        kmeans.fit(swarm)
        cluster_centers = kmeans.cluster_centers_
        
        # Gradually reduce perturbation range
        initial_perturbation *= perturbation_decay
    
    # Final local refinement on the best solution
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    cons = LinearConstraint([[1.0]*n_points], [n_points/2], [n_points/2])
    final_result = optimize.minimize(lambda x: compute_overlap(x), best_solution, bounds=bounds, constraints=cons, method='SLSQP')
    
    if final_result.success:
        best_solution = final_result.x
        best_c5 = final_result.fun
    
    return (best_solution.tolist(), best_c5, n_points)
